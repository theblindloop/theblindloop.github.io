"""Renderer and analytic arm for an area-weighted balance relation.

Each panel contains two separated solid disks on opposite sides of a fixed
pivot.  The question defines disk weight as proportional to visible area, so
the executable relation is

    red_radius**2 * left_distance == blue_radius**2 * right_distance.

The image exposes all of those quantities, but the relation requires binding
the size and lever arm of each disk before comparing the two products.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 820

BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
FRAME = (102, 116, 134)
INK = (34, 43, 56)
GUIDE = (218, 225, 233)
RED = (208, 60, 60)
BLUE = (49, 101, 194)

PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
PANEL_BOXES = (
    (35, 100, 685, 410),
    (715, 100, 1365, 410),
    (35, 440, 685, 750),
    (715, 440, 1365, 750),
)

PIVOT_X_OFFSET = 325
BEAM_Y_OFFSET = 215
DISK_Y_OFFSET = 167
DIMENSION_Y_OFFSET = 91
BEAM_LEFT_OFFSET = 68
BEAM_RIGHT_OFFSET = 605
MIN_DISTANCE = 108
MAX_DISTANCE = 202
MIN_RADIUS = 18
MAX_RADIUS = 31

BALANCE_TOLERANCE = 0.06
DECOY_CLEARANCE = 0.20
TARGET_HEADROOM = 0.02

_RED_RADIUS_KEY = "red_radius"
_BLUE_RADIUS_KEY = "blue_radius"
_LEFT_DISTANCE_KEY = "left_distance"
_RIGHT_DISTANCE_KEY = "right_distance"

# These pairs make exact integer area moments while giving the winner a range
# of relative sizes and lever arms.  (red radius, blue radius, left d, right d)
TARGET_TEMPLATES = (
    (20, 25, 200, 128),
    (25, 20, 128, 200),
    (21, 28, 192, 108),
    (28, 21, 108, 192),
    (24, 30, 200, 128),
    (30, 24, 128, 200),
    (18, 24, 192, 108),
    (24, 18, 108, 192),
    (27, 30, 200, 162),
    (30, 27, 162, 200),
)


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _moment(panel: dict, side: str) -> float:
    if side == "red":
        return float(panel[_RED_RADIUS_KEY] ** 2 * panel[_LEFT_DISTANCE_KEY])
    if side == "blue":
        return float(panel[_BLUE_RADIUS_KEY] ** 2 * panel[_RIGHT_DISTANCE_KEY])
    raise KeyError(side)


def _normalized_residual(panel: dict) -> float:
    left = _moment(panel, "red")
    right = _moment(panel, "blue")
    return (left - right) / max(left, right)


def _decoy(rng: random.Random, desired_sign: int) -> dict[str, int]:
    for _ in range(5000):
        panel = {
            "red_radius": rng.randint(MIN_RADIUS, MAX_RADIUS),
            "blue_radius": rng.randint(MIN_RADIUS, MAX_RADIUS),
            "left_distance": rng.randint(MIN_DISTANCE, MAX_DISTANCE),
            "right_distance": rng.randint(MIN_DISTANCE, MAX_DISTANCE),
        }
        residual = _normalized_residual(panel)
        if desired_sign * residual >= DECOY_CLEARANCE:
            return panel
    raise RuntimeError("could not sample a separated balance decoy")


def sample_scene(seed: int) -> dict:
    """Sample four panels with one exact area-weighted moment balance."""
    rng = random.Random(int(seed))
    target_index = rng.randrange(len(PANEL_LABELS))
    red_radius, blue_radius, left_distance, right_distance = rng.choice(TARGET_TEMPLATES)
    target = {
        "red_radius": red_radius,
        "blue_radius": blue_radius,
        "left_distance": left_distance,
        "right_distance": right_distance,
    }
    panels: list[dict[str, int]] = []
    decoy_rank = 0
    for index in range(len(PANEL_LABELS)):
        if index == target_index:
            panels.append(dict(target))
        else:
            desired_sign = -1 if decoy_rank % 2 == 0 else 1
            panels.append(_decoy(rng, desired_sign))
            decoy_rank += 1
    scene = {"panels": panels}
    if is_quarantined(scene):
        raise AssertionError("sampler produced a quarantined scene")
    return scene


def _matches(panels: Iterable[dict]) -> list[int]:
    return [
        index
        for index, panel in enumerate(panels)
        if abs(_normalized_residual(panel)) <= BALANCE_TOLERANCE
    ]


def render(scene: dict) -> Image.Image:
    """Render four framed balance beams with disk and dimension evidence."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27)
    legend_font = _font(18)
    panel_font = _font(29)
    small_font = _font(16)

    draw.text((40, 18), "Which beam balances?", fill=INK, font=title_font)
    draw.text(
        (40, 59),
        "red disk = left weight   blue disk = right weight   dark triangle = pivot",
        fill=FRAME,
        font=legend_font,
    )

    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 25, y0 + 20), PANEL_LABELS[index][-1], fill=INK, font=panel_font)

        pivot_x = x0 + PIVOT_X_OFFSET
        beam_y = y0 + BEAM_Y_OFFSET
        disk_y = y0 + DISK_Y_OFFSET
        dimension_y = y0 + DIMENSION_Y_OFFSET

        # Neutral construction marks make the two lever arms easy to inspect.
        beam_left = x0 + BEAM_LEFT_OFFSET
        beam_right = x0 + BEAM_RIGHT_OFFSET
        draw.line((beam_left, beam_y, beam_right, beam_y), fill=INK, width=7)
        draw.line((pivot_x, beam_y - 12, pivot_x, beam_y + 17), fill=INK, width=4)
        draw.polygon(
            ((pivot_x, beam_y + 18), (pivot_x - 22, beam_y + 47), (pivot_x + 22, beam_y + 47)),
            fill=FRAME,
        )
        draw.text((pivot_x - 28, beam_y + 53), "PIVOT", fill=FRAME, font=small_font)

        left_x = pivot_x - int(panel[_LEFT_DISTANCE_KEY])
        right_x = pivot_x + int(panel[_RIGHT_DISTANCE_KEY])
        for disk_x in (left_x, right_x):
            draw.line((disk_x, disk_y + 34, disk_x, beam_y - 4), fill=GUIDE, width=2)
            draw.line((disk_x, dimension_y, disk_x, dimension_y + 18), fill=FRAME, width=2)
        draw.line((left_x, dimension_y + 9, pivot_x, dimension_y + 9), fill=GUIDE, width=3)
        draw.line((pivot_x, dimension_y + 9, right_x, dimension_y + 9), fill=GUIDE, width=3)
        for endpoint in (left_x, pivot_x, right_x):
            draw.line((endpoint, dimension_y + 1, endpoint, dimension_y + 17), fill=FRAME, width=2)

        red_radius = int(panel[_RED_RADIUS_KEY])
        blue_radius = int(panel[_BLUE_RADIUS_KEY])
        draw.ellipse(
            (left_x - red_radius, disk_y - red_radius, left_x + red_radius, disk_y + red_radius),
            fill=RED,
        )
        draw.ellipse(
            (right_x - blue_radius, disk_y - blue_radius, right_x + blue_radius, disk_y + blue_radius),
            fill=BLUE,
        )

    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique panel with equal area-weighted lever arms."""
    matches = _matches(scene["panels"])
    if len(matches) != 1:
        raise ValueError(f"expected one balanced panel, found {matches}")
    return PANEL_LABELS[matches[0]]


def margin(scene: dict) -> float:
    """Return separation between the target band and its nearest decoy."""
    try:
        residuals = sorted(abs(_normalized_residual(panel)) for panel in scene["panels"])
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return -1.0
    if len(residuals) != len(PANEL_LABELS):
        return -1.0
    return min(BALANCE_TOLERANCE - residuals[0], residuals[1] - BALANCE_TOLERANCE)


def is_quarantined(scene: dict) -> bool:
    """Exclude ambiguous equality bands and insufficiently separated decoys."""
    try:
        residuals = sorted(abs(_normalized_residual(panel)) for panel in scene["panels"])
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True
    if len(residuals) != len(PANEL_LABELS):
        return True
    return not (
        residuals[0] <= TARGET_HEADROOM
        and residuals[1] >= DECOY_CLEARANCE
        and len(_matches(scene["panels"])) == 1
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; all label inputs are visibly rendered."""
    return []
