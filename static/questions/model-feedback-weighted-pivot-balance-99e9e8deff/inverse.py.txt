"""Independent image-only inverse for area-weighted pivot balance.

This module intentionally duplicates only the raster layout constants it needs.
It does not import the renderer, prompts, generator, verifier, or latent scene.
The two disk radii come from fitted component bounding boxes and the distances
come from recovered component centroids relative to the visible pivot.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 820
PANEL_BOXES = (
    (35, 100, 685, 410),
    (715, 100, 1365, 410),
    (35, 440, 685, 750),
    (715, 440, 1365, 750),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
RED = (208, 60, 60)
BLUE = (49, 101, 194)
PIVOT_X_OFFSET = 325
DISK_Y_OFFSET = 167
BALANCE_TOLERANCE = 0.06
MIN_DISK_PIXELS = 700


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected component coordinates for a boolean crop mask."""
    visited = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    result: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if visited[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int32))
    return result


def _disk_stats(arr: np.ndarray, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> tuple[float, float, float]:
    x0, y0, x1, y1 = box
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
    components = _components(mask)
    if len(components) != 1:
        raise ValueError("each weight color must form one connected component")
    points = components[0]
    if len(points) < MIN_DISK_PIXELS:
        raise ValueError("weight disk is missing or too small")
    ys = points[:, 0]
    xs = points[:, 1]
    xmin, xmax = int(xs.min()), int(xs.max())
    ymin, ymax = int(ys.min()), int(ys.max())
    width = xmax - xmin
    height = ymax - ymin
    if width < 34 or height < 34 or abs(width - height) > 2:
        raise ValueError("weight component is not a readable disk")
    radius = width / 2.0
    circle_area = np.pi * radius * radius
    if not 0.80 <= len(points) / circle_area <= 1.20:
        raise ValueError("weight component is not sufficiently disk-like")
    center_x = float(xs.mean()) + x0
    center_y = float(ys.mean()) + y0
    expected_y = y0 + DISK_Y_OFFSET
    if abs(center_y - expected_y) > 1.5:
        raise ValueError("weight disk is not on the beam axis")
    return center_x, radius, center_y


def _relative_residual(arr: np.ndarray, box: tuple[int, int, int, int]) -> float:
    x0, y0, _x1, _y1 = box
    pivot_x = float(x0 + PIVOT_X_OFFSET)
    red_x, red_radius, _ = _disk_stats(arr, box, RED)
    blue_x, blue_radius, _ = _disk_stats(arr, box, BLUE)
    left_distance = pivot_x - red_x
    right_distance = blue_x - pivot_x
    if left_distance < 50 or right_distance < 50:
        raise ValueError("both disks must be visibly on opposite sides of the pivot")
    left_moment = red_radius * red_radius * left_distance
    right_moment = blue_radius * blue_radius * right_distance
    return (left_moment - right_moment) / max(left_moment, right_moment)


def decision_from_image(image: Image.Image) -> str:
    """Measure both disk moments in each panel and select the unique balance."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb, dtype=np.uint8)
    residuals = [_relative_residual(arr, box) for box in PANEL_BOXES]
    matches = [index for index, residual in enumerate(residuals) if abs(residual) <= BALANCE_TOLERANCE]
    if len(matches) != 1:
        raise ValueError(f"expected one area-moment balance, found {matches}")
    return PANEL_LABELS[matches[0]]
