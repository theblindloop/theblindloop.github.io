"""Deterministic sampler and raster renderer for Corridor Width Palindrome."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 600
BACKGROUND = (248, 247, 242)
NAVY = (27, 48, 64)
HALO = (190, 207, 213)
BLUE = (39, 105, 201)
ORANGE = (232, 126, 38)
GOLD = (235, 175, 42)
BASE_WIDTH = 62
FULL_RADIUS = 64
GAUGE_OFFSET = 44
GAUGE_RADIUS = 6


def _d4(point: tuple[int, int], code: int) -> tuple[int, int]:
    x, y = point
    m = CANVAS_SIZE - 1
    return (
        (x, y), (m - x, y), (x, m - y), (m - x, m - y),
        (y, x), (m - y, x), (y, m - x), (m - y, m - x),
    )[code]


def _transform_vector(vector: tuple[int, int], code: int) -> tuple[int, int]:
    origin = _d4((300, 300), code)
    endpoint = _d4((300 + vector[0], 300 + vector[1]), code)
    dx, dy = endpoint[0] - origin[0], endpoint[1] - origin[1]
    if dx:
        return (1 if dx > 0 else -1, 0)
    return (0, 1 if dy > 0 else -1)


def _station(start: tuple[int, int], end: tuple[int, int], fraction: float) -> tuple[int, int]:
    return (
        int(round(start[0] + fraction * (end[0] - start[0]))),
        int(round(start[1] + fraction * (end[1] - start[1]))),
    )


def _paired_words(rng: random.Random) -> tuple[list[str], list[str]]:
    """Create equal-histogram palindrome/non-palindrome words."""
    n_full = rng.randint(2, 5)
    center_full = n_full % 2
    full_pairs = set(rng.sample(range(3), (n_full - center_full) // 2))
    palindrome = ["split"] * 7
    if center_full:
        palindrome[3] = "full"
    for index in full_pairs:
        palindrome[index] = palindrome[6 - index] = "full"

    for _ in range(100):
        full_positions = set(rng.sample(range(7), n_full))
        other = ["full" if index in full_positions else "split" for index in range(7)]
        if other != list(reversed(other)):
            return palindrome, other
    raise RuntimeError("could not construct a non-palindromic paired word")


def sample_scene(seed: int) -> dict:
    pair_seed = seed - seed % 2
    rng = random.Random((pair_seed * 1_000_003) ^ 0x6A09E667)
    left = 84 + rng.randint(-5, 5)
    right = 515 + rng.randint(-5, 5)
    top = 84 + rng.randint(-5, 5)
    middle = 300 + rng.randint(-7, 7)
    bottom = 515 + rng.randint(-5, 5)
    base_route = [(left, top), (right, top), (right, middle), (left, middle), (left, bottom), (right, bottom)]
    specs = ((0, .28), (0, .70), (1, .48), (2, .28), (2, .70), (3, .48), (4, .50))
    base_stations = [_station(base_route[i], base_route[i + 1], fraction) for i, fraction in specs]
    base_tangents = []
    for segment, _ in specs:
        a, b = base_route[segment], base_route[segment + 1]
        base_tangents.append((1, 0) if a[1] == b[1] else (0, 1))
    transform = rng.randrange(8)
    route = [_d4(point, transform) for point in base_route]
    stations = [_d4(point, transform) for point in base_stations]
    tangents = [_transform_vector(vector, transform) for vector in base_tangents]
    yes_word, no_word = _paired_words(rng)
    modes = yes_word if seed % 2 == 0 else no_word
    return {
        "route_points": [list(point) for point in route],
        "start_center": list(route[0]),
        "finish_center": list(route[-1]),
        "station_centers": [list(point) for point in stations],
        "station_modes": modes,
        "station_tangents": [list(vector) for vector in tangents],
    }


def _filled_circle(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, fill: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def _stroke_polyline(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], fill: tuple[int, int, int], width: int) -> None:
    radius = width // 2
    for a, b in zip(points, points[1:]):
        # Canonical endpoint order removes Pillow's even-width direction alias.
        first, second = sorted((a, b))
        draw.line((first, second), fill=fill, width=width)
    for point in points:
        _filled_circle(draw, point, radius, fill)


def _role_ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - 19, y - 19, x + 19, y + 19), fill=BACKGROUND, outline=color, width=7)


def render(scene: dict) -> Image.Image:
    points = [tuple(map(int, value)) for value in scene["route_points"]]
    start = tuple(map(int, scene["start_center"]))
    finish = tuple(map(int, scene["finish_center"]))
    stations = [tuple(map(int, value)) for value in scene["station_centers"]]
    modes = list(scene["station_modes"])
    tangents = [tuple(map(int, value)) for value in scene["station_tangents"]]
    if len(stations) != 7 or len(modes) != 7 or len(tangents) != 7:
        raise ValueError("exactly seven stations are required")
    if any(mode not in {"full", "split"} for mode in modes):
        raise ValueError("station modes must be full or split")

    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((17, 17, 582, 582), radius=22, outline=(221, 222, 215), width=2)
    _stroke_polyline(draw, points, HALO, BASE_WIDTH + 18)
    for center, mode in zip(stations, modes, strict=True):
        if mode == "full":
            _filled_circle(draw, center, FULL_RADIUS + 9, HALO)
    _stroke_polyline(draw, points, NAVY, BASE_WIDTH)
    for center, mode in zip(stations, modes, strict=True):
        if mode == "full":
            _filled_circle(draw, center, FULL_RADIUS, NAVY)

    for center, tangent in zip(stations, tangents, strict=True):
        nx, ny = -tangent[1], tangent[0]
        for sign in (-1, 1):
            dot = (center[0] + sign * GAUGE_OFFSET * nx, center[1] + sign * GAUGE_OFFSET * ny)
            _filled_circle(draw, dot, GAUGE_RADIUS, GOLD)

    _role_ring(draw, start, BLUE)
    _role_ring(draw, finish, ORANGE)
    return image


def _mismatched_pairs(scene: dict) -> int:
    modes = list(scene["station_modes"])
    if len(modes) != 7 or any(mode not in {"full", "split"} for mode in modes):
        raise ValueError("station_modes must contain seven visible states")
    return sum(modes[index] != modes[6 - index] for index in range(3))


def analytic_gold(scene: dict) -> str:
    return "yes" if _mismatched_pairs(scene) == 0 else "no"


def margin(scene: dict) -> float:
    mismatches = _mismatched_pairs(scene)
    return float(max(1, mismatches))


def is_quarantined(scene: dict) -> bool:
    modes = list(scene["station_modes"])
    return len(modes) != 7 or any(mode not in {"full", "split"} for mode in modes)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["route_points"] = list(reversed(twin["route_points"]))
    return [("reverse_undirected_polyline_storage", twin)]
