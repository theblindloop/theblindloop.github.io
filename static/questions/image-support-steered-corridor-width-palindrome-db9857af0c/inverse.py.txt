"""Independent pixel-only inverse for Corridor Width Palindrome."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_NAVY = np.array((27, 48, 64), dtype=np.int32)
_HALO = np.array((190, 207, 213), dtype=np.int32)
_BLUE = np.array((39, 105, 201), dtype=np.int32)
_ORANGE = np.array((232, 126, 38), dtype=np.int32)
_GOLD = np.array((235, 175, 42), dtype=np.int32)
_TOL_SQ = 24 * 24
_DOWNSAMPLE = 4


class PixelOracleError(ValueError):
    """Raised when the final raster does not support one stable answer."""


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= _TOL_SQ


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        sy, sx = int(raw_y), int(raw_x)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        stack = [(sy, sx)]
        points: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centers(mask: np.ndarray, expected: int, label: str, area: tuple[int, int]) -> list[tuple[float, float]]:
    components = _components(mask, minimum=20)
    if len(components) != expected:
        raise PixelOracleError(f"expected {expected} {label} components, found {len(components)}")
    low, high = area
    centers = []
    for component in components:
        if not low <= len(component) <= high:
            raise PixelOracleError(f"{label} component area {len(component)} outside [{low}, {high}]")
        y, x = component.mean(axis=0)
        centers.append((float(y), float(x)))
    return centers


def _coarse_floor(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    if height % _DOWNSAMPLE or width % _DOWNSAMPLE:
        raise PixelOracleError("image dimensions are incompatible with the public floor scale")
    blocks = mask.reshape(height // _DOWNSAMPLE, _DOWNSAMPLE, width // _DOWNSAMPLE, _DOWNSAMPLE)
    return blocks.mean(axis=(1, 3)) >= 0.25


def _boundary_breaches(navy: np.ndarray, allowed: np.ndarray) -> int:
    """Count floor edges lacking the renderer's visible pale safety border."""
    breaches = 0
    for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        neighbor = np.roll(allowed, (dy, dx), axis=(0, 1))
        if dy == -1:
            neighbor[-1, :] = False
        elif dy == 1:
            neighbor[0, :] = False
        elif dx == -1:
            neighbor[:, -1] = False
        else:
            neighbor[:, 0] = False
        breaches += int(np.count_nonzero(navy & ~neighbor))
    return breaches


def _nearest(points: np.ndarray, center: tuple[float, float]) -> tuple[int, int, float]:
    delta = points.astype(np.float64) - np.asarray(center, dtype=np.float64)
    squared = np.sum(delta * delta, axis=1)
    index = int(np.argmin(squared))
    y, x = points[index]
    return int(y), int(x), float(squared[index])


def _geodesic(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    distances = np.full(mask.shape, -1, dtype=np.int32)
    distances[start] = 0
    queue: deque[tuple[int, int]] = deque([start])
    while queue:
        y, x = queue.popleft()
        value = int(distances[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] and mask[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = value
                queue.append((ny, nx))
    return distances


def _pair_dots(centers: list[tuple[float, float]]) -> list[tuple[tuple[float, float], tuple[float, float]]]:
    remaining = set(range(len(centers)))
    pairs = []
    while remaining:
        best = None
        for i in remaining:
            for j in remaining:
                if i >= j:
                    continue
                distance = math.dist(centers[i], centers[j])
                if best is None or distance < best[0]:
                    best = (distance, i, j)
        if best is None or not 78.0 <= best[0] <= 98.0:
            raise PixelOracleError("gold dots do not form seven separated fixed-span pairs")
        _, i, j = best
        remaining.remove(i)
        remaining.remove(j)
        pairs.append((centers[i], centers[j]))
    return pairs


def _support_fraction(navy: np.ndarray, center: tuple[float, float]) -> float:
    cy, cx = center
    y0, y1 = max(0, int(cy) - 13), min(navy.shape[0], int(cy) + 14)
    x0, x1 = max(0, int(cx) - 13), min(navy.shape[1], int(cx) + 14)
    yy, xx = np.ogrid[y0:y1, x0:x1]
    radius = np.hypot(yy - cy, xx - cx)
    annulus = (radius >= 8.0) & (radius <= 12.0)
    if annulus.sum() < 80:
        raise PixelOracleError("gauge support annulus is clipped")
    return float(navy[y0:y1, x0:x1][annulus].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (600, 600, 3):
        raise PixelOracleError(f"expected a 600 by 600 RGB raster, got {array.shape}")
    navy = _mask(array, _NAVY)
    halo = _mask(array, _HALO)
    blue_mask = _mask(array, _BLUE)
    orange_mask = _mask(array, _ORANGE)
    gold_mask = _mask(array, _GOLD)
    if not 90000 <= int(navy.sum()) <= 190000:
        raise PixelOracleError("navy floor area is outside the supported broad-corridor range")
    breaches = _boundary_breaches(navy, navy | halo | blue_mask | orange_mask | gold_mask)
    if breaches > 4:
        raise PixelOracleError(f"navy floor has {breaches} pixels without its visible pale border")
    coarse = _coarse_floor(navy)
    components = sorted(_components(coarse, minimum=12), key=len, reverse=True)
    if len(components) != 1:
        raise PixelOracleError(f"navy floor is not one connected coarse component ({len(components)})")
    floor_points = components[0]
    y_span = int(floor_points[:, 0].max() - floor_points[:, 0].min())
    x_span = int(floor_points[:, 1].max() - floor_points[:, 1].min())
    if y_span < 0.70 * coarse.shape[0] or x_span < 0.70 * coarse.shape[1]:
        raise PixelOracleError("navy floor does not span both image axes")

    blue = _centers(blue_mask, 1, "blue ring", (300, 900))[0]
    orange = _centers(orange_mask, 1, "orange ring", (300, 900))[0]
    dots = _centers(gold_mask, 14, "gold dot", (75, 170))
    pairs = _pair_dots(dots)

    coarse_start = (blue[0] / _DOWNSAMPLE, blue[1] / _DOWNSAMPLE)
    sy, sx, start_gap = _nearest(floor_points, coarse_start)
    if start_gap > 45.0:
        raise PixelOracleError("blue ring is detached from the floor")
    distances = _geodesic(coarse, (sy, sx))
    fy, fx, finish_gap = _nearest(floor_points, (orange[0] / _DOWNSAMPLE, orange[1] / _DOWNSAMPLE))
    if finish_gap > 45.0 or distances[fy, fx] < 300:
        raise PixelOracleError("orange ring is detached or the traced floor is too short")

    ordered = []
    for first, second in pairs:
        fractions = (_support_fraction(navy, first), _support_fraction(navy, second))
        if min(fractions) >= 0.78:
            mode = "full"
        elif max(fractions) <= 0.12:
            mode = "split"
        else:
            raise PixelOracleError(f"gauge support is ambiguous: {fractions}")
        midpoint = ((first[0] + second[0]) / (2 * _DOWNSAMPLE), (first[1] + second[1]) / (2 * _DOWNSAMPLE))
        y, x, gap = _nearest(floor_points, midpoint)
        along = int(distances[y, x])
        if gap > 36.0 or along < 0:
            raise PixelOracleError("gauge pair is detached from the traced floor")
        ordered.append((along, mode))
    ordered.sort()
    ranks = [distance for distance, _ in ordered]
    if any(b - a < 24 for a, b in zip(ranks, ranks[1:])):
        raise PixelOracleError("gauge order has inadequate pixel separation")
    word = [mode for _, mode in ordered]
    return "yes" if word == list(reversed(word)) else "no"
