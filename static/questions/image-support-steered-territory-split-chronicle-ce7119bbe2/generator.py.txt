from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
FRAME = (31, 42, 58)
PANEL = (245, 246, 240)
COLORS = {
    "cyan": (31, 169, 190),
    "coral": (232, 91, 86),
    "violet": (116, 86, 179),
}
SLOTS = [(148, 148), (268, 148), (208, 268)]
PANEL_SIZE = 96


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(COLORS)
    first = names[int(seed) % 3]
    others = [name for name in names if name != first]
    rng.shuffle(others)
    baseline, second = others
    split_steps = {baseline: 0, first: 1, second: 2}
    panel_slots = [0, 1, 2]
    rng.shuffle(panel_slots)
    lane_orders = []
    for _ in range(3):
        order = names[:]
        rng.shuffle(order)
        lane_orders.append(order)
    return {
        "seed": int(seed),
        "split_steps": split_steps,
        "panel_slots": panel_slots,
        "lane_orders": lane_orders,
        "gap_px": [5, 8, 11][(int(seed) // 3) % 3],
    }


def _draw_panel(draw: ImageDraw.ImageDraw, scene: dict, stage: int) -> None:
    slot = int(scene["panel_slots"][stage])
    x, y = SLOTS[slot]
    s = PANEL_SIZE
    draw.rectangle((x, y, x + s - 1, y + s - 1), fill=FRAME)
    draw.rectangle((x + 3, y + 20, x + s - 4, y + s - 4), fill=PANEL)

    dot_count = stage + 1
    spacing = 13
    start = x + s // 2 - spacing * (dot_count - 1) // 2
    for i in range(dot_count):
        cx = start + i * spacing
        draw.ellipse((cx - 4, y + 7, cx + 4, y + 15), fill=BACKGROUND)

    for lane, name in enumerate(scene["lane_orders"][stage]):
        cy = y + 34 + lane * 19
        left, right = x + 16, x + 79
        top, bottom = cy, cy + 12
        if int(scene["split_steps"][name]) <= stage:
            gap = int(scene["gap_px"])
            mid = (left + right) // 2
            draw.rounded_rectangle((left, top, mid - gap // 2 - 1, bottom), radius=4, fill=COLORS[name])
            draw.rounded_rectangle((mid + (gap + 1) // 2, top, right, bottom), radius=4, fill=COLORS[name])
        else:
            draw.rounded_rectangle((left, top, right, bottom), radius=4, fill=COLORS[name])


def render(scene: dict) -> Image.Image:
    # Read every declared latent field: seed selects a harmless visible corner pip.
    seed = int(scene["seed"])
    _ = scene["split_steps"]  # direct access keeps the rendered-field audit explicit
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((132, 132, 380, 380), radius=28, fill=(226, 231, 229))
    for stage in range(3):
        _draw_panel(draw, scene, stage)
    # A tiny neutral pip exposes seed parity without carrying the answer.
    px = 252 + (seed % 2) * 8
    draw.ellipse((px - 2, 386, px + 2, 390), fill=FRAME)
    return image


def analytic_gold(scene: dict) -> str:
    matches = [name for name, step in scene["split_steps"].items() if int(step) == 1]
    if len(matches) != 1:
        raise ValueError("scene must have exactly one first post-baseline split")
    return matches[0]


def margin(scene: dict) -> float:
    return float(scene["gap_px"])


def is_quarantined(scene: dict) -> bool:
    return int(scene["gap_px"]) <= 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
