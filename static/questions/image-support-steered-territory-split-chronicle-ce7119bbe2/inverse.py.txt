from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PANEL_RGB = np.array((245, 246, 240), dtype=np.int16)
FRAME_RGB = np.array((31, 42, 58), dtype=np.int16)
PALETTE = {
    "cyan": np.array((31, 169, 190), dtype=np.int16),
    "coral": np.array((232, 91, 86), dtype=np.int16),
    "violet": np.array((116, 86, 179), dtype=np.int16),
}


def _mask_near(arr: np.ndarray, rgb: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - rgb), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((len(xs), min(xs), min(ys), max(xs), max(ys)))
    return output


def _panel_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    comps = _components(_mask_near(arr, PANEL_RGB))
    boxes = []
    for area, x0, y0, x1, y1 in comps:
        if area > 3200 and 80 <= x1 - x0 <= 94 and 65 <= y1 - y0 <= 78:
            boxes.append((x0 - 3, y0 - 20, x1 + 4, y1 + 4))
    if len(boxes) != 3:
        raise ValueError(f"expected three intact panel interiors, found {len(boxes)}")
    return boxes


def _header_number(arr: np.ndarray, box: tuple[int, int, int, int]) -> int:
    x0, y0, x1, _ = box
    crop = arr[y0 + 4 : y0 + 18, x0 + 4 : x1 - 3]
    white = np.min(crop.astype(np.int16), axis=2) >= 245
    dots = [c for c in _components(white) if 35 <= c[0] <= 100]
    if len(dots) not in (1, 2, 3):
        raise ValueError("panel header dots are missing or ambiguous")
    return len(dots)


def _territory_counts(arr: np.ndarray, box: tuple[int, int, int, int]) -> dict[str, int]:
    x0, y0, x1, y1 = box
    crop = arr[y0 + 22 : y1 - 2, x0 + 5 : x1 - 4]
    counts = {}
    for name, rgb in PALETTE.items():
        pieces = [c for c in _components(_mask_near(crop, rgb)) if c[0] >= 80]
        if len(pieces) not in (1, 2):
            raise ValueError(f"{name} territory has {len(pieces)} valid pieces")
        counts[name] = len(pieces)
    return counts


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 200:
        raise ValueError("image is too small")
    stages = {}
    for box in _panel_boxes(arr):
        number = _header_number(arr, box)
        if number in stages:
            raise ValueError("duplicate panel number")
        stages[number] = _territory_counts(arr, box)
    if set(stages) != {1, 2, 3}:
        raise ValueError("panel order is incomplete")
    split_sets = [{name for name, count in stages[k].items() if count == 2} for k in (1, 2, 3)]
    if [len(item) for item in split_sets] != [1, 2, 3]:
        raise ValueError("territory chronology is not uniquely replayable")
    if not (split_sets[0] < split_sets[1] < split_sets[2]):
        raise ValueError("a territory rejoined or multiple transitions occurred")
    new_first = split_sets[1] - split_sets[0]
    if len(new_first) != 1:
        raise ValueError("first post-baseline split is ambiguous")
    return next(iter(new_first))
