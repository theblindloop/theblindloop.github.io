"""Renderer for a texture-defined hidden silhouette."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
PALETTES = ((38, 72, 95), (71, 55, 96), (33, 79, 67), (91, 54, 48))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    rows = rng.choice((17, 19))
    cols = rows
    cell = 24 if rows == 17 else 22
    span = (rows - 1) * cell
    base = (CANVAS - span) / 2
    shape = "round" if seed % 2 == 0 else "diamond"
    return {
        "shape": shape,
        "rows": rows,
        "cols": cols,
        "cell": cell,
        "origin_x": round(base + rng.uniform(-8, 8), 3),
        "origin_y": round(base + rng.uniform(-8, 8), 3),
        "center_x": rows // 2 + rng.choice((-1, 0, 1)),
        "center_y": rows // 2 + rng.choice((-1, 0, 1)),
        "target_cells": rng.choice((57, 61, 65)),
        "angle": rng.choice((31, 35, 39, 43)),
        "target_slash": bool(rng.randrange(2)),
        "palette_index": rng.randrange(len(PALETTES)),
        "jitter_seed": rng.randrange(1, 2**31),
    }


def _target_mask(scene: dict) -> set[tuple[int, int]]:
    shape = scene["shape"]
    rows, cols = int(scene["rows"]), int(scene["cols"])
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    count = int(scene["target_cells"])
    perturb = random.Random(int(scene["jitter_seed"]) ^ 0x51A71E)
    ranked = []
    for y in range(rows):
        for x in range(cols):
            dx, dy = x - cx, y - cy
            base = dx * dx + dy * dy if shape == "round" else (abs(dx) + abs(dy)) ** 2
            ranked.append((base + perturb.uniform(-0.08, 0.08), y, x))
    ranked.sort()
    return {(y, x) for _, y, x in ranked[:count]}


def render(scene: dict) -> Image.Image:
    rows, cols = int(scene["rows"]), int(scene["cols"])
    cell = float(scene["cell"])
    ox, oy = float(scene["origin_x"]), float(scene["origin_y"])
    angle = math.radians(float(scene["angle"]))
    target_slash = bool(scene["target_slash"])
    ink = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    target = _target_mask(scene)
    jitter = random.Random(int(scene["jitter_seed"]))

    scale = 3
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), (247, 243, 234))
    draw = ImageDraw.Draw(image)
    half = cell * 0.34
    for y in range(rows):
        for x in range(cols):
            cx = ox + x * cell + jitter.uniform(-1.5, 1.5)
            cy = oy + y * cell + jitter.uniform(-1.5, 1.5)
            belongs = (y, x) in target
            slash = target_slash if belongs else not target_slash
            signed = -angle if slash else angle
            dx, dy = half * math.cos(signed), half * math.sin(signed)
            draw.line(
                ((cx - dx) * scale, (cy - dy) * scale, (cx + dx) * scale, (cy + dy) * scale),
                fill=ink,
                width=4 * scale,
            )
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["shape"])


def _ideal_fill(scene: dict) -> float:
    points = _target_mask(scene)
    xs = [x for _, x in points]
    ys = [y for y, _ in points]
    return len(points) / ((max(xs) - min(xs) + 1) * (max(ys) - min(ys) + 1))


def margin(scene: dict) -> float:
    return abs(_ideal_fill(scene) - 0.67)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.025


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
