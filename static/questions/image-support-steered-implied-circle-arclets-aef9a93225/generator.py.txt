from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

SIZE = 384
TOLERANCE = 4.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    aligned = (seed % 2) == 0
    shift = rng.choice([-2, -1, 0, 1, 2]) if aligned else rng.choice([-22, -20, -18, 18, 20, 22])
    return {
        "radial_shift": shift,
        "cx": rng.randint(120, 264),
        "cy": rng.randint(120, 264),
        "radius": rng.randint(38, 48),
        "rotation": rng.randint(0, 359),
        "deviant": rng.randrange(3),
        "half_span": rng.randint(27, 31),
        "width": 6,
        "color": rng.choice([[35, 92, 181], [24, 118, 151], [74, 72, 172]]),
    }


def _points(cx: float, cy: float, radius: float, start: float, stop: float) -> list[tuple[float, float]]:
    values = []
    for i in range(31):
        angle = math.radians(start + (stop - start) * i / 30.0)
        values.append((cx + radius * math.cos(angle), cy + radius * math.sin(angle)))
    return values


def render(scene: dict) -> Image.Image:
    shift = float(scene["radial_shift"])
    image = Image.new("RGB", (SIZE, SIZE), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    for index in range(3):
        angle = float(scene["rotation"]) + index * 120.0
        radius = float(scene["radius"]) + (shift if index == int(scene["deviant"]) else 0.0)
        pts = _points(float(scene["cx"]), float(scene["cy"]), radius, angle - float(scene["half_span"]), angle + float(scene["half_span"]))
        draw.line(pts, fill=tuple(scene["color"]), width=int(scene["width"]), joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if abs(float(scene["radial_shift"])) <= TOLERANCE else "no"


def margin(scene: dict) -> float:
    return abs(abs(float(scene["radial_shift"])) - TOLERANCE)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
