from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    parts = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            py, px = stack.pop()
            pixels.append((py, px))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = py + dy, px + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= 20:
            parts.append(np.asarray(pixels, dtype=float))
    return parts


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    background = np.array([250, 249, 246], dtype=np.int16)
    mask = np.max(np.abs(rgb - background), axis=2) > 35
    parts = _components(mask)
    if len(parts) != 3:
        return "abstain"
    points = np.concatenate([part[:, ::-1] for part in parts], axis=0)
    x, y = points[:, 0], points[:, 1]
    design = np.column_stack((x, y, np.ones_like(x)))
    rhs = -(x * x + y * y)
    a, b, c = np.linalg.lstsq(design, rhs, rcond=None)[0]
    cx, cy = -a / 2.0, -b / 2.0
    radial = np.sqrt((x - cx) ** 2 + (y - cy) ** 2)
    residual = float(np.std(radial))
    return "yes" if residual <= 2.05 else "no"
