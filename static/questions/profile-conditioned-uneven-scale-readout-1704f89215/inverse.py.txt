"""Pixel-only decision arm for uneven_scale_readout.

Receives the final PNG and nothing else.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
_RAIL = np.array([70, 70, 70], dtype=np.int16)
_PALETTE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
}
_MIN_RAIL_RUN = 400
_MIN_TICKS = 6
_MIN_POINTER_PIXELS = 150
_MIN_EDGE_CLEARANCE = 3.0
_MIN_GAP = 0.10


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def decision_from_image(image: Image.Image) -> str:
    """Recover the colour whose pointer reads the largest value."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = (np.abs(rgb - _RAIL).max(axis=2) <= 45)

    rail_rows = dark.sum(axis=1) >= _MIN_RAIL_RUN
    bands = _runs(rail_rows)
    if len(bands) != 4:
        return ABSTAIN

    rails: list[dict] = []
    for top, bottom in bands:
        tick_row = bottom + 10
        if tick_row >= dark.shape[0]:
            return ABSTAIN
        strokes = _runs(dark[tick_row])
        ticks = [0.5 * (a + b) for a, b in strokes if (b - a) <= 6]
        if len(ticks) < _MIN_TICKS:
            return ABSTAIN
        ticks.sort()
        if min(ticks[k + 1] - ticks[k] for k in range(len(ticks) - 1)) < 20.0:
            return ABSTAIN
        rails.append({"top": top, "bottom": bottom, "ticks": ticks})
    rails.sort(key=lambda item: item["top"])

    readings: dict[str, float] = {}
    for name, target in _PALETTE.items():
        mask = np.abs(rgb - np.array(target, dtype=np.int16)).max(axis=2) <= 60
        if int(mask.sum()) < _MIN_POINTER_PIXELS:
            return ABSTAIN
        rows = np.flatnonzero(mask.any(axis=1))
        cols = np.flatnonzero(mask.any(axis=0))
        if rows.size == 0 or cols.size == 0:
            return ABSTAIN
        if (rows[-1] - rows[0]) > 60 or (cols[-1] - cols[0]) > 60:
            return ABSTAIN
        pointer_x = 0.5 * (float(cols[0]) + float(cols[-1]))
        below = [rail for rail in rails if rail["top"] > rows[-1]]
        if not below:
            return ABSTAIN
        rail = min(below, key=lambda item: item["top"] - rows[-1])
        ticks = rail["ticks"]
        if not (ticks[0] < pointer_x < ticks[-1]):
            return ABSTAIN
        index = None
        for k in range(len(ticks) - 1):
            if ticks[k] <= pointer_x <= ticks[k + 1]:
                index = k
                break
        if index is None:
            return ABSTAIN
        low, high = ticks[index], ticks[index + 1]
        if pointer_x - low < _MIN_EDGE_CLEARANCE or high - pointer_x < _MIN_EDGE_CLEARANCE:
            return ABSTAIN
        readings[name] = index + (pointer_x - low) / (high - low)

    if len(readings) != 4:
        return ABSTAIN
    ranked = sorted(readings.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] - ranked[1][1] < _MIN_GAP:
        return ABSTAIN
    return ranked[0][0]
