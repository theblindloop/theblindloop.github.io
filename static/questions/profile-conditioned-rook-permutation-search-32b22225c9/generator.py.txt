"""Deterministic renderer and analytic arm for row-and-column rook search."""

from __future__ import annotations

import random
from itertools import product

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 560
BLUE = (36, 105, 198)
INK = (45, 49, 56)
GRID = (125, 132, 143)
PANEL = (224, 228, 235)
BACKGROUND = (248, 247, 243)
LABELS = "ABCD"
CENTERS = ((190, 170), (530, 170), (190, 410), (530, 410))
CELL = 48


def _score(board: list[list[int]]) -> int:
    return len({int(p[0]) for p in board}) + len({int(p[1]) for p in board})


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    cols = [0, 1, 2]
    rng.shuffle(cols)
    target = [[row, cols[row]] for row in range(3)]

    distractors: list[list[list[int]]] = []
    all_cells = list(product(range(3), repeat=2))
    while len(distractors) < 3:
        cells = rng.sample(all_cells, 3)
        board = [[r, c] for r, c in cells]
        if _score(board) >= 6 or board in distractors:
            continue
        distractors.append(board)

    boards: list[list[list[int]]] = []
    take = 0
    for index in range(4):
        if index == target_index:
            boards.append(target)
        else:
            boards.append(distractors[take])
            take += 1
    return {"boards": boards}


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        font = ImageFont.truetype(
            "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 24 * scale
        )
    except OSError:
        font = ImageFont.load_default()

    for index, (cx, cy) in enumerate(CENTERS):
        cx *= scale
        cy *= scale
        half = (3 * CELL // 2) * scale
        pad = 18 * scale
        draw.rounded_rectangle(
            (cx - half - pad, cy - half - pad, cx + half + pad, cy + half + pad),
            radius=10 * scale,
            fill=(255, 255, 255),
            outline=PANEL,
            width=3 * scale,
        )
        for k in range(4):
            x = cx + int((k - 1.5) * CELL * scale)
            y = cy + int((k - 1.5) * CELL * scale)
            draw.line((x, cy - half, x, cy + half), fill=GRID, width=2 * scale)
            draw.line((cx - half, y, cx + half, y), fill=GRID, width=2 * scale)
        label_pos = (cx - half - pad, cy - half - pad - 31 * scale)
        draw.text(label_pos, LABELS[index], font=font, fill=INK)
        for row, col in scene["boards"][index]:
            x = cx + int((int(col) - 1) * CELL * scale)
            y = cy + int((int(row) - 1) * CELL * scale)
            radius = 13 * scale
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=BLUE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    winners = [index for index, board in enumerate(scene["boards"]) if _score(board) == 6]
    if len(winners) != 1:
        raise ValueError(f"expected one permutation board, found {winners}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_score(board) for board in scene["boards"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0 or sum(_score(board) == 6 for board in scene["boards"]) != 1
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
