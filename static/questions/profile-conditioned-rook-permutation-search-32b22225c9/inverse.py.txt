"""PNG-only inverse arm. This module deliberately imports no world code."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = "ABCD"


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 150:
            found.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return found


def _cluster_count(values: list[float], tolerance: float = 12.0) -> int:
    groups: list[list[float]] = []
    for value in sorted(values):
        if not groups or value - sum(groups[-1]) / len(groups[-1]) > tolerance:
            groups.append([value])
        else:
            groups[-1].append(value)
    return len(groups)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    mask = (blue > 145) & (blue - red > 70) & (blue - green > 35) & (red < 100)
    dots = _components(mask)
    if len(dots) != 12:
        raise ValueError(f"expected 12 blue dots, recovered {len(dots)}")

    x_sorted = sorted(dot[0] for dot in dots)
    y_sorted = sorted(dot[1] for dot in dots)
    split_x = (x_sorted[5] + x_sorted[6]) / 2.0
    split_y = (y_sorted[5] + y_sorted[6]) / 2.0
    boards: list[list[tuple[float, float]]] = [[] for _ in range(4)]
    for x, y, _area in dots:
        index = (2 if y > split_y else 0) + (1 if x > split_x else 0)
        boards[index].append((x, y))
    if any(len(board) != 3 for board in boards):
        raise ValueError(f"could not recover four equal-count boards: {[len(b) for b in boards]}")

    scores = [
        _cluster_count([x for x, _ in board]) + _cluster_count([y for _, y in board])
        for board in boards
    ]
    winners = [index for index, score in enumerate(scores) if score == 6]
    if len(winners) != 1:
        raise ValueError(f"pixel evidence has no unique winner: scores={scores}")
    ordered = sorted(scores, reverse=True)
    if ordered[0] - ordered[1] < 1:
        raise ValueError(f"pixel runner-up margin is insufficient: scores={scores}")
    return LABELS[winners[0]]
