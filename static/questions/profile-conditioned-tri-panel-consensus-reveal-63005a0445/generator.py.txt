"""Deterministic three-layer consensus question world."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

N = 9
CELL = 18
PANEL = N * CELL
CANVAS = (640, 248)
INKS = [(20, 24, 32), (20, 35, 72), (52, 24, 66)]
FRAMES = [(0, 165, 190), (20, 150, 210), (0, 185, 180)]
LABELS = ("diamond", "plus", "X", "square")


def _template(label: str) -> np.ndarray:
    a = np.zeros((N, N), dtype=np.uint8)
    if label == "plus":
        a[4, :] = 1
        a[:, 4] = 1
    elif label == "X":
        for i in range(N):
            a[i, i] = a[i, N - 1 - i] = 1
    elif label == "square":
        a[2, 2:7] = a[6, 2:7] = 1
        a[2:7, 2] = a[2:7, 6] = 1
        a[4, 4] = 1
    elif label == "diamond":
        for y in range(N):
            for x in range(N):
                if abs(y - 4) + abs(x - 4) == 4:
                    a[y, x] = 1
        a[4, 4] = 1
    else:
        raise ValueError(f"unknown label {label!r}")
    assert int(a.sum()) == 17
    return a


def _vote(panels: Iterable) -> np.ndarray:
    a = np.asarray(list(panels), dtype=np.uint8)
    if a.shape != (3, N, N):
        raise ValueError(f"panels must have shape (3,{N},{N})")
    return (a.sum(axis=0) >= 2).astype(np.uint8)


def _distances(mask: np.ndarray) -> list[tuple[int, str]]:
    return sorted((int(np.count_nonzero(mask != _template(name))), name) for name in LABELS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    target = _template(label)
    panels = np.zeros((3, N, N), dtype=np.uint8)
    # Exactly two votes for target cells and exactly one for non-target cells.
    # Thus total source ink is 81 + 17 for every answer identity.
    for y in range(N):
        for x in range(N):
            order = [0, 1, 2]
            rng.shuffle(order)
            take = order[: 2 if target[y, x] else 1]
            panels[take, y, x] = 1
    order = [0, 1, 2]
    rng.shuffle(order)
    panels = panels[order]
    distractors = []
    for _ in range(2 + (int(seed) // 4) % 4):
        distractors.append(
            [rng.randint(12, CANVAS[0] - 13), rng.choice([18, 226]), rng.randint(2, 5)]
        )
    return {
        "panels": panels.tolist(),
        "ink": list(INKS[(int(seed) // 4) % len(INKS)]),
        "frame": list(FRAMES[(int(seed) // 12) % len(FRAMES)]),
        "origin_x": 42 + 3 * ((int(seed) // 12) % 3),
        "origin_y": 47 + 2 * ((int(seed) // 36) % 3),
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    ink = tuple(scene["ink"])
    frame = tuple(scene["frame"])
    origin_x = int(scene["origin_x"])
    origin_y = int(scene["origin_y"])
    distractors = scene["distractors"]
    image = Image.new("RGB", CANVAS, (248, 247, 243))
    draw = ImageDraw.Draw(image)
    for dx, dy, radius in distractors:
        draw.ellipse((dx - radius, dy - radius, dx + radius, dy + radius), fill=(164, 165, 168))
    for p, panel in enumerate(panels):
        x0 = origin_x + p * (PANEL + 30)
        y0 = origin_y
        draw.rectangle((x0 - 4, y0 - 4, x0 + PANEL + 3, y0 + PANEL + 3), outline=frame, width=4)
        for i in range(N + 1):
            q = i * CELL
            draw.line((x0 + q, y0, x0 + q, y0 + PANEL), fill=(210, 211, 208), width=1)
            draw.line((x0, y0 + q, x0 + PANEL, y0 + q), fill=(210, 211, 208), width=1)
        for y in range(N):
            for x in range(N):
                if panel[y][x]:
                    draw.rectangle(
                        (x0 + x * CELL + 2, y0 + y * CELL + 2,
                         x0 + (x + 1) * CELL - 2, y0 + (y + 1) * CELL - 2),
                        fill=ink,
                    )
    return image


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    return _distances(_vote(panels))[0][1]


def margin(scene: dict) -> float:
    distances = _distances(_vote(scene["panels"]))
    return float(distances[1][0] - distances[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
