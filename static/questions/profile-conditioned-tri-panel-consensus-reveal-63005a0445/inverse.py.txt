"""Independent inverse arm; receives only the final raster image."""

from __future__ import annotations

import numpy as np

N = 9
LABELS = ("diamond", "plus", "X", "square")


def _template(label):
    a = np.zeros((N, N), dtype=np.uint8)
    if label == "plus":
        a[4, :] = 1
        a[:, 4] = 1
    elif label == "X":
        for i in range(N):
            a[i, i] = a[i, N - 1 - i] = 1
    elif label == "square":
        a[2, 2:7] = a[6, 2:7] = 1
        a[2:7, 2] = a[2:7, 6] = 1
        a[4, 4] = 1
    elif label == "diamond":
        for y in range(N):
            for x in range(N):
                if abs(y - 4) + abs(x - 4) == 4:
                    a[y, x] = 1
        a[4, 4] = 1
    return a


def _components(mask):
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    stack.append((yy, xx))
        if len(points) >= 400:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            found.append((min(xs), min(ys), max(xs), max(ys), len(points)))
    return found


def _read_panels(image):
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Cyan family, deliberately broader than any one renderer palette entry.
    cyan = (arr[:, :, 0] < 70) & (arr[:, :, 1] > 115) & (arr[:, :, 2] > 125)
    boxes = sorted(_components(cyan), key=lambda box: box[0])
    if len(boxes) != 3:
        raise ValueError(f"expected three visible cyan frames, found {len(boxes)}")
    panels = []
    for left, top, right, bottom, _ in boxes:
        span_x = right - left + 1
        span_y = bottom - top + 1
        if not (165 <= span_x <= 175 and 165 <= span_y <= 175):
            raise ValueError("frame geometry is not a square nine-cell panel")
        x0, y0 = left + 4, top + 4
        usable_x, usable_y = span_x - 8, span_y - 8
        panel = np.zeros((N, N), dtype=np.uint8)
        for y in range(N):
            for x in range(N):
                xa = round(x0 + (x + 0.31) * usable_x / N)
                xb = round(x0 + (x + 0.69) * usable_x / N)
                ya = round(y0 + (y + 0.31) * usable_y / N)
                yb = round(y0 + (y + 0.69) * usable_y / N)
                patch = arr[ya:yb, xa:xb]
                luminance = patch.mean(axis=2)
                panel[y, x] = int(float(np.median(luminance)) < 105.0)
        panels.append(panel)
    return np.stack(panels)


def _decision_and_margin(image):
    panels = _read_panels(image)
    consensus = (panels.sum(axis=0) >= 2).astype(np.uint8)
    ranked = sorted(
        (int(np.count_nonzero(consensus != _template(label))), label) for label in LABELS
    )
    gap = ranked[1][0] - ranked[0][0]
    if ranked[0][0] != 0 or gap < 6:
        raise ValueError(f"ambiguous transformed symbol: best={ranked[0][0]}, gap={gap}")
    return ranked[0][1], float(gap)


def decision_from_image(image):
    return _decision_and_margin(image)[0]
