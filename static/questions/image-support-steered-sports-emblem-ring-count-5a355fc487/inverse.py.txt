"""Independent image-only inverse arm for counting complete chromatic rings."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[list[tuple[int, int]]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            pixels: list[tuple[int, int]] = []
            while queue:
                cy, cx = queue.popleft()
                pixels.append((cy, cx))
                for ny in range(max(0, cy - 1), min(height, cy + 2)):
                    for nx in range(max(0, cx - 1), min(width, cx + 2)):
                        if mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            queue.append((ny, nx))
            found.append(pixels)
    return found


def _is_complete_ring(component: list[tuple[int, int]], mask: np.ndarray) -> bool:
    if len(component) < 400:
        return False
    ys = [p[0] for p in component]
    xs = [p[1] for p in component]
    y0, y1, x0, x1 = min(ys), max(ys), min(xs), max(xs)
    height, width = y1 - y0 + 1, x1 - x0 + 1
    if not (40 <= width <= 58 and 40 <= height <= 58 and 0.82 <= width / height <= 1.22):
        return False
    cy, cx = (y0 + y1) / 2.0, (x0 + x1) / 2.0
    inner = max(3, int(min(width, height) * 0.22))
    yy0, yy1 = max(0, int(cy) - inner), min(mask.shape[0], int(cy) + inner + 1)
    xx0, xx1 = max(0, int(cx) - inner), min(mask.shape[1], int(cx) + inner + 1)
    if float(mask[yy0:yy1, xx0:xx1].mean()) > 0.06:
        return False
    fill_fraction = len(component) / float(width * height)
    return 0.34 <= fill_fraction <= 0.66


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    maximum = rgb.max(axis=2)
    minimum = rgb.min(axis=2)
    chromatic = (maximum - minimum > 62) & (maximum > 105) & (minimum < 185)
    count = sum(_is_complete_ring(component, chromatic) for component in _components(chromatic))
    return str(int(count))
