"""Deterministic renderer and analytic arm for the sports-emblem ring world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
PALETTES = (
    ("#1677b8", "#e4572e", "#00a878", "#8f4bb8", "#d6284e", "#008fa8"),
    ("#3155a6", "#ef6c23", "#149b68", "#a23fa3", "#cf334d", "#087fbd"),
    ("#1769aa", "#d95f2a", "#079b83", "#794db2", "#df3157", "#0092a3"),
)


def _base_centers(count: int) -> list[tuple[int, int]]:
    if count == 4:
        return [(180, 190), (180, 270), (460, 190), (460, 270)]
    if count == 5:
        return [(180, 170), (180, 230), (180, 290), (460, 195), (460, 265)]
    return [(180, 170), (180, 230), (180, 290), (460, 170), (460, 230), (460, 290)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    ring_count = 4 + (int(seed) % 3)
    dx, dy = rng.randint(-18, 18), rng.randint(-10, 10)
    centers = [[x + dx, y + dy] for x, y in _base_centers(ring_count)]
    palette = list(PALETTES[rng.randrange(len(PALETTES))])
    rng.shuffle(palette)
    radius = rng.randint(23, 25)
    thickness = rng.randint(8, 10)
    card_box = [88 + dx // 4, 76 + dy // 4, 552 + dx // 4, 344 + dy // 4]
    accent_dots = [[card_box[0] + 22, card_box[1] + 22], [card_box[2] - 22, card_box[1] + 22]]
    return {
        "ring_count": ring_count,
        "palette": palette,
        "centers": centers,
        "radius": radius,
        "thickness": thickness,
        "card_box": card_box,
        "accent_dots": accent_dots,
    }


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (248, 246, 240))
    draw = ImageDraw.Draw(image)
    card = [int(v) * scale for v in scene["card_box"]]
    draw.rounded_rectangle(card, radius=24 * scale, fill=(255, 255, 255), outline=(55, 62, 70), width=3 * scale)
    draw.text((card[0] + 24 * scale, card[1] + 16 * scale), "SPORTS EMBLEM", fill=(70, 74, 82), stroke_width=0)
    for dot in scene["accent_dots"]:
        x, y = int(dot[0]) * scale, int(dot[1]) * scale
        draw.ellipse([x - 4 * scale, y - 4 * scale, x + 4 * scale, y + 4 * scale], fill=(98, 103, 110))
    radius = int(scene["radius"]) * scale
    thickness = int(scene["thickness"]) * scale
    visible_centers = scene["centers"][: int(scene["ring_count"])]
    for index, center in enumerate(visible_centers):
        x, y = int(center[0]) * scale, int(center[1]) * scale
        color = scene["palette"][index]
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=color)
        inner = radius - thickness
        draw.ellipse([x - inner, y - inner, x + inner, y + inner], fill=(255, 255, 255))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["ring_count"])


def margin(scene: dict) -> float:
    centers = scene["centers"]
    radius = float(scene["radius"])
    pair_clearances = [
        math.hypot(a[0] - b[0], a[1] - b[1]) - 2.0 * radius
        for i, a in enumerate(centers)
        for b in centers[i + 1 :]
    ]
    card = scene["card_box"]
    edge_clearances = [
        min(x - radius - card[0], card[2] - x - radius, y - radius - card[1], card[3] - y - radius)
        for x, y in centers
    ]
    return float(min(pair_clearances + edge_clearances + [float(scene["thickness"])]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 7.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["accent_dots"] = list(reversed(twin["accent_dots"]))
    return [("reverse_neutral_dot_draw_order", twin)]
