"""Deterministic sampler, cubical replay, and compact raster renderer."""

from __future__ import annotations

import itertools
import random
from collections import Counter
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BADGE_SIZE = 96

FACE_POSITIONS = ("top", "bottom", "front", "back", "left", "right")
DIRECTIONS = ("up", "down", "left", "right")
COLOR_NAMES = ("red", "blue", "green", "yellow", "purple", "orange")
PALETTE = {
    "red": (222, 68, 72),
    "blue": (52, 121, 211),
    "green": (55, 166, 105),
    "yellow": (238, 188, 47),
    "purple": (149, 83, 191),
    "orange": (232, 126, 45),
}
THEMES = (
    (28, 72, 104),
    (86, 51, 105),
    (31, 91, 82),
)
BACKGROUND = (248, 247, 242)
DECOR = (210, 213, 216)
INK = (14, 19, 24)
PANEL = (235, 234, 224)
RAIL_LINE = (151, 164, 172)

# The net is L-F-R-B across, with TOP above FRONT and BOTTOM below FRONT.
NET_CELLS = {
    "top": (19, 20),
    "left": (5, 34),
    "front": (19, 34),
    "right": (33, 34),
    "back": (47, 34),
    "bottom": (19, 48),
}
ARROW_CENTERS = tuple((79, 12 + 14 * index) for index in range(6))


def _roll(orientation: dict[str, str], command: str) -> dict[str, str]:
    """Return the six face identities after one physical quarter-roll."""

    old = dict(orientation)
    if command == "up":
        return {
            "top": old["front"], "back": old["top"],
            "bottom": old["back"], "front": old["bottom"],
            "left": old["left"], "right": old["right"],
        }
    if command == "down":
        return {
            "top": old["back"], "front": old["top"],
            "bottom": old["front"], "back": old["bottom"],
            "left": old["left"], "right": old["right"],
        }
    if command == "left":
        return {
            "top": old["right"], "left": old["top"],
            "bottom": old["left"], "right": old["bottom"],
            "front": old["front"], "back": old["back"],
        }
    if command == "right":
        return {
            "top": old["left"], "right": old["top"],
            "bottom": old["right"], "left": old["bottom"],
            "front": old["front"], "back": old["back"],
        }
    raise ValueError(f"unknown roll command {command!r}")


def _terminal_source_face(commands: tuple[str, ...] | list[str]) -> str:
    orientation = {face: face for face in FACE_POSITIONS}
    for command in commands:
        orientation = _roll(orientation, command)
    return orientation["top"]


def _program_pool() -> dict[str, tuple[tuple[str, ...], ...]]:
    """Fixed-ink programs: 2 up, 2 down, 1 left, and 1 right."""

    inverse = {("up", "down"), ("down", "up"), ("left", "right"), ("right", "left")}
    unique = sorted(set(itertools.permutations(("up", "up", "down", "down", "left", "right"))))
    pools: dict[str, list[tuple[str, ...]]] = {face: [] for face in FACE_POSITIONS}
    for program in unique:
        if any(pair in inverse for pair in zip(program, program[1:])):
            continue
        pools[_terminal_source_face(program)].append(program)
    if any(not programs for programs in pools.values()):
        raise AssertionError("the fixed-multiset roll pool must reach every face")
    return {face: tuple(programs) for face, programs in pools.items()}


PROGRAMS_BY_TOP_FACE = _program_pool()


def _decorations(rng: random.Random, badge_x: int, badge_y: int) -> list[dict]:
    lattice = []
    for y in range(28, HEIGHT - 27, 48):
        for x in range(28, WIDTH - 27, 48):
            if badge_x - 24 <= x <= badge_x + BADGE_SIZE + 24:
                if badge_y - 24 <= y <= badge_y + BADGE_SIZE + 24:
                    continue
            lattice.append((x, y))
    rng.shuffle(lattice)
    kinds = ("ring", "diamond", "cross")
    return [
        {"x": x, "y": y, "kind": kinds[rng.randrange(3)], "size": 5 + rng.randrange(3)}
        for x, y in lattice[:12]
    ]


def sample_scene(seed: int) -> dict:
    """Sample exactly balanced answers with position and command-order variation."""

    seed = int(seed)
    rng = random.Random((seed * 1_000_003) ^ 0xC0BE7011)
    target_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    desired_face = FACE_POSITIONS[((seed // 6) + (seed % 6)) % len(FACE_POSITIONS)]
    programs = PROGRAMS_BY_TOP_FACE[desired_face]
    commands = list(programs[rng.randrange(len(programs))])

    remaining_colors = [color for color in COLOR_NAMES if color != target_color]
    rng.shuffle(remaining_colors)
    face_colors: dict[str, str] = {desired_face: target_color}
    for face, color in zip((f for f in FACE_POSITIONS if f != desired_face), remaining_colors):
        face_colors[face] = color

    grid_col = rng.randrange(1, 7)
    grid_row = rng.randrange(1, 7)
    badge_x = grid_col * 64 - 16
    badge_y = grid_row * 64 - 16
    return {
        "face_colors": face_colors,
        "commands": commands,
        "grid_col": grid_col,
        "grid_row": grid_row,
        "theme": rng.randrange(len(THEMES)),
        "decorations": _decorations(rng, badge_x, badge_y),
    }


def analytic_gold(scene: dict) -> str:
    orientation = {face: str(scene["face_colors"][face]) for face in FACE_POSITIONS}
    for command in scene["commands"]:
        orientation = _roll(orientation, str(command))
    return orientation["top"]


def margin(scene: dict) -> float:
    """One-sided nominal pixel margin for four-way arrow-centroid decoding."""

    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_unordered_decorations", twin)]


def _draw_decoration(draw: ImageDraw.ImageDraw, item: dict) -> None:
    x, y, size = int(item["x"]), int(item["y"]), int(item["size"])
    if item["kind"] == "ring":
        draw.ellipse((x - size, y - size, x + size, y + size), outline=DECOR, width=2)
    elif item["kind"] == "diamond":
        draw.polygon(((x, y - size), (x + size, y), (x, y + size), (x - size, y)), outline=DECOR)
    elif item["kind"] == "cross":
        draw.line((x - size, y, x + size, y), fill=DECOR, width=2)
        draw.line((x, y - size, x, y + size), fill=DECOR, width=2)
    else:
        raise ValueError(f"unknown decoration {item['kind']!r}")


def _arrow_vertices(cx: int, cy: int, direction: str) -> tuple[tuple[int, int], ...]:
    if direction == "up":
        return ((cx, cy - 5), (cx - 5, cy + 4), (cx + 5, cy + 4))
    if direction == "down":
        return ((cx, cy + 5), (cx - 5, cy - 4), (cx + 5, cy - 4))
    if direction == "left":
        return ((cx - 5, cy), (cx + 4, cy - 5), (cx + 4, cy + 5))
    if direction == "right":
        return ((cx + 5, cy), (cx - 4, cy - 5), (cx - 4, cy + 5))
    raise ValueError(f"unknown direction {direction!r}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    face_colors = {str(k): str(v) for k, v in scene["face_colors"].items()}
    commands = [str(command) for command in scene["commands"]]
    grid_col = int(scene["grid_col"])
    grid_row = int(scene["grid_row"])
    theme = int(scene["theme"])
    decorations = list(scene["decorations"])
    if set(face_colors) != set(FACE_POSITIONS) or set(face_colors.values()) != set(COLOR_NAMES):
        raise ValueError("face colors must be a permutation of the six public colors")
    if len(commands) != 6 or Counter(commands) != Counter({"up": 2, "down": 2, "left": 1, "right": 1}):
        raise ValueError("invalid fixed-multiset roll program")
    if not 1 <= grid_col <= 6 or not 1 <= grid_row <= 6 or not 0 <= theme < len(THEMES):
        raise ValueError("invalid visible style latent")

    for item in decorations:
        _draw_decoration(draw, item)

    x0 = grid_col * 64 - 16
    y0 = grid_row * 64 - 16
    draw.rectangle((x0, y0, x0 + 95, y0 + 95), fill=THEMES[theme])

    # A small bridge makes the net and event rail read as one postage-stamp console.
    draw.rectangle((x0 + 58, y0 + 38, x0 + 69, y0 + 43), fill=PANEL)
    for face, (rx, ry) in NET_CELLS.items():
        draw.rectangle((x0 + rx, y0 + ry, x0 + rx + 13, y0 + ry + 13), fill=INK)
        draw.rectangle((x0 + rx + 2, y0 + ry + 2, x0 + rx + 11, y0 + ry + 11), fill=PALETTE[face_colors[face]])

    draw.rectangle((x0 + 67, y0 + 5, x0 + 90, y0 + 90), fill=INK)
    draw.rectangle((x0 + 69, y0 + 7, x0 + 88, y0 + 88), fill=PANEL)
    for index in range(1, 6):
        y = y0 + 5 + index * 14
        draw.line((x0 + 68, y, x0 + 89, y), fill=RAIL_LINE, width=1)
    for command, (cx, cy) in zip(commands, ARROW_CENTERS):
        draw.polygon(_arrow_vertices(x0 + cx, y0 + cy, command), fill=INK)

    return image
