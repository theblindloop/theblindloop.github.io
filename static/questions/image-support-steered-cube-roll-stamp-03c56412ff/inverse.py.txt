"""Independent pixel-only inverse for the compact cube-roll stamp.

The inverse locates the translated connected console, validates its frame,
recovers all six face colors and all six arrow directions, composes physical
cube rolls, and returns the terminal top color.  It receives no scene data and
imports no renderer, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

FACE_POSITIONS = ("top", "bottom", "front", "back", "left", "right")
PALETTE = {
    "red": (222, 68, 72),
    "blue": (52, 121, 211),
    "green": (55, 166, 105),
    "yellow": (238, 188, 47),
    "purple": (149, 83, 191),
    "orange": (232, 126, 45),
}
NET_CELLS = {
    "top": (19, 20),
    "left": (5, 34),
    "front": (19, 34),
    "right": (33, 34),
    "back": (47, 34),
    "bottom": (19, 48),
}
ARROW_CENTERS = tuple((79, 12 + 14 * index) for index in range(6))


def _roll(orientation: dict[str, str], command: str) -> dict[str, str]:
    old = dict(orientation)
    if command == "up":
        return {"top": old["front"], "back": old["top"], "bottom": old["back"], "front": old["bottom"], "left": old["left"], "right": old["right"]}
    if command == "down":
        return {"top": old["back"], "front": old["top"], "bottom": old["front"], "back": old["bottom"], "left": old["left"], "right": old["right"]}
    if command == "left":
        return {"top": old["right"], "left": old["top"], "bottom": old["left"], "right": old["bottom"], "front": old["front"], "back": old["back"]}
    if command == "right":
        return {"top": old["left"], "right": old["top"], "bottom": old["right"], "left": old["bottom"], "front": old["front"], "back": old["back"]}
    raise ValueError("unknown decoded roll")


def _decode_face(patch: np.ndarray) -> str:
    pixels = patch.astype(np.int16).reshape(-1, 3)
    palette = np.asarray(list(PALETTE.values()), dtype=np.int16)
    distances = np.linalg.norm(pixels[:, None, :] - palette[None, :, :], axis=2)
    labels = np.argmin(distances, axis=1)
    winner, count = Counter(labels.tolist()).most_common(1)[0]
    if count < int(0.9 * len(labels)) or float(np.median(distances[:, winner])) > 8.0:
        raise ValueError("a face color is missing or ambiguous")
    ordered = np.sort(np.mean(distances, axis=0))
    if float(ordered[1] - ordered[0]) < 60.0:
        raise ValueError("face color runner-up margin collapsed")
    return tuple(PALETTE)[int(winner)]


def _decode_arrow(crop: np.ndarray) -> tuple[str, float]:
    dark = np.all(crop < 45, axis=2)
    py, px = np.nonzero(dark)
    if not 45 <= len(px) <= 80:
        raise ValueError("an arrow is missing or malformed")
    dx = float(px.mean() - 5.0)
    dy = float(py.mean() - 5.0)
    axis_gap = abs(abs(dx) - abs(dy))
    magnitude = max(abs(dx), abs(dy))
    if axis_gap < 0.65 or magnitude < 0.75:
        raise ValueError("arrow orientation is tied")
    if abs(dx) > abs(dy):
        return ("left" if dx > 0 else "right"), min(axis_gap, magnitude)
    return ("up" if dy > 0 else "down"), min(axis_gap, magnitude)


def _decode(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise ValueError("expected a 512 by 512 RGB raster")

    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    saturated = spread > 35
    ys, xs = np.nonzero(saturated)
    if len(xs) < 5400:
        raise ValueError("compact console is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 + 1 != 96 or y1 - y0 + 1 != 96:
        raise ValueError("console bounds are incomplete or ambiguous")

    # Four substantial interior corner anchors make frame loss observable.
    for rx, ry in ((1, 1), (91, 1), (1, 91), (91, 91)):
        patch = saturated[y0 + ry:y0 + ry + 4, x0 + rx:x0 + rx + 4]
        if patch.shape != (4, 4) or float(patch.mean()) < 0.95:
            raise ValueError("connected console frame is incomplete")

    orientation: dict[str, str] = {}
    for face, (rx, ry) in NET_CELLS.items():
        patch = array[y0 + ry + 3:y0 + ry + 11, x0 + rx + 3:x0 + rx + 11]
        if patch.shape != (8, 8, 3):
            raise ValueError("cube net is cropped")
        orientation[face] = _decode_face(patch)
    if set(orientation.values()) != set(PALETTE):
        raise ValueError("the six face identities are not unique")

    commands = []
    margins = []
    for rx, ry in ARROW_CENTERS:
        crop = array[y0 + ry - 5:y0 + ry + 6, x0 + rx - 5:x0 + rx + 6]
        if crop.shape != (11, 11, 3):
            raise ValueError("roll rail is cropped")
        command, arrow_margin = _decode_arrow(crop)
        commands.append(command)
        margins.append(arrow_margin)
    if Counter(commands) != Counter({"up": 2, "down": 2, "left": 1, "right": 1}):
        raise ValueError("the fixed-ink arrow contract is broken")

    for command in commands:
        orientation = _roll(orientation, command)
    return orientation["top"], min(margins)


def decision_from_image(image: Image.Image) -> str:
    decision, _pixel_margin = _decode(image)
    return decision
