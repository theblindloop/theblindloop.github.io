"""Latent sampler, analytic arm, and Pillow renderer for Echo Pair Trace."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
DARK = (35, 44, 58)
GOLD = (238, 177, 43)
PALE = (226, 231, 237)
PALETTES = (
    ((20, 174, 196), (220, 82, 92)),
    ((41, 158, 105), (139, 83, 190)),
)
FACES = ("circle", "triangle")


def _final_face(start_face: str, span_colors: list[int]) -> str:
    face = start_face
    for before, current in zip(span_colors, span_colors[1:]):
        if before == current:
            face = FACES[1 - FACES.index(face)]
    return face


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    start_face = FACES[(int(seed) // 2) % 2]
    target = FACES[int(seed) % 2]
    for _ in range(1000):
        colors = [rng.randrange(2) for _ in range(6)]
        count_one = sum(colors)
        repeats = sum(a == b for a, b in zip(colors, colors[1:]))
        if 2 <= count_one <= 4 and 1 <= repeats <= 4 and _final_face(start_face, colors) == target:
            break
    else:
        raise AssertionError("could not balance a span sequence")

    return {
        "start_face": start_face,
        "span_colors": colors,
        "path_variant": rng.randrange(3),
        "direction": 1 if rng.randrange(2) == 0 else -1,
        "palette_variant": rng.randrange(2),
        "translation": [rng.randint(-18, 18), rng.randint(-14, 14)],
        "decorations": [[rng.randint(28, 484), rng.randint(28, 484), rng.choice([2, 3])] for _ in range(10)],
    }


def _ordered_points(scene: dict[str, Any]) -> list[tuple[int, int]]:
    variants = (
        [(-126, -46), (-84, -8), (-42, -51), (0, -8), (42, 34), (84, -8), (126, 34)],
        [(-126, 34), (-84, -8), (-42, 34), (0, -8), (42, -51), (84, -8), (126, -46)],
        [(-126, -32), (-84, 15), (-42, -32), (0, 15), (42, -32), (84, 15), (126, -32)],
    )
    dx, dy = (int(v) for v in scene["translation"])
    points = [(256 + dx + x, 256 + dy + y) for x, y in variants[int(scene["path_variant"])]]
    if int(scene["direction"]) == -1:
        points.reverse()
    return points


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        if not (95 <= x <= 417 and 155 <= y <= 355):
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    points = _ordered_points(scene)
    palette = PALETTES[int(scene["palette_variant"])]
    for index, (a, b) in enumerate(zip(points, points[1:])):
        draw.line((a, b), fill=palette[int(scene["span_colors"][index])], width=16)

    # Dark junctions visibly divide even same-color consecutive spans while the
    # union remains a single connected trace.
    for x, y in points[1:-1]:
        draw.ellipse((x - 14, y - 14, x + 14, y + 14), fill=DARK)

    sx, sy = points[0]
    draw.ellipse((sx - 17, sy - 17, sx + 17, sy + 17), fill=DARK)
    draw.ellipse((sx - 11, sy - 11, sx + 11, sy + 11), fill=BACKGROUND)
    if scene["start_face"] == "circle":
        draw.ellipse((sx - 7, sy - 7, sx + 7, sy + 7), fill=GOLD)
    else:
        draw.polygon(((sx, sy - 9), (sx - 9, sy + 8), (sx + 9, sy + 8)), fill=GOLD)

    ex, ey = points[-1]
    px, py = points[-2]
    ux, uy = ex - px, ey - py
    length = math.hypot(ux, uy)
    ux, uy = ux / length, uy / length
    nx, ny = -uy, ux
    tip = (ex + 16 * ux, ey + 16 * uy)
    left = (ex - 10 * ux + 13 * nx, ey - 10 * uy + 13 * ny)
    right = (ex - 10 * ux - 13 * nx, ey - 10 * uy - 13 * ny)
    draw.polygon((tip, left, right), fill=DARK)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return _final_face(str(scene["start_face"]), [int(v) for v in scene["span_colors"]])


def margin(scene: dict[str, Any]) -> float:
    # Runner-up separation is governed by the 15 px face scale and the large
    # RGB separation of both legal palettes; the smaller geometric lever wins.
    return 15.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    colors = list(scene["span_colors"])
    return bool(
        scene["start_face"] not in FACES
        or len(colors) != 6
        or any(int(v) not in (0, 1) for v in colors)
        or not 2 <= sum(int(v) for v in colors) <= 4
        or int(scene["path_variant"]) not in range(3)
        or int(scene["direction"]) not in (-1, 1)
        or int(scene["palette_variant"]) not in (0, 1)
    )


def nuisance_transforms(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    shifted = dict(scene)
    dx, dy = (int(v) for v in scene["translation"])
    shifted["translation"] = [dx + (-12 if dx > 0 else 12), dy]
    recolored = dict(scene)
    recolored["palette_variant"] = 1 - int(scene["palette_variant"])
    return [("horizontal_translation", shifted), ("operation_palette_exchange", recolored)]


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
