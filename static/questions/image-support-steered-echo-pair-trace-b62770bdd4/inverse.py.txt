"""Independent PNG-only inverse arm for Echo Pair Trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_DARK = np.array((35, 44, 58), dtype=np.int16)
_GOLD = np.array((238, 177, 43), dtype=np.int16)
_OP_COLORS = (
    np.array((20, 174, 196), dtype=np.int16),
    np.array((220, 82, 92), dtype=np.int16),
    np.array((41, 158, 105), dtype=np.int16),
    np.array((139, 83, 190), dtype=np.int16),
)
_FACES = ("circle", "triangle")


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int32))
    return result


def _bbox(component: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = component[:, 0], component[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas size")

    dark = _mask(array, _DARK)
    gold = _mask(array, _GOLD)
    op_masks = [_mask(array, color) for color in _OP_COLORS]
    trace = dark.copy()
    for mask in op_masks:
        trace |= mask

    large_trace = [component for component in _components(trace) if len(component) >= 2500]
    if len(large_trace) != 1:
        raise ValueError("the connected decisive trace is missing or split")
    x0, y0, x1, y1 = _bbox(large_trace[0])
    if not 250 <= x1 - x0 + 1 <= 310 or not 70 <= y1 - y0 + 1 <= 135:
        raise ValueError("trace extent is ambiguous")

    gold_components = [component for component in _components(gold) if len(component) >= 80]
    if len(gold_components) != 1:
        raise ValueError("starting face is missing or ambiguous")
    icon = gold_components[0]
    ix0, iy0, ix1, iy1 = _bbox(icon)
    iw, ih = ix1 - ix0 + 1, iy1 - iy0 + 1
    if not 13 <= iw <= 20 or not 13 <= ih <= 19:
        raise ValueError("starting face geometry is damaged")
    start_face = "circle" if iw <= 16 else "triangle"
    start_x = float(icon[:, 1].mean())

    spans = []
    for color_id, mask in enumerate(op_masks):
        for component in _components(mask):
            if len(component) < 350:
                continue
            bx0, by0, bx1, by1 = _bbox(component)
            width, height = bx1 - bx0 + 1, by1 - by0 + 1
            if not 30 <= width <= 53 or not 26 <= height <= 58 or not 350 <= len(component) <= 900:
                raise ValueError("colored span geometry is ambiguous")
            spans.append((float(component[:, 1].mean()), color_id))
    if len(spans) != 6:
        raise ValueError("expected exactly six intact colored spans")

    center_x = sum(x for x, _ in spans) / len(spans)
    direction = 1 if center_x > start_x else -1
    spans.sort(key=lambda item: direction * item[0])
    observed = [color for _, color in spans]
    # A legal scene uses exactly one of the two two-color palettes.
    palette_ids = {color // 2 for color in observed}
    if len(palette_ids) != 1 or len(set(observed)) != 2:
        raise ValueError("operation palette is incomplete or mixed")

    face = start_face
    for before, current in zip(observed, observed[1:]):
        if before == current:
            face = _FACES[1 - _FACES.index(face)]
    return face
