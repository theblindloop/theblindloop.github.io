"""Deterministic latent sampler and raster renderer for branch_rejoin_badge."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

BG = (246, 244, 238)
PANEL = (232, 226, 212)
NAVY = (30, 52, 70)
GREEN = (42, 166, 91)
ORANGE = (239, 132, 36)
PALE = (205, 208, 202)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    near_boundary = int(seed) % 11 == 0
    rejoins = bool(rng.getrandbits(1))
    return {
        "width": 512,
        "height": 512,
        "cx": rng.randint(105, 407),
        "cy": rng.randint(105, 407),
        "rotation": rng.choice((0, 90, 180, 270)),
        "rejoins": rejoins,
        "line_width": 6 if near_boundary and rejoins else rng.choice((9, 10, 11)),
        "gap_px": 6 if near_boundary and not rejoins else rng.choice((13, 15, 17)),
        "decorations": [
            [rng.randint(28, 484), rng.randint(28, 484), rng.choice((3, 4, 5))]
            for _ in range(5)
        ],
    }


def _turn(point: tuple[float, float], rotation: int) -> tuple[float, float]:
    x, y = point
    if rotation == 0:
        return x, y
    if rotation == 90:
        return -y, x
    if rotation == 180:
        return -x, -y
    return y, -x


def _points(scene: dict, points: Iterable[tuple[float, float]]) -> list[tuple[int, int]]:
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    rotation = int(scene["rotation"])
    return [(round(cx + x), round(cy + y)) for x, y in (_turn(p, rotation) for p in points)]


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    rotation = int(scene["rotation"])
    rejoins = bool(scene["rejoins"])
    line_width = int(scene["line_width"])
    gap_px = int(scene["gap_px"])
    decorations = list(scene["decorations"])

    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    for x, y, radius in decorations:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    # A filled inset visually groups all decisive evidence into one local badge.
    draw.rounded_rectangle((cx - 62, cy - 48, cx + 62, cy + 48), radius=18, fill=PANEL)

    split = (-30.0, 0.0)
    join = (30.0, 0.0)
    upper = [split, (-18.0, -23.0), (16.0, -23.0)]
    lower = [split, (-18.0, 23.0), (18.0, 23.0), join]
    if rejoins:
        upper.append(join)
    else:
        # Move the dead end left/right by the declared visible gap while keeping
        # the rest of the route unchanged. gap_px is therefore rendered even
        # though the oracle never receives it.
        upper[-1] = (30.0 - float(gap_px), -23.0)

    start_stub = [(-50.0, 0.0), split]
    finish_stub = [join, (50.0, 0.0)]
    for path in (start_stub, upper, lower, finish_stub):
        draw.line(_points(scene, path), fill=NAVY, width=line_width, joint="curve")
    node_radius = max(4, line_width // 2)
    for node in (split, join):
        px, py = _points(scene, [node])[0]
        draw.ellipse(
            (px - node_radius, py - node_radius, px + node_radius, py + node_radius),
            fill=NAVY,
        )

    start, finish = _points(scene, [(-50.0, 0.0), (50.0, 0.0)])
    marker_radius = 8
    draw.ellipse(
        (start[0] - marker_radius, start[1] - marker_radius,
         start[0] + marker_radius, start[1] + marker_radius),
        fill=GREEN,
        outline=(255, 255, 255),
        width=2,
    )
    draw.ellipse(
        (finish[0] - marker_radius, finish[1] - marker_radius,
         finish[0] + marker_radius, finish[1] + marker_radius),
        fill=ORANGE,
        outline=(255, 255, 255),
        width=2,
    )
    # Consume the local variables explicitly so every declared scene field is
    # part of the visible rendering contract under source audit.
    _ = rotation
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["rejoins"]) else "no"


def margin(scene: dict) -> float:
    return float(scene["line_width"] if bool(scene["rejoins"]) else scene["gap_px"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
