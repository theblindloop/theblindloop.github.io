"""Independent image-only inverse arm for compact route reconvergence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = np.array((30, 52, 70), dtype=np.int16)
_GREEN = np.array((42, 166, 91), dtype=np.int16)
_ORANGE = np.array((239, 132, 36), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, radius: float = 24.0) -> np.ndarray:
    # int32 keeps squared RGB distances from overflowing signed int16.
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[list[tuple[int, int]]] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(y, x)]
        comp: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            comp.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(comp)
    return output


def _marker_centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    components = [item for item in _components(mask) if len(item) >= 45]
    if len(components) != 1:
        raise ValueError(f"expected one intact {name} marker")
    points = np.asarray(components[0], dtype=float)
    return float(points[:, 0].mean()), float(points[:, 1].mean())


def _enclosed_background_count(route: np.ndarray) -> int:
    ys, xs = np.nonzero(route)
    if len(ys) < 430:
        raise ValueError("route ink is missing or materially erased")
    y0, y1 = max(0, int(ys.min()) - 2), min(route.shape[0], int(ys.max()) + 3)
    x0, x1 = max(0, int(xs.min()) - 2), min(route.shape[1], int(xs.max()) + 3)
    crop = route[y0:y1, x0:x1]
    height, width = crop.shape
    open_space = ~crop
    exterior = np.zeros_like(open_space)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if open_space[y, x] and not exterior[y, x]:
                exterior[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if open_space[y, x] and not exterior[y, x]:
                exterior[y, x] = True
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and open_space[ny, nx] and not exterior[ny, nx]:
                exterior[ny, nx] = True
                queue.append((ny, nx))
    holes = open_space & ~exterior
    return sum(len(component) >= 80 for component in _components(holes))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    route = _mask(array, _NAVY)
    route_components = [item for item in _components(route) if len(item) >= 80]
    if len(route_components) != 1:
        raise ValueError("expected one intact compact route")
    points = np.asarray(route_components[0])
    span_y = int(points[:, 0].max() - points[:, 0].min() + 1)
    span_x = int(points[:, 1].max() - points[:, 1].min() + 1)
    if min(span_x, span_y) < 42 or max(span_x, span_y) > 125:
        raise ValueError("route dimensions fall outside the public badge design")

    green = _marker_centroid(_mask(array, _GREEN), "green")
    orange = _marker_centroid(_mask(array, _ORANGE), "orange")
    if float(np.hypot(green[0] - orange[0], green[1] - orange[1])) < 75.0:
        raise ValueError("start and finish markers are not distinct")

    holes = _enclosed_background_count(route)
    if holes > 1:
        raise ValueError("ambiguous route topology")
    return "yes" if holes == 1 else "no"
