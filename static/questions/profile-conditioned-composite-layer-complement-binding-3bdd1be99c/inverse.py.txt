"""Pixel-only inverse arm.

Receives a rendered PNG and the public question contract (six 5x5 grids: the
upper band holds PLATE then RESULT, the lower band holds four colored stencils).
It recovers panel geometry from the black frames, reads cell occupancy and ink
color, then solves the complement assignment. It imports no world code and never
sees a scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 5
AMBIGUOUS = "ambiguous"


def _runs(indices: np.ndarray, gap: int = 3) -> list[tuple[int, int]]:
    if indices.size == 0:
        return []
    spans = []
    start = previous = int(indices[0])
    for value in indices[1:]:
        value = int(value)
        if value - previous > gap:
            spans.append((start, previous))
            start = value
        previous = value
    spans.append((start, previous))
    return spans


def _classify(rgb) -> str:
    r, g, b = (float(rgb[0]), float(rgb[1]), float(rgb[2]))
    if max(r, g, b) - min(r, g, b) < 30:
        return "gray"
    if b > r + 30 and b > g + 30:
        return "blue"
    if g > r + 20 and g > b + 20:
        return "green"
    if r > 150 and g > 95:
        return "orange"
    if r > 120 and g < 95 and b < 95:
        return "red"
    return "unknown"


def _read_panel(arr: np.ndarray, box: tuple[int, int, int, int]):
    x0, x1, y0, y1 = box
    dark = arr.max(axis=2) < 40
    mid = (y0 + y1) // 2
    border = 0
    while border < 12 and dark[mid, x0 + border]:
        border += 1
    if border == 0:
        return None
    ix0, ix1 = x0 + border, x1 - border
    iy0, iy1 = y0 + border, y1 - border
    width = (ix1 - ix0 + 1) / GRID
    height = (iy1 - iy0 + 1) / GRID
    if width < 8 or height < 8:
        return None
    cells = {}
    for row in range(GRID):
        for col in range(GRID):
            cx0 = int(round(ix0 + (col + 0.3) * width))
            cx1 = int(round(ix0 + (col + 0.7) * width))
            cy0 = int(round(iy0 + (row + 0.3) * height))
            cy1 = int(round(iy0 + (row + 0.7) * height))
            patch = arr[cy0:cy1, cx0:cx1]
            if patch.size == 0:
                return None
            mean = patch.reshape(-1, 3).mean(axis=0)
            if float(mean.min()) < 200.0:
                cells[(row, col)] = _classify(mean)
    return cells


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    dark = arr.max(axis=2) < 40
    bands = _runs(np.flatnonzero(dark.any(axis=1)))
    if len(bands) != 2:
        return AMBIGUOUS
    panels: list[list[tuple[int, int, int, int]]] = []
    for y0, y1 in bands:
        strip = dark[y0 : y1 + 1]
        row_boxes = []
        for x0, x1 in _runs(np.flatnonzero(strip.any(axis=0))):
            local = np.flatnonzero(strip[:, x0 : x1 + 1].any(axis=1))
            if local.size == 0:
                return AMBIGUOUS
            row_boxes.append((x0, x1, y0 + int(local[0]), y0 + int(local[-1])))
        panels.append(row_boxes)
    if len(panels[0]) != 2 or len(panels[1]) != 4:
        return AMBIGUOUS

    top = [_read_panel(arr, box) for box in panels[0]]
    bottom = [_read_panel(arr, box) for box in panels[1]]
    if any(item is None for item in top + bottom):
        return AMBIGUOUS

    plate = frozenset(top[0])
    result = frozenset(top[1])
    target = result - plate
    if not target:
        return AMBIGUOUS

    hits = []
    for cells in bottom:
        inks = {name for name in cells.values()}
        if len(inks) != 1:
            return AMBIGUOUS
        ink = inks.pop()
        if ink in ("gray", "unknown"):
            return AMBIGUOUS
        if frozenset(cells) == target:
            hits.append(ink)
    if len(hits) != 1:
        return AMBIGUOUS
    return hits[0]
