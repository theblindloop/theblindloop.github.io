"""Composite-layer complement world.

A PLATE grid and a RESULT grid are shown. RESULT is the union of PLATE with a
hidden added layer, drawn in one uniform ink so that provenance of a filled cell
is not visible. Four colored stencils are offered; exactly one of them holds
precisely the cells of the hidden added layer (RESULT minus PLATE). The decision
is the color of that stencil.

Only ``sample_scene``/``render``/``analytic_gold``/``margin``/``is_quarantined``/
``latent_symmetries`` are public. The pixel oracle lives in ``oracle.py`` and
shares no code with this module.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 5
CELL = 36
BORDER = 3
INNER = GRID * CELL
PANEL = INNER + 2 * BORDER
CANVAS_W = 874
CANVAS_H = 566
TOP_Y = 70
BOTTOM_Y = 350
TOP_X = (179, 509)
BOTTOM_X = (26, 238, 450, 662)

BG = (255, 255, 255)
FRAME = (0, 0, 0)
GRIDLINE = (216, 216, 216)
DARK = (95, 95, 95)
TEXT = (120, 120, 120)

COLOR_ORDER = ("red", "blue", "green", "orange")
COLORS = {
    "red": (198, 42, 42),
    "blue": (36, 72, 196),
    "green": (26, 132, 66),
    "orange": (232, 140, 16),
}

_ALL_CELLS = [(r, c) for r in range(GRID) for c in range(GRID)]


def _key(cells) -> frozenset:
    return frozenset((int(a), int(b)) for a, b in cells)


def _sorted(cells) -> list[list[int]]:
    return [list(cell) for cell in sorted((int(a), int(b)) for a, b in cells)]


def sample_scene(seed: int) -> dict:
    """Balanced rejection sampler.

    The correct color cycles with ``seed`` and the near-boundary regime cycles
    every four seeds, so answer identity is decorrelated from slot position,
    ink, and difficulty.
    """
    rng = random.Random(9_176_003 * (int(seed) + 1) + 41)
    wanted_color = COLOR_ORDER[int(seed) % len(COLOR_ORDER)]
    near_boundary = ((int(seed) // len(COLOR_ORDER)) % 2) == 1

    for _ in range(4000):
        plate = rng.sample(_ALL_CELLS, rng.randint(6, 9))
        free = [cell for cell in _ALL_CELLS if cell not in plate]
        added = rng.sample(free, rng.randint(4, 6))
        outside = [cell for cell in free if cell not in added]
        if len(outside) < 5:
            continue

        types = ["swap", "swap", "swap"]
        if near_boundary:
            types[rng.randrange(3)] = rng.choice(["drop", "add"])

        variants = []
        for kind in types:
            cells = set(added)
            if kind in ("swap", "drop"):
                cells.discard(rng.choice(sorted(cells)))
            if kind in ("swap", "add"):
                cells.add(rng.choice(outside))
            variants.append(cells)

        sets = [set(added)] + variants
        keys = {frozenset(item) for item in sets}
        if len(keys) != 4 or min(len(item) for item in variants) < 2:
            continue

        slots = list(range(4))
        correct_slot = rng.randrange(4)
        others = [name for name in COLOR_ORDER if name != wanted_color]
        rng.shuffle(others)
        rng.shuffle(variants)

        stencils = []
        pool = iter(variants)
        spare = iter(others)
        for slot in slots:
            if slot == correct_slot:
                stencils.append({"color": wanted_color, "cells": _sorted(added)})
            else:
                stencils.append({"color": next(spare), "cells": _sorted(next(pool))})

        return {
            "grid": GRID,
            "plate": _sorted(plate),
            "result": _sorted(set(plate) | set(added)),
            "stencils": stencils,
        }
    raise RuntimeError("scene sampling failed to converge")


def _draw_panel(draw: ImageDraw.ImageDraw, x: int, y: int, painted: list) -> None:
    draw.rectangle([x, y, x + PANEL - 1, y + PANEL - 1], fill=BG, outline=FRAME, width=BORDER)
    ix, iy = x + BORDER, y + BORDER
    for k in range(1, GRID):
        draw.line([ix + k * CELL, iy, ix + k * CELL, iy + INNER - 1], fill=GRIDLINE)
        draw.line([ix, iy + k * CELL, ix + INNER - 1, iy + k * CELL], fill=GRIDLINE)
    for (row, col), rgb in painted:
        x0 = ix + int(col) * CELL + 4
        y0 = iy + int(row) * CELL + 4
        draw.rectangle([x0, y0, x0 + CELL - 9, y0 + CELL - 9], fill=rgb)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BG)
    draw = ImageDraw.Draw(image)

    plate = [((int(a), int(b)), DARK) for a, b in scene["plate"]]
    result = [((int(a), int(b)), DARK) for a, b in scene["result"]]
    draw.text((TOP_X[0], TOP_Y - 18), "PLATE", fill=TEXT)
    draw.text((TOP_X[1], TOP_Y - 18), "RESULT", fill=TEXT)
    _draw_panel(draw, TOP_X[0], TOP_Y, plate)
    _draw_panel(draw, TOP_X[1], TOP_Y, result)

    draw.text((BOTTOM_X[0], BOTTOM_Y - 18), "STENCILS", fill=TEXT)
    for slot, stencil in enumerate(scene["stencils"]):
        rgb = COLORS[stencil["color"]]
        painted = [((int(a), int(b)), rgb) for a, b in stencil["cells"]]
        _draw_panel(draw, BOTTOM_X[slot], BOTTOM_Y, painted)
    return image


def analytic_gold(scene: dict) -> str:
    plate = frozenset((int(a), int(b)) for a, b in scene["plate"])
    result = frozenset((int(a), int(b)) for a, b in scene["result"])
    target = result - plate
    hits = [
        item["color"]
        for item in scene["stencils"]
        if frozenset((int(a), int(b)) for a, b in item["cells"]) == target
    ]
    if not target or len(hits) != 1:
        return "ambiguous"
    return hits[0]


def margin(scene: dict) -> float:
    """Smallest cell-wise symmetric difference between the runner-up and truth."""
    target = _key(scene["result"]) - _key(scene["plate"])
    gaps = [
        len(_key(item["cells"]) ^ target)
        for item in scene["stencils"]
        if _key(item["cells"]) != target
    ]
    return float(min(gaps)) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    """Single-cell runner-ups are the minimal visible evidence regime."""
    return margin(scene) <= 1.0 + 1e-9 or analytic_gold(scene) == "ambiguous"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cell listings are sets; their order is a latent alias with no pixel effect."""

    def rebuild(order):
        return {
            "grid": scene["grid"],
            "plate": order(scene["plate"]),
            "result": order(scene["result"]),
            "stencils": [
                {"color": item["color"], "cells": order(item["cells"])}
                for item in scene["stencils"]
            ],
        }

    reverse = lambda cells: [list(cell) for cell in reversed(list(cells))]  # noqa: E731
    rotate = lambda cells: [list(cell) for cell in list(cells)[1:] + list(cells)[:1]]  # noqa: E731
    return [
        ("cell_listing_order_reversed", rebuild(reverse)),
        ("cell_listing_order_rotated", rebuild(rotate)),
    ]
