"""Deterministic renderer and analytic arm for two-group centroid separation."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 600
HEIGHT = 480
PANEL_WIDTH = 250
PANEL_HEIGHT = 180
PANEL_RECTS = (
    (25, 40, 275, 220),
    (325, 40, 575, 220),
    (25, 270, 275, 450),
    (325, 270, 575, 450),
)
BLUE = (35, 103, 180)
ORANGE = (230, 126, 34)
BACKGROUND = (255, 255, 255)
PANEL_FILL = (248, 249, 250)
INK = (32, 38, 45)
DOT_RADIUS = 5
QUARANTINE_GAP = 10.0
LABELS = "ABCD"

_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _four_points(cx: int, cy: int, sx: int, sy: int) -> list[list[int]]:
    return [
        [cx - sx, cy - sy],
        [cx - sx, cy + sy],
        [cx + sx, cy - sy],
        [cx + sx, cy + sy],
    ]


def sample_scene(seed: int) -> dict:
    """Sample four balanced panels; ``seed % 4`` counterbalances the winner."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    smaller = [40, 60, 80]
    rng.shuffle(smaller)
    separations = smaller[:]
    separations.insert(winner, 100)
    direction_bases = ((5, 0), (0, 5), (3, 4), (4, 3), (-3, 4), (-4, 3))
    panels: list[dict] = []
    for distance in separations:
        ux5, uy5 = rng.choice(direction_bases)
        if rng.randrange(2):
            ux5, uy5 = -ux5, -uy5
        scale = distance // 5
        dx, dy = ux5 * scale, uy5 * scale
        mx = 125 + rng.randint(-4, 4)
        my = 90 + rng.randint(-4, 4)
        blue_center = (mx - dx // 2, my - dy // 2)
        orange_center = (mx + dx // 2, my + dy // 2)
        blue_points = _four_points(
            blue_center[0], blue_center[1], rng.randint(7, 10), rng.randint(7, 10)
        )
        orange_points = _four_points(
            orange_center[0], orange_center[1], rng.randint(7, 10), rng.randint(7, 10)
        )
        rng.shuffle(blue_points)
        rng.shuffle(orange_points)
        panels.append({"blue_points": blue_points, "orange_points": orange_points})
    return {"panels": panels}


def _centroid(points: list[list[int]]) -> tuple[float, float]:
    return (
        sum(float(point[0]) for point in points) / len(points),
        sum(float(point[1]) for point in points) / len(points),
    )


def _separations(panels: list[dict]) -> list[float]:
    values = []
    for panel in panels:
        bx, by = _centroid(panel["blue_points"])
        ox, oy = _centroid(panel["orange_points"])
        values.append(math.hypot(bx - ox, by - oy))
    return values


def analytic_gold(scene: dict) -> str:
    values = _separations(scene["panels"])
    return LABELS[max(range(4), key=values.__getitem__)]


def margin(scene: dict) -> float:
    values = sorted(_separations(scene["panels"]), reverse=True)
    return values[0] - values[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Same-color draw order is latent and cannot affect these disjoint disks."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["blue_points"].reverse()
        panel["orange_points"].reverse()
    return [("reverse_same_color_point_order", twin)]


def swap_color_roles(scene: dict) -> dict:
    """A label-preserving, pixel-changing nuisance transform used by tests."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["blue_points"], panel["orange_points"] = (
            panel["orange_points"],
            panel["blue_points"],
        )
    return twin


def translate_all_groups(scene: dict, dx: int, dy: int) -> dict:
    """Translate every group equally inside its panel; centroid gaps are invariant."""
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        for key in ("blue_points", "orange_points"):
            panel[key] = [[x + dx, y + dy] for x, y in panel[key]]
    return twin


def _draw_glyph(draw: ImageDraw.ImageDraw, label: str, x: int, y: int, scale: int = 3) -> None:
    for row, bits in enumerate(_GLYPHS[label]):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (x + col * scale, y + row * scale,
                     x + (col + 1) * scale - 1, y + (row + 1) * scale - 1),
                    fill=INK,
                )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    panels = scene["panels"]
    if len(panels) != 4:
        raise ValueError("scene must contain four panels")
    for index, (panel, rect) in enumerate(zip(panels, PANEL_RECTS, strict=True)):
        x0, y0, x1, y1 = rect
        draw.rectangle(rect, fill=PANEL_FILL, outline=INK, width=3)
        _draw_glyph(draw, LABELS[index], x0 + 5, y0 - 27)
        for key, color in (("blue_points", BLUE), ("orange_points", ORANGE)):
            points = panel[key]
            if len(points) != 4:
                raise ValueError("every color group must contain four dots")
            for px, py in points:
                cx, cy = x0 + int(px), y0 + int(py)
                draw.ellipse(
                    (cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS),
                    fill=color,
                )
    return image
