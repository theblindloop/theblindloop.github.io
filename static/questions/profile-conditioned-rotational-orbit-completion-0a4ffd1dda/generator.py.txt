"""Sampler, renderer, and analytic arm for rotational_orbit_completion."""

from __future__ import annotations

import copy
import math
import random

import numpy as np
from PIL import Image, ImageDraw

COLORS: tuple[str, ...] = ("red", "blue", "green", "orange")
RGB = {
    "red": (205, 67, 85),
    "blue": (45, 103, 210),
    "green": (38, 151, 105),
    "orange": (232, 132, 39),
}
BACKGROUND = (248, 247, 242)
DARK = (38, 46, 57)
CENTER_GRAY = (118, 126, 138)
QUARANTINE_MARGIN_PX = 12.0
_TURN = 2.0 * math.pi / 3.0


def _rotate(point: list[float] | tuple[float, float], center: list[float], angle: float) -> np.ndarray:
    px, py = float(point[0]) - center[0], float(point[1]) - center[1]
    cosine, sine = math.cos(angle), math.sin(angle)
    return np.array(
        [center[0] + cosine * px - sine * py, center[1] + sine * px + cosine * py],
        dtype=np.float64,
    )


def _orbit_score(points: list[list[float]], center: list[float]) -> float:
    """Mean of the four worst rotation-to-set residuals, in pixels."""

    array = np.asarray(points, dtype=np.float64)
    residuals: list[float] = []
    for angle in (_TURN, 2.0 * _TURN):
        shifted = array - np.asarray(center, dtype=np.float64)
        cosine, sine = math.cos(angle), math.sin(angle)
        rotated = np.column_stack(
            (
                center[0] + cosine * shifted[:, 0] - sine * shifted[:, 1],
                center[1] + sine * shifted[:, 0] + cosine * shifted[:, 1],
            )
        )
        distances = np.linalg.norm(rotated[:, None, :] - array[None, :, :], axis=2)
        residuals.extend(distances.min(axis=1).tolist())
    worst = sorted(residuals, reverse=True)[:4]
    return float(sum(worst) / len(worst))


def _ranked_scores(scene: dict) -> list[tuple[float, str]]:
    center = scene["center"]
    points = scene["points"]
    markers = scene["markers"]
    return sorted(
        (_orbit_score(points + [marker["xy"]], center), str(marker["color"]))
        for marker in markers
    )


def _far_enough(point: list[float], others: list[list[float]], clearance: float) -> bool:
    return all(math.hypot(point[0] - other[0], point[1] - other[1]) >= clearance for other in others)


def sample_scene(seed: int) -> dict:
    """Sample a randomized incomplete threefold constellation.

    Consecutive quartets share all geometry. Only the mapping from the four
    public colors to the four candidate positions changes, so answer identity is
    exactly counterbalanced against position, size, count, and total ink.
    """

    geometry_id = int(seed) // 4
    answer_index = int(seed) % len(COLORS)
    rng = random.Random(0x3F01D + geometry_id * 104729)
    width = height = 512

    for _layout_attempt in range(800):
        center = [float(rng.randint(190, 322)), float(rng.randint(190, 322))]
        full: list[list[float]] = []
        for _orbit in range(6):
            radius = rng.uniform(58.0, 166.0)
            phase = rng.uniform(0.0, _TURN)
            for turn in range(3):
                angle = phase + turn * _TURN
                full.append(
                    [
                        round(center[0] + radius * math.cos(angle), 5),
                        round(center[1] + radius * math.sin(angle), 5),
                    ]
                )
        if not all(18.0 <= p[0] <= width - 18.0 and 18.0 <= p[1] <= height - 18.0 for p in full):
            continue
        if any(
            math.hypot(full[i][0] - full[j][0], full[i][1] - full[j][1]) < 24.0
            for i in range(len(full))
            for j in range(i)
        ):
            continue

        missing_index = rng.randrange(len(full))
        missing = full[missing_index]
        points = [point for index, point in enumerate(full) if index != missing_index]
        positions = [list(missing)]

        # A controlled near runner-up supplies several difficulty bands without
        # coupling difficulty to answer color.
        near_distance = (34.0, 48.0, 64.0)[geometry_id % 3]
        for _ in range(300):
            angle = rng.uniform(0.0, 2.0 * math.pi)
            proposal = [
                round(missing[0] + near_distance * math.cos(angle), 5),
                round(missing[1] + near_distance * math.sin(angle), 5),
            ]
            if not (18.0 <= proposal[0] <= width - 18.0 and 18.0 <= proposal[1] <= height - 18.0):
                continue
            if math.hypot(proposal[0] - center[0], proposal[1] - center[1]) < 30.0:
                continue
            if _far_enough(proposal, points, 27.0):
                positions.append(proposal)
                break
        if len(positions) != 2:
            continue

        for _decoy in range(2):
            for _ in range(500):
                radius = rng.uniform(48.0, 174.0)
                angle = rng.uniform(0.0, 2.0 * math.pi)
                proposal = [
                    round(center[0] + radius * math.cos(angle), 5),
                    round(center[1] + radius * math.sin(angle), 5),
                ]
                if not (18.0 <= proposal[0] <= width - 18.0 and 18.0 <= proposal[1] <= height - 18.0):
                    continue
                if _far_enough(proposal, points, 27.0) and _far_enough(proposal, positions, 31.0):
                    positions.append(proposal)
                    break
        if len(positions) != 4:
            continue

        assigned = [COLORS[(answer_index + offset) % 4] for offset in range(4)]
        markers = [
            {"color": color, "xy": position}
            for color, position in zip(assigned, positions, strict=True)
        ]
        rng.shuffle(markers)
        scene = {
            "width": width,
            "height": height,
            "center": center,
            "points": points,
            "markers": markers,
            "dot_radius": 6,
            "marker_radius": 10,
            "center_radius": 13,
            "center_stroke": 4,
            "background": list(BACKGROUND),
            "dark": list(DARK),
            "center_gray": list(CENTER_GRAY),
        }
        ranked = _ranked_scores(scene)
        if ranked[0][1] == COLORS[answer_index] and ranked[1][0] - ranked[0][0] >= 14.0:
            return scene
    raise RuntimeError(f"could not sample an unambiguous rotational constellation for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Render the visible evidence with deterministic supersampling."""

    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    dot_radius = int(scene["dot_radius"]) * scale
    for px, py in scene["points"]:
        x, y = int(round(px * scale)), int(round(py * scale))
        draw.ellipse([x - dot_radius, y - dot_radius, x + dot_radius, y + dot_radius], fill=tuple(scene["dark"]))

    cx, cy = scene["center"]
    center_radius = int(scene["center_radius"]) * scale
    center_stroke = int(scene["center_stroke"]) * scale
    box = [cx * scale - center_radius, cy * scale - center_radius, cx * scale + center_radius, cy * scale + center_radius]
    draw.ellipse(box, outline=tuple(scene["center_gray"]), width=center_stroke)

    marker_radius = int(scene["marker_radius"]) * scale
    halo_radius = marker_radius + 3 * scale
    for marker in scene["markers"]:
        px, py = marker["xy"]
        x, y = int(round(px * scale)), int(round(py * scale))
        draw.ellipse([x - halo_radius, y - halo_radius, x + halo_radius, y + halo_radius], fill=tuple(scene["background"]))
        draw.ellipse([x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius], fill=RGB[marker["color"]])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _ranked_scores(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _ranked_scores(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_unordered_dark_point_list", twin)]
