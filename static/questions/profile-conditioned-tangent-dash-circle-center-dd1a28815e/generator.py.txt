"""Deterministic renderer for a circle implied by disconnected tangent dashes."""

from __future__ import annotations

import math
import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ANTIALIAS = 4
BACKGROUND = (248, 246, 239)
DARK = (31, 38, 48)
PALETTE = {
    "red": (214, 54, 63),
    "blue": (41, 105, 200),
    "green": (39, 151, 88),
    "orange": (235, 137, 35),
}
COLORS = tuple(PALETTE)
QUARANTINE_GAP = 8.0


def _candidate_layout(seed: int, rng: random.Random) -> list[dict]:
    bases = [(146.0, 146.0), (366.0, 146.0), (366.0, 366.0), (146.0, 366.0)]
    positions = [
        (x + rng.uniform(-10.0, 10.0), y + rng.uniform(-10.0, 10.0))
        for x, y in bases
    ]
    shift = (seed // 4) % 4
    shade = (-8, 0, 8)[(seed // 4) % 3]
    rings = []
    for color_index, color in enumerate(COLORS):
        x, y = positions[(color_index + shift) % 4]
        rgb = [max(0, min(255, channel + shade)) for channel in PALETTE[color]]
        rings.append(
            {"color": color, "x": round(x, 3), "y": round(y, 3), "rgb": rgb}
        )
    return rings


def _point_segment_distance(
    point: tuple[float, float], segment: list[float]
) -> float:
    px, py = point
    x1, y1, x2, y2 = segment
    dx, dy = x2 - x1, y2 - y1
    scale = dx * dx + dy * dy
    if scale <= 0.0:
        return math.hypot(px - x1, py - y1)
    t = max(0.0, min(1.0, ((px - x1) * dx + (py - y1) * dy) / scale))
    return math.hypot(px - (x1 + t * dx), py - (y1 + t * dy))


def _segments_intersect(left: list[float], right: list[float]) -> bool:
    def orient(ax: float, ay: float, bx: float, by: float, cx: float, cy: float) -> float:
        return (bx - ax) * (cy - ay) - (by - ay) * (cx - ax)

    a, b, c, d = left[:2], left[2:], right[:2], right[2:]
    return (
        orient(*a, *b, *c) * orient(*a, *b, *d) <= 0.0
        and orient(*c, *d, *a) * orient(*c, *d, *b) <= 0.0
    )


def _segment_gap(left: list[float], right: list[float]) -> float:
    if _segments_intersect(left, right):
        return 0.0
    endpoints_left = [(left[0], left[1]), (left[2], left[3])]
    endpoints_right = [(right[0], right[1]), (right[2], right[3])]
    return min(
        *(_point_segment_distance(point, right) for point in endpoints_left),
        *(_point_segment_distance(point, left) for point in endpoints_right),
    )


def _make_dashes(
    center: tuple[float, float], rings: list[dict], rng: random.Random
) -> list[list[float]]:
    cx, cy = center
    for _configuration_attempt in range(1200):
        radius = rng.uniform(80.0, 90.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        dashes: list[list[float]] = []
        valid = True
        for index in range(10):
            theta = phase + index * (2.0 * math.pi / 10.0) + rng.uniform(-0.10, 0.10)
            nx, ny = math.cos(theta), math.sin(theta)
            tx, ty = -ny, nx
            length = rng.uniform(28.0, 34.0)
            along = rng.uniform(-74.0, 74.0)
            mx = cx + radius * nx + along * tx
            my = cy + radius * ny + along * ty
            candidate = [
                mx - 0.5 * length * tx,
                my - 0.5 * length * ty,
                mx + 0.5 * length * tx,
                my + 0.5 * length * ty,
            ]
            if not all(18.0 <= value <= 494.0 for value in candidate):
                valid = False
                break
            if any(
                _point_segment_distance((ring["x"], ring["y"]), candidate) < 25.0
                for ring in rings
            ):
                valid = False
                break
            if any(_segment_gap(candidate, prior) < 12.0 for prior in dashes):
                valid = False
                break
            dashes.append([round(value, 3) for value in candidate])
        if valid and len(dashes) == 10:
            return dashes
    raise RuntimeError("could not sample a separated tangent-dash configuration")


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene solely from ``seed``."""

    rng = random.Random((int(seed) ^ 0x5A17C1E) & 0xFFFFFFFFFFFFFFFF)
    rings = _candidate_layout(int(seed), rng)
    answer_color = COLORS[int(seed) % len(COLORS)]
    answer_ring = next(ring for ring in rings if ring["color"] == answer_color)
    dashes = _make_dashes((answer_ring["x"], answer_ring["y"]), rings, rng)
    scene = {"dash_segments": dashes, "candidate_rings": rings}
    if is_quarantined(scene):
        raise RuntimeError("deterministic sampler produced an ambiguous scene")
    return scene


def _line_distance(point: tuple[float, float], segment: list[float]) -> float:
    x, y = point
    x1, y1, x2, y2 = segment
    dx, dy = x2 - x1, y2 - y1
    length = math.hypot(dx, dy)
    if length <= 0.0:
        raise ValueError("zero-length dash")
    return abs(dy * (x - x1) - dx * (y - y1)) / length


def _support_spreads(scene: dict) -> list[tuple[float, str]]:
    segments = scene["dash_segments"]
    rings = scene["candidate_rings"]
    values = []
    for ring in rings:
        distances = [_line_distance((ring["x"], ring["y"]), item) for item in segments]
        values.append((float(np.std(np.asarray(distances, dtype=float))), ring["color"]))
    return sorted(values)


def analytic_gold(scene: dict) -> str:
    return _support_spreads(scene)[0][1]


def margin(scene: dict) -> float:
    scores = _support_spreads(scene)
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Dash storage order is a genuine render- and label-invariant alias."""

    twin = deepcopy(scene)
    twin["dash_segments"] = list(reversed(twin["dash_segments"]))
    return [("reverse_dash_storage_order", twin)]


def render(scene: dict) -> Image.Image:
    segments = scene["dash_segments"]
    rings = scene["candidate_rings"]
    scale = ANTIALIAS
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x1, y1, x2, y2 in segments:
        draw.line(
            (x1 * scale, y1 * scale, x2 * scale, y2 * scale),
            fill=DARK,
            width=5 * scale,
        )

    for ring in rings:
        x, y = ring["x"] * scale, ring["y"] * scale
        radius = 11 * scale
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            outline=tuple(ring["rgb"]),
            width=5 * scale,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
