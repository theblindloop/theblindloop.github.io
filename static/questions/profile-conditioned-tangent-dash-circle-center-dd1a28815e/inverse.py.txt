"""Independent pixel-only inverse for the tangent-support constellation."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_PALETTE = {
    "red": np.asarray((214, 54, 63), dtype=float),
    "blue": np.asarray((41, 105, 200), dtype=float),
    "green": np.asarray((39, 151, 88), dtype=float),
    "orange": np.asarray((235, 137, 35), dtype=float),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        stack = [(int(y0), int(x0))]
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    yy, xx = y + dy, x + dx
                    if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                        seen[yy, xx] = True
                        stack.append((yy, xx))
        groups.append(np.asarray(pixels, dtype=float))
    return groups


def _colored_centers(rgb: np.ndarray) -> list[tuple[str, float, float]]:
    points = []
    pixels = rgb.astype(float)
    for name, target in _PALETTE.items():
        distance2 = np.sum((pixels - target) ** 2, axis=2)
        ys, xs = np.nonzero(distance2 < 58.0**2)
        if len(xs) < 60:
            raise ValueError(f"missing {name} candidate ring")
        points.append((name, float(xs.mean()), float(ys.mean())))
    return points


def _dash_lines(rgb: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    dark = np.max(rgb, axis=2) < 92
    groups = [group for group in _components(dark) if len(group) >= 45]
    if len(groups) != 10:
        raise ValueError(f"expected 10 disconnected dark dashes, found {len(groups)}")
    lines = []
    for group in groups:
        xy = group[:, ::-1]
        center = xy.mean(axis=0)
        centered = xy - center
        covariance = centered.T @ centered
        values, vectors = np.linalg.eigh(covariance)
        direction = vectors[:, int(np.argmax(values))]
        if float(values.max()) < 8.0 * float(values.min() + 1e-9):
            raise ValueError("dark component is not line-like")
        lines.append((center, direction))
    return lines


def _spread(point: tuple[float, float], lines: list[tuple[np.ndarray, np.ndarray]]) -> float:
    p = np.asarray(point, dtype=float)
    distances = []
    for center, direction in lines:
        offset = p - center
        distances.append(abs(float(direction[0] * offset[1] - direction[1] * offset[0])))
    return float(np.std(np.asarray(distances, dtype=float)))


def decision_from_image(image: Image.Image) -> str:
    """Recover colored rings and dash support lines from final RGB pixels only."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers = _colored_centers(rgb)
    lines = _dash_lines(rgb)
    scores = sorted((_spread((x, y), lines), name) for name, x, y in centers)
    if not math.isfinite(scores[0][0]) or scores[1][0] - scores[0][0] < 6.0:
        raise ValueError("ambiguous tangent-circle center")
    return scores[0][1]
