"""Independent pixel-only inverse for the color between blue and yellow arcs."""

from __future__ import annotations

import numpy as np
from PIL import Image


_COLOR_NAMES = ("red", "orange", "yellow", "green", "blue")


def _semantic_masks(image: Image.Image) -> dict[str, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red_channel = arr[:, :, 0]
    green_channel = arr[:, :, 1]
    blue_channel = arr[:, :, 2]
    return {
        "red": (red_channel > 165) & (green_channel < 95) & (blue_channel < 125) & (red_channel > green_channel + 85),
        "orange": (red_channel > 190) & (green_channel >= 95) & (green_channel < 176) & (blue_channel < 105) & (red_channel > green_channel + 60) & (green_channel > blue_channel + 50),
        "yellow": (red_channel > 190) & (green_channel >= 176) & (blue_channel < 125) & (red_channel - green_channel < 50),
        "green": (green_channel > 120) & (red_channel < 125) & (blue_channel < 155) & (green_channel > red_channel + 38),
        "blue": (blue_channel > 150) & (red_channel < 125) & (green_channel < 160) & (blue_channel > green_channel + 42),
    }


def _measure(mask: np.ndarray, name: str) -> dict[str, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 650:
        raise ValueError(f"required {name} band is absent or too small")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    width = right - left + 1
    height = bottom - top + 1
    if width < 70 or height < 30 or width / height < 1.45:
        raise ValueError(f"{name} pixels do not form a clear semicircular band")
    return {
        "width": float(width),
        "height": float(height),
        "center_x": float((left + right) / 2.0),
        "bottom": float(bottom),
        "area": float(len(xs)),
    }


def decision_from_image(image: Image.Image) -> str:
    """Recover radial order and return the unique color between blue and yellow."""
    masks = _semantic_masks(image)
    measurements = {name: _measure(masks[name], name) for name in _COLOR_NAMES}

    centers_x = [value["center_x"] for value in measurements.values()]
    bottoms = [value["bottom"] for value in measurements.values()]
    if max(centers_x) - min(centers_x) > 8.0:
        raise ValueError("colored bands are not visibly concentric")
    if max(bottoms) - min(bottoms) > 10.0:
        raise ValueError("colored arcs do not share a clear baseline")

    outer_to_inner = sorted(_COLOR_NAMES, key=lambda name: measurements[name]["width"], reverse=True)
    widths = [measurements[name]["width"] for name in outer_to_inner]
    radial_gaps = [widths[index] - widths[index + 1] for index in range(4)]
    if min(radial_gaps) < 25.0:
        raise ValueError("adjacent radial ranks are visually ambiguous")

    blue_rank = outer_to_inner.index("blue")
    yellow_rank = outer_to_inner.index("yellow")
    if abs(blue_rank - yellow_rank) != 2:
        raise ValueError("blue and yellow do not visibly delimit one intervening band")
    answer = outer_to_inner[(blue_rank + yellow_rank) // 2]
    if answer not in ("red", "orange", "green"):
        raise ValueError("intervening band is outside the public answer set")
    return answer
