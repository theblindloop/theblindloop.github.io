"""Deterministic latent sampler and raster renderer for a rainbow-order prior conflict."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


COLORS = ("red", "orange", "yellow", "green", "blue")
CANONICAL_ORDER = ("red", "orange", "yellow", "green", "blue")

# Twelve outer-to-inner orders. Blue and yellow are always exactly two ranks apart;
# red/orange/green is the unique intervening color. Each answer occurs four times
# with the same absolute-rank multiset, blocking position and arc-ink shortcuts.
BAND_ORDERS = (
    ("red", "orange", "yellow", "green", "blue"),
    ("blue", "green", "yellow", "red", "orange"),
    ("red", "blue", "green", "yellow", "orange"),
    ("orange", "yellow", "green", "blue", "red"),
    ("blue", "red", "yellow", "orange", "green"),
    ("orange", "yellow", "red", "blue", "green"),
    ("green", "blue", "red", "yellow", "orange"),
    ("orange", "green", "yellow", "red", "blue"),
    ("blue", "orange", "yellow", "red", "green"),
    ("red", "yellow", "orange", "blue", "green"),
    ("green", "blue", "orange", "yellow", "red"),
    ("red", "green", "yellow", "orange", "blue"),
)

_PALETTES = (
    {"red": (216, 48, 61), "orange": (239, 126, 35), "yellow": (244, 202, 45), "green": (42, 158, 88), "blue": (47, 103, 194)},
    {"red": (228, 58, 66), "orange": (232, 116, 27), "yellow": (235, 194, 38), "green": (37, 145, 82), "blue": (42, 91, 184)},
    {"red": (204, 42, 57), "orange": (244, 139, 42), "yellow": (248, 211, 55), "green": (48, 168, 94), "blue": (53, 112, 207)},
    {"red": (222, 52, 72), "orange": (236, 131, 31), "yellow": (241, 199, 50), "green": (33, 151, 76), "blue": (45, 98, 199)},
)
_BACKGROUNDS = ((248, 246, 239), (239, 247, 250), (246, 241, 249), (241, 248, 240))
_DECORATION_COLORS = ((83, 91, 99), (91, 84, 78), (75, 91, 87), (88, 82, 94))
_SUPERSAMPLE = 4


def sample_scene(seed: int) -> dict:
    """Sample one scene; each consecutive twelve seeds share all nuisances."""
    seed = int(seed)
    group = seed // len(BAND_ORDERS)
    order = list(BAND_ORDERS[seed % len(BAND_ORDERS)])
    rng = random.Random(0x71A1B0A + group * 1_000_033)

    # Every fourth matched group is a declared near-clearance quarantine set.
    gap = rng.randint(2, 3) if group % 4 == 0 else rng.randint(7, 12)
    palette = rng.choice(_PALETTES)
    decorations = [
        {"kind": "diamond", "x": rng.randint(45, 90), "y": rng.randint(46, 90), "size": rng.randint(5, 8)},
        {"kind": "drop", "x": rng.randint(420, 468), "y": rng.randint(55, 105), "size": rng.randint(6, 9)},
        {"kind": "diamond", "x": rng.randint(424, 474), "y": rng.randint(270, 310), "size": rng.randint(4, 7)},
    ]
    return {
        "width": 512,
        "height": 384,
        "background_rgb": list(rng.choice(_BACKGROUNDS)),
        "center_x": rng.randint(247, 265),
        "center_y": rng.randint(326, 340),
        "outer_radius": rng.randint(194, 207),
        "band_width": rng.randint(22, 27),
        "band_gap": gap,
        "band_palette": {name: list(palette[name]) for name in COLORS},
        "decoration_rgb": list(rng.choice(_DECORATION_COLORS)),
        "decorations": decorations,
        "band_order": order,
    }


def _scaled_box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in values)  # type: ignore[return-value]


def _draw_decoration(draw: ImageDraw.ImageDraw, item: dict, color: tuple[int, int, int], scale: int) -> None:
    x = int(item["x"]) * scale
    y = int(item["y"]) * scale
    size = int(item["size"]) * scale
    if item["kind"] == "diamond":
        draw.polygon(((x, y - size), (x + size, y), (x, y + size), (x - size, y)), fill=color)
    elif item["kind"] == "drop":
        draw.polygon(((x, y - size), (x + size, y + size), (x, y + size // 2), (x - size, y + size)), fill=color)
    else:
        raise ValueError(f"unknown decoration kind {item['kind']!r}")


def render(scene: dict) -> Image.Image:
    """Render five separated semicircular color bands from public scene fields."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background_rgb"])
    center_x = int(scene["center_x"])
    center_y = int(scene["center_y"])
    outer_radius = int(scene["outer_radius"])
    band_width = int(scene["band_width"])
    band_gap = int(scene["band_gap"])
    palette = {name: tuple(value) for name, value in scene["band_palette"].items()}
    decoration_rgb = tuple(scene["decoration_rgb"])
    decorations = list(scene["decorations"])
    order = list(scene["band_order"])

    if sorted(order) != sorted(COLORS):
        raise ValueError("band_order must contain each declared color exactly once")
    scale = _SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    for item in decorations:
        _draw_decoration(draw, item, decoration_rgb, scale)

    step = band_width + band_gap
    for rank, color_name in enumerate(order):
        radius = outer_radius - rank * step
        if radius <= band_width:
            raise ValueError("inner rainbow band has nonpositive usable radius")
        box = _scaled_box(
            (center_x - radius, center_y - radius, center_x + radius, center_y + radius),
            scale,
        )
        draw.arc(box, start=180, end=360, fill=palette[color_name], width=band_width * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    order = list(scene["band_order"])
    blue = order.index("blue")
    yellow = order.index("yellow")
    if abs(blue - yellow) != 2:
        raise ValueError("blue and yellow must have exactly one intervening band")
    return order[(blue + yellow) // 2]


def margin(scene: dict) -> float:
    """Minimum declared white radial clearance between adjacent colored bands."""
    return float(scene["band_gap"])


def is_quarantined(scene: dict) -> bool:
    return int(scene["band_gap"]) < 5 or int(scene["band_width"]) < 18


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All declared fields affect pixels, so there is no intentional latent alias.
    return []
