"""Independent final-PNG topology inverse; no latent scene data is used."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas")
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    delta = np.linalg.norm(array - background, axis=2)
    trace_raw = delta > 58.0
    groups = [group for group in _components(trace_raw) if len(group) >= 80]
    if len(groups) != 1:
        raise ValueError("the dark trace is missing, disconnected, or ambiguous")

    trace = np.zeros(trace_raw.shape, dtype=bool)
    ys, xs = zip(*groups[0])
    ys_array, xs_array = np.asarray(ys), np.asarray(xs)
    trace[ys_array, xs_array] = True
    y0, y1 = int(ys_array.min()), int(ys_array.max())
    x0, x1 = int(xs_array.min()), int(xs_array.max())
    box_width, box_height = x1 - x0 + 1, y1 - y0 + 1
    if not (104 <= box_width <= 121 and 24 <= box_height <= 33):
        raise ValueError("paperclip trace geometry is corrupted")
    if not (650 <= len(groups[0]) <= 1800):
        raise ValueError("paperclip ink amount is implausible")

    # Work in a small crop around the connected trace. The only free component
    # touching this crop's border is the unbounded canvas background; every
    # other sufficiently large component is a visible enclosed white pocket.
    pad = 3
    crop = ~trace[y0 - pad:y1 + pad + 1, x0 - pad:x1 + pad + 1]
    free_groups = _components(crop)
    crop_height, crop_width = crop.shape
    bounded: list[list[tuple[int, int]]] = []
    for group in free_groups:
        touches_border = any(
            y == 0 or x == 0 or y == crop_height - 1 or x == crop_width - 1
            for y, x in group
        )
        if touches_border:
            continue
        ys_local, xs_local = zip(*group)
        region_width = max(xs_local) - min(xs_local) + 1
        region_height = max(ys_local) - min(ys_local) + 1
        if len(group) >= 85 and region_width >= 6 and region_height >= 14:
            bounded.append(group)
    count = len(bounded)
    if count not in (1, 2, 3, 4):
        raise ValueError("visible enclosure count is outside the public answer set")
    areas = sorted(len(group) for group in bounded)
    if areas[0] < 100:
        raise ValueError("an enclosed pocket is too small to resolve")
    return str(count)
