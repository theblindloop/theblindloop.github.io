"""Deterministic renderer and analytic arm for paperclip pocket counting."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


PALETTES = (
    ((34, 76, 154), (250, 248, 242)),
    ((126, 48, 122), (248, 250, 246)),
    ((10, 112, 108), (252, 248, 244)),
    ((174, 70, 24), (247, 250, 252)),
)

# These anchors move the short trace around the canvas while keeping it away
# from the deterministic descriptor grid's halo boundaries.
ANCHORS = ((84, 213), (212, 213), (340, 213), (84, 341), (212, 341), (340, 85))


def sample_scene(seed: int) -> dict:
    """Consecutive groups of four share style while varying visible topology."""
    pocket_count = 1 + (int(seed) % 4)
    style_rng = random.Random(47_000 + int(seed) // 4)
    ink, background = PALETTES[style_rng.randrange(len(PALETTES))]
    anchor_x, anchor_y = ANCHORS[style_rng.randrange(len(ANCHORS))]
    bridge_rng = random.Random(83_000 + int(seed))
    bridge_positions = sorted(bridge_rng.sample(range(1, 7), pocket_count - 1))
    return {
        "pocket_count": pocket_count,
        "bridge_positions": bridge_positions,
        "left": anchor_x + style_rng.randint(0, 3),
        "top": anchor_y + style_rng.randint(0, 3),
        "clip_width": style_rng.randint(108, 116),
        "pocket_height": style_rng.randint(26, 29),
        "stroke_width": 4,
        "ink_color": list(ink),
        "background": list(background),
        "width": 512,
        "height": 512,
    }


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    ink = tuple(int(value) for value in scene["ink_color"])
    left, top = int(scene["left"]), int(scene["top"])
    clip_width = int(scene["clip_width"])
    pocket_height = int(scene["pocket_height"])
    stroke_width = int(scene["stroke_width"])
    pocket_count = int(scene["pocket_count"])
    bridge_positions = [int(value) for value in scene["bridge_positions"]]
    if pocket_count != len(set(bridge_positions)) + 1:
        raise ValueError("pocket_count disagrees with the visible bridge set")
    if not set(bridge_positions) <= set(range(1, 7)):
        raise ValueError("bridge position is outside the paperclip")

    canvas = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(canvas)
    box = (
        left * scale,
        top * scale,
        (left + clip_width) * scale,
        (top + pocket_height) * scale,
    )
    draw.rounded_rectangle(
        box,
        radius=(pocket_height * scale) // 2,
        outline=ink,
        width=stroke_width * scale,
    )
    inner_top = (top + 1) * scale
    inner_bottom = (top + pocket_height - 1) * scale
    for slot in bridge_positions:
        x = round((left + clip_width * slot / 7.0) * scale)
        draw.line(
            (x, inner_top, x, inner_bottom),
            fill=ink,
            width=stroke_width * scale,
        )
    return canvas.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["pocket_count"]))


def margin(scene: dict) -> float:
    # Runner-up margin: the narrowest nominal pocket width after subtracting ink.
    positions = [0, *sorted(int(value) for value in scene["bridge_positions"]), 7]
    minimum_slots = min(right - left for left, right in zip(positions, positions[1:]))
    return float(scene["clip_width"]) * minimum_slots / 7.0 - 2.0 * float(scene["stroke_width"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0 or float(scene["pocket_height"]) - 2.0 * float(scene["stroke_width"]) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["bridge_positions"] = list(reversed(scene["bridge_positions"]))
    return [("reverse_bridge_traversal", transformed)]
