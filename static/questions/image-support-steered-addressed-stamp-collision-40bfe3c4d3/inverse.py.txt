"""Independent image-only inverse arm for addressed paint stamps.

The arm finds the two saturated stamp frames, independently finds the two dark
address markers, binds each marker to its nearest frame, recovers its quadrant,
orders the frames left to right, and replays the two writes. It imports no
renderer, scene, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    remaining = {tuple(map(int, point)) for point in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        points = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    points.append(neighbor)
        if len(points) >= minimum:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        [
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    spread = array.max(axis=2) - array.min(axis=2)
    contrast = np.linalg.norm(array - background, axis=2)
    saturated = (spread >= 52) & (contrast >= 62) & (array.max(axis=2) >= 105)
    dark = (array.max(axis=2) <= 92) & (spread <= 45) & (contrast >= 100)
    return saturated, dark


def _center(points: np.ndarray) -> tuple[float, float]:
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _bbox(points: np.ndarray) -> tuple[float, float]:
    width = float(points[:, 1].max() - points[:, 1].min() + 1)
    height = float(points[:, 0].max() - points[:, 0].min() + 1)
    return width, height


def _address(frame: np.ndarray, marker: np.ndarray) -> tuple[str, float, tuple[float, float]]:
    fx, fy = _center(frame)
    mx, my = _center(marker)
    dx, dy = mx - fx, my - fy
    distance = float(np.hypot(dx, dy))
    confidence = min(abs(dx), abs(dy))
    if not 3.0 <= distance <= 10.5 or confidence < 2.2:
        raise OracleAbstention("dark marker is not decisively inside one address quadrant")
    vertical = "top" if dy < 0 else "bottom"
    horizontal = "left" if dx < 0 else "right"
    return f"{vertical}_{horizontal}", confidence, (fx, fy)


def decision_from_image(image: Image.Image) -> str:
    saturated, dark = _masks(image)
    frames = _components(saturated, minimum=80)
    markers = _components(dark, minimum=9)
    frames = [group for group in frames if 18 <= _bbox(group)[0] <= 32 and 18 <= _bbox(group)[1] <= 32]
    markers = [group for group in markers if 3 <= _bbox(group)[0] <= 9 and 3 <= _bbox(group)[1] <= 9]
    if len(frames) != 2:
        raise OracleAbstention(f"expected two saturated stamp frames, found {len(frames)}")
    if len(markers) != 2:
        raise OracleAbstention(f"expected two dark address markers, found {len(markers)}")

    frames.sort(key=lambda group: _center(group)[0])
    available = list(markers)
    decoded: list[str] = []
    frame_centers: list[tuple[float, float]] = []
    for frame in frames:
        fx, fy = _center(frame)
        nearest_index = min(
            range(len(available)),
            key=lambda index: np.hypot(
                _center(available[index])[0] - fx,
                _center(available[index])[1] - fy,
            ),
        )
        nearest = available.pop(nearest_index)
        if np.hypot(_center(nearest)[0] - fx, _center(nearest)[1] - fy) > 11.0:
            raise OracleAbstention("an address marker cannot be bound to a stamp")
        address, _, center = _address(frame, nearest)
        decoded.append(address)
        frame_centers.append(center)
    if frame_centers[1][0] - frame_centers[0][0] < 70.0:
        raise OracleAbstention("stamp order is not visibly separated left to right")
    if abs(frame_centers[1][1] - frame_centers[0][1]) > 18.0:
        raise OracleAbstention("stamps do not form one ordered row")
    return "overwrite" if decoded[0] == decoded[1] else "separate"
