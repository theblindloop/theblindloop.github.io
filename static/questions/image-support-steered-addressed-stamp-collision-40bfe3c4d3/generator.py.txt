"""Deterministic renderer and analytic arm for addressed paint stamps."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
CELLS = ("top_left", "top_right", "bottom_left", "bottom_right")
PALE_INK = "#cdd2d8"
GRID_INK = "#a8afb8"
DARK_INK = "#2c3040"
PANEL_INK = "#f7f4ec"
PALE_BLUE = "#dbe7ee"
BACKGROUNDS = ("#ffffff", "#fffdf7")
PALETTES = (
    ("#1769aa", "#c13f75"),
    ("#07866f", "#c2541b"),
    ("#7042b5", "#ad7200"),
    ("#b13f3f", "#167a9c"),
)


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; adjacent seeds share all nuisance fields."""

    pair = seed // 2
    first_index = pair % 4
    overwrite = seed % 2 == 0
    if overwrite:
        second_index = first_index
    else:
        second_index = (first_index + 1 + (pair // 4) % 3) % 4

    nuisance = random.Random(pair * 1_000_003 + 73)
    decorations = ["circle", "square", "diamond", "cross"]
    nuisance.shuffle(decorations)
    return {
        "first_cell": CELLS[first_index],
        "second_cell": CELLS[second_index],
        "left_cell": 1 + (pair * 3 + pair // 5) % 4,
        "row_cell": 1 + (pair * 5 + pair // 7) % 6,
        "palette_index": pair % len(PALETTES),
        "glyph_radius": 13 + pair % 2,
        "halo_style": "double" if nuisance.randrange(2) else "single",
        "decorations": decorations,
        "background": BACKGROUNDS[(pair // 3) % len(BACKGROUNDS)],
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
    }


def _scaled(points: Iterable[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def _centers(scene: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    left_cell = int(scene["left_cell"])
    row_cell = int(scene["row_cell"])
    cell_w = width / 8.0
    cell_h = height / 8.0
    first = ((left_cell + 0.5) * cell_w, (row_cell + 0.5) * cell_h)
    second = ((left_cell + 2.5) * cell_w, (row_cell + 0.5) * cell_h)
    return first, second


def _marker_offset(cell: str, radius: int) -> tuple[float, float]:
    offset = radius * 0.34
    mapping = {
        "top_left": (-offset, -offset),
        "top_right": (offset, -offset),
        "bottom_left": (-offset, offset),
        "bottom_right": (offset, offset),
    }
    return mapping[cell]


def _draw_stamp(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    cell: str,
    color: str,
    radius: int,
    kind: str,
    halo_style: str,
    scale: int,
) -> None:
    cx, cy = center
    halo_radius = radius + 8
    halo_width = 2 * scale
    if kind == "round":
        halo_box = tuple(
            round(value * scale)
            for value in (cx - halo_radius, cy - halo_radius, cx + halo_radius, cy + halo_radius)
        )
        draw.ellipse(halo_box, outline=PALE_BLUE, width=halo_width)
        if halo_style == "double":
            inset = halo_radius - 4
            draw.ellipse(
                tuple(round(value * scale) for value in (cx - inset, cy - inset, cx + inset, cy + inset)),
                outline=PALE_INK,
                width=scale,
            )
        outer = tuple(
            round(value * scale)
            for value in (cx - radius, cy - radius, cx + radius, cy + radius)
        )
        draw.ellipse(outer, fill=color)
    else:
        halo = _scaled(
            [(cx, cy - halo_radius), (cx + halo_radius, cy), (cx, cy + halo_radius), (cx - halo_radius, cy)],
            scale,
        )
        draw.line(halo + [halo[0]], fill=PALE_BLUE, width=halo_width, joint="curve")
        if halo_style == "double":
            inset = halo_radius - 4
            inner_halo = _scaled(
                [(cx, cy - inset), (cx + inset, cy), (cx, cy + inset), (cx - inset, cy)],
                scale,
            )
            draw.line(inner_halo + [inner_halo[0]], fill=PALE_INK, width=scale, joint="curve")
        draw.polygon(
            _scaled([(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)], scale),
            fill=color,
        )

    inner = radius * 0.61
    inner_box = tuple(
        round(value * scale)
        for value in (cx - inner, cy - inner, cx + inner, cy + inner)
    )
    draw.ellipse(inner_box, fill=PANEL_INK)
    line_extent = inner * 0.76
    draw.line(
        _scaled([(cx, cy - line_extent), (cx, cy + line_extent)], scale),
        fill=GRID_INK,
        width=scale,
    )
    draw.line(
        _scaled([(cx - line_extent, cy), (cx + line_extent, cy)], scale),
        fill=GRID_INK,
        width=scale,
    )
    dx, dy = _marker_offset(cell, radius)
    half = 2.4
    marker_box = tuple(
        round(value * scale)
        for value in (cx + dx - half, cy + dy - half, cx + dx + half, cy + dy + half)
    )
    draw.rounded_rectangle(marker_box, radius=scale, fill=DARK_INK)


def _draw_decoration(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    kind: str,
    scale: int,
) -> None:
    cx, cy = center
    radius = 9
    box = tuple(round(value * scale) for value in (cx - radius, cy - radius, cx + radius, cy + radius))
    if kind == "circle":
        draw.ellipse(box, outline=PALE_INK, width=2 * scale)
    elif kind == "square":
        draw.rectangle(box, outline=PALE_INK, width=2 * scale)
    elif kind == "diamond":
        draw.line(
            _scaled([(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy), (cx, cy - radius)], scale),
            fill=PALE_INK,
            width=2 * scale,
            joint="curve",
        )
    else:
        draw.line(_scaled([(cx - radius, cy), (cx + radius, cy)], scale), fill=PALE_INK, width=2 * scale)
        draw.line(_scaled([(cx, cy - radius), (cx, cy + radius)], scale), fill=PALE_INK, width=2 * scale)


def render(scene: dict) -> Image.Image:
    """Render all public scene fields into one deterministic RGB image."""

    first_cell = str(scene["first_cell"])
    second_cell = str(scene["second_cell"])
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    background = str(scene["background"])
    palette_index = int(scene["palette_index"])
    radius = int(scene["glyph_radius"])
    halo_style = str(scene["halo_style"])
    decorations = list(scene["decorations"])
    first_center, second_center = _centers(scene)
    first_color, second_color = PALETTES[palette_index]

    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    _draw_stamp(draw, first_center, first_cell, first_color, radius, "round", halo_style, scale)
    _draw_stamp(draw, second_center, second_cell, second_color, radius, "diamond", halo_style, scale)

    remote = [(48, 48), (width - 48, 48), (48, height - 48), (width - 48, height - 48)]
    for center, kind in zip(remote, decorations, strict=True):
        _draw_decoration(draw, center, kind, scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "overwrite" if scene["first_cell"] == scene["second_cell"] else "separate"


def margin(scene: dict) -> float:
    """Pixel margin from each marker center to the address-grid midlines."""

    radius = float(scene["glyph_radius"])
    return float(radius * 0.34 - 2.4)


def is_quarantined(scene: dict) -> bool:
    return int(scene["glyph_radius"]) < 12 or margin(scene) < 1.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
