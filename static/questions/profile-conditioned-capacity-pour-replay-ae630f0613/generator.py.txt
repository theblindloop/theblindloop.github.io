"""Deterministic renderer and analytic arm for capacity-limited vessel pours."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1320
HEIGHT = 500
CAPACITY = 4
N_VESSELS = 3
N_OPS = 7
TOTAL_UNITS = 6

# Only the fill color changes. Identity, endpoint role, and query status remain
# encoded by geometry, so palette is an answer-preserving nuisance.
FILL_PALETTES = (
    (28, 132, 181),
    (31, 145, 96),
    (43, 103, 194),
    (168, 82, 170),
)
SOURCE_INK = (104, 67, 174)
TARGET_INK = (229, 126, 34)
QUERY_INK = (211, 52, 59)
_DARK = (37, 43, 49)
_MID = (112, 120, 126)
_PAPER = (249, 249, 246)
_WHITE = (255, 255, 253)

_INITIALS = tuple(
    values
    for values in itertools.product(range(CAPACITY + 1), repeat=N_VESSELS)
    if sum(values) == TOTAL_UNITS
)
_DIRECTED_PAIRS = tuple(
    (source, target)
    for source in range(N_VESSELS)
    for target in range(N_VESSELS)
    if source != target
)


def _replay(initial_levels: list[int] | tuple[int, ...], operations: list[list[int]]) -> tuple[list[int], list[str]]:
    levels = [int(value) for value in initial_levels]
    event_kinds: list[str] = []
    for source, target in operations:
        source_level = levels[int(source)]
        vacancy = CAPACITY - levels[int(target)]
        amount = min(source_level, vacancy)
        if amount == 0:
            event_kinds.append("no_op")
        elif source_level < vacancy:
            event_kinds.append("source_empty")
        elif source_level > vacancy:
            event_kinds.append("target_full")
        else:
            event_kinds.append("both_boundaries")
        levels[int(source)] -= amount
        levels[int(target)] += amount
    return levels, event_kinds


def sample_scene(seed: int) -> dict:
    """Sample one answer-stratified, order-sensitive seven-pour program.

    Every scene begins with exactly six filled cells. This holds total initial ink
    and conserved quantity constant across all five answer classes. The target
    class cycles with the seed, while query identity cycles independently.
    """

    seed = int(seed)
    rng = random.Random(seed)
    desired = seed % (CAPACITY + 1)
    query_vessel = (seed // (CAPACITY + 1)) % N_VESSELS

    for _ in range(25000):
        initial = list(rng.choice(_INITIALS))
        operations = [list(rng.choice(_DIRECTED_PAIRS)) for _ in range(N_OPS)]
        final, kinds = _replay(initial, operations)
        reverse, _ = _replay(initial, list(reversed(operations)))
        if (
            final[query_vessel] == desired
            and initial[query_vessel] != desired
            and reverse[query_vessel] != desired
            and kinds.count("no_op") >= 1
            and sum(kind != "no_op" for kind in kinds) >= 3
        ):
            break
    else:
        raise RuntimeError("could not sample a constrained pour program")

    display_order = list(range(N_VESSELS))
    rng.shuffle(display_order)
    return {
        "initial_levels": initial,
        "query_vessel": query_vessel,
        "operations": operations,
        "display_order": display_order,
        "palette": rng.randrange(len(FILL_PALETTES)),
        "offset_x": rng.choice((-5, 0, 5)),
        "offset_y": rng.choice((-4, 0, 4)),
        "card_gap": rng.choice((-2, 0, 2)),
        "decorations": rng.randrange(4),
    }


def _font(size: int) -> ImageFont.ImageFont:
    return ImageFont.load_default(size=size)


def _pip_offsets(count: int) -> tuple[tuple[int, int], ...]:
    if count == 1:
        return ((0, 0),)
    if count == 2:
        return ((0, -10), (0, 10))
    if count == 3:
        return ((-8, -9), (8, -9), (0, 9))
    raise ValueError("pip count must be 1, 2, or 3")


def _draw_pips(draw: ImageDraw.ImageDraw, center: tuple[int, int], count: int, color: tuple[int, int, int], radius: int = 6) -> None:
    cx, cy = center
    for ox, oy in _pip_offsets(count):
        draw.ellipse((cx + ox - radius, cy + oy - radius, cx + ox + radius, cy + oy + radius), fill=color)


def render(scene: dict) -> Image.Image:
    """Rasterize every answer-bearing field as separated, discrete marks."""

    if is_quarantined(scene):
        raise ValueError("cannot render malformed scene")
    initial_levels = [int(value) for value in scene["initial_levels"]]
    query_vessel = int(scene["query_vessel"])
    operations = [[int(a), int(b)] for a, b in scene["operations"]]
    display_order = [int(value) for value in scene["display_order"]]
    fill = FILL_PALETTES[int(scene["palette"])]
    dx = int(scene["offset_x"])
    dy = int(scene["offset_y"])
    card_gap = int(scene["card_gap"])
    decorations = int(scene["decorations"])

    image = Image.new("RGB", (WIDTH, HEIGHT), _PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 34 + dy), "CAPACITY-LIMITED POUR PROGRAM", font=_font(25), fill=_DARK, anchor="mm")
    draw.text((WIDTH // 2, 65 + dy), "each arrow pours until SOURCE EMPTY or DESTINATION FULL", font=_font(17), fill=_MID, anchor="mm")

    # Initial state panel. Vessels may be permuted on the page; their pip labels
    # bind visible position to persistent identity.
    draw.rounded_rectangle((22 + dx, 82 + dy, 349 + dx, 447 + dy), 13, fill=_WHITE, outline=_DARK, width=3)
    draw.text((185 + dx, 108 + dy), "START  -  CAPACITY 4", font=_font(19), fill=_DARK, anchor="mm")
    vessel_centers = (87, 185, 283)
    bottom = 345
    cell_height = 38
    cell_gap = 7
    for display_slot, identity in enumerate(display_order):
        cx = vessel_centers[display_slot] + dx
        if identity == query_vessel:
            draw.rounded_rectangle((cx - 38, 153 + dy, cx + 38, 356 + dy), 9, outline=QUERY_INK, width=6)
        for level_index in range(CAPACITY):
            y1 = bottom - level_index * (cell_height + cell_gap) + dy
            y0 = y1 - cell_height
            draw.rounded_rectangle((cx - 29, y0, cx + 29, y1), 4, fill=_WHITE, outline=_DARK, width=2)
            if level_index < initial_levels[identity]:
                draw.rectangle((cx - 22, y0 + 6, cx + 22, y1 - 6), fill=fill)
        _draw_pips(draw, (cx, 390 + dy), identity + 1, _DARK)
    draw.text((185 + dx, 428 + dy), "pips name vessels  -  red ring marks query", font=_font(13), fill=_MID, anchor="mm")

    # Seven cards. Purple pips on the left are the source; orange pips on the
    # right are the destination. Geometry, not text or color names, defines role.
    stride = 128 + card_gap
    first_x = 378 + dx
    card_y0, card_y1 = 119 + dy, 407 + dy
    for index, (source, target) in enumerate(operations):
        x0 = first_x + index * stride
        x1 = x0 + 112
        cx = (x0 + x1) // 2
        draw.rounded_rectangle((x0, card_y0, x1, card_y1), 10, fill=_WHITE, outline=_DARK, width=3)
        draw.text((cx, card_y0 + 22), str(index + 1), font=_font(17), fill=_MID, anchor="mm")
        draw.text((x0 + 26, card_y0 + 70), "FROM", font=_font(11), fill=_MID, anchor="mm")
        draw.text((x0 + 86, card_y0 + 70), "TO", font=_font(11), fill=_MID, anchor="mm")
        source_center = (x0 + 26, card_y0 + 139)
        target_center = (x0 + 86, card_y0 + 139)
        _draw_pips(draw, source_center, source + 1, SOURCE_INK)
        _draw_pips(draw, target_center, target + 1, TARGET_INK)
        arrow_y = card_y0 + 139
        draw.line((x0 + 43, arrow_y, x0 + 66, arrow_y), fill=_DARK, width=4)
        draw.polygon(((x0 + 72, arrow_y), (x0 + 63, arrow_y - 7), (x0 + 63, arrow_y + 7)), fill=_DARK)
        draw.text((cx, card_y1 - 47), "POUR", font=_font(15), fill=_DARK, anchor="mm")
        draw.text((cx, card_y1 - 23), "until bound", font=_font(12), fill=_MID, anchor="mm")

    # Low-contrast decorations vary neutral ink without entering any evidence mask.
    decoration_x = (420, 610, 800, 990)
    for index, x in enumerate(decoration_x):
        if (index + decorations) % 2 == 0:
            y = 462 + (index % 2) * 7
            draw.polygon(((x + dx, y - 5), (x + 5 + dx, y), (x + dx, y + 5), (x - 5 + dx, y)), fill=(220, 224, 224), outline=(165, 171, 174))
    return image


def analytic_gold(scene: dict) -> str:
    final, _ = _replay(scene["initial_levels"], scene["operations"])
    return str(final[int(scene["query_vessel"])])


def margin(scene: dict) -> float:
    """Conservative pixel runner-up margin for the smallest evidence separation."""

    return 0.0 if is_quarantined(scene) else 11.0


def is_quarantined(scene: dict) -> bool:
    try:
        initial = scene.get("initial_levels")
        query = scene.get("query_vessel")
        operations = scene.get("operations")
        display = scene.get("display_order")
        palette = scene.get("palette")
        offset_x = scene.get("offset_x")
        offset_y = scene.get("offset_y")
        card_gap = scene.get("card_gap")
        decorations = scene.get("decorations")
        valid_operations = (
            isinstance(operations, list)
            and len(operations) == N_OPS
            and all(
                isinstance(pair, list)
                and len(pair) == 2
                and all(isinstance(value, int) and 0 <= value < N_VESSELS for value in pair)
                and pair[0] != pair[1]
                for pair in operations
            )
        )
        return not (
            isinstance(initial, list)
            and len(initial) == N_VESSELS
            and all(isinstance(value, int) and 0 <= value <= CAPACITY for value in initial)
            and sum(initial) == TOTAL_UNITS
            and isinstance(query, int)
            and 0 <= query < N_VESSELS
            and valid_operations
            and isinstance(display, list)
            and sorted(display) == list(range(N_VESSELS))
            and isinstance(palette, int)
            and 0 <= palette < len(FILL_PALETTES)
            and isinstance(offset_x, int)
            and -8 <= offset_x <= 8
            and isinstance(offset_y, int)
            and -8 <= offset_y <= 8
            and isinstance(card_gap, int)
            and -3 <= card_gap <= 3
            and isinstance(decorations, int)
            and 0 <= decorations < 4
        )
    except (AttributeError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
