"""Deterministic renderer for marked-edge route redundancy."""

from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 400
BG = (250, 249, 245)
NAVY = (28, 52, 82)
YELLOW = (244, 183, 35)
GREEN = (36, 174, 92)
RED = (218, 58, 69)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    y_mid = 198 + rng.randint(-18, 18)
    side = -1 if rng.randrange(2) == 0 else 1
    y_bypass = y_mid + side * rng.randint(112, 136)
    x_a = 116 + rng.randint(-12, 12)
    x_b = 524 + rng.randint(-12, 12)
    nodes = {
        "s": [42 + rng.randint(-5, 5), y_mid],
        "a": [x_a, y_mid],
        "c": [x_a, y_bypass],
        "d": [x_b, y_bypass],
        "b": [x_b, y_mid],
        "t": [598 + rng.randint(-5, 5), y_mid],
    }
    # Two out of every three integer seeds have a genuine alternate route.
    redundant = int(seed) % 3 != 0
    marked = ["a", "b"]
    edges = [["s", "a"], marked, ["a", "c"], ["c", "d"], ["d", "b"]]
    if not redundant:
        # Replace one side of the bypass by an equal-length outward spur.  This
        # balances edge count and total ink while breaking terminal redundancy.
        missing_left = rng.randrange(2) == 0
        if missing_left:
            edges.remove(["a", "c"])
            spur_y = y_mid - side * abs(y_bypass - y_mid)
            nodes["u"] = [x_a, spur_y]
            edges.append(["a", "u"])
        else:
            edges.remove(["d", "b"])
            spur_y = y_mid - side * abs(y_bypass - y_mid)
            nodes["u"] = [x_b, spur_y]
            edges.append(["b", "u"])
    edges.append(["b", "t"])
    return {
        "nodes": nodes,
        "edges": edges,
        "marked_edge": marked,
        "start_node": "s",
        "finish_node": "t",
        "line_width": [5, 7, 9][int(seed) % 3],
    }


def _edge_key(edge: list[str]) -> frozenset[str]:
    return frozenset(edge)


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    marked = scene["marked_edge"]
    start = scene["start_node"]
    finish = scene["finish_node"]
    width = int(scene.get("line_width", 7))
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    marked_key = _edge_key(marked)
    for edge in edges:
        if _edge_key(edge) != marked_key:
            draw.line([tuple(nodes[edge[0]]), tuple(nodes[edge[1]])], fill=NAVY, width=width)
    p, q = (tuple(nodes[name]) for name in marked)
    # Yellow is a visibly removable edge, with navy junction caps restored.
    draw.line([p, q], fill=YELLOW, width=width + 2)
    radius = width // 2 + 2
    for name in marked:
        x, y = nodes[name]
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=NAVY)
    # Clear terminal identities sit at distant ends of the connected network.
    for name, color in ((start, GREEN), (finish, RED)):
        x, y = nodes[name]
        r = 11
        draw.ellipse((x-r, y-r, x+r, y+r), fill=color, outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    marked = scene["marked_edge"]
    start = scene["start_node"]
    finish = scene["finish_node"]
    adjacency = {name: [] for name in nodes}
    marked_key = _edge_key(marked)
    for left, right in edges:
        if _edge_key([left, right]) == marked_key:
            continue
        adjacency[left].append(right)
        adjacency[right].append(left)
    seen = {start}
    queue = deque([start])
    while queue:
        node = queue.popleft()
        for nxt in adjacency[node]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return "yes" if finish in seen else "no"


def margin(scene: dict) -> float:
    # The narrowest generated line remains two pixels above quarantine.
    return float(int(scene.get("line_width", 7)) - 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
