"""Independent image-only inverse arm for marked-edge route redundancy."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _mask_near(rgb: np.ndarray, target: tuple[int, int, int], tolerance: int = 10) -> np.ndarray:
    delta = rgb.astype(np.int16) - np.asarray(target, dtype=np.int16)
    return np.max(np.abs(delta), axis=2) <= tolerance


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    out = mask.copy()
    for _ in range(rounds):
        padded = np.pad(out, 1)
        grown = np.zeros_like(out)
        for dy in range(3):
            for dx in range(3):
                grown |= padded[dy:dy + out.shape[0], dx:dx + out.shape[1]]
        out = grown
    return out


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    navy = _mask_near(rgb, (28, 52, 82))
    green = _mask_near(rgb, (36, 174, 92))
    red = _mask_near(rgb, (218, 58, 69))
    if navy.sum() < 100 or green.sum() < 20 or red.sum() < 20:
        raise ValueError("required route or terminal colors are absent")
    start_zone = _dilate(green, 4) & navy
    finish_zone = _dilate(red, 4) & navy
    starts = np.argwhere(start_zone)
    if not len(starts) or not finish_zone.any():
        raise ValueError("a terminal does not visibly meet the navy network")
    h, w = navy.shape
    visited = np.zeros_like(navy)
    queue = deque((int(y), int(x)) for y, x in starts)
    for y, x in queue:
        visited[y, x] = True
    while queue:
        y, x = queue.popleft()
        if finish_zone[y, x]:
            return "yes"
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                yy, xx = y + dy, x + dx
                if 0 <= yy < h and 0 <= xx < w and navy[yy, xx] and not visited[yy, xx]:
                    visited[yy, xx] = True
                    queue.append((yy, xx))
    return "no"
