"""Pixel-only decision arm for slope_extension_finish.

Receives one rasterized PNG and nothing else: no scene, no renderer import, no
analytic gold, no metadata. It locates the black vertical finish line by column
ink, segments the four coloured segments by exact colour, fits a straight line
to each segment's per-column centroids, extends every fit to the finish-line
abscissa and reports the colour arriving nearest the top. It abstains whenever
the expected structure is absent, a fit is not straight, or the top-two arrival
gap is too small to be read reliably at this resolution.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "slope_extension_finish-oracle-1.0.0"

ABSTAIN = "abstain"

_MARK_RGB = {
    "red": (214, 28, 28),
    "green": (22, 142, 62),
    "blue": (34, 68, 210),
    "orange": (242, 150, 16),
}

_DARK = 70  # a finish-line pixel is dark in every channel
_MIN_FINISH_INK = 200  # the finish line spans most of the picture height
_MIN_COLUMNS = 120  # a drawn segment covers at least this many columns
_EDGE_TRIM = 3  # end columns of a thick stroke have biased centroids
_MAX_RESIDUAL = 1.5  # per-column centroid RMS around the fitted straight line
_MIN_GAP_PX = 0.8  # below this the arrival ordering is not stable in pixels


def _finish_abscissa(dark: np.ndarray) -> float | None:
    counts = dark.sum(axis=0)
    peak = int(counts.max()) if counts.size else 0
    if peak < _MIN_FINISH_INK:
        return None
    columns = np.flatnonzero(counts >= peak * 0.5)
    if columns.size == 0 or columns.size > 9:
        return None
    if int(columns[-1] - columns[0]) != columns.size - 1:
        # More than one tall vertical structure: the finish line is ambiguous.
        return None
    return float(columns.mean())


def _fit(mask: np.ndarray) -> tuple[float, float] | None:
    columns = np.flatnonzero(mask.any(axis=0))
    if columns.size < _MIN_COLUMNS:
        return None
    if int(columns[-1] - columns[0]) != columns.size - 1:
        return None  # a split or partly erased stroke
    columns = columns[_EDGE_TRIM : columns.size - _EDGE_TRIM]
    if columns.size < _MIN_COLUMNS - 2 * _EDGE_TRIM:
        return None
    rows = np.arange(mask.shape[0], dtype=np.float64)
    sub = mask[:, columns].astype(np.float64)
    weight = sub.sum(axis=0)
    if float(weight.min()) <= 0.0:
        return None
    centre = (rows[:, None] * sub).sum(axis=0) / weight
    xs = columns.astype(np.float64)
    slope, intercept = np.polyfit(xs, centre, 1)
    residual = float(np.sqrt(np.mean((centre - (slope * xs + intercept)) ** 2)))
    if residual > _MAX_RESIDUAL:
        return None
    return float(slope), float(intercept)


def decision_from_image(image: Image.Image) -> str:
    data = np.asarray(image.convert("RGB"), dtype=np.int16)
    if data.ndim != 3 or data.shape[2] != 3:
        return ABSTAIN

    finish_x = _finish_abscissa((data < _DARK).all(axis=2))
    if finish_x is None:
        return ABSTAIN

    readings = []
    for name, target in sorted(_MARK_RGB.items()):
        mask = (data == np.array(target, dtype=np.int16)).all(axis=2)
        if int(mask.sum()) < 3 * _MIN_COLUMNS:
            return ABSTAIN
        fit = _fit(mask)
        if fit is None:
            return ABSTAIN
        slope, intercept = fit
        arrival = slope * finish_x + intercept
        if not 0.0 <= arrival < data.shape[0]:
            return ABSTAIN
        readings.append((arrival, name))

    readings.sort()
    if readings[1][0] - readings[0][0] < _MIN_GAP_PX:
        return ABSTAIN
    return readings[0][1]
