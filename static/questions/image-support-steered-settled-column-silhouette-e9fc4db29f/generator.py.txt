"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

N = 9
FRAME = (88, 54, 140)
AMBER = (239, 174, 55)
BACKGROUND = (248, 246, 241)
PALETTES = (
    ((31, 58, 86), (220, 235, 239)),
    ((45, 68, 48), (231, 238, 213)),
    ((91, 53, 46), (242, 226, 205)),
)
PROFILES = {
    "rising stair": (1, 2, 3, 4, 5, 6, 7, 8, 9),
    "falling stair": (9, 8, 7, 6, 5, 4, 3, 2, 1),
    "central peak": (2, 3, 5, 8, 9, 8, 5, 3, 2),
    "central notch": (8, 7, 5, 2, 1, 2, 5, 7, 8),
}
LABELS = tuple(PROFILES)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    profile = PROFILES[LABELS[int(seed) % len(LABELS)]]
    columns: list[list[int]] = []
    for height in profile:
        column = [1] * height + [0] * (N - height)
        rng.shuffle(column)
        columns.append(column)
    cell_size = rng.choice((34, 36, 38, 40))
    total = 14 + N * cell_size
    offset_x = rng.randint(34, 512 - total - 34)
    offset_y = rng.randint(34, 512 - total - 34)
    return {
        "label": LABELS[int(seed) % len(LABELS)],
        "columns": columns,
        "cell_size": cell_size,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "palette": rng.randrange(len(PALETTES)),
    }


def _column_counts(scene: dict) -> tuple[int, ...]:
    columns = scene["columns"]
    return tuple(sum(int(value) for value in column) for column in columns)


def _distances(counts: tuple[int, ...]) -> list[tuple[int, str]]:
    return sorted((sum(abs(a - b) for a, b in zip(counts, profile)), label)
                  for label, profile in PROFILES.items())


def analytic_gold(scene: dict) -> str:
    distances = _distances(_column_counts(scene))
    if len(distances) < 2 or distances[0][0] == distances[1][0]:
        return "abstain"
    return distances[0][1]


def margin(scene: dict) -> float:
    distances = _distances(_column_counts(scene))
    return float(distances[1][0] - distances[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    columns = scene["columns"]
    cell = int(scene["cell_size"])
    x0 = int(scene["offset_x"])
    y0 = int(scene["offset_y"])
    palette_index = int(scene["palette"])
    dark, pale = PALETTES[palette_index]
    border = 7
    line = 2
    total = 2 * border + N * cell

    image = Image.new("RGB", (512, 512), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x0 + total - 1, y0 + total - 1), fill=FRAME)
    gx, gy = x0 + border, y0 + border
    for col in range(N):
        for row in range(N):
            color = dark if int(columns[col][row]) else pale
            xa = gx + col * cell + line
            ya = gy + row * cell + line
            xb = gx + (col + 1) * cell - line - 1
            yb = gy + (row + 1) * cell - line - 1
            draw.rectangle((xa, ya, xb, yb), fill=color)

    # A visible downward fiducial is integrated into the top border.
    cx = x0 + total // 2
    draw.polygon(((cx - 5, y0 + 1), (cx + 5, y0 + 1), (cx, y0 + 6)), fill=AMBER)
    return image
