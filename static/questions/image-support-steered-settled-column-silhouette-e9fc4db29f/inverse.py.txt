"""Independent image-only inverse for column compaction."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 9
FRAME = np.array((88, 54, 140), dtype=np.int16)
PROFILES = {
    "rising stair": (1, 2, 3, 4, 5, 6, 7, 8, 9),
    "falling stair": (9, 8, 7, 6, 5, 4, 3, 2, 1),
    "central peak": (2, 3, 5, 8, 9, 8, 5, 3, 2),
    "central notch": (8, 7, 5, 2, 1, 2, 5, 7, 8),
}


def _recover_cells(image: Image.Image) -> np.ndarray | None:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    frame_mask = np.max(np.abs(arr - FRAME), axis=2) <= 2
    ys, xs = np.nonzero(frame_mask)
    if len(xs) < 500:
        return None
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if abs(width - height) > 2 or not (320 <= width <= 380):
        return None

    # Infer the repeated pitch from the located frame, never from absolute position.
    border = 7
    pitch = (width - 2 * border) / N
    if abs(pitch - round(pitch)) > 0.05:
        return None
    pitch = int(round(pitch))
    samples = np.zeros((N, N, 3), dtype=np.float64)
    for row in range(N):
        for col in range(N):
            cx = int(round(x0 + border + (col + 0.5) * pitch))
            cy = int(round(y0 + border + (row + 0.5) * pitch))
            patch = arr[cy - 2:cy + 3, cx - 2:cx + 3].astype(np.float64)
            if patch.shape != (5, 5, 3) or float(patch.std(axis=(0, 1)).max()) > 2.0:
                return None
            samples[row, col] = patch.mean(axis=(0, 1))

    flat = samples.reshape(-1, 3)
    luminance = flat @ np.array((0.2126, 0.7152, 0.0722))
    order = np.argsort(luminance)
    # Every class has exactly 45 dark cells, explicitly balancing total ink.
    dark_idx, pale_idx = order[:45], order[45:]
    gap = float(luminance[pale_idx].min() - luminance[dark_idx].max())
    if gap < 60.0 or float(luminance.max()) > 246.0:
        return None
    dark_center = flat[dark_idx].mean(axis=0)
    pale_center = flat[pale_idx].mean(axis=0)
    if np.linalg.norm(flat[dark_idx] - dark_center, axis=1).max() > 5.0:
        return None
    if np.linalg.norm(flat[pale_idx] - pale_center, axis=1).max() > 5.0:
        return None
    binary = np.zeros(N * N, dtype=np.uint8)
    binary[dark_idx] = 1
    return binary.reshape(N, N)


def decision_from_image(image: Image.Image) -> str:
    cells = _recover_cells(image)
    if cells is None:
        return "abstain"
    counts = tuple(int(cells[:, col].sum()) for col in range(N))
    ranked = sorted((sum(abs(a - b) for a, b in zip(counts, profile)), label)
                    for label, profile in PROFILES.items())
    if ranked[0][0] != 0 or ranked[0][0] == ranked[1][0]:
        return "abstain"
    return ranked[0][1]
