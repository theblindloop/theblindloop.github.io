"""Fold-net opposite-face world.

A cream board carries several large square tiles of identical size. Exactly six
of them are joined edge to edge into one connected piece: the net. Folding that
net along its shared edges closes it into a cube, and the face bearing the white
dot ends up directly opposite exactly one other face. The decision is the colour
of that opposite face.

All decisive evidence lives inside the one small connected net blob. Loose tiles
of the same size lying apart from the net, and the small decorative squares, are
nondiagnostic.
"""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

RENDERER_VERSION = "foldnet_face_opposite_bind-0.1.0"

WIDTH = 896
HEIGHT = 628
CELL = 32
BG = (243, 238, 226)
MARK_RGB = (250, 250, 252)
DOT_RADIUS = 6
QUARANTINE_MARGIN = 12.0

COLOR_ORDER = ("red", "orange", "yellow", "green", "blue", "purple")
PALETTE = {
    "red": (198, 42, 38),
    "orange": (226, 124, 20),
    "yellow": (222, 196, 32),
    "green": (46, 140, 58),
    "blue": (38, 86, 190),
    "purple": (126, 52, 166),
}
DECOR_GREY = (176, 170, 158)

_DIRS = ((-1, 0), (1, 0), (0, -1), (0, 1))


# --------------------------------------------------------------------------
# cube folding by rolling a die across the paper squares
# --------------------------------------------------------------------------
_BOTTOM, _TOP, _NORTH, _SOUTH, _EAST, _WEST = range(6)
_START_STATE = (0, 1, 2, 3, 4, 5)


def _roll(state: tuple, dr: int, dc: int) -> tuple:
    """Roll the die one square in paper direction (dr, dc); rows grow downward.

    The state is a tuple of face ids ordered bottom, top, north, south, east,
    west, so that ``f`` and ``f ^ 1`` stay opposite faces of the cube.
    """
    new = list(state)
    if (dr, dc) == (-1, 0):  # north
        new[_BOTTOM], new[_NORTH], new[_TOP], new[_SOUTH] = (
            state[_NORTH], state[_TOP], state[_SOUTH], state[_BOTTOM],
        )
    elif (dr, dc) == (1, 0):  # south
        new[_BOTTOM], new[_SOUTH], new[_TOP], new[_NORTH] = (
            state[_SOUTH], state[_TOP], state[_NORTH], state[_BOTTOM],
        )
    elif (dr, dc) == (0, 1):  # east
        new[_BOTTOM], new[_EAST], new[_TOP], new[_WEST] = (
            state[_EAST], state[_TOP], state[_WEST], state[_BOTTOM],
        )
    elif (dr, dc) == (0, -1):  # west
        new[_BOTTOM], new[_WEST], new[_TOP], new[_EAST] = (
            state[_WEST], state[_TOP], state[_EAST], state[_BOTTOM],
        )
    else:
        raise ValueError("not a unit paper step")
    return tuple(new)


def fold_faces(cells: frozenset) -> dict | None:
    """Map each net square to a cube face id, or None when the net is invalid.

    Face ids are paired so that ``f`` and ``f ^ 1`` are opposite faces of the
    cube. A conflicting revisit (for example any 2x2 block) rejects the net.
    """
    if len(cells) != 6:
        return None
    start = min(cells)
    seen = {start: _START_STATE}
    faces = {start: _START_STATE[_BOTTOM]}
    queue = deque([start])
    while queue:
        cell = queue.popleft()
        state = seen[cell]
        for dr, dc in _DIRS:
            nxt = (cell[0] + dr, cell[1] + dc)
            if nxt not in cells:
                continue
            rolled = _roll(state, dr, dc)
            if nxt in seen:
                if seen[nxt][_BOTTOM] != rolled[_BOTTOM]:
                    return None
                continue
            seen[nxt] = rolled
            faces[nxt] = rolled[_BOTTOM]
            queue.append(nxt)
    if len(faces) != 6 or len(set(faces.values())) != 6:
        return None
    return faces


def opposite_cell(cells: frozenset, mark: tuple) -> tuple | None:
    faces = fold_faces(cells)
    if faces is None or mark not in faces:
        return None
    wanted = faces[mark] ^ 1
    for cell, face in faces.items():
        if face == wanted:
            return cell
    return None


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _rect(x: int, y: int, size: int) -> tuple:
    return (x, y, x + size, y + size)


def _clearance(a: tuple, b: tuple) -> float:
    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return 0.0
    return float((dx * dx + dy * dy) ** 0.5)


def _net_rects(scene: dict) -> list:
    ox, oy = scene["origin"]
    return [_rect(ox + c * CELL, oy + r * CELL, CELL) for r, c, _ in scene["cells"]]


def _grow_polyomino(rng: random.Random) -> frozenset:
    cells = {(0, 0)}
    while len(cells) < 6:
        frontier = sorted(
            {
                (r + dr, c + dc)
                for r, c in cells
                for dr, dc in _DIRS
                if (r + dr, c + dc) not in cells
            }
        )
        cells.add(frontier[rng.randrange(len(frontier))])
    rows = min(r for r, _ in cells)
    cols = min(c for _, c in cells)
    return frozenset((r - rows, c - cols) for r, c in cells)


def _mark_choices(cells: frozenset) -> list:
    """Marks that force real folding rather than a cheap board-level cue."""
    faces = fold_faces(cells)
    if faces is None:
        return []
    good = []
    for mark in sorted(cells):
        degree = sum(1 for dr, dc in _DIRS if (mark[0] + dr, mark[1] + dc) in cells)
        if degree > 2:
            continue  # a high-degree square leaves only one non-touching answer
        if any((mark[0] + 2 * dr, mark[1] + 2 * dc) in cells for dr, dc in _DIRS):
            continue  # colinear-at-two is a known shortcut for "opposite"
        answer = opposite_cell(cells, mark)
        if answer is None:
            continue
        dists = {
            cell: abs(cell[0] - mark[0]) + abs(cell[1] - mark[1])
            for cell in cells
            if cell != mark
        }
        far = max(dists.values())
        farthest = [cell for cell, d in dists.items() if d == far]
        if len(farthest) == 1 and farthest[0] == answer:
            continue  # "pick the farthest square" must not solve the scene
        good.append(mark)
    return good


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    shape = None
    mark = None
    for _ in range(6000):
        candidate = _grow_polyomino(rng)
        choices = _mark_choices(candidate)
        if choices:
            shape = candidate
            mark = choices[rng.randrange(len(choices))]
            break
    if shape is None:
        raise RuntimeError("no foldable net found")

    answer = opposite_cell(shape, mark)
    target_colour = COLOR_ORDER[seed % 6]
    rest = [c for c in COLOR_ORDER if c != target_colour]
    rng.shuffle(rest)
    ordered = sorted(shape)
    colours = {}
    spare = iter(rest)
    for cell in ordered:
        colours[cell] = target_colour if cell == answer else next(spare)

    rows = 1 + max(r for r, _ in shape)
    cols = 1 + max(c for _, c in shape)
    pad = 80
    ox = rng.randrange(pad, WIDTH - pad - cols * CELL)
    oy = rng.randrange(pad, HEIGHT - pad - rows * CELL)
    cells = [[r, c, colours[(r, c)]] for r, c in ordered]
    scene = {"origin": [ox, oy], "cells": cells, "mark_cell": [mark[0], mark[1]]}
    net = _net_rects(scene)

    near_boundary = (seed // 6) % 5 == 0
    gap = rng.randrange(7, 14) if near_boundary else rng.randrange(16, 71)

    loose = []
    placed = list(net)
    for _ in range(400):
        r, c, _colour = cells[rng.randrange(len(cells))]
        dr, dc = _DIRS[rng.randrange(4)]
        x = ox + c * CELL + dc * (CELL + gap)
        y = oy + r * CELL + dr * (CELL + gap)
        if not (8 <= x <= WIDTH - CELL - 8 and 8 <= y <= HEIGHT - CELL - 8):
            continue
        box = _rect(x, y, CELL)
        if min(_clearance(box, other) for other in placed) < gap - 0.01:
            continue
        loose.append([x, y, COLOR_ORDER[rng.randrange(6)]])
        placed.append(box)
        break
    if not loose:
        raise RuntimeError("no near loose tile placed")

    for _ in range(3000):
        if len(loose) >= 2 + rng.randrange(3):
            break
        x = rng.randrange(8, WIDTH - CELL - 8)
        y = rng.randrange(8, HEIGHT - CELL - 8)
        box = _rect(x, y, CELL)
        if min(_clearance(box, other) for other in placed) < max(gap + 6, 22):
            continue
        loose.append([x, y, COLOR_ORDER[rng.randrange(6)]])
        placed.append(box)

    decor_colours = list(COLOR_ORDER) + ["grey", "grey"]
    decor = []
    wanted = 22 + rng.randrange(9)
    for _ in range(9000):
        if len(decor) >= wanted:
            break
        size = 7 + rng.randrange(9)
        x = rng.randrange(6, WIDTH - size - 6)
        y = rng.randrange(6, HEIGHT - size - 6)
        box = _rect(x, y, size)
        if min(_clearance(box, other) for other in net) < 19:
            continue
        if min(_clearance(box, other) for other in placed) < 9:
            continue
        decor.append([x, y, size, decor_colours[rng.randrange(len(decor_colours))]])
        placed.append(box)

    scene["loose"] = sorted(loose)
    scene["decor"] = sorted(decor)
    return scene


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    for x, y, size, name in sorted(scene["decor"]):
        rgb = DECOR_GREY if name == "grey" else PALETTE[name]
        draw.rectangle([x, y, x + size - 1, y + size - 1], fill=rgb)

    for x, y, name in sorted(scene["loose"]):
        draw.rectangle([x, y, x + CELL - 1, y + CELL - 1], fill=PALETTE[name])

    ox, oy = scene["origin"]
    for r, c, name in sorted(scene["cells"]):
        x = ox + c * CELL
        y = oy + r * CELL
        draw.rectangle([x, y, x + CELL - 1, y + CELL - 1], fill=PALETTE[name])

    mr, mc = scene["mark_cell"]
    cx = ox + mc * CELL + CELL // 2
    cy = oy + mr * CELL + CELL // 2
    draw.ellipse(
        [cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS],
        fill=MARK_RGB,
    )
    return image


# --------------------------------------------------------------------------
# analytic gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    cells = frozenset((r, c) for r, c, _ in scene["cells"])
    mark = (scene["mark_cell"][0], scene["mark_cell"][1])
    answer = opposite_cell(cells, mark)
    if answer is None:
        raise ValueError("scene does not carry a foldable net")
    for r, c, name in scene["cells"]:
        if (r, c) == answer:
            return name
    raise ValueError("opposite face carries no colour")


def margin(scene: dict) -> float:
    """Pixel clearance between the net and the nearest other mark on the board.

    As that clearance shrinks the nearest loose tile stops being separable from
    the net; at zero it would join the piece, so a seventh square would either
    destroy the cube net or repair it into a different folding.
    """
    net = _net_rects(scene)
    others = [_rect(x, y, CELL) for x, y, _ in scene["loose"]]
    others += [_rect(x, y, size) for x, y, size, _ in scene["decor"]]
    if not others:
        return float(CELL)
    return min(_clearance(a, b) for a in net for b in others)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical stored-order aliases; every one must preserve the gold."""
    out = []

    rotated = dict(scene)
    cells = list(scene["cells"])
    rotated["cells"] = cells[1:] + cells[:1]
    out.append(("cell_list_rotation", rotated))

    reversed_loose = dict(scene)
    reversed_loose["loose"] = list(reversed(scene["loose"]))
    out.append(("loose_tile_list_reversal", reversed_loose))

    reversed_decor = dict(scene)
    reversed_decor["decor"] = list(reversed(scene["decor"]))
    out.append(("decor_list_reversal", reversed_decor))

    return out
