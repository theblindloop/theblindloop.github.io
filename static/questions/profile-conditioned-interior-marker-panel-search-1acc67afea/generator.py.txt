"""Deterministic renderer and analytic gold for an induced-hull search."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

CANVAS_SIZE = (1040, 380)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (12, 18, 264, 362),
    (270, 18, 522, 362),
    (528, 18, 780, 362),
    (786, 18, 1038, 362),
)
BACKGROUND = (252, 252, 249)
PANEL_FILL = (246, 247, 243)
PANEL_EDGE = (184, 188, 181)
LABEL_INK = (104, 106, 100)
MARKER_INK = (22, 28, 32)
MARKER_RADIUS = 9

# Every template has the same point bounding box.  The four equal disks therefore
# have equal count, ink area, and rendered bounding-box dimensions in every panel.
_TARGET_TEMPLATES = (
    ((36, 54), (216, 60), (130, 278), (126, 148)),
    ((36, 60), (216, 54), (126, 278), (132, 148)),
)
_CONVEX_TEMPLATES = (
    ((36, 54), (216, 60), (204, 278), (48, 270)),
    ((36, 54), (216, 90), (198, 278), (58, 252)),
    ((36, 80), (216, 54), (208, 278), (52, 278)),
)


def _cross(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _strict_inside(
    point: tuple[float, float], triangle: tuple[tuple[float, float], ...], epsilon: float = 2.0
) -> bool:
    if len(triangle) != 3:
        return False
    values = [_cross(triangle[i], triangle[(i + 1) % 3], point) for i in range(3)]
    positive = all(value > epsilon for value in values)
    negative = all(value < -epsilon for value in values)
    return positive or negative


def _panel_score(points: list[tuple[float, float]]) -> int:
    if len(points) != 4:
        raise ValueError("each panel must contain four marker centers")
    score = 0
    for index, point in enumerate(points):
        triangle = tuple(points[j] for j in range(4) if j != index)
        if _strict_inside(point, triangle):
            score += 1
    return score


def _panel_points(scene: dict) -> list[list[tuple[float, float]]]:
    panels = scene["panels"]
    if not isinstance(panels, list) or len(panels) != 4:
        raise ValueError("scene must contain four panels")
    result: list[list[tuple[float, float]]] = []
    for panel in panels:
        raw = panel["points"]
        if not isinstance(raw, list) or len(raw) != 4:
            raise ValueError("panel must contain four points")
        result.append([(float(point[0]), float(point[1])) for point in raw])
    return result


def _reflect(template: tuple[tuple[int, int], ...], mode: int) -> tuple[tuple[int, int], ...]:
    """Apply a bbox-preserving reflection to vary nuisance orientation."""

    min_x = min(point[0] for point in template)
    max_x = max(point[0] for point in template)
    min_y = min(point[1] for point in template)
    max_y = max(point[1] for point in template)
    reflected = []
    for x, y in template:
        if mode & 1:
            x = min_x + max_x - x
        if mode & 2:
            y = min_y + max_y - y
        reflected.append((x, y))
    return tuple(reflected)


def _sample_panel_points(
    panel_box: tuple[int, int, int, int],
    template: tuple[tuple[int, int], ...],
    local: random.Random,
) -> list[list[int]]:
    x0, y0, _, _ = panel_box
    offset_x = 16 + local.randrange(12)
    # Keep the lowest disk fully inside the pixel-oracle crop and panel card.
    offset_y = 34 + local.randrange(13)
    mode = local.randrange(4)
    transformed = _reflect(template, mode)
    return [[x0 + offset_x + x, y0 + offset_y + y] for x, y in transformed]


def sample_scene(seed: int) -> dict:
    """Sample four equal marker sets with exactly one strict hull interior."""

    rng = random.Random(seed * 1009 + 37)
    target_slot = seed % 4
    target_variant = (seed // 4) % len(_TARGET_TEMPLATES)
    panels: list[dict] = []
    for index, (label, box) in enumerate(zip(LABELS, PANEL_BOXES, strict=True)):
        local = random.Random(rng.randrange(1 << 30) + index * 7919)
        if index == target_slot:
            template = _TARGET_TEMPLATES[target_variant]
        else:
            template = _CONVEX_TEMPLATES[(seed + index + target_variant) % len(_CONVEX_TEMPLATES)]
        points = _sample_panel_points(box, template, local)
        panels.append({"label": label, "points": points})

    scene = {"panels": panels}
    scores = [_panel_score(points) for points in _panel_points(scene)]
    if scores.count(1) != 1 or any(score not in (0, 1) for score in scores):
        raise AssertionError(f"sampled hull scores are invalid: {scores}")
    return scene


def render(scene: dict) -> Image.Image:
    """Render the public four-panel raster with identical dark markers."""

    image = Image.new("RGB", CANVAS_SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for panel, box in zip(scene["panels"], PANEL_BOXES, strict=True):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(
            (x0, y0, x1 - 1, y1 - 1),
            radius=12,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2,
        )
        draw.text((x0 + 12, y0 + 12), str(panel["label"]), font=font, fill=LABEL_INK)
        for raw_x, raw_y in panel["points"]:
            x, y = int(raw_x), int(raw_y)
            draw.ellipse(
                (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
                fill=MARKER_INK,
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Select the label of the unique panel containing an interior marker."""

    scores = [_panel_score(points) for points in _panel_points(scene)]
    best = max(scores)
    if best != 1 or scores.count(best) != 1:
        raise ValueError(f"scene has no unique interior-marker winner: {scores}")
    return str(scene["panels"][scores.index(best)]["label"])


def margin(scene: dict) -> float:
    scores = sorted((_panel_score(points) for points in _panel_points(scene)), reverse=True)
    if len(scores) != 4:
        raise ValueError("need four panel scores")
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = [_panel_score(points) for points in _panel_points(scene)]
    return max(scores) != 1 or scores.count(max(scores)) != 1 or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """All label-bearing scene fields are rendered; there is no latent alias."""

    return []
