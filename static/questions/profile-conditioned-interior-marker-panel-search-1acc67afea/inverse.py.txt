"""Independent image-only inverse for the induced-hull search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANVAS_SIZE = (1040, 380)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (12, 18, 264, 362),
    (270, 18, 522, 362),
    (528, 18, 780, 362),
    (786, 18, 1038, 362),
)


class OracleAmbiguous(ValueError):
    """Raised when the raster does not expose a unique hull-interior panel."""


def _dark_crop(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    crop = array[y0 + 55 : y1 - 8, x0 + 30 : x1 - 10]
    return np.max(crop, axis=2) < 80


def _components(mask: np.ndarray) -> list[tuple[float, float]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            visited[y, x] = True
            queue: deque[tuple[int, int]] = deque([(x, y)])
            points: list[tuple[int, int]] = []
            while queue:
                cx, cy = queue.popleft()
                points.append((cx, cy))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((nx, ny))
            if len(points) < 80:
                raise OracleAmbiguous("marker component is too small")
            centers.append(
                (sum(point[0] for point in points) / len(points), sum(point[1] for point in points) / len(points))
            )
    return centers


def _cross(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _strict_inside(
    point: tuple[float, float], triangle: tuple[tuple[float, float], ...], epsilon: float = 2.0
) -> bool:
    values = [_cross(triangle[i], triangle[(i + 1) % 3], point) for i in range(3)]
    return (all(value > epsilon for value in values) or all(value < -epsilon for value in values))


def _panel_score(points: list[tuple[float, float]]) -> int:
    if len(points) != 4:
        raise OracleAmbiguous("panel does not contain four markers")
    score = 0
    for index, point in enumerate(points):
        triangle = tuple(points[j] for j in range(4) if j != index)
        if abs(_cross(triangle[0], triangle[1], triangle[2])) <= 4.0:
            raise OracleAmbiguous("triangle vertices are nearly collinear")
        if _strict_inside(point, triangle):
            score += 1
    return score


def decision_from_image(image: Image.Image) -> str:
    """Recover all panel scores from the final RGB pixels only."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    scores: list[int] = []
    for box in PANEL_BOXES:
        centers = _components(_dark_crop(array, box))
        if len(centers) != 4:
            raise OracleAmbiguous(f"expected four marker components, received {len(centers)}")
        scores.append(_panel_score(centers))
    best = max(scores)
    if best != 1 or scores.count(best) != 1:
        raise OracleAmbiguous(f"hull scores are ambiguous: {scores}")
    return LABELS[scores.index(best)]
