"""Deterministic renderer and analytic arm for paired motion interventions."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)

TOKEN_RADIUS = 22.0
ARROW_GAP = 6.0
ARROW_WIDTH = 8
ARROW_HEAD_LENGTH = 16.0
ARROW_HEAD_WIDTH = 9.0
COLLISION_DIAMETER = TOKEN_RADIUS * 2.0
QUARANTINE_MARGIN = 8.0

BACKGROUND = (238, 242, 246)
PANEL_FILL = (250, 251, 253)
FRAME = (112, 126, 140)
FRAME_ACCENT = (205, 213, 220)
BLUE_DISK = (42, 116, 214)
ORANGE_DISK = (232, 126, 37)
BLUE_ARROW = (29, 74, 139)
ORANGE_ARROW = (164, 78, 24)
LABEL = (57, 70, 86)
WHITE = (255, 255, 255)

PANEL_LABELS = ("A", "B", "C", "D")
_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _point(center: tuple[float, float], radius: float, angle_deg: float) -> tuple[float, float]:
    angle = math.radians(angle_deg)
    return center[0] + radius * math.cos(angle), center[1] + radius * math.sin(angle)


def _vector(angle_deg: float, length: float) -> tuple[float, float]:
    angle = math.radians(angle_deg)
    return length * math.cos(angle), length * math.sin(angle)


def _post_center(x: float, y: float, angle_deg: float, length: float) -> tuple[float, float]:
    dx, dy = _vector(angle_deg, length)
    return x + dx, y + dy


def _panel_local_center(rect: tuple[int, int, int, int], local: tuple[float, float]) -> tuple[float, float]:
    return float(rect[0]) + local[0], float(rect[1]) + local[1]


def _arrow_tip(center: tuple[float, float], angle_deg: float, length: float) -> tuple[float, float]:
    # The arrowhead marks the counterfactual destination of the disk center.
    return _point(center, length, angle_deg)


def _arrow_fits(rect: tuple[int, int, int, int], center: tuple[float, float], angle: float, length: float) -> bool:
    tip = _arrow_tip(center, angle, length)
    x0, y0, x1, y1 = rect
    return x0 + 28.0 < tip[0] < x1 - 28.0 and y0 + 28.0 < tip[1] < y1 - 28.0


def _point_segment_distance(
    point: tuple[float, float],
    start: tuple[float, float],
    end: tuple[float, float],
) -> float:
    px, py = point
    ax, ay = start
    bx, by = end
    dx, dy = bx - ax, by - ay
    length_sq = dx * dx + dy * dy
    if length_sq == 0.0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length_sq))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _segments_intersect(
    first_start: tuple[float, float],
    first_end: tuple[float, float],
    second_start: tuple[float, float],
    second_end: tuple[float, float],
) -> bool:
    def cross(
        a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]
    ) -> float:
        return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])

    first_a = cross(first_start, first_end, second_start)
    first_b = cross(first_start, first_end, second_end)
    second_a = cross(second_start, second_end, first_start)
    second_b = cross(second_start, second_end, first_end)
    return first_a * first_b <= 0.0 and second_a * second_b <= 0.0


def _segment_distance(
    first_start: tuple[float, float],
    first_end: tuple[float, float],
    second_start: tuple[float, float],
    second_end: tuple[float, float],
) -> float:
    if _segments_intersect(first_start, first_end, second_start, second_end):
        return 0.0
    return min(
        _point_segment_distance(first_start, second_start, second_end),
        _point_segment_distance(first_end, second_start, second_end),
        _point_segment_distance(second_start, first_start, first_end),
        _point_segment_distance(second_end, first_start, first_end),
    )


def _sample_motion(
    rng: random.Random,
    rect: tuple[int, int, int, int],
    center: tuple[float, float],
) -> tuple[float, float]:
    for _ in range(400):
        angle = rng.uniform(-175.0, 175.0)
        length = rng.uniform(55.0, 88.0)
        if _arrow_fits(rect, center, angle, length):
            return angle, length
    # The broad interior leaves this fallback unreachable for the public
    # sampler, but it keeps sampling total if a future layout is tightened.
    return 0.0, 60.0


def sample_scene(seed: int) -> dict:
    """Sample four clear panels with exactly one post-motion collision."""

    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = []
    for index, rect in enumerate(PANEL_RECTS):
        blue = _panel_local_center(
            rect,
            (rng.uniform(125.0, 155.0), rng.uniform(140.0, 171.0)),
        )
        orange = _panel_local_center(
            rect,
            (rng.uniform(266.0, 296.0), rng.uniform(104.0, 135.0)),
        )
        current_distance = math.hypot(blue[0] - orange[0], blue[1] - orange[1])
        if current_distance <= COLLISION_DIAMETER + 24.0:
            # This deterministic correction is only a guard against future
            # changes to the local ranges; the present ranges are well clear.
            orange = (blue[0] + 145.0, blue[1] - 18.0)

        if index == answer_index:
            # Two nearby but separate endpoints make the post-action overlap
            # deep while keeping the two visible arrowheads distinct.
            separation = math.hypot(orange[0] - blue[0], orange[1] - blue[1])
            perpendicular = (
                -(orange[1] - blue[1]) / separation,
                (orange[0] - blue[0]) / separation,
            )
            midpoint = ((blue[0] + orange[0]) / 2.0, (blue[1] + orange[1]) / 2.0)
            landing_blue = (
                midpoint[0] + perpendicular[0] * 14.0,
                midpoint[1] + perpendicular[1] * 14.0,
            )
            landing_orange = (
                midpoint[0] - perpendicular[0] * 14.0,
                midpoint[1] - perpendicular[1] * 14.0,
            )
            blue_angle = math.degrees(
                math.atan2(landing_blue[1] - blue[1], landing_blue[0] - blue[0])
            )
            orange_angle = math.degrees(
                math.atan2(landing_orange[1] - orange[1], landing_orange[0] - orange[0])
            )
            blue_length = math.hypot(landing_blue[0] - blue[0], landing_blue[1] - blue[1])
            orange_length = math.hypot(
                landing_orange[0] - orange[0], landing_orange[1] - orange[1]
            )
        else:
            for _ in range(400):
                blue_angle, blue_length = _sample_motion(rng, rect, blue)
                orange_angle, orange_length = _sample_motion(rng, rect, orange)
                blue_post = _post_center(blue[0], blue[1], blue_angle, blue_length)
                orange_post = _post_center(orange[0], orange[1], orange_angle, orange_length)
                post_distance = math.hypot(
                    blue_post[0] - orange_post[0], blue_post[1] - orange_post[1]
                )
                blue_tail = _point(blue, TOKEN_RADIUS + ARROW_GAP, blue_angle)
                orange_tail = _point(orange, TOKEN_RADIUS + ARROW_GAP, orange_angle)
                if (
                    post_distance > COLLISION_DIAMETER + QUARANTINE_MARGIN + 18.0
                    and _segment_distance(blue_tail, blue_post, orange_tail, orange_post) > 14.0
                ):
                    break

        panels.append(
            {
                "blue_x": round(blue[0], 3),
                "blue_y": round(blue[1], 3),
                "blue_angle_deg": round(blue_angle, 6),
                "blue_length": round(blue_length, 6),
                "orange_x": round(orange[0], 3),
                "orange_y": round(orange[1], 3),
                "orange_angle_deg": round(orange_angle, 6),
                "orange_length": round(orange_length, 6),
                "token_radius": TOKEN_RADIUS,
            }
        )
    return {"canvas_width": WIDTH, "canvas_height": HEIGHT, "panels": panels}


def _draw_bitmap(draw: ImageDraw.ImageDraw, glyph: str, xy: tuple[int, int], scale: int = 4) -> None:
    bitmap = _BITMAPS[glyph]
    x0, y0 = xy
    for row, bits in enumerate(bitmap):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (
                        x0 + col * scale,
                        y0 + row * scale,
                        x0 + (col + 1) * scale - 1,
                        y0 + (row + 1) * scale - 1,
                    ),
                    fill=LABEL,
                )


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    angle_deg: float,
    length: float,
    color: tuple[int, int, int],
) -> None:
    angle = math.radians(angle_deg)
    ux, uy = math.cos(angle), math.sin(angle)
    px, py = -uy, ux
    tail = (
        center[0] + (TOKEN_RADIUS + ARROW_GAP) * ux,
        center[1] + (TOKEN_RADIUS + ARROW_GAP) * uy,
    )
    tip = (
        center[0] + length * ux,
        center[1] + length * uy,
    )
    draw.line((tail, tip), fill=color, width=ARROW_WIDTH)
    base = (
        tip[0] - ARROW_HEAD_LENGTH * ux,
        tip[1] - ARROW_HEAD_LENGTH * uy,
    )
    left = (base[0] + ARROW_HEAD_WIDTH * px, base[1] + ARROW_HEAD_WIDTH * py)
    right = (base[0] - ARROW_HEAD_WIDTH * px, base[1] - ARROW_HEAD_WIDTH * py)
    draw.polygon((tip, left, right), fill=color)


def _draw_disk(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = center
    radius = TOKEN_RADIUS
    draw.ellipse(
        (round(x - radius), round(y - radius), round(x + radius), round(y + radius)),
        fill=color,
        outline=WHITE,
        width=2,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((35, 42, 925, 45), fill=FRAME_ACCENT)
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_RECTS[index]
        draw.rounded_rectangle(
            (x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3
        )
        draw.rectangle((x0 + 14, y0 + 14, x0 + 44, y0 + 44), fill=WHITE, outline=FRAME_ACCENT, width=1)
        _draw_bitmap(draw, PANEL_LABELS[index], (x0 + 19, y0 + 17))

        blue = (float(panel["blue_x"]), float(panel["blue_y"]))
        orange = (float(panel["orange_x"]), float(panel["orange_y"]))
        _draw_arrow(
            draw,
            blue,
            float(panel["blue_angle_deg"]),
            float(panel["blue_length"]),
            BLUE_ARROW,
        )
        _draw_arrow(
            draw,
            orange,
            float(panel["orange_angle_deg"]),
            float(panel["orange_length"]),
            ORANGE_ARROW,
        )
        _draw_disk(draw, blue, BLUE_DISK)
        _draw_disk(draw, orange, ORANGE_DISK)
    return image


def _post_distance(
    blue_x: float,
    blue_y: float,
    blue_angle: float,
    blue_length: float,
    orange_x: float,
    orange_y: float,
    orange_angle: float,
    orange_length: float,
) -> float:
    blue = _post_center(
        blue_x,
        blue_y,
        blue_angle,
        blue_length,
    )
    orange = _post_center(
        orange_x,
        orange_y,
        orange_angle,
        orange_length,
    )
    return math.hypot(blue[0] - orange[0], blue[1] - orange[1])


def _collision_score(
    blue_x: float,
    blue_y: float,
    blue_angle: float,
    blue_length: float,
    orange_x: float,
    orange_y: float,
    orange_angle: float,
    orange_length: float,
    token_radius: float,
) -> float:
    """Signed clearance after both visible motion vectors are applied."""

    return _post_distance(
        blue_x,
        blue_y,
        blue_angle,
        blue_length,
        orange_x,
        orange_y,
        orange_angle,
        orange_length,
    ) - 2.0 * token_radius


def _current_distance(blue_x: float, blue_y: float, orange_x: float, orange_y: float) -> float:
    return math.hypot(
        blue_x - orange_x,
        blue_y - orange_y,
    )


def analytic_gold(scene: dict) -> str:
    scores = [
        _collision_score(
            float(panel["blue_x"]),
            float(panel["blue_y"]),
            float(panel["blue_angle_deg"]),
            float(panel["blue_length"]),
            float(panel["orange_x"]),
            float(panel["orange_y"]),
            float(panel["orange_angle_deg"]),
            float(panel["orange_length"]),
            float(panel["token_radius"]),
        )
        for panel in scene["panels"]
    ]
    inside = [score < 0.0 for score in scores]
    if sum(inside) == 1:
        index = inside.index(True)
    else:
        index = min(range(len(scores)), key=lambda item: scores[item])
    return f"panel-{chr(65 + index)}"


def margin(scene: dict) -> float:
    scores = [
        _collision_score(
            float(panel["blue_x"]),
            float(panel["blue_y"]),
            float(panel["blue_angle_deg"]),
            float(panel["blue_length"]),
            float(panel["orange_x"]),
            float(panel["orange_y"]),
            float(panel["orange_angle_deg"]),
            float(panel["orange_length"]),
            float(panel["token_radius"]),
        )
        for panel in scene["panels"]
    ]
    current_clearance = [
        _current_distance(
            float(panel["blue_x"]),
            float(panel["blue_y"]),
            float(panel["orange_x"]),
            float(panel["orange_y"]),
        )
        - 2.0 * float(panel["token_radius"])
        for panel in scene["panels"]
    ]
    return float(min([abs(score) for score in scores] + current_clearance))


def is_quarantined(scene: dict) -> bool:
    scores = [
        _collision_score(
            float(panel["blue_x"]),
            float(panel["blue_y"]),
            float(panel["blue_angle_deg"]),
            float(panel["blue_length"]),
            float(panel["orange_x"]),
            float(panel["orange_y"]),
            float(panel["orange_angle_deg"]),
            float(panel["orange_length"]),
            float(panel["token_radius"]),
        )
        for panel in scene["panels"]
    ]
    current_clearance = [
        _current_distance(
            float(panel["blue_x"]),
            float(panel["blue_y"]),
            float(panel["orange_x"]),
            float(panel["orange_y"]),
        )
        - 2.0 * float(panel["token_radius"])
        for panel in scene["panels"]
    ]
    return (
        sum(score < 0.0 for score in scores) != 1
        or min(abs(score) for score in scores) < QUARANTINE_MARGIN
        or min(current_clearance) < QUARANTINE_MARGIN
    )
