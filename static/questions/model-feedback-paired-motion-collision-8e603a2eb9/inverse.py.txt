"""Independent image-only inverse for paired motion collision scenes."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)
TOKEN_RADIUS = 22.0
COLLISION_DIAMETER = 44.0
BLUE_DISK = np.array([42, 116, 214], dtype=np.uint8)
ORANGE_DISK = np.array([232, 126, 37], dtype=np.uint8)
BLUE_ARROW = np.array([29, 74, 139], dtype=np.uint8)
ORANGE_ARROW = np.array([164, 78, 24], dtype=np.uint8)


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(arr == color, axis=2)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected (row, column) components of a small exact mask."""

    coords = np.argwhere(mask)
    pending = {(int(row), int(col)) for row, col in coords}
    components: list[np.ndarray] = []
    while pending:
        start = pending.pop()
        stack = [start]
        found = [start]
        while stack:
            row, col = stack.pop()
            for drow in (-1, 0, 1):
                for dcol in (-1, 0, 1):
                    if not drow and not dcol:
                        continue
                    neighbor = (row + drow, col + dcol)
                    if neighbor in pending:
                        pending.remove(neighbor)
                        stack.append(neighbor)
                        found.append(neighbor)
        components.append(np.asarray(found, dtype=float))
    return components


def _centroid(points: np.ndarray) -> tuple[float, float]:
    if len(points) == 0:
        raise ValueError("missing colored component")
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _crop_mask(mask: np.ndarray, rect: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = rect
    return mask[y0 : y1 + 1, x0 : x1 + 1]


def _disk_center(arr: np.ndarray, rect: tuple[int, int, int, int], color: np.ndarray) -> tuple[float, float]:
    points = np.argwhere(_crop_mask(_mask(arr, color), rect))
    if len(points) < 300:
        raise ValueError("disk component is missing")
    x0, y0, _, _ = rect
    points = points.astype(float)
    points[:, 0] += y0
    points[:, 1] += x0
    return _centroid(points)


def _arrow_components(
    arr: np.ndarray, rect: tuple[int, int, int, int], color: np.ndarray
) -> list[np.ndarray]:
    components = _components(_crop_mask(_mask(arr, color), rect))
    x0, y0, _, _ = rect
    translated = []
    for component in components:
        if len(component) >= 80:
            component = component.copy()
            component[:, 0] += y0
            component[:, 1] += x0
            translated.append(component)
    if len(translated) != 1:
        raise ValueError(f"expected one motion arrow, found {len(translated)}")
    return translated


def _motion_from_arrow(component: np.ndarray, center: tuple[float, float]) -> tuple[float, float]:
    xy = np.column_stack((component[:, 1], component[:, 0]))
    relative = xy - np.asarray(center, dtype=float)
    radii = np.linalg.norm(relative, axis=1)
    maximum = float(radii.max())
    tip_points = xy[radii >= maximum - 1.0]
    tip = tip_points.mean(axis=0)
    vector = tip - np.asarray(center, dtype=float)
    visible_radius = float(np.linalg.norm(vector))
    if visible_radius <= 35.0:
        raise ValueError("motion arrow is too short")
    return float(vector[0]), float(vector[1])


def _post_score(arr: np.ndarray, index: int) -> float:
    rect = PANEL_RECTS[index]
    blue = _disk_center(arr, rect, BLUE_DISK)
    orange = _disk_center(arr, rect, ORANGE_DISK)
    blue_arrows = _arrow_components(arr, rect, BLUE_ARROW)
    orange_arrows = _arrow_components(arr, rect, ORANGE_ARROW)
    blue_move = _motion_from_arrow(blue_arrows[0], blue)
    orange_move = _motion_from_arrow(orange_arrows[0], orange)
    blue_post = (blue[0] + blue_move[0], blue[1] + blue_move[1])
    orange_post = (orange[0] + orange_move[0], orange[1] + orange_move[1])
    return math.hypot(blue_post[0] - orange_post[0], blue_post[1] - orange_post[1]) - COLLISION_DIAMETER


def decision_from_image(image: Image.Image) -> str:
    """Return the unique panel whose imagined moved disks overlap."""

    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        return "ambiguous"
    arr = np.asarray(rgb)
    scores = [_post_score(arr, index) for index in range(4)]
    inside = [score < 0.0 for score in scores]
    if sum(inside) != 1:
        return "ambiguous"
    return f"panel-{chr(65 + inside.index(True))}"
