"""Deterministic renderer for global compatibility matching."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 590
N = 7
GRID_X = 260
GRID_Y = 130
CELL = 58
ANSWERS = tuple("ABCDEFG")

RGB = {
    "background": (248, 247, 242),
    "ink": (31, 40, 54),
    "muted": (103, 116, 132),
    "cell": (235, 240, 244),
    "mark": (27, 116, 184),
    "query": (208, 50, 62),
    "panel": (222, 228, 234),
}

GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "G": ("01111", "10000", "10000", "10111", "10001", "10001", "01111"),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _unique_matchings(allowed: list[list[bool]], limit: int = 2) -> list[tuple[int, ...]]:
    results: list[tuple[int, ...]] = []
    assignment = [-1] * N

    def visit(used: set[int]) -> None:
        if len(results) >= limit:
            return
        if len(used) == N:
            results.append(tuple(assignment))
            return
        remaining = [row for row in range(N) if assignment[row] < 0]
        row = min(remaining, key=lambda item: sum(allowed[item][col] and col not in used for col in range(N)))
        for column in range(N):
            if allowed[row][column] and column not in used:
                assignment[row] = column
                visit(used | {column})
                assignment[row] = -1

    visit(set())
    return results


def _validate_scene(scene: dict) -> None:
    if set(scene) != {"allowed", "query_row"}:
        raise ValueError("scene fields do not match the world schema")
    allowed = scene["allowed"]
    if not isinstance(allowed, list) or len(allowed) != N:
        raise ValueError("allowed must contain seven rows")
    if any(not isinstance(row, list) or len(row) != N for row in allowed):
        raise ValueError("allowed must be a seven by seven matrix")
    if any(not isinstance(value, bool) for row in allowed for value in row):
        raise ValueError("allowed entries must be Boolean")
    query_row = scene["query_row"]
    if isinstance(query_row, bool) or not isinstance(query_row, int) or not 0 <= query_row < N:
        raise ValueError("query_row is outside the matrix")


def sample_scene(seed: int) -> dict:
    """Build a shuffled DAG-shaped compatibility graph with one perfect matching."""

    seed = int(seed)
    rng = random.Random((seed ^ 0x6C0B41A7) & 0xFFFFFFFFFFFFFFFF)
    topological = list(range(N))
    rng.shuffle(topological)
    logical = [[False] * N for _ in range(N)]
    for rank, row_id in enumerate(topological):
        logical[row_id][row_id] = True
        if rank:
            logical[row_id][topological[rank - 1]] = True
        if rank > 1:
            logical[row_id][rng.choice(topological[: rank - 1])] = True
        if rank > 3 and rng.random() < 0.7:
            logical[row_id][rng.choice(topological[: rank - 2])] = True

    query_id = topological[4]
    row_order = list(range(N))
    rng.shuffle(row_order)
    query_row = row_order.index(query_id)

    desired_column = seed % N
    other_columns = [value for value in range(N) if value != query_id]
    rng.shuffle(other_columns)
    column_order = other_columns[:desired_column] + [query_id] + other_columns[desired_column:]
    allowed = [
        [logical[row_order[row]][column_order[column]] for column in range(N)]
        for row in range(N)
    ]
    scene = {"allowed": allowed, "query_row": query_row}
    _validate_scene(scene)
    if len(_unique_matchings(allowed)) != 1:
        raise RuntimeError("sampler failed to construct a unique matching")
    return scene


def _draw_glyph(draw: ImageDraw.ImageDraw, center_x: int, top: int, glyph: str) -> None:
    scale = 4
    left = center_x - 10
    for row, bits in enumerate(GLYPHS[glyph]):
        for column, bit in enumerate(bits):
            if bit == "1":
                x = left + column * scale
                y = top + row * scale
                draw.rectangle((x, y, x + scale - 1, y + scale - 1), fill=RGB["ink"])


def render(scene: dict) -> Image.Image:
    """Render one high-contrast compatibility worksheet."""

    _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["background"])
    draw = ImageDraw.Draw(image)
    title = _font(24, bold=True)
    label = _font(17, bold=True)
    small = _font(15, bold=True)

    draw.text((34, 24), "ONE-TO-ONE PAIRING", fill=RGB["ink"], font=title)
    draw.ellipse((39, 77, 65, 103), fill=RGB["mark"])
    draw.text((76, 80), "allowed", fill=RGB["muted"], font=label)
    draw.rounded_rectangle((432, 25, 716, 66), radius=8, fill=RGB["panel"])
    draw.text((451, 35), "USE EACH COLUMN ONCE", fill=RGB["ink"], font=small)

    for column, answer in enumerate(ANSWERS):
        center_x = GRID_X + column * CELL + CELL // 2
        _draw_glyph(draw, center_x, 88, answer)

    for row in range(N):
        center_y = GRID_Y + row * CELL + CELL // 2
        for column in range(N):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=RGB["ink"])
            draw.rectangle((left + 3, top + 3, left + CELL - 4, top + CELL - 4), fill=RGB["cell"])
            if scene["allowed"][row][column]:
                center_x = left + CELL // 2
                draw.ellipse((center_x - 17, center_y - 17, center_x + 17, center_y + 17), fill=RGB["mark"])

    query_y = GRID_Y + int(scene["query_row"]) * CELL + CELL // 2
    draw.text((37, query_y - 12), "QUERY", fill=RGB["query"], font=label)
    draw.rectangle((145, query_y - 5, 231, query_y + 5), fill=RGB["query"])
    draw.polygon(((252, query_y), (230, query_y - 16), (230, query_y + 16)), fill=RGB["query"])
    draw.text((260, 554), "blue dot = allowed pairing", fill=RGB["muted"], font=small)
    return image


def analytic_gold(scene: dict) -> str:
    """Solve the visible all-different constraint and return the query partner."""

    _validate_scene(scene)
    matchings = _unique_matchings(scene["allowed"])
    if len(matchings) != 1:
        raise ValueError("matrix does not have exactly one complete matching")
    return ANSWERS[matchings[0][scene["query_row"]]]


def margin(scene: dict) -> float:
    """One discrete compatibility-cell change is the nearest assignment boundary."""

    _validate_scene(scene)
    if len(_unique_matchings(scene["allowed"])) != 1:
        return 0.0
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        _validate_scene(scene)
        return len(_unique_matchings(scene["allowed"])) != 1
    except (KeyError, TypeError, ValueError):
        return True
