"""Independent image-only inverse for global compatibility matching."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 760
HEIGHT = 590
N = 7
GRID_X = 260
GRID_Y = 130
CELL = 58
ANSWERS = tuple("ABCDEFG")

BACKGROUND = (248, 247, 242)
INK = (31, 40, 54)
CELL_RGB = (235, 240, 244)
MARK = (27, 116, 184)
QUERY = (208, 50, 62)

GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "G": ("01111", "10000", "10000", "10111", "10001", "10001", "01111"),
}


class OracleError(ValueError):
    """Raised when pixels do not encode one intact matching worksheet."""


def _uniform(array: np.ndarray, box: tuple[int, int, int, int], color: tuple[int, int, int], what: str) -> None:
    left, top, right, bottom = box
    patch = array[top:bottom, left:right]
    if patch.size == 0 or not np.all(patch == np.asarray(color, dtype=np.uint8)):
        raise OracleError(f"{what} is missing or damaged")


def _validate_glyph(array: np.ndarray, center_x: int, top: int, glyph: str) -> None:
    scale = 4
    left = center_x - 10
    for row, bits in enumerate(GLYPHS[glyph]):
        for column, bit in enumerate(bits):
            color = INK if bit == "1" else BACKGROUND
            x = left + column * scale + 1
            y = top + row * scale + 1
            _uniform(array, (x, y, x + 2, y + 2), color, f"column glyph {glyph}")


def _solve(allowed: list[list[bool]]) -> tuple[int, ...]:
    results: list[tuple[int, ...]] = []
    assignment = [-1] * N

    def visit(used: set[int]) -> None:
        if len(results) > 1:
            return
        if len(used) == N:
            results.append(tuple(assignment))
            return
        remaining = [row for row in range(N) if assignment[row] < 0]
        row = min(remaining, key=lambda item: sum(allowed[item][col] and col not in used for col in range(N)))
        for column in range(N):
            if allowed[row][column] and column not in used:
                assignment[row] = column
                visit(used | {column})
                assignment[row] = -1

    visit(set())
    if len(results) != 1:
        raise OracleError(f"visible matrix has {len(results)} complete matchings, expected one")
    return results[0]


def decision_from_image(image: Image.Image) -> str:
    """Reconstruct and solve the compatibility assignment from final pixels only."""

    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected a {WIDTH} by {HEIGHT} raster")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (HEIGHT, WIDTH, 3):
        raise OracleError("expected one RGB worksheet")
    for x, y in ((0, 0), (WIDTH - 1, 0), (0, HEIGHT - 1), (WIDTH - 1, HEIGHT - 1)):
        if tuple(int(value) for value in array[y, x]) != BACKGROUND:
            raise OracleError("canvas corners do not contain the declared background")

    for column, glyph in enumerate(ANSWERS):
        _validate_glyph(array, GRID_X + column * CELL + CELL // 2, 88, glyph)

    query_rows: list[int] = []
    allowed: list[list[bool]] = []
    mark_array = np.asarray(MARK, dtype=np.uint8)
    cell_array = np.asarray(CELL_RGB, dtype=np.uint8)
    for row in range(N):
        center_y = GRID_Y + row * CELL + CELL // 2
        arrow_patch = array[center_y - 4 : center_y + 5, 160:210]
        if np.all(arrow_patch == np.asarray(QUERY, dtype=np.uint8)):
            query_rows.append(row)
        elif not np.all(arrow_patch == np.asarray(BACKGROUND, dtype=np.uint8)):
            raise OracleError(f"query marker region {row} is ambiguous")

        values: list[bool] = []
        for column in range(N):
            left = GRID_X + column * CELL
            top = GRID_Y + row * CELL
            _uniform(array, (left + 8, top, left + CELL - 8, top + 3), INK, "matrix frame")
            center_x = left + CELL // 2
            patch = array[center_y - 7 : center_y + 8, center_x - 7 : center_x + 8]
            if np.all(patch == mark_array):
                values.append(True)
            elif np.all(patch == cell_array):
                values.append(False)
            else:
                raise OracleError(f"matrix cell {row},{column} is ambiguous")
        allowed.append(values)

    if len(query_rows) != 1:
        raise OracleError("worksheet must contain exactly one broad query arrow")
    matching = _solve(allowed)
    return ANSWERS[matching[query_rows[0]]]
