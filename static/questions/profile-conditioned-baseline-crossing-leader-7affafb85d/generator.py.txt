"""Deterministic scientific small-multiple renderer and analytic arm."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 600
PANEL_W, PANEL_H = 330, 210
PANEL_ORIGINS = ((35, 92), (395, 92), (35, 342), (395, 342))
X_OFFSETS = (38, 81, 124, 167, 210, 253, 296)
BASELINE_OFFSET = 112
LABELS = ("A", "B", "C", "D")
THEMES = (
    {
        "line": (89, 163, 181),
        "marker": (20, 73, 92),
        "baseline": (91, 99, 110),
        "accent": (231, 244, 247),
    },
    {
        "line": (198, 139, 97),
        "marker": (111, 60, 34),
        "baseline": (82, 96, 113),
        "accent": (250, 241, 231),
    },
)


def _font(size: int) -> ImageFont.ImageFont:
    return ImageFont.load_default(size=size)


def _crossing_count(values: Iterable[int]) -> int:
    values = list(values)
    return sum((a > 0) != (b > 0) for a, b in zip(values, values[1:]))


def _offset_sequence(rng: random.Random, crossing_count: int) -> list[int]:
    """Create seven clear signed offsets with exactly the requested sign changes."""
    flip_edges = set(rng.sample(range(6), crossing_count))
    sign = rng.choice((-1, 1))
    values: list[int] = []
    for index in range(7):
        # Shared amplitude range prevents crossing count from being a height cue.
        amplitude = 24 + rng.choice((-3, -2, -1, 0, 1, 2, 3))
        values.append(sign * amplitude)
        if index in flip_edges:
            sign *= -1
    assert _crossing_count(values) == crossing_count
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x4C6F6F70)
    winner = int(seed) % 4
    counts = [2, 3, 4, 5] if int(seed) % 3 else [1, 2, 3, 6]
    winning_count = max(counts)
    distractors = [value for value in counts if value != winning_count]
    rng.shuffle(distractors)
    assigned: list[int] = []
    for panel in range(4):
        assigned.append(winning_count if panel == winner else distractors.pop())
    return {
        "offsets": [_offset_sequence(rng, count) for count in assigned],
        "theme": rng.randrange(len(THEMES)),
        "shift_x": rng.randint(-4, 4),
        "shift_y": rng.randint(-3, 3),
    }


def analytic_gold(scene: dict) -> str:
    counts = [_crossing_count(values) for values in scene["offsets"]]
    best = max(counts)
    if counts.count(best) != 1:
        raise ValueError("analytic gold is undefined for a tied maximum")
    return LABELS[counts.index(best)]


def margin(scene: dict) -> float:
    counts = sorted((_crossing_count(values) for values in scene["offsets"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    sequences = scene["offsets"]
    counts = [_crossing_count(values) for values in sequences]
    if counts.count(max(counts)) != 1:
        return True
    if any(abs(value) < 12 for values in sequences for value in values):
        return True
    # A crossing fraction closer than 0.18 of an edge to either endpoint is fragile.
    for values in sequences:
        for left, right in zip(values, values[1:]):
            if (left > 0) != (right > 0):
                fraction = abs(left) / (abs(left) + abs(right))
                if min(fraction, 1.0 - fraction) < 0.18:
                    return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    offsets = scene["offsets"]
    theme = THEMES[int(scene["theme"])]
    shift_x = int(scene["shift_x"])
    shift_y = int(scene["shift_y"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 249, 251))
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((24, 15, 736, 70), radius=12, fill=(255, 255, 255), outline=(218, 222, 228), width=2)
    draw.text((42, 25), "BASELINE CROSSING COUNT", fill=(31, 41, 55), font=_font(20))
    draw.text(
        (42, 49),
        "Which panel's curve crosses its dashed baseline most often?",
        fill=(76, 86, 99),
        font=_font(14),
    )

    for panel, ((origin_x, origin_y), values) in enumerate(zip(PANEL_ORIGINS, offsets)):
        px, py = origin_x + shift_x, origin_y + shift_y
        draw.rounded_rectangle(
            (px, py, px + PANEL_W, py + PANEL_H),
            radius=10,
            fill=(255, 255, 255),
            outline=(205, 211, 219),
            width=2,
        )
        draw.rounded_rectangle((px + 10, py + 9, px + 39, py + 37), radius=7, fill=theme["accent"])
        draw.text((px + 19, py + 12), LABELS[panel], fill=(48, 55, 66), font=_font(19))
        draw.text((px + 48, py + 14), "7 checkpoints", fill=(107, 116, 128), font=_font(13))

        baseline_y = py + BASELINE_OFFSET
        for x_offset in X_OFFSETS:
            x = px + x_offset
            draw.line((x, py + 48, x, py + 184), fill=(236, 239, 243), width=1)
        # Unique exact baseline color is intentionally recoverable by the inverse arm.
        dash_start, dash_end = px + 28, px + 310
        for x in range(dash_start, dash_end, 16):
            draw.line((x, baseline_y, min(x + 9, dash_end), baseline_y), fill=theme["baseline"], width=2)
        draw.text((px + 235, baseline_y - 20), "baseline", fill=(126, 134, 145), font=_font(12))

        points = [(px + xoff, baseline_y - int(value)) for xoff, value in zip(X_OFFSETS, values)]
        # Four equal dots per edge make every panel's connector ink identical;
        # crossing count cannot be recovered from a total-ink shortcut.
        for (x1, y1), (x2, y2) in zip(points, points[1:]):
            for fraction in (0.23, 0.41, 0.59, 0.77):
                x = round(x1 + fraction * (x2 - x1))
                y = round(y1 + fraction * (y2 - y1))
                draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=theme["line"])
        for x, y in points:
            draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=theme["marker"], outline=(255, 255, 255), width=2)
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=theme["marker"])

    draw.text(
        (380, 580),
        "Count adjacent-marker side changes; no marker lies on a baseline.",
        anchor="mm",
        fill=(86, 96, 109),
        font=_font(13),
    )
    return image
