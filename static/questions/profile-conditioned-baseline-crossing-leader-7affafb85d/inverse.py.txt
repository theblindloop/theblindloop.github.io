"""Independent image-only inverse arm for baseline-crossing decisions."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
MARKER_COLORS = ((20, 73, 92), (111, 60, 34))
BASELINE_COLORS = ((91, 99, 110), (82, 96, 113))


def _exact_palette_mask(array: np.ndarray, colors: tuple[tuple[int, int, int], ...]) -> np.ndarray:
    mask = np.zeros(array.shape[:2], dtype=bool)
    for color in colors:
        mask |= np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    return mask


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    """Return x/y centroids and sizes of 8-connected components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(xs) >= minimum:
            result.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys)), len(xs)))
    return result


def _split_at_largest_gap(values: list[tuple[float, float, int]], coordinate: int) -> tuple[list, list]:
    ordered = sorted(values, key=lambda item: item[coordinate])
    gaps = [ordered[index + 1][coordinate] - ordered[index][coordinate] for index in range(len(ordered) - 1)]
    if not gaps:
        raise ValueError("cannot split an empty or singleton component set")
    split = max(range(len(gaps)), key=gaps.__getitem__) + 1
    return ordered[:split], ordered[split:]


def _panel_marker_groups(components: list[tuple[float, float, int]]) -> list[list[tuple[float, float, int]]]:
    if len(components) != 28:
        raise ValueError(f"expected 28 checkpoint markers, found {len(components)}")
    top, bottom = _split_at_largest_gap(components, 1)
    if len(top) != 14 or len(bottom) != 14:
        raise ValueError("checkpoint rows are not separable into two groups of fourteen")
    groups: list[list[tuple[float, float, int]]] = []
    for row in (top, bottom):
        left, right = _split_at_largest_gap(row, 0)
        if len(left) != 7 or len(right) != 7:
            raise ValueError("checkpoint row is not separable into two seven-marker panels")
        groups.extend((sorted(left), sorted(right)))
    return groups


def _baseline_y(mask: np.ndarray, group: list[tuple[float, float, int]]) -> float:
    xs = [item[0] for item in group]
    ys = [item[1] for item in group]
    left = max(0, int(min(xs)) - 14)
    right = min(mask.shape[1], int(max(xs)) + 15)
    target = float(np.median(ys))
    y_coords = np.argwhere(mask[:, left:right])[:, 0]
    if len(y_coords) < 20:
        raise ValueError("visible dashed baseline could not be recovered")
    counts = np.bincount(y_coords, minlength=mask.shape[0])
    plausible = np.flatnonzero(counts >= 8)
    if not len(plausible):
        raise ValueError("dashed baseline has no supported raster row")
    chosen = int(plausible[np.argmin(np.abs(plausible - target))])
    close = y_coords[np.abs(y_coords - chosen) <= 1]
    return float(np.median(close))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    marker_mask = _exact_palette_mask(array, MARKER_COLORS)
    baseline_mask = _exact_palette_mask(array, BASELINE_COLORS)
    markers = _components(marker_mask, minimum=45)
    groups = _panel_marker_groups(markers)
    counts: list[int] = []
    for group in groups:
        baseline = _baseline_y(baseline_mask, group)
        ordered = sorted(group, key=lambda item: item[0])
        sides = [centroid_y < baseline for _, centroid_y, _ in ordered]
        if any(abs(centroid_y - baseline) < 10 for _, centroid_y, _ in ordered):
            raise ValueError("checkpoint is too close to its baseline")
        counts.append(sum(left != right for left, right in zip(sides, sides[1:])))
    best = max(counts)
    if counts.count(best) != 1:
        raise ValueError(f"pixel decision is tied: {counts}")
    return LABELS[counts.index(best)]
