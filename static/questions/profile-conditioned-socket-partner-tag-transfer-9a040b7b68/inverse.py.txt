from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_IDENTITY_RGB = {
    "teal": (0, 166, 166),
    "orange": (242, 142, 43),
    "lime": (124, 179, 66),
    "navy": (76, 91, 155),
}
_TAG_RGB = {
    "red": (209, 73, 91),
    "blue": (52, 120, 192),
    "gold": (212, 167, 44),
    "purple": (141, 90, 199),
}
_CONNECTOR_RGB = (24, 175, 196)
_RING_RGB = (232, 62, 140)


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 34.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    outputs = []
    ys, xs = np.nonzero(mask)
    for y0, x0 in zip(ys.tolist(), xs.tolist(), strict=True):
        if seen[y0, x0]:
            continue
        q = deque([(y0, x0)])
        seen[y0, x0] = True
        sx = sy = count = 0
        while q:
            y, x = q.popleft()
            sx += x
            sy += y
            count += 1
            for yy, xx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    q.append((yy, xx))
        if count >= minimum:
            outputs.append((sx / count, sy / count, count))
    return outputs


def _line_support(mask: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> float:
    hits = 0
    total = 0
    h, w = mask.shape
    for t in np.linspace(0.22, 0.78, 45):
        x = int(round(a[0] * (1-t) + b[0] * t))
        y = int(round(a[1] * (1-t) + b[1] * t))
        if 2 <= x < w-2 and 2 <= y < h-2:
            hits += bool(mask[y-2:y+3, x-2:x+3].any())
            total += 1
    return hits / max(total, 1)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[1] < 700 or arr.shape[0] < 450:
        raise ValueError("unexpected raster size")

    reference: dict[str, tuple[float, float]] = {}
    lookup: dict[str, tuple[float, float]] = {}
    for name, rgb in _IDENTITY_RGB.items():
        comps = _components(_mask(arr, rgb), minimum=700)
        left = [(x, y) for x, y, _ in comps if x < arr.shape[1] / 2]
        right = [(x, y) for x, y, _ in comps if x > arr.shape[1] / 2]
        if len(left) != 1 or len(right) != 1:
            raise ValueError(f"identity component failure for {name}: {len(left)}/{len(right)}")
        reference[name] = left[0]
        lookup[name] = right[0]

    rings = _components(_mask(arr, _RING_RGB), minimum=500)
    if len(rings) != 1:
        raise ValueError("missing or ambiguous magenta marker")
    rx, ry, _ = rings[0]
    ranked_sources = sorted(
        ((math.hypot(x-rx, y-ry), name) for name, (x, y) in reference.items())
    )
    if ranked_sources[0][0] > 8 or ranked_sources[1][0] - ranked_sources[0][0] < 40:
        raise ValueError("marker is not uniquely bound to a socket")
    source = ranked_sources[0][1]

    connector_mask = _mask(arr, _CONNECTOR_RGB)
    supports = sorted(
        ((_line_support(connector_mask, reference[source], pos), name)
         for name, pos in reference.items() if name != source),
        reverse=True,
    )
    if supports[0][0] < 0.72 or supports[0][0] - supports[1][0] < 0.35:
        raise ValueError(f"connector partner is ambiguous: {supports}")
    partner = supports[0][1]

    tag_centers: dict[str, tuple[float, float]] = {}
    for tag, rgb in _TAG_RGB.items():
        comps = _components(_mask(arr, rgb), minimum=700)
        comps = [(x, y) for x, y, _ in comps if x > arr.shape[1] / 2]
        if len(comps) != 1:
            raise ValueError(f"tag component failure for {tag}: {len(comps)}")
        tag_centers[tag] = comps[0]

    ix, iy = lookup[partner]
    ranked_tags = sorted(
        (abs(tx-ix) + 0.22*abs((ty-iy)-74), tag)
        for tag, (tx, ty) in tag_centers.items()
        if ty > iy
    )
    if len(ranked_tags) < 1 or ranked_tags[0][0] > 18:
        raise ValueError("no square tag is directly below matched lookup socket")
    if len(ranked_tags) > 1 and ranked_tags[1][0] - ranked_tags[0][0] < 25:
        raise ValueError("lookup tag binding is ambiguous")
    return ranked_tags[0][1]
