from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 900, 600, 2
IDENTITIES = ("teal", "orange", "lime", "navy")
ID_RGB = {
    "teal": "#00A6A6",
    "orange": "#F28E2B",
    "lime": "#7CB342",
    "navy": "#4C5B9B",
}
TAGS = ("red", "blue", "gold", "purple")
TAG_RGB = {
    "red": "#D1495B",
    "blue": "#3478C0",
    "gold": "#D4A72C",
    "purple": "#8D5AC7",
}
CONNECTOR = "#18AFC4"
RING = "#E83E8C"
INK = "#22252A"
GRAY = "#7A8088"

REF_SLOTS = ((135, 225), (330, 225), (135, 420), (330, 420))
LOOKUP_SLOTS = ((565, 205), (765, 205), (565, 395), (765, 395))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    ref_ids = list(IDENTITIES)
    rng.shuffle(ref_ids)
    orientation = int(seed) % 2
    slot_pairs = ((0, 1), (2, 3)) if orientation == 0 else ((0, 2), (1, 3))
    reference_positions = {identity: list(REF_SLOTS[i]) for i, identity in enumerate(ref_ids)}
    reference_pairs = [[ref_ids[a], ref_ids[b]] for a, b in slot_pairs]
    marked_identity = ref_ids[rng.randrange(4)]
    partner = next(b if a == marked_identity else a for a, b in reference_pairs if marked_identity in (a, b))

    lookup_ids = list(IDENTITIES)
    rng.shuffle(lookup_ids)
    lookup_positions = {identity: list(LOOKUP_SLOTS[i]) for i, identity in enumerate(lookup_ids)}

    desired = TAGS[int(seed) % len(TAGS)]
    remaining = [tag for tag in TAGS if tag != desired]
    rng.shuffle(remaining)
    tag_map = {partner: desired}
    for identity, tag in zip([x for x in IDENTITIES if x != partner], remaining, strict=True):
        tag_map[identity] = tag

    return {
        "reference_positions": reference_positions,
        "reference_pairs": reference_pairs,
        "marked_identity": marked_identity,
        "lookup_positions": lookup_positions,
        "tag_map": tag_map,
        "audit_nonce": int(seed) * 17 + 3,
    }


def _partner(scene: dict) -> str:
    marked = scene["marked_identity"]
    for pair in scene["reference_pairs"]:
        if marked in pair:
            return pair[1] if pair[0] == marked else pair[0]
    raise ValueError("marked socket is not in exactly one pair")


def analytic_gold(scene: dict) -> str:
    marked = scene["marked_identity"]
    pairs = scene["reference_pairs"]
    tags = scene["tag_map"]
    hits = [pair for pair in pairs if marked in pair]
    if len(hits) != 1 or len(hits[0]) != 2:
        raise ValueError("invalid reference matching")
    partner = hits[0][1] if hits[0][0] == marked else hits[0][0]
    return str(tags[partner])


def margin(scene: dict) -> float:
    positions = scene["reference_positions"]
    pairs = scene["reference_pairs"]
    lengths = []
    for a, b in pairs:
        ax, ay = positions[a]
        bx, by = positions[b]
        lengths.append(math.hypot(ax - bx, ay - by) - 58.0)
    return float(min(lengths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["audit_nonce"] = int(twin.get("audit_nonce", 0)) + 1
    return [("audit_nonce_shift", twin)]


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE)


def _ellipse(draw: ImageDraw.ImageDraw, box, **kwargs) -> None:
    draw.ellipse(tuple(int(v * SCALE) for v in box), **kwargs)


def _rect(draw: ImageDraw.ImageDraw, box, **kwargs) -> None:
    draw.rounded_rectangle(tuple(int(v * SCALE) for v in box), radius=8 * SCALE, **kwargs)


def render(scene: dict) -> Image.Image:
    ref_pos = scene["reference_positions"]
    pairs = scene["reference_pairs"]
    marked = scene["marked_identity"]
    lookup_pos = scene["lookup_positions"]
    tags = scene["tag_map"]
    _ = scene.get("audit_nonce", 0)

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "#F7F4EE")
    d = ImageDraw.Draw(image)
    d.rounded_rectangle((30*SCALE, 90*SCALE, 430*SCALE, 555*SCALE), radius=18*SCALE,
                        fill="#FFFFFF", outline="#484C52", width=4*SCALE)
    d.rounded_rectangle((470*SCALE, 90*SCALE, 870*SCALE, 555*SCALE), radius=18*SCALE,
                        fill="#FFFFFF", outline="#484C52", width=4*SCALE)
    d.text((45*SCALE, 28*SCALE), "REFERENCE", font=_font(23), fill=INK)
    d.text((490*SCALE, 28*SCALE), "LOOKUP", font=_font(23), fill=INK)

    for a, b in pairs:
        ax, ay = ref_pos[a]
        bx, by = ref_pos[b]
        d.line((ax*SCALE, ay*SCALE, bx*SCALE, by*SCALE), fill=INK, width=18*SCALE)
        d.line((ax*SCALE, ay*SCALE, bx*SCALE, by*SCALE), fill=CONNECTOR, width=11*SCALE)

    for identity in IDENTITIES:
        x, y = ref_pos[identity]
        if identity == marked:
            _ellipse(d, (x-39, y-39, x+39, y+39), fill=RING, outline=INK, width=2*SCALE)
            _ellipse(d, (x-33, y-33, x+33, y+33), fill="#FFFFFF")
        _ellipse(d, (x-27, y-27, x+27, y+27), fill=ID_RGB[identity], outline=INK, width=3*SCALE)
        _ellipse(d, (x-7, y-7, x+7, y+7), fill="#FFFFFF", outline=INK, width=2*SCALE)

    for identity in IDENTITIES:
        x, y = lookup_pos[identity]
        d.line((x*SCALE, (y+27)*SCALE, x*SCALE, (y+58)*SCALE), fill=GRAY, width=6*SCALE)
        _rect(d, (x-22, y+54, x+22, y+94), fill=TAG_RGB[tags[identity]], outline=INK, width=3*SCALE)
        _ellipse(d, (x-27, y-27, x+27, y+27), fill=ID_RGB[identity], outline=INK, width=3*SCALE)
        _ellipse(d, (x-7, y-7, x+7, y+7), fill="#FFFFFF", outline=INK, width=2*SCALE)

    d.text((95*SCALE, 510*SCALE), "follow the marked pair", font=_font(15), fill="#555A62")
    d.text((545*SCALE, 510*SCALE), "read its square tag", font=_font(15), fill="#555A62")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
