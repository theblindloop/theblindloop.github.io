"""Pixel-only inverse for the spiral-pair carpet.

This module intentionally duplicates the small raster measurement rule rather
than importing any renderer state. It receives only a final RGB image.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

FRAME = np.asarray((28, 51, 76), dtype=np.float32)
N = 16
FRAME_BORDER = 12
TEMPLATE_A = (
    "....########....", "...##......##...", "..##........##..", "..##........##..",
    "..##........##..", "..##........##..", "...##......##...", "....########....",
)
TEMPLATE_B = (
    ".......##.......", "......####......", ".....######.....", "....########....",
    "....########....", ".....######.....", "......####......", ".......##.......",
)


def _spiral_order(n: int = N) -> list[tuple[int, int]]:
    top, left, bottom, right = 0, 0, n - 1, n - 1
    result: list[tuple[int, int]] = []
    while top <= bottom and left <= right:
        result.extend((top, col) for col in range(left, right + 1))
        top += 1
        result.extend((row, right) for row in range(top, bottom + 1))
        right -= 1
        if top <= bottom:
            result.extend((bottom, col) for col in range(right, left - 1, -1))
            bottom -= 1
        if left <= right:
            result.extend((row, left) for row in range(bottom, top - 1, -1))
            left += 1
    return result


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    ys, xs = np.nonzero(mask)
    if not len(xs):
        return []
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    sub = mask[y0:y1, x0:x1]
    h, w = sub.shape
    seen = np.zeros_like(sub, dtype=bool)
    output = []
    for sy, sx in zip(*np.nonzero(sub)):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        count = 0
        miny = maxy = int(sy)
        minx = maxx = int(sx)
        while stack:
            cy, cx = stack.pop()
            count += 1
            miny, maxy = min(miny, cy), max(maxy, cy)
            minx, maxx = min(minx, cx), max(maxx, cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and sub[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        output.append((count, x0 + minx, y0 + miny, x0 + maxx, y0 + maxy))
    return output


def _template_bits(rows: tuple[str, ...]) -> np.ndarray:
    return np.asarray([char == "#" for row in rows for char in row], dtype=np.uint8)


def _measure(image: Image.Image) -> list[int]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    frame_distance = np.linalg.norm(array - FRAME, axis=2)
    frame_mask = frame_distance < 12.0
    components = _components(frame_mask)
    if not components:
        raise ValueError("no visible navy frame")
    count, x0, y0, x1, y1 = max(components)
    if count < 1000 or (x1 - x0 + 1) < 16 * 13 + 2 * FRAME_BORDER:
        raise ValueError("frame is too small or absent")
    side = x1 - x0 + 1
    inner_side = side - 2 * FRAME_BORDER
    if inner_side <= 0 or inner_side % N:
        raise ValueError("cell pitch is not integral")
    cell = inner_side // N
    if not 13 <= cell <= 17:
        raise ValueError("unsupported cell pitch")
    inner_x, inner_y = x0 + FRAME_BORDER, y0 + FRAME_BORDER
    samples = []
    for row in range(N):
        for col in range(N):
            cy = inner_y + row * cell + cell // 2
            cx = inner_x + col * cell + cell // 2
            samples.append(array[cy, cx])
    samples_array = np.asarray(samples)
    # The question declares two tones; luminance ordering is image-observable.
    luminance = samples_array @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32)
    low, high = float(np.min(luminance)), float(np.max(luminance))
    if high - low < 25.0:
        raise ValueError("two raster tones are not separated")
    tone_low = samples_array[int(np.argmin(luminance))]
    tone_high = samples_array[int(np.argmax(luminance))]
    # A destructive ablation must be observable: every cell is expected to be
    # a solid raster tone. This independent integrity check turns erased cell
    # interiors into explicit oracle abstention rather than silently treating
    # the background as a third label.
    for row in range(N):
        for col in range(N):
            top = inner_y + row * cell
            left = inner_x + col * cell
            patch = array[top : top + cell, left : left + cell]
            d_low = np.linalg.norm(patch - tone_low, axis=2)
            d_high = np.linalg.norm(patch - tone_high, axis=2)
            if float(np.min(np.minimum(d_low, d_high))) < 0.0:
                raise ValueError("unreachable raster check")
            if float(np.max(np.minimum(d_low, d_high))) > 16.0:
                raise ValueError("raster cell is incomplete or corrupted")
    values = (luminance < (low + high) / 2.0).astype(np.uint8).tolist()
    stream = [values[row * N + col] for row, col in _spiral_order()]
    return [stream[index] ^ stream[index + 1] for index in range(0, len(stream), 2)]


def decision_from_image(image: Image.Image) -> str:
    output = np.asarray(_measure(image), dtype=np.uint8)
    score_a = int(np.count_nonzero(output != _template_bits(TEMPLATE_A)))
    score_b = int(np.count_nonzero(output != _template_bits(TEMPLATE_B)))
    return "A" if score_a <= score_b else "B"
