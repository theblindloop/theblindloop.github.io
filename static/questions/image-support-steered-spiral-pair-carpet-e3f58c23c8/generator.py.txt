"""Independent analytic side of the spiral-pair carpet world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
FRAME_BORDER = 12
N = 16
OUT_H = 8
OUT_W = 16
BACKGROUND = (247, 245, 239)
FRAME = (28, 51, 76)
NOTCH = (204, 51, 64)

TEMPLATE_A = (
    "....########....",
    "...##......##...",
    "..##........##..",
    "..##........##..",
    "..##........##..",
    "..##........##..",
    "...##......##...",
    "....########....",
)
TEMPLATE_B = (
    ".......##.......",
    "......####......",
    ".....######.....",
    "....########....",
    "....########....",
    ".....######.....",
    "......####......",
    ".......##.......",
)

TONE_PAIRS = (
    ((55, 76, 96), (226, 214, 180)),
    ((70, 63, 98), (220, 232, 202)),
    ((58, 91, 79), (233, 211, 190)),
    ((84, 66, 63), (211, 226, 224)),
)


def _spiral_order(n: int = N) -> list[tuple[int, int]]:
    """Clockwise square spiral beginning at the upper-left cell."""
    top, left, bottom, right = 0, 0, n - 1, n - 1
    result: list[tuple[int, int]] = []
    while top <= bottom and left <= right:
        for col in range(left, right + 1):
            result.append((top, col))
        top += 1
        for row in range(top, bottom + 1):
            result.append((row, right))
        right -= 1
        if top <= bottom:
            for col in range(right, left - 1, -1):
                result.append((bottom, col))
            bottom -= 1
        if left <= right:
            for row in range(bottom, top - 1, -1):
                result.append((row, left))
            left += 1
    return result


SPIRAL = _spiral_order()


def _template_bits(name: str) -> list[int]:
    rows = TEMPLATE_A if name == "A" else TEMPLATE_B
    return [int(char == "#") for row in rows for char in row]


def _output_bits(raster: list[list[int]]) -> list[int]:
    stream = [int(raster[row][col]) for row, col in SPIRAL]
    return [stream[index] ^ stream[index + 1] for index in range(0, len(stream), 2)]


def _scores(raster: list[list[int]]) -> tuple[int, int]:
    output = _output_bits(raster)
    a = _template_bits("A")
    b = _template_bits("B")
    return sum(x != y for x, y in zip(output, a)), sum(x != y for x, y in zip(output, b))


def _raster_for(seed: int, target: str) -> list[list[int]]:
    """Invert the declared transform while balancing the visible input ink."""
    rng = random.Random(seed * 1009 + 17)
    base = _template_bits(target)
    other = _template_bits("B" if target == "A" else "A")
    output = list(base)

    # A few deterministic near-boundary scenes make quarantine visible. Flip an
    # equal number in each direction so both labels keep the same output weight.
    if seed % 8 in (0, 1):
        one_to_zero = [i for i, (x, y) in enumerate(zip(base, other)) if x == 1 and y == 0]
        zero_to_one = [i for i, (x, y) in enumerate(zip(base, other)) if x == 0 and y == 1]
        rng.shuffle(one_to_zero)
        rng.shuffle(zero_to_one)
        for index in one_to_zero[:17] + zero_to_one[:17]:
            output[index] = other[index]
    elif seed % 8 in (2, 3):
        one_to_zero = [i for i, (x, y) in enumerate(zip(base, other)) if x == 1 and y == 0]
        zero_to_one = [i for i, (x, y) in enumerate(zip(base, other)) if x == 0 and y == 1]
        rng.shuffle(one_to_zero)
        rng.shuffle(zero_to_one)
        for index in one_to_zero[:15] + zero_to_one[:15]:
            output[index] = other[index]

    stream = [0] * (N * N)
    zero_pairs = [index for index, bit in enumerate(output) if bit == 0]
    rng.shuffle(zero_pairs)
    high_count = len(zero_pairs) // 2
    high_pairs = set(zero_pairs[:high_count])
    for pair_index, bit in enumerate(output):
        first = 2 * pair_index
        if bit:
            values = (1, 0) if rng.randrange(2) == 0 else (0, 1)
        else:
            values = (1, 1) if pair_index in high_pairs else (0, 0)
        stream[first], stream[first + 1] = values

    raster = [[0 for _ in range(N)] for _ in range(N)]
    for value, (row, col) in zip(stream, SPIRAL):
        raster[row][col] = int(value)
    return raster


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = "A" if seed % 2 == 0 else "B"
    cell_size = rng.choice((14, 15, 16))
    x = rng.randrange(112, 161)
    y = rng.randrange(112, 161)
    tone_a, tone_b = TONE_PAIRS[rng.randrange(len(TONE_PAIRS))]
    return {
        "raster": _raster_for(seed, target),
        "x": x,
        "y": y,
        "cell_size": cell_size,
        "tone_a": list(tone_a),
        "tone_b": list(tone_b),
    }


def render(scene: dict) -> Image.Image:
    cell_size = int(scene["cell_size"])
    x0, y0 = int(scene["x"]), int(scene["y"])
    side = N * cell_size + 2 * FRAME_BORDER
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x0 + side - 1, y0 + side - 1), fill=FRAME)
    inner_x, inner_y = x0 + FRAME_BORDER, y0 + FRAME_BORDER
    raster = scene["raster"]
    tone_a = tuple(int(value) for value in scene["tone_a"])
    tone_b = tuple(int(value) for value in scene["tone_b"])
    for row in range(N):
        for col in range(N):
            left = inner_x + col * cell_size
            top = inner_y + row * cell_size
            fill = tone_a if int(raster[row][col]) else tone_b
            draw.rectangle((left, top, left + cell_size - 1, top + cell_size - 1), fill=fill)
    # The notch is fully on the frame and cannot be confused with a raster tone.
    draw.rectangle((x0 + 3, y0 + 3, x0 + 9, y0 + 9), fill=NOTCH)
    return image


def analytic_gold(scene: dict) -> str:
    raster = scene["raster"]
    score_a, score_b = _scores(raster)
    return "A" if score_a <= score_b else "B"


def margin(scene: dict) -> float:
    score_a, score_b = _scores(scene["raster"])
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
