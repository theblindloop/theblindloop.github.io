from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

BACKGROUND = (252, 252, 248)
HALO = (161, 190, 218)
TRACE = (23, 52, 93)
START = (24, 165, 88)
FINISH = (220, 63, 69)
WHITE = (255, 255, 255)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    size = 512
    # Alternation makes every consecutive 24-scene stress run exactly balanced.
    if int(seed) % 2 == 0:
        signs = [1, 1] if rng.randrange(2) else [-1, -1]
    else:
        signs = [1, -1] if rng.randrange(2) else [-1, 1]
    return {
        "canvas_size": size,
        "center": [rng.choice((160, 224, 288, 352)), rng.choice((160, 224, 288, 352))],
        "rotation": rng.randrange(4),
        "turn_signs": signs,
        "shape_variant": rng.randrange(3),
    }


def _cross(a: tuple[float, float], b: tuple[float, float]) -> float:
    return a[0] * b[1] - a[1] * b[0]


def _template(signs: list[int], variant: int) -> list[tuple[float, float]]:
    # Equal x spacing keeps endpoint distance and segment count fixed.  The
    # finite set was chosen for large, plainly visible turn angles.
    amplitude = (12, 13, 14)[int(variant)]
    if signs[0] == signs[1]:
        if signs[0] > 0:
            ys = (0, -amplitude, -amplitude, 0)
        else:
            ys = (0, amplitude, amplitude, 0)
    else:
        if signs[0] > 0:
            ys = (0, -amplitude, amplitude, 0)
        else:
            ys = (0, amplitude, -amplitude, 0)
    xs = (-54, -18, 18, 54)
    points = [(float(x), float(y)) for x, y in zip(xs, ys, strict=True)]
    observed = []
    for i in (1, 2):
        u = (points[i][0] - points[i - 1][0], points[i][1] - points[i - 1][1])
        v = (points[i + 1][0] - points[i][0], points[i + 1][1] - points[i][1])
        observed.append(1 if _cross(u, v) > 0 else -1)
    if observed != [int(x) for x in signs]:
        raise AssertionError((observed, signs))
    return points


def _points(scene: dict) -> list[tuple[int, int]]:
    points = _template(scene["turn_signs"], scene["shape_variant"])
    turns = int(scene["rotation"]) % 4
    cx, cy = (int(x) for x in scene["center"])
    output = []
    for x, y in points:
        for _ in range(turns):
            x, y = -y, x
        output.append((int(round(cx + x)), int(round(cy + y))))
    return output


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    draw.line(points, fill=HALO, width=13, joint="curve")
    draw.line(points, fill=TRACE, width=3, joint="curve")
    for point, color in ((points[0], START), (points[-1], FINISH)):
        x, y = point
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=WHITE)
        draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    signs = scene["turn_signs"]
    return "same" if int(signs[0]) == int(signs[1]) else "opposite"


def margin(scene: dict) -> float:
    points = _template(scene["turn_signs"], scene["shape_variant"])
    values = []
    for i in (1, 2):
        u = (points[i][0] - points[i - 1][0], points[i][1] - points[i - 1][1])
        v = (points[i + 1][0] - points[i][0], points[i + 1][1] - points[i][1])
        values.append(abs(_cross(u, v)) / (math.hypot(*u) * math.hypot(*v)))
    return round(min(values), 12)


def is_quarantined(scene: dict) -> bool:
    size = int(scene["canvas_size"])
    edge = min(min(x, y, size - 1 - x, size - 1 - y) for x, y in _points(scene))
    return margin(scene) < 0.30 or edge < 24


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
