"""Pixel-only inverse arm for the station-removal question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 62, 62)),
    "green": np.array((42, 166, 91)),
    "blue": np.array((55, 112, 210)),
}


def _centroid(mask: np.ndarray, label: str) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError(f"cannot segment {label} station")
    return int(round(xs.mean())), int(round(ys.mean()))


def _component_reaches(mask: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    width = mask.shape[1]
    height = mask.shape[0]
    sx, sy = start
    gx, gy = goal
    if not (0 <= sx < width and 0 <= sy < height and mask[sy, sx]):
        return False
    pending = deque([(sx, sy)])
    seen = np.zeros(mask.shape, dtype=bool)
    seen[sy, sx] = True
    while pending:
        x, y = pending.popleft()
        if abs(x - gx) <= 2 and abs(y - gy) <= 2:
            return True
        for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                pending.append((nx, ny))
    return False


def _dark_components(arr: np.ndarray) -> list[tuple[int, int, int]]:
    mask = np.all(arr < 36, axis=2)
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        pending = deque([(int(x), int(y))])
        seen[y, x] = True
        pixels = []
        while pending:
            px, py = pending.popleft()
            pixels.append((px, py))
            for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    pending.append((nx, ny))
        if len(pixels) > 300:
            found.append((len(pixels), int(round(sum(p[0] for p in pixels) / len(pixels))), int(round(sum(p[1] for p in pixels) / len(pixels)))))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.shape[0] < 32 or arr.shape[1] < 32:
        raise ValueError("image is too small")

    centers = {}
    for name, rgb in PALETTE.items():
        distance = np.max(np.abs(arr - rgb), axis=2)
        centers[name] = _centroid(distance <= 12, name)

    terminals = _dark_components(arr)
    if len(terminals) != 2:
        raise ValueError(f"expected two black square terminals, found {len(terminals)}")
    terminal_centers = [(x, y) for _, x, y in sorted(terminals, reverse=True)]

    # Every route pixel is visibly non-background. Removing a station means
    # deleting a disk slightly larger than its outlined body, severing all
    # incident links exactly as the public question states.
    network = np.any(arr < 235, axis=2)
    yy, xx = np.ogrid[: arr.shape[0], : arr.shape[1]]
    cuts = []
    for name, (cx, cy) in centers.items():
        remaining = network.copy()
        remaining[(xx - cx) ** 2 + (yy - cy) ** 2 <= 23**2] = False
        if not _component_reaches(remaining, terminal_centers[0], terminal_centers[1]):
            cuts.append(name)
    if len(cuts) != 1:
        raise ValueError(f"expected one pixel-derived cut station, found {cuts}")
    return cuts[0]
