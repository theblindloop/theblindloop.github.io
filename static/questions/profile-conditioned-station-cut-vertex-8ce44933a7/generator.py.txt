"""Render cycle-rich station graphs with one terminal-separating colored vertex."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 250, 247)
LINK = (92, 105, 118)
OUTLINE = (48, 55, 62)
NEUTRAL = (214, 220, 225)
TERMINAL = (24, 27, 30)
COLORS = {
    "red": (220, 62, 62),
    "green": (42, 166, 91),
    "blue": (55, 112, 210),
}


def _transform(point: tuple[float, float], quarter_turns: int, dx: int, dy: int) -> list[int]:
    x, y = point
    cx, cy = WIDTH / 2, HEIGHT / 2
    for _ in range(quarter_turns % 4):
        x, y = cx - (y - cy), cy + (x - cx)
    return [int(round(x + dx)), int(round(y + dy))]


def sample_scene(seed: int) -> dict:
    """Return a balanced deterministic scene for one integer seed."""

    rng = random.Random(seed)
    # Two visible cycles joined through vertex a. Vertices b and c are degree-two
    # candidates on the cycles, so local degree cannot identify the articulation.
    base = {
        "t0": (120, 232),
        "b": (190, 132),
        "l1": (275, 205),
        "l2": (197, 326),
        "a": (335, 190),
        "r1": (395, 202),
        "c": (465, 116),
        "t1": (520, 218),
        "r2": (462, 324),
        "d0": (150, 273),
        "d1": (495, 274),
    }
    quarter_turns = (seed // 3) % 4
    dx = rng.randint(-11, 11)
    dy = rng.randint(-9, 9)
    # Neutral leaf stations vary visible distractor ink without affecting routes.
    if seed % 2:
        base["d0"] = (145, 286)
        base["d1"] = (492, 287)
    nodes = {name: _transform(point, quarter_turns, dx, dy) for name, point in base.items()}
    edges = [
        ["t0", "b"], ["b", "l1"], ["l1", "l2"], ["l2", "t0"],
        ["l1", "a"], ["a", "r1"],
        ["r1", "c"], ["c", "t1"], ["t1", "r2"], ["r2", "r1"],
        ["t0", "d0"], ["t1", "d1"],
    ]
    # Rotate the color assignment exactly with seed mod 3 for class balance.
    names = ["red", "green", "blue"]
    shift = seed % 3
    ordered_nodes = ["a", "b", "c"]
    colored_nodes = {names[(i + shift) % 3]: node for i, node in enumerate(ordered_nodes)}
    return {
        "nodes": nodes,
        "edges": edges,
        "terminals": ["t0", "t1"],
        "colored_nodes": colored_nodes,
        "variant": int(seed % 8),
    }


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    terminals = scene["terminals"]
    colored_nodes = scene["colored_nodes"]
    variant = scene["variant"]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # Draw links first so station bodies visibly define their incident endpoints.
    for u, v in edges:
        draw.line([tuple(nodes[u]), tuple(nodes[v])], fill=LINK, width=11)

    inverse_colors = {node: name for name, node in colored_nodes.items()}
    terminal_set = set(terminals)
    for name, (x, y) in nodes.items():
        if name in terminal_set:
            radius = 14
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=TERMINAL)
        else:
            radius = 18
            fill = COLORS[inverse_colors[name]] if name in inverse_colors else NEUTRAL
            draw.ellipse(
                (x - radius, y - radius, x + radius, y + radius),
                fill=fill,
                outline=OUTLINE,
                width=4,
            )
    # Variant is deliberately visible through a harmless tiny neutral legend dot.
    # It changes distractor ink but is disconnected from the route graph.
    lx = 18 + 9 * (variant % 4)
    ly = 18 + 9 * (variant // 4)
    draw.ellipse((lx - 3, ly - 3, lx + 3, ly + 3), fill=NEUTRAL)
    return image


def _connected_after_removal(edges: list[list[str]], terminals: list[str], removed: str) -> bool:
    graph: dict[str, set[str]] = {}
    for u, v in edges:
        if u == removed or v == removed:
            continue
        graph.setdefault(u, set()).add(v)
        graph.setdefault(v, set()).add(u)
    start, goal = terminals
    pending = deque([start])
    seen = {start}
    while pending:
        node = pending.popleft()
        if node == goal:
            return True
        for neighbor in graph.get(node, ()):
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return False


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    terminals = scene["terminals"]
    colored_nodes = scene["colored_nodes"]
    cuts = [
        color
        for color, node in colored_nodes.items()
        if not _connected_after_removal(edges, terminals, node)
    ]
    if len(cuts) != 1:
        raise ValueError(f"expected one colored terminal cut, found {cuts}")
    return cuts[0]


def margin(scene: dict) -> float:
    """Conservative raster clearance between a removed station and severed links."""

    return 18.0


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
        points = list(scene["nodes"].values())
        return any(math.dist(a, b) < 42 for i, a in enumerate(points) for b in points[i + 1 :])
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = [[v, u] for u, v in scene["edges"]]
    return [("reverse_undirected_edge_endpoints", twin)]
