"""Direct raster renderer and analytic gold for the transpose-overlay world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "transpose_overlay_symbol-0.1.0"
CANVAS_SIZE = (900, 520)
SYMBOLS = ("arch", "loop", "fork", "kite")
PANEL_SIZES = (176, 192, 208)
MARK_RADIUS = 7
MARK_COLOR_RED = (224, 58, 62)
MARK_COLOR_BLUE = (54, 92, 224)
BACKGROUND = (250, 250, 250)
PANEL_BACKGROUND = (236, 236, 236)
FRAME = (72, 72, 72)
BASE_LEVEL = 145.0
SIGNAL_LEVEL = 32.0
MARGIN_QUARANTINE = 0.08


def _points(symbol: str, size: int) -> list[tuple[int, int]]:
    s = float(size - 1)
    values = {
        "arch": ((0.27, 0.74), (0.27, 0.42), (0.50, 0.20), (0.73, 0.42), (0.73, 0.74)),
        "fork": ((0.50, 0.79), (0.50, 0.48), (0.27, 0.23)),
        "kite": ((0.50, 0.15), (0.76, 0.50), (0.50, 0.85), (0.24, 0.50), (0.50, 0.15)),
    }
    return [(int(round(s * x)), int(round(s * y))) for x, y in values[symbol]]


def glyph_mask(symbol: str, size: int, stroke_variant: int = 0) -> np.ndarray:
    """Draw a broad, low-frequency line glyph into a binary mask."""

    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    canvas = Image.new("L", (int(size), int(size)), 0)
    draw = ImageDraw.Draw(canvas)
    width = max(7, int(round(size * (0.052 + 0.006 * int(stroke_variant)))))
    if symbol == "loop":
        pad_x = int(round(size * 0.25))
        pad_y = int(round(size * 0.18))
        draw.ellipse((pad_x, pad_y, size - 1 - pad_x, size - 1 - pad_y), outline=255, width=width)
    elif symbol == "fork":
        points = _points(symbol, size)
        draw.line(points, fill=255, width=width, joint="curve")
        branch = _points(symbol, size)
        draw.line((branch[1], (int(round((size - 1) * 0.73)), int(round((size - 1) * 0.23)))), fill=255, width=width)
    else:
        draw.line(_points(symbol, size), fill=255, width=width, joint="curve")
    return np.asarray(canvas, dtype=np.float32) / 255.0


def _carrier(size: int, phase: int) -> np.ndarray:
    """Strong deterministic grain whose transpose is visibly different."""

    # A seeded pixel grain prevents the broad signal from being read in either
    # raw panel.  The paired panels carry exact opposite samples, so the
    # declared transpose followed by averaging cancels it without resampling.
    generator = np.random.default_rng(7000 + int(size) * 31 + int(phase) % 11)
    return generator.integers(-108, 109, size=(int(size), int(size))).astype(np.float32)


def _glyph_level(scene: dict) -> np.ndarray:
    size = int(scene["panel_size"])
    mask = glyph_mask(str(scene["glyph"]), size, int(scene["stroke_variant"]))
    return BASE_LEVEL - SIGNAL_LEVEL * mask


def _panel_pixels(scene: dict, transposed_recording: bool) -> np.ndarray:
    size = int(scene["panel_size"])
    carrier = _carrier(size, int(scene["carrier_phase"]))
    glyph = _glyph_level(scene)
    if transposed_recording:
        values = (glyph - carrier).T
    else:
        values = glyph + carrier
    return np.clip(np.rint(values), 0, 255).astype(np.uint8)


def _layout(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    size = int(scene["panel_size"])
    outer = size + 48
    layouts = (
        ((58, 142), (588, 128)),
        ((86, 96), (592, 170)),
        ((48, 172), (566, 88)),
    )
    left, right = layouts[int(scene["layout"]) % len(layouts)]
    # The layout tables are deliberately expressed for the largest panel.
    # Small panels retain translation variation without any clipping.
    max_right = CANVAS_SIZE[0] - outer - 42
    right = (min(right[0], max_right), right[1])
    return left, right


def _draw_panel(
    image: Image.Image,
    origin: tuple[int, int],
    pixels: np.ndarray,
    color: tuple[int, int, int],
) -> None:
    x0, y0 = (int(origin[0]), int(origin[1]))
    size = int(pixels.shape[0])
    outer = size + 48
    tile = Image.fromarray(pixels, mode="L").convert("RGB")
    image.paste(tile, (x0 + 24, y0 + 24))
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x0 + outer - 1, y0 + outer - 1), outline=FRAME, width=2)
    centers = (
        (x0 + 12, y0 + 12),
        (x0 + size + 36, y0 + 12),
        (x0 + 12, y0 + size + 36),
        (x0 + size + 36, y0 + size + 36),
    )
    radius = MARK_RADIUS
    for cx, cy in centers:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, BACKGROUND)
    left, right = _layout(scene)
    red_origin, blue_origin = (left, right) if int(scene["role_order"]) == 0 else (right, left)
    _draw_panel(image, red_origin, _panel_pixels(scene, False), MARK_COLOR_RED)
    _draw_panel(image, blue_origin, _panel_pixels(scene, True), MARK_COLOR_BLUE)
    return image


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    return {
        "seed": seed,
        # Consecutive generation seeds cycle through all answer classes while
        # the independent nuisance fields remain pseudorandom.
        "glyph": SYMBOLS[(seed + 1) % len(SYMBOLS)],
        "role_order": rng.randrange(2),
        "layout": rng.randrange(3),
        "panel_size": PANEL_SIZES[rng.randrange(len(PANEL_SIZES))],
        "stroke_variant": rng.randrange(2),
        "carrier_phase": rng.randrange(11),
    }


def _template_gap(image: np.ndarray, size: int, stroke_variant: int) -> float:
    """Return a normalized top-vs-runner-up gap on a transformed interior."""

    arr = image.astype(np.float32)
    candidates: list[tuple[float, str]] = []
    for symbol in SYMBOLS:
        for variant in (0, 1):
            target = BASE_LEVEL - SIGNAL_LEVEL * glyph_mask(symbol, size, variant)
            # Central support suppresses any residual border/crop uncertainty.
            lo = max(0, int(round(0.10 * size)))
            hi = min(size, int(round(0.90 * size)))
            mse = float(np.mean((arr[lo:hi, lo:hi] - target[lo:hi, lo:hi]) ** 2))
            candidates.append((mse, symbol))
    candidates.sort(key=lambda item: (item[0], item[1]))
    best = candidates[0][0]
    runner = next(score for score, symbol in candidates if symbol != candidates[0][1])
    return max(0.0, (runner - best) / 1000.0)


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    transformed = _glyph_level(scene)
    return float(_template_gap(transformed, int(scene["panel_size"]), int(scene["stroke_variant"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The renderer declares no aliases: every answer-bearing field is visible,
    # and nuisance variants are intentionally rendered rather than hidden.
    return []
