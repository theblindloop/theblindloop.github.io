"""Renderer for the filament_pointer_relay world.

One thin dark filament wanders across the whole canvas as a single connected
open curve. A red hub with an inset blue bar sits on the filament's starting
end and shows the pointer's initial compass direction. Small coloured tabs are
welded to the side of the filament along its length: an orange tab turns the
carried pointer one quarter turn clockwise, a teal tab mirrors it left-to-right
(east and west swap, north and south are unchanged). The tabs act in the order
in which they are met while walking the filament from the hub, so the answer is
the compass direction the pointer holds when the far end of the filament is
reached.

The scene is sampled deterministically from a seed; every sampled field is
painted into the raster.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "filament_pointer_relay-renderer-0.1.0"

WIDTH = 760
HEIGHT = 560
BG = (247, 247, 243)

WIRE = (38, 42, 58)
HUB = (206, 54, 66)
BAR = (24, 108, 190)
TURN = (236, 146, 26)
MIRROR = (28, 148, 122)
DECOR = (152, 154, 152)

WIRE_HALF = 2  # painted half-width of the filament (stroke width 5)
TAB_HALF = 6  # painted half-width of an operator tab
TAB_OFFSET = 8  # tab centre distance from the filament centreline
HUB_RADIUS = 15
BAR_LEN = 12
BAR_HALF = 4
LEAD = 46  # straight lead-in joining the hub to the serpentine body
TERMINAL_HALF = 6

X_LEFT = 76
X_RIGHT = 690
ROWS = 4
ROW_PITCH = 120
ROW_TOP = 78
JOG = 26
JOG_RUN = 46

HEADINGS = ("north", "east", "south", "west")

QUARANTINE_MARGIN = 5.0

# hard sampling floors (pixels)
MIN_TAB_GAP = 92.0
MIN_TAB_VERTEX = 11.0
MIN_DECOR_CLEAR = 18.0
MIN_LANE_GAP = 88.0


# --------------------------------------------------------------------------
# state algebra
# --------------------------------------------------------------------------
def _apply(heading: int, kind: str) -> int:
    if kind == "turn":
        return (heading + 1) % 4
    if kind == "mirror":
        return heading if heading % 2 == 0 else (heading + 2) % 4
    raise ValueError(f"unknown operator kind: {kind}")


def _replay(start: int, kinds: list[str]) -> int:
    heading = start
    for kind in kinds:
        heading = _apply(heading, kind)
    return heading


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _cumulative(path: list[list[int]]) -> list[float]:
    out = [0.0]
    for (x0, y0), (x1, y1) in zip(path, path[1:]):
        out.append(out[-1] + math.hypot(x1 - x0, y1 - y0))
    return out


def _point_at(path: list[list[int]], t: float) -> tuple[float, float, tuple[float, float]]:
    """Return (x, y, unit direction) at arclength t along the polyline."""
    cum = _cumulative(path)
    total = cum[-1]
    t = max(0.0, min(total, t))
    for index in range(len(path) - 1):
        if t <= cum[index + 1] or index == len(path) - 2:
            x0, y0 = path[index]
            x1, y1 = path[index + 1]
            seg = max(1e-9, cum[index + 1] - cum[index])
            frac = (t - cum[index]) / seg
            ux, uy = (x1 - x0) / seg, (y1 - y0) / seg
            return x0 + (x1 - x0) * frac, y0 + (y1 - y0) * frac, (ux, uy)
    raise AssertionError("unreachable")


def _seg_distance(px: float, py: float, x0: float, y0: float, x1: float, y1: float) -> float:
    dx, dy = x1 - x0, y1 - y0
    length2 = dx * dx + dy * dy
    if length2 <= 1e-9:
        return math.hypot(px - x0, py - y0)
    frac = max(0.0, min(1.0, ((px - x0) * dx + (py - y0) * dy) / length2))
    return math.hypot(px - (x0 + dx * frac), py - (y0 + dy * frac))


def _path_distance(px: float, py: float, path: list[list[int]]) -> float:
    return min(
        _seg_distance(px, py, a[0], a[1], b[0], b[1]) for a, b in zip(path, path[1:])
    )


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _build_path(rng: random.Random, start_left: bool, top_down: bool) -> list[list[int]]:
    ys = []
    for row in range(ROWS):
        ys.append(ROW_TOP + row * ROW_PITCH + rng.randint(-14, 14))
    ys.sort()
    if not top_down:
        ys = list(reversed(ys))

    path: list[list[int]] = []
    going_right = start_left
    for row, y in enumerate(ys):
        x_from = X_LEFT if going_right else X_RIGHT
        x_to = X_RIGHT if going_right else X_LEFT
        path.append([x_from, y])
        if rng.random() < 0.65:
            span = x_to - x_from
            frac = rng.uniform(0.34, 0.62)
            jx = int(x_from + span * frac)
            jog = JOG if rng.random() < 0.5 else -JOG
            run = JOG_RUN if going_right else -JOG_RUN
            path.append([jx, y])
            path.append([jx, y + jog])
            path.append([jx + run, y + jog])
            path.append([jx + run, y])
        path.append([x_to, y])
        if row < ROWS - 1:
            path.append([x_to, ys[row + 1]])
        going_right = not going_right

    # straight lead-in in front of the very first point, carrying the hub
    x0, y0 = path[0]
    x1, y1 = path[1]
    lead_x = x0 - LEAD if x1 > x0 else x0 + LEAD
    path.insert(0, [lead_x, y0])
    return path


def _tab_positions(rng: random.Random, path: list[list[int]]) -> list[float] | None:
    cum = _cumulative(path)
    total = cum[-1]
    vertices = cum[1:-1]
    count = rng.choice([3, 3, 4, 4, 5])
    lo, hi = 0.055 * total, 0.965 * total
    for _ in range(400):
        picks = sorted(rng.uniform(lo, hi) for _ in range(count))
        gaps = [b - a for a, b in zip(picks, picks[1:])]
        if gaps and min(gaps) < MIN_TAB_GAP:
            continue
        vertex_clear = min(
            abs(p - v) for p in picks for v in vertices
        )
        if vertex_clear < MIN_TAB_VERTEX:
            continue
        if picks[0] > 0.32 * total or picks[-1] < 0.70 * total:
            continue
        return picks
    return None


def _decor(rng: random.Random, path: list[list[int]], hub: tuple[float, float]) -> list[dict]:
    marks: list[dict] = []
    attempts = 0
    wanted = rng.choice([5, 6, 7, 8])
    while len(marks) < wanted and attempts < 500:
        attempts += 1
        radius = rng.choice([5, 6, 7, 8])
        cx = rng.randint(24, WIDTH - 24)
        cy = rng.randint(20, HEIGHT - 20)
        clear = _path_distance(cx, cy, path) - radius
        clear = min(clear, math.hypot(cx - hub[0], cy - hub[1]) - radius - HUB_RADIUS)
        for other in marks:
            clear = min(
                clear,
                math.hypot(cx - other["cx"], cy - other["cy"]) - radius - other["radius"],
            )
        if clear < MIN_DECOR_CLEAR:
            continue
        marks.append(
            {
                "cx": cx,
                "cy": cy,
                "radius": radius,
                "kind": rng.choice(["ring", "dash"]),
                "clear": round(clear, 3),
            }
        )
    return marks


def sample_scene(seed: int) -> dict:
    """Deterministically sample one filament scene."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(600):
        start_left = rng.random() < 0.5
        top_down = rng.random() < 0.5
        path = _build_path(rng, start_left, top_down)

        ys = sorted({p[1] for p in path})
        lane_gap = min(
            (b - a for a, b in zip(ys, ys[1:]) if b - a > JOG + 4), default=999.0
        )
        if lane_gap < MIN_LANE_GAP:
            continue

        picks = _tab_positions(rng, path)
        if picks is None:
            continue

        # Every tab must actually move the pointer when it is met, so a mirror is
        # only welded on while the running heading is east or west.
        start_heading = rng.randrange(4)
        kinds: list[str] = []
        heading = start_heading
        for _ in picks:
            allowed = ["turn"] if heading % 2 == 0 else ["turn", "mirror"]
            kind = rng.choice(allowed)
            kinds.append(kind)
            heading = _apply(heading, kind)
        if "turn" not in kinds or "mirror" not in kinds:
            continue
        final = _replay(start_heading, kinds)
        # erasing any single tab must change the answer
        if any(
            _replay(start_heading, kinds[:index] + kinds[index + 1 :]) == final
            for index in range(len(kinds))
        ):
            continue
        order_sensitive = False
        for index in range(len(kinds) - 1):
            swapped = list(kinds)
            swapped[index], swapped[index + 1] = swapped[index + 1], swapped[index]
            if _replay(start_heading, swapped) != final:
                order_sensitive = True
                break
        if not order_sensitive:
            continue

        sides = [1 if rng.random() < 0.5 else -1 for _ in picks]
        tabs = [
            {"t": round(t, 3), "kind": kind, "side": side}
            for t, kind, side in zip(picks, kinds, sides)
        ]

        hub = (float(path[0][0]), float(path[0][1]))
        decor = _decor(rng, path, hub)

        cum = _cumulative(path)
        vertices = cum[1:-1]
        gaps = [b["t"] - a["t"] for a, b in zip(tabs, tabs[1:])]
        vertex_clear = min(abs(tab["t"] - v) for tab in tabs for v in vertices)

        scene = {
            "path": [[int(x), int(y)] for x, y in path],
            "start_heading": start_heading,
            "tabs": tabs,
            "decor": decor,
            "start_left": start_left,
            "top_down": top_down,
            "clearances": {
                "min_tab_gap": round(min(gaps), 3),
                "min_tab_vertex": round(vertex_clear, 3),
                "min_lane_gap": round(float(lane_gap), 3),
                "min_decor_clear": round(min((m["clear"] for m in decor), default=99.0), 3),
            },
        }
        # every tab must sit inside the canvas with its painted body
        ok = True
        for tab in tabs:
            x, y, (ux, uy) = _point_at(path, tab["t"])
            tx = x - uy * TAB_OFFSET * tab["side"]
            ty = y + ux * TAB_OFFSET * tab["side"]
            if not (TAB_HALF + 2 <= tx <= WIDTH - TAB_HALF - 2):
                ok = False
            if not (TAB_HALF + 2 <= ty <= HEIGHT - TAB_HALF - 2):
                ok = False
        if not ok:
            continue
        return scene
    raise RuntimeError(f"failed to sample a legible scene for seed {seed}")


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Compass direction of the pointer once the filament has been walked."""
    kinds = [tab["kind"] for tab in sorted(scene["tabs"], key=lambda tab: tab["t"])]
    return HEADINGS[_replay(int(scene["start_heading"]), kinds)]


def margin(scene: dict) -> float:
    """Painted legibility margin in pixels of the decisive filament structure."""
    clear = scene["clearances"]
    return round(
        min(
            clear["min_tab_gap"] / 20.0,
            clear["min_tab_vertex"] - 6.0,
            (clear["min_lane_gap"] - 43.0) / 4.0,
        ),
        4,
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted (see candidate.json)."""
    return []


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    path = [tuple(point) for point in scene["path"]]

    for mark in scene["decor"]:
        cx, cy, radius = mark["cx"], mark["cy"], mark["radius"]
        if mark["kind"] == "ring":
            draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], outline=DECOR, width=2)
        else:
            draw.line([cx - radius, cy, cx + radius, cy], fill=DECOR, width=3)

    draw.line(path, fill=WIRE, width=2 * WIRE_HALF + 1, joint="curve")
    for x, y in path:
        draw.rectangle([x - WIRE_HALF, y - WIRE_HALF, x + WIRE_HALF, y + WIRE_HALF], fill=WIRE)

    ex, ey = path[-1]
    draw.rectangle(
        [ex - TERMINAL_HALF, ey - TERMINAL_HALF, ex + TERMINAL_HALF, ey + TERMINAL_HALF],
        fill=WIRE,
    )

    for tab in scene["tabs"]:
        x, y, (ux, uy) = _point_at(scene["path"], tab["t"])
        tx = x - uy * TAB_OFFSET * tab["side"]
        ty = y + ux * TAB_OFFSET * tab["side"]
        colour = TURN if tab["kind"] == "turn" else MIRROR
        draw.rectangle(
            [
                round(tx) - TAB_HALF,
                round(ty) - TAB_HALF,
                round(tx) + TAB_HALF,
                round(ty) + TAB_HALF,
            ],
            fill=colour,
        )

    hx, hy = path[0]
    draw.ellipse([hx - HUB_RADIUS, hy - HUB_RADIUS, hx + HUB_RADIUS, hy + HUB_RADIUS], fill=HUB)
    heading = int(scene["start_heading"])
    dx, dy = [(0, -1), (1, 0), (0, 1), (-1, 0)][heading]
    x0 = hx + min(0, dx * BAR_LEN) - (BAR_HALF if dx == 0 else 0)
    x1 = hx + max(0, dx * BAR_LEN) + (BAR_HALF if dx == 0 else 0)
    y0 = hy + min(0, dy * BAR_LEN) - (BAR_HALF if dy == 0 else 0)
    y1 = hy + max(0, dy * BAR_LEN) + (BAR_HALF if dy == 0 else 0)
    draw.rectangle([x0, y0, x1, y1], fill=BAR)
    return image
