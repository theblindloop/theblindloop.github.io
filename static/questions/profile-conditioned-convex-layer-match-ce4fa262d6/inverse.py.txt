"""Independent inverse arm receiving only the final raster image."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

IDENTITY_PALETTE = {
    "cyan": (25, 166, 190), "magenta": (200, 66, 148),
    "green": (42, 158, 86), "orange": (230, 125, 34),
    "violet": (126, 76, 188), "lime": (139, 181, 43),
    "pink": (235, 108, 116), "brown": (143, 94, 54),
}
BORDER_PALETTE = {"red": (211, 48, 48), "blue": (43, 94, 207), "gold": (218, 166, 32)}


def _mask(rgb: np.ndarray, target: tuple[int, int, int]) -> np.ndarray:
    return np.max(np.abs(rgb.astype(np.int16) - np.asarray(target, dtype=np.int16)), axis=2) <= 4


def _components(mask: np.ndarray) -> list[tuple[int, float, float, tuple[int, int, int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        todo = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while todo:
            yy, xx = todo.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        ys = [p[0] for p in pixels]
        xs = [p[1] for p in pixels]
        result.append((len(pixels), sum(xs) / len(xs), sum(ys) / len(ys), (min(xs), min(ys), max(xs), max(ys))))
    return result


def _turn(a: tuple[float, float, str], b: tuple[float, float, str], c: tuple[float, float, str]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _outside_ids(points: list[tuple[float, float, str]]) -> set[int]:
    ordered = sorted((x, y, color, i) for i, (x, y, color) in enumerate(points))
    if len(ordered) <= 1:
        return {item[3] for item in ordered}
    low = []
    for item in ordered:
        while len(low) >= 2 and _turn(low[-2][:3], low[-1][:3], item[:3]) <= 0.5:
            low.pop()
        low.append(item)
    high = []
    for item in reversed(ordered):
        while len(high) >= 2 and _turn(high[-2][:3], high[-1][:3], item[:3]) <= 0.5:
            high.pop()
        high.append(item)
    return {item[3] for item in low[:-1] + high[:-1]}


def _peel(points: list[tuple[float, float, str]]) -> tuple[tuple[str, ...], ...]:
    remaining = list(points)
    layers = []
    while remaining:
        outside = _outside_ids(remaining)
        layers.append(tuple(sorted(remaining[i][2] for i in outside)))
        remaining = [p for i, p in enumerate(remaining) if i not in outside]
    if tuple(map(len, layers)) != (4, 3, 1):
        raise ValueError("ambiguous convex-layer structure")
    return tuple(layers)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.ndim != 3 or rgb.shape[0] < 600 or rgb.shape[1] < 800:
        raise ValueError("unexpected raster dimensions")

    option_boxes = {}
    for label, value in BORDER_PALETTE.items():
        components = _components(_mask(rgb, value))
        boxes = [box for area, _, _, box in components if area > 2500 and box[2] - box[0] > 260 and box[3] - box[1] > 240]
        if len(boxes) != 1:
            raise ValueError(f"could not isolate {label} option frame")
        option_boxes[label] = boxes[0]

    all_dots = []
    for color, value in IDENTITY_PALETTE.items():
        components = _components(_mask(rgb, value))
        dots = [(cx, cy, color) for area, cx, cy, box in components if 300 <= area <= 520 and 20 <= box[2] - box[0] <= 26 and 20 <= box[3] - box[1] <= 26]
        if len(dots) != 4:
            raise ValueError(f"expected four raster dots for {color}")
        all_dots.extend(dots)

    options = {}
    claimed = set()
    for label, (x0, y0, x1, y1) in option_boxes.items():
        card = [(x, y, color) for x, y, color in all_dots if x0 < x < x1 and y0 < y < y1]
        if len(card) != 8 or len({p[2] for p in card}) != 8:
            raise ValueError("option card lacks one dot of each identity")
        options[label] = _peel(card)
        claimed.update((round(x, 3), round(y, 3), color) for x, y, color in card)
    reference = [p for p in all_dots if (round(p[0], 3), round(p[1], 3), p[2]) not in claimed]
    if len(reference) != 8 or len({p[2] for p in reference}) != 8:
        raise ValueError("reference card is not recoverable")
    target = _peel(reference)
    matches = [label for label, signature in options.items() if signature == target]
    if len(matches) != 1:
        raise ValueError("raster does not identify one matching option")
    return matches[0]
