"""Renderer and analytic arm for convex-layer correspondence."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 740
IDENTITY_COLORS = (
    "cyan", "magenta", "green", "orange",
    "violet", "lime", "pink", "brown",
)
BORDER_COLORS = ("red", "blue", "gold")
PALETTE = {
    "cyan": (25, 166, 190),
    "magenta": (200, 66, 148),
    "green": (42, 158, 86),
    "orange": (230, 125, 34),
    "violet": (126, 76, 188),
    "lime": (139, 181, 43),
    "pink": (235, 108, 116),
    "brown": (143, 94, 54),
    "red": (211, 48, 48),
    "blue": (43, 94, 207),
    "gold": (218, 166, 32),
}
REF_BOX = (290, 45, 710, 325)
OPTION_BOXES = ((25, 420, 325, 700), (350, 420, 650, 700), (675, 420, 975, 700))
DOT_RADIUS = 12


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull_indices(points: list[dict]) -> list[int]:
    ordered = sorted((float(p["x"]), float(p["y"]), i) for i, p in enumerate(points))
    if len(ordered) <= 1:
        return [item[2] for item in ordered]
    lower: list[tuple[float, float, int]] = []
    for item in ordered:
        while len(lower) >= 2 and _cross(lower[-2][:2], lower[-1][:2], item[:2]) <= 1e-7:
            lower.pop()
        lower.append(item)
    upper: list[tuple[float, float, int]] = []
    for item in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2][:2], upper[-1][:2], item[:2]) <= 1e-7:
            upper.pop()
        upper.append(item)
    return [item[2] for item in lower[:-1] + upper[:-1]]


def _layer_signature(points: list[dict]) -> tuple[tuple[str, ...], ...]:
    remaining = [dict(p) for p in points]
    layers: list[tuple[str, ...]] = []
    while remaining:
        hull = _hull_indices(remaining)
        layers.append(tuple(sorted(remaining[i]["color"] for i in hull)))
        remove = set(hull)
        remaining = [p for i, p in enumerate(remaining) if i not in remove]
    return tuple(layers)


def _geometry(rng: random.Random, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    base = [
        (-1.00, -0.72), (0.82, -0.92), (1.00, 0.72), (-0.82, 0.94),
        (-0.55, 0.36), (0.00, -0.56), (0.55, 0.36),
        (0.00, 0.05),
    ]
    angle = rng.uniform(0, 2 * math.pi)
    ca, sa = math.cos(angle), math.sin(angle)
    sx, sy = rng.uniform(0.90, 1.10), rng.uniform(0.90, 1.10)
    shear = rng.uniform(-0.15, 0.15)
    raw = []
    for x, y in base:
        x1, y1 = sx * x + shear * y, sy * y
        raw.append((x1 * ca - y1 * sa, x1 * sa + y1 * ca))
    x0, y0, x1, y1 = box
    room_x, room_y = x1 - x0 - 74, y1 - y0 - 74
    span_x = max(x for x, _ in raw) - min(x for x, _ in raw)
    span_y = max(y for _, y in raw) - min(y for _, y in raw)
    scale = min(room_x / span_x, room_y / span_y) * rng.uniform(0.82, 0.94)
    values = [(x * scale, y * scale) for x, y in raw]
    extent_x = max(abs(x) for x, _ in values)
    extent_y = max(abs(y) for _, y in values)
    cx = (x0 + x1) / 2 + rng.uniform(-max(0, room_x / 2 - extent_x), max(0, room_x / 2 - extent_x))
    cy = (y0 + y1) / 2 + rng.uniform(-max(0, room_y / 2 - extent_y), max(0, room_y / 2 - extent_y))
    return [(cx + x - x0, cy + y - y0) for x, y in values]


def _card(rng: random.Random, box: tuple[int, int, int, int], layer_colors: list[list[str]]) -> dict:
    positions = _geometry(rng, box)
    points = []
    offset = 0
    for colors in layer_colors:
        shuffled = list(colors)
        rng.shuffle(shuffled)
        for color, (x, y) in zip(shuffled, positions[offset:offset + len(shuffled)]):
            points.append({"color": color, "x": round(x, 5), "y": round(y, 5)})
        offset += len(shuffled)
    rng.shuffle(points)
    return {"points": points}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    shuffled = list(IDENTITY_COLORS)
    rng.shuffle(shuffled)
    reference_layers = [shuffled[:4], shuffled[4:7], shuffled[7:]]

    distractor_one = [list(layer) for layer in reference_layers]
    distractor_one[0][0], distractor_one[1][0] = distractor_one[1][0], distractor_one[0][0]
    distractor_two = [list(layer) for layer in reference_layers]
    distractor_two[0][1], distractor_two[1][1] = distractor_two[1][1], distractor_two[0][1]
    distractor_two[1][2], distractor_two[2][0] = distractor_two[2][0], distractor_two[1][2]
    variants = [reference_layers, distractor_one, distractor_two]
    match_slot = rng.randrange(3)
    option_layers = [None, None, None]
    option_layers[match_slot] = variants[0]
    distractors = [variants[1], variants[2]]
    rng.shuffle(distractors)
    for i in range(3):
        if option_layers[i] is None:
            option_layers[i] = distractors.pop()

    border_colors = list(BORDER_COLORS)
    rng.shuffle(border_colors)
    cards = [_card(rng, REF_BOX, reference_layers)]
    cards.extend(_card(rng, box, layers) for box, layers in zip(OPTION_BOXES, option_layers))
    return {"width": WIDTH, "height": HEIGHT, "cards": cards, "border_colors": border_colors}


def _draw_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], border: tuple[int, int, int], card: dict) -> None:
    draw.rounded_rectangle(box, radius=15, fill=(251, 251, 248), outline=border, width=7)
    x0, y0, _, _ = box
    for point in card["points"]:
        cx = int(round(x0 + point["x"]))
        cy = int(round(y0 + point["y"]))
        draw.ellipse((cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS), fill=PALETTE[point["color"]], outline=(35, 35, 35), width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (245, 244, 239))
    draw = ImageDraw.Draw(image)
    _draw_card(draw, REF_BOX, (65, 65, 65), scene["cards"][0])
    for box, border_color, card in zip(OPTION_BOXES, scene["border_colors"], scene["cards"][1:]):
        _draw_card(draw, box, PALETTE[border_color], card)
    draw.text((500, 17), "REFERENCE", fill=(45, 45, 45), anchor="ma")
    draw.text((500, 378), "OPTIONS", fill=(45, 45, 45), anchor="ma")
    return image


def analytic_gold(scene: dict) -> str:
    target = _layer_signature(scene["cards"][0]["points"])
    matches = [i for i, card in enumerate(scene["cards"][1:]) if _layer_signature(card["points"]) == target]
    if len(matches) != 1:
        raise ValueError("scene does not contain one layer-profile match")
    return scene["border_colors"][matches[0]]


def _point_segment_distance(p: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    vx, vy = b[0] - a[0], b[1] - a[1]
    wx, wy = p[0] - a[0], p[1] - a[1]
    denom = vx * vx + vy * vy
    t = max(0.0, min(1.0, (wx * vx + wy * vy) / denom))
    return math.hypot(p[0] - (a[0] + t * vx), p[1] - (a[1] + t * vy))


def _card_clearance(card: dict) -> float:
    remaining = [dict(p) for p in card["points"]]
    clearances = []
    while len(remaining) >= 3:
        hull_ids = _hull_indices(remaining)
        if len(hull_ids) < 3:
            return 0.0
        hull = [remaining[i] for i in hull_ids]
        inner = [p for i, p in enumerate(remaining) if i not in set(hull_ids)]
        for point in inner:
            p = (point["x"], point["y"])
            clearances.append(min(_point_segment_distance(p, (hull[j]["x"], hull[j]["y"]), (hull[(j + 1) % len(hull)]["x"], hull[(j + 1) % len(hull)]["y"])) for j in range(len(hull))))
        remaining = inner
    distances = []
    pts = card["points"]
    for i, left in enumerate(pts):
        for right in pts[i + 1:]:
            distances.append(math.hypot(left["x"] - right["x"], left["y"] - right["y"]) - 2 * DOT_RADIUS)
    return min(clearances + distances)


def margin(scene: dict) -> float:
    target = _layer_signature(scene["cards"][0]["points"])
    mismatch_counts = []
    for card in scene["cards"][1:]:
        sig = _layer_signature(card["points"])
        if sig != target:
            mismatch_counts.append(sum(set(a) != set(b) for a, b in zip(sig, target)))
    structural = 18.0 * min(mismatch_counts) if mismatch_counts else 0.0
    return float(min(structural, min(_card_clearance(card) for card in scene["cards"])))


def is_quarantined(scene: dict) -> bool:
    target = _layer_signature(scene["cards"][0]["points"])
    return sum(_layer_signature(card["points"]) == target for card in scene["cards"][1:]) != 1 or margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = deepcopy(scene)
    for card in transformed["cards"]:
        card["points"] = list(reversed(card["points"]))
    return [("reverse_nonoverlapping_point_serialization", transformed)]
