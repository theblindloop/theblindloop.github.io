"""Independent pixel-only inverse for the notched swap ribbon.

The parser derives its frame from colored endpoint components, follows the
single connected blue trace, divides only the detected endpoint span into six
ordered intervals, and replays the recovered adjacent swaps.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PATH_REFS = np.asarray(((25, 95, 175), (40, 80, 165), (15, 110, 185)), dtype=np.int32)
_START_REFS = np.asarray(((30, 160, 76), (22, 145, 61), (53, 166, 67)), dtype=np.int32)
_END_REFS = np.asarray(((203, 61, 72), (190, 67, 83), (213, 73, 59)), dtype=np.int32)
_LANES = ("A", "B", "C")


class OracleAbstention(ValueError):
    pass


def _color_mask(array: np.ndarray, references: np.ndarray, radius: float = 46.0) -> np.ndarray:
    pixels = array.astype(np.int32)
    squared = ((pixels[:, :, None, :] - references[None, None, :, :]) ** 2).sum(axis=3)
    return squared.min(axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        groups.append(np.asarray(points, dtype=np.float64))
    return sorted(groups, key=len, reverse=True)


def _single_component(mask: np.ndarray, name: str, minimum: int) -> np.ndarray:
    groups = _components(mask)
    if not groups or len(groups[0]) < minimum:
        raise OracleAbstention(f"{name} is missing")
    if len(groups) > 1 and len(groups[1]) >= 12:
        raise OracleAbstention(f"{name} is fragmented")
    return groups[0]


def _centroid(points: np.ndarray) -> tuple[float, float]:
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def decision_from_image(image: Image.Image) -> str:
    """Recover and replay the six swaps from the final raster alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 64:
        raise OracleAbstention("unsupported raster")
    blue = _color_mask(array, _PATH_REFS)
    green = _color_mask(array, _START_REFS)
    red = _color_mask(array, _END_REFS)

    path_points = _single_component(blue, "ribbon", 450)
    start_points = _single_component(green, "green endpoint", 45)
    end_points = _single_component(red, "red endpoint", 45)
    gx, gy = _centroid(start_points)
    rx, ry = _centroid(end_points)
    span = rx - gx
    if not 160.0 <= span <= 200.0 or abs(gy - ry) > 18.0:
        raise OracleAbstention("endpoint geometry is ambiguous")

    endpoint_radius = max(3.0, np.sqrt(len(end_points) / np.pi))
    delta = gy - ry
    if delta < -endpoint_radius:
        start_lane = 0
    elif delta > endpoint_radius:
        start_lane = 2
    elif abs(delta) <= endpoint_radius * 0.65:
        start_lane = 1
    else:
        raise OracleAbstention("start bead lies between lanes")

    # Require that the answer-bearing pixels themselves form one elongated
    # component whose detected extent bridges the two endpoint neighborhoods.
    px_min, px_max = path_points[:, 1].min(), path_points[:, 1].max()
    if px_min > gx + 10.0 or px_max < rx - 10.0:
        raise OracleAbstention("ribbon does not reach both endpoints")

    inner_left = gx + 0.078 * span
    inner_right = rx - 0.056 * span
    width = (inner_right - inner_left) / 6.0
    operations: list[str] = []
    for index in range(6):
        left = inner_left + (index + 0.24) * width
        right = inner_left + (index + 0.76) * width
        selected = path_points[(path_points[:, 1] >= left) & (path_points[:, 1] <= right)]
        if len(selected) < 28:
            raise OracleAbstention(f"notch {index + 1} is incomplete")
        upper = ry - float(selected[:, 0].min())
        lower = float(selected[:, 0].max()) - ry
        if max(upper, lower) < 7.0 or abs(upper - lower) < 4.5:
            raise OracleAbstention(f"notch {index + 1} is flat or ambiguous")
        operations.append("upper" if upper > lower else "lower")

    lane = start_lane
    for operation in operations:
        if operation == "upper" and lane in (0, 1):
            lane = 1 - lane
        elif operation == "lower" and lane in (1, 2):
            lane = 3 - lane
    return _LANES[lane]
