"""Deterministic renderer and analytic arm for the notched swap ribbon."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 768
HEIGHT = 512
SCALE = 4
BACKGROUND = (247, 244, 235)
INK = (47, 53, 61)
PALE = (214, 211, 201)

PATH_COLORS = ((25, 95, 175), (40, 80, 165), (15, 110, 185))
START_COLORS = ((30, 160, 76), (22, 145, 61), (53, 166, 67))
END_COLORS = ((203, 61, 72), (190, 67, 83), (213, 73, 59))

LANES = ("A", "B", "C")
OPERATIONS = ("upper", "lower")


def _replay(operations: list[str], start_lane: str) -> str:
    lane = LANES.index(start_lane)
    for operation in operations:
        if operation == "upper" and lane < 2:
            if lane in (0, 1):
                lane = 1 - lane
        elif operation == "lower" and lane > 0:
            if lane in (1, 2):
                lane = 3 - lane
    return LANES[lane]


_BALANCED_WORDS = tuple(
    tuple("upper" if index in upper_positions else "lower" for index in range(6))
    for upper_positions in itertools.combinations(range(6), 3)
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with fixed operation ink and answer cycling."""

    rng = random.Random(int(seed))
    desired = LANES[int(seed) % 3]
    start_lane = LANES[rng.randrange(3)]
    eligible = [word for word in _BALANCED_WORDS if _replay(list(word), start_lane) == desired]
    operations = list(eligible[rng.randrange(len(eligible))])
    return {
        "operations": operations,
        "start_lane": start_lane,
        "anchor_col": rng.choice((2, 3, 4, 5)),
        "anchor_row": rng.choice((2, 3, 4, 5)),
        "palette_variant": rng.randrange(len(PATH_COLORS)),
        "decoration_variant": rng.randrange(4),
    }


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf", size * SCALE)


def render(scene: dict) -> Image.Image:
    """Render all answer-bearing evidence as one short connected blue trace."""

    operations = list(scene["operations"])
    start_lane = str(scene["start_lane"])
    anchor_col = int(scene["anchor_col"])
    anchor_row = int(scene["anchor_row"])
    palette_variant = int(scene["palette_variant"]) % len(PATH_COLORS)
    decoration_variant = int(scene["decoration_variant"]) % 2
    if len(operations) != 6 or any(item not in OPERATIONS for item in operations):
        raise ValueError("exactly six upper/lower operations are required")
    if start_lane not in LANES:
        raise ValueError("invalid start lane")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    cx = (48 + 96 * anchor_col) * SCALE
    cy = (32 + 64 * anchor_row) * SCALE
    sx, ex = cx - 90 * SCALE, cx + 90 * SCALE
    lane_offset = (-12, 0, 12)[LANES.index(start_lane)] * SCALE
    sy = cy + lane_offset
    path_color = PATH_COLORS[palette_variant]
    start_color = START_COLORS[palette_variant]
    end_color = END_COLORS[palette_variant]

    # Pale card and local row key are explanatory but ignored by the oracle.
    draw.rounded_rectangle(
        (cx - 112 * SCALE, cy - 27 * SCALE, cx + 112 * SCALE, cy + 27 * SCALE),
        radius=10 * SCALE,
        fill=(252, 250, 244),
        outline=PALE,
        width=1 * SCALE,
    )
    font = _font(9)
    for label, offset in zip(LANES, (-12, 0, 12), strict=True):
        y = cy + offset * SCALE
        draw.line((sx - 17 * SCALE, y, sx - 10 * SCALE, y), fill=PALE, width=2 * SCALE)
        draw.text((sx - 29 * SCALE, y - 6 * SCALE), label, font=font, fill=(105, 105, 101))

    # The nuisance decoration has a real two-state appearance, but values two
    # apart are an intentional latent alias exercised by latent_symmetries().
    deco_y = cy + (-19 if decoration_variant == 0 else 19) * SCALE
    draw.ellipse(
        (cx - 3 * SCALE, deco_y - 2 * SCALE, cx + 1 * SCALE, deco_y + 2 * SCALE),
        fill=(229, 198, 104),
    )

    points: list[tuple[int, int]] = [(sx, sy), (sx + 14 * SCALE, cy)]
    inner_left = sx + 14 * SCALE
    inner_right = ex - 10 * SCALE
    step = (inner_right - inner_left) / 6.0
    for index, operation in enumerate(operations):
        left = inner_left + index * step
        right = inner_left + (index + 1) * step
        peak = (left + right) / 2.0
        peak_y = cy + (-11 if operation == "upper" else 11) * SCALE
        if index and points[-1] != (round(left), cy):
            points.append((round(left), cy))
        points.extend(((round(peak), peak_y), (round(right), cy)))
    points.append((ex, cy))
    draw.line(points, fill=path_color, width=5 * SCALE, joint="curve")

    radius = 7 * SCALE
    draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=INK)
    draw.ellipse(
        (sx - 5 * SCALE, sy - 5 * SCALE, sx + 5 * SCALE, sy + 5 * SCALE),
        fill=start_color,
    )
    draw.ellipse((ex - radius, cy - radius, ex + radius, cy + radius), fill=INK)
    draw.ellipse(
        (ex - 5 * SCALE, cy - 5 * SCALE, ex + 5 * SCALE, cy + 5 * SCALE),
        fill=end_color,
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _replay(list(scene["operations"]), str(scene["start_lane"]))


def margin(scene: dict) -> float:
    """Designed minimum separation, in pixels, from a flat/ambiguous notch."""

    return 8.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decoration_variant"] = int(twin["decoration_variant"]) + 2
    return [("decoration_alias_plus_two", twin)]
