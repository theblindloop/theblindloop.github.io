"""Independent pixel-only inverse arm for salt_pepper_median_reveal.

Receives ONLY the final PNG. It:
  1. reads the mean luminance of every lattice cell;
  2. thresholds each cell to bright/dark;
  3. applies a nonlinear median (majority) filter over a neighborhood to strip
     the salt-and-pepper outliers;
  4. correlates the cleaned binary field against the two declared diagonal-band
     templates A (rising) and B (falling);
  5. returns the template with the larger inside-band occupancy fraction.

No scene data, no analytic gold, no renderer imports. The decision relies on
the wide-band majority structure that survives only after the median filter.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "salt_pepper_median_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_WIDTH = 7
RADIUS = 2


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def _cell_means(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    g = np.zeros((NGRID, NGRID), dtype=np.float64)
    for i in range(NGRID):
        for j in range(NGRID):
            cell = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
            g[i, j] = cell.mean()
    return g


def _majority(field: np.ndarray, radius: int = RADIUS) -> np.ndarray:
    out = np.zeros_like(field)
    n = field.shape[0]
    for i in range(n):
        lo = max(0, i - radius)
        hi = min(n, i + radius + 1)
        for j in range(n):
            jlo = max(0, j - radius)
            jhi = min(n, j + radius + 1)
            win = field[lo:hi, jlo:jhi]
            out[i, j] = 1.0 if win.mean() >= 0.5 else 0.0
    return out


def decision_from_image(image: Image.Image) -> str:
    g = _cell_means(image)
    binary = _majority((g > 128.0).astype(np.float64))
    masks = _band_masks()
    score_a = float((binary * masks["A"]).sum()) / float(masks["A"].sum())
    score_b = float((binary * masks["B"]).sum()) / float(masks["B"].sum())
    return "A" if score_a >= score_b else "B"


def reveal_margin(image: Image.Image) -> float:
    g = _cell_means(image)
    binary = _majority((g > 128.0).astype(np.float64))
    masks = _band_masks()
    score_a = float((binary * masks["A"]).sum()) / float(masks["A"].sum())
    score_b = float((binary * masks["B"]).sum()) / float(masks["B"].sum())
    return abs(score_a - score_b)
