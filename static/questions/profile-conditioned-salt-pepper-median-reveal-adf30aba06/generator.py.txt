"""salt_pepper_median_reveal: median (majority) outlier-removal glyph reveal.

Declared transform: apply a median (nonlinear majority) filter to remove the
pure salt-and-pepper outlier noise, then read the coarse diagonal-band glyph
that the filtered field reveals. Every cell of the raw raster is either a
bright or dark band sample or, with probability P_FLIP, an extreme random
salt-or-pepper outlier; at P_FLIP = 0.5 the raw field is visually uniform
random black-and-white speckle with no band visible from any local patch. Only
removing the outliers with a nonlinear median/majority filter exposes the wide
diagonal band, whose orientation is the glyph A (rising) or B (falling).

Latent scene z = {noise, glyph}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "salt_pepper_median_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # cells per side
CELL = W // NGRID   # 16
BRIGHT = 240.0      # luminance of an in-band (on) cell
DARK = 20.0         # luminance of an out-of-band (off) cell
P_FLIP = 0.5        # per-cell probability of an extreme salt-or-pepper outlier
BAND_WIDTH = 7      # half-width of the diagonal band in cell units
Q_TOL = 0.15        # quarantine threshold on the recovered template margin


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A rising '/', B falling '\\'."""
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) cell-luminance field with salt-and-pepper outliers."""
    n = NGRID
    inside = _glyph_mask(scene["glyph"])
    base = np.where(inside, BRIGHT, DARK)
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    flip = rng.random((n, n)) < P_FLIP
    peaks = np.where(rng.random((n, n)) < 0.5, 255.0, 0.0)
    return np.where(flip, peaks, base)


def majority_filter(field: np.ndarray, radius: int = 2) -> np.ndarray:
    """Binary majority (median-of-neighborhood) filter over a 2*radius+1 window."""
    out = np.zeros_like(field)
    n = field.shape[0]
    for i in range(n):
        lo = max(0, i - radius)
        hi = min(n, i + radius + 1)
        for j in range(n):
            jlo = max(0, j - radius)
            jhi = min(n, j + radius + 1)
            win = field[lo:hi, jlo:jhi]
            out[i, j] = 1.0 if win.mean() >= 0.5 else 0.0
    return out


def _decision_from_field(field: np.ndarray) -> tuple[str, float]:
    binary = majority_filter((field > 128.0).astype(np.float64))
    masks = band_masks()
    score_a = float((binary * masks["A"]).sum()) / float(masks["A"].sum())
    score_b = float((binary * masks["B"]).sum()) / float(masks["B"].sum())
    glyph = "A" if score_a >= score_b else "B"
    return glyph, round(abs(score_a - score_b), 6)


def render(scene: dict) -> Image.Image:
    mat = render_matrix(scene)
    arr = np.repeat(np.repeat(mat, CELL, axis=0), CELL, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    _, m = _decision_from_field(render_matrix(scene))
    return m


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: the per-cell noise seed and the answer glyph each
    directly change the raster (every cell's value depends on the noise seed
    and the band mask), so no distinct latent vector reproduces the same image.
    Declared empty by design."""
    return iter(())
