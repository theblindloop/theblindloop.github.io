"""Collinearity-degree comparison world: which colored triplet is more collinear.

A light panel holds two colored dot triplets: exactly three red filled dots and
exactly three blue filled dots, each triplet drawn near a straight line with a
controlled perpendicular bend. The question asks which colour's three dots lie
most nearly on a single straight line, i.e. which triplet has the smaller
measured collinearity error.

The measured per-colour scalar is the normalised collinearity error of the
triplet of dot centres:

    error = (2 * triangle_area) / span^2

where span is the maximum pairwise distance between the three dot centres. For
three points this is (height of the triangle from its longest side) divided by
that longest side, a scale-invariant "straightness" measure. The pixel arm
recovers the three dot centres of each colour from the raster and computes the
same ratio, then selects the colour with the smaller error.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "more_collinear_color-0.1.0"

RED_RGB = (200, 20, 20)
BLUE_RGB = (20, 20, 200)
BACKGROUND = (250, 250, 250)

WIDTH = 460
HEIGHT = 320
DOT_RADIUS = 7.0
DOT_DIAM = int(2 * DOT_RADIUS)

# Perpendicular offset of the middle dot from the line through the other two.
_D_MIN = 7.0
_D_MAX = 42.0
# End-to-end span of each triplet (distance between the two outer dots).
_SPAN_MIN = 90.0
_SPAN_MAX = 160.0

# Minimum pairwise distance between any two dot centres (same or other colour)
# so that dot centroids stay separable in the raster.
MIN_SEP = 40.0

# A scene is only emitted when the analytic-error margin is at least this large,
# guaranteeing a clean winner for the pixel oracle.
MIN_MARGIN = 0.05
# Below this margin a scene is flagged quarantined as a finite-resolution
# boundary case, but it is still emitted and certified.
QUARANTINE_MARGIN = 0.10


def _triangle_error(points: list) -> float:
    """Normalised collinearity error (2*area / span^2) of three dot centres."""
    (ax, ay), (bx, by), (cx, cy) = points
    area = 0.5 * abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))
    span2 = max(
        (bx - ax) ** 2 + (by - ay) ** 2,
        (cx - bx) ** 2 + (cy - by) ** 2,
        (ax - cx) ** 2 + (ay - cy) ** 2,
    )
    if span2 <= 1e-9:
        return 0.0
    return 2.0 * area / span2


def _error_for(scene: dict, colour: str) -> float:
    return _triangle_error(scene["dots"][colour])


def _in_bounds(x: float, y: float) -> bool:
    margin = DOT_RADIUS + 3.0
    return margin <= x <= WIDTH - margin and margin <= y <= HEIGHT - margin


def _pairwise_ok(points_a: list, points_b: list) -> bool:
    all_pts = points_a + points_b
    for i in range(len(all_pts)):
        for j in range(i + 1, len(all_pts)):
            ax, ay = all_pts[i]
            bx, by = all_pts[j]
            if (ax - bx) ** 2 + (ay - by) ** 2 < MIN_SEP * MIN_SEP:
                return False
    return True


def _sample_triplet(rng: random.Random) -> list:
    """Sample three dot centres: two outer dots plus a middle dot bent off axis."""
    while True:
        cx = rng.uniform(120.0, WIDTH - 120.0)
        cy = rng.uniform(95.0, HEIGHT - 95.0)
        ang = rng.uniform(0.0, math.pi)
        ux, uy = math.cos(ang), math.sin(ang)
        vx, vy = -uy, ux
        span = rng.uniform(_SPAN_MIN, _SPAN_MAX)
        d = rng.uniform(_D_MIN, _D_MAX)
        half = span / 2.0
        pts = [
            [cx - ux * half, cy - uy * half],
            [cx + ux * half, cy + uy * half],
            [cx + vx * d, cy + vy * d],
        ]
        if all(_in_bounds(p[0], p[1]) for p in pts):
            return pts


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1013 + 11)
    while True:
        red = _sample_triplet(rng)
        blue = _sample_triplet(rng)
        if not _pairwise_ok(red, blue):
            continue
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
            "dots": {"red": red, "blue": blue},
        }
        if margin(scene) >= MIN_MARGIN:
            return scene


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for colour, rgb in (("red", RED_RGB), ("blue", BLUE_RGB)):
        for (x, y) in scene["dots"][colour]:
            left = int(round(x - DOT_RADIUS))
            top = int(round(y - DOT_RADIUS))
            right = int(round(x + DOT_RADIUS))
            bottom = int(round(y + DOT_RADIUS))
            draw.ellipse([left, top, right, bottom], fill=rgb)
    return img


def analytic_gold(scene: dict) -> str:
    """Colour whose triplet has the smaller collinearity error."""
    red_err = _error_for(scene, "red")
    blue_err = _error_for(scene, "blue")
    if red_err <= blue_err:
        return "red"
    return "blue"


def margin(scene: dict) -> float:
    """Difference between the two collinearity errors (rounding-safe)."""
    return float(abs(_error_for(scene, "red") - _error_for(scene, "blue")))


def is_quarantined(scene: dict) -> bool:
    """Near-ambiguous: the two triplets' collinearity errors are too close."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field (the dot centres) is drawn verbatim.

    Each of the three per-colour dot centres becomes visible filled ink, and the
    analytic decision reads exactly those dot centres. There is no hidden scene
    field that could change which colour is more collinear without changing the
    raster, so no nontrivial pixel-identical latent transformation exists.
    """
    return []
