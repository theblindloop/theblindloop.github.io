"""Independent pixel-only inverse arm for the collinearity-degree world.

This module receives ONLY the rasterized PNG and decides which colour's three
dots lie most nearly on a single straight line. It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or any analytic gold code.

Pipeline: (1) label each pixel to its nearest fixed colour; (2) group each
colour's ink into connected components and take the centroid of every dot;
(3) compute each triplet's normalised collinearity error (2*area / span^2 from
the three centroids); (4) select the colour with the smaller error.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "more_collinear_color-oracle-0.1.0"

RED_RGB = (200, 20, 20)
BLUE_RGB = (20, 20, 200)
BACKGROUND = (250, 250, 250)
_COLOURS = [BACKGROUND, RED_RGB, BLUE_RGB]


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest fixed colour (0 bg, 1 red, 2 blue)."""
    refs = np.stack([np.array(c, dtype=np.float32) for c in _COLOURS], axis=0)
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _centroids(labels: np.ndarray, colour: int) -> list[tuple[float, float]]:
    """Centroid of each connected component of the given labelled colour."""
    ys, xs = np.nonzero(labels == colour)
    if len(ys) == 0:
        return []
    points = {}
    for x, y in zip(xs.tolist(), ys.tolist()):
        points.setdefault((x, y), True)
    seen = set()
    out: list[tuple[float, float]] = []
    for (sx, sy) in points:
        if (sx, sy) in seen:
            continue
        stack = [(sx, sy)]
        component: list[tuple[int, int]] = []
        seen.add((sx, sy))
        while stack:
            cx, cy = stack.pop()
            component.append((cx, cy))
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nx, ny = cx + dx, cy + dy
                if (nx, ny) in points and (nx, ny) not in seen:
                    seen.add((nx, ny))
                    stack.append((nx, ny))
        mx = sum(p[0] for p in component) / len(component)
        my = sum(p[1] for p in component) / len(component)
        out.append((mx, my))
    return out


def _error(centres: list[tuple[float, float]]) -> float:
    if len(centres) < 3:
        return 0.0
    (ax, ay), (bx, by), (cx, cy) = centres
    area = 0.5 * abs((bx - ax) * (cy - ay) - (cx - ax) * (by - ay))
    span2 = max(
        (bx - ax) ** 2 + (by - ay) ** 2,
        (cx - bx) ** 2 + (cy - by) ** 2,
        (ax - cx) ** 2 + (ay - cy) ** 2,
    )
    if span2 <= 1e-6:
        return 0.0
    return 2.0 * area / span2


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    red = _error(_centroids(labels, 1))
    blue = _error(_centroids(labels, 2))
    if red <= blue:
        return "red"
    return "blue"
