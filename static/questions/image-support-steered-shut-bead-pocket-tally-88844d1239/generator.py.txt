"""Renderer and analytic gold for `shut_bead_pocket_tally`.

Three small black slabs stand together inside one small inset of the page. Each
slab carries one or two square holes; three of the five holes hold a coloured
bead. Exactly one hole in the scene has a channel cut clean through its wall, so
that hole is joined to the open page. A bead is *shut in* when the white area
holding it is completely surrounded by black ink. The gold is the number of
shut-in beads, which is two when the channel happens to sit at a bead hole and
three when it sits at an empty hole.

Every sampled field is painted into the raster; nothing is gold-only.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (1280, 768)
BAND = 96
HOLE = (20, 20)
PLATE_GAP = 16
BLACK = (18, 18, 18)
BEAD_COLORS = [
    (214, 42, 52),
    (30, 90, 220),
    (24, 150, 60),
    (232, 140, 20),
    (150, 60, 200),
]
GREY_INK = [(176, 176, 176), (158, 158, 158), (192, 192, 192)]
WHITE = (255, 255, 255)
QUARANTINE_PX = 6.0


def _plate_size(wall: int, holes: int) -> tuple[int, int]:
    width = 2 * wall + holes * HOLE[0] + (holes - 1) * (wall + 4)
    height = 2 * wall + HOLE[1]
    return width, height


def sample_scene(seed: int) -> dict:
    rng = random.Random(f"shut_bead_pocket_tally|{seed}")
    tight = rng.random() < 0.18
    if tight:
        wall = rng.choice([6, 7, 8])
        channel = rng.choice([5, 6, 7])
    else:
        wall = rng.choice([9, 10, 11])
        channel = rng.choice([9, 10, 11])
    bead_w = rng.choice([6, 8])
    bead_h = rng.choice([6, 8])

    # two slabs standing far apart in one small inset: they share five holes and
    # three beads, split two and one, so the page always shows two slabs, five
    # holes, three beads in holes and one channel regardless of the answer, and
    # every slab keeps at least one bead of its own.
    plate_holes = rng.choice([[2, 3], [3, 2]])
    plate_beads = rng.choice([[2, 1], [1, 2]])

    sizes = [_plate_size(wall, n) for n in plate_holes]
    block_w = max(w for w, _ in sizes) + 8
    # the two slabs stand in two separate horizontal bands of the page with one
    # whole empty band of white page between them, so background always
    # separates the two groups of decisive ink.
    band = rng.randint(1, 4)
    tops = []
    for step in (0, 2):
        base = BAND * (band + step) + BAND // 4
        tops.append(base + rng.randint(0, max(0, BAND // 2 - sizes[step // 2][1])))
    block_h = tops[1] + sizes[1][1] - tops[0]
    ox = rng.randint(80, CANVAS[0] - block_w - 200)
    oy = tops[0]

    plates = []
    palette = rng.sample(BEAD_COLORS, 2)
    for index, count in enumerate(plate_holes):
        pw, ph = sizes[index]
        px = ox + rng.randint(0, 8)
        cursor = tops[index]
        rect = [px, cursor, pw, ph]
        bead_slots = sorted(rng.sample(range(count), plate_beads[index]))
        holes = []
        for slot in range(count):
            hx = px + wall + slot * (HOLE[0] + wall + 4)
            hy = cursor + wall
            bead = None
            if slot in bead_slots:
                bead = {
                    "color": list(palette[index]),
                    "rect": [hx + (HOLE[0] - bead_w) // 2,
                             hy + (HOLE[1] - bead_h) // 2, bead_w, bead_h],
                }
            holes.append({"rect": [hx, hy, HOLE[0], HOLE[1]],
                          "bead": bead, "breached": False})
        plates.append({"rect": rect, "holes": holes})

    every = [(pi, hi) for pi, plate in enumerate(plates)
             for hi in range(len(plate["holes"]))]
    with_bead = [k for k in every if plates[k[0]]["holes"][k[1]]["bead"] is not None]
    without = [k for k in every if plates[k[0]]["holes"][k[1]]["bead"] is None]
    if rng.random() < 0.5:
        # bias the cut towards the slab that holds two beads, so that both slabs
        # almost always keep decisive ink of their own
        rich = 0 if plate_beads[0] == 2 else 1
        pool = [k for k in with_bead if k[0] == rich] if rng.random() < 0.85 else with_bead
        target = rng.choice(pool)
    else:
        target = rng.choice(without)
    plates[target[0]]["holes"][target[1]]["breached"] = True

    block = [ox, oy, block_w, block_h]
    decor = _sample_decor(rng, block)
    return {
        "canvas": list(CANVAS),
        "wall": wall,
        "channel": channel,
        "bead_size": [bead_w, bead_h],
        "block": block,
        "plates": plates,
        "decor": decor,
    }


def _sample_decor(rng: random.Random, block: list[int]) -> list[dict]:
    bx, by, bw, bh = block
    keep = (bx - 30, by - 30, bx + bw + 30, by + bh + 30)
    decor: list[dict] = []
    tries = 0
    while len(decor) < 9 and tries < 400:
        tries += 1
        kind = rng.choice(["slab", "ring", "bead", "bar"])
        w = rng.randint(34, 96) if kind in ("slab", "ring") else rng.randint(12, 26)
        h = rng.randint(30, 56) if kind in ("slab", "ring") else rng.randint(10, 22)
        if kind == "bar":
            w, h = rng.randint(40, 120), rng.randint(8, 14)
        x = rng.randint(12, CANVAS[0] - w - 12)
        y = rng.randint(12, CANVAS[1] - h - 12)
        if x + w + 8 > keep[0] and x - 8 < keep[2] and y + h + 8 > keep[1] and y - 8 < keep[3]:
            continue
        if any(x + w + 6 > d["rect"][0] and x - 6 < d["rect"][0] + d["rect"][2]
               and y + h + 6 > d["rect"][1] and y - 6 < d["rect"][1] + d["rect"][3]
               for d in decor):
            continue
        item = {"kind": kind, "rect": [x, y, w, h],
                "ink": list(rng.choice(GREY_INK)),
                "bead": list(rng.choice(BEAD_COLORS)) if kind in ("ring", "bead") else None}
        decor.append(item)
    return decor


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), WHITE)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        x, y, w, h = item["rect"]
        ink = tuple(item["ink"])
        if item["kind"] == "slab":
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=ink)
            draw.rectangle([x + 10, y + 9, x + w - 11, y + h - 10], fill=WHITE)
        elif item["kind"] == "ring":
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=ink)
            draw.rectangle([x + 9, y + 8, x + w - 10, y + h - 9], fill=WHITE)
            bx, by = x + w // 2, y + h // 2
            draw.rectangle([bx - 6, by - 4, bx + 5, by + 4], fill=tuple(item["bead"]))
        elif item["kind"] == "bar":
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=ink)
        else:
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=tuple(item["bead"]))

    channel = scene["channel"]
    for plate in scene["plates"]:
        px, py, pw, ph = plate["rect"]
        draw.rectangle([px, py, px + pw - 1, py + ph - 1], fill=BLACK)
        for hole in plate["holes"]:
            hx, hy, hw, hh = hole["rect"]
            draw.rectangle([hx, hy, hx + hw - 1, hy + hh - 1], fill=WHITE)
            if hole["breached"]:
                cx = hx + (hw - channel) // 2
                draw.rectangle([cx, hy + hh, cx + channel - 1, py + ph - 1], fill=WHITE)
        for hole in plate["holes"]:
            if hole["bead"] is not None:
                bx, by, bw, bh = hole["bead"]["rect"]
                draw.rectangle([bx, by, bx + bw - 1, by + bh - 1],
                               fill=tuple(hole["bead"]["color"]))
    return image


def _shut_beads(scene: dict) -> int:
    return sum(1 for plate in scene["plates"] for hole in plate["holes"]
               if hole["bead"] is not None and not hole["breached"])


def analytic_gold(scene: dict) -> str:
    return str(_shut_beads(scene))


def margin(scene: dict) -> float:
    wall = float(scene["wall"])
    bead_w, bead_h = scene["bead_size"]
    clearance = min((HOLE[0] - bead_w) / 2.0, (HOLE[1] - bead_h) / 2.0)
    widths = [wall, float(scene["channel"]), clearance, float(wall + 4)]
    return float(min(widths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    return []
