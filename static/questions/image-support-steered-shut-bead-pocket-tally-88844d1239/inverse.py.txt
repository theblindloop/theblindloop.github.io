"""Independent pixel-only arm for `shut_bead_pocket_tally`.

Receives one RGB image and nothing else: no scene, no palette table, no canvas
size, no plate positions and no analytic gold. It thresholds near-black barrier
ink, labels the barrier components, and for each component decides which white
regions are true holes of that component by testing whether a white region
reaches the border of a padded box around it. A hole counts when it holds a
saturated bead blob; the reading is the number of such holes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BLACK_MAX = 90
SAT_MIN = 70
BEAD_MIN_PX = 12
BEAD_AMBIGUOUS = 4
BARRIER_MIN_PX = 250
HOLE_MIN_PX = 40


def _components(mask: np.ndarray, min_size: int) -> list[np.ndarray]:
    """Label 4-connected components of a boolean mask; return index arrays."""
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[np.ndarray] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        seen[start_y, start_x] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            if y > 0 and mask[y - 1, x] and not seen[y - 1, x]:
                seen[y - 1, x] = True
                stack.append((y - 1, x))
            if y + 1 < height and mask[y + 1, x] and not seen[y + 1, x]:
                seen[y + 1, x] = True
                stack.append((y + 1, x))
            if x > 0 and mask[y, x - 1] and not seen[y, x - 1]:
                seen[y, x - 1] = True
                stack.append((y, x - 1))
            if x + 1 < width and mask[y, x + 1] and not seen[y, x + 1]:
                seen[y, x + 1] = True
                stack.append((y, x + 1))
        if len(pixels) >= min_size:
            out.append(np.array(pixels, dtype=np.int64))
    return out


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB")).astype(np.int16)
    barrier = array.max(axis=2) < BLACK_MAX
    spread = array.max(axis=2) - array.min(axis=2)
    bead = (spread >= SAT_MIN) & (~barrier)

    shut = 0
    ambiguous = False
    for pixels in _components(barrier, BARRIER_MIN_PX):
        y0, x0 = pixels[:, 0].min(), pixels[:, 1].min()
        y1, x1 = pixels[:, 0].max(), pixels[:, 1].max()
        pad = 3
        ty0, tx0 = max(0, y0 - pad), max(0, x0 - pad)
        ty1, tx1 = min(array.shape[0] - 1, y1 + pad), min(array.shape[1] - 1, x1 + pad)
        box = barrier[ty0:ty1 + 1, tx0:tx1 + 1]
        free = ~box
        for hole in _components(free, HOLE_MIN_PX):
            hy, hx = hole[:, 0], hole[:, 1]
            touches = (hy.min() == 0 or hx.min() == 0
                       or hy.max() == box.shape[0] - 1 or hx.max() == box.shape[1] - 1)
            if touches:
                continue
            local = bead[ty0:ty1 + 1, tx0:tx1 + 1]
            count = int(local[hy, hx].sum())
            if count >= BEAD_MIN_PX:
                shut += 1
            elif count > BEAD_AMBIGUOUS:
                ambiguous = True
    if ambiguous:
        return "abstain"
    return str(shut)
