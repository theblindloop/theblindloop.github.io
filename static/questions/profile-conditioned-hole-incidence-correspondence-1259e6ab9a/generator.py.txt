"""Deterministic renderer and analytic arm for hole-incidence correspondence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 620
BACKGROUND = (246, 248, 252)
CARD_FILL = (255, 255, 255)
INK = (28, 33, 42)
MUTED = (91, 101, 117)
MAGENTA = (216, 42, 144)
FRAME_PALETTE = {
    "red": (218, 56, 66),
    "green": (27, 153, 91),
    "blue": (45, 101, 218),
    "orange": (232, 126, 30),
}
ANSWER_NAMES = tuple(FRAME_PALETTE)
SIGNATURES = ((0, 0, 4), (0, 1, 3), (0, 2, 2), (1, 1, 2))
REFERENCE_BOX = (403, 58, 597, 278)
TARGET_BOXES = tuple((27 + 243 * slot, 365, 221 + 243 * slot, 585) for slot in range(4))


def _hole_offsets(count: int, variant: int) -> list[list[int]]:
    if count == 0:
        return []
    if count == 1:
        return [[0, 0]]
    if count == 2:
        return [[-10, 0], [10, 0]] if variant % 2 else [[0, -23], [0, 23]]
    if count == 3:
        points = [[0, -25], [-11, 16], [11, 16]]
        return [[x, -y] for x, y in points] if variant % 2 else points
    if count == 4:
        return [[-10, -24], [10, -24], [-10, 24], [10, 24]]
    raise ValueError(f"unsupported hole count {count}")


def _make_emblem(rng: random.Random, signature: tuple[int, int, int], frame_color: str, slot: int) -> dict:
    counts = list(signature)
    rng.shuffle(counts)
    shapes = ["roundrect", "ellipse", "hexagon"]
    rng.shuffle(shapes)
    pieces = []
    for piece_slot, (count, shape) in enumerate(zip(counts, shapes, strict=True)):
        variant = rng.randrange(2)
        radius = rng.choice((5, 6))
        pieces.append({
            "slot": piece_slot,
            "shape": shape,
            "width": rng.choice((44, 46, 48)),
            "height": rng.choice((120, 126, 132)),
            "dx": rng.randint(-2, 2),
            "dy": rng.randint(-5, 5),
            "hole_radius": radius,
            "holes": _hole_offsets(count, variant),
        })
    rng.shuffle(pieces)
    return {"frame_color": frame_color, "slot": slot, "pieces": pieces}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = ANSWER_NAMES[int(seed) % len(ANSWER_NAMES)]
    reference_signature = SIGNATURES[(int(seed) // len(ANSWER_NAMES)) % len(SIGNATURES)]
    reference = _make_emblem(rng, reference_signature, "magenta", 0)

    remaining_colors = [name for name in ANSWER_NAMES if name != answer]
    remaining_signatures = [sig for sig in SIGNATURES if sig != reference_signature]
    rng.shuffle(remaining_colors)
    rng.shuffle(remaining_signatures)
    assignments = [(answer, reference_signature), *zip(remaining_colors, remaining_signatures, strict=True)]
    rng.shuffle(assignments)
    targets = [
        _make_emblem(rng, signature, color, slot)
        for slot, (color, signature) in enumerate(assignments)
    ]
    rng.shuffle(targets)
    return {"width": WIDTH, "height": HEIGHT, "reference": reference, "targets": targets}


def _topology(pieces: list[dict]) -> tuple[int, ...]:
    return tuple(sorted(len(piece["holes"]) for piece in pieces))


def analytic_gold(scene: dict) -> str:
    reference_signature = _topology(scene["reference"]["pieces"])
    matches = [
        target["frame_color"]
        for target in scene["targets"]
        if _topology(target["pieces"]) == reference_signature
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one topology match, found {matches}")
    return matches[0]


def _distance(left: tuple[int, ...], right: tuple[int, ...]) -> int:
    return sum(abs(a - b) for a, b in zip(left, right, strict=True))


def margin(scene: dict) -> float:
    reference_signature = _topology(scene["reference"]["pieces"])
    distances = sorted(_distance(reference_signature, _topology(target["pieces"])) for target in scene["targets"])
    return float(distances[1] - distances[0]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    ref = _topology(scene["reference"]["pieces"])
    targets = [_topology(target["pieces"]) for target in scene["targets"]]
    all_emblems = [ref, *targets]
    return (
        any(len(sig) != 3 or sum(sig) != 4 for sig in all_emblems)
        or len(set(targets)) != 4
        or targets.count(ref) != 1
        or margin(scene) <= 0
    )


def _card_box(emblem: dict, reference: bool) -> tuple[int, int, int, int]:
    return REFERENCE_BOX if reference else TARGET_BOXES[int(emblem["slot"])]


def _piece_center(box: tuple[int, int, int, int], piece: dict) -> tuple[int, int]:
    x0, y0, x1, y1 = box
    x = x0 + (31, 97, 163)[int(piece["slot"])] + int(piece["dx"])
    y = (y0 + y1) // 2 + 6 + int(piece["dy"])
    return x, y


def _draw_piece(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], piece: dict) -> None:
    cx, cy = _piece_center(box, piece)
    half_w, half_h = int(piece["width"]) // 2, int(piece["height"]) // 2
    bounds = (cx - half_w, cy - half_h, cx + half_w, cy + half_h)
    if piece["shape"] == "roundrect":
        draw.rounded_rectangle(bounds, radius=12, fill=INK)
    elif piece["shape"] == "ellipse":
        draw.ellipse(bounds, fill=INK)
    elif piece["shape"] == "hexagon":
        draw.polygon([
            (cx, cy - half_h), (cx + half_w, cy - half_h // 2),
            (cx + half_w, cy + half_h // 2), (cx, cy + half_h),
            (cx - half_w, cy + half_h // 2), (cx - half_w, cy - half_h // 2),
        ], fill=INK)
    else:
        raise ValueError(f"unknown piece shape {piece['shape']!r}")
    radius = int(piece["hole_radius"])
    for dx, dy in piece["holes"]:
        hx, hy = cx + int(dx), cy + int(dy)
        draw.ellipse((hx - radius, hy - radius, hx + radius, hy + radius), fill=CARD_FILL)


def _draw_emblem(draw: ImageDraw.ImageDraw, emblem: dict, reference: bool) -> None:
    box = _card_box(emblem, reference)
    color = MAGENTA if reference else FRAME_PALETTE[emblem["frame_color"]]
    draw.rounded_rectangle(box, radius=15, fill=CARD_FILL, outline=color, width=7)
    for piece in sorted(emblem["pieces"], key=lambda item: item["slot"]):
        _draw_piece(draw, box, piece)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((466, 27), "REFERENCE", fill=MUTED, font=font)
    draw.text((475, 333), "TARGETS", fill=MUTED, font=font)
    draw.line((70, 316, 930, 316), fill=(205, 211, 221), width=2)
    _draw_emblem(draw, scene["reference"], True)
    for target in sorted(scene["targets"], key=lambda item: item["slot"]):
        _draw_emblem(draw, target, False)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["reference"]["pieces"] = list(reversed(twin["reference"]["pieces"]))
    twin["targets"] = list(reversed(twin["targets"]))
    for target in twin["targets"]:
        target["pieces"] = list(reversed(target["pieces"]))
    return [("reverse_slot_keyed_record_order", twin)]
