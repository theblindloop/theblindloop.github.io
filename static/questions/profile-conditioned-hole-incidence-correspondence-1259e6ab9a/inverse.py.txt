"""Independent PNG-only inverse for hole-incidence emblem correspondence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

FRAME_PALETTE = {
    "magenta": (216, 42, 144),
    "red": (218, 56, 66),
    "green": (27, 153, 91),
    "blue": (45, 101, 218),
    "orange": (232, 126, 30),
}
COLOR_RADIUS = 55.0


def _color_mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= COLOR_RADIUS * COLOR_RADIUS


def _frame_box(array: np.ndarray, name: str) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(_color_mask(array, FRAME_PALETTE[name]))
    if len(xs) < 1200:
        raise ValueError(f"{name} frame is missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if x1 - x0 < 150 or y1 - y0 < 170:
        raise ValueError(f"{name} frame is malformed")
    return x0, y0, x1, y1


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    records = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        touches = False
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            touches = touches or cy == 0 or cx == 0 or cy == height - 1 or cx == width - 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = [p[0] for p in pixels]
            xs = [p[1] for p in pixels]
            records.append({
                "area": len(pixels), "touches": touches,
                "box": (min(xs), min(ys), max(xs), max(ys)), "pixels": pixels,
            })
    return records


def _holes_in_component(component: dict) -> int:
    x0, y0, x1, y1 = component["box"]
    local = np.zeros((y1 - y0 + 3, x1 - x0 + 3), dtype=bool)
    for y, x in component["pixels"]:
        local[y - y0 + 1, x - x0 + 1] = True
    background = _components(~local, minimum=20)
    return sum(not record["touches"] for record in background)


def _signature(array: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, ...]:
    x0, y0, x1, y1 = box
    interior = array[y0 + 11:y1 - 10, x0 + 11:x1 - 10]
    black = np.max(interior, axis=2) < 105
    pieces = _components(black, minimum=500)
    if len(pieces) != 3:
        raise ValueError(f"emblem has {len(pieces)} black pieces instead of three")
    signature = tuple(sorted(_holes_in_component(piece) for piece in pieces))
    if sum(signature) != 4:
        raise ValueError(f"emblem has hole signature {signature}, not four total holes")
    return signature


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.shape[1] < 800 or array.shape[0] < 500:
        raise ValueError("image is too small for the public emblem design")
    reference = _signature(array, _frame_box(array, "magenta"))
    matches = []
    for name in ("red", "green", "blue", "orange"):
        if _signature(array, _frame_box(array, name)) == reference:
            matches.append(name)
    if len(matches) != 1:
        raise ValueError(f"raster has {len(matches)} topology matches")
    return matches[0]
