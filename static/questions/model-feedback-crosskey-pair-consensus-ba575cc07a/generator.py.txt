"""Render four locally re-keyed badge-pair ledgers."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

N = 5
BOARD_COUNT = 4
WIDTH = 1400
HEIGHT = 930
CARD_W = 660
CARD_H = 350
CARD_POSITIONS = ((30, 130), (710, 130), (30, 510), (710, 510))
ROW_OFFSET = 108
ROW_STEP = 43

BG = (228, 236, 240)
PAPER = (255, 254, 249)
CARD = (249, 251, 250)
INK = (30, 41, 51)
MUTED = (79, 92, 103)
RULE = (174, 187, 195)
ROW_TINT = (237, 243, 245)
BLUE_HEAD = (38, 116, 166)
AMBER_HEAD = (190, 112, 31)

SOURCE_COLORS = (
    (207, 70, 69),
    (53, 111, 176),
    (61, 147, 91),
    (139, 84, 171),
    (221, 138, 42),
)
LEFT_COLORS = (
    (18, 143, 151),
    (190, 72, 126),
    (54, 75, 137),
    (116, 151, 47),
    (31, 157, 200),
)
RIGHT_COLORS = (
    (232, 96, 80),
    (71, 131, 207),
    (75, 169, 103),
    (158, 97, 191),
    (232, 162, 48),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _draw_badge(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    identity: int,
    palette: tuple[tuple[int, int, int], ...],
) -> None:
    """Identity is redundant in both saturated color and silhouette."""
    cx, cy = center
    color = palette[identity]
    r = 15
    if identity == 0:
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=color, outline=INK, width=2)
    elif identity == 1:
        draw.rounded_rectangle((cx - r, cy - r, cx + r, cy + r), radius=4,
                               fill=color, outline=INK, width=2)
    elif identity == 2:
        points = ((cx, cy - 18), (cx + 18, cy), (cx, cy + 18), (cx - 18, cy))
        draw.polygon(points, fill=color)
        draw.line((*points, points[0]), fill=INK, width=2)
    elif identity == 3:
        points = ((cx, cy - 18), (cx + 18, cy + 15), (cx - 18, cy + 15))
        draw.polygon(points, fill=color)
        draw.line((*points, points[0]), fill=INK, width=2)
    else:
        points = ((cx - 15, cy - 12), (cx, cy - 18), (cx + 15, cy - 12),
                  (cx + 15, cy + 12), (cx, cy + 18), (cx - 15, cy + 12))
        draw.polygon(points, fill=color)
        draw.line((*points, points[0]), fill=INK, width=2)


def _draw_source(draw: ImageDraw.ImageDraw, center: tuple[int, int], source: int) -> None:
    cx, cy = center
    draw.rounded_rectangle((cx - 17, cy - 17, cx + 17, cy + 17), radius=8,
                           fill=SOURCE_COLORS[source], outline=INK, width=2)
    draw.text((cx, cy), chr(ord("A") + source), font=_font(19, True),
              fill=(255, 255, 255), anchor="mm")


def _draw_arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    x0, y0 = start
    x1, y1 = end
    draw.line((x0, y0, x1 - 7, y1), fill=MUTED, width=3)
    draw.polygon(((x1, y1), (x1 - 9, y1 - 6), (x1 - 9, y1 + 6)), fill=MUTED)


def _accepted_sources(scene: dict, board_index: int) -> list[int]:
    board = scene["boards"][board_index]
    left = board["left_for_source"]
    right = board["right_for_source"]
    partner = board["partner_for_left"]
    return [source for source in range(N) if partner[left[source]] == right[source]]


def _consensus_source(scene: dict) -> int:
    accepted = [_accepted_sources(scene, index) for index in range(len(scene["boards"]))]
    winners = [source for source in range(N) if all(source in group for group in accepted)]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one all-board partner consensus")
    return winners[0]


def _make_boards(rng: random.Random, answer: int) -> list[dict]:
    distractors = [source for source in range(N) if source != answer]
    rng.shuffle(distractors)
    d1, d2, d3, d4 = distractors
    invalid_by_board = ((d2, d3), (d3, d4), (d1, d4), (d1, d2))

    for _attempt in range(2000):
        left_identity = list(range(N))
        right_identity = list(range(N))
        rng.shuffle(left_identity)
        rng.shuffle(right_identity)
        left_offsets = rng.sample(range(N), BOARD_COUNT)
        right_offsets = rng.sample(range(N), BOARD_COUNT)
        rng.shuffle(left_offsets)
        rng.shuffle(right_offsets)
        boards: list[dict] = []
        for board_index in range(BOARD_COUNT):
            left = [left_identity[(source + left_offsets[board_index]) % N]
                    for source in range(N)]
            right = [right_identity[(source + right_offsets[board_index]) % N]
                     for source in range(N)]
            source_partner = list(range(N))
            first_bad, second_bad = invalid_by_board[board_index]
            source_partner[first_bad], source_partner[second_bad] = second_bad, first_bad
            partner = [-1] * N
            for source in range(N):
                partner[left[source]] = right[source_partner[source]]
            source_order = list(range(N))
            key_order = list(range(N))
            rng.shuffle(source_order)
            rng.shuffle(key_order)
            boards.append({
                "left_for_source": left,
                "right_for_source": right,
                "partner_for_left": partner,
                "source_order": source_order,
                "key_order": key_order,
            })
        if all(len({board["partner_for_left"][token] for board in boards}) >= 2
               for token in range(N)):
            rng.shuffle(boards)
            return boards
    raise RuntimeError("could not construct varied local partner keys")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0xC2055E7)
    answer = seed % N
    scene = {"width": WIDTH, "height": HEIGHT, "boards": _make_boards(rng, answer)}
    if analytic_gold(scene) != f"letter-{chr(ord('A') + answer)}":
        raise RuntimeError("construction lost intended consensus")
    if is_quarantined(scene):
        raise RuntimeError("constructed scene was unexpectedly quarantined")
    return scene


def _draw_board(draw: ImageDraw.ImageDraw, board: dict, index: int,
                origin: tuple[int, int]) -> None:
    x, y = origin
    draw.rounded_rectangle((x, y, x + CARD_W, y + CARD_H), radius=16,
                           fill=CARD, outline=INK, width=3)
    draw.text((x + CARD_W // 2, y + 22), f"BOARD {index + 1}",
              font=_font(19, True), fill=INK, anchor="mm")
    draw.line((x + 15, y + 44, x + CARD_W - 15, y + 44), fill=RULE, width=2)
    draw.text((x + 157, y + 65), "LETTER'S TWO BADGES", font=_font(15, True),
              fill=MUTED, anchor="mm")
    draw.text((x + 500, y + 65), "PAIR KEY", font=_font(15, True),
              fill=MUTED, anchor="mm")
    draw.text((x + 148, y + 88), "LEFT", font=_font(13, True), fill=BLUE_HEAD, anchor="mm")
    draw.text((x + 260, y + 88), "RIGHT", font=_font(13, True), fill=AMBER_HEAD, anchor="mm")
    draw.text((x + 400, y + 88), "LEFT", font=_font(13, True), fill=BLUE_HEAD, anchor="mm")
    draw.text((x + 562, y + 88), "RIGHT", font=_font(13, True), fill=AMBER_HEAD, anchor="mm")
    draw.line((x + 326, y + 52, x + 326, y + CARD_H - 14), fill=RULE, width=2)

    for row in range(N):
        cy = y + ROW_OFFSET + row * ROW_STEP
        if row % 2 == 0:
            draw.rounded_rectangle((x + 12, cy - 19, x + 312, cy + 19), radius=8, fill=ROW_TINT)
            draw.rounded_rectangle((x + 342, cy - 19, x + 648, cy + 19), radius=8, fill=ROW_TINT)

        source = int(board["source_order"][row])
        _draw_source(draw, (x + 42, cy), source)
        _draw_badge(draw, (x + 148, cy), int(board["left_for_source"][source]), LEFT_COLORS)
        draw.text((x + 204, cy), "+", font=_font(21, True), fill=MUTED, anchor="mm")
        _draw_badge(draw, (x + 260, cy), int(board["right_for_source"][source]), RIGHT_COLORS)

        left_token = int(board["key_order"][row])
        _draw_badge(draw, (x + 400, cy), left_token, LEFT_COLORS)
        _draw_arrow(draw, (x + 429, cy), (x + 531, cy))
        _draw_badge(draw, (x + 562, cy), int(board["partner_for_left"][left_token]), RIGHT_COLORS)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((10, 10, WIDTH - 10, HEIGHT - 10), radius=24,
                           fill=PAPER, outline=INK, width=3)
    draw.text((WIDTH // 2, 34), "FOUR LOCAL PAIR KEYS", font=_font(27, True), fill=INK, anchor="mm")
    draw.text((WIDTH // 2, 68), "A letter passes a board only when its LEFT + RIGHT badges appear together in that board's PAIR KEY.",
              font=_font(18, True), fill=MUTED, anchor="mm")
    draw.text((WIDTH // 2, 99), "Which letter passes every board?", font=_font(22, True),
              fill=(174, 57, 61), anchor="mm")
    for index, (board, position) in enumerate(zip(scene["boards"], CARD_POSITIONS)):
        _draw_board(draw, board, index, position)
    draw.text((WIDTH // 2, 894), "ANSWER:  LETTER A   •   LETTER B   •   LETTER C   •   LETTER D   •   LETTER E",
              font=_font(18, True), fill=INK, anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    return f"letter-{chr(ord('A') + _consensus_source(scene))}"


def margin(scene: dict) -> float:
    counts = [sum(source in _accepted_sources(scene, index)
                  for index in range(len(scene["boards"])))
              for source in range(N)]
    answer = _consensus_source(scene)
    return float(counts[answer] - max(count for source, count in enumerate(counts)
                                      if source != answer))


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["width"]) != WIDTH or int(scene["height"]) != HEIGHT:
            return True
        boards = scene["boards"]
        if len(boards) != BOARD_COUNT:
            return True
        for board_index, board in enumerate(boards):
            for field in ("left_for_source", "right_for_source", "partner_for_left",
                          "source_order", "key_order"):
                if sorted(map(int, board[field])) != list(range(N)):
                    return True
            if len(_accepted_sources(scene, board_index)) != 3:
                return True
        answer = _consensus_source(scene)
        counts = [sum(source in _accepted_sources(scene, index)
                      for index in range(len(boards)))
                  for source in range(N)]
        if counts[answer] != BOARD_COUNT:
            return True
        if any(count != 2 for source, count in enumerate(counts) if source != answer):
            return True
        for source in range(N):
            if len({board["left_for_source"][source] for board in boards}) != BOARD_COUNT:
                return True
            if len({board["right_for_source"][source] for board in boards}) != BOARD_COUNT:
                return True
        if any(len({board["partner_for_left"][token] for board in boards}) < 2
               for token in range(N)):
            return True
        return margin(scene) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
