"""Independent fixed-layout RGB decoder for local badge-pair consensus."""

from __future__ import annotations

from PIL import Image

N = 5
BOARD_COUNT = 4
WIDTH = 1400
HEIGHT = 930
CARD_POSITIONS = ((30, 130), (710, 130), (30, 510), (710, 510))
ROW_OFFSET = 108
ROW_STEP = 43

SOURCE_COLORS = (
    (207, 70, 69), (53, 111, 176), (61, 147, 91), (139, 84, 171), (221, 138, 42),
)
LEFT_COLORS = (
    (18, 143, 151), (190, 72, 126), (54, 75, 137), (116, 151, 47), (31, 157, 200),
)
RIGHT_COLORS = (
    (232, 96, 80), (71, 131, 207), (75, 169, 103), (158, 97, 191), (232, 162, 48),
)


def _classify(pixels, cx: int, cy: int,
              palette: tuple[tuple[int, int, int], ...], radius: int = 18) -> int:
    counts = []
    for color in palette:
        count = 0
        for y in range(cy - radius, cy + radius + 1):
            for x in range(cx - radius, cx + radius + 1):
                pixel = pixels[x, y]
                if max(abs(int(pixel[channel]) - color[channel]) for channel in range(3)) <= 8:
                    count += 1
        counts.append(count)
    best = max(range(len(counts)), key=counts.__getitem__)
    ordered = sorted(counts, reverse=True)
    if counts[best] < 180 or ordered[1] > counts[best] // 3:
        raise ValueError("badge identity is missing or ambiguous")
    return best


def _decode_board(pixels, origin: tuple[int, int]) -> tuple[list[int], list[int], list[int]]:
    x, y = origin
    left = [-1] * N
    right = [-1] * N
    partner = [-1] * N
    for row in range(N):
        cy = y + ROW_OFFSET + row * ROW_STEP
        source = _classify(pixels, x + 42, cy, SOURCE_COLORS)
        if left[source] != -1:
            raise ValueError("duplicate visible source row")
        left[source] = _classify(pixels, x + 148, cy, LEFT_COLORS)
        right[source] = _classify(pixels, x + 260, cy, RIGHT_COLORS)

        left_token = _classify(pixels, x + 400, cy, LEFT_COLORS)
        if partner[left_token] != -1:
            raise ValueError("duplicate visible pair-key row")
        partner[left_token] = _classify(pixels, x + 562, cy, RIGHT_COLORS)
    if sorted(left) != list(range(N)) or sorted(right) != list(range(N)):
        raise ValueError("visible source assignments are not complete bijections")
    if sorted(partner) != list(range(N)):
        raise ValueError("visible pair key is not a complete bijection")
    return left, right, partner


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    pixels = raster.load()
    boards = [_decode_board(pixels, position) for position in CARD_POSITIONS]
    pass_sets = [
        {source for source in range(N) if partner[left[source]] == right[source]}
        for left, right, partner in boards
    ]
    if any(len(group) != 3 for group in pass_sets):
        raise ValueError("each visible board must accept exactly three letters")
    counts = [sum(source in group for group in pass_sets) for source in range(N)]
    winners = [source for source, count in enumerate(counts) if count == BOARD_COUNT]
    if len(winners) != 1:
        raise ValueError("visible boards do not identify one consensus letter")
    if any(count != 2 for source, count in enumerate(counts) if source != winners[0]):
        raise ValueError("visible distractor evidence is degenerate")
    return f"letter-{chr(ord('A') + winners[0])}"
