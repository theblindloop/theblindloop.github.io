"""Deterministic renderer and analytic measurement for paired box plots."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "box-whisker-asymmetry-0.1.0"
SCALE = 4
WIDTH, HEIGHT = 960, 500
PANELS = ((36, 460), (500, 924))
PANEL_TOP, PANEL_BOTTOM = 38, 462
PLOT_Y = (170.0, 294.0)
PALETTE = (
    "#d1495b",
    "#0077b6",
    "#2a9d8f",
    "#e09f3e",
    "#7b2cbf",
    "#e76f51",
    "#118ab2",
    "#6a994e",
)
INK = "#33404d"
FRAME = "#c8d0d8"
PANEL_FILL = "#fbfcfe"


def _font(size: int):
    try:
        return ImageFont.truetype(
            "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
        )
    except OSError:
        return ImageFont.load_default()


def _tail_stat(a_min: float, a_q1: float, a_q3: float, a_max: float) -> float:
    total = a_max - a_min
    left_tail = a_q1 - a_min
    right_tail = a_max - a_q3
    return abs(left_tail - right_tail) / total


def _sample_summary(rng: random.Random, target: str) -> tuple[float, float, float, float]:
    """Sample an ordered five-number summary with a controlled tail imbalance."""

    if target == "near":
        imbalance = rng.uniform(0.040, 0.060)
    elif target == "low":
        imbalance = rng.uniform(0.105, 0.150)
    else:
        imbalance = rng.uniform(0.300, 0.460)
    total = rng.uniform(0.53, 0.68)
    origin = rng.uniform(0.12, 0.88 - total)
    left_tail = rng.uniform(0.10, 0.25)
    sign = rng.choice((-1.0, 1.0))
    right_tail = left_tail + sign * imbalance
    if right_tail < 0.085 or left_tail + right_tail > 0.78:
        right_tail = left_tail + imbalance
        if right_tail > 0.54:
            left_tail = max(0.09, 0.70 - right_tail)
    values = (
        origin,
        origin + left_tail * total,
        origin + (1.0 - right_tail) * total,
        origin + total,
    )
    if not (0.0 < values[0] < values[1] < values[2] < values[3] < 1.0):
        raise RuntimeError("invalid sampled box summary")
    return tuple(round(value, 6) for value in values)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair with independent visible nuisance geometry."""

    rng = random.Random(int(seed))
    # The odd/even seed parity gives an exact alternating answer balance for the
    # public stress schedules, while every geometric field remains randomized.
    winner = "A" if int(seed) % 2 == 0 else "B"
    low_target = "near" if int(seed) % 11 == 0 else "low"
    a_target, b_target = ("high", low_target) if winner == "A" else (low_target, "high")
    a_min, a_q1, a_q3, a_max = _sample_summary(rng, a_target)
    b_min, b_q1, b_q3, b_max = _sample_summary(rng, b_target)
    a_left, a_right = PANELS[0]
    b_left, b_right = PANELS[1]
    a_plot_left = round(rng.uniform(a_left + 54.0, a_left + 82.0), 6)
    a_plot_right = round(rng.uniform(a_right - 76.0, a_right - 48.0), 6)
    b_plot_left = round(rng.uniform(b_left + 48.0, b_left + 78.0), 6)
    b_plot_right = round(rng.uniform(b_right - 82.0, b_right - 50.0), 6)
    a_y = round(rng.uniform(PLOT_Y[0], PLOT_Y[1]), 6)
    b_y = round(rng.uniform(PLOT_Y[0], PLOT_Y[1]), 6)
    return {
        "a_min": a_min,
        "a_q1": a_q1,
        "a_q3": a_q3,
        "a_max": a_max,
        "b_min": b_min,
        "b_q1": b_q1,
        "b_q3": b_q3,
        "b_max": b_max,
        "a_plot_left": a_plot_left,
        "a_plot_right": a_plot_right,
        "b_plot_left": b_plot_left,
        "b_plot_right": b_plot_right,
        "a_y": a_y,
        "b_y": b_y,
        "a_box_half": round(rng.uniform(28.0, 37.0), 6),
        "b_box_half": round(rng.uniform(28.0, 37.0), 6),
        "a_color": rng.choice(PALETTE),
        "b_color": rng.choice(PALETTE),
    }


def _x(value: float, left: float, right: float) -> float:
    return left + value * (right - left)


def _draw_panel_frame(draw: ImageDraw.ImageDraw, left: int, right: int, label: str) -> None:
    draw.rounded_rectangle(
        (left * SCALE, PANEL_TOP * SCALE, right * SCALE, PANEL_BOTTOM * SCALE),
        radius=13 * SCALE,
        fill=PANEL_FILL,
        outline=FRAME,
        width=2 * SCALE,
    )
    font = _font(25)
    bbox = draw.textbbox((0, 0), label, font=font)
    text_width = bbox[2] - bbox[0]
    draw.text(
        (((left + right) * SCALE - text_width) / 2.0, 55 * SCALE),
        label,
        fill=INK,
        font=font,
    )


def _draw_axis(draw: ImageDraw.ImageDraw, left: float, right: float) -> None:
    axis_y = 350.0
    draw.line((left * SCALE, axis_y * SCALE, right * SCALE, axis_y * SCALE), fill=INK, width=2 * SCALE)
    tick_font = _font(12)
    for fraction, text in ((0.0, "0"), (0.25, "¼"), (0.50, "½"), (0.75, "¾"), (1.0, "1")):
        x = _x(fraction, left, right)
        draw.line((x * SCALE, (axis_y - 6) * SCALE, x * SCALE, (axis_y + 6) * SCALE), fill=INK, width=2 * SCALE)
        bbox = draw.textbbox((0, 0), text, font=tick_font)
        draw.text((x * SCALE - (bbox[2] - bbox[0]) / 2.0, (axis_y + 11) * SCALE), text, fill=INK, font=tick_font)


def _draw_boxplot(
    draw: ImageDraw.ImageDraw,
    summary: tuple[float, float, float, float],
    left: float,
    right: float,
    center_y: float,
    box_half: float,
    color: str,
) -> None:
    minimum, q1, q3, maximum = summary
    x_min, x_q1, x_q3, x_max = (_x(value, left, right) for value in summary)
    y = center_y
    # The complete glyph uses one saturated color, so a pixel-only inverse can
    # recover its four ordered breakpoints without reading renderer state.
    line_width = 8 * SCALE
    cap_width = 5 * SCALE
    draw.line((x_min * SCALE, y * SCALE, x_q1 * SCALE, y * SCALE), fill=color, width=line_width)
    draw.line((x_q3 * SCALE, y * SCALE, x_max * SCALE, y * SCALE), fill=color, width=line_width)
    draw.line((x_min * SCALE, (y - box_half) * SCALE, x_min * SCALE, (y + box_half) * SCALE), fill=color, width=cap_width)
    draw.line((x_max * SCALE, (y - box_half) * SCALE, x_max * SCALE, (y + box_half) * SCALE), fill=color, width=cap_width)
    draw.rectangle(
        (x_q1 * SCALE, (y - box_half) * SCALE, x_q3 * SCALE, (y + box_half) * SCALE),
        fill=color,
        outline=color,
        width=4 * SCALE,
    )
    # Median is a standard visible glyph element but deliberately not part of
    # the requested outer-whisker statistic.
    median_x = (x_q1 + x_q3) / 2.0
    draw.line(
        (median_x * SCALE, (y - box_half + 5) * SCALE, median_x * SCALE, (y + box_half - 5) * SCALE),
        fill=INK,
        width=4 * SCALE,
    )


def render(scene: dict) -> Image.Image:
    """Render two labeled, horizontal box-and-whisker plots."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(image)
    _draw_panel_frame(draw, *PANELS[0], "A")
    _draw_panel_frame(draw, *PANELS[1], "B")
    a_left = float(scene["a_plot_left"])
    a_right = float(scene["a_plot_right"])
    a_summary = (
        float(scene["a_min"]),
        float(scene["a_q1"]),
        float(scene["a_q3"]),
        float(scene["a_max"]),
    )
    _draw_axis(draw, a_left, a_right)
    _draw_boxplot(
        draw,
        a_summary,
        a_left,
        a_right,
        float(scene["a_y"]),
        float(scene["a_box_half"]),
        str(scene["a_color"]),
    )
    b_left = float(scene["b_plot_left"])
    b_right = float(scene["b_plot_right"])
    b_summary = (
        float(scene["b_min"]),
        float(scene["b_q1"]),
        float(scene["b_q3"]),
        float(scene["b_max"]),
    )
    _draw_axis(draw, b_left, b_right)
    _draw_boxplot(
        draw,
        b_summary,
        b_left,
        b_right,
        float(scene["b_y"]),
        float(scene["b_box_half"]),
        str(scene["b_color"]),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Select the panel with the larger normalized tail imbalance."""

    a_min = float(scene["a_min"])
    a_q1 = float(scene["a_q1"])
    a_q3 = float(scene["a_q3"])
    a_max = float(scene["a_max"])
    b_min = float(scene["b_min"])
    b_q1 = float(scene["b_q1"])
    b_q3 = float(scene["b_q3"])
    b_max = float(scene["b_max"])
    a_value = _tail_stat(a_min, a_q1, a_q3, a_max)
    b_value = _tail_stat(b_min, b_q1, b_q3, b_max)
    return "A" if a_value > b_value else "B"


def margin(scene: dict) -> float:
    """Signed A-minus-B difference; positive means A wins."""

    a_min = float(scene["a_min"])
    a_q1 = float(scene["a_q1"])
    a_q3 = float(scene["a_q3"])
    a_max = float(scene["a_max"])
    b_min = float(scene["b_min"])
    b_q1 = float(scene["b_q1"])
    b_q3 = float(scene["b_q3"])
    b_max = float(scene["b_max"])
    return _tail_stat(a_min, a_q1, a_q3, a_max) - _tail_stat(b_min, b_q1, b_q3, b_max)


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed because all label inputs are rendered."""

    return []
