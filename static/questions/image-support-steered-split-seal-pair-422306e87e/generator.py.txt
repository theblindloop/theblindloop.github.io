"""Renderer and analytic arm for shared-turn detached-pair correspondence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 768
GRID = 4
PLATE = 72
CELL = 11
STEP = 14
REFERENCE_CENTERS = ((348, 190), (424, 190))
CANDIDATE_CENTERS = {
    "left": ((280, 568), (356, 568)),
    "right": ((436, 568), (512, 568)),
}
TAG_CENTERS = {"left": (318, 625), "right": (474, 625)}
RGB = {
    "paper": (255, 255, 255),
    "plate": (218, 229, 237),
    "ink": (29, 38, 50),
    "red": (220, 55, 60),
    "blue": (45, 100, 210),
}


def _matrix(value) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(int(cell) for cell in row) for row in value)


def _rotate(value, turns: int) -> tuple[tuple[int, ...], ...]:
    out = _matrix(value)
    for _ in range(int(turns) % 4):
        out = tuple(tuple(out[GRID - 1 - c][r] for c in range(GRID)) for r in range(GRID))
    return out


def _hamming(left, right) -> int:
    a, b = _matrix(left), _matrix(right)
    return sum(a[r][c] != b[r][c] for r in range(GRID) for c in range(GRID))


def _pair_score(reference, pair) -> int:
    return min(
        _hamming(reference[0], _rotate(pair[0], k))
        + _hamming(reference[1], _rotate(pair[1], k))
        for k in range(4)
    )


def _random_pattern(rng: random.Random) -> tuple[tuple[int, ...], ...]:
    while True:
        values = [1] * 6 + [0] * 10
        rng.shuffle(values)
        item = tuple(tuple(values[r * GRID + c] for c in range(GRID)) for r in range(GRID))
        if min(_hamming(item, _rotate(item, k)) for k in (1, 2, 3)) >= 6:
            return item


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        reference = (_random_pattern(rng), _random_pattern(rng))
        if min(_hamming(reference[0], _rotate(reference[1], k)) for k in range(4)) >= 4:
            break
    common_turn = rng.randrange(4)
    match = (_rotate(reference[0], common_turn), _rotate(reference[1], common_turn))
    while True:
        first_turn, second_turn = rng.sample(range(4), 2)
        distractor = (_rotate(reference[0], first_turn), _rotate(reference[1], second_turn))
        if _pair_score(reference, distractor) >= 4:
            break

    match_slot = "left" if int(seed) % 2 == 0 else "right"
    answer = "red" if (int(seed) // 2) % 2 == 0 else "blue"
    other = "blue" if answer == "red" else "red"
    records = [
        {"slot": match_slot, "pair": match, "tag_color": answer},
        {"slot": "right" if match_slot == "left" else "left", "pair": distractor, "tag_color": other},
    ]
    rng.shuffle(records)
    return {
        "reference": [[list(row) for row in item] for item in reference],
        "records": [
            {"slot": record["slot"], "pair": [[list(row) for row in item] for item in record["pair"]], "tag_color": record["tag_color"]}
            for record in records
        ],
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_plate(draw: ImageDraw.ImageDraw, pattern, center: tuple[int, int]) -> None:
    cx, cy = center
    x0, y0 = cx - PLATE // 2, cy - PLATE // 2
    draw.rectangle((x0, y0, x0 + PLATE - 1, y0 + PLATE - 1), fill=RGB["plate"])
    for row in range(GRID):
        for col in range(GRID):
            if int(pattern[row][col]):
                left = x0 + 8 + col * STEP
                top = y0 + 8 + row * STEP
                draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=RGB["ink"])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), RGB["paper"])
    draw = ImageDraw.Draw(image)
    for pattern, center in zip(scene["reference"], REFERENCE_CENTERS, strict=True):
        _draw_plate(draw, pattern, center)
    for record in sorted(scene["records"], key=lambda item: item["slot"]):
        centers = CANDIDATE_CENTERS[record["slot"]]
        for pattern, center in zip(record["pair"], centers, strict=True):
            _draw_plate(draw, pattern, center)
        tx, ty = TAG_CENTERS[record["slot"]]
        draw.ellipse((tx - 13, ty - 13, tx + 13, ty + 13), fill=RGB[record["tag_color"]])
    return image


def _records_by_slot(scene: dict) -> list[dict]:
    return sorted(scene["records"], key=lambda item: item["slot"])


def analytic_gold(scene: dict) -> str:
    reference = tuple(_matrix(item) for item in scene["reference"])
    matches = [record for record in _records_by_slot(scene) if _pair_score(reference, record["pair"]) == 0]
    if len(matches) != 1:
        raise ValueError("scene lacks one unique shared-turn match")
    return str(matches[0]["tag_color"])


def margin(scene: dict) -> float:
    reference = tuple(_matrix(item) for item in scene["reference"])
    scores = sorted(_pair_score(reference, record["pair"]) for record in scene["records"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        reference = tuple(_matrix(item) for item in scene["reference"])
        scores = [_pair_score(reference, record["pair"]) for record in scene["records"]]
    except Exception:
        return True
    return scores.count(0) != 1 or sorted(scores)[1] < 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["records"] = list(reversed(twin["records"]))
    return [("reverse_slotted_record_storage", twin)]
