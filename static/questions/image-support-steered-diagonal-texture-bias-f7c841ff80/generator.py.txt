"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations
import random
from PIL import Image, ImageDraw

N = 12
PALETTES = [
    ["#1565C0", "#F9A825"],
    ["#6A1B9A", "#2E7D32"],
    ["#C62828", "#00838F"],
]

def _scores(grid: list[list[int]]) -> tuple[int, int]:
    rising = sum(grid[y][x] == grid[y-1][x+1] for y in range(1, N) for x in range(N-1))
    falling = sum(grid[y][x] == grid[y+1][x+1] for y in range(N-1) for x in range(N-1))
    return rising, falling

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = "rising" if int(seed) % 2 == 0 else "falling"
    for _ in range(20000):
        grid = [[rng.randrange(2) for _ in range(N)] for _ in range(N)]
        ones = sum(map(sum, grid))
        if not 62 <= ones <= 82:
            continue
        rising, falling = _scores(grid)
        diff = rising - falling
        if abs(diff) >= 14 and ((diff > 0) == (wanted == "rising")):
            break
    else:
        raise RuntimeError("could not sample a well-separated texture")
    palette = list(rng.choice(PALETTES))
    if rng.randrange(2):
        palette.reverse()
    cell_size = rng.choice([28, 30, 32])
    board = cell_size * N
    origin = [rng.randint(48, 512 - board - 48), rng.randint(48, 512 - board - 48)]
    return {"grid": grid, "palette": palette, "cell_size": cell_size, "origin": origin}

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (512, 512), "#F4F0E8")
    draw = ImageDraw.Draw(image)
    grid = scene["grid"]
    palette = scene["palette"]
    cell = int(scene["cell_size"])
    ox, oy = map(int, scene["origin"])
    for y, row in enumerate(grid):
        for x, value in enumerate(row):
            x0, y0 = ox + x * cell, oy + y * cell
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=palette[value])
    draw.rectangle((ox - 3, oy - 3, ox + N * cell + 2, oy + N * cell + 2), outline="#263238", width=3)
    return image

def analytic_gold(scene: dict) -> str:
    rising, falling = _scores(scene["grid"])
    return "rising" if rising > falling else "falling"

def margin(scene: dict) -> float:
    rising, falling = _scores(scene["grid"])
    return float(abs(rising - falling))

def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["grid"] = [[1 - value for value in row] for row in scene["grid"]]
    twin["palette"] = [scene["palette"][1], scene["palette"][0]]
    return [("complement_grid_and_swap_palette", twin)]
