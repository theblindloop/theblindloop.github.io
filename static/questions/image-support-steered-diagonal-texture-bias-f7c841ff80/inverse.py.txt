"""Independent image-only inverse arm for the dense swatch."""

from __future__ import annotations
import numpy as np
from PIL import Image

N = 12
BACKGROUND = np.array([244, 240, 232], dtype=np.int16)
BORDER = np.array([38, 50, 56], dtype=np.int16)

def _recover_grid(image: Image.Image) -> list[list[int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    bgdist = np.max(np.abs(arr - BACKGROUND), axis=2)
    borderdist = np.max(np.abs(arr - BORDER), axis=2)
    ink = (bgdist > 25) & (borderdist > 12)
    ys, xs = np.nonzero(ink)
    if len(xs) < 1000:
        raise ValueError("swatch is missing")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()) + 1, int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if abs(width - height) > 2 or width < 250:
        raise ValueError("swatch bounds are invalid")
    cell = width / N
    samples = np.array([
        arr[int(y0 + (y + 0.5) * cell), int(x0 + (x + 0.5) * cell)]
        for y in range(N) for x in range(N)
    ], dtype=np.float64)
    # Recover the two colors without knowing their RGB values: farthest pair seeds,
    # then deterministic nearest-center refinement.
    distances = ((samples[:, None, :] - samples[None, :, :]) ** 2).sum(axis=2)
    a, b = np.unravel_index(int(distances.argmax()), distances.shape)
    centers = np.stack([samples[a], samples[b]])
    for _ in range(6):
        labels = ((samples[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2).argmin(axis=1)
        if len(set(labels.tolist())) != 2:
            raise ValueError("could not recover two tile colors")
        centers = np.stack([samples[labels == k].mean(axis=0) for k in range(2)])
    return labels.reshape(N, N).tolist()

def decision_from_image(image: Image.Image) -> str:
    grid = _recover_grid(image)
    rising = sum(grid[y][x] == grid[y-1][x+1] for y in range(1, N) for x in range(N-1))
    falling = sum(grid[y][x] == grid[y+1][x+1] for y in range(N-1) for x in range(N-1))
    if rising == falling:
        raise ValueError("diagonal texture decision is tied")
    return "rising" if rising > falling else "falling"
