"""Independent pixel-only inverse arm for the toggle-register accumulator.

Receives only the final PNG. From the raster alone it:
  1. segments the panel frames by their gray border color;
  2. recovers panel order (left to right);
  3. classifies each token's binary state (filled disk vs hollow ring) from the
     ink at the token's center;
  4. reads each inter-panel operator glyph by its fill color (red circle =
     flip, blue square = keep);
  5. composes the sequence from the marked start state and returns the final
     state.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageColor

BG = (255, 255, 255)
INK = (0, 0, 0)
FLIP_COLOR = (210, 40, 40)
SAME_COLOR = (40, 60, 210)
RING_MARK = (240, 150, 20)
BORDER = (200, 200, 200)

PALETTE = [BG, INK, FLIP_COLOR, SAME_COLOR, RING_MARK, BORDER]
_BG, _INK, _FLIP, _SAME, _RING, _BRD = range(6)

MIN_PANELS = 4
WINDOW = 3


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """8-connected components of a boolean mask; returns sorted index pairs."""
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < mask.shape[0]
                        and 0 <= nx < mask.shape[1]
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        comps.append(points)
    return comps


def _bbox(points) -> tuple[int, int, int, int]:
    ys = [p[0] for p in points]
    xs = [p[1] for p in points]
    return min(xs), min(ys), max(xs), max(ys)


def _token_state(labels: np.ndarray, cx: int, cy: int) -> int:
    y0 = max(0, cy - WINDOW)
    y1 = min(labels.shape[0], cy + WINDOW + 1)
    x0 = max(0, cx - WINDOW)
    x1 = min(labels.shape[1], cx + WINDOW + 1)
    win = labels[y0:y1, x0:x1]
    bg_count = int((win == _BG).sum())
    ink_count = int((win == _INK).sum())
    if ink_count > bg_count:
        return 1
    if bg_count > ink_count:
        return 0
    raise UnreadableImageError("token state is ambiguous at the marked start")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    frames = []
    for points in _components(labels == _BRD):
        if len(points) < 40:  # drop tiny border specks
            continue
        frames.append(_bbox(points))
    frames.sort(key=lambda box: box[0])  # left-to-right panel order
    if len(frames) < MIN_PANELS:
        raise UnreadableImageError(f"found {len(frames)} panels; expected >= {MIN_PANELS}")

    centers = [((x0 + x1) // 2, (y0 + y1) // 2) for (x0, y0, x1, y1) in frames]
    states = [_token_state(labels, cx, cy) for (cx, cy) in centers]

    # Operators live in the horizontal gap between consecutive panel frames.
    ops = []
    for i in range(len(frames) - 1):
        _, _, rx, _ = frames[i]
        lx, _, _, _ = frames[i + 1]
        x0 = max(0, rx + 1)
        x1 = min(labels.shape[1], lx)
        y0 = max(0, centers[i][1] - 30)
        y1 = min(labels.shape[0], centers[i][1] + 30)
        region = labels[y0:y1, x0:x1]
        red = int((region == _FLIP).sum())
        blue = int((region == _SAME).sum())
        ops.append("flip" if red > blue else "same")
    if len(ops) < len(states) - 1:
        raise UnreadableImageError("not enough operator glyphs detected")

    composed = states[0]
    for op in ops:
        if op == "flip":
            composed = 1 - composed
    return "filled" if composed == 1 else "outline"
