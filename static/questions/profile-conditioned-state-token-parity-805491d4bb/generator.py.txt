"""Sequential toggle-register accumulator (state tracking).

The image shows a single binary token carried through K=4 ordered panels (left
to right). Each panel draws the token's current state: a FILLED black disk is
state 1, a HOLLOW ring is state 0. A red marker ring around the leftmost token
marks the start of the tracked chain. Between each consecutive panel sits an
operator glyph: a RED circle means "flip the state", a BLUE square means "keep
the state".

The latent scene stores only ``states`` (each panel's binary token state,
already consistent with the operators) and ``ops`` (the operators drawn between
the panels). The decision is the final state, which is a cumulative function of
the whole sequence: final = initial XOR parity(flips).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 200
PANEL_W = 140
PANEL_H = 150
GAP = 60
MARGIN_LEFT = 30
PITCH = PANEL_W + GAP
TOKEN_R = 26
GLYPH_R = 14
HALF_H = 75
CENTER_Y = HEIGHT // 2

BG = (255, 255, 255)
BORDER = (200, 200, 200)
INK = (0, 0, 0)
RING_MARK = (240, 150, 20)
FLIP_COLOR = (210, 40, 40)
SAME_COLOR = (40, 60, 210)

STATE_LABELS = {0: "outline", 1: "filled"}
PANEL_COUNT = 4


def _panel_center(index: int) -> tuple[int, int]:
    return (MARGIN_LEFT + PITCH * index + PANEL_W // 2, CENTER_Y)


def _glyph_center(index: int) -> tuple[int, int]:
    left = MARGIN_LEFT + PITCH * index + PANEL_W
    return (left + GAP // 2, CENTER_Y)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    initial = rng.randint(0, 1)
    ops = [rng.choice(["flip", "same"]) for _ in range(PANEL_COUNT - 1)]
    states = [initial]
    state = initial
    for op in ops:
        if op == "flip":
            state = 1 - state
        states.append(state)
    return {"states": states, "ops": ops}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    states = scene["states"]
    ops = scene["ops"]

    for i in range(PANEL_COUNT):
        left = MARGIN_LEFT + PITCH * i
        top = CENTER_Y - HALF_H
        draw.rectangle([left, top, left + PANEL_W, top + PANEL_H], outline=BORDER, width=3)
        cx, cy = _panel_center(i)
        box = [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R]
        if states[i] == 1:
            draw.ellipse(box, fill=INK)
        else:
            draw.ellipse(box, outline=INK, width=8)

    cx, cy = _panel_center(0)
    draw.ellipse(
        [cx - TOKEN_R - 7, cy - TOKEN_R - 7, cx + TOKEN_R + 7, cy + TOKEN_R + 7],
        outline=RING_MARK,
        width=4,
    )

    for i, op in enumerate(ops):
        gx, gy = _glyph_center(i)
        if op == "flip":
            draw.ellipse(
                [gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=FLIP_COLOR
            )
        else:
            gx, gy = _glyph_center(i)
            draw.rectangle(
                [gx - GLYPH_R, gy - GLYPH_R, gx + GLYPH_R, gy + GLYPH_R], fill=SAME_COLOR
            )
    return img


def analytic_gold(scene: dict) -> str:
    final = scene["states"][-1]
    return STATE_LABELS[final]


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: every latent field
    (``states``, ``ops``) directly changes the pixels -- ``states`` control the
    drawn token and ``ops`` control the drawn operator glyphs. Any other
    permutation of slots would reorder the panels, changing the image."""
    return []
