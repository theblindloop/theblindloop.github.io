"""Renderer and analytic state-transition arm for the two-lamp gate strip."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 430
START_RECT = (24, 110, 210, 320)
GATE_RECTS = tuple((245 + 135 * i, 110, 345 + 135 * i, 320) for i in range(8))
START_CENTERS = ((77, 220), (157, 220))
GATE_CENTERS = tuple((x0 + 35, 220, x0 + 78, 220) for x0, _, _, _ in GATE_RECTS)

BACKGROUND = (244, 247, 250)
PANEL = (255, 255, 255)
FRAME = (116, 130, 145)
INK = (34, 44, 58)
LIT = (34, 44, 58)
DARK = (225, 233, 240)
FLIP_LEFT = (215, 73, 65)
FLIP_RIGHT = (48, 111, 201)
EXCHANGE = (122, 71, 171)
START = (45, 157, 104)
CONNECTOR = (157, 169, 181)

STATE_LABELS = {
    (False, False): "both-dark",
    (True, False): "left-lit",
    (False, True): "right-lit",
    (True, True): "both-lit",
}


def _apply(state: tuple[bool, bool], op: int) -> tuple[bool, bool]:
    left, right = state
    if op == 0:
        return (not left, right)
    if op == 1:
        return (left, not right)
    if op == 2:
        return (right, left)
    raise ValueError(f"bad gate {op}")


def _two_steps(start: tuple[bool, bool], target: tuple[bool, bool]) -> list[int]:
    for first in range(3):
        for second in range(3):
            if _apply(_apply(start, first), second) == target:
                return [first, second]
    raise AssertionError("two visible operators should reach every binary state")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    initial = (bool(rng.randrange(2)), bool(rng.randrange(2)))
    gates = [rng.randrange(3) for _ in range(6)]
    state = initial
    for op in gates:
        state = _apply(state, op)
    targets = ((False, False), (True, False), (False, True), (True, True))
    gates.extend(_two_steps(state, targets[int(seed) % 4]))
    return {"initial": [int(initial[0]), int(initial[1])], "gates": gates}


def _arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], angle: float, color: tuple[int, int, int], size: float = 10.0) -> None:
    ux, uy = math.cos(angle), math.sin(angle)
    px, py = -uy, ux
    back = (tip[0] - ux * size * 1.45, tip[1] - uy * size * 1.45)
    draw.polygon(
        (tip, (back[0] + px * size, back[1] + py * size),
         (back[0] - px * size, back[1] - py * size)),
        fill=color,
    )


def _lamp_pair(draw: ImageDraw.ImageDraw, left: tuple[float, float], right: tuple[float, float]) -> None:
    for cx, cy in (left, right):
        draw.ellipse((cx - 14, cy - 14, cx + 14, cy + 14), fill=DARK, outline=INK, width=2)


def _circular_operator(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int]) -> None:
    cx, cy = center
    box = (cx - 27, cy - 27, cx + 27, cy + 27)
    draw.arc(box, start=35, end=320, fill=color, width=7)
    theta = math.radians(320)
    tip = (cx + 27 * math.cos(theta), cy + 27 * math.sin(theta))
    _arrowhead(draw, tip, theta + math.pi / 2, color, size=8)


def _exchange_operator(draw: ImageDraw.ImageDraw, left: tuple[float, float], right: tuple[float, float]) -> None:
    y = left[1]
    draw.line((left[0] + 18, y - 9, right[0] - 18, y - 9), fill=EXCHANGE, width=6)
    draw.line((right[0] - 18, y + 9, left[0] + 18, y + 9), fill=EXCHANGE, width=6)
    _arrowhead(draw, (right[0] - 18, y - 9), 0.0, EXCHANGE, size=7)
    _arrowhead(draw, (left[0] + 18, y + 9), math.pi, EXCHANGE, size=7)


def _draw_state(draw: ImageDraw.ImageDraw, centers: tuple[tuple[float, float], tuple[float, float]], state: list[int] | tuple[bool, bool]) -> None:
    for (cx, cy), on in zip(centers, state):
        fill = LIT if bool(on) else DARK
        draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), fill=fill, outline=INK, width=3)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.rectangle((25, 48, 1375, 52), fill=(210, 218, 226))
    draw.rounded_rectangle(START_RECT, radius=16, fill=PANEL, outline=FRAME, width=3)
    draw.rectangle((45, 132, 67, 154), fill=START)
    draw.text((76, 132), "START", fill=INK, font=font)
    _draw_state(draw, START_CENTERS, scene["initial"])
    draw.text((56, 282), "two lamps", fill=FRAME, font=font)

    for index, rect in enumerate(GATE_RECTS):
        x0, y0, x1, y1 = rect
        draw.rounded_rectangle(rect, radius=16, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 12, y0 + 16), str(index + 1), fill=FRAME, font=font)
        left, right = GATE_CENTERS[index][:2], GATE_CENTERS[index][2:]
        _lamp_pair(draw, left, right)
        op = int(scene["gates"][index])
        if op == 0:
            _circular_operator(draw, left, FLIP_LEFT)
        elif op == 1:
            _circular_operator(draw, right, FLIP_RIGHT)
        elif op == 2:
            _exchange_operator(draw, left, right)
        else:
            raise ValueError(f"bad gate {op}")
        if index < len(GATE_RECTS) - 1:
            next_x = GATE_RECTS[index + 1][0]
            y = 220
            draw.line((x1 + 6, y, next_x - 6, y), fill=CONNECTOR, width=4)
            _arrowhead(draw, (next_x - 6, y), 0.0, CONNECTOR, size=7)
    last_x = GATE_RECTS[-1][2]
    draw.line((last_x + 8, 220, 1370, 220), fill=CONNECTOR, width=4)
    _arrowhead(draw, (1370, 220), 0.0, CONNECTOR, size=7)
    draw.rectangle((1368, 175, 1374, 265), fill=FRAME)
    return image


def _final_state(scene: dict) -> tuple[bool, bool]:
    state = (bool(scene["initial"][0]), bool(scene["initial"][1]))
    for op in scene["gates"]:
        state = _apply(state, int(op))
    return state


def analytic_gold(scene: dict) -> str:
    return STATE_LABELS[_final_state(scene)]


def margin(scene: dict) -> float:
    # The sampled strip has no analogical threshold; 24 px is the minimum
    # deliberate separation of a gate glyph from its card's neighboring marks.
    return 24.0


def is_quarantined(scene: dict) -> bool:
    return len(scene.get("gates", [])) != 8 or len(scene.get("initial", [])) != 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
