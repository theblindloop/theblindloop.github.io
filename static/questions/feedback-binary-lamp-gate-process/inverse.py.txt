"""Independent pixel-only inverse for the ordered two-lamp gate strip."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 430
START_CENTERS = ((77, 220), (157, 220))
GATE_X = tuple(245 + 135 * i for i in range(8))
GATE_Y0, GATE_Y1 = 110, 320
LIT = np.array([34, 44, 58], dtype=np.uint8)
DARK = np.array([225, 233, 240], dtype=np.uint8)
FLIP_LEFT = np.array([215, 73, 65], dtype=np.uint8)
FLIP_RIGHT = np.array([48, 111, 201], dtype=np.uint8)
EXCHANGE = np.array([122, 71, 171], dtype=np.uint8)


def _mask(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(arr == color, axis=2)


def _apply(state: tuple[bool, bool], op: int) -> tuple[bool, bool]:
    left, right = state
    if op == 0:
        return (not left, right)
    if op == 1:
        return (left, not right)
    if op == 2:
        return (right, left)
    raise ValueError("unknown visible operator")


def _center_rgb(arr: np.ndarray, x: int, y: int) -> np.ndarray:
    return arr[y, x]


def decision_from_image(image: Image.Image) -> str:
    """Read START lamp fills and each exact-color operator glyph, then execute."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb)
    state = []
    for x, y in START_CENTERS:
        pixel = _center_rgb(arr, x, y)
        if np.array_equal(pixel, LIT):
            state.append(True)
        elif np.array_equal(pixel, DARK):
            state.append(False)
        else:
            raise ValueError("START lamp fill is unreadable")

    masks = (_mask(arr, FLIP_LEFT), _mask(arr, FLIP_RIGHT), _mask(arr, EXCHANGE))
    ops = []
    for x0 in GATE_X:
        crop = (slice(GATE_Y0, GATE_Y1 + 1), slice(x0, x0 + 101))
        counts = [int(mask[crop].sum()) for mask in masks]
        op = int(np.argmax(counts))
        if counts[op] < 20 or sum(value >= 20 for value in counts) != 1:
            raise ValueError("gate glyph is unreadable")
        ops.append(op)

    current = (bool(state[0]), bool(state[1]))
    for op in ops:
        current = _apply(current, op)
    labels = {
        (False, False): "both-dark",
        (True, False): "left-lit",
        (False, True): "right-lit",
        (True, True): "both-lit",
    }
    return labels[current]
