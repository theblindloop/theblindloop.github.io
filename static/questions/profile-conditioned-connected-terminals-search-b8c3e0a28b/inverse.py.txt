"""Independent image-only inverse arm for connected-terminal search.

This module deliberately shares no source, constants, geometry, scene state, or
gold code with the image producer.  It discovers visible panel borders, locates
red and blue pixels, and flood-fills the union of visible graph ink.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _near(rgb: np.ndarray, target: tuple[int, int, int], radius: int) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(target, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _border_boxes(rgb: np.ndarray) -> list[tuple[int, int, int, int]]:
    border = _near(rgb, (74, 104, 128), 34)
    height, width = border.shape
    seen = np.zeros_like(border, dtype=bool)
    boxes: list[tuple[int, int, int, int]] = []
    for sy, sx in np.argwhere(border):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        pending = [(sy, sx)]
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while pending:
            y, x = pending.pop()
            xs.append(x)
            ys.append(y)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and border[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        pending.append((ny, nx))
        x0, x1, y0, y1 = min(xs), max(xs), min(ys), max(ys)
        if len(xs) > 1000 and x1 - x0 > 250 and y1 - y0 > 180:
            boxes.append((x0, y0, x1, y1))
    if len(boxes) != 4:
        raise ValueError(f"expected four visible panel borders, found {len(boxes)}")
    boxes.sort(key=lambda box: (box[1] // 100, box[0]))
    return boxes


def _terminals_connected(crop: np.ndarray) -> bool:
    red = _near(crop, (220, 62, 66), 62)
    blue = _near(crop, (42, 126, 203), 62)
    if int(red.sum()) < 80 or int(blue.sum()) < 80:
        raise ValueError("a panel does not contain one clearly visible red and blue terminal")
    # The graph palette is deliberately dark. This threshold includes colored
    # terminal interiors and anti-aliased contact pixels but excludes the white panel.
    graph = np.min(crop, axis=2) < 205
    visited = np.zeros_like(graph, dtype=bool)
    pending: deque[tuple[int, int]] = deque()
    for y, x in np.argwhere(red):
        y, x = int(y), int(x)
        visited[y, x] = True
        pending.append((y, x))
    height, width = graph.shape
    while pending:
        y, x = pending.popleft()
        if blue[y, x]:
            return True
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dx == 0 and dy == 0:
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and graph[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    pending.append((ny, nx))
    return False


def decision_from_image(image: Image.Image) -> str:
    """Return A-D using only raster evidence, or raise on an ambiguous raster."""
    rgb = np.asarray(image.convert("RGB"))
    boxes = _border_boxes(rgb)
    winners: list[int] = []
    for index, (x0, y0, x1, y1) in enumerate(boxes):
        # Inset past the discovered border and the visible label line.
        crop = rgb[y0 + 34 : y1 - 8, x0 + 8 : x1 - 8]
        if _terminals_connected(crop):
            winners.append(index)
    if len(winners) != 1:
        raise ValueError(f"connectivity evidence is ambiguous: winners={winners}")
    return ("A", "B", "C", "D")[winners[0]]
