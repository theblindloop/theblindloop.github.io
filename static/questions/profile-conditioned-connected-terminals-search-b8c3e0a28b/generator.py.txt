"""Deterministic latent sampler and raster renderer for graph connectivity search.

Every panel contains the same two disconnected three-node paths.  All four
red/blue terminal pairs have the same separation, displayed orientation, node
degree, palette, graph size, edge count, component sizes, and nominal ink.  In
exactly one panel the terminals belong to the same path.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 600
SCALE = 2
LABELS = ("A", "B", "C", "D")
PANEL_BOUNDS = ((35, 55, 365, 285), (403, 55, 733, 285),
                (35, 335, 365, 565), (403, 335, 733, 565))

BACKGROUND = (244, 247, 250)
PANEL_FILL = (255, 255, 255)
BORDER = (74, 104, 128)
DARK = (43, 52, 61)
RED = (220, 62, 66)
BLUE = (42, 126, 203)
MUTED = (103, 116, 128)

EDGES = ((0, 2), (2, 1), (3, 5), (5, 4))
CONNECTED_PAIRS = ((0, 1), (3, 4))
DISCONNECTED_PAIRS = ((0, 3), (1, 4))
QUARANTINE_FLOOR_PX = 6.0


def _rotated(x: float, y: float, quarter_turns: int, mirror: bool) -> tuple[float, float]:
    if mirror:
        x = -x
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return x, y


def _node_offsets(panel: dict) -> list[tuple[float, float]]:
    spacing = float(panel["spacing"])
    shoulder = float(panel["shoulder"])
    raw = [
        (-spacing / 2, -spacing / 2),
        (spacing / 2, -spacing / 2),
        (0.0, -spacing / 2 - shoulder),
        (-spacing / 2, spacing / 2),
        (spacing / 2, spacing / 2),
        (0.0, spacing / 2 + shoulder),
    ]
    return [_rotated(x, y, int(panel["rotation_quarters"]), bool(panel["mirror"])) for x, y in raw]


def sample_scene(seed: int) -> dict:
    """Sample one balanced four-panel scene from an integer seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    # These yield visible inter-component clearances of 8, 14, 22, or 34 px.
    clearance = (8, 14, 22, 34)[(int(seed) // 4) % 4]
    node_radius = 12
    spacing = 2 * node_radius + clearance
    shoulder = 31
    desired_axis = rng.randrange(4)
    translations = [(-5, -2), (5, -2), (-5, 3), (5, 3)]
    rng.shuffle(translations)
    wrong_pairs = [DISCONNECTED_PAIRS[i % 2] for i in range(3)]
    rng.shuffle(wrong_pairs)
    wrong_index = 0
    panels = []
    for index in range(4):
        if index == winner:
            pair = list(rng.choice(CONNECTED_PAIRS))
            rotation = desired_axis
        else:
            pair = list(wrong_pairs[wrong_index])
            wrong_index += 1
            # Cross-component pairs are vertical in canonical coordinates;
            # rotate once more so every displayed terminal pair has one axis.
            rotation = (desired_axis + 1) % 4
        if rng.randrange(2):
            pair.reverse()
        panels.append(
            {
                "edges": [list(edge) for edge in EDGES],
                "red_node": pair[0],
                "blue_node": pair[1],
                "spacing": spacing,
                "shoulder": shoulder,
                "node_radius": node_radius,
                "rotation_quarters": rotation,
                "mirror": bool(rng.randrange(2)),
                "translation": list(translations[index]),
            }
        )
    return {"panels": panels}


def analytic_gold(scene: dict) -> str:
    """Return the unique panel whose colored terminals share a graph component."""
    winners: list[int] = []
    for index, panel in enumerate(scene["panels"]):
        adjacency = {node: set() for node in range(6)}
        for left, right in panel["edges"]:
            adjacency[int(left)].add(int(right))
            adjacency[int(right)].add(int(left))
        pending = [int(panel["red_node"])]
        reached: set[int] = set()
        while pending:
            node = pending.pop()
            if node in reached:
                continue
            reached.add(node)
            pending.extend(adjacency[node] - reached)
        if int(panel["blue_node"]) in reached:
            winners.append(index)
    if len(winners) != 1:
        raise ValueError(f"expected one connected terminal panel, found {winners}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    """Runner-up separation: narrowest visible gap between the two components."""
    values = []
    for panel in scene["panels"]:
        values.append(float(panel["spacing"]) - 2.0 * float(panel["node_radius"]))
    return min(values)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_FLOOR_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The full rendered panel specification is injective over the sampled domain.
    return []


def _font(size: int) -> ImageFont.ImageFont:
    return ImageFont.load_default(size=size)


def _scaled_box(box: Iterable[float]) -> tuple[int, ...]:
    return tuple(int(round(value * SCALE)) for value in box)


def _draw_panel(draw: ImageDraw.ImageDraw, bounds: tuple[int, int, int, int], label: str, panel: dict) -> None:
    x0, y0, x1, y1 = bounds
    draw.rounded_rectangle(_scaled_box((x0 + 3, y0 + 5, x1 + 3, y1 + 5)), radius=14 * SCALE,
                           fill=(220, 226, 232))
    draw.rounded_rectangle(_scaled_box(bounds), radius=14 * SCALE, fill=PANEL_FILL,
                           outline=BORDER, width=4 * SCALE)
    draw.text((int((x0 + 14) * SCALE), int((y0 + 9) * SCALE)), label,
              font=_font(24 * SCALE), fill=BORDER)

    tx, ty = panel["translation"]
    cx = (x0 + x1) / 2 + float(tx)
    cy = (y0 + y1) / 2 + 11 + float(ty)
    offsets = _node_offsets(panel)
    points = [(cx + dx, cy + dy) for dx, dy in offsets]
    edge_width = 8 * SCALE
    for left, right in panel["edges"]:
        p0 = (int(round(points[int(left)][0] * SCALE)), int(round(points[int(left)][1] * SCALE)))
        p1 = (int(round(points[int(right)][0] * SCALE)), int(round(points[int(right)][1] * SCALE)))
        draw.line((p0, p1), fill=DARK, width=edge_width)

    radius = float(panel["node_radius"])
    red_node = int(panel["red_node"])
    blue_node = int(panel["blue_node"])
    for node, (px, py) in enumerate(points):
        outer = _scaled_box((px - radius, py - radius, px + radius, py + radius))
        draw.ellipse(outer, fill=DARK)
        if node == red_node or node == blue_node:
            inner_r = radius - 3.0
            inner = _scaled_box((px - inner_r, py - inner_r, px + inner_r, py + inner_r))
            draw.ellipse(inner, fill=RED if node == red_node else BLUE)


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    if len(panels) != 4:
        raise ValueError("scene must contain four panels")
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((35 * SCALE, 14 * SCALE), "TRACE THE CONNECTION", font=_font(24 * SCALE), fill=DARK)
    draw.text((321 * SCALE, 19 * SCALE), "same node count  •  same edge count",
              font=_font(14 * SCALE), fill=MUTED)
    for bounds, label, panel in zip(PANEL_BOUNDS, LABELS, panels):
        _draw_panel(draw, bounds, label, panel)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
