"""Deterministic renderer for turn-majority measurement on a compact route."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 768
PALETTES = [
    ((19, 58, 112), (150, 216, 228), (225, 83, 68), (48, 101, 196), (250, 248, 242)),
    ((22, 64, 128), (154, 222, 215), (214, 91, 65), (57, 116, 205), (248, 250, 247)),
    ((31, 68, 122), (166, 211, 230), (231, 96, 72), (62, 107, 188), (251, 249, 245)),
]
# Screen-coordinate directions in clockwise order: east, south, west, north.
DIRECTIONS = ((1, 0), (0, 1), (-1, 0), (0, -1))


def _route_pixels(points: list[list[int]]) -> set[tuple[int, int]]:
    pixels: set[tuple[int, int]] = set()
    expected = 1
    for index, ((x0, y0), (x1, y1)) in enumerate(zip(points, points[1:])):
        if (x0 == x1) == (y0 == y1):
            raise ValueError("route legs must be nonzero and orthogonal")
        dx = 0 if x0 == x1 else (1 if x1 > x0 else -1)
        dy = 0 if y0 == y1 else (1 if y1 > y0 else -1)
        length = abs(x1 - x0) + abs(y1 - y0)
        expected += length
        start = 0 if index == 0 else 1
        for step in range(start, length + 1):
            pixels.add((x0 + dx * step, y0 + dy * step))
    if len(pixels) != expected:
        raise ValueError("route self-intersects")
    for x, y in pixels:
        degree = sum((x + dx, y + dy) in pixels for dx, dy in DIRECTIONS)
        if degree not in (1, 2):
            raise ValueError("route touches itself or branches")
    if sum(sum((x + dx, y + dy) in pixels for dx, dy in DIRECTIONS) == 1 for x, y in pixels) != 2:
        raise ValueError("route is not a simple path")
    return pixels


def _turns(points: list[list[int]]) -> list[str]:
    if len(points) != 7:
        raise ValueError("route must have six legs and five bends")
    turns: list[str] = []
    for a, b, c in zip(points, points[1:], points[2:]):
        ux, uy = b[0] - a[0], b[1] - a[1]
        vx, vy = c[0] - b[0], c[1] - b[1]
        cross_screen = ux * vy - uy * vx
        if cross_screen == 0:
            raise ValueError("every junction must be a visible right-angle bend")
        # Image y increases downward, so a negative screen cross is a left turn.
        turns.append("left" if cross_screen < 0 else "right")
    return turns


def _point_segment_distance_sq(point: list[int], left: list[int], right: list[int]) -> int:
    px, py = point
    x0, y0 = left
    x1, y1 = right
    if y0 == y1:
        nearest_x = min(max(px, min(x0, x1)), max(x0, x1))
        return (px - nearest_x) ** 2 + (py - y0) ** 2
    nearest_y = min(max(py, min(y0, y1)), max(y0, y1))
    return (px - x0) ** 2 + (py - nearest_y) ** 2


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "left" if int(seed) % 2 == 0 else "right"
    majority_delta = -1 if desired == "left" else 1
    start_direction = rng.randrange(4)
    points: list[list[int]] | None = None
    for _ in range(80):
        deltas = [majority_delta] * 4 + [-majority_delta]
        rng.shuffle(deltas)
        directions = [start_direction]
        for delta in deltas:
            directions.append((directions[-1] + delta) % 4)
        # The three legs parallel to the initial axis contain two movements in
        # one direction and one in the opposite direction. Give the repeated
        # direction two long runs and the opposing direction a shorter run;
        # transverse legs remain short. This creates a thin local ribbon rather
        # than a square spiral, without changing or revealing turn majority.
        axial_indices = [index for index, direction in enumerate(directions) if direction % 2 == start_direction % 2]
        counts = {direction: sum(directions[index] == direction for index in axial_indices) for direction in set(directions[index] for index in axial_indices)}
        repeated = max(counts, key=counts.get)
        long_lengths = [62 + rng.randint(0, 2), 82 + rng.randint(0, 2)]
        rng.shuffle(long_lengths)
        short_lengths = [12 + rng.randint(0, 1), 15 + rng.randint(0, 1), 18 + rng.randint(0, 1)]
        rng.shuffle(short_lengths)
        lengths = []
        for direction in directions:
            if direction % 2 != start_direction % 2:
                lengths.append(short_lengths.pop())
            elif direction == repeated:
                lengths.append(long_lengths.pop())
            else:
                lengths.append(22 + rng.randint(0, 2))
        direction = start_direction
        trial = [[0, 0]]
        for index, length in enumerate(lengths):
            dx, dy = DIRECTIONS[direction]
            trial.append([trial[-1][0] + dx * length, trial[-1][1] + dy * length])
            if index < len(deltas):
                direction = (direction + deltas[index]) % 4
        try:
            _route_pixels(trial)
        except ValueError:
            continue
        # Endpoint disks must identify direction without masking or crowding a
        # nonincident part of the route.
        if min(_point_segment_distance_sq(trial[0], a, b) for a, b in zip(trial[1:], trial[2:])) <= 100:
            continue
        if min(_point_segment_distance_sq(trial[-1], a, b) for a, b in zip(trial[:-2], trial[1:-1])) <= 100:
            continue
        points = trial
        break
    if points is None:
        raise RuntimeError("unable to sample a simple compact route")

    min_x, max_x = min(p[0] for p in points), max(p[0] for p in points)
    min_y, max_y = min(p[1] for p in points), max(p[1] for p in points)
    x_span, y_span = max_x - min_x, max_y - min_y
    band_center = rng.randint(1, 6) * 96 + 48
    if x_span >= y_span:
        tx = rng.randint(70 - min_x, CANVAS - 70 - max_x)
        ty = band_center - round((min_y + max_y) / 2)
    else:
        tx = band_center - round((min_x + max_x) / 2)
        ty = rng.randint(70 - min_y, CANVAS - 70 - max_y)
    points = [[x + tx, y + ty] for x, y in points]

    route, halo, start, finish, background = PALETTES[rng.randrange(len(PALETTES))]
    distractors = []
    box = (min(p[0] for p in points) - 30, min(p[1] for p in points) - 30,
           max(p[0] for p in points) + 30, max(p[1] for p in points) + 30)
    for x in (52, CANVAS - 54):
        for index in range(3):
            y = 128 + index * 250 + rng.randint(-28, 28)
            if not (box[0] <= x <= box[2] and box[1] <= y <= box[3]):
                distractors.append([x, y, rng.choice([10, 14, 18])])
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": list(background),
        "route_color": list(route),
        "halo_color": list(halo),
        "start_color": list(start),
        "finish_color": list(finish),
        "stroke": rng.choice([7, 9]),
        "points": points,
        "distractors": distractors,
    }


def _draw_axis_route(draw: ImageDraw.ImageDraw, points: list[list[int]], color: tuple[int, int, int], width: int) -> None:
    radius = width // 2
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        draw.rectangle((min(x0, x1) - radius, min(y0, y1) - radius,
                        max(x0, x1) + radius, max(y0, y1) + radius), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for x, y, length in scene["distractors"]:
        draw.line((x - length, y, x + length, y), fill=(210, 211, 207), width=2)
    points = [[int(x), int(y)] for x, y in scene["points"]]
    _draw_axis_route(draw, points, tuple(scene["halo_color"]), int(scene["stroke"]))
    for left, right in zip(points, points[1:]):
        draw.line((*left, *right), fill=tuple(scene["route_color"]), width=1)
    for point, color in ((points[0], tuple(scene["start_color"])), (points[-1], tuple(scene["finish_color"]))):
        x, y = point
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=tuple(scene["background"]))
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    turns = _turns(scene["points"])
    return "left" if turns.count("left") > turns.count("right") else "right"


def margin(scene: dict) -> float:
    turns = _turns(scene["points"])
    return float(abs(turns.count("left") - turns.count("right")))


def is_quarantined(scene: dict) -> bool:
    try:
        _route_pixels(scene["points"])
        turns = _turns(scene["points"])
    except ValueError:
        return True
    return len(turns) != 5 or margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
