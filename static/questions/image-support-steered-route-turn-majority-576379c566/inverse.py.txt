"""Independent pixel-only inverse for majority turn direction."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    route = (blue > red + 55) & (blue > green + 28) & (blue < 170) & (red < 75)
    start = (red > 180) & (red > green + 100) & (blue < 110)
    finish = (blue > 170) & (blue > red + 80) & (blue > green + 45)
    return route, start, finish


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if not (45 <= len(xs) <= 90):
        raise ValueError(f"missing or malformed {name} marker")
    return float(xs.mean()), float(ys.mean())


def _component(mask: np.ndarray) -> set[tuple[int, int]]:
    pixels = {(int(x), int(y)) for y, x in np.argwhere(mask)}
    if not pixels:
        raise ValueError("measurement centerline is absent")
    start = next(iter(pixels))
    seen = {start}
    queue = deque([start])
    while queue:
        x, y = queue.popleft()
        for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if point in pixels and point not in seen:
                seen.add(point)
                queue.append(point)
    if seen != pixels or not (170 <= len(seen) <= 260):
        raise ValueError("centerline is split, corrupted, or implausible")
    return seen


def _ordered_path(pixels: set[tuple[int, int]], start_xy: tuple[float, float], finish_xy: tuple[float, float]) -> list[tuple[int, int]]:
    def neighbors(point: tuple[int, int]) -> list[tuple[int, int]]:
        x, y = point
        return [q for q in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)) if q in pixels]
    endpoints = [point for point in pixels if len(neighbors(point)) == 1]
    if len(endpoints) != 2 or any(len(neighbors(point)) not in (1, 2) for point in pixels):
        raise ValueError("centerline is not one simple unbranched path")
    distance_to_start = [((x - start_xy[0]) ** 2 + (y - start_xy[1]) ** 2, (x, y)) for x, y in endpoints]
    distance_to_start.sort()
    if distance_to_start[0][0] > 150 or distance_to_start[1][0] - distance_to_start[0][0] < 100:
        raise ValueError("start marker does not identify one route endpoint")
    ordered = [distance_to_start[0][1]]
    previous = None
    current = ordered[0]
    while True:
        options = [point for point in neighbors(current) if point != previous]
        if not options:
            break
        if len(options) != 1:
            raise ValueError("centerline traversal is ambiguous")
        previous, current = current, options[0]
        ordered.append(current)
        if len(ordered) > len(pixels):
            raise ValueError("centerline contains a cycle")
    if len(ordered) != len(pixels):
        raise ValueError("centerline traversal missed pixels")
    ex, ey = ordered[-1]
    if (ex - finish_xy[0]) ** 2 + (ey - finish_xy[1]) ** 2 > 150:
        raise ValueError("finish marker does not meet the other endpoint")
    return ordered


def _directions(ordered: list[tuple[int, int]]) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    last = None
    run_length = 0
    for (x0, y0), (x1, y1) in zip(ordered, ordered[1:]):
        direction = (x1 - x0, y1 - y0)
        if direction == last:
            run_length += 1
        else:
            if last is not None:
                if run_length < 6:
                    raise ValueError("route leg is too short")
                runs.append(last)
            last = direction
            run_length = 1
    if last is not None:
        if run_length < 6:
            raise ValueError("route leg is too short")
        runs.append(last)
    if len(runs) != 6:
        raise ValueError("expected exactly six visible straight legs")
    return runs


def decision_from_image(image: Image.Image) -> str:
    route_mask, start_mask, finish_mask = _masks(image)
    try:
        start = _centroid(start_mask, "start")
        finish = _centroid(finish_mask, "finish")
        ordered = _ordered_path(_component(route_mask), start, finish)
        directions = _directions(ordered)
    except ValueError:
        return "abstain"
    turns = []
    for (ux, uy), (vx, vy) in zip(directions, directions[1:]):
        cross_screen = ux * vy - uy * vx
        if cross_screen == 0:
            return "abstain"
        turns.append("left" if cross_screen < 0 else "right")
    difference = turns.count("left") - turns.count("right")
    if abs(difference) < 3:
        return "abstain"
    return "left" if difference > 0 else "right"
