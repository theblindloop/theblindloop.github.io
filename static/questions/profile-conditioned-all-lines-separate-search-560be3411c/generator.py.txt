"""Deterministic latent sampler and Pillow renderer for all-lines separation search."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
PANEL_ORIGINS = ((34, 34), (374, 34), (34, 374), (374, 374))
PANEL_SIZE = (312, 312)
SUPER_SAMPLE = 3
THEMES = (
    {
        "background_rgb": [242, 243, 238],
        "card_rgb": [255, 255, 253],
        "frame_rgb": [188, 193, 184],
        "badge_rgb": [226, 139, 57],
        "line_rgb": [49, 57, 66],
        "red_rgb": [220, 67, 64],
        "blue_rgb": [43, 111, 190],
    },
    {
        "background_rgb": [244, 240, 244],
        "card_rgb": [255, 254, 255],
        "frame_rgb": [198, 187, 199],
        "badge_rgb": [211, 126, 55],
        "line_rgb": [58, 51, 68],
        "red_rgb": [202, 61, 72],
        "blue_rgb": [39, 105, 181],
    },
    {
        "background_rgb": [238, 244, 246],
        "card_rgb": [254, 255, 255],
        "frame_rgb": [181, 194, 198],
        "badge_rgb": [219, 132, 48],
        "line_rgb": [44, 61, 64],
        "red_rgb": [214, 70, 58],
        "blue_rgb": [47, 104, 184],
    },
)


def _point(center: tuple[float, float], tangent: tuple[float, float], normal: tuple[float, float],
           along: float, across: float) -> list[float]:
    return [
        round(center[0] + along * tangent[0] + across * normal[0], 4),
        round(center[1] + along * tangent[1] + across * normal[1], 4),
    ]


def _make_panel(rng: random.Random, target: bool) -> dict[str, Any]:
    angle = math.radians(rng.choice((-28, -21, -14, -7, 0, 7, 14, 21, 28)))
    tangent = (math.cos(angle), math.sin(angle))
    normal = (-math.sin(angle), math.cos(angle))
    center = (156.0 + rng.randint(-3, 3), 156.0 + rng.randint(-3, 3))
    cluster = float(rng.randint(-8, 8))
    line_half = 106.0
    lines: list[list[list[float]]] = []
    for relative in (-48.0, 0.0, 48.0):
        line_center = _point(center, tangent, normal, 0.0, cluster + relative)
        lines.append([
            _point((line_center[0], line_center[1]), tangent, normal, -line_half, 0.0),
            _point((line_center[0], line_center[1]), tangent, normal, line_half, 0.0),
        ])

    # Both classes have the same 144 px dot separation and identical inventory.
    # The winner's projected midpoint is centered among all three lines. Each
    # distractor shifts the pair so exactly one outer line no longer separates it.
    midpoint = float(rng.randint(-4, 4)) if target else float(rng.choice((-47, -46, 46, 47)))
    along = float(rng.randint(-8, 8))
    low = _point(center, tangent, normal, along, cluster + midpoint - 72.0)
    high = _point(center, tangent, normal, along, cluster + midpoint + 72.0)
    red, blue = (low, high) if rng.randrange(2) == 0 else (high, low)
    rng.shuffle(lines)
    return {"red": red, "blue": blue, "lines": lines}


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle the answer A/B/C/D."""
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = [_make_panel(rng, index == answer_index) for index in range(4)]
    theme = THEMES[(int(seed) // 4) % len(THEMES)]
    return {
        "width": 720,
        "height": 720,
        **copy.deepcopy(theme),
        "dot_radius": 13,
        "line_width": 7,
        "panels": panels,
    }


def _signed_distance(point: list[float], line: list[list[float]]) -> float:
    (x0, y0), (x1, y1) = line
    dx, dy = x1 - x0, y1 - y0
    length = math.hypot(dx, dy)
    if length <= 0:
        raise ValueError("zero-length line")
    return (dx * (point[1] - y0) - dy * (point[0] - x0)) / length


def _panel_scores(scene: dict) -> list[int]:
    scores: list[int] = []
    for panel in scene["panels"]:
        score = sum(
            _signed_distance(panel["red"], line) * _signed_distance(panel["blue"], line) < 0.0
            for line in panel["lines"]
        )
        scores.append(int(score))
    return scores


def _minimum_clearance(scene: dict) -> float:
    return min(
        abs(_signed_distance(panel[color], line))
        for panel in scene["panels"]
        for color in ("red", "blue")
        for line in panel["lines"]
    )


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    winners = [index for index, score in enumerate(scores) if score == 3]
    if len(winners) != 1:
        raise ValueError(f"expected one score-three panel, got {winners} from {scores}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    ordered = sorted(_panel_scores(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    return sorted(scores) != [2, 2, 2, 3] or _minimum_clearance(scene) < 18.0 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The direct scene encoding retains no pixel-identical aliases."""
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    ss = SUPER_SAMPLE
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * ss, height * ss), tuple(scene["background_rgb"]))
    draw = ImageDraw.Draw(image)
    font = _font(25 * ss)
    radius = int(scene["dot_radius"]) * ss
    line_width = int(scene["line_width"]) * ss
    for index, panel in enumerate(scene["panels"]):
        ox, oy = PANEL_ORIGINS[index]
        pw, ph = PANEL_SIZE
        draw.rounded_rectangle(
            (ox * ss, oy * ss, (ox + pw) * ss, (oy + ph) * ss),
            radius=15 * ss,
            fill=tuple(scene["card_rgb"]), outline=tuple(scene["frame_rgb"]), width=3 * ss,
        )
        badge_x, badge_y = ox + 28, oy + 28
        draw.ellipse(
            ((badge_x - 19) * ss, (badge_y - 19) * ss,
             (badge_x + 19) * ss, (badge_y + 19) * ss),
            fill=tuple(scene["badge_rgb"]),
        )
        draw.text((badge_x * ss, (badge_y - 1) * ss), LABELS[index], font=font,
                  fill=(255, 255, 255), anchor="mm")
        for line in panel["lines"]:
            points = [((ox + point[0]) * ss, (oy + point[1]) * ss) for point in line]
            draw.line(points, fill=tuple(scene["line_rgb"]), width=line_width)
            cap = line_width / 2.0
            for x, y in points:
                draw.ellipse((x - cap, y - cap, x + cap, y + cap), fill=tuple(scene["line_rgb"]))
        for key, color in (("red", scene["red_rgb"]), ("blue", scene["blue_rgb"])):
            x, y = panel[key]
            cx, cy = (ox + x) * ss, (oy + y) * ss
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(color))
    return image.resize((width, height), Image.Resampling.LANCZOS)
