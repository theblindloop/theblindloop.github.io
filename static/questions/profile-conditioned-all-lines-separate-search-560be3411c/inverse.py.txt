"""Independent raster-only inverse arm for repeated line separation.

The arm discovers all colored dots and dark elongated components, groups them
into the four visible cards from large pixel-space gaps, fits each line using
principal-component analysis, and evaluates both dot centers against every fit.
It receives no latent scene and shares no candidate-side geometry code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


LABELS = ("A", "B", "C", "D")


class OracleAbstention(ValueError):
    """Raised when the raster does not support a unique, high-clearance answer."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    result: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if not work[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        work[sy, sx] = False
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.float64))
    return result


def _dot_centers(arr: np.ndarray, color: str) -> list[np.ndarray]:
    red = arr[:, :, 0].astype(np.int16)
    green = arr[:, :, 1].astype(np.int16)
    blue = arr[:, :, 2].astype(np.int16)
    if color == "red":
        mask = (red > 145) & (red > green + 55) & (red > blue + 45)
    else:
        mask = (blue > 125) & (blue > red + 45) & (blue > green + 35)
    centers: list[np.ndarray] = []
    for points in _components(mask):
        if 380 <= len(points) <= 750:
            y0, x0 = points.min(axis=0)
            y1, x1 = points.max(axis=0)
            if 20 <= x1 - x0 + 1 <= 32 and 20 <= y1 - y0 + 1 <= 32:
                centers.append(np.array([points[:, 1].mean(), points[:, 0].mean()]))
    if len(centers) != 4:
        raise OracleAbstention(f"expected four {color} dots, found {len(centers)}")
    return centers


def _line_fits(arr: np.ndarray) -> list[dict[str, np.ndarray | float]]:
    dark = arr.max(axis=2) < 135
    fits: list[dict[str, np.ndarray | float]] = []
    for points_yx in _components(dark):
        if not 900 <= len(points_yx) <= 2300:
            continue
        points = points_yx[:, ::-1]
        center = points.mean(axis=0)
        centered = points - center
        covariance = centered.T @ centered / len(points)
        values, vectors = np.linalg.eigh(covariance)
        if values[1] < 900.0 or values[0] <= 0.0 or values[1] / values[0] < 180.0:
            continue
        direction = vectors[:, 1]
        normal = np.array([-direction[1], direction[0]])
        projected = centered @ direction
        visible_length = float(projected.max() - projected.min())
        if not 195.0 <= visible_length <= 225.0:
            continue
        fits.append({"center": center, "normal": normal, "length": visible_length})
    if len(fits) != 12:
        raise OracleAbstention(f"expected twelve dark lines, found {len(fits)}")
    return fits


def _largest_gap_cut(values: list[float]) -> float:
    ordered = sorted(values)
    if len(ordered) < 2:
        raise OracleAbstention("cannot divide fewer than two positions")
    gaps = [ordered[index + 1] - ordered[index] for index in range(len(ordered) - 1)]
    index = int(np.argmax(np.asarray(gaps)))
    if gaps[index] < 70.0:
        raise OracleAbstention(f"card separation gap too small: {gaps[index]:.1f}")
    return (ordered[index] + ordered[index + 1]) / 2.0


def _quadrant(point: np.ndarray, x_cut: float, y_cut: float) -> int:
    return (2 if point[1] >= y_cut else 0) + (1 if point[0] >= x_cut else 0)


def analyze_image(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red = _dot_centers(arr, "red")
    blue = _dot_centers(arr, "blue")
    fits = _line_fits(arr)
    # The parallel-line triplets stay centered inside their cards, whereas a
    # card's two dots deliberately span much of its height. Derive the card
    # cuts from line centroids, then assign every independently found mark.
    fit_centers = [fit["center"] for fit in fits]
    x_cut = _largest_gap_cut([float(point[0]) for point in fit_centers])
    y_cut = _largest_gap_cut([float(point[1]) for point in fit_centers])
    groups = [{"red": [], "blue": [], "lines": []} for _ in range(4)]
    for color, points in (("red", red), ("blue", blue)):
        for point in points:
            groups[_quadrant(point, x_cut, y_cut)][color].append(point)
    for fit in fits:
        center = fit["center"]
        assert isinstance(center, np.ndarray)
        groups[_quadrant(center, x_cut, y_cut)]["lines"].append(fit)
    inventory = [(len(group["red"]), len(group["blue"]), len(group["lines"])) for group in groups]
    if inventory != [(1, 1, 3)] * 4:
        raise OracleAbstention(f"invalid per-card inventory {inventory}")

    scores: list[int] = []
    clearances: list[float] = []
    for group in groups:
        red_point = group["red"][0]
        blue_point = group["blue"][0]
        score = 0
        for fit in group["lines"]:
            center, normal = fit["center"], fit["normal"]
            assert isinstance(center, np.ndarray) and isinstance(normal, np.ndarray)
            red_side = float((red_point - center) @ normal)
            blue_side = float((blue_point - center) @ normal)
            clearances.extend((abs(red_side), abs(blue_side)))
            score += int(red_side * blue_side < 0.0)
        scores.append(score)
    winners = [index for index, score in enumerate(scores) if score == 3]
    ordered = sorted(scores, reverse=True)
    minimum_clearance = min(clearances)
    if len(winners) != 1 or ordered != [3, 2, 2, 2]:
        raise OracleAbstention(f"no unique all-lines winner: {scores}")
    if minimum_clearance < 15.0 or ordered[0] - ordered[1] < 1:
        raise OracleAbstention(
            f"weak raster margin: score={scores}, clearance={minimum_clearance:.2f}"
        )
    return {
        "decision": LABELS[winners[0]],
        "scores": scores,
        "runner_up_margin": float(ordered[0] - ordered[1]),
        "minimum_clearance": float(minimum_clearance),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
