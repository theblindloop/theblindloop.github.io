"""Deterministic renderer and analytic arm for route_cut_checkpoint."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw


CANVAS = (960, 460)
SCALE = 3
NODE_RADIUS = 14
COLORS = ("red", "blue", "orange")
PALETTE = {
    "background": (250, 249, 246),
    "ink": (29, 32, 38),
    "generic": (211, 217, 224),
    "green": (38, 174, 91),
    "purple": (132, 79, 188),
    "red": (221, 61, 61),
    "blue": (48, 107, 215),
    "orange": (239, 143, 33),
}


def _node(node_id: str, role: str, fill: str, x: int, y: int) -> dict:
    return {
        "id": node_id,
        "role": role,
        "fill": fill,
        "x": int(x),
        "y": int(y),
        "r": NODE_RADIUS,
    }


def _reflect_x(x: int, left: int, right: int, flip: bool) -> int:
    return left + right - x if flip else x


def sample_scene(seed: int) -> dict:
    """Sample a crossing-free graph with one colored START/FINISH cut vertex."""

    rng = random.Random(int(seed))
    answer_color = COLORS[int(seed) % 3]
    cut_module = (int(seed) // 3) % 3
    remaining = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining)
    module_colors = [None, None, None]
    module_colors[cut_module] = answer_color
    for index, color in zip((i for i in range(3) if i != cut_module), remaining):
        module_colors[index] = color

    baselines = [rng.choice((188, 230, 272)) for _ in range(3)]
    sides = [rng.choice((-1, 1)) for _ in range(3)]
    flips = [bool(rng.getrandbits(1)) for _ in range(3)]
    optional_slots = [bool(rng.getrandbits(1)) for _ in range(3)]

    nodes: list[dict] = []
    edges: list[list[str]] = []
    module_lefts = (70, 360, 650)
    module_rights = (310, 600, 890)

    start = _node("start", "start", "green", 20, baselines[0])
    finish = _node("finish", "finish", "purple", 940, baselines[2])
    nodes.extend((start, finish))

    prior = "start"
    for index, (left, right, base, side, flip) in enumerate(
        zip(module_lefts, module_rights, baselines, sides, flips)
    ):
        entry_id = f"m{index}_entry"
        exit_id = f"m{index}_exit"
        candidate_id = f"m{index}_candidate"
        nodes.append(_node(entry_id, "junction", "generic", left, base))
        nodes.append(_node(exit_id, "junction", "generic", right, base))
        edges.append([prior, entry_id])

        if index == cut_module:
            # Two routes merge before (or split after) the colored degree-two cut node.
            raw = {
                "a": (left + 55, base - 65),
                "b": (left + 55, base + 65),
                "merge": (left + 125, base),
                "candidate": (left + 180, base + side * 34),
            }
            coords = {
                key: (_reflect_x(x, left, right, flip), y) for key, (x, y) in raw.items()
            }
            for key in ("a", "b", "merge"):
                x, y = coords[key]
                nodes.append(_node(f"m{index}_{key}", "junction", "generic", x, y))
            x, y = coords["candidate"]
            nodes.append(_node(candidate_id, "checkpoint", module_colors[index], x, y))
            if not flip:
                edges.extend(
                    [
                        [entry_id, f"m{index}_a"],
                        [f"m{index}_a", f"m{index}_merge"],
                        [entry_id, f"m{index}_b"],
                        [f"m{index}_b", f"m{index}_merge"],
                        [f"m{index}_merge", candidate_id],
                        [candidate_id, exit_id],
                    ]
                )
            else:
                edges.extend(
                    [
                        [entry_id, candidate_id],
                        [candidate_id, f"m{index}_merge"],
                        [f"m{index}_merge", f"m{index}_a"],
                        [f"m{index}_a", exit_id],
                        [f"m{index}_merge", f"m{index}_b"],
                        [f"m{index}_b", exit_id],
                    ]
                )
        else:
            upper = -side * 65
            lower = side * 65
            first_x, second_x = left + 72, left + 168
            candidate_x = second_x if optional_slots[index] else first_x
            partner_x = first_x if optional_slots[index] else second_x
            raw = {
                "candidate": (candidate_x, base + upper),
                "partner": (partner_x, base + upper),
                "bypass_a": (first_x, base + lower),
                "bypass_b": (second_x, base + lower),
            }
            coords = {
                key: (_reflect_x(x, left, right, flip), y) for key, (x, y) in raw.items()
            }
            x, y = coords["candidate"]
            nodes.append(_node(candidate_id, "checkpoint", module_colors[index], x, y))
            for key in ("partner", "bypass_a", "bypass_b"):
                x, y = coords[key]
                nodes.append(_node(f"m{index}_{key}", "junction", "generic", x, y))
            upper_pair = [candidate_id, f"m{index}_partner"]
            upper_pair.sort(key=lambda node_id: next(n["x"] for n in nodes if n["id"] == node_id))
            lower_pair = [f"m{index}_bypass_a", f"m{index}_bypass_b"]
            lower_pair.sort(key=lambda node_id: next(n["x"] for n in nodes if n["id"] == node_id))
            edges.extend(
                [
                    [entry_id, upper_pair[0]],
                    [upper_pair[0], upper_pair[1]],
                    [upper_pair[1], exit_id],
                    [entry_id, lower_pair[0]],
                    [lower_pair[0], lower_pair[1]],
                    [lower_pair[1], exit_id],
                ]
            )
        prior = exit_id

    edges.append([prior, "finish"])
    return {"canvas": list(CANVAS), "nodes": nodes, "edges": edges}


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width * SCALE, height * SCALE), PALETTE["background"])
    draw = ImageDraw.Draw(image)
    by_id = {node["id"]: node for node in scene["nodes"]}
    for left_id, right_id in scene["edges"]:
        left, right = by_id[left_id], by_id[right_id]
        draw.line(
            (left["x"] * SCALE, left["y"] * SCALE, right["x"] * SCALE, right["y"] * SCALE),
            fill=PALETTE["ink"],
            width=7 * SCALE,
        )
    for node in scene["nodes"]:
        x, y, radius = node["x"] * SCALE, node["y"] * SCALE, node["r"] * SCALE
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=PALETTE[node["fill"]],
            outline=PALETTE["ink"],
            width=3 * SCALE,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _connected(nodes: list[dict], edges: list[list[str]], removed: str | None) -> bool:
    ids = {node["id"] for node in nodes if node["id"] != removed}
    adjacency = {node_id: set() for node_id in ids}
    for left, right in edges:
        if left in ids and right in ids:
            adjacency[left].add(right)
            adjacency[right].add(left)
    if "start" not in ids or "finish" not in ids:
        return False
    queue = deque(["start"])
    seen = {"start"}
    while queue:
        current = queue.popleft()
        if current == "finish":
            return True
        for neighbor in adjacency[current] - seen:
            seen.add(neighbor)
            queue.append(neighbor)
    return False


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    checkpoints = [node for node in nodes if node["role"] == "checkpoint"]
    winners = [node["fill"] for node in checkpoints if not _connected(nodes, edges, node["id"])]
    if len(winners) != 1 or winners[0] not in COLORS:
        raise ValueError(f"expected one colored route cut checkpoint, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    nodes = scene["nodes"]
    edges = scene["edges"]
    scores = [
        1.0 if not _connected(nodes, edges, node["id"]) else 0.0
        for node in nodes
        if node["role"] == "checkpoint"
    ]
    scores.sort(reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    if margin({"nodes": nodes, "edges": edges}) < 1.0:
        return True
    for index, left in enumerate(nodes):
        for right in nodes[index + 1 :]:
            clearance = math.hypot(left["x"] - right["x"], left["y"] - right["y"]) - left["r"] - right["r"]
            if clearance < 12:
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
