"""Floor-clamped counter accumulator (state tracking).

The image is a left-to-right program. The leftmost panel shows a vertical level
track with four tick levels 0 (bottom) to 3 (top). A dark marker disk sits on
the tick of the START level, and an orange ring marks it as the tracked start.
To the right sit three operator panels, read left to right. Each holds one
arrow: a GREEN up-arrow means "raise the level by one" (+1) and a RED down-arrow
means "lower the level by one" (-1).

The tracked level is clamped to the [0, 3] interval after every step: it can
never fall below zero nor rise above three. The FINAL level is never drawn
anywhere; it must be recovered by replaying the declared operators from the
marked start.

The latent scene stores only ``start`` (the marked starting level 0..3) and
``ops`` (the ordered list of operator directions). The decision is the final
level after clamped replay.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 840
HEIGHT = 270
CENTER_Y = 135

START_LEFT = 20
START_TOP = 30
START_W = 190
START_H = 210
START_RIGHT = START_LEFT + START_W

TRACK_X = START_LEFT + 70
TRACK_X0 = TRACK_X - 45
TRACK_X1 = TRACK_X + 30
TRACK_BOTTOM = START_TOP + START_H - 30
LEVEL_PX = 50
LEVELS = 4

OP_W = 120
OP_H = 120
OP_TOP = CENTER_Y - OP_H // 2
OP_GAP = 40
OP_FIRST = START_RIGHT + 45
OP_PITCH = OP_W + OP_GAP
UP_R = 34

BG = (255, 255, 255)
BORDER = (200, 200, 200)
INK = (0, 0, 0)
TICK = (95, 95, 95)
RING_MARK = (240, 150, 20)
GREEN = (40, 170, 60)
RED = (200, 45, 45)

OP_COUNT = 3


def _tick_y(level: int) -> int:
    return TRACK_BOTTOM - (level + 1) * LEVEL_PX + LEVEL_PX // 2


def _op_center(index: int) -> tuple[int, int]:
    return (OP_FIRST + index * OP_PITCH + OP_W // 2, OP_TOP + OP_H // 2)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    start = rng.randint(0, LEVELS - 1)
    ops = [rng.choice(["up", "down"]) for _ in range(OP_COUNT)]
    return {"start": start, "ops": ops}


def _replay(start: int, ops: list) -> int:
    level = start
    for op in ops:
        step = 1 if op == "up" else -1
        level = max(0, min(LEVELS - 1, level + step))
    return level


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    draw.rectangle(
        [START_LEFT, START_TOP, START_RIGHT, START_TOP + START_H], outline=BORDER, width=3
    )

    for level in range(LEVELS):
        y = _tick_y(level)
        draw.line([TRACK_X0, y, TRACK_X1, y], fill=TICK, width=3)
        draw.text((START_RIGHT - 30, y - 9), str(level), fill=INK)

    start = scene["start"]
    my = _tick_y(start)
    draw.ellipse([TRACK_X - 11, my - 11, TRACK_X + 11, my + 11], fill=INK)
    draw.ellipse(
        [TRACK_X - 16, my - 16, TRACK_X + 16, my + 16],
        outline=RING_MARK,
        width=4,
    )

    for i, op in enumerate(scene["ops"]):
        cx, cy = _op_center(i)
        draw.rectangle(
            [cx - OP_W // 2, OP_TOP, cx + OP_W // 2, OP_TOP + OP_H], outline=BORDER, width=3
        )
        color = GREEN if op == "up" else RED
        if op == "up":
            draw.polygon(
                [(cx, cy - UP_R), (cx - UP_R, cy + UP_R), (cx + UP_R, cy + UP_R)],
                fill=color,
            )
        else:
            draw.polygon(
                [(cx, cy + UP_R), (cx - UP_R, cy - UP_R), (cx + UP_R, cy - UP_R)],
                fill=color,
            )

    return img


def analytic_gold(scene: dict) -> str:
    return str(_replay(scene["start"], scene["ops"]))


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry.

    ``start`` fixes the drawn marker level and ``ops`` fix the drawn arrow
    directions in their fixed left-to-right order. Reordering the operators or
    changing either field changes the pixels, and any other field would be
    render-irrelevant to the decision.
    """
    return []
