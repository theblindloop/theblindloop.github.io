"""Deterministic renderer and analytic world for the planet/moon binding task."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 420
TARGET_BLUE_SWATCHES = (
    (42, 118, 210),
    (38, 142, 215),
    (58, 105, 190),
)
OTHER_SWATCHES = (
    (222, 96, 68),
    (224, 166, 46),
    (45, 164, 101),
    (194, 75, 126),
)
MOON_SWATCHES = (
    (78, 89, 105),
    (92, 100, 112),
    (72, 87, 96),
)
ORBIT_SWATCHES = (
    (168, 179, 191),
    (183, 190, 201),
    (151, 166, 181),
)
BACKGROUND_SWATCHES = (
    (247, 249, 246),
    (244, 247, 250),
)
CARD_SWATCHES = (
    (255, 255, 255),
    (252, 253, 250),
)
OUTLINE_SWATCHES = (
    (211, 218, 226),
    (207, 216, 221),
)
TITLE_COLOR = (39, 56, 77)
MOON_ANGLES = (25.0, 155.0, 205.0, 335.0)
SUPERSAMPLE = 4


def _as_rgb(value: list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    return int(value[0]), int(value[1]), int(value[2])


def _sample_system(rng: random.Random, x: float, y: float, target_blue: tuple[int, int, int], moon_color: tuple[int, int, int]) -> dict[str, Any]:
    role = "blue" if rng.random() < 0.5 else "other"
    radius = rng.randint(34, 44)
    moon_radius = rng.randint(7, 10)
    clearance = rng.uniform(4.0, 8.0) if rng.random() < 0.13 else rng.uniform(18.0, 28.0)
    orbit_radius = radius + moon_radius + clearance
    orbit_rx = round(orbit_radius, 3)
    orbit_ry = round(orbit_radius * rng.uniform(0.68, 0.76), 3)
    moon_count = rng.randint(0, 2)
    angles = list(MOON_ANGLES)
    rng.shuffle(angles)
    angles = sorted(round(angle, 3) for angle in angles[:moon_count])
    other_fill = rng.choice(OTHER_SWATCHES)
    current_fill = target_blue if role == "blue" else other_fill
    return {
        "planet_role": role,
        "planet_fill": list(current_fill),
        "alternate_fill": list(other_fill if role == "blue" else target_blue),
        "planet_center": [round(x, 3), round(y, 3)],
        "planet_radius": radius,
        "orbit_rx": orbit_rx,
        "orbit_ry": orbit_ry,
        "orbit_width": rng.choice([2, 3]),
        "moon_radius": moon_radius,
        "moon_angles": angles,
        "moon_fill": list(moon_color),
    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one four-card scene with independent style and role nuisance draws."""

    rng = random.Random(int(seed))
    target_blue = rng.choice(TARGET_BLUE_SWATCHES)
    moon_color = rng.choice(MOON_SWATCHES)
    background = rng.choice(BACKGROUND_SWATCHES)
    card_fill = rng.choice(CARD_SWATCHES)
    card_outline = rng.choice(OUTLINE_SWATCHES)
    orbit_color = rng.choice(ORBIT_SWATCHES)
    slot_x = [120.0, 360.0, 600.0, 840.0]
    rng.shuffle(slot_x)
    systems = []
    for base_x in slot_x:
        x = base_x + rng.uniform(-10.0, 10.0)
        y = 228.0 + rng.uniform(-7.0, 7.0)
        systems.append(_sample_system(rng, x, y, target_blue, moon_color))
    # The direct count is not stored as a separate hidden answer field. It is
    # computed from the visible systems list by analytic_gold().
    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "background": list(background),
        "card_fill": list(card_fill),
        "card_outline": list(card_outline),
        "title": "SOLAR ARRAY",
        "title_color": list(TITLE_COLOR),
        "orbit_color": list(orbit_color),
        "moon_color": list(moon_color),
        "systems": systems,
    }


def counterfactual_scene(scene: dict[str, Any]) -> dict[str, Any]:
    """Return a matched visible counterfactual by flipping one one-moon role."""

    twin = copy.deepcopy(scene)
    for system in twin["systems"]:
        if len(system["moon_angles"]) == 1:
            old_role = system["planet_role"]
            system["planet_role"] = "other" if old_role == "blue" else "blue"
            system["planet_fill"], system["alternate_fill"] = (
                system["alternate_fill"],
                system["planet_fill"],
            )
            return twin
    # A no-one-moon scene has no eligible visible relation to flip. It remains
    # a valid matched control, and the generator records it without a claim of
    # changed answer.
    return twin


def render(scene: dict[str, Any]) -> Image.Image:
    ss = SUPERSAMPLE
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width * ss, height * ss), _as_rgb(scene["background"]))
    draw = ImageDraw.Draw(image)

    # The small title is semantic framing only; it is not used by analytic gold
    # or by the pixel inverse.
    font = ImageFont.load_default()
    draw.text((24 * ss, 20 * ss), str(scene["title"]), fill=_as_rgb(scene["title_color"]), font=font)

    for system in scene["systems"]:
        cx, cy = (float(v) for v in system["planet_center"])
        radius = float(system["planet_radius"])
        rx = float(system["orbit_rx"])
        ry = float(system["orbit_ry"])
        moon_radius = float(system["moon_radius"])
        card_left = cx - 104.0
        card_top = cy - 145.0
        card_right = cx + 104.0
        card_bottom = cy + 145.0
        draw.rounded_rectangle(
            [round(card_left * ss), round(card_top * ss), round(card_right * ss), round(card_bottom * ss)],
            radius=18 * ss,
            fill=_as_rgb(scene["card_fill"]),
            outline=_as_rgb(scene["card_outline"]),
            width=2 * ss,
        )
        draw.ellipse(
            [round((cx - rx) * ss), round((cy - ry) * ss), round((cx + rx) * ss), round((cy + ry) * ss)],
            outline=_as_rgb(scene["orbit_color"]),
            width=int(system["orbit_width"]) * ss,
        )
        for angle in system["moon_angles"]:
            theta = math.radians(float(angle))
            mx = cx + rx * math.cos(theta)
            my = cy + ry * math.sin(theta)
            draw.ellipse(
                [
                    round((mx - moon_radius) * ss),
                    round((my - moon_radius) * ss),
                    round((mx + moon_radius) * ss),
                    round((my + moon_radius) * ss),
                ],
                fill=_as_rgb(system["moon_fill"]),
            )
        draw.ellipse(
            [
                round((cx - radius) * ss),
                round((cy - radius) * ss),
                round((cx + radius) * ss),
                round((cy + radius) * ss),
            ],
            fill=_as_rgb(system["planet_fill"]),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute the label from the visible role and moon markers only."""

    return str(
        sum(
            system["planet_role"] == "blue" and len(system["moon_angles"]) == 1
            for system in scene["systems"]
        )
    )


def margin(scene: dict[str, Any]) -> float:
    """Smallest visible clearance used as the finite-resolution confidence margin."""

    values: list[float] = []
    systems = scene["systems"]
    for system in systems:
        cx, cy = (float(v) for v in system["planet_center"])
        radius = float(system["planet_radius"])
        moon_radius = float(system["moon_radius"])
        rx = float(system["orbit_rx"])
        ry = float(system["orbit_ry"])
        values.extend([104.0 - rx - moon_radius, 145.0 - ry - moon_radius])
        positions: list[tuple[float, float]] = []
        for angle in system["moon_angles"]:
            theta = math.radians(float(angle))
            positions.append((cx + rx * math.cos(theta), cy + ry * math.sin(theta)))
        for mx, my in positions:
            values.append(math.hypot(mx - cx, my - cy) - radius - moon_radius)
        for index, first in enumerate(positions):
            for second in positions[index + 1 :]:
                values.append(math.hypot(first[0] - second[0], first[1] - second[1]) - 2.0 * moon_radius)
    for index, first in enumerate(systems):
        fx, fy = (float(v) for v in first["planet_center"])
        fr = float(first["orbit_rx"]) + float(first["moon_radius"])
        for second in systems[index + 1 :]:
            sx, sy = (float(v) for v in second["planet_center"])
            sr = float(second["orbit_rx"]) + float(second["moon_radius"])
            values.append(math.hypot(fx - sx, fy - sy) - fr - sr)
    return round(min(values), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed; all answer-bearing fields are rendered."""

    return []
