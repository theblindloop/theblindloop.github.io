"""Independent pixel-only inverse for the planet/moon binding question."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "planet-moon-binding-oracle-0.1.0"

_MOON_SWATCHES = np.asarray(
    ((78, 89, 105), (92, 100, 112), (72, 87, 96)), dtype=np.float32
)


def _components(mask: np.ndarray, minimum: int) -> list[tuple[np.ndarray, np.ndarray]]:
    """Return (ys, xs) for 8-connected components above an area floor."""

    height, width = mask.shape
    remaining = mask.copy()
    found: list[tuple[np.ndarray, np.ndarray]] = []
    while remaining.any():
        y0, x0 = np.argwhere(remaining)[0]
        queue = deque([(int(y0), int(x0))])
        remaining[y0, x0] = False
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    yy = y + dy
                    xx = x + dx
                    if 0 <= yy < height and 0 <= xx < width and remaining[yy, xx]:
                        remaining[yy, xx] = False
                        queue.append((yy, xx))
        if len(ys) >= minimum:
            found.append((np.asarray(ys, dtype=np.float64), np.asarray(xs, dtype=np.float64)))
    return found


def _centers(components: list[tuple[np.ndarray, np.ndarray]]) -> list[tuple[float, float, float]]:
    centers = []
    for ys, xs in components:
        centers.append((float(xs.mean()) + 0.5, float(ys.mean()) + 0.5, math.sqrt(len(xs) / math.pi)))
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Recover the count using only raster colours, components, and distances."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    red = arr[..., 0]
    green = arr[..., 1]
    blue = arr[..., 2]

    # Interior pixels of the blue planet swatches satisfy this hue predicate;
    # antialiased boundary pixels are unnecessary because the interiors are large.
    blue_planet_mask = (blue > 145.0) & (blue > red * 1.25) & (blue > green * 1.12)
    planet_components = _components(blue_planet_mask, minimum=700)
    planets = _centers(planet_components)

    # Moon interiors are dark, low-saturation pixels near one of the public
    # moon swatches. This excludes the pale orbit strokes and card borders.
    distances = np.stack(
        [np.linalg.norm(arr - swatch, axis=-1) for swatch in _MOON_SWATCHES], axis=-1
    )
    moon_mask = (distances.min(axis=-1) < 38.0) & (arr.mean(axis=-1) > 65.0)
    moon_components = _components(moon_mask, minimum=45)
    moons = [(x, y) for x, y, _ in _centers(moon_components)]

    count = 0
    for px, py, estimated_radius in planets:
        attached = 0
        for mx, my in moons:
            distance = math.hypot(mx - px, my - py)
            if estimated_radius * 1.08 < distance < 112.0:
                attached += 1
        if attached == 1:
            count += 1
    return str(count)
