"""Independent image-only inverse arm for fragmented loop closure."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageFilter

PALETTE = {
    "red": np.array((220, 48, 58), dtype=np.int32),
    "green": np.array((20, 146, 96), dtype=np.int32),
    "blue": np.array((42, 101, 211), dtype=np.int32),
}


def _closed(mask: np.ndarray) -> np.ndarray:
    # Radius 7 square closing bridges seams no wider than the declared 12px stroke.
    pic = Image.fromarray((mask * 255).astype(np.uint8), mode="L")
    pic = pic.filter(ImageFilter.MaxFilter(15)).filter(ImageFilter.MinFilter(15))
    return np.asarray(pic) > 127


def _hole_count(mask: np.ndarray) -> int:
    h, w = mask.shape
    background = ~mask
    seen = np.zeros_like(background, dtype=bool)
    q: deque[tuple[int, int]] = deque()
    for x in range(w):
        if background[0, x]: q.append((0, x))
        if background[h - 1, x]: q.append((h - 1, x))
    for y in range(h):
        if background[y, 0]: q.append((y, 0))
        if background[y, w - 1]: q.append((y, w - 1))
    while q:
        y, x = q.popleft()
        if seen[y, x] or not background[y, x]:
            continue
        seen[y, x] = True
        if y: q.append((y - 1, x))
        if y + 1 < h: q.append((y + 1, x))
        if x: q.append((y, x - 1))
        if x + 1 < w: q.append((y, x + 1))
    unseen = background & ~seen
    holes = 0
    for y, x in zip(*np.nonzero(unseen)):
        if seen[y, x]:
            continue
        holes += 1
        q.append((int(y), int(x)))
        while q:
            yy, xx = q.popleft()
            if seen[yy, xx] or not unseen[yy, xx]:
                continue
            seen[yy, xx] = True
            if yy: q.append((yy - 1, xx))
            if yy + 1 < h: q.append((yy + 1, xx))
            if xx: q.append((yy, xx - 1))
            if xx + 1 < w: q.append((yy, xx + 1))
    return holes


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = np.stack([np.sum((arr - rgb) ** 2, axis=2) for rgb in PALETTE.values()])
    labels = distances.argmin(axis=0)
    nearest = distances.min(axis=0)
    winners = []
    for index, color in enumerate(PALETTE):
        mask = (labels == index) & (nearest < 5000)
        if int(mask.sum()) < 1000:
            continue
        if _hole_count(_closed(mask)) == 1:
            winners.append(color)
    if len(winners) != 1:
        raise ValueError(f"expected one closed-loop color, found {winners}")
    return winners[0]
