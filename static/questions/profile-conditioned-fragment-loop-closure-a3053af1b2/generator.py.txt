"""Deterministic fragmented-contour world rendered with Pillow."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

COLORS = ("red", "green", "blue")
RGB = {"red": (220, 48, 58), "green": (20, 146, 96), "blue": (42, 101, 211)}
BACKGROUND = (248, 247, 242)
BRIDGE_LIMIT = 12.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = COLORS[int(seed) % len(COLORS)]
    positions = [(120, 140), (360, 140), (600, 140)]
    rng.shuffle(positions)
    open_shifts = [22.0 + rng.uniform(0, 5), 28.0 + rng.uniform(0, 5)]
    rng.shuffle(open_shifts)
    contours = []
    oi = 0
    for color, center in zip(COLORS, positions):
        is_target = color == target_color
        contours.append(
            {
                "color": color,
                "center": list(center),
                "radius": round(rng.uniform(67.0, 74.0), 3),
                "rotation": round(rng.uniform(0.0, 30.0), 3),
                "shift_index": rng.randrange(12),
                "shift_degrees": 0.0 if is_target else round(open_shifts[oi], 3),
                "max_seam": 7.8 if is_target else round(26.0 + open_shifts[oi] * 0.35, 3),
            }
        )
        if not is_target:
            oi += 1
    return {
        "contours": contours,
        "target_color": target_color,
        "width": 720,
        "height": 280,
        "background": list(BACKGROUND),
        "stroke": 12,
    }


def _arc_points(cx: float, cy: float, radius: float, a0: float, a1: float) -> list[tuple[float, float]]:
    points = []
    for j in range(13):
        a = math.radians(a0 + (a1 - a0) * j / 12.0)
        points.append((cx + radius * math.cos(a), cy + radius * math.sin(a)))
    return points


def render(scene: dict) -> Image.Image:
    # Explicit access makes the label-bearing field part of the rendered-field audit.
    if scene["target_color"] not in COLORS:
        raise ValueError("invalid target color")
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    bg = tuple(scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), bg)
    draw = ImageDraw.Draw(image)
    stroke = int(scene["stroke"]) * scale
    for contour in scene["contours"]:
        cx, cy = (float(v) * scale for v in contour["center"])
        radius = float(contour["radius"]) * scale
        rotation = float(contour["rotation"])
        shifted = int(contour["shift_index"])
        delta = float(contour["shift_degrees"])
        color = RGB[contour["color"]]
        # Twelve equal-ink 24-degree arcs with nominal 6-degree seams.
        for index in range(12):
            offset = delta if index == shifted else 0.0
            a0 = rotation + index * 30.0 + 3.0 + offset
            a1 = a0 + 24.0
            pts = _arc_points(cx, cy, radius, a0, a1)
            draw.line(pts, fill=color, width=stroke, joint="curve")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["target_color"])


def margin(scene: dict) -> float:
    target = scene["target_color"]
    target_seam = next(float(c["max_seam"]) for c in scene["contours"] if c["color"] == target)
    runner_up = min(float(c["max_seam"]) for c in scene["contours"] if c["color"] != target)
    return round(runner_up - max(BRIDGE_LIMIT, target_seam), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["contours"] = list(reversed(twin["contours"]))
    return [("reverse_nonoverlapping_draw_order", twin)]
