"""Independent image-only inverse for directional shadow fill."""

from __future__ import annotations

from PIL import Image

_PROFILES = {
    "crown": (2, 3, 5, 7, 8, 8, 7, 5, 3, 2),
    "bowl": (8, 7, 5, 3, 2, 2, 3, 5, 7, 8),
    "stairs": (1, 2, 3, 4, 5, 6, 6, 7, 8, 8),
    "wave": (5, 7, 8, 7, 5, 3, 2, 3, 5, 5),
}


def _magenta_center(image: Image.Image) -> tuple[float, float]:
    points = []
    for y in range(image.height):
        for x in range(image.width):
            red, green, blue = image.getpixel((x, y))
            if red > 150 and red > green * 2 and red > blue * 1.25:
                points.append((x, y))
    if len(points) < 200:
        raise ValueError("missing projection-side marker")
    return (sum(x for x, _ in points) / len(points), sum(y for _, y in points) / len(points))


def _upright(image: Image.Image) -> Image.Image:
    mx, my = _magenta_center(image)
    cx, cy = image.width / 2, image.height / 2
    if my > cy + 100:
        turns = 0
    elif mx < cx - 100:
        turns = 1
    elif my < cy - 100:
        turns = 2
    elif mx > cx + 100:
        turns = 3
    else:
        raise ValueError("ambiguous projection-side marker")
    return image.rotate(90 * turns, resample=Image.Resampling.NEAREST)


def _dark(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return max(pixel) < 90 and max(pixel) - min(pixel) < 35 and red < 60


def decision_from_image(image: Image.Image) -> str:
    if not isinstance(image, Image.Image):
        raise TypeError("expected PIL image")
    raster = _upright(image.convert("RGB"))
    if raster.size != (640, 640):
        raise ValueError("unexpected raster size")
    heights = []
    for col in range(10):
        occupied = []
        for row in range(10):
            x, y = 120 + col * 40 + 20, 100 + row * 40 + 20
            if _dark(raster.getpixel((x, y))):
                occupied.append(row)
        if not occupied:
            raise ValueError("empty projection ray")
        heights.append(10 - min(occupied))
    scores = sorted(
        (sum(abs(a - b) for a, b in zip(heights, profile)), label)
        for label, profile in _PROFILES.items()
    )
    if len(scores) < 2 or scores[1][0] - scores[0][0] < 2:
        raise ValueError("ambiguous skyline")
    return scores[0][1]
