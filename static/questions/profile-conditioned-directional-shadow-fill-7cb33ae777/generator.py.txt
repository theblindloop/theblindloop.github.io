"""Deterministic latent sampler and raster renderer for directional shadow fill."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

LABELS = ("crown", "bowl", "stairs", "wave")
PROFILES = {
    "crown": (2, 3, 5, 7, 8, 8, 7, 5, 3, 2),
    "bowl": (8, 7, 5, 3, 2, 2, 3, 5, 7, 8),
    "stairs": (1, 2, 3, 4, 5, 6, 6, 7, 8, 8),
    "wave": (5, 7, 8, 7, 5, 3, 2, 3, 5, 5),
}
PALETTES = (
    {"dark": (24, 28, 34), "cyan": (0, 151, 181), "magenta": (214, 37, 126), "paper": (250, 248, 241)},
    {"dark": (38, 31, 25), "cyan": (0, 132, 168), "magenta": (190, 39, 121), "paper": (247, 250, 246)},
    {"dark": (20, 33, 30), "cyan": (8, 145, 169), "magenta": (207, 45, 117), "paper": (251, 249, 244)},
)


def _canonical_dark_cells(label: str, rng: random.Random) -> list[list[int]]:
    profile = PROFILES[label]
    cells = {(10 - height, col) for col, height in enumerate(profile)}
    available = [
        (row, col)
        for col, height in enumerate(profile)
        for row in range(10 - height, 10)
        if (row, col) not in cells
    ]
    rng.shuffle(available)
    cells.update(available[:16])  # exactly 26 dark cells in every class
    return [list(cell) for cell in sorted(cells)]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    dark_cells = _canonical_dark_cells(label, rng)
    free = [(r, c) for r in range(10) for c in range(10) if [r, c] not in dark_cells]
    rng.shuffle(free)
    return {
        "label": label,
        "dark_cells": dark_cells,
        "orientation": rng.randrange(4),
        "palette": rng.randrange(len(PALETTES)),
        "distractors": [list(cell) for cell in sorted(free[:12])],
    }


def _canonical_image(scene: dict[str, Any]) -> Image.Image:
    # Reading label here makes the gold-bearing variable visibly audited; the
    # assertion rejects inconsistent latent records rather than hiding a key.
    label = scene["label"]
    dark_cells = {tuple(cell) for cell in scene["dark_cells"]}
    profile = PROFILES[label]
    recovered = tuple(10 - min(r for r, c2 in dark_cells if c2 == c) for c in range(10))
    if recovered != profile:
        raise ValueError("dark cells do not encode the declared label")
    colors = PALETTES[int(scene["palette"])]
    image = Image.new("RGB", (640, 640), colors["paper"])
    draw = ImageDraw.Draw(image)
    x0, y0, cell = 120, 100, 40
    draw.rounded_rectangle((78, 48, 562, 570), radius=22, fill=(255, 255, 255), outline=(215, 218, 216), width=3)
    for row in range(10):
        for col in range(10):
            xa, ya = x0 + col * cell, y0 + row * cell
            draw.rectangle((xa + 1, ya + 1, xa + cell - 1, ya + cell - 1), fill=(244, 244, 240))
    for row, col in scene["distractors"]:
        cx, cy = x0 + col * cell + cell // 2, y0 + row * cell + cell // 2
        draw.ellipse((cx - 6, cy - 6, cx + 6, cy + 6), fill=(157, 162, 164))
    for row, col in dark_cells:
        xa, ya = x0 + col * cell, y0 + row * cell
        draw.rectangle((xa + 6, ya + 6, xa + cell - 6, ya + cell - 6), fill=colors["dark"])
    for index in range(11):
        x = x0 + index * cell
        y = y0 + index * cell
        draw.line((x, y0, x, y0 + 400), fill=(197, 203, 202), width=1)
        draw.line((x0, y, x0 + 400, y), fill=(197, 203, 202), width=1)
    draw.rectangle((x0 - 3, y0 - 3, x0 + 403, y0 + 403), outline=colors["cyan"], width=6)
    draw.rounded_rectangle((225, 520, 415, 542), radius=7, fill=colors["magenta"])
    draw.polygon(((310, 510), (330, 510), (320, 520)), fill=colors["magenta"])
    return image


def render(scene: dict[str, Any]) -> Image.Image:
    image = _canonical_image(scene)
    turns = int(scene["orientation"]) % 4
    return image.rotate(-90 * turns, resample=Image.Resampling.NEAREST)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["label"])


def _profile_from_scene(scene: dict[str, Any]) -> tuple[int, ...]:
    cells = {tuple(cell) for cell in scene["dark_cells"]}
    return tuple(10 - min(row for row, col2 in cells if col2 == col) for col in range(10))


def margin(scene: dict[str, Any]) -> float:
    observed = _profile_from_scene(scene)
    scores = sorted(sum(abs(a - b) for a, b in zip(observed, profile)) for profile in PROFILES.values())
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
