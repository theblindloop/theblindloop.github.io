"""Billiard-ball rack count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of billiard balls) when a familiar canonical prior ("a full billiard
rack holds fifteen balls") is contradicted by the actual image, which in the
counterfactual scenes shows a rack holding a visibly different number of balls
arranged as a pyramid (triangular) rack.

Latent z = {n, rows, ball_radius, gap, jitter_dx, jitter_dy, width, height}.
``n`` is the number of billiard balls actually drawn in the rack; it is the only
field that determines the analytic gold. Every field is directly visible in the
raster: the rack is a filled pyramid triangle of ``n`` ball-coloured disks on a
plain background, so there is no hidden or latent-only state and the world
declares latent-alias mode ``not_applicable``.

The closed-form gold is

    decision = str(n)

a scene parameter, not a pixel measurement.

The pixel-only inverse arm segments the ball colour and counts the connected
components of ball-coloured pixels, which equals the number of separated disks
because adjacent balls are spaced by a clear, always-present gap. The continuous
verifiability margin is the smallest centre-to-centre clearance (in pixels)
between two neighbouring balls; when that clearance is too small the disks merge
and become pixel-ambiguous, so such scenes are quarantined.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_pool_rack-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
OUTLINE = (236, 238, 242)   # light rack frame (ignored by the oracle)
BALL = (52, 84, 160)        # the only counted, saturated pixels

BALL_RADIUS = 30
GAP = 16                      # horizontal clearance between adjacent balls
D = 2 * BALL_RADIUS + GAP     # centre-to-centre horizontal spacing
VSTEP = D * 0.866             # vertical row step of the pyramid lattice

SUPPORTED_COUNTS = (6, 10, 15, 21)   # triangular numbers T3..T6
QUAR_CLEARANCE = 8.0


def _rows_for(n: int) -> int:
    return {6: 3, 10: 4, 15: 5, 21: 6}.get(int(n), 0)


def margin(scene: dict) -> float:
    """Nearest-neighbour ball clearance in pixels (count robustness)."""
    return float(scene["gap"]) + BALL_RADIUS


def is_quarantined(scene: dict) -> bool:
    n = int(scene["n"])
    if n not in SUPPORTED_COUNTS:
        return True
    return margin(scene) < QUAR_CLEARANCE + BALL_RADIUS


def _centers(n: int, jitter_dx: float, jitter_dy: float) -> list[tuple[float, float]]:
    rows = _rows_for(n)
    cx, cy = CANVAS[0] / 2.0, CANVAS[1] / 2.0
    height = (rows - 1) * VSTEP
    y_top = cy - height / 2.0
    centers = []
    for row in range(rows):
        y = y_top + row * VSTEP
        for k in range(row + 1):
            x = cx + (k - row / 2.0) * D
            centers.append((x + jitter_dx, y + jitter_dy))
    return centers


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    n = SUPPORTED_COUNTS[seed % len(SUPPORTED_COUNTS)]
    scene = {
        "n": n,
        "rows": _rows_for(n),
        "ball_radius": BALL_RADIUS,
        "gap": GAP,
        "jitter_dx": rng.uniform(-24.0, 24.0),
        "jitter_dy": rng.uniform(-24.0, 24.0),
        "width": CANVAS[0],
        "height": CANVAS[1],
    }
    return scene


def render(scene: dict) -> Image.Image:
    n = int(scene["n"])
    radius = int(scene["ball_radius"])
    img = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(img)
    centers = _centers(n, float(scene["jitter_dx"]), float(scene["jitter_dy"]))
    for (x, y) in centers:
        draw.ellipse(
            [x - radius, y - radius, x + radius, y + radius], fill=BALL, outline=OUTLINE, width=2
        )
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["n"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """None: every scene field is visible, so there is no latent alias to test."""
    return []
