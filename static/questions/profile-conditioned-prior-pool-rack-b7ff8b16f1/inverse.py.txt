"""Independent pixel-only inverse arm for the billiard-ball rack world.

Receives ONLY the final PNG and returns the number of balls drawn in the rack as
a string. It shares no geometry constants with renderer.render except the ball
colour, which is part of the visible image. It never imports renderer, prompts,
generate, verify, or analytic-gold code, and accepts no latent scene data.

Method: the rack's balls are the only saturated disk-shaped pixels; the light
rack frame and background are ignored. The oracle thresholds the ball colour to
a binary mask and counts the connected components of ball-coloured pixels. The
pyramid-lattice balls are always separated by a clean gap, so each disk is its
own component and the component count equals the number of balls.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_pool_rack-oracle-0.1.0"

_BALL = np.array((52, 84, 160), dtype=np.float32)
_COLOR_DIST = 80.0


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected foreground components in a boolean mask via BFS."""
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return 0
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    count = 0
    for start in range(len(ys)):
        sy, sx = int(ys[start]), int(xs[start])
        if visited[sy, sx]:
            continue
        count += 1
        stack = deque([(sy, sx)])
        visited[sy, sx] = True
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    h, w = arr.shape[:2]
    ball = np.linalg.norm(arr - _BALL, axis=-1) < _COLOR_DIST
    if not ball.any():
        return "0"
    return str(_count_components(ball))


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    ball = np.linalg.norm(arr - _BALL, axis=-1) < _COLOR_DIST
    return {
        "ball_px": int(ball.sum()),
        "components": _count_components(ball),
        "size": image.size,
    }
