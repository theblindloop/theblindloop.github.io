"""Renderer and analytic arm for alternating-deal correspondence."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH = HEIGHT = 800
TAG_COLORS = ("ruby", "blue")
RGB = {
    "paper": (248, 245, 235),
    "faint": (226, 222, 207),
    "ink": (52, 61, 70),
    "ref": (132, 139, 147),
    "teal": (38, 164, 154),
    "amber": (239, 166, 54),
    "ruby": (205, 55, 78),
    "blue": (51, 105, 206),
}


def _deal(reference):
    return [list(reference[0::2]), list(reference[1::2])]


def _flat(candidate):
    return tuple(int(v) for row in candidate for v in row)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = [0, 0, 0, 1, 1, 1]
    rng.shuffle(reference)
    correct = _deal(reference)
    distractor_flat = list(_flat(correct))
    zeros = [i for i, value in enumerate(distractor_flat) if value == 0]
    ones = [i for i, value in enumerate(distractor_flat) if value == 1]
    rng.shuffle(zeros)
    rng.shuffle(ones)
    swap_pairs = 1 + (int(seed) % 3)
    for zi, oi in zip(zeros[:swap_pairs], ones[:swap_pairs], strict=True):
        distractor_flat[zi], distractor_flat[oi] = distractor_flat[oi], distractor_flat[zi]
    distractor = [distractor_flat[:3], distractor_flat[3:]]

    correct_index = int(seed) % 2
    candidates = [distractor]
    candidates.insert(correct_index, correct)

    desired = TAG_COLORS[(int(seed) // 2) % 2]
    tags = list(TAG_COLORS)
    if tags[correct_index] != desired:
        tags.reverse()

    left_cell = 1 + ((int(seed) // 4) % 4)
    y_cell = 2 + ((int(seed) // 16) % 4)
    return {
        "reference": reference,
        "candidates": candidates,
        "tag_colors": tags,
        "badge_anchor": [left_cell, y_cell],
        "width": WIDTH,
        "height": HEIGHT,
    }


def _bead(draw, x: int, y: int, value: int) -> None:
    color = RGB["amber"] if int(value) else RGB["teal"]
    draw.rounded_rectangle((x - 3, y - 3, x + 3, y + 3), radius=1, fill=color)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), RGB["paper"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    # A quiet sorting-sheet world; these remote marks are deliberately nondiagnostic.
    draw.rounded_rectangle((32, 32, width - 33, height - 33), radius=28,
                           outline=RGB["faint"], width=3)
    for y in range(100, height - 80, 100):
        draw.line((72, y, width - 72, y), fill=RGB["faint"], width=1)
    draw.text((64, 58), "ALTERNATING DEAL / POSITION CORRESPONDENCE", fill=RGB["ink"], font=font)
    draw.text((64, height - 72), "odd positions -> top     even positions -> bottom",
              fill=RGB["ink"], font=font)

    left_cell, y_cell = (int(v) for v in scene["badge_anchor"])
    rx = left_cell * 100 + 50
    cx = (left_cell + 2) * 100 + 50
    cy = y_cell * 100 + 50

    # Two visibly separate local islands with open paper between them.
    draw.rounded_rectangle((rx - 29, cy - 14, rx + 29, cy + 14), radius=6,
                           fill=(236, 234, 227), outline=RGB["faint"], width=1)
    draw.rectangle((rx - 2, cy - 12, rx + 2, cy - 8), fill=RGB["ref"])
    for index, value in enumerate(scene["reference"]):
        _bead(draw, rx - 20 + 8 * index, cy + 4, int(value))
    draw.text((rx - 16, cy - 27), "RIBBON", fill=RGB["ink"], font=font)

    draw.rounded_rectangle((cx - 27, cy - 25, cx + 27, cy + 25), radius=6,
                           fill=(236, 234, 227), outline=RGB["faint"], width=1)
    for candidate_index, (candidate, tag) in enumerate(
        zip(scene["candidates"], scene["tag_colors"], strict=True)
    ):
        base_y = cy - 13 + candidate_index * 26
        for row in range(2):
            for col in range(3):
                _bead(draw, cx - 14 + 8 * col, base_y - 4 + 8 * row,
                      int(candidate[row][col]))
        draw.ellipse((cx + 13, base_y - 5, cx + 21, base_y + 3), fill=RGB[str(tag)])
    draw.text((cx - 13, cy - 39), "DEALS", fill=RGB["ink"], font=font)
    return image


def analytic_gold(scene: dict) -> str:
    target = _flat(_deal(scene["reference"]))
    matches = [i for i, candidate in enumerate(scene["candidates"])
               if _flat(candidate) == target]
    if len(matches) != 1:
        raise ValueError("scene lacks one unique alternating deal")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _flat(_deal(scene["reference"]))
    distances = sorted(sum(a != b for a, b in zip(_flat(c), target, strict=True))
                       for c in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _flat(_deal(scene["reference"]))
        distances = [sum(a != b for a, b in zip(_flat(c), target, strict=True))
                     for c in scene["candidates"]]
        counts_ok = len(scene["reference"]) == 6 and sum(scene["reference"]) == 3
        candidates_ok = all(len(c) == 2 and all(len(row) == 3 for row in c)
                            and sum(map(sum, c)) == 3 for c in scene["candidates"])
        return not (counts_ok and candidates_ok and distances.count(0) == 1
                    and sorted(distances)[1] >= 2)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
