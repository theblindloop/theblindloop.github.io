"""Independent image-only inverse arm for alternating-deal correspondence."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = {
    "ref": np.array((132, 139, 147), dtype=np.int16),
    "teal": np.array((38, 164, 154), dtype=np.int16),
    "amber": np.array((239, 166, 54), dtype=np.int16),
    "ruby": np.array((205, 55, 78), dtype=np.int16),
    "blue": np.array((51, 105, 206), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 8):
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            pts = np.asarray(points)
            output.append((float(pts[:, 1].mean()), float(pts[:, 0].mean()), len(points)))
    return output


def _single_marker(array: np.ndarray, name: str):
    items = _components(_mask(array, COLORS[name]), minimum=12)
    if len(items) != 1:
        raise ValueError(f"missing or duplicated {name} marker")
    return items[0]


def _beads(array: np.ndarray):
    items = []
    for value, name in ((0, "teal"), (1, "amber")):
        for x, y, area in _components(_mask(array, COLORS[name]), minimum=20):
            if not 25 <= area <= 70:
                raise ValueError("malformed bead")
            items.append((x, y, value))
    if len(items) != 18:
        raise ValueError("expected eighteen visible beads")
    return items


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 300:
        raise ValueError("unsupported raster")
    ref_x, ref_y, _ = _single_marker(array, "ref")
    markers = {name: _single_marker(array, name) for name in ("ruby", "blue")}
    beads = _beads(array)

    reference_items = sorted((x, value) for x, y, value in beads
                             if abs(y - (ref_y + 14.0)) < 7 and abs(x - ref_x) < 30)
    if len(reference_items) != 6:
        raise ValueError("reference ribbon is incomplete")
    reference = [value for _, value in reference_items]
    if sum(reference) != 3:
        raise ValueError("reference colors are unbalanced")
    target = tuple(reference[0::2] + reference[1::2])

    matches = []
    for label, (tag_x, tag_y, _) in markers.items():
        local = [(x, y, value) for x, y, value in beads
                 if tag_x - 42 < x < tag_x - 6 and abs(y - tag_y) < 12]
        if len(local) != 6:
            raise ValueError("candidate deal is incomplete")
        ys = sorted(y for _, y, _ in local)
        split = (ys[2] + ys[3]) / 2.0
        top = sorted((x, value) for x, y, value in local if y < split)
        bottom = sorted((x, value) for x, y, value in local if y > split)
        if len(top) != 3 or len(bottom) != 3:
            raise ValueError("candidate rows are malformed")
        candidate = tuple(value for _, value in top + bottom)
        if sum(candidate) != 3:
            raise ValueError("candidate colors are unbalanced")
        if candidate == target:
            matches.append(label)
    if len(matches) != 1:
        raise ValueError("image has no unique matching deal")
    return matches[0]
