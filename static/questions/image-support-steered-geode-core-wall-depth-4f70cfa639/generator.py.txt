"""Geode world: how many black walls shut the green patch in.

One small solid black stamp is carved into white chambers. Some chambers hold a
black pebble with a white core, and a pebble wall may carry a visible mouth that
opens its core into the surrounding chamber. Exactly one small green patch is
painted somewhere in the white space of the stamp. The decision is the number of
black walls that must be crossed to walk from the green patch out to the page
edge: one when the green patch sits in a chamber or in the core of a pebble
whose wall is broken, two when it sits in the core of a pebble whose wall runs
all the way round.

Everything decisive lives inside one small square inset of the page. The stamp
is a dense area-like block: erasing any window that overlaps it either opens a
wall, removes the green patch, or destroys the enclosure being counted.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS = 704
INSET = 192
BACKGROUND = (255, 255, 255)
BLACK = (20, 20, 20)
GREEN = (26, 150, 60)
DECOR_COLORS = [
    (54, 92, 214),    # blue
    (226, 132, 24),   # orange
    (176, 176, 176),  # light grey
    (168, 84, 198),   # purple
]

# green-patch host schedule, cycled by scene index so classes stay balanced
# and so that "there is a pebble" and "the green patch is small" carry no
# information about the answer.
_SCHEDULE = ["chamber", "sealed", "cut", "sealed", "cut", "sealed"]


def _rect_inside(inner, outer) -> bool:
    ix, iy, iw, ih = inner
    ox, oy, ow, oh = outer
    return ix >= ox and iy >= oy and ix + iw <= ox + ow and iy + ih <= oy + oh


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    index = seed % 1000003
    host_kind = _SCHEDULE[index % len(_SCHEDULE)]
    thin = (index % 7) == 3
    narrow = (index % 7) == 5

    wall = 6 if thin else (9 if narrow else rng.choice([11, 12, 13]))
    origin_x = rng.randrange(48, CANVAS - INSET - 48)
    origin_y = rng.randrange(48, CANVAS - INSET - 48)
    blob = [origin_x, origin_y, INSET, INSET]

    # four chambers carved out of the solid stamp in a 2x2 arrangement
    cell = (INSET - 3 * wall) // 2
    chambers = []
    for row in range(2):
        for col in range(2):
            cx = origin_x + wall + col * (cell + wall)
            cy = origin_y + wall + row * (cell + wall)
            chambers.append([cx, cy, cell, cell])

    # exactly two chambers carry a pebble; which two is independent of the answer
    pebble_slots = rng.sample(range(4), 2)
    pebble_inset = 4 if thin else (6 if narrow else 9)
    pebbles = []
    for slot in pebble_slots:
        cx, cy, cw, ch = chambers[slot]
        px, py = cx + pebble_inset, cy + pebble_inset
        pw, ph = cw - 2 * pebble_inset, ch - 2 * pebble_inset
        pwall = 5 if thin else (7 if narrow else rng.choice([10, 11, 12]))
        core = [px + pwall, py + pwall, pw - 2 * pwall, ph - 2 * pwall]
        pebbles.append({
            "chamber": slot,
            "rect": [px, py, pw, ph],
            "core": core,
            "mouth": None,
        })

    def set_mouth(pebble: dict, present: bool) -> None:
        if not present:
            pebble["mouth"] = None
            return
        px, py, pw, ph = pebble["rect"]
        cxr, cyr, cwr, chr_ = pebble["core"]
        pwall = cxr - px
        span = 8 if thin else (10 if narrow else rng.choice([14, 16, 18]))
        side = rng.choice(["n", "s", "e", "w"])
        if side in ("n", "s"):
            mx = cxr + (cwr - span) // 2
            my = py if side == "n" else cyr + chr_
            pebble["mouth"] = [mx, my, span, pwall]
        else:
            my = cyr + (chr_ - span) // 2
            mx = px if side == "w" else cxr + cwr
            pebble["mouth"] = [mx, my, pwall, span]

    if host_kind == "chamber":
        for pebble in pebbles:
            set_mouth(pebble, rng.random() < 0.5)
        free_chambers = [i for i in range(4) if i not in pebble_slots]
        host_index = rng.choice(free_chambers)
        host_rect = chambers[host_index]
    else:
        host_pebble = rng.randrange(2)
        set_mouth(pebbles[host_pebble], host_kind == "cut")
        set_mouth(pebbles[1 - host_pebble], rng.random() < 0.5)
        host_rect = pebbles[host_pebble]["core"]

    # the green patch has the same size distribution whatever it sits in, so
    # green ink is decorrelated from the answer
    gsize = 12 if thin else (14 if narrow else rng.choice([16, 18]))
    hx, hy, hw, hh = host_rect
    gx = hx + (hw - gsize) // 2
    gy = hy + (hh - gsize) // 2
    green = [gx, gy, gsize, gsize]

    decor = []
    for _ in range(rng.randrange(4, 7)):
        for _attempt in range(60):
            dw = rng.randrange(26, 62)
            dh = rng.randrange(26, 62)
            dx = rng.randrange(10, CANVAS - dw - 10)
            dy = rng.randrange(10, CANVAS - dh - 10)
            if (dx + dw + 26 < origin_x or dx - 26 > origin_x + INSET
                    or dy + dh + 26 < origin_y or dy - 26 > origin_y + INSET):
                nested = rng.random() < 0.6
                decor.append({
                    "rect": [dx, dy, dw, dh],
                    "color": rng.randrange(len(DECOR_COLORS)),
                    "nested": nested,
                })
                break

    return {
        "canvas": CANVAS,
        "blob": blob,
        "chambers": chambers,
        "pebbles": pebbles,
        "green": green,
        "decor": decor,
    }


def _fill(buf: np.ndarray, rect, color) -> None:
    x, y, w, h = [int(v) for v in rect]
    buf[y:y + h, x:x + w] = color


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    buf = np.zeros((size, size, 3), dtype=np.uint8)
    buf[:, :] = BACKGROUND

    # decoration first: never black, so it is free space for the wall count
    for item in scene["decor"]:
        color = DECOR_COLORS[int(item["color"])]
        dx, dy, dw, dh = [int(v) for v in item["rect"]]
        _fill(buf, [dx, dy, dw, dh], color)
        if item.get("nested"):
            ring = 7
            _fill(buf, [dx + ring, dy + ring, dw - 2 * ring, dh - 2 * ring], BACKGROUND)
            inner = 2 * ring
            if dw - 2 * inner > 8 and dh - 2 * inner > 8:
                _fill(buf, [dx + inner, dy + inner, dw - 2 * inner, dh - 2 * inner], color)

    _fill(buf, scene["blob"], BLACK)
    for chamber in scene["chambers"]:
        _fill(buf, chamber, BACKGROUND)
    for pebble in scene["pebbles"]:
        _fill(buf, pebble["rect"], BLACK)
        _fill(buf, pebble["core"], BACKGROUND)
        if pebble["mouth"] is not None:
            _fill(buf, pebble["mouth"], BACKGROUND)
    _fill(buf, scene["green"], GREEN)
    return Image.fromarray(buf, mode="RGB")


def _green_host(scene: dict):
    """Innermost painted white region that contains the green patch."""
    green = scene["green"]
    for pebble in scene["pebbles"]:
        if _rect_inside(green, pebble["core"]):
            return ("core", pebble)
    for index, chamber in enumerate(scene["chambers"]):
        if _rect_inside(green, chamber):
            return ("chamber", index)
    return ("outside", None)


def analytic_gold(scene: dict) -> str:
    kind, item = _green_host(scene)
    if kind == "core":
        # a mouth cut through the pebble wall joins the core to its chamber, so
        # the pebble wall no longer has to be crossed
        return "d1" if item["mouth"] is not None else "d2"
    if kind == "chamber":
        return "d1"
    return "d0"


def _wall_widths(scene: dict) -> list:
    widths = []
    bx, by, bw, bh = scene["blob"]
    for cx, cy, cw, ch in scene["chambers"]:
        widths.extend([cx - bx, cy - by, bx + bw - (cx + cw), by + bh - (cy + ch)])
    for pebble in scene["pebbles"]:
        px, py, pw, ph = pebble["rect"]
        kx, ky, kw, kh = pebble["core"]
        widths.extend([kx - px, ky - py, px + pw - (kx + kw), py + ph - (ky + kh)])
        cx, cy, cw, ch = scene["chambers"][pebble["chamber"]]
        widths.extend([px - cx, py - cy, cx + cw - (px + pw), cy + ch - (py + ph)])
        if pebble["mouth"] is not None:
            mx, my, mw, mh = pebble["mouth"]
            widths.append(min(mw, mh))
    return [float(v) for v in widths]


def margin(scene: dict) -> float:
    """Weakest decisive width in pixels on this page."""
    widths = _wall_widths(scene)
    kind, item = _green_host(scene)
    host_rect = item["core"] if kind == "core" else scene["chambers"][item]
    hx, hy, hw, hh = host_rect
    gx, gy, gw, gh = scene["green"]
    widths.extend([
        float(gx - hx), float(gy - hy),
        float(hx + hw - (gx + gw)), float(hy + hh - (gy + gh)),
    ])
    return float(min(widths))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every sampled field is painted."""
    return []
