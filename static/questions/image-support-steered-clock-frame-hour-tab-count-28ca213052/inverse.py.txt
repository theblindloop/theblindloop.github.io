"""Independent image-only inverse based on chromatic angular occupancy."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _sample(mask: np.ndarray, cx: float, cy: float, radius: float, angles: np.ndarray) -> np.ndarray:
    xs = np.rint(cx + radius * np.cos(angles)).astype(int)
    ys = np.rint(cy + radius * np.sin(angles)).astype(int)
    valid = (xs >= 0) & (xs < mask.shape[1]) & (ys >= 0) & (ys < mask.shape[0])
    values = np.zeros(len(angles), dtype=bool)
    values[valid] = mask[ys[valid], xs[valid]]
    return values


def _circular_close(signal: np.ndarray, gap: int = 3) -> np.ndarray:
    output = signal.astype(bool).copy()
    size = len(output)
    for start in range(size):
        if not output[start]:
            continue
        length = 1
        while length <= gap and not output[(start + length) % size]:
            length += 1
        if length <= gap and output[(start + length) % size]:
            for offset in range(1, length):
                output[(start + offset) % size] = True
    return output


def _run_lengths(signal: np.ndarray) -> list[int]:
    if signal.all():
        return [len(signal)]
    if not signal.any():
        return []
    start = int(np.flatnonzero(~signal)[0])
    rotated = np.roll(signal, -start)
    lengths = []
    current = 0
    for value in rotated:
        if value:
            current += 1
        elif current:
            lengths.append(current)
            current = 0
    if current:
        lengths.append(current)
    return lengths


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    colored = (high - low > 55) & (low < 225)
    points = np.argwhere(colored)
    if len(points) < 20_000:
        raise ValueError("no substantial colored clock frame")

    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width = float(x1 - x0 + 1)
    height = float(y1 - y0 + 1)
    if min(width, height) < 420 or max(width, height) / min(width, height) > 1.08:
        raise ValueError("colored frame extent is missing or distorted")
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    outer = (width + height) / 4.0

    angles = np.linspace(0.0, 2.0 * math.pi, 1440, endpoint=False)

    # A tab counts only as fused to one intact frame, so continuity is evidence-bearing.
    ring_votes = np.stack(
        [_sample(colored, cx, cy, factor * outer, angles) for factor in (0.81, 0.85, 0.89)]
    ).sum(axis=0)
    intact = ring_votes >= 2
    missing_runs = _run_lengths(~intact)
    if float(intact.mean()) < 0.985 or (missing_runs and max(missing_runs) > 5):
        raise ValueError("colored annular frame is not visibly continuous")

    tab_votes = np.stack(
        [_sample(colored, cx, cy, factor * outer, angles) for factor in (0.57, 0.61, 0.65, 0.68)]
    ).sum(axis=0)
    tabs = _circular_close(tab_votes >= 3, gap=3)
    runs = [length for length in _run_lengths(tabs) if length >= 10]
    count = len(runs)
    if count not in {10, 11, 12, 13}:
        raise ValueError(f"ambiguous inward-tab count: {count}")
    if not runs or min(runs) < 14 or max(runs) > 70:
        raise ValueError("tab widths are not cleanly resolved")
    return str(count)
