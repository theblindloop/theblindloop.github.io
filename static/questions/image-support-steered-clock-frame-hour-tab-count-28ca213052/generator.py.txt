"""Deterministic Pillow renderer for a clock frame with fused inward hour tabs."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIZE = 640
SUPERSAMPLE = 3
BACKGROUND = (248, 247, 242)
PALETTES = (
    (34, 105, 210),
    (155, 62, 202),
    (18, 151, 126),
    (211, 67, 88),
)
TOTAL_TAB_ANGLE = 2.20


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC10C5A7)
    return {
        "tab_count": 10 + (int(seed) % 4),
        "center_x": rng.randint(310, 330),
        "center_y": rng.randint(310, 330),
        "outer_radius": rng.randint(262, 276),
        "rotation": rng.uniform(0.0, 2.0 * math.pi),
        "palette_index": rng.randrange(len(PALETTES)),
        "hand_angle": rng.uniform(0.0, 2.0 * math.pi),
        "render_nonce": rng.randrange(1_000_000),
    }


def _scaled_point(cx: float, cy: float, radius: float, angle: float) -> tuple[int, int]:
    return (
        round((cx + radius * math.cos(angle)) * SUPERSAMPLE),
        round((cy + radius * math.sin(angle)) * SUPERSAMPLE),
    )


def render(scene: dict) -> Image.Image:
    # The zero-amplitude nonce access is a declared render-invariant latent alias.
    cx = float(scene["center_x"]) + 0.0 * int(scene["render_nonce"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    count = int(scene["tab_count"])
    rotation = float(scene["rotation"])
    color = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    hand_angle = float(scene["hand_angle"])

    canvas = Image.new("RGB", (SIZE * SUPERSAMPLE, SIZE * SUPERSAMPLE), BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    box = tuple(round(value * SUPERSAMPLE) for value in (cx - outer, cy - outer, cx + outer, cy + outer))
    draw.ellipse(box, fill=color)

    inner = 0.72 * outer
    inner_box = tuple(
        round(value * SUPERSAMPLE)
        for value in (cx - inner, cy - inner, cx + inner, cy + inner)
    )
    draw.ellipse(inner_box, fill=BACKGROUND)

    # Total colored tab angle is fixed, so tab identity is not predicted by total ink.
    half_width = TOTAL_TAB_ANGLE / (2.0 * count)
    tab_inner = 0.54 * outer
    tab_outer = 0.735 * outer
    for index in range(count):
        angle = rotation + 2.0 * math.pi * index / count
        polygon = [
            _scaled_point(cx, cy, tab_inner, angle - half_width),
            _scaled_point(cx, cy, tab_outer, angle - half_width),
            _scaled_point(cx, cy, tab_outer, angle + half_width),
            _scaled_point(cx, cy, tab_inner, angle + half_width),
        ]
        draw.polygon(polygon, fill=color)

    # Neutral hands and pin are explicitly excluded by the wording and color inverse.
    neutral = (91, 95, 101)
    hand_width = 9 * SUPERSAMPLE
    long_end = _scaled_point(cx, cy, 0.39 * outer, hand_angle)
    short_end = _scaled_point(cx, cy, 0.28 * outer, hand_angle + 2.05)
    center = (round(cx * SUPERSAMPLE), round(cy * SUPERSAMPLE))
    draw.line((center, long_end), fill=neutral, width=hand_width)
    draw.line((center, short_end), fill=neutral, width=hand_width)
    pin = 12 * SUPERSAMPLE
    draw.ellipse((center[0] - pin, center[1] - pin, center[0] + pin, center[1] + pin), fill=neutral)

    return canvas.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["tab_count"])


def margin(scene: dict) -> float:
    count = int(scene["tab_count"])
    outer = float(scene["outer_radius"])
    gap_angle = (2.0 * math.pi - TOTAL_TAB_ANGLE) / count
    return float(0.54 * outer * gap_angle)


def is_quarantined(scene: dict) -> bool:
    outer = float(scene["outer_radius"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    count = int(scene["tab_count"])
    return (
        count not in {10, 11, 12, 13}
        or margin(scene) < 38.0
        or min(cx - outer, cy - outer, SIZE - cx - outer, SIZE - cy - outer) < 20.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["render_nonce"] = int(scene["render_nonce"]) + 1
    return [("render_nonce_alias", twin)]
