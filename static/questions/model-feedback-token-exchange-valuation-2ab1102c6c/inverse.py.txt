"""Independent pixel-only inverse arm for the token-exchange valuation world.

Reads nothing but the rendered PNG: it segments the black ink into connected
components, classifies each glyph by how much of its bounding box it fills,
recovers the two drawn equalities, solves them for the worth of every glyph, and
then sums each colour-framed purse.  It never imports the renderer, the prompts,
the generator or the verifier, and it never receives latent scene data.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (37, 86, 196),
    "green": (24, 140, 66),
    "orange": (232, 124, 20),
    "purple": (128, 52, 168),
}

_CIRCLE, _TRIANGLE, _SQUARE = "circle", "triangle", "square"


def _components(mask: np.ndarray) -> list:
    """Four-connected components of a boolean mask, as bounding boxes + areas."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for sy in range(height):
        row = mask[sy]
        if not row.any():
            continue
        for sx in range(width):
            if not row[sx] or seen[sy, sx]:
                continue
            queue = deque([(sy, sx)])
            seen[sy, sx] = True
            pixels = 0
            y0 = y1 = sy
            x0 = x1 = sx
            while queue:
                y, x = queue.popleft()
                pixels += 1
                if y < y0:
                    y0 = y
                if y > y1:
                    y1 = y
                if x < x0:
                    x0 = x
                if x > x1:
                    x1 = x
                for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                    if 0 <= ny < height and 0 <= nx < width:
                        if mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            queue.append((ny, nx))
            out.append(
                {
                    "x0": x0,
                    "y0": y0,
                    "x1": x1,
                    "y1": y1,
                    "w": x1 - x0 + 1,
                    "h": y1 - y0 + 1,
                    "area": pixels,
                    "cx": (x0 + x1) / 2.0,
                    "cy": (y0 + y1) / 2.0,
                }
            )
    return out


def _classify(part: dict) -> str:
    ratio = part["w"] / float(part["h"])
    if ratio > 2.0 or ratio < 0.5:
        return "bar"
    fill = part["area"] / float(part["w"] * part["h"])
    if fill >= 0.90:
        return _SQUARE
    if 0.66 <= fill <= 0.88:
        return _CIRCLE
    if 0.38 <= fill <= 0.64:
        return _TRIANGLE
    raise ValueError(f"unclassifiable glyph with fill {fill:.3f}")


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB")).astype(np.int16)

    # --- purse frames -------------------------------------------------------
    boxes = {}
    for name, hue in _PALETTE.items():
        mask = np.all(rgb == np.array(hue, dtype=np.int16), axis=-1)
        if not mask.any():
            continue
        ys, xs = np.nonzero(mask)
        boxes[name] = (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
    if len(boxes) != 4:
        raise ValueError("expected exactly four colour-framed purses")

    # --- black ink ----------------------------------------------------------
    black = np.all(rgb <= 40, axis=-1)
    parts = _components(black)
    if not parts:
        raise ValueError("no black ink found")

    purse_parts = {name: [] for name in boxes}
    loose = []
    for part in parts:
        home = None
        for name, (bx0, by0, bx1, by1) in boxes.items():
            if bx0 <= part["cx"] <= bx1 and by0 <= part["cy"] <= by1:
                home = name
                break
        if home is None:
            loose.append(part)
        else:
            purse_parts[home].append(part)

    # --- the drawn equalities ----------------------------------------------
    bars = [p for p in loose if _classify(p) == "bar"]
    glyphs = [p for p in loose if _classify(p) != "bar"]
    if len(bars) != 4:
        raise ValueError("expected two equality signs outside the purses")

    bars.sort(key=lambda p: p["cy"])
    rows = [bars[0:2], bars[2:4]]
    relations = []
    for pair in rows:
        eq_x = sum(p["cx"] for p in pair) / 2.0
        eq_y = sum(p["cy"] for p in pair) / 2.0
        band = [g for g in glyphs if abs(g["cy"] - eq_y) < 45]
        left = [g for g in band if g["cx"] < eq_x]
        right = [g for g in band if g["cx"] > eq_x]
        if len(left) != 1 or not right:
            raise ValueError("malformed equality row")
        right_kinds = {_classify(g) for g in right}
        if len(right_kinds) != 1:
            raise ValueError("mixed glyphs on the right of an equality")
        relations.append((_classify(left[0]), right_kinds.pop(), len(right)))

    # --- solve the value system --------------------------------------------
    worth = {_CIRCLE: 1}
    for _ in range(len(relations) + 1):
        for lhs, rhs, count in relations:
            if rhs in worth:
                worth[lhs] = count * worth[rhs]
    if len(worth) < 3:
        raise ValueError("the drawn equalities do not determine every glyph")

    totals = {}
    for name, items in purse_parts.items():
        totals[name] = sum(worth[_classify(p)] for p in items)

    best = max(totals.values())
    winners = [name for name, value in totals.items() if value == best]
    if len(winners) != 1:
        raise ValueError("no unique richest purse")
    return winners[0]
