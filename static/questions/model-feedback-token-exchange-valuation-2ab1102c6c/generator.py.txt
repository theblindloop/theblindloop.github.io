"""Scene model and renderer for the token-exchange valuation world.

Two drawn equalities fix a value system over three black token glyphs: one
triangle balances a run of circles, and one square balances a run of triangles.
The square's worth is therefore only obtainable by composing both equalities.
Four colour-framed purses hold interleaved mixtures of the three glyphs, and the
answer is the frame colour of the purse whose total worth is largest.

Only axis-aligned filled rectangles, filled ellipses and filled polygons are
drawn, so the raster is free of antialiasing and byte-deterministic.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 900
HEIGHT = 860

WHITE = (255, 255, 255)
BLACK = (0, 0, 0)

HUES = {
    "blue": (37, 86, 196),
    "green": (24, 140, 66),
    "orange": (232, 124, 20),
    "purple": (128, 52, 168),
}
COLOR_NAMES = ("blue", "green", "orange", "purple")

# Glyph geometry.  Every glyph is drawn inside the same square bounding box, so
# the three kinds differ only by how much of that box they fill: square 1.00,
# circle 0.79, triangle 0.50.
GLYPH = 56
CIRCLE, TRIANGLE, SQUARE = 0, 1, 2

# Conversion strip.
CONV_ROW_Y = (80, 175)
CONV_LEFT_X = 70
EQ_X = 170
EQ_W = 52
EQ_H = 12
EQ_GAP = 20
CONV_RIGHT_X = 270
CONV_PITCH = 84

# Purses.
PURSE_W = 390
PURSE_H = 280
FRAME = 12
PURSE_X = (30, 480)
PURSE_Y = (265, 575)
CELL_PITCH_X = 100
CELL_PITCH_Y = 100
CELL_X0 = 50
CELL_Y0 = 55
GRID_COLS = 3
GRID_ROWS = 2

QUARANTINE_MARGIN = 11.0

# Fixed geometric clearances of the layout, in pixels.
_GLYPH_GAP = CELL_PITCH_X - GLYPH  # 44
_ROW_GAP = CELL_PITCH_Y - GLYPH  # 44
_PURSE_GAP_X = PURSE_X[1] - (PURSE_X[0] + PURSE_W)  # 60
_PURSE_GAP_Y = PURSE_Y[1] - (PURSE_Y[0] + PURSE_H)  # 30
_FILL_SEPARATION = 21.5  # 100 x smallest fill-ratio gap between glyph kinds
_MAX_JITTER = 12


def _values(k: int, m: int) -> tuple:
    """Worth of (circle, triangle, square) in circle units."""
    return (1, k, k * m)


def _purse_values(scene: dict) -> list:
    vals = _values(scene["k"], scene["m"])
    return [sum(vals[t] for t in tokens) for tokens in scene["tokens"]]


def _flat_chain_values(scene: dict) -> list:
    """Worth if the second equality is dropped, i.e. a square read as a triangle."""
    vals = (1, scene["k"], scene["k"])
    return [sum(vals[t] for t in tokens) for tokens in scene["tokens"]]


def _counts(tokens: list, kind: int) -> int:
    return sum(1 for t in tokens if t == kind)


def _strict_argmax(values: list) -> int:
    best = max(values)
    if values.count(best) != 1:
        return -1
    return values.index(best)


def _sample_purse(rng: random.Random) -> list:
    n = rng.randint(4, GRID_COLS * GRID_ROWS)
    while True:
        tokens = [rng.randrange(3) for _ in range(n)]
        if len(set(tokens)) >= 2:
            rng.shuffle(tokens)
            return tokens


def sample_scene(seed: int) -> dict:
    """Sample one accepted scene deterministically from ``seed``."""
    rng = random.Random(seed)
    # Most scenes are sampled with a comfortable two-unit win.  One seed in
    # seven also admits a single-unit win, which the margin rule quarantines, so
    # the gallery still covers the near-boundary regime.
    least_gap = 1 if seed % 7 == 0 else 2
    for _ in range(4000):
        k = rng.choice((2, 3))
        m = rng.choice((2, 3))
        tokens = [_sample_purse(rng) for _ in range(4)]
        values = [sum(_values(k, m)[t] for t in purse) for purse in tokens]

        winner = _strict_argmax(values)
        if winner < 0:
            continue
        runner = sorted(values)[-2]
        gap = values[winner] - runner
        if not least_gap <= gap <= 5:
            continue

        # Decoy 1: the purse holding the most tokens must not be the answer.
        sizes = [len(p) for p in tokens]
        if _strict_argmax(sizes) == winner:
            continue

        # Decoy 2: the purse holding the most squares must not be the answer.
        squares = [_counts(p, SQUARE) for p in tokens]
        if _strict_argmax(squares) == winner:
            continue

        # Decoy 3: the purse holding the most triangles must not be the answer.
        triangles = [_counts(p, TRIANGLE) for p in tokens]
        if _strict_argmax(triangles) == winner:
            continue

        # Decoy 4: reading only the first equality, so a square counts as a
        # triangle, must select a different purse.
        flat = [sum((1, k, k)[t] for t in purse) for purse in tokens]
        if _strict_argmax(flat) == winner:
            continue

        # The second equality must be load bearing: repricing the square by
        # changing that row alone moves the label to a different purse.
        other_m = 3 if m == 2 else 2
        alt = [sum(_values(k, other_m)[t] for t in purse) for purse in tokens]
        if _strict_argmax(alt) == winner:
            continue

        colors = list(COLOR_NAMES)
        rng.shuffle(colors)
        return {
            "colors": colors,
            "tokens": tokens,
            "k": k,
            "m": m,
            "jitters": [rng.randrange(_MAX_JITTER + 1) for _ in range(4)],
            "conv_jitters": [rng.randrange(_MAX_JITTER + 1) for _ in range(2)],
        }
    raise RuntimeError("scene sampling failed to converge")


def _draw_glyph(draw: ImageDraw.ImageDraw, kind: int, x: int, y: int) -> None:
    x1, y1 = x + GLYPH - 1, y + GLYPH - 1
    if kind == SQUARE:
        draw.rectangle([x, y, x1, y1], fill=BLACK)
    elif kind == CIRCLE:
        draw.ellipse([x, y, x1, y1], fill=BLACK)
    else:
        draw.polygon([(x + GLYPH // 2, y), (x1, y1), (x, y1)], fill=BLACK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)

    # --- the two drawn equalities -------------------------------------------
    rows = (
        (TRIANGLE, CIRCLE, scene["k"]),
        (SQUARE, TRIANGLE, scene["m"]),
    )
    for idx, (left_kind, right_kind, count) in enumerate(rows):
        cy = CONV_ROW_Y[idx]
        jx = scene["conv_jitters"][idx]
        top = cy - GLYPH // 2
        _draw_glyph(draw, left_kind, CONV_LEFT_X + jx, top)
        ex = EQ_X + jx
        for sign in (-1, 1):
            by = cy + sign * (EQ_GAP // 2 + EQ_H // 2) - EQ_H // 2
            draw.rectangle([ex, by, ex + EQ_W - 1, by + EQ_H - 1], fill=BLACK)
        for j in range(count):
            _draw_glyph(draw, right_kind, CONV_RIGHT_X + jx + j * CONV_PITCH, top)

    # --- the four purses ----------------------------------------------------
    for slot in range(4):
        x0 = PURSE_X[slot % 2]
        y0 = PURSE_Y[slot // 2]
        hue = HUES[scene["colors"][slot]]
        draw.rectangle(
            [x0, y0, x0 + PURSE_W - 1, y0 + PURSE_H - 1], outline=hue, width=FRAME
        )
        jx = scene["jitters"][slot]
        for pos, kind in enumerate(scene["tokens"][slot]):
            col = pos % GRID_COLS
            row = pos // GRID_COLS
            gx = x0 + CELL_X0 + jx + col * CELL_PITCH_X
            gy = y0 + CELL_Y0 + row * CELL_PITCH_Y
            _draw_glyph(draw, kind, gx, gy)
    return image


def analytic_gold(scene: dict) -> str:
    values = _purse_values(scene)
    winner = values.index(max(values))
    return scene["colors"][winner]


def margin(scene: dict) -> float:
    values = _purse_values(scene)
    best = max(values)
    runner = sorted(values)[-2]
    frame_clearance = CELL_X0 - FRAME - _MAX_JITTER
    return float(
        min(
            _GLYPH_GAP,
            _ROW_GAP,
            _PURSE_GAP_X,
            _PURSE_GAP_Y,
            frame_clearance,
            _FILL_SEPARATION,
            6.0 * (best - runner),
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No transform holds the raster fixed: every sampled field is drawn."""
    return []
