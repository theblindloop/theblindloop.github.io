"""Analytic sampler and forward rasterizer for a contour-partition world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
ANCHOR_Y = (-16, 52, 120, 188, 256, 324, 392, 460, 528)
PATH_PALETTES = ("#1d3040", "#382a46", "#193d37", "#493225")
RED = "#d84b45"
BLUE = "#3478cf"


def sample_scene(seed: int) -> dict:
    """Sample pair-matched classes with independently randomized visible nuisances."""
    seed = int(seed)
    relation = "same" if seed % 2 == 0 else "opposite"
    # Consecutive even/odd seeds share all nuisance variables.  Only the blue
    # dot crosses the visible barrier, making color, trail geometry, position,
    # length, width, and total ink exactly balanced between the two classes.
    pair_index = seed // 2
    rng = random.Random(pair_index ^ 0x5A17C9)
    path_x = [rng.randint(226, 286)]
    for _ in range(len(ANCHOR_Y) - 1):
        delta = rng.choice((-36, -28, -20, 20, 28, 36))
        proposed = path_x[-1] + delta
        if proposed < 150 or proposed > 362:
            proposed = path_x[-1] - delta
        path_x.append(proposed)
    return {
        "relation": relation,
        "path_x": path_x,
        "red_y": rng.choice((96, 128, 160, 192)),
        "blue_y": rng.choice((320, 352, 384, 416)),
        "primary_side": rng.choice((-1, 1)),
        "dot_offset": 28 if pair_index % 6 == 0 else rng.choice((44, 48, 52)),
        "dot_radius": rng.choice((12, 13)),
        "stroke": rng.choice((10, 11)),
        "path_palette_index": rng.randrange(len(PATH_PALETTES)),
    }


def _path_x_slope(scene: dict, y: float) -> tuple[float, float]:
    anchors = scene["path_x"]
    for index in range(len(ANCHOR_Y) - 1):
        y0, y1 = ANCHOR_Y[index], ANCHOR_Y[index + 1]
        if y <= y1:
            fraction = (float(y) - y0) / (y1 - y0)
            x0, x1 = float(anchors[index]), float(anchors[index + 1])
            return x0 + fraction * (x1 - x0), (x1 - x0) / (y1 - y0)
    x0, x1 = float(anchors[-2]), float(anchors[-1])
    return x1, (x1 - x0) / (ANCHOR_Y[-1] - ANCHOR_Y[-2])


def _trail_points(scene: dict) -> list[tuple[float, float]]:
    return [(float(x), float(y)) for x, y in zip(scene["path_x"], ANCHOR_Y, strict=True)]


def _dot_positions(scene: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    red_y = float(scene["red_y"])
    blue_y = float(scene["blue_y"])
    red_path_x, _ = _path_x_slope(scene, red_y)
    blue_path_x, _ = _path_x_slope(scene, blue_y)
    red_side = int(scene["primary_side"])
    blue_side = red_side if scene["relation"] == "same" else -red_side
    offset = float(scene["dot_offset"])
    return (
        (red_path_x + red_side * offset, red_y),
        (blue_path_x + blue_side * offset, blue_y),
    )


def render(scene: dict) -> Image.Image:
    """Draw one unbroken edge-spanning trail and the two comparison dots."""
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    points = [(round(x * scale), round(y * scale)) for x, y in _trail_points(scene)]
    draw.line(
        points,
        fill=PATH_PALETTES[int(scene["path_palette_index"])],
        width=int(scene["stroke"] * scale),
        joint="curve",
    )
    radius = float(scene["dot_radius"])
    for (x, y), color in zip(_dot_positions(scene), (RED, BLUE), strict=True):
        draw.ellipse(
            (
                round((x - radius) * scale),
                round((y - radius) * scale),
                round((x + radius) * scale),
                round((y + radius) * scale),
            ),
            fill=color,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["relation"])


def margin(scene: dict) -> float:
    """Conservative minimum perpendicular clearance between dot and trail edges."""
    values = []
    for y in (float(scene["red_y"]), float(scene["blue_y"])):
        _, slope = _path_x_slope(scene, y)
        center_distance = float(scene["dot_offset"]) / math.sqrt(1.0 + slope * slope)
        values.append(
            center_distance - float(scene["dot_radius"]) - float(scene["stroke"]) / 2.0
        )
    return float(min(values))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
