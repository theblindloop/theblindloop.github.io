"""Deterministic crosshair-offset scene model and Pillow renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 480
COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (214, 55, 52),
    "blue": (42, 105, 190),
    "green": (35, 145, 78),
    "purple": (142, 68, 173),
}
REFERENCE_RGB = (92, 98, 105)


def _products(scene: dict) -> dict[str, float]:
    rx, ry = float(scene["reference_x"]), float(scene["reference_y"])
    return {
        color: abs(float(point[0]) - rx) * abs(float(point[1]) - ry)
        for color, point in zip(scene["colors"], scene["points"], strict=True)
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, well-separated scene with no one-axis winner shortcut."""
    rng = random.Random(int(seed))
    desired_winner = int(seed) % len(COLORS)
    quadrants = [(-1, -1), (1, -1), (-1, 1), (1, 1)]
    rng.shuffle(quadrants)
    reference_x = rng.randint(342, 378)
    reference_y = rng.randint(248, 278)

    # Role 0 has the largest product, role 1 the largest horizontal offset,
    # and role 2 the largest vertical offset. Rejection keeps the raster margin safe.
    for _ in range(500):
        role_offsets = [
            (rng.randint(104, 110), rng.randint(94, 100)),
            (rng.randint(137, 143), rng.randint(64, 68)),
            (rng.randint(69, 75), rng.randint(115, 121)),
            (rng.randint(84, 90), rng.randint(82, 88)),
        ]
        role_products = [x * y for x, y in role_offsets]
        ordered = sorted(role_products, reverse=True)
        if role_products[0] > role_products[1] and ordered[0] - ordered[1] >= 240:
            break
    else:
        raise RuntimeError("could not construct a safe product margin")

    remaining = [index for index in range(4) if index != desired_winner]
    rng.shuffle(remaining)
    roles_by_color = [None] * 4
    roles_by_color[desired_winner] = 0
    for color_index, role in zip(remaining, (1, 2, 3), strict=True):
        roles_by_color[color_index] = role

    points: list[list[int]] = []
    for color_index, role in enumerate(roles_by_color):
        dx, dy = role_offsets[int(role)]
        sx, sy = quadrants[color_index]
        points.append([reference_x + sx * dx, reference_y + sy * dy])
    return {
        "colors": list(COLORS),
        "points": points,
        "reference_x": reference_x,
        "reference_y": reference_y,
    }


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    rx, ry = int(scene["reference_x"]), int(scene["reference_y"])

    draw.text(
        (WIDTH * scale // 2, 27 * scale),
        "Offset rectangles from a reference crosshair",
        fill=(31, 35, 39), font=_font(22 * scale, True), anchor="mm",
    )
    draw.text(
        (WIDTH * scale // 2, 59 * scale),
        "Area = horizontal distance × vertical distance",
        fill=(62, 67, 72), font=_font(15 * scale), anchor="mm",
    )

    def dashed_line(start: tuple[int, int], end: tuple[int, int]) -> None:
        x1, y1 = start
        x2, y2 = end
        length = int(round(math.hypot(x2 - x1, y2 - y1)))
        for pos in range(0, length + 1, 21):
            stop = min(pos + 12, length)
            ax = x1 + (x2 - x1) * pos / length
            ay = y1 + (y2 - y1) * pos / length
            bx = x1 + (x2 - x1) * stop / length
            by = y1 + (y2 - y1) * stop / length
            draw.line((ax * scale, ay * scale, bx * scale, by * scale), fill=REFERENCE_RGB, width=3 * scale)

    dashed_line((66, ry), (654, ry))
    dashed_line((rx, 94), (rx, 438))
    draw.text((rx * scale + 9 * scale, 446 * scale), "reference", fill=(69, 74, 79), font=_font(12 * scale), anchor="la")

    for color, point in zip(scene["colors"], scene["points"], strict=True):
        cx, cy = map(int, point)
        radius = 12
        draw.ellipse(
            ((cx - radius) * scale, (cy - radius) * scale, (cx + radius) * scale, (cy + radius) * scale),
            fill=RGB[color], outline=(255, 255, 255), width=2 * scale,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _products(scene)
    ordered = sorted(scores, key=scores.get, reverse=True)
    if scores[ordered[0]] == scores[ordered[1]]:
        return "abstain"
    return ordered[0]


def margin(scene: dict) -> float:
    values = sorted(_products(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    rx, ry = float(scene["reference_x"]), float(scene["reference_y"])
    points = [(float(x), float(y)) for x, y in scene["points"]]
    line_clearance = min(min(abs(x - rx), abs(y - ry)) for x, y in points)
    point_clearance = min(
        math.hypot(a[0] - b[0], a[1] - b[1])
        for index, a in enumerate(points) for b in points[index + 1:]
    )
    return margin(scene) < 200.0 or line_clearance < 45.0 or point_clearance < 42.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
