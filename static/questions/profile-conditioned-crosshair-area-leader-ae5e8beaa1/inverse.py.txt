"""Independent image-only inverse for the crosshair offset-product plot."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (214, 55, 52),
    "blue": (42, 105, 190),
    "green": (35, 145, 78),
    "purple": (142, 68, 173),
}
_REFERENCE = np.asarray((92, 98, 105), dtype=np.int32)


def _point_centers(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    centers: dict[str, tuple[float, float]] = {}
    work = arr.astype(np.int32)
    for name, rgb in _PALETTE.items():
        delta = work - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2, dtype=np.int32) < 1600
        ys, xs = np.nonzero(mask)
        if len(xs) < 180:
            raise ValueError(f"missing or damaged {name} point")
        if int(xs.max() - xs.min()) > 30 or int(ys.max() - ys.min()) > 30:
            raise ValueError(f"invalid {name} component geometry")
        centers[name] = (float(xs.mean()), float(ys.mean()))
    return centers


def _reference_crosshair(arr: np.ndarray) -> tuple[float, float]:
    work = arr.astype(np.int32)
    delta = work - _REFERENCE
    neutral = np.sum(delta * delta, axis=2, dtype=np.int32) < 750
    # Long repeated neutral runs distinguish the crosshair from text glyphs.
    column_scores = neutral.sum(axis=0)
    row_scores = neutral.sum(axis=1)
    columns = np.flatnonzero(column_scores > 115)
    rows = np.flatnonzero(row_scores > 190)
    if len(columns) < 2 or len(rows) < 2:
        raise ValueError("missing or damaged dashed reference crosshair")
    if int(columns.max() - columns.min()) > 8 or int(rows.max() - rows.min()) > 8:
        raise ValueError("ambiguous reference crosshair geometry")
    return float(columns.mean()), float(rows.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 400 or arr.shape[1] < 600:
        raise ValueError("image is not the declared crosshair plot raster")
    centers = _point_centers(arr)
    rx, ry = _reference_crosshair(arr)
    scores = {
        color: abs(center[0] - rx) * abs(center[1] - ry)
        for color, center in centers.items()
    }
    ordered = sorted(scores, key=scores.get, reverse=True)
    if scores[ordered[0]] - scores[ordered[1]] < 80.0:
        return "abstain"
    return ordered[0]
