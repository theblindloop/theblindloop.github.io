"""Render four colored-node graphs and expose their analytic correspondence."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 720
BACKGROUND = (248, 247, 242)
INK = (42, 47, 55)
NEUTRAL = (112, 118, 126)
NODE_COLORS = {
    "cyan": (20, 156, 190),
    "magenta": (190, 54, 132),
    "orange": (232, 132, 34),
    "green": (59, 153, 88),
}
BORDER_COLORS = {
    "red": (207, 56, 63),
    "blue": (54, 102, 190),
    "gold": (218, 164, 35),
}
NODES = tuple(NODE_COLORS)
BOX_CENTERS = ((250, 210), (710, 210), (250, 515), (710, 515))
BOX_HALF = (175, 115)
BASE_OFFSETS = ((-92, -52), (92, -52), (92, 52), (-92, 52))


def _canon(edges: Iterable[Iterable[str]]) -> tuple[tuple[str, str], ...]:
    return tuple(sorted(tuple(sorted(pair)) for pair in edges))


def _path_edges(order: list[str] | tuple[str, ...]) -> list[list[str]]:
    return [[order[i], order[i + 1]] for i in range(3)]


def _all_paths() -> list[list[list[str]]]:
    unique: dict[tuple[tuple[str, str], ...], list[list[str]]] = {}
    for order in itertools.permutations(NODES):
        edges = _path_edges(order)
        unique.setdefault(_canon(edges), edges)
    return [unique[key] for key in sorted(unique)]


def _layout(rng: random.Random) -> dict[str, list[int]]:
    offsets = list(BASE_OFFSETS)
    rng.shuffle(offsets)
    # Small visible translation variation prevents exact coordinate templates.
    dx, dy = rng.randint(-12, 12), rng.randint(-10, 10)
    return {name: [x + dx, y + dy] for name, (x, y) in zip(NODES, offsets)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    paths = _all_paths()
    reference_edges = paths[rng.randrange(len(paths))]
    ref_key = _canon(reference_edges)
    alternatives = [
        edges
        for edges in paths
        if len(set(ref_key).symmetric_difference(_canon(edges))) >= 2
    ]
    rng.shuffle(alternatives)
    match_index = (int(seed) // 3) % 3
    candidate_edges = [alternatives.pop(), alternatives.pop(), alternatives.pop()]
    candidate_edges[match_index] = [list(pair) for pair in reference_edges]

    answer_border = ("red", "blue", "gold")[int(seed) % 3]
    other_borders = [name for name in BORDER_COLORS if name != answer_border]
    rng.shuffle(other_borders)
    candidate_borders: list[str] = []
    for index in range(3):
        candidate_borders.append(
            answer_border if index == match_index else other_borders.pop()
        )
    return {
        "reference_layout": _layout(rng),
        "reference_edges": reference_edges,
        "candidate_layouts": [_layout(rng) for _ in range(3)],
        "candidate_edges": candidate_edges,
        "candidate_borders": candidate_borders,
    }


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _draw_box(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    layout: dict[str, list[int]],
    edges: list[list[str]],
    border: tuple[int, int, int],
) -> None:
    cx, cy = center
    hx, hy = BOX_HALF
    draw.rounded_rectangle(
        (cx - hx, cy - hy, cx + hx, cy + hy),
        radius=17,
        fill=(255, 255, 255),
        outline=border,
        width=9,
    )
    points = {name: (cx + xy[0], cy + xy[1]) for name, xy in layout.items()}
    for left, right in edges:
        draw.line((points[left], points[right]), fill=INK, width=8)
    radius = 18
    for name in NODES:
        x, y = points[name]
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=NODE_COLORS[name],
            outline=(255, 255, 255),
            width=3,
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font, label_font, small_font = _font(27, True), _font(18, True), _font(15)
    draw.text((WIDTH // 2, 21), "MATCH THE COLORED CONNECTIONS", font=title_font, fill=INK, anchor="ma")
    draw.text(
        (WIDTH // 2, 57),
        "Same connected color pairs — rotation and rearrangement do not matter",
        font=small_font,
        fill=(82, 88, 96),
        anchor="ma",
    )
    _draw_box(draw, BOX_CENTERS[0], scene["reference_layout"], scene["reference_edges"], NEUTRAL)
    draw.text((BOX_CENTERS[0][0], 80), "REFERENCE", font=label_font, fill=NEUTRAL, anchor="ma")
    for index in range(3):
        center = BOX_CENTERS[index + 1]
        border_name = scene["candidate_borders"][index]
        _draw_box(
            draw,
            center,
            scene["candidate_layouts"][index],
            scene["candidate_edges"][index],
            BORDER_COLORS[border_name],
        )
    draw.text((WIDTH // 2, 665), "Answer with the border color:  red   •   blue   •   gold", font=_font(19, True), fill=INK, anchor="ma")
    return image


def analytic_gold(scene: dict) -> str:
    target = _canon(scene["reference_edges"])
    matches = [index for index, edges in enumerate(scene["candidate_edges"]) if _canon(edges) == target]
    if len(matches) != 1:
        return "abstain"
    return str(scene["candidate_borders"][matches[0]])


def margin(scene: dict) -> float:
    target = set(_canon(scene["reference_edges"]))
    distances = sorted(len(target.symmetric_difference(set(_canon(edges)))) for edges in scene["candidate_edges"])
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    target = _canon(scene["reference_edges"])
    matches = sum(_canon(edges) == target for edges in scene["candidate_edges"])
    return matches != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "reference_layout": scene["reference_layout"],
        "reference_edges": list(reversed(scene["reference_edges"])),
        "candidate_layouts": scene["candidate_layouts"],
        "candidate_edges": [list(reversed(edges)) for edges in scene["candidate_edges"]],
        "candidate_borders": scene["candidate_borders"],
    }
    return [("reverse_same_color_edge_draw_order", twin)]
