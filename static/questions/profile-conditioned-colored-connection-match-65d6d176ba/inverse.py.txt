"""Independent pixel-only inverse for colored connection correspondence."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

NODE_RGB = {
    "cyan": (20, 156, 190),
    "magenta": (190, 54, 132),
    "orange": (232, 132, 34),
    "green": (59, 153, 88),
}
BORDER_RGB = {
    "reference": (112, 118, 126),
    "red": (207, 56, 63),
    "blue": (54, 102, 190),
    "gold": (218, 164, 35),
}
INK = np.array((42, 47, 55), dtype=np.int16)


class OracleError(ValueError):
    pass


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 18.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.array(rgb, dtype=np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2)) <= tolerance


def _bounds_for_color(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    mask = _mask(arr, rgb)
    height, width = mask.shape
    seen = np.zeros_like(mask)
    largest: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(component) > len(largest):
            largest = component
    if len(largest) < 500:
        raise OracleError("border color is missing or too small")
    ys = np.array([point[0] for point in largest])
    xs = np.array([point[1] for point in largest])
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 250:
        raise OracleError("colored node is missing")
    return float(xs.mean()), float(ys.mean())


def _edge_present(dark: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    ax, ay = a
    bx, by = b
    hits = 0
    samples = 45
    height, width = dark.shape
    for t in np.linspace(0.22, 0.78, samples):
        x = int(round(ax + t * (bx - ax)))
        y = int(round(ay + t * (by - ay)))
        x0, x1 = max(0, x - 2), min(width, x + 3)
        y0, y1 = max(0, y - 2), min(height, y + 3)
        hits += bool(dark[y0:y1, x0:x1].any())
    return hits / samples >= 0.72


def _graph_in_box(arr: np.ndarray, bounds: tuple[int, int, int, int]) -> frozenset[tuple[str, str]]:
    x0, y0, x1, y1 = bounds
    pad = 13
    region = np.zeros(arr.shape[:2], dtype=bool)
    region[y0 + pad : y1 - pad + 1, x0 + pad : x1 - pad + 1] = True
    centers = {name: _centroid(_mask(arr, rgb) & region) for name, rgb in NODE_RGB.items()}
    delta = arr.astype(np.int32) - INK.astype(np.int32)
    dark = (np.sqrt(np.sum(delta * delta, axis=2)) <= 20.0) & region
    edges = []
    for left, right in itertools.combinations(sorted(NODE_RGB), 2):
        if _edge_present(dark, centers[left], centers[right]):
            edges.append((left, right))
    if len(edges) != 3:
        raise OracleError(f"expected three visible connections, recovered {len(edges)}")
    return frozenset(edges)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    boxes = {name: _bounds_for_color(arr, rgb) for name, rgb in BORDER_RGB.items()}
    reference = _graph_in_box(arr, boxes["reference"])
    matches = [name for name in ("red", "blue", "gold") if _graph_in_box(arr, boxes[name]) == reference]
    if len(matches) != 1:
        raise OracleError(f"expected one matching border, found {matches}")
    return matches[0]
