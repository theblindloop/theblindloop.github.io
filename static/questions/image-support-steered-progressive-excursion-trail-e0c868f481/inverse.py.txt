"""Independent pixel-only inverse for the progressive excursion trail."""

from __future__ import annotations

import numpy as np
from PIL import Image


START = np.asarray([210.0, 63.0, 65.0])
END = np.asarray([38.0, 151.0, 102.0])
TRAIL = np.asarray([35.0, 79.0, 126.0])


def _mask(array: np.ndarray, target: np.ndarray, tolerance: float = 38.0) -> np.ndarray:
    return np.linalg.norm(array.astype(np.float32) - target, axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> np.ndarray:
    points = np.argwhere(mask)
    if len(points) < 80:
        raise ValueError("endpoint evidence is absent")
    return points[:, ::-1].mean(axis=0)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    start = _centroid(_mask(array, START))
    end = _centroid(_mask(array, END))
    axis = end - start
    length = float(np.linalg.norm(axis))
    if not 135.0 <= length <= 160.0:
        raise ValueError("trail endpoints do not define the expected local baseline")
    unit = axis / length
    normal = np.asarray([-unit[1], unit[0]])

    trail_points = np.argwhere(_mask(array, TRAIL))[:, ::-1].astype(np.float64)
    if len(trail_points) < 500:
        raise ValueError("too little connected trail ink")
    relative = trail_points - start
    along = relative @ unit
    across = np.abs(relative @ normal)
    amplitudes = []
    for index in range(3):
        lo = length * index / 3.0 + 0.04 * length
        hi = length * (index + 1) / 3.0 - 0.04 * length
        values = across[(along >= lo) & (along <= hi)]
        if len(values) < 100:
            raise ValueError("an interior excursion is incomplete")
        amplitudes.append(float(np.percentile(values, 99.0)))

    first_gap = amplitudes[1] - amplitudes[0]
    second_gap = amplitudes[2] - amplitudes[1]
    if first_gap > 3.5 and second_gap > 3.5:
        return "grow"
    if first_gap < -3.5 and second_gap < -3.5:
        return "shrink"
    raise ValueError("excursion trend is ambiguous or damaged")
