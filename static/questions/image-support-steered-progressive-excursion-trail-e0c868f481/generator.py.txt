"""Deterministic local trail world.

The label is carried only by the order of three excursion amplitudes along one
connected trail.  Reversing that order changes the label while preserving the
amplitude multiset, total ink, count, scale, and palette.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    step = rng.randint(24, 25)
    low = rng.randint(3, 4)
    gap1 = 7
    gap2 = 7
    ordered = [low, low + gap1, low + gap1 + gap2]
    amplitudes = ordered if int(seed) % 2 == 0 else list(reversed(ordered))
    return {
        "width": 640,
        "height": 640,
        # Cell centers keep the narrow trail from straddling an arbitrary
        # ablation row while still randomizing its absolute location widely.
        "center_x": rng.choice([200, 280, 360, 440]),
        "center_y": rng.choice([200, 280, 360, 440]),
        "quarter_turn": rng.randrange(4),
        "reflect": bool(rng.randrange(2)),
        "step": step,
        "amplitudes": amplitudes,
        "stroke": 4,
        "trail_color": [35, 79, 126],
        "start_color": [210, 63, 65],
        "end_color": [38, 151, 102],
        "background": [247, 244, 236],
    }


def _transform(x: float, y: float, scene: dict) -> tuple[float, float]:
    if scene["reflect"]:
        y = -y
    turns = int(scene["quarter_turn"]) % 4
    for _ in range(turns):
        x, y = -y, x
    return x + float(scene["center_x"]), y + float(scene["center_y"])


def _points(scene: dict) -> list[tuple[float, float]]:
    step = float(scene["step"])
    amplitudes = [float(value) for value in scene["amplitudes"]]
    x = -3.0 * step
    points = [(x, 0.0)]
    for index, amplitude in enumerate(amplitudes):
        sign = -1.0 if index % 2 == 0 else 1.0
        points.append((x + step, sign * amplitude))
        x += 2.0 * step
        points.append((x, 0.0))
    return [_transform(x, y, scene) for x, y in points]


def render(scene: dict) -> Image.Image:
    # Every accessed field is an explicit part of the raster state.
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    stroke = int(scene["stroke"])
    trail = tuple(int(v) for v in scene["trail_color"])
    draw.line(points, fill=trail, width=stroke, joint="curve")

    radius = stroke + 3
    sx, sy = points[0]
    ex, ey = points[-1]
    start = tuple(int(v) for v in scene["start_color"])
    end = tuple(int(v) for v in scene["end_color"])
    draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=start)
    draw.ellipse((ex - radius, ey - radius, ex + radius, ey + radius), fill=end)
    return image


def analytic_gold(scene: dict) -> str:
    amplitudes = [float(value) for value in scene["amplitudes"]]
    if amplitudes[0] < amplitudes[1] < amplitudes[2]:
        return "grow"
    if amplitudes[0] > amplitudes[1] > amplitudes[2]:
        return "shrink"
    raise ValueError("amplitudes are not strictly monotone")


def margin(scene: dict) -> float:
    amplitudes = [float(value) for value in scene["amplitudes"]]
    return min(abs(amplitudes[1] - amplitudes[0]), abs(amplitudes[2] - amplitudes[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
