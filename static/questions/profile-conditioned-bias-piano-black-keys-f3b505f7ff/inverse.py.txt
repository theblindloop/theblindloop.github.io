"""Independent image-only inverse arm for raised black piano keys."""

from __future__ import annotations

from collections import deque

from PIL import Image


def _components(mask: list[bytearray]) -> list[tuple[int, tuple[int, int, int, int], bool]]:
    height = len(mask)
    width = len(mask[0]) if height else 0
    seen = [bytearray(width) for _ in range(height)]
    found: list[tuple[int, tuple[int, int, int, int], bool]] = []
    for y0 in range(height):
        for x0 in range(width):
            if not mask[y0][x0] or seen[y0][x0]:
                continue
            queue = deque([(x0, y0)])
            seen[y0][x0] = 1
            area = 0
            min_x = max_x = x0
            min_y = max_y = y0
            border = False
            while queue:
                x, y = queue.popleft()
                area += 1
                min_x, max_x = min(min_x, x), max(max_x, x)
                min_y, max_y = min(min_y, y), max(max_y, y)
                border = border or x == 0 or y == 0 or x == width - 1 or y == height - 1
                for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny][nx] and not seen[ny][nx]:
                        seen[ny][nx] = 1
                        queue.append((nx, ny))
            found.append((area, (min_x, min_y, max_x, max_y), border))
    return found


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 160 or height < 160:
        return "abstain"
    pixels = rgb.load()
    dark = [bytearray(width) for _ in range(height)]
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y]
            # The count label is public: raised keys are the only large,
            # near-neutral, near-black regions. Case, seams, and shadows are gray.
            dark[y][x] = int(max(r, g, b) < 78 and max(r, g, b) - min(r, g, b) < 24)

    candidates = []
    for area, (x0, y0, x1, y1), border in _components(dark):
        box_w = x1 - x0 + 1
        box_h = y1 - y0 + 1
        fill = area / float(box_w * box_h)
        if border or area < 1400:
            continue
        if not (18 <= box_w <= 36 and 80 <= box_h <= 132 and box_h >= 2.8 * box_w and fill >= 0.78):
            continue
        candidates.append((x0, x1, y0, y1))
    candidates.sort()
    if len(candidates) not in (4, 5, 6):
        return "abstain"
    # All counted keys must share the same upper row and remain visibly separate.
    if max(c[2] for c in candidates) - min(c[2] for c in candidates) > 3:
        return "abstain"
    if any(candidates[i + 1][0] - candidates[i][1] < 9 for i in range(len(candidates) - 1)):
        return "abstain"
    return str(len(candidates))
