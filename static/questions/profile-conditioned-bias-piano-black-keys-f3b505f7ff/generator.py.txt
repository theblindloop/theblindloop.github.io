"""Deterministic prior-conflict piano-key world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 440
SCALE = 4
QUARANTINE_GAP_PX = 10.0
DARK_PALETTE = ((19, 23, 29), (27, 30, 35), (31, 27, 32), (22, 29, 31))
WHITE_PALETTE = ((244, 241, 232), (247, 245, 238), (239, 237, 229))
BACKGROUND_PALETTE = ((218, 210, 196), (225, 216, 202), (211, 218, 217))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x51A90)
    count = int(seed) % 3 + 4
    white_width = rng.randint(47, 51)
    keyboard_x = rng.randint(35, CANVAS - 7 * white_width - 35)
    keyboard_y = rng.randint(72, 91)
    white_height = rng.randint(260, 278)
    # Hold total target-key area nearly constant across the three answer classes
    # so a global dark-pixel sum cannot substitute for component counting.
    target_dark_area = rng.randint(13800, 14600)
    width_ranges = {4: (29, 32), 5: (25, 30), 6: (21, 27)}
    black_width = rng.randint(*width_ranges[count])
    black_height = int(round(target_dark_area / (count * black_width)))

    # Seven white keys have six internal gaps. Five keys use the conventional
    # C-D-E / F-G-A-B pattern; four removes one conventional key; six fills the
    # normally empty E-F gap. The omission rotates deterministically.
    canonical = [1, 2, 4, 5, 6]
    if count == 4:
        omitted_index = rng.randrange(5)
        occupied = [gap for index, gap in enumerate(canonical) if index != omitted_index]
    elif count == 5:
        occupied = canonical
    else:
        occupied = [1, 2, 3, 4, 5, 6]
    return {
        "black_key_count": count,
        "occupied_gaps": occupied,
        "canvas_size": CANVAS,
        "keyboard_x": keyboard_x,
        "keyboard_y": keyboard_y,
        "white_key_width": white_width,
        "white_key_height": white_height,
        "black_key_width": black_width,
        "black_key_height": black_height,
        "dark_rgb": list(DARK_PALETTE[rng.randrange(len(DARK_PALETTE))]),
        "white_rgb": list(WHITE_PALETTE[rng.randrange(len(WHITE_PALETTE))]),
        "background_rgb": list(BACKGROUND_PALETTE[rng.randrange(len(BACKGROUND_PALETTE))]),
    }


def _box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in values)  # type: ignore[return-value]


def _key_boxes(scene: dict[str, Any]) -> list[tuple[float, float, float, float]]:
    x = float(scene["keyboard_x"])
    y = float(scene["keyboard_y"])
    ww = float(scene["white_key_width"])
    bw = float(scene["black_key_width"])
    bh = float(scene["black_key_height"])
    return [(x + gap * ww - bw / 2, y, x + gap * ww + bw / 2, y + bh) for gap in scene["occupied_gaps"]]


def render(scene: dict[str, Any]) -> Image.Image:
    declared_count = int(scene["black_key_count"])
    if declared_count != len(scene["occupied_gaps"]):
        raise ValueError("black_key_count must equal the number of occupied gaps")
    size = int(scene["canvas_size"])
    x = float(scene["keyboard_x"])
    y = float(scene["keyboard_y"])
    ww = float(scene["white_key_width"])
    wh = float(scene["white_key_height"])
    dark = tuple(int(v) for v in scene["dark_rgb"])
    white = tuple(int(v) for v in scene["white_rgb"])
    background = tuple(int(v) for v in scene["background_rgb"])

    image = Image.new("RGB", (size * SCALE, size * SCALE), background)
    draw = ImageDraw.Draw(image)
    # A soft keyboard shadow and warm case make the object recognizable without
    # adding near-black components that the independent oracle could count.
    draw.rounded_rectangle(_box((x - 13, y - 18, x + 7 * ww + 13, y + wh + 17)), radius=9 * SCALE, fill=(121, 112, 102))
    for index in range(7):
        x0 = x + index * ww
        draw.rectangle(_box((x0, y, x0 + ww, y + wh)), fill=white, outline=(104, 102, 99), width=2 * SCALE)
        draw.line(_box((x0 + 5, y + wh - 8, x0 + ww - 5, y + wh - 8)), fill=(193, 188, 178), width=SCALE)

    # Raised keys are separated opaque components and cast identical soft shadows.
    for x0, y0, x1, y1 in _key_boxes(scene):
        draw.rounded_rectangle(_box((x0 + 3, y0 + 4, x1 + 5, y1 + 7)), radius=3 * SCALE, fill=(103, 96, 91))
        draw.rounded_rectangle(_box((x0, y0, x1, y1)), radius=2 * SCALE, fill=dark)
        draw.line(_box((x0 + 3, y1 - 5, x1 - 3, y1 - 5)), fill=(64, 67, 70), width=SCALE)

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["black_key_count"]))


def margin(scene: dict[str, Any]) -> float:
    boxes = _key_boxes(scene)
    if len(boxes) < 2:
        return float("inf")
    return min(boxes[index + 1][0] - boxes[index][2] for index in range(len(boxes) - 1))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_GAP_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
