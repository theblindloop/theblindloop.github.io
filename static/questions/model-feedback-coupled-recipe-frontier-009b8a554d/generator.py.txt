"""Render three workshops with coupled two-resource recipe allocation."""

from __future__ import annotations
import random
from PIL import Image, ImageDraw

CANVAS = (1200, 760)
ANSWERS = ("amber", "cyan", "violet")
RGB = {"background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
       "card": (232, 237, 242), "amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
ROW_Y = (25, 270, 515)
ROW_H = 220
STOCK_X = (150, 350)
RECIPE_X = (760, 1020)
RESOURCE_Y_OFF = (105, 170)
DIGIT_SEGMENTS = {0: "abcdef", 1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg", 5: "acdfg", 6: "acdefg", 7: "abc", 8: "abcdefg", 9: "abcdfg"}
BLOCK_FONT = {
    "A": ("01110","10001","10001","11111","10001","10001","10001"),
    "B": ("11110","10001","10001","11110","10001","10001","11110"),
    "C": ("01111","10000","10000","10000","10000","10000","01111"),
    "E": ("11111","10000","10000","11110","10000","10000","11111"),
    "I": ("111","010","010","010","010","010","111"),
    "K": ("10001","10010","10100","11000","10100","10010","10001"),
    "O": ("01110","10001","10001","10001","10001","10001","01110"),
    "P": ("11110","10001","10001","11110","10000","10000","10000"),
    "R": ("11110","10001","10001","11110","10100","10010","10001"),
    "S": ("11111","10000","10000","11111","00001","00001","11111"),
    "T": ("11111","00100","00100","00100","00100","00100","00100"),
    "X": ("10001","10001","01010","00100","01010","10001","10001"),
    "Y": ("10001","10001","01010","00100","00100","00100","00100"),
}


def _optimum(stocks: list[int], recipes: list[list[int]]) -> tuple[int, list[tuple[int, int]]]:
    best = -1; pairs: list[tuple[int, int]] = []
    for x in range(stocks[0] // recipes[0][0] + 1):
        for y in range(stocks[0] // recipes[1][0] + 1):
            if x * recipes[0][0] + y * recipes[1][0] <= stocks[0] and x * recipes[0][1] + y * recipes[1][1] <= stocks[1]:
                total = x + y
                if total > best: best, pairs = total, [(x, y)]
                elif total == best: pairs.append((x, y))
    return best, pairs


def _good_line(stocks: list[int], recipes: list[list[int]]) -> bool:
    if not (recipes[0][0] < recipes[1][0] and recipes[0][1] > recipes[1][1]): return False
    score, pairs = _optimum(stocks, recipes)
    pure = max(min(stocks[0] // recipes[i][0], stocks[1] // recipes[i][1]) for i in range(2))
    return 7 <= score <= 15 and score > pure and any(x > 0 and y > 0 for x, y in pairs)


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xC0A1ED) & 0xFFFFFFFFFFFFFFFF)
    for _ in range(20000):
        lines = []
        for _slot in range(3):
            for _attempt in range(500):
                ax = rng.randint(2, 5); ay = rng.randint(ax + 1, 9)
                by = rng.randint(2, 5); bx = rng.randint(by + 1, 9)
                stocks = [rng.randint(30, 75), rng.randint(30, 75)]
                recipes = [[ax, bx], [ay, by]]
                if _good_line(stocks, recipes):
                    lines.append((stocks, recipes)); break
        if len(lines) != 3: continue
        scores = [_optimum(s, r)[0] for s, r in lines]
        ordered = sorted(scores, reverse=True)
        if len(set(scores)) != 3 or ordered[0] - ordered[1] != 1 or ordered[0] - ordered[2] > 3: continue
        winner = scores.index(max(scores)); answer_color = ANSWERS[(int(seed) // 3) % 3]
        others = [c for c in ANSWERS if c != answer_color]; rng.shuffle(others)
        colors: list[str | None] = [None, None, None]; colors[winner] = answer_color
        for i in range(3):
            if colors[i] is None: colors[i] = others.pop()
        scene = {"stocks": [x[0] for x in lines], "recipes": [x[1] for x in lines], "tag_colors": colors}
        if not is_quarantined(scene): return scene
    raise RuntimeError("could not construct coupled recipe scene")


def _draw_digit(draw: ImageDraw.ImageDraw, digit: int, x: int, y: int) -> None:
    boxes = {"a":(x+4,y,x+19,y+4),"b":(x+19,y+4,x+23,y+21),"c":(x+19,y+23,x+23,y+40),
             "d":(x+4,y+40,x+19,y+44),"e":(x,y+23,x+4,y+40),"f":(x,y+4,x+4,y+21),"g":(x+4,y+20,x+19,y+24)}
    for segment in DIGIT_SEGMENTS[int(digit)]: draw.rectangle(boxes[segment], fill=RGB["ink"])


def _draw_number(draw: ImageDraw.ImageDraw, value: int, cx: int, cy: int) -> None:
    tens, ones = divmod(int(value), 10); y = cy - 22
    _draw_digit(draw, tens, cx - 27, y); _draw_digit(draw, ones, cx + 4, y)


def _block_text(draw: ImageDraw.ImageDraw, text: str, cx: int, top: int, scale: int = 3) -> None:
    widths = [len(BLOCK_FONT[ch][0]) * scale for ch in text]; x = cx - (sum(widths) + max(0, len(text)-1)*2*scale)//2
    for ch, width in zip(text, widths):
        for row, bits in enumerate(BLOCK_FONT[ch]):
            for col, bit in enumerate(bits):
                if bit == "1": draw.rectangle((x+col*scale,top+row*scale,x+(col+1)*scale-1,top+(row+1)*scale-1), fill=RGB["ink"])
        x += width + 2*scale


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"]); draw = ImageDraw.Draw(image)
    for y, stocks, recipes, color in zip(ROW_Y, scene["stocks"], scene["recipes"], scene["tag_colors"]):
        draw.rounded_rectangle((18,y,1182,y+ROW_H), radius=22, fill=RGB["panel"], outline=RGB[str(color)], width=10)
        _block_text(draw, "STOCK", 250, y+17, 2); _block_text(draw, "RECIPE", 890, y+17, 2)
        for cx, label, stock in zip(STOCK_X, ("A","B"), stocks):
            draw.rounded_rectangle((cx-70,y+65,cx+70,y+202), radius=14, fill=RGB["card"], outline=RGB["ink"], width=5)
            _block_text(draw, label, cx, y+76, 3); _draw_number(draw, int(stock), cx, y+155)
        draw.line((505,y+58,505,y+205), fill=RGB["ink"], width=5)
        _block_text(draw, "A", 590, y+87, 3); _block_text(draw, "B", 590, y+152, 3)
        for cx, label, recipe in zip(RECIPE_X, ("X","Y"), recipes):
            _block_text(draw, label, cx, y+51, 3)
            for cy, value in zip((y+RESOURCE_Y_OFF[0], y+RESOURCE_Y_OFF[1]), recipe):
                draw.rounded_rectangle((cx-62,cy-28,cx+62,cy+28), radius=10, fill=RGB["background"], outline=RGB["ink"], width=4)
                _draw_digit(draw, int(value), cx-11, cy-22)
    return image


def analytic_gold(scene: dict) -> str:
    if len(scene["stocks"]) != 3 or len(scene["recipes"]) != 3 or set(scene["tag_colors"]) != set(ANSWERS): raise ValueError("malformed workshop binding")
    scores = [_optimum(s, r)[0] for s, r in zip(scene["stocks"], scene["recipes"])]
    if scores.count(max(scores)) != 1: raise ValueError("workshop optima are tied")
    return str(scene["tag_colors"][scores.index(max(scores))])


def margin(scene: dict) -> float:
    scores = sorted((_optimum(s, r)[0] for s, r in zip(scene["stocks"], scene["recipes"])), reverse=True)
    return float((scores[0]-scores[1])/scores[1]) if len(scores)==3 and scores[0]>scores[1] else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["stocks"]) != 3 or len(scene["recipes"]) != 3 or set(scene["tag_colors"]) != set(ANSWERS): return True
        if any(not _good_line(s, r) for s, r in zip(scene["stocks"], scene["recipes"])): return True
        scores = sorted((_optimum(s, r)[0] for s, r in zip(scene["stocks"], scene["recipes"])), reverse=True)
        return len(set(scores)) != 3 or scores[0]-scores[1] != 1 or scores[0]-scores[2] > 3
    except (KeyError, TypeError, ValueError, ZeroDivisionError): return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]: return []
