"""Independent image-only inverse for coupled recipe frontiers."""

from __future__ import annotations
import numpy as np
from PIL import Image

CANVAS=(1200,760); INK=(31,38,46); BACKGROUND=(255,255,255); PANEL=(247,249,252); CARD=(232,237,242)
TAG_RGB={"amber":(226,151,28),"cyan":(18,157,181),"violet":(141,77,190)}
ALLOWED={INK,BACKGROUND,PANEL,CARD,*TAG_RGB.values()}; ROW_Y=(25,270,515); STOCK_X=(150,350); RECIPE_X=(760,1020); RESOURCE_Y_OFF=(105,170)
POINTS={"a":(11,2),"b":(21,12),"c":(21,32),"d":(11,42),"e":(2,32),"f":(2,12),"g":(11,22)}
DIGITS={frozenset("abcdef"):0,frozenset("bc"):1,frozenset("abdeg"):2,frozenset("abcdg"):3,frozenset("bcfg"):4,frozenset("acdfg"):5,frozenset("acdefg"):6,frozenset("abc"):7,frozenset("abcdefg"):8,frozenset("abcdfg"):9}

class OracleError(ValueError): pass

def _digit(arr: np.ndarray, x: int, y: int) -> int:
    active=set()
    for name,(dx,dy) in POINTS.items():
        patch=arr[y+dy-1:y+dy+2,x+dx-1:x+dx+2]
        if patch.shape!=(3,3,3): raise OracleError("digit outside canvas")
        count=int(np.all(patch==INK,axis=2).sum())
        if count>=7: active.add(name)
        elif count!=0: raise OracleError("damaged digit stroke")
    try: return DIGITS[frozenset(active)]
    except KeyError as exc: raise OracleError("invalid digit") from exc

def _stock(arr: np.ndarray, cx: int, y: int) -> int:
    cy=y+155; value=10*_digit(arr,cx-27,cy-22)+_digit(arr,cx+4,cy-22)
    if not 30<=value<=75: raise OracleError("stock outside range")
    patch=arr[y+65:y+203,cx-70:cx+71]
    if int(np.all(patch==CARD,axis=2).sum())<9000: raise OracleError("stock card damaged")
    return value

def _recipe(arr: np.ndarray, cx: int, y: int) -> list[int]:
    values=[]
    for off in RESOURCE_Y_OFF:
        cy=y+off; values.append(_digit(arr,cx-11,cy-22))
        patch=arr[cy-28:cy+29,cx-62:cx+63]
        if int(np.all(patch==BACKGROUND,axis=2).sum())<4500: raise OracleError("recipe card damaged")
    if any(v<2 or v>9 for v in values): raise OracleError("recipe cost outside range")
    return values

def _color(arr: np.ndarray, y: int) -> str:
    counts={name:int(np.all(arr[y:y+12,200:1000]==rgb,axis=2).sum()) for name,rgb in TAG_RGB.items()}
    names=[name for name,count in counts.items() if count>6000]
    if len(names)!=1: raise OracleError("workshop border ambiguous")
    return names[0]

def _score(stocks: list[int], recipes: list[list[int]]) -> int:
    best=-1
    for x in range(stocks[0]//recipes[0][0]+1):
        for y in range(stocks[0]//recipes[1][0]+1):
            if x*recipes[0][0]+y*recipes[1][0]<=stocks[0] and x*recipes[0][1]+y*recipes[1][1]<=stocks[1]: best=max(best,x+y)
    return best

def decision_from_image(image: Image.Image) -> str:
    rgb=image.convert("RGB")
    if rgb.size!=CANVAS: raise OracleError("wrong canvas")
    colors=rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(c) for _,c in colors}<=ALLOWED: raise OracleError("unknown raster color")
    arr=np.asarray(rgb); seen=[]; scores=[]
    for y in ROW_Y:
        seen.append(_color(arr,y)); stocks=[_stock(arr,x,y) for x in STOCK_X]; recipes=[_recipe(arr,x,y) for x in RECIPE_X]
        if not (recipes[0][0]<recipes[1][0] and recipes[0][1]>recipes[1][1]): raise OracleError("recipes lack visible tradeoff")
        scores.append(_score(stocks,recipes))
    if set(seen)!=set(TAG_RGB): raise OracleError("colors are not a permutation")
    if scores.count(max(scores))!=1: raise OracleError("scores tied")
    return seen[scores.index(max(scores))]
