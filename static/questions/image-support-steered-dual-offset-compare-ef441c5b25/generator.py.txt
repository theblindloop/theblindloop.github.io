"""Dual-offset binding world.

Latent z = {a_box, b_box, a_mx, b_mx}: two colored tracks placed in distant,
disconnected image regions (red top-left, blue bottom-right), each bearing a
black marker at a fractional x-offset within its own track. The decision is a
cross-component comparison of the two normalized fractional offsets: which
marker sits proportionally farther right within its own track.

This makes the decisive support spatially distributed (the two tracks are far
apart) and multipart (both tracks must be bound and compared; each contributes
to the answer). There is no decisive path joining them.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "dual_offset_compare-0.1.0"

W, H = 512, 512
TRACK_H = 26
MARKER_R = 11
RED = (220, 30, 30)
BLUE = (30, 60, 220)
BLACK = (0, 0, 0)
QUARANTINE_MARGIN = 0.13


def _frac(scene: dict, box_key: str, mx_key: str) -> float:
    box = scene[box_key]
    mx = scene[mx_key]
    return (mx - box[0]) / (box[2] - box[0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        a_w = rng.uniform(170, 240)
        a_box = [
            rng.uniform(20, 150),
            rng.uniform(30, 120),
            0.0,
            0.0,
        ]
        a_box[2] = a_box[0] + a_w
        a_box[3] = a_box[1] + TRACK_H

        b_w = rng.uniform(170, 240)
        b_x0 = rng.uniform(240, W - b_w - 12)
        b_box = [b_x0, rng.uniform(360, 450), b_x0 + b_w, 0.0]
        b_box[3] = b_box[1] + TRACK_H

        fa = rng.uniform(0.16, 0.84)
        fb = rng.uniform(0.16, 0.84)
        a_mx = a_box[0] + fa * (a_box[2] - a_box[0])
        b_mx = b_box[0] + fb * (b_box[2] - b_box[0])

        scene = {"a_box": a_box, "b_box": b_box, "a_mx": a_mx, "b_mx": b_mx}
        if margin(scene) >= QUARANTINE_MARGIN:
            return scene


def _marker(draw, mx: float, cy: float, s: int) -> None:
    r = MARKER_R * s
    box = [(mx - MARKER_R) * s, (cy - MARKER_R) * s, (mx + MARKER_R) * s, (cy + MARKER_R) * s]
    draw.ellipse(box, fill=BLACK)


def render(scene: dict, supersample: int = 3) -> Image.Image:
    s = supersample
    img = Image.new("RGB", (W * s, H * s), "white")
    d = ImageDraw.Draw(img)
    a = scene["a_box"]
    b = scene["b_box"]
    d.rectangle([a[0] * s, a[1] * s, a[2] * s, a[3] * s], fill=RED)
    d.rectangle([b[0] * s, b[1] * s, b[2] * s, b[3] * s], fill=BLUE)
    _marker(d, scene["a_mx"], (a[1] + a[3]) / 2.0, s)
    _marker(d, scene["b_mx"], (b[1] + b[3]) / 2.0, s)
    return img.resize((W, H), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    a_box = scene["a_box"]
    b_box = scene["b_box"]
    a_mx = scene["a_mx"]
    b_mx = scene["b_mx"]
    fa = (a_mx - a_box[0]) / (a_box[2] - a_box[0])
    fb = (b_mx - b_box[0]) / (b_box[2] - b_box[0])
    return "red" if fa > fb else "blue"


def margin(scene: dict) -> float:
    a_box = scene["a_box"]
    b_box = scene["b_box"]
    a_mx = scene["a_mx"]
    b_mx = scene["b_mx"]
    fa = (a_mx - a_box[0]) / (a_box[2] - a_box[0])
    fb = (b_mx - b_box[0]) / (b_box[2] - b_box[0])
    return abs(fa - fb)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    return iter(())
