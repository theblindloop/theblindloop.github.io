"""Independent pixel-only inverse arm for the dual-offset binding world.

Receives only the final RGB PNG. Segments the two colored tracks by color, finds
each track's black marker centroid, computes each marker's normalized fractional
offset within its own track, and compares the two fractions. Must never read
renderer, prompts, generate, verify, or analytic-gold code or latent scene data.

Abstains (raises) when either track or either marker cannot be recovered.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _red_mask(arr: np.ndarray) -> np.ndarray:
    return (arr[..., 0] > 120) & (arr[..., 1] < 80) & (arr[..., 2] < 80)


def _blue_mask(arr: np.ndarray) -> np.ndarray:
    return (arr[..., 2] > 120) & (arr[..., 1] < 90) & (arr[..., 0] < 90)


def _black_mask(arr: np.ndarray) -> np.ndarray:
    return (arr[..., 0] < 80) & (arr[..., 1] < 80) & (arr[..., 2] < 80)


def _bbox(mask: np.ndarray) -> tuple[float, float, float, float]:
    rows = np.flatnonzero(mask.any(axis=1))
    cols = np.flatnonzero(mask.any(axis=0))
    if not len(rows) or not len(cols):
        raise ValueError("no track pixels found")
    return float(cols[0]), float(rows[0]), float(cols[-1]), float(rows[-1])


def _marker_center(arr: np.ndarray, black: np.ndarray, box: tuple, margin_px: float) -> float:
    left, top, right, bottom = box
    band = (
        black
        & (np.arange(arr.shape[0])[:, None] >= top)
        & (np.arange(arr.shape[0])[:, None] <= bottom)
        & (np.arange(arr.shape[1])[None, :] >= left + margin_px)
        & (np.arange(arr.shape[1])[None, :] <= right - margin_px)
    )
    ys, xs = np.nonzero(band)
    if not len(xs):
        raise ValueError("marker not found within track")
    return float(xs.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red = _red_mask(arr)
    blue = _blue_mask(arr)
    black = _black_mask(arr)

    a_box = _bbox(red)
    b_box = _bbox(blue)

    if not a_box[0] <= b_box[0]:
        pass
    a_mx = _marker_center(arr, black, a_box, 4.0)
    b_mx = _marker_center(arr, black, b_box, 4.0)

    fa = (a_mx - a_box[0]) / (a_box[2] - a_box[0])
    fb = (b_mx - b_box[0]) / (b_box[2] - b_box[0])
    return "red" if fa - fb > 0.0 else "blue"
