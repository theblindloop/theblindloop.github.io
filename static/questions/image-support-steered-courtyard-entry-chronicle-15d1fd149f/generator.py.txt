"""Deterministic renderer for a compact four-snapshot region-entry chronicle."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
TABLET = 54
PANEL = 21
BORDER = 4
GUTTER = 4

BACKGROUND = (247, 244, 235)
NAVY = (29, 45, 68)
PANEL_BG = (255, 252, 237)
GOLD = (238, 183, 52)
BEAD_RGB = {"red": (220, 55, 65), "green": (35, 164, 105), "blue": (42, 105, 210)}
DECOR = (207, 210, 207)


def _first_entries(states: list[list[int]]) -> list[int | None]:
    out: list[int | None] = []
    for history in states:
        out.append(next((t for t in range(1, 4) if history[t - 1] == 0 and history[t] == 1), None))
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = ["red", "green", "blue"]
    winner = int(seed) % 3
    entry = 1 + (int(seed) // 3) % 3

    states = [[0, 0, 0, 0] for _ in colors]
    for t in range(entry, 4):
        states[winner][t] = 1 if t == entry else rng.randint(0, 1)
    if entry < 3:
        follower = (winner + 1 + rng.randrange(2)) % 3
        states[follower][entry + 1] = 1
        for t in range(entry + 2, 4):
            states[follower][t] = rng.randint(0, 1)
    for idx in range(3):
        if idx == winner:
            continue
        for t in range(entry + 1, 4):
            if states[idx][t] == 0 and rng.random() < 0.28:
                states[idx][t] = 1

    inside_slots = [(8, 8), (12, 8), (10, 12)]
    outside_slots = [(3, 3), (10, 3), (17, 3), (3, 10), (17, 10), (3, 17), (10, 17), (17, 17)]
    positions: list[list[list[int]]] = []
    for frame in range(4):
        ins = inside_slots[:]
        outs = outside_slots[:]
        rng.shuffle(ins)
        rng.shuffle(outs)
        frame_positions = []
        for color_index in range(3):
            slot = ins.pop() if states[color_index][frame] else outs.pop()
            frame_positions.append([slot[0], slot[1]])
        positions.append(frame_positions)

    tablet_x = rng.randint(58, WIDTH - TABLET - 58)
    tablet_y = rng.randint(58, HEIGHT - TABLET - 58)
    decorations: list[list[int]] = []
    for _ in range(18):
        for _attempt in range(200):
            x, y = rng.randint(18, WIDTH - 19), rng.randint(18, HEIGHT - 19)
            if not (tablet_x - 28 <= x <= tablet_x + TABLET + 28 and tablet_y - 28 <= y <= tablet_y + TABLET + 28):
                decorations.append([x, y, rng.choice([2, 3, 4])])
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "tablet_x": tablet_x,
        "tablet_y": tablet_y,
        "colors": colors,
        "states": states,
        "positions": positions,
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    tx, ty = int(scene["tablet_x"]), int(scene["tablet_y"])
    colors = list(scene["colors"])
    states = scene["states"]
    positions = scene["positions"]
    decorations = scene["decorations"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, radius in decorations:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=DECOR, width=1)

    draw.rectangle((tx, ty, tx + TABLET - 1, ty + TABLET - 1), fill=NAVY)
    for frame in range(4):
        row, col = divmod(frame, 2)
        px = tx + BORDER + col * (PANEL + GUTTER)
        py = ty + BORDER + row * (PANEL + GUTTER)
        draw.rectangle((px, py, px + PANEL - 1, py + PANEL - 1), fill=PANEL_BG)
        draw.rectangle((px + 6, py + 6, px + 14, py + 14), fill=GOLD)
        for color_index, color in enumerate(colors):
            bx, by = positions[frame][color_index]
            expected_inside = 6 <= bx <= 14 and 6 <= by <= 14
            if bool(states[color_index][frame]) != expected_inside:
                raise ValueError("state and rendered bead position disagree")
            draw.rectangle((px + bx - 1, py + by - 1, px + bx + 1, py + by + 1), fill=BEAD_RGB[color])
    return image


def analytic_gold(scene: dict) -> str:
    states = scene["states"]
    colors = scene["colors"]
    entries = _first_entries(states)
    valid = [(step, idx) for idx, step in enumerate(entries) if step is not None]
    if not valid:
        raise ValueError("no entry event")
    first = min(step for step, _ in valid)
    winners = [idx for step, idx in valid if step == first]
    if len(winners) != 1:
        raise ValueError("ambiguous earliest entry")
    return str(colors[winners[0]])


def margin(scene: dict) -> float:
    entries = sorted(step for step in _first_entries(scene["states"]) if step is not None)
    if not entries:
        return 0.0
    runner = entries[1] if len(entries) > 1 else 4
    return float(runner - entries[0])


def is_quarantined(scene: dict) -> bool:
    entries = _first_entries(scene["states"])
    present = [step for step in entries if step is not None]
    return not present or present.count(min(present)) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
