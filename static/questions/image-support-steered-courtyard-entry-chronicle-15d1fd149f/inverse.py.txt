"""Independent pixel-only inverse arm for the courtyard chronicle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = (29, 45, 68)
_PANEL = (255, 252, 237)
_GOLD = (238, 183, 52)
_COLORS = {"red": (220, 55, 65), "green": (35, 164, 105), "blue": (42, 105, 210)}


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    result: list[np.ndarray] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=int))
    return result


def _bbox(points: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = points[:, 0], points[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy = _mask(array, _NAVY)
    navy_components = [part for part in _components(navy) if len(part) >= 100]
    if len(navy_components) != 1 or len(navy_components[0]) != 1152:
        raise ValueError("chronicle tablet is absent or damaged")
    left, top, right, bottom = _bbox(navy_components[0])
    if right - left + 1 != 54 or bottom - top + 1 != 54:
        raise ValueError("chronicle tablet geometry is invalid")

    panel_parts = []
    for part in _components(_mask(array, _PANEL)):
        x0, y0, x1, y1 = _bbox(part)
        if left < x0 and x1 < right and top < y0 and y1 < bottom and len(part) >= 200:
            panel_parts.append(part)
    if len(panel_parts) != 4:
        raise ValueError("four intact snapshot panels were not found")
    panel_parts.sort(key=lambda pts: (_bbox(pts)[1], _bbox(pts)[0]))

    histories = {name: [] for name in _COLORS}
    for part in panel_parts:
        px0, py0, px1, py1 = _bbox(part)
        if px1 - px0 + 1 != 21 or py1 - py0 + 1 != 21:
            raise ValueError("snapshot panel is damaged")
        crop = array[py0 : py1 + 1, px0 : px1 + 1]
        gold_points = np.argwhere(_mask(crop, _GOLD))
        if len(gold_points) < 50:
            raise ValueError("courtyard region is damaged")
        gy0, gx0 = gold_points.min(axis=0)
        gy1, gx1 = gold_points.max(axis=0)
        if int(gx1 - gx0 + 1) != 9 or int(gy1 - gy0 + 1) != 9:
            raise ValueError("courtyard boundary is unresolved")
        for name, rgb in _COLORS.items():
            points = np.argwhere(_mask(crop, rgb))
            if len(points) != 9:
                raise ValueError(f"{name} bead is absent or corrupted")
            cy, cx = points.mean(axis=0)
            inside = gx0 < cx < gx1 and gy0 < cy < gy1
            clearance = min(abs(cx - gx0), abs(cx - gx1), abs(cy - gy0), abs(cy - gy1))
            if clearance < 2.0:
                raise ValueError("bead lies too near the courtyard boundary")
            histories[name].append(int(inside))

    events = []
    for name, history in histories.items():
        first = next((step for step in range(1, 4) if history[step - 1] == 0 and history[step] == 1), None)
        if first is not None:
            events.append((first, name))
    if not events:
        raise ValueError("no entry event is visible")
    earliest = min(step for step, _ in events)
    winners = [name for step, name in events if step == earliest]
    if len(winners) != 1:
        raise ValueError("earliest entry is ambiguous")
    return winners[0]
