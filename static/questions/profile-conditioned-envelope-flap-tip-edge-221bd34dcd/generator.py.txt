"""Deterministic renderer for the envelope flap-tip edge world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SCALE = 3
EDGES = ("top", "right", "bottom", "left")
FLAP_PALETTES = ((31, 91, 196), (38, 112, 184), (54, 82, 185), (25, 105, 170))
PAPER_PALETTES = ((250, 247, 238), (246, 249, 252), (252, 245, 241), (247, 247, 244))
BACKGROUND_PALETTES = ((224, 232, 238), (235, 229, 219), (226, 235, 225), (235, 226, 232))


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one balanced, fully visible scene from an integer seed."""
    rng = random.Random(int(seed))
    # The label schedule is independent of all subsequent random style draws.
    tip_edge = EDGES[int(seed) % len(EDGES)]
    box_w = rng.randrange(244, 277, 4)
    box_h = rng.randrange(174, 203, 4)
    cx = rng.randrange(186, 199)
    cy = rng.randrange(186, 199)
    palette_index = rng.randrange(len(FLAP_PALETTES))
    paper_index = rng.randrange(len(PAPER_PALETTES))
    background_index = rng.randrange(len(BACKGROUND_PALETTES))
    return {
        "tip_edge": tip_edge,
        "cx": cx,
        "cy": cy,
        "box_w": box_w,
        "box_h": box_h,
        "flap_rgb": list(FLAP_PALETTES[palette_index]),
        "paper_rgb": list(PAPER_PALETTES[paper_index]),
        "background_rgb": list(BACKGROUND_PALETTES[background_index]),
    }


def _box(scene: dict[str, Any]) -> tuple[int, int, int, int]:
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    box_w = int(scene["box_w"])
    box_h = int(scene["box_h"])
    return cx - box_w // 2, cy - box_h // 2, cx + box_w // 2, cy + box_h // 2


def _flap_vertices(scene: dict[str, Any]) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = _box(scene)
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    edge = str(scene["tip_edge"])
    half_base = 55
    depth = 96
    if edge == "top":
        return [(cx, y0), (cx - half_base, y0 + depth), (cx + half_base, y0 + depth)]
    if edge == "right":
        return [(x1, cy), (x1 - depth, cy - half_base), (x1 - depth, cy + half_base)]
    if edge == "bottom":
        return [(cx, y1), (cx + half_base, y1 - depth), (cx - half_base, y1 - depth)]
    if edge == "left":
        return [(x0, cy), (x0 + depth, cy + half_base), (x0 + depth, cy - half_base)]
    raise ValueError(f"invalid tip_edge {edge!r}")


def _validate(scene: dict[str, Any]) -> None:
    if str(scene["tip_edge"]) not in EDGES:
        raise ValueError("tip_edge must be an image-relative edge")
    x0, y0, x1, y1 = _box(scene)
    if not (24 <= x0 < x1 <= WIDTH - 24 and 24 <= y0 < y1 <= HEIGHT - 24):
        raise ValueError("envelope must remain fully visible")
    for key in ("flap_rgb", "paper_rgb", "background_rgb"):
        rgb = scene[key]
        if not isinstance(rgb, list) or len(rgb) != 3 or any(not 0 <= int(v) <= 255 for v in rgb):
            raise ValueError(f"invalid {key}")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a familiar envelope icon; the answer is encoded only in pixels."""
    _validate(scene)
    x0, y0, x1, y1 = _box(scene)
    cx = int(scene["cx"])
    cy = int(scene["cy"])
    flap = tuple(int(v) for v in scene["flap_rgb"])
    paper = tuple(int(v) for v in scene["paper_rgb"])
    background = tuple(int(v) for v in scene["background_rgb"])

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), background)
    draw = ImageDraw.Draw(image)
    s = SCALE
    rect = tuple(v * s for v in (x0, y0, x1, y1))
    draw.rounded_rectangle(rect, radius=8 * s, fill=paper, outline=(34, 40, 48), width=5 * s)

    # Light seam lines and postal marks make the object recognizable while using
    # colors that the independent blue-mask oracle rejects.
    seam = (158, 164, 171)
    draw.line([(x0 * s, y0 * s), (cx * s, cy * s), (x1 * s, y0 * s)], fill=seam, width=2 * s)
    draw.line([(x0 * s, y1 * s), (cx * s, cy * s), (x1 * s, y1 * s)], fill=seam, width=2 * s)
    stamp_box = ((x1 - 43) * s, (y0 + 17) * s, (x1 - 17) * s, (y0 + 43) * s)
    draw.rectangle(stamp_box, fill=(236, 174, 78), outline=(126, 83, 38), width=2 * s)
    for offset in (-15, 0, 15):
        draw.line([((cx - 48) * s, (cy + offset) * s), ((cx + 48) * s, (cy + offset) * s)], fill=(184, 83, 76), width=3 * s)

    vertices = [(x * s, y * s) for x, y in _flap_vertices(scene)]
    draw.polygon(vertices, fill=flap, outline=(15, 57, 125), width=2 * s)
    # Tiny white crease at the broad base emphasizes which vertex is the tip.
    draw.line([vertices[1], vertices[2]], fill=(224, 232, 244), width=2 * s)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["tip_edge"])


def margin(scene: dict[str, Any]) -> float:
    """Runner-up gap: next nearest border minus the apex's zero distance."""
    edge = str(scene["tip_edge"])
    if edge in ("top", "bottom"):
        return float(min(int(scene["box_w"]) // 2, int(scene["box_h"])))
    return float(min(int(scene["box_h"]) // 2, int(scene["box_w"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    flap = tuple(int(v) for v in scene["flap_rgb"])
    paper = tuple(int(v) for v in scene["paper_rgb"])
    color_gap = math.sqrt(sum((a - b) ** 2 for a, b in zip(flap, paper)))
    return margin(scene) < 70.0 or color_gap < 90.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # No two distinct declared scenes intentionally rasterize to identical pixels.
    return []
