"""Independent inverse arm: recover the flap tip edge from final RGB pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the raster does not contain unambiguous declared evidence."""


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    blue_mask = (blue >= 125) & (blue - red >= 45) & (blue - green >= 20)
    ys, xs = np.nonzero(blue_mask)
    if len(xs) < 1200:
        raise OracleAbstention("solid blue triangular flap is missing or too small")

    # The outer envelope is the largest near-dark rectangular stroke. Postal
    # decorations are inside it, so extrema of all dark ink recover its border.
    dark_mask = (red + green + blue < 205) & (np.max(arr, axis=2) < 105)
    dys, dxs = np.nonzero(dark_mask)
    if len(dxs) < 300:
        raise OracleAbstention("dark outer envelope border is missing")
    left, right = int(dxs.min()), int(dxs.max())
    top, bottom = int(dys.min()), int(dys.max())
    if right - left < 180 or bottom - top < 130:
        raise OracleAbstention("outer border geometry is implausible")

    distances = {
        "top": abs(int(ys.min()) - top),
        "right": abs(right - int(xs.max())),
        "bottom": abs(bottom - int(ys.max())),
        "left": abs(int(xs.min()) - left),
    }
    ranked = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    if ranked[0][1] > 6 or ranked[1][1] - ranked[0][1] < 30:
        raise OracleAbstention(f"blue apex-to-border contact is ambiguous: {distances}")
    return ranked[0][0]
