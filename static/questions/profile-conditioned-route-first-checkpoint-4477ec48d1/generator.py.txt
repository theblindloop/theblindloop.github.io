"""Scene sampling, RGB rendering, analytic gold and margins for route_first_checkpoint.

A single continuous gray pipe is drawn as a self-avoiding walk on a coarse
lattice.  One end of the pipe carries an orange ring.  Three colored squares
(blue, red, purple) sit on the pipe at well separated arc-length positions.
The decision is which square is met first when walking along the pipe from the
ringed end.  Sampling deliberately decorrelates the answer from the cheap
straight-line cue: the first square along the pipe is also the square nearest
the ring in straight-line distance in only one third of scenes.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route_first_checkpoint-renderer-0.1.0"

CANVAS = 512
ORIGIN = 46
SPACING = 84
GRID = 6
STROKE = 9
RING_RADIUS = 20
RING_WIDTH = 5
MARK_SIZE = 26

BACKGROUND = (255, 255, 255)
PIPE = (85, 85, 85)
RING = (255, 150, 0)
MARK_COLORS = {
    "blue": (30, 90, 220),
    "red": (215, 35, 35),
    "purple": (150, 45, 190),
}
COLOR_NAMES = ("blue", "red", "purple")

MIN_ARC_SEPARATION = 100.0
MIN_MARK_DISTANCE = 55.0
MIN_START_DISTANCE = 70.0
QUARANTINE_MARGIN = 130.0


def _cell_xy(cell: tuple[int, int]) -> tuple[float, float]:
    return (float(ORIGIN + SPACING * cell[0]), float(ORIGIN + SPACING * cell[1]))


def _walk(rng: random.Random, target: int) -> list[tuple[int, int]] | None:
    """Randomised depth-first self-avoiding lattice walk of `target` vertices."""
    start = (rng.randrange(GRID), rng.randrange(GRID))
    path = [start]
    visited = {start}
    stack = [_shuffled_neighbours(rng, start, visited)]
    steps = 0
    while path and steps < 4000:
        steps += 1
        if len(path) == target:
            return path
        options = stack[-1]
        while options and options[-1] in visited:
            options.pop()
        if not options:
            visited.discard(path.pop())
            stack.pop()
            continue
        nxt = options.pop()
        path.append(nxt)
        visited.add(nxt)
        stack.append(_shuffled_neighbours(rng, nxt, visited))
    return None


def _shuffled_neighbours(
    rng: random.Random, cell: tuple[int, int], visited: set
) -> list[tuple[int, int]]:
    out = []
    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        nb = (cell[0] + dx, cell[1] + dy)
        if 0 <= nb[0] < GRID and 0 <= nb[1] < GRID and nb not in visited:
            out.append(nb)
    rng.shuffle(out)
    return out


def _point_at(polyline: list[list[float]], t: float) -> tuple[float, float]:
    remaining = t
    for i in range(len(polyline) - 1):
        ax, ay = polyline[i]
        bx, by = polyline[i + 1]
        seg = math.hypot(bx - ax, by - ay)
        if remaining <= seg:
            f = remaining / seg
            return (ax + (bx - ax) * f, ay + (by - ay) * f)
        remaining -= seg
    return (polyline[-1][0], polyline[-1][1])


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for `seed`, balanced by construction."""
    rng = random.Random(1_000_003 + 7919 * int(seed))
    target_color = COLOR_NAMES[int(seed) % 3]
    euclid_agrees = (int(seed) // 3) % 3 == 0

    for _ in range(600):
        n_vertices = rng.randrange(12, 17)
        cells = _walk(rng, n_vertices)
        if cells is None:
            continue
        polyline = [list(_cell_xy(c)) for c in cells]
        total = SPACING * (len(polyline) - 1)

        # Arc positions sit on straight runs, away from every corner.
        slots = []
        for k in range(1, len(polyline) - 2):
            slots.append(84.0 * k + rng.randrange(28, 57))
        if len(slots) < 3:
            continue
        chosen = sorted(rng.sample(slots, 3))
        if chosen[0] < 110.0 or chosen[-1] > total - 90.0:
            continue
        if min(chosen[1] - chosen[0], chosen[2] - chosen[1]) < MIN_ARC_SEPARATION:
            continue

        points = [_point_at(polyline, t) for t in chosen]
        start = (polyline[0][0], polyline[0][1])
        pair_ok = all(
            math.dist(points[i], points[j]) >= MIN_MARK_DISTANCE
            for i in range(3)
            for j in range(i + 1, 3)
        )
        if not pair_ok:
            continue
        straight = [math.dist(start, p) for p in points]
        if min(straight) < MIN_START_DISTANCE:
            continue
        nearest = min(range(3), key=lambda i: straight[i])
        if abs(straight[sorted(range(3), key=lambda i: straight[i])[1]] - min(straight)) < 20.0:
            continue
        if (nearest == 0) != euclid_agrees:
            continue

        others = [c for c in COLOR_NAMES if c != target_color]
        rng.shuffle(others)
        names = [target_color, others[0], others[1]]
        checkpoints = [
            {
                "color": names[i],
                "t": round(chosen[i], 4),
                "x": round(points[i][0], 4),
                "y": round(points[i][1], 4),
            }
            for i in range(3)
        ]
        return {
            "canvas": [CANVAS, CANVAS],
            "background": list(BACKGROUND),
            "grid": [GRID, SPACING, ORIGIN],
            "polyline": [[round(x, 4), round(y, 4)] for x, y in polyline],
            "checkpoints": checkpoints,
            "start_xy": [round(start[0], 4), round(start[1], 4)],
            "stroke_width": STROKE,
            "ring_radius": RING_RADIUS,
            "ring_width": RING_WIDTH,
            "mark_size": MARK_SIZE,
            "pipe_color": list(PIPE),
            "ring_color": list(RING),
        }
    raise RuntimeError(f"failed to sample a valid scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    cx, cy = scene["start_xy"]
    r = scene["ring_radius"]
    draw.ellipse(
        [cx - r, cy - r, cx + r, cy + r],
        outline=tuple(scene["ring_color"]),
        width=int(scene["ring_width"]),
    )

    w = int(scene["stroke_width"])
    pipe = tuple(scene["pipe_color"])
    points = [(float(x), float(y)) for x, y in scene["polyline"]]
    for (ax, ay), (bx, by) in zip(points, points[1:]):
        draw.line([ax, ay, bx, by], fill=pipe, width=w)
    half = w // 2
    for x, y in points:
        draw.rectangle([x - half, y - half, x + half, y + half], fill=pipe)

    size = int(scene["mark_size"])
    for mark in scene["checkpoints"]:
        mx, my = float(mark["x"]), float(mark["y"])
        draw.rectangle(
            [mx - size / 2, my - size / 2, mx + size / 2, my + size / 2],
            fill=MARK_COLORS[mark["color"]],
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Colour name of the square met first when walking from the ringed end."""
    marks = scene["checkpoints"]
    first = min(marks, key=lambda mark: mark["t"])
    return str(first["color"])


def margin(scene: dict) -> float:
    """Along-pipe gap in pixels between the first and the runner-up square."""
    order = sorted(float(mark["t"]) for mark in scene["checkpoints"])
    return float(order[1] - order[0])


def is_quarantined(scene: dict) -> bool:
    marks = scene["checkpoints"]
    centres = [(float(m["x"]), float(m["y"])) for m in marks]
    start = (float(scene["start_xy"][0]), float(scene["start_xy"][1]))
    pair = min(
        math.dist(centres[i], centres[j]) for i in range(len(centres)) for j in range(i + 1, len(centres))
    )
    near = min(math.dist(start, c) for c in centres)
    return bool(
        margin(scene) < QUARANTINE_MARGIN
        or pair < MIN_MARK_DISTANCE
        or near < MIN_START_DISTANCE
    )


def latent_symmetries(scene: dict) -> list:
    """No render-invariant relabelling exists; see candidate.json latent_alias."""
    return []
