"""Image-only oracle for route_first_checkpoint.

Receives the final PNG and nothing else.  It segments the pipe, the ring and
the three colored squares by exact colour, builds one traversable mask from
pipe plus squares, and runs a breadth-first flood along that mask from the
pipe pixel under the ring centre.  The square with the smallest along-pipe
flood distance is the one met first.  Returns "unknown" when the evidence is
missing, unreachable, or the two leading distances are too close to separate.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

ORACLE_VERSION = "route_first_checkpoint-oracle-0.1.0"

ABSTAIN = "unknown"
_PIPE = (85, 85, 85)
_RING = (255, 150, 0)
_MARKS = {
    "blue": (30, 90, 220),
    "red": (215, 35, 35),
    "purple": (150, 45, 190),
}
_TOLERANCE = 30
_MIN_PIXELS = 40
_MIN_GAP = 25


def _close(pixel: tuple[int, int, int], target: tuple[int, int, int]) -> bool:
    return (
        abs(pixel[0] - target[0]) <= _TOLERANCE
        and abs(pixel[1] - target[1]) <= _TOLERANCE
        and abs(pixel[2] - target[2]) <= _TOLERANCE
    )


def decision_from_image(image):
    """Colour name of the first square along the pipe, or "unknown"."""
    if not isinstance(image, Image.Image):
        image = Image.open(image)
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    traversable = bytearray(width * height)
    mark_pixels: dict[str, list[int]] = {name: [] for name in _MARKS}
    ring_sum_x = ring_sum_y = ring_n = 0

    for y in range(height):
        row = y * width
        for x in range(width):
            pixel = pixels[x, y]
            if pixel == (255, 255, 255):
                continue
            if _close(pixel, _PIPE):
                traversable[row + x] = 1
                continue
            if _close(pixel, _RING):
                ring_sum_x += x
                ring_sum_y += y
                ring_n += 1
                continue
            for name, target in _MARKS.items():
                if _close(pixel, target):
                    traversable[row + x] = 1
                    mark_pixels[name].append(row + x)
                    break

    if ring_n < _MIN_PIXELS:
        return ABSTAIN
    if any(len(hits) < _MIN_PIXELS for hits in mark_pixels.values()):
        return ABSTAIN

    start = _nearest_traversable(traversable, width, height, ring_sum_x // ring_n, ring_sum_y // ring_n)
    if start is None:
        return ABSTAIN

    distance = _flood(traversable, width, height, start)

    reached = {}
    for name, hits in mark_pixels.items():
        best = min((distance[index] for index in hits if distance[index] >= 0), default=-1)
        if best < 0:
            return ABSTAIN
        reached[name] = best

    order = sorted(reached.items(), key=lambda item: item[1])
    if order[1][1] - order[0][1] < _MIN_GAP:
        return ABSTAIN
    return order[0][0]


def _nearest_traversable(traversable, width, height, cx, cy):
    for radius in range(0, 40):
        for dy in range(-radius, radius + 1):
            y = cy + dy
            if not 0 <= y < height:
                continue
            span = radius - abs(dy)
            for dx in (-span, span):
                x = cx + dx
                if 0 <= x < width and traversable[y * width + x]:
                    return y * width + x
    return None


def _flood(traversable, width, height, start):
    distance = [-1] * (width * height)
    distance[start] = 0
    queue = deque([start])
    while queue:
        index = queue.popleft()
        step = distance[index] + 1
        x = index % width
        if x > 0 and traversable[index - 1] and distance[index - 1] < 0:
            distance[index - 1] = step
            queue.append(index - 1)
        if x + 1 < width and traversable[index + 1] and distance[index + 1] < 0:
            distance[index + 1] = step
            queue.append(index + 1)
        if index >= width and traversable[index - width] and distance[index - width] < 0:
            distance[index - width] = step
            queue.append(index - width)
        if index + width < width * height and traversable[index + width] and distance[index + width] < 0:
            distance[index + width] = step
            queue.append(index + width)
    return distance
