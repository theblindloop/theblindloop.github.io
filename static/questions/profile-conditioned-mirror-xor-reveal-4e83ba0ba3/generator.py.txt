"""mirror_xor_reveal: symmetry-difference icon reveal world.

Declared transform: take the left-right mirror difference (XOR) of the two
halves of the canvas. The two halves are both dense block lattices of random
ink, so neither half alone shows the answer; only their pixel-wise XOR reveals
a single block glyph, which is either the letter A or the letter B. The
question asks which letter the difference reveals.

Latent scene z = {noise, glyph, invert, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "mirror_xor_reveal-0.1.0"

W = 512
H = 512
W2 = W // 2
CELL = 8
NROWS = H // CELL          # 64
NCOLS = W2 // CELL         # 32

# Fixed block-glyph templates: 15 rows x 12 columns of cells.
TEMPLATE_A = [
    list("............"),
    list("....##......"),
    list("....##......"),
    list("...#..#....."),
    list("...#..#....."),
    list("..#....#...."),
    list("..#....#...."),
    list(".#......#..."),
    list(".#......#..."),
    list(".########..."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("............"),
    list("............"),
]
TEMPLATE_B = [
    list("............"),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
]
TEMPLATES = {"A": TEMPLATE_A, "B": TEMPLATE_B}

# Glyph box placement in the LEFT field cell coordinates.
BOX_R0, BOX_R1 = 8, 8 + 15   # rows [8, 23)
BOX_C0, BOX_C1 = 10, 10 + 12  # cols [10, 22)
Q_TOL = 0.15  # quarantine threshold on the A/B correlation margin


def _template_mask(glyph: str) -> np.ndarray:
    """(NROWS, NCOLS) boolean: glyph template placed in the left field."""
    grid = TEMPLATES[glyph]
    mask = np.zeros((NROWS, NCOLS), dtype=bool)
    mask[BOX_R0:BOX_R1, BOX_C0:BOX_C1] = np.array(
        [[c == "#" for c in row] for row in grid], dtype=bool
    )
    return mask


def build_scene(noise: int, glyph: str, invert: bool) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "invert": bool(invert),
        "cell": CELL,
    }


def sample_scene(seed: int) -> dict:
    """One deterministic scene from a scalar seed (interface requirement).

    The class glyph alternates with seed parity and invert cycles so a single
    stress seed still yields both classes when generate.py mixes seeds.
    """
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph, invert=bool(int(seed) % 4 == 0))


def _left_field(rng: np.random.Generator) -> np.ndarray:
    return rng.integers(0, 2, size=(NROWS, NCOLS)).astype(bool)


def render_mask(scene: dict) -> np.ndarray:
    """Full (H, W) boolean ink mask of the rendered canvas."""
    cell = int(scene.get("cell", CELL))
    nrows, ncols = H // cell, W2 // cell
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    left = rng.integers(0, 2, size=(nrows, ncols)).astype(bool)
    glyph_mask = _template_mask(scene["glyph"])
    right = np.copy(left)
    right[:, :] = left[:, ::-1] ^ glyph_mask[:, ::-1]  # R[i,k] = L[i,31-k] ^ M[i,31-k]
    cells = np.concatenate([left, right], axis=1)  # (nrows, 2*ncols)
    mask = np.repeat(np.repeat(cells, cell, axis=0), cell, axis=1)
    return mask[:H, :W]


def render(scene: dict) -> Image.Image:
    mask = render_mask(scene)
    invert = bool(scene.get("invert", False))
    bg = (0, 0, 0) if invert else (255, 255, 255)
    fg = (255, 255, 255) if invert else (0, 0, 0)
    arr = np.where(mask[:, :, None], np.array(fg, dtype=np.uint8), np.array(bg, dtype=np.uint8))
    return Image.fromarray(arr.astype(np.uint8), "RGB")


def margin(scene: dict) -> float:
    """A/B template correlation margin, analytic, deterministic."""
    chosen = _template_mask(scene["glyph"])
    other = _template_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_chosen = float((chosen & chosen).sum()) / max(1, c_filled)
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(corr_chosen - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: the parametrization is minimal and fully visible; every
    latent field (the random lattice, the answer glyph, the color inversion,
    the block size) directly changes the rendered pixels, so no distinct latent
    vector reproduces the same image. Declared empty."""
    return iter(())
