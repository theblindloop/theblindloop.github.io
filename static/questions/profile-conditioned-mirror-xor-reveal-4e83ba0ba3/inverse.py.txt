"""Independent pixel-only inverse arm for mirror_xor_reveal.

Receives ONLY the final PNG. It re-discovers the revealed symbol by taking the
left-right mirror pixel difference (XOR), then classifies the resulting shape
against the two fixed block-glyph candidates A and B that the public question
contract declares. No scene data, no analytic gold.

The reveal operation is polarity-invariant (a global fg/bg swap complements
both halves, leaving the XOR unchanged), so the threshold direction does not
matter.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mirror_xor_reveal-oracle-0.1.0"

W2 = 256
CELL = 8
BOX_R0, BOX_R1 = 64, 184  # pixel rows of glyph box (left field)
BOX_C0, BOX_C1 = 80, 176  # pixel cols of glyph box (left field)

TEMPLATE_A = [
    list("............"),
    list("....##......"),
    list("....##......"),
    list("...#..#....."),
    list("...#..#....."),
    list("..#....#...."),
    list("..#....#...."),
    list(".#......#..."),
    list(".#......#..."),
    list(".########..."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("............"),
    list("............"),
]
TEMPLATE_B = [
    list("............"),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("#........#.."),
    list("#........#.."),
    list("#........#.."),
    list("##########.."),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
    list("............"),
]
TEMPLATES = {"A": TEMPLATE_A, "B": TEMPLATE_B}


def _template_pixel_mask(glyph: str) -> np.ndarray:
    grid = np.array([[c == "#" for c in row] for row in TEMPLATES[glyph]], dtype=bool)
    return np.repeat(np.repeat(grid, CELL, axis=0), CELL, axis=1)


def _reveal(mask: np.ndarray) -> np.ndarray:
    """XOR of the left half with the mirror of the right half: the hidden glyph."""
    left = mask[:, :W2]
    right = mask[:, W2:]
    return left ^ right[:, ::-1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("L"), dtype=np.uint8)
    ink = arr < 128  # fg assumed darker; polarity-invariant because XOR complements
    revealed = _reveal(ink)
    crop = revealed[BOX_R0:BOX_R1, BOX_C0:BOX_C1]
    scores = {}
    for glyph, tmpl in TEMPLATES.items():
        tp = _template_pixel_mask(glyph)
        scores[glyph] = float((crop & tp).sum()) / float(tp.sum())
    return "A" if scores["A"] >= scores["B"] else "B"


def reveal_margin(image: Image.Image) -> float:
    """Decision margin recovered from pixels alone (diagnostic / quarantine aid)."""
    arr = np.asarray(image.convert("L"), dtype=np.uint8)
    revealed = _reveal(arr < 128)
    crop = revealed[BOX_R0:BOX_R1, BOX_C0:BOX_C1]
    scores = {}
    for glyph, tmpl in TEMPLATES.items():
        tp = _template_pixel_mask(glyph)
        scores[glyph] = float((crop & tp).sum()) / float(tp.sum())
    return abs(scores["A"] - scores["B"])
