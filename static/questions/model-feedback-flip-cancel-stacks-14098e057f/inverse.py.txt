"""Independent fixed-layout PNG decoder and stack-state executor."""

from __future__ import annotations

from PIL import Image

WIDTH = 1120
HEIGHT = 820
DARK = (42, 57, 73)
LIGHT = (226, 231, 235)
PAPER = (255, 255, 255)
INK = (34, 42, 54)
BADGES = ((32, 119, 180), (230, 126, 34), (52, 154, 91), (151, 91, 176))
STACK_X = (105, 275, 445, 615)
STACK_Y = 142
STACK_CELL_H = 42
STACK_GAP = 7
TARGET_X = 940
TARGET_Y = 138
TARGET_CELL_H = 27
TARGET_GAP = 5
TARGET_SLOTS = 10
CARD_X = (30, 300, 570, 840)
CARD_Y = (545, 680)
SOURCE_CENTER = (78, 65)
DEST_CENTER = (177, 65)


class OracleAbstention(ValueError):
    pass


def _near(pixel: tuple[int, ...], color: tuple[int, int, int], tolerance: int = 8) -> bool:
    return len(pixel) >= 3 and max(abs(int(pixel[i]) - color[i]) for i in range(3)) <= tolerance


def _face(pixel: tuple[int, ...]) -> int:
    if _near(pixel, DARK):
        return 1
    if _near(pixel, LIGHT):
        return 0
    raise OracleAbstention(f"unrecognized tile color {pixel[:3]}")


def _badge(pixel: tuple[int, ...]) -> int:
    matches = [index for index, color in enumerate(BADGES) if _near(pixel, color)]
    if len(matches) != 1:
        raise OracleAbstention(f"unrecognized endpoint color {pixel[:3]}")
    return matches[0]


def _simulate(stacks: list[list[int]], moves: list[tuple[int, int]]) -> list[list[int]]:
    live = [list(stack) for stack in stacks]
    for source, destination in moves:
        if source == destination or not live[source]:
            raise OracleAbstention("invalid visible transfer program")
        moved = 1 - live[source].pop(0)
        live[destination].insert(0, moved)
        if len(live[destination]) >= 2 and live[destination][0] == live[destination][1]:
            del live[destination][:2]
    return live


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected image size {rgb.size}")
    px = rgb.load()

    stacks: list[list[int]] = []
    for x in STACK_X:
        stack = []
        for row in range(4):
            y = STACK_Y + row * (STACK_CELL_H + STACK_GAP) + 8
            stack.append(_face(px[x - 28, y]))
        stacks.append(stack)

    moves: list[tuple[int, int]] = []
    for number in range(8):
        x = CARD_X[number % 4]
        y = CARD_Y[number // 4]
        if not _near(px[x + 24, y + 10], INK):
            raise OracleAbstention(f"missing order marker on card {number + 1}")
        source_xy = (x + SOURCE_CENTER[0] - 15, y + SOURCE_CENTER[1] + 11)
        destination_xy = (x + DEST_CENTER[0] - 15, y + DEST_CENTER[1] + 11)
        moves.append((_badge(px[source_xy[0], source_xy[1]]), _badge(px[destination_xy[0], destination_xy[1]])))

    target: list[int] = []
    ended = False
    for row in range(TARGET_SLOTS):
        y = TARGET_Y + row * (TARGET_CELL_H + TARGET_GAP) + 7
        pixel = px[TARGET_X - 30, y]
        if _near(pixel, PAPER):
            ended = True
            continue
        if ended:
            raise OracleAbstention("target strip has a gap")
        target.append(_face(pixel))
    if not target:
        raise OracleAbstention("empty target strip")

    final = _simulate(stacks, moves)
    matches = [index for index, stack in enumerate(final) if stack == target]
    if len(matches) != 1:
        raise OracleAbstention("target does not match one unique final stack")
    return f"stack-{'ABCD'[matches[0]]}"
