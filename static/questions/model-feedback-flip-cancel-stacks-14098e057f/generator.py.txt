"""Deterministic sampler, simulator, and rasterizer for mutable stack programs."""

from __future__ import annotations

import copy
import random
from functools import lru_cache

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1120
HEIGHT = 820
BG = (246, 245, 240)
PAPER = (255, 255, 255)
INK = (34, 42, 54)
MUTED = (111, 120, 130)
DARK = (42, 57, 73)
LIGHT = (226, 231, 235)
BADGES = ((32, 119, 180), (230, 126, 34), (52, 154, 91), (151, 91, 176))
STACK_X = (105, 275, 445, 615)
STACK_Y = 142
STACK_CELL_W = 88
STACK_CELL_H = 42
STACK_GAP = 7
TARGET_X = 940
TARGET_Y = 138
TARGET_CELL_W = 92
TARGET_CELL_H = 27
TARGET_GAP = 5
TARGET_SLOTS = 10
CARD_X = (30, 300, 570, 840)
CARD_Y = (545, 680)
CARD_W = 250
CARD_H = 108
SOURCE_CENTER = (78, 65)
DEST_CENTER = (177, 65)


@lru_cache(maxsize=None)
def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _simulate(initial: list[list[int]], moves: list[list[int]]) -> tuple[list[list[int]], int]:
    stacks = [list(stack) for stack in initial]
    cancellations = 0
    for source, destination in moves:
        if source == destination or not 0 <= source < 4 or not 0 <= destination < 4:
            raise ValueError("invalid transfer endpoints")
        if not stacks[source]:
            raise ValueError("transfer from empty stack")
        face = 1 - stacks[source].pop(0)
        stacks[destination].insert(0, face)
        if len(stacks[destination]) >= 2 and stacks[destination][0] == stacks[destination][1]:
            del stacks[destination][:2]
            cancellations += 1
    return stacks, cancellations


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    for _ in range(20000):
        stacks = [[rng.randrange(2) for _ in range(4)] for _ in range(4)]
        if any(len(set(stack)) < 2 for stack in stacks):
            continue
        live = [list(stack) for stack in stacks]
        moves: list[list[int]] = []
        destinations_seen: set[int] = set()
        dynamic_reads = 0
        cancellations = 0
        endpoints: set[int] = set()
        valid = True
        for _step in range(8):
            sources = [index for index, stack in enumerate(live) if stack]
            dependent = [index for index in sources if index in destinations_seen]
            source = rng.choice(dependent if dependent and rng.random() < 0.72 else sources)
            destination = rng.choice([index for index in range(4) if index != source])
            if source in destinations_seen:
                dynamic_reads += 1
            moves.append([source, destination])
            endpoints.update((source, destination))
            destinations_seen.add(destination)
            before = sum(map(len, live))
            live, _ = _simulate(live, [[source, destination]])
            if sum(map(len, live)) == before - 2:
                cancellations += 1
            if any(len(stack) > TARGET_SLOTS for stack in live):
                valid = False
                break
        if not valid or not 2 <= cancellations <= 5 or dynamic_reads < 3 or endpoints != {0, 1, 2, 3}:
            continue
        patterns = [tuple(stack) for stack in live]
        target_length = len(live[target_index])
        same_length = sum(len(stack) == target_length for stack in live)
        if not 2 <= target_length <= 6 or same_length < 2 or len(set(patterns)) != 4:
            continue
        draw_order = list(range(8))
        rng.shuffle(draw_order)
        return {
            "stacks": stacks,
            "moves": moves,
            "target_pattern": list(live[target_index]),
            "move_draw_order": draw_order,
        }
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    final, _ = _simulate(scene["stacks"], scene["moves"])
    target = list(scene["target_pattern"])
    matches = [index for index, stack in enumerate(final) if stack == target]
    if len(matches) != 1:
        raise ValueError("target must match exactly one final stack")
    return f"stack-{'ABCD'[matches[0]]}"


def margin(scene: dict) -> float:
    analytic_gold(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    analytic_gold(scene)
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["move_draw_order"] = list(reversed(scene["move_draw_order"]))
    return [("reverse_disjoint_move_card_draw_order", twin)]


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], index: int, radius: int = 24) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=BADGES[index], outline=INK, width=3)
    _centered(draw, center, "ABCD"[index], _font(24, True), PAPER)


def _tile(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], face: int) -> None:
    fill = DARK if int(face) else LIGHT
    draw.rounded_rectangle(box, radius=8, fill=fill, outline=INK, width=3)
    cx = (box[0] + box[2]) // 2
    cy = (box[1] + box[3]) // 2
    if face:
        draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=PAPER)
    else:
        draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=INK)


def _draw_stack(draw: ImageDraw.ImageDraw, index: int, stack: list[int]) -> None:
    x = STACK_X[index]
    _badge(draw, (x, 91), index, 25)
    draw.text((x + 34, 75), f"STACK {'ABCD'[index]}", font=_font(19, True), fill=INK)
    frame = (x - 58, STACK_Y - 10, x + 58, STACK_Y + 4 * (STACK_CELL_H + STACK_GAP) + 7)
    draw.rounded_rectangle(frame, radius=12, fill=PAPER, outline=MUTED, width=3)
    for row, face in enumerate(stack):
        y = STACK_Y + row * (STACK_CELL_H + STACK_GAP)
        _tile(draw, (x - STACK_CELL_W // 2, y, x + STACK_CELL_W // 2, y + STACK_CELL_H), face)


def _draw_target(draw: ImageDraw.ImageDraw, pattern: list[int]) -> None:
    draw.rounded_rectangle((848, 68, 1033, 488), radius=18, fill=PAPER, outline=INK, width=4)
    _centered(draw, (TARGET_X, 96), "TARGET", _font(25, True), INK)
    draw.polygon(((TARGET_X - 8, 115), (TARGET_X + 8, 115), (TARGET_X, 128)), fill=INK)
    for row, face in enumerate(pattern):
        y = TARGET_Y + row * (TARGET_CELL_H + TARGET_GAP)
        _tile(draw, (TARGET_X - TARGET_CELL_W // 2, y, TARGET_X + TARGET_CELL_W // 2, y + TARGET_CELL_H), face)


def _draw_move(draw: ImageDraw.ImageDraw, number: int, move: list[int]) -> None:
    x = CARD_X[number % 4]
    y = CARD_Y[number // 4]
    draw.rounded_rectangle((x, y, x + CARD_W, y + CARD_H), radius=14, fill=PAPER, outline=MUTED, width=3)
    draw.ellipse((x + 9, y + 8, x + 39, y + 38), fill=INK)
    _centered(draw, (x + 24, y + 23), str(number + 1), _font(17, True), PAPER)
    source = (x + SOURCE_CENTER[0], y + SOURCE_CENTER[1])
    destination = (x + DEST_CENTER[0], y + DEST_CENTER[1])
    _badge(draw, source, int(move[0]), 24)
    _badge(draw, destination, int(move[1]), 24)
    draw.line((source[0] + 30, source[1], destination[0] - 31, destination[1]), fill=INK, width=5)
    draw.polygon(((destination[0] - 31, destination[1] - 9), (destination[0] - 31, destination[1] + 9), (destination[0] - 17, destination[1])), fill=INK)


def render(scene: dict) -> Image.Image:
    analytic_gold(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((45, 20), "START STACKS", font=_font(31, True), fill=INK)
    draw.text((45, 48), "Read each stack from TOP downward", font=_font(18), fill=MUTED)
    for index, stack in enumerate(scene["stacks"]):
        _draw_stack(draw, index, stack)
    _draw_target(draw, scene["target_pattern"])

    draw.rounded_rectangle((45, 378, 792, 475), radius=14, fill=PAPER, outline=MUTED, width=3)
    draw.text((68, 392), "EACH MOVE", font=_font(17, True), fill=MUTED)
    _tile(draw, (205, 395, 275, 431), 1)
    draw.text((286, 400), "FLIP", font=_font(22, True), fill=INK)
    _tile(draw, (359, 395, 429, 431), 0)
    draw.text((472, 397), "MATCHING TOP PAIR", font=_font(18, True), fill=INK)
    draw.line((696, 397, 746, 447), fill=(193, 57, 57), width=7)
    draw.line((746, 397, 696, 447), fill=(193, 57, 57), width=7)
    draw.text((45, 500), "MOVES — FOLLOW 1 TO 8", font=_font(25, True), fill=INK)
    for number in scene["move_draw_order"]:
        _draw_move(draw, int(number), scene["moves"][int(number)])
    return image
