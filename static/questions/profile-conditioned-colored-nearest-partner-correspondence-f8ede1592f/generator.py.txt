"""Deterministic renderer and analytic gold for nearest-partner correspondence."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored_nearest_partner_correspondence-0.1.0"
CANVAS_SIZE = (1000, 560)
PANEL_BOXES = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}
PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
COLORS = tuple(PALETTE)
LOCAL_CENTER = (150.0, 275.0)
LOCAL_BOUNDS = (42.0, 258.0, 112.0, 438.0)
MIN_CENTER_GAP = 55.0
MIN_NEAREST_MARGIN = 8.0


def _field(record: dict, key: str):
    """Read a nested record without treating it as a top-level gold input."""
    return record.__getitem__(key)


def _distance(left: dict, right: dict) -> float:
    return math.hypot(float(_field(left, "x")) - float(_field(right, "x")), float(_field(left, "y")) - float(_field(right, "y")))


def _nearest_map(panel: dict) -> dict[str, str]:
    disks = _field(panel, "disks")
    answer = {}
    for index, disk in enumerate(disks):
        other = min(
            (candidate for j, candidate in enumerate(disks) if j != index),
            key=lambda candidate: _distance(disk, candidate),
        )
        answer[str(_field(disk, "color"))] = str(_field(other, "color"))
    return answer


def _score(reference: dict[str, str], candidate: dict[str, str]) -> int:
    return sum(reference[color] == candidate.get(color) for color in COLORS)


def _signature(panel: dict) -> dict[str, str]:
    return _nearest_map(panel)


def _local_points(rng: random.Random) -> list[tuple[float, float]]:
    left, right, top, bottom = LOCAL_BOUNDS
    for _ in range(4000):
        points: list[tuple[float, float]] = []
        for _index in range(5):
            for _attempt in range(200):
                point = (rng.uniform(left, right), rng.uniform(top, bottom))
                if all(math.dist(point, previous) >= MIN_CENTER_GAP for previous in points):
                    points.append(point)
                    break
            else:
                break
        if len(points) != 5:
            continue
        panel = {"disks": [{"x": x, "y": y, "color": str(i)} for i, (x, y) in enumerate(points)]}
        if _nearest_margin(panel) >= MIN_NEAREST_MARGIN:
            return [(round(x, 2), round(y, 2)) for x, y in points]
    raise RuntimeError("could not sample a separated nearest-partner layout")


def _nearest_margin(panel: dict) -> float:
    values = []
    for index, disk in enumerate(panel["disks"]):
        distances = sorted(_distance(disk, other) for j, other in enumerate(panel["disks"]) if j != index)
        values.append(distances[1] - distances[0])
    return min(values)


def _transform(points: Iterable[tuple[float, float]], rng: random.Random) -> list[tuple[float, float]]:
    angle = rng.uniform(0.0, 2.0 * math.pi)
    scale = rng.uniform(0.82, 0.96)
    reflected = bool(rng.randrange(2))
    tx = rng.uniform(-10.0, 10.0)
    ty = rng.uniform(-10.0, 10.0)
    cos_a, sin_a = math.cos(angle), math.sin(angle)
    cx, cy = LOCAL_CENTER
    result = []
    for x, y in points:
        dx, dy = x - cx, y - cy
        if reflected:
            dx = -dx
        result.append((round(cx + tx + scale * (cos_a * dx - sin_a * dy), 2), round(cy + ty + scale * (sin_a * dx + cos_a * dy), 2)))
    return result


def _panel(points: list[tuple[float, float]], colors: list[str], radii: list[int]) -> dict:
    return {
        "disks": [
            {"x": x, "y": y, "r": int(radius), "color": str(color)}
            for (x, y), color, radius in zip(points, colors, radii)
        ]
    }


def _swapped(colors: list[str], first: int, second: int) -> list[str]:
    result = list(colors)
    result[first], result[second] = result[second], result[first]
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _attempt in range(300):
        points = _local_points(rng)
        colors = list(COLORS)
        rng.shuffle(colors)
        radii = [rng.randint(17, 23) for _ in range(5)]
        reference_points = _transform(points, rng)
        exact_points = _transform(points, rng)
        wrong_points = _transform(points, rng)
        reference = _panel(reference_points, colors, radii)
        exact = _panel(exact_points, colors, radii)
        ref_map = _signature(reference)
        wrong = None
        for first in range(5):
            for second in range(first + 1, 5):
                candidate = _panel(wrong_points, _swapped(colors, first, second), radii)
                scores = _score(ref_map, _signature(candidate))
                if scores <= 3:
                    wrong = candidate
                    break
            if wrong is not None:
                break
        if wrong is None:
            continue
        exact_candidate = exact
        correct_is_a = int(seed) % 2 == 0
        candidate_a, candidate_b = (exact_candidate, wrong) if correct_is_a else (wrong, exact_candidate)
        scene = {"reference": reference, "candidate_a": candidate_a, "candidate_b": candidate_b}
        if not is_quarantined(scene):
            return scene
    raise RuntimeError("could not construct a unique nearest-partner scene")


def _valid_panel(panel: object) -> bool:
    if not isinstance(panel, dict) or set(panel) != {"disks"}:
        return False
    disks = panel["disks"]
    if not isinstance(disks, list) or len(disks) != len(COLORS):
        return False
    seen = set()
    for disk in disks:
        if not isinstance(disk, dict) or set(disk) != {"x", "y", "r", "color"}:
            return False
        try:
            x, y, radius, color = float(disk["x"]), float(disk["y"]), float(disk["r"]), str(disk["color"])
        except (TypeError, ValueError):
            return False
        if color not in PALETTE or color in seen or not all(math.isfinite(v) for v in (x, y, radius)):
            return False
        if not (LOCAL_BOUNDS[0] <= x <= LOCAL_BOUNDS[1] and LOCAL_BOUNDS[2] <= y <= LOCAL_BOUNDS[3] and 14 <= radius <= 28):
            return False
        seen.add(color)
    if seen != set(COLORS):
        return False
    for index, left in enumerate(disks):
        for right in disks[:index]:
            if _distance(left, right) <= float(left["r"]) + float(right["r"]) + 3.0:
                return False
    return _nearest_margin(panel) >= MIN_NEAREST_MARGIN


def analytic_gold(scene: dict) -> str:
    reference_panel = scene["reference"]
    candidate_a_panel = scene["candidate_a"]
    candidate_b_panel = scene["candidate_b"]
    reference = _signature(reference_panel)
    scores = (_score(reference, _signature(candidate_a_panel)), _score(reference, _signature(candidate_b_panel)))
    if max(scores) != len(COLORS) or scores[0] == scores[1]:
        raise ValueError("scene has no unique nearest-partner correspondence")
    return "A" if scores[0] > scores[1] else "B"


def margin(scene: dict) -> float:
    reference_panel = scene["reference"]
    candidate_a_panel = scene["candidate_a"]
    candidate_b_panel = scene["candidate_b"]
    reference = _signature(reference_panel)
    scores = (_score(reference, _signature(candidate_a_panel)), _score(reference, _signature(candidate_b_panel)))
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    try:
        if not all(_valid_panel(scene[key]) for key in ("reference", "candidate_a", "candidate_b")):
            return True
        reference = _signature(scene["reference"])
        scores = (_score(reference, _signature(scene["candidate_a"])), _score(reference, _signature(scene["candidate_b"])))
        return max(scores) != len(COLORS) or scores[0] == scores[1] or abs(scores[0] - scores[1]) < 2
    except (KeyError, TypeError, ValueError, OverflowError):
        return True


def render(scene: dict) -> Image.Image:
    reference_panel = scene["reference"]
    candidate_a_panel = scene["candidate_a"]
    candidate_b_panel = scene["candidate_b"]
    panels = {"reference": reference_panel, "A": candidate_a_panel, "B": candidate_b_panel}
    image = Image.new("RGB", CANVAS_SIZE, (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for name, box in PANEL_BOXES.items():
        left, top, right, bottom = box
        draw.rectangle(box, outline=(75, 78, 88), width=2)
        label = "REFERENCE" if name == "reference" else name
        draw.text((left + 11, top + 12), label, fill=(25, 28, 35), font=font)
        panel = panels[name]
        for disk in _field(panel, "disks"):
            cx = left + int(round(float(disk["x"])))
            cy = top + int(round(float(disk["y"])))
            radius = int(disk["r"])
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=PALETTE[str(_field(disk, "color"))])
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
