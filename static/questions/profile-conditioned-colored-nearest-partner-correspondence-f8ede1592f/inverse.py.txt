"""Independent pixel-only inverse for the nearest-partner map."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_PANELS = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}
_PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
_COLORS = tuple(_PALETTE)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected pixel components without using renderer geometry."""
    height, width = mask.shape
    remaining = mask.copy()
    components = []
    for y, x in zip(*np.where(remaining)):
        if not remaining[y, x]:
            continue
        stack = [(int(y), int(x))]
        remaining[y, x] = False
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cx, cy))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and remaining[ny, nx]:
                        remaining[ny, nx] = False
                        stack.append((ny, nx))
        components.append(np.asarray(points, dtype=float))
    return components


def _centers(image: Image.Image, box: tuple[int, int, int, int]) -> dict[str, tuple[float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    left, top, right, bottom = box
    crop = array[top + 3 : bottom - 3, left + 3 : right - 3]
    centers = {}
    for color, rgb in _PALETTE.items():
        mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
        parts = [part for part in _components(mask) if len(part) >= 100]
        if len(parts) != 1:
            raise ValueError("expected one visible disk component per color")
        points = parts[0]
        centers[color] = (float(points[:, 0].mean() + left + 3), float(points[:, 1].mean() + top + 3))
    return centers


def _partner_map(nodes: dict[str, tuple[float, float]]) -> dict[str, str]:
    result = {}
    for color in _COLORS:
        other = min(
            (candidate for candidate in _COLORS if candidate != color),
            key=lambda candidate: math.dist(nodes[color], nodes[candidate]),
        )
        result[color] = other
    return result


def _score(reference: dict[str, str], candidate: dict[str, str]) -> int:
    return sum(reference[color] == candidate.get(color) for color in _COLORS)


def decision_from_image(image: Image.Image) -> str:
    """Recover A or B from colored pixels alone."""
    reference = _partner_map(_centers(image, _PANELS["reference"]))
    candidate_a = _partner_map(_centers(image, _PANELS["A"]))
    candidate_b = _partner_map(_centers(image, _PANELS["B"]))
    scores = (_score(reference, candidate_a), _score(reference, candidate_b))
    if max(scores) != len(_COLORS) or scores[0] == scores[1]:
        raise ValueError("image has no unique nearest-partner correspondence")
    return "A" if scores[0] > scores[1] else "B"
