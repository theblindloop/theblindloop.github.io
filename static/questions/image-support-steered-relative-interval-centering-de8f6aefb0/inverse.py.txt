"""Independent image-only inverse arm for relative interval centering.

This module intentionally knows only the public ink palette and visible glyph
grammar.  It receives no scene, renderer state, source geometry, or gold data.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

CYAN = np.array((14, 165, 184), dtype=np.int16)
ORANGE = np.array((235, 124, 45), dtype=np.int16)
DARK = np.array((35, 42, 52), dtype=np.int16)


@dataclass(frozen=True)
class Component:
    area: int
    x0: int
    y0: int
    x1: int
    y1: int


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 20) -> np.ndarray:
    signed = array.astype(np.int16) - color
    return np.max(np.abs(signed), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[Component]:
    ys, xs = np.where(mask)
    if len(xs) == 0:
        return []
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    local = mask[y0 : y1 + 1, x0 : x1 + 1]
    seen = np.zeros_like(local, dtype=bool)
    found: list[Component] = []
    for sy, sx in np.argwhere(local):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < local.shape[0]
                    and 0 <= nx < local.shape[1]
                    and local[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        py = [point[0] for point in points]
        px = [point[1] for point in points]
        found.append(
            Component(
                len(points),
                min(px) + x0,
                min(py) + y0,
                max(px) + x0,
                max(py) + y0,
            )
        )
    return sorted(found, key=lambda item: (item.x0, item.y0))


def _notch_components(array: np.ndarray) -> list[Component]:
    components = [item for item in _components(_near(array, DARK)) if item.area >= 25]
    if len(components) != 2:
        raise ValueError(f"expected two visible estimate notches, found {len(components)}")
    for item in components:
        width, height = item.x1 - item.x0 + 1, item.y1 - item.y0 + 1
        if item.area < 130 or not (20 <= width <= 22 and 6 <= height <= 8):
            raise ValueError("an estimate notch is materially incomplete")
    return components


def _measure_interval(
    array: np.ndarray, color: np.ndarray, notches: list[Component]
) -> tuple[float, float]:
    mask = _near(array, color)
    pieces = [item for item in _components(mask) if item.area >= 10]
    if len(pieces) != 2 or min(item.area for item in pieces) < 90:
        raise ValueError("a colored interval is missing one of its two visible pieces")

    ys, xs = np.where(mask)
    if len(xs) < 250:
        raise ValueError("too little colored interval ink")
    center_x = float(np.median(xs))
    if int(xs.max() - xs.min() + 1) not in range(28, 31):
        raise ValueError("colored endpoint caps are malformed")

    row_counts = mask.sum(axis=1)
    cap_rows = np.where(row_counts >= 20)[0]
    if len(cap_rows) != 10:
        raise ValueError("both five-row endpoint caps must remain visible")
    if not (np.all(np.diff(cap_rows[:5]) == 1) and np.all(np.diff(cap_rows[5:]) == 1)):
        raise ValueError("endpoint cap rows are discontinuous")
    top = float(np.mean(cap_rows[:5]))
    bottom = float(np.mean(cap_rows[5:]))
    if not 70 <= bottom - top <= 98:
        raise ValueError("visible interval height is outside the public glyph grammar")

    bound = [
        item
        for item in notches
        if abs((item.x0 + item.x1) / 2.0 - center_x) <= 18
        and top < (item.y0 + item.y1) / 2.0 < bottom
    ]
    if len(bound) != 1:
        raise ValueError("could not bind exactly one notch to the colored interval")
    notch = (bound[0].y0 + bound[0].y1) / 2.0
    relative_offset = abs(notch - (top + bottom) / 2.0) / (bottom - top)
    return relative_offset, center_x


def decision_from_image(image: Image.Image) -> str:
    """Recover both normalized offsets from RGB pixels and compare them."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    notches = _notch_components(array)
    cyan, cyan_x = _measure_interval(array, CYAN, notches)
    orange, orange_x = _measure_interval(array, ORANGE, notches)
    if abs(cyan_x - orange_x) < 80:
        raise ValueError("the two interval glyphs are not visibly disconnected")
    if abs(cyan - orange) < 0.005:
        raise ValueError("raster comparison is too close to resolve")
    return "CYAN" if cyan < orange else "ORANGE"
