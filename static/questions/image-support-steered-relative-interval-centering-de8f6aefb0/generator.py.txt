"""Deterministic latent sampler and renderer for relative interval centering."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
CELL = 64
CYAN = (14, 165, 184)
ORANGE = (235, 124, 45)
NOTCH = (35, 42, 52)
BACKGROUND = (244, 247, 250)
CARD = (255, 255, 255)
QUARANTINE_MARGIN = 0.035


def _relative_offsets(scene: dict[str, Any]) -> tuple[float, float]:
    cyan_mid = (scene["cyan_top"] + scene["cyan_bottom"]) / 2.0
    orange_mid = (scene["orange_top"] + scene["orange_bottom"]) / 2.0
    cyan = abs(scene["cyan_notch"] - cyan_mid) / (
        scene["cyan_bottom"] - scene["cyan_top"]
    )
    orange = abs(scene["orange_notch"] - orange_mid) / (
        scene["orange_bottom"] - scene["orange_top"]
    )
    return cyan, orange


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, local two-glyph measurement scene."""

    rng = random.Random(int(seed))
    left_col = rng.choice((1, 2, 3, 4))
    left_x = left_col * CELL + CELL // 2
    right_x = left_x + 2 * CELL
    cyan_left = (seed // 2) % 2 == 0
    cyan_x, orange_x = (left_x, right_x) if cyan_left else (right_x, left_x)

    row = rng.choice((2, 3, 4, 5))
    base_y = row * CELL + CELL // 2
    cyan_center = base_y + rng.randint(-7, 7)
    orange_center = base_y + rng.randint(-7, 7)
    cyan_height = rng.randrange(72, 97, 2)
    orange_height = rng.randrange(72, 97, 2)

    # Every eighth scene is deliberately close to the visible equality boundary.
    target_gap = (0.025, 0.050, 0.065, 0.085, 0.110, 0.140, 0.170, 0.205)[seed % 8]
    winner = "CYAN" if seed % 2 == 0 else "ORANGE"
    for _ in range(100):
        loser_fraction = rng.uniform(max(0.245, target_gap + 0.075), 0.365)
        winner_fraction = loser_fraction - target_gap
        if winner_fraction < 0.035:
            continue
        cyan_fraction = winner_fraction if winner == "CYAN" else loser_fraction
        orange_fraction = winner_fraction if winner == "ORANGE" else loser_fraction
        cyan_offset = max(3, round(cyan_fraction * cyan_height))
        orange_offset = max(3, round(orange_fraction * orange_height))
        cyan_notch = cyan_center + rng.choice((-1, 1)) * cyan_offset
        orange_notch = orange_center + rng.choice((-1, 1)) * orange_offset
        cyan_top = cyan_center - cyan_height // 2
        cyan_bottom = cyan_center + cyan_height // 2
        orange_top = orange_center - orange_height // 2
        orange_bottom = orange_center + orange_height // 2
        cyan_actual = abs(cyan_notch - cyan_center) / cyan_height
        orange_actual = abs(orange_notch - orange_center) / orange_height
        actual_winner = "CYAN" if cyan_actual < orange_actual else "ORANGE"
        if actual_winner == winner and abs(cyan_actual - orange_actual) >= 0.015:
            break
    else:  # pragma: no cover - deterministic ranges above always succeed
        raise RuntimeError("could not sample separated normalized offsets")

    top_extent = min(cyan_top, orange_top) - 25
    bottom_extent = max(cyan_bottom, orange_bottom) + 25
    card = [left_x - 31, top_extent, right_x + 31, bottom_extent]

    distractors: list[list[int]] = []
    attempts = 0
    while len(distractors) < 12 and attempts < 500:
        attempts += 1
        x, y = rng.randint(26, CANVAS - 27), rng.randint(26, CANVAS - 27)
        if card[0] - 24 <= x <= card[2] + 24 and card[1] - 24 <= y <= card[3] + 24:
            continue
        distractors.append([x, y, rng.choice((3, 4, 5))])

    return {
        "canvas_size": [CANVAS, CANVAS],
        "card": card,
        "cyan_x": cyan_x,
        "cyan_top": cyan_top,
        "cyan_bottom": cyan_bottom,
        "cyan_notch": cyan_notch,
        "orange_x": orange_x,
        "orange_top": orange_top,
        "orange_bottom": orange_bottom,
        "orange_notch": orange_notch,
        "distractors": distractors,
    }


def _draw_interval(
    draw: ImageDraw.ImageDraw,
    x: int,
    top: int,
    bottom: int,
    notch: int,
    color: tuple[int, int, int],
) -> None:
    draw.rectangle((x - 4, top, x + 4, bottom), fill=color)
    draw.rectangle((x - 14, top - 2, x + 14, top + 2), fill=color)
    draw.rectangle((x - 14, bottom - 2, x + 14, bottom + 2), fill=color)
    # Overpainting the complete stem width makes the internal estimate tick
    # independently recoverable and leaves two visible colored range pieces.
    draw.rectangle((x - 10, notch - 3, x + 10, notch + 3), fill=NOTCH)


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = scene["canvas_size"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Remote chart-paper context and gray crosses are deliberately nondiagnostic.
    grid = (232, 236, 241)
    for coordinate in range(CELL, width, CELL):
        draw.line((coordinate, 14, coordinate, height - 15), fill=grid, width=1)
    for coordinate in range(CELL, height, CELL):
        draw.line((14, coordinate, width - 15, coordinate), fill=grid, width=1)
    for x, y, size in scene["distractors"]:
        color = (190, 198, 207)
        draw.line((x - size, y, x + size, y), fill=color, width=1)
        draw.line((x, y - size, x, y + size), fill=color, width=1)

    left, top, right, bottom = scene["card"]
    draw.rounded_rectangle(
        (left, top, right, bottom), radius=11, fill=CARD, outline=(211, 218, 226), width=2
    )
    _draw_interval(
        draw,
        scene["cyan_x"],
        scene["cyan_top"],
        scene["cyan_bottom"],
        scene["cyan_notch"],
        CYAN,
    )
    _draw_interval(
        draw,
        scene["orange_x"],
        scene["orange_top"],
        scene["orange_bottom"],
        scene["orange_notch"],
        ORANGE,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    cyan, orange = _relative_offsets(scene)
    return "CYAN" if cyan < orange else "ORANGE"


def margin(scene: dict[str, Any]) -> float:
    cyan, orange = _relative_offsets(scene)
    return abs(cyan - orange)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
