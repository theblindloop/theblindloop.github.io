"""Deterministic renderer and analytic arm for local whisker overlap."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 640
BACKGROUND = (249, 248, 244)
INK = (48, 54, 61)
GRID = (216, 218, 216)
PALETTES = (
    (22, 133, 171),
    (205, 91, 52),
    (77, 139, 72),
    (137, 82, 168),
    (183, 117, 20),
)
PANEL_TONES = (
    (238, 239, 236),
    (235, 238, 240),
    (241, 237, 233),
)
CAP_HALF = 12
HALF_STEM = 3
QUARANTINE_PX = 2


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Return pair-balanced scenes with no label cue in ink, size, or placement."""

    seed = int(seed)
    pair_seed = seed // 2
    rng = random.Random(0x71A5_2026 + pair_seed * 1_000_003)
    x_left = (120, 200, 280, 360)[pair_seed % 4]
    x_right = x_left + 160
    center_y = (240, 320, 400)[(pair_seed // 4) % 3]
    length_left = rng.randint(32, 40)
    length_right = rng.randint(32, 40)
    near_boundary = pair_seed % 6 == 0
    separation = rng.choice((1, 2)) if near_boundary else (7, 9, 11, 13)[pair_seed % 4]

    # The canonical left whisker ends at center_y. The right whisker begins
    # before it for overlap and after it for separation. Odd/even seeds switch
    # only this relation while retaining both glyph lengths and total ink.
    top_left = center_y - length_left
    bottom_left = center_y
    if seed % 2 == 0:
        top_right = center_y - separation
    else:
        top_right = center_y + separation
    bottom_right = top_right + length_right

    # Mirror half the pairs so neither upper/lower direction nor left identity
    # is a shortcut. Reflection preserves interval overlap and all lengths.
    if rng.random() < 0.5:
        mirror = 2 * center_y
        top_left, bottom_left = mirror - bottom_left, mirror - top_left
        top_right, bottom_right = mirror - bottom_right, mirror - top_right

    return {
        "x_left": x_left,
        "x_right": x_right,
        "top_left": top_left,
        "bottom_left": bottom_left,
        "top_right": top_right,
        "bottom_right": bottom_right,
        "whisker_rgb": list(PALETTES[pair_seed % len(PALETTES)]),
        "panel_rgb": list(PANEL_TONES[(pair_seed // 3) % len(PANEL_TONES)]),
    }


def _endpoints(scene: dict[str, Any]) -> tuple[int, int, int, int]:
    top_left = int(scene["top_left"])
    bottom_left = int(scene["bottom_left"])
    top_right = int(scene["top_right"])
    bottom_right = int(scene["bottom_right"])
    if not (top_left < bottom_left and top_right < bottom_right):
        raise ValueError("each interval must have ordered distinct endpoints")
    return top_left, bottom_left, top_right, bottom_right


def _signed_overlap(scene: dict[str, Any]) -> int:
    top_left, bottom_left, top_right, bottom_right = _endpoints(scene)
    return min(bottom_left, bottom_right) - max(top_left, top_right)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if _signed_overlap(scene) >= 0 else "no"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin in pixels from the first-contact boundary."""

    return float(abs(_signed_overlap(scene)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _draw_whisker(draw: ImageDraw.ImageDraw, x: int, top: int, bottom: int, color: tuple[int, int, int]) -> None:
    draw.rectangle((x - HALF_STEM, top, x + HALF_STEM, bottom), fill=color)
    draw.rectangle((x - CAP_HALF, top - HALF_STEM, x + CAP_HALF, top + HALF_STEM), fill=color)
    draw.rectangle((x - CAP_HALF, bottom - HALF_STEM, x + CAP_HALF, bottom + HALF_STEM), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    x_left = int(scene["x_left"])
    x_right = int(scene["x_right"])
    top_left, bottom_left, top_right, bottom_right = _endpoints(scene)
    whisker = tuple(int(value) for value in scene["whisker_rgb"])
    panel = tuple(int(value) for value in scene["panel_rgb"])
    if x_right - x_left != 160 or x_left < 100 or x_right > 540:
        raise ValueError("whiskers must occupy the bounded local inset")
    if min(top_left, top_right) < 120 or max(bottom_left, bottom_right) > 520:
        raise ValueError("whisker endpoints do not fit")
    if len(whisker) != 3 or len(panel) != 3:
        raise ValueError("RGB fields must contain three channels")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    mid_y = round((top_left + bottom_left + top_right + bottom_right) / 4)
    panel_box = (x_left - 66, mid_y - 92, x_right + 66, mid_y + 92)
    draw.rounded_rectangle(panel_box, radius=14, fill=panel, outline=GRID, width=2)
    for offset in (-48, 0, 48):
        y = mid_y + offset
        draw.line((x_left - 42, y, x_right + 42, y), fill=GRID, width=1)
        draw.line((x_left - 51, y, x_left - 43, y), fill=INK, width=2)
    draw.text((x_left - 52, mid_y - 78), "LOCAL UNCERTAINTY WINDOW", font=_font(13, bold=True), fill=INK)
    draw.text((x_left - 7, mid_y + 60), "A", font=_font(15, bold=True), fill=INK)
    draw.text((x_right - 7, mid_y + 60), "B", font=_font(15, bold=True), fill=INK)

    _draw_whisker(draw, x_left, top_left, bottom_left, whisker)
    _draw_whisker(draw, x_right, top_right, bottom_right, whisker)
    return image
