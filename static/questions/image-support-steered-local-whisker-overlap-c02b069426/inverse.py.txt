"""Independent final-PNG inverse arm for capped-whisker interval overlap.

Only saturated pixels are used. The arm discovers two connected glyphs,
validates their complete cap/stem templates, recovers four endpoints from their
bounding boxes, and compares the implied vertical closed intervals. It uses no
renderer state, scene payload, metadata, filenames, or fixed glyph coordinates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _colored_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    return ((high - low) >= 55) & (high >= 105) & (low <= 190)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 40:
            components.append(np.asarray(points, dtype=np.int32))
        else:
            raise ValueError("unexpected colored speckle or damaged whisker fragment")
    return components


def _interval(component: np.ndarray, mask: np.ndarray) -> tuple[float, int, int]:
    ys = component[:, 0]
    xs = component[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width != 25 or not 35 <= height <= 47:
        raise ValueError("colored component has an invalid whisker extent")
    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    expected = np.zeros_like(sub)
    center_x = width // 2
    expected[:, center_x - 3 : center_x + 4] = True
    expected[:7, :] = True
    expected[-7:, :] = True
    if not np.array_equal(sub, expected):
        raise ValueError("whisker cap or stem is incomplete")
    return float((x0 + x1) / 2.0), y0 + 3, y1 - 3


def decision_from_image(image: Image.Image) -> str:
    mask = _colored_mask(image)
    components = _components(mask)
    if len(components) != 2:
        raise ValueError("exactly two disconnected colored whiskers are required")
    intervals = sorted((_interval(component, mask) for component in components), key=lambda item: item[0])
    (x_a, top_a, bottom_a), (x_b, top_b, bottom_b) = intervals
    if x_b - x_a < 60:
        raise ValueError("whiskers are not visibly separated")
    signed_overlap = min(bottom_a, bottom_b) - max(top_a, top_b)
    return "yes" if signed_overlap >= 0 else "no"
