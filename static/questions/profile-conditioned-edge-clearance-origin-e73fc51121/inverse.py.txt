"""Pixel-only decision arm for edge_clearance_origin.

Receives one rasterized PNG and nothing else: no scene, no renderer import, no
analytic gold, no metadata. It segments the black cross and the four colored
disks by exact color, recovers each disk's centre and radius from its component
bounding box, and reports the color whose outline is farthest from the centre of
the cross. It abstains whenever the structure it expects is not present or the
top-two clearance gap is too small to be read reliably from pixels.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "edge_clearance_origin-oracle-1.0.0"

ABSTAIN = "abstain"

_MARK_RGB = {
    "red": (222, 32, 32),
    "green": (24, 148, 64),
    "blue": (36, 72, 214),
    "orange": (240, 152, 24),
}

_DARK = 60  # a cross pixel is dark in every channel
_MIN_DISK_PIXELS = 400
_MIN_CROSS_PIXELS = 60
_MIN_GAP_PX = 1.0  # below this the reading is not stable at this resolution
_FILL_TOLERANCE = 0.06  # a filled disk covers about pi/4 of its bounding box


def _bbox(mask: np.ndarray) -> tuple | None:
    """Bounding box ``(x0, y0, x1, y1, count)`` of a boolean pixel mask."""
    count = int(mask.sum())
    if count == 0:
        return None
    rows = np.flatnonzero(mask.any(axis=1))
    cols = np.flatnonzero(mask.any(axis=0))
    return int(cols[0]), int(rows[0]), int(cols[-1]), int(rows[-1]), count


def decision_from_image(image: Image.Image) -> str:
    data = np.asarray(image.convert("RGB"), dtype=np.int16)
    if data.ndim != 3 or data.shape[2] != 3:
        return ABSTAIN

    cross = _bbox((data < _DARK).all(axis=2))
    if cross is None or cross[4] < _MIN_CROSS_PIXELS:
        return ABSTAIN
    cx0, cy0, cx1, cy1, _ = cross
    if (cx1 - cx0) != (cy1 - cy0) or (cx1 - cx0) < 4:
        # The origin mark is a symmetric cross; anything else is not readable.
        return ABSTAIN
    origin_x = (cx0 + cx1) / 2.0
    origin_y = (cy0 + cy1) / 2.0

    readings = []
    for name, target in sorted(_MARK_RGB.items()):
        found = _bbox((data == np.array(target, dtype=np.int16)).all(axis=2))
        if found is None:
            return ABSTAIN
        x0, y0, x1, y1, count = found
        if count < _MIN_DISK_PIXELS:
            return ABSTAIN
        box_w, box_h = x1 - x0 + 1, y1 - y0 + 1
        if abs(box_w - box_h) > 1:
            # A solid disk has a square bounding box; a clipped, split or
            # partly erased one does not, and its radius is not trustworthy.
            return ABSTAIN
        if abs(count / float(box_w * box_h) - math.pi / 4.0) > _FILL_TOLERANCE:
            return ABSTAIN
        centre_x = (x0 + x1) / 2.0
        centre_y = (y0 + y1) / 2.0
        radius = (box_w + box_h - 2) / 4.0
        clearance = math.hypot(centre_x - origin_x, centre_y - origin_y) - radius
        if clearance <= 0.0:
            return ABSTAIN
        readings.append((clearance, name))

    readings.sort(reverse=True)
    if readings[0][0] - readings[1][0] < _MIN_GAP_PX:
        return ABSTAIN
    return readings[0][1]
