"""Scene sampler, renderer and analytic gold for edge_clearance_origin.

A black cross marks the ORIGIN. Four solid colored disks of different sizes sit
around it and never touch each other or the cross. Each disk's CLEARANCE is the
shortest distance from the centre of the cross to that disk's outline, i.e. the
centre distance minus the disk's radius. The answer names the disk with the
greatest clearance.

Neither the biggest disk nor the disk whose centre is farthest away decides the
answer: the sampler draws a balanced cue profile so that the radius cue and the
centre-distance cue each disagree with the answer in a controlled fraction of
scenes. Everything is drawn with hard-edged Pillow primitives, so the raster
carries the exact centres and radii the analytic arm uses.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "edge_clearance_origin-renderer-1.0.0"

WIDTH = 560
HEIGHT = 420
BACKGROUND = (255, 255, 255)

CROSS_RGB = (0, 0, 0)
CROSS_ARM = 9  # half-length of each cross arm, in pixels
CROSS_HALF_THICK = 1  # half-thickness of each cross arm, in pixels

COLOR_NAMES = ("red", "green", "blue", "orange")
COLOR_RGB = {
    "red": (222, 32, 32),
    "green": (24, 148, 64),
    "blue": (36, 72, 214),
    "orange": (240, 152, 24),
}

MIN_RADIUS = 14
MAX_RADIUS = 40
MIN_CLEARANCE = 18.0  # every disk stays well clear of the cross
MIN_CLEARANCE_GAP = 3.0  # smallest permitted gap between any two clearances
DISK_GAP = 12  # smallest permitted gap between two disk outlines
EDGE_PAD = 8  # smallest permitted gap between a disk and the canvas edge

QUARANTINE_PX = 12.0  # runner-up clearance gap, in pixels

# Balanced cue profile: (answer is the largest disk, answer is the farthest
# centre).  Sampling one profile per scene keeps either shortcut from carrying
# the answer on its own.
CUE_PROFILES = (
    (False, False),
    (False, False),
    (False, True),
    (True, True),
)


def _sample_once(rng: random.Random) -> dict | None:
    origin = (rng.randint(240, 320), rng.randint(180, 240))
    radii = []
    centers = []
    for slot in range(4):
        radius = rng.randint(MIN_RADIUS, MAX_RADIUS)
        angle = math.radians(90.0 * slot + rng.uniform(14.0, 76.0))
        distance = rng.uniform(radius + MIN_CLEARANCE + 2.0, 215.0)
        cx = origin[0] + int(round(distance * math.cos(angle)))
        cy = origin[1] + int(round(distance * math.sin(angle)))
        if not (EDGE_PAD <= cx - radius and cx + radius <= WIDTH - 1 - EDGE_PAD):
            return None
        if not (EDGE_PAD <= cy - radius and cy + radius <= HEIGHT - 1 - EDGE_PAD):
            return None
        radii.append(radius)
        centers.append([cx, cy])

    for a in range(4):
        for b in range(a + 1, 4):
            span = math.hypot(
                centers[a][0] - centers[b][0], centers[a][1] - centers[b][1]
            )
            if span < radii[a] + radii[b] + DISK_GAP:
                return None

    clearances = [
        math.hypot(centers[i][0] - origin[0], centers[i][1] - origin[1]) - radii[i]
        for i in range(4)
    ]
    if min(clearances) < MIN_CLEARANCE:
        return None
    order = sorted(clearances, reverse=True)
    for first, second in zip(order, order[1:]):
        if first - second < MIN_CLEARANCE_GAP:
            return None

    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    return {
        "origin": list(origin),
        "centers": centers,
        "radii": radii,
        "slot_colors": colors,
    }


def sample_scene(seed: int) -> dict:
    """Deterministic scene for ``seed``; rejection-sampled to a cue profile."""
    rng = random.Random(seed)
    wanted = CUE_PROFILES[rng.randrange(len(CUE_PROFILES))]
    for _ in range(20000):
        scene = _sample_once(rng)
        if scene is None:
            continue
        cues = cue_congruence(scene)
        if (cues["largest_disk"], cues["farthest_centre"]) == wanted:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def clearances(scene: dict) -> list[float]:
    """Centre-to-outline distance of each disk, in pixels, by slot."""
    ox, oy = scene["origin"]
    return [
        math.hypot(cx - ox, cy - oy) - radius
        for (cx, cy), radius in zip(scene["centers"], scene["radii"])
    ]


def _winning_slot(scene: dict) -> int:
    values = clearances(scene)
    return max(range(len(values)), key=lambda slot: values[slot])


def analytic_gold(scene: dict) -> str:
    """Colour name of the disk whose nearest edge is farthest from the origin."""
    return scene["slot_colors"][_winning_slot(scene)]


def margin(scene: dict) -> float:
    """Runner-up decision margin: the top-two clearance gap, in pixels."""
    values = sorted(clearances(scene), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_PX)


def cue_congruence(scene: dict) -> dict:
    """Whether each cheap local cue happens to agree with the answer."""
    win = _winning_slot(scene)
    ox, oy = scene["origin"]
    centre_distance = [
        math.hypot(cx - ox, cy - oy) for cx, cy in scene["centers"]
    ]
    radii = scene["radii"]
    tops = [cy - radius for (_, cy), radius in zip(scene["centers"], radii)]
    lefts = [cx - radius for (cx, _), radius in zip(scene["centers"], radii)]
    return {
        "largest_disk": win == max(range(4), key=lambda s: radii[s]),
        "farthest_centre": win == max(range(4), key=lambda s: centre_distance[s]),
        "topmost": win == min(range(4), key=lambda s: tops[s]),
        "leftmost": win == min(range(4), key=lambda s: lefts[s]),
    }


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias exists; see candidate.json latent_alias.justification."""
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for (cx, cy), radius, name in zip(
        scene["centers"], scene["radii"], scene["slot_colors"]
    ):
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=COLOR_RGB[name],
        )

    ox, oy = scene["origin"]
    draw.rectangle(
        [ox - CROSS_ARM, oy - CROSS_HALF_THICK, ox + CROSS_ARM, oy + CROSS_HALF_THICK],
        fill=CROSS_RGB,
    )
    draw.rectangle(
        [ox - CROSS_HALF_THICK, oy - CROSS_ARM, ox + CROSS_HALF_THICK, oy + CROSS_ARM],
        fill=CROSS_RGB,
    )
    return image
