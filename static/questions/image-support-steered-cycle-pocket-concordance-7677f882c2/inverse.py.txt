"""Independent image-only inverse arm for Cycle–Pocket Concordance.

The oracle segments three ink families, recovers four circular graph stations
and the straight links between their centroids, computes E-V+C, independently
counts bounded complement components of the filled teal sheet, and compares
the two counts. It receives no latent values or renderer state.
"""

from __future__ import annotations

from itertools import combinations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _color_masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    teal = (red < 80) & (green < 125) & (blue < 140) & ((blue - red) > 20)
    amber = (
        (red > 160)
        & (green > 65)
        & (green < 165)
        & (blue < 90)
        & ((red - blue) > 115)
    )
    purple = (
        (red > 75)
        & (red < 155)
        & (green < 125)
        & (blue > 135)
        & ((blue - red) > 25)
    )
    return teal, amber, purple


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    remaining = {tuple(int(value) for value in point) for point in np.argwhere(mask)}
    output: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        output.append(component)
    return output


def _sheet_pockets(teal: np.ndarray) -> int:
    sheets = [component for component in _components(teal) if len(component) >= 500]
    if len(sheets) != 1:
        raise OracleAbstention(f"expected one teal sheet, found {len(sheets)}")
    component = sheets[0]
    ys = [point[0] for point in component]
    xs = [point[1] for point in component]
    y0, y1 = min(ys) - 1, max(ys) + 2
    x0, x1 = min(xs) - 1, max(xs) + 2
    if y0 < 0 or x0 < 0 or y1 > teal.shape[0] or x1 > teal.shape[1]:
        raise OracleAbstention("teal sheet touches the image boundary")
    local = np.zeros((y1 - y0, x1 - x0), dtype=bool)
    for y, x in component:
        local[y - y0, x - x0] = True
    holes = 0
    height, width = local.shape
    for region in _components(~local):
        touches_border = any(
            y in {0, height - 1} or x in {0, width - 1} for y, x in region
        )
        if not touches_border and len(region) >= 40:
            holes += 1
    if holes not in {1, 2}:
        raise OracleAbstention(f"unsupported teal pocket count {holes}")
    return holes


def _station_centroids(purple: np.ndarray) -> list[tuple[float, float]]:
    stations = [component for component in _components(purple) if len(component) >= 20]
    if len(stations) != 4:
        raise OracleAbstention(f"expected four purple stations, found {len(stations)}")
    centroids: list[tuple[float, float]] = []
    for component in stations:
        points = np.asarray(component, dtype=np.float64)
        centroids.append((float(points[:, 1].mean()), float(points[:, 0].mean())))
    return centroids


def _edge_present(
    amber: np.ndarray,
    first: tuple[float, float],
    second: tuple[float, float],
) -> bool:
    hits = 0
    samples = np.linspace(0.27, 0.73, 17)
    for fraction in samples:
        x = first[0] + fraction * (second[0] - first[0])
        y = first[1] + fraction * (second[1] - first[1])
        ix, iy = int(round(x)), int(round(y))
        y0, y1 = max(0, iy - 2), min(amber.shape[0], iy + 3)
        x0, x1 = max(0, ix - 2), min(amber.shape[1], ix + 3)
        hits += int(bool(amber[y0:y1, x0:x1].any()))
    return hits >= 13


def _graph_cycle_rank(amber: np.ndarray, purple: np.ndarray) -> int:
    if int(amber.sum()) < 250:
        raise OracleAbstention("too little amber trail ink")
    vertices = _station_centroids(purple)
    edges: list[tuple[int, int]] = []
    for left, right in combinations(range(len(vertices)), 2):
        if _edge_present(amber, vertices[left], vertices[right]):
            edges.append((left, right))
    adjacency = {index: set() for index in range(len(vertices))}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    components = 0
    remaining = set(adjacency)
    while remaining:
        components += 1
        stack = [remaining.pop()]
        while stack:
            vertex = stack.pop()
            for neighbor in adjacency[vertex]:
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
    rank = len(edges) - len(vertices) + components
    if components != 1 or rank not in {1, 2}:
        raise OracleAbstention(
            f"unsupported graph topology E={len(edges)} V={len(vertices)} C={components}"
        )
    return rank


def decision_from_image(image: Image.Image) -> str:
    teal, amber, purple = _color_masks(image)
    cycle_rank = _graph_cycle_rank(amber, purple)
    pocket_count = _sheet_pockets(teal)
    return "yes" if cycle_rank == pocket_count else "no"
