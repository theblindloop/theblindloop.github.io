"""Deterministic analytic and rendering arm for Cycle–Pocket Concordance."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
CELL = 64
SHEET_SIZE = (40, 48)
STATION_RADIUS = 5
LINK_WIDTH = 5
MIN_CLEARANCE = 5.0


def _rect(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    size: tuple[int, int],
    fill: str,
) -> None:
    cx, cy = center
    width, height = size
    x0 = cx - width // 2
    y0 = cy - height // 2
    draw.rectangle((x0, y0, x0 + width - 1, y0 + height - 1), fill=fill)


def _sample_decorations(
    rng: random.Random,
    graph_center: tuple[int, int],
    sheet_center: tuple[int, int],
) -> list[list[int]]:
    decorations: list[list[int]] = []
    attempts = 0
    while len(decorations) < 7 and attempts < 500:
        attempts += 1
        x = rng.randrange(28, CANVAS - 28)
        y = rng.randrange(28, CANVAS - 28)
        if min(math.dist((x, y), graph_center), math.dist((x, y), sheet_center)) < 105:
            continue
        if any(math.dist((x, y), (old_x, old_y)) < 25 for old_x, old_y, _ in decorations):
            continue
        decorations.append([x, y, rng.randrange(2)])
    return decorations


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC1C1E57)
    combinations = ((1, 1), (1, 2), (2, 1), (2, 2))
    cycle_rank, pocket_count = combinations[int(seed) % len(combinations)]
    pair_axis = "horizontal" if rng.randrange(2) == 0 else "vertical"
    if pair_axis == "horizontal":
        row = rng.randrange(1, 7)
        first = rng.randrange(1, 5)
        centers = [
            (first * CELL + CELL // 2, row * CELL + CELL // 2),
            ((first + 2) * CELL + CELL // 2, row * CELL + CELL // 2),
        ]
    else:
        column = rng.randrange(1, 7)
        first = rng.randrange(1, 5)
        centers = [
            (column * CELL + CELL // 2, first * CELL + CELL // 2),
            (column * CELL + CELL // 2, (first + 2) * CELL + CELL // 2),
        ]
    if rng.randrange(2):
        centers.reverse()
    graph_center, sheet_center = centers
    palettes = (
        ("#174d5b", "#cf7a1d", "#7853a6", "#d8dee2"),
        ("#205466", "#d48324", "#7051a3", "#d2dadd"),
        ("#1f5960", "#c96e2c", "#8355aa", "#dde1e4"),
    )
    teal, amber, purple, decoration_color = rng.choice(palettes)
    hole_axis = "horizontal" if rng.randrange(2) == 0 else "vertical"
    diagonal = "none"
    if cycle_rank == 2:
        diagonal = "main" if rng.randrange(2) == 0 else "anti"
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": "#ffffff",
        "teal": teal,
        "amber": amber,
        "purple": purple,
        "decoration_color": decoration_color,
        "cycle_rank": cycle_rank,
        "pocket_count": pocket_count,
        "graph_center": list(graph_center),
        "sheet_center": list(sheet_center),
        "pair_axis": pair_axis,
        "hole_axis": hole_axis,
        "diagonal": diagonal,
        "decorations": _sample_decorations(rng, graph_center, sheet_center),
    }


def _draw_sheet(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    count: int,
    axis: str,
    teal: str,
    background: str,
) -> None:
    _rect(draw, center, SHEET_SIZE, teal)
    cx, cy = center
    if count == 1:
        _rect(draw, center, (20, 20), background)
        return
    if count != 2:
        raise ValueError(f"unsupported pocket_count={count!r}")
    if axis == "horizontal":
        for pocket_center in ((cx - 8, cy), (cx + 8, cy)):
            _rect(draw, pocket_center, (10, 20), background)
    elif axis == "vertical":
        for pocket_center in ((cx, cy - 8), (cx, cy + 8)):
            _rect(draw, pocket_center, (20, 10), background)
    else:
        raise ValueError(f"unsupported hole_axis={axis!r}")


def _graph_geometry(center: tuple[int, int], cycle_rank: int, diagonal: str):
    cx, cy = center
    vertices = [
        (cx - 16, cy - 16),
        (cx + 16, cy - 16),
        (cx + 16, cy + 16),
        (cx - 16, cy + 16),
    ]
    edges = [(0, 1), (1, 2), (2, 3), (3, 0)]
    if cycle_rank == 2:
        if diagonal == "main":
            edges.append((0, 2))
        elif diagonal == "anti":
            edges.append((1, 3))
        else:
            raise ValueError("rank-two graph requires a visible diagonal")
    elif cycle_rank != 1 or diagonal != "none":
        raise ValueError("rank-one graph must have no diagonal")
    return vertices, edges


def _draw_graph(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    cycle_rank: int,
    diagonal: str,
    amber: str,
    purple: str,
) -> None:
    vertices, edges = _graph_geometry(center, cycle_rank, diagonal)
    for left, right in edges:
        draw.line((vertices[left], vertices[right]), fill=amber, width=LINK_WIDTH)
    for x, y in vertices:
        draw.ellipse(
            (x - STATION_RADIUS, y - STATION_RADIUS, x + STATION_RADIUS, y + STATION_RADIUS),
            fill=purple,
        )


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = str(scene["background"])
    teal = str(scene["teal"])
    amber = str(scene["amber"])
    purple = str(scene["purple"])
    decoration_color = str(scene["decoration_color"])
    cycle_rank = int(scene["cycle_rank"])
    pocket_count = int(scene["pocket_count"])
    graph_center = tuple(int(value) for value in scene["graph_center"])
    sheet_center = tuple(int(value) for value in scene["sheet_center"])
    pair_axis = str(scene["pair_axis"])
    hole_axis = str(scene["hole_axis"])
    diagonal = str(scene["diagonal"])
    decorations = list(scene["decorations"])
    if pair_axis not in {"horizontal", "vertical"}:
        raise ValueError(f"unsupported pair_axis={pair_axis!r}")

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for x, y, kind in decorations:
        x, y, kind = int(x), int(y), int(kind)
        if kind == 0:
            draw.rectangle((x - 4, y - 4, x + 3, y + 3), outline=decoration_color, width=2)
        else:
            draw.rectangle((x - 1, y - 5, x, y + 4), fill=decoration_color)
            draw.rectangle((x - 5, y - 1, x + 4, y), fill=decoration_color)
    _draw_graph(draw, graph_center, cycle_rank, diagonal, amber, purple)
    _draw_sheet(draw, sheet_center, pocket_count, hole_axis, teal, background)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["cycle_rank"]) == int(scene["pocket_count"]) else "no"


def margin(scene: dict) -> float:
    graph_center = tuple(int(value) for value in scene["graph_center"])
    sheet_center = tuple(int(value) for value in scene["sheet_center"])
    pocket_count = int(scene["pocket_count"])
    hole_axis = str(scene["hole_axis"])
    cycle_rank = int(scene["cycle_rank"])
    diagonal = str(scene["diagonal"])
    width = int(scene["width"])
    height = int(scene["height"])
    separator = 6.0 if pocket_count == 2 else 10.0
    graph_clearance = float(min(LINK_WIDTH, STATION_RADIUS))
    pair_clearance = math.dist(graph_center, sheet_center) - 48.0
    border_clearance = min(
        graph_center[0] - 22,
        width - (graph_center[0] + 22),
        graph_center[1] - 22,
        height - (graph_center[1] + 22),
        sheet_center[0] - 20,
        width - (sheet_center[0] + 20),
        sheet_center[1] - 24,
        height - (sheet_center[1] + 24),
    )
    structural_check = 8.0 if hole_axis in {"horizontal", "vertical"} else 0.0
    diagonal_check = 8.0 if (cycle_rank, diagonal) in {(1, "none"), (2, "main"), (2, "anti")} else 0.0
    return float(
        min(
            separator,
            graph_clearance,
            pair_clearance,
            border_clearance,
            structural_check,
            diagonal_check,
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
