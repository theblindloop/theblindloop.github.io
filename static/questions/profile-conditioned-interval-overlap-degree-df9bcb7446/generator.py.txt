"""Deterministic interval-overlap graph renderer and analytic arm."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

PALETTE = {
    "blue": (38, 106, 196),
    "orange": (230, 126, 34),
    "green": (35, 155, 86),
    "purple": (132, 78, 190),
}
COLOR_NAMES = tuple(PALETTE)


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one robust star-shaped interval graph with counterbalanced label."""
    rng = random.Random(int(seed))
    winner = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    others = [color for color in COLOR_NAMES if color != winner]
    rng.shuffle(others)

    shift = rng.randint(-30, 30)
    hub_left = 324 + shift + rng.randint(-7, 7)
    hub_right = 574 + shift + rng.randint(-7, 7)
    geometries = [
        (hub_left, hub_right),
        (84 + shift + rng.randint(-10, 10), hub_left + rng.randint(28, 45)),
        (hub_left + rng.randint(76, 92), hub_left + rng.randint(126, 145)),
        (hub_left + rng.randint(176, 194), 812 + shift + rng.randint(-10, 10)),
    ]
    colors = [winner, *others]
    rows = [0, 1, 2, 3]
    rng.shuffle(rows)
    intervals = [
        {"color": color, "x0": int(pair[0]), "x1": int(pair[1]), "row": row}
        for color, pair, row in zip(colors, geometries, rows)
    ]
    if rng.random() < 0.5:
        # A visible reflection prevents fixed left/right endpoint templates.
        for item in intervals:
            item["x0"], item["x1"] = 896 - item["x1"], 896 - item["x0"]
    return {"width": 900, "height": 540, "intervals": intervals}


def _degrees(intervals: list[dict[str, Any]]) -> dict[str, int]:
    degree = {str(item["color"]): 0 for item in intervals}
    for index, left in enumerate(intervals):
        for right in intervals[index + 1 :]:
            overlap = min(int(left["x1"]), int(right["x1"])) - max(
                int(left["x0"]), int(right["x0"])
            )
            if overlap > 0:
                degree[str(left["color"])] += 1
                degree[str(right["color"])] += 1
    return degree


def analytic_gold(scene: dict[str, Any]) -> str:
    degree = _degrees(scene["intervals"])
    ordered = sorted(degree, key=lambda color: (-degree[color], color))
    if len(ordered) < 2 or degree[ordered[0]] == degree[ordered[1]]:
        raise ValueError("scene has no unique overlap-degree leader")
    return ordered[0]


def margin(scene: dict[str, Any]) -> float:
    values = sorted(_degrees(scene["intervals"]).values(), reverse=True)
    return float(values[0] - values[1])


def _clearance(scene: dict[str, Any]) -> int:
    intervals = scene["intervals"]
    clearances = []
    for index, left in enumerate(intervals):
        for right in intervals[index + 1 :]:
            overlap = min(int(left["x1"]), int(right["x1"])) - max(
                int(left["x0"]), int(right["x0"])
            )
            clearances.append(abs(overlap))
    return min(clearances) if clearances else 0


def is_quarantined(scene: dict[str, Any]) -> bool:
    intervals = scene["intervals"]
    if len(intervals) != 4 or {item.get("color") for item in intervals} != set(COLOR_NAMES):
        return True
    if any(int(item["x1"]) - int(item["x0"]) < 30 for item in intervals):
        return True
    return margin(scene) < 1.0 or _clearance(scene) < 16


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    intervals = scene["intervals"]
    image = Image.new("RGB", (width, height), (250, 251, 253))
    draw = ImageDraw.Draw(image)
    draw.text((42, 28), "INTERVAL OVERLAP MAP", fill=(24, 31, 43), font=_font(25, True))
    draw.text(
        (42, 68),
        "Count how many other intervals each colored interval overlaps",
        fill=(70, 78, 91),
        font=_font(17),
    )
    plot_left, plot_right, plot_top, plot_bottom = 62, width - 42, 116, height - 70
    draw.rounded_rectangle(
        (plot_left, plot_top, plot_right, plot_bottom),
        radius=10,
        fill=(255, 255, 255),
        outline=(210, 216, 226),
        width=2,
    )
    for x in range(100, 851, 100):
        draw.line((x, plot_top + 2, x, plot_bottom - 2), fill=(230, 234, 240), width=1)
    row_y = [164, 242, 320, 398]
    for row, y in enumerate(row_y, start=1):
        draw.text((76, y - 10), str(row), fill=(104, 111, 123), font=_font(15, True))
        draw.line((105, y, plot_right - 18, y), fill=(239, 241, 245), width=2)
    for item in intervals:
        x0, x1 = int(item["x0"]), int(item["x1"])
        y = row_y[int(item["row"])]
        color = PALETTE[str(item["color"])]
        draw.line((x0, y, x1, y), fill=color, width=11)
        draw.line((x0, y - 14, x0, y + 14), fill=color, width=7)
        draw.line((x1, y - 14, x1, y + 14), fill=color, width=7)
    draw.text(
        (62, height - 45),
        "Intervals overlap when they share a positive-width horizontal span.",
        fill=(70, 78, 91),
        font=_font(15),
    )
    return image
