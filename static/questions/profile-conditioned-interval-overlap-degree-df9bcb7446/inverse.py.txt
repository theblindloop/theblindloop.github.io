"""Independent PNG-only inverse arm for interval overlap degree."""

from __future__ import annotations

from PIL import Image

PALETTE = {
    "blue": (38, 106, 196),
    "orange": (230, 126, 34),
    "green": (35, 155, 86),
    "purple": (132, 78, 190),
}


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    pixels = rgb.load()
    spans: dict[str, tuple[int, int]] = {}
    for name, target in PALETTE.items():
        xs: list[int] = []
        count = 0
        for y in range(rgb.height):
            for x in range(rgb.width):
                if pixels[x, y] == target:
                    xs.append(x)
                    count += 1
        if count < 250 or not xs:
            raise ValueError(f"missing or damaged {name} interval")
        x0, x1 = min(xs), max(xs)
        if x1 - x0 < 28:
            raise ValueError(f"unstable {name} interval")
        spans[name] = (x0, x1)

    degree = {name: 0 for name in PALETTE}
    names = list(PALETTE)
    for index, left_name in enumerate(names):
        left = spans[left_name]
        for right_name in names[index + 1 :]:
            right = spans[right_name]
            # Three pixels of tolerance keeps endpoint-only contact excluded.
            if min(left[1], right[1]) - max(left[0], right[0]) > 3:
                degree[left_name] += 1
                degree[right_name] += 1
    ordered = sorted(names, key=lambda name: (-degree[name], name))
    if degree[ordered[0]] == degree[ordered[1]]:
        raise ValueError("ambiguous overlap-degree leader")
    return ordered[0]
