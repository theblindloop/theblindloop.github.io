"""Deterministic bowling-ball grip renderer and analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
CELL_SIZE = 64
SUPERSAMPLE = 4
BELOW_THRESHOLD_PX = 28.0

BACKGROUND_TONES = (
    (247, 244, 237),
    (240, 245, 247),
    (247, 241, 246),
)
BALL_PALETTE = (
    (48, 116, 213),
    (174, 65, 190),
    (34, 157, 128),
    (222, 104, 45),
)
HOLE_RIM = (82, 91, 102)
HOLE_CORE = (21, 27, 34)


def sample_scene(seed: int) -> dict:
    """Return adjacent matched control/counterfactual pairs.

    Seeds 2k and 2k+1 share every nuisance field and the same multiset of
    aperture radii.  The even scene assigns the large thumb aperture to the
    lower slot; the odd scene assigns it to one of the upper slots.
    """

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0xB0111A9)
    left_cell = rng.randrange(1, 5)
    top_row = rng.randrange(2, 5)
    top_y = top_row * CELL_SIZE + CELL_SIZE // 2
    bottom_y = (top_row + 1) * CELL_SIZE + CELL_SIZE // 2
    left_x = left_cell * CELL_SIZE + CELL_SIZE // 2
    middle_x = (left_cell + 1) * CELL_SIZE + CELL_SIZE // 2
    right_x = (left_cell + 2) * CELL_SIZE + CELL_SIZE // 2

    small_radius = 12 + (pair % 2)
    large_radius = 18 + (pair % 2)
    if seed % 2 == 0:
        radii = (small_radius, small_radius, large_radius)
    elif (pair // 2) % 2 == 0:
        radii = (large_radius, small_radius, small_radius)
    else:
        radii = (small_radius, large_radius, small_radius)

    holes = [
        [left_x, top_y, radii[0]],
        [right_x, top_y, radii[1]],
        [middle_x, bottom_y, radii[2]],
    ]
    return {
        "background_index": pair % len(BACKGROUND_TONES),
        "ball_color_index": (pair // 3) % len(BALL_PALETTE),
        "ball_center": [middle_x, top_y + CELL_SIZE // 2],
        "ball_radius": 146 + (pair % 3) * 2,
        "canvas_size": CANVAS_SIZE,
        "gloss_variant": rng.randrange(6),
        "holes": holes,
        "lane_variant": rng.randrange(12),
    }


def _box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def _shade(color: tuple[int, int, int], amount: int) -> tuple[int, int, int]:
    return tuple(max(0, min(255, channel + amount)) for channel in color)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    background = BACKGROUND_TONES[int(scene["background_index"]) % len(BACKGROUND_TONES)]
    ball_color = BALL_PALETTE[int(scene["ball_color_index"]) % len(BALL_PALETTE)]
    ball_x, ball_y = (int(value) for value in scene["ball_center"])
    ball_radius = int(scene["ball_radius"])
    lane_variant = int(scene["lane_variant"])
    gloss_variant = int(scene["gloss_variant"])
    holes = [[int(value) for value in hole] for hole in scene["holes"]]
    if len(holes) != 3:
        raise ValueError("the bowling grip must contain exactly three holes")

    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Nondiagnostic lane boards and pin-deck marks are achromatic and remote.
    board = _shade(background, -17)
    seam = _shade(background, -27)
    for index in range(4):
        x = 22 + ((lane_variant * 47 + index * 139) % 468)
        draw.line(
            [(x * ss, 0), ((x + ((index % 2) * 18 - 9)) * ss, size * ss)],
            fill=board,
            width=2 * ss,
        )
    for index in range(3):
        x = 42 + ((lane_variant * 31 + index * 181) % 425)
        y = 32 if (lane_variant + index) % 2 == 0 else size - 31
        draw.ellipse(_box((x - 7, y - 3, x + 7, y + 3), ss), fill=seam)

    # A soft shadow and colored spherical silhouette make the object legible.
    draw.ellipse(
        _box(
            (
                ball_x - ball_radius + 13,
                ball_y + ball_radius - 12,
                ball_x + ball_radius + 21,
                ball_y + ball_radius + 13,
            ),
            ss,
        ),
        fill=(191, 190, 187),
    )
    draw.ellipse(
        _box(
            (
                ball_x - ball_radius,
                ball_y - ball_radius,
                ball_x + ball_radius,
                ball_y + ball_radius,
            ),
            ss,
        ),
        fill=ball_color,
        outline=_shade(ball_color, -31),
        width=4 * ss,
    )

    # Gloss varies independently of the answer and never enters the dark mask.
    gloss_dx = (gloss_variant % 3) * 4
    gloss_dy = (gloss_variant // 3) * 5
    draw.arc(
        _box(
            (
                ball_x - ball_radius + 30 + gloss_dx,
                ball_y - ball_radius + 28 + gloss_dy,
                ball_x + 20 + gloss_dx,
                ball_y + 18 + gloss_dy,
            ),
            ss,
        ),
        start=198,
        end=282,
        fill=_shade(ball_color, 63),
        width=7 * ss,
    )
    draw.arc(
        _box(
            (
                ball_x - ball_radius + 48 + gloss_dx,
                ball_y - ball_radius + 47 + gloss_dy,
                ball_x - 9 + gloss_dx,
                ball_y - 3 + gloss_dy,
            ),
            ss,
        ),
        start=202,
        end=263,
        fill=_shade(ball_color, 88),
        width=3 * ss,
    )

    # The three nonoverlapping apertures are the only near-black components.
    for x, y, radius in holes:
        draw.ellipse(
            _box((x - radius - 4, y - radius - 4, x + radius + 4, y + radius + 4), ss),
            fill=HOLE_RIM,
        )
        draw.ellipse(
            _box((x - radius, y - radius, x + radius, y + radius), ss),
            fill=HOLE_CORE,
        )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def _hole_relation(scene: dict) -> tuple[float, float]:
    holes = [[float(value) for value in hole] for hole in scene["holes"]]
    if len(holes) != 3:
        raise ValueError("expected three holes")
    ordered = sorted(holes, key=lambda hole: hole[2], reverse=True)
    area_gap = math.pi * (ordered[0][2] ** 2 - ordered[1][2] ** 2)
    vertical_delta = ordered[0][1] - max(hole[1] for hole in ordered[1:])
    return area_gap, vertical_delta


def analytic_gold(scene: dict) -> str:
    holes = [[float(value) for value in hole] for hole in scene["holes"]]
    ordered = sorted(holes, key=lambda hole: hole[2], reverse=True)
    largest_y = ordered[0][1]
    other_y = max(hole[1] for hole in ordered[1:])
    return "yes" if largest_y - other_y > BELOW_THRESHOLD_PX else "no"


def margin(scene: dict) -> float:
    area_gap, vertical_delta = _hole_relation(scene)
    return float(min(area_gap, abs(vertical_delta - BELOW_THRESHOLD_PX)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["holes"] = list(reversed(twin["holes"]))
    return [("reverse_serialized_hole_records", twin)]
