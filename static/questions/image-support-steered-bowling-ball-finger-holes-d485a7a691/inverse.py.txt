"""Independent image-only inverse for bowling-ball grip hierarchy."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MAX_DARK_CHANNEL = 60
_MIN_COMPONENT_AREA = 330
_MAX_COMPONENT_AREA = 1350
_MIN_AREA_GAP = 180
_BELOW_THRESHOLD_PX = 28.0
_BOUNDARY_BAND_PX = 10.0


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _dark_apertures(array: np.ndarray) -> list[dict[str, float]]:
    mask = array.max(axis=2) <= _MAX_DARK_CHANNEL
    apertures: list[dict[str, float]] = []
    for points in _components(mask):
        area = len(points)
        if area < 40:
            continue
        ys, xs = points[:, 0], points[:, 1]
        height = int(ys.max() - ys.min() + 1)
        width = int(xs.max() - xs.min() + 1)
        fill = area / float(width * height)
        aspect = width / float(height)
        if not _MIN_COMPONENT_AREA <= area <= _MAX_COMPONENT_AREA:
            raise OracleAbstention(f"dark component has unsupported area {area}")
        if not (20 <= width <= 42 and 20 <= height <= 42):
            raise OracleAbstention("dark component has unsupported diameter")
        if not (0.82 <= aspect <= 1.20 and 0.68 <= fill <= 0.88):
            raise OracleAbstention("dark component is not a complete aperture")
        apertures.append(
            {
                "area": float(area),
                "x": float(xs.mean()),
                "y": float(ys.mean()),
            }
        )
    if len(apertures) != 3:
        raise OracleAbstention(f"expected three dark apertures, found {len(apertures)}")

    xs = [item["x"] for item in apertures]
    ys = [item["y"] for item in apertures]
    if max(xs) - min(xs) > 175.0 or max(ys) - min(ys) > 90.0:
        raise OracleAbstention("apertures do not form one compact grip")
    return apertures


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise OracleAbstention("raster is too small")
    apertures = sorted(_dark_apertures(array), key=lambda item: item["area"], reverse=True)
    area_gap = apertures[0]["area"] - apertures[1]["area"]
    if area_gap < _MIN_AREA_GAP:
        raise OracleAbstention("no uniquely largest aperture")
    vertical_delta = apertures[0]["y"] - max(item["y"] for item in apertures[1:])
    if abs(vertical_delta - _BELOW_THRESHOLD_PX) <= _BOUNDARY_BAND_PX:
        raise OracleAbstention("largest aperture is too close to the below boundary")
    return "yes" if vertical_delta > _BELOW_THRESHOLD_PX else "no"
