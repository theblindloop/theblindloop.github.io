"""Analytic sampler and Pillow renderer for Dual-Frame Selector Replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT, SCALE = 720, 440, 3
TOKEN_RGB = {
    "red": (220, 55, 55),
    "green": (45, 165, 85),
    "blue": (65, 105, 225),
    "yellow": (235, 195, 45),
}
CARD_RGB = {"pointer": (52, 145, 220), "tray": (238, 143, 50)}
MAGENTA = (210, 45, 150)


def _outcome(tray: list[str], pointer: int, operations: list[dict]) -> str:
    current = list(tray)
    aimed = int(pointer)
    for operation in operations:
        step = 1 if operation["direction"] == "cw" else -1
        if operation["frame"] == "pointer":
            aimed = (aimed + step) % 4
        else:
            moved = [""] * 4
            for old_index, color in enumerate(current):
                moved[(old_index + step) % 4] = color
            current = moved
    return current[aimed]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    tray = list(TOKEN_RGB)
    rng.shuffle(tray)
    pointer = rng.randrange(4)
    operations = []
    # Force both moving frames to appear, then randomize the full ordered stream.
    frames = ["pointer", "tray"] + [rng.choice(("pointer", "tray")) for _ in range(4)]
    rng.shuffle(frames)
    for frame in frames:
        operations.append({"frame": frame, "direction": rng.choice(("cw", "ccw"))})
    return {"tray": tray, "pointer": pointer, "operations": operations}


def render(scene: dict) -> Image.Image:
    tray = list(scene["tray"])
    pointer = int(scene["pointer"])
    operations = list(scene["operations"])
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (250, 249, 246))
    draw = ImageDraw.Draw(image)

    def box(coords, **kwargs):
        draw.rounded_rectangle(tuple(int(v * SCALE) for v in coords), radius=9 * SCALE, **kwargs)

    cx, cy, radius = 180, 170, 102
    draw.ellipse(
        ((cx - radius) * SCALE, (cy - radius) * SCALE,
         (cx + radius) * SCALE, (cy + radius) * SCALE),
        fill=(236, 234, 227), outline=(125, 124, 120), width=4 * SCALE,
    )
    slot_centers = ((cx, cy - 78), (cx + 78, cy), (cx, cy + 78), (cx - 78, cy))
    for color, (x, y) in zip(tray, slot_centers, strict=True):
        draw.ellipse(
            ((x - 23) * SCALE, (y - 23) * SCALE, (x + 23) * SCALE, (y + 23) * SCALE),
            fill=TOKEN_RGB[color], outline=(255, 255, 255), width=4 * SCALE,
        )

    # The sole near-black component is an asymmetric pointer; its centroid points
    # in the selected direction and is independently measurable from pixels.
    pointer_polygons = (
        ((cx - 13, cy + 14), (cx + 13, cy + 14), (cx, cy - 60)),
        ((cx - 14, cy - 13), (cx - 14, cy + 13), (cx + 60, cy)),
        ((cx - 13, cy - 14), (cx + 13, cy - 14), (cx, cy + 60)),
        ((cx + 14, cy - 13), (cx + 14, cy + 13), (cx - 60, cy)),
    )
    draw.polygon([(x * SCALE, y * SCALE) for x, y in pointer_polygons[pointer]], fill=(28, 31, 36))
    draw.ellipse(
        ((cx - 13) * SCALE, (cy - 13) * SCALE, (cx + 13) * SCALE, (cy + 13) * SCALE),
        fill=(28, 31, 36),
    )

    # Six operation cards. Blue moves the pointer; orange moves the tray. The
    # magenta dot is left for CCW and right for CW.
    start_x, card_y, card_w, gap = 43, 326, 94, 20
    for index, operation in enumerate(operations):
        x0 = start_x + index * (card_w + gap)
        box((x0, card_y, x0 + card_w, card_y + 78), fill=CARD_RGB[operation["frame"]])
        dot_x = x0 + (70 if operation["direction"] == "cw" else 24)
        draw.ellipse(
            ((dot_x - 10) * SCALE, (card_y + 29) * SCALE,
             (dot_x + 10) * SCALE, (card_y + 49) * SCALE),
            fill=MAGENTA,
        )
        # A pale center hub reinforces that each card means a quarter-turn.
        draw.ellipse(
            ((x0 + 41) * SCALE, (card_y + 34) * SCALE,
             (x0 + 53) * SCALE, (card_y + 46) * SCALE),
            fill=(250, 249, 246),
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _outcome(list(scene["tray"]), int(scene["pointer"]), list(scene["operations"]))


def margin(scene: dict) -> float:
    # Minimum nominal gap between independently parsed visual components.
    return 20.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
