"""Dash-count ring oddball search world.

Four equal-size dark, dashed circles (rings) are drawn, one in each quadrant of
the image. Each ring is broken into a whole number of equally spaced dashes of
equal arc length with equal gaps between them. Three of the rings share the same
number of dashes, while the fourth ring has a different (unique) number of
dashes. The decision is the quadrant containing the ring whose dash count is
unique. All geometry, dash counts, and the decision are visible in the raster;
rendering is deterministic.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
INK = (30, 30, 30)
R = 90        # radius of every ring, in pixels
WIDTH = 7     # line width of every dash, in pixels

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 120, "TR": 360, "BL": 120, "BR": 360}
SECTOR_CY = {"TL": 120, "TR": 120, "BL": 360, "BR": 360}

DASH_COUNTS = (5, 6, 7, 8, 9, 10, 11)
QUARANTINE_THRESH = 2.0  # count-units margin below which a scene is quarantined


def _dashed_ring(cx: int, cy: int, n: int) -> list[tuple[float, float]]:
    step = 360.0 / n
    dash = step / 2.0
    return [(k * step, k * step + dash) for k in range(n)]


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    winner_count = int(DASH_COUNTS[rng.integers(0, len(DASH_COUNTS))])
    offsets = [o for o in (-4, -3, -2, -1, 1, 2, 3, 4)
               if winner_count + o in DASH_COUNTS]
    distractor_count = winner_count + int(offsets[rng.integers(0, len(offsets))])

    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    rings = [
        {
            "sector": sector,
            "count": winner_count if sector == winner else distractor_count,
            "cx": SECTOR_CX[sector],
            "cy": SECTOR_CY[sector],
        }
        for sector in SECTORS
    ]
    return {"rings": rings}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    for ring in scene["rings"]:
        arcs = _dashed_ring(ring["cx"], ring["cy"], ring["count"])
        bbox = [
            ring["cx"] - R,
            ring["cy"] - R,
            ring["cx"] + R,
            ring["cy"] + R,
        ]
        for start, end in arcs:
            draw.arc(bbox, start, end, fill=INK, width=WIDTH)
    return img


def _decision(scene: dict) -> tuple[str, float]:
    freq: dict[int, int] = {}
    for ring in scene["rings"]:
        freq[ring["count"]] = freq.get(ring["count"], 0) + 1
    unique = [ring["sector"] for ring in scene["rings"] if freq[ring["count"]] == 1]
    if len(unique) != 1:
        return "TL", 0.0
    winner = unique[0]
    winner_count = next(ring["count"] for ring in scene["rings"] if ring["sector"] == winner)
    others = [ring["count"] for ring in scene["rings"] if ring["sector"] != winner]
    margin = float(min(abs(winner_count - other) for other in others))
    return winner, margin


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
