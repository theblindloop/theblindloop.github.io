"""Independent pixel-only inverse arm for dash_count_ring_oddball.

Receives only a PIL image and the public question contract. It separates the
image into four quadrant windows, extracts the dark ring pixels in each window,
and counts the number of distinct disconnected dashes (line segments) that make
up each dashed ring. It then returns the quadrant whose dash count occurs
exactly once among the four.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

from collections import deque

import numpy as np

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 120, "TR": 360, "BL": 120, "BR": 360}
SECTOR_CY = {"TL": 120, "TR": 120, "BL": 360, "BR": 360}
WIN = 100          # half-width of each quadrant analysis window, in pixels
DARK = 150         # grayscale threshold below which a pixel counts as ink


def _count_components(binary: np.ndarray) -> int:
    h, w = binary.shape
    seen = np.zeros((h, w), dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if not binary[y, x] or seen[y, x]:
                continue
            count += 1
            queue = deque([(y, x)])
            seen[y, x] = True
            while queue:
                cy, cx = queue.popleft()
                for dy in (-1, 0, 1):
                    for dx in (-1, 0, 1):
                        ny, nx = cy + dy, cx + dx
                        if (
                            0 <= ny < h
                            and 0 <= nx < w
                            and binary[ny, nx]
                            and not seen[ny, nx]
                        ):
                            seen[ny, nx] = True
                            queue.append((ny, nx))
    return count


def _dash_count(arr: np.ndarray, cx: int, cy: int) -> int:
    x0, x1 = cx - WIN, cx + WIN
    y0, y1 = cy - WIN, cy + WIN
    window = arr[y0:y1, x0:x1]
    binary = np.min(window, axis=2) < DARK
    if binary.sum() < 20:
        return 0
    return _count_components(binary)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    counts = {
        sector: _dash_count(arr, SECTOR_CX[sector], SECTOR_CY[sector]) for sector in SECTORS
    }
    values = list(counts.values())
    unique_value = next((v for v in values if values.count(v) == 1), None)
    if unique_value is None:
        return "NONE"
    for sector in SECTORS:
        if counts[sector] == unique_value:
            return sector
    return "NONE"
