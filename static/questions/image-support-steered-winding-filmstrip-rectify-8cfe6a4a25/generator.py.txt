"""Deterministic winding-filmstrip visual world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("X", "plus", "T", "L")

TEMPLATES = {
    "X": (
        "1000001", "0100010", "0010100", "0001000",
        "0010100", "0100010", "1000001",
    ),
    "plus": (
        "0001000", "0001000", "0001000", "1111111",
        "0001000", "0001000", "0001000",
    ),
    "T": (
        "1111111", "0001000", "0001000", "0001000",
        "0001000", "0001000", "0001000",
    ),
    "L": (
        "1000000", "1000000", "1000000", "1000000",
        "1000000", "1000000", "1111111",
    ),
}

# One seven-frame chain. The other variants are rigid rotations/reflections,
# which counterbalance route geometry without changing the transported glyph.
_BASE_ROUTE = ((-116, -58), (-58, -58), (0, -58), (0, 0), (58, 0), (116, 0), (116, 58))
_BACKGROUNDS = ((250, 248, 241), (244, 249, 250))
_ROUTE_COLORS = ((38, 105, 168), (27, 126, 151))
_GREEN = (38, 166, 91)
_ORANGE = (238, 133, 42)
_START = (171, 66, 187)
_END = (236, 194, 42)
_BLACK = (25, 27, 31)
_WHITE = (255, 255, 255)


def _route(variant: int, ox: int, oy: int) -> list[tuple[float, float]]:
    points = []
    for x, y in _BASE_ROUTE:
        if variant == 0:
            tx, ty = x, y
        elif variant == 1:
            tx, ty = -x, y
        elif variant == 2:
            tx, ty = y, x
        else:
            tx, ty = -y, x
        points.append((256 + ox + tx, 270 + oy + ty))
    return points


def _glyph_grid(symbol: str) -> list[list[int]]:
    # Stored templates are rows; frames transport their columns.
    return [[int(value) for value in row] for row in TEMPLATES[symbol]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A17E)
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "route_variant": (int(seed) // 4) % 4,
        "palette_index": (int(seed) // 16) % 2,
        "offset_x": rng.randint(-10, 10),
        "offset_y": rng.randint(-10, 10),
        "nuisance_mask": rng.randrange(128),
    }


def render(scene: dict) -> Image.Image:
    symbol = scene["symbol"]
    variant = int(scene["route_variant"]) % 4
    palette = int(scene["palette_index"]) % 2
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    nuisance = int(scene["nuisance_mask"]) % 128
    background = _BACKGROUNDS[palette]
    blue = _ROUTE_COLORS[palette]
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)

    points = _route(variant, ox, oy)
    # The connected blue carrier is deliberately drawn before its framed
    # cross-sections. Missing any interior link makes route order undefined.
    draw.line(points, fill=blue, width=12, joint="curve")

    grid = _glyph_grid(symbol)
    for index, (cx, cy) in enumerate(points):
        # Orientation varies independently of the answer. Green-to-orange is
        # the public top-to-bottom direction after rectification.
        horizontal = bool((nuisance >> index) & 1)
        ux, uy = ((1.0, 0.0) if horizontal else (0.0, 1.0))
        vx, vy = -uy, ux
        half_axis, half_width = 29.0, 8.0
        polygon = [
            (cx - ux * half_axis - vx * half_width, cy - uy * half_axis - vy * half_width),
            (cx + ux * half_axis - vx * half_width, cy + uy * half_axis - vy * half_width),
            (cx + ux * half_axis + vx * half_width, cy + uy * half_axis + vy * half_width),
            (cx - ux * half_axis + vx * half_width, cy - uy * half_axis + vy * half_width),
        ]
        draw.polygon(polygon, fill=(255, 253, 247), outline=blue, width=3)
        column = [grid[row][index] for row in range(7)]
        for row, bit in enumerate(column):
            distance = (row - 3) * 5.0
            px, py = cx + ux * distance, cy + uy * distance
            radius = 2.4
            fill = _BLACK if bit else _WHITE
            outline = _BLACK if bit else (183, 184, 180)
            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=fill, outline=outline)
        for distance, color in ((-24.0, _GREEN), (24.0, _ORANGE)):
            px, py = cx + ux * distance, cy + uy * distance
            draw.ellipse((px - 4, py - 4, px + 4, py + 4), fill=color, outline=(255, 255, 255))

    # Visible route endpoints: triangular purple START and round yellow END.
    def cap_position(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
        dx, dy = a[0] - b[0], a[1] - b[1]
        length = math.hypot(dx, dy)
        return a[0] + 42 * dx / length, a[1] + 42 * dy / length

    sx, sy = cap_position(points[0], points[1])
    ex, ey = cap_position(points[-1], points[-2])
    draw.line((sx, sy, points[0][0], points[0][1]), fill=blue, width=12)
    draw.line((points[-1][0], points[-1][1], ex, ey), fill=blue, width=12)
    draw.polygon(((sx - 8, sy + 7), (sx - 8, sy - 7), (sx + 8, sy)), fill=_START, outline=(255, 255, 255))
    draw.ellipse((ex - 8, ey - 8, ex + 8, ey + 8), fill=_END, outline=(255, 255, 255), width=2)

    # Repaint all measured samples and bead fiducials after every overlapping
    # frame and endpoint cap. This makes occlusion order a nuisance only for the
    # carrier artwork, never a hidden loss of decisive evidence.
    for index, (cx, cy) in enumerate(points):
        horizontal = bool((nuisance >> index) & 1)
        ux, uy = ((1.0, 0.0) if horizontal else (0.0, 1.0))
        column = [grid[row][index] for row in range(7)]
        for row, bit in enumerate(column):
            distance = (row - 3) * 5.0
            px, py = cx + ux * distance, cy + uy * distance
            radius = 2.4
            fill = _BLACK if bit else _WHITE
            outline = _BLACK if bit else (183, 184, 180)
            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=fill, outline=outline)
        for distance, color in ((-24.0, _GREEN), (24.0, _ORANGE)):
            px, py = cx + ux * distance, cy + uy * distance
            draw.ellipse((px - 4, py - 4, px + 4, py + 4), fill=color, outline=(255, 255, 255))

    # Label-free neutral perforation distractors. Adding 128 to nuisance_mask
    # is a tested latent alias because both orientation and these positions use
    # only the low seven bits.
    decor_rng = random.Random(nuisance * 7919 + 17)
    for _ in range(14):
        side = decor_rng.randrange(4)
        if side == 0:
            x, y = decor_rng.randint(50, 462), decor_rng.randint(55, 105)
        elif side == 1:
            x, y = decor_rng.randint(50, 462), decor_rng.randint(425, 465)
        elif side == 2:
            x, y = decor_rng.randint(45, 90), decor_rng.randint(120, 410)
        else:
            x, y = decor_rng.randint(422, 467), decor_rng.randint(120, 410)
        draw.ellipse((x - 2, y - 2, x + 2, y + 2), outline=(174, 177, 174))
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    symbol = scene["symbol"]
    source = "".join(TEMPLATES[symbol])
    distances = [
        sum(a != b for a, b in zip(source, "".join(template), strict=True))
        for name, template in TEMPLATES.items()
        if name != symbol
    ]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["nuisance_mask"] = int(twin["nuisance_mask"]) + 128
    return [("low_seven_bit_alias", twin)]
