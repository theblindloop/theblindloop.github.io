"""Independent pixel-only inverse arm for visible bilateral correspondence."""

from __future__ import annotations

import itertools
import math
from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    remaining = {tuple(p) for p in np.argwhere(mask)}
    found = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for q in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= q[0] < height and 0 <= q[1] < width and q in remaining:
                    remaining.remove(q)
                    queue.append(q)
                    points.append(q)
        if len(points) >= minimum:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def _quadrants(points: list[tuple[float, float, int]]) -> list[list[tuple[float, float, int]]]:
    xs = sorted(p[0] for p in points)
    ys = sorted(p[1] for p in points)
    x_cut = (xs[len(xs) // 2 - 1] + xs[len(xs) // 2]) / 2.0
    y_cut = (ys[len(ys) // 2 - 1] + ys[len(ys) // 2]) / 2.0
    groups = [[] for _ in range(4)]
    for point in points:
        x, y, _ = point
        group = (2 if y > y_cut else 0) + (1 if x > x_cut else 0)
        groups[group].append(point)
    return groups


def _reflection_score(dots: list[tuple[float, float, int]], axis_x: float) -> float:
    left = [(x, y, area) for x, y, area in dots if x < axis_x]
    right = [(x, y, area) for x, y, area in dots if x > axis_x]
    if len(left) != 3 or len(right) != 3:
        raise ValueError("expected three dots on each side of the axis")
    best = math.inf
    for order in itertools.permutations(right):
        errors = []
        for (lx, ly, la), (rx, ry, ra) in zip(left, order):
            position = math.hypot((2.0 * axis_x - lx) - rx, ly - ry)
            size = abs(math.sqrt(la / math.pi) - math.sqrt(ra / math.pi))
            errors.append(position + size)
        best = min(best, max(errors))
    return best


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flat = array.reshape(-1, 3)
    spread = flat.max(axis=1) - flat.min(axis=1)
    vivid = flat[spread > 45]
    if len(vivid) < 7000:
        raise ValueError("insufficient colored evidence")
    colors, counts = np.unique(vivid, axis=0, return_counts=True)
    ranked = np.argsort(counts)[::-1]
    if len(ranked) < 2 or int(counts[ranked[0]]) < 7000 or int(counts[ranked[1]]) < 700:
        raise ValueError("dot and axis inks are not recoverable")
    dot_color = colors[int(ranked[0])]
    axis_color = colors[int(ranked[1])]
    dots = _components(np.all(array == dot_color, axis=2), 250)
    axes = _components(np.all(array == axis_color, axis=2), 20)
    if len(dots) != 24 or len(axes) < 24:
        raise ValueError("expected twenty-four dots and four dashed axes")
    dot_groups = _quadrants(dots)
    axis_groups = _quadrants(axes)
    if any(len(group) != 6 for group in dot_groups) or any(len(group) < 6 for group in axis_groups):
        raise ValueError("could not assign visible marks to four panels")
    scores = []
    for dot_group, axis_group in zip(dot_groups, axis_groups):
        axis_x = float(np.mean([x for x, _, _ in axis_group]))
        scores.append(_reflection_score(dot_group, axis_x))
    order = sorted(range(4), key=lambda i: scores[i])
    if scores[order[0]] > 1.5 or scores[order[1]] - scores[order[0]] < 8.0:
        raise ValueError("ambiguous or non-symmetric scene; abstaining")
    return LABELS[order[0]]
