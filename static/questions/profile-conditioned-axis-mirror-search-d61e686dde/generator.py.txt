"""Deterministic latent sampler, Pillow renderer, and analytic reflection arm."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
LABELS = ("A", "B", "C", "D")
QUARANTINE_MARGIN = 8.0
INKS = ((31, 91, 145), (151, 63, 116), (28, 127, 111), (183, 73, 50), (101, 78, 163))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 4
    panels = []
    for panel in range(4):
        pairs = []
        for y in (-55, 0, 55):
            distance = rng.randint(43, 72)
            yy = y + rng.randint(-3, 3)
            pairs.append([-distance, yy, distance, yy])
        if panel != target:
            slot = rng.randrange(3)
            defect = rng.randint(12, 17) * rng.choice((-1, 1))
            if rng.randrange(2):
                pairs[slot][2] += defect
            else:
                pairs[slot][3] += defect
        rng.shuffle(pairs)
        panels.append(pairs)
    dot_index = rng.randrange(len(INKS))
    axis_index = rng.randrange(len(INKS) - 1)
    if axis_index >= dot_index:
        axis_index += 1
    return {
        "panel_pairs": panels,
        "dot_color": list(INKS[dot_index]),
        "axis_color": list(INKS[axis_index]),
        "shift_x": rng.randint(-6, 6),
        "shift_y": rng.randint(-5, 5),
        "dot_radius": rng.choice((11, 13, 15)),
    }


def _panel_score(pairs: list[list[int]]) -> float:
    left = [(float(p[0]), float(p[1])) for p in pairs]
    right = [(float(p[2]), float(p[3])) for p in pairs]
    best = math.inf
    for order in itertools.permutations(right):
        mismatch = max(math.hypot(-lx - rx, ly - ry) for (lx, ly), (rx, ry) in zip(left, order))
        best = min(best, mismatch)
    return best


def _scores(scene: dict) -> list[float]:
    return [_panel_score(pairs) for pairs in scene["panel_pairs"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    order = sorted(range(4), key=lambda i: scores[i])
    if scores[order[1]] - scores[order[0]] <= 0:
        raise ValueError("scene lacks a unique most-symmetric panel")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for pairs in twin["panel_pairs"]:
        pairs[0], pairs[1] = pairs[1], pairs[0]
    return [("swap_first_two_disjoint_pair_records", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 247, 243))
    draw = ImageDraw.Draw(image)
    sx, sy = int(scene["shift_x"]), int(scene["shift_y"])
    centers = ((170, 133), (470, 133), (170, 353), (470, 353))
    dot_color = tuple(int(v) for v in scene["dot_color"])
    axis_color = tuple(int(v) for v in scene["axis_color"])
    radius = int(scene["dot_radius"])
    font = ImageFont.load_default()
    for index, ((cx0, cy0), pairs) in enumerate(zip(centers, scene["panel_pairs"])):
        cx, cy = cx0 + sx, cy0 + sy
        draw.rounded_rectangle((cx - 135, cy - 92, cx + 135, cy + 92), radius=13,
                               fill=(255, 255, 255), outline=(158, 163, 168), width=3)
        draw.text((cx - 125, cy - 80), LABELS[index], fill=(54, 57, 62), font=font)
        for top in range(cy - 76, cy + 77, 16):
            draw.rectangle((cx - 2, top, cx + 2, min(top + 8, cy + 76)), fill=axis_color)
        for lx, ly, rx, ry in pairs:
            for x, y in ((cx + int(lx), cy + int(ly)), (cx + int(rx), cy + int(ry))):
                draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=dot_color)
    return image


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = deepcopy(scene)
    twin["shift_x"] += int(dx)
    twin["shift_y"] += int(dy)
    return twin
