from __future__ import annotations

import math
import numpy as np
from PIL import Image

CYAN = np.array([10, 154, 184], dtype=np.int32)
AXIS = np.array([71, 76, 82], dtype=np.int32)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[np.ndarray] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        stack = [(y, x)]
        pts = []
        while stack:
            cy, cx = stack.pop()
            pts.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        out.append(np.asarray(pts, dtype=np.int32))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    route = np.sqrt(np.sum((arr - CYAN) ** 2, axis=2)) < 48
    axis = np.sqrt(np.sum((arr - AXIS) ** 2, axis=2)) < 44
    comps = [c for c in _components(route) if len(c) >= 80]
    if len(comps) != 1:
        raise ValueError("trace must be one intact connected component")
    comp = comps[0]
    xmin, xmax = int(comp[:, 1].min()), int(comp[:, 1].max())
    if xmax - xmin < 445:
        raise ValueError("trace does not span the distributed measurement range")
    counts = route[:, xmin : xmax + 1].sum(axis=0)
    if np.any(counts < 2):
        raise ValueError("trace has an erased or disconnected interior column")
    ay, ax = np.nonzero(axis)
    if len(ay) < 250 or int(ax.max() - ax.min()) < 455:
        raise ValueError("visible dashed reference is absent or damaged")
    baseline = float(np.median(ay))
    center = np.array([np.median(np.nonzero(route[:, x])[0]) for x in range(xmin, xmax + 1)], dtype=float)
    # Recover the centerline independently from its raster columns. A compact
    # symmetric filter suppresses subpixel edge ringing without assuming vertex
    # locations or reading renderer state.
    kernel = np.ones(7, dtype=float) / 7.0
    smooth = np.convolve(np.pad(center, (3, 3), mode="edge"), kernel, mode="valid")
    above = below = 0.0
    for y1, y2 in zip(smooth, smooth[1:]):
        a, b = y1 - baseline, y2 - baseline
        seg = math.hypot(1.0, y2 - y1)
        if a < 0 and b < 0:
            above += seg
        elif a > 0 and b > 0:
            below += seg
        elif a == b == 0:
            raise ValueError("trace lies on the decision boundary")
        else:
            frac = abs(a) / (abs(a) + abs(b))
            first, second = seg * frac, seg * (1.0 - frac)
            if a < 0:
                above += first
                below += second
            else:
                below += first
                above += second
    pixel_margin = abs(above - below)
    if pixel_margin < 16.0:
        raise ValueError("raster arc-length comparison is ambiguous")
    return "above" if above > below else "below"
