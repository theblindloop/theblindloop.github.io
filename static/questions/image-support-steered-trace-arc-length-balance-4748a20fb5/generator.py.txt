from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

SCALE = 3
MIN_MARGIN = 75.0


def _lengths(scene: dict) -> tuple[float, float]:
    """Exact arc length of the latent polyline on each side of the baseline."""
    y0 = float(scene["baseline_y"])
    above = below = 0.0
    points = scene["points"]
    for (x1, y1), (x2, y2) in zip(points, points[1:]):
        a, b = y1 - y0, y2 - y0
        length = math.hypot(x2 - x1, y2 - y1)
        if a < 0 and b < 0:
            above += length
        elif a > 0 and b > 0:
            below += length
        else:
            # No sampled vertex lies on the reference. A crossing divides a
            # straight segment in the same ratio as its signed endpoint offsets.
            frac = abs(a) / (abs(a) + abs(b))
            first, second = length * frac, length * (1.0 - frac)
            if a < 0:
                above += first
                below += second
            else:
                below += first
                above += second
    return above, below


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 512, 384
    baseline_y = rng.randint(181, 203)
    target = "above" if int(seed) % 2 == 0 else "below"
    band = int(seed) % 4
    # Near-boundary scenes are deliberately represented, but remain outside
    # the finite-resolution quarantine band.
    ranges = ((84.0, 110.0), (110.0, 155.0), (155.0, 220.0), (220.0, 1000.0))
    lo, hi = ranges[band]
    for _ in range(20000):
        xs = [30]
        for i in range(1, 9):
            xs.append(30 + i * 50 + rng.randint(-6, 6))
        xs.append(482)
        xs = sorted(xs)
        offsets = []
        sign = -1 if rng.random() < 0.5 else 1
        for i in range(10):
            # Frequent sign changes prevent endpoint and longest-lobe shortcuts.
            if i and rng.random() < 0.58:
                sign *= -1
            offsets.append(sign * rng.randint(30, 105))
        points = [[x, baseline_y + off] for x, off in zip(xs, offsets)]
        scene = {
            "width": width,
            "height": height,
            "baseline_y": baseline_y,
            "points": points,
            "route_color": [10, 154, 184],
            "axis_color": [71, 76, 82],
            "background": [249, 248, 244],
        }
        above, below = _lengths(scene)
        decision = "above" if above > below else "below"
        m = abs(above - below)
        if decision == target and lo <= m < hi:
            return scene
    raise RuntimeError(f"unable to sample balanced scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    above, below = _lengths(scene)
    return "above" if above > below else "below"


def margin(scene: dict) -> float:
    above, below = _lengths(scene)
    return float(abs(above - below))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (w * SCALE, h * SCALE), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    y0 = int(scene["baseline_y"] * SCALE)
    # Light plot guides are deliberately not used by either answer arm.
    for gx in (86, 150, 214, 278, 342, 406):
        draw.line([(gx * SCALE, 55 * SCALE), (gx * SCALE, 329 * SCALE)], fill=(225, 226, 221), width=SCALE)
    axis = tuple(scene["axis_color"])
    x = 20 * SCALE
    while x < 492 * SCALE:
        draw.line([(x, y0), (min(x + 16 * SCALE, 492 * SCALE), y0)], fill=axis, width=3 * SCALE)
        x += 26 * SCALE
    points = [(int(x * SCALE), int(y * SCALE)) for x, y in scene["points"]]
    route = tuple(scene["route_color"])
    draw.line(points, fill=route, width=7 * SCALE, joint="curve")
    for px, py in points:
        r = 3 * SCALE
        draw.ellipse([px - r, py - r, px + r, py + r], fill=route)
    return image.resize((w, h), Image.Resampling.LANCZOS).convert("RGB")
