from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((24, 151, 181), dtype=np.int16)
AMBER = np.array((221, 132, 23), dtype=np.int16)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    found: list[np.ndarray] = []
    height, width = work.shape
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        work[sy, sx] = False
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if len(points) >= 55:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _bearing(points: np.ndarray) -> int:
    xy = points[:, [1, 0]]
    centered = xy - xy.mean(axis=0)
    values, vectors = np.linalg.eigh(centered.T @ centered)
    vx, vy = vectors[:, int(np.argmax(values))]
    angle = math.atan2(vy, vx) % math.pi
    return int(round(angle / (math.pi / 4.0))) % 4


def _measure_color(array: np.ndarray, target: np.ndarray) -> tuple[tuple[float, float, int, int], list[tuple[float, float, int, int]]]:
    delta = np.linalg.norm(array.astype(np.int16) - target, axis=2)
    pieces = _components(delta < 72.0)
    records = []
    for points in pieces:
        cy, cx = points.mean(axis=0)
        records.append((float(cx), float(cy), len(points), _bearing(points)))
    if len(records) < 3:
        raise ValueError("too few colored bearing components")
    records.sort(key=lambda row: row[2], reverse=True)
    guide = records[0]
    candidate_areas = [row[2] for row in records[1:]]
    if guide[2] < 1.55 * max(candidate_areas):
        raise ValueError("remote bearing guide is absent or ambiguous")
    return guide, records[1:]


def _slot(cx: float, cy: float, width: int, height: int) -> int:
    if not (0.24 * width < cx < 0.76 * width and 0.39 * height < cy < 0.94 * height):
        raise ValueError("candidate bearing lies outside visible badge field")
    return (0 if cy < 0.665 * height else 2) + (0 if cx < 0.50 * width else 1)


def _distance(a: int, b: int) -> int:
    d = abs(a - b)
    return min(d, 4 - d)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    cyan_guide, cyan_candidates = _measure_color(array, CYAN)
    amber_guide, amber_candidates = _measure_color(array, AMBER)
    cyan_slots = {_slot(x, y, width, height): bearing for x, y, area, bearing in cyan_candidates if area >= 55}
    amber_slots = {_slot(x, y, width, height): bearing for x, y, area, bearing in amber_candidates if area >= 55}
    common = sorted(set(cyan_slots) & set(amber_slots))
    if not common:
        raise ValueError("no complete two-color badge remains")
    scores = []
    for slot in common:
        score = _distance(cyan_slots[slot], cyan_guide[3]) + _distance(amber_slots[slot], amber_guide[3])
        scores.append((score, slot))
    scores.sort()
    if scores[0][0] != 0 or (len(scores) > 1 and scores[1][0] == scores[0][0]):
        raise ValueError("no unique exact two-bearing match")
    return LABELS[scores[0][1]]
