from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 512
SCALE = 3
BG = (248, 247, 242)
CYAN = (24, 151, 181)
AMBER = (221, 132, 23)
INK = (35, 42, 48)
PANEL = (109, 116, 121)
LABELS = ("A", "B", "C", "D")
VECTORS = ((1, 0), (1, -1), (0, 1), (1, 1))


def _mismatch(a: int, b: int) -> int:
    d = abs((int(a) % 4) - (int(b) % 4))
    return min(d, 4 - d)


def _scores(scene: dict) -> list[int]:
    refs = scene["reference_bearings"]
    pairs = scene["candidate_bearings"]
    return [_mismatch(pair[0], refs[0]) + _mismatch(pair[1], refs[1]) for pair in pairs]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    refs = [rng.randrange(4), rng.randrange(4)]
    target = int(seed) % 4
    other = [i for i in range(4) if i != target]
    rng.shuffle(other)
    cyan_only, amber_only, neither = other
    da = rng.choice((1, 2, 3))
    dc = rng.choice((1, 2, 3))
    dn_c = rng.choice((1, 2, 3))
    dn_a = rng.choice((1, 2, 3))
    pairs = [[0, 0] for _ in range(4)]
    pairs[target] = [refs[0], refs[1]]
    pairs[cyan_only] = [refs[0], (refs[1] + da) % 4]
    pairs[amber_only] = [(refs[0] + dc) % 4, refs[1]]
    pairs[neither] = [(refs[0] + dn_c) % 4, (refs[1] + dn_a) % 4]
    return {
        "reference_bearings": refs,
        "candidate_bearings": pairs,
        "canvas": [WIDTH, HEIGHT],
        "palette": {"background": list(BG), "cyan": list(CYAN), "amber": list(AMBER)},
    }


def _font(size: int, bold: bool = False):
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(path, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _bar(draw: ImageDraw.ImageDraw, center: tuple[int, int], bearing: int, length: int, width: int, color: tuple[int, int, int]) -> None:
    vx, vy = VECTORS[int(bearing) % 4]
    if vx and vy:
        half = round(length / (2 ** 0.5) / 2)
        dx, dy = vx * half, vy * half
    else:
        dx, dy = vx * length // 2, vy * length // 2
    cx, cy = center
    p0 = ((cx - dx) * SCALE, (cy - dy) * SCALE)
    p1 = ((cx + dx) * SCALE, (cy + dy) * SCALE)
    draw.line((p0, p1), fill=color, width=width * SCALE)
    r = width * SCALE // 2
    for px, py in (p0, p1):
        draw.ellipse((px - r, py - r, px + r, py + r), fill=color)


def render(scene: dict) -> Image.Image:
    w, h = map(int, scene["canvas"])
    palette = scene["palette"]
    bg = tuple(palette["background"])
    cyan = tuple(palette["cyan"])
    amber = tuple(palette["amber"])
    refs = scene["reference_bearings"]
    pairs = scene["candidate_bearings"]
    image = Image.new("RGB", (w * SCALE, h * SCALE), bg)
    draw = ImageDraw.Draw(image)
    title_font = _font(18, True)
    small_font = _font(14, True)
    label_font = _font(23, True)
    draw.text((w * SCALE // 2, 18 * SCALE), "MATCH BOTH REMOTE BEARINGS", font=title_font, fill=INK, anchor="ma")

    for cx, color, bearing, caption in (
        (72, cyan, refs[0], "CYAN GUIDE"),
        (696, amber, refs[1], "AMBER GUIDE"),
    ):
        draw.rounded_rectangle(((18 if cx < w // 2 else 642) * SCALE, 54 * SCALE, (126 if cx < w // 2 else 750) * SCALE, 160 * SCALE), radius=14 * SCALE, outline=PANEL, width=3 * SCALE)
        _bar(draw, (cx, 104), bearing, 74, 15, color)
        draw.text((cx * SCALE, 174 * SCALE), caption, font=small_font, fill=INK, anchor="ma")

    centers = ((310, 274), (458, 274), (310, 410), (458, 410))
    for idx, ((cx, cy), pair) in enumerate(zip(centers, pairs, strict=True)):
        draw.rounded_rectangle(((cx - 61) * SCALE, (cy - 50) * SCALE, (cx + 61) * SCALE, (cy + 50) * SCALE), radius=13 * SCALE, fill=(255, 255, 255), outline=PANEL, width=3 * SCALE)
        _bar(draw, (cx - 27, cy), pair[0], 46, 9, cyan)
        _bar(draw, (cx + 27, cy), pair[1], 46, 9, amber)
        draw.ellipse(((cx - 10) * SCALE, (cy - 10) * SCALE, (cx + 10) * SCALE, (cy + 10) * SCALE), fill=bg, outline=PANEL, width=2 * SCALE)
        draw.text((cx * SCALE, (cy + 69) * SCALE), LABELS[idx], font=label_font, fill=INK, anchor="mm")
    return image.resize((w, h), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    refs = scene["reference_bearings"]
    pairs = scene["candidate_bearings"]
    scores = [_mismatch(p[0], refs[0]) + _mismatch(p[1], refs[1]) for p in pairs]
    best = min(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("bearing search has no unique analytic winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene))
    return float((values[1] - values[0]) * 45)


def is_quarantined(scene: dict) -> bool:
    values = sorted(_scores(scene))
    return values[0] != 0 or values[1] - values[0] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["reference_bearings"] = [int(value) + 4 for value in scene["reference_bearings"]]
    twin["candidate_bearings"] = [[int(a) + 4, int(b) + 4] for a, b in scene["candidate_bearings"]]
    return [("undirected_half_turn_alias", twin)]
