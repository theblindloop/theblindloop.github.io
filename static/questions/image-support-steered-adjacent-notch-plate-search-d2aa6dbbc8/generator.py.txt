"""Latent sampler and independent analytic arm for the notched-plate world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = [248, 248, 244]
PLATE_SIZE = 60
NOTCH_WIDTH = 16
NOTCH_DEPTH = 13
COLORS: dict[str, tuple[int, int, int]] = {
    "red": (210, 57, 64),
    "blue": (45, 105, 181),
    "green": (42, 145, 84),
    "purple": (126, 72, 168),
}
SIDES = ("top", "right", "bottom", "left")
ADJACENT_PAIRS = (
    ("top", "right"),
    ("right", "bottom"),
    ("bottom", "left"),
    ("left", "top"),
)
OPPOSITE_PAIRS = (("top", "bottom"), ("left", "right"))
BASE_CENTERS = ((138, 140), (374, 138), (140, 374), (372, 372))


def _cyclic_separation(notches: list[str]) -> int:
    if len(notches) != 2 or len(set(notches)) != 2:
        raise ValueError("a plate must have two distinct notch sides")
    a, b = (SIDES.index(side) for side in notches)
    delta = abs(a - b)
    return min(delta, 4 - delta)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(COLORS)
    target = names[int(seed) % len(names)]
    shuffled = names[:]
    rng.shuffle(shuffled)
    centers = []
    for base_x, base_y in BASE_CENTERS:
        centers.append([base_x + rng.randint(-22, 22), base_y + rng.randint(-22, 22)])
    plates = []
    target_pair = list(rng.choice(ADJACENT_PAIRS))
    rng.shuffle(target_pair)
    for index, color in enumerate(shuffled):
        if color == target:
            notches = target_pair
        else:
            notches = list(OPPOSITE_PAIRS[(rng.randrange(2) + index) % 2])
            rng.shuffle(notches)
        plates.append({"color": color, "center": centers[index], "notches": notches})
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "plates": plates,
    }


def _notch_box(x0: int, y0: int, side: str) -> tuple[int, int, int, int]:
    offset = (PLATE_SIZE - NOTCH_WIDTH) // 2
    if side == "top":
        return x0 + offset, y0, x0 + offset + NOTCH_WIDTH - 1, y0 + NOTCH_DEPTH - 1
    if side == "bottom":
        return (
            x0 + offset,
            y0 + PLATE_SIZE - NOTCH_DEPTH,
            x0 + offset + NOTCH_WIDTH - 1,
            y0 + PLATE_SIZE - 1,
        )
    if side == "left":
        return x0, y0 + offset, x0 + NOTCH_DEPTH - 1, y0 + offset + NOTCH_WIDTH - 1
    if side == "right":
        return (
            x0 + PLATE_SIZE - NOTCH_DEPTH,
            y0 + offset,
            x0 + PLATE_SIZE - 1,
            y0 + offset + NOTCH_WIDTH - 1,
        )
    raise ValueError(f"unknown side {side!r}")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    # Pale neutral marks add search clutter without entering the oracle palette.
    for y in range(54, height - 40, 67):
        for x in range(48, width - 40, 83):
            if ((x // 7 + y // 11) % 3) == 0:
                draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=(224, 224, 218))
    for plate in scene["plates"]:
        cx, cy = (int(value) for value in plate["center"])
        x0 = cx - PLATE_SIZE // 2
        y0 = cy - PLATE_SIZE // 2
        color = COLORS[str(plate["color"])]
        draw.rectangle((x0, y0, x0 + PLATE_SIZE - 1, y0 + PLATE_SIZE - 1), fill=color)
        for side in plate["notches"]:
            draw.rectangle(_notch_box(x0, y0, str(side)), fill=background)
    return image


def analytic_gold(scene: dict) -> str:
    winners = []
    for plate in scene["plates"]:
        notches = list(plate["notches"])
        indices = [SIDES.index(str(side)) for side in notches]
        delta = abs(indices[0] - indices[1])
        if min(delta, 4 - delta) == 1:
            winners.append(str(plate["color"]))
    if len(winners) != 1:
        raise ValueError(f"expected one adjacent-notch plate, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(
        (2 - _cyclic_separation(list(plate["notches"])) for plate in scene["plates"]),
        reverse=True,
    )
    if len(scores) < 2:
        return 0.0
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return len(scene["plates"]) != 4 or margin(scene) < 1.0 or analytic_gold(scene) not in COLORS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["plates"] = list(reversed(twin["plates"]))
    return [("reverse_nonoverlapping_draw_order", twin)]
