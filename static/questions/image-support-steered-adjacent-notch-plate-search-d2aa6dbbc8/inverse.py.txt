"""Scene-free pixel inverse for adjacent-edge notch search.

This module intentionally duplicates the public palette and plate measurements.
It receives only a final PIL image and never imports the latent renderer.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (210, 57, 64),
    "blue": (45, 105, 181),
    "green": (42, 145, 84),
    "purple": (126, 72, 168),
}
_SIDES = ("top", "right", "bottom", "left")
_SIZE = 60
_NOTCH_WIDTH = 16
_NOTCH_DEPTH = 13
_EXPECTED_AREA = _SIZE * _SIZE - 2 * _NOTCH_WIDTH * _NOTCH_DEPTH


def _largest_component(mask: np.ndarray) -> np.ndarray | None:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(best):
            best = points
    if not best:
        return None
    component = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*best, strict=True)
    component[np.asarray(ys), np.asarray(xs)] = True
    return component


def _side_missing(local: np.ndarray) -> list[float]:
    offset = (_SIZE - _NOTCH_WIDTH) // 2
    probes = (
        local[0:_NOTCH_DEPTH, offset : offset + _NOTCH_WIDTH],
        local[offset : offset + _NOTCH_WIDTH, _SIZE - _NOTCH_DEPTH : _SIZE],
        local[_SIZE - _NOTCH_DEPTH : _SIZE, offset : offset + _NOTCH_WIDTH],
        local[offset : offset + _NOTCH_WIDTH, 0:_NOTCH_DEPTH],
    )
    return [1.0 - float(np.mean(probe)) for probe in probes]


def _measure_candidate(mask: np.ndarray) -> tuple[int, list[str]] | None:
    component = _largest_component(mask)
    if component is None:
        return None
    ys, xs = np.nonzero(component)
    if len(xs) < _EXPECTED_AREA - 24 or len(xs) > _EXPECTED_AREA + 24:
        return None
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if (x1 - x0 + 1, y1 - y0 + 1) != (_SIZE, _SIZE):
        return None
    local = component[y0 : y1 + 1, x0 : x1 + 1]
    missing = _side_missing(local)
    notch_indices = [index for index, value in enumerate(missing) if value >= 0.92]
    if len(notch_indices) != 2:
        return None
    if any(value > 0.08 for index, value in enumerate(missing) if index not in notch_indices):
        return None
    delta = abs(notch_indices[0] - notch_indices[1])
    separation = min(delta, 4 - delta)
    if separation not in (1, 2):
        return None
    return 2 - separation, [_SIDES[index] for index in notch_indices]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    measured: list[tuple[int, str]] = []
    for name, rgb in _COLORS.items():
        distance = np.linalg.norm(array - np.asarray(rgb, dtype=np.int16), axis=2)
        result = _measure_candidate(distance <= 8.0)
        if result is not None:
            score, _notches = result
            measured.append((score, name))
    if len(measured) < 2:
        raise ValueError("fewer than two intact colored plate candidates")
    measured.sort(reverse=True)
    if measured[0][0] - measured[1][0] < 1:
        raise ValueError("no unique adjacent-notch winner with a runner-up margin")
    return measured[0][1]
