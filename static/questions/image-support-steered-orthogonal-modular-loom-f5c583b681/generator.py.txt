from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

SIZE = (960, 720)
N = 5
DECISIONS = ("rising_1", "rising_2", "falling_2", "falling_1")
STEP_FOR_DECISION = {"falling_1": 1, "falling_2": 2, "rising_2": 3, "rising_1": 4}
DECISION_FOR_STEP = {value: key for key, value in STEP_FOR_DECISION.items()}

PALETTES = (
    ((220, 145, 30), (34, 150, 165)),
    ((210, 105, 35), (29, 135, 190)),
    ((235, 165, 25), (40, 160, 140)),
    ((200, 125, 40), (25, 145, 180)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    decision = DECISIONS[int(seed) % len(DECISIONS)]
    step = STEP_FOR_DECISION[decision]
    row_rate = rng.randrange(1, N)
    column_rate = (-step * row_rate) % N
    row_start = rng.randrange(N)
    column_start = rng.randrange(N)
    row_values = [(row_start + row_rate * index) % N for index in range(N)]
    column_values = [
        (column_start + column_rate * index) % N for index in range(N)
    ]
    # A compensating visible phase shift changes no answer and makes phase cheap cues useless.
    compensation = rng.randrange(N)
    row_values = [(value + compensation) % N for value in row_values]
    column_values = [(value - compensation) % N for value in column_values]
    return {
        "row_values": row_values,
        "column_values": column_values,
        "palette_index": rng.randrange(len(PALETTES)),
        "layout_dx": rng.randrange(-15, 16),
        "layout_dy": rng.randrange(-12, 13),
    }


def _zero_rows(row_values: Iterable[int], column_values: Iterable[int]) -> list[int]:
    rows = list(row_values)
    cols = list(column_values)
    result = []
    for column in cols:
        hits = [index for index, row in enumerate(rows) if (row + column) % N == 0]
        if len(hits) != 1:
            raise ValueError("each column must have exactly one modular zero")
        result.append(hits[0])
    return result


def _classify(row_values: Iterable[int], column_values: Iterable[int]) -> str:
    zero_rows = _zero_rows(row_values, column_values)
    steps = [(zero_rows[index + 1] - zero_rows[index]) % N for index in range(N - 1)]
    if not steps or len(set(steps)) != 1 or steps[0] not in DECISION_FOR_STEP:
        raise ValueError("zero-set is not a supported constant-step path")
    return DECISION_FOR_STEP[steps[0]]


def analytic_gold(scene: dict) -> str:
    return _classify(scene["row_values"], scene["column_values"])


def _binary_raster(row_values: Iterable[int], column_values: Iterable[int]) -> list[list[int]]:
    rows = list(row_values)
    cols = list(column_values)
    return [[int((rows[r] + cols[c]) % N == 0) for c in range(N)] for r in range(N)]


def margin(scene: dict) -> float:
    actual = _binary_raster(scene["row_values"], scene["column_values"])
    scores = []
    for step in (1, 2, 3, 4):
        best = -1
        for phase in range(N):
            template = [[int(r == (phase + step * c) % N) for c in range(N)] for r in range(N)]
            score = sum(actual[r][c] == template[r][c] for r in range(N) for c in range(N))
            best = max(best, score)
        scores.append(best)
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0
    except (TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width, height = SIZE
    image = Image.new("RGB", SIZE, (248, 246, 240))
    draw = ImageDraw.Draw(image)
    warm, cool = PALETTES[int(scene["palette_index"])]
    dx, dy = int(scene["layout_dx"]), int(scene["layout_dy"])
    row_values = [int(value) for value in scene["row_values"]]
    column_values = [int(value) for value in scene["column_values"]]

    # Faint, non-decisive loom: the detached rulers are aligned to its axes.
    grid_x, grid_y, cell = 530 + dx, 350 + dy, 42
    for index in range(N + 1):
        x = grid_x + index * cell
        y = grid_y + index * cell
        draw.line((x, grid_y, x, grid_y + N * cell), fill=(205, 202, 195), width=2)
        draw.line((grid_x, y, grid_x + N * cell, y), fill=(205, 202, 195), width=2)

    # Amber row ruler card: five horizontal unary mini-rulers.
    rx, ry, rw, rh = 270 + dx, 350 + dy, 170, 230
    draw.rounded_rectangle((rx, ry, rx + rw, ry + rh), radius=9, outline=warm, width=5)
    for row, value in enumerate(row_values):
        y = ry + 14 + row * 42
        for slot in range(N):
            x = rx + 20 + slot * 27
            draw.rounded_rectangle(
                (x, y, x + 22, y + 26),
                radius=3,
                fill=warm if slot < value else (238, 235, 227),
                outline=(135, 132, 126),
                width=2,
            )

    # Teal column ruler card: five vertical unary mini-rulers.
    tx, ty, tw, th = 530 + dx, 110 + dy, 230, 150
    draw.rounded_rectangle((tx, ty, tx + tw, ty + th), radius=9, outline=cool, width=5)
    for column, value in enumerate(column_values):
        x = tx + 14 + column * 42
        for slot in range(N):
            y = ty + 12 + slot * 24
            draw.rounded_rectangle(
                (x, y, x + 26, y + 19),
                radius=3,
                fill=cool if slot >= N - value else (238, 235, 227),
                outline=(135, 132, 126),
                width=2,
            )

    # Visible axis-end fiducials clarify direction without carrying label information.
    draw.polygon(
        [(rx - 16, ry + 8), (rx - 4, ry + 15), (rx - 16, ry + 22)], fill=(105, 102, 98)
    )
    draw.polygon(
        [(tx + tw - 22, ty - 16), (tx + tw - 8, ty - 16), (tx + tw - 15, ty - 4)],
        fill=(105, 102, 98),
    )
    return image
