from __future__ import annotations

import numpy as np
from PIL import Image

N = 5
DECISION_FOR_STEP = {1: "falling_1", 2: "falling_2", 3: "rising_2", 4: "rising_1"}


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    warm = (red > 170) & (green > 80) & (green < 190) & (blue < 105) & (red > green + 35)
    cool = (red < 105) & (green > 105) & (blue > 105) & (blue < 215)
    return warm, cool


def _bbox(mask: np.ndarray, name: str) -> tuple[int, int, int, int]:
    points = np.argwhere(mask)
    if len(points) < 200:
        raise ValueError(f"{name} ruler is missing")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def _fraction(mask: np.ndarray, x: int, y: int, width: int, height: int) -> float:
    patch = mask[y : y + height, x : x + width]
    if patch.shape != (height, width):
        raise ValueError("ruler sample extends beyond image")
    return float(patch.mean())


def _decode(image: Image.Image) -> tuple[list[int], list[int]]:
    warm, cool = _masks(image)
    wx0, wy0, wx1, wy1 = _bbox(warm, "amber")
    tx0, ty0, tx1, ty1 = _bbox(cool, "teal")
    if not (165 <= wx1 - wx0 <= 172 and 225 <= wy1 - wy0 <= 232):
        raise ValueError("amber frame fiducial is damaged")
    if not (225 <= tx1 - tx0 <= 232 and 145 <= ty1 - ty0 <= 152):
        raise ValueError("teal frame fiducial is damaged")

    rows = []
    for row in range(N):
        count = 0
        for slot in range(N):
            # Sample well inside each rounded cell, away from neutral outlines.
            x = wx0 + 24 + slot * 27
            y = wy0 + 19 + row * 42
            count += _fraction(warm, x, y, 14, 15) > 0.72
        rows.append(int(count))

    columns = []
    for column in range(N):
        count = 0
        for slot in range(N):
            x = tx0 + 20 + column * 42
            y = ty0 + 16 + slot * 24
            count += _fraction(cool, x, y, 15, 11) > 0.72
        columns.append(int(count))

    if sorted(rows) != list(range(N)) or sorted(columns) != list(range(N)):
        raise ValueError("ruler residues are incomplete or ambiguous")
    return rows, columns


def decision_from_image(image: Image.Image) -> str:
    rows, columns = _decode(image)
    zero_rows = []
    for column in columns:
        hits = [row_index for row_index, row in enumerate(rows) if (row + column) % N == 0]
        if len(hits) != 1:
            raise ValueError("a woven column has no unique modular zero")
        zero_rows.append(hits[0])
    steps = [(zero_rows[index + 1] - zero_rows[index]) % N for index in range(N - 1)]
    if not steps or len(set(steps)) != 1 or steps[0] not in DECISION_FOR_STEP:
        raise ValueError("woven path has no stable cyclic step")
    return DECISION_FOR_STEP[steps[0]]
