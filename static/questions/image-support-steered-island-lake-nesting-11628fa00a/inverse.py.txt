"""Independent image-only inverse arm using segmentation and flood fill."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(y, x)]
        group = []
        while queue:
            cy, cx = queue.pop()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    yellow = (red > 205) & (green > 125) & (green < 225) & (blue < 105) & ((red - blue) > 120)
    yellow_groups = [group for group in _components(yellow) if len(group) >= 45]
    if len(yellow_groups) != 1:
        raise ValueError("expected one visible yellow island")
    island = np.zeros(yellow.shape, dtype=bool)
    for y, x in yellow_groups[0]:
        island[y, x] = True
    black = (red < 90) & (green < 90) & (blue < 90)
    traversable = ~(black | island)
    height, width = traversable.shape
    reachable = np.zeros_like(traversable, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if traversable[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if traversable[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and traversable[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                queue.append((ny, nx))
    touches_outer_water = False
    for y, x in yellow_groups[0]:
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and reachable[ny, nx]:
                touches_outer_water = True
                break
        if touches_outer_water:
            break
    return "no" if touches_outer_water else "yes"
