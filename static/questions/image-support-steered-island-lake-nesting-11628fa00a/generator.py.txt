"""Deterministic compact topological lake renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

BACKGROUND = (247, 247, 242)
BLACK = (29, 33, 37)
YELLOW = (246, 184, 44)
GRAY = (174, 181, 184)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 384
    outer = rng.choice((29, 31, 33))
    wall = rng.choice((8, 9, 10, 11))
    inner = outer - wall
    gap = 7 if seed % 6 == 0 else rng.choice((10, 11, 12, 13))
    # Randomize among many safe absolute positions while centering the compact
    # cue near a fine-grid intersection. This prevents a small area cue from
    # being characterized as an accidental one-cell-wide strip.
    cx = rng.choice((96, 144, 192, 240, 288)) + rng.randint(-3, 3)
    cy = rng.choice((96, 144, 192, 240, 288)) + rng.randint(-3, 3)
    side = rng.choice(("north", "east", "south", "west"))
    distractors = []
    while len(distractors) < 5:
        x = rng.randint(20, width - 21)
        y = rng.randint(20, height - 21)
        if abs(x - cx) > outer + 45 or abs(y - cy) > outer + 45:
            distractors.append([x, y, rng.choice((2, 3, 4))])
    return {
        "width": width,
        "height": height,
        "cx": cx,
        "cy": cy,
        "closed": bool(seed % 2 == 0),
        "side": side,
        "outer": outer,
        "inner": inner,
        "wall": wall,
        "gap": gap,
        "gate_depth": wall + 3,
        "distractors": distractors,
    }


def _channel_box(scene: dict) -> tuple[int, int, int, int]:
    cx, cy = int(scene["cx"]), int(scene["cy"])
    outer, inner, gap = int(scene["outer"]), int(scene["inner"]), int(scene["gap"])
    half = gap // 2
    side = scene["side"]
    if side == "north":
        return (cx - half, cy - outer - 1, cx + half, cy - inner + 1)
    if side == "south":
        return (cx - half, cy + inner - 1, cx + half, cy + outer + 1)
    if side == "west":
        return (cx - outer - 1, cy - half, cx - inner + 1, cy + half)
    return (cx + inner - 1, cy - half, cx + outer + 1, cy + half)


def _gate_box(scene: dict, closed: bool) -> tuple[int, int, int, int]:
    x0, y0, x1, y1 = _channel_box(scene)
    if closed:
        return (x0, y0, x1, y1)
    side = scene["side"]
    shift = int(scene["gate_depth"]) + 7
    if side == "north":
        return (x0, y0 - shift, x1, y1 - shift)
    if side == "south":
        return (x0, y0 + shift, x1, y1 + shift)
    if side == "west":
        return (x0 - shift, y0, x1 - shift, y1)
    return (x0 + shift, y0, x1 + shift, y1)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    cx, cy = int(scene["cx"]), int(scene["cy"])
    outer, inner = int(scene["outer"]), int(scene["inner"])
    # Direct reads make every declared latent field visibly auditable.
    _ = int(scene["wall"]), int(scene["gap"]), int(scene["gate_depth"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["distractors"]:
        draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=GRAY)
    draw.rectangle((cx - outer, cy - outer, cx + outer, cy + outer), fill=BLACK)
    draw.rectangle((cx - inner, cy - inner, cx + inner, cy + inner), fill=BACKGROUND)
    draw.rectangle(_channel_box(scene), fill=BACKGROUND)
    draw.rectangle(_gate_box(scene, bool(scene["closed"])), fill=BLACK)
    island = max(5, inner // 3)
    draw.rectangle((cx - island, cy - island, cx + island, cy + island), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["closed"]) else "no"


def margin(scene: dict) -> float:
    return float(min(int(scene["wall"]), int(scene["gap"])))


def is_quarantined(scene: dict) -> bool:
    return min(int(scene["wall"]), int(scene["gap"])) < 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
