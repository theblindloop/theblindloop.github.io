"""Deterministic paired enclosure-depth world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 420
BACKGROUND = (255, 255, 255)
BLACK = (24, 28, 34)
CYAN = (14, 174, 214)
ORANGE = (240, 118, 34)
DOT_RADIUS = 9
LINE_WIDTH = 9


def _marker_position(cx: int, cy: int, rectangles: list[list[int]], depth: int, rng: random.Random) -> list[int]:
    outer, middle, inner = rectangles
    side = rng.choice((-1, 1))
    if depth == 0:
        return [cx + side * (outer[0] + rng.choice((24, 28))), cy + rng.choice((-76, -38, 38, 76))]
    if depth == 1:
        return [cx + side * ((outer[0] + middle[0]) // 2), cy + rng.choice((-66, -28, 28, 66))]
    if depth == 2:
        return [cx + side * ((middle[0] + inner[0]) // 2), cy + rng.choice((-43, -17, 17, 43))]
    return [cx + rng.choice((-20, 0, 20)), cy + rng.choice((-28, 0, 28))]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # The four codes exactly counterbalance yes/no and which color is deeper in
    # the left and right panels over every four consecutive seeds.
    deeper_codes = (("cyan", "cyan"), ("orange", "orange"), ("cyan", "orange"), ("orange", "cyan"))
    deeper_colors = deeper_codes[int(seed) % 4]
    depth_pairs = ((3, 1), (3, 2), (2, 0), (1, 0), (3, 0), (2, 1))
    panels = []
    for panel_index, base_cx in enumerate((170, 790)):
        cx = base_cx + rng.randint(-8, 8)
        cy = 210 + rng.randint(-7, 7)
        jitter = rng.randint(-2, 2)
        rectangles = [[120 + jitter, 145 + jitter], [78 + jitter, 98 + jitter], [39 + jitter, 55 + jitter]]
        deep, shallow = depth_pairs[rng.randrange(len(depth_pairs))]
        cyan_depth = deep if deeper_colors[panel_index] == "cyan" else shallow
        orange_depth = deep if deeper_colors[panel_index] == "orange" else shallow
        markers = {
            "cyan": [*_marker_position(cx, cy, rectangles, cyan_depth, rng), cyan_depth],
            "orange": [*_marker_position(cx, cy, rectangles, orange_depth, rng), orange_depth],
        }
        panels.append({"cx": cx, "cy": cy, "rectangles": rectangles, "markers": markers})
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        cx, cy = int(panel["cx"]), int(panel["cy"])
        for half_width, half_height in panel["rectangles"]:
            draw.rectangle(
                (cx - half_width, cy - half_height, cx + half_width, cy + half_height),
                outline=BLACK,
                width=LINE_WIDTH,
            )
        for color_name, fill in (("cyan", CYAN), ("orange", ORANGE)):
            x, y, _depth = panel["markers"][color_name]
            draw.ellipse(
                (x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS),
                fill=fill,
                outline=BACKGROUND,
                width=2,
            )
    return image


def analytic_gold(scene: dict) -> str:
    signs = []
    for panel in scene["panels"]:
        cyan_depth = int(panel["markers"]["cyan"][2])
        orange_depth = int(panel["markers"]["orange"][2])
        if cyan_depth == orange_depth:
            raise ValueError("marker depths must differ")
        signs.append(cyan_depth > orange_depth)
    if len(signs) != 2:
        raise ValueError("exactly two panels are required")
    return "yes" if signs[0] == signs[1] else "no"


def _point_rect_distance(x: float, y: float, cx: float, cy: float, hx: float, hy: float) -> float:
    dx, dy = abs(x - cx), abs(y - cy)
    if dx <= hx and dy <= hy:
        return min(hx - dx, hy - dy)
    return math.hypot(max(dx - hx, 0.0), max(dy - hy, 0.0))


def margin(scene: dict) -> float:
    clearances = []
    for panel in scene["panels"]:
        cx, cy = panel["cx"], panel["cy"]
        points = []
        for color_name in ("cyan", "orange"):
            x, y, _depth = panel["markers"][color_name]
            points.append((x, y))
            for hx, hy in panel["rectangles"]:
                clearances.append(_point_rect_distance(x, y, cx, cy, hx, hy) - DOT_RADIUS - LINE_WIDTH / 2)
        clearances.append(math.dist(points[0], points[1]) - 2 * DOT_RADIUS)
    return round(float(min(clearances)), 6)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0 or any(
            len(panel["rectangles"]) != 3 for panel in scene["panels"]
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("reverse_storage_order", twin)]
