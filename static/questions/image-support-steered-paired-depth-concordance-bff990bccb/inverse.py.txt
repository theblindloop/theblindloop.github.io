"""Independent image-only inverse for paired enclosure-depth concordance."""

from __future__ import annotations

from collections import deque

import numpy as np


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    height, width = mask.shape
    unseen = {tuple(map(int, point)) for point in np.argwhere(mask)}
    found = []
    while unseen:
        start = unseen.pop()
        queue = [start]
        points = [start]
        while queue:
            y, x = queue.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        if len(points) >= minimum:
            ys = np.fromiter((p[0] for p in points), dtype=np.int32)
            xs = np.fromiter((p[1] for p in points), dtype=np.int32)
            found.append({
                "area": len(points),
                "bbox": (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())),
                "cx": float(xs.mean()),
                "cy": float(ys.mean()),
            })
    return found


def _side_coverage(mask: np.ndarray, bbox: tuple[int, int, int, int]) -> float:
    x0, y0, x1, y1 = bbox
    band = 11
    top = mask[y0 : min(y0 + band, y1 + 1), x0 : x1 + 1].any(axis=0).mean()
    bottom = mask[max(y0, y1 - band + 1) : y1 + 1, x0 : x1 + 1].any(axis=0).mean()
    left = mask[y0 : y1 + 1, x0 : min(x0 + band, x1 + 1)].any(axis=1).mean()
    right = mask[y0 : y1 + 1, max(x0, x1 - band + 1) : x1 + 1].any(axis=1).mean()
    return float(min(top, bottom, left, right))


def _marker_centers(array: np.ndarray, target: tuple[int, int, int]) -> list[tuple[float, float]]:
    delta = np.abs(array.astype(np.int16) - np.asarray(target, dtype=np.int16)).sum(axis=2)
    components = _components(delta < 45, minimum=120)
    if len(components) != 2 or any(not 140 <= item["area"] <= 400 for item in components):
        raise ValueError("expected two intact markers of each color")
    return sorted(((item["cx"], item["cy"]) for item in components), key=lambda point: point[0])


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 700:
        raise ValueError("unexpected raster dimensions")
    black = np.all(array < np.asarray((70, 70, 75), dtype=np.uint8), axis=2)
    loops = _components(black, minimum=450)
    if len(loops) != 6:
        raise ValueError("expected exactly six intact closed loops")
    if any(_side_coverage(black, item["bbox"]) < 0.84 for item in loops):
        raise ValueError("a loop is visibly open or materially erased")
    loops.sort(key=lambda item: item["cx"])
    panel_loops = [loops[:3], loops[3:]]
    if panel_loops[1][0]["cx"] - panel_loops[0][-1]["cx"] < 300:
        raise ValueError("the two loop fields are not separated")
    for group in panel_loops:
        if max(item["cx"] for item in group) - min(item["cx"] for item in group) > 20:
            raise ValueError("loop centers do not form one nest")
        group.sort(key=lambda item: (item["bbox"][2] - item["bbox"][0]), reverse=True)
        for outer, inner in zip(group, group[1:]):
            ox0, oy0, ox1, oy1 = outer["bbox"]
            ix0, iy0, ix1, iy1 = inner["bbox"]
            if not (ox0 + 18 < ix0 and oy0 + 18 < iy0 and ix1 < ox1 - 18 and iy1 < oy1 - 18):
                raise ValueError("loops are not safely and strictly nested")
    cyan = _marker_centers(array, (14, 174, 214))
    orange = _marker_centers(array, (240, 118, 34))
    signs = []
    for index, group in enumerate(panel_loops):
        def depth(point: tuple[float, float]) -> int:
            x, y = point
            return sum(x0 < x < x1 and y0 < y < y1 for x0, y0, x1, y1 in (item["bbox"] for item in group))
        cyan_depth = depth(cyan[index])
        orange_depth = depth(orange[index])
        if cyan_depth == orange_depth:
            raise ValueError("the panel does not expose a strict depth order")
        signs.append(cyan_depth > orange_depth)
    return "yes" if signs[0] == signs[1] else "no"
