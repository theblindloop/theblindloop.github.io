"""Independent pixel-only inverse arm for the ladybug world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[dict] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        arr = np.asarray(points, dtype=np.int32)
        output.append({
            "area": len(points),
            "y0": int(arr[:, 0].min()), "y1": int(arr[:, 0].max()),
            "x0": int(arr[:, 1].min()), "x1": int(arr[:, 1].max()),
            "cy": float(arr[:, 0].mean()), "cx": float(arr[:, 1].mean()),
        })
    return output


def _inside_cover(mark: dict, cover: dict) -> bool:
    cx = (cover["x0"] + cover["x1"]) / 2.0
    cy = (cover["y0"] + cover["y1"]) / 2.0
    rx = (cover["x1"] - cover["x0"] + 1) / 2.0
    ry = (cover["y1"] - cover["y0"] + 1) / 2.0
    return ((mark["cx"] - cx) / (rx - 5.0)) ** 2 + ((mark["cy"] - cy) / (ry - 5.0)) ** 2 < 0.78


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.shape[0] < 300 or arr.shape[1] < 500:
        raise OracleAbstention("canvas is too small")
    red = (arr[:, :, 0] > 155) & (arr[:, :, 0] > arr[:, :, 1] * 2) & (arr[:, :, 0] > arr[:, :, 2] * 1.6)
    covers = [c for c in _components(red) if c["area"] > 2000]
    if len(covers) != 2:
        raise OracleAbstention(f"expected two red covers, found {len(covers)}")
    covers.sort(key=lambda c: c["cx"])
    for cover in covers:
        box_w = cover["x1"] - cover["x0"] + 1
        box_h = cover["y1"] - cover["y0"] + 1
        # The lower area bound is intentionally strict: material erasure of
        # either decisive host region forces abstention instead of hallucination.
        if not (185 <= box_w <= 195 and 283 <= box_h <= 293 and cover["area"] >= 37000):
            raise OracleAbstention("a red cover is incomplete or malformed")

    dark = (arr[:, :, 0] < 45) & (arr[:, :, 1] < 45) & (arr[:, :, 2] < 45)
    marks = []
    for component in _components(dark):
        w = component["x1"] - component["x0"] + 1
        h = component["y1"] - component["y0"] + 1
        if 500 <= component["area"] <= 1150 and 27 <= w <= 39 and 27 <= h <= 39 and abs(w - h) <= 3:
            marks.append(component)

    counts = []
    for cover in covers:
        enclosed = [mark for mark in marks if _inside_cover(mark, cover)]
        if not 2 <= len(enclosed) <= 4:
            raise OracleAbstention("enclosed spot count is outside the declared range")
        counts.append(len(enclosed))
    if len(marks) != sum(counts):
        raise OracleAbstention("unbound dark circular mark detected")
    return "yes" if counts[0] == counts[1] else "no"
