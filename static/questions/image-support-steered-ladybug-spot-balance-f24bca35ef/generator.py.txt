"""Deterministic renderer for bilateral ladybug spot-count comparison."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 400
SCALE = 3
BACKGROUND = (250, 246, 235)
RED = (218, 52, 68)
RED_EDGE = (137, 28, 43)
BLACK = (15, 18, 23)
NAVY = (38, 52, 70)
CREAM = (255, 226, 151)


def _box(cx: float, cy: float, rx: float, ry: float) -> list[float]:
    return [cx - rx, cy - ry, cx + rx, cy + ry]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    equal = int(seed) % 2 == 0
    left_count = 2 + ((int(seed) // 2) % 3)
    if equal:
        right_count = left_count
    else:
        # Both sides take every count in both classes over consecutive seeds.
        step = 1 if ((int(seed) // 6) % 2 == 0) else 2
        right_count = 2 + ((left_count - 2 + step) % 3)

    dx = rng.randint(-7, 7)
    dy = rng.randint(-8, 8)
    # Identical host geometry across controls and counterfactuals prevents cover
    # size from becoming a label cue; only global translation varies.
    rx = 96
    ry = 146
    left_cover = _box(120 + dx, 201 + dy, rx, ry)
    right_cover = _box(520 + dx, 201 + dy, rx, ry)
    body_box = _box(320 + dx, 218 + dy, 33, 118)
    head_box = _box(320 + dx, 75 + dy, 48, 43)

    slots = [(-34, -82), (31, -61), (-40, -9), (36, 25), (-5, 83)]

    def choose_spots(count: int, side: str) -> list[list[float]]:
        chosen = rng.sample(slots, count)
        spots = []
        center_x = (120 if side == "left" else 520) + dx
        center_y = 201 + dy
        for ox, oy in chosen:
            # Mirror the slot field but retain independent small pixel-visible jitter.
            sx = center_x + (ox if side == "left" else -ox) + rng.randint(-3, 3)
            sy = center_y + oy + rng.randint(-3, 3)
            radius = rng.randint(15, 18)
            spots.append([sx, sy, radius])
        rng.shuffle(spots)
        return spots

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "left_cover": left_cover,
        "right_cover": right_cover,
        "left_spots": choose_spots(left_count, "left"),
        "right_spots": choose_spots(right_count, "right"),
        "body_box": body_box,
        "head_box": head_box,
        "palette": {
            "background": list(BACKGROUND),
            "red": list(RED),
            "red_edge": list(RED_EDGE),
            "black": list(BLACK),
            "navy": list(NAVY),
            "cream": list(CREAM),
        },
    }


def _scaled_box(box: list[float]) -> tuple[int, int, int, int]:
    return tuple(round(float(v) * SCALE) for v in box)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    palette = scene["palette"]
    image = Image.new("RGB", (width * SCALE, height * SCALE), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)

    for cover in (scene["left_cover"], scene["right_cover"]):
        draw.ellipse(_scaled_box(cover), fill=tuple(palette["red"]), outline=tuple(palette["red_edge"]), width=3 * SCALE)

    # Neutral central pieces make the iconic prior recognizable without joining
    # the two decisive red regions.
    draw.ellipse(_scaled_box(scene["body_box"]), fill=tuple(palette["navy"]))
    draw.ellipse(_scaled_box(scene["head_box"]), fill=tuple(palette["navy"]))
    hb = scene["head_box"]
    hx = (hb[0] + hb[2]) / 2
    hy = (hb[1] + hb[3]) / 2
    for sign in (-1, 1):
        draw.line(
            [(round((hx + sign * 18) * SCALE), round((hy - 25) * SCALE)),
             (round((hx + sign * 38) * SCALE), round((hy - 55) * SCALE))],
            fill=tuple(palette["navy"]), width=4 * SCALE,
        )
        r = 6
        draw.ellipse(_scaled_box(_box(hx + sign * 40, hy - 58, r, r)), fill=tuple(palette["cream"]), outline=tuple(palette["navy"]), width=2 * SCALE)

    for spot in [*scene["left_spots"], *scene["right_spots"]]:
        cx, cy, radius = map(float, spot)
        draw.ellipse(_scaled_box(_box(cx, cy, radius, radius)), fill=tuple(palette["black"]))

    resampling = getattr(Image, "Resampling", Image).LANCZOS
    return image.resize((width, height), resampling)


def analytic_gold(scene: dict) -> str:
    return "yes" if len(scene["left_spots"]) == len(scene["right_spots"]) else "no"


def margin(scene: dict) -> float:
    difference = abs(len(scene["left_spots"]) - len(scene["right_spots"]))
    return float(max(1, difference))


def is_quarantined(scene: dict) -> bool:
    all_spots = [*scene["left_spots"], *scene["right_spots"]]
    if any(float(spot[2]) < 13 for spot in all_spots):
        return True
    for spots in (scene["left_spots"], scene["right_spots"]):
        for i, a in enumerate(spots):
            for b in spots[i + 1:]:
                if math.hypot(a[0] - b[0], a[1] - b[1]) <= a[2] + b[2] + 5:
                    return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["left_spots"] = list(reversed(twin["left_spots"]))
    twin["right_spots"] = list(reversed(twin["right_spots"]))
    return [("reverse_spot_storage_order", twin)]
