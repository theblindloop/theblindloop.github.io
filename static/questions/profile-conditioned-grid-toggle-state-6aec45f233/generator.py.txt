"""Deterministic raster renderer and analytic side for ordered grid toggles."""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("ON", "OFF")
CANVAS = (1600, 520)
PANEL_X0 = 20
PANEL_STEP = 310
PANEL_TOP = 100
PANEL_W = 220
PANEL_H = 360
PANEL_ORDER = (1, 2, 3, 4, 5)
GRID_LEFT = 46
GRID_TOP = 76
CELL_SIZE = 32
GRID_SIZE = 4
GLYPH_CELL = 8
GLYPH_SIZE = GLYPH_CELL * GRID_SIZE
GLYPH_TOP = 55

BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
INK = (35, 44, 54)
MUTED = (100, 111, 122)
GRID_LINE = (145, 157, 169)
ON_FILL = (45, 116, 193)
OFF_FILL = (255, 255, 255)
RING = (229, 139, 30)
OP_MARK = (196, 83, 64)
OP_GRID = (207, 214, 221)

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _panel_left(index: int) -> int:
    return PANEL_X0 + index * PANEL_STEP


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _replay(initial_grid: list[list[int]], operations: list[dict], count: int | None = None) -> list[list[int]]:
    state = [[int(value) for value in row] for row in initial_grid]
    if len(state) != GRID_SIZE or any(len(row) != GRID_SIZE for row in state):
        raise ValueError("initial grid must be four by four")
    events = operations if count is None else operations[:count]
    for operation in events:
        axis = str(operation["axis"])
        index = int(operation["index"])
        if axis not in {"row", "col"} or not 0 <= index < GRID_SIZE:
            raise ValueError("invalid row or column operation")
        if axis == "row":
            for column in range(GRID_SIZE):
                state[index][column] ^= 1
        else:
            for row in range(GRID_SIZE):
                state[row][index] ^= 1
    return state


def sample_scene(seed: int) -> dict:
    """Sample balanced five-panel grid scenes with independent style noise."""

    event_rng = random.Random(seed * 1000003 + 17)
    style_rng = random.Random(seed * 1000033 + 31)

    ringed_cell = [event_rng.randrange(GRID_SIZE), event_rng.randrange(GRID_SIZE)]
    axes = ["row", "row", "col", "col"]
    event_rng.shuffle(axes)
    operations = [
        {"axis": axis, "index": event_rng.randrange(GRID_SIZE)} for axis in axes
    ]

    # The class schedule is independent of visual style and alternates under
    # the generator stride, while the marked initial bit is solved so that the
    # final state has the requested class.
    desired = int((seed + 1) % 2)
    initial_grid = [
        [event_rng.randrange(2) for _ in range(GRID_SIZE)] for _ in range(GRID_SIZE)
    ]
    target_row, target_col = ringed_cell
    target_flip_parity = sum(
        operation["axis"] == "row" and operation["index"] == target_row
        or operation["axis"] == "col" and operation["index"] == target_col
        for operation in operations
    ) % 2
    initial_grid[target_row][target_col] = desired ^ target_flip_parity

    near_boundary = seed % 7 == 0
    ring_radius = 25 if near_boundary else 21
    ring_clearance = round(float(24 - ring_radius), 6)
    scene = {
        "canvas": list(CANVAS),
        "panel_order": list(PANEL_ORDER),
        "initial_grid": initial_grid,
        "ringed_cell": ringed_cell,
        "operations": operations,
        "cell_size": CELL_SIZE,
        "ring_radius": ring_radius,
        "ring_clearance": ring_clearance,
    }
    final_grid = _replay(initial_grid, operations)
    if final_grid[target_row][target_col] != desired:
        raise AssertionError("sampler failed to balance the requested final class")
    return scene


def _draw_operation(draw: ImageDraw.ImageDraw, panel_index: int, operation: dict, small_font) -> None:
    left = _panel_left(panel_index)
    glyph_left = left + PANEL_W + (PANEL_STEP - PANEL_W - GLYPH_SIZE) // 2
    glyph_top = PANEL_TOP + GLYPH_TOP
    axis = str(operation["axis"])
    index = int(operation["index"])
    for row in range(GRID_SIZE):
        for column in range(GRID_SIZE):
            box = (
                glyph_left + column * GLYPH_CELL,
                glyph_top + row * GLYPH_CELL,
                glyph_left + (column + 1) * GLYPH_CELL - 1,
                glyph_top + (row + 1) * GLYPH_CELL - 1,
            )
            selected = (axis == "row" and row == index) or (axis == "col" and column == index)
            draw.rectangle(box, fill=OP_MARK if selected else PANEL_FILL, outline=OP_MARK if selected else OP_GRID)
    _centered_text(draw, (glyph_left + GLYPH_SIZE / 2, PANEL_TOP + 112), f"{axis.upper()} {index + 1}", small_font, OP_MARK)
    _centered_text(draw, (glyph_left + GLYPH_SIZE / 2, PANEL_TOP + 130), "flip", small_font, MUTED)

    arrow_y = PANEL_TOP + 193
    next_left = _panel_left(panel_index + 1)
    draw.line((left + PANEL_W + 5, arrow_y, next_left - 9, arrow_y), fill=MUTED, width=2)
    draw.polygon(
        ((next_left - 9, arrow_y), (next_left - 16, arrow_y - 4), (next_left - 16, arrow_y + 4)),
        fill=MUTED,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(16)
    panel_font = _font(16)
    small_font = _font(11)
    cell_font = _font(10)

    draw.text((20, 16), "GRID TOGGLES", font=title_font, fill=INK)
    draw.text(
        (20, 43),
        "Filled = ON, empty = OFF; each arrow flips one complete row or column",
        font=small_font,
        fill=MUTED,
    )
    draw.text((1325, 22), "P1  →  P5", font=small_font, fill=MUTED)
    draw.rectangle((325, 23, 343, 41), fill=ON_FILL, outline=INK, width=1)
    draw.text((350, 25), "ON", font=small_font, fill=INK)
    draw.rectangle((395, 23, 413, 41), fill=OFF_FILL, outline=INK, width=1)
    draw.text((420, 25), "OFF", font=small_font, fill=INK)
    draw.ellipse((485, 24, 505, 44), outline=RING, width=2)
    draw.text((512, 25), "ringed cell", font=small_font, fill=INK)

    for panel_index, panel_number in enumerate(scene["panel_order"]):
        left = _panel_left(panel_index)
        top = PANEL_TOP
        right = left + PANEL_W
        bottom = top + PANEL_H
        draw.rounded_rectangle((left, top, right, bottom), radius=8, fill=PANEL_FILL, outline=INK, width=2)
        draw.text((left + 10, top + 10), f"P{panel_number}", font=panel_font, fill=INK)
        state = _replay(scene["initial_grid"], scene["operations"], panel_index)
        for row in range(GRID_SIZE):
            for column in range(GRID_SIZE):
                x0 = left + GRID_LEFT + column * scene["cell_size"]
                y0 = top + GRID_TOP + row * scene["cell_size"]
                x1 = x0 + scene["cell_size"] - 1
                y1 = y0 + scene["cell_size"] - 1
                fill = ON_FILL if state[row][column] else OFF_FILL
                draw.rectangle((x0, y0, x1, y1), fill=fill, outline=GRID_LINE, width=2)
        if panel_index == 0:
            target_row, target_col = scene["ringed_cell"]
            cx = left + GRID_LEFT + target_col * scene["cell_size"] + scene["cell_size"] / 2
            cy = top + GRID_TOP + target_row * scene["cell_size"] + scene["cell_size"] / 2
            radius = scene["ring_radius"]
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), outline=RING, width=3)

        if panel_index < len(scene["panel_order"]) - 1:
            _draw_operation(draw, panel_index, scene["operations"][panel_index], small_font)

    return image


def analytic_gold(scene: dict) -> str:
    state = _replay(scene["initial_grid"], scene["operations"])
    row, column = scene["ringed_cell"]
    return LABELS[0] if state[row][column] else LABELS[1]


def margin(scene: dict) -> float:
    """Clearance of the ring from the neighboring-cell decision boundary."""

    return float(scene["ring_clearance"])


def is_quarantined(scene: dict) -> bool:
    return float(scene["ring_clearance"]) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # There is no unrendered record-order field: all answer-bearing inputs are
    # explicitly drawn, so this candidate declares latent aliases inapplicable.
    return []
