"""Independent image-only inverse for the ordered grid-toggle world."""

from __future__ import annotations

import math

from PIL import Image


CANVAS = (1600, 520)
PANEL_X0 = 20
PANEL_STEP = 310
PANEL_TOP = 100
PANEL_W = 220
PANEL_H = 360
GRID_LEFT = 46
GRID_TOP = 76
CELL_SIZE = 32
GRID_SIZE = 4
GLYPH_CELL = 8
GLYPH_SIZE = 32
GLYPH_TOP = 55

ON_FILL = (45, 116, 193)
OFF_FILL = (255, 255, 255)
RING = (229, 139, 30)
OP_MARK = (196, 83, 64)


def _panel_left(index: int) -> int:
    return PANEL_X0 + index * PANEL_STEP


def _ringed_cell(frame: Image.Image, left: int) -> tuple[int, int]:
    points: list[tuple[int, int]] = []
    pixels = frame.load()
    for y in range(PANEL_TOP, PANEL_TOP + PANEL_H):
        for x in range(left, left + PANEL_W):
            if pixels[x, y] == RING:
                points.append((x, y))
    if len(points) < 20:
        raise ValueError("P1 does not expose a ring")
    center = (sum(point[0] for point in points) / len(points), sum(point[1] for point in points) / len(points))
    centers = [
        (
            left + GRID_LEFT + column * CELL_SIZE + CELL_SIZE / 2,
            PANEL_TOP + GRID_TOP + row * CELL_SIZE + CELL_SIZE / 2,
            row,
            column,
        )
        for row in range(GRID_SIZE)
        for column in range(GRID_SIZE)
    ]
    nearest = min(centers, key=lambda item: math.hypot(center[0] - item[0], center[1] - item[1]))
    if math.hypot(center[0] - nearest[0], center[1] - nearest[1]) > 10.0:
        raise ValueError("ring does not identify one grid cell")
    return int(nearest[2]), int(nearest[3])


def _panel_state(frame: Image.Image, left: int) -> list[list[int]]:
    state: list[list[int]] = []
    pixels = frame.load()
    for row in range(GRID_SIZE):
        values: list[int] = []
        for column in range(GRID_SIZE):
            x = left + GRID_LEFT + column * CELL_SIZE + CELL_SIZE // 2
            y = PANEL_TOP + GRID_TOP + row * CELL_SIZE + CELL_SIZE // 2
            color = pixels[x, y]
            if color == ON_FILL:
                values.append(1)
            elif color == OFF_FILL:
                values.append(0)
            else:
                raise ValueError("grid cell center is not a clear ON/OFF pixel")
        state.append(values)
    return state


def _operation(frame: Image.Image, panel_index: int) -> tuple[str, int]:
    left = _panel_left(panel_index)
    glyph_left = left + PANEL_W + (PANEL_STEP - PANEL_W - GLYPH_SIZE) // 2
    glyph_top = PANEL_TOP + GLYPH_TOP
    pixels = frame.load()
    marks = [
        [
            pixels[glyph_left + column * GLYPH_CELL + GLYPH_CELL // 2, glyph_top + row * GLYPH_CELL + GLYPH_CELL // 2] == OP_MARK
            for column in range(GRID_SIZE)
        ]
        for row in range(GRID_SIZE)
    ]
    if sum(sum(int(value) for value in row) for row in marks) != GRID_SIZE:
        raise ValueError("operation glyph does not mark exactly one line")
    row_counts = [sum(int(value) for value in row) for row in marks]
    col_counts = [sum(int(marks[row][column]) for row in range(GRID_SIZE)) for column in range(GRID_SIZE)]
    if 4 in row_counts and row_counts.count(4) == 1 and all(value in {0, 4} for value in row_counts):
        return "row", row_counts.index(4)
    if 4 in col_counts and col_counts.count(4) == 1 and all(value in {0, 4} for value in col_counts):
        return "col", col_counts.index(4)
    raise ValueError("operation glyph has ambiguous orientation")


def _flip(state: list[list[int]], axis: str, index: int) -> None:
    if axis == "row":
        for column in range(GRID_SIZE):
            state[index][column] ^= 1
    elif axis == "col":
        for row in range(GRID_SIZE):
            state[row][index] ^= 1
    else:
        raise ValueError("unknown operation axis")


def decision_from_image(image: Image.Image) -> str:
    """Decode the visible panels and replay every line toggle from P1."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    ring_row, ring_column = _ringed_cell(frame, _panel_left(0))
    states = [_panel_state(frame, _panel_left(index)) for index in range(5)]
    operations = [_operation(frame, index) for index in range(4)]
    replay = [row[:] for row in states[0]]
    for panel_index, (axis, index) in enumerate(operations, start=1):
        _flip(replay, axis, index)
        if replay != states[panel_index]:
            raise ValueError("visible panel does not match the declared toggle")
    return "ON" if replay[ring_row][ring_column] else "OFF"
