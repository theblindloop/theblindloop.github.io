"""Independent inverse arm: only final RGB pixels enter this module."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

INK = np.array([37, 42, 48], dtype=np.int16)
GREEN = np.array([32, 166, 106], dtype=np.int16)
TERMINALS = {
    "orange": np.array([242, 142, 43], dtype=np.int16),
    "purple": np.array([148, 103, 189], dtype=np.int16),
    "blue": np.array([59, 130, 208], dtype=np.int16),
}


def _near(arr: np.ndarray, color: np.ndarray, radius: int) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return (delta * delta).sum(axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > 450:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _centroid(mask: np.ndarray, label: str) -> np.ndarray:
    points = np.argwhere(mask)
    if len(points) < 60:
        raise ValueError(f"missing {label} marker")
    return points.mean(axis=0)


def _nearest_component(center: np.ndarray, components: list[np.ndarray]) -> tuple[int, float, float]:
    distances = []
    for component in components:
        distances.append(float(np.sqrt(((component - center) ** 2).sum(axis=1)).min()))
    order = np.argsort(distances)
    if len(order) < 2:
        raise ValueError("insufficient cable components")
    return int(order[0]), distances[int(order[0])], distances[int(order[1])]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    components = _components(_near(arr, INK, 38))
    if len(components) != 3:
        raise ValueError(f"expected three disconnected dark cables, found {len(components)}")
    green_center = _centroid(_near(arr, GREEN, 48), "green start")
    target, nearest, runner_up = _nearest_component(green_center, components)
    if runner_up - nearest < 24.0:
        raise ValueError("ambiguous green-to-cable association")

    matches = []
    for label, color in TERMINALS.items():
        center = _centroid(_near(arr, color, 50), label)
        component, first, second = _nearest_component(center, components)
        if second - first < 24.0:
            raise ValueError(f"ambiguous {label}-to-cable association")
        if component == target:
            matches.append(label)
    if len(matches) != 1:
        raise ValueError(f"target cable has {len(matches)} terminal matches")
    return matches[0]
