"""Deterministic renderer and analytic arm for cable terminal tracing."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BG = "#f7f6f2"
INK = "#252a30"
GREEN = "#20a66a"
GRAY = "#a8adb4"
COLORS = {"orange": "#f28e2b", "purple": "#9467bd", "blue": "#3b82d0"}

# Three disjoint, nonbranching routes occupying different interlocking regions.
# The transforms below move all visible geometry together and preserve topology.
BASE_ROUTES = [
    [[54, 70], [244, 70], [244, 132], [104, 132], [104, 232], [52, 232]],
    [[300, 50], [300, 190], [438, 190], [438, 286], [370, 286], [370, 320]],
    [[54, 330], [208, 330], [208, 250], [314, 250], [314, 382], [462, 382]],
]


def _transform(points: list[list[int]], mode: int) -> list[list[int]]:
    result = []
    for x, y in points:
        if mode == 0:
            nx, ny = x, y
        elif mode == 1:
            nx, ny = WIDTH - x, y
        elif mode == 2:
            nx, ny = x, HEIGHT - y
        else:
            nx, ny = WIDTH - y, x
        result.append([int(nx), int(ny)])
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 3
    shift = (int(seed) // 3) % 3
    palette = ["orange", "purple", "blue"]
    terminal_colors = [palette[(i + shift) % 3] for i in range(3)]
    mode = rng.randrange(4)
    reverse_all = bool(rng.getrandbits(1))
    routes = []
    for index, base in enumerate(BASE_ROUTES):
        points = _transform(base, mode)
        if reverse_all:
            points = list(reversed(points))
        routes.append(
            {
                "points": points,
                "start": "green" if index == target else "gray",
                "terminal": terminal_colors[index],
            }
        )
    # Shuffle draw/list order so the answer is not tied to a record slot.
    rng.shuffle(routes)
    return {"routes": routes}


def analytic_gold(scene: dict) -> str:
    green_routes = [route for route in scene["routes"] if route["start"] == "green"]
    if len(green_routes) != 1:
        raise ValueError("scene must contain exactly one green start")
    terminal = green_routes[0]["terminal"]
    if terminal not in COLORS:
        raise ValueError("unknown terminal color")
    return terminal


def _point_distance(a: list[int], b: list[int]) -> float:
    return math.hypot(float(a[0] - b[0]), float(a[1] - b[1]))


def margin(scene: dict) -> float:
    routes = scene["routes"]
    endpoints = [point for route in routes for point in (route["points"][0], route["points"][-1])]
    return float(min(_point_distance(a, b) for i, a in enumerate(endpoints) for b in endpoints[i + 1 :]))


def is_quarantined(scene: dict) -> bool:
    try:
        return len(scene["routes"]) != 3 or margin(scene) < 24.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BG)
    draw = ImageDraw.Draw(image)

    def box(values):
        return tuple(int(v * scale) for v in values)

    # A subtle frame establishes the visual field without participating in the trace.
    draw.rounded_rectangle(box((20, 20, 492, 492)), radius=22 * scale, fill="#ffffff", outline="#d9dce0", width=2 * scale)
    for route in scene["routes"]:
        points = [tuple(int(v * scale) for v in point) for point in route["points"]]
        draw.line(points, fill=INK, width=9 * scale, joint="curve")

    # Markers are drawn after strokes, making the intended endpoints explicit.
    for route in scene["routes"]:
        sx, sy = route["points"][0]
        ex, ey = route["points"][-1]
        ring = GREEN if route["start"] == "green" else GRAY
        draw.ellipse(box((sx - 14, sy - 14, sx + 14, sy + 14)), fill="#ffffff", outline=ring, width=6 * scale)
        draw.rounded_rectangle(
            box((ex - 14, ey - 14, ex + 14, ey + 14)),
            radius=4 * scale,
            fill=COLORS[route["terminal"]],
            outline="#ffffff",
            width=3 * scale,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
