"""Independent pixel-only oracle for the focal twin-glyph world.

Receives only the final PNG. It never sees the latent scene, renderer geometry
API, or analytic gold. It re-measures the raster independently:

1. It performs nearest-palette colour segmentation against the (white background,
   three-colour palette) world.
2. It extracts the connected components of non-background pixels; a legitimate
   scene contains exactly the two small disconnected discs.
3. If the number of such components is not exactly two (a disc was removed,
   occluded, or replaced by background), the oracle abstains by raising. This is
   what makes each of the two disconnected components jointly decisive.
4. For each of the two discs it reads the dominant bound attribute colour and
   answers whether they are the same.

The colour letters and segmentation constants are re-derived independently from
the raster; no renderer/gold code is imported.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

ORACLE_VERSION = "focal_disc_attribute_match-oracle-0.1.0"

CANVAS = 512
WHITE = (255, 255, 255)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)


class TwinDiscsCompromisedError(ValueError):
    """One or both of the two decisive disconnected discs are missing/damaged."""


def _nearest_palette_color(pixel: np.ndarray) -> str | None:
    best_name, best_dist = None, None
    for name, rgb in PALETTE.items():
        d = float(np.linalg.norm(pixel - np.array(rgb, dtype=np.float32)))
        if best_dist is None or d < best_dist:
            best_dist, best_name = d, name
    return best_name


def _component_labels(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = set()
    components = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or (y, x) in seen:
                continue
            stack = [(y, x)]
            seen.add((y, x))
            cells = []
            while stack:
                cy, cx = stack.pop()
                cells.append((cy, cx))
                for neighbor in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    ny, nx = neighbor
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and neighbor not in seen
                    ):
                        seen.add(neighbor)
                        stack.append(neighbor)
            components.append(cells)
    return components


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (CANVAS, CANVAS):
        raise TwinDiscsCompromisedError("unexpected image size")
    arr = np.asarray(image, dtype=np.uint8)
    flat = arr.reshape(-1, 3).astype(np.float32)
    palette_dist = np.stack(
        [np.linalg.norm(flat - np.array(rgb, dtype=np.float32), axis=1) for rgb in PALETTE.values()],
        axis=-1,
    )
    nearest_idx = palette_dist.argmin(axis=1)
    background = (
        np.linalg.norm(flat - np.array(WHITE, dtype=np.float32), axis=1) <= 24.0
    )
    labels = np.where(background, -1, nearest_idx).reshape(arr.shape[:2])

    mask = (labels >= 0)
    components = _component_labels(mask)
    if len(components) != 2:
        raise TwinDiscsCompromisedError(
            f"expected exactly two disconnected discs, found {len(components)}"
        )

    disc_colours = []
    for cells in components:
        votes: Counter = Counter()
        for y, x in cells:
            name = PALETTE_NAMES[int(labels[y, x])]
            votes[name] += 1
        if not votes:
            raise TwinDiscsCompromisedError("a disc carries no dominant colour")
        disc_colours.append(votes.most_common(1)[0][0])

    return "yes" if disc_colours[0] == disc_colours[1] else "no"
