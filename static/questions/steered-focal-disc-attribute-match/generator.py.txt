"""Focal disjoint twin-attribute correspondence world.

A small local inset (position randomized across scenes, well inside the canvas)
contains exactly two disconnected coloured discs, separated by visible white
background. The decision is an attribute-correspondence predicate: do the two
disconnected glyphs carry the SAME bound attribute colour? Both discs are jointly
decisive and both sit in one small local span.

Latent state z = {disc_left_color, disc_right_color, ox, oy}. Only the two colour
fields influence analytic gold; the inset position (ox, oy) is painted into the
raster so the decisive support sits in one focal neighborhood and is randomized
across scenes. The parametrisation is minimal / injective over its label inputs
(no nontrivial pixel-identical symmetry), declared ``not_applicable``.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "focal_disc_attribute_match-0.1.0"

CANVAS = 512
CELL = 64
DISC_RADIUS = 12

WHITE = (255, 255, 255)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

# Coarse cell-aligned origins so the two discs deterministically occupy two
# grid-adjacent-but-disjoint cells with a full empty cell between them in the
# protected 8x8 ablation grid, preserving focal + multipart support everywhere.
ORIGIN_X = [CELL, 2 * CELL, 3 * CELL, 4 * CELL]
ORIGIN_Y = [CELL, 2 * CELL, 3 * CELL, 4 * CELL]

# Relative disc centres inside the inset: 2 cells apart horizontally.
_DISC_1_OFF = (32, 32)
_DISC_2_OFF = (160, 32)


def _disc_centres(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    ox, oy = scene["ox"], scene["oy"]
    return (ox + _DISC_1_OFF[0], oy + _DISC_1_OFF[1]), (
        ox + _DISC_2_OFF[0],
        oy + _DISC_2_OFF[1],
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "disc_left_color": rng.choice(PALETTE_NAMES),
        "disc_right_color": rng.choice(PALETTE_NAMES),
        "ox": rng.choice(ORIGIN_X),
        "oy": rng.choice(ORIGIN_Y),
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    for center, color in zip(
        _disc_centres(scene), (scene["disc_left_color"], scene["disc_right_color"]), strict=True
    ):
        cx, cy = center
        draw.ellipse(
            [cx - DISC_RADIUS, cy - DISC_RADIUS, cx + DISC_RADIUS, cy + DISC_RADIUS],
            fill=PALETTE[color],
        )
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["disc_left_color"] == scene["disc_right_color"] else "no"


def margin(scene: dict) -> float:
    a = np.array(PALETTE[scene["disc_left_color"]], dtype=float)
    b = np.array(PALETTE[scene["disc_right_color"]], dtype=float)
    return float(np.linalg.norm(a - b))


def is_quarantined(scene: dict) -> bool:
    # The palette uses well-separated, pairwise-distinct colours and the discs are
    # solid-filled, so colour identity is a crisp finite-resolution event. Equal
    # colour scenes (margin 0) are a legitimate class, not an ambiguous boundary.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "disc_left_color": scene["disc_left_color"],
        "disc_right_color": scene["disc_right_color"],
        "ox": scene["ox"],
        "oy": scene["oy"],
    }
