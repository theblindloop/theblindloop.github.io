"""Deterministic analytic scene sampler and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480
SUPERSAMPLE = 3
WALL_WIDTH = 10
DOT_RADIUS = 11
BACKGROUND = (250, 249, 245)
NAVY = (28, 55, 82)
ORANGE = (230, 122, 35)
PURPLE = (132, 76, 184)
PALE = (207, 213, 218)


def _wall_x(path: list[list[int]], y: float) -> float:
    """Interpolate the x coordinate of the y-monotone crosscut."""
    for first, second in zip(path, path[1:]):
        x0, y0 = first
        x1, y1 = second
        if y0 <= y <= y1:
            t = (y - y0) / (y1 - y0)
            return x0 + t * (x1 - x0)
    if y < path[0][1]:
        return float(path[0][0])
    return float(path[-1][0])


def _side_for(path: list[list[int]], marker: list[int]) -> int:
    return 1 if marker[0] > _wall_x(path, marker[1]) else -1


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC2055C17)
    same_bank = (int(seed) % 2) == 0
    reverse_bend = bool(rng.getrandbits(1))
    left = rng.randint(132, 174)
    right = rng.randint(466, 508)
    if reverse_bend:
        left, right = right, left
    middle_y = rng.randint(218, 262)
    path = [
        [left + rng.randint(-10, 10), -24],
        [right + rng.randint(-10, 10), middle_y],
        [left + rng.randint(-10, 10), HEIGHT + 24],
    ]

    upper_y = rng.randint(104, 184)
    lower_y = rng.randint(296, 376)
    if same_bank:
        orange_side = purple_side = rng.choice((-1, 1))
    else:
        orange_side = rng.choice((-1, 1))
        purple_side = -orange_side
    orange_y, purple_y = upper_y, lower_y
    if rng.random() < 0.5:
        orange_y, purple_y = purple_y, orange_y
    orange_offset = rng.randint(48, 66)
    purple_offset = rng.randint(48, 66)
    orange = [round(_wall_x(path, orange_y) + orange_side * orange_offset), orange_y]
    purple = [round(_wall_x(path, purple_y) + purple_side * purple_offset), purple_y]

    decorations = [[24, 70], [616, 116], [24, 230], [616, 278], [24, 410], [616, 438]]
    return {
        "path": path,
        "orange": orange,
        "purple": purple,
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y in scene["decorations"]:
        radius = 3 * scale
        draw.ellipse(
            (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
            fill=PALE,
        )

    path = [(x * scale, y * scale) for x, y in scene["path"]]
    draw.line(path, fill=NAVY, width=WALL_WIDTH * scale, joint="curve")
    for x, y in (scene["orange"], scene["purple"]):
        if [x, y] == scene["orange"]:
            color = ORANGE
        else:
            color = PURPLE
        radius = DOT_RADIUS * scale
        draw.ellipse(
            (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
            fill=color,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    path = scene["path"]
    orange = scene["orange"]
    purple = scene["purple"]
    return "yes" if _side_for(path, orange) == _side_for(path, purple) else "no"


def margin(scene: dict) -> float:
    path = scene["path"]
    orange = scene["orange"]
    purple = scene["purple"]
    clearances = [
        abs(marker[0] - _wall_x(path, marker[1])) - DOT_RADIUS - WALL_WIDTH / 2
        for marker in (orange, purple)
    ]
    segment_spans = [second[1] - first[1] for first, second in zip(path, path[1:])]
    return float(min(*clearances, min(segment_spans) / 4.0))


def is_quarantined(scene: dict) -> bool:
    return (not math.isfinite(margin(scene))) or margin(scene) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_identical_decorations", twin)]
