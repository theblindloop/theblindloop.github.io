"""Independent inverse arm: final RGB pixels are its only input."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((28, 55, 82), dtype=np.int16)
ORANGE = np.array((230, 122, 35), dtype=np.int16)
PURPLE = np.array((132, 76, 184), dtype=np.int16)
DOWNSAMPLE = 4


def _near(array: np.ndarray, color: np.ndarray, radius: float) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < 180:
        raise ValueError(f"{name} marker is missing or materially erased")
    if len(points) > 650:
        raise ValueError(f"{name} marker segmentation is ambiguous")
    y, x = points.mean(axis=0)
    return float(x), float(y)


def _validate_crosscut(wall: np.ndarray) -> np.ndarray:
    height, width = wall.shape
    centers = np.empty(height, dtype=np.float64)
    for y in range(height):
        xs = np.flatnonzero(wall[y])
        if len(xs) < 4:
            raise ValueError("navy wall is not an unbroken top-to-bottom crosscut")
        if xs[-1] - xs[0] > 24:
            raise ValueError("navy wall has a branch or ambiguous row")
        centers[y] = float(np.median(xs))
    if np.max(np.abs(np.diff(centers))) > 9.0:
        raise ValueError("navy crosscut is disconnected or changes too abruptly")
    return centers


def _reachable(blocked: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> bool:
    height, width = blocked.shape
    sx, sy = start
    gx, gy = goal
    if not (0 <= sx < width and 0 <= sy < height and 0 <= gx < width and 0 <= gy < height):
        raise ValueError("marker centroid falls outside the raster")
    if blocked[sy, sx] or blocked[gy, gx]:
        raise ValueError("a marker touches the wall")
    seen = np.zeros_like(blocked, dtype=np.uint8)
    queue = deque([(sy, sx)])
    seen[sy, sx] = 1
    while queue:
        y, x = queue.popleft()
        if x == gx and y == gy:
            return True
        if y and not blocked[y - 1, x] and not seen[y - 1, x]:
            seen[y - 1, x] = 1
            queue.append((y - 1, x))
        if y + 1 < height and not blocked[y + 1, x] and not seen[y + 1, x]:
            seen[y + 1, x] = 1
            queue.append((y + 1, x))
        if x and not blocked[y, x - 1] and not seen[y, x - 1]:
            seen[y, x - 1] = 1
            queue.append((y, x - 1))
        if x + 1 < width and not blocked[y, x + 1] and not seen[y, x + 1]:
            seen[y, x + 1] = 1
            queue.append((y, x + 1))
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if (width, height) != (640, 480):
        raise ValueError("unexpected canvas size")
    wall = _near(array, NAVY, 62.0)
    orange = _centroid(_near(array, ORANGE, 66.0), "orange")
    purple = _centroid(_near(array, PURPLE, 66.0), "purple")
    centers = _validate_crosscut(wall)
    for x, y in (orange, purple):
        row = max(0, min(height - 1, round(y)))
        if abs(x - centers[row]) < 24.0:
            raise ValueError("marker is too close to the wall for a stable decision")

    factor = DOWNSAMPLE
    blocked = wall.reshape(height // factor, factor, width // factor, factor).any(axis=(1, 3))
    start = (round(orange[0] / factor), round(orange[1] / factor))
    goal = (round(purple[0] / factor), round(purple[1] / factor))
    return "yes" if _reachable(blocked, start, goal) else "no"
