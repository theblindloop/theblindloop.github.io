"""Independent inverse arm: pixels in, motif decision out.

This module deliberately duplicates neither renderer state nor latent access.
It segments the two public role palettes, recovers each binary chip from its
two disconnected dark cells, executes block substitution via a Kronecker
product, and template-classifies the resulting raster.
"""

from __future__ import annotations

import numpy as np

OUTER_DARKS = ((22, 137, 160), (45, 104, 177))
INNER_DARKS = ((226, 91, 76), (184, 72, 146))

DIAG = np.asarray([[1, 0], [0, 1]], dtype=np.uint8)
ANTI = np.asarray([[0, 1], [1, 0]], dtype=np.uint8)
TEMPLATES = {
    "one falling diagonal": np.kron(DIAG, DIAG),
    "two rising dashes": np.kron(DIAG, ANTI),
    "two falling dashes": np.kron(ANTI, DIAG),
    "one rising diagonal": np.kron(ANTI, ANTI),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        # Filled cells are 10x10.  The role-colored chip outline is a larger
        # hollow component and is a visible locator, not binary payload.
        if 70 <= len(points) <= 130:
            ys, xs = zip(*points)
            found.append((float(np.mean(ys)), float(np.mean(xs)), len(points)))
    return sorted(found)


def _recover_factor(array: np.ndarray, colors: tuple[tuple[int, int, int], ...]) -> np.ndarray:
    mask = np.zeros(array.shape[:2], dtype=bool)
    for color in colors:
        mask |= np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    components = _components(mask)
    if len(components) != 2:
        raise ValueError(f"expected two dark cells for a chip, found {len(components)}")
    (y0, x0, a0), (y1, x1, a1) = components
    if min(a0, a1) < 80 or abs(y1 - y0) < 8 or abs(x1 - x0) < 8:
        raise ValueError("chip cells are incomplete or not diagonally separated")
    return DIAG.copy() if x0 < x1 else ANTI.copy()


def decision_with_margin(image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    outer = _recover_factor(array, OUTER_DARKS)
    inner = _recover_factor(array, INNER_DARKS)
    product = np.kron(outer, inner)
    ranked = sorted(
        ((16 - int(np.count_nonzero(product != template)), name) for name, template in TEMPLATES.items()),
        reverse=True,
    )
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("ambiguous output template")
    return ranked[0][1], float(ranked[0][0] - ranked[1][0])


def decision_from_image(image) -> str:
    return decision_with_margin(image)[0]
