"""Cut-point selection world renderer.

A small network is drawn from black connection points (filled disks) joined by
thick black links. The network is a single simple cycle (a ring of m points)
with one extra leaf point attached to exactly one cycle point by one link. Two
of the points are highlighted: one is drawn red and the other blue. The single
point at which the leaf attaches is a *cut point*: deleting it and all its links
splits the ring from the leaf, increasing the number of connected pieces. Every
other point (other ring points and the leaf) is not a cut point. The question is
which highlighted color marks the cut point.

The scene is a plain dict so the verifier's AST reachability audit stays simple.

    z = {
        "width": 400, "height": 400,
        "center": [cx, cy],
        "ring_points": [[x, y], ...],     # cycle nodes on the ring
        "leaf": [x, y],
        "attachment_index": a,            # index into ring_points that the leaf joins
        "distractor_index": d,            # index into ring_points highlighted as distractor
        "cut_is_red": bool,               # whether the attachment point is drawn red
        "node_radius": 12.0,
        "link_width": 7.0,
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "cut_point_selection-0.4.0"

WHITE = (255, 255, 255)
BLACK = (20, 20, 20)
RED = (220, 40, 40)
BLUE = (40, 60, 220)

NODE_R = 12.0
LINK_W = 7.0
CANVAS_MARGIN = 16.0
QUARANTINE_MIN = 8.0


def _seg_dist(px: float, py: float, x1: float, y1: float, x2: float, y2: float) -> float:
    dx = x2 - x1
    dy = y2 - y1
    l2 = dx * dx + dy * dy
    if l2 == 0:
        return math.hypot(px - x1, py - y1)
    t = max(0.0, min(1.0, ((px - x1) * dx + (py - y1) * dy) / l2))
    return math.hypot(px - (x1 + t * dx), py - (y1 + t * dy))


def sample_scene(rng: random.Random, width: int = 400, height: int = 400) -> dict:
    for _attempt in range(700):
        cx = rng.uniform(185, 215)
        cy = rng.uniform(185, 215)
        m = rng.choice([4, 5, 6, 7])
        R = rng.uniform(60, 100)
        base = rng.uniform(0.0, 2.0 * math.pi)
        ring = []
        for i in range(m):
            a = base + i * 2.0 * math.pi / m
            ring.append([cx + R * math.cos(a), cy + R * math.sin(a)])

        a_idx = rng.randrange(m)
        d_idx = rng.randrange(m)
        while d_idx == a_idx:
            d_idx = rng.randrange(m)

        ax, ay = ring[a_idx]
        ang = math.atan2(ay - cy, ax - cx)
        leaf_dist = rng.uniform(26.0, 50.0)
        leaf = [ax + leaf_dist * math.cos(ang), ay + leaf_dist * math.sin(ang)]
        if not (CANVAS_MARGIN < leaf[0] < width - CANVAS_MARGIN):
            continue
        if not (CANVAS_MARGIN < leaf[1] < height - CANVAS_MARGIN):
            continue

        nodes = ring + [leaf]
        links = set()
        for i in range(m):
            j = (i + 1) % m
            links.add((min(i, j), max(i, j)))
        links.add((min(a_idx, m), max(a_idx, m)))
        links = sorted(links)

        ok = True
        for i in range(len(nodes)):
            for j in range(i + 1, len(nodes)):
                if (i, j) in links or (j, i) in links:
                    continue
                if math.hypot(nodes[i][0] - nodes[j][0], nodes[i][1] - nodes[j][1]) < 2 * NODE_R + 8:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            continue

        for i, (nx, ny) in enumerate(nodes):
            for si, (ii, jj) in enumerate(links):
                if i in (ii, jj):
                    continue
                x1, y1 = nodes[ii]
                x2, y2 = nodes[jj]
                dd = _seg_dist(nx, ny, x1, y1, x2, y2)
                if dd < NODE_R + LINK_W / 2.0 + 6.0:
                    ok = False
                    break
            if not ok:
                break
        if not ok:
            continue

        cut_is_red = rng.random() < 0.5
        return {
            "width": width,
            "height": height,
            "center": [cx, cy],
            "ring_points": ring,
            "leaf": leaf,
            "attachment_index": a_idx,
            "distractor_index": d_idx,
            "cut_is_red": cut_is_red,
            "node_radius": NODE_R,
            "link_width": LINK_W,
        }
    return _fallback_scene()


def _fallback_scene() -> dict:
    cx, cy = 200.0, 200.0
    m = 5
    base = 0.0
    ring = []
    for i in range(m):
        a = base + i * 2.0 * math.pi / m
        ring.append([cx + 80.0 * math.cos(a), cy + 80.0 * math.sin(a)])
    leaf = [cx + 130.0, cy]
    return {
        "width": 400,
        "height": 400,
        "center": [cx, cy],
        "ring_points": ring,
        "leaf": leaf,
        "attachment_index": 0,
        "distractor_index": 2,
        "cut_is_red": True,
        "node_radius": NODE_R,
        "link_width": LINK_W,
    }


def _node_colors(scene: dict) -> list:
    n = len(scene["ring_points"])
    a = scene["attachment_index"]
    d = scene["distractor_index"]
    cut_is_red = bool(scene["cut_is_red"])
    colors = [BLACK] * (n + 1)
    colors[a] = RED if cut_is_red else BLUE
    colors[d] = BLUE if cut_is_red else RED
    return colors


def _incident_links(scene: dict) -> list[tuple[int, int]]:
    n = len(scene["ring_points"])
    links = []
    for i in range(n):
        j = (i + 1) % n
        links.append((i, j))
    links.append((scene["attachment_index"], n))
    return links


def render(scene: dict, supersample: int = 1) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    r = float(scene["node_radius"]) * ss
    lw = max(1, int(round(float(scene["link_width"]) * ss)))
    nodes = list(scene["ring_points"]) + [scene["leaf"]]
    colors = _node_colors(scene)
    for (i, j) in _incident_links(scene):
        x1, y1 = nodes[i]
        x2, y2 = nodes[j]
        draw.line([(x1 * ss, y1 * ss), (x2 * ss, y2 * ss)], fill=BLACK, width=lw)
    for (x, y), col in zip(nodes, colors):
        draw.ellipse(
            [(x - r) * ss, (y - r) * ss, (x + r) * ss, (y + r) * ss], fill=col
        )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    return "red" if bool(scene["cut_is_red"]) else "blue"


def margin(scene: dict) -> float:
    links = _incident_links(scene)
    nodes = list(scene["ring_points"]) + [scene["leaf"]]
    r = float(scene["node_radius"])
    lw = float(scene["link_width"])
    best = float("inf")
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            if (i, j) in links or (j, i) in links:
                continue
            best = min(
                best,
                math.hypot(nodes[i][0] - nodes[j][0], nodes[i][1] - nodes[j][1]) - 2.0 * r,
            )
    for i, (nx, ny) in enumerate(nodes):
        for (ii, jj) in links:
            if i in (ii, jj):
                continue
            x1, y1 = nodes[ii]
            x2, y2 = nodes[jj]
            dd = _seg_dist(nx, ny, x1, y1, x2, y2)
            best = min(best, dd - r - lw / 2.0)
    if math.isinf(best):
        return float("inf")
    return max(0.0, float(best))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render-invariant symmetry: every stored point coordinate
    and the cut/is-red label change the raster, and the analytic decision is a
    pure function of those same label fields. Declared not_applicable."""
    return iter(())
