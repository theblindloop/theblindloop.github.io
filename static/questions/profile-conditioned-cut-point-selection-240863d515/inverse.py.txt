"""Independent pixel-only inverse arm for the cut-point-selection world.

Receives ONLY the final PNG. It never imports or reads the renderer, the analytic
scene, prompts, or renderer state. It recovers the decision by:

1. segmenting the red and blue disks and taking their centroids (the two
   highlighted points);
2. finding every black connection point by eroding the black ink (which removes
   the thin links but keeps the thick disks) and taking each surviving core's
   centroid;
3. reconstructing the network graph by joining any two point centers whose
   straight chord is covered by ink;
4. computing the articulation (cut) points of that graph by the standard
   depth-first biconnected decomposition;
5. returning the color whose highlighted point is a cut point (red on a
   red-cut / blue-not-cut result, blue on the opposite).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageFilter

ORACLE_VERSION = "cut_point_selection-oracle-0.4.0"

_RED = (220, 40, 40)
_BLUE = (40, 60, 220)
_ERODE_BOX = 13


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    labels = np.zeros_like(mask, dtype=np.int32)
    current = 0
    result: list[np.ndarray] = []
    for y in range(h):
        for x in range(w):
            if not mask[y, x] or labels[y, x]:
                continue
            current += 1
            labels[y, x] = current
            stack = [(x, y)]
            while stack:
                cx, cy = stack.pop()
                for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                    if (
                        0 <= nx < w
                        and 0 <= ny < h
                        and mask[ny, nx]
                        and labels[ny, nx] == 0
                    ):
                        labels[ny, nx] = current
                        stack.append((nx, ny))
    for label in range(1, current + 1):
        result.append(labels == label)
    return result


def _articulation(n: int, adj: list[list[int]]) -> set[int]:
    disc = [-1] * n
    low = [0] * n
    parent = [-1] * n
    art: set[int] = set()
    time = [0]

    def dfs(u: int) -> None:
        disc[u] = low[u] = time[0]
        time[0] += 1
        children = 0
        for v in adj[u]:
            if disc[v] == -1:
                parent[v] = u
                children += 1
                dfs(v)
                low[u] = min(low[u], low[v])
                if parent[u] == -1 and children > 1:
                    art.add(u)
                if parent[u] != -1 and low[v] >= disc[u]:
                    art.add(u)
            elif v != parent[u]:
                low[u] = min(low[u], disc[v])

    for u in range(n):
        if disc[u] == -1:
            dfs(u)
    return art


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    h, w = arr.shape[0], arr.shape[1]

    nonwhite = (arr[..., 0] < 225) | (arr[..., 1] < 225) | (arr[..., 2] < 225)
    black = (arr[..., 0] < 100) & (arr[..., 1] < 100) & (arr[..., 2] < 100)

    red_cent = _centroid((arr[..., 0] == _RED[0]) & (arr[..., 1] == _RED[1]) & (arr[..., 2] == _RED[2]))
    blue_cent = _centroid((arr[..., 0] == _BLUE[0]) & (arr[..., 1] == _BLUE[1]) & (arr[..., 2] == _BLUE[2]))

    erode = Image.fromarray((black * 255).astype(np.uint8)).filter(ImageFilter.MinFilter(_ERODE_BOX))
    eroded = np.asarray(erode) > 127

    centers: list[tuple[float, float]] = []
    for comp in _components(eroded):
        c = _centroid(comp)
        if c is not None:
            centers.append(c)
    if red_cent is not None:
        centers.append(red_cent)
    if blue_cent is not None:
        centers.append(blue_cent)

    dedup: list[tuple[float, float]] = []
    for c in centers:
        if any(math.hypot(c[0] - k[0], c[1] - k[1]) < 8.0 for k in dedup):
            continue
        dedup.append(c)
    centers = dedup
    n = len(centers)

    nw = nonwhite
    adj: list[list[int]] = [[] for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            length = int(math.hypot(x2 - x1, y2 - y1)) + 1
            ink = 0
            tot = 0
            for t in range(length):
                tt = t / (length - 1) if length > 1 else 0.0
                px = int(round(x1 + (x2 - x1) * tt))
                py = int(round(y1 + (y2 - y1) * tt))
                if 0 <= px < w and 0 <= py < h:
                    tot += 1
                    if nw[py, px]:
                        ink += 1
            if tot and ink / tot > 0.9:
                adj[i].append(j)
                adj[j].append(i)

    art = _articulation(n, adj)

    red_idx = None
    blue_idx = None
    for i, c in enumerate(centers):
        if red_cent is not None and math.hypot(c[0] - red_cent[0], c[1] - red_cent[1]) < 8.0:
            red_idx = i
        if blue_cent is not None and math.hypot(c[0] - blue_cent[0], c[1] - blue_cent[1]) < 8.0:
            blue_idx = i

    red_cut = red_idx is not None and red_idx in art
    blue_cut = blue_idx is not None and blue_idx in art
    if red_cut and not blue_cut:
        return "red"
    return "blue"
