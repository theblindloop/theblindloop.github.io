"""Deterministic renderer for visible playing-card rank/pip agreement."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS_WIDTH = 512
CANVAS_HEIGHT = 768
SUPERSAMPLE = 4

BACKGROUNDS = (
    (232, 238, 244),
    (244, 238, 228),
    (237, 232, 244),
    (230, 241, 235),
)
HEART_PALETTE = (
    (210, 43, 70),
    (188, 35, 77),
    (224, 65, 72),
    (198, 48, 96),
)
CARD_FILL = (253, 251, 247)
CARD_OUTLINE = (126, 137, 147)
INK = (27, 32, 45)
DECOR_COLORS = ((168, 181, 194), (176, 165, 190), (158, 184, 173))

DIGIT_SEGMENTS = {
    3: ("a", "b", "c", "d", "g"),
    4: ("b", "c", "f", "g"),
    5: ("a", "c", "d", "f", "g"),
    6: ("a", "c", "d", "e", "f", "g"),
    7: ("a", "b", "c"),
}

# Seven possible large-pip centers. Every prefix layout deliberately retains an
# upper-right and a remote lower pip, making a one-sided crop insufficient.
PIP_SLOTS = (
    (365, 176),
    (188, 252),
    (365, 314),
    (238, 390),
    (366, 474),
    (184, 566),
    (382, 636),
)
PIP_LAYOUTS = {
    3: (0, 3, 6),
    4: (0, 1, 4, 6),
    5: (0, 1, 3, 4, 6),
    6: (0, 1, 2, 4, 5, 6),
    7: (0, 1, 2, 3, 4, 5, 6),
}


def sample_scene(seed: int) -> dict:
    """Create adjacent matched control/counterfactual pairs.

    Seeds 2k and 2k+1 share rank, translation, palette, background, and
    decoration. The even control has count == rank. The odd counterfactual uses
    a cyclically displaced count. Across complete cycles both rank and heart
    count have identical marginals in the yes and no classes.
    """

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0xC4A7D5)
    rank = 3 + (pair % 5)
    if seed % 2 == 0:
        count = rank
    else:
        displacement = 1 + ((pair // 5) % 4)
        count = 3 + ((rank - 3 + displacement) % 5)
    return {
        "background_code": (pair // 3) % len(BACKGROUNDS),
        "card_offset_x": (-6, 0, 6)[rng.randrange(3)],
        "card_offset_y": (-5, 0, 5)[rng.randrange(3)],
        "count": count,
        "decor_variant": rng.randrange(12),
        "heart_code": pair % len(HEART_PALETTE),
        "rank": rank,
    }


def _box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def _point(point: tuple[float, float], scale: int) -> tuple[int, int]:
    return round(point[0] * scale), round(point[1] * scale)


def _heart(draw: ImageDraw.ImageDraw, cx: float, cy: float, color: tuple[int, int, int], scale: int, size: float = 1.0) -> None:
    """Draw one connected filled heart from overlapping lobes and a point."""

    radius = 14.0 * size
    lobe_dx = 11.0 * size
    lobe_y = cy - 8.0 * size
    for lx in (cx - lobe_dx, cx + lobe_dx):
        draw.ellipse(_box((lx - radius, lobe_y - radius, lx + radius, lobe_y + radius), scale), fill=color)
    draw.polygon(
        [
            _point((cx - 25.0 * size, cy - 7.0 * size), scale),
            _point((cx + 25.0 * size, cy - 7.0 * size), scale),
            _point((cx, cy + 29.0 * size), scale),
        ],
        fill=color,
    )


def _segment_boxes(origin_x: float, origin_y: float) -> dict[str, tuple[float, float, float, float]]:
    return {
        "a": (origin_x + 10, origin_y, origin_x + 42, origin_y + 8),
        "b": (origin_x + 44, origin_y + 11, origin_x + 52, origin_y + 31),
        "c": (origin_x + 44, origin_y + 46, origin_x + 52, origin_y + 66),
        "d": (origin_x + 10, origin_y + 69, origin_x + 42, origin_y + 77),
        "e": (origin_x, origin_y + 46, origin_x + 8, origin_y + 66),
        "f": (origin_x, origin_y + 11, origin_x + 8, origin_y + 31),
        "g": (origin_x + 10, origin_y + 34, origin_x + 42, origin_y + 42),
    }


def pip_centers(scene: dict) -> tuple[tuple[int, int], ...]:
    dx = int(scene["card_offset_x"])
    dy = int(scene["card_offset_y"])
    indices = PIP_LAYOUTS[int(scene["count"])]
    return tuple((PIP_SLOTS[index][0] + dx, PIP_SLOTS[index][1] + dy) for index in indices)


def render(scene: dict) -> Image.Image:
    background = BACKGROUNDS[int(scene["background_code"]) % len(BACKGROUNDS)]
    heart = HEART_PALETTE[int(scene["heart_code"]) % len(HEART_PALETTE)]
    dx = int(scene["card_offset_x"])
    dy = int(scene["card_offset_y"])
    decor = int(scene["decor_variant"])
    rank = int(scene["rank"])
    count = int(scene["count"])
    if rank not in DIGIT_SEGMENTS or count not in PIP_LAYOUTS:
        raise ValueError("rank and count must lie in [3, 7]")

    ss = SUPERSAMPLE
    image = Image.new("RGB", (CANVAS_WIDTH * ss, CANVAS_HEIGHT * ss), background)
    draw = ImageDraw.Draw(image)

    # Remote low-chroma ornaments vary without entering either oracle mask.
    decor_color = DECOR_COLORS[decor % len(DECOR_COLORS)]
    anchors = ((24, 105), (480, 218), (28, 616), (475, 704))
    rotation = decor % len(anchors)
    for index in range(3):
        x, y = anchors[(index + rotation) % len(anchors)]
        radius = 7 + ((decor + index * 3) % 8)
        draw.ellipse(_box((x - radius, y - radius, x + radius, y + radius), ss), fill=decor_color)

    draw.rounded_rectangle(
        _box((52 + dx, 19 + dy, 460 + dx, 749 + dy), ss),
        radius=25 * ss,
        fill=CARD_FILL,
        outline=CARD_OUTLINE,
        width=4 * ss,
    )
    # A faint inner rule reinforces the playing-card object but stays irrelevant.
    draw.rounded_rectangle(
        _box((61 + dx, 28 + dy, 451 + dx, 740 + dy), ss),
        radius=20 * ss,
        outline=(218, 218, 212),
        width=2 * ss,
    )

    origin_x, origin_y = 83 + dx, 58 + dy
    boxes = _segment_boxes(origin_x, origin_y)
    for segment in DIGIT_SEGMENTS[rank]:
        draw.rounded_rectangle(_box(boxes[segment], ss), radius=2 * ss, fill=INK)

    # The explicitly excluded small corner suit mark makes the card convention
    # recognizable while remaining below the inverse's large-pip threshold.
    _heart(draw, 109 + dx, 164 + dy, heart, ss, size=0.34)
    for cx, cy in pip_centers(scene):
        _heart(draw, cx, cy, heart, ss)

    return image.resize((CANVAS_WIDTH, CANVAS_HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["rank"]) == int(scene["count"]) else "no"


def margin(scene: dict) -> float:
    difference = abs(int(scene["rank"]) - int(scene["count"]))
    return float(max(1, difference))


def is_quarantined(scene: dict) -> bool:
    return int(scene["rank"]) not in DIGIT_SEGMENTS or int(scene["count"]) not in PIP_LAYOUTS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["heart_code"] = int(twin["heart_code"]) + len(HEART_PALETTE)
    return [("one_full_heart_palette_cycle", twin)]
