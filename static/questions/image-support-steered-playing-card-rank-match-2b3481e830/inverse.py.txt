"""Independent pixel-only inverse for rank-to-heart agreement.

The inverse uses only raster topology. It finds dark rectangular components and
decodes ranks 3--7 from their horizontal/vertical segment pattern, then counts
large red heart-shaped connected components while ignoring the small corner
suit mark. No renderer, latent scene, filename, metadata, or fixed image
coordinate is consulted.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _geometry(points: np.ndarray) -> tuple[int, int, int, float, float, float]:
    ys, xs = points[:, 0], points[:, 1]
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    fill = len(points) / float(width * height)
    return width, height, len(points), fill, float(xs.mean()), float(ys.mean())


def _decode_rank(array: np.ndarray) -> int:
    dark = array.max(axis=2) < 92
    horizontal: list[tuple[int, int, int, float, float, float]] = []
    vertical: list[tuple[int, int, int, float, float, float]] = []
    for points in _components(dark):
        width, height, area, fill, cx, cy = _geometry(points)
        if area < 18:
            continue
        if not (115 <= area <= 360 and fill >= 0.72):
            raise OracleAbstention("dark ink contains a malformed numeral segment")
        if 28 <= width <= 37 and 7 <= height <= 11 and width >= 2.7 * height:
            horizontal.append((width, height, area, fill, cx, cy))
        elif 7 <= width <= 11 and 17 <= height <= 24 and height >= 1.7 * width:
            vertical.append((width, height, area, fill, cx, cy))
        else:
            raise OracleAbstention("dark component is not a supported numeral segment")

    counts = (len(horizontal), len(vertical))
    if counts == (1, 3):
        rank = 4
    elif counts == (3, 3):
        rank = 6
    elif counts == (1, 2):
        rank = 7
    elif counts == (3, 2):
        upper, lower = sorted(vertical, key=lambda item: item[5])
        horizontal_width = float(np.median([item[0] for item in horizontal]))
        x_difference = lower[4] - upper[4]
        if abs(x_difference) < 0.35 * horizontal_width:
            rank = 3
        elif x_difference > 0.9 * horizontal_width:
            rank = 5
        else:
            raise OracleAbstention("two vertical segments do not identify rank 3 or 5")
    else:
        raise OracleAbstention(f"unsupported numeral topology H{counts[0]} V{counts[1]}")

    all_segments = horizontal + vertical
    xs = [item[4] for item in all_segments]
    ys = [item[5] for item in all_segments]
    if max(xs) - min(xs) > 58 or max(ys) - min(ys) > 82:
        raise OracleAbstention("numeral segments do not form one compact corner glyph")
    if len(horizontal) == 3:
        hys = sorted(item[5] for item in horizontal)
        if min(hys[1] - hys[0], hys[2] - hys[1]) < 25:
            raise OracleAbstention("horizontal numeral rows are ambiguous")
    return rank


def _count_large_hearts(array: np.ndarray) -> int:
    signed = array.astype(np.int16)
    red = signed[:, :, 0]
    green = signed[:, :, 1]
    blue = signed[:, :, 2]
    mask = (red >= 145) & (red - green >= 72) & (red - blue >= 54)
    count = 0
    for points in _components(mask):
        width, height, area, fill, _cx, _cy = _geometry(points)
        if area < 35:
            continue
        if area < 430:
            # The small corner suit mark is intentionally ignored by the rule.
            continue
        if not (1050 <= area <= 2100):
            raise OracleAbstention(f"red component has ambiguous large-heart area {area}")
        if not (44 <= width <= 56 and 43 <= height <= 58):
            raise OracleAbstention("large red component has unsupported dimensions")
        if not (0.55 <= fill <= 0.82 and 0.82 <= width / height <= 1.20):
            raise OracleAbstention("large red component is not heart-like")
        count += 1
    if count not in (3, 4, 5, 6, 7):
        raise OracleAbstention(f"expected 3--7 large red hearts, found {count}")
    return count


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 600 or array.shape[1] < 400:
        raise OracleAbstention("raster is too small")
    rank = _decode_rank(array)
    count = _count_large_hearts(array)
    return "yes" if rank == count else "no"
