"""Deterministic renderer and analytic arm for the power-ring gap task."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIDES = ("top", "right", "bottom", "left")
SIDE_ANGLES = {"top": -90.0, "right": 0.0, "bottom": 90.0, "left": 180.0}
RING_COLORS = ("#1677B8", "#19866F", "#7952B3", "#D66A24")
MARKER_COLORS = ("#D84A4A", "#2A78C5", "#E0A526", "#3A9B67")
BAR_COLOR = "#252A34"
SUPER_SAMPLE = 4


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; consecutive seeds cycle through all four labels."""

    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C3)
    gap_index = seed % 4
    block = seed // 4
    narrow = block % 6 == 0
    return {
        "gap_side": SIDES[gap_index],
        "center_x": rng.randint(181, 203),
        "center_y": rng.randint(181, 203),
        "outer_radius": rng.randint(105, 124),
        "ring_width": rng.randint(21, 28),
        "gap_half_angle": round(
            rng.uniform(11.5, 14.5) if narrow else rng.uniform(24.0, 30.0), 4
        ),
        "ring_color": RING_COLORS[(gap_index + 3 * block) % len(RING_COLORS)],
        "stem_direction": SIDES[(gap_index + block + 1) % 4],
        "marker_phase": rng.randrange(4),
        "canvas_size": 384,
        "background": "#F7F4EC",
    }


def _point(cx: float, cy: float, radius: float, angle_degrees: float) -> tuple[float, float]:
    angle = math.radians(angle_degrees)
    return cx + radius * math.cos(angle), cy + radius * math.sin(angle)


def _ring_mask(scene: dict, scale: int) -> Image.Image:
    size = int(scene["canvas_size"]) * scale
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    outer = float(scene["outer_radius"]) * scale
    inner = (float(scene["outer_radius"]) - float(scene["ring_width"])) * scale
    half = float(scene["gap_half_angle"])
    center_angle = SIDE_ANGLES[str(scene["gap_side"])]

    mask = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(mask)
    draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=255)
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=0)

    reach = outer * 1.8
    p1 = _point(cx, cy, reach, center_angle - half)
    p2 = _point(cx, cy, reach, center_angle + half)
    draw.polygon([(cx, cy), p1, p2], fill=0)
    return mask


def _draw_center_bar(scene: dict, image: Image.Image, scale: int) -> None:
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    inner = (float(scene["outer_radius"]) - float(scene["ring_width"])) * scale
    angle = SIDE_ANGLES[str(scene["stem_direction"])]
    start = _point(cx, cy, 5.0 * scale, angle)
    end = _point(cx, cy, inner - 12.0 * scale, angle)
    width = 15 * scale
    draw.line([start, end], fill=BAR_COLOR, width=width)
    radius = width / 2.0
    draw.ellipse(
        (end[0] - radius, end[1] - radius, end[0] + radius, end[1] + radius),
        fill=BAR_COLOR,
    )


def _draw_corner_markers(scene: dict, image: Image.Image, scale: int) -> None:
    draw = ImageDraw.Draw(image)
    size = int(scene["canvas_size"])
    phase = int(scene["marker_phase"]) % 4
    centers = ((29, 29), (size - 29, 29), (size - 29, size - 29), (29, size - 29))
    radius = 7 * scale
    for index, (x, y) in enumerate(centers):
        cx, cy = x * scale, y * scale
        color = MARKER_COLORS[(index + phase) % 4]
        draw.polygon(
            [(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)],
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    """Render only fields in ``scene``; no answer-specific metadata is embedded."""

    scale = SUPER_SAMPLE
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size * scale, size * scale), str(scene["background"]))
    mask = _ring_mask(scene, scale)
    ring_layer = Image.new("RGB", image.size, str(scene["ring_color"]))
    image.paste(ring_layer, (0, 0), mask)
    _draw_center_bar(scene, image, scale)
    _draw_corner_markers(scene, image, scale)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["gap_side"])


def margin(scene: dict) -> float:
    """Ideal missing-ink area separating the chosen sector from an intact runner-up."""

    outer = float(scene["outer_radius"])
    inner = outer - float(scene["ring_width"])
    fraction = (2.0 * float(scene["gap_half_angle"])) / 360.0
    return math.pi * (outer * outer - inner * inner) * fraction


def is_quarantined(scene: dict) -> bool:
    return float(scene["gap_half_angle"]) < 18.0 or float(scene["ring_width"]) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the genuine period-four alias in the corner-palette phase."""

    twin = dict(scene)
    twin["marker_phase"] = int(scene["marker_phase"]) + 4
    return [("marker_phase_plus_four", twin)]
