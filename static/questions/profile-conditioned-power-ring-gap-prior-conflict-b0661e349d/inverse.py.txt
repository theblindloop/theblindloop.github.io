"""Independent inverse arm using only RGB pixels from the final PNG."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_SIDE_ANGLES = {"top": -90.0, "right": 0.0, "bottom": 90.0, "left": 180.0}


def _dominant_ring_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    colors, counts = np.unique(arr.reshape(-1, 3), axis=0, return_counts=True)
    candidates: list[tuple[int, np.ndarray]] = []
    for color, count in zip(colors, counts):
        chroma = int(color.max()) - int(color.min())
        if count >= 900 and chroma >= 45 and int(color.max()) <= 235:
            candidates.append((int(count), color))
    if not candidates:
        raise ValueError("no large saturated annular stroke is visible")
    _, ring_color = max(candidates, key=lambda item: item[0])
    distance = np.sqrt(((arr.astype(np.float64) - ring_color.astype(np.float64)) ** 2).sum(axis=2))
    mask = distance <= 24.0
    if int(mask.sum()) < 3500:
        raise ValueError("the candidate ring has too little visible ink")
    return mask


def _fit_center(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) > 16000:
        stride = max(1, len(xs) // 16000)
        xs, ys = xs[::stride], ys[::stride]
    x = xs.astype(np.float64)
    y = ys.astype(np.float64)
    design = np.column_stack((2.0 * x, 2.0 * y, np.ones_like(x)))
    target = x * x + y * y
    cx, cy, _ = np.linalg.lstsq(design, target, rcond=None)[0]
    if not (20.0 < cx < mask.shape[1] - 20.0 and 20.0 < cy < mask.shape[0] - 20.0):
        raise ValueError("fitted ring center lies outside the visible symbol")
    return float(cx), float(cy)


def measurement_from_image(image: Image.Image) -> dict:
    """Return the gap decision, four sector scores, and runner-up pixel margin."""

    mask = _dominant_ring_mask(image)
    cy_pixels, cx_pixels = np.nonzero(mask)
    cx, cy = _fit_center(mask)
    angles = np.degrees(np.arctan2(cy_pixels.astype(float) - cy, cx_pixels.astype(float) - cx))

    scores: dict[str, int] = {}
    sector_half_width = 35.0
    for side, target in _SIDE_ANGLES.items():
        delta = (angles - target + 180.0) % 360.0 - 180.0
        scores[side] = int(np.count_nonzero(np.abs(delta) <= sector_half_width))

    ranked = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    decision, best = ranked[0]
    runner_up = ranked[1][1]
    runner_up_margin = float(runner_up - best)
    if runner_up_margin < max(180.0, 0.12 * max(1, runner_up)):
        raise ValueError("no unique cardinal ring gap is recoverable")
    return {
        "decision": decision,
        "scores": scores,
        "runner_up_margin": runner_up_margin,
        "fitted_center": [round(cx, 3), round(cy, 3)],
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measurement_from_image(image)["decision"])
