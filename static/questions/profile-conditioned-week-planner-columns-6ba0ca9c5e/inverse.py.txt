"""Independent image-only inverse for the WEEK planner.

Only the final RGB raster is accepted.  The inverse finds chromatic connected
components in the public header band and counts components large enough to be
header tabs.  It does not import renderer, prompts, generator, verifier, or
analytic gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for 8-connected components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    ys, xs = np.nonzero(mask)
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        area = 0
        min_x = max_x = x0
        min_y = max_y = y0
        while stack:
            y, x = stack.pop()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        found.append((area, min_x, min_y, max_x + 1, max_y + 1))
    return found


def measure_header(image: Image.Image) -> tuple[int, float]:
    """Measure tab count and the smallest component separation margin."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # The public visual contract places colored tabs in y=100..150.  Neutral
    # lane fills, outlines, title, and row marks are excluded by chroma.
    crop = array[100:151]
    spread = crop.max(axis=2) - crop.min(axis=2)
    mask = (spread >= 45) & (crop.min(axis=2) <= 238)
    components = [item for item in _components(mask) if item[0] >= 120 and item[3] - item[1] >= 10 and item[4] - item[2] >= 10]
    components.sort(key=lambda item: item[1])
    if not components:
        raise ValueError("no separated colored WEEK header tabs found")
    separations = [components[i + 1][1] - components[i][3] for i in range(len(components) - 1)]
    separation = float(min(separations)) if separations else 999.0
    return len(components), separation


def decision_from_image(image: Image.Image) -> str:
    count, _ = measure_header(image)
    return str(count)
