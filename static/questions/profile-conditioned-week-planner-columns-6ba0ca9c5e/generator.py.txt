"""Deterministic renderer for the labeled WEEK planner candidate.

The answer unit is a compound visual correspondence: one colored tab in the
header band is aligned with one full-height day lane below it.  The renderer
uses only ordinary raster drawing; the independent inverse arm counts header
components from the PNG without importing this module.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 400
BACKGROUND = (248, 248, 246)
ACCENTS = (
    (42, 112, 204),
    (213, 82, 82),
    (28, 151, 145),
    (133, 79, 184),
    (226, 137, 39),
)
GRID = (107, 116, 126)
LANE_FILLS = ((255, 255, 255), (243, 246, 248), (250, 248, 242))
INK = (45, 52, 60)
ROW_INK = (170, 177, 184)


def _rng(seed: int, salt: int) -> random.Random:
    value = (int(seed) * 6364136223846793005 + salt * 1442695040888963407) & ((1 << 63) - 1)
    return random.Random(value)


def _font(size: int):
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    """Sample one JSON-serializable scene with count/style streams separated."""
    seed = int(seed)
    count = 4 + (seed % 6)
    layout_rng = _rng(seed, 11)
    palette_rng = _rng(seed, 23)
    detail_rng = _rng(seed, 37)
    position_rng = _rng(seed, 53)

    grid_width = float(layout_rng.randint(504, 548))
    grid_x = float(round((WIDTH - grid_width) / 2.0 + layout_rng.uniform(-5.0, 5.0), 2))
    grid_x = max(38.0, min(grid_x, WIDTH - grid_width - 38.0))
    body_top = float(layout_rng.randint(156, 166))
    body_bottom = float(layout_rng.randint(320, 338))
    tab_height = float(layout_rng.randint(28, 34))
    tight_layout = ((seed // 7) % 5) == 0
    pitch = grid_width / count
    if tight_layout:
        gap = layout_rng.uniform(4.5, 9.0)
        tab_width = max(21.0, pitch - gap)
    else:
        # Approximately constant accent ink across answer classes.
        target_tab_area = layout_rng.uniform(6900.0, 7900.0)
        tab_width = target_tab_area / (count * tab_height)
        tab_width = min(tab_width, pitch - 13.0)
        tab_width = max(tab_width, 23.0)
    tab_width = float(round(tab_width, 2))

    accent = list(ACCENTS[palette_rng.randrange(len(ACCENTS))])
    title_offset = float(round(position_rng.uniform(-18.0, 18.0), 2))
    appointment_budget = 12 + detail_rng.randrange(5)
    appointments_left = appointment_budget
    columns = []
    for index in range(count):
        center = grid_x + pitch * (index + 0.5)
        jitter = position_rng.uniform(-1.8, 1.8) if count > 1 else 0.0
        x = float(round(center + jitter, 2))
        lane_half = pitch * 0.455
        left = float(round(x - lane_half, 2))
        right = float(round(x + lane_half, 2))
        remaining = count - index
        if index == count - 1:
            appointment_count = appointments_left
        else:
            appointment_count = detail_rng.randrange(max(1, appointments_left // remaining + 2))
            appointments_left -= appointment_count
        appointments = []
        for slot in range(appointment_count):
            y = float(round(body_top + 25 + ((slot * 29 + detail_rng.randrange(17)) % max(30, int(body_bottom - body_top - 44))), 2))
            width = float(round(min(pitch * 0.64, 20 + detail_rng.randrange(14)), 2))
            appointments.append({"y": y, "width": width, "tone": int(detail_rng.randrange(3))})
        columns.append(
            {
                "x": x,
                "left": left,
                "right": right,
                "appointments": appointments,
                "fill_tone": int(detail_rng.randrange(len(LANE_FILLS))),
            }
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "column_count": count,
        "columns": columns,
        "accent": accent,
        "grid_x": grid_x,
        "grid_width": float(round(grid_width, 2)),
        "body_top": body_top,
        "body_bottom": body_bottom,
        "tab_width": tab_width,
        "tab_height": tab_height,
        "tight_layout": tight_layout,
        "title_offset": title_offset,
        "row_labels": [17, 31, 22, 28, 15],
        "ink_budget": appointment_budget,
    }


def render(scene: dict) -> Image.Image:
    """Render a scene; sorting makes column-record order a true latent alias."""
    ss = 3
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    def box(values):
        return [int(round(float(value) * ss)) for value in values]

    def rounded(values, radius, **kwargs):
        draw.rounded_rectangle(box(values), radius=int(round(radius * ss)), **kwargs)

    count = int(scene["column_count"])
    columns = sorted(scene["columns"], key=lambda item: float(item["x"]))[:count]
    accent = tuple(int(v) for v in scene["accent"])
    grid_x = float(scene["grid_x"])
    grid_width = float(scene["grid_width"])
    body_top = float(scene["body_top"])
    body_bottom = float(scene["body_bottom"])
    tab_width = float(scene["tab_width"])
    tab_height = float(scene["tab_height"])

    # Familiar context is deliberately small and never used as gold.
    draw.text(
        (int(round((width / 2 + float(scene["title_offset"])) * ss)), 51 * ss),
        "WEEK",
        fill=INK,
        anchor="mm",
        font=_font(31 * ss),
    )
    draw.line(box((grid_x, 91, grid_x + grid_width, 91)), fill=(205, 210, 214), width=2 * ss)

    # Row labels are neutral gray marks outside the answer-bearing structure.
    row_y = [177, 211, 245, 279, 313]
    for y, length in zip(row_y, scene["row_labels"], strict=True):
        draw.rounded_rectangle(box((17, y, 17 + float(length), y + 4)), radius=2 * ss, fill=ROW_INK)

    # The lanes come first; each record carries its own visible geometry.
    for column in columns:
        left, right = float(column["left"]), float(column["right"])
        rounded(
            (left, body_top, right, body_bottom),
            6,
            fill=LANE_FILLS[int(column["fill_tone"])],
            outline=GRID,
            width=2 * ss,
        )
        # Neutral appointment marks vary total ink independently of the count.
        for appointment in column["appointments"]:
            y = float(appointment["y"])
            w = float(appointment["width"])
            tone = ((210, 218, 226), (225, 205, 184), (204, 221, 209))[int(appointment["tone"])]
            rounded((float(column["x"]) - w / 2, y, float(column["x"]) + w / 2, y + 9), 3, fill=tone)

    # Separated colored tabs are the explicit visible labels for the lanes.
    tab_y = body_top - tab_height - 17.0
    for column in columns:
        x = float(column["x"])
        rounded((x - tab_width / 2, tab_y, x + tab_width / 2, tab_y + tab_height), 7, fill=accent)
        draw.line(
            box((x - tab_width * 0.22, tab_y + tab_height * 0.52, x + tab_width * 0.22, tab_y + tab_height * 0.52)),
            fill=(255, 255, 255),
            width=max(1, ss),
        )

    # One neutral baseline makes the lane correspondence legible.
    draw.line(box((grid_x, body_bottom + 8, grid_x + grid_width, body_bottom + 8)), fill=GRID, width=2 * ss)
    image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["column_count"]))


def margin(scene: dict) -> float:
    columns = sorted(scene["columns"], key=lambda item: float(item["x"]))
    tab_width = float(scene["tab_width"])
    if len(columns) < 2:
        return 999.0
    gaps = [float(b["x"]) - float(a["x"]) - tab_width for a, b in zip(columns, columns[1:])]
    return float(round(min(gaps), 6))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0 or float(scene["tab_width"]) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["columns"] = list(reversed(twin["columns"]))
    return [("permute_unordered_column_records", twin)]
