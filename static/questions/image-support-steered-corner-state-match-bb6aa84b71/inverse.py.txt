"""Independent pixel-only inverse arm for corner_state_match.

Receives only the final RGB image. It must never import the renderer, prompts,
generator, verifier, or any analytic-gold/scene code.

It locates the two state-tracking registers (one in the top-left, one in the
bottom-right), reads each register's start-disk colour (blue = OFF, orange =
ON), counts the red toggle disks attached to that register, replays each
register's tracked state, and finally BINDS the two distant registers by
comparing their final states. If either distant register is missing (e.g. after
evidence erasure of a corner), it abstains by raising.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = np.array([255, 255, 255])
OFF = np.array([40, 80, 220])
ON = np.array([235, 150, 30])
TOGGLE = np.array([210, 40, 50])

# Register bounding boxes (generous; the two corners are far apart).
BOX_A = (0, 0, 210, 210)          # x0,y0,x1,y1 top-left register
BOX_B = (302, 302, 512, 512)      # x0,y0,x1,y1 bottom-right register

MIN_COLOR_PIXELS = 120


def _near(pixel: np.ndarray, target: np.ndarray, tol: int = 80) -> bool:
    return bool(np.abs(pixel.astype(int) - target.astype(int)).sum() <= tol)


def _components(mask2d: np.ndarray) -> int:
    hs, ws = mask2d.shape
    visited = np.zeros_like(mask2d, dtype=bool)
    count = 0
    for py in range(hs):
        for px in range(ws):
            if not mask2d[py, px] or visited[py, px]:
                continue
            count += 1
            stack = [(py, px)]
            visited[py, px] = True
            while stack:
                cy, cx = stack.pop()
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < hs and 0 <= nx < ws and mask2d[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
    return count


def _register_state(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, int]:
    x0, y0, x1, y1 = box
    region = arr[y0:y1, x0:x1]
    blue = np.abs(region.astype(int) - OFF.astype(int)).sum(axis=2) <= 80
    orange = np.abs(region.astype(int) - ON.astype(int)).sum(axis=2) <= 80
    red = np.abs(region.astype(int) - TOGGLE.astype(int)).sum(axis=2) <= 80
    colored = int(blue.sum()) + int(orange.sum())
    if colored < MIN_COLOR_PIXELS:
        raise ValueError("abstain: a distant state register is unreadable")
    start = 1 if int(orange.sum()) > int(blue.sum()) else 0
    n_toggle = _components(red)
    final = (start + (n_toggle % 2)) % 2
    return final, colored


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    final_a, _ = _register_state(arr, BOX_A)
    final_b, _ = _register_state(arr, BOX_B)
    return "yes" if final_a == final_b else "no"
