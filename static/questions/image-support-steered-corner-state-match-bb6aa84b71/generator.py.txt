"""corner_state_match renderer.

Two disconnected state-tracking registers live at widely separated corners of
the canvas: one register in the TOP-LEFT and one register in the BOTTOM-RIGHT.
Each register independently tracks a binary state: it starts from its own
start-disk colour (blue = OFF, orange = ON) and flips that state once per red
toggle disk stacked beside the start disk. The two registers are NOT joined by
any path — they are two multipart decisive components that must be bound by a
comparison.

The question binds the two distant registers: "do the two tracked final states
match?" (both ON or both OFF). Removing either corner register makes the answer
unrecoverable, so the decisive support is distributed (wide span) and multipart
(two disconnected components).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512

# Palette
BG = (255, 255, 255)
OFF = (40, 80, 220)     # blue  -> OFF
ON = (235, 150, 30)     # orange-> ON
TOGGLE = (210, 40, 50)  # red

DISC_RADIUS = 14
PAD = 90  # start-disk centre distance from its corner

# Registers sit on two 8x8-grid diagonal corners: top-left and bottom-right.
AX, AY = PAD, PAD                     # register A centre
BX, BY = CANVAS - PAD, CANVAS - PAD   # register B centre


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start_a = rng.randint(0, 1)
    start_b = rng.randint(0, 1)
    n_a = rng.randint(0, 3)
    n_b = rng.randint(0, 3)
    toggles_a = [[round(k * 24.0 + 12.0, 1)] for k in range(n_a)]
    toggles_b = [[round(k * 24.0 + 12.0, 1)] for k in range(n_b)]
    return {
        "start_a": start_a,
        "toggles_a": toggles_a,
        "start_b": start_b,
        "toggles_b": toggles_b,
        "canvas": CANVAS,
    }


def _draw_register(draw: ImageDraw.ImageDraw, cx: int, cy: int,
                   start: int, toggles: list, down: bool) -> None:
    col = ON if start == 1 else OFF
    draw.ellipse([cx - DISC_RADIUS, cy - DISC_RADIUS,
                  cx + DISC_RADIUS, cy + DISC_RADIUS], fill=col)
    for k, (dx,) in enumerate(toggles):
        sign = 1 if down else -1
        ox = cx + int(dx)
        oy = cy + sign * (DISC_RADIUS + 16 + k * 24)
        r = 10
        draw.ellipse([ox - r, oy - r, ox + r, oy + r], fill=TOGGLE)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas"], scene["canvas"]), BG)
    draw = ImageDraw.Draw(img)
    _draw_register(draw, AX, AY, scene["start_a"], scene["toggles_a"], down=True)
    _draw_register(draw, BX, BY, scene["start_b"], scene["toggles_b"], down=False)
    return img


def _flip(start: int, toggles: list) -> int:
    return (start + (len(toggles) % 2)) % 2


def analytic_gold(scene: dict) -> str:
    final_a = _flip(scene["start_a"], scene["toggles_a"])
    final_b = _flip(scene["start_b"], scene["toggles_b"])
    return "yes" if final_a == final_b else "no"


def margin(scene: dict) -> float:
    return float(10 + len(scene["toggles_a"]) + len(scene["toggles_b"]))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    return []
