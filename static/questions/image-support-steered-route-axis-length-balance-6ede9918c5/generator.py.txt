"""Deterministic scene sampler and renderer for route-axis length balance."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
QUARANTINE_MARGIN = 50.0
ROUTE_PALETTE = (
    (0, 119, 128),
    (45, 91, 174),
    (151, 66, 126),
    (21, 126, 82),
)
BACKGROUND_PALETTE = (
    (250, 249, 245),
    (246, 249, 250),
    (250, 247, 251),
)
DIFFICULTIES = (4.00, 4.20, 4.40, 4.60, 4.75)


def _base_vertices(horizontal_major: bool, a: float, d: float) -> list[tuple[float, float]]:
    """Paired ten-run paths with one 10-by-2.5 envelope and matched total length."""
    if horizontal_major:
        c = a - d
        return [
            (0.0, 0.0),
            (a, 0.0),
            (a, 0.5),
            (c, 0.5),
            (c, 1.0),
            (c + 1.5, 1.0),
            (c + 1.5, 1.5),
            (c + 3.0, 1.5),
            (c + 3.0, 2.0),
            (10.0, 2.0),
            (10.0, 2.5),
        ]
    q = 5.0 - d
    return [
        (0.0, 0.0),
        (0.0, 2.5),
        (2.0, 2.5),
        (2.0, 0.0),
        (4.0, 0.0),
        (4.0, 2.5),
        (6.0, 2.5),
        (6.0, q),
        (8.0, q),
        (8.0, 2.5),
        (10.0, 2.5),
    ]


def _transform(
    points: Iterable[tuple[float, float]],
    *,
    transpose: bool,
    reflect_x: bool,
    reflect_y: bool,
) -> list[tuple[float, float]]:
    output = []
    for x, y in points:
        if reflect_x:
            x = 10.0 - x
        if reflect_y:
            y = 2.5 - y
        if transpose:
            x, y = y, x
        output.append((x, y))
    return output


def sample_scene(seed: int) -> dict:
    """Sample balanced pairs; odd seeds transpose the even-class construction."""
    seed = int(seed)
    pair_seed = seed // 2
    rng = random.Random(pair_seed ^ 0x5A17D3)
    scale = rng.randint(32, 35)
    stroke = rng.choice((9, 11, 13))
    difficulty = DIFFICULTIES[pair_seed % len(DIFFICULTIES)]
    a = rng.uniform(difficulty + 1.15, 6.15)
    shape_transposed = bool(rng.getrandbits(1))
    desired_horizontal = not bool(seed & 1)
    base_horizontal = desired_horizontal != shape_transposed
    points = _base_vertices(base_horizontal, a, difficulty)
    points = _transform(
        points,
        transpose=shape_transposed,
        reflect_x=bool(rng.getrandbits(1)),
        reflect_y=bool(rng.getrandbits(1)),
    )
    span_x = (2.5 if shape_transposed else 10) * scale
    span_y = (10 if shape_transposed else 2.5) * scale
    x0 = int(round((WIDTH - span_x) / 2.0)) + rng.randint(-16, 16)
    y0 = int(round((HEIGHT - span_y) / 2.0)) + rng.randint(-16, 16)
    vertices = [[int(round(x0 + scale * x)), int(round(y0 + scale * y))] for x, y in points]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "vertices": vertices,
        "route_rgb": list(rng.choice(ROUTE_PALETTE)),
        "background_rgb": list(rng.choice(BACKGROUND_PALETTE)),
        "stroke_width": stroke,
    }


def _axis_totals(scene: dict) -> tuple[float, float]:
    vertices = scene["vertices"]
    horizontal = 0.0
    vertical = 0.0
    for first, second in zip(vertices, vertices[1:]):
        dx = abs(float(second[0]) - float(first[0]))
        dy = abs(float(second[1]) - float(first[1]))
        if dx > 0 and dy > 0:
            raise ValueError("route segment is not orthogonal")
        horizontal += dx
        vertical += dy
    return horizontal, vertical


def analytic_gold(scene: dict) -> str:
    horizontal, vertical = _axis_totals(scene)
    if horizontal == vertical:
        raise ValueError("route has an exact axis-length tie")
    return "horizontal" if horizontal > vertical else "vertical"


def margin(scene: dict) -> float:
    horizontal, vertical = _axis_totals(scene)
    return abs(horizontal - vertical)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def _segment_box(first: list[int], second: list[int], half: int) -> tuple[int, int, int, int]:
    x1, y1 = first
    x2, y2 = second
    return (
        min(x1, x2) - half,
        min(y1, y2) - half,
        max(x1, x2) + half,
        max(y1, y2) + half,
    )


def render(scene: dict) -> Image.Image:
    """Render the route as the union of six antialiased square-capped rectangles."""
    width = int(scene["width"])
    height = int(scene["height"])
    vertices = scene["vertices"]
    route_rgb = tuple(int(value) for value in scene["route_rgb"])
    background_rgb = tuple(int(value) for value in scene["background_rgb"])
    stroke_width = int(scene["stroke_width"])
    supersample = 4
    image = Image.new("RGB", (width * supersample, height * supersample), background_rgb)
    draw = ImageDraw.Draw(image)
    half = stroke_width * supersample // 2
    scaled = [[int(x) * supersample, int(y) * supersample] for x, y in vertices]
    for first, second in zip(scaled, scaled[1:]):
        draw.rectangle(_segment_box(first, second, half), fill=route_rgb)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["vertices"] = list(reversed([list(point) for point in scene["vertices"]]))
    return [("reverse_route_storage", twin)]
