"""Independent pixel-only inverse for the route-axis length comparison."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MIN_COMPONENT_PIXELS = 2500
MIN_LONG_RUN = 16
MIN_BBOX_SPAN = 70
MAX_BBOX_SPAN = 390
MIN_PIXEL_MARGIN = 8.0


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0)
    delta = array.astype(np.float64) - background
    return np.sqrt(np.sum(delta * delta, axis=2)) > 48.0


def _component_sizes(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    sizes: list[int] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _long_ink_per_scanline(mask: np.ndarray) -> np.ndarray:
    values = np.zeros(mask.shape[0], dtype=np.float64)
    for index, row in enumerate(mask):
        padded = np.pad(row.astype(np.int8), (1, 1))
        changes = np.diff(padded)
        starts = np.flatnonzero(changes == 1)
        ends = np.flatnonzero(changes == -1)
        lengths = ends - starts
        values[index] = float(lengths[lengths >= MIN_LONG_RUN].sum())
    return values


def _band_total(mask: np.ndarray) -> tuple[float, int]:
    values = _long_ink_per_scanline(mask)
    occupied = values > 0
    padded = np.pad(occupied.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    bands = []
    for start, end in zip(starts, ends):
        core = values[start:end]
        if len(core):
            bands.append(float(np.median(core)))
    return sum(bands), len(bands)


def _measure_axis_totals(image: Image.Image) -> tuple[float, float]:
    mask = _foreground(image)
    sizes = _component_sizes(mask)
    if len(sizes) != 1 or sizes[0] < MIN_COMPONENT_PIXELS:
        raise ValueError("the colored route is missing or severed")
    points = np.argwhere(mask)
    height = int(points[:, 0].max() - points[:, 0].min() + 1)
    width = int(points[:, 1].max() - points[:, 1].min() + 1)
    if not (MIN_BBOX_SPAN <= width <= MAX_BBOX_SPAN and MIN_BBOX_SPAN <= height <= MAX_BBOX_SPAN):
        raise ValueError("the route extent is outside the supported regional frame")
    horizontal, horizontal_bands = _band_total(mask)
    vertical, vertical_bands = _band_total(mask.T)
    if not (2 <= horizontal_bands <= 5 and 2 <= vertical_bands <= 5):
        raise ValueError("the image does not contain the expected ten-run orthogonal route")
    return horizontal, vertical


def decision_from_image(image: Image.Image) -> str:
    """Return the larger recovered centerline-axis total using final RGB pixels only."""
    horizontal, vertical = _measure_axis_totals(image)
    difference = horizontal - vertical
    if abs(difference) <= MIN_PIXEL_MARGIN:
        raise ValueError("pixel measurement is too close to the equality boundary")
    return "horizontal" if difference > 0 else "vertical"
