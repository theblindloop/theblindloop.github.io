"""Independent image-only inverse for normalized slider-rail correspondence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((213, 75, 75), dtype=np.int16),
    "blue": np.array((57, 118, 197), dtype=np.int16),
    "gold": np.array((214, 161, 29), dtype=np.int16),
    "teal": np.array((0, 140, 145), dtype=np.int16),
    "orange": np.array((225, 123, 50), dtype=np.int16),
    "green": np.array((75, 155, 88), dtype=np.int16),
    "purple": np.array((132, 85, 184), dtype=np.int16),
    "dark": np.array((38, 50, 56), dtype=np.int16),
}
BORDERS = ("red", "blue", "gold")
SLIDERS = ("teal", "orange", "green", "purple")


def _mask(arr: np.ndarray, name: str, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - PALETTE[name]), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = sumx = sumy = 0
        minx = maxx = sx
        miny = maxy = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            sumx += x
            sumy += y
            minx, maxx = min(minx, x), max(maxx, x)
            miny, maxy = min(miny, y), max(maxy, y)
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append((area, sumx / area, sumy / area, minx, miny, maxx, maxy))
    return output


def _inside(point: tuple[float, float], bounds: tuple[int, int, int, int]) -> bool:
    x, y = point
    minx, miny, maxx, maxy = bounds
    return minx < x < maxx and miny < y < maxy


def _signature(
    endpoints: tuple[tuple[float, float], tuple[float, float]],
    sliders: dict[str, list[tuple[float, float]]],
) -> dict[str, float]:
    left, right = sorted(endpoints, key=lambda point: point[0])
    length = right[0] - left[0]
    if length < 150.0 or abs(right[1] - left[1]) > 3.0:
        raise ValueError("invalid visible rail endpoints")
    signature = {}
    for color, points in sliders.items():
        candidates = [
            point
            for point in points
            if left[0] < point[0] < right[0] and abs(point[1] - left[1]) <= 4.0
        ]
        if len(candidates) != 1:
            raise ValueError(f"rail does not contain exactly one {color} slider")
        signature[color] = (candidates[0][0] - left[0]) / length
    return signature


def _error(left: dict[str, float], right: dict[str, float]) -> float:
    return sum(abs(left[color] - right[color]) for color in SLIDERS)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 600 or arr.shape[1] < 900:
        raise ValueError("unexpected raster")
    comps = {name: _components(_mask(arr, name)) for name in PALETTE}

    endpoint_points = [
        (item[1], item[2]) for item in comps["dark"] if 620 <= item[0] <= 820
    ]
    if len(endpoint_points) != 8:
        raise ValueError("expected eight visible end stops")
    endpoint_points.sort(key=lambda point: (round(point[1] / 20), point[0]))
    top = sorted(endpoint_points[:2])
    bottom = sorted(endpoint_points[2:], key=lambda point: point[0])
    rails = [(top[0], top[1])] + [tuple(bottom[i : i + 2]) for i in range(0, 6, 2)]

    slider_points = {}
    for color in SLIDERS:
        points = [(item[1], item[2]) for item in comps[color] if 260 <= item[0] <= 430]
        if len(points) != 4:
            raise ValueError(f"expected four {color} sliders")
        slider_points[color] = points
    signatures = [_signature(pair, slider_points) for pair in rails]

    border_bounds = {}
    for border in BORDERS:
        candidates = [item for item in comps[border] if item[0] >= 2500]
        if len(candidates) != 1:
            raise ValueError(f"expected one connected {border} option border")
        item = candidates[0]
        border_bounds[border] = (item[3], item[4], item[5], item[6])

    reference_index = min(range(4), key=lambda i: sum(point[1] for point in rails[i]))
    reference = signatures[reference_index]
    scored = []
    for index, pair in enumerate(rails):
        if index == reference_index:
            continue
        midpoint = ((pair[0][0] + pair[1][0]) / 2.0, (pair[0][1] + pair[1][1]) / 2.0)
        owners = [name for name, bounds in border_bounds.items() if _inside(midpoint, bounds)]
        if len(owners) != 1:
            raise ValueError("option rail is not visibly bound to one border")
        scored.append((_error(reference, signatures[index]), owners[0]))
    scored.sort()
    if len(scored) != 3 or scored[0][0] > 0.035 or scored[1][0] - scored[0][0] < 0.12:
        raise ValueError("pixel rail signatures do not yield a unique match")
    return scored[0][1]
