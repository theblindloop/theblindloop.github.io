"""Deterministic latent sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 700
BG = "#f7f4ed"
CARD = "#fffefd"
DARK = "#263238"
GRAY = "#879296"
PALE = "#d7dcdd"
BORDER_RGB = {"red": "#d54b4b", "blue": "#3976c5", "gold": "#d6a11d"}
SLIDER_RGB = {
    "teal": "#008c91",
    "orange": "#e17b32",
    "green": "#4b9b58",
    "purple": "#8455b8",
}
BORDERS = ("red", "blue", "gold")
SLIDERS = ("teal", "orange", "green", "purple")
OPTION_XS = (180, 500, 820)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _signature_error(left: dict, right: dict) -> float:
    return sum(abs(float(left[color]) - float(right[color])) for color in SLIDERS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Every rail has the same four occupied fractions, counterbalancing total
    # ink and local position.  Only the binding between color and fraction varies.
    position_sets = (
        (0.16, 0.37, 0.61, 0.84),
        (0.18, 0.41, 0.64, 0.82),
        (0.15, 0.39, 0.66, 0.83),
        (0.17, 0.35, 0.63, 0.81),
    )
    positions = list(position_sets[(seed // 3) % len(position_sets)])
    rng.shuffle(positions)
    reference_signature = dict(zip(SLIDERS, positions))

    swaps = [(a, b) for a in range(4) for b in range(a + 1, 4)]
    rng.shuffle(swaps)
    distractors = []
    for a, b in swaps[:2]:
        altered = dict(reference_signature)
        ca, cb = SLIDERS[a], SLIDERS[b]
        altered[ca], altered[cb] = altered[cb], altered[ca]
        distractors.append(altered)

    answer_border = BORDERS[seed % len(BORDERS)]
    signatures = [reference_signature, *distractors]
    rng.shuffle(signatures)
    # Bind the exact signature to the balanced answer border, independently of
    # its horizontal card position.
    correct_index = next(i for i, signature in enumerate(signatures) if signature == reference_signature)
    borders = list(BORDERS)
    current = borders.index(answer_border)
    borders[current], borders[correct_index] = borders[correct_index], borders[current]

    reference = {
        "x": 500,
        "y": 205,
        "length": rng.choice((460, 500, 540)),
        "fractions": reference_signature,
    }
    # Even the smallest leftmost fraction stays visibly clear of an end stop.
    option_lengths = [200, 215, 230]
    rng.shuffle(option_lengths)
    options = [
        {
            "x": x,
            "y": 520,
            "length": length,
            "border": border,
            "fractions": signature,
        }
        for x, length, border, signature in zip(OPTION_XS, option_lengths, borders, signatures)
    ]
    scene = {"reference": reference, "options": options}
    if is_quarantined(scene):
        raise RuntimeError("sampler produced an ambiguous rail scene")
    return scene


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    options = scene["options"]
    scores = sorted(
        (_signature_error(reference["fractions"], option["fractions"]), option["border"])
        for option in options
    )
    if len(scores) != 3 or scores[0][0] != 0.0 or scores[1][0] <= 0.0:
        raise ValueError("scene lacks one exact normalized-coordinate match")
    return scores[0][1]


def margin(scene: dict) -> float:
    try:
        reference = scene["reference"]
        errors = sorted(
            _signature_error(reference["fractions"], option["fractions"])
            for option in scene["options"]
        )
        return round((errors[1] - errors[0]) * 100.0, 6)
    except (KeyError, TypeError, ValueError, IndexError):
        return 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        reference = scene["reference"]
        options = scene["options"]
        if len(options) != 3 or {item["border"] for item in options} != set(BORDERS):
            return True
        records = [reference, *options]
        for record in records:
            fractions = record["fractions"]
            if set(fractions) != set(SLIDERS):
                return True
            values = sorted(float(value) for value in fractions.values())
            if values[0] < 0.12 or values[-1] > 0.88:
                return True
            if min(b - a for a, b in zip(values, values[1:])) < 0.14:
                return True
            if int(record["length"]) < 170:
                return True
        analytic_gold(scene)
        return margin(scene) < 15.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["options"].reverse()
    return [("option_record_order_alias", twin)]


def _draw_rail(draw: ImageDraw.ImageDraw, record: dict) -> None:
    cx, cy = int(record["x"]), int(record["y"])
    length = int(record["length"])
    left, right = cx - length // 2, cx + length // 2
    draw.line((left, cy, right, cy), fill=GRAY, width=8)
    draw.ellipse((left - 15, cy - 15, left + 15, cy + 15), fill=DARK)
    draw.ellipse((right - 15, cy - 15, right + 15, cy + 15), fill=DARK)
    for color in SLIDERS:
        x = round(left + float(record["fractions"][color]) * length)
        draw.ellipse((x - 12, cy - 12, x + 12, cy + 12), fill=SLIDER_RGB[color], outline=DARK, width=2)


def render(scene: dict) -> Image.Image:
    reference = scene["reference"]
    options = scene["options"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((500, 22), "REFERENCE", fill=DARK, font=_font(25, True), anchor="ma")
    draw.rounded_rectangle((105, 70, 895, 335), radius=18, fill=CARD, outline=PALE, width=4)
    _draw_rail(draw, reference)
    draw.text((500, 374), "OPTIONS", fill=DARK, font=_font(25, True), anchor="ma")
    for option in options:
        cx = int(option["x"])
        draw.rounded_rectangle(
            (cx - 140, 415, cx + 140, 635),
            radius=18,
            fill=CARD,
            outline=BORDER_RGB[option["border"]],
            width=7,
        )
        _draw_rail(draw, option)
    return image
