"""Independent pixel-only inverse for the bilateral fold-overlay world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("anchor", "fork", "loop", "bolt")
PAD = 14
GUTTER = 8
OUT_W = 132
OUT_H = 252
SCALES = (0.92, 1.0, 1.08)
OFFSETS = (-10, 0, 10)


def _mask(symbol: str, dx: int, dy: int, scale: float) -> np.ndarray:
    image = Image.new("L", (OUT_W, OUT_H), 0)
    draw = ImageDraw.Draw(image)
    cx = OUT_W / 2 + dx
    cy = OUT_H / 2 + dy
    sx = 52.0 * scale
    sy = 88.0 * scale

    def p(x: float, y: float) -> tuple[int, int]:
        return (round(x), round(y))

    if symbol == "anchor":
        lw = max(8, round(12 * scale))
        draw.line([p(cx, cy - 0.72 * sy), p(cx, cy + 0.40 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.52 * sx, cy - 0.24 * sy), p(cx + 0.52 * sx, cy - 0.24 * sy)], fill=255, width=lw)
        draw.arc([round(cx - 0.62 * sx), round(cy - 0.02 * sy), round(cx + 0.62 * sx), round(cy + 0.78 * sy)], 0, 180, fill=255, width=lw)
        draw.line([p(cx - 0.62 * sx, cy + 0.38 * sy), p(cx - 0.48 * sx, cy + 0.24 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.62 * sx, cy + 0.38 * sy), p(cx + 0.48 * sx, cy + 0.24 * sy)], fill=255, width=lw)
    elif symbol == "fork":
        lw = max(8, round(13 * scale))
        draw.line([p(cx, cy + 0.72 * sy), p(cx, cy - 0.02 * sy)], fill=255, width=lw)
        draw.line([p(cx, cy - 0.02 * sy), p(cx - 0.70 * sx, cy - 0.67 * sy)], fill=255, width=lw)
        draw.line([p(cx, cy - 0.02 * sy), p(cx + 0.70 * sx, cy - 0.67 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.70 * sx, cy - 0.67 * sy), p(cx - 0.53 * sx, cy - 0.49 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.70 * sx, cy - 0.67 * sy), p(cx + 0.53 * sx, cy - 0.49 * sy)], fill=255, width=lw)
    elif symbol == "loop":
        lw = max(8, round(11 * scale))
        draw.ellipse([round(cx - 0.58 * sx), round(cy - 0.64 * sy), round(cx + 0.58 * sx), round(cy + 0.28 * sy)], outline=255, width=lw)
        draw.line([p(cx, cy + 0.20 * sy), p(cx, cy + 0.77 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.30 * sx, cy + 0.72 * sy), p(cx + 0.30 * sx, cy + 0.72 * sy)], fill=255, width=lw)
    elif symbol == "bolt":
        lw = max(8, round(12 * scale))
        pts = [p(cx - 0.52 * sx, cy - 0.76 * sy), p(cx + 0.30 * sx, cy - 0.76 * sy), p(cx - 0.06 * sx, cy - 0.08 * sy), p(cx + 0.55 * sx, cy - 0.08 * sy), p(cx - 0.40 * sx, cy + 0.79 * sy), p(cx - 0.15 * sx, cy + 0.17 * sy), p(cx - 0.58 * sx, cy + 0.17 * sy)]
        draw.line(pts, fill=255, width=lw, joint="curve")
    else:
        raise ValueError(symbol)
    return np.asarray(image, dtype=np.float32) / 255.0


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[tuple[float, float, int]] = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 30:
            found.append((float(np.mean([p[1] for p in points])), float(np.mean([p[0] for p in points])), len(points)))
    return found


def _fold(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = rgb.max(axis=2).astype(int) - rgb.min(axis=2).astype(int)
    colored = (spread > 55) & (rgb.max(axis=2) > 100) & (rgb.min(axis=2) < 220)
    marks = _components(colored)
    if len(marks) < 4:
        raise ValueError("four colored corner fiducials were not found")
    marks = sorted(marks, key=lambda item: item[2], reverse=True)[:4]
    min_x = min(item[0] for item in marks)
    max_x = max(item[0] for item in marks)
    min_y = min(item[1] for item in marks)
    max_y = max(item[1] for item in marks)
    # Corner squares at the image edge are clipped by one pixel.  Their
    # measured centroids are therefore half-pixel shifted relative to the
    # nominal marker centers used by the renderer.
    tile_x = int(round(min_x - 6.5))
    tile_y = int(round(min_y - 6.5))
    tile_w = int(round(max_x - min_x + 14))
    tile_h = int(round(max_y - min_y + 14))
    out_w = (tile_w - 2 * PAD - GUTTER) // 2
    out_h = tile_h - 2 * PAD
    if out_w != OUT_W or out_h != OUT_H:
        raise ValueError(f"unexpected marked rectangle dimensions {(tile_w, tile_h)}")
    gray = np.asarray(image.convert("L"), dtype=np.float32)
    y0 = tile_y + PAD
    left_x = tile_x + PAD
    right_x = left_x + out_w + GUTTER
    left = gray[y0 : y0 + out_h, left_x : left_x + out_w]
    right = gray[y0 : y0 + out_h, right_x : right_x + out_w]
    if left.shape != (OUT_H, OUT_W) or right.shape != (OUT_H, OUT_W):
        raise ValueError("fold panels were cropped outside the image")
    return (left + right[:, ::-1]) / 2.0


def _score_templates(folded: np.ndarray) -> dict[str, float]:
    signal = np.median(folded) - folded
    vector = signal.reshape(-1).astype(np.float32)
    vector -= vector.mean()
    vector_norm = float(np.linalg.norm(vector)) + 1e-6
    scores: dict[str, float] = {}
    for symbol in SYMBOLS:
        centered, norms = _TEMPLATE_BANK[symbol]
        scores[symbol] = float(np.max((centered @ vector) / (norms * vector_norm)))
    return scores


def _build_template_bank() -> dict[str, tuple[np.ndarray, np.ndarray]]:
    bank: dict[str, tuple[np.ndarray, np.ndarray]] = {}
    for symbol in SYMBOLS:
        templates = []
        for scale in SCALES:
            for dx in OFFSETS:
                for dy in (-14, 0, 14):
                    templates.append(_mask(symbol, dx, dy, scale).reshape(-1))
        values = np.stack(templates).astype(np.float32)
        centered = values - values.mean(axis=1, keepdims=True)
        norms = np.linalg.norm(centered, axis=1) + 1e-6
        bank[symbol] = (centered, norms)
    return bank


_TEMPLATE_BANK = _build_template_bank()


def decision_from_image(image: Image.Image) -> str:
    """Return the broad symbol recovered from pixels only."""

    scores = _score_templates(_fold(image))
    return max(SYMBOLS, key=lambda symbol: scores[symbol])
