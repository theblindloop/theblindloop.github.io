"""Renderer and analytic answer for the bilateral fold-overlay world.

The gray tile contains two equal panels.  The right panel stores the same broad
signal in reverse order and an opposite fine carrier.  Reflecting that panel
across the marked gutter and averaging the pair cancels the carrier.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "fold_overlay_symbol-0.1.0"
SYMBOLS = ("anchor", "fork", "loop", "bolt")
WIDTH, HEIGHT = 512, 384
TILE_W, TILE_H = 300, 280
PAD = 14
GUTTER = 8
OUT_W = 132
OUT_H = 252
PALETTES = (
    (18, 158, 181),
    (222, 64, 119),
    (75, 95, 211),
    (226, 132, 25),
)


def _symbol_mask(
    symbol: str,
    h: int = OUT_H,
    w: int = OUT_W,
    dx: int = 0,
    dy: int = 0,
    scale: float = 1.0,
) -> np.ndarray:
    """Make a broad binary glyph; this helper is deliberately self-contained."""

    image = Image.new("L", (w, h), 0)
    draw = ImageDraw.Draw(image)
    cx = w / 2 + dx
    cy = h / 2 + dy
    sx = 52.0 * scale
    sy = 88.0 * scale

    def point(x: float, y: float) -> tuple[int, int]:
        return (round(x), round(y))

    if symbol == "anchor":
        lw = max(8, round(12 * scale))
        draw.line([point(cx, cy - 0.72 * sy), point(cx, cy + 0.40 * sy)], fill=255, width=lw)
        draw.line([point(cx - 0.52 * sx, cy - 0.24 * sy), point(cx + 0.52 * sx, cy - 0.24 * sy)], fill=255, width=lw)
        draw.arc(
            [round(cx - 0.62 * sx), round(cy - 0.02 * sy), round(cx + 0.62 * sx), round(cy + 0.78 * sy)],
            0,
            180,
            fill=255,
            width=lw,
        )
        draw.line([point(cx - 0.62 * sx, cy + 0.38 * sy), point(cx - 0.48 * sx, cy + 0.24 * sy)], fill=255, width=lw)
        draw.line([point(cx + 0.62 * sx, cy + 0.38 * sy), point(cx + 0.48 * sx, cy + 0.24 * sy)], fill=255, width=lw)
    elif symbol == "fork":
        lw = max(8, round(13 * scale))
        draw.line([point(cx, cy + 0.72 * sy), point(cx, cy - 0.02 * sy)], fill=255, width=lw)
        draw.line([point(cx, cy - 0.02 * sy), point(cx - 0.70 * sx, cy - 0.67 * sy)], fill=255, width=lw)
        draw.line([point(cx, cy - 0.02 * sy), point(cx + 0.70 * sx, cy - 0.67 * sy)], fill=255, width=lw)
        draw.line([point(cx - 0.70 * sx, cy - 0.67 * sy), point(cx - 0.53 * sx, cy - 0.49 * sy)], fill=255, width=lw)
        draw.line([point(cx + 0.70 * sx, cy - 0.67 * sy), point(cx + 0.53 * sx, cy - 0.49 * sy)], fill=255, width=lw)
    elif symbol == "loop":
        lw = max(8, round(11 * scale))
        box = [round(cx - 0.58 * sx), round(cy - 0.64 * sy), round(cx + 0.58 * sx), round(cy + 0.28 * sy)]
        draw.ellipse(box, outline=255, width=lw)
        draw.line([point(cx, cy + 0.20 * sy), point(cx, cy + 0.77 * sy)], fill=255, width=lw)
        draw.line([point(cx - 0.30 * sx, cy + 0.72 * sy), point(cx + 0.30 * sx, cy + 0.72 * sy)], fill=255, width=lw)
    elif symbol == "bolt":
        lw = max(8, round(12 * scale))
        pts = [
            point(cx - 0.52 * sx, cy - 0.76 * sy),
            point(cx + 0.30 * sx, cy - 0.76 * sy),
            point(cx - 0.06 * sx, cy - 0.08 * sy),
            point(cx + 0.55 * sx, cy - 0.08 * sy),
            point(cx - 0.40 * sx, cy + 0.79 * sy),
            point(cx - 0.15 * sx, cy + 0.17 * sy),
            point(cx - 0.58 * sx, cy + 0.17 * sy),
        ]
        draw.line(pts, fill=255, width=lw, joint="curve")
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(image, dtype=np.float32) / 255.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    weak = seed % 5 == 0
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "tile_x": rng.randint(48, 158),
        "tile_y": rng.randint(36, 82),
        "carrier_phase": rng.randrange(8),
        "carrier_amplitude": round(rng.uniform(72.0, 88.0) if weak else rng.uniform(52.0, 72.0), 3),
        "signal_strength": round(rng.uniform(25.0, 38.0) if weak else rng.uniform(50.0, 68.0), 3),
        "base_level": rng.randint(150, 205),
        "symbol_dx": rng.choice((-10, 0, 10)),
        "symbol_dy": rng.choice((-14, 0, 14)),
        "symbol_scale": rng.choice((0.92, 1.0, 1.08)),
        "marker_palette_id": rng.randrange(len(PALETTES)),
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "seed": int(seed),
    }


def _carrier(phase: int, amplitude: float) -> np.ndarray:
    yy, xx = np.indices((OUT_H, OUT_W))
    blocks = (yy // 4 + (xx + int(phase)) // 4) % 2
    return np.where(blocks == 0, amplitude, -amplitude).astype(np.float32)


def render(scene: dict) -> Image.Image:
    tile = np.full((TILE_H, TILE_W), float(scene["base_level"]), dtype=np.float32)
    mask = _symbol_mask(
        str(scene["symbol"]),
        dx=int(scene["symbol_dx"]),
        dy=int(scene["symbol_dy"]),
        scale=float(scene["symbol_scale"]),
    )
    carrier = _carrier(int(scene["carrier_phase"]), float(scene["carrier_amplitude"]))
    strength = float(scene["signal_strength"])
    base = float(scene["base_level"])
    left = np.clip(base - strength * mask + carrier, 0, 255)
    right = np.clip(base - strength * mask[:, ::-1] - carrier[:, ::-1], 0, 255)
    y0, x0 = PAD, PAD
    tile[y0 : y0 + OUT_H, x0 : x0 + OUT_W] = left
    right_x = x0 + OUT_W + GUTTER
    tile[y0 : y0 + OUT_H, right_x : right_x + OUT_W] = right
    tile_rgb = Image.fromarray(np.rint(tile).astype(np.uint8), mode="L").convert("RGB")
    draw = ImageDraw.Draw(tile_rgb)
    draw.rectangle([0, 0, TILE_W - 1, TILE_H - 1], outline=(34, 34, 34), width=3)
    fold_x = PAD + OUT_W + GUTTER // 2
    for yy in range(PAD + 3, PAD + OUT_H - 2, 14):
        draw.line([fold_x, yy, fold_x, min(yy + 7, PAD + OUT_H - 1)], fill=(86, 86, 86), width=2)
    color = PALETTES[int(scene["marker_palette_id"]) % len(PALETTES)]
    for cx, cy in (
        (6, 6),
        (TILE_W - 7, 6),
        (6, TILE_H - 7),
        (TILE_W - 7, TILE_H - 7),
    ):
        draw.rectangle([cx - 7, cy - 7, cx + 7, cy + 7], fill=color)
    canvas = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (255, 255, 255))
    canvas.paste(tile_rgb, (int(scene["tile_x"]), int(scene["tile_y"])))
    return canvas


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return round(float(scene["signal_strength"]) - 0.70 * float(scene["carrier_amplitude"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 8
    return [("carrier_period_alias", twin)]
