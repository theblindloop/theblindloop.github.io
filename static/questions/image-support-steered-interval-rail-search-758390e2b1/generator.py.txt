"""Deterministic latent sampler and renderer for interval_rail_search."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
BACKGROUND = (248, 246, 239)
LABELS = ("A", "B", "C", "D")
PALETTES = (
    {"rail": [24, 139, 160], "ink": [35, 42, 48], "guide": [218, 214, 202]},
    {"rail": [38, 128, 176], "ink": [38, 42, 48], "guide": [218, 214, 202]},
    {"rail": [24, 145, 128], "ink": [35, 42, 48], "guide": [218, 214, 202]},
)


def _scores(scene: dict) -> list[int]:
    scores = []
    for centers in scene["section_bump_centers"]:
        gaps = [centers[i + 1] - centers[i] for i in range(3)]
        scores.append(max(gaps) - min(gaps))
    return scores


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    x0 = rng.randint(96, 108)
    section_width = rng.choice((106, 108, 110))
    width = 4 * section_width
    x1 = x0 + width
    y = rng.randint(222, 238)
    base_gap = rng.choice((21, 22, 23))
    delta = rng.choice((4, 5, 6))

    target = [base_gap, base_gap, base_gap]
    distractors = []
    for multiplier in (1, 1, 1):
        gaps = [base_gap - delta, base_gap, base_gap + delta]
        rng.shuffle(gaps)
        distractors.append(gaps)
    rng.shuffle(distractors)
    target_index = int(seed) % 4
    patterns = []
    remaining = iter(distractors)
    for index in range(4):
        patterns.append(target if index == target_index else next(remaining))
    centers = []
    for section, gaps in enumerate(patterns):
        first = x0 + section * section_width + (section_width - 3 * base_gap) // 2
        row = [first]
        for gap in gaps:
            row.append(row[-1] + gap)
        centers.append(row)

    return {
        "section_bump_centers": centers,
        "rail_bounds": [x0, x1, y],
        "palette": deepcopy(rng.choice(PALETTES)),
    }


def analytic_gold(scene: dict) -> str:
    centers = scene["section_bump_centers"]
    scores = []
    for row in centers:
        gaps = [row[i + 1] - row[i] for i in range(3)]
        scores.append(max(gaps) - min(gaps))
    winner = min(range(4), key=scores.__getitem__)
    if scores.count(scores[winner]) != 1:
        raise ValueError("scene has no unique uniform-spacing stretch")
    return LABELS[winner]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    x0, x1, y = [int(v) for v in scene["rail_bounds"]]
    centers = [[int(x) for x in row] for row in scene["section_bump_centers"]]
    rail = tuple(scene["palette"]["rail"])
    ink = tuple(scene["palette"]["ink"])
    guide = tuple(scene["palette"]["guide"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((x0, y - 92), "ONE CONTINUOUS RAIL", fill=ink, font=font)
    draw.line((x0, y - 72, x1, y - 72), fill=guide, width=1)
    draw.line((x0, y, x1, y), fill=rail, width=9)

    section_width = (x1 - x0) // 4
    for index, (label, bumps) in enumerate(zip(LABELS, centers)):
        left = x0 + index * section_width
        if index:
            draw.line((left, y + 12, left, y + 25), fill=ink, width=1)
        draw.text((left + section_width // 2 - 3, y + 32), label, fill=ink, font=font)
        for x in bumps:
            draw.rectangle((x - 4, y - 22, x + 4, y - 4), fill=rail)

    draw.text((x0, y + 68), "COMPARE THE THREE GAPS IN EACH STRETCH", fill=ink, font=font)
    return image
