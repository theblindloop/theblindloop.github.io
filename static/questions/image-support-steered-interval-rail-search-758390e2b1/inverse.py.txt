"""Independent pixel-only inverse arm for interval_rail_search.

The implementation receives only the final raster.  It segments the saturated
rail, validates that the rail is one connected component, enumerates four equal
stretches and their attached square bumps, then compares the three recovered
gaps in every stretch. It contains no renderer or latent-state dependency.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _rail_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    saturation = high - low
    mask = (saturation > 55) & (high > 90) & (low < 110)
    if int(mask.sum()) < 4000:
        raise ValueError("rail is missing or too small")
    return mask


def _component_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    unseen = set(map(tuple, np.argwhere(mask)))
    count = 0
    while unseen:
        count += 1
        if count > 1:
            return count
        start = unseen.pop()
        queue = deque([start])
        while queue:
            y, x = queue.popleft()
            for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                py, px = point
                if 0 <= py < height and 0 <= px < width and point in unseen:
                    unseen.remove(point)
                    queue.append(point)
    return count


def _runs(indices: np.ndarray) -> list[tuple[int, int]]:
    if not len(indices):
        return []
    runs = []
    start = previous = int(indices[0])
    for value in map(int, indices[1:]):
        if value != previous + 1:
            runs.append((start, previous))
            start = value
        previous = value
    runs.append((start, previous))
    return runs


def decision_from_image(image: Image.Image) -> str:
    mask = _rail_mask(image)
    if _component_count(mask) != 1:
        raise ValueError("rail must be one unbroken connected component")

    ys, xs = np.nonzero(mask)
    x_min, x_max = int(xs.min()), int(xs.max())
    row_ink = mask.sum(axis=1)
    baseline_y = int(np.argmax(row_ink))
    if int(row_ink[baseline_y]) < 380:
        raise ValueError("continuous rail baseline is incomplete")

    upper = mask[max(0, baseline_y - 24) : max(0, baseline_y - 6), :].sum(axis=0)
    bump_runs = [(a, b) for a, b in _runs(np.flatnonzero(upper >= 10)) if b - a >= 6]
    if len(bump_runs) != 16:
        raise ValueError(f"expected sixteen recoverable bumps, found {len(bump_runs)}")
    bump_centers = [(a + b) / 2.0 for a, b in bump_runs]

    section_width = (x_max - x_min + 1) / 4.0
    scores = []
    for section in range(4):
        left = x_min + section * section_width
        right = x_min + (section + 1) * section_width
        centers = [
            x
            for x in bump_centers
            if left <= x and (x < right or (section == 3 and x <= right))
        ]
        if len(centers) != 4:
            raise ValueError(f"stretch has {len(centers)} recoverable bumps")
        gaps = [centers[i + 1] - centers[i] for i in range(3)]
        scores.append(max(gaps) - min(gaps))

    order = sorted(range(4), key=scores.__getitem__)
    if scores[order[1]] - scores[order[0]] < 6.0:
        raise ValueError("spacing winner has inadequate runner-up margin")
    return LABELS[order[0]]
