"""Two-hop attribute relay world.

Three columns of four tiles each. The left column binds a black shape to a
colored bar and to a dot count; the middle column binds a black shape to a dot
count; the right column binds a dot count to a colored bar. One left tile is
ringed in black.

The label is obtained by relaying an identity through two *different* matching
attributes:

    marked left tile -> (shape) -> middle tile -> (dot count) -> right tile
    answer = that right tile's bar color

Every column is independently permuted, so no row index survives the relay, and
the marked tile carries its own bar color and its own dot count as decoys whose
answers are constructed to differ from the gold.

Latent state z is fully rendered: every field of the scene dict draws ink.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "two_hop_attribute_relay-0.1.0"

# ---------------------------------------------------------------- palette ---
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GRAY = (150, 150, 150)

COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "blue": (30, 90, 220),
    "green": (25, 150, 60),
    "orange": (240, 140, 20),
    "purple": (150, 45, 190),
}
COLOR_NAMES = ("blue", "green", "orange", "purple")
GLYPHS = ("square", "up_tri", "down_tri", "plus")

# ----------------------------------------------------------------- layout ---
TILE_W = 200
TILE_H = 224
COL_GAP = 52
ROW_GAP = 52
MARGIN_X = 60
MARGIN_Y = 46
N_ROWS = 4
N_COLS = 3

CANVAS_W = 2 * MARGIN_X + N_COLS * TILE_W + (N_COLS - 1) * COL_GAP
CANVAS_H = 2 * MARGIN_Y + N_ROWS * TILE_H + (N_ROWS - 1) * ROW_GAP

OUTLINE_W = 3
SWATCH_W = 84
SWATCH_H = 50
SWATCH_TOP = 16          # relative to tile origin; bottom row = 65
SWATCH_BOTTOM = SWATCH_TOP + SWATCH_H - 1

PIP_D = 20
PIP_TOP = 186            # relative; bottom row = 205
PIP_BOTTOM = PIP_TOP + PIP_D - 1

RING_GAP = 12
RING_W = 8

QUARANTINE_MARGIN = 11.0

_JITTER = 6
_GLYPH_SIZES = tuple(range(78, 93, 2))     # 78..92, even
_GLYPH_CY = tuple(range(124, 130))         # 124..129
_PIP_GAPS = tuple(range(10, 27))           # 10..26


# ------------------------------------------------------------- sampling ----
def _tile_origin(scene: dict, col: int, row: int) -> tuple[int, int]:
    k = col * N_ROWS + row
    x = MARGIN_X + col * (TILE_W + COL_GAP) + scene["tile_dx"][k]
    y = MARGIN_Y + row * (TILE_H + ROW_GAP) + scene["tile_dy"][k]
    return x, y


def _glyph_box(scene: dict, col: int, row: int) -> tuple[int, int, int, int]:
    """Absolute inclusive bbox of the black glyph in tile (col, row)."""
    k = col * N_ROWS + row          # col is 0 (left) or 1 (middle)
    size = scene["glyph_size"][k]
    cy = scene["glyph_cy"][k]
    x, y = _tile_origin(scene, col, row)
    gx0 = x + (TILE_W - size) // 2
    gy0 = y + cy - size // 2
    return gx0, gy0, gx0 + size - 1, gy0 + size - 1


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(4000):
        left_glyphs = list(GLYPHS)
        rng.shuffle(left_glyphs)
        mid_glyphs = list(GLYPHS)
        rng.shuffle(mid_glyphs)
        left_pips = [1, 2, 3, 4]
        rng.shuffle(left_pips)
        mid_pips = [1, 2, 3, 4]
        rng.shuffle(mid_pips)
        right_pips = [1, 2, 3, 4]
        rng.shuffle(right_pips)
        left_colors = list(COLOR_NAMES)
        rng.shuffle(left_colors)
        right_colors = list(COLOR_NAMES)
        rng.shuffle(right_colors)
        marked_row = rng.randrange(N_ROWS)

        mid_idx = mid_glyphs.index(left_glyphs[marked_row])
        right_idx = right_pips.index(mid_pips[mid_idx])
        gold = right_colors[right_idx]

        # Every positional shortcut must fail.
        if mid_idx == marked_row or right_idx == mid_idx or right_idx == marked_row:
            continue
        # The marked tile's own bar color must not be the answer.
        if gold == left_colors[marked_row]:
            continue
        # Skipping the middle column via the marked tile's own dot count must fail.
        if gold == right_colors[right_pips.index(left_pips[marked_row])]:
            continue

        scene = {
            "left_colors": left_colors,
            "left_glyphs": left_glyphs,
            "left_pips": left_pips,
            "mid_glyphs": mid_glyphs,
            "mid_pips": mid_pips,
            "right_pips": right_pips,
            "right_colors": right_colors,
            "marked_row": marked_row,
            "tile_dx": [rng.randint(-_JITTER, _JITTER) for _ in range(N_COLS * N_ROWS)],
            "tile_dy": [rng.randint(-_JITTER, _JITTER) for _ in range(N_COLS * N_ROWS)],
            "glyph_size": [rng.choice(_GLYPH_SIZES) for _ in range(2 * N_ROWS)],
            "glyph_cy": [rng.choice(_GLYPH_CY) for _ in range(2 * N_ROWS)],
            "pip_gap": rng.choice(_PIP_GAPS),
        }
        return scene
    raise RuntimeError("failed to sample a scene satisfying the decoy constraints")


# -------------------------------------------------------------- analytic ---
def _relay(scene: dict) -> tuple[int, int]:
    mid_idx = scene["mid_glyphs"].index(scene["left_glyphs"][scene["marked_row"]])
    right_idx = scene["right_pips"].index(scene["mid_pips"][mid_idx])
    return mid_idx, right_idx


def analytic_gold(scene: dict) -> str:
    _, right_idx = _relay(scene)
    return scene["right_colors"][right_idx]


def margin(scene: dict) -> float:
    vals: list[float] = []

    # Vertical clearances inside every glyph-bearing tile.
    for col in (0, 1):
        for row in range(N_ROWS):
            k = col * N_ROWS + row
            size = scene["glyph_size"][k]
            cy = scene["glyph_cy"][k]
            gy0 = cy - size // 2
            gy1 = gy0 + size - 1
            if col == 0:                       # left tiles carry a bar above
                vals.append(float(gy0 - SWATCH_BOTTOM - 1))
            else:
                vals.append(float(gy0 - OUTLINE_W))
            vals.append(float(PIP_TOP - gy1 - 1))
            vals.append(float(size // 3))      # thinnest decisive glyph feature

    # Bar-to-outline and pip-to-outline clearances (layout constants).
    vals.append(float(SWATCH_TOP - OUTLINE_W))
    vals.append(float(TILE_H - OUTLINE_W - PIP_BOTTOM - 1))
    vals.append(float(PIP_D))
    vals.append(float(scene["pip_gap"]))

    # Tile-to-tile clearances after jitter, ring included where it is drawn.
    boxes = {}
    for col in range(N_COLS):
        for row in range(N_ROWS):
            x, y = _tile_origin(scene, col, row)
            boxes[(col, row)] = (x, y, x + TILE_W - 1, y + TILE_H - 1)
    ring_row = scene["marked_row"]
    for (col, row), box in boxes.items():
        grown = box
        if col == 0 and row == ring_row:
            pad = RING_GAP + RING_W
            grown = (box[0] - pad, box[1] - pad, box[2] + pad, box[3] + pad)
        for (ocol, orow), other in boxes.items():
            if (ocol, orow) <= (col, row):
                continue
            dx = max(grown[0] - other[2], other[0] - grown[2]) - 1
            dy = max(grown[1] - other[3], other[1] - grown[3]) - 1
            vals.append(float(max(dx, dy)))

    return float(min(vals))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias exists: every scene field draws ink, so R is injective."""
    return []


# ---------------------------------------------------------------- render ---
def _draw_glyph(draw: ImageDraw.ImageDraw, kind: str, box: tuple[int, int, int, int]) -> None:
    x0, y0, x1, y1 = box
    cx = (x0 + x1) // 2
    if kind == "square":
        draw.rectangle([x0, y0, x1, y1], fill=BLACK)
    elif kind == "up_tri":
        draw.polygon([(cx, y0), (x1, y1), (x0, y1)], fill=BLACK)
    elif kind == "down_tri":
        draw.polygon([(x0, y0), (x1, y0), (cx, y1)], fill=BLACK)
    elif kind == "plus":
        arm = (x1 - x0 + 1) // 3
        half = arm // 2
        cyy = (y0 + y1) // 2
        draw.rectangle([x0, cyy - half, x1, cyy - half + arm - 1], fill=BLACK)
        draw.rectangle([cx - half, y0, cx - half + arm - 1, y1], fill=BLACK)
    else:
        raise ValueError(f"unknown glyph {kind!r}")


def _draw_pips(draw: ImageDraw.ImageDraw, x: int, y: int, count: int, gap: int) -> None:
    total = count * PIP_D + (count - 1) * gap
    sx = x + (TILE_W - total) // 2
    for i in range(count):
        px = sx + i * (PIP_D + gap)
        py = y + PIP_TOP
        draw.ellipse([px, py, px + PIP_D - 1, py + PIP_D - 1], fill=BLACK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), WHITE)
    draw = ImageDraw.Draw(image)

    for col in range(N_COLS):
        for row in range(N_ROWS):
            x, y = _tile_origin(scene, col, row)
            draw.rectangle(
                [x, y, x + TILE_W - 1, y + TILE_H - 1], outline=GRAY, width=OUTLINE_W
            )
            if col in (0, 2):
                name = (scene["left_colors"] if col == 0 else scene["right_colors"])[row]
                sx = x + (TILE_W - SWATCH_W) // 2
                sy = y + SWATCH_TOP
                draw.rectangle(
                    [sx, sy, sx + SWATCH_W - 1, sy + SWATCH_H - 1], fill=COLOR_RGB[name]
                )
            if col in (0, 1):
                kind = (scene["left_glyphs"] if col == 0 else scene["mid_glyphs"])[row]
                _draw_glyph(draw, kind, _glyph_box(scene, col, row))
            pips = (
                scene["left_pips"] if col == 0
                else scene["mid_pips"] if col == 1
                else scene["right_pips"]
            )[row]
            _draw_pips(draw, x, y, pips, scene["pip_gap"])

    x, y = _tile_origin(scene, 0, scene["marked_row"])
    pad = RING_GAP + RING_W
    draw.rectangle(
        [x - pad, y - pad, x + TILE_W - 1 + pad, y + TILE_H - 1 + pad],
        outline=BLACK,
        width=RING_W,
    )
    return image
