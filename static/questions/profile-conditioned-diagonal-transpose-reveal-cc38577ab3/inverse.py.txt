"""Independent pixel-only inverse arm for diagonal_transpose_reveal.

Receives ONLY the final PNG. It re-discovers the revealed glyph by:
  1. reading the mean luminance of every cell of the dense lattice;
  2. subtracting the transpose (main-diagonal reflection) of that cell matrix
     (the declared transpose difference), which cancels the transposed noise;
  3. correlating the surviving anti-symmetric difference against the two
     declared diagonal-band templates A and B (transpose images of each other);
  4. reporting the template with the larger signed correlation.

No scene data, no analytic gold, no renderer imports.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "diagonal_transpose_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_OFF = 12
BAND_HALF = 4


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((c - r) - BAND_OFF) <= BAND_HALF
    mask_b = np.transpose(mask_a)
    return {"A": mask_a, "B": mask_b}


def _cell_means(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    n = NGRID
    mat = np.zeros((n, n), dtype=np.float64)
    for i in range(n):
        for j in range(n):
            block = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
            mat[i, j] = block.mean()
    return mat


def _diff_signal(image: Image.Image) -> np.ndarray:
    mat = _cell_means(image)
    return mat - np.transpose(mat)


def _correlate(sig: np.ndarray, mask: np.ndarray) -> float:
    return float(np.sum(sig * mask))


def decision_from_image(image: Image.Image) -> str:
    sig = _diff_signal(image)
    masks = _band_masks()
    sa = _correlate(sig, masks["A"])
    sb = _correlate(sig, masks["B"])
    return "A" if sa >= sb else "B"


def reveal_margin(image: Image.Image) -> float:
    sig = _diff_signal(image)
    masks = _band_masks()
    sa = _correlate(sig, masks["A"])
    sb = _correlate(sig, masks["B"])
    return abs(abs(sa) - abs(sb))
