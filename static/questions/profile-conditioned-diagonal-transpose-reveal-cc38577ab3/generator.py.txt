"""diagonal_transpose_reveal: main-diagonal transpose difference glyph reveal world.

Declared transform: subtract the transpose (reflection across the main
diagonal, i.e. swapping rows and columns) of the field from the field itself.
The raw image is a flat field of noise that is exactly symmetric under
transposition (M = M^T) so the transposed noise cancels perfectly; only the
symbol, written as an anti-symmetric luminance bias around the main-diagonal
reflection (bright inside the answer band and dark inside its transpose-image
band), survives the difference. The surviving field is strongly positive inside
one diagonal band and negative inside the opposite diagonal band, giving a clean
single diagonal band (A) versus the opposite diagonal band (B).

Latent scene z = {noise, glyph, cell}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "diagonal_transpose_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # lattice cells per side
CELL = W // NGRID   # 16
BASE = 128.0
BIAS = 40.0
NOISE_SIGMA = 15.0
BAND_OFF = 12       # |c - r - off| within this is A; B = A transposed
BAND_HALF = 4       # half-width of each diagonal band in grid cells
Q_TOL = 0.15        # quarantine threshold on the A/B template-overlap margin


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A = one diagonal band, B = its transpose.

    B is exactly the transpose (main-diagonal reflection) of A, so under the
    transpose difference the two bands separate cleanly.
    """
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((c - r) - BAND_OFF) <= BAND_HALF
    mask_b = np.transpose(mask_a)
    return {"A": mask_a, "B": mask_b}


def build_scene(noise: int, glyph: str, cell: int = CELL) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "cell": int(cell),
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(noise=int(seed), glyph=glyph)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(NGRID, NGRID) mean-luminance of each lattice cell."""
    glyph = scene["glyph"]
    rng = np.random.default_rng(int(scene["noise"]) & 0xFFFFFFFF)
    n = NGRID
    t = rng.normal(0.0, NOISE_SIGMA, size=(n, n))
    symmetric = t + np.transpose(t)
    g = _glyph_mask(glyph)
    antisym = g.astype(np.float64) - np.transpose(g).astype(np.float64)
    val = BASE + symmetric + BIAS * antisym
    return val


def render(scene: dict) -> Image.Image:
    cell = int(scene.get("cell", CELL))
    n = W // cell
    mat = render_matrix(scene)[:n, :n]
    arr = np.repeat(np.repeat(mat, cell, axis=0), cell, axis=1)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the per-cell noise seed, the answer
    glyph, the cell pitch) directly changes the rendered pixels, so no distinct
    latent vector reproduces the same image. Declared empty."""
    return iter(())
