"""Deterministic latent sampler and renderer for Midline Ribbon Reveal."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = (512, 320)
QUARANTINE_GAP = 0.14
SHAPES = ("arch", "cup", "wave")
OUTLINES = ((28, 42, 57), (55, 38, 72), (25, 59, 64))
BACKGROUNDS = ((248, 246, 240), (243, 248, 246), (248, 244, 249))
RED = (211, 52, 61)
GOLD = (232, 169, 39)


def _normalized_scores(scene: dict) -> dict[str, float]:
    c, s = (float(v) for v in scene["shape_coefficients"])
    norm = math.hypot(c, s)
    if norm <= 1e-9:
        return {name: 0.0 for name in SHAPES}
    return {"arch": c / norm, "cup": -c / norm, "wave": s / norm}


def analytic_gold(scene: dict) -> str:
    scores = _normalized_scores(scene)
    return max(SHAPES, key=lambda name: (scores[name], -SHAPES.index(name)))


def margin(scene: dict) -> float:
    values = sorted(_normalized_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A7D1)
    label_index = int(seed) % len(SHAPES)
    label = SHAPES[label_index]
    near = int(seed) % 8 == 0
    if label == "arch":
        coefficients = [0.53, 0.47] if near else [1.0, rng.uniform(-0.12, 0.12)]
    elif label == "cup":
        coefficients = [-0.53, 0.47] if near else [-1.0, rng.uniform(-0.12, 0.12)]
    else:
        side = -1.0 if (int(seed) // 3) % 2 else 1.0
        coefficients = [side * 0.47, 0.53] if near else [rng.uniform(-0.12, 0.12), 1.0]

    terms = [
        [rng.uniform(5.0, 8.0), 3, rng.uniform(0.0, 2.0 * math.pi)],
        [rng.uniform(2.0, 4.0), 5, rng.uniform(0.0, 2.0 * math.pi)],
    ]
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "x0": rng.randint(48, 58),
        "x1": rng.randint(454, 464),
        "center_y": rng.uniform(154.0, 166.0),
        "tilt": rng.uniform(-8.0, 8.0),
        "amplitude": rng.uniform(18.0, 24.0),
        "shape_coefficients": coefficients,
        "width_base": rng.uniform(42.0, 50.0),
        "width_terms": terms,
        "stroke_width": rng.choice([4, 5]),
        "outline_rgb": list(rng.choice(OUTLINES)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
    }


def _center(scene: dict, t: float) -> float:
    c, s = (float(v) for v in scene["shape_coefficients"])
    phase_path = c * math.cos(2.0 * math.pi * t) + s * math.sin(2.0 * math.pi * t)
    return (
        float(scene["center_y"])
        + float(scene["tilt"]) * (t - 0.5)
        + float(scene["amplitude"]) * phase_path
    )


def _ribbon_width(scene: dict, t: float) -> float:
    total = float(scene["width_base"])
    canonical_terms: Iterable[list[float]] = sorted(
        scene["width_terms"], key=lambda term: (int(term[1]), float(term[0]), float(term[2]))
    )
    for amplitude, frequency, phase in canonical_terms:
        total += float(amplitude) * math.sin(2.0 * math.pi * int(frequency) * t + float(phase))
    return total


def render(scene: dict) -> Image.Image:
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background_rgb"])
    outline = tuple(int(v) for v in scene["outline_rgb"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    x0, x1 = int(scene["x0"]), int(scene["x1"])
    top: list[tuple[int, int]] = []
    bottom: list[tuple[int, int]] = []
    for x in range(x0, x1 + 1):
        t = (x - x0) / (x1 - x0)
        mid = _center(scene, t)
        half = _ribbon_width(scene, t) / 2.0
        top.append((x * scale, round((mid - half) * scale)))
        bottom.append((x * scale, round((mid + half) * scale)))
    stroke = int(scene["stroke_width"]) * scale
    draw.line(top, fill=outline, width=stroke, joint="curve")
    draw.line(bottom, fill=outline, width=stroke, joint="curve")
    draw.line([top[0], bottom[0]], fill=outline, width=stroke)
    draw.line([top[-1], bottom[-1]], fill=outline, width=stroke)

    left_mid = _center(scene, 0.0)
    right_mid = _center(scene, 1.0)
    ring_x, ring_y, ring_r = (x0 - 18) * scale, round(left_mid * scale), 8 * scale
    draw.ellipse(
        [ring_x - ring_r, ring_y - ring_r, ring_x + ring_r, ring_y + ring_r],
        outline=RED,
        width=4 * scale,
    )
    draw.line([(ring_x + ring_r, ring_y), (x0 * scale, round(left_mid * scale))], fill=RED, width=3 * scale)
    tri_x, tri_y = (x1 + 18) * scale, round(right_mid * scale)
    triangle = [
        (tri_x + 9 * scale, tri_y),
        (tri_x - 7 * scale, tri_y - 9 * scale),
        (tri_x - 7 * scale, tri_y + 9 * scale),
    ]
    draw.polygon(triangle, fill=GOLD)
    draw.line([(x1 * scale, round(right_mid * scale)), (tri_x - 7 * scale, tri_y)], fill=GOLD, width=3 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["width_terms"] = list(reversed(twin["width_terms"]))
    return [("permute_commutative_width_harmonics", twin)]
