"""recolour_rule_replay: sample, render and analytically solve one scene.

A drawn board of five coloured tiles is the starting state; exactly one tile
carries a heavy black ring.  Below the board a left-to-right ordered row of
rule cards is drawn.  Each card holds two colour swatches: the left swatch is
the colour a rule looks for, the right swatch is the colour it writes in its
place.  The cards act on the whole board one at a time in drawn left-to-right
order; a card whose left colour is not the colour a tile currently has leaves
that tile alone.  The board after the cards have acted is never drawn.

The answer is the colour the ringed tile ends with (and, for the second prompt
family, whether it ends different from the colour it starts with).

Every quantity that decides the answer is drawn: the ringed tile's starting
colour, the ordered pair of colours on each card, and the drawn left-to-right
order of the cards.  The decision boundary is the hue boundary between the four
named colours, and ``margin`` measures the smallest drawn clearance to it.
"""

from __future__ import annotations

import colorsys
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "recolour_rule_replay-renderer-0.1.0"

# ---------------------------------------------------------------- palette
# The four public colour names, as hue/saturation/value anchors.  The hue
# boundaries below are the classification boundaries the pixel arm uses too;
# they are properties of the colour names, not of this renderer.
COLOURS = ("red", "amber", "green", "blue")
_ANCHOR = {
    "red": (2.0, 0.76, 0.81),
    "amber": (41.0, 0.83, 0.89),
    "green": (133.0, 0.68, 0.57),
    "blue": (225.0, 0.76, 0.77),
}
BOUNDARIES = (21.5, 87.0, 179.0, 293.0)
MARGIN_CAP = 20.0
QUARANTINE_BELOW = 6.0

# ---------------------------------------------------------------- layout
CANVAS = (880, 470)
BG = (250, 247, 240)
RING = (25, 25, 25)
CARD_BG = (255, 255, 255)
CARD_EDGE = (170, 170, 170)
ARROW = (105, 105, 105)

TILE = 72
BOARD_GAP = 26
BOARD_Y = 52
N_TILES = 5

CARD_W = 132
CARD_H = 92
CARD_GAP = 26
CARD_Y = 250
SW = 42
SW_DX1 = 16
SW_DX2 = 74
SW_DY = 25


def _hue(rgb) -> float:
    r, g, b = (v / 255.0 for v in rgb)
    h, _, _ = colorsys.rgb_to_hsv(r, g, b)
    return h * 360.0


def _rgb(h: float, s: float, v: float):
    r, g, b = colorsys.hsv_to_rgb((h % 360.0) / 360.0, s, v)
    return (int(round(r * 255)), int(round(g * 255)), int(round(b * 255)))


def _circ(a: float, b: float) -> float:
    d = abs((a - b) % 360.0)
    return min(d, 360.0 - d)


def classify(rgb) -> str:
    """Name the colour of a drawn square from its hue alone."""
    h = _hue(rgb) % 360.0
    if h < BOUNDARIES[0] or h >= BOUNDARIES[3]:
        return "red"
    if h < BOUNDARIES[1]:
        return "amber"
    if h < BOUNDARIES[2]:
        return "green"
    return "blue"


def clearance(rgb) -> float:
    """Angular clearance of a drawn square to the nearest hue boundary."""
    h = _hue(rgb)
    return min(_circ(h, b) for b in BOUNDARIES)


def _make(rng: random.Random, name: str, push: float | None) -> list[int]:
    """Draw one square colour for ``name``; ``push`` targets a clearance."""
    h0, s0, v0 = _ANCHOR[name]
    s = min(0.95, max(0.55, s0 + rng.uniform(-0.07, 0.07)))
    v = min(0.95, max(0.50, v0 + rng.uniform(-0.07, 0.07)))
    if push is None:
        h = h0 + rng.uniform(-7.0, 7.0)
    else:
        near = min(BOUNDARIES, key=lambda b: _circ(h0, b))
        sign = 1.0 if ((near - h0) % 360.0) < 180.0 else -1.0
        h = near - sign * push
    rgb = _rgb(h, s, v)
    return [int(rgb[0]), int(rgb[1]), int(rgb[2])]


# ---------------------------------------------------------------- sampling
def _replay(start: str, rules) -> str:
    cur = start
    for rule in sorted(rules, key=lambda r: r["step"]):
        if cur == rule["src"]:
            cur = rule["dst"]
    return cur


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x9E37 ^ (int(seed) * 7919 + 13))
    mode = "normal"
    roll = rng.random()
    if roll < 0.20:
        mode = "tight"
    elif roll < 0.44:
        mode = "near"

    want_colour = COLOURS[rng.randrange(len(COLOURS))]
    want_changed = rng.random() < 0.5

    for _ in range(4000):
        n_cards = rng.choice((3, 4, 5))
        rules = []
        for step in range(n_cards):
            src = COLOURS[rng.randrange(4)]
            dst = COLOURS[rng.randrange(4)]
            while dst == src:
                dst = COLOURS[rng.randrange(4)]
            rules.append({"step": step, "src": src, "dst": dst})
        start = COLOURS[rng.randrange(4)]
        final = _replay(start, rules)
        if final != want_colour:
            continue
        if ((final != start) != want_changed):
            continue
        # shortcut block: the last card's right swatch names the answer only
        # on a minority of scenes.
        if rules[-1]["dst"] == final and rng.random() > 0.25:
            continue
        ring_pos = rng.randrange(N_TILES)
        board = []
        for pos in range(N_TILES):
            name = start if pos == ring_pos else COLOURS[rng.randrange(4)]
            board.append({"pos": pos, "colour": name})
        counts = {c: sum(1 for t in board if t["colour"] == c) for c in COLOURS}
        top = max(counts.values())
        # shortcut block: the answer is never the single most frequent colour
        # drawn on the starting board.
        if counts[final] == top and sum(1 for c in counts.values() if c == top) == 1:
            continue
        break
    else:  # pragma: no cover - the search above succeeds in a few draws
        raise RuntimeError("scene search failed")

    squares = [("board", t["pos"]) for t in board]
    squares += [("src", r["step"]) for r in rules] + [("dst", r["step"]) for r in rules]
    if mode == "tight":
        push = rng.uniform(1.6, 5.4)
    elif mode == "near":
        push = rng.uniform(6.6, 10.5)
    else:
        push = None
    special = squares[rng.randrange(len(squares))] if push is not None else None

    for tile in board:
        p = push if special == ("board", tile["pos"]) else None
        tile["rgb"] = _make(rng, tile["colour"], p)
    for rule in rules:
        for key, tag in (("src", "src"), ("dst", "dst")):
            p = push if special == (tag, rule["step"]) else None
            rule[f"{key}_rgb"] = _make(rng, rule[key], p)

    rng.shuffle(board)
    rng.shuffle(rules)
    return {
        "seed": int(seed),
        "mode": mode,
        "ring_pos": ring_pos,
        "start": start,
        "board": board,
        "rules": rules,
        "layout": {
            "n_cards": n_cards,
            "board_dx": rng.randint(-10, 10),
            "board_dy": rng.randint(-6, 6),
            "cards_dx": rng.randint(-8, 8),
            "cards_dy": rng.randint(-6, 6),
        },
    }


# ---------------------------------------------------------------- gold
def _ring_tile(scene: dict) -> dict:
    for tile in scene["board"]:
        if tile["pos"] == scene["ring_pos"]:
            return tile
    raise KeyError("ringed tile is missing from the board bag")


def analytic_gold(scene: dict) -> str:
    """``"<final colour>|<changed>"`` read off the drawn squares."""
    start = classify(_ring_tile(scene)["rgb"])
    cur = start
    for rule in sorted(scene["rules"], key=lambda r: r["step"]):
        if cur == classify(rule["src_rgb"]):
            cur = classify(rule["dst_rgb"])
    return f"{cur}|{'yes' if cur != start else 'no'}"


def _squares(scene: dict):
    for tile in scene["board"]:
        yield tile["rgb"]
    for rule in scene["rules"]:
        yield rule["src_rgb"]
        yield rule["dst_rgb"]


def margin(scene: dict) -> float:
    worst = min(clearance(rgb) for rgb in _squares(scene))
    return round(min(MARGIN_CAP, worst), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_BELOW


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    out = []
    alt = dict(scene)
    alt["board"] = list(reversed(scene["board"]))
    out.append(("board_bag_reversed", alt))
    alt = dict(scene)
    alt["rules"] = list(reversed(scene["rules"]))
    out.append(("rule_bag_reversed", alt))
    alt = dict(scene)
    alt["board"] = scene["board"][2:] + scene["board"][:2]
    alt["rules"] = scene["rules"][1:] + scene["rules"][:1]
    out.append(("both_bags_rotated", alt))
    return out


# ---------------------------------------------------------------- drawing
def _row_x(count: int, width: int, gap: int, dx: int) -> list[int]:
    total = count * width + (count - 1) * gap
    x0 = (CANVAS[0] - total) // 2 + dx
    return [x0 + i * (width + gap) for i in range(count)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    d = ImageDraw.Draw(img)
    lay = scene["layout"]

    xs = _row_x(N_TILES, TILE, BOARD_GAP, lay["board_dx"])
    by = BOARD_Y + lay["board_dy"]
    for tile in sorted(scene["board"], key=lambda t: t["pos"]):
        x = xs[tile["pos"]]
        d.rectangle([x, by, x + TILE - 1, by + TILE - 1], fill=tuple(tile["rgb"]))
        if tile["pos"] == scene["ring_pos"]:
            d.rectangle(
                [x - 12, by - 12, x + TILE + 11, by + TILE + 11], outline=RING, width=5
            )

    n = lay["n_cards"]
    cxs = _row_x(n, CARD_W, CARD_GAP, lay["cards_dx"])
    cy = CARD_Y + lay["cards_dy"]
    for rule in sorted(scene["rules"], key=lambda r: r["step"]):
        x = cxs[rule["step"]]
        d.rectangle([x, cy, x + CARD_W - 1, cy + CARD_H - 1], fill=CARD_BG,
                    outline=CARD_EDGE, width=3)
        sy = cy + SW_DY
        d.rectangle([x + SW_DX1, sy, x + SW_DX1 + SW - 1, sy + SW - 1],
                    fill=tuple(rule["src_rgb"]))
        d.rectangle([x + SW_DX2, sy, x + SW_DX2 + SW - 1, sy + SW - 1],
                    fill=tuple(rule["dst_rgb"]))
        my = sy + SW // 2
        d.rectangle([x + SW_DX1 + SW + 3, my - 2, x + SW_DX2 - 8, my + 2], fill=ARROW)
        d.polygon(
            [(x + SW_DX2 - 3, my), (x + SW_DX2 - 11, my - 7), (x + SW_DX2 - 11, my + 7)],
            fill=ARROW,
        )

    # the drawn order arrow under the card row
    ay = cy + CARD_H + 34
    x0, x1 = cxs[0], cxs[-1] + CARD_W
    d.rectangle([x0, ay - 3, x1 - 14, ay + 3], fill=ARROW)
    d.polygon([(x1, ay), (x1 - 16, ay - 12), (x1 - 16, ay + 12)], fill=ARROW)
    return img
