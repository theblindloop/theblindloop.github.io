"""Renderer and analytic arm for the connected checkpoint-route world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "route-toggle-accumulator-0.1.0"
CANVAS = (512, 512)
EVENT_COUNT = 6
POINT_COUNT = EVENT_COUNT + 2

PALETTE = {
    "background": (250, 247, 240),
    "route": (30, 67, 102),
    "gate": (222, 126, 24),
    "start": (25, 132, 82),
    "finish": (125, 78, 158),
    "distractor": (155, 164, 172),
}


def _palette(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    raw = scene["palette"]
    return {name: tuple(int(channel) for channel in value) for name, value in raw.items()}


def _route_for(rng: random.Random) -> list[list[int]]:
    # All templates stay inside the middle of the canvas.  Their 8 checkpoints
    # are separated enough for the pixel oracle while the full route remains a
    # single thin component in a medium-sized region.
    template = rng.randrange(4)
    phase = rng.uniform(-0.12, 0.12)
    shift_x = rng.randint(-7, 7)
    shift_y = rng.randint(-7, 7)
    points: list[list[int]] = []
    for index in range(POINT_COUNT):
        t = index / (POINT_COUNT - 1)
        x = 138.0 + 232.0 * t + 5.0 * math.sin(math.pi * (t + phase))
        if template == 0:
            y = 148.0 + 214.0 * t + 14.0 * math.sin(math.pi * t)
        elif template == 1:
            y = 158.0 + 194.0 * t + 14.0 * math.sin(2.0 * math.pi * t)
        elif template == 2:
            y = 150.0 + 216.0 * t - 12.0 * math.sin(math.pi * t)
        else:
            y = 160.0 + 200.0 * t + 10.0 * math.sin(1.5 * math.pi * t)
        # Small, deterministic jitter changes geometry without changing the
        # route's order or its medium regional footprint.
        jitter = 3.0 if 0 < index < POINT_COUNT - 1 else 0.0
        x += rng.uniform(-jitter, jitter) + shift_x
        y += rng.uniform(-jitter, jitter) + shift_y
        points.append([int(round(x)), int(round(y))])
    return points


def _distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(62, 74), (448, 84), (67, 432), (449, 444), (92, 260), (424, 262)]
    rng.shuffle(anchors)
    result = []
    for index, (x, y) in enumerate(anchors[:4]):
        result.append(
            {
                "kind": index % 2,
                "x": x + rng.randint(-10, 10),
                "y": y + rng.randint(-10, 10),
                "size": rng.randint(5, 9),
            }
        )
    return result


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, unambiguous route and six visible operations."""
    rng = random.Random(int(seed))
    events = [rng.randrange(2) for _ in range(EVENT_COUNT)]
    desired_parity = int(seed) & 1
    if sum(events) % 2 != desired_parity:
        events[rng.randrange(EVENT_COUNT)] ^= 1
    return {
        "events": events,
        "route_points": _route_for(rng),
        "distractors": _distractors(rng),
        "route_width": rng.choice([6, 7]),
        "marker_radius": 9,
        "palette": {name: list(value) for name, value in PALETTE.items()},
        "canvas": list(CANVAS),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible route, gate, endpoint, and nuisance geometry."""
    width, height = (int(value) for value in scene["canvas"])
    colors = _palette(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=colors["distractor"], width=3)
        else:
            draw.rectangle((x - size, y - size, x + size, y + size), outline=colors["distractor"], width=3)

    points = [(int(point[0]), int(point[1])) for point in scene["route_points"]]
    draw.line(points, fill=colors["route"], width=int(scene["route_width"]), joint="curve")
    radius = int(scene["marker_radius"])

    # Filled gate silhouettes share one color.  Shape, not color, carries the
    # operation semantics: diamond toggles; circle holds.
    for point, event in zip(points[1:-1], scene["events"], strict=True):
        x, y = point
        if int(event):
            draw.polygon(
                [(x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)],
                fill=colors["gate"],
            )
        else:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=colors["gate"])

    start_x, start_y = points[0]
    draw.rectangle(
        (start_x - radius, start_y - radius, start_x + radius, start_y + radius),
        fill=colors["start"],
    )
    finish_x, finish_y = points[-1]
    draw.ellipse(
        (finish_x - radius, finish_y - radius, finish_x + radius, finish_y + radius),
        fill=colors["finish"],
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    state = 0
    for event in scene["events"]:
        state ^= int(event)
    return "yes" if state else "no"


def margin(scene: dict[str, Any]) -> float:
    points = [tuple(float(value) for value in point) for point in scene["route_points"]]
    radius = float(scene["marker_radius"])
    width = float(scene["route_width"])
    clearances = [math.dist(left, right) - (2.0 * radius + width) for left, right in zip(points, points[1:])]
    return round(min(clearances), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # No unrendered or render-invariant label-bearing transformation is
    # declared.  The v0.4 verifier separately audits the label input fields.
    return []
