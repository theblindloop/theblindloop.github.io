"""Independent pixel-only inverse for the checkpoint lamp raster."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _near(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - np.array(color, dtype=np.int16)), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append({"area": len(xs), "x0": min(xs), "x1": max(xs), "y0": min(ys), "y1": max(ys), "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
    return result


def _state_at(cyan: np.ndarray, cx: float, cy: float, radius: float) -> bool:
    yy, xx = np.ogrid[:cyan.shape[0], :cyan.shape[1]]
    disk = (xx - cx) ** 2 + (yy - cy) ** 2 <= (radius * 0.72) ** 2
    return int(np.count_nonzero(cyan & disk)) > 180


def _pip_count(ink_components: list[dict], x0: float, x1: float, y0: float, y1: float) -> int:
    pips = [c for c in ink_components if x0 < c["cx"] < x1 and y0 < c["cy"] < y1 and 110 <= c["area"] <= 190 and 12 <= c["x1"] - c["x0"] + 1 <= 16 and 12 <= c["y1"] - c["y0"] + 1 <= 16]
    if len(pips) not in (1, 2, 3):
        raise ValueError(f"ambiguous pip count {len(pips)}")
    return len(pips)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 600:
        raise ValueError("invalid raster")
    ink = _near(array, (24, 43, 61))
    cyan = _near(array, (34, 211, 238))
    gold = _near(array, (245, 184, 35))
    orange = _near(array, (244, 128, 48))
    green = _near(array, (38, 166, 91))
    purple = _near(array, (145, 82, 190))
    card_fill = _near(array, (244, 247, 249))
    ink_components = _components(ink)

    # Recover the six card interiors as the large pale connected regions.
    cards = [c for c in _components(card_fill) if c["area"] > 9000 and c["y0"] > array.shape[0] * 0.45]
    cards.sort(key=lambda c: c["cx"])
    if len(cards) != 6:
        raise ValueError(f"expected six cards, found {len(cards)}")

    # Initial lamp outlines are the three near-square ink components above cards.
    outlines = [c for c in ink_components if 50 <= c["x1"] - c["x0"] + 1 <= 70 and 50 <= c["y1"] - c["y0"] + 1 <= 70 and c["cy"] < min(card["y0"] for card in cards)]
    outlines.sort(key=lambda c: c["cx"])
    if len(outlines) != 3:
        raise ValueError(f"expected three initial lamps, found {len(outlines)}")
    states = [False, False, False]
    query = None
    gold_components = [c for c in _components(gold) if c["area"] > 500]
    if len(gold_components) != 1:
        raise ValueError("query ring missing or ambiguous")
    query_x = gold_components[0]["cx"]
    for outline in outlines:
        identity = _pip_count(ink_components, outline["cx"] - 45, outline["cx"] + 45, outline["cy"] + 35, outline["cy"] + 100) - 1
        states[identity] = _state_at(cyan, outline["cx"], outline["cy"], 30)
        if abs(outline["cx"] - query_x) < 50:
            query = identity
    if query is None:
        raise ValueError("cannot bind query ring")

    saved = None
    for card in cards:
        x0, x1, y0, y1 = card["x0"], card["x1"], card["y0"], card["y1"]
        region = np.zeros(ink.shape, dtype=bool)
        region[y0:y1 + 1, x0:x1 + 1] = True
        counts = {
            "set": int(np.count_nonzero(orange & region)),
            "save": int(np.count_nonzero(green & region)),
            "restore": int(np.count_nonzero(purple & region)),
        }
        kind, strength = max(counts.items(), key=lambda item: item[1])
        if strength < 1000:
            raise ValueError("card type is ambiguous")
        if kind == "set":
            target = _pip_count(ink_components, x0 + 15, x1 - 15, y0 + 50, y0 + 115) - 1
            states[target] = _state_at(cyan, card["cx"], y0 + 150, 28)
        elif kind == "save":
            saved = list(states)
        else:
            if saved is None:
                raise ValueError("RESTORE before SAVE")
            states = list(saved)
    return "ON" if states[query] else "OFF"
