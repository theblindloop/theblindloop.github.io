"""Deterministic renderer and analytic replay for checkpointed lamps."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1024, 560
INK = (24, 43, 61)
ON = (34, 211, 238)
QUERY = (245, 184, 35)
SET = (244, 128, 48)
SAVE = (38, 166, 91)
RESTORE = (145, 82, 190)
CARD = (244, 247, 249)
BG = (255, 255, 255)


def _replay(initial_states: list[bool], operations: list[dict]) -> list[bool]:
    current = list(initial_states)
    saved = None
    for operation in operations:
        kind = operation["kind"]
        if kind == "set":
            current[int(operation["target"])] = bool(operation["state"])
        elif kind == "save":
            saved = list(current)
        elif kind == "restore":
            if saved is None:
                raise ValueError("RESTORE without SAVE")
            current = list(saved)
        else:
            raise ValueError(f"unknown operation {kind}")
    return current


def _restore_is_relevant(initial_states: list[bool], query: int, operations: list[dict]) -> bool:
    full = _replay(initial_states, operations)[query]
    without = _replay(initial_states, [op for op in operations if op["kind"] != "restore"])[query]
    return full != without


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(10000):
        initial = [bool(rng.getrandbits(1)) for _ in range(3)]
        query = rng.randrange(3)
        identity_order = list(range(3))
        rng.shuffle(identity_order)
        save_at = rng.choice((1, 2))
        restore_at = rng.choice(tuple(i for i in (3, 4) if i > save_at))
        operations = []
        for i in range(6):
            if i == save_at:
                operations.append({"kind": "save"})
            elif i == restore_at:
                operations.append({"kind": "restore"})
            else:
                operations.append(
                    {"kind": "set", "target": rng.randrange(3), "state": bool(rng.getrandbits(1))}
                )
        if _restore_is_relevant(initial, query, operations):
            return {
                "initial_states": initial,
                "query": query,
                "identity_order": identity_order,
                "operations": operations,
            }
    raise RuntimeError("could not sample a causally relevant checkpoint scene")


def analytic_gold(scene: dict) -> str:
    states = _replay(scene["initial_states"], scene["operations"])
    return "ON" if states[int(scene["query"])] else "OFF"


def margin(scene: dict) -> float:
    # Runner-up margin in raster units: the smallest intentional clearance
    # between categorical color regions and their decision thresholds.
    _ = scene["initial_states"], scene["query"], scene["operations"]
    return 18.0


def is_quarantined(scene: dict) -> bool:
    initial = scene["initial_states"]
    query = int(scene["query"])
    operations = scene["operations"]
    seen_save = False
    valid = True
    for operation in operations:
        if operation["kind"] == "save":
            seen_save = True
        elif operation["kind"] == "restore" and not seen_save:
            valid = False
    return not (valid and _restore_is_relevant(initial, query, operations))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False):
    # Pillow's bundled bitmap font keeps rendering independent of host fonts.
    return ImageFont.load_default(size=size)


def _pips(draw: ImageDraw.ImageDraw, center_x: int, y: int, count: int, radius: int = 7) -> None:
    spacing = 21
    start = center_x - spacing * (count - 1) // 2
    for j in range(count):
        x = start + spacing * j
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=INK)


def _lamp(draw: ImageDraw.ImageDraw, x: int, y: int, state: bool, radius: int = 30) -> None:
    fill = ON if state else BG
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=INK, width=5)


def render(scene: dict) -> Image.Image:
    initial_states = scene["initial_states"]
    query = int(scene["query"])
    identity_order = scene["identity_order"]
    operations = scene["operations"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    title_font = _font(24)
    card_font = _font(18)
    draw.text((35, 22), "START", fill=INK, font=title_font)
    draw.line((35, 57, 989, 57), fill=(205, 215, 222), width=3)

    lamp_xs = (230, 512, 794)
    for slot, identity in enumerate(identity_order):
        x = lamp_xs[slot]
        if identity == query:
            draw.ellipse((x - 42, 80, x + 42, 164), outline=QUERY, width=7)
        _lamp(draw, x, 122, bool(initial_states[identity]))
        _pips(draw, x, 192, identity + 1)

    draw.text((35, 254), "COMMANDS: read left to right", fill=INK, font=title_font)
    card_w, card_h, gap, left, top = 142, 214, 19, 35, 306
    for index, operation in enumerate(operations):
        x0 = left + index * (card_w + gap)
        x1, y1 = x0 + card_w, top + card_h
        draw.rounded_rectangle((x0, top, x1, y1), radius=12, fill=CARD, outline=INK, width=4)
        kind = operation["kind"]
        color = SET if kind == "set" else SAVE if kind == "save" else RESTORE
        draw.rounded_rectangle((x0 + 8, top + 8, x1 - 8, top + 51), radius=7, fill=color)
        label = kind.upper()
        label_box = draw.textbbox((0, 0), label, font=card_font)
        draw.text(((x0 + x1 - (label_box[2] - label_box[0])) // 2, top + 20), label, fill=BG, font=card_font)
        if kind == "set":
            _pips(draw, (x0 + x1) // 2, top + 91, int(operation["target"]) + 1)
            _lamp(draw, (x0 + x1) // 2, top + 158, bool(operation["state"]), radius=25)
        elif kind == "save":
            cx = (x0 + x1) // 2
            draw.rectangle((cx - 31, top + 88, cx + 31, top + 155), outline=SAVE, width=7)
            draw.polygon(((cx, top + 173), (cx - 26, top + 140), (cx + 26, top + 140)), fill=SAVE)
        else:
            cx = (x0 + x1) // 2
            draw.rectangle((cx - 31, top + 112, cx + 31, top + 179), outline=RESTORE, width=7)
            draw.polygon(((cx, top + 78), (cx - 26, top + 111), (cx + 26, top + 111)), fill=RESTORE)
    return image
