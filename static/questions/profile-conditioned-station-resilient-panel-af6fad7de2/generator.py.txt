"""Deterministic station-network world rendered with flat, measurable ink."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 330
PANEL_W, PANEL_H = 270, 270
PANEL_X = (15, 315, 615)
PANEL_Y = 30
FRAME = (37, 99, 235)
LINK = (100, 116, 139)
STATION = (8, 145, 178)
BACKGROUND = (255, 255, 255)
RADIUS = 10


def _templates() -> list[dict]:
    # All three graphs have n=8, m=9 and degrees [3,3,2,2,2,2,2,2].
    theta_points = [
        [25, 130], [82, 42], [158, 42], [130, 130],
        [72, 218], [130, 225], [188, 218], [235, 130],
    ]
    theta_edges = [
        [0, 1], [1, 2], [2, 7],
        [0, 3], [3, 7],
        [0, 4], [4, 5], [5, 6], [6, 7],
    ]

    square_points = [
        [28, 72], [76, 28], [124, 72], [76, 116],
        [142, 166], [190, 122], [238, 166], [190, 214],
    ]
    square_edges = [
        [0, 1], [1, 2], [2, 3], [3, 0],
        [4, 5], [5, 6], [6, 7], [7, 4], [3, 4],
    ]

    mixed_points = [
        [26, 82], [76, 28], [126, 82],
        [145, 146], [184, 112], [232, 135], [222, 200], [166, 220],
    ]
    mixed_edges = [
        [0, 1], [1, 2], [2, 0],
        [3, 4], [4, 5], [5, 6], [6, 7], [7, 3], [2, 3],
    ]
    return [
        {"points": theta_points, "edges": theta_edges},
        {"points": square_points, "edges": square_edges},
        {"points": mixed_points, "edges": mixed_edges},
    ]


def _connected_without(edges: list[list[int]], n: int, removed: int) -> bool:
    remaining = [v for v in range(n) if v != removed]
    if not remaining:
        return True
    adj = {v: [] for v in remaining}
    for a, b in edges:
        if a != removed and b != removed:
            adj[a].append(b)
            adj[b].append(a)
    seen = {remaining[0]}
    queue = deque([remaining[0]])
    while queue:
        for nxt in adj[queue.popleft()]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return len(seen) == len(remaining)


def _resilient(panel: dict) -> bool:
    return all(
        _connected_without(panel["edges"], len(panel["points"]), removed)
        for removed in range(len(panel["points"]))
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    panels = copy.deepcopy(_templates())
    # Reflections alter geometry without changing any abstract graph.
    for panel in panels:
        if rng.randrange(2):
            panel["points"] = [[250 - x, y] for x, y in panel["points"]]
        if rng.randrange(2):
            panel["points"] = [[x, 250 - y] for x, y in panel["points"]]
    # Exact cyclic answer balance for consecutive generator seeds.
    answer_slot = int(seed) % 3
    good, bad_a, bad_b = panels
    bads = [bad_a, bad_b]
    if rng.randrange(2):
        bads.reverse()
    ordered = [None, None, None]
    ordered[answer_slot] = good
    j = 0
    for index in range(3):
        if ordered[index] is None:
            ordered[index] = bads[j]
            j += 1
    return {"panels": ordered}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot, panel in enumerate(scene["panels"]):
        ox, oy = PANEL_X[slot], PANEL_Y
        draw.rounded_rectangle(
            (ox, oy, ox + PANEL_W, oy + PANEL_H), radius=12, outline=FRAME, width=4
        )
        points = [(ox + 10 + x, oy + 10 + y) for x, y in panel["points"]]
        for a, b in panel["edges"]:
            draw.line((points[a], points[b]), fill=LINK, width=7)
        for x, y in points:
            draw.ellipse(
                (x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS),
                fill=STATION,
                outline=(14, 116, 144),
                width=2,
            )
    return image


def analytic_gold(scene: dict) -> str:
    labels = ("left", "center", "right")
    winners = []
    for index, panel in enumerate(scene["panels"]):
        n = len(panel["points"])
        ok = True
        for removed in range(n):
            remaining = [v for v in range(n) if v != removed]
            adj = {v: [] for v in remaining}
            for a, b in panel["edges"]:
                if a != removed and b != removed:
                    adj[a].append(b)
                    adj[b].append(a)
            seen = {remaining[0]}
            queue = [remaining[0]]
            while queue:
                here = queue.pop()
                for nxt in adj[here]:
                    if nxt not in seen:
                        seen.add(nxt)
                        queue.append(nxt)
            if len(seen) != len(remaining):
                ok = False
                break
        if ok:
            winners.append(index)
    if len(winners) != 1:
        raise ValueError(f"expected one resilient panel, got {winners}")
    return labels[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((1.0 if _resilient(p) else 0.0 for p in scene["panels"]), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    try:
        if margin(scene) < 1.0:
            return True
        for panel in scene["panels"]:
            if len(panel["points"]) != 8 or len(panel["edges"]) != 9:
                return True
            for i, (x1, y1) in enumerate(panel["points"]):
                for x2, y2 in panel["points"][i + 1 :]:
                    if (x1 - x2) ** 2 + (y1 - y2) ** 2 < 35**2:
                        return True
        return False
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"] = list(reversed(panel["edges"]))
    return [("reverse_link_storage_order", twin)]
