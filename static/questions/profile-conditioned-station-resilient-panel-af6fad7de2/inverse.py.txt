"""Independent pixel-only graph reconstruction from the final PNG."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

STATION_RGB = np.array([8, 145, 178], dtype=np.int16)
LINK_RGB = np.array([100, 116, 139], dtype=np.int16)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 80:
            result.append(pixels)
    return result


def _station_centers(arr: np.ndarray) -> list[tuple[float, float]]:
    delta = np.abs(arr.astype(np.int16) - STATION_RGB).max(axis=2)
    groups = _components(delta <= 8)
    centers = []
    for pixels in groups:
        ys = np.array([p[0] for p in pixels], dtype=float)
        xs = np.array([p[1] for p in pixels], dtype=float)
        centers.append((float(xs.mean()), float(ys.mean())))
    return centers


def _has_link(link_mask: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    hits = 0
    total = 0
    # Test the mathematical centerline, not a wide corridor. A bent two-link
    # route can lie near the chord between its endpoints, but cannot sustain
    # exact gray centerline support across this many samples.
    for t in np.linspace(0.18, 0.82, 41):
        x = int(round(a[0] * (1 - t) + b[0] * t))
        y = int(round(a[1] * (1 - t) + b[1] * t))
        hits += bool(link_mask[y, x])
        total += 1
    return hits / total >= 0.88


def _resilient(n: int, edges: list[tuple[int, int]]) -> bool:
    for removed in range(n):
        remaining = [v for v in range(n) if v != removed]
        adj = {v: [] for v in remaining}
        for a, b in edges:
            if a != removed and b != removed:
                adj[a].append(b)
                adj[b].append(a)
        seen = {remaining[0]}
        queue = [remaining[0]]
        while queue:
            for nxt in adj[queue.pop()]:
                if nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
        if len(seen) != len(remaining):
            return False
    return True


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[1] < 600 or arr.shape[0] < 250:
        raise ValueError("unexpected image dimensions")
    centers = _station_centers(arr)
    link_delta = np.abs(arr.astype(np.int16) - LINK_RGB).max(axis=2)
    link_mask = link_delta <= 8
    labels = ("left", "center", "right")
    winners = []
    width = arr.shape[1]
    for slot in range(3):
        lo, hi = slot * width / 3, (slot + 1) * width / 3
        points = sorted((p for p in centers if lo < p[0] < hi), key=lambda p: (p[1], p[0]))
        if len(points) != 8:
            raise ValueError(f"panel {slot} has {len(points)} recoverable stations")
        edges = []
        for i in range(8):
            for j in range(i + 1, 8):
                if _has_link(link_mask, points[i], points[j]):
                    edges.append((i, j))
        if len(edges) != 9:
            raise ValueError(f"panel {slot} has {len(edges)} recoverable links")
        if _resilient(8, edges):
            winners.append(slot)
    if len(winners) != 1:
        raise ValueError(f"expected one resilient pixel graph, got {winners}")
    return labels[winners[0]]
