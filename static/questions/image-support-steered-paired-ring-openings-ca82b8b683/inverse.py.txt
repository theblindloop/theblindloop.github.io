"""Pixel-only inverse arm for the paired-ring-openings world.

The arm receives the final PNG and nothing else: no latent scene, no renderer
import, no analytic gold.  It rebuilds the declared global evidence itself --
the dark support, its connected components, each component's own circle centre,
and each component's angular occupancy field -- then reads the opening
(negative-space) direction out of that field and classifies only the resulting
two-node relation against the axis joining the two recovered centres.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "paired_ring_openings-oracle-0.1.0"

DARK_LUMA = 140.0
MIN_SPECK = 12
MIN_AREA = 120
MAX_AREA = 900
BINS = 36
MIN_GAP_BINS = 5
MAX_RUNNER_BINS = 3
SEARCH = 8
QUARANTINE = 0.30
UNCLEAR = "unclear"


def _dark_support(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    luma = 0.299 * array[:, :, 0] + 0.587 * array[:, :, 1] + 0.114 * array[:, :, 2]
    return luma < DARK_LUMA


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    rows, columns = np.nonzero(mask)
    remaining = set(zip(rows.tolist(), columns.tolist()))
    components: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbour in (
                (y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1),
                (y - 1, x - 1), (y - 1, x + 1), (y + 1, x - 1), (y + 1, x + 1),
            ):
                if neighbour in remaining:
                    remaining.discard(neighbour)
                    stack.append(neighbour)
                    component.append(neighbour)
        components.append(component)
    return components


def _fit_centre(points: np.ndarray) -> tuple[float, float]:
    """Centre that makes the component's radii most uniform (annulus fit)."""
    ys = points[:, 0]
    xs = points[:, 1]
    cy0 = int(round((ys.min() + ys.max()) / 2.0))
    cx0 = int(round((xs.min() + xs.max()) / 2.0))
    best = None
    for dy in range(-SEARCH, SEARCH + 1):
        for dx in range(-SEARCH, SEARCH + 1):
            cy = cy0 + dy
            cx = cx0 + dx
            spread = float(np.hypot(ys - cy, xs - cx).var())
            if best is None or spread < best[0]:
                best = (spread, float(cy), float(cx))
    return best[1], best[2]


def _opening_direction(points: np.ndarray, cy: float, cx: float):
    """Bisector of the single empty sector of the component's angular field."""
    angles = np.arctan2(points[:, 0] - cy, points[:, 1] - cx) % (2.0 * math.pi)
    index = np.minimum((angles / (2.0 * math.pi) * BINS).astype(int), BINS - 1)
    occupancy = np.bincount(index, minlength=BINS)
    empty = occupancy == 0
    filled = np.nonzero(~empty)[0]
    if not len(filled) or not empty.any():
        return None
    anchor = int(filled[0])
    runs: list[tuple[int, int]] = []
    length = 0
    start = 0
    for step in range(BINS):
        position = (anchor + 1 + step) % BINS
        if empty[position]:
            if length == 0:
                start = position
            length += 1
        elif length:
            runs.append((length, start))
            length = 0
    if length:
        runs.append((length, start))
    if not runs:
        return None
    runs.sort(reverse=True)
    top, top_start = runs[0]
    runner_up = runs[1][0] if len(runs) > 1 else 0
    if top < MIN_GAP_BINS or runner_up > MAX_RUNNER_BINS:
        return None
    middle = (top_start + top / 2.0) * (2.0 * math.pi / BINS)
    return math.cos(middle), math.sin(middle)


def decision_from_image(image: Image.Image) -> str:
    support = _dark_support(image)
    components = [item for item in _components(support) if len(item) >= MIN_SPECK]
    if len(components) != 2:
        return UNCLEAR
    if any(not (MIN_AREA <= len(item) <= MAX_AREA) for item in components):
        return UNCLEAR

    centres = []
    directions = []
    for component in components:
        points = np.asarray(component, dtype=np.float64)
        cy, cx = _fit_centre(points)
        direction = _opening_direction(points, cy, cx)
        if direction is None:
            return UNCLEAR
        centres.append((cx, cy))
        directions.append(direction)

    (ax, ay), (bx, by) = centres
    length = math.hypot(bx - ax, by - ay)
    if length < 1.0:
        return UNCLEAR
    ux, uy = (bx - ax) / length, (by - ay) / length
    cos_a = directions[0][0] * ux + directions[0][1] * uy
    cos_b = directions[1][0] * (-ux) + directions[1][1] * (-uy)
    if min(abs(cos_a), abs(cos_b)) < QUARANTINE:
        return UNCLEAR
    if cos_a > 0.0 and cos_b > 0.0:
        return "facing"
    if cos_a < 0.0 and cos_b < 0.0:
        return "apart"
    return "same_way"
