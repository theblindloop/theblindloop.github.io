"""Deterministic renderer and analytic scorer for polygon-parallelism search."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "parallel-sides-search-0.1.0"
WIDTH, HEIGHT = 900, 700
BACKGROUND = (250, 249, 246)
INK = (33, 52, 78)
PANEL_FILL = (246, 248, 251)
BORDER = (184, 196, 210)
LABEL = (75, 93, 115)
PANEL_BOXES = (
    (40, 80, 430, 370),
    (470, 80, 860, 370),
    (40, 400, 430, 690),
    (470, 400, 860, 690),
)
PANEL_CENTERS = ((235, 225), (665, 225), (235, 545), (665, 545))
PANEL_LABELS = "ABCD"

# Every shape has the same x/y extent and area before the independent nuisance
# transform.  The target is a skewed parallelogram.  Each near miss preserves
# the area exactly while moving the two vertices that define one side pair by
# the same horizontal amount, so ink mass and bounding-box extent are not
# answer-coded.
_TARGET = ((-72.0, -48.0), (48.0, -48.0), (72.0, 48.0), (-48.0, 48.0))
_DISTRACTORS = (
    ((-72.0, -48.0), (72.0, -48.0), (72.0, 48.0), (-24.0, 48.0)),
    ((-72.0, -48.0), (24.0, -48.0), (72.0, 48.0), (-72.0, 48.0)),
    ((-72.0, -48.0), (64.0, -48.0), (72.0, 48.0), (-32.0, 48.0)),
)
_NEAR_MISS = ((-72.0, -48.0), (56.0, -48.0), (72.0, 48.0), (-40.0, 48.0))


def _transform(points: tuple[tuple[float, float], ...], theta: float, scale: float, dx: float, dy: float) -> list[list[float]]:
    c, s = math.cos(theta), math.sin(theta)
    return [
        [
            round(scale * (x * c - y * s) + dx, 4),
            round(scale * (x * s + y * c) + dy, 4),
        ]
        for x, y in points
    ]


def sample_scene(seed: int) -> dict:
    """Return a JSON-serializable scene whose answer is balanced by panel."""
    rng = random.Random(int(seed))
    target_panel = rng.randrange(4)
    theta = rng.uniform(-math.pi, math.pi)
    distractors = list(_DISTRACTORS)
    if int(seed) % 13 == 0:
        distractors[0] = _NEAR_MISS
    rng.shuffle(distractors)

    panels: list[list[list[float]]] = []
    distractor_index = 0
    for panel_index in range(4):
        if panel_index == target_panel:
            base = _TARGET
        else:
            base = distractors[distractor_index]
            distractor_index += 1
        # Independent translation and scale are nuisance variables, not label
        # inputs: the parallelism score is invariant to both.
        scale = rng.uniform(0.96, 1.04)
        dx = rng.uniform(-8.0, 8.0)
        dy = rng.uniform(-8.0, 8.0)
        panels.append(_transform(base, theta, scale, dx, dy))
    return {"panels": panels}


def _undirected_angle_difference(a: float, b: float) -> float:
    """Small angle in radians between two undirected lines."""
    return abs((a - b + math.pi / 2.0) % math.pi - math.pi / 2.0)


def _polygon_score(points: list[list[float]]) -> float:
    if len(points) != 4:
        return float("inf")
    vectors = []
    for index in range(4):
        x0, y0 = points[index]
        x1, y1 = points[(index + 1) % 4]
        vectors.append((x1 - x0, y1 - y0))
    angles = [math.atan2(y, x) for x, y in vectors]
    first = math.degrees(_undirected_angle_difference(angles[0], angles[2]))
    second = math.degrees(_undirected_angle_difference(angles[1], angles[3]))
    return float(max(first, second))


def _panel_scores(scene: dict) -> list[float]:
    return [_polygon_score(panel) for panel in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    if len(scores) != 4 or not all(math.isfinite(value) for value in scores):
        raise ValueError("scene must contain four valid quadrilaterals")
    best = min(scores)
    winners = [index for index, value in enumerate(scores) if abs(value - best) < 1e-9]
    if len(winners) != 1:
        raise ValueError(f"no unique parallel-side panel: {scores}")
    return PANEL_LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene))
    if len(scores) < 2 or not all(math.isfinite(value) for value in scores[:2]):
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _panel_scores(scene)
    if len(scores) != 4 or not all(math.isfinite(value) for value in scores):
        return True
    if len({round(value, 8) for value in scores}) < 2:
        return True
    return margin(scene) < 10.0


def _draw_panel(draw: ImageDraw.ImageDraw, panel_index: int, points: list[list[float]], font: ImageFont.ImageFont) -> None:
    x0, y0, x1, y1 = PANEL_BOXES[panel_index]
    cx, cy = PANEL_CENTERS[panel_index]
    draw.rectangle((x0, y0, x1, y1), fill=PANEL_FILL, outline=BORDER, width=2)
    draw.text((x0 + 16, y0 + 14), PANEL_LABELS[panel_index], fill=LABEL, font=font)
    translated = [(cx + point[0], cy + point[1]) for point in points]
    draw.polygon(translated, fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((40, 22), "FIND THE QUADRILATERAL WITH PARALLEL OPPOSITE SIDES", fill=LABEL, font=font)
    for panel_index, points in enumerate(scene["panels"]):
        _draw_panel(draw, panel_index, points, font)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden fields or pixel-identical latent aliases are declared."""
    return []
