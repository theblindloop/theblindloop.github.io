"""Independent pixel-only inverse for the polygon-parallelism search."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_SIZE = (900, 700)
_CENTERS = ((235, 225), (665, 225), (235, 545), (665, 545))
_INK = np.array((33, 52, 78), dtype=np.float64)
_CROP_HALF_X = 150
_CROP_HALF_Y = 112


def _mask_for_panel(image: Image.Image, center: tuple[int, int]) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float64)
    cx, cy = center
    left, right = cx - _CROP_HALF_X, cx + _CROP_HALF_X
    top, bottom = cy - _CROP_HALF_Y, cy + _CROP_HALF_Y
    if left < 0 or top < 0 or right > array.shape[1] or bottom > array.shape[0]:
        raise ValueError("panel crop falls outside image")
    patch = array[top:bottom, left:right]
    distance = np.linalg.norm(patch - _INK, axis=2)
    mask = distance <= 78.0
    if int(mask.sum()) < 1000:
        raise ValueError("panel has no measurable filled quadrilateral")
    return mask


def _cross(o: tuple[int, int], a: tuple[int, int], b: tuple[int, int]) -> int:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _convex_hull(mask: np.ndarray) -> list[tuple[int, int]]:
    ys, xs = np.where(mask)
    points = sorted(set(zip(xs.tolist(), ys.tolist())))
    if len(points) < 4:
        raise ValueError("fewer than four boundary points")
    lower: list[tuple[int, int]] = []
    for point in points:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 0:
            lower.pop()
        lower.append(point)
    upper: list[tuple[int, int]] = []
    for point in reversed(points):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 0:
            upper.pop()
        upper.append(point)
    hull = lower[:-1] + upper[:-1]
    if len(hull) < 4:
        raise ValueError("mask hull is not polygonal")
    return hull


def _point_line_distance(point: tuple[int, int], start: tuple[int, int], end: tuple[int, int]) -> float:
    x, y = point
    x0, y0 = start
    x1, y1 = end
    dx, dy = x1 - x0, y1 - y0
    if dx == 0 and dy == 0:
        return math.hypot(x - x0, y - y0)
    return abs(dy * x - dx * y + x1 * y0 - y1 * x0) / math.hypot(dx, dy)


def _rdp(points: list[tuple[int, int]], epsilon: float) -> list[tuple[int, int]]:
    if len(points) < 3:
        return points
    distances = [
        (_point_line_distance(point, points[0], points[-1]), index)
        for index, point in enumerate(points[1:-1], start=1)
    ]
    distance, split = max(distances, default=(0.0, 0))
    if distance > epsilon:
        left = _rdp(points[: split + 1], epsilon)
        right = _rdp(points[split:], epsilon)
        return left[:-1] + right
    return [points[0], points[-1]]


def _corner_candidates(hull: list[tuple[int, int]]) -> list[tuple[int, int]]:
    """Simplify the pixel hull into four corners without latent geometry."""
    best: list[tuple[int, int]] | None = None
    for start in range(len(hull)):
        rotated = hull[start:] + hull[:start]
        reduced = _rdp(rotated + [rotated[0]], 4.0)[:-1]
        if best is None or abs(len(reduced) - 4) < abs(len(best) - 4):
            best = reduced
        if len(reduced) == 4:
            return reduced
    if best is None or len(best) != 4:
        raise ValueError(f"corner recovery found {0 if best is None else len(best)} corners")
    return best


def _undirected_angle_difference(a: float, b: float) -> float:
    return abs((a - b + math.pi / 2.0) % math.pi - math.pi / 2.0)


def _score(corners: list[tuple[int, int]]) -> float:
    if len(corners) != 4:
        raise ValueError("expected four corners")
    angles = []
    for index in range(4):
        x0, y0 = corners[index]
        x1, y1 = corners[(index + 1) % 4]
        angles.append(math.atan2(y1 - y0, x1 - x0))
    return math.degrees(max(
        _undirected_angle_difference(angles[0], angles[2]),
        _undirected_angle_difference(angles[1], angles[3]),
    ))


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B/C/D using only the final raster and public layout."""
    if image.size != _SIZE:
        raise ValueError(f"expected {_SIZE} image, got {image.size}")
    scores = []
    for center in _CENTERS:
        mask = _mask_for_panel(image, center)
        hull = _convex_hull(mask)
        corners = _corner_candidates(hull)
        scores.append(_score(corners))
    best = min(scores)
    winners = [index for index, value in enumerate(scores) if abs(value - best) < 1.5]
    if len(winners) != 1:
        raise ValueError(f"ambiguous parallel-side scores: {scores}")
    return "ABCD"[winners[0]]
