"""Deterministic renderer for addressed ripple-carry lamp sequences."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 520

PALETTES = (
    {"bg": "#F8F7F2", "ink": "#263449", "lamp": "#1677C8", "query": "#D62F84", "card": "#F4C84A", "pip": "#6842C2"},
    {"bg": "#F4F8F6", "ink": "#29343A", "lamp": "#178C75", "query": "#D64272", "card": "#F2B84B", "pip": "#6448B8"},
)


def _advance(bits: list[int], address: int) -> None:
    """Flip address, carrying right after every 1 -> 0 transition."""
    for index in range(address, 4):
        before = bits[index]
        bits[index] = 1 - bits[index]
        if before == 0:
            break


def _replay(initial_bits: Iterable[int], operations: Iterable[int]) -> list[int]:
    bits = list(initial_bits)
    for address in operations:
        _advance(bits, int(address))
    return bits


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    query_index = rng.randrange(4)
    operations = [rng.randrange(4) for _ in range(6)]
    desired = int(seed) & 1
    viable = []
    for value in range(16):
        bits = [(value >> index) & 1 for index in range(4)]
        if _replay(bits, operations)[query_index] == desired:
            viable.append(bits)
    initial_bits = list(rng.choice(viable))
    return {
        "initial_bits": initial_bits,
        "query_index": query_index,
        "operations": operations,
        "palette_variant": rng.randrange(len(PALETTES)),
        "layout_variant": rng.randrange(3),
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = [
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ]
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    initial_bits = [int(v) for v in scene["initial_bits"]]
    query_index = int(scene["query_index"])
    operations = [int(v) for v in scene["operations"]]
    palette_variant = int(scene["palette_variant"])
    layout_variant = int(scene["layout_variant"])
    colors = PALETTES[palette_variant]
    image = Image.new("RGB", (WIDTH, HEIGHT), colors["bg"])
    draw = ImageDraw.Draw(image)

    shift_x = (-18, 0, 18)[layout_variant]
    lamp_y = 168 + (8, 0, -8)[layout_variant]
    lamp_xs = [240 + shift_x + 145 * i for i in range(4)]
    card_y = 350 + (5, 0, -5)[layout_variant]
    card_xs = [105 + shift_x + 150 * i for i in range(6)]

    draw.text((48, 28), "RIPPLE ADVANCE", fill=colors["ink"], font=_font(30, True))
    draw.text((48, 68), "Read cards left to right", fill=colors["ink"], font=_font(20))
    draw.text((92, lamp_y), "START", anchor="mm", fill=colors["ink"], font=_font(22, True))
    draw.line((145, lamp_y, 176, lamp_y), fill=colors["ink"], width=4)
    draw.polygon([(176, lamp_y), (164, lamp_y - 7), (164, lamp_y + 7)], fill=colors["ink"])

    for index, (x, bit) in enumerate(zip(lamp_xs, initial_bits)):
        if index == query_index:
            draw.ellipse((x - 43, lamp_y - 43, x + 43, lamp_y + 43), outline=colors["query"], width=7)
        draw.ellipse((x - 29, lamp_y - 29, x + 29, lamp_y + 29), fill=colors["lamp"])
        if not bit:
            draw.ellipse((x - 20, lamp_y - 20, x + 20, lamp_y + 20), fill=colors["bg"])
        draw.text((x, lamp_y + 57), str(index + 1), anchor="mm", fill=colors["ink"], font=_font(20, True))
        draw.text((x, lamp_y - 55), "ON" if bit else "OFF", anchor="mm", fill=colors["ink"], font=_font(15, True))

    draw.text((70, 270), "ADVANCE CARDS", fill=colors["ink"], font=_font(20, True))
    draw.line((70, 302, 890, 302), fill=colors["ink"], width=2)
    for order, (x, address) in enumerate(zip(card_xs, operations), start=1):
        box = (x - 55, card_y - 48, x + 55, card_y + 48)
        draw.rounded_rectangle(box, radius=13, fill=colors["card"], outline=colors["ink"], width=3)
        draw.text((x, card_y - 27), "ADV", anchor="mm", fill=colors["ink"], font=_font(17, True))
        count = address + 1
        spacing = 18
        start = x - spacing * (count - 1) / 2
        for pip_index in range(count):
            px = start + spacing * pip_index
            draw.ellipse((px - 6, card_y + 8, px + 6, card_y + 20), fill=colors["pip"])
        draw.text((x, card_y + 66), str(order), anchor="mm", fill=colors["ink"], font=_font(16))

    draw.text((480, 475), "1→0: carry right   •   0→1: stop", anchor="mm", fill=colors["ink"], font=_font(22, True))
    return image


def analytic_gold(scene: dict) -> str:
    bits = _replay(scene["initial_bits"], scene["operations"])
    return "ON" if bits[int(scene["query_index"])] else "OFF"


def margin(scene: dict) -> float:
    # Minimum intended separation (pixels) between independently parsed marks.
    return 7.0 + float((int(scene["query_index"]) + len(scene["operations"])) * 0)


def is_quarantined(scene: dict) -> bool:
    return bool(len(scene["initial_bits"]) != 4 or len(scene["operations"]) != 6)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
