"""Independent image-only inverse for ripple-carry lamps."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            output.append({"area": len(xs), "x0": min(xs), "x1": max(xs), "y0": min(ys), "y1": max(ys), "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
    return output


def _advance(bits: list[int], address: int) -> None:
    for index in range(address, 4):
        before = bits[index]
        bits[index] = 1 - bits[index]
        if before == 0:
            break


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]

    # Both renderer palettes make lamp ink strongly blue/cyan.
    lamp_mask = (b > 95) & (g > 70) & (b > r + 25) & (g > r + 20)
    lamps = [c for c in _components(lamp_mask, 180) if 45 <= c["x1"] - c["x0"] <= 70 and 45 <= c["y1"] - c["y0"] <= 70]
    lamps.sort(key=lambda c: c["cx"])
    if len(lamps) != 4:
        raise ValueError(f"expected four lamp components, found {len(lamps)}")
    bits = [1 if c["area"] > 1500 else 0 for c in lamps]

    query_mask = (r > 150) & (r > g + 60) & (b > g + 10)
    rings = [c for c in _components(query_mask, 250) if c["x1"] - c["x0"] > 70 and c["y1"] - c["y0"] > 70]
    if len(rings) != 1:
        raise ValueError(f"expected one query ring, found {len(rings)}")
    query_index = min(range(4), key=lambda i: abs(lamps[i]["cx"] - rings[0]["cx"]))

    card_mask = (r > 205) & (g > 135) & (g < 225) & (b < 110) & (r > b + 110)
    cards = [c for c in _components(card_mask, 2000) if c["x1"] - c["x0"] > 80 and c["y1"] - c["y0"] > 70]
    cards.sort(key=lambda c: c["cx"])
    if len(cards) != 6:
        raise ValueError(f"expected six operation cards, found {len(cards)}")

    pip_mask = (b > 115) & (b > r + 25) & (r > 55) & (r < 140) & (g < 105)
    pip_components = [c for c in _components(pip_mask, 50) if 8 <= c["x1"] - c["x0"] <= 16 and 8 <= c["y1"] - c["y0"] <= 16]
    operations = []
    for card in cards:
        count = sum(card["x0"] < pip["cx"] < card["x1"] and card["y0"] < pip["cy"] < card["y1"] for pip in pip_components)
        if count not in (1, 2, 3, 4):
            raise ValueError(f"invalid card address pip count {count}")
        operations.append(count - 1)

    for address in operations:
        _advance(bits, address)
    return "ON" if bits[query_index] else "OFF"
