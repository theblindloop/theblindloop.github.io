"""Independent image-only inverse for the gap-rhythm world."""

from __future__ import annotations

from math import hypot

import numpy as np
from PIL import Image

TEAL = (28, 137, 151)
TAG_COLORS = {
    "red": (211, 61, 68),
    "blue": (50, 103, 188),
    "gold": (217, 157, 31),
}


def _near(pixel: tuple[int, int, int], color: tuple[int, int, int], tolerance: float = 18.0) -> bool:
    return sum((int(a) - int(b)) ** 2 for a, b in zip(pixel, color)) <= tolerance * tolerance


def _components(image: Image.Image, color: tuple[int, int, int], min_area: int) -> list[tuple[float, float, int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    target = np.asarray(color, dtype=np.int32)
    mask = np.sum((array - target) ** 2, axis=2) <= 18 * 18
    remaining = {tuple(value) for value in np.argwhere(mask)}
    found = []
    while remaining:
        start = remaining.pop()
        queue = [start]
        count = sx = sy = 0
        while queue:
            y, x = queue.pop()
            count += 1
            sx += x
            sy += y
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                neighbor = (ny, nx)
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
        if count >= min_area:
            found.append((sx / count, sy / count, count))
    return found


def _cluster_rows(points: list[tuple[float, float, int]]) -> list[list[tuple[float, float, int]]]:
    bands: list[list[tuple[float, float, int]]] = []
    for point in sorted(points, key=lambda item: item[1]):
        if not bands or abs(point[1] - sum(p[1] for p in bands[-1]) / len(bands[-1])) > 35:
            bands.append([point])
        else:
            bands[-1].append(point)
    rows: list[list[tuple[float, float, int]]] = []
    for band in bands:
        ordered = sorted(band, key=lambda item: item[0])
        if len(ordered) % 6:
            raise ValueError("bead band cannot be divided into six-bead rows")
        rows.extend(ordered[start : start + 6] for start in range(0, len(ordered), 6))
    return rows


def _pattern(row: list[tuple[float, float, int]]) -> tuple[int, ...]:
    xs = sorted(point[0] for point in row)
    if len(xs) != 6:
        raise ValueError("each row must contain six visible beads")
    gaps = [b - a for a, b in zip(xs, xs[1:])]
    ordered = sorted(range(5), key=lambda index: gaps[index], reverse=True)
    if gaps[ordered[1]] - gaps[ordered[2]] < 12:
        raise ValueError("long and short gaps are not visibly separated")
    long_indices = set(ordered[:2])
    return tuple(int(index in long_indices) for index in range(5))


def decision_from_image(image: Image.Image) -> str:
    beads = _components(image, TEAL, min_area=120)
    rows = _cluster_rows(beads)
    if len(rows) != 4 or any(len(row) != 6 for row in rows):
        raise ValueError("could not recover four six-bead rows")
    target = min(rows, key=lambda row: sum(p[1] for p in row) / len(row))
    references = [row for row in rows if row is not target]
    references.sort(key=lambda row: sum(p[0] for p in row) / len(row))
    target_pattern = _pattern(target)
    patterns = [_pattern(row) for row in references]
    matches = [index for index, pattern in enumerate(patterns) if pattern == target_pattern]
    if len(matches) != 1:
        raise ValueError("gap-word correspondence is not unique")

    tags = []
    for name, rgb in TAG_COLORS.items():
        for cx, cy, area in _components(image, rgb, min_area=1200):
            tags.append((cx, cy, name, area))
    if len(tags) != 3:
        raise ValueError("could not recover exactly three answer tags")
    row_x = [sum(p[0] for p in row) / len(row) for row in references]
    tag = min(tags, key=lambda item: hypot(item[0] - row_x[matches[0]], item[1] - max(p[1] for p in references[matches[0]])))
    if abs(tag[0] - row_x[matches[0]]) > 90:
        raise ValueError("matched tag is not visibly bound to its row")
    return tag[2]
