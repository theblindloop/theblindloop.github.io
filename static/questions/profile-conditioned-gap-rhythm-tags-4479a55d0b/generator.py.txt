"""Deterministic Pillow world for ordered adjacent-gap correspondence."""

from __future__ import annotations

import copy
import random
from itertools import combinations

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 660
BACKGROUND = (248, 247, 242)
INK = (35, 40, 46)
MUTED = (105, 112, 120)
CARD = (255, 255, 252)
TEAL = (28, 137, 151)
TARGET = (182, 57, 154)
TAG_RGB = {
    "red": (211, 61, 68),
    "blue": (50, 103, 188),
    "gold": (217, 157, 31),
}
TAG_NAMES = tuple(TAG_RGB)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _patterns() -> list[list[int]]:
    rows = []
    for long_places in combinations(range(5), 2):
        rows.append([int(i in long_places) for i in range(5)])
    return rows


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    patterns = _patterns()
    target_pattern = list(rng.choice(patterns))
    distractors = [p for p in patterns if p != target_pattern]
    rng.shuffle(distractors)
    option_patterns = [list(target_pattern), list(distractors[0]), list(distractors[1])]
    rng.shuffle(option_patterns)

    answer_tag = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    other_tags = [name for name in TAG_NAMES if name != answer_tag]
    rng.shuffle(other_tags)
    option_tags = []
    for pattern in option_patterns:
        option_tags.append(answer_tag if pattern == target_pattern else other_tags.pop())

    return {
        "seed": int(seed),
        "target_pattern": target_pattern,
        "option_patterns": option_patterns,
        "option_tags": option_tags,
        "target_scale": rng.choice([0.88, 0.96, 1.04, 1.12]),
        "option_scales": [rng.choice([0.86, 0.94, 1.02, 1.10]) for _ in range(3)],
    }


def _valid_pattern(pattern: object) -> bool:
    return (
        isinstance(pattern, list)
        and len(pattern) == 5
        and all(value in (0, 1) for value in pattern)
        and sum(pattern) == 2
    )


def analytic_gold(scene: dict) -> str:
    target = scene["target_pattern"]
    options = scene["option_patterns"]
    tags = scene["option_tags"]
    matches = [i for i, pattern in enumerate(options) if pattern == target]
    if len(matches) != 1 or len(tags) != 3:
        raise ValueError("scene does not have one matching reference")
    return str(tags[matches[0]])


def margin(scene: dict) -> float:
    target = scene["target_pattern"]
    distances = sorted(sum(a != b for a, b in zip(target, pattern)) for pattern in scene["option_patterns"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    patterns = [scene.get("target_pattern"), *scene.get("option_patterns", [])]
    if len(patterns) != 4 or not all(_valid_pattern(pattern) for pattern in patterns):
        return True
    if sum(pattern == patterns[0] for pattern in patterns[1:]) != 1:
        return True
    scales = [scene.get("target_scale"), *scene.get("option_scales", [])]
    return len(scales) != 4 or any(not isinstance(s, (int, float)) or s < 0.8 or s > 1.2 for s in scales)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _bead_centers(pattern: list[int], cx: float, cy: float, scale: float) -> list[tuple[float, float]]:
    gaps = [(72 if value else 38) * scale for value in pattern]
    offsets = [0.0]
    for gap in gaps:
        offsets.append(offsets[-1] + gap)
    shift = (offsets[-1] / 2.0)
    return [(cx + value - shift, cy) for value in offsets]


def _rounded(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], *, outline, width: int, radius: int = 18, fill=CARD) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=fill, outline=outline, width=width)


def _row(draw: ImageDraw.ImageDraw, pattern: list[int], scale: float, cx: int, cy: int) -> None:
    radius = int(round(11 * scale))
    for x, y in _bead_centers(pattern, cx, cy, scale):
        box = (int(round(x - radius)), int(round(y - radius)), int(round(x + radius)), int(round(y + radius)))
        draw.ellipse(box, fill=TEAL, outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    # All accesses are explicit for the rendered-field audit.
    _ = scene["seed"]
    target_pattern = scene["target_pattern"]
    option_patterns = scene["option_patterns"]
    option_tags = scene["option_tags"]
    target_scale = float(scene["target_scale"])
    option_scales = scene["option_scales"]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((42, 24), "GAP-RHYTHM MATCH", font=_font(27, True), fill=INK)
    draw.text((42, 58), "Match the left-to-right SHORT/LONG gap order; row size does not matter.", font=_font(18), fill=MUTED)

    _rounded(draw, (410, 100, 790, 286), outline=TARGET, width=6)
    draw.text((600, 123), "TARGET", anchor="ma", font=_font(20, True), fill=TARGET)
    _row(draw, target_pattern, target_scale, 600, 205)
    draw.text((600, 258), "read left  →  right", anchor="ma", font=_font(15), fill=MUTED)

    centers = (205, 600, 995)
    for index, cx in enumerate(centers):
        _rounded(draw, (cx - 180, 335, cx + 180, 610), outline=(179, 184, 188), width=3)
        draw.text((cx, 360), f"REFERENCE {index + 1}", anchor="ma", font=_font(16, True), fill=MUTED)
        _row(draw, option_patterns[index], float(option_scales[index]), cx, 455)
        color = TAG_RGB[option_tags[index]]
        draw.rounded_rectangle((cx - 66, 523, cx + 66, 576), radius=12, fill=color, outline=INK, width=2)
        draw.text((cx, 549), option_tags[index].upper(), anchor="mm", font=_font(18, True), fill=(255, 255, 255))
    return image
