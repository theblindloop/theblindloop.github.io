"""Independent RGB-only inverse arm for cumulative turn-balance search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = (250, 248, 243)
INK = (36, 43, 50)
JOINT = (151, 160, 168)
GOLD = (224, 170, 38)
PALETTE = {
    "red": (211, 67, 61),
    "blue": (48, 113, 190),
    "green": (46, 145, 91),
    "purple": (139, 82, 174),
}


class OracleError(ValueError):
    pass


def _near(array: np.ndarray, rgb: tuple[int, int, int], tolerance: float) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            components.append(np.asarray(points, dtype=np.float64))
    return components


def _single_center(array: np.ndarray, rgb: tuple[int, int, int], label: str) -> np.ndarray:
    components = _components(_near(array, rgb, 38.0), 14)
    if len(components) != 1:
        raise OracleError(f"expected one visible {label} ring, found {len(components)}")
    yx = components[0].mean(axis=0)
    return np.asarray([yx[1], yx[0]], dtype=np.float64)


def _joint_centers(array: np.ndarray) -> list[np.ndarray]:
    components = _components(_near(array, JOINT, 25.0), 7)
    if not 16 <= len(components) <= 24:
        raise OracleError(f"expected 16 to 24 visible route nodes, found {len(components)}")
    centers = []
    for component in components:
        yx = component.mean(axis=0)
        centers.append(np.asarray([yx[1], yx[0]], dtype=np.float64))
    return centers


def _corridor_present(trace: np.ndarray, first: np.ndarray, second: np.ndarray) -> bool:
    dx, dy = second - first
    if abs(dx) <= 2.2 and abs(dy) >= 18.0:
        x = int(round((first[0] + second[0]) / 2.0))
        low, high = sorted((first[1], second[1]))
        coords = range(int(round(low)) + 6, int(round(high)) - 5)
        hits = [bool(trace[y, max(0, x - 2):x + 3].any()) for y in coords]
    elif abs(dy) <= 2.2 and abs(dx) >= 18.0:
        y = int(round((first[1] + second[1]) / 2.0))
        low, high = sorted((first[0], second[0]))
        coords = range(int(round(low)) + 6, int(round(high)) - 5)
        hits = [bool(trace[max(0, y - 2):y + 3, x].any()) for x in coords]
    else:
        return False
    return bool(hits) and sum(hits) / len(hits) >= 0.92


def _between(point: np.ndarray, first: np.ndarray, second: np.ndarray) -> bool:
    if abs(first[0] - second[0]) <= 2.2:
        return abs(point[0] - first[0]) <= 2.2 and min(first[1], second[1]) < point[1] < max(first[1], second[1])
    return abs(point[1] - first[1]) <= 2.2 and min(first[0], second[0]) < point[0] < max(first[0], second[0])


def _ordered_route(array: np.ndarray, nodes: list[np.ndarray], start: np.ndarray) -> list[np.ndarray]:
    trace = ~_near(array, BACKGROUND, 34.0)
    count = len(nodes)
    graph = [set() for _ in range(count)]
    for left in range(count):
        for right in range(left + 1, count):
            if any(_between(nodes[mid], nodes[left], nodes[right]) for mid in range(count) if mid not in (left, right)):
                continue
            if _corridor_present(trace, nodes[left], nodes[right]):
                graph[left].add(right)
                graph[right].add(left)
    degrees = sorted(len(edges) for edges in graph)
    if degrees != [1, 1] + [2] * (count - 2):
        raise OracleError(f"visible route graph is not one unbranched chain: degrees={degrees}")
    start_index = int(np.argmin([np.linalg.norm(node - start) for node in nodes]))
    if np.linalg.norm(nodes[start_index] - start) > 2.5 or len(graph[start_index]) != 1:
        raise OracleError("gold ring does not mark a route endpoint")
    ordered = []
    previous = -1
    current = start_index
    while True:
        ordered.append(nodes[current])
        nxt = [index for index in graph[current] if index != previous]
        if not nxt:
            break
        if len(nxt) != 1:
            raise OracleError("route traversal became ambiguous")
        previous, current = current, nxt[0]
        if len(ordered) > count:
            raise OracleError("route graph contains a cycle")
    if len(ordered) != count:
        raise OracleError("route graph is disconnected")
    return ordered


def _measure(image: Image.Image) -> tuple[str, dict[str, int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 64:
        raise OracleError("unsupported raster")
    start = _single_center(array, GOLD, "gold START")
    color_centers = {
        color: _single_center(array, rgb, color) for color, rgb in PALETTE.items()
    }
    nodes = _joint_centers(array)
    ordered = _ordered_route(array, nodes, start)

    candidate_positions: dict[str, int] = {}
    for color, center in color_centers.items():
        distances = [float(np.linalg.norm(node - center)) for node in ordered]
        position = int(np.argmin(distances))
        if distances[position] > 2.8 or position in (0, len(ordered) - 1):
            raise OracleError(f"{color} ring is not centered on an interior route bend")
        candidate_positions[color] = position
    if len(set(candidate_positions.values())) != 4:
        raise OracleError("candidate rings do not mark four distinct bends")

    cumulative: dict[int, int] = {}
    running = 0
    for position in range(1, len(ordered) - 1):
        a, b, c = ordered[position - 1], ordered[position], ordered[position + 1]
        first = b - a
        second = c - b
        cross_screen = first[0] * second[1] - first[1] * second[0]
        if abs(cross_screen) < 25.0:
            raise OracleError("visible route contains an unresolved bend")
        running += 1 if cross_screen < 0 else -1
        cumulative[position] = running
    imbalances = {color: abs(cumulative[position]) for color, position in candidate_positions.items()}
    best = min(imbalances.values())
    winners = sorted(color for color, value in imbalances.items() if value == best)
    if best != 0 or len(winners) != 1:
        raise OracleError("colored checkpoints have no unique zero-balance winner")
    ranked = sorted(imbalances.values())
    if ranked[1] < 2:
        raise OracleError("runner-up checkpoint is too close to the equality boundary")
    return winners[0], imbalances


def decision_from_image(image: Image.Image) -> str:
    decision, _ = _measure(image)
    return decision
