"""Renderer for swatchfield_window_lock_bind.

One slate BOARD of medium span carries the whole decision.  On the left of the
board a 7-by-7 FIELD of flat colour swatches is painted; on the right of the
same board, detached from the field and drawn at a visibly smaller cell size,
sits a 3-by-3 KEY patch.  The key reproduces the colours of exactly one 3-by-3
window of the field and of no other window.  Small round MARKS lie on top of
some field swatches; the key never carries a mark.  The decision is the colour
of the mark sitting on the centre swatch of the unique matching window.

The evidence is deliberately one thick connected area: the key is useless
without the field, no window of the field can be certified as the unique match
without every other window being checked, and the answer is a mark that the key
itself never shows.  Everything outside the board is inert clutter.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "swatchfield_window_lock_bind-0.1.0"

CANVAS_W = 840
CANVAS_H = 630

FIELD_N = 7
WIN_N = 3
CELL = 28
GAP = 3
KEY_CELL = 20
KEY_GAP = 3
PAD = 14
GUTTER = 22

FIELD_SPAN = FIELD_N * CELL + (FIELD_N - 1) * GAP          # 214
KEY_SPAN = WIN_N * KEY_CELL + (WIN_N - 1) * KEY_GAP        # 66
BOARD_W = PAD + FIELD_SPAN + GUTTER + KEY_SPAN + PAD       # 330
BOARD_H = 2 * PAD + FIELD_SPAN                             # 242
BOARD_MARGIN = 30

DOT_R = 7

CREAM = (238, 231, 210)
SLATE = (44, 52, 66)

SWATCH_NAMES = ("sand", "teal", "rose", "olive", "indigo")
SWATCH_RGB = {
    "sand": (226, 188, 96),
    "teal": (46, 150, 150),
    "rose": (208, 96, 128),
    "olive": (120, 158, 66),
    "indigo": (92, 102, 190),
}

MARK_NAMES = ("red", "amber", "cyan", "violet")
MARK_RGB = {
    "red": (200, 32, 32),
    "amber": (246, 176, 30),
    "cyan": (52, 200, 212),
    "violet": (150, 54, 196),
}

CLUTTER_SIZES = (8, 10, 12, 14)
CLUTTER_COUNT = (11, 16)
QUARANTINE_CLEARANCE = 3


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def _windows() -> list[tuple[int, int]]:
    span = FIELD_N - WIN_N + 1
    return [(r, c) for r in range(span) for c in range(span)]


def _window_cells(top: int, left: int, field: list[list[str]]) -> list[str]:
    return [field[top + i][left + j] for i in range(WIN_N) for j in range(WIN_N)]


def _mismatch(win: list[str], key: list[str]) -> int:
    return sum(1 for a, b in zip(win, key) if a != b)


def _disjoint(a: tuple[int, int], b: tuple[int, int]) -> bool:
    return abs(a[0] - b[0]) >= WIN_N or abs(a[1] - b[1]) >= WIN_N


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    wins = _windows()
    # window slot and answer colour cycle on coprime periods (25 and 4), so
    # neither the matched position nor the answer is predictable from the other
    target = wins[(seed * 7) % len(wins)]
    answer = MARK_NAMES[seed % len(MARK_NAMES)]

    field: list[list[str]] = []
    key_cells: list[str] = []
    decoys: list[tuple[int, int]] = []
    for _ in range(400):
        field = [[rng.choice(SWATCH_NAMES) for _ in range(FIELD_N)] for _ in range(FIELD_N)]
        key_cells = _window_cells(target[0], target[1], field)

        # plant one or two near-duplicate windows that never overlap the target
        far = [w for w in wins if w != target and _disjoint(w, target)]
        rng.shuffle(far)
        wanted_decoys = rng.choice((0, 1, 1, 2))
        decoys = []
        for cand in far:
            if len(decoys) >= wanted_decoys:
                break
            if all(_disjoint(cand, d) for d in decoys):
                decoys.append(cand)
        for top, left in decoys:
            flipped = rng.sample(range(WIN_N * WIN_N), rng.choice((1, 1, 2)))
            for idx in range(WIN_N * WIN_N):
                want = key_cells[idx]
                if idx in flipped:
                    want = rng.choice([n for n in SWATCH_NAMES if n != want])
                field[top + idx // WIN_N][left + idx % WIN_N] = want

        key_cells = _window_cells(target[0], target[1], field)
        exact = [w for w in wins if _mismatch(_window_cells(w[0], w[1], field), key_cells) == 0]
        if exact == [target]:
            break
    else:  # pragma: no cover - the loop converges in one or two tries
        raise RuntimeError("failed to sample a uniquely matched window")

    # marks: the target centre always carries one, plus balanced distractors
    centre = (target[0] + 1, target[1] + 1)
    others = [(r, c) for r in range(FIELD_N) for c in range(FIELD_N) if (r, c) != centre]
    rng.shuffle(others)
    extra = others[: rng.choice((5, 6, 7))]
    palette = list(MARK_NAMES) + [answer]
    while len(palette) < len(extra):
        palette.append(rng.choice(MARK_NAMES))
    palette = palette[: len(extra)]
    rng.shuffle(palette)
    marks = [{"r": centre[0], "c": centre[1], "colour": answer}]
    marks += [{"r": r, "c": c, "colour": col} for (r, c), col in zip(extra, palette)]
    rng.shuffle(marks)

    bx = rng.randint(BOARD_MARGIN, CANVAS_W - BOARD_W - BOARD_MARGIN)
    by = rng.randint(BOARD_MARGIN, CANVAS_H - BOARD_H - BOARD_MARGIN)

    clearance = rng.choice((2, 3, 4, 5, 6, 9, 14, 20, 26, 32))
    clutter = _sample_clutter(rng, bx, by, clearance)

    return {
        "canvas": [CANVAS_W, CANVAS_H],
        "board": [bx, by],
        "field": field,
        "key": [key_cells[i * WIN_N : (i + 1) * WIN_N] for i in range(WIN_N)],
        "marks": marks,
        "clutter": clutter,
        "clearance": clearance,
    }


def _sample_clutter(rng: random.Random, bx: int, by: int, clearance: int) -> list[dict]:
    placed: list[dict] = []
    boxes: list[tuple[int, int, int, int]] = []

    def free(x: int, y: int, size: int, gap: int) -> bool:
        if not (10 <= x <= CANVAS_W - size - 10 and 10 <= y <= CANVAS_H - size - 10):
            return False
        if (
            x + size + gap > bx
            and x - gap < bx + BOARD_W
            and y + size + gap > by
            and y - gap < by + BOARD_H
        ):
            return False
        for ox, oy, osz, _g in boxes:
            if x + size + 4 > ox and x - 4 < ox + osz and y + size + 4 > oy and y - 4 < oy + osz:
                return False
        return True

    # one deliberately tight mark hugging the board at the sampled clearance
    size = rng.choice(CLUTTER_SIZES)
    side = rng.choice(("left", "right", "top", "bottom"))
    if side == "left":
        x, y = bx - clearance - size, rng.randint(by, by + BOARD_H - size)
    elif side == "right":
        x, y = bx + BOARD_W + clearance, rng.randint(by, by + BOARD_H - size)
    elif side == "top":
        x, y = rng.randint(bx, bx + BOARD_W - size), by - clearance - size
    else:
        x, y = rng.randint(bx, bx + BOARD_W - size), by + BOARD_H + clearance
    if 6 <= x <= CANVAS_W - size - 6 and 6 <= y <= CANVAS_H - size - 6:
        placed.append({"x": x, "y": y, "size": size, "colour": rng.choice(SWATCH_NAMES + MARK_NAMES)})
        boxes.append((x, y, size, clearance))

    wanted = rng.randint(*CLUTTER_COUNT)
    guard = 0
    while len(placed) < wanted and guard < 4000:
        guard += 1
        size = rng.choice(CLUTTER_SIZES)
        x = rng.randint(10, CANVAS_W - size - 10)
        y = rng.randint(10, CANVAS_H - size - 10)
        if not free(x, y, size, 8):
            continue
        placed.append({"x": x, "y": y, "size": size, "colour": rng.choice(SWATCH_NAMES + MARK_NAMES)})
        boxes.append((x, y, size, 8))
    return placed


# --------------------------------------------------------------------------- #
# geometry, rendering
# --------------------------------------------------------------------------- #
def field_cell_box(scene: dict, r: int, c: int) -> tuple[int, int, int, int]:
    bx, by = scene["board"]
    x = bx + PAD + c * (CELL + GAP)
    y = by + PAD + r * (CELL + GAP)
    return (x, y, x + CELL, y + CELL)


def key_cell_box(scene: dict, r: int, c: int) -> tuple[int, int, int, int]:
    bx, by = scene["board"]
    x0 = bx + PAD + FIELD_SPAN + GUTTER
    y0 = by + PAD + (FIELD_SPAN - KEY_SPAN) // 2
    x = x0 + c * (KEY_CELL + KEY_GAP)
    y = y0 + r * (KEY_CELL + KEY_GAP)
    return (x, y, x + KEY_CELL, y + KEY_CELL)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["canvas"][0]), int(scene["canvas"][1])), CREAM)
    draw = ImageDraw.Draw(image)

    for piece in sorted(scene["clutter"], key=lambda p: (p["y"], p["x"], p["size"])):
        colour = SWATCH_RGB.get(piece["colour"]) or MARK_RGB[piece["colour"]]
        draw.rectangle(
            [piece["x"], piece["y"], piece["x"] + piece["size"] - 1, piece["y"] + piece["size"] - 1],
            fill=colour,
        )

    bx, by = scene["board"]
    draw.rectangle([bx, by, bx + BOARD_W - 1, by + BOARD_H - 1], fill=SLATE)

    for r in range(FIELD_N):
        for c in range(FIELD_N):
            x0, y0, x1, y1 = field_cell_box(scene, r, c)
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=SWATCH_RGB[scene["field"][r][c]])

    for mark in sorted(scene["marks"], key=lambda m: (m["r"], m["c"])):
        x0, y0, x1, y1 = field_cell_box(scene, mark["r"], mark["c"])
        cx, cy = (x0 + x1 - 1) / 2, (y0 + y1 - 1) / 2
        draw.ellipse([cx - DOT_R, cy - DOT_R, cx + DOT_R, cy + DOT_R], fill=MARK_RGB[mark["colour"]])

    for r in range(WIN_N):
        for c in range(WIN_N):
            x0, y0, x1, y1 = key_cell_box(scene, r, c)
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=SWATCH_RGB[scene["key"][r][c]])

    return image


# --------------------------------------------------------------------------- #
# analytic gold, margin, quarantine, aliases
# --------------------------------------------------------------------------- #
def _solve(scene: dict) -> tuple[tuple[int, int], int]:
    """Return the unique matching window and the runner-up mismatch count."""
    field = scene["field"]
    key = [scene["key"][i][j] for i in range(WIN_N) for j in range(WIN_N)]
    exact: list[tuple[int, int]] = []
    runner = WIN_N * WIN_N
    for top, left in _windows():
        diff = _mismatch(_window_cells(top, left, field), key)
        if diff == 0:
            exact.append((top, left))
        else:
            runner = min(runner, diff)
    if len(exact) != 1:
        raise ValueError(f"scene has {len(exact)} exact windows")
    return exact[0], runner


def analytic_gold(scene: dict) -> str:
    (top, left), _ = _solve(scene)
    centre = (top + 1, left + 1)
    hits = [m["colour"] for m in scene["marks"] if (m["r"], m["c"]) == centre]
    if len(hits) != 1:
        raise ValueError("the centre of the matching window must carry exactly one mark")
    return hits[0]


def margin(scene: dict) -> float:
    """Runner-up decision margin: swatches by which the closest wrong window fails."""
    _, runner = _solve(scene)
    return float(runner)


def is_quarantined(scene: dict) -> bool:
    return int(scene["clearance"]) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Stored-order aliases: render sorts clutter and marks into a canonical order."""
    return [
        ("reversed_clutter_order", {**scene, "clutter": list(reversed(scene["clutter"]))}),
        ("reversed_mark_order", {**scene, "marks": list(reversed(scene["marks"]))}),
    ]
