"""Standalone pixel-only inverse arm for enclosed_cavity_area.

Receives only a PIL RGB image. It segments the dark ink, flood-fills the
white background from the image border so every remaining white region is an
enclosed (negative-space) cavity, computes the area of each cavity, and answers
with the side (left / right) whose enclosed empty region is larger. Returns the
sentinel "ABSTAIN" when the raster does not contain exactly two well-resolved
enclosed cavities.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

SIDES = ("left", "right")
_MIN_HOLE_AREA = 120
_ABSTAIN = "ABSTAIN"


def _flood_background(mask):
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    queue = deque()
    for y in range(height):
        for x in (0, width - 1):
            if not mask[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    for x in range(width):
        for y in (0, height - 1):
            if not mask[y, x] and not seen[y, x]:
                seen[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < height and 0 <= nx < width:
                if not mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return seen


def _hole_components(mask, bg):
    height, width = mask.shape
    visited = bg.copy()
    regions = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] and not visited[y, x]:
                visited[y, x] = True
                queue = deque([(y, x)])
                area = 0
                sum_x = 0.0
                while queue:
                    py, px = queue.popleft()
                    area += 1
                    sum_x += px
                    for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                        ny, nx = py + dy, px + dx
                        if 0 <= ny < height and 0 <= nx < width:
                            if not mask[ny, nx] and not visited[ny, nx]:
                                visited[ny, nx] = True
                                queue.append((ny, nx))
                regions.append((area, sum_x / area))
    return regions


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    arr = np.asarray(image)
    if arr.ndim != 3 or arr.shape[2] != 3:
        return _ABSTAIN
    gray = arr.astype(np.float64).mean(axis=2)
    mask = gray < 128.0

    bg = _flood_background(mask)
    holes = [r for r in _hole_components(mask, bg) if r[0] >= _MIN_HOLE_AREA]
    if len(holes) != 2:
        return _ABSTAIN

    (area_a, cx_a), (area_b, cx_b) = holes
    left_area, left_x = (area_a, cx_a) if cx_a < cx_b else (area_b, cx_b)
    right_area, right_x = (area_b, cx_b) if cx_a < cx_b else (area_a, cx_a)

    if left_area == right_area:
        return _ABSTAIN
    if left_area > right_area:
        return "left"
    return "right"
