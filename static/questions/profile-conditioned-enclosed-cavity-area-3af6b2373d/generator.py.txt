"""Enclosed-cavity-area world.

Two thick dark rings are drawn on a white canvas. Each ring fully encloses one
white (negative-space) cavity; the two cavities have different areas. The
question asks which of the two enclosed empty regions is larger: left or
right. The answer is a property of the global figure-ground organization (the
relative size of the two negative-space lobes), not of any single dot, colour,
count, total ink, or absolute position.
"""

from __future__ import annotations

import hashlib
import math
import random

from PIL import Image, ImageDraw

WIDTH = 256
HEIGHT = 256
BACKGROUND = (255, 255, 255)
RING_COLOR = (22, 28, 74)
SIDES = ("left", "right")

_MARGIN_QUARANTINE = 0.12


def sample_scene(seed: int) -> dict:
    rng = random.Random(hashlib.sha256(f"enclosed_cavity_area:{seed}".encode()).hexdigest())

    winner = rng.choice(SIDES)
    r_hi = rng.uniform(30.0, 40.0)
    delta = rng.uniform(0.24, 0.42)
    r_lo = r_hi * (1.0 - delta)

    thickness = rng.uniform(8.0, 12.0)

    hole_l = r_hi if winner == "left" else r_lo
    hole_r = r_hi if winner == "right" else r_lo

    cy = rng.uniform(116.0, 140.0)
    cxl = rng.uniform(58.0, 72.0)
    cxr = rng.uniform(WIDTH - 72.0, WIDTH - 58.0)

    return {
        "center_l": [cxl, cy],
        "center_r": [cxr, cy],
        "hole_l_radius": hole_l,
        "hole_r_radius": hole_r,
        "ring_thickness": thickness,
    }


def _draw_ring(draw, center, hole_radius, thickness):
    cx, cy = center
    outer = hole_radius + thickness
    def _ellipse(cx, cy, r):
        return (cx - r, cy - r, cx + r, cy + r)
    draw.ellipse(_ellipse(cx, cy, outer), fill=RING_COLOR)
    draw.ellipse(_ellipse(cx, cy, hole_radius), fill=BACKGROUND)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_ring(draw, scene["center_l"], scene["hole_l_radius"], scene["ring_thickness"])
    _draw_ring(draw, scene["center_r"], scene["hole_r_radius"], scene["ring_thickness"])
    return image


def _hole_area(radius: float) -> float:
    return math.pi * radius * radius


def analytic_gold(scene: dict) -> str:
    if scene["hole_l_radius"] > scene["hole_r_radius"]:
        return "left"
    return "right"


def margin(scene: dict) -> float:
    al = _hole_area(scene["hole_l_radius"])
    ar = _hole_area(scene["hole_r_radius"])
    denom = al + ar
    if denom <= 0.0:
        return 0.0
    return abs(al - ar) / denom


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < _MARGIN_QUARANTINE:
        return True
    cxl, cy = scene["center_l"]
    cxr, _ = scene["center_r"]
    left_outer = scene["hole_l_radius"] + scene["ring_thickness"]
    right_outer = scene["hole_r_radius"] + scene["ring_thickness"]
    if cxl - left_outer < 8.0 or cxl + left_outer > cxr - 8.0:
        return True
    if cxr - right_outer < cxl + 8.0 or cxr + right_outer > WIDTH - 8.0:
        return True
    if cy - left_outer < 8.0 or cy + left_outer > HEIGHT - 8.0:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
