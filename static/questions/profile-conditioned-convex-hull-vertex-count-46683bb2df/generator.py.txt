"""Convex-hull vertex-count world renderer.

Latent scene z = {hull_vertices, center, circumradius, dot_radius, canvas,
background, dot_color, dots}. A scattered group of small filled disks is placed
in the canvas: its `hull_vertices` (3, 4, 5 or 6) extreme disks sit on the
vertices of one strictly convex regular k-gon, and the remaining random disks
lie strictly inside that polygon. The answer is the number of corners (vertices)
of the convex hull that tightly wraps the whole dot group --- a global Gestalt
boundary revealed by the arrangement of an otherwise unconnected dot cloud.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "convex_hull_vertex_count-0.1.0"

VERTEX_OPTIONS = (3, 4, 5, 6)
RADIUS_RANGE = (95.0, 150.0)
DOT_RADIUS = 6.0
CANVAS = 512
INTERIOR_COUNT_RANGE = (6, 12)
INTERIOR_FRACTION = 0.75

# Hulls below this circumradius make corners ambiguous at pixel scale (adjacent
# corner clusters merge), so scenes sampled near the floor are quarantined.
QUARANTINE_RADIUS = 60.0


def _k_gon_vertices(k: int, radius: float) -> list[tuple[float, float]]:
    return [
        (radius * math.cos(2 * math.pi * n / k), radius * math.sin(2 * math.pi * n / k))
        for n in range(k)
    ]


def sample_scene(seed: int) -> dict:
    """Deterministically sample a full latent scene from an integer seed."""
    rng = random.Random(seed)
    k = rng.choice(VERTEX_OPTIONS)
    radius = rng.uniform(*RADIUS_RANGE)
    cx = CANVAS / 2
    cy = CANVAS / 2
    verts = _k_gon_vertices(k, radius)

    dots: list[list[float]] = [list(vert) for vert in verts]

    inradius = abs(radius * math.cos(math.pi / k))
    interior_count = rng.randint(*INTERIOR_COUNT_RANGE)
    for _ in range(interior_count):
        ang = rng.uniform(0, 2 * math.pi)
        rad = rng.uniform(0.0, inradius * INTERIOR_FRACTION)
        dots.append([rad * math.cos(ang), rad * math.sin(ang)])

    rng.shuffle(dots)

    return {
        "hull_vertices": k,
        "center": [cx, cy],
        "circumradius": round(radius, 3),
        "dot_radius": DOT_RADIUS,
        "canvas": CANVAS,
        "background": "white",
        "dot_color": "black",
        "dots": [list(round(v, 3) for v in dot) for dot in dots],
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render the scattered dot group using supersampled antialiased disks."""
    k = int(scene["hull_vertices"])
    if k not in VERTEX_OPTIONS:
        raise ValueError(f"unsupported hull vertex count {k}")
    width = int(scene["canvas"])
    ss = supersample
    img = Image.new("RGB", (width * ss, width * ss), scene["background"])
    draw = ImageDraw.Draw(img)
    radius = float(scene["dot_radius"]) * ss
    cx, cy = float(scene["center"][0]), float(scene["center"][1])
    for dot in scene["dots"]:
        dx = cx + float(dot[0])
        dy = cy + float(dot[1])
        draw.ellipse(
            [dx * ss - radius, dy * ss - radius, dx * ss + radius, dy * ss + radius],
            fill=scene["dot_color"],
        )
    return img.resize((width, width), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The answer is the convex hull's vertex count as a string."""
    return str(scene["hull_vertices"])


def margin(scene: dict) -> float:
    """One-sided verifiability margin: polygon circumradius in pixels.

    Larger hulls separate their corner clusters more clearly; hulls whose
    circumradius drops near the resolution floor are quarantined.
    """
    return float(scene["circumradius"])


def is_quarantined(scene: dict) -> bool:
    """Exclude very small hulls where corners merge at pixel scale."""
    return bool(float(scene["circumradius"]) < QUARANTINE_RADIUS)


def latent_symmetries(scene: dict) -> list:
    """No declared latent symmetry.

    The hull vertex count is an injective, pixel-revealed property of the dot
    cloud: distinct vertex counts always change the boundary structure, the dot
    layout is fully visible, and no render-invariant relabeling leaves both
    pixels and gold fixed. Declared empty.
    """
    return []
