"""Independent PNG-only station/link graph recovery and girth decision."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(int(y), int(x))]
        pixels = []
        for cy, cx in queue:
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            found.append(pixels)
    return found


def _point_segment_distance(p, a, b) -> tuple[float, float]:
    px, py = p
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return 1e9, 0.0
    t = ((px - ax) * dx + (py - ay) * dy) / denom
    qx, qy = ax + t * dx, ay + t * dy
    return math.hypot(px - qx, py - qy), t


def _has_link(a, b, centers, ink: np.ndarray) -> bool:
    for c in centers:
        if c == a or c == b:
            continue
        distance, t = _point_segment_distance(c, a, b)
        if 0.08 < t < 0.92 and distance < 13.0:
            return False
    length = math.dist(a, b)
    samples = max(20, int(length * 1.2))
    good = 0
    total = 0
    for k in range(samples + 1):
        t = k / samples
        if t * length < 10 or (1 - t) * length < 10:
            continue
        x = round(a[0] + (b[0] - a[0]) * t)
        y = round(a[1] + (b[1] - a[1]) * t)
        patch = ink[max(0, y - 2) : y + 3, max(0, x - 2) : x + 3]
        total += 1
        good += bool(patch.any())
    return total > 0 and good / total > 0.965


def _girth(adjacency: list[set[int]]) -> int:
    best = 10**9
    for root in range(len(adjacency)):
        distance = [-1] * len(adjacency)
        parent = [-1] * len(adjacency)
        distance[root] = 0
        queue = deque([root])
        while queue:
            u = queue.popleft()
            for v in adjacency[u]:
                if distance[v] < 0:
                    distance[v] = distance[u] + 1
                    parent[v] = u
                    queue.append(v)
                elif parent[u] != v:
                    best = min(best, distance[u] + distance[v] + 1)
    return best if best < 10**9 else 0


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    height, width = array.shape[:2]
    if width < 600 or height < 250:
        return "abstain"
    # Cyan station fill has high green/blue, low red; dark outlines and gray links form ink.
    cyan = (array[:, :, 0] < 45) & (array[:, :, 1] > 135) & (array[:, :, 2] > 155) & ((array[:, :, 2] - array[:, :, 0]) > 100)
    gray = (np.max(array, axis=2) - np.min(array, axis=2) < 32) & (array[:, :, 0] < 135)
    dark_cyan = (array[:, :, 0] < 40) & (array[:, :, 1] < 125) & (array[:, :, 2] < 145)
    ink = gray | cyan | dark_cyan
    labels = ("left", "center", "right")
    hits = []
    for panel_index in range(3):
        lo = round(panel_index * width / 3)
        hi = round((panel_index + 1) * width / 3)
        comps = _components(cyan[:, lo:hi], minimum=80)
        if len(comps) != 8:
            return "abstain"
        centers = []
        for pixels in comps:
            ys = [p[0] for p in pixels]
            xs = [p[1] + lo for p in pixels]
            centers.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys))))
        adjacency = [set() for _ in centers]
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if _has_link(centers[i], centers[j], centers, ink):
                    adjacency[i].add(j)
                    adjacency[j].add(i)
        if sum(map(len, adjacency)) // 2 != 9 or sorted(map(len, adjacency)) != [2, 2, 2, 2, 2, 2, 3, 3]:
            return "abstain"
        if _girth(adjacency) == 4:
            hits.append(panel_index)
    return labels[hits[0]] if len(hits) == 1 else "abstain"
