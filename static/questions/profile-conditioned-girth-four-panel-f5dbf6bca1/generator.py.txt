"""Deterministic raster world containing three degree-matched theta graphs."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = (960, 360)
PANEL_X = (20, 340, 660)
PANEL_Y = 34
PANEL_SIZE = 280
LABELS = ("left", "center", "right")
ROUTE_LENGTHS = {3: (1, 2, 6), 4: (2, 2, 5), 5: (2, 3, 4)}


def _make_panel(girth: int, rng: random.Random) -> dict:
    lengths = list(ROUTE_LENGTHS[girth])
    if girth == 3:
        offsets = [0, -96, 96]
    elif girth == 4:
        offsets = [-96, 96, 0]
    else:
        offsets = [-96, 0, 96]
    if rng.randrange(2):
        offsets = [-value for value in offsets]
    dx, dy = rng.randint(-4, 4), rng.randint(-3, 3)
    nodes: list[list[int]] = [[24 + dx, 140 + dy], [256 + dx, 140 + dy]]
    edges: list[list[int]] = []
    next_id = 2
    for length, offset in zip(lengths, offsets):
        route = [0]
        for step in range(1, length):
            t = step / length
            x = round(24 + 232 * t) + dx
            y = round(140 + offset * 4 * t * (1 - t)) + dy
            nodes.append([x, y])
            route.append(next_id)
            next_id += 1
        route.append(1)
        edges.extend([[a, b] for a, b in zip(route, route[1:])])
    if rng.randrange(2):
        nodes = [[280 - x, y] for x, y in nodes]
    return {"target_girth": girth, "nodes": nodes, "edges": edges}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    girths = [3, 4, 5]
    rng.shuffle(girths)
    palettes = [
        {"background": [250, 250, 247], "panel": [255, 255, 255], "border": [190, 199, 210], "edge": [91, 99, 109], "node": [18, 181, 199]},
        {"background": [248, 249, 246], "panel": [255, 255, 255], "border": [184, 196, 208], "edge": [83, 94, 105], "node": [12, 174, 195]},
    ]
    palette = copy.deepcopy(palettes[rng.randrange(len(palettes))])
    return {
        "canvas": list(CANVAS),
        "panels": [_make_panel(girth, rng) for girth in girths],
        "palette": palette,
        "style": {"edge_width": 7, "node_radius": 11, "panel_x": list(PANEL_X), "panel_y": PANEL_Y, "panel_size": PANEL_SIZE},
    }


def render(scene: dict) -> Image.Image:
    canvas = tuple(scene["canvas"])
    palette = scene["palette"]
    style = scene["style"]
    image = Image.new("RGB", canvas, tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    for index, panel in enumerate(scene["panels"]):
        ox, oy = style["panel_x"][index], style["panel_y"]
        size = style["panel_size"]
        draw.rounded_rectangle((ox, oy, ox + size, oy + size), radius=12, fill=tuple(palette["panel"]), outline=tuple(palette["border"]), width=3)
        points = [(ox + x, oy + y) for x, y in panel["nodes"]]
        for a, b in panel["edges"]:
            draw.line((points[a], points[b]), fill=tuple(palette["edge"]), width=style["edge_width"])
        radius = style["node_radius"]
        for x, y in points:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=tuple(palette["node"]), outline=(5, 92, 109), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    hits = [index for index, panel in enumerate(scene["panels"]) if panel["target_girth"] == 4]
    if len(hits) != 1:
        return "abstain"
    return LABELS[hits[0]]


def _graph_girth(panel: dict) -> int:
    n = len(panel["nodes"])
    adjacency = [set() for _ in range(n)]
    for a, b in panel["edges"]:
        adjacency[a].add(b)
        adjacency[b].add(a)
    best = 10**9
    for root in range(n):
        distance = [-1] * n
        parent = [-1] * n
        distance[root] = 0
        queue = deque([root])
        while queue:
            u = queue.popleft()
            for v in adjacency[u]:
                if distance[v] < 0:
                    distance[v] = distance[u] + 1
                    parent[v] = u
                    queue.append(v)
                elif parent[u] != v:
                    best = min(best, distance[u] + distance[v] + 1)
    return best if best < 10**9 else 0


def margin(scene: dict) -> float:
    girths = [_graph_girth(panel) for panel in scene["panels"]]
    if girths.count(4) != 1:
        return 0.0
    return float(min(abs(value - 4) for value in girths if value != 4))


def is_quarantined(scene: dict) -> bool:
    if tuple(scene.get("canvas", ())) != CANVAS or len(scene.get("panels", [])) != 3:
        return True
    girths = []
    for panel in scene["panels"]:
        if len(panel.get("nodes", [])) != 8 or len(panel.get("edges", [])) != 9:
            return True
        if min((x1 - x2) ** 2 + (y1 - y2) ** 2 for i, (x1, y1) in enumerate(panel["nodes"]) for x2, y2 in panel["nodes"][i + 1 :]) < 24**2:
            return True
        girths.append(_graph_girth(panel))
    return sorted(girths) != [3, 4, 5]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"] = [[b, a] for a, b in reversed(panel["edges"])]
    return [("reverse_undirected_edge_serialization", twin)]
