"""Sole circuit room: which coloured room stands on a closed corridor loop.

The scene is a 6 by 6 lattice of square rooms.  Every room boundary carries a
thick wall; a boundary that is a *doorway* is drawn with a central gap instead
of a solid run.  The doorway graph is a spanning tree over all thirty-six rooms
plus exactly one extra doorway, so the whole free space is connected and holds
exactly one circuit.  Four rooms carry a flat coloured floor.  Exactly one of
them stands on that circuit; the other three sit on tree branches, each of them
one single solid wall away from being on a loop of its own, and each of them
with more doorways than the answer room has.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1000
CELLS = 6
CELL_PX = 150
WALL_PX = 12

BACKGROUND = (247, 246, 242)
WALL = (28, 30, 34)

COLOURS: dict[str, tuple[int, int, int]] = {
    "orange": (235, 130, 25),
    "green": (26, 150, 60),
    "blue": (32, 82, 214),
    "pink": (240, 105, 185),
}
COLOUR_ORDER = ("orange", "green", "blue", "pink")

GAP_MIN = 48
GAP_MAX = 78
QUARANTINE_GAP = 56
MIN_CIRCUIT = 12


# --------------------------------------------------------------------------- #
# lattice helpers
# --------------------------------------------------------------------------- #
def _neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    row, col = cell
    out = []
    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        nr, nc = row + dr, col + dc
        if 0 <= nr < CELLS and 0 <= nc < CELLS:
            out.append((nr, nc))
    return out


def _edge_key(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    first, second = sorted((a, b))
    return (first[0], first[1], second[0], second[1])


def _graph(doorways: set[tuple[int, int, int, int]]) -> dict:
    adjacency: dict[tuple[int, int], list[tuple[int, int]]] = {}
    for row in range(CELLS):
        for col in range(CELLS):
            cell = (row, col)
            adjacency[cell] = [
                n for n in _neighbours(cell) if _edge_key(cell, n) in doorways
            ]
    return adjacency


def _reach(adjacency: dict, source: tuple[int, int]) -> dict[tuple[int, int], int]:
    seen = {source: 0}
    queue = [source]
    head = 0
    while head < len(queue):
        current = queue[head]
        head += 1
        for n in adjacency[current]:
            if n not in seen:
                seen[n] = seen[current] + 1
                queue.append(n)
    return seen


def circuit_rooms(doorways: set[tuple[int, int, int, int]]) -> set[tuple[int, int]]:
    """Rooms that lie on a closed corridor walk repeating no doorway.

    Iteratively strip rooms with at most one doorway; what survives is exactly
    the union of the circuits of the free-space graph.
    """
    adjacency = {cell: set(n) for cell, n in _graph(doorways).items()}
    alive = {cell for cell in adjacency}
    changed = True
    while changed:
        changed = False
        for cell in sorted(alive):
            if len(adjacency[cell]) <= 1:
                for n in adjacency[cell]:
                    adjacency[n].discard(cell)
                adjacency[cell] = set()
                alive.discard(cell)
                changed = True
    return alive


def _spanning_tree(rng: random.Random) -> list[tuple[int, int, int, int]]:
    cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    start = cells[rng.randrange(len(cells))]
    visited = {start}
    frontier = [start]
    edges: list[tuple[int, int, int, int]] = []
    while frontier:
        index = rng.randrange(len(frontier)) if rng.random() < 0.35 else len(frontier) - 1
        current = frontier[index]
        options = [n for n in _neighbours(current) if n not in visited]
        if not options:
            frontier.pop(index)
            continue
        nxt = options[rng.randrange(len(options))]
        visited.add(nxt)
        edges.append(_edge_key(current, nxt))
        frontier.append(nxt)
    return sorted(set(edges))


def _tree_path(parent: dict, depth: dict, a, b) -> list:
    left, right = a, b
    path_left, path_right = [a], [b]
    while depth[left] > depth[right]:
        left = parent[left]
        path_left.append(left)
    while depth[right] > depth[left]:
        right = parent[right]
        path_right.append(right)
    while left != right:
        left = parent[left]
        right = parent[right]
        path_left.append(left)
        path_right.append(right)
    return path_left + list(reversed(path_right[:-1]))


def _manhattan(a, b) -> int:
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    all_cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    for _ in range(6000):
        tree = _spanning_tree(rng)
        tree_set = set(tree)
        tree_adj = _graph(tree_set)
        root = (0, 0)
        parent: dict = {root: root}
        depth = {root: 0}
        order = [root]
        head = 0
        while head < len(order):
            current = order[head]
            head += 1
            for n in tree_adj[current]:
                if n not in depth:
                    parent[n] = current
                    depth[n] = depth[current] + 1
                    order.append(n)
        solid = [
            _edge_key(a, b)
            for a in all_cells
            for b in _neighbours(a)
            if a < b and _edge_key(a, b) not in tree_set
        ]
        long_enough = [
            edge
            for edge in solid
            if len(_tree_path(parent, depth, (edge[0], edge[1]), (edge[2], edge[3])))
            >= MIN_CIRCUIT
        ]
        if not long_enough:
            continue
        extra = long_enough[rng.randrange(len(long_enough))]
        cycle = set(
            _tree_path(parent, depth, (extra[0], extra[1]), (extra[2], extra[3]))
        )
        doorways = sorted(tree_set | {extra})
        door_set = set(doorways)
        adjacency = _graph(door_set)
        if circuit_rooms(door_set) != cycle:
            continue
        # rooms that a single extra doorway would put on a loop of their own
        near_cyclic: set = set()
        for edge in solid:
            if edge == extra:
                continue
            near_cyclic.update(
                _tree_path(parent, depth, (edge[0], edge[1]), (edge[2], edge[3]))
            )
        answers = [c for c in sorted(cycle) if len(adjacency[c]) == 2]
        if not answers:
            continue
        answer = answers[rng.randrange(len(answers))]
        decoys = [
            c
            for c in sorted(set(all_cells) - cycle)
            if len(adjacency[c]) >= 3
            and c in near_cyclic
            and _manhattan(c, answer) >= 2
        ]
        touching = [c for c in decoys if any(n in cycle for n in _neighbours(c))]
        if len(touching) < 2 or len(decoys) < 3:
            continue
        rng.shuffle(touching)
        picks = touching[:2]
        rest = [c for c in decoys if c not in picks]
        rng.shuffle(rest)
        picks = picks + rest[:1]
        if len(picks) != 3:
            continue
        rooms = [answer] + picks
        if len(set(rooms)) != 4:
            continue
        if min(_manhattan(a, b) for i, a in enumerate(rooms) for b in rooms[i + 1:]) < 2:
            continue
        colours = list(COLOUR_ORDER)
        rng.shuffle(colours)
        assignment = dict(zip(colours, rooms))
        gap = rng.randrange(GAP_MIN, GAP_MAX + 1, 2)
        scene = {
            "canvas": CANVAS,
            "cells": CELLS,
            "cell_px": CELL_PX,
            "wall_px": WALL_PX,
            "gap_px": int(gap),
            "circuit_px": len(cycle),
            "doorways": [list(edge) for edge in doorways],
            "rooms": {
                colour: [int(assignment[colour][0]), int(assignment[colour][1])]
                for colour in COLOUR_ORDER
            },
            "draw_order": [COLOUR_ORDER.index(c) for c in colours],
        }
        try:
            analytic_gold(scene)
        except ValueError:
            continue
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _origin(scene: dict) -> int:
    return (int(scene["canvas"]) - int(scene["cells"]) * int(scene["cell_px"])) // 2


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    cells = int(scene["cells"])
    step = int(scene["cell_px"])
    half = int(scene["wall_px"]) // 2
    gap = int(scene["gap_px"])
    off = _origin(scene)
    doorways = {tuple(int(v) for v in edge) for edge in scene["doorways"]}

    image = Image.new("RGB", (canvas, canvas), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for index in scene["draw_order"]:
        colour = COLOUR_ORDER[int(index)]
        row, col = (int(v) for v in scene["rooms"][colour])
        x0 = off + col * step + half
        y0 = off + row * step + half
        draw.rectangle(
            [x0, y0, x0 + step - 2 * half - 1, y0 + step - 2 * half - 1],
            fill=COLOURS[colour],
        )

    def horizontal(row: int, col: int, open_gap: bool) -> None:
        y = off + row * step
        x0 = off + col * step
        x1 = x0 + step
        if open_gap:
            mid = (x0 + x1) // 2
            draw.rectangle([x0, y - half, mid - gap // 2, y + half - 1], fill=WALL)
            draw.rectangle([mid + gap // 2, y - half, x1 - 1, y + half - 1], fill=WALL)
        else:
            draw.rectangle([x0, y - half, x1 - 1, y + half - 1], fill=WALL)

    def vertical(row: int, col: int, open_gap: bool) -> None:
        x = off + col * step
        y0 = off + row * step
        y1 = y0 + step
        if open_gap:
            mid = (y0 + y1) // 2
            draw.rectangle([x - half, y0, x + half - 1, mid - gap // 2], fill=WALL)
            draw.rectangle([x - half, mid + gap // 2, x + half - 1, y1 - 1], fill=WALL)
        else:
            draw.rectangle([x - half, y0, x + half - 1, y1 - 1], fill=WALL)

    for row in range(cells):
        for col in range(cells + 1):
            if col == 0 or col == cells:
                vertical(row, col, False)
            else:
                vertical(row, col, _edge_key((row, col - 1), (row, col)) in doorways)
    for col in range(cells):
        for row in range(cells + 1):
            if row == 0 or row == cells:
                horizontal(row, col, False)
            else:
                horizontal(row, col, _edge_key((row - 1, col), (row, col)) in doorways)
    for row in range(cells + 1):
        for col in range(cells + 1):
            x = off + col * step
            y = off + row * step
            draw.rectangle([x - half, y - half, x + half - 1, y + half - 1], fill=WALL)
    return image


# --------------------------------------------------------------------------- #
# analytic gold
# --------------------------------------------------------------------------- #
def _looped(scene: dict) -> list[str]:
    door_set = {tuple(int(v) for v in edge) for edge in scene["doorways"]}
    adjacency = _graph(door_set)
    if len(_reach(adjacency, (0, 0))) != CELLS * CELLS:
        raise ValueError("the free space is not one connected region")
    core = circuit_rooms(door_set)
    if len(core) < MIN_CIRCUIT:
        raise ValueError("the circuit is too short to read")
    out = []
    for colour in COLOUR_ORDER:
        room = (int(scene["rooms"][colour][0]), int(scene["rooms"][colour][1]))
        if room in core:
            out.append(colour)
    return out


def analytic_gold(scene: dict) -> str:
    winners = _looped(scene)
    if len(winners) != 1:
        raise ValueError(f"scene does not have exactly one circuit room: {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    gap = int(scene["gap_px"])
    stub = (int(scene["cell_px"]) - gap) // 2
    return float(min(gap, stub * 2))


def is_quarantined(scene: dict) -> bool:
    return int(scene["gap_px"]) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_doors = dict(scene)
    reversed_doors["doorways"] = list(reversed(scene["doorways"]))
    rotated = dict(scene)
    rotated["doorways"] = list(scene["doorways"][1:]) + [scene["doorways"][0]]
    reordered = dict(scene)
    reordered["draw_order"] = list(reversed(scene["draw_order"]))
    return [
        ("reverse_doorway_serialisation", reversed_doors),
        ("rotate_doorway_serialisation", rotated),
        ("reverse_floor_draw_order", reordered),
    ]
