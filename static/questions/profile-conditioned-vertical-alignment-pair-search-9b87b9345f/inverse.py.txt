"""Independent PNG-only inverse for the vertical-alignment pair search.

This module intentionally contains its own panel geometry and component
measurement.  It receives one raster image and never imports renderer code or
reads a scene record.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_SIZE = (900, 650)
_PANELS = ((35, 90, 425, 325), (475, 90, 865, 325), (35, 365, 425, 600), (475, 365, 865, 600))
_ALIGN_TOLERANCE = 4.0
_MIN_COMPONENT = 120


def _marker_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    # The marker is the only saturated blue object.  This mask intentionally
    # ignores the gray panel borders and labels.
    return (blue - red >= 30) & (green - red >= 18) & (blue >= 120)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for row in range(height):
        for col in range(width):
            if not mask[row, col] or seen[row, col]:
                continue
            stack = [(row, col)]
            seen[row, col] = True
            points: list[tuple[int, int]] = []
            while stack:
                y, x = stack.pop()
                points.append((y, x))
                for dy in (-1, 0, 1):
                    for dx in (-1, 0, 1):
                        if dx == 0 and dy == 0:
                            continue
                        yy, xx = y + dy, x + dx
                        if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                            seen[yy, xx] = True
                            stack.append((yy, xx))
            if len(points) >= _MIN_COMPONENT:
                mean_y = sum(point[0] + 0.5 for point in points) / len(points)
                mean_x = sum(point[1] + 0.5 for point in points) / len(points)
                found.append((mean_x, mean_y, len(points)))
    return found


def _panel_centers(mask: np.ndarray, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    left, top, right, bottom = box
    components = _components(mask[top:bottom, left:right])
    return [(x + left, y + top) for x, y, _size in components]


def _minimum_x_gap(centers: list[tuple[float, float]]) -> float:
    if len(centers) < 2:
        return math.inf
    return min(abs(first[0] - second[0]) for index, first in enumerate(centers) for second in centers[index + 1 :])


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B/C/D using only marker pixels in the supplied PNG."""

    if image.size != _SIZE:
        raise ValueError(f"expected {_SIZE} raster, received {image.size}")
    mask = _marker_mask(image)
    scores = []
    for box in _PANELS:
        centers = _panel_centers(mask, box)
        if len(centers) != 4:
            raise ValueError(f"expected four marker components in panel, got {len(centers)}")
        scores.append(_minimum_x_gap(centers))
    best = min(range(4), key=lambda index: (scores[index], index))
    return ("A", "B", "C", "D")[best]
