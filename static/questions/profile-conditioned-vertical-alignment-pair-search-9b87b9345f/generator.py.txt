"""Deterministic raster world for the vertical-alignment pair search."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw


WIDTH, HEIGHT = 900, 650
PANEL_BOXES = (
    (35, 90, 425, 325),
    (475, 90, 865, 325),
    (35, 365, 425, 600),
    (475, 365, 865, 600),
)
LABELS = ("A", "B", "C", "D")
ALIGN_TOLERANCE = 4.0
RUNNER_UP_GAP = 18.0
MIN_MARKER_DISTANCE = 34.0
MARKER_RADIUS = 14.0
SUPER_SAMPLE = 4

BACKGROUND = (246, 248, 250)
PANEL_FILL = (255, 255, 255)
PANEL_BORDER = (183, 192, 201)
MARKER_INK = (36, 92, 154)
LABEL_INK = (72, 80, 88)


def _pairwise_min_x(markers: list[list[float]]) -> float:
    values = [abs(float(a[0]) - float(b[0])) for i, a in enumerate(markers) for b in markers[i + 1 :]]
    return min(values) if values else float("inf")


def _pairwise_min_distance(markers: list[list[float]]) -> float:
    values = [
        math.hypot(float(a[0]) - float(b[0]), float(a[1]) - float(b[1]))
        for i, a in enumerate(markers)
        for b in markers[i + 1 :]
    ]
    return min(values) if values else float("inf")


def _valid_markers(markers: list[list[float]]) -> bool:
    return (
        len(markers) == 4
        and all(len(point) == 2 for point in markers)
        and all(54 <= float(point[0]) <= 336 and 58 <= float(point[1]) <= 204 for point in markers)
        and _pairwise_min_distance(markers) >= MIN_MARKER_DISTANCE
    )


def _sample_unaligned(rng: random.Random) -> list[list[int]]:
    """Sample four separated points with no close x pair."""

    for _ in range(5000):
        points: list[list[int]] = []
        for _point in range(4):
            accepted = False
            for _attempt in range(300):
                candidate = [rng.randint(62, 328), rng.randint(70, 190)]
                if all(abs(candidate[0] - other[0]) >= 28 for other in points) and all(
                    math.hypot(candidate[0] - other[0], candidate[1] - other[1]) >= MIN_MARKER_DISTANCE
                    for other in points
                ):
                    points.append(candidate)
                    accepted = True
                    break
            if not accepted:
                break
        if len(points) == 4 and _pairwise_min_x(points) > ALIGN_TOLERANCE:
            return points
    raise RuntimeError("could not sample an unaligned panel")


def _sample_aligned(rng: random.Random) -> list[list[int]]:
    """Sample four separated points with exactly one vertically aligned pair."""

    for _ in range(5000):
        x0 = rng.randint(78, 312)
        y1 = rng.randint(70, 120)
        y2 = rng.randint(148, 190)
        if abs(y2 - y1) < 55:
            continue
        points: list[list[int]] = [[x0, y1], [x0, y2]]
        for _point in range(2):
            accepted = False
            for _attempt in range(500):
                candidate = [rng.randint(62, 328), rng.randint(70, 190)]
                if abs(candidate[0] - x0) < 28:
                    continue
                if any(
                    abs(candidate[0] - other[0]) < 28
                    or math.hypot(candidate[0] - other[0], candidate[1] - other[1]) < MIN_MARKER_DISTANCE
                    for other in points
                ):
                    continue
                points.append(candidate)
                accepted = True
                break
            if not accepted:
                break
        if len(points) == 4 and _pairwise_min_x(points) <= ALIGN_TOLERANCE:
            return points
    raise RuntimeError("could not sample an aligned panel")


def _sample_scene_rng(rng: random.Random) -> dict[str, Any]:
    target = rng.randrange(4)
    panels = []
    for index in range(4):
        markers = _sample_aligned(rng) if index == target else _sample_unaligned(rng)
        rng.shuffle(markers)
        panels.append({"markers": markers})
    return {"panels": panels}


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one deterministic scene from an integer seed."""

    return _sample_scene_rng(random.Random(int(seed)))


def _scene_scores(scene: dict[str, Any]) -> list[float]:
    return [_pairwise_min_x(panel["markers"]) for panel in scene["panels"]]


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the panel with the smallest pairwise horizontal separation."""

    scores = _scene_scores(scene)
    best = min(range(len(scores)), key=lambda index: (scores[index], index))
    return LABELS[best]


def margin(scene: dict[str, Any]) -> float:
    """Runner-up gap in pixels between the two best panel scores."""

    scores = sorted(_scene_scores(scene))
    if len(scores) < 2:
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject ambiguous or poorly separated finite-resolution arrangements."""

    try:
        panels = scene["panels"]
        if len(panels) != 4 or any(not _valid_markers(panel["markers"]) for panel in panels):
            return True
        scores = _scene_scores(scene)
        matching = sum(score <= ALIGN_TOLERANCE for score in scores)
        return matching != 1 or margin(scene) < RUNNER_UP_GAP
    except (KeyError, TypeError, ValueError):
        return True


def _draw_marker(draw: ImageDraw.ImageDraw, center: list[float], origin: tuple[int, int], scale: int) -> None:
    x = (origin[0] + float(center[0])) * scale
    y = (origin[1] + float(center[1])) * scale
    radius = MARKER_RADIUS * scale
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=MARKER_INK)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a scene as one deterministic RGB PNG-sized image."""

    scale = SUPER_SAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for index, (left, top, right, bottom) in enumerate(PANEL_BOXES):
        draw.rectangle(
            (left * scale, top * scale, right * scale, bottom * scale),
            fill=PANEL_FILL,
            outline=PANEL_BORDER,
            width=2 * scale,
        )
        origin = (left, top)
        for marker in scene["panels"][index]["markers"]:
            _draw_marker(draw, marker, origin, scale)
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    labels = ImageDraw.Draw(image)
    try:
        from PIL import ImageFont

        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 18)
    except (OSError, IOError):
        font = None
    for label, (left, top, _right, _bottom) in zip(LABELS, PANEL_BOXES):
        labels.text((left + 14, top + 12), label, fill=LABEL_INK, font=font)
    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Declare a nontrivial draw-order permutation that preserves the raster."""

    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["markers"] = list(reversed(panel["markers"]))
    return [("reverse_marker_order", twin)]
