"""Deterministic renderer and analytic gold for nested-loop parity."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 900
HEIGHT = 700
SCALE = 4
BACKGROUND = (250, 251, 253)
LOOP_INK = (38, 48, 61)
MARKER_COLORS = {
    "red": (214, 66, 77),
    "blue": (49, 112, 205),
    "green": (48, 154, 99),
}


def sample_scene(seed: int) -> dict:
    """Sample a clear nested-loop scene with both parity classes reachable."""

    rng = random.Random(int(seed))
    ring_count = rng.choice((3, 4, 5))
    center = [
        round(435.0 + rng.uniform(0.0, 30.0), 3),
        round(342.0 + rng.uniform(0.0, 16.0), 3),
    ]
    outer = rng.uniform(232.0, 250.0)
    gap = rng.uniform(45.0, 52.0)
    radii = [round(outer - gap * (ring_count - 1 - i), 3) for i in range(ring_count)]
    stroke = round(rng.uniform(8.0, 10.0), 3)

    # Choose the parity first, then a depth having that parity.  This keeps
    # stress batches balanced without tying the answer to marker colour or
    # direction.
    desired_parity = rng.randrange(2)
    depths = [depth for depth in range(ring_count + 1) if depth % 2 == desired_parity]
    depth = rng.choice(depths)
    if depth == 0:
        distance = radii[-1] + rng.uniform(27.0, 38.0)
    elif depth == ring_count:
        distance = rng.uniform(12.0, max(13.0, radii[0] - 22.0))
    else:
        inner_boundary = radii[ring_count - depth - 1]
        outer_boundary = radii[ring_count - depth]
        distance = inner_boundary + rng.uniform(18.0, outer_boundary - inner_boundary - 18.0)
    angle = rng.uniform(-math.pi, math.pi)
    marker = {
        "x": round(center[0] + distance * math.cos(angle), 3),
        "y": round(center[1] + distance * math.sin(angle), 3),
        "color": rng.choice(tuple(MARKER_COLORS)),
        "radius": round(rng.uniform(12.0, 16.0), 3),
    }
    return {
        "rings": {
            "center": center,
            "radii": radii,
            "stroke": stroke,
        },
        "marker": marker,
    }


def render(scene: dict) -> Image.Image:
    """Render only visible loop and marker geometry; no latent state is embedded."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    rings = scene["rings"]
    center = rings["center"]
    cx, cy = float(center[0]), float(center[1])
    stroke = max(1, int(round(float(rings["stroke"]) * SCALE)))
    for radius in rings["radii"]:
        r = float(radius)
        box = (
            int(round((cx - r) * SCALE)),
            int(round((cy - r) * SCALE)),
            int(round((cx + r) * SCALE)),
            int(round((cy + r) * SCALE)),
        )
        draw.ellipse(box, outline=LOOP_INK, width=stroke)

    marker = scene["marker"]
    mx, my = float(marker["x"]), float(marker["y"])
    mr = float(marker["radius"])
    color = MARKER_COLORS[str(marker["color"])]
    box = (
        int(round((mx - mr) * SCALE)),
        int(round((my - mr) * SCALE)),
        int(round((mx + mr) * SCALE)),
        int(round((my + mr) * SCALE)),
    )
    draw.ellipse(box, fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The point marker is enclosed by a loop when its centre is inside it."""

    rings = scene["rings"]
    marker = scene["marker"]
    cx, cy = (float(value) for value in rings["center"])
    distance = math.hypot(float(marker["x"]) - cx, float(marker["y"]) - cy)
    depth = sum(distance < float(radius) for radius in rings["radii"])
    return "yes" if depth % 2 else "no"


def margin(scene: dict) -> float:
    """Distance from the marker centre to the nearest loop boundary in pixels."""

    if is_quarantined(scene):
        return 0.0
    rings = scene["rings"]
    marker = scene["marker"]
    cx, cy = (float(value) for value in rings["center"])
    distance = math.hypot(float(marker["x"]) - cx, float(marker["y"]) - cy)
    return min(abs(distance - float(radius)) for radius in rings["radii"])


def is_quarantined(scene: dict) -> bool:
    """Reject clipped, touching, malformed, or raster-ambiguous scenes."""

    try:
        rings = scene["rings"]
        marker = scene["marker"]
        center = tuple(float(value) for value in rings["center"])
        radii = tuple(float(value) for value in rings["radii"])
        cx, cy = center
        stroke = float(rings["stroke"])
        mx, my = float(marker["x"]), float(marker["y"])
        mr = float(marker["radius"])
        if not 3 <= len(radii) <= 5 or any(radius <= 0 for radius in radii):
            return True
        if tuple(sorted(radii)) != radii or any(b - a < 32.0 for a, b in zip(radii, radii[1:])):
            return True
        if not 5.0 <= stroke <= 14.0 or not 8.0 <= mr <= 22.0:
            return True
        if not (80.0 < cx < WIDTH - 80.0 and 80.0 < cy < HEIGHT - 80.0):
            return True
        if cx - radii[-1] < 35.0 or cx + radii[-1] > WIDTH - 35.0:
            return True
        if cy - radii[-1] < 35.0 or cy + radii[-1] > HEIGHT - 35.0:
            return True
        if not (mr + 3.0 < mx < WIDTH - mr - 3.0 and mr + 3.0 < my < HEIGHT - mr - 3.0):
            return True
        distance = math.hypot(mx - cx, my - cy)
        if min(abs(distance - radius) for radius in radii) < 16.0:
            return True
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden render-invariant label aliases in this encoding."""

    return []
