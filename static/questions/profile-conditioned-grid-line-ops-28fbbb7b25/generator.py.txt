"""Scene sampler, renderer, analytic gold, margin and latent symmetries for grid_line_ops.

A 4x4 grid of cells is drawn once, each cell either dark-filled or empty. One
cell is marked with a red ring. To the right, an ordered row of operation cards
is drawn; each card shows a small 4x4 diagram in which exactly one whole row or
one whole column is coloured. Green means "fill that whole line"; magenta means
"flip every cell of that whole line". The cards are applied one at a time in
drawn left-to-right order to the grid. The resulting grid is never drawn, so the
answer is a consequence of replaying the program, not a measurement of any drawn
configuration.

The scene stores the cells and the operation cards as unordered bags; the drawn
order of the cards comes from each card's own ``slot`` field, and the marked cell
is recovered from the drawn ring geometry. Gold therefore depends on what the
image shows and not on Python list position, which ``latent_symmetries``
exercises directly.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "grid_line_ops-renderer-0.1.0"

WHITE = (255, 255, 255)
NAVY = (28, 52, 120)
PALE = (236, 236, 236)
GRID_EDGE = (90, 90, 90)
RED = (214, 38, 38)
CARD_BG = (252, 246, 228)
MINI_EDGE = (152, 152, 152)
SET_C = (26, 150, 72)
TOG_C = (200, 36, 150)
BLACK = (15, 15, 15)

N = 4
CELL = 62
GAP = 12
PITCH = CELL + GAP  # 74
ORIGIN = (40, 58)

CARD = 78
CGAP = 14
CARD_ORIGIN = (390, 150)
MCELL = 12
MGAP = 4
MPITCH = MCELL + MGAP  # 16
MOFF = 9

RING_R = 21
RING_W = 4
CANVAS = (880, 400)
N_OPS = 5
QUARANTINE_MARGIN = 6.0


# --------------------------------------------------------------------------- geometry


def _cell_center(r: int, c: int) -> tuple[float, float]:
    return (
        ORIGIN[0] + c * PITCH + CELL / 2.0,
        ORIGIN[1] + r * PITCH + CELL / 2.0,
    )


def _ring_center(scene: dict) -> tuple[float, float]:
    ring = scene["ring"]
    cx, cy = _cell_center(int(ring["r"]), int(ring["c"]))
    return cx + float(ring["dx"]), cy + float(ring["dy"])


def _ring_cell(scene: dict) -> tuple[int, int]:
    """The marked cell is the drawn cell whose centre is nearest the ring centre."""
    rx, ry = _ring_center(scene)
    best = None
    for r in range(N):
        for c in range(N):
            cx, cy = _cell_center(r, c)
            d = math.hypot(rx - cx, ry - cy)
            if best is None or d < best[0]:
                best = (d, r, c)
    return best[1], best[2]


def _ops_in_drawn_order(scene: dict) -> list[dict]:
    return sorted(scene["ops"], key=lambda o: int(o["slot"]))


def _initial_state(scene: dict, r: int, c: int) -> bool:
    for cell in scene["cells"]:
        if int(cell["r"]) == r and int(cell["c"]) == c:
            return bool(cell["on"])
    raise ValueError("scene has no cell at the marked position")


def _replay(ops: list[dict], state: bool, r: int, c: int) -> tuple[bool, int]:
    changes = 0
    for op in sorted(ops, key=lambda o: int(o["slot"])):
        hit = (op["axis"] == "row" and int(op["idx"]) == r) or (
            op["axis"] == "col" and int(op["idx"]) == c
        )
        if not hit:
            continue
        new = True if op["op"] == "set" else (not state)
        if new != state:
            changes += 1
        state = new
    return state, changes


# --------------------------------------------------------------------------- sampling


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    tight = rng.random() < 0.18
    want_final = rng.random() < 0.5
    want_init = rng.random() < 0.5

    accepted = None
    for _ in range(4000):
        r = rng.randrange(N)
        c = rng.randrange(N)
        others = [(i, j) for i in range(N) for j in range(N) if (i, j) != (r, c)]
        rng.shuffle(others)
        n_other_on = 8 - (1 if want_init else 0)
        on = set(others[:n_other_on])
        if want_init:
            on.add((r, c))

        k = rng.choice([1, 2, 3])
        slots = sorted(rng.sample(range(N_OPS), k))
        ops = []
        for s in range(N_OPS):
            axis = "row" if rng.random() < 0.5 else "col"
            forbidden = r if axis == "row" else c
            if s in slots:
                idx = forbidden
            else:
                idx = rng.choice([v for v in range(N) if v != forbidden])
            ops.append(
                {
                    "slot": s,
                    "axis": axis,
                    "idx": idx,
                    "op": "set" if rng.random() < 0.5 else "toggle",
                }
            )

        state, changes = _replay(ops, want_init, r, c)
        if state != want_final or changes < 1:
            continue
        # evidence sensitivity: erasing the last card that touches the marked
        # cell must move the final state
        without, _ = _replay([o for o in ops if int(o["slot"]) != slots[-1]], want_init, r, c)
        if without == state:
            continue
        accepted = (r, c, on, ops)
        break

    if accepted is None:
        raise RuntimeError(f"scene sampling failed for seed {seed}")
    r, c, on, ops = accepted

    if tight:
        m = rng.randrange(28, 36)
        dirs = []
        if r > 0:
            dirs.append((0, -m))
        if r < N - 1:
            dirs.append((0, m))
        if c > 0:
            dirs.append((-m, 0))
        if c < N - 1:
            dirs.append((m, 0))
    else:
        m = rng.randrange(0, 6)
        dirs = [(m, 0), (-m, 0), (0, m), (0, -m)]
    dx, dy = dirs[rng.randrange(len(dirs))]

    cells = [{"r": i, "c": j, "on": (i, j) in on} for i in range(N) for j in range(N)]
    rng.shuffle(cells)
    bag = list(ops)
    rng.shuffle(bag)

    return {
        "canvas": list(CANVAS),
        "origin": list(ORIGIN),
        "cards_origin": list(CARD_ORIGIN),
        "n_ops": N_OPS,
        "cells": cells,
        "ring": {"r": r, "c": c, "dx": dx, "dy": dy},
        "ops": bag,
    }


# --------------------------------------------------------------------------- gold


def analytic_gold(scene: dict) -> str:
    r, c = _ring_cell(scene)
    state = _initial_state(scene, r, c)
    state, changes = _replay(scene["ops"], state, r, c)
    return f"{'filled' if state else 'empty'}|{changes}"


def margin(scene: dict) -> float:
    rx, ry = _ring_center(scene)
    dists = sorted(
        math.hypot(rx - _cell_center(r, c)[0], ry - _cell_center(r, c)[1])
        for r in range(N)
        for c in range(N)
    )
    ring_sep = (dists[1] - dists[0]) / 2.0
    # clearance between the drawn ring stroke and every cell centre, so that the
    # dark/empty reading of a cell is never taken on top of the marker
    fill_clear = min(abs(d - RING_R) for d in dists)
    # half-pitch of the card diagrams, rescaled to main-grid pixels so that the
    # row/column index read on a card is comparable with the ring reading
    card_sep = (MPITCH / 2.0) * (PITCH / MPITCH)
    return round(min(ring_sep, fill_clear, card_sep), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Bag re-orderings that must redraw the identical raster and keep gold fixed."""
    ops = list(scene["ops"])
    cells = list(scene["cells"])
    out = [
        ("ops_reversed", dict(scene, ops=list(reversed(ops)))),
        ("ops_rotated", dict(scene, ops=ops[2:] + ops[:2])),
        ("cells_reversed", dict(scene, cells=list(reversed(cells)))),
        ("cells_rotated", dict(scene, cells=cells[5:] + cells[:5])),
    ]
    return out


# --------------------------------------------------------------------------- render


def render(scene: dict) -> Image.Image:
    w, h = int(scene["canvas"][0]), int(scene["canvas"][1])
    img = Image.new("RGB", (w, h), WHITE)
    d = ImageDraw.Draw(img)
    ox, oy = int(scene["origin"][0]), int(scene["origin"][1])

    for cell in sorted(scene["cells"], key=lambda x: (int(x["r"]), int(x["c"]))):
        r, c = int(cell["r"]), int(cell["c"])
        x0 = ox + c * PITCH
        y0 = oy + r * PITCH
        d.rectangle(
            [x0, y0, x0 + CELL - 1, y0 + CELL - 1],
            fill=NAVY if cell["on"] else PALE,
            outline=GRID_EDGE,
            width=3,
        )

    rx, ry = _ring_center(scene)
    d.ellipse(
        [rx - RING_R, ry - RING_R, rx + RING_R, ry + RING_R],
        outline=RED,
        width=RING_W,
    )

    cx0, cy0 = int(scene["cards_origin"][0]), int(scene["cards_origin"][1])
    for op in _ops_in_drawn_order(scene):
        s = int(op["slot"])
        bx = cx0 + s * (CARD + CGAP)
        by = cy0
        d.rectangle([bx, by, bx + CARD - 1, by + CARD - 1], fill=CARD_BG)
        colour = SET_C if op["op"] == "set" else TOG_C
        idx = int(op["idx"])
        for i in range(N):
            for j in range(N):
                mx = bx + MOFF + j * MPITCH
                my = by + MOFF + i * MPITCH
                d.rectangle(
                    [mx, my, mx + MCELL - 1, my + MCELL - 1],
                    outline=MINI_EDGE,
                    width=2,
                )
                hit = (op["axis"] == "row" and i == idx) or (op["axis"] == "col" and j == idx)
                if hit:
                    d.rectangle([mx + 2, my + 2, mx + MCELL - 3, my + MCELL - 3], fill=colour)

    ay = cy0 + CARD + 34
    ax0 = cx0
    ax1 = cx0 + (int(scene["n_ops"]) - 1) * (CARD + CGAP) + CARD - 1
    d.line([ax0, ay, ax1 - 14, ay], fill=BLACK, width=4)
    d.polygon([(ax1, ay), (ax1 - 18, ay - 9), (ax1 - 18, ay + 9)], fill=BLACK)
    return img
