"""Scene-free pixel inverse arm for grid_line_ops.

Receives only the rendered RGB image. It re-discovers the main grid from its
drawn cell borders, re-discovers the marked cell from the red ring, reads that
cell's drawn dark/empty state, re-discovers the operation cards from their cream
backgrounds in drawn left-to-right order, reads each card's operation colour and
the row or column its coloured line occupies, replays the program on the marked
cell, and returns the final state together with the number of cards that changed
it.

It imports no renderer, prompt, generator, verifier or gold code and never sees
latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "grid_line_ops-oracle-0.1.0"

_GRID_EDGE = (90, 90, 90)
_MINI_EDGE = (152, 152, 152)
_CARD_BG = (252, 246, 228)
_RED = (214, 38, 38)
_SET = (26, 150, 72)
_TOG = (200, 36, 150)
_DARK = (28, 52, 120)
_LIGHT = (236, 236, 236)


def _mask(arr: np.ndarray, ref, tol: int) -> np.ndarray:
    return (
        (np.abs(arr[:, :, 0] - ref[0]) <= tol)
        & (np.abs(arr[:, :, 1] - ref[1]) <= tol)
        & (np.abs(arr[:, :, 2] - ref[2]) <= tol)
    )


def _runs(flags: np.ndarray, min_len: int) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, on in enumerate(flags):
        if on and start is None:
            start = i
        elif not on and start is not None:
            if i - start >= min_len:
                out.append((start, i - 1))
            start = None
    if start is not None and len(flags) - start >= min_len:
        out.append((start, len(flags) - 1))
    return out


def _centres(runs: list[tuple[int, int]]) -> list[float]:
    return [(a + b) / 2.0 for a, b in runs]


def _nearest(value: float, centres: list[float]) -> int:
    return min(range(len(centres)), key=lambda i: abs(value - centres[i]))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int32)

    # 1. main grid: the only ink drawn in the cell-border grey
    grid = _mask(arr, _GRID_EDGE, 12)
    cols = _runs(grid.any(axis=0), min_len=20)
    rows = _runs(grid.any(axis=1), min_len=20)
    if len(cols) != 4 or len(rows) != 4:
        return "unreadable|0"
    col_c = _centres(cols)
    row_c = _centres(rows)

    # 2. marked cell: nearest cell centre to the red ring's centre
    red = _mask(arr, _RED, 40)
    ys, xs = np.nonzero(red)
    if ys.size < 60:
        return "unreadable|0"
    ring_x = (float(xs.min()) + float(xs.max())) / 2.0
    ring_y = (float(ys.min()) + float(ys.max())) / 2.0
    mr = _nearest(ring_y, row_c)
    mc = _nearest(ring_x, col_c)

    # 3. its drawn state, read from a small patch at the cell centre
    py, px = int(round(row_c[mr])), int(round(col_c[mc]))
    patch = arr[py - 3 : py + 4, px - 3 : px + 4, :].reshape(-1, 3)
    if patch.shape[0] == 0:
        return "unreadable|0"
    mean = patch.mean(axis=0)
    d_dark = float(np.sum((mean - np.array(_DARK)) ** 2))
    d_light = float(np.sum((mean - np.array(_LIGHT)) ** 2))
    state = d_dark < d_light

    # 4. operation cards, in drawn left-to-right order
    card = _mask(arr, _CARD_BG, 8)
    card_cols = _runs(card.any(axis=0), min_len=30)
    if not card_cols:
        return "unreadable|0"
    card_rows = _runs(card.any(axis=1), min_len=30)
    if len(card_rows) != 1:
        return "unreadable|0"
    ry0, ry1 = card_rows[0]

    changes = 0
    for cx0, cx1 in card_cols:
        sub = arr[ry0 : ry1 + 1, cx0 : cx1 + 1, :]
        set_m = _mask(sub, _SET, 30)
        tog_m = _mask(sub, _TOG, 30)
        n_set, n_tog = int(set_m.sum()), int(tog_m.sum())
        if max(n_set, n_tog) < 80:
            return "unreadable|0"
        is_set = n_set > n_tog
        line = set_m if is_set else tog_m

        mini = _mask(sub, _MINI_EDGE, 12)
        mcols = _runs(mini.any(axis=0), min_len=4)
        mrows = _runs(mini.any(axis=1), min_len=4)
        if len(mcols) != 4 or len(mrows) != 4:
            return "unreadable|0"

        lys, lxs = np.nonzero(line)
        w = float(lxs.max() - lxs.min())
        h = float(lys.max() - lys.min())
        if abs(w - h) < 6:
            return "unreadable|0"
        if w > h:  # a whole row of the card diagram is coloured
            idx = _nearest((float(lys.min()) + float(lys.max())) / 2.0, _centres(mrows))
            hit = idx == mr
        else:  # a whole column is coloured
            idx = _nearest((float(lxs.min()) + float(lxs.max())) / 2.0, _centres(mcols))
            hit = idx == mc
        if not hit:
            continue
        new = True if is_set else (not state)
        if new != state:
            changes += 1
        state = new

    return f"{'filled' if state else 'empty'}|{changes}"
