"""Independent PNG-only inverse for the hollow chamber world.

This module intentionally contains no renderer or scene imports.  It detects
the colored foreground from the raster, flood-fills the exterior background,
selects the largest enclosed complement component, and compares its pixel mass
above and below the colored shell's horizontal center.  A damaged contour is
rejected instead of being guessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _background_color(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _exterior_background(background: np.ndarray) -> np.ndarray:
    """Return background pixels reachable from the image border."""

    height, width = background.shape
    exterior = np.zeros((height, width), dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if background[0, x]:
            exterior[0, x] = True
            queue.append((0, x))
        if background[height - 1, x] and not exterior[height - 1, x]:
            exterior[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(1, height - 1):
        if background[y, 0] and not exterior[y, 0]:
            exterior[y, 0] = True
            queue.append((y, 0))
        if background[y, width - 1] and not exterior[y, width - 1]:
            exterior[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width:
                if background[ny, nx] and not exterior[ny, nx]:
                    exterior[ny, nx] = True
                    queue.append((ny, nx))
    return exterior


def _largest_enclosed_component(mask: np.ndarray) -> tuple[np.ndarray, int]:
    """Extract the largest enclosed component and count all enclosed voids."""

    remaining = mask.copy()
    best: list[tuple[int, int]] = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(remaining)):
        if not remaining[y, x]:
            continue
        queue = [(int(y), int(x))]
        remaining[y, x] = False
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and remaining[ny, nx]:
                    remaining[ny, nx] = False
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    result = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        result[np.asarray(ys), np.asarray(xs)] = True
    return result, int(1 if best else 0)


def _enclosed_components(mask: np.ndarray) -> tuple[np.ndarray, int]:
    """Return the largest void plus the number of void components.

    A clean scene has exactly one enclosed complement component.  A white tile
    cut out of the thick shell creates a second enclosed void; rejecting that
    damaged contour makes the oracle's structural support broad and explicit.
    """

    remaining = mask.copy()
    best: list[tuple[int, int]] = []
    count = 0
    height, width = mask.shape
    for y, x in zip(*np.nonzero(remaining)):
        if not remaining[y, x]:
            continue
        count += 1
        queue = [(int(y), int(x))]
        remaining[y, x] = False
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and remaining[ny, nx]:
                    remaining[ny, nx] = False
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    result = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        result[np.asarray(ys), np.asarray(xs)] = True
    return result, count


def _chamber_mask(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    background_rgb = _background_color(array)
    distance = np.linalg.norm(array.astype(np.float32) - background_rgb, axis=2)
    foreground = distance > 28.0
    if int(foreground.sum()) < 8000:
        raise ValueError("insufficient colored shell")
    ys, xs = np.nonzero(foreground)
    if int(xs.max() - xs.min()) < 170 or int(ys.max() - ys.min()) < 145:
        raise ValueError("shell contour is incomplete")
    # The cavity label is only meaningful when the whole outer contour is
    # present.  Probe many angular rays through the broad shell; a destructive
    # grid tile that removes an outer arc must make the oracle abstain too.
    center_x = 0.5 * (float(xs.min()) + float(xs.max()))
    center_y = 0.5 * (float(ys.min()) + float(ys.max()))
    radius_x = 0.5 * float(xs.max() - xs.min())
    radius_y = 0.5 * float(ys.max() - ys.min())
    for index in range(48):
        theta = 2.0 * np.pi * index / 48.0
        samples = 0
        for radial in np.linspace(0.88, 1.015, 20):
            x = int(round(center_x + radius_x * radial * np.cos(theta)))
            y = int(round(center_y + radius_y * radial * np.sin(theta)))
            if 0 <= y < foreground.shape[0] and 0 <= x < foreground.shape[1]:
                samples += int(foreground[y, x])
        if samples < 5:
            raise ValueError("outer contour has a missing angular sector")
    exterior = _exterior_background(~foreground)
    enclosed = (~foreground) & ~exterior
    chamber, enclosed_count = _enclosed_components(enclosed)
    if enclosed_count != 1:
        raise ValueError("contour has an extra enclosed void")
    if int(chamber.sum()) < 6000:
        raise ValueError("no stable enclosed chamber")
    chamber_y, chamber_x = np.nonzero(chamber)
    # Reject a false interior component that is not actually inside the shell.
    if chamber_x.min() <= xs.min() + 8 or chamber_x.max() >= xs.max() - 8:
        raise ValueError("chamber is not enclosed laterally")
    if chamber_y.min() <= ys.min() + 8 or chamber_y.max() >= ys.max() - 8:
        raise ValueError("chamber is not enclosed vertically")
    return chamber, foreground


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff the upper cavity half contains more pixels than the lower."""

    chamber, foreground = _chamber_mask(image)
    ys, xs = np.nonzero(foreground)
    center_y = 0.5 * (float(ys.min()) + float(ys.max()))
    chamber_y, _ = np.nonzero(chamber)
    upper = int(np.count_nonzero(chamber_y < center_y))
    lower = int(np.count_nonzero(chamber_y > center_y))
    total = upper + lower
    if total < 6000 or abs(upper - lower) / total < 0.018:
        raise ValueError("cavity-area difference is below pixel confidence")
    return "yes" if upper > lower else "no"
