"""Renderer and analytic scene contract for the hollow chamber world.

The foreground is one broad, filled shell.  Its inner boundary has a smooth
signed vertical swell.  Positive swell makes the upper half of the enclosed
white chamber larger; negative swell makes the lower half larger.  All scene
fields are JSON-friendly so the manifest is also the analytic audit record.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SUPER = 4
BACKGROUND = (255, 255, 255)
PALETTE: tuple[tuple[int, int, int], ...] = (
    (35, 82, 145),
    (166, 57, 64),
    (48, 122, 83),
    (116, 70, 151),
    (205, 112, 34),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene; seed controls nuisances as well as label sign."""

    rng = random.Random(int(seed))
    sign = 1.0 if rng.random() < 0.5 else -1.0
    magnitude = rng.uniform(0.12, 0.30)
    return {
        "bias": round(sign * magnitude, 6),
        "cx": round(rng.uniform(172.0, 212.0), 4),
        "cy": round(rng.uniform(172.0, 212.0), 4),
        "scale": round(rng.uniform(0.90, 1.02), 5),
        "ink": list(rng.choice(PALETTE)),
    }


def _ellipse_points(cx: float, cy: float, rx: float, ry: float, count: int = 360):
    return [
        (cx + rx * math.cos(2.0 * math.pi * i / count),
         cy + ry * math.sin(2.0 * math.pi * i / count))
        for i in range(count)
    ]


def _inner_points(scene: dict[str, Any], count: int = 360):
    """Create the asymmetric but smooth inner contour in image coordinates."""

    cx = float(scene["cx"])
    cy = float(scene["cy"])
    scale = float(scene["scale"])
    bias = float(scene["bias"])
    # A small but substantial chamber leaves one thick, area-like shell.  The
    # shell is intentionally broad so destructive grid cells cover one
    # connected support region rather than four thin arc fragments.
    rx = 70.0 * scale
    ry = 54.0 * scale
    points = []
    for i in range(count):
        theta = 2.0 * math.pi * i / count
        sine = math.sin(theta)
        # Image y increases downward.  A positive bias enlarges the top.
        vertical_swell = 1.0 - bias * sine
        points.append(
            (cx + rx * math.cos(theta), cy + ry * sine * vertical_swell)
        )
    return points


def _scaled(points, factor: int):
    return [(round(x * factor), round(y * factor)) for x, y in points]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a single connected RGB shell with an enclosed white chamber."""

    scale = float(scene["scale"])
    outer = _ellipse_points(
        float(scene["cx"]),
        float(scene["cy"]),
        154.0 * scale,
        133.0 * scale,
    )
    image = Image.new("RGB", (WIDTH * SUPER, HEIGHT * SUPER), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.polygon(_scaled(outer, SUPER), fill=tuple(scene["ink"]))
    # The cutout is painted after the shell so it is a true negative-space
    # component rather than a second disconnected foreground object.
    draw.polygon(_scaled(_inner_points(scene), SUPER), fill=BACKGROUND)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """The declared label is the sign of the rendered inner-contour swell."""

    return "yes" if float(scene["bias"]) > 0.0 else "no"


def margin(scene: dict[str, Any]) -> float:
    """Signed distance from the balanced-cavity boundary in normalized units."""

    return float(scene["bias"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return abs(float(scene["bias"])) < 0.08


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No distinct latent aliases are claimed for this continuous scene."""

    return []
