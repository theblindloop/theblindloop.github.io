"""Deterministic renderer and analytic decision for three_route_terminal."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 840
HEIGHT = 360
COLORS = ("red", "blue", "orange")
RGB = {
    "red": (224, 58, 62),
    "blue": (55, 116, 224),
    "orange": (238, 143, 47),
    "green": (36, 166, 91),
    "node": (207, 219, 231),
    "ink": (25, 28, 33),
    "border": (190, 198, 207),
    "background": (255, 255, 255),
}


def _simple_path_count(nodes: list[dict], edges: list[list[str]]) -> int:
    """Count undirected START-to-terminal paths that never revisit a node."""
    adjacency = {node["id"]: set() for node in nodes}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    start = next(node["id"] for node in nodes if node["role"] == "start")
    terminal = next(node["id"] for node in nodes if node["role"] == "terminal")

    def visit(current: str, seen: frozenset[str]) -> int:
        if current == terminal:
            return 1
        return sum(
            visit(neighbor, seen | {neighbor})
            for neighbor in adjacency[current]
            if neighbor not in seen
        )

    return visit(start, frozenset({start}))


def _blueprint(kind: str) -> tuple[dict[str, tuple[int, int]], list[list[str]]]:
    """Return one of three nine-node, ten-link planar graphs with 2/3/4 routes."""
    if kind == "two":
        coords = {
            "s": (24, 180), "a": (68, 124), "b": (68, 236),
            "x": (112, 180), "y": (156, 180), "c": (202, 180),
            "t": (240, 180), "d": (156, 94), "z": (202, 94),
        }
        edges = [
            ["s", "a"], ["s", "b"], ["a", "x"], ["b", "x"],
            ["x", "y"], ["y", "c"], ["c", "t"],
            ["y", "d"], ["d", "z"], ["z", "y"],
        ]
    elif kind == "three":
        coords = {
            "s": (24, 180), "a": (72, 100), "b": (76, 180),
            "c": (72, 260), "x": (128, 180), "y": (184, 180),
            "t": (240, 180), "d": (128, 86), "z": (184, 86),
        }
        edges = [
            ["s", "a"], ["a", "x"], ["s", "b"], ["b", "x"],
            ["s", "c"], ["c", "x"], ["x", "y"], ["y", "t"],
            ["x", "d"], ["d", "z"],
        ]
    elif kind == "four":
        coords = {
            "s": (24, 180), "a": (68, 124), "b": (68, 236),
            "x": (124, 180), "c": (180, 124), "d": (180, 236),
            "t": (240, 180), "y": (124, 76), "z": (180, 76),
        }
        edges = [
            ["s", "a"], ["s", "b"], ["a", "x"], ["b", "x"],
            ["x", "c"], ["x", "d"], ["c", "t"], ["d", "t"],
            ["x", "y"], ["y", "z"],
        ]
    else:
        raise ValueError(f"unknown graph kind {kind!r}")
    return coords, edges


def _base_panel(kind: str, color: str, slot: int, mirror_x: bool, mirror_y: bool) -> dict:
    coords, edges = _blueprint(kind)
    x_offset = 10 + slot * 275
    nodes = []
    for node_id, (x, y) in coords.items():
        if mirror_x:
            x = 264 - x
        if mirror_y:
            y = 360 - y
        role = "start" if node_id == "s" else "terminal" if node_id == "t" else "node"
        fill = "green" if role == "start" else color if role == "terminal" else "node"
        nodes.append({"id": node_id, "x": x_offset + x, "y": y, "role": role, "fill": fill})
    return {"slot": slot, "kind": kind, "terminal_color": color, "nodes": nodes, "edges": edges}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = COLORS[int(seed) % 3]
    target_slot = (int(seed) // 3) % 3
    remaining_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining_colors)
    loser_kinds = ["two", "four"]
    rng.shuffle(loser_kinds)
    panels = []
    for slot in range(3):
        if slot == target_slot:
            kind = "three"
            color = answer_color
        else:
            kind = loser_kinds.pop()
            color = remaining_colors.pop()
        panels.append(
            _base_panel(kind, color, slot, bool(rng.getrandbits(1)), bool(rng.getrandbits(1)))
        )
    return {"width": WIDTH, "height": HEIGHT, "panels": panels}


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), RGB["background"])
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        slot = int(panel["slot"])
        x0 = (10 + slot * 275) * scale
        draw.rounded_rectangle(
            (x0 + 2 * scale, 16 * scale, x0 + 264 * scale, 344 * scale),
            radius=12 * scale,
            outline=RGB["border"],
            width=2 * scale,
        )
        lookup = {node["id"]: node for node in panel["nodes"]}
        for left, right in panel["edges"]:
            a, b = lookup[left], lookup[right]
            draw.line(
                (a["x"] * scale, a["y"] * scale, b["x"] * scale, b["y"] * scale),
                fill=RGB["ink"],
                width=7 * scale,
            )
        for node in panel["nodes"]:
            radius = 13
            box = (
                (node["x"] - radius) * scale,
                (node["y"] - radius) * scale,
                (node["x"] + radius) * scale,
                (node["y"] + radius) * scale,
            )
            draw.ellipse(box, fill=RGB[node["fill"]], outline=RGB["ink"], width=3 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _counts(scene: dict) -> list[int]:
    return [_simple_path_count(panel["nodes"], panel["edges"]) for panel in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    counts = _counts(scene)
    winners = [
        panel["terminal_color"]
        for panel, count in zip(scene["panels"], counts, strict=True)
        if count == 3
    ]
    if sorted(counts) != [2, 3, 4] or len(winners) != 1 or winners[0] not in COLORS:
        raise ValueError(f"expected panel route counts [2, 3, 4], found {counts}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted((-abs(count - 3) for count in _counts(scene)), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    counts = _counts(scene)
    return sorted(counts) != [2, 3, 4] or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"].reverse()
    return [("reverse_edge_storage", twin)]
