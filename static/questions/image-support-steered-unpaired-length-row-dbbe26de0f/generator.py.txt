"""Unpaired-length row world.

Two small detached panels sit apart on one pale sheet with a wide empty
corridor between them: the left panel holds four solid dark bars, the right
panel holds three. Every right bar has exactly one left bar of exactly the same
drawn length; one left bar has no partner at all. The decision is which left row
is the unpaired one, so both panels are decisive and neither panel alone carries
the answer.

Latent state z is a pure drawing description: canvas size, ink colour, the two
panel rectangles, the pale guide furniture, the seven bar rectangles (left edge,
top edge, own length, thickness), and the decoration marks outside the panels.
Nothing in z stores the answer: the gold is recomputed from the bar rectangles
by a length-matching pass.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "unpaired_length_row-0.2.0"

CANVAS = (760, 460)
BG = (246, 245, 240)
INK = (24, 32, 58)
FRAME = (205, 207, 212)
GUIDE = (223, 224, 228)

# The two decisive panels. They are fixed, well inside the sheet, and separated
# by a wide corridor of empty background: the decisive support is therefore two
# disconnected clusters spread over one medium region of the sheet.
LEFT_BOX = (194, 194, 315, 266)     # x0, y0, x1, y1
RIGHT_BOX = (574, 194, 695, 266)

N_LEFT, N_RIGHT = 4, 3
LEN_MIN, LEN_MAX = 26, 110
THICK = 8
X_JIT = 7
GAP_RANGE = (12, 26)      # per-scene floor on the spacing of the four lengths
MARGIN_FLOOR = 12.0       # never emitted below this
QUARANTINE_AT = 14.0      # flagged at or below this

DECOR_COLORS = [
    (232, 158, 74), (108, 178, 168), (196, 138, 196),
    (166, 178, 108), (150, 168, 208), (214, 152, 140),
]


# ---------------------------------------------------------------- sampling

def _lengths(rng: random.Random) -> list[int] | None:
    """Four increasing lengths spaced by a per-scene floor.

    The floor is resampled for every scene, so the decision margin sweeps the
    whole legible band instead of piling up at one value, and near-boundary
    scenes occur naturally.
    """
    span = LEN_MAX - LEN_MIN
    n_gaps = N_LEFT - 1
    for _ in range(400):
        floor = rng.randint(*GAP_RANGE)
        gaps = [floor] * n_gaps
        tight = rng.randrange(n_gaps)
        gaps[tight] += rng.randint(0, 2)
        rem = span - sum(gaps)
        if rem < 0:
            continue
        cuts = sorted(rng.randint(0, rem) for _ in range(n_gaps))
        parts = [cuts[0]] + [cuts[i] - cuts[i - 1] for i in range(1, n_gaps)]
        parts.append(rem - cuts[-1])
        for k, idx in enumerate(i for i in range(n_gaps) if i != tight):
            gaps[idx] += parts[k]
        vals = [LEN_MIN + parts[-1]]
        for g in gaps:
            vals.append(vals[-1] + g)
        return vals
    return None


def _rows(box, count, pitch, top_pad, rng, lengths):
    bars = []
    for i, ln in enumerate(lengths):
        bars.append({
            "x": box[0] + 2 + rng.randrange(0, X_JIT),
            "y": box[1] + top_pad + i * pitch,
            "length": int(ln),
            "thick": THICK,
        })
    return bars


def sample_scene(seed: int) -> dict:
    rng = random.Random(1000003 * (seed + 1) + 17)
    for _ in range(4000):
        vals = _lengths(rng)
        if vals is None:
            continue
        # The unpaired length is never the longest or the shortest of the four,
        # so "pick the extreme bar" is not a substitute rule.
        odd_val = vals[rng.randrange(1, N_LEFT - 1)]
        partners = [v for v in vals if v != odd_val]

        rows = vals[:]
        rng.shuffle(rows)                     # unpaired row index is uniform
        right_rows = partners[:]
        rng.shuffle(right_rows)               # partner order carries no signal

        scene = {
            "width": CANVAS[0],
            "height": CANVAS[1],
            "ink": list(INK),
            "left_box": list(LEFT_BOX),
            "right_box": list(RIGHT_BOX),
            "left": _rows(LEFT_BOX, N_LEFT, 18, 3, rng, rows),
            "right": _rows(RIGHT_BOX, N_RIGHT, 23, 6, rng, right_rows),
            "decor": _decor(rng),
            "guides": _guides(rng),
        }
        if not _fits(scene):
            continue
        if margin(scene) < MARGIN_FLOOR:
            continue
        return scene
    raise RuntimeError("could not sample a legible scene")


def _fits(scene: dict) -> bool:
    for key, box in (("left", scene["left_box"]), ("right", scene["right_box"])):
        for bar in scene[key]:
            if bar["x"] < box[0] + 2 or bar["x"] + bar["length"] > box[2] - 2:
                return False
            if bar["y"] < box[1] + 2 or bar["y"] + bar["thick"] > box[3] - 2:
                return False
    lmax = max(b["x"] + b["length"] for b in scene["left"])
    rmin = min(b["x"] for b in scene["right"])
    return rmin - lmax >= 200          # wide empty corridor between the panels


def _decor(rng: random.Random) -> list[dict]:
    """Pale coloured decoration, always clear of both panels."""
    out = []
    for _ in range(rng.randrange(6, 11)):
        for _ in range(60):
            cx = rng.randrange(24, 736)
            cy = rng.randrange(24, 436)
            if all(not (b[0] - 26 <= cx <= b[2] + 26 and b[1] - 26 <= cy <= b[3] + 26)
                   for b in (LEFT_BOX, RIGHT_BOX)):
                break
        out.append({
            "kind": rng.choice(("dot", "ring", "tick", "chip")),
            "cx": cx, "cy": cy,
            "r": rng.randrange(6, 15),
            "color": list(DECOR_COLORS[rng.randrange(len(DECOR_COLORS))]),
            "w": rng.randrange(2, 5),
        })
    return out


def _guides(rng: random.Random) -> list[dict]:
    """Very light grey guide hairlines across the sheet (never dark ink)."""
    return [{"y": rng.randrange(40, 420), "x0": rng.randrange(20, 300),
             "len": rng.randrange(80, 420)}
            for _ in range(rng.randrange(3, 7))]


# ---------------------------------------------------------------- gold

def _match(scene: dict) -> tuple[int, float]:
    """Return (unpaired left row index 0-based, runner-up margin px)."""
    lefts = [b["length"] for b in scene["left"]]
    rights = [b["length"] for b in scene["right"]]
    dists = [min(abs(L - R) for R in rights) for L in lefts]
    order = sorted(range(len(lefts)), key=lambda i: -dists[i])
    odd = order[0]
    runner = dists[order[1]]
    slack = min(
        dists[odd] - runner,
        min(abs(lefts[i] - lefts[j])
            for i in range(len(lefts)) for j in range(i + 1, len(lefts))),
    )
    return odd, float(slack)


def analytic_gold(scene: dict) -> str:
    odd, _ = _match(scene)
    return str(odd + 1)


def margin(scene: dict) -> float:
    return _match(scene)[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_AT


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias: every sampled field is a drawing primitive."""
    return []


# ------------------------------------------- label-preserving perturbations
# Declared symmetries of the ANSWER (not of the pixels); exercised by
# verify.py and the candidate tests, not by the latent-alias gate.

def _copy(scene: dict) -> dict:
    out = {k: (list(v) if isinstance(v, list) else v) for k, v in scene.items()}
    out["left"] = [dict(b) for b in scene["left"]]
    out["right"] = [dict(b) for b in scene["right"]]
    out["decor"] = [dict(d) for d in scene["decor"]]
    out["guides"] = [dict(g) for g in scene["guides"]]
    return out


def relabel_decor(scene: dict, shift: int = 1) -> dict:
    out = _copy(scene)
    for dec, src in zip(out["decor"], scene["decor"]):
        idx = DECOR_COLORS.index(tuple(src["color"]))
        dec["color"] = list(DECOR_COLORS[(idx + shift) % len(DECOR_COLORS)])
    return out


def translate_block(scene: dict, dx: int = 0, dy: int = 0) -> dict:
    """Shift both panels together; the pairing is unchanged."""
    out = _copy(scene)
    for key in ("left_box", "right_box"):
        x0, y0, x1, y1 = scene[key]
        out[key] = [x0 + dx, y0 + dy, x1 + dx, y1 + dy]
    for key in ("left", "right"):
        for bar in out[key]:
            bar["x"] += dx
            bar["y"] += dy
    return out


def permute_right_rows(scene: dict) -> dict:
    """Reorder the right panel vertically; the partner set is unchanged."""
    out = _copy(scene)
    lens = [b["length"] for b in out["right"]]
    lens = lens[1:] + lens[:1]
    for bar, ln in zip(out["right"], lens):
        bar["length"] = ln
    return out


# ---------------------------------------------------------------- render

def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BG)
    d = ImageDraw.Draw(img)
    for g in scene["guides"]:
        d.line([g["x0"], g["y"], g["x0"] + g["len"], g["y"]], fill=GUIDE, width=2)
    for box in (scene["left_box"], scene["right_box"]):
        d.rectangle(list(box), outline=FRAME, width=2)
    for dec in scene["decor"]:
        c = tuple(dec["color"])
        cx, cy, r = dec["cx"], dec["cy"], dec["r"]
        if dec["kind"] == "dot":
            d.ellipse([cx - r, cy - r, cx + r, cy + r], fill=c)
        elif dec["kind"] == "ring":
            d.ellipse([cx - r, cy - r, cx + r, cy + r], outline=c, width=dec["w"])
        elif dec["kind"] == "tick":
            d.line([cx, cy - r, cx + r, cy + r], fill=c, width=dec["w"])
        else:
            d.rectangle([cx - r, cy - r // 2, cx + r, cy + r // 2], fill=c)
    ink = tuple(scene["ink"])
    for bar in scene["left"] + scene["right"]:
        d.rectangle(
            [bar["x"], bar["y"],
             bar["x"] + bar["length"] - 1, bar["y"] + bar["thick"] - 1],
            fill=ink,
        )
    return img
