from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIZE = (640, 480)
BACKGROUND = (247, 244, 236)
NAVY = (29, 43, 61)
GREEN = (46, 160, 96)
RED = (211, 69, 74)
CYAN = (25, 174, 201)
ORANGE = (238, 139, 43)
WHITE = (255, 255, 255)


def _polyline_lengths(points: list[list[int]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for left, right in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(left, right))
    return cumulative, cumulative[-1]


def _point_at(points: list[list[int]], fraction: float) -> tuple[float, float]:
    cumulative, total = _polyline_lengths(points)
    target = fraction * total
    for index in range(len(points) - 1):
        if target <= cumulative[index + 1]:
            span = cumulative[index + 1] - cumulative[index]
            ratio = (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return x0 + ratio * (x1 - x0), y0 + ratio * (y1 - y0)
    return tuple(points[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    points = [
        [54 + rng.randint(-6, 6), 414 + rng.randint(-6, 6)],
        [208 + rng.randint(-9, 9), 336 + rng.randint(-7, 7)],
        [119 + rng.randint(-8, 8), 258 + rng.randint(-7, 7)],
        [322 + rng.randint(-9, 9), 226 + rng.randint(-7, 7)],
        [247 + rng.randint(-8, 8), 145 + rng.randint(-7, 7)],
        [438 + rng.randint(-9, 9), 174 + rng.randint(-7, 7)],
        [586 + rng.randint(-6, 6), 58 + rng.randint(-6, 6)],
    ]
    transform = rng.randrange(4)
    if transform & 1:
        points = [[SIZE[0] - x, y] for x, y in points]
    if transform & 2:
        points = [[x, SIZE[1] - y] for x, y in points]
    pattern_index = seed % 4
    patterns = [
        ["cyan", "orange", "cyan", "orange"],
        ["cyan", "orange", "orange", "cyan"],
        ["orange", "cyan", "orange", "cyan"],
        ["orange", "cyan", "cyan", "orange"],
    ]
    jitter = [rng.uniform(-0.012, 0.012) for _ in range(4)]
    positions = [round(base + delta, 5) for base, delta in zip((0.12, 0.37, 0.63, 0.88), jitter)]
    return {
        "points": points,
        "checkpoint_positions": positions,
        "checkpoint_colors": patterns[pattern_index],
        "marker_radius": 10 + rng.randrange(2),
        "line_width": 8 + 2 * rng.randrange(2),
    }


def render(scene: dict) -> Image.Image:
    points = [tuple(value) for value in scene["points"]]
    positions = [float(value) for value in scene["checkpoint_positions"]]
    colors = list(scene["checkpoint_colors"])
    radius = int(scene["marker_radius"])
    width = int(scene["line_width"])
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.line(points, fill=WHITE, width=width + 8, joint="curve")
    draw.line(points, fill=NAVY, width=width, joint="curve")
    palette = {"cyan": CYAN, "orange": ORANGE}
    for fraction, name in zip(positions, colors):
        x, y = _point_at(scene["points"], fraction)
        box = [round(x - radius), round(y - radius), round(x + radius), round(y + radius)]
        draw.ellipse(box, fill=palette[name])
    endpoint_radius = radius + 4
    for (x, y), color in ((points[0], GREEN), (points[-1], RED)):
        box = [x - endpoint_radius, y - endpoint_radius, x + endpoint_radius, y + endpoint_radius]
        draw.ellipse(box, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    sequence = list(scene["checkpoint_colors"])
    return "interleaved" if sequence[0] == sequence[2] else "nested"


def margin(scene: dict) -> float:
    points = scene["points"]
    positions = sorted(float(value) for value in scene["checkpoint_positions"])
    _, total = _polyline_lengths(points)
    gaps = [(right - left) * total for left, right in zip(positions, positions[1:])]
    return round(min(gaps) - 2.0 * float(scene["marker_radius"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 35.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
