from __future__ import annotations

from collections import deque

import numpy as np

NAVY = (29, 43, 61)
GREEN = (46, 160, 96)
RED = (211, 69, 74)
CYAN = (25, 174, 201)
ORANGE = (238, 139, 43)


def _mask(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(color, dtype=np.uint8), axis=2)


def _components(mask: np.ndarray, minimum: int = 12) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        group = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(group) >= minimum:
            groups.append(group)
    return groups


def _distances(traversable: np.ndarray, sources: np.ndarray) -> np.ndarray:
    height, width = traversable.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    queue = deque()
    for y, x in np.argwhere(sources):
        y, x = int(y), int(x)
        distance[y, x] = 0
        queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        step = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and traversable[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = step
                queue.append((ny, nx))
    return distance


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = {name: _mask(array, color) for name, color in {
        "navy": NAVY, "green": GREEN, "red": RED, "cyan": CYAN, "orange": ORANGE
    }.items()}
    cyan_groups = _components(masks["cyan"])
    orange_groups = _components(masks["orange"])
    green_groups = _components(masks["green"])
    red_groups = _components(masks["red"])
    if len(cyan_groups) != 2 or len(orange_groups) != 2 or len(green_groups) != 1 or len(red_groups) != 1:
        raise ValueError("expected two checkpoints of each color and one endpoint of each kind")
    traversable = np.logical_or.reduce(tuple(masks.values()))
    distance = _distances(traversable, masks["green"])
    if not np.any(distance[masks["red"]] >= 0):
        raise ValueError("finish is not connected to start through the visible route")
    ordered = []
    for name, groups in (("cyan", cyan_groups), ("orange", orange_groups)):
        for group in groups:
            values = [int(distance[y, x]) for y, x in group if distance[y, x] >= 0]
            if not values:
                raise ValueError("checkpoint is not connected to the visible route")
            ordered.append((min(values), name))
    ordered.sort()
    if any(right[0] - left[0] < 20 for left, right in zip(ordered, ordered[1:])):
        raise ValueError("checkpoint order is not resolved at this raster scale")
    sequence = [name for _, name in ordered]
    if sequence in (["cyan", "orange", "cyan", "orange"], ["orange", "cyan", "orange", "cyan"]):
        return "interleaved"
    if sequence in (["cyan", "orange", "orange", "cyan"], ["orange", "cyan", "cyan", "orange"]):
        return "nested"
    raise ValueError("visible checkpoint arrangement is outside the question contract")
