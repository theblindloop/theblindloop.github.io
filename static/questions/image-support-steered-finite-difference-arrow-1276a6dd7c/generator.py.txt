"""Deterministic renderer for the neighbor-difference arrow world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

ROWS = 28
RAW_COLS = 29
OUT_COLS = 28
CANVAS = 512
FRAME = 6
FRAME_COLOR = (111, 48, 151)
GRID_COLOR = (205, 201, 211)
BACKGROUND = (248, 246, 239)
SYMBOLS = ("up", "down", "left", "right")
PALETTES = (
    ((30, 73, 112), (238, 214, 127)),
    ((38, 105, 82), (242, 187, 174)),
    ((112, 45, 63), (178, 226, 220)),
    ((49, 52, 63), (231, 174, 88)),
)


def symbol_template(symbol: str) -> np.ndarray:
    """Return one thick, connected, area-like 28x28 arrow mask."""
    if symbol not in SYMBOLS:
        raise ValueError(f"unknown symbol {symbol!r}")
    image = Image.new("1", (28, 28), 0)
    draw = ImageDraw.Draw(image)
    # Right-facing arrow. Rotations preserve total ink exactly.
    draw.polygon([(1, 8), (14, 8), (14, 2), (26, 14), (14, 26), (14, 20), (1, 20)], fill=1)
    array = np.asarray(image, dtype=np.uint8)
    rotations = {"right": 0, "up": 1, "left": 2, "down": 3}
    return np.rot90(array, rotations[symbol]).copy()


def _raw_bits(scene: dict) -> np.ndarray:
    target = symbol_template(scene["symbol"])
    phases = np.asarray(scene["row_phases"], dtype=np.uint8)
    if phases.shape != (ROWS,):
        raise ValueError("row_phases must contain exactly 28 binary entries")
    raw = np.zeros((ROWS, RAW_COLS), dtype=np.uint8)
    raw[:, 0] = phases
    for x in range(OUT_COLS):
        raw[:, x + 1] = np.bitwise_xor(raw[:, x], target[:, x])
    return raw


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    cell_size = rng.choice((11, 12, 13))
    field_w, field_h = RAW_COLS * cell_size, ROWS * cell_size
    jitter_x = rng.randint(-22, 22)
    jitter_y = rng.randint(-22, 22)
    origin_x = (CANVAS - field_w) // 2 + jitter_x
    origin_y = (CANVAS - field_h) // 2 + jitter_y
    # Seed-mixed row phases prevent the integrated pattern from having a stable polarity.
    row_phases = [rng.randrange(2) for _ in range(ROWS)]
    return {
        "symbol": symbol,
        "row_phases": row_phases,
        "palette_index": rng.randrange(len(PALETTES)),
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
    }


def render(scene: dict) -> Image.Image:
    symbol = scene["symbol"]
    phases = scene["row_phases"]
    palette_index = int(scene["palette_index"])
    cell = int(scene["cell_size"])
    ox = int(scene["origin_x"])
    oy = int(scene["origin_y"])
    # Explicit reads keep the rendered-field audit transparent.
    working = dict(scene)
    working["symbol"] = symbol
    working["row_phases"] = phases
    bits = _raw_bits(working)
    dark, light = PALETTES[palette_index]
    field_w, field_h = RAW_COLS * cell, ROWS * cell

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        [ox - FRAME, oy - FRAME, ox + field_w + FRAME - 1, oy + field_h + FRAME - 1],
        fill=FRAME_COLOR,
    )
    for y in range(ROWS):
        for x in range(RAW_COLS):
            x0, y0 = ox + x * cell, oy + y * cell
            draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=dark if bits[y, x] else light)
    # Faint lines make every square addressable even when neighboring values match.
    for x in range(1, RAW_COLS):
        xx = ox + x * cell
        draw.line([(xx, oy), (xx, oy + field_h - 1)], fill=GRID_COLOR, width=1)
    for y in range(1, ROWS):
        yy = oy + y * cell
        draw.line([(ox, yy), (ox + field_w - 1, yy)], fill=GRID_COLOR, width=1)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def margin(scene: dict) -> float:
    target = symbol_template(scene["symbol"])
    distances = [int(np.count_nonzero(target != symbol_template(other))) for other in SYMBOLS if other != scene["symbol"]]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
