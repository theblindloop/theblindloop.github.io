"""Independent image-only inverse for Cross-Field Badge Splice."""

from __future__ import annotations

from PIL import Image, ImageDraw

CANVAS = (1000, 800)
INK = (35, 40, 48)
MUTED = (222, 226, 229)
SHAPE = (25, 126, 111)
BADGE = (58, 48, 92)
PANEL = (255, 255, 255)
COLORS = {"amber": (219, 139, 22), "cyan": (18, 153, 177), "violet": (133, 76, 181)}
ROW_Y = (315, 395, 475, 555)


class OracleError(ValueError):
    pass


def _shape_template(name: str) -> tuple[int, ...]:
    image = Image.new("RGB", (80, 80), PANEL)
    draw, cx, cy = ImageDraw.Draw(image), 40, 40
    if name == "circle": draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), fill=SHAPE)
    elif name == "square": draw.rectangle((cx - 23, cy - 23, cx + 23, cy + 23), fill=SHAPE)
    elif name == "triangle": draw.polygon(((cx, cy - 29), (cx - 29, cy + 23), (cx + 29, cy + 23)), fill=SHAPE)
    elif name == "diamond": draw.polygon(((cx, cy - 30), (cx - 30, cy), (cx, cy + 30), (cx + 30, cy)), fill=SHAPE)
    return tuple(1 if pixel == SHAPE else 0 for pixel in image.getdata())


def _read_shape(image: Image.Image, cx: int, cy: int) -> str:
    crop = image.crop((cx - 40, cy - 40, cx + 40, cy + 40))
    observed = tuple(1 if pixel == SHAPE else 0 for pixel in crop.getdata())
    scores = {name: sum(a != b for a, b in zip(observed, _shape_template(name)))
              for name in ("circle", "square", "triangle", "diamond")}
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 1500:
        raise OracleError("damaged or unknown shape")
    return best


def _read_badge(image: Image.Image, cx: int, cy: int) -> tuple[int, ...]:
    px = image.load()
    if px[cx - 48, cy] != INK or px[cx + 48, cy] != INK:
        raise OracleError("badge border missing")
    bits = []
    for i in range(6):
        col, row = i % 3, i // 3
        x0, y0 = cx - 34 + col * 24, cy - 23 + row * 28
        samples = [px[x, y] for x in range(x0 + 3, x0 + 16, 4)
                   for y in range(y0 + 3, y0 + 16, 4)]
        if all(pixel == BADGE for pixel in samples): bits.append(1)
        elif all(pixel == MUTED for pixel in samples): bits.append(0)
        else: raise OracleError("badge cell damaged")
    if sum(bits) != 3:
        raise OracleError("invalid badge")
    return tuple(bits)


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != CANVAS:
        raise OracleError("wrong canvas")
    px = image.load()
    for x, y in ((500, 35), (80, 400), (920, 400)):
        if px[x, y] != INK: raise OracleError("panel boundary missing")
    first_key = _read_shape(image, 360, 145)
    second_key = _read_shape(image, 640, 145)
    if first_key == second_key:
        raise OracleError("query keys must differ")
    records = {}
    badges_seen = set()
    for y in ROW_Y:
        if px[320, y] != INK: raise OracleError("record arrow missing")
        key = _read_shape(image, 220, y)
        left, right = _read_badge(image, 500, y), _read_badge(image, 700, y)
        records[key] = (left, right)
        badges_seen.update((left, right))
    if len(records) != 4 or len(badges_seen) != 8:
        raise OracleError("duplicate table mark")
    target = (records[first_key][0], records[second_key][1])
    matches = []
    for cx in (175, 500, 825):
        border = px[cx, 630]
        names = [name for name, value in COLORS.items() if border == value]
        if len(names) != 1: raise OracleError("answer border damaged")
        pair = (_read_badge(image, cx - 58, 720), _read_badge(image, cx + 58, 720))
        if pair == target: matches.append(names[0])
    if len(matches) != 1:
        raise OracleError("non-unique spliced pair")
    return matches[0]
