"""Independent image-only inverse for the smooth-path constellation.

This module deliberately duplicates only the public pixel palette and the
measurement idea.  It receives a PNG, segments each visible color, recovers
dot centroids from connected components, and compares full-set quadratic
residuals.  It imports no renderer, scene, generator, prompt, or gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


PIXEL_COLORS = {
    "red": np.asarray((205, 55, 67), dtype=np.int16),
    "blue": np.asarray((48, 103, 176), dtype=np.int16),
    "green": np.asarray((47, 143, 82), dtype=np.int16),
}
EXPECTED_DOTS = 11
COLOR_TOLERANCE_SQUARED = 2200


def _connected_components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    """Return 8-connected foreground components using only the binary mask."""

    ys, xs = np.nonzero(mask)
    remaining = set(zip(xs.tolist(), ys.tolist()))
    components: list[list[tuple[int, int]]] = []
    while remaining:
        first = remaining.pop()
        queue = deque([first])
        component = [first]
        while queue:
            x, y = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    neighbor = (x + dx, y + dy)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        queue.append(neighbor)
                        component.append(neighbor)
        components.append(component)
    return components


def _dot_centroids(image: Image.Image, color: str) -> list[tuple[float, float]]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    prototype = PIXEL_COLORS[color].astype(np.int32)
    distance_squared = np.sum((pixels - prototype) ** 2, axis=2)
    mask = distance_squared <= COLOR_TOLERANCE_SQUARED
    components = [item for item in _connected_components(mask) if len(item) >= 20]
    if len(components) != EXPECTED_DOTS:
        return []
    centers = [
        (
            float(sum(point[0] for point in component) / len(component)),
            float(sum(point[1] for point in component) / len(component)),
        )
        for component in components
    ]
    centers.sort()
    return centers


def _quadratic_residual(points: list[tuple[float, float]]) -> float:
    if len(points) != EXPECTED_DOTS:
        return float("inf")
    x = np.asarray([point[0] for point in points], dtype=float)
    y = np.asarray([point[1] for point in points], dtype=float)
    if float(np.ptp(x)) < 12.0:
        return float("inf")
    coefficients = np.polyfit(x, y, 2)
    residuals = y - np.polyval(coefficients, x)
    return float(np.sqrt(np.mean(residuals * residuals)))


def decision_from_image(image: Image.Image) -> str:
    """Choose the color with the smallest whole-set quadratic residual."""

    scores = {
        color: _quadratic_residual(_dot_centroids(image, color))
        for color in PIXEL_COLORS
    }
    best = min(scores, key=lambda color: (scores[color], color))
    if not np.isfinite(scores[best]):
        return "ambiguous"
    return best
