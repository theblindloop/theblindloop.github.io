"""Renderer and analytic contract for the smooth-path constellation world.

Three same-size colored dot sets share the same abscissae and the same set of
ordinate offsets.  The target color receives those offsets in their smooth
quadratic order; each other color receives a deliberately scrambled order.
The answer is therefore a relation among all points, not a count or a local
marker.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw


RENDERER_VERSION = "smooth_path_constellation-0.1.0"
COLOR_NAMES = ("red", "blue", "green")
PALETTE = {
    "red": (205, 55, 67),
    "blue": (48, 103, 176),
    "green": (47, 143, 82),
}
BACKGROUND = (248, 247, 242)
WIDTH = 900
HEIGHT = 620
POINT_COUNT = 11
RADIUS_RANGE = (6.0, 8.0)
QUARANTINE_FLOOR = 8.0


def _quadratic_residual(xs: Iterable[float], ys: Iterable[float]) -> float:
    """Return RMS residual of the best quadratic through the supplied points."""

    x = np.asarray(list(xs), dtype=float)
    y = np.asarray(list(ys), dtype=float)
    if len(x) < 3 or float(np.ptp(x)) < 1e-9:
        return float("inf")
    coefficients = np.polyfit(x, y, 2)
    residuals = y - np.polyval(coefficients, x)
    return float(np.sqrt(np.mean(residuals * residuals)))


def _permutation_score(x_offsets: list[float], values: list[float], permutation: list[int]) -> float:
    ys = [values[index] for index in permutation]
    return _quadratic_residual(x_offsets, ys)


def _curve_offsets(rng: random.Random, x_offsets: list[float]) -> list[float]:
    """Sample one curved ordinate multiset, with a small common measurement wobble."""

    curvature = rng.uniform(54.0, 82.0) * (1.0 if rng.random() < 0.5 else -1.0)
    slope = rng.uniform(-23.0, 23.0)
    noise = [rng.uniform(-1.4, 1.4) for _ in x_offsets]
    values = [
        curvature * (u * u - 0.36) + slope * u + wobble
        for u, wobble in zip(x_offsets, noise)
    ]
    mean = sum(values) / len(values)
    return [round(value - mean, 4) for value in values]


def _decoy_permutation(
    rng: random.Random,
    x_offsets: list[float],
    values: list[float],
    minimum_residual: float = 18.0,
) -> list[int]:
    """Find a scrambled ordering that is visibly unlike the smooth ordering."""

    identity = list(range(len(values)))
    for _ in range(5000):
        candidate = identity[:]
        rng.shuffle(candidate)
        if candidate == identity:
            continue
        if _permutation_score(x_offsets, values, candidate) >= minimum_residual:
            return candidate
    # This fallback is deterministic and is only reachable if the sampled
    # curve is unexpectedly flat; sample_scene rejects that scene below.
    return list(reversed(identity))


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene from an integer seed."""

    rng = random.Random(int(seed))
    target_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    x_offsets = [
        round(-1.0 + 2.0 * i / (POINT_COUNT - 1) + rng.uniform(-0.025, 0.025), 5)
        for i in range(POINT_COUNT)
    ]
    x_offsets.sort()
    values = _curve_offsets(rng, x_offsets)
    identity = list(range(POINT_COUNT))
    lane_centers = [120.0, 310.0, 500.0]
    rng.shuffle(lane_centers)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)

    lanes = []
    for color, center_y in zip(colors, lane_centers):
        lanes.append(
            {
                "color": color,
                "cx": round(rng.uniform(435.0, 465.0), 4),
                "cy": round(center_y + rng.uniform(-10.0, 10.0), 4),
                "x_scale": round(rng.uniform(245.0, 275.0), 4),
                "y_scale": round(rng.uniform(0.88, 1.12), 4),
                "decoy_permutation": identity
                if color == target_color
                else _decoy_permutation(rng, x_offsets, values),
            }
        )

    scene = {
        "target_color": target_color,
        "colors": list(COLOR_NAMES),
        "lanes": lanes,
        "x_offsets": x_offsets,
        "curve_offsets": values,
        "palette": {name: list(rgb) for name, rgb in PALETTE.items()},
        "background": list(BACKGROUND),
        "width": WIDTH,
        "height": HEIGHT,
        "radius": round(rng.uniform(*RADIUS_RANGE), 4),
    }
    # Keep the public generator away from marginal permutations.  The target
    # is always the unique best fit, while this check also protects the raster
    # oracle from an accidental near-tie at a stress seed.
    if margin(scene) < 18.0:
        return sample_scene(int(seed) + 104729)
    return scene


def _lane_points(scene: dict, lane: dict) -> list[tuple[float, float]]:
    values = scene["curve_offsets"]
    identity = list(range(len(values)))
    permutation = identity if lane["color"] == scene["target_color"] else lane["decoy_permutation"]
    return [
        (
            float(lane["cx"]) + float(lane["x_scale"]) * float(x),
            float(lane["cy"]) + float(lane["y_scale"]) * float(values[index]),
        )
        for x, index in zip(scene["x_offsets"], permutation)
    ]


def render(scene: dict) -> Image.Image:
    """Render the scene with deterministic supersampling."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    target_color = scene["target_color"]
    declared_colors = set(scene["colors"])
    radius = float(scene["radius"])
    for lane in scene["lanes"]:
        color_name = lane["color"]
        if color_name not in declared_colors:
            raise ValueError("lane color is not declared")
        # Reading target_color here is substantive: it assigns the smooth
        # correspondence to the visible target color during rendering.
        rgb = tuple(scene["palette"][color_name])
        for x, y in _lane_points(scene, lane):
            box = [
                (x - radius) * scale,
                (y - radius) * scale,
                (x + radius) * scale,
                (y + radius) * scale,
            ]
            draw.ellipse(box, fill=rgb)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the declared answer label; it is visibly assigned in render()."""

    return str(scene["target_color"])


def _scores(scene: dict) -> dict[str, float]:
    scores = {}
    for lane in scene["lanes"]:
        points = _lane_points(scene, lane)
        scores[str(lane["color"])] = _quadratic_residual(
            [point[0] for point in points], [point[1] for point in points]
        )
    return scores


def margin(scene: dict) -> float:
    """Return the best-vs-runner-up curve residual gap in image pixels."""

    ordered = sorted(_scores(scene).values())
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This minimal parameterization declares no latent pixel-identical aliases."""

    return []
