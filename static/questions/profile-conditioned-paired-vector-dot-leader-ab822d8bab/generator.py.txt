"""Deterministic renderer for paired three-entry vector dot products."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 500
LABELS = ("A", "B", "C", "D")
ROW_Y = (150, 235, 320, 405)
ENTRY_DY = (-18, 0, 18)
SCALE = 13
P_X, Q_X = 155, 475
PALETTES = (
    ((31, 119, 180), (238, 119, 35)),
    ((34, 150, 110), (126, 82, 176)),
    ((25, 132, 168), (220, 82, 96)),
)


def _dot(p: list[int], q: list[int]) -> int:
    return sum(int(a) * int(b) for a, b in zip(p, q))


def _composition(rng: random.Random, total: int) -> list[int]:
    while True:
        values = [rng.randint(1, 10), rng.randint(1, 10)]
        values.append(total - sum(values))
        if 1 <= values[2] <= 10:
            return values


def sample_scene(seed: int) -> dict:
    """Sample four equal-ink rows with a balanced, well-separated winner."""
    rng = random.Random(int(seed))
    target = int(seed) % 4
    for _ in range(10_000):
        raw = [
            {"p": _composition(rng, 16), "q": _composition(rng, 18)}
            for _ in range(4)
        ]
        scores = [_dot(row["p"], row["q"]) for row in raw]
        order = sorted(range(4), key=lambda i: scores[i], reverse=True)
        if len(set(scores)) == 4 and scores[order[0]] - scores[order[1]] >= 18:
            winner = raw.pop(order[0])
            rng.shuffle(raw)
            arranged = list(raw)
            arranged.insert(target, winner)
            rows = [
                {"label": LABELS[index], "p": row["p"], "q": row["q"]}
                for index, row in enumerate(arranged)
            ]
            rng.shuffle(rows)  # latent storage order is deliberately nonvisual
            columns = [0, 1, 2]
            rng.shuffle(columns)
            return {
                "rows": rows,
                "palette": int(seed) % len(PALETTES),
                "offset": {"x": rng.randint(-9, 9), "y": rng.randint(-5, 5)},
                "column_order": columns,
            }
    raise RuntimeError("could not sample a separated dot-product scene")


def analytic_gold(scene: dict) -> str:
    rows = scene["rows"]
    scores = [_dot(row["p"], row["q"]) for row in rows]
    return rows[max(range(len(rows)), key=lambda index: scores[index])]["label"]


def margin(scene: dict) -> float:
    scores = sorted((_dot(row["p"], row["q"]) for row in scene["rows"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["rows"] = list(reversed(twin["rows"]))
    return [("reverse_nonvisual_row_storage", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    title_font = ImageFont.load_default(size=20)
    text_font = ImageFont.load_default(size=17)
    small_font = ImageFont.load_default(size=14)
    p_color, q_color = PALETTES[int(scene.get("palette", 0)) % len(PALETTES)]
    dx = int(scene.get("offset", {}).get("x", 0))
    dy = int(scene.get("offset", {}).get("y", 0))
    column_order = [int(value) for value in scene.get("column_order", [0, 1, 2])]
    by_label = {row["label"]: row for row in scene["rows"]}

    draw.text((WIDTH // 2, 27), "Paired three-entry vectors", fill=(30, 32, 35), anchor="mm", font=title_font)
    draw.text((P_X + 65 + dx, 70 + dy), "P", fill=p_color, anchor="mm", font=title_font)
    draw.text((Q_X + 65 + dx, 70 + dy), "Q", fill=q_color, anchor="mm", font=title_font)
    draw.text((WIDTH // 2, 96 + dy), "aligned entries 1   2   3", fill=(82, 84, 88), anchor="mm", font=small_font)

    for row_index, label in enumerate(LABELS):
        row = by_label[label]
        cy = ROW_Y[row_index] + dy
        draw.rounded_rectangle((38 + dx, cy - 34, 710 + dx, cy + 34), radius=8, fill=(244, 244, 240), outline=(220, 221, 217), width=1)
        draw.text((72 + dx, cy), label, fill=(25, 27, 30), anchor="mm", font=text_font)
        for base_x in (P_X + dx, Q_X + dx):
            draw.line((base_x - 1, cy - 29, base_x - 1, cy + 29), fill=(166, 168, 166), width=2)
            draw.line((base_x, cy + 29, base_x + 132, cy + 29), fill=(224, 224, 220), width=1)
        for visual_slot, source_index in enumerate(column_order):
            y = cy + ENTRY_DY[visual_slot]
            p_len = int(row["p"][source_index]) * SCALE
            q_len = int(row["q"][source_index]) * SCALE
            draw.rectangle((P_X + dx, y - 5, P_X + dx + p_len, y + 5), fill=p_color)
            draw.rectangle((Q_X + dx, y - 5, Q_X + dx + q_len, y + 5), fill=q_color)
            draw.text((124 + dx, y), str(visual_slot + 1), fill=(105, 107, 109), anchor="mm", font=small_font)
            draw.text((444 + dx, y), str(visual_slot + 1), fill=(105, 107, 109), anchor="mm", font=small_font)
    return image
