"""Independent PNG-only inverse for paired bar-vector dot products."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 100:
            result.append((min(xs), max(xs), min(ys), max(ys), len(xs)))
    return result


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    saturated = (high - low > 58) & (high > 105)
    saturated[:105, :] = False
    marks = _components(saturated)
    left = sorted((item for item in marks if item[1] < 360), key=lambda item: (item[2] + item[3]) / 2)
    right = sorted((item for item in marks if item[0] > 400), key=lambda item: (item[2] + item[3]) / 2)
    if len(left) != 12 or len(right) != 12:
        raise ValueError("expected twelve bars for each vector")
    for p_mark, q_mark in zip(left, right):
        py = (p_mark[2] + p_mark[3]) / 2
        qy = (q_mark[2] + q_mark[3]) / 2
        if abs(py - qy) > 2.0:
            raise ValueError("aligned entries cannot be bound")
    p_widths = [item[1] - item[0] for item in left]
    q_widths = [item[1] - item[0] for item in right]
    scores = []
    for row in range(4):
        start = row * 3
        scores.append(sum(p_widths[i] * q_widths[i] for i in range(start, start + 3)))
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] - scores[order[1]] < 1000:
        raise ValueError("pixel dot products are too close to resolve")
    return LABELS[order[0]]
