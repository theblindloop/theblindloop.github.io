"""Independent pixel-only inverse arm."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _centers_from_pixels(image: Image.Image) -> list[tuple[float, float]]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = np.max(rgb, axis=2) < 100
    ys, xs = np.nonzero(mask)
    remaining = set(zip(ys.tolist(), xs.tolist()))
    centers: list[tuple[float, float]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
                        component.append(neighbor)
        if len(component) >= 40:
            centers.append(
                (
                    sum(item[1] for item in component) / len(component),
                    sum(item[0] for item in component) / len(component),
                )
            )
    if len(centers) < 4:
        raise ValueError("image does not contain a recoverable dot constellation")
    return sorted(centers)


def _minimum_tree(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    n = len(points)
    parent = list(range(n))

    def find(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    edges = sorted(
        (math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1]), i, j)
        for i in range(n)
        for j in range(i + 1, n)
    )
    chosen: list[tuple[int, int]] = []
    for _, i, j in edges:
        a, b = find(i), find(j)
        if a == b:
            continue
        parent[a] = b
        chosen.append((i, j))
        if len(chosen) == n - 1:
            break
    return chosen


def decision_from_image(image: Image.Image) -> str:
    points = _centers_from_pixels(image)
    degree = [0] * len(points)
    for i, j in _minimum_tree(points):
        degree[i] += 1
        degree[j] += 1
    return "chain" if max(degree) <= 2 else "branch"
