"""Latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 3
QUARANTINE_MARGIN = 2.5


def _distance(points: list[list[int]], i: int, j: int) -> float:
    return math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1])


def _mst(points: list[list[int]]) -> list[tuple[int, int, float]]:
    """Kruskal MST with deterministic tie ordering."""
    n = len(points)
    parent = list(range(n))

    def root(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    edges = sorted(
        (_distance(points, i, j), i, j) for i in range(n) for j in range(i + 1, n)
    )
    tree: list[tuple[int, int, float]] = []
    for length, i, j in edges:
        ri, rj = root(i), root(j)
        if ri == rj:
            continue
        parent[ri] = rj
        tree.append((i, j, length))
        if len(tree) == n - 1:
            break
    return tree


def _kind(n: int, edges: Iterable[tuple[int, int, float]]) -> str:
    degree = [0] * n
    for i, j, _ in edges:
        degree[i] += 1
        degree[j] += 1
    return "chain" if max(degree) <= 2 else "branch"


def _opposite_exchange_margin(points: list[list[int]]) -> float:
    """Closest one-edge MST exchange whose degree class is opposite."""
    n = len(points)
    tree = _mst(points)
    original = _kind(n, tree)
    tree_pairs = {tuple(sorted((i, j))) for i, j, _ in tree}
    adjacency: list[list[tuple[int, float]]] = [[] for _ in range(n)]
    for i, j, length in tree:
        adjacency[i].append((j, length))
        adjacency[j].append((i, length))

    best = float("inf")
    for u in range(n):
        for v in range(u + 1, n):
            if (u, v) in tree_pairs:
                continue
            stack = [(u, -1, [])]
            path: list[tuple[int, int, float]] | None = None
            while stack:
                node, previous, used = stack.pop()
                if node == v:
                    path = used
                    break
                for nxt, length in adjacency[node]:
                    if nxt != previous:
                        stack.append((nxt, node, used + [(node, nxt, length)]))
            assert path is not None
            added = _distance(points, u, v)
            for a, b, removed in path:
                alternative = [edge for edge in tree if tuple(sorted(edge[:2])) != tuple(sorted((a, b)))]
                alternative.append((u, v, added))
                if _kind(n, alternative) != original:
                    best = min(best, max(0.0, added - removed))
    if math.isinf(best):
        # This is a conservative robustness score for rare multiply-branched trees.
        lengths = sorted(_distance(points, i, j) for i in range(n) for j in range(i + 1, n))
        best = min(b - a for a, b in zip(lengths, lengths[1:]))
    return best


def _valid_spacing(points: list[list[int]], minimum: float = 30.0) -> bool:
    return all(
        _distance(points, i, j) >= minimum
        for i in range(len(points))
        for j in range(i + 1, len(points))
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = "chain" if int(seed) % 2 == 0 else "branch"
    n = 7 + ((int(seed) // 2) % 3)
    fallback = None
    for _ in range(30000):
        points = [[rng.randint(58, 454), rng.randint(58, 454)] for _ in range(n)]
        if not _valid_spacing(points):
            continue
        tree = _mst(points)
        if _kind(n, tree) != target:
            continue
        fallback = points
        # Keep most scenes comfortably resolvable while retaining naturally
        # occurring low-margin cases for explicit quarantine coverage.
        if _opposite_exchange_margin(points) >= 0.20:
            return {"points": points, "dot_radius": 8}
    if fallback is None:
        raise RuntimeError("could not sample the requested network topology")
    return {"points": fallback, "dot_radius": 8}


def render(scene: dict) -> Image.Image:
    points = scene["points"]
    radius = int(scene["dot_radius"])
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (255, 255, 255))
    draw = ImageDraw.Draw(canvas)
    for x, y in points:
        box = ((x - radius) * SCALE, (y - radius) * SCALE,
               (x + radius) * SCALE, (y + radius) * SCALE)
        draw.ellipse(box, fill=(18, 22, 28))
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    return _kind(len(points), _mst(points))


def margin(scene: dict) -> float:
    points = scene["points"]
    return _opposite_exchange_margin(points)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["points"] = list(reversed(scene["points"]))
    return [("reverse_point_order", twin)]
