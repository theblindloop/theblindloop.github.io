from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

BLACK = np.array([32, 36, 42], dtype=np.int16)
ORANGE = np.array([242, 142, 43], dtype=np.int16)
BLUE = np.array([40, 120, 199], dtype=np.int16)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _centers(mask: np.ndarray) -> list[list[float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    points: list[list[float]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 45:
            points.append([float(np.mean(xs)), float(np.mean(ys))])
    if len(points) < 6:
        raise ValueError("insufficient black dot components")
    return points


def _fit_line(mask: np.ndarray) -> tuple[float, float, float]:
    coords = np.argwhere(mask)[:, ::-1].astype(float)
    if len(coords) < 100:
        raise ValueError("candidate line is missing")
    mean = coords.mean(axis=0)
    covariance = np.cov(coords - mean, rowvar=False)
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    nx, ny = -float(direction[1]), float(direction[0])
    return nx, ny, nx * float(mean[0]) + ny * float(mean[1])


def _cross(o: list[float], a: list[float], b: list[float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[list[float]]) -> list[list[float]]:
    pts = sorted(points)
    lower: list[list[float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[list[float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _area(poly: list[list[float]]) -> float:
    return abs(sum(p[0] * q[1] - q[0] * p[1] for p, q in zip(poly, poly[1:] + poly[:1]))) / 2


def _clip(poly: list[list[float]], nx: float, ny: float, c: float) -> list[list[float]]:
    result: list[list[float]] = []
    for p, q in zip(poly, poly[1:] + poly[:1]):
        a = nx * p[0] + ny * p[1] - c
        b = nx * q[0] + ny * q[1] - c
        if a >= 0:
            result.append(p)
        if (a >= 0) != (b >= 0):
            t = a / (a - b)
            result.append([p[0] + t * (q[0] - p[0]), p[1] + t * (q[1] - p[1])])
    return result


def _imbalance(poly: list[list[float]], line: tuple[float, float, float]) -> float:
    total = _area(poly)
    positive = _area(_clip(poly, *line))
    return abs(2 * positive - total) / total


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    poly = _hull(_centers(_mask(arr, BLACK)))
    scores = {
        "orange": _imbalance(poly, _fit_line(_mask(arr, ORANGE))),
        "blue": _imbalance(poly, _fit_line(_mask(arr, BLUE))),
    }
    ordered = sorted((value, color) for color, value in scores.items())
    if ordered[1][0] - ordered[0][0] < 0.055:
        raise ValueError("ambiguous raster area partition")
    return ordered[0][1]
