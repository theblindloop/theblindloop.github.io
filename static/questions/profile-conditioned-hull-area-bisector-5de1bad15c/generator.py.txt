from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BLACK = "#20242a"
ORANGE = "#f28e2b"
BLUE = "#2878c7"
BACKGROUND = "#faf9f5"


def _cross(o: list[float], a: list[float], b: list[float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: Iterable[list[float]]) -> list[list[float]]:
    pts = sorted([list(map(float, p)) for p in points])
    lower: list[list[float]] = []
    for p in pts:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[list[float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _area(poly: list[list[float]]) -> float:
    return abs(sum(p[0] * q[1] - q[0] * p[1] for p, q in zip(poly, poly[1:] + poly[:1]))) / 2


def _clip(poly: list[list[float]], nx: float, ny: float, c: float) -> list[list[float]]:
    out: list[list[float]] = []
    for p, q in zip(poly, poly[1:] + poly[:1]):
        fp = nx * p[0] + ny * p[1] - c
        fq = nx * q[0] + ny * q[1] - c
        pin, qin = fp >= 0, fq >= 0
        if pin:
            out.append(p)
        if pin != qin:
            t = fp / (fp - fq)
            out.append([p[0] + t * (q[0] - p[0]), p[1] + t * (q[1] - p[1])])
    return out


def _imbalance(points: list[list[float]], line: dict) -> float:
    poly = _hull(points)
    total = _area(poly)
    theta = float(line["angle"])
    nx, ny = -math.sin(theta), math.cos(theta)
    c = nx * float(line["point"][0]) + ny * float(line["point"][1])
    positive = _area(_clip(poly, nx, ny, c))
    return abs(2 * positive - total) / total


def _scores(scene: dict) -> list[tuple[float, str]]:
    return sorted((_imbalance(scene["dots"], line), str(line["color"])) for line in scene["lines"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cx = rng.uniform(238, 274)
    cy = rng.uniform(238, 274)
    base = rng.uniform(-0.22, 0.22)
    dots: list[list[float]] = []
    # Opposite pairs make the convex envelope centrally symmetric while its
    # silhouette, orientation, size, and vertex count remain randomized.
    for i in range(5):
        angle = base + i * math.pi / 5 + rng.uniform(-0.055, 0.055)
        radius = rng.uniform(126, 171)
        dx, dy = radius * math.cos(angle), radius * math.sin(angle)
        dots.extend([[round(cx + dx, 3), round(cy + dy, 3)], [round(cx - dx, 3), round(cy - dy, 3)]])
    for _ in range(2):
        angle = rng.uniform(0, math.pi)
        radius = rng.uniform(38, 88)
        dx, dy = radius * math.cos(angle), radius * math.sin(angle)
        dots.extend([[round(cx + dx, 3), round(cy + dy, 3)], [round(cx - dx, 3), round(cy - dy, 3)]])
    rng.shuffle(dots)

    correct_color = "orange" if int(seed) % 2 == 0 else "blue"
    wrong_color = "blue" if correct_color == "orange" else "orange"
    good_angle = rng.uniform(0.12, math.pi - 0.12)
    bad_angle = (good_angle + rng.uniform(0.55, 1.25)) % math.pi
    offset = rng.choice([-1, 1]) * rng.uniform(42, 72)
    bx = cx - math.sin(bad_angle) * offset
    by = cy + math.cos(bad_angle) * offset
    lines = [
        {"color": correct_color, "point": [round(cx, 3), round(cy, 3)], "angle": round(good_angle, 6)},
        {"color": wrong_color, "point": [round(bx, 3), round(by, 3)], "angle": round(bad_angle, 6)},
    ]
    rng.shuffle(lines)
    return {"dots": dots, "lines": lines, "width": WIDTH, "height": HEIGHT, "dot_radius": 6}


def _draw_dashed(draw: ImageDraw.ImageDraw, point: list[float], angle: float, color: str) -> None:
    dx, dy = math.cos(angle), math.sin(angle)
    for start in range(-760, 761, 28):
        end = start + 17
        p = (point[0] + start * dx, point[1] + start * dy)
        q = (point[0] + end * dx, point[1] + end * dy)
        draw.line([p, q], fill=ORANGE if color == "orange" else BLUE, width=5)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for line in scene["lines"]:
        _draw_dashed(draw, line["point"], float(line["angle"]), str(line["color"]))
    r = int(scene["dot_radius"])
    for x, y in scene["dots"]:
        draw.ellipse([x - r, y - r, x + r, y + r], fill=BLACK)
    return image


def analytic_gold(scene: dict) -> str:
    return _scores(scene)[0][1]


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return scores[1][0] - scores[0][0]


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return scores[0][0] > 0.025 or scores[1][0] - scores[0][0] < 0.10


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {key: value for key, value in scene.items()}
    twin["dots"] = [list(p) for p in scene["dots"]]
    twin["dots"][0], twin["dots"][1] = twin["dots"][1], twin["dots"][0]
    twin["lines"] = [{**line, "point": list(line["point"])} for line in scene["lines"]]
    return [("permute_same_appearance_dots", twin)]
