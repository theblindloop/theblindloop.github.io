"""Deterministic renderer and analytic arm for Annular Cell Unfurl."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


CANVAS = 512
BACKGROUND = (248, 247, 242)
CYAN = (16, 166, 190)
MAGENTA = (218, 48, 122)
TEMPLATES: dict[str, tuple[str, ...]] = {
    "hook": ("10000", "10000", "10000", "10001", "01110"),
    "bolt": ("11100", "00100", "00100", "00010", "00011"),
    "fork": ("10001", "10001", "01110", "00100", "00000"),
}
INKS = ((35, 45, 62), (73, 43, 91), (31, 75, 66))
LABELS = tuple(TEMPLATES)


def _matrix(name: str) -> list[list[int]]:
    return [[int(value) for value in row] for row in TEMPLATES[name]]


def _distances(cells: list[list[int]]) -> dict[str, int]:
    return {
        name: sum(
            int(int(template_row[x]) != int(cells[y][x]))
            for y, template_row in enumerate(rows)
            for x in range(5)
        )
        for name, rows in TEMPLATES.items()
    }


def _classify(cells: list[list[int]]) -> tuple[str, float]:
    distances = _distances(cells)
    ordered = sorted((distance, name) for name, distance in distances.items())
    gap = float(ordered[1][0] - ordered[0][0])
    return ("ambiguous" if gap == 0 else ordered[0][1], gap)


def _perturb_equal_ink(cells: list[list[int]], target: str, runner: str, swaps: int) -> None:
    target_rows = TEMPLATES[target]
    runner_rows = TEMPLATES[runner]
    remove = []
    add = []
    for y in range(5):
        for x in range(5):
            left, right = int(target_rows[y][x]), int(runner_rows[y][x])
            if left == 1 and right == 0:
                remove.append((y, x))
            elif left == 0 and right == 1:
                add.append((y, x))
    for index in range(swaps):
        ry, rx = remove[index]
        ay, ax = add[index]
        cells[ry][rx] = 0
        cells[ay][ax] = 1


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    target_index = seed % 3
    target = LABELS[target_index]
    runner = LABELS[(target_index + 1) % 3]
    swap_cycle = (0, 1, 2, 0, 1, 3)
    swaps = swap_cycle[(seed // 3) % len(swap_cycle)]
    cells = _matrix(target)
    _perturb_equal_ink(cells, target, runner, swaps)

    rng = random.Random(seed * 1_000_003 + 91_127)
    centers = (96, 160, 224, 288, 352, 416)
    center = [rng.choice(centers), rng.choice(centers)]
    outer_radius = rng.choice((36, 38, 40))
    rotation = math.radians(rng.randrange(20) * 18)
    ink_rgb = list(INKS[rng.randrange(len(INKS))])

    distractors = []
    for _ in range(200):
        if len(distractors) >= 9:
            break
        radius = rng.choice((3, 4, 5))
        x = rng.randrange(28, CANVAS - 27)
        y = rng.randrange(28, CANVAS - 27)
        if math.hypot(x - center[0], y - center[1]) < outer_radius + radius + 36:
            continue
        if any(math.hypot(x - item[0], y - item[1]) < radius + item[2] + 6 for item in distractors):
            continue
        shade = rng.choice((207, 215, 223))
        distractors.append([x, y, radius, shade])
    if len(distractors) < 4:
        raise RuntimeError("could not place neutral distractors")

    return {
        "cells": cells,
        "center": center,
        "outer_radius": outer_radius,
        "rotation": rotation,
        "ink_rgb": ink_rgb,
        "distractors": distractors,
    }


def _point(cx: float, cy: float, radius: float, angle: float) -> tuple[float, float]:
    return (cx + radius * math.cos(angle), cy + radius * math.sin(angle))


def _sector_polygon(
    cx: float, cy: float, inner: float, outer: float, start: float, end: float
) -> list[tuple[float, float]]:
    steps = 12
    outside = [_point(cx, cy, outer, start + (end - start) * i / steps) for i in range(steps + 1)]
    inside = [_point(cx, cy, inner, start + (end - start) * i / steps) for i in range(steps, -1, -1)]
    return outside + inside


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius, shade in scene["distractors"]:
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=(shade, shade, shade),
        )

    cx, cy = (float(value) for value in scene["center"])
    outer = float(scene["outer_radius"])
    inner = outer / 4.0
    band = (outer - inner) / 5.0
    start = float(scene["rotation"])
    ink = tuple(int(value) for value in scene["ink_rgb"])
    cells = scene["cells"]

    for row in range(5):
        cell_outer = outer - row * band
        cell_inner = outer - (row + 1) * band
        for column in range(5):
            if int(cells[row][column]):
                angle0 = start + column * 2.0 * math.pi / 5.0
                angle1 = start + (column + 1) * 2.0 * math.pi / 5.0
                draw.polygon(
                    _sector_polygon(cx, cy, cell_inner, cell_outer, angle0, angle1),
                    fill=ink,
                )

    grid_width = 2
    for index in range(6):
        radius = inner + index * band
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            outline=CYAN,
            width=grid_width,
        )
    for column in range(5):
        angle = start + column * 2.0 * math.pi / 5.0
        draw.line(
            (*_point(cx, cy, inner, angle), *_point(cx, cy, outer, angle)),
            fill=CYAN,
            width=grid_width,
        )

    marker_angle = start
    left = _point(cx, cy, outer + 2, marker_angle - 0.11)
    right = _point(cx, cy, outer + 2, marker_angle + 0.11)
    tip = _point(cx, cy, outer + 9, marker_angle)
    draw.polygon((left, tip, right), fill=MAGENTA)
    return image


def analytic_gold(scene: dict) -> str:
    return _classify(scene["cells"])[0]


def margin(scene: dict) -> float:
    return _classify(scene["cells"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_disjoint_distractor_order", twin)]
