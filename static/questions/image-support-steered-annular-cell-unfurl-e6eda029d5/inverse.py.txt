"""Independent PNG-only inverse arm for Annular Cell Unfurl.

The implementation locates the visible cyan polar grid and magenta start notch,
estimates its radial bounds, samples the centers of the declared curved cells,
unfurls them into a 5-by-5 binary raster, and compares that raster with the
public answer templates. It imports no renderer or analytic-gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


TEMPLATES: dict[str, tuple[str, ...]] = {
    "hook": ("10000", "10000", "10000", "10001", "01110"),
    "bolt": ("11100", "00100", "00100", "00010", "00011"),
    "fork": ("10001", "10001", "01110", "00100", "00000"),
}


def _mask_centroid(mask: np.ndarray) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) == 0:
        raise ValueError("required visible fiducial is missing")
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _sample_dark(array: np.ndarray, x: float, y: float) -> int:
    ix, iy = int(round(x)), int(round(y))
    if not (2 <= ix < array.shape[1] - 2 and 2 <= iy < array.shape[0] - 2):
        raise ValueError("polar cell sample leaves the raster")
    patch = array[iy - 1 : iy + 2, ix - 1 : ix + 2].astype(np.float32)
    return int(float(patch.mean()) < 155.0)


def analyze_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    cyan = (red < 80) & (green > 120) & (blue > 145) & (blue > green)
    magenta = (red > 160) & (green < 100) & (blue > 80)
    if int(cyan.sum()) < 180:
        raise ValueError("cyan annular grid is missing or too incomplete")
    if int(magenta.sum()) < 8:
        raise ValueError("magenta start notch is missing or too incomplete")

    cx, cy = _mask_centroid(cyan)
    mx, my = _mask_centroid(magenta)
    start = math.atan2(my - cy, mx - cx)
    points = np.argwhere(cyan)
    radii = np.hypot(points[:, 1] - cx, points[:, 0] - cy)
    inner = float(np.quantile(radii, 0.002))
    outer = float(np.quantile(radii, 0.998))
    if not 24.0 <= outer - inner <= 36.0 or outer < 32.0:
        raise ValueError("visible annular grid has inconsistent radial bounds")

    cells: list[list[int]] = []
    for row in range(5):
        radius = outer - (row + 0.5) * (outer - inner) / 5.0
        values = []
        for column in range(5):
            angle = start + (column + 0.5) * 2.0 * math.pi / 5.0
            x = cx + radius * math.cos(angle)
            y = cy + radius * math.sin(angle)
            values.append(_sample_dark(array, x, y))
        cells.append(values)

    distances = {
        name: sum(
            int(int(template_row[x]) != cells[y][x])
            for y, template_row in enumerate(rows)
            for x in range(5)
        )
        for name, rows in TEMPLATES.items()
    }
    ordered = sorted((distance, name) for name, distance in distances.items())
    gap = int(ordered[1][0] - ordered[0][0])
    decision = "ambiguous" if gap == 0 else ordered[0][1]
    return {
        "decision": decision,
        "margin": float(gap),
        "distances": distances,
        "cells": cells,
        "center": [round(cx, 3), round(cy, 3)],
        "radial_bounds": [round(inner, 3), round(outer, 3)],
    }


def decision_from_image(image: Image.Image) -> str:
    return analyze_image(image)["decision"]
