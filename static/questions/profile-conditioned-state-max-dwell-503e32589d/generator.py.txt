"""Temporal region-dwell argmax: which token is inside the shaded region for the
most panels (state tracking).

Four ordered panels are drawn left to right (P1, P2, P3, P4). Every panel shows
the same fixed five-slot track, and the middle slot column (slot index 2) is
always the same shaded vertical region. Four coloured tokens -- red, green,
blue, orange -- each keep their own fixed colour and occupy one slot in every
panel (slots 0..4 left to right; at most one token may sit in the shaded middle
slot per panel, so no two tokens overlap). A token is *inside the region* in a
given panel exactly when it sits in the middle (shaded) slot of that panel.

The question is: counting the four ordered panels P1..P4, which colour token is
inside the shaded region for the MOST panels? The answer is the argmax over the
four tokens of their per-token region-dwell counts. The construction always
produces a strict (unique) argmax, so no scene is quarantined.

The latent scene stores ``slots``: a 4x4 list, where ``slots[panel][token]`` is
the slot index (0..4) that token occupies in that panel. The token order maps to
fixed colours via the module constant ``COLORS``. All latent fields are
render-evident, so no nontrivial render-invariant latent symmetry exists.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 260
PANEL_W = 200
PANEL_H = 180
PANEL_TOP = 40
PANEL_GAP = 240
SLOT_X_FRAC = (0.1, 0.3, 0.5, 0.7, 0.9)
MIDDLE_SLOT = 2
DISK_R = 14

BG = (255, 255, 255)
BORDER = (200, 200, 200)
REGION_FILL = (235, 235, 235)

COLOR_NAMES = ["red", "green", "blue", "orange"]
COLORS = {
    "red": (220, 40, 40),
    "green": (40, 170, 80),
    "blue": (40, 80, 220),
    "orange": (235, 150, 20),
}


def _panel_left(index: int) -> int:
    return 30 + index * PANEL_GAP


def _slot_x(panel_left: int, slot: int) -> int:
    return int(panel_left + PANEL_W * SLOT_X_FRAC[slot])


def _counts(slots: list) -> list[int]:
    """Per-token count of panels where the token sits in the middle (shaded)
    region slot (slot index 2)."""
    return [sum(1 for panel in range(len(slots)) if slots[panel][token] == MIDDLE_SLOT) for token in range(len(COLORS))]


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene with a strict (unique) argmax dwell count."""
    rng = random.Random(seed)

    while True:
        argmax = rng.randrange(4)
        c_max = rng.randint(1, 4)
        others = []
        for token in range(4):
            if token == argmax:
                continue
            others.append(rng.randint(0, c_max - 1))
        counts = [0, 0, 0, 0]
        counts[argmax] = c_max
        k = 0
        for token in range(4):
            if token != argmax:
                counts[token] = others[k]
                k += 1
        # At most one token can occupy the shaded middle slot in a given panel,
        # so across 4 panels the sum of all dwell counts cannot exceed 4.
        if sum(counts) <= 4:
            break

    # Build the per-panel inside/outside schedule.
    pool = []
    for token in range(4):
        pool += [token] * counts[token]
    rng.shuffle(pool)
    inside_panels = rng.sample(range(4), sum(counts))
    inside_at = {panel: token for panel, token in zip(inside_panels, pool)}

    slots = []
    for panel in range(4):
        row = [None] * 4
        token_in = inside_at.get(panel)
        if token_in is not None:
            row[token_in] = MIDDLE_SLOT
        outside = [token for token in range(4) if row[token] is None]
        rng.shuffle(outside)
        outside_slots = rng.sample([0, 1, 3, 4], len(outside))
        for token, slot in zip(outside, outside_slots):
            row[token] = slot
        slots.append(row)

    return {"slots": slots}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    slots = scene["slots"]
    center_y = PANEL_TOP + PANEL_H // 2

    for panel in range(4):
        left = _panel_left(panel)
        # shaded middle-slot region (a vertical band over slot index 2)
        mid_x = _slot_x(left, MIDDLE_SLOT)
        band_half = 18
        draw.rectangle(
            [mid_x - band_half, PANEL_TOP, mid_x + band_half, PANEL_TOP + PANEL_H],
            fill=REGION_FILL,
        )
        # panel border
        draw.rectangle(
            [left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H],
            outline=BORDER,
            width=3,
        )
        # tokens
        row = slots[panel]
        for token, slot in enumerate(row):
            cx = _slot_x(left, slot)
            color = COLORS[COLOR_NAMES[token]]
            box = [cx - DISK_R, center_y - DISK_R, cx + DISK_R, center_y + DISK_R]
            draw.ellipse(box, fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    counts = _counts(scene["slots"])
    return COLOR_NAMES[max(range(4), key=lambda token: counts[token])]


def margin(scene: dict) -> float:
    """Discrete one-sided decision margin: the gap between the top dwell count
    and the runner-up count."""
    counts = sorted(_counts(scene["slots"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    # Construction guarantees a strict (unique) argmax, so nothing is ambiguous.
    return False


def latent_symmetries(scene: dict) -> list:
    # Every latent field (slots) directly fixes the drawn per-panel positions;
    # any reparametrization that changed which token is the max-dwell token
    # would change the raster. There is no nontrivial render-invariant symmetry.
    return []
