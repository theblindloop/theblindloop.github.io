"""Independent PNG-only inverse for the disjoint portal linkage world."""

from __future__ import annotations

from PIL import Image
import numpy as np

WIDTH = 1040
HEIGHT = 720
ROWS = 11
COLS = 15
CELL = 38
ORIGIN_X = 55
ORIGIN_Y = 178
OPEN_RGB = (248, 250, 252)
WALL_RGB = (35, 49, 67)
DOT_RGB = (25, 42, 65)
START_RGB = {
    "pink": (211, 54, 111),
    "teal": (0, 145, 151),
    "gold": (226, 137, 20),
}
EXIT_RGB = {"1": (73, 91, 185), "2": (58, 151, 82), "3": (202, 67, 54)}
PORTAL_RGB = {"A": (49, 105, 215), "B": (139, 72, 190)}
START_NAMES = tuple(START_RGB)
EXIT_NAMES = tuple(EXIT_RGB)
OPTION_LABELS = "ABCD"
CARD_X = (670, 850)
CARD_Y = (178, 410)
MATRIX_DX = 58
MATRIX_DY = 82
OPTION_CELL = 27


class OracleError(ValueError):
    pass


def _near(patch: np.ndarray, rgb: tuple[int, int, int], radius: float = 30.0) -> int:
    delta = patch.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return int((np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius).sum())


def _recover(image: Image.Image) -> tuple[
    set[tuple[int, int]],
    dict[str, tuple[int, int]],
    dict[str, tuple[int, int]],
    dict[str, tuple[tuple[int, int], tuple[int, int]]],
    list[tuple[str, str, str]],
]:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    special_palette = {
        **{f"start:{name}": rgb for name, rgb in START_RGB.items()},
        **{f"exit:{name}": rgb for name, rgb in EXIT_RGB.items()},
        **{f"portal:{name}": rgb for name, rgb in PORTAL_RGB.items()},
    }
    found: dict[str, list[tuple[int, int]]] = {name: [] for name in special_palette}
    open_cells: set[tuple[int, int]] = set()
    for row in range(ROWS):
        for col in range(COLS):
            cx = ORIGIN_X + col * CELL + CELL // 2
            cy = ORIGIN_Y + row * CELL + CELL // 2
            patch = arr[cy - 11 : cy + 12, cx - 11 : cx + 12]
            scores = {name: _near(patch, rgb) for name, rgb in special_palette.items()}
            best = max(scores, key=scores.get)
            ranked = sorted(scores.values(), reverse=True)
            if scores[best] >= 100 and scores[best] >= 2 * max(1, ranked[1]):
                found[best].append((row, col))
                open_cells.add((row, col))
                continue
            open_score = _near(patch, OPEN_RGB)
            wall_score = _near(patch, WALL_RGB)
            if open_score >= 430 and open_score >= 5 * max(1, wall_score):
                open_cells.add((row, col))
            elif wall_score >= 430 and wall_score >= 5 * max(1, open_score):
                pass
            else:
                raise OracleError(f"cell {(row, col)} has no stable interior class")
    starts: dict[str, tuple[int, int]] = {}
    exits: dict[str, tuple[int, int]] = {}
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for name in START_NAMES:
        values = found[f"start:{name}"]
        if len(values) != 1:
            raise OracleError(f"expected one {name} start, found {len(values)}")
        starts[name] = values[0]
    for name in EXIT_NAMES:
        values = found[f"exit:{name}"]
        if len(values) != 1:
            raise OracleError(f"expected one exit {name}, found {len(values)}")
        exits[name] = values[0]
    for name in PORTAL_RGB:
        values = found[f"portal:{name}"]
        if len(values) != 2:
            raise OracleError(f"expected two {name} portals, found {len(values)}")
        portals[name] = (values[0], values[1])

    options: list[tuple[str, str, str]] = []
    for index in range(4):
        card_x = CARD_X[index % 2]
        card_y = CARD_Y[index // 2]
        matrix_x = card_x + MATRIX_DX
        matrix_y = card_y + MATRIX_DY
        values: list[str] = []
        for row_index in range(3):
            hits: list[str] = []
            for col_index, exit_name in enumerate(EXIT_NAMES):
                cx = int(round(matrix_x + col_index * OPTION_CELL + OPTION_CELL / 2))
                cy = int(round(matrix_y + row_index * OPTION_CELL + OPTION_CELL / 2))
                patch = arr[cy - 5 : cy + 6, cx - 5 : cx + 6]
                score = _near(patch, DOT_RGB, radius=25.0)
                if score >= 70:
                    hits.append(exit_name)
                elif score > 5:
                    raise OracleError(f"option {OPTION_LABELS[index]} has ambiguous dot at {(row_index, col_index)}")
            if len(hits) != 1:
                raise OracleError(f"option {OPTION_LABELS[index]} row {row_index} has {len(hits)} dots")
            values.append(hits[0])
        if set(values) != set(EXIT_NAMES):
            raise OracleError(f"option {OPTION_LABELS[index]} is not a permutation")
        options.append(tuple(values))  # type: ignore[arg-type]
    if len(set(options)) != 4:
        raise OracleError("option cards are not distinct")
    return open_cells, starts, exits, portals, options


def _graph(open_cells, portals):
    graph = {cell: set() for cell in open_cells}
    for row, col in open_cells:
        for neighbor in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            if neighbor in open_cells:
                graph[(row, col)].add(neighbor)
    for left, right in portals.values():
        graph[left].add(right)
        graph[right].add(left)
    return graph


def _simple_paths(graph, start, target, limit=64):
    found = []

    def visit(node, path, seen):
        if len(found) >= limit:
            return
        if node == target:
            found.append(tuple(path))
            return
        for neighbor in sorted(graph[node]):
            if neighbor not in seen:
                visit(neighbor, path + [neighbor], seen | {neighbor})

    visit(start, [start], {start})
    return found


def _feasible(open_cells, starts, exits, portals, option):
    graph = _graph(open_cells, portals)
    path_sets = []
    for index, start_name in enumerate(START_NAMES):
        paths = _simple_paths(graph, starts[start_name], exits[option[index]])
        if not paths:
            return False
        path_sets.append(paths)
    order = sorted(range(3), key=lambda index: len(path_sets[index]))

    def allocate(depth, used):
        if depth == len(order):
            return True
        index = order[depth]
        for path in path_sets[index]:
            cells = set(path)
            if cells.isdisjoint(used) and allocate(depth + 1, used | cells):
                return True
        return False

    return allocate(0, set())


def decision_from_image(image: Image.Image) -> str:
    open_cells, starts, exits, portals, options = _recover(image)
    feasible = [
        OPTION_LABELS[index]
        for index, option in enumerate(options)
        if _feasible(open_cells, starts, exits, portals, option)
    ]
    if len(feasible) != 1:
        raise OracleError(f"expected one feasible displayed linkage, found {feasible}")
    return feasible[0]
