"""Deterministic portal maze with a unique displayed three-pair linkage."""

from __future__ import annotations

import random
from itertools import permutations
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1040
HEIGHT = 720
SCALE = 3
ROWS = 11
COLS = 15
CELL = 38
ORIGIN_X = 55
ORIGIN_Y = 178

OPEN_RGB = (248, 250, 252)
WALL_RGB = (35, 49, 67)
GRID_RGB = (104, 118, 134)
INK_RGB = (24, 34, 47)
MUTED_RGB = (88, 103, 120)
DOT_RGB = (25, 42, 65)
CARD_RGB = (255, 255, 255)
START_RGB = {
    "pink": (211, 54, 111),
    "teal": (0, 145, 151),
    "gold": (226, 137, 20),
}
EXIT_RGB = {
    "1": (73, 91, 185),
    "2": (58, 151, 82),
    "3": (202, 67, 54),
}
PORTAL_RGB = {
    "A": (49, 105, 215),
    "B": (139, 72, 190),
}
START_NAMES = tuple(START_RGB)
EXIT_NAMES = tuple(EXIT_RGB)
OPTION_LABELS = "ABCD"
CARD_X = (670, 850)
CARD_Y = (178, 410)
CARD_W = 160
CARD_H = 190
MATRIX_DX = 58
MATRIX_DY = 82
OPTION_CELL = 27


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _transform(cell: tuple[int, int], variant: int) -> tuple[int, int]:
    row, col = cell
    if variant & 1:
        col = COLS - 1 - col
    if variant & 2:
        row = ROWS - 1 - row
    return row, col


def _base_geometry() -> tuple[
    set[tuple[int, int]],
    list[tuple[int, int]],
    list[tuple[int, int]],
    dict[str, list[tuple[int, int]]],
]:
    # Two spatially separate snakes become one cycle only after both portal
    # identifications are added.  Short branches are topological dead ends.
    upper = (
        [(1, col) for col in range(2, 7)]
        + [(row, 6) for row in range(2, 5)]
        + [(4, col) for col in range(7, 12)]
        + [(row, 11) for row in (3, 2)]
        + [(2, col) for col in (12, 13)]
    )
    lower = (
        [(9, col) for col in range(2, 6)]
        + [(row, 5) for row in (8, 7)]
        + [(7, col) for col in range(6, 10)]
        + [(row, 9) for row in (8, 9, 10)]
        + [(10, col) for col in range(10, 14)]
    )
    branches = [
        [(1, 4), (0, 4)],
        [(4, 8), (5, 8), (5, 9)],
        [(7, 7), (6, 7), (6, 6)],
        [(9, 3), (10, 3)],
    ]
    open_cells = set(upper) | set(lower) | {cell for branch in branches for cell in branch}

    # Around the portal-completed cycle the order is S0,E0,S1,E1,S2,E2.
    start_roles = [upper[3], upper[13], lower[9]]
    exit_roles = [upper[8], lower[14], lower[4]]
    portals = {"A": [upper[0], lower[0]], "B": [upper[-1], lower[-1]]}
    return open_cells, start_roles, exit_roles, portals


def _graph(
    open_cells: set[tuple[int, int]],
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
) -> dict[tuple[int, int], set[tuple[int, int]]]:
    graph = {cell: set() for cell in open_cells}
    for row, col in open_cells:
        for neighbor in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            if neighbor in open_cells:
                graph[(row, col)].add(neighbor)
    for left, right in portals.values():
        graph[left].add(right)
        graph[right].add(left)
    return graph


def _simple_paths(
    graph: dict[tuple[int, int], set[tuple[int, int]]],
    start: tuple[int, int],
    target: tuple[int, int],
    limit: int = 64,
) -> list[tuple[tuple[int, int], ...]]:
    found: list[tuple[tuple[int, int], ...]] = []

    def visit(node: tuple[int, int], path: list[tuple[int, int]], seen: set[tuple[int, int]]) -> None:
        if len(found) >= limit:
            return
        if node == target:
            found.append(tuple(path))
            return
        for neighbor in sorted(graph[node]):
            if neighbor not in seen:
                visit(neighbor, path + [neighbor], seen | {neighbor})

    visit(start, [start], {start})
    return found


def _linkage_feasible(
    open_cells: set[tuple[int, int]],
    starts: dict[str, tuple[int, int]],
    exits: dict[str, tuple[int, int]],
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]],
    option: tuple[str, str, str] | list[str],
) -> bool:
    graph = _graph(open_cells, portals)
    path_sets: list[list[tuple[tuple[int, int], ...]]] = []
    for index, start_name in enumerate(START_NAMES):
        paths = _simple_paths(graph, starts[start_name], exits[str(option[index])])
        if not paths:
            return False
        path_sets.append(paths)
    order = sorted(range(3), key=lambda index: len(path_sets[index]))

    def allocate(depth: int, used: set[tuple[int, int]]) -> bool:
        if depth == len(order):
            return True
        index = order[depth]
        for path in path_sets[index]:
            cells = set(path)
            if cells.isdisjoint(used) and allocate(depth + 1, used | cells):
                return True
        return False

    return allocate(0, set())


def sample_scene(seed: int) -> dict[str, Any]:
    seed = int(seed)
    rng = random.Random(seed)
    open_cells, start_roles, exit_roles, raw_portals = _base_geometry()
    variant = (seed // 4) % 4
    transform = lambda cell: _transform(cell, variant)

    start_assignment = list(START_NAMES)
    exit_assignment = list(EXIT_NAMES)
    rng.shuffle(start_assignment)
    rng.shuffle(exit_assignment)
    starts = {
        name: transform(cell) for name, cell in zip(start_assignment, start_roles)
    }
    exits = {
        name: transform(cell) for name, cell in zip(exit_assignment, exit_roles)
    }
    portals = {
        letter: (transform(pair[0]), transform(pair[1]))
        for letter, pair in raw_portals.items()
    }
    transformed_open = {transform(cell) for cell in open_cells}

    all_options = [tuple(values) for values in permutations(EXIT_NAMES)]
    feasible = [
        option
        for option in all_options
        if _linkage_feasible(transformed_open, starts, exits, portals, option)
    ]
    infeasible = [option for option in all_options if option not in feasible]
    if len(feasible) != 2 or len(infeasible) != 4:
        raise RuntimeError("base topology does not have the expected linkage classes")
    target = feasible[(seed // 16) % len(feasible)]
    decoys = rng.sample(infeasible, 3)
    target_index = seed % 4
    options = list(decoys)
    options.insert(target_index, target)

    return {
        "open_cells": [list(cell) for cell in sorted(transformed_open)],
        "starts": {name: list(starts[name]) for name in START_NAMES},
        "exits": {name: list(exits[name]) for name in EXIT_NAMES},
        "portals": {
            letter: [list(portals[letter][0]), list(portals[letter][1])]
            for letter in PORTAL_RGB
        },
        "options": [list(option) for option in options],
    }


def _cell(value: object) -> tuple[int, int] | None:
    if not isinstance(value, list) or len(value) != 2:
        return None
    if any(isinstance(item, bool) or not isinstance(item, int) for item in value):
        return None
    row, col = int(value[0]), int(value[1])
    if not (0 <= row < ROWS and 0 <= col < COLS):
        return None
    return row, col


def _decode(scene: dict[str, Any]) -> tuple[
    set[tuple[int, int]],
    dict[str, tuple[int, int]],
    dict[str, tuple[int, int]],
    dict[str, tuple[tuple[int, int], tuple[int, int]]],
    list[tuple[str, str, str]],
] | None:
    try:
        raw_open = scene["open_cells"]
        raw_starts = scene["starts"]
        raw_exits = scene["exits"]
        raw_portals = scene["portals"]
        raw_options = scene["options"]
    except (KeyError, TypeError):
        return None
    if not isinstance(raw_open, list) or not isinstance(raw_starts, dict):
        return None
    if not isinstance(raw_exits, dict) or not isinstance(raw_portals, dict):
        return None
    if set(raw_starts) != set(START_NAMES) or set(raw_exits) != set(EXIT_NAMES):
        return None
    if set(raw_portals) != set(PORTAL_RGB) or not isinstance(raw_options, list):
        return None
    converted = [_cell(value) for value in raw_open]
    if any(value is None for value in converted):
        return None
    open_cells = {value for value in converted if value is not None}
    if len(open_cells) != len(converted):
        return None
    starts = {name: _cell(raw_starts[name]) for name in START_NAMES}
    exits = {name: _cell(raw_exits[name]) for name in EXIT_NAMES}
    if any(value is None for value in starts.values()) or any(value is None for value in exits.values()):
        return None
    clean_starts = {name: value for name, value in starts.items() if value is not None}
    clean_exits = {name: value for name, value in exits.items() if value is not None}
    portals: dict[str, tuple[tuple[int, int], tuple[int, int]]] = {}
    for letter in PORTAL_RGB:
        pair = raw_portals[letter]
        if not isinstance(pair, list) or len(pair) != 2:
            return None
        cells = tuple(_cell(value) for value in pair)
        if any(value is None for value in cells) or cells[0] == cells[1]:
            return None
        portals[letter] = (cells[0], cells[1])  # type: ignore[arg-type]
    options: list[tuple[str, str, str]] = []
    if len(raw_options) != 4:
        return None
    for raw in raw_options:
        if not isinstance(raw, list) or len(raw) != 3:
            return None
        option = tuple(str(value) for value in raw)
        if set(option) != set(EXIT_NAMES):
            return None
        options.append(option)  # type: ignore[arg-type]
    if len(set(options)) != 4:
        return None
    special = [
        *clean_starts.values(), *clean_exits.values(),
        *(cell for pair in portals.values() for cell in pair),
    ]
    if len(special) != len(set(special)) or any(cell not in open_cells for cell in special):
        return None
    return open_cells, clean_starts, clean_exits, portals, options


def _feasible_labels(scene: dict[str, Any]) -> list[str]:
    decoded = _decode(scene)
    if decoded is None:
        return []
    open_cells, starts, exits, portals, options = decoded
    return [
        OPTION_LABELS[index]
        for index, option in enumerate(options)
        if _linkage_feasible(open_cells, starts, exits, portals, option)
    ]


def analytic_gold(scene: dict[str, Any]) -> str:
    labels = _feasible_labels(scene)
    if len(labels) != 1:
        raise ValueError(f"scene has {len(labels)} feasible displayed linkages")
    return labels[0]


def margin(scene: dict[str, Any]) -> float:
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    decoded = _decode(scene)
    if decoded is None or len(_feasible_labels(scene)) != 1:
        return True
    open_cells, starts, exits, portals, options = decoded
    graph = _graph(open_cells, portals)
    # A local reachability shortcut must be uninformative for every displayed pair.
    for option in options:
        for index, start_name in enumerate(START_NAMES):
            if not _simple_paths(graph, starts[start_name], exits[option[index]], limit=1):
                return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    text: str,
    font: ImageFont.FreeTypeFont,
    fill: tuple[int, int, int] | str,
) -> None:
    cx, cy = center
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((cx * SCALE - width / 2, cy * SCALE - height / 2 - 1 * SCALE), text, font=font, fill=fill)


def render(scene: dict[str, Any]) -> Image.Image:
    decoded = _decode(scene)
    if decoded is None:
        raise ValueError("cannot render malformed scene")
    open_cells, starts, exits, portals, options = decoded
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (237, 242, 247))
    draw = ImageDraw.Draw(image)
    draw.text((34 * SCALE, 20 * SCALE), "THREE ROUTES  •  NO SHARED CELLS", font=_font(23), fill=INK_RGB)
    draw.text((35 * SCALE, 60 * SCALE), "White cells touch  |  equal letters jump  |  route all three pairs at once", font=_font(14), fill=MUTED_RGB)

    # Visible endpoint key; the same colors and labels repeat inside the maze and cards.
    draw.text((55 * SCALE, 112 * SCALE), "STARTS", font=_font(12), fill=INK_RGB)
    for index, name in enumerate(START_NAMES):
        cx = 145 + index * 100
        draw.ellipse(((cx - 12) * SCALE, 107 * SCALE, (cx + 12) * SCALE, 131 * SCALE), fill=START_RGB[name], outline=INK_RGB, width=2 * SCALE)
        draw.text(((cx + 18) * SCALE, 109 * SCALE), name, font=_font(10), fill=INK_RGB)
    draw.text((430 * SCALE, 112 * SCALE), "EXITS", font=_font(12), fill=INK_RGB)
    for index, name in enumerate(EXIT_NAMES):
        cx = 505 + index * 48
        draw.rounded_rectangle(((cx - 12) * SCALE, 107 * SCALE, (cx + 12) * SCALE, 131 * SCALE), radius=4 * SCALE, fill=EXIT_RGB[name], outline=INK_RGB, width=2 * SCALE)
        _centered_text(draw, (cx, 119), name, _font(11), "white")

    left = ORIGIN_X * SCALE
    top = ORIGIN_Y * SCALE
    right = (ORIGIN_X + COLS * CELL) * SCALE
    bottom = (ORIGIN_Y + ROWS * CELL) * SCALE
    draw.rectangle((left - 5 * SCALE, top - 5 * SCALE, right + 5 * SCALE, bottom + 5 * SCALE), fill=WALL_RGB, outline=INK_RGB, width=2 * SCALE)
    for row in range(ROWS):
        for col in range(COLS):
            x0 = (ORIGIN_X + col * CELL) * SCALE
            y0 = (ORIGIN_Y + row * CELL) * SCALE
            x1 = x0 + CELL * SCALE
            y1 = y0 + CELL * SCALE
            draw.rectangle((x0, y0, x1, y1), fill=OPEN_RGB if (row, col) in open_cells else WALL_RGB)

    def center(cell: tuple[int, int]) -> tuple[int, int]:
        row, col = cell
        return ORIGIN_X + col * CELL + CELL // 2, ORIGIN_Y + row * CELL + CELL // 2

    for letter, pair in portals.items():
        for cell in pair:
            cx, cy = center(cell)
            draw.ellipse(((cx - 15) * SCALE, (cy - 15) * SCALE, (cx + 15) * SCALE, (cy + 15) * SCALE), fill=PORTAL_RGB[letter], outline=INK_RGB, width=2 * SCALE)
            _centered_text(draw, (cx, cy), letter, _font(15), "white")
    for name, cell in starts.items():
        cx, cy = center(cell)
        draw.ellipse(((cx - 15) * SCALE, (cy - 15) * SCALE, (cx + 15) * SCALE, (cy + 15) * SCALE), fill=START_RGB[name], outline=INK_RGB, width=2 * SCALE)
        _centered_text(draw, (cx, cy), "S", _font(14), "white")
    for name, cell in exits.items():
        cx, cy = center(cell)
        draw.rounded_rectangle(((cx - 15) * SCALE, (cy - 15) * SCALE, (cx + 15) * SCALE, (cy + 15) * SCALE), radius=5 * SCALE, fill=EXIT_RGB[name], outline=INK_RGB, width=2 * SCALE)
        _centered_text(draw, (cx, cy), name, _font(14), "white")
    for col in range(COLS + 1):
        x = (ORIGIN_X + col * CELL) * SCALE
        draw.line((x, top, x, bottom), fill=GRID_RGB, width=1 * SCALE)
    for row in range(ROWS + 1):
        y = (ORIGIN_Y + row * CELL) * SCALE
        draw.line((left, y, right, y), fill=GRID_RGB, width=1 * SCALE)

    draw.text((676 * SCALE, 112 * SCALE), "OPTIONS", font=_font(15), fill=INK_RGB)
    draw.text((676 * SCALE, 140 * SCALE), "rows = starts   columns = exits", font=_font(11), fill=MUTED_RGB)
    for index, option in enumerate(options):
        card_x = CARD_X[index % 2]
        card_y = CARD_Y[index // 2]
        draw.rounded_rectangle((card_x * SCALE, card_y * SCALE, (card_x + CARD_W) * SCALE, (card_y + CARD_H) * SCALE), radius=10 * SCALE, fill=CARD_RGB, outline=(117, 130, 146), width=2 * SCALE)
        draw.rounded_rectangle(((card_x + 10) * SCALE, (card_y + 10) * SCALE, (card_x + 42) * SCALE, (card_y + 42) * SCALE), radius=5 * SCALE, fill=INK_RGB)
        _centered_text(draw, (card_x + 26, card_y + 26), OPTION_LABELS[index], _font(15), "white")
        matrix_x = card_x + MATRIX_DX
        matrix_y = card_y + MATRIX_DY
        for col_index, exit_name in enumerate(EXIT_NAMES):
            cx = matrix_x + col_index * OPTION_CELL + OPTION_CELL / 2
            cy = card_y + 61
            draw.rounded_rectangle(((cx - 10) * SCALE, (cy - 10) * SCALE, (cx + 10) * SCALE, (cy + 10) * SCALE), radius=3 * SCALE, fill=EXIT_RGB[exit_name])
            _centered_text(draw, (cx, cy), exit_name, _font(9), "white")
        for row_index, start_name in enumerate(START_NAMES):
            cy = matrix_y + row_index * OPTION_CELL + OPTION_CELL / 2
            draw.ellipse(((card_x + 25 - 9) * SCALE, (cy - 9) * SCALE, (card_x + 25 + 9) * SCALE, (cy + 9) * SCALE), fill=START_RGB[start_name], outline=INK_RGB, width=1 * SCALE)
        for offset in range(4):
            x = (matrix_x + offset * OPTION_CELL) * SCALE
            y = (matrix_y + offset * OPTION_CELL) * SCALE
            draw.line((x, matrix_y * SCALE, x, (matrix_y + 3 * OPTION_CELL) * SCALE), fill=(155, 165, 177), width=1 * SCALE)
            draw.line((matrix_x * SCALE, y, (matrix_x + 3 * OPTION_CELL) * SCALE, y), fill=(155, 165, 177), width=1 * SCALE)
        for row_index, exit_name in enumerate(option):
            col_index = EXIT_NAMES.index(exit_name)
            cx = matrix_x + col_index * OPTION_CELL + OPTION_CELL / 2
            cy = matrix_y + row_index * OPTION_CELL + OPTION_CELL / 2
            draw.ellipse(((cx - 9) * SCALE, (cy - 9) * SCALE, (cx + 9) * SCALE, (cy + 9) * SCALE), fill=DOT_RGB)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
