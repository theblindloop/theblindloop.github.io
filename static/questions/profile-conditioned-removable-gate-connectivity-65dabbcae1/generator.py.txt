"""Deterministic latent sampler, raster renderer, and analytic topology arm."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

SIZE = 512
PALETTES = (
    {"background": [249, 248, 244], "wall": [51, 58, 66], "green": [25, 163, 108], "purple": [143, 76, 211]},
    {"background": [247, 249, 246], "wall": [61, 64, 72], "green": [35, 154, 96], "purple": [155, 73, 198]},
    {"background": [250, 247, 243], "wall": [43, 54, 64], "green": [19, 169, 116], "purple": [128, 72, 190]},
)


def _all_edges(rows: int, cols: int) -> tuple[list[str], list[str]]:
    outer: list[str] = []
    internal: list[str] = []
    for r in range(rows + 1):
        for c in range(cols):
            edge = f"h:{r}:{c}"
            (outer if r in {0, rows} else internal).append(edge)
    for r in range(rows):
        for c in range(cols + 1):
            edge = f"v:{r}:{c}"
            (outer if c in {0, cols} else internal).append(edge)
    return outer, internal


def _parse_edge(edge: str) -> tuple[str, int, int]:
    kind, first, second = edge.split(":")
    return kind, int(first), int(second)


def _incident_cells(edge: str, rows: int, cols: int) -> set[tuple[int, int]]:
    kind, r, c = _parse_edge(edge)
    cells: set[tuple[int, int]] = set()
    if kind == "h":
        if r > 0:
            cells.add((r - 1, c))
        if r < rows:
            cells.add((r, c))
    else:
        if c > 0:
            cells.add((r, c - 1))
        if c < cols:
            cells.add((r, c))
    return cells


def _cell_components(rows: int, cols: int, walls: Iterable[str]) -> dict[tuple[int, int], int]:
    wall_set = set(walls)
    parent = {(r, c): (r, c) for r in range(rows) for c in range(cols)}

    def find(item: tuple[int, int]) -> tuple[int, int]:
        while parent[item] != item:
            parent[item] = parent[parent[item]]
            item = parent[item]
        return item

    def union(left: tuple[int, int], right: tuple[int, int]) -> None:
        a, b = find(left), find(right)
        if a != b:
            parent[b] = a

    for r in range(rows):
        for c in range(cols):
            if c + 1 < cols and f"v:{r}:{c + 1}" not in wall_set:
                union((r, c), (r, c + 1))
            if r + 1 < rows and f"h:{r + 1}:{c}" not in wall_set:
                union((r, c), (r + 1, c))
    roots = {find(cell) for cell in parent}
    root_ids = {root: index for index, root in enumerate(sorted(roots))}
    return {cell: root_ids[find(cell)] for cell in parent}


def _gate_components(edge: str, rows: int, cols: int, components: dict[tuple[int, int], int]) -> set[int]:
    return {components[cell] for cell in _incident_cells(edge, rows, cols)}


def _edge_midpoint(edge: str, xs: list[int], ys: list[int]) -> tuple[float, float]:
    kind, r, c = _parse_edge(edge)
    if kind == "h":
        return ((xs[c] + xs[c + 1]) / 2.0, float(ys[r]))
    return (float(xs[c]), (ys[r] + ys[r + 1]) / 2.0)


def _choose_gate_pair(
    rng: random.Random,
    walls: list[str],
    rows: int,
    cols: int,
    xs: list[int],
    ys: list[int],
    want_shared: bool,
) -> tuple[str, str] | None:
    components = _cell_components(rows, cols, walls)
    shuffled = list(walls)
    rng.shuffle(shuffled)
    pairs: list[tuple[str, str]] = []
    for index, left in enumerate(shuffled):
        lx, ly = _edge_midpoint(left, xs, ys)
        left_components = _gate_components(left, rows, cols, components)
        for right in shuffled[index + 1 :]:
            rx, ry = _edge_midpoint(right, xs, ys)
            if (lx - rx) ** 2 + (ly - ry) ** 2 < 92**2:
                continue
            shared = bool(left_components & _gate_components(right, rows, cols, components))
            if shared == want_shared:
                pairs.append((left, right))
    return rng.choice(pairs) if pairs else None


def _jittered_axis(rng: random.Random, count: int) -> list[int]:
    start = rng.randint(54, 68)
    end = rng.randint(444, 458)
    span = end - start
    values = [start]
    for index in range(1, count):
        nominal = start + round(span * index / count)
        values.append(nominal + rng.randint(-9, 9))
    values.append(end)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    rows = cols = 4
    want_shared = int(seed) % 2 == 0
    for _attempt in range(200):
        xs = _jittered_axis(rng, cols)
        ys = _jittered_axis(rng, rows)
        outer, internal = _all_edges(rows, cols)
        walls = outer + [edge for edge in internal if rng.random() < rng.uniform(0.48, 0.68)]
        components = _cell_components(rows, cols, walls)
        component_count = len(set(components.values()))
        if not 3 <= component_count <= 9:
            continue
        pair = _choose_gate_pair(rng, walls, rows, cols, xs, ys, want_shared)
        if pair is None:
            continue
        if rng.random() < 0.5:
            pair = (pair[1], pair[0])
        return {
            "rows": rows,
            "cols": cols,
            "xs": xs,
            "ys": ys,
            "walls": sorted(walls),
            "green_gate": pair[0],
            "purple_gate": pair[1],
            "wall_width": rng.choice([14, 16, 18]),
            "gate_length": rng.choice([38, 42, 46, 50]),
            "palette": dict(PALETTES[rng.randrange(len(PALETTES))]),
        }
    raise RuntimeError(f"could not sample a valid partition for seed {seed}")


def _edge_box(edge: str, xs: list[int], ys: list[int], width: int) -> tuple[int, int, int, int]:
    kind, r, c = _parse_edge(edge)
    half = width // 2
    if kind == "h":
        return (xs[c] - half, ys[r] - half, xs[c + 1] + half, ys[r] + half)
    return (xs[c] - half, ys[r] - half, xs[c] + half, ys[r + 1] + half)


def _gate_box(edge: str, xs: list[int], ys: list[int], width: int, length: int) -> tuple[int, int, int, int]:
    kind, r, c = _parse_edge(edge)
    half_width = width // 2
    half_length = length // 2
    if kind == "h":
        center = (xs[c] + xs[c + 1]) // 2
        return (center - half_length, ys[r] - half_width, center + half_length, ys[r] + half_width)
    center = (ys[r] + ys[r + 1]) // 2
    return (xs[c] - half_width, center - half_length, xs[c] + half_width, center + half_length)


def render(scene: dict) -> Image.Image:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    walls = [str(value) for value in scene["walls"]]
    green_gate = str(scene["green_gate"])
    purple_gate = str(scene["purple_gate"])
    wall_width = int(scene["wall_width"])
    gate_length = int(scene["gate_length"])
    palette = scene["palette"]
    if len(xs) != cols + 1 or len(ys) != rows + 1:
        raise ValueError("axis lengths disagree with grid dimensions")
    image = Image.new("RGB", (SIZE, SIZE), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    wall_color = tuple(palette["wall"])
    for edge in walls:
        draw.rectangle(_edge_box(edge, xs, ys, wall_width), fill=wall_color)
    draw.rectangle(_gate_box(green_gate, xs, ys, wall_width, gate_length), fill=tuple(palette["green"]))
    draw.rectangle(_gate_box(purple_gate, xs, ys, wall_width, gate_length), fill=tuple(palette["purple"]))
    return image


def analytic_gold(scene: dict) -> str:
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    walls = [str(value) for value in scene["walls"]]
    green_gate = str(scene["green_gate"])
    purple_gate = str(scene["purple_gate"])
    components = _cell_components(rows, cols, walls)
    green = _gate_components(green_gate, rows, cols, components)
    purple = _gate_components(purple_gate, rows, cols, components)
    return "yes" if green & purple else "no"


def margin(scene: dict) -> float:
    width = float(scene["wall_width"])
    gate_length = float(scene["gate_length"])
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    shortest_edge = min(min(b - a for a, b in zip(xs, xs[1:])), min(b - a for a, b in zip(ys, ys[1:])))
    endpoint_clearance = (shortest_edge - gate_length) / 2.0 - width / 2.0
    return float(min(width / 2.0, endpoint_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
