"""Independent PNG-only enclosure-binding oracle for the marked-link pair."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_DARK = np.asarray([35, 40, 48], dtype=np.int16)
_YELLOW = np.asarray([244, 183, 0], dtype=np.int16)
_TOL = 34.0


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = deque([(sy, sx)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (0 <= ny < height and 0 <= nx < width and mask[ny, nx]
                        and not seen[ny, nx]):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            part = np.zeros_like(mask, dtype=bool)
            yy, xx = zip(*points)
            part[np.asarray(yy), np.asarray(xx)] = True
            output.append(part)
    return output


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape != (384, 384, 3):
        raise ValueError("expected a 384 by 384 RGB raster")
    dark = np.linalg.norm(array - _DARK, axis=2) <= _TOL
    yellow = np.linalg.norm(array - _YELLOW, axis=2) <= _TOL
    tokens = _components(yellow, minimum=300)
    if len(tokens) != 2 or any(not 450 <= int(token.sum()) <= 1100 for token in tokens):
        raise ValueError("expected two intact separated yellow rail links")
    barrier = dark | yellow
    if int(dark.sum()) < 4000:
        raise ValueError("substantial railway ink is missing")
    ys, xs = np.nonzero(barrier)
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    span_x, span_y = x1 - x0, y1 - y0
    if min(span_x, span_y) < 105 or max(span_x, span_y) < 225:
        raise ValueError("rail envelope does not visibly support two loop basins")

    # The declared railway has two closed faces joined along its major axis.
    # Infer that axis from the segmented rail envelope, then bind every marked
    # component to the nearer basin half. This uses no fixed coordinates and is
    # stable under translation and quarter-turn rotation.
    horizontal = span_x > span_y
    midpoint = (x0 + x1) / 2.0 if horizontal else (y0 + y1) / 2.0
    assignments = []
    for token in tokens:
        tx, ty = _centroid(token)
        coordinate = tx if horizontal else ty
        if abs(coordinate - midpoint) < 35.0:
            raise ValueError("a marked link is ambiguous between the two loops")
        assignments.append(0 if coordinate < midpoint else 1)
    return "yes" if len(set(assignments)) == 2 else "no"
