"""Deterministic renderer for a joint cycle-breaking pair of rail links."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw


WIDTH = 384
HEIGHT = 384
BACKGROUND = [249, 248, 243]
TRACK = [35, 40, 48]
MARKED = [244, 183, 0]
TRACK_WIDTH = 13


def _rotate(point: tuple[int, int], turns: int) -> list[int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = WIDTH - 1 - y, x
    return [x, y]


def _rotate_box(box: list[int], turns: int) -> list[int]:
    x0, y0, x1, y1 = box
    points = [_rotate((x0, y0), turns), _rotate((x1, y0), turns),
              _rotate((x1, y1), turns), _rotate((x0, y1), turns)]
    return [min(p[0] for p in points), min(p[1] for p in points),
            max(p[0] for p in points), max(p[1] for p in points)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    loop_w = rng.randint(112, 120)
    loop_h = rng.randint(132, 142)
    gap = rng.randint(26, 34)
    total_w = 2 * loop_w + gap
    x0 = (WIDTH - total_w) // 2 + rng.randint(-5, 5)
    y0 = (HEIGHT - loop_h) // 2 + rng.randint(-7, 7)
    boxes = [
        [x0, y0, x0 + loop_w, y0 + loop_h],
        [x0 + loop_w + gap, y0, x0 + 2 * loop_w + gap, y0 + loop_h],
    ]
    seg_len = rng.randint(58, 66)

    def slot(loop_index: int, edge: str) -> list[list[int]]:
        bx0, by0, bx1, by1 = boxes[loop_index]
        cx = (bx0 + bx1) // 2
        y = by0 if edge == "top" else by1
        return [[cx - seg_len // 2, y], [cx + seg_len // 2, y]]

    # Even seeds: one marked link per loop (a minimal feedback-edge pair).
    # Odd seeds: two separated links on one loop (the other loop survives).
    if int(seed) % 2 == 0:
        marked = [slot(0, rng.choice(("top", "bottom"))),
                  slot(1, rng.choice(("top", "bottom")))]
    else:
        chosen = rng.randrange(2)
        marked = [slot(chosen, "top"), slot(chosen, "bottom")]

    bridge = [[boxes[0][2], (boxes[0][1] + boxes[0][3]) // 2],
              [boxes[1][0], (boxes[1][1] + boxes[1][3]) // 2]]
    turns = rng.randrange(4)
    boxes = [_rotate_box(box, turns) for box in boxes]
    bridge = [_rotate(tuple(point), turns) for point in bridge]
    marked = [[_rotate(tuple(point), turns) for point in segment] for segment in marked]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "track_color": list(TRACK),
        "marked_color": list(MARKED),
        "loop_boxes": boxes,
        "bridge": bridge,
        "marked_segments": marked,
        "track_width": TRACK_WIDTH,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])),
                      tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    width = int(scene["track_width"])
    track = tuple(scene["track_color"])
    for x0, y0, x1, y1 in scene["loop_boxes"]:
        draw.line([(x0, y0), (x1, y0), (x1, y1), (x0, y1), (x0, y0)],
                  fill=track, width=width, joint="curve")
    draw.line([tuple(p) for p in scene["bridge"]], fill=track, width=width)
    for segment in scene["marked_segments"]:
        draw.line([tuple(p) for p in segment], fill=tuple(scene["marked_color"]), width=width)
    return image


def _segment_loop(scene: dict, segment: list[list[int]]) -> int:
    cx = sum(point[0] for point in segment) / 2.0
    cy = sum(point[1] for point in segment) / 2.0
    centers = [((box[0] + box[2]) / 2.0, (box[1] + box[3]) / 2.0)
               for box in scene["loop_boxes"]]
    distances = [(cx - x) ** 2 + (cy - y) ** 2 for x, y in centers]
    if distances[0] == distances[1]:
        raise ValueError("marked link is not visibly assigned to one loop")
    return 0 if distances[0] < distances[1] else 1


def analytic_gold(scene: dict) -> str:
    loops_hit = {_segment_loop(scene, segment) for segment in scene["marked_segments"]}
    return "yes" if len(loops_hit) == 2 else "no"


def margin(scene: dict) -> float:
    # The runner-up differs by one surviving independent cycle.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    if len(scene["marked_segments"]) != 2 or int(scene["track_width"]) < 9:
        return True
    lengths = [abs(seg[1][0] - seg[0][0]) + abs(seg[1][1] - seg[0][1])
               for seg in scene["marked_segments"]]
    return min(lengths) < 45


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["marked_segments"] = list(reversed(twin["marked_segments"]))
    return [("reverse_marked_link_storage", twin)]
