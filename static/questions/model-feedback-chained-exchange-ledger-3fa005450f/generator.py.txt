"""Deterministic chained-exchange sampler and raster renderer."""

from __future__ import annotations

from collections import defaultdict
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

SIZE = (768, 640)
BACKGROUND = (247, 245, 239)
INK = (39, 43, 48)
PANEL = (255, 254, 250)
MUTED = (219, 216, 207)
TOKEN_COLORS = ((23, 137, 180), (151, 78, 163), (228, 132, 32), (55, 153, 93))
WEIGHTS = (30, 20, 8, 6)
DECISIONS = ("crate-1", "crate-2", "crate-3", "crate-4")
EQUATIONS = (((0, 2), (1, 3)), ((1, 2), (2, 5)), ((2, 3), (3, 4)))
CARD_X = (24, 272, 520)
CRATE_X = (24, 206, 388, 570)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _value(counts: list[int] | tuple[int, ...]) -> int:
    return sum(int(a) * b for a, b in zip(counts, WEIGHTS))


def _eligible_groups() -> list[tuple[int, list[tuple[int, int, int, int]]]]:
    groups: dict[int, list[tuple[int, int, int, int]]] = defaultdict(list)
    for counts in itertools.product(range(6), repeat=4):
        if sum(counts) == 6 and sum(value > 0 for value in counts) >= 2:
            groups[_value(counts)].append(counts)
    return sorted(
        (value, counts)
        for value, counts in groups.items()
        if len(counts) >= 2
    )


GROUPS = _eligible_groups()
ALL_GROUPS = [
    counts
    for counts in itertools.product(range(6), repeat=4)
    if sum(counts) == 6 and sum(value > 0 for value in counts) >= 2
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    _target_value, representations = GROUPS[rng.randrange(len(GROUPS))]
    target, correct = rng.sample(representations, 2)
    target_value = _value(target)
    distractor_pool = [
        counts for counts in ALL_GROUPS
        if _value(counts) != target_value and 2 <= abs(_value(counts) - target_value) <= 16
    ]
    rng.shuffle(distractor_pool)
    distractors: list[tuple[int, int, int, int]] = []
    used_values: set[int] = set()
    for counts in distractor_pool:
        value = _value(counts)
        if value not in used_values:
            distractors.append(counts)
            used_values.add(value)
        if len(distractors) == 3:
            break
    if len(distractors) != 3:
        raise RuntimeError("insufficient distractors")
    rng.shuffle(distractors)
    correct_index = int(seed) % 4
    crates = list(distractors)
    crates.insert(correct_index, correct)
    equation_order = list(range(3))
    rng.shuffle(equation_order)
    return {
        "target_counts": list(target),
        "crate_counts": [list(counts) for counts in crates],
        "equation_order": equation_order,
        "layout_variant": rng.randrange(8),
    }


def _validated(scene: dict) -> tuple[tuple[int, ...], tuple[tuple[int, ...], ...]]:
    target = tuple(int(value) for value in scene["target_counts"])
    crates = tuple(tuple(int(value) for value in counts) for counts in scene["crate_counts"])
    order = [int(value) for value in scene["equation_order"]]
    variant = int(scene["layout_variant"])
    if len(target) != 4 or len(crates) != 4 or any(len(counts) != 4 for counts in crates):
        raise ValueError("target and four crates require four token counts")
    if any(value < 0 or value > 5 for value in target) or any(value < 0 or value > 5 for counts in crates for value in counts):
        raise ValueError("token counts out of range")
    if sum(target) != 6 or any(sum(counts) != 6 for counts in crates):
        raise ValueError("all compared groups must contain six tokens")
    if sorted(order) != [0, 1, 2] or not 0 <= variant < 8:
        raise ValueError("invalid equation layout")
    return target, crates


def analytic_gold(scene: dict) -> str:
    try:
        target, crates = _validated(scene)
        matches = [index for index, counts in enumerate(crates) if _value(counts) == _value(target)]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    target, crates = _validated(scene)
    differences = [abs(_value(counts) - _value(target)) for counts in crates]
    nonzero = [value for value in differences if value]
    return float(min(nonzero)) if differences.count(0) == 1 and nonzero else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return analytic_gold(scene) == "abstain" or margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _tokens(draw: ImageDraw.ImageDraw, counts: tuple[int, ...] | list[int], box: tuple[int, int, int, int], reverse: bool) -> None:
    x0, y0, x1, y1 = box
    types = [kind for kind, count in enumerate(counts) for _ in range(int(count))]
    if reverse:
        types.reverse()
    columns = 3
    gap_x, gap_y = 27, 27
    start_x = (x0 + x1) // 2 - gap_x
    start_y = (y0 + y1) // 2 - (gap_y // 2 if len(types) > 3 else 0)
    for index, kind in enumerate(types):
        cx = start_x + (index % columns) * gap_x
        cy = start_y + (index // columns) * gap_y
        draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=TOKEN_COLORS[kind], outline=INK, width=1)


def render(scene: dict) -> Image.Image:
    target, crates = _validated(scene)
    order = [int(value) for value in scene["equation_order"]]
    variant = int(scene["layout_variant"])
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font, label_font, small_font = _font(24), _font(18), _font(15)
    draw.text((24, 18), "EXACT EXCHANGES", font=title_font, fill=INK)

    for slot, equation_index in enumerate(order):
        x0 = CARD_X[slot]
        draw.rounded_rectangle((x0, 58, x0 + 224, 218), radius=12, fill=PANEL, outline=INK, width=3)
        (left_type, left_count), (right_type, right_count) = EQUATIONS[equation_index]
        if (variant >> equation_index) & 1:
            left_type, right_type = right_type, left_type
            left_count, right_count = right_count, left_count
        left = [0, 0, 0, 0]
        right = [0, 0, 0, 0]
        left[left_type], right[right_type] = left_count, right_count
        _tokens(draw, left, (x0 + 8, 84, x0 + 94, 202), bool(slot & 1))
        _tokens(draw, right, (x0 + 130, 84, x0 + 216, 202), not bool(slot & 1))
        draw.text((x0 + 102, 128), "=", font=title_font, fill=INK)

    draw.text((24, 244), "TARGET", font=label_font, fill=INK)
    draw.rounded_rectangle((24, 272, 744, 398), radius=12, fill=PANEL, outline=(185, 91, 52), width=4)
    _tokens(draw, target, (286, 280, 458, 390), bool(variant & 1))

    draw.text((24, 414), "CRATES", font=label_font, fill=INK)
    for index, counts in enumerate(crates):
        x0 = CRATE_X[index]
        draw.rounded_rectangle((x0, 444, x0 + 158, 612), radius=12, fill=PANEL, outline=INK, width=3)
        label = f"crate {index + 1}"
        label_box = draw.textbbox((0, 0), label, font=small_font)
        draw.text((x0 + 79 - (label_box[2] - label_box[0]) // 2, 454), label, font=small_font, fill=INK)
        draw.line((x0 + 12, 480, x0 + 146, 480), fill=MUTED, width=2)
        _tokens(draw, counts, (x0 + 22, 488, x0 + 136, 598), bool((variant + index) & 1))
    return image
