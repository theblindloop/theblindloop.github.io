"""Independent image-only inverse for chained exchange ledgers."""

from __future__ import annotations

from fractions import Fraction

import numpy as np
from PIL import Image

TOKEN_COLORS = ((23, 137, 180), (151, 78, 163), (228, 132, 32), (55, 153, 93))
CARD_X = (24, 272, 520)
CRATE_X = (24, 206, 388, 570)


def _component_count(mask: np.ndarray) -> int | None:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    sizes: list[int] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        size = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            size += 1
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        if size >= 100:
            sizes.append(size)
    if any(not 200 <= size <= 260 for size in sizes):
        return None
    return len(sizes)


def _counts(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, ...] | None:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1, x0:x1]
    result: list[int] = []
    for color in TOKEN_COLORS:
        delta = np.max(np.abs(crop.astype(np.int16) - np.asarray(color, dtype=np.int16)), axis=2)
        count = _component_count(delta <= 2)
        if count is None:
            return None
        result.append(count)
    return tuple(result)


def _recover_weights(equations: list[tuple[int, int, int, int]]) -> list[Fraction] | None:
    weights: list[Fraction | None] = [Fraction(1), None, None, None]
    changed = True
    while changed:
        changed = False
        for left_type, left_count, right_type, right_count in equations:
            if weights[left_type] is not None and weights[right_type] is None:
                weights[right_type] = weights[left_type] * left_count / right_count
                changed = True
            elif weights[right_type] is not None and weights[left_type] is None:
                weights[left_type] = weights[right_type] * right_count / left_count
                changed = True
            elif weights[left_type] is not None and weights[right_type] is not None:
                if left_count * weights[left_type] != right_count * weights[right_type]:
                    return None
    if any(value is None for value in weights):
        return None
    return [value for value in weights if value is not None]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (640, 768, 3):
        return "abstain"
    equations: list[tuple[int, int, int, int]] = []
    for x0 in CARD_X:
        left = _counts(arr, (x0 + 6, 80, x0 + 98, 206))
        right = _counts(arr, (x0 + 126, 80, x0 + 218, 206))
        if left is None or right is None:
            return "abstain"
        left_types = [index for index, count in enumerate(left) if count]
        right_types = [index for index, count in enumerate(right) if count]
        if len(left_types) != 1 or len(right_types) != 1:
            return "abstain"
        lt, rt = left_types[0], right_types[0]
        equations.append((lt, left[lt], rt, right[rt]))
    weights = _recover_weights(equations)
    if weights is None:
        return "abstain"
    target = _counts(arr, (280, 276, 464, 394))
    crates = [_counts(arr, (x0 + 16, 484, x0 + 142, 604)) for x0 in CRATE_X]
    if target is None or any(counts is None for counts in crates):
        return "abstain"
    if sum(target) != 6 or any(sum(counts) != 6 for counts in crates if counts is not None):
        return "abstain"
    target_value = sum(count * weight for count, weight in zip(target, weights))
    matches = [
        index for index, counts in enumerate(crates)
        if counts is not None and sum(count * weight for count, weight in zip(counts, weights)) == target_value
    ]
    return f"crate-{matches[0] + 1}" if len(matches) == 1 else "abstain"
