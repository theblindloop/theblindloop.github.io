"""Deterministic renderer for a reflected cyclic bead-order question.

Each panel is an isolated ring with six equal beads at fixed, top-anchored
positions.  The decision compares the complete clockwise color order of the
reference with each candidate after a horizontal reflection.  The cards are
deliberately separated so the pixel oracle needs all three distant witnesses.
"""

from __future__ import annotations

import copy
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "cyclic-bead-order-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 560
PANEL_CENTERS = ((160, 215), (800, 215), (480, 440))
PANEL_BOXES = ((45, 105, 275, 330), (685, 105, 915, 330), (365, 335, 595, 555))
RING_RADIUS = 62
BEAD_RADIUS = 17
BEAD_COUNT = 6

PAPER = (249, 247, 241)
CARD = (236, 232, 222)
INK = (38, 49, 62)
RING = (154, 162, 168)
PALETTE = (
    (214, 76, 84),
    (53, 117, 185),
    (57, 158, 112),
    (223, 164, 58),
    (131, 89, 174),
    (35, 157, 170),
)
QUARANTINE_MARGIN = 0.010


def _font(size: int):
    """Use a fixed installed font with a deterministic Pillow fallback."""

    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size)
    return ImageFont.load_default()


def _reflect_order(order: list[int]) -> list[int]:
    """Map a pre-reflection order to positions after a left-right reflection."""

    return [int(order[0]), *[int(value) for value in reversed(order[1:])]]


def _slot_orders(scene: dict) -> dict[int, list[int]]:
    """Resolve candidate records by visible slot, not latent list order."""

    return {
        int(record["slot"]): [int(value) for value in record["order"]]
        for record in scene["candidate_panels"]
    }


def _order_error(reference: list[int], observed: list[int]) -> float:
    """Mean squared circular slot displacement for a full color permutation."""

    if sorted(reference) != list(range(BEAD_COUNT)) or sorted(observed) != list(range(BEAD_COUNT)):
        raise ValueError("order is not a permutation of the six bead labels")
    reference_positions = {label: index for index, label in enumerate(reference)}
    observed_positions = {label: index for index, label in enumerate(observed)}
    total = 0.0
    for label in range(BEAD_COUNT):
        distance = abs(reference_positions[label] - observed_positions[label])
        distance = min(distance, BEAD_COUNT - distance)
        total += (distance / BEAD_COUNT) ** 2
    return total / BEAD_COUNT


def sample_scene(seed: int) -> dict:
    """Sample balanced far and adjacent-swap correspondence scenes."""

    rng = random.Random(int(seed))
    reference = list(range(BEAD_COUNT))
    rng.shuffle(reference)

    # Reflection is an involution, so applying it constructs the matching
    # pre-reflection candidate and also keeps the candidate color multiset
    # exactly equal to the reference.
    matched_pre = _reflect_order(reference)
    matched_after = _reflect_order(matched_pre)
    near_boundary = (int(seed) % 5) == 0
    wrong_after = list(matched_after)
    if near_boundary:
        # A one-step adjacent swap produces a small but visible runner-up gap.
        wrong_after[1], wrong_after[2] = wrong_after[2], wrong_after[1]
    else:
        # A two-step cyclic shift produces a comfortably separated competitor.
        wrong_after = wrong_after[2:] + wrong_after[:2]
    wrong_pre = _reflect_order(wrong_after)

    match_slot = 1 if rng.randrange(2) == 0 else 2
    records = [
        {"slot": match_slot, "order": matched_pre},
        {"slot": 3 - match_slot, "order": wrong_pre},
    ]
    rng.shuffle(records)
    return {
        "reference_order": [int(value) for value in reference],
        "candidate_panels": records,
    }


def analytic_gold(scene: dict) -> str:
    """Select the candidate with the smallest transformed circular-order error."""

    reference = [int(value) for value in scene["reference_order"]]
    candidates = _slot_orders(scene)
    scores = {
        slot: _order_error(reference, _reflect_order(order))
        for slot, order in candidates.items()
    }
    if scores[1] == scores[2]:
        raise ValueError("candidate cyclic-order scores are tied")
    return "A" if scores[1] < scores[2] else "B"


def margin(scene: dict) -> float:
    """Return the absolute gap between the two transformed order errors."""

    reference = [int(value) for value in scene["reference_order"]]
    candidates = _slot_orders(scene)
    scores = [
        _order_error(reference, _reflect_order(candidates[1])),
        _order_error(reference, _reflect_order(candidates[2])),
    ]
    return abs(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    """Quarantine near-ties that do not have a comfortable raster margin."""

    return margin(scene) < QUARANTINE_MARGIN


def _scaled_box(box: tuple[float, float, float, float], scale: int):
    return tuple(int(round(value * scale)) for value in box)


def _draw_text(draw: ImageDraw.ImageDraw, xy, text: str, font, fill, scale: int) -> None:
    draw.text((int(xy[0] * scale), int(xy[1] * scale)), text, font=font, fill=fill)


def _bead_positions(center_x: int, center_y: int) -> list[tuple[int, int]]:
    positions = []
    for index in range(BEAD_COUNT):
        angle = -math.pi / 2.0 + 2.0 * math.pi * index / BEAD_COUNT
        positions.append(
            (
                int(round(center_x + RING_RADIUS * math.cos(angle))),
                int(round(center_y + RING_RADIUS * math.sin(angle))),
            )
        )
    return positions


def render(scene: dict) -> Image.Image:
    """Render a deterministic antialiased RGB image."""

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25 * scale)
    label_font = _font(18 * scale)
    small_font = _font(14 * scale)

    _draw_text(draw, (40, 25), "CYCLIC BEAD CORRESPONDENCE", title_font, INK, scale)
    _draw_text(
        draw,
        (40, 66),
        "Start at the top bead, move clockwise; reflect candidates left to right.",
        small_font,
        INK,
        scale,
    )

    labels = ("REFERENCE  R", "CANDIDATE  A", "CANDIDATE  B")
    orders_by_slot = _slot_orders(scene)
    panel_orders = (
        [int(value) for value in scene["reference_order"]],
        orders_by_slot[1],
        orders_by_slot[2],
    )
    for (center_x, center_y), box, label, order in zip(PANEL_CENTERS, PANEL_BOXES, labels, panel_orders):
        draw.rounded_rectangle(
            _scaled_box(box, scale),
            radius=16 * scale,
            fill=CARD,
            outline=INK,
            width=4 * scale,
        )
        bbox = draw.textbbox((0, 0), label, font=label_font)
        text_width = bbox[2] - bbox[0]
        _draw_text(
            draw,
            (center_x - text_width / (2 * scale), box[1] + 20),
            label,
            label_font,
            INK,
            scale,
        )

        ring_box = _scaled_box(
            (
                center_x - RING_RADIUS,
                center_y - RING_RADIUS,
                center_x + RING_RADIUS,
                center_y + RING_RADIUS,
            ),
            scale,
        )
        draw.ellipse(ring_box, outline=RING, width=3 * scale)
        # A top tick makes the public starting convention visible without
        # carrying any answer-specific information.
        draw.line(
            _scaled_box((center_x, center_y - RING_RADIUS - 11, center_x, center_y - RING_RADIUS - 1), scale),
            fill=INK,
            width=3 * scale,
        )
        for position, label_id in zip(_bead_positions(center_x, center_y), order):
            x, y = position
            bead_box = _scaled_box(
                (x - BEAD_RADIUS, y - BEAD_RADIUS, x + BEAD_RADIUS, y + BEAD_RADIUS),
                scale,
            )
            draw.ellipse(bead_box, fill=PALETTE[int(label_id)], outline=PAPER, width=2 * scale)

        _draw_text(draw, (center_x - 76, box[3] - 34), "six equal beads", small_font, INK, scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the harmless permutation of slot-addressed candidate records."""

    twin = copy.deepcopy(scene)
    twin["candidate_panels"] = list(reversed(twin["candidate_panels"]))
    return [("candidate_record_order", twin)]
