"""Independent RGB-only inverse for the cyclic bead-order question.

The inverse has its own geometry and recovers each colored bead from connected
components in the supplied raster.  It never imports the renderer or reads a
latent scene.  Candidate orders are then reflected and compared as complete
circular permutations.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 560
PANEL_CENTERS = ((160, 215), (800, 215), (480, 440))
RING_RADIUS = 62.0
BEAD_COUNT = 6
PALETTE = np.asarray(
    (
        (214, 76, 84),
        (53, 117, 185),
        (57, 158, 112),
        (223, 164, 58),
        (131, 89, 174),
        (35, 157, 170),
    ),
    dtype=np.float32,
)


def _component_centroid(mask: np.ndarray) -> tuple[float, float]:
    """Return the centroid of the largest 4-connected mask component."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        iy, ix = int(y), int(x)
        if seen[iy, ix]:
            continue
        stack = [(iy, ix)]
        seen[iy, ix] = True
        component: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(component) > len(best):
            best = component
    if len(best) < 120:
        raise ValueError("colored bead component is missing or too small")
    values = np.asarray(best, dtype=np.float64)
    return float(values[:, 1].mean()), float(values[:, 0].mean())


def _expected_positions(center_x: int, center_y: int) -> list[tuple[float, float]]:
    return [
        (
            center_x + RING_RADIUS * math.cos(-math.pi / 2.0 + 2.0 * math.pi * index / BEAD_COUNT),
            center_y + RING_RADIUS * math.sin(-math.pi / 2.0 + 2.0 * math.pi * index / BEAD_COUNT),
        )
        for index in range(BEAD_COUNT)
    ]


def _recover_order(array: np.ndarray, center_x: int, center_y: int) -> list[int]:
    """Recover the six color labels at the six fixed ring positions."""

    height, width, _ = array.shape
    order: list[int | None] = [None] * BEAD_COUNT
    used_slots: set[int] = set()
    expected = _expected_positions(center_x, center_y)
    x_left, x_right = center_x - 118, center_x + 118
    y_top, y_bottom = max(0, center_y - 118), min(height, center_y + 118)
    crop = array[y_top:y_bottom, x_left:x_right]
    for label, color in enumerate(PALETTE):
        distance = np.sqrt(np.sum((crop - color) ** 2, axis=2))
        local_x, local_y = _component_centroid(distance < 62.0)
        x = local_x + x_left
        y = local_y + y_top
        distances = [math.hypot(x - px, y - py) for px, py in expected]
        slot = int(np.argmin(distances))
        if distances[slot] > 22.0 or slot in used_slots:
            raise ValueError("bead centroid is not at a unique anchored ring position")
        used_slots.add(slot)
        order[slot] = label
    if any(value is None for value in order) or len(used_slots) != BEAD_COUNT:
        raise ValueError("ring does not contain six distinct colored beads")
    return [int(value) for value in order]


def _reflected(order: list[int]) -> list[int]:
    return [int(order[0]), *[int(value) for value in reversed(order[1:])]]


def _order_error(reference: list[int], observed: list[int]) -> float:
    reference_positions = {label: index for index, label in enumerate(reference)}
    observed_positions = {label: index for index, label in enumerate(observed)}
    total = 0.0
    for label in range(BEAD_COUNT):
        distance = abs(reference_positions[label] - observed_positions[label])
        distance = min(distance, BEAD_COUNT - distance)
        total += (distance / BEAD_COUNT) ** 2
    return total / BEAD_COUNT


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from the final RGB raster alone."""

    if image.width != WIDTH or image.height != HEIGHT:
        raise ValueError("unexpected raster dimensions")
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    reference = _recover_order(array, *PANEL_CENTERS[0])
    candidate_a = _recover_order(array, *PANEL_CENTERS[1])
    candidate_b = _recover_order(array, *PANEL_CENTERS[2])
    error_a = _order_error(reference, _reflected(candidate_a))
    error_b = _order_error(reference, _reflected(candidate_b))
    if not math.isfinite(error_a + error_b) or error_a == error_b:
        raise ValueError("cyclic-order comparison is tied or non-finite")
    return "A" if error_a < error_b else "B"
