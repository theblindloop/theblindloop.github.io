"""Deterministic renderer for four counterfactual Boolean gate cascades."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 760
SCALE = 2
BACKGROUND = (246, 245, 240)
PANEL_FILL = (255, 255, 255)
INK = (35, 40, 47)
OFF = (255, 255, 255)
RED = (205, 54, 54)
GATE_COLORS = {
    "and": (48, 112, 190),
    "or": (224, 148, 39),
    "xor": (132, 76, 183),
}
GATE_LABELS = {"and": "ALL", "or": "ANY", "xor": "ONE"}
PANEL_ORIGINS = ((24, 116), (492, 116), (24, 432), (492, 432))
INPUT_OFFSETS = ((58, 74), (58, 124), (58, 174), (58, 224))
GATE_OFFSETS = ((158, 99), (269, 149), (380, 199))
OUTPUT_OFFSET = (430, 199)


def _gate(kind: str, left: bool, right: bool) -> bool:
    if kind == "and":
        return left and right
    if kind == "or":
        return left or right
    if kind == "xor":
        return left != right
    raise ValueError(f"unknown gate {kind!r}")


def _evaluate(inputs: list[int], gates: list[str]) -> tuple[bool, bool, bool]:
    a = _gate(gates[0], bool(inputs[0]), bool(inputs[1]))
    b = _gate(gates[1], a, bool(inputs[2]))
    c = _gate(gates[2], b, bool(inputs[3]))
    return a, b, c


def _effect_depth(panel: dict) -> int:
    factual = _evaluate(panel["inputs"], panel["gates"])
    changed_inputs = list(panel["inputs"])
    marked = int(panel["marked"])
    changed_inputs[marked] = 1 - changed_inputs[marked]
    counter = _evaluate(changed_inputs, panel["gates"])
    for index, (before, after) in enumerate(zip(factual, counter, strict=True)):
        if before == after:
            return index
    return 3


def _sample_panel(rng: random.Random, desired_depth: int) -> dict:
    for _ in range(20000):
        inputs = [rng.randrange(2) for _ in range(4)]
        gates = [rng.choice(("and", "or", "xor")) for _ in range(3)]
        if len(set(gates)) == 1:
            continue
        panel = {
            "inputs": inputs,
            "gates": gates,
            "marked": rng.randrange(2),
            "output": int(_evaluate(inputs, gates)[-1]),
        }
        if _effect_depth(panel) == desired_depth:
            return panel
    raise RuntimeError(f"could not sample effect depth {desired_depth}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    distractor_depths = [0, 1, 2]
    rng.shuffle(distractor_depths)
    panels: list[dict] = []
    cursor = 0
    for slot in range(4):
        depth = 3 if slot == target_slot else distractor_depths[cursor]
        if slot != target_slot:
            cursor += 1
        panels.append(_sample_panel(rng, depth))
    return {"panels": panels, "panel_order": [0, 1, 2, 3]}


def analytic_gold(scene: dict) -> str:
    # Keep the closed-form gold visibly rooted in the single declared top-level
    # label input. Nested panel fields are evaluated here rather than delegated,
    # which also makes the rendered-field audit exact at the scene boundary.
    changed: list[int] = []
    for index, panel in enumerate(scene["panels"]):
        values = [bool(value) for value in panel["inputs"]]
        kinds = panel["gates"]

        def apply(kind: str, left: bool, right: bool) -> bool:
            if kind == "and":
                return left and right
            if kind == "or":
                return left or right
            if kind == "xor":
                return left != right
            raise ValueError(kind)

        factual_a = apply(kinds[0], values[0], values[1])
        factual_b = apply(kinds[1], factual_a, values[2])
        factual = apply(kinds[2], factual_b, values[3])
        marked = int(panel["marked"])
        values[marked] = not values[marked]
        counter_a = apply(kinds[0], values[0], values[1])
        counter_b = apply(kinds[1], counter_a, values[2])
        counter = apply(kinds[2], counter_b, values[3])
        if factual != counter:
            changed.append(index)
    if len(changed) != 1:
        raise ValueError(f"expected one changed output, got {changed}")
    return f"panel-{changed[0] + 1}"


def margin(scene: dict) -> float:
    # The rule is discrete; one whole Boolean value separates change from no change.
    _ = analytic_gold(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["panel_order"] = list(reversed(twin["panel_order"]))
    return [("reverse_disjoint_draw_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _pt(value: tuple[int, int]) -> tuple[int, int]:
    return value[0] * SCALE, value[1] * SCALE


def _box(value: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(v * SCALE for v in value)  # type: ignore[return-value]


def _draw_panel(draw: ImageDraw.ImageDraw, slot: int, panel: dict) -> None:
    ox, oy = PANEL_ORIGINS[slot]
    draw.rounded_rectangle(_box((ox, oy, ox + 444, oy + 296)), radius=14 * SCALE,
                           fill=PANEL_FILL, outline=INK, width=3 * SCALE)
    draw.text(_pt((ox + 16, oy + 12)), f"PANEL {slot + 1}", fill=INK, font=_font(18, True))

    inputs = [(ox + dx, oy + dy) for dx, dy in INPUT_OFFSETS]
    gates = [(ox + dx, oy + dy) for dx, dy in GATE_OFFSETS]
    output = (ox + OUTPUT_OFFSET[0], oy + OUTPUT_OFFSET[1])
    line = (91, 99, 110)

    # Fixed causal wiring, drawn first so nodes remain crisp and unobscured.
    g1, g2, g3 = gates
    paths = [
        [inputs[0], (g1[0] - 34, g1[1] - 9)],
        [inputs[1], (g1[0] - 34, g1[1] + 9)],
        [(g1[0] + 34, g1[1]), (g1[0] + 55, g1[1]), (g1[0] + 55, g2[1] - 9), (g2[0] - 34, g2[1] - 9)],
        [inputs[2], (g2[0] - 53, inputs[2][1]), (g2[0] - 53, g2[1] + 9), (g2[0] - 34, g2[1] + 9)],
        [(g2[0] + 34, g2[1]), (g2[0] + 55, g2[1]), (g2[0] + 55, g3[1] - 9), (g3[0] - 34, g3[1] - 9)],
        [inputs[3], (g3[0] - 53, inputs[3][1]), (g3[0] - 53, g3[1] + 9), (g3[0] - 34, g3[1] + 9)],
        [(g3[0] + 34, g3[1]), (output[0] - 14, output[1])],
    ]
    for path in paths:
        draw.line([_pt(p) for p in path], fill=line, width=4 * SCALE, joint="curve")

    for index, (cx, cy) in enumerate(inputs):
        if index == int(panel["marked"]):
            draw.ellipse(_box((cx - 21, cy - 21, cx + 21, cy + 21)), outline=RED, width=5 * SCALE)
        fill = INK if panel["inputs"][index] else OFF
        draw.ellipse(_box((cx - 12, cy - 12, cx + 12, cy + 12)), fill=fill, outline=INK, width=3 * SCALE)
        draw.text(_pt((cx - 42, cy - 9)), f"I{index + 1}", fill=INK, font=_font(13, True))

    for index, (cx, cy) in enumerate(gates):
        kind = panel["gates"][index]
        draw.rounded_rectangle(_box((cx - 34, cy - 22, cx + 34, cy + 22)), radius=8 * SCALE,
                               fill=GATE_COLORS[kind], outline=INK, width=2 * SCALE)
        label = GATE_LABELS[kind]
        bbox = draw.textbbox((0, 0), label, font=_font(13, True))
        tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
        draw.text((cx * SCALE - tw // 2, cy * SCALE - th // 2 - 1 * SCALE), label,
                  fill=(255, 255, 255), font=_font(13, True))

    fill = INK if panel["output"] else OFF
    draw.ellipse(_box((output[0] - 13, output[1] - 13, output[0] + 13, output[1] + 13)),
                 fill=fill, outline=INK, width=3 * SCALE)
    draw.text(_pt((output[0] - 24, output[1] + 20)), "OUT", fill=INK, font=_font(12, True))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text(_pt((28, 20)), "FLIP EACH RED-RINGED INPUT", fill=INK, font=_font(29, True))
    draw.text(_pt((30, 64)), "blue ALL  •  amber ANY  •  violet ONE", fill=INK, font=_font(19))
    panels = scene["panels"]
    for slot in scene["panel_order"]:
        _draw_panel(draw, int(slot), panels[int(slot)])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
