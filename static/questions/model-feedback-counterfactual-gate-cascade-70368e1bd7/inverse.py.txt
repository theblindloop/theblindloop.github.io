"""Independent fixed-layout inverse arm for the counterfactual gate raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 760
INK = np.array((35, 40, 47), dtype=np.float64)
OFF = np.array((255, 255, 255), dtype=np.float64)
RED = np.array((205, 54, 54), dtype=np.float64)
GATE_COLORS = {
    "and": np.array((48, 112, 190), dtype=np.float64),
    "or": np.array((224, 148, 39), dtype=np.float64),
    "xor": np.array((132, 76, 183), dtype=np.float64),
}
PANEL_ORIGINS = ((24, 116), (492, 116), (24, 432), (492, 432))
INPUT_OFFSETS = ((58, 74), (58, 124), (58, 174), (58, 224))
GATE_OFFSETS = ((158, 99), (269, 149), (380, 199))
OUTPUT_OFFSET = (430, 199)


class OracleAbstention(ValueError):
    pass


def _patch_rgb(arr: np.ndarray, x: int, y: int, radius: int = 4) -> np.ndarray:
    patch = arr[y - radius:y + radius + 1, x - radius:x + radius + 1]
    return np.median(patch.reshape(-1, 3), axis=0)


def _binary(arr: np.ndarray, x: int, y: int) -> int:
    rgb = _patch_rgb(arr, x, y)
    d_on = float(np.linalg.norm(rgb - INK))
    d_off = float(np.linalg.norm(rgb - OFF))
    if min(d_on, d_off) > 35 or abs(d_on - d_off) < 55:
        raise OracleAbstention(f"unclear binary lamp at {(x, y)}: {rgb.tolist()}")
    return int(d_on < d_off)


def _gate_kind(arr: np.ndarray, x: int, y: int) -> str:
    rgb = _patch_rgb(arr, x, y, 6)
    distances = {kind: float(np.linalg.norm(rgb - color)) for kind, color in GATE_COLORS.items()}
    kind = min(distances, key=distances.get)  # type: ignore[arg-type]
    if distances[kind] > 45:
        raise OracleAbstention(f"unclear gate at {(x, y)}: {rgb.tolist()}")
    return kind


def _marked(arr: np.ndarray, x: int, y: int) -> bool:
    crop = arr[y - 25:y + 26, x - 25:x + 26].astype(np.int16)
    yy, xx = np.ogrid[-25:26, -25:26]
    annulus = ((xx * xx + yy * yy) >= 15 * 15) & ((xx * xx + yy * yy) <= 24 * 24)
    red_like = (crop[:, :, 0] > 155) & (crop[:, :, 1] < 110) & (crop[:, :, 2] < 110)
    return int(np.count_nonzero(red_like & annulus)) >= 55


def _apply(kind: str, left: bool, right: bool) -> bool:
    if kind == "and":
        return left and right
    if kind == "or":
        return left or right
    if kind == "xor":
        return left != right
    raise OracleAbstention("unknown gate")


def _evaluate(values: list[int], kinds: list[str]) -> bool:
    first = _apply(kinds[0], bool(values[0]), bool(values[1]))
    second = _apply(kinds[1], first, bool(values[2]))
    return _apply(kinds[2], second, bool(values[3]))


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)
    changed: list[int] = []
    for slot, (ox, oy) in enumerate(PANEL_ORIGINS):
        inputs = [_binary(arr, ox + dx, oy + dy) for dx, dy in INPUT_OFFSETS]
        marks = [_marked(arr, ox + dx, oy + dy) for dx, dy in INPUT_OFFSETS]
        if sum(marks) != 1:
            raise OracleAbstention(f"panel {slot + 1} has {sum(marks)} intervention marks")
        kinds = [_gate_kind(arr, ox + dx, oy + dy) for dx, dy in GATE_OFFSETS]
        shown = _binary(arr, ox + OUTPUT_OFFSET[0], oy + OUTPUT_OFFSET[1])
        factual = int(_evaluate(inputs, kinds))
        if shown != factual:
            raise OracleAbstention(f"panel {slot + 1} output contradicts visible inputs")
        changed_inputs = list(inputs)
        index = marks.index(True)
        changed_inputs[index] = 1 - changed_inputs[index]
        if int(_evaluate(changed_inputs, kinds)) != shown:
            changed.append(slot)
    if len(changed) != 1:
        raise OracleAbstention(f"expected one changed output, recovered {changed}")
    return f"panel-{changed[0] + 1}"
