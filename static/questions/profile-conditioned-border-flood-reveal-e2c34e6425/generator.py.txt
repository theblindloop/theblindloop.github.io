"""Analytic sampler and crisp Pillow renderer for border-flood reveal."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

GRID_N = 20
SYMBOLS = ("diamond", "square", "plus", "X")


def _templates() -> dict[str, set[tuple[int, int]]]:
    square = {(r, c) for r in range(9) for c in range(9) if r in (0, 8) or c in (0, 8)}
    diamond = {(r, c) for r in range(9) for c in range(9) if abs(r - 4) + abs(c - 4) in (3, 4)}
    diamond |= {(2, 4), (6, 4), (4, 2), (4, 6)}
    plus = {(r, c) for r in range(9) for c in (3, 4, 5)}
    plus |= {(4, c) for c in range(9)}
    plus.remove((4, 4))
    xmark = {(i, i) for i in range(9)} | {(i, 8 - i) for i in range(9)}
    xmark.remove((4, 4))
    for r0, c0, sr, sc in ((0, 0, 1, 1), (0, 8, 1, -1), (8, 0, -1, 1), (8, 8, -1, -1)):
        xmark |= {(r0, c0 + sc), (r0 + sr, c0), (r0 + sr, c0 + 2 * sc), (r0 + 2 * sr, c0 + sc)}
    result = {"diamond": diamond, "square": square, "plus": plus, "X": xmark}
    assert all(len(cells) == 32 for cells in result.values())
    assert all(min(r for r, _ in cells) == 0 and max(r for r, _ in cells) == 8 for cells in result.values())
    assert all(min(c for _, c in cells) == 0 and max(c for _, c in cells) == 8 for cells in result.values())
    return result


TEMPLATES = _templates()
INKS = ((20, 25, 32), (35, 25, 28), (24, 34, 27), (28, 27, 42))
FRAMES = ((0, 156, 188), (0, 171, 183), (16, 148, 196))
GATES = ((224, 28, 132), (210, 34, 151), (232, 40, 119))


def _neighbors(cell: tuple[int, int]) -> list[tuple[int, int]]:
    r, c = cell
    return [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]


def _make_clutter(rng: random.Random, target: set[tuple[int, int]], side: int, index: int) -> list[list[int]]:
    start = ((index, 0), (0, index), (index, GRID_N - 1), (GRID_N - 1, index))[side]
    forbidden = set(target)
    for cell in target:
        forbidden.update(_neighbors(cell))
    occupied = {start}
    allowed = {(r, c) for r in range(GRID_N) for c in range(GRID_N)} - forbidden
    # First grow a connected, canvas-spanning backbone. This prevents the isolated
    # glyph from being a clean object beside one compact distractor blob.
    goals = [(0, 0), (0, GRID_N - 1), (GRID_N - 1, 0), (GRID_N - 1, GRID_N - 1),
             (0, GRID_N // 2), (GRID_N - 1, GRID_N // 2),
             (GRID_N // 2, 0), (GRID_N // 2, GRID_N - 1)]
    rng.shuffle(goals)
    for goal in goals:
        if goal not in allowed or goal in occupied:
            continue
        parent = {goal: None}
        queue = [goal]
        hit = None
        for cell in queue:
            if cell in occupied:
                hit = cell
                break
            nxt = [p for p in _neighbors(cell) if p in allowed and p not in parent]
            rng.shuffle(nxt)
            for p in nxt:
                parent[p] = cell
                queue.append(p)
        if hit is None:
            continue
        cell = hit
        while cell is not None:
            occupied.add(cell)
            cell = parent[cell]
    frontier = {p for p in _neighbors(start) if p in allowed and p not in occupied}
    for cell in list(occupied):
        frontier.update(p for p in _neighbors(cell) if p in allowed and p not in occupied)
    while len(occupied) < 170:
        if not frontier:
            raise RuntimeError("clutter growth exhausted")
        choices = sorted(frontier)
        cell = choices[rng.randrange(len(choices))]
        frontier.remove(cell)
        occupied.add(cell)
        for p in _neighbors(cell):
            if 0 <= p[0] < GRID_N and 0 <= p[1] < GRID_N and p not in forbidden and p not in occupied:
                frontier.add(p)
    cells = [list(p) for p in sorted(occupied)]
    rng.shuffle(cells)
    return cells


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    top = rng.randint(5, 6)
    left = rng.randint(5, 6)
    symbol_cells = [[r + top, c + left] for r, c in sorted(TEMPLATES[symbol])]
    target = {tuple(p) for p in symbol_cells}
    side = rng.randrange(4)
    index = rng.choice([i for i in range(2, GRID_N - 2)])
    clutter_cells = _make_clutter(rng, target, side, index)
    cell_size = rng.choice((14, 15, 16))
    span = GRID_N * cell_size
    grid_x = rng.randint(48, 480 - span - 48)
    grid_y = rng.randint(48, 480 - span - 48)
    return {
        "symbol": symbol,
        "symbol_cells": symbol_cells,
        "clutter_cells": clutter_cells,
        "gate_side": side,
        "gate_index": index,
        "grid_x": grid_x,
        "grid_y": grid_y,
        "cell_size": cell_size,
        "ink_rgb": list(rng.choice(INKS)),
        "frame_rgb": list(rng.choice(FRAMES)),
        "gate_rgb": list(rng.choice(GATES)),
    }


def _gate_start(scene: dict) -> tuple[int, int]:
    side = int(scene["gate_side"])
    index = int(scene["gate_index"])
    return ((index, 0), (0, index), (index, GRID_N - 1), (GRID_N - 1, index))[side]


def render(scene: dict) -> Image.Image:
    # Direct accesses make every declared rendered field visible to the field audit.
    symbol = scene["symbol"]
    symbol_cells = scene["symbol_cells"]
    clutter_cells = scene["clutter_cells"]
    gate_side = scene["gate_side"]
    gate_index = scene["gate_index"]
    x = int(scene["grid_x"])
    y = int(scene["grid_y"])
    s = int(scene["cell_size"])
    ink = tuple(scene["ink_rgb"])
    frame = tuple(scene["frame_rgb"])
    gate = tuple(scene["gate_rgb"])
    if symbol not in TEMPLATES:
        raise ValueError("unknown symbol")
    if int(gate_side) not in range(4) or not 0 <= int(gate_index) < GRID_N:
        raise ValueError("invalid gate")
    image = Image.new("RGB", (480, 480), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    for r, c in [*clutter_cells, *symbol_cells]:
        x0, y0 = x + c * s + 2, y + r * s + 2
        draw.rectangle((x0, y0, x + (c + 1) * s - 2, y + (r + 1) * s - 2), fill=ink)
    grid_color = (190, 203, 207)
    for k in range(1, GRID_N):
        draw.line((x + k * s, y, x + k * s, y + GRID_N * s), fill=grid_color, width=1)
        draw.line((x, y + k * s, x + GRID_N * s, y + k * s), fill=grid_color, width=1)
    draw.rectangle((x, y, x + GRID_N * s, y + GRID_N * s), outline=frame, width=4)
    r, c = _gate_start(scene)
    cx, cy = x + c * s + s // 2, y + r * s + s // 2
    if c == 0:
        poly = [(x - 20, cy), (x - 6, cy - 8), (x - 6, cy + 8)]
    elif c == GRID_N - 1:
        poly = [(x + GRID_N * s + 20, cy), (x + GRID_N * s + 6, cy - 8), (x + GRID_N * s + 6, cy + 8)]
    elif r == 0:
        poly = [(cx, y - 20), (cx - 8, y - 6), (cx + 8, y - 6)]
    else:
        poly = [(cx, y + GRID_N * s + 20), (cx - 8, y + GRID_N * s + 6), (cx + 8, y + GRID_N * s + 6)]
    draw.polygon(poly, fill=gate)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def _iou(a: set[tuple[int, int]], b: set[tuple[int, int]]) -> float:
    return len(a & b) / len(a | b)


def margin(scene: dict) -> float:
    target = TEMPLATES[scene["symbol"]]
    scores = sorted((_iou(target, cells) for cells in TEMPLATES.values()), reverse=True)
    return round(100.0 * (scores[0] - scores[1]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0 or int(scene["cell_size"]) < 12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["clutter_cells"] = list(reversed(twin["clutter_cells"]))
    return [("reverse_clutter_storage_order", twin)]
