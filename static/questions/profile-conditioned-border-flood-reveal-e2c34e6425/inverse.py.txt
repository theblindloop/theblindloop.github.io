"""Independent PNG-only inverse: detect grid, flood gated ink, classify residue."""

from __future__ import annotations

import numpy as np


def _templates() -> dict[str, set[tuple[int, int]]]:
    square = {(r, c) for r in range(9) for c in range(9) if r in (0, 8) or c in (0, 8)}
    diamond = {(r, c) for r in range(9) for c in range(9) if abs(r - 4) + abs(c - 4) in (3, 4)}
    diamond |= {(2, 4), (6, 4), (4, 2), (4, 6)}
    plus = {(r, c) for r in range(9) for c in (3, 4, 5)} | {(4, c) for c in range(9)}
    plus.remove((4, 4))
    xmark = {(i, i) for i in range(9)} | {(i, 8 - i) for i in range(9)}
    xmark.remove((4, 4))
    for r0, c0, sr, sc in ((0, 0, 1, 1), (0, 8, 1, -1), (8, 0, -1, 1), (8, 8, -1, -1)):
        xmark |= {(r0, c0 + sc), (r0 + sr, c0), (r0 + sr, c0 + 2 * sc), (r0 + 2 * sr, c0 + sc)}
    return {"diamond": diamond, "square": square, "plus": plus, "X": xmark}


TEMPLATES = _templates()


def _groups(values: np.ndarray) -> list[int]:
    values = sorted(int(v) for v in values)
    groups: list[list[int]] = []
    for value in values:
        if not groups or value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [int(round(sum(group) / len(group))) for group in groups]


def _iou(a: set[tuple[int, int]], b: set[tuple[int, int]]) -> float:
    return len(a & b) / max(1, len(a | b))


def decision_with_margin(image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    cyan = (r < 80) & (g > 110) & (b > 135) & (b - r > 80)
    yy, xx = np.nonzero(cyan)
    if len(xx) < 100:
        raise ValueError("cyan frame missing")
    x0, x1, y0, y1 = int(xx.min()), int(xx.max()), int(yy.min()), int(yy.max())
    gridish = (abs(r - 190) <= 3) & (abs(g - 203) <= 3) & (abs(b - 207) <= 3)
    vscore = gridish[y0 + 3:y1 - 2, x0:x1 + 1].mean(axis=0)
    hscore = gridish[y0:y1 + 1, x0 + 3:x1 - 2].mean(axis=1)
    vx = _groups(np.flatnonzero(vscore > 0.82) + x0)
    hy = _groups(np.flatnonzero(hscore > 0.82) + y0)
    xs = [x0] + [v for v in vx if x0 + 4 < v < x1 - 4] + [x1]
    ys = [y0] + [v for v in hy if y0 + 4 < v < y1 - 4] + [y1]
    if len(xs) != 21 or len(ys) != 21:
        raise ValueError(f"ambiguous grid lines: {len(xs)}x{len(ys)}")
    occupied = np.zeros((20, 20), dtype=bool)
    for row in range(20):
        for col in range(20):
            patch = arr[ys[row] + 3:ys[row + 1] - 2, xs[col] + 3:xs[col + 1] - 2]
            occupied[row, col] = bool((patch.mean(axis=2) < 100).mean() > 0.5)
    magenta = (r > 160) & (g < 90) & (b > 85) & (r - g > 100)
    gy, gx = np.nonzero(magenta)
    if len(gx) < 20:
        raise ValueError("magenta gate missing")
    gcx, gcy = float(gx.mean()), float(gy.mean())
    distances = (abs(gcx - x0), abs(gcy - y0), abs(gcx - x1), abs(gcy - y1))
    side = int(np.argmin(distances))
    if side in (0, 2):
        row = min(19, max(0, int((gcy - y0) * 20 / (y1 - y0))))
        start = (row, 0 if side == 0 else 19)
    else:
        col = min(19, max(0, int((gcx - x0) * 20 / (x1 - x0))))
        start = (0 if side == 1 else 19, col)
    if not occupied[start]:
        raise ValueError("gated cell is not inked")
    seen = {start}
    stack = [start]
    while stack:
        row, col = stack.pop()
        for nr, nc in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            if 0 <= nr < 20 and 0 <= nc < 20 and occupied[nr, nc] and (nr, nc) not in seen:
                seen.add((nr, nc)); stack.append((nr, nc))
    residue = {(row, col) for row in range(20) for col in range(20) if occupied[row, col] and (row, col) not in seen}
    if not residue:
        raise ValueError("flood erases all ink")
    minr, minc = min(r for r, _ in residue), min(c for _, c in residue)
    normalized = {(r - minr, c - minc) for r, c in residue}
    scores = sorted(((name, _iou(normalized, cells)) for name, cells in TEMPLATES.items()), key=lambda item: (-item[1], item[0]))
    margin = 100.0 * (scores[0][1] - scores[1][1])
    if scores[0][1] < 0.90 or margin < 20.0:
        raise ValueError(f"ambiguous transformed symbol margin={margin:.2f}")
    return scores[0][0], round(margin, 6)


def decision_from_image(image) -> str:
    return decision_with_margin(image)[0]
