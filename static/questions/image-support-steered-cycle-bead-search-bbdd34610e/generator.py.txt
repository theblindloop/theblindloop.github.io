"""Deterministic scene sampler and renderer for loop-bead rail search."""

from __future__ import annotations

import random
from math import hypot

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 249, 246)
RAIL = (18, 151, 178)
OUTLINE = (35, 43, 48)
COLORS = {
    "red": (214, 57, 65),
    "gold": (224, 157, 28),
    "violet": (126, 75, 181),
    "green": (42, 147, 83),
}
RADIUS = 12
RAIL_WIDTH = 9


def _anchors(box: list[int]) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = box
    return [((x0 + x1) // 2, y0), (x1, (y0 + y1) // 2),
            ((x0 + x1) // 2, y1), (x0, (y0 + y1) // 2)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = tuple(COLORS)[int(seed) % 4]
    target_side = (int(seed) // 4) % 4
    width = 316 + 8 * rng.randrange(3)
    height = 78 + 6 * rng.randrange(3)
    cx = 320 + rng.randrange(-18, 19)
    cy = 250 + rng.randrange(-12, 13)
    box = [cx - width // 2, cy - height // 2, cx + width // 2, cy + height // 2]
    anchors = _anchors(box)

    remaining_colors = [name for name in COLORS if name != answer]
    rng.shuffle(remaining_colors)
    remaining_sides = [side for side in range(4) if side != target_side]
    beads = [{"color": answer, "center": list(anchors[target_side])}]
    branches = []
    normals = [(0, -1), (1, 0), (0, 1), (-1, 0)]
    tangents = [(1, 0), (0, 1), (-1, 0), (0, -1)]
    for color, side in zip(remaining_colors, remaining_sides, strict=True):
        ax, ay = anchors[side]
        nx, ny = normals[side]
        tx, ty = tangents[side]
        bend = rng.choice((-1, 1)) * (16 + 4 * rng.randrange(3))
        reach = 34 + 3 * rng.randrange(4)
        p1 = (ax + nx * reach, ay + ny * reach)
        p2 = (p1[0] + tx * bend, p1[1] + ty * bend)
        branches.append({"points": [[ax, ay], list(p1), list(p2)]})
        beads.append({"color": color, "center": list(p2)})
    rng.shuffle(beads)
    return {
        "cycle_box": box,
        "branches": branches,
        "beads": beads,
        "answer": answer,
    }


def render(scene: dict) -> Image.Image:
    box = [int(v) for v in scene["cycle_box"]]
    branches = scene["branches"]
    beads = scene["beads"]
    answer = scene["answer"]
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    x0, y0, x1, y1 = box
    loop = [(x0, y0), (x1, y0), (x1, y1), (x0, y1), (x0, y0)]
    draw.line(loop, fill=OUTLINE, width=RAIL_WIDTH + 4, joint="curve")
    draw.line(loop, fill=RAIL, width=RAIL_WIDTH, joint="curve")
    for branch in branches:
        points = [tuple(map(int, point)) for point in branch["points"]]
        draw.line(points, fill=OUTLINE, width=RAIL_WIDTH + 4, joint="curve")
        draw.line(points, fill=RAIL, width=RAIL_WIDTH, joint="curve")
    for bead in beads:
        x, y = map(int, bead["center"])
        fill = COLORS[str(bead["color"])]
        draw.ellipse((x - RADIUS - 2, y - RADIUS - 2, x + RADIUS + 2, y + RADIUS + 2), fill=OUTLINE)
        draw.ellipse((x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS), fill=fill)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["answer"])


def margin(scene: dict) -> float:
    box = [float(v) for v in scene["cycle_box"]]
    x0, y0, x1, y1 = box
    answer = str(scene["answer"])
    distances = []
    for bead in scene["beads"]:
        if str(bead["color"]) == answer:
            continue
        x, y = map(float, bead["center"])
        dx = max(x0 - x, 0.0, x - x1)
        dy = max(y0 - y, 0.0, y - y1)
        distances.append(hypot(dx, dy))
    return min(distances) - RADIUS if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
