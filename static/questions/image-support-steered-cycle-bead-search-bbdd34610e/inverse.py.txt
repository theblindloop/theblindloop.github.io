"""Pixel-only inverse arm: recover the bead bordering the enclosed background hole."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((214, 57, 65), dtype=np.int16),
    "gold": np.array((224, 157, 28), dtype=np.int16),
    "violet": np.array((126, 75, 181), dtype=np.int16),
    "green": np.array((42, 147, 83), dtype=np.int16),
}


def _enclosed_background(array: np.ndarray) -> np.ndarray:
    # Infer the paper color from the border, then work only in the visible-ink bbox.
    border = np.concatenate((array[:2].reshape(-1, 3), array[-2:].reshape(-1, 3),
                             array[:, :2].reshape(-1, 3), array[:, -2:].reshape(-1, 3)))
    paper = np.median(border, axis=0)
    ink = np.linalg.norm(array.astype(np.float32) - paper, axis=2) > 35.0
    ys, xs = np.nonzero(ink)
    if not len(xs):
        raise ValueError("no rail ink")
    y0, y1 = max(0, int(ys.min()) - 2), min(array.shape[0], int(ys.max()) + 3)
    x0, x1 = max(0, int(xs.min()) - 2), min(array.shape[1], int(xs.max()) + 3)
    local_ink = ink[y0:y1, x0:x1]
    background = ~local_ink
    outside = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    h, w = background.shape
    for x in range(w):
        if background[0, x]:
            outside[0, x] = True; queue.append((0, x))
        if background[h - 1, x] and not outside[h - 1, x]:
            outside[h - 1, x] = True; queue.append((h - 1, x))
    for y in range(h):
        if background[y, 0] and not outside[y, 0]:
            outside[y, 0] = True; queue.append((y, 0))
        if background[y, w - 1] and not outside[y, w - 1]:
            outside[y, w - 1] = True; queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and background[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    local_holes = background & ~outside
    if int(local_holes.sum()) < 2500:
        raise ValueError("no robust closed loop")
    holes = np.zeros(ink.shape, dtype=bool)
    holes[y0:y1, x0:x1] = local_holes
    return holes


def _dilate(mask: np.ndarray, steps: int = 3) -> np.ndarray:
    out = mask.copy()
    for _ in range(steps):
        grown = out.copy()
        grown[1:] |= out[:-1]
        grown[:-1] |= out[1:]
        grown[:, 1:] |= out[:, :-1]
        grown[:, :-1] |= out[:, 1:]
        out = grown
    return out


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    holes = _enclosed_background(array)
    neighborhood = _dilate(holes, 3)
    scores = []
    signed = array.astype(np.int16)
    for name, rgb in PALETTE.items():
        marker = np.linalg.norm(signed - rgb, axis=2) < 18.0
        score = int(np.count_nonzero(marker & neighborhood))
        scores.append((score, name))
    scores.sort(reverse=True)
    if scores[0][0] < 8 or scores[0][0] - scores[1][0] < 8:
        raise ValueError("no unique cycle bead")
    return scores[0][1]
