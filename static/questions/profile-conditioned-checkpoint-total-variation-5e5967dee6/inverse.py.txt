"""Independent pixel-only inverse arm for checkpoint total variation."""

from __future__ import annotations

import numpy as np
from PIL import Image

PROTOTYPES = {
    "blue": np.array((32, 112, 210), dtype=np.int16),
    "orange": np.array((232, 126, 35), dtype=np.int16),
    "green": np.array((38, 154, 91), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float]] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(xs) >= 120:
            found.append((len(xs), float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    totals: dict[str, float] = {}
    for name, prototype in PROTOTYPES.items():
        distance2 = ((arr - prototype.astype(np.int32)) ** 2).sum(axis=2)
        parts = _components(distance2 < 3600)
        if len(parts) != 5:
            raise ValueError(f"expected five isolated {name} checkpoints, found {len(parts)}")
        points = sorted(parts, key=lambda item: item[1])
        ys = [item[2] for item in points]
        totals[name] = sum(abs(b - a) for a, b in zip(ys, ys[1:]))
    ordered = sorted(totals.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 10.0:
        raise ValueError("pixel decision is inside the ambiguity band")
    return ordered[0][0]
