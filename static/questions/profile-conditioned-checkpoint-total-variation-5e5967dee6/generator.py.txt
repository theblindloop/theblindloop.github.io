"""Renderer and analytic arm for checkpoint total variation."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
PLOT = (76, 45, 606, 352)
COLORS = {
    "blue": [(32, 112, 210), (27, 105, 198), (40, 118, 218)],
    "orange": [(232, 126, 35), (221, 116, 27), (240, 137, 43)],
    "green": [(38, 154, 91), (30, 145, 83), (48, 164, 100)],
}


def _variation(values: list[int]) -> int:
    return sum(abs(b - a) for a, b in zip(values, values[1:]))


def _walk(start: int, steps: list[int]) -> list[int]:
    out = [start]
    sign = -1
    for step in steps:
        out.append(out[-1] + sign * step)
        sign *= -1
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = ["blue", "orange", "green"]
    winner = names[int(seed) % 3]
    remainder = [name for name in names if name != winner]
    rng.shuffle(remainder)
    if int(seed) % 8 == 0:
        step_sets = ([45, 42, 40, 38], [41, 39, 36, 34], [26, 25, 25, 24])
    else:
        step_sets = ([105, 118, 98, 110], [78, 73, 76, 70], [42, 38, 40, 36])
    assigned = {winner: step_sets[0], remainder[0]: step_sets[1], remainder[1]: step_sets[2]}
    series: dict[str, list[int]] = {}
    for name in names:
        steps = list(assigned[name])
        if rng.random() < 0.5:
            steps.reverse()
        values = _walk(205 + rng.randint(-12, 12), steps)
        if rng.random() < 0.5:
            values = [410 - value for value in values]
        if rng.random() < 0.5:
            values.reverse()
        lo, hi = min(values), max(values)
        shift = rng.randint(76 - lo, 337 - hi)
        series[name] = [value + shift for value in values]
    base_x = [106, 223, 340, 457, 574]
    x_positions = [x + rng.randint(-10, 10) for x in base_x]
    palette = {name: list(rng.choice(COLORS[name])) for name in names}
    return {"x_positions": x_positions, "series": series, "palette": palette}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    left, top, right, bottom = PLOT
    for y in range(80, 341, 52):
        draw.line((left, y, right, y), fill=(220, 223, 224), width=1)
    draw.line((left, top, left, bottom), fill=(45, 48, 50), width=2)
    draw.line((left, bottom, right, bottom), fill=(45, 48, 50), width=2)
    for index, x in enumerate(scene["x_positions"], start=1):
        draw.line((x, bottom, x, bottom + 5), fill=(45, 48, 50), width=2)
        draw.text((x - 3, bottom + 9), str(index), fill=(45, 48, 50))
    draw.text((253, 389), "CHECKPOINT (LEFT TO RIGHT)", fill=(45, 48, 50))
    draw.text((14, 17), "VERTICAL VALUE", fill=(45, 48, 50))
    x_dodge = {"blue": -22, "orange": 0, "green": 22}
    for name in ("blue", "orange", "green"):
        fill = tuple(scene["palette"][name])
        for x, y in zip(scene["x_positions"], scene["series"][name]):
            x += x_dodge[name]
            radius = 9
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    totals = {name: _variation(values) for name, values in scene["series"].items()}
    return max(totals, key=totals.get)


def margin(scene: dict) -> float:
    totals = sorted((_variation(values) for values in scene["series"].values()), reverse=True)
    return float(totals[0] - totals[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
