"""Independent pixel-only inverse arm for the attributed-graph world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (this file's own specification):

  * the top panel holds one ``reference`` graph: four colored nodes at the
    corners of a square, some of the four sides drawn as dark edges;
  * the bottom panel holds a row of three ``candidate`` graphs (left to right
    labeled A, B, C) with the same node colors and corner layout, differing
    only in which sides carry a dark edge;
  * the task: decide which candidate has the same adjacency graph (same set of
    drawn sides) as the reference.

The oracle recovers, from pixels alone: the divider, each graph's node corner
positions, each graph's set of drawn side-edges, and finally which candidate's
edge set equals the reference's.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "attributed_graph_match-oracle-0.1.0"

NODE_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}
_BACKGROUND_THRESH = 200.0
_EDGE_MAX = 120.0
NODE_R = 24
SIDE_CORNER_PAIRS = [(0, 1), (1, 3), (2, 3), (0, 2)]  # T, R, B, L
MIN_NODE_PX = 300


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    return arr.max(axis=2) < _EDGE_MAX


def _node_mask(arr: np.ndarray) -> np.ndarray:
    rgbs = np.array(list(NODE_PALETTE.values()), dtype=np.float32)
    d = np.linalg.norm(arr[..., None, :] - rgbs[None, None, :, :], axis=-1)
    return d.min(axis=-1) < 60.0


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connectivity; returns pixel (x, y) arrays per component."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    comps = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(x, y)])
                visited[y, x] = True
                cur = []
                while queue:
                    cx, cy = queue.popleft()
                    cur.append((cx, cy))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = cx + dx, cy + dy
                        if 0 <= nx < w and 0 <= ny < h and mask[ny, nx] and not visited[ny, nx]:
                            visited[ny, nx] = True
                            queue.append((nx, ny))
                comps.append(np.array(cur, dtype=np.float32))
    return comps


def _graph_edge_bits(arr: np.ndarray, node_pts: list[np.ndarray]) -> int:
    """Recover the adjacency bitmask of one graph from its four node points."""
    centers = np.array([pts.mean(axis=0) for pts in node_pts], dtype=np.float32)
    cx, cy = centers.mean(axis=0)
    corners = {}
    for i, c in enumerate(centers):
        row = 0 if c[1] < cy else 1
        col = 0 if c[0] < cx else 1
        corners[row * 2 + col] = c
    bits = 0
    for side, (a, b) in enumerate(SIDE_CORNER_PAIRS):
        p1, p2 = corners[a], corners[b]
        present = _side_has_edge(arr, p1, p2)
        if present:
            bits |= 1 << side
    return bits


def _side_has_edge(arr: np.ndarray, p1: np.ndarray, p2: np.ndarray) -> bool:
    """True if a dark edge is visible between two corner nodes (mid-band probe)."""
    h, w = arr.shape[:2]
    dx, dy = p2 - p1
    length = float(np.hypot(dx, dy))
    if length < 1.0:
        return False
    dark = 0
    count = 0
    for t in np.linspace(0.35, 0.65, 21):
        x = int(round(p1[0] + dx * t))
        y = int(round(p1[1] + dy * t))
        if 0 <= x < w and 0 <= y < h:
            count += 1
            if arr[y, x].max() < _EDGE_MAX:
                dark += 1
    return count > 0 and dark / count > 0.5


def _extract_graphs(arr: np.ndarray, region: slice) -> list[dict]:
    """Find all node discs in a region and cluster them into graphs."""
    sub = arr[region]
    yoff = region.start
    node_mask = _node_mask(sub)
    comps = [c for c in _connected_components(node_mask) if len(c) >= MIN_NODE_PX]
    if not comps:
        return []
    comps = [np.column_stack([c[:, 0], c[:, 1] + yoff]) for c in comps]
    comps.sort(key=lambda c: (c[:, 0].mean(), c[:, 1].mean()))
    # Group components that belong to the same graph (same x-cluster).
    clusters: list[list[np.ndarray]] = []
    for comp in comps:
        xm = comp[:, 0].mean()
        placed = False
        for cl in clusters:
            cxm = np.mean([c[:, 0].mean() for c in cl])
            if abs(cxm - xm) < 150.0:
                cl.append(comp)
                placed = True
                break
        if not placed:
            clusters.append([comp])
    return clusters


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    h, w = arr.shape[:2]

    dark = _dark_mask(arr)

    grid = np.zeros(h, dtype=bool)
    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * w
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = int(round(float(rows.mean())))
    divider_low = int(rows.max()) + 1

    top = slice(0, divider_mid)
    ref_clusters = _extract_graphs(arr, top)
    if len(ref_clusters) != 1 or len(ref_clusters[0]) != 4:
        raise ValueError("reference graph nodes not recovered")
    ref_bits = _graph_edge_bits(arr, ref_clusters[0])

    bottom = slice(divider_low, h)
    cand_clusters = _extract_graphs(arr, bottom)
    cand_clusters.sort(key=lambda cl: np.mean([c[:, 0].mean() for c in cl]))
    if len(cand_clusters) != 3:
        raise ValueError(f"expected 3 candidate graphs, got {len(cand_clusters)}")
    for cluster in cand_clusters:
        if len(cluster) != 4:
            raise ValueError("candidate graph node recovery failed")

    for label, cluster in zip("ABC", cand_clusters, strict=True):
        bits = _graph_edge_bits(arr, cluster)
        if bits == ref_bits:
            return label
    raise ValueError("no candidate matched the reference adjacency")
