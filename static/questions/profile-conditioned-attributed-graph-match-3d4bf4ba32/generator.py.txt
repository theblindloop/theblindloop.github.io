"""Attributed-graph correspondence world renderer (paper-semantic family).

A cross-panel structural-correspondence task. The top panel shows a ``reference``
graph: four colored nodes placed at the corners of a square, with some of the
four square sides drawn as dark connecting edges (an adjacency structure). The
bottom panel shows a row of three ``candidate`` graphs that share the *same*
node colors and same corner layout, but differ in which sides are drawn (which
adjacencies are present). Exactly one candidate has the identical adjacency
graph as the reference. The question asks which candidate (labeled A, B, C)
preserves the reference's colored-part adjacency graph.

The pixel-only inverse arm must extract each node as a local colored part and
its relational edge structure (which corner-to-corner sides carry a visible
edge), then solve the cross-panel assignment by structural equality.

Latent state z (all influencing fields are rendered -- nothing hidden):

    ref_edges   : reference adjacency bitmask; bits index the four sides
                  [Top, Right, Bottom, Left] present in the reference graph.
    cand_edges  : length-3 adjacency bitmask for candidates A, B, C. Exactly
                  one equals ``ref_edges`` (the structural match).
    node_colors : colors assigned to the four corners [TL, TR, BL, BR]
                  (same assignment for every panel of a scene).
    graph_center, cand_centers, corner_offsets, width, height : geometry.

The analytic gold is the label (A/B/C) of the candidate whose adjacency bitmask
equals the reference's. Because every such field (edge presence and node color)
is rendered into the raster, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "attributed_graph_match-0.1.0"

BACKGROUND = "white"
EDGE_COLOR = (45, 45, 45)
DIVIDER = (80, 80, 80)

NODE_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for counterbalance of ink).
WIDTH = 900
HEIGHT = 560
DIVIDER_Y = 220
DIVIDER_THICK = 3
NODE_R = 24
EDGE_HALF = 6          # half-thickness of a drawn edge
REF_CENTER = (450, 112)
CAND_Y = 388
CAND_XS = [210, 510, 810]
# Corners of a square centered on each graph; [TL, TR, BL, BR].
CORNER_OFFSETS = [(-60, -60), (60, -60), (-60, 60), (60, 60)]
# Four sides as (start_corner_index, end_corner_index) ; bit order matters.
SIDES: list[tuple[int, int]] = [(0, 1), (1, 3), (2, 3), (0, 2)]  # T, R, B, L

_MARGIN_FLOOR = 2.0  # require distractors to differ in at least two edges


def _node_positions(center) -> list[tuple[float, float]]:
    cx, cy = center
    return [(cx + ox, cy + oy) for ox, oy in CORNER_OFFSETS]


def _side_centers(corners, side) -> list[tuple[float, float]]:
    a, b = SIDES[side]
    ax, ay = corners[a]
    bx, by = corners[b]
    return [(ax * (1 - t) + bx * t, ay * (1 - t) + by * t) for t in (0.30, 0.50, 0.70)]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    names = list(NODE_PALETTE)
    rng.shuffle(names)
    node_colors = list(names)  # assign to [TL, TR, BL, BR]

    ref_edges = rng.randint(1, 14)
    candidates = [ref_edges]
    while len(candidates) < 3:
        m = rng.randint(1, 14)
        if m == ref_edges or m in candidates:
            continue
        if all(bin(m ^ ref_edges).count("1") >= 2 for _ in [0]):
            candidates.append(m)
    rng.shuffle(candidates)
    return {
        "ref_edges": ref_edges,
        "cand_edges": candidates,
        "node_colors": node_colors,
        "graph_center": list(REF_CENTER),
        "cand_centers": [[x, CAND_Y] for x in CAND_XS],
        "corner_offsets": [list(o) for o in CORNER_OFFSETS],
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_graph(draw, center, edges_mask, node_colors):
    corners = _node_positions(center)
    # Edges first so they sit beneath node discs.
    for side, (a, b) in enumerate(SIDES):
        if edges_mask & (1 << side):
            ax, ay = corners[a]
            bx, by = corners[b]
            draw.line([ax, ay, bx, by], fill=EDGE_COLOR, width=EDGE_HALF * 2)
    x, y = center
    for (ax, ay), color in zip(corners, node_colors, strict=True):
        draw.ellipse(
            [ax - NODE_R, ay - NODE_R, ax + NODE_R, ay + NODE_R],
            fill=NODE_PALETTE[color],
        )


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution (crisp edges)."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_graph(draw, tuple(scene["graph_center"]), int(scene["ref_edges"]), scene["node_colors"])
    for center, mask in zip(scene["cand_centers"], scene["cand_edges"], strict=True):
        _draw_graph(draw, tuple(center), int(mask), scene["node_colors"])
    draw.rectangle([0, DIVIDER_Y, w, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _matched_index(scene: dict) -> int:
    return scene["cand_edges"].index(scene["ref_edges"])


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate with the reference adjacency."""
    return "ABC"[_matched_index(scene)]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum number of differing edges between the reference
    adjacency and any distractor candidate. Large -> unambiguous match."""
    ref = int(scene["ref_edges"])
    diffs = [
        bin(ref ^ int(m)).count("1")
        for m in scene["cand_edges"]
        if int(m) != ref
    ]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene: never produced, because the sampler
    keeps distractors at least two edges away from the reference."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (edge presence, node
    colors) is rendered verbatim, so the raster uniquely determines the gold;
    no pixel-identical transform is declared."""
    return []
