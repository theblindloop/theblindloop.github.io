"""Deterministic renderer for rectangle-constellation visual search."""

from __future__ import annotations

import math
import random
from itertools import combinations

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
LABELS = ("A", "B", "C", "D")
CARD_BOXES = (
    (105, 72, 535, 142),
    (105, 158, 535, 228),
    (105, 244, 535, 314),
    (105, 330, 535, 400),
)
CYAN = (18, 176, 200)
INK = (38, 48, 58)
BACKGROUND = (248, 246, 239)


def _pairings() -> tuple[tuple[tuple[int, int], tuple[int, int]], ...]:
    return (((0, 1), (2, 3)), ((0, 2), (1, 3)), ((0, 3), (1, 2)))


def _rectangle_mismatch(points: list[list[float]]) -> float:
    """Zero iff some two equal diagonals bisect each other (a rectangle)."""
    best = float("inf")
    for (a, b), (c, d) in _pairings():
        p, q, r, s = points[a], points[b], points[c], points[d]
        midpoint_gap = math.hypot((p[0] + q[0]) - (r[0] + s[0]), (p[1] + q[1]) - (r[1] + s[1]))
        len_pq = math.hypot(p[0] - q[0], p[1] - q[1])
        len_rs = math.hypot(r[0] - s[0], r[1] - s[1])
        best = min(best, midpoint_gap + abs(len_pq - len_rs))
    return best


def _rotated_rectangle(cx: float, cy: float, half_w: float, half_h: float, angle: float) -> list[list[float]]:
    ca, sa = math.cos(angle), math.sin(angle)
    points = []
    for x, y in ((-half_w, -half_h), (half_w, -half_h), (half_w, half_h), (-half_w, half_h)):
        points.append([cx + x * ca - y * sa, cy + x * sa + y * ca])
    return points


def _points_clear(points: list[list[float]], box: tuple[int, int, int, int]) -> bool:
    x0, y0, x1, y1 = box
    if any(not (x0 + 24 <= x <= x1 - 18 and y0 + 11 <= y <= y1 - 11) for x, y in points):
        return False
    return all(math.hypot(points[a][0] - points[b][0], points[a][1] - points[b][1]) >= 24 for a, b in combinations(range(4), 2))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    cards = []
    for index, (x0, y0, x1, y1) in enumerate(CARD_BOXES):
        cx = (x0 + x1) / 2 + rng.uniform(-3, 3)
        cy = (y0 + y1) / 2 + rng.uniform(-1.5, 1.5)
        half_w = rng.uniform(142, 151)
        half_h = rng.uniform(17, 19)
        angle = math.radians(rng.uniform(-1, 1))
        points = _rotated_rectangle(cx, cy, half_w, half_h, angle)
        if index != winner:
            # Move one corner: count, color, size, and approximate extent remain fixed,
            # while the equal-bisecting-diagonals invariant is broken.
            base = [point[:] for point in points]
            for _ in range(100):
                points = [point[:] for point in base]
                corner = rng.randrange(4)
                direction = rng.uniform(0, 2 * math.pi)
                displacement = rng.uniform(18, 27)
                points[corner][0] += displacement * math.cos(direction)
                points[corner][1] += displacement * math.sin(direction)
                if _points_clear(points, CARD_BOXES[index]) and _rectangle_mismatch(points) >= 14.0:
                    break
            else:
                raise RuntimeError("could not sample a clear non-rectangular constellation")
        cards.append({"label": LABELS[index], "points": [[round(x, 3), round(y, 3)] for x, y in points]})

    return {"cards": cards}


def analytic_gold(scene: dict) -> str:
    scores = [_rectangle_mismatch(card["points"]) for card in scene["cards"]]
    return LABELS[min(range(4), key=scores.__getitem__)]


def margin(scene: dict) -> float:
    scores = sorted(_rectangle_mismatch(card["points"]) for card in scene["cards"])
    return scores[1] - scores[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = _font(19 * scale)
    title_font = _font(20 * scale)
    draw.text((WIDTH * scale / 2, 38 * scale), "FIND THE RECTANGLE", fill=INK, font=title_font, anchor="mm")
    for index, card in enumerate(scene["cards"]):
        x0, y0, x1, y1 = CARD_BOXES[index]
        box = tuple(int(v * scale) for v in (x0, y0, x1, y1))
        draw.rounded_rectangle(box, radius=12 * scale, fill=(255, 255, 255), outline=(118, 126, 132), width=3 * scale)
        draw.text(((x0 + 16) * scale, (y0 + 14) * scale), card["label"], fill=INK, font=font, anchor="mm")
        for x, y in card["points"]:
            r = 10 * scale
            cx, cy = x * scale, y * scale
            draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=CYAN)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
