"""Independent pixel-only inverse arm for rectangle-constellation search."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            yy, xx = stack.pop()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 120:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _lane_bounds(arr: np.ndarray) -> list[tuple[float, float]]:
    """Enumerate the four visible card lanes from their gray horizontal rims."""
    border = np.array((118, 126, 132), dtype=np.int32)
    gray = np.sum((arr - border) ** 2, axis=2) < 42**2
    active = np.count_nonzero(gray, axis=1) > 150
    runs: list[list[int]] = []
    for y in np.flatnonzero(active):
        if not runs or y > runs[-1][-1] + 1:
            runs.append([int(y)])
        else:
            runs[-1].append(int(y))
    centers = [float(np.mean(run)) for run in runs if len(run) >= 1]
    if len(centers) != 8:
        raise ValueError(f"expected eight visible horizontal card rims, found {len(centers)}")
    return [(centers[i], centers[i + 1]) for i in range(0, 8, 2)]


def _mismatch(points: list[tuple[float, float]]) -> float:
    pairings = (((0, 1), (2, 3)), ((0, 2), (1, 3)), ((0, 3), (1, 2)))
    best = float("inf")
    for (a, b), (c, d) in pairings:
        p, q, r, s = points[a], points[b], points[c], points[d]
        mid = math.hypot((p[0] + q[0]) - (r[0] + s[0]), (p[1] + q[1]) - (r[1] + s[1]))
        d1 = math.hypot(p[0] - q[0], p[1] - q[1])
        d2 = math.hypot(r[0] - s[0], r[1] - s[1])
        best = min(best, mid + abs(d1 - d2))
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    target = np.array((18, 176, 200), dtype=np.int32)
    mask = np.sum((arr - target) ** 2, axis=2) < 45**2
    dots = _components(mask)
    lanes = _lane_bounds(arr)
    scored: list[tuple[float, int]] = []
    for index, (top, bottom) in enumerate(lanes):
        group = [(x, y) for x, y, _ in dots if top < y < bottom]
        if len(group) == 4:
            scored.append((_mismatch(group), index))
    if not scored:
        raise ValueError("no intact four-dot candidate lane remains")
    scored.sort()
    if len(dots) == 16 and (len(scored) != 4 or scored[1][0] - scored[0][0] < 7.0):
        raise ValueError("rectangle winner lacks a pixel-measured runner-up margin")
    return LABELS[scored[0][1]]
