"""Independent pixel-only inverse for state/motion concordance."""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 6
IDENTITY_COUNT = 4
SLOT_COUNT = 4
STATE_COUNT = 3
LABELS = ("A", "B", "C", "D")
CANVAS = (1940, 640)
INK = (28, 45, 58)
PANEL_X0 = 20
PANEL_TOP = 210
PANEL_WIDTH = 300
PANEL_GAP = 8
CARD_ORIGINS = ((8, 72), (82, 72), (156, 72), (230, 72))
LABEL_OFFSET = (8, 42)
STATE_OFFSET = (35, 42)
STATE_BOX_SIZE = 22
STATE_KEY_XS = (430, 650, 870)
KEY_Y = 104
STATE_CODE_OFFSET = (32, 5)

LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)
STATE_CODE_PATTERNS = (
    ("01110", "10001", "10011", "10101", "11001", "10001", "01110"),
    ("00100", "01100", "00100", "00100", "00100", "00100", "01110"),
    ("01110", "10001", "00001", "00010", "00100", "01000", "11111"),
)


def _points(pattern: tuple[str, ...], scale: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * scale + dx, y * scale + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(scale)
        for dy in range(scale)
    )


LABEL_POINTS = tuple(_points(pattern, 3) for pattern in LABEL_PATTERNS)
STATE_CODE_POINTS = tuple(_points(pattern, 2) for pattern in STATE_CODE_PATTERNS)


def _read_exact(frame: Image.Image, origin: tuple[int, int], width: int, height: int) -> frozenset[tuple[int, int]]:
    ox, oy = origin
    return frozenset(
        (x, y)
        for x in range(width)
        for y in range(height)
        if frame.getpixel((ox + x, oy + y)) == INK
    )


def _read_identity(frame: Image.Image, origin: tuple[int, int]) -> int:
    observed = _read_exact(frame, origin, 18, 23)
    matches = [index for index, points in enumerate(LABEL_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("identity glyph is missing, altered, or ambiguous")
    return matches[0]


def _read_code(frame: Image.Image, origin: tuple[int, int]) -> int:
    observed = _read_exact(frame, origin, 12, 16)
    matches = [index for index, points in enumerate(STATE_CODE_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("state key code is missing or ambiguous")
    return matches[0]


def _decode_state_key(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    mapping: dict[tuple[int, int, int], int] = {}
    for key_x in STATE_KEY_XS:
        color = frame.getpixel((key_x + STATE_BOX_SIZE // 2, KEY_Y + STATE_BOX_SIZE // 2))
        state = _read_code(frame, (key_x + STATE_CODE_OFFSET[0], KEY_Y + STATE_CODE_OFFSET[1]))
        if color in mapping:
            raise ValueError("state key repeats a color")
        mapping[color] = state
    if len(mapping) != STATE_COUNT or set(mapping.values()) != set(range(STATE_COUNT)):
        raise ValueError("state key does not expose LOW, MID, HIGH")
    return mapping


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique winner from raster pixels alone."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_key = _decode_state_key(frame)
    state_tables: list[dict[int, int]] = []
    position_tables: list[dict[int, int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        states: dict[int, int] = {}
        positions: dict[int, int] = {}
        for slot, (slot_x, slot_y) in enumerate(CARD_ORIGINS):
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            label = _read_identity(frame, (card_x + LABEL_OFFSET[0], card_y + LABEL_OFFSET[1]))
            if label in states:
                raise ValueError("panel repeats an identity")
            color = frame.getpixel(
                (card_x + STATE_OFFSET[0] + STATE_BOX_SIZE // 2, card_y + STATE_OFFSET[1] + STATE_BOX_SIZE // 2)
            )
            if color not in state_key:
                raise ValueError("tile state swatch is absent or not keyed")
            states[label] = state_key[color]
            positions[label] = slot
        if set(states) != set(range(IDENTITY_COUNT)):
            raise ValueError("panel does not expose all four identities")
        state_tables.append(states)
        position_tables.append(positions)

    scores = [0] * IDENTITY_COUNT
    for label in range(IDENTITY_COUNT):
        for panel in range(PANEL_COUNT - 1):
            state_delta = (state_tables[panel + 1][label] > state_tables[panel][label]) - (
                state_tables[panel + 1][label] < state_tables[panel][label]
            )
            move_delta = (position_tables[panel + 1][label] > position_tables[panel][label]) - (
                position_tables[panel + 1][label] < position_tables[panel][label]
            )
            if state_delta == move_delta:
                scores[label] += 1
    ordered = sorted(scores, reverse=True)
    winners = [label for label, score in enumerate(scores) if score == ordered[0]]
    if len(winners) != 1 or ordered[0] - ordered[1] < 2:
        raise ValueError("image does not expose one unique concordance winner")
    return LABELS[winners[0]]
