"""Deterministic renderer for the Confluence Checkpoint world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (250, 250, 247)
ROUTE = (28, 40, 58)
COLORS = {
    "blue": (40, 108, 220),
    "cyan": (24, 178, 190),
    "orange": (235, 132, 35),
    "red": (220, 62, 62),
    "green": (58, 166, 91),
}


def _transform(point: tuple[int, int], rotation: int, reflect: bool, dx: int, dy: int) -> list[int]:
    x, y = point
    x -= 256
    y -= 256
    if reflect:
        x = -x
    for _ in range(rotation % 4):
        x, y = -y, x
    return [x + 256 + dx, y + 256 + dy]


def _path(points: Iterable[tuple[int, int]], rotation: int, reflect: bool, dx: int, dy: int) -> list[list[int]]:
    return [_transform(p, rotation, reflect, dx, dy) for p in points]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, nuisance-varied tree with two arms and one trunk."""
    rng = random.Random(int(seed))
    rotation = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    dx = rng.randint(-4, 4)
    dy = rng.randint(-4, 4)
    route_width = rng.choice((14, 16, 18))
    marker_radius = rng.choice((9, 10))

    left = [(190, 174), (190, 200), (226, 200), (226, 222), (190, 222),
            (190, 244), (226, 244), (226, 266), (244, 266), (256, 278)]
    right = [(322, 174), (322, 200), (286, 200), (286, 222), (322, 222),
             (322, 244), (286, 244), (286, 266), (268, 266), (256, 278)]
    trunk = [(256, 278), (256, 294), (198, 294), (198, 318),
             (314, 318), (314, 340), (256, 340), (256, 356)]

    branches = [_path(left, rotation, reflect, dx, dy), _path(right, rotation, reflect, dx, dy)]
    if rng.randrange(2):
        branches.reverse()  # draw-list position is deliberately meaningless
    trunk_t = _path(trunk, rotation, reflect, dx, dy)
    merge = trunk_t[0]
    decoy = trunk_t[3]
    merge_color = "red" if int(seed) % 2 == 0 else "green"
    red_checkpoint = merge if merge_color == "red" else decoy
    green_checkpoint = merge if merge_color == "green" else decoy

    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": list(BACKGROUND),
        "route_color": list(ROUTE),
        "route_width": route_width,
        "marker_radius": marker_radius,
        "branch_paths": branches,
        "trunk_path": trunk_t,
        "start_blue": _transform(left[0], rotation, reflect, dx, dy),
        "start_cyan": _transform(right[0], rotation, reflect, dx, dy),
        "finish": trunk_t[-1],
        "red_checkpoint": list(red_checkpoint),
        "green_checkpoint": list(green_checkpoint),
        "merge_color": merge_color,
    }


def _scaled(points: list[list[int]], scale: int) -> list[tuple[int, int]]:
    return [(int(x) * scale, int(y) * scale) for x, y in points]


def render(scene: dict) -> Image.Image:
    """Render only information declared in the public scene dictionary."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    route_color = tuple(scene["route_color"])
    route_width = int(scene["route_width"])
    radius = int(scene["marker_radius"])
    branches = scene["branch_paths"]
    trunk = scene["trunk_path"]
    merge_color = scene["merge_color"]
    red_checkpoint = scene["red_checkpoint"]
    green_checkpoint = scene["green_checkpoint"]

    expected_merge = red_checkpoint if merge_color == "red" else green_checkpoint
    if list(expected_merge) != list(trunk[0]):
        raise ValueError("merge-color label is inconsistent with the visible junction")
    if sorted(tuple(p) for p in (red_checkpoint, green_checkpoint)) != sorted(
        (tuple(trunk[0]), tuple(trunk[3]))
    ):
        raise ValueError("checkpoint locations must be the merge and downstream decoy")

    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)
    for points in [*branches, trunk]:
        draw.line(_scaled(points, ss), fill=route_color, width=route_width * ss, joint="curve")

    markers = (
        (scene["start_blue"], "blue"),
        (scene["start_cyan"], "cyan"),
        (scene["finish"], "orange"),
        (red_checkpoint, "red"),
        (green_checkpoint, "green"),
    )
    for (x, y), name in markers:
        box = ((x - radius) * ss, (y - radius) * ss, (x + radius) * ss, (y + radius) * ss)
        draw.ellipse(box, fill=COLORS[name])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["merge_color"])


def _polyline_distance_until(points: list[list[int]], target: list[int]) -> float:
    total = 0.0
    for a, b in zip(points, points[1:]):
        total += math.dist(a, b)
        if list(b) == list(target):
            return total
    raise ValueError("target is not a trunk vertex")


def margin(scene: dict) -> float:
    """Geodesic clearance between the true confluence and the runner-up marker."""
    merge = scene["trunk_path"][0]
    decoy = scene["green_checkpoint"] if scene["merge_color"] == "red" else scene["red_checkpoint"]
    return _polyline_distance_until(scene["trunk_path"], decoy) - 2.0 * float(scene["marker_radius"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["branch_paths"] = list(reversed(twin["branch_paths"]))
    return [("swap_unlabeled_branch_draw_order", twin)]
