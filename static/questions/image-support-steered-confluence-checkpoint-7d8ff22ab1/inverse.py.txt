"""Independent pixel-only inverse for confluence localization.

The oracle segments five marker colors and the dark stroke union.  It then
reconstructs the route graph through connectivity.  Removing the true
confluence checkpoint separates blue, cyan, and orange into three different
components; removing the downstream checkpoint leaves blue and cyan together.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "route": (28, 40, 58),
    "blue": (40, 108, 220),
    "cyan": (24, 178, 190),
    "orange": (235, 132, 35),
    "red": (220, 62, 62),
    "green": (58, 166, 91),
}
_MARKERS = ("blue", "cyan", "orange", "red", "green")


def _near(array: np.ndarray, rgb: tuple[int, int, int], tolerance: float) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _marker(array: np.ndarray, name: str) -> tuple[tuple[int, int], float]:
    mask = _near(array, _COLORS[name], 48.0)
    ys, xs = np.nonzero(mask)
    area = len(xs)
    if not 170 <= area <= 430:
        raise ValueError(f"{name} marker area is missing or corrupted: {area}")
    cx = int(round(float(xs.mean())))
    cy = int(round(float(ys.mean())))
    return (cy, cx), math.sqrt(area / math.pi)


def _seed(mask: np.ndarray, point: tuple[int, int], radius: int = 8) -> tuple[int, int]:
    y, x = point
    h, w = mask.shape
    best = None
    for yy in range(max(0, y - radius), min(h, y + radius + 1)):
        for xx in range(max(0, x - radius), min(w, x + radius + 1)):
            if mask[yy, xx]:
                d = (yy - y) ** 2 + (xx - x) ** 2
                if best is None or d < best[0]:
                    best = (d, yy, xx)
    if best is None:
        raise ValueError("marker is detached from the route mask")
    return best[1], best[2]


def _flood(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sy, sx = _seed(mask, start)
    seen[sy, sx] = True
    queue = deque([(sy, sx)])
    while queue:
        y, x = queue.popleft()
        if y and mask[y - 1, x] and not seen[y - 1, x]:
            seen[y - 1, x] = True; queue.append((y - 1, x))
        if y + 1 < h and mask[y + 1, x] and not seen[y + 1, x]:
            seen[y + 1, x] = True; queue.append((y + 1, x))
        if x and mask[y, x - 1] and not seen[y, x - 1]:
            seen[y, x - 1] = True; queue.append((y, x - 1))
        if x + 1 < w and mask[y, x + 1] and not seen[y, x + 1]:
            seen[y, x + 1] = True; queue.append((y, x + 1))
    return seen


def _labels_for(mask: np.ndarray, points: list[tuple[int, int]]) -> list[int]:
    labels = np.zeros(mask.shape, dtype=np.int16)
    component = 0
    for point in points:
        sy, sx = _seed(mask, point)
        if labels[sy, sx] == 0:
            component += 1
            reached = _flood(mask, (sy, sx))
            labels[reached] = component
    return [int(labels[_seed(mask, point)]) for point in points]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 128 or array.shape[1] < 128:
        raise ValueError("unsupported image size")

    found = {name: _marker(array, name) for name in _MARKERS}
    palette_masks = [_near(array, rgb, 72.0) for rgb in _COLORS.values()]
    route_graph = np.logical_or.reduce(palette_masks)

    connected = _flood(route_graph, found["orange"][0])
    for name in _MARKERS:
        y, x = _seed(route_graph, found[name][0])
        if not connected[y, x]:
            raise ValueError(f"{name} is not connected to the complete route tree")

    winners: list[str] = []
    yy, xx = np.ogrid[: route_graph.shape[0], : route_graph.shape[1]]
    endpoints = [found["blue"][0], found["cyan"][0], found["orange"][0]]
    for color in ("red", "green"):
        (cy, cx), radius = found[color]
        cut_radius = max(20.0, 2.25 * radius)
        cut = (yy - cy) ** 2 + (xx - cx) ** 2 <= cut_radius * cut_radius
        reduced = route_graph & ~cut
        labels = _labels_for(reduced, endpoints)
        if len(set(labels)) == 3:
            winners.append(color)
    if len(winners) != 1:
        raise ValueError(f"ambiguous confluence checkpoints: {winners}")
    return winners[0]
