"""Independent PNG-only inverse for the first USB branch junction.

No scene, renderer state, coordinates, metadata, or filenames are accepted.
The inverse segments the sole saturated connected trace, estimates its vertical
trunk from column support, then finds the row peak where each branch leaves the
near-trunk strip. Remote terminals and gray proof marks are ignored.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _peak_row(mask: np.ndarray, x0: int, x1: int, y0: int, y1: int) -> float:
    if x1 <= x0 or y1 <= y0:
        raise OracleAbstention("empty branch search strip")
    counts = mask[y0:y1, x0:x1].sum(axis=1).astype(np.float64)
    maximum = float(counts.max(initial=0.0))
    if maximum < 7.0:
        raise OracleAbstention("a branch-to-trunk join is not visibly supported")
    selected = np.flatnonzero(counts >= max(5.0, 0.72 * maximum))
    if not len(selected):
        raise OracleAbstention("no stable junction-row peak")
    weights = counts[selected]
    return float(y0 + np.average(selected, weights=weights))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleAbstention("RGB raster required")
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    saturation = array.max(axis=2) - array.min(axis=2)
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    ink = (saturation > 52) & (distance > 62.0)
    ys, xs = np.nonzero(ink)
    if len(xs) < 700:
        raise OracleAbstention("colored USB trace is missing or too damaged")

    # The trunk contributes the longest vertical run. Averaging near-maximal
    # support columns makes this robust to antialiasing and line-width changes.
    col_counts = ink.sum(axis=0)
    max_col = int(col_counts.max(initial=0))
    trunk_columns = np.flatnonzero(col_counts >= max(28, int(0.82 * max_col)))
    if not len(trunk_columns):
        raise OracleAbstention("vertical trunk could not be localized")
    cx = int(round(float(np.median(trunk_columns))))

    y_min, y_max = int(ys.min()), int(ys.max())
    height = y_max - y_min + 1
    if not 125 <= height <= 215:
        raise OracleAbstention("colored trace has an implausible vertical extent")
    # Both junctions lie in the central 56% of the emblem. The side strips stop
    # well before either terminal, so endpoint erasure is nondiagnostic.
    y0 = y_min + int(round(0.22 * height))
    y1 = y_min + int(round(0.78 * height))
    inner = max(8, int(round(0.050 * height)))
    outer = max(inner + 15, int(round(0.205 * height)))
    left_y = _peak_row(ink, max(0, cx - outer), max(0, cx - inner), y0, y1)
    right_y = _peak_row(
        ink, min(ink.shape[1], cx + inner), min(ink.shape[1], cx + outer), y0, y1
    )
    separation = abs(left_y - right_y)
    if separation < 12.0:
        raise OracleAbstention("junction order is a finite-resolution tie")
    return "left" if left_y < right_y else "right"
