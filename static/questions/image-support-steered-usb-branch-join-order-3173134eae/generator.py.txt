"""Deterministic latent sampler and Pillow renderer for a compact USB-like tree."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


INKS = (
    (20, 105, 210),
    (0, 145, 135),
    (155, 55, 190),
    (225, 92, 32),
    (30, 150, 75),
)
BACKGROUNDS = ((250, 249, 246), (248, 250, 252), (252, 250, 247))


def sample_scene(seed: int) -> dict:
    numeric_seed = int(seed)
    # Consecutive even/odd seeds are an exact matched pair: every nuisance is
    # shared and only the visible branch order changes.
    pair_seed = numeric_seed // 2
    rng = random.Random(pair_seed)
    scale = rng.choice((0.90, 1.00, 1.10))
    # Parity guarantees class balance for every ordinary consecutive seed range.
    upper_side = "left" if numeric_seed % 2 == 0 else "right"
    junction_gap = 30 if pair_seed % 12 == 0 else rng.choice((54, 56, 58))
    return {
        "width": 512,
        "height": 512,
        # Multiple grid-aligned anchors retain broad absolute translation while
        # preventing an arbitrary ablation-cell boundary from splitting the
        # same short junction cue differently across scenes.
        "cx": rng.choice((160, 224, 288, 352)),
        "cy": rng.choice((160, 224, 288, 352)),
        "scale": scale,
        "line_width": rng.choice((8, 9, 10)),
        "upper_side": upper_side,
        "junction_gap": junction_gap,
        "terminal_offset": rng.choice((-5, -2, 0, 2, 5)),
        "ink_rgb": list(rng.choice(INKS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "decor_variant": rng.randrange(8),
    }


def _point(cx: float, cy: float, scale: float, dx: float, dy: float) -> tuple[int, int]:
    return (round(cx + scale * dx), round(cy + scale * dy))


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    scale = float(scene["scale"])
    line_width = int(scene["line_width"])
    upper_side = str(scene["upper_side"])
    junction_gap = float(scene["junction_gap"])
    terminal_offset = float(scene["terminal_offset"])
    ink = tuple(int(value) for value in scene["ink_rgb"])
    background = tuple(int(value) for value in scene["background_rgb"])
    decor_variant = int(scene["decor_variant"])

    ss = 4
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)
    gray = (188, 191, 196)

    # Low-saturation proof marks are intentionally remote and nondiagnostic.
    for index in range(6):
        phase = (decor_variant * 3 + index * 5) % 17
        x = (40 + index * 79 + phase * 3) % (width - 40) + 20
        y = 34 + ((decor_variant * 41 + index * 67) % (height - 68))
        if abs(x - cx) < 105 and abs(y - cy) < 115:
            x = 24 if x < cx else width - 24
        r = 2 + (index + decor_variant) % 3
        draw.ellipse(
            ((x - r) * ss, (y - r) * ss, (x + r) * ss, (y + r) * ss),
            fill=gray,
        )

    def p(dx: float, dy: float) -> tuple[int, int]:
        x, y = _point(cx, cy, scale, dx, dy)
        return x * ss, y * ss

    lw = max(7, round(line_width * scale)) * ss
    half_gap = junction_gap / 2.0
    upper_y = -half_gap
    lower_y = half_gap
    left_y = upper_y if upper_side == "left" else lower_y
    right_y = upper_y if upper_side == "right" else lower_y
    end_y = terminal_offset

    # Branches are painted first and the central trunk last, yielding one
    # uninterrupted connected foreground component with plainly visible joins.
    left_path = [p(0, left_y), p(-28, left_y), p(-53, end_y), p(-66, end_y)]
    right_path = [p(0, right_y), p(28, right_y), p(53, end_y), p(66, end_y)]
    draw.line(left_path, fill=ink, width=lw, joint="curve")
    draw.line(right_path, fill=ink, width=lw, joint="curve")
    draw.line([p(0, -65), p(0, 70)], fill=ink, width=lw)

    # Standard-looking USB terminals: arrowhead, left circle, right square,
    # and a round base. Their locations and identities never encode the answer.
    arrow = [p(0, -84), p(-14, -64), p(14, -64)]
    draw.polygon(arrow, fill=ink)
    lr = round(10 * scale) * ss
    lx, ly = p(-72, end_y)
    draw.ellipse((lx - lr, ly - lr, lx + lr, ly + lr), fill=ink)
    rr = round(10 * scale) * ss
    rx, ry = p(72, end_y)
    draw.rectangle((rx - rr, ry - rr, rx + rr, ry + rr), fill=ink)
    br = round(10 * scale) * ss
    bx, by = p(0, 76)
    draw.ellipse((bx - br, by - br, bx + br, by + br), fill=ink)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    value = str(scene["upper_side"])
    if value not in {"left", "right"}:
        raise ValueError("upper_side must be left or right")
    return value


def margin(scene: dict) -> float:
    return float(scene["junction_gap"]) * float(scene["scale"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 36.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
