"""Pixel-only decision arm for peak_overlap_depth.

Receives the final PNG and nothing else: no latent scene, no renderer module, no
analytic gold. It segments the coloured strips by row band, recovers each span
from column occupancy, and sweeps for the deepest vertical overlap.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_SATURATION = 45
_MIN_ROWS = 8
_MIN_COLS = 40
_EXPECTED_STRIPS = 6
_MIN_CLEARANCE = 5.0


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    """Inclusive index ranges of consecutive true entries."""

    out: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _spans(image: Image.Image) -> list[tuple[float, float]] | None:
    data = np.asarray(image.convert("RGB")).astype(np.int16)
    saturation = data.max(axis=2) - data.min(axis=2)
    mark = saturation > _SATURATION
    bands = [band for band in _runs(mark.any(axis=1)) if band[1] - band[0] + 1 >= _MIN_ROWS]
    if len(bands) != _EXPECTED_STRIPS:
        return None
    spans: list[tuple[float, float]] = []
    for top, bottom in bands:
        columns = mark[top : bottom + 1].any(axis=0)
        segments = [seg for seg in _runs(columns) if seg[1] - seg[0] + 1 >= _MIN_COLS]
        if len(segments) != 1:
            return None
        left, right = segments[0]
        if int(columns.sum()) != right - left + 1:
            return None
        spans.append((float(left) - 0.5, float(right) + 0.5))
    return spans


def _profile(spans: list[tuple[float, float]]) -> list[tuple[float, float, int]]:
    cuts = sorted({value for span in spans for value in span})
    out: list[tuple[float, float, int]] = []
    for start, end in zip(cuts, cuts[1:]):
        mid = 0.5 * (start + end)
        out.append((start, end, sum(1 for left, right in spans if left < mid < right)))
    return out


def _lift_cost(spans: list[tuple[float, float]], size: int) -> float:
    best = float("inf")
    order = sorted(range(len(spans)), key=lambda i: spans[i][0])

    def walk(start: int, chosen: list[int]) -> None:
        nonlocal best
        if len(chosen) == size:
            left = max(spans[i][0] for i in chosen)
            right = min(spans[i][1] for i in chosen)
            best = min(best, max(0.0, left - right))
            return
        for index in range(start, len(order)):
            walk(index + 1, chosen + [order[index]])

    if size <= len(spans):
        walk(0, [])
    return best


def decision_from_image(image: Image.Image) -> str:
    """Deepest overlap of the coloured strips, read from the raster alone."""

    spans = _spans(image)
    if spans is None:
        return ABSTAIN
    profile = _profile(spans)
    if not profile:
        return ABSTAIN
    depth = max(value for _, _, value in profile)
    if depth < 2:
        return ABSTAIN
    runs: list[tuple[float, float]] = []
    for start, end, value in profile:
        if value != depth:
            continue
        if runs and abs(runs[-1][1] - start) < 1e-9:
            runs[-1] = (runs[-1][0], end)
        else:
            runs.append((start, end))
    if len(runs) != 1:
        return ABSTAIN
    clearance = min(runs[0][1] - runs[0][0], _lift_cost(spans, depth + 1))
    if clearance < _MIN_CLEARANCE:
        return ABSTAIN
    return str(depth)
