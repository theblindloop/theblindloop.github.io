"""Scene sampling, rasterisation and analytic gold for peak_overlap_depth.

Seven solid horizontal strips are drawn, each in its own row. The decision is
the greatest number of strips that are cut by one single vertical line, i.e. the
maximum overlap depth of the seven horizontal spans.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W = 760
H = 396
X0 = 70.0
X1 = 700.0
SPAN = X1 - X0
ROW_Y = [24 + index * 60 for index in range(6)]
STRIP_H = 24
BG = (255, 255, 255)
GUIDE = (226, 226, 226)
FRAME = (150, 150, 150)

PALETTE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
    "brown": (140, 86, 45),
}
COLORS = tuple(sorted(PALETTE))

N_STRIPS = 6
_SPREAD = {2: (0.45, 1.00), 3: (0.22, 0.55), 4: (0.11, 0.30), 5: (0.03, 0.17)}
LEN_MIN = 44.0
LEN_MAX = 96.0
MIN_MARGIN = 10.0
QUARANTINE_MARGIN = 14.0


def _segments(strips: list[dict]) -> list[tuple[float, float, int]]:
    """Constant-depth segments of the overlap profile, left to right."""

    cuts = sorted({round(value, 6) for strip in strips for value in (strip["l"], strip["r"])})
    out: list[tuple[float, float, int]] = []
    for start, end in zip(cuts, cuts[1:]):
        mid = 0.5 * (start + end)
        depth = sum(1 for strip in strips if strip["l"] < mid < strip["r"])
        out.append((start, end, depth))
    return out


def _depth(strips: list[dict]) -> int:
    return max((depth for _, _, depth in _segments(strips)), default=0)


def _plateaus(strips: list[dict], depth: int) -> list[tuple[float, float]]:
    """Merged maximal runs where the profile equals ``depth``."""

    runs: list[tuple[float, float]] = []
    for start, end, value in _segments(strips):
        if value != depth:
            continue
        if runs and abs(runs[-1][1] - start) < 1e-9:
            runs[-1] = (runs[-1][0], end)
        else:
            runs.append((start, end))
    return runs


def _lift_cost(strips: list[dict], size: int) -> float:
    """Smallest total end displacement that would make ``size`` strips share a point."""

    best = float("inf")
    order = sorted(range(len(strips)), key=lambda i: strips[i]["l"])

    def walk(start: int, chosen: list[int]) -> None:
        nonlocal best
        if len(chosen) == size:
            left = max(strips[i]["l"] for i in chosen)
            right = min(strips[i]["r"] for i in chosen)
            best = min(best, max(0.0, left - right))
            return
        for index in range(start, len(order)):
            walk(index + 1, chosen + [order[index]])

    if size <= len(strips):
        walk(0, [])
    return best


def _pair_clearance(strips: list[dict]) -> float:
    """Smallest overlap or gap between any two spans, in pixels."""

    best = float("inf")
    for i in range(len(strips)):
        for j in range(i + 1, len(strips)):
            reach = min(strips[i]["r"], strips[j]["r"]) - max(strips[i]["l"], strips[j]["l"])
            best = min(best, abs(reach))
    return best


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one integer seed."""

    rng = random.Random(seed)
    target = rng.choice([2, 3, 4, 5])
    lengths = [round(rng.uniform(LEN_MIN, LEN_MAX), 3) for _ in range(N_STRIPS)]
    for _ in range(20000):
        low_s, high_s = _SPREAD[target]
        spread = rng.uniform(low_s, high_s) * SPAN
        centre = rng.uniform(X0 + 0.35 * SPAN, X1 - 0.35 * SPAN)
        strips: list[dict] = []
        for length in lengths:
            low = X0 + 0.5 * length
            high = X1 - 0.5 * length
            mid = min(max(centre + rng.uniform(-spread, spread), low), high)
            strips.append({"l": round(mid - 0.5 * length, 3), "r": round(mid + 0.5 * length, 3)})
        depth = _depth(strips)
        if depth != target:
            continue
        plateaus = _plateaus(strips, depth)
        if len(plateaus) != 1:
            continue
        width = plateaus[0][1] - plateaus[0][0]
        lift = _lift_cost(strips, depth + 1)
        if min(width, lift, _pair_clearance(strips)) < MIN_MARGIN:
            continue
        rows = list(range(N_STRIPS))
        rng.shuffle(rows)
        colors = list(COLORS)
        rng.shuffle(colors)
        records = [
            {"color": colors[i], "row": rows[i], "l": strips[i]["l"], "r": strips[i]["r"]}
            for i in range(N_STRIPS)
        ]
        rng.shuffle(records)
        guides = sorted(round(rng.uniform(X0 + 30, X1 - 30), 2) for _ in range(2))
        return {"strips": records, "guides": guides}
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Rasterise a latent scene; the strip list is drawn as an unordered set."""

    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    for guide in scene["guides"]:
        x = int(round(float(guide)))
        draw.line([(x, 8), (x, H - 8)], fill=GUIDE, width=2)
    draw.line([(int(X0) - 24, H - 6), (int(X1) + 24, H - 6)], fill=FRAME, width=2)
    for strip in sorted(scene["strips"], key=lambda item: item["row"]):
        top = ROW_Y[int(strip["row"])]
        draw.rectangle(
            [int(round(strip["l"])), top, int(round(strip["r"])), top + STRIP_H],
            fill=PALETTE[strip["color"]],
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Greatest number of strips crossed by one vertical line."""

    return str(_depth(scene["strips"]))


def margin(scene: dict) -> float:
    """Smallest horizontal displacement in pixels that would change the depth."""

    strips = scene["strips"]
    depth = _depth(strips)
    plateaus = _plateaus(strips, depth)
    shrink = min(end - start for start, end in plateaus)
    return float(round(min(shrink, _lift_cost(strips, depth + 1), _pair_clearance(strips)), 6))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes excluded from headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabelings that must preserve the gold."""

    strips = scene["strips"]
    return [
        ("reversed_strip_list", {"strips": list(reversed(strips)), "guides": list(scene["guides"])}),
        ("rotated_strip_list", {"strips": strips[2:] + strips[:2], "guides": list(scene["guides"])}),
        ("reversed_guides", {"strips": list(strips), "guides": list(reversed(scene["guides"]))}),
    ]
