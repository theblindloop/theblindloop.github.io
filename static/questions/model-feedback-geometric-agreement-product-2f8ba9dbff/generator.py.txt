"""Fresh renderer and analytic specification for a multiplicative agreement board.

Three horizontal ledgers show one bar for each of four repeated colors.  A
candidate score is the product, over colors, of the smaller height divided by
the larger height in the candidate/reference pair.  The product is used
directly because taking a fourth root would not change the comparison.
"""

from __future__ import annotations

import itertools
import random
from fractions import Fraction
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "geometric_agreement_product-renderer-0.1.0"

WIDTH = 1280
HEIGHT = 820
PANEL_X = (30, 440, 850)
PANEL_WIDTH = 400
PANEL_TOP = 145
PANEL_BOTTOM = 680
TRACK_LEFT_OFFSET = 102
UNIT = 38
BAR_HEIGHT = 42
ROW_CENTERS = (248, 338, 428, 518)
RULER_Y = 612
COLOR_NAMES = ("red", "blue", "green", "violet")
COLOR_SHORT = {"red": "R", "blue": "B", "green": "G", "violet": "V"}
COLOR_LABELS = {"red": "RED", "blue": "BLUE", "green": "GREEN", "violet": "VIOLET"}
BAR_COLORS = {
    "red": (210, 70, 67),
    "blue": (52, 101, 184),
    "green": (47, 145, 91),
    "violet": (145, 78, 174),
}

BACKGROUND = (247, 249, 247)
PANEL_FILL = (255, 255, 255)
INK = (38, 49, 58)
SUBTLE = (111, 128, 137)
GUIDE = (216, 225, 226)
ACCENT = (225, 174, 55)
LEVEL_CHOICES = tuple(range(1, 8))


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
        text,
        font=font,
        fill=fill,
    )


def _validate_scene(scene: dict) -> tuple[list[int], list[int], list[int]]:
    reference = [int(value) for value in scene["reference_values"]]
    candidate_a = [int(value) for value in scene["candidate_a_values"]]
    candidate_b = [int(value) for value in scene["candidate_b_values"]]
    values = (reference, candidate_a, candidate_b)
    if any(len(item) != len(COLOR_NAMES) for item in values):
        raise ValueError("the ledger needs four values in each panel")
    if any(value not in LEVEL_CHOICES for item in values for value in item):
        raise ValueError("bar heights must be integer ruler levels from 1 through 7")
    if any(len(set(item)) != len(COLOR_NAMES) for item in values):
        raise ValueError("each panel needs four distinct visible levels")
    if any(sorted(item) != sorted(reference) for item in (candidate_a, candidate_b)):
        raise ValueError("all panels must use one shared height multiset")
    return reference, candidate_a, candidate_b


def _coverage(reference: Iterable[int], candidate: Iterable[int]) -> list[Fraction]:
    return [
        Fraction(min(int(value), int(target)), max(int(value), int(target)))
        for value, target in zip(candidate, reference, strict=True)
    ]


def _product(reference: Iterable[int], candidate: Iterable[int]) -> Fraction:
    answer = Fraction(1, 1)
    for ratio in _coverage(reference, candidate):
        answer *= ratio
    return answer


def _bottleneck(reference: Iterable[int], candidate: Iterable[int]) -> Fraction:
    return min(_coverage(reference, candidate))


def _l1(reference: Iterable[int], candidate: Iterable[int]) -> int:
    return sum(abs(int(value) - int(target)) for value, target in zip(candidate, reference, strict=True))


def _pairs_for_target(
    reference: tuple[int, ...], target: str, min_gap: Fraction = Fraction(1, 200)
) -> list[tuple[tuple[int, ...], tuple[int, ...]]]:
    """Find pairs where familiar additive and bottleneck shortcuts disagree."""

    permutations = sorted(set(itertools.permutations(reference)))
    records = []
    for values in permutations:
        records.append(
            (
                values,
                _product(reference, values),
                _bottleneck(reference, values),
                _l1(reference, values),
            )
        )
    pairs: list[tuple[tuple[int, ...], tuple[int, ...]]] = []
    for left in records:
        for right in records:
            if left[0] == right[0]:
                continue
            product_delta = left[1] - right[1]
            if target == "A":
                valid = (
                    product_delta >= min_gap
                    and left[2] < right[2]
                    and left[3] < right[3]
                )
            elif target == "B":
                valid = (
                    product_delta <= -min_gap
                    and left[2] > right[2]
                    and left[3] > right[3]
                )
            elif target == "tie":
                valid = product_delta == 0
            else:
                raise ValueError(target)
            if valid:
                pairs.append((left[0], right[0]))
    return pairs


def sample_scene(seed: int) -> dict:
    """Sample a deterministic product board with both strict signs and ties."""

    rng = random.Random(int(seed))
    levels = tuple(sorted(rng.sample(list(LEVEL_CHOICES), len(COLOR_NAMES))))
    reference = tuple(rng.sample(list(levels), len(levels)))
    if int(seed) % 19 == 0:
        target = "tie"
    else:
        target = "A" if int(seed) % 2 == 0 else "B"
    pairs = _pairs_for_target(reference, target)
    if not pairs:
        raise RuntimeError(f"no {target} product pair exists for levels {levels}")
    candidate_a, candidate_b = rng.choice(pairs)
    return {
        "reference_values": list(reference),
        "candidate_a_values": list(candidate_a),
        "candidate_b_values": list(candidate_b),
    }


def render(scene: dict) -> Image.Image:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    body_font = _font(17)
    panel_font = _font(21)
    label_font = _font(15)
    ruler_font = _font(14)

    draw.text((40, 22), "MULTIPLICATIVE AGREEMENT LEDGER", font=title_font, fill=INK)
    draw.text(
        (40, 63),
        "Match each color to REFERENCE; every bar ends on the same integer ruler.",
        font=body_font,
        fill=INK,
    )
    draw.text(
        (40, 93),
        "Each candidate score is the product of its four smaller-height / larger-height ratios.",
        font=label_font,
        fill=SUBTLE,
    )

    panel_values = (reference, candidate_a, candidate_b)
    panel_titles = ("REFERENCE", "CANDIDATE A", "CANDIDATE B")
    for x0, values, title in zip(PANEL_X, panel_values, panel_titles, strict=True):
        _centered_text(draw, (x0 + PANEL_WIDTH / 2, 125), title, panel_font, INK)
        draw.rounded_rectangle(
            (x0, PANEL_TOP, x0 + PANEL_WIDTH, PANEL_BOTTOM),
            radius=8,
            fill=PANEL_FILL,
            outline=INK,
            width=2,
        )
        track_left = x0 + TRACK_LEFT_OFFSET
        track_right = track_left + 7 * UNIT
        for row_index, (color_name, value, center_y) in enumerate(
            zip(COLOR_NAMES, values, ROW_CENTERS, strict=True)
        ):
            draw.text((x0 + 14, center_y - 12), COLOR_LABELS[color_name], font=label_font, fill=INK)
            draw.line((track_left, center_y, track_right, center_y), fill=GUIDE, width=2)
            top = center_y - BAR_HEIGHT // 2
            right = track_left + int(value) * UNIT
            draw.rectangle((track_left, top, right, center_y + BAR_HEIGHT // 2), fill=BAR_COLORS[color_name])
            _centered_text(draw, (track_left - 20, center_y), COLOR_SHORT[color_name], label_font, SUBTLE)
            if row_index == 0:
                draw.text((track_left, center_y - 31), "height", font=ruler_font, fill=SUBTLE)

        draw.line((track_left, RULER_Y, track_right, RULER_Y), fill=SUBTLE, width=2)
        for level in range(0, 8):
            x = track_left + level * UNIT
            draw.line((x, RULER_Y - 7, x, RULER_Y + 7), fill=SUBTLE, width=2)
            _centered_text(draw, (x, RULER_Y + 22), str(level), ruler_font, INK)
        draw.text((x0 + 15, 646), "four colored measurements", font=ruler_font, fill=SUBTLE)

    draw.rounded_rectangle((40, 718, 1240, 782), radius=8, fill=(255, 252, 237), outline=ACCENT, width=1)
    draw.text(
        (58, 731),
        "For each candidate: multiply the four color agreements, then compare the two products.",
        font=body_font,
        fill=INK,
    )
    draw.text(
        (58, 756),
        "A fourth root would preserve the same ordering; equal products are a tie.",
        font=label_font,
        fill=SUBTLE,
    )
    return image


def analytic_gold(scene: dict) -> str:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    score_a = _product(reference, candidate_a)
    score_b = _product(reference, candidate_b)
    if score_a > score_b:
        return "A"
    if score_b > score_a:
        return "B"
    return "tie"


def margin(scene: dict) -> float:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    return float(_product(reference, candidate_a) - _product(reference, candidate_b))


def is_quarantined(scene: dict) -> bool:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    return _product(reference, candidate_a) == _product(reference, candidate_b)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:

    # unrendered scene parameter with an alternate pixel-identical transform.
