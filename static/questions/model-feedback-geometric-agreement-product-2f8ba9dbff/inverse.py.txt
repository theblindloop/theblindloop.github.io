"""Independent image-only inverse for the multiplicative agreement ledger."""

from __future__ import annotations

from collections import deque
from fractions import Fraction

import numpy as np
from PIL import Image

ORACLE_VERSION = "geometric_agreement_product-oracle-0.1.0"

_WIDTH = 1280
_HEIGHT = 820
_PANEL_X = (30, 440, 850)
_PANEL_WIDTH = 400
_PANEL_TOP = 145
_PANEL_BOTTOM = 680
_TRACK_LEFT_OFFSET = 102
_UNIT = 38.0
_BASE_COLORS = {
    "red": np.array((210, 70, 67), dtype=np.uint8),
    "blue": np.array((52, 101, 184), dtype=np.uint8),
    "green": np.array((47, 145, 91), dtype=np.uint8),
    "violet": np.array((145, 78, 174), dtype=np.uint8),
}
_COLORS = tuple(_BASE_COLORS)
_MIN_BAR_PIXELS = 1000


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected components without sharing renderer geometry code."""

    ys, xs = np.nonzero(mask)
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _panel_values(arr: np.ndarray, panel_x: int) -> dict[str, int]:
    values: dict[str, int] = {}
    x_lo = panel_x + 1
    x_hi = panel_x + _PANEL_WIDTH - 1
    for color_name, color in _BASE_COLORS.items():
        mask = np.all(arr == color, axis=2)
        mask[:_PANEL_TOP, :] = False
        mask[_PANEL_BOTTOM + 1 :, :] = False
        mask[:, :x_lo] = False
        mask[:, x_hi + 1 :] = False
        components = [part for part in _components(mask) if len(part) >= _MIN_BAR_PIXELS]
        if len(components) != 1:
            raise ValueError(f"expected one {color_name} bar in panel, found {len(components)}")
        component = components[0]
        left = int(component[:, 1].min())
        right = int(component[:, 1].max())
        top = int(component[:, 0].min())
        bottom = int(component[:, 0].max())
        expected_left = panel_x + _TRACK_LEFT_OFFSET
        if left != expected_left or bottom - top < 35:
            raise ValueError(f"{color_name} bar has an unexpected raster footprint")
        raw_level = (right - left) / _UNIT
        level = int(round(raw_level))
        if level < 1 or level > 7 or abs(raw_level - level) > 0.04:
            raise ValueError(f"{color_name} bar does not end on an integer ruler mark")
        values[color_name] = level
    if len(set(values.values())) != 4:
        raise ValueError("panel does not contain four distinct levels")
    return values


def _score(reference: dict[str, int], candidate: dict[str, int]) -> Fraction:
    value = Fraction(1, 1)
    for color_name in _COLORS:
        value *= Fraction(
            min(candidate[color_name], reference[color_name]),
            max(candidate[color_name], reference[color_name]),
        )
    return value


def decision_from_image(image: Image.Image) -> str:
    """Return ``A``, ``B``, or ``tie`` using only the final raster."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (_HEIGHT, _WIDTH, 3):
        raise ValueError(f"unexpected image shape {arr.shape}")
    reference = _panel_values(arr, _PANEL_X[0])
    candidate_a = _panel_values(arr, _PANEL_X[1])
    candidate_b = _panel_values(arr, _PANEL_X[2])
    common = sorted(reference.values())
    if sorted(candidate_a.values()) != common or sorted(candidate_b.values()) != common:
        raise ValueError("panels do not share one height multiset")
    score_a = _score(reference, candidate_a)
    score_b = _score(reference, candidate_b)
    if score_a > score_b:
        return "A"
    if score_b > score_a:
        return "B"
    return "tie"
