"""Independent image-only inverse for the global stripe-gap contour."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _ink_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # All permitted backgrounds are pale and nearly neutral; colored evidence has
    # at least one substantially darker channel. This derives only from pixels.
    return arr.min(axis=2) < 225


def _band_rows(mask: np.ndarray) -> list[int]:
    active = mask.sum(axis=1) >= 65
    groups: list[list[int]] = []
    for y in np.flatnonzero(active).tolist():
        if not groups or y > groups[-1][-1] + 1:
            groups.append([y])
        else:
            groups[-1].append(y)
    return [group[len(group) // 2] for group in groups if len(group) >= 2]


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    points = np.flatnonzero(values)
    if len(points) == 0:
        return []
    out = []
    start = previous = int(points[0])
    for point in points[1:]:
        point = int(point)
        if point != previous + 1:
            out.append((start, previous))
            start = point
        previous = point
    out.append((start, previous))
    return out


def _extract_width_profile(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    mask = _ink_mask(image)
    ys = _band_rows(mask)
    if len(ys) < 16:
        raise ValueError("insufficient colored bands for a global contour")
    candidates = []
    for y in ys:
        runs = _runs(mask[y])
        for left, right in zip(runs, runs[1:]):
            gap_left = left[1] + 1
            gap_right = right[0] - 1
            gap_width = gap_right - gap_left + 1
            center = (gap_left + gap_right) / 2.0
            if gap_width >= 26 and image.width * 0.22 < center < image.width * 0.78:
                candidates.append((gap_width, center))
    if len(candidates) < 5:
        raise ValueError("no coherent central negative-space opening")
    # Large gaps are overwhelmingly central; taking the median rejects narrow
    # independently scattered arm breaks.
    candidates.sort(reverse=True)
    core = candidates[: max(5, len(candidates) * 3 // 4)]
    center_x = float(np.median([item[1] for item in core]))
    if not image.width * 0.30 < center_x < image.width * 0.70:
        raise ValueError("central opening is not reliably localized")
    widths = []
    x_probe = int(round(center_x))
    for y in ys:
        row = mask[y]
        if row[x_probe]:
            widths.append(0.0)
            continue
        left = x_probe
        while left > 0 and not row[left - 1]:
            left -= 1
        right = x_probe
        while right + 1 < row.size and not row[right + 1]:
            right += 1
        width = float(right - left + 1)
        # An unbounded pale row would be background, not an enclosed band gap.
        if left == 0 or right == row.size - 1:
            raise ValueError("opening is not bounded by colored endpoints")
        widths.append(width)
    return np.asarray(ys, dtype=np.float64), np.asarray(widths, dtype=np.float64)


def _best_error(kind: str, ys: np.ndarray, widths: np.ndarray) -> float:
    best = float("inf")
    y_lo, y_hi = int(ys.min()), int(ys.max())
    for cy in range(y_lo + 25, y_hi - 24):
        dy = np.abs(ys - float(cy))
        for radius in range(45, 101):
            t = dy / float(radius)
            if kind == "circle":
                predicted = 2.0 * radius * np.sqrt(np.maximum(0.0, 1.0 - t * t))
            else:
                predicted = 2.0 * radius * np.maximum(0.0, 1.0 - t)
            error = float(np.mean(np.abs(predicted - widths)))
            if error < best:
                best = error
    return best


def decision_from_image(image: Image.Image) -> str:
    ys, widths = _extract_width_profile(image)
    circle_error = _best_error("circle", ys, widths)
    diamond_error = _best_error("diamond", ys, widths)
    ordered = sorted(((circle_error, "circle"), (diamond_error, "diamond")))
    if not math.isfinite(ordered[0][0]) or ordered[1][0] - ordered[0][0] < 3.0:
        raise ValueError("ambiguous contour reconstruction")
    return ordered[0][1]
