"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 360, 300
PALETTES = (
    ((29, 91, 121), (211, 83, 69), (227, 163, 46), (68, 126, 91)),
    ((76, 70, 142), (25, 137, 145), (211, 91, 62), (190, 139, 36)),
    ((25, 105, 137), (171, 67, 106), (77, 132, 64), (214, 137, 45)),
)
BACKGROUNDS = ((248, 247, 241), (245, 247, 246), (249, 246, 242))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    profile = "circle" if int(seed) % 2 == 0 else "diamond"
    center_x = rng.randint(166, 194)
    center_y = rng.randint(139, 161)
    radius = rng.randint(61, 79)
    band_height = rng.choice((4, 5, 6))
    pitch = rng.choice((9, 10, 11))
    phase = rng.randint(0, pitch - 1)
    palette = list(rng.choice(PALETTES))
    rng.shuffle(palette)
    background = rng.choice(BACKGROUNDS)
    bands = []
    index = 0
    for y in range(25 + phase, HEIGHT - 24, pitch):
        arm = rng.randint(58, 70)
        jitter = rng.choice((-1, 0, 0, 0, 1))
        cuts = []
        if rng.random() < 0.42:
            side = rng.choice((-1, 1))
            offset = rng.randint(28, max(29, arm - 15))
            cuts.append([side, offset, rng.randint(5, 10)])
        bands.append(
            {
                "y": y,
                "color": index % len(palette),
                "arm": arm,
                "edge_jitter": jitter,
                "cuts": cuts,
            }
        )
        index += 1
    rng.shuffle(bands)
    return {
        "profile": profile,
        "canvas": [WIDTH, HEIGHT],
        "center_x": center_x,
        "center_y": center_y,
        "radius": radius,
        "band_height": band_height,
        "palette": [list(color) for color in palette],
        "background": list(background),
        "bands": bands,
    }


def _half_width(profile: str, dy: float, radius: float) -> float:
    t = abs(dy) / radius
    if t >= 1.0:
        return 0.0
    if profile == "circle":
        return radius * math.sqrt(max(0.0, 1.0 - t * t))
    if profile == "diamond":
        return radius * (1.0 - t)
    raise ValueError(f"unknown profile {profile!r}")


def _render_bands(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    background = tuple(int(value) for value in scene["background"])
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    radius = float(scene["radius"])
    band_height = int(scene["band_height"])
    profile = str(scene["profile"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for band in sorted(scene["bands"], key=lambda item: int(item["y"])):
        y = int(band["y"])
        half = int(round(_half_width(profile, y - cy, radius)))
        if half > 0:
            half = max(2, half + int(band["edge_jitter"]))
        arm = int(band["arm"])
        color = palette[int(band["color"]) % len(palette)]
        top = y - band_height // 2
        bottom = top + band_height - 1
        left_inner = cx - half
        right_inner = cx + half
        draw.rectangle((left_inner - arm, top, left_inner - 1, bottom), fill=color)
        draw.rectangle((right_inner, top, right_inner + arm - 1, bottom), fill=color)
        for side, offset, cut_width in band["cuts"]:
            if int(side) < 0:
                cut_center = left_inner - int(offset)
            else:
                cut_center = right_inner + int(offset)
            cut_left = cut_center - int(cut_width) // 2
            draw.rectangle((cut_left, top, cut_left + int(cut_width) - 1, bottom), fill=background)
    return image


def render(scene: dict) -> Image.Image:
    return _render_bands(scene)


def analytic_gold(scene: dict) -> str:
    return str(scene["profile"])


def margin(scene: dict) -> float:
    radius = float(scene["radius"])
    cy = float(scene["center_y"])
    profile = str(scene["profile"])
    other = "diamond" if profile == "circle" else "circle"
    diffs = []
    for band in scene["bands"]:
        y = float(band["y"])
        true_width = 2.0 * _half_width(profile, y - cy, radius)
        other_width = 2.0 * _half_width(other, y - cy, radius)
        diffs.append(abs(true_width - other_width))
    return round(sum(diffs) / len(diffs), 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["bands"] = list(reversed(twin["bands"]))
    return [("reverse_disjoint_band_records", twin)]
