"""Independent PNG-only inverse arm for counterfactual provenance mosaics."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 780
INK, OFF, RED, BLUE = (np.array(color, dtype=np.float64) for color in
                       ((35, 40, 47), (255, 255, 255), (205, 54, 54), (35, 105, 210)))
GATE_COLORS = {
    "and": np.array((48, 112, 190), dtype=np.float64),
    "or": np.array((224, 148, 39), dtype=np.float64),
    "xor": np.array((132, 76, 183), dtype=np.float64),
}
TOKEN_COLORS = tuple(np.array(color, dtype=np.float64) for color in (
    (0, 150, 180), (0, 140, 70), (225, 100, 0), (190, 45, 135),
))
ROW_Y, TOP_X0, CELL_STEP = (130, 225, 320, 415, 510), 84, 60
CARD_BOXES = ((35, 625, 245, 755), (260, 625, 470, 755),
              (485, 625, 695, 755), (710, 625, 920, 755))
CARD_PATTERN_X = (34, 81, 128, 175)


class OracleAbstention(ValueError):
    pass


def _patch(array: np.ndarray, x: int, y: int, radius: int = 4) -> np.ndarray:
    pixels = array[y - radius:y + radius + 1, x - radius:x + radius + 1]
    return np.median(pixels.reshape(-1, 3), axis=0)


def _binary(array: np.ndarray, x: int, y: int) -> int:
    rgb = _patch(array, x, y)
    on_distance = float(np.linalg.norm(rgb - INK))
    off_distance = float(np.linalg.norm(rgb - OFF))
    if min(on_distance, off_distance) > 35 or abs(on_distance - off_distance) < 60:
        raise OracleAbstention(f"unclear binary cell at {(x, y)}: {rgb.tolist()}")
    return int(on_distance < off_distance)


def _gate(array: np.ndarray, y: int) -> str:
    rgb = _patch(array, 662, y, 5)
    distances = {kind: float(np.linalg.norm(rgb - color)) for kind, color in GATE_COLORS.items()}
    kind = min(distances, key=distances.get)  # type: ignore[arg-type]
    if distances[kind] > 45:
        raise OracleAbstention(f"unclear row rule at y={y}: {rgb.tolist()}")
    return kind


def _annulus(array: np.ndarray, x: int, y: int) -> np.ndarray:
    crop = array[y - 31:y + 32, x - 31:x + 32].astype(np.float64)
    yy, xx = np.ogrid[-31:32, -31:32]
    mask = ((xx * xx + yy * yy) >= 21 * 21) & ((xx * xx + yy * yy) <= 30 * 30)
    return crop[mask]


def _color_count(pixels: np.ndarray, color: np.ndarray, tolerance: float = 58.0) -> int:
    return int(np.count_nonzero(np.linalg.norm(pixels - color, axis=1) < tolerance))


def _source_color(array: np.ndarray, x: int, y: int) -> int | None:
    pixels = _annulus(array, x, y)
    red_count = _color_count(pixels, RED)
    blue_count = _color_count(pixels, BLUE)
    if max(red_count, blue_count) < 100:
        return None
    if abs(red_count - blue_count) < 45:
        raise OracleAbstention(f"ambiguous source ring at {(x, y)}")
    return int(blue_count > red_count)


def _token_ring(array: np.ndarray, x: int, y: int) -> int | None:
    pixels = _annulus(array, x, y)
    counts = [_color_count(pixels, color) for color in TOKEN_COLORS]
    best = max(range(len(counts)), key=counts.__getitem__)
    ordered = sorted(counts, reverse=True)
    if counts[best] < 100:
        return None
    if ordered[0] - ordered[1] < 45:
        raise OracleAbstention(f"ambiguous mediator ring at {(x, y)}: {counts}")
    return best


def _card_source(array: np.ndarray, x: int, y: int) -> int:
    crop = array[y - 19:y + 20, x - 15:x + 16].astype(np.float64).reshape(-1, 3)
    red_count = _color_count(crop, RED, 52.0)
    blue_count = _color_count(crop, BLUE, 52.0)
    if max(red_count, blue_count) < 550 or abs(red_count - blue_count) < 300:
        raise OracleAbstention(f"unclear card source at {(x, y)}: {(red_count, blue_count)}")
    return int(blue_count > red_count)


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise OracleAbstention("unknown gate")


def _next_row(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[index], row[index + 1]) for index in range(len(row) - 1)]


def _run(top: list[int], gates: list[str]) -> list[list[int]]:
    rows = [list(top)]
    for kind in gates:
        rows.append(_next_row(rows[-1], kind))
    return rows


def _flip(top: list[int], *marked: int) -> list[int]:
    changed = list(top)
    for index in marked:
        changed[index] ^= 1
    return changed


def _hybrid(top: list[int], gates: list[str], red_rows: list[list[int]],
            blue_rows: list[list[int]], mediators: list[tuple[int, int]],
            card: tuple[int, ...]) -> list[list[int]]:
    rows = [list(top)]
    for row_index, kind in enumerate(gates, 1):
        row = _next_row(rows[-1], kind)
        for token, (selected_row, selected_cell) in enumerate(mediators):
            if selected_row == row_index:
                source = blue_rows if card[token] else red_rows
                row[selected_cell] = source[row_index][selected_cell]
        rows.append(row)
    return rows


def _distance(left: list[int], right: list[int]) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    array = np.asarray(raster)
    visible_rows: list[list[int]] = []
    source_marks: list[int | None] = []
    mediator_by_token: dict[int, tuple[int, int]] = {}
    for row_index, y in enumerate(ROW_Y):
        x0 = TOP_X0 + row_index * (CELL_STEP // 2)
        row = []
        for cell_index in range(9 - row_index):
            x = x0 + cell_index * CELL_STEP
            row.append(_binary(array, x, y))
            if row_index == 0:
                source_marks.append(_source_color(array, x, y))
            elif row_index <= 3:
                token = _token_ring(array, x, y)
                if token is not None:
                    if token in mediator_by_token:
                        raise OracleAbstention(f"duplicate mediator token {token}")
                    mediator_by_token[token] = (row_index, cell_index)
        visible_rows.append(row)
    red_positions = [index for index, value in enumerate(source_marks) if value == 0]
    blue_positions = [index for index, value in enumerate(source_marks) if value == 1]
    if len(red_positions) != 1 or len(blue_positions) != 1:
        raise OracleAbstention(f"found source rings red={red_positions}, blue={blue_positions}")
    red_marked, blue_marked = red_positions[0], blue_positions[0]
    if not (1 <= red_marked <= 7 and 1 <= blue_marked <= 7):
        raise OracleAbstention("a source ring is not internal")
    if abs(red_marked - blue_marked) > 3:
        raise OracleAbstention("source rings are too widely separated")
    if set(mediator_by_token) != set(range(4)):
        raise OracleAbstention(f"expected mediator tokens A-D, found {sorted(mediator_by_token)}")
    mediators = [mediator_by_token[index] for index in range(4)]
    if sum(row == 2 for row, _ in mediators) != 2 or any(row not in (2, 3) for row, _ in mediators):
        raise OracleAbstention("mediators do not span rows two and three equally")
    gates = [_gate(array, ROW_Y[index]) for index in range(1, 5)]
    if len(set(gates)) < 3:
        raise OracleAbstention("fewer than three distinct rules")
    factual = _run(visible_rows[0], gates)
    if factual != visible_rows:
        raise OracleAbstention("visible factual lattice contradicts its row rules")
    red_rows = _run(_flip(visible_rows[0], red_marked), gates)
    blue_rows = _run(_flip(visible_rows[0], blue_marked), gates)
    both_rows = _run(_flip(visible_rows[0], red_marked, blue_marked), gates)
    if any(_distance(rows[-1], both_rows[-1]) < 2 for rows in (factual, red_rows, blue_rows)):
        raise OracleAbstention("joint target is insufficiently separated")
    if any(red_rows[row][cell] == blue_rows[row][cell] for row, cell in mediators):
        raise OracleAbstention("a mediator does not distinguish source provenance")
    cards: list[tuple[int, ...]] = []
    for x0, y0, _x1, _y1 in CARD_BOXES:
        card = tuple(_card_source(array, x0 + offset, y0 + 90) for offset in CARD_PATTERN_X)
        if sum(card) != 2:
            raise OracleAbstention(f"card is not provenance-balanced: {card}")
        cards.append(card)
    if len(set(cards)) != 4:
        raise OracleAbstention("duplicate provenance card")
    if any(len({card[column] for card in cards}) != 2 for column in range(4)):
        raise OracleAbstention("a mediator source is constant across cards")
    outcomes = [_hybrid(visible_rows[0], gates, red_rows, blue_rows, mediators, card)[-1]
                for card in cards]
    matches = [index for index, outcome in enumerate(outcomes) if outcome == both_rows[-1]]
    if len(matches) != 1:
        raise OracleAbstention(f"expected one matching card, recovered {matches}")
    wrong = [outcome for index, outcome in enumerate(outcomes) if index != matches[0]]
    if len({tuple(outcome) for outcome in wrong}) != 3:
        raise OracleAbstention("distractor outcomes are not distinct")
    if any(_distance(outcome, both_rows[-1]) < 2 for outcome in wrong):
        raise OracleAbstention("a distractor is too close to the joint target")
    return f"card-{matches[0] + 1}"
