"""Conjunctive dual-attribute binding correspondence world.

A cross-panel correspondence task whose identity key is a *pair* of independent
bound attributes that must co-vary together, so that no single attribute can
locate the object.

Top (reference) panel: one object, a solid fill disk surrounded by a thin
marker ring. The object therefore exposes two bound attributes simultaneously
(fill color and marker color).

Bottom (candidate) panel: a row of three candidate objects, each a fill disk
plus marker ring plus an outermost halo ring. Each candidate carries a distinct
halo color that appears ONLY in the candidate panel.

Three candidate archetypes are present in every scene:
  * the TRUE match   -- fill == ref fill  AND  marker == ref marker;
  * a FILL decoy     -- fill == ref fill,  marker != ref marker;
  * a MARKER decoy   -- marker == ref marker, fill != ref fill.

Because a fill-only matcher sees two candidates with the reference fill and a
marker-only matcher sees two candidates with the reference marker, the object
is uniquely recoverable only by demanding BOTH attributes match. The question
asks for the halo bound to that uniquely matched object.

Latent scene fields z (every field below is rendered; nothing is hidden):

    ref_fill     : reference fill color        (FILL_PALETTE)
    ref_marker   : reference marker color      (MARKER_PALETTE)
    cand_fills   : fill color per candidate slot
    cand_markers : marker color per candidate slot
    cand_halos   : halo color per candidate slot (panel-only answer attribute)
    cand_centers : [x, y] of each candidate slot
    width, height: canvas pixels

Analytic gold is derived from these rendered fields: locate the slot whose fill
and marker both equal the reference's, then report that slot's halo.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "correspondence_dual_binding-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)

# Three pairwise-disjoint palettes so the inverse arm can classify pixels by
# nearest color without ambiguity. Colors are deliberately far apart.
FILL_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (255, 0, 0),
    "green": (0, 200, 0),
    "blue": (0, 0, 255),
    "orange": (255, 140, 0),
}

MARKER_PALETTE: dict[str, tuple[int, int, int]] = {
    "cyan": (0, 220, 220),
    "magenta": (255, 0, 255),
    "yellow": (255, 235, 0),
    "purple": (200, 40, 220),
}

HALO_PALETTE: dict[str, tuple[int, int, int]] = {
    "navy": (20, 20, 120),
    "teal": (0, 110, 110),
    "gold": (200, 165, 0),
    "maroon": (150, 40, 90),
}

# Constant public geometry (identical ink budget across all scenes).
WIDTH = 760
HEIGHT = 470
DIVIDER_Y = 240
DIVIDER_THICK = 3
FILL_R = 26
MARK_R = 44
HALO_R = 62
REF_CENTER = (380, 118)
CAND_Y = 366
_SLOTS = [168, 380, 592]

_MARGIN_FLOOR = 60.0


def _color_dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.dist(a, b)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    fills = list(FILL_PALETTE)
    markers = list(MARKER_PALETTE)

    ref_fill = rng.choice(fills)
    ref_marker = rng.choice(markers)
    fill_decoy_marker = rng.choice([m for m in markers if m != ref_marker])
    marker_decoy_fill = rng.choice([f for f in fills if f != ref_fill])
    h0, h1, h2 = rng.sample(list(HALO_PALETTE), 3)

    objects = [
        (ref_fill, ref_marker, h0),
        (ref_fill, fill_decoy_marker, h1),
        (marker_decoy_fill, ref_marker, h2),
    ]
    rng.shuffle(objects)

    return {
        "ref_fill": ref_fill,
        "ref_marker": ref_marker,
        "cand_fills": [o[0] for o in objects],
        "cand_markers": [o[1] for o in objects],
        "cand_halos": [o[2] for o in objects],
        "cand_centers": [[_SLOTS[i], CAND_Y] for i in range(3)],
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_object(draw: ImageDraw.ImageDraw, cx: int, cy: int, fill, marker, halo=None):
    if halo is not None:
        draw.ellipse([cx - HALO_R, cy - HALO_R, cx + HALO_R, cy + HALO_R], fill=HALO_PALETTE[halo])
    draw.ellipse([cx - MARK_R, cy - MARK_R, cx + MARK_R, cy + MARK_R], fill=MARKER_PALETTE[marker])
    draw.ellipse([cx - FILL_R, cy - FILL_R, cx + FILL_R, cy + FILL_R], fill=FILL_PALETTE[fill])


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    rcx, rcy = REF_CENTER
    _draw_object(draw, rcx, rcy, scene["ref_fill"], scene["ref_marker"])

    for (fx, fy), fill, marker, halo in zip(
        scene["cand_centers"], scene["cand_fills"], scene["cand_markers"], scene["cand_halos"], strict=True
    ):
        _draw_object(draw, int(fx), int(fy), fill, marker, halo)

    draw.rectangle([0, DIVIDER_Y, w, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _matched_index(scene: dict) -> int:
    for i, (f, m) in enumerate(zip(scene["cand_fills"], scene["cand_markers"], strict=True)):
        if f == scene["ref_fill"] and m == scene["ref_marker"]:
            return i
    raise ValueError("no dual-attribute match")


def analytic_gold(scene: dict) -> str:
    """Decision: the halo bound to the object matching BOTH fill and marker."""
    idx = _matched_index(scene)
    return scene["cand_halos"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: the nearest confusable alternative is the decoy most
    nearly satisfying both binding criteria."""
    rf = FILL_PALETTE[scene["ref_fill"]]
    rm = MARKER_PALETTE[scene["ref_marker"]]
    residuals = []
    for i, (f, m) in enumerate(zip(scene["cand_fills"], scene["cand_markers"], strict=True)):
        if f == scene["ref_fill"] and m == scene["ref_marker"]:
            continue
        df = _color_dist(FILL_PALETTE[f], rf)
        dm = _color_dist(MARKER_PALETTE[m], rm)
        if df == 0.0:
            residual = dm
        elif dm == 0.0:
            residual = df
        else:
            residual = min(df, dm)
        residuals.append(residual)
    return float(min(residuals))


def is_quarantined(scene: dict) -> bool:
    """Ambiguity threshold; never triggered because palettes keep the two decoys
    far from also satisfying the missing criterion."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing scene field is rendered verbatim,
    so the raster uniquely determines the dual-attribute match and its bound
    halo; no pixel-identical transform with unchanged gold is declared."""
    return []
