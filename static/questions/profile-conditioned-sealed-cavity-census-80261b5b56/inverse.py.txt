"""Pixel-only inverse arm for the sealed-cavity census.

Receives a rendered RGB image and nothing else.  It segments paper-white from
plate ink, floods the white that the image border can reach, keeps the white
that the border cannot reach, erodes away every narrow channel so that only
chamber cores survive, and counts those cores.  No renderer, prompt, generator,
verifier, scene, or gold input is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WHITE_LEVEL = 225
CHANNEL_KILL_HALFWIDTH = 13  # square erosion half-width: kills any run <= 26 px wide
MIN_CORE_AREA = 40


def _flood(seed: np.ndarray, allowed: np.ndarray) -> np.ndarray:
    """4-connected reachability inside `allowed`, seeded by `seed`."""
    current = seed & allowed
    height, width = allowed.shape
    while True:
        before = int(current.sum())
        for col in range(1, width):
            current[:, col] |= current[:, col - 1] & allowed[:, col]
        for col in range(width - 2, -1, -1):
            current[:, col] |= current[:, col + 1] & allowed[:, col]
        for row in range(1, height):
            current[row] |= current[row - 1] & allowed[row]
        for row in range(height - 2, -1, -1):
            current[row] |= current[row + 1] & allowed[row]
        if int(current.sum()) == before:
            return current


def _erode_square(mask: np.ndarray, half: int) -> np.ndarray:
    out = mask.copy()
    for axis in (0, 1):
        rolled = out.copy()
        for shift in range(1, half + 1):
            forward = np.zeros_like(out)
            backward = np.zeros_like(out)
            if axis == 0:
                forward[:-shift] = out[shift:]
                backward[shift:] = out[:-shift]
            else:
                forward[:, :-shift] = out[:, shift:]
                backward[:, shift:] = out[:, :-shift]
            rolled &= forward & backward
        out = rolled
    return out


def _count_components(mask: np.ndarray, min_area: int) -> int:
    remaining = mask.copy()
    count = 0
    while True:
        hits = np.argwhere(remaining)
        if hits.size == 0:
            return count
        seed = np.zeros_like(remaining)
        seed[hits[0][0], hits[0][1]] = True
        blob = _flood(seed, remaining)
        if int(blob.sum()) >= min_area:
            count += 1
        remaining &= ~blob


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB")).astype(np.int16)
    white = array.min(axis=2) >= WHITE_LEVEL

    border = np.zeros_like(white)
    border[0, :] = True
    border[-1, :] = True
    border[:, 0] = True
    border[:, -1] = True

    outside = _flood(border, white)
    enclosed = white & ~outside
    cores = _erode_square(enclosed, CHANNEL_KILL_HALFWIDTH)
    return str(_count_components(cores, MIN_CORE_AREA))
