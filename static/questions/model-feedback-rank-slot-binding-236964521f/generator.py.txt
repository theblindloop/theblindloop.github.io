"""Rank-slot binding correspondence world.

A KEY row shows N coloured tokens, each with a rank ruler beneath it: one
small filled cell per unit of that token's rank, so the ruler's filled-cell
count is the token's rank (1..N). The KEY records a colour-identity to -rank
binding. Two candidate panels A and B below each show the same N colours as a
2-by-3 grid of square cells; every cell carries a printed rank label (1..N)
drawn in a fixed non-monotone arrangement. A panel is *matched* exactly when
every coloured token sits inside the cell whose printed label equals that
token's keyed rank. The solver must bind each colour to its KEY rank, carry
that binding into each two-dimensional panel, and verify the per-cell slot
placement of every token against its labelled rank-cell. This is a
correspondence-binding program: unlike the parent's linear-order predicate on
a one-dimensional row, it composes the identity-to-rank binding with a 2D read
of printed per-cell rank labels and a per-cell positional equality. The other
panel is a permutation of the same colours with at least three tokens sitting
in the wrong cell.

The renderer owns scene sampling, rasterization and the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same decision
from the final PNG alone: it detects each coloured token, reads the filled-cell
rank ruler under each KEY token to build the colour->rank map, then for each
candidate panel verifies that the colour in every labelled cell is the token
whose keyed rank equals that cell's printed label.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 780
HEIGHT = 740
BACKGROUND = (255, 255, 255)
LABEL = (35, 35, 35)
PIP = (45, 45, 45)
BORDER = (90, 90, 90)

N = 6
COLOR_NAMES = ["red", "green", "blue", "orange", "purple", "brown"]
COLOR_RGB = {
    "red": (220, 40, 40),
    "green": (40, 170, 40),
    "blue": (40, 80, 220),
    "orange": (255, 150, 20),
    "purple": (150, 60, 190),
    "brown": (120, 80, 40),
}

TOKEN = 46
PITCH = 96
FIRST_CX = 150
KEY_TOP = 96
LABEL_Y = 48

CELL = 9
CELL_GAP = 2
RULER_GAP = 12
RULER_W = N * CELL + (N - 1) * CELL_GAP

# Candidate panels: each is a 2x3 grid of cells; cells are labelled (printed)
# in this fixed non-monotone order over row-major positions 0..5.
CELLSIZE = 92
CELLGAP = 12
PANEL_L = 120
PANEL_COLS = 3
PANEL_ROWS = 2
A_TOP = 360
B_TOP = 560

LABELS = [2, 5, 1, 6, 3, 4]
QUAR_DECOY_MISPLACE = 3

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_CX = [FIRST_CX + i * PITCH for i in range(N)]


def _panel_cell_rect(panel_top: int, pos: int) -> tuple[int, int, int, int]:
    col = pos % PANEL_COLS
    row = pos // PANEL_COLS
    x0 = PANEL_L + col * (CELLSIZE + CELLGAP)
    y0 = panel_top + row * (CELLSIZE + CELLGAP)
    return x0, y0, x0 + CELLSIZE, y0 + CELLSIZE


def _perm(rng: np.random.Generator, n: int) -> list[int]:
    return list(rng.permutation(n))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)

    key_colors = [COLOR_NAMES[i] for i in _perm(rng, N)]
    key_ranks = [int(r) + 1 for r in _perm(rng, N)]
    color_to_rank = dict(zip(key_colors, key_ranks))
    rank_to_color = {rank: color for color, rank in color_to_rank.items()}

    correct_placement = [rank_to_color[label] for label in LABELS]

    while True:
        decoy = [COLOR_NAMES[i] for i in _perm(rng, N)]
        displaced = sum(
            color_to_rank[decoy[pos]] != LABELS[pos] for pos in range(N)
        )
        if displaced >= QUAR_DECOY_MISPLACE:
            break

    correct_index = int(rng.integers(0, 2))
    if correct_index == 0:
        a_placement = list(correct_placement)
        b_placement = decoy
    else:
        a_placement = decoy
        b_placement = list(correct_placement)

    return {
        "key_colors": key_colors,
        "key_ranks": key_ranks,
        "a_placement": a_placement,
        "b_placement": b_placement,
        "correct_index": correct_index,
    }


def _draw_token(draw: ImageDraw.ImageDraw, cx: int, top: int, color: str) -> None:
    x0 = cx - TOKEN // 2
    draw.rectangle([x0, top, x0 + TOKEN, top + TOKEN], fill=COLOR_RGB[color], outline=(255, 255, 255), width=1)


def _draw_ruler(draw: ImageDraw.ImageDraw, cx: int, top: int, rank: int) -> None:
    x0 = cx - RULER_W // 2
    y0 = top
    for i in range(rank):
        x = x0 + i * (CELL + CELL_GAP)
        draw.rectangle([x, y0, x + CELL, y0 + CELL], fill=PIP)


def _draw_panel(draw: ImageDraw.ImageDraw, font: ImageFont.FreeTypeFont, top: int, placement: list[str]) -> None:
    for pos, color in enumerate(placement):
        x0, y0, x1, y1 = _panel_cell_rect(top, pos)
        draw.rectangle([x0, y0, x1, y1], fill=BACKGROUND, outline=BORDER, width=2)
        label_text = str(LABELS[pos])
        draw.text((x0 + 4, y0 + 2), label_text, font=font, fill=LABEL)
        cx = (x0 + x1) // 2
        token_top = y0 + (CELLSIZE - TOKEN) // 2
        _draw_token(draw, cx, token_top, color)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype(_FONT_PATH, 24)
    except OSError:
        font = ImageFont.load_default()

    draw.text((24, LABEL_Y), "KEY", font=font, fill=LABEL)
    for i, color in enumerate(scene["key_colors"]):
        _draw_token(draw, _CX[i], KEY_TOP, color)
        _draw_ruler(draw, _CX[i], KEY_TOP + TOKEN + RULER_GAP, scene["key_ranks"][i])

    draw.text((24, A_TOP - 6), "A", font=font, fill=LABEL)
    _draw_panel(draw, font, A_TOP, scene["a_placement"])

    draw.text((24, B_TOP - 6), "B", font=font, fill=LABEL)
    _draw_panel(draw, font, B_TOP, scene["b_placement"])

    return img


def _color_to_rank(scene: dict) -> dict:
    return dict(zip(scene["key_colors"], scene["key_ranks"]))


def _matched(placement: list[str], color_to_rank: dict) -> bool:
    for pos, color in enumerate(placement):
        if color_to_rank[color] != LABELS[pos]:
            return False
    return True


def analytic_gold(scene: dict) -> str:
    """Structural gold: which candidate panel is bound to the KEY rank slots."""
    color_to_rank = _color_to_rank(scene)
    a_match = _matched(scene["a_placement"], color_to_rank)
    b_match = _matched(scene["b_placement"], color_to_rank)
    if a_match:
        return "A"
    if b_match:
        return "B"
    raise ValueError("neither candidate panel is bound to the rank slots")


def margin(scene: dict) -> float:
    """Commitment magnitude: how many decoy tokens sit in the wrong cell."""
    color_to_rank = _color_to_rank(scene)
    a_match = _matched(scene["a_placement"], color_to_rank)
    b_match = _matched(scene["b_placement"], color_to_rank)
    decoy = scene["b_placement"] if a_match else scene["a_placement"]
    displaced = sum(
        color_to_rank[decoy[pos]] != LABELS[pos] for pos in range(N)
    )
    return float(displaced)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where the decoy is barely different (ambiguous)."""
    return margin(scene) < QUAR_DECOY_MISPLACE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
