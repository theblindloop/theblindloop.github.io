"""Independent pixel-only inverse arm for the rank-slot binding world.

Recovers the decision (which candidate panel is bound to the KEY rank slots)
from the final rasterized PNG alone. It locates each coloured KEY token by
exact colour, reads the filled-cell rank ruler beneath it to build the
colour->rank map, then for each candidate 2-by-3 panel detects which colour
occupies every labelled cell and returns the label of the panel in which every
token sits in the cell printed with its keyed rank. It never receives the
latent scene or analytic gold and imports only numpy/PIL.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (220, 40, 40),
    "green": (40, 170, 40),
    "blue": (40, 80, 220),
    "orange": (255, 150, 20),
    "purple": (150, 60, 190),
    "brown": (120, 80, 40),
}
PIP = (45, 45, 45)

TOKEN = 46
KEY_TOP = 96
CELL = 9
CELL_GAP = 2
RULER_GAP = 12
RULER_W = 6 * CELL + 5 * CELL_GAP
RULER_TOP = KEY_TOP + TOKEN + RULER_GAP

CELLSIZE = 92
CELLGAP = 12
PANEL_L = 120
PANEL_COLS = 3
PANEL_ROWS = 2
A_TOP = 360
B_TOP = 560

LABELS = [2, 5, 1, 6, 3, 4]


def _panel_cell_rect(panel_top: int, pos: int) -> tuple[int, int, int, int]:
    col = pos % PANEL_COLS
    row = pos // PANEL_COLS
    x0 = PANEL_L + col * (CELLSIZE + CELLGAP)
    y0 = panel_top + row * (CELLSIZE + CELLGAP)
    return x0, y0, x0 + CELLSIZE, y0 + CELLSIZE


def _tokens_in_band(arr: np.ndarray, top: int) -> list[tuple[str, int]]:
    """Return (color_name, center_x) for each KEY token in a row band, left-to-right."""
    band = arr[top : top + TOKEN]
    found = []
    for name, rgb in COLOR_RGB.items():
        mask = (band == np.array(rgb)).all(axis=-1)
        cols = np.nonzero(np.any(mask, axis=0))[0]
        if cols.size == 0:
            continue
        cx = int(cols.min() + cols.max()) // 2
        found.append((name, cx))
    found.sort(key=lambda item: item[1])
    return found


def _ruler_rank(arr: np.ndarray, cx: int) -> int:
    """Read the filled-cell count (rank) under a KEY token at center x."""
    band = arr[RULER_TOP : RULER_TOP + CELL]
    pip = (band == np.array(PIP)).all(axis=-1)
    x0 = cx - RULER_W // 2
    runs = 0
    prev_on = False
    for dx in range(RULER_W):
        x = x0 + dx
        if x < 0 or x >= band.shape[1]:
            continue
        on = bool(np.any(pip[:, x]))
        if on and not prev_on:
            runs += 1
        prev_on = on
    return runs


def _cell_colour(arr: np.ndarray, panel_top: int, pos: int) -> str:
    x0, y0, x1, y1 = _panel_cell_rect(panel_top, pos)
    window = arr[y0 : y1 + 1, x0 : x1 + 1]
    counts = {}
    for name, rgb in COLOR_RGB.items():
        counts[name] = int(np.count_nonzero((window == np.array(rgb)).all(axis=-1)))
    best = max(counts, key=counts.get)
    return best if counts[best] > 0 else ""


def _panel_matched(arr: np.ndarray, panel_top: int, color_to_rank: dict[str, int]) -> bool:
    for pos, label in enumerate(LABELS):
        colour = _cell_colour(arr, panel_top, pos)
        if color_to_rank.get(colour) != label:
            return False
    return True


def decision_from_image(image: Image.Image) -> str:
    """Return A or B: which candidate panel is bound to the KEY rank slots."""
    arr = np.asarray(image.convert("RGB"))

    key_tokens = _tokens_in_band(arr, KEY_TOP)
    if len(key_tokens) != 6:
        return "A"
    color_to_rank = {}
    for name, cx in key_tokens:
        r = _ruler_rank(arr, cx)
        color_to_rank[name] = r if r >= 1 else 1
    if len(color_to_rank) != 6:
        return "A"

    a_match = _panel_matched(arr, A_TOP, color_to_rank)
    b_match = _panel_matched(arr, B_TOP, color_to_rank)
    return "A" if a_match else "B"
