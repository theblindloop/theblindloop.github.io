"""Deterministic renderer and analytic arm for route bottleneck capacity."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

BACKGROUND = (247, 249, 252)
DARK = (31, 36, 44)
GREEN = (32, 170, 92)
GRAY = (132, 141, 153)
COLORS = {
    "orange": (242, 139, 38),
    "purple": (142, 68, 173),
    "blue": (45, 117, 220),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    dx, dy = rng.randint(-6, 6), rng.randint(-5, 5)
    lane_y = [57 + dy, 165 + dy, 273 + dy]

    # Equal link-width sums (34) suppress total-ink shortcuts while their
    # bottlenecks remain separated by two visible pixels.
    profiles = [
        [4, 10, 10, 10],
        [6, 8, 10, 10],
        [8, 8, 9, 9],
    ]
    for profile in profiles:
        rng.shuffle(profile)
    rng.shuffle(profiles)

    target_lane = next(i for i, profile in enumerate(profiles) if min(profile) == 8)
    target_color = list(COLORS)[int(seed) % 3]
    other_colors = [name for name in COLORS if name != target_color]
    rng.shuffle(other_colors)
    lane_colors = other_colors[:]
    lane_colors.insert(target_lane, target_color)

    positions: dict[str, list[int]] = {}
    routes: list[list[str]] = []
    link_widths: dict[str, int] = {}
    terminal_colors: dict[str, str] = {}
    start_nodes: list[str] = []
    waypoint_nodes: list[str] = []
    terminal_nodes: list[str] = []
    xs = [43 + dx, 128 + dx, 214 + dx, 300 + dx, 387 + dx]
    for lane, (base_y, profile, color) in enumerate(zip(lane_y, profiles, lane_colors)):
        start = f"start_{lane}"
        waypoints = [f"waypoint_{lane}_{i}" for i in range(3)]
        terminal = f"terminal_{lane}"
        nodes = [start, *waypoints, terminal]
        offsets = [0, rng.randint(-13, 13), rng.randint(-13, 13), rng.randint(-13, 13), 0]
        # Keep adjacent segments visibly non-horizontal often, without allowing
        # routes in neighboring lanes to approach one another.
        if offsets[1:4] == [0, 0, 0]:
            offsets[2] = 9
        for x, off, node in zip(xs, offsets, nodes):
            positions[node] = [x, base_y + off]
        routes.append(nodes)
        start_nodes.append(start)
        waypoint_nodes.extend(waypoints)
        terminal_nodes.append(terminal)
        terminal_colors[terminal] = color
        for index, stroke_width in enumerate(profile):
            link_widths[f"{nodes[index]}|{nodes[index + 1]}"] = stroke_width

    return {
        "width": 430,
        "height": 330,
        "positions": positions,
        "routes": routes,
        "link_widths": link_widths,
        "terminal_colors": terminal_colors,
        "start_nodes": start_nodes,
        "waypoint_nodes": waypoint_nodes,
        "terminal_nodes": terminal_nodes,
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def point(node: str) -> tuple[int, int]:
        x, y = scene["positions"][node]
        return int(x * scale), int(y * scale)

    for route in scene["routes"]:
        for a, b in zip(route, route[1:]):
            key = f"{a}|{b}"
            draw.line([point(a), point(b)], fill=DARK, width=int(scene["link_widths"][key]) * scale)

    start_radius = 16 * scale
    for node in scene["start_nodes"]:
        x, y = point(node)
        draw.ellipse((x - start_radius, y - start_radius, x + start_radius, y + start_radius), fill=GREEN)
    waypoint_radius = 10 * scale
    for node in scene["waypoint_nodes"]:
        x, y = point(node)
        draw.ellipse((x - waypoint_radius, y - waypoint_radius, x + waypoint_radius, y + waypoint_radius), fill=GRAY)
    terminal_half = 15 * scale
    for node in scene["terminal_nodes"]:
        x, y = point(node)
        color = COLORS[scene["terminal_colors"][node]]
        draw.rectangle((x - terminal_half, y - terminal_half, x + terminal_half, y + terminal_half), fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _capacities(scene: dict) -> dict[str, int]:
    result = {}
    for route in scene["routes"]:
        widths = [scene["link_widths"][f"{a}|{b}"] for a, b in zip(route, route[1:])]
        result[scene["terminal_colors"][route[-1]]] = min(widths)
    return result


def analytic_gold(scene: dict) -> str:
    capacities = _capacities(scene)
    best = max(capacities.values())
    winners = [color for color, value in capacities.items() if value == best]
    if len(winners) != 1:
        raise ValueError(f"expected one maximum bottleneck, got {capacities}")
    return winners[0]


def margin(scene: dict) -> float:
    capacities = sorted(_capacities(scene).values(), reverse=True)
    return float(capacities[0] - capacities[1])


def is_quarantined(scene: dict) -> bool:
    try:
        capacities = _capacities(scene)
        return len(capacities) != 3 or list(capacities.values()).count(max(capacities.values())) != 1 or margin(scene) < 2.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
