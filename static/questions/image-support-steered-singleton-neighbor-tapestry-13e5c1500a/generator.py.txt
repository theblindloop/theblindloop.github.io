"""Deterministic renderer and analytic arm for the singleton-neighborhood tapestry."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 768
TAG_COLORS = {
    "red": "#e34a4a",
    "green": "#35ad6f",
    "blue": "#4385df",
}
DEFAULT_PALETTE = {
    "background": "#f7f3e8",
    "board": "#172b4d",
    "on": "#48d7d0",
    "off": "#f4d78b",
    "gold": "#f0b93a",
    **TAG_COLORS,
}


def _update(bits: Iterable[int]) -> list[int]:
    values = [int(v) for v in bits]
    n = len(values)
    return [
        int(values[(i - 1) % n] + values[i] + values[(i + 1) % n] == 1)
        for i in range(n)
    ]


def _hamming(left: Iterable[int], right: Iterable[int]) -> int:
    return sum(int(a) != int(b) for a, b in zip(left, right))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(1000):
        reference = [rng.randrange(2) for _ in range(7)]
        target = _update(reference)
        if 2 <= sum(target) <= 5:
            break
    else:
        raise RuntimeError("could not sample a nondegenerate source")

    ones = [i for i, value in enumerate(target) if value]
    zeros = [i for i, value in enumerate(target) if not value]
    alternatives: list[list[int]] = []
    for one in ones:
        for zero in zeros:
            changed = target.copy()
            changed[one], changed[zero] = 0, 1
            alternatives.append(changed)
    rng.shuffle(alternatives)
    distractors: list[list[int]] = []
    for row in alternatives:
        if row != target and row not in distractors:
            distractors.append(row)
        if len(distractors) == 2:
            break

    rows = [target, *distractors]
    rng.shuffle(rows)
    tags = list(TAG_COLORS)
    rng.shuffle(tags)
    return {
        "reference": reference,
        "candidate_rows": rows,
        "tags": tags,
        "palette": dict(DEFAULT_PALETTE),
    }


def analytic_gold(scene: dict) -> str:
    target = _update(scene["reference"])
    matches = [
        index for index, row in enumerate(scene["candidate_rows"]) if list(row) == target
    ]
    if len(matches) != 1:
        raise ValueError("scene does not contain one unique transformed row")
    return str(scene["tags"][matches[0]])


def margin(scene: dict) -> float:
    target = _update(scene["reference"])
    distances = sorted(_hamming(row, target) for row in scene["candidate_rows"])
    if not distances or distances[0] != 0:
        return 0.0
    return float(distances[1]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (ValueError, IndexError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    reference = [int(v) for v in scene["reference"]]
    rows = [[int(v) for v in row] for row in scene["candidate_rows"]]
    tags = [str(tag) for tag in scene["tags"]]
    if len(reference) != 7 or len(rows) != 3 or any(len(row) != 7 for row in rows):
        raise ValueError("expected one source and three seven-stitch candidates")
    if sorted(tags) != ["blue", "green", "red"]:
        raise ValueError("candidate tags must be red, green, and blue")

    image = Image.new("RGB", (WIDTH, HEIGHT), palette["background"])
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((24, 24, 743, 743), radius=22, fill=palette["board"])

    y_centers = (105, 293, 481, 669)
    x_centers = (145, 235, 325, 415, 505, 595, 685)
    all_rows = [reference, *rows]
    all_tags = ["gold", *tags]
    for row_index, (bits, tag) in enumerate(zip(all_rows, all_tags)):
        cy = y_centers[row_index]
        color = palette[tag]
        if tag == "gold":
            draw.polygon(((68, cy - 27), (95, cy), (68, cy + 27), (41, cy)), fill=color)
            draw.ellipse((61, cy - 7, 75, cy + 7), fill=palette["board"])
        else:
            draw.ellipse((41, cy - 27, 95, cy + 27), fill=color)
            draw.ellipse((61, cy - 7, 75, cy + 7), fill=palette["board"])
        for bit, cx in zip(bits, x_centers):
            fill = palette["on"] if bit else palette["off"]
            draw.rounded_rectangle((cx - 29, cy - 29, cx + 29, cy + 29), radius=7, fill=fill)
            # Four navy corner stitches keep each tile visibly integrated with the carrier.
            for dx, dy in ((-20, -20), (20, -20), (-20, 20), (20, 20)):
                draw.ellipse((cx + dx - 3, cy + dy - 3, cx + dx + 3, cy + dy + 3), fill=palette["board"])
    return image
