"""Independent pixel-only inverse arm for the tapestry."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "board": (23, 43, 77),
    "on": (72, 215, 208),
    "off": (244, 215, 139),
    "gold": (240, 185, 58),
    "red": (227, 74, 74),
    "green": (53, 173, 111),
    "blue": (67, 133, 223),
}


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        output.append((len(points), float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), max(xs) - min(xs) + 1, max(ys) - min(ys) + 1))
    return output


def _exact_mask(array: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(color, dtype=np.uint8), axis=2)


def _update(bits: list[int]) -> list[int]:
    n = len(bits)
    return [int(bits[(i - 1) % n] + bits[i] + bits[(i + 1) % n] == 1) for i in range(n)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if min(height, width) < 300:
        raise ValueError("raster too small")

    board = _exact_mask(array, COLORS["board"])
    board_points = np.argwhere(board)
    if len(board_points) < height * width * 0.70:
        raise ValueError("continuous tapestry carrier is missing or materially erased")
    bh = int(board_points[:, 0].max() - board_points[:, 0].min() + 1)
    bw = int(board_points[:, 1].max() - board_points[:, 1].min() + 1)
    if bw < width * 0.82 or bh < height * 0.82:
        raise ValueError("tapestry does not span the raster")

    cells = []
    for bit, name in ((1, "on"), (0, "off")):
        for area, cx, cy, cw, ch in _components(_exact_mask(array, COLORS[name])):
            if area >= 3000:
                if not (52 <= cw <= 62 and 52 <= ch <= 62 and area >= 3200):
                    raise ValueError("a stitch is materially damaged")
                cells.append((cy, cx, bit))
    if len(cells) != 28:
        raise ValueError(f"expected 28 intact stitches, found {len(cells)}")

    cells.sort()
    rows: list[list[tuple[float, float, int]]] = []
    for cy, cx, bit in cells:
        if not rows or abs(cy - np.mean([v[0] for v in rows[-1]])) > height * 0.09:
            rows.append([])
        rows[-1].append((cy, cx, bit))
    if len(rows) != 4 or any(len(row) != 7 for row in rows):
        raise ValueError("stitches do not form four complete seven-cell rows")
    binary_rows = []
    row_y = []
    for row in rows:
        row.sort(key=lambda item: item[1])
        xs = [item[1] for item in row]
        if min(np.diff(xs)) < width * 0.08:
            raise ValueError("cell columns are ambiguous")
        binary_rows.append([item[2] for item in row])
        row_y.append(float(np.mean([item[0] for item in row])))

    tag_rows: dict[str, float] = {}
    for name in ("red", "green", "blue"):
        components = [item for item in _components(_exact_mask(array, COLORS[name])) if item[0] > 900]
        if len(components) != 1 or components[0][0] < 1200:
            raise ValueError(f"{name} tag is missing or damaged")
        tag_rows[name] = components[0][2]
    gold_components = [item for item in _components(_exact_mask(array, COLORS["gold"])) if item[0] > 900]
    if len(gold_components) != 1 or gold_components[0][0] < 1200:
        raise ValueError("gold source tag is missing or damaged")

    target = _update(binary_rows[0])
    matches = [index for index, row in enumerate(binary_rows[1:], start=1) if row == target]
    if len(matches) != 1:
        raise ValueError("transformed row is not uniquely represented")
    match_y = row_y[matches[0]]
    distances = sorted((abs(y - match_y), name) for name, y in tag_rows.items())
    if len(distances) < 2 or distances[0][0] > height * 0.04 or distances[1][0] < height * 0.10:
        raise ValueError("tag-to-row binding is ambiguous")
    return distances[0][1]
