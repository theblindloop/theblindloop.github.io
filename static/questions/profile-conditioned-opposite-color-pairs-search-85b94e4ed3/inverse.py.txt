"""Pixel-only inverse arm; deliberately independent of world generation code."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_COLORS = ((232, 74, 95), (22, 138, 173), (244, 162, 97))
_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    work = mask.copy()
    found = []
    height, width = work.shape
    while bool(work.any()):
        y, x = (int(v) for v in np.argwhere(work)[0])
        work[y, x] = False
        queue = deque([(y, x)])
        sx = sy = count = 0
        while queue:
            yy, xx = queue.popleft()
            sx += xx
            sy += yy
            count += 1
            for ny, nx in ((yy-1, xx), (yy+1, xx), (yy, xx-1), (yy, xx+1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if count >= 250:
            found.append((sx / count, sy / count, count))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    beads: list[tuple[float, float, int]] = []
    for color_id, rgb in enumerate(_COLORS):
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) < 42 * 42
        for x, y, _area in _components(mask):
            beads.append((x, y, color_id))
    if len(beads) != 24:
        raise ValueError(f"expected 24 colored beads, found {len(beads)}")

    ordered = sorted(beads, key=lambda item: item[0])
    panels = [ordered[i:i+6] for i in range(0, 24, 6)]
    scores = []
    for group in panels:
        if len(group) != 6 or max(p[0] for p in group) - min(p[0] for p in group) > 155:
            raise ValueError("could not separate bead panels")
        cx = sum(p[0] for p in group) / 6.0
        cy = sum(p[1] for p in group) / 6.0
        around = sorted(group, key=lambda p: math.atan2(p[1] - cy, p[0] - cx))
        score = sum(around[i][2] == around[i+3][2] for i in range(3))
        scores.append(score)
    best = max(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if best != 3 or len(winners) != 1 or best - sorted(scores, reverse=True)[1] < 2:
        raise ValueError(f"ambiguous opposite-pair scores {scores}")
    return _LABELS[winners[0]]
