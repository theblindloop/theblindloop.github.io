"""Deterministic sampler and renderer for opposite-color pairing search."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 900, 360, 3
INK = "#263238"
PANEL = "#90a4ae"
GUIDE = "#c5ced3"
BACKGROUND = "#fbfaf6"
BASE_PALETTE = ("#e84a5f", "#168aad", "#f4a261")
LABELS = ("A", "B", "C", "D")


def _score(sequence: list[int]) -> int:
    return sum(sequence[i] == sequence[i + 3] for i in range(3))


def sample_scene(seed: int) -> dict:
    """Sample equal-histogram panels with one well-separated three-pair winner."""
    rng = random.Random(int(seed))
    palette = list(BASE_PALETTE)
    rng.shuffle(palette)
    first = [0, 1, 2]
    rng.shuffle(first)
    target = first + first

    distractor_pool = []
    for seq in sorted(set(itertools.permutations((0, 0, 1, 1, 2, 2)))):
        candidate = list(seq)
        if _score(candidate) <= 1 and candidate != target:
            distractor_pool.append(candidate)
    distractors = rng.sample(distractor_pool, 3)
    answer_index = int(seed) % 4
    panels = distractors[:]
    panels.insert(answer_index, target)
    return {
        "panels": panels,
        "rotation": rng.choice((-14, -8, -3, 3, 8, 14)),
        "palette": palette,
    }


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    rotation = float(scene["rotation"])
    palette = scene["palette"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=17 * SCALE)
    centers = (120, 340, 560, 780)
    cy, radius, bead_r = 166, 67, 17

    def line(points, **kw):
        draw.line([(int(x * SCALE), int(y * SCALE)) for x, y in points], **kw)

    for panel_index, (cx, sequence) in enumerate(zip(centers, panels, strict=True)):
        box = (cx - 96, 48, cx + 96, 294)
        draw.rounded_rectangle(tuple(int(v * SCALE) for v in box), radius=12 * SCALE,
                               outline=PANEL, width=2 * SCALE, fill="#ffffff")
        positions = []
        for i in range(6):
            theta = math.radians(rotation + i * 60 - 90)
            positions.append((cx + radius * math.cos(theta), cy + radius * math.sin(theta)))
        for i in range(3):
            line((positions[i], positions[i + 3]), fill=GUIDE, width=3 * SCALE)
        cr = 5 * SCALE
        draw.ellipse((cx*SCALE-cr, cy*SCALE-cr, cx*SCALE+cr, cy*SCALE+cr), fill=INK)
        for color_index, (x, y) in zip(sequence, positions, strict=True):
            r = bead_r * SCALE
            draw.ellipse((int(x*SCALE-r), int(y*SCALE-r), int(x*SCALE+r), int(y*SCALE+r)),
                         fill=palette[color_index], outline=INK, width=3*SCALE)
        label = LABELS[panel_index]
        bbox = draw.textbbox((0, 0), label, font=font)
        tw = bbox[2] - bbox[0]
        draw.text((cx*SCALE-tw/2, 315*SCALE), label, fill=INK, font=font)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [_score(sequence) for sequence in scene["panels"]]
    winners = [i for i, score in enumerate(scores) if score == max(scores)]
    if len(winners) != 1:
        raise ValueError("scene has no unique winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_score(sequence) for sequence in scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = [_score(sequence) for sequence in scene["panels"]]
    return scores.count(3) != 1 or max(scores) != 3 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
