"""Gauge-needle nearest-tick selection world.

A single dark needle pivots at the centre of a light circular gauge face. Three
coloured tick marks (red, blue, green) sit on the annular track at fixed
reference angles. The question is a pure angular measurement: which coloured
tick is the needle closest to, i.e. which tick has the smallest angular gap
(in circular degrees) to the needle's pointing direction.

The inverse arm recovers the single angular reading of the needle from the
raster by fitting its black ink to a direction and compares it to the fixed
calibration angles. This is an instrument-gauge read: one rotating pointer,
fixed graduated reference marks, and a nearest-mark selection based on an
angular distance measurement.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "gauge_needle_nearest-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (248, 248, 248)
FACE_RING = (190, 190, 190)
HUB_RGB = (120, 120, 120)
NEEDLE_RGB = (10, 10, 10)

WIDTH = 640
HEIGHT = 500
CENTER_X = 320
CENTER_Y = 260
FACE_RADIUS = 205
TICK_RADIUS_IN = 135
TICK_RADIUS_OUT = 165
NEEDLE_RADIUS = 125
NEEDLE_RADIUS_IN = 22
NEEDLE_WIDTH = 9
HUB_RADIUS = 13

TICK_ANGLES = {"red": 200.0, "blue": 270.0, "green": 340.0}

MIN_MARGIN_DEG = 20.0
QUARANTINE_DEG = 25.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced nearest tick.

    The needle angle is drawn so that the top-two tick angular gaps differ by at
    least MIN_MARGIN_DEG, mirroring a real unambiguous gauge read.
    """
    rng = random.Random(seed * 131 + 17)
    tick_angles = dict(TICK_ANGLES)
    while True:
        needle_angle = rng.uniform(200.0, 340.0)
        gaps = sorted(
            _circular_gap(needle_angle, tick_angles[color])
            for color in PALETTE
        )
        if gaps[1] - gaps[0] >= MIN_MARGIN_DEG:
            break
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": CENTER_X,
        "center_y": CENTER_Y,
        "tick_angles": tick_angles,
        "needle_angle": needle_angle,
    }


def _circular_gap(a: float, b: float) -> float:
    diff = abs(a - b) % 360.0
    return float(min(diff, 360.0 - diff))


def _point(cx: float, cy: float, radius: float, angle_deg: float) -> tuple[float, float]:
    rad = math.radians(angle_deg)
    return (cx + radius * math.cos(rad), cy + radius * math.sin(rad))


def render(scene: dict) -> Image.Image:
    """Render I = R(z): a circular gauge with coloured ticks and one needle."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    cx, cy = scene["center_x"], scene["center_y"]

    # Face rim.
    box = (cx - FACE_RADIUS, cy - FACE_RADIUS, cx + FACE_RADIUS, cy + FACE_RADIUS)
    draw.ellipse(box, outline=FACE_RING, width=4)

    # Coloured calibration ticks.
    tick_angles = scene["tick_angles"]
    for color in PALETTE:
        angle = tick_angles[color]
        p_in = _point(cx, cy, TICK_RADIUS_IN, angle)
        p_out = _point(cx, cy, TICK_RADIUS_OUT, angle)
        draw.line([p_in, p_out], fill=COLOR_RGB[color], width=7)

    # Central hub.
    hub_box = (cx - HUB_RADIUS, cy - HUB_RADIUS, cx + HUB_RADIUS, cy + HUB_RADIUS)
    draw.ellipse(hub_box, fill=HUB_RGB)

    # Needle: dark line from near the hub out to the track.
    angle = scene["needle_angle"]
    p_in = _point(cx, cy, NEEDLE_RADIUS_IN, angle)
    p_out = _point(cx, cy, NEEDLE_RADIUS, angle)
    draw.line([p_in, p_out], fill=NEEDLE_RGB, width=NEEDLE_WIDTH)
    return img


def _nearest_tick_reading(scene: dict) -> tuple[str, float, float]:
    """Return (nearest color, its angular gap, top-two margin in degrees)."""
    needle = scene["needle_angle"]
    tick_angles = scene["tick_angles"]
    keyed = [
        (color, _circular_gap(needle, tick_angles[color]))
        for color in PALETTE
    ]
    keyed.sort(key=lambda item: item[1])
    (c0, g0), (c1, g1) = keyed[0], keyed[1]
    return c0, g0, g1 - g0


def analytic_gold(scene: dict) -> str:
    """The colour of the tick with the smallest angular gap to the needle."""
    nearest, _gap, _margin = _nearest_tick_reading(scene)
    return nearest


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two angular-gap difference in degrees."""
    _nearest, _gap, margin_deg = _nearest_tick_reading(scene)
    return float(margin_deg)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the pointer sits almost exactly between two ticks."""
    return margin(scene) < QUARANTINE_DEG


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Rotating the needle to a different angle changes which tick is nearest (and
    changes the raster), and any global flip/rotation maps the raster to a
    distinct image. Every latent field (tick angles and the needle angle) is
    directly rendered as visible ink, so the parametrisation has no hidden
    label alias.
    """
    return []
