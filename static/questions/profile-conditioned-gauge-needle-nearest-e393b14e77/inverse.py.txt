"""Independent pixel-only inverse arm for the gauge-needle world.

This module receives ONLY the rasterized PNG (plus the public palette and the
fixed gauge geometry, which are the world's declared contract) and re-measures
which coloured tick the dark needle points closest to. It shares no geometry
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytical gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "gauge_needle_nearest-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (248, 248, 248)
NEEDLE_RGB = (10, 10, 10)

TICK_ANGLES = {"red": 200.0, "blue": 270.0, "green": 340.0}
CENTER_X = 320
CENTER_Y = 260


def _needle_angle(arr: np.ndarray) -> float:
    """Recover the needle's pointing angle from its dark ink alone.

    Every non-needle pixel is light (background / gray face / colored tick),
    so labelling pixels by nearest reference colour yields an isolated black
    mask for the needle. The centroid of that mask lies along the needle's
    length, so its direction from the fixed pivot gives the pointer angle.
    """
    refs = np.stack(
        [
            np.array(NEEDLE_RGB, dtype=np.float32),
            np.array(BACKGROUND, dtype=np.float32),
            np.array(HUB_SAMPLE, dtype=np.float32),
        ]
        + [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE],
        axis=0,
    )
    diffs = (
        arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    )
    labels = np.linalg.norm(diffs, axis=-1).argmin(axis=-1)
    ys, xs = np.nonzero(labels == 0)
    if len(ys) < 10:
        return float("nan")
    cy = float(np.mean(ys)) - CENTER_Y
    cx = float(np.mean(xs)) - CENTER_X
    return float(math.degrees(math.atan2(cy, cx)) % 360.0)


HUB_SAMPLE = (120, 120, 120)


def _circular_gap(a: float, b: float) -> float:
    diff = abs(a - b) % 360.0
    return float(min(diff, 360.0 - diff))


def decision_from_image(image: Image.Image) -> str:
    """Return the colour of the tick nearest to the needle's angle."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    angle = _needle_angle(arr)
    if math.isnan(angle):
        return PALETTE[0]
    gaps = {
        color: _circular_gap(angle, TICK_ANGLES[color]) for color in PALETTE
    }
    return min(gaps, key=gaps.get)
