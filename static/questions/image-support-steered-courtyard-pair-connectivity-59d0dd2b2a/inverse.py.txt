"""Independent inverse arm using only raster pixels and public color roles."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_NAMES = ("background", "sheet", "orange", "purple", "green", "pink")
_COLORS = np.asarray(
    [
        [248, 250, 252],
        [13, 116, 110],
        [234, 113, 16],
        [126, 34, 206],
        [22, 163, 74],
        [219, 39, 119],
    ],
    dtype=np.int32,
)
_COLOR_INDEX = {name: index for index, name in enumerate(_NAMES)}
_DOWNSAMPLE = 4


def _nearest_labels(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    if array.shape != (512, 512, 3):
        raise ValueError(f"expected a 512x512 RGB raster, got {array.shape}")
    distances = ((array[:, :, None, :] - _COLORS[None, None, :, :]) ** 2).sum(axis=3)
    return distances.argmin(axis=2).astype(np.uint8)


def _block_sheet(labels: np.ndarray) -> np.ndarray:
    sheet = labels == _COLOR_INDEX["sheet"]
    height, width = sheet.shape
    factor = _DOWNSAMPLE
    blocks = sheet.reshape(height // factor, factor, width // factor, factor)
    return blocks.mean(axis=(1, 3)) >= 0.5


def _components(mask: np.ndarray) -> tuple[np.ndarray, dict[int, int]]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    areas: dict[int, int] = {}
    component = 0
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if labels[y, x]:
            continue
        component += 1
        labels[y, x] = component
        queue = deque([(y, x)])
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and labels[ny, nx] == 0
                ):
                    labels[ny, nx] = component
                    queue.append((ny, nx))
        areas[component] = area
    return labels, areas


def _marker_component(
    pixel_labels: np.ndarray, region_labels: np.ndarray, color_name: str
) -> int:
    points = np.argwhere(pixel_labels == _COLOR_INDEX[color_name])
    if len(points) < 250:
        raise ValueError(f"{color_name} marker is missing or materially erased")
    y, x = np.rint(points.mean(axis=0) / _DOWNSAMPLE).astype(int)
    y = min(max(int(y), 0), region_labels.shape[0] - 1)
    x = min(max(int(x), 0), region_labels.shape[1] - 1)
    component = int(region_labels[y, x])
    if component == 0:
        raise ValueError(f"{color_name} marker does not lie in traversable white space")
    return component


def decision_from_image(image: Image.Image) -> str:
    """Select the pair sharing a 4-connected enclosed complement component."""

    pixel_labels = _nearest_labels(image)
    sheet = _block_sheet(pixel_labels)
    sheet_labels, sheet_areas = _components(sheet)
    substantial_sheet = [area for area in sheet_areas.values() if area >= 20]
    if len(substantial_sheet) != 1 or substantial_sheet[0] < 8500:
        raise ValueError("the thick teal sheet is missing, split, or materially corrupted")
    sheet_points = np.argwhere(sheet)
    y0, x0 = sheet_points.min(axis=0)
    y1, x1 = sheet_points.max(axis=0)
    if x0 > 6 or y0 > 6 or x1 < 121 or y1 < 121:
        raise ValueError("the teal sheet no longer spans the image")

    traversable_labels, areas = _components(~sheet)
    border_ids = set(int(value) for value in traversable_labels[0, :])
    border_ids.update(int(value) for value in traversable_labels[-1, :])
    border_ids.update(int(value) for value in traversable_labels[:, 0])
    border_ids.update(int(value) for value in traversable_labels[:, -1])
    border_ids.discard(0)
    holes = {key: area for key, area in areas.items() if key not in border_ids and area >= 40}
    if len(holes) != 5:
        raise ValueError(f"expected five enclosed chambers, recovered {len(holes)}")
    ordered_areas = sorted(holes.values(), reverse=True)
    if not (2400 <= ordered_areas[0] <= 3500):
        raise ValueError("the broad central chamber has an unsafe or corrupted area")
    if any(not 180 <= area <= 430 for area in ordered_areas[1:]):
        raise ValueError("a small chamber has an unsafe or corrupted area")

    marker_components = {
        name: _marker_component(pixel_labels, traversable_labels, name)
        for name in ("orange", "purple", "green", "pink")
    }
    if any(value in border_ids for value in marker_components.values()):
        raise ValueError("a marked chamber is open to the image exterior")
    orange_purple = marker_components["orange"] == marker_components["purple"]
    green_pink = marker_components["green"] == marker_components["pink"]
    if orange_purple == green_pink:
        raise ValueError("raster does not expose exactly one connected candidate pair")
    return "orange-purple" if orange_purple else "green-pink"
