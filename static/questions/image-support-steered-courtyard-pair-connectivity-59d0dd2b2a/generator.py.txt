"""Deterministic visual world for paired courtyard connectivity."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


CANVAS_SIZE = 512
PAIR_COLORS = {
    "orange-purple": ("orange", "purple"),
    "green-pink": ("green", "pink"),
}
PALETTE = {
    "background": [248, 250, 252],
    "sheet": [13, 116, 110],
    "orange": [234, 113, 16],
    "purple": [126, 34, 206],
    "green": [22, 163, 74],
    "pink": [219, 39, 119],
}


def _shift_box(box: list[int], dx: int, dy: int) -> list[int]:
    return [box[0] + dx, box[1] + dy, box[2] + dx, box[3] + dy]


def _center(box: list[int]) -> list[int]:
    return [(box[0] + box[2]) // 2, (box[1] + box[3]) // 2]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, safely rasterized scene from one integer seed."""

    rng = random.Random(int(seed))
    connected_pair = "orange-purple" if int(seed) % 2 == 0 else "green-pink"
    other_pair = "green-pink" if connected_pair == "orange-purple" else "orange-purple"
    dx, dy = rng.randint(-3, 3), rng.randint(-3, 3)

    central = _shift_box([150, 150, 362, 362], dx, dy)
    corner_boxes = [
        _shift_box([42, 50, 114, 122], dx, dy),
        _shift_box([398, 50, 470, 122], dx, dy),
        _shift_box([42, 390, 114, 462], dx, dy),
        _shift_box([398, 390, 470, 462], dx, dy),
    ]

    central_positions = (
        [[202 + dx, 202 + dy], [310 + dx, 310 + dy]]
        if rng.randrange(2) == 0
        else [[310 + dx, 202 + dy], [202 + dx, 310 + dy]]
    )
    connected_colors = list(PAIR_COLORS[connected_pair])
    single_colors = list(PAIR_COLORS[other_pair])
    rng.shuffle(connected_colors)
    rng.shuffle(single_colors)
    connected_markers = [
        {"color": color, "center": center}
        for color, center in zip(connected_colors, central_positions, strict=True)
    ]

    opposite_choices = ([0, 3], [1, 2])
    single_indices = list(opposite_choices[rng.randrange(2)])
    rng.shuffle(single_indices)
    single_chambers = [
        {"color": color, "box": corner_boxes[index], "center": _center(corner_boxes[index])}
        for color, index in zip(single_colors, single_indices, strict=True)
    ]
    distractor_chambers = [
        corner_boxes[index] for index in range(4) if index not in set(single_indices)
    ]

    return {
        "canvas_size": CANVAS_SIZE,
        "sheet_box": [18, 18, 494, 494],
        "connected_pair": connected_pair,
        "central_chamber": central,
        "connected_markers": connected_markers,
        "single_chambers": single_chambers,
        "distractor_chambers": distractor_chambers,
        "palette": copy.deepcopy(PALETTE),
        "clearance": 21.0,
    }


def _scaled_box(box: list[int], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(value * scale) for value in box)


def render(scene: dict) -> Image.Image:
    """Render a scene with supersampling; every analytic label input is visible."""

    size = int(scene["canvas_size"])
    palette = scene["palette"]
    _ = scene["connected_pair"]  # explicitly rendered through marker assignment
    scale = 3
    image = Image.new("RGB", (size * scale, size * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        _scaled_box(scene["sheet_box"], scale),
        radius=24 * scale,
        fill=tuple(palette["sheet"]),
    )

    chamber_fill = tuple(palette["background"])
    draw.rounded_rectangle(
        _scaled_box(scene["central_chamber"], scale),
        radius=30 * scale,
        fill=chamber_fill,
    )
    for item in scene["single_chambers"]:
        draw.rounded_rectangle(
            _scaled_box(item["box"], scale), radius=18 * scale, fill=chamber_fill
        )
    for box in scene["distractor_chambers"]:
        draw.rounded_rectangle(_scaled_box(box, scale), radius=18 * scale, fill=chamber_fill)

    marker_radius = 14 * scale
    for item in [*scene["connected_markers"], *scene["single_chambers"]]:
        cx, cy = (int(value * scale) for value in item["center"])
        draw.ellipse(
            (cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius),
            fill=tuple(palette[item["color"]]),
        )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["connected_pair"])


def margin(scene: dict) -> float:
    return float(scene["clearance"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractor_chambers"] = list(reversed(twin["distractor_chambers"]))
    return [("reverse_disjoint_distractor_draw_order", twin)]
