"""Deterministic renderer and analytic arm for Second Checkpoint Trace."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
ROUTE = (28, 43, 63)
START = (42, 105, 210)
CHECKPOINT_RGB = {
    "red": (220, 58, 70),
    "gold": (235, 171, 24),
    "green": (42, 158, 96),
    "purple": (145, 76, 190),
}
CHECKPOINTS = tuple(CHECKPOINT_RGB)


def _dense_polyline(vertices: Iterable[tuple[int, int]], step: int = 3) -> list[list[int]]:
    vertices = list(vertices)
    points: list[list[int]] = [[int(vertices[0][0]), int(vertices[0][1])]]
    for (x0, y0), (x1, y1) in zip(vertices, vertices[1:]):
        length = max(abs(x1 - x0), abs(y1 - y0))
        count = max(1, (length + step - 1) // step)
        for k in range(1, count + 1):
            point = [round(x0 + (x1 - x0) * k / count), round(y0 + (y1 - y0) * k / count)]
            if point != points[-1]:
                points.append(point)
    return points


def _reflect(points: list[list[int]], horizontal: bool, vertical: bool) -> list[list[int]]:
    reflected = []
    for x, y in points:
        reflected.append([WIDTH - x if horizontal else x, HEIGHT - y if vertical else y])
    return reflected


def sample_scene(seed: int) -> dict:
    """Sample a balanced four-answer scene with a compact U-folded route."""

    rng = random.Random(int(seed) ^ 0x5EC0A7)
    dx = rng.randint(-11, 11)
    dy = rng.randint(-10, 10)
    vertices = [
        (132 + dx, 226 + dy),
        (371 + dx + rng.randint(-7, 7), 226 + dy),
        (371 + dx + rng.randint(-7, 7), 286 + dy + rng.randint(-3, 3)),
        (157 + dx + rng.randint(-8, 8), 286 + dy + rng.randint(-3, 3)),
    ]
    points = _dense_polyline(vertices)
    points = _reflect(points, bool((seed // 8) % 2), bool((seed // 16) % 2))

    fractions = (0.16, 0.37, 0.63, 0.84)
    checkpoint_indices = [round(fraction * (len(points) - 1)) for fraction in fractions]
    start_end = (seed // 4) % 2
    physical_order = list(range(4)) if start_end == 0 else list(reversed(range(4)))

    answer_color = CHECKPOINTS[seed % len(CHECKPOINTS)]
    remaining = [color for color in CHECKPOINTS if color != answer_color]
    rng.shuffle(remaining)
    traversal_colors = [remaining[0], answer_color, remaining[1], remaining[2]]
    checkpoint_colors = [""] * 4
    for traversal_slot, physical_slot in enumerate(physical_order):
        checkpoint_colors[physical_slot] = traversal_colors[traversal_slot]

    return {
        "route_points": points,
        "checkpoint_indices": checkpoint_indices,
        "checkpoint_colors": checkpoint_colors,
        "start_end": start_end,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int], inner: int) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=BACKGROUND)


def render(scene: dict) -> Image.Image:
    points = [[int(p[0]), int(p[1])] for p in scene["route_points"]]
    indices = [int(value) for value in scene["checkpoint_indices"]]
    colors = [str(value) for value in scene["checkpoint_colors"]]
    start_end = int(scene["start_end"])
    width = int(scene["width"])
    height = int(scene["height"])
    if width != WIDTH or height != HEIGHT or len(points) < 20:
        raise ValueError("invalid canvas or route")
    if len(indices) != 4 or len(colors) != 4 or set(colors) != set(CHECKPOINTS):
        raise ValueError("scene must contain the four checkpoint colors exactly once")
    if start_end not in (0, 1) or any(index < 0 or index >= len(points) for index in indices):
        raise ValueError("invalid route station or start end")

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for index, color_name in zip(indices, colors):
        _ring(draw, tuple(points[index]), 15, CHECKPOINT_RGB[color_name], 8)
    _ring(draw, tuple(points[0] if start_end == 0 else points[-1]), 18, START, 10)
    draw.line([tuple(point) for point in points], fill=ROUTE, width=9, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    route_length = len(scene["route_points"])
    indices = [int(value) for value in scene["checkpoint_indices"]]
    colors = [str(value) for value in scene["checkpoint_colors"]]
    start_end = int(scene["start_end"])
    if route_length < 2 or len(indices) != 4 or len(colors) != 4 or start_end not in (0, 1):
        raise ValueError("invalid analytic scene")
    order = sorted(range(4), key=lambda item: indices[item], reverse=bool(start_end))
    return colors[order[1]]


def margin(scene: dict) -> float:
    count = len(scene["route_points"])
    indices = sorted(int(value) for value in scene["checkpoint_indices"])
    start_end = int(scene["start_end"])
    ordered = indices if start_end == 0 else list(reversed(indices))
    stations = ordered if start_end == 0 else [count - 1 - value for value in ordered]
    gaps = [stations[0], *(b - a for a, b in zip(stations, stations[1:])), count - 1 - stations[-1]]
    return float(min(gaps) * 3 - 38)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 42.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the route representation while preserving every visible fact."""

    twin = dict(scene)
    points = [list(point) for point in scene["route_points"]]
    twin["route_points"] = list(reversed(points))
    twin["checkpoint_indices"] = [len(points) - 1 - int(value) for value in scene["checkpoint_indices"]]
    twin["checkpoint_colors"] = list(scene["checkpoint_colors"])
    twin["start_end"] = 1 - int(scene["start_end"])
    return [("reverse_route_representation", twin)]
