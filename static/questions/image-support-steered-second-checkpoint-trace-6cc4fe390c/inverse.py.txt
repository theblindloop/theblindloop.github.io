"""Independent raster-only inverse arm for Second Checkpoint Trace.

The implementation knows the public drawing vocabulary, but receives only a
PIL image. It segments the route and rings, computes pixel-graph geodesic
distances from the visible blue start, and selects the second checkpoint.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ROUTE_RGB = np.array((28, 43, 63), dtype=np.int16)
START_RGB = np.array((42, 105, 210), dtype=np.int16)
COLOR_RGB = {
    "red": np.array((220, 58, 70), dtype=np.int16),
    "gold": np.array((235, 171, 24), dtype=np.int16),
    "green": np.array((42, 158, 96), dtype=np.int16),
    "purple": np.array((145, 76, 190), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 2) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _component_count(mask: np.ndarray) -> tuple[int, int]:
    visited = np.zeros(mask.shape, dtype=np.bool_)
    components = 0
    largest = 0
    height, width = mask.shape
    for y_raw, x_raw in np.argwhere(mask):
        y, x = int(y_raw), int(x_raw)
        if visited[y, x]:
            continue
        components += 1
        size = 0
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        largest = max(largest, size)
    return components, largest


def _center(mask: np.ndarray, minimum: int, maximum: int) -> tuple[float, float]:
    points = np.argwhere(mask)
    if not minimum <= len(points) <= maximum:
        raise ValueError("marker is missing, clipped, or corrupted")
    center = points.mean(axis=0)
    spread = np.ptp(points, axis=0)
    if np.any(spread < 16) or np.any(spread > 40):
        raise ValueError("marker geometry is ambiguous")
    return float(center[0]), float(center[1])


def _nearest_route(route: np.ndarray, center: tuple[float, float], radius: int = 12) -> tuple[int, int]:
    cy, cx = center
    y0, y1 = max(0, round(cy) - radius), min(route.shape[0], round(cy) + radius + 1)
    x0, x1 = max(0, round(cx) - radius), min(route.shape[1], round(cx) + radius + 1)
    points = np.argwhere(route[y0:y1, x0:x1])
    if not len(points):
        raise ValueError("marker is not centered on the route")
    points = points + np.array((y0, x0))
    squared = (points[:, 0] - cy) ** 2 + (points[:, 1] - cx) ** 2
    point = points[int(np.argmin(squared))]
    if float(np.min(squared)) > 16.0:
        raise ValueError("marker-to-route ownership is ambiguous")
    return int(point[0]), int(point[1])


def _distances(route: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = route.shape
    distance = np.full(route.shape, -1, dtype=np.int32)
    queue = deque([start])
    distance[start] = 0
    while queue:
        cy, cx = queue.popleft()
        next_distance = int(distance[cy, cx]) + 1
        for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
            if 0 <= ny < height and 0 <= nx < width and route[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = next_distance
                queue.append((ny, nx))
    return distance


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise ValueError("unexpected raster size")

    route = _mask(array, ROUTE_RGB)
    components, route_pixels = _component_count(route)
    if components != 1 or not 3800 <= route_pixels <= 7000:
        raise ValueError("the dark route is broken, missing, or malformed")

    start_center = _center(_mask(array, START_RGB), 300, 750)
    checkpoint_centers = {
        name: _center(_mask(array, rgb), 220, 600) for name, rgb in COLOR_RGB.items()
    }
    start_pixel = _nearest_route(route, start_center)
    distances = _distances(route, start_pixel)

    measured = []
    for name, center in checkpoint_centers.items():
        pixel = _nearest_route(route, center)
        value = int(distances[pixel])
        if value < 0:
            raise ValueError("checkpoint is unreachable from the visible start")
        measured.append((value, name))
    measured.sort()
    gaps = [b[0] - a[0] for a, b in zip(measured, measured[1:])]
    if len(measured) != 4 or min(gaps) < 55:
        raise ValueError("checkpoint ordering has insufficient pixel margin")
    return measured[1][1]
