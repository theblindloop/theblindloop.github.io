"""Deterministic renderer and analytic arm for Odd-Sum Constellation Reveal."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

SIDE = 5
CANVAS = 768
CELL_STEP = 7
CARD_HALF = 22
DOT_RADIUS = 2.35
FRAME = (62, 68, 78)
MARKER = (190, 40, 145)
BACKGROUND = (248, 247, 242)
PALE = (226, 229, 226)
PALE_2 = (236, 232, 222)
PALETTES = (((0, 145, 183), (224, 142, 8)), ((224, 142, 8), (0, 145, 183)))
LABELS = ("plus", "x", "box-dot")


def _templates() -> dict[str, np.ndarray]:
    plus = np.zeros((SIDE, SIDE), dtype=np.uint8)
    plus[2, :] = 1
    plus[:, 2] = 1
    xmark = np.eye(SIDE, dtype=np.uint8) | np.fliplr(np.eye(SIDE, dtype=np.uint8))
    boxdot = np.zeros((SIDE, SIDE), dtype=np.uint8)
    boxdot[1, 1:4] = 1
    boxdot[3, 1:4] = 1
    boxdot[1:4, 1] = 1
    boxdot[1:4, 3] = 1
    boxdot[2, 2] = 1
    return {"plus": plus, "x": xmark.astype(np.uint8), "box-dot": boxdot}


TEMPLATES = _templates()


def _array(mask: Iterable[Iterable[int]]) -> np.ndarray:
    value = np.asarray(mask, dtype=np.uint8)
    if value.shape != (SIDE, SIDE) or np.any((value != 0) & (value != 1)):
        raise ValueError("mask must be a binary 5 by 5 array")
    return value


def _cyclic_parity(left: np.ndarray, right: np.ndarray) -> np.ndarray:
    """Centered circular convolution over GF(2), with index 2 as zero."""
    out = np.zeros((SIDE, SIDE), dtype=np.uint8)
    for ay, ax in np.argwhere(left):
        for by, bx in np.argwhere(right):
            oy = (int(ay) + int(by) - 2) % SIDE
            ox = (int(ax) + int(bx) - 2) % SIDE
            out[oy, ox] ^= 1
    return out


def _distances(output: np.ndarray) -> list[tuple[int, str]]:
    return sorted((int(np.count_nonzero(output ^ template)), label) for label, template in TEMPLATES.items())


def _classification(output: np.ndarray) -> str:
    ranked = _distances(output)
    return "ambiguous" if ranked[0][0] == ranked[1][0] else ranked[0][1]


def _solve_right(left: np.ndarray, target: np.ndarray) -> np.ndarray | None:
    """Solve left (*) right = target over GF(2), returning None if singular."""
    matrix = np.zeros((SIDE * SIDE, SIDE * SIDE), dtype=np.uint8)
    for index in range(SIDE * SIDE):
        basis = np.zeros((SIDE, SIDE), dtype=np.uint8)
        basis.flat[index] = 1
        matrix[:, index] = _cyclic_parity(left, basis).reshape(-1)
    augmented = np.concatenate((matrix, target.reshape(-1, 1)), axis=1)
    row = 0
    for column in range(SIDE * SIDE):
        pivots = np.flatnonzero(augmented[row:, column])
        if not len(pivots):
            continue
        pivot = row + int(pivots[0])
        augmented[[row, pivot]] = augmented[[pivot, row]]
        other = np.flatnonzero(augmented[:, column])
        other = other[other != row]
        augmented[other] ^= augmented[row]
        row += 1
    if row != SIDE * SIDE:
        return None
    return augmented[:, -1].reshape(SIDE, SIDE)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = LABELS[int(seed) % len(LABELS)]
    noise_count = (int(seed) // len(LABELS)) % 3
    base_target = TEMPLATES[wanted]
    for _ in range(4000):
        target = base_target.copy()
        for index in rng.sample(range(SIDE * SIDE), noise_count):
            target.flat[index] ^= 1
        if _classification(target) != wanted or (_distances(target)[1][0] - _distances(target)[0][0]) <= 2:
            continue
        left = np.zeros((SIDE, SIDE), dtype=np.uint8)
        left.flat[rng.sample(range(SIDE * SIDE), 7)] = 1
        right = _solve_right(left, target)
        # Circular GF(2) convolution preserves parity for an odd-weight left
        # mask, so a one-cell target perturbation necessarily has even right
        # ink.  The 12/13 split is a difficulty cue, never an answer cue: every
        # label is crossed with every noise level by the deterministic sweep.
        wanted_right_ink = 12 if noise_count % 2 else 13
        if right is None or int(right.sum()) != wanted_right_ink:
            continue

        gx, gy = rng.randint(1, 5), rng.randint(1, 5)
        diagonal = rng.choice((-1, 1))
        slots = [((gx + 0.5) * 96, (gy + 0.5) * 96), ((gx + 1.5) * 96, (gy + 0.5 + diagonal) * 96)]
        if rng.random() < 0.5:
            slots.reverse()
        return {
            "width": CANVAS,
            "height": CANVAS,
            "a_mask": left.tolist(),
            "b_mask": right.tolist(),
            "rotation_a": rng.randrange(4),
            "rotation_b": rng.randrange(4),
            "center_a": [int(slots[0][0]), int(slots[0][1])],
            "center_b": [int(slots[1][0]), int(slots[1][1])],
            "palette_variant": rng.randrange(len(PALETTES)),
        }
    raise RuntimeError("could not sample a fixed-ink invertible constellation pair")


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def _draw_card(draw: ImageDraw.ImageDraw, center: list[int], logical: list[list[int]], rotation: int, color: tuple[int, int, int], scale: int) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        _scaled_box((cx - CARD_HALF, cy - CARD_HALF, cx + CARD_HALF, cy + CARD_HALF), scale),
        radius=4 * scale,
        fill=(255, 255, 252),
        outline=FRAME,
        width=3 * scale,
    )
    observed = np.rot90(_array(logical), k=int(rotation))
    for row, col in np.argwhere(observed):
        x = cx + (int(col) - 2) * CELL_STEP
        y = cy + (int(row) - 2) * CELL_STEP
        draw.ellipse(_scaled_box((x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS), scale), fill=color)

    corners = ((cx - 18, cy - 18), (cx - 18, cy + 18), (cx + 18, cy + 18), (cx + 18, cy - 18))
    mx, my = corners[int(rotation) % 4]
    draw.rectangle(_scaled_box((mx - 2.2, my - 2.2, mx + 2.2, my + 2.2), scale), fill=MARKER)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # Nondiagnostic pale landmarks make translation changes visibly explicit.
    for x in (width // 4, width // 2, 3 * width // 4):
        draw.line((x * scale, 0, x * scale, height * scale), fill=PALE, width=scale)
    for y in (height // 4, height // 2, 3 * height // 4):
        draw.line((0, y * scale, width * scale, y * scale), fill=PALE_2, width=scale)
    colors = PALETTES[int(scene["palette_variant"])]
    _draw_card(draw, scene["center_a"], scene["a_mask"], int(scene["rotation_a"]), colors[0], scale)
    _draw_card(draw, scene["center_b"], scene["b_mask"], int(scene["rotation_b"]), colors[1], scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    left = _array(scene["a_mask"])
    right = _array(scene["b_mask"])
    return _classification(_cyclic_parity(left, right))


def margin(scene: dict) -> float:
    output = _cyclic_parity(_array(scene["a_mask"]), _array(scene["b_mask"]))
    ranked = _distances(output)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "ambiguous" or margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
