"""Independent final-PNG inverse for Odd-Sum Constellation Reveal."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

SIDE = 5
FRAME_RGB = np.asarray((62, 68, 78), dtype=np.int16)
MARKER_RGB = np.asarray((190, 40, 145), dtype=np.int16)
INK_RGBS = (np.asarray((0, 145, 183), dtype=np.int16), np.asarray((224, 142, 8), dtype=np.int16))


def _components(mask: np.ndarray, minimum: int = 1) -> list[tuple[int, int, int, int, int, float, float]]:
    points = {(int(y), int(x)) for y, x in np.argwhere(mask)}
    output = []
    while points:
        first = points.pop()
        stack = [first]
        component = [first]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in points:
                    points.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        if len(component) >= minimum:
            ys = [p[0] for p in component]
            xs = [p[1] for p in component]
            output.append((min(xs), min(ys), max(xs), max(ys), len(component), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _near(array: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    return np.linalg.norm(array.astype(np.int16) - color, axis=2) < tolerance


def _templates() -> dict[str, np.ndarray]:
    plus = np.zeros((SIDE, SIDE), dtype=np.uint8)
    plus[2, :] = 1
    plus[:, 2] = 1
    xmark = np.eye(SIDE, dtype=np.uint8) | np.fliplr(np.eye(SIDE, dtype=np.uint8))
    boxdot = np.zeros((SIDE, SIDE), dtype=np.uint8)
    boxdot[1, 1:4] = 1
    boxdot[3, 1:4] = 1
    boxdot[1:4, 1] = 1
    boxdot[1:4, 3] = 1
    boxdot[2, 2] = 1
    return {"plus": plus, "x": xmark.astype(np.uint8), "box-dot": boxdot}


def _extract_cards(array: np.ndarray) -> list[np.ndarray]:
    frame_parts = [item for item in _components(_near(array, FRAME_RGB, 38.0), 80) if 37 <= item[2] - item[0] + 1 <= 49 and 37 <= item[3] - item[1] + 1 <= 49]
    if len(frame_parts) != 2:
        raise ValueError(f"expected two intact card frames, found {len(frame_parts)}")
    marker_parts = _components(_near(array, MARKER_RGB, 45.0), 5)
    ink_mask = _near(array, INK_RGBS[0], 48.0) | _near(array, INK_RGBS[1], 48.0)
    ink_parts = _components(ink_mask, 5)
    cards = []
    for x0, y0, x1, y1, _, _, _ in sorted(frame_parts, key=lambda item: (item[6], item[5])):
        cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
        markers = [part for part in marker_parts if x0 <= part[5] <= x1 and y0 <= part[6] <= y1]
        if len(markers) != 1:
            raise ValueError("each card must contain exactly one magenta corner marker")
        mx, my = markers[0][5], markers[0][6]
        if mx < cx and my < cy:
            rotation = 0
        elif mx < cx and my > cy:
            rotation = 1
        elif mx > cx and my > cy:
            rotation = 2
        else:
            rotation = 3
        observed = np.zeros((SIDE, SIDE), dtype=np.uint8)
        dots = [part for part in ink_parts if x0 < part[5] < x1 and y0 < part[6] < y1]
        if len(dots) not in (7, 12, 13):
            raise ValueError(f"card has {len(dots)} point marks; expected 7, 12, or 13")
        for part in dots:
            col = int(round((part[5] - cx) / 7.0)) + 2
            row = int(round((part[6] - cy) / 7.0)) + 2
            if not (0 <= row < SIDE and 0 <= col < SIDE):
                raise ValueError("point mark lies outside the five-by-five lattice")
            expected_x, expected_y = cx + (col - 2) * 7.0, cy + (row - 2) * 7.0
            if math.hypot(part[5] - expected_x, part[6] - expected_y) > 1.8 or observed[row, col]:
                raise ValueError("point mark does not occupy a unique lattice cell")
            observed[row, col] = 1
        cards.append(np.rot90(observed, k=(-rotation) % 4).astype(np.uint8))
    weights = sorted(int(card.sum()) for card in cards)
    if weights not in ([7, 12], [7, 13]):
        raise ValueError("fixed-ink card pair is incomplete")
    return cards


def _cyclic_parity(left: np.ndarray, right: np.ndarray) -> np.ndarray:
    output = np.zeros((SIDE, SIDE), dtype=np.uint8)
    for ay, ax in np.argwhere(left):
        for by, bx in np.argwhere(right):
            output[(int(ay) + int(by) - 2) % SIDE, (int(ax) + int(bx) - 2) % SIDE] ^= 1
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cards = _extract_cards(array)
    output = _cyclic_parity(cards[0], cards[1])
    ranked = sorted((int(np.count_nonzero(output ^ template)), label) for label, template in _templates().items())
    if ranked[0][0] == ranked[1][0] or ranked[1][0] - ranked[0][0] <= 2:
        return "ambiguous"
    return ranked[0][1]
