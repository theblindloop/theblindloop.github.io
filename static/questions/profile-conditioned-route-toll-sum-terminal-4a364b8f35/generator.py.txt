"""Analytic sampler and renderer for a diamond-weighted rooted route tree."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
TARGET = 5
PALETTE = {
    "orange": (242, 137, 38),
    "purple": (142, 74, 196),
    "blue": (47, 112, 210),
}
DARK = (39, 45, 54)
GREEN = (35, 176, 91)
GRAY = (135, 145, 157)
CYAN = (20, 190, 205)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_slot = int(seed) % 3
    colors = ["orange", "purple", "blue"]
    rng.shuffle(colors)
    # Rotate bindings so answer identity and vertical slot both cycle over seeds.
    answer_color = colors[int(seed) % 3]
    other_colors = [c for c in colors if c != answer_color]
    slot_colors = [None, None, None]
    slot_colors[answer_slot] = answer_color
    free_slots = [i for i in range(3) if i != answer_slot]
    for slot, color in zip(free_slots, other_colors):
        slot_colors[slot] = color

    near = int(seed) % 2 == 0
    other_totals = [4, 6] if near else [3, 7]
    totals = [None, None, None]
    totals[answer_slot] = TARGET
    for slot, value in zip(free_slots, other_totals):
        totals[slot] = value

    dx = rng.randint(-12, 12)
    dy = rng.randint(-10, 10)
    root = [76 + dx, 256 + dy]
    leaf_y = [92 + dy, 256 + dy, 420 + dy]
    mid_y = [154 + dy, 256 + dy, 358 + dy]
    positions = [root]
    positions.extend([[252 + dx + rng.randint(-8, 8), y] for y in mid_y])
    positions.extend([[438 + dx, y] for y in leaf_y])
    edges = []
    bead_counts = []
    for slot in range(3):
        mid = 1 + slot
        leaf = 4 + slot
        edges.extend([[0, mid], [mid, leaf]])
        total = int(totals[slot])
        first = 1 + rng.randrange(max(1, min(3, total - 1)))
        first = min(first, total - 1)
        bead_counts.extend([first, total - first])

    return {
        "positions": positions,
        "edges": edges,
        "bead_counts": bead_counts,
        "root": 0,
        "terminal_colors": {str(4 + i): str(slot_colors[i]) for i in range(3)},
    }


def _validate_scene(scene: dict) -> None:
    positions = scene["positions"]
    edges = scene["edges"]
    counts = scene["bead_counts"]
    root = int(scene["root"])
    terminals = scene["terminal_colors"]
    if len(positions) != 7 or root >= len(positions) or len(edges) != len(counts):
        raise ValueError("invalid graph dimensions")
    if set(terminals.values()) != set(PALETTE):
        raise ValueError("terminal colors must be unique")
    for a, b in edges:
        if not (0 <= int(a) < len(positions) and 0 <= int(b) < len(positions)):
            raise ValueError("invalid edge")
        if math.dist(positions[int(a)], positions[int(b)]) < 90:
            raise ValueError("edge too short")


def _path_totals(scene: dict) -> dict[str, int]:
    positions = scene["positions"]
    edges = scene["edges"]
    counts = scene["bead_counts"]
    root = int(scene["root"])
    terminals = scene["terminal_colors"]
    if not positions:
        raise ValueError("positions missing")
    graph = {i: [] for i in range(len(positions))}
    for (a, b), count in zip(edges, counts):
        graph[int(a)].append((int(b), int(count)))
        graph[int(b)].append((int(a), int(count)))
    out = {}
    stack = [(root, -1, 0)]
    while stack:
        node, parent, total = stack.pop()
        if str(node) in terminals:
            out[str(terminals[str(node)])] = total
        for nxt, weight in graph[node]:
            if nxt != parent:
                stack.append((nxt, node, total + weight))
    return out


def analytic_gold(scene: dict) -> str:
    _validate_scene(scene)
    totals = _path_totals(scene)
    winners = [color for color, total in totals.items() if total == TARGET]
    if len(winners) != 1:
        raise ValueError("scene has no unique exact-total terminal")
    return winners[0]


def margin(scene: dict) -> float:
    totals = _path_totals(scene)
    distances = sorted(abs(total - TARGET) for total in totals.values())
    return float(distances[1]) if len(distances) >= 2 and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        totals = _path_totals(scene)
        return sum(total == TARGET for total in totals.values()) != 1 or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    twin["bead_counts"] = list(reversed(scene["bead_counts"]))
    return [("reverse_serialized_edges", twin)]


def _diamond(draw: ImageDraw.ImageDraw, x: float, y: float, r: int = 6) -> None:
    draw.polygon([(x, y - r), (x + r, y), (x, y + r), (x - r, y)], fill=CYAN, outline=(0, 115, 130))


def render(scene: dict) -> Image.Image:
    _validate_scene(scene)
    positions = scene["positions"]
    edges = scene["edges"]
    counts = scene["bead_counts"]
    root = int(scene["root"])
    terminals = scene["terminal_colors"]
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        draw.line([tuple(positions[int(a)]), tuple(positions[int(b)])], fill=DARK, width=7)
    for (a, b), count in zip(edges, counts):
        x1, y1 = positions[int(a)]
        x2, y2 = positions[int(b)]
        vx, vy = x2 - x1, y2 - y1
        length = math.hypot(vx, vy)
        nx, ny = -vy / length, vx / length
        # Push marks away from neighboring branches rather than into the fan.
        branch_y = (y1 + y2) / 2
        offset_sign = 1 if branch_y > positions[root][1] + 20 else -1
        offset = 8 if abs(branch_y - positions[root][1]) <= 30 else 13
        for j in range(int(count)):
            t = (j + 1) / (int(count) + 1)
            x = x1 + t * vx + offset_sign * offset * nx
            y = y1 + t * vy + offset_sign * offset * ny
            _diamond(draw, x, y)
    for index, (x, y) in enumerate(positions):
        if index == root:
            color, radius = GREEN, 15
        elif str(index) in terminals:
            color, radius = PALETTE[str(terminals[str(index)])], 15
        else:
            color, radius = GRAY, 11
        draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=color, outline=DARK, width=3)
    return image
