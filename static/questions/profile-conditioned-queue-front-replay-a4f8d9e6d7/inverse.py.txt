"""Independent pixel-only inverse for Queue Front Replay.

It uses only color-connected components and their relative layout. It neither
imports nor reconstructs the renderer's scene state.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[dict]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    results: list[dict] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if visited[y, x]:
            continue
        todo = deque([(y, x)])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while todo:
            cy, cx = todo.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    todo.append((ny, nx))
        if len(points) < minimum:
            continue
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        results.append(
            {
                "area": len(points),
                "x0": min(xs),
                "x1": max(xs),
                "y0": min(ys),
                "y1": max(ys),
                "cx": sum(xs) / len(xs),
                "cy": sum(ys) / len(ys),
                "points": points,
            }
        )
    return results


def _shape_name(component: dict, blue_mask: np.ndarray) -> str:
    width = component["x1"] - component["x0"] + 1
    height = component["y1"] - component["y0"] + 1
    if not 40 <= width <= 58 or not 40 <= height <= 58:
        raise ValueError("symbol component has invalid dimensions")
    center_x = int(round((component["x0"] + component["x1"]) / 2))
    center_y = int(round((component["y0"] + component["y1"]) / 2))
    if not blue_mask[center_y, center_x]:
        return "ring"
    fill = component["area"] / float(width * height)
    if fill > 0.92:
        return "square"
    if fill > 0.68:
        return "circle"
    if 0.42 < fill < 0.62:
        return "triangle"
    raise ValueError(f"ambiguous symbol fill ratio {fill:.3f}")


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    blue_mask = (blue > 125) & (blue - red > 55) & (blue - green > 35)
    green_mask = (green > 115) & (green - red > 45) & (green - blue > 35)
    red_mask = (red > 155) & (red - green > 75) & (red - blue > 70)

    symbol_components = []
    for component in _components(blue_mask, minimum=300):
        width = component["x1"] - component["x0"] + 1
        height = component["y1"] - component["y0"] + 1
        if 40 <= width <= 58 and 40 <= height <= 58:
            symbol_components.append(component)

    headers: list[tuple[str, dict]] = []
    for name, mask in (("add", green_mask), ("take", red_mask)):
        for component in _components(mask, minimum=1200):
            width = component["x1"] - component["x0"] + 1
            height = component["y1"] - component["y0"] + 1
            if width > 80 and 25 <= height <= 50:
                headers.append((name, component))
    headers.sort(key=lambda pair: pair[1]["cx"])
    if len(headers) != 6:
        raise ValueError(f"expected six operation headers, found {len(headers)}")

    header_top = min(item[1]["y0"] for item in headers)
    initial_parts = sorted(
        [component for component in symbol_components if component["cy"] < header_top],
        key=lambda component: component["cx"],
    )
    if not 2 <= len(initial_parts) <= 4:
        raise ValueError(f"invalid initial queue component count {len(initial_parts)}")
    queue = [_shape_name(component, blue_mask) for component in initial_parts]

    for kind, header in headers:
        if kind == "take":
            if queue:
                queue.pop(0)
            continue
        candidates = [
            component
            for component in symbol_components
            if component["cy"] > header["y1"]
            and abs(component["cx"] - header["cx"]) < 35
        ]
        if len(candidates) != 1:
            raise ValueError("ADD card does not contain exactly one symbol")
        queue.append(_shape_name(candidates[0], blue_mask))

    if not queue:
        raise ValueError("queue is empty after replay")
    return queue[0]
