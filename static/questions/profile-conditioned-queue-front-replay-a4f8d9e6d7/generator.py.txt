"""Deterministic Pillow renderer and analytic FIFO replay arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 600
SHAPES = ("circle", "square", "triangle", "ring")
QUESTION = (
    "ADD places a symbol at the back. TAKE removes the front. "
    "Which symbol finishes at FRONT?"
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _normal_op(value: str) -> str:
    if value in {"add", "append"}:
        return "add"
    if value in {"take", "remove"}:
        return "take"
    raise ValueError(f"unknown operation {value!r}")


def _replay(initial: list[str], operations: list[dict]) -> list[str]:
    queue = list(initial)
    for item in operations:
        kind = _normal_op(item["op"])
        if kind == "add":
            shape = item["shape"]
            if shape not in SHAPES:
                raise ValueError("invalid shape")
            queue.append(shape)
        elif queue:
            queue.pop(0)
    return queue


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 104729 + 37)
    target = SHAPES[seed % len(SHAPES)]
    for _ in range(10000):
        initial = [rng.choice(SHAPES) for _ in range(rng.choice((2, 3, 4)))]
        operations: list[dict] = []
        for _step in range(6):
            if rng.random() < 0.53:
                operations.append({"op": "add", "shape": rng.choice(SHAPES)})
            else:
                operations.append({"op": "take", "shape": None})
        kinds = [item["op"] for item in operations]
        result = _replay(initial, operations)
        if (
            result
            and result[0] == target
            and kinds.count("add") >= 2
            and kinds.count("take") >= 2
        ):
            break
    else:
        raise RuntimeError("could not sample a balanced queue scene")

    distractors = []
    for index in range(4):
        distractors.append([70 + index * 245 + rng.randint(-8, 8), 292 + rng.randint(-5, 5)])
    return {
        "initial": initial,
        "operations": operations,
        "palette_variant": (seed // 4) % 2,
        "shift_x": (-18, 0, 18)[(seed // 8) % 3],
        "distractors": distractors,
    }


def _draw_symbol(
    draw: ImageDraw.ImageDraw,
    shape: str,
    cx: int,
    cy: int,
    color: tuple[int, int, int],
    background: tuple[int, int, int],
) -> None:
    half = 25
    box = (cx - half, cy - half, cx + half, cy + half)
    if shape == "circle":
        draw.ellipse(box, fill=color)
    elif shape == "square":
        draw.rectangle(box, fill=color)
    elif shape == "triangle":
        draw.polygon([(cx, cy - half), (cx - half, cy + half), (cx + half, cy + half)], fill=color)
    elif shape == "ring":
        draw.ellipse(box, fill=color)
        draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), fill=background)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict) -> Image.Image:
    initial = scene["initial"]
    operations = scene["operations"]
    variant = int(scene["palette_variant"])
    shift_x = int(scene["shift_x"])
    distractors = scene["distractors"]
    blue = ((35, 91, 157), (66, 83, 170))[variant]
    green = (46, 145, 74)
    red = (205, 66, 61)
    ink = (32, 42, 54)
    muted = (103, 113, 126)
    card_bg = (248, 250, 252)
    pale = (232, 236, 241)
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    draw.text((48, 32), "QUEUE FRONT REPLAY", fill=ink, font=_font(29, True))
    draw.text((48, 74), QUESTION, fill=muted, font=_font(18))

    for x, y in distractors:
        draw.ellipse((x - 4 + shift_x, y - 4, x + 4 + shift_x, y + 4), fill=pale)

    left = 200 + shift_x
    right = 760 + shift_x
    draw.rounded_rectangle((left, 138, right, 260), radius=18, fill=card_bg, outline=pale, width=3)
    draw.text((left + 18, 150), "FRONT", fill=ink, font=_font(16, True))
    draw.polygon([(left + 78, 160), (left + 98, 151), (left + 98, 169)], fill=ink)
    draw.text((right - 70, 150), "BACK", fill=ink, font=_font(16, True))
    spacing = 94
    start_x = 310 + shift_x - (len(initial) - 3) * spacing // 2
    for index, shape in enumerate(initial):
        cx = start_x + index * spacing
        _draw_symbol(draw, shape, cx, 216, blue, card_bg)

    draw.text((48, 310), "READ CARDS LEFT TO RIGHT", fill=ink, font=_font(18, True))
    for index, item in enumerate(operations):
        cx = 142 + shift_x + index * 135
        x0, x1 = cx - 56, cx + 56
        draw.rounded_rectangle((x0, 350, x1, 526), radius=14, fill=card_bg, outline=pale, width=3)
        kind = _normal_op(item["op"])
        header = green if kind == "add" else red
        draw.rectangle((x0 + 3, 353, x1 - 3, 392), fill=header)
        label = "ADD" if kind == "add" else "TAKE"
        bounds = draw.textbbox((0, 0), label, font=_font(17, True))
        draw.text((cx - (bounds[2] - bounds[0]) // 2, 362), label, fill=(255, 255, 255), font=_font(17, True))
        if kind == "add":
            _draw_symbol(draw, item["shape"], cx, 456, blue, card_bg)
        else:
            draw.line((cx - 23, 456, cx + 23, 456), fill=red, width=9)
        draw.text((cx - 5, 535), str(index + 1), fill=muted, font=_font(15, True))

    draw.text((48, 570), "Green ADD = back   |   Red TAKE = remove front", fill=muted, font=_font(16))
    return image


def analytic_gold(scene: dict) -> str:
    final = _replay(scene["initial"], scene["operations"])
    if not final:
        raise ValueError("empty final queue")
    return final[0]


def margin(scene: dict) -> float:
    return 12.0


def is_quarantined(scene: dict) -> bool:
    try:
        return not bool(_replay(scene["initial"], scene["operations"]))
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twins: list[tuple[str, dict]] = []
    if len(scene["distractors"]) > 1:
        twin = deepcopy(scene)
        twin["distractors"] = list(reversed(twin["distractors"]))
        twins.append(("distractor_serialization_permutation", twin))
    for index, item in enumerate(scene["operations"]):
        twin = deepcopy(scene)
        if _normal_op(item["op"]) == "add":
            twin["operations"][index]["op"] = "append"
        else:
            twin["operations"][index]["op"] = "remove"
        twins.append(("operation_keyword_alias", twin))
        break
    return twins
