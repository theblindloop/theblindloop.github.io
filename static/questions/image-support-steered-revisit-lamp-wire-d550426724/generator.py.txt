from __future__ import annotations

import random
from PIL import Image, ImageDraw

W, H = 768, 256
CYAN = (25, 180, 205)
CORAL = (235, 100, 85)
GOLD = (235, 180, 45)
DARK = (35, 42, 52)
BG = (248, 246, 238)

def _replay(steps, initial=False):
    balance, lamp, seen = 0, bool(initial), {0}
    for step in steps:
        balance += step
        if balance in seen:
            lamp = not lamp
        seen.add(balance)
    return lamp

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Rejection balances both decisions without answer-specific rendering.
    wanted = bool(seed % 2)
    for _ in range(10000):
        steps = [rng.choice((-1, 1)) for _ in range(11)]
        if _replay(steps) == wanted:
            break
    return {
        "steps": steps,
        "initial_lamp": False,
        "route_y": rng.choice((108, 128, 148)),
        "bead_radius": rng.choice((13, 15, 17)),
        "wire_width": rng.choice((5, 6, 7)),
    }

def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene["steps"], scene["initial_lamp"]) else "no"

def margin(scene: dict) -> float:
    return float(scene["bead_radius"] - 8)

def is_quarantined(scene: dict) -> bool:
    return len(scene["steps"]) != 11 or any(x not in (-1, 1) for x in scene["steps"]) or margin(scene) <= 0

def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(img)
    y = int(scene["route_y"])
    xs = [84 + 55 * i for i in range(12)]
    # A single thin, connected support from the distant start to arrow.
    d.line([(48, y), (700, y)], fill=DARK, width=int(scene["wire_width"]))
    r = int(scene["bead_radius"])
    d.ellipse((48-r, y-r, 48+r, y+r), fill=GOLD, outline=DARK, width=3)
    # lamp glyph inside START: hollow means OFF, dark means ON.
    d.ellipse((41, y-7, 55, y+7), fill=(DARK if scene["initial_lamp"] else BG), outline=DARK, width=2)
    for x, step in zip(xs[1:], scene["steps"]):
        color = CYAN if step == 1 else CORAL
        d.ellipse((x-r, y-r, x+r, y+r), fill=color, outline=DARK, width=3)
    d.polygon([(700, y), (724, y-14), (724, y+14)], fill=DARK)
    return img

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
