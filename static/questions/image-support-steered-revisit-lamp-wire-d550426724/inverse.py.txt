from __future__ import annotations

import numpy as np
from PIL import Image

CYAN = np.array((25, 180, 205), dtype=np.int16)
CORAL = np.array((235, 100, 85), dtype=np.int16)
GOLD = np.array((235, 180, 45), dtype=np.int16)
DARK = np.array((35, 42, 52), dtype=np.int16)

def _mask(arr, color, tol=12):
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tol

def _components(mask):
    seen = np.zeros(mask.shape, bool)
    out = []
    h, w = mask.shape
    for yy, xx in zip(*np.nonzero(mask)):
        if seen[yy, xx]:
            continue
        stack, pts = [(yy, xx)], []
        seen[yy, xx] = True
        while stack:
            y, x = stack.pop(); pts.append((y, x))
            for dy, dx in ((1,0),(-1,0),(0,1),(0,-1)):
                ny, nx = y+dy, x+dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; stack.append((ny, nx))
        if len(pts) >= 80:
            out.append(pts)
    return out

def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (256, 768):
        return "abstain"
    # Integrity of the one connected thin support: a nearly full-width dark component is required.
    dark_components = _components(_mask(arr, DARK))
    if not dark_components:
        return "abstain"
    main = max(dark_components, key=len)
    xs = np.array([p[1] for p in main]); ys = np.array([p[0] for p in main])
    if xs.max() - xs.min() < 670 or ys.max() - ys.min() > 45:
        return "abstain"
    colored = _mask(arr, CYAN) | _mask(arr, CORAL)
    comps = _components(colored)
    if len(comps) != 11:
        return "abstain"
    beads = []
    for pts in comps:
        yy = int(round(sum(p[0] for p in pts) / len(pts)))
        xx = int(round(sum(p[1] for p in pts) / len(pts)))
        pix = arr[yy, xx].astype(np.int16)
        dc, dr = np.max(np.abs(pix-CYAN)), np.max(np.abs(pix-CORAL))
        if min(dc, dr) > 12:
            return "abstain"
        beads.append((xx, 1 if dc < dr else -1))
    beads.sort()
    gaps = np.diff([x for x, _ in beads])
    if len(gaps) != 10 or gaps.min() < 42 or gaps.max() > 68:
        return "abstain"
    gold = _components(_mask(arr, GOLD))
    if len(gold) != 1:
        return "abstain"
    steps = [v for _, v in beads]
    balance, lamp, seen = 0, False, {0}
    for step in steps:
        balance += step
        if balance in seen:
            lamp = not lamp
        seen.add(balance)
    return "yes" if lamp else "no"
