"""Independent pixel-only inverse arm for the constellation long axis."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

DARK_RGB = np.array((35, 43, 51), dtype=np.int32)
COLOR_RGB = {
    "red": np.array((214, 61, 68), dtype=np.int32),
    "blue": np.array((42, 105, 200), dtype=np.int32),
    "green": np.array((35, 148, 91), dtype=np.int32),
    "orange": np.array((231, 132, 38), dtype=np.int32),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = {tuple(value) for value in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        group = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    group.append(neighbor)
        groups.append(np.asarray(group, dtype=np.float64))
    return groups


def _axis_angle(xy: np.ndarray) -> float:
    centered = xy - xy.mean(axis=0)
    covariance = centered.T @ centered
    values, vectors = np.linalg.eigh(covariance)
    vector = vectors[:, int(np.argmax(values))]
    return math.atan2(float(vector[1]), float(vector[0])) % math.pi


def _angular_error(a: float, b: float) -> float:
    return abs((a - b + math.pi / 2.0) % math.pi - math.pi / 2.0)


def decision_from_image(image: Image.Image) -> str:
    """Recover dot centroids and guide orientations from RGB pixels only."""
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    dark_distance = np.sqrt(((array - DARK_RGB) ** 2).sum(axis=2))
    groups = _components(dark_distance < 18.0)
    centers = []
    for group in groups:
        height = group[:, 0].max() - group[:, 0].min() + 1
        width = group[:, 1].max() - group[:, 1].min() + 1
        if 35 <= len(group) <= 130 and width <= 13 and height <= 13:
            centers.append([float(group[:, 1].mean()), float(group[:, 0].mean())])
    if len(centers) < 12:
        raise ValueError("dark-dot evidence is missing or too sparse")
    cloud_angle = _axis_angle(np.asarray(centers, dtype=np.float64))

    scores: list[tuple[float, str]] = []
    for name, rgb in COLOR_RGB.items():
        distance = np.sqrt(((array - rgb) ** 2).sum(axis=2))
        pixels = np.argwhere(distance < 18.0)
        if len(pixels) < 120:
            raise ValueError(f"colored guide {name} is missing")
        xy = pixels[:, ::-1].astype(np.float64)
        guide_angle = _axis_angle(xy)
        scores.append((_angular_error(cloud_angle, guide_angle), name))
    scores.sort()
    if math.degrees(scores[1][0] - scores[0][0]) < 4.0:
        return "abstain"
    return scores[0][1]
