"""Deterministic renderer for the principal-axis constellation world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 247, 243)
FIELD = (255, 255, 255)
DARK = (35, 43, 51)
PALETTE = {
    "red": (214, 61, 68),
    "blue": (42, 105, 200),
    "green": (35, 148, 91),
    "orange": (231, 132, 38),
}
COLORS = tuple(PALETTE)


def _principal_angle(points: list[list[int]]) -> float:
    """Dominant centered-covariance direction, modulo pi."""
    mx = sum(p[0] for p in points) / len(points)
    my = sum(p[1] for p in points) / len(points)
    xx = sum((p[0] - mx) ** 2 for p in points)
    yy = sum((p[1] - my) ** 2 for p in points)
    xy = sum((p[0] - mx) * (p[1] - my) for p in points)
    return (0.5 * math.atan2(2.0 * xy, xx - yy)) % math.pi


def _angular_error(a: float, b: float) -> float:
    return abs((a - b + math.pi / 2.0) % math.pi - math.pi / 2.0)


def _axis_scores(scene: dict) -> list[tuple[float, str]]:
    target = _principal_angle(scene["points"])
    return sorted(
        (_angular_error(target, float(candidate["angle"])), str(candidate["color"]))
        for candidate in scene["candidates"]
    )


def _elongation(points: list[list[int]]) -> float:
    mx = sum(p[0] for p in points) / len(points)
    my = sum(p[1] for p in points) / len(points)
    xx = sum((p[0] - mx) ** 2 for p in points)
    yy = sum((p[1] - my) ** 2 for p in points)
    xy = sum((p[0] - mx) * (p[1] - my) for p in points)
    trace = xx + yy
    spread = math.sqrt((xx - yy) ** 2 + 4.0 * xy * xy)
    major = max((trace + spread) / 2.0, 1e-9)
    minor = max((trace - spread) / 2.0, 1e-9)
    return major / minor


def sample_scene(seed: int) -> dict:
    """Sample a balanced asymmetric elongated cloud and four equal-ink guides."""
    rng = random.Random(int(seed))
    theta = rng.uniform(0.0, math.pi)
    ct, st = math.cos(theta), math.sin(theta)
    cx = rng.uniform(235.0, 277.0)
    cy = rng.uniform(180.0, 214.0)
    count = 22 + (int(seed) % 7)
    points: list[list[int]] = []
    radii: list[int] = []
    attempts = 0
    while len(points) < count and attempts < 10000:
        attempts += 1
        # The non-Gaussian skew term makes mirror symmetry vanishingly unlikely.
        u = max(-2.0, min(2.0, rng.gauss(0.0, 0.85))) * 72.0
        v = max(-2.0, min(2.0, rng.gauss(0.0, 0.85))) * 22.0
        v += 0.0008 * (u * u - 3000.0) * (1.0 if (seed & 1) else -1.0)
        x = int(round(cx + u * ct - v * st))
        y = int(round(cy + u * st + v * ct))
        if not (38 <= x <= 474 and 34 <= y <= 346):
            continue
        if any((x - q[0]) ** 2 + (y - q[1]) ** 2 < 17 ** 2 for q in points):
            continue
        points.append([x, y])
        radii.append(4 + rng.randrange(2))
    if len(points) != count:
        raise RuntimeError("could not sample separated dot constellation")

    axis = _principal_angle(points)
    correct_color = COLORS[int(seed) % len(COLORS)]
    near = int(seed) % 11 == 0
    distractor_offsets = [15.0 if near else 29.0, 53.0, 79.0]
    rng.shuffle(distractor_offsets)
    angles: dict[str, float] = {}
    distractor_number = 0
    for color in COLORS:
        if color == correct_color:
            offset = rng.uniform(-1.2, 1.2)
        else:
            offset = distractor_offsets[distractor_number]
            distractor_number += 1
            if rng.randrange(2):
                offset = -offset
        angles[color] = (axis + math.radians(offset)) % math.pi

    slots = [82, 198, 314, 430]
    rng.shuffle(slots)
    candidates = [
        {"color": color, "x": slots[index], "y": 435, "angle": angles[color]}
        for index, color in enumerate(COLORS)
    ]
    return {"points": points, "radii": radii, "candidates": candidates}


def render(scene: dict) -> Image.Image:
    """Render only declared scene fields onto a fixed public visual grammar."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 18, 494, 366), radius=14, fill=FIELD, outline=(205, 208, 210), width=2)
    draw.line((28, 386, 484, 386), fill=(190, 193, 196), width=2)

    for point, radius in zip(scene["points"], scene["radii"]):
        x, y = point
        r = int(radius)
        draw.ellipse((x - r, y - r, x + r, y + r), fill=DARK)

    half_length = 31.0
    for candidate in scene["candidates"]:
        x = float(candidate["x"])
        y = float(candidate["y"])
        angle = float(candidate["angle"])
        dx = half_length * math.cos(angle)
        dy = half_length * math.sin(angle)
        color = PALETTE[str(candidate["color"])]
        draw.rounded_rectangle((x - 48, y - 40, x + 48, y + 40), radius=10, fill=FIELD, outline=(210, 212, 214), width=2)
        endpoints = (int(round(x - dx)), int(round(y - dy)), int(round(x + dx)), int(round(y + dy)))
        draw.line(endpoints, fill=color, width=7)
        for ex, ey in ((endpoints[0], endpoints[1]), (endpoints[2], endpoints[3])):
            draw.ellipse((ex - 4, ey - 4, ex + 4, ey + 4), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return _axis_scores(scene)[0][1]


def margin(scene: dict) -> float:
    scores = _axis_scores(scene)
    return math.degrees(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0 or _elongation(scene["points"]) < 2.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"].reverse()
    twin["radii"].reverse()
    twin["candidates"].reverse()
    return [("reverse_storage_order", twin)]
