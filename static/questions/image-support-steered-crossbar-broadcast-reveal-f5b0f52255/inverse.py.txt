"""Independent pixel-only inverse for the row/column broadcast world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 20:
            component = np.asarray(points, dtype=np.int32)
            span_y = int(np.ptp(component[:, 0])) + 1
            span_x = int(np.ptp(component[:, 1])) + 1
            fill = len(component) / float(span_y * span_x)
            if 5 <= span_x <= 8 and 5 <= span_y <= 8 and fill >= 0.80:
                output.append(component)
    return output


def _selected_index(array: np.ndarray, components: list[np.ndarray], vertical: bool) -> int:
    if len(components) != 5:
        raise ValueError(f"expected five detached strip cells, found {len(components)}")
    centers = []
    for points in components:
        ys, xs = points[:, 0], points[:, 1]
        if np.ptp(ys) < 4 or np.ptp(xs) < 4:
            raise ValueError("damaged strip cell")
        brightness = float(array[ys, xs].astype(np.float32).mean())
        centers.append((float(ys.mean()), float(xs.mean()), brightness))
    centers.sort(key=(lambda item: item[0]) if vertical else (lambda item: item[1]))
    orthogonal = [item[1] if vertical else item[0] for item in centers]
    if max(orthogonal) - min(orthogonal) > 3.0:
        raise ValueError("strip cells are not collinear")
    values = np.asarray([item[2] for item in centers])
    order = np.argsort(values)
    if float(values[order[1]] - values[order[0]]) < 30.0:
        raise ValueError("selected cell lacks a decisive contrast margin")
    return int(order[0])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    teal_mask = (green - red > 24) & (blue - red > 18) & (np.abs(green - blue) < 28)
    coral_mask = (red - green > 28) & (red - blue > 42) & (green > blue)
    teal = _components(teal_mask)
    coral = _components(coral_mask)
    row_index = _selected_index(array, teal, vertical=True)
    column_index = _selected_index(array, coral, vertical=False)
    mapping = {(0, 2): "top", (2, 4): "right", (4, 2): "bottom", (2, 0): "left"}
    try:
        return mapping[(row_index, column_index)]
    except KeyError as exc:
        raise ValueError("the two strips do not broadcast a supported T") from exc
