"""Deterministic latent sampler and renderer for Crossbar Broadcast Reveal."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

ORIENTATIONS = ("top", "right", "bottom", "left")
BACKGROUND = (248, 247, 243)
TEAL_PALE = (172, 224, 216)
CORAL_PALE = (247, 190, 169)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x5EEDB0A)
    cell = rng.choice((6, 7))
    gap = 1
    # Grid-cell-centered anchors keep both detached cues focal under raster ablation,
    # while these choices translate the whole local world across scenes.
    anchors = ((200, 200), (280, 200), (200, 280), (280, 280))
    badge_x, badge_y = anchors[rng.randrange(len(anchors))]
    layout = rng.randrange(2)
    contrast = 48 if int(seed) % 12 == 0 else (88, 112, 136)[rng.randrange(3)]
    distractors = []
    for _ in range(3):
        # Remote hollow marks are visibly unrelated and never use strip hues.
        x = rng.choice((55, 575)) + rng.randint(-12, 12)
        y = rng.randint(70, 570)
        r = rng.choice((4, 5, 6))
        distractors.append([x, y, r, rng.randrange(2)])
    return {
        "width": 640,
        "height": 640,
        "badge_x": badge_x,
        "badge_y": badge_y,
        "cell": cell,
        "gap": gap,
        "layout": layout,
        "orientation": ORIENTATIONS[int(seed) % 4],
        "contrast": contrast,
        "distractors": distractors,
    }


def _selected_indices(scene: dict[str, Any]) -> tuple[int, int]:
    side = scene["orientation"]
    return {
        "top": (0, 2),
        "right": (2, 4),
        "bottom": (4, 2),
        "left": (2, 0),
    }[side]


def _dark(pale: tuple[int, int, int], contrast: int) -> tuple[int, int, int]:
    return tuple(max(12, channel - contrast) for channel in pale)


def _strip_cells(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    vertical: bool,
    selected: int,
    cell: int,
    gap: int,
    pale: tuple[int, int, int],
    contrast: int,
) -> None:
    step = cell + gap
    length = 5 * cell + 4 * gap
    x0 = center[0] - (cell // 2 if vertical else length // 2)
    y0 = center[1] - (length // 2 if vertical else cell // 2)
    dark = _dark(pale, contrast)
    for index in range(5):
        x = x0 if vertical else x0 + index * step
        y = y0 + index * step if vertical else y0
        draw.rectangle((x, y, x + cell - 1, y + cell - 1), fill=dark if index == selected else pale)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    anchor = (int(scene["badge_x"]), int(scene["badge_y"]))
    offset = (160, 160)
    if int(scene["layout"]) == 0:
        vertical_center = anchor
        horizontal_center = (anchor[0] + offset[0], anchor[1] + offset[1])
    else:
        horizontal_center = anchor
        vertical_center = (anchor[0] + offset[0], anchor[1] + offset[1])
    row_index, column_index = _selected_indices(scene)
    cell, gap, contrast = int(scene["cell"]), int(scene["gap"]), int(scene["contrast"])
    _strip_cells(draw, vertical_center, True, row_index, cell, gap, TEAL_PALE, contrast)
    _strip_cells(draw, horizontal_center, False, column_index, cell, gap, CORAL_PALE, contrast)
    for x, y, radius, style in scene["distractors"]:
        color = (174, 151, 205) if int(style) == 0 else (220, 178, 76)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=2)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["orientation"])


def margin(scene: dict[str, Any]) -> float:
    return float(scene["contrast"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return int(scene["contrast"]) <= 52


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
