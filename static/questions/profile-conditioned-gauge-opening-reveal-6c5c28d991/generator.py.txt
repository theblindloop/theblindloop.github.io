"""gauge_opening_reveal: sample, render and score the latent sheet.

The sheet prints strokes of two ink widths along a 4x4 lattice.  A solid
reference bar printed below the sheet declares the sanding width: strokes
thinner than the bar are erased, strokes at least as thick as the bar survive.
The surviving strokes spell one of five named symbols.
"""
from __future__ import annotations

import random

from PIL import Image, ImageDraw

NODES = 4
EDGE_IDS = ["H%d%d" % (r, c) for r in range(4) for c in range(3)] + [
    "V%d%d" % (r, c) for r in range(3) for c in range(4)
]

SYMBOLS = {
    "comb": ["H00", "H01", "H02", "V01", "V02", "V03", "V12"],
    "hood": ["H00", "H01", "H02", "V00", "V10", "V03", "V13"],
    "jhook": ["V00", "V10", "V20", "H30", "H31", "H32", "V23"],
    "rungs": ["H10", "H11", "H12", "H30", "H31", "H32", "V21"],
    "zig": ["H00", "V01", "H11", "V12", "H22", "V23", "H32"],
}
SYMBOL_NAMES = tuple(sorted(SYMBOLS))

PALETTES = [
    {"bg": (247, 245, 239), "ink": (30, 32, 36), "mark": (212, 46, 118)},
    {"bg": (240, 243, 247), "ink": (38, 38, 40), "mark": (32, 132, 214)},
    {"bg": (245, 242, 232), "ink": (26, 28, 28), "mark": (222, 108, 24)},
    {"bg": (238, 240, 236), "ink": (34, 34, 38), "mark": (128, 52, 200)},
]

CANVAS_W = 420
CANVAS_H = 500
FIDUCIAL = 15
GAUGE_LEN = 120


def _rng(seed: int) -> random.Random:
    return random.Random(1000003 + int(seed) * 7919)


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = _rng(seed)
    symbol = SYMBOL_NAMES[seed % len(SYMBOL_NAMES)]
    others = [s for s in SYMBOL_NAMES if s != symbol]
    decoy = others[rng.randrange(len(others))]

    gauge_w = [9, 11, 13][rng.randrange(3)]
    near = (seed % 7) == 0
    dt = 1 if near else rng.choice([2, 3, 4])
    dn = 1 if (near and rng.random() < 0.5) else rng.choice([2, 3, 4])
    thick_w = gauge_w + dt
    thin_w = gauge_w - dn

    sym_edges = list(SYMBOLS[symbol])
    decoy_edges = [e for e in SYMBOLS[decoy] if e not in sym_edges]
    pool = [e for e in EDGE_IDS if e not in sym_edges and e not in decoy_edges]
    rng.shuffle(pool)
    clutter = sorted(pool[: rng.randint(2, 5)])

    step = [104, 110, 116][rng.randrange(3)]
    span = step * (NODES - 1)
    ox = (CANVAS_W - span) // 2 + rng.randint(-7, 7)
    oy = 52 + rng.randint(-7, 7)
    return {
        "symbol": symbol,
        "decoy": decoy,
        "decoy_edges": decoy_edges,
        "clutter": clutter,
        "gauge_w": gauge_w,
        "thick_w": thick_w,
        "thin_w": thin_w,
        "step": step,
        "ox": ox,
        "oy": oy,
        "palette": rng.randrange(len(PALETTES)),
        "gauge_x": (CANVAS_W - GAUGE_LEN) // 2 + rng.randint(-18, 18),
    }


def _node(scene: dict, r: int, c: int) -> tuple:
    return (scene["ox"] + c * scene["step"], scene["oy"] + r * scene["step"])


def _stroke(draw, scene: dict, edge: str, width: int, ink) -> None:
    kind, r, c = edge[0], int(edge[1]), int(edge[2])
    half = width // 2
    if kind == "H":
        x0, y0 = _node(scene, r, c)
        x1, _ = _node(scene, r, c + 1)
        draw.rectangle([x0 - half, y0 - half, x1 + half, y0 - half + width - 1], fill=ink)
    else:
        x0, y0 = _node(scene, r, c)
        _, y1 = _node(scene, r + 1, c)
        draw.rectangle([x0 - half, y0 - half, x0 - half + width - 1, y1 + half], fill=ink)


def render(scene: dict) -> Image.Image:
    pal = PALETTES[int(scene["palette"])]
    img = Image.new("RGB", (CANVAS_W, CANVAS_H), pal["bg"])
    draw = ImageDraw.Draw(img)
    for edge in list(scene["decoy_edges"]) + list(scene["clutter"]):
        _stroke(draw, scene, edge, int(scene["thin_w"]), pal["ink"])
    for edge in SYMBOLS[scene["symbol"]]:
        _stroke(draw, scene, edge, int(scene["thick_w"]), pal["ink"])
    h = FIDUCIAL // 2
    for r in (0, NODES - 1):
        for c in (0, NODES - 1):
            x, y = _node(scene, r, c)
            draw.rectangle([x - h, y - h, x + h, y + h], fill=pal["mark"])
    gx = int(scene["gauge_x"])
    gy = scene["oy"] + scene["step"] * (NODES - 1) + 56
    draw.rectangle([gx, gy, gx + GAUGE_LEN - 1, gy + int(scene["gauge_w"]) - 1], fill=pal["ink"])
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return float(min(int(scene["thick_w"]) - int(scene["gauge_w"]),
                     int(scene["gauge_w"]) - int(scene["thin_w"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list:
    a = dict(scene)
    a["clutter"] = list(reversed(list(scene["clutter"])))
    b = dict(scene)
    b["decoy_edges"] = list(reversed(list(scene["decoy_edges"])))
    c = dict(scene)
    c["clutter"] = list(scene["clutter"]) + list(scene["clutter"])[:1]
    return [
        ("reverse_clutter_paint_order", a),
        ("reverse_decoy_paint_order", b),
        ("repeat_one_clutter_stroke", c),
    ]
