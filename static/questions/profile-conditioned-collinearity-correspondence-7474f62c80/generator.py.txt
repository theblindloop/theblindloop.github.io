"""Collinearity-correspondence world renderer (paper-semantic family).

A cross-panel *collinearity* correspondence task. Each panel shows two gray
``anchors`` and one colored ``target`` dot. The two anchors define a line. The
reference panel's target is placed either exactly ``on`` that line (collinear
with the two anchors) or clearly ``off`` it (non-collinear) - randomized. Three
candidate panels A, B, C each show their own two anchors (at different positions:
a visible re-arrangement of the same relational structure) and one colored
target. Exactly one candidate reproduces the reference's collinear/non-collinear
*relationship* (its target on/off its own anchor line matches the reference's
state); the other two have the opposite state. The question asks which candidate
A, B, or C reproduces the same collinearity relationship as the reference.

The pixel-only inverse arm must, in every panel, recover the two anchors and the
colored target, reconstruct the anchor-to-anchor line, and read the target's
on/off position relative to that line, then match the candidate whose relation
equals the reference's - a genuine cross-panel relational correspondence.

Latent state z (every scene field is rendered - none hidden):

    panels : ordered list of four panels [reference, A, B, C]; each panel is
             {an1: [x, y], an2: [x, y], target: [x, y]}. The target position of
             each panel encodes that panel's on/off collinear state.
    width  : raster width.
    height : raster height.

The analytic gold is the label (A/B/C) of the candidate panel whose target's
collinear/non-collinear relationship to its own anchors equals the reference's.
Because panels (all anchor and target positions) are drawn verbatim, the gold is
recoverable from pixels alone; no gold-only field exists.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "collinearity_correspondence-0.1.0"

BACKGROUND = (255, 255, 255)
ANCHOR_COLOR = (95, 95, 95)
LINE_COLOR = (150, 150, 150, 90)

TARGET_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for ink/position counterbalance).
WIDTH = 900
HEIGHT = 940
BAND_CY = [160, 390, 620, 850]   # reference, A, B, C top to bottom
RADIUS = 15
ANCHOR_T = 70                    # anchors sit +/-ANCHOR_T along the panel line
TARGET_T = [-30, 0, 30]          # on-line target locations inside the anchors
OFF_DIST = 45.0                  # perpendicular offset for a non-collinear target
THRESH = 20.0                    # on/off classifier threshold (pixels)
MAX_ANGLE = 0.45                 # radians; keeps panels compact / bands separated

_MARGIN_FLOOR = 2.0


def _perp_distance(panel: dict) -> float:
    ax1, ay1 = panel["an1"]
    ax2, ay2 = panel["an2"]
    tx, ty = panel["target"]
    dx, dy = ax2 - ax1, ay2 - ay1
    length = math.hypot(dx, dy)
    if length == 0:
        return float("inf")
    num = abs(dx * (ay1 - ty) - dy * (ax1 - tx))
    return num / length


def _build_panel(rng: random.Random, yc: float, on: bool) -> dict:
    angle = rng.uniform(-MAX_ANGLE, MAX_ANGLE)
    c, s = math.cos(angle), math.sin(angle)
    ux, uy = c, s          # unit along-line direction
    nx_, ny_ = -s, c       # unit normal
    cx = WIDTH // 2
    t_tgt = rng.choice(TARGET_T)
    sign = rng.choice((-1, 1))
    on_x = cx + t_tgt * ux
    on_y = yc + t_tgt * uy
    if on:
        target = [round(on_x), round(on_y)]
    else:
        target = [round(on_x + OFF_DIST * sign * nx_), round(on_y + OFF_DIST * sign * ny_)]
    return {
        "an1": [round(cx - ANCHOR_T * ux), round(yc - ANCHOR_T * uy)],
        "an2": [round(cx + ANCHOR_T * ux), round(yc + ANCHOR_T * uy)],
        "target": target,
        "target_color": rng.choice(list(TARGET_PALETTE)),
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_on = rng.random() < 0.5
    match_slot = rng.randrange(3)
    cand_states = [not ref_on, not ref_on, not ref_on]
    cand_states[match_slot] = ref_on

    panels = [_build_panel(rng, BAND_CY[0], ref_on)]
    for i in range(3):
        panels.append(_build_panel(rng, BAND_CY[i + 1], cand_states[i]))

    return {
        "panels": panels,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_dot(draw, pos, color, radius: int = RADIUS) -> None:
    x, y = int(pos[0]), int(pos[1])
    draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=color)


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    for index, panel in enumerate(scene["panels"]):
        ax1, ay1 = panel["an1"]
        ax2, ay2 = panel["an2"]
        draw.ellipse(
            [ax1 - RADIUS, ay1 - RADIUS, ax1 + RADIUS, ay1 + RADIUS], fill=ANCHOR_COLOR
        )
        draw.ellipse(
            [ax2 - RADIUS, ay2 - RADIUS, ax2 + RADIUS, ay2 + RADIUS], fill=ANCHOR_COLOR
        )
        _draw_dot(draw, panel["target"], TARGET_PALETTE[panel["target_color"]])
    return img


def _state(panel: dict) -> str:
    return "on" if _perp_distance(panel) < THRESH else "off"


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate panel whose target on/off
    relationship to its own anchors matches the reference panel's state."""
    panels = scene["panels"]
    distances = []
    for panel in panels:
        ax1, ay1 = panel["an1"]
        ax2, ay2 = panel["an2"]
        tx, ty = panel["target"]
        dx = ax2 - ax1
        dy = ay2 - ay1
        length = math.hypot(dx, dy)
        distances.append(abs(dx * (ay1 - ty) - dy * (ax1 - tx)) / length)
    states = ["on" if d < THRESH else "off" for d in distances]
    ref_state = states[0]
    for i in range(1, 4):
        if states[i] == ref_state:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference collinear relationship")


def margin(scene: dict) -> float:
    """One-sided verifiability margin: minimum, over every panel, of the distance
    from that panel's target to the on/off classification threshold boundary."""
    distances = [_perp_distance(p) for p in scene["panels"]]
    return float(min(abs(abs(d) - THRESH) for d in distances))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous scene. With targets placed exactly on
    the line (distance ~0) or at +/-OFF_DIST (45), the margin is always large, so
    ambiguous scenes are never produced by the sampler."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (all anchor and target
    positions) is rendered verbatim, so the raster uniquely determines each
    panel's on/off state and the matching candidate; no pixel-identical transform
    is declared."""
    return []
