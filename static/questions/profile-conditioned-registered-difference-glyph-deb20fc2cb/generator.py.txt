"""Deterministic renderer and analytic gold for a registered raster difference."""

from __future__ import annotations

import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = (600, 380)
PANEL_SIZE = 184
INNER_SIZE = 144
INNER_OFFSET = 20
TILE_SIZE = 16
TARGET_INK = 1400
SYMBOLS = ("triangle", "ring", "cross", "zigzag")
FIDUCIAL = (28, 180, 98)
REFERENCE_MARK = (42, 104, 224)
TRANSFORM_MARK = (232, 122, 28)


def _segment_distance(x: np.ndarray, y: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> np.ndarray:
    ax, ay = a
    bx, by = b
    vx = bx - ax
    vy = by - ay
    denom = vx * vx + vy * vy
    t = ((x - ax) * vx + (y - ay) * vy) / denom
    t = np.clip(t, 0.0, 1.0)
    dx = x - (ax + t * vx)
    dy = y - (ay + t * vy)
    return np.hypot(dx, dy)


def _glyph_mask(symbol: str, dx: int, dy: int) -> np.ndarray:
    """Make an equal-area abstract glyph; translation is a nuisance, not its label."""

    yy, xx = np.indices((INNER_SIZE, INNER_SIZE), dtype=np.float64)
    x = xx - (INNER_SIZE / 2.0 + dx)
    y = yy - (INNER_SIZE / 2.0 + dy)
    if symbol == "ring":
        score = np.abs(np.hypot(x, y) - 28.0) + 0.012 * np.hypot(x, y)
    elif symbol == "cross":
        radius = np.maximum(np.abs(x), np.abs(y))
        score = np.minimum(np.abs(x), np.abs(y)) + 0.025 * radius
        score = np.where(radius <= 40.0, score, score + 100.0)
    elif symbol == "triangle":
        top, bottom = -36.0, 36.0
        progress = np.clip((y - top) / (bottom - top), 0.0, 1.0)
        half_width = 3.0 + 34.0 * progress
        inside = (y >= top) & (y <= bottom) & (np.abs(x) <= half_width)
        score = np.where(
            inside,
            0.055 * np.abs(x) + 0.014 * np.abs(y),
            100.0 + np.abs(y) + 0.1 * np.abs(x),
        )
    elif symbol == "zigzag":
        points = [(-38.0, -25.0), (-13.0, 25.0), (13.0, -25.0), (38.0, 25.0)]
        score = np.minimum.reduce(
            [_segment_distance(x, y, a, b) for a, b in zip(points, points[1:])]
        )
        score = np.where((np.abs(x) <= 44.0) & (np.abs(y) <= 38.0), score, score + 100.0)
        score = score + 0.01 * np.hypot(x, y)
    else:
        raise ValueError(f"unknown symbol: {symbol}")
    order = np.lexsort((xx.ravel(), yy.ravel(), score.ravel()))
    mask = np.zeros(INNER_SIZE * INNER_SIZE, dtype=bool)
    mask[order[:TARGET_INK]] = True
    return mask.reshape((INNER_SIZE, INNER_SIZE))


def _carrier(texture_seed: int, phase: int) -> np.ndarray:
    rng = np.random.default_rng(int(texture_seed))
    tile = rng.integers(68, 148, size=(TILE_SIZE, TILE_SIZE), dtype=np.uint8)
    yy, xx = np.indices((INNER_SIZE, INNER_SIZE))
    phase = int(phase) % TILE_SIZE
    return tile[(yy + phase) % TILE_SIZE, (xx + 3 * phase) % TILE_SIZE]


def _panel_arrays(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    base = _carrier(scene["texture_seed"], scene["texture_phase"])
    modified = base.copy()
    mask = _glyph_mask(scene["symbol"], scene["symbol_dx"], scene["symbol_dy"])
    boosted = modified.astype(np.int16)
    boosted[mask] = np.clip(boosted[mask] + int(scene["contrast"]), 0, 255)
    modified = boosted.astype(np.uint8)
    transform_source = np.rot90(modified, 2)
    if int(scene["reference_side"]) == 0:
        return base, transform_source
    return transform_source, base


def _draw_panel(image: Image.Image, x: int, y: int, pixels: np.ndarray, role: str) -> None:
    draw = ImageDraw.Draw(image)
    draw.rectangle((x, y, x + PANEL_SIZE - 1, y + PANEL_SIZE - 1), fill=(235, 235, 235))
    inner = Image.fromarray(pixels, mode="L").convert("RGB")
    image.paste(inner, (x + INNER_OFFSET, y + INNER_OFFSET))
    draw.rectangle(
        (x, y, x + PANEL_SIZE - 1, y + PANEL_SIZE - 1),
        outline=(72, 72, 72),
        width=2,
    )
    corners = (
        (x + 8, y + 8),
        (x + PANEL_SIZE - 17, y + 8),
        (x + 8, y + PANEL_SIZE - 17),
        (x + PANEL_SIZE - 17, y + PANEL_SIZE - 17),
    )
    for cx, cy in corners:
        draw.rectangle((cx, cy, cx + 8, cy + 8), fill=FIDUCIAL)
    mark = REFERENCE_MARK if role == "reference" else TRANSFORM_MARK
    cx = x + PANEL_SIZE // 2 - 5
    draw.rectangle((cx, y + 6, cx + 9, y + 15), fill=mark)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    seed = int(seed)
    symbol = SYMBOLS[(3 * seed + 1) % len(SYMBOLS)]
    contrast_levels = (42, 58, 74, 90)
    contrast = contrast_levels[(seed // 2 + seed) % len(contrast_levels)]
    return {
        "seed": seed,
        "symbol": symbol,
        "texture_seed": rng.randrange(1, 2**31 - 1),
        "texture_phase": rng.randrange(TILE_SIZE),
        "contrast": contrast,
        "symbol_dx": rng.randint(-14, 14),
        "symbol_dy": rng.randint(-12, 12),
        "reference_side": rng.randrange(2),
        "pair_shift": rng.choice((-8, -4, 0, 4, 8)),
    }


def render(scene: dict) -> Image.Image:
    left_pixels, right_pixels = _panel_arrays(scene)
    image = Image.new("RGB", CANVAS_SIZE, (250, 250, 250))
    shift = int(scene["pair_shift"])
    left_x = 24 + shift
    right_x = 392 + shift
    y = 98
    reference_side = int(scene["reference_side"])
    left_role = "reference" if reference_side == 0 else "transform"
    right_role = "transform" if reference_side == 0 else "reference"
    _draw_panel(image, left_x, y, left_pixels, left_role)
    _draw_panel(image, right_x, y, right_pixels, right_role)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def margin(scene: dict) -> float:
    return float(int(scene["contrast"]) - 50)


def is_quarantined(scene: dict) -> bool:
    return int(scene["contrast"]) <= 50


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["texture_phase"] = int(scene["texture_phase"]) + TILE_SIZE
    return [("carrier_phase_period", twin)]
