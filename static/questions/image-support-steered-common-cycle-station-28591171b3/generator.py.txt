"""Deterministic scene model and raster renderer for Common Cycle Station."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


CANVAS = 512
BACKGROUND = (248, 247, 243)
NAVY = (29, 47, 67)
STATION = (218, 224, 228)
RED = (216, 55, 70)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, translated and reflected eight-station graph."""
    rng = random.Random(int(seed))
    decision = "yes" if int(seed) % 2 == 0 else "no"
    spacing = rng.choice((62, 64, 66))
    x0 = (CANVAS - 3 * spacing) // 2 + rng.randint(-6, 6)
    y0 = (CANVAS - spacing) // 2 + rng.randint(-2, 2)
    nodes = [[x0 + col * spacing, y0 + row * spacing] for row in range(2) for col in range(4)]

    # Every scene has six horizontal/vertical slots plus four rungs, exactly
    # nine of which are inked.  Equal station spacing makes class ink totals
    # identical.  In YES, the far rung is absent and every cycle uses node 1.
    # In NO, the middle lower link is absent and a right-hand cycle avoids it.
    all_edges = [
        [0, 1], [1, 2], [2, 3],
        [4, 5], [5, 6], [6, 7],
        [0, 4], [1, 5], [2, 6], [3, 7],
    ]
    omitted = [3, 7] if decision == "yes" else [5, 6]
    edges = [edge for edge in all_edges if edge != omitted]

    if rng.random() < 0.5:
        nodes = [[CANVAS - x, y] for x, y in nodes]
    if rng.random() < 0.5:
        nodes = [[x, CANVAS - y] for x, y in nodes]

    return {
        "canvas_size": CANVAS,
        "nodes": nodes,
        "edges": edges,
        "red_node": 1,
        "stroke_width": 9,
        "node_radius": 13,
    }


def _has_cycle(edges: list[list[int]], node_count: int, removed: int | None = None) -> bool:
    parent = list(range(node_count))

    def find(value: int) -> int:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    for left, right in edges:
        if left == removed or right == removed:
            continue
        a, b = find(left), find(right)
        if a == b:
            return True
        parent[a] = b
    return False


def analytic_gold(scene: dict) -> str:
    """Return whether the red station intersects every simple cycle."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    red_node = int(scene["red_node"])
    if not _has_cycle(edges, len(nodes)):
        raise ValueError("the public world requires at least one loop")
    return "yes" if not _has_cycle(edges, len(nodes), removed=red_node) else "no"


def margin(scene: dict) -> float:
    """One-sided raster clearance: shortest station gap after ink thickness."""
    nodes = scene["nodes"]
    radius = float(scene["node_radius"])
    stroke = float(scene["stroke_width"])
    nearest = min(
        math.dist(nodes[i], nodes[j])
        for i in range(len(nodes))
        for j in range(i + 1, len(nodes))
    )
    return float(nearest - 2.0 * radius - stroke)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def render(scene: dict) -> Image.Image:
    """Render from visible scene fields only, with deterministic antialiasing."""
    size = int(scene["canvas_size"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    red_node = int(scene["red_node"])
    stroke = int(scene["stroke_width"])
    radius = int(scene["node_radius"])
    scale = 3
    image = Image.new("RGB", (size * scale, size * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for left, right in edges:
        p1 = tuple(int(v * scale) for v in nodes[left])
        p2 = tuple(int(v * scale) for v in nodes[right])
        draw.line([p1, p2], fill=NAVY, width=stroke * scale)

    for index, (x, y) in enumerate(nodes):
        box = [
            (x - radius) * scale,
            (y - radius) * scale,
            (x + radius) * scale,
            (y + radius) * scale,
        ]
        fill = RED if index == red_node else STATION
        draw.ellipse(box, fill=fill, outline=NAVY, width=4 * scale)

    return image.resize((size, size), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Edge-list order is latent: same-color drawing and graph gold are invariant."""
    twin = dict(scene)
    twin["edges"] = [list(edge) for edge in reversed(scene["edges"])]
    return [("reverse_edge_storage_order", twin)]
