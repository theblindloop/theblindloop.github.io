"""Independent final-PNG inverse arm for directional lamina spread.

The oracle finds the sole saturated filled sheet without fixed coordinates,
checks that it is a broad square-bounded lamina with no enclosed erasure, then
computes second central moments directly from the foreground pixels.  It has no
access to renderer state, latent cells, metadata, filenames, or analytic gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _filled_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low
    return (chroma >= 55) & (high >= 105) & (low <= 190)


def _has_enclosed_background(mask: np.ndarray) -> bool:
    """Return true when background inside the foreground bounds cannot reach an edge."""

    background = ~mask
    height, width = background.shape
    reached = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if background[0, x]:
            reached[0, x] = True
            queue.append((0, x))
        if background[height - 1, x] and not reached[height - 1, x]:
            reached[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(height):
        if background[y, 0] and not reached[y, 0]:
            reached[y, 0] = True
            queue.append((y, 0))
        if background[y, width - 1] and not reached[y, width - 1]:
            reached[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and background[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                queue.append((ny, nx))
    return bool(np.any(background & ~reached))


def _measure(image: Image.Image) -> tuple[float, float]:
    mask = _filled_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < image.width * image.height * 0.24:
        raise ValueError("the filled lamina is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < image.width * 0.56 or height < image.height * 0.68:
        raise ValueError("the filled lamina has implausibly small extent")
    if abs(width - height) > max(3, int(0.02 * max(width, height))):
        raise ValueError("the public square bounding extent is incomplete")
    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    # A valid specimen is visibly a high-fill sheet rather than a sparse path.
    # This scene-relative ratio also makes material removal explicit: a raster
    # with a large erased patch is outside the inverse arm's support.
    if float(sub.mean()) < 0.91:
        raise ValueError("the broad lamina is materially erased")
    if _has_enclosed_background(sub):
        raise ValueError("the lamina contains an enclosed erasure")

    # Translation cancels because both quantities are central moments.  Every
    # foreground pixel is equally weighted, exactly matching uniform density.
    horizontal = float(np.var(xs.astype(np.float64)))
    vertical = float(np.var(ys.astype(np.float64)))
    if abs(horizontal - vertical) < 1e-9:
        raise ValueError("directional pixel moments tie")
    return horizontal, vertical


def decision_from_image(image: Image.Image) -> str:
    horizontal, vertical = _measure(image)
    return "horizontal" if horizontal > vertical else "vertical"
