"""Deterministic sampler, Pillow renderer, and analytic arm."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

W, H = 768, 512
NAVY = (20, 43, 69)
CYAN = (69, 214, 208)
CREAM = (243, 229, 192)
GOLD = (242, 193, 78)
TAG_RGB = {"red": (233, 93, 98), "green": (84, 182, 106), "blue": (86, 133, 216)}
TAGS = ("red", "green", "blue")


def _crossweave(source: list[list[int]]) -> list[list[int]]:
    """Outer conjunction of odd row parity and odd column parity."""
    row_odd = [sum(row) % 2 for row in source]
    col_odd = [sum(source[r][c] for r in range(5)) % 2 for c in range(5)]
    return [[int(bool(row_odd[r] and col_odd[c])) for c in range(5)] for r in range(5)]


def _equal_ink_distractor(target: list[list[int]], rng: random.Random) -> list[list[int]]:
    flat = [value for row in target for value in row]
    ones = [i for i, value in enumerate(flat) if value]
    zeros = [i for i, value in enumerate(flat) if not value]
    swaps = 2 if rng.random() < 0.6 else 3
    for one, zero in zip(rng.sample(ones, swaps), rng.sample(zeros, swaps)):
        flat[one], flat[zero] = 0, 1
    return [flat[i:i + 5] for i in range(0, 25, 5)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Rejection is deterministic and excludes nearly blank/full parity products.
    for _ in range(200):
        source = [[rng.randrange(2) for _ in range(5)] for _ in range(5)]
        target = _crossweave(source)
        if 4 <= sum(map(sum, target)) <= 18:
            break
    else:
        raise RuntimeError("could not sample a nondegenerate parity weave")
    distractor_a = _equal_ink_distractor(target, rng)
    distractor_b = _equal_ink_distractor(target, rng)
    while distractor_b == distractor_a:
        distractor_b = _equal_ink_distractor(target, rng)
    correct_index = int(seed) % 3
    candidates = [distractor_a, distractor_b]
    candidates.insert(correct_index, target)
    return {"source": source, "candidates": candidates, "tags": list(TAGS)}


def _draw_grid(draw: ImageDraw.ImageDraw, grid: list[list[int]], x: int, y: int,
               cell: int, gap: int) -> None:
    for row_index, row in enumerate(grid):
        for col_index, value in enumerate(row):
            x0 = x + col_index * (cell + gap)
            y0 = y + row_index * (cell + gap)
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1),
                           fill=CYAN if value else CREAM)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    # One thick, uninterrupted atlas surrounds every decisive tile and tag.
    draw.rounded_rectangle((16, 16, W - 17, H - 17), radius=22, fill=NAVY)
    sx, sy, sc, sg = 54, 137, 42, 6
    _draw_grid(draw, scene["source"], sx, sy, sc, sg)
    draw.rectangle((sx - 8, sy - 8, sx + 5 * sc + 4 * sg + 7,
                    sy + 5 * sc + 4 * sg + 7), outline=GOLD, width=4)
    cx, cc, cg = 452, 22, 5
    for matrix, tag, cy in zip(scene["candidates"], scene["tags"], (34, 190, 346)):
        _draw_grid(draw, matrix, cx, cy, cc, cg)
        center_y = cy + (5 * cc + 4 * cg) // 2
        draw.ellipse((638, center_y - 20, 678, center_y + 20), fill=TAG_RGB[tag])
    return image


def analytic_gold(scene: dict) -> str:
    target = _crossweave(scene["source"])
    matches = [i for i, candidate in enumerate(scene["candidates"]) if candidate == target]
    if len(matches) != 1:
        raise ValueError("scene does not contain one unique parity-weave match")
    return str(scene["tags"][matches[0]])


def margin(scene: dict) -> float:
    target = _crossweave(scene["source"])
    distances = [sum(a != b for ra, rb in zip(target, candidate) for a, b in zip(ra, rb))
                 for candidate in scene["candidates"] if candidate != target]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0 or analytic_gold(scene) not in TAGS
    except (ValueError, IndexError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
