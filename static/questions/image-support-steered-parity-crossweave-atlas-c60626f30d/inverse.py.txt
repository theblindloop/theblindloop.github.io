"""Independent PNG-only inverse based on color components and grid recovery."""

from __future__ import annotations

import numpy as np
from PIL import Image

NAVY = np.array((20, 43, 69), dtype=np.int16)
CYAN = np.array((69, 214, 208), dtype=np.int16)
CREAM = np.array((243, 229, 192), dtype=np.int16)
TAG_COLORS = {
    "red": np.array((233, 93, 98), dtype=np.int16),
    "green": np.array((84, 182, 106), dtype=np.int16),
    "blue": np.array((86, 133, 216), dtype=np.int16),
}


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x); ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _ordered_grid(items: list[tuple[float, float, int]], size: int = 5) -> list[list[int]]:
    if len(items) != size * size:
        raise ValueError(f"expected {size * size} tiles, found {len(items)}")
    ordered = sorted(items, key=lambda item: item[1])
    groups = [ordered[i * size:(i + 1) * size] for i in range(size)]
    if any(max(item[1] for item in group) - min(item[1] for item in group) > 4 for group in groups):
        raise ValueError("tile rows are not aligned")
    return [[value for _, _, value in sorted(group)] for group in groups]


def _crossweave(source: list[list[int]]) -> list[list[int]]:
    row_odd = [sum(row) % 2 for row in source]
    col_odd = [sum(source[r][c] for r in range(5)) % 2 for c in range(5)]
    return [[int(bool(row_odd[r] and col_odd[c])) for c in range(5)] for r in range(5)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy_points = np.argwhere(_mask(arr, NAVY))
    if len(navy_points) < arr.shape[0] * arr.shape[1] * 0.30:
        raise ValueError("connected atlas is absent")
    y0, x0 = navy_points.min(axis=0)
    y1, x1 = navy_points.max(axis=0)
    interior = arr[y0:y1 + 1, x0:x1 + 1]
    # The visible contract says any background-colored hole invalidates the atlas.
    if float(np.all(interior >= 245, axis=2).mean()) > 0.002:
        raise ValueError("atlas has a background-colored hole")

    tiles = []
    for value, color in ((1, CYAN), (0, CREAM)):
        for left, top, right, bottom, area in _components(_mask(arr, color), minimum=350):
            width, height = right - left + 1, bottom - top + 1
            if 20 <= width <= 44 and 20 <= height <= 44 and area >= width * height * 0.96:
                tiles.append(((left + right) / 2, (top + bottom) / 2, value, area))
    if len(tiles) != 100:
        raise ValueError(f"expected 100 visible tiles, found {len(tiles)}")
    source_items = [(x, y, value) for x, y, value, area in tiles if area > 1000]
    candidate_items = [(x, y, value) for x, y, value, area in tiles if area <= 1000]
    source = _ordered_grid(source_items)
    if len(candidate_items) != 75:
        raise ValueError("candidate tile bank is malformed")
    by_y = sorted(candidate_items, key=lambda item: item[1])
    candidates = [_ordered_grid(by_y[i * 25:(i + 1) * 25]) for i in range(3)]

    tag_positions = []
    for name, color in TAG_COLORS.items():
        found = _components(_mask(arr, color), minimum=500)
        if len(found) != 1:
            raise ValueError(f"expected one {name} tag")
        left, top, right, bottom, _ = found[0]
        tag_positions.append(((top + bottom) / 2, name))
    tags = [name for _, name in sorted(tag_positions)]
    target = _crossweave(source)
    matches = [i for i, candidate in enumerate(candidates) if candidate == target]
    if len(matches) != 1:
        raise ValueError("no unique pixel-derived parity-weave match")
    return tags[matches[0]]
