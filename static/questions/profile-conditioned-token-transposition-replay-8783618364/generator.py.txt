"""Deterministic renderer and analytic arm for Token Transposition Replay."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1024, 520
BG = ((250, 252, 253), (255, 249, 239))
PANEL_BG = ((240, 247, 247), (249, 242, 232))
INK = (35, 45, 58)
BOARD = (20, 100, 102)
CARD = (102, 82, 160)
PIP = (232, 145, 35)
RING = (212, 48, 116)
TOKEN = (118, 134, 150)

PAIRS = tuple((a, b) for a in range(1, 5) for b in range(a + 1, 5))


def _swap(slot: int, pair: list[int] | tuple[int, int]) -> int:
    a, b = int(pair[0]), int(pair[1])
    if slot == a:
        return b
    if slot == b:
        return a
    return slot


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x71A95)
    target_slot = (3 * int(seed) + 1) % 4 + 1
    swaps = [list(pair) for pair in rng.sample(PAIRS, 5)]
    rng.shuffle(swaps)
    start_slot = target_slot
    for pair in reversed(swaps):
        start_slot = _swap(start_slot, pair)
    return {
        "start_slot": start_slot,
        "swaps": swaps,
        "palette": (int(seed) // 4) % 2,
    }


def analytic_gold(scene: dict) -> str:
    slot = int(scene["start_slot"])
    for pair in scene["swaps"]:
        slot = _swap(slot, pair)
    return str(slot)


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["start_slot"]) not in range(1, 5):
            return True
        if int(scene["palette"]) not in (0, 1):
            return True
        if len(scene["swaps"]) != 5:
            return True
        normalized = []
        for pair in scene["swaps"]:
            if len(pair) != 2:
                return True
            a, b = int(pair[0]), int(pair[1])
            if a not in range(1, 5) or b not in range(1, 5) or a >= b:
                return True
            normalized.append((a, b))
        if len(set(normalized)) != 5:
            return True
    except (KeyError, TypeError, ValueError):
        return True
    return False


def margin(scene: dict) -> float:
    return 0.0 if is_quarantined(scene) else 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int) -> None:
    gap = 25
    y0 = cy - gap * (count - 1) / 2
    for index in range(count):
        y = int(round(y0 + index * gap))
        draw.ellipse((cx - 7, y - 7, cx + 7, y + 7), fill=PIP)


def _double_arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    draw.line((x0, y, x1, y), fill=INK, width=4)
    draw.polygon([(x0, y), (x0 + 10, y - 7), (x0 + 10, y + 7)], fill=INK)
    draw.polygon([(x1, y), (x1 - 10, y - 7), (x1 - 10, y + 7)], fill=INK)


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("malformed scene")
    palette = int(scene["palette"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG[palette])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((40, 27), "TOKEN TRANSPOSITIONS", fill=INK, font=font)
    draw.text((40, 50), "START: ringed token", fill=BOARD, font=font)
    draw.text((334, 50), "READ FIVE SWAP CARDS LEFT TO RIGHT", fill=INK, font=font)

    # A single connected board-colored frame exposes the four equal slot cells.
    bx, by, cell, bh = 40, 150, 64, 150
    draw.rectangle((bx, by, bx + 4 * cell, by + bh), fill=PANEL_BG[palette], outline=BOARD, width=5)
    for k in range(1, 4):
        draw.line((bx + k * cell, by, bx + k * cell, by + bh), fill=BOARD, width=3)
    for slot in range(1, 5):
        cx = bx + int((slot - 0.5) * cell)
        draw.text((cx - 3, by + 17), str(slot), fill=INK, font=font)
        draw.ellipse((cx - 13, by + 76, cx + 13, by + 102), fill=TOKEN)
        if slot == int(scene["start_slot"]):
            draw.ellipse((cx - 23, by + 66, cx + 23, by + 112), outline=RING, width=7)
            draw.ellipse((cx - 5, by + 85, cx + 5, by + 95), fill=RING)

    for index, pair in enumerate(scene["swaps"]):
        x0, y0, cw, ch = 334 + 134 * index, 106, 120, 302
        draw.rounded_rectangle(
            (x0, y0, x0 + cw, y0 + ch), radius=9,
            fill=PANEL_BG[palette], outline=CARD, width=4,
        )
        draw.text((x0 + 44, y0 + 17), f"{index + 1}", fill=INK, font=font)
        draw.text((x0 + 44, y0 + 40), "SWAP", fill=INK, font=font)
        _pips(draw, x0 + 31, y0 + 144, int(pair[0]))
        _pips(draw, x0 + 89, y0 + 144, int(pair[1]))
        _double_arrow(draw, x0 + 48, x0 + 72, y0 + 144)
        draw.text((x0 + 18, y0 + 248), "dot count", fill=INK, font=font)
        draw.text((x0 + 24, y0 + 264), "= slot", fill=INK, font=font)

    draw.text(
        (40, 454),
        "Each card exchanges the two dot-counted slots; every other token stays.",
        fill=INK,
        font=font,
    )
    return image
