"""Independent PNG-only inverse arm for Token Transposition Replay."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BOARD = np.array((20, 100, 102), dtype=np.uint8)
CARD = np.array((102, 82, 160), dtype=np.uint8)
PIP = np.array((232, 145, 35), dtype=np.uint8)
RING = np.array((212, 48, 116), dtype=np.uint8)


def _mask(rgb: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(rgb == color, axis=2)


def _bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("required visual evidence is absent")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _x_groups(xs: np.ndarray) -> list[tuple[int, int]]:
    values = sorted(set(int(x) for x in xs))
    if not values:
        return []
    groups: list[list[int]] = [[values[0]]]
    for value in values[1:]:
        if value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [(group[0], group[-1]) for group in groups]


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    """Return (area, cx, cy) for 4-connected exact-color components."""
    work = mask.copy()
    height, width = work.shape
    found: list[tuple[int, float, float]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        work[y0, x0] = False
        queue = deque([(int(y0), int(x0))])
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        found.append((len(xs), float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return found


def _swap(slot: int, a: int, b: int) -> int:
    if slot == a:
        return b
    if slot == b:
        return a
    return slot


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 350 or rgb.shape[1] < 900:
        raise ValueError("unexpected raster")

    # Infer the marked initial slot from the visible board frame and ring center.
    board_box = _bbox(_mask(rgb, BOARD))
    bx0, by0, bx1, by1 = board_box
    if bx1 - bx0 < 220 or by1 - by0 < 120:
        raise ValueError("initial board is malformed")
    ring_components = [c for c in _components(_mask(rgb, RING)) if c[0] >= 50]
    if len(ring_components) != 2:
        raise ValueError("ring evidence is not unique")
    ring_x = sum(area * cx for area, cx, _ in ring_components) / sum(c[0] for c in ring_components)
    if not (bx0 < ring_x < bx1):
        raise ValueError("ring is outside the board")
    slot = min(4, max(1, int(4 * (ring_x - bx0) / max(1, bx1 - bx0 + 1)) + 1))

    # Card outlines create five disjoint wide x-runs. Count gold components on
    # each side of each card; the inverse receives neither coordinates nor scene.
    card_mask = _mask(rgb, CARD)
    _, card_xs = np.nonzero(card_mask)
    cards = [(left, right) for left, right in _x_groups(card_xs) if right - left >= 100]
    if len(cards) != 5:
        raise ValueError("card order is ambiguous")

    pip_components = [c for c in _components(_mask(rgb, PIP)) if c[0] >= 100]
    if not 15 <= len(pip_components) <= 35:
        raise ValueError("pip evidence is malformed")
    for left, right in cards:
        center = (left + right) / 2
        local = [c for c in pip_components if left < c[1] < right]
        a = sum(1 for _, cx, _ in local if cx < center)
        b = sum(1 for _, cx, _ in local if cx > center)
        if a not in range(1, 5) or b not in range(1, 5) or a >= b or len(local) != a + b:
            raise ValueError("a swap endpoint is ambiguous")
        slot = _swap(slot, a, b)
    return str(slot)
