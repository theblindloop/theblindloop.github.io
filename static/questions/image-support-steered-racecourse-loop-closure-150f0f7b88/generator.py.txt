"""Deterministic raster world for visible racecourse closure."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640

ROUTE_PALETTES = (
    (24, 105, 214),
    (214, 58, 64),
    (31, 158, 92),
    (214, 124, 26),
    (132, 63, 187),
    (16, 153, 163),
)
BACKGROUND_PALETTES = (
    (248, 248, 245),
    (243, 247, 249),
    (250, 246, 239),
    (246, 244, 249),
)
FRAME_PALETTES = (
    (78, 84, 88),
    (93, 84, 78),
    (72, 87, 91),
    (101, 91, 107),
)
MARK_PALETTES = (
    (151, 151, 146),
    (139, 148, 151),
    (164, 145, 135),
    (143, 137, 156),
)


def _rng(seed: int, salt: int) -> random.Random:
    return random.Random(int(seed) * 1_000_003 + salt * 97_409 + 17)


def _rounded_loop_points(
    center_x: float,
    center_y: float,
    outer_width: float,
    outer_height: float,
    corner_radius: float,
    wobble: float,
) -> list[list[float]]:
    """Sample a smooth, slightly asymmetric rounded circuit centerline."""

    left = center_x - outer_width / 2.0
    right = center_x + outer_width / 2.0
    top = center_y - outer_height / 2.0
    bottom = center_y + outer_height / 2.0
    radius = min(corner_radius, outer_width / 2.0 - 18.0, outer_height / 2.0 - 18.0)
    centers = (
        (right - radius, top + radius, -90.0, 0.0),
        (right - radius, bottom - radius, 0.0, 90.0),
        (left + radius, bottom - radius, 90.0, 180.0),
        (left + radius, top + radius, 180.0, 270.0),
    )
    points: list[list[float]] = []
    for corner_index, (cx, cy, start, finish) in enumerate(centers):
        for step in range(18):
            fraction = step / 18.0
            angle = math.radians(start + (finish - start) * fraction)
            # The wobble is deterministic, small, and independent of closure.
            radial = 1.0 + wobble * math.sin((corner_index * 18 + step) * 0.71)
            points.append(
                [
                    round(cx + radius * radial * math.cos(angle), 6),
                    round(cy + radius * radial * math.sin(angle), 6),
                ]
            )
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one JSON-serializable scene with answer-independent nuisances."""

    seed = int(seed)
    label_rng = _rng(seed, 1)
    geometry_rng = _rng(seed, 2)
    nuisance_rng = _rng(seed, 3)

    # The label is drawn by a separate stream from all visible nuisance fields.
    loop_closed = label_rng.randrange(2) == 0

    outer_width = geometry_rng.uniform(520.0, 548.0)
    # A wide, low-profile circuit makes the one connected support elongated
    # rather than an isotropic ring, while still spanning most of the canvas.
    outer_height = geometry_rng.uniform(188.0, 238.0)
    center_x = geometry_rng.uniform(316.0, 324.0)
    center_y = geometry_rng.uniform(304.0, 336.0)
    corner_radius = geometry_rng.uniform(58.0, 96.0)
    wobble = geometry_rng.uniform(0.015, 0.045)
    route_points = _rounded_loop_points(
        center_x,
        center_y,
        outer_width,
        outer_height,
        corner_radius,
        wobble,
    )
    route_width = geometry_rng.uniform(9.0, 13.5)
    gap_start = nuisance_rng.randrange(len(route_points))
    gap_span = nuisance_rng.randrange(6, 10)

    neutral_marks: list[dict[str, Any]] = []
    anchors = ((39.0, 41.0), (601.0, 43.0), (42.0, 600.0), (597.0, 598.0), (320.0, 31.0))
    mark_color = MARK_PALETTES[nuisance_rng.randrange(len(MARK_PALETTES))]
    for index, (anchor_x, anchor_y) in enumerate(anchors):
        jitter = nuisance_rng.uniform(-11.0, 11.0)
        size = nuisance_rng.uniform(4.0, 8.0)
        neutral_marks.append(
            {
                "x": round(anchor_x + jitter * (1 if index % 2 == 0 else -1), 6),
                "y": round(anchor_y + nuisance_rng.uniform(-9.0, 9.0), 6),
                "size": round(size, 6),
                "kind": "dot" if index % 2 == 0 else "bar",
                "color": list(mark_color),
            }
        )

    return {
        "loop_closed": loop_closed,
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "outer_width": round(outer_width, 6),
        "outer_height": round(outer_height, 6),
        "corner_radius": round(corner_radius, 6),
        "route_points": route_points,
        "gap_start": gap_start,
        "gap_span": gap_span,
        "route_width": round(route_width, 6),
        "route_color": list(ROUTE_PALETTES[nuisance_rng.randrange(len(ROUTE_PALETTES))]),
        "frame_color": list(FRAME_PALETTES[nuisance_rng.randrange(len(FRAME_PALETTES))]),
        "frame_width": round(nuisance_rng.uniform(2.5, 4.5), 6),
        "background": list(BACKGROUND_PALETTES[nuisance_rng.randrange(len(BACKGROUND_PALETTES))]),
        "neutral_marks": neutral_marks,
        "label_offset": round(nuisance_rng.uniform(-9.0, 9.0), 6),
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)  # type: ignore[return-value]


def _scaled_point(point: list[float], scale: int) -> tuple[int, int]:
    return int(round(float(point[0]) * scale)), int(round(float(point[1]) * scale))


def _gap_edges(scene: dict[str, Any]) -> set[int]:
    count = len(scene["route_points"])
    start = int(scene["gap_start"]) % count
    span = int(scene["gap_span"])
    return {(start + offset) % count for offset in range(span)}


def _draw_route(draw: ImageDraw.ImageDraw, scene: dict[str, Any], scale: int, closed: bool) -> None:
    points = [_scaled_point(point, scale) for point in scene["route_points"]]
    count = len(points)
    skipped = set() if closed else _gap_edges(scene)
    color = tuple(int(value) for value in scene["route_color"])
    width = max(4, int(round(float(scene["route_width"]) * scale)))
    radius = width / 2.0

    for index, start in enumerate(points):
        if index in skipped:
            continue
        end = points[(index + 1) % count]
        draw.line((start, end), fill=color, width=width)
        draw.ellipse(
            (int(round(start[0] - radius)), int(round(start[1] - radius)),
             int(round(start[0] + radius)), int(round(start[1] + radius))),
            fill=color,
        )
        draw.ellipse(
            (int(round(end[0] - radius)), int(round(end[1] - radius)),
             int(round(end[0] + radius)), int(round(end[1] + radius))),
            fill=color,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one page whose decisive evidence is a single thin route."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = 4 if bool(scene["antialias"]) else 1
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer_width = float(scene["outer_width"])
    outer_height = float(scene["outer_height"])
    radius = float(scene["corner_radius"])
    frame_pad = 19.0
    frame_box = (
        cx - outer_width / 2.0 - frame_pad,
        cy - outer_height / 2.0 - frame_pad,
        cx + outer_width / 2.0 + frame_pad,
        cy + outer_height / 2.0 + frame_pad,
    )
    draw.rounded_rectangle(
        _scaled_box(frame_box, scale),
        radius=max(4, int(round((radius + frame_pad) * scale))),
        outline=tuple(scene["frame_color"]),
        width=max(2, int(round(float(scene["frame_width"]) * scale))),
    )

    # Neutral corner marks and a tiny neutral title plaque are nuisance ink.
    for mark in sorted(scene["neutral_marks"], key=lambda item: (item["x"], item["y"], item["kind"])):
        x = float(mark["x"])
        y = float(mark["y"])
        size = float(mark["size"])
        color = tuple(int(value) for value in mark["color"])
        if mark["kind"] == "dot":
            draw.ellipse(_scaled_box((x - size, y - size, x + size, y + size), scale), fill=color)
        else:
            draw.rounded_rectangle(
                _scaled_box((x - size * 1.6, y - size / 2.0, x + size * 1.6, y + size / 2.0), scale),
                radius=max(1, int(round(size * scale / 3.0))),
                fill=color,
            )
    plaque_x = cx - 26.0 + float(scene["label_offset"])
    plaque_y = cy - outer_height / 2.0 - 13.0
    plaque_color = tuple(int(value) for value in scene["frame_color"])
    for offset, length in ((0.0, 13.0), (6.0, 21.0), (12.0, 16.0)):
        draw.line(
            (int(round(plaque_x * scale)), int(round((plaque_y + offset) * scale)),
             int(round((plaque_x + length) * scale)), int(round((plaque_y + offset) * scale))),
            fill=plaque_color,
            width=max(1, int(round(2.0 * scale))),
        )

    # Read the label-bearing field in this entrypoint as well as passing it to
    # the drawing helper; the route topology is visibly controlled here.
    closed = bool(scene["loop_closed"])
    _draw_route(draw, scene, scale, closed)
    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """The sole analytic label input is the visible closure class."""

    return "yes" if bool(scene["loop_closed"]) else "no"


def margin(scene: dict[str, Any]) -> float:
    """Return a one-sided raster margin for route closure or the open gap."""

    points = scene["route_points"]
    route_width = float(scene["route_width"])
    outer_width = float(scene["outer_width"])
    outer_height = float(scene["outer_height"])
    frame_clearance = min(
        float(scene["center_x"]) - outer_width / 2.0 - 28.0,
        float(scene["width"]) - float(scene["center_x"]) - outer_width / 2.0 - 28.0,
        float(scene["center_y"]) - outer_height / 2.0 - 28.0,
        float(scene["height"]) - float(scene["center_y"]) - outer_height / 2.0 - 28.0,
    )
    inner_clearance = min(outer_width, outer_height) - 2.0 * route_width - 80.0
    if bool(scene["loop_closed"]):
        closure_clearance = inner_clearance
    else:
        count = len(points)
        start = int(scene["gap_start"]) % count
        end = (start + int(scene["gap_span"])) % count
        first = points[start]
        last = points[end]
        gap_distance = math.hypot(float(first[0]) - float(last[0]), float(first[1]) - float(last[1]))
        closure_clearance = gap_distance - route_width - 12.0
    return float(min(route_width - 5.0, inner_clearance, closure_clearance, frame_clearance))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reverse only the sorted-at-render neutral-mark serialization."""

    twin = dict(scene)
    twin["neutral_marks"] = list(reversed(scene["neutral_marks"]))
    return [("reversed_neutral_mark_serialization", twin)]
