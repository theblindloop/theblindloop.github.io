"""Independent pixel-only inverse for visible route closure."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstain(ValueError):
    """The raster does not expose one unambiguous route topology."""


def _route_mask(image: Image.Image) -> np.ndarray:
    values = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_max = values.max(axis=2)
    channel_min = values.min(axis=2)
    saturation = channel_max - channel_min
    # The route is the only high-saturation foreground. The frame, plaque,
    # marks, and every background variant are intentionally low-saturation.
    return (saturation >= 68) & (channel_max >= 125) & (channel_min <= 205)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dy and not dx:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 40:
            components.append(np.asarray(points, dtype=np.int32))
    return components


def _component_mask(points: np.ndarray, shape: tuple[int, int]) -> np.ndarray:
    mask = np.zeros(shape, dtype=bool)
    mask[points[:, 0], points[:, 1]] = True
    return mask


def _enclosed_area(component: np.ndarray) -> int:
    """Count non-route pixels not reachable from the crop boundary."""

    y_min = max(0, int(component[:, 0].min()) - 3)
    y_max = min(component.shape[0] if component.ndim == 2 else 640, int(component[:, 0].max()) + 4)
    x_min = max(0, int(component[:, 1].min()) - 3)
    x_max = min(640, int(component[:, 1].max()) + 4)
    # The bounds are replaced by the actual image-sized extrema by the caller
    # through the stored coordinates; the fixed canvas is part of this world.
    sub_shape = (y_max - y_min, x_max - x_min)
    route = np.zeros(sub_shape, dtype=bool)
    route[component[:, 0] - y_min, component[:, 1] - x_min] = True
    factor = 2
    small_height = sub_shape[0] // factor
    small_width = sub_shape[1] // factor
    if small_height < 20 or small_width < 20:
        raise OracleAbstain("route crop is below the topology visibility floor")
    route_small = route[: small_height * factor, : small_width * factor].reshape(
        small_height, factor, small_width, factor
    ).any(axis=(1, 3))
    free = ~route_small
    reachable = np.zeros_like(free, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for y in range(small_height):
        for x in (0, small_width - 1):
            if free[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    for x in range(small_width):
        for y in (0, small_height - 1):
            if free[y, x] and not reachable[y, x]:
                reachable[y, x] = True
                queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = cy + dy, cx + dx
            if 0 <= ny < small_height and 0 <= nx < small_width:
                if free[ny, nx] and not reachable[ny, nx]:
                    reachable[ny, nx] = True
                    queue.append((ny, nx))
    return int(np.count_nonzero(free & ~reachable)) * factor * factor


def decision_from_image(image: Image.Image) -> str:
    """Recover yes/no closure from the PNG without scene or renderer access."""

    rgb = image.convert("RGB")
    mask = _route_mask(rgb)
    components = _components(mask)
    if not components:
        raise OracleAbstain("no saturated route found")
    components.sort(key=len, reverse=True)
    largest = components[0]
    if len(largest) < 5000:
        raise OracleAbstain("route has too little connected ink")
    if len(components) > 1 and len(components[1]) > 0.08 * len(largest):
        raise OracleAbstain("more than one substantial route component")
    y_span = int(largest[:, 0].max() - largest[:, 0].min())
    x_span = int(largest[:, 1].max() - largest[:, 1].min())
    if min(x_span, y_span) < 140:
        raise OracleAbstain("route does not span the canvas region")
    enclosed = _enclosed_area(largest)
    crop_area = (y_span + 8) * (x_span + 8)
    if enclosed > max(1000, int(0.025 * crop_area)):
        return "yes"
    if enclosed < 200:
        return "no"
    raise OracleAbstain("enclosure is too close to the raster boundary")
