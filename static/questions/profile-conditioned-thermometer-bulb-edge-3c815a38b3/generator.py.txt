"""Deterministic thermometer renderer and analytic arm."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 480
HEIGHT = 480
AA = 3
EDGES = ("top", "right", "bottom", "left")
VECTORS = {
    "top": (0, -1),
    "right": (1, 0),
    "bottom": (0, 1),
    "left": (-1, 0),
}
RED = (215, 40, 48)


def sample_scene(seed: int) -> dict:
    """Cycle answer identity while sampling styling independently."""
    rng = random.Random(int(seed))
    body_length = rng.randint(248, 276)
    body_width = rng.randint(34, 42)
    bulb_radius = rng.randint(31, 37)
    return {
        "bulb_edge": EDGES[int(seed) % 4],
        "center_x": rng.randint(226, 254),
        "center_y": rng.randint(226, 254),
        "body_length": body_length,
        "body_width": body_width,
        "bulb_radius": bulb_radius,
        "fluid_extent": rng.randint(int(body_length * 0.56), int(body_length * 0.78)),
        "casing_tone": rng.choice([174, 186, 198, 210]),
        "background_tone": rng.choice([239, 245, 251]),
    }


def _p(point: tuple[float, float]) -> tuple[int, int]:
    return (round(point[0] * AA), round(point[1] * AA))


def _ellipse(cx: float, cy: float, radius: float) -> tuple[int, int, int, int]:
    return tuple(round(value * AA) for value in (cx - radius, cy - radius, cx + radius, cy + radius))


def render(scene: dict) -> Image.Image:
    """Draw one familiar thermometer, rotated to a cardinal direction."""
    bg = int(scene["background_tone"])
    image = Image.new("RGB", (WIDTH * AA, HEIGHT * AA), (bg, bg, bg))
    draw = ImageDraw.Draw(image)
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    length = float(scene["body_length"])
    width = int(scene["body_width"])
    radius = float(scene["bulb_radius"])
    vx, vy = VECTORS[str(scene["bulb_edge"])]
    nx, ny = -vy, vx
    bulb = (cx + vx * length / 2.0, cy + vy * length / 2.0)
    cap = (cx - vx * length / 2.0, cy - vy * length / 2.0)
    dark = (52, 58, 64)
    tone = int(scene["casing_tone"])
    casing = (tone, tone + 3, tone + 6)
    well = (251, 251, 247)

    # A dark outer silhouette, pale well, and circular reservoir establish the object.
    draw.line((_p(cap), _p(bulb)), fill=dark, width=(width + 10) * AA)
    draw.ellipse(_ellipse(*cap, (width + 10) / 2), fill=dark)
    draw.ellipse(_ellipse(*bulb, radius + 9), fill=dark)
    draw.line((_p(cap), _p(bulb)), fill=casing, width=width * AA)
    draw.ellipse(_ellipse(*cap, width / 2), fill=casing)
    draw.ellipse(_ellipse(*bulb, radius + 3), fill=casing)
    draw.line((_p(cap), _p(bulb)), fill=well, width=max(18, width - 14) * AA)
    draw.ellipse(_ellipse(*cap, max(18, width - 14) / 2), fill=well)

    # Neutral scale ticks rotate with the instrument and are identical across answers.
    for fraction in (0.18, 0.31, 0.44, 0.57, 0.70):
        px = cap[0] + vx * length * fraction
        py = cap[1] + vy * length * fraction
        inner = (px + nx * (width / 2 + 7), py + ny * (width / 2 + 7))
        tick_len = 14 if fraction in (0.31, 0.57) else 10
        outer = (px + nx * (width / 2 + 7 + tick_len), py + ny * (width / 2 + 7 + tick_len))
        draw.line((_p(inner), _p(outer)), fill=dark, width=3 * AA)

    # The thin fluid path and wide reservoir are one connected red region.
    fluid_end = (
        bulb[0] - vx * float(scene["fluid_extent"]),
        bulb[1] - vy * float(scene["fluid_extent"]),
    )
    tube_width = max(8, width // 4)
    draw.line((_p(bulb), _p(fluid_end)), fill=RED, width=tube_width * AA)
    draw.ellipse(_ellipse(*fluid_end, tube_width / 2), fill=RED)
    draw.ellipse(_ellipse(*bulb, radius), fill=RED)
    # Small fixed highlight prevents the reservoir from being a featureless disk.
    glint = max(5, int(radius // 5))
    draw.ellipse(
        _ellipse(bulb[0] - nx * radius * 0.28 - vx * radius * 0.18,
                 bulb[1] - ny * radius * 0.28 - vy * radius * 0.18, glint),
        fill=(250, 178, 181),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["bulb_edge"])


def margin(scene: dict) -> float:
    """Winning nearest-edge score minus the best runner-up score in pixels."""
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    length = float(scene["body_length"])
    vx, vy = VECTORS[str(scene["bulb_edge"])]
    bx, by = cx + vx * length / 2.0, cy + vy * length / 2.0
    scores = {
        "top": HEIGHT - by,
        "right": bx,
        "bottom": by,
        "left": WIDTH - bx,
    }
    ordered = sorted(scores.values(), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return bool(
        float(scene["bulb_radius"]) < 24
        or float(scene["bulb_radius"]) < float(scene["body_width"]) * 0.68
        or margin(scene) < 45.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
