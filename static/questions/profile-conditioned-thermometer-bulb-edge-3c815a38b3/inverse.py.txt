"""Independent PNG-only inverse arm for the thermometer-bulb question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_RED = np.array((215, 40, 48), dtype=np.int32)
_MAX_COLOR_DISTANCE = 62.0
_MIN_AREA = 1800
_MIN_BULB_WIDTH = 34


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return a mask for the largest 4-connected foreground component."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y0, x0 in zip(*np.nonzero(mask), strict=True):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) > len(best):
            best = pixels
    if len(best) < _MIN_AREA:
        raise ValueError("missing or too-small connected red thermometer fluid")
    result = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*best, strict=True)
    result[np.asarray(ys), np.asarray(xs)] = True
    return result


def decision_from_image(image: Image.Image) -> str:
    """Find the thick reservoir end using only red-component projections."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    distance = np.sqrt(np.sum((arr - _RED) ** 2, axis=2))
    component = _largest_component(distance <= _MAX_COLOR_DISTANCE)
    ys, xs = np.nonzero(component)
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    span_x, span_y = x1 - x0 + 1, y1 - y0 + 1
    if max(span_x, span_y) < 2.35 * min(span_x, span_y):
        raise ValueError("red component has no unambiguous long thermometer axis")

    if span_y > span_x:
        profile = component[:, x0:x1 + 1].sum(axis=1).astype(float)
        positive = profile[profile > 0]
        baseline = float(np.median(positive))
        wide = np.flatnonzero(profile >= max(_MIN_BULB_WIDTH, baseline * 2.2))
        if len(wide) < 12:
            raise ValueError("no sufficiently wide round red bulb is visible")
        bulb_coordinate = float(np.average(wide, weights=profile[wide]))
        midpoint = (y0 + y1) / 2.0
        separation = abs(bulb_coordinate - midpoint)
        if separation < 35.0:
            raise ValueError("bulb endpoint is vertically ambiguous")
        return "top" if bulb_coordinate < midpoint else "bottom"

    profile = component[y0:y1 + 1, :].sum(axis=0).astype(float)
    positive = profile[profile > 0]
    baseline = float(np.median(positive))
    wide = np.flatnonzero(profile >= max(_MIN_BULB_WIDTH, baseline * 2.2))
    if len(wide) < 12:
        raise ValueError("no sufficiently wide round red bulb is visible")
    bulb_coordinate = float(np.average(wide, weights=profile[wide]))
    midpoint = (x0 + x1) / 2.0
    separation = abs(bulb_coordinate - midpoint)
    if separation < 35.0:
        raise ValueError("bulb endpoint is horizontally ambiguous")
    return "left" if bulb_coordinate < midpoint else "right"
