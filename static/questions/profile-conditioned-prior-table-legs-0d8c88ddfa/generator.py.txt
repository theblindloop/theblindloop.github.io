"""Prior-conflict table world: count the visible legs.

The latent scene holds a minimal line-style table with ``n`` legs supporting a
tabletop. A familiar canonical prior asserts a table has FOUR legs; the
counterfactual renders a non-canonical count and the decision must be read from
the visible legs alone, never from the prior. The pixel oracle counts the actual
leg components.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in the tabletop geometry (width, vertical position, thickness) or in
the discrete visible leg set (count, spacing, jitter), so the rendering is
injective in the latent scene and there is no render-invariant alias. The
canonical-prior name "table" is a matched-control/counterfactual manipulation
of the *question*, not of any latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0

# Supported off-quarantine counts. The canonical prior is 4; 5 and 6 are the
# counterfactuals that contradict it. Each count keeps adjacent legs separated
# by a clear horizontal gap so every leg is its own discrete blob.
SUPPORTED_N = (4, 5, 6)

# A count is quarantined when adjacent legs crowd so closely they can touch and
# be miscounted at pixel scale. For the sampled ranges the minimum clear gap is
# always well above this threshold, so supported counts are never ambiguous.
QUAR_MIN_GAP_PX = 12.0

BACKGROUND = (255, 255, 255)
TABLETOP_COLOR = (172, 138, 96)   # warm wood tabletop
LEG_COLOR = (58, 46, 34)          # dark legs (the counted object)


def sample_scene(seed: int) -> dict:
    """Deterministic scene rgba from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # Cheap-cue counterbalance: tabletop width, vertical position, thickness,
    # leg height, leg width and per-leg jitter all vary independently of n so
    # that neither position nor size predicts the count.
    top_width = rng.uniform(340.0, 430.0)
    top_x0 = CX - top_width / 2.0 + rng.uniform(-24.0, 24.0)
    top_y0 = rng.uniform(120.0, 190.0)
    top_thick = rng.uniform(18.0, 30.0)
    leg_w = rng.uniform(30.0, 38.0)
    leg_h = rng.uniform(150.0, 210.0)
    floor_y = top_y0 + top_thick + leg_h
    # Evenly distribute n legs across the tabletop with a per-leg horizontal
    # jitter that never erases the inter-leg clear gap.
    leg_px = []
    slot = (top_width - n * leg_w) / (n + 1)
    for i in range(n):
        left = top_x0 + slot + i * (leg_w + slot)
        left += rng.uniform(-3.0, 3.0)
        leg_px.append(round(left, 3))
    gap = slot - 6.0
    return {
        "n": n,
        "canvas": CANVAS,
        "top_x0": round(top_x0, 3),
        "top_width": round(top_width, 3),
        "top_y0": round(top_y0, 3),
        "top_thick": round(top_thick, 3),
        "leg_w": round(leg_w, 3),
        "leg_h": round(leg_h, 3),
        "floor_y": round(floor_y, 3),
        "leg_px": leg_px,
        "min_gap": round(max(gap, 0.0), 3),
    }


def render(scene: dict) -> Image.Image:
    """Render the table (tabletop + n legs) to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    top_x0 = float(scene["top_x0"])
    top_y0 = float(scene["top_y0"])
    top_thick = float(scene["top_thick"])
    floor_y = float(scene["floor_y"])
    draw.rectangle(
        [top_x0, top_y0, top_x0 + float(scene["top_width"]), top_y0 + top_thick],
        fill=TABLETOP_COLOR,
    )
    leg_w = float(scene["leg_w"])
    n = int(scene["n"])
    leg_px = list(scene["leg_px"])
    if len(leg_px) != n:
        raise ValueError("leg_px length must equal n")
    for left in leg_px:
        draw.rectangle([left, top_y0 + top_thick, left + leg_w, floor_y], fill=LEG_COLOR)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible count of legs, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = minimum clear horizontal gap between adjacent legs."""
    return float(scene["min_gap"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine counts whose legs are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_GAP_PX


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
