"""Renderer and analytic gold for the folded-wire echo-run world.

One thin wire is folded into a staircase of eight straight runs and confined to
a small randomly placed inset of a large plain canvas. One end of the wire
carries a solid square block. Walking the wire from the blocked end, exactly one
of runs three to six repeats the length of the run immediately before it.

Everything is drawn with integer rectangles and no antialiasing so that the
analytic run lengths and the pixel-measured run lengths are the same integers.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

RENDERER_VERSION = "wire_walk_echo_search-renderer-0.1.0"

WIDTH = 640
HEIGHT = 960
BACKGROUND = (238, 238, 234)
INKS = (
    (24, 28, 36),
    (30, 60, 110),
    (120, 30, 40),
    (28, 90, 70),
    (90, 40, 110),
    (70, 60, 20),
)
STROKE_HALF = 1           # drawn wire is three pixels thick
CAP_HALF = 5              # end block is eleven by eleven
RUN_COUNT = 8
MIN_RUN = 12
MAX_RUN = 44
VERTICAL_SPAN = 48        # keeps the folded wire inside a thin horizontal strip
BAND = HEIGHT // 8        # the wire is confined to one horizontal band of the page
MIN_SPAN_X = 104          # the strip is always clearly wider than it is tall
MAX_SPAN_X = 150
MAX_SPAN_Y = 46
BAND_INSET = 36           # keeps the strip clear of the neighbouring bands
CANVAS_MARGIN = 28
CLEAR_SPECKS = 26         # keeps stray specks off the wire component
QUARANTINE_MARGIN = 9

WORDS = ("third", "fourth", "fifth", "sixth")
FIRST_CANDIDATE_RUN = 3   # runs three to six can carry the echo


# ---------------------------------------------------------------- sampling
def _sample_lengths(rng: random.Random, echo_run: int, gap_floor: int) -> list[int]:
    """Eight run lengths in walk order; only run ``echo_run`` repeats its predecessor."""

    echo = echo_run - 1  # zero-based index of the echoing run
    for _ in range(6000):
        lengths = [rng.randrange(MIN_RUN, MAX_RUN + 1)]
        ok = True
        for index in range(1, RUN_COUNT):
            if index == echo:
                lengths.append(lengths[-1])
                continue
            for _try in range(80):
                value = rng.randrange(MIN_RUN, MAX_RUN + 1)
                if abs(value - lengths[-1]) >= gap_floor:
                    break
            else:
                ok = False
                break
            lengths.append(value)
        if not ok:
            continue
        diffs = [
            abs(lengths[i] - lengths[i - 1]) for i in range(1, RUN_COUNT) if i != echo
        ]
        smallest = min(diffs)
        if gap_floor >= QUARANTINE_MARGIN:
            if smallest < QUARANTINE_MARGIN:
                continue
        elif not 4 <= smallest <= QUARANTINE_MARGIN - 1:
            continue
        if sum(1 for i in range(1, RUN_COUNT) if lengths[i] == lengths[i - 1]) != 1:
            continue
        if lengths[echo] in (max(lengths), min(lengths)):
            continue  # the echoed length is never the longest or shortest run
        return lengths
    raise RuntimeError("length sampling failed")


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one folded wire."""

    rng = random.Random(seed * 7919 + 13)
    gap_floor = 4 if seed % 7 == 3 else QUARANTINE_MARGIN
    echo_run = FIRST_CANDIDATE_RUN + rng.randrange(len(WORDS))
    cap_at_left = bool(rng.getrandbits(1))

    # The folded wire must stay a thin horizontal strip: clearly wider than tall,
    # and short enough vertically to sit inside one horizontal band of the page.
    for _ in range(400):
        walk = _sample_lengths(rng, echo_run, gap_floor)
        geometric = list(walk) if cap_at_left else list(reversed(walk))
        points = [(0, 0)]
        x = y = 0
        for index in range(RUN_COUNT):
            run = geometric[index]
            if index % 2 == 0:
                x += run
            else:
                options = [d for d in (1, -1) if 0 <= y + d * run <= VERTICAL_SPAN]
                step = options[rng.randrange(len(options))] if options else 1
                y += step * run
            points.append((x, y))
        ys = [point[1] for point in points]
        span_x = points[-1][0]
        span_y = max(ys) - min(ys)
        if MIN_SPAN_X <= span_x <= MAX_SPAN_X and span_y <= MAX_SPAN_Y:
            break
    else:
        raise RuntimeError("geometry sampling failed")

    low = min(ys)
    off_x = rng.randrange(CANVAS_MARGIN, WIDTH - CANVAS_MARGIN - span_x)
    band = rng.randrange(HEIGHT // BAND)
    off_y = band * BAND + BAND_INSET + rng.randrange(BAND_INSET + 13 - span_y)
    vertices = [[px + off_x, py - low + off_y] for px, py in points]

    box = (
        off_x - CLEAR_SPECKS,
        off_y - CLEAR_SPECKS,
        off_x + span_x + CLEAR_SPECKS,
        off_y + span_y + CLEAR_SPECKS,
    )
    specks: list[list[int]] = []
    wanted = 4 + rng.randrange(5)
    for _ in range(600):
        if len(specks) >= wanted:
            break
        sx = rng.randrange(12, WIDTH - 12)
        sy = rng.randrange(12, HEIGHT - 12)
        if box[0] <= sx <= box[2] and box[1] <= sy <= box[3]:
            continue
        if any(abs(sx - ox) < 16 and abs(sy - oy) < 16 for ox, oy in specks):
            continue
        specks.append([sx, sy])

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "vertices": vertices,
        "cap_at_left": cap_at_left,
        "color": rng.randrange(len(INKS)),
        "specks": specks,
    }


# ---------------------------------------------------------------- geometry
def geometric_lengths(scene: dict) -> list[int]:
    """Run lengths in left-to-right drawing order."""

    points = scene["vertices"]
    return [
        abs(points[i + 1][0] - points[i][0]) + abs(points[i + 1][1] - points[i][1])
        for i in range(len(points) - 1)
    ]


def walk_lengths(scene: dict) -> list[int]:
    """Run lengths in walk order, starting from the run that touches the block."""

    lengths = geometric_lengths(scene)
    return lengths if scene["cap_at_left"] else lengths[::-1]


def _echo_index(scene: dict) -> int:
    lengths = walk_lengths(scene)
    hits = [
        run
        for run in range(FIRST_CANDIDATE_RUN, FIRST_CANDIDATE_RUN + len(WORDS))
        if lengths[run - 1] == lengths[run - 2]
    ]
    if len(hits) != 1:
        raise ValueError(f"scene has {len(hits)} echoing runs")
    return hits[0]


def analytic_gold(scene: dict) -> str:
    """Ordinal, along the walk, of the run that repeats its predecessor."""

    return WORDS[_echo_index(scene) - FIRST_CANDIDATE_RUN]


def margin(scene: dict) -> float:
    """Smallest length difference of any competing neighbouring pair, in pixels."""

    lengths = walk_lengths(scene)
    echo = _echo_index(scene) - 1
    return float(
        min(
            abs(lengths[i] - lengths[i - 1])
            for i in range(1, RUN_COUNT)
            if i != echo
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent rewrites that are byte-identical rasters with the same gold."""

    specks = scene["specks"]
    out: list[tuple[str, dict]] = []
    if len(specks) >= 2:
        out.append(("specks_reversed", {**scene, "specks": specks[::-1]}))
        out.append(("specks_rotated", {**scene, "specks": specks[1:] + specks[:1]}))
    return out


# ---------------------------------------------------------------- rendering
def render(scene: dict) -> Image.Image:
    array = np.zeros((scene["height"], scene["width"], 3), dtype=np.uint8)
    array[:, :] = np.array(BACKGROUND, dtype=np.uint8)
    ink = np.array(INKS[int(scene["color"]) % len(INKS)], dtype=np.uint8)
    points = [(int(px), int(py)) for px, py in scene["vertices"]]

    for index in range(len(points) - 1):
        (x1, y1), (x2, y2) = points[index], points[index + 1]
        if y1 == y2:
            array[y1 - STROKE_HALF : y1 + STROKE_HALF + 1, min(x1, x2) : max(x1, x2) + 1] = ink
        else:
            array[min(y1, y2) : max(y1, y2) + 1, x1 - STROKE_HALF : x1 + STROKE_HALF + 1] = ink
    for x, y in points:
        array[y - STROKE_HALF : y + STROKE_HALF + 1, x - STROKE_HALF : x + STROKE_HALF + 1] = ink

    cap_x, cap_y = points[0] if scene["cap_at_left"] else points[-1]
    array[cap_y - CAP_HALF : cap_y + CAP_HALF + 1, cap_x - CAP_HALF : cap_x + CAP_HALF + 1] = ink

    for sx, sy in scene["specks"]:
        sx, sy = int(sx), int(sy)
        array[sy - STROKE_HALF : sy + STROKE_HALF + 1, sx - STROKE_HALF : sx + STROKE_HALF + 1] = ink

    return Image.fromarray(array, mode="RGB")
