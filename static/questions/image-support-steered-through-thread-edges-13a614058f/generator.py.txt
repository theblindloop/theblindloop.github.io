"""Through-thread world: one connected thin thread joins two page sides.

A white page carries several thin dark threads of identical colour and
thickness.  Exactly one of them -- the through-thread -- runs from one side of
the page to another side.  Every other thread starts on a side and dead-ends
deep inside the page.  The number of thread ends on each of the four sides is
always exactly two, so counting ends at any single side is uninformative; the
answer is the unordered pair of sides linked by the one thread that actually
gets across, which can only be established by following a thread along its
whole length.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "through_thread_edges-renderer-0.1.0"

CANVAS = 800
GRID = 40
ORIGIN = 20
NODES = (CANVAS - 2 * ORIGIN) // GRID + 1  # 20 lattice nodes per axis
THREAD_WIDTH = 5
BACKGROUND = (255, 255, 255)
CLUTTER_COLOR = (222, 222, 222)
THREAD_PALETTE = (
    (26, 26, 26),
    (16, 42, 68),
    (40, 26, 52),
    (20, 54, 40),
)
EDGES = ("top", "bottom", "left", "right")
EDGE_PAIRS = (
    "top-bottom",
    "top-left",
    "top-right",
    "bottom-left",
    "bottom-right",
    "left-right",
)
QUARANTINE_MARGIN = 14.0
BAND = 140.0
_STEPS = ((1, 0), (-1, 0), (0, 1), (0, -1))


def _node_xy(node: tuple[int, int]) -> tuple[float, float]:
    return (float(ORIGIN + GRID * node[0]), float(ORIGIN + GRID * node[1]))


def _edge_of_node(node: tuple[int, int]) -> list[str]:
    sides = []
    if node[1] == 0:
        sides.append("top")
    if node[1] == NODES - 1:
        sides.append("bottom")
    if node[0] == 0:
        sides.append("left")
    if node[0] == NODES - 1:
        sides.append("right")
    return sides


LOW_RANGE = (3, 9)
HIGH_RANGE = (NODES - 9, NODES - 3)


def _place(side: str, inner: int) -> tuple[int, int]:
    if side == "top":
        return (inner, 0)
    if side == "bottom":
        return (inner, NODES - 1)
    if side == "left":
        return (0, inner)
    return (NODES - 1, inner)


def _side_nodes(side: str, rng: random.Random) -> tuple[int, int]:
    return _place(side, rng.randrange(3, NODES - 3))


def _far_inner(avoid: str, rng: random.Random) -> int:
    """Along-side coordinate kept away from the side we must not hug."""
    if avoid in ("left", "top"):
        return rng.randrange(*HIGH_RANGE)
    return rng.randrange(*LOW_RANGE)


def _endpoints(
    side_a: str, side_b: str, rng: random.Random
) -> tuple[tuple[int, int], tuple[int, int]]:
    opposite = {"top": "bottom", "bottom": "top", "left": "right", "right": "left"}
    if opposite[side_a] == side_b:
        low = rng.randrange(*LOW_RANGE)
        high = rng.randrange(*HIGH_RANGE)
        if rng.random() < 0.5:
            low, high = high, low
        return _place(side_a, low), _place(side_b, high)
    return _place(side_a, _far_inner(side_b, rng)), _place(side_b, _far_inner(side_a, rng))


def _stub_point(node: tuple[int, int], side: str) -> tuple[float, float]:
    x, y = _node_xy(node)
    if side == "top":
        return (x, -8.0)
    if side == "bottom":
        return (x, CANVAS + 8.0)
    if side == "left":
        return (-8.0, y)
    return (CANVAS + 8.0, y)


def _blocked(reserved: set[tuple[int, int]], node: tuple[int, int]) -> bool:
    if node in reserved:
        return True
    for dx, dy in _STEPS:
        if (node[0] + dx, node[1] + dy) in reserved:
            return True
    return False


def _reserve(reserved: set[tuple[int, int]], path: list[tuple[int, int]]) -> None:
    reserved.update(path)


def _in_bounds(node: tuple[int, int]) -> bool:
    return 0 <= node[0] < NODES and 0 <= node[1] < NODES


def _lateral(node: tuple[int, int], start: tuple[int, int], goal: tuple[int, int]) -> float:
    ax, ay = _node_xy(start)
    bx, by = _node_xy(goal)
    px, py = _node_xy(node)
    vx, vy = bx - ax, by - ay
    length = math.hypot(vx, vy)
    if length < 1e-9:
        return 0.0
    return abs((px - ax) * vy - (py - ay) * vx) / length


def _walk_to_goal(
    rng: random.Random,
    reserved: set[tuple[int, int]],
    start: tuple[int, int],
    goal: tuple[int, int],
    min_steps: int,
) -> list[tuple[int, int]] | None:
    path = [start]
    own = {start}
    for _ in range(6 * NODES):
        current = path[-1]
        if current == goal and len(path) >= min_steps:
            return path
        options = []
        for dx, dy in _STEPS:
            nxt = (current[0] + dx, current[1] + dy)
            if not _in_bounds(nxt) or nxt in own:
                continue
            if nxt != goal and _edge_of_node(nxt):
                continue
            if nxt != goal and _blocked(reserved, nxt):
                continue
            here = abs(current[0] - goal[0]) + abs(current[1] - goal[1])
            there = abs(nxt[0] - goal[0]) + abs(nxt[1] - goal[1])
            if there < here:
                weight = 6 if len(path) >= min_steps - 2 else 2
            else:
                if _lateral(nxt, start, goal) > BAND:
                    continue
                weight = 1 if len(path) >= min_steps - 2 else 4
            options.append((nxt, weight))
        if not options:
            return None
        nodes = [item[0] for item in options]
        weights = [item[1] for item in options]
        choice = rng.choices(nodes, weights=weights, k=1)[0]
        path.append(choice)
        own.add(choice)
    return None


def _walk_stub(
    rng: random.Random,
    reserved: set[tuple[int, int]],
    start: tuple[int, int],
    steps: int,
) -> list[tuple[int, int]] | None:
    path = [start]
    own = {start}
    for _ in range(steps):
        current = path[-1]
        options = []
        for dx, dy in _STEPS:
            nxt = (current[0] + dx, current[1] + dy)
            if not _in_bounds(nxt) or nxt in own:
                continue
            if _edge_of_node(nxt) or _blocked(reserved, nxt):
                continue
            options.append(nxt)
        if not options:
            return None
        path.append(rng.choice(options))
        own.add(path[-1])
    tip = path[-1]
    if not (2 <= tip[0] <= NODES - 3 and 2 <= tip[1] <= NODES - 3):
        return None
    return path


def _chaikin(points: list[tuple[float, float]], rounds: int = 2) -> list[tuple[float, float]]:
    current = list(points)
    for _ in range(rounds):
        if len(current) < 3:
            break
        nxt = [current[0]]
        for index in range(len(current) - 1):
            ax, ay = current[index]
            bx, by = current[index + 1]
            nxt.append((0.75 * ax + 0.25 * bx, 0.75 * ay + 0.25 * by))
            nxt.append((0.25 * ax + 0.75 * bx, 0.25 * ay + 0.75 * by))
        nxt.append(current[-1])
        current = nxt
    return [(round(x, 3), round(y, 3)) for x, y in current]


def _polyline(path: list[tuple[int, int]], sides: list[str]) -> list[list[float]]:
    points = [_node_xy(node) for node in path]
    if sides:
        points = [_stub_point(path[0], sides[0])] + points
    smooth = _chaikin(points)
    return [[x, y] for x, y in smooth]


def _attempt(rng: random.Random, pair: str) -> dict | None:
    side_a, side_b = pair.split("-")
    reserved: set[tuple[int, int]] = set()
    start, goal = _endpoints(side_a, side_b, rng)
    if start == goal:
        return None
    main = _walk_to_goal(rng, reserved, start, goal, rng.randrange(13, 18))
    if main is None:
        return None
    _reserve(reserved, main)
    route = _polyline(main, [side_a])
    exit_points = [_stub_point(goal, side_b)]
    route = route + [[exit_points[0][0], exit_points[0][1]]]

    quota: dict[str, int] = {side: 2 for side in EDGES}
    quota[side_a] -= 1
    quota[side_b] -= 1
    wanted: list[str] = []
    for side in EDGES:
        wanted.extend([side] * quota[side])
    rng.shuffle(wanted)
    long_index = rng.randrange(len(wanted))

    decoys: list[dict] = []
    for index, side in enumerate(wanted):
        steps = rng.randrange(len(main) + 2, len(main) + 6) if index == long_index else rng.randrange(5, 10)
        built = None
        for _ in range(40):
            node = _side_nodes(side, rng)
            if _blocked(reserved, node):
                continue
            built = _walk_stub(rng, reserved, node, steps)
            if built is not None:
                break
        if built is None:
            return None
        _reserve(reserved, built)
        decoys.append({"side": side, "points": _polyline(built, [side])})

    if decoys and rng.random() < 0.75:
        chosen = decoys[rng.randrange(len(decoys))]
        tip = chosen["points"][-1]
        gaps = {
            "left": tip[0],
            "right": CANVAS - tip[0],
            "top": tip[1],
            "bottom": CANVAS - tip[1],
        }
        near = min(gaps, key=lambda side: gaps[side])
        target = rng.uniform(10.0, 46.0)
        if gaps[near] - target > 14.0:
            if near == "left":
                reach = [round(target, 3), tip[1]]
            elif near == "right":
                reach = [round(CANVAS - target, 3), tip[1]]
            elif near == "top":
                reach = [tip[0], round(target, 3)]
            else:
                reach = [tip[0], round(CANVAS - target, 3)]
            chosen["points"] = chosen["points"] + [reach]

    clutter = []
    for _ in range(rng.randrange(9, 15)):
        radius = rng.randrange(6, 17)
        clutter.append(
            [
                rng.randrange(30, CANVAS - 30),
                rng.randrange(30, CANVAS - 30),
                radius,
            ]
        )
    return {
        "canvas": CANVAS,
        "route": route,
        "decoys": decoys,
        "clutter": clutter,
        "thread_color": list(THREAD_PALETTE[rng.randrange(len(THREAD_PALETTE))]),
        "width": THREAD_WIDTH,
    }


def sample_scene(seed: int) -> dict:
    for attempt in range(600):
        rng = random.Random((int(seed) * 7919 + attempt * 104729) % (2**63))
        pair = EDGE_PAIRS[int(seed) % len(EDGE_PAIRS)]
        scene = _attempt(rng, pair)
        if scene is None:
            continue
        if _edge_set(scene["route"]) != tuple(sorted(pair.split("-"), key=EDGES.index)):
            continue
        if margin(scene) < QUARANTINE_MARGIN:
            continue
        if _max_decoy_length(scene) <= _length(scene["route"]) * 1.02:
            continue
        if not _spans_page(scene["route"]):
            continue
        return scene
    raise RuntimeError(f"no feasible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["clutter"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=CLUTTER_COLOR)
    color = tuple(int(value) for value in scene["thread_color"])
    width = int(scene["width"])
    for decoy in scene["decoys"]:
        draw.line(
            [(float(x), float(y)) for x, y in decoy["points"]],
            fill=color,
            width=width,
            joint="curve",
        )
    draw.line(
        [(float(x), float(y)) for x, y in scene["route"]],
        fill=color,
        width=width,
        joint="curve",
    )
    return image


def _edge_set(route: list) -> tuple[str, ...]:
    sides = set()
    for x, y in route:
        if y <= 1.0:
            sides.add("top")
        if y >= CANVAS - 2.0:
            sides.add("bottom")
        if x <= 1.0:
            sides.add("left")
        if x >= CANVAS - 2.0:
            sides.add("right")
    return tuple(sorted(sides, key=EDGES.index))


def analytic_gold(scene: dict) -> str:
    sides = _edge_set(scene["route"])
    if len(sides) != 2:
        raise ValueError(f"route reaches {sides}, expected exactly two page sides")
    return "-".join(sides)


def _samples(points: list) -> list[tuple[float, float]]:
    out: list[tuple[float, float]] = []
    for index in range(len(points) - 1):
        ax, ay = float(points[index][0]), float(points[index][1])
        bx, by = float(points[index + 1][0]), float(points[index + 1][1])
        span = math.hypot(bx - ax, by - ay)
        steps = max(1, int(span / 3.0))
        for step in range(steps):
            t = step / steps
            out.append((ax + (bx - ax) * t, ay + (by - ay) * t))
    if points:
        out.append((float(points[-1][0]), float(points[-1][1])))
    return out


def _min_distance(first: list[tuple[float, float]], second: list[tuple[float, float]]) -> float:
    best = float("inf")
    for ax, ay in first:
        for bx, by in second:
            gap = (ax - bx) ** 2 + (ay - by) ** 2
            if gap < best:
                best = gap
    return math.sqrt(best)


def _length(points: list) -> float:
    total = 0.0
    for index in range(len(points) - 1):
        total += math.hypot(
            float(points[index + 1][0]) - float(points[index][0]),
            float(points[index + 1][1]) - float(points[index][1]),
        )
    return total


def _spans_page(route: list) -> bool:
    """The traced component must reach across the page on both axes, so no half
    of the picture is on its own sufficient for the inverse arm."""
    xs = [max(0.0, min(float(CANVAS), float(point[0]))) for point in route]
    ys = [max(0.0, min(float(CANVAS), float(point[1]))) for point in route]
    return (max(xs) - min(xs)) >= 0.55 * CANVAS and (max(ys) - min(ys)) >= 0.55 * CANVAS


def _max_decoy_length(scene: dict) -> float:
    return max((_length(decoy["points"]) for decoy in scene["decoys"]), default=0.0)


def margin(scene: dict) -> float:
    threads = [_samples(scene["route"])] + [
        _samples(decoy["points"]) for decoy in scene["decoys"]
    ]
    best = float("inf")
    for i in range(len(threads)):
        for j in range(i + 1, len(threads)):
            best = min(best, _min_distance(threads[i], threads[j]) - float(scene["width"]))
    for decoy in scene["decoys"]:
        tip = decoy["points"][-1]
        x, y = float(tip[0]), float(tip[1])
        best = min(best, x, y, CANVAS - x, CANVAS - y)
    return round(float(best), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twins: list[tuple[str, dict]] = []
    if len(scene["decoys"]) > 1:
        flipped = dict(scene)
        flipped["decoys"] = list(reversed(scene["decoys"]))
        twins.append(("decoy_draw_order_reversed", flipped))
    if len(scene["clutter"]) > 1:
        rotated = dict(scene)
        rotated["clutter"] = scene["clutter"][1:] + scene["clutter"][:1]
        twins.append(("clutter_order_rotated", rotated))
    return twins
