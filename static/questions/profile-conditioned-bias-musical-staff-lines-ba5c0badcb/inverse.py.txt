"""Independent raster-only inverse for sustained horizontal staff paths."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _row_groups(mask: np.ndarray) -> tuple[list[tuple[int, int]], np.ndarray]:
    support = mask.sum(axis=1)
    active = support >= max(100, int(mask.shape[1] * 0.55))
    groups: list[tuple[int, int]] = []
    start = None
    for y, value in enumerate(active.tolist() + [False]):
        if value and start is None:
            start = y
        elif not value and start is not None:
            groups.append((start, y - 1))
            start = None
    return groups, support


def measure_from_image(image: Image.Image) -> tuple[str, float]:
    """Return count and weakest-accepted versus strongest-rejected row margin."""
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Both staff palettes and note ink are far from the near-white paper colors.
    mask = np.min(255 - rgb, axis=2) > 55
    groups, support = _row_groups(mask)
    if not 4 <= len(groups) <= 6:
        raise ValueError(f"ambiguous staff evidence: found {len(groups)} long row groups")
    accepted_rows = {y for lo, hi in groups for y in range(lo, hi + 1)}
    accepted = min(int(support[y]) for y in accepted_rows)
    rejected = max((int(value) for y, value in enumerate(support) if y not in accepted_rows), default=0)
    runner_up_margin = float(accepted - rejected)
    if runner_up_margin < 80:
        raise ValueError(f"ambiguous staff evidence: support margin {runner_up_margin}")
    return str(len(groups)), runner_up_margin


def decision_from_image(image: Image.Image) -> str:
    return measure_from_image(image)[0]
