"""Deterministic Pillow renderer and analytic arm for a music-staff world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 360
STAFF_PALETTE = (
    (35, 67, 112),
    (27, 96, 93),
    (88, 57, 119),
    (117, 72, 45),
)
NOTE_PALETTE = ((35, 37, 42), (52, 48, 58), (48, 55, 62))


def sample_scene(seed: int) -> dict:
    """Sample a balanced answer with all nuisance factors seeded independently."""
    rng = random.Random(int(seed) ^ 0x5A17C9)
    line_count = 4 + (int(seed) % 3)
    line_left = rng.randint(58, 88)
    line_length = rng.randint(330, 382)
    spacing = rng.randint(28, 34)
    thickness = rng.randint(3, 5)
    total_height = (line_count - 1) * spacing
    top_y = rng.randint(150 - total_height // 2, 180 - total_height // 2)
    note_radius = rng.randint(7, 9)
    note_count = rng.randint(6, 10)
    notes = []
    x_min = line_left + 42
    x_max = line_left + line_length - 24
    pitch_steps = list(range(-2, 2 * line_count + 2))
    for _ in range(note_count):
        x = rng.randint(x_min, x_max)
        step = rng.choice(pitch_steps)
        y = top_y + round(step * spacing / 2)
        stem_up = bool(rng.getrandbits(1))
        stem_height = rng.randint(28, 45)
        notes.append({"x": x, "y": y, "stem_up": stem_up, "stem_height": stem_height})
    return {
        "line_count": line_count,
        "line_left": line_left,
        "line_length": line_length,
        "top_y": top_y,
        "spacing": spacing,
        "thickness": thickness,
        "staff_color": list(rng.choice(STAFF_PALETTE)),
        "note_color": list(rng.choice(NOTE_PALETTE)),
        "note_radius": note_radius,
        "notes": notes,
    }


def render(scene: dict) -> Image.Image:
    """Render only from public scene fields; no hidden label payload is used."""
    line_count = int(scene["line_count"])
    line_left = int(scene["line_left"])
    line_length = int(scene["line_length"])
    top_y = int(scene["top_y"])
    spacing = int(scene["spacing"])
    thickness = int(scene["thickness"])
    staff_color = tuple(scene["staff_color"])
    note_color = tuple(scene["note_color"])
    note_radius = int(scene["note_radius"])
    notes = scene["notes"]

    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)

    # A faint paper wash is deliberately non-directional and answer-independent.
    draw.rounded_rectangle((28, 36, 484, 324), radius=18, fill=(255, 255, 253))

    half_lo = thickness // 2
    half_hi = thickness - half_lo - 1
    right = line_left + line_length - 1
    for index in range(line_count):
        y = top_y + index * spacing
        draw.rectangle((line_left, y - half_lo, right, y + half_hi), fill=staff_color)

    # Compact note symbols cross the staff but never approach full-row coverage.
    for note in notes:
        x = int(note["x"])
        y = int(note["y"])
        r = note_radius
        draw.ellipse((x - r, y - r + 2, x + r, y + r - 2), fill=note_color)
        if bool(note["stem_up"]):
            draw.rectangle((x + r - 2, y - int(note["stem_height"]), x + r + 1, y), fill=note_color)
        else:
            draw.rectangle((x - r - 1, y, x - r + 2, y + int(note["stem_height"])), fill=note_color)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["line_count"]))


def margin(scene: dict) -> float:
    """Conservative accepted-row versus strongest-distractor support gap."""
    full_support = int(scene["line_length"])
    distractor_bound = len(scene["notes"]) * (2 * int(scene["note_radius"]) + 1)
    return float(full_support - distractor_bound)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 100.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["notes"] = list(reversed(twin["notes"]))
    return [("reverse_same_color_note_draw_order", twin)]
