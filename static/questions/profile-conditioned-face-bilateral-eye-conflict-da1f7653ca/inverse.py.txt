"""Independent pixel-only inverse for bilateral eye coverage around a nose."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_FACE_PALETTE = np.array(((244, 202, 91), (236, 177, 111), (226, 193, 151)), dtype=np.int16)
_EYE_PALETTE = np.array(((61, 142, 203), (132, 93, 184), (38, 151, 139)), dtype=np.int16)
_NOSE_PALETTE = np.array(((205, 66, 61), (222, 103, 38), (190, 55, 112)), dtype=np.int16)
_TOLERANCE = 27.0


class AmbiguousImage(ValueError):
    """Raised when the visible parts or side boundary cannot be recovered."""


def _palette_mask(rgb: np.ndarray, palette: np.ndarray) -> np.ndarray:
    pixels = rgb.astype(np.int16)
    delta = pixels[:, :, None, :] - palette[None, None, :, :]
    distances = np.sqrt(np.sum(delta.astype(np.float32) ** 2, axis=3))
    return np.min(distances, axis=2) <= _TOLERANCE


def _components(mask: np.ndarray, minimum_area: int) -> list[dict]:
    work = mask.copy()
    height, width = work.shape
    found: list[dict] = []
    while bool(work.any()):
        y0, x0 = np.argwhere(work)[0]
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(xs) >= minimum_area:
            found.append({
                "left": min(xs), "right": max(xs), "top": min(ys), "bottom": max(ys),
                "width": max(xs) - min(xs) + 1, "height": max(ys) - min(ys) + 1,
                "area": len(xs), "cx": (min(xs) + max(xs)) / 2.0,
            })
    return sorted(found, key=lambda item: item["cx"])


def _mask_box(mask: np.ndarray) -> dict:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise AmbiguousImage("required color family is absent")
    return {
        "left": int(xs.min()), "right": int(xs.max()),
        "top": int(ys.min()), "bottom": int(ys.max()),
        "width": int(xs.max() - xs.min() + 1),
        "height": int(ys.max() - ys.min() + 1),
        "area": int(len(xs)),
    }


def measure_image(image: Image.Image) -> dict:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 260 or rgb.shape[1] < 340:
        raise AmbiguousImage("image is too small")

    face = _mask_box(_palette_mask(rgb, _FACE_PALETTE))
    eyes = _components(_palette_mask(rgb, _EYE_PALETTE), 650)
    noses = _components(_palette_mask(rgb, _NOSE_PALETTE), 350)
    if face["area"] < 36_000 or face["width"] < 260 or face["height"] < 225:
        raise AmbiguousImage("one intact face region was not recovered")
    if len(eyes) != 2:
        raise AmbiguousImage("exactly two complete colored eye interiors were not recovered")
    if len(noses) != 1:
        raise AmbiguousImage("one triangular nose was not recovered")

    dark = np.max(rgb, axis=2) < 95
    for eye in eyes:
        if not (43 <= eye["width"] <= 65 and 28 <= eye["height"] <= 47):
            raise AmbiguousImage("an eye has implausible visible dimensions")
        inset = dark[eye["top"]:eye["bottom"] + 1, eye["left"]:eye["right"] + 1]
        if int(inset.sum()) < 120:
            raise AmbiguousImage("an eye lacks its visible dark pupil")
    nose = noses[0]
    if not (25 <= nose["width"] <= 43 and 32 <= nose["height"] <= 51):
        raise AmbiguousImage("the nose has implausible visible dimensions")

    sides: list[str] = []
    gaps: list[float] = []
    for eye in eyes:
        left_gap = float(nose["left"] - eye["right"])
        right_gap = float(eye["left"] - nose["right"])
        if left_gap >= 8.0:
            sides.append("left")
            gaps.append(left_gap)
        elif right_gap >= 8.0:
            sides.append("right")
            gaps.append(right_gap)
        else:
            raise AmbiguousImage("an eye lies in the nose boundary band")

    decision = "yes" if set(sides) == {"left", "right"} else "no"
    return {
        "decision": decision,
        "margin": min(gaps),
        "eye_sides": sides,
        "eye_boxes": eyes,
        "nose_box": nose,
    }


def decision_from_image(image: Image.Image) -> str:
    try:
        return str(measure_image(image)["decision"])
    except (AmbiguousImage, ValueError, IndexError, TypeError):
        return "abstain"
