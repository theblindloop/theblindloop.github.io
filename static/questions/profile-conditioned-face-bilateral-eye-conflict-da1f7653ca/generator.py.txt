"""Matched renderer for bilateral versus one-sided eye placement on a face."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


FACE_COLORS = ((244, 202, 91), (236, 177, 111), (226, 193, 151))
EYE_COLORS = ((61, 142, 203), (132, 93, 184), (38, 151, 139))
NOSE_COLORS = ((205, 66, 61), (222, 103, 38), (190, 55, 112))
BACKGROUNDS = ((249, 248, 244), (245, 249, 251), (251, 246, 249))
INK = (38, 42, 49)


def sample_scene(seed: int) -> dict:
    """Return deterministic even/odd control-counterfactual pairs."""

    pair_seed = int(seed) // 2
    rng = random.Random(0xFACEB17 ^ (pair_seed * 1_000_003))
    return {
        "width": 420,
        "height": 340,
        "bilateral": int(seed) % 2 == 0,
        "one_sided_side": -1 if rng.randrange(2) == 0 else 1,
        "face_center_x": 210 + rng.randint(-11, 11),
        "face_center_y": 174 + rng.randint(-6, 6),
        "face_rx": rng.randrange(142, 153, 2),
        "face_ry": rng.randrange(124, 135, 2),
        "eye_inner_offset": rng.randrange(68, 77, 2),
        "eye_outer_offset": rng.randrange(96, 109, 2),
        "eye_center_y": 135 + rng.randint(-5, 5),
        "eye_width": rng.randrange(54, 63, 2),
        "eye_height": rng.randrange(38, 45, 2),
        "eye_y_offsets": [-25 + rng.randint(-2, 2), 25 + rng.randint(-2, 2)],
        "pupil_x_offsets": [rng.randint(-3, 3), rng.randint(-3, 3)],
        "pupil_y_offsets": [rng.randint(-2, 2), rng.randint(-2, 2)],
        "nose_width": rng.randrange(32, 39, 2),
        "nose_height": rng.randrange(42, 49, 2),
        "face_color_index": rng.randrange(len(FACE_COLORS)),
        "eye_color_index": rng.randrange(len(EYE_COLORS)),
        "nose_color_index": rng.randrange(len(NOSE_COLORS)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "mouth_curve": rng.choice([-1, 1]),
        "eye_draw_order": [0, 1] if rng.randrange(2) == 0 else [1, 0],
    }


def _eye_centers(scene: dict) -> list[tuple[float, float]]:
    side = int(scene["one_sided_side"])
    outer = side * float(scene["eye_outer_offset"])
    inner = side * float(scene["eye_inner_offset"])
    if bool(scene["bilateral"]):
        inner = -inner
    cx = float(scene["face_center_x"])
    cy = float(scene["eye_center_y"])
    ys = scene["eye_y_offsets"]
    return [(cx + outer, cy + float(ys[0])), (cx + inner, cy + float(ys[1]))]


def _scale_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def render(scene: dict) -> Image.Image:
    """Render a recognizable face with two equally styled complete eye motifs."""

    ss = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = BACKGROUNDS[int(scene["background_index"])]
    face_color = FACE_COLORS[int(scene["face_color_index"])]
    eye_color = EYE_COLORS[int(scene["eye_color_index"])]
    nose_color = NOSE_COLORS[int(scene["nose_color_index"])]
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    cx = float(scene["face_center_x"])
    cy = float(scene["face_center_y"])
    rx = float(scene["face_rx"])
    ry = float(scene["face_ry"])
    draw.ellipse(
        _scale_box((cx - rx, cy - ry, cx + rx, cy + ry), ss),
        fill=face_color,
        outline=INK,
        width=5 * ss,
    )

    eye_w = float(scene["eye_width"])
    eye_h = float(scene["eye_height"])
    centers = _eye_centers(scene)
    for index in scene["eye_draw_order"]:
        ex, ey = centers[int(index)]
        draw.ellipse(
            _scale_box((ex - eye_w / 2, ey - eye_h / 2, ex + eye_w / 2, ey + eye_h / 2), ss),
            fill=eye_color,
            outline=INK,
            width=3 * ss,
        )
        px = ex + float(scene["pupil_x_offsets"][int(index)])
        py = ey + float(scene["pupil_y_offsets"][int(index)])
        draw.ellipse(_scale_box((px - 7, py - 8, px + 7, py + 8), ss), fill=INK)

    nose_w = float(scene["nose_width"])
    nose_h = float(scene["nose_height"])
    nose_top = cy - 16.0
    nose_points = [
        (int(round(cx * ss)), int(round(nose_top * ss))),
        (int(round((cx - nose_w / 2) * ss)), int(round((nose_top + nose_h) * ss))),
        (int(round((cx + nose_w / 2) * ss)), int(round((nose_top + nose_h) * ss))),
    ]
    draw.polygon(nose_points, fill=nose_color, outline=INK)
    draw.line(nose_points + [nose_points[0]], fill=INK, width=3 * ss, joint="curve")

    mouth_y = cy + 70.0
    mouth_box = _scale_box((cx - 55, mouth_y - 18, cx + 55, mouth_y + 18), ss)
    if int(scene["mouth_curve"]) > 0:
        draw.arc(mouth_box, start=10, end=170, fill=INK, width=5 * ss)
    else:
        draw.arc(mouth_box, start=190, end=350, fill=INK, width=5 * ss)

    # Identical brows and cheek marks are non-answer-bearing facial context.
    for sign in (-1, 1):
        bx = cx + sign * 72
        draw.line(
            [(int((bx - 22) * ss), int((cy - 82) * ss)),
             (int((bx + 22) * ss), int((cy - 86) * ss))],
            fill=INK,
            width=4 * ss,
        )
        draw.ellipse(_scale_box((cx + sign * 103 - 4, cy + 42, cx + sign * 103 + 4, cy + 50), ss), fill=nose_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["bilateral"]) else "no"


def margin(scene: dict) -> float:
    half_eye = float(scene["eye_width"]) / 2.0
    half_nose = float(scene["nose_width"]) / 2.0
    return min(float(scene["eye_inner_offset"]), float(scene["eye_outer_offset"])) - half_eye - half_nose


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0 or int(scene["eye_height"]) < 34 or int(scene["nose_width"]) < 28


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["eye_draw_order"] = list(reversed(twin["eye_draw_order"]))
    return [("reverse_disjoint_eye_draw_order", twin)]
