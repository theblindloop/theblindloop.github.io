"""Accumulated-travel state tracking (state_travel_distance).

The image shows FOUR ordered panels P1, P2, P3, P4 read LEFT to RIGHT. Each
panel contains the SAME three colour-coded markers -- red, green, and blue --
each travelling along its OWN horizontal lane (red on the top lane, green on
the middle lane, blue on the bottom lane). Along each lane there are NINE
evenly spaced sites (site 1 leftmost ... site 9 rightmost). In every panel each
marker sits on exactly one site of its lane.

A marker's TOTAL TRAVEL is the sum, measured in sites, of the absolute steps it
takes between consecutive panels (|P2-P1| + |P3-P2| + |P4-P3|). The task is to
report WHICH marker colour accumulates the GREATEST total travel.

Because the markers are identity-locked by colour and their per-panel site
positions change over the sequence, the answer is recovered only by following
each identity across the ordered panels and integrating its movement magnitude
-- a temporal state-tracking decision, not a single-frame cue.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 945
HEIGHT = 300
NPANELS = 4
NTOK = 3
SITES = 9

PANEL_W = 210
PANEL_H = 250
PANEL_TOP = 20
PANEL_GAP = 30
PANEL_X0 = 15

SITE_MARGIN = 25
SITE_STEP = 20
R = 9

TRACK_YS = [PANEL_TOP + 70, PANEL_TOP + 125, PANEL_TOP + 180]

BG = (255, 255, 255)
BORDER = (120, 120, 120)
TRACK = (228, 228, 228)

COLOR_NAMES = ["red", "green", "blue"]
COLORS = {
    "red": (220, 40, 40),
    "green": (40, 170, 80),
    "blue": (40, 80, 220),
}


def _panel_left(p: int) -> int:
    return PANEL_X0 + p * (PANEL_W + PANEL_GAP)


def _site_x(panel_left: int, idx: int) -> int:
    return panel_left + SITE_MARGIN + idx * SITE_STEP


def _travel(pos: list[list[int]]) -> list[int]:
    return [
        sum(abs(pos[t][p + 1] - pos[t][p]) for p in range(NPANELS - 1)) for t in range(NTOK)
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    winner = seed % NTOK
    for _ in range(4000):
        positions = []
        for t in range(NTOK):
            while True:
                path = [rng.randrange(SITES) for _ in range(NPANELS)]
                dist = sum(abs(path[i + 1] - path[i]) for i in range(NPANELS - 1))
                if t == winner:
                    if dist >= 6:
                        break
                else:
                    if dist <= 3:
                        break
            positions.append(path)
        dists = _travel(positions)
        winner_dist = dists[winner]
        others = max(d for i, d in enumerate(dists) if i != winner)
        if winner_dist - others >= 2:
            return {"positions": positions}
    return {"positions": [[0, 0, 0, 0] for _ in range(NTOK)]}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]
    for p in range(NPANELS):
        left = _panel_left(p)
        draw.rectangle(
            [left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H],
            outline=BORDER,
            width=3,
        )
        for t in range(NTOK):
            y = TRACK_YS[t]
            draw.line(
                [(left + SITE_MARGIN - 5, y), (left + SITE_STEP * (SITES - 1) + SITE_MARGIN + 5, y)],
                fill=TRACK,
                width=2,
            )
        for t in range(NTOK):
            y = TRACK_YS[t]
            idx = positions[t][p]
            cx = _site_x(left, idx)
            draw.ellipse(
                [cx - R, y - R, cx + R, y + R],
                fill=COLORS[COLOR_NAMES[t]],
            )
    return img


def _winner_index(positions: list[list[int]]) -> int:
    dists = _travel(positions)
    return dists.index(max(dists))


def analytic_gold(scene: dict) -> str:
    positions = scene["positions"]
    return COLOR_NAMES[_winner_index(positions)]


def margin(scene: dict) -> float:
    positions = scene["positions"]
    dists = sorted(_travel(positions), reverse=True)
    if len(dists) < 2:
        return 0.0
    return float(dists[0] - dists[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry. The ``positions`` field
    directly sets each marker's drawn site in every panel, and analytic gold
    reads exactly those same positions (the per-identity travelled distance is
    a function of the positions alone). Any reparametrization that changed a
    marker's site would change the raster; permuting the panel order or the
    colour lanes would change the pixels. Hence no pixel-identical twin with a
    different gold exists."""
    return []
