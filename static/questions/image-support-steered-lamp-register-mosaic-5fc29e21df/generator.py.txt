"""Independent scene sampler and renderer for a compact register program."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
PALETTES = (
    ((70, 181, 176), (230, 124, 139)),
    ((88, 151, 212), (226, 154, 70)),
    ((120, 177, 92), (186, 125, 201)),
    ((69, 170, 214), (219, 121, 174)),
)
ACCENTS = ((241, 198, 73), (236, 183, 83), (224, 196, 92))
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    bits = [rng.randrange(2) for _ in range(3)]
    operations = [rng.choice("RF") for _ in range(15)]
    # Avoid visually repetitive programs while leaving both operation counts variable.
    if operations.count("R") < 4:
        operations[:4] = list("RRRR")
        rng.shuffle(operations)
    if operations.count("F") < 4:
        operations[:4] = list("FFFF")
        rng.shuffle(operations)
    return {
        "initial_bits": bits,
        "operations": "".join(operations),
        "palette": rng.randrange(len(PALETTES)),
        "accent": rng.randrange(len(ACCENTS)),
        "cell_size": rng.choice((60, 62, 64)),
        "board_shift": [rng.randint(-10, 10), rng.randint(-10, 10)],
    }


def _replay(initial_bits: list[int], operations: str) -> list[int]:
    bits = list(initial_bits)
    for operation in operations:
        if operation == "R":
            bits = [bits[1], bits[2], bits[0]]
        elif operation == "F":
            bits[0] = 1 - bits[0]
        else:
            raise ValueError("operation must be R or F")
    return bits


def analytic_gold(scene: dict) -> str:
    bits = _replay(scene["initial_bits"], scene["operations"])
    return "".join(str(bit) for bit in bits)


def margin(scene: dict) -> float:
    baseline = analytic_gold(scene)
    distances = []
    for index, value in enumerate(scene["operations"]):
        alternative = scene["operations"][:index] + ("F" if value == "R" else "R") + scene["operations"][index + 1 :]
        outcome = "".join(str(bit) for bit in _replay(scene["initial_bits"], alternative))
        if outcome != baseline:
            distances.append(sum(a != b for a, b in zip(outcome, baseline)))
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    initial_bits = list(scene["initial_bits"])
    operations = str(scene["operations"])
    palette = PALETTES[int(scene["palette"])]
    accent = ACCENTS[int(scene["accent"])]
    cell = int(scene["cell_size"])
    shift_x, shift_y = (int(value) for value in scene["board_shift"])
    if len(initial_bits) != 3 or any(bit not in (0, 1) for bit in initial_bits):
        raise ValueError("initial_bits must contain three binary values")
    if len(operations) != 15 or any(value not in "RF" for value in operations):
        raise ValueError("operations must contain fifteen R/F symbols")

    image = Image.new("RGB", (CANVAS, CANVAS), "white")
    draw = ImageDraw.Draw(image)
    board = 4 * cell
    left = (CANVAS - board) // 2 + shift_x
    top = (CANVAS - board) // 2 + shift_y
    font = ImageFont.truetype(str(FONT_PATH), max(30, int(cell * 0.58)))
    small = ImageFont.truetype(str(FONT_PATH), max(9, int(cell * 0.14)))
    title = ImageFont.truetype(str(FONT_PATH), 20)
    draw.text((CANVAS // 2, 46), "THREE-LAMP REGISTER", fill=(30, 38, 52), anchor="mm", font=title)
    draw.text((CANVAS // 2, 72), "ROW ORDER  →", fill=(65, 73, 88), anchor="mm", font=small)

    symbols = ["S", *operations]
    for index, symbol in enumerate(symbols):
        row, column = divmod(index, 4)
        x0 = left + column * cell
        y0 = top + row * cell
        fill = accent if symbol == "S" else palette[0 if symbol == "R" else 1]
        draw.rectangle((x0, y0, x0 + cell, y0 + cell), fill=fill)
        if symbol == "S":
            draw.text((x0 + cell / 2, y0 + cell * 0.18), "START", fill=(25, 28, 34), anchor="mm", font=small)
            radius = max(5, int(cell * 0.105))
            for lamp, bit in enumerate(initial_bits):
                cx = x0 + cell * (0.24 + 0.26 * lamp)
                cy = y0 + cell * 0.59
                box = (cx - radius, cy - radius, cx + radius, cy + radius)
                draw.ellipse(box, fill=(26, 29, 35) if bit else fill, outline=(26, 29, 35), width=2)
        else:
            draw.text((x0 + cell / 2, y0 + cell / 2 - 1), symbol, fill=(24, 27, 34), anchor="mm", font=font)

    # One continuous frame and internal lattice make completeness directly visible.
    frame = max(3, cell // 16)
    draw.rectangle((left, top, left + board, top + board), outline=(24, 27, 34), width=frame)
    for step in range(1, 4):
        draw.line((left + step * cell, top, left + step * cell, top + board), fill=(24, 27, 34), width=frame)
        draw.line((left, top + step * cell, left + board, top + step * cell), fill=(24, 27, 34), width=frame)
    draw.text((CANVAS // 2, 458), "R = ROTATE LEFT     F = FLIP LEFT LAMP", fill=(45, 52, 66), anchor="mm", font=small)
    return image
