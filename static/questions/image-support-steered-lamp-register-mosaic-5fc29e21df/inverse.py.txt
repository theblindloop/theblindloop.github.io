"""Pixel-only inverse for the lamp-register mosaic.

The implementation locates the saturated connected board from pixels, infers
its lattice, validates all sixteen cell interiors, decodes the start lamps and
operation glyphs, and replays the register. It receives no latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    output = []
    start = previous = int(indices[0])
    for value in indices[1:]:
        value = int(value)
        if value > previous + 1:
            output.append((start, previous + 1))
            start = value
        previous = value
    output.append((start, previous + 1))
    return output


def _replay(bits: list[int], operations: list[str]) -> str:
    state = list(bits)
    for operation in operations:
        if operation == "R":
            state = [state[1], state[2], state[0]]
        elif operation == "F":
            state[0] = 1 - state[0]
        else:
            raise ValueError("unrecognized operation")
    return "".join(str(value) for value in state)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 300:
        raise ValueError("invalid canvas")
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    saturated = (maximum - minimum > 38) & (maximum < 248) & (minimum < 235)
    points = np.argwhere(saturated)
    if len(points) < 20000:
        raise ValueError("complete board not found")
    sy0, sx0 = points.min(axis=0)
    sy1, sx1 = points.max(axis=0) + 1
    cell_y = (sy1 - sy0 + 5) / 4.0
    cell_x = (sx1 - sx0 + 5) / 4.0
    cell = (cell_x + cell_y) / 2.0
    if not 58 <= cell <= 66 or abs(cell_x - cell_y) > 2.0:
        raise ValueError("invalid board geometry")
    left = int(round(sx0 - 3))
    top = int(round(sy0 - 3))

    # Every tile must retain most of its colored interior. This is a visible
    # completeness check, and makes destructive regional erasure abstain.
    for index in range(16):
        row, column = divmod(index, 4)
        x0 = int(round(left + column * cell + cell * 0.10))
        x1 = int(round(left + (column + 1) * cell - cell * 0.10))
        y0 = int(round(top + row * cell + cell * 0.10))
        y1 = int(round(top + (row + 1) * cell - cell * 0.10))
        purity = float(saturated[y0:y1, x0:x1].mean())
        if purity < 0.69:
            raise ValueError("damaged or ambiguous tile")

    dark = maximum < 95
    bits = []
    for lamp in range(3):
        cx = int(round(left + cell * (0.24 + 0.26 * lamp)))
        cy = int(round(top + cell * 0.59))
        radius = max(3, int(round(cell * 0.055)))
        patch = dark[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
        fraction = float(patch.mean())
        if fraction > 0.70:
            bits.append(1)
        elif fraction < 0.20:
            bits.append(0)
        else:
            raise ValueError("ambiguous start lamp")

    operations = []
    for index in range(1, 16):
        row, column = divmod(index, 4)
        x0 = int(round(left + column * cell + cell * 0.20))
        x1 = int(round(left + (column + 1) * cell - cell * 0.20))
        y0 = int(round(top + row * cell + cell * 0.20))
        y1 = int(round(top + (row + 1) * cell - cell * 0.20))
        glyph = dark[y0:y1, x0:x1]
        ink = float(glyph.mean())
        # In the shared bold font, R has a diagonal leg and enclosed bowl and
        # therefore materially more central dark ink than F at every cell size.
        if ink > 0.240:
            operations.append("R")
        elif ink < 0.225:
            operations.append("F")
        else:
            raise ValueError("ambiguous operation glyph")
    return _replay(bits, operations)
