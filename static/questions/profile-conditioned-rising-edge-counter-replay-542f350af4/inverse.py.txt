"""Independent image-only inverse for the rising-edge counter world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AmbiguousImage(ValueError):
    pass


def _components(mask: np.ndarray):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for ny, nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    queue.append((ny,nx))
        if len(points) >= 180:
            output.append(points)
    return output


def _read_states(image: Image.Image) -> list[int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    # Signal marks are the only high-saturation objects. This deliberately does
    # not assume a particular palette or sample coordinates.
    mask = (high - low > 55) & (high > 80) & (low < 170)
    marks = []
    for points in _components(mask):
        ys = np.fromiter((p[0] for p in points), dtype=int)
        xs = np.fromiter((p[1] for p in points), dtype=int)
        bw = int(xs.max() - xs.min() + 1)
        bh = int(ys.max() - ys.min() + 1)
        if 32 <= bw <= 55 and 32 <= bh <= 55 and 0.75 <= bw / bh <= 1.3:
            ratio = len(points) / float(bw * bh)
            if ratio > 0.61:
                state = 1
            elif ratio < 0.46:
                state = 0
            else:
                raise AmbiguousImage(f"signal fill ratio {ratio:.3f} has no clear state")
            marks.append((float(xs.mean()), float(ys.mean()), state))
    if len(marks) != 8:
        raise AmbiguousImage(f"expected 8 signal marks, recovered {len(marks)}")
    marks.sort()
    ys = [m[1] for m in marks]
    if max(ys) - min(ys) > 12:
        raise AmbiguousImage("signal marks do not form one ordered row")
    gaps = [marks[i+1][0] - marks[i][0] for i in range(7)]
    if min(gaps) < 55:
        raise AmbiguousImage("signal order is unresolved")
    return [m[2] for m in marks]


def decision_from_image(image: Image.Image) -> str:
    states = _read_states(image)
    previous = states[0]
    pulses = 0
    for current in states[1:]:
        pulses += int(previous == 0 and current == 1)
        previous = current
    return str(pulses)
