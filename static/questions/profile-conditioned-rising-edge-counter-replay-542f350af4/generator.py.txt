"""Latent sampler and Pillow renderer for Rising-Edge Pulse Counter."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1160, 420
N_OBSERVATIONS = 7
PALETTE = ("#1769aa", "#167c5a", "#7546a8", "#b54b2a")


def _rise_count(initial: int, observations: list[int]) -> int:
    previous = initial
    count = 0
    for current in observations:
        count += int(previous == 0 and current == 1)
        previous = current
    return count


def _visual_observations(scene: dict) -> list[int]:
    values = list(scene["observations"])
    if scene["display_reversed"]:
        values.reverse()
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 5
    possibilities = []
    for initial in (0, 1):
        for values in itertools.product((0, 1), repeat=N_OBSERVATIONS):
            if _rise_count(initial, list(values)) == target:
                possibilities.append((initial, list(values)))
    initial, visual = rng.choice(possibilities)
    display_reversed = bool(rng.randrange(2))
    stored = list(reversed(visual)) if display_reversed else list(visual)
    return {
        "initial_state": initial,
        "observations": stored,
        "on_color": PALETTE[rng.randrange(len(PALETTE))],
        "display_reversed": display_reversed,
    }


def analytic_gold(scene: dict) -> str:
    return str(_rise_count(int(scene["initial_state"]), _visual_observations(scene)))


def margin(scene: dict) -> float:
    # Runner-up separation between the expected normalized ink fill of a solid
    # disk/ring and the public pixel parser's 0.53 decision boundary.
    outer, inner = 22.0, 15.0
    solid_fill = math.pi * outer * outer / ((2 * outer + 1) ** 2)
    ring_fill = math.pi * (outer * outer - inner * inner) / ((2 * outer + 1) ** 2)
    return round(min(solid_fill - 0.53, 0.53 - ring_fill), 6)


def is_quarantined(scene: dict) -> bool:
    values = [scene.get("initial_state"), *scene.get("observations", [])]
    return len(values) != N_OBSERVATIONS + 1 or any(v not in (0, 1) for v in values)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["observations"] = list(reversed(twin["observations"]))
    twin["display_reversed"] = not bool(twin["display_reversed"])
    return [("reverse_storage_and_direction", twin)]


def _font(size: int, bold: bool = False):
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, box, text: str, font, fill):
    b = draw.textbbox((0, 0), text, font=font)
    x = (box[0] + box[2] - (b[2] - b[0])) / 2
    y = (box[1] + box[3] - (b[3] - b[1])) / 2 - b[1]
    draw.text((x, y), text, font=font, fill=fill)


def _draw_signal(draw: ImageDraw.ImageDraw, cx: int, cy: int, state: int, color: str, scale: int):
    r = 22 * scale
    box = (cx * scale - r, cy * scale - r, cx * scale + r, cy * scale + r)
    if state:
        draw.ellipse(box, fill=color)
    else:
        draw.ellipse(box, outline=color, width=7 * scale)


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "#fbfcfe")
    draw = ImageDraw.Draw(image)
    ink = "#202733"
    muted = "#687180"
    border = "#8b95a3"
    signal = scene["on_color"]
    title_font = _font(25 * scale, True)
    label_font = _font(15 * scale, True)
    small_font = _font(13 * scale, False)
    _center_text(draw, (0, 18 * scale, WIDTH * scale, 62 * scale), "RISING-EDGE PULSE COUNTER", title_font, ink)
    _center_text(draw, (0, 62 * scale, WIDTH * scale, 94 * scale), "Count only changes from hollow OFF to solid ON", small_font, muted)

    y0, y1 = 132, 286
    start_x0, start_x1 = 35, 170
    draw.rounded_rectangle((start_x0*scale, y0*scale, start_x1*scale, y1*scale), radius=12*scale, fill="white", outline=border, width=2*scale)
    _center_text(draw, (start_x0*scale, (y0+10)*scale, start_x1*scale, (y0+42)*scale), "START", label_font, ink)
    _draw_signal(draw, (start_x0+start_x1)//2, 220, int(scene["initial_state"]), signal, scale)
    _center_text(draw, (start_x0*scale, 250*scale, start_x1*scale, 279*scale), "ON" if scene["initial_state"] else "OFF", small_font, muted)

    draw.line((183*scale, 209*scale, 224*scale, 209*scale), fill=muted, width=3*scale)
    draw.polygon(((224*scale,209*scale),(211*scale,201*scale),(211*scale,217*scale)), fill=muted)

    observations = _visual_observations(scene)
    card_w, gap, first_x = 105, 13, 238
    for index, state in enumerate(observations, start=1):
        x0 = first_x + (index - 1) * (card_w + gap)
        x1 = x0 + card_w
        draw.rounded_rectangle((x0*scale, y0*scale, x1*scale, y1*scale), radius=12*scale, fill="white", outline=border, width=2*scale)
        _center_text(draw, (x0*scale, (y0+10)*scale, x1*scale, (y0+42)*scale), f"STEP {index}", label_font, ink)
        _draw_signal(draw, (x0+x1)//2, 220, int(state), signal, scale)
        _center_text(draw, (x0*scale, 250*scale, x1*scale, 279*scale), "ON" if state else "OFF", small_font, muted)

    _center_text(draw, (0, 317*scale, WIDTH*scale, 348*scale), "Read left to right", label_font, ink)
    _center_text(draw, (0, 348*scale, WIDTH*scale, 382*scale), "A repeated ON records no new pulse", small_font, muted)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
