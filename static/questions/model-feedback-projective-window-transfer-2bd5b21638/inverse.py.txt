"""Independent image-only inverse for projective-window transfer."""

from __future__ import annotations

import numpy as np
from PIL import Image

CORNER_COLORS = [np.array(c, float) for c in ((216,67,67),(42,117,203),(43,157,94),(145,83,181))]
MARKER = np.array((246,190,34), float)
CUP_COLORS = {"A":np.array((229,121,45),float), "B":np.array((30,166,184),float),
              "C":np.array((215,71,139),float), "D":np.array((112,86,64),float)}


def _pixels(arr, color, tolerance=32.0):
    ys,xs = np.nonzero(np.linalg.norm(arr.astype(float)-color, axis=2) <= tolerance)
    return xs.astype(float), ys.astype(float)


def _centroid(arr, color, minimum=220):
    xs,ys = _pixels(arr,color)
    if len(xs) < minimum:
        raise ValueError("required colored mark is missing")
    return np.array([xs.mean(),ys.mean()])


def _paired_corner(arr, color):
    xs,ys = _pixels(arr,color)
    if len(xs) < 650:
        raise ValueError("corner pair is missing")
    left = xs < 490
    right = xs > 500
    if left.sum() < 220 or right.sum() < 220:
        raise ValueError("could not separate corner pair")
    return np.array([xs[left].mean(),ys[left].mean()]), np.array([xs[right].mean(),ys[right].mean()])


def _homography(src,dst):
    rows,rhs=[],[]
    for (x,y),(u,v) in zip(src,dst):
        rows += [[x,y,1,0,0,0,-u*x,-u*y],[0,0,0,x,y,1,-v*x,-v*y]]
        rhs += [u,v]
    try:
        h=np.linalg.solve(np.asarray(rows,float),np.asarray(rhs,float))
    except np.linalg.LinAlgError as exc:
        raise ValueError("degenerate visible corners") from exc
    return np.append(h,1.0).reshape(3,3)


def decision_from_image(image: Image.Image) -> str:
    arr=np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (720,1000):
        raise ValueError("unexpected canvas size")
    pairs=[_paired_corner(arr,c) for c in CORNER_COLORS]
    src=[p[0] for p in pairs]
    dst=[p[1] for p in pairs]
    marker=_centroid(arr,MARKER,300)
    q=_homography(src,dst) @ np.array([marker[0],marker[1],1.0])
    if abs(q[2]) < 1e-8:
        raise ValueError("endpoint at infinity")
    endpoint=q[:2]/q[2]
    cups={label:_centroid(arr,color,420) for label,color in CUP_COLORS.items()}
    distances={label:float(np.linalg.norm(endpoint-center)) for label,center in cups.items()}
    ordered=sorted(distances,key=distances.get)
    if distances[ordered[0]] > 9.0 or distances[ordered[1]]-distances[ordered[0]] < 42.0:
        raise ValueError("transferred point does not identify one cup")
    return ordered[0]
