"""Independent image-only inverse for braided trace terminal identity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


START_GREEN = np.array((39.0, 155.0, 104.0), dtype=np.float32)
TERMINAL_RGB = {
    "red": np.array((213.0, 67.0, 76.0), dtype=np.float32),
    "blue": np.array((45.0, 112.0, 195.0), dtype=np.float32),
    "gold": np.array((226.0, 165.0, 35.0), dtype=np.float32),
}


class OracleFitError(RuntimeError):
    pass


def _colour_components(arr: np.ndarray, prototype: np.ndarray, tolerance: float) -> list[np.ndarray]:
    distances = np.sqrt(np.sum((arr - prototype) ** 2, axis=2))
    mask = distances <= tolerance
    return _components(mask, minimum=35)


def _route_mask(arr: np.ndarray) -> np.ndarray:
    # The route is the only very dark ink; colored markers and pale strokes are
    # intentionally excluded by the all-channel ceiling.
    return np.max(arr, axis=2) < 136.0


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    result = mask.copy()
    height, width = mask.shape
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            if dx * dx + dy * dy > radius * radius:
                continue
            y0, y1 = max(0, dy), min(height, height + dy)
            x0, x1 = max(0, dx), min(width, width + dx)
            sy0, sy1 = max(0, -dy), min(height, height - dy)
            sx0, sx1 = max(0, -dx), min(width, width - dx)
            result[y0:y1, x0:x1] |= mask[sy0:sy1, sx0:sx1]
    return result


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(mask):
        start_y, start_x = int(start_y), int(start_x)
        if visited[start_y, start_x]:
            continue
        visited[start_y, start_x] = True
        queue: deque[tuple[int, int]] = deque([(start_y, start_x)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            components.append(np.asarray(points, dtype=np.int32))
    return components


def _marker_center(arr: np.ndarray, prototype: np.ndarray) -> tuple[float, float]:
    # Use the solid interior color rather than a broad hue neighborhood: the
    # orange source marker is intentionally close to gold in RGB space.
    components = _colour_components(arr, prototype, tolerance=36.0)
    if not components:
        raise OracleFitError("colored marker is missing")
    component = max(components, key=len)
    y, x = component.mean(axis=0)
    return float(x), float(y)


def _route_components(arr: np.ndarray) -> list[np.ndarray]:
    components = _components(_dilate(_route_mask(arr), radius=2), minimum=180)
    if len(components) != 3:
        raise OracleFitError(f"expected three separated route components, found {len(components)}")
    return components


def _nearest_component(center: tuple[float, float], components: list[np.ndarray]) -> tuple[int, float]:
    x, y = center
    best_index = -1
    best_distance = float("inf")
    for index, component in enumerate(components):
        dy = component[:, 0].astype(np.float32) - y
        dx = component[:, 1].astype(np.float32) - x
        distance = float(np.min(dx * dx + dy * dy))
        if distance < best_distance:
            best_distance = distance
            best_index = index
    if best_index < 0:
        raise OracleFitError("route component binding failed")
    return best_index, best_distance


def decision_from_image(image: Image.Image) -> str:
    """Return red, blue, or gold using only the final PNG."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    green_center = _marker_center(arr, START_GREEN)
    components = _route_components(arr)
    green_component, green_distance = _nearest_component(green_center, components)
    if green_distance > 900.0:
        raise OracleFitError("green port is not attached to a route")
    terminal_components = {
        color: _nearest_component(arrival_center, components)[0]
        for color, prototype in TERMINAL_RGB.items()
        for arrival_center in [_marker_center(arr, prototype)]
    }
    matches = [color for color, component in terminal_components.items() if component == green_component]
    if len(matches) != 1:
        raise OracleFitError(f"green route has {len(matches)} terminal bindings")
    return matches[0]
