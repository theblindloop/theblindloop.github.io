"""Deterministic raster renderer and analytic contract for braided traces.

The answer is the terminal color attached to the same visible route component
as the green start port.  No answer-only field is sampled: the two label
bindings used by ``analytic_gold`` are both rendered marker colors.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "braided_trace_terminal_identity-0.1.0"
WIDTH = 640
HEIGHT = 512
BACKGROUND = (247, 246, 241)
START_COLORS = {
    "green": (39, 155, 104),
    "violet": (130, 82, 174),
    "orange": (226, 128, 44),
}
TERMINAL_COLORS = {
    "red": (213, 67, 76),
    "blue": (45, 112, 195),
    "gold": (226, 165, 35),
}
ROUTE_VARIANTS = ((35, 48, 66), (44, 53, 72), (32, 58, 73))
DISTRACTOR_COLOR = (205, 209, 207)


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _sub(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    return (a[0] - b[0], a[1] - b[1])


def _cross(a: tuple[float, float], b: tuple[float, float]) -> float:
    return a[0] * b[1] - a[1] * b[0]


def _point_segment_distance(
    p: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    ab = _sub(b, a)
    denom = ab[0] * ab[0] + ab[1] * ab[1]
    if denom <= 1e-9:
        return _distance(p, a)
    t = max(0.0, min(1.0, ((p[0] - a[0]) * ab[0] + (p[1] - a[1]) * ab[1]) / denom))
    q = (a[0] + t * ab[0], a[1] + t * ab[1])
    return _distance(p, q)


def _segments_intersect(
    a: tuple[float, float],
    b: tuple[float, float],
    c: tuple[float, float],
    d: tuple[float, float],
) -> bool:
    ab = _sub(b, a)
    cd = _sub(d, c)
    values = (_cross(ab, _sub(c, a)), _cross(ab, _sub(d, a)), _cross(cd, _sub(a, c)), _cross(cd, _sub(b, c)))
    eps = 1e-7
    return values[0] * values[1] < -eps and values[2] * values[3] < -eps


def _segment_distance(
    a: tuple[float, float],
    b: tuple[float, float],
    c: tuple[float, float],
    d: tuple[float, float],
) -> float:
    if _segments_intersect(a, b, c, d):
        return 0.0
    return min(
        _point_segment_distance(a, c, d),
        _point_segment_distance(b, c, d),
        _point_segment_distance(c, a, b),
        _point_segment_distance(d, a, b),
    )


def _polyline_length(points: Iterable[Iterable[float]]) -> float:
    pts = [(float(p[0]), float(p[1])) for p in points]
    return sum(_distance(a, b) for a, b in zip(pts, pts[1:]))


def _bezier(
    start: tuple[float, float],
    control_a: tuple[float, float],
    control_b: tuple[float, float],
    end: tuple[float, float],
    count: int = 13,
) -> list[list[float]]:
    points: list[list[float]] = []
    for index in range(count):
        t = index / (count - 1)
        u = 1.0 - t
        x = u**3 * start[0] + 3.0 * u * u * t * control_a[0] + 3.0 * u * t * t * control_b[0] + t**3 * end[0]
        y = u**3 * start[1] + 3.0 * u * u * t * control_a[1] + 3.0 * u * t * t * control_b[1] + t**3 * end[1]
        points.append([round(x, 4), round(y, 4)])
    return points


def _clearance(paths: list[list[list[float]]]) -> float:
    """Return the smallest centerline or endpoint-marker clearance."""

    polyline_paths = [[(float(p[0]), float(p[1])) for p in path] for path in paths]
    minimum = float("inf")
    # Non-neighboring pieces of the same curve must not fold over themselves.
    for path in polyline_paths:
        for i in range(len(path) - 1):
            for j in range(i + 2, len(path) - 1):
                minimum = min(minimum, _segment_distance(path[i], path[i + 1], path[j], path[j + 1]))
    # Every pair of different traces is independent and visibly separated.
    for first in range(len(polyline_paths)):
        for second in range(first + 1, len(polyline_paths)):
            left, right = polyline_paths[first], polyline_paths[second]
            for a, b in zip(left, left[1:]):
                for c, d in zip(right, right[1:]):
                    minimum = min(minimum, _segment_distance(a, b, c, d))
    # Marker edge clearance matters at both ends of every route.
    endpoints = [path[0] for path in polyline_paths] + [path[-1] for path in polyline_paths]
    for i, first in enumerate(endpoints):
        for second in endpoints[i + 1 :]:
            minimum = min(minimum, _distance(first, second) - 30.0)
    for route_index, path in enumerate(polyline_paths):
        for endpoint in (path[0], path[-1]):
            for other_index, other in enumerate(polyline_paths):
                if other_index == route_index:
                    continue
                for a, b in zip(other, other[1:]):
                    minimum = min(minimum, _point_segment_distance(endpoint, a, b) - 16.0)
    return float(minimum)


def _valid_paths(paths: list[list[list[float]]]) -> bool:
    if len(paths) != 3:
        return False
    for path in paths:
        if len(path) < 8 or not 260.0 <= _polyline_length(path) <= 1100.0:
            return False
        if any(not (38.0 <= float(p[0]) <= WIDTH - 38.0 and 38.0 <= float(p[1]) <= HEIGHT - 38.0) for p in path):
            return False
    return _clearance(paths) >= 19.0


def _sample_paths(rng: random.Random) -> list[list[list[float]]]:
    for _attempt in range(7000):
        endpoints: list[tuple[float, float]] = []
        for _ in range(6):
            for _retry in range(80):
                candidate = (rng.uniform(72.0, WIDTH - 72.0), rng.uniform(70.0, HEIGHT - 70.0))
                if all(_distance(candidate, prior) >= 61.0 for prior in endpoints):
                    endpoints.append(candidate)
                    break
            else:
                break
        if len(endpoints) != 6:
            continue
        paths: list[list[list[float]]] = []
        for route_index in range(3):
            start = endpoints[2 * route_index]
            end = endpoints[2 * route_index + 1]
            direct = _sub(end, start)
            direct_length = math.hypot(direct[0], direct[1])
            if not 180.0 <= direct_length <= 520.0:
                paths = []
                break
            length = max(direct_length, 1.0)
            perpendicular = (-direct[1] / length, direct[0] / length)
            offset_a = rng.uniform(-145.0, 145.0)
            offset_b = rng.uniform(-145.0, 145.0)
            control_a = (start[0] + direct[0] * 0.34 + perpendicular[0] * offset_a, start[1] + direct[1] * 0.34 + perpendicular[1] * offset_a)
            control_b = (start[0] + direct[0] * 0.66 + perpendicular[0] * offset_b, start[1] + direct[1] * 0.66 + perpendicular[1] * offset_b)
            paths.append(_bezier(start, control_a, control_b, end))
        if paths and _valid_paths(paths):
            return paths
    raise RuntimeError("could not sample three separated traces")


def _sample_distractors(rng: random.Random, paths: list[list[list[float]]]) -> list[list[list[float]]]:
    route_points = [[(float(p[0]), float(p[1])) for p in path] for path in paths]
    endpoints = [point for path in route_points for point in (path[0], path[-1])]
    distractors: list[list[list[float]]] = []
    for _ in range(7):
        for _attempt in range(200):
            count = rng.choice((2, 3, 4))
            x = rng.uniform(52.0, WIDTH - 52.0)
            y = rng.uniform(52.0, HEIGHT - 52.0)
            heading = rng.uniform(0.0, 2.0 * math.pi)
            points = [(x, y)]
            for _step in range(count - 1):
                heading += rng.uniform(-0.75, 0.75)
                step = rng.uniform(28.0, 58.0)
                points.append((points[-1][0] + step * math.cos(heading), points[-1][1] + step * math.sin(heading)))
            if any(not (32.0 <= px <= WIDTH - 32.0 and 32.0 <= py <= HEIGHT - 32.0) for px, py in points):
                continue
            if any(_distance(point, endpoint) < 28.0 for point in points for endpoint in endpoints):
                continue
            too_close = False
            for point in points:
                for route in route_points:
                    for a, b in zip(route, route[1:]):
                        if _point_segment_distance(point, a, b) < 25.0:
                            too_close = True
                            break
                    if too_close:
                        break
                if too_close:
                    break
            if too_close:
                continue
            distractors.append([[round(px, 4), round(py, 4)] for px, py in points])
            break
    return distractors


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    paths = _sample_paths(rng)
    start_colors = ["green", "violet", "orange"]
    terminal_colors = ["red", "blue", "gold"]
    rng.shuffle(start_colors)
    rng.shuffle(terminal_colors)
    start_radii = [round(rng.uniform(15.0, 18.0), 3) for _ in range(3)]
    terminal_radii = [round(rng.uniform(14.0, 17.0), 3) for _ in range(3)]
    route_color = list(rng.choice(ROUTE_VARIANTS))
    route_width = round(rng.uniform(6.0, 8.0), 3)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "route_color": route_color,
        "route_width": route_width,
        "paths": paths,
        "start_colors": start_colors,
        "terminal_colors": terminal_colors,
        "start_radii": start_radii,
        "terminal_radii": terminal_radii,
        "distractors": _sample_distractors(rng, paths),
        "start_outline": [247, 246, 241],
        "terminal_outline": [247, 246, 241],
        "marker_stroke_width": round(rng.uniform(2.0, 3.0), 3),
    }


def _draw_polyline(draw: ImageDraw.ImageDraw, points: list[list[float]], scale: int, fill: tuple[int, int, int], width: int) -> None:
    draw.line(
        [(int(round(float(point[0]) * scale)), int(round(float(point[1]) * scale))) for point in points],
        fill=fill,
        width=width,
        joint="curve",
    )


def _draw_marker(
    draw: ImageDraw.ImageDraw,
    point: list[float],
    radius: float,
    fill: tuple[int, int, int],
    outline: tuple[int, int, int],
    outline_width: int,
    scale: int,
) -> None:
    x, y = float(point[0]) * scale, float(point[1]) * scale
    r = float(radius) * scale
    draw.ellipse((int(round(x - r)), int(round(y - r)), int(round(x + r)), int(round(y + r))), fill=fill, outline=outline, width=outline_width)


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    for distractor in scene["distractors"]:
        _draw_polyline(draw, distractor, scale, DISTRACTOR_COLOR, max(2, scale))
    route_color = tuple(int(value) for value in scene["route_color"])
    route_width = max(1, int(round(float(scene["route_width"]) * scale)))
    for path in scene["paths"]:
        _draw_polyline(draw, path, scale, route_color, route_width)
    start_outline = tuple(int(value) for value in scene["start_outline"])
    terminal_outline = tuple(int(value) for value in scene["terminal_outline"])
    outline_width = max(1, int(round(float(scene["marker_stroke_width"]) * scale)))
    for index, path in enumerate(scene["paths"]):
        _draw_marker(draw, path[0], float(scene["start_radii"][index]), START_COLORS[scene["start_colors"][index]], start_outline, outline_width, scale)
        _draw_marker(draw, path[-1], float(scene["terminal_radii"][index]), TERMINAL_COLORS[scene["terminal_colors"][index]], terminal_outline, outline_width, scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    green_index = list(scene["start_colors"]).index("green")
    return str(scene["terminal_colors"][green_index])


def margin(scene: dict) -> float:
    return round(_clearance(scene["paths"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
