"""Band-containment count world: how many markers are strictly inside the band.

A light panel shows a horizontal blue band formed by two full-width parallel
lines. Several black circular markers lie scattered at various heights. The
question asks how many of the black markers are strictly inside the band, i.e.
have their vertical centre strictly between the two blue lines. This is a pure
*measurement* task: the pixel arm recovers the two band lines and each marker
vertical position, then counts the markers whose height sits between them.

``band_containment_count`` is a count decision, so it does not select a single
coloured extremum. The answer is a whole number (0..N), counterbalanced over the
arbitrary spatial scatter of the markers.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "band_containment_count-0.1.0"

# Marker palette (single colour) and the band colour; the oracle re-derives the
# palette from the fixed rendering constants without importing this module.
MARKER_COLOR = "black"
MARKER_RGB = (20, 20, 20)
BAND_RGB = (0, 0, 205)
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
N_MARKERS = 6
MARKER_RADIUS = 6
BAND_WIDTH = 5

# A marker centre must sit at least this far from either band line so the
# inside/outside decision is well above pixel-fit noise.
MIN_MARGIN_PX = 12.0
# Below this centre-to-bandline clearance a scene is quarantined as a
# finite-resolution boundary case, but it is still emitted and certified.
QUARANTINE_PX = 20.0
# Minimum clearance used when placing marker centres from a band line.
_ZONE_CLEARANCE = 16.0
_BAND_TOP_MIN = 55.0
_BAND_BOTTOM_MAX = 285.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed-clear count decision.

    Returns a plain JSON-serialisable dict with ``markers`` (list of {x, y})
    and ``band`` ({y_top, y_bottom}).
    """
    rng = random.Random(seed * 1009 + 7)
    while True:
        y_c = rng.uniform(120.0, 220.0)
        half = rng.uniform(45.0, 80.0)
        y_top = y_c - half
        y_bottom = y_c + half
        if y_top >= _BAND_TOP_MIN and y_bottom <= _BAND_BOTTOM_MAX:
            break

    xs = _scatter_x(rng)
    markers = []
    for i in range(N_MARKERS):
        if i == 0:
            zone = rng.choice(["above", "below"])
        else:
            zone = rng.choice(["above", "inside", "below"])
        x = xs[i]
        if zone == "above":
            y = rng.uniform(25.0, y_top - _ZONE_CLEARANCE)
        elif zone == "inside":
            y = rng.uniform(y_top + _ZONE_CLEARANCE, y_bottom - _ZONE_CLEARANCE)
        else:
            y = rng.uniform(y_bottom + _ZONE_CLEARANCE, 315.0)
        markers.append({"x": x, "y": y})

    scene = {
        "markers": markers,
        "band": {"y_top": float(y_top), "y_bottom": float(y_bottom)},
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }
    # Guarantee the analytic margin clears the minimum (defensive; the zone
    # clearance already does this).
    if margin(scene) < MIN_MARGIN_PX:
        return sample_scene(seed + 1)
    return scene


def _scatter_x(rng: random.Random) -> list[float]:
    """Return N x positions separated by at least ~42 px so markers don't touch."""
    xs: list[float] = []
    while len(xs) < N_MARKERS:
        candidate = rng.uniform(20.0, WIDTH - 20.0)
        if all(abs(candidate - taken) >= 42.0 for taken in xs):
            xs.append(candidate)
    return xs


def render(scene: dict) -> Image.Image:
    """Render I = R(z): two full-width blue lines plus N black circular markers."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    band = scene["band"]
    y_top = min(band["y_top"], band["y_bottom"])
    y_bottom = max(band["y_top"], band["y_bottom"])
    draw.line([(0, y_top), (scene["width"], y_top)], fill=BAND_RGB, width=BAND_WIDTH)
    draw.line([(0, y_bottom), (scene["width"], y_bottom)], fill=BAND_RGB, width=BAND_WIDTH)
    for m in scene["markers"]:
        r = MARKER_RADIUS
        draw.ellipse(
            [m["x"] - r, m["y"] - r, m["x"] + r, m["y"] + r],
            fill=MARKER_RGB,
        )
    return img


def _inside_count(markers: list[dict], band: dict) -> int:
    y_top = min(band["y_top"], band["y_bottom"])
    y_bottom = max(band["y_top"], band["y_bottom"])
    return int(sum(1 for m in markers if y_top < m["y"] < y_bottom))


def analytic_gold(scene: dict) -> int:
    """Number of markers whose vertical centre lies strictly inside the band."""
    return _inside_count(scene["markers"], scene["band"])


def margin(scene: dict) -> float:
    """Verifiability margin: least centre-to-bandline clearance over all markers."""
    band = scene["band"]
    y_top = min(band["y_top"], band["y_bottom"])
    y_bottom = max(band["y_top"], band["y_bottom"])
    best = float("inf")
    for m in scene["markers"]:
        best = min(best, abs(m["y"] - y_top), abs(m["y"] - y_bottom))
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: some marker sits uncomfortably close to a band line."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: the counted mark positions are all directly visible.

    Every latent field (each marker's x/y and the two band line heights) is
    drawn verbatim as ink, so there is no hidden scene field that could change
    the gold without changing the raster. No nontrivial pixel-identical latent
    transformation exists.
    """
    return []
