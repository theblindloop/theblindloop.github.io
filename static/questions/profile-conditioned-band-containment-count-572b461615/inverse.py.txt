"""Independent pixel-only inverse arm for the band-containment count world.

This module receives ONLY the rasterized PNG and re-measures how many black
markers sit strictly between the two blue band lines. It shares no geometry
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytic gold code.

Pipeline: (1) label each pixel to its nearest fixed colour (black markers, blue
band, light background); (2) recover the two blue line rows; (3) locate black
marker blobs by connected components; (4) count blobs whose vertical centre is
strictly between the two band rows.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "band_containment_count-oracle-0.1.0"

# Fixed colour book, shared conceptually with the renderer's naming convention.
MARKER_RGB = (20, 20, 20)
BAND_RGB = (0, 0, 205)
BACKGROUND = (250, 250, 250)


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest fixed colour (0=background,1=band,2=marker)."""
    refs = np.stack(
        [np.array(BACKGROUND, dtype=np.float32),
         np.array(BAND_RGB, dtype=np.float32),
         np.array(MARKER_RGB, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _band_rows(labels: np.ndarray) -> tuple[float, float]:
    """Recover (top, bottom) line rows from the blue (band=1) ink.

    The band is drawn as two horizontal lines, each a few rows thick. We take
    the set of rows that contain band-coloured pixels, split them into the two
    contiguous clusters by a large vertical gap, and return the two median rows.
    """
    rows = np.nonzero((labels == 1).any(axis=1))[0]
    if len(rows) == 0:
        return 0.0, 0.0
    clusters = []
    current = [int(rows[0])]
    for value in rows[1:]:
        value = int(value)
        if value - current[-1] <= 4:
            current.append(value)
        else:
            clusters.append(current)
            current = [value]
    clusters.append(current)
    clusters.sort(key=lambda cluster: -len(cluster))
    top_cluster = clusters[0]
    bottom_cluster = clusters[1] if len(clusters) > 1 else clusters[0]
    top = float(np.median(top_cluster))
    bottom = float(np.median(bottom_cluster))
    return (top, bottom) if top < bottom else (bottom, top)


def _marker_centres(labels: np.ndarray) -> list[float]:
    """Centroid y for each connected black-marker blob (marker=2 ink)."""
    ys, xs = np.nonzero(labels == 2)
    if len(ys) == 0:
        return []
    pixels = set(zip(xs.tolist(), ys.tolist()))
    centres: list[float] = []
    while pixels:
        seed = pixels.pop()
        stack = [(seed[0], seed[1])]
        comp_xs: list[int] = []
        comp_ys: list[int] = []
        while stack:
            px, py = stack.pop()
            comp_xs.append(px)
            comp_ys.append(py)
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nb = (px + dx, py + dy)
                if nb in pixels:
                    pixels.discard(nb)
                    stack.append(nb)
        centres.append(float(np.mean(comp_ys)))
    return centres


def decision_from_image(image: Image.Image) -> int:
    """Return how many black markers are strictly between the two band lines."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    top, bottom = _band_rows(labels)
    count = 0
    for y in _marker_centres(labels):
        if top < y < bottom:
            count += 1
    return int(count)
