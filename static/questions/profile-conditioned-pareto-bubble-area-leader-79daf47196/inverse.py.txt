"""Independent PNG-only inverse for the Pareto-frontier bubble task."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (35, 103, 205),
    "orange": (224, 112, 30),
    "green": (24, 151, 91),
    "purple": (126, 78, 190),
    "red": (205, 62, 71),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    output: list[np.ndarray] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append(np.asarray(pixels, dtype=float))
    return output


def _measure_bubble(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float, float]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    components = [component for component in _components(mask) if len(component) > 400]
    if len(components) != 1:
        raise ValueError("expected exactly one intact bubble for each answer color")
    pixels = components[0]
    ys, xs = pixels[:, 0], pixels[:, 1]
    span_x = float(xs.max() - xs.min() + 1)
    span_y = float(ys.max() - ys.min() + 1)
    if not 0.94 <= span_x / span_y <= 1.06:
        raise ValueError("colored mark is not circular")
    radius = math.sqrt(float(len(pixels)) / math.pi)
    if not 12.0 <= radius <= 38.0:
        raise ValueError("bubble radius is outside the public visual contract")
    return float(xs.mean()), float(ys.mean()), radius


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (620, 760):
        raise ValueError("unexpected canvas dimensions")
    measured = {color: _measure_bubble(arr, rgb) for color, rgb in _COLORS.items()}
    eligible = [
        color for color, (x, y, _radius) in measured.items()
        if not any(other != color and ox > x + 3.0 and oy < y - 3.0
                   for other, (ox, oy, _oradius) in measured.items())
    ]
    if len(eligible) < 2:
        raise ValueError("frontier is unresolved or implausibly degenerate")
    ordered = sorted(((color, measured[color][2]) for color in eligible), key=lambda item: item[1], reverse=True)
    if len(ordered) > 1 and ordered[0][1] - ordered[1][1] < 4.5:
        raise ValueError("frontier bubble areas are unresolved")
    return ordered[0][0]
