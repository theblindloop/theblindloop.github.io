"""Deterministic Pillow renderer for a Pareto-frontier bubble measurement task."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "blue": (35, 103, 205),
    "orange": (224, 112, 30),
    "green": (24, 151, 91),
    "purple": (126, 78, 190),
    "red": (205, 62, 71),
}
BACKGROUND = (247, 249, 251)
PANEL = (255, 255, 255)
FRAME = (65, 75, 88)
GRID = (219, 225, 231)
TEXT = (31, 37, 45)


def _frontier(scene: dict[str, Any]) -> list[dict[str, Any]]:
    items = scene["bubbles"]
    return [
        item for item in items
        if not any(
            other is not item
            and float(other["center"][0]) > float(item["center"][0])
            and float(other["center"][1]) < float(item["center"][1])
            for other in items
        )
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    names = list(COLORS)
    winner = names[int(seed) % len(names)]
    rest = [name for name in names if name != winner]
    rng.shuffle(rest)
    roles = {winner: "winner", rest[0]: "frontier", rest[1]: "large_decoy", rest[2]: "decoy_a", rest[3]: "decoy_b"}

    # In every mode the first two points trade off x against height; every other
    # point is strictly dominated. Odd modes swap which frontier position wins.
    mode = (int(seed) // len(names)) % 4
    frontier_pairs = [
        ((338, 220), (532, 354)),
        ((532, 354), (338, 220)),
        ((452, 206), (574, 350)),
        ((574, 350), (452, 206)),
    ]
    win_pos, frontier_pos = frontier_pairs[mode]
    positions = {
        "winner": win_pos,
        "frontier": frontier_pos,
        "large_decoy": (248, 336),
        "decoy_a": (418, 438),
        "decoy_b": (184, 470),
    }
    radii = {
        "winner": rng.randint(26, 29),
        "frontier": rng.randint(16, 19),
        "large_decoy": rng.randint(33, 36),
        "decoy_a": rng.randint(19, 23),
        "decoy_b": rng.randint(13, 17),
    }
    dx, dy = rng.randint(-14, 14), rng.randint(-10, 10)
    bubbles: list[dict[str, Any]] = []
    for color in names:
        role = roles[color]
        x, y = positions[role]
        bubbles.append({"color": color, "center": [x + dx, y + dy], "radius": radii[role]})
    rng.shuffle(bubbles)
    scene = {
        "width": 760,
        "height": 620,
        "plot_box": [88 + dx, 102 + dy, 686 + dx, 544 + dy],
        "bubbles": bubbles,
        "axis_width": 4,
    }
    if is_quarantined(scene):
        raise RuntimeError("internal scene template produced a quarantined scene")
    return scene


def analytic_gold(scene: dict[str, Any]) -> str:
    eligible = _frontier(scene)
    ordered = sorted(eligible, key=lambda item: float(item["radius"]), reverse=True)
    if len(ordered) < 1 or (len(ordered) > 1 and float(ordered[0]["radius"]) == float(ordered[1]["radius"])):
        return "abstain"
    return str(ordered[0]["color"])


def margin(scene: dict[str, Any]) -> float:
    eligible_radii = sorted((float(item["radius"]) for item in _frontier(scene)), reverse=True)
    radius_clearance = eligible_radii[0] if len(eligible_radii) == 1 else eligible_radii[0] - eligible_radii[1]
    items = scene["bubbles"]
    coordinate_clearance = min(
        min(abs(float(items[i]["center"][0]) - float(items[j]["center"][0])),
            abs(float(items[i]["center"][1]) - float(items[j]["center"][1])))
        for i in range(len(items)) for j in range(i + 1, len(items))
    )
    return float(min(radius_clearance, coordinate_clearance))


def is_quarantined(scene: dict[str, Any]) -> bool:
    items = scene["bubbles"]
    if len(items) != len(COLORS) or {str(item["color"]) for item in items} != set(COLORS):
        return True
    if analytic_gold(scene) == "abstain" or len(_frontier(scene)) < 2 or margin(scene) < 6.0:
        return True
    left, top, right, bottom = map(float, scene["plot_box"])
    for item in items:
        x, y = map(float, item["center"])
        radius = float(item["radius"])
        if min(x - radius - left, right - x - radius, y - radius - top, bottom - y - radius) < 18.0:
            return True
    for i in range(len(items)):
        for j in range(i + 1, len(items)):
            if math.dist(items[i]["center"], items[j]["center"]) - float(items[i]["radius"]) - float(items[j]["radius"]) < 24.0:
                return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    transformed = copy.deepcopy(scene)
    transformed["bubbles"] = list(reversed(transformed["bubbles"]))
    return [("reverse_nonoverlapping_bubble_records", transformed)]


def _font(name: str, size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font("DejaVuSans-Bold.ttf", 20)
    note_font = _font("DejaVuSans.ttf", 14)
    axis_font = _font("DejaVuSans-Bold.ttf", 14)
    draw.text((34, 20), "EFFICIENCY FRONTIER", fill=TEXT, font=title_font)
    draw.text((34, 51), "Upper-right is better on both measures", fill=TEXT, font=note_font)
    left, top, right, bottom = map(int, scene["plot_box"])
    draw.rectangle((left, top, right, bottom), fill=PANEL, outline=FRAME, width=int(scene["axis_width"]))
    for fraction in (0.25, 0.5, 0.75):
        x = int(round(left + fraction * (right - left)))
        y = int(round(top + fraction * (bottom - top)))
        draw.line((x, top + 3, x, bottom - 3), fill=GRID, width=1)
        draw.line((left + 3, y, right - 3, y), fill=GRID, width=1)
    draw.text((right - 76, bottom + 18), "MEASURE X", fill=TEXT, font=axis_font)
    draw.text((left - 54, top - 25), "HIGH Y", fill=TEXT, font=axis_font)
    for item in scene["bubbles"]:
        x, y = map(int, item["center"])
        radius = int(item["radius"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[str(item["color"])])
    return image
