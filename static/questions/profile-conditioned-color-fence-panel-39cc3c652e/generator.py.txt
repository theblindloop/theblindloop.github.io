"""Colour-fence panel world.

Latent scene z = {width, height, palette, dot_radius, fence_width, panels}.

Four square panels sit side by side. Each panel holds one straight fence line
drawn right across it and the same number of small filled dots, half of them in
a warm colour and half in a cool colour. A panel is cleanly fenced when every
warm dot lies on one side of its own fence and every cool dot lies on the other
side. Exactly one panel is cleanly fenced; in every other panel at least two
dots sit on the wrong side under either reading of the fence.

The decision is the cleanly fenced panel. The margin is measured in pixels: the
smallest distance from any dot centre to its own fence, capped by ten pixels per
surplus wrong-side dot in the closest competing panel.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "color_fence_panel-0.1.0"

WIDTH = 800
HEIGHT = 200
PANEL_COUNT = 4
PANEL_WIDTH = WIDTH // PANEL_COUNT
DOT_RADIUS = 7
FENCE_WIDTH = 3
SUPERSAMPLE = 3

DOTS_PER_PANEL = 10
EDGE_INSET = 18.0
MIN_DOT_SEPARATION = 22.0
MIN_FENCE_CLEARANCE = 14.0
MIN_WRONG_SIDE = 2
QUARANTINE_MARGIN = 14.5

PANELS = ("panel_1", "panel_2", "panel_3", "panel_4")

PALETTES = (
    {
        "background": (255, 255, 255),
        "warm": (178, 40, 34),
        "cool": (34, 62, 170),
        "fence": (38, 38, 42),
        "frame": (178, 178, 178),
    },
    {
        "background": (248, 246, 240),
        "warm": (168, 62, 20),
        "cool": (26, 88, 148),
        "fence": (44, 44, 46),
        "frame": (172, 174, 176),
    },
    {
        "background": (244, 247, 250),
        "warm": (162, 34, 74),
        "cool": (30, 76, 158),
        "fence": (34, 40, 44),
        "frame": (170, 178, 186),
    },
    {
        "background": (250, 250, 244),
        "warm": (186, 66, 26),
        "cool": (24, 70, 176),
        "fence": (40, 36, 34),
        "frame": (182, 178, 168),
    },
)


def _panel_index(cx: float) -> int:
    return min(PANEL_COUNT - 1, max(0, int(cx // PANEL_WIDTH)))


def _signed_offset(panel: dict, dot: dict) -> float:
    """Signed distance from a dot centre to its panel fence, in pixels."""
    theta = math.radians(panel["fence_deg"])
    nx = -math.sin(theta)
    ny = math.cos(theta)
    return (dot["x"] - panel["fence_x"]) * nx + (dot["y"] - panel["fence_y"]) * ny


def _wrong_side(panel: dict) -> int:
    """Fewest dots that sit on the wrong side under either reading of the fence."""
    warm_positive = sum(
        1 for dot in panel["dots"] if dot["warm"] and _signed_offset(panel, dot) > 0.0
    )
    warm_negative = sum(1 for dot in panel["dots"] if dot["warm"]) - warm_positive
    cool_positive = sum(
        1 for dot in panel["dots"] if not dot["warm"] and _signed_offset(panel, dot) > 0.0
    )
    cool_negative = sum(1 for dot in panel["dots"] if not dot["warm"]) - cool_positive
    return int(min(warm_positive + cool_negative, warm_negative + cool_positive))


def _clearance(panel: dict) -> float:
    return min(abs(_signed_offset(panel, dot)) for dot in panel["dots"])


def _reports(scene: dict) -> dict:
    """Per panel lane: wrong-side count under the better polarity, and clearance."""
    reports: dict = {}
    for panel in scene["panels"]:
        name = PANELS[_panel_index(panel["cx"])]
        if name in reports:
            raise ValueError("two records share one panel lane")
        theta = math.radians(panel["fence_deg"])
        nx = -math.sin(theta)
        ny = math.cos(theta)
        offsets = [
            (
                (dot["x"] - panel["fence_x"]) * nx + (dot["y"] - panel["fence_y"]) * ny,
                bool(dot["warm"]),
            )
            for dot in panel["dots"]
        ]
        if not offsets:
            raise ValueError("panel holds no dot")
        warm_positive = sum(1 for offset, warm in offsets if warm and offset > 0.0)
        warm_negative = sum(1 for offset, warm in offsets if warm and offset <= 0.0)
        cool_positive = sum(1 for offset, warm in offsets if not warm and offset > 0.0)
        cool_negative = sum(1 for offset, warm in offsets if not warm and offset <= 0.0)
        reports[name] = {
            "wrong": int(min(warm_positive + cool_negative, warm_negative + cool_positive)),
            "clearance": min(abs(offset) for offset, _warm in offsets),
        }
    return reports


def analytic_gold(scene: dict) -> str:
    """Panel whose own fence separates warm dots from cool dots without error."""
    reports = _reports(scene)
    if len(reports) != PANEL_COUNT:
        raise ValueError("scene does not populate every panel exactly once")
    clean = [name for name, report in reports.items() if report["wrong"] == 0]
    if len(clean) != 1:
        raise ValueError(f"scene has no unique cleanly fenced panel: {sorted(clean)}")
    return clean[0]


def margin(scene: dict) -> float:
    """Smallest pixel clearance in the image, capped by the competing panels."""
    reports = _reports(scene)
    if not reports:
        raise ValueError("scene has no panel")
    closest = min(report["clearance"] for report in reports.values())
    surplus = min(
        (report["wrong"] for report in reports.values() if report["wrong"] > 0),
        default=0,
    )
    return round(min(closest, 10.0 * surplus), 6)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes whose tightest clearance is hard to resolve."""
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict):
    """Storage aliases: record order and the two-sided fence angle representative."""
    reordered = dict(scene, panels=[scene["panels"][i] for i in (2, 0, 3, 1)])
    reversed_order = dict(scene, panels=list(reversed(scene["panels"])))
    dots_shuffled = dict(
        scene,
        panels=[dict(panel, dots=list(reversed(panel["dots"]))) for panel in scene["panels"]],
    )
    half_turn = dict(
        scene,
        panels=[dict(panel, fence_deg=panel["fence_deg"] + 180.0) for panel in scene["panels"]],
    )
    return [
        ("panel_record_order", reordered),
        ("panel_record_reversal", reversed_order),
        ("dot_record_order", dots_shuffled),
        ("fence_half_turn_representative", half_turn),
    ]


def _sample_positions(rng: random.Random, panel_left: float, fence: tuple) -> list:
    """Well separated dot centres that keep clear of the fence and the frame."""
    fence_x, fence_y, fence_deg = fence
    probe = {"fence_x": fence_x, "fence_y": fence_y, "fence_deg": fence_deg}
    points: list = []
    for _ in range(4000):
        x = panel_left + EDGE_INSET + rng.randrange(int(PANEL_WIDTH - 2 * EDGE_INSET) + 1)
        y = EDGE_INSET + rng.randrange(int(HEIGHT - 2 * EDGE_INSET) + 1)
        candidate = {"x": float(x), "y": float(y)}
        if abs(_signed_offset(probe, candidate)) < MIN_FENCE_CLEARANCE:
            continue
        if any(math.hypot(x - p["x"], y - p["y"]) < MIN_DOT_SEPARATION for p in points):
            continue
        points.append(candidate)
        if len(points) == DOTS_PER_PANEL:
            return points
    return []


def _sample_panel(rng: random.Random, index: int, clean: bool, warm_first: bool) -> dict:
    """One panel: a fence across it plus dots that are clean or corrupted."""
    panel_left = float(index * PANEL_WIDTH)
    for _ in range(400):
        fence_deg = float(rng.randrange(360))
        fence_x = panel_left + PANEL_WIDTH / 2.0 + rng.randrange(-34, 35)
        fence_y = HEIGHT / 2.0 + rng.randrange(-30, 31)
        points = _sample_positions(rng, panel_left, (fence_x, fence_y, fence_deg))
        if not points:
            continue
        panel = {
            "cx": panel_left + PANEL_WIDTH / 2.0,
            "fence_x": float(fence_x),
            "fence_y": float(fence_y),
            "fence_deg": fence_deg,
            "dots": points,
        }
        positive = [dot for dot in points if _signed_offset(panel, dot) > 0.0]
        negative = [dot for dot in points if _signed_offset(panel, dot) <= 0.0]
        if min(len(positive), len(negative)) < 3:
            continue
        for dot in positive:
            dot["warm"] = warm_first
        for dot in negative:
            dot["warm"] = not warm_first
        if not clean:
            flips = rng.choice((MIN_WRONG_SIDE, MIN_WRONG_SIDE + 1))
            larger = positive if len(positive) >= len(negative) else negative
            if len(larger) - flips < 3:
                continue
            for dot in rng.sample(larger, flips):
                dot["warm"] = not dot["warm"]
        warm_count = sum(1 for dot in points if dot["warm"])
        if not 3 <= warm_count <= DOTS_PER_PANEL - 3:
            continue
        expected = 0 if clean else _wrong_side(panel)
        if clean and _wrong_side(panel) != 0:
            continue
        if not clean and expected < MIN_WRONG_SIDE:
            continue
        rng.shuffle(panel["dots"])
        return panel
    raise RuntimeError("could not sample a panel")


def sample_scene(seed: int) -> dict:
    rng = random.Random(repr(("color_fence_panel", seed)))
    target = seed % PANEL_COUNT
    palette_index = rng.randrange(len(PALETTES))
    for _ in range(200):
        panels = []
        try:
            for index in range(PANEL_COUNT):
                warm_first = rng.choice((True, False))
                panels.append(_sample_panel(rng, index, index == target, warm_first))
        except RuntimeError:
            continue
        rng.shuffle(panels)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "palette": palette_index,
            "dot_radius": DOT_RADIUS,
            "fence_width": FENCE_WIDTH,
            "panels": panels,
        }
        try:
            if analytic_gold(scene) != PANELS[target]:
                continue
        except ValueError:
            continue
        if margin(scene) < MIN_FENCE_CLEARANCE:
            continue
        return scene
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def _fence_endpoints(panel: dict) -> tuple:
    """Canonical clipped endpoints of a panel fence, independent of angle sign."""
    left = panel["cx"] - PANEL_WIDTH / 2.0
    right = panel["cx"] + PANEL_WIDTH / 2.0
    theta = math.radians(panel["fence_deg"])
    dx = math.cos(theta)
    dy = math.sin(theta)
    hits = []
    for bound, axis in ((left, 0), (right, 0), (0.0, 1), (float(HEIGHT), 1)):
        step = dx if axis == 0 else dy
        if abs(step) < 1e-9:
            continue
        origin = panel["fence_x"] if axis == 0 else panel["fence_y"]
        t = (bound - origin) / step
        x = panel["fence_x"] + t * dx
        y = panel["fence_y"] + t * dy
        if left - 1e-6 <= x <= right + 1e-6 and -1e-6 <= y <= HEIGHT + 1e-6:
            hits.append((round(x, 6), round(y, 6)))
    hits = sorted(set(hits))
    if len(hits) < 2:
        raise ValueError("fence does not cross its panel")
    return hits[0], hits[-1]


def render(scene: dict) -> Image.Image:
    palette = PALETTES[scene["palette"] % len(PALETTES)]
    width = scene["width"]
    height = scene["height"]
    radius = scene["dot_radius"]
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), palette["background"])
    draw = ImageDraw.Draw(image)
    line = max(1, ss // 2)

    draw.rectangle(
        [(0, 0), (width * ss - 1, height * ss - 1)], outline=palette["frame"], width=line
    )
    for index in range(1, PANEL_COUNT):
        x = index * PANEL_WIDTH * ss
        draw.line([(x, 0), (x, height * ss)], fill=palette["frame"], width=line)

    for panel in sorted(scene["panels"], key=lambda item: item["cx"]):
        start, end = _fence_endpoints(panel)
        draw.line(
            [(start[0] * ss, start[1] * ss), (end[0] * ss, end[1] * ss)],
            fill=palette["fence"],
            width=scene["fence_width"] * ss,
        )
    for panel in sorted(scene["panels"], key=lambda item: item["cx"]):
        for dot in sorted(panel["dots"], key=lambda item: (item["x"], item["y"])):
            colour = palette["warm"] if dot["warm"] else palette["cool"]
            cx = dot["x"] * ss
            cy = dot["y"] * ss
            draw.ellipse(
                [(cx - radius * ss, cy - radius * ss), (cx + radius * ss, cy + radius * ss)],
                fill=colour,
            )
    return image.resize((width, height), Image.LANCZOS)
