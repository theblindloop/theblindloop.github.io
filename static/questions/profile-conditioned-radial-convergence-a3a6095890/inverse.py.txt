"""Independent pixel-only inverse arm for the rod-concurrence world.

Recovers each thin dark rod's supporting line from the final raster alone (by
connected-component analysis and principal-axis line fitting) and decides
whether all the extended lines converge on a single shared center point (yes)
or point in unrelated directions (no). It receives only the image, so the
answer is recoverable from the pixels -- the implicit global concurrence of the
extended rod axes -- without any latent scene input.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

MIN_PIXELS = 20
RESIDUAL_THRESHOLD = 1.2


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    labelled = np.zeros((h, w), dtype=bool)
    components = []
    for sy, sx in np.argwhere(mask):
        if labelled[sy, sx]:
            continue
        comp = []
        stack = [(int(sy), int(sx))]
        labelled[sy, sx] = True
        while stack:
            y, x = stack.pop()
            comp.append((y, x))
            if y + 1 < h and mask[y + 1, x] and not labelled[y + 1, x]:
                labelled[y + 1, x] = True
                stack.append((y + 1, x))
            if y - 1 >= 0 and mask[y - 1, x] and not labelled[y - 1, x]:
                labelled[y - 1, x] = True
                stack.append((y - 1, x))
            if x + 1 < w and mask[y, x + 1] and not labelled[y, x + 1]:
                labelled[y, x + 1] = True
                stack.append((y, x + 1))
            if x - 1 >= 0 and mask[y, x - 1] and not labelled[y, x - 1]:
                labelled[y, x - 1] = True
                stack.append((y, x - 1))
        if len(comp) >= MIN_PIXELS:
            components.append(comp)
    return components


def _lines_from_mask(mask: np.ndarray) -> list[tuple[float, float, float]]:
    lines = []
    for comp in _components(mask):
        pts = np.asarray(comp, dtype=float)
        ys = pts[:, 0]
        xs = pts[:, 1]
        xc = xs.mean()
        yc = ys.mean()
        xx = ((xs - xc) * (xs - xc)).sum()
        yy = ((ys - yc) * (ys - yc)).sum()
        xy = ((xs - xc) * (ys - yc)).sum()
        cov = np.array([[xx, xy], [xy, yy]])
        vals, vecs = np.linalg.eigh(cov)
        v = vecs[:, int(np.argmax(vals))]
        nx, ny = -v[1], v[0]
        length = float(np.sqrt(nx * nx + ny * ny))
        if length < 1e-9:
            continue
        nx /= length
        ny /= length
        d = -(nx * xc + ny * yc)
        lines.append((float(nx), float(ny), float(d)))
    return lines


def _concurrence_residual(mask: np.ndarray) -> float:
    lines = _lines_from_mask(mask)
    if len(lines) < 3:
        return float(RESIDUAL_THRESHOLD * 4.0)
    ns = np.array([[nx, ny] for nx, ny, _ in lines])
    ds = np.array([d for _, _, d in lines])
    a = ns.T @ ns
    b = ds @ ns
    try:
        point = -np.linalg.solve(a, b)
    except np.linalg.LinAlgError:
        return float(RESIDUAL_THRESHOLD * 4.0)
    return float((((ns @ point) + ds) ** 2).mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    mask = (arr[..., 0].astype(int) < 150) & (arr[..., 1].astype(int) < 150)
    residual = _concurrence_residual(mask)
    return "yes" if residual < RESIDUAL_THRESHOLD else "no"
