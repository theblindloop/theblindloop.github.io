"""Rod-concurrence world: a hidden global ray-convergence point.

A set of thin dark rods is scattered across a white canvas. In the "convergent"
state every rod sits on its own radial line and all of those extended lines pass
through one shared, undrawn center point -- the rods radiate like the spokes of
an implicit wheel. In the "divergent" state the rods have unrelated orientations
so their extended lines share no common point. The figure to reveal is the
implicit global concurrence of the extended axes, not any single object, dot
count, colour, or the (never drawn) center itself.

The renderer owns scene sampling, rasterization, and the structural analytic
gold (the boolean agreement flag). The independent pixel-only inverse arm
(``oracle.py``) reconstructs each rod's supporting line from the raster and
measures how closely all the extended lines concur on a single best point.
"""

from __future__ import annotations

import numpy as np
from collections import deque
from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
CENTRE_X = WIDTH // 2
CENTRE_Y = HEIGHT // 2
ROD_LEN = 64          # rod length in pixels
ROD_WIDTH = 5         # rod thickness in pixels
MIN_SEP = 16          # minimum pixel distance between rod centres
MIN_ANG_GAP = 0.44    # minimum angular gap between convergent rods (radians)
CLEARANCE = 7.0       # minimum pixel gap between any two rod segments
BACKGROUND = (255, 255, 255)
ROD_COLOUR = (30, 40, 90)
MIN_PIXELS = 20       # minimum connected-component size to count as a rod
RESIDUAL_THRESHOLD = 1.2   # mean squared concurrence residual (px^2)
MARGIN_QUARANTINE = 0.30   # ambiguity band around the residual threshold

_MARGIN = 40


def _seg_point_dist(px, py, ax, ay, bx, by) -> float:
    dx = bx - ax
    dy = by - ay
    l2 = dx * dx + dy * dy
    if l2 == 0.0:
        return float(np.hypot(px - ax, py - ay))
    t = ((px - ax) * dx + (py - ay) * dy) / l2
    t = min(1.0, max(0.0, t))
    qx = ax + t * dx
    qy = ay + t * dy
    return float(np.hypot(px - qx, py - qy))


def _seg_seg_dist(a1, a2, b1, b2) -> float:
    (ax, ay), (bx, by) = a1, a2
    (cx, cy), (dx2, dy2) = b1, b2
    best = min(
        _seg_point_dist(cx, cy, ax, ay, bx, by),
        _seg_point_dist(dx2, dy2, ax, ay, bx, by),
        _seg_point_dist(ax, ay, cx, cy, dx2, dy2),
        _seg_point_dist(bx, by, cx, cy, dx2, dy2),
    )
    return best


def _place_rod(rng, rod_count: int, rng_mode: str):
    shared_cx = CENTRE_X + int(rng.integers(-10, 11))
    shared_cy = CENTRE_Y + int(rng.integers(-10, 11))
    rods = []
    thetas = []
    attempts = 0
    while len(rods) < rod_count and attempts < rod_count * 5000:
        attempts += 1
        if rng_mode == "convergent":
            cx, cy = shared_cx, shared_cy
            theta = float(rng.uniform(0.0, 2.0 * np.pi))
            radius = float(rng.uniform(62.0, 95.0))
            ux = float(np.cos(theta))
            uy = float(np.sin(theta))
            mx = cx + radius * ux
            my = cy + radius * uy
            ang = theta
            angle_ok = True
            for prior in thetas:
                diff = abs((theta - prior + np.pi) % (2.0 * np.pi) - np.pi)
                if diff < MIN_ANG_GAP:
                    angle_ok = False
                    break
            if not angle_ok:
                continue
        else:
            mx = float(rng.uniform(_MARGIN + 20, WIDTH - _MARGIN - 20))
            my = float(rng.uniform(_MARGIN + 20, HEIGHT - _MARGIN - 20))
            ang = float(rng.uniform(0.0, np.pi))
            ux = float(np.cos(ang))
            uy = float(np.sin(ang))
            cx, cy = 0.0, 0.0
        half = ROD_LEN / 2.0
        x1 = mx - half * ux
        y1 = my - half * uy
        x2 = mx + half * ux
        y2 = my + half * uy
        seg = ((x1, y1), (x2, y2))
        if _MARGIN + 2 > min(x1, x2) - ROD_WIDTH or max(x1, x2) + ROD_WIDTH >= WIDTH - _MARGIN:
            continue
        if _MARGIN + 2 > min(y1, y2) - ROD_WIDTH or max(y1, y2) + ROD_WIDTH >= HEIGHT - _MARGIN:
            continue
        place_ok = True
        for rod in rods:
            other = ((rod["x1"], rod["y1"]), (rod["x2"], rod["y2"]))
            if _seg_seg_dist(seg[0], seg[1], other[0], other[1]) < CLEARANCE:
                place_ok = False
                break
        if not place_ok:
            continue
        rods.append(
            {
                "x1": round(x1, 4),
                "y1": round(y1, 4),
                "x2": round(x2, 4),
                "y2": round(y2, 4),
            }
        )
        thetas.append(ang)
    return rods


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    convergent = bool(rng.integers(0, 2))
    rod_count = int(rng.integers(6, 10))
    rods = _place_rod(rng, rod_count, "convergent" if convergent else "divergent")
    return {"convergent": convergent, "rods": rods}


def _draw_scene(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    if not scene["convergent"] in (True, False):
        raise ValueError(f"unknown convergence flag {scene.get('convergent')!r}")
    draw = ImageDraw.Draw(image)
    for rod in scene["rods"]:
        draw.line(
            [
                (rod["x1"], rod["y1"]),
                (rod["x2"], rod["y2"]),
            ],
            fill=ROD_COLOUR,
            width=ROD_WIDTH,
        )
    return image


def render(scene: dict) -> Image.Image:
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["convergent"] else "no"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    labelled = np.zeros((h, w), dtype=bool)
    components = []
    for sy, sx in np.argwhere(mask):
        if labelled[sy, sx]:
            continue
        comp = []
        stack = [(int(sy), int(sx))]
        labelled[sy, sx] = True
        while stack:
            y, x = stack.pop()
            comp.append((y, x))
            if y + 1 < h and mask[y + 1, x] and not labelled[y + 1, x]:
                labelled[y + 1, x] = True
                stack.append((y + 1, x))
            if y - 1 >= 0 and mask[y - 1, x] and not labelled[y - 1, x]:
                labelled[y - 1, x] = True
                stack.append((y - 1, x))
            if x + 1 < w and mask[y, x + 1] and not labelled[y, x + 1]:
                labelled[y, x + 1] = True
                stack.append((y, x + 1))
            if x - 1 >= 0 and mask[y, x - 1] and not labelled[y, x - 1]:
                labelled[y, x - 1] = True
                stack.append((y, x - 1))
        if len(comp) >= MIN_PIXELS:
            components.append(comp)
    return components


def _lines_from_mask(mask: np.ndarray) -> list[tuple[float, float, float]]:
    """Return unit-normal (nx, ny, d) lines for each rod's principal axis."""
    lines = []
    for comp in _components(mask):
        pts = np.asarray(comp, dtype=float)
        ys = pts[:, 0]
        xs = pts[:, 1]
        n = len(pts)
        xc = xs.mean()
        yc = ys.mean()
        xx = ((xs - xc) * (xs - xc)).sum()
        yy = ((ys - yc) * (ys - yc)).sum()
        xy = ((xs - xc) * (ys - yc)).sum()
        cov = np.array([[xx, xy], [xy, yy]])
        vals, vecs = np.linalg.eigh(cov)
        v = vecs[:, int(np.argmax(vals))]
        nx, ny = -v[1], v[0]
        length = float(np.sqrt(nx * nx + ny * ny))
        if length < 1e-9:
            continue
        nx /= length
        ny /= length
        d = -(nx * xc + ny * yc)
        lines.append((float(nx), float(ny), float(d)))
    return lines


def _concurrence_residual(mask: np.ndarray) -> float:
    """Mean squared distance of every rod line to the best-shared point."""
    lines = _lines_from_mask(mask)
    if len(lines) < 3:
        return float(RESIDUAL_THRESHOLD * 4.0)
    ns = np.array([[nx, ny] for nx, ny, _ in lines])
    ds = np.array([d for _, _, d in lines])
    a = ns.T @ ns
    b = ds @ ns
    try:
        point = -np.linalg.solve(a, b)
    except np.linalg.LinAlgError:
        return float(RESIDUAL_THRESHOLD * 4.0)
    residual = float((((ns @ point) + ds) ** 2).mean())
    return residual


def margin(scene: dict) -> float:
    arr = np.asarray(_draw_scene(scene).convert("RGB"))
    mask = (arr[..., 0].astype(int) < 150) & (arr[..., 1].astype(int) < 150)
    residual = _concurrence_residual(mask)
    return float(RESIDUAL_THRESHOLD - residual)


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
