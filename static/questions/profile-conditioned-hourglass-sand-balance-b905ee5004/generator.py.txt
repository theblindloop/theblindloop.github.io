"""Deterministic renderer for the hourglass sand-balance question world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 512


def sample_scene(seed: int) -> dict:
    """Sample balanced top/bottom outcomes with independently varied styling."""
    rng = random.Random(seed)
    low = rng.choice((2, 3, 4))
    gap = rng.choice((1, 2, 3, 4))
    high = min(8, low + gap)
    if seed % 2 == 0:
        top_units, bottom_units = high, low
    else:
        top_units, bottom_units = low, high
    return {
        "top_units": top_units,
        "bottom_units": bottom_units,
        "frame_width": rng.choice((190, 210, 230)),
        "shift_x": rng.choice((-18, 0, 18)),
        "palette_id": rng.choice((0, 1)),
    }


def _sand_polygon(cx: int, units: int, top: bool) -> list[tuple[int, int]]:
    width = 42 + 14 * units
    height = 16 + 5 * units
    shoulder = int(round(width * 0.34))
    if top:
        base_y = 216
        return [
            (cx - width // 2, base_y - height),
            (cx + width // 2, base_y - height),
            (cx + shoulder, base_y),
            (cx - shoulder, base_y),
        ]
    base_y = 414
    return [
        (cx - shoulder, base_y - height),
        (cx + shoulder, base_y - height),
        (cx + width // 2, base_y),
        (cx - width // 2, base_y),
    ]


def _polygon_area(points: list[tuple[int, int]]) -> float:
    total = 0
    for index, (x1, y1) in enumerate(points):
        x2, y2 = points[(index + 1) % len(points)]
        total += x1 * y2 - x2 * y1
    return abs(total) / 2.0


def render(scene: dict) -> Image.Image:
    top_units = int(scene["top_units"])
    bottom_units = int(scene["bottom_units"])
    frame_width = int(scene["frame_width"])
    shift_x = int(scene["shift_x"])
    palette_id = int(scene["palette_id"])

    image = Image.new("RGB", (CANVAS, CANVAS), (248, 246, 239))
    draw = ImageDraw.Draw(image)
    cx = CANVAS // 2 + shift_x
    half = frame_width // 2
    ink = (48, 45, 42)
    frame = (176, 108, 34) if palette_id == 0 else (142, 82, 42)
    glass = (229, 239, 239)
    sand = (26, 111, 226) if palette_id == 0 else (4, 151, 190)

    top_glass = [(cx - half + 12, 83), (cx + half - 12, 83), (cx + 27, 238), (cx - 27, 238)]
    bottom_glass = [(cx - 27, 274), (cx + 27, 274), (cx + half - 12, 429), (cx - half + 12, 429)]
    draw.polygon(top_glass, fill=glass, outline=ink, width=5)
    draw.polygon(bottom_glass, fill=glass, outline=ink, width=5)

    # The two solid regions are deliberately separated, directly measurable,
    # and drawn above the glass but beneath no other marks.
    draw.polygon(_sand_polygon(cx, top_units, True), fill=sand)
    draw.polygon(_sand_polygon(cx, bottom_units, False), fill=sand)

    # Familiar hourglass frame, constant across the two answer classes.
    draw.rounded_rectangle((cx - half, 65, cx + half, 88), radius=7, fill=frame, outline=ink, width=3)
    draw.rounded_rectangle((cx - half, 424, cx + half, 447), radius=7, fill=frame, outline=ink, width=3)
    draw.line((cx - half + 7, 82, cx - half + 7, 430), fill=frame, width=11)
    draw.line((cx + half - 7, 82, cx + half - 7, 430), fill=frame, width=11)
    draw.ellipse((cx - 31, 236, cx + 31, 276), fill=frame, outline=ink, width=3)
    draw.ellipse((cx - 17, 246, cx + 17, 266), fill=(248, 246, 239), outline=ink, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    top_units = int(scene["top_units"])
    bottom_units = int(scene["bottom_units"])
    if top_units == bottom_units:
        return "equal"
    return "top" if top_units > bottom_units else "bottom"


def margin(scene: dict) -> float:
    cx = CANVAS // 2
    top_area = _polygon_area(_sand_polygon(cx, int(scene["top_units"]), True))
    bottom_area = _polygon_area(_sand_polygon(cx, int(scene["bottom_units"]), False))
    return float(abs(top_area - bottom_area))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 400.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
