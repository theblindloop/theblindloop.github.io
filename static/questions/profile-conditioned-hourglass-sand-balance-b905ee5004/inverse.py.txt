"""Independent PNG-only inverse arm for visible blue sand area."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        count = 0
        sum_x = 0
        sum_y = 0
        while queue:
            yy, xx = queue.popleft()
            count += 1
            sum_x += xx
            sum_y += yy
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 300:
            found.append((count, sum_x / count, sum_y / count))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    # Covers both renderer palettes while excluding brown frame, dark outline,
    # pale glass, and off-white background.
    mask = (blue >= 150) & (blue > red + 55) & (green > red + 18)
    regions = _components(mask)
    if len(regions) != 2:
        raise ValueError(f"expected exactly two substantial blue regions, found {len(regions)}")
    regions.sort(key=lambda item: item[2])
    top_area, _, top_y = regions[0]
    bottom_area, _, bottom_y = regions[1]
    if bottom_y - top_y < image.height * 0.18:
        raise ValueError("blue regions are not visibly separated into top and bottom chambers")
    difference = top_area - bottom_area
    if abs(difference) < 200:
        return "equal"
    return "top" if difference > 0 else "bottom"
