from __future__ import annotations

import math
import random
from collections import deque

import numpy as np
from PIL import Image, ImageDraw


PALETTES = [
    {"background": [250, 249, 246], "dot": [24, 27, 31], "scale": [112, 116, 122]},
    {"background": [246, 249, 250], "dot": [31, 25, 29], "scale": [103, 110, 118]},
    {"background": [249, 247, 251], "dot": [20, 30, 29], "scale": [119, 113, 124]},
]


def _polar(center: tuple[int, int], radius: float, angle: float, jitter: float) -> list[int]:
    d = radius + jitter
    return [int(round(center[0] + d * math.cos(angle))), int(round(center[1] + d * math.sin(angle)))]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    marker_radius = rng.choice([5, 6, 7])
    influence_radius = rng.randint(48, 64)
    origin = (rng.randint(226, 286), rng.randint(205, 255))
    rotation = rng.uniform(0, 2 * math.pi)
    label_yes = (int(seed) % 2 == 0)
    centers: list[list[int]] = []

    if label_yes:
        ring_radius = rng.uniform(1.92, 2.12) * influence_radius
        for i in range(12):
            a = rotation + 2 * math.pi * i / 12
            centers.append(_polar(origin, ring_radius, a, rng.uniform(-2.0, 2.0)))
    elif ((int(seed) // 2) % 2) == 0:
        # A visibly scaled but incomplete union: one angular opening exceeds 2R.
        ring_radius = rng.uniform(2.55, 2.72) * influence_radius
        large_gap = rng.uniform(0.82, 0.90)
        small_gap = (2 * math.pi - large_gap) / 11
        angle = rotation
        for i in range(12):
            centers.append(_polar(origin, ring_radius, angle, rng.uniform(-1.5, 1.5)))
            angle += large_gap if i == 0 else small_gap
    else:
        # The outside chain closes, but an identically styled central disk fills its cavity.
        ring_radius = rng.uniform(1.68, 1.82) * influence_radius
        for i in range(11):
            a = rotation + 2 * math.pi * i / 11
            centers.append(_polar(origin, ring_radius, a, rng.uniform(-1.5, 1.5)))
        centers.append([origin[0] + rng.randint(-3, 3), origin[1] + rng.randint(-3, 3)])

    rng.shuffle(centers)
    return {
        "centers": centers,
        "influence_radius": influence_radius,
        "width": width,
        "height": height,
        "marker_radius": marker_radius,
        "palette": rng.choice(PALETTES),
    }


def render(scene: dict) -> Image.Image:
    centers = scene["centers"]
    influence_radius = int(scene["influence_radius"])
    width = int(scene["width"])
    height = int(scene["height"])
    marker_radius = int(scene["marker_radius"])
    palette = scene["palette"]
    image = Image.new("RGB", (width, height), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    dot = tuple(palette["dot"])
    for x, y in centers:
        draw.ellipse(
            [x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius],
            fill=dot,
        )
    x0 = (width - influence_radius) // 2
    y0 = height - 35
    draw.rectangle([x0, y0, x0 + influence_radius - 1, y0 + 6], fill=tuple(palette["scale"]))
    return image


def _has_bounded_complement(centers: list[list[int]], radius: int, width: int, height: int) -> bool:
    pad = 3
    x0 = max(0, int(min(p[0] for p in centers) - radius - pad))
    x1 = min(width - 1, int(max(p[0] for p in centers) + radius + pad))
    y0 = max(0, int(min(p[1] for p in centers) - radius - pad))
    y1 = min(height - 1, int(max(p[1] for p in centers) + radius + pad))
    yy, xx = np.ogrid[y0 : y1 + 1, x0 : x1 + 1]
    union = np.zeros((y1 - y0 + 1, x1 - x0 + 1), dtype=bool)
    rr = radius * radius
    for cx, cy in centers:
        union |= (xx - cx) ** 2 + (yy - cy) ** 2 <= rr

    outside = np.zeros_like(union)
    queue: deque[tuple[int, int]] = deque()
    h, w = union.shape
    for x in range(w):
        if not union[0, x]: outside[0, x] = True; queue.append((0, x))
        if not union[h - 1, x] and not outside[h - 1, x]: outside[h - 1, x] = True; queue.append((h - 1, x))
    for y in range(1, h - 1):
        if not union[y, 0]: outside[y, 0] = True; queue.append((y, 0))
        if not union[y, w - 1]: outside[y, w - 1] = True; queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and not union[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    return bool(np.any(~union & ~outside))


def analytic_gold(scene: dict) -> str:
    centers = scene["centers"]
    radius = int(scene["influence_radius"])
    width = int(scene["width"])
    height = int(scene["height"])
    return "yes" if _has_bounded_complement(centers, radius, width, height) else "no"


def margin(scene: dict) -> float:
    centers = scene["centers"]
    radius = float(scene["influence_radius"])
    # Robust median is not pulled far by a possible central filler marker.
    xs = sorted(float(p[0]) for p in centers)
    ys = sorted(float(p[1]) for p in centers)
    cx = (xs[5] + xs[6]) / 2
    cy = (ys[5] + ys[6]) / 2
    ordered = sorted(centers, key=lambda p: math.atan2(p[1] - cy, p[0] - cx))
    gaps = [math.dist(ordered[i], ordered[(i + 1) % len(ordered)]) for i in range(len(ordered))]
    opening_clearance = max(gaps) - 2 * radius
    center_coverage = radius - min(math.hypot(p[0] - cx, p[1] - cy) for p in centers)
    if analytic_gold(scene) == "yes":
        return float(max(0.0, min(-opening_clearance, -center_coverage)))
    return float(max(abs(opening_clearance), center_coverage, 0.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["centers"] = list(reversed(scene["centers"]))
    return [("reverse_center_serialization", twin)]
