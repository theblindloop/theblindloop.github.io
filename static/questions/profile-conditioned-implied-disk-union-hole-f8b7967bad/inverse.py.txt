from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    h, w = mask.shape
    for y, x in np.argwhere(mask):
        y = int(y); x = int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        q = deque([(y, x)])
        group: list[tuple[int, int]] = []
        while q:
            py, px = q.popleft()
            group.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        groups.append(group)
    return groups


def _read_geometry(image: Image.Image) -> tuple[list[list[int]], int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    lum = arr.mean(axis=2)
    dark = lum < 70
    centers: list[list[int]] = []
    for group in _components(dark):
        if 55 <= len(group) <= 190:
            centers.append([int(round(sum(p[1] for p in group) / len(group))), int(round(sum(p[0] for p in group) / len(group)))])
    neutral = (lum >= 70) & (lum < 180) & ((arr.max(axis=2) - arr.min(axis=2)) < 28)
    bars = []
    for group in _components(neutral):
        ys = [p[0] for p in group]; xs = [p[1] for p in group]
        bw = max(xs) - min(xs) + 1; bh = max(ys) - min(ys) + 1
        if bw >= 30 and bh <= 10:
            bars.append((bw, len(group)))
    if len(centers) != 12 or len(bars) != 1:
        raise ValueError(f"could not recover 12 dots and one scale bar: {len(centers)}, {len(bars)}")
    return centers, int(bars[0][0])


def _union_has_hole(centers: list[list[int]], radius: int, image_size: tuple[int, int]) -> bool:
    width, height = image_size
    pad = 3
    x0 = max(0, min(p[0] for p in centers) - radius - pad)
    x1 = min(width - 1, max(p[0] for p in centers) + radius + pad)
    y0 = max(0, min(p[1] for p in centers) - radius - pad)
    y1 = min(height - 1, max(p[1] for p in centers) + radius + pad)
    yy, xx = np.ogrid[y0 : y1 + 1, x0 : x1 + 1]
    occupied = np.zeros((y1 - y0 + 1, x1 - x0 + 1), dtype=bool)
    for cx, cy in centers:
        occupied |= (xx - cx) ** 2 + (yy - cy) ** 2 <= radius * radius
    h, w = occupied.shape
    exterior = np.zeros_like(occupied)
    q: deque[tuple[int, int]] = deque()
    for x in range(w):
        for y in (0, h - 1):
            if not occupied[y, x] and not exterior[y, x]: exterior[y, x] = True; q.append((y, x))
    for y in range(1, h - 1):
        for x in (0, w - 1):
            if not occupied[y, x] and not exterior[y, x]: exterior[y, x] = True; q.append((y, x))
    while q:
        y, x = q.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and not occupied[ny, nx] and not exterior[ny, nx]:
                exterior[ny, nx] = True
                q.append((ny, nx))
    return bool(np.any(~occupied & ~exterior))


def decision_from_image(image: Image.Image) -> str:
    centers, radius = _read_geometry(image)
    return "yes" if _union_has_hole(centers, radius, image.size) else "no"
