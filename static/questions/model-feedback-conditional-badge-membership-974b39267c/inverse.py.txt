"""Independent image-only inverse for Conditional Badge Membership."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
DECISIONS = tuple("ABCDEF")
INK = np.array((35, 39, 48), dtype=np.int16)
START = np.array((198, 38, 139), dtype=np.int16)
SOURCE_COLORS = tuple(np.array(color, dtype=np.int16) for color in ((220, 55, 62), (35, 105, 210), (20, 153, 100), (235, 137, 25), (126, 70, 190), (10, 157, 177)))
TARGET_COLORS = tuple(np.array(color, dtype=np.int16) for color in ((119, 63, 45), (224, 91, 168), (88, 126, 35), (34, 65, 139), (191, 103, 26), (72, 72, 78)))
CELL = 58
GRID_Y = 284
SOURCE_X = 76
TARGET_X = 1036
ROUTE_SOURCE_X = 535
FILLED_TARGET_X = 721
HOLLOW_TARGET_X = 896
ROUTE_Y = 302
ROUTE_STEP = 55


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _tag_in_crop(crop: np.ndarray, colors: tuple[np.ndarray, ...]) -> int | None:
    scores = [int(_mask(crop, color).sum()) for color in colors]
    winners = [index for index, score in enumerate(scores) if score > 170]
    return winners[0] if len(winners) == 1 else None


def _read_board(arr: np.ndarray, x0: int, colors: tuple[np.ndarray, ...]) -> tuple[list[list[int]], list[int]] | None:
    matrix, tags = [], []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        tag = _tag_in_crop(arr[cy - 22:cy + 23, x0 - 69:x0 - 20], colors)
        if tag is None:
            return None
        tags.append(tag)
        values = []
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            values.append(1 if int(_mask(arr[cy - 9:cy + 10, cx - 9:cx + 10], INK).sum()) > 250 else 0)
        matrix.append(values)
    if sorted(tags) != list(range(N)) or any(sum(row) != 3 for row in matrix):
        return None
    if any(sum(matrix[row][col] for row in range(N)) != 3 for col in range(N)):
        return None
    return matrix, tags


def _read_routes(arr: np.ndarray) -> tuple[list[int], list[int]] | None:
    filled: list[int | None] = [None] * N
    hollow: list[int | None] = [None] * N
    for row in range(N):
        cy = ROUTE_Y + row * ROUTE_STEP
        source = _tag_in_crop(arr[cy - 22:cy + 23, ROUTE_SOURCE_X - 23:ROUTE_SOURCE_X + 24], SOURCE_COLORS)
        f_target = _tag_in_crop(arr[cy - 22:cy + 23, FILLED_TARGET_X - 23:FILLED_TARGET_X + 24], TARGET_COLORS)
        h_target = _tag_in_crop(arr[cy - 22:cy + 23, HOLLOW_TARGET_X - 23:HOLLOW_TARGET_X + 24], TARGET_COLORS)
        if source is None or f_target is None or h_target is None or filled[source] is not None:
            return None
        filled[source], hollow[source] = f_target, h_target
    if sorted(value for value in filled if value is not None) != list(range(N)):
        return None
    if sorted(value for value in hollow if value is not None) != list(range(N)):
        return None
    if any(filled[index] == hollow[index] for index in range(N)):
        return None
    return [int(value) for value in filled], [int(value) for value in hollow]


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {tag: row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (1460, 800):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        source = _read_board(arr, SOURCE_X, SOURCE_COLORS)
        target = _read_board(arr, TARGET_X, TARGET_COLORS)
        routes = _read_routes(arr)
        if source is None or target is None or routes is None:
            return "abstain"
        starts = []
        for col in range(N):
            cx = SOURCE_X + col * CELL + CELL // 2
            starts.append(int(_mask(arr[188:238, cx - 27:cx + 28], START).sum()) > 750)
        if sum(starts) != 1:
            return "abstain"
        source_signature = _signatures(*source)[starts.index(True)]
        filled, hollow = routes
        routed: list[int | None] = [None] * N
        for source_tag, value in enumerate(source_signature):
            destination = filled[source_tag] if value else hollow[source_tag]
            if routed[destination] is not None:
                return "abstain"
            routed[destination] = value
        if any(value is None for value in routed):
            return "abstain"
        target_signatures = _signatures(*target)
        matches = [index for index, signature in enumerate(target_signatures) if signature == tuple(routed)]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except Exception:
        return "abstain"
