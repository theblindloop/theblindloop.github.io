"""Deterministic renderer and analytic gold for compact patch turn replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-patch-turn-replay-0.1.0"
WIDTH = 768
HEIGHT = 512
SUPERSAMPLE = 4
GRID_SIZE = 4
OPERATION_COUNT = 5

BACKGROUND = (249, 251, 252)
PANEL_FILL = (231, 239, 243)
PANEL_EDGE = (48, 68, 80)
CARD_FILL = (214, 226, 233)
CARD_EDGE = (178, 195, 204)
MARKER = (36, 56, 70)
TEXT = (49, 68, 80)
DISTRACTOR = (157, 169, 177)
TILE_COLORS = (
    (228, 86, 82),
    (71, 139, 216),
    (75, 172, 116),
    (231, 174, 61),
)
ROW_LABELS = ("A", "B", "C", "D")
COL_LABELS = ("1", "2", "3", "4")
CELL_LABELS = tuple(f"{row}{col}" for row in ROW_LABELS for col in COL_LABELS)

PANEL_WIDTH = 500
PANEL_HEIGHT = 300
GRID_LEFT = 34
GRID_TOP = 58
CELL_SIZE = 48
CELL_GAP = 8
GRID_WIDTH = GRID_SIZE * CELL_SIZE + (GRID_SIZE - 1) * CELL_GAP
COMMAND_LEFT = 282
COMMAND_TOP = 55
COMMAND_WIDTH = 182
COMMAND_HEIGHT = 39
COMMAND_STEP = 46

FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(FONT_PATH, size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, ...], scale: int) -> list[int]:
    return [int(round(value * scale)) for value in values]


def _center_text(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: float,
    top_y: float,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    bounds = draw.textbbox((0, 0), text, font=font)
    width = bounds[2] - bounds[0]
    draw.text(
        (int(round(center_x * scale - width / 2)), int(round(top_y * scale))),
        text,
        fill=fill,
        font=font,
    )


def _replay_index(initial_cell: int, operations: list[str] | tuple[str, ...]) -> int:
    row, column = divmod(int(initial_cell), GRID_SIZE)
    for operation in operations:
        if operation == "R":
            row, column = column, GRID_SIZE - 1 - row
        elif operation == "L":
            row, column = GRID_SIZE - 1 - column, row
        else:
            raise ValueError(f"unknown turn operation {operation!r}")
    return row * GRID_SIZE + column


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, visible patch-turn scene."""

    rng = random.Random(int(seed))
    initial_cell = (int(seed) * 7 + rng.randrange(GRID_SIZE * GRID_SIZE)) % (GRID_SIZE * GRID_SIZE)
    operations = [rng.choice(("R", "L")) for _ in range(OPERATION_COUNT)]
    palette = [list(color) for color in TILE_COLORS]
    rng.shuffle(palette)
    tile_states = [rng.randrange(len(TILE_COLORS)) for _ in range(GRID_SIZE * GRID_SIZE)]

    origin_x = rng.choice((105, 120, 135, 150, 165))
    origin_y = rng.choice((70, 82, 94, 106))
    panel_box = (origin_x - 20, origin_y - 20, origin_x + PANEL_WIDTH + 20, origin_y + PANEL_HEIGHT + 20)
    distractors: list[dict[str, int]] = []
    for _ in range(6):
        for _attempt in range(300):
            x = rng.randint(24, WIDTH - 24)
            y = rng.randint(24, HEIGHT - 24)
            if not (panel_box[0] <= x <= panel_box[2] and panel_box[1] <= y <= panel_box[3]):
                distractors.append(
                    {"x": x, "y": y, "kind": rng.randrange(2), "size": rng.randint(4, 8)}
                )
                break

    return {
        "initial_cell": initial_cell,
        "operations": operations,
        "origin": [origin_x, origin_y],
        "palette": palette,
        "tile_states": tile_states,
        "distractors": distractors,
    }


def _tile_center(origin_x: float, origin_y: float, row: int, column: int) -> tuple[float, float]:
    return (
        origin_x + GRID_LEFT + column * (CELL_SIZE + CELL_GAP) + CELL_SIZE / 2,
        origin_y + GRID_TOP + row * (CELL_SIZE + CELL_GAP) + CELL_SIZE / 2,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one bounded state patch and its ordered operation cards."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    small_font = _font(12 * scale)
    number_font = _font(13 * scale)
    header_font = _font(12 * scale)

    for distractor in scene["distractors"]:
        x = float(distractor["x"])
        y = float(distractor["y"])
        size = float(distractor["size"])
        if int(distractor["kind"]) == 0:
            draw.ellipse(_box((x - size, y - size, x + size, y + size), scale), fill=DISTRACTOR)
        else:
            draw.polygon(
                [
                    (int(round(x * scale)), int(round((y - size) * scale))),
                    (int(round((x + size) * scale)), int(round(y * scale))),
                    (int(round(x * scale)), int(round((y + size) * scale))),
                    (int(round((x - size) * scale)), int(round(y * scale))),
                ],
                fill=DISTRACTOR,
            )

    origin_x, origin_y = (float(value) for value in scene["origin"])
    draw.rounded_rectangle(
        _box((origin_x, origin_y, origin_x + PANEL_WIDTH, origin_y + PANEL_HEIGHT), scale),
        radius=18 * scale,
        fill=PANEL_FILL,
        outline=PANEL_EDGE,
        width=3 * scale,
    )
    _center_text(draw, "PATCH TURN LOG", origin_x + PANEL_WIDTH / 2, origin_y + 12, header_font, TEXT, scale)

    palette = [tuple(int(value) for value in color) for color in scene["palette"]]
    for row in range(GRID_SIZE):
        _center_text(
            draw,
            ROW_LABELS[row],
            origin_x + 17,
            origin_y + GRID_TOP + row * (CELL_SIZE + CELL_GAP) + 16,
            small_font,
            TEXT,
            scale,
        )
    for column in range(GRID_SIZE):
        _center_text(
            draw,
            COL_LABELS[column],
            origin_x + GRID_LEFT + column * (CELL_SIZE + CELL_GAP) + CELL_SIZE / 2,
            origin_y + GRID_TOP - 19,
            small_font,
            TEXT,
            scale,
        )

    states = [int(value) for value in scene["tile_states"]]
    for index, state in enumerate(states):
        row, column = divmod(index, GRID_SIZE)
        left = origin_x + GRID_LEFT + column * (CELL_SIZE + CELL_GAP)
        top = origin_y + GRID_TOP + row * (CELL_SIZE + CELL_GAP)
        draw.rounded_rectangle(
            _box((left, top, left + CELL_SIZE, top + CELL_SIZE), scale),
            radius=5 * scale,
            fill=palette[state],
            outline=PANEL_EDGE,
            width=2 * scale,
        )
        _center_text(
            draw,
            CELL_LABELS[index],
            left + CELL_SIZE / 2,
            top + 16,
            small_font,
            (255, 255, 255),
            scale,
        )
        if index == int(scene["initial_cell"]):
            cx, cy = _tile_center(origin_x, origin_y, row, column)
            # Leave a deliberate small opening in the ring.  This keeps the
            # colored tile one connected component for the independent mask
            # inverse while remaining an unmistakable ring marker.
            draw.arc(
                _box((cx - 20, cy - 20, cx + 20, cy + 20), scale),
                fill=MARKER,
                width=4 * scale,
                start=24,
                end=336,
            )

    command_font = _font(11 * scale)
    for index, operation in enumerate(scene["operations"]):
        top = origin_y + COMMAND_TOP + index * COMMAND_STEP
        left = origin_x + COMMAND_LEFT
        draw.rounded_rectangle(
            _box((left, top, left + COMMAND_WIDTH, top + COMMAND_HEIGHT), scale),
            radius=8 * scale,
            fill=CARD_FILL,
            outline=CARD_EDGE,
            width=2 * scale,
        )
        _center_text(draw, str(index + 1), left + 19, top + 11, number_font, TEXT, scale)
        _center_text(draw, "turn", left + 143, top + 12, command_font, TEXT, scale)
        cx = left + 88
        cy = top + COMMAND_HEIGHT / 2
        if operation == "R":
            points = ((cx - 15, cy - 15), (cx + 15, cy), (cx - 15, cy + 15))
        else:
            points = ((cx + 15, cy - 15), (cx - 15, cy), (cx + 15, cy + 15))
        draw.polygon([(int(round(x * scale)), int(round(y * scale))) for x, y in points], fill=MARKER)

    if scale > 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the terminal labeled cell after the visible turn sequence."""

    initial_cell = int(scene["initial_cell"])
    operations = list(scene["operations"])
    if not (0 <= initial_cell < GRID_SIZE * GRID_SIZE) or len(operations) != OPERATION_COUNT:
        raise ValueError("scene does not contain one valid start and five operations")
    return CELL_LABELS[_replay_index(initial_cell, operations)]


def margin(scene: dict[str, Any]) -> float:
    """One-sided discrete margin: a valid replay has one terminal cell."""

    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    operations = scene.get("operations")
    initial_cell = scene.get("initial_cell")
    return not (
        isinstance(initial_cell, int)
        and 0 <= initial_cell < GRID_SIZE * GRID_SIZE
        and isinstance(operations, list)
        and len(operations) == OPERATION_COUNT
        and all(operation in {"R", "L"} for operation in operations)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No hidden label-preserving transform is claimed for this encoding."""

    return []
