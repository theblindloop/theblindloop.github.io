"""Independent pixel-only inverse arm based on dark-stroke connectivity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MIN_STROKE_AREA = 500
MIN_BACKGROUND_FRACTION = 0.55


def _components(mask: np.ndarray) -> list[tuple[int, bool]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, bool]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        area = 0
        border = False
        while queue:
            y, x = queue.popleft()
            area += 1
            border = border or y == 0 or x == 0 or y == height - 1 or x == width - 1
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        found.append((area, border))
    return found


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Estimate the paper color from a robust median of a 12-pixel border band.
    border = np.concatenate(
        [rgb[:12].reshape(-1, 3), rgb[-12:].reshape(-1, 3),
         rgb[:, :12].reshape(-1, 3), rgb[:, -12:].reshape(-1, 3)], axis=0
    )
    paper = np.median(border, axis=0)
    distance = np.sqrt(np.sum((rgb - paper) ** 2, axis=2))
    background = distance < 34.0
    if float(background.mean()) < MIN_BACKGROUND_FRACTION:
        raise ValueError("background segmentation is implausible")
    strokes = _components(~background)
    plausible = [area for area, _ in strokes if area >= MIN_STROKE_AREA]
    ambiguous = [area for area, _ in strokes if 80 <= area < MIN_STROKE_AREA]
    if ambiguous:
        raise ValueError("ambiguous stroke component")
    if len(plausible) not in (1, 2):
        raise ValueError(f"unsupported number of stroke groups: {len(plausible)}")
    return "yes" if len(plausible) == 1 else "no"
