"""Deterministic latent sampler and raster renderer for emblem loop counting."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
SCALE = 4
CONTACT_CLEARANCE = 12.0
INKS = ("#172033", "#263238", "#202124", "#18242f")
BACKGROUNDS = ("#f7f4ed", "#f4f7f8", "#faf7f2", "#f5f3f8")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    connected = bool(int(seed) % 2)
    loop_width = rng.randrange(108, 121, 4)
    loop_height = rng.randrange(76, 97, 4)
    # Broad overlap in thickness across classes while retaining a safe hole.
    stroke_width = rng.randrange(9, 15)
    return {
        "connected": connected,
        "phase": rng.randrange(2),
        "loop_width": loop_width,
        "loop_height": loop_height,
        "stroke_width": stroke_width,
        "ink": INKS[rng.randrange(len(INKS))],
        "background": BACKGROUNDS[rng.randrange(len(BACKGROUNDS))],
        "offset_x": rng.randrange(-17, 18),
        "offset_y": rng.randrange(-15, 16),
    }


def _centers(scene: dict) -> list[tuple[float, float]]:
    connected = bool(scene["connected"])
    phase = int(scene["phase"])
    width = float(scene["loop_width"])
    height = float(scene["loop_height"])
    ox = float(scene["offset_x"])
    oy = float(scene["offset_y"])
    gap_x = width - 18.0
    centers: list[tuple[float, float]] = []
    for i in range(5):
        x = CANVAS / 2 + (i - 2) * gap_x + ox
        y = CANVAS / 2 + (-12.0 if (i + phase) % 2 == 0 else 12.0) + oy
        if not connected and i == (0 if phase else 4):
            y += 118.0 if phase == 0 else -118.0
        centers.append((x, y))
    return centers


def render(scene: dict) -> Image.Image:
    # Every declared latent field is deliberately read on the render path.
    width = float(scene["loop_width"])
    height = float(scene["loop_height"])
    stroke = int(scene["stroke_width"])
    ink = str(scene["ink"])
    background = str(scene["background"])
    centers = _centers(scene)

    image = Image.new("RGB", (CANVAS * SCALE, CANVAS * SCALE), background)
    draw = ImageDraw.Draw(image)
    for cx, cy in centers:
        box = (
            int(round((cx - width / 2) * SCALE)),
            int(round((cy - height / 2) * SCALE)),
            int(round((cx + width / 2) * SCALE)),
            int(round((cy + height / 2) * SCALE)),
        )
        draw.ellipse(box, outline=ink, width=stroke * SCALE)
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["connected"]) else "no"


def margin(scene: dict) -> float:
    # Runner-up margin to the touch/no-touch boundary, in designed pixels.
    return CONTACT_CLEARANCE


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
