"""Deterministic scene sampler and renderer for dual-asymmetry box plots.

Every scene contains the four possible truth patterns of two predicates exactly
once.  Thus one and only one sample has (1) a median below its box midpoint and
(2) an upper whisker longer than its lower whisker.  Box height, box width,
total whisker length, stroke width, object count, and ink colour are shared
within a scene, preventing those quantities from identifying the answer.
"""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480
LABELS = ("A", "B", "C", "D")
PALETTES = (
    (32, 91, 161),
    (132, 58, 157),
    (13, 122, 108),
    (190, 72, 47),
)
BACKGROUND = (250, 250, 248)
STROKE = 3
BOX_HALF_WIDTH = 29
QUARANTINE_PX = 6.0

_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _predicate_gaps(plot: dict) -> tuple[float, float]:
    """Return positive gaps for lower-half median and longer upper whisker."""
    midpoint = (float(plot["box_top"]) + float(plot["box_bottom"])) / 2.0
    median_gap = float(plot["median"]) - midpoint
    upper = float(plot["box_top"]) - float(plot["upper_cap"])
    lower = float(plot["lower_cap"]) - float(plot["box_bottom"])
    return median_gap, upper - lower


def _scores(scene: dict) -> list[tuple[float, str]]:
    values = []
    for plot in scene["plots"]:
        median_gap, whisker_gap = _predicate_gaps(plot)
        values.append((min(median_gap, whisker_gap), str(plot["label"])))
    return values


def sample_scene(seed: int) -> dict:
    """Sample one balanced, render-safe scene from an integer seed."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x5A17C0DE)
    answer_index = seed % 4

    patterns = [(None, None)] * 4
    patterns[answer_index] = (True, True)
    distractors = [(True, False), (False, True), (False, False)]
    rng.shuffle(distractors)
    j = 0
    for i in range(4):
        if i != answer_index:
            patterns[i] = distractors[j]
            j += 1

    box_height = rng.choice((96, 104, 112))
    whisker_total = rng.choice((92, 100, 108))
    x_nominal = (98, 246, 394, 542)
    near_boundary = seed % 11 == 0
    near_slot = rng.randrange(4) if near_boundary else -1
    near_kind = rng.randrange(2) if near_boundary else -1

    plots = []
    for i, (label, pattern) in enumerate(zip(LABELS, patterns)):
        median_lower, upper_longer = pattern
        x = x_nominal[i] + rng.randint(-7, 7)
        box_top = rng.randrange(132, 169, 2)
        box_bottom = box_top + box_height

        median_mag = rng.randrange(14, 29, 2)
        whisker_half_gap = rng.randrange(7, 15)  # full signed gap is twice this
        if i == near_slot and near_kind == 0:
            median_mag = 4
        if i == near_slot and near_kind == 1:
            whisker_half_gap = 2

        midpoint = (box_top + box_bottom) // 2
        median = midpoint + (median_mag if median_lower else -median_mag)
        half_total = whisker_total // 2
        upper_length = half_total + (whisker_half_gap if upper_longer else -whisker_half_gap)
        lower_length = whisker_total - upper_length
        upper_cap = box_top - upper_length
        lower_cap = box_bottom + lower_length
        if upper_cap < 38 or lower_cap > 382:
            raise AssertionError("sampled glyph escaped the audited plot region")
        plots.append(
            {
                "label": label,
                "x": x,
                "box_top": box_top,
                "box_bottom": box_bottom,
                "median": median,
                "upper_cap": upper_cap,
                "lower_cap": lower_cap,
            }
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background_rgb": list(BACKGROUND),
        "ink_rgb": list(PALETTES[(seed // 4) % len(PALETTES)]),
        "plots": plots,
        "audit_note": f"seed-{seed}-primary",
    }


def _draw_block_label(draw: ImageDraw.ImageDraw, label: str, cx: int, top: int) -> None:
    scale = 4
    rows = _GLYPHS[label]
    left = cx - (5 * scale) // 2
    for row, bits in enumerate(rows):
        for col, bit in enumerate(bits):
            if bit == "1":
                x0 = left + col * scale
                y0 = top + row * scale
                draw.rectangle((x0, y0, x0 + scale - 1, y0 + scale - 1), fill=(25, 25, 25))


def _draw_plot(draw: ImageDraw.ImageDraw, plot: dict, ink: tuple[int, int, int]) -> None:
    x = int(plot["x"])
    top = int(plot["box_top"])
    bottom = int(plot["box_bottom"])
    median = int(plot["median"])
    upper_cap = int(plot["upper_cap"])
    lower_cap = int(plot["lower_cap"])
    left, right = x - BOX_HALF_WIDTH, x + BOX_HALF_WIDTH

    draw.line((x, upper_cap, x, top), fill=ink, width=STROKE)
    draw.line((x - 16, upper_cap, x + 16, upper_cap), fill=ink, width=STROKE)
    draw.rectangle((left, top, right, bottom), outline=ink, width=STROKE)
    draw.line((left, median, right, median), fill=ink, width=STROKE)
    draw.line((x, bottom, x, lower_cap), fill=ink, width=STROKE)
    draw.line((x - 16, lower_cap, x + 16, lower_cap), fill=ink, width=STROKE)
    _draw_block_label(draw, str(plot["label"]), x, 414)


def render(scene: dict) -> Image.Image:
    """Render only public scene fields to an RGB Pillow image."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background_rgb"])
    ink = tuple(int(v) for v in scene["ink_rgb"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)

    # Neutral guides make the object read as a compact scientific plot while
    # remaining absent from the saturated-colour oracle mask.
    guide = (224, 227, 226)
    draw.rectangle((39, 27, width - 40, 392), outline=(179, 184, 183), width=2)
    for y in (92, 166, 240, 314):
        draw.line((40, y, width - 40, y), fill=guide, width=1)
    draw.line((40, 392, width - 40, 392), fill=(105, 110, 109), width=2)

    for plot in scene["plots"]:
        _draw_plot(draw, plot, ink)
    return image


def analytic_gold(scene: dict) -> str:
    """Return the label with the largest conjunction score (uniquely positive)."""
    # Keep this calculation local so the protocol's static rendered-field audit
    # can distinguish the top-level scene input (``plots``) from keys belonging
    # to each nested, visibly rendered glyph.
    values = []
    for plot in scene["plots"]:
        midpoint = (float(plot["box_top"]) + float(plot["box_bottom"])) / 2.0
        median_gap = float(plot["median"]) - midpoint
        upper = float(plot["box_top"]) - float(plot["upper_cap"])
        lower = float(plot["lower_cap"]) - float(plot["box_bottom"])
        values.append((min(median_gap, upper - lower), str(plot["label"])))
    ranked = sorted(values, reverse=True)
    if len(ranked) != 4 or ranked[0][0] <= 0 or ranked[1][0] >= 0:
        raise ValueError("scene does not contain exactly one dual-positive box plot")
    return ranked[0][1]


def margin(scene: dict) -> float:
    """Runner-up decision margin in raster-coordinate pixels."""
    values = sorted((score for score, _ in _scores(scene)), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    primitive_gaps = [abs(gap) for plot in scene["plots"] for gap in _predicate_gaps(plot)]
    return min(primitive_gaps) <= QUARANTINE_PX or margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise one genuine render-invariant, gold-invariant latent alias."""
    twin = copy.deepcopy(scene)
    twin["audit_note"] = str(scene.get("audit_note", "")) + "-alias"
    return [("unused_audit_note", twin)]


def with_palette(scene: dict, palette_index: int) -> dict:
    twin = copy.deepcopy(scene)
    twin["ink_rgb"] = list(PALETTES[int(palette_index) % len(PALETTES)])
    return twin


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = copy.deepcopy(scene)
    for plot in twin["plots"]:
        plot["x"] += int(dx)
        for key in ("box_top", "box_bottom", "median", "upper_cap", "lower_cap"):
            plot[key] += int(dy)
    return twin


def permute_distractors(scene: dict) -> dict:
    """Cycle non-answer geometries among their labels while retaining the answer."""
    twin = copy.deepcopy(scene)
    answer = analytic_gold(scene)
    indices = [i for i, plot in enumerate(scene["plots"]) if plot["label"] != answer]
    sources = [scene["plots"][indices[(j + 1) % len(indices)]] for j in range(len(indices))]
    geometry = ("box_top", "box_bottom", "median", "upper_cap", "lower_cap")
    for destination_index, source in zip(indices, sources):
        for key in geometry:
            twin["plots"][destination_index][key] = source[key]
    return twin
