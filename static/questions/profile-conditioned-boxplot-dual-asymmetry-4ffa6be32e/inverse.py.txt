"""Independent pixel-only inverse for the dual-asymmetry box-plot task.

The oracle receives only an RGB raster.  It discovers the saturated plot ink,
splits four glyphs by horizontal connected support, recovers five horizontal
landmark bands per glyph from row projections, and recomputes both signed gaps.
It contains no renderer coordinates, scene values, analytic gold, or metadata.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _runs(values: np.ndarray, *, max_gap: int = 1) -> list[np.ndarray]:
    if len(values) == 0:
        return []
    breaks = np.flatnonzero(np.diff(values) > max_gap) + 1
    return [part for part in np.split(values, breaks) if len(part)]


def _ink_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    # All allowed plot palettes are saturated; gray guides and black labels are not.
    return ((high - low) >= 55) & (high <= 225) & (low <= 170)


def _measure(image: Image.Image) -> list[tuple[float, str]]:
    mask = _ink_mask(image)
    active_x = np.flatnonzero(mask.any(axis=0))
    groups = _runs(active_x, max_gap=2)
    if len(groups) != 4:
        raise ValueError(f"expected four separated box-plot glyphs, found {len(groups)}")

    measured = []
    for label, columns in zip(LABELS, groups):
        crop = mask[:, int(columns[0]) : int(columns[-1]) + 1]
        row_counts = crop.sum(axis=1)
        # Caps span about 33 pixels and box/median bars about 59. Vertical strokes
        # contribute at most nine pixels, so this threshold isolates horizontal bars.
        active_y = np.flatnonzero(row_counts >= 18)
        bands = _runs(active_y, max_gap=1)
        if len(bands) != 5:
            raise ValueError(f"sample {label}: expected five horizontal landmarks, found {len(bands)}")
        landmarks = [float(part.mean()) for part in bands]
        upper_cap, box_top, median, box_bottom, lower_cap = landmarks
        if not (upper_cap < box_top < median < box_bottom < lower_cap):
            raise ValueError(f"sample {label}: invalid landmark order")
        median_gap = median - (box_top + box_bottom) / 2.0
        upper_length = box_top - upper_cap
        lower_length = lower_cap - box_bottom
        measured.append((min(median_gap, upper_length - lower_length), label))
    return measured


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique dual-positive sample label from final pixels alone."""
    ranked = sorted(_measure(image), reverse=True)
    if ranked[0][0] <= 0 or ranked[1][0] >= 0:
        raise ValueError("raster does not contain exactly one dual-positive sample")
    return ranked[0][1]
