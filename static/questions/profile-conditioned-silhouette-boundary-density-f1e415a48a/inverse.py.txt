"""Independent pixel-only inverse for normalized silhouette boundary density."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


PANEL_RECTS = ((60, 120, 480, 550), (520, 120, 940, 550))


def _saturated_mask(image: Image.Image, rect: tuple[int, int, int, int]) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    x0, y0, x1, y1 = rect
    crop = pixels[y0 + 14 : y1 - 14, x0 + 14 : x1 - 14]
    spread = crop.max(axis=2) - crop.min(axis=2)
    brightness = crop.max(axis=2)
    darkness = crop.min(axis=2)
    return (spread >= 45) & (brightness < 248) & (darkness < 230)


def _marching_perimeter(mask: np.ndarray) -> float:
    """Approximate a closed boundary with independent marching-square segments."""

    mask = np.asarray(mask, dtype=bool)
    height, width = mask.shape
    perimeter = 0.0
    for y in range(height - 1):
        top_left = mask[y, :-1]
        top_right = mask[y, 1:]
        bottom_right = mask[y + 1, 1:]
        bottom_left = mask[y + 1, :-1]
        for x in range(width - 1):
            a = bool(top_left[x])
            b = bool(top_right[x])
            c = bool(bottom_right[x])
            d = bool(bottom_left[x])
            crossings: list[tuple[float, float]] = []
            if a != b:
                crossings.append((x + 0.5, y))
            if b != c:
                crossings.append((x + 1.0, y + 0.5))
            if c != d:
                crossings.append((x + 0.5, y + 1.0))
            if d != a:
                crossings.append((x, y + 0.5))
            if len(crossings) == 2:
                perimeter += math.dist(crossings[0], crossings[1])
            elif len(crossings) == 4:
                perimeter += math.dist(crossings[0], crossings[1])
                perimeter += math.dist(crossings[2], crossings[3])
    return perimeter


def _shape_score(image: Image.Image, rect: tuple[int, int, int, int]) -> float:
    mask = _saturated_mask(image, rect)
    area = int(mask.sum())
    if area < 3000:
        raise ValueError(f"colored silhouette is too small: {area} pixels")
    rows, columns = np.nonzero(mask)
    if len(rows) == 0 or rows.min() == 0 or columns.min() == 0:
        raise ValueError("colored silhouette is clipped or absent")
    perimeter = _marching_perimeter(mask)
    if perimeter <= 0.0:
        raise ValueError("colored silhouette boundary is not measurable")
    return perimeter / math.sqrt(float(area))


def decision_from_image(image: Image.Image) -> str:
    """Recover both ratios from colored pixels and return the larger panel."""

    score_a = _shape_score(image, PANEL_RECTS[0])
    score_b = _shape_score(image, PANEL_RECTS[1])
    return "A" if score_a > score_b else "B"
