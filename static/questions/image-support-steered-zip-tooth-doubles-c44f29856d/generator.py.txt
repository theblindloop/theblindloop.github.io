"""Deterministic scene sampling and RGB rendering for zip_tooth_doubles.

One strongly coloured zipper is drawn as a single connected comb: a straight
thin tape with short teeth attached along it, each tooth sticking out from
exactly one side. The decision is how many neighbouring tooth pairs stick out
from the *same* side, i.e. how many times the left/right alternation of the
strip breaks while walking along the tape. The canonical prior (a zipper
interlocks perfectly, so zero breaks) is recorded for analysis only.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (255, 255, 255)
QUARANTINE_MARGIN = 4.0
CANONICAL_PRIOR = "0"
MAX_BREAKS = 3

# Strongly chromatic tape colours; chroma (max-min) is far above the oracle's
# chroma threshold and is sampled independently of the answer.
TAPE_COLORS = (
    (28, 60, 168),
    (176, 32, 52),
    (16, 116, 116),
    (108, 36, 160),
    (150, 84, 12),
)
CHROMA_THRESHOLD = 46


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _zipper_rects(params: dict) -> list[tuple[int, int, int, int]]:
    """Inclusive-pixel rectangles of one zipper comb, in canvas coordinates."""
    anchor_x, anchor_y = params["anchor"]
    spacing = params["spacing"]
    thickness = params["tape_thickness"]
    tooth_w = params["tooth_width"]
    reach = params["protrusion"]
    end = params["end_margin"]
    length = _tape_length(params)

    local: list[tuple[int, int, int, int]] = []
    seam_top = reach
    local.append((0, seam_top, length - 1, seam_top + thickness - 1))
    for index, side in params["teeth"]:
        center = end + index * spacing
        x0 = center - tooth_w // 2
        x1 = x0 + tooth_w - 1
        if side < 0:
            local.append((x0, seam_top - reach, x1, seam_top - 1))
        else:
            local.append((x0, seam_top + thickness, x1, seam_top + thickness + reach - 1))

    horizontal = params["orientation"] == "h"
    rects = []
    for x0, y0, x1, y1 in local:
        if horizontal:
            rects.append((anchor_x + x0, anchor_y + y0, anchor_x + x1, anchor_y + y1))
        else:
            rects.append((anchor_x + y0, anchor_y + x0, anchor_x + y1, anchor_y + x1))
    return rects


def _tape_length(params: dict) -> int:
    return 2 * params["end_margin"] + (params["n_teeth"] - 1) * params["spacing"] + 1


def _zipper_bbox(params: dict) -> tuple[int, int, int, int]:
    long_span = _tape_length(params)
    short_span = 2 * params["protrusion"] + params["tape_thickness"]
    anchor_x, anchor_y = params["anchor"]
    if params["orientation"] == "h":
        return (anchor_x, anchor_y, anchor_x + long_span - 1, anchor_y + short_span - 1)
    return (anchor_x, anchor_y, anchor_x + short_span - 1, anchor_y + long_span - 1)


def _inflate(box: tuple[int, int, int, int], pad: int) -> tuple[int, int, int, int]:
    return (box[0] - pad, box[1] - pad, box[2] + pad, box[3] + pad)


def _overlaps(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> bool:
    return not (a[2] < b[0] or b[2] < a[0] or a[3] < b[1] or b[3] < a[1])


def _sides_from_breaks(rng: random.Random, n_teeth: int, breaks: int) -> list[int]:
    slots = list(range(n_teeth - 1))
    same_side_slots = set(rng.sample(slots, breaks))
    sides = [rng.choice((-1, 1))]
    for slot in slots:
        sides.append(sides[-1] if slot in same_side_slots else -sides[-1])
    return sides


def _make_zipper(rng: random.Random, color: tuple[int, int, int], breaks: int) -> dict:
    n_teeth = rng.choice((6, 7, 8))
    while True:
        spacing = rng.randint(11, 17)
        if (n_teeth - 1) * spacing <= 100:
            break
    tooth_width = rng.choice((5, 6))
    protrusion = rng.randint(3, 14)
    params = {
        "orientation": rng.choice(("h", "v")),
        "n_teeth": n_teeth,
        "spacing": spacing,
        "tape_thickness": rng.choice((5, 6)),
        "tooth_width": tooth_width,
        "protrusion": protrusion,
        "end_margin": 7,
        "color": list(color),
        "anchor": [0, 0],
        "teeth": [[index, side] for index, side in enumerate(_sides_from_breaks(rng, n_teeth, breaks))],
    }
    return params


# --------------------------------------------------------------------------
# contract interface
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    breaks = rng.randint(0, MAX_BREAKS)
    zipper = _make_zipper(rng, rng.choice(TAPE_COLORS), breaks)

    occupied: list[tuple[int, int, int, int]] = []
    for _ in range(200):
        box = _zipper_bbox(zipper)
        span_x = box[2] - box[0]
        span_y = box[3] - box[1]
        zipper["anchor"] = [
            rng.randint(26, CANVAS - 27 - span_x),
            rng.randint(26, CANVAS - 27 - span_y),
        ]
        break
    occupied.append(_inflate(_zipper_bbox(zipper), 10))

    decor: list[dict] = []
    for _ in range(rng.randint(2, 4)):
        shade = rng.randint(105, 175)
        grey = _make_zipper(rng, (shade, shade, shade), rng.randint(0, MAX_BREAKS))
        placed = False
        for _ in range(80):
            box = _zipper_bbox(grey)
            span_x, span_y = box[2] - box[0], box[3] - box[1]
            if CANVAS - 13 - span_x <= 12 or CANVAS - 13 - span_y <= 12:
                break
            grey["anchor"] = [
                rng.randint(12, CANVAS - 13 - span_x),
                rng.randint(12, CANVAS - 13 - span_y),
            ]
            candidate = _inflate(_zipper_bbox(grey), 8)
            if all(not _overlaps(candidate, taken) for taken in occupied):
                occupied.append(candidate)
                placed = True
                break
        if placed:
            decor.append({"kind": "grey_zipper", "params": grey})

    for _ in range(rng.randint(4, 9)):
        radius = rng.randint(3, 8)
        shade = rng.randint(110, 185)
        for _ in range(60):
            cx = rng.randint(12, CANVAS - 13)
            cy = rng.randint(12, CANVAS - 13)
            box = (cx - radius, cy - radius, cx + radius, cy + radius)
            if all(not _overlaps(_inflate(box, 6), taken) for taken in occupied):
                occupied.append(_inflate(box, 6))
                decor.append(
                    {
                        "kind": "grey_dot" if radius % 2 else "grey_square",
                        "box": list(box),
                        "shade": shade,
                    }
                )
                break

    return {
        "width": CANVAS,
        "height": CANVAS,
        "zipper": zipper,
        "decor": decor,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for item in scene["decor"]:
        if item["kind"] == "grey_zipper":
            params = item["params"]
            shade = tuple(params["color"])
            for rect in _zipper_rects(params):
                draw.rectangle(list(rect), fill=shade)
        else:
            shade = (item["shade"], item["shade"], item["shade"])
            if item["kind"] == "grey_dot":
                draw.ellipse(list(item["box"]), fill=shade)
            else:
                draw.rectangle(list(item["box"]), fill=shade)
    zipper = scene["zipper"]
    color = tuple(zipper["color"])
    for rect in _zipper_rects(zipper):
        draw.rectangle(list(rect), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    """Number of neighbouring tooth pairs on the same side of the tape.

    Gold reads only the coloured zipper's teeth list, and sorts it by position
    index, so the stored order of that list cannot influence the answer.
    """
    teeth = sorted((int(index), int(side)) for index, side in scene["zipper"]["teeth"])
    breaks = sum(1 for (_, a), (_, b) in zip(teeth, teeth[1:]) if a == b)
    return str(breaks)


def margin(scene: dict) -> float:
    """Pixel clearance to the nearest visible configuration change.

    Two events end the reading: a tooth becoming too short for its side to be
    legible (protrusion shrinking to zero), and two neighbouring teeth merging
    into one stub (the along-tape gap closing).
    """
    zipper = scene["zipper"]
    gap = zipper["spacing"] - zipper["tooth_width"]
    return float(min(zipper["protrusion"], gap))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent aliases of the stored scene record.

    Teeth and decoration items are pairwise disjoint and each is painted in one
    flat colour, so the stored order of those lists is a genuine latent alias:
    permuting it yields a byte-identical raster while the record differs.
    """
    twins: list[tuple[str, dict]] = []
    teeth = scene["zipper"]["teeth"]

    reversed_scene = _copy(scene)
    reversed_scene["zipper"]["teeth"] = list(reversed(teeth))
    twins.append(("teeth_order_reversal", reversed_scene))

    if len(teeth) > 1:
        rotated = _copy(scene)
        rotated["zipper"]["teeth"] = teeth[1:] + teeth[:1]
        twins.append(("teeth_order_rotation", rotated))

    if len(scene["decor"]) > 1:
        shuffled = _copy(scene)
        shuffled["decor"] = list(reversed(scene["decor"]))
        twins.append(("decor_order_reversal", shuffled))
    return twins


def _copy(scene: dict) -> dict:
    zipper = dict(scene["zipper"])
    zipper["teeth"] = [list(t) for t in scene["zipper"]["teeth"]]
    zipper["color"] = list(scene["zipper"]["color"])
    zipper["anchor"] = list(scene["zipper"]["anchor"])
    return {
        "width": scene["width"],
        "height": scene["height"],
        "zipper": zipper,
        "decor": [dict(item) for item in scene["decor"]],
    }


def target_bbox(scene: dict) -> tuple[int, int, int, int]:
    """Convenience for tests and erasure probes; never used by gold."""
    return _zipper_bbox(scene["zipper"])
