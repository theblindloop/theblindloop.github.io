"""Swatch-key sieve reveal world.

One small bordered panel carries a 5x5 field of cells, each filled with one of
exactly four rendered grays. Two glyphs are superposed on that field at once:
the cells of one class belong to both glyphs, a second class belongs only to the
first glyph, a third class only to the second glyph, and a fourth class to
neither. A separate small bordered key, two lattice steps away and touching
nothing, shows two of those four grays as flat swatches. The declared raster
operation is a palette sieve: mark every panel cell whose gray matches either
key swatch. The marked cells spell exactly one glyph. The panel alone leaves two
candidate glyphs; the key alone carries no field at all.

All fills are axis-aligned flat blocks, so the raster is byte-reproducible and
every cell is one exact gray.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "swatch_key_sieve_reveal-renderer-0.1.0"

CANVAS = 800
LATTICE = 8
LATTICE_PX = CANVAS // LATTICE
WINDOW_LO = 25
WINDOW_HI = 75
GRID = 5
CELL_PX = 8
BORDER_PX = 2
PANEL_PX = GRID * CELL_PX + 2 * BORDER_PX
SWATCH_PX = 16
KEY_LONG = 2 * BORDER_PX + 2 * SWATCH_PX + 2
KEY_SHORT = 2 * BORDER_PX + SWATCH_PX
DECOY_GRID = 3
DECOY_PX = DECOY_GRID * CELL_PX + 2 * BORDER_PX

QUARANTINE_CLEARANCE = 11.0
BACKGROUND = (238, 236, 230)
BORDER_INK = (18, 18, 20)
DECOY_INKS = ((38, 112, 122), (126, 92, 38), (72, 92, 156), (140, 66, 106))
DECOR_INKS = ((150, 186, 202), (196, 176, 132), (168, 152, 196), (122, 168, 148))

CLASSES = ("both", "target_only", "other_only", "neither")

GLYPHS: dict[str, tuple[tuple[int, int], ...]] = {
    "arrow": ((0, 2), (1, 1), (1, 2), (1, 3), (2, 0), (2, 4), (3, 2), (4, 2)),
    "cross": ((0, 0), (0, 4), (1, 1), (1, 3), (3, 1), (3, 3), (4, 0), (4, 4)),
    "ring": ((1, 1), (1, 2), (1, 3), (2, 1), (2, 3), (3, 1), (3, 2), (3, 3)),
    "stair": ((0, 0), (1, 0), (1, 1), (2, 1), (2, 2), (3, 2), (3, 3), (4, 3)),
}
GLYPH_NAMES = tuple(sorted(GLYPHS))
GLYPH_CELLS = 8


def glyph_mask(name: str) -> np.ndarray:
    """Boolean 5x5 template of one declared glyph."""
    mask = np.zeros((GRID, GRID), dtype=bool)
    for row, column in GLYPHS[name]:
        mask[row, column] = True
    return mask


def glyph_separation(name: str) -> int:
    """Smallest cell-position disagreement between one glyph and any other."""
    base = glyph_mask(name)
    return min(
        int(np.count_nonzero(base ^ glyph_mask(other)))
        for other in GLYPH_NAMES
        if other != name
    )


def _class_field(target: str, other: str) -> np.ndarray:
    """Per-cell class index of the superposed glyph pair."""
    in_target = glyph_mask(target)
    in_other = glyph_mask(other)
    field = np.full((GRID, GRID), CLASSES.index("neither"), dtype=int)
    field[in_target & in_other] = CLASSES.index("both")
    field[in_target & ~in_other] = CLASSES.index("target_only")
    field[~in_target & in_other] = CLASSES.index("other_only")
    return field


def _cells_far_from(used: list[tuple[int, int]], radius: int = 2) -> list[tuple[int, int]]:
    free = []
    for row in range(LATTICE):
        for column in range(LATTICE):
            if all(
                max(abs(row - ur), abs(column - uc)) >= radius for ur, uc in used
            ):
                free.append((row, column))
    return free


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; every field controls visible pixels."""
    rng = np.random.default_rng(int(seed))

    order = rng.permutation(len(GLYPH_NAMES))
    target = GLYPH_NAMES[int(order[0])]
    other = GLYPH_NAMES[int(order[1])]

    bucket = float(rng.random())
    if bucket < 0.72:
        separation = int(rng.integers(30, 45))
    elif bucket < 0.88:
        separation = int(rng.integers(22, 30))
    else:
        separation = int(rng.integers(12, 22))
    extras = [int(value) for value in rng.integers(0, 7, size=2)]
    gaps = [separation, separation + extras[0], separation + extras[1]]
    total = sum(gaps)
    start = int(rng.integers(70, max(71, 212 - total)))
    ladder = [start]
    for gap in gaps:
        ladder.append(ladder[-1] + gap)
    assignment = rng.permutation(4)
    levels = {CLASSES[index]: int(ladder[int(assignment[index])]) for index in range(4)}

    axis = int(rng.integers(0, 2))
    sign = 1 if rng.random() < 0.5 else -1
    if sign > 0:
        base = int(rng.integers(0, LATTICE - 2))
    else:
        base = int(rng.integers(2, LATTICE))
    free_axis = int(rng.integers(0, LATTICE))
    if axis == 0:
        panel_cell = (free_axis, base)
        key_cell = (free_axis, base + 2 * sign)
    else:
        panel_cell = (base, free_axis)
        key_cell = (base + 2 * sign, free_axis)

    slack = (WINDOW_HI - WINDOW_LO) - PANEL_PX
    panel_jitter = [int(rng.integers(0, slack + 1)), int(rng.integers(0, slack + 1))]
    key_horizontal = bool(rng.integers(0, 2))
    key_width = KEY_LONG if key_horizontal else KEY_SHORT
    key_height = KEY_SHORT if key_horizontal else KEY_LONG
    key_jitter = [
        int(rng.integers(0, (WINDOW_HI - WINDOW_LO) - key_width + 1)),
        int(rng.integers(0, (WINDOW_HI - WINDOW_LO) - key_height + 1)),
    ]
    key_swapped = bool(rng.integers(0, 2))

    used = [list(panel_cell), list(key_cell)]
    free = _cells_far_from([tuple(cell) for cell in used])
    picks = rng.permutation(len(free))
    decoys = []
    decor = []
    decoy_count = int(rng.integers(3, 6))
    decor_count = int(rng.integers(4, 8))
    cursor = 0
    while len(decoys) < decoy_count and cursor < len(picks):
        cell = free[int(picks[cursor])]
        cursor += 1
        slack_decoy = (WINDOW_HI - WINDOW_LO) - DECOY_PX
        decoys.append(
            {
                "cell": [int(cell[0]), int(cell[1])],
                "jitter": [
                    int(rng.integers(0, slack_decoy + 1)),
                    int(rng.integers(0, slack_decoy + 1)),
                ],
                "ink": int(rng.integers(0, len(DECOY_INKS))),
                "levels": [int(rng.integers(0, 4)) for _ in range(DECOY_GRID * DECOY_GRID)],
            }
        )
    while len(decor) < decor_count and cursor < len(picks):
        cell = free[int(picks[cursor])]
        cursor += 1
        decor.append(
            {
                "cell": [int(cell[0]), int(cell[1])],
                "jitter": [int(rng.integers(0, 21)), int(rng.integers(0, 21))],
                "ink": int(rng.integers(0, len(DECOR_INKS))),
                "kind": int(rng.integers(0, 3)),
                "size": int(rng.integers(10, 21)),
            }
        )

    return {
        "canvas": CANVAS,
        "lattice": LATTICE,
        "grid": GRID,
        "cell_px": CELL_PX,
        "border_px": BORDER_PX,
        "target": target,
        "other": other,
        "separation": separation,
        "levels": levels,
        "panel": {"cell": [int(panel_cell[0]), int(panel_cell[1])], "jitter": panel_jitter},
        "key": {
            "cell": [int(key_cell[0]), int(key_cell[1])],
            "jitter": key_jitter,
            "horizontal": key_horizontal,
            "swapped": key_swapped,
        },
        "decoys": decoys,
        "decor": decor,
    }


def _origin(cell: list[int], jitter: list[int]) -> tuple[int, int]:
    x = LATTICE_PX * int(cell[1]) + WINDOW_LO + int(jitter[0])
    y = LATTICE_PX * int(cell[0]) + WINDOW_LO + int(jitter[1])
    return x, y


def panel_box(scene: dict) -> tuple[int, int, int, int]:
    x, y = _origin(scene["panel"]["cell"], scene["panel"]["jitter"])
    return x, y, x + PANEL_PX, y + PANEL_PX


def key_box(scene: dict) -> tuple[int, int, int, int]:
    x, y = _origin(scene["key"]["cell"], scene["key"]["jitter"])
    width = KEY_LONG if scene["key"]["horizontal"] else KEY_SHORT
    height = KEY_SHORT if scene["key"]["horizontal"] else KEY_LONG
    return x, y, x + width, y + height


def key_levels(scene: dict) -> tuple[int, int]:
    """The two grays printed as key swatches, in drawn order."""
    first = int(scene["levels"]["both"])
    second = int(scene["levels"]["target_only"])
    if scene["key"]["swapped"]:
        return second, first
    return first, second


def marked_mask(scene: dict) -> np.ndarray:
    """The 5x5 mask produced by the declared palette sieve."""
    field = _class_field(scene["target"], scene["other"])
    keyed = {CLASSES.index("both"), CLASSES.index("target_only")}
    return np.isin(field, list(keyed))


def _fill(array: np.ndarray, x0: int, y0: int, x1: int, y1: int, color) -> None:
    array[max(0, y0) : max(0, y1), max(0, x0) : max(0, x1)] = color


def render(scene: dict) -> Image.Image:
    """Rasterize one scene with flat, non-antialiased block fills."""
    array = np.zeros((CANVAS, CANVAS, 3), dtype=np.uint8)
    array[:, :] = BACKGROUND

    for item in scene["decor"]:
        x, y = _origin(item["cell"], item["jitter"])
        ink = DECOR_INKS[int(item["ink"]) % len(DECOR_INKS)]
        size = int(item["size"])
        kind = int(item["kind"])
        if kind == 0:
            _fill(array, x, y, x + size, y + size, ink)
        elif kind == 1:
            _fill(array, x, y + size // 2 - 1, x + size, y + size // 2 + 2, ink)
            _fill(array, x + size // 2 - 1, y, x + size // 2 + 2, y + size, ink)
        else:
            _fill(array, x, y, x + size, y + 3, ink)
            _fill(array, x, y + size - 3, x + size, y + size, ink)
            _fill(array, x, y, x + 3, y + size, ink)
            _fill(array, x + size - 3, y, x + size, y + size, ink)

    for decoy in scene["decoys"]:
        x, y = _origin(decoy["cell"], decoy["jitter"])
        ink = DECOY_INKS[int(decoy["ink"]) % len(DECOY_INKS)]
        _fill(array, x, y, x + DECOY_PX, y + DECOY_PX, ink)
        for index, level_index in enumerate(decoy["levels"]):
            row, column = divmod(index, DECOY_GRID)
            gray = int(scene["levels"][CLASSES[int(level_index) % 4]])
            cx = x + BORDER_PX + column * CELL_PX
            cy = y + BORDER_PX + row * CELL_PX
            _fill(array, cx, cy, cx + CELL_PX, cy + CELL_PX, (gray, gray, gray))

    px0, py0, px1, py1 = panel_box(scene)
    _fill(array, px0, py0, px1, py1, BORDER_INK)
    field = _class_field(scene["target"], scene["other"])
    for row in range(GRID):
        for column in range(GRID):
            gray = int(scene["levels"][CLASSES[int(field[row, column])]])
            cx = px0 + BORDER_PX + column * CELL_PX
            cy = py0 + BORDER_PX + row * CELL_PX
            _fill(array, cx, cy, cx + CELL_PX, cy + CELL_PX, (gray, gray, gray))

    kx0, ky0, kx1, ky1 = key_box(scene)
    _fill(array, kx0, ky0, kx1, ky1, BORDER_INK)
    first, second = key_levels(scene)
    for index, gray in enumerate((first, second)):
        if scene["key"]["horizontal"]:
            sx = kx0 + BORDER_PX + index * (SWATCH_PX + 2)
            sy = ky0 + BORDER_PX
        else:
            sx = kx0 + BORDER_PX
            sy = ky0 + BORDER_PX + index * (SWATCH_PX + 2)
        _fill(array, sx, sy, sx + SWATCH_PX, sy + SWATCH_PX, (gray, gray, gray))

    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """The glyph selected by the two grays printed in the key."""
    return str(scene["target"])


def margin(scene: dict) -> float:
    """Per-cell gray clearance scaled by the runner-up glyph separation."""
    clearance = float(scene["separation"]) / 2.0
    return round(clearance * glyph_separation(str(scene["target"])) / GLYPH_CELLS, 6)


def is_quarantined(scene: dict) -> bool:
    """True when the rendered gray clearance is below the declared floor."""
    return bool(float(scene["separation"]) / 2.0 < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered."""
    return []
