"""Deterministic two-card interval-centering renderer and analytic reference."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
TARGET_PALETTE = {
    "amber": [226, 158, 34],
    "violet": [128, 82, 190],
}
LABEL_PALETTE = {
    "P": [54, 102, 190],
    "Q": [213, 94, 56],
}
CARD_PALETTE = ([246, 248, 252], [242, 247, 246], [248, 245, 250])
GUIDE_PALETTE = ([68, 79, 96], [73, 88, 82], [81, 73, 94])


def _target_for_interval(
    rng: random.Random, lo: int, hi: int, relative_offset: float
) -> int:
    midpoint = (lo + hi) / 2.0
    direction = -1 if rng.randrange(2) == 0 else 1
    return int(round(midpoint + direction * relative_offset * (hi - lo)))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced scenes with two separated, raster-safe interval cards."""
    rng = random.Random(int(seed))
    card_width = int(rng.choice((166, 174, 182)))
    card_height = int(rng.choice((294, 306, 318)))
    left_card_x = int(rng.choice((18, 26, 34)))
    right_card_x = int(rng.choice((296, 304, 312)))
    card_y = int(rng.choice((92, 100, 108)))

    left_lo = left_card_x + int(rng.choice((24, 28, 32)))
    left_hi = left_card_x + card_width - int(rng.choice((24, 28, 32)))
    right_lo = right_card_x + int(rng.choice((24, 28, 32)))
    right_hi = right_card_x + card_width - int(rng.choice((24, 28, 32)))
    if left_hi - left_lo < 100 or right_hi - right_lo < 100:
        raise AssertionError("sampled interval is too short")

    # The parity-controlled winner makes the three public stress sets balanced.
    left_wins = int(seed) % 2 == 0
    close_ratio = rng.choice((0.07, 0.10, 0.13, 0.16))
    far_ratio = rng.choice((0.31, 0.35, 0.39, 0.43))
    if left_wins:
        left_ratio, right_ratio = close_ratio, far_ratio
    else:
        left_ratio, right_ratio = far_ratio, close_ratio

    panel_labels = ["P", "Q"]
    if rng.randrange(2):
        panel_labels.reverse()
    target_names = ["amber", "violet"]
    rng.shuffle(target_names)
    return {
        "left_target_x": _target_for_interval(rng, left_lo, left_hi, left_ratio),
        "left_interval_lo": int(left_lo),
        "left_interval_hi": int(left_hi),
        "right_target_x": _target_for_interval(rng, right_lo, right_hi, right_ratio),
        "right_interval_lo": int(right_lo),
        "right_interval_hi": int(right_hi),
        "panel_labels": panel_labels,
        "left_card_x": left_card_x,
        "right_card_x": right_card_x,
        "card_y": card_y,
        "card_width": card_width,
        "card_height": card_height,
        "target_color_map": {
            "left": list(TARGET_PALETTE[target_names[0]]),
            "right": list(TARGET_PALETTE[target_names[1]]),
        },
        "guide_color": list(rng.choice(GUIDE_PALETTE)),
        "card_color": list(rng.choice(CARD_PALETTE)),
        "background": [255, 255, 255],
        "target_half_size": int(rng.choice((15, 17, 19))),
        "stroke_width": int(rng.choice((4, 5, 6))),
        "target_y_offset": int(rng.choice((112, 120, 128))),
    }


def _draw_label(
    draw: ImageDraw.ImageDraw,
    label: str,
    x: int,
    y: int,
    color: tuple[int, int, int],
) -> None:
    # The colored chip makes the panel identity recoverable without OCR. The
    # large letter is a human-facing redundant label, not an answer cue.
    draw.rounded_rectangle([x, y, x + 36, y + 32], radius=7, fill=color)
    try:
        font = ImageFont.truetype("DejaVuSans-Bold.ttf", 21)
    except OSError:
        font = ImageFont.load_default()
    draw.text((x + 10, y + 3), label, fill=(255, 255, 255), font=font)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    side: str,
    card_x: int,
    label: str,
    target_x: int,
    interval_lo: int,
    interval_hi: int,
) -> None:
    card_y = int(scene["card_y"])
    card_width = int(scene["card_width"])
    card_height = int(scene["card_height"])
    card_color = tuple(int(value) for value in scene["card_color"])
    guide = tuple(int(value) for value in scene["guide_color"])
    stroke = int(scene["stroke_width"])
    draw.rounded_rectangle(
        [card_x, card_y, card_x + card_width, card_y + card_height],
        radius=18,
        fill=card_color,
        outline=(211, 218, 228),
        width=2,
    )
    _draw_label(draw, label, card_x + 20, card_y + 22, tuple(LABEL_PALETTE[label]))

    target_y = card_y + int(scene["target_y_offset"])
    half = int(scene["target_half_size"])
    target_color = tuple(int(value) for value in scene["target_color_map"][side])
    draw.polygon(
        [
            (target_x, target_y - half),
            (target_x + half, target_y),
            (target_x, target_y + half),
            (target_x - half, target_y),
        ],
        fill=target_color,
    )
    ruler_y = card_y + int(card_height * 0.70)
    draw.line([interval_lo, ruler_y, interval_hi, ruler_y], fill=guide, width=stroke)
    tick_half = 24
    draw.line(
        [interval_lo, ruler_y - tick_half, interval_lo, ruler_y + tick_half],
        fill=guide,
        width=stroke,
    )
    draw.line(
        [interval_hi, ruler_y - tick_half, interval_hi, ruler_y + tick_half],
        fill=guide,
        width=stroke,
    )
    midpoint = (interval_lo + interval_hi) / 2.0
    draw.line(
        [int(round(midpoint)), ruler_y - 12, int(round(midpoint)), ruler_y + 12],
        fill=guide,
        width=max(2, stroke - 2),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary."""
    image = Image.new(
        "RGB", (WIDTH, HEIGHT), tuple(int(value) for value in scene["background"])
    )
    draw = ImageDraw.Draw(image)
    _draw_card(
        draw,
        scene,
        "left",
        int(scene["left_card_x"]),
        str(scene["panel_labels"][0]),
        int(scene["left_target_x"]),
        int(scene["left_interval_lo"]),
        int(scene["left_interval_hi"]),
    )
    _draw_card(
        draw,
        scene,
        "right",
        int(scene["right_card_x"]),
        str(scene["panel_labels"][1]),
        int(scene["right_target_x"]),
        int(scene["right_interval_lo"]),
        int(scene["right_interval_hi"]),
    )
    return image


def _normalized_offset(target: int, lo: int, hi: int) -> float:
    if hi <= lo:
        raise ValueError("interval endpoints are not ordered")
    return abs(float(target) - (float(lo) + float(hi)) / 2.0) / float(hi - lo)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the label of the card with the smaller normalized offset."""
    left = _normalized_offset(
        int(scene["left_target_x"]),
        int(scene["left_interval_lo"]),
        int(scene["left_interval_hi"]),
    )
    right = _normalized_offset(
        int(scene["right_target_x"]),
        int(scene["right_interval_lo"]),
        int(scene["right_interval_hi"]),
    )
    if left == right:
        raise ValueError("centering comparison is tied")
    return str(scene["panel_labels"][0] if left < right else scene["panel_labels"][1])


def margin(scene: dict[str, Any]) -> float:
    """Return the absolute difference between the two normalized offsets."""
    left = _normalized_offset(
        int(scene["left_target_x"]),
        int(scene["left_interval_lo"]),
        int(scene["left_interval_hi"]),
    )
    right = _normalized_offset(
        int(scene["right_target_x"]),
        int(scene["right_interval_lo"]),
        int(scene["right_interval_hi"]),
    )
    return float(abs(left - right))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude near ties, short rulers, and targets outside safe clearance."""
    try:
        analytic_gold(scene)
        left_lo = int(scene["left_interval_lo"])
        left_hi = int(scene["left_interval_hi"])
        right_lo = int(scene["right_interval_lo"])
        right_hi = int(scene["right_interval_hi"])
        left_target = int(scene["left_target_x"])
        right_target = int(scene["right_target_x"])
        left_score = _normalized_offset(left_target, left_lo, left_hi)
        right_score = _normalized_offset(right_target, right_lo, right_hi)
    except (KeyError, TypeError, ValueError):
        return True
    return bool(
        margin(scene) < 0.10
        or left_hi - left_lo < 82
        or right_hi - right_lo < 82
        or left_score > 0.46
        or right_score > 0.46
        or not (left_lo < left_target < left_hi and right_lo < right_target < right_hi)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No render-invariant latent aliases are claimed for this world."""
    return []
