"""Deterministic scene sampler and Pillow renderer."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 360
COLORS = {
    "teal": (18, 154, 143),
    "amber": (236, 153, 36),
    "magenta": (199, 63, 126),
}
INK = (39, 48, 62)
FRAME = (188, 197, 208)
PANEL_X = (20, 270, 520, 770)
Y_LEVELS = (98, 180, 262)


def _match_count(panel: dict) -> int:
    return sum(
        panel["left_colors"][left] == panel["right_colors"][right]
        for left, right in enumerate(panel["wiring"])
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    labels = ["A", "B", "C", "D"]
    target = seed % 4
    wiring = list(rng.choice(list(itertools.permutations(range(3)))))
    base = list(COLORS)
    panels = []
    for index, label in enumerate(labels):
        left = base[:]
        rng.shuffle(left)
        if index == target:
            right = [None, None, None]
            for li, ri in enumerate(wiring):
                right[ri] = left[li]
        else:
            choices = []
            for perm in itertools.permutations(base):
                candidate = {
                    "left_colors": left,
                    "right_colors": list(perm),
                    "wiring": wiring,
                }
                count = _match_count(candidate)
                if count <= 1:
                    choices.append((count, list(perm)))
            desired = 0 if (index + seed) % 3 == 0 else 1
            pool = [perm for count, perm in choices if count == desired]
            right = list(rng.choice(pool))
        panels.append(
            {
                "label": label,
                "left_colors": left,
                "right_colors": right,
                "wiring": wiring,
            }
        )
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=22)
    for panel_index, panel in enumerate(panels):
        x0 = PANEL_X[panel_index]
        draw.rounded_rectangle((x0, 42, x0 + 210, 302), radius=12, outline=FRAME, width=3)
        left_x, right_x = x0 + 38, x0 + 172
        for left_index, right_index in enumerate(panel["wiring"]):
            draw.line(
                (left_x, Y_LEVELS[left_index], right_x, Y_LEVELS[right_index]),
                fill=INK,
                width=7,
            )
        for side_x, key in ((left_x, "left_colors"), (right_x, "right_colors")):
            for y, color_name in zip(Y_LEVELS, panel[key]):
                fill = COLORS[color_name]
                draw.ellipse((side_x - 16, y - 16, side_x + 16, y + 16), fill=fill, outline=(255, 255, 255), width=3)
        bbox = draw.textbbox((0, 0), panel["label"], font=font)
        tw = bbox[2] - bbox[0]
        draw.text((x0 + 105 - tw / 2, 316), panel["label"], fill=INK, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    # Keep the gold entrypoint's scene access explicit for the protected
    # rendered-field audit; nested panel members are part of this one field.
    winners = []
    for panel in panels:
        score = sum(
            panel["left_colors"][left] == panel["right_colors"][right]
            for left, right in enumerate(panel["wiring"])
        )
        if score == 3:
            winners.append(panel["label"])
    if len(winners) != 1:
        raise ValueError("scene does not have a unique color-preserving panel")
    return winners[0]


def margin(scene: dict) -> float:
    panels = scene["panels"]
    scores = sorted((_match_count(panel) for panel in panels), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
