"""Independent final-PNG-only inverse arm."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "teal": np.array((18, 154, 143), dtype=float),
    "amber": np.array((236, 153, 36), dtype=float),
    "magenta": np.array((199, 63, 126), dtype=float),
}
INK = np.array((39, 48, 62), dtype=float)


def _components(mask: np.ndarray, minimum: int = 80) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            stack = [(y, x)]
            seen[y, x] = True
            xs, ys = [], []
            while stack:
                cy, cx = stack.pop()
                xs.append(cx)
                ys.append(cy)
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
            if len(xs) >= minimum:
                found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def _segment_coverage(ink: np.ndarray, start: tuple[float, float], end: tuple[float, float]) -> float:
    x0, y0 = start
    x1, y1 = end
    hits = 0
    total = 0
    for step in range(18, 83):
        t = step / 100.0
        x = x0 + t * (x1 - x0)
        y = y0 + t * (y1 - y0)
        local = False
        for oy in range(-3, 4):
            for ox in range(-3, 4):
                yy, xx = int(round(y + oy)), int(round(x + ox))
                if 0 <= yy < ink.shape[0] and 0 <= xx < ink.shape[1] and ink[yy, xx]:
                    local = True
        hits += int(local)
        total += 1
    return hits / total


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=float)
    endpoints = []
    for name, reference in PALETTE.items():
        distance = np.linalg.norm(rgb - reference, axis=2)
        for x, y, size in _components(distance < 25.0):
            endpoints.append({"x": x, "y": y, "color": name, "size": size})
    if len(endpoints) != 24:
        raise ValueError(f"expected 24 colored endpoints, found {len(endpoints)}")

    endpoints.sort(key=lambda item: item["x"])
    columns = []
    for endpoint in endpoints:
        if not columns or abs(endpoint["x"] - sum(item["x"] for item in columns[-1]) / len(columns[-1])) > 30:
            columns.append([endpoint])
        else:
            columns[-1].append(endpoint)
    if len(columns) != 8 or any(len(column) != 3 for column in columns):
        raise ValueError("could not recover eight three-endpoint columns")
    groups = [columns[index] + columns[index + 1] for index in range(0, 8, 2)]

    ink_distance = np.linalg.norm(rgb - INK, axis=2)
    ink = ink_distance < 35.0
    scores = []
    for group in groups:
        group = sorted(group, key=lambda item: item["x"])
        left, right = group[:3], group[3:]
        left.sort(key=lambda item: item["y"])
        right.sort(key=lambda item: item["y"])
        used = set()
        pairs = []
        for li, source in enumerate(left):
            options = []
            for ri, target in enumerate(right):
                coverage = _segment_coverage(ink, (source["x"], source["y"]), (target["x"], target["y"]))
                options.append((coverage, ri))
            options.sort(reverse=True)
            pairs.append((options, li))
        best = None
        for a in range(3):
            for b in range(3):
                for c in range(3):
                    mapping = (a, b, c)
                    if len(set(mapping)) < 3:
                        continue
                    coverage = sum(pairs[li][0][next(k for k, (_, ri) in enumerate(pairs[li][0]) if ri == mapping[li])][0] for li in range(3))
                    if best is None or coverage > best[0]:
                        best = (coverage, mapping)
        if best is None or best[0] < 2.55:
            raise ValueError("wire pairing is visually ambiguous")
        match_count = sum(left[li]["color"] == right[ri]["color"] for li, ri in enumerate(best[1]))
        scores.append(match_count)
    winners = [index for index, score in enumerate(scores) if score == 3]
    if len(winners) != 1 or 3 - sorted(scores, reverse=True)[1] < 2:
        raise ValueError("no unique pixel-supported winner")
    return "ABCD"[winners[0]]
