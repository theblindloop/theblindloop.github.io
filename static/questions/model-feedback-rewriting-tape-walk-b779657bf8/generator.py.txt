"""Rewriting tape walk: the strip the machine reads is destroyed as it runs.

The raster shows a rule table whose four rows are labelled by state colors and
whose two columns are labelled by a black square and a black circle.  Every cell
holds one colored arrowhead: its hue is the successor state and the side it
points is the direction the head then steps.  A black annulus marks the start
state; a black bar under one strip cell marks where the head starts.

The single mechanism that separates this world from a read-only tape walk is
that every cell the head reads is *rewritten*: a square becomes a circle and a
circle becomes a square the moment it is read.  The drawn strip is therefore
only the tape's initial condition.  From the first revisit onwards, the symbol
the machine consumes is not the symbol drawn in that cell, and no raster shows
the tape the machine is actually working on.  The reported decision is the state
color after seven steps.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- constants

WIDTH = 1040
HEIGHT = 900

WHITE = (255, 255, 255)
BLACK = (18, 18, 18)

STATES = ("amber", "teal", "violet", "crimson")
STATE_RGB = {
    "amber": (240, 170, 20),
    "teal": (20, 160, 160),
    "violet": (150, 60, 200),
    "crimson": (210, 40, 60),
}
SHAPES = ("square", "circle")
OTHER = {"square": "circle", "circle": "square"}

DISK_R = 38
RING_INNER_R = 52
RING_OUTER_R = 66
ARROW_HALF = 40        # colored cell arrowhead half-extent
GLYPH_HALF = 30        # black silhouette half-extent

HEADER_Y = 86
ROW_Y0 = 210
ROW_PITCH = 150
LABEL_X = 200
CELL_X0 = 500
CELL_PITCH = 230
TAPE_Y = 790
TAPE_X0 = 120
TAPE_PITCH = 100
TAPE_LEN = 9
BAR_Y = 852
BAR_H = 12
BAR_HALF_W = 30

STEPS = 7
MIN_MARGIN = 11.0


# ---------------------------------------------------------------- machine

def _run(table, start, tape, head, steps, rewrite=True):
    """Run the machine.

    ``rewrite=True`` is this world's mechanism: the cell under the head flips to
    the other shape as it is read.  ``rewrite=False`` is the read-only tape walk
    of the archive, kept here only so sampling can guarantee that the two
    readings disagree.

    Returns ``(final_state, visited_positions, effective_symbols)`` or ``None``
    when the head walks off the strip.
    """
    cur = start
    pos = head
    band = list(tape)
    visited = [pos]
    consumed = []
    for _ in range(steps):
        if pos < 0 or pos >= len(band):
            return None
        sym = band[pos]
        consumed.append(sym)
        cell = table[cur][sym]
        if rewrite:
            band[pos] = OTHER[sym]
        cur = cell[0]
        pos = pos - 1 if cell[1] == "left" else pos + 1
        visited.append(pos)
    if pos < 0 or pos >= len(band):
        return None
    return cur, visited, consumed


def _rightward(table, start, tape, head, steps):
    """Shortcut reading: ignore the arrowhead sides, always step right."""
    cur = start
    pos = head
    band = list(tape)
    for _ in range(steps):
        p = min(pos, len(band) - 1)
        sym = band[p]
        band[p] = OTHER[sym]
        cur = table[cur][sym][0]
        pos += 1
    return cur


def _left_to_right(table, start, tape, steps):
    """Shortcut reading: consume the drawn strip from its left end."""
    cur = start
    for k in range(steps):
        cur = table[cur][tape[k]][0]
    return cur


def _mode_cell_color(table):
    counts = {s: 0 for s in STATES}
    for row in STATES:
        for sym in SHAPES:
            counts[table[row][sym][0]] += 1
    best = max(counts.values())
    winners = [s for s in STATES if counts[s] == best]
    return winners[0] if len(winners) == 1 else None


# ---------------------------------------------------------------- sampling

def sample_scene(seed: int) -> dict:
    """Sample one latent scene.  Every field below is drawn in the raster."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(40000):
        table = {
            row: {
                sym: (rng.choice(STATES), rng.choice(("left", "right")))
                for sym in SHAPES
            }
            for row in STATES
        }
        # the successor color must depend on the symbol within every row and on
        # the row within every column, so no row or column can be collapsed
        if any(len({table[r][s][0] for s in SHAPES}) < 2 for r in STATES):
            continue
        if any(len({table[r][s][0] for r in STATES}) < 3 for s in SHAPES):
            continue
        # the step direction must likewise depend on both indices
        if any(len({table[r][s][1] for s in SHAPES}) < 2 for r in STATES):
            continue
        if any(len({table[r][s][1] for r in STATES}) < 2 for s in SHAPES):
            continue

        start = rng.choice(STATES)
        tape = [rng.choice(SHAPES) for _ in range(TAPE_LEN)]
        if any(tape.count(s) < 3 for s in SHAPES):
            continue
        head = rng.randint(3, TAPE_LEN - 4)

        res = _run(table, start, tape, head, STEPS)
        if res is None:
            continue
        final, visited, consumed = res

        # the walk must wander and re-read, otherwise rewriting is inert
        steps_dir = [b - a for a, b in zip(visited, visited[1:])]
        reversals = sum(1 for a, b in zip(steps_dir, steps_dir[1:]) if a != b)
        if reversals < 2:
            continue
        read = visited[:-1]
        if len(set(read)) > STEPS - 3:      # at least three reads land on a revisit
            continue
        # the rewriting must actually bite: at least three consumed symbols must
        # differ from the symbol drawn in that cell
        altered = sum(
            1 for pos, sym in zip(read, consumed) if sym != tape[pos]
        )
        if altered < 3:
            continue
        states = [start]
        cur = start
        for sym in consumed:
            cur = table[cur][sym][0]
            states.append(cur)
        if len(set(states)) < 3:
            continue

        # the archive's read-only tape walk must land somewhere else
        plain = _run(table, start, tape, head, STEPS, rewrite=False)
        if plain is None or plain[0] == final:
            continue
        # so must every other cheap reading
        if final == start:
            continue
        if final == _rightward(table, start, tape, head, STEPS):
            continue
        if final == _left_to_right(table, start, tape, STEPS):
            continue
        if final == table[start][tape[head]][0]:
            continue
        short = _run(table, start, tape, head, STEPS - 1)
        if short is None or final == short[0]:
            continue
        longer = _run(table, start, tape, head, STEPS + 1)
        if longer is not None and final == longer[0]:
            continue
        mode = _mode_cell_color(table)
        if mode is not None and final == mode:
            continue

        rows = list(STATES)
        rng.shuffle(rows)
        cols = list(SHAPES)
        rng.shuffle(cols)
        jit = {
            "rows": [rng.randint(-4, 4) for _ in range(4)],
            "cols": [rng.randint(-4, 4) for _ in range(2)],
            "tape": [rng.randint(-3, 3) for _ in range(TAPE_LEN)],
            "header": rng.randint(-3, 3),
        }
        return {
            "seed": seed,
            "rows": rows,
            "cols": cols,
            "table": table,
            "start": start,
            "tape": tape,
            "head": head,
            "jitter": jit,
        }
    raise RuntimeError("rejection sampling failed for seed %r" % (seed,))


# ---------------------------------------------------------------- geometry

def _row_y(scene, i):
    return ROW_Y0 + ROW_PITCH * i + scene["jitter"]["rows"][i]


def _col_x(scene, j):
    return CELL_X0 + CELL_PITCH * j + scene["jitter"]["cols"][j]


def _tape_x(scene, k):
    return TAPE_X0 + TAPE_PITCH * k + scene["jitter"]["tape"][k]


# ---------------------------------------------------------------- drawing

def _disk(draw, cx, cy, r, rgb):
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=rgb)


def _arrowhead(draw, cx, cy, side, rgb):
    if side == "right":
        pts = [(cx - ARROW_HALF, cy - ARROW_HALF),
               (cx - ARROW_HALF, cy + ARROW_HALF),
               (cx + ARROW_HALF, cy)]
    else:
        pts = [(cx + ARROW_HALF, cy - ARROW_HALF),
               (cx + ARROW_HALF, cy + ARROW_HALF),
               (cx - ARROW_HALF, cy)]
    draw.polygon(pts, fill=rgb)


def _glyph(draw, cx, cy, shape):
    box = [cx - GLYPH_HALF, cy - GLYPH_HALF, cx + GLYPH_HALF, cy + GLYPH_HALF]
    if shape == "square":
        draw.rectangle(box, fill=BLACK)
    elif shape == "circle":
        draw.ellipse(box, fill=BLACK)
    else:  # pragma: no cover - defensive
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(img)

    hy = HEADER_Y + scene["jitter"]["header"]
    for j, shape in enumerate(scene["cols"]):
        _glyph(draw, _col_x(scene, j), hy, shape)

    for i, row_state in enumerate(scene["rows"]):
        cy = _row_y(scene, i)
        if row_state == scene["start"]:
            _disk(draw, LABEL_X, cy, RING_OUTER_R, BLACK)
            _disk(draw, LABEL_X, cy, RING_INNER_R, WHITE)
        _disk(draw, LABEL_X, cy, DISK_R, STATE_RGB[row_state])
        for j, shape in enumerate(scene["cols"]):
            nxt, side = scene["table"][row_state][shape]
            _arrowhead(draw, _col_x(scene, j), cy, side, STATE_RGB[nxt])

    for k, shape in enumerate(scene["tape"]):
        _glyph(draw, _tape_x(scene, k), TAPE_Y, shape)

    bx = _tape_x(scene, scene["head"])
    draw.rectangle(
        [bx - BAR_HALF_W, BAR_Y, bx + BAR_HALF_W, BAR_Y + BAR_H], fill=BLACK
    )
    return img


# ---------------------------------------------------------------- contract

def analytic_gold(scene: dict) -> str:
    res = _run(scene["table"], scene["start"], scene["tape"], scene["head"], STEPS)
    if res is None:  # pragma: no cover - sampling forbids this
        raise ValueError("head left the strip")
    return res[0]


def margin(scene: dict) -> float:
    gaps = [
        float(RING_OUTER_R - RING_INNER_R),
        float(RING_INNER_R - DISK_R),
        float(BAR_H),
    ]
    rows = [_row_y(scene, i) for i in range(4)]
    for a, b in zip(rows, rows[1:]):
        gaps.append(float(b - a - 2 * ARROW_HALF))
        gaps.append(float(b - a - RING_OUTER_R - ARROW_HALF))
    cols = [_col_x(scene, j) for j in range(2)]
    for a, b in zip(cols, cols[1:]):
        gaps.append(float(b - a - 2 * ARROW_HALF))
        gaps.append(float(b - a - 2 * GLYPH_HALF))
    gaps.append(float(cols[0] - LABEL_X - RING_OUTER_R - ARROW_HALF))
    taps = [_tape_x(scene, k) for k in range(TAPE_LEN)]
    for a, b in zip(taps, taps[1:]):
        gaps.append(float(b - a - 2 * GLYPH_HALF))
        gaps.append(float(b - a - 2 * BAR_HALF_W))
    hy = HEADER_Y + scene["jitter"]["header"]
    gaps.append(float(rows[0] - ARROW_HALF - hy - GLYPH_HALF))
    gaps.append(float(TAPE_Y - GLYPH_HALF - rows[-1] - RING_OUTER_R))
    gaps.append(float(BAR_Y - TAPE_Y - GLYPH_HALF))
    gaps.append(float(TAPE_X0 - GLYPH_HALF))
    gaps.append(float(WIDTH - taps[-1] - GLYPH_HALF))
    gaps.append(float(HEIGHT - BAR_Y - BAR_H))
    # square versus circle is decided by 2*(1 - pi/4) of the glyph box area
    gaps.append(float(2 * GLYPH_HALF * (1.0 - 0.7853981633974483)))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Rendering is injective over sampled scenes; no nontrivial alias exists."""
    return []
