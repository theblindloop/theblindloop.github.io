"""Pixel-only inverse arm for the rewriting tape walk.

Receives a PIL image and nothing else.  It segments the exact mark colors,
recovers the four table rows and their state colors, each cell's successor color
and pointing side, the annulus-marked start state, the ordered strip of black
squares and circles and the bar-marked head cell, then runs the machine itself
for seven steps while maintaining its own copy of the tape and flipping every
cell as it is read.  It imports no renderer, prompt, generator, verifier or gold
code and never sees latent scene data.
"""

from __future__ import annotations

import numpy as np

_STATE_RGB = {
    "amber": (240, 170, 20),
    "teal": (20, 160, 160),
    "violet": (150, 60, 200),
    "crimson": (210, 40, 60),
}
_BLACK = (18, 18, 18)
_STEPS = 7
_OTHER = {"square": "circle", "circle": "square"}


def _mask(arr, rgb):
    return (
        (arr[:, :, 0] == rgb[0])
        & (arr[:, :, 1] == rgb[1])
        & (arr[:, :, 2] == rgb[2])
    )


def _runs(flags, min_gap=6):
    idx = np.flatnonzero(flags)
    if idx.size == 0:
        return []
    out = [[int(idx[0]), int(idx[0])]]
    for v in idx[1:]:
        v = int(v)
        if v - out[-1][1] <= min_gap:
            out[-1][1] = v
        else:
            out.append([v, v])
    return [tuple(r) for r in out]


def _classify(sub):
    """square or circle, from the filled fraction of the bounding box."""
    ys, xs = np.nonzero(sub)
    h = ys.max() - ys.min() + 1
    w = xs.max() - xs.min() + 1
    ratio = float(sub.sum()) / float(h * w)
    return "square" if ratio >= 0.90 else "circle"


def _glyphs_in_band(black, y0, y1):
    band = black[y0 : y1 + 1, :]
    out = []
    for x0, x1 in _runs(band.any(axis=0)):
        sub = band[:, x0 : x1 + 1]
        out.append(((x0 + x1) / 2.0, _classify(sub)))
    out.sort()
    return out


def decision_from_image(image):
    arr = np.asarray(image.convert("RGB"))

    color_masks = {n: _mask(arr, rgb) for n, rgb in _STATE_RGB.items()}
    any_color = np.zeros(arr.shape[:2], dtype=bool)
    for m in color_masks.values():
        any_color |= m
    black = _mask(arr, _BLACK)

    col_runs = _runs(any_color.any(axis=0))
    row_runs = _runs(any_color.any(axis=1))
    if len(col_runs) != 3 or len(row_runs) != 4:
        raise ValueError(
            "unexpected table layout: %d columns, %d rows"
            % (len(col_runs), len(row_runs))
        )
    label_run = col_runs[0]
    cell_runs = col_runs[1:]

    def _region(y0, y1, x0, x1):
        best, best_n = None, 0
        for name, m in color_masks.items():
            n = int(m[y0 : y1 + 1, x0 : x1 + 1].sum())
            if n > best_n:
                best, best_n = name, n
        if best is None:
            raise ValueError("no colored mark at that grid position")
        return best, color_masks[best][y0 : y1 + 1, x0 : x1 + 1]

    row_states = [_region(y0, y1, *label_run)[0] for y0, y1 in row_runs]
    if len(set(row_states)) != 4:
        raise ValueError("row labels are not four distinct colors")

    # black y-bands, top to bottom: column headers, the start annulus,
    # the symbol strip, the head bar.
    bands = _runs(black.any(axis=1))
    if len(bands) != 4:
        raise ValueError("expected 4 black bands, found %d" % len(bands))
    header_band, ring_band, tape_band, bar_band = bands

    ring_cy = (ring_band[0] + ring_band[1]) / 2.0
    start_idx = min(
        range(4),
        key=lambda i: abs((row_runs[i][0] + row_runs[i][1]) / 2.0 - ring_cy),
    )
    start = row_states[start_idx]

    head_glyphs = _glyphs_in_band(black, *header_band)
    if len(head_glyphs) != 2:
        raise ValueError("expected 2 column headers, found %d" % len(head_glyphs))
    col_shapes = []
    for x0, x1 in cell_runs:
        cx = (x0 + x1) / 2.0
        col_shapes.append(min(head_glyphs, key=lambda g: abs(g[0] - cx))[1])
    if len(set(col_shapes)) != 2:
        raise ValueError("column headers are not two distinct shapes")

    strip = _glyphs_in_band(black, *tape_band)
    if len(strip) < 5:
        raise ValueError("symbol strip too short")
    tape = [s for _, s in strip]

    bar = _runs(black[bar_band[0] : bar_band[1] + 1, :].any(axis=0))
    if len(bar) != 1:
        raise ValueError("expected exactly one head bar, found %d" % len(bar))
    bar_cx = (bar[0][0] + bar[0][1]) / 2.0
    head = min(range(len(strip)), key=lambda k: abs(strip[k][0] - bar_cx))

    table = {}
    for i, (y0, y1) in enumerate(row_runs):
        row = {}
        for j, (x0, x1) in enumerate(cell_runs):
            color, sub = _region(y0, y1, x0, x1)
            ys, xs = np.nonzero(sub)
            lo, hi = int(xs.min()), int(xs.max())
            mid = (lo + hi) / 2.0
            left = int((xs <= mid).sum())
            right = int((xs > mid).sum())
            if left == right:
                raise ValueError("cell arrowhead has no pointing side")
            row[col_shapes[j]] = (color, "right" if left > right else "left")
        table[row_states[i]] = row

    cur = start
    pos = head
    band = list(tape)
    for _ in range(_STEPS):
        if pos < 0 or pos >= len(band):
            raise ValueError("head stepped off the strip")
        sym = band[pos]
        color, side = table[cur][sym]
        band[pos] = _OTHER[sym]
        cur = color
        pos = pos - 1 if side == "left" else pos + 1
    return cur
