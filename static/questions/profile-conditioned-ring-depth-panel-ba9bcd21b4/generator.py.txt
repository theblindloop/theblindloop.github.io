"""Renderer for the ring-depth panel world.

Five grey-framed square panels stand in a row. Every panel holds exactly two
non-touching circular ring outlines of the same two radii (one scene-wide radius
pair, so ink is identical in every panel) and exactly one small solid black dot.
In a panel the pair is either nested (small ring inside the large one) or fully
apart (side by side). The dot lies inside both rings in exactly one panel; in
the other panels it lies inside exactly one ring or inside neither.

The analytic decision is the panel index whose enclosure depth is 2. The margin
is the smallest pixel clearance that certifies any panel's reading: the gap
between the dot's edge and every ring outline it must be inside or outside of,
and the wall-to-wall gap between the two rings of a pair.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ring_depth_panel-0.1.0"

PANELS = 5
PANEL_SIZE = 210
PAD = 12
FRAME_WIDTH = 3
INSET = 7
CANVAS_W = PANELS * PANEL_SIZE + (PANELS + 1) * PAD
CANVAS_H = PANEL_SIZE + 2 * PAD

BACKGROUND = (255, 255, 255)
FRAME_COLOR = (140, 140, 140)
DOT_COLOR = (18, 18, 18)
DOT_R = 6
RING_STROKE = 4

PALETTE = [
    (196, 62, 48),
    (36, 96, 176),
    (34, 132, 92),
    (168, 84, 176),
    (194, 132, 30),
    (60, 60, 148),
]

MIN_CLEARANCE = 9.0          # sampling floor for dot-edge to ring wall
MIN_RING_GAP = 14.0          # sampling floor for wall-to-wall ring separation
QUARANTINE_MARGIN = 6.0      # below this the reading is not certified

TYPES = ("nested_annulus", "nested_outside", "apart_big", "apart_small", "apart_outside")


def panel_box(index: int) -> tuple[int, int, int, int]:
    """Outer frame rectangle of one panel (x0, y0, x1, y1)."""
    x0 = PAD + index * (PANEL_SIZE + PAD)
    return x0, PAD, x0 + PANEL_SIZE, PAD + PANEL_SIZE


def panel_inner(index: int) -> tuple[float, float, float, float]:
    """Drawable interior of one panel."""
    x0, y0, x1, y1 = panel_box(index)
    return x0 + INSET, y0 + INSET, x1 - INSET, y1 - INSET


def _fits(cx: float, cy: float, r: float, inner: tuple[float, float, float, float]) -> bool:
    ix0, iy0, ix1, iy1 = inner
    return ix0 <= cx - r and cx + r <= ix1 and iy0 <= cy - r and cy + r <= iy1


def _sample_panel(rng: random.Random, index: int, kind: str, r_big: float, r_small: float) -> dict:
    """One panel record: two rings plus a dot realising the requested kind."""
    inner = panel_inner(index)
    ix0, iy0, ix1, iy1 = inner
    nested = kind in ("target", "nested_annulus", "nested_outside")
    for _ in range(20000):
        if nested:
            bx = rng.uniform(ix0 + r_big, ix1 - r_big)
            by = rng.uniform(iy0 + r_big, iy1 - r_big)
            slack = r_big - r_small - MIN_RING_GAP
            if slack < 0:
                continue
            off = rng.uniform(0.0, slack)
            ang = rng.uniform(0.0, 2 * math.pi)
            sx, sy = bx + off * math.cos(ang), by + off * math.sin(ang)
        else:
            gap = rng.uniform(MIN_RING_GAP, MIN_RING_GAP + 14.0)
            d = r_big + r_small + gap
            bx = rng.uniform(ix0 + r_big, ix1 - r_big)
            by = rng.uniform(iy0 + r_big, iy1 - r_big)
            ang = rng.uniform(0.0, 2 * math.pi)
            sx, sy = bx + d * math.cos(ang), by + d * math.sin(ang)
            if not _fits(sx, sy, r_small, inner):
                continue
        dot = _sample_dot(rng, kind, (bx, by, r_big), (sx, sy, r_small), inner)
        if dot is None:
            continue
        rings = [
            {"cx": round(bx, 3), "cy": round(by, 3), "r": round(r_big, 3), "color": rng.randrange(len(PALETTE))},
            {"cx": round(sx, 3), "cy": round(sy, 3), "r": round(r_small, 3), "color": rng.randrange(len(PALETTE))},
        ]
        if rng.random() < 0.5:
            rings.reverse()
        return {"rings": rings, "dot": {"cx": round(dot[0], 3), "cy": round(dot[1], 3)}}
    raise RuntimeError(f"could not place panel {index} of kind {kind}")


def _sample_dot(rng, kind, big, small, inner):
    bx, by, r_big = big
    sx, sy, r_small = small
    ix0, iy0, ix1, iy1 = inner
    for _ in range(4000):
        if kind == "target":
            t = math.sqrt(rng.random()) * max(0.0, r_small - DOT_R - MIN_CLEARANCE)
            a = rng.uniform(0.0, 2 * math.pi)
            px, py = sx + t * math.cos(a), sy + t * math.sin(a)
        else:
            px = rng.uniform(ix0 + DOT_R, ix1 - DOT_R)
            py = rng.uniform(iy0 + DOT_R, iy1 - DOT_R)
        db = math.dist((px, py), (bx, by))
        ds = math.dist((px, py), (sx, sy))
        if min(abs(db - r_big), abs(ds - r_small)) - DOT_R < MIN_CLEARANCE:
            continue
        if not (ix0 + DOT_R <= px <= ix1 - DOT_R and iy0 + DOT_R <= py <= iy1 - DOT_R):
            continue
        inside_big, inside_small = db < r_big, ds < r_small
        want = {
            "target": (True, True),
            "nested_annulus": (True, False),
            "nested_outside": (False, False),
            "apart_big": (True, False),
            "apart_small": (False, True),
            "apart_outside": (False, False),
        }[kind]
        if (inside_big, inside_small) == want:
            return px, py
    return None


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene. The winning slot is uniform over the five panels."""
    rng = random.Random(seed)
    for _ in range(400):
        r_small = rng.uniform(22.0, 28.0)
        r_big = rng.uniform(r_small + 26.0, min(r_small + 40.0, 88.0 - r_small))
        winner = rng.randrange(PANELS)
        others = ["nested_annulus", "apart_small"]
        others += [rng.choice(TYPES) for _ in range(PANELS - 3)]
        rng.shuffle(others)
        kinds = others[:winner] + ["target"] + others[winner:]
        try:
            panels = [
                _sample_panel(rng, i, kind, r_big, r_small) for i, kind in enumerate(kinds)
            ]
        except RuntimeError:
            continue
        scene = {
            "canvas": [CANVAS_W, CANVAS_H],
            "palette": rng.randrange(len(PALETTE)) + len(PALETTE),
            "panels": panels,
        }
        if _depths(scene).count(2) == 1 and margin(scene) >= MIN_CLEARANCE - 1.0:
            return scene
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def _depths(scene: dict) -> list[int]:
    depths = []
    for panel in scene["panels"]:
        dot = panel["dot"]
        depths.append(
            sum(
                1
                for ring in panel["rings"]
                if math.dist((dot["cx"], dot["cy"]), (ring["cx"], ring["cy"])) < ring["r"]
            )
        )
    return depths


def margin(scene: dict) -> float:
    """Smallest pixel clearance certifying any panel's enclosure reading."""
    best = float("inf")
    for panel in scene["panels"]:
        dot = panel["dot"]
        pt = (dot["cx"], dot["cy"])
        outside = []
        for ring in panel["rings"]:
            d = math.dist(pt, (ring["cx"], ring["cy"]))
            clear = abs(d - ring["r"]) - DOT_R
            if d < ring["r"]:
                best = min(best, clear)
            else:
                outside.append(clear)
        if outside:
            best = min(best, max(outside))
        a, b = panel["rings"]
        centre_gap = math.dist((a["cx"], a["cy"]), (b["cx"], b["cy"]))
        if centre_gap < abs(a["r"] - b["r"]):
            wall = abs(a["r"] - b["r"]) - centre_gap
        else:
            wall = centre_gap - (a["r"] + b["r"])
        best = min(best, wall)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    return _depths(scene).count(2) != 1 or margin(scene) < QUARANTINE_MARGIN


def analytic_gold(scene: dict) -> str:
    depths = _depths(scene)
    if depths.count(2) != 1:
        raise ValueError(f"scene has {depths.count(2)} doubly enclosed dots, not one")
    return f"panel-{depths.index(2) + 1}"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical re-parametrisations that must preserve the answer."""
    out = []
    swapped = copy.deepcopy(scene)
    for panel in swapped["panels"]:
        panel["rings"].reverse()
    out.append(("ring_slot_swap", swapped))
    shifted = copy.deepcopy(scene)
    shifted["palette"] = scene["palette"] + len(PALETTE)
    out.append(("palette_period", shifted))
    return out


def render(scene: dict) -> Image.Image:
    """Flat-colour raster; no antialiasing, so every ink pixel is an exact colour."""
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for index in range(PANELS):
        draw.rectangle(panel_box(index), outline=FRAME_COLOR, width=FRAME_WIDTH)
    for panel in scene["panels"]:
        for ring in panel["rings"]:
            cx, cy, r = ring["cx"], ring["cy"], ring["r"]
            colour = PALETTE[ring["color"] % len(PALETTE)]
            draw.ellipse([cx - r, cy - r, cx + r, cy + r], outline=colour, width=RING_STROKE)
        dot = panel["dot"]
        draw.ellipse(
            [dot["cx"] - DOT_R, dot["cy"] - DOT_R, dot["cx"] + DOT_R, dot["cy"] + DOT_R],
            fill=DOT_COLOR,
        )
    return image
