"""Pixel-only inverse arm for the ring-depth panel world.

The arm receives a rendered PNG and nothing else: no scene, no renderer, no
analytic gold. It rules the grey frames to recover the five panel interiors,
segments the flat inks (saturated ring outlines versus the near-black dot),
locates the dot centroid in each panel, and counts how many closed ring
outlines enclose it by casting rays outward and taking the smallest number of
ink crossings over all directions. A ray leaving a point enclosed by k closed
curves must cross each of them, and some direction misses every curve that does
not enclose the point, so the minimum crossing count is the enclosure depth.
The decision is the unique panel of depth two; anything else abstains.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "ring_depth_panel-oracle-0.1.0"
AMBIGUOUS = "ambiguous"

PANELS = 5
GREY_TOL = 14
FRAME_LO = 110
FRAME_HI = 175
DARK_HI = 70
SAT_LO = 45
RAYS = 144
STEP = 0.5
MIN_DOT_PX = 60
MAX_DOT_PX = 240


def _masks(arr: np.ndarray):
    chan_max = arr.max(axis=2).astype(np.int16)
    chan_min = arr.min(axis=2).astype(np.int16)
    spread = chan_max - chan_min
    mean = arr.mean(axis=2)
    frame = (spread <= GREY_TOL) & (mean >= FRAME_LO) & (mean <= FRAME_HI)
    dot = (spread <= GREY_TOL) & (mean <= DARK_HI)
    ring = spread >= SAT_LO
    return frame, dot, ring


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, flag in enumerate(flags):
        if flag and start is None:
            start = i
        elif not flag and start is not None:
            out.append((start, i))
            start = None
    if start is not None:
        out.append((start, len(flags)))
    return out


def _bands(counts: np.ndarray) -> list[tuple[int, int]]:
    peak = counts.max()
    if peak <= 0:
        return []
    return _runs(counts >= 0.5 * peak)


def _panel_boxes(frame: np.ndarray):
    cols = _bands(frame.sum(axis=0))
    rows = _bands(frame.sum(axis=1))
    if len(cols) != 2 * PANELS or len(rows) != 2:
        return None
    y0, y1 = rows[0][1], rows[1][0]
    if y1 - y0 < 20:
        return None
    boxes = []
    for i in range(PANELS):
        x0, x1 = cols[2 * i][1], cols[2 * i + 1][0]
        if x1 - x0 < 20:
            return None
        boxes.append((x0, y0, x1, y1))
    return boxes


def _depth(ring: np.ndarray, box, px: float, py: float) -> int:
    x0, y0, x1, y1 = box
    best = None
    for k in range(RAYS):
        angle = 2.0 * math.pi * k / RAYS
        dx, dy = math.cos(angle), math.sin(angle)
        crossings = 0
        inside_run = False
        t = 0.0
        while True:
            t += STEP
            x = int(round(px + t * dx))
            y = int(round(py + t * dy))
            if not (x0 <= x < x1 and y0 <= y < y1):
                break
            if ring[y, x]:
                if not inside_run:
                    crossings += 1
                    inside_run = True
            else:
                inside_run = False
        if inside_run:
            return 99
        best = crossings if best is None else min(best, crossings)
        if best == 0:
            break
    return 99 if best is None else best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    frame, dot, ring = _masks(arr)
    boxes = _panel_boxes(frame)
    if boxes is None:
        return AMBIGUOUS
    depths = []
    for box in boxes:
        x0, y0, x1, y1 = box
        sub = dot[y0:y1, x0:x1]
        ys, xs = np.nonzero(sub)
        if not (MIN_DOT_PX <= xs.size <= MAX_DOT_PX):
            return AMBIGUOUS
        px = float(xs.mean()) + x0
        py = float(ys.mean()) + y0
        depths.append(_depth(ring, box, px, py))
    if depths.count(2) != 1:
        return AMBIGUOUS
    return f"panel-{depths.index(2) + 1}"
