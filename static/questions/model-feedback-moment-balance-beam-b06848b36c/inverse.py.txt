"""Scene-free pixel oracle for the moment-balance-beam world.

Receives only the rendered PNG. It splits the canvas into four bands from the
rows of grey notch ticks, and inside each band it locates the red wedge apex,
the notch pitch from the tick centres, and every blue load stack. A stack's
weight is its number of separated blocks and its lever arm is its notch
distance from the wedge, so the band's two moment sums follow. It reports the
unique band whose two sums are equal.

It imports no renderer, prompt, generator, verifier or gold code and never
sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_TICK = (128, 136, 148)
_WEDGE = (198, 42, 52)
_BLOCK = (46, 104, 186)

_NO_ANSWER = "beam-0"


def _mask(arr: np.ndarray, colour: tuple[int, int, int]) -> np.ndarray:
    return (
        (arr[:, :, 0] == colour[0])
        & (arr[:, :, 1] == colour[1])
        & (arr[:, :, 2] == colour[2])
    )


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, on in enumerate(flags):
        if on and start is None:
            start = i
        elif not on and start is not None:
            out.append((start, i - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _median(values: list[float]) -> float:
    ordered = sorted(values)
    mid = len(ordered) // 2
    if len(ordered) % 2:
        return float(ordered[mid])
    return (ordered[mid - 1] + ordered[mid]) / 2.0


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    ticks = _mask(arr, _TICK)
    wedges = _mask(arr, _WEDGE)
    blocks = _mask(arr, _BLOCK)

    tick_bands = _runs(ticks.any(axis=1))
    if len(tick_bands) < 2:
        return _NO_ANSWER

    tops = [band[0] for band in tick_bands]
    limits = tops[1:] + [arr.shape[0]]

    balanced: list[int] = []
    for index, (top, stop) in enumerate(zip(tops, limits)):
        tick_band = ticks[top:stop]
        wedge_band = wedges[top:stop]
        block_band = blocks[top:stop]

        tick_runs = _runs(tick_band.any(axis=0))
        if len(tick_runs) < 4:
            return _NO_ANSWER
        centres = [(a + b) / 2.0 for a, b in tick_runs]
        pitch = _median([b - a for a, b in zip(centres, centres[1:])])
        if pitch <= 1:
            return _NO_ANSWER
        base = centres[0]

        wedge_cols = np.flatnonzero(wedge_band.any(axis=0))
        if wedge_cols.size == 0:
            return _NO_ANSWER
        pivot_notch = round(((wedge_cols[0] + wedge_cols[-1]) / 2.0 - base) / pitch)

        left = right = 0
        for x0, x1 in _runs(block_band.any(axis=0)):
            column = block_band[:, x0 : x1 + 1].any(axis=1)
            weight = len(_runs(column))
            notch = round(((x0 + x1) / 2.0 - base) / pitch)
            arm = notch - pivot_notch
            if arm < 0:
                left += weight * (-arm)
            elif arm > 0:
                right += weight * arm
        if left == right:
            balanced.append(index)

    if len(balanced) != 1:
        return _NO_ANSWER
    return f"beam-{balanced[0] + 1}"
