"""Moment-balance beams: which bar carries the same total turning effect on both sides?

Four horizontal bars are stacked down the canvas. Each bar rests on a red
wedge whose apex sits exactly on one of the bar's evenly spaced notches, and
several loads hang from other notches; a load is a vertical stack of flat blue
blocks, so its weight is a countable integer. A load's moment is its block
count multiplied by its distance from the wedge measured in that bar's own
notches, and a bar balances when the moments summed on its left equal the
moments summed on its right.

Exactly one of the four bars balances. The sampler forces the two magnitude
shortcuts to name definite wrong bars: one non-answer bar carries the same
total number of blocks on both sides, and a different non-answer bar hangs its
loads at mirror-image distances about its wedge, while the answer bar is
neither weight-equal nor mirror-symmetric.

Every mark is an axis-aligned flat-filled rectangle or a flat-filled triangle
with no anti-aliasing, so the raster is byte-deterministic.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 1220
HEIGHT = 1124
N_BEAMS = 4

TOP_PAD = 26
STATION_H = 268

TICK_DY = 8
TICK_H = 22
TICK_W = 8
BEAM_DY = 30
BEAM_H = 18
WEDGE_HALF = 34
WEDGE_H = 54
HANGER_W = 8
HANGER_H = 12
BLOCK_W = 44
BLOCK_H = 30
BLOCK_GAP = 6
END_CAP = 26

MIN_PITCH = 62
MAX_PITCH = 76
MIN_NOTCHES = 9
MAX_NOTCHES = 12
MAX_WEIGHT = 5

QUARANTINE_MARGIN = 3.0

BG = (247, 246, 242)
BEAM = (34, 38, 46)
TICK = (128, 136, 148)
WEDGE = (198, 42, 52)
BLOCK = (46, 104, 186)

_KINDS = ("balanced", "equal_weight", "mirror", "free")


# --------------------------------------------------------------------------
# latent scene
# --------------------------------------------------------------------------
def _moments(pivot: int, loads: list) -> tuple[int, int]:
    """(left moment, right moment) of one bar, in block-notch units."""
    left = right = 0
    for notch, weight in loads:
        if notch < pivot:
            left += weight * (pivot - notch)
        else:
            right += weight * (notch - pivot)
    return left, right


def _weights(pivot: int, loads: list) -> tuple[int, int]:
    left = sum(w for n, w in loads if n < pivot)
    right = sum(w for n, w in loads if n > pivot)
    return left, right


def _is_mirror(pivot: int, loads: list) -> bool:
    left = sorted(pivot - n for n, _ in loads if n < pivot)
    right = sorted(n - pivot for n, _ in loads if n > pivot)
    return left == right


def _sample_bar(rng: random.Random, kind: str) -> tuple[int, int, int, int, list]:
    """Return (n_notches, pitch, pivot, left_x, loads) for one bar of the given kind."""
    for _ in range(20000):
        n_notches = rng.randrange(MIN_NOTCHES, MAX_NOTCHES + 1)
        pitch = rng.randrange(MIN_PITCH, MAX_PITCH + 1)
        pivot = rng.randrange(3, n_notches - 3)
        n_left = rng.choice((2, 3))
        n_right = rng.choice((2, 3))
        left_d = rng.sample(range(1, pivot + 1), n_left)
        right_d = rng.sample(range(1, n_notches - pivot), n_right)
        loads = [[pivot - d, rng.randrange(1, MAX_WEIGHT + 1)] for d in left_d]
        loads += [[pivot + d, rng.randrange(1, MAX_WEIGHT + 1)] for d in right_d]
        loads.sort()
        beam_len = (n_notches - 1) * pitch + 2 * END_CAP
        if beam_len > WIDTH - 80:
            continue
        left_x = rng.randrange(40, WIDTH - 40 - beam_len + 1)
        lm, rm = _moments(pivot, loads)
        lw, rw = _weights(pivot, loads)
        mirror = _is_mirror(pivot, loads)
        equal_weight = lw == rw
        if kind == "balanced":
            if lm == rm and not mirror and not equal_weight and abs(lw - rw) >= 2:
                return n_notches, pitch, pivot, left_x, loads
        elif kind == "equal_weight":
            if equal_weight and not mirror and abs(lm - rm) >= 2:
                return n_notches, pitch, pivot, left_x, loads
        elif kind == "mirror":
            if mirror and not equal_weight and abs(lm - rm) >= 2:
                return n_notches, pitch, pivot, left_x, loads
        else:
            if not mirror and not equal_weight and abs(lm - rm) >= 2:
                return n_notches, pitch, pivot, left_x, loads
    raise RuntimeError(f"could not sample a {kind} bar")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(0x5EED ^ (seed * 2654435761 % (1 << 32)))
    for _ in range(200):
        answer = rng.randrange(N_BEAMS)
        others = ["equal_weight", "mirror", "free"]
        rng.shuffle(others)
        pool = iter(others)
        kinds = ["balanced" if i == answer else next(pool) for i in range(N_BEAMS)]
        bars = [_sample_bar(rng, kind) for kind in kinds]
        totals = [sum(w for _, w in bar[4]) for bar in bars]
        if totals[answer] == max(totals) or totals[answer] == min(totals):
            continue  # total ink must not name the answer
        paint_order = list(range(N_BEAMS))
        rng.shuffle(paint_order)
        return {
            "n_notches": [bar[0] for bar in bars],
            "pitch": [bar[1] for bar in bars],
            "pivot": [bar[2] for bar in bars],
            "left_x": [bar[3] for bar in bars],
            "loads": [bar[4] for bar in bars],
            "paint_order": paint_order,
        }
    raise RuntimeError("could not sample a scene")


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def _answer_index(scene: dict) -> int:
    pivots = scene["pivot"]
    loads = scene["loads"]
    hits = [i for i in range(len(pivots)) if _moments(pivots[i], loads[i])[0] == _moments(pivots[i], loads[i])[1]]
    if len(hits) != 1:
        raise ValueError("scene does not have a unique balanced bar")
    return hits[0]


def analytic_gold(scene: dict) -> str:
    return f"beam-{_answer_index(scene) + 1}"


def margin(scene: dict) -> float:
    """Smallest moment imbalance, in block-notch units, on any non-answer bar."""
    answer = _answer_index(scene)
    pivots = scene["pivot"]
    loads = scene["loads"]
    gaps = []
    for i in range(len(pivots)):
        if i == answer:
            continue
        left, right = _moments(pivots[i], loads[i])
        gaps.append(abs(left - right))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """paint_order is photometrically invisible: bars never share a pixel."""
    order = list(scene["paint_order"])
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    reversed_ = dict(scene)
    reversed_["paint_order"] = list(reversed(order))
    return [("rotate_paint_order", rotated), ("reverse_paint_order", reversed_)]


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _notch_x(left_x: int, pitch: int, notch: int) -> int:
    return left_x + END_CAP + notch * pitch


def _draw_bar(
    draw: ImageDraw.ImageDraw,
    band_top: int,
    n_notches: int,
    pitch: int,
    pivot: int,
    left_x: int,
    loads: list,
) -> None:
    beam_top = band_top + BEAM_DY
    beam_bottom = beam_top + BEAM_H
    beam_len = (n_notches - 1) * pitch + 2 * END_CAP
    draw.rectangle([left_x, beam_top, left_x + beam_len - 1, beam_bottom - 1], fill=BEAM)

    tick_top = band_top + TICK_DY
    for j in range(n_notches):
        cx = _notch_x(left_x, pitch, j)
        draw.rectangle(
            [cx - TICK_W // 2, tick_top, cx - TICK_W // 2 + TICK_W - 1, tick_top + TICK_H - 1],
            fill=TICK,
        )

    px = _notch_x(left_x, pitch, pivot)
    draw.polygon(
        [
            (px - WEDGE_HALF, beam_bottom + WEDGE_H),
            (px + WEDGE_HALF, beam_bottom + WEDGE_H),
            (px, beam_bottom),
        ],
        fill=WEDGE,
    )

    for notch, weight in loads:
        cx = _notch_x(left_x, pitch, notch)
        draw.rectangle(
            [
                cx - HANGER_W // 2,
                beam_bottom,
                cx - HANGER_W // 2 + HANGER_W - 1,
                beam_bottom + HANGER_H - 1,
            ],
            fill=BEAM,
        )
        top = beam_bottom + HANGER_H
        for k in range(weight):
            y = top + k * (BLOCK_H + BLOCK_GAP)
            draw.rectangle(
                [cx - BLOCK_W // 2, y, cx - BLOCK_W // 2 + BLOCK_W - 1, y + BLOCK_H - 1],
                fill=BLOCK,
            )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for index in scene["paint_order"]:
        _draw_bar(
            draw,
            TOP_PAD + index * STATION_H,
            scene["n_notches"][index],
            scene["pitch"][index],
            scene["pivot"][index],
            scene["left_x"][index],
            scene["loads"][index],
        )
    return image
