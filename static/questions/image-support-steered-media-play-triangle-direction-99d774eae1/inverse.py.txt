"""Independent inverse arm using final RGB pixels only.

The oracle finds the sole saturated area and fits the left-to-right height
profile of its filled silhouette.  A triangle pointing right is tall on the
left and tapers toward the right; a left-pointing triangle has the opposite
slope.  Strict shape checks turn destructive erasures into abstentions.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _measure(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleAbstention("expected an RGB raster")

    spread = array.max(axis=2) - array.min(axis=2)
    mask = (spread > 48) & (array.max(axis=2) > 105)
    ys, xs = np.nonzero(mask)
    if len(xs) < 6500:
        raise OracleAbstention("large colored region is missing")

    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if not (120 <= width <= 280 and 110 <= height <= 250):
        raise OracleAbstention("colored region has implausible extent")
    fill = len(xs) / float(width * height)
    if not 0.43 <= fill <= 0.57:
        raise OracleAbstention("colored region is not a filled triangle")

    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    # A clean filled triangle intersects every row and column in one interval.
    row_counts = sub.sum(axis=1)
    col_counts = sub.sum(axis=0)
    row_span = np.zeros_like(row_counts)
    for index in np.flatnonzero(row_counts):
        coords = np.flatnonzero(sub[index])
        row_span[index] = coords[-1] - coords[0] + 1
    col_span = np.zeros_like(col_counts)
    for index in np.flatnonzero(col_counts):
        coords = np.flatnonzero(sub[:, index])
        col_span[index] = coords[-1] - coords[0] + 1
    missing_inside = int((row_span - row_counts).sum() + (col_span - col_counts).sum())
    if missing_inside > max(30, int(0.006 * len(xs))):
        raise OracleAbstention("colored silhouette contains an erasure or gap")

    heights_all = col_counts.astype(np.float64)
    # Ignore sub-five-percent antialias fringe columns.  Depending on which
    # edge lands on a subpixel, Pillow can emit one isolated outer rim column;
    # it is not part of the broad silhouette profile.
    valid = heights_all >= max(3.0, 0.05 * float(heights_all.max()))
    x = np.arange(width, dtype=np.float64)[valid]
    heights = heights_all[valid]
    slope, intercept = np.polyfit(x, heights, 1)
    predicted = slope * x + intercept
    rmse = float(np.sqrt(np.mean((heights - predicted) ** 2)))
    corr = float(np.corrcoef(x, heights)[0, 1])
    if not np.isfinite(corr) or abs(corr) < 0.985:
        raise OracleAbstention("silhouette lacks a decisive monotone taper")
    if rmse / max(float(heights.max()), 1.0) > 0.045:
        raise OracleAbstention("triangle taper fit is ambiguous")

    # Runner-up separation in pixel units: centroid displacement from box center.
    centroid_x = float(xs.mean())
    box_mid_x = (x0 + x1) / 2.0
    margin_px = abs(box_mid_x - centroid_x)
    if margin_px < 18.0:
        raise OracleAbstention("left/right occupancy margin is too small")
    direction = "right" if slope < 0.0 else "left"
    return direction, margin_px


def decision_from_image(image: Image.Image) -> str:
    return _measure(image)[0]
