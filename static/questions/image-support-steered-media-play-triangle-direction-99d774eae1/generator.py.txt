"""Deterministic renderer and analytic arm for the media play-direction world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

PALETTE = (
    (18, 125, 190),
    (0, 150, 136),
    (194, 58, 98),
    (111, 70, 180),
    (220, 100, 20),
    (40, 155, 80),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a matched control/counterfactual scene for an integer seed.

    Seeds 2k and 2k+1 share all nuisance values.  Only the direction differs,
    making color, position, scale, chrome, and colored-ink area exactly matched.
    """

    pair_seed = int(seed) // 2
    rng = random.Random(0x5A17_2026 ^ pair_seed)
    return {
        "direction": "right" if int(seed) % 2 == 0 else "left",
        "color": list(PALETTE[pair_seed % len(PALETTE)]),
        "center_x": rng.randint(246, 266),
        "center_y": rng.randint(218, 238),
        "triangle_width": rng.randrange(204, 229, 4),
        "triangle_height": rng.randrange(176, 201, 4),
        "panel_tone": rng.randint(229, 239),
        "control_variant": pair_seed % 3,
        "distractor_shift": rng.randint(-10, 10),
    }


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(round(value * SUPERSAMPLE) for value in box)  # type: ignore[return-value]


def _scaled_points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SUPERSAMPLE), round(y * SUPERSAMPLE)) for x, y in points]


def _triangle_points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    width = float(scene["triangle_width"])
    height = float(scene["triangle_height"])
    top = cy - height / 2.0
    bottom = cy + height / 2.0
    if scene["direction"] == "right":
        return [(cx - width / 2.0, top), (cx - width / 2.0, bottom), (cx + width / 2.0, cy)]
    return [(cx + width / 2.0, top), (cx + width / 2.0, bottom), (cx - width / 2.0, cy)]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a familiar media-player panel with one saturated answer region."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 249, 247))
    draw = ImageDraw.Draw(image)
    tone = int(scene["panel_tone"])
    panel_fill = (tone, tone + 1, tone + 2)
    draw.rounded_rectangle(
        _scaled_box((52, 58, 460, 452)),
        radius=30 * scale,
        fill=(205, 207, 210),
    )
    draw.rounded_rectangle(
        _scaled_box((58, 52, 454, 446)),
        radius=28 * scale,
        fill=panel_fill,
        outline=(102, 107, 113),
        width=3 * scale,
    )

    # Screen bezel and quiet horizontal highlights establish a recognizable
    # player while remaining achromatic and therefore invisible to the oracle.
    draw.rounded_rectangle(
        _scaled_box((82, 86, 430, 354)),
        radius=20 * scale,
        fill=(247, 247, 246),
        outline=(154, 158, 163),
        width=3 * scale,
    )
    shift = int(scene["distractor_shift"])
    draw.rounded_rectangle(
        _scaled_box((104 + shift, 384, 408 + shift, 396)),
        radius=6 * scale,
        fill=(185, 188, 192),
    )
    draw.rounded_rectangle(
        _scaled_box((104 + shift, 384, 214 + shift, 396)),
        radius=6 * scale,
        fill=(112, 116, 121),
    )
    draw.ellipse(_scaled_box((206 + shift, 378, 224 + shift, 402)), fill=(92, 97, 103))

    variant = int(scene["control_variant"])
    if variant == 0:
        draw.rectangle(_scaled_box((244, 409, 251, 428)), fill=(126, 130, 135))
        draw.rectangle(_scaled_box((259, 409, 266, 428)), fill=(126, 130, 135))
    elif variant == 1:
        draw.ellipse(_scaled_box((242, 411, 252, 421)), fill=(126, 130, 135))
        draw.ellipse(_scaled_box((258, 411, 268, 421)), fill=(126, 130, 135))
    else:
        draw.polygon(_scaled_points([(244, 410), (244, 426), (254, 418)]), fill=(126, 130, 135))
        draw.polygon(_scaled_points([(260, 410), (260, 426), (270, 418)]), fill=(126, 130, 135))

    points = _triangle_points(scene)
    # An achromatic offset shadow is visual chrome, not an answer-specific cue.
    draw.polygon(_scaled_points([(x + 5, y + 7) for x, y in points]), fill=(186, 188, 190))
    color = tuple(int(value) for value in scene["color"])
    draw.polygon(_scaled_points(points), fill=color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    direction = str(scene["direction"])
    if direction not in {"left", "right"}:
        raise ValueError("direction must be left or right")
    return direction


def margin(scene: dict[str, Any]) -> float:
    """Ideal centroid-to-box-midpoint separation (the runner-up gap), in px."""

    return float(scene["triangle_width"]) / 6.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return (
        float(scene["triangle_width"]) < 108.0
        or float(scene["triangle_height"]) < 108.0
        or margin(scene) < 18.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
