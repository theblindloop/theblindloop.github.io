"""Sampler, simulator, and rasterizer for a remote-instruction cursor walk."""

from __future__ import annotations

import copy
import random
from functools import lru_cache

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1120
HEIGHT = 820
GRID_N = 7
GRID_X = 48
GRID_Y = 112
CELL = 70
GAP = 7
PITCH = CELL + GAP
STEPS = 20
PAIR_COUNT = 10

BG = (246, 245, 240)
PAPER = (255, 255, 255)
INK = (34, 42, 54)
MUTED = (101, 111, 122)
DARK = (45, 57, 75)
LIGHT = (226, 232, 228)
START = (24, 151, 96)
GOALS = ((35, 112, 180), (224, 132, 31), (155, 80, 166), (207, 63, 67))
PAIR_COLORS = (
    (0, 126, 167), (230, 134, 28), (126, 87, 194), (206, 70, 65), (22, 143, 82),
    (53, 95, 184), (197, 75, 140), (116, 108, 56), (0, 151, 144), (177, 75, 0),
)
DIRECTIONS = ((-1, 0), (0, 1), (1, 0), (0, -1))


@lru_cache(maxsize=None)
def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _pair_maps(pairs: list) -> tuple[dict[tuple[int, int], tuple[int, int]], dict[tuple[int, int], int]]:
    twins: dict[tuple[int, int], tuple[int, int]] = {}
    labels: dict[tuple[int, int], int] = {}
    if len(pairs) != PAIR_COUNT:
        raise ValueError("wrong pair count")
    for index, pair in enumerate(pairs):
        if len(pair) != 2:
            raise ValueError("each letter must mark two cells")
        first, second = (tuple(map(int, cell)) for cell in pair)
        if first == second or first in twins or second in twins:
            raise ValueError("pair cells must be distinct")
        if any(not 0 <= value < GRID_N for cell in (first, second) for value in cell):
            raise ValueError("pair cell outside board")
        twins[first], twins[second] = second, first
        labels[first] = labels[second] = index
    return twins, labels


def _walk(
    tiles: list[list[int]],
    pairs: list,
    start: list[int] | tuple[int, int],
    heading: int,
    steps: int,
    *,
    read_mode: str = "twin",
    write_mode: str = "twin",
) -> tuple[tuple[int, int], list[list[int]], list[tuple[int, int]], dict]:
    state = [list(map(int, row)) for row in tiles]
    if len(state) != GRID_N or any(len(row) != GRID_N for row in state):
        raise ValueError("invalid grid")
    twins, labels = _pair_maps(pairs)
    row, col = map(int, start)
    direction = int(heading)
    trace = [(row, col)]
    remote_reads = 0
    divergent_remote_reads = 0
    mutated_reads: list[tuple[int, int]] = []
    initial = [line[:] for line in state]
    for _ in range(int(steps)):
        if not (0 <= row < GRID_N and 0 <= col < GRID_N):
            raise ValueError("cursor left board")
        here = (row, col)
        target = twins.get(here, here)
        if read_mode == "twin":
            read_at = target
        elif read_mode == "self":
            read_at = here
        else:
            raise ValueError("unknown read mode")
        face = state[read_at[0]][read_at[1]]
        if face not in (0, 1):
            raise ValueError("invalid face")
        if target != here:
            remote_reads += 1
            divergent_remote_reads += int(state[target[0]][target[1]] != state[row][col])
            if state[target[0]][target[1]] != initial[target[0]][target[1]]:
                mutated_reads.append(target)
        direction = (direction + (-1 if face else 1)) % 4
        if write_mode == "twin":
            state[target[0]][target[1]] ^= 1
        elif write_mode == "self":
            state[row][col] ^= 1
        elif write_mode == "none":
            pass
        else:
            raise ValueError("unknown write mode")
        dr, dc = DIRECTIONS[direction]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID_N and 0 <= col < GRID_N):
            raise ValueError("cursor left board")
        trace.append((row, col))
    diagnostics = {
        "remote_reads": remote_reads,
        "divergent_remote_reads": divergent_remote_reads,
        "mutated_remote_reads": len(mutated_reads),
        "mutated_remote_cells": len(set(mutated_reads)),
    }
    return (row, col), state, trace, diagnostics


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    cells = [(row, col) for row in range(GRID_N) for col in range(GRID_N)]
    for _ in range(60000):
        tiles = [[rng.randrange(2) for _ in range(GRID_N)] for _ in range(GRID_N)]
        if not 18 <= sum(map(sum, tiles)) <= 31:
            continue
        selected = rng.sample(cells, 2 * PAIR_COUNT)
        rng.shuffle(selected)
        pairs = [[list(selected[2 * index]), list(selected[2 * index + 1])] for index in range(PAIR_COUNT)]
        start = [rng.randrange(2, 5), rng.randrange(2, 5)]
        heading = rng.randrange(4)
        try:
            final, _, trace, diagnostics = _walk(tiles, pairs, start, heading, STEPS)
            parent_read, _, _, _ = _walk(tiles, pairs, start, heading, STEPS, read_mode="self")
            wrong_write, _, _, _ = _walk(tiles, pairs, start, heading, STEPS, write_mode="self")
        except ValueError:
            continue
        one_short = trace[-2]
        endpoints = {final, parent_read, wrong_write, one_short}
        if len(endpoints) != 4 or tuple(start) in endpoints:
            continue
        revisits = len(trace) - len(set(trace))
        if revisits < 3 or len(set(trace)) < 10:
            continue
        if (diagnostics["remote_reads"] < 8 or diagnostics["divergent_remote_reads"] < 4
                or diagnostics["mutated_remote_reads"] < 2 or diagnostics["mutated_remote_cells"] < 1):
            continue
        distractors = [parent_read, wrong_write, one_short]
        rng.shuffle(distractors)
        goals: list[list[int] | None] = [None] * 4
        goals[answer_index] = list(final)
        for index, endpoint in zip((i for i in range(4) if i != answer_index), distractors):
            goals[index] = list(endpoint)
        draw_order = list(range(GRID_N * GRID_N))
        rng.shuffle(draw_order)
        return {
            "tiles": tiles,
            "pairs": pairs,
            "start": start,
            "heading": heading,
            "steps": STEPS,
            "goal_positions": goals,
            "cell_draw_order": draw_order,
        }
    raise RuntimeError(f"could not sample scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    final, _, _, _ = _walk(scene["tiles"], scene["pairs"], scene["start"], scene["heading"], scene["steps"])
    goals = [tuple(map(int, position)) for position in scene["goal_positions"]]
    if len(goals) != 4 or len(set(goals)) != 4:
        raise ValueError("goals must be unique")
    matches = [index for index, position in enumerate(goals) if position == final]
    if len(matches) != 1:
        raise ValueError("final cursor must occupy one goal")
    return f"goal-{'ABCD'[matches[0]]}"


def margin(scene: dict) -> float:
    analytic_gold(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    analytic_gold(scene)
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cell_draw_order"] = list(reversed(scene["cell_draw_order"]))
    return [("reverse_disjoint_cell_draw_order", twin)]


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _cell_box(row: int, col: int) -> tuple[int, int, int, int]:
    x = GRID_X + col * PITCH
    y = GRID_Y + row * PITCH
    return x, y, x + CELL, y + CELL


def _turn_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], right: bool, color=INK) -> None:
    x, y = center
    if right:
        draw.arc((x - 19, y - 18, x + 19, y + 20), 205, 30, fill=color, width=5)
        draw.polygon(((x + 18, y - 1), (x + 7, y - 4), (x + 14, y + 8)), fill=color)
    else:
        draw.arc((x - 19, y - 18, x + 19, y + 20), 150, 335, fill=color, width=5)
        draw.polygon(((x - 18, y - 1), (x - 7, y - 4), (x - 14, y + 8)), fill=color)


def _start_arrow(draw: ImageDraw.ImageDraw, row: int, col: int, heading: int) -> None:
    left, top, right, bottom = _cell_box(row, col)
    cx, cy = (left + right) // 2, (top + bottom) // 2
    dr, dc = DIRECTIONS[int(heading)]
    pr, pc = dc, -dr
    draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), outline=START, width=6)
    draw.line((cx - 15 * dc, cy - 15 * dr, cx + 3 * dc, cy + 3 * dr), fill=START, width=8)
    tip = (cx + 20 * dc, cy + 20 * dr)
    base = (cx, cy)
    draw.polygon((tip, (base[0] + 10 * pc, base[1] + 10 * pr), (base[0] - 10 * pc, base[1] - 10 * pr)), fill=START)


def _goal_badge(draw: ImageDraw.ImageDraw, row: int, col: int, index: int) -> None:
    _, _, right, bottom = _cell_box(row, col)
    cx, cy = right - 17, bottom - 17
    draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=GOALS[index], outline=PAPER, width=3)
    _centered(draw, (cx, cy), "ABCD"[index], _font(17, True), PAPER)


def render(scene: dict) -> Image.Image:
    analytic_gold(scene)
    _, labels = _pair_maps(scene["pairs"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((48, 22), "TWIN-READ RELAY", font=_font(31, True), fill=INK)
    draw.text((48, 61), f"{scene['steps']} MOVES  •  a letter points to its twin", font=_font(19, True), fill=MUTED)

    for flat in scene["cell_draw_order"]:
        row, col = divmod(int(flat), GRID_N)
        box = _cell_box(row, col)
        fill = DARK if int(scene["tiles"][row][col]) else LIGHT
        draw.rounded_rectangle(box, radius=10, fill=fill, outline=INK, width=3)
        index = labels.get((row, col))
        if index is not None:
            left, top, _, _ = box
            draw.rounded_rectangle((left + 6, top + 7, left + 39, top + 35), radius=6, fill=PAIR_COLORS[index], outline=PAPER, width=2)
            _centered(draw, (left + 22.5, top + 21), chr(65 + index), _font(18, True), PAPER)

    for index, position in enumerate(scene["goal_positions"]):
        _goal_badge(draw, int(position[0]), int(position[1]), index)
    _start_arrow(draw, int(scene["start"][0]), int(scene["start"][1]), int(scene["heading"]))

    draw.rounded_rectangle((622, 92, 1074, 714), radius=20, fill=PAPER, outline=INK, width=4)
    _centered(draw, (848, 128), "CHOOSE SHADE TO READ", _font(22, True), INK)
    draw.line((661, 153, 1035, 153), fill=LIGHT, width=4)
    draw.rounded_rectangle((663, 174, 1033, 258), radius=13, fill=LIGHT, outline=MUTED, width=3)
    draw.text((682, 190), "LIGHT", font=_font(22, True), fill=INK)
    draw.text((790, 190), "TURN RIGHT", font=_font(21, True), fill=INK)
    _turn_icon(draw, (991, 216), True)
    draw.rounded_rectangle((663, 274, 1033, 358), radius=13, fill=DARK, outline=MUTED, width=3)
    draw.text((682, 290), "DARK", font=_font(22, True), fill=PAPER)
    draw.text((790, 290), "TURN LEFT", font=_font(21, True), fill=PAPER)
    _turn_icon(draw, (991, 316), False, PAPER)

    draw.line((661, 382, 1035, 382), fill=LIGHT, width=4)
    _centered(draw, (848, 414), "READ AND FLIP", _font(22, True), MUTED)
    draw.rounded_rectangle((680, 440, 1016, 491), radius=10, fill=(238, 242, 247), outline=MUTED, width=2)
    _centered(draw, (848, 466), "LETTER  →  OTHER SAME LETTER", _font(18, True), INK)
    draw.rounded_rectangle((680, 504, 1016, 555), radius=10, fill=(238, 242, 247), outline=MUTED, width=2)
    _centered(draw, (848, 530), "BLANK  →  THIS SQUARE", _font(18, True), INK)
    _centered(draw, (848, 580), "READ  →  TURN  →  FLIP  →  MOVE", _font(18, True), INK)

    for index, color in enumerate(GOALS):
        cx = 704 + index * 96
        cy = 635
        draw.ellipse((cx - 22, cy - 22, cx + 22, cy + 22), fill=color, outline=INK, width=3)
        _centered(draw, (cx, cy), "ABCD"[index], _font(21, True), PAPER)
        _centered(draw, (cx, 677), f"goal-{'ABCD'[index]}", _font(14, True), INK)
    return image
