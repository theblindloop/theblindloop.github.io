"""Deterministic sampler, renderer, and analytic arm.

The analytic decision is deliberately derived from ``target_cells`` by the
same public four-neighbor rule stated in the question.  No class tag exists in
the scene.  Rendering uses only ordinary Pillow primitives.
"""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

DECISIONS = ("one_island", "ring", "two_islands")
INKS = ("#183B66", "#49306B", "#174F45", "#5A2C32")
BACKGROUNDS = ("#FCFAF4", "#F7FAFC", "#FAF7FC", "#F8FBF7")


def _normalize(cells: set[tuple[int, int]]) -> set[tuple[int, int]]:
    r0 = min(r for r, _ in cells)
    c0 = min(c for _, c in cells)
    return {(r - r0, c - c0) for r, c in cells}


def _transform(cells: set[tuple[int, int]], turns: int, mirror: bool) -> set[tuple[int, int]]:
    cells = _normalize(cells)
    for _ in range(turns % 4):
        height = max(r for r, _ in cells) + 1
        cells = {(c, height - 1 - r) for r, c in cells}
        cells = _normalize(cells)
    if mirror:
        width = max(c for _, c in cells) + 1
        cells = {(r, width - 1 - c) for r, c in cells}
    return _normalize(cells)


def _base_mask(kind: str, rng: random.Random) -> set[tuple[int, int]]:
    if kind == "one_island":
        h, w = rng.choice(((4, 6), (3, 8), (2, 12)))
        return {(r, c) for r in range(h) for c in range(w)}
    if kind == "ring":
        h, w = rng.choice(((6, 8), (5, 9), (4, 10)))
        return {
            (r, c)
            for r in range(h)
            for c in range(w)
            if r in (0, h - 1) or c in (0, w - 1)
        }
    if kind == "two_islands":
        offset = rng.randrange(0, 4)
        first = {(r, c) for r in range(3) for c in range(4)}
        second = {(r + offset, c + 8) for r in range(3) for c in range(4)}
        return first | second
    raise ValueError(kind)


def _topology(target_cells: list[list[int]]) -> tuple[int, int]:
    cells = {tuple(map(int, cell)) for cell in target_cells}
    if not cells:
        return 0, 0
    components = 0
    unseen = set(cells)
    while unseen:
        components += 1
        queue = [unseen.pop()]
        while queue:
            r, c = queue.pop()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in unseen:
                    unseen.remove(nxt)
                    queue.append(nxt)

    min_r, max_r = min(r for r, _ in cells) - 1, max(r for r, _ in cells) + 1
    min_c, max_c = min(c for _, c in cells) - 1, max(c for _, c in cells) + 1
    exterior = {(min_r, min_c)}
    queue = deque(exterior)
    while queue:
        r, c = queue.popleft()
        for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            nr, nc = nxt
            if (
                min_r <= nr <= max_r
                and min_c <= nc <= max_c
                and nxt not in cells
                and nxt not in exterior
            ):
                exterior.add(nxt)
                queue.append(nxt)
    holes = 0
    remaining = {
        (r, c)
        for r in range(min_r, max_r + 1)
        for c in range(min_c, max_c + 1)
        if (r, c) not in cells and (r, c) not in exterior
    }
    while remaining:
        holes += 1
        queue2 = [remaining.pop()]
        while queue2:
            r, c = queue2.pop()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in remaining:
                    remaining.remove(nxt)
                    queue2.append(nxt)
    return components, holes


def analytic_gold(scene: dict) -> str:
    components, holes = _topology(scene["target_cells"])
    if components == 1 and holes == 0:
        return "one_island"
    if components == 1 and holes == 1:
        return "ring"
    if components == 2 and holes == 0:
        return "two_islands"
    raise ValueError(f"unsupported target topology: components={components}, holes={holes}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    kind = DECISIONS[int(seed) % len(DECISIONS)]
    grid_size = 15
    cells = _transform(_base_mask(kind, rng), rng.randrange(4), bool(rng.randrange(2)))
    height = max(r for r, _ in cells) + 1
    width = max(c for _, c in cells) + 1
    row_shift = rng.randrange(1, grid_size - height)
    col_shift = rng.randrange(1, grid_size - width)
    cells = {(r + row_shift, c + col_shift) for r, c in cells}
    target_cells = [[r, c] for r, c in sorted(cells)]

    spacing = rng.choice((24, 25, 26))
    dash_length = rng.choice((15, 16, 17, 18))
    dash_width = rng.choice((3, 4))
    canvas_size = 440
    span = spacing * (grid_size - 1)
    slack = canvas_size - span
    origin = [slack / 2 + rng.uniform(-3.0, 3.0), slack / 2 + rng.uniform(-3.0, 3.0)]
    jitter = [
        [round(rng.uniform(-1.25, 1.25), 3), round(rng.uniform(-1.25, 1.25), 3)]
        for _ in range(grid_size * grid_size)
    ]
    scene = {
        "target_cells": target_cells,
        "grid_size": grid_size,
        "spacing": spacing,
        "origin": origin,
        "jitter": jitter,
        "dash_length": dash_length,
        "dash_width": dash_width,
        "target_color": rng.choice(INKS),
        "background_color": rng.choice(BACKGROUNDS),
        "canvas_size": canvas_size,
    }
    assert len(target_cells) == 24
    assert analytic_gold(scene) == kind
    return scene


def render(scene: dict) -> Image.Image:
    target_cells = {tuple(map(int, cell)) for cell in scene["target_cells"]}
    grid_size = int(scene["grid_size"])
    spacing = float(scene["spacing"])
    ox, oy = map(float, scene["origin"])
    jitter = scene["jitter"]
    dash_length = float(scene["dash_length"])
    dash_width = int(scene["dash_width"])
    ink = scene["target_color"]
    background = scene["background_color"]
    canvas_size = int(scene["canvas_size"])

    scale = 4
    image = Image.new("RGB", (canvas_size * scale, canvas_size * scale), background)
    draw = ImageDraw.Draw(image)
    half = dash_length / (2.0 * math.sqrt(2.0))
    for row in range(grid_size):
        for col in range(grid_size):
            dx, dy = map(float, jitter[row * grid_size + col])
            cx = ox + col * spacing + dx
            cy = oy + row * spacing + dy
            if (row, col) in target_cells:  # forward slash: negative image-coordinate slope
                endpoints = ((cx - half, cy + half), (cx + half, cy - half))
            else:
                endpoints = ((cx - half, cy - half), (cx + half, cy + half))
            draw.line(
                tuple((round(x * scale), round(y * scale)) for x, y in endpoints),
                fill=ink,
                width=dash_width * scale,
            )
    return image.resize((canvas_size, canvas_size), Image.Resampling.LANCZOS)


def margin(scene: dict) -> float:
    # Pixel clearance between neighboring dash bounding boxes; the competing
    # orientation modes remain 90 degrees apart throughout this range.
    return float(scene["spacing"]) - float(scene["dash_length"])


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
