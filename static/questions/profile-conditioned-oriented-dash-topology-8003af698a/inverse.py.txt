"""Independent PNG-only inverse arm for the oriented-dash topology task."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        todo = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while todo:
            yy, xx = todo.pop()
            points.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        if len(points) >= 12:
            result.append(np.asarray(points, dtype=np.float64))
    return result


def _clusters(values: list[float]) -> list[float]:
    ordered = sorted(values)
    if len(ordered) < 4:
        raise ValueError("too few marks")
    gaps = np.diff(ordered)
    large = gaps[gaps > 4.0]
    if not len(large):
        raise ValueError("lattice axes are not separable")
    threshold = max(4.0, float(np.median(large)) * 0.45)
    groups: list[list[float]] = [[ordered[0]]]
    for previous, value in zip(ordered, ordered[1:]):
        if value - previous > threshold:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [float(np.mean(group)) for group in groups]


def _topology(matrix: np.ndarray) -> tuple[int, int]:
    targets = {tuple(map(int, pos)) for pos in np.argwhere(matrix)}
    unseen = set(targets)
    components = 0
    while unseen:
        components += 1
        todo = [unseen.pop()]
        while todo:
            r, c = todo.pop()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in unseen:
                    unseen.remove(nxt)
                    todo.append(nxt)

    rows, cols = matrix.shape
    exterior = set()
    todo = deque()
    for r in range(rows):
        for c in (0, cols - 1):
            if not matrix[r, c] and (r, c) not in exterior:
                exterior.add((r, c)); todo.append((r, c))
    for c in range(cols):
        for r in (0, rows - 1):
            if not matrix[r, c] and (r, c) not in exterior:
                exterior.add((r, c)); todo.append((r, c))
    while todo:
        r, c = todo.popleft()
        for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= nr < rows and 0 <= nc < cols and not matrix[nr, nc] and (nr, nc) not in exterior:
                exterior.add((nr, nc)); todo.append((nr, nc))
    enclosed = {
        (r, c)
        for r in range(rows)
        for c in range(cols)
        if not matrix[r, c] and (r, c) not in exterior
    }
    holes = 0
    while enclosed:
        holes += 1
        stack = [enclosed.pop()]
        while stack:
            r, c = stack.pop()
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in enclosed:
                    enclosed.remove(nxt); stack.append(nxt)
    return components, holes


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate((arr[0], arr[-1], arr[:, 0], arr[:, -1]), axis=0)
    background = np.median(border, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    marks = _components(distance > 28.0)
    if len(marks) < 100:
        raise ValueError(f"insufficient visible dash evidence: {len(marks)} marks")

    fitted = []
    for points_yx in marks:
        y = points_yx[:, 0]
        x = points_yx[:, 1]
        centered_x = x - x.mean()
        centered_y = y - y.mean()
        covariance = float(np.mean(centered_x * centered_y))
        spread = float(np.mean(centered_x**2 + centered_y**2))
        if spread < 3.0 or abs(covariance) / spread < 0.18:
            raise ValueError("ambiguous dash orientation")
        fitted.append((float(x.mean()), float(y.mean()), covariance < 0.0))

    xs = _clusters([item[0] for item in fitted])
    ys = _clusters([item[1] for item in fitted])
    if len(xs) != len(ys) or not 9 <= len(xs) <= 25:
        raise ValueError(f"invalid recovered lattice: {len(xs)} by {len(ys)}")
    matrix = np.zeros((len(ys), len(xs)), dtype=bool)
    occupied = set()
    for x, y, forward_slash in fitted:
        col = int(np.argmin(np.abs(np.asarray(xs) - x)))
        row = int(np.argmin(np.abs(np.asarray(ys) - y)))
        if (row, col) in occupied:
            raise ValueError("multiple marks mapped to one cell")
        occupied.add((row, col))
        matrix[row, col] = forward_slash
    if len(occupied) != len(xs) * len(ys):
        raise ValueError("incomplete lattice")
    if int(matrix.sum()) < 8:
        raise ValueError("target orientation was erased")

    components, holes = _topology(matrix)
    if components == 1 and holes == 0:
        return "one_island"
    if components == 1 and holes == 1:
        return "ring"
    if components == 2 and holes == 0:
        return "two_islands"
    raise ValueError(f"ambiguous topology: components={components}, holes={holes}")
