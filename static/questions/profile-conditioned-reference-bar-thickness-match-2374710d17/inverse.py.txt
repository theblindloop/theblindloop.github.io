"""Independent pixel-only inverse arm for reference_bar_thickness_match.

Receives only the final RGB image. It separates the reference box at the top
from the four quadrant cells below, thresholds the dark bar ink, and measures
the thickness (vertical extent in pixels) of every dark bar. The reference bar
thickness is measured in the top box; each quadrant's bar thickness is measured
within its cell. The decision is the quadrant whose measured bar thickness
equals the reference bar thickness.

The geometry below (box at top, four quadrants below, grid split) is part of the
public question contract. Imports nothing from renderer/prompts/generate/verify.
"""

from __future__ import annotations

import numpy as np

SIZE = 480
DARK_SUM = 300  # ink (25,25,25) sums to 75; grey divider (200,200,200) to 600

# Reference box region.
REF_X0, REF_X1 = 60, 420
REF_Y0, REF_Y1 = 20, 130

# Grid geometry.
MID_X = 240
GRID_Y0 = 150
MID_Y = 310

SECTORS = ("TL", "TR", "BL", "BR")
_GRID_BOXES = {
    "TL": (0, GRID_Y0, MID_X, MID_Y),
    "TR": (MID_X, GRID_Y0, SIZE, MID_Y),
    "BL": (0, MID_Y, MID_X, SIZE),
    "BR": (MID_X, MID_Y, SIZE, SIZE),
}


def _bar_thickness(dark: np.ndarray, x0: int, y0: int, x1: int, y1: int) -> int:
    cell = dark[y0:y1, x0:x1]
    rows = np.where(cell.any(axis=1))[0]  # y offsets of rows containing dark ink
    if rows.size == 0:
        return 0
    return int(rows.max() - rows.min() + 1)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < DARK_SUM
    ref_thick = _bar_thickness(dark, REF_X0, REF_Y0, REF_X1, REF_Y1)
    best = None
    best_diff = None
    for sector in SECTORS:
        x0, y0, x1, y1 = _GRID_BOXES[sector]
        t = _bar_thickness(dark, x0, y0, x1, y1)
        diff = abs(t - ref_thick)
        if best_diff is None or diff < best_diff:
            best_diff = diff
            best = sector
    return best if best is not None else SECTORS[0]
