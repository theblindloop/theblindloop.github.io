"""Reference stroke-thickness search world.

A dark reference bar is drawn inside the grey box at the top of the image. The
thickness (stroke width) of a bar is how tall it is: the vertical extent in
pixels between its top and bottom edges. Below the box the image is split into
four quadrants (top-left, top-right, bottom-left, bottom-right), each holding
one dark horizontal bar. Exactly one quadrant's bar has the SAME thickness as
the reference bar; the other three bars each carry a clearly different
thickness. The decision is the quadrant whose bar thickness equals the
reference thickness.

Every attribute that decides the answer (the reference thickness and each
quadrant's bar thickness) is drawn in the raster; the bar widths and placement
positions are nuisance fields that do not carry the answer.
"""

from __future__ import annotations

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
INK = (25, 25, 25)
DIV = (200, 200, 200)

# Reference box geometry (top of image).
REF_X0 = 60
REF_X1 = 420
REF_Y0 = 20
REF_Y1 = 130
REF_BAR_Y = 78          # top row of the reference bar
REF_BAR_X0 = 150
REF_BAR_X1 = 330

# Grid geometry below the reference box.
GRID_Y0 = 150
MID_Y = 310
MID_X = 240

SECTORS = ("TL", "TR", "BL", "BR")
_GRID_BOXES = {
    "TL": (0, GRID_Y0, MID_X, MID_Y),
    "TR": (MID_X, GRID_Y0, SIZE, MID_Y),
    "BL": (0, MID_Y, MID_X, SIZE),
    "BR": (MID_X, MID_Y, SIZE, SIZE),
}

# Discrete bar thicknesses in pixels (stroke heights). Chosen values are spaced
# by 4px so any two distinct thicknesses are visibly separated at pixel scale.
THICKNESSES = [6, 10, 14, 18, 22, 26, 30, 34]
REF_POOL = [10, 14, 18, 22, 26, 30]
WIDTH_MIN = 70
WIDTH_MAX = 170
PAD = 12

QUARANTINE_THRESH = 4  # thicknesses are spaced >=4px; only a <4px gap is fragile


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    ref_thick = int(REF_POOL[int(rng.integers(0, len(REF_POOL)))])
    distractor_pool = [t for t in THICKNESSES if t != ref_thick]
    others = [int(x) for x in rng.choice(distractor_pool, size=3, replace=False)]
    thicknesses = {}
    widths = {}
    positions = {}
    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    pool_iter = iter(others)
    for sector in SECTORS:
        if sector == winner:
            thicknesses[sector] = ref_thick
        else:
            thicknesses[sector] = next(pool_iter)
        widths[sector] = int(rng.integers(WIDTH_MIN, WIDTH_MAX + 1))
        x0, y0, x1, y1 = _GRID_BOXES[sector]
        bw = widths[sector]
        bh = thicknesses[sector]
        px = int(rng.integers(x0 + PAD, x1 - PAD - bw + 1))
        py = int(rng.integers(y0 + PAD, y1 - PAD - bh + 1))
        positions[sector] = (px, py)
    return {
        "ref_thick": ref_thick,
        "thicknesses": thicknesses,
        "widths": widths,
        "positions": positions,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    # Reference box outline (light grey, excluded from dark ink).
    draw.rectangle([REF_X0, REF_Y0, REF_X1, REF_Y1], outline=DIV, width=2)
    # Reference bar.
    ref_thick = scene["ref_thick"]
    draw.rectangle([REF_BAR_X0, REF_BAR_Y, REF_BAR_X1, REF_BAR_Y + ref_thick - 1], fill=INK)
    # Grid dividers (light grey).
    draw.line([0, GRID_Y0, SIZE, GRID_Y0], fill=DIV, width=2)
    draw.line([MID_X, GRID_Y0, MID_X, SIZE], fill=DIV, width=2)
    draw.line([0, MID_Y, SIZE, MID_Y], fill=DIV, width=2)
    # Quadrant bars.
    for sector in SECTORS:
        t = scene["thicknesses"][sector]
        px, py = scene["positions"][sector]
        w = scene["widths"][sector]
        draw.rectangle([px, py, px + w - 1, py + t - 1], fill=INK)
    return img


def analytic_gold(scene: dict) -> str:
    ref_thick = scene["ref_thick"]
    thicknesses = scene["thicknesses"]
    for sector in SECTORS:
        if thicknesses[sector] == ref_thick:
            return sector
    raise RuntimeError("no quadrant matches the reference thickness")


def margin(scene: dict) -> float:
    ref_thick = scene["ref_thick"]
    others = [scene["thicknesses"][s] for s in SECTORS if scene["thicknesses"][s] != ref_thick]
    if not others:
        return float(QUARANTINE_THRESH)
    return float(min(abs(t - ref_thick) for t in others))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
