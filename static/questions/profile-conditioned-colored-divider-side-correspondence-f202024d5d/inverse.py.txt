"""Independent pixel-only inverse for the divider-side world."""

from __future__ import annotations

import numpy as np
from PIL import Image


PANEL_ORIGINS = ((20, 30), (420, 30), (820, 30))
PANEL_WIDTH = 360
PANEL_HEIGHT = 360
PALETTE = np.asarray(
    (
        (218, 54, 58),
        (46, 101, 211),
        (42, 169, 91),
        (235, 143, 39),
        (145, 76, 190),
        (30, 163, 170),
    ),
    dtype=np.float32,
)


def _token_centers(array: np.ndarray, origin: tuple[int, int]) -> list[tuple[float, float]]:
    """Segment each known color and measure its centroid inside one panel."""

    ox, oy = origin
    crop = array[oy + 72 : oy + PANEL_HEIGHT - 8, ox + 12 : ox + PANEL_WIDTH - 12]
    centers: list[tuple[float, float]] = []
    for color in PALETTE:
        distances = np.linalg.norm(crop - color, axis=2)
        # The closest palette colors are blue/teal, so a broad radius would
        # merge their disks.  The renderer uses flat fills; a tight distance
        # band retains the visible interior of each component independently.
        mask = distances < 38.0
        ys, xs = np.where(mask)
        if len(xs) < 60:
            raise ValueError("colored token is missing from a panel")
        centers.append((float(xs.mean() + ox + 12), float(ys.mean() + oy + 72)))
    return centers


def _divider_x(array: np.ndarray, origin: tuple[int, int]) -> float:
    """Fit the visible vertical divider from its dark interior pixels."""

    ox, oy = origin
    crop = array[oy + 72 : oy + PANEL_HEIGHT - 8, ox + 12 : ox + PANEL_WIDTH - 12]
    dark = np.max(crop, axis=2) < 100.0
    ys, xs = np.where(dark)
    if len(xs) < 100:
        raise ValueError("divider line is not measurable")
    # The panel border and header are outside this crop; colored disks have a
    # channel above 100, leaving only the divider in the dark mask.
    return float(np.median(xs) + ox + 12)


def _side_signature(array: np.ndarray, origin: tuple[int, int]) -> tuple[int, ...]:
    centers = _token_centers(array, origin)
    divider = _divider_x(array, origin)
    return tuple(1 if x < divider else 0 for x, _ in centers)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only the final RGB pixels and public panel geometry."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.shape[0] < 390 or array.shape[1] < 1180:
        raise ValueError("unexpected canvas size")
    signatures = [_side_signature(array, origin) for origin in PANEL_ORIGINS]
    if signatures[1] == signatures[0]:
        return "A"
    if signatures[2] == signatures[0]:
        return "B"
    raise ValueError("neither candidate matches the reference signature")
