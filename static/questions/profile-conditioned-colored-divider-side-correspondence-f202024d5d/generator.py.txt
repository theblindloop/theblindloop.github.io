"""Renderer and analytic world for color-bound half-plane correspondence."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored-divider-side-correspondence-0.1.0"
PANEL_ORIGINS: tuple[tuple[int, int], ...] = ((20, 30), (420, 30), (820, 30))
PANEL_SIZE = (360, 360)
PANEL_LABELS = ("REFERENCE", "A", "B")
PALETTE: tuple[tuple[int, int, int], ...] = (
    (218, 54, 58),
    (46, 101, 211),
    (42, 169, 91),
    (235, 143, 39),
    (145, 76, 190),
    (30, 163, 170),
)
BACKGROUND = (248, 249, 245)
PANEL_BORDER = (215, 219, 213)
INK = (39, 43, 48)
TOKEN_MARGIN = 42.0
QUARANTINE_MARGIN = 12.0


def _all_fields(scene: dict[str, Any]) -> tuple[Any, ...]:
    """Read every declared scene field used by both renderer and label."""

    return (
        scene["reference_centers"],
        scene["candidate_a_centers"],
        scene["candidate_b_centers"],
        scene["divider_xs"],
        scene["token_radii"],
        scene["palette"],
        scene["canvas"],
        scene["background"],
        scene["stroke_width"],
    )


def _side_bits(centers: list[list[float]], divider_x: float) -> tuple[int, ...]:
    """Return 1 for left and 0 for right, using the token center."""

    return tuple(1 if float(center[0]) < float(divider_x) else 0 for center in centers)


def _signature(scene: dict[str, Any]) -> tuple[tuple[int, ...], ...]:
    reference, candidate_a, candidate_b, divider_xs, *_ = _all_fields(scene)
    return (
        _side_bits(reference, divider_xs[0]),
        _side_bits(candidate_a, divider_xs[1]),
        _side_bits(candidate_b, divider_xs[2]),
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, well-separated three-panel scene."""

    rng = random.Random(int(seed))
    reference_bits = [1, 1, 1, 0, 0, 0]
    rng.shuffle(reference_bits)
    wrong_bits = list(reference_bits)
    left = [index for index, bit in enumerate(wrong_bits) if bit == 1]
    right = [index for index, bit in enumerate(wrong_bits) if bit == 0]
    swap_left = rng.choice(left)
    swap_right = rng.choice(right)
    wrong_bits[swap_left], wrong_bits[swap_right] = (
        wrong_bits[swap_right],
        wrong_bits[swap_left],
    )

    # The panel identity is balanced by seed parity, independent of palette and
    # geometry. The other panel has the same 3/3 marginal count.
    matching_panel = 0 if int(seed) % 2 == 0 else 1
    panel_bits = [
        reference_bits,
        reference_bits if matching_panel == 0 else wrong_bits,
        reference_bits if matching_panel == 1 else wrong_bits,
    ]
    divider_xs = [
        float(origin[0] + 180.0 + rng.uniform(-28.0, 28.0))
        for origin in PANEL_ORIGINS
    ]
    centers: list[list[list[float]]] = []
    radii: list[list[float]] = []
    for panel_index, (origin, bits, divider_x) in enumerate(
        zip(PANEL_ORIGINS, panel_bits, divider_xs)
    ):
        panel_centers: list[list[float]] = []
        panel_radii: list[float] = []
        for bit in bits:
            for _ in range(20_000):
                x = rng.uniform(origin[0] + 30.0, origin[0] + 330.0)
                y = rng.uniform(origin[1] + 92.0, origin[1] + 338.0)
                if (x < divider_x) != bool(bit):
                    continue
                if abs(x - divider_x) < TOKEN_MARGIN:
                    continue
                if any(math.dist((x, y), center) < 47.0 for center in panel_centers):
                    continue
                panel_centers.append([round(x, 3), round(y, 3)])
                panel_radii.append(round(rng.uniform(15.0, 20.0), 3))
                break
            else:
                raise RuntimeError("could not place separated colored tokens")
        centers.append(panel_centers)
        radii.append(panel_radii)

    return {
        "reference_centers": centers[0],
        "candidate_a_centers": centers[1],
        "candidate_b_centers": centers[2],
        "divider_xs": divider_xs,
        "token_radii": radii,
        "palette": [list(color) for color in PALETTE],
        "canvas": [1220, 430],
        "background": list(BACKGROUND),
        "stroke_width": 5,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the three labeled panels as a single RGB raster."""

    reference, candidate_a, candidate_b, divider_xs, token_radii, palette, canvas, background, stroke_width = _all_fields(scene)
    image = Image.new("RGB", tuple(int(value) for value in canvas), tuple(background))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    panel_centers = (reference, candidate_a, candidate_b)

    for panel_index, (origin, label, token_centers, line_x) in enumerate(
        zip(PANEL_ORIGINS, PANEL_LABELS, panel_centers, divider_xs)
    ):
        ox, oy = origin
        panel_box = (ox, oy, ox + PANEL_SIZE[0], oy + PANEL_SIZE[1])
        draw.rounded_rectangle(panel_box, radius=12, fill=tuple(background), outline=PANEL_BORDER, width=2)
        draw.text((ox + PANEL_SIZE[0] / 2, oy + 28), label, fill=INK, anchor="mm", font=font)
        draw.line(
            (float(line_x), oy + 70, float(line_x), oy + PANEL_SIZE[1] - 10),
            fill=INK,
            width=max(3, int(stroke_width)),
        )
        for token_index, center in enumerate(token_centers):
            x, y = float(center[0]), float(center[1])
            radius = float(token_radii[panel_index][token_index])
            color = tuple(int(value) for value in palette[token_index])
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the candidate with the identical color-to-side signature."""

    reference, candidate_a, candidate_b, divider_xs, token_radii, palette, canvas, background, stroke_width = _all_fields(scene)
    _ = (token_radii, palette, canvas, background, stroke_width)
    reference_signature = _side_bits(reference, divider_xs[0])
    if _side_bits(candidate_a, divider_xs[1]) == reference_signature:
        return "A"
    if _side_bits(candidate_b, divider_xs[2]) == reference_signature:
        return "B"
    raise ValueError("scene has no matching candidate")


def margin(scene: dict[str, Any]) -> float:
    """Smallest center-to-divider distance over all visible tokens."""

    reference, candidate_a, candidate_b, divider_xs, *_ = _all_fields(scene)
    distances = []
    for centers, divider_x in zip((reference, candidate_a, candidate_b), divider_xs):
        distances.extend(abs(float(center[0]) - float(divider_x)) for center in centers)
    return float(min(distances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No unobserved render-invariant latent transform is declared."""

    return []
