from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BG = np.array((248, 246, 240), dtype=np.int16)
NAVY = np.array((30, 48, 75), dtype=np.int16)
GOLD = np.array((230, 174, 45), dtype=np.int16)
PALETTE = {
    "coral": np.array((226, 91, 84), dtype=np.int16),
    "green": np.array((67, 158, 105), dtype=np.int16),
    "blue": np.array((58, 128, 189), dtype=np.int16),
    "purple": np.array((145, 94, 174), dtype=np.int16),
}


def _mask_near(arr: np.ndarray, rgb: np.ndarray, tol: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - rgb), axis=2) <= tol


def _components(mask: np.ndarray, minimum: int) -> list[tuple[np.ndarray, np.ndarray]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        q = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        ys, xs = [], []
        while q:
            y, x = q.popleft()
            ys.append(y); xs.append(x)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; q.append((ny,nx))
        if len(ys) >= minimum:
            output.append((np.asarray(ys), np.asarray(xs)))
    return output


def _shape_at(navy: np.ndarray, cx: int, cy: int) -> str:
    # Measure horizontal silhouette widths well inside the halo; stems are hidden
    # beneath the halo and do not enter these rows.
    def width(dy: int) -> int:
        row = navy[cy + dy, max(0, cx - 15):cx + 16]
        return int(row.sum())
    top, mid, bottom = width(-8), width(0), width(8)
    if top >= 19 and bottom >= 19:
        return "square"
    if bottom - top >= 8:
        return "triangle"
    if mid >= 23 and top <= 11 and bottom <= 11:
        return "diamond"
    if 11 <= top <= 19 and 11 <= bottom <= 19 and mid >= 21:
        return "circle"
    raise ValueError(f"ambiguous station silhouette {top, mid, bottom}")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape != (520, 720, 3):
        raise ValueError("unexpected raster size")
    navy = _mask_near(arr, NAVY)
    navy_parts = _components(navy, 25)
    # The rail and stems form one dominant component; eight shape silhouettes
    # are deliberately detached within their halos. Erasing an interior rail
    # region splits or shrinks the dominant component and is therefore visible.
    sizes = sorted((len(part[0]) for part in navy_parts), reverse=True)
    if len(navy_parts) != 9 or not sizes or sizes[0] < 5500:
        raise ValueError("navy rail is broken or incomplete")
    if len(_components(_mask_near(arr, GOLD), 200)) != 1:
        raise ValueError("gold start bead missing")

    stations = []
    for name, rgb in PALETTE.items():
        parts = _components(_mask_near(arr, rgb), 250)
        if len(parts) != 2:
            raise ValueError("each halo color must occur at two stations")
        for ys, xs in parts:
            cx, cy = int(round(float(xs.mean()))), int(round(float(ys.mean())))
            stations.append((cy, cx, _shape_at(navy, cx, cy), name))
    if len(stations) != 8:
        raise ValueError("expected eight stations")
    gold_y, gold_x = (lambda p: (float(p[0].mean()), float(p[1].mean())))(_components(_mask_near(arr, GOLD), 200)[0])
    ordered = sorted(stations, key=lambda s: s[1], reverse=gold_x > arr.shape[1] / 2)
    if len(ordered) != 8 or abs(gold_y - 260) > 10:
        raise ValueError("station traversal is ambiguous")
    ref = {shape: color for _, _, shape, color in ordered[:4]}
    cand = {shape: color for _, _, shape, color in ordered[4:]}
    if set(ref) != {"circle", "square", "triangle", "diamond"} or set(cand) != set(ref):
        raise ValueError("each shape must occur once per run")
    return "yes" if ref == cand else "no"
