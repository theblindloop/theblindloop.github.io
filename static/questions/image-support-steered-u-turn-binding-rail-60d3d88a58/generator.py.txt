from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 520
BG = (248, 246, 240)
NAVY = (30, 48, 75)
GOLD = (230, 174, 45)
HALOS = {
    "coral": (226, 91, 84),
    "green": (67, 158, 105),
    "blue": (58, 128, 189),
    "purple": (145, 94, 174),
}
SHAPES = ("circle", "square", "triangle", "diamond")
COLORS = tuple(HALOS)


def _gold_for_reference(reference_order: list[str]) -> dict[str, str]:
    # The reference uses a deterministic palette assignment recoverable from its pixels.
    # Order determines which shape receives which shuffled color without a hidden label field.
    seed = sum((i + 1) * SHAPES.index(shape) for i, shape in enumerate(reference_order))
    colors = list(COLORS)
    random.Random(seed).shuffle(colors)
    return dict(zip(reference_order, colors, strict=True))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_order = list(SHAPES)
    rng.shuffle(reference_order)
    candidate_order = list(SHAPES)
    rng.shuffle(candidate_order)
    mapping = _gold_for_reference(reference_order)
    candidate_halos = [mapping[s] for s in candidate_order]
    if seed % 2:
        a, b = rng.sample(range(4), 2)
        candidate_halos[a], candidate_halos[b] = candidate_halos[b], candidate_halos[a]
    return {
        "reference_order": reference_order,
        "candidate_order": candidate_order,
        "candidate_halos": candidate_halos,
        "mirror_x": bool(rng.randrange(2)),
    }


def _shape(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int) -> None:
    if name == "circle":
        draw.ellipse((cx - 11, cy - 11, cx + 11, cy + 11), fill=NAVY)
    elif name == "square":
        draw.rectangle((cx - 10, cy - 10, cx + 10, cy + 10), fill=NAVY)
    elif name == "triangle":
        draw.polygon(((cx, cy - 12), (cx - 12, cy + 10), (cx + 12, cy + 10)), fill=NAVY)
    elif name == "diamond":
        draw.polygon(((cx, cy - 13), (cx - 13, cy), (cx, cy + 13), (cx + 13, cy)), fill=NAVY)
    else:
        raise ValueError("unknown shape")


def render(scene: dict) -> Image.Image:
    reference_order = list(scene["reference_order"])
    candidate_order = list(scene["candidate_order"])
    candidate_halos = list(scene["candidate_halos"])
    mirror = bool(scene["mirror_x"])
    if sorted(reference_order) != sorted(SHAPES) or sorted(candidate_order) != sorted(SHAPES):
        raise ValueError("each run must contain every shape once")
    if sorted(candidate_halos) != sorted(COLORS):
        raise ValueError("candidate halos must be a color permutation")
    mapping = _gold_for_reference(reference_order)
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    def tx(x: int) -> int:
        return WIDTH - x if mirror else x

    # One continuous regional relay rail; all eight stations occupy a narrow
    # band so the evidence support follows the rail rather than a filled panel.
    draw.line([(tx(165), 260), (tx(555), 260)], fill=NAVY, width=12)
    ref_x = [205, 250, 295, 340]
    cand_x = [385, 430, 475, 520]
    for x in ref_x + cand_x:
        draw.line([(tx(x), 260), (tx(x), 230)], fill=NAVY, width=7)

    draw.ellipse((tx(165) - 18, 242, tx(165) + 18, 278), fill=GOLD, outline=NAVY, width=4)
    for x, shape in zip(ref_x, reference_order, strict=True):
        cx, cy = tx(x), 210
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=HALOS[mapping[shape]], outline=BG, width=2)
        _shape(draw, shape, cx, cy)
    for x, shape, halo in zip(cand_x, candidate_order, candidate_halos, strict=True):
        cx, cy = tx(x), 210
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=HALOS[halo], outline=BG, width=2)
        _shape(draw, shape, cx, cy)
    return image


def analytic_gold(scene: dict) -> str:
    reference_order = list(scene["reference_order"])
    candidate_order = list(scene["candidate_order"])
    candidate_halos = list(scene["candidate_halos"])
    mapping = _gold_for_reference(reference_order)
    return "yes" if all(mapping[s] == c for s, c in zip(candidate_order, candidate_halos, strict=True)) else "no"


def margin(scene: dict) -> float:
    reference_order = list(scene["reference_order"])
    candidate_order = list(scene["candidate_order"])
    candidate_halos = list(scene["candidate_halos"])
    mapping = _gold_for_reference(reference_order)
    correct = sum(mapping[s] == c for s, c in zip(candidate_order, candidate_halos, strict=True))
    return float(abs(correct - 3.0))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 0.5
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
