"""Deterministic scene sampler and renderer for dual ruler calibration."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


BACKGROUND = (248, 250, 252)
DARK = (30, 41, 55)
PALETTES = (
    (231, 111, 38),
    (220, 86, 34),
    (239, 132, 42),
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, comfortably raster-resolvable comparison."""

    rng = random.Random(int(seed))
    winner = "LEFT" if int(seed) % 2 == 0 else "RIGHT"
    smaller = rng.choice((0.50, 0.75, 1.00, 1.25, 1.50, 1.75, 2.00, 2.25))
    gap = rng.choice((0.50, 0.75, 1.00, 1.25))
    larger = smaller + gap
    left_reading, right_reading = (
        (larger, smaller) if winner == "LEFT" else (smaller, larger)
    )

    left_spacing = rng.choice((52, 58, 64, 70))
    right_spacing = rng.choice((52, 58, 64, 70))
    if right_spacing == left_spacing:
        right_spacing = (52, 58, 64, 70)[((52, 58, 64, 70).index(right_spacing) + 1) % 4]

    return {
        "width": 800,
        "height": 500,
        "left_x": rng.randint(46, 70),
        "right_x": rng.randint(638, 662),
        "left_bottom": rng.randint(372, 420),
        "right_bottom": rng.randint(372, 420),
        "left_spacing": left_spacing,
        "right_spacing": right_spacing,
        "left_reading": left_reading,
        "right_reading": right_reading,
        "palette_index": rng.randrange(len(PALETTES)),
    }


def _draw_ruler(
    draw: ImageDraw.ImageDraw,
    *,
    x: int,
    bottom: int,
    spacing: int,
    reading: float,
    marker: tuple[int, int, int],
) -> None:
    top = bottom - 4 * spacing
    draw.line((x, top, x, bottom), fill=DARK, width=5)
    for index in range(5):
        y = bottom - index * spacing
        draw.line((x - 34, y, x + 34, y), fill=DARK, width=5)

    marker_y = int(round(bottom - reading * spacing))
    # The pointer remains visibly separated from the ruler ink, so the two
    # distant ruler/marker assemblies remain disconnected evidence components.
    draw.line((x + 47, marker_y, x + 76, marker_y), fill=marker, width=7)
    draw.polygon(
        ((x + 41, marker_y), (x + 51, marker_y - 8), (x + 51, marker_y + 8)),
        fill=marker,
    )
    draw.ellipse((x + 72, marker_y - 12, x + 96, marker_y + 12), fill=marker)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    marker = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    left_x = int(scene["left_x"])
    right_x = int(scene["right_x"])
    left_bottom = int(scene["left_bottom"])
    right_bottom = int(scene["right_bottom"])
    left_spacing = int(scene["left_spacing"])
    right_spacing = int(scene["right_spacing"])
    left_reading = float(scene["left_reading"])
    right_reading = float(scene["right_reading"])

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_ruler(
        draw,
        x=left_x,
        bottom=left_bottom,
        spacing=left_spacing,
        reading=left_reading,
        marker=marker,
    )
    _draw_ruler(
        draw,
        x=right_x,
        bottom=right_bottom,
        spacing=right_spacing,
        reading=right_reading,
        marker=marker,
    )
    return image


def analytic_gold(scene: dict) -> str:
    left = float(scene["left_reading"])
    right = float(scene["right_reading"])
    if left == right:
        raise ValueError("equal calibrated readings have no unique answer")
    return "LEFT" if left > right else "RIGHT"


def margin(scene: dict) -> float:
    return abs(float(scene["left_reading"]) - float(scene["right_reading"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.35


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
