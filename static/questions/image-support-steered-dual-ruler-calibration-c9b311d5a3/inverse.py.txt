"""Independent raster-only inverse for the calibrated ruler comparison."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 60:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _cluster_rows(rows: np.ndarray) -> list[float]:
    if not len(rows):
        return []
    values = sorted(int(value) for value in rows)
    groups = [[values[0]]]
    for value in values[1:]:
        if value <= groups[-1][-1] + 1:
            groups[-1].append(value)
        else:
            groups.append([value])
    return [float(np.mean(group)) for group in groups]


def _reading(dark: np.ndarray, marker_y: float, marker_x: float) -> float:
    height, width = dark.shape
    x0 = max(0, int(round(marker_x)) - 145)
    x1 = min(width, int(round(marker_x)) - 25)
    if x1 - x0 < 60:
        raise ValueError("ruler search strip is missing")
    row_ink = dark[:, x0:x1].sum(axis=1)
    tick_rows = np.flatnonzero(row_ink >= 35)
    centers = _cluster_rows(tick_rows)
    if len(centers) != 5:
        raise ValueError(f"expected five visible ruler ticks, found {len(centers)}")
    centers.sort()
    gaps = np.diff(np.asarray(centers, dtype=np.float64))
    spacing = float(np.median(gaps))
    if spacing < 40.0 or spacing > 80.0 or float(np.max(np.abs(gaps - spacing))) > 2.0:
        raise ValueError("tick spacing is not stably recoverable")
    bottom = centers[-1]
    reading = (bottom - float(marker_y)) / spacing
    if not -0.1 <= reading <= 4.1:
        raise ValueError("marker falls outside the calibrated ruler")
    return reading


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    orange = (red >= 185) & (green >= 55) & (green <= 165) & (blue <= 105) & (red - green >= 65)
    marker_groups = _components(orange)
    if len(marker_groups) != 2:
        raise ValueError(f"expected two orange marker components, found {len(marker_groups)}")
    markers = sorted(
        ((float(group[:, 1].mean()), float(group[:, 0].mean())) for group in marker_groups),
        key=lambda item: item[0],
    )
    if markers[1][0] - markers[0][0] < image.width * 0.55:
        raise ValueError("the two ruler markers are not distinctly separated")

    dark = (red <= 75) & (green <= 80) & (blue <= 90)
    left = _reading(dark, marker_y=markers[0][1], marker_x=markers[0][0])
    right = _reading(dark, marker_y=markers[1][1], marker_x=markers[1][0])
    difference = left - right
    if abs(difference) < 0.30:
        raise ValueError("calibrated comparison is raster-ambiguous")
    return "LEFT" if difference > 0 else "RIGHT"
