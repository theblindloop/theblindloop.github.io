from __future__ import annotations

import random
from PIL import Image, ImageDraw

COLORS = ("red", "blue", "green")
RGB = {"red": (220, 45, 55), "blue": (35, 105, 210), "green": (25, 155, 85)}


def _iou(pair: list[list[int]]) -> float:
    a, b = pair
    overlap = max(0, min(a[1], b[1]) - max(a[0], b[0]))
    union = max(a[1], b[1]) - min(a[0], b[0])
    return overlap / union


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    winner = COLORS[seed % len(COLORS)]
    order = list(COLORS)
    rng.shuffle(order)
    ys = (72, 154, 236)
    rows = []
    loser_offsets = [rng.randint(72, 88), rng.randint(94, 108)]
    rng.shuffle(loser_offsets)
    loser_i = 0
    for color, y in zip(order, ys, strict=True):
        offset = rng.randint(24, 34) if color == winner else loser_offsets[loser_i]
        loser_i += color != winner
        length = rng.randint(126, 150)
        left = rng.randint(38, 300 - length - offset)
        rows.append({
            "color": color,
            "y": y,
            "intervals": [[left, left + length], [left + offset, left + offset + length]],
        })
    return {"width": 360, "height": 308, "rows": rows}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    for x in range(40, scene["width"] - 20, 40):
        draw.line((x, 24, x, scene["height"] - 24), fill=(225, 225, 222), width=1)
    for row in scene["rows"]:
        color = RGB[row["color"]]
        y = row["y"]
        for dy, interval in zip((-11, 11), row["intervals"], strict=True):
            x1, x2 = interval
            draw.rectangle((x1, y + dy - 5, x2, y + dy + 5), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    scores = {row["color"]: _iou(row["intervals"]) for row in scene["rows"]}
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted((_iou(row["intervals"]) for row in scene["rows"]), reverse=True)
    return values[0] - values[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
