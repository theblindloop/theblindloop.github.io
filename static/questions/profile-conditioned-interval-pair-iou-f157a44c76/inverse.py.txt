from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_COLORS = {"red": (220, 45, 55), "blue": (35, 105, 210), "green": (25, 155, 85)}


def _components(mask: np.ndarray) -> list[tuple[int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for sy, sx in zip(*np.nonzero(mask), strict=True):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            for yy, xx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        if len(xs) >= 100:
            result.append((min(xs), max(xs), len(xs)))
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    scores = {}
    for name, rgb in _COLORS.items():
        parts = _components(np.all(arr == np.array(rgb, dtype=np.uint8), axis=2))
        if len(parts) != 2:
            raise ValueError(f"expected two visible {name} intervals, found {len(parts)}")
        spans = sorted((part[0], part[1]) for part in parts)
        a, b = spans
        overlap = max(0, min(a[1], b[1]) - max(a[0], b[0]))
        union = max(a[1], b[1]) - min(a[0], b[0])
        if union <= 0:
            raise ValueError("degenerate interval union")
        scores[name] = overlap / union
    ordered = sorted(scores, key=scores.get, reverse=True)
    if scores[ordered[0]] - scores[ordered[1]] < 0.05:
        raise ValueError("ambiguous pixel decision")
    return ordered[0]
