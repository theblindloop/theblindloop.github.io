"""Independent pixel-only inverse arm for the ruler-threshold graph.

This module deliberately duplicates segmentation, ruler measurement, graph
construction, and articulation scoring.  It accepts only the final raster.
"""

from __future__ import annotations

import math
from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image

_POINT_COLORS: dict[str, np.ndarray] = {
    "dark": np.array((38, 43, 50), dtype=np.float64),
    "red": np.array((214, 58, 67), dtype=np.float64),
    "blue": np.array((48, 105, 205), dtype=np.float64),
    "green": np.array((34, 151, 91), dtype=np.float64),
    "orange": np.array((229, 133, 35), dtype=np.float64),
}
_RULER_COLOR = np.array((112, 120, 130), dtype=np.float64)
_ANSWER_COLORS = ("red", "blue", "green", "orange")
_POINT_COUNT = 15


@dataclass(frozen=True)
class PixelMeasurement:
    points: tuple[tuple[float, float], ...]
    colors: tuple[str, ...]
    threshold: float
    scores: tuple[int, ...]
    margin: float
    edge_clearance: float


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y0, x0 = int(raw_y), int(raw_x)
        if visited[y0, x0]:
            continue
        visited[y0, x0] = True
        pending = [(y0, x0)]
        pixels: list[tuple[int, int]] = []
        while pending:
            y, x = pending.pop()
            pixels.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not visited[ny, nx]
                ):
                    visited[ny, nx] = True
                    pending.append((ny, nx))
        output.append(np.asarray(pixels, dtype=np.int32))
    return output


def _classify_center(arr: np.ndarray, x: float, y: float) -> str:
    xx, yy = int(round(x)), int(round(y))
    patch = arr[yy - 3 : yy + 4, xx - 3 : xx + 4].astype(np.float64)
    if patch.shape[:2] != (7, 7):
        raise ValueError("dot center is too close to the raster boundary")
    rgb = np.median(patch.reshape(-1, 3), axis=0)
    distances = {
        name: float(np.linalg.norm(rgb - reference))
        for name, reference in _POINT_COLORS.items()
    }
    label = min(distances, key=distances.get)
    if distances[label] > 20.0:
        raise ValueError(f"unrecognized dot-center color at {(x, y)}")
    return label


def _measure_ruler(arr: np.ndarray) -> float:
    gray_distance = np.linalg.norm(arr.astype(np.float64) - _RULER_COLOR, axis=2)
    parts = [part for part in _components(gray_distance < 8.0) if len(part) >= 100]
    if len(parts) != 1:
        raise ValueError(f"expected one gray distance ruler, found {len(parts)}")
    pixels = parts[0]
    # Recover the centers of the two tall end caps.  Using the full bounding
    # box would incorrectly include half the stroke width at both ends.
    cap_columns: list[tuple[int, float]] = []
    for raw_x in np.unique(pixels[:, 1]):
        x = int(raw_x)
        ys = pixels[pixels[:, 1] == x, 0]
        if int(ys.max()) - int(ys.min()) >= 14:
            cap_columns.append((x, float(np.median(ys))))
    if len(cap_columns) < 4:
        raise ValueError("distance ruler has no measurable end caps")
    midpoint = (cap_columns[0][0] + cap_columns[-1][0]) / 2.0
    left = [(x, y) for x, y in cap_columns if x < midpoint]
    right = [(x, y) for x, y in cap_columns if x > midpoint]
    if not left or not right:
        raise ValueError("distance ruler end caps overlap")
    left_x = float(np.mean([x for x, _ in left]))
    left_y = float(np.mean([y for _, y in left]))
    right_x = float(np.mean([x for x, _ in right]))
    right_y = float(np.mean([y for _, y in right]))
    threshold = math.hypot(right_x - left_x, right_y - left_y)
    if not 55.0 <= threshold <= 90.0:
        raise ValueError(f"implausible ruler length {threshold}")
    return threshold


def _adjacency(
    points: tuple[tuple[float, float], ...], threshold: float
) -> list[set[int]]:
    graph = [set() for _ in points]
    for left in range(len(points)):
        for right in range(left + 1, len(points)):
            if math.dist(points[left], points[right]) < threshold:
                graph[left].add(right)
                graph[right].add(left)
    return graph


def _graph_components(graph: list[set[int]], removed: int | None = None) -> list[list[int]]:
    unseen = set(range(len(graph)))
    if removed is not None:
        unseen.discard(removed)
    output: list[list[int]] = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        pending = deque([start])
        component: list[int] = []
        while pending:
            node = pending.popleft()
            component.append(node)
            for neighbor in graph[node]:
                if neighbor != removed and neighbor in unseen:
                    unseen.remove(neighbor)
                    pending.append(neighbor)
        output.append(component)
    return output


def _score(graph: list[set[int]], removed: int) -> int:
    parts = _graph_components(graph, removed)
    return sum(
        len(parts[left]) * len(parts[right])
        for left in range(len(parts))
        for right in range(left + 1, len(parts))
    )


def measure_image(image: Image.Image) -> PixelMeasurement:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[2] != 3 or min(arr.shape[:2]) < 32:
        raise ValueError("expected a bounded RGB raster")

    distances = np.stack(
        [
            np.linalg.norm(arr.astype(np.float64) - reference, axis=2)
            for reference in _POINT_COLORS.values()
        ],
        axis=0,
    )
    point_mask = distances.min(axis=0) < 8.0
    parts = [part for part in _components(point_mask) if 200 <= len(part) <= 360]
    if len(parts) != _POINT_COUNT:
        raise ValueError(f"expected {_POINT_COUNT} dot components, found {len(parts)}")

    recovered: list[tuple[float, float, str]] = []
    for part in parts:
        y = float(part[:, 0].mean())
        x = float(part[:, 1].mean())
        recovered.append((x, y, _classify_center(arr, x, y)))
    recovered.sort(key=lambda item: (item[1], item[0]))
    points = tuple((x, y) for x, y, _ in recovered)
    colors = tuple(color for _, _, color in recovered)
    if any(colors.count(color) != 1 for color in _ANSWER_COLORS):
        raise ValueError("expected exactly one dot of every answer color")

    threshold = _measure_ruler(arr)
    graph = _adjacency(points, threshold)
    if len(_graph_components(graph)) != 1:
        raise ValueError("recovered threshold graph is not connected")
    scores = tuple(_score(graph, index) for index in range(len(points)))
    colored_scores = sorted(
        (score for score, color in zip(scores, colors, strict=True) if color != "dark"),
        reverse=True,
    )
    if len(colored_scores) != 4:
        raise ValueError("expected four colored candidates")
    margin = float(colored_scores[0] - colored_scores[1])
    edge_clearance = min(
        abs(math.dist(points[left], points[right]) - threshold)
        for left in range(len(points))
        for right in range(left + 1, len(points))
    )
    return PixelMeasurement(
        points=points,
        colors=colors,
        threshold=threshold,
        scores=scores,
        margin=margin,
        edge_clearance=edge_clearance,
    )


def decision_from_image(image: Image.Image) -> str:
    measured = measure_image(image)
    candidates = sorted(
        (
            (score, color)
            for score, color in zip(measured.scores, measured.colors, strict=True)
            if color != "dark"
        ),
        reverse=True,
    )
    if candidates[0][0] <= 0 or candidates[0][0] == candidates[1][0]:
        raise ValueError(f"no unique colored articulation point: {candidates}")
    if sum(score > 0 for score, _ in candidates) != 1:
        raise ValueError(f"more than one colored bridge: {candidates}")
    return candidates[0][1]
