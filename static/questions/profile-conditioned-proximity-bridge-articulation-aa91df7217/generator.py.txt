"""Deterministic scene generator and analytic arm for an implicit graph task.

Every answer-bearing quantity is rendered: vertices are the visible dot centers
and the edge threshold is the visible end-to-end span of the gray bracket.
"""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 248, 244)
DARK = (38, 43, 50)
RULER = (112, 120, 130)
PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (214, 58, 67),
    "blue": (48, 105, 205),
    "green": (34, 151, 91),
    "orange": (229, 133, 35),
}
ANSWER_COLORS = tuple(PALETTE)
POINT_COUNT = 15
OUTER_RADIUS = 9
INNER_RADIUS = 7
SAMPLE_EDGE_FLOOR = 0.45
QUARANTINE_EDGE_GAP = 2.5


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _threshold_from_ruler(ruler: dict) -> float:
    return _distance(
        (float(ruler["x0"]), float(ruler["y0"])),
        (float(ruler["x1"]), float(ruler["y1"])),
    )


def _adjacency(
    centers: list[tuple[float, float]], threshold: float
) -> list[set[int]]:
    graph = [set() for _ in centers]
    for left in range(len(centers)):
        for right in range(left + 1, len(centers)):
            if _distance(centers[left], centers[right]) < threshold:
                graph[left].add(right)
                graph[right].add(left)
    return graph


def _components(graph: list[set[int]], removed: int | None = None) -> list[list[int]]:
    unseen = set(range(len(graph)))
    if removed is not None:
        unseen.discard(removed)
    output: list[list[int]] = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        pending = deque([start])
        component: list[int] = []
        while pending:
            node = pending.popleft()
            component.append(node)
            for neighbor in graph[node]:
                if neighbor != removed and neighbor in unseen:
                    unseen.remove(neighbor)
                    pending.append(neighbor)
        output.append(component)
    return output


def _separated_pair_score(graph: list[set[int]], removed: int) -> int:
    """Pairs connected before removal but separated afterward.

    Accepted scenes have a connected base graph.  For post-removal component
    sizes s_i, the newly separated unordered pairs are sum_{i<j} s_i*s_j.
    """

    parts = _components(graph, removed)
    score = 0
    for left in range(len(parts)):
        for right in range(left + 1, len(parts)):
            score += len(parts[left]) * len(parts[right])
    return score


def _scene_graph(scene: dict) -> tuple[list[dict], list[set[int]], float]:
    points = scene["points"]
    ruler = scene["ruler"]
    centers = [(float(point["x"]), float(point["y"])) for point in points]
    # Keep the public top-level scene boundary explicit for the rendered-field
    # audit: nested ruler coordinates are representation details of ``ruler``.
    threshold = _distance(
        (float(ruler["x0"]), float(ruler["y0"])),
        (float(ruler["x1"]), float(ruler["y1"])),
    )
    return points, _adjacency(centers, threshold), threshold


def _colored_scores(scene: dict) -> list[tuple[int, str]]:
    points, graph, _ = _scene_graph(scene)
    return [
        (_separated_pair_score(graph, index), str(point["color"]))
        for index, point in enumerate(points)
        if point["color"] != "dark"
    ]


def edge_clearance(scene: dict) -> float:
    points, _, threshold = _scene_graph(scene)
    centers = [(float(point["x"]), float(point["y"])) for point in points]
    return min(
        abs(_distance(centers[left], centers[right]) - threshold)
        for left in range(len(centers))
        for right in range(left + 1, len(centers))
    )


def _base_layout(near_boundary: bool) -> list[tuple[float, float]]:
    outer = 1.90 if near_boundary else 1.80
    left = [
        (-0.72, -0.23),
        (-0.72, 0.23),
        (-1.25, -0.45),
        (-1.40, 0.00),
        (-1.25, 0.45),
        (-outer, -0.32),
        (-outer, 0.32),
    ]
    return [(0.0, 0.0), *left, *((-x, y) for x, y in left)]


def sample_scene(seed: int) -> dict:
    """Sample a varied, balanced graph with one colored articulation vertex."""

    rng = random.Random(int(seed))
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    near_boundary = (int(seed) // len(ANSWER_COLORS)) % 6 == 0

    for _attempt in range(6000):
        threshold = rng.randint(68, 78)
        angle = rng.uniform(0.0, 2.0 * math.pi)
        cosine, sine = math.cos(angle), math.sin(angle)
        center_x = rng.randint(236, 276)
        center_y = rng.randint(214, 258)
        cross_scale = rng.uniform(0.91, 1.09)
        local = _base_layout(near_boundary)
        centers: list[tuple[int, int]] = []
        for index, (along, across) in enumerate(local):
            # Independent perturbations prevent the rendered population from
            # being a rigidly transformed template while topology is checked.
            if index:
                along += rng.uniform(-0.035, 0.035)
                across += rng.uniform(-0.040, 0.040)
            else:
                along += rng.uniform(-0.012, 0.012)
                across += rng.uniform(-0.012, 0.012)
            local_x = threshold * along
            local_y = threshold * across * cross_scale
            x = int(round(center_x + local_x * cosine - local_y * sine))
            y = int(round(center_y + local_x * sine + local_y * cosine))
            centers.append((x, y))

        if any(
            not 28 <= x <= WIDTH - 28 or not 28 <= y <= 410
            for x, y in centers
        ):
            continue
        if min(
            _distance(centers[left], centers[right])
            for left in range(len(centers))
            for right in range(left + 1, len(centers))
        ) < 24.0:
            continue

        ruler_x0 = rng.randint(175, 245)
        ruler_y = rng.randint(462, 478)
        ruler = {
            "x0": ruler_x0,
            "y0": ruler_y,
            "x1": ruler_x0 + threshold,
            "y1": ruler_y,
        }
        plain = [
            {"x": x, "y": y, "color": "dark"}
            for x, y in centers
        ]
        graph = _adjacency(
            [(float(x), float(y)) for x, y in centers], float(threshold)
        )
        if len(_components(graph)) != 1:
            continue
        articulations = [
            index
            for index in range(len(centers))
            if _separated_pair_score(graph, index) > 0
        ]
        if articulations != [0]:
            continue

        clearance = min(
            abs(_distance(centers[left], centers[right]) - threshold)
            for left in range(len(centers))
            for right in range(left + 1, len(centers))
        )
        if near_boundary:
            if not SAMPLE_EDGE_FLOOR < clearance < QUARANTINE_EDGE_GAP:
                continue
        elif clearance < 3.2:
            continue

        distractor_pool = list(range(1, len(plain)))
        rng.shuffle(distractor_pool)
        colored_indices = [0, *distractor_pool[:3]]
        other_colors = [color for color in ANSWER_COLORS if color != answer]
        rng.shuffle(other_colors)
        assigned = {0: answer}
        assigned.update(zip(colored_indices[1:], other_colors, strict=True))
        for index, color in assigned.items():
            plain[index]["color"] = color
        rng.shuffle(plain)
        scene = {"points": plain, "ruler": ruler}
        if analytic_gold(scene) != answer:
            continue
        return scene
    raise RuntimeError(f"could not sample a stable scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    points = scene["points"]
    ruler = scene["ruler"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for point in points:
        x, y = int(point["x"]), int(point["y"])
        draw.ellipse(
            (x - OUTER_RADIUS, y - OUTER_RADIUS, x + OUTER_RADIUS, y + OUTER_RADIUS),
            fill=DARK,
        )
        color = str(point["color"])
        if color != "dark":
            draw.ellipse(
                (x - INNER_RADIUS, y - INNER_RADIUS, x + INNER_RADIUS, y + INNER_RADIUS),
                fill=PALETTE[color],
            )
    x0, y0 = int(ruler["x0"]), int(ruler["y0"])
    x1, y1 = int(ruler["x1"]), int(ruler["y1"])
    draw.line((x0, y0, x1, y1), fill=RULER, width=4)
    draw.line((x0, y0 - 9, x0, y0 + 9), fill=RULER, width=4)
    draw.line((x1, y1 - 9, x1, y1 + 9), fill=RULER, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    points, graph, _ = _scene_graph(scene)
    if len(_components(graph)) != 1:
        raise ValueError("the threshold graph must be connected before removal")
    scored = sorted(_colored_scores(scene), reverse=True)
    if len(scored) != len(ANSWER_COLORS):
        raise ValueError("expected one point of every answer color")
    if scored[0][0] <= 0 or scored[0][0] == scored[1][0]:
        raise ValueError(f"expected one colored articulation point, got {scored}")
    # Check the prompt's stronger 'sole colored bridge' wording explicitly.
    positive = [color for score, color in scored if score > 0]
    if len(positive) != 1:
        raise ValueError(f"expected sole colored bridge, got {positive}")
    return scored[0][1]


def margin(scene: dict) -> float:
    scores = sorted((score for score, _ in _colored_scores(scene)), reverse=True)
    if len(scores) < 2:
        raise ValueError("runner-up margin requires at least two candidates")
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return edge_clearance(scene) < QUARANTINE_EDGE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return [
        (
            "reverse_unordered_points",
            {"points": list(reversed(scene["points"])), "ruler": dict(scene["ruler"])},
        )
    ]
