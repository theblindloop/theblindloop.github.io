"""Deterministic latent sampler, equal-ink renderer, and analytic arm."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

DECISIONS = ("red", "blue", "green", "orange")
RING_COLORS = {
    "red": "#D32F2F",
    "blue": "#1976D2",
    "green": "#2E8B57",
    "orange": "#ED7D18",
}
INKS = ("#20354F", "#3C294F", "#173F3A", "#4B2A2F")
BACKGROUNDS = ("#FCFAF2", "#F8FBFD", "#FBF8FC", "#F7FBF7")
GRIDS = ("#9EA6AD", "#A6A0AC", "#99A8A3", "#AAA29E")
GRID_SIZE = 13
BLOCK_SIZES = (2, 3, 4, 6)


def _correlation(left: list[float], right: list[float]) -> float:
    if len(left) != len(right) or len(left) < 8:
        raise ValueError("insufficient field samples")
    lm = sum(left) / len(left)
    rm = sum(right) / len(right)
    dl = [value - lm for value in left]
    dr = [value - rm for value in right]
    denom = math.sqrt(sum(value * value for value in dl) * sum(value * value for value in dr))
    if denom <= 1e-12:
        raise ValueError("texture field has no recoverable frequency gradient")
    return sum(a * b for a, b in zip(dl, dr)) / denom


def _rank(
    tile_blocks: list[list[int]],
    candidate_cells: list[list[int]],
    candidate_colors: list[str],
) -> list[tuple[float, str]]:
    if len(candidate_cells) != 4 or len(candidate_colors) != 4:
        raise ValueError("exactly four visible candidates are required")
    excluded = {tuple(map(int, cell)) for cell in candidate_cells}
    ranked = []
    for cell, color in zip(candidate_cells, candidate_colors):
        focus_r, focus_c = map(float, cell)
        observed: list[float] = []
        predicted: list[float] = []
        for row, values in enumerate(tile_blocks):
            for col, block in enumerate(values):
                if (row, col) in excluded:
                    continue
                observed.append(float(block))
                predicted.append(-math.hypot(row - focus_r, col - focus_c))
        ranked.append((_correlation(observed, predicted), str(color)))
    return sorted(ranked, reverse=True)


def analytic_gold(scene: dict) -> str:
    return _rank(
        scene["tile_blocks"], scene["candidate_cells"], scene["candidate_colors"]
    )[0][1]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Color and spatial slot are crossed deterministically over every 16 seeds.
    winning_color = DECISIONS[int(seed) % 4]
    winning_slot = (int(seed) // 4) % 4
    layouts = (
        [[3, 3], [3, 9], [9, 3], [9, 9]],
        [[2, 6], [6, 10], [10, 6], [6, 2]],
    )
    candidate_cells = [list(cell) for cell in layouts[(int(seed) // 16) % 2]]
    focus_r, focus_c = candidate_cells[winning_slot]

    other_colors = [color for color in DECISIONS if color != winning_color]
    rng.shuffle(other_colors)
    candidate_colors = other_colors[:]
    candidate_colors.insert(winning_slot, winning_color)

    tile_blocks: list[list[int]] = []
    for row in range(GRID_SIZE):
        values = []
        for col in range(GRID_SIZE):
            distance = math.hypot(row - focus_r, col - focus_c) + rng.uniform(-0.32, 0.32)
            if distance < 2.55:
                block = 6
            elif distance < 4.70:
                block = 4
            elif distance < 6.75:
                block = 3
            else:
                block = 2
            values.append(block)
        tile_blocks.append(values)

    phases = [[rng.randrange(2) for _ in range(GRID_SIZE)] for _ in range(GRID_SIZE)]
    palette_index = (int(seed) // 8) % len(INKS)
    scene = {
        "tile_blocks": tile_blocks,
        "phases": phases,
        "candidate_cells": candidate_cells,
        "candidate_colors": candidate_colors,
        "origin": [50 + rng.randrange(-9, 10), 50 + rng.randrange(-9, 10)],
        "cell_size": 26,
        "line_width": 2,
        "ink_color": INKS[palette_index],
        "background_color": BACKGROUNDS[palette_index],
        "grid_color": GRIDS[palette_index],
        "canvas_size": 440,
    }
    if analytic_gold(scene) != winning_color:
        raise RuntimeError("sampled texture field did not preserve its intended focus")
    return scene


def render(scene: dict) -> Image.Image:
    tile_blocks = scene["tile_blocks"]
    phases = scene["phases"]
    candidate_cells = scene["candidate_cells"]
    candidate_colors = scene["candidate_colors"]
    origin_x, origin_y = map(int, scene["origin"])
    cell_size = int(scene["cell_size"])
    line_width = int(scene["line_width"])
    ink = str(scene["ink_color"])
    background = str(scene["background_color"])
    grid = str(scene["grid_color"])
    canvas_size = int(scene["canvas_size"])
    interior = cell_size - line_width
    if interior != 24:
        raise ValueError("this renderer requires a 24-pixel checker interior")

    image = Image.new("RGB", (canvas_size, canvas_size), background)
    draw = ImageDraw.Draw(image)
    for row in range(GRID_SIZE):
        for col in range(GRID_SIZE):
            block = int(tile_blocks[row][col])
            phase = int(phases[row][col])
            if block not in BLOCK_SIZES or interior % block:
                raise ValueError("invalid checker block size")
            left = origin_x + col * cell_size + line_width
            top = origin_y + row * cell_size + line_width
            for yy in range(0, interior, block):
                for xx in range(0, interior, block):
                    if ((xx // block) + (yy // block) + phase) % 2 == 0:
                        draw.rectangle(
                            (left + xx, top + yy, left + xx + block - 1, top + yy + block - 1),
                            fill=ink,
                        )

    panel_end = GRID_SIZE * cell_size + line_width
    for index in range(GRID_SIZE + 1):
        x = origin_x + index * cell_size
        y = origin_y + index * cell_size
        draw.rectangle((x, origin_y, x + line_width - 1, origin_y + panel_end - 1), fill=grid)
        draw.rectangle((origin_x, y, origin_x + panel_end - 1, y + line_width - 1), fill=grid)

    radius = 9
    for (row, col), color in zip(candidate_cells, candidate_colors):
        cx = origin_x + int(col) * cell_size + line_width + interior // 2
        cy = origin_y + int(row) * cell_size + line_width + interior // 2
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            outline=RING_COLORS[str(color)],
            width=3,
        )
    return image


def margin(scene: dict) -> float:
    ranked = _rank(scene["tile_blocks"], scene["candidate_cells"], scene["candidate_colors"])
    return float(ranked[0][0] - ranked[1][0])


def is_quarantined(scene: dict) -> bool:
    try:
        ranked = _rank(scene["tile_blocks"], scene["candidate_cells"], scene["candidate_colors"])
    except ValueError:
        return True
    return ranked[0][0] < 0.45 or ranked[0][0] - ranked[1][0] < 0.08


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
