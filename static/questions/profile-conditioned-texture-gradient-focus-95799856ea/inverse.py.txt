"""Independent PNG-only inverse arm for the checker-frequency focus task."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

RING_RGB = {
    "red": (211, 47, 47),
    "blue": (25, 118, 210),
    "green": (46, 139, 87),
    "orange": (237, 125, 24),
}


def _runs(indices: np.ndarray) -> list[np.ndarray]:
    if not len(indices):
        return []
    groups = [[int(indices[0])]]
    for value in map(int, indices[1:]):
        if value == groups[-1][-1] + 1:
            groups[-1].append(value)
        else:
            groups.append([value])
    return [np.asarray(group, dtype=int) for group in groups]


def _corr(a: list[float], b: list[float]) -> float:
    aa = np.asarray(a, dtype=np.float64)
    bb = np.asarray(b, dtype=np.float64)
    if aa.size < 8 or aa.size != bb.size:
        raise ValueError("insufficient radial evidence")
    aa -= aa.mean()
    bb -= bb.mean()
    denom = float(np.linalg.norm(aa) * np.linalg.norm(bb))
    if denom < 1e-9:
        raise ValueError("checker frequencies carry no gradient")
    return float(np.dot(aa, bb) / denom)


def _palette_by_population(arr: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    colors, counts = np.unique(arr.reshape(-1, 3), axis=0, return_counts=True)
    if len(colors) < 7:
        raise ValueError("image lacks the grid, texture, or four candidate colors")
    order = np.argsort(counts)[::-1]
    # Equal-ink construction makes background, checker ink, and grid the three
    # dominant exact raster colors, in that population order.
    return colors[order[0]], colors[order[1]], colors[order[2]]


def _cell_index(value: float, lines: list[np.ndarray]) -> int:
    for index in range(len(lines) - 1):
        if lines[index][-1] < value < lines[index + 1][0]:
            return index
    raise ValueError("colored ring is not inside the recovered lattice")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    _, ink_rgb, grid_rgb = _palette_by_population(arr)
    grid_mask = np.all(arr == grid_rgb, axis=2)
    col_counts = grid_mask.sum(axis=0)
    row_counts = grid_mask.sum(axis=1)
    if int(col_counts.max(initial=0)) < 100 or int(row_counts.max(initial=0)) < 100:
        raise ValueError("continuous outlined lattice is absent")
    x_lines = _runs(np.flatnonzero(col_counts > 0.70 * col_counts.max()))
    y_lines = _runs(np.flatnonzero(row_counts > 0.70 * row_counts.max()))
    if len(x_lines) != 14 or len(y_lines) != 14:
        raise ValueError(f"expected a 13-by-13 lattice, recovered {len(x_lines)-1} by {len(y_lines)-1}")

    candidate_cells: list[tuple[int, int]] = []
    candidate_colors: list[str] = []
    for name, rgb in RING_RGB.items():
        target = np.asarray(rgb, dtype=np.int16)
        distance = np.linalg.norm(arr.astype(np.int16) - target, axis=2)
        points = np.argwhere(distance < 8.0)
        if len(points) < 40:
            raise ValueError(f"{name} candidate ring is missing")
        cy, cx = points.mean(axis=0)
        row = _cell_index(float(cy), y_lines)
        col = _cell_index(float(cx), x_lines)
        candidate_cells.append((row, col))
        candidate_colors.append(name)
    if len(set(candidate_cells)) != 4:
        raise ValueError("candidate rings do not mark four distinct cells")

    ink_mask = np.all(arr == ink_rgb, axis=2)
    excluded = set(candidate_cells)
    blocks: list[list[int | None]] = [[None] * 13 for _ in range(13)]
    for row in range(13):
        top = int(y_lines[row][-1] + 1)
        bottom = int(y_lines[row + 1][0])
        for col in range(13):
            if (row, col) in excluded:
                continue
            left = int(x_lines[col][-1] + 1)
            right = int(x_lines[col + 1][0])
            patch = ink_mask[top:bottom, left:right]
            if patch.shape != (24, 24):
                raise ValueError(f"unexpected checker patch size {patch.shape}")
            horizontal = float(np.count_nonzero(patch[:, 1:] != patch[:, :-1])) / patch.shape[0]
            vertical = float(np.count_nonzero(patch[1:, :] != patch[:-1, :])) / patch.shape[1]
            transitions = (horizontal + vertical) / 2.0
            block = int(round(24.0 / (transitions + 1.0)))
            if block not in (2, 3, 4, 6) or abs(transitions - (24.0 / block - 1.0)) > 0.15:
                raise ValueError("a tile has ambiguous spatial frequency")
            blocks[row][col] = block

    ranked: list[tuple[float, str]] = []
    for (focus_r, focus_c), color in zip(candidate_cells, candidate_colors):
        observed: list[float] = []
        radial: list[float] = []
        for row in range(13):
            for col in range(13):
                value = blocks[row][col]
                if value is None:
                    continue
                observed.append(float(value))
                radial.append(-math.hypot(row - focus_r, col - focus_c))
        ranked.append((_corr(observed, radial), color))
    ranked.sort(reverse=True)
    if ranked[0][0] < 0.45 or ranked[0][0] - ranked[1][0] < 0.08:
        raise ValueError("radial texture focus is ambiguous")
    return ranked[0][1]
