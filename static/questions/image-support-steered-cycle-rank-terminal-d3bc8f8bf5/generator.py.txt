"""Deterministic renderer and analytic answer for the cycle-rank world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ROUTE_RGB = (37, 52, 68)
GREEN_RGB = (37, 171, 104)
AMBER_RGB = (230, 153, 35)
VIOLET_RGB = (135, 84, 193)
TERMINAL_RGB = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
ROUTE_WIDTH = 9
NODE_RADIUS = 5.5
PORT_RADIUS = 14.0
QUARANTINE_THRESHOLD = 12.0


def _cycle_rank(vertices: dict[str, Any], edges: list[list[str]]) -> int:
    """Return E-V+1 for one connected route graph."""

    vertex_names = set(vertices)
    if not vertex_names or not edges:
        raise ValueError("empty route network")
    if any(edge[0] not in vertex_names or edge[1] not in vertex_names for edge in edges):
        raise ValueError("route edge references an unknown vertex")
    return len(edges) - len(vertex_names) + 1


def _single_loop(network_id: str, x0: float, y0: float, terminal_color: str) -> dict[str, Any]:
    # The large single loop has approximately the same total centerline length
    # and bounding box as the two smaller loops in _double_loop.
    vertices = {
        "s": [x0 + 8.0, y0 + 105.0],
        "p": [x0 + 32.0, y0 + 105.0],
        "a": [x0 + 60.0, y0 + 35.0],
        "b": [x0 + 170.0, y0 + 35.0],
        "c": [x0 + 198.0, y0 + 105.0],
        "d": [x0 + 170.0, y0 + 175.0],
        "e": [x0 + 60.0, y0 + 175.0],
        "t": [x0 + 220.0, y0 + 105.0],
    }
    return {
        "network_id": network_id,
        "topology": "single_loop",
        "vertices": vertices,
        "edges": [
            ["s", "p"],
            ["p", "a"],
            ["a", "b"],
            ["b", "c"],
            ["c", "d"],
            ["d", "e"],
            ["e", "p"],
            ["c", "t"],
        ],
        "start": "s",
        "terminal": "t",
        "terminal_color": terminal_color,
    }

    vertices = {

        "p": [x0 + 32.0, y0 + 105.0],

        "b": [x0 + 83.0, y0 + 55.0],
        "c": [x0 + 99.0, y0 + 105.0],
        "d": [x0 + 83.0, y0 + 155.0],
        "q": [x0 + 121.0, y0 + 105.0],
        "e": [x0 + 137.0, y0 + 55.0],
        "f": [x0 + 172.0, y0 + 55.0],
        "g": [x0 + 188.0, y0 + 105.0],
        "h": [x0 + 172.0, y0 + 155.0],
        "t": [x0 + 210.0, y0 + 105.0],

    return {

        "topology": "double_loop",
        "vertices": vertices,
        "edges": [
            ["s", "p"],
            ["p", "a"],
            ["a", "b"],
            ["b", "c"],
            ["c", "d"],
            ["d", "p"],
            ["c", "q"],

            ["e", "f"],
            ["f", "g"],

    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced topology, panel, color, and small layout variations."""

    rng = random.Random(int(seed))
    double_on_left = bool(rng.randrange(2))
    amber_on_left = bool(rng.randrange(2))
    # Diagonal placement makes the two decisive supports widely separated on
    # the image while the topology and answer identity remain counterbalanced.
    left_x = 8.0 + float(rng.randrange(-3, 4))
    right_x = 278.0 + float(rng.randrange(-3, 4))
    left_y = 42.0 + float(rng.randrange(-6, 7))
    right_y = 287.0 + float(rng.randrange(-6, 7))
    left_color = "amber" if amber_on_left else "violet"
    right_color = "violet" if amber_on_left else "amber"
    left = (
        _double_loop("left", left_x, left_y, left_color)
        if double_on_left
        else _single_loop("left", left_x, left_y, left_color)
    )
    right = (
        _single_loop("right", right_x, right_y, right_color)
        if double_on_left
        else _double_loop("right", right_x, right_y, right_color)
    )
    return {"networks": [left, right]}


def _point(network: dict[str, Any], name: str) -> tuple[float, float]:
    raw = network["vertices"][name]
    return float(raw[0]), float(raw[1])


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the visible network field with deterministic supersampling."""

    supersample = 4
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    def scaled(point: tuple[float, float]) -> tuple[float, float]:
        return (point[0] * supersample, point[1] * supersample)

    def disk(center: tuple[float, float], radius: float, fill: tuple[int, int, int]) -> None:
        cx, cy = scaled(center)
        rr = radius * supersample
        draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=fill)

    for network in scene["networks"]:
        vertices = network["vertices"]
        for edge in network["edges"]:
            start = _point(network, edge[0])
            end = _point(network, edge[1])
            draw.line(
                [scaled(start), scaled(end)],
                fill=ROUTE_RGB,
                width=round(ROUTE_WIDTH * supersample),
                joint="curve",
            )
        for name in vertices:
            disk(_point(network, name), NODE_RADIUS, ROUTE_RGB)

    # Ports are drawn last so the terminal identity is visible but the route
    # remains an unambiguous dark connected component behind each port.
    for network in scene["networks"]:
        disk(_point(network, network["start"]), PORT_RADIUS, GREEN_RGB)
        disk(
            _point(network, network["terminal"]),
            PORT_RADIUS,
            TERMINAL_RGB[network["terminal_color"]],
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    networks = scene["networks"]
    if len(networks) != 2:
        raise ValueError("scene must contain exactly two networks")
    ranks = [_cycle_rank(network["vertices"], network["edges"]) for network in networks]
    if sorted(ranks) != [1, 2]:
        raise ValueError(f"expected cycle ranks one and two, got {ranks}")
    winner = networks[ranks.index(max(ranks))]
    return str(winner["terminal_color"])


def _network_clearance(network: dict[str, Any]) -> float:
    points = [_point(network, name) for name in network["vertices"]]
    xs = [point[0] for point in points]
    ys = [point[1] for point in points]
    width = max(xs) - min(xs)
    height = max(ys) - min(ys)
    # The smallest inner pocket is the limiting finite-resolution cue in the
    # double-loop geometry; retain a continuous clearance proxy for margins.
    if network["topology"] == "double_loop":
        pocket_width = 35.0 - 2.0 * ROUTE_WIDTH
        pocket_height = 50.0 - 2.0 * ROUTE_WIDTH
        return min(pocket_width, pocket_height, width, height)
    return min(width - 2.0 * ROUTE_WIDTH, height - 2.0 * ROUTE_WIDTH)


def margin(scene: dict[str, Any]) -> float:
    networks = scene["networks"]
    ranks = [_cycle_rank(network["vertices"], network["edges"]) for network in networks]
    rank_gap = abs(float(ranks[0] - ranks[1]))
    all_points = [[_point(network, name) for name in network["vertices"]] for network in networks]
    separation = min(
        math.hypot(a[0] - b[0], a[1] - b[1])
        for a in all_points[0]
        for b in all_points[1]
    ) - 2.0 * ROUTE_WIDTH
    clearances = [_network_clearance(network) for network in networks]
    return float(min(40.0 * rank_gap, separation, *clearances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < QUARANTINE_THRESHOLD or abs(
            _cycle_rank(scene["networks"][0]["vertices"], scene["networks"][0]["edges"])
            - _cycle_rank(scene["networks"][1]["vertices"], scene["networks"][1]["edges"])
        ) == 0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # No latent alias is claimed: the single scene field is fully rendered and
    # all answer-bearing values remain visible in the final raster.
    return []
