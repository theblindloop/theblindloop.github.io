from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
SUPERSAMPLE = 4
QUARANTINE_MARGIN = 200.0
PALETTES = (
    ("#f7f3e8", "#235f78"),
    ("#f4f1f8", "#75458b"),
    ("#f8f2e9", "#a24e32"),
    ("#eef6f1", "#287056"),
    ("#f5f3ed", "#435a7a"),
    ("#f8f0f2", "#9a3f61"),
)


def _polygon_area(points: list[list[int]] | list[tuple[float, float]]) -> float:
    total = 0.0
    for index, point in enumerate(points):
        following = points[(index + 1) % len(points)]
        total += float(point[0]) * float(following[1]) - float(following[0]) * float(point[1])
    return abs(total) / 2.0


def _clip_halfplane(
    points: list[list[int]] | list[tuple[float, float]], a: float, b: float, c: float
) -> list[tuple[float, float]]:
    output: list[tuple[float, float]] = []
    if not points:
        return output
    previous = (float(points[-1][0]), float(points[-1][1]))
    previous_value = a * previous[0] + b * previous[1] + c
    for raw in points:
        current = (float(raw[0]), float(raw[1]))
        current_value = a * current[0] + b * current[1] + c
        current_inside = current_value >= 0.0
        previous_inside = previous_value >= 0.0
        if current_inside != previous_inside:
            fraction = previous_value / (previous_value - current_value)
            output.append(
                (
                    previous[0] + fraction * (current[0] - previous[0]),
                    previous[1] + fraction * (current[1] - previous[1]),
                )
            )
        if current_inside:
            output.append(current)
        previous = current
        previous_value = current_value
    return output


def _area_scores(points: list[list[int]]) -> tuple[float, float]:
    total = _polygon_area(points)
    if total < 1.0:
        raise ValueError("degenerate patch")
    rising_side = _polygon_area(_clip_halfplane(points, 1.0, 1.0, -(WIDTH - 1.0)))
    falling_side = _polygon_area(_clip_halfplane(points, 1.0, -1.0, 0.0))
    return abs(2.0 * rising_side - total), abs(2.0 * falling_side - total)


def _convex_hull(points: list[list[int]]) -> list[list[int]]:
    ordered = sorted(set((int(x), int(y)) for x, y in points))
    if len(ordered) < 3:
        raise ValueError("too few boundary points")

    def cross(origin, first, second):
        return (first[0] - origin[0]) * (second[1] - origin[1]) - (first[1] - origin[1]) * (second[0] - origin[0])

    lower = []
    for point in ordered:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], point) <= 0:
            lower.pop()
        lower.append(point)
    upper = []
    for point in reversed(ordered):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], point) <= 0:
            upper.pop()
        upper.append(point)
    return [[x, y] for x, y in lower[:-1] + upper[:-1]]


def _polygon_centroid(points: list[list[int]]) -> tuple[float, float]:
    signed_twice_area = 0.0
    x_numerator = 0.0
    y_numerator = 0.0
    for index, point in enumerate(points):
        following = points[(index + 1) % len(points)]
        cross = float(point[0] * following[1] - following[0] * point[1])
        signed_twice_area += cross
        x_numerator += float(point[0] + following[0]) * cross
        y_numerator += float(point[1] + following[1]) * cross
    if abs(signed_twice_area) < 1.0:
        raise ValueError("degenerate patch")
    return (
        x_numerator / (3.0 * signed_twice_area),
        y_numerator / (3.0 * signed_twice_area),
    )


def _random_boundary(rng: random.Random) -> list[list[int]]:
    count = 48
    base_radius = rng.uniform(60.0, 70.0)
    center_x = (WIDTH - 1.0) / 2.0
    center_y = (HEIGHT - 1.0) / 2.0
    phases = [rng.uniform(0.0, 2.0 * math.pi) for _ in range(3)]
    coefficients = [rng.uniform(0.025, 0.065), rng.uniform(0.055, 0.12), rng.uniform(0.015, 0.05)]
    raw_points: list[list[int]] = []
    for index in range(count):
        theta = 2.0 * math.pi * index / count
        modulation = sum(
            coefficients[order - 2] * math.sin(order * theta + phases[order - 2])
            for order in range(2, 5)
        )
        radius = base_radius * (1.0 + modulation)
        raw_points.append(
            [
                int(round(center_x + radius * math.cos(theta))),
                int(round(center_y + radius * math.sin(theta))),
            ]
        )
    points = _convex_hull(raw_points)
    centroid_x, centroid_y = _polygon_centroid(points)
    shift_x = int(round(center_x - centroid_x))
    shift_y = int(round(center_y - centroid_y))
    return [[x + shift_x, y + shift_y] for x, y in points]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "rising" if int(seed) % 2 == 0 else "falling"
    for _ in range(1000):
        points = _random_boundary(rng)
        rising, falling = _area_scores(points)
        if abs(rising - falling) < 350.0:
            continue
        current = "rising" if rising < falling else "falling"
        if current != desired:
            points = [[WIDTH - 1 - int(x), int(y)] for x, y in points]
            points.reverse()
        background, color = PALETTES[rng.randrange(len(PALETTES))]
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "background": background,
            "color": color,
            "points": points,
        }
        if analytic_gold(scene) == desired and margin(scene) >= 325.0:
            return scene
    raise RuntimeError("could not sample a well-separated diagonal partition")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    points = [(int(p[0]) * scale, int(p[1]) * scale) for p in scene["points"]]
    draw.polygon(points, fill=scene["color"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    rising, falling = _area_scores(scene["points"])
    return "rising" if rising < falling else "falling"


def margin(scene: dict) -> float:
    rising, falling = _area_scores(scene["points"])
    return abs(rising - falling)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
