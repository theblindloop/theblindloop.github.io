from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    largest: list[tuple[int, int]] = []
    component_count = 0
    for y, x in np.argwhere(mask):
        yi, xi = int(y), int(x)
        if seen[yi, xi]:
            continue
        component_count += 1
        seen[yi, xi] = True
        queue = deque([(yi, xi)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) > len(largest):
            largest = points
    output = np.zeros_like(mask, dtype=bool)
    if largest:
        yy, xx = zip(*largest)
        output[np.asarray(yy), np.asarray(xx)] = True
    return output, component_count


def _longest_internal_plateau(values: np.ndarray, outer_value: int) -> int:
    longest = 0
    run = 0
    previous = None
    for raw in values.tolist():
        value = int(raw)
        if value == previous and value != outer_value:
            run += 1
        else:
            run = 1
        longest = max(longest, run)
        previous = value
    return longest


def _require_complete_smooth_patch(mask: np.ndarray) -> None:
    ys, xs = np.nonzero(mask)
    ymin, ymax = int(ys.min()), int(ys.max())
    xmin, xmax = int(xs.min()), int(xs.max())
    lefts: list[int] = []
    rights: list[int] = []
    for y in range(ymin, ymax + 1):
        row = np.flatnonzero(mask[y])
        if not len(row) or len(row) + 2 < int(row[-1] - row[0] + 1):
            raise ValueError("patch has a missing horizontal bite")
        lefts.append(int(row[0]))
        rights.append(int(row[-1]))
    tops: list[int] = []
    bottoms: list[int] = []
    for x in range(xmin, xmax + 1):
        column = np.flatnonzero(mask[:, x])
        if not len(column) or len(column) + 2 < int(column[-1] - column[0] + 1):
            raise ValueError("patch has a missing vertical bite")
        tops.append(int(column[0]))
        bottoms.append(int(column[-1]))
    for values in (np.asarray(lefts), np.asarray(rights), np.asarray(tops), np.asarray(bottoms)):
        if len(values) > 1 and int(np.max(np.abs(np.diff(values)))) > 32:
            raise ValueError("patch contour has an abrupt erased edge")
    if _longest_internal_plateau(np.asarray(lefts), xmin) > 42:
        raise ValueError("patch has a long left cut")
    if _longest_internal_plateau(np.asarray(rights), xmax) > 42:
        raise ValueError("patch has a long right cut")
    if _longest_internal_plateau(np.asarray(tops), ymin) > 42:
        raise ValueError("patch has a long top cut")
    if _longest_internal_plateau(np.asarray(bottoms), ymax) > 42:
        raise ValueError("patch has a long bottom cut")


def _measure(image: Image.Image) -> tuple[float, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.shape[0] != array.shape[1] or array.shape[0] < 128:
        raise ValueError("world requires a square image")
    border = np.concatenate(
        [
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    raw = np.linalg.norm(array - background, axis=2) > 45.0
    mask, component_count = _largest_component(raw)
    ys, xs = np.nonzero(mask)
    if len(xs) < 7000 or component_count != 1:
        raise ValueError("no single complete filled patch")
    extent_x = int(xs.max() - xs.min() + 1)
    extent_y = int(ys.max() - ys.min() + 1)
    if not (105 <= extent_x <= 180 and 105 <= extent_y <= 180):
        raise ValueError("patch extent is outside the declared world")
    _require_complete_smooth_patch(mask)
    yy, xx = np.indices(mask.shape)
    size = mask.shape[0]
    rising_sign = np.sign((xx + 0.5) + (yy + 0.5) - float(size))
    falling_sign = np.sign((xx + 0.5) - (yy + 0.5))
    rising = abs(float(np.sum(rising_sign[mask])))
    falling = abs(float(np.sum(falling_sign[mask])))
    return rising, falling


def decision_from_image(image: Image.Image) -> str:
    rising, falling = _measure(image)
    if abs(rising - falling) < 60.0:
        raise ValueError("diagonal area comparison is raster-ambiguous")
    return "rising" if rising < falling else "falling"
