"""Independent image-only inverse for visible mug-handle attachment side."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _largest_component_centroid(mask: np.ndarray) -> tuple[float, float, int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        y = int(y); x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        comp: list[tuple[int, int]] = []
        while stack:
            yy, xx = stack.pop()
            comp.append((yy, xx))
            for ny, nx in ((yy-1, xx), (yy+1, xx), (yy, xx-1), (yy, xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(comp) > len(best):
            best = comp
    if len(best) < 250:
        raise ValueError("required colored region is missing or too small")
    ys = np.fromiter((p[0] for p in best), dtype=float)
    xs = np.fromiter((p[1] for p in best), dtype=float)
    return float(xs.mean()), float(ys.mean()), len(best)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    # Public visual roles, detected broadly across all declared shades.
    # The high blue/red chroma requirement excludes the answer-neutral slate steam/saucer.
    blue = (b > r + 80) & (b > g + 9) & (b > 130)
    amber = (r > 185) & (g > 75) & (g < 205) & (b < 100) & (r > g + 28)
    hx, _, hn = _largest_component_centroid(blue)
    bx, _, bn = _largest_component_centroid(amber)
    if hn < 350 or bn < 1800:
        raise ValueError("colored handle or cup body lacks enough visible ink")
    delta = hx - bx
    if abs(delta) < 35:
        raise ValueError("ambiguous handle/body horizontal separation")
    return "left" if delta < 0 else "right"
