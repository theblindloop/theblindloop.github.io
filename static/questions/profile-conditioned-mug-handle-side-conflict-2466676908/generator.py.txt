"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

W = H = 512
AA = 3
BODY_PALETTES = ((245, 158, 11), (249, 115, 22), (234, 179, 8))
HANDLE_PALETTES = ((37, 99, 235), (8, 145, 178), (79, 70, 229))


def sample_scene(seed: int) -> dict:
    # Consecutive odd/even seeds are style-matched left/right counterfactuals.
    style_rng = random.Random((int(seed) - 1) // 2)
    return {
        "side": "left" if int(seed) % 2 else "right",
        "cup_x": style_rng.randint(228, 284),
        "cup_y": style_rng.randint(158, 194),
        "scale": round(style_rng.uniform(0.72, 1.12), 4),
        "palette": style_rng.randrange(len(BODY_PALETTES)),
        "steam_variant": style_rng.randrange(3),
        "saucer_variant": style_rng.randrange(3),
    }


def _xy(values):
    return tuple(int(round(v * AA)) for v in values)


def render(scene: dict) -> Image.Image:
    side = scene["side"]
    cx = float(scene["cup_x"])
    y = float(scene["cup_y"])
    s = float(scene["scale"])
    palette = int(scene["palette"])
    steam_variant = int(scene["steam_variant"])
    saucer_variant = int(scene["saucer_variant"])

    bg = (250, 248, 242)
    black = (30, 41, 59)
    grey = ((148, 163, 184), (100, 116, 139), (71, 85, 105))[saucer_variant]
    body = BODY_PALETTES[palette]
    handle = HANDLE_PALETTES[palette]
    coffee = (92, 51, 23)

    image = Image.new("RGB", (W * AA, H * AA), bg)
    d = ImageDraw.Draw(image)
    sw = max(5, int(round(3.5 * AA * s)))

    # Saucer is answer-neutral and always centered under the independently translated cup.
    d.ellipse(_xy((cx - 112*s, y + 147*s, cx + 112*s, y + 184*s)), fill=grey, outline=black, width=sw)
    d.ellipse(_xy((cx - 83*s, y + 153*s, cx + 83*s, y + 174*s)), fill=bg, outline=grey, width=max(2, sw//2))

    # Handle is painted behind the body. Mirroring changes only the declared side.
    sign = -1 if side == "left" else 1
    hx = cx + sign * 91*s
    outer = (hx - 53*s, y + 35*s, hx + 53*s, y + 129*s)
    inner = (hx - 34*s, y + 53*s, hx + 34*s, y + 111*s)
    d.ellipse(_xy(outer), fill=black)
    d.ellipse(_xy((outer[0]+5*s, outer[1]+5*s, outer[2]-5*s, outer[3]-5*s)), fill=handle)
    d.ellipse(_xy(inner), fill=bg, outline=black, width=max(3, sw//2))

    # A slightly tapered cup body with an open coffee rim.
    body_poly = [
        (cx - 75*s, y + 19*s), (cx + 75*s, y + 19*s),
        (cx + 61*s, y + 151*s), (cx + 48*s, y + 162*s),
        (cx - 48*s, y + 162*s), (cx - 61*s, y + 151*s),
    ]
    d.polygon([_xy(p) for p in body_poly], fill=body, outline=black, width=sw)
    d.ellipse(_xy((cx - 75*s, y, cx + 75*s, y + 42*s)), fill=body, outline=black, width=sw)
    d.ellipse(_xy((cx - 61*s, y + 8*s, cx + 61*s, y + 33*s)), fill=coffee, outline=black, width=max(2, sw//2))
    d.arc(_xy((cx - 58*s, y + 137*s, cx + 58*s, y + 166*s)), 5, 175, fill=(180, 83, 9), width=max(2, sw//2))

    # Three answer-neutral steam designs; their grey pixels are ignored by the oracle.
    dx = (-16, 0, 16)[steam_variant]
    for k in (-1, 1):
        x0 = cx + dx*s + k*24*s
        pts = [(x0, y-8*s), (x0-9*s, y-28*s), (x0+7*s, y-49*s), (x0, y-68*s)]
        d.line([_xy(p) for p in pts], fill=grey, width=max(3, sw//2), joint="curve")

    return image.resize((W, H), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    side = scene["side"]
    if side not in {"left", "right"}:
        raise ValueError(f"invalid side {side!r}")
    return side


def margin(scene: dict) -> float:
    # Analytic horizontal separation between the handle and body centers.
    return round(91.0 * float(scene["scale"]), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
