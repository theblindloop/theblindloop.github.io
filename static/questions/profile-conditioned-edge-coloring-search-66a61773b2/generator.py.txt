"""Deterministic renderer and analytic arm for Edge-Neighbor Color Search."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 820, 620
LABELS = ("A", "B", "C", "D")
RGB = {
    "coral": (222, 79, 72),
    "blue": (52, 111, 196),
    "green": (54, 151, 99),
    "gold": (232, 169, 45),
    "violet": (137, 91, 178),
    "cyan": (31, 157, 174),
}


def _conflicts(board: list[list[int]]) -> int:
    total = 0
    for row in range(3):
        for col in range(3):
            if col < 2 and board[row][col] == board[row][col + 1]:
                total += 1
            if row < 2 and board[row][col] == board[row + 1][col]:
                total += 1
    return total


def _conflict_counts(boards: list[list[list[int]]]) -> list[int]:
    return [_conflicts(board) for board in boards]


def _proper_board(rng: random.Random) -> list[list[int]]:
    rows = list(range(3))
    cols = list(range(3))
    rng.shuffle(rows)
    rng.shuffle(cols)
    offset = rng.randrange(3)
    return [[(rows[r] + cols[c] + offset) % 3 for c in range(3)] for r in range(3)]


def _distractor(rng: random.Random, proper: list[list[int]], used: set[tuple[int, ...]]) -> list[list[int]]:
    flat = [value for row in proper for value in row]
    for _ in range(1000):
        trial = flat[:]
        rng.shuffle(trial)
        key = tuple(trial)
        board = [trial[i : i + 3] for i in range(0, 9, 3)]
        if key not in used and 2 <= _conflicts(board) <= 7:
            used.add(key)
            return board
    raise RuntimeError("could not sample a distinct structured distractor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    proper = _proper_board(rng)
    used = {tuple(value for row in proper for value in row)}
    boards = [proper]
    boards.extend(_distractor(rng, proper, used) for _ in range(3))
    target = int(seed) % 4
    boards[0], boards[target] = boards[target], boards[0]
    palette = rng.sample(list(RGB), 3)
    cell_orders = []
    for _ in range(4):
        order = list(range(9))
        rng.shuffle(order)
        cell_orders.append(order)
    return {"boards": boards, "palette": palette, "cell_orders": cell_orders}


def render(scene: dict) -> Image.Image:
    boards = scene["boards"]
    palette = scene["palette"]
    cell_orders = scene["cell_orders"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=24)
    small = ImageFont.load_default(size=14)
    panel_origins = ((55, 74), (435, 74), (55, 334), (435, 334))
    pitch = 52
    board_offset = (88, 39)
    for panel_index, (panel_x, panel_y) in enumerate(panel_origins):
        draw.rounded_rectangle(
            (panel_x, panel_y, panel_x + 330, panel_y + 225),
            radius=12,
            fill=(255, 255, 255),
            outline=(74, 78, 84),
            width=3,
        )
        draw.text((panel_x + 19, panel_y + 91), LABELS[panel_index], fill=(30, 34, 40), font=font)
        grid_x = panel_x + board_offset[0]
        grid_y = panel_y + board_offset[1]
        board = boards[panel_index]
        for cell_index in cell_orders[panel_index]:
            row, col = divmod(cell_index, 3)
            left = grid_x + col * pitch + 4
            top = grid_y + row * pitch + 4
            draw.rectangle(
                (left, top, left + pitch - 8, top + pitch - 8),
                fill=RGB[palette[board[row][col]]],
            )
        for k in range(4):
            x = grid_x + k * pitch
            y = grid_y + k * pitch
            draw.line((x, grid_y, x, grid_y + 3 * pitch), fill=(45, 49, 55), width=3)
            draw.line((grid_x, y, grid_x + 3 * pitch, y), fill=(45, 49, 55), width=3)
        draw.text((panel_x + 107, panel_y + 202), "3 x 3 board", fill=(88, 92, 98), font=small)
    return image


def analytic_gold(scene: dict) -> str:
    counts = _conflict_counts(scene["boards"])
    winners = [index for index, count in enumerate(counts) if count == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have one proper coloring")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    counts = sorted(_conflict_counts(scene["boards"]))
    if not counts:
        return 0.0
    return float(counts[1] - counts[0])


def is_quarantined(scene: dict) -> bool:
    counts = _conflict_counts(scene["boards"])
    return counts.count(0) != 1 or sorted(counts)[1] - sorted(counts)[0] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cell_orders"] = [list(reversed(order)) for order in twin["cell_orders"]]
    return [("reverse_disjoint_cell_draw_order", twin)]
