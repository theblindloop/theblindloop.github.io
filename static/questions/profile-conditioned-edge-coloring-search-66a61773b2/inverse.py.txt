"""Independent pixel-only inverse arm.

The implementation knows the public palette and the declared reading-order rule,
but it receives no scene, renderer state, coordinates, labels, or gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
PALETTE = (
    (222, 79, 72),
    (52, 111, 196),
    (54, 151, 99),
    (232, 169, 45),
    (137, 91, 178),
    (31, 157, 174),
)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        area = len(xs)
        if area >= 900:
            span_x = max(xs) - min(xs) + 1
            span_y = max(ys) - min(ys) + 1
            if not (0.88 <= span_x / span_y <= 1.14 and area >= 0.85 * span_x * span_y):
                raise ValueError("colored component is not a solid cell interior")
            found.append(((min(xs) + max(xs)) / 2, (min(ys) + max(ys)) / 2, area))
    return found


def _largest_gap_split(values: list[float]) -> float:
    ordered = sorted(values)
    gaps = [(ordered[i + 1] - ordered[i], i) for i in range(len(ordered) - 1)]
    gap, index = max(gaps)
    if gap < 40:
        raise ValueError("cannot separate the four boards from pixels")
    return (ordered[index] + ordered[index + 1]) / 2


def _rank_three(values: list[float]) -> dict[float, int]:
    unique = sorted(set(values))
    if len(unique) != 3 or min(unique[i + 1] - unique[i] for i in range(2)) < 20:
        raise ValueError("board is not a recoverable 3-by-3 layout")
    return {value: rank for rank, value in enumerate(unique)}


def _conflicts(board: list[list[int]]) -> int:
    return sum(
        (col < 2 and board[row][col] == board[row][col + 1])
        + (row < 2 and board[row][col] == board[row + 1][col])
        for row in range(3)
        for col in range(3)
    )


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"))
    cells: list[tuple[float, float, int]] = []
    for color_index, rgb in enumerate(PALETTE):
        mask = np.all(pixels == np.array(rgb, dtype=np.uint8), axis=2)
        for x, y, _area in _components(mask):
            cells.append((x, y, color_index))
    if len(cells) != 36:
        raise ValueError(f"expected 36 visible color cells, recovered {len(cells)}")

    split_x = _largest_gap_split([cell[0] for cell in cells])
    split_y = _largest_gap_split([cell[1] for cell in cells])
    panels: list[list[tuple[float, float, int]]] = [[] for _ in range(4)]
    for x, y, color in cells:
        panel = (2 if y > split_y else 0) + (1 if x > split_x else 0)
        panels[panel].append((x, y, color))

    scores: list[int] = []
    for panel_cells in panels:
        if len(panel_cells) != 9:
            raise ValueError("each visible board must have nine cells")
        x_rank = _rank_three([x for x, _, _ in panel_cells])
        y_rank = _rank_three([y for _, y, _ in panel_cells])
        board = [[-1] * 3 for _ in range(3)]
        for x, y, color in panel_cells:
            row, col = y_rank[y], x_rank[x]
            if board[row][col] != -1:
                raise ValueError("two colored components occupy one recovered cell")
            board[row][col] = color
        if any(value < 0 for row in board for value in row):
            raise ValueError("recovered board has a missing cell")
        histogram = sorted(sum(row.count(color) for row in board) for color in set(sum(board, [])))
        if histogram != [3, 3, 3]:
            raise ValueError("board color histogram is not the declared balanced 3-3-3")
        scores.append(_conflicts(board))

    winners = [index for index, score in enumerate(scores) if score == 0]
    if len(winners) != 1:
        raise ValueError("pixels do not contain a unique zero-conflict board")
    runner_up_gap = sorted(scores)[1] - sorted(scores)[0]
    if runner_up_gap < 2:
        raise ValueError("pixel decision margin is below two conflict edges")
    return LABELS[winners[0]]
