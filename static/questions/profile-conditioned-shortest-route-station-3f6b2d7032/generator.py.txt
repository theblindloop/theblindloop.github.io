"""Deterministic visible graph for universal shortest-route membership."""

from __future__ import annotations

import itertools
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 520
CANDIDATE_NODES = (2, 3, 4)
COLOR_NAMES = ("orange", "green", "purple")
COLORS = {
    "orange": "#e5632f",
    "green": "#2c9653",
    "purple": "#8a53b5",
    "cyan": "#21b7cc",
    "yellow": "#f4c225",
    "edge": "#4a5058",
    "outline": "#17212a",
}

# Terminals 0/1, winning structural candidate 2, distractor candidates 3/4.
BASE_POSITIONS = (
    (65, 245), (835, 245), (450, 155), (300, 355), (600, 355),
    (235, 80), (235, 190), (665, 80), (665, 190), (165, 355),
    (395, 315), (505, 315), (735, 355), (450, 395), (450, 465),
)
BASE_EDGES = (
    (0, 5), (0, 6), (5, 2), (6, 2), (2, 7), (2, 8), (7, 1), (8, 1),
    (0, 9), (9, 3), (3, 10), (10, 11), (11, 4), (4, 12), (12, 1),
    (3, 13), (13, 4), (3, 14), (14, 4),
)


def _adjacency(edges: list[list[int]] | tuple[tuple[int, int], ...], removed: int | None = None) -> list[set[int]]:
    graph = [set() for _ in range(15)]
    for a, b in edges:
        if removed in (a, b):
            continue
        graph[a].add(b)
        graph[b].add(a)
    return graph


def _distance(edges: list[list[int]] | tuple[tuple[int, int], ...], removed: int | None = None) -> int:
    graph = _adjacency(edges, removed)
    if removed in (0, 1):
        return 10_000
    distances = {0: 0}
    pending = deque([0])
    while pending:
        node = pending.popleft()
        for neighbor in graph[node]:
            if neighbor not in distances:
                distances[neighbor] = distances[node] + 1
                pending.append(neighbor)
    return distances.get(1, 10_000)


def _penalties(scene: dict) -> list[int]:
    edges = scene["edges"]
    baseline = _distance(edges)
    return [_distance(edges, node) - baseline for node in CANDIDATE_NODES]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    mode = rng.randrange(4)
    positions = []
    for x, y in BASE_POSITIONS:
        x += rng.randint(-5, 5)
        y += rng.randint(-5, 5)
        if mode in (1, 3):
            x = WIDTH - x
        if mode in (2, 3):
            y = HEIGHT - y
        positions.append([x, y])
    permutations = tuple(itertools.permutations(COLOR_NAMES))
    candidate_colors = list(permutations[int(seed) % len(permutations)])
    return {
        "positions": positions,
        "edges": [list(edge) for edge in BASE_EDGES],
        "candidate_colors": candidate_colors,
    }


def analytic_gold(scene: dict) -> str:
    penalties = _penalties(scene)
    winner = max(range(3), key=lambda index: penalties[index])
    if penalties.count(penalties[winner]) != 1 or penalties[winner] <= 0:
        raise ValueError(f"no unique shortest-route station: {penalties}")
    return scene["candidate_colors"][winner]


def margin(scene: dict) -> float:
    penalties = sorted(_penalties(scene), reverse=True)
    return float(penalties[0] - penalties[1])


def is_quarantined(scene: dict) -> bool:
    try:
        positions = scene["positions"]
        edges = scene["edges"]
        colors = scene["candidate_colors"]
        if len(positions) != 15 or set(colors) != set(COLOR_NAMES):
            return True
        normalized = {tuple(sorted(edge)) for edge in edges}
        if len(normalized) != len(BASE_EDGES) or any(a == b or not (0 <= a < 15 and 0 <= b < 15) for a, b in normalized):
            return True
        if _distance(edges) != 4:
            return True
        penalties = _penalties(scene)
        if sorted(penalties) != [0, 0, 2] or max(penalties) >= 9990:
            return True
        degrees = [len(_adjacency(edges)[node]) for node in CANDIDATE_NODES]
        if degrees != [4, 4, 4]:
            return True
        for a, b in itertools.combinations(range(15), 2):
            if math.dist(positions[a], positions[b]) < 58:
                return True
        for a, b in normalized:
            px, py = positions[a]
            qx, qy = positions[b]
            length2 = (qx - px) ** 2 + (qy - py) ** 2
            for c, (rx, ry) in enumerate(positions):
                if c in (a, b):
                    continue
                along = ((rx - px) * (qx - px) + (ry - py) * (qy - py)) / length2
                if 0.08 < along < 0.92:
                    sx, sy = px + along * (qx - px), py + along * (qy - py)
                    if math.hypot(rx - sx, ry - sy) < 25:
                        return True
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    positions = scene["positions"]
    edges = scene["edges"]
    candidate_colors = scene["candidate_colors"]
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    points = [(x * scale, y * scale) for x, y in positions]
    for a, b in edges:
        draw.line((points[a], points[b]), fill=COLORS["edge"], width=9 * scale)

    candidate_map = dict(zip(CANDIDATE_NODES, candidate_colors))
    for index, (x, y) in enumerate(points):
        if index in (0, 1):
            half = 18 * scale
            draw.rounded_rectangle((x - half, y - half, x + half, y + half), radius=3 * scale,
                                   fill=COLORS["yellow"], outline=COLORS["outline"], width=4 * scale)
        else:
            radius = 16 * scale
            fill = COLORS[candidate_map[index]] if index in candidate_map else COLORS["cyan"]
            draw.ellipse((x - radius, y - radius, x + radius, y + radius),
                         fill=fill, outline=COLORS["outline"], width=4 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
