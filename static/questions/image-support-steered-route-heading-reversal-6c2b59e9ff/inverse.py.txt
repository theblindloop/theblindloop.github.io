from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_TRACE = np.array((23, 52, 93), dtype=np.int16)
_START = np.array((24, 165, 88), dtype=np.int16)
_FINISH = np.array((220, 63, 69), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    yy, xx = np.nonzero(mask)
    if not 40 <= len(xx) <= 120:
        raise ValueError(f"missing or malformed {name} marker")
    return float(xx.mean()), float(yy.mean())


def _components(mask: np.ndarray) -> int:
    unseen = {tuple(p) for p in np.argwhere(mask).tolist()}
    count = 0
    while unseen:
        count += 1
        stack = [unseen.pop()]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    p = (y + dy, x + dx)
                    if p in unseen:
                        unseen.remove(p)
                        stack.append(p)
    return count


def _nearest(points: np.ndarray, center: tuple[float, float]) -> tuple[int, int]:
    delta = points.astype(float) - np.array((center[1], center[0]))
    y, x = points[int(np.argmin(np.sum(delta * delta, axis=1)))]
    return int(y), int(x)


def _shortest_path(mask: np.ndarray, start: tuple[int, int], finish: tuple[int, int]):
    queue = deque([start])
    parent = {start: None}
    while queue:
        p = queue.popleft()
        if p == finish:
            break
        y, x = p
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dx or dy):
                    continue
                q = (y + dy, x + dx)
                if (0 <= q[0] < mask.shape[0] and 0 <= q[1] < mask.shape[1]
                        and mask[q] and q not in parent):
                    parent[q] = p
                    queue.append(q)
    if finish not in parent:
        raise ValueError("navy centerline is broken")
    path = []
    p = finish
    while p is not None:
        path.append((float(p[1]), float(p[0])))
        p = parent[p]
    return list(reversed(path))


def _distance_to_segment(point, a, b) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    d = vx * vx + vy * vy
    if d == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / d))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def _rdp(points, epsilon: float):
    if len(points) <= 2:
        return points
    ds = [_distance_to_segment(p, points[0], points[-1]) for p in points[1:-1]]
    if not ds or max(ds) <= epsilon:
        return [points[0], points[-1]]
    i = 1 + int(np.argmax(ds))
    left = _rdp(points[:i + 1], epsilon)
    right = _rdp(points[i:], epsilon)
    return left[:-1] + right


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("unsupported raster")
    start_center = _centroid(_mask(array, _START), "green")
    finish_center = _centroid(_mask(array, _FINISH), "red")
    trace = _mask(array, _TRACE)
    points = np.argwhere(trace)
    if not 250 <= len(points) <= 900 or _components(trace) != 1:
        raise ValueError("missing, broken, or extraneous navy centerline")
    ordered = _shortest_path(trace, _nearest(points, start_center), _nearest(points, finish_center))
    # A long first chord estimates the initial heading well inside its 70-pixel
    # run. Shorter scan chords still fit wholly inside the later 30-pixel
    # antiparallel run while suppressing one-pixel raster staircases.
    first_lag = 20
    lag = 8
    if len(ordered) < 60:
        raise ValueError("route is too short to trace")
    first = (ordered[first_lag][0] - ordered[0][0],
             ordered[first_lag][1] - ordered[0][1])
    first_norm = math.hypot(*first)
    opposition = []
    for index in range(first_lag, len(ordered) - lag, 2):
        vector = (ordered[index + lag][0] - ordered[index][0],
                  ordered[index + lag][1] - ordered[index][1])
        norm = math.hypot(*vector)
        if norm:
            opposition.append(-(first[0] * vector[0] + first[1] * vector[1]) /
                              (first_norm * norm))
    best = max(opposition)
    if abs(best - 0.98) < 0.005:
        raise ValueError("ambiguous heading comparison")
    return "yes" if best > 0.98 else "no"
