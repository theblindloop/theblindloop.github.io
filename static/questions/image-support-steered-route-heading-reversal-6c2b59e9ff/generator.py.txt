from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = "#fcfcf8"
HALO = "#a1beda"
TRACE = "#17345d"
START = "#18a558"
FINISH = "#dc3f45"
WHITE = "#ffffff"


def _transform(points: list[tuple[int, int]], rotation: int, reflect: bool,
               center: tuple[int, int]) -> list[list[int]]:
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    ox = (min(xs) + max(xs)) / 2.0
    oy = (min(ys) + max(ys)) / 2.0
    out: list[list[int]] = []
    for px, py in points:
        x, y = px - ox, py - oy
        if reflect:
            y = -y
        for _ in range(int(rotation) % 4):
            x, y = -y, x
        out.append([int(round(center[0] + x)), int(round(center[1] + y))])
    return out


def _template(decision: str, variant: int) -> list[tuple[int, int]]:
    # Both classes have five runs, the same endpoint displacement and the same
    # slender 108x24 footprint.  The positive class contains one exact westward
    # run after an eastward initial run; negative back-slants are visibly oblique.
    if decision == "no":
        raw = [(0, 0), (70, 0), (40, -12), (70, -24), (90, -12), (108, -24)]
    else:
        raw = [(0, 0), (70, 0), (70, -12), (40, -12), (70, -24), (108, -24)]
    if int(variant) % 2:
        # Reverse nuisance ordering of the two perpendicular run lengths without
        # changing the heading-reversal decision or any aggregate geometry.
        raw = [(x, -y) for x, y in raw]
    return raw


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    decision = "yes" if int(seed) % 2 == 0 else "no"
    # Grid-cell centers vary broadly while keeping this very thin trace away
    # from deterministic ablation seams in either quarter-turn orientation.
    centers = (96, 160, 224, 288, 352, 416)
    center = (rng.choice(centers), rng.choice(centers))
    points = _transform(_template(decision, rng.randrange(2)), rng.randrange(4),
                        bool(rng.randrange(2)), center)
    return {
        "width": 512,
        "height": 512,
        "points": points,
        "stroke": rng.choice((3, 4)),
        "route_color": TRACE,
        "start_color": START,
        "finish_color": FINISH,
        "background": BACKGROUND,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), scene["background"])
    draw = ImageDraw.Draw(image)
    points = [tuple(map(int, p)) for p in scene["points"]]
    stroke = int(scene["stroke"])
    draw.line(points, fill=HALO, width=stroke + 6, joint="curve")
    draw.line(points, fill=scene["route_color"], width=stroke, joint="curve")
    for point, color in ((points[0], scene["start_color"]), (points[-1], scene["finish_color"])):
        x, y = point
        draw.ellipse((x - 6, y - 6, x + 6, y + 6), fill=WHITE)
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=color)
    return image


def _opposition_scores(scene: dict) -> list[float]:
    pts = [tuple(map(float, p)) for p in scene["points"]]
    vectors = [(b[0] - a[0], b[1] - a[1]) for a, b in zip(pts, pts[1:])]
    first = vectors[0]
    scores = []
    for vector in vectors[1:]:
        dot = first[0] * vector[0] + first[1] * vector[1]
        scores.append(-dot / (math.hypot(*first) * math.hypot(*vector)))
    return scores


def analytic_gold(scene: dict) -> str:
    return "yes" if max(_opposition_scores(scene)) > 0.98 else "no"


def margin(scene: dict) -> float:
    # Runner-up distance from the declared cosine decision threshold.
    return round(abs(max(_opposition_scores(scene)) - 0.98), 12)


def is_quarantined(scene: dict) -> bool:
    points = [tuple(map(int, p)) for p in scene["points"]]
    lengths = [math.dist(a, b) for a, b in zip(points, points[1:])]
    edge = min(min(x, y, int(scene["width"]) - 1 - x, int(scene["height"]) - 1 - y)
               for x, y in points)
    return margin(scene) < 0.015 or min(lengths) < 10 or edge < 24


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["route_color"] = str(scene["route_color"]).upper()
    return [("hex_case_alias", transformed)]
