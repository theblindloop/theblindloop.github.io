"""Pixel-only inverse arm for rotary-dial cyan aperture count."""

from __future__ import annotations

from functools import lru_cache

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        y0 = int(y0)
        x0 = int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        count = 0
        min_x = max_x = x0
        min_y = max_y = y0
        while stack:
            y, x = stack.pop()
            count += 1
            min_x = min(min_x, x)
            max_x = max(max_x, x)
            min_y = min(min_y, y)
            max_y = max(max_y, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        found.append((count, min_x, min_y, max_x, max_y))
    return found


@lru_cache(maxsize=256)
def _decision_from_rgb(size: tuple[int, int], pixels: bytes) -> str:
    arr = np.frombuffer(pixels, dtype=np.uint8).reshape(size[1], size[0], 3)
    red, green, blue = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    cyan = (red < 85) & (green > 135) & (blue > 145) & ((green.astype(int) + blue.astype(int) - 2 * red.astype(int)) > 190)
    accepted = []
    for area, min_x, min_y, max_x, max_y in _components(cyan):
        w = max_x - min_x + 1
        h = max_y - min_y + 1
        fill = area / float(w * h)
        if 350 <= area <= 1250 and 22 <= w <= 44 and 22 <= h <= 44 and 0.62 <= fill <= 0.90 and abs(w - h) <= 3:
            accepted.append((area, min_x, min_y, max_x, max_y))
    count = len(accepted)
    if count not in (8, 10, 12):
        raise ValueError(f"ambiguous cyan aperture count: {count}")
    return str(count)


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    return _decision_from_rgb(rgb.size, rgb.tobytes())
