"""Deterministic rotary-telephone world with a visible aperture count."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
COUNTS = (8, 10, 12)
PALETTES = (
    {"bg": "#e8f0ee", "body": "#f3c867", "dark": "#18334b", "hole": "#18b7c9", "shade": "#dcae4b"},
    {"bg": "#eceaf2", "body": "#efb86b", "dark": "#293650", "hole": "#20a9d0", "shade": "#d39a48"},
    {"bg": "#e4efec", "body": "#eecb73", "dark": "#243d47", "hole": "#21b6a8", "shade": "#d3ad50"},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    count = COUNTS[int(seed) % len(COUNTS)]
    # Constant target-color area suppresses answer leakage through aggregate ink.
    radius = round(math.sqrt(2722.5 / count), 4)
    base = rng.uniform(0.0, 360.0 / count)
    return {
        "hole_count": count,
        "hole_radius": radius,
        "palette_id": rng.randrange(len(PALETTES)),
        "rotation_deg": round(base, 4),
        "x_shift": rng.randint(-14, 14),
        "y_shift": rng.randint(-8, 10),
        "screw_angles": [round((base + 37.0 + i * 180.0) % 360.0, 4) for i in range(2)],
    }


def _palette(scene: dict) -> dict:
    return PALETTES[int(scene["palette_id"]) % len(PALETTES)]


def _geometry(scene: dict) -> tuple[float, float, float, float, list[tuple[float, float]]]:
    count = int(scene["hole_count"])
    if count not in COUNTS:
        raise ValueError("hole_count must be 8, 10, or 12")
    radius = float(scene["hole_radius"])
    cx = 256.0 + float(scene["x_shift"])
    cy = 291.0 + float(scene["y_shift"])
    ring_radius = 88.0
    phase = math.radians(float(scene["rotation_deg"]))
    centers = []
    for i in range(count):
        angle = phase + 2.0 * math.pi * i / count - math.pi / 2.0
        centers.append((cx + ring_radius * math.cos(angle), cy + ring_radius * math.sin(angle)))
    return cx, cy, ring_radius, radius, centers


def _S(value: float) -> int:
    return int(round(value * 3))


def _ellipse(draw: ImageDraw.ImageDraw, box, *, fill, outline=None, width=1) -> None:
    draw.ellipse(tuple(_S(v) for v in box), fill=fill, outline=outline, width=_S(width))


def _rounded(draw: ImageDraw.ImageDraw, box, radius, *, fill, outline=None, width=1) -> None:
    draw.rounded_rectangle(tuple(_S(v) for v in box), radius=_S(radius), fill=fill, outline=outline, width=_S(width))


def render(scene: dict) -> Image.Image:
    palette = _palette(scene)
    cx, cy, ring_radius, hole_radius, centers = _geometry(scene)
    canvas = Image.new("RGB", (WIDTH * 3, HEIGHT * 3), palette["bg"])
    draw = ImageDraw.Draw(canvas)

    # Matched telephone housing and shadow.
    _rounded(draw, (73, 101, 439, 467), 48, fill=palette["dark"])
    _rounded(draw, (80, 94, 432, 456), 45, fill=palette["body"], outline=palette["dark"], width=4)
    _rounded(draw, (102, 121, 410, 177), 25, fill=palette["shade"], outline=palette["dark"], width=3)

    # Handset silhouette supplies the familiar-object prior without touching the dial evidence.
    _rounded(draw, (83, 57, 429, 127), 28, fill=palette["dark"])
    _rounded(draw, (139, 72, 373, 112), 18, fill=palette["body"])
    _rounded(draw, (88, 52, 154, 137), 25, fill=palette["dark"])
    _rounded(draw, (358, 52, 424, 137), 25, fill=palette["dark"])

    # Annular rotary dial.
    _ellipse(draw, (cx - 132, cy - 132, cx + 132, cy + 132), fill=palette["dark"])
    _ellipse(draw, (cx - 116, cy - 116, cx + 116, cy + 116), fill=palette["bg"], outline=palette["dark"], width=3)
    _ellipse(draw, (cx - 56, cy - 56, cx + 56, cy + 56), fill=palette["body"], outline=palette["dark"], width=4)

    # Counted apertures: separated target-color disks with constant total nominal area.
    for hx, hy in centers:
        _ellipse(
            draw,
            (hx - hole_radius, hy - hole_radius, hx + hole_radius, hy + hole_radius),
            fill=palette["hole"],
            outline=palette["dark"],
            width=2,
        )

    # Gray screws are distractors; their list order is a tested pixel-identical alias.
    for degrees in scene["screw_angles"]:
        a = math.radians(float(degrees))
        sx = cx + 43.0 * math.cos(a)
        sy = cy + 43.0 * math.sin(a)
        _ellipse(draw, (sx - 4, sy - 4, sx + 4, sy + 4), fill="#8d969c", outline=palette["dark"], width=1)

    # Short center label, visually distinct from cyan evidence.
    font = ImageFont.load_default(size=18)
    label = "DIAL"
    box = draw.textbbox((0, 0), label, font=font)
    tx = _S(cx) - (box[2] - box[0]) // 2
    ty = _S(cy) - (box[3] - box[1]) // 2
    draw.text((tx, ty), label, font=font, fill=palette["dark"])

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict) -> float:
    _, _, ring_radius, radius, _ = _geometry(scene)
    count = int(scene["hole_count"])
    clearance = 2.0 * ring_radius * math.sin(math.pi / count) - 2.0 * radius
    return float(min(radius - 12.0, clearance - 8.0))


def is_quarantined(scene: dict) -> bool:
    _, _, ring_radius, radius, _ = _geometry(scene)
    count = int(scene["hole_count"])
    clearance = 2.0 * ring_radius * math.sin(math.pi / count) - 2.0 * radius
    return radius < 12.0 or clearance < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["screw_angles"] = list(reversed(scene["screw_angles"]))
    return [("reverse_separated_screw_draw_order", transformed)]
