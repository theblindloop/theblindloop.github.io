"""Independent image-only inverse for the bilateral-glyph search.

This module intentionally knows only the public raster layout: four equal tile
boxes, dark glyph ink, and the reflection rule in the question.  It does not
import any candidate renderer or latent-scene code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANVAS_SIZE = (760, 580)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (20, 20, 370, 275),
    (390, 20, 740, 275),
    (20, 305, 370, 560),
    (390, 305, 740, 560),
)


class OracleAmbiguous(ValueError):
    """Raised when the raster does not expose a unique measurable glyph."""


def _one_component(mask: np.ndarray) -> bool:
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return False
    start = (int(ys[0]), int(xs[0]))
    seen = {start}
    queue: deque[tuple[int, int]] = deque([start])
    height, width = mask.shape
    while queue:
        y, x = queue.popleft()
        for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            ny, nx = y + dy, x + dx
            point = (ny, nx)
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and point not in seen:
                seen.add(point)
                queue.append(point)
    return len(seen) == int(mask.sum())


def _candidate_mask(arr: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    # The public glyph is near-black; panel rules and labels are intentionally
    # lighter.  The inset also removes the panel border from consideration.
    crop = arr[y0 + 35 : y1 - 10, x0 + 45 : x1 - 10]
    dark = np.max(crop, axis=2) < 80
    ys, xs = np.nonzero(dark)
    if len(ys) == 0:
        raise OracleAmbiguous("tile contains no dark glyph pixels")
    mask = dark[int(ys.min()) : int(ys.max()) + 1, int(xs.min()) : int(xs.max()) + 1]
    if not _one_component(mask):
        raise OracleAmbiguous("tile does not contain one connected glyph")
    return mask


def _residual(mask: np.ndarray) -> int:
    return int(np.count_nonzero(mask != np.fliplr(mask)))


def decision_from_image(image: Image.Image) -> str:
    """Measure every tile and return the label with the smallest residual."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    scores = [_residual(_candidate_mask(arr, box)) for box in PANEL_BOXES]
    best = min(scores)
    if scores.count(best) != 1:
        raise OracleAmbiguous("reflection residual has no unique winner")
    return LABELS[scores.index(best)]
