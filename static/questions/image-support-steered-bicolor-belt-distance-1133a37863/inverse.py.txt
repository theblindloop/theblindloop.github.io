"""Independent image-only inverse arm for bicolor belt distance."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((38, 188, 201), dtype=np.int16)
ORANGE = np.array((239, 143, 37), dtype=np.int16)
BACKGROUND = np.array((246, 244, 236), dtype=np.int16)


def _groups(indices: np.ndarray) -> list[np.ndarray]:
    if not len(indices):
        return []
    cuts = np.where(np.diff(indices) > 1)[0] + 1
    return [group for group in np.split(indices, cuts) if len(group)]


def _connected(mask: np.ndarray) -> bool:
    points = np.argwhere(mask)
    if not len(points):
        return False
    start = tuple(int(v) for v in points[0])
    seen = {start}
    queue = deque([start])
    height, width = mask.shape
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            point = (ny, nx)
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and point not in seen:
                seen.add(point)
                queue.append(point)
    return len(seen) == int(mask.sum())


def _patch_color(cyan: np.ndarray, orange: np.ndarray, x: float, y: float) -> str:
    xi, yi = int(round(x)), int(round(y))
    y0, y1 = max(0, yi - 6), min(cyan.shape[0], yi + 7)
    x0, x1 = max(0, xi - 6), min(cyan.shape[1], xi + 7)
    c = int(cyan[y0:y1, x0:x1].sum())
    o = int(orange[y0:y1, x0:x1].sum())
    if max(c, o) < 80 or abs(c - o) < 50:
        raise ValueError("segment color is missing or ambiguous")
    return "cyan" if c > o else "orange"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[0] < 128 or array.shape[1] < 128:
        raise ValueError("image is too small")
    cyan = np.max(np.abs(array - CYAN), axis=2) <= 2
    orange = np.max(np.abs(array - ORANGE), axis=2) <= 2
    interior = cyan | orange
    if int(cyan.sum()) < 10000 or int(orange.sum()) < 10000:
        raise ValueError("both pavement colors must be substantial")
    if not _connected(interior):
        raise ValueError("colored belt interior is not one connected component")

    foreground = np.linalg.norm(array.astype(np.float32) - BACKGROUND, axis=2) > 30.0
    height, width = foreground.shape
    tile_fractions = []
    for gy in range(8):
        y0, y1 = height * gy // 8, height * (gy + 1) // 8
        for gx in range(8):
            x0, x1 = width * gx // 8, width * (gx + 1) // 8
            tile_fractions.append(float(foreground[y0:y1, x0:x1].mean()))
    if min(tile_fractions) < 0.46:
        raise ValueError("a distant belt tile is missing; full trace is not recoverable")

    projection = interior.sum(axis=1)
    row_groups = _groups(np.flatnonzero(projection > 0.58 * width))
    if len(row_groups) != 5:
        raise ValueError(f"expected five horizontal belt runs, found {len(row_groups)}")
    centers = [float(group.mean()) for group in row_groups]
    thickness = float(np.median([len(group) for group in row_groups]))
    if not 56.0 <= thickness <= 70.0:
        raise ValueError("belt width is outside the declared uniform range")

    signed = 0.0
    left_centers = []
    right_centers = []
    for y in centers:
        yi = int(round(y))
        xs = np.flatnonzero(interior[yi])
        if len(xs) < 350:
            raise ValueError("horizontal belt run is incomplete")
        left = float(xs.min()) + thickness / 2.0
        right = float(xs.max()) - thickness / 2.0
        if right - left < 350.0:
            raise ValueError("horizontal centerline span is too short")
        left_centers.append(left)
        right_centers.append(right)
        color = _patch_color(cyan, orange, (left + right) / 2.0, y)
        length = right - left
        signed += length if color == "cyan" else -length

    for index in range(4):
        y = (centers[index] + centers[index + 1]) / 2.0
        left = (left_centers[index] + left_centers[index + 1]) / 2.0
        right = (right_centers[index] + right_centers[index + 1]) / 2.0
        candidates = []
        for x in (left, right):
            try:
                candidates.append((_patch_color(cyan, orange, x, y), x))
            except ValueError:
                pass
        if len(candidates) != 1:
            raise ValueError("vertical connector ownership is ambiguous")
        color = candidates[0][0]
        length = centers[index + 1] - centers[index]
        signed += length if color == "cyan" else -length

    if abs(signed) < 1.0:
        raise ValueError("pixel runner-up margin is too small")
    return "cyan" if signed > 0 else "orange"
