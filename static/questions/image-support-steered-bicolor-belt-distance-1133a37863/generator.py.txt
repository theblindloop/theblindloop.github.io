"""Deterministic scene sampler and renderer for the bicolor belt world."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (246, 244, 236)
NAVY = (27, 45, 74)
CYAN = (38, 188, 201)
ORANGE = (239, 143, 37)
GREEN = (42, 165, 92)
RED = (213, 63, 58)
OUTER_WIDTH = 88.0
INNER_WIDTH = 64.0
QUARANTINE_MARGIN = 80.0


def _points(scene: dict) -> list[tuple[float, float]]:
    left = float(scene["x_left"])
    right = float(scene["x_right"])
    ys = [float(value) for value in scene["ys"]]
    begin_left = bool(scene["start_from_left"])
    points: list[tuple[float, float]] = []
    for index, y in enumerate(ys):
        starts_left = begin_left if index % 2 == 0 else not begin_left
        a, b = (left, right) if starts_left else (right, left)
        if not points:
            points.append((a, y))
        points.append((b, y))
        if index + 1 < len(ys):
            points.append((b, ys[index + 1]))
    return points


def _segment_lengths(scene: dict) -> list[float]:
    horizontal = abs(float(scene["x_right"]) - float(scene["x_left"]))
    ys = [float(value) for value in scene["ys"]]
    lengths: list[float] = []
    for index, y in enumerate(ys):
        lengths.append(horizontal)
        if index + 1 < len(ys):
            lengths.append(abs(ys[index + 1] - y))
    return lengths


def _signed_length(scene: dict) -> float:
    colors = scene["colors"]
    lengths = _segment_lengths(scene)
    return sum(length if color == "cyan" else -length for color, length in zip(colors, lengths))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    x_left = rng.randint(43, 48)
    x_right = rng.randint(464, 469)
    base = rng.randint(52, 57)
    gaps = [rng.randint(97, 101) for _ in range(4)]
    ys = [base]
    for gap in gaps:
        ys.append(ys[-1] + gap)
    start_from_left = bool(rng.randrange(2))
    lengths = []
    for index in range(9):
        lengths.append(float(x_right - x_left) if index % 2 == 0 else float(gaps[index // 2]))

    desired = "cyan" if int(seed) % 2 == 0 else "orange"
    want_near = int(seed) % 6 == 0
    chosen = None
    fallback = None
    for _ in range(20000):
        colors = ["cyan" if rng.randrange(2) else "orange" for _ in lengths]
        if len(set(colors)) < 2:
            continue
        signed = sum(length if color == "cyan" else -length for color, length in zip(colors, lengths))
        if (signed > 0) != (desired == "cyan"):
            continue
        absolute = abs(signed)
        if fallback is None or absolute < fallback[0]:
            fallback = (absolute, colors)
        if want_near and absolute <= QUARANTINE_MARGIN:
            chosen = colors
            break
        if not want_near and 220.0 <= absolute <= 1050.0:
            chosen = colors
            break
    if chosen is None:
        if fallback is None:
            raise RuntimeError("could not construct a balanced color assignment")
        chosen = fallback[1]
    return {
        "colors": chosen,
        "x_left": x_left,
        "x_right": x_right,
        "ys": ys,
        "start_from_left": start_from_left,
    }


def _distance_and_owner(points: list[tuple[float, float]]) -> tuple[np.ndarray, np.ndarray]:
    yy, xx = np.mgrid[0:HEIGHT, 0:WIDTH]
    distances = []
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        if y0 == y1:
            lo, hi = sorted((x0, x1))
            dx = np.maximum(np.maximum(lo - xx, 0.0), xx - hi)
            dy = np.abs(yy - y0)
        else:
            lo, hi = sorted((y0, y1))
            dx = np.abs(xx - x0)
            dy = np.maximum(np.maximum(lo - yy, 0.0), yy - hi)
        distances.append(dx * dx + dy * dy)
    stack = np.stack(distances, axis=0)
    return np.sqrt(stack.min(axis=0)), stack.argmin(axis=0)


def render(scene: dict) -> Image.Image:
    points = _points(scene)
    distance, owner = _distance_and_owner(points)
    array = np.empty((HEIGHT, WIDTH, 3), dtype=np.uint8)
    array[:] = BACKGROUND
    array[distance <= OUTER_WIDTH / 2.0] = NAVY
    colors = scene["colors"]
    inner = distance <= INNER_WIDTH / 2.0
    for index, color in enumerate(colors):
        array[inner & (owner == index)] = CYAN if color == "cyan" else ORANGE
    image = Image.fromarray(array, mode="RGB")
    draw = ImageDraw.Draw(image)
    for point, color in ((points[0], GREEN), (points[-1], RED)):
        x, y = point
        box = (x - 18, y - 18, x + 18, y + 18)
        draw.ellipse(box, fill=color)
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=(255, 255, 255))
    return image


def analytic_gold(scene: dict) -> str:
    signed = _signed_length(scene)
    return "cyan" if signed > 0 else "orange"


def margin(scene: dict) -> float:
    return abs(_signed_length(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
