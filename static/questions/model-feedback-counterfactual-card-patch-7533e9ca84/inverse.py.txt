"""Independent pixel-only inverse for the counterfactual card-patch board."""
from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (920, 700)
BOARD_W, BOARD_H = 880, 660
BORDER = (55, 64, 72)
PATCH = (204, 43, 55)
MUTED = (100, 108, 114)
REGISTERS = ("A", "B", "C", "D")
BADGES = {(198, 82, 52): "A", (22, 139, 150): "B", (121, 75, 180): "C", (65, 104, 174): "D"}
SHAPES = {(42, 124, 202): "circle", (38, 157, 91): "triangle", (230, 158, 26): "star", (205, 69, 119): "diamond"}
DIFF_LABELS = {"A-B-C", "A-B-D", "A-C-D", "B-C-D"}


class OracleError(ValueError):
    pass


def _patch(array: np.ndarray, x: int, y: int, radius: int = 2) -> tuple[int, int, int]:
    patch = array[y - radius : y + radius + 1, x - radius : x + radius + 1]
    colors = {tuple(int(v) for v in row) for row in patch.reshape(-1, 3)}
    if len(colors) != 1:
        raise OracleError("a required marker is damaged")
    return next(iter(colors))


def _shape(array: np.ndarray, x: int, y: int) -> str:
    color = _patch(array, x, y)
    if color not in SHAPES:
        raise OracleError("unknown shape marker")
    return SHAPES[color]


def _badge(array: np.ndarray, cx: int, y: int, large: bool = False) -> str:
    color = _patch(array, cx - (18 if large else 14), y)
    if color not in BADGES:
        raise OracleError("unknown register badge")
    return BADGES[color]


def _run(initial: list[str], cards: list[tuple], patch_card: int | None, patch_swap: tuple[str, str] | None) -> tuple[list[str], list[bool]]:
    state = list(initial)
    choices: list[bool] = []
    for index, (test_slot, test_shape, yes, no) in enumerate(cards):
        hit = state[REGISTERS.index(test_slot)] == test_shape
        pair = patch_swap if index == patch_card and hit else (yes if hit else no)
        i, j = REGISTERS.index(pair[0]), REGISTERS.index(pair[1])
        state[i], state[j] = state[j], state[i]
        choices.append(hit)
    return state, choices


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if (array.shape[1], array.shape[0]) != CANVAS:
        raise OracleError("unexpected canvas size")
    mask = np.all(array == np.asarray(BORDER, dtype=np.uint8), axis=2)
    points = np.argwhere(mask)
    if not len(points):
        raise OracleError("board absent")
    top, left = points.min(axis=0); bottom, right = points.max(axis=0)
    if int(right - left + 1) != BOARD_W or int(bottom - top + 1) != BOARD_H:
        raise OracleError("wrong board extent")
    ax, ay = int(left), int(top)
    if not (
        np.all(mask[ay, ax : ax + BOARD_W]) and np.all(mask[ay + BOARD_H - 1, ax : ax + BOARD_W])
        and np.all(mask[ay : ay + BOARD_H, ax]) and np.all(mask[ay : ay + BOARD_H, ax + BOARD_W - 1])
    ):
        raise OracleError("broken board border")
    initial = [_shape(array, ax + x, ay + 128) for x in (110, 330, 550, 770)]
    if sorted(initial) != sorted(SHAPES.values()):
        raise OracleError("invalid start inventory")
    indicators = [_patch(array, ax + 275 + 31 * i, ay + 223) for i in range(6)]
    selected = [i for i, color in enumerate(indicators) if color == PATCH]
    if len(selected) != 1 or any(color not in {PATCH, MUTED} for color in indicators):
        raise OracleError("malformed patch-card selector")
    patch_card = selected[0]
    if patch_card not in range(1, 4):
        raise OracleError("patch card outside supported range")
    patch_swap = (_badge(array, ax + 610, ay + 210), _badge(array, ax + 690, ay + 210))
    if REGISTERS.index(patch_swap[0]) >= REGISTERS.index(patch_swap[1]):
        raise OracleError("malformed patch swap")
    cards = []
    card_xs, card_ys = (25, 315, 605), (285, 455)
    for index in range(6):
        x, y = ax + card_xs[index % 3], ay + card_ys[index // 3]
        test_slot = _badge(array, x + 93, y + 49)
        test_shape = _shape(array, x + 165, y + 49)
        yes = (_badge(array, x + 105, y + 94), _badge(array, x + 190, y + 94))
        no = (_badge(array, x + 105, y + 132), _badge(array, x + 190, y + 132))
        if REGISTERS.index(yes[0]) >= REGISTERS.index(yes[1]) or REGISTERS.index(no[0]) >= REGISTERS.index(no[1]) or yes == no:
            raise OracleError("malformed branch swap")
        cards.append((test_slot, test_shape, yes, no))
    if patch_swap == cards[patch_card][2]:
        raise OracleError("patch changes nothing")
    factual, factual_choices = _run(initial, cards, None, None)
    patched, patched_choices = _run(initial, cards, patch_card, patch_swap)
    changed = [REGISTERS[i] for i in range(4) if factual[i] != patched[i]]
    later_divergence = sum(
        factual_choices[i] != patched_choices[i] for i in range(patch_card + 1, 6)
    )
    if not factual_choices[patch_card] or len(changed) != 3 or later_divergence < 2:
        raise OracleError("patch is not causally deep and uniquely answerable")
    answer = "-".join(changed)
    if answer not in DIFF_LABELS:
        raise OracleError("unknown answer triple")
    return answer
