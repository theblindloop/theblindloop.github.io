"""Deterministic renderer for factual versus instruction-patched swap programs."""
from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw, ImageFont

CANVAS = (920, 700)
BOARD_W, BOARD_H = 880, 660
REGISTERS = ("A", "B", "C", "D")
SHAPES = ("circle", "triangle", "star", "diamond")
PAIRS = tuple((a, b) for i, a in enumerate(REGISTERS) for b in REGISTERS[i + 1 :])
DIFF_TRIPLES = tuple(tuple(r for r in REGISTERS if r != omitted) for omitted in reversed(REGISTERS))
DIFF_LABELS = tuple("-".join(triple) for triple in DIFF_TRIPLES)
RGB = {
    "white": (255, 255, 255), "board": (246, 248, 244), "card": (255, 255, 252),
    "ink": (37, 43, 49), "muted": (100, 108, 114), "border": (55, 64, 72),
    "patch": (204, 43, 55), "yes": (31, 133, 83), "no": (185, 73, 48),
    "A": (198, 82, 52), "B": (22, 139, 150), "C": (121, 75, 180), "D": (65, 104, 174),
    "circle": (42, 124, 202), "triangle": (38, 157, 91),
    "star": (230, 158, 26), "diamond": (205, 69, 119),
}
FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _run(initial: list[str], cards: list[dict], patch_card: int | None, patch_swap: list[str] | None) -> tuple[list[str], list[bool]]:
    state = list(initial)
    choices: list[bool] = []
    for index, card in enumerate(cards):
        hit = state[REGISTERS.index(card["test_slot"])] == card["test_shape"]
        pair = patch_swap if index == patch_card and hit else (card["yes"] if hit else card["no"])
        i, j = REGISTERS.index(pair[0]), REGISTERS.index(pair[1])
        state[i], state[j] = state[j], state[i]
        choices.append(hit)
    return state, choices


def _worlds(scene: dict) -> tuple[list[str], list[str], list[bool], list[bool]]:
    initial = [str(v) for v in scene["initial"]]
    cards = [dict(c) for c in scene["cards"]]
    patch_card = int(scene["patch_card"])
    patch_swap = [str(v) for v in scene["patch_swap"]]
    factual, factual_choices = _run(initial, cards, None, None)
    patched, patched_choices = _run(initial, cards, patch_card, patch_swap)
    return factual, patched, factual_choices, patched_choices


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xC0FA71C4) & 0xFFFFFFFFFFFFFFFF)
    wanted = DIFF_TRIPLES[int(seed) % len(DIFF_TRIPLES)]
    for _ in range(250000):
        initial = list(SHAPES)
        rng.shuffle(initial)
        patch_card = rng.randrange(1, 4)
        cards = []
        for _card in range(6):
            yes = rng.choice(PAIRS)
            no = rng.choice([pair for pair in PAIRS if pair != yes])
            cards.append({
                "test_slot": rng.choice(REGISTERS),
                "test_shape": rng.choice(SHAPES),
                "yes": list(yes),
                "no": list(no),
            })
        original_yes = tuple(cards[patch_card]["yes"])
        patch_swap = list(rng.choice([pair for pair in PAIRS if pair != original_yes]))
        probe = {
            "initial": initial, "cards": cards, "patch_card": patch_card,
            "patch_swap": patch_swap, "anchor_x": 0, "anchor_y": 0,
        }
        factual, patched, factual_choices, patched_choices = _worlds(probe)
        changed = tuple(REGISTERS[i] for i in range(4) if factual[i] != patched[i])
        later_divergence = sum(
            factual_choices[i] != patched_choices[i] for i in range(patch_card + 1, 6)
        )
        if factual_choices[patch_card] and changed == wanted and later_divergence >= 2:
            return {**probe, "anchor_x": rng.randint(18, 22), "anchor_y": rng.randint(18, 22)}
    raise RuntimeError("could not sample a deep, answer-balanced instruction patch")


def _validate(scene: dict) -> None:
    initial = [str(v) for v in scene["initial"]]
    cards = [dict(c) for c in scene["cards"]]
    if len(initial) != 4 or sorted(initial) != sorted(SHAPES) or len(cards) != 6:
        raise ValueError("invalid start row or card count")
    patch_card = int(scene["patch_card"])
    patch_swap = [str(v) for v in scene["patch_swap"]]
    if patch_card not in range(1, 4) or len(patch_swap) != 2 or tuple(patch_swap) not in PAIRS:
        raise ValueError("invalid patch")
    for card in cards:
        if card.get("test_slot") not in REGISTERS or card.get("test_shape") not in SHAPES:
            raise ValueError("invalid predicate")
        yes, no = card.get("yes"), card.get("no")
        if not isinstance(yes, list) or not isinstance(no, list) or len(yes) != 2 or len(no) != 2:
            raise ValueError("invalid branch action")
        if tuple(yes) not in PAIRS or tuple(no) not in PAIRS or yes == no:
            raise ValueError("invalid swap pair")
    if patch_swap == cards[patch_card]["yes"]:
        raise ValueError("patch changes nothing")
    factual, patched, factual_choices, patched_choices = _worlds(scene)
    changed = [i for i in range(4) if factual[i] != patched[i]]
    later_divergence = sum(
        factual_choices[i] != patched_choices[i] for i in range(patch_card + 1, 6)
    )
    if not factual_choices[patch_card] or len(changed) != 3 or later_divergence < 2:
        raise ValueError("patch is not causally deep and uniquely answerable")


def analytic_gold(scene: dict) -> str:
    _validate(scene)
    factual, patched, _, _ = _worlds(scene)
    changed = [REGISTERS[i] for i in range(4) if factual[i] != patched[i]]
    answer = "-".join(changed)
    if answer not in DIFF_LABELS:
        raise ValueError("invalid difference triple")
    return answer


def margin(scene: dict) -> float:
    try:
        return 1.0 if analytic_gold(scene) in DIFF_LABELS else 0.0
    except (KeyError, TypeError, ValueError):
        return 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False):
    return ImageFont.truetype(BOLD if bold else FONT, size)


def _text(draw, xy, value, size, *, fill=None, bold=False, anchor="mm"):
    draw.text(xy, value, font=_font(size, bold), fill=fill or RGB["ink"], anchor=anchor)


def _shape(draw, center, kind: str, scale: int = 18) -> None:
    x, y = center
    color = RGB[kind]
    if kind == "circle":
        draw.ellipse((x - scale, y - scale, x + scale, y + scale), fill=color, outline=RGB["ink"], width=2)
    elif kind == "triangle":
        pts = ((x, y - scale - 2), (x - scale - 2, y + scale), (x + scale + 2, y + scale))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=2)
    elif kind == "diamond":
        pts = ((x, y - scale - 2), (x - scale - 2, y), (x, y + scale + 2), (x + scale + 2, y))
        draw.polygon(pts, fill=color); draw.line(pts + (pts[0],), fill=RGB["ink"], width=2)
    elif kind == "star":
        pts = []
        for i in range(10):
            radius = scale + 2 if i % 2 == 0 else scale * 0.45
            angle = -math.pi / 2 + i * math.pi / 5
            pts.append((x + radius * math.cos(angle), y + radius * math.sin(angle)))
        draw.polygon(pts, fill=color); draw.line(pts + [pts[0]], fill=RGB["ink"], width=2, joint="curve")
    else:
        raise ValueError(kind)


def _badge(draw, center, label: str, small: bool = False) -> None:
    x, y = center
    w, h = ((38, 27) if small else (48, 34))
    draw.rounded_rectangle((x - w // 2, y - h // 2, x + w // 2, y + h // 2), radius=6, fill=RGB[label], outline=RGB["ink"], width=2)
    _text(draw, center, label, 16 if small else 19, fill=RGB["white"], bold=True)


def render(scene: dict) -> Image.Image:
    _validate(scene)
    ax, ay = int(scene["anchor_x"]), int(scene["anchor_y"])
    image = Image.new("RGB", CANVAS, RGB["white"])
    draw = ImageDraw.Draw(image)
    draw.rectangle((ax, ay, ax + BOARD_W - 1, ay + BOARD_H - 1), fill=RGB["board"], outline=RGB["border"], width=4)
    _text(draw, (ax + 440, ay + 24), "ORIGINAL vs PATCHED", 22, bold=True)
    _text(draw, (ax + 440, ay + 50), "Same START for both runs", 16, fill=RGB["muted"], bold=True)
    for label, kind, xoff in zip(REGISTERS, scene["initial"], (110, 330, 550, 770), strict=True):
        x = ax + xoff
        draw.rounded_rectangle((x - 75, ay + 67, x + 75, ay + 157), radius=9, fill=RGB["card"], outline=RGB[label], width=4)
        _badge(draw, (x - 45, ay + 91), label, small=True)
        _shape(draw, (x, ay + 128), str(kind), 20)
    draw.rounded_rectangle((ax + 90, ay + 174, ax + 790, ay + 247), radius=10, fill=(255, 243, 243), outline=RGB["patch"], width=4)
    _text(draw, (ax + 153, ay + 210), "PATCH", 17, fill=RGB["patch"], bold=True)
    _text(draw, (ax + 220, ay + 210), "CARD", 14, bold=True)
    selected = int(scene["patch_card"])
    for i in range(6):
        x = ax + 275 + 31 * i
        fill = RGB["patch"] if i == selected else RGB["white"]
        ink = RGB["white"] if i == selected else RGB["ink"]
        draw.ellipse((x - 12, ay + 190, x + 12, ay + 214), fill=fill, outline=RGB["patch"] if i == selected else RGB["muted"], width=2)
        _text(draw, (x, ay + 202), str(i + 1), 13, fill=ink, bold=True)
        draw.rectangle((x - 8, ay + 221, x + 8, ay + 226), fill=RGB["patch"] if i == selected else RGB["muted"])
    _text(draw, (ax + 500, ay + 210), "YES becomes", 14, bold=True)
    _badge(draw, (ax + 610, ay + 210), str(scene["patch_swap"][0]), small=True)
    _text(draw, (ax + 650, ay + 210), "↔", 22, bold=True)
    _badge(draw, (ax + 690, ay + 210), str(scene["patch_swap"][1]), small=True)
    _text(draw, (ax + 440, ay + 267), "Each card: test current shape, then do YES or NO swap", 16, bold=True)
    card_xs, card_ys = (25, 315, 605), (285, 455)
    for index, card in enumerate(scene["cards"]):
        x, y = ax + card_xs[index % 3], ay + card_ys[index // 3]
        outline = RGB["patch"] if index == selected else RGB["border"]
        draw.rounded_rectangle((x, y, x + 250, y + 150), radius=9, fill=RGB["card"], outline=outline, width=4 if index == selected else 3)
        draw.ellipse((x + 8, y + 8, x + 38, y + 38), fill=RGB["patch"] if index == selected else RGB["ink"])
        _text(draw, (x + 23, y + 23), str(index + 1), 16, fill=RGB["white"], bold=True)
        _text(draw, (x + 54, y + 49), "IF", 15, bold=True)
        _badge(draw, (x + 93, y + 49), str(card["test_slot"]), small=True)
        _text(draw, (x + 126, y + 49), "=", 18, bold=True)
        _shape(draw, (x + 165, y + 49), str(card["test_shape"]), 15)
        _text(draw, (x + 39, y + 94), "YES", 14, fill=RGB["yes"], bold=True)
        _badge(draw, (x + 105, y + 94), str(card["yes"][0]), small=True)
        _text(draw, (x + 145, y + 94), "↔", 22, bold=True)
        _badge(draw, (x + 190, y + 94), str(card["yes"][1]), small=True)
        _text(draw, (x + 39, y + 132), "NO", 14, fill=RGB["no"], bold=True)
        _badge(draw, (x + 105, y + 132), str(card["no"][0]), small=True)
        _text(draw, (x + 145, y + 132), "↔", 22, bold=True)
        _badge(draw, (x + 190, y + 132), str(card["no"][1]), small=True)
    return image
