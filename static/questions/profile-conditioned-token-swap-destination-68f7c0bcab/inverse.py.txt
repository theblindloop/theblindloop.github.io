"""Independent image-only inverse for the visible-token-swaps world."""

from __future__ import annotations

import math

from PIL import Image


SLOT_NAMES = ("NW", "NE", "SW", "SE")
PALETTE = (
    (214, 70, 70),
    (45, 116, 193),
    (41, 151, 96),
    (145, 79, 174),
)
RING = (229, 139, 30)
PANEL_X0 = 20
PANEL_STEP = 250
PANEL_TOP = 90
PANEL_W = 238
PANEL_H = 400
SLOT_CENTERS = ((61, 184), (177, 184), (61, 330), (177, 330))


def _exact_color_points(
    frame: Image.Image, left: int, color: tuple[int, int, int]
) -> list[tuple[int, int]]:
    pixels = frame.load()
    points: list[tuple[int, int]] = []
    x_start = left + 10
    x_stop = left + PANEL_W - 10
    y_start = PANEL_TOP + 45
    y_stop = PANEL_TOP + PANEL_H - 18
    for y in range(y_start, y_stop + 1):
        for x in range(x_start, x_stop + 1):
            if pixels[x, y] == color:
                points.append((x, y))
    return points


def _center_of_token(frame: Image.Image, left: int, color: tuple[int, int, int]) -> tuple[float, float]:
    points = _exact_color_points(frame, left, color)
    if len(points) < 120:
        raise ValueError("panel is missing a colored token")
    return (
        sum(point[0] for point in points) / len(points),
        sum(point[1] for point in points) / len(points),
    )


def _nearest_slot(point: tuple[float, float], left: int) -> int:
    distances = [
        math.hypot(point[0] - (left + cx), point[1] - (PANEL_TOP + cy))
        for cx, cy in SLOT_CENTERS
    ]
    slot = min(range(4), key=distances.__getitem__)
    if distances[slot] > 42.0:
        raise ValueError("token center is too far from every named slot")
    return slot


def _panel_state(frame: Image.Image, left: int) -> list[int]:
    state: list[int | None] = [None, None, None, None]
    for palette_index, color in enumerate(PALETTE):
        center = _center_of_token(frame, left, color)
        slot = _nearest_slot(center, left)
        if state[slot] is not None:
            raise ValueError("two tokens occupy one slot")
        state[slot] = palette_index
    if any(value is None for value in state):
        raise ValueError("panel does not expose all four slots")
    return [int(value) for value in state]


def _ringed_slot(frame: Image.Image, left: int) -> int:
    points = _exact_color_points(frame, left, RING)
    if len(points) < 20:
        raise ValueError("P1 does not expose a ring")
    center = (
        sum(point[0] for point in points) / len(points),
        sum(point[1] for point in points) / len(points),
    )
    return _nearest_slot(center, left)


def _infer_swap(previous: list[int], current: list[int]) -> tuple[int, int]:
    changed = [slot for slot in range(4) if previous[slot] != current[slot]]
    if len(changed) != 2:
        raise ValueError("adjacent panels do not show exactly one two-slot swap")
    left, right = changed
    if previous[left] != current[right] or previous[right] != current[left]:
        raise ValueError("changed slots are not a consistent swap")
    for slot in range(4):
        if slot not in changed and previous[slot] != current[slot]:
            raise ValueError("an undeclared slot changed")
    return left, right


def decision_from_image(image: Image.Image) -> str:
    """Recover the final slot using only the submitted RGB raster."""

    frame = image.convert("RGB")
    if frame.size != (1280, 560):
        raise ValueError(f"unexpected raster size {frame.size}")
    first_left = PANEL_X0
    start_slot = _ringed_slot(frame, first_left)
    states = [
        _panel_state(frame, PANEL_X0 + panel_index * PANEL_STEP)
        for panel_index in range(5)
    ]
    target_color = states[0][start_slot]
    replay = list(states[0])
    for panel_index in range(1, len(states)):
        left, right = _infer_swap(states[panel_index - 1], states[panel_index])
        replay[left], replay[right] = replay[right], replay[left]
    if replay != states[-1]:
        raise ValueError("swap replay does not reach the final panel")
    final_slot = replay.index(target_color)
    return SLOT_NAMES[final_slot]
