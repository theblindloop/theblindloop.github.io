"""Deterministic raster renderer and analytic side for visible token swaps."""

from __future__ import annotations

import copy
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
SLOT_NAMES = ("NW", "NE", "SW", "SE")
CANVAS = (1280, 560)
PANEL_X0 = 20
PANEL_STEP = 250
PANEL_TOP = 90
PANEL_W = 238
PANEL_H = 400
PANEL_ORDER = (1, 2, 3, 4, 5)
SLOT_CENTERS = ((61, 184), (177, 184), (61, 330), (177, 330))
RING_RADIUS = 37

PALETTE = (
    (214, 70, 70),
    (45, 116, 193),
    (41, 151, 96),
    (145, 79, 174),
)
BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
SLOT_FILL = (246, 248, 250)
SLOT_OUTLINE = (194, 202, 210)
INK = (35, 44, 54)
ARROW = (100, 111, 122)
RING = (229, 139, 30)

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _panel_left(index: int) -> int:
    return PANEL_X0 + index * PANEL_STEP


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _replay(initial_slots: list[str], swaps: list[list[int]], count: int | None = None) -> list[str]:
    state = list(initial_slots)
    events = swaps if count is None else swaps[:count]
    for pair in events:
        if len(pair) != 2:
            raise ValueError("every event must name two slots")
        left, right = (int(pair[0]), int(pair[1]))
        if left == right or not (0 <= left < 4 and 0 <= right < 4):
            raise ValueError("invalid swap pair")
        state[left], state[right] = state[right], state[left]
    return state


def sample_scene(seed: int) -> dict:
    """Sample an identity-preserving, balanced five-panel swap sequence."""

    event_rng = random.Random(seed * 1000003 + 17)
    style_rng = random.Random(seed * 1000033 + 31)

    initial_slots = event_rng.sample(list(LABELS), len(LABELS))
    start_slot = event_rng.randrange(4)
    desired_final_slot = (seed * 17 + 5) % 4
    pairs = [(left, right) for left in range(4) for right in range(left + 1, 4)]
    swaps: list[list[int]] = []
    for _ in range(256):
        trial = [list(event_rng.choice(pairs)) for _ in range(4)]
        final_state = _replay(initial_slots, trial)
        target_label = initial_slots[start_slot]
        if final_state.index(target_label) == desired_final_slot:
            swaps = trial
            break
    if not swaps:
        raise AssertionError("could not sample a balanced final destination")

    color_indices = style_rng.sample(list(range(4)), 4)
    radii = [style_rng.choice((19, 20, 21, 22, 23)) for _ in LABELS]
    target_label = initial_slots[start_slot]
    # This nuisance schedule is intentionally independent of the four answer
    # classes; seed modulo four would quarantine one destination class every
    # time the evidence stride is used.
    near_ring = seed % 5 == 0
    if near_ring:
        target_index = LABELS.index(target_label)
        radii[target_index] = 31

    tokens = [
        {
            "label": label,
            "color_index": color_indices[index],
            "radius": radii[index],
        }
        for index, label in enumerate(LABELS)
    ]
    ring_gap = float(RING_RADIUS - radii[LABELS.index(target_label)] - 2)
    scene = {
        "canvas": list(CANVAS),
        "panel_order": list(PANEL_ORDER),
        "slot_centers": [list(point) for point in SLOT_CENTERS],
        "tokens": tokens,
        "initial_slots": initial_slots,
        "swaps": swaps,
        "start_slot": start_slot,
        "ring_gap": round(ring_gap, 6),
    }
    if _replay(initial_slots, swaps).index(target_label) != desired_final_slot:
        raise AssertionError("sampler lost its requested destination")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(16)
    panel_font = _font(16)
    label_font = _font(12)
    token_font = _font(14)
    tiny_font = _font(11)

    draw.text((20, 15), "VISIBLE SWAPS", font=title_font, fill=INK)
    draw.text((20, 40), "Follow the arrows; each step swaps exactly two slots", font=label_font, fill=ARROW)
    draw.text((1010, 22), "P1  →  P5", font=tiny_font, fill=ARROW)

    draw.text((235, 17), "token identity:", font=label_font, fill=INK)
    token_by_label = {token["label"]: token for token in scene["tokens"]}
    for index, label in enumerate(LABELS):
        x = 330 + index * 63
        color = PALETTE[token_by_label[label]["color_index"]]
        draw.ellipse((x - 9, 24, x + 9, 42), fill=color, outline=INK, width=1)
        _centered_text(draw, (x, 33), label, tiny_font, (255, 255, 255))
    draw.ellipse((610, 26, 628, 44), outline=RING, width=2)
    draw.text((637, 27), "ringed slot = start", font=label_font, fill=INK)

    for panel_index, panel_number in enumerate(scene["panel_order"]):
        left = _panel_left(panel_index)
        top = PANEL_TOP
        right = left + PANEL_W
        bottom = top + PANEL_H
        draw.rounded_rectangle(
            (left, top, right, bottom), radius=8, fill=PANEL_FILL, outline=INK, width=2
        )
        draw.text((left + 10, top + 9), f"P{panel_number}", font=panel_font, fill=INK)
        if panel_index < len(scene["panel_order"]) - 1:
            next_left = _panel_left(panel_index + 1)
            arrow_y = top - 23
            draw.line((right + 4, arrow_y, next_left - 10, arrow_y), fill=ARROW, width=2)
            draw.polygon(
                ((next_left - 10, arrow_y), (next_left - 17, arrow_y - 4), (next_left - 17, arrow_y + 4)),
                fill=ARROW,
            )
            _centered_text(draw, ((right + next_left) / 2, arrow_y - 10), "swap", tiny_font, ARROW)

        state = _replay(scene["initial_slots"], scene["swaps"], panel_index)
        for slot_index, (local_x, local_y) in enumerate(scene["slot_centers"]):
            cx = left + local_x
            cy = top + local_y
            draw.rounded_rectangle(
                (cx - 50, cy - 50, cx + 50, cy + 50),
                radius=10,
                fill=SLOT_FILL,
                outline=SLOT_OUTLINE,
                width=1,
            )
            _centered_text(draw, (cx, cy - 40), SLOT_NAMES[slot_index], label_font, ARROW)
            label = state[slot_index]
            token = token_by_label[label]
            radius = token["radius"]
            color = PALETTE[token["color_index"]]
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                fill=color,
                outline=INK,
                width=2,
            )
            _centered_text(draw, (cx, cy), label, token_font, (255, 255, 255))
            if panel_index == 0 and slot_index == scene["start_slot"]:
                draw.ellipse(
                    (cx - RING_RADIUS, cy - RING_RADIUS, cx + RING_RADIUS, cy + RING_RADIUS),
                    outline=RING,
                    width=3,
                )

    return image


def analytic_gold(scene: dict) -> str:
    state = _replay(scene["initial_slots"], scene["swaps"])
    target_label = scene["initial_slots"][scene["start_slot"]]
    return SLOT_NAMES[state.index(target_label)]


def margin(scene: dict) -> float:
    """Visual clearance between the start ring and the marked token."""

    return float(scene["ring_gap"])


def is_quarantined(scene: dict) -> bool:
    return float(scene["ring_gap"]) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["tokens"] = list(reversed(twin["tokens"]))
    return [("token_record_order", twin)]
