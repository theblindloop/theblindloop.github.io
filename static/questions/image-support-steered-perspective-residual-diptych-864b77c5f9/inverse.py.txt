"""Independent raster inverse for the perspective residual diptych.

Only the final RGB image is accepted. Visible colored corner markers determine
two projective coordinate frames; cell interiors are sampled in those frames,
compared symmetrically, and matched to four public binary glyph templates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 7
MARKERS = (
    (232, 62, 70),
    (45, 168, 92),
    (52, 101, 210),
    (240, 184, 48),
)


def _glyph(label: str) -> list[int]:
    values = [0] * (N * N)
    def put(x: int, y: int) -> None:
        values[y * N + x] = 1
    if label == "X":
        for i in range(N):
            put(i, i); put(N - 1 - i, i)
    elif label == "plus":
        for i in range(N):
            put(i, 3); put(3, i)
    elif label == "diamond":
        for y in range(N):
            for x in range(N):
                if abs(x - 3) + abs(y - 3) == 3:
                    put(x, y)
        put(3, 3)
    elif label == "U":
        for y in range(1, 6):
            put(1, y); put(5, y)
        for x in range(2, 5):
            put(x, 5)
    return values


GLYPHS = {name: _glyph(name) for name in ("X", "plus", "diamond", "U")}


def _find_quads(array: np.ndarray) -> list[list[tuple[float, float]]]:
    height, width = array.shape[:2]
    found = [[], []]
    rgb = array.astype(np.float32)
    for target in MARKERS:
        distance = np.sqrt(((rgb - np.asarray(target, dtype=np.float32)) ** 2).sum(axis=2))
        mask = distance < 54
        for side, region in enumerate((np.arange(width) < width / 2, np.arange(width) > width / 2)):
            ys, xs = np.nonzero(mask & region[None, :])
            if len(xs) < 35:
                raise ValueError("missing or damaged corner fiducial")
            if xs.max() - xs.min() > 30 or ys.max() - ys.min() > 30:
                raise ValueError("ambiguous corner fiducial")
            found[side].append((float(xs.mean()), float(ys.mean())))
    return found


def _map_for(quad: list[tuple[float, float]]) -> np.ndarray:
    unit = ((0.0, 0.0), (1.0, 0.0), (1.0, 1.0), (0.0, 1.0))
    matrix, rhs = [], []
    for (u, v), (x, y) in zip(unit, quad):
        matrix.extend(([u, v, 1, 0, 0, 0, -u*x, -v*x],
                       [0, 0, 0, u, v, 1, -u*y, -v*y]))
        rhs.extend((x, y))
    solution = np.linalg.solve(np.asarray(matrix), np.asarray(rhs))
    return np.append(solution, 1.0).reshape(3, 3)


def _point(mapping: np.ndarray, u: float, v: float) -> tuple[int, int]:
    value = mapping @ np.array([u, v, 1.0])
    return round(float(value[0] / value[2])), round(float(value[1] / value[2]))


def _read_panel(array: np.ndarray, quad: list[tuple[float, float]]) -> list[int]:
    mapping = _map_for(quad)
    height, width = array.shape[:2]
    result = []
    lo, span = 0.105, 0.79
    for y in range(N):
        for x in range(N):
            u = lo + span * (x + 0.5) / N
            v = lo + span * (y + 0.5) / N
            px, py = _point(mapping, u, v)
            if not (2 <= px < width - 2 and 2 <= py < height - 2):
                raise ValueError("rectified sample lies outside image")
            patch = array[py-2:py+3, px-2:px+3].astype(float)
            luminance = np.median(0.2126*patch[..., 0] + 0.7152*patch[..., 1] + 0.0722*patch[..., 2])
            if 112 <= luminance <= 172:
                raise ValueError("ambiguous cell shade")
            result.append(1 if luminance < 142 else 0)
    return result


def _measure(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or min(array.shape[:2]) < 100:
        raise ValueError("unsupported raster")
    quads = _find_quads(array)
    first, second = (_read_panel(array, quad) for quad in quads)
    residual = [a ^ b for a, b in zip(first, second)]
    scores = sorted((sum(a != b for a, b in zip(residual, glyph)), name)
                    for name, glyph in GLYPHS.items())
    gap = float(scores[1][0] - scores[0][0])
    if gap <= 1:
        raise ValueError("ambiguous residual template")
    return scores[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return _measure(image)[0]


def decision_margin_from_image(image: Image.Image) -> float:
    """Public diagnostic; classification still consumes only raster pixels."""
    return _measure(image)[1]
