"""Compass-needle heading world.

A compass dial is drawn with a permanently fixed decorative frame: a rim circle,
four identical cardinal tick marks at up/right/down/left, four shorter diagonal
"halfway" marks that draw the decision boundary itself, and a letter ``N`` mark
welded to the top of the rim.  Those frame elements are byte-identical in every
scene, so nothing about them predicts the answer -- they exist to install the
familiar canonical prior "the compass needle points north, and north is up".

Through the centre of the dial runs a needle made of two mirror-image triangles
that share one base.  Exactly one of them is filled with a saturated colour; the
other is only a grey outline.  The decision is the direction the SOLID triangle
points, rounded to the nearest of up/right/down/left.

Latent state z = {heading_deg, needle_color, pip_angles, canvas}.  The needle
silhouette is centrally symmetric and the coloured area is constant, so neither
position, size, total ink, nor colour carries the answer: only which of the two
mirrored halves carries the fill does.

Angle convention: ``heading_deg`` is measured CLOCKWISE FROM UP (the ordinary
compass bearing), so the unit direction in image coordinates is
``(sin a, -cos a)``.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "compass_needle_heading-renderer-0.1.0"

WIDTH = 448
HEIGHT = 448
CENTER = (224.0, 224.0)
RIM_RADIUS = 172.0

# Frame ink: deliberately achromatic so the pixel oracle's saturation test can
# never confuse the frame with the needle fill.
FRAME_GRAY = (60, 60, 60)
PIP_GRAY = (120, 120, 120)
BACKGROUND = (255, 255, 255)

# Saturated needle-fill palette. Colour is sampled independently of the heading.
NEEDLE_PALETTE: tuple[tuple[int, int, int], ...] = (
    (214, 24, 58),    # crimson
    (0, 138, 62),     # green
    (28, 62, 198),    # blue
    (228, 118, 0),    # orange
    (138, 40, 172),   # purple
    (0, 138, 150),    # teal
)

NEEDLE_LENGTH = 138.0   # centre -> apex
NEEDLE_HALFBASE = 17.0  # half width of the shared base

CARDINALS: tuple[tuple[str, float], ...] = (
    ("up", 0.0),
    ("right", 90.0),
    ("down", 180.0),
    ("left", 270.0),
)

# Deterministic cardinal cycle. "up" is the prior-consistent (canonical control)
# heading; the other three are the counterfactuals.  The cycle over-samples the
# canonical case 3:1 relative to each counterfactual while keeping every class
# present, and it is matched: wording, palette, frame, scale and layout are
# identical across canonical and counterfactual scenes.
_CARDINAL_CYCLE = ("up", "right", "up", "down", "up", "left")

# Offset bands, in degrees away from the target cardinal. The last band lands
# inside the quarantine ring.
_BANDS: tuple[tuple[str, float, float], ...] = (
    ("centered", 0.0, 15.0),
    ("offset", 15.0, 33.0),
    ("near_boundary", 33.0, 40.8),
    ("boundary", 41.2, 44.0),
)

# A heading exactly 45 deg from two cardinals is a tie. Below this angular
# clearance the rounding is not certifiable at raster resolution.
QUARANTINE_MARGIN_DEG = 4.0


def _unit(a_deg: float) -> tuple[float, float]:
    a = math.radians(a_deg)
    return (math.sin(a), -math.cos(a))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene.

    Every stored field is rendered into the image; there are no hidden or
    unrendered latent fields, and only ``heading_deg`` feeds the analytic gold.
    """
    rng = random.Random((seed * 0x9E3779B1) & 0xFFFFFFFF)

    name = _CARDINAL_CYCLE[seed % len(_CARDINAL_CYCLE)]
    base = dict(CARDINALS)[name]
    _, lo, hi = _BANDS[(seed // len(_CARDINAL_CYCLE)) % len(_BANDS)]
    offset = rng.uniform(lo, hi) * rng.choice((-1.0, 1.0))
    heading = round((base + offset) % 360.0, 4)

    n_pips = rng.randint(3, 7)
    pips = sorted(round(rng.uniform(0.0, 360.0), 3) for _ in range(n_pips))

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "heading_deg": heading,
        "needle_color": list(NEEDLE_PALETTE[rng.randrange(len(NEEDLE_PALETTE))]),
        "pip_angles": pips,
    }


def _nearest_cardinal(heading_deg: float) -> tuple[str, float]:
    """(name, absolute angular distance) of the closest cardinal direction."""
    best_name = ""
    best_delta = 360.0
    for name, ang in CARDINALS:
        delta = abs((heading_deg - ang + 180.0) % 360.0 - 180.0)
        if delta < best_delta - 1e-12:
            best_name, best_delta = name, delta
    return best_name, best_delta


def analytic_gold(scene: dict) -> str:
    """The direction the solid triangle points, rounded to the nearest cardinal."""
    return _nearest_cardinal(float(scene["heading_deg"]))[0]


def margin(scene: dict) -> float:
    """Angular clearance in degrees to the nearest 45 deg tie line (0 .. 45)."""
    return round(45.0 - _nearest_cardinal(float(scene["heading_deg"]))[1], 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: declared `not_applicable`.

    Every field of the scene dict is drawn into the raster (canvas size, needle
    heading, needle fill colour, and the grey pip angles), and the only field
    that feeds the analytic gold -- ``heading_deg`` -- is directly visible as the
    needle's pointing direction against the drawn tick marks.  Two scenes with
    different gold necessarily differ by at least 4 deg of needle heading (the
    quarantine ring) and therefore cannot produce the same raster, so there is no
    pixel collision with disagreeing labels.  Gold-preserving perturbations do
    exist (repalette, pip permutation) but they visibly change the raster, so
    they are nuisance covariates, not aliases; they are exercised as explicit
    invariance checks in ``verify.py`` rather than declared here.
    """
    return []


# -- gold-preserving nuisance perturbations (invariance checks, not aliases) ---


def repalette(scene: dict, k: int = 1) -> dict:
    """Rotate the needle fill colour. Gold-preserving by construction."""
    cur = tuple(scene["needle_color"])
    idx = NEEDLE_PALETTE.index(cur) if cur in NEEDLE_PALETTE else 0
    out = dict(scene)
    out["needle_color"] = list(NEEDLE_PALETTE[(idx + k) % len(NEEDLE_PALETTE)])
    return out


def permute_pips(scene: dict, seed: int = 0) -> dict:
    """Re-place the grey rim pips. Gold-preserving by construction."""
    rng = random.Random(seed)
    out = dict(scene)
    out["pip_angles"] = sorted(
        round(rng.uniform(0.0, 360.0), 3) for _ in range(len(scene["pip_angles"]))
    )
    return out


def rotate_heading(scene: dict, quarter_turns: int) -> dict:
    """Label-TRANSFORMING symmetry: rotating the needle by k*90 deg must rotate
    the answer by exactly k steps along up -> right -> down -> left."""
    out = dict(scene)
    out["heading_deg"] = round((float(scene["heading_deg"]) + 90.0 * quarter_turns) % 360.0, 4)
    return out


def expected_rotated_gold(gold: str, quarter_turns: int) -> str:
    order = [name for name, _ in CARDINALS]
    return order[(order.index(gold) + quarter_turns) % 4]


# -- rendering ----------------------------------------------------------------


def _tri(cx: float, cy: float, ux: float, uy: float, length: float, half: float):
    px, py = -uy, ux
    return [
        (cx + ux * length, cy + uy * length),
        (cx + px * half, cy + py * half),
        (cx - px * half, cy - py * half),
    ]


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """I = R(z). Area-averaged downscale (no ringing, no out-of-gamut halos)."""
    ss = int(supersample)
    w, h = int(scene["width"]), int(scene["height"])
    img = Image.new("RGB", (w * ss, h * ss), BACKGROUND)
    d = ImageDraw.Draw(img)
    cx, cy = CENTER[0] * ss, CENTER[1] * ss
    rim = RIM_RADIUS * ss

    # Rim.
    d.ellipse([cx - rim, cy - rim, cx + rim, cy + rim], outline=FRAME_GRAY, width=4 * ss)

    # Four identical cardinal ticks and four shorter diagonal halfway marks.
    # The diagonal marks ARE the decision boundary: a needle aimed at one of them
    # is exactly 45 deg from two cardinals.
    for k in range(4):
        for ang, r0, r1, wid in (
            (90.0 * k, rim - 34 * ss, rim - 6 * ss, 5 * ss),
            (90.0 * k + 45.0, rim - 18 * ss, rim - 6 * ss, 3 * ss),
        ):
            ux, uy = _unit(ang)
            d.line([(cx + ux * r0, cy + uy * r0), (cx + ux * r1, cy + uy * r1)],
                   fill=FRAME_GRAY, width=int(wid))

    # The canonical "north" letter, welded to the top of the rim in every scene.
    nx, ny = cx, cy - rim + 56 * ss
    hw, hh = 13 * ss, 17 * ss
    d.line([(nx - hw, ny + hh), (nx - hw, ny - hh)], fill=FRAME_GRAY, width=4 * ss)
    d.line([(nx - hw, ny - hh), (nx + hw, ny + hh)], fill=FRAME_GRAY, width=4 * ss)
    d.line([(nx + hw, ny + hh), (nx + hw, ny - hh)], fill=FRAME_GRAY, width=4 * ss)

    # Grey decorative pips (achromatic distractors), drawn UNDER the needle.
    for ang in scene["pip_angles"]:
        ux, uy = _unit(float(ang))
        r = rim - 52 * ss
        pr = 5 * ss
        d.ellipse([cx + ux * r - pr, cy + uy * r - pr, cx + ux * r + pr, cy + uy * r + pr],
                  fill=PIP_GRAY)

    ux, uy = _unit(float(scene["heading_deg"]))
    length = NEEDLE_LENGTH * ss
    half = NEEDLE_HALFBASE * ss

    # Hollow (grey outline) half: the exact mirror image of the solid half, so
    # the needle silhouette is centrally symmetric and only the FILL polarity
    # encodes the answer.
    d.polygon(_tri(cx, cy, -ux, -uy, length, half), outline=FRAME_GRAY, width=4 * ss)
    # Solid half: the answer-bearing element.
    d.polygon(_tri(cx, cy, ux, uy, length, half), fill=tuple(scene["needle_color"]))

    return img.resize((w, h), Image.BOX) if ss > 1 else img
