"""Independent image-only inverse arm for the ticket queue raster."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

REFERENCE = {
    "red": ((220, 58, 64), (198, 72, 82)),
    "green": ((48, 166, 92), (62, 150, 98)),
    "blue": ((54, 105, 210), (70, 112, 191)),
    "swap": ((38, 185, 207), (54, 170, 191)),
    "serve": ((242, 139, 78), (224, 128, 72)),
}


def _mask(array: np.ndarray, references: tuple[tuple[int, int, int], ...]) -> np.ndarray:
    rgb = array.astype(np.int32)
    distances = [np.sqrt(np.sum((rgb - np.asarray(color, dtype=np.int32)) ** 2, axis=2))
                 for color in references]
    return np.minimum.reduce(distances) < 24.0


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, float, float]] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        points: list[tuple[int, int]] = []
        todo = deque([(y, x)])
        seen[y, x] = True
        while todo:
            cy, cx = todo.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        if len(points) >= 8:
            ys = np.fromiter((p[0] for p in points), dtype=np.float64)
            xs = np.fromiter((p[1] for p in points), dtype=np.float64)
            output.append((len(points), float(xs.mean()), float(ys.mean())))
    return output


def _one_component(array: np.ndarray, name: str, minimum_area: int) -> tuple[float, float]:
    found = [item for item in _components(_mask(array, REFERENCE[name])) if item[0] >= minimum_area]
    if len(found) != 1:
        raise ValueError(f"expected one intact {name} component, found {len(found)}")
    return found[0][1], found[0][2]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    tickets = []
    for name in ("red", "green", "blue"):
        x, y = _one_component(array, name, 24)
        tickets.append((x, y, name))
    if max(item[1] for item in tickets) - min(item[1] for item in tickets) > 8.0:
        raise ValueError("ticket row is not aligned")
    tickets.sort()
    queue = [item[2] for item in tickets]

    operations = []
    for name in ("swap", "serve"):
        x, y = _one_component(array, name, 32)
        operations.append((x, y, name))
    if max(item[1] for item in operations) - min(item[1] for item in operations) > 8.0:
        raise ValueError("operation row is not aligned")
    operations.sort()
    if operations[0][0] <= tickets[-1][0] + 25.0:
        raise ValueError("operation badges do not follow the ticket rail")
    for _, _, operation in operations:
        if operation == "swap":
            queue[0], queue[1] = queue[1], queue[0]
        else:
            queue.pop(0)
    return queue[0]
