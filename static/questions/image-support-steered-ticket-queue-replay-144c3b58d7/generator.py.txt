"""Deterministic renderer and analytic arm for a compact FIFO ticket queue."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
COLORS = ("red", "green", "blue")
PALETTES = {
    "bright": {
        "red": (220, 58, 64),
        "green": (48, 166, 92),
        "blue": (54, 105, 210),
        "swap": (38, 185, 207),
        "serve": (242, 139, 78),
    },
    "soft": {
        "red": (198, 72, 82),
        "green": (62, 150, 98),
        "blue": (70, 112, 191),
        "swap": (54, 170, 191),
        "serve": (224, 128, 72),
    },
}


def sample_scene(seed: int) -> dict[str, Any]:
    """Create balanced scenes while translating the decisive micro-inset."""
    rng = random.Random(int(seed))
    target = COLORS[int(seed) % len(COLORS)]
    operations = ["swap", "serve"]
    rng.shuffle(operations)
    rest = [color for color in COLORS if color != target]
    rng.shuffle(rest)
    if operations == ["swap", "serve"]:
        initial = [target, rest[0], rest[1]]
    else:
        initial = [rest[0], rest[1], target]
    # Centers remain aligned with the public 64 px ablation lattice, but the
    # whole two-part cue moves among many absolute locations across scenes.
    col = rng.choice((1, 2, 3, 4))
    row = rng.choice((1, 2, 3, 4, 5, 6))
    return {
        "anchor_x": 64 * col + 32,
        "anchor_y": 64 * row + 32,
        "initial": initial,
        "operations": operations,
        "palette": rng.choice(tuple(PALETTES)),
    }


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    palette = PALETTES[str(scene["palette"])]
    x0 = int(scene["anchor_x"])
    y0 = int(scene["anchor_y"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    ink = (42, 46, 54)
    quiet = (112, 118, 126)

    # Nondiagnostic remote registration marks prevent a bare cue-only canvas.
    for px, py in ((28, 28), (484, 28), (28, 484), (484, 484)):
        draw.ellipse((px - 2, py - 2, px + 2, py + 2), fill=(220, 218, 211))

    draw.text((x0 - 28, y0 - 31), "FRONT", fill=quiet, font=_font())
    draw.line((x0 - 32, y0 - 11, x0 + 33, y0 - 11), fill=(188, 190, 188), width=2)
    draw.polygon([(x0 - 35, y0 - 11), (x0 - 28, y0 - 15), (x0 - 28, y0 - 7)], fill=quiet)
    for offset, color in zip((-18, 0, 18), scene["initial"], strict=True):
        cx = x0 + offset
        draw.ellipse((cx - 8, y0 - 8, cx + 8, y0 + 8), fill=palette[str(color)], outline=ink, width=2)

    op_x = x0 + 128
    draw.text((op_x - 22, y0 - 31), "1       2", fill=quiet, font=_font())
    for offset, operation in zip((-16, 16), scene["operations"], strict=True):
        cx = op_x + offset
        draw.rounded_rectangle((cx - 10, y0 - 10, cx + 10, y0 + 10), radius=3,
                               fill=palette[str(operation)], outline=ink, width=2)
        if operation == "swap":
            draw.line((cx - 5, y0 - 3, cx + 5, y0 - 3), fill=ink, width=1)
            draw.line((cx + 5, y0 + 3, cx - 5, y0 + 3), fill=ink, width=1)
            draw.polygon([(cx + 6, y0 - 3), (cx + 2, y0 - 6), (cx + 2, y0)], fill=ink)
            draw.polygon([(cx - 6, y0 + 3), (cx - 2, y0), (cx - 2, y0 + 6)], fill=ink)
        else:
            draw.line((cx - 5, y0, cx + 5, y0), fill=ink, width=2)

    # A faint nonsemantic separator keeps the two evidence components visibly detached.
    draw.text((x0 + 52, y0 - 4), "then", fill=(178, 178, 174), font=_font())
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    queue = list(scene["initial"])
    for operation in scene["operations"]:
        if operation == "swap":
            queue[0], queue[1] = queue[1], queue[0]
        elif operation == "serve":
            queue.pop(0)
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return str(queue[0])


def margin(scene: dict[str, Any]) -> float:
    # One discrete glyph substitution is the runner-up edit boundary.
    return 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
