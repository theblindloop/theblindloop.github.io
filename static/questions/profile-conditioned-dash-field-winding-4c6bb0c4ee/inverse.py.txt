"""Pixel-only inverse arm for the dash-field winding sign."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 35:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _extract(image: Image.Image) -> list[tuple[float, float, float]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate((arr[:3].reshape(-1, 3), arr[-3:].reshape(-1, 3), arr[:, :3].reshape(-1, 3), arr[:, -3:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    components = _components(distance > 45.0)
    measurements = []
    for points in components:
        # Component points arrive as (y,x); PCA is performed in Cartesian (x,y).
        xy = points[:, ::-1] + 0.5
        center = xy.mean(axis=0)
        covariance = np.cov(xy, rowvar=False)
        values, vectors = np.linalg.eigh(covariance)
        if values[1] < 2.8 * max(values[0], 1e-9):
            continue
        vector = vectors[:, 1]
        phi = math.atan2(float(vector[1]), float(vector[0]))
        measurements.append((float(center[0]), float(center[1]), phi))
    if len(measurements) < 12:
        raise ValueError(f"insufficient elongated dash evidence: {len(measurements)} components")
    return measurements


def decision_from_image(image: Image.Image) -> str:
    measurements = _extract(image)
    cx = sum(item[0] for item in measurements) / len(measurements)
    cy = sum(item[1] for item in measurements) / len(measurements)
    strengths = {}
    for sign in (1, -1):
        phases = []
        for x, y, phi in measurements:
            theta = math.atan2(y - cy, x - cx)
            phases.append(phi - sign * theta)
        c = sum(math.cos(2 * value) for value in phases) / len(phases)
        s = sum(math.sin(2 * value) for value in phases) / len(phases)
        strengths[sign] = math.hypot(c, s)
    if abs(strengths[1] - strengths[-1]) < 0.12:
        raise ValueError("ambiguous winding evidence")
    return "whirl" if strengths[1] > strengths[-1] else "saddle"
