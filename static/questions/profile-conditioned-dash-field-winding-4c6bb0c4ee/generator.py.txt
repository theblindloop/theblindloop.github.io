"""Deterministic renderer for a global unoriented dash field."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

W = H = 360
SCALE = 4
PALETTES = [
    ("#f7f4ec", "#17324d"),
    ("#f3f7f4", "#3b244a"),
    ("#f8f2f2", "#16433d"),
    ("#f3f4f8", "#4a2a20"),
]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    field_type = "whirl" if int(seed) % 2 == 0 else "saddle"
    background, ink = PALETTES[rng.randrange(len(PALETTES))]
    cx = rng.uniform(170.0, 190.0)
    cy = rng.uniform(170.0, 190.0)
    # Phase is a nuisance parameter. Whirls remain visually tangential; saddle
    # phase freely rotates the hyperbolic texture without affecting its label.
    phase = math.pi / 2 + rng.uniform(-0.08, 0.08) if field_type == "whirl" else rng.uniform(-math.pi / 2, math.pi / 2)
    dashes: list[dict[str, float]] = []
    angular_offset = rng.uniform(-math.pi, math.pi)
    for i in range(12):
        theta = angular_offset + i * math.pi / 12 + rng.uniform(-0.035, 0.035)
        radius = (82.0 if i % 2 == 0 else 119.0) + rng.uniform(-4.0, 4.0)
        length = rng.uniform(21.0, 26.0)
        thickness = rng.uniform(5.0, 6.5)
        jitter = rng.uniform(-0.07, 0.07)
        # Exact antipodal pairing makes the field center recoverable from pixels
        # without a privileged marker or fixed canvas coordinate.
        for antipode in (0.0, math.pi):
            a = theta + antipode
            dashes.append({
                "x": cx + radius * math.cos(a),
                "y": cy + radius * math.sin(a),
                "length": length,
                "thickness": thickness,
                "jitter": jitter,
            })
    return {
        "field_type": field_type,
        "width": W,
        "height": H,
        "background": background,
        "ink": ink,
        "center": [cx, cy],
        "phase": phase,
        "dashes": dashes,
    }


def _axis_angle(scene: dict, dash: dict) -> float:
    cx, cy = scene["center"]
    theta = math.atan2(dash["y"] - cy, dash["x"] - cx)
    winding = 1.0 if scene["field_type"] == "whirl" else -1.0
    return winding * theta + scene["phase"] + dash["jitter"]


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), scene["background"])
    draw = ImageDraw.Draw(image)
    ink = scene["ink"]
    for dash in scene["dashes"]:
        phi = _axis_angle(scene, dash)
        ux, uy = math.cos(phi), math.sin(phi)
        vx, vy = -uy, ux
        half_l = dash["length"] * SCALE / 2
        half_t = dash["thickness"] * SCALE / 2
        x, y = dash["x"] * SCALE, dash["y"] * SCALE
        polygon = [
            (x - half_l * ux - half_t * vx, y - half_l * uy - half_t * vy),
            (x + half_l * ux - half_t * vx, y + half_l * uy - half_t * vy),
            (x + half_l * ux + half_t * vx, y + half_l * uy + half_t * vy),
            (x - half_l * ux + half_t * vx, y - half_l * uy + half_t * vy),
        ]
        draw.polygon(polygon, fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["field_type"])


def _model_scores(scene: dict) -> tuple[float, float]:
    cx, cy = scene["center"]
    phases = {1: [], -1: []}
    for dash in scene["dashes"]:
        theta = math.atan2(dash["y"] - cy, dash["x"] - cx)
        phi = _axis_angle(scene, dash)
        for sign in (1, -1):
            phases[sign].append(phi - sign * theta)
    scores = []
    for sign in (1, -1):
        values = phases[sign]
        c = sum(math.cos(2 * value) for value in values) / len(values)
        s = sum(math.sin(2 * value) for value in values) / len(values)
        scores.append(math.hypot(c, s))
    return scores[0], scores[1]


def margin(scene: dict) -> float:
    positive, negative = _model_scores(scene)
    return abs(positive - negative)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.35


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
