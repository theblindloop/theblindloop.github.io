"""Finite-state machine transition-table register (state tracking).

A single tracked token carries a colour state (red, green, or blue). On the
left the START panel shows the token in its initial colour, ringed in orange.
To the right a row of four operator boxes (read left to right) each holds one
of three dark operator glyphs (filled circle, square, or triangle). A visible
3x3 transition table below defines the automaton: its left column lists the
three current states as coloured disks (red, green, blue, top to bottom) and
its top row lists the three operators as the same three dark glyphs; every
table cell holds the coloured disk of the next state for the (current-state
row, operator column) pair.

The decision is the colour of the machine after all four operators are applied
in order from the ringed start state, where each transition is a (possibly
non-bijective, non-cyclic) lookup in the table. The final colour is never drawn
anywhere; it must be recovered by replaying the ordered transitions from the
start token, so the gold is an order-dependent composition of the whole chain,
not a single frame, a count, or a net displacement.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 620

BG = (255, 255, 255)
DARK = (50, 50, 50)
BORDER = (210, 210, 210)
RING = (250, 140, 0)

COLOR_NAMES = ["red", "green", "blue"]
COLORS = {
    "red": (220, 40, 40),
    "green": (40, 170, 80),
    "blue": (40, 80, 220),
}
OPERATOR_NAMES = ["circle", "square", "triangle"]

START_CENTER = (115, 115)
START_DISK_R = 32
START_RING_R = 39
RING_W = 7

OPERATOR_CENTERS = [(250 + i * 120, 118) for i in range(4)]
OPERATOR_GLYPH_R = 26

LEGEND_ROW_X = 90
LEGEND_ROW_YS = [330, 420, 510]
LEGEND_COL_XS = [210, 300, 390]
LEGEND_COL_Y = 295
LEGEND_HEADER_DISK_R = 20
LEGEND_GLYPH_R = 20
LEGEND_CELL_R = 15


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    start = rng.randrange(3)
    operators = [rng.randrange(3) for _ in range(4)]
    table = [[rng.randrange(3) for _ in range(3)] for _ in range(3)]
    return {"start_color": start, "operators": operators, "table": table}


def final_index(scene: dict) -> int:
    cur = scene["start_color"]
    table = scene["table"]
    for op in scene["operators"]:
        cur = table[cur][op]
    return cur


def analytic_gold(scene: dict) -> str:
    return COLOR_NAMES[final_index(scene)]


def _draw_filled_disk(draw, center, radius, color) -> None:
    cx, cy = center
    draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=color)


def _draw_glyph(draw, center, radius, shape: str) -> None:
    cx, cy = center
    if shape == "circle":
        _draw_filled_disk(draw, center, radius, DARK)
    elif shape == "square":
        draw.rectangle(
            [cx - radius, cy - radius, cx + radius, cy + radius], fill=DARK
        )
    elif shape == "triangle":
        pts = [
            (cx, cy - radius),
            (cx - radius, cy + radius),
            (cx + radius, cy + radius),
        ]
        draw.polygon(pts, fill=DARK)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    start = scene["start_color"]
    operators = scene["operators"]
    table = scene["table"]

    # Start panel box.
    draw.rectangle([40, 40, 190, 190], outline=BORDER, width=4)
    _draw_filled_disk(draw, START_CENTER, START_DISK_R, COLORS[COLOR_NAMES[start]])
    draw.ellipse(
        [
            START_CENTER[0] - START_RING_R,
            START_CENTER[1] - START_RING_R,
            START_CENTER[0] + START_RING_R,
            START_CENTER[1] + START_RING_R,
        ],
        outline=RING,
        width=RING_W,
    )

    # Operator boxes.
    for i, op in enumerate(operators):
        cx, cy = OPERATOR_CENTERS[i]
        draw.rectangle([cx - 48, cy - 48, cx + 48, cy + 48], outline=BORDER, width=4)
        _draw_glyph(draw, (cx, cy), OPERATOR_GLYPH_R, OPERATOR_NAMES[op])

    # Legend box.
    draw.rectangle([40, 250, 500, 600], outline=BORDER, width=4)
    # Row headers (current states, red green blue top to bottom).
    for r, y in enumerate(LEGEND_ROW_YS):
        _draw_filled_disk(draw, (LEGEND_ROW_X, y), LEGEND_HEADER_DISK_R, COLORS[COLOR_NAMES[r]])
    # Column headers (operators).
    for c, x in enumerate(LEGEND_COL_XS):
        _draw_glyph(draw, (x, LEGEND_COL_Y), LEGEND_GLYPH_R, OPERATOR_NAMES[c])
    # Cells.
    for r, y in enumerate(LEGEND_ROW_YS):
        for c, x in enumerate(LEGEND_COL_XS):
            _draw_filled_disk(
                draw, (x, y), LEGEND_CELL_R, COLORS[COLOR_NAMES[table[r][c]]]
            )
    return img


def _replay_with(table, start, operators) -> int:
    cur = start
    for op in operators:
        cur = table[cur][op]
    return cur


def margin(scene: dict) -> float:
    """Evidence-dependency count: how many single table-cell changes would alter
    the final answer. Positive (at least the on-path cells count) and
    deterministic, so erasing any on-path transition changes the decision."""
    start = scene["start_color"]
    operators = scene["operators"]
    table = scene["table"]
    base = _replay_with(table, start, operators)
    dep = 0
    for r in range(3):
        for c in range(3):
            original = table[r][c]
            for alt in range(3):
                if alt == original:
                    continue
                new_table = [list(row) for row in table]
                new_table[r][c] = alt
                if _replay_with(new_table, start, operators) != base:
                    dep += 1
    return float(dep)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: start_color fixes the
    ringed start disk colour, operators fix which glyphs appear in the four
    operator boxes in order, and table fixes every transition cell colour. Red
   rawing any latent field changes the corresponding pixels, and the analytic
    replay reads exactly those fields, so no reparametrization yields a
    pixel-identical raster with a different gold."""
    return []
