"""Independent pixel-only inverse for tapered-aperture polarity."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) > len(best):
            best = points
    output = np.zeros_like(mask, dtype=bool)
    if best:
        yy, xx = zip(*best)
        output[np.asarray(yy), np.asarray(xx)] = True
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    dark = arr.mean(axis=2) < 205
    plaque = _largest_component((chroma > 35) & dark)
    ys, xs = np.nonzero(plaque)
    if len(xs) < 4500:
        raise ValueError("plaque missing or damaged")
    x0, x1, y0, y1 = xs.min(), xs.max(), ys.min(), ys.max()
    if min(x1-x0, y1-y0) < 85 or max(x1-x0, y1-y0) > 116:
        raise ValueError("invalid plaque extent")
    cx, cy = (x0+x1)/2.0, (y0+y1)/2.0
    radius = min(x1-x0, y1-y0)/2.0
    yy, xx = np.indices(plaque.shape)
    inside = (xx-cx)**2 + (yy-cy)**2 < (radius-5.0)**2
    bg = np.median(np.concatenate((arr[:3].reshape(-1,3), arr[-3:].reshape(-1,3), arr[:,:3].reshape(-1,3), arr[:,-3:].reshape(-1,3))), axis=0)
    pale = np.linalg.norm(arr.astype(float)-bg, axis=2) < 12.0
    hole = _largest_component(inside & pale)
    hy, hx = np.nonzero(hole)
    if not 450 <= len(hx) <= 1700:
        raise ValueError("aperture missing or damaged")
    vx, vy = hx.mean()-cx, hy.mean()-cy
    norm = float(np.hypot(vx, vy))
    if norm < 7.0:
        raise ValueError("aperture has no radial direction")
    ux, uy = vx/norm, vy/norm
    radial = (hx-cx)*ux + (hy-cy)*uy
    lateral = -(hx-cx)*uy + (hy-cy)*ux
    qlo, qhi = np.quantile(radial, [0.18, 0.82])
    low = lateral[radial <= qlo]
    high = lateral[radial >= qhi]
    low_width = float(np.quantile(low, .9)-np.quantile(low, .1))
    high_width = float(np.quantile(high, .9)-np.quantile(high, .1))
    if abs(high_width-low_width) < 5.0:
        raise ValueError("aperture taper is ambiguous")
    return "inward" if low_width < high_width else "outward"
