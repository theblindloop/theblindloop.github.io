"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

PALETTE = [(20, 116, 142), (173, 55, 76), (87, 86, 166), (31, 132, 91)]
BACKGROUND = (246, 244, 238)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    radius = rng.randint(47, 55)
    cx = rng.randint(radius + 24, 512 - radius - 24)
    cy = rng.randint(radius + 24, 512 - radius - 24)
    polarity = "inward" if int(seed) % 2 == 0 else "outward"
    distractors = []
    for _ in range(7):
        for _attempt in range(100):
            x, y = rng.randint(18, 494), rng.randint(18, 494)
            if math.hypot(x - cx, y - cy) > radius + 45:
                distractors.append([x, y, rng.randint(2, 4)])
                break
    return {
        "width": 512,
        "height": 512,
        "center_x": cx,
        "center_y": cy,
        "radius": radius,
        "angle": rng.uniform(0.0, 2.0 * math.pi),
        "polarity": polarity,
        "aperture_depth": rng.randint(30, 38),
        "base_half_width": rng.randint(13, 18),
        "tip_half_width": rng.randint(2, 4),
        "plaque_color": list(PALETTE[rng.randrange(len(PALETTE))]),
        "background_color": list(BACKGROUND),
        "distractors": distractors,
    }


def _point(cx: float, cy: float, radial: float, lateral: float, angle: float) -> tuple[float, float]:
    ux, uy = math.cos(angle), math.sin(angle)
    vx, vy = -uy, ux
    return cx + radial * ux + lateral * vx, cy + radial * uy + lateral * vy


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    bg = tuple(scene["background_color"])
    image = Image.new("RGB", (width, height), bg)
    draw = ImageDraw.Draw(image)
    for x, y, r in scene["distractors"]:
        shade = (181, 181, 176)
        draw.ellipse((x-r, y-r, x+r, y+r), fill=shade)
    cx, cy, radius = scene["center_x"], scene["center_y"], scene["radius"]
    color = tuple(scene["plaque_color"])
    draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), fill=color)
    angle = float(scene["angle"])
    depth = float(scene["aperture_depth"])
    base = float(scene["base_half_width"])
    tip = float(scene["tip_half_width"])
    polarity = scene["polarity"]
    inner_r, outer_r = 5.0, 5.0 + depth
    inner_w, outer_w = (tip, base) if polarity == "inward" else (base, tip)
    polygon = [
        _point(cx, cy, inner_r, -inner_w, angle),
        _point(cx, cy, outer_r, -outer_w, angle),
        _point(cx, cy, outer_r, outer_w, angle),
        _point(cx, cy, inner_r, inner_w, angle),
    ]
    draw.polygon(polygon, fill=bg)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["polarity"])


def margin(scene: dict) -> float:
    return float(scene["base_half_width"] - scene["tip_half_width"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
