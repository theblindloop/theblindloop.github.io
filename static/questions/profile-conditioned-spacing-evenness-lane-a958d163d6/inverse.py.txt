"""Pixel-only decision arm for spacing_evenness_lane.

Receives the final PNG and nothing else: no latent scene, no renderer module, no
analytic gold. It segments the coloured tick marks by hue, groups them into
connected blobs, orders each colour's blob centres from left to right, measures
the gaps between neighbours, and reports the colour whose widest gap minus
narrowest gap is smallest.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_REFERENCE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
_TOLERANCE = 40
_SATURATION = 45
_EXPECTED_COLORS = 5
_EXPECTED_MARKS = 8
_MIN_AREA = 220
_MAX_AREA = 430
_MIN_SEPARATION = 22.0
_MIN_CLEARANCE = 8.0


def _blobs(mask: np.ndarray) -> list[tuple[float, int, int, int]] | None:
    """Connected components of a boolean mask as (centre_x, area, height, width)."""

    points = {(int(r), int(c)) for r, c in np.argwhere(mask)}
    out: list[tuple[float, int, int, int]] = []
    while points:
        stack = [points.pop()]
        blob = list(stack)
        while stack:
            row, col = stack.pop()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                neighbour = (row + dr, col + dc)
                if neighbour in points:
                    points.discard(neighbour)
                    stack.append(neighbour)
                    blob.append(neighbour)
        rows = [point[0] for point in blob]
        cols = [point[1] for point in blob]
        out.append(
            (
                sum(cols) / len(cols),
                len(blob),
                max(rows) - min(rows) + 1,
                max(cols) - min(cols) + 1,
            )
        )
        if len(out) > 60:
            return None
    return out


def decision_from_image(image: Image.Image) -> str:
    """Colour of the most evenly spaced lane, read from the raster alone."""

    data = np.asarray(image.convert("RGB")).astype(np.int16)
    saturation = data.max(axis=2) - data.min(axis=2)
    coloured = saturation > _SATURATION
    unevenness: dict[str, float] = {}
    for name, reference in _REFERENCE.items():
        distance = np.abs(data - np.array(reference, dtype=np.int16)).max(axis=2)
        mask = coloured & (distance <= _TOLERANCE)
        if not mask.any():
            continue
        blobs = _blobs(mask)
        if blobs is None:
            return ABSTAIN
        kept = [blob for blob in blobs if blob[1] >= _MIN_AREA]
        if len(kept) != _EXPECTED_MARKS:
            return ABSTAIN
        for _, area, height, width in kept:
            if area > _MAX_AREA or not 20 <= height <= 32 or not 8 <= width <= 18:
                return ABSTAIN
        centres = sorted(blob[0] for blob in kept)
        gaps = [b - a for a, b in zip(centres, centres[1:])]
        if min(gaps) < _MIN_SEPARATION:
            return ABSTAIN
        unevenness[name] = max(gaps) - min(gaps)
    if len(unevenness) != _EXPECTED_COLORS:
        return ABSTAIN
    ranked = sorted(unevenness.items(), key=lambda item: item[1])
    if ranked[1][1] - ranked[0][1] < _MIN_CLEARANCE:
        return ABSTAIN
    return ranked[0][0]
