"""Scene sampler, raster renderer and analytic gold for spacing_evenness_lane.

Five horizontal lanes each carry the same number of tick marks between the same
two end marks, so count, span, mean spacing and total ink are constant. The
deciding quantity is the EVENNESS of one lane's own spacing: its widest gap
between neighbouring marks minus its narrowest gap. The answer is the colour
whose difference is smallest.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W, H = 920, 560
X0, X1 = 70, 840
SPAN = X1 - X0
N_MARKS = 8
N_GAPS = N_MARKS - 1
MIN_GAP = 40
MARK_W, MARK_H = 12, 26
ROW_Y = (90, 185, 280, 375, 470)

COLORS = ("red", "green", "blue", "orange", "purple")
RGB = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}

MIN_MARGIN = 20.0
QUARANTINE_MARGIN = 30.0
WINNER_LO, WINNER_HI = 4.0, 18.0
LOSER_HI = 135.0
LOSER_CLEAR = 24.0


def _gaps(marks: list[int]) -> list[int]:
    ordered = sorted(marks)
    return [b - a for a, b in zip(ordered, ordered[1:])]


def _score(marks: list[int]) -> float:
    """Unevenness of one lane: its widest gap minus its narrowest gap, in pixels."""

    gaps = _gaps(marks)
    return float(max(gaps) - min(gaps))


def _marks_for(rng: random.Random, target: float) -> list[int] | None:
    """Integer mark positions from X0 to X1 whose gap range is close to target."""

    raw = [rng.random() for _ in range(N_GAPS)]
    low, high = min(raw), max(raw)
    if high - low < 1e-6:
        return None
    unit = [(value - low) / (high - low) for value in raw]
    base = (SPAN - target * sum(unit)) / N_GAPS
    if base < MIN_GAP:
        return None
    gaps = [base + target * value for value in unit]
    positions = [X0]
    total = 0.0
    for gap in gaps[:-1]:
        total += gap
        positions.append(X0 + int(round(total)))
    positions.append(X1)
    if min(_gaps(positions)) < MIN_GAP:
        return None
    return positions


def _lane_scores(scene: dict) -> list[tuple[float, str]]:
    return sorted((_score(lane["marks"]), lane["color"]) for lane in scene["lanes"])


def analytic_gold(scene: dict) -> str:
    """Colour of the lane whose widest gap minus narrowest gap is smallest."""

    return _lane_scores(scene)[0][1]


def margin(scene: dict) -> float:
    """Pixels of unevenness separating the evenest lane from the runner-up."""

    ranked = _lane_scores(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _cheap_cues_are_clean(scene: dict) -> bool:
    """The widest and the narrowest single gap of the picture must not name the answer."""

    gold = analytic_gold(scene)
    widest = max(scene["lanes"], key=lambda lane: max(_gaps(lane["marks"])))
    narrowest = min(scene["lanes"], key=lambda lane: min(_gaps(lane["marks"])))
    longest_run = max(scene["lanes"], key=lambda lane: _gaps(lane["marks"])[0])
    return gold not in {widest["color"], narrowest["color"], longest_run["color"]}


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene as an unordered set of lane records."""

    rng = random.Random(int(seed))
    for _ in range(4000):
        winner = rng.uniform(WINNER_LO, WINNER_HI)
        losers = [rng.uniform(winner + LOSER_CLEAR, LOSER_HI) for _ in range(4)]
        targets = [winner] + losers
        colours = list(COLORS)
        rng.shuffle(colours)
        rows = list(range(len(ROW_Y)))
        rng.shuffle(rows)
        lanes = []
        for colour, row, target in zip(colours, rows, targets):
            marks = _marks_for(rng, target)
            if marks is None:
                break
            lanes.append({"color": colour, "row": row, "marks": marks})
        if len(lanes) != len(COLORS):
            continue
        rules = sorted(rng.randrange(X0 + 60, X1 - 60) for _ in range(3))
        scene = {"lanes": lanes, "rules": rules}
        if margin(scene) < MIN_MARGIN:
            continue
        if not _cheap_cues_are_clean(scene):
            continue
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Deterministic integer raster of the scene."""

    image = Image.new("RGB", (W, H), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    draw.rectangle((18, 18, W - 19, H - 19), outline=(150, 150, 150), width=2)
    for x in scene.get("rules", []):
        draw.line((x, 30, x, H - 31), fill=(232, 232, 232), width=2)
    for lane in scene["lanes"]:
        y = ROW_Y[lane["row"]]
        draw.line((X0 - 34, y, X1 + 34, y), fill=(216, 216, 216), width=2)
    for lane in scene["lanes"]:
        y = ROW_Y[lane["row"]]
        colour = RGB[lane["color"]]
        for x in lane["marks"]:
            draw.rectangle(
                (x - MARK_W // 2, y - MARK_H // 2, x + MARK_W // 2 - 1, y + MARK_H // 2 - 1),
                fill=colour,
            )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent reorderings that leave the raster byte-identical and the gold fixed."""

    lanes = scene["lanes"]
    rules = list(scene.get("rules", []))
    return [
        ("reverse_lane_list", {"lanes": list(reversed(lanes)), "rules": rules}),
        ("rotate_lane_list", {"lanes": lanes[2:] + lanes[:2], "rules": rules}),
        (
            "reverse_marks_in_every_lane",
            {
                "lanes": [
                    {"color": lane["color"], "row": lane["row"], "marks": list(reversed(lane["marks"]))}
                    for lane in lanes
                ],
                "rules": rules,
            },
        ),
        ("reverse_rule_list", {"lanes": lanes, "rules": list(reversed(rules))}),
    ]
