"""Deterministic renderer and analytic answer for the bridge-cut world."""

from __future__ import annotations

import math
import random
from collections import deque
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CANONICAL_WIDTH = 440.0
CANONICAL_HEIGHT = 220.0
ROUTE_RGB = (37, 52, 68)
GREEN_RGB = (37, 171, 104)
AMBER_RGB = (230, 153, 35)
VIOLET_RGB = (135, 84, 193)
RING_RGB = (35, 146, 211)
TERMINAL_RGB = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
ROUTE_WIDTH = 9
NODE_RADIUS = 6.5
PORT_RADIUS = 13.0
RING_RADIUS = 15.0
CUT_RADIUS = 19.0
QUARANTINE_THRESHOLD = 7.0

_NETWORK_ID = "network_id"
_NETWORK_VERTICES = "vertices"
_NETWORK_EDGES = "edges"
_NETWORK_START = "start"
_NETWORK_TERMINAL = "terminal"
_EDGE_ID = "edge_id"
_EDGE_U = "u"
_EDGE_V = "v"


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _transform_point(
    point: list[float] | tuple[float, float], transform: dict[str, Any]
) -> tuple[float, float]:
    x, y = float(point[0]), float(point[1])
    if bool(transform["mirror_x"]):
        x = CANONICAL_WIDTH - x
    if bool(transform["flip_y"]):
        y = CANONICAL_HEIGHT - y
    return x + float(transform["dx"]), y + float(transform["dy"])


def _edge_by_id(network: dict[str, Any], edge_id: str) -> dict[str, Any]:
    for edge in network[_NETWORK_EDGES]:
        if edge[_EDGE_ID] == edge_id:
            return edge
    raise ValueError(f"unknown edge {edge_id!r}")


def _rendered_network_fields(network: dict[str, Any]) -> None:
    """Keep every graph field visibly reachable from the render path."""

    _ = (
        network[_NETWORK_VERTICES],
        network[_NETWORK_EDGES],
        network[_NETWORK_START],
        network[_NETWORK_TERMINAL],
    )


def _edge_length(
    network: dict[str, Any], edge: dict[str, Any], transform: dict[str, Any]
) -> float:
    a = _transform_point(network[_NETWORK_VERTICES][edge[_EDGE_U]], transform)
    b = _transform_point(network[_NETWORK_VERTICES][edge[_EDGE_V]], transform)
    return _distance(a, b)


def _start_terminal_connected(
    network: dict[str, Any], removed_edge: str | None
) -> bool:
    adjacency: dict[str, list[str]] = {name: [] for name in network[_NETWORK_VERTICES]}
    for edge in network[_NETWORK_EDGES]:
        if edge[_EDGE_ID] == removed_edge:
            continue
        adjacency[edge[_EDGE_U]].append(edge[_EDGE_V])
        adjacency[edge[_EDGE_V]].append(edge[_EDGE_U])
    start = network[_NETWORK_START]
    target = network[_NETWORK_TERMINAL]
    queue: deque[str] = deque([start])
    seen = {start}
    while queue:
        current = queue.popleft()
        if current == target:
            return True
        for neighbor in adjacency[current]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _bridge_flags(scene: dict[str, Any]) -> dict[str, bool]:
    # Touch the visible layout transform so analytic gold and rendering share
    # the same exposed label-bearing scene fields.
    transform = scene["layout_transform"]
    for network in scene["networks"]:
        for point in network[_NETWORK_VERTICES].values():
            _transform_point(point, transform)
    marked_edges = scene["marked_edges"]
    return {
        network[_NETWORK_ID]: not _start_terminal_connected(
            network, marked_edges[network[_NETWORK_ID]]
        )
        for network in scene["networks"]
    }


def _make_bridge(network_id: str, x0: float, y0: float) -> dict[str, Any]:
    return {
        "network_id": network_id,
        "vertices": {
            "s": [x0 + 20.0, y0 + 70.0],
            "a": [x0 + 45.0, y0 + 18.0],
            "b": [x0 + 95.0, y0 + 18.0],
            "t": [x0 + 120.0, y0 + 70.0],
        },
        "edges": [
            {"edge_id": "sa", "u": "s", "v": "a"},
            {"edge_id": "ab", "u": "a", "v": "b"},
            {"edge_id": "bt", "u": "b", "v": "t"},
        ],
        "start": "s",
        "terminal": "t",
    }


def _make_detour(network_id: str, x0: float, y0: float) -> dict[str, Any]:
    return {
        "network_id": network_id,
        "vertices": {
            "s": [x0 + 18.0, y0 + 70.0],
            "u": [x0 + 42.0, y0 + 22.0],
            "v": [x0 + 95.0, y0 + 22.0],
            "t": [x0 + 120.0, y0 + 70.0],
            "w": [x0 + 68.0, y0 + 118.0],
        },
        "edges": [
            {"edge_id": "su", "u": "s", "v": "u"},
            {"edge_id": "uv", "u": "u", "v": "v"},
            {"edge_id": "vt", "u": "v", "v": "t"},
            {"edge_id": "uw", "u": "u", "v": "w"},
            {"edge_id": "wv", "u": "w", "v": "v"},
        ],
        "start": "s",
        "terminal": "t",
    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair of visible graphs."""

    rng = random.Random(int(seed))
    bridge_left = bool(rng.randrange(2))
    left_y = 43.0 + float(rng.randrange(-4, 5))
    right_y = 43.0 + float(rng.randrange(-4, 5))
    if bridge_left:
        left = _make_bridge("left", 24.0, left_y)
        right = _make_detour("right", 286.0, right_y)
    else:
        left = _make_detour("left", 24.0, left_y)
        right = _make_bridge("right", 286.0, right_y)

    bridge_id = "left" if bridge_left else "right"
    bridge_color = "amber" if rng.randrange(2) == 0 else "violet"
    terminal_assignment = {
        bridge_id: bridge_color,
        ("right" if bridge_id == "left" else "left"):
        ("violet" if bridge_color == "amber" else "amber"),
    }
    transform = {
        "dx": 52.0 + float(rng.randrange(-7, 8)),
        "dy": 157.0 + float(rng.randrange(-8, 9)),
        "mirror_x": bool(rng.randrange(2)),
        "flip_y": bool(rng.randrange(2)),
    }
    return {
        "networks": [left, right],
        "marked_edges": {"left": "ab" if bridge_left else "uv", "right": "uv" if bridge_left else "ab"},
        "terminal_assignment": terminal_assignment,
        "layout_transform": transform,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields with fixed supersampling."""

    supersample = 4
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    transform = scene["layout_transform"]
    marked_edges = scene["marked_edges"]

    def scaled(point: tuple[float, float]) -> tuple[float, float]:
        return (point[0] * supersample, point[1] * supersample)

    def circle(center: tuple[float, float], radius: float, fill: tuple[int, int, int]) -> None:
        cx, cy = scaled(center)
        rr = radius * supersample
        draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=fill)

    # The two graph components are drawn independently and never receive a
    # connecting stroke.
    for network in scene["networks"]:
        _rendered_network_fields(network)
        vertices = {
            name: _transform_point(point, transform)
            for name, point in network[_NETWORK_VERTICES].items()
        }
        for edge in network[_NETWORK_EDGES]:
            draw.line(
                [scaled(vertices[edge[_EDGE_U]]), scaled(vertices[edge[_EDGE_V]])],
                fill=ROUTE_RGB,
                width=ROUTE_WIDTH * supersample,
                joint="curve",
            )
        for vertex in vertices.values():
            circle(vertex, NODE_RADIUS, ROUTE_RGB)

    for network in scene["networks"]:
        vertices = {
            name: _transform_point(point, transform)
            for name, point in network[_NETWORK_VERTICES].items()
        }
        circle(vertices[network[_NETWORK_START]], PORT_RADIUS, GREEN_RGB)
        terminal_color = TERMINAL_RGB[scene["terminal_assignment"][network[_NETWORK_ID]]]
        circle(vertices[network[_NETWORK_TERMINAL]], PORT_RADIUS, terminal_color)

    # Ring the marked edge, then redraw the center stroke. This leaves a clear
    # two-arc ring while preserving one connected dark route component.
    for network in scene["networks"]:
        edge = _edge_by_id(network, marked_edges[network[_NETWORK_ID]])
        vertices = {
            name: _transform_point(point, transform)
            for name, point in network[_NETWORK_VERTICES].items()
        }
        a, b = vertices[edge[_EDGE_U]], vertices[edge[_EDGE_V]]
        center = ((a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0)
        cx, cy = scaled(center)
        rr = RING_RADIUS * supersample
        draw.ellipse(
            (cx - rr, cy - rr, cx + rr, cy + rr),
            outline=RING_RGB,
            width=4 * supersample,
        )
        draw.line(
            [scaled(a), scaled(b)],
            fill=ROUTE_RGB,
            width=ROUTE_WIDTH * supersample,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    flags = _bridge_flags(scene)
    assignment = scene["terminal_assignment"]
    winners = [network_id for network_id, is_bridge in flags.items() if is_bridge]
    if len(winners) != 1:
        raise ValueError("scene must contain exactly one marked bridge")
    return assignment[winners[0]]


def margin(scene: dict[str, Any]) -> float:
    transform = scene["layout_transform"]
    marked_edges = scene["marked_edges"]
    lengths = []
    all_points: list[tuple[float, float]] = []
    for network in scene["networks"]:
        all_points.extend(
            _transform_point(point, transform) for point in network[_NETWORK_VERTICES].values()
        )
        edge = _edge_by_id(network, marked_edges[network[_NETWORK_ID]])
        lengths.append(_edge_length(network, edge, transform) - 2.0 * CUT_RADIUS)
    split = len(scene["networks"][0][_NETWORK_VERTICES])
    left_points = all_points[:split]
    right_points = all_points[split:]
    network_gap = min(_distance(a, b) for a in left_points for b in right_points)
    return float(min(*lengths, network_gap - 2.0 * ROUTE_WIDTH))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All answer-bearing fields are rendered; there is no declared latent alias.
    return []
