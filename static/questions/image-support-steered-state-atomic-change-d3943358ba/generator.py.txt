"""Deterministic renderer and analytic gold for a paired-panel state change."""

from __future__ import annotations

import random
from pathlib import Path
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-atomic-change-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

BACKGROUND = (248, 250, 252)
PANEL_FILL = (232, 241, 245)
PANEL_EDGE = (52, 72, 87)
TEXT = (45, 64, 78)
DISTRACTOR = (150, 161, 169)
STATE_COLORS = (
    (218, 78, 78),
    (63, 124, 211),
    (70, 166, 111),
    (224, 164, 48),
)
SLOT_LABELS = ("A", "B", "C", "D")

# The two panels are intentionally close: all answer-bearing marks occupy one
# compact inset, while the gray nuisance marks are placed outside it.
PANEL_STEP_X = 70
PANEL_STEP_Y = 64
PANEL_WIDTH = 48
SLOT_OFFSETS = ((14, 34), (32, 34), (14, 52), (32, 52))
PANEL_TOP = 12
PANEL_BOTTOM = 68
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(FONT_PATH, size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, ...], scale: int) -> list[int]:
    return [int(round(value * scale)) for value in values]


def _draw_centered(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: float,
    top_y: float,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    scale: int,
) -> None:
    bounds = draw.textbbox((0, 0), text, font=font)
    width = bounds[2] - bounds[0]
    draw.text(
        (int(round(center_x * scale - width / 2)), int(round(top_y * scale))),
        text,
        fill=TEXT,
        font=font,
    )


def _panel_origin(scene: dict[str, Any], panel: int) -> tuple[float, float]:
    x0, y0 = scene["origin"]
    return float(x0) + panel * PANEL_STEP_X, float(y0) + panel * PANEL_STEP_Y


def _changed_slots(scene: dict[str, Any]) -> list[int]:
    before = scene["before_states"]
    after = scene["after_states"]
    return [index for index, (left, right) in enumerate(zip(before, after)) if left != right]


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a balanced two-panel scene with exactly one state transition."""

    rng = random.Random(int(seed))
    # Consecutive scene indices cycle the answer identity, independent of the
    # random state palette, placement, size, and nuisance marks.
    target_hint = (int(seed) * 5 + 1) % len(SLOT_LABELS)
    before_states = [rng.randrange(len(STATE_COLORS)) for _ in SLOT_LABELS]
    after_states = list(before_states)
    change_rng = random.Random(int(seed) ^ 0xC0FFEE)
    delta = 1 + change_rng.randrange(len(STATE_COLORS) - 1)
    after_states[target_hint] = (before_states[target_hint] + delta) % len(STATE_COLORS)

    # Align each panel's compact four-glyph cluster near the center of a
    # different 8x8 ablation cell. The cell choices randomize absolute
    # position while keeping the two clusters close and diagonally separate.
    x0 = 137 + 64 * rng.randrange(3)
    y0 = 117 + 64 * rng.randrange(3)
    glyph_radius = rng.choice((5, 6))

    distractors: list[dict[str, int]] = []
    local_box = (
        x0 - 14,
        y0 - 8,
        x0 + PANEL_STEP_X + PANEL_WIDTH + 14,
        y0 + PANEL_STEP_Y + PANEL_BOTTOM + 10,
    )
    for _ in range(4 + rng.randrange(4)):
        for _attempt in range(200):
            px = rng.randint(24, WIDTH - 24)
            py = rng.randint(24, HEIGHT - 24)
            if not (local_box[0] <= px <= local_box[2] and local_box[1] <= py <= local_box[3]):
                distractors.append(
                    {
                        "x": px,
                        "y": py,
                        "kind": rng.randrange(2),
                        "size": 3 + rng.randrange(3),
                    }
                )
                break

    return {
        "before_states": before_states,
        "after_states": after_states,
        "origin": [x0, y0],
        "glyph_radius": glyph_radius,
        "distractors": distractors,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render two ordered panels and gray marks that are irrelevant to gold."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    small_font = _font(8 * ss)
    header_font = _font(9 * ss)

    for mark in scene["distractors"]:
        px, py, size = mark["x"], mark["y"], mark["size"]
        if mark["kind"] == 0:
            draw.ellipse(_box((px - size, py - size, px + size, py + size), ss), fill=DISTRACTOR)
        else:
            draw.polygon(
                [
                    (int(px * ss), int((py - size) * ss)),
                    (int((px + size) * ss), int(py * ss)),
                    (int(px * ss), int((py + size) * ss)),
                    (int((px - size) * ss), int(py * ss)),
                ],
                fill=DISTRACTOR,
            )

    for panel, states in enumerate((scene["before_states"], scene["after_states"])):
        px, py = _panel_origin(scene, panel)
        draw.rounded_rectangle(
            _box((px + 2, py + PANEL_TOP, px + PANEL_WIDTH - 2, py + PANEL_BOTTOM), ss),
            radius=7 * ss,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2 * ss,
        )
        _draw_centered(draw, "I" if panel == 0 else "F", px + PANEL_WIDTH / 2, py + 1, header_font, ss)
        for index, (offset, state) in enumerate(zip(SLOT_OFFSETS, states, strict=True)):
            cx, cy = px + offset[0], py + offset[1]
            _draw_centered(draw, SLOT_LABELS[index], cx, cy - 18, small_font, ss)
            radius = float(scene["glyph_radius"])
            draw.ellipse(
                _box((cx - radius, cy - radius, cx + radius, cy + radius), ss),
                fill=STATE_COLORS[int(state)],
                outline=PANEL_EDGE,
                width=max(1, ss),
            )

    if ss > 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the label of the one slot whose visible state changed."""

    changed = _changed_slots(scene)
    if len(changed) != 1:
        raise ValueError("scene does not contain exactly one state change")
    return SLOT_LABELS[changed[0]]


def margin(scene: dict[str, Any]) -> float:
    """Discrete one-sided margin: one valid change is separated from ambiguity."""

    return 1.0 if len(_changed_slots(scene)) == 1 else 0.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return len(_changed_slots(scene)) != 1


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No label-preserving latent alias is claimed for this scene encoding."""

    return []
