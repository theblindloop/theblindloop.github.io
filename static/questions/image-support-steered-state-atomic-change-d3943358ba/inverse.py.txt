"""Independent pixel-only inverse for the paired-panel state world.

This module deliberately reconstructs the two four-slot grids from RGB pixels.
It does not import the renderer or consume scene data, filenames, or metadata.
Gray remote marks and panel decoration are ignored; the eight colored state
components are the only evidence used for the decision.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


WIDTH = 512
HEIGHT = 512
STATE_COLORS = np.asarray(
    (
        (218, 78, 78),
        (63, 124, 211),
        (70, 166, 111),
        (224, 164, 48),
    ),
    dtype=np.float32,
)
SLOT_LABELS = ("A", "B", "C", "D")
# The pure and antialiased glyph interior is within this radius of its palette
# color; the nearest panel-edge blend is farther away and must not become a
# false state component.
COLOR_TOLERANCE = 48.0
MIN_COMPONENT_AREA = 24
MAX_COMPONENT_AREA = 220


class PixelOracleError(ValueError):
    """The raster does not expose one complete, unambiguous panel comparison."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Find 8-connected components without sharing renderer geometry."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dy and not dx:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) < MIN_COMPONENT_AREA or len(points) > MAX_COMPONENT_AREA:
            continue
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _recover_state_components(array: np.ndarray) -> list[tuple[float, float, int, int]]:
    found: list[tuple[float, float, int, int]] = []
    for state, color in enumerate(STATE_COLORS):
        distance = np.linalg.norm(array.astype(np.float32) - color, axis=2)
        for points in _components(distance <= COLOR_TOLERANCE):
            ys = points[:, 0]
            xs = points[:, 1]
            found.append((float(xs.mean()), float(ys.mean()), state, int(len(points))))
    return found


def _panel_states(items: list[tuple[float, float, int, int]]) -> tuple[int, int, int, int]:
    if len(items) != 4:
        raise PixelOracleError("each panel must expose exactly four state glyphs")
    by_x = sorted(items, key=lambda item: item[0])
    if by_x[2][0] - by_x[1][0] < 14.0:
        raise PixelOracleError("panel columns are merged or unresolved")
    columns = (by_x[:2], by_x[2:])
    result: list[int | None] = [None, None, None, None]
    for column, members in enumerate(columns):
        by_y = sorted(members, key=lambda item: item[1])
        if by_y[1][1] - by_y[0][1] < 14.0:
            raise PixelOracleError("panel rows are merged or unresolved")
        for row, item in enumerate(by_y):
            index = row * 2 + column
            if result[index] is not None:
                raise PixelOracleError("duplicate panel slot")
            result[index] = item[2]
    if any(value is None for value in result):
        raise PixelOracleError("panel slot assignment is incomplete")
    return tuple(int(value) for value in result)  # type: ignore[arg-type]


def decision_from_image(image: Image.Image) -> str:
    """Compare the left and right colored slot grids using pixels alone."""

    if image.size != (WIDTH, HEIGHT):
        raise PixelOracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    components = _recover_state_components(array)
    if len(components) != 8:
        raise PixelOracleError("the raster does not expose all eight state glyphs")

    ordered = sorted(components, key=lambda item: (item[0], item[1]))
    left = ordered[:4]
    right = ordered[4:]
    if right[0][0] - left[-1][0] < 24.0:
        raise PixelOracleError("left/right panel order is ambiguous")
    left_states = _panel_states(left)
    right_states = _panel_states(right)

    changed = [index for index, (before, after) in enumerate(zip(left_states, right_states)) if before != after]
    if len(changed) != 1:
        raise PixelOracleError("pixel replay found zero or multiple state changes")
    return SLOT_LABELS[changed[0]]
