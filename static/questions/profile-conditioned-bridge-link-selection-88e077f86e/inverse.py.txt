"""Independent raster-only inverse for the bridge-link world.

Only the final RGB image is accepted.  This module duplicates the public
diagram geometry, segments visible link strokes, reconstructs their node
adjacency, and computes component changes after edge removal.  It does not
import renderer, prompts, generate, verify, or analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


WIDTH = 1100
HEIGHT = 760
CENTER = (550.0, 400.0)
BASE_NODES = (
    (320.0, 290.0),
    (320.0, 510.0),
    (470.0, 400.0),
    (630.0, 400.0),
    (780.0, 290.0),
    (780.0, 510.0),
)
EDGES = (
    (0, 1),
    (1, 2),
    (2, 0),
    (3, 4),
    (4, 5),
    (5, 3),
    (2, 3),
)
COLOR_RGB = {
    "red": np.asarray((218, 65, 74), dtype=np.float32),
    "blue": np.asarray((49, 112, 205), dtype=np.float32),
    "green": np.asarray((48, 154, 99), dtype=np.float32),
}


def _positions(variant: int) -> tuple[tuple[int, int], ...]:
    result = []
    cx, cy = CENTER
    for point in BASE_NODES:
        x, y = point
        if int(variant) % 2:
            x, y = cx - (y - cy), cy + (x - cx)
        result.append((int(round(x)), int(round(y))))
    return tuple(result)


def _sample_strip(array: np.ndarray, first: tuple[int, int], second: tuple[int, int]) -> np.ndarray:
    values = []
    x0, y0 = first
    x1, y1 = second
    for t in np.linspace(0.20, 0.80, 36):
        x = int(round(x0 + (x1 - x0) * float(t)))
        y = int(round(y0 + (y1 - y0) * float(t)))
        for dy in range(-2, 3):
            for dx in range(-2, 3):
                yy = min(max(y + dy, 0), array.shape[0] - 1)
                xx = min(max(x + dx, 0), array.shape[1] - 1)
                values.append(array[yy, xx])
    return np.asarray(values, dtype=np.float32)


def _edge_signal(array: np.ndarray, first: tuple[int, int], second: tuple[int, int]) -> float:
    pixels = _sample_strip(array, first, second)
    # The page and panel are light; every visible link has a dark or saturated
    # centerline in this strip.
    return float(np.mean(np.min(pixels, axis=1) < 210.0))


def _color_signal(
    array: np.ndarray,
    first: tuple[int, int],
    second: tuple[int, int],
    target: np.ndarray,
) -> float:
    pixels = _sample_strip(array, first, second)
    distance = np.sqrt(((pixels - target) ** 2).sum(axis=1))
    return float(np.mean(distance < 92.0))


def _layout(array: np.ndarray) -> tuple[tuple[tuple[int, int], ...], list[float]]:
    candidates = []
    for variant in (0, 1):
        positions = _positions(variant)
        signals = [
            _edge_signal(array, positions[left], positions[right])
            for left, right in EDGES
        ]
        candidates.append((positions, signals))
    candidates.sort(key=lambda item: sum(value > 0.38 for value in item[1]), reverse=True)
    positions, signals = candidates[0]
    if sum(value > 0.38 for value in signals) < len(EDGES):
        raise ValueError("could not recover all visible network links")
    return positions, signals


def _component_count(active_edges: list[int], removed: int | None = None) -> int:
    adjacency = [[] for _ in range(len(BASE_NODES))]
    for edge_index in active_edges:
        if edge_index == removed:
            continue
        left, right = EDGES[edge_index]
        adjacency[left].append(right)
        adjacency[right].append(left)
    seen = [False] * len(adjacency)
    components = 0
    for start in range(len(adjacency)):
        if seen[start]:
            continue
        components += 1
        seen[start] = True
        queue = deque([start])
        while queue:
            node = queue.popleft()
            for neighbor in adjacency[node]:
                if not seen[neighbor]:
                    seen[neighbor] = True
                    queue.append(neighbor)
    return components


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the unique colored edge that disconnects the graph."""

    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected image size {rgb.size}")
    array = np.asarray(rgb)
    positions, signals = _layout(array)
    if min(signals) < 0.38:
        raise ValueError("one or more links are too weak to classify")

    edge_colors: dict[int, str] = {}
    for edge_index, (left, right) in enumerate(EDGES):
        scores = {
            name: _color_signal(array, positions[left], positions[right], target)
            for name, target in COLOR_RGB.items()
        }
        name, score = max(scores.items(), key=lambda item: item[1])
        if score >= 0.28:
            edge_colors[edge_index] = name

    if len(edge_colors) != 3 or len(set(edge_colors.values())) != 3:
        raise ValueError(f"expected three distinct colored links, recovered {edge_colors}")

    active_edges = list(range(len(EDGES)))
    baseline = _component_count(active_edges)
    bridge_edges = [
        edge_index
        for edge_index in edge_colors
        if _component_count(active_edges, removed=edge_index) > baseline
    ]
    if len(bridge_edges) != 1:
        raise ValueError(f"expected one colored bridge, found {bridge_edges}")
    return edge_colors[bridge_edges[0]]
