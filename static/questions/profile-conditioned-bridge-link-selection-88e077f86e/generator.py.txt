"""Renderer and analytic answer for the bridge-link topology world."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1100
HEIGHT = 760
SCALE = 3
CENTER = (550.0, 400.0)

# Two triangles joined by one edge.  The bridge has no leaf endpoint: both of
# its endpoints also belong to a cycle, so bridge status is a global property.
BASE_NODES = (
    (320.0, 290.0),
    (320.0, 510.0),
    (470.0, 400.0),
    (630.0, 400.0),
    (780.0, 290.0),
    (780.0, 510.0),
)
EDGES = (
    (0, 1),
    (1, 2),
    (2, 0),
    (3, 4),
    (4, 5),
    (5, 3),
    (2, 3),
)
BRIDGE_EDGE = 6
RING_EDGES = tuple(range(6))

COLORS = {
    "red": (218, 65, 74),
    "blue": (49, 112, 205),
    "green": (48, 154, 99),
}
BASE_INK = (53, 64, 75)
TEXT = (38, 47, 57)
SUBTEXT = (92, 103, 114)
PAGE = (245, 247, 249)
PANEL = (255, 255, 255)
PANEL_EDGE = (172, 181, 190)
NODE_FILL = (250, 251, 248)


def _font(size: int):
    try:
        return ImageFont.load_default(size=size * SCALE)
    except TypeError:
        return ImageFont.load_default()


def _rotate(point: tuple[float, float], quarter_turns: int) -> tuple[int, int]:
    x, y = point
    cx, cy = CENTER
    for _ in range(int(quarter_turns) % 4):
        x, y = cx - (y - cy), cy + (x - cx)
    return int(round(x)), int(round(y))


def node_positions(layout_variant: int) -> tuple[tuple[int, int], ...]:
    """Return one of two visible 90-degree orientations of the same graph."""

    return tuple(_rotate(point, int(layout_variant) % 2) for point in BASE_NODES)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    ring_targets = sorted(rng.sample(RING_EDGES, 2))
    target_edges = [BRIDGE_EDGE, *ring_targets]
    labels = ["red", "blue", "green"]
    rng.shuffle(labels)
    edge_colors = {str(edge): label for edge, label in zip(target_edges, labels)}
    return {
        "edge_colors": edge_colors,
        "ring_targets": ring_targets,
        "layout_variant": rng.randrange(2),
    }


def _scaled_point(point: tuple[int, int]) -> tuple[int, int]:
    return (point[0] * SCALE, point[1] * SCALE)


def _draw_link(
    draw: ImageDraw.ImageDraw,
    first: tuple[int, int],
    second: tuple[int, int],
    color: tuple[int, int, int],
    width: int,
) -> None:
    p1 = _scaled_point(first)
    p2 = _scaled_point(second)
    draw.line((p1, p2), fill=color, width=width * SCALE)
    radius = (width * SCALE) // 2
    for x, y in (p1, p2):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _draw_node(draw: ImageDraw.ImageDraw, point: tuple[int, int]) -> None:
    x, y = _scaled_point(point)
    outer = 28 * SCALE
    draw.ellipse(
        (x - outer, y - outer, x + outer, y + outer),
        fill=NODE_FILL,
        outline=BASE_INK,
        width=5 * SCALE,
    )
    dot = 5 * SCALE
    draw.ellipse((x - dot, y - dot, x + dot, y + dot), fill=BASE_INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAGE)
    draw = ImageDraw.Draw(image)
    draw.text((32 * SCALE, 24 * SCALE), "BRIDGE LINK", fill=TEXT, font=_font(25))
    draw.text(
        (34 * SCALE, 66 * SCALE),
        "One colored link disconnects the network when removed; ignore the frame and text.",
        fill=SUBTEXT,
        font=_font(13),
    )
    frame = (118 * SCALE, 122 * SCALE, 982 * SCALE, 692 * SCALE)
    draw.rounded_rectangle(frame, radius=16 * SCALE, fill=PANEL, outline=PANEL_EDGE, width=2 * SCALE)

    positions = node_positions(int(scene["layout_variant"]))
    edge_colors = scene["edge_colors"]
    for edge_index, (left, right) in enumerate(EDGES):
        _draw_link(draw, positions[left], positions[right], BASE_INK, 18)
        color_name = edge_colors.get(str(edge_index))
        if color_name is not None:
            _draw_link(draw, positions[left], positions[right], COLORS[color_name], 11)
    for point in positions:
        _draw_node(draw, point)

    draw.text(
        (142 * SCALE, 655 * SCALE),
        "Trace alternate routes between the outlined nodes.",
        fill=SUBTEXT,
        font=_font(12),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    edge_colors = scene["edge_colors"]
    answer = edge_colors.get(str(BRIDGE_EDGE))
    if answer not in COLORS:
        raise ValueError("scene must color the unique bridge")
    return str(answer)


def margin(scene: dict) -> float:
    """The bridge deletion changes one component; each ring deletion changes zero."""

    if is_quarantined(scene):
        return 0.0
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["layout_variant"]) not in (0, 1):
            return True
        ring_targets = tuple(int(value) for value in scene["ring_targets"])
        if len(ring_targets) != 2 or len(set(ring_targets)) != 2:
            return True
        if any(value not in RING_EDGES for value in ring_targets):
            return True
        edge_colors = scene["edge_colors"]
        if set(edge_colors) != {str(BRIDGE_EDGE), *(str(value) for value in ring_targets)}:
            return True
        if set(edge_colors.values()) != set(COLORS):
            return True
    except (KeyError, TypeError, ValueError):
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene encoding has no declared render-invariant latent alias."""

    return []
