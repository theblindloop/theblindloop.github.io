"""Pixel-only inverse for the compact saturating-counter console.

The implementation locates the translated saturated console, identifies the
yellow start dot, measures all six triangle orientations, counts the center
meter bars, and independently replays the bounded state transitions.  It has
no access to renderer state or analytic gold.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PAD_CENTERS = ((48, 15), (76, 31), (76, 65), (48, 81), (20, 65), (20, 31))
MARKER_CENTERS = ((48, 4), (87, 23), (87, 73), (48, 92), (9, 73), (9, 23))


def _decode(image: Image.Image) -> tuple[int, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise ValueError("expected a 512 by 512 RGB raster")

    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    saturated = spread > 35
    ys, xs = np.nonzero(saturated)
    if len(xs) < 7000:
        raise ValueError("compact console is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 + 1 != 96 or y1 - y0 + 1 != 96:
        raise ValueError("console bounds are incomplete or ambiguous")

    # The square's four substantial corners form a connected integrity cue.
    # This rejects cropped/ablated consoles before any state is guessed.
    for xa, ya in ((2, 2), (82, 2), (2, 82), (82, 82)):
        patch = saturated[y0 + ya : y0 + ya + 12, x0 + xa : x0 + xa + 12]
        if patch.shape != (12, 12) or float(patch.mean()) < 0.88:
            raise ValueError("console integrity frame is incomplete")

    marker = (
        (array[:, :, 0] > 235)
        & (array[:, :, 1] > 160)
        & (array[:, :, 1] < 220)
        & (array[:, :, 2] < 90)
    )
    my, mx = np.nonzero(marker)
    if not 20 <= len(mx) <= 60:
        raise ValueError("start dot is missing or ambiguous")
    marker_point = (float(mx.mean() - x0), float(my.mean() - y0))
    distances = [math.dist(marker_point, point) for point in MARKER_CENTERS]
    order = np.argsort(distances)
    if distances[int(order[0])] > 2.0 or distances[int(order[1])] - distances[int(order[0])] < 8.0:
        raise ValueError("cannot resolve the first command")
    first_index = int(order[0])

    command_by_position: dict[int, str] = {}
    orientation_margins = []
    for position, (cx_rel, cy_rel) in enumerate(PAD_CENTERS):
        cx, cy = x0 + cx_rel, y0 + cy_rel
        crop = array[cy - 7 : cy + 8, cx - 7 : cx + 8]
        dark = np.all(crop < 32, axis=2)
        py, px = np.nonzero(dark)
        if not 55 <= len(px) <= 115:
            raise ValueError("a command triangle is missing or malformed")
        signed = float(py.mean() - 7.0)
        if abs(signed) < 1.0:
            raise ValueError("triangle orientation is tied")
        command_by_position[position] = "up" if signed > 0 else "down"
        orientation_margins.append(abs(signed))

    bars = []
    for index in range(4):
        y = y0 + 37 + index * 7
        patch = array[y : y + 3, x0 + 45 : x0 + 52].astype(np.float32)
        mean = patch.mean(axis=(0, 1))
        if 224 <= float(mean.mean()) <= 244 and float(mean.max() - mean.min()) < 12:
            bars.append(True)
        elif 95 <= float(mean.mean()) <= 135 and float(mean.max() - mean.min()) < 35:
            bars.append(False)
        else:
            raise ValueError("a meter bar is missing or ambiguous")
    if bars not in ([True, False, False, False], [True, True, False, False], [True, True, True, False], [True, True, True, True]):
        raise ValueError("meter lights are not a valid contiguous level")
    level = sum(bars)

    ordered_commands = [command_by_position[(first_index + step) % 6] for step in range(6)]
    if ordered_commands.count("up") != 3:
        raise ValueError("the balanced six-command contract is broken")
    for command in ordered_commands:
        level = min(4, level + 1) if command == "up" else max(1, level - 1)
    return level, min(orientation_margins)


def decision_from_image(image: Image.Image) -> str:
    level, _pixel_margin = _decode(image)
    return str(level)
