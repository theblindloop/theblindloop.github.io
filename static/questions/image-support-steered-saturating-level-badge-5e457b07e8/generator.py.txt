"""Sampler, analytic replay, and deterministic raster renderer."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BADGE_SIZE = 96

# Clockwise, beginning at twelve o'clock.  These are console-relative slots,
# not absolute scene coordinates; the console is translated between scenes.
PAD_CENTERS = ((48, 15), (76, 31), (76, 65), (48, 81), (20, 65), (20, 31))
MARKER_CENTERS = ((48, 4), (87, 23), (87, 73), (48, 92), (9, 73), (9, 23))

THEMES = (
    {"outer": (38, 62, 104), "pad": (52, 181, 176)},
    {"outer": (91, 48, 109), "pad": (86, 170, 107)},
    {"outer": (30, 91, 94), "pad": (201, 132, 55)},
)
BACKGROUND = (248, 247, 242)
DECOR = (211, 214, 218)
TRACK = (168, 183, 197)
INK = (14, 19, 24)
MARKER = (250, 191, 48)
LIT = (238, 238, 229)
UNLIT = (105, 115, 130)


def _replay(start_level: int, commands: list[str] | tuple[str, ...]) -> int:
    level = int(start_level)
    for command in commands:
        if command == "up":
            level = min(4, level + 1)
        elif command == "down":
            level = max(1, level - 1)
        else:
            raise ValueError(f"unknown command {command!r}")
    return level


def _eligible_programs(start_level: int, final_level: int) -> list[tuple[str, ...]]:
    programs = []
    for bits in itertools.product(("down", "up"), repeat=6):
        if bits.count("up") == 3 and _replay(start_level, bits) == final_level:
            programs.append(bits)
    return programs


def _decorations(rng: random.Random, badge_x: int, badge_y: int) -> list[dict]:
    lattice = []
    for y in range(28, HEIGHT - 27, 48):
        for x in range(28, WIDTH - 27, 48):
            if badge_x - 24 <= x <= badge_x + BADGE_SIZE + 24:
                if badge_y - 24 <= y <= badge_y + BADGE_SIZE + 24:
                    continue
            lattice.append((x, y))
    rng.shuffle(lattice)
    kinds = ("ring", "diamond", "cross")
    return [
        {"x": x, "y": y, "kind": kinds[rng.randrange(len(kinds))], "size": 5 + rng.randrange(3)}
        for x, y in lattice[:12]
    ]


def sample_scene(seed: int) -> dict:
    """Sample balanced final states and fixed-ink six-command programs."""

    rng = random.Random((int(seed) * 1_000_003) ^ 0x51A7C0DE)
    final_level = int(seed) % 4 + 1
    start_level = rng.randrange(1, 5)
    programs = _eligible_programs(start_level, final_level)
    commands = list(programs[rng.randrange(len(programs))])
    grid_col = rng.randrange(1, 7)
    grid_row = rng.randrange(1, 7)
    badge_x = grid_col * 64 - 16
    badge_y = grid_row * 64 - 16
    return {
        "start_level": start_level,
        "commands": commands,
        "first_index": rng.randrange(6),
        "grid_col": grid_col,
        "grid_row": grid_row,
        "theme": rng.randrange(len(THEMES)),
        "decorations": _decorations(rng, badge_x, badge_y),
    }


def analytic_gold(scene: dict) -> str:
    return str(_replay(scene["start_level"], scene["commands"]))


def margin(scene: dict) -> float:
    """Nominal pixel runner-up margin for triangle orientation decoding."""

    # The two orientations have centroids on opposite sides of the pad centre.
    # Nominal rasterization leaves at least two pixels of centroid separation.
    return 2.0


def is_quarantined(scene: dict) -> bool:
    # The sampler has no continuous near-boundary mode: every glyph is full
    # scale.  The inverse arm abstains on raster corruption instead.
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_unordered_decorations", twin)]


def _draw_decoration(draw: ImageDraw.ImageDraw, item: dict) -> None:
    x, y, size = int(item["x"]), int(item["y"]), int(item["size"])
    kind = item["kind"]
    if kind == "ring":
        draw.ellipse((x - size, y - size, x + size, y + size), outline=DECOR, width=2)
    elif kind == "diamond":
        draw.polygon(((x, y - size), (x + size, y), (x, y + size), (x - size, y)), outline=DECOR)
    elif kind == "cross":
        draw.line((x - size, y, x + size, y), fill=DECOR, width=2)
        draw.line((x, y - size, x, y + size), fill=DECOR, width=2)
    else:
        raise ValueError(f"unknown decoration {kind!r}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Access every declared visual latent explicitly; this also keeps the
    # rendered-field audit transparent.
    start_level = int(scene["start_level"])
    commands = list(scene["commands"])
    first_index = int(scene["first_index"])
    grid_col = int(scene["grid_col"])
    grid_row = int(scene["grid_row"])
    theme_index = int(scene["theme"])
    decorations = list(scene["decorations"])
    if not 1 <= start_level <= 4 or len(commands) != 6:
        raise ValueError("invalid counter scene")
    if not 0 <= first_index < 6 or not 0 <= theme_index < len(THEMES):
        raise ValueError("invalid visible style latent")

    for item in decorations:
        _draw_decoration(draw, item)

    x0 = grid_col * 64 - 16
    y0 = grid_row * 64 - 16
    theme = THEMES[theme_index]
    draw.rectangle((x0, y0, x0 + 95, y0 + 95), fill=theme["outer"])

    absolute_pads = [(x0 + x, y0 + y) for x, y in PAD_CENTERS]
    draw.line(absolute_pads + [absolute_pads[0]], fill=TRACK, width=2, joint="curve")

    command_by_position: dict[int, str] = {}
    for step, command in enumerate(commands):
        command_by_position[(first_index + step) % 6] = command

    for position, (cx_rel, cy_rel) in enumerate(PAD_CENTERS):
        cx, cy = x0 + cx_rel, y0 + cy_rel
        draw.rounded_rectangle((cx - 8, cy - 8, cx + 8, cy + 8), radius=3, fill=theme["pad"])
        if command_by_position[position] == "up":
            vertices = ((cx, cy - 6), (cx - 6, cy + 5), (cx + 6, cy + 5))
        else:
            vertices = ((cx - 6, cy - 5), (cx + 6, cy - 5), (cx, cy + 6))
        draw.polygon(vertices, fill=INK)

    marker_x, marker_y = MARKER_CENTERS[first_index]
    draw.ellipse(
        (x0 + marker_x - 3, y0 + marker_y - 3, x0 + marker_x + 3, y0 + marker_y + 3),
        fill=MARKER,
    )

    # Four center bars show the starting level.  Lit and unlit bars have equal
    # geometry, preventing total meter area from becoming an answer style.
    draw.rounded_rectangle((x0 + 39, y0 + 32, x0 + 57, y0 + 65), radius=4, fill=INK)
    for index in range(4):
        y = y0 + 36 + index * 7
        fill = LIT if index < start_level else UNLIT
        draw.rectangle((x0 + 43, y, x0 + 53, y + 4), fill=fill)

    return image
