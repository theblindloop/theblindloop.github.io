"""Independent final-PNG inverse arm for Pearson-correlation selection."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

_LABELS = ("A", "B", "C", "D")
_EXPECTED_POINTS = 32


def _components(mask: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    result: list[tuple[np.ndarray, np.ndarray]] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if visited[sy, sx]:
            continue
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        visited[sy, sx] = True
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for ny, nx in (
                (y - 1, x - 1), (y - 1, x), (y - 1, x + 1),
                (y, x - 1), (y, x + 1),
                (y + 1, x - 1), (y + 1, x), (y + 1, x + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        result.append((np.asarray(ys), np.asarray(xs)))
    return result


def _split_threshold(values: list[float]) -> float:
    ordered = sorted(values)
    gaps = [(ordered[index + 1] - ordered[index], index) for index in range(len(ordered) - 1)]
    gap, index = max(gaps)
    if gap < 45:
        raise ValueError("scatter panels are not visibly separated")
    return (ordered[index] + ordered[index + 1]) / 2.0


def _coefficient(points: list[tuple[float, float]]) -> float:
    if len(points) != 8:
        raise ValueError("each scatter panel must contain exactly eight markers")
    points = sorted(points)
    xs = [point[0] for point in points]
    # Raster rows grow downward, while the visibly labelled y-axis grows upward.
    ys = [-point[1] for point in points]
    if min(np.diff(xs)) < 18:
        raise ValueError("panel markers do not expose eight distinct x coordinates")
    if len({round(value, 1) for value in ys}) != 8:
        raise ValueError("panel markers do not expose eight distinct y coordinates")
    mean_x = sum(xs) / len(xs)
    mean_y = sum(ys) / len(ys)
    numerator = sum((x - mean_x) * (y - mean_y) for x, y in zip(xs, ys, strict=True))
    denominator = math.sqrt(
        sum((x - mean_x) ** 2 for x in xs) * sum((y - mean_y) ** 2 for y in ys)
    )
    if denominator <= 0:
        raise ValueError("recovered coordinates are constant")
    return numerator / denominator


def decision_from_image(image: Image.Image) -> str:
    """Recover all marker centroids and select the reading-order panel with max r."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3 or array.shape[0] < 500 or array.shape[1] < 650:
        raise ValueError("unsupported raster")
    channel_max = array.max(axis=2)
    channel_min = array.min(axis=2)
    saturated = (channel_max - channel_min >= 70) & (channel_max >= 120) & (channel_min <= 140)

    centers: list[tuple[float, float]] = []
    for ys, xs in _components(saturated):
        count = len(xs)
        box_width = int(xs.max() - xs.min() + 1)
        box_height = int(ys.max() - ys.min() + 1)
        if 80 <= count <= 110 and 10 <= box_width <= 12 and 10 <= box_height <= 12:
            centers.append((float(xs.mean()), float(ys.mean())))
    if len(centers) != _EXPECTED_POINTS:
        raise ValueError(f"expected 32 isolated scatter markers, found {len(centers)}")

    split_x = _split_threshold([point[0] for point in centers])
    split_y = _split_threshold([point[1] for point in centers])
    panels: list[list[tuple[float, float]]] = [[], [], [], []]
    for x, y in centers:
        panel = (2 if y > split_y else 0) + (1 if x > split_x else 0)
        panels[panel].append((x, y))

    coefficients = [_coefficient(points) for points in panels]
    ranked = sorted(((value, panel) for panel, value in enumerate(coefficients)), reverse=True)
    if ranked[0][0] - ranked[1][0] < 0.02:
        raise ValueError("pixel Pearson leader is tied or ambiguous")
    return _LABELS[ranked[0][1]]
