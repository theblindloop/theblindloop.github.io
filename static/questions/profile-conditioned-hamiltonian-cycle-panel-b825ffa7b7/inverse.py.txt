"""Independent raster-only inverse for Hamiltonian-cycle panel selection."""

from __future__ import annotations

from collections import deque
from math import hypot

import numpy as np
from PIL import Image

LABELS = ("left", "center", "right")


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = np.fromiter((p[0] for p in pixels), dtype=float)
            xs = np.fromiter((p[1] for p in pixels), dtype=float)
            output.append((float(xs.mean()), float(ys.mean()), len(pixels)))
    return output


def _line_score(link_mask: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> float:
    x0, y0 = first
    x1, y1 = second
    length = hypot(x1 - x0, y1 - y0)
    samples = max(20, int(length))
    hits = 0
    total = 0
    for step in range(samples + 1):
        t = 0.20 + 0.60 * step / samples
        x = int(round(x0 + t * (x1 - x0)))
        y = int(round(y0 + t * (y1 - y0)))
        patch = link_mask[max(0, y - 2): y + 3, max(0, x - 2): x + 3]
        hits += int(bool(patch.any()))
        total += 1
    return hits / total


def _has_hamiltonian(adjacency: list[set[int]]) -> bool:
    count = len(adjacency)

    def visit(path: list[int], used: set[int]) -> bool:
        if len(path) == count:
            return path[0] in adjacency[path[-1]]
        return any(visit(path + [nxt], used | {nxt}) for nxt in adjacency[path[-1]] - used)

    return visit([0], {0})


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    cyan = (red < 65) & (green > 130) & (green < 220) & (blue > 145) & (blue < 230)
    link = (red > 75) & (red < 135) & (green > 85) & (green < 145) & (blue > 90) & (blue < 155)
    stations = _components(cyan, minimum=150)
    if len(stations) != 18:
        raise ValueError(f"expected 18 visible stations, found {len(stations)}")
    panel_nodes: list[list[tuple[float, float]]] = [[], [], []]
    for x, y, _area in stations:
        index = min(2, max(0, int(x * 3 // image.width)))
        panel_nodes[index].append((x, y))
    winners = []
    for panel_index, nodes in enumerate(panel_nodes):
        if len(nodes) != 6:
            raise ValueError(f"panel {panel_index} has {len(nodes)} stations")
        nodes.sort(key=lambda point: (point[1], point[0]))
        adjacency = [set() for _ in nodes]
        scores = []
        for a in range(6):
            for b in range(a + 1, 6):
                score = _line_score(link, nodes[a], nodes[b])
                scores.append(score)
                if score > 0.82:
                    adjacency[a].add(b)
                    adjacency[b].add(a)
        edge_count = sum(map(len, adjacency)) // 2
        if edge_count != 7:
            raise ValueError(f"panel {panel_index} recovered {edge_count} links")
        if _has_hamiltonian(adjacency):
            winners.append(panel_index)
    if len(winners) != 1:
        raise ValueError(f"expected one Hamiltonian panel, found {winners}")
    return LABELS[winners[0]]
