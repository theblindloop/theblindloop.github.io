"""Atomic cross-panel change identification (state tracking).

The image shows two panels side by side: panel A (left) and panel B (right).
Both panels contain the SAME four distinctly colored disks at the SAME fixed
slot positions (read left to right: red, green, blue, purple). In each panel
every disk is independently either FILLED (a solid disk) or HOLLOW (a ring with
a white centre). Exactly ONE disk changes its fill state between panel A and
panel B; all other disks are pixel-identical across the two panels.

The task is to report WHICH disk (by its color) underwent the single declared
visual state change. Because fills differ between the disks within each panel,
no single frame reveals the changed disk on its own; the answer is recovered
only by comparing the two panels (cross-panel state tracking).

The latent scene stores ``start_fills`` (panel A fill bits) and ``end_fills``
(panel B fill bits). The answer is the color of the single slot whose fill bit
differs between the two lists.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 880
HEIGHT = 240
PANEL_W = 400
PANEL_H = 160
PANEL_TOP = 40
PANEL_A_LEFT = 30
PANEL_B_LEFT = 450
DISK_R = 24
RING_W = 7
CENTER_Y = PANEL_TOP + PANEL_H // 2

BG = (255, 255, 255)
BORDER = (200, 200, 200)

COLOR_NAMES = ["red", "green", "blue", "purple"]
COLORS = {
    "red": (220, 40, 40),
    "green": (40, 170, 80),
    "blue": (40, 80, 220),
    "purple": (150, 60, 190),
}


def _slot_centers(panel_left: int) -> list[tuple[int, int]]:
    centers = []
    for i in range(4):
        x = panel_left + PANEL_W * (i + 0.5) / 4
        centers.append((int(x), CENTER_Y))
    return centers


def _draw_disk(draw: ImageDraw.ImageDraw, center: tuple[int, int], color, filled: bool) -> None:
    cx, cy = center
    box = [cx - DISK_R, cy - DISK_R, cx + DISK_R, cy + DISK_R]
    if filled:
        draw.ellipse(box, fill=color)
    else:
        draw.ellipse(box, fill=BG, outline=color, width=RING_W)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    start = [rng.randint(0, 1) for _ in range(4)]
    changed = rng.randrange(4)
    end = list(start)
    end[changed] = 1 - end[changed]
    return {"start_fills": start, "end_fills": end}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    start = scene["start_fills"]
    end = scene["end_fills"]

    draw.rectangle(
        [PANEL_A_LEFT, PANEL_TOP, PANEL_A_LEFT + PANEL_W, PANEL_TOP + PANEL_H],
        outline=BORDER,
        width=3,
    )
    draw.rectangle(
        [PANEL_B_LEFT, PANEL_TOP, PANEL_B_LEFT + PANEL_W, PANEL_TOP + PANEL_H],
        outline=BORDER,
        width=3,
    )

    for i, name in enumerate(COLOR_NAMES):
        color = COLORS[name]
        a_center = _slot_centers(PANEL_A_LEFT)[i]
        b_center = _slot_centers(PANEL_B_LEFT)[i]
        _draw_disk(draw, a_center, color, bool(start[i]))
        _draw_disk(draw, b_center, color, bool(end[i]))
    return img


def analytic_gold(scene: dict) -> str:
    start = scene["start_fills"]
    end = scene["end_fills"]
    for i in range(4):
        if start[i] != end[i]:
            return COLOR_NAMES[i]
    return COLOR_NAMES[-1]


def margin(scene: dict) -> float:
    """Robust discrete margin: number of pixels that differ between the two
    panels (a single disk's fill toggles, giving a large positive area)."""
    start = scene["start_fills"]
    end = scene["end_fills"]
    img_a = render({"start_fills": start, "end_fills": list(start)})
    img_b = render({"start_fills": start, "end_fills": end})
    from PIL import ImageChops

    diff = ImageChops.difference(img_a, img_b)
    bbox = diff.getbbox()
    if bbox is None:
        return 0.0
    region = diff.crop(bbox)
    # Count pixels that actually changed colour (non-black).
    count = sum(1 for pixel in region.getdata() if pixel != (0, 0, 0))
    return float(count)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: start_fills and
    end_fills directly determine the drawn disk fills in each panel, and any
    reparametrization that changed which slot differs would change the pixels.
    Permuting panel A and panel B would change the image; permuting slots would
    reorder the colours, changing the raster."""
    return []
