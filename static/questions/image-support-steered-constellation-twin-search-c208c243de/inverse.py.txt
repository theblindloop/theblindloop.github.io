"""Independent image-only inverse arm for twin-constellation search."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (216, 55, 62),
    "blue": (45, 105, 210),
    "green": (18, 151, 92),
    "purple": (145, 63, 191),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        stack = [(y0, x0)]
        seen[y0, x0] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 25:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _pattern_for_side(hub: np.ndarray, dots: list[np.ndarray]) -> frozenset[tuple[int, int]]:
    hy, hx = hub.mean(axis=0)
    pattern = set()
    for dot in dots:
        dy, dx = dot.mean(axis=0) - np.asarray([hy, hx])
        qx = 0 if abs(dx) < 9 else (1 if dx > 0 else -1)
        qy = 0 if abs(dy) < 9 else (1 if dy > 0 else -1)
        if (qx, qy) == (0, 0) or abs(dx) > 28 or abs(dy) > 28:
            raise ValueError("dot is not on a visible outer grid cell")
        pattern.add((qx, qy))
    if len(pattern) != 4:
        raise ValueError("card does not expose four distinct dot cells")
    return frozenset(pattern)


def _measure_color(array: np.ndarray, rgb: tuple[int, int, int]) -> tuple[frozenset, frozenset]:
    distance = np.linalg.norm(array - np.asarray(rgb, dtype=np.float32), axis=2)
    comps = _components(distance < 64.0)
    hubs = []
    dots = []
    for points in comps:
        ys, xs = points[:, 0], points[:, 1]
        height = ys.max() - ys.min() + 1
        width = xs.max() - xs.min() + 1
        area = len(points)
        if 38 <= area <= 90 and 8 <= width <= 15 and 8 <= height <= 15:
            hubs.append(points)
        elif 92 <= area <= 180 and 10 <= width <= 15 and 10 <= height <= 15:
            dots.append(points)
    if len(hubs) != 2 or len(dots) != 8:
        raise ValueError("color does not expose exactly two alignment pluses and eight dots")
    hubs.sort(key=lambda points: float(points[:, 1].mean()))
    divider = (hubs[0][:, 1].mean() + hubs[1][:, 1].mean()) / 2.0
    left_dots = [points for points in dots if points[:, 1].mean() < divider]
    right_dots = [points for points in dots if points[:, 1].mean() >= divider]
    if len(left_dots) != 4 or len(right_dots) != 4:
        raise ValueError("dot groups cannot be split into two four-dot cards")
    return _pattern_for_side(hubs[0], left_dots), _pattern_for_side(hubs[1], right_dots)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    mismatches = {}
    for color, rgb in PALETTE.items():
        left, right = _measure_color(array, rgb)
        mismatches[color] = len(left ^ right)
    winners = [color for color, value in mismatches.items() if value == 0]
    if len(winners) != 1:
        raise ValueError("image does not expose one exact constellation twin")
    winner = winners[0]
    runner = min(value for color, value in mismatches.items() if color != winner)
    if runner < 2:
        raise ValueError("insufficient runner-up mismatch margin")
    return winner
