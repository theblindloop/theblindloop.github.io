"""Renderer for `wind_flags_same_side`.

One picture, two flagpoles.  Each pole carries one triangular flag whose cloth
reaches sideways from the pole by a signed number of pixels.  The decision is
whether the two flags reach out to the SAME side of their own poles or to
OPPOSITE sides.

The canonical prior is that one wind blows across a whole scene, so every wind
indicator in a picture leans the same way; the counterfactual paints two flags
disagreeing.  The prior is recorded for analysis only.  Gold is always the pair
of signed reaches actually painted into the raster.

Everything is drawn as flat blocks of one exact colour with no antialiasing, so
the raster is byte-deterministic and the pixel-only inverse arm can segment the
page by exact colour.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "wind_flags_same_side-renderer-0.1.0"

DECISIONS = ("same", "opposite")

# Declared in the question: no flag in this set reaches less than 40 px sideways.
QUARANTINE_MARGIN = 24.0

WIDTH, HEIGHT = 660, 440
GROUND_Y = 392
POLE_W = 9

FLAG_INKS = [
    [200, 40, 40],
    [30, 90, 200],
    [20, 140, 70],
    [225, 150, 20],
    [130, 50, 180],
    [0, 150, 160],
]
POLE_INK = [55, 55, 55]
FURNITURE_INK = [170, 170, 170]
CLOUD_INK = [208, 208, 208]


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    """Sample one scene.  Every field below is painted into the raster."""
    rng = random.Random(seed)

    pc = [rng.randrange(150, 196), rng.randrange(455, 511)]
    pole_top = [rng.randrange(70, 151), rng.randrange(70, 151)]
    y0 = [pole_top[0] + rng.randrange(8, 27), pole_top[1] + rng.randrange(8, 27)]
    wa = [rng.randrange(46, 75), rng.randrange(46, 75)]
    reach = [rng.randrange(58, 106), rng.randrange(58, 106)]
    droop = [rng.randrange(-6, 21), rng.randrange(-6, 21)]

    inks = rng.sample(FLAG_INKS, 2)

    # The class and the side of the first flag are drawn AFTER every cheap cue
    # above, so palette, pole heights, pole positions, flag sizes, reaches and
    # droops are independent of the answer.
    base = 1 if rng.getrandbits(1) else -1
    decision = "same" if rng.getrandbits(1) else "opposite"
    sign = [base, base if decision == "same" else -base]

    clouds = []
    for _ in range(3):
        clouds.append(
            [
                rng.randrange(60, 600),
                rng.randrange(40, 120),
                rng.randrange(26, 52),
                rng.randrange(12, 22),
            ]
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "ground_y": GROUND_Y,
        "frame_inset": rng.randrange(10, 21),
        "tick_len": rng.randrange(14, 27),
        "pole_w": POLE_W,
        "pole_cx": pc,
        "pole_top": pole_top,
        "flag_y0": y0,
        "flag_wa": wa,
        "flag_e": [sign[0] * reach[0], sign[1] * reach[1]],
        "flag_droop": droop,
        "flag_ink": [list(inks[0]), list(inks[1])],
        "pole_ink": list(POLE_INK),
        "furniture_ink": list(FURNITURE_INK),
        "cloud_ink": list(CLOUD_INK),
        "clouds": clouds,
    }


# --------------------------------------------------------------------------
# gold, margin, quarantine
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Gold reads exactly two things: the sign of each painted reach."""
    e0, e1 = scene["flag_e"]
    return "same" if (e0 > 0) == (e1 > 0) else "opposite"


def margin(scene: dict) -> float:
    """Smaller of the two sideways reaches, in pixels.

    This is how far the shorter flag's cloth must be pulled back towards its
    pole before it no longer picks a side and the reported answer dies.
    """
    return float(min(abs(scene["flag_e"][0]), abs(scene["flag_e"][1])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def runner_up(scene: dict) -> str:
    return "opposite" if analytic_gold(scene) == "same" else "same"


def variant(scene: dict) -> str:
    """Canonical = the picture agrees with the one-wind prior."""
    return "canonical" if analytic_gold(scene) == "same" else "counterfactual"


def prior_answer(scene: dict) -> str:
    """What the canonical prior would say regardless of the picture."""
    return "same"


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every field is painted and recoverable."""
    return []


def mirror_class(decision: str) -> str:
    """Mirroring the page flips both signs, so the class is preserved."""
    return decision


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _flag_polygon(scene: dict, i: int) -> list:
    pc = scene["pole_cx"][i]
    y0 = scene["flag_y0"][i]
    wa = scene["flag_wa"][i]
    e = scene["flag_e"][i]
    tip_y = y0 + wa // 2 + scene["flag_droop"][i]
    return [(pc, y0), (pc, y0 + wa), (pc + e, tip_y)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), (255, 255, 255))
    d = ImageDraw.Draw(img)

    cloud = tuple(scene["cloud_ink"])
    for cx, cy, rx, ry in scene["clouds"]:
        d.ellipse([cx - rx, cy - ry, cx + rx, cy + ry], fill=cloud)

    furn = tuple(scene["furniture_ink"])
    ins = scene["frame_inset"]
    d.rectangle([ins, ins, scene["width"] - 1 - ins, scene["height"] - 1 - ins],
                outline=furn, width=2)
    tl = scene["tick_len"]
    x0, y0f = ins + 9, ins + 9
    x1, y1f = scene["width"] - 10 - ins, scene["height"] - 10 - ins
    d.line([x0, y0f, x0, y0f + tl], fill=furn, width=2)
    d.line([x1, y1f, x1, y1f - tl], fill=furn, width=2)
    d.line([ins + 6, scene["ground_y"], scene["width"] - 7 - ins, scene["ground_y"]],
           fill=furn, width=3)

    pole = tuple(scene["pole_ink"])
    half = scene["pole_w"] // 2
    for i in (0, 1):
        pc = scene["pole_cx"][i]
        d.rectangle([pc - half, scene["pole_top"][i], pc + half, scene["ground_y"]],
                    fill=pole)
    for i in (0, 1):
        d.polygon(_flag_polygon(scene, i), fill=tuple(scene["flag_ink"][i]))
    return img


def ink_pixels(image: Image.Image) -> int:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    return int(np.count_nonzero(np.any(arr < 250, axis=-1)))


def flag_pixels(image: Image.Image, scene: dict) -> int:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    total = 0
    for ink in scene["flag_ink"]:
        total += int(np.count_nonzero(np.all(arr == np.array(ink, dtype=np.uint8), axis=-1)))
    return total


# --------------------------------------------------------------------------
# declared edits (used by verify.py and the tests)
# --------------------------------------------------------------------------
def _copy(scene: dict) -> dict:
    out = dict(scene)
    for k, v in scene.items():
        if isinstance(v, list):
            out[k] = [list(x) if isinstance(x, list) else x for x in v]
    return out


def translate(scene: dict, dx: int, dy: int) -> dict:
    """Gold-preserving: slide the whole scene."""
    s = _copy(scene)
    s["pole_cx"] = [p + dx for p in s["pole_cx"]]
    s["pole_top"] = [p + dy for p in s["pole_top"]]
    s["flag_y0"] = [p + dy for p in s["flag_y0"]]
    s["ground_y"] = s["ground_y"] + dy
    s["clouds"] = [[c[0] + dx, c[1] + dy, c[2], c[3]] for c in s["clouds"]]
    return s


def recolor(scene: dict) -> dict:
    """Gold-preserving: swap the flag inks and repaint poles and furniture."""
    s = _copy(scene)
    s["flag_ink"] = [list(s["flag_ink"][1]), list(s["flag_ink"][0])]
    s["pole_ink"] = [38, 44, 38]
    s["furniture_ink"] = [182, 176, 168]
    s["cloud_ink"] = [214, 214, 220]
    return s


def restyle(scene: dict, seed: int) -> dict:
    """Gold-preserving: permute the colourless page furniture."""
    rng = random.Random(seed ^ 0x5EED)
    s = _copy(scene)
    s["frame_inset"] = rng.randrange(10, 21)
    s["tick_len"] = rng.randrange(14, 27)
    s["clouds"] = [
        [rng.randrange(60, 600), rng.randrange(40, 120), rng.randrange(26, 52),
         rng.randrange(12, 22)]
        for _ in s["clouds"]
    ]
    return s


def rescale(scene: dict) -> dict:
    """Gold-preserving: grow both flags together (signs and sides untouched)."""
    s = _copy(scene)
    s["flag_wa"] = [int(round(w * 1.10)) for w in s["flag_wa"]]
    s["flag_e"] = [int(round(e * 1.10)) for e in s["flag_e"]]
    return s


def set_class(scene: dict, decision: str) -> dict:
    """Label-transforming: repaint the same two reaches as the target class.

    Only the sign of the second flag moves, so both cloth areas, both reaches,
    the palette, the poles and the page furniture are untouched.
    """
    if decision not in DECISIONS:
        raise ValueError(f"unknown decision: {decision}")
    s = _copy(scene)
    e0, e1 = s["flag_e"]
    want = 1 if ((e0 > 0) == (decision == "same")) else -1
    s["flag_e"] = [e0, want * abs(e1)]
    return s


def collapse_margin(scene: dict) -> dict:
    """Pull both flags back to the boundary: limp cloth, no side picked."""
    s = _copy(scene)
    s["flag_e"] = [(6 if e > 0 else -6) for e in s["flag_e"]]
    return s
