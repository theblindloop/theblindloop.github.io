"""Deterministic latent scene sampler and Pillow renderer."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

QUARANTINE_MARGIN = 120.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    pairs = [
        (0.44, 0.25),
        (0.56, 0.75),
        (0.46, 0.68),
        (0.54, 0.32),
    ]
    near, far = pairs[int(seed) % len(pairs)]
    assignment = {"A": "near", "B": "far"}
    if int(seed) % 2:
        assignment = {"A": "far", "B": "near"}
    return {
        "seed": int(seed),
        "width": 384,
        "height": 384,
        "belt_width": 36,
        "lane_gap": 14,
        "inset": 30 + 2 * rng.randrange(0, 3),
        "checkpoint_fractions": {"near": near, "far": far},
        "label_assignment": assignment,
        "rotation": int(seed) % 4,
        "background": [246, 243, 234],
        "belt_color": [27, 55, 86],
        "start_color": [25, 166, 92],
        "finish_color": [213, 55, 67],
        "a_color": [194, 52, 154],
        "b_color": [239, 132, 38],
    }


def _polyline(scene: dict) -> list[tuple[float, float]]:
    width = int(scene["width"])
    height = int(scene["height"])
    inset = int(scene["inset"])
    belt_width = int(scene["belt_width"])
    lane_gap = int(scene["lane_gap"])
    pitch = belt_width + lane_gap
    top = 42
    ys = [top + i * pitch for i in range(7)]
    if ys[-1] + belt_width / 2 >= height:
        raise ValueError("belt does not fit canvas")
    left, right = inset, width - inset
    points: list[tuple[float, float]] = []
    for i, y in enumerate(ys):
        row = [(left, y), (right, y)] if i % 2 == 0 else [(right, y), (left, y)]
        if not points:
            points.extend(row)
        else:
            points.append(row[0])
            points.append(row[1])
    return points


def _length(points: Iterable[tuple[float, float]]) -> float:
    pts = list(points)
    return sum(math.hypot(x1 - x0, y1 - y0) for (x0, y0), (x1, y1) in zip(pts, pts[1:]))


def _point_at(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    target = float(fraction) * _length(points)
    walked = 0.0
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        segment = math.hypot(x1 - x0, y1 - y0)
        if walked + segment >= target:
            ratio = (target - walked) / segment if segment else 0.0
            return x0 + ratio * (x1 - x0), y0 + ratio * (y1 - y0)
        walked += segment
    return points[-1]


def _marker(draw: ImageDraw.ImageDraw, center: tuple[float, float], radius: int, color: tuple[int, ...], label: str) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    box = draw.textbbox((0, 0), label)
    tw, th = box[2] - box[0], box[3] - box[1]
    draw.text((x - tw / 2, y - th / 2 - 1), label, fill=(255, 255, 255))


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    belt_color = tuple(scene["belt_color"])
    start_color = tuple(scene["start_color"])
    finish_color = tuple(scene["finish_color"])
    a_color = tuple(scene["a_color"])
    b_color = tuple(scene["b_color"])
    points = _polyline(scene)
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.line(points, fill=belt_color, width=int(scene["belt_width"]), joint="curve")
    cap = int(scene["belt_width"]) // 2
    for x, y in points:
        draw.ellipse((x - cap, y - cap, x + cap, y + cap), fill=belt_color)

    fractions = scene["checkpoint_fractions"]
    assignment = scene["label_assignment"]
    for label, color in (("A", a_color), ("B", b_color)):
        fraction = float(fractions[assignment[label]])
        _marker(draw, _point_at(points, fraction), 11, color, label)
    _marker(draw, points[0], 13, start_color, "S")
    _marker(draw, points[-1], 13, finish_color, "F")

    rotation = int(scene["rotation"])
    if rotation == 1:
        image = image.transpose(Image.Transpose.ROTATE_90)
    elif rotation == 2:
        image = image.transpose(Image.Transpose.ROTATE_180)
    elif rotation == 3:
        image = image.transpose(Image.Transpose.ROTATE_270)
    return image


def analytic_gold(scene: dict) -> str:
    fractions = scene["checkpoint_fractions"]
    assignment = scene["label_assignment"]
    scores = {
        label: abs(float(fractions[assignment[label]]) - 0.5)
        for label in ("A", "B")
    }
    return min(scores, key=scores.get)


def margin(scene: dict) -> float:
    points = _polyline(scene)
    fractions = scene["checkpoint_fractions"]
    assignment = scene["label_assignment"]
    scores = sorted(abs(float(fractions[assignment[label]]) - 0.5) for label in ("A", "B"))
    return (scores[1] - scores[0]) * _length(points)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
