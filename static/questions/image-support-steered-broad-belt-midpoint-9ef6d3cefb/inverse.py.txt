"""Independent pixel-only inverse arm using color masks and graph geodesics."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public visual palette, duplicated deliberately rather than imported.
_COLORS = {
    "belt": np.array([27, 55, 86], dtype=np.int16),
    "start": np.array([25, 166, 92], dtype=np.int16),
    "finish": np.array([213, 55, 67], dtype=np.int16),
    "A": np.array([194, 52, 154], dtype=np.int16),
    "B": np.array([239, 132, 38], dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _adjacent(mask: np.ndarray, domain: np.ndarray) -> np.ndarray:
    out = np.zeros_like(mask)
    out[1:] |= mask[:-1]
    out[:-1] |= mask[1:]
    out[:, 1:] |= mask[:, :-1]
    out[:, :-1] |= mask[:, 1:]
    return out & domain


def _distances(domain: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    if int(seeds.sum()) < 6:
        raise ValueError("endpoint does not visibly touch the belt")
    distance = np.full(domain.shape, -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for y, x in np.argwhere(seeds):
        yi, xi = int(y), int(x)
        distance[yi, xi] = 0
        queue.append((yi, xi))
    height, width = domain.shape
    while queue:
        y, x = queue.popleft()
        nd = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and domain[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = nd
                queue.append((ny, nx))
    return distance


def _distance_to_marker(distance: np.ndarray, marker: np.ndarray, domain: np.ndarray) -> float:
    boundary = _adjacent(marker, domain)
    values = distance[boundary]
    values = values[values >= 0]
    if len(values) < 6:
        raise ValueError("checkpoint is disconnected from an endpoint")
    return float(values.min())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = {name: _mask(array, color) for name, color in _COLORS.items()}
    # Clean worlds contain at least 88k navy pixels.  A grid-sized erasure from
    # any part of this high-fill connected region is material damage, even when
    # the remaining fragment happens to retain one candidate's local distance.
    if int(masks["belt"].sum()) < 87000:
        raise ValueError("broad belt has lost a material part of its filled area")
    for name in ("start", "finish", "A", "B"):
        if int(masks[name].sum()) < 120:
            raise ValueError(f"{name} marker is missing or materially corrupted")
    ds = _distances(masks["belt"], _adjacent(masks["start"], masks["belt"]))
    df = _distances(masks["belt"], _adjacent(masks["finish"], masks["belt"]))
    scores: dict[str, float] = {}
    for label in ("A", "B"):
        start_distance = _distance_to_marker(ds, masks[label], masks["belt"])
        finish_distance = _distance_to_marker(df, masks[label], masks["belt"])
        scores[label] = abs(start_distance - finish_distance)
    gap = abs(scores["A"] - scores["B"])
    if gap < 80.0:
        raise ValueError("pixel midpoint comparison is ambiguous")
    return min(scores, key=scores.get)
