"""Deterministic renderer for a compact checkpoint cross-section question.

The answer is a local measurement: compare the horizontal span of one connected
filled ribbon at two visible colored marker heights.  The card is translated and
the marker order, profile context, and remote decoration are randomized so none
of those presentation choices identifies the answer.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "ribbon_checkpoint_width-renderer-0.1.0"
CANVAS = (512, 512)
CARD_W = 126
CARD_H = 126
CARD_BG = (249, 251, 253)
BACKGROUND = (255, 255, 255)
RIBBON = (42, 104, 190)
AMBER = (232, 145, 28)
VIOLET = (124, 70, 174)
DARK = (44, 55, 72)
LIGHT = (218, 224, 232)
REMOTE = (231, 235, 241)


def sample_scene(seed: int) -> dict:
    """Sample one JSON-serializable latent scene from an integer seed."""

    rng = random.Random(int(seed))
    card_x = rng.randint(24, CANVAS[0] - CARD_W - 24)
    card_y = rng.randint(24, CANVAS[1] - CARD_H - 24)

    low_level = rng.randint(32, 43)
    high_level = rng.randint(78, 90)
    if rng.random() < 0.5:
        amber_y, violet_y = low_level, high_level
    else:
        amber_y, violet_y = high_level, low_level

    # The parity of the public scene seed counterbalances the answer identity;
    # all geometric nuisance parameters below are sampled independently.
    sign = 1 if int(seed) % 2 == 0 else -1
    near_boundary = int(seed) % 6 in (0, 1)
    difference = rng.randint(4, 7) if near_boundary else rng.randint(12, 30)
    narrow = rng.randint(44, 57)
    if sign > 0:
        amber_width, violet_width = narrow + difference, narrow
    else:
        amber_width, violet_width = narrow, narrow + difference

    return {
        "amber_width": int(amber_width),
        "violet_width": int(violet_width),
        "amber_y": int(amber_y),
        "violet_y": int(violet_y),
        "card_x": int(card_x),
        "card_y": int(card_y),
        "top_width": int(rng.randint(46, 70)),
        "bottom_width": int(rng.randint(46, 70)),
        "center_shift": int(rng.randint(-6, 6)),
        "remote_variant": int(rng.randrange(4)),
        "background": "white",
        "style": int(rng.randrange(3)),
    }


def analytic_gold(scene: dict) -> str:
    """Return the answer from the two latent widths, with no raster inspection."""

    return "amber" if scene["amber_width"] > scene["violet_width"] else "violet"


def margin(scene: dict) -> float:
    """Continuous signed width difference in displayed pixels."""

    return float(scene["amber_width"] - scene["violet_width"])


def is_quarantined(scene: dict) -> bool:
    """Near-equal spans are retained as explicit but non-headline cases."""

    return abs(margin(scene)) <= 7.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This parametrization has no declared render-identical latent aliases."""

    return []


def _remote_marks(draw: ImageDraw.ImageDraw, scene: dict) -> None:
    """Add low-contrast context that is deliberately not used by the oracle."""

    variant = int(scene["remote_variant"])
    offset = (variant * 17) % 31
    for row in range(4):
        y = 66 + row * 88 + (offset % 9)
        x = 30 + ((row * 71 + offset) % 310)
        draw.rectangle((x, y, min(488, x + 34 + variant * 5), y + 2), fill=REMOTE)
        draw.rectangle((x + 10, y + 10, x + 12, y + 20 + variant), fill=REMOTE)
    if variant % 2:
        draw.rectangle((398, 34, 470, 36), fill=REMOTE)
        draw.rectangle((420, 42, 422, 72), fill=REMOTE)


def _control_points(scene: dict) -> list[tuple[int, int, int]]:
    """Return (relative y, left inset, right inset) controls for the blue fill."""

    levels = [
        (18, int(scene["top_width"])),
        (int(scene["amber_y"]), int(scene["amber_width"])),
        (int(scene["violet_y"]), int(scene["violet_width"])),
        (106, int(scene["bottom_width"])),
    ]
    ordered = sorted(levels, key=lambda item: item[0])
    center = 80 + int(scene["center_shift"])
    controls: list[tuple[int, int, int]] = []
    for y, width in ordered:
        left = center - width // 2
        right = center + (width - 1) // 2
        controls.append((y, left, right))
    return controls


def _ribbon_polygon(scene: dict) -> list[tuple[int, int]]:
    card_x, card_y = int(scene["card_x"]), int(scene["card_y"])
    controls = _control_points(scene)
    left_edge = [(card_x + left, card_y + y) for y, left, _ in controls]
    right_edge = [(card_x + right, card_y + y) for y, _, right in reversed(controls)]
    return left_edge + right_edge


def _draw_marker(
    draw: ImageDraw.ImageDraw, *, x: int, y: int, color: tuple[int, int, int]
) -> None:
    draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=color)


def render(scene: dict) -> Image.Image:
    """Render the final RGB raster seen by the independent inverse arm."""

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _remote_marks(draw, scene)

    card_x, card_y = int(scene["card_x"]), int(scene["card_y"])
    card_box = (card_x, card_y, card_x + CARD_W - 1, card_y + CARD_H - 1)
    draw.rectangle(card_box, fill=CARD_BG, outline=DARK, width=2)

    # Neutral chart furniture frames the inset without encoding the answer.
    draw.rectangle((card_x + 10, card_y + 10, card_x + 30, card_y + 12), fill=DARK)
    draw.rectangle((card_x + 34, card_y + 10, card_x + 53, card_y + 12), fill=LIGHT)
    draw.rectangle((card_x + 12, card_y + 114, card_x + 113, card_y + 116), fill=LIGHT)
    for tick in (25, 50, 75, 100):
        draw.rectangle((card_x + tick, card_y + 111, card_x + tick + 1, card_y + 119), fill=LIGHT)

    draw.polygon(_ribbon_polygon(scene), fill=RIBBON)
    _draw_marker(
        draw,
        x=card_x + 16,
        y=card_y + int(scene["amber_y"]),
        color=AMBER,
    )
    _draw_marker(
        draw,
        x=card_x + 16,
        y=card_y + int(scene["violet_y"]),
        color=VIOLET,
    )
    return image
