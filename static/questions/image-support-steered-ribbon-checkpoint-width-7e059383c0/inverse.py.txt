"""Independent pixel-only inverse for the ribbon checkpoint question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


ORACLE_VERSION = "ribbon_checkpoint_width-oracle-0.1.0"
_RIBBON = np.asarray((42, 104, 190), dtype=np.int16)
_AMBER = np.asarray((232, 145, 28), dtype=np.int16)
_VIOLET = np.asarray((124, 70, 174), dtype=np.int16)


def _palette_mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return the largest 4-connected component as a boolean mask."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        visited[y, x] = True
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    output = np.zeros_like(mask, dtype=bool)
    for y, x in best:
        output[y, x] = True
    return output


def _marker_y(mask: np.ndarray, name: str) -> int:
    component = _largest_component(mask)
    points = np.argwhere(component)
    if len(points) < 70:
        raise ValueError(f"{name} marker is missing or too small")
    return int(round(float(points[:, 0].mean())))


def _span_at(ribbon: np.ndarray, y: int) -> int:
    if y < 0 or y >= ribbon.shape[0]:
        raise ValueError("checkpoint is outside the raster")
    xs = np.flatnonzero(ribbon[y])
    if len(xs) < 8:
        raise ValueError("ribbon does not cross a checkpoint")
    return int(xs[-1] - xs[0] + 1)


def decision_from_image(image: Image.Image) -> str:
    """Recover amber/violet solely from the supplied final PNG pixels."""

    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    ribbon = _largest_component(_palette_mask(array, _RIBBON))
    if int(ribbon.sum()) < 1200:
        raise ValueError("filled ribbon is missing or too small")
    amber_y = _marker_y(_palette_mask(array, _AMBER), "amber")
    violet_y = _marker_y(_palette_mask(array, _VIOLET), "violet")
    amber_span = _span_at(ribbon, amber_y)
    violet_span = _span_at(ribbon, violet_y)
    observed_margin = amber_span - violet_span
    if abs(observed_margin) < 2:
        raise ValueError("checkpoint spans are unresolved at pixel scale")
    return "amber" if observed_margin > 0 else "violet"
