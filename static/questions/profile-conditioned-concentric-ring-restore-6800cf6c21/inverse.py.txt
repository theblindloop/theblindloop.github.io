"""Independent image-only inverse for the concentric ring-order world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("crescent", "lattice", "plume", "arrow")
SIDE = 224
FRAME = 18
MARK = 14
OUTER = SIDE + 2 * FRAME
SCALES = (0.92, 1.0, 1.08)
OFFSETS = (-10, 0, 10)


def _symbol_mask(symbol: str, dx: int, dy: int, scale: float) -> np.ndarray:
    image = Image.new("L", (SIDE, SIDE), 0)
    draw = ImageDraw.Draw(image)
    cx = SIDE / 2 + dx
    cy = SIDE / 2 + dy
    sx = 66.0 * scale
    sy = 72.0 * scale

    def p(x: float, y: float) -> tuple[int, int]:
        return round(x), round(y)

    if symbol == "crescent":
        draw.ellipse([round(cx - 0.72 * sx), round(cy - sy), round(cx + 0.72 * sx), round(cy + sy)], fill=255)
        draw.ellipse([round(cx - 0.16 * sx), round(cy - 0.84 * sy), round(cx + 1.02 * sx), round(cy + 0.78 * sy)], fill=0)
    elif symbol == "lattice":
        lw = max(9, round(13 * scale))
        for frac in (-0.50, 0.0, 0.50):
            draw.line([p(cx - 0.72 * sx, cy + frac * sy), p(cx + 0.72 * sx, cy + frac * sy)], fill=255, width=lw)
            draw.line([p(cx + frac * sx, cy - 0.70 * sy), p(cx + frac * sx, cy + 0.70 * sy)], fill=255, width=lw)
    elif symbol == "plume":
        lw = max(9, round(14 * scale))
        draw.line([p(cx, cy + 0.76 * sy), p(cx - 0.05 * sx, cy - 0.28 * sy)], fill=255, width=lw)
        draw.arc([round(cx - 0.72 * sx), round(cy - 0.90 * sy), round(cx + 0.62 * sx), round(cy + 0.25 * sy)], 248, 112, fill=255, width=lw)
        draw.line([p(cx - 0.05 * sx, cy - 0.28 * sy), p(cx + 0.48 * sx, cy - 0.64 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.06 * sx, cy + 0.34 * sy), p(cx - 0.50 * sx, cy + 0.57 * sy)], fill=255, width=lw)
    elif symbol == "arrow":
        lw = max(10, round(15 * scale))
        draw.line([p(cx - 0.70 * sx, cy + 0.52 * sy), p(cx + 0.42 * sx, cy - 0.58 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.42 * sx, cy - 0.58 * sy), p(cx + 0.26 * sx, cy - 0.05 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.42 * sx, cy - 0.58 * sy), p(cx - 0.10 * sx, cy - 0.42 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.62 * sx, cy + 0.66 * sy), p(cx - 0.32 * sx, cy + 0.36 * sy)], fill=255, width=lw)
    else:
        raise ValueError(symbol)
    return np.asarray(image, dtype=np.float32) / 255.0


def _ring_coords(side: int) -> list[list[tuple[int, int]]]:
    rings: list[list[tuple[int, int]]] = []
    for layer in range((side + 1) // 2):
        top = left = layer
        bottom = right = side - 1 - layer
        coords: list[tuple[int, int]] = []
        for x in range(left, right + 1):
            coords.append((top, x))
        for y in range(top + 1, bottom + 1):
            coords.append((y, right))
        if bottom > top:
            for x in range(right - 1, left - 1, -1):
                coords.append((bottom, x))
        if right > left:
            for y in range(bottom - 1, top, -1):
                coords.append((y, left))
        rings.append(coords)
    return rings


RINGS = _ring_coords(SIDE)


def _connected_components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 60:
            ys = [point[0] for point in points]
            xs = [point[1] for point in points]
            components.append((min(xs), min(ys), max(xs), max(ys), len(points)))
    return components


def _locate_tile(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = rgb.max(axis=2).astype(int) - rgb.min(axis=2).astype(int)
    colored = (spread > 55) & (rgb.max(axis=2) > 100) & (rgb.min(axis=2) < 220)
    components = sorted(_connected_components(colored), key=lambda item: item[4], reverse=True)
    marks = [component for component in components if 120 <= component[4] <= 260][:4]
    if len(marks) != 4:
        raise ValueError("four colored corner fiducials were not found")
    min_x = min(item[0] for item in marks)
    min_y = min(item[1] for item in marks)
    max_x = max(item[2] for item in marks)
    max_y = max(item[3] for item in marks)
    if max_x - min_x + 1 != OUTER or max_y - min_y + 1 != OUTER:
        raise ValueError("corner fiducials do not define the expected square")
    x0 = min_x + FRAME
    y0 = min_y + FRAME
    gray = np.asarray(image.convert("L"), dtype=np.float32)
    tile = gray[y0 : y0 + SIDE, x0 : x0 + SIDE]
    if tile.shape != (SIDE, SIDE):
        raise ValueError("marked square crop is outside the image")
    return tile


def _restore(encoded: np.ndarray) -> np.ndarray:
    stream = encoded.reshape(-1)
    restored = np.empty((SIDE, SIDE), dtype=np.float32)
    cursor = 0
    for ring_index, coords in enumerate(RINGS):
        count = len(coords)
        values = np.asarray(stream[cursor : cursor + count], dtype=np.float32)
        cursor += count
        if ring_index % 2:
            values = values[::-1]
        for (y, x), value in zip(coords, values):
            restored[y, x] = value
    if cursor != SIDE * SIDE:
        raise ValueError("ring stream did not cover the square")
    return restored


def _mean9(image: np.ndarray) -> np.ndarray:
    padded = np.pad(image, 4, mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant")
    integral = integral.cumsum(axis=0).cumsum(axis=1)
    size = 9
    return (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    ) / float(size * size)


def _build_template_bank() -> dict[str, tuple[np.ndarray, np.ndarray]]:
    bank: dict[str, tuple[np.ndarray, np.ndarray]] = {}
    for symbol in SYMBOLS:
        templates = []
        for scale in SCALES:
            for dx in OFFSETS:
                for dy in OFFSETS:
                    templates.append(_symbol_mask(symbol, dx, dy, scale).reshape(-1))
        values = np.stack(templates).astype(np.float32)
        centered = values - values.mean(axis=1, keepdims=True)
        norms = np.linalg.norm(centered, axis=1) + 1e-6
        bank[symbol] = centered, norms
    return bank


_TEMPLATE_BANK = _build_template_bank()


def _scores(restored: np.ndarray) -> dict[str, float]:
    smoothed = _mean9(restored)
    signal = np.median(smoothed) - smoothed
    vector = signal.reshape(-1).astype(np.float32)
    vector -= vector.mean()
    vector_norm = float(np.linalg.norm(vector)) + 1e-6
    scores: dict[str, float] = {}
    for symbol in SYMBOLS:
        templates, norms = _TEMPLATE_BANK[symbol]
        scores[symbol] = float(np.max((templates @ vector) / (norms * vector_norm)))
    return scores


def decision_from_image(image: Image.Image) -> str:
    """Recover the symbol using only pixels of the final PNG."""

    encoded = _locate_tile(image)
    restored = _restore(encoded)
    scores = _scores(restored)
    return max(SYMBOLS, key=lambda symbol: scores[symbol])
