"""Renderer and analytic gold for a marked concentric-ring raster world.

The broad symbol is written into a square raster, then serialized one square
perimeter at a time from outside to inside.  Odd-numbered perimeter runs are
reversed before the stream is written row-major into the visible tile.  The
pixel-only inverse in ``oracle.py`` reconstructs the raster from the rendered
PNG and is intentionally implemented without importing this module.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "concentric_ring_restore-0.1.0"
SYMBOLS = ("crescent", "lattice", "plume", "arrow")
PALETTES = (
    (16, 151, 183),
    (218, 61, 116),
    (70, 96, 211),
    (225, 132, 22),
)

WIDTH, HEIGHT = 512, 384
SIDE = 224
FRAME = 18
MARK = 14
OUTER = SIDE + 2 * FRAME


def _symbol_mask(
    symbol: str,
    *,
    dx: int = 0,
    dy: int = 0,
    scale: float = 1.0,
) -> np.ndarray:
    """Return one broad binary glyph at the declared nuisance pose."""

    image = Image.new("L", (SIDE, SIDE), 0)
    draw = ImageDraw.Draw(image)
    cx = SIDE / 2 + dx
    cy = SIDE / 2 + dy
    sx = 66.0 * scale
    sy = 72.0 * scale

    def p(x: float, y: float) -> tuple[int, int]:
        return round(x), round(y)

    if symbol == "crescent":
        draw.ellipse(
            [round(cx - 0.72 * sx), round(cy - sy), round(cx + 0.72 * sx), round(cy + sy)],
            fill=255,
        )
        draw.ellipse(
            [round(cx - 0.16 * sx), round(cy - 0.84 * sy), round(cx + 1.02 * sx), round(cy + 0.78 * sy)],
            fill=0,
        )
    elif symbol == "lattice":
        lw = max(9, round(13 * scale))
        for frac in (-0.50, 0.0, 0.50):
            draw.line(
                [p(cx - 0.72 * sx, cy + frac * sy), p(cx + 0.72 * sx, cy + frac * sy)],
                fill=255,
                width=lw,
            )
            draw.line(
                [p(cx + frac * sx, cy - 0.70 * sy), p(cx + frac * sx, cy + 0.70 * sy)],
                fill=255,
                width=lw,
            )
    elif symbol == "plume":
        lw = max(9, round(14 * scale))
        draw.line([p(cx, cy + 0.76 * sy), p(cx - 0.05 * sx, cy - 0.28 * sy)], fill=255, width=lw)
        draw.arc(
            [round(cx - 0.72 * sx), round(cy - 0.90 * sy), round(cx + 0.62 * sx), round(cy + 0.25 * sy)],
            248,
            112,
            fill=255,
            width=lw,
        )
        draw.line([p(cx - 0.05 * sx, cy - 0.28 * sy), p(cx + 0.48 * sx, cy - 0.64 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.06 * sx, cy + 0.34 * sy), p(cx - 0.50 * sx, cy + 0.57 * sy)], fill=255, width=lw)
    elif symbol == "arrow":
        lw = max(10, round(15 * scale))
        draw.line([p(cx - 0.70 * sx, cy + 0.52 * sy), p(cx + 0.42 * sx, cy - 0.58 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.42 * sx, cy - 0.58 * sy), p(cx + 0.26 * sx, cy - 0.05 * sy)], fill=255, width=lw)
        draw.line([p(cx + 0.42 * sx, cy - 0.58 * sy), p(cx - 0.10 * sx, cy - 0.42 * sy)], fill=255, width=lw)
        draw.line([p(cx - 0.62 * sx, cy + 0.66 * sy), p(cx - 0.32 * sx, cy + 0.36 * sy)], fill=255, width=lw)
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(image, dtype=np.float32) / 255.0


def _ring_coords(side: int) -> list[list[tuple[int, int]]]:
    rings: list[list[tuple[int, int]]] = []
    for layer in range((side + 1) // 2):
        top = left = layer
        bottom = right = side - 1 - layer
        coords: list[tuple[int, int]] = []
        for x in range(left, right + 1):
            coords.append((top, x))
        for y in range(top + 1, bottom + 1):
            coords.append((y, right))
        if bottom > top:
            for x in range(right - 1, left - 1, -1):
                coords.append((bottom, x))
        if right > left:
            for y in range(bottom - 1, top, -1):
                coords.append((y, left))
        rings.append(coords)
    return rings


RINGS = _ring_coords(SIDE)


def _encode(restored: np.ndarray) -> np.ndarray:
    stream: list[float] = []
    for ring_index, coords in enumerate(RINGS):
        values = np.asarray([restored[y, x] for y, x in coords], dtype=np.float32)
        if ring_index % 2:
            values = values[::-1]
        stream.extend(float(value) for value in values)
    return np.asarray(stream, dtype=np.float32).reshape((SIDE, SIDE))


def _carrier(phase: int, amplitude: float) -> np.ndarray:
    yy, xx = np.indices((SIDE, SIDE))
    phase = int(phase) % 16
    blocks = ((xx + phase) // 4 + yy // 4) % 2
    return np.where(blocks == 0, amplitude, -amplitude).astype(np.float32)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    weak = int(seed) % 13 == 0
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "tile_x": rng.randint(44, WIDTH - OUTER - 44),
        "tile_y": rng.randint(34, HEIGHT - OUTER - 30),
        "carrier_phase": rng.randrange(16),
        "carrier_amplitude": round(rng.uniform(48.0, 56.0) if weak else rng.uniform(32.0, 48.0), 3),
        "signal_strength": round(rng.uniform(42.0, 50.0) if weak else rng.uniform(64.0, 82.0), 3),
        "base_level": rng.randint(164, 198),
        "symbol_dx": rng.choice((-10, 0, 10)),
        "symbol_dy": rng.choice((-10, 0, 10)),
        "symbol_scale": rng.choice((0.92, 1.0, 1.08)),
        "marker_palette_id": rng.randrange(len(PALETTES)),
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "seed": int(seed),
    }


def render(scene: dict) -> Image.Image:
    mask = _symbol_mask(
        str(scene["symbol"]),
        dx=int(scene["symbol_dx"]),
        dy=int(scene["symbol_dy"]),
        scale=float(scene["symbol_scale"]),
    )
    base = float(scene["base_level"])
    strength = float(scene["signal_strength"])
    carrier = _carrier(int(scene["carrier_phase"]), float(scene["carrier_amplitude"]))
    restored = np.clip(base - strength * mask + carrier, 0.0, 255.0)
    encoded = _encode(restored)

    tile = np.full((OUTER, OUTER), 242.0, dtype=np.float32)
    tile[FRAME : FRAME + SIDE, FRAME : FRAME + SIDE] = encoded
    tile_rgb = Image.fromarray(np.rint(tile).astype(np.uint8), mode="L").convert("RGB")
    draw = ImageDraw.Draw(tile_rgb)
    draw.rectangle([0, 0, OUTER - 1, OUTER - 1], outline=(40, 40, 40), width=3)
    color = PALETTES[int(scene["marker_palette_id"]) % len(PALETTES)]
    for mx, my in ((0, 0), (OUTER - MARK, 0), (0, OUTER - MARK), (OUTER - MARK, OUTER - MARK)):
        draw.rectangle([mx, my, mx + MARK - 1, my + MARK - 1], fill=color)

    canvas = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (255, 255, 255))
    canvas.paste(tile_rgb, (int(scene["tile_x"]), int(scene["tile_y"])))
    return canvas


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """Analytic proxy for the recovered top-versus-runner-up template gap."""

    return round(float(scene["signal_strength"]) - 0.85 * float(scene["carrier_amplitude"]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 16
    return [("carrier_phase_period_alias", twin)]
