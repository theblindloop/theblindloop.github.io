"""Analytic sampler and raster renderer for an alternating-route search world."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "route-alternation-search-0.1.0"
QUADRANTS = ("NW", "NE", "SW", "SE")
CANVAS = (960, 720)
_CENTER = {
    "NW": (240.0, 180.0),
    "NE": (720.0, 180.0),
    "SW": (240.0, 540.0),
    "SE": (720.0, 540.0),
}
_PATTERN_SIGNS: dict[str, tuple[int, int, int, int]] = {
    "alternating": (1, -1, 1, -1),
    "repeat_then_switch": (1, 1, -1, 1),
    "middle_repeat": (1, -1, -1, 1),
    "mirrored_repeat": (-1, 1, 1, -1),
}
_INK_PALETTE = (
    (28, 79, 145),
    (137, 52, 59),
    (29, 112, 87),
    (105, 57, 145),
)


def _route_points(
    pattern: str,
    turn_angle: float,
    step_length: float,
    transform: dict[str, Any],
    jitter: tuple[float, float],
    quadrant: str,
) -> list[tuple[float, float]]:
    """Construct one visible route from turn signs, then apply only nuisance transforms."""

    signs = _PATTERN_SIGNS[pattern]
    heading = 0.0
    points = [(0.0, 0.0)]
    for segment in range(5):
        last_x, last_y = points[-1]
        points.append(
            (
                last_x + step_length * math.cos(heading),
                last_y + step_length * math.sin(heading),
            )
        )
        if segment < 4:
            heading += math.radians(turn_angle) * signs[segment]

    xs = [point[0] for point in points]
    ys = [point[1] for point in points]
    center_x = (min(xs) + max(xs)) / 2.0
    center_y = (min(ys) + max(ys)) / 2.0
    local = [(x - center_x, y - center_y) for x, y in points]
    if bool(transform["mirror"]):
        local = [(-x, y) for x, y in local]
    scale = float(transform["scale"])
    angle = float(transform["angle"])
    cosine, sine = math.cos(angle), math.sin(angle)
    cx, cy = _CENTER[quadrant]
    jx, jy = jitter
    return [
        (
            cx + jx + scale * (x * cosine - y * sine),
            cy + jy + scale * (x * sine + y * cosine),
        )
        for x, y in local
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, deterministic four-sector route scene."""

    rng = random.Random(int(seed))
    target = rng.choice(QUADRANTS)
    distractor_patterns = ["repeat_then_switch", "middle_repeat", "mirrored_repeat"]
    rng.shuffle(distractor_patterns)
    specs: list[dict[str, str]] = []
    distractor_index = 0
    for quadrant in QUADRANTS:
        if quadrant == target:
            pattern = "alternating"
        else:
            pattern = distractor_patterns[distractor_index]
            distractor_index += 1
        specs.append({"quadrant": quadrant, "pattern": pattern})
    rng.shuffle(specs)

    transforms: dict[str, dict[str, Any]] = {}
    jitters: dict[str, list[float]] = {}
    for quadrant in QUADRANTS:
        transforms[quadrant] = {
            "angle": round(rng.uniform(-0.08, 0.08), 5),
            "scale": round(rng.uniform(0.95, 1.05), 5),
            "mirror": bool(rng.randrange(2)),
        }
        jitters[quadrant] = [
            round(rng.uniform(-14.0, 14.0), 3),
            round(rng.uniform(-12.0, 12.0), 3),
        ]

    return {
        "target_quadrant": target,
        "route_specs": specs,
        "route_transforms": transforms,
        "route_jitter": jitters,
        "turn_angle": round(rng.uniform(24.0, 68.0), 3),
        "step_length": round(rng.uniform(80.0, 88.0), 3),
        "stroke_width": round(rng.uniform(9.0, 12.0), 3),
        "ink_color": list(rng.choice(_INK_PALETTE)),
        "background": [248, 249, 246],
        "guide_color": [213, 219, 222],
        "width": CANVAS[0],
        "height": CANVAS[1],
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render all routes from scene fields into an RGB PNG."""

    supersample = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * supersample, height * supersample), background)
    draw = ImageDraw.Draw(image)

    guide = tuple(int(value) for value in scene["guide_color"])
    guide_width = max(1, round(2.0 * supersample))
    draw.line(
        [(width * supersample / 2.0, 0), (width * supersample / 2.0, height * supersample)],
        fill=guide,
        width=guide_width,
    )
    draw.line(
        [(0, height * supersample / 2.0), (width * supersample, height * supersample / 2.0)],
        fill=guide,
        width=guide_width,
    )

    ink = tuple(int(value) for value in scene["ink_color"])
    width_px = max(1, round(float(scene["stroke_width"]) * supersample))
    radius = width_px / 2.0
    for spec in scene["route_specs"]:
        quadrant = str(spec["quadrant"])
        pattern = "alternating" if quadrant == scene["target_quadrant"] else str(spec["pattern"])
        transform = scene["route_transforms"][quadrant]
        jitter = tuple(float(value) for value in scene["route_jitter"][quadrant])
        points = _route_points(
            pattern,
            float(scene["turn_angle"]),
            float(scene["step_length"]),
            transform,
            jitter,
            quadrant,
        )
        scaled = [(round(x * supersample), round(y * supersample)) for x, y in points]
        draw.line(scaled, fill=ink, width=width_px, joint="curve")
        # Rounded caps and joins keep the route one connected thin component.
        for x, y in scaled:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ink)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """The answer is the visible sector containing the alternating route."""

    return str(scene["target_quadrant"])


def _alternation_count(signs: tuple[int, int, int, int]) -> int:
    return sum(int(first != second) for first, second in zip(signs, signs[1:]))


def margin(scene: dict[str, Any]) -> float:
    """Continuous top-vs-runner-up gap, modulated by bend separation from straight."""

    scores = []
    for spec in scene["route_specs"]:
        quadrant = str(spec["quadrant"])
        pattern = "alternating" if quadrant == scene["target_quadrant"] else str(spec["pattern"])
        scores.append(_alternation_count(_PATTERN_SIGNS[pattern]))
    ordered = sorted(scores, reverse=True)
    bend_strength = abs(math.sin(math.radians(float(scene["turn_angle"]))))
    return float((ordered[0] - ordered[1]) * bend_strength)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude shallow bends whose raster turn sign is close to its finite-resolution boundary."""

    return margin(scene) < 0.42


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reverse the unordered, nonoverlapping route-record list."""

    twin = copy.deepcopy(scene)
    twin["route_specs"] = list(reversed(twin["route_specs"]))
    return [("reverse_unordered_route_records", twin)]
