"""Independent inverse arm using only the final raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_RGB = {
    "red": (210, 48, 61),
    "blue": (38, 103, 205),
    "green": (29, 145, 88),
    "purple": (139, 64, 170),
}
BENCHMARK_RGB = (231, 133, 30)


def _near(array: np.ndarray, rgb: tuple[int, int, int], radius: float = 18.0) -> np.ndarray:
    delta = array.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    by, bx = np.nonzero(_near(array, BENCHMARK_RGB))
    if len(bx) < 120 or float(np.std(bx)) > 3.0:
        raise ValueError("vertical dashed benchmark is absent or malformed")
    benchmark = float(np.median(bx))

    intervals: dict[str, tuple[float, float]] = {}
    for name, rgb in COLOR_RGB.items():
        ys, xs = np.nonzero(_near(array, rgb))
        if len(xs) < 300:
            raise ValueError(f"{name} interval is absent")
        lo, hi = float(xs.min()), float(xs.max())
        if hi - lo < 45 or float(ys.max() - ys.min()) < 20:
            raise ValueError(f"{name} interval geometry is malformed")
        intervals[name] = (lo, hi)

    qualifying = {name: bounds for name, bounds in intervals.items() if bounds[0] < benchmark < bounds[1]}
    if len(qualifying) < 2:
        raise ValueError("too few benchmark-crossing intervals")
    ordered = sorted(qualifying, key=lambda name: qualifying[name][1] - qualifying[name][0])
    widths = [qualifying[name][1] - qualifying[name][0] for name in ordered]
    if widths[1] - widths[0] < 12:
        raise ValueError("narrowest qualifying interval is pixel-ambiguous")
    return ordered[0]
