"""Deterministic interval-plot sampler and Pillow renderer."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
COLOR_RGB = {
    "red": (210, 48, 61),
    "blue": (38, 103, 205),
    "green": (29, 145, 88),
    "purple": (139, 64, 170),
}
COLORS = tuple(COLOR_RGB)
BENCHMARK_RGB = (231, 133, 30)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    benchmark = rng.randint(282, 358)
    answer = COLORS[int(seed) % len(COLORS)]
    rows = [116, 180, 244, 308]
    rng.shuffle(rows)

    answer_width = rng.randint(92, 118)
    qualifying_widths = [answer_width, answer_width + rng.randint(42, 61), answer_width + rng.randint(83, 112)]
    other_colors = [color for color in COLORS if color != answer]
    rng.shuffle(other_colors)
    width_by_color = {answer: qualifying_widths[0], other_colors[0]: qualifying_widths[1], other_colors[1]: qualifying_widths[2]}

    intervals = []
    for index, color in enumerate(COLORS):
        if color in width_by_color:
            width = width_by_color[color]
            left_clear = rng.randint(28, width - 28)
            lo = benchmark - left_clear
            hi = lo + width
        else:
            # The excluded interval is deliberately shorter than the answer,
            # so the benchmark-filtering step cannot be skipped.
            width = rng.randint(55, 78)
            if rng.random() < 0.5:
                hi = benchmark - rng.randint(24, 54)
                lo = hi - width
            else:
                lo = benchmark + rng.randint(24, 54)
                hi = lo + width
        intervals.append({"color": color, "row": rows[index], "lo": lo, "hi": hi})
    return {"intervals": intervals, "benchmark_x": benchmark}


def render(scene: dict) -> Image.Image:
    intervals = scene["intervals"]
    benchmark = int(scene["benchmark_x"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 248, 246))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((55, 54, 608, 360), radius=8, fill=(255, 255, 255), outline=(93, 98, 104), width=2)
    for y in (116, 180, 244, 308):
        draw.line((78, y, 585, y), fill=(226, 229, 231), width=2)
    for x in range(94, 586, 82):
        draw.line((x, 345, x, 351), fill=(86, 91, 97), width=2)
    draw.line((78, 345, 585, 345), fill=(86, 91, 97), width=3)

    # Visible dashed benchmark; intervals are painted over it at crossings.
    for y in range(72, 333, 20):
        draw.line((benchmark, y, benchmark, min(y + 12, 333)), fill=BENCHMARK_RGB, width=5)

    for item in intervals:
        color = COLOR_RGB[item["color"]]
        y, lo, hi = int(item["row"]), int(item["lo"]), int(item["hi"])
        draw.line((lo, y, hi, y), fill=color, width=7)
        draw.line((lo, y - 13, lo, y + 13), fill=color, width=7)
        draw.line((hi, y - 13, hi, y + 13), fill=color, width=7)
        draw.ellipse((lo + (hi - lo) // 2 - 6, y - 6, lo + (hi - lo) // 2 + 6, y + 6), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    intervals = scene["intervals"]
    benchmark = float(scene["benchmark_x"])
    qualifying = [item for item in intervals if float(item["lo"]) < benchmark < float(item["hi"])]
    if len(qualifying) < 2:
        raise ValueError("fewer than two benchmark-crossing intervals")
    return min(qualifying, key=lambda item: float(item["hi"]) - float(item["lo"]))["color"]


def margin(scene: dict) -> float:
    intervals = scene["intervals"]
    benchmark = float(scene["benchmark_x"])
    qualifying = [item for item in intervals if float(item["lo"]) < benchmark < float(item["hi"])]
    widths = sorted(float(item["hi"]) - float(item["lo"]) for item in qualifying)
    endpoint_clearance = min(min(abs(benchmark - float(item["lo"])), abs(float(item["hi"]) - benchmark)) for item in intervals)
    return min(widths[1] - widths[0], endpoint_clearance)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
