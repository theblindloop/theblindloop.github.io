"""Independent raster-only inverse for the public rhythm-gap question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MARKER = (20, 44, 82)
MAGENTA = (211, 45, 143)
FRAME_PALETTE = {
    "red": (214, 50, 56),
    "green": (24, 151, 91),
    "blue": (43, 101, 214),
}
MASK_RADIUS = 28.0


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= MASK_RADIUS * MASK_RADIUS


def _components(mask: np.ndarray, minimum: int = 80) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append(
                {
                    "area": len(xs),
                    "cx": float(sum(xs)) / len(xs),
                    "cy": float(sum(ys)) / len(ys),
                    "bbox": (min(xs), min(ys), max(xs), max(ys)),
                }
            )
    return found


def _frame_bbox(array: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(_mask(array, rgb))
    if len(xs) < 900:
        raise ValueError("a required colored frame is missing")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _normalized_centroid_gaps(components: list[dict], bounds: tuple[int, int, int, int]) -> tuple[float, ...]:
    x0, y0, x1, y1 = bounds
    dots = [
        item for item in components
        if x0 < item["cx"] < x1 and y0 < item["cy"] < y1 and 220 <= item["area"] <= 450
    ]
    if len(dots) != 5:
        raise ValueError(f"a rhythm frame contains {len(dots)} recoverable dots, expected five")
    xs = sorted(item["cx"] for item in dots)
    gaps = [right - left for left, right in zip(xs, xs[1:])]
    total = sum(gaps)
    if total <= 0 or min(gaps) < 20:
        raise ValueError("dot gaps are unresolved")
    return tuple(value / total for value in gaps)


def _distance(left: tuple[float, ...], right: tuple[float, ...]) -> float:
    return sum(abs(a - b) for a, b in zip(left, right, strict=True))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.shape[1] < 700 or array.shape[0] < 500:
        raise ValueError("image is too small for the public layout")
    dots = _components(_mask(array, MARKER))
    if len(dots) != 20:
        raise ValueError(f"raster contains {len(dots)} dot components, expected twenty")
    reference = _normalized_centroid_gaps(dots, _frame_bbox(array, MAGENTA))
    scores = []
    for color, rgb in FRAME_PALETTE.items():
        signature = _normalized_centroid_gaps(dots, _frame_bbox(array, rgb))
        scores.append((_distance(reference, signature), color))
    scores.sort()
    if scores[0][0] > 0.035 or scores[1][0] - scores[0][0] < 0.08:
        raise ValueError("raster does not expose a unique normalized-gap match")
    return scores[0][1]
