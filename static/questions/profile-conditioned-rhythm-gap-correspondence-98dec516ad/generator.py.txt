"""Deterministic renderer and analytic arm for ordered rhythm-gap matching."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 640
BACKGROUND = (245, 247, 251)
CARD_FILL = (255, 255, 255)
INK = (56, 65, 79)
MARKER = (20, 44, 82)
GUIDE = (188, 196, 207)
MAGENTA = (211, 45, 143)
FRAME_PALETTE = {
    "red": (214, 50, 56),
    "green": (24, 151, 91),
    "blue": (43, 101, 214),
}
FRAME_NAMES = tuple(FRAME_PALETTE)
REFERENCE_BOUNDS = (70, 65, 930, 205)
CANDIDATE_BOUNDS = (
    (70, 245, 930, 345),
    (70, 375, 930, 475),
    (70, 505, 930, 605),
)
_PERMUTATIONS = tuple(itertools.permutations((1, 2, 3, 4)))


def _normalized(gaps: list[int] | tuple[int, ...]) -> tuple[float, ...]:
    values = tuple(float(value) for value in gaps)
    total = sum(values)
    if len(values) != 4 or total <= 0 or any(value <= 0 for value in values):
        raise ValueError("a rhythm must have four positive consecutive gaps")
    return tuple(value / total for value in values)


def _residual(left: list[int] | tuple[int, ...], right: list[int] | tuple[int, ...]) -> float:
    return sum(abs(a - b) for a, b in zip(_normalized(left), _normalized(right), strict=True))


def _dot_positions(start: int, unit: int, gaps: list[int] | tuple[int, ...]) -> list[int]:
    positions = [int(start)]
    for gap in gaps:
        positions.append(positions[-1] + int(unit) * int(gap))
    return positions


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seed triples cycle all answer colors."""
    value = int(seed)
    rng = random.Random(value ^ 0x5A17C3)
    reference_gaps = list(_PERMUTATIONS[(value // 3) % len(_PERMUTATIONS)])
    answer_color = FRAME_NAMES[value % len(FRAME_NAMES)]

    ranked = sorted(
        (
            (_residual(reference_gaps, permutation), permutation)
            for permutation in _PERMUTATIONS
            if list(permutation) != reference_gaps
        ),
        key=lambda item: (item[0], item[1]),
    )
    # Alternate genuinely close and comfortably separated runner-up cases.
    if (value // 3) % 2 == 0:
        distractors = [ranked[0][1], ranked[2][1]]
    else:
        distractors = [ranked[len(ranked) // 2][1], ranked[-2][1]]
    if distractors[0] == distractors[1]:
        raise RuntimeError("distractor construction collapsed")

    slot_shift = (value // 3) % 3
    unit_shift = (value // 9) % 3
    offset_shift = (value // 27) % 3
    units = (40, 48, 56)
    offsets = (-24, 0, 24)
    remaining = iter(distractors)
    cards = []
    for color_index, color in enumerate(FRAME_NAMES):
        gaps = tuple(reference_gaps) if color == answer_color else next(remaining)
        unit = units[(color_index + unit_shift) % 3]
        center = 500 + offsets[(color_index + offset_shift) % 3]
        cards.append(
            {
                "color": color,
                "slot": (color_index + slot_shift) % 3,
                "gaps": list(gaps),
                "unit": unit,
                "start": center - 5 * unit,
            }
        )
    rng.shuffle(cards)  # latent serialization order, intentionally invisible

    reference_unit = (42, 48, 54)[(value // 5) % 3]
    reference_center = 500 + (-18, 0, 18)[(value // 15) % 3]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "reference_gaps": reference_gaps,
        "reference_unit": reference_unit,
        "reference_start": reference_center - 5 * reference_unit,
        "candidate_cards": cards,
    }


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_gaps"]
    scores = sorted(
        (_residual(reference, card["gaps"]), card["color"])
        for card in scene["candidate_cards"]
    )
    if len(scores) != 3 or scores[0][0] > 1e-9 or scores[1][0] - scores[0][0] <= 1e-9:
        raise ValueError("scene does not contain one exact rhythm match")
    return scores[0][1]


def margin(scene: dict) -> float:
    reference = scene["reference_gaps"]
    scores = sorted(_residual(reference, card["gaps"]) for card in scene["candidate_cards"])
    return float(100.0 * (scores[1] - scores[0])) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        cards = scene["candidate_cards"]
        valid = (
            len(cards) == 3
            and len({card["color"] for card in cards}) == 3
            and len({int(card["slot"]) for card in cards}) == 3
            and all(len(_dot_positions(card["start"], card["unit"], card["gaps"])) == 5 for card in cards)
            and len(_dot_positions(scene["reference_start"], scene["reference_unit"], scene["reference_gaps"])) == 5
        )
        return not valid or margin(scene) < 10.0
    except (KeyError, TypeError, ValueError):
        return True


def _draw_rhythm(
    draw: ImageDraw.ImageDraw,
    *,
    y: int,
    start: int,
    unit: int,
    gaps: list[int] | tuple[int, ...],
) -> None:
    positions = _dot_positions(start, unit, gaps)
    draw.line((positions[0], y, positions[-1], y), fill=GUIDE, width=4)
    for x in positions:
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=MARKER)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((70, 18), "MATCH THE LEFT-TO-RIGHT GAP PROPORTIONS", fill=INK, font=font)
    draw.rounded_rectangle(REFERENCE_BOUNDS, radius=16, fill=CARD_FILL, outline=MAGENTA, width=7)
    draw.text((92, 82), "REFERENCE", fill=INK, font=font)
    _draw_rhythm(
        draw,
        y=155,
        start=int(scene["reference_start"]),
        unit=int(scene["reference_unit"]),
        gaps=scene["reference_gaps"],
    )
    for card in sorted(scene["candidate_cards"], key=lambda item: int(item["slot"])):
        bounds = CANDIDATE_BOUNDS[int(card["slot"])]
        draw.rounded_rectangle(
            bounds,
            radius=15,
            fill=CARD_FILL,
            outline=FRAME_PALETTE[card["color"]],
            width=7,
        )
        draw.text((92, bounds[1] + 12), "CANDIDATE", fill=INK, font=font)
        _draw_rhythm(
            draw,
            y=(bounds[1] + bounds[3]) // 2 + 15,
            start=int(card["start"]),
            unit=int(card["unit"]),
            gaps=card["gaps"],
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["candidate_cards"] = list(reversed(twin["candidate_cards"]))
    return [("reverse_candidate_record_order", twin)]
