"""Supersampled renderer and analytic labels for visible star-point counting."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "star_point_count-renderer-0.1.0"
WIDTH = 640
HEIGHT = 512
SUPERSAMPLE = 4
TAU = 2.0 * math.pi
QUARANTINE_CONTRAST = 48.0

BACKGROUND_PALETTE = (
    (246, 242, 232),
    (237, 243, 246),
    (245, 238, 246),
    (235, 244, 238),
)
STAR_PALETTE = (
    (31, 47, 86),
    (78, 34, 55),
    (25, 76, 67),
    (64, 42, 91),
    (49, 51, 57),
)


def _color(value: object) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[arg-type]


def _vertices(scene: dict) -> list[tuple[float, float]]:
    """Construct alternating outer and inner vertices from visible scene fields."""

    count = int(scene["point_count"])
    angles = sorted(float(value) % TAU for value in scene["point_angles"])
    if len(angles) != count:
        raise ValueError("point_angles must contain one angle per outward point")
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    outer = float(scene["outer_radius"])
    inner = float(scene["inner_radius"])
    phase = math.pi / count
    result: list[tuple[float, float]] = []
    for angle in angles:
        result.append((cx + outer * math.cos(angle), cy + outer * math.sin(angle)))
        valley = angle + phase
        result.append((cx + inner * math.cos(valley), cy + inner * math.sin(valley)))
    return result


def sample_scene(seed: int) -> dict:
    """Sample balanced four-to-seven-point stars with explicit geometry."""

    seed_value = int(seed)
    rng = random.Random(seed_value)
    point_count = 4 + (seed_value % 4)
    near_boundary = seed_value % 11 == 0

    # Keep the filled area approximately stable across point counts and valley
    # depths.  This makes total ink a poor proxy for the answer.
    target_area = rng.uniform(34_000.0, 38_000.0)
    if near_boundary:
        inner_ratio = rng.uniform(0.70, 0.74)
    else:
        inner_ratio = rng.uniform(0.46, 0.54)
    outer_radius = math.sqrt(
        target_area
        / (point_count * inner_ratio * math.sin(math.pi / point_count))
    )
    # Independent nuisance variation is deliberately applied before the scene
    # is serialized, while retaining a comfortable canvas margin.
    outer_radius *= rng.uniform(0.985, 1.015)
    inner_radius = outer_radius * inner_ratio
    center_x = rng.uniform(210.0, 430.0)
    center_y = rng.uniform(205.0, 250.0)
    rotation = rng.uniform(0.0, TAU)
    point_angles = [rotation + TAU * index / point_count for index in range(point_count)]
    rng.shuffle(point_angles)

    background = BACKGROUND_PALETTE[rng.randrange(len(BACKGROUND_PALETTE))]
    star_color = STAR_PALETTE[rng.randrange(len(STAR_PALETTE))]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(background),
        "star_color": list(star_color),
        "center_x": round(center_x, 4),
        "center_y": round(center_y, 4),
        "outer_radius": round(outer_radius, 4),
        "inner_radius": round(inner_radius, 4),
        "point_count": point_count,
        "point_angles": [round(angle, 8) for angle in point_angles],
    }


def render(scene: dict) -> Image.Image:
    """Render one filled star as a deterministic RGB PNG-sized image."""

    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), _color(scene["background"]))
    draw = ImageDraw.Draw(image)
    vertices = _vertices(scene)
    scaled = [(round(x * scale), round(y * scale)) for x, y in vertices]
    draw.polygon(scaled, fill=_color(scene["star_color"]))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The visible truth is the number of outward tips in the silhouette."""

    return str(int(scene["point_count"]))


def margin(scene: dict) -> float:
    """Return the visible radial separation between each tip and its valley."""

    return round(float(scene["outer_radius"]) - float(scene["inner_radius"]), 6)


def is_quarantined(scene: dict) -> bool:
    """Exclude stars whose tip/valley contrast is too small for stable pixels."""

    return margin(scene) < QUARANTINE_CONTRAST


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the ordered angle list; rendering sorts this same set of angles."""

    twin = dict(scene)
    twin["point_angles"] = list(reversed(scene["point_angles"]))
    return [("reverse_outer_tip_angle_order", twin)]
