"""Independent image-only inverse for counting star tips.

The inverse segments the largest foreground component, estimates its centroid,
constructs a circular radial boundary profile, and counts separated high-radius
lobes.  It does not import renderer state, scene records, prompts, or analytic
labels.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "star_point_count-oracle-0.1.0"
FOREGROUND_THRESHOLD = 48.0
PROFILE_BINS = 720
SMOOTH_RADIUS = 5


def _background_estimate(array: np.ndarray) -> np.ndarray:
    height, width = array.shape[:2]
    size = min(10, height // 4, width // 4)
    patches = np.concatenate(
        (
            array[:size, :size].reshape(-1, 3),
            array[:size, width - size : width].reshape(-1, 3),
            array[height - size : height, :size].reshape(-1, 3),
            array[height - size : height, width - size : width].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(patches.astype(np.float32), axis=0)


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return the largest 8-connected foreground component."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    foreground = np.argwhere(mask)
    best: list[tuple[int, int]] = []
    for start_y, start_x in foreground.tolist():
        y0, x0 = int(start_y), int(start_x)
        if visited[y0, x0]:
            continue
        stack = [(y0, x0)]
        visited[y0, x0] = True
        current: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            current.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
        if len(current) > len(best):
            best = current
    result = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        result[np.asarray(ys), np.asarray(xs)] = True
    return result


def _radial_profile(component: np.ndarray) -> np.ndarray:
    ys, xs = np.nonzero(component)
    if len(xs) == 0:
        return np.zeros(PROFILE_BINS, dtype=np.float64)
    center_x = float(xs.mean())
    center_y = float(ys.mean())
    dx = xs.astype(np.float64) - center_x
    dy = ys.astype(np.float64) - center_y
    radius = np.hypot(dx, dy)
    angles = np.mod(np.arctan2(dy, dx), 2.0 * math.pi)
    bins = np.minimum(
        PROFILE_BINS - 1,
        (angles * PROFILE_BINS / (2.0 * math.pi)).astype(np.int64),
    )
    profile = np.full(PROFILE_BINS, -np.inf, dtype=np.float64)
    np.maximum.at(profile, bins, radius)
    known = np.flatnonzero(np.isfinite(profile))
    if len(known) == 0:
        return np.zeros(PROFILE_BINS, dtype=np.float64)
    if len(known) < PROFILE_BINS:
        extended_x = np.concatenate((known - PROFILE_BINS, known, known + PROFILE_BINS))
        extended_y = np.tile(profile[known], 3)
        profile = np.interp(np.arange(PROFILE_BINS), extended_x, extended_y)
    padded = np.concatenate((profile[-SMOOTH_RADIUS:], profile, profile[:SMOOTH_RADIUS]))
    kernel = np.ones(2 * SMOOTH_RADIUS + 1, dtype=np.float64) / (2 * SMOOTH_RADIUS + 1)
    return np.convolve(padded, kernel, mode="valid")


def _count_lobes(profile: np.ndarray) -> int:
    if not np.any(profile):
        return 0
    low = float(np.percentile(profile, 5.0))
    high = float(np.percentile(profile, 95.0))
    contrast = high - low
    if contrast < 8.0:
        return 0
    threshold = low + 0.58 * contrast
    active = profile >= threshold
    return int(np.sum(active & ~np.roll(active, 1)))


def decision_from_image(image: Image.Image) -> str:
    """Return the outward-tip count recovered solely from the RGB raster."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    background = _background_estimate(array)
    distance = np.linalg.norm(array - background[None, None, :], axis=2)
    foreground = distance > FOREGROUND_THRESHOLD
    component = _largest_component(foreground)
    return str(_count_lobes(_radial_profile(component)))
