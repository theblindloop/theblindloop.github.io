"""Independent pixel-only decision arm for the port-weave world.

Receives one final RGB image. It locates the single green port, recovers the
opaque plate that carries it by solid-run scanning, segments the dark bands
inside that plate, follows the band attached to the green port, and reports the
colour of the port it reaches. Anything inconsistent with the declared grammar
raises, which is an abstention rather than a guess.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BG = np.asarray((255, 255, 255), dtype=np.float64)
_PLATE = np.asarray((214, 214, 214), dtype=np.float64)
_GREEN = np.asarray((30, 150, 70), dtype=np.float64)
_TARGETS = {
    "orange": (235, 130, 25),
    "purple": (140, 60, 190),
    "blue": (35, 95, 215),
}
_SIDE_LEN = 85


def _solid_rect(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if not len(ys):
        raise ValueError("empty colour blob")
    y0, y1, x0, x1 = int(ys.min()), int(ys.max()), int(xs.min()), int(xs.max())
    height, width = y1 - y0 + 1, x1 - x0 + 1
    if int(mask.sum()) != height * width:
        raise ValueError("colour blob is not a solid rectangle")
    if not (9 <= height <= 22 and 9 <= width <= 22):
        raise ValueError("colour blob has an unexpected size")
    return y0, y1, x0, x1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float64)
    if array.ndim != 3:
        raise ValueError("expected an RGB raster")

    green = np.linalg.norm(array - _GREEN, axis=2) < 60.0
    gy0, gy1, gx0, gx1 = _solid_rect(green)
    cy, cx = (gy0 + gy1) // 2, (gx0 + gx1) // 2

    nonbg = np.linalg.norm(array - _BG, axis=2) > 20.0
    height, width = nonbg.shape
    x0, x1, y0, y1 = cx, cx, cy, cy
    while x0 - 1 >= 0 and nonbg[cy, x0 - 1]:
        x0 -= 1
    while x1 + 1 < width and nonbg[cy, x1 + 1]:
        x1 += 1
    while y0 - 1 >= 0 and nonbg[y0 - 1, cx]:
        y0 -= 1
    while y1 + 1 < height and nonbg[y1 + 1, cx]:
        y1 += 1
    if (x1 - x0 + 1) != _SIDE_LEN or (y1 - y0 + 1) != _SIDE_LEN:
        raise ValueError("the plate carrying the green port is not a complete square")
    if not nonbg[y0 : y1 + 1, x0 : x1 + 1].all():
        raise ValueError("the plate carrying the green port is punctured")

    patch = array[y0 : y1 + 1, x0 : x1 + 1]
    dark = patch.max(axis=2) < 110.0
    colored = (
        (~dark)
        & (np.linalg.norm(patch - _PLATE, axis=2) > 50.0)
        & (np.linalg.norm(patch - _BG, axis=2) > 50.0)
    )
    local_green = np.linalg.norm(patch - _GREEN, axis=2) < 60.0
    if int(local_green.sum()) != int(green.sum()):
        raise ValueError("the green port is not wholly inside its plate")

    ports = {}
    for name, value in _TARGETS.items():
        mask = colored & (np.linalg.norm(patch - np.asarray(value, dtype=np.float64), axis=2) < 60.0)
        if not mask.any():
            continue
        ports[name] = _solid_rect(mask)
    if len(ports) != 3:
        raise ValueError("the plate does not carry exactly three non-green ports")
    if int(colored.sum()) != int(local_green.sum()) + sum(
        (r[1] - r[0] + 1) * (r[3] - r[2] + 1) for r in ports.values()
    ):
        raise ValueError("unexpected coloured ink on the plate")

    def _halo(rect, pad=3):
        ry0 = max(0, rect[0] - pad)
        ry1 = min(patch.shape[0] - 1, rect[1] + pad)
        rx0 = max(0, rect[2] - pad)
        rx1 = min(patch.shape[1] - 1, rect[3] + pad)
        box = np.zeros_like(dark)
        box[ry0 : ry1 + 1, rx0 : rx1 + 1] = True
        return box

    gys, gxs = np.nonzero(local_green)
    green_rect = (int(gys.min()), int(gys.max()), int(gxs.min()), int(gxs.max()))
    seeds = np.argwhere(_halo(green_rect) & dark)
    if not len(seeds):
        raise ValueError("no band is attached to the green port")

    reached = np.zeros_like(dark)
    queue = deque()
    for sy, sx in seeds:
        if not reached[sy, sx]:
            reached[sy, sx] = True
            queue.append((int(sy), int(sx)))
    rows, cols = dark.shape
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < rows and 0 <= nx < cols and dark[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                queue.append((ny, nx))

    hits = [name for name, rect in ports.items() if (_halo(rect) & reached).any()]
    if len(hits) != 1:
        raise ValueError(f"the green band reaches {len(hits)} coloured ports")
    return hits[0]
