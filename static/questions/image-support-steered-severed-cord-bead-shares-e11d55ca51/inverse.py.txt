"""Independent pixel-only inverse arm for `severed_cord_bead_shares`.

Receives one rasterised RGB image and nothing else: no scene, no renderer
constant, no route, no palette table, no canvas pitch, no analytic gold. The
colour conventions used here are exactly the ones the public question declares
(a black cord, a green anchor square, a red anchor square, orange squares that
count only when they touch the black cord, grey and blue decoration).

Procedure: segment black ink, label its four-connected components, require the
single visible cut to have produced exactly two pieces, attribute each anchor
square to the piece it touches, attribute each orange component to the piece it
touches, require the five counting squares the question declares, then compare
the two touching-square totals.

Returns "green_more", "red_more", or "abstain" when the raster does not present
one severed black cord with two distinctly anchored pieces carrying exactly five
counting squares between them.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
MIN_CORD = 300
MIN_MARK = 40
DECLARED_SQUARES = 5      # the question declares exactly five counting squares


def _masks(array: np.ndarray) -> dict[str, np.ndarray]:
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    return {
        "cord": (red <= 70) & (green <= 70) & (blue <= 70),
        "orange": (red >= 200) & (green >= 90) & (green <= 190) & (blue <= 90),
        "green": (green >= 110) & (red <= 90) & (blue <= 90),
        "red": (red >= 150) & (green <= 80) & (blue <= 80),
    }


def _components(mask: np.ndarray, min_size: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    out = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        pixels = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= min_size:
            out.append(np.asarray(pixels, dtype=np.int64))
    return out


def _dilate(pixels: np.ndarray, shape: tuple[int, int]) -> np.ndarray:
    """One-step eight-connected dilation of a component, as a boolean mask."""
    field = np.zeros(shape, dtype=bool)
    field[pixels[:, 0], pixels[:, 1]] = True
    grown = field.copy()
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            grown |= np.roll(np.roll(field, dy, axis=0), dx, axis=1)
    return grown


def _touching(pixels: np.ndarray, pieces: list[np.ndarray], shape) -> list[int]:
    grown = _dilate(pixels, shape)
    hits = []
    for index, piece in enumerate(pieces):
        if bool(grown[piece[:, 0], piece[:, 1]].any()):
            hits.append(index)
    return hits


def decision_from_image(image: Image.Image) -> str:
    try:
        array = np.asarray(image.convert("RGB"), dtype=np.uint8)
        shape = array.shape[:2]
        masks = _masks(array)
        pieces = _components(masks["cord"], MIN_CORD)
        if len(pieces) != 2:
            return ABSTAIN
        anchor_piece = {}
        for name in ("green", "red"):
            blobs = _components(masks[name], MIN_MARK)
            if len(blobs) != 1:
                return ABSTAIN
            hits = _touching(blobs[0], pieces, shape)
            if len(hits) != 1:
                return ABSTAIN
            anchor_piece[name] = hits[0]
        if anchor_piece["green"] == anchor_piece["red"]:
            return ABSTAIN
        tally = [0, 0]
        for blob in _components(masks["orange"], MIN_MARK):
            hits = _touching(blob, pieces, shape)
            if len(hits) > 1:
                return ABSTAIN
            if hits:
                tally[hits[0]] += 1
        if sum(tally) != DECLARED_SQUARES:
            return ABSTAIN
        green = tally[anchor_piece["green"]]
        red = tally[anchor_piece["red"]]
        if green == red:
            return ABSTAIN
        return "green_more" if green > red else "red_more"
    except Exception:
        return ABSTAIN
