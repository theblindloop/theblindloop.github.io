"""Independent image-only inverse for the two visible die tiles.

The inverse identifies large dark rounded frame components, then counts small
dark circular components inside each frame. It shares no renderer geometry or
scene data and intentionally fails closed when the raster is incomplete.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleError(ValueError):
    """The raster does not expose two confidently measurable die faces."""


def _dark_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    return np.max(array, axis=2) < 132


def _components(mask: np.ndarray) -> list[dict[str, float]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    coords = np.argwhere(mask)
    output: list[dict[str, float]] = []
    for start_y, start_x in coords:
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        area = 0
        sum_x = 0
        sum_y = 0
        min_x = max_x = sx
        min_y = max_y = sy
        while stack:
            y, x = stack.pop()
            area += 1
            sum_x += x
            sum_y += y
            min_x = min(min_x, x)
            max_x = max(max_x, x)
            min_y = min(min_y, y)
            max_y = max(max_y, y)
            for ny, nx in (
                (y - 1, x),
                (y + 1, x),
                (y, x - 1),
                (y, x + 1),
                (y - 1, x - 1),
                (y - 1, x + 1),
                (y + 1, x - 1),
                (y + 1, x + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        output.append(
            {
                "area": float(area),
                "x0": float(min_x),
                "y0": float(min_y),
                "x1": float(max_x),
                "y1": float(max_y),
                "cx": float(sum_x / area),
                "cy": float(sum_y / area),
            }
        )
    return output


def _is_frame(component: dict[str, float]) -> bool:
    width = component["x1"] - component["x0"] + 1.0
    height = component["y1"] - component["y0"] + 1.0
    aspect = max(width, height) / min(width, height)
    return (
        component["area"] >= 1700.0
        and 130.0 <= width <= 250.0
        and 130.0 <= height <= 250.0
        and aspect <= 1.12
    )


def _is_pip(component: dict[str, float], frame: dict[str, float]) -> bool:
    width = component["x1"] - component["x0"] + 1.0
    height = component["y1"] - component["y0"] + 1.0
    aspect = max(width, height) / min(width, height)
    inside = (
        frame["x0"] + 15.0 < component["cx"] < frame["x1"] - 15.0
        and frame["y0"] + 15.0 < component["cy"] < frame["y1"] - 15.0
    )
    return (
        inside
        and 130.0 <= component["area"] <= 1500.0
        and 8.0 <= width <= 36.0
        and 8.0 <= height <= 36.0
        and aspect <= 1.35
    )


def _face_is_intact(image: Image.Image, frame: dict[str, float]) -> bool:
    """Reject ablations that remove a visible patch of a die's face.

    A valid face has one dominant light fill color inside its dark frame. The
    modal quantized color is inferred from the raster, so this check does not
    rely on renderer palette constants. It is intentionally one-sided: a
    damaged or partially erased tile is ambiguous and must be quarantined by
    the inverse rather than silently treated as a smaller die.
    """

    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    x0 = int(frame["x0"]) + 25
    y0 = int(frame["y0"]) + 25
    x1 = int(frame["x1"]) - 24
    y1 = int(frame["y1"]) - 24
    if x1 <= x0 or y1 <= y0:
        return False
    crop = array[y0 : y1 + 1, x0 : x1 + 1]
    light = np.max(crop, axis=2) >= 132
    pixels = crop[light]
    if len(pixels) < 1000:
        return False
    quantized = pixels // 8
    _, counts = np.unique(quantized, axis=0, return_counts=True)
    # The most common light quantization bin is the die fill in an intact face.
    mode_fraction = float(counts.max() / len(pixels))
    return mode_fraction >= 0.78


def _pip_counts(image: Image.Image) -> list[int]:
    mask = _dark_mask(image)
    frames = [component for component in _components(mask) if _is_frame(component)]
    if len(frames) != 2:
        raise OracleError(f"expected two separated die frames, found {len(frames)}")
    frames.sort(key=lambda item: (item["cy"], item["cx"]))
    if any(not _face_is_intact(image, frame) for frame in frames):
        raise OracleError("one or more die faces are incomplete or ablated")
    components = _components(mask)
    counts = [sum(_is_pip(component, frame) for component in components) for frame in frames]
    if any(count < 1 or count > 6 for count in counts):
        raise OracleError(f"unreliable pip counts: {counts}")
    return counts


def decision_from_image(image: Image.Image) -> str:
    """Return the yes/no label from the final RGB pixels alone."""

    counts = _pip_counts(image)
    return "yes" if sum(counts) == 7 else "no"
