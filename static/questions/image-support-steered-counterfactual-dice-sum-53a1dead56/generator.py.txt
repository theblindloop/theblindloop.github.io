"""Deterministic renderer for the visible two-die pip-sum world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 620
BACKGROUND = (246, 241, 230)
PIP_COLOR = (24, 29, 36)
FRAME_WIDTH = 9

FACE_COLORS = (
    (235, 246, 255),
    (255, 236, 222),
    (235, 250, 236),
    (248, 236, 255),
    (255, 247, 211),
    (225, 246, 245),
)
OUTLINE_COLORS = (
    (35, 57, 80),
    (73, 46, 86),
    (42, 78, 61),
    (83, 58, 36),
)

# Coordinates are fractional positions in the usable face square. The layout
# is deliberately standard-looking but the answer is read from the visible
# pixels, never from a remembered die convention.
PIP_LAYOUTS: dict[int, tuple[tuple[float, float], ...]] = {
    1: ((0.50, 0.50),),
    2: ((0.28, 0.28), (0.72, 0.72)),
    3: ((0.28, 0.28), (0.50, 0.50), (0.72, 0.72)),
    4: ((0.28, 0.28), (0.72, 0.28), (0.28, 0.72), (0.72, 0.72)),
    5: ((0.28, 0.28), (0.72, 0.28), (0.50, 0.50), (0.28, 0.72), (0.72, 0.72)),
    6: (
        (0.28, 0.24),
        (0.72, 0.24),
        (0.28, 0.50),
        (0.72, 0.50),
        (0.28, 0.76),
        (0.72, 0.76),
    ),
}


def _pair_for_seed(seed: int) -> tuple[int, int]:
    """Use parity for balanced stress sweeps, then rotate nuisance pairings."""

    rng = random.Random(seed ^ 0x5EEDC0DE)
    if seed % 2:
        pairs = ((1, 6), (2, 5), (3, 4), (4, 3), (5, 2), (6, 1))
    else:
        # Near-boundary counterfactuals make the familiar seven prior tempting
        # without allowing a count of seven to be guessed from a layout cue.
        pairs = ((1, 5), (2, 4), (3, 3), (4, 2), (5, 1), (2, 6), (3, 5), (4, 4))
    return pairs[rng.randrange(len(pairs))]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, answer-counterbalanced two-tile scene."""

    rng = random.Random(seed ^ 0xA17C91)
    first, second = _pair_for_seed(seed)
    # The two support regions remain near opposite diagonal corners, while the
    # small independent jitter prevents a fixed crop or position from carrying
    # the answer. Both tiles stay comfortably inside the 1000x620 canvas.
    center_a = (165 + rng.randint(-20, 20), 150 + rng.randint(-18, 18))
    center_b = (835 + rng.randint(-20, 20), 470 + rng.randint(-18, 18))
    size_a = rng.randrange(174, 199, 2)
    size_b = rng.randrange(174, 199, 2)
    pip_radius_a = rng.randrange(11, 14)
    pip_radius_b = rng.randrange(11, 14)
    face_order = rng.sample(range(len(FACE_COLORS)), 2)
    outline_order = rng.sample(range(len(OUTLINE_COLORS)), 2)

    records = [
        {
            "slot": 0,
            "center": [center_a[0], center_a[1]],
            "size": size_a,
            "corner_radius": 19,
            "face_color": list(FACE_COLORS[face_order[0]]),
            "outline_color": list(OUTLINE_COLORS[outline_order[0]]),
            "pip_radius": pip_radius_a,
            "pips": first,
        },
        {
            "slot": 1,
            "center": [center_b[0], center_b[1]],
            "size": size_b,
            "corner_radius": 19,
            "face_color": list(FACE_COLORS[face_order[1]]),
            "outline_color": list(OUTLINE_COLORS[outline_order[1]]),
            "pip_radius": pip_radius_b,
            "pips": second,
        },
    ]
    # Serialization order is deliberately nuisance-only; rendering and gold
    # sort by the visible slot, which gives a concrete pixel-identical alias.
    if rng.randrange(2):
        records.reverse()
    return {
        "dice": records,
        "canvas_size": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
        "frame_width": FRAME_WIDTH,
        "pip_color": list(PIP_COLOR),
    }


def _tile_box(die: dict[str, Any]) -> tuple[int, int, int, int]:
    cx, cy = (float(value) for value in die["center"])
    size = int(die["size"])
    half = size / 2.0
    return (
        int(round(cx - half)),
        int(round(cy - half)),
        int(round(cx + half)),
        int(round(cy + half)),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible geometry from the serialized scene."""

    width, height = (int(value) for value in scene["canvas_size"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    frame_width = int(scene["frame_width"])
    pip_color = tuple(scene["pip_color"])

    for die in sorted(scene["dice"], key=lambda item: int(item["slot"])):
        x0, y0, x1, y1 = _tile_box(die)
        size = int(die["size"])
        radius = int(die["corner_radius"])
        draw.rounded_rectangle(
            (x0, y0, x1, y1),
            radius=radius,
            fill=tuple(die["face_color"]),
            outline=tuple(die["outline_color"]),
            width=frame_width,
        )
        usable = size - 2 * frame_width
        pip_radius = int(die["pip_radius"])
        for px, py in PIP_LAYOUTS[int(die["pips"])]:
            center_x = x0 + frame_width + px * usable
            center_y = y0 + frame_width + py * usable
            draw.ellipse(
                (
                    int(round(center_x - pip_radius)),
                    int(round(center_y - pip_radius)),
                    int(round(center_x + pip_radius)),
                    int(round(center_y + pip_radius)),
                ),
                fill=pip_color,
            )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute the visible question label from the rendered die records."""

    total = sum(int(die["pips"]) for die in scene["dice"])
    return "yes" if total == 7 else "no"


def margin(scene: dict[str, Any]) -> float:
    """Signed distance to the half-integer boundary around the exact total."""

    total = sum(int(die["pips"]) for die in scene["dice"])
    return float(0.5 - abs(total - 7))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject only scenes that would be ambiguous at the declared raster size."""

    try:
        width, height = (int(value) for value in scene["canvas_size"])
        dice = scene["dice"]
        if len(dice) != 2 or width < 500 or height < 300:
            return True
        boxes = [_tile_box(die) for die in dice]
        for die, box in zip(dice, boxes, strict=True):
            x0, y0, x1, y1 = box
            if int(die["size"]) < 150 or int(die["pip_radius"]) < 9:
                return True
            if x0 < 8 or y0 < 8 or x1 >= width - 8 or y1 >= height - 8:
                return True
        ax = (boxes[0][0] + boxes[0][2]) / 2.0
        ay = (boxes[0][1] + boxes[0][3]) / 2.0
        bx = (boxes[1][0] + boxes[1][2]) / 2.0
        by = (boxes[1][1] + boxes[1][3]) / 2.0
        clearance = math.hypot(ax - bx, ay - by) - (int(dice[0]["size"]) + int(dice[1]["size"])) / 2.0
        return clearance < 80.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Expose the unordered-record alias used by the v0.4 latent gate."""

    twin = dict(scene)
    twin["dice"] = list(reversed(scene["dice"]))
    return [("unordered_dice_records", twin)]
