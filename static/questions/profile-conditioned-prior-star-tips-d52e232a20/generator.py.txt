"""Star tip-count world under a contradictory canonical prior.

A single filled star polygon is drawn on a plain background. The canonical
prior (an ordinary five-pointed star) has five points, but the drawn star can
have any point count in ALLOWED_POINTS. The gold answer is exactly the number
of pointed tips that are actually rendered.

The observable margin of the count is the radial tip prominence, i.e. the
difference between the star's outer (tip) radius and its inner (notch) radius,
in pixels. When this prominence is too small the tips become ambiguous at pixel
scale and the scene is quarantined (the sampler never produces such scenes).
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_star_tips-0.1.0"

ALLOWED_POINTS = (4, 5, 6, 7, 8, 9, 10)
COLORS = ["blue", "crimson", "darkgreen", "purple", "navy", "teal"]
CANVAS = (512, 512)
OUTER_MIN, OUTER_MAX = 170.0, 220.0
INNER_RATIO_MIN, INNER_RATIO_MAX = 0.42, 0.55
MIN_PROMINENCE_PX = 14.0


def _star_vertices(cx, cy, outer, inner, n, rot):
    verts = []
    for i in range(2 * n):
        r = outer if i % 2 == 0 else inner
        ang = rot + i * math.pi / n
        verts.append((cx + r * math.cos(ang), cy + r * math.sin(ang)))
    return verts


def sample_scene(seed):
    rng = random.Random(seed)
    n = rng.choice(ALLOWED_POINTS)
    outer = rng.uniform(OUTER_MIN, OUTER_MAX)
    ratio = rng.uniform(INNER_RATIO_MIN, INNER_RATIO_MAX)
    inner = outer * ratio
    cx = rng.uniform(OUTER_MAX + 20, CANVAS[0] - OUTER_MAX - 20)
    cy = rng.uniform(OUTER_MAX + 20, CANVAS[1] - OUTER_MAX - 20)
    rot = rng.uniform(0.0, math.pi / n)
    color = rng.choice(COLORS)
    return {
        "points": n,
        "outer_radius": float(round(outer, 4)),
        "inner_radius": float(round(inner, 4)),
        "center_x": float(round(cx, 4)),
        "center_y": float(round(cy, 4)),
        "rotation": float(round(rot, 4)),
        "color": color,
        "canvas": list(CANVAS),
    }


def render(scene, supersample=4):
    ss = supersample
    W, H = scene["canvas"]
    img = Image.new("RGB", (W * ss, H * ss), "white")
    draw = ImageDraw.Draw(img)
    verts = _star_vertices(
        scene["center_x"] * ss,
        scene["center_y"] * ss,
        scene["outer_radius"] * ss,
        scene["inner_radius"] * ss,
        scene["points"],
        scene["rotation"],
    )
    draw.polygon(verts, fill=scene["color"])
    img = img.resize((W, H), Image.LANCZOS)
    return img


def analytic_gold(scene):
    return str(scene["points"])


def margin(scene):
    return round(scene["outer_radius"] - scene["inner_radius"], 6)


def is_quarantined(scene):
    return (scene["outer_radius"] - scene["inner_radius"]) < MIN_PROMINENCE_PX


def latent_symmetries(scene):
    """No nontrivial render-invariant latent symmetry.

    Every scene field (point count, radii, centre, rotation, colour, canvas) is
    directly and uniquely reflected in the raster, so the parametrization is
    injective and the count is not aliasable by any re-labelling of the scene.
    Declared empty (mode not_applicable).
    """
    return iter(())
