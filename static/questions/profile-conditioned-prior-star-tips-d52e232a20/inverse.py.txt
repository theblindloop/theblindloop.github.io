"""Scene-free pixel oracle for the star tip-count world.

This module never receives the latent scene and imports nothing from renderer,
prompts, generate, or verify. It reconstructs the number of pointed tips of a
single filled star from the raster alone.

Pipeline:
  1. segment the non-background (star) pixels,
  2. compute the centroid of the star region,
  3. build the radial-extent profile r(theta) = max distance from the centroid
     to a star pixel at each angle,
  4. lightly smooth the profile to suppress anti-aliasing noise,
  5. split the profile at the mid-radius between the outer tips and inner
     notches, and count the contiguous angular runs above that level; each run
     is one outward tip.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

NBINS = 720
SMOOTH = 9
MIN_RUN = 8


def decision_from_image(image):
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    bg = np.abs(arr - 255).sum(axis=2) > 40
    ys, xs = np.nonzero(bg)
    if xs.size == 0:
        return None
    cx = float(xs.mean())
    cy = float(ys.mean())
    dx = xs - cx
    dy = ys - cy
    radius = np.sqrt(dx * dx + dy * dy)
    theta = np.degrees(np.arctan2(dy, dx)) % 360.0
    bins = ((theta / 360.0) * NBINS).astype(np.int64) % NBINS
    rmax = np.zeros(NBINS, dtype=np.float64)
    np.maximum.at(rmax, bins, radius)
    rmax = np.where(rmax > 0.0, rmax, rmax.max())
    if rmax.max() <= 0.0:
        return None
    radius = np.concatenate([rmax[-SMOOTH // 2 :], rmax, rmax[: SMOOTH // 2]])
    kernel = np.ones(SMOOTH) / SMOOTH
    smoothed = np.convolve(radius, kernel, mode="valid")
    lo = smoothed.min()
    mid = (smoothed.max() + lo) / 2.0
    above = smoothed > mid
    n = len(above)
    widths = []
    in_run = False
    start = 0
    for i in range(n):
        if above[i]:
            if not in_run:
                start = i
                in_run = True
        elif in_run:
            widths.append(i - start)
            in_run = False
    if in_run:
        widths.append(n - start)
    if above[0] and above[-1] and n > 1:
        widths[0] += widths[-1]
        widths.pop()
    return str(sum(1 for width in widths if width >= MIN_RUN))
