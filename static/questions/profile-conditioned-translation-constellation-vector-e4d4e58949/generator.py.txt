"""Deterministic renderer for a repeated-translation constellation task."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
FIELD_BOTTOM = 394
DOT_RADIUS = 6
LANDING_RADIUS = 7.0
COLORS = {"orange": (232, 101, 24), "blue": (28, 112, 214)}
VECTOR_POOL = [
    (42, 0), (-42, 0), (36, 24), (36, -24), (-36, 24), (-36, -24),
    (28, 32), (28, -32), (-28, 32), (-28, -32),
]


def _landing_count(dots: list[list[int]], dx: int, dy: int) -> int:
    count = 0
    for i, (x, y) in enumerate(dots):
        tx, ty = x + dx, y + dy
        if any(
            j != i and math.hypot(tx - qx, ty - qy) <= LANDING_RADIUS
            for j, (qx, qy) in enumerate(dots)
        ):
            count += 1
    return count


def _scores(scene: dict) -> list[tuple[str, int]]:
    dots = scene["dots"]
    vectors = scene["vectors"]
    return [
        (item["color"], _landing_count(dots, int(item["dx"]), int(item["dy"])))
        for item in vectors
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    true_vector = VECTOR_POOL[rng.randrange(len(VECTOR_POOL))]
    false_choices = [
        value
        for value in VECTOR_POOL
        if value != true_vector and value != (-true_vector[0], -true_vector[1])
    ]
    false_vector = rng.choice(false_choices)
    pair_count = 4 + (abs(int(seed)) % 4)

    for _attempt in range(2000):
        dots: list[list[int]] = []

        def add(point: tuple[int, int]) -> bool:
            x, y = point
            if not (24 <= x <= WIDTH - 24 and 24 <= y <= FIELD_BOTTOM - 24):
                return False
            if any(math.hypot(x - qx, y - qy) < 17 for qx, qy in dots):
                return False
            dots.append([x, y])
            return True

        bases: list[tuple[int, int]] = []
        for _ in range(pair_count):
            placed = False
            for _ in range(1000):
                x = rng.randint(35, WIDTH - 35)
                y = rng.randint(35, FIELD_BOTTOM - 35)
                target = (x + true_vector[0], y + true_vector[1])
                before = len(dots)
                if add((x, y)) and add(target):
                    bases.append((x, y))
                    placed = True
                    break
                del dots[before:]
            if not placed:
                break
        if len(bases) != pair_count:
            continue

        while len(dots) < 18:
            if not add((rng.randint(28, WIDTH - 28), rng.randint(28, FIELD_BOTTOM - 28))):
                continue

        true_score = _landing_count(dots, *true_vector)
        false_score = _landing_count(dots, *false_vector)
        if true_score - false_score < 2:
            continue

        true_color = "orange" if int(seed) % 2 == 0 else "blue"
        false_color = "blue" if true_color == "orange" else "orange"
        slots = [0, 1]
        rng.shuffle(slots)
        vectors = [
            {"color": true_color, "dx": true_vector[0], "dy": true_vector[1], "slot": slots[0]},
            {"color": false_color, "dx": false_vector[0], "dy": false_vector[1], "slot": slots[1]},
        ]
        vectors.sort(key=lambda item: item["color"])
        return {"width": WIDTH, "height": HEIGHT, "dots": dots, "vectors": vectors}
    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    dots = scene["dots"]
    vectors = scene["vectors"]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    draw.line([(18 * scale, 402 * scale), ((width - 18) * scale, 402 * scale)], fill=(205, 205, 200), width=2 * scale)
    for x, y in dots:
        r = DOT_RADIUS * scale
        draw.ellipse((x * scale - r, y * scale - r, x * scale + r, y * scale + r), fill=(24, 27, 29))
    for item in vectors:
        color = COLORS[item["color"]]
        sx = (124 if int(item["slot"]) == 0 else 364) * scale
        sy = 456 * scale
        ex = sx + int(item["dx"]) * scale
        ey = sy + int(item["dy"]) * scale
        draw.line([(sx, sy), (ex, ey)], fill=(174, 174, 169), width=2 * scale)
        rr = 9 * scale
        draw.ellipse((sx - rr, sy - rr, sx + rr, sy + rr), outline=color, width=3 * scale)
        h = 7 * scale
        draw.rectangle((ex - h, ey - h, ex + h, ey + h), fill=color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    ranked = sorted(_scores(scene), key=lambda item: (-item[1], item[0]))
    return ranked[0][0]


def margin(scene: dict) -> float:
    values = sorted((score for _, score in _scores(scene)), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["dots"] = list(reversed(twin["dots"]))
    return [("reverse_dot_order", twin)]
