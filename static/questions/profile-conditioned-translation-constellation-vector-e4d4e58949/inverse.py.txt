"""Independent inverse arm: only the final RGB raster enters this module."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {"orange": np.array((232, 101, 24)), "blue": np.array((28, 112, 214))}
LANDING_RADIUS = 8.5


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        groups.append(np.asarray(pixels, dtype=float))
    return groups


def _dot_centers(arr: np.ndarray) -> list[tuple[float, float]]:
    mask = (arr[:398, :, :].max(axis=2) < 90)
    groups = [group for group in _components(mask) if len(group) > 45]
    centers = [(float(group[:, 1].mean()), float(group[:, 0].mean())) for group in groups]
    if len(centers) < 8:
        raise ValueError("too few black dot components")
    return centers


def _vector(arr: np.ndarray, name: str) -> tuple[float, float]:
    target = PALETTE[name]
    delta = arr.astype(float) - target
    mask = (np.sqrt((delta * delta).sum(axis=2)) < 75)
    mask[:405, :] = False
    groups = [group for group in _components(mask) if len(group) > 35]
    if len(groups) != 2:
        raise ValueError(f"expected two {name} glyph endpoints, found {len(groups)}")
    centers = [(float(group[:, 1].mean()), float(group[:, 0].mean()), group) for group in groups]
    ring = None
    square = None
    for cx, cy, group in centers:
        yy, xx = int(round(cy)), int(round(cx))
        if mask[yy, xx]:
            square = (cx, cy)
        else:
            ring = (cx, cy)
    if ring is None or square is None:
        raise ValueError(f"could not distinguish {name} ring and square")
    return (square[0] - ring[0], square[1] - ring[1])


def _count(dots: list[tuple[float, float]], vector: tuple[float, float]) -> int:
    dx, dy = vector
    count = 0
    for i, (x, y) in enumerate(dots):
        if any(
            j != i and math.hypot(x + dx - qx, y + dy - qy) <= LANDING_RADIUS
            for j, (qx, qy) in enumerate(dots)
        ):
            count += 1
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    dots = _dot_centers(arr)
    scores = {name: _count(dots, _vector(arr, name)) for name in PALETTE}
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ranked[0][1] == ranked[1][1]:
        raise ValueError("pixel correspondence scores are tied")
    return ranked[0][0]
