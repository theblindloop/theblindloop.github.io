"""Scene sampling and rendering for overlapping-neighborhood density search."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

SIZE = 400
GRID_SIDE = 11
CELL = 24
X0 = (SIZE - GRID_SIDE * CELL) // 2
Y0 = X0
CYAN = ((35, 178, 190), (79, 205, 210))
BACKGROUND = (255, 255, 255)
MARKER_RGB = {
    "red": (220, 62, 55),
    "gold": (238, 173, 42),
    "violet": (133, 79, 186),
    "green": (58, 151, 86),
}
COLORS = tuple(MARKER_RGB)
# Their 5x5 neighborhoods meet along row/column 5, so the union is one compact
# 9x9 regional support rather than four disjoint local windows.
POSITIONS = ((3, 3), (3, 7), (7, 3), (7, 7))
HOLE_COUNT = 22


def _scores(grid: list[list[int]], markers: list[dict[str, Any]]) -> dict[str, int]:
    values: dict[str, int] = {}
    for marker in markers:
        row, col = int(marker["row"]), int(marker["col"])
        values[str(marker["color"])] = sum(
            int(grid[y][x])
            for y in range(row - 2, row + 3)
            for x in range(col - 2, col + 3)
        )
    return values


def _connected(grid: list[list[int]]) -> bool:
    occupied = {(y, x) for y in range(GRID_SIDE) for x in range(GRID_SIDE) if grid[y][x]}
    if not occupied:
        return False
    seen = {next(iter(occupied))}
    stack = list(seen)
    while stack:
        y, x = stack.pop()
        for point in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if point in occupied and point not in seen:
                seen.add(point)
                stack.append(point)
    return seen == occupied


def sample_scene(seed: int) -> dict:
    """Create a balanced scene with a unique, visibly separated maximum."""

    rng = random.Random(int(seed))
    desired = COLORS[int(seed) % len(COLORS)]
    positions = list(POSITIONS)
    rng.shuffle(positions)
    markers = [
        {"color": color, "row": position[0], "col": position[1]}
        for color, position in zip(COLORS, positions, strict=True)
    ]
    marker_cells = {(int(item["row"]), int(item["col"])) for item in markers}
    removable = [
        (y, x)
        for y in range(1, GRID_SIDE - 1)
        for x in range(1, GRID_SIDE - 1)
        if (y, x) not in marker_cells
    ]
    for _ in range(20000):
        holes = set(rng.sample(removable, HOLE_COUNT))
        grid = [
            [0 if (y, x) in holes else 1 for x in range(GRID_SIDE)]
            for y in range(GRID_SIDE)
        ]
        if not _connected(grid):
            continue
        scores = _scores(grid, markers)
        ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
        if ranked[0][0] == desired and ranked[0][1] - ranked[1][1] >= 2:
            return {"grid": grid, "markers": markers}
    raise RuntimeError(f"could not sample a separated scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Render only the declared occupancy grid and marker bindings."""

    grid = scene["grid"]
    markers = scene["markers"]
    image = Image.new("RGB", (SIZE, SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for row in range(GRID_SIDE):
        for col in range(GRID_SIDE):
            if int(grid[row][col]):
                x0, y0 = X0 + col * CELL, Y0 + row * CELL
                draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=CYAN[(row + col) % 2])
    for marker in markers:
        row, col = int(marker["row"]), int(marker["col"])
        cx = X0 + col * CELL + CELL // 2
        cy = Y0 + row * CELL + CELL // 2
        radius = 6
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=MARKER_RGB[str(marker["color"])],
            outline=(35, 35, 40),
            width=1,
        )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["grid"], scene["markers"])
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if len(ranked) != 4 or ranked[0][1] == ranked[1][1]:
        raise ValueError("scene has no unique density winner")
    return ranked[0][0]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene["grid"], scene["markers"]).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
