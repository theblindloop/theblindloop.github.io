"""Independent PNG-only inverse for overlapping-neighborhood density search."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID_SIDE = 11
KNOWN_CYAN = np.asarray(((35, 178, 190), (79, 205, 210)), dtype=np.int16)
MARKER_RGB = {
    "red": (220, 62, 55),
    "gold": (238, 173, 42),
    "violet": (133, 79, 186),
    "green": (58, 151, 86),
}


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 38.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2)) <= tolerance


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    points = set(map(tuple, np.argwhere(mask).tolist()))
    found: list[list[tuple[int, int]]] = []
    while points:
        start = points.pop()
        component = [start]
        queue = deque([start])
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in points:
                    points.remove(neighbor)
                    component.append(neighbor)
                    queue.append(neighbor)
        found.append(component)
    return found


def _marker_centers(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    result: dict[str, tuple[float, float]] = {}
    for color, rgb in MARKER_RGB.items():
        candidates = [component for component in _components(_near(arr, rgb)) if len(component) >= 40]
        if len(candidates) != 1:
            raise ValueError(f"expected one visible {color} marker")
        points = np.asarray(candidates[0], dtype=np.float64)
        result[color] = (float(points[:, 0].mean()), float(points[:, 1].mean()))
    return result


def decision_from_image(image: Image.Image) -> str:
    """Recover the grid, enumerate all markers, and require a unique winner."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 100 or arr.shape[1] < 100:
        raise ValueError("image is too small")
    cyan_delta = arr.astype(np.int32)[:, :, None, :] - KNOWN_CYAN.astype(np.int32)[None, None, :, :]
    cyan = np.sqrt(np.sum(cyan_delta * cyan_delta, axis=3)).min(axis=2) <= 38.0
    marker_masks = [_near(arr, rgb) for rgb in MARKER_RGB.values()]
    foreground = cyan.copy()
    for mask in marker_masks:
        foreground |= mask
    ys, xs = np.nonzero(foreground)
    if len(xs) < 1000:
        raise ValueError("mosaic evidence is missing")
    left, right = int(xs.min()), int(xs.max()) + 1
    top, bottom = int(ys.min()), int(ys.max()) + 1
    width, height = right - left, bottom - top
    if abs(width - height) > 2 or not 160 <= width <= 280:
        raise ValueError("could not recover the square mosaic boundary")
    cell_x, cell_y = width / GRID_SIDE, height / GRID_SIDE
    if abs(cell_x - cell_y) > 0.3:
        raise ValueError("recovered lattice cells are not square")

    grid = [[0] * GRID_SIDE for _ in range(GRID_SIDE)]
    for row in range(GRID_SIDE):
        for col in range(GRID_SIDE):
            x0 = int(round(left + col * cell_x))
            x1 = int(round(left + (col + 1) * cell_x))
            y0 = int(round(top + row * cell_y))
            y1 = int(round(top + (row + 1) * cell_y))
            patch = cyan[y0:y1, x0:x1]
            grid[row][col] = int(float(patch.mean()) >= 0.35)

    centers = _marker_centers(arr)
    scores: dict[str, int] = {}
    occupied = {(row, col) for row in range(GRID_SIDE) for col in range(GRID_SIDE) if grid[row][col]}
    for color, (cy, cx) in centers.items():
        row = int((cy - top) // cell_y)
        col = int((cx - left) // cell_x)
        if not (2 <= row < GRID_SIDE - 2 and 2 <= col < GRID_SIDE - 2):
            raise ValueError(f"{color} marker is outside a complete 5-by-5 neighborhood")
        occupied.add((row, col))
        scores[color] = sum(
            (y, x) in occupied
            for y in range(row - 2, row + 3)
            for x in range(col - 2, col + 3)
        )
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if len(ranked) != 4 or ranked[0][1] == ranked[1][1]:
        raise ValueError("pixel evidence has no unique winner")
    return ranked[0][0]
