"""Independent image-only inverse arm using only final RGB pixels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
EXPECTED_BAR = np.array((38, 66, 86), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (x0,y0,x1,y1,area) for 4-connected mask components."""
    work = mask.copy()
    h, w = work.shape
    out: list[tuple[int, int, int, int, int]] = []
    for sy, sx in zip(*np.nonzero(work), strict=True):
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(int(sy), int(sx))])
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if len(xs) >= 250:
            out.append((min(xs), min(ys), max(xs) + 1, max(ys) + 1, len(xs)))
    return out


def _cluster_panels(bars: list[tuple[int, int, int, int, int]]) -> list[list[tuple[int, int, int, int, int]]]:
    ordered = sorted(bars, key=lambda box: (box[0] + box[2]) / 2)
    if len(ordered) != 16:
        raise ValueError(f"expected 16 visible bars, found {len(ordered)}")
    centers = [(box[0] + box[2]) / 2 for box in ordered]
    gaps = [b - a for a, b in zip(centers, centers[1:])]
    split_indices = sorted(range(len(gaps)), key=lambda i: gaps[i], reverse=True)[:3]
    if not split_indices or min(gaps[i] for i in split_indices) < 1.8 * max(
        gap for i, gap in enumerate(gaps) if i not in split_indices
    ):
        raise ValueError("panel grouping is not visually separated")
    cuts = sorted(i + 1 for i in split_indices)
    groups = []
    start = 0
    for stop in [*cuts, len(ordered)]:
        groups.append(ordered[start:stop])
        start = stop
    if [len(group) for group in groups] != [4, 4, 4, 4]:
        raise ValueError("each panel must contain four bars")
    return groups


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    target = EXPECTED_BAR.astype(np.int32)
    distance = np.sqrt(np.square(arr - target).sum(axis=2))
    mask = distance < 42.0
    bars = _components(mask)
    groups = _cluster_panels(bars)

    scores: list[int] = []
    for group in groups:
        group = sorted(group, key=lambda box: (box[0] + box[2]) / 2)
        heights = [box[3] - box[1] for box in group]
        if min(abs(a - b) for i, a in enumerate(heights) for b in heights[i + 1 :]) < 8:
            raise ValueError("bar heights are not cleanly distinguishable")
        scores.append(sum(a < b for a, b in zip(heights, heights[1:])))
    best = max(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    runner_up = sorted(scores, reverse=True)[1]
    if len(winners) != 1 or best != 3 or best - runner_up < 1:
        raise ValueError("no unique monotone winner with runner-up separation")
    return LABELS[winners[0]]
