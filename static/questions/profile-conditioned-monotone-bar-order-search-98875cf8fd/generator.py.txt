"""Deterministic scene sampler and renderer for monotone bar-order search."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 420
SCALE = 3
LABELS = ("A", "B", "C", "D")
BAR_RGB = (38, 66, 86)
INK_RGB = (31, 38, 46)
PANEL_RGB = (180, 190, 198)
ACCENT_RGB = (214, 112, 48)
BACKGROUND = (249, 247, 242)


def _ascent_score(order: list[int]) -> int:
    return sum(a < b for a, b in zip(order, order[1:]))


def sample_scene(seed: int) -> dict:
    """Sample four equal-ink bar permutations with one monotone winner."""
    rng = random.Random(int(seed))
    gaps = [rng.randint(22, 30) for _ in range(3)]
    heights = [rng.randint(48, 60)]
    for gap in gaps:
        heights.append(heights[-1] + gap)
    width = rng.randint(16, 21)

    target = [0, 1, 2, 3]
    # Hard distractors satisfy two of the three adjacent comparisons. This makes
    # every distractor locally plausible while leaving a one-constraint gap.
    hard = [
        list(p)
        for p in itertools.permutations(range(4))
        if _ascent_score(list(p)) == 2
    ]
    rng.shuffle(hard)
    orders = [target, *hard[:3]]
    rng.shuffle(orders)
    panels = [
        {"order": order, "heights": heights, "bar_width": width}
        for order in orders
    ]
    return {"panels": panels}


def _panel_scores(scene: dict) -> list[int]:
    panels = scene["panels"]
    return [_ascent_score(panel["order"]) for panel in panels]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    winners = [i for i, score in enumerate(scores) if score == 3]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one increasing panel")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    panels = scene["panels"]
    if len(panels) != 4:
        return True
    try:
        heights = sorted(int(v) for v in panels[0]["heights"])
        if len(heights) != 4 or min(b - a for a, b in zip(heights, heights[1:])) < 12:
            return True
        for panel in panels:
            if sorted(panel["order"]) != [0, 1, 2, 3]:
                return True
            if list(panel["heights"]) != list(panels[0]["heights"]):
                return True
            if int(panel["bar_width"]) != int(panels[0]["bar_width"]):
                return True
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every declared latent field is visibly rendered; no pixel-identical latent
    # aliases are claimed.
    return []


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    if len(panels) != 4:
        raise ValueError("render requires four panels")

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=21 * SCALE)
    title_font = ImageFont.load_default(size=17 * SCALE)
    draw.text(
        (30 * SCALE, 22 * SCALE),
        "FIND THE PANEL THAT RISES LEFT TO RIGHT",
        fill=INK_RGB,
        font=title_font,
    )

    panel_w = 154
    panel_h = 270
    panel_y = 82
    x0s = [24, 197, 370, 543]
    for idx, (x0, panel) in enumerate(zip(x0s, panels, strict=True)):
        box = (x0 * SCALE, panel_y * SCALE, (x0 + panel_w) * SCALE, (panel_y + panel_h) * SCALE)
        draw.rounded_rectangle(box, radius=10 * SCALE, fill=(255, 255, 253), outline=PANEL_RGB, width=2 * SCALE)
        baseline = (panel_y + 218) * SCALE
        draw.line(
            ((x0 + 15) * SCALE, baseline + 1 * SCALE, (x0 + panel_w - 15) * SCALE, baseline + 1 * SCALE),
            fill=(215, 219, 220),
            width=2 * SCALE,
        )
        centers = [x0 + 30, x0 + 61, x0 + 92, x0 + 123]
        bar_width = int(panel["bar_width"])
        heights = panel["heights"]
        order = panel["order"]
        for cx, rank in zip(centers, order, strict=True):
            height = int(heights[rank])
            left = int((cx - bar_width / 2) * SCALE)
            right = int((cx + bar_width / 2) * SCALE)
            draw.rounded_rectangle(
                (left, baseline - height * SCALE, right, baseline),
                radius=3 * SCALE,
                fill=BAR_RGB,
            )
        label = LABELS[idx]
        bbox = draw.textbbox((0, 0), label, font=font)
        tx = (x0 + panel_w / 2) * SCALE - (bbox[2] - bbox[0]) / 2
        draw.text((tx, (panel_y + 232) * SCALE), label, fill=ACCENT_RGB, font=font)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
