"""Renderer for crosslap_over_order_bind.

One slate SLAB carries four crossing-piles on trays: the pale tray holds the KEY
pile, the three grey trays hold candidate piles.  Every pile is built from the
same three bar colours: one SPINE bar that crosses the two other bars, and two
CROSS bars that never touch each other.  Each pile is drawn under its own
independent rotation/reflection, so a colour's orientation and position carry no
information.  The only thing that can differ between piles is which colour lies
on top at each of the two overlaps.  Exactly one candidate pile repeats both of
the key's over/under facts; the answer is that candidate's marker colour.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "crosslap_over_order_bind-0.1.0"
QUARANTINE_MARGIN = 10.0

CANVAS = 640
SLAB = 276
TRAY = 124
PILE = 112
GAP = 8
PAD = 10

PAGE = (250, 246, 236)
SLAB_COLOR = (60, 66, 78)
KEY_TRAY = (232, 226, 210)
CAND_TRAY = (150, 156, 168)

BAR_COLORS = {
    "red": (214, 52, 48),
    "blue": (40, 86, 200),
    "green": (28, 140, 70),
    "purple": (128, 52, 168),
    "yellow": (228, 180, 30),
}
MARKER_COLORS = {
    "orange": (240, 124, 20),
    "teal": (0, 150, 152),
    "magenta": (206, 44, 140),
    "olive": (120, 124, 20),
}
BAR_NAMES = tuple(sorted(BAR_COLORS))
MARKER_NAMES = tuple(sorted(MARKER_COLORS))


# ---------------------------------------------------------------- geometry ---
def _dihedral(x: int, y: int, d: int, side: int) -> tuple[int, int]:
    """Map a local pile pixel through one of the eight symmetries of a square."""
    if d >= 4:
        x = side - 1 - x
    for _ in range(d % 4):
        x, y = side - 1 - y, x
    return x, y


def _map_rect(rect: tuple[int, int, int, int], d: int, side: int) -> tuple[int, int, int, int]:
    x0, y0, x1, y1 = rect
    a = _dihedral(x0, y0, d, side)
    b = _dihedral(x1, y1, d, side)
    return (min(a[0], b[0]), min(a[1], b[1]), max(a[0], b[0]), max(a[1], b[1]))


def _pile_rects(pile: dict) -> dict:
    """Local (pre-transform) rectangles of the three bars, keyed by colour name."""
    spine_t = pile["spine_thickness"]
    cy = pile["spine_center"]
    rects = {
        pile["spine"]: (0, cy - spine_t // 2, PILE - 1, cy - spine_t // 2 + spine_t - 1)
    }
    for name, cx, thick in zip(pile["cross"], pile["cross_centers"], pile["cross_thickness"]):
        rects[name] = (cx - thick // 2, 0, cx - thick // 2 + thick - 1, PILE - 1)
    return rects


def _stack_order(pile: dict) -> list[str]:
    """Bottom-to-top drawing order consistent with the pile's two over/under bits."""
    spine, (c0, c1) = pile["spine"], pile["cross"]
    b0, b1 = pile["bits"]  # 1 == spine lies on top of that cross bar
    below = [c for c, bit in ((c0, b0), (c1, b1)) if bit == 1]
    above = [c for c, bit in ((c0, b0), (c1, b1)) if bit == 0]
    return below + [spine] + above


def _tray_origins(slab_x: int, slab_y: int) -> list[tuple[int, int]]:
    step = TRAY + GAP
    return [
        (slab_x + PAD + (i % 2) * step, slab_y + PAD + (i // 2) * step) for i in range(4)
    ]


# ------------------------------------------------------------------ sampling ---
def _sample_pile(rng: random.Random, spine: str, cross: tuple[str, str], bits: tuple[int, int]) -> dict:
    thick = [rng.randint(14, 26) for _ in range(3)]
    if rng.random() < 0.18:
        thick[rng.randrange(3)] = rng.randint(7, 14)
    return {
        "spine": spine,
        "cross": list(cross),
        "bits": list(bits),
        "orientation": rng.randrange(8),
        "spine_center": rng.randint(46, 66),
        "spine_thickness": thick[0],
        "cross_centers": [rng.randint(24, 34), rng.randint(78, 88)],
        "cross_thickness": thick[1:],
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x5CA1AB1E ^ (seed * 2654435761 % (1 << 32)))

    palette = rng.sample(BAR_NAMES, 3)
    spine = palette[0]
    cross = (palette[1], palette[2])

    key_bits = (rng.randrange(2), rng.randrange(2))
    others = [b for b in ((0, 0), (0, 1), (1, 0), (1, 1)) if b != key_bits]
    rng.shuffle(others)

    key_slot = rng.randrange(4)
    cand_slots = [s for s in range(4) if s != key_slot]
    match_index = rng.randrange(3)

    target = MARKER_NAMES[seed % len(MARKER_NAMES)]
    spare = [m for m in MARKER_NAMES if m != target]
    rng.shuffle(spare)
    markers = list(spare[:2])
    markers.insert(match_index, target)

    piles = {}
    piles[key_slot] = _sample_pile(rng, spine, cross, key_bits)
    for i, slot in enumerate(cand_slots):
        bits = key_bits if i == match_index else others[i if i < match_index else i - 1]
        piles[slot] = _sample_pile(rng, spine, cross, bits)

    slab_x = rng.randint(40, CANVAS - SLAB - 40)
    slab_y = rng.randint(40, CANVAS - SLAB - 40)

    decoration = []
    taken = []
    for _ in range(120):
        if len(decoration) >= 26:
            break
        side = rng.choice((10, 13, 16))
        x = rng.randint(8, CANVAS - side - 8)
        y = rng.randint(8, CANVAS - side - 8)
        box = (x - 6, y - 6, x + side + 6, y + side + 6)
        if not (box[2] < slab_x or box[0] > slab_x + SLAB or box[3] < slab_y or box[1] > slab_y + SLAB):
            continue
        if any(not (box[2] < t[0] or box[0] > t[2] or box[3] < t[1] or box[1] > t[3]) for t in taken):
            continue
        taken.append(box)
        decoration.append(
            {"x": x, "y": y, "side": side, "color": rng.choice(BAR_NAMES + MARKER_NAMES)}
        )

    return {
        "slab": [slab_x, slab_y],
        "key_slot": key_slot,
        "cand_slots": cand_slots,
        "markers": markers,
        "spine": spine,
        "cross": list(cross),
        "piles": [piles[i] for i in range(4)],
        "decoration": decoration,
        "slot_order": list(range(4)),
        "decoration_order": list(range(len(decoration))),
    }


# ------------------------------------------------------------------- render ---
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), PAGE)
    draw = ImageDraw.Draw(image)

    for index in sorted(scene["decoration_order"]):
        deco = scene["decoration"][index]
        color = {**BAR_COLORS, **MARKER_COLORS}[deco["color"]]
        draw.rectangle(
            [deco["x"], deco["y"], deco["x"] + deco["side"] - 1, deco["y"] + deco["side"] - 1],
            fill=color,
        )

    slab_x, slab_y = scene["slab"]
    draw.rectangle([slab_x, slab_y, slab_x + SLAB - 1, slab_y + SLAB - 1], fill=SLAB_COLOR)

    origins = _tray_origins(slab_x, slab_y)
    for slot in sorted(scene["slot_order"]):
        ox, oy = origins[slot]
        is_key = slot == scene["key_slot"]
        draw.rectangle([ox, oy, ox + TRAY - 1, oy + TRAY - 1], fill=KEY_TRAY if is_key else CAND_TRAY)

        pile = scene["piles"][slot]
        px, py = ox + (TRAY - PILE) // 2, oy + (TRAY - PILE) // 2
        rects = _pile_rects(pile)
        for name in _stack_order(pile):
            x0, y0, x1, y1 = _map_rect(rects[name], pile["orientation"], PILE)
            draw.rectangle([px + x0, py + y0, px + x1, py + y1], fill=BAR_COLORS[name])

        if not is_key:
            marker = scene["markers"][scene["cand_slots"].index(slot)]
            draw.rectangle([ox + 1, oy + 1, ox + 8, oy + 8], fill=MARKER_COLORS[marker])

    return image


# ---------------------------------------------------------------- analytics ---
def _key_bits(scene: dict) -> tuple[int, int]:
    return tuple(scene["piles"][scene["key_slot"]]["bits"])


def _matching_slots(scene: dict) -> list[int]:
    key = _key_bits(scene)
    return [s for s in scene["cand_slots"] if tuple(scene["piles"][s]["bits"]) == key]


def analytic_gold(scene: dict) -> str:
    matches = _matching_slots(scene)
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one matching candidate pile")
    return scene["markers"][scene["cand_slots"].index(matches[0])]


def margin(scene: dict) -> float:
    """Smallest visible over/under patch side anywhere in the decisive slab."""
    smallest = float(PILE)
    for pile in scene["piles"]:
        for cross_t in pile["cross_thickness"]:
            smallest = min(smallest, float(min(pile["spine_thickness"], cross_t)))
    return smallest


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    import copy

    reversed_deco = copy.deepcopy(scene)
    reversed_deco["decoration_order"] = list(reversed(scene["decoration_order"]))

    reversed_slots = copy.deepcopy(scene)
    reversed_slots["slot_order"] = list(reversed(scene["slot_order"]))

    return [
        ("decoration_draw_order_reversed", reversed_deco),
        ("tray_draw_order_reversed", reversed_slots),
    ]
