"""Independent pixel-only inverse for the state-change parity world.

The parser reads only the final RGB image. It discovers the state-color mapping
from the visible key, identifies cards from their fixed bitmap glyphs, and
computes adjacent state changes. It deliberately shares no renderer, prompt,
generator, verifier, or analytic-gold imports.
"""

from __future__ import annotations

from PIL import Image


CANVAS = (1940, 820)
PANEL_COUNT = 7
IDENTITY_COUNT = 4
STATE_COUNT = 3
LABELS = ("A", "B", "C", "D")
INK = (31, 48, 61)
PANEL_X0 = 24
PANEL_TOP = 222
PANEL_WIDTH = 260
PANEL_GAP = 8
SLOT_ORIGINS = ((14, 88), (134, 88), (14, 332), (134, 332))
LABEL_X_OFFSET = 14
LABEL_Y_OFFSET = 44
LABEL_SCALE = 3
STATE_X_OFFSET = 76
STATE_Y_OFFSET = 43
STATE_BOX_SIZE = 26
KEY_XS = (470, 720, 970)
KEY_Y = 112

LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)


def _label_points(label_index: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * LABEL_SCALE + dx, y * LABEL_SCALE + dy)
        for y, row in enumerate(LABEL_PATTERNS[label_index])
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(LABEL_SCALE)
        for dy in range(LABEL_SCALE)
    )


LABEL_POINTS = tuple(_label_points(index) for index in range(IDENTITY_COUNT))


def _decode_key(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    mapping: dict[tuple[int, int, int], int] = {}
    for state, key_x in enumerate(KEY_XS):
        color = frame.getpixel((key_x + STATE_BOX_SIZE // 2, KEY_Y + STATE_BOX_SIZE // 2))
        if color in mapping:
            raise ValueError("state key repeats a palette color")
        mapping[color] = state
    if len(mapping) != STATE_COUNT:
        raise ValueError("state key is incomplete")
    return mapping


def _read_label(frame: Image.Image, origin: tuple[int, int]) -> int:
    ox, oy = origin
    observed = frozenset(
        (x, y)
        for x in range(15)
        for y in range(21)
        if frame.getpixel((ox + x, oy + y)) == INK
    )
    matches = [label for label, points in enumerate(LABEL_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("identity glyph is missing, altered, or ambiguous")
    return matches[0]


def decision_from_image(image: Image.Image) -> str:
    """Return the unique label with an odd number of adjacent state changes."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_key = _decode_key(frame)
    state_tables: list[dict[int, int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        states: dict[int, int] = {}
        for slot_x, slot_y in SLOT_ORIGINS:
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            label = _read_label(frame, (card_x + LABEL_X_OFFSET, card_y + LABEL_Y_OFFSET))
            if label in states:
                raise ValueError("a panel repeats one identity")
            color = frame.getpixel(
                (
                    card_x + STATE_X_OFFSET + STATE_BOX_SIZE // 2,
                    card_y + STATE_Y_OFFSET + STATE_BOX_SIZE // 2,
                )
            )
            if color not in state_key:
                raise ValueError("a keyed state swatch is missing or altered")
            states[label] = state_key[color]
        if set(states) != set(range(IDENTITY_COUNT)):
            raise ValueError("a panel does not contain all four identities")
        state_tables.append(states)

    change_counts = [
        sum(state_tables[panel][label] != state_tables[panel - 1][label] for panel in range(1, PANEL_COUNT))
        for label in range(IDENTITY_COUNT)
    ]
    winners = [label for label, count in enumerate(change_counts) if count % 2 == 1]
    if len(winners) != 1:
        raise ValueError("image has zero or multiple odd-parity labels")
    return LABELS[winners[0]]
