"""Independent inverse arm: infer hole ownership from final RGB pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

DARK_LUMA_MAX = 115.0
MIN_ISLAND_PIXELS = 2500
MIN_HOLE_PIXELS = 250


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return four-connected components as arrays of (row, column) pixels."""

    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=np.bool_)
    found: list[np.ndarray] = []
    for row in range(height):
        for col in range(width):
            if not mask[row, col] or seen[row, col]:
                continue
            queue = deque([(row, col)])
            seen[row, col] = True
            pixels = []
            while queue:
                rr, cc = queue.popleft()
                pixels.append((rr, cc))
                if rr and mask[rr - 1, cc] and not seen[rr - 1, cc]:
                    seen[rr - 1, cc] = True
                    queue.append((rr - 1, cc))
                if rr + 1 < height and mask[rr + 1, cc] and not seen[rr + 1, cc]:
                    seen[rr + 1, cc] = True
                    queue.append((rr + 1, cc))
                if cc and mask[rr, cc - 1] and not seen[rr, cc - 1]:
                    seen[rr, cc - 1] = True
                    queue.append((rr, cc - 1))
                if cc + 1 < width and mask[rr, cc + 1] and not seen[rr, cc + 1]:
                    seen[rr, cc + 1] = True
                    queue.append((rr, cc + 1))
            found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _dark_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    luma = 0.2126 * array[..., 0] + 0.7152 * array[..., 1] + 0.0722 * array[..., 2]
    return luma < DARK_LUMA_MAX


def _enclosed_light_count(dark: np.ndarray, component: np.ndarray) -> int:
    rows = component[:, 0]
    cols = component[:, 1]
    top, bottom = int(rows.min()), int(rows.max())
    left, right = int(cols.min()), int(cols.max())
    local_light = ~dark[top : bottom + 1, left : right + 1]
    holes = 0
    height, width = local_light.shape
    for pixels in _components(local_light):
        rr = pixels[:, 0]
        cc = pixels[:, 1]
        touches_box = bool(
            (rr == 0).any()
            or (rr == height - 1).any()
            or (cc == 0).any()
            or (cc == width - 1).any()
        )
        if not touches_box and len(pixels) >= MIN_HOLE_PIXELS:
            holes += 1
    return holes


def decision_from_image(image: Image.Image) -> str:
    """Return yes/no solely from component and enclosure relations in the PNG."""

    dark = _dark_mask(image)
    islands = [part for part in _components(dark) if len(part) >= MIN_ISLAND_PIXELS]
    if len(islands) != 3:
        raise ValueError(f"expected three visible dark islands, recovered {len(islands)}")
    counts = [_enclosed_light_count(dark, island) for island in islands]
    if sum(counts) != 3:
        raise ValueError(f"expected three visible enclosed holes, recovered {sum(counts)}")
    return "yes" if all(count == 1 for count in counts) else "no"
