"""Deterministic latent sampler and renderer for per-component hole ownership."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 680
HEIGHT = 340
SUPERSAMPLE = 4
QUARANTINE_CLEARANCE = 9.0

_BACKGROUNDS = (
    (248, 246, 239),
    (242, 247, 248),
    (247, 243, 248),
)
_SHADOWS = (
    (205, 211, 214),
    (210, 207, 214),
    (207, 213, 207),
)
_INKS = (
    (28, 39, 52),
    (52, 31, 49),
    (31, 51, 45),
)


def sample_scene(seed: int) -> dict:
    """Sample balanced geometry; seed parity alternates the two decisions."""

    rng = random.Random(int(seed))
    positive = int(seed) % 2 == 0
    counts = [1, 1, 1] if positive else [0, 1, 2]
    if not positive:
        rng.shuffle(counts)

    outer_w = rng.choice((164, 168, 172))
    outer_h = rng.choice((150, 154, 158))
    hole_w = rng.choice((36, 38, 40))
    hole_h = rng.choice((54, 58, 62))
    clearance = rng.choice((12, 14, 16, 18, 20, 22))
    base_x = 42 + rng.randrange(-3, 4)
    step = 216
    base_y = 88 + rng.randrange(-7, 8)

    inks = list(_INKS)
    rng.shuffle(inks)
    islands = []
    for index, hole_count in enumerate(counts):
        x0 = base_x + index * step + rng.randrange(-3, 4)
        y0 = base_y + rng.randrange(-8, 9)
        x1 = x0 + outer_w
        y1 = y0 + outer_h
        left_x = x0 + clearance
        right_x = x1 - clearance - hole_w
        hole_y = y0 + (outer_h - hole_h) // 2
        slots = [left_x, right_x]
        if hole_count == 1:
            chosen = [slots[rng.randrange(2)]]
        elif hole_count == 2:
            chosen = slots
        else:
            chosen = []
        holes = [[hx, hole_y, hx + hole_w, hole_y + hole_h] for hx in chosen]
        islands.append(
            {
                "outer": [x0, y0, x1, y1],
                "holes": holes,
                "ink": list(inks[index]),
                "radius": 20,
                "hole_radius": 8,
            }
        )

    return {
        "canvas": [WIDTH, HEIGHT],
        "background": list(rng.choice(_BACKGROUNDS)),
        "shadow": list(rng.choice(_SHADOWS)),
        "islands": islands,
    }


def _scaled_box(box: list[int], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(value) * scale for value in box)


def render(scene: dict) -> Image.Image:
    """Render only observable scene fields, with robust antialiased boundaries."""

    width, height = (int(value) for value in scene["canvas"])
    scale = SUPERSAMPLE
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    shadow = tuple(int(value) for value in scene["shadow"])
    for island in scene["islands"]:
        outer = list(island["outer"])
        shadow_box = [outer[0] + 5, outer[1] + 7, outer[2] + 5, outer[3] + 7]
        draw.rounded_rectangle(
            _scaled_box(shadow_box, scale),
            radius=int(island["radius"]) * scale,
            fill=shadow,
        )
    for island in scene["islands"]:
        draw.rounded_rectangle(
            _scaled_box(island["outer"], scale),
            radius=int(island["radius"]) * scale,
            fill=tuple(int(value) for value in island["ink"]),
        )
        for hole in island["holes"]:
            draw.rounded_rectangle(
                _scaled_box(hole, scale),
                radius=int(island["hole_radius"]) * scale,
                fill=background,
            )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def _hole_counts(scene: dict) -> list[int]:
    return [len(island["holes"]) for island in scene["islands"]]


def analytic_gold(scene: dict) -> str:
    return "yes" if all(count == 1 for count in _hole_counts(scene)) else "no"


def margin(scene: dict) -> float:
    """Minimum visible clearance to a hole-opening or component-merger event."""

    clearances: list[float] = []
    ordered = sorted(scene["islands"], key=lambda item: item["outer"][0])
    for left, right in zip(ordered, ordered[1:]):
        clearances.append(float(right["outer"][0] - left["outer"][2]))
    for island in scene["islands"]:
        x0, y0, x1, y1 = island["outer"]
        holes = sorted(island["holes"], key=lambda box: box[0])
        for hx0, hy0, hx1, hy1 in holes:
            clearances.extend(
                [float(hx0 - x0), float(x1 - hx1), float(hy0 - y0), float(y1 - hy1)]
            )
        for left, right in zip(holes, holes[1:]):
            clearances.append(float(right[0] - left[2]))
    return min(clearances) if clearances else 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["islands"] = list(reversed(twin["islands"]))
    return [("reverse_disjoint_draw_records", twin)]
