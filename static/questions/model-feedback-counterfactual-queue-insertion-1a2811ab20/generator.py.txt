"""Renderer for a counterfactual ordered-queue insertion question.

The displayed rows are the pre-action state.  A marked source card is moved to
an explicitly drawn gap in the other row; the answer is read from a later slot
after the destination row has been reindexed.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "counterfactual_queue_insertion-0.1.0"

WIDTH = 1000
HEIGHT = 620
BACKGROUND = (255, 255, 255)
PANEL_FILL = (246, 249, 253)
PANEL_BORDER = (58, 74, 96)
INK = (35, 47, 63)
GOLD = (232, 157, 24)
CYAN = (20, 174, 196)

ANSWER_COLORS = ("red", "blue", "green", "violet")
CARD_COLORS = {
    "red": (215, 58, 66),
    "blue": (48, 103, 190),
    "green": (44, 156, 95),
    "violet": (137, 76, 190),
}

PANEL_BOXES = (
    (70.0, 105.0, 930.0, 285.0),
    (70.0, 335.0, 930.0, 515.0),
)


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene with a visible post-insertion shift."""

    rng = random.Random(
        (int(seed) * 6364136223846793005 + 1442695040888963407) & ((1 << 64) - 1)
    )
    source_row = int(rng.randrange(2))
    source_cards = [rng.choice(ANSWER_COLORS) for _ in range(7)]

    destination_cards = [rng.choice(ANSWER_COLORS) for _ in range(8)]
    query_index = 4 + int(rng.randrange(4))
    distance = 1 + int(rng.randrange(min(3, query_index)))
    insertion_index = query_index - distance

    # The answer is deliberately an existing destination card shifted into the
    # query slot.  The initially bracketed card is forced to differ, making a
    # no-action read visibly produce the wrong class.
    target_color = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    destination_cards[query_index - 1] = target_color
    target_position = ANSWER_COLORS.index(target_color)
    destination_cards[query_index] = ANSWER_COLORS[(target_position + 1) % len(ANSWER_COLORS)]

    return {
        "source_row": source_row,
        "source_cards": list(source_cards),
        "destination_cards": list(destination_cards),
        "marked_index": int(rng.randrange(len(source_cards))),
        "insertion_index": int(insertion_index),
        "query_index": int(query_index),
        "layout_variant": int(rng.randrange(3)),
        "width": WIDTH,
        "height": HEIGHT,
        "card_width": 64,
        "card_height": 86,
        "card_gap": 18 + int(rng.randrange(5)),
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    paths = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    )
    path = paths[0 if bold else 1]
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _row_y(row: int) -> float:
    box = PANEL_BOXES[int(row)]
    return (box[1] + box[3]) / 2.0 + 18.0


def _slot_centres(count: int, row: int, scene: dict) -> list[tuple[float, float]]:
    box = PANEL_BOXES[int(row)]
    card_width = float(scene["card_width"])
    gap = float(scene["card_gap"])
    total = float(count) * card_width + float(max(0, count - 1)) * gap
    start = (box[0] + box[2] - total) / 2.0 + card_width / 2.0
    base_y = _row_y(row)
    centres = []
    for index in range(count):
        y = base_y
        variant = int(scene["layout_variant"])
        if variant == 1:
            y += 4.0 if index % 2 else -3.0
        elif variant == 2:
            y += (5.0, -4.0, 2.0)[index % 3]
        centres.append((start + index * (card_width + gap), y))
    return centres


def _card_box(center: tuple[float, float], scene: dict) -> tuple[float, float, float, float]:
    cx, cy = center
    half_w = float(scene["card_width"]) / 2.0
    half_h = float(scene["card_height"]) / 2.0
    return (cx - half_w, cy - half_h, cx + half_w, cy + half_h)


def _gap_x(scene: dict, destination_row: int) -> float:
    centres = _slot_centres(len(scene["destination_cards"]), destination_row, scene)
    index = int(scene["insertion_index"])
    return (centres[index - 1][0] + centres[index][0]) / 2.0


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    source_row: int,
    destination_row: int,
    source_center: tuple[float, float],
    gap_x: float,
) -> None:
    source_y = _row_y(source_row)
    destination_y = _row_y(destination_row)
    downward = destination_row > source_row
    direction = 1.0 if downward else -1.0
    source_exit = source_y + direction * 58.0
    destination_entry = destination_y - direction * 63.0
    lane_y = (source_exit + destination_entry) / 2.0
    points = [
        source_center,
        (source_center[0], source_exit),
        (gap_x, lane_y),
        (gap_x, destination_entry),
    ]
    draw.line(points, fill=GOLD, width=8, joint="curve")

    tip_y = destination_y - direction * 53.0
    if downward:
        arrowhead = [
            (gap_x - 15.0, tip_y - 18.0),
            (gap_x + 15.0, tip_y - 18.0),
            (gap_x, tip_y + 5.0),
        ]
    else:
        arrowhead = [
            (gap_x - 15.0, tip_y + 18.0),
            (gap_x + 15.0, tip_y + 18.0),
            (gap_x, tip_y - 5.0),
        ]
    draw.polygon(arrowhead, fill=GOLD)


def _draw_marker_pill(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    label: str,
    fill: tuple[int, int, int],
    width: float,
    y_offset: float,
) -> None:
    cx, cy = center
    box = (cx - width / 2.0, cy + y_offset - 13.0, cx + width / 2.0, cy + y_offset + 13.0)
    draw.rounded_rectangle(box, radius=8, fill=fill)
    draw.text(
        (cx, cy + y_offset),
        label,
        fill=(255, 255, 255),
        anchor="mm",
        font=_font(13, bold=True),
    )


def render(scene: dict) -> Image.Image:
    """Render only the visible pre-action state and its action annotations."""

    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    source_row = int(scene["source_row"])
    destination_row = 1 - source_row
    source_cards = list(scene["source_cards"])
    destination_cards = list(scene["destination_cards"])

    draw.text(
        (width / 2.0, 40.0),
        "COUNTERFACTUAL CARD EDIT",
        fill=INK,
        anchor="mm",
        font=_font(25, bold=True),
    )
    draw.text(
        (width / 2.0, 72.0),
        "The rows show the cards before the move",
        fill=(92, 105, 122),
        anchor="mm",
        font=_font(16),
    )

    for row, box in enumerate(PANEL_BOXES):
        draw.rounded_rectangle(box, radius=24, fill=PANEL_FILL, outline=PANEL_BORDER, width=4)
        label = "SOURCE ROW" if row == source_row else "DESTINATION ROW"
        label_color = GOLD if row == source_row else CYAN
        draw.text(
            (box[0] + 26.0, box[1] + 24.0),
            label,
            fill=label_color,
            anchor="lm",
            font=_font(16, bold=True),
        )

    source_centres = _slot_centres(len(source_cards), source_row, scene)
    destination_centres = _slot_centres(len(destination_cards), destination_row, scene)
    marked_index = int(scene["marked_index"])
    insertion_index = int(scene["insertion_index"])
    query_index = int(scene["query_index"])
    source_center = source_centres[marked_index]
    gap_x = (destination_centres[insertion_index - 1][0] + destination_centres[insertion_index][0]) / 2.0
    query_center = destination_centres[query_index]

    _draw_arrow(draw, scene, source_row, destination_row, source_center, gap_x)

    # The gold ring and cyan bracket are laid down before card fills so each is
    # a clean, broad annotation outside the large colored component.
    selected_box = _card_box(source_center, scene)
    pad = 13.0
    draw.rounded_rectangle(
        (selected_box[0] - pad, selected_box[1] - pad, selected_box[2] + pad, selected_box[3] + pad),
        radius=17,
        outline=GOLD,
        width=7,
    )
    query_box = _card_box(query_center, scene)
    query_pad = 12.0
    draw.rounded_rectangle(
        (query_box[0] - query_pad, query_box[1] - query_pad, query_box[2] + query_pad, query_box[3] + query_pad),
        radius=17,
        outline=CYAN,
        width=7,
    )

    # A wide vertical marker makes the insertion location distinct from the
    # query bracket and from the arrow's route.
    destination_y = _row_y(destination_row)
    draw.line(
        [(gap_x, destination_y - 55.0), (gap_x, destination_y + 55.0)],
        fill=GOLD,
        width=7,
    )
    draw.ellipse(
        (gap_x - 10.0, destination_y - 10.0, gap_x + 10.0, destination_y + 10.0),
        fill=GOLD,
    )
    _draw_marker_pill(draw, (gap_x, destination_y), "INSERT", GOLD, 76.0, -70.0)
    _draw_marker_pill(draw, query_center, "QUERY", CYAN, 67.0, 70.0)

    for row, cards in ((source_row, source_cards), (destination_row, destination_cards)):
        centres = _slot_centres(len(cards), row, scene)
        for index, color_name in enumerate(cards):
            box = _card_box(centres[index], scene)
            draw.rounded_rectangle(
                box,
                radius=13,
                fill=CARD_COLORS[color_name],
                outline=INK,
                width=3,
            )

    return image


def _post_destination(scene: dict) -> list[str]:
    """Apply the visible deletion/insertion operation to ordered card lists."""

    source = list(scene["source_cards"])
    destination = list(scene["destination_cards"])
    moved = source.pop(int(scene["marked_index"]))
    destination.insert(int(scene["insertion_index"]), moved)
    return destination


def analytic_gold(scene: dict) -> str:
    """Return the color at the post-insertion query slot."""

    return _post_destination(scene)[int(scene["query_index"])]


def margin(scene: dict) -> float:
    """Discrete separation between the insertion gap and the query slot."""

    return float(abs(int(scene["query_index"]) - int(scene["insertion_index"])))


def is_quarantined(scene: dict) -> bool:
    """Exact gap/query alignment is excluded as the discrete boundary case."""

    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This parametrization has no intended render-invariant aliases."""

    return []
