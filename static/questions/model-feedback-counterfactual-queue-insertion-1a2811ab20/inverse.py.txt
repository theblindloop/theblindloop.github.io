"""Independent pixel inverse for the queue-insertion question.

Only the raster is accepted.  Large flat card interiors are located by their
color runs; the source is bound by the gold ring, the insertion point by the
gold vertical marker, and the queried slot by the cyan bracket.  No renderer
or scene metadata is imported.
"""

from __future__ import annotations

import numpy as np


ORACLE_VERSION = "counterfactual_queue_insertion-oracle-0.1.0"
ABSTAIN = "abstain"

_COLORS = {
    "red": np.array((215, 58, 66), dtype=np.int16),
    "blue": np.array((48, 103, 190), dtype=np.int16),
    "green": np.array((44, 156, 95), dtype=np.int16),
    "violet": np.array((137, 76, 190), dtype=np.int16),
}
_GOLD = np.array((232, 157, 24), dtype=np.int16)
_CYAN = np.array((20, 174, 196), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 14) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _x_runs(columns: np.ndarray) -> list[tuple[int, int]]:
    if columns.size == 0:
        return []
    runs: list[tuple[int, int]] = []
    start = previous = int(columns[0])
    for value in columns[1:].tolist():
        value = int(value)
        if value != previous + 1:
            runs.append((start, previous))
            start = value
        previous = value
    runs.append((start, previous))
    return runs


def _y_runs(rows: np.ndarray) -> list[tuple[int, int]]:
    return _x_runs(rows)


def _card_components(array: np.ndarray) -> list[dict]:
    """Find large colored card interiors using independent column runs."""

    components: list[dict] = []
    for name, color in _COLORS.items():
        mask = _near(array, color)
        rows = np.flatnonzero(mask.any(axis=1))
        for y_start, y_end in _y_runs(rows):
            row_mask = mask[y_start : y_end + 1]
            columns = np.flatnonzero(row_mask.any(axis=0))
            for x_start, x_end in _x_runs(columns):
                crop = row_mask[:, x_start : x_end + 1]
                ys, xs = np.nonzero(crop)
                if len(ys) < 1800 or int(ys.max()) - int(ys.min()) < 48:
                    continue
                components.append(
                    {
                        "color": name,
                        "area": int(len(ys)),
                        "x0": int(x_start),
                        "x1": int(x_end),
                        "y0": int(y_start + ys.min()),
                        "y1": int(y_start + ys.max()),
                        "cx": (float(x_start) + float(x_end)) / 2.0,
                        "cy": (float(y_start + ys.min()) + float(y_start + ys.max())) / 2.0,
                    }
                )
    return components


def _ring_score(gold: np.ndarray, component: dict) -> int:
    x0, x1 = component["x0"], component["x1"]
    y0, y1 = component["y0"], component["y1"]
    height, width = gold.shape
    outer = gold[max(0, y0 - 20) : min(height, y1 + 21), max(0, x0 - 20) : min(width, x1 + 21)]
    inner = gold[max(0, y0 - 5) : min(height, y1 + 6), max(0, x0 - 5) : min(width, x1 + 6)]
    return int(outer.sum()) - int(inner.sum())


def _marker_x(gold: np.ndarray, destination_y: float) -> float | None:
    y0 = max(0, int(round(destination_y - 58.0)))
    y1 = min(gold.shape[0], int(round(destination_y + 59.0)))
    counts = gold[y0:y1].sum(axis=0)
    if counts.size == 0:
        return None
    peak = int(counts.max())
    if peak < 45:
        return None
    return float(int(np.flatnonzero(counts == peak)[0]))


def _query_x(cyan: np.ndarray, destination_y: float) -> float | None:
    y0 = max(0, int(round(destination_y - 70.0)))
    y1 = min(cyan.shape[0], int(round(destination_y + 78.0)))
    ys, xs = np.nonzero(cyan[y0:y1])
    if len(xs) < 80:
        return None
    return (float(xs.min()) + float(xs.max())) / 2.0


def decision_from_image(image) -> str:
    """Recover the post-insertion queried card color from one RGB image."""

    try:
        array = np.asarray(image.convert("RGB"))
    except Exception:
        return ABSTAIN
    if array.ndim != 3 or array.shape[2] != 3 or array.shape[0] < 200 or array.shape[1] < 400:
        return ABSTAIN

    cards = _card_components(array)
    if len(cards) < 12:
        return ABSTAIN
    gold = _near(array, _GOLD)
    ring_scores = [( _ring_score(gold, card), index) for index, card in enumerate(cards)]
    ring_scores.sort(reverse=True)
    if not ring_scores or ring_scores[0][0] < 700:
        return ABSTAIN
    if len(ring_scores) > 1 and ring_scores[0][0] == ring_scores[1][0]:
        return ABSTAIN
    source = cards[ring_scores[0][1]]

    source_row = [card for card in cards if abs(card["cy"] - source["cy"]) < 35.0]
    destination_row = [card for card in cards if abs(card["cy"] - source["cy"]) >= 100.0]
    if len(source_row) < 5 or len(destination_row) < 6:
        return ABSTAIN
    destination_row.sort(key=lambda card: card["cx"])

    gap_x = _marker_x(gold, sum(card["cy"] for card in destination_row) / len(destination_row))
    destination_y = sum(card["cy"] for card in destination_row) / len(destination_row)
    query_x = _query_x(_near(array, _CYAN), destination_y)
    if gap_x is None or query_x is None:
        return ABSTAIN

    query_index = min(
        range(len(destination_row)),
        key=lambda index: abs(destination_row[index]["cx"] - query_x),
    )
    insertion_index = sum(card["cx"] < gap_x for card in destination_row)
    if insertion_index < 0 or insertion_index > len(destination_row):
        return ABSTAIN

    if insertion_index < query_index:
        return destination_row[query_index - 1]["color"]
    if insertion_index == query_index:
        return source["color"]
    return destination_row[query_index]["color"]
