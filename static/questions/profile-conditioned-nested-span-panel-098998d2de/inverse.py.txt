"""Image-only oracle for the nested-span panel world.

The oracle receives a rendered RGB image and nothing else.  It recovers the
panel columns from the grey frames, recovers each bar's horizontal stretch from
the coloured pixel rows, and names the single panel whose bars never cross.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOUR_SATURATION = 45
GREY_TOLERANCE = 12
GREY_RANGE = (70, 165)
FRAME_COLUMN_FRACTION = 0.3
PIXEL_TOLERANCE = 6.0
ABSTAIN = "abstain"


def _runs(flags):
    runs = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            runs.append((start, index))
            start = None
    if start is not None:
        runs.append((start, len(flags)))
    return runs


def _pair_state(first, second):
    low = max(first[0], second[0])
    high = min(first[1], second[1])
    if high <= low:
        return "separate", float(low - high)
    if first[0] <= second[0] and second[1] <= first[1]:
        return "nested", float(min(second[0] - first[0], first[1] - second[1]))
    if second[0] <= first[0] and first[1] <= second[1]:
        return "nested", float(min(first[0] - second[0], second[1] - first[1]))
    left, right = (first, second) if first[0] <= second[0] else (second, first)
    return "cross", float(min(left[1] - right[0], right[0] - left[0], right[1] - left[1]))


def decision_from_image(image) -> str:
    if not isinstance(image, Image.Image):
        return ABSTAIN
    array = np.asarray(image.convert("RGB")).astype(int)
    height, width = array.shape[0], array.shape[1]
    high = array.max(axis=2)
    low = array.min(axis=2)
    colour = (high - low) > COLOUR_SATURATION
    grey = (
        ((high - low) <= GREY_TOLERANCE)
        & (array[:, :, 0] >= GREY_RANGE[0])
        & (array[:, :, 0] <= GREY_RANGE[1])
    )
    frame_columns = grey.sum(axis=0) > FRAME_COLUMN_FRACTION * height
    borders = _runs(frame_columns.tolist())
    if len(borders) % 2 or len(borders) < 4:
        return ABSTAIN
    panels = [(borders[i][1], borders[i + 1][0]) for i in range(0, len(borders), 2)]
    if not panels:
        return ABSTAIN

    winners = []
    for index, (x_low, x_high) in enumerate(panels):
        if x_high - x_low < 8:
            return ABSTAIN
        block = colour[:, x_low:x_high]
        row_runs = _runs(block.any(axis=1).tolist())
        spans = []
        for top, bottom in row_runs:
            columns = np.nonzero(block[top:bottom].any(axis=0))[0]
            if columns.size == 0:
                return ABSTAIN
            spans.append((float(x_low + columns[0]), float(x_low + columns[-1] + 1)))
        if len(spans) < 2:
            return ABSTAIN
        crossing = False
        for i in range(len(spans)):
            for j in range(i + 1, len(spans)):
                state, value = _pair_state(spans[i], spans[j])
                if value < PIXEL_TOLERANCE:
                    return ABSTAIN
                if state == "cross":
                    crossing = True
        if not crossing:
            winners.append(index)
    if len(winners) != 1:
        return ABSTAIN
    return f"panel-{winners[0] + 1}"
