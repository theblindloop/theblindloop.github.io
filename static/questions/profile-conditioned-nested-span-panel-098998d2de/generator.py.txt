"""Nested-span panel world.

Five framed panels each carry five horizontal bars.  Read every bar as the
horizontal stretch it covers.  Two bars are *separate* when their stretches do
not touch, *nested* when one stretch lies entirely inside the other, and
*crossing* when the stretches overlap while neither lies inside the other.
Exactly one panel is free of crossing pairs; the decision names that panel.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

UNIT = 8
AXIS_UNITS = 24
PANEL_W = 210
PANEL_H = 190
PANEL_GAP = 18
MARGIN_X = 20
MARGIN_TOP = 20
INNER_INSET = 9
SLOT_TOP = 22
SLOT_STEP = 30
BAR_H = 14
N_PANELS = 5
N_BARS = 5
TOL_UNITS = 2.0
INK_BAND_UNITS = 4

WIDTH = 2 * MARGIN_X + N_PANELS * PANEL_W + (N_PANELS - 1) * PANEL_GAP
HEIGHT = MARGIN_TOP + PANEL_H + 30

BACKGROUND = (255, 255, 255)
FRAME = (110, 110, 110)
LABEL = (0, 0, 0)
PALETTE = [
    [198, 40, 40],
    [21, 101, 192],
    [46, 125, 50],
    [230, 120, 0],
    [106, 27, 154],
    [0, 131, 143],
    [146, 64, 14],
    [60, 60, 160],
]


# --------------------------------------------------------------------------
# geometry of the decision
# --------------------------------------------------------------------------
def _pair_state(first, second):
    """Return (state, robustness) for two spans, in axis units."""
    low = max(first[0], second[0])
    high = min(first[1], second[1])
    if high <= low:
        return "separate", float(low - high)
    if first[0] <= second[0] and second[1] <= first[1]:
        return "nested", float(min(second[0] - first[0], first[1] - second[1]))
    if second[0] <= first[0] and first[1] <= second[1]:
        return "nested", float(min(first[0] - second[0], second[1] - first[1]))
    left, right = (first, second) if first[0] <= second[0] else (second, first)
    overlap = left[1] - right[0]
    return "cross", float(min(overlap, right[0] - left[0], right[1] - left[1]))


def _panel_report(spans):
    crossings = []
    clean = []
    for i in range(len(spans)):
        for j in range(i + 1, len(spans)):
            state, value = _pair_state(spans[i], spans[j])
            (crossings if state == "cross" else clean).append(value)
    return crossings, clean


def _analysis(panels):
    """Winners and the decision margin, reading only the panel records."""
    winners = []
    values = []
    for index, panel in enumerate(panels):
        spans = [(bar["x0"], bar["x1"]) for bar in panel["bars"]]
        crossings, clean = _panel_report(spans)
        clearance = min(clean) if clean else 0.0
        if crossings:
            values.append(min(min(crossings), clearance))
        else:
            winners.append(index)
            values.append(clearance)
    return {"winners": winners, "margin_units": float(min(values)) if values else 0.0}


def analytic_gold(scene) -> str:
    report = _analysis(scene["panels"])
    if len(report["winners"]) != 1:
        return "abstain"
    return f"panel-{report['winners'][0] + 1}"


def margin(scene) -> float:
    return float(_analysis(scene["panels"])["margin_units"]) * UNIT


def is_quarantined(scene) -> bool:
    report = _analysis(scene["panels"])
    if len(report["winners"]) != 1:
        return True
    return report["margin_units"] < TOL_UNITS


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _composition(rng, total, parts):
    cuts = sorted(rng.sample(range(1, total), parts - 1)) if parts > 1 else []
    sizes = []
    previous = 0
    for cut in cuts + [total]:
        sizes.append(cut - previous)
        previous = cut
    return sizes


def _place_family(rng, low, high, sizes):
    """Place sibling subtrees inside [low, high] with two-unit gaps."""
    need = sum(5 * size - 2 for size in sizes) + 2 * (len(sizes) - 1)
    slack = (high - low) - need
    if slack < 0:
        return None
    buckets = len(sizes) + max(0, len(sizes) - 1) + 2
    extra = [0] * buckets
    for _ in range(slack):
        extra[rng.randrange(buckets)] += 1
    spans = []
    cursor = low + extra[-2]
    for index, size in enumerate(sizes):
        width = 5 * size - 2 + extra[index]
        span = (cursor, cursor + width)
        spans.append(span)
        if size > 1:
            parts = rng.randint(1, size - 1)
            inner = _place_family(rng, span[0] + 2, span[1] - 2, _composition(rng, size - 1, parts))
            if inner is None:
                return None
            spans.extend(inner)
        cursor = span[1] + 2
        if index < len(sizes) - 1:
            cursor += extra[len(sizes) + index]
    return spans


def _laminar(rng):
    for _ in range(80):
        parts = rng.randint(1, N_BARS)
        spans = _place_family(rng, 0, AXIS_UNITS, _composition(rng, N_BARS, parts))
        if spans is None or len(spans) != N_BARS:
            continue
        crossings, clean = _panel_report(spans)
        if crossings or (clean and min(clean) < TOL_UNITS):
            continue
        if all(0 <= a and b <= AXIS_UNITS and b - a >= 3 for a, b in spans):
            rng.shuffle(spans)
            return spans
    return None


def _valid_dirty(spans):
    if not all(0 <= a and b <= AXIS_UNITS and b - a >= 3 for a, b in spans):
        return False
    crossings, clean = _panel_report(spans)
    if not 1 <= len(crossings) <= 2:
        return False
    if min(crossings) < TOL_UNITS:
        return False
    return not clean or min(clean) >= TOL_UNITS


def _perturb(rng, base):
    """Push a single endpoint of a laminar panel until one pair crosses."""
    for _ in range(80):
        spans = list(base)
        i, j = rng.sample(range(len(spans)), 2)
        first, second = spans[i], spans[j]
        state, _value = _pair_state(first, second)
        if state == "separate":
            left, right = (first, second) if first[0] <= second[0] else (second, first)
            index = spans.index(left)
            span_room = right[1] - right[0]
            if span_room < 4:
                continue
            new_end = right[0] + rng.randint(2, span_room - 2)
            spans[index] = (left[0], new_end)
        elif state == "nested":
            outer, inner = (first, second) if first[1] - first[0] >= second[1] - second[0] else (second, first)
            if rng.random() < 0.5:
                room = AXIS_UNITS - outer[1]
                if room < 2:
                    continue
                spans[spans.index(inner)] = (inner[0], outer[1] + rng.randint(2, min(room, 5)))
            else:
                if inner[1] - inner[0] < 4:
                    continue
                spans[spans.index(outer)] = (outer[0], rng.randint(inner[0] + 2, inner[1] - 2))
        else:
            continue
        if _valid_dirty(spans):
            rng.shuffle(spans)
            return spans
    return None


def _total(spans):
    return sum(b - a for a, b in spans)


def _panel_record(rng, spans):
    colours = rng.sample(PALETTE, N_BARS)
    slots = list(range(N_BARS))
    rng.shuffle(slots)
    bars = [
        {"slot": slots[index], "x0": span[0], "x1": span[1], "color": list(colours[index])}
        for index, span in enumerate(spans)
    ]
    rng.shuffle(bars)
    return {"bars": bars}


def _try_scene(rng, seed):
    winner = _laminar(rng)
    if winner is None:
        return None
    target_total = _total(winner)
    distractors = []
    for _ in range(N_PANELS - 1):
        chosen = None
        for _attempt in range(40):
            base = _laminar(rng)
            if base is None:
                continue
            spans = _perturb(rng, base)
            if spans is None:
                continue
            if abs(_total(spans) - target_total) <= INK_BAND_UNITS:
                chosen = spans
                break
        if chosen is None:
            return None
        distractors.append(chosen)
    totals = [_total(spans) for spans in distractors]
    if not (any(value > target_total for value in totals) and any(value < target_total for value in totals)):
        return None
    target = rng.randrange(N_PANELS)
    ordered = list(distractors)
    ordered.insert(target, winner)
    panels = [_panel_record(rng, spans) for spans in ordered]
    scene = {
        "seed": int(seed),
        "width": WIDTH,
        "height": HEIGHT,
        "unit": UNIT,
        "target_index": target,
        "panels": panels,
    }
    if is_quarantined(scene) or analytic_gold(scene) != f"panel-{target + 1}":
        return None
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random(1_000_003 * int(seed) + 7)
    for _ in range(200):
        scene = _try_scene(rng, seed)
        if scene is not None:
            return scene
    raise RuntimeError(f"could not sample a balanced scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    unit = scene["unit"]
    for index, panel in enumerate(scene["panels"]):
        left = MARGIN_X + index * (PANEL_W + PANEL_GAP)
        top = MARGIN_TOP
        draw.rectangle([left, top, left + PANEL_W, top + PANEL_H], outline=FRAME, width=2)
        for bar in panel["bars"]:
            y0 = top + SLOT_TOP + bar["slot"] * SLOT_STEP
            x0 = left + INNER_INSET + bar["x0"] * unit
            x1 = left + INNER_INSET + bar["x1"] * unit
            draw.rectangle([x0, y0, x1 - 1, y0 + BAR_H - 1], fill=tuple(bar["color"]))
        draw.text((left + PANEL_W // 2 - 22, top + PANEL_H + 8), f"panel-{index + 1}", fill=LABEL)
    return image


def latent_symmetries(scene):
    """Bar-list order is latent only: rendering keys off each bar's own slot."""
    rotated = {key: value for key, value in scene.items()}
    rotated["panels"] = [
        {"bars": panel["bars"][1:] + panel["bars"][:1]} for panel in scene["panels"]
    ]
    reversed_bars = {key: value for key, value in scene.items()}
    reversed_bars["panels"] = [
        {"bars": list(reversed(panel["bars"]))} for panel in scene["panels"]
    ]
    return [("bar_list_rotation", rotated), ("bar_list_reversal", reversed_bars)]
