"""Pixel-only inverse arm for the pipe-path connectivity search world.

The oracle receives a rasterized image and nothing else.  It recovers the row
of framed panels from the mid-tone frame ink, isolates the dark pipe ink and
the saturated warm terminal dots inside each panel, labels the pipe-plus-dot
ink into connected components, and reports the single panel whose two terminal
dots fall in the same component.  It abstains when the layout, the terminal
count, or the winner is not uniquely recovered.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

N_PANELS = 5
PANEL_INSET = 4
FRAME_MAX_LO = 130
FRAME_MAX_HI = 210
FRAME_SATURATION = 30
DARK_MAX = 140
PORT_MIN_RED = 150
PORT_CHANNEL_GAP = 60
MIN_FRAME_PIXELS = 400
MIN_PORT_PIXELS = 40


def _label(mask: np.ndarray) -> tuple:
    """Four-connected component labelling of a boolean mask."""
    height, width = mask.shape
    labels = np.zeros(mask.shape, dtype=np.int32)
    flat_mask = mask.reshape(-1)
    flat_labels = labels.reshape(-1)
    current = 0
    for start in np.flatnonzero(flat_mask):
        if flat_labels[start]:
            continue
        current += 1
        flat_labels[start] = current
        queue = deque([int(start)])
        while queue:
            index = queue.popleft()
            row, col = divmod(index, width)
            if col > 0:
                nxt = index - 1
                if flat_mask[nxt] and not flat_labels[nxt]:
                    flat_labels[nxt] = current
                    queue.append(nxt)
            if col + 1 < width:
                nxt = index + 1
                if flat_mask[nxt] and not flat_labels[nxt]:
                    flat_labels[nxt] = current
                    queue.append(nxt)
            if row > 0:
                nxt = index - width
                if flat_mask[nxt] and not flat_labels[nxt]:
                    flat_labels[nxt] = current
                    queue.append(nxt)
            if row + 1 < height:
                nxt = index + width
                if flat_mask[nxt] and not flat_labels[nxt]:
                    flat_labels[nxt] = current
                    queue.append(nxt)
    return labels, current


def _panel_boxes(rgb: np.ndarray) -> list:
    channel_max = rgb.max(axis=2)
    channel_min = rgb.min(axis=2)
    frame = (
        (channel_max >= FRAME_MAX_LO)
        & (channel_max <= FRAME_MAX_HI)
        & ((channel_max - channel_min) <= FRAME_SATURATION)
    )
    labels, count = _label(frame)
    boxes = []
    for value in range(1, count + 1):
        rows, cols = np.nonzero(labels == value)
        if rows.size < MIN_FRAME_PIXELS:
            continue
        boxes.append((int(cols.min()), int(rows.min()), int(cols.max()), int(rows.max())))
    boxes.sort(key=lambda box: box[0])
    return boxes


def analyse(image: Image.Image) -> dict:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    boxes = _panel_boxes(rgb)
    if len(boxes) != N_PANELS:
        return {"decision": AMBIGUOUS, "panels": len(boxes), "joined": []}

    joined = []
    for index, (x0, y0, x1, y1) in enumerate(boxes):
        patch = rgb[
            y0 + PANEL_INSET: y1 + 1 - PANEL_INSET,
            x0 + PANEL_INSET: x1 + 1 - PANEL_INSET,
        ]
        if patch.size == 0:
            return {"decision": AMBIGUOUS, "panels": len(boxes), "joined": []}
        red, green, blue = patch[:, :, 0], patch[:, :, 1], patch[:, :, 2]
        dark = patch.max(axis=2) < DARK_MAX
        port = (
            (red >= PORT_MIN_RED)
            & ((red - green) > PORT_CHANNEL_GAP)
            & ((red - blue) > PORT_CHANNEL_GAP)
        )
        port_labels, port_count = _label(port)
        seats = []
        for value in range(1, port_count + 1):
            rows, cols = np.nonzero(port_labels == value)
            if rows.size < MIN_PORT_PIXELS:
                continue
            seats.append((int(rows[0]), int(cols[0])))
        if len(seats) != 2:
            return {"decision": AMBIGUOUS, "panels": len(boxes), "joined": []}
        ink_labels, _ = _label(dark | port)
        first = int(ink_labels[seats[0]])
        second = int(ink_labels[seats[1]])
        if first and first == second:
            joined.append(index)

    if len(joined) != 1:
        return {"decision": AMBIGUOUS, "panels": len(boxes), "joined": joined}
    return {"decision": f"panel-{joined[0] + 1}", "panels": len(boxes), "joined": joined}


def decision_from_image(image: Image.Image) -> str:
    return analyse(image)["decision"]
