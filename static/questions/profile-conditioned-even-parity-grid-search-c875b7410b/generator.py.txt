"""Deterministic renderer for a distributed row/column parity search."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 690
GRID_N = 4
CELL = 48
GRID_SIZE = GRID_N * CELL
BASE_POSITIONS = ((120, 120), (590, 120), (120, 410), (590, 410))
LABELS = ("A", "B", "C", "D")
PALE = (244, 246, 242)
GRID_INK = (69, 102, 126)
PALE_BG = (252, 250, 245)
INK_PALETTES = (
    (38, 45, 54),
    (52, 36, 62),
    (31, 55, 52),
    (59, 42, 33),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _cells(mask: int) -> list[list[int]]:
    return [[(mask >> (r * GRID_N + c)) & 1 for c in range(GRID_N)] for r in range(GRID_N)]


def _score(cells: list[list[int]]) -> int:
    row_score = sum(sum(row) % 2 == 0 for row in cells)
    col_score = sum(sum(cells[r][c] for r in range(GRID_N)) % 2 == 0 for c in range(GRID_N))
    return int(row_score + col_score)


def _mask_pools() -> dict[int, list[int]]:
    pools: dict[int, list[int]] = {0: [], 2: [], 4: [], 6: [], 8: []}
    for occupied in itertools.combinations(range(GRID_N * GRID_N), 8):
        mask = sum(1 << bit for bit in occupied)
        pools[_score(_cells(mask))].append(mask)
    return pools


_POOLS = _mask_pools()


def sample_scene(seed: int) -> dict:
    """Sample four equal-inventory grids with one all-even winner."""
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    requested_scores = [6, 4, 2]
    rng.shuffle(requested_scores)
    masks_by_slot: list[int | None] = [None] * 4
    masks_by_slot[target_slot] = rng.choice(_POOLS[8])
    for slot, requested in zip((s for s in range(4) if s != target_slot), requested_scores):
        masks_by_slot[slot] = rng.choice(_POOLS[requested])
    if len(set(masks_by_slot)) != 4:
        # This is overwhelmingly unlikely across different score pools, but the
        # explicit assertion documents the visual-uniqueness invariant.
        raise RuntimeError("sampled duplicate panel masks")

    panels = []
    for slot, mask in enumerate(masks_by_slot):
        assert mask is not None
        panels.append(
            {
                "slot": slot,
                "cells": _cells(mask),
                "dx": rng.randint(-12, 12),
                "dy": rng.randint(-8, 8),
            }
        )
    return {
        "panels": panels,
        "labels": list(LABELS),
        "ink_color": list(INK_PALETTES[(int(seed) // 4) % len(INK_PALETTES)]),
        "background": list(PALE_BG),
    }


def _panel_scores(scene: dict) -> list[tuple[int, int]]:
    return [(int(panel["slot"]), _score(panel["cells"])) for panel in scene["panels"]]


def analytic_gold(scene: dict) -> str:
    scores = _panel_scores(scene)
    best = max(score for _, score in scores)
    winners = [slot for slot, score in scores if score == best]
    if best != 8 or len(winners) != 1:
        raise ValueError("scene does not have one all-even grid")
    return str(scene["labels"][winners[0]])


def margin(scene: dict) -> float:
    ordered = sorted((score for _, score in _panel_scores(scene)), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    try:
        panels = scene["panels"]
        if len(panels) != 4 or sorted(int(p["slot"]) for p in panels) != [0, 1, 2, 3]:
            return True
        for panel in panels:
            cells = panel["cells"]
            if len(cells) != 4 or any(len(row) != 4 for row in cells):
                return True
            flat = [value for row in cells for value in row]
            if any(value not in (0, 1) for value in flat) or sum(flat) != 8:
                return True
        scores = _panel_scores(scene)
        return margin(scene) < 2.0 or sum(score == 8 for _, score in scores) != 1
    except (KeyError, TypeError, ValueError):
        return True


def render(scene: dict) -> Image.Image:
    """Render using panel-owned slots, so panel-list order is a true alias."""
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(scene.get("background", PALE_BG)))
    draw = ImageDraw.Draw(image)
    labels = scene["labels"]
    ink = tuple(scene.get("ink_color", INK_PALETTES[0]))

    title_font = _font(28, bold=True)
    small_font = _font(17)
    label_font = _font(38, bold=True)
    draw.text((WIDTH // 2, 30), "EVEN IN EVERY ROW + COLUMN", fill=(37, 48, 57), font=title_font, anchor="mm")
    draw.text((WIDTH // 2, 64), "Count the dark squares along each horizontal and vertical line.", fill=(74, 78, 80), font=small_font, anchor="mm")

    for panel in scene["panels"]:
        slot = int(panel["slot"])
        base_x, base_y = BASE_POSITIONS[slot]
        x0 = base_x + int(panel.get("dx", 0))
        y0 = base_y + int(panel.get("dy", 0))
        draw.text((x0 - 48, y0 + GRID_SIZE // 2), str(labels[slot]), fill=(31, 39, 45), font=label_font, anchor="mm")

        # Pale cells first, then inset dark fills, then the connected slate grid.
        draw.rectangle((x0, y0, x0 + GRID_SIZE, y0 + GRID_SIZE), fill=PALE)
        for r, row in enumerate(panel["cells"]):
            for c, occupied in enumerate(row):
                if occupied:
                    pad = 6
                    draw.rectangle(
                        (
                            x0 + c * CELL + pad,
                            y0 + r * CELL + pad,
                            x0 + (c + 1) * CELL - pad,
                            y0 + (r + 1) * CELL - pad,
                        ),
                        fill=ink,
                    )
        for k in range(GRID_N + 1):
            xx = x0 + k * CELL
            yy = y0 + k * CELL
            draw.line((xx, y0, xx, y0 + GRID_SIZE), fill=GRID_INK, width=4)
            draw.line((x0, yy, x0 + GRID_SIZE, yy), fill=GRID_INK, width=4)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("reverse_panel_storage_order", twin)]
