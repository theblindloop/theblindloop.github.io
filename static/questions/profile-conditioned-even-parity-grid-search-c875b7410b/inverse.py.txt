"""Independent pixel-only inverse for the parity-grid search.

The inverse discovers candidate grids as connected components of the visible
slate lattice. It then measures cell interiors relative to each discovered
bounding box. It receives no latent scene and imports no candidate modules.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
GRID_N = 4


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _discover_grids(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    # Public slate grid ink is chromatically blue-gray, unlike neutral text,
    # neutral dark cells, and the pale background.
    rgb = arr.astype(np.int16)
    mask = (
        (rgb[:, :, 2] - rgb[:, :, 0] >= 35)
        & (rgb[:, :, 2] - rgb[:, :, 1] >= 15)
        & (rgb[:, :, 0] < 110)
        & (rgb[:, :, 1] < 140)
    )
    boxes = []
    for x0, y0, x1, y1, area in _components(mask):
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if area > 1500 and 150 <= width <= 260 and 150 <= height <= 260 and abs(width - height) <= 12:
            boxes.append((x0, y0, x1, y1))
    if len(boxes) != 4:
        raise OracleAbstention(f"expected four discovered grids, found {len(boxes)}")
    # The visible labels define ordinary row-major A/B/C/D order. Ordering is
    # relative among discovered regions; no canvas coordinate is assumed.
    by_height = sorted(boxes, key=lambda box: (box[1] + box[3]) / 2.0)
    top = sorted(by_height[:2], key=lambda box: (box[0] + box[2]) / 2.0)
    bottom = sorted(by_height[2:], key=lambda box: (box[0] + box[2]) / 2.0)
    return top + bottom


def _read_cells(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    cells: list[list[int]] = []
    for r in range(GRID_N):
        row = []
        for c in range(GRID_N):
            # Central 42% of each inferred cell avoids all lattice strokes.
            cx = x0 + (c + 0.5) * width / GRID_N
            cy = y0 + (r + 0.5) * height / GRID_N
            rx = max(5, int(width / GRID_N * 0.21))
            ry = max(5, int(height / GRID_N * 0.21))
            crop = arr[int(cy - ry) : int(cy + ry), int(cx - rx) : int(cx + rx)]
            if crop.size == 0:
                raise OracleAbstention("empty inferred cell interior")
            luminance = 0.2126 * crop[:, :, 0] + 0.7152 * crop[:, :, 1] + 0.0722 * crop[:, :, 2]
            dark_fraction = float(np.mean(luminance < 125.0))
            if dark_fraction >= 0.8:
                row.append(1)
            elif dark_fraction <= 0.2:
                row.append(0)
            else:
                raise OracleAbstention(f"ambiguous cell darkness fraction {dark_fraction:.3f}")
        cells.append(row)
    return cells


def _parity_score(cells: list[list[int]]) -> int:
    rows = sum(sum(row) % 2 == 0 for row in cells)
    cols = sum(sum(cells[r][c] for r in range(4)) % 2 == 0 for c in range(4))
    return int(rows + cols)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    boxes = _discover_grids(arr)
    scores = []
    for index, box in enumerate(boxes):
        cells = _read_cells(arr, box)
        if sum(value for row in cells for value in row) != 8:
            raise OracleAbstention("a candidate grid does not have the declared equal inventory")
        scores.append((LABELS[index], _parity_score(cells)))
    ordered = sorted(scores, key=lambda item: item[1], reverse=True)
    if ordered[0][1] != 8 or ordered[0][1] - ordered[1][1] < 2:
        raise OracleAbstention(f"no unique robust all-even winner: {scores}")
    return ordered[0][0]
