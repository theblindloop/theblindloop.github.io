"""Deterministic renderer for the bilateral butterfly spot-order world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 420
SCALE = 3
BACKGROUND = (249, 247, 242)
INK = (47, 43, 53)

_SPOT_COLORS = (
    (220, 55, 67),
    (42, 112, 222),
    (35, 168, 102),
    (240, 151, 34),
    (145, 69, 205),
    (18, 166, 186),
)
_WING_STYLES = (
    ((235, 222, 246), (120, 101, 139)),
    ((249, 226, 205), (148, 111, 91)),
    ((218, 239, 232), (91, 132, 121)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced control/counterfactual scene from one integer seed."""

    # Adjacent even/odd seeds form an explicitly matched pair. Every nuisance
    # draw is shared; only the right-wing order differs.
    rng = random.Random(int(seed) // 2)
    palette = [list(color) for color in rng.sample(_SPOT_COLORS, 4)]
    left_order = list(range(4))
    rng.shuffle(left_order)
    swap_at = rng.randrange(3)
    right_order = list(left_order)
    if int(seed) % 2:
        right_order[swap_at], right_order[swap_at + 1] = (
            right_order[swap_at + 1],
            right_order[swap_at],
        )
    wing_fill, wing_accent = rng.choice(_WING_STYLES)
    return {
        "left_order": left_order,
        "right_order": right_order,
        "palette": palette,
        "spot_radius": rng.randint(16, 19),
        "center_x": 320 + rng.randint(-12, 12),
        "center_y": 210 + rng.randint(-5, 5),
        "x_spread": rng.randint(215, 225),
        "y_gap": rng.randint(72, 76),
        "spot_jitter": [
            [rng.randint(-7, 7), rng.randint(-3, 3)] for _ in range(8)
        ],
        "wing_fill": list(wing_fill),
        "wing_accent": list(wing_accent),
        "body_tilt": rng.randint(-5, 5),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return whether the two visible ordinal color sequences are equal."""

    return "yes" if tuple(scene["left_order"]) == tuple(scene["right_order"]) else "no"


def _color_separation(palette: list[list[int]]) -> float:
    return min(
        math.sqrt(sum((float(a) - float(b)) ** 2 for a, b in zip(palette[i], palette[j])))
        for i in range(4)
        for j in range(i + 1, 4)
    )


def margin(scene: dict[str, Any]) -> float:
    """Visible runner-up margin: weakest color or spatial segmentation clearance."""

    palette_clearance = _color_separation(scene["palette"]) - 55.0
    jitter = scene["spot_jitter"]
    worst_vertical_jitter = max(abs(int(item[1])) for item in jitter)
    spatial_clearance = (
        float(scene["y_gap"])
        - 2.0 * float(scene["spot_radius"])
        - 2.0 * worst_vertical_jitter
        - 8.0
    )
    bilateral_clearance = 2.0 * float(scene["x_spread"]) - 120.0
    return round(min(palette_clearance, spatial_clearance, bilateral_clearance), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """There are no pixel-identical latent aliases in this explicit scene model."""

    return []


def _scaled(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _ellipse(
    draw: ImageDraw.ImageDraw,
    box: tuple[float, float, float, float],
    *,
    fill: tuple[int, int, int],
    outline: tuple[int, int, int] | None = None,
    width: int = 1,
) -> None:
    draw.ellipse(
        tuple(round(value * SCALE) for value in box),
        fill=fill,
        outline=outline,
        width=width * SCALE,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a recognizable butterfly with two far-separated spot sequences."""

    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    spread = int(scene["x_spread"])
    gap = int(scene["y_gap"])
    radius = int(scene["spot_radius"])
    jitter = scene["spot_jitter"]
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    left_order = [int(value) for value in scene["left_order"]]
    right_order = [int(value) for value in scene["right_order"]]
    wing_fill = tuple(int(channel) for channel in scene["wing_fill"])
    wing_accent = tuple(int(channel) for channel in scene["wing_accent"])
    tilt = int(scene["body_tilt"])

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    left_wing = [
        (cx - 13, cy - 11),
        (cx - 78, cy - 174),
        (cx - 230, cy - 161),
        (cx - 207, cy - 25),
        (cx - 237, cy + 154),
        (cx - 76, cy + 174),
        (cx - 13, cy + 15),
    ]
    right_wing = [(2 * cx - x, y) for x, y in left_wing]
    draw.polygon(_scaled(left_wing), fill=wing_fill, outline=INK, width=5 * SCALE)
    draw.polygon(_scaled(right_wing), fill=wing_fill, outline=INK, width=5 * SCALE)

    # Low-saturation veins establish the butterfly world but are irrelevant to
    # the high-saturation spot oracle.
    for side in (-1, 1):
        for offset in (-92, 6, 100):
            draw.line(
                _scaled(
                    [
                        (cx + side * 18, cy + offset * 0.12),
                        (cx + side * 188, cy + offset),
                    ]
                ),
                fill=wing_accent,
                width=3 * SCALE,
            )

    for index in range(4):
        base_y = cy + (index - 1.5) * gap
        for side_index, (side, order) in enumerate(((-1, left_order), (1, right_order))):
            dx, dy = jitter[side_index * 4 + index]
            x = cx + side * spread + int(dx)
            y = base_y + int(dy)
            color = palette[order[index]]
            _ellipse(
                draw,
                (x - radius - 3, y - radius - 3, x + radius + 3, y + radius + 3),
                fill=BACKGROUND,
                outline=INK,
                width=2,
            )
            _ellipse(
                draw,
                (x - radius, y - radius, x + radius, y + radius),
                fill=color,
            )

    body_points = [
        (cx + tilt, cy - 139),
        (cx + 15, cy - 105),
        (cx + 11, cy + 125),
        (cx - tilt, cy + 150),
        (cx - 11, cy + 125),
        (cx - 15, cy - 105),
    ]
    draw.polygon(_scaled(body_points), fill=INK)
    _ellipse(draw, (cx - 19, cy - 166, cx + 19, cy - 128), fill=INK)
    draw.arc(
        tuple(round(value * SCALE) for value in (cx - 78, cy - 211, cx, cy - 132)),
        195,
        310,
        fill=INK,
        width=3 * SCALE,
    )
    draw.arc(
        tuple(round(value * SCALE) for value in (cx, cy - 211, cx + 78, cy - 132)),
        230,
        345,
        fill=INK,
        width=3 * SCALE,
    )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
