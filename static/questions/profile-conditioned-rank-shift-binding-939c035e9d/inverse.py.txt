"""Independent image-only inverse for cross-panel rank displacement."""

from __future__ import annotations

import numpy as np
from PIL import Image


_NAMES = ("red", "green", "blue")
_COLORS = np.asarray(((220, 65, 65), (45, 175, 95), (65, 110, 220)), dtype=np.int16)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        seen[start_y, start_x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dx and not dy:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1]:
                        if mask[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))
        components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _marker_centers(rgb: np.ndarray, color: np.ndarray) -> list[tuple[float, float]]:
    distance = np.linalg.norm(rgb.astype(np.int16) - color[None, None, :], axis=2)
    mask = distance <= 42.0
    centers: list[tuple[float, float]] = []
    for component in _components(mask):
        if len(component) < 350:
            continue
        y0, x0 = component.min(axis=0)
        y1, x1 = component.max(axis=0)
        if not (28 <= x1 - x0 + 1 <= 58 and 28 <= y1 - y0 + 1 <= 58):
            continue
        centers.append((float(component[:, 1].mean()), float(component[:, 0].mean())))
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Recover the winning color from only the final raster."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[1] < 500 or rgb.shape[0] < 300:
        raise ValueError("image is not a usable paired marker panel")
    split = rgb.shape[1] / 2.0
    panel_y: list[list[float]] = [[], []]
    for color in _COLORS:
        centers = _marker_centers(rgb, color)
        left = [center[1] for center in centers if center[0] < split]
        right = [center[1] for center in centers if center[0] >= split]
        if len(left) != 1 or len(right) != 1:
            raise ValueError("expected one marker of each color in each panel")
        panel_y[0].append(left[0])
        panel_y[1].append(right[0])

    order_a = sorted(range(3), key=lambda identity: panel_y[0][identity])
    order_b = sorted(range(3), key=lambda identity: panel_y[1][identity])
    rank_a = {identity: rank for rank, identity in enumerate(order_a)}
    rank_b = {identity: rank for rank, identity in enumerate(order_b)}
    scores = [abs(rank_a[identity] - rank_b[identity]) for identity in range(3)]
    best = max(scores)
    if scores.count(best) != 1:
        raise ValueError("image has a tied rank-displacement decision")
    return _NAMES[scores.index(best)]
