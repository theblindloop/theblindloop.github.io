"""Deterministic renderer and analytic gold for cross-panel rank displacement."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "rank-shift-binding-0.1.0"
CANVAS = (900, 520)
PANEL_X = (40, 480)
PANEL_WIDTH = 380
PANEL_TOP = 70
PANEL_BOTTOM = 465
RANK_Y = (140.0, 260.0, 380.0)
COLOR_NAMES = ("red", "green", "blue")
PALETTE = {
    0: (220, 65, 65),
    1: (45, 175, 95),
    2: (65, 110, 220),
}


def _panel_geometry(rng: random.Random, panel_x: int, order: list[int]) -> tuple[list[float], list[float]]:
    """Place one ordered panel with independently shuffled horizontal slots."""

    slots = [85.0, 190.0, 295.0]
    rng.shuffle(slots)
    x_by_identity = [0.0, 0.0, 0.0]
    for identity, slot in zip(range(3), slots):
        x_by_identity[identity] = round(panel_x + slot + rng.uniform(-16.0, 16.0), 3)

    y_by_rank = [round(base + rng.uniform(-12.0, 12.0), 3) for base in RANK_Y]
    return x_by_identity, y_by_rank


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a unique three-cycle rank transition with balanced winner identity."""

    rng = random.Random(int(seed))
    winner = int(seed) % 3
    others = [identity for identity in range(3) if identity != winner]
    rng.shuffle(others)
    if rng.choice((0, 1)) == 0:
        order_a = [winner, others[0], others[1]]
        order_b = [others[0], others[1], winner]
    else:
        order_a = [others[0], others[1], winner]
        order_b = [winner, others[0], others[1]]

    x_a, y_a = _panel_geometry(rng, PANEL_X[0], order_a)
    x_b, y_b = _panel_geometry(rng, PANEL_X[1], order_b)
    return {
        "order_a": order_a,
        "order_b": order_b,
        "x_a": x_a,
        "x_b": x_b,
        "y_a": y_a,
        "y_b": y_b,
        "marker_radius": rng.randint(18, 22),
        "canvas": list(CANVAS),
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary into an RGB PNG."""

    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (247, 249, 252))
    draw = ImageDraw.Draw(image)

    def sc(value: float) -> int:
        return int(round(float(value) * scale))

    try:
        label_font = _font(24 * scale)
    except Exception:
        label_font = ImageFont.load_default()

    for panel_index, panel_x in enumerate(PANEL_X):
        x0, x1 = panel_x, panel_x + PANEL_WIDTH
        draw.rounded_rectangle(
            (sc(x0), sc(PANEL_TOP), sc(x1), sc(PANEL_BOTTOM)),
            radius=sc(14),
            fill=(255, 255, 255),
            outline=(142, 151, 163),
            width=sc(2),
        )
        draw.text(
            (sc(x0 + 18), sc(PANEL_TOP + 14)),
            "A" if panel_index == 0 else "B",
            fill=(40, 48, 58),
            font=label_font,
        )
        for guide_y in RANK_Y:
            draw.line(
                (sc(x0 + 34), sc(guide_y), sc(x1 - 24), sc(guide_y)),
                fill=(231, 235, 240),
                width=sc(1),
            )
        xs = scene["x_a"] if panel_index == 0 else scene["x_b"]
        ys = scene["y_a"] if panel_index == 0 else scene["y_b"]
        order = scene["order_a"] if panel_index == 0 else scene["order_b"]
        radius = int(scene["marker_radius"])
        for rank, identity in enumerate(order):
            cx, cy = float(xs[identity]), float(ys[rank])
            draw.ellipse(
                (sc(cx - radius), sc(cy - radius), sc(cx + radius), sc(cy + radius)),
                fill=PALETTE[identity],
            )

    draw.line((sc(450), sc(86), sc(450), sc(448)), fill=(225, 229, 234), width=sc(1))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _rank_scores(scene: dict[str, Any]) -> dict[int, int]:
    """Return absolute rank displacement for each identity, from visible orders."""

    order_a = scene["order_a"]
    order_b = scene["order_b"]
    rank_a = {identity: rank for rank, identity in enumerate(order_a)}
    rank_b = {identity: rank for rank, identity in enumerate(order_b)}
    return {identity: abs(rank_a[identity] - rank_b[identity]) for identity in range(3)}


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the unique color with the greatest absolute rank displacement."""

    scores = _rank_scores(scene)
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if len(ranked) < 2 or ranked[0][1] == ranked[1][1]:
        raise ValueError("rank displacement is tied")
    return COLOR_NAMES[ranked[0][0]]


def margin(scene: dict[str, Any]) -> float:
    """Return the top-score minus runner-up gap."""

    values = sorted(_rank_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject ties and marker arrangements too close for stable rank recovery."""

    if margin(scene) < 1.0:
        return True
    for ys in (scene["y_a"], scene["y_b"]):
        for first in range(3):
            for second in range(first + 1, 3):
                if abs(float(ys[first]) - float(ys[second])) < 45.0:
                    return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed for this directly rendered parameterization."""

    return []
