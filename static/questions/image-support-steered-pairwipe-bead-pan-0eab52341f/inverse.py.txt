"""Scene-free pixel oracle for the `pairwipe_bead_pan` world.

It receives a rasterised PNG and nothing else.  It segments the four saturated
bead colours, groups them into solid square blobs, separates the single blob
that stands off the shared row (the bead already in the pan) from the evenly
spaced queue row, replays the wipe rule left to right and reports the colour
left on top.  It abstains whenever the picture does not present exactly that
structure.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "pairwipe_bead_pan-oracle-0.1.0"
ABSTAIN = "abstain"

_NAMES = ("red", "green", "blue", "yellow")
_RGB = (
    (208, 32, 40),
    (26, 140, 58),
    (36, 72, 196),
    (226, 168, 20),
)
_MIN_BEAD_PIXELS = 400
_SQUARENESS = 0.82
_FILL = 0.9


def _components(mask: np.ndarray) -> list:
    """Four-connected components of a boolean mask, as (y0, y1, x0, x1, count)."""
    height, width = mask.shape
    seen = np.zeros((height, width), dtype=bool)
    found = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        seen[start_y, start_x] = True
        y0 = y1 = start_y
        x0 = x1 = start_x
        count = 0
        while stack:
            cy, cx = stack.pop()
            count += 1
            if cy < y0:
                y0 = cy
            if cy > y1:
                y1 = cy
            if cx < x0:
                x0 = cx
            if cx > x1:
                x1 = cx
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width:
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        found.append((y0, y1, x0, x1, count))
    return found


def _beads(image: Image.Image) -> list:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    beads = []
    for index, rgb in enumerate(_RGB):
        mask = np.all(arr == np.array(rgb, dtype=np.int16), axis=2)
        if not mask.any():
            continue
        for y0, y1, x0, x1, count in _components(mask):
            if count < _MIN_BEAD_PIXELS:
                return []
            height = y1 - y0 + 1
            width = x1 - x0 + 1
            if min(height, width) / max(height, width) < _SQUARENESS:
                return []
            if count < _FILL * height * width:
                return []
            beads.append(
                {
                    "colour": index,
                    "cx": (x0 + x1) / 2.0,
                    "cy": (y0 + y1) / 2.0,
                    "size": (height + width) / 2.0,
                }
            )
    return beads


def decision_from_image(image: Image.Image) -> str:
    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image accepts a PIL image only")

    beads = _beads(image)
    if len(beads) < 4 or len(beads) > 12:
        return ABSTAIN

    typical = sorted(bead["size"] for bead in beads)[len(beads) // 2]
    if any(abs(bead["size"] - typical) > 0.25 * typical for bead in beads):
        return ABSTAIN

    # the queue is the largest set of beads sharing a row; the pan bead stands off
    best_row: list = []
    for anchor in beads:
        row = [b for b in beads if abs(b["cy"] - anchor["cy"]) <= 0.4 * typical]
        if len(row) > len(best_row):
            best_row = row
    if len(best_row) != len(beads) - 1:
        return ABSTAIN

    lone = [b for b in beads if b not in best_row]
    if len(lone) != 1:
        return ABSTAIN
    pan = lone[0]

    row_cy = sum(b["cy"] for b in best_row) / len(best_row)
    if abs(pan["cy"] - row_cy) < 1.2 * typical:
        return ABSTAIN
    if any(abs(b["cy"] - row_cy) > 0.4 * typical for b in best_row):
        return ABSTAIN

    queue = sorted(best_row, key=lambda b: b["cx"])
    if len(queue) < 3:
        return ABSTAIN

    # the queue must be a single evenly spaced run, so its order is unambiguous
    steps = [queue[i + 1]["cx"] - queue[i]["cx"] for i in range(len(queue) - 1)]
    if min(steps) <= typical:
        return ABSTAIN
    if max(steps) - min(steps) > 0.35 * typical:
        return ABSTAIN

    pile = [pan["colour"]]
    for bead in queue:
        if pile and pile[-1] == bead["colour"]:
            pile.pop()
        else:
            pile.append(bead["colour"])
    if not pile:
        return ABSTAIN
    return _NAMES[pile[-1]]
