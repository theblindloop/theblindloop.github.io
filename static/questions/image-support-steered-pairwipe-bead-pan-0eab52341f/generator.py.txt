"""Renderer and analytic gold for the `pairwipe_bead_pan` world.

One bead stands alone above a short evenly spaced row of beads.  The lone bead
is the bead already lying in the pan; the row is a queue that is tipped into the
pan one bead at a time, from left to right.  A bead that arrives on top of a
bead of its own colour wipes that bead out and is destroyed with it; otherwise
it simply comes to rest on top.  The question asks which colour is left showing
on top of the pan once the whole queue has been tipped in.

Nothing draws the pan's interior, so the state being tracked is never visible:
only the queue and the single seed bead are drawn, and the running pile has to
be maintained by the solver.

`render` is a pure function of the scene dictionary.  Every scene field is a
plain JSON type, so a JSON round trip reproduces the identical raster.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pairwipe_bead_pan-renderer-0.1.0"

CANVAS_W = 800
CANVAS_H = 520
BG = (246, 246, 246)
DECOR = (204, 204, 204)

COLOUR_NAMES = ("red", "green", "blue", "yellow")
COLOURS = {
    "red": (208, 32, 40),
    "green": (26, 140, 58),
    "blue": (36, 72, 196),
    "yellow": (226, 168, 20),
}
RGB_BY_INDEX = tuple(COLOURS[name] for name in COLOUR_NAMES)

QUEUE_MIN = 5
QUEUE_MAX = 6
SIZE_MIN, SIZE_MAX = 32, 36
GAP_MIN, GAP_MAX = 14, 20
VGAP_MIN, VGAP_MAX = 130, 200
BAND_STEP = 65
BAND_TOPS = (65, 130)
OX_MIN, OX_MAX = 180, 240
MARGIN_FLOOR = 6.0
ALT_PAN_MIN = 1
DROP_MIN = 4
DECOR_CLEAR = 14


# --------------------------------------------------------------------------
# geometry helpers (used by the renderer, the analytic arm and the tests)
# --------------------------------------------------------------------------
def _bead_boxes(scene: dict) -> list:
    """Return [(x, y, size)] for the queue beads, left to right."""
    ox, oy = int(scene["origin"][0]), int(scene["origin"][1])
    size = int(scene["size"])
    gap = int(scene["gap"])
    top = oy + size + int(scene["vgap"])
    return [(ox + i * (size + gap), top, size) for i in range(len(scene["queue"]))]


def _pan_box(scene: dict) -> tuple:
    ox, oy = int(scene["origin"][0]), int(scene["origin"][1])
    return (ox + int(scene["pan_dx"]), oy, int(scene["size"]))


def _support_box(scene: dict) -> tuple:
    """Bounding box of every decisive mark: (x0, y0, x1, y1), exclusive end."""
    boxes = _bead_boxes(scene) + [_pan_box(scene)]
    x0 = min(b[0] for b in boxes)
    y0 = min(b[1] for b in boxes)
    x1 = max(b[0] + b[2] for b in boxes)
    y1 = max(b[1] + b[2] for b in boxes)
    return (x0, y0, x1, y1)


def _queue_box(scene: dict) -> tuple:
    boxes = _bead_boxes(scene)
    x0 = min(b[0] for b in boxes)
    y0 = min(b[1] for b in boxes)
    x1 = max(b[0] + b[2] for b in boxes)
    y1 = max(b[1] + b[2] for b in boxes)
    return (x0, y0, x1, y1)


# --------------------------------------------------------------------------
# the replay itself
# --------------------------------------------------------------------------
def _replay(pan: int, queue: list) -> list:
    """Tip `queue` into a pan already holding `pan`; return the resulting pile."""
    pile = [int(pan)]
    for bead in queue:
        bead = int(bead)
        if pile and pile[-1] == bead:
            pile.pop()
        else:
            pile.append(bead)
    return pile


def analytic_gold(scene: dict) -> str:
    pile = _replay(scene["pan"], list(scene["queue"]))
    if not pile:
        raise ValueError("the pan finished empty; such scenes are quarantined")
    return COLOUR_NAMES[pile[-1]]


def margin(scene: dict) -> float:
    """Smallest decisive clearance in pixels.

    The decision is a discrete replay, so the finite-resolution risk is purely
    one of reading the drawn beads apart: the horizontal gaps inside the queue,
    the vertical stand-off of the lone pan bead, and the clearance between any
    inert decoration and the decisive band.
    """
    clearances = [float(int(scene["gap"])), float(int(scene["vgap"]))]
    x0, y0, x1, y1 = _support_box(scene)
    for dx, dy, dw, dh in scene["decor"]:
        gx = max(x0 - (dx + dw), dx - x1, 0)
        gy = max(y0 - (dy + dh), dy - y1, 0)
        clearances.append(float(max(gx, gy)))
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    queue = [int(v) for v in scene["queue"]]
    pan = int(scene["pan"])
    if not QUEUE_MIN <= len(queue) <= QUEUE_MAX:
        return True
    pile = _replay(pan, queue)
    if not pile:
        return True
    if margin(scene) < MARGIN_FLOOR:
        return True
    x0, y0, x1, y1 = _support_box(scene)
    if x0 < 40 or y0 < 40 or x1 > CANVAS_W - 40 or y1 > CANVAS_H - 40:
        return True
    # the seed bead must matter, the last tipped bead must not be the answer,
    # and every queue bead must be load bearing
    answer = pile[-1]
    if answer == pan:
        return True
    if answer == queue[-1]:
        return True
    swings = 0
    for alternative in range(4):
        if alternative == pan:
            continue
        other = _replay(alternative, queue)
        if not other or other[-1] != answer:
            swings += 1
    if swings < ALT_PAN_MIN:
        return True
    drawn = [pan] + queue
    counts = [drawn.count(index) for index in range(4)]
    if counts[answer] == max(counts) and counts.count(max(counts)) == 1:
        return True
    load_bearing = 0
    for drop in range(len(queue)):
        shorter = queue[:drop] + queue[drop + 1 :]
        other = _replay(pan, shorter)
        if not other or other[-1] != answer:
            load_bearing += 1
    if load_bearing < DROP_MIN:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every label input is drawn flat."""
    return []


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _sample_decor(rng: random.Random, support: tuple) -> list:
    x0, y0, x1, y1 = support
    marks = []
    for _ in range(rng.randint(6, 12)):
        for _attempt in range(60):
            w = rng.randint(14, 30)
            h = rng.choice([w, rng.randint(10, 26)])
            dx = rng.randint(12, CANVAS_W - w - 12)
            dy = rng.randint(12, CANVAS_H - h - 12)
            if (
                dx + w + DECOR_CLEAR < x0
                or dx > x1 + DECOR_CLEAR
                or dy + h + DECOR_CLEAR < y0
                or dy > y1 + DECOR_CLEAR
            ):
                marks.append([dx, dy, w, h])
                break
    return marks


def _geometry(rng: random.Random, length: int) -> dict:
    size = rng.randint(SIZE_MIN, SIZE_MAX)
    gap = rng.randint(GAP_MIN, GAP_MAX)
    # the seed bead and the queue are held far enough apart that a wide empty
    # horizontal band always separates them; nothing decisive ever joins them
    band = rng.choice(BAND_TOPS)
    seed_top = band + rng.randint(0, 22)
    queue_top = band + 3 * BAND_STEP + rng.randint(2, 26)
    vgap = queue_top - seed_top - size
    span = length * size + (length - 1) * gap
    ox = rng.randint(OX_MIN, min(OX_MAX, CANVAS_W - 60 - span))
    oy = seed_top
    pan_dx = rng.randint(0, min(60, span - size))
    return {
        "origin": [ox, oy],
        "size": size,
        "gap": gap,
        "vgap": vgap,
        "pan_dx": pan_dx,
    }


def sample_scene(seed: int) -> dict:
    """Sample one non-quarantined scene; the answer colour is fixed by the seed."""
    rng = random.Random(seed)
    target = int(seed) % 4
    for _attempt in range(40000):
        length = rng.randint(QUEUE_MIN, QUEUE_MAX)
        pan = rng.randrange(4)
        queue = [rng.randrange(4) for _ in range(length)]
        pile = _replay(pan, queue)
        if not pile or pile[-1] != target:
            continue
        if len(pile) > 4:
            continue
        scene = {"pan": pan, "queue": queue}
        scene.update(_geometry(rng, length))
        scene["decor"] = _sample_decor(rng, _support_box(scene))
        if is_quarantined(scene):
            continue
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BG)
    draw = ImageDraw.Draw(image)

    for dx, dy, w, h in scene["decor"]:
        draw.rectangle([dx, dy, dx + w - 1, dy + h - 1], fill=DECOR)

    px, py, size = _pan_box(scene)
    draw.rectangle(
        [px, py, px + size - 1, py + size - 1], fill=RGB_BY_INDEX[int(scene["pan"])]
    )
    for (bx, by, bsize), bead in zip(_bead_boxes(scene), scene["queue"]):
        draw.rectangle(
            [bx, by, bx + bsize - 1, by + bsize - 1], fill=RGB_BY_INDEX[int(bead)]
        )
    return image
