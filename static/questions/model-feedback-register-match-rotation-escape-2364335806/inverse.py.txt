"""Independent fixed-layout pixel inverse for match-selected rotations."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1040, 800
N, CELL = 4, 108
GRID_X, GRID_Y = 304, 210
QUEUE_XS, QUEUE_Y = (385, 475, 565, 655), 105
BACKGROUND = (247, 246, 241)
GOLD = (244, 190, 51)
BLUE = (54, 128, 194)
RED = (211, 46, 57)
VIOLET = (124, 58, 188)
LINK_COLORS = ((17, 150, 132), (238, 112, 28), (86, 91, 190), (190, 76, 151),
               (94, 148, 45), (0, 158, 204), (177, 130, 0), (112, 112, 112))
DIRS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> np.ndarray:
    return np.all(np.abs(arr.astype(np.int16) - np.array(color, dtype=np.int16)) <= tolerance, axis=2)


def _near(pixel: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _classify(pixel: np.ndarray, colors: tuple[tuple[int, int, int], ...]) -> int | None:
    hits = [index for index, color in enumerate(colors) if _near(pixel, color)]
    return hits[0] if len(hits) == 1 else None


def _token(arr: np.ndarray, color: tuple[int, int, int]) -> tuple[list[int], int] | None:
    crop = arr[GRID_Y:GRID_Y + N * CELL, GRID_X:GRID_X + N * CELL]
    mask = _mask(crop, color)
    hits = []
    for row in range(N):
        for col in range(N):
            patch = mask[row * CELL:(row + 1) * CELL, col * CELL:(col + 1) * CELL].copy()
            patch[:43, :54] = False
            patch[67:, 67:] = False
            if int(patch.sum()) > 350:
                hits.append((row, col))
    if len(hits) != 1:
        return None
    row, col = hits[0]
    patch = mask[row * CELL:(row + 1) * CELL, col * CELL:(col + 1) * CELL].copy()
    patch[:43, :54] = False
    patch[67:, 67:] = False
    ys, xs = np.nonzero(patch)
    if len(xs) < 350:
        return None
    cx, cy = 54, 56
    extents = (cy - int(ys.min()), int(xs.max()) - cx,
               int(ys.max()) - cy, cx - int(xs.min()))
    best = max(extents)
    if extents.count(best) != 1 or best < 23:
        return None
    return [row, col], extents.index(best)


def _mate(links: list[list[int]], row: int, col: int) -> tuple[int, int] | None:
    hits = [(r, c) for r in range(N) for c in range(N) if links[r][c] == links[row][col]]
    if len(hits) != 2:
        return None
    return hits[1] if hits[0] == (row, col) else hits[0]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (HEIGHT, WIDTH, 3) or not _near(arr[785, 1025], BACKGROUND):
        return "abstain"
    turns, registers, links = [], [], []
    for row in range(N):
        turn_row, reg_row, link_row = [], [], []
        for col in range(N):
            x0, y0 = GRID_X + col * CELL, GRID_Y + row * CELL
            turn = _classify(arr[y0 + 94, x0 + 18], (GOLD, BLUE))
            reg = _classify(arr[y0 + 93, x0 + 78], (RED, VIOLET))
            link = _classify(arr[y0 + 17, x0 + 14], LINK_COLORS)
            if turn is None or reg is None or link is None:
                return "abstain"
            turn_row.append(turn)
            reg_row.append(reg)
            link_row.append(link)
        turns.append(turn_row)
        registers.append(reg_row)
        links.append(link_row)
    if sorted(v for row in links for v in row) != sorted(list(range(8)) * 2):
        return "abstain"
    queue = []
    for cx in QUEUE_XS:
        actor = _classify(arr[QUEUE_Y - 15, cx], (RED, VIOLET))
        if actor is None:
            return "abstain"
        queue.append(actor)
    red, violet = _token(arr, RED), _token(arr, VIOLET)
    if red is None or violet is None or red[0] == violet[0]:
        return "abstain"
    positions = [red[0], violet[0]]
    headings = [red[1], violet[1]]
    for _ in range(80):
        actor = queue.pop(0)
        row, col = positions[actor]
        mate = _mate(links, row, col)
        if mate is None:
            return "abstain"
        mr, mc = mate
        old_turn = turns[row][col]
        old_here, old_mate = registers[row][col], registers[mr][mc]
        headings[actor] = (headings[actor] + (1 if old_turn == 0 else -1)) % 4
        turns[row][col] ^= 1
        if old_here == old_mate:
            registers[row][col] = actor
            registers[mr][mc] = old_here
            queue.append(old_mate)
        else:
            registers[mr][mc] = actor
            registers[row][col] = old_mate
            queue.append(old_here)
        dr, dc = DIRS[headings[actor]]
        positions[actor] = [row + dr, col + dc]
        nr, nc = positions[actor]
        if nr < 0 or nr >= N or nc < 0 or nc >= N:
            return ("token-red", "token-violet")[actor]
    return "abstain"
