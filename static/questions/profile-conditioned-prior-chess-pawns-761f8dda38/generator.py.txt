"""Prior-conflict chess-pawn world: count visible pawns.

The latent scene holds a row of ``n`` chess pawns standing on a dark board. A
familiar canonical prior asserts that one side of a chess set has EIGHT pawns;
the counterfactual renders a non-canonical count (6 or 4) and the decision must
be read from the visible pawns alone, never from the prior. The pixel oracle
counts the actual pawn-colour components.

Latent alias mode is ``not_applicable``: every latent field (pawn count, size,
horizontal cell, vertical placement, canvas) is directly reflected in the
raster, so the rendering is injective in the latent scene and there is no
render-invariant alias. The canonical-prior name "chess pawns" is a
matched-control/counterfactual manipulation of the *question*, not of any
latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
# Fixed horizontal inset so the row is centred on the board.
MARGIN_X = 70
# Ivory pawn silhouettes on a dark felt board keep a two-colour world whose
# connected pawn components are trivially separable.
BACKGROUND = (46, 58, 52)
PAWN = (236, 224, 190)
# Base ellipse half-width as a fraction of the pawn body width.
BASE_FRAC = 0.68

# Supported off-quarantine counts. The canonical prior is 8; 6 and 4 are the
# counterfactuals that contradict it. All remain far enough apart at this
# canvas that each pawn is an isolated ivory component.
SUPPORTED_N = (8, 6, 4)

# A count is quarantined when adjacent pawn components can merge at pixel
# scale. The margin for the supported counts stays well above this threshold.
QUAR_MIN_GAP = 6.0


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    cell = (CANVAS - 2 * MARGIN_X) / n
    # pawn size and vertical placement vary independently of n so that neither
    # size nor position predicts the count (cheap-cue counterbalance).
    pawn_w = rng.uniform(19.0, 23.0)
    row_y = rng.uniform(300.0, 392.0)
    return {
        "n": n,
        "cell": round(cell, 3),
        "pawn_w": round(pawn_w, 3),
        "row_y": round(row_y, 3),
        "canvas": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    """Render the row of pawns to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    n = int(scene["n"])
    cell = float(scene["cell"])
    w = float(scene["pawn_w"])
    row_y = float(scene["row_y"])
    h_rad = 0.5 * w
    total_h = 2.95 * w
    top_y = row_y - total_h
    for k in range(n):
        x = MARGIN_X + cell * (k + 0.5)
        # Wide base, then a slender body, then a ball head; they all overlap so
        # each pawn is one connected ivory component.
        draw.ellipse(
            [x - BASE_FRAC * w, row_y - 0.32 * w, x + BASE_FRAC * w, row_y + 0.16 * w],
            fill=PAWN,
        )
        draw.rounded_rectangle(
            [x - 0.5 * w, top_y + 0.9 * h_rad, x + 0.5 * w, row_y],
            radius=0.25 * w,
            fill=PAWN,
        )
        draw.ellipse([x - h_rad, top_y, x + h_rad, top_y + 2 * h_rad], fill=PAWN)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible pawn count, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = smallest horizontal gap between adjacent pawns."""
    cell = float(scene["cell"])
    w = float(scene["pawn_w"])
    return round(cell - 2 * BASE_FRAC * w, 3)


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded counts that are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_GAP


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
