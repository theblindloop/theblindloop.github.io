"""Independent pixel-only inverse arm for the signed-run crossing world.

Reads nothing but the rendered PNG.  It locates the gray separator rule, takes
the black ink above it as the target bar and the black ink below it as the
spine, measures every colored bar's drawn length and the side of the spine it
hangs on, then runs the signed accumulation itself and reports the color of the
first row whose running total reaches the target length.

It never imports the renderer, the prompts, the generator or the verifier, and
it never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (37, 86, 196),
    "crimson": (200, 26, 64),
    "green": (24, 140, 66),
    "orange": (232, 124, 20),
    "purple": (128, 52, 168),
    "teal": (0, 150, 160),
}
_BLACK = (0, 0, 0)
_GRAY = (150, 150, 150)


def _mask(arr: np.ndarray, color) -> np.ndarray:
    return (
        (arr[:, :, 0] == color[0])
        & (arr[:, :, 1] == color[1])
        & (arr[:, :, 2] == color[2])
    )


def _box(mask: np.ndarray):
    rows = np.flatnonzero(mask.any(axis=1))
    cols = np.flatnonzero(mask.any(axis=0))
    if rows.size == 0 or cols.size == 0:
        return None
    return int(cols[0]), int(rows[0]), int(cols[-1]), int(rows[-1])


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    rule = _box(_mask(arr, _GRAY))
    if rule is None:
        raise ValueError("no separator rule found")
    rule_top, rule_bottom = rule[1], rule[3]

    black = _mask(arr, _BLACK)
    above = black.copy()
    above[rule_top:, :] = False
    below = black.copy()
    below[: rule_bottom + 1, :] = False

    target_box = _box(above)
    spine_box = _box(below)
    if target_box is None or spine_box is None:
        raise ValueError("target bar or spine missing")
    if spine_box[3] - spine_box[1] < 4 * (spine_box[2] - spine_box[0]):
        raise ValueError("spine is not a tall narrow stroke")

    target_px = target_box[2] - target_box[0] + 1
    spine_center = (spine_box[0] + spine_box[2]) / 2.0

    bars = []
    for name, color in _PALETTE.items():
        box = _box(_mask(arr, color))
        if box is None:
            raise ValueError(f"missing bar color {name}")
        x0, y0, x1, y1 = box
        length = x1 - x0 + 1
        sign = 1 if (x0 + x1) / 2.0 > spine_center else -1
        bars.append(((y0 + y1) / 2.0, name, sign * length))
    bars.sort()

    # Unit inference is unnecessary: every quantity is a pixel length on one
    # shared scale, so the accumulation runs directly in pixels.
    total = 0
    for _, name, value in bars:
        total += value
        if total >= target_px:
            return name
    raise ValueError("running total never reaches the target length")
