"""Comb-wall rank-partner world.

One solid wall of ten flush teeth spans the whole canvas. Five teeth hang from
the top edge, five rise from the bottom edge, and neighbouring teeth always
overlap in height, so the inked stimulus is a single connected area-like mass.
Top teeth and bottom teeth are paired by *length rank* only: the k-th longest
top tooth partners the k-th longest bottom tooth. One top tooth carries a black
key square; the answer is the colour of its partner.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "combwall_rank_pair_bind-0.1.0"

WIDTH = 720
HEIGHT = 560
SLOTS = 10
SLOT_W = WIDTH // SLOTS

BACKGROUND = (246, 242, 232)
MARKER = (16, 16, 16)

TOP_COLORS = {
    "red": (198, 42, 44),
    "blue": (38, 84, 190),
    "green": (38, 138, 64),
    "purple": (132, 58, 168),
    "orange": (233, 142, 26),
}
BOTTOM_COLORS = {
    "teal": (20, 146, 146),
    "yellow": (226, 198, 44),
    "pink": (232, 116, 176),
    "brown": (132, 88, 44),
    "slate": (84, 96, 120),
}
TOP_NAMES = tuple(sorted(TOP_COLORS))
BOTTOM_NAMES = ("teal", "yellow", "pink", "brown", "slate")

TOP_SLOTS = (0, 2, 4, 6, 8)
BOTTOM_SLOTS = (1, 3, 5, 7, 9)

TOP_RANGE = (300, 540)
BOTTOM_RANGE = (380, 530)

MARKER_SIZE = 24
MARKER_TOP = 34

QUARANTINE_MARGIN = 22.0
NEAR_BOUNDARY_MIN_GAP = 16


def _sample_lengths(rng: random.Random, lo: int, hi: int, near_boundary: bool) -> list[int] | None:
    """Five ascending lengths in [lo, hi] with a controlled smallest gap."""
    span = hi - lo
    for _ in range(400):
        if near_boundary:
            gmin = rng.randint(NEAR_BOUNDARY_MIN_GAP, 21)
            extras = [rng.randint(0, 15) for _ in range(3)]
        else:
            gmin = rng.randint(26, 32)
            extras = [rng.randint(0, 5) for _ in range(3)]
        gaps = [gmin] + [gmin + e for e in extras]
        rng.shuffle(gaps)
        if sum(gaps) > span:
            continue
        start = lo + rng.randint(0, span - sum(gaps))
        values = [start]
        for gap in gaps:
            values.append(values[-1] + gap)
        return values
    return None


def sample_scene(seed: int) -> dict:
    """Sample one wall.

    Counterbalancing is deterministic in the seed: the answer colour cycles with
    ``seed % 5`` and the key tooth's length rank cycles with ``(seed // 5) % 5``,
    so answer identity is independent of rank, colour frequency and position.
    """
    rng = random.Random(seed * 7919 + 13)
    answer_colour = BOTTOM_NAMES[seed % 5]
    key_rank = (seed // 5) % 5
    near_top = seed % 7 == 3
    near_bottom = seed % 7 == 5

    for _ in range(200):
        top_vals = _sample_lengths(rng, TOP_RANGE[0], TOP_RANGE[1], near_top)
        bot_vals = _sample_lengths(rng, BOTTOM_RANGE[0], BOTTOM_RANGE[1], near_bottom)
        if top_vals is None or bot_vals is None:
            continue

        top_order = list(TOP_SLOTS)
        rng.shuffle(top_order)
        bot_order = list(BOTTOM_SLOTS)
        rng.shuffle(bot_order)

        lengths = [0] * SLOTS
        colors = [""] * SLOTS
        # descending length rank -> slot
        top_rank_slot = {}
        for rank, value in enumerate(reversed(top_vals)):
            slot = top_order[rank]
            lengths[slot] = value
            top_rank_slot[rank] = slot
        bot_rank_slot = {}
        for rank, value in enumerate(reversed(bot_vals)):
            slot = bot_order[rank]
            lengths[slot] = value
            bot_rank_slot[rank] = slot

        top_palette = list(TOP_NAMES)
        rng.shuffle(top_palette)
        for i, slot in enumerate(TOP_SLOTS):
            colors[slot] = top_palette[i]

        answer_slot = bot_rank_slot[key_rank]
        rest = [name for name in BOTTOM_NAMES if name != answer_colour]
        rng.shuffle(rest)
        colors[answer_slot] = answer_colour
        for slot in BOTTOM_SLOTS:
            if slot != answer_slot:
                colors[slot] = rest.pop()

        key_slot = top_rank_slot[key_rank]
        key_len = lengths[key_slot]

        # Anti-shortcut 1: the numerically closest bottom tooth must not be the
        # partner, so absolute length matching gives the wrong tooth.
        closest = min(BOTTOM_SLOTS, key=lambda s: (abs(lengths[s] - key_len), s))
        if closest == answer_slot:
            continue
        # Anti-shortcut 2: the partner is never a side-by-side neighbour of the key.
        if abs(answer_slot - key_slot) == 1:
            continue

        return {
            "colors": colors,
            "lengths": lengths,
            "key_slot": key_slot,
        }

    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _slot_box(slot: int, length: int) -> tuple[int, int, int, int]:
    x0 = slot * SLOT_W
    x1 = x0 + SLOT_W - 1
    if slot % 2 == 0:
        return x0, 0, x1, length - 1
    return x0, HEIGHT - length, x1, HEIGHT - 1


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot in range(SLOTS):
        name = scene["colors"][slot]
        rgb = TOP_COLORS[name] if slot % 2 == 0 else BOTTOM_COLORS[name]
        draw.rectangle(_slot_box(slot, int(scene["lengths"][slot])), fill=rgb)
    key = int(scene["key_slot"])
    mx0 = key * SLOT_W + (SLOT_W - MARKER_SIZE) // 2
    draw.rectangle(
        (mx0, MARKER_TOP, mx0 + MARKER_SIZE - 1, MARKER_TOP + MARKER_SIZE - 1),
        fill=MARKER,
    )
    return image


def _ranked(scene: dict, slots: tuple[int, ...]) -> list[int]:
    return sorted(slots, key=lambda s: (-int(scene["lengths"][s]), s))


def analytic_gold(scene: dict) -> str:
    top_ranked = _ranked(scene, TOP_SLOTS)
    bottom_ranked = _ranked(scene, BOTTOM_SLOTS)
    rank = top_ranked.index(int(scene["key_slot"]))
    return scene["colors"][bottom_ranked[rank]]


def margin(scene: dict) -> float:
    """Smallest visible length separation that the rank ordering relies on."""
    gaps = []
    for slots in (TOP_SLOTS, BOTTOM_SLOTS):
        values = sorted(int(scene["lengths"][s]) for s in slots)
        gaps.extend(b - a for a, b in zip(values, values[1:]))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every scene field is drawn into the raster."""
    return []
