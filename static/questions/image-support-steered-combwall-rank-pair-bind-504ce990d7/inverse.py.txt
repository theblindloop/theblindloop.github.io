"""Pixel-only inverse arm for the comb-wall rank-partner world.

Receives a rendered PNG and the public question contract only. It segments the
ten teeth, checks that each is one solid edge-anchored rectangle, recovers both
length orders, locates the black key square, and only then reads the colour of
the bottom tooth holding the key tooth's length rank. Any structural anomaly
(missing or split tooth, wrong anchor, tied lengths inside a comb, absent or
ambiguous key square, off-contract colour) makes it abstain.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

BACKGROUND = (246, 242, 232)
MARKER = (16, 16, 16)
TOP_ANCHORS = {
    "red": (198, 42, 44),
    "blue": (38, 84, 190),
    "green": (38, 138, 64),
    "purple": (132, 58, 168),
    "orange": (233, 142, 26),
}
BOTTOM_ANCHORS = {
    "teal": (20, 146, 146),
    "yellow": (226, 198, 44),
    "pink": (232, 116, 176),
    "brown": (132, 88, 44),
    "slate": (84, 96, 120),
}

_LABELS = ["__bg__", "__marker__"] + list(TOP_ANCHORS) + list(BOTTOM_ANCHORS)
_ANCHOR_RGB = np.array(
    [BACKGROUND, MARKER] + list(TOP_ANCHORS.values()) + list(BOTTOM_ANCHORS.values()),
    dtype=np.int32,
)

MAX_ANCHOR_DISTANCE = 60.0
MIN_TOOTH_AREA = 4000


def _label_map(image: Image.Image) -> np.ndarray | None:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    diff = arr[:, :, None, :] - _ANCHOR_RGB[None, None, :, :]
    dist2 = (diff * diff).sum(axis=3)
    labels = dist2.argmin(axis=2)
    if np.sqrt(dist2.min(axis=2)).max() > MAX_ANCHOR_DISTANCE:
        return None  # off-contract ink: corrupted or ablated raster
    return labels


def _solid_rect(mask: np.ndarray, marker: np.ndarray) -> tuple[int, int, int, int] | None:
    rows = np.flatnonzero(mask.any(axis=1))
    cols = np.flatnonzero(mask.any(axis=0))
    if rows.size == 0 or cols.size == 0:
        return None
    y0, y1 = int(rows[0]), int(rows[-1])
    x0, x1 = int(cols[0]), int(cols[-1])
    window = mask[y0 : y1 + 1, x0 : x1 + 1] | marker[y0 : y1 + 1, x0 : x1 + 1]
    if not window.all():
        return None  # tooth is not one solid rectangle
    return x0, y0, x1, y1


def decision_from_image(image: Image.Image) -> str:
    labels = _label_map(image)
    if labels is None:
        return ABSTAIN
    height, width = labels.shape
    marker = labels == 1

    boxes: dict[str, tuple[int, int, int, int]] = {}
    for index, name in enumerate(_LABELS):
        if index < 2:
            continue
        mask = labels == index
        if int(mask.sum()) < MIN_TOOTH_AREA:
            continue
        box = _solid_rect(mask, marker)
        if box is None:
            return ABSTAIN
        boxes[name] = box

    top: dict[str, int] = {}
    bottom: dict[str, int] = {}
    for name, (_x0, y0, _x1, y1) in boxes.items():
        length = y1 - y0 + 1
        if name in TOP_ANCHORS:
            if y0 != 0 or y1 >= height - 1:
                return ABSTAIN  # top tooth must hang from the top edge only
            top[name] = length
        else:
            if y1 != height - 1 or y0 <= 0:
                return ABSTAIN
            bottom[name] = length
    if len(top) != 5 or len(bottom) != 5:
        return ABSTAIN
    if len(set(top.values())) != 5 or len(set(bottom.values())) != 5:
        return ABSTAIN

    marker_rows = np.flatnonzero(marker.any(axis=1))
    marker_cols = np.flatnonzero(marker.any(axis=0))
    if marker_rows.size == 0 or marker_cols.size == 0:
        return ABSTAIN
    my0, my1 = int(marker_rows[0]), int(marker_rows[-1])
    mx0, mx1 = int(marker_cols[0]), int(marker_cols[-1])
    block = marker[my0 : my1 + 1, mx0 : mx1 + 1]
    if not block.all() or int(marker.sum()) != block.size:
        return ABSTAIN  # key square must be one solid block

    holders = [
        name
        for name, (x0, y0, x1, y1) in boxes.items()
        if name in TOP_ANCHORS and x0 <= mx0 and mx1 <= x1 and y0 <= my0 and my1 <= y1
    ]
    if len(holders) != 1:
        return ABSTAIN

    top_order = sorted(top, key=lambda n: -top[n])
    bottom_order = sorted(bottom, key=lambda n: -bottom[n])
    if width < 10 or height < 10:
        return ABSTAIN
    rank = top_order.index(holders[0])
    return bottom_order[rank]
