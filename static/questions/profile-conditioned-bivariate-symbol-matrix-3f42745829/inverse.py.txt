"""Scene-free inverse arm: recover normalized relation signatures from pixels only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((213, 75, 75), dtype=np.int16),
    "blue": np.array((57, 118, 197), dtype=np.int16),
    "gold": np.array((214, 161, 29), dtype=np.int16),
    "teal": np.array((0, 140, 145), dtype=np.int16),
    "dark": np.array((38, 50, 56), dtype=np.int16),
}
ROWS = ("red", "blue", "gold")
SHAPES = ("circle", "triangle", "square")


def _mask(arr: np.ndarray, name: str, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - PALETTE[name]), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    """Return area, bbox and centroid for 8-connected true-pixel components."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = 0
        minx = maxx = sx
        miny = maxy = sy
        sumx = sumy = 0
        while queue:
            y, x = queue.popleft()
            area += 1
            sumx += x
            sumy += y
            minx = min(minx, x)
            maxx = max(maxx, x)
            miny = min(miny, y)
            maxy = max(maxy, y)
            for ny in range(max(0, y - 1), min(h, y + 2)):
                for nx in range(max(0, x - 1), min(w, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append((area, minx, miny, maxx, maxy, sumx / area, sumy / area))
    return output


def _large_border(components: list[tuple], min_y: int = 0) -> tuple[int, int, int, int]:
    candidates = [c for c in components if c[3] - c[1] > 240 and c[4] - c[2] > 210 and c[2] >= min_y]
    if len(candidates) != 1:
        raise ValueError("expected one large card border")
    c = candidates[0]
    return c[1], c[2], c[3], c[4]


def _inside(c: tuple, box: tuple[int, int, int, int]) -> bool:
    _, _, _, _, _, cx, cy = c
    x0, y0, x1, y1 = box
    return x0 < cx < x1 and y0 < cy < y1


def _signature(box: tuple[int, int, int, int], components: dict[str, list[tuple]]) -> frozenset[tuple[str, str]]:
    x0, y0, x1, y1 = box
    width, height = x1 - x0, y1 - y0
    row_centers: dict[str, float] = {}
    for row in ROWS:
        pieces = [
            c for c in components[row] if _inside(c, box) and 150 <= c[0] <= 800
            and x0 + 0.07 * width < c[5] < x0 + 0.28 * width
            and y0 + 0.25 * height < c[6] < y0 + 0.95 * height
        ]
        if len(pieces) != 1:
            raise ValueError(f"could not recover {row} row label")
        row_centers[row] = pieces[0][6]

    header_components = [
        c for c in components["dark"] if _inside(c, box) and 120 <= c[0] <= 900
        and y0 + 0.06 * height < c[6] < y0 + 0.28 * height
        and x0 + 0.25 * width < c[5] < x0 + 0.94 * width
    ]
    if len(header_components) != 3:
        raise ValueError("could not recover three shape headers")
    # Header ink areas are intentionally separated: circle < triangle < square.
    by_area = sorted(header_components, key=lambda c: c[0])
    col_centers = {shape: comp[5] for shape, comp in zip(SHAPES, by_area)}

    marks = [
        c for c in components["teal"] if _inside(c, box) and 180 <= c[0] <= 500
        and y0 + 0.27 * height < c[6] < y0 + 0.96 * height
    ]
    if len(marks) != 5:
        raise ValueError("every card must show exactly five teal marks")
    relation = set()
    for mark in marks:
        row = min(ROWS, key=lambda name: abs(mark[6] - row_centers[name]))
        shape = min(SHAPES, key=lambda name: abs(mark[5] - col_centers[name]))
        relation.add((row, shape))
    if len(relation) != 5:
        raise ValueError("two marks mapped to one matrix cell")
    return frozenset(relation)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 500 or arr.shape[1] < 800:
        raise ValueError("unexpected raster")
    masks = {name: _mask(arr, name) for name in PALETTE}
    components = {name: _components(mask) for name, mask in masks.items()}
    # A large dark outline is the reference; the three large colored outlines are candidates.
    reference_box = _large_border(components["dark"])
    option_boxes = {name: _large_border(components[name], min_y=250) for name in ROWS}
    reference = _signature(reference_box, components)
    matches = [name for name, box in option_boxes.items() if _signature(box, components) == reference]
    if len(matches) != 1:
        raise ValueError("pixel evidence does not identify one matching option")
    return matches[0]
