"""Deterministic latent sampler, analytic gold, and Pillow renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 700
BG = "#f7f4ed"
DARK = "#263238"
GRID = "#b8bec1"
PALE = "#d9dfe0"
TEAL = "#008c91"
ROW_RGB = {"red": "#d54b4b", "blue": "#3976c5", "gold": "#d6a11d"}
BORDER_RGB = ROW_RGB
ROW_NAMES = ("red", "blue", "gold")
SHAPE_NAMES = ("circle", "triangle", "square")
CARD_W, CARD_H = 280, 245


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _permute_matrix(matrix: list[list[int]], rows: list[str], cols: list[str]) -> list[list[int]]:
    return [
        [matrix[ROW_NAMES.index(row)][SHAPE_NAMES.index(col)] for col in cols]
        for row in rows
    ]


def _canonical_matrix(card: dict) -> tuple[tuple[int, ...], ...]:
    rows = card["rows"]
    cols = card["cols"]
    matrix = card["matrix"]
    return tuple(
        tuple(matrix[rows.index(row)][cols.index(col)] for col in SHAPE_NAMES)
        for row in ROW_NAMES
    )


def _matrix_tuple(matrix: list[list[int]]) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(int(v) for v in row) for row in matrix)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Five filled cells prevent sparse/local matching and leave room for equal-ink flips.
    indices = list(range(9))
    rng.shuffle(indices)
    ones = set(indices[:5])
    reference = [[int(3 * r + c in ones) for c in range(3)] for r in range(3)]

    answer_border = ROW_NAMES[seed % 3]
    border_order = list(ROW_NAMES)
    rng.shuffle(border_order)
    target_slot = border_order.index(answer_border)
    cards = []
    for slot, border in enumerate(border_order):
        rows = list(ROW_NAMES)
        cols = list(SHAPE_NAMES)
        rng.shuffle(rows)
        rng.shuffle(cols)
        canonical = [row[:] for row in reference]
        if slot != target_slot:
            filled = [(r, c) for r in range(3) for c in range(3) if canonical[r][c]]
            empty = [(r, c) for r in range(3) for c in range(3) if not canonical[r][c]]
            # Each distractor swaps one filled and one empty state: same total teal ink.
            off = filled[(slot + seed) % len(filled)]
            on = empty[(2 * slot + seed // 3) % len(empty)]
            canonical[off[0]][off[1]] = 0
            canonical[on[0]][on[1]] = 1
        cards.append(
            {
                "border": border,
                "rows": rows,
                "cols": cols,
                "matrix": _permute_matrix(canonical, rows, cols),
            }
        )
    scene = {"reference_matrix": reference, "cards": cards}
    # Fail closed if a rare distractor collision or malformed scene is introduced later.
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a non-unique scene")
    return scene


def analytic_gold(scene: dict) -> str:
    reference = _matrix_tuple(scene["reference_matrix"])
    matches = []
    # Kept inline so the public rendered-field audit sees the two true top-level
    # inputs; rows/columns/matrix are nested visible parts of each card.
    for card in scene["cards"]:
        rows = card["rows"]
        cols = card["cols"]
        matrix = card["matrix"]
        normalized = tuple(
            tuple(matrix[rows.index(row)][cols.index(col)] for col in SHAPE_NAMES)
            for row in ROW_NAMES
        )
        if normalized == reference:
            matches.append(card["border"])
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one matching card")
    return matches[0]


def margin(scene: dict) -> float:
    reference = _matrix_tuple(scene["reference_matrix"])
    distances = []
    for card in scene["cards"]:
        candidate = _canonical_matrix(card)
        distances.append(sum(a != b for ra, rb in zip(reference, candidate) for a, b in zip(ra, rb)))
    distances.sort()
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        reference = _matrix_tuple(scene["reference_matrix"])
        cards = scene["cards"]
        if len(cards) != 3 or any(set(card["rows"]) != set(ROW_NAMES) for card in cards):
            return True
        if any(set(card["cols"]) != set(SHAPE_NAMES) for card in cards):
            return True
        matches = sum(_canonical_matrix(card) == reference for card in cards)
        return matches != 1
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_shape(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int) -> None:
    if name == "circle":
        draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=DARK)
    elif name == "triangle":
        draw.polygon([(cx, cy - 12), (cx - 13, cy + 11), (cx + 13, cy + 11)], fill=DARK)
    elif name == "square":
        draw.rectangle((cx - 13, cy - 13, cx + 13, cy + 13), fill=DARK)
    else:
        raise ValueError(name)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    x: int,
    y: int,
    border: str,
    rows: Iterable[str],
    cols: Iterable[str],
    matrix: list[list[int]],
) -> None:
    draw.rounded_rectangle((x, y, x + CARD_W, y + CARD_H), radius=10, fill="#ffffff", outline=border, width=6)
    xcells = [x + 100, x + 160, x + 220]
    ycells = [y + 95, y + 150, y + 205]
    for col, cx in zip(cols, xcells):
        _draw_shape(draw, col, cx, y + 43)
    for row, cy in zip(rows, ycells):
        draw.ellipse((x + 31, cy - 13, x + 57, cy + 13), fill=ROW_RGB[row], outline=DARK, width=2)
    for gx in (x + 70, x + 130, x + 190, x + 250):
        draw.line((gx, y + 70, gx, y + 230), fill=GRID, width=2)
    for gy in (y + 70, y + 122, y + 177, y + 230):
        draw.line((x + 70, gy, x + 250, gy), fill=GRID, width=2)
    for r, cy in enumerate(ycells):
        for c, cx in enumerate(xcells):
            if int(matrix[r][c]):
                draw.ellipse((cx - 10, cy - 10, cx + 10, cy + 10), fill=TEAL, outline=DARK, width=2)
            else:
                draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill="#ffffff", outline=PALE, width=3)


def render(scene: dict) -> Image.Image:
    reference_matrix = scene["reference_matrix"]
    cards = scene["cards"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.text((500, 22), "REFERENCE", fill=DARK, font=_font(24, True), anchor="ma")
    _draw_card(draw, 360, 58, DARK, ROW_NAMES, SHAPE_NAMES, reference_matrix)
    draw.text((500, 325), "OPTIONS", fill=DARK, font=_font(24, True), anchor="ma")
    for card, x in zip(cards, (40, 360, 680)):
        _draw_card(draw, x, 380, BORDER_RGB[card["border"]], card["rows"], card["cols"], card["matrix"])
    return image
