from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

N = 12


def _templates() -> dict[str, np.ndarray]:
    y, x = np.mgrid[:N, :N]
    c = 5.5
    plus = ((abs(x - c) <= 0.5) & (y >= 1) & (y <= 10)) | ((abs(y - c) <= 0.5) & (x >= 1) & (x <= 10))
    circle = abs(np.hypot(x - c, y - c) - 4.6) <= 0.75
    diag = (x == y) | (x + y == 11)
    extra_rows = np.isin(y, [2, 3, 4, 7, 8, 9])
    extra = extra_rows & ((abs(x - y) == 1) | (abs((x + y) - 11) == 1))
    extra &= ((x < c) == (y < c))
    xmark = diag | extra
    manhattan = abs(x - c) + abs(y - c)
    diamond = (manhattan == 6) | ((manhattan == 5) & (np.maximum(abs(x-c), abs(y-c)) <= 3.5))
    return {"circle": circle, "diamond": diamond, "plus": plus, "x": xmark}


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    comps = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x); ys.append(y)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    queue.append((ny,nx))
        comps.append((len(xs), min(xs), min(ys), max(xs), max(ys), float(np.mean(xs)), float(np.mean(ys))))
    return comps


def _recover(image: Image.Image) -> tuple[np.ndarray, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    accent = (chroma > 55) & (arr.max(axis=2) < 235) & (arr.min(axis=2) < 180)
    comps = _components(accent)
    frames = sorted([c for c in comps if c[0] > 1000], key=lambda c: c[5])
    if len(frames) != 2:
        raise ValueError(f"expected two colored frames, found {len(frames)}")
    markers = [c for c in comps if 80 <= c[0] <= 400]
    if not markers:
        raise ValueError("missing colored start triangle")
    marker = max(markers, key=lambda c: c[0])
    start = int(np.argmin([abs(marker[5] - frame[5]) for frame in frames]))
    panel_masks = []
    for frame in frames:
        _, xmin, ymin, xmax, ymax, _, _ = frame
        side = xmax - xmin + 1
        grid = side - 8
        if grid <= 0 or grid % N:
            raise ValueError("frame does not enclose a 12 by 12 grid")
        cell = grid // N
        px, py = xmin + 4, ymin + 4
        cells = np.zeros((N, N), dtype=bool)
        gray = arr.mean(axis=2)
        for row in range(N):
            for col in range(N):
                y0, y1 = py + row * cell + 2, py + (row + 1) * cell - 2
                x0, x1 = px + col * cell + 2, px + (col + 1) * cell - 2
                cells[row, col] = float(gray[y0:y1, x0:x1].mean()) < 105.0
        panel_masks.append(cells)
    recovered = np.zeros((N, N), dtype=bool)
    for row in range(N):
        source = start if row % 2 == 0 else 1 - start
        recovered[row] = panel_masks[source][row]
    distances = sorted((int(np.count_nonzero(recovered != mask)), name) for name, mask in TEMPLATES.items())
    return recovered, float(distances[1][0] - distances[0][0])


def decision_from_image(image: Image.Image) -> str:
    recovered, gap = _recover(image)
    ranked = sorted((int(np.count_nonzero(recovered != mask)), name) for name, mask in TEMPLATES.items())
    if gap < 8 or ranked[0][0] > 3:
        raise ValueError("ambiguous or corrupted row weave")
    return ranked[0][1]
