from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("circle", "diamond", "plus", "x")
ACCENTS = ("#168aad", "#7b2cbf", "#168f6a", "#d05a18")
N = 12


def _templates() -> dict[str, np.ndarray]:
    y, x = np.mgrid[:N, :N]
    c = 5.5
    plus = ((abs(x - c) <= 0.5) & (y >= 1) & (y <= 10)) | (
        (abs(y - c) <= 0.5) & (x >= 1) & (x <= 10)
    )
    circle = abs(np.hypot(x - c, y - c) - 4.6) <= 0.75
    diag = (x == y) | (x + y == 11)
    extra_rows = np.isin(y, [2, 3, 4, 7, 8, 9])
    extra = extra_rows & ((abs(x - y) == 1) | (abs((x + y) - 11) == 1))
    # One symmetric adjacent cell per arm row keeps the X at 36 cells.
    extra &= ((x < c) == (y < c))
    xmark = diag | extra
    manhattan = abs(x - c) + abs(y - c)
    diamond = (manhattan == 6) | ((manhattan == 5) & (np.maximum(abs(x-c), abs(y-c)) <= 3.5))
    result = {"circle": circle, "diamond": diamond, "plus": plus, "x": xmark}
    if any(int(mask.sum()) != 36 for mask in result.values()):
        raise RuntimeError({name: int(mask.sum()) for name, mask in result.items()})
    return result


TEMPLATES = _templates()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    decoys = [name for name in SYMBOLS if name != symbol]
    return {
        "symbol": symbol,
        "decoy": rng.choice(decoys),
        "start_panel": rng.randrange(2),
        "cell_size": rng.choice([10, 11, 12]),
        "offset_x": rng.randint(-18, 18),
        "offset_y": rng.randint(-12, 12),
        "accent": rng.choice(ACCENTS),
    }


def _panel_masks(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    target = TEMPLATES[scene["symbol"]]
    decoy = TEMPLATES[scene["decoy"]]
    panels = [np.zeros_like(target), np.zeros_like(target)]
    start = int(scene["start_panel"])
    for row in range(N):
        chosen = start if row % 2 == 0 else 1 - start
        panels[chosen][row] = target[row]
        panels[1 - chosen][row] = decoy[row]
    return panels[0], panels[1]


def render(scene: dict) -> Image.Image:
    # Explicit reads make the complete visible latent surface auditable.
    cell = int(scene["cell_size"])
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    accent = scene["accent"]
    start = int(scene["start_panel"])
    _ = scene["symbol"], scene["decoy"]
    grid = N * cell
    gap = 62
    total = 2 * grid + gap
    left = (640 - total) // 2 + ox
    top = (420 - grid) // 2 + 18 + oy
    origins = [(left, top), (left + grid + gap, top)]
    image = Image.new("RGB", (640, 420), "#f8f7f2")
    draw = ImageDraw.Draw(image)
    masks = _panel_masks(scene)
    for (px, py), mask in zip(origins, masks):
        draw.rectangle([px - 4, py - 4, px + grid + 3, py + grid + 3], outline=accent, width=4)
        for k in range(1, N):
            q = k * cell
            draw.line([px + q, py, px + q, py + grid - 1], fill="#deddd7", width=1)
            draw.line([px, py + q, px + grid - 1, py + q], fill="#deddd7", width=1)
        for row, col in np.argwhere(mask):
            x0, y0 = px + int(col) * cell + 2, py + int(row) * cell + 2
            draw.rectangle([x0, y0, x0 + cell - 5, y0 + cell - 5], fill="#20242b")
    mx = origins[start][0] + grid // 2
    draw.polygon([(mx, top - 8), (mx - 12, top - 25), (mx + 12, top - 25)], fill=accent)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def margin(scene: dict) -> float:
    target = TEMPLATES[scene["symbol"]]
    distances = sorted(int(np.count_nonzero(target != other)) for other in TEMPLATES.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
