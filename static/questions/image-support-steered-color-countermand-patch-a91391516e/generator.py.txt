"""Deterministic Pillow world for an order-sensitive cancellation counter."""

from __future__ import annotations

import random
from collections import Counter
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
CELL = 72
GRID = 3
BOARD = CELL * GRID
COLORS = ("cyan", "coral", "violet")
PALETTES = (
    {"cyan": (35, 169, 187), "coral": (232, 102, 91), "violet": (119, 87, 184)},
    {"cyan": (27, 160, 180), "coral": (225, 94, 84), "violet": (112, 80, 176)},
    {"cyan": (43, 177, 194), "coral": (239, 110, 98), "violet": (127, 95, 191)},
)
BACKGROUND = (248, 246, 240)
GRID_INK = (42, 48, 58)
GOLD = (241, 185, 48)


def _replay(tiles: list[str]) -> tuple[str, int]:
    held = tiles[0]
    balance = 1
    for color in tiles[1:]:
        if color == held:
            balance += 1
        else:
            balance -= 1
            if balance == 0:
                held = color
                balance = 1
    return held, balance


def sample_scene(seed: int) -> dict:
    """Sample equal-ink sequences while cycling the requested final class."""
    rng = random.Random(int(seed))
    wanted = COLORS[int(seed) % len(COLORS)]
    base = [color for color in COLORS for _ in range(3)]
    for _ in range(20000):
        tiles = base[:]
        rng.shuffle(tiles)
        held, _ = _replay(tiles)
        if held == wanted:
            break
    else:
        raise RuntimeError("could not sample requested final color")
    return {
        "tiles": tiles,
        "offset_x": rng.randint(-18, 18),
        "offset_y": rng.randint(-14, 14),
        "shade_variant": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    tiles = list(scene["tiles"])
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    shade = int(scene["shade_variant"])
    palette = PALETTES[shade]
    x0 = (WIDTH - BOARD) // 2 + ox
    y0 = (HEIGHT - BOARD) // 2 + oy
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((x0 - 10, y0 - 10, x0 + BOARD + 9, y0 + BOARD + 9), radius=14, fill=GRID_INK)
    for index, color in enumerate(tiles):
        row, col = divmod(index, GRID)
        left = x0 + col * CELL + 6
        top = y0 + row * CELL + 6
        right = x0 + (col + 1) * CELL - 7
        bottom = y0 + (row + 1) * CELL - 7
        draw.rounded_rectangle((left, top, right, bottom), radius=8, fill=palette[color])
    # The outline is fully visible and identifies the initial tile from pixels.
    draw.rounded_rectangle((x0 + 2, y0 + 2, x0 + CELL - 3, y0 + CELL - 3), radius=11, outline=GOLD, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(list(scene["tiles"]))[0]


def margin(scene: dict) -> float:
    # Runner-up decision margin: unmatched votes still backing the held candidate.
    return float(_replay(list(scene["tiles"]))[1])


def is_quarantined(scene: dict) -> bool:
    tiles = list(scene["tiles"])
    return (
        len(tiles) != 9
        or set(tiles) != set(COLORS)
        or any(count != 3 for count in Counter(tiles).values())
        or int(scene["shade_variant"]) not in range(len(PALETTES))
        or abs(int(scene["offset_x"])) > 18
        or abs(int(scene["offset_y"])) > 14
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All declared latent fields affect pixels; no render-invariant aliases exist.
    return []


def translated(scene: dict, dx: int, dy: int) -> dict:
    """Public label-preserving transform used by tests, not by analytic gold."""
    twin = deepcopy(scene)
    twin["offset_x"] = int(twin["offset_x"]) + dx
    twin["offset_y"] = int(twin["offset_y"]) + dy
    return twin


def reshaded(scene: dict, variant: int) -> dict:
    """Public label-preserving palette transform used by tests."""
    twin = deepcopy(scene)
    twin["shade_variant"] = int(variant)
    return twin
