"""Independent pixel-only inverse for Color Countermand Patch.

This module knows only the public visual contract and receives one raster.  It
segments tile interiors by palette family, reconstructs the translated grid,
checks the gold START mark, and replays the countermand rule.
"""

from __future__ import annotations

from collections import deque

import numpy as np

ANCHORS = {
    "cyan": np.array((35, 169, 187), dtype=np.int16),
    "coral": np.array((232, 102, 91), dtype=np.int16),
    "violet": np.array((119, 87, 184), dtype=np.int16),
}
GOLD = np.array((241, 185, 48), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) < 100:
            continue
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        output.append((len(points), float(sum(xs) / len(xs)), float(sum(ys) / len(ys)), max(xs) - min(xs) + 1, max(ys) - min(ys) + 1))
    return output


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("RGB image required")
    labels = []
    for name, anchor in ANCHORS.items():
        distance = np.linalg.norm(array - anchor, axis=2)
        for area, cx, cy, bw, bh in _components(distance < 38.0):
            if area >= 2300 and 49 <= bw <= 62 and 49 <= bh <= 62:
                labels.append((cy, cx, name, area))
    if len(labels) != 9:
        raise ValueError(f"expected nine intact tile interiors, found {len(labels)}")
    # Cluster into three rows by centroid. Translation is recovered, never assumed.
    labels.sort()
    rows = [sorted(labels[start:start + 3], key=lambda item: item[1]) for start in (0, 3, 6)]
    if any(max(item[0] for item in row) - min(item[0] for item in row) > 5 for row in rows):
        raise ValueError("tile row alignment is ambiguous")
    if min(rows[i + 1][0][0] - rows[i][0][0] for i in range(2)) < 60:
        raise ValueError("tile rows are unresolved")
    ordered = [item for row in rows for item in row]
    if any(ordered[i + 1][1] - ordered[i][1] < 60 for row in rows for i in range(rows.index(row) * 3, rows.index(row) * 3 + 2)):
        raise ValueError("tile columns are unresolved")
    # Detect and bind the unique gold outline to the row-order first tile.
    gold_mask = np.linalg.norm(array - GOLD, axis=2) < 35.0
    gold_points = np.argwhere(gold_mask)
    if len(gold_points) < 700:
        raise ValueError("START outline is missing")
    gy, gx = gold_points.mean(axis=0)
    if abs(gx - ordered[0][1]) > 8 or abs(gy - ordered[0][0]) > 8:
        raise ValueError("START outline does not mark the first tile")
    held = ordered[0][2]
    balance = 1
    for _, _, color, _ in ordered[1:]:
        if color == held:
            balance += 1
        else:
            balance -= 1
            if balance == 0:
                held = color
                balance = 1
    return held
