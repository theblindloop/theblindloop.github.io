"""Pixel-only inverse arm.

Receives the final PNG and nothing else: no scene, no renderer, no gold. It
re-finds the three role objects by colour-role rules (yellow disk, neutral-grey
shape, brown vertical bar), takes their horizontal centres of mass, and reports
whether the grey shape and the yellow disk sit on the same side of the bar.
Abstains when a role is missing or when either horizontal displacement is too
small to carry a side.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "shadow_sun_side-oracle-0.1.0"
ABSTAIN = "uncertain"

MIN_ROLE_PIXELS = 400
MIN_OFFSET_PX = 30.0


def _roles(arr: np.ndarray) -> tuple:
    r = arr[..., 0]
    g = arr[..., 1]
    b = arr[..., 2]
    mx = arr.max(axis=-1)
    mn = arr.min(axis=-1)
    sun = (r >= 225) & (g >= 140) & (g <= 238) & (b <= 118)
    shadow = ((mx - mn) <= 12) & (mn >= 95) & (mx <= 185)
    trunk = ((r - g) >= 20) & ((g - b) >= 12) & (r >= 85) & (r <= 175) & (b <= 95)
    return sun, shadow, trunk


def _centre_x(mask: np.ndarray) -> float:
    xs = np.nonzero(mask.any(axis=0))[0]
    cols = mask.sum(axis=0).astype(np.float64)
    total = cols.sum()
    return float((cols * np.arange(mask.shape[1])).sum() / total) if total and len(xs) else float("nan")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    sun, shadow, trunk = _roles(arr)
    if min(int(sun.sum()), int(shadow.sum()), int(trunk.sum())) < MIN_ROLE_PIXELS:
        return ABSTAIN
    sun_x = _centre_x(sun)
    shadow_x = _centre_x(shadow)
    trunk_x = _centre_x(trunk)
    if not all(np.isfinite([sun_x, shadow_x, trunk_x])):
        return ABSTAIN
    d_sun = sun_x - trunk_x
    d_shadow = shadow_x - trunk_x
    if abs(d_sun) < MIN_OFFSET_PX or abs(d_shadow) < MIN_OFFSET_PX:
        return ABSTAIN
    return "same" if (d_sun > 0) == (d_shadow > 0) else "opposite"
