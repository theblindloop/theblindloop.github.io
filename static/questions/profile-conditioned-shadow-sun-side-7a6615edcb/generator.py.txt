"""Shadow-side world: which side of the trunk carries the painted shadow.

Latent scene z = page geometry + sun offset + signed shadow length + nuisance
style. Gold is the AGREEMENT of two signs measured from the same anchor:

    d_sun    = sun_x    - trunk_x
    d_shadow = shadow_centre_x - trunk_x
    gold     = "same" if sign(d_sun) == sign(d_shadow) else "opposite"

The physically canonical picture is "opposite" (a shadow falls away from the
light). Counterfactual pictures paint the shadow on the sun's own side. The
canonical prior is recorded for analysis only; gold is always the painted sign
agreement.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "shadow_sun_side-renderer-0.1.0"

WIDTH = 560
HEIGHT = 560
SS = 3  # supersampling factor

# Sampler floors: the two signed offsets that carry the answer never fall below
# these magnitudes, so the evidence is never sub-pixel.
MIN_SUN_OFFSET_PX = 118
MIN_SHADOW_LEN_PX = 118
QUARANTINE_MARGIN_PX = 66.0

# Role palettes. Every variant keeps the same colour ROLE separable by a simple
# hue rule (yellow sun, neutral-grey shadow, brown trunk), so the pixel arm never
# needs to know which variant was drawn.
PALETTES = (
    {"sun": (250, 205, 60), "shadow": (140, 140, 140), "trunk": (120, 74, 42),
     "crown": (46, 125, 50), "ground": (38, 92, 44), "frame": (150, 175, 215),
     "cloud": (198, 214, 240)},
    {"sun": (245, 170, 50), "shadow": (118, 118, 118), "trunk": (140, 90, 52),
     "crown": (60, 140, 66), "ground": (46, 104, 52), "frame": (162, 186, 222),
     "cloud": (206, 220, 244)},
    {"sun": (252, 222, 88), "shadow": (158, 158, 158), "trunk": (104, 64, 36),
     "crown": (34, 110, 44), "ground": (30, 80, 38), "frame": (140, 168, 210),
     "cloud": (190, 208, 236)},
)


def sample_scene(seed: int) -> dict:
    """Sample one scene. The two answer-carrying bits come from the seed itself:

    bit 0 -> which side of the trunk the sun is on;
    bit 1 -> whether the shadow shares that side.

    Both bits are independent of every geometric and stylistic field, so over any
    contiguous block of scene seeds the sun side, the shadow side and the answer
    are each balanced, and the answer is uncorrelated with either single side.
    """
    rng = random.Random(seed * 7919 + 13)
    sun_left = bool(seed & 1)
    same_side = bool((seed >> 1) & 1)

    ground_y = rng.randint(408, 432)
    trunk_x = rng.randint(245, 315)
    trunk_w = rng.randint(22, 30)
    trunk_h = rng.randint(148, 190)
    crown_r = rng.randint(56, 76)
    sun_r = rng.randint(26, 38)
    sun_y = rng.randint(66, 124)
    sun_sign = -1 if sun_left else 1
    sun_off = sun_sign * rng.randint(MIN_SUN_OFFSET_PX, 190)
    shadow_sign = sun_sign if same_side else -sun_sign
    shadow_len = shadow_sign * rng.randint(MIN_SHADOW_LEN_PX, 200)
    shadow_h = rng.randint(20, 30)
    palette = rng.randrange(len(PALETTES))

    clouds = []
    sun_x = trunk_x + sun_off
    for _ in range(rng.randint(1, 3)):
        for _try in range(40):
            cw = rng.randint(46, 92)
            ch = rng.randint(18, 30)
            cx = rng.randint(60, WIDTH - 60)
            cy = rng.randint(44, 168)
            if abs(cx - sun_x) < cw / 2 + sun_r + 18 and abs(cy - sun_y) < sun_r + 46:
                continue
            clouds.append([cx, cy, cw, ch])
            break

    return {
        "width": WIDTH, "height": HEIGHT, "ground_y": ground_y,
        "trunk_x": trunk_x, "trunk_w": trunk_w, "trunk_h": trunk_h,
        "crown_r": crown_r, "sun_r": sun_r, "sun_y": sun_y, "sun_off": sun_off,
        "shadow_len": shadow_len, "shadow_h": shadow_h, "palette": palette,
        "clouds": clouds,
    }


def analytic_gold(scene: dict) -> str:
    """Sign agreement of the two painted horizontal offsets."""
    sun_right = scene["sun_off"] > 0
    shadow_right = scene["shadow_len"] > 0
    return "same" if sun_right == shadow_right else "opposite"


def canonical_prior(scene: dict) -> str:
    """Recorded for analysis only; never used as gold."""
    return "opposite"


def margin(scene: dict) -> float:
    """Pixels of horizontal displacement carried by the weaker of the two cues.

    The sun contributes |sun_off|; the shadow contributes half its painted
    length, which is where its centre of mass sits relative to the trunk. When
    either shrinks to zero that object straddles the trunk and its side is no
    longer defined, so the smaller of the two is the whole decision margin.
    """
    return float(min(abs(scene["sun_off"]), abs(scene["shadow_len"]) / 2.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list:
    """No latent aliases: every field is painted and pixel-visible (see
    candidate.json latent_alias). Declared empty -> gate 9 not_applicable."""
    return []


def render(scene: dict) -> Image.Image:
    pal = PALETTES[int(scene["palette"])]
    w, h = int(scene["width"]), int(scene["height"])
    img = Image.new("RGB", (w * SS, h * SS), (255, 255, 255))
    d = ImageDraw.Draw(img)
    s = SS

    def box(x0, y0, x1, y1):
        return [x0 * s, y0 * s, x1 * s, y1 * s]

    d.rectangle(box(10, 10, w - 10, h - 10), outline=pal["frame"], width=5 * s)
    for cx, cy, cw, ch in scene["clouds"]:
        d.ellipse(box(cx - cw / 2, cy - ch / 2, cx + cw / 2, cy + ch / 2), fill=pal["cloud"])
        d.ellipse(box(cx - cw / 5, cy - ch, cx + cw / 3, cy + ch / 3), fill=pal["cloud"])

    gy = int(scene["ground_y"])
    sun_x = int(scene["trunk_x"]) + int(scene["sun_off"])
    sr = int(scene["sun_r"])
    d.ellipse(box(sun_x - sr, scene["sun_y"] - sr, sun_x + sr, scene["sun_y"] + sr),
              fill=pal["sun"])
    d.rectangle(box(16, gy - 3, w - 16, gy + 3), fill=pal["ground"])

    tx = int(scene["trunk_x"])
    sl = int(scene["shadow_len"])
    sh = int(scene["shadow_h"])
    cyy = gy + 14
    d.ellipse(box(min(tx, tx + sl), cyy - sh / 2, max(tx, tx + sl), cyy + sh / 2),
              fill=pal["shadow"])

    tw = int(scene["trunk_w"])
    th = int(scene["trunk_h"])
    d.rectangle(box(tx - tw / 2, gy - th, tx + tw / 2, gy + 2), fill=pal["trunk"])
    cr = int(scene["crown_r"])
    d.ellipse(box(tx - cr, gy - th - cr, tx + cr, gy - th + cr), fill=pal["crown"])

    return img.resize((w, h), Image.LANCZOS)


# ---------------------------------------------------------------------------
# Declared scene transforms (used by verify.py and tests/).
# ---------------------------------------------------------------------------

def translate(scene: dict, dx: int, dy: int) -> dict:
    out = dict(scene)
    out["trunk_x"] = int(scene["trunk_x"]) + dx
    out["ground_y"] = int(scene["ground_y"]) + dy
    out["sun_y"] = int(scene["sun_y"]) + dy
    out["clouds"] = [[c[0] + dx, c[1] + dy, c[2], c[3]] for c in scene["clouds"]]
    return out


def recolor(scene: dict) -> dict:
    out = dict(scene)
    out["palette"] = (int(scene["palette"]) + 1) % len(PALETTES)
    return out


def restyle(scene: dict) -> dict:
    """Nuisance resize of sun and shadow that keeps both signs."""
    out = dict(scene)
    out["sun_r"] = int(scene["sun_r"]) + 5
    out["shadow_h"] = int(scene["shadow_h"]) + 6
    sl = int(scene["shadow_len"])
    out["shadow_len"] = int(sl * 1.08) if abs(int(sl * 1.08)) >= MIN_SHADOW_LEN_PX else sl
    out["crown_r"] = int(scene["crown_r"]) - 6
    return out


def permute_clouds(scene: dict, key: int) -> dict:
    out = dict(scene)
    rng = random.Random(key)
    sun_x = int(scene["trunk_x"]) + int(scene["sun_off"])
    clouds = []
    for c in scene["clouds"]:
        for _try in range(60):
            cx = rng.randint(60, WIDTH - 60)
            cy = rng.randint(44, 168)
            if abs(cx - sun_x) < c[2] / 2 + int(scene["sun_r"]) + 18 and \
                    abs(cy - int(scene["sun_y"])) < int(scene["sun_r"]) + 46:
                continue
            clouds.append([cx, cy, c[2], c[3]])
            break
    out["clouds"] = clouds
    return out


def mirror_page(scene: dict) -> dict:
    """Whole-page left-right mirror. Both signs flip, so their AGREEMENT — the
    answer — is preserved. This is the sharpest contrast with a chirality world,
    where the same operation would invert the label."""
    out = dict(scene)
    w = int(scene["width"])
    out["trunk_x"] = w - int(scene["trunk_x"])
    out["sun_off"] = -int(scene["sun_off"])
    out["shadow_len"] = -int(scene["shadow_len"])
    out["clouds"] = [[w - c[0], c[1], c[2], c[3]] for c in scene["clouds"]]
    return out


def flip_shadow(scene: dict) -> dict:
    """Label-transforming: repaint the shadow on the other side of the trunk."""
    out = dict(scene)
    out["shadow_len"] = -int(scene["shadow_len"])
    return out


def flip_sun(scene: dict) -> dict:
    """Label-transforming: move the sun to the other side of the trunk."""
    out = dict(scene)
    out["sun_off"] = -int(scene["sun_off"])
    return out


def shrink_shadow(scene: dict, length: int) -> dict:
    """Collapse the shadow towards the trunk: the margin evidence disappears."""
    out = dict(scene)
    sign = 1 if int(scene["shadow_len"]) > 0 else -1
    out["shadow_len"] = sign * length
    return out
