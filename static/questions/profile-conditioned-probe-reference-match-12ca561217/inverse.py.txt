from __future__ import annotations

import numpy as np
from PIL import Image

REFERENCE_COLORS = {
    "orange": (232, 126, 38),
    "green": (40, 158, 104),
    "purple": (145, 86, 190),
}
TRACE_COLOR = (34, 96, 205)
PROBE_COLOR = (92, 98, 108)


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 8.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    probe_ys, probe_xs = np.nonzero(_mask(arr, PROBE_COLOR))
    if len(probe_xs) < 30:
        raise ValueError("vertical probe is missing")
    probe_x = float(np.median(probe_xs))

    trace_ys, trace_xs = np.nonzero(_mask(arr, TRACE_COLOR))
    local = np.abs(trace_xs.astype(float) - probe_x) <= 1.0
    if int(local.sum()) < 15:
        raise ValueError("trace cannot be measured at the probe")
    trace_y = float(np.median(trace_ys[local]))

    distances: dict[str, float] = {}
    for name, rgb in REFERENCE_COLORS.items():
        ys, xs = np.nonzero(_mask(arr, rgb))
        if len(xs) < 200:
            raise ValueError(f"{name} reference is missing")
        distances[name] = abs(float(np.median(ys)) - trace_y)
    ordered = sorted(distances, key=lambda name: (distances[name], name))
    if distances[ordered[1]] - distances[ordered[0]] < 1.0:
        raise ValueError("pixel decision is tied")
    return ordered[0]
