from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

REFERENCE_COLORS = {
    "orange": (232, 126, 38),
    "green": (40, 158, 104),
    "purple": (145, 86, 190),
}
TRACE_COLOR = (34, 96, 205)
PROBE_COLOR = (92, 98, 108)
BACKGROUND = (250, 251, 253)
INK = (34, 40, 48)
GRID = (222, 226, 232)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    width, height = 640, 440
    left = rng.randint(58, 78)
    top = rng.randint(48, 64)
    right = left + 520
    bottom = top + 310
    probe_x = rng.randint(left + 235, left + 285)

    levels = [top + rng.randint(55, 70), top + rng.randint(145, 165), top + rng.randint(240, 255)]
    names = list(REFERENCE_COLORS)
    rng.shuffle(names)
    reference_y = {name: float(level) for name, level in zip(names, levels)}

    desired = list(REFERENCE_COLORS)[int(seed) % 3]
    if int(seed) % 11 == 0:
        other = min((name for name in reference_y if name != desired), key=lambda n: abs(reference_y[n] - reference_y[desired]))
        midpoint = (reference_y[desired] + reference_y[other]) / 2.0
        direction = 1.0 if reference_y[desired] > midpoint else -1.0
        probe_y = midpoint + 3.0 * direction
    else:
        probe_y = reference_y[desired] + rng.uniform(-8.0, 8.0)

    xs = [left + 18, left + 112, probe_x - 68, probe_x, probe_x + 78, right - 18]
    ys = [rng.uniform(top + 24, bottom - 24) for _ in xs]
    ys[3] = probe_y
    trace_points = [[float(x), float(y)] for x, y in zip(xs, ys)]
    return {
        "width": width,
        "height": height,
        "plot_left": left,
        "plot_right": right,
        "plot_top": top,
        "plot_bottom": bottom,
        "probe_x": float(probe_x),
        "reference_y": reference_y,
        "trace_points": trace_points,
    }


def _trace_y_at_probe(scene: dict[str, Any]) -> float:
    probe_x = float(scene["probe_x"])
    points = [[float(v) for v in pair] for pair in scene["trace_points"]]
    for x, y in points:
        if abs(x - probe_x) < 1e-9:
            return y
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        if min(x0, x1) <= probe_x <= max(x0, x1):
            t = (probe_x - x0) / (x1 - x0)
            return y0 + t * (y1 - y0)
    raise ValueError("probe lies outside the trace")


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    left, right = int(scene["plot_left"]), int(scene["plot_right"])
    top, bottom = int(scene["plot_top"]), int(scene["plot_bottom"])
    probe_x = int(round(float(scene["probe_x"])))
    reference_y = scene["reference_y"]
    points = [(int(round(x)), int(round(y))) for x, y in scene["trace_points"]]

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((left, 18), "TRACE VALUE AT THE VERTICAL PROBE", fill=INK, font=font)
    for frac in (0.25, 0.5, 0.75):
        y = int(round(top + frac * (bottom - top)))
        draw.line((left, y, right, y), fill=GRID, width=1)
    draw.rectangle((left, top, right, bottom), outline=INK, width=2)

    for y0 in range(top + 5, bottom - 4, 15):
        draw.line((probe_x, y0, probe_x, min(y0 + 8, bottom)), fill=PROBE_COLOR, width=3)

    for name, rgb in REFERENCE_COLORS.items():
        y = int(round(float(reference_y[name])))
        draw.line((left + 2, y, right - 2, y), fill=rgb, width=5)

    draw.line(points, fill=TRACE_COLOR, width=7, joint="curve")
    py = int(round(_trace_y_at_probe(scene)))
    draw.ellipse((probe_x - 7, py - 7, probe_x + 7, py + 7), fill=TRACE_COLOR)

    draw.line((left, bottom, right, bottom), fill=INK, width=3)
    draw.line((left, top, left, bottom), fill=INK, width=3)
    draw.text((probe_x - 18, bottom + 10), "probe", fill=INK, font=font)
    draw.text((right - 35, bottom + 10), "x", fill=INK, font=font)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    trace_y = _trace_y_at_probe(scene)
    reference_y = scene["reference_y"]
    return min(reference_y, key=lambda name: (abs(float(reference_y[name]) - trace_y), name))


def margin(scene: dict[str, Any]) -> float:
    trace_y = _trace_y_at_probe(scene)
    distances = sorted(abs(float(y) - trace_y) for y in scene["reference_y"].values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
