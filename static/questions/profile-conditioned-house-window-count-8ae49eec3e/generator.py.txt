"""Deterministic raster renderer for the house-window count world.

The scene is deliberately ordinary: a small house illustration carries a
semantic prior, while the answer is the number of visibly separated square
openings.  The inverse arm does not share this module.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WINDOW_PALETTES = (
    (38, 106, 173),
    (28, 137, 132),
    (117, 82, 166),
    (187, 128, 38),
    (190, 83, 77),
)
WALL_PALETTES = (
    (219, 190, 145),
    (205, 211, 204),
    (222, 178, 151),
    (202, 194, 221),
)
ROOF_PALETTES = (
    (158, 70, 64),
    (73, 107, 135),
    (91, 116, 75),
    (126, 83, 133),
)
DOOR_PALETTES = (
    (128, 80, 48),
    (103, 91, 72),
    (97, 67, 93),
    (86, 106, 119),
)
BACKGROUND_PALETTES = (
    (248, 247, 242),
    (240, 246, 249),
    (248, 242, 249),
)
OUTLINE = (57, 61, 67)
LABEL_PALETTES = ((44, 48, 55), (68, 62, 77), (46, 75, 84))

# Eight stable facade slots.  Sampling a subset, rather than filling from a
# fixed side, removes a cheap position-to-answer cue.
BASE_SLOTS = (
    (-96.0, -36.0),
    (-32.0, -36.0),
    (32.0, -36.0),
    (96.0, -36.0),
    (-96.0, 46.0),
    (-32.0, 46.0),
    (32.0, 46.0),
    (96.0, 46.0),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one fully specified, JSON-serializable house scene."""

    rng = random.Random(int(seed))
    width, height = 640, 480
    count = rng.randint(2, 7)
    slots = list(rng.sample(BASE_SLOTS, count))
    scale = rng.uniform(0.93, 1.08)
    # A small minority is retained as an explicit finite-resolution boundary
    # case.  The opening is still measurable, but it is quarantined in claims.
    base_window = 17.0 if int(seed) % 17 == 0 else rng.uniform(27.0, 37.0)
    return {
        "width": width,
        "height": height,
        "window_count": count,
        "window_slots": [[float(x), float(y)] for x, y in slots],
        "center_x": rng.uniform(286.0, 354.0),
        "center_y": rng.uniform(252.0, 290.0),
        "scale": scale,
        "window_size": base_window,
        "window_color": list(rng.choice(WINDOW_PALETTES)),
        "wall_color": list(rng.choice(WALL_PALETTES)),
        "roof_color": list(rng.choice(ROOF_PALETTES)),
        "door_color": list(rng.choice(DOOR_PALETTES)),
        "outline_color": list(OUTLINE),
        "background": list(rng.choice(BACKGROUND_PALETTES)),
        "label_color": list(rng.choice(LABEL_PALETTES)),
        "label_text": "HOUSE",
        "antialias": bool(rng.random() < 0.82),
        "door_side": rng.choice((-1, 1)),
    }


def _point(x: float, y: float, ss: int) -> tuple[int, int]:
    return (int(round(x * ss)), int(round(y * ss)))


def _polygon(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], fill: tuple[int, ...], outline: tuple[int, ...], width: int, ss: int) -> None:
    draw.polygon([_point(x, y, ss) for x, y in points], fill=fill)
    if width > 0:
        draw.line([_point(x, y, ss) for x, y in [*points, points[0]]], fill=outline, width=max(1, width * ss), joint="curve")


def _rectangle(draw: ImageDraw.ImageDraw, box: tuple[float, float, float, float], fill: tuple[int, ...], outline: tuple[int, ...] | None, width: int, ss: int) -> None:
    scaled = tuple(int(round(value * ss)) for value in box)
    draw.rectangle(scaled, fill=fill, outline=outline, width=max(1, width * ss) if outline is not None else 1)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the scene to an RGB PNG-compatible image."""

    width, height = int(scene["width"]), int(scene["height"])
    ss = 3 if bool(scene["antialias"]) else 1
    background = tuple(scene["background"])
    outline = tuple(scene["outline_color"])
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)

    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    body_w = 306.0 * scale
    body_h = 207.0 * scale
    body_left = cx - body_w / 2.0
    body_right = cx + body_w / 2.0
    body_top = cy - 62.0 * scale
    body_bottom = body_top + body_h
    roof_peak = (cx, body_top - 112.0 * scale)
    roof_base_y = body_top + 7.0 * scale
    overhang = 18.0 * scale

    roof_points = [
        (body_left - overhang, roof_base_y),
        roof_peak,
        (body_right + overhang, roof_base_y),
    ]
    _polygon(
        draw,
        roof_points,
        tuple(scene["roof_color"]),
        outline,
        5,
        ss,
    )
    _rectangle(
        draw,
        (body_left, body_top, body_right, body_bottom),
        tuple(scene["wall_color"]),
        outline,
        5,
        ss,
    )

    # A side-swapped door is a nuisance factor.  It is deliberately not the
    # window colour and is tall rather than square for inverse separation.
    door_w = 56.0 * scale
    door_h = 75.0 * scale
    door_cx = cx + float(scene["door_side"]) * 52.0 * scale
    door_box = (
        door_cx - door_w / 2.0,
        body_bottom - door_h,
        door_cx + door_w / 2.0,
        body_bottom,
    )
    _rectangle(draw, door_box, tuple(scene["door_color"]), outline, 5, ss)
    knob_x = door_cx + (15.0 if scene["door_side"] < 0 else -15.0) * scale
    knob_y = body_bottom - door_h * 0.52
    draw.ellipse(
        (
            int(round((knob_x - 3.5 * scale) * ss)),
            int(round((knob_y - 3.5 * scale) * ss)),
            int(round((knob_x + 3.5 * scale) * ss)),
            int(round((knob_y + 3.5 * scale) * ss)),
        ),
        fill=outline,
    )

    # Window interiors are single solid square components inside dark frames;
    # no crossbars are drawn because the public rule counts each opening once.
    interior = float(scene["window_size"]) * scale
    frame = 5.0 * scale
    for raw_x, raw_y in scene["window_slots"][: int(scene["window_count"])]:
        wx = cx + float(raw_x) * scale
        wy = body_top + 58.0 * scale + float(raw_y) * scale
        outer_half = interior / 2.0 + frame
        _rectangle(
            draw,
            (wx - outer_half, wy - outer_half, wx + outer_half, wy + outer_half),
            outline,
            None,
            0,
            ss,
        )
        _rectangle(
            draw,
            (wx - interior / 2.0, wy - interior / 2.0, wx + interior / 2.0, wy + interior / 2.0),
            tuple(scene["window_color"]),
            None,
            0,
            ss,
        )

    # The label is intentionally small and fixed; it names the familiar
    # object but never carries a numeric answer.
    font = ImageFont.load_default()
    label = str(scene["label_text"])
    bbox = draw.textbbox((0, 0), label, font=font)
    label_w = bbox[2] - bbox[0]
    label_x = int(round((cx - label_w / (2.0 * ss)) * ss))
    label_y = int(round((body_bottom + 20.0 * scale) * ss))
    draw.text((label_x, label_y), label, fill=tuple(scene["label_color"]), font=font)

    if ss > 1:
        img = img.resize((width, height), Image.Resampling.LANCZOS)
    return img


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return only the visible opening count encoded by the scene."""

    return str(int(scene["window_count"]))


def margin(scene: dict[str, Any]) -> float:
    """One-sided finite-resolution clearance for the component count."""

    scale = float(scene["scale"])
    interior = float(scene["window_size"]) * scale
    horizontal_gap = 64.0 * scale - interior
    vertical_gap = 82.0 * scale - interior
    return round(min(interior - 16.0, horizontal_gap - 8.0, vertical_gap - 8.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases: all label inputs are directly visible in the raster."""

    return []
