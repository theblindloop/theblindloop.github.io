"""Independent PNG-only inverse for the house-window count.

This module intentionally contains its own colour segmentation and connected
component code.  It receives only the final RGB image and returns a constrained
numeric decision; no renderer state, scene dictionary, or metadata is read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_WINDOW_COLORS = np.array(
    [
        (38, 106, 173),
        (28, 137, 132),
        (117, 82, 166),
        (187, 128, 38),
        (190, 83, 77),
    ],
    dtype=np.float32,
)


def _window_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.stack(
        [np.linalg.norm(pixels - colour, axis=-1) for colour in _WINDOW_COLORS],
        axis=-1,
    )
    return distances.min(axis=-1) <= 58.0


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            queue = deque([(y, x)])
            seen[y, x] = True
            size = 0
            min_x = max_x = x
            min_y = max_y = y
            while queue:
                cy, cx = queue.popleft()
                size += 1
                min_x = min(min_x, cx)
                max_x = max(max_x, cx)
                min_y = min(min_y, cy)
                max_y = max(max_y, cy)
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
            found.append((size, min_x, min_y, max_x + 1, max_y + 1))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Count square, medium-sized window-colour components in the PNG alone."""

    mask = _window_mask(image)
    count = 0
    for size, left, top, right, bottom in _components(mask):
        width = right - left
        height = bottom - top
        if width < 8 or height < 8 or width > 48 or height > 48:
            continue
        ratio = width / float(height)
        fill = size / float(width * height)
        if 0.68 <= ratio <= 1.45 and fill >= 0.42 and size >= 80:
            count += 1
    if count not in {2, 3, 4, 5, 6, 7}:
        raise ValueError(f"raster contains {count} qualifying square openings; expected 2-7")
    return str(count)
