from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIZE = 512
BACKGROUND = (250, 249, 246)
NAVY = (31, 61, 89)
GREEN = (35, 170, 105)
RED = (220, 70, 75)
YELLOW = (245, 190, 45)
GRAY = (100, 105, 110)


def _nearest_segment(points, beacon):
    best = None
    bx, by = beacon
    for index, (a, b) in enumerate(zip(points, points[1:])):
        ax, ay = a
        ex, ey = b
        dx, dy = ex - ax, ey - ay
        length2 = dx * dx + dy * dy
        t = max(0.0, min(1.0, ((bx - ax) * dx + (by - ay) * dy) / length2))
        qx, qy = ax + t * dx, ay + t * dy
        distance = math.hypot(bx - qx, by - qy)
        candidate = (distance, index, t, qx, qy, dx, dy)
        if best is None or candidate < best:
            best = candidate
    return best


def _flank_signs(points, beacons):
    signs = []
    clearances = []
    for beacon in beacons:
        distance, _index, t, qx, qy, dx, dy = _nearest_segment(points, beacon)
        if not 0.18 <= t <= 0.82:
            raise ValueError("beacon is too close to a bend")
        cross = dx * (beacon[1] - qy) - dy * (beacon[0] - qx)
        signs.append(1 if cross > 0 else -1)
        clearances.append(distance)
    return signs, clearances


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    tx = rng.choice((-18, -9, 0, 9, 18))
    ty = rng.choice((-14, -7, 0, 7, 14))
    points = [
        [135 + tx, 155 + ty], [225 + tx, 155 + ty],
        [225 + tx, 215 + ty], [290 + tx, 215 + ty],
        [290 + tx, 275 + ty], [355 + tx, 275 + ty],
        [355 + tx, 335 + ty], [405 + tx, 335 + ty],
    ]
    if rng.randrange(2):
        points = [[SIZE - x, y] for x, y in points]
    balanced = int(seed) % 2 == 0
    if balanced:
        flank_signs = [1, 1, 1, -1, -1, -1]
    else:
        majority = rng.choice((-1, 1))
        flank_signs = [majority] * 4 + [-majority] * 2
    base_signs = flank_signs
    for _attempt in range(200):
        flank_signs = list(base_signs)
        rng.shuffle(flank_signs)
        beacons = []
        for segment_id, sign in enumerate(flank_signs):
            ax, ay = points[segment_id]
            bx, by = points[segment_id + 1]
            t = 0.5 if segment_id in (1, 3, 5) else rng.choice((0.44, 0.56))
            qx, qy = ax + t * (bx - ax), ay + t * (by - ay)
            dx, dy = bx - ax, by - ay
            length = math.hypot(dx, dy)
            nx, ny = -dy / length, dx / length
            offset = rng.choice((24, 26, 28))
            beacons.append([round(qx + sign * offset * nx), round(qy + sign * offset * ny)])
        separated = all(
            math.dist(beacons[i], beacons[j]) >= 22.0
            for i in range(6) for j in range(i + 1, 6)
        )
        correctly_bound = all(_nearest_segment(points, beacon)[1] == index for index, beacon in enumerate(beacons))
        if separated and correctly_bound:
            break
    else:
        raise RuntimeError("could not place six unambiguous beacons")
    return {"points": points, "beacons": beacons, "start": list(points[0]), "finish": list(points[-1])}


def render(scene: dict) -> Image.Image:
    points = [tuple(value) for value in scene["points"]]
    beacons = [tuple(value) for value in scene["beacons"]]
    start = tuple(scene["start"])
    finish = tuple(scene["finish"])
    image = Image.new("RGB", (SIZE, SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.line(points, fill=NAVY, width=9, joint="curve")
    for x, y in beacons:
        draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=YELLOW)
    for (x, y), color in ((start, GREEN), (finish, RED)):
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=color)
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=BACKGROUND)
    draw.text((start[0] - 29, start[1] - 34), "START", fill=GRAY)
    draw.text((finish[0] - 29, finish[1] + 19), "FINISH", fill=GRAY)
    return image


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    beacons = scene["beacons"]
    start = scene["start"]
    finish = scene["finish"]
    if list(points[0]) != list(start) or list(points[-1]) != list(finish):
        raise ValueError("declared endpoints disagree with route")
    signs, _ = _flank_signs(points, beacons)
    return "balanced" if signs.count(1) == signs.count(-1) == 3 else "unbalanced"


def margin(scene: dict) -> float:
    signs, clearances = _flank_signs(scene["points"], scene["beacons"])
    count_gap = abs(signs.count(1) - signs.count(-1))
    class_flip_margin = 1.0 if count_gap in (0, 2) else count_gap / 2.0
    return float(min(clearances) + class_flip_margin)


def is_quarantined(scene: dict) -> bool:
    try:
        signs, clearances = _flank_signs(scene["points"], scene["beacons"])
    except (ValueError, ZeroDivisionError):
        return True
    return len(signs) != 6 or min(clearances, default=0.0) < 20.0
