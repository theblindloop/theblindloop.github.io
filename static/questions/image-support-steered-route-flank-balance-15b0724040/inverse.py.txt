from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((31, 61, 89), dtype=np.int16)
GREEN = np.array((35, 170, 105), dtype=np.int16)
RED = np.array((220, 70, 75), dtype=np.int16)
YELLOW = np.array((245, 190, 45), dtype=np.int16)


def _mask(array, color, tolerance=10):
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask, minimum=20):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        group = []
        while stack:
            cy, cx = stack.pop()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(group) >= minimum:
            groups.append(np.asarray(group, dtype=np.int32))
    return groups


def _centroid(component):
    yx = component.mean(axis=0)
    return np.array((yx[1], yx[0]), dtype=np.float64)


def _route_distances(route, start_xy):
    height, width = route.shape
    coords_yx = np.argwhere(route)
    coords_xy = coords_yx[:, ::-1].astype(np.float64)
    start_index = int(np.argmin(np.sum((coords_xy - start_xy) ** 2, axis=1)))
    sy, sx = map(int, coords_yx[start_index])
    distances = np.full((height, width), -1, dtype=np.int32)
    distances[sy, sx] = 0
    queue = deque([(sy, sx)])
    while queue:
        y, x = queue.popleft()
        next_distance = int(distances[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and route[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = next_distance
                queue.append((ny, nx))
    return coords_yx, coords_xy, distances


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    route_groups = _components(_mask(array, NAVY), minimum=300)
    green_groups = _components(_mask(array, GREEN), minimum=80)
    red_groups = _components(_mask(array, RED), minimum=80)
    beacon_groups = _components(_mask(array, YELLOW), minimum=80)
    if len(route_groups) != 1 or len(green_groups) != 1 or len(red_groups) != 1:
        raise ValueError("route or endpoint segmentation is ambiguous")
    if len(beacon_groups) != 6:
        raise ValueError("expected six separated yellow beacons")
    route = _mask(array, NAVY)
    start_xy = _centroid(green_groups[0])
    finish_xy = _centroid(red_groups[0])
    coords_yx, coords_xy, distances = _route_distances(route, start_xy)
    finish_index = int(np.argmin(np.sum((coords_xy - finish_xy) ** 2, axis=1)))
    fy, fx = map(int, coords_yx[finish_index])
    if distances[fy, fx] < 0:
        raise ValueError("navy route is not continuous from start to finish")
    route_d = distances[coords_yx[:, 0], coords_yx[:, 1]]
    signs = []
    for group in beacon_groups:
        beacon_xy = _centroid(group)
        nearest = int(np.argmin(np.sum((coords_xy - beacon_xy) ** 2, axis=1)))
        ay, ax = map(int, coords_yx[nearest])
        anchor_d = int(distances[ay, ax])
        if anchor_d < 20:
            raise ValueError("beacon is too close to route start")
        before = coords_xy[(route_d >= anchor_d - 24) & (route_d <= anchor_d - 14)]
        after = coords_xy[(route_d >= anchor_d + 14) & (route_d <= anchor_d + 24)]
        local = coords_xy[(route_d >= anchor_d - 2) & (route_d <= anchor_d + 2)]
        if min(len(before), len(after), len(local)) < 3:
            raise ValueError("local directed tangent is not recoverable")
        p0, p1, anchor = before.mean(axis=0), after.mean(axis=0), local.mean(axis=0)
        tangent, offset = p1 - p0, beacon_xy - anchor
        cross = float(tangent[0] * offset[1] - tangent[1] * offset[0])
        normalized = abs(cross) / max(1.0, float(np.linalg.norm(tangent)))
        if normalized < 14.0:
            raise ValueError("beacon side is too close to the visible boundary")
        signs.append(1 if cross > 0 else -1)
    return "balanced" if signs.count(1) == signs.count(-1) == 3 else "unbalanced"
