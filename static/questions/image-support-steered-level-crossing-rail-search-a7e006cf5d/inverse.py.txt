"""Independent image-only inverse for the local level-crossing rail."""

from __future__ import annotations

import numpy as np
from PIL import Image

_CYAN = np.array((20, 151, 178), dtype=np.int16)
_PALETTE = {
    "red": np.array((220, 62, 67), dtype=np.int16),
    "gold": np.array((224, 158, 35), dtype=np.int16),
    "violet": np.array((126, 78, 181), dtype=np.int16),
    "green": np.array((41, 158, 103), dtype=np.int16),
}


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _center(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("required colored bead is absent")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    gates = []
    for name, color in _PALETTE.items():
        x, y = _center(_mask(arr, color))
        gates.append((x, y, name))
    gates.sort()
    gaps = np.diff([g[0] for g in gates])
    if len(gaps) != 3 or float(gaps.min()) < 20 or float(gaps.max() - gaps.min()) > 2.0:
        raise ValueError("four equally spaced candidate gates were not recovered")
    excursion_width = float(np.median(gaps))
    step_x = excursion_width / 8.0
    cyan = _mask(arr, _CYAN, tolerance=20)
    scored = []
    for gx, baseline, name in gates:
        signs = []
        for k in range(1, 8):
            expected_x = gx + k * step_x
            x0 = max(0, int(round(expected_x)) - 1)
            x1 = min(cyan.shape[1], int(round(expected_x)) + 2)
            ys, _ = np.nonzero(cyan[:, x0:x1])
            if len(ys) < 2:
                raise ValueError("cyan rail vertex is unresolved")
            center_y = float(np.median(ys))
            delta = center_y - baseline
            if abs(delta) < 2.0:
                raise ValueError("rail vertex ambiguously touches the reference level")
            signs.append(1 if delta > 0 else -1)
        crossings = sum(a != b for a, b in zip(signs, signs[1:]))
        scored.append((crossings, name))
    winners = [name for count, name in scored if count == 3]
    runner_gap = min((abs(count - 3) for count, _ in scored if count != 3), default=0)
    if len(winners) != 1 or runner_gap < 1:
        raise ValueError("no unique three-crossing excursion with runner-up margin")
    return winners[0]
