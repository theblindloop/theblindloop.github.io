"""Deterministic renderer for a compact, single-rail level-crossing search."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 1280
HEIGHT = 1280
CYAN = (20, 151, 178)
INK = (31, 41, 55)
PALETTE = {
    "red": (220, 62, 67),
    "gold": (224, 158, 35),
    "violet": (126, 78, 181),
    "green": (41, 158, 103),
}
GATE_NAMES = ("red", "gold", "violet", "green")
TARGET_CROSSINGS = 3


def _words_for_counts(rng: random.Random, counts: list[int]) -> list[list[int]]:
    words: list[list[int]] = []
    for count in counts:
        # Seven nonzero vertices create six possible strict sign transitions.
        signs = [rng.choice((-1, 1))]
        switches_left = count
        slots_left = 6
        for _ in range(6):
            switch = switches_left > 0 and (
                switches_left == slots_left or rng.randrange(slots_left) < switches_left
            )
            signs.append(-signs[-1] if switch else signs[-1])
            switches_left -= int(switch)
            slots_left -= 1
        words.append(signs)
    return words


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    distractor_pool = [0, 1, 2, 4, 5, 6]
    rng.shuffle(distractor_pool)
    counts = distractor_pool[:3]
    counts.insert(target_index, TARGET_CROSSINGS)
    gate_colors = list(GATE_NAMES)
    rng.shuffle(gate_colors)
    sign_words = _words_for_counts(rng, counts)
    amplitudes = [[rng.choice((1, 2, 3)) for _ in word] for word in sign_words]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "origin_x": 480 + rng.randrange(-180, 181),
        # Keep the short trace away from coarse-grid row boundaries while still
        # randomizing its absolute band and sub-cell position.
        "origin_y": rng.choice((400, 560, 720, 880)) + rng.randrange(-10, 11),
        "gate_colors": gate_colors,
        "sign_words": sign_words,
        "amplitudes": amplitudes,
        "step_x": 8,
        "step_y": 7,
        "rail_width": 3,
        "bead_radius": 7,
    }


def _crossing_counts(scene: dict) -> list[int]:
    counts = []
    for word in scene["sign_words"]:
        counts.append(sum(a != b for a, b in zip(word, word[1:])))
    return counts


def analytic_gold(scene: dict) -> str:
    counts = _crossing_counts(scene)
    winners = [i for i, value in enumerate(counts) if value == TARGET_CROSSINGS]
    if len(winners) != 1:
        raise ValueError("scene does not have one three-crossing excursion")
    return scene["gate_colors"][winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((-abs(value - TARGET_CROSSINGS) for value in _crossing_counts(scene)), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    counts = _crossing_counts(scene)
    return counts.count(TARGET_CROSSINGS) != 1 or margin(scene) < 1.0 or scene["step_y"] < 7


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    ox = int(scene["origin_x"])
    baseline = int(scene["origin_y"])
    gate_colors = list(scene["gate_colors"])
    sign_words = [list(word) for word in scene["sign_words"]]
    amplitudes = [list(values) for values in scene["amplitudes"]]
    step_x = int(scene["step_x"])
    step_y = int(scene["step_y"])
    rail_width = int(scene["rail_width"])
    bead_radius = int(scene["bead_radius"])

    image = Image.new("RGB", (width, height), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    points: list[tuple[int, int]] = [(ox, baseline)]
    gate_points: list[tuple[int, int]] = []
    x = ox
    for word, amps in zip(sign_words, amplitudes):
        gate_points.append((x, baseline))
        for sign, amp in zip(word, amps):
            x += step_x
            points.append((x, baseline + sign * amp * step_y))
        x += step_x
        points.append((x, baseline))

    # Nondiagnostic registration ticks emphasize that all four excursions share a level.
    for gx, gy in gate_points:
        draw.line((gx - 1, gy - 13, gx - 1, gy + 13), fill=(218, 216, 210), width=1)
    draw.line(points, fill=CYAN, width=rail_width, joint="curve")
    for (gx, gy), name in zip(gate_points, gate_colors):
        color = PALETTE[name]
        draw.ellipse((gx - bead_radius, gy - bead_radius, gx + bead_radius, gy + bead_radius), fill=color, outline=INK, width=2)
    draw.ellipse((x - 5, baseline - 5, x + 5, baseline + 5), fill=(235, 234, 229), outline=INK, width=2)
    return image
