"""Inverse arm: pixel-only recovery of the cut band that splits the plate.

Receives the rendered PNG and nothing else.  It segments paper, plate and the
three coloured bands by colour alone, then for each colour deletes that band
from the plate - the two other bands still count as plate, since only one cut is
made - and counts the surviving connected pieces.  The colour that leaves more
than one piece is the answer; anything else abstains.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

STRIDE = 4
COLOURS = ("red", "green", "blue")
MIN_PIECE = 40  # sampled lattice cells


def _classify(image: Image.Image) -> np.ndarray:
    """Label every sampled pixel: 0 paper, 1 plate, 2 red, 3 green, 4 blue."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)[::STRIDE, ::STRIDE, :]
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    hi = arr.max(axis=2)
    lo = arr.min(axis=2)
    out = np.zeros(arr.shape[:2], dtype=np.uint8)
    grey = (hi - lo) < 45
    out[grey & (hi >= 200)] = 0
    out[grey & (hi < 200)] = 1
    coloured = ~grey
    dom = np.argmax(arr, axis=2)
    out[coloured & (dom == 0)] = 2
    out[coloured & (dom == 1)] = 3
    out[coloured & (dom == 2)] = 4
    return out


def _piece_count(mask: np.ndarray) -> int:
    """Number of 4-connected components of ``mask`` above the size floor."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    pieces = 0
    for sy in range(h):
        row = mask[sy]
        for sx in range(w):
            if not row[sx] or seen[sy, sx]:
                continue
            size = 0
            queue = deque([(sy, sx)])
            seen[sy, sx] = True
            while queue:
                y, x = queue.popleft()
                size += 1
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
            if size >= MIN_PIECE:
                pieces += 1
    return pieces


def decision_from_image(image: Image.Image) -> str:
    labels = _classify(image)
    plate = labels == 1
    bands = {c: labels == (2 + i) for i, c in enumerate(COLOURS)}
    if plate.sum() < 200 or any(m.sum() < 20 for m in bands.values()):
        return "abstain"

    splitting = []
    for colour in COLOURS:
        region = plate.copy()
        for other in COLOURS:
            if other != colour:
                region |= bands[other]
        if _piece_count(region) > 1:
            splitting.append(colour)
    if len(splitting) != 1:
        return "abstain"
    return splitting[0]
