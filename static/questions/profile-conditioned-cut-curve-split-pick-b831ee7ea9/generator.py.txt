"""Forward arm: a holed grey plate crossed by three coloured cut bands.

Exactly one of the three bands, if the plate were cut along it, would fall the
plate into two separate pieces.  A band separates the plate exactly when its two
ends land on the same boundary of the plate (the outer rim counts as one
boundary and every hole counts as another); a band running from one boundary to
a different one leaves the plate in one piece.  Gold is not read off that rule -
it is recomputed by deleting each band from the plate and counting the connected
pieces that remain - but the rule is what makes the scenes constructible.
"""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

CELL = 16
COLS, ROWS = 40, 28
CANVAS = (COLS * CELL, ROWS * CELL)
PLATE = (2, 2, COLS - 3, ROWS - 3)  # inclusive cell bounds of the plate
COLORS = ("red", "green", "blue")
INK = {"red": (206, 58, 58), "green": (36, 146, 64), "blue": (48, 86, 204)}
PLATE_INK = (132, 132, 132)
PAPER = (255, 255, 255)

QUARANTINE_MARGIN = 10.0
SAMPLE_MARGIN = 15.0
MIN_PIECE_CELLS = 8
CUT_CLEARANCE = 3  # minimum Chebyshev cell distance between two different cuts

_STEPS4 = ((1, 0), (-1, 0), (0, 1), (0, -1))
_STEPS8 = tuple((dx, dy) for dx in (-1, 0, 1) for dy in (-1, 0, 1) if (dx, dy) != (0, 0))


# --------------------------------------------------------------------------- #
# cell geometry
# --------------------------------------------------------------------------- #
def _in_plate_rect(c) -> bool:
    x, y = c
    return PLATE[0] <= x <= PLATE[2] and PLATE[1] <= y <= PLATE[3]


def _hole_of(c, holes):
    for idx, (x0, y0, x1, y1) in enumerate(holes):
        if x0 <= c[0] <= x1 and y0 <= c[1] <= y1:
            return idx + 1
    return None


def _is_plate(c, holes) -> bool:
    return _in_plate_rect(c) and _hole_of(c, holes) is None


def _boundary_of(c, holes) -> int:
    """Boundary id of a NON-plate cell: 0 for the outer rim, 1+ per hole."""
    h = _hole_of(c, holes)
    return 0 if h is None else h


def _plate_cells(holes):
    return {
        (x, y)
        for x in range(PLATE[0], PLATE[2] + 1)
        for y in range(PLATE[1], PLATE[3] + 1)
        if _hole_of((x, y), holes) is None
    }


def _components(cells):
    """4-connected components of a set of cells."""
    remaining = set(cells)
    out = []
    while remaining:
        seed = remaining.pop()
        comp = {seed}
        queue = deque([seed])
        while queue:
            x, y = queue.popleft()
            for dx, dy in _STEPS4:
                n = (x + dx, y + dy)
                if n in remaining:
                    remaining.discard(n)
                    comp.add(n)
                    queue.append(n)
        out.append(comp)
    return out


def _deep(c, holes) -> bool:
    """A cell whose whole 3x3 neighbourhood is plate: safe for a band interior."""
    if not _is_plate(c, holes):
        return False
    return all(_is_plate((c[0] + dx, c[1] + dy), holes) for dx, dy in _STEPS8)


def _attach_cells(holes, boundary):
    """Plate cells flush against exactly one boundary, that boundary."""
    out = []
    for x in range(PLATE[0], PLATE[2] + 1):
        for y in range(PLATE[1], PLATE[3] + 1):
            c = (x, y)
            if not _is_plate(c, holes):
                continue
            touched = set()
            flush = False
            for dx, dy in _STEPS8:
                n = (x + dx, y + dy)
                if _is_plate(n, holes):
                    continue
                touched.add(_boundary_of(n, holes))
                if (dx, dy) in _STEPS4:
                    flush = flush or _boundary_of(n, holes) == boundary
            if flush and touched == {boundary}:
                out.append(c)
    return out


def _cheb(a, b) -> int:
    return max(abs(a[0] - b[0]), abs(a[1] - b[1]))


# --------------------------------------------------------------------------- #
# band routing
# --------------------------------------------------------------------------- #
def _route(start, goal, deep, forbidden, rng, minlen, maxlen, budget=4000):
    """Randomised depth-first self-avoiding band from ``start`` to ``goal``."""
    path = [start]
    used = {start}
    steps = [0]

    def free(c) -> bool:
        if c in used or c in forbidden:
            return False
        if c != goal and c not in deep:
            return False
        recent = set(path[-2:])
        for dx, dy in _STEPS8:
            n = (c[0] + dx, c[1] + dy)
            if n in used and n not in recent:
                return False
        return True

    def walk() -> bool:
        steps[0] += 1
        if steps[0] > budget:
            return False
        cur = path[-1]
        if cur == goal:
            return len(path) >= minlen
        if len(path) >= maxlen:
            return False
        nbrs = [(cur[0] + dx, cur[1] + dy) for dx, dy in _STEPS4]
        rng.shuffle(nbrs)
        for n in nbrs:
            reach = abs(n[0] - goal[0]) + abs(n[1] - goal[1])
            if len(path) + 1 + reach > maxlen:
                continue
            if free(n):
                path.append(n)
                used.add(n)
                if walk():
                    return True
                path.pop()
                used.discard(n)
        return False

    if start in forbidden or goal in forbidden:
        return None
    if not walk():
        return None
    return list(path)


def _separates(cut, plate, holes) -> bool:
    return len(_components(plate - set(map(tuple, cut)))) > 1


def _min_piece(cut, plate, holes) -> int:
    comps = _components(plate - set(map(tuple, cut)))
    return min(len(c) for c in comps) if len(comps) > 1 else len(plate)


def _sample_holes(rng):
    for _ in range(200):
        holes = []
        ok = True
        for _ in range(2):
            w = rng.randint(3, 6)
            h = rng.randint(3, 5)
            x0 = rng.randint(PLATE[0] + 3, PLATE[2] - 3 - w)
            y0 = rng.randint(PLATE[1] + 3, PLATE[3] - 3 - h)
            rect = [x0, y0, x0 + w, y0 + h]
            for other in holes:
                if (
                    rect[0] - 4 <= other[2]
                    and other[0] - 4 <= rect[2]
                    and rect[1] - 4 <= other[3]
                    and other[1] - 4 <= rect[3]
                ):
                    ok = False
            holes.append(rect)
        if ok:
            return holes
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    answer = COLORS[seed % 3]
    others = [c for c in COLORS if c != answer]

    for _ in range(60):
        holes = _sample_holes(rng)
        if holes is None:
            continue
        plate = _plate_cells(holes)
        deep = {c for c in plate if _deep(c, holes)}
        attach = {b: _attach_cells(holes, b) for b in (0, 1, 2)}
        if any(len(v) < 6 for v in attach.values()):
            continue

        pairs = [(0, 1), (0, 2), (1, 2)]
        rng.shuffle(pairs)
        same = rng.choice((0, 1, 2))
        plan = [(answer, same, same), (others[0], *pairs[0]), (others[1], *pairs[1])]

        cuts = {}
        forbidden = set()
        target = None
        good = True
        for colour, ba, bb in plan:
            placed = None
            for _ in range(14):
                start = rng.choice(attach[ba])
                goal = rng.choice(attach[bb])
                if _cheb(start, goal) < 6 or start == goal:
                    continue
                lo, hi = (16, 24) if target is None else (target - 2, target + 2)
                path = _route(start, goal, deep, forbidden, rng, lo, hi)
                if path is None:
                    continue
                placed = path
                break
            if placed is None:
                good = False
                break
            cuts[colour] = [list(c) for c in placed]
            for cell in placed:
                for dx in range(-(CUT_CLEARANCE - 1), CUT_CLEARANCE):
                    for dy in range(-(CUT_CLEARANCE - 1), CUT_CLEARANCE):
                        forbidden.add((cell[0] + dx, cell[1] + dy))
            if target is None:
                target = len(placed)
        if not good:
            continue

        seps = [c for c in COLORS if _separates(cuts[c], plate, holes)]
        if seps != [answer] and set(seps) != {answer}:
            continue
        if len(seps) != 1:
            continue
        if _min_piece(cuts[answer], plate, holes) < MIN_PIECE_CELLS:
            continue

        scene = {
            "seed": seed,
            "canvas": list(CANVAS),
            "cell": CELL,
            "plate": list(PLATE),
            "holes": holes,
            "cuts": {c: cuts[c] for c in COLORS},
        }
        if margin(scene) < SAMPLE_MARGIN:
            continue
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------- #
# gold, margin, quarantine
# --------------------------------------------------------------------------- #
def analytic_gold(scene: dict) -> str:
    holes = [list(h) for h in scene["holes"]]
    plate = _plate_cells(holes)
    seps = [c for c in COLORS if _separates(scene["cuts"][c], plate, holes)]
    if len(seps) != 1:
        raise ValueError(f"scene does not have a unique separating cut: {seps}")
    return seps[0]


def margin(scene: dict) -> float:
    """Smallest visible slack in the drawn evidence, in pixels."""
    holes = [list(h) for h in scene["holes"]]
    plate = _plate_cells(holes)
    cuts = {c: [tuple(p) for p in scene["cuts"][c]] for c in COLORS}

    gaps = []
    for i, a in enumerate(COLORS):
        for b in COLORS[i + 1 :]:
            gaps.append(min(_cheb(p, q) for p in cuts[a] for q in cuts[b]))
    between = (min(gaps) - 1) * CELL

    boundary = []
    non_plate = {
        (x, y)
        for x in range(PLATE[0] - 1, PLATE[2] + 2)
        for y in range(PLATE[1] - 1, PLATE[3] + 2)
        if not _is_plate((x, y), holes)
    }
    for c in COLORS:
        for p in cuts[c][2:-2]:
            boundary.append(min(_cheb(p, q) for q in non_plate))
    to_boundary = ((min(boundary) if boundary else 3) - 1) * CELL

    gold = [c for c in COLORS if _separates(cuts[c], plate, holes)]
    piece = _min_piece(cuts[gold[0]], plate, holes) if gold else 0
    piece_px = CELL * math.sqrt(piece)
    return float(round(min(between, to_boundary, piece_px), 3))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    """Pixel-identical relabellings of the latent record."""
    out = []

    a = {k: (v if k != "cuts" else dict(v)) for k, v in scene.items()}
    a["cuts"] = {c: [list(p) for p in reversed(scene["cuts"][c])] for c in COLORS}
    out.append(("band_traversal_direction", a))

    b = dict(scene)
    b["holes"] = [list(h) for h in reversed(scene["holes"])]
    out.append(("hole_record_order", b))

    d = dict(scene)
    d["cuts"] = {c: [list(p) for p in scene["cuts"][c]] for c in reversed(COLORS)}
    out.append(("cut_record_order", d))
    return out


# --------------------------------------------------------------------------- #
# raster
# --------------------------------------------------------------------------- #
def _box(cell):
    x, y = cell
    return (x * CELL, y * CELL, x * CELL + CELL - 1, y * CELL + CELL - 1)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), PAPER)
    draw = ImageDraw.Draw(img)
    x0, y0, x1, y1 = scene["plate"]
    draw.rectangle(
        (x0 * CELL, y0 * CELL, (x1 + 1) * CELL - 1, (y1 + 1) * CELL - 1), fill=PLATE_INK
    )
    for hx0, hy0, hx1, hy1 in scene["holes"]:
        draw.rectangle(
            (hx0 * CELL, hy0 * CELL, (hx1 + 1) * CELL - 1, (hy1 + 1) * CELL - 1), fill=PAPER
        )
    for colour in COLORS:
        for cell in scene["cuts"][colour]:
            draw.rectangle(_box(tuple(cell)), fill=INK[colour])
    return img
