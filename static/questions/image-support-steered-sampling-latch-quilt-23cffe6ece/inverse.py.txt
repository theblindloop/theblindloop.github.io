"""Independent image-only inverse arm. It intentionally imports no world module."""
from __future__ import annotations
import numpy as np
from PIL import Image

_PALETTES = [
    {"cyan": (36,174,194), "coral": (231,104,82), "gold": (245,190,48), "paper": (248,246,238)},
    {"cyan": (45,154,178), "coral": (222,112,74), "gold": (239,184,51), "paper": (250,247,239)}
]

def _near(pixel, color, tol=18.0):
    return float(np.linalg.norm(pixel.astype(float)-np.asarray(color, dtype=float))) <= tol

def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape != (620, 720, 3):
        raise ValueError("unexpected raster size")
    # Identify palette from large flat-color populations, not metadata.
    scores = []
    for p in _PALETTES:
        scores.append(sum(np.count_nonzero(np.linalg.norm(arr.astype(float)-np.asarray(p[k]), axis=2) < 12) for k in ("cyan","coral")))
    p = _PALETTES[int(np.argmax(scores))]
    # Sample a 3x3 interior patch in every cell. Requiring all nine to agree makes
    # grid ablations/corruptions abstain instead of silently inventing an operation.
    ops = []
    for r in range(6):
        for c in range(7):
            if r == 0 and c == 0:
                continue
            cx, cy = 45+c*90+45, 164+r*70+35
            patch = arr[cy-1:cy+2, cx-1:cx+2]
            if all(_near(px, p["cyan"]) for px in patch.reshape(-1,3)):
                ops.append("T")
            elif all(_near(px, p["coral"]) for px in patch.reshape(-1,3)):
                ops.append("S")
            else:
                raise ValueError("missing or ambiguous command cell")
    if len(ops) != 41 or ops.count("T") < 12 or ops.count("S") < 12:
        raise ValueError("incomplete command field")
    # Recover the two initial lamp fills from broad center disks.
    states = []
    for cy in (184, 215):
        disk = arr[cy-5:cy+6, 45+45-5:45+45+6]
        gold_frac = sum(_near(px, p["gold"], 22) for px in disk.reshape(-1,3)) / disk.reshape(-1,3).shape[0]
        paper_frac = sum(_near(px, p["paper"], 22) for px in disk.reshape(-1,3)) / disk.reshape(-1,3).shape[0]
        if gold_frac > .75: states.append(1)
        elif paper_frac > .75: states.append(0)
        else: raise ValueError("ambiguous START lamp")
    sensor, recorder = states
    for op in ops:
        if op == "T": sensor ^= 1
        else: recorder = sensor
    return "yes" if recorder else "no"
