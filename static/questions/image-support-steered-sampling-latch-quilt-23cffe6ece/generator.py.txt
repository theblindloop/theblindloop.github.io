"""Deterministic latent scene and Pillow renderer for Sampling Latch Quilt."""
from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw, ImageFont

W, H = 720, 620
ROWS, COLS = 6, 7
X0, Y0, CW, CH = 45, 164, 90, 70
PALETTES = [
    {"cyan": [36, 174, 194], "coral": [231, 104, 82], "gold": [245, 190, 48], "ink": [28, 35, 48], "paper": [248, 246, 238]},
    {"cyan": [45, 154, 178], "coral": [222, 112, 74], "gold": [239, 184, 51], "ink": [31, 38, 49], "paper": [250, 247, 239]}
]

def _replay(scene: dict) -> tuple[int, int]:
    sensor, recorder = int(scene["sensor"]), int(scene["recorder"])
    for op in scene["commands"]:
        if op == "T":
            sensor ^= 1
        else:
            recorder = sensor
    return sensor, recorder

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Balance final class deterministically without making any rendered local cue predictive.
    target = seed & 1
    for _ in range(10000):
        sensor, recorder = rng.randrange(2), rng.randrange(2)
        commands = [rng.choice("TS") for _ in range(ROWS * COLS - 1)]
        # Both operations are abundant and the recorder is sampled in multiple distant columns.
        if commands.count("T") < 12 or commands.count("S") < 12:
            continue
        if _replay({"sensor": sensor, "recorder": recorder, "commands": commands})[1] == target:
            return {"rows": ROWS, "cols": COLS, "sensor": sensor, "recorder": recorder,
                    "commands": commands, "palette": rng.randrange(len(PALETTES)), "audit_nonce": rng.randrange(1 << 30)}
    raise RuntimeError("balanced sampling failed")

def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene)[1] else "no"

def margin(scene: dict) -> float:
    # Minimum flat-color distance to the nearest competing palette color, in RGB units.
    p = PALETTES[int(scene["palette"])]
    colors = [p["cyan"], p["coral"], p["gold"], p["ink"], p["paper"]]
    return float(min(sum((a-b)**2 for a,b in zip(x,y)) ** 0.5 for i,x in enumerate(colors) for y in colors[i+1:]))

def is_quarantined(scene: dict) -> bool:
    return False

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["audit_nonce"] = int(scene.get("audit_nonce", 0)) + 1
    return [("unused_audit_nonce", twin)]

def _font(size: int, bold: bool = False):
    paths = ["/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"]
    for path in paths:
        try: return ImageFont.truetype(path, size)
        except OSError: pass
    return ImageFont.load_default()

def render(scene: dict) -> Image.Image:
    p = PALETTES[int(scene["palette"])]
    im = Image.new("RGB", (W, H), tuple(p["paper"]))
    d = ImageDraw.Draw(im)
    ink, gold = tuple(p["ink"]), tuple(p["gold"])
    title = "SAMPLING LATCH QUILT"
    d.text((W//2, 24), title, anchor="ma", font=_font(25, True), fill=ink)
    d.rounded_rectangle((45, 62, 675, 140), radius=12, fill=(236,233,224), outline=ink, width=3)
    d.text((61, 76), "START: SENSOR above RECORDER", font=_font(16, True), fill=ink)
    d.text((61, 105), "CYAN toggles SENSOR     CORAL copies SENSOR to RECORDER", font=_font(15), fill=ink)
    # Full edge-sharing grid: its rectangular footprint is one dense connected evidence field.
    command_i = 0
    for r in range(ROWS):
        for c in range(COLS):
            x0, y0 = X0+c*CW, Y0+r*CH
            if r == 0 and c == 0:
                fill = (224, 220, 205)
            else:
                fill = tuple(p["cyan"] if scene["commands"][command_i] == "T" else p["coral"])
                command_i += 1
            d.rectangle((x0, y0, x0+CW, y0+CH), fill=fill, outline=ink, width=3)
    # Start cell lamps and clear ordering labels.
    sx, sy = X0 + CW//2, Y0
    for j, (label, state) in enumerate((("S", int(scene["sensor"])), ("R", int(scene["recorder"])))):
        cy = sy + 20 + j*31
        d.text((X0+15, cy), label, anchor="mm", font=_font(15, True), fill=ink)
        d.ellipse((sx-10, cy-10, sx+10, cy+10), fill=gold if state else tuple(p["paper"]), outline=ink, width=3)
    d.text((W//2, 597), "READ ROWS LEFT TO RIGHT, TOP TO BOTTOM", anchor="mm", font=_font(15, True), fill=ink)
    return im
