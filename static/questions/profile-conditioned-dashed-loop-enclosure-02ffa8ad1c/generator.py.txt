"""Dashed closed loops: how many loops enclose the black dot?

Every loop is painted as a ring of short dashes, so no loop has a continuous
outline and no region is ever filled.  The decision is the enclosure depth of
one marked point inside an arrangement of pairwise non-crossing closed curves.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

W = 720
H = 720
BG = (246, 245, 240)
DOT_COLOR = (18, 18, 18)
DOT_R = 9
LOOP_WIDTH = 5
DASH_LEN = 17.0
GAP_LEN = 12.0
N_LOOPS = 4
CLEAR = 20.0          # sampling floor: dot clearance from every loop boundary
QUARANTINE_CLEAR = 13.0
RADII = (56.0, 101.0, 146.0)
PALETTE = (
    (196, 46, 54),
    (36, 122, 62),
    (40, 76, 190),
    (212, 122, 18),
    (124, 50, 156),
)
_SAMPLES = 720


# A loop is a flat record [cx, cy, r, phase, a2, p2, a3, p3, color_index]; the
# flat form keeps the scene free of nested field names.
CX, CY, R, PHASE, A2, P2, A3, P3, COLOR = range(9)


def _radius_at(loop: list, theta: float) -> float:
    """Wobbled radius of a loop; the wobble is small enough to stay convex."""
    return loop[R] * (
        1.0 + loop[A2] * math.cos(2.0 * theta + loop[P2]) + loop[A3] * math.cos(3.0 * theta + loop[P3])
    )


def _outline(loop: list) -> list[tuple[float, float]]:
    pts = []
    for i in range(_SAMPLES):
        theta = 2.0 * math.pi * i / _SAMPLES
        rad = _radius_at(loop, theta)
        pts.append((loop[CX] + rad * math.cos(theta), loop[CY] + rad * math.sin(theta)))
    return pts


def _point_in_loop(loop: list, x: float, y: float) -> bool:
    """Exact-for-star-shaped test: compare the radius at the point's angle."""
    dx, dy = x - loop[CX], y - loop[CY]
    return math.hypot(dx, dy) < _radius_at(loop, math.atan2(dy, dx))


def _boundary_distance(loop: list, x: float, y: float) -> float:
    return min(math.hypot(px - x, py - y) for px, py in _outline(loop))


def _depth(scene: dict, x: float, y: float) -> int:
    return sum(1 for loop in scene["loops"] if _point_in_loop(loop, x, y))


def _clearance(scene: dict, x: float, y: float) -> float:
    return min(_boundary_distance(loop, x, y) for loop in scene["loops"])


def _make_loop(rng: random.Random, cx: float, cy: float, r: float, color_index: int) -> list:
    return [
        round(cx, 3),
        round(cy, 3),
        round(r, 3),
        round(rng.uniform(0.0, 2.0 * math.pi), 6),
        round(rng.uniform(-0.035, 0.035), 6),
        round(rng.uniform(0.0, 2.0 * math.pi), 6),
        round(rng.uniform(-0.030, 0.030), 6),
        round(rng.uniform(0.0, 2.0 * math.pi), 6),
        color_index,
    ]


def _fits(loops: list, cx: float, cy: float, r: float) -> bool:
    """Disjoint from every placed loop, with a wide visible corridor."""
    if not (r + 26.0 <= cx <= W - r - 26.0 and r + 26.0 <= cy <= H - r - 26.0):
        return False
    for other in loops:
        d = math.hypot(cx - other[CX], cy - other[CY])
        if d < r + other[R] + 40.0:
            return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target = seed % 4                      # balanced enclosure depth 0..3
    colors = list(range(len(PALETTE)))
    rng.shuffle(colors)

    for _ in range(600):
        loops: list = []
        # nested chain of `target` loops, sharing one drifting centre
        if target:
            base_x = rng.uniform(200.0, W - 200.0)
            base_y = rng.uniform(200.0, H - 200.0)
            chain_r = list(RADII[:target])
            outer = chain_r[-1]
            base_x = min(max(base_x, outer + 26.0), W - outer - 26.0)
            base_y = min(max(base_y, outer + 26.0), H - outer - 26.0)
            for level, r in enumerate(chain_r):
                drift = 0.0 if level == target - 1 else rng.uniform(0.0, 12.0)
                ang = rng.uniform(0.0, 2.0 * math.pi)
                loops.append(
                    _make_loop(
                        rng,
                        base_x + drift * math.cos(ang),
                        base_y + drift * math.sin(ang),
                        r,
                        colors[len(loops)],
                    )
                )
            chain_centre = (base_x, base_y)
        else:
            chain_centre = None

        # decoy loops, radii drawn from the same pool so size carries no signal
        pool = list(RADII)
        rng.shuffle(pool)
        ok = True
        for k in range(N_LOOPS - target):
            placed = False
            for r in sorted(pool, reverse=True):
                for _try in range(160):
                    cx = rng.uniform(r + 26.0, W - r - 26.0)
                    cy = rng.uniform(r + 26.0, H - r - 26.0)
                    if _fits(loops, cx, cy, r) and (
                        chain_centre is None
                        or math.hypot(cx - chain_centre[0], cy - chain_centre[1]) > r + RADII[target - 1] + 40.0
                    ):
                        loops.append(_make_loop(rng, cx, cy, r, colors[len(loops)]))
                        placed = True
                        break
                if placed:
                    break
            if not placed:
                ok = False
                break
        if not ok:
            continue

        scene = {
            "width": W,
            "height": H,
            "bg_color": list(BG),
            "dot_color": list(DOT_COLOR),
            "dot_radius": DOT_R,
            "loop_width": LOOP_WIDTH,
            "dash_len": DASH_LEN,
            "gap_len": GAP_LEN,
            "loops": loops,
            "dot": [0.0, 0.0],
        }

        for _try in range(4000):
            if target:
                inner = loops[0]
                ang = rng.uniform(0.0, 2.0 * math.pi)
                rad = rng.uniform(0.0, inner[R] * 0.72)
                x, y = inner[CX] + rad * math.cos(ang), inner[CY] + rad * math.sin(ang)
            else:
                x, y = rng.uniform(40.0, W - 40.0), rng.uniform(40.0, H - 40.0)
            if _depth(scene, x, y) != target:
                continue
            if _clearance(scene, x, y) < CLEAR:
                continue
            scene["dot"] = [round(x, 3), round(y, 3)]
            return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _dashes(loop: list) -> list[tuple[tuple[float, float], tuple[float, float]]]:
    pts = _outline(loop)
    seg: list[tuple[tuple[float, float], tuple[float, float]]] = []
    period = DASH_LEN + GAP_LEN
    travelled = loop[PHASE] * loop[R] % period
    for i in range(len(pts)):
        a = pts[i]
        b = pts[(i + 1) % len(pts)]
        step = math.hypot(b[0] - a[0], b[1] - a[1])
        if (travelled % period) < DASH_LEN:
            seg.append((a, b))
        travelled += step
    return seg


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), tuple(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    for loop in scene["loops"]:
        color = PALETTE[loop[COLOR]]
        for a, b in _dashes(loop):
            draw.line([a, b], fill=color, width=scene["loop_width"])
    x, y = scene["dot"]
    r = scene["dot_radius"]
    draw.ellipse([x - r, y - r, x + r, y + r], fill=tuple(scene["dot_color"]))
    return image


def analytic_gold(scene: dict) -> str:
    return str(_depth(scene, scene["dot"][0], scene["dot"][1]))


def margin(scene: dict) -> float:
    """Smallest visible clearance, in pixels, between the dot and any loop."""
    return float(round(_clearance(scene, scene["dot"][0], scene["dot"][1]), 3))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEAR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabellings of the unordered loop set."""
    loops = scene["loops"]
    out: list[tuple[str, dict]] = []
    reversed_scene = dict(scene)
    reversed_scene["loops"] = list(reversed(loops))
    out.append(("reverse_loop_order", reversed_scene))
    shifted = dict(scene)
    shifted["loops"] = loops[1:] + loops[:1]
    out.append(("cyclic_shift_loop_order", shifted))
    return out
