"""Image-only oracle: enclosure depth from angular coverage of each dash ring.

Receives a rendered RGB image and nothing else.  For every coloured dash ring
it measures the directions, seen from the black dot, in which that ring paints
ink.  A closed curve that contains the dot surrounds it, so its dashes leave
only small angular gaps; a curve that does not contain the dot occupies a
single sector and therefore leaves a gap of at least 180 degrees.
"""

from __future__ import annotations

import math

from PIL import Image

MIN_INK = 250          # pixels; a dash ring paints far more than this
INSIDE_GAP_DEG = 90.0  # decision boundary between surrounded and not surrounded


def _groups(image: Image.Image):
    px = image.convert("RGB").load()
    width, height = image.size
    dark: list[tuple[int, int]] = []
    colored: dict[tuple[int, int, int], list[tuple[int, int]]] = {}
    for y in range(height):
        for x in range(width):
            r, g, b = px[x, y]
            if r > 200 and g > 200 and b > 190:
                continue
            if r < 80 and g < 80 and b < 80:
                dark.append((x, y))
                continue
            colored.setdefault((r, g, b), []).append((x, y))
    return dark, {k: v for k, v in colored.items() if len(v) >= MIN_INK}


def _max_gap_deg(points: list[tuple[int, int]], cx: float, cy: float) -> float:
    angles = sorted(math.atan2(y - cy, x - cx) for x, y in points)
    gap = angles[0] + 2.0 * math.pi - angles[-1]
    for i in range(1, len(angles)):
        gap = max(gap, angles[i] - angles[i - 1])
    return math.degrees(gap)


def decision_from_image(image: Image.Image) -> str:
    dark, colored = _groups(image)
    if not dark:
        return "0"
    cx = sum(p[0] for p in dark) / len(dark)
    cy = sum(p[1] for p in dark) / len(dark)
    depth = 0
    for points in colored.values():
        if _max_gap_deg(points, cx, cy) < INSIDE_GAP_DEG:
            depth += 1
    return str(min(depth, 3))
