"""Independent pixel-only inverse for the marked mosaic world.

This module deliberately duplicates the public raster specification instead of
importing any renderer or scene code. It receives only the final PNG.
"""

from __future__ import annotations

import math
from collections import defaultdict
from functools import lru_cache

import numpy as np
from PIL import Image

TILE = 160
MOSAIC = 320
MARK = 14
PERIOD = 16
INK_PER_TILE = 2300
BLUR_SIZE = 9
SYMBOLS = ("spiral", "fork", "bracket", "petal")
TRANSLATIONS = (-12, -6, 0, 6, 12)
SLOT_OFFSETS = {
    0: ((-24, 18), (18, -24)),
    1: ((TILE + 6, 18), (TILE - 38, -24)),
    2: ((-24, TILE - 38), (18, TILE + 6)),
    3: ((TILE + 6, TILE - 38), (TILE - 38, TILE + 6)),
}


def _segment_distance(x, y, a, b):
    ax, ay = a
    bx, by = b
    vx = bx - ax
    vy = by - ay
    denom = vx * vx + vy * vy
    t = ((x - ax) * vx + (y - ay) * vy) / denom
    t = np.clip(t, 0.0, 1.0)
    return np.hypot(x - (ax + t * vx), y - (ay + t * vy))


def _polyline_distance(x, y, points, closed=False):
    segments = list(zip(points, points[1:]))
    if closed:
        segments.append((points[-1], points[0]))
    return np.minimum.reduce([_segment_distance(x, y, a, b) for a, b in segments])


def _glyph_score(symbol, x, y, dx, dy):
    cx = (MOSAIC - 1) / 2.0 + float(dx)
    cy = (MOSAIC - 1) / 2.0 + float(dy)
    u = x - cx
    v = y - cy
    if symbol == "spiral":
        points = []
        for theta in np.linspace(0.15, 4.8 * math.pi, 160):
            radius = 5.0 + 5.2 * theta
            points.append((radius * math.cos(theta), radius * math.sin(theta)))
        score = _polyline_distance(u, v, points)
    elif symbol == "fork":
        score = _polyline_distance(
            u,
            v,
            [(-76.0, 64.0), (0.0, 4.0), (0.0, -76.0), (0.0, 4.0), (76.0, 64.0)],
        )
    elif symbol == "bracket":
        left = [(-48.0, -74.0), (-78.0, -74.0), (-78.0, 74.0), (-48.0, 74.0)]
        right = [(48.0, -74.0), (78.0, -74.0), (78.0, 74.0), (48.0, 74.0)]
        score = np.minimum(_polyline_distance(u, v, left), _polyline_distance(u, v, right))
    elif symbol == "petal":
        points = []
        for theta in np.linspace(0.0, 2.0 * math.pi, 220, endpoint=False):
            radius = 58.0 + 22.0 * math.sin(4.0 * theta)
            points.append((radius * math.cos(theta), radius * math.sin(theta)))
        score = _polyline_distance(u, v, points, closed=True)
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return score + 0.001 * np.hypot(u, v)


@lru_cache(maxsize=256)
def _template(symbol: str, dx: int, dy: int) -> np.ndarray:
    yy, xx = np.indices((MOSAIC, MOSAIC), dtype=np.float64)
    scores = _glyph_score(symbol, xx, yy, int(dx), int(dy))
    mask = np.zeros((MOSAIC, MOSAIC), dtype=bool)
    for qy in range(2):
        for qx in range(2):
            part = scores[qy * TILE : (qy + 1) * TILE, qx * TILE : (qx + 1) * TILE]
            order = np.argsort(part, axis=None, kind="stable")[:INK_PER_TILE]
            flat = np.zeros(TILE * TILE, dtype=np.float64)
            flat[order] = 1.0
            mask[qy * TILE : (qy + 1) * TILE, qx * TILE : (qx + 1) * TILE] = flat.reshape(
                (TILE, TILE)
            ).astype(bool)
    return _pool(_box_blur(mask.astype(np.float64)))


def _pool(values: np.ndarray) -> np.ndarray:
    factor = 4
    height, width = values.shape
    return values.reshape(height // factor, factor, width // factor, factor).mean(axis=(1, 3))


def _box_blur(values: np.ndarray) -> np.ndarray:
    radius = BLUR_SIZE // 2
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant").cumsum(0).cumsum(1)
    size = BLUR_SIZE
    return (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    ) / float(size * size)


def _marker_components(rgb: np.ndarray):
    chroma = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    ys, xs = np.nonzero(chroma > 60)
    remaining = {(int(y), int(x)) for y, x in zip(ys.tolist(), xs.tolist())}
    found = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        points = [start]
        while stack:
            y, x = stack.pop()
            for ny in range(y - 1, y + 2):
                for nx in range(x - 1, x + 2):
                    if (ny, nx) in remaining:
                        remaining.remove((ny, nx))
                        stack.append((ny, nx))
                        points.append((ny, nx))
        if len(points) >= MARK * MARK // 2:
            p = np.asarray(points, dtype=np.int32)
            y0, x0 = p.min(axis=0)
            y1, x1 = p.max(axis=0)
            found.append(
                {
                    "area": len(points),
                    "x0": int(x0),
                    "y0": int(y0),
                    "x1": int(x1),
                    "y1": int(y1),
                    "color": tuple(int(v) for v in rgb[int(y0), int(x0)]),
                }
            )
    return found


def _panel_anchors(rgb: np.ndarray):
    components = _marker_components(rgb)
    if len(components) != 8:
        raise ValueError(f"expected eight colored fiducial components, found {len(components)}")
    grouped = defaultdict(list)
    for component in components:
        grouped[component["color"]].append(component)
    if len(grouped) != 4 or any(len(items) != 2 for items in grouped.values()):
        raise ValueError("expected four colors, each used by one fiducial pair")
    panels = []
    used_slots = set()
    for items in grouped.values():
        candidates = []
        for ordered_items in (items, list(reversed(items))):
            for slot, offsets in SLOT_OFFSETS.items():
                estimates = []
                for item, (ox, oy) in zip(ordered_items, offsets):
                    estimates.append((item["x0"] - ox, item["y0"] - oy))
                error = float(np.linalg.norm(np.asarray(estimates[0]) - np.asarray(estimates[1])))
                anchor = tuple(int(round(value)) for value in np.mean(np.asarray(estimates), axis=0))
                candidates.append((error, slot, anchor))
        error, slot, anchor = min(candidates)
        if error > 1.0 or slot in used_slots:
            raise ValueError("fiducial pair does not identify a unique panel corner")
        used_slots.add(slot)
        panels.append((slot, anchor))
    if used_slots != {0, 1, 2, 3}:
        raise ValueError("not all four destination corners were identified")
    return panels


def _reconstruct(rgb: np.ndarray) -> np.ndarray:
    gray = np.asarray(Image.fromarray(rgb, mode="RGB").convert("L"), dtype=np.float64)
    assembled = np.zeros((MOSAIC, MOSAIC), dtype=np.float64)
    for slot, (x, y) in _panel_anchors(rgb):
        if x < 0 or y < 0 or x + TILE > gray.shape[1] or y + TILE > gray.shape[0]:
            raise ValueError("fiducial-defined panel is outside the raster")
        qx = slot % 2
        qy = slot // 2
        assembled[qy * TILE : (qy + 1) * TILE, qx * TILE : (qx + 1) * TILE] = gray[
            y : y + TILE, x : x + TILE
        ]
    return assembled


def _decode(image: Image.Image) -> tuple[str, float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    assembled = _reconstruct(rgb)
    blurred = _box_blur(assembled)
    edge = np.concatenate(
        (blurred[:24, :].ravel(), blurred[-24:, :].ravel(), blurred[:, :24].ravel(), blurred[:, -24:].ravel())
    )
    baseline = float(np.median(edge))
    observed = _pool(np.maximum(baseline - blurred, 0.0))[5:75, 5:75]
    observed = observed - observed.mean()
    norm = float(np.linalg.norm(observed))
    if norm < 1e-4:
        raise ValueError("reassembled evidence is flat")
    scores = []
    for symbol in SYMBOLS:
        for dx in TRANSLATIONS:
            for dy in TRANSLATIONS:
                template = _template(symbol, int(dx), int(dy))[5:75, 5:75]
                centered = template - template.mean()
                denominator = norm * float(np.linalg.norm(centered))
                if denominator > 0.0:
                    scores.append((float(np.sum(observed * centered) / denominator), symbol, dx, dy))
    scores.sort(reverse=True)
    top = scores[0]
    runner = next(item for item in scores if item[1] != top[1])
    margin = float(top[0] - runner[0])
    if top[0] < 0.25 or margin < 0.012:
        raise ValueError(f"reassembled symbol is ambiguous (score={top[0]:.3f}, margin={margin:.3f})")
    return top[1], margin


def decision_from_image(image: Image.Image) -> str:
    """Classify only the supplied final raster; no latent scene is accepted."""

    return _decode(image)[0]


def confidence_margin(image: Image.Image) -> float:
    return _decode(image)[1]
