"""Pixel-only inverse arm for the lamp-shadow gate world.

The oracle receives a rasterized image and nothing else.  It splits the row into
panels using the dark frame ink, then inside each panel separates three saturated
inks by dominant channel: the red-dominant lamp disc, the blue-dominant bar and
the green-dominant gateway segment lying on the floor.  It projects every bar
pixel from the lamp centroid onto the gateway's row, measures the clearance
between that projected span and the gateway ends, and returns the single panel
whose clearance is positive by a decisive runner-up gap.  It abstains when the
layout, an ink, or the winner is not uniquely recovered.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

N_PANELS = 5
GREY_SATURATION = 24
FRAME_MAX = 110
INK_SATURATION = 45
MIN_PANEL_COLUMNS = 60
MIN_LAMP_PIXELS = 45
MIN_BAR_PIXELS = 90
MIN_DOOR_PIXELS = 90
SLACK_TOL = 5.0
QUARANTINE_MARGIN = 10.0


def _runs(flags: np.ndarray, minimum: int) -> list:
    spans = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            if index - start >= minimum:
                spans.append((start, index - 1))
            start = None
    if start is not None and len(flags) - start >= minimum:
        spans.append((start, len(flags) - 1))
    return spans


def _panel_slack(lamp_mask, bar_mask, door_mask, x_offset: int):
    if (
        int(lamp_mask.sum()) < MIN_LAMP_PIXELS
        or int(bar_mask.sum()) < MIN_BAR_PIXELS
        or int(door_mask.sum()) < MIN_DOOR_PIXELS
    ):
        return None
    lamp_rows, lamp_cols = np.nonzero(lamp_mask)
    lamp_x = float(lamp_cols.mean()) + x_offset
    lamp_y = float(lamp_rows.mean())

    door_rows, door_cols = np.nonzero(door_mask)
    floor_y = float(door_rows.mean())
    door_lo = float(door_cols.min()) + x_offset
    door_hi = float(door_cols.max()) + x_offset
    if door_hi - door_lo < 20.0:
        return None

    bar_rows, bar_cols = np.nonzero(bar_mask)
    depth = bar_rows.astype(np.float64) - lamp_y
    if float(depth.min()) <= 1.0 or floor_y - lamp_y <= 1.0:
        return None
    projected = lamp_x + (bar_cols.astype(np.float64) + x_offset - lamp_x) * (
        (floor_y - lamp_y) / depth
    )
    return min(float(projected.min()) - door_lo, door_hi - float(projected.max()))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB")).astype(np.int16)
    highest = array.max(axis=2)
    lowest = array.min(axis=2)
    saturation = highest - lowest
    frame = (saturation <= GREY_SATURATION) & (highest < FRAME_MAX)
    coloured = saturation >= INK_SATURATION
    dominant = np.argmax(array, axis=2)
    lamp_ink = coloured & (dominant == 0)
    door_ink = coloured & (dominant == 1)
    bar_ink = coloured & (dominant == 2)

    spans = _runs(frame.any(axis=0), MIN_PANEL_COLUMNS)
    if len(spans) != N_PANELS:
        return AMBIGUOUS

    slacks = []
    for left, right in spans:
        inner = slice(left + 3, right - 2)
        slack = _panel_slack(
            lamp_ink[:, inner], bar_ink[:, inner], door_ink[:, inner], inner.start
        )
        if slack is None:
            return AMBIGUOUS
        slacks.append(slack)

    order = sorted(range(N_PANELS), key=lambda index: slacks[index], reverse=True)
    best, runner_up = order[0], order[1]
    if slacks[best] < SLACK_TOL:
        return AMBIGUOUS
    if slacks[best] - slacks[runner_up] < QUARANTINE_MARGIN:
        return AMBIGUOUS
    return f"panel-{best + 1}"
