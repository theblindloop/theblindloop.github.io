"""Renderer for the ladder_rung_walk world.

A short, thin ladder is painted somewhere on the canvas: four vertical rails of
equal length, joined by horizontal rungs that each connect two neighbouring
rails at distinct heights.  A solid dark tab above one rail marks where the walk
starts.  The walk goes down that rail and switches rail at every rung it meets,
in top-to-bottom order.  A small coloured square is welded under the bottom of
every rail; the colour of the square under the finishing rail is the decision.

The decisive evidence is therefore the interior of one thin connected route: the
walked rail segments, the rungs actually crossed and the finishing cap.  Rungs
that the walk never meets, the other caps, and every grey mark outside the
ladder are inert.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ladder_rung_walk-renderer-0.1.0"

CANVAS = 448
BACKGROUND = (246, 246, 242)
INK = (28, 28, 34)
DECOR = (201, 201, 196)

RAILS = 4
RUNG_SLOTS = 6

CAP_COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (198, 42, 42),
    "green": (32, 138, 66),
    "blue": (40, 84, 200),
    "purple": (142, 48, 168),
}
CAP_NAMES = ("red", "green", "blue", "purple")

QUARANTINE_MARGIN = 3.0


# ---------------------------------------------------------------- geometry ---
def _ladder_metrics(scene: dict) -> dict:
    """Pixel geometry of the ladder, derived only from painted scene fields."""
    pitch = scene["pitch"]
    stroke = scene["stroke"]
    rung_pitch = scene["rung_pitch"]
    x0 = scene["x0"]
    y0 = scene["y0"]
    height = rung_pitch * (RUNG_SLOTS + 1)
    rail_x = [x0 + pitch * i for i in range(RAILS)]
    slot_y = [y0 + rung_pitch * (k + 1) for k in range(RUNG_SLOTS)]
    return {
        "pitch": pitch,
        "stroke": stroke,
        "rail_x": rail_x,
        "slot_y": slot_y,
        "top": y0,
        "bottom": y0 + height,
        "cap_size": scene["cap_size"],
        "arrow_gap": scene["arrow_gap"],
    }


def _rungs(scene: dict) -> list[tuple[int, int]]:
    """Ordered (slot_index, left_rail) rungs, top to bottom."""
    return [(k, left) for k, left in enumerate(scene["slots"]) if left >= 0]


def _walk(scene: dict) -> list[int]:
    """Rail index after each rung slot, starting at the marked rail."""
    current = scene["start"]
    trace = [current]
    for _slot, left in _rungs(scene):
        if current == left:
            current = left + 1
        elif current == left + 1:
            current = left
        trace.append(current)
    return trace


def _crossings(scene: dict) -> int:
    current = scene["start"]
    crossed = 0
    for _slot, left in _rungs(scene):
        if current in (left, left + 1):
            current = left + 1 if current == left else left
            crossed += 1
    return crossed


def _bbox(scene: dict) -> tuple[int, int, int, int]:
    metrics = _ladder_metrics(scene)
    half = metrics["stroke"]
    cap = metrics["cap_size"]
    x_lo = metrics["rail_x"][0] - max(half, cap // 2) - 2
    x_hi = metrics["rail_x"][-1] + max(half, cap // 2) + 2
    y_lo = metrics["top"] - metrics["arrow_gap"] - 12
    y_hi = metrics["bottom"] + cap + 6
    return x_lo, y_lo, x_hi, y_hi


# ---------------------------------------------------------------- sampling ---
def _sample_once(rng: random.Random) -> dict | None:
    margin_tier = rng.choice([3, 4, 5, 7, 9, 11])
    stroke = 3
    pitch = rng.randint(14, 16)
    rung_pitch = min(14, max(13, stroke + margin_tier + rng.randint(0, 1)))
    cap_size = max(6, min(12, pitch - margin_tier))
    arrow_gap = max(3, margin_tier)

    slots: list[int] = [-1] * RUNG_SLOTS
    n_rungs = rng.randint(4, 5)
    chosen = rng.sample(range(RUNG_SLOTS), n_rungs)
    for slot in chosen:
        slots[slot] = rng.randint(0, RAILS - 2)

    start = rng.randrange(RAILS)
    scene_stub = {"slots": slots, "start": start}
    finish = _walk(scene_stub)[-1]
    if finish == start:
        return None
    if _crossings(scene_stub) < 2:
        return None

    caps = list(CAP_NAMES)
    rng.shuffle(caps)

    height = rung_pitch * (RUNG_SLOTS + 1)
    width = pitch * (RAILS - 1)
    pad_x = max(cap_size, 8) + 14
    pad_top = arrow_gap + 26
    pad_bottom = cap_size + 20
    x0 = rng.randint(pad_x, CANVAS - width - pad_x)
    y0 = rng.randint(pad_top, CANVAS - height - pad_bottom)

    scene = {
        "pitch": pitch,
        "stroke": stroke,
        "rung_pitch": rung_pitch,
        "cap_size": cap_size,
        "arrow_gap": arrow_gap,
        "margin_tier": margin_tier,
        "slots": slots,
        "start": start,
        "caps": caps,
        "x0": x0,
        "y0": y0,
        "decor": [],
    }
    scene["decor"] = _sample_decor(rng, scene)
    if margin(scene) < QUARANTINE_MARGIN:
        return None
    return scene


def _sample_decor(rng: random.Random, scene: dict) -> list[dict]:
    """Inert grey marks, kept well clear of the ladder."""
    x_lo, y_lo, x_hi, y_hi = _bbox(scene)
    marks: list[dict] = []
    attempts = 0
    wanted = rng.randint(7, 11)
    while len(marks) < wanted and attempts < 400:
        attempts += 1
        kind = rng.choice(["dot", "box", "bar"])
        size = rng.randint(4, 11)
        length = rng.randint(9, 26)
        cx = rng.randint(10, CANVAS - 10)
        cy = rng.randint(10, CANVAS - 10)
        w = length if kind == "bar" else size
        h = size if kind != "bar" else rng.randint(3, 5)
        left, top = cx - w // 2, cy - h // 2
        if left < 6 or top < 6 or left + w > CANVAS - 6 or top + h > CANVAS - 6:
            continue
        if left - 22 < x_hi and x_lo - 22 < left + w and top - 22 < y_hi and y_lo - 22 < top + h:
            continue
        marks.append({"kind": kind, "left": left, "top": top, "w": w, "h": h})
    return marks


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(600):
        scene = _sample_once(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"could not sample a legible ladder scene for seed {seed}")


# ------------------------------------------------------------------ answers ---
def analytic_gold(scene: dict) -> str:
    """Colour name of the cap under the rail where the walk finishes."""
    finish = _walk(scene)[-1]
    return scene["caps"][finish]


def margin(scene: dict) -> float:
    """Smallest visible clearance that the replay depends on, in pixels."""
    metrics = _ladder_metrics(scene)
    stroke = metrics["stroke"]
    rung_ys = [metrics["slot_y"][k] for k, _left in _rungs(scene)]
    gaps = [b - a - stroke for a, b in zip(rung_ys, rung_ys[1:])]
    clearances = [
        float(min(gaps)) if gaps else float(metrics["rung_pitch"] if "rung_pitch" in metrics else 99),
        float(metrics["pitch"] - stroke),
        float(metrics["pitch"] - metrics["cap_size"]),
        float(metrics["arrow_gap"]),
    ]
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    """True when a decisive clearance is too small to resolve on the raster."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted (see candidate.json)."""
    return []


# ------------------------------------------------------------------ drawing ---
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    metrics = _ladder_metrics(scene)
    stroke = metrics["stroke"]
    half = stroke // 2
    rail_x = metrics["rail_x"]
    top, bottom = metrics["top"], metrics["bottom"]

    for mark in scene["decor"]:
        box = [mark["left"], mark["top"], mark["left"] + mark["w"], mark["top"] + mark["h"]]
        if mark["kind"] == "dot":
            draw.ellipse(box, fill=DECOR)
        else:
            draw.rectangle(box, fill=DECOR)

    for x in rail_x:
        draw.rectangle([x - half, top, x + half, bottom], fill=INK)

    for slot, left in _rungs(scene):
        y = metrics["slot_y"][slot]
        draw.rectangle([rail_x[left], y - half, rail_x[left + 1], y + half], fill=INK)

    cap = metrics["cap_size"]
    cap_top = bottom + metrics["arrow_gap"]
    for index, x in enumerate(rail_x):
        colour = CAP_COLOURS[scene["caps"][index]]
        draw.rectangle([x - cap // 2, cap_top, x - cap // 2 + cap, cap_top + cap], fill=colour)

    tab_x = rail_x[scene["start"]]
    tab_bottom = top - metrics["arrow_gap"]
    draw.polygon(
        [
            (tab_x, tab_bottom),
            (tab_x - 6, tab_bottom - 10),
            (tab_x + 6, tab_bottom - 10),
        ],
        fill=INK,
    )
    return image
