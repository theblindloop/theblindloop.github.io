"""Compact directed-route renderer and analytic arm."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 247, 243)
UNDERLAY = (70, 184, 203)
DARK_ROUTES = [(20, 62, 92), (35, 68, 112), (27, 78, 91)]
START = (224, 91, 73)
FINISH = (235, 177, 52)
QUARANTINE_PX = 2.0


def _vertices(scene: dict) -> list[tuple[int, int]]:
    x, y = int(scene["x0"]), int(scene["y0"])
    points = [(x, y)]
    for direction, length in zip(scene["directions"], scene["lengths"], strict=True):
        if direction == "R":
            x += int(length)
        elif direction == "U":
            y -= int(length)
        elif direction == "D":
            y += int(length)
        else:
            raise ValueError("unsupported direction")
        points.append((x, y))
    return points


def _fraction_measure(scene: dict) -> tuple[str, float]:
    lengths = [float(v) for v in scene["lengths"]]
    target = sum(lengths) / 3.0
    cumulative = 0.0
    boundaries = []
    selected = None
    for direction, length in zip(scene["directions"], lengths, strict=True):
        if cumulative < target < cumulative + length:
            selected = "horizontal" if direction == "R" else "vertical"
        cumulative += length
        boundaries.append(cumulative)
    if selected is None:
        raise ValueError("fractional point lies on a corner")
    clearance = min(abs(target - value) for value in [0.0, *boundaries])
    return selected, clearance


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = "horizontal" if seed % 2 == 0 else "vertical"
    vertical_signs = rng.choice([("U", "D", "U"), ("D", "U", "D")])
    directions = ["R", vertical_signs[0], "R", vertical_signs[1], "R", vertical_signs[2]]
    for _ in range(20000):
        # A roughly 90 px by 10 px footprint is local yet clearly pathlike.
        lengths = [rng.randint(25, 34), rng.randint(6, 10), rng.randint(25, 34),
                   rng.randint(6, 10), rng.randint(25, 34), rng.randint(6, 10)]
        probe = {"directions": directions, "lengths": lengths}
        try:
            answer, clear = _fraction_measure(probe)
        except ValueError:
            continue
        if answer == wanted and clear >= 3.0:
            break
    else:
        raise RuntimeError("could not sample a clear fractional point")
    x0 = rng.randint(70, 330)
    y0 = rng.choice([96, 160, 224, 288, 352, 416]) + rng.randint(-3, 3)
    # Alternating vertical signs return near the start band; this conservative
    # placement leaves room for all sampled legs and endpoint dots.
    route_color = list(rng.choice(DARK_ROUTES))
    return {
        "x0": x0,
        "y0": y0,
        "directions": directions,
        "lengths": lengths,
        "stroke": rng.choice([5, 7]),
        "route_color": route_color,
        "start_color": list(START),
        "finish_color": list(FINISH),
        "width": CANVAS,
        "height": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _vertices(scene)
    stroke = int(scene["stroke"])
    draw.line(points, fill=UNDERLAY, width=stroke, joint="curve")
    radius = 6
    for point, color in ((points[0], tuple(scene["start_color"])),
                         (points[-1], tuple(scene["finish_color"]))):
        x, y = point
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=color)
    # The one-pixel centerline is the sole decisive ordered support. It remains
    # visible through both endpoint markers and is independently traced by the oracle.
    draw.line(points, fill=tuple(scene["route_color"]), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    return _fraction_measure(scene)[0]


def margin(scene: dict) -> float:
    return _fraction_measure(scene)[1]


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < QUARANTINE_PX
    except ValueError:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
