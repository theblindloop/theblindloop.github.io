"""Independent inverse arm: trace the dark one-pixel route from raster pixels only."""

from __future__ import annotations

import numpy as np
from PIL import Image

ROUTE_COLORS = np.asarray([(20, 62, 92), (35, 68, 112), (27, 78, 91)], dtype=np.int16)
START_COLOR = np.asarray((224, 91, 73), dtype=np.int16)


def _mask_near(array: np.ndarray, colors: np.ndarray, tolerance: int = 4) -> np.ndarray:
    delta = np.abs(array.astype(np.int16)[:, :, None, :] - colors[None, None, :, :])
    return np.any(np.max(delta, axis=3) <= tolerance, axis=2)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    route = _mask_near(array, ROUTE_COLORS)
    points_yx = np.argwhere(route)
    if len(points_yx) < 45 or len(points_yx) > 190:
        raise ValueError("route centerline is missing or corrupted")
    points = {(int(x), int(y)) for y, x in points_yx}
    neighbors: dict[tuple[int, int], list[tuple[int, int]]] = {}
    for x, y in points:
        adjacent = [q for q in ((x-1,y), (x+1,y), (x,y-1), (x,y+1)) if q in points]
        if len(adjacent) > 2:
            raise ValueError("route is not a simple thin path")
        neighbors[(x, y)] = adjacent
    endpoints = [p for p, adjacent in neighbors.items() if len(adjacent) == 1]
    if len(endpoints) != 2 or any(len(v) == 0 for v in neighbors.values()):
        raise ValueError("route is disconnected or lacks two endpoints")

    start_delta = np.max(np.abs(array.astype(np.int16) - START_COLOR), axis=2)
    syx = np.argwhere(start_delta <= 4)
    if len(syx) < 20:
        raise ValueError("visible start marker is missing")
    start_xy = np.asarray([syx[:, 1].mean(), syx[:, 0].mean()])
    start = min(endpoints, key=lambda p: float(np.linalg.norm(np.asarray(p) - start_xy)))
    other = endpoints[1] if endpoints[0] == start else endpoints[0]
    if np.linalg.norm(np.asarray(start) - start_xy) > 2.5:
        raise ValueError("route endpoint is not bound to start marker")

    ordered = [start]
    previous = None
    current = start
    while current != other:
        choices = [p for p in neighbors[current] if p != previous]
        if len(choices) != 1:
            raise ValueError("ambiguous route traversal")
        previous, current = current, choices[0]
        ordered.append(current)
        if len(ordered) > len(points) + 1:
            raise ValueError("route contains a cycle")
    if len(ordered) != len(points):
        raise ValueError("route contains detached pixels")

    step_count = len(ordered) - 1
    target = step_count / 3.0
    index = int(np.floor(target))
    if index <= 0 or index >= step_count:
        raise ValueError("invalid fractional index")
    # Clear submitted scenes put the target far inside one leg, so either
    # neighboring edge has the same orientation.
    p, q = ordered[index], ordered[index + 1]
    if p[1] == q[1] and p[0] != q[0]:
        return "horizontal"
    if p[0] == q[0] and p[1] != q[1]:
        return "vertical"
    raise ValueError("fractional edge is not orthogonal")
