"""Independent pixel-only inverse arm for the connected-piece-count world.

It receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking "dark ink" pixels (anything clearly not the white background);
2. counting the number of 4-connected components of that ink mask.

Every separate drawn piece is one connected ink component, so its count equals
the analytic piece count exactly on robust scenes (where no two distinct pieces
are bridged by antialiasing).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "piece_count-oracle-0.4.0"

_INK_THRESHOLD = 210


def _is_ink(arr: np.ndarray, threshold: int = _INK_THRESHOLD) -> np.ndarray:
    """True where any RGB channel is clearly darker than the white background."""
    return arr.min(axis=-1) < threshold


def _count_components(mask: np.ndarray) -> int:
    """Count 4-connected components of a boolean mask (BFS over true pixels)."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                count += 1
                seen[y, x] = True
                stack = [(x, y)]
                while stack:
                    cx, cy = stack.pop()
                    for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                        if (
                            0 <= nx < w
                            and 0 <= ny < h
                            and mask[ny, nx]
                            and not seen[ny, nx]
                        ):
                            seen[ny, nx] = True
                            stack.append((nx, ny))
    return count


def decision_from_image(image: Image.Image) -> int:
    """Number of connected line pieces recovered from the final PNG alone."""
    arr = np.asarray(image.convert("RGB"))
    ink = _is_ink(arr)
    return _count_components(ink)
