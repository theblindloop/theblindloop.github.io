"""Deterministic renderer for length-ranked row reassembly."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

ROWS = 8
COLS = 9
LABELS = ("left_chevron", "right_chevron", "two_rising_strokes", "two_falling_strokes")

# Pairs of source rows are pixel-identical.  Every class uses the same multiset
# of horizontal three-tile strokes, so neither raw pieces nor total ink predicts
# the answer.  Only the tab-length-to-row binding does.
ROW_STARTS = (0, 0, 2, 2, 4, 4, 6, 6)
ORDERS: dict[str, tuple[int, ...]] = {
    "left_chevron": (6, 4, 2, 0, 1, 3, 5, 7),
    "right_chevron": (0, 2, 4, 6, 7, 5, 3, 1),
    "two_rising_strokes": (0, 2, 4, 6, 1, 3, 5, 7),
    "two_falling_strokes": (6, 4, 2, 0, 7, 5, 3, 1),
}

TILE_PALETTES = (
    {"dark": (29, 48, 66), "light": (218, 229, 230)},
    {"dark": (70, 43, 91), "light": (232, 222, 234)},
    {"dark": (42, 74, 52), "light": (226, 232, 213)},
)
BACKGROUNDS = ((248, 245, 237), (239, 245, 247), (246, 240, 245))
GOLD = (226, 160, 25)
PANEL = (229, 232, 229)
FRAME = (55, 62, 68)
CANVAS = (760, 540)


def _strip_cells(strip_id: int) -> tuple[int, ...]:
    strip_id = int(strip_id)
    if not 0 <= strip_id < ROWS:
        raise ValueError("strip identifier is outside 0..7")
    start = ROW_STARTS[strip_id]
    return tuple(int(start <= column < start + 3) for column in range(COLS))


def _template(label: str) -> tuple[tuple[int, ...], ...]:
    return tuple(_strip_cells(strip_id) for strip_id in ORDERS[str(label)])


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    block = seed // len(LABELS)
    label = LABELS[seed % len(LABELS)]
    # All four labels in a balance block receive exactly the same nuisance state.
    rng = random.Random((block ^ 0x7A8B_91C2_D3E4) & 0xFFFFFFFFFFFFFFFF)
    source_order = list(range(ROWS))
    rng.shuffle(source_order)
    return {
        "label": label,
        "source_order": source_order,
        "cell_width": (34, 38, 42)[block % 3],
        "cell_height": (24, 27, 30)[(block // 3) % 3],
        "cell_gap": (3, 4, 5)[(block // 5) % 3],
        "row_gap": (18, 21, 24)[(block // 7) % 3],
        "origin_x": (276, 288, 300)[(block // 11) % 3],
        "origin_y": (58, 68, 78)[(block // 13) % 3],
        "tab_origin_x": (54, 66, 78)[(block // 17) % 3],
        "tab_base": (24, 29, 34)[(block // 19) % 3],
        "tab_step": (16, 18, 20)[(block // 23) % 3],
        "tile_palette_index": block % len(TILE_PALETTES),
        "background_index": (block // 2) % len(BACKGROUNDS),
    }


def render(scene: dict) -> Image.Image:
    label = str(scene["label"])
    if label not in ORDERS:
        raise ValueError("unknown label")
    source_order = [int(value) for value in scene["source_order"]]
    if sorted(source_order) != list(range(ROWS)):
        raise ValueError("source_order must be a permutation of eight strips")
    cell_width = int(scene["cell_width"])
    cell_height = int(scene["cell_height"])
    cell_gap = int(scene["cell_gap"])
    row_gap = int(scene["row_gap"])
    origin_x = int(scene["origin_x"])
    origin_y = int(scene["origin_y"])
    tab_x = int(scene["tab_origin_x"])
    tab_base = int(scene["tab_base"])
    tab_step = int(scene["tab_step"])
    if min(cell_width, cell_height, cell_gap, row_gap, tab_base, tab_step) <= 0:
        raise ValueError("render dimensions must be positive")

    background = BACKGROUNDS[int(scene["background_index"]) % len(BACKGROUNDS)]
    palette = TILE_PALETTES[int(scene["tile_palette_index"]) % len(TILE_PALETTES)]
    image = Image.new("RGB", CANVAS, background)
    draw = ImageDraw.Draw(image)
    grid_width = COLS * cell_width + (COLS - 1) * cell_gap
    grid_height = ROWS * cell_height + (ROWS - 1) * row_gap
    panel_left = tab_x - 18
    panel_top = origin_y - 18
    panel_right = origin_x + grid_width + 18
    panel_bottom = origin_y + grid_height + 18
    if not (0 <= panel_left < panel_right < CANVAS[0] and 0 <= panel_top < panel_bottom < CANVAS[1]):
        raise ValueError("scene geometry exceeds canvas")
    draw.rounded_rectangle((panel_left, panel_top, panel_right, panel_bottom), radius=14,
                           fill=PANEL, outline=FRAME, width=2)

    rank_for_strip = {strip_id: rank for rank, strip_id in enumerate(ORDERS[label])}
    for source_slot, strip_id in enumerate(source_order):
        y0 = origin_y + source_slot * (cell_height + row_gap)
        cy = y0 + (cell_height - 1) // 2
        rank = rank_for_strip[strip_id]
        tab_width = tab_base + rank * tab_step
        tab_height = max(10, cell_height // 2)
        draw.rounded_rectangle((tab_x, cy - tab_height // 2, tab_x + tab_width - 1,
                                cy + (tab_height - 1) // 2), radius=4, fill=GOLD)
        for column, value in enumerate(_strip_cells(strip_id)):
            x0 = origin_x + column * (cell_width + cell_gap)
            draw.rectangle((x0, y0, x0 + cell_width - 1, y0 + cell_height - 1),
                           fill=palette["dark" if value else "light"])
        draw.rectangle((origin_x - 3, y0 - 3, origin_x + grid_width + 2,
                        y0 + cell_height + 2), outline=FRAME, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["label"])


def margin(scene: dict) -> float:
    target = _template(str(scene["label"]))
    distances = []
    for name in LABELS:
        if name == str(scene["label"]):
            continue
        rival = _template(name)
        distances.append(sum(target[r][c] != rival[r][c]
                             for r in range(ROWS) for c in range(COLS)))
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return (margin(scene) < 20.0 or int(scene["tab_step"]) < 12
            or int(scene["cell_width"]) < 30 or int(scene["cell_height"]) < 20)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["tile_palette_index"] = int(scene["tile_palette_index"]) + len(TILE_PALETTES)
    twin["background_index"] = int(scene["background_index"]) + len(BACKGROUNDS)
    return [("add_complete_palette_periods", twin)]
