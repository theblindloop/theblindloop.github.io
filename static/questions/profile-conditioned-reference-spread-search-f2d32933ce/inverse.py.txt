"""Independent pixel-only inverse arm for reference_spread_search.

Receives only a PIL image and the public question contract: the image has a
reference cluster of three dark dots in the top panel and one cluster of three
dark dots per quadrant. The spread of a cluster is the distance between its two
farthest-apart dots (its diameter). This arm locates the dark dots in the
reference panel and in each quadrant with the fixed public layout, measures each
cluster's diameter, and selects the quadrant whose measured diameter is closest
to the reference cluster's diameter.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import math

import numpy as np

DOT_THRESH = 150  # pixels darker than this are dot ink (dividers are much lighter)

WINDOWS = {
    "REF": (120, 400, 10, 150),
    "TL": (20, 248, 145, 320),
    "TR": (272, 500, 145, 320),
    "BL": (20, 248, 330, 510),
    "BR": (272, 500, 330, 510),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return the pixel coordinate lists of the 4-connected foreground blobs."""
    h, w = mask.shape
    comps = []
    seen = np.zeros((h, w), dtype=bool)
    for start in np.argwhere(mask):
        sy, sx = int(start[0]), int(start[1])
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((x, y))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        comps.append(np.asarray(pixels, dtype=np.float64))
    return comps


def _clusters_in_window(arr: np.ndarray, name: str) -> list[np.ndarray]:
    x0, x1, y0, y1 = WINDOWS[name]
    region = arr[y0:y1, x0:x1]
    dark = np.min(region, axis=2) < DOT_THRESH
    return _components(dark)


def _spread(centroids: list[tuple[float, float]]) -> float:
    best = 0.0
    n = len(centroids)
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(centroids[i][0] - centroids[j][0], centroids[i][1] - centroids[j][1])
            if d > best:
                best = d
    return float(best)


def _cluster_spread(arr: np.ndarray, name: str) -> float:
    clusters = _clusters_in_window(arr, name)
    centroids = []
    for comp in clusters:
        if len(comp) < 5:  # ignore stray antialiasing specks
            continue
        centroids.append((float(comp[:, 0].mean()), float(comp[:, 1].mean())))
    if len(centroids) < 2:
        return 0.0
    return _spread(centroids)


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    ref_spread = _cluster_spread(arr, "REF")
    best_sector = None
    best_diff = float("inf")
    for sector in ("TL", "TR", "BL", "BR"):
        spread = _cluster_spread(arr, sector)
        diff = abs(spread - ref_spread)
        if diff < best_diff:
            best_diff = diff
            best_sector = sector
    return best_sector if best_sector is not None else "TL"
