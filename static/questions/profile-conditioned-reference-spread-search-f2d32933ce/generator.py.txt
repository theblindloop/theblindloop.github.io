"""Reference-guided cluster-spread (dispersion) search world.

A reference cluster of three dark dots is shown in the top panel. The spread of
a cluster is the distance between the two dots in it that are farthest apart
(the cluster's widest extent, its diameter). The image below is split into four
quadrants, each holding its own cluster of three dark dots. Exactly one
quadrant's cluster has the same spread as the reference cluster; the other
three each have a clearly different spread. The decision is the quadrant whose
cluster's spread matches the reference's spread. All geometry and the decision
are visible in the raster; rendering is deterministic.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

SIZE = 520
BG = (255, 255, 255)
DOT = (25, 25, 25)
DIVIDER = (230, 230, 230)
R = 11  # dot radius in pixels

REF_CX, REF_CY = 260, 80

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 125, "TR": 395, "BL": 125, "BR": 395}
SECTOR_CY = {"TL": 245, "TR": 245, "BL": 408, "BR": 408}

SPREAD_LO = 48
SPREAD_HI = 84
MIN_SEP = 26.0  # minimum centre distance between any two dots in a cluster
MIN_DIFF = 8  # minimum |spread - reference spread| for every distractor
QUARANTINE_THRESH = 6.0

# Public fixed layout windows used also by the independent oracle.
Y_REF_LO, Y_REF_HI = (10, 150)
Y_ROW1_LO, Y_ROW1_HI = (145, 320)
Y_ROW2_LO, Y_ROW2_HI = (330, 510)


def _centroid(points: list[tuple[float, float]]) -> tuple[float, float]:
    n = len(points)
    return (sum(p[0] for p in points) / n, sum(p[1] for p in points) / n)


def _make_cluster(rng, spread: int, center: tuple[float, float]):
    """Three dark-dot centres with exact diameter (max pairwise distance) = spread.

    A and C are fixed a distance ``spread`` apart so their distance is exactly the
    diameter D. B is sampled in the lens of points within D of both A and C but at
    least MIN_SEP from each, so the triangle is scattered (non-zero separation) yet
    its widest extent stays exactly D. The whole cluster is then rotated by a
    random angle about its centroid and translated to the cell centre, so no fixed
    orientation or absolute position is a cue.
    """
    d = float(spread)
    a = (0.0, 0.0)
    c = (d, 0.0)
    b = None
    half = d / 2.0
    for _ in range(20000):
        u = float(rng.uniform(-half, half))
        v = float(rng.uniform(-0.7 * d, 0.7 * d))
        b = (half + u, v)
        d1 = math.hypot(b[0] - a[0], b[1] - a[1])
        d2 = math.hypot(b[0] - c[0], b[1] - c[1])
        if MIN_SEP <= d1 <= d and MIN_SEP <= d2 <= d:
            break
    if b is None:  # pragma: no cover - spread range guarantees a feasible point
        raise RuntimeError(f"could not place scattered cluster for spread={spread}")
    pts = [a, b, c]
    cx0, cy0 = _centroid(pts)
    local = [(x - cx0, y - cy0) for x, y in pts]
    theta = float(rng.uniform(0.0, 2.0 * math.pi))
    cos_t, sin_t = math.cos(theta), math.sin(theta)
    cx, cy = center
    out = []
    for x, y in local:
        rx = x * cos_t - y * sin_t + cx
        ry = x * sin_t + y * cos_t + cy
        out.append((rx, ry))
    return out


def _in_panel(pts, panel: str) -> bool:
    if panel == "REF":
        x0, x1, y0, y1 = 120, 400, Y_REF_LO + R, Y_REF_HI - R
    elif panel == "TL":
        x0, x1, y0, y1 = 20, 248, Y_ROW1_LO + R, Y_ROW1_HI - R
    elif panel == "TR":
        x0, x1, y0, y1 = 272, 500, Y_ROW1_LO + R, Y_ROW1_HI - R
    elif panel == "BL":
        x0, x1, y0, y1 = 20, 248, Y_ROW2_LO + R, Y_ROW2_HI - R
    else:  # BR
        x0, x1, y0, y1 = 272, 500, Y_ROW2_LO + R, Y_ROW2_HI - R
    return all(x0 <= x <= x1 and y0 <= y <= y1 for x, y in pts)


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    ref_spread = int(rng.integers(SPREAD_LO, SPREAD_HI + 1))
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]

    spreads: dict[str, int] = {winner_sector: ref_spread}
    for sector in SECTORS:
        if sector in spreads:
            continue
        for _ in range(2000):
            cand = int(rng.integers(SPREAD_LO, SPREAD_HI + 1))
            if abs(cand - ref_spread) < MIN_DIFF:
                continue
            spreads[sector] = cand
            break
        else:  # pragma: no cover - pathologically rare
            off = MIN_DIFF if ref_spread >= SPREAD_LO + MIN_DIFF else -MIN_DIFF
            spreads[sector] = ref_spread + off

    def _place(center, spread, panel):
        for _ in range(20000):
            pts = _make_cluster(rng, spread, center)
            if _in_panel(pts, panel):
                return pts
        return _make_cluster(rng, spread, center)  # pragma: no cover

    reference = {"spread": float(ref_spread), "points": _place((REF_CX, REF_CY), ref_spread, "REF")}
    centers = {
        "TL": (SECTOR_CX["TL"], SECTOR_CY["TL"]),
        "TR": (SECTOR_CX["TR"], SECTOR_CY["TR"]),
        "BL": (SECTOR_CX["BL"], SECTOR_CY["BL"]),
        "BR": (SECTOR_CX["BR"], SECTOR_CY["BR"]),
    }
    candidates = [
        {
            "sector": sector,
            "spread": float(spreads[sector]),
            "points": _place(centers[sector], spreads[sector], sector),
        }
        for sector in SECTORS
    ]
    return {"reference": reference, "candidates": candidates}


def _draw_dots(draw, points: list) -> None:
    for x, y in points:
        draw.ellipse([x - R, y - R, x + R, y + R], fill=DOT)


def _draw_dividers(draw) -> None:
    draw.line([0, Y_REF_LO, SIZE, Y_REF_LO], fill=DIVIDER, width=2)
    draw.line([0, Y_ROW1_LO, SIZE, Y_ROW1_LO], fill=DIVIDER, width=2)
    draw.line([0, Y_ROW2_LO, SIZE, Y_ROW2_LO], fill=DIVIDER, width=2)
    draw.line([260, Y_ROW1_LO, 260, SIZE], fill=DIVIDER, width=2)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    _draw_dividers(draw)
    _draw_dots(draw, scene["reference"]["points"])
    for cand in scene["candidates"]:
        _draw_dots(draw, cand["points"])
    return img


def _spread_of(points: list) -> float:
    best = 0.0
    n = len(points)
    for i in range(n):
        for j in range(i + 1, n):
            d = math.dist(points[i], points[j])
            if d > best:
                best = d
    return float(best)


def _decision(scene: dict) -> tuple[str, float]:
    ref = scene["reference"]["spread"]
    best_sector = None
    best_diff = float("inf")
    for cand in scene["candidates"]:
        diff = abs(cand["spread"] - ref)
        if diff < best_diff:
            best_diff = diff
            best_sector = cand["sector"]
    runner_up = min(
        abs(c["spread"] - ref)
        for c in scene["candidates"]
        if c["sector"] != best_sector
    )
    return best_sector, float(runner_up)


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
