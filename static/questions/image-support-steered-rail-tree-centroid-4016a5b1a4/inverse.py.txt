"""Independent pixel-only reconstruction of the station-track tree."""

from __future__ import annotations

import numpy as np

CYAN = np.array((0, 164, 190), dtype=np.int16)
YELLOW = np.array((244, 174, 36), dtype=np.int16)
BLACK = np.array((27, 31, 36), dtype=np.int16)
TRACK = np.array((79, 86, 94), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        output.append(np.asarray(points, dtype=np.float64))
    return output


def _station_centers(array: np.ndarray) -> tuple[list[tuple[float, float]], int, int]:
    groups: list[tuple[str, np.ndarray]] = []
    for name, color in (("black", BLACK), ("cyan", CYAN), ("yellow", YELLOW)):
        for component in _components(_near(array, color)):
            if len(component) >= 180:
                groups.append((name, component))
    if len(groups) != 13:
        raise ValueError(f"expected 13 round stations, recovered {len(groups)}")
    if sum(name == "cyan" for name, _ in groups) != 1 or sum(name == "yellow" for name, _ in groups) != 1:
        raise ValueError("expected exactly one station of each candidate color")
    centers = [(float(points[:, 1].mean()), float(points[:, 0].mean())) for _, points in groups]
    cyan = next(i for i, (name, _) in enumerate(groups) if name == "cyan")
    yellow = next(i for i, (name, _) in enumerate(groups) if name == "yellow")
    return centers, cyan, yellow


def _track_edges(array: np.ndarray, centers: list[tuple[float, float]]) -> list[tuple[int, int]]:
    components = [points for points in _components(_near(array, TRACK)) if len(points) >= 25]
    if len(components) != 12:
        raise ValueError(f"expected 12 intact track segments, recovered {len(components)}")
    edges = []
    center_array = np.asarray(centers, dtype=np.float64)
    for points in components:
        ys, xs = points[:, 0], points[:, 1]
        if xs.max() - xs.min() >= ys.max() - ys.min():
            endpoints = np.asarray(((xs.min(), np.median(ys[xs == xs.min()])), (xs.max(), np.median(ys[xs == xs.max()]))))
        else:
            endpoints = np.asarray(
                (
                    (np.median(xs[ys == ys.min()]), ys.min()),
                    (np.median(xs[ys == ys.max()]), ys.max()),
                )
            )
        selected = []
        for endpoint in endpoints:
            distances = np.linalg.norm(center_array - endpoint, axis=1)
            index = int(np.argmin(distances))
            if distances[index] > 25.0:
                raise ValueError("track endpoint does not meet a visible station")
            selected.append(index)
        edge = tuple(sorted(selected))
        if edge[0] == edge[1] or edge in edges:
            raise ValueError("track recovery produced a loop or duplicate edge")
        edges.append(edge)
    return edges


def _largest_after(adjacency: list[set[int]], removed: int) -> int:
    unseen = set(range(len(adjacency))) - {removed}
    largest = 0
    while unseen:
        root = unseen.pop()
        stack = [root]
        size = 0
        while stack:
            node = stack.pop()
            size += 1
            for neighbor in adjacency[node]:
                if neighbor != removed and neighbor in unseen:
                    unseen.remove(neighbor)
                    stack.append(neighbor)
        largest = max(largest, size)
    return largest


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers, cyan, yellow = _station_centers(array)
    edges = _track_edges(array, centers)
    adjacency = [set() for _ in centers]
    for a, b in edges:
        adjacency[a].add(b)
        adjacency[b].add(a)
    seen = {0}
    stack = [0]
    while stack:
        for neighbor in adjacency[stack.pop()]:
            if neighbor not in seen:
                seen.add(neighbor)
                stack.append(neighbor)
    if len(seen) != 13:
        raise ValueError("recovered tracks are not one connected tree")
    limit = 6
    cyan_ok = _largest_after(adjacency, cyan) <= limit
    yellow_ok = _largest_after(adjacency, yellow) <= limit
    if cyan_ok == yellow_ok:
        raise ValueError("image does not expose exactly one balancing candidate")
    return "cyan" if cyan_ok else "yellow"
