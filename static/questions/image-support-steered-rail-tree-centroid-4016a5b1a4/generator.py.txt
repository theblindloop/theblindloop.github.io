"""Deterministic Pillow world for a vertex-centroid decision on a visible tree."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 760, 360
BACKGROUND = (248, 247, 242)
TRACK = (79, 86, 94)
BLACK = (27, 31, 36)
CYAN = (0, 164, 190)
YELLOW = (244, 174, 36)


def _adjacency(scene: dict) -> list[set[int]]:
    nodes = scene["nodes"]
    adjacency = [set() for _ in nodes]
    for a, b in scene["edges"]:
        adjacency[a].add(b)
        adjacency[b].add(a)
    return adjacency


def _largest_after(adjacency: list[set[int]], removed: int) -> int:
    unseen = set(range(len(adjacency))) - {removed}
    largest = 0
    while unseen:
        root = next(iter(unseen))
        unseen.remove(root)
        queue = [root]
        size = 0
        while queue:
            node = queue.pop()
            size += 1
            for neighbor in adjacency[node]:
                if neighbor != removed and neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
        largest = max(largest, size)
    return largest


def _qualifying_nodes(scene: dict) -> list[int]:
    adjacency = _adjacency(scene)
    limit = (len(adjacency) - 1) // 2
    return [i for i in range(len(adjacency)) if _largest_after(adjacency, i) <= limit]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Seven widely spaced spine stations plus six leaves form a 13-vertex tree.
    # Sampling six of ten upper/lower leaf slots moves the graph centroid while
    # keeping the visible support one connected, horizontally elongated trace.
    spine = [[80 + 100 * i, 150] for i in range(7)]
    slots = [(parent, side) for parent in range(1, 6) for side in (-1, 1)]
    chosen = rng.sample(slots, 6)
    nodes = list(spine)
    edges = [[i, i + 1] for i in range(6)]
    for parent, side in chosen:
        x = spine[parent][0]
        # Small deterministic lateral offsets prevent identical-looking crowns.
        x += (-12 if (parent + side) % 2 else 12)
        nodes.append([x, 150 + side * 42])
        edges.append([parent, len(nodes) - 1])

    provisional = {"nodes": nodes, "edges": edges, "cyan_node": 0, "yellow_node": 1}
    centroid = _qualifying_nodes(provisional)[0]
    candidates = [i for i in range(7) if i != centroid and abs(i - centroid) >= 2]
    decoy = rng.choice(candidates)
    if seed % 2 == 0:
        cyan_node, yellow_node = centroid, decoy
    else:
        cyan_node, yellow_node = decoy, centroid
    return {
        "nodes": nodes,
        "edges": edges,
        "cyan_node": cyan_node,
        "yellow_node": yellow_node,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for a, b in scene["edges"]:
        draw.line([tuple(nodes[a]), tuple(nodes[b])], fill=TRACK, width=9)
    for index, (x, y) in enumerate(nodes):
        color = CYAN if index == scene["cyan_node"] else YELLOW if index == scene["yellow_node"] else BLACK
        draw.ellipse((x - 14, y - 14, x + 14, y + 14), fill=color, outline=BACKGROUND, width=2)
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    qualifying = set(_qualifying_nodes(scene))
    cyan = scene["cyan_node"] in qualifying
    yellow = scene["yellow_node"] in qualifying
    if cyan == yellow:
        raise ValueError("scene does not have exactly one qualifying colored station")
    return "cyan" if cyan else "yellow"


def margin(scene: dict) -> float:
    adjacency = _adjacency(scene)
    cyan = _largest_after(adjacency, scene["cyan_node"])
    yellow = _largest_after(adjacency, scene["yellow_node"])
    return float(abs(cyan - yellow))


def is_quarantined(scene: dict) -> bool:
    try:
        n = len(scene["nodes"])
        adjacency = _adjacency(scene)
        if n != 13 or len(scene["edges"]) != n - 1:
            return True
        seen = {0}
        queue = deque([0])
        while queue:
            for neighbor in adjacency[queue.popleft()]:
                if neighbor not in seen:
                    seen.add(neighbor)
                    queue.append(neighbor)
        if len(seen) != n or len(_qualifying_nodes(scene)) != 1:
            return True
        if scene["cyan_node"] == scene["yellow_node"] or margin(scene) < 1.0:
            return True
        return False
    except (KeyError, IndexError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
