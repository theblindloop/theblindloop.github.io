"""Independent pixel-only inverse for the compare-and-swap world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

GREEN = (22, 163, 74)
BLUE = (37, 99, 235)
MAGENTA = (219, 39, 119)
BLACK = (24, 24, 27)


def _components(mask: np.ndarray) -> list[dict]:
    work = mask.copy()
    height, width = work.shape
    out: list[dict] = []
    for y, x in zip(*np.nonzero(work), strict=True):
        if not work[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        work[y, x] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        out.append({
            "area": len(xs),
            "bbox": (min(xs), min(ys), max(xs), max(ys)),
            "cx": sum(xs) / len(xs),
            "cy": sum(ys) / len(ys),
        })
    return out


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.array(rgb, dtype=np.uint8), axis=2)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 400 or arr.shape[1] < 700:
        raise ValueError("image is too small for the public visual grammar")

    black = _components(_mask(arr, BLACK))
    slots = sorted(
        [c for c in black if c["bbox"][2] - c["bbox"][0] > 100 and c["bbox"][3] - c["bbox"][1] > 130],
        key=lambda c: c["cx"],
    )
    if len(slots) != 4:
        raise ValueError(f"expected four start slots, found {len(slots)}")

    green_pips = [c for c in _components(_mask(arr, GREEN)) if 150 <= c["area"] <= 400]
    initial: list[int] = []
    for slot in slots:
        x0, y0, x1, y1 = slot["bbox"]
        count = sum(x0 < p["cx"] < x1 and y0 < p["cy"] < y1 for p in green_pips)
        if count not in (1, 2, 3, 4):
            raise ValueError("unreadable token pip count")
        initial.append(count)
    if sorted(initial) != [1, 2, 3, 4]:
        raise ValueError("start tokens do not form the declared value set")

    magenta = _components(_mask(arr, MAGENTA))
    rings = [c for c in magenta if c["bbox"][2] - c["bbox"][0] > 120]
    if len(rings) != 1:
        raise ValueError("expected one query ring")
    query_slot = min(range(4), key=lambda i: abs(slots[i]["cx"] - rings[0]["cx"]))

    blue = _components(_mask(arr, BLUE))
    frames = sorted(
        [c for c in blue if c["bbox"][2] - c["bbox"][0] > 140 and c["bbox"][3] - c["bbox"][1] > 170],
        key=lambda c: c["cx"],
    )
    pips = [c for c in blue if 100 <= c["area"] <= 260 and c["bbox"][2] - c["bbox"][0] < 25]
    if len(frames) != 5:
        raise ValueError(f"expected five comparator cards, found {len(frames)}")
    comparators: list[tuple[int, int]] = []
    for frame in frames:
        x0, y0, x1, y1 = frame["bbox"]
        inside = [p for p in pips if x0 < p["cx"] < x1 and y0 < p["cy"] < y1]
        left = sum(p["cx"] < frame["cx"] for p in inside)
        right = sum(p["cx"] > frame["cx"] for p in inside)
        if left not in (1, 2, 3, 4) or right not in (1, 2, 3, 4):
            raise ValueError("unreadable comparator operands")
        comparators.append((left, right))

    values = list(initial)
    for left_slot, right_slot in comparators:
        left = values[left_slot - 1]
        right = values[right_slot - 1]
        values[left_slot - 1] = min(left, right)
        values[right_slot - 1] = max(left, right)
    return str(values[query_slot])
