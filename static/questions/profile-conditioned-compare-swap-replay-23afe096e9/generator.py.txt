"""Deterministic visual world for replaying compare-and-swap cards."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 560
GREEN = (22, 163, 74)
BLUE = (37, 99, 235)
MAGENTA = (219, 39, 119)
BLACK = (24, 24, 27)
GRAY = (100, 116, 139)
PAPER = (248, 250, 252)
WHITE = (255, 255, 255)


def _replay(initial_values: list[int], comparators: list[list[int]]) -> list[int]:
    values = list(initial_values)
    for left_slot, right_slot in comparators:
        left = values[left_slot - 1]
        right = values[right_slot - 1]
        values[left_slot - 1] = min(left, right)
        values[right_slot - 1] = max(left, right)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    initial = [1, 2, 3, 4]
    rng.shuffle(initial)

    # The first card is guaranteed to perform a real swap. Later cards include
    # both effective and ineffective comparisons as the state evolves.
    high_slot = initial.index(max(initial[0], initial[1])) + 1
    low_slot = initial.index(min(initial[0], initial[1])) + 1
    comparators: list[list[int]] = [[high_slot, low_slot]]
    pairs = [(a, b) for a in range(1, 5) for b in range(1, 5) if a != b]
    for _ in range(4):
        comparators.append(list(rng.choice(pairs)))

    final = _replay(initial, comparators)
    desired_value = int(seed) % 4 + 1
    query_slot = final.index(desired_value) + 1
    return {
        "initial_values": initial,
        "comparators": comparators,
        "query_slot": query_slot,
        "pip_radius": 9,
        "frame_width": 5,
        "clearance": 15,
    }


def _pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int, radius: int, color: tuple[int, int, int]) -> None:
    offsets = {
        1: [(0, 0)],
        2: [(-13, 0), (13, 0)],
        3: [(-15, -12), (15, -12), (0, 14)],
        4: [(-14, -14), (14, -14), (-14, 14), (14, 14)],
    }[count]
    for dx, dy in offsets:
        draw.ellipse((cx + dx - radius, cy + dy - radius, cx + dx + radius, cy + dy + radius), fill=color)


def render(scene: dict) -> Image.Image:
    initial_values = scene["initial_values"]
    comparators = scene["comparators"]
    query_slot = scene["query_slot"]
    radius = int(scene.get("pip_radius", 9))
    frame_width = int(scene.get("frame_width", 5))

    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((34, 20), "START: token value = green-dot count", fill=GRAY, font=font)

    slot_xs = [135, 335, 535, 735]
    slot_y0, slot_y1 = 65, 235
    for index, (x0, value) in enumerate(zip(slot_xs, initial_values, strict=True), start=1):
        x1 = x0 + 130
        if index == query_slot:
            draw.rectangle((x0 - 12, slot_y0 - 12, x1 + 12, slot_y1 + 12), outline=MAGENTA, width=6)
            draw.text((x0 + 43, slot_y1 + 17), "QUERY", fill=MAGENTA, font=font)
        draw.rectangle((x0, slot_y0, x1, slot_y1), fill=WHITE, outline=BLACK, width=frame_width)
        _pips(draw, x0 + 65, 150, int(value), radius, GREEN)

    draw.text((34, 280), "COMPARE CARDS (left to right)", fill=GRAY, font=font)
    card_y0, card_y1 = 310, 515
    for card_index, (left_slot, right_slot) in enumerate(comparators):
        x0 = 25 + card_index * 195
        x1 = x0 + 170
        draw.rectangle((x0, card_y0, x1, card_y1), fill=WHITE, outline=BLUE, width=frame_width)
        _pips(draw, x0 + 43, 412, int(left_slot), 7, BLUE)
        _pips(draw, x0 + 127, 412, int(right_slot), 7, BLUE)
        draw.line((x0 + 72, 412, x0 + 98, 412), fill=GRAY, width=5)
        draw.polygon([(x0 + 98, 412), (x0 + 87, 404), (x0 + 87, 420)], fill=GRAY)
    return image


def analytic_gold(scene: dict) -> str:
    values = _replay(scene["initial_values"], scene["comparators"])
    return str(values[scene["query_slot"] - 1])


def margin(scene: dict) -> float:
    return float(min(2 * int(scene.get("pip_radius", 9)), int(scene.get("clearance", 15))))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
