"""Analytic sampler and renderer for an ordered adjacent-swap program."""

from __future__ import annotations

import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("anchor", "boot", "chair", "sail")
TEMPLATES = {
    "anchor": (
        "...##...", "...##...", ".######.", "...##...",
        "...##...", ".#.##.#.", "..####..", "...##...",
    ),
    "boot": (
        ".##.....", ".##.....", ".##.....", ".##.....",
        ".##.....", ".##.##..", ".#####..", ".#####..",
    ),
    "chair": (
        ".##..##.", ".##..##.", ".##..##.", ".######.",
        "....##..", "....##..", "..##....", "........",
    ),
    "sail": (
        "...#....", "..##....", ".###....", "####....",
        "..##....", "..##....", ".####...", "######..",
    ),
}

INKS = ((23, 29, 39), (49, 28, 70), (17, 55, 65))
CYAN = (0, 174, 205)
MAGENTA = (224, 32, 126)
PAPER = (249, 248, 244)
CELL_BG = (242, 242, 238)
GUIDE = (174, 180, 182)
DISTRACTOR_COLORS = ((236, 104, 68), (67, 157, 90), (232, 174, 35))


def _template(name: str) -> np.ndarray:
    return np.array([[ch == "#" for ch in row] for row in TEMPLATES[name]], dtype=bool)


def _classification(mask: np.ndarray) -> tuple[str, float]:
    scores = sorted(
        ((64 - int(np.count_nonzero(mask != _template(name))), name) for name in SYMBOLS),
        reverse=True,
    )
    return scores[0][1], float(scores[0][0] - scores[1][0])


def _decoded_output(scene: dict[str, Any]) -> np.ndarray:
    mask = _template(scene["target"]).copy()
    for index in scene["error_cells"]:
        mask.reshape(-1)[int(index)] = ~mask.reshape(-1)[int(index)]
    return mask


def _apply_swaps(mask: np.ndarray, swaps: list[int]) -> np.ndarray:
    result = mask.copy()
    for pair in swaps:
        p = int(pair)
        result[:, [p, p + 1]] = result[:, [p + 1, p]]
    return result


def _source_grid(scene: dict[str, Any]) -> np.ndarray:
    # Adjacent transpositions are self-inverse, so undo the public program in
    # reverse order to construct the visible source raster.
    return _apply_swaps(_decoded_output(scene), list(reversed(scene["swaps"])))


def _sample_swaps(rng: random.Random, target_mask: np.ndarray, length: int) -> list[int]:
    for _ in range(400):
        swaps: list[int] = []
        while len(swaps) < length:
            pair = rng.randrange(7)
            if swaps and pair == swaps[-1]:
                continue
            swaps.append(pair)
        source = _apply_swaps(target_mask, list(reversed(swaps)))
        winner, gap = _classification(source)
        order = list(range(8))
        for pair in swaps:
            order[pair], order[pair + 1] = order[pair + 1], order[pair]
        fixed = sum(index == value for index, value in enumerate(order))
        if fixed <= 3 and (winner != _classification(target_mask)[0] or gap < 4):
            return swaps
    raise RuntimeError("could not sample a sufficiently nontrivial swap ledger")


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    target = SYMBOLS[seed % len(SYMBOLS)]
    error_goal = (seed // 4) % 7
    candidates = list(range(64))
    rng.shuffle(candidates)
    error_cells: list[int] = []
    for index in candidates:
        trial = _template(target).copy()
        for prior in [*error_cells, index]:
            trial.reshape(-1)[prior] = ~trial.reshape(-1)[prior]
        winner, gap = _classification(trial)
        if winner == target and gap >= 4:
            error_cells.append(index)
        if len(error_cells) == error_goal:
            break
    decoded = _template(target).copy()
    for index in error_cells:
        decoded.reshape(-1)[index] = ~decoded.reshape(-1)[index]
    length = 9 + ((seed // 28) % 4)
    swaps = _sample_swaps(rng, decoded, length)
    cell_size = (22, 24, 26)[(seed // 12) % 3]
    offset_x = 116 + rng.randrange(-24, 25)
    offset_y = 43 + rng.randrange(-8, 9)
    distractors = []
    for i in range(2 + (seed // 4) % 3):
        x = 35 + 445 * (i % 2)
        y = 75 + ((seed * 31 + i * 97) % 360)
        distractors.append([x, y, 7 + (seed + i) % 5, i % 3])
    return {
        "target": target,
        "swaps": swaps,
        "cell_size": cell_size,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "ink_index": (seed // 4) % len(INKS),
        "distractors": distractors,
        "error_cells": sorted(error_cells),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (520, 520), PAPER)
    draw = ImageDraw.Draw(image)
    for x, y, radius, color_index in sorted(scene["distractors"]):
        color = DISTRACTOR_COLORS[int(color_index) % len(DISTRACTOR_COLORS)]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    cell = int(scene["cell_size"])
    x0, y0 = int(scene["offset_x"]), int(scene["offset_y"])
    border = 5
    draw.rectangle(
        (x0 - border, y0 - border, x0 + 8 * cell + border - 1, y0 + 8 * cell + border - 1),
        fill=CYAN,
    )
    draw.rectangle((x0, y0, x0 + 8 * cell - 1, y0 + 8 * cell - 1), fill=CELL_BG)
    source = _source_grid(scene)
    ink = INKS[int(scene["ink_index"]) % len(INKS)]
    for row in range(8):
        for col in range(8):
            left, top = x0 + col * cell, y0 + row * cell
            draw.rectangle((left, top, left + cell - 1, top + cell - 1), outline=(192, 195, 194), width=1)
            if bool(source[row, col]):
                draw.rectangle((left + 3, top + 3, left + cell - 4, top + cell - 4), fill=ink)

    ledger_top = y0 + 8 * cell + 28
    step = 15
    # A visible down arrow fixes execution direction without a text payload.
    arrow_x = x0 - 22
    draw.line((arrow_x, ledger_top - 6, arrow_x, ledger_top + step * (len(scene["swaps"]) - 1) + 7), fill=GUIDE, width=2)
    draw.polygon(
        ((arrow_x - 5, ledger_top + step * (len(scene["swaps"]) - 1) + 2),
         (arrow_x + 5, ledger_top + step * (len(scene["swaps"]) - 1) + 2),
         (arrow_x, ledger_top + step * (len(scene["swaps"]) - 1) + 9)),
        fill=GUIDE,
    )
    for stage, pair in enumerate(scene["swaps"]):
        cy = ledger_top + stage * step
        centers = [x0 + col * cell + cell // 2 for col in range(8)]
        draw.line((centers[0], cy, centers[-1], cy), fill=(216, 218, 216), width=1)
        for cx in centers:
            draw.ellipse((cx - 2, cy - 2, cx + 2, cy + 2), fill=GUIDE)
        left, right = centers[int(pair)], centers[int(pair) + 1]
        draw.rectangle((left, cy - 3, right, cy + 3), fill=MAGENTA)
        draw.ellipse((left - 4, cy - 4, left + 4, cy + 4), fill=MAGENTA)
        draw.ellipse((right - 4, cy - 4, right + 4, cy + 4), fill=MAGENTA)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["target"])


def margin(scene: dict) -> float:
    winner, gap = _classification(_decoded_output(scene))
    return gap if winner == scene["target"] else -gap


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractors"] = list(reversed(scene["distractors"]))
    return [("reverse_distractor_storage", twin)]
