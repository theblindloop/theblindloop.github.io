"""Independent pixel-only inverse for the adjacent-swap ledger world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAMES = ("anchor", "boot", "chair", "sail")
_ROWS = {
    "anchor": ("...##...", "...##...", ".######.", "...##...", "...##...", ".#.##.#.", "..####..", "...##..."),
    "boot": (".##.....", ".##.....", ".##.....", ".##.....", ".##.....", ".##.##..", ".#####..", ".#####.."),
    "chair": (".##..##.", ".##..##.", ".##..##.", ".######.", "....##..", "....##..", "..##....", "........"),
    "sail": ("...#....", "..##....", ".###....", "####....", "..##....", "..##....", ".####...", "######.."),
}
_GLYPHS = {
    name: np.array([[value == "#" for value in row] for row in rows], dtype=bool)
    for name, rows in _ROWS.items()
}


def _color_mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 18) -> np.ndarray:
    delta = np.abs(arr.astype(np.int16) - np.array(color, dtype=np.int16))
    return np.max(delta, axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    found: list[tuple[int, int, int, int, int]] = []
    for start_y, start_x in zip(*np.nonzero(work)):
        if not work[start_y, start_x]:
            continue
        queue = deque([(int(start_y), int(start_x))])
        work[start_y, start_x] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _decode(image: Image.Image) -> tuple[np.ndarray, str, float]:
    arr = np.asarray(image.convert("RGB"))
    cyan = _color_mask(arr, (0, 174, 205))
    ys, xs = np.nonzero(cyan)
    if len(xs) < 100:
        raise ValueError("cyan grid frame is missing")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())
    outer_w, outer_h = right - left + 1, bottom - top + 1
    if abs(outer_w - outer_h) > 2 or not 180 <= outer_w <= 220:
        raise ValueError("cyan grid frame is ambiguous")
    middle_x = (left + right) // 2
    border = 0
    while top + border <= bottom and cyan[top + border, middle_x]:
        border += 1
    if not 3 <= border <= 8:
        raise ValueError("grid border cannot be measured")
    interior = outer_w - 2 * border
    if interior % 8:
        raise ValueError("grid does not contain eight equal columns")
    cell = interior // 8
    x0, y0 = left + border, top + border

    source = np.zeros((8, 8), dtype=bool)
    for row in range(8):
        for col in range(8):
            cx = x0 + col * cell + cell // 2
            cy = y0 + row * cell + cell // 2
            patch = arr[cy - 3 : cy + 4, cx - 3 : cx + 4].astype(np.float32)
            luminance = 0.2126 * patch[:, :, 0] + 0.7152 * patch[:, :, 1] + 0.0722 * patch[:, :, 2]
            source[row, col] = float(luminance.mean()) < 125.0

    magenta = _color_mask(arr, (224, 32, 126), 20)
    parts = [part for part in _components(magenta) if part[4] >= 80]
    if not 9 <= len(parts) <= 12:
        raise ValueError(f"expected 9 to 12 swap stages, found {len(parts)}")
    parts.sort(key=lambda part: part[1])
    swaps: list[int] = []
    prior_y = -100
    for x_min, y_min, x_max, y_max, area in parts:
        cy = (y_min + y_max) / 2.0
        cx = (x_min + x_max) / 2.0
        if cy <= bottom + 15 or cy - prior_y < 10:
            raise ValueError("swap stages are missing or out of order")
        pair = int(round((cx - x0) / cell)) - 1
        expected_center = x0 + (pair + 1) * cell
        if not 0 <= pair < 7 or abs(cx - expected_center) > 2.0:
            raise ValueError("swap bar is not aligned to adjacent columns")
        if not cell - 4 <= x_max - x_min <= cell + 10 or y_max - y_min > 12:
            raise ValueError("swap bar geometry is invalid")
        swaps.append(pair)
        prior_y = cy

    output = source.copy()
    for pair in swaps:
        output[:, [pair, pair + 1]] = output[:, [pair + 1, pair]]
    scores = sorted(
        ((64 - int(np.count_nonzero(output != glyph)), name) for name, glyph in _GLYPHS.items()),
        reverse=True,
    )
    gap = float(scores[0][0] - scores[1][0])
    if scores[0][0] == scores[1][0] or gap < 4.0:
        raise ValueError(f"ambiguous glyph; runner-up gap={gap}")
    return output, scores[0][1], gap


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    _, decision, gap = _decode(image)
    return decision, gap


def decision_from_image(image: Image.Image) -> str:
    return _decode(image)[1]
