"""Independent image-only inverse for dark-component ownership of holes."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _label_components(mask: np.ndarray) -> tuple[np.ndarray, int]:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int16)
    next_label = 0
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if labels[y, x] >= 0:
            continue
        labels[y, x] = next_label
        queue = deque([y * width + x])
        while queue:
            value = queue.popleft()
            cy, cx = divmod(value, width)
            if cy and mask[cy - 1, cx] and labels[cy - 1, cx] < 0:
                labels[cy - 1, cx] = next_label
                queue.append(value - width)
            if cy + 1 < height and mask[cy + 1, cx] and labels[cy + 1, cx] < 0:
                labels[cy + 1, cx] = next_label
                queue.append(value + width)
            if cx and mask[cy, cx - 1] and labels[cy, cx - 1] < 0:
                labels[cy, cx - 1] = next_label
                queue.append(value - 1)
            if cx + 1 < width and mask[cy, cx + 1] and labels[cy, cx + 1] < 0:
                labels[cy, cx + 1] = next_label
                queue.append(value + 1)
        next_label += 1
    return labels, next_label


def _marker_centroids(colored: np.ndarray) -> list[tuple[int, int]]:
    labels, count = _label_components(colored)
    centers: list[tuple[int, int]] = []
    for label in range(count):
        ys, xs = np.nonzero(labels == label)
        if len(xs) >= 30:
            centers.append((int(round(float(ys.mean()))), int(round(float(xs.mean())))))
    return centers


def hole_owner_ids_from_image(image: Image.Image) -> tuple[int, int]:
    """Return dark connected-component labels owning the two marker holes."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 128 or rgb.shape[1] < 128:
        raise ValueError("image is too small")
    rgb = rgb[::2, ::2]
    high = rgb.max(axis=2).astype(np.int16)
    low = rgb.min(axis=2).astype(np.int16)
    dark = high < 90
    colored = (high - low > 45) & (high < 245) & ~dark
    centers = _marker_centroids(colored)
    if len(centers) != 2:
        raise ValueError(f"expected two colored marker components, found {len(centers)}")
    dark_labels, dark_count = _label_components(dark)
    if dark_count != 2:
        raise ValueError(f"expected two dark islands, found {dark_count}")
    free_labels, _ = _label_components(~dark)
    owners: list[int] = []
    height, width = dark.shape
    for y, x in centers:
        region = int(free_labels[y, x])
        if region < 0:
            raise ValueError("marker centroid lies on dark foreground")
        region_mask = free_labels == region
        if region_mask[0].any() or region_mask[-1].any() or region_mask[:, 0].any() or region_mask[:, -1].any():
            raise ValueError("marker is not in an enclosed white hole")
        boundary_labels: set[int] = set()
        ys, xs = np.nonzero(region_mask)
        for py, px in zip(ys, xs, strict=True):
            if py and dark[py - 1, px]:
                boundary_labels.add(int(dark_labels[py - 1, px]))
            if py + 1 < height and dark[py + 1, px]:
                boundary_labels.add(int(dark_labels[py + 1, px]))
            if px and dark[py, px - 1]:
                boundary_labels.add(int(dark_labels[py, px - 1]))
            if px + 1 < width and dark[py, px + 1]:
                boundary_labels.add(int(dark_labels[py, px + 1]))
        if len(boundary_labels) != 1:
            raise ValueError(f"hole boundary has {len(boundary_labels)} dark owners")
        owners.append(next(iter(boundary_labels)))
    return owners[0], owners[1]


def decision_from_image(image: Image.Image) -> str:
    first, second = hole_owner_ids_from_image(image)
    return "yes" if first == second else "no"
