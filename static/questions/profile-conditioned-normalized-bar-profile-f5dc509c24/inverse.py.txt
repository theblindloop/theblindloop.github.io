"""Independent pixel-only inverse for normalized bar profiles."""

from __future__ import annotations

from PIL import Image

BAR_COLORS = ((0, 150, 205), (205, 45, 145), (40, 165, 80), (238, 135, 35))
TAG_RGB = {"red": (220, 55, 55), "blue": (55, 100, 220), "gold": (225, 170, 35)}


def _card_index(x: int, y: int) -> int | None:
    if y < 330:
        return 0
    if y <= 350:
        return None
    return 1 + min(2, x // 333)


def _measure(image: Image.Image) -> tuple[list[list[int]], list[str]]:
    """One raster pass; no renderer geometry or scene state is used."""
    bar_index = {rgb: index for index, rgb in enumerate(BAR_COLORS)}
    tag_name = {rgb: name for name, rgb in TAG_RGB.items()}
    bounds = [[None for _ in BAR_COLORS] for _ in range(4)]
    tag_counts = [{name: 0 for name in TAG_RGB} for _ in range(3)]
    pixels = image.load()
    for y in range(image.height):
        for x in range(image.width):
            rgb = pixels[x, y]
            card = _card_index(x, y)
            if card is None:
                continue
            if rgb in bar_index:
                color = bar_index[rgb]
                current = bounds[card][color]
                bounds[card][color] = [x, y, x, y] if current is None else [
                    min(current[0], x), min(current[1], y), max(current[2], x), max(current[3], y)
                ]
            if card > 0 and rgb in tag_name:
                tag_counts[card - 1][tag_name[rgb]] += 1
    profiles = []
    for card_bounds in bounds:
        if any(item is None for item in card_bounds):
            raise ValueError("missing identity-colored bar")
        profiles.append([item[3] - item[1] + 1 for item in card_bounds])
    tags = []
    for counts in tag_counts:
        ordered = sorted(counts, key=counts.get, reverse=True)
        if counts[ordered[0]] < 1000 or counts[ordered[0]] == counts[ordered[1]]:
            raise ValueError("missing or ambiguous option border")
        tags.append(ordered[0])
    return profiles, tags


def _distance(left: list[int], right: list[int]) -> float:
    lm, rm = max(left), max(right)
    return sum(abs(a / lm - b / rm) for a, b in zip(left, right)) / 4


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (1000, 720):
        raise ValueError("unexpected raster size")
    profiles, tags = _measure(rgb)
    distances = [_distance(profiles[0], profiles[index]) for index in range(1, 4)]
    ordered = sorted(distances)
    if ordered[1] - ordered[0] < 0.05:
        raise ValueError("ambiguous normalized-profile assignment")
    match = min(range(3), key=lambda index: distances[index])
    return tags[match]
