"""Latent sampler and deterministic Pillow renderer."""

from __future__ import annotations

import random
from fractions import Fraction

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 720
BAR_COLORS = ((0, 150, 205), (205, 45, 145), (40, 165, 80), (238, 135, 35))
TAG_RGB = {"red": (220, 55, 55), "blue": (55, 100, 220), "gold": (225, 170, 35)}
CARD_BOXES = ((250, 58, 750, 316), (22, 365, 318, 690), (352, 365, 648, 690), (682, 365, 978, 690))


def _profile_distance(left: list[int], right: list[int]) -> float:
    lm = max(left)
    rm = max(right)
    return sum(abs(a / lm - b / rm) for a, b in zip(left, right)) / len(left)


def _scaled(values: list[int], factor: Fraction) -> list[int]:
    return [int(v * factor) for v in values]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    base = rng.sample([48, 56, 64, 72, 80, 88, 96], 4)
    reference = _scaled(base, rng.choice((Fraction(1, 1), Fraction(5, 4))))
    match = _scaled(base, rng.choice((Fraction(5, 4), Fraction(3, 2))))

    distractors: list[list[int]] = []
    attempts = 0
    while len(distractors) < 2:
        attempts += 1
        candidate = _scaled(base, rng.choice((Fraction(1, 1), Fraction(5, 4), Fraction(3, 2))))
        indices = rng.sample(range(4), 2)
        for index in indices:
            candidate[index] = max(32, min(148, candidate[index] + rng.choice((-24, -16, 16, 24))))
        if _profile_distance(reference, candidate) >= 0.09 and candidate not in distractors:
            distractors.append(candidate)
        if attempts > 1000:
            raise RuntimeError("could not sample separated distractors")

    match_index = (seed // 3) % 3
    options = distractors[:]
    options.insert(match_index, match)

    desired_answer = ("red", "blue", "gold")[seed % 3]
    remaining = [value for value in ("red", "blue", "gold") if value != desired_answer]
    rng.shuffle(remaining)
    tags = [None, None, None]
    tags[match_index] = desired_answer
    for index, value in zip((i for i in range(3) if i != match_index), remaining):
        tags[index] = value
    return {"heights": [reference, *options], "option_tags": tags}


def _distances(scene: dict) -> list[float]:
    profiles = scene["heights"]
    return [_profile_distance(profiles[0], item) for item in profiles[1:]]


def analytic_gold(scene: dict) -> str:
    profiles = scene["heights"]
    tags = scene["option_tags"]
    distances = [_profile_distance(profiles[0], item) for item in profiles[1:]]
    best = min(range(3), key=lambda index: distances[index])
    return tags[best]


def margin(scene: dict) -> float:
    ordered = sorted(_distances(scene))
    return (ordered[1] - ordered[0]) * 100.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 7.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], value: str, anchor: str | None = None) -> None:
    draw.text(xy, value, fill=(25, 30, 38), font=ImageFont.load_default(), anchor=anchor)


def _draw_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], heights: list[int], border: tuple[int, int, int], label: str) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=14, fill=(250, 250, 247), outline=border, width=6)
    _text(draw, ((x0 + x1) // 2, y0 + 16), label, anchor="mm")
    baseline = y1 - 34
    draw.line((x0 + 34, baseline + 1, x1 - 34, baseline + 1), fill=(38, 42, 48), width=4)
    available = x1 - x0 - 96
    centers = [x0 + 48 + round(i * available / 3) for i in range(4)]
    for center, height, color in zip(centers, heights, BAR_COLORS):
        draw.rectangle((center - 18, baseline - height + 1, center + 17, baseline), fill=color)


def render(scene: dict) -> Image.Image:
    profiles = scene["heights"]
    tags = scene["option_tags"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (245, 247, 250))
    draw = ImageDraw.Draw(image)
    _text(draw, (WIDTH // 2, 18), "MATCH THE COMPLETE COLOR-TO-HEIGHT PROFILE", anchor="mm")
    _draw_card(draw, CARD_BOXES[0], profiles[0], (75, 82, 92), "REFERENCE")
    for index in range(3):
        _draw_card(draw, CARD_BOXES[index + 1], profiles[index + 1], TAG_RGB[tags[index]], f"OPTION {index + 1}")
    _text(draw, (WIDTH // 2, 708), "Cards may be uniformly resized. Answer with the matching border color.", anchor="ms")
    return image
