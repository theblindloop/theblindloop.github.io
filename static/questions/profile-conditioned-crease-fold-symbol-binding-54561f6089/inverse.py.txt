"""Pixel-only decision arm.

Receives a rendered PNG and nothing else: it segments the dashed crease, fits
its line, locates the red disk and the outlined landing circles, reflects the
red disk across the fitted crease, assigns it to the nearest circle, and only
then classifies the black symbol inside that circle.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_CREASE = (90, 90, 165)
_RING = (120, 120, 120)
_RED = (215, 45, 45)
_TOL = 24
_DARK = 90

_MIN_CREASE_PIXELS = 900
_MIN_DISK_PIXELS = 400
_MIN_RING_PIXELS = 200
_MAX_LANDING_ERROR = 26.0
_MIN_SEPARATION = 20.0


def _mask(array: np.ndarray, colour: tuple[int, int, int]) -> np.ndarray:
    target = np.array(colour, dtype=np.int16)
    return (np.abs(array.astype(np.int16) - target).max(axis=2) <= _TOL)


def _components(mask: np.ndarray, min_size: int) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[np.ndarray] = []
    rows, cols = np.nonzero(mask)
    for start_r, start_c in zip(rows.tolist(), cols.tolist()):
        if seen[start_r, start_c]:
            continue
        stack = deque([(start_r, start_c)])
        seen[start_r, start_c] = True
        pixels = []
        while stack:
            r, c = stack.pop()
            pixels.append((r, c))
            for dr in (-1, 0, 1):
                for dc in (-1, 0, 1):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < mask.shape[0] and 0 <= nc < mask.shape[1]:
                        if mask[nr, nc] and not seen[nr, nc]:
                            seen[nr, nc] = True
                            stack.append((nr, nc))
        if len(pixels) >= min_size:
            out.append(np.array(pixels, dtype=float))
    return out


def _fit_line(points: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    centre = points.mean(axis=0)
    centred = points - centre
    _, _, vectors = np.linalg.svd(centred, full_matrices=False)
    return centre, vectors[0]


def _classify_symbol(dark: np.ndarray) -> str | None:
    rows, cols = np.nonzero(dark)
    if rows.size < 80:
        return None
    height = rows.max() - rows.min() + 1
    width = cols.max() - cols.min() + 1
    if height < 6 or width < 6:
        return None
    fill = rows.size / float(height * width)
    aspect = width / float(height)
    centroid_y = (rows.mean() - rows.min()) / float(height)
    if aspect >= 1.9:
        return "bar"
    if fill >= 0.85:
        return "square"
    if centroid_y >= 0.58:
        return "triangle"
    return "cross"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))

    crease = np.argwhere(_mask(array, _CREASE)).astype(float)
    if crease.shape[0] < _MIN_CREASE_PIXELS:
        return ABSTAIN
    origin, direction = _fit_line(crease)

    red = np.argwhere(_mask(array, _RED)).astype(float)
    if red.shape[0] < _MIN_DISK_PIXELS:
        return ABSTAIN
    red_centre = red.mean(axis=0)

    ring_components = _components(_mask(array, _RING), _MIN_RING_PIXELS)
    if len(ring_components) < 3:
        return ABSTAIN
    centres = np.array([component.mean(axis=0) for component in ring_components])

    offset = red_centre - origin
    landed = origin + (2.0 * float(offset @ direction) * direction - offset)

    distances = np.linalg.norm(centres - landed, axis=1)
    order = np.argsort(distances)
    best = int(order[0])
    if distances[best] > _MAX_LANDING_ERROR:
        return ABSTAIN
    if float(distances[order[1]] - distances[best]) < _MIN_SEPARATION:
        return ABSTAIN

    cr, cc = centres[best]
    r0 = max(0, int(cr) - 24)
    r1 = min(array.shape[0], int(cr) + 25)
    c0 = max(0, int(cc) - 24)
    c1 = min(array.shape[1], int(cc) + 25)
    window = array[r0:r1, c0:c1].astype(np.int16)
    dark = window.max(axis=2) <= _DARK
    symbol = _classify_symbol(dark)
    if symbol is None:
        return ABSTAIN
    return symbol
