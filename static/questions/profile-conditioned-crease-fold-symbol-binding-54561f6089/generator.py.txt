"""Scene sampler, renderer, analytic gold, margin and aliases.

World: a sheet with a dashed crease. Colored disks sit on one side of the
crease; outlined landing circles sit on the other side, each holding one black
symbol. Folding the sheet along the crease reflects every disk onto exactly one
landing circle. The decision is the symbol inside the circle that the red disk
lands on.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = (760, 520)
BACKGROUND = (255, 255, 255)
CREASE = (90, 90, 165)
RING = (120, 120, 120)
GLYPH = (25, 25, 25)

PALETTE = {
    "red": (215, 45, 45),
    "blue": (45, 80, 210),
    "green": (35, 140, 70),
    "orange": (240, 150, 25),
}
COLORS = ("red", "blue", "green", "orange")
SHAPES = ("square", "triangle", "cross", "bar")
QUERY_COLOR = "red"

DISK_RADIUS = 20
RING_RADIUS = 30
RING_WIDTH = 5
CREASE_WIDTH = 7
DASH_ON = 26
DASH_OFF = 18
QUARANTINE_MARGIN = 45.0

_MARGIN_X = 62
_MARGIN_Y = 62


def _unit(angle_deg: float) -> tuple[float, float]:
    rad = math.radians(angle_deg)
    return math.cos(rad), math.sin(rad)


def _normal(angle_deg: float) -> tuple[float, float]:
    ux, uy = _unit(angle_deg)
    return -uy, ux


def _reflect(point, line_point, angle_deg: float) -> tuple[float, float]:
    nx, ny = _normal(angle_deg)
    tx = point[0] - line_point[0]
    ty = point[1] - line_point[1]
    t = tx * nx + ty * ny
    return point[0] - 2.0 * t * nx, point[1] - 2.0 * t * ny


def _signed_distance(point, line_point, angle_deg: float) -> float:
    nx, ny = _normal(angle_deg)
    return (point[0] - line_point[0]) * nx + (point[1] - line_point[1]) * ny


def _dist(a, b) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _inside(point) -> bool:
    return (
        _MARGIN_X <= point[0] <= CANVAS[0] - _MARGIN_X
        and _MARGIN_Y <= point[1] <= CANVAS[1] - _MARGIN_Y
    )


def _rank(values: list[float], index: int) -> int:
    return sorted(range(len(values)), key=lambda i: values[i]).index(index)


def sample_scene(seed: int) -> dict:
    """Deterministically sample one balanced, cue-controlled scene."""
    rng = random.Random(seed)
    for _ in range(6000):
        angle = rng.uniform(50.0, 130.0)
        line_point = [
            CANVAS[0] / 2 + rng.uniform(-70.0, 70.0),
            CANVAS[1] / 2 + rng.uniform(-50.0, 50.0),
        ]
        side = rng.choice((-1.0, 1.0))
        separation_floor = rng.uniform(78.0, 140.0)

        disks: list[list[float]] = []
        rings: list[list[float]] = []
        ok = True
        for _slot in range(4):
            placed = None
            for _try in range(400):
                point = (
                    rng.uniform(_MARGIN_X, CANVAS[0] - _MARGIN_X),
                    rng.uniform(_MARGIN_Y, CANVAS[1] - _MARGIN_Y),
                )
                signed = _signed_distance(point, line_point, angle)
                if not 58.0 <= side * signed <= 190.0:
                    continue
                mirror = _reflect(point, line_point, angle)
                if not _inside(mirror):
                    continue
                if any(_dist(point, other) < separation_floor for other in disks):
                    continue
                if any(_dist(mirror, other) < 78.0 for other in rings):
                    continue
                if any(_dist(point, other) < 78.0 for other in rings):
                    continue
                if any(_dist(mirror, other) < 78.0 for other in disks):
                    continue
                placed = (point, mirror)
                break
            if placed is None:
                ok = False
                break
            disks.append([placed[0][0], placed[0][1]])
            rings.append([placed[1][0], placed[1][1]])
        if not ok:
            continue

        disk_colors = list(COLORS)
        rng.shuffle(disk_colors)
        query = disk_colors.index(QUERY_COLOR)

        # Cheap-cue control: the landing circle must not be the circle nearest to
        # the red disk, nor the circle at the red disk's own vertical rank, nor
        # the circle picked by mirroring about the canvas midline.
        nearest = min(range(4), key=lambda i: _dist(disks[query], rings[i]))
        if nearest == query:
            continue
        if _rank([d[1] for d in disks], query) == _rank([r[1] for r in rings], query):
            continue
        fake = (CANVAS[0] - disks[query][0], disks[query][1])
        if min(range(4), key=lambda i: _dist(fake, rings[i])) == query:
            continue

        shapes = list(SHAPES)
        rng.shuffle(shapes)

        # Independent storage orders for disks and rings (label-preserving).
        disk_order = list(range(4))
        rng.shuffle(disk_order)
        ring_order = list(range(4))
        rng.shuffle(ring_order)
        scene = {
            "line_point": [round(line_point[0], 4), round(line_point[1], 4)],
            "line_angle_deg": round(angle, 4),
            "disks": [[round(disks[i][0], 4), round(disks[i][1], 4)] for i in disk_order],
            "disk_colors": [disk_colors[i] for i in disk_order],
            "rings": [[round(rings[i][0], 4), round(rings[i][1], 4)] for i in ring_order],
            "ring_shapes": [shapes[i] for i in ring_order],
        }
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _query_disk(scene: dict) -> list[float]:
    index = scene["disk_colors"].index(QUERY_COLOR)
    return scene["disks"][index]


def landing_index(scene: dict) -> int:
    """Index (in storage order) of the circle the red disk folds onto."""
    target = _reflect(_query_disk(scene), scene["line_point"], scene["line_angle_deg"])
    return min(range(len(scene["rings"])), key=lambda i: _dist(target, scene["rings"][i]))


def analytic_gold(scene: dict) -> str:
    """Symbol inside the landing circle of the red disk."""
    return scene["ring_shapes"][landing_index(scene)]


def margin(scene: dict) -> float:
    """Half the gap between the landing circle and its nearest rival circle."""
    index = landing_index(scene)
    centre = scene["rings"][index]
    rivals = [
        _dist(centre, other) for i, other in enumerate(scene["rings"]) if i != index
    ]
    return round(0.5 * min(rivals), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, label-preserving re-orderings of the latent lists."""
    out: list[tuple[str, dict]] = []

    rotated_disks = {
        **scene,
        "disks": scene["disks"][1:] + scene["disks"][:1],
        "disk_colors": scene["disk_colors"][1:] + scene["disk_colors"][:1],
    }
    out.append(("disk_index_rotation", rotated_disks))

    order = [2, 0, 3, 1]
    permuted_rings = {
        **scene,
        "rings": [scene["rings"][i] for i in order],
        "ring_shapes": [scene["ring_shapes"][i] for i in order],
    }
    out.append(("ring_index_permutation", permuted_rings))

    both = {
        **permuted_rings,
        "disks": rotated_disks["disks"],
        "disk_colors": rotated_disks["disk_colors"],
    }
    out.append(("joint_index_permutation", both))
    return out


def _draw_glyph(draw: ImageDraw.ImageDraw, centre, shape: str) -> None:
    cx, cy = int(round(centre[0])), int(round(centre[1]))
    if shape == "square":
        draw.rectangle([cx - 11, cy - 11, cx + 10, cy + 10], fill=GLYPH)
    elif shape == "bar":
        draw.rectangle([cx - 15, cy - 5, cx + 14, cy + 5], fill=GLYPH)
    elif shape == "triangle":
        draw.polygon(
            [(cx, cy - 12), (cx + 13, cy + 11), (cx - 13, cy + 11)], fill=GLYPH
        )
    elif shape == "cross":
        draw.rectangle([cx - 13, cy - 4, cx + 13, cy + 4], fill=GLYPH)
        draw.rectangle([cx - 4, cy - 13, cx + 4, cy + 13], fill=GLYPH)
    else:  # pragma: no cover - guarded by the sampler
        raise ValueError(f"unknown shape {shape!r}")


def _draw_crease(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    ux, uy = _unit(scene["line_angle_deg"])
    px, py = scene["line_point"]
    step = DASH_ON + DASH_OFF
    start = -900.0
    while start < 900.0:
        end = start + DASH_ON
        draw.line(
            [
                int(round(px + ux * start)),
                int(round(py + uy * start)),
                int(round(px + ux * end)),
                int(round(py + uy * end)),
            ],
            fill=CREASE,
            width=CREASE_WIDTH,
        )
        start += step


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_crease(scene, draw)
    for centre, shape in zip(scene["rings"], scene["ring_shapes"]):
        cx, cy = int(round(centre[0])), int(round(centre[1]))
        draw.ellipse(
            [cx - RING_RADIUS, cy - RING_RADIUS, cx + RING_RADIUS, cy + RING_RADIUS],
            outline=RING,
            width=RING_WIDTH,
        )
        _draw_glyph(draw, (cx, cy), shape)
    for centre, colour in zip(scene["disks"], scene["disk_colors"]):
        cx, cy = int(round(centre[0])), int(round(centre[1]))
        draw.ellipse(
            [cx - DISK_RADIUS, cy - DISK_RADIUS, cx + DISK_RADIUS, cy + DISK_RADIUS],
            fill=PALETTE[colour],
        )
    return image
