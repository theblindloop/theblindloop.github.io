"""Independent image-only inverse for Latin-quilt search."""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

PALETTE = np.asarray(
    ((34, 112, 168), (221, 73, 64), (63, 151, 91), (239, 178, 45)),
    dtype=np.uint8,
)
LABELS = ("A", "B", "C", "D")


def _score(patch: np.ndarray) -> int:
    rows = sum(len(set(int(value) for value in patch[row])) == 4 for row in range(4))
    columns = sum(
        len(set(int(value) for value in patch[:, column])) == 4 for column in range(4)
    )
    return rows + columns


def _read_grid(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    matches = np.all(array[:, :, None, :] == PALETTE[None, None, :, :], axis=3)
    colored = matches.any(axis=2)
    points = np.argwhere(colored)
    if len(points) < 10000:
        raise ValueError("insufficient colored quilt area")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width = int(x1 - x0 + 1)
    height = int(y1 - y0 + 1)
    cell_x = int(round(width / 8.0))
    cell_y = int(round(height / 8.0))
    if cell_x < 45 or cell_y < 45 or abs(width - 8 * cell_x) > 2 or abs(height - 8 * cell_y) > 2:
        raise ValueError("quilt bounds do not describe an 8 by 8 grid")
    if abs(cell_x - cell_y) > 1:
        raise ValueError("quilt cells are not square")

    grid = np.empty((8, 8), dtype=np.int8)
    for row in range(8):
        for column in range(8):
            cx = int(round(x0 + (column + 0.5) * cell_x - 0.5))
            cy = int(round(y0 + (row + 0.5) * cell_y - 0.5))
            window = array[cy - 4 : cy + 5, cx - 4 : cx + 5]
            counts = np.asarray(
                [np.all(window == color, axis=2).sum() for color in PALETTE], dtype=int
            )
            best = int(np.argmax(counts))
            if counts[best] < 65 or (np.sort(counts)[-1] - np.sort(counts)[-2]) < 50:
                raise ValueError("a quilt cell is missing or ambiguous")
            grid[row, column] = best
    return grid


def decision_from_image(image: Image.Image) -> str:
    grid = _read_grid(image)
    patches = (
        grid[0:4, 0:4],
        grid[0:4, 4:8],
        grid[4:8, 0:4],
        grid[4:8, 4:8],
    )
    scores = [_score(patch) for patch in patches]
    winners = [index for index, score in enumerate(scores) if score == 8]
    ranked = sorted(scores, reverse=True)
    if len(winners) != 1 or ranked[0] - ranked[1] < 2:
        raise ValueError("no unique well-separated Latin patch")
    return LABELS[winners[0]]
