"""Deterministic analytic world for the Latin-quilt visual search."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
BASE_COLORS = (
    (34, 112, 168),
    (221, 73, 64),
    (63, 151, 91),
    (239, 178, 45),
)
LABELS = ("A", "B", "C", "D")


def _patch_score(patch: list[list[int]]) -> int:
    rows = sum(len(set(row)) == 4 for row in patch)
    columns = sum(len({patch[row][column] for row in range(4)}) == 4 for column in range(4))
    return rows + columns


def _scores(scene: dict[str, Any]) -> list[int]:
    return [_patch_score(patch) for patch in scene["patches"]]


def _latin_patch(rng: random.Random) -> list[list[int]]:
    # Alternate between the cyclic and Klein-four Latin tables, then independently
    # permute rows, columns, and symbols to prevent a fixed diagonal texture.
    if rng.randrange(2):
        base = [[(row + column) % 4 for column in range(4)] for row in range(4)]
    else:
        base = [[row ^ column for column in range(4)] for row in range(4)]
    rows = list(range(4))
    columns = list(range(4))
    symbols = list(range(4))
    rng.shuffle(rows)
    rng.shuffle(columns)
    rng.shuffle(symbols)
    return [[symbols[base[rows[row]][columns[column]]] for column in range(4)] for row in range(4)]


def _near_latin(source: list[list[int]], rng: random.Random, kind: int) -> list[list[int]]:
    """Preserve each color's total count while breaking different line constraints."""
    for _ in range(200):
        patch = copy.deepcopy(source)
        if kind == 0:
            row = rng.randrange(4)
            left, right = rng.sample(range(4), 2)
            patch[row][left], patch[row][right] = patch[row][right], patch[row][left]
        elif kind == 1:
            column = rng.randrange(4)
            top, bottom = rng.sample(range(4), 2)
            patch[top][column], patch[bottom][column] = patch[bottom][column], patch[top][column]
        else:
            for _swap in range(2):
                a, b = rng.sample(range(16), 2)
                ar, ac = divmod(a, 4)
                br, bc = divmod(b, 4)
                patch[ar][ac], patch[br][bc] = patch[br][bc], patch[ar][ac]
        if patch != source and _patch_score(patch) <= 6:
            return patch
    raise RuntimeError("could not construct a near-Latin distractor")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    target = _latin_patch(rng)
    distractors: list[list[list[int]]] = []
    for kind in range(3):
        for _ in range(100):
            candidate = _near_latin(target, rng, kind)
            if candidate not in distractors:
                distractors.append(candidate)
                break
        else:
            raise RuntimeError("could not construct distinct distractors")

    target_index = int(seed) % 4
    patches = []
    distractor_iter = iter(distractors)
    for index in range(4):
        patches.append(copy.deepcopy(target if index == target_index else next(distractor_iter)))

    palette = list(BASE_COLORS)
    rng.shuffle(palette)
    cell_size = rng.choice((50, 51, 52, 53))
    quilt_size = 8 * cell_size
    origin_x = (CANVAS - quilt_size) // 2 + rng.randint(-3, 3)
    origin_y = (CANVAS - quilt_size) // 2 + rng.randint(-3, 3)
    return {
        "patches": patches,
        "palette": [list(color) for color in palette],
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
    }


def _cell_value(scene: dict[str, Any], global_row: int, global_column: int) -> int:
    patch_index = (global_row // 4) * 2 + global_column // 4
    return scene["patches"][patch_index][global_row % 4][global_column % 4]


def render(scene: dict[str, Any]) -> Image.Image:
    palette = [tuple(color) for color in scene["palette"]]
    cell = int(scene["cell_size"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    image = Image.new("RGB", (CANVAS, CANVAS), (248, 246, 239))
    draw = ImageDraw.Draw(image)
    for row in range(8):
        for column in range(8):
            value = _cell_value(scene, row, column)
            left = x0 + column * cell
            top = y0 + row * cell
            draw.rectangle((left, top, left + cell - 1, top + cell - 1), fill=palette[value])

    # One perimeter and one cross articulate four candidates without separating
    # the underlying high-fill carpet into detached panels.
    dark = (43, 47, 48)
    gold = (255, 224, 132)
    size = 8 * cell
    draw.rectangle((x0 - 5, y0 - 5, x0 + size + 4, y0 + size + 4), outline=dark, width=5)
    draw.line((x0 + 4 * cell, y0, x0 + 4 * cell, y0 + size), fill=gold, width=7)
    draw.line((x0, y0 + 4 * cell, x0 + size, y0 + 4 * cell), fill=gold, width=7)
    draw.line((x0 + 4 * cell, y0, x0 + 4 * cell, y0 + size), fill=dark, width=2)
    draw.line((x0, y0 + 4 * cell, x0 + size, y0 + 4 * cell), fill=dark, width=2)

    font = ImageFont.load_default()
    for index, label in enumerate(LABELS):
        patch_row, patch_column = divmod(index, 2)
        # The center of a 4x4 patch is a four-cell intersection, deliberately
        # distant from every cell-center sample used by the independent oracle.
        center_x = x0 + patch_column * 4 * cell + 2 * cell
        center_y = y0 + patch_row * 4 * cell + 2 * cell
        draw.rounded_rectangle(
            (center_x - 11, center_y - 9, center_x + 11, center_y + 9),
            radius=4,
            fill=gold,
            outline=dark,
            width=2,
        )
        draw.text((center_x, center_y), label, fill=dark, anchor="mm", font=font)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = _scores(scene)
    winners = [index for index, score in enumerate(scores) if score == 8]
    if len(winners) != 1:
        raise ValueError("scene has no unique Latin patch")
    return LABELS[winners[0]]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    scores = sorted(_scores(scene), reverse=True)
    return scores[0] != 8 or scores.count(8) != 1 or scores[0] - scores[1] < 2


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["patches"] = [
        [[(value + 1) % 4 for value in row] for row in patch] for patch in scene["patches"]
    ]
    old_palette = scene["palette"]
    new_palette: list[list[int] | None] = [None, None, None, None]
    for old_value in range(4):
        new_palette[(old_value + 1) % 4] = list(old_palette[old_value])
    twin["palette"] = new_palette
    return [("cyclic_category_relabel", twin)]
