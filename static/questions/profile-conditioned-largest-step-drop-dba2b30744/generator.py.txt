"""Deterministic latent sampling, rendering, and analytic decision arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 600
TOTAL_DECLINE = 64
QUARANTINE_GAP_PX = 8

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (214, 54, 64),
    "blue": (40, 112, 184),
    "green": (35, 142, 82),
    "orange": (226, 126, 34),
}
COLORS = tuple(PALETTE)


def _partition_with_unique_max(rng: random.Random, peak: int) -> list[int]:
    """Return four positive integer drops totaling TOTAL_DECLINE with max=peak."""
    for _ in range(10000):
        remaining = TOTAL_DECLINE - peak
        a = rng.randint(7, min(peak - 1, remaining - 14))
        b = rng.randint(7, min(peak - 1, remaining - a - 7))
        c = remaining - a - b
        if 7 <= c < peak and len({peak, a, b, c}) >= 3:
            values = [peak, a, b, c]
            rng.shuffle(values)
            return values
    raise RuntimeError(f"could not construct drop partition for peak={peak}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5EED5A17)
    winner_index = int(seed) % len(COLORS)
    near_boundary = int(seed) % 11 == 0
    ordered_peaks = [30, 25, 22, 19] if near_boundary else [36, 26, 23, 20]

    other_colors = [color for index, color in enumerate(COLORS) if index != winner_index]
    rng.shuffle(other_colors)
    peak_by_color = {COLORS[winner_index]: ordered_peaks[0]}
    for color, peak in zip(other_colors, ordered_peaks[1:], strict=True):
        peak_by_color[color] = peak

    panel_colors = list(COLORS)
    rng.shuffle(panel_colors)
    curves = []
    for panel, color in enumerate(panel_colors):
        base_x = [220, 335, 450, 560]
        xs = [value + rng.randint(-13, 13) for value in base_x]
        curves.append(
            {
                "color": color,
                "panel": panel,
                "drops": _partition_with_unique_max(rng, peak_by_color[color]),
                "drop_x": xs,
            }
        )
    return {"curves": curves}


def analytic_gold(scene: dict) -> str:
    curves = scene["curves"]
    ranked = sorted(
        ((max(int(value) for value in curve["drops"]), str(curve["color"])) for curve in curves),
        reverse=True,
    )
    if len(ranked) != 4 or ranked[0][0] == ranked[1][0]:
        raise ValueError("scene does not have a unique largest one-step drop")
    return ranked[0][1]


def margin(scene: dict) -> float:
    peaks = sorted(
        (max(int(value) for value in curve["drops"]) for curve in scene["curves"]),
        reverse=True,
    )
    return float(peaks[0] - peaks[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["curves"] = list(reversed(twin["curves"]))
    return [("reverse_curve_list", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (246, 248, 251))
    draw = ImageDraw.Draw(image)
    draw.text((82, 20), "ONE-STEP DECLINES", fill=(37, 45, 58), font=_font(22, bold=True))
    draw.text(
        (82, 50),
        "Equal total decline  •  compare the longest vertical step",
        fill=(91, 101, 116),
        font=_font(14),
    )

    for panel in range(4):
        top = 88 + panel * 124
        bottom = top + 104
        draw.rounded_rectangle(
            (72, top, 680, bottom), radius=8, fill=(255, 255, 255), outline=(218, 223, 231), width=1
        )
        draw.line((132, top + 15, 132, top + 85), fill=(205, 211, 220), width=1)
        draw.line((132, top + 85, 650, top + 85), fill=(205, 211, 220), width=1)
        for offset in (31, 47, 63):
            draw.line((133, top + offset, 650, top + offset), fill=(238, 241, 245), width=1)
        draw.text((88, top + 42), str(panel + 1), fill=(116, 125, 139), font=_font(13, bold=True))

    for curve in scene["curves"]:
        color_name = str(curve["color"])
        panel = int(curve["panel"])
        drops = [int(value) for value in curve["drops"]]
        xs = [int(value) for value in curve["drop_x"]]
        if color_name not in PALETTE or sorted(xs) != xs or len(drops) != 4 or sum(drops) != TOTAL_DECLINE:
            raise ValueError("invalid curve scene")
        top = 88 + panel * 124
        y = top + 16
        points: list[tuple[int, int]] = [(145, y)]
        for x, drop in zip(xs, drops, strict=True):
            points.append((x, y))
            y += drop
            points.append((x, y))
        points.append((642, y))
        draw.line(points, fill=PALETTE[color_name], width=7, joint="curve")

    draw.text(
        (515, 576), "time  →", fill=(91, 101, 116), font=_font(13, bold=True)
    )
    return image
