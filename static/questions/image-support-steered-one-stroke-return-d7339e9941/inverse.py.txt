"""Independent image-only inverse arm for the localized Euler-tour graph."""

from __future__ import annotations

from collections import deque
from itertools import combinations

import numpy as np
from PIL import Image


_INKS = np.asarray([(25, 52, 68), (66, 42, 92)], dtype=np.int32)
_STATIONS = np.asarray([(238, 166, 62), (70, 177, 164)], dtype=np.int32)


class OracleAbstention(ValueError):
    """The visible public graph contract is missing or only partially present."""


def _near_any(array: np.ndarray, colors: np.ndarray, squared_radius: int) -> np.ndarray:
    delta = array[:, :, None, :].astype(np.int32) - colors[None, None, :, :]
    return np.min(np.sum(delta * delta, axis=3), axis=2) <= squared_radius


def _components(mask: np.ndarray, minimum: int = 18) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _station_centers(array: np.ndarray) -> list[tuple[float, float]]:
    parts = _components(_near_any(array, _STATIONS, 1_500))
    if len(parts) != 6:
        raise OracleAbstention(f"expected six visible station interiors, found {len(parts)}")
    centers: list[tuple[float, float]] = []
    for part in parts:
        ys, xs = part[:, 0], part[:, 1]
        width, height = xs.max() - xs.min(), ys.max() - ys.min()
        if not 28 <= len(part) <= 150 or not 7 <= width <= 14 or not 7 <= height <= 14:
            raise OracleAbstention("a station interior is clipped or malformed")
        centers.append((float(xs.mean()), float(ys.mean())))
    return centers


def _corridor_score(
    ink: np.ndarray, first: tuple[float, float], second: tuple[float, float]
) -> float:
    x1, y1 = first
    x2, y2 = second
    length = float(np.hypot(x2 - x1, y2 - y1))
    # The template has no links this short; station outlines can nearly bridge
    # the small gap between this one nonadjacent pair, so reject it before
    # corridor scoring rather than mistaking two rims for an edge.
    if length < 35.0:
        return 0.0
    guard = 9.0
    count = max(12, int(length - 2.0 * guard) + 1)
    ts = np.linspace(guard / length, 1.0 - guard / length, count)
    supported = 0
    for value in ts:
        x = int(round(x1 + value * (x2 - x1)))
        y = int(round(y1 + value * (y2 - y1)))
        patch = ink[max(0, y - 1): y + 2, max(0, x - 1): x + 2]
        supported += bool(patch.size and float(patch.mean()) >= 0.34)
    return supported / len(ts)


def _connected(edges: list[tuple[int, int]], count: int) -> bool:
    adjacency = [set() for _ in range(count)]
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    reached = {0}
    pending = [0]
    while pending:
        current = pending.pop()
        for neighbor in adjacency[current]:
            if neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return len(reached) == count


def decision_from_image(image: Image.Image) -> str:
    """Recover the closed-Euler-tour answer from final RGB pixels alone."""
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    centers = _station_centers(array)
    ink = _near_any(array, _INKS, 1_800)
    edges: list[tuple[int, int]] = []
    for left, right in combinations(range(len(centers)), 2):
        score = _corridor_score(ink, centers[left], centers[right])
        if score >= 0.96:
            edges.append((left, right))
        elif score > 0.90:
            raise OracleAbstention(f"a possible link is only partially visible ({score:.2f})")
    if len(edges) != 7:
        raise OracleAbstention(f"expected seven complete links, found {len(edges)}")
    if not _connected(edges, 6):
        raise OracleAbstention("the recovered station graph is disconnected")
    degree = [0] * 6
    for left, right in edges:
        degree[left] += 1
        degree[right] += 1
    ordered = sorted(degree)
    if ordered == [2, 2, 2, 2, 2, 4]:
        return "yes"
    if ordered == [1, 2, 2, 2, 3, 4]:
        return "no"
    raise OracleAbstention(f"degree pattern {ordered} is outside the public world")
