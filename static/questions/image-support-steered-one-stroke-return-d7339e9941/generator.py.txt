"""Deterministic localized graph world for a closed one-stroke tour."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


CANVAS = 640
BACKGROUND = (249, 248, 244)
PALE = (218, 220, 216)
PALE_2 = (231, 228, 220)
PALETTES = (
    ((25, 52, 68), (238, 166, 62)),
    ((66, 42, 92), (70, 177, 164)),
)


def _template_nodes() -> list[list[int]]:
    # Central station, far-left station, upper-left station, upper-right
    # station, far-right station, lower-right station.  The whole graph is a
    # 190-by-36 pixel strip including station discs.
    return [[0, 0], [-88, 0], [-44, -11], [44, -11], [88, 0], [44, 11]]


def _transform_nodes(
    nodes: list[list[int]], *, reflect_x: bool, reflect_y: bool, vertical: bool,
    center_x: int, center_y: int,
) -> list[list[int]]:
    output: list[list[int]] = []
    for x, y in nodes:
        if reflect_x:
            x = -x
        if reflect_y:
            y = -y
        if vertical:
            x, y = -y, x
        output.append([center_x + x, center_y + y])
    return output


def sample_scene(seed: int) -> dict:
    """Sample exact class balance with paired nuisance configurations."""
    seed = int(seed)
    decision_yes = seed % 2 == 0
    # Consecutive even/odd scenes share every nuisance, preventing class cues
    # from position, orientation, reflection, palette, or decoration.
    rng = random.Random((seed // 2) * 1_000_003 + 71)
    vertical = bool(rng.randrange(2))
    reflect_x = bool(rng.randrange(2))
    reflect_y = bool(rng.randrange(2))
    palette_variant = rng.randrange(len(PALETTES))

    triplet_start = rng.randrange(6)
    orthogonal_lane = rng.randrange(8)
    if vertical:
        center_x = 80 * orthogonal_lane + 40
        center_y = 80 * triplet_start + 120
    else:
        center_x = 80 * triplet_start + 120
        center_y = 80 * orthogonal_lane + 40
    nodes = _transform_nodes(
        _template_nodes(), reflect_x=reflect_x, reflect_y=reflect_y,
        vertical=vertical, center_x=center_x, center_y=center_y,
    )

    # YES is a triangle and a four-cycle sharing the central station.  NO
    # replaces the equal-length central-to-left link by a central-to-right
    # chord.  Both classes therefore have six stations, seven links, identical
    # link-length multisets, and identical bounding extents.
    common = [[1, 2], [2, 0], [0, 3], [3, 4], [4, 5], [5, 0]]
    class_link = [0, 1] if decision_yes else [0, 4]
    edges = [class_link, *common]

    min_x = min(point[0] for point in nodes) - 28
    max_x = max(point[0] for point in nodes) + 28
    min_y = min(point[1] for point in nodes) - 28
    max_y = max(point[1] for point in nodes) + 28
    decorations: list[list[int | str]] = []
    attempts = 0
    while len(decorations) < 10 and attempts < 500:
        attempts += 1
        x, y = rng.randint(20, CANVAS - 21), rng.randint(20, CANVAS - 21)
        if min_x <= x <= max_x and min_y <= y <= max_y:
            continue
        kind = "ring" if len(decorations) % 2 == 0 else "dash"
        decorations.append([x, y, rng.choice((5, 6, 7)), kind])

    return {
        "canvas_size": CANVAS,
        "nodes": nodes,
        "edges": edges,
        "stroke_width": 5,
        "node_radius": 7,
        "palette_variant": palette_variant,
        "decorations": decorations,
    }


def _degree_counts(scene: dict) -> list[int]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    degree = [0] * len(nodes)
    for left, right in edges:
        degree[int(left)] += 1
        degree[int(right)] += 1
    return degree


def _is_connected(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    adjacency = [set() for _ in nodes]
    for left, right in edges:
        adjacency[int(left)].add(int(right))
        adjacency[int(right)].add(int(left))
    reached = {0}
    pending = [0]
    while pending:
        current = pending.pop()
        for neighbor in adjacency[current]:
            if neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return len(reached) == len(nodes)


def analytic_gold(scene: dict) -> str:
    """Euler's theorem: a connected graph has a closed tour iff all degrees are even."""
    if not _is_connected(scene):
        raise ValueError("the public scene requires one connected graph")
    return "yes" if all(value % 2 == 0 for value in _degree_counts(scene)) else "no"


def margin(scene: dict) -> float:
    """Visible link-interior clearance after station discs and stroke thickness."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    radius = float(scene["node_radius"])
    stroke = float(scene["stroke_width"])
    shortest = min(math.dist(nodes[left], nodes[right]) for left, right in edges)
    return float(shortest - 2.0 * radius - stroke)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def render(scene: dict) -> Image.Image:
    """Render only public scene fields, using deterministic supersampling."""
    size = int(scene["canvas_size"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    stroke = int(scene["stroke_width"])
    radius = int(scene["node_radius"])
    ink, station = PALETTES[int(scene["palette_variant"])]
    decorations = scene["decorations"]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, extent, kind in decorations:
        x, y, extent = int(x) * scale, int(y) * scale, int(extent) * scale
        if kind == "ring":
            draw.ellipse([x - extent, y - extent, x + extent, y + extent], outline=PALE, width=2 * scale)
        else:
            draw.line([x - extent, y, x + extent, y], fill=PALE_2, width=2 * scale)

    for left, right in edges:
        first = tuple(int(value * scale) for value in nodes[int(left)])
        second = tuple(int(value * scale) for value in nodes[int(right)])
        draw.line([first, second], fill=ink, width=stroke * scale)

    for x, y in nodes:
        box = [
            (x - radius) * scale, (y - radius) * scale,
            (x + radius) * scale, (y + radius) * scale,
        ]
        draw.ellipse(box, fill=station, outline=ink, width=2 * scale)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Same-color edge storage order is unobservable and label preserving."""
    twin = dict(scene)
    twin["edges"] = [list(edge) for edge in reversed(scene["edges"])]
    return [("reverse_link_storage_order", twin)]
