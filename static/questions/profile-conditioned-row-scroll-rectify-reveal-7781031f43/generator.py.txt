"""row_scroll_rectify_reveal: row-scroll (shear) cipher glyph reveal world.

Declared transform: undo a deterministic CYCLIC ROW SCROLL. The image is a coarse
32x32 cell grid whose pixel rows have each been cyclically scrolled by a fixed
ramp offset (shift_r = 7*r mod 32), smearing the glyph into an oblique shear that
is not readable directly. Applying the declared inverse cyclic scroll to every
row restores the aligned binary grid, which is one of two declared abstract
symbols: a hollow square ring (A) or a diagonal cross X (B).

The raw image alone shows only a sheared smear; the answer A or B appears only
after executing the declared unscroll (each row cyclically rolled back by 7*r
mod 32) and reading the reconstructed symbol.

Latent scene z = {glyph, invert}. label = glyph. The row-scroll ramp is a fixed,
question-declared constant (stride 7), never latent.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "row_scroll_rectify_reveal-0.1.0"

W = 256
H = 256
NCELL = 32
CELL = W // NCELL   # 8 px per cell
STRIDE = 7          # fixed, question-declared row-scroll ramp factor (coprime to 32)
SCROLL_TOL = 40.0   # quarantine threshold (cells) on the A/B reconstruction gap


def _glyph_grid(glyph: str) -> np.ndarray:
    """(NCELL, NCELL) uint8 grid: 255 = dark cell, 0 = light cell."""
    g = np.zeros((NCELL, NCELL), dtype=np.uint8)
    y, x = np.indices((NCELL, NCELL))
    if glyph == "A":
        inside = (y >= 8) & (y <= 23) & (x >= 8) & (x <= 23)
        edge = (y == 8) | (y == 23) | (x == 8) | (x == 23)
        g[inside & edge] = 255
    else:
        band1 = np.abs(y - x) <= 2
        band2 = np.abs(y + x - (NCELL - 1)) <= 2
        g[band1 | band2] = 255
    return g


def _scrolled(grid: np.ndarray, rolls: np.ndarray) -> np.ndarray:
    """Cyclically scroll row r left by rolls[r] cell positions... applied once."""
    out = np.empty_like(grid)
    for r in range(NCELL):
        out[r] = np.roll(grid[r], -int(rolls[r]))
    return out


def _unscrolled(grid: np.ndarray, rolls: np.ndarray) -> np.ndarray:
    """Cyclically scroll row r right by rolls[r] (inverse of _scrolled)."""
    out = np.empty_like(grid)
    for r in range(NCELL):
        out[r] = np.roll(grid[r], int(rolls[r]))
    return out


def _rolls() -> np.ndarray:
    r = np.arange(NCELL)
    return (STRIDE * r) % NCELL


def build_scene(glyph: str, invert: bool) -> dict:
    return {"glyph": glyph, "invert": bool(invert)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(glyph=glyph, invert=bool(int(seed) % 4 == 0))


def _to_pixels(grid: np.ndarray, invert: bool) -> np.ndarray:
    arr = np.repeat(np.repeat(grid, CELL, axis=0), CELL, axis=1)
    if invert:
        arr = 255 - arr
    return arr.astype(np.uint8)


def render(scene: dict) -> Image.Image:
    """Render the sheared (row-scrolled) cipher image; symbol hidden directly."""
    grid = _glyph_grid(scene["glyph"])
    scrambled = _scrolled(grid, _rolls())
    arr = _to_pixels(scrambled, bool(scene.get("invert", False)))
    return Image.fromarray(arr, "L").convert("RGB")


def reconstruct(grid: np.ndarray) -> np.ndarray:
    """Apply the declared inverse row scroll to a (NCELL, NCELL) cell-grid image."""
    return _unscrolled(grid, _rolls())


def _extract_grid(image: Image.Image) -> np.ndarray:
    """Downsample rendered pixels into an NCELLxNCELL bool grid (dark=True)."""
    arr = np.asarray(image.convert("L"), dtype=np.uint8).reshape(NCELL, CELL, NCELL, CELL)
    avg = arr.mean(axis=(1, 3))
    dark = avg < 128
    if dark.sum() > (NCELL * NCELL) / 2:
        dark = ~dark
    return dark


def margin(scene: dict) -> float:
    """A/B reconstruction gap: popcount(diffB) - popcount(diffA) in cells."""
    grid = _extract_grid(render(scene)) * 1.0
    ga = _glyph_grid("A") > 0
    gb = _glyph_grid("B") > 0
    diffA = float(np.count_nonzero(grid != ga))
    diffB = float(np.count_nonzero(grid != gb))
    return round(diffB - diffA, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < SCROLL_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (glyph, invert) directly changes the
    rendered pixels; the row-scroll transform is a fixed deterministic cipher so
    no two distinct latent vectors produce the same raster with different labels.
    Declared empty."""
    return iter(())
