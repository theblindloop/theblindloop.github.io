"""Independent pixel-only inverse arm for row_scroll_rectify_reveal.

Receives ONLY the final PNG. It executes the declared inverse row scroll: it
downsamples the raster to the declared 32x32 cell grid, normalizes polarity
(dark minority), cyclically rolls each row back by 7*r mod 32 (the question's
declared cipher), and reads the reconstructed binary symbol by matching the
recovered grid against the two declared reference glyphs A (hollow square ring)
and B (diagonal cross X). It reads no scene data and no analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "row_scroll_rectify_reveal-oracle-0.1.0"

NCELL = 32
CELL = 8
STRIDE = 7

_A: np.ndarray | None = None
_B: np.ndarray | None = None


def _glyph_grid(glyph: str) -> np.ndarray:
    g = np.zeros((NCELL, NCELL), dtype=bool)
    y, x = np.indices((NCELL, NCELL))
    if glyph == "A":
        inside = (y >= 8) & (y <= 23) & (x >= 8) & (x <= 23)
        edge = (y == 8) | (y == 23) | (x == 8) | (x == 23)
        g[inside & edge] = True
    else:
        band1 = np.abs(y - x) <= 2
        band2 = np.abs(y + x - (NCELL - 1)) <= 2
        g[band1 | band2] = True
    return g


def _ref(glyph: str) -> np.ndarray:
    global _A, _B
    if glyph == "A":
        if _A is None:
            _A = _glyph_grid("A")
        return _A
    if _B is None:
        _B = _glyph_grid("B")
    return _B


def _rolls() -> np.ndarray:
    r = np.arange(NCELL)
    return (STRIDE * r) % NCELL


def _reconstruct(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("L"), dtype=np.uint8).reshape(NCELL, CELL, NCELL, CELL)
    avg = arr.mean(axis=(1, 3))
    dark = avg < 128
    if dark.sum() > (NCELL * NCELL) / 2:
        dark = ~dark
    rolls = _rolls()
    out = np.empty_like(dark)
    for r in range(NCELL):
        out[r] = np.roll(dark[r], int(rolls[r]))
    return out


def decision_from_image(image: Image.Image) -> str:
    rec = _reconstruct(image)
    diffA = int(np.count_nonzero(rec != _ref("A")))
    diffB = int(np.count_nonzero(rec != _ref("B")))
    return "A" if diffA <= diffB else "B"


def reveal_margin(image: Image.Image) -> float:
    """A/B reconstruction-gap margin recovered from pixels alone."""
    rec = _reconstruct(image)
    diffA = float(np.count_nonzero(rec != _ref("A")))
    diffB = float(np.count_nonzero(rec != _ref("B")))
    return abs(diffB - diffA)
