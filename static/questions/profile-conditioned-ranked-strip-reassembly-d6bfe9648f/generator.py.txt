"""Latent sampler and raster renderer for ranked strip reassembly."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 560
GRID_TOP = 222
CELL_H = 24
FRAME_W = 42
FRAME_H = 12 * CELL_H

# Every 6x6 glyph has twelve dark cells and is expanded to 12x12 (48 dark cells).
_COARSE = {
    "flag": ("011110", "011000", "011100", "010000", "010000", "010000"),
    "hook": ("010000", "010000", "010000", "010001", "010001", "011111"),
    "stair": ("000011", "000110", "001100", "011000", "110000", "110000"),
    "fork": ("010010", "010010", "010010", "011110", "001000", "001000"),
}
SYMBOLS = tuple(_COARSE)
PALETTES = (
    {"frame": (0, 157, 188), "tab": (216, 35, 147), "ink": (26, 29, 34)},
    {"frame": (16, 132, 181), "tab": (185, 43, 168), "ink": (35, 31, 37)},
)


def _expand(rows: Iterable[str]) -> list[list[int]]:
    coarse = [[int(value) for value in row] for row in rows]
    return [[coarse[r // 2][c // 2] for c in range(12)] for r in range(12)]


TEMPLATES = {name: _expand(rows) for name, rows in _COARSE.items()}


def _effective_grid(scene: dict) -> list[list[int]]:
    grid = [list(map(int, row)) for row in scene["grid"]]
    for row, col in scene["cell_noise"]:
        grid[int(row)][int(col)] ^= 1
    return grid


def _classify(grid: list[list[int]]) -> tuple[str, float]:
    scores = []
    for name, template in TEMPLATES.items():
        distance = sum(
            int(grid[row][col]) != template[row][col]
            for row in range(12)
            for col in range(12)
        )
        scores.append((distance, name))
    scores.sort()
    return scores[0][1], float(scores[1][0] - scores[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[rng.randrange(len(SYMBOLS))]
    grid = [row[:] for row in TEMPLATES[symbol]]

    # Toggle equal numbers of on and off cells, preserving total ink and identity.
    pair_count = rng.randrange(4)  # 0, 2, 4, or 6 toggled cells
    on = [(r, c) for r in range(12) for c in range(12) if grid[r][c]]
    off = [(r, c) for r in range(12) for c in range(12) if not grid[r][c]]
    noise = rng.sample(on, pair_count) + rng.sample(off, pair_count)
    rng.shuffle(noise)

    permutation = list(range(12))
    rng.shuffle(permutation)
    strip_x = [42 + 70 * pos + rng.randint(-3, 3) for pos in range(12)]
    tab_y = [34 + 12 * rank + rng.randint(-1, 1) for rank in range(12)]
    return {
        "grid": grid,
        "permutation": permutation,
        "tab_y": tab_y,
        "strip_x": strip_x,
        "palette_variant": rng.randrange(len(PALETTES)),
        "cell_noise": [list(item) for item in noise],
    }


def render(scene: dict) -> Image.Image:
    grid = _effective_grid(scene)
    permutation = [int(value) for value in scene["permutation"]]
    tab_y = [float(value) for value in scene["tab_y"]]
    strip_x = [float(value) for value in scene["strip_x"]]
    palette = PALETTES[int(scene["palette_variant"])]

    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    draw.line((24, GRID_TOP - 12, WIDTH - 24, GRID_TOP - 12), fill=(210, 208, 202), width=2)
    for position, column in enumerate(permutation):
        x0 = int(round(strip_x[position]))
        x1 = x0 + FRAME_W
        ty = int(round(tab_y[column]))

        # A neutral tether makes the tab visibly belong to its strip.
        center_x = (x0 + x1) // 2
        draw.line((center_x, ty + 9, center_x, GRID_TOP - 13), fill=(188, 186, 181), width=2)
        draw.rounded_rectangle((center_x - 9, ty, center_x + 9, ty + 9), radius=2,
                               fill=palette["tab"])
        draw.rectangle((x0, GRID_TOP, x1, GRID_TOP + FRAME_H), outline=palette["frame"], width=3)
        for row in range(1, 12):
            y = GRID_TOP + row * CELL_H
            draw.line((x0 + 2, y, x1 - 2, y), fill=(218, 216, 210), width=1)
        for row in range(12):
            if grid[row][column]:
                y0 = GRID_TOP + row * CELL_H + 4
                y1 = GRID_TOP + (row + 1) * CELL_H - 4
                draw.rectangle((x0 + 6, y0, x1 - 6, y1), fill=palette["ink"])
    return image


def analytic_gold(scene: dict) -> str:
    return _classify(_effective_grid(scene))[0]


def margin(scene: dict) -> float:
    return _classify(_effective_grid(scene))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Raster coordinates round to integral pixels; +0.1 remains in the same bin.
    twin = {key: (value[:] if isinstance(value, list) else value) for key, value in scene.items()}
    twin["strip_x"] = [float(value) + 0.1 for value in scene["strip_x"]]
    return [("subpixel_strip_shift_same_raster_bin", twin)]
