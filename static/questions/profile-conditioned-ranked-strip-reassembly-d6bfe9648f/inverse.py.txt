"""Independent pixel-only inverse for ranked strip reassembly.

This module deliberately duplicates the public glyph vocabulary and discovers all
geometry from color components in the final raster. It imports no world module.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COARSE = {
    "flag": ("011110", "011000", "011100", "010000", "010000", "010000"),
    "hook": ("010000", "010000", "010000", "010001", "010001", "011111"),
    "stair": ("000011", "000110", "001100", "011000", "110000", "110000"),
    "fork": ("010010", "010010", "010010", "011110", "001000", "001000"),
}


def _templates() -> dict[str, np.ndarray]:
    result = {}
    for name, rows in _COARSE.items():
        coarse = np.array([[int(value) for value in row] for row in rows], dtype=np.uint8)
        result[name] = np.repeat(np.repeat(coarse, 2, axis=0), 2, axis=1)
    return result


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for start_y, start_x in zip(*np.nonzero(mask)):
        if seen[start_y, start_x]:
            continue
        queue = deque([(int(start_y), int(start_x))])
        seen[start_y, start_x] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs), max(ys), len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return output


def _measure(image: Image.Image) -> tuple[np.ndarray, list[tuple[int, str]]]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    cyan = (red < 80) & (green > 95) & (blue > 140) & ((blue - red) > 80)
    magenta = (red > 145) & (green < 95) & (blue > 105)
    dark = (red < 90) & (green < 90) & (blue < 90)

    frames = [c for c in _components(cyan) if (c[2] - c[0]) > 30 and (c[3] - c[1]) > 240 and c[4] > 250]
    tabs = [c for c in _components(magenta) if c[4] > 80 and (c[2] - c[0]) > 10]
    if len(frames) != 12 or len(tabs) != 12:
        raise ValueError(f"expected 12 frames and tabs, found {len(frames)} and {len(tabs)}")
    frames.sort(key=lambda item: item[5])

    ranked_columns = []
    diagnostics = []
    for frame in frames:
        x0, y0, x1, y1, _, center_x, _ = frame
        tab = min(tabs, key=lambda item: abs(item[5] - center_x))
        if abs(tab[5] - center_x) > 8.0:
            raise ValueError("tab-to-strip association is ambiguous")
        column = []
        frame_height = (y1 - y0)
        for row in range(12):
            top = int(round(y0 + row * frame_height / 12.0)) + 4
            bottom = int(round(y0 + (row + 1) * frame_height / 12.0)) - 3
            left, right = x0 + 6, x1 - 5
            occupancy = float(dark[top:bottom, left:right].mean())
            if 0.15 < occupancy < 0.55:
                raise ValueError("cell occupancy is ambiguous")
            column.append(1 if occupancy >= 0.55 else 0)
        ranked_columns.append((tab[6], column))

    ranked_columns.sort(key=lambda item: item[0])
    gaps = np.diff([item[0] for item in ranked_columns])
    if len(gaps) and float(gaps.min()) < 5.0:
        raise ValueError("tab ranks are ambiguous")
    grid = np.array([item[1] for item in ranked_columns], dtype=np.uint8).T
    templates = _templates()
    distances = sorted((int(np.count_nonzero(grid != template)), name) for name, template in templates.items())
    diagnostics = [(distance, name) for distance, name in distances]
    return grid, diagnostics


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    _, scores = _measure(image)
    margin = float(scores[1][0] - scores[0][0])
    if margin < 8.0:
        raise ValueError(f"ambiguous symbol: runner-up gap {margin}")
    return scores[0][1], margin


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
