"""Independent image-only inverse for the keyed corridor cell grammar."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 900
HEIGHT = 680
ROWS = 13
COLS = 15
CELL = 38
ORIGIN_X = 55
ORIGIN_Y = 155

OPEN_RGB = (248, 250, 252)
WALL_RGB = (34, 48, 65)
START_RGB = (20, 132, 74)
LETTERS = tuple("ABCDEFG")
KEY_RGB = {
    "A": (45, 88, 230),
    "B": (188, 40, 196),
    "C": (0, 157, 132),
    "D": (218, 82, 12),
    "E": (119, 55, 221),
    "F": (185, 30, 91),
    "G": (0, 133, 170),
}
GATE_RGB = {
    "A": (148, 188, 252),
    "B": (228, 145, 237),
    "C": (112, 220, 190),
    "D": (250, 179, 109),
    "E": (190, 169, 247),
    "F": (246, 143, 182),
    "G": (100, 215, 236),
}
EXIT_RGB = {
    "red": (216, 48, 58),
    "blue": (18, 100, 180),
    "orange": (238, 115, 18),
    "violet": (105, 64, 177),
}
EXIT_COLORS = tuple(EXIT_RGB)


class OracleError(ValueError):
    pass


def _near(patch: np.ndarray, rgb: tuple[int, int, int], radius: float = 18.0) -> int:
    delta = patch.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return int((np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius).sum())


def _recover(image: Image.Image) -> tuple[
    set[tuple[int, int]],
    tuple[int, int],
    dict[str, tuple[int, int]],
    dict[str, tuple[tuple[int, int], ...]],
    dict[str, tuple[int, int]],
]:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    special_palette = {
        "start": START_RGB,
        **{f"key:{letter}": rgb for letter, rgb in KEY_RGB.items()},
        **{f"gate:{letter}": rgb for letter, rgb in GATE_RGB.items()},
        **{f"exit:{color}": rgb for color, rgb in EXIT_RGB.items()},
    }
    found: dict[str, list[tuple[int, int]]] = {name: [] for name in special_palette}
    open_cells: set[tuple[int, int]] = set()

    for row in range(ROWS):
        for col in range(COLS):
            cx = ORIGIN_X + col * CELL + CELL // 2
            cy = ORIGIN_Y + row * CELL + CELL // 2
            patch = arr[cy - 12 : cy + 13, cx - 12 : cx + 13]
            scores = {name: _near(patch, rgb) for name, rgb in special_palette.items()}
            best = max(scores, key=scores.get)
            ranked = sorted(scores.values(), reverse=True)
            if scores[best] >= 150 and scores[best] >= 2 * max(1, ranked[1]):
                found[best].append((row, col))
                if not best.startswith("gate:"):
                    open_cells.add((row, col))
                continue
            open_score = _near(patch, OPEN_RGB)
            wall_score = _near(patch, WALL_RGB)
            if open_score >= 520 and open_score >= 5 * max(1, wall_score):
                open_cells.add((row, col))
            elif wall_score >= 520 and wall_score >= 5 * max(1, open_score):
                pass
            else:
                raise OracleError(f"cell {(row, col)} has no stable interior class")

    if len(found["start"]) != 1:
        raise OracleError(f"expected one START marker, found {len(found['start'])}")
    keys: dict[str, tuple[int, int]] = {}
    gates: dict[str, tuple[tuple[int, int], ...]] = {}
    for letter in LETTERS:
        key_values = found[f"key:{letter}"]
        gate_values = found[f"gate:{letter}"]
        if len(key_values) != 1:
            raise OracleError(f"expected one {letter} key, found {len(key_values)}")
        if len(gate_values) not in (1, 2):
            raise OracleError(f"expected one or two {letter} gates, found {len(gate_values)}")
        keys[letter] = key_values[0]
        gates[letter] = tuple(gate_values)
    if sorted(len(cells) for cells in gates.values()) != [1, 1, 1, 1, 1, 1, 2]:
        raise OracleError("expected exactly one duplicated gate letter")
    exits: dict[str, tuple[int, int]] = {}
    for color in EXIT_COLORS:
        values = found[f"exit:{color}"]
        if len(values) != 1:
            raise OracleError(f"expected one {color} EXIT, found {len(values)}")
        exits[color] = values[0]
    return open_cells, found["start"][0], keys, gates, exits


def _flood(
    traversable: set[tuple[int, int]], start: tuple[int, int]
) -> set[tuple[int, int]]:
    if start not in traversable:
        return set()
    seen = {start}
    queue = deque([start])
    while queue:
        row, col = queue.popleft()
        for neighbor in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            if neighbor in traversable and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return seen


def _closure(
    open_cells: set[tuple[int, int]],
    start: tuple[int, int],
    keys: dict[str, tuple[int, int]],
    gates: dict[str, tuple[tuple[int, int], ...]],
) -> tuple[set[tuple[int, int]], set[str]]:
    held: set[str] = set()
    for _ in range(len(LETTERS) + 1):
        unlocked = {cell for letter in held for cell in gates[letter]}
        reachable = _flood(open_cells | unlocked, start)
        found = {letter for letter, cell in keys.items() if cell in reachable}
        if found == held:
            return reachable, held
        held = found
    raise OracleError("key closure failed to stabilize")


def decision_from_image(image: Image.Image) -> str:
    open_cells, start, keys, gates, exits = _recover(image)
    reachable, held = _closure(open_cells, start, keys, gates)
    if len(held) != len(LETTERS) - 2:
        raise OracleError(f"expected five reached keys, found {sorted(held)}")
    answers = [color for color, cell in exits.items() if cell in reachable]
    if len(answers) != 1:
        raise OracleError(f"expected one reachable EXIT, found {answers}")
    return answers[0]
