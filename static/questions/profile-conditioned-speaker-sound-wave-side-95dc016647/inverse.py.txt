"""Independent PNG-only inverse arm for the public color-and-side contract."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    body = (blue > 115) & (blue > red + 55) & (blue > green + 25)
    waves = (red > 190) & (green > 65) & (green < 190) & (blue < 100) & (red > green + 55)
    if int(body.sum()) < 3500:
        raise ValueError("blue speaker body is missing or too small")
    if int(waves.sum()) < 500:
        raise ValueError("orange sound-wave evidence is missing or too small")
    return body, waves


def measurement_from_image(image: Image.Image) -> dict[str, float | str]:
    body, waves = _masks(image)
    _, body_x = np.nonzero(body)
    _, wave_x = np.nonzero(waves)
    body_center = float(np.median(body_x))
    wave_center = float(np.median(wave_x))
    delta = wave_center - body_center
    if abs(delta) < 45.0:
        raise ValueError("sound-wave side has insufficient pixel margin")
    return {
        "decision": "right" if delta > 0 else "left",
        "runner_up_margin": abs(delta),
        "body_center_x": body_center,
        "wave_center_x": wave_center,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measurement_from_image(image)["decision"])
