"""Deterministic matched-pair speaker icon renderer."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BODY_PALETTE = ("#235cbe", "#2d70d2", "#1e55a5")
WAVE_PALETTE = ("#f08219", "#e66914", "#f59623")


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one scene; consecutive even/odd seeds are exact mirror treatments."""
    rng = random.Random(int(seed) // 2)
    count = rng.choice((2, 3))
    start = rng.uniform(47.0, 58.0)
    step = rng.uniform(28.0, 35.0)
    return {
        "wave_side": "left" if int(seed) % 2 == 0 else "right",
        "center_x": round(rng.uniform(238.0, 274.0), 3),
        "center_y": round(rng.uniform(226.0, 286.0), 3),
        "scale": round(rng.uniform(0.88, 1.10), 4),
        "gap": round(rng.uniform(*((8.0, 11.0), (12.0, 16.0), (18.0, 23.0), (25.0, 30.0))[(int(seed) // 2) % 4]), 3),
        "arc_radii": [round(start + i * step, 3) for i in range(count)],
        "body_color": rng.choice(BODY_PALETTE),
        "wave_color": rng.choice(WAVE_PALETTE),
        "decoration_offsets": [[-12, -17], [12, 17]],
    }


def _mirror_x(x: float, cx: float, side: str) -> float:
    return x if side == "right" else 2.0 * cx - x


def _geometry(scene: dict[str, Any], ss: int) -> tuple[list[tuple[float, float]], list[float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    side = str(scene["wave_side"])
    # Defined in the right-facing frame, then reflected for the counterfactual.
    points = [
        (cx - 68 * scale, cy - 39 * scale),
        (cx - 34 * scale, cy - 39 * scale),
        (cx + 27 * scale, cy - 78 * scale),
        (cx + 27 * scale, cy + 78 * scale),
        (cx - 34 * scale, cy + 39 * scale),
        (cx - 68 * scale, cy + 39 * scale),
    ]
    points = [(_mirror_x(x, cx, side) * ss, y * ss) for x, y in points]
    return points, [float(value) * scale for value in scene["arc_radii"]]


def render(scene: dict[str, Any]) -> Image.Image:
    ss = 4
    canvas = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), "white")
    draw = ImageDraw.Draw(canvas)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    side = str(scene["wave_side"])
    body_color = str(scene["body_color"])
    wave_color = str(scene["wave_color"])
    body, radii = _geometry(scene, ss)
    draw.polygon(body, fill=body_color)

    # Neutral screws are balanced decorations inside the cabinet. Their list is a set.
    for dx, dy in scene["decoration_offsets"]:
        x = _mirror_x(cx - 52 * scale + float(dx) * 0.18, cx, side)
        y = cy + float(dy) * scale
        r = 3.2 * scale
        draw.ellipse(((x-r)*ss, (y-r)*ss, (x+r)*ss, (y+r)*ss), fill="#6f7782")

    face_x = _mirror_x(cx + 27 * scale, cx, side)
    direction = 1.0 if side == "right" else -1.0
    virtual_x = face_x + direction * float(scene["gap"])
    stroke = max(7, round(7.5 * scale * ss))
    for radius in radii:
        box = ((virtual_x-radius)*ss, (cy-radius)*ss, (virtual_x+radius)*ss, (cy+radius)*ss)
        angles = (-55, 55) if side == "right" else (125, 235)
        draw.arc(box, start=angles[0], end=angles[1], fill=wave_color, width=stroke)
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["wave_side"])


def margin(scene: dict[str, Any]) -> float:
    return float(scene["gap"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["arc_radii"] = list(reversed(scene["arc_radii"]))
    twin["decoration_offsets"] = list(reversed(scene["decoration_offsets"]))
    return [("reverse_unordered_drawing_sets", twin)]
