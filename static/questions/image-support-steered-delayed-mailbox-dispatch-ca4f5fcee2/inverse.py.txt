"""Independent pixel-only inverse for Delayed Mailbox Dispatch."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = np.array((35, 42, 52), dtype=np.int16)
GOLD = np.array((218, 164, 52), dtype=np.int16)
ORDER = np.array((67, 171, 102), dtype=np.int16)
OP_COLORS = {
    "flip": np.array((45, 174, 196), dtype=np.int16),
    "send": np.array((238, 137, 63), dtype=np.int16),
    "deliver": np.array((142, 91, 184), dtype=np.int16),
}


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            output.append({
                "area": len(points),
                "bbox": (min(xs), min(ys), max(xs) + 1, max(ys) + 1),
                "centroid": (sum(xs) / len(xs), sum(ys) / len(ys)),
            })
    return output


def _parse_start(array: np.ndarray) -> tuple[int, int, int]:
    dark = _mask(array, INK)
    lamps = []
    for part in _components(dark, minimum=110):
        bx0, by0, bx1, by1 = part["bbox"]
        width, height = bx1 - bx0, by1 - by0
        if 27 <= width <= 33 and 27 <= height <= 33:
            lamps.append(part)
    lamps.sort(key=lambda item: item["centroid"][0])
    if len(lamps) != 3:
        raise ValueError("abstain: three initial lamps were not recovered")
    if max(lamp["centroid"][1] for lamp in lamps) - min(lamp["centroid"][1] for lamp in lamps) > 3:
        raise ValueError("abstain: initial lamps do not share a visible row")
    gaps = [lamps[i + 1]["centroid"][0] - lamps[i]["centroid"][0] for i in range(2)]
    if min(gaps) < 45 or max(gaps) > 80:
        raise ValueError("abstain: initial lamp order is ambiguous")
    return tuple(1 if lamp["area"] > 430 else 0 for lamp in lamps)


def _parse_operations(array: np.ndarray) -> list[str]:
    order_mask = _mask(array, ORDER)
    order_dots = [part for part in _components(order_mask, minimum=45)
                  if 8 <= part["bbox"][2] - part["bbox"][0] <= 13
                  and 8 <= part["bbox"][3] - part["bbox"][1] <= 13]
    icons = []
    for operation, color in OP_COLORS.items():
        for part in _components(_mask(array, color), minimum=1800):
            x0, y0, x1, y1 = part["bbox"]
            if not (72 <= x1 - x0 <= 78 and 49 <= y1 - y0 <= 55 and part["area"] >= 3200):
                continue
            cx, _ = part["centroid"]
            dots = [dot for dot in order_dots
                    if x0 - 10 <= dot["centroid"][0] <= x1 + 10
                    and y0 - 32 <= dot["centroid"][1] <= y0 - 10]
            if not dots:
                raise ValueError("abstain: operation number is missing")
            dot_xs = sorted(dot["centroid"][0] for dot in dots)
            if abs(sum(dot_xs) / len(dot_xs) - cx) > 8:
                raise ValueError("abstain: operation number is displaced")
            icons.append((len(dots), operation))
    if len(icons) != 4 or sorted(number for number, _ in icons) != [1, 2, 3, 4]:
        raise ValueError("abstain: four uniquely numbered operations were not recovered")
    return [operation for _, operation in sorted(icons)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    source, mail, display = _parse_start(array)
    for operation in _parse_operations(array):
        if operation == "flip":
            source = 1 - source
        elif operation == "send":
            mail = source
        elif operation == "deliver":
            display = mail
    return "yes" if display else "no"
