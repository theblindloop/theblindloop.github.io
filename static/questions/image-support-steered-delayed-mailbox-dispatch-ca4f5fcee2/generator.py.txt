"""Deterministic scene sampler and raster renderer for Delayed Mailbox Dispatch."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 640
BACKGROUND = (248, 247, 243)
INK = (35, 42, 52)
GOLD = (218, 164, 52)
ORDER = (67, 171, 102)
OP_COLORS = {
    "flip": (45, 174, 196),
    "send": (238, 137, 63),
    "deliver": (142, 91, 184),
}
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

TEMPLATES = (
    ("send", "flip", "deliver", "flip"),
    ("flip", "send", "flip", "deliver"),
    ("send", "flip", "flip", "deliver"),
    ("flip", "send", "deliver", "flip"),
    ("deliver", "flip", "send", "deliver"),
    ("flip", "deliver", "send", "deliver"),
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    operations = list(TEMPLATES[rng.randrange(len(TEMPLATES))])
    return {
        "initial_source": rng.randrange(2),
        "initial_mail": rng.randrange(2),
        "initial_display": rng.randrange(2),
        "operations": operations,
        "layout_variant": rng.randrange(3),
    }


def _replay(scene: dict) -> tuple[int, int, int]:
    source = int(scene["initial_source"])
    mail = int(scene["initial_mail"])
    display = int(scene["initial_display"])
    for operation in scene["operations"]:
        if operation == "flip":
            source = 1 - source
        elif operation == "send":
            mail = source
        elif operation == "deliver":
            display = mail
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return source, mail, display


def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene)[2] else "no"


def margin(scene: dict) -> float:
    # One-sided discrete margin: every legal lamp and event is separated from
    # the nearest competing code by at least one complete 18 px visual stroke.
    _replay(scene)
    return 18.0


def is_quarantined(scene: dict) -> bool:
    try:
        return (
            len(scene["operations"]) != 4
            or any(op not in OP_COLORS for op in scene["operations"])
            or int(scene["initial_source"]) not in (0, 1)
            or int(scene["initial_mail"]) not in (0, 1)
            or int(scene["initial_display"]) not in (0, 1)
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], text: str,
                   font: ImageFont.FreeTypeFont, fill: tuple[int, int, int]) -> None:
    left, top, right, bottom = box
    bounds = draw.textbbox((0, 0), text, font=font)
    tw, th = bounds[2] - bounds[0], bounds[3] - bounds[1]
    draw.text(((left + right - tw) // 2, (top + bottom - th) // 2 - bounds[1]),
              text, font=font, fill=fill)


def _lamp(draw: ImageDraw.ImageDraw, center: tuple[int, int], on: int) -> None:
    x, y = center
    box = (x - 15, y - 15, x + 15, y + 15)
    draw.ellipse(box, fill=INK if on else BACKGROUND, outline=INK, width=5)


def _start_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], scene: dict) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=18, fill=(255, 253, 246), outline=GOLD, width=7)
    _centered_text(draw, (x0, y0 + 8, x1, y0 + 36), "START", _font(18), INK)
    xs = (x0 + 48, x0 + 110, x0 + 172)
    values = (scene["initial_source"], scene["initial_mail"], scene["initial_display"])
    labels = ("SOURCE", "MAIL", "DISPLAY")
    for x, value, label in zip(xs, values, labels):
        _lamp(draw, (x, y0 + 78), int(value))
        _centered_text(draw, (x - 38, y0 + 103, x + 38, y0 + 129), label, _font(10), INK)
    _centered_text(draw, (x0 + 10, y1 - 26, x1 - 10, y1 - 5), "filled = ON", _font(11), INK)


def _operation_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int],
                    number: int, operation: str) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=15, fill=(255, 255, 255), outline=INK, width=4)
    spacing = 17
    start = (x0 + x1 - spacing * (number - 1)) // 2
    for index in range(number):
        cx = start + index * spacing
        draw.ellipse((cx - 5, y0 + 10, cx + 5, y0 + 20), fill=ORDER)
    icon = (x0 + 18, y0 + 35, x1 - 18, y0 + 86)
    draw.rounded_rectangle(icon, radius=8, fill=OP_COLORS[operation])
    _centered_text(draw, icon, operation.upper(), _font(15), (255, 255, 255))
    if operation == "flip":
        caption = "SOURCE toggles"
    elif operation == "send":
        caption = "SOURCE → MAIL"
    else:
        caption = "MAIL → DISPLAY"
    _centered_text(draw, (x0 + 3, y0 + 90, x1 - 3, y1 - 5), caption, _font(8), INK)


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render malformed scene")
    variant = int(scene["layout_variant"])
    shifts = ((0, 0), (-10, 8), (9, -7))
    dx, dy = shifts[variant]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (160, 28, 800, 72), "DELAYED MAILBOX DISPATCH", _font(27), INK)
    _centered_text(draw, (180, 78, 780, 106), "Read the numbered cards in order", _font(15), (88, 94, 104))
    _start_card(draw, (370 + dx, 70 + dy, 590 + dx, 230 + dy), scene)
    positions = (
        (250 + dx, 360 + dy, 360 + dx, 480 + dy),
        (375 + dx, 360 + dy, 485 + dx, 480 + dy),
        (500 + dx, 360 + dy, 610 + dx, 480 + dy),
        (625 + dx, 360 + dy, 735 + dx, 480 + dy),
    )
    for number, (operation, box) in enumerate(zip(scene["operations"], positions), start=1):
        _operation_card(draw, box, number, operation)
    _centered_text(draw, (190, 525, 770, 568), "FLIP changes SOURCE   •   SEND stores SOURCE   •   DELIVER shows MAIL",
                   _font(14), (88, 94, 104))
    return image
