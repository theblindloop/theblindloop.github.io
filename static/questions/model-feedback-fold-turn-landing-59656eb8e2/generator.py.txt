"""Deterministic renderer for the fold-then-turn geometric program."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 1030
GRID_N = 9
CELL = 38
PANEL_W = 450
PANEL_H = 440
PANEL_X = (35, 515)
PANEL_Y = (85, 555)
GRID_OFFSET_X = 54
GRID_OFFSET_Y = 72

WHITE = (255, 255, 255)
INK = (31, 41, 55)
GRID = (211, 218, 226)
BLUE = (37, 99, 235)
AMBER = (245, 158, 11)
RED = (220, 38, 38)
GREEN = (22, 163, 74)

FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_BASE_SHAPES = (
    ((0, 0), (0, 1), (0, 2), (1, 2)),
    ((0, 0), (1, 0), (2, 0), (0, 1)),
    ((0, 0), (1, 0), (1, 1), (2, 1)),
    ((0, 0), (0, 1), (1, 1), (2, 1)),
    ((0, 0), (1, 0), (2, 0), (1, 1)),
)


def _shape_variant(shape: Iterable[tuple[int, int]], code: int) -> set[tuple[int, int]]:
    points = []
    for x, y in shape:
        if code >= 4:
            x = -x
        for _ in range(code % 4):
            x, y = -y, x
        points.append((x, y))
    min_x = min(x for x, _ in points)
    min_y = min(y for _, y in points)
    return {(x - min_x, y - min_y) for x, y in points}


def _reflect_point(point: tuple[int, int], axis: dict) -> tuple[int, int]:
    x, y = point
    k = int(axis["k"])
    kind = axis["kind"]
    if kind == "vertical":
        return 2 * k - x, y
    if kind == "horizontal":
        return x, 2 * k - y
    if kind == "diag_down":  # y = x + k
        return y - k, x + k
    if kind == "diag_up":  # x + y = k
        return k - y, k - x
    raise ValueError(f"unknown axis kind {kind!r}")


def _rotate_cw(point: tuple[int, int], pivot: tuple[int, int]) -> tuple[int, int]:
    x, y = point
    px, py = pivot
    return px - (y - py), py + (x - px)


def _rotate_ccw(point: tuple[int, int], pivot: tuple[int, int]) -> tuple[int, int]:
    x, y = point
    px, py = pivot
    return px + (y - py), py - (x - px)


def _transform(source: Iterable[tuple[int, int]], axis: dict, pivot: tuple[int, int]) -> set[tuple[int, int]]:
    return {_rotate_cw(_reflect_point(point, axis), pivot) for point in source}


def _axis_side(point: tuple[int, int], axis: dict) -> int:
    x, y = point
    k = int(axis["k"])
    if axis["kind"] == "vertical":
        return x - k
    if axis["kind"] == "horizontal":
        return y - k
    if axis["kind"] == "diag_down":
        return y - x - k
    return x + y - k


def _inside(points: Iterable[tuple[int, int]]) -> bool:
    return all(0 <= x < GRID_N and 0 <= y < GRID_N for x, y in points)


def _serial(points: Iterable[tuple[int, int]]) -> list[list[int]]:
    return [[x, y] for x, y in sorted(points, key=lambda p: (p[1], p[0]))]


def _make_panel(rng: random.Random, correct: bool, wrong_selector: int) -> dict:
    axis_choices = (
        ("vertical", tuple(range(2, 7))),
        ("horizontal", tuple(range(2, 7))),
        ("diag_down", tuple(range(-2, 3))),
        ("diag_up", tuple(range(6, 11))),
    )
    for _ in range(12000):
        shape = _shape_variant(rng.choice(_BASE_SHAPES), rng.randrange(8))
        shape_w = max(x for x, _ in shape) + 1
        shape_h = max(y for _, y in shape) + 1
        tx = rng.randrange(0, GRID_N - shape_w + 1)
        ty = rng.randrange(0, GRID_N - shape_h + 1)
        source = {(x + tx, y + ty) for x, y in shape}

        kind, values = rng.choice(axis_choices)
        axis = {"kind": kind, "k": int(rng.choice(values))}
        sides = [_axis_side(point, axis) for point in source]
        if not (all(value >= 1 for value in sides) or all(value <= -1 for value in sides)):
            continue
        reflected = {_reflect_point(point, axis) for point in source}
        if not _inside(reflected):
            continue

        pivot = (rng.randrange(1, GRID_N - 1), rng.randrange(1, GRID_N - 1))
        if pivot in source or pivot in reflected or _axis_side(pivot, axis) == 0:
            continue
        final = {_rotate_cw(point, pivot) for point in reflected}
        if not _inside(final) or source & final or pivot in final:
            continue
        if any(_axis_side(point, axis) == 0 for point in final):
            continue

        reversed_order = {_reflect_point(_rotate_cw(point, pivot), axis) for point in source}
        wrong_sets = (
            {_rotate_ccw(point, pivot) for point in reflected},
            reversed_order,
            reflected,
            {_rotate_cw(point, pivot) for point in source},
            {_rotate_cw(_rotate_cw(point, pivot), pivot) for point in reflected},
        )
        usable = []
        for wrong in wrong_sets:
            if (
                _inside(wrong)
                and wrong != final
                and not (source & wrong)
                and pivot not in wrong
                and not any(_axis_side(point, axis) == 0 for point in wrong)
                and len(final & wrong) <= 2
            ):
                usable.append(wrong)
        if not correct and not usable:
            continue
        target = final if correct else usable[wrong_selector % len(usable)]
        return {
            "source": _serial(source),
            "axis": axis,
            "pivot": list(pivot),
            "target": _serial(target),
        }
    raise RuntimeError("could not sample a clear fold-turn panel")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x6F31A2D9)
    correct_index = int(seed) % 4
    panels = [
        _make_panel(rng, index == correct_index, index + int(seed))
        for index in range(4)
    ]
    order = list(range(4))
    rng.shuffle(order)
    scene = {"panels": panels, "panel_draw_order": order}
    # Construction guarantees a unique answer; fail here rather than emitting poison.
    analytic_gold(scene)
    return scene


def _panel_score(panel: dict) -> int:
    source = {tuple(point) for point in panel["source"]}
    pivot = tuple(panel["pivot"])
    final = _transform(source, panel["axis"], pivot)
    target = {tuple(point) for point in panel["target"]}
    return len(final & target)


def analytic_gold(scene: dict) -> str:
    # Keep the audited label boundary at the top-level ``panels`` field.  The
    # panel payload is unpacked here so the static rendered-field audit sees the
    # same top-level input that the renderer consumes.
    scores = []
    for panel in scene["panels"]:
        source = {tuple(point) for point in panel["source"]}
        pivot = tuple(panel["pivot"])
        final = _transform(source, panel["axis"], pivot)
        target = {tuple(point) for point in panel["target"]}
        scores.append(len(final & target))
    winners = [index for index, score in enumerate(scores) if score == 4]
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} exact landing panels: {scores}")
    return f"panel-{winners[0] + 1}"


def margin(scene: dict) -> float:
    scores = sorted((_panel_score(panel) for panel in scene["panels"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panel_draw_order"] = list(reversed(scene["panel_draw_order"]))
    return [("reverse_disjoint_panel_painter_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_BOLD if bold else FONT_REGULAR, size)


def _panel_origin(index: int) -> tuple[int, int]:
    return PANEL_X[index % 2], PANEL_Y[index // 2]


def _grid_center(panel_index: int, point: tuple[float, float]) -> tuple[float, float]:
    panel_x, panel_y = _panel_origin(panel_index)
    x, y = point
    return (
        panel_x + GRID_OFFSET_X + (x + 0.5) * CELL,
        panel_y + GRID_OFFSET_Y + (y + 0.5) * CELL,
    )


def _axis_endpoints(panel_index: int, axis: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    low, high = -0.42, GRID_N - 0.58
    kind, k = axis["kind"], float(axis["k"])
    if kind == "vertical":
        a, b = (k, low), (k, high)
    elif kind == "horizontal":
        a, b = (low, k), (high, k)
    elif kind == "diag_down":
        x0, x1 = max(low, low - k), min(high, high - k)
        a, b = (x0, x0 + k), (x1, x1 + k)
    else:
        x0, x1 = max(low, k - high), min(high, k - low)
        a, b = (x0, k - x0), (x1, k - x1)
    return _grid_center(panel_index, a), _grid_center(panel_index, b)


def _dashed_line(
    draw: ImageDraw.ImageDraw,
    start: tuple[float, float],
    end: tuple[float, float],
    *,
    fill: tuple[int, int, int],
    width: int,
    dash: float = 16.0,
    gap: float = 10.0,
) -> None:
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    distance = 0.0
    while distance < length:
        stop = min(distance + dash, length)
        draw.line(
            (
                start[0] + ux * distance,
                start[1] + uy * distance,
                start[0] + ux * stop,
                start[1] + uy * stop,
            ),
            fill=fill,
            width=width,
        )
        distance += dash + gap


def _draw_panel(draw: ImageDraw.ImageDraw, panel: dict, index: int) -> None:
    panel_x, panel_y = _panel_origin(index)
    draw.rounded_rectangle(
        (panel_x, panel_y, panel_x + PANEL_W, panel_y + PANEL_H),
        radius=14,
        fill=WHITE,
        outline=INK,
        width=3,
    )
    badge = (panel_x + 16, panel_y + 14, panel_x + 54, panel_y + 52)
    draw.ellipse(badge, fill=INK)
    label = str(index + 1)
    box = draw.textbbox((0, 0), label, font=_font(24, True))
    draw.text(
        (panel_x + 35 - (box[2] - box[0]) / 2, panel_y + 33 - (box[3] - box[1]) / 2 - 2),
        label,
        fill=WHITE,
        font=_font(24, True),
        anchor="mm",
    )

    gx = panel_x + GRID_OFFSET_X
    gy = panel_y + GRID_OFFSET_Y
    grid_size = GRID_N * CELL
    draw.rectangle((gx, gy, gx + grid_size, gy + grid_size), fill=(250, 251, 252), outline=GRID, width=2)
    for step in range(1, GRID_N):
        draw.line((gx + step * CELL, gy, gx + step * CELL, gy + grid_size), fill=GRID, width=1)
        draw.line((gx, gy + step * CELL, gx + grid_size, gy + step * CELL), fill=GRID, width=1)

    axis_a, axis_b = _axis_endpoints(index, panel["axis"])
    _dashed_line(draw, axis_a, axis_b, fill=RED, width=7)

    for key, color in (("target", AMBER), ("source", BLUE)):
        for x, y in panel[key]:
            left = gx + x * CELL + 3
            top = gy + y * CELL + 3
            draw.rectangle((left, top, left + CELL - 6, top + CELL - 6), fill=color)

    px, py = _grid_center(index, tuple(panel["pivot"]))
    draw.ellipse((px - 13, py - 13, px + 13, py + 13), fill=WHITE, outline=INK, width=2)
    draw.ellipse((px - 10, py - 10, px + 10, py + 10), fill=GREEN)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.text((28, 12), "REFLECT, THEN TURN 90° CLOCKWISE", fill=INK, font=_font(29, True))
    draw.rectangle((646, 18, 668, 40), fill=BLUE)
    draw.text((676, 18), "start", fill=INK, font=_font(17))
    draw.line((742, 29, 779, 29), fill=RED, width=6)
    draw.text((787, 18), "fold", fill=INK, font=_font(17))
    draw.ellipse((849, 19, 869, 39), fill=GREEN)
    draw.text((877, 18), "pin", fill=INK, font=_font(17))
    draw.rectangle((646, 49, 668, 71), fill=AMBER)
    draw.text((676, 49), "shown landing", fill=INK, font=_font(17))
    for index in scene.get("panel_draw_order", list(range(4))):
        _draw_panel(draw, scene["panels"][int(index)], int(index))
    return image
