"""Independent image-only inverse for trace direction alternation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
ROUTE = np.asarray([24.0, 73.0, 98.0], dtype=np.float32)
ROUTE_TOLERANCE = 34.0
MIN_ROUTE_PIXELS = 450
MIN_COMPONENT_FRACTION = 0.90
PIECE_COUNT = 7
MIN_SLOPE = 0.15


def _route_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError("unexpected canvas size")
    return np.linalg.norm(array - ROUTE.reshape(1, 1, 3), axis=2) <= ROUTE_TOLERANCE


def _largest_component(mask: np.ndarray) -> int:
    points = np.argwhere(mask)
    if not len(points):
        return 0
    start = (int(points[0, 0]), int(points[0, 1]))
    seen = {start}
    queue: deque[tuple[int, int]] = deque([start])
    while queue:
        y, x = queue.popleft()
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dy or dx):
                    continue
                neighbor = (y + dy, x + dx)
                ny, nx = neighbor
                if (
                    0 <= ny < mask.shape[0]
                    and 0 <= nx < mask.shape[1]
                    and mask[ny, nx]
                    and neighbor not in seen
                ):
                    seen.add(neighbor)
                    queue.append(neighbor)
    return len(seen)


def _centerline(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    counts = mask.sum(axis=0)
    columns = np.flatnonzero(counts > 0)
    if len(columns) < 160:
        raise ValueError("trace is absent or too short")
    if int(counts[columns].max()) < 3:
        raise ValueError("trace has no measurable stroke width")
    gaps = np.diff(columns)
    if len(gaps) and int(gaps.max()) > 3:
        raise ValueError("trace has a destructive horizontal gap")
    left = int(columns[0] + 5)
    right = int(columns[-1] - 5)
    if right - left < 180:
        raise ValueError("trace support is too small")
    xs: list[float] = []
    ys: list[float] = []
    for x in range(left, right + 1):
        row = np.flatnonzero(mask[:, x])
        if len(row) == 0:
            raise ValueError("trace centerline is incomplete")
        xs.append(float(x))
        ys.append(float(np.median(row)))
    return np.asarray(xs, dtype=np.float64), np.asarray(ys, dtype=np.float64)


def _piece_slopes(xs: np.ndarray, ys: np.ndarray) -> list[float]:
    boundaries = np.linspace(float(xs[0]), float(xs[-1]), PIECE_COUNT + 1)
    slopes: list[float] = []
    for index in range(PIECE_COUNT):
        # Leave a small margin at each bend so a rounded joint cannot dominate
        # the local linear fit.
        lo = boundaries[index] + 3.0
        hi = boundaries[index + 1] - 3.0
        select = (xs >= lo) & (xs <= hi)
        if int(select.sum()) < 12:
            raise ValueError("one of seven trace pieces is under-observed")
        slope, _intercept = np.polyfit(xs[select], ys[select], 1)
        slope = float(slope)
        if abs(slope) < MIN_SLOPE:
            raise ValueError("a piece direction is unresolved at raster scale")
        slopes.append(slope)
    return slopes


def decision_from_image(image: Image.Image) -> str:
    """Recover yes/no from the final PNG only."""

    mask = _route_mask(image)
    pixels = int(mask.sum())
    if pixels < MIN_ROUTE_PIXELS:
        raise ValueError("route ink is absent or too small")
    component = _largest_component(mask)
    if component < MIN_COMPONENT_FRACTION * pixels:
        raise ValueError("route is not one connected component")
    xs, ys = _centerline(mask)
    slopes = _piece_slopes(xs, ys)
    signs = [1 if slope > 0 else -1 for slope in slopes]
    alternating = all(left != right for left, right in zip(signs, signs[1:]))
    return "yes" if alternating else "no"
