"""Renderer and analytic arm for the trace-direction alternation world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
ROUTE_RGB = (24, 73, 98)
BACKGROUND_COLORS = ((248, 251, 255), (255, 252, 246), (249, 255, 250))
FRAME_COLORS = ((158, 176, 191), (175, 164, 151), (151, 180, 164))
STROKES = (9, 10)
MIN_CLEARANCE = 28.0

# Each row has seven successive vertical displacements with a large margin.
# The near versions keep one displacement visibly nonzero but below the
# quarantine band, supplying explicit raster-boundary evidence.
Y_TEMPLATES: dict[tuple[str, int, bool], tuple[int, ...]] = {
    ("alternating", 0, False): (280, 225, 275, 220, 270, 225, 280, 230),
    ("alternating", 0, True): (280, 225, 249, 220, 270, 225, 280, 230),
    ("non_alternating", 0, False): (280, 225, 275, 315, 250, 295, 235, 280),
    ("non_alternating", 0, True): (280, 225, 249, 315, 250, 295, 235, 280),
    ("alternating", 1, False): (235, 285, 230, 280, 235, 290, 240, 285),
    ("alternating", 1, True): (235, 285, 261, 280, 235, 290, 240, 285),
    ("non_alternating", 1, False): (235, 285, 230, 270, 315, 255, 300, 250),
    ("non_alternating", 1, True): (235, 285, 261, 270, 315, 255, 300, 250),
}


def _route_points(scene: dict[str, Any]) -> list[tuple[int, int]]:
    pattern = str(scene["pattern"])
    variant = int(scene["variant"])
    near_boundary = bool(scene["near_boundary"])
    y_values = Y_TEMPLATES[(pattern, variant, near_boundary)]
    x_scale = float(scene["x_scale"])
    y_scale = float(scene["y_scale"])
    x_shift = float(scene["x_shift"])
    y_shift = float(scene["y_shift"])
    flip_vertical = int(scene["flip_vertical"])
    points: list[tuple[int, int]] = []
    for index, base_y in enumerate(y_values):
        base_x = 261.0 + (index - 3.5) * 35.0
        x = 261.0 + (base_x - 261.0) * x_scale + x_shift
        centered_y = 260.0 + (float(base_y) - 260.0) * y_scale
        y = 260.0 + flip_vertical * (centered_y - 260.0) + y_shift
        points.append((round(x), round(y)))
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a deterministic, answer-balanced raster scene."""

    rng = random.Random(int(seed))
    # The generator's 1009-step schedule alternates this identity exactly,
    # while presentation variables remain independently randomized.
    pattern = "alternating" if int(seed) % 2 == 0 else "non_alternating"
    return {
        "pattern": pattern,
        "variant": rng.randrange(2),
        "near_boundary": rng.random() < 0.22,
        "x_scale": round(rng.uniform(0.96, 1.04), 6),
        "y_scale": round(rng.uniform(0.96, 1.04), 6),
        "x_shift": round(rng.uniform(-5.0, 5.0), 3),
        "y_shift": round(rng.uniform(-5.0, 5.0), 3),
        "flip_vertical": -1 if rng.randrange(2) else 1,
        "background": list(BACKGROUND_COLORS[rng.randrange(len(BACKGROUND_COLORS))]),
        "route_color": list(ROUTE_RGB),
        "stroke": STROKES[rng.randrange(len(STROKES))],
        "frame_color": list(FRAME_COLORS[rng.randrange(len(FRAME_COLORS))]),
        "frame_width": 2 + rng.randrange(2),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one final RGB PNG with no hidden payload or external asset."""

    background = tuple(int(value) for value in scene["background"])
    frame_color = tuple(int(value) for value in scene["frame_color"])
    route_color = tuple(int(value) for value in scene["route_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle((112, 128, 408, 388), outline=frame_color, width=int(scene["frame_width"]))
    points = _route_points(scene)
    draw.line(points, fill=route_color, width=int(scene["stroke"]), joint="curve")
    # Small same-color terminal caps make the single route visually deliberate
    # without adding a second answer-bearing mark.
    radius = max(4, int(scene["stroke"]) // 2)
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=route_color)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """The label is the visible route-pattern predicate."""

    return "yes" if scene["pattern"] == "alternating" else "no"


def margin(scene: dict[str, Any]) -> float:
    """Minimum absolute vertical displacement between adjacent vertices."""

    points = _route_points(scene)
    return float(min(abs(b[1] - a[1]) for a, b in zip(points, points[1:])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """This minimal parametrization declares no pixel-identical transform."""

    return []
