from __future__ import annotations

import random
from PIL import Image, ImageDraw

W, H = 640, 360
BG = (250, 249, 246)
INK = (25, 29, 36)
GRAY = (126, 134, 146)
PALE = (235, 238, 241)
MAGENTA = (220, 35, 135)
COLORS = {"red": (220, 55, 58), "green": (35, 165, 92), "purple": (119, 72, 190)}
TYPES = {"color": (38, 112, 210), "shape": (238, 139, 36)}
SLOTS = (140, 320, 500)
CARD_X = (80, 200, 320, 440, 560)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    shapes = ["circle", "square", "triangle"]
    rng.shuffle(colors)
    rng.shuffle(shapes)
    operations = []
    for _ in range(5):
        a, b = rng.sample(range(3), 2)
        operations.append({"kind": rng.choice(("color", "shape")), "a": a, "b": b})
    return {
        "initial_colors": colors,
        "initial_shapes": shapes,
        "operations": operations,
        "query_slot": rng.randrange(3),
    }


def _shape(draw: ImageDraw.ImageDraw, name: str, cx: int, cy: int, fill: tuple[int, int, int]) -> None:
    r = 28
    if name == "circle":
        draw.ellipse((cx-r, cy-r, cx+r, cy+r), fill=fill)
    elif name == "square":
        draw.rectangle((cx-r, cy-r, cx+r, cy+r), fill=fill)
    elif name == "triangle":
        draw.polygon(((cx, cy-r-2), (cx-r-3, cy+r), (cx+r+3, cy+r)), fill=fill)
    else:
        raise ValueError("unknown shape")


def _pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int) -> None:
    for j in range(count):
        y = cy + (j - (count - 1) / 2) * 12
        draw.ellipse((cx-4, y-4, cx+4, y+4), fill=INK)


def render(scene: dict) -> Image.Image:
    colors = scene["initial_colors"]
    shapes = scene["initial_shapes"]
    operations = scene["operations"]
    query = int(scene["query_slot"])
    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    draw.line((70, 160, 570, 160), fill=GRAY, width=2)
    for i, x in enumerate(SLOTS):
        draw.rounded_rectangle((x-54, 50, x+54, 151), radius=12, fill=PALE, outline=GRAY, width=2)
        _shape(draw, shapes[i], x, 101, COLORS[colors[i]])
        # Three neutral slot ticks make fixed slot order visible without text.
        for k in range(i + 1):
            tx = x + (k - i / 2) * 12
            draw.ellipse((tx-3, 137, tx+3, 143), fill=GRAY)
    draw.ellipse((SLOTS[query]-43, 58, SLOTS[query]+43, 144), outline=MAGENTA, width=6)

    for op, cx in zip(operations, CARD_X):
        draw.rounded_rectangle((cx-50, 193, cx+50, 326), radius=10, fill=PALE, outline=GRAY, width=2)
        draw.rectangle((cx-36, 211, cx+36, 227), fill=TYPES[op["kind"]])
        draw.line((cx-8, 267, cx+8, 267), fill=GRAY, width=3)
        draw.polygon(((cx+8, 267), (cx+1, 262), (cx+1, 272)), fill=GRAY)
        _pips(draw, cx-25, 267, int(op["a"]) + 1)
        _pips(draw, cx+25, 267, int(op["b"]) + 1)
    return image


def _replay(scene: dict) -> tuple[list[str], list[str]]:
    colors = list(scene["initial_colors"])
    shapes = list(scene["initial_shapes"])
    operations = scene["operations"]
    # Read query too so analytic label inputs exactly match all answer-relevant fields.
    int(scene["query_slot"])
    for op in operations:
        a, b = int(op["a"]), int(op["b"])
        target = colors if op["kind"] == "color" else shapes
        target[a], target[b] = target[b], target[a]
    return colors, shapes


def analytic_gold(scene: dict) -> str:
    colors, shapes = _replay(scene)
    q = int(scene["query_slot"])
    return f"{colors[q]} {shapes[q]}"


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
