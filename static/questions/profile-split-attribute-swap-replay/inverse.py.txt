from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 58),
    "green": (35, 165, 92),
    "purple": (119, 72, 190),
}
TYPE_RGB = {"color": (38, 112, 210), "shape": (238, 139, 36)}
INK = (25, 29, 36)
MAGENTA = (220, 35, 135)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            yy, xx = stack.pop(); pts.append((yy, xx))
            for dy, dx in ((1,0),(-1,0),(0,1),(0,-1)):
                ny, nx = yy+dy, xx+dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; stack.append((ny, nx))
        ys = [p[0] for p in pts]; xs = [p[1] for p in pts]
        out.append((min(xs), min(ys), max(xs), max(ys), len(pts), float(sum(xs))/len(xs), float(sum(ys))/len(ys)))
    return out


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.array(rgb, dtype=np.uint8), axis=2)


def _shape_name(component: tuple[int, int, int, int, int, float, float]) -> str:
    x0, y0, x1, y1, area, _, _ = component
    ratio = area / ((x1-x0+1) * (y1-y0+1))
    if ratio > 0.88:
        return "square"
    if ratio > 0.63:
        return "circle"
    if ratio > 0.38:
        return "triangle"
    raise ValueError("unreadable shape")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    badges = []
    for color, rgb in PALETTE.items():
        comps = [c for c in _components(_mask(arr, rgb)) if c[4] > 800]
        if len(comps) != 1:
            raise ValueError("missing or ambiguous badge color")
        c = comps[0]
        badges.append({"x": c[5], "color": color, "shape": _shape_name(c)})
    badges.sort(key=lambda b: b["x"])
    mag = np.nonzero(_mask(arr, MAGENTA))
    if len(mag[0]) < 100:
        raise ValueError("query ring missing")
    qx = float(np.mean(mag[1]))
    query = min(range(3), key=lambda i: abs(badges[i]["x"] - qx))
    if abs(badges[query]["x"] - qx) > 55:
        raise ValueError("query ring unbound")

    stripes = []
    for kind, rgb in TYPE_RGB.items():
        for c in _components(_mask(arr, rgb)):
            if c[4] > 700:
                stripes.append((c[5], c[6], kind))
    if len(stripes) != 5:
        raise ValueError("operation stripes missing")
    stripes.sort()
    dots = [c for c in _components(_mask(arr, INK)) if 35 <= c[4] <= 100]
    if len(dots) < 10:
        raise ValueError("operation dots missing")
    operations = []
    for cx, cy, kind in stripes:
        nearby = [c for c in dots if abs(c[5]-cx) < 46 and 238 < c[6] < 300]
        left = sum(c[5] < cx-8 for c in nearby)
        right = sum(c[5] > cx+8 for c in nearby)
        if left not in (1,2,3) or right not in (1,2,3) or left == right:
            raise ValueError("ambiguous dot-coded slot pair")
        operations.append((kind, left-1, right-1))
    colors = [b["color"] for b in badges]
    shapes = [b["shape"] for b in badges]
    for kind, a, b in operations:
        values = colors if kind == "color" else shapes
        values[a], values[b] = values[b], values[a]
    return f"{colors[query]} {shapes[query]}"
