"""Scene-free pixel arm for the mirror-fold defect world.

Receives only the rendered RGB image and the public question contract.  It
locates the coloured frame and fold-axis fiducials, reads the lean direction of
every printed bar, folds the lattice about the axis, and names the symbol drawn
by the cells whose partner leans the wrong way.  It abstains whenever the
fiducials, the lattice, the lean directions, or the symbol match are unsafe.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

ROWS = 7
COLS = 12
HALF = COLS // 2
SYM_SIZE = 5

ACCENT_SAT = 70
INK_SAT = 50
INK_VAL = 130

ANGLE_FLOOR_DEG = 3.5
MAX_BEST_DIST = 2
MIN_RUNNERUP_GAP = 2

# Independent transcription of the five symbol descriptions published in the
# question text; five cells per row, nine marked cells per symbol.
_SHAPES = {
    "plus": ("..#..", "..#..", "#####", "..#..", "..#.."),
    "ex": ("#...#", ".#.#.", "..#..", ".#.#.", "#...#"),
    "tee": ("#####", "..#..", "..#..", "..#..", "..#.."),
    "ell": ("#....", "#....", "#....", "#....", "#####"),
    "cee": (".###.", "#....", "#....", "#....", ".###."),
}


def _masks(rgb: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    hi = rgb.max(axis=2).astype(np.int16)
    lo = rgb.min(axis=2).astype(np.int16)
    sat = hi - lo
    accent = sat >= ACCENT_SAT
    ink = (sat <= INK_SAT) & (hi <= INK_VAL)
    return accent, ink


def _bands(profile: np.ndarray, lo: int) -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    start = None
    for i, value in enumerate(profile):
        if value > 0 and start is None:
            start = i
        elif value == 0 and start is not None:
            out.append((start + lo, i + lo))
            start = None
    if start is not None:
        out.append((start + lo, len(profile) + lo))
    return out


def _tilt(patch: np.ndarray) -> float | None:
    ys, xs = np.nonzero(patch)
    if xs.size < 40:
        return None
    x = xs.astype(np.float64) - xs.mean()
    y = ys.astype(np.float64) - ys.mean()
    cxx = float((x * x).mean())
    cyy = float((y * y).mean())
    cxy = float((x * y).mean())
    theta = 0.5 * math.atan2(2.0 * cxy, cxx - cyy)
    vx, vy = math.cos(theta), math.sin(theta)
    if abs(vy) < abs(vx):  # keep the elongated (near vertical) principal axis
        vx, vy = -vy, vx
    if vy < 0:
        vx, vy = -vx, -vy
    return math.degrees(math.atan2(vx, vy))


def _place(name: str, r0: int, c0: int) -> np.ndarray:
    grid = np.zeros((ROWS, HALF), dtype=bool)
    rows = _SHAPES[name]
    for r in range(SYM_SIZE):
        for c in range(SYM_SIZE):
            if rows[r][c] == "#":
                grid[r0 + r, c0 + c] = True
    return grid


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    accent, ink = _masks(rgb)
    if not accent.any() or not ink.any():
        return ABSTAIN

    iy, ix = np.nonzero(ink)
    y0, y1 = int(iy.min()), int(iy.max()) + 1
    x0, x1 = int(ix.min()), int(ix.max()) + 1

    inner = np.zeros_like(accent)
    inner[y0:y1, x0 + 3 : x1 - 3] = accent[y0:y1, x0 + 3 : x1 - 3]
    ay, ax = np.nonzero(inner)
    if ax.size < 30:
        return ABSTAIN
    axis_x = float(np.median(ax))
    if abs(float(ax.max()) - float(ax.min())) > 8:
        return ABSTAIN

    cols = _bands(ink[y0:y1, x0:x1].sum(axis=0), x0)
    rows = _bands(ink[y0:y1, x0:x1].sum(axis=1), y0)
    if len(cols) != COLS or len(rows) != ROWS:
        return ABSTAIN
    if not cols[HALF - 1][1] < axis_x < cols[HALF][0]:
        return ABSTAIN

    signs = np.zeros((ROWS, COLS), dtype=np.int8)
    for r, (ra, rb) in enumerate(rows):
        for c, (ca, cb) in enumerate(cols):
            angle = _tilt(ink[ra:rb, ca:cb])
            if angle is None or abs(angle) < ANGLE_FLOOR_DEG:
                return ABSTAIN
            signs[r, c] = 1 if angle > 0 else -1

    folded = np.zeros((ROWS, HALF), dtype=bool)
    for r in range(ROWS):
        for c in range(HALF):
            folded[r, c] = signs[r, COLS - 1 - c] != -signs[r, c]

    scored = []
    for name in _SHAPES:
        best = min(
            int(np.count_nonzero(folded != _place(name, r0, c0)))
            for r0 in range(ROWS - SYM_SIZE + 1)
            for c0 in range(HALF - SYM_SIZE + 1)
        )
        scored.append((best, name))
    scored.sort()
    if scored[0][0] > MAX_BEST_DIST:
        return ABSTAIN
    if scored[1][0] - scored[0][0] < MIN_RUNNERUP_GAP:
        return ABSTAIN
    return scored[0][1]
