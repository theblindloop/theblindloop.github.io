"""Independent PNG-only inverse for widest_checkpoint_spread."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_PALETTE = (
    (220, 55, 50),
    (45, 105, 210),
    (25, 150, 95),
    (235, 135, 35),
)
_COLOR_TOLERANCE = 48.0
_MIN_MARKER_PIXELS = 90
_COLUMN_GAP_PX = 45
_ABSTAIN_GAP_PX = 5.0


class OracleError(ValueError):
    pass


def _marker_centroids(image: Image.Image, rgb: tuple[int, int, int]) -> list[tuple[float, float]]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    target = np.asarray(rgb, dtype=np.float32)
    mask = np.linalg.norm(pixels - target, axis=2) <= _COLOR_TOLERANCE
    ys, xs = np.nonzero(mask)
    if len(xs) < 4 * _MIN_MARKER_PIXELS:
        raise OracleError(f"insufficient ink for palette color {rgb}")

    order = np.argsort(xs, kind="stable")
    xs, ys = xs[order], ys[order]
    breaks = np.flatnonzero(np.diff(xs) > _COLUMN_GAP_PX) + 1
    groups = np.split(np.arange(len(xs)), breaks)
    groups = [group for group in groups if len(group) >= _MIN_MARKER_PIXELS]
    if len(groups) != 4:
        raise OracleError(f"expected four separated markers for {rgb}, recovered {len(groups)}")
    centers = [(float(xs[group].mean()), float(ys[group].mean())) for group in groups]
    centers.sort(key=lambda point: point[0])
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Recover the winning left-to-right checkpoint solely from marker pixels."""
    if image.width < 320 or image.height < 240:
        raise OracleError("image is too small for the public marker specification")
    by_color = [_marker_centroids(image, rgb) for rgb in _PALETTE]
    ranges = [
        max(points[column][1] for points in by_color)
        - min(points[column][1] for points in by_color)
        for column in range(4)
    ]
    order = sorted(range(4), key=lambda column: ranges[column], reverse=True)
    if not math.isfinite(ranges[order[0]]) or ranges[order[0]] - ranges[order[1]] < _ABSTAIN_GAP_PX:
        raise OracleError("top two checkpoint ranges are raster-ambiguous")
    return f"T{order[0] + 1}"
