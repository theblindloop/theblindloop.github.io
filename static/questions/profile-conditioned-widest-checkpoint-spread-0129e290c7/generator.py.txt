"""Deterministic renderer for the widest-checkpoint-spread question world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 440
PALETTE = ("#dc3732", "#2d69d2", "#19965f", "#eb8723")
CHECKPOINTS = ("T1", "T2", "T3", "T4")
QUESTION = (
    "At which checkpoint is the vertical distance between the highest and lowest "
    "colored marker greatest? Answer T1, T2, T3, or T4."
)
QUARANTINE_PX = 8.0
SAMPLED_FLOOR_PX = 12.0


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _column_spreads(scene: dict) -> list[float]:
    values = scene["values"]
    return [
        max(row[column] for row in values) - min(row[column] for row in values)
        for column in range(4)
    ]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed modulo four fixes only the winning column."""
    rng = random.Random(int(seed))
    plot_height = 292
    winner = int(seed) % 4
    remaining = [index for index in range(4) if index != winner]
    runner = rng.choice(remaining)

    winning_range = rng.uniform(66.0, 76.0)
    if (int(seed) // 4) % 3 == 0:
        requested_gap_px = rng.uniform(SAMPLED_FLOOR_PX, 17.0)
    else:
        requested_gap_px = rng.uniform(25.0, 45.0)
    runner_range = winning_range - requested_gap_px * 100.0 / plot_height
    lower = [runner_range - rng.uniform(8.0, 13.0), runner_range - rng.uniform(17.0, 23.0)]
    lower.sort(reverse=True)

    ranges = [0.0] * 4
    ranges[winner] = winning_range
    ranges[runner] = runner_range
    other_columns = [index for index in remaining if index != runner]
    rng.shuffle(other_columns)
    for column, value in zip(other_columns, lower, strict=True):
        ranges[column] = max(30.0, value)

    values = [[0.0 for _ in range(4)] for _ in range(4)]
    for column, span in enumerate(ranges):
        center = rng.uniform(48.0, 52.0)
        levels = [
            center - span / 2.0,
            center - span / 2.0 + span * rng.uniform(0.27, 0.35),
            center - span / 2.0 + span * rng.uniform(0.65, 0.73),
            center + span / 2.0,
        ]
        permutation = list(range(4))
        rng.shuffle(permutation)
        for series in range(4):
            values[series][column] = round(levels[permutation[series]], 6)

    scene = {
        "values": values,
        "palette": list(PALETTE),
        "plot_left": 98 + rng.randint(-10, 10),
        "plot_top": 62 + rng.randint(-7, 7),
        "plot_width": 450,
        "plot_height": plot_height,
        "width": WIDTH,
        "height": HEIGHT,
    }
    if margin(scene) + 1e-9 < SAMPLED_FLOOR_PX:
        raise AssertionError("sampler produced insufficient runner-up clearance")
    return scene


def _xy(scene: dict, column: int, value: float) -> tuple[float, float]:
    x = scene["plot_left"] + column * scene["plot_width"] / 3.0
    y = scene["plot_top"] + scene["plot_height"] * (1.0 - value / 100.0)
    return x, y


def render(scene: dict) -> Image.Image:
    """Render a chart with neutral connectors and independently colored markers."""
    ss = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * ss, height * ss), "#fbfcfe")
    draw = ImageDraw.Draw(image)

    left = float(scene["plot_left"])
    top = float(scene["plot_top"])
    plot_width = float(scene["plot_width"])
    plot_height = float(scene["plot_height"])
    right, bottom = left + plot_width, top + plot_height

    def scaled(point: tuple[float, float]) -> tuple[int, int]:
        return int(round(point[0] * ss)), int(round(point[1] * ss))

    grid_color = "#dfe3ea"
    axis_color = "#414854"
    for tick in (0, 25, 50, 75, 100):
        y = top + plot_height * (1.0 - tick / 100.0)
        draw.line([scaled((left, y)), scaled((right, y))], fill=grid_color, width=2 * ss)
        draw.text(
            scaled((left - 11, y)),
            str(tick),
            font=_font(12 * ss),
            fill=axis_color,
            anchor="rm",
        )
    draw.line([scaled((left, top)), scaled((left, bottom))], fill=axis_color, width=2 * ss)
    draw.line([scaled((left, bottom)), scaled((right, bottom))], fill=axis_color, width=2 * ss)

    for column, label in enumerate(CHECKPOINTS):
        x, _ = _xy(scene, column, 0.0)
        draw.line([scaled((x, bottom)), scaled((x, bottom + 5))], fill=axis_color, width=2 * ss)
        draw.text(
            scaled((x, bottom + 13)),
            label,
            font=_font(15 * ss),
            fill=axis_color,
            anchor="ma",
        )

    draw.text(
        scaled((left - 56, top + plot_height / 2.0)),
        "VALUE",
        font=_font(13 * ss),
        fill=axis_color,
        anchor="mm",
    )
    draw.text(
        scaled((left + plot_width / 2.0, top - 29)),
        "MEASUREMENTS BY CHECKPOINT",
        font=_font(16 * ss),
        fill="#202631",
        anchor="mm",
    )

    values = scene["values"]
    palette = scene["palette"]
    connector = "#aeb6c2"
    for row in values:
        points = [_xy(scene, column, row[column]) for column in range(4)]
        draw.line([scaled(point) for point in points], fill=connector, width=2 * ss)

    radius = 8.0
    for color, row in zip(palette, values, strict=True):
        for column, value in enumerate(row):
            x, y = _xy(scene, column, value)
            box = [
                int(round((x - radius) * ss)),
                int(round((y - radius) * ss)),
                int(round((x + radius) * ss)),
                int(round((y + radius) * ss)),
            ]
            draw.ellipse(box, fill=color, outline="#ffffff", width=1 * ss)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    spreads = _column_spreads(scene)
    winner = max(range(4), key=lambda index: spreads[index])
    return CHECKPOINTS[winner]


def margin(scene: dict) -> float:
    spreads = sorted(_column_spreads(scene), reverse=True)
    return (spreads[0] - spreads[1]) * float(scene["plot_height"]) / 100.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["values"] = list(reversed(twin["values"]))
    twin["palette"] = list(reversed(twin["palette"]))
    return [("reverse_series_storage", twin)]
