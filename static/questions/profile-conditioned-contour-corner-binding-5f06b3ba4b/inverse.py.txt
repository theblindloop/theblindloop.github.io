"""Independent image-only inverse arm for contour-corner bindings."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

IDENTITY = [
    np.array((24, 174, 190)),
    np.array((210, 61, 151)),
    np.array((55, 158, 91)),
    np.array((239, 133, 39)),
]
TAGS = {
    "red": np.array((214, 63, 58)),
    "blue": np.array((51, 105, 196)),
    "gold": np.array((222, 169, 35)),
}


def _components(mask: np.ndarray, minimum: int = 15) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx); ys.append(yy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys)), len(xs)))
    return found


def _color_centers(rgb: np.ndarray, target: np.ndarray, minimum: int = 15) -> list[tuple[float, float, int]]:
    distance = np.max(np.abs(rgb.astype(np.int16) - target.astype(np.int16)), axis=2)
    return _components(distance <= 18, minimum)


def _cluster_cards(markers: list[tuple[int, float, float]]) -> list[list[tuple[int, float, float]]]:
    if len(markers) != 16:
        raise ValueError("expected sixteen identity landmarks")
    by_y = sorted(markers, key=lambda item: item[2])
    gaps = [by_y[i + 1][2] - by_y[i][2] for i in range(len(by_y) - 1)]
    split = max(range(len(gaps)), key=gaps.__getitem__) + 1
    upper, lower = by_y[:split], by_y[split:]
    if len(upper) != 4 or len(lower) != 12:
        raise ValueError("could not separate reference row from option row")
    ordered_lower = sorted(lower, key=lambda item: item[1])
    groups = [upper] + [ordered_lower[i:i + 4] for i in range(0, 12, 4)]
    valid = [group for group in groups if len(group) == 4 and {item[0] for item in group} == set(range(4))]
    if len(valid) != 4:
        raise ValueError(f"expected four complete landmark cards, found {len(valid)}")
    return valid


def _signature(group: list[tuple[int, float, float]]) -> tuple[int, ...]:
    cx = sum(item[1] for item in group) / 4
    cy = sum(item[2] for item in group) / 4
    radii = [(item[0], ((item[1] - cx) ** 2 + (item[2] - cy) ** 2) ** 0.5) for item in group]
    ordered = sorted(radii, key=lambda item: item[1])
    if ordered[2][1] - ordered[1][1] < 16:
        raise ValueError("corner radius classes are not visibly separated")
    bits = [0, 0, 0, 0]
    for identity, _ in ordered[2:]:
        bits[identity] = 1
    return tuple(bits)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    markers = []
    for identity, target in enumerate(IDENTITY):
        centers = _color_centers(rgb, target, 45)
        if len(centers) != 4:
            raise ValueError(f"identity color {identity} has {len(centers)} landmarks")
        markers.extend((identity, x, y) for x, y, _ in centers)
    groups = _cluster_cards(markers)
    groups.sort(key=lambda group: sum(item[2] for item in group) / 4)
    reference = _signature(groups[0])
    options = sorted(groups[1:], key=lambda group: sum(item[1] for item in group) / 4)
    matches = [index for index, group in enumerate(options) if _signature(group) == reference]
    if len(matches) != 1:
        raise ValueError("pixel evidence does not yield one matching option")
    option_x = sum(item[1] for item in options[matches[0]]) / 4
    tag_hits = []
    for name, target in TAGS.items():
        for x, y, area in _color_centers(rgb, target, 80):
            if y > rgb.shape[0] * 0.70:
                tag_hits.append((name, x, y, area))
    if len(tag_hits) != 3:
        raise ValueError(f"expected three visible tags, found {len(tag_hits)}")
    chosen = min(tag_hits, key=lambda item: abs(item[1] - option_x))
    if abs(chosen[1] - option_x) > 70:
        raise ValueError("matching option has no spatially bound tag")
    return chosen[0]
