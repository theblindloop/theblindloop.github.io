"""Deterministic latent scenes and raster renderer."""

from __future__ import annotations

import math
import random
from itertools import combinations

from PIL import Image, ImageDraw, ImageFont

W, H = 900, 700
IDENTITY = [
    (24, 174, 190),   # cyan
    (210, 61, 151),   # magenta
    (55, 158, 91),    # green
    (239, 133, 39),   # orange
]
TAG_RGB = {
    "red": (214, 63, 58),
    "blue": (51, 105, 196),
    "gold": (222, 169, 35),
}
ALL_BITS = [list(map(int, (i in pair for i in range(4)))) for pair in combinations(range(4), 2)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(ALL_BITS[rng.randrange(len(ALL_BITS))])
    distractors = [bits for bits in ALL_BITS if bits != reference]
    rng.shuffle(distractors)
    options = [list(reference), list(distractors[0]), list(distractors[1])]
    rng.shuffle(options)
    tags = list(TAG_RGB)
    rng.shuffle(tags)
    rotations = [rng.randrange(4) for _ in range(4)]
    return {
        "reference_bits": reference,
        "option_bits": options,
        "rotations": rotations,
        "tag_colors": tags,
    }


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    names = [
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ]
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _point(cx: float, cy: float, radius: float, degrees: float) -> tuple[int, int]:
    angle = math.radians(degrees)
    return round(cx + radius * math.cos(angle)), round(cy + radius * math.sin(angle))


def _draw_card(draw: ImageDraw.ImageDraw, cx: int, cy: int, bits: list[int], rotation: int, tag: str | None) -> None:
    draw.rounded_rectangle((cx - 110, cy - 108, cx + 110, cy + 108), radius=12, fill=(251, 250, 247), outline=(142, 145, 151), width=3)
    points: list[tuple[int, int]] = []
    turn = rotation * 90
    for index in range(4):
        base = turn + index * 90
        points.append(_point(cx, cy, 61, base - 27))
        points.append(_point(cx, cy, 80 if bits[index] else 42, base))
        points.append(_point(cx, cy, 61, base + 27))
    draw.polygon(points, fill=(225, 232, 240), outline=(36, 42, 50), width=7)
    # Identity markers are centered on the signed local contour turn.
    for index, color in enumerate(IDENTITY):
        base = turn + index * 90
        px, py = _point(cx, cy, 80 if bits[index] else 42, base)
        draw.ellipse((px - 10, py - 10, px + 10, py + 10), fill=color, outline=(255, 255, 255), width=3)
        draw.ellipse((px - 10, py - 10, px + 10, py + 10), outline=(31, 35, 41), width=2)
    if tag is not None:
        color = TAG_RGB[tag]
        draw.rounded_rectangle((cx - 37, cy + 116, cx + 37, cy + 148), radius=10, fill=color, outline=(40, 40, 40), width=2)
        label = tag.upper()
        font = _font(15, True)
        box = draw.textbbox((0, 0), label, font=font)
        draw.text((cx - (box[2] - box[0]) / 2, cy + 123), label, fill=(255, 255, 255), font=font)


def render(scene: dict) -> Image.Image:
    reference = scene["reference_bits"]
    options = scene["option_bits"]
    rotations = scene["rotations"]
    tags = scene["tag_colors"]
    image = Image.new("RGB", (W, H), (246, 247, 249))
    draw = ImageDraw.Draw(image)
    title_font = _font(25, True)
    small_font = _font(18, True)
    draw.text((450, 22), "MATCH EACH COLOR TO ITS CONTOUR CORNER", anchor="ma", fill=(30, 35, 42), font=title_font)
    draw.text((450, 62), "OUTWARD TOOTH  or  INWARD NOTCH", anchor="ma", fill=(74, 78, 86), font=_font(18))
    draw.text((450, 91), "REFERENCE", anchor="ma", fill=(38, 43, 50), font=small_font)
    _draw_card(draw, 450, 220, reference, rotations[0], None)
    draw.line((58, 362, 842, 362), fill=(184, 187, 193), width=2)
    draw.text((450, 376), "WHICH TAG HAS THE SAME FOUR COLOR–CORNER BINDINGS?", anchor="ma", fill=(38, 43, 50), font=_font(17, True))
    for idx, cx in enumerate((180, 450, 720)):
        _draw_card(draw, cx, 520, options[idx], rotations[idx + 1], tags[idx])
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_bits"]
    options = scene["option_bits"]
    tags = scene["tag_colors"]
    matches = [i for i, bits in enumerate(options) if bits == reference]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one match")
    return str(tags[matches[0]])


def margin(scene: dict) -> float:
    reference = scene["reference_bits"]
    options = scene["option_bits"]
    distances = [sum(int(a != b) for a, b in zip(reference, bits)) for bits in options]
    nonzero = [value for value in distances if value > 0]
    return float(min(nonzero)) if nonzero else 0.0


def is_quarantined(scene: dict) -> bool:
    reference = scene["reference_bits"]
    options = scene["option_bits"]
    if len(reference) != 4 or len(options) != 3:
        return True
    distances = [sum(int(a != b) for a, b in zip(reference, bits)) for bits in options]
    return distances.count(0) != 1 or any(value < 2 for value in distances if value)

    return []
