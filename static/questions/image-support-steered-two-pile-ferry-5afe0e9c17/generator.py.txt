"""Deterministic renderer for Two-Pile Ferry."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 600
BG = (248, 247, 242)
CARD = (231, 233, 232)
INK = (31, 38, 48)
GOLD = (218, 169, 43)
CYAN = (27, 169, 190)
CORAL = (222, 91, 78)
TOKEN_COLORS = {
    "RED": (205, 55, 67),
    "GREEN": (41, 151, 91),
    "BLUE": (55, 101, 202),
}


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)


def _replay(pile_a: list[str], pile_b: list[str], operations: list[str]) -> tuple[list[str], list[str]]:
    a, b = list(pile_a), list(pile_b)
    for operation in operations:
        if operation == "A_TO_B" and a:
            b.append(a.pop())
        elif operation == "B_TO_A" and b:
            a.append(b.pop())
    return a, b


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene deterministically by requested final top color."""
    rng = random.Random(seed)
    target = ("RED", "GREEN", "BLUE")[seed % 3]
    colors = tuple(TOKEN_COLORS)
    for _ in range(20000):
        pile_a = [rng.choice(colors) for _ in range(rng.randint(2, 4))]
        pile_b = [rng.choice(colors) for _ in range(rng.randint(2, 4))]
        operations = [rng.choice(("A_TO_B", "B_TO_A")) for _ in range(4)]
        # Both directions make the two-pile coupling visible and avoid a unary walk.
        if len(set(operations)) < 2:
            continue
        final_a, _ = _replay(pile_a, pile_b, operations)
        if final_a and final_a[-1] == target:
            return {"pile_a": pile_a, "pile_b": pile_b, "operations": operations}
    raise RuntimeError("balanced sampler exhausted")


def _draw_arrow(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], right: bool) -> None:
    x0, y0, x1, y1 = box
    cy = (y0 + y1) // 2 + 22
    left, right_x = x0 + 36, x1 - 36
    if right:
        draw.line((left, cy, right_x - 14, cy), fill=INK, width=9)
        draw.polygon([(right_x, cy), (right_x - 25, cy - 19), (right_x - 25, cy + 19)], fill=INK)
        draw.text((left - 5, cy + 25), "A", font=_font(22), fill=INK)
        draw.text((right_x - 12, cy + 25), "B", font=_font(22), fill=INK)
    else:
        draw.line((left + 14, cy, right_x, cy), fill=INK, width=9)
        draw.polygon([(left, cy), (left + 25, cy - 19), (left + 25, cy + 19)], fill=INK)
        draw.text((left - 5, cy + 25), "A", font=_font(22), fill=INK)
        draw.text((right_x - 12, cy + 25), "B", font=_font(22), fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    pile_a = list(scene["pile_a"])
    pile_b = list(scene["pile_b"])
    operations = list(scene["operations"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    # A distant, disconnected START component.
    start = (24, 48, 326, 552)
    draw.rectangle(start, fill=CARD, outline=GOLD, width=7)
    draw.text((72, 68), "START", font=_font(32), fill=INK)
    draw.line((70, 490, 280, 490), fill=INK, width=5)
    for x, label, pile in ((112, "A", pile_a), (238, "B", pile_b)):
        draw.text((x - 10, 506), label, font=_font(28), fill=INK)
        for level, color in enumerate(pile):
            cy = 452 - 58 * level
            draw.ellipse((x - 23, cy - 23, x + 23, cy + 23), fill=TOKEN_COLORS[color], outline=INK, width=3)

    # Numbered command cards occupy four separated regions across the other side.
    slots = ((380, 58, 562, 276), (594, 58, 776, 276), (380, 324, 562, 542), (594, 324, 776, 542))
    for number, (operation, box) in enumerate(zip(operations, slots), start=1):
        border = CYAN if operation == "A_TO_B" else CORAL
        draw.rectangle(box, fill=CARD, outline=border, width=7)
        x0, y0, x1, _ = box
        spacing = 22
        start_x = (x0 + x1) // 2 - spacing * (number - 1) // 2
        for index in range(number):
            cx = start_x + index * spacing
            draw.ellipse((cx - 7, y0 + 22, cx + 7, y0 + 36), fill=INK)
        _draw_arrow(draw, box, operation == "A_TO_B")
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    final_a, _ = _replay(list(scene["pile_a"]), list(scene["pile_b"]), list(scene["operations"]))
    return final_a[-1] if final_a else "EMPTY"


def margin(scene: dict[str, Any]) -> float:
    # Minimum clear gap between same-pile tokens and between detached cards.
    _ = (scene["pile_a"], scene["pile_b"], scene["operations"])
    return 12.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    _ = (scene["pile_a"], scene["pile_b"], scene["operations"])
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _ = (scene["pile_a"], scene["pile_b"], scene["operations"])
    return []
