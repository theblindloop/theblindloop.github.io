"""Independent pixel-only inverse for Two-Pile Ferry.

This module deliberately knows only raster colors and generic component geometry.
It does not import any forward-world code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GOLD = (218, 169, 43)
CYAN = (27, 169, 190)
CORAL = (222, 91, 78)
INK = (31, 38, 48)
TOKENS = {
    "RED": (205, 55, 67),
    "GREEN": (41, 151, 91),
    "BLUE": (55, 101, 202),
}


def _mask(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 18) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 1) -> list[tuple[int, int, int, int, int]]:
    remaining = {tuple(item) for item in np.argwhere(mask)}
    found: list[tuple[int, int, int, int, int]] = []
    while remaining:
        first = remaining.pop()
        queue = deque([first])
        ys, xs = [first[0]], [first[1]]
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    ys.append(neighbor[0])
                    xs.append(neighbor[1])
        if len(xs) >= minimum:
            found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _inside(component: tuple[int, int, int, int, int], box: tuple[int, int, int, int, int]) -> bool:
    x0, y0, x1, y1, _ = component
    bx0, by0, bx1, by1, _ = box
    return bx0 < (x0 + x1) / 2 < bx1 and by0 < (y0 + y1) / 2 < by1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    gold = _components(_mask(array, GOLD), minimum=500)
    if len(gold) != 1:
        raise ValueError("START card is not uniquely visible")
    start = gold[0]

    colored_cards: list[tuple[str, tuple[int, int, int, int, int]]] = []
    for operation, color in (("A_TO_B", CYAN), ("B_TO_A", CORAL)):
        for component in _components(_mask(array, color), minimum=500):
            colored_cards.append((operation, component))
    if len(colored_cards) != 4:
        raise ValueError("four transfer cards are required")

    dark_components = _components(_mask(array, INK), minimum=35)
    numbered: dict[int, str] = {}
    for operation, box in colored_cards:
        bx0, by0, bx1, by1, _ = box
        dots = 0
        for component in dark_components:
            x0, y0, x1, y1, pixels = component
            cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
            width, height = x1 - x0 + 1, y1 - y0 + 1
            if bx0 < cx < bx1 and by0 < cy < by0 + (by1 - by0) * 0.28 and 9 <= width <= 18 and 9 <= height <= 18 and 100 <= pixels <= 230:
                dots += 1
        if dots not in (1, 2, 3, 4) or dots in numbered:
            raise ValueError("card dot counts must uniquely encode 1 through 4")
        numbered[dots] = operation
    if set(numbered) != {1, 2, 3, 4}:
        raise ValueError("incomplete operation order")

    token_components: list[tuple[str, tuple[int, int, int, int, int]]] = []
    for name, color in TOKENS.items():
        for component in _components(_mask(array, color), minimum=500):
            if _inside(component, start):
                token_components.append((name, component))
    if not 4 <= len(token_components) <= 8:
        raise ValueError("START piles are incomplete")
    centers_x = sorted({round((component[0] + component[2]) / 2) for _, component in token_components})
    if len(centers_x) != 2 or centers_x[1] - centers_x[0] < 60:
        raise ValueError("two separate piles are required")
    split = sum(centers_x) / 2
    a_tokens = [(name, (component[1] + component[3]) / 2) for name, component in token_components if (component[0] + component[2]) / 2 < split]
    b_tokens = [(name, (component[1] + component[3]) / 2) for name, component in token_components if (component[0] + component[2]) / 2 > split]
    if not a_tokens or not b_tokens:
        raise ValueError("each START pile must be visible")
    # Pixel y grows downward, so bottom-to-top is descending y.
    pile_a = [name for name, _ in sorted(a_tokens, key=lambda item: -item[1])]
    pile_b = [name for name, _ in sorted(b_tokens, key=lambda item: -item[1])]
    for number in range(1, 5):
        operation = numbered[number]
        if operation == "A_TO_B" and pile_a:
            pile_b.append(pile_a.pop())
        elif operation == "B_TO_A" and pile_b:
            pile_a.append(pile_b.pop())
    return pile_a[-1] if pile_a else "EMPTY"
