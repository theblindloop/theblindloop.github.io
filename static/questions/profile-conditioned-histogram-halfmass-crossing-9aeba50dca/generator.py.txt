"""Deterministic renderer and analytic arm for a cumulative histogram decision."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 600
BAR_WIDTH = 54
BAR_GAP = 18
TOTAL_HEIGHT = 600
HALF_HEIGHT = TOTAL_HEIGHT / 2
QUARANTINE_MARGIN = 18.0
BAR_COLORS = ("#2678B2", "#2A9D8F", "#7656A8", "#C76D1D")
DIVIDER_COLOR = "#BE185D"
INK = "#243447"
GRID = "#D8DEE8"
PAPER = "#FAFBFD"


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size=size)


def _partition(total: int, count: int, rng: random.Random) -> list[int]:
    """Partition total into visibly large positive integer parts."""
    if count == 0:
        if total != 0:
            raise ValueError("nonzero total with no parts")
        return []
    minimum, maximum = 42, 285
    values: list[int] = []
    remaining = total
    for index in range(count):
        left = count - index - 1
        lo = max(minimum, remaining - maximum * left)
        hi = min(maximum, remaining - minimum * left)
        if lo > hi:
            raise ValueError("unpackable partition")
        value = rng.randint(lo, hi)
        values.append(value)
        remaining -= value
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    crossing_bin = 2 + (int(seed) % 4)  # 1-based bin index: 2, 3, 4, or 5
    # One quarter of consecutive seeds deliberately approaches the half boundary
    # without entering quarantine; the other scenes span comfortable clearances.
    if (int(seed) // 4) % 4 == 0:
        before = 300 - rng.randint(22, 30)
        through = 300 + rng.randint(22, 30)
    else:
        before = rng.randint(220, 258)
        through = rng.randint(342, 380)
    crossing_height = through - before
    heights = (
        _partition(before, crossing_bin - 1, rng)
        + [crossing_height]
        + _partition(TOTAL_HEIGHT - through, 6 - crossing_bin, rng)
    )
    x_shift = rng.randint(-10, 10)
    baseline = 462 + rng.randint(-7, 7)
    start_x = 82 + x_shift
    color = BAR_COLORS[(int(seed) // 4) % len(BAR_COLORS)]
    bars = [
        {"x": start_x + index * (BAR_WIDTH + BAR_GAP), "height": height, "color": color}
        for index, height in enumerate(heights)
    ]
    decorations = [
        {"x": start_x + rng.randint(0, 5 * (BAR_WIDTH + BAR_GAP) + BAR_WIDTH),
         "y": 112 + rng.randint(-5, 5)}
        for _ in range(3)
    ]
    return {"bars": bars, "baseline": baseline, "decorations": decorations}


def _ordered_bars(scene: dict) -> list[dict]:
    bars = scene["bars"]
    if len(bars) != 6:
        raise ValueError("exactly six bars are required")
    return sorted(bars, key=lambda item: int(item["x"]))


def _prefixes(scene: dict) -> tuple[list[int], int]:
    heights = [int(bar["height"]) for bar in _ordered_bars(scene)]
    prefixes: list[int] = []
    running = 0
    for value in heights:
        if value <= 0:
            raise ValueError("bar heights must be positive")
        running += value
        prefixes.append(running)
    return prefixes, running


def analytic_gold(scene: dict) -> str:
    prefixes, total = _prefixes(scene)
    for divider_index, bin_index in enumerate(range(1, 5), start=1):
        if 2 * prefixes[bin_index] >= total:
            return f"D{divider_index}"
    raise ValueError("half-total crossing is outside D1-D4")


def margin(scene: dict) -> float:
    prefixes, total = _prefixes(scene)
    half = total / 2.0
    crossing = next(index for index, value in enumerate(prefixes) if value >= half)
    before = prefixes[crossing - 1] if crossing else 0
    return float(min(half - before, prefixes[crossing] - half))


def is_quarantined(scene: dict) -> bool:
    bars = _ordered_bars(scene)
    return margin(scene) < QUARANTINE_MARGIN or min(int(bar["height"]) for bar in bars) < 36


def _divider_xs(bars: list[dict]) -> list[int]:
    # Four candidates lie after bins 2 through 5.
    return [
        (int(bars[index]["x"]) + BAR_WIDTH + int(bars[index + 1]["x"])) // 2
        for index in range(1, 5)
    ]


def render(scene: dict) -> Image.Image:
    bars = _ordered_bars(scene)
    baseline = int(scene["baseline"])
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    left = int(bars[0]["x"]) - 20
    right = int(bars[-1]["x"]) + BAR_WIDTH + 20

    draw.text((left, 28), "CUMULATIVE HALF", font=_font(26, True), fill=INK)
    draw.text((left, 65), "equal-width bars  •  accumulate from left", font=_font(17), fill=INK)
    for y in range(baseline - 240, baseline, 60):
        draw.line((left, y, right, y), fill=GRID, width=1)
    draw.line((left, baseline, right, baseline), fill=INK, width=3)

    # Answer-neutral decorations make erasure/permutation checks nontrivial.
    for item in scene.get("decorations", []):
        x, y = int(item["x"]), int(item["y"])
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill="#AAB4C3")

    for bar in scene["bars"]:  # latent draw order is deliberately irrelevant
        x = int(bar["x"])
        height = int(bar["height"])
        draw.rectangle((x, baseline - height, x + BAR_WIDTH - 1, baseline - 1), fill=bar["color"])

    dividers = _divider_xs(bars)
    for number, x in enumerate(dividers, start=1):
        for y in range(baseline - 316, baseline - 10, 20):
            draw.rectangle((x - 2, y, x + 1, min(y + 11, baseline - 10)), fill=DIVIDER_COLOR)
        label = f"D{number}"
        box = draw.textbbox((0, 0), label, font=_font(18, True))
        draw.text((x - (box[2] - box[0]) / 2, baseline + 12), label, font=_font(18, True), fill=INK)

    # A visible, answer-neutral total reference exposes the 50% boundary.
    ref_left = right + 48
    ref_right = ref_left + 72
    ref_bottom = baseline
    ref_top = baseline - 270
    draw.rectangle((ref_left, ref_top, ref_right, ref_bottom), fill="#E5E9F0", outline=INK, width=3)
    half_y = (ref_top + ref_bottom) // 2
    for x in range(ref_left - 8, ref_right + 9, 14):
        draw.line((x, half_y, min(x + 8, ref_right + 8), half_y), fill=INK, width=3)
    draw.text((ref_left + 8, ref_top - 30), "TOTAL", font=_font(16, True), fill=INK)
    draw.text((ref_right + 10, half_y - 10), "50%", font=_font(16, True), fill=INK)
    draw.text((left, baseline + 60), "Which divider is the first cumulative half crossing?", font=_font(17), fill=INK)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["bars"] = list(reversed(twin["bars"]))
    return [("reverse_disjoint_bar_draw_order", twin)]
