"""Independent PNG-only inverse for the histogram cumulative-half decision."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Public ink specification duplicated intentionally; no renderer state is imported.
_BAR_RGBS = ((38, 120, 178), (42, 157, 143), (118, 86, 168), (199, 109, 29))
_DIVIDER_RGB = (190, 24, 93)


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    if values.size == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = previous = int(values[0])
    for raw in values[1:]:
        value = int(raw)
        if value != previous + 1:
            groups.append((start, previous))
            start = value
        previous = value
    groups.append((start, previous))
    return groups


def _measure_bars(array: np.ndarray) -> list[dict]:
    mask = np.zeros(array.shape[:2], dtype=bool)
    for rgb in _BAR_RGBS:
        mask |= np.all(array == np.array(rgb, dtype=np.uint8), axis=2)
    # A real bar supplies many exact-color pixels in every interior column.
    x_groups = _runs(np.flatnonzero(mask.sum(axis=0) >= 30))
    if len(x_groups) != 6:
        raise ValueError(f"expected six separable colored bars, recovered {len(x_groups)}")
    bars: list[dict] = []
    for x0, x1 in x_groups:
        sub = mask[:, x0 : x1 + 1]
        ys, xs = np.nonzero(sub)
        if len(ys) < 1200:
            raise ValueError("colored component is too small to be a bar")
        y0, y1 = int(ys.min()), int(ys.max())
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        occupancy = float(sub[y0 : y1 + 1].mean())
        if not 45 <= width <= 62 or height < 36 or occupancy < 0.985:
            raise ValueError("colored component is not a solid equal-width bar")
        bars.append({"x0": x0, "x1": x1, "center": (x0 + x1) / 2, "height": height, "bottom": y1})
    widths = [bar["x1"] - bar["x0"] + 1 for bar in bars]
    bottoms = [bar["bottom"] for bar in bars]
    if max(widths) - min(widths) > 1 or max(bottoms) - min(bottoms) > 1:
        raise ValueError("bars do not share a width and baseline")
    return sorted(bars, key=lambda item: item["center"])


def _measure_dividers(array: np.ndarray, bars: list[dict]) -> list[float]:
    mask = np.all(array == np.array(_DIVIDER_RGB, dtype=np.uint8), axis=2)
    # Each dashed vertical divider has substantial ink in four adjacent columns.
    x_groups = _runs(np.flatnonzero(mask.sum(axis=0) >= 80))
    if len(x_groups) != 4:
        raise ValueError(f"expected four visible dividers, recovered {len(x_groups)}")
    centers = [(x0 + x1) / 2 for x0, x1 in x_groups]
    counts_left = [sum(bar["center"] < divider for bar in bars) for divider in centers]
    if counts_left != [2, 3, 4, 5]:
        raise ValueError("divider ordering is not the declared D1-D4 layout")
    return centers


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 300:
        raise ValueError("image is too small for stable measurement")
    bars = _measure_bars(array)
    dividers = _measure_dividers(array, bars)
    total = sum(int(bar["height"]) for bar in bars)
    if total <= 0:
        raise ValueError("bar total is empty")
    for index, divider in enumerate(dividers, start=1):
        cumulative = sum(int(bar["height"]) for bar in bars if bar["center"] < divider)
        if 2 * cumulative == total:
            raise ValueError("exact half crossing is raster-ambiguous")
        if 2 * cumulative > total:
            clearance = min(cumulative - total / 2, total / 2 - (cumulative - int(bars[index]["height"])))
            if clearance < 8:
                raise ValueError("half crossing has insufficient pixel clearance")
            return f"D{index}"
    raise ValueError("half-total crossing is outside the four visible dividers")
