from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 520
BACKGROUND = (248, 250, 252)
FRAME = (166, 179, 190)
INK = (32, 38, 44)
CYAN = (0, 174, 194)
MARKERS = {
    "red": (220, 54, 66),
    "gold": (226, 157, 28),
    "violet": (125, 72, 190),
    "green": (34, 154, 92),
}
PANEL_BOXES = ((34, 100, 302, 286), (338, 100, 606, 286),
               (34, 310, 302, 496), (338, 310, 606, 496))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A7C3)
    guide = "H" if int(seed) % 2 == 0 else "V"
    winner = int(seed) % 4
    answer_color = (int(seed) // 4) % 4
    palette = ["red", "gold", "violet", "green"]
    marker_colors = [None] * 4
    marker_colors[winner] = palette[answer_color]
    remaining_colors = [c for c in palette if c != palette[answer_color]]
    rng.shuffle(remaining_colors)
    for panel in range(4):
        if marker_colors[panel] is None:
            marker_colors[panel] = remaining_colors.pop()

    win_count = 6 + ((int(seed) // 16) % 2)
    counts = [win_count - 1, win_count - 2, win_count - 3]
    rng.shuffle(counts)
    match_counts = []
    for panel in range(4):
        match_counts.append(win_count if panel == winner else counts.pop())

    patterns = []
    other = "V" if guide == "H" else "H"
    for count in match_counts:
        values = [guide] * count + [other] * (9 - count)
        rng.shuffle(values)
        patterns.append("".join(values))
    return {"guide": guide, "marker_colors": marker_colors, "patterns": patterns}


def _match_counts(scene: dict) -> list[int]:
    key = scene["guide"]
    return [pattern.count(key) for pattern in scene["patterns"]]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Remote visual key, centered and well separated from all candidate fields.
    draw.rounded_rectangle((256, 18, 384, 82), radius=12, outline=FRAME, width=3, fill=(255, 255, 255))
    if scene["guide"] == "H":
        draw.rounded_rectangle((282, 43, 358, 57), radius=7, fill=CYAN)
    else:
        draw.rounded_rectangle((313, 26, 327, 74), radius=7, fill=CYAN)

    for panel, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=10, outline=FRAME, width=3, fill=(255, 255, 255))
        color = MARKERS[scene["marker_colors"][panel]]
        draw.ellipse((x0 + 13, y0 + 12, x0 + 35, y0 + 34), fill=color)
        pattern = scene["patterns"][panel]
        for index, orientation in enumerate(pattern):
            row, col = divmod(index, 3)
            cx = x0 + 77 + col * 70
            cy = y0 + 62 + row * 48
            if orientation == "H":
                rect = (cx - 23, cy - 7, cx + 23, cy + 7)
            else:
                rect = (cx - 7, cy - 19, cx + 7, cy + 19)
            draw.rounded_rectangle(rect, radius=6, fill=INK)
    return image


def analytic_gold(scene: dict) -> str:
    counts = _match_counts(scene)
    best = max(counts)
    winners = [i for i, value in enumerate(counts) if value == best]
    if len(winners) != 1:
        raise ValueError("no unique analytic winner")
    return scene["marker_colors"][winners[0]]


def margin(scene: dict) -> float:
    counts = sorted(_match_counts(scene), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    return len(scene["patterns"]) != 4 or any(len(p) != 9 for p in scene["patterns"]) or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
