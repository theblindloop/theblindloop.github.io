from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_INK = np.array((32, 38, 44), dtype=np.int16)
_CYAN = np.array((0, 174, 194), dtype=np.int16)
_MARKERS = {
    "red": np.array((220, 54, 66), dtype=np.int16),
    "gold": np.array((226, 157, 28), dtype=np.int16),
    "violet": np.array((125, 72, 190), dtype=np.int16),
    "green": np.array((34, 154, 92), dtype=np.int16),
}


def _mask_near(arr: np.ndarray, rgb: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - rgb), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx); ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs) + 1, max(ys) + 1, len(xs)))
    return output


def _single_bbox(mask: np.ndarray, label: str) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 50:
        raise ValueError(f"missing {label}")
    return int(xs.min()), int(ys.min()), int(xs.max()) + 1, int(ys.max()) + 1


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 200:
        raise ValueError("unsupported raster")

    guide_box = _single_bbox(_mask_near(arr, _CYAN), "cyan key")
    guide_w, guide_h = guide_box[2] - guide_box[0], guide_box[3] - guide_box[1]
    if max(guide_w, guide_h) / max(1, min(guide_w, guide_h)) < 2.5:
        raise ValueError("ambiguous cyan key orientation")
    guide = "H" if guide_w > guide_h else "V"

    # Discover one marker of every declared answer color. Their visible locations
    # anchor the four candidate regions; pale frames are presentation, not a
    # brittle dependency of the inverse arm.
    marker_regions = []
    for name, rgb in _MARKERS.items():
        ys, xs = np.nonzero(_mask_near(arr, rgb))
        if len(xs) < 80:
            raise ValueError(f"missing {name} field marker")
        marker_regions.append((float(xs.mean()), float(ys.mean()), name))

    bars = []
    for x0, y0, x1, y1, area in _components(_mask_near(arr, _INK)):
        w, h = x1 - x0, y1 - y0
        if area >= 350 and max(w, h) >= 32 and min(w, h) >= 10:
            if max(w, h) / min(w, h) < 2.0:
                raise ValueError("ambiguous black bar orientation")
            bars.append((x0, y0, x1, y1, "H" if w > h else "V"))
    if len(bars) < 8:
        raise ValueError("too few visible black bars")

    scored = []
    for mx, my, name in marker_regions:
        # The field extends rightward and downward from its marker. These bounds
        # are recovered relative to a visible anchor, never from renderer state.
        local = [bar for bar in bars
                 if mx + 20 < (bar[0] + bar[2]) / 2 < mx + 255
                 and my + 20 < (bar[1] + bar[3]) / 2 < my + 170]
        if len(local) < 3:
            raise ValueError(f"too little visible evidence in {name} field")
        scored.append((sum(bar[4] == guide for bar in local), name))

    scored.sort(reverse=True)
    if len(scored) != 4 or scored[0][0] - scored[1][0] < 1:
        raise ValueError("no unique runner-up-separated winner")
    return scored[0][1]
