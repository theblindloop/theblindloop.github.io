"""Deterministic renderer and analytic arm for Bent Route Interval Comparison."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
ROUTE = (24, 38, 58)
DISTRACTOR = (224, 230, 236)
MARKER_RGB = {
    "red": (218, 62, 70),
    "gold": (238, 174, 24),
    "green": (38, 157, 91),
}
DECISIONS = ("red-gold", "gold-green")


def _half_points(start: int, end: int, secondary: int, long: bool) -> list[tuple[int, int]]:
    primary = [start, start + 7, start + 14, start + 22, start + 29, end]
    offsets = [0, -9, 9, -9, 9, 0] if long else [0, 1, -1, 1, -1, 0]
    return list(zip(primary, [secondary + value for value in offsets]))


def _xy(primary: int, secondary: int, horizontal: bool) -> list[int]:
    return [primary, secondary] if horizontal else [secondary, primary]


def sample_scene(seed: int) -> dict:
    """Sample independent side, endpoint-color, orientation, and location factors."""

    seed = int(seed)
    rng = random.Random(seed ^ 0xA17C04)
    horizontal = ((seed // 4) % 2) == 0
    primary_center = rng.choice((128, 192, 256, 320, 384))
    secondary_center = rng.choice((96, 160, 224, 288, 352, 416))
    long_first = (seed % 2) == 0
    red_first = ((seed // 2) % 2) == 0

    first = _half_points(primary_center - 36, primary_center, secondary_center, long_first)
    second = _half_points(primary_center, primary_center + 36, secondary_center, not long_first)
    ordered = first + second[1:]
    route_points = [_xy(primary, secondary, horizontal) for primary, secondary in ordered]
    marker_colors = ["red", "gold", "green"] if red_first else ["green", "gold", "red"]

    # Remote marks are deliberately nondiagnostic and never touch the local route inset.
    decorations: list[list[int]] = []
    route_x = [point[0] for point in route_points]
    route_y = [point[1] for point in route_points]
    for _ in range(10):
        for _attempt in range(100):
            x = rng.randint(34, WIDTH - 35)
            y = rng.randint(34, HEIGHT - 35)
            if not (min(route_x) - 48 <= x <= max(route_x) + 48 and min(route_y) - 48 <= y <= max(route_y) + 48):
                decorations.append([x, y, rng.choice((-1, 1))])
                break
        else:
            raise RuntimeError("could not place nuisance dash")

    return {
        "route_points": route_points,
        "marker_indices": [0, 5, 10],
        "marker_colors": marker_colors,
        "decorations": decorations,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=color)
    draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=BACKGROUND)


def _validated(scene: dict) -> tuple[list[list[int]], list[int], list[str], list[list[int]]]:
    points = [[int(value) for value in point] for point in scene["route_points"]]
    indices = [int(value) for value in scene["marker_indices"]]
    colors = [str(value) for value in scene["marker_colors"]]
    decorations = [[int(value) for value in item] for item in scene["decorations"]]
    if int(scene["width"]) != WIDTH or int(scene["height"]) != HEIGHT:
        raise ValueError("unexpected canvas")
    if len(points) != 11 or any(len(point) != 2 for point in points):
        raise ValueError("route must contain eleven ordered vertices")
    if len(indices) != 3 or len(colors) != 3 or set(colors) != set(MARKER_RGB):
        raise ValueError("scene must contain red, gold, and green rings exactly once")
    if len(set(indices)) != 3 or any(index < 0 or index >= len(points) for index in indices):
        raise ValueError("invalid marker stations")
    if len(decorations) != 10 or any(len(item) != 3 for item in decorations):
        raise ValueError("invalid nuisance marks")
    return points, indices, colors, decorations


def render(scene: dict) -> Image.Image:
    points, indices, colors, decorations = _validated(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, slope in decorations:
        draw.line((x - 3, y - 3 * slope, x + 3, y + 3 * slope), fill=DISTRACTOR, width=2)
    for index, color in zip(indices, colors):
        _ring(draw, tuple(points[index]), MARKER_RGB[color])
    draw.line([tuple(point) for point in points], fill=ROUTE, width=5, joint="curve")
    return image


def _interval_lengths(scene: dict) -> dict[str, float]:
    points, indices, colors, _ = _validated(scene)
    stations = sorted((index, color) for index, color in zip(indices, colors))
    if [color for _, color in stations][1] != "gold":
        raise ValueError("gold must be the middle station")
    if {stations[0][1], stations[2][1]} != {"red", "green"}:
        raise ValueError("red and green must be endpoint stations")

    def length(a: int, b: int) -> float:
        return sum(
            math.hypot(points[i + 1][0] - points[i][0], points[i + 1][1] - points[i][1])
            for i in range(a, b)
        )

    first = length(stations[0][0], stations[1][0])
    second = length(stations[1][0], stations[2][0])
    if stations[0][1] == "red":
        return {"red-gold": first, "gold-green": second}
    return {"red-gold": second, "gold-green": first}


def analytic_gold(scene: dict) -> str:
    lengths = _interval_lengths(scene)
    if abs(lengths["red-gold"] - lengths["gold-green"]) < 1e-9:
        raise ValueError("route intervals tie")
    return max(lengths, key=lengths.get)


def margin(scene: dict) -> float:
    lengths = _interval_lengths(scene)
    return float(abs(lengths["red-gold"] - lengths["gold-green"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    points = [list(point) for point in scene["route_points"]]
    twin = dict(scene)
    twin["route_points"] = list(reversed(points))
    twin["marker_indices"] = [len(points) - 1 - int(index) for index in scene["marker_indices"]]
    twin["marker_colors"] = list(scene["marker_colors"])
    twin["decorations"] = [list(item) for item in scene["decorations"]]
    return [("reverse_route_representation", twin)]
