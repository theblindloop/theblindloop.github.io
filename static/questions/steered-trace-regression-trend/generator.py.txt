"""Deterministic sampler and renderer for a distributed regression-trend task."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw


MIN_EDGE_RISE = 28.0


def _centerline(points: list[list[int]]) -> tuple[np.ndarray, np.ndarray]:
    """Sample the piecewise-linear route once at each covered integer x."""
    xs = np.arange(int(points[0][0]), int(points[-1][0]) + 1, dtype=np.float64)
    vertex_x = np.asarray([point[0] for point in points], dtype=np.float64)
    vertex_y = np.asarray([point[1] for point in points], dtype=np.float64)
    return xs, np.interp(xs, vertex_x, vertex_y)


def _edge_rise(points: list[list[int]]) -> float:
    """Return fitted image-y change over the route span (positive is downward)."""
    xs, ys = _centerline(points)
    centered = xs - xs.mean()
    slope = float(np.dot(centered, ys - ys.mean()) / np.dot(centered, centered))
    return slope * float(xs[-1] - xs[0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width, height = 768, 480
    x0 = rng.randint(45, 58)
    x1 = rng.randint(710, 723)
    xs = [round(x0 + (x1 - x0) * index / 12) for index in range(13)]
    wanted_image_slope = -1 if int(seed) % 2 == 0 else 1
    endpoint_same = (int(seed) // 2) % 2 == 0

    for _ in range(5000):
        base = rng.randint(222, 258)
        offsets = [rng.randint(-112, 112) for _ in xs]
        # Mild smoothing keeps the route legible while retaining strong reversals.
        offsets = [
            round((offsets[max(0, i - 1)] + 2 * offsets[i] + offsets[min(12, i + 1)]) / 4)
            for i in range(13)
        ]
        points = [[x, base + offset] for x, offset in zip(xs, offsets)]
        if min(y for _, y in points) < 78 or max(y for _, y in points) > 392:
            continue
        edge_rise = _edge_rise(points)
        if not 42.0 <= abs(edge_rise) <= 112.0:
            continue
        if (1 if edge_rise > 0 else -1) != wanted_image_slope:
            continue
        endpoint_delta = points[-1][1] - points[0][1]
        if abs(endpoint_delta) < 24:
            continue
        endpoint_sign = 1 if endpoint_delta > 0 else -1
        if (endpoint_sign == wanted_image_slope) != endpoint_same:
            continue
        steps = [points[i + 1][1] - points[i][1] for i in range(12)]
        turns = sum((a > 0) != (b > 0) for a, b in zip(steps, steps[1:]) if a and b)
        if turns < 4 or max(y for _, y in points) - min(y for _, y in points) < 105:
            continue
        break
    else:
        raise RuntimeError(f"could not sample a robust trace for seed {seed}")

    return {
        "width": width,
        "height": height,
        "points": points,
        "reference_y": rng.randint(210, 270),
        "stroke_width": rng.choice([6, 7, 8]),
        "route_color": [18, 157, 187],
        "reference_color": [219, 126, 37],
        "background": [250, 250, 247],
    }


def analytic_gold(scene: dict) -> str:
    return "downward" if _edge_rise(scene["points"]) > 0 else "upward"


def margin(scene: dict) -> float:
    return abs(_edge_rise(scene["points"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_EDGE_RISE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    frame_color = (105, 112, 116)
    draw.line([(28 * scale, 54 * scale), (28 * scale, 428 * scale)], fill=frame_color, width=2 * scale)
    draw.line(
        [(28 * scale, 428 * scale), ((width - 28) * scale, 428 * scale)],
        fill=frame_color,
        width=2 * scale,
    )
    reference_y = int(scene["reference_y"])
    reference_color = tuple(scene["reference_color"])
    for x in range(40, width - 40, 28):
        draw.line(
            [(x * scale, reference_y * scale), (min(x + 15, width - 40) * scale, reference_y * scale)],
            fill=reference_color,
            width=3 * scale,
        )
    points = [(int(x) * scale, int(y) * scale) for x, y in scene["points"]]
    stroke = int(scene["stroke_width"]) * scale
    route_color = tuple(scene["route_color"])
    draw.line(points, fill=route_color, width=stroke, joint="curve")
    radius = max(3, int(scene["stroke_width"]) // 2) * scale
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=route_color)
    return image.resize((width, height), Image.Resampling.LANCZOS)
