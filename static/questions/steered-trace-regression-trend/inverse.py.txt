"""Independent pixel-only inverse arm for the global trace trend."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _cyan_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    return (red < 105) & (green > 105) & (blue > 125) & ((blue - red) > 70)


def _edge_rise_from_pixels(image: Image.Image) -> float:
    cyan = _cyan_mask(image)
    height, width = cyan.shape
    if int(cyan.sum()) < 2100:
        raise OracleAbstention("too little cyan trace ink")
    xs = np.flatnonzero(cyan.any(axis=0))
    if len(xs) < width * 0.78 or xs[0] > width * 0.10 or xs[-1] < width * 0.90:
        raise OracleAbstention("route does not span the image")
    if len(xs) > 1 and int(np.diff(xs).max()) > 3:
        raise OracleAbstention("route is visibly broken")

    full_x = np.arange(int(xs[0]), int(xs[-1]) + 1, dtype=np.float64)
    center_y = np.empty(len(full_x), dtype=np.float64)
    for index, x in enumerate(full_x.astype(int)):
        ys = np.flatnonzero(cyan[:, x])
        if not len(ys):
            raise OracleAbstention("route has a missing column")
        center_y[index] = float(np.median(ys))
    centered = full_x - full_x.mean()
    slope = float(np.dot(centered, center_y - center_y.mean()) / np.dot(centered, centered))
    edge_rise = slope * float(full_x[-1] - full_x[0])
    if abs(edge_rise) < 20.0:
        raise OracleAbstention("pixel regression is too close to horizontal")
    return edge_rise


def decision_from_image(image: Image.Image) -> str:
    return "downward" if _edge_rise_from_pixels(image) > 0 else "upward"
