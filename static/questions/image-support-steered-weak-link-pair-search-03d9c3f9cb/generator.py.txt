"""Renderer for the weak-link pair search world.

Four coloured pairs of free-floating upright bars lie on a plain page. The two
bars of one pair share a colour and sit diagonally next to each other with plain
background visible between them; the four pairs sit far apart. Each pair is
scored by the length of its *shorter* bar, and exactly one pair has the greatest
such score. The tallest bar on the page always belongs to a losing pair, so the
answer can never be read off a single salient bar.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "weak_link_pair_search-renderer-0.1.0"

CANVAS_W = 1024
CANVAS_H = 768
GRID = 8                      # the page is thought of as an eight-by-eight lattice
CELL_W = CANVAS_W // GRID     # 128
CELL_H = CANVAS_H // GRID     # 96
INSET_X = 32                  # keeps a bar clear of its lattice cell's side edges
INSET_Y = 24                  # keeps a bar clear of its lattice cell's top and bottom
ZONE_W = CELL_W - 2 * INSET_X  # 64 px of horizontal freedom for one bar
ZONE_H = CELL_H - 2 * INSET_Y  # 48 px of vertical freedom for one bar

BAR_W = 14                    # every bar is drawn with this fixed width
MIN_H = 12
MAX_H = 40
SPECK = 3
SPECK_CLEARANCE = 26
QUARANTINE_MARGIN = 5.0

COLOUR_NAMES = ("red", "blue", "green", "purple")
BAR_RGB = {
    "red": (192, 44, 40),
    "blue": (34, 78, 176),
    "green": (28, 124, 60),
    "purple": (122, 56, 158),
}
SPECK_RGB = (110, 110, 110)
BACKGROUNDS = (
    (246, 244, 240),
    (240, 243, 246),
    (244, 241, 236),
    (238, 240, 238),
)


def _pair_heights(rng: random.Random, winner: int, sum_rank: int, margin: int) -> list | None:
    """Four (short, tall) height pairs meeting every counterbalancing rule."""

    for _ in range(600):
        short_w = rng.randint(MIN_H + margin, MIN_H + margin + 6)
        shorts = [0, 0, 0, 0]
        shorts[winner] = short_w
        losers = [index for index in range(4) if index != winner]
        rng.shuffle(losers)
        shorts[losers[0]] = short_w - margin
        for index in losers[1:]:
            shorts[index] = rng.randint(MIN_H, short_w - margin)
        talls = [
            shorts[index] + rng.randint(3, min(22, MAX_H - shorts[index]))
            for index in range(4)
        ]
        if max(talls) > MAX_H:
            continue
        # The single tallest bar on the page must belong to a losing pair.
        if talls[winner] >= max(talls) - 2:
            continue
        # The most nearly equal pair must not be the winning pair either.
        spreads = [talls[index] - shorts[index] for index in range(4)]
        if spreads[winner] <= min(spreads) + 1:
            continue
        totals = [shorts[index] + talls[index] for index in range(4)]
        if len(set(totals)) != 4:
            continue
        # Total ink rank of the winner is driven to a prescribed value, so ink
        # cannot indicate the answer in either direction across the dataset.
        rank = sorted(range(4), key=lambda i: -totals[i]).index(winner) + 1
        if rank != sum_rank:
            continue
        return [[shorts[index], talls[index]] for index in range(4)]
    return None


def sample_scene(seed: int) -> dict:
    """Deterministic scene: four coloured bar pairs plus inert background specks."""

    rng = random.Random(seed * 6151 + 29)
    winner = (seed // 4) % 4
    winner_colour = COLOUR_NAMES[seed % 4]
    decision_margin = 3 + (seed // 4) % 8
    sum_rank = (seed // 4) % 4 + 1

    heights = None
    # The prescribed total-ink rank of the winner is tried first; the remaining
    # ranks are tried in a fixed order so that sampling stays deterministic even
    # when a rank is unreachable for the drawn decision margin.
    for rank in [sum_rank] + [value for value in (2, 3, 1, 4) if value != sum_rank]:
        heights = _pair_heights(rng, winner, rank, decision_margin)
        if heights is not None:
            break
    if heights is None:  # pragma: no cover - the search succeeds well inside its budget
        raise RuntimeError("failed to sample counterbalanced pair heights")

    quadrants = [0, 1, 2, 3]
    rng.shuffle(quadrants)
    # The winning colour cycles with the seed, independently of which record
    # holds the winning heights and of where the pairs are placed on the page.
    rest = [name for name in COLOUR_NAMES if name != winner_colour]
    rng.shuffle(rest)
    colours = []
    for index in range(4):
        colours.append(winner_colour if index == winner else rest[len(colours) - (index > winner)])

    pairs = []
    for index in range(4):
        quadrant = quadrants[index]
        row = 4 * (quadrant // 2) + rng.randint(0, 1)
        col = 4 * (quadrant % 2) + rng.randint(0, 1)
        short_h, tall_h = heights[index]
        order = [short_h, tall_h]
        if rng.random() < 0.5:
            order = [tall_h, short_h]
        bars = []
        for step, height in enumerate(order):
            cell_r, cell_c = row + step, col + step
            bars.append(
                {
                    "cell_r": cell_r,
                    "cell_c": cell_c,
                    "jx": rng.randint(0, ZONE_W - BAR_W),
                    "jy": rng.randint(0, ZONE_H - height),
                    "height": height,
                }
            )
        pairs.append({"colour": colours[index], "bars": bars})

    scene = {
        "width": CANVAS_W,
        "height": CANVAS_H,
        "background": rng.randrange(len(BACKGROUNDS)),
        "pairs": pairs,
        "specks": [],
    }

    boxes = [bar_box(bar) for pair in pairs for bar in pair["bars"]]
    specks: list = []
    target = rng.randint(6, 14)
    guard = 0
    while len(specks) < target and guard < 4000:
        guard += 1
        sx = rng.randint(10, CANVAS_W - 10 - SPECK)
        sy = rng.randint(10, CANVAS_H - 10 - SPECK)
        if any(
            sx + SPECK + SPECK_CLEARANCE >= x0
            and sx <= x1 + SPECK_CLEARANCE
            and sy + SPECK + SPECK_CLEARANCE >= y0
            and sy <= y1 + SPECK_CLEARANCE
            for x0, y0, x1, y1 in boxes
        ):
            continue
        if any(abs(sx - ox) < 4 * SPECK and abs(sy - oy) < 4 * SPECK for ox, oy in specks):
            continue
        specks.append([sx, sy])
    scene["specks"] = specks
    return scene


def bar_box(bar: dict) -> tuple:
    """Inclusive pixel rectangle of one bar."""

    x0 = int(bar["cell_c"]) * CELL_W + INSET_X + int(bar["jx"])
    y0 = int(bar["cell_r"]) * CELL_H + INSET_Y + int(bar["jy"])
    return (x0, y0, x0 + BAR_W - 1, y0 + int(bar["height"]) - 1)


def pair_score(bars: list) -> int:
    """Length in pixels of the shorter bar of one coloured pair."""

    return min(int(bar["height"]) for bar in bars)


def analytic_gold(scene: dict) -> str:
    """Colour of the pair whose shorter bar is the longest."""

    scores = {pair["colour"]: pair_score(pair["bars"]) for pair in scene["pairs"]}
    ranked = sorted(scores, key=lambda name: -scores[name])
    if scores[ranked[0]] <= scores[ranked[1]]:
        raise ValueError("scene does not have a unique weak-link winner")
    return ranked[0]


def margin(scene: dict) -> float:
    """Runner-up decision margin, in pixels of shorter-bar length."""

    scores = sorted((pair_score(pair["bars"]) for pair in scene["pairs"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes where a second pair almost matches the winner."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving rewrites of the latent scene."""

    out = []
    if len(scene["specks"]) > 1:
        flipped = dict(scene)
        flipped["specks"] = list(reversed(scene["specks"]))
        out.append(("speck_order_reversed", flipped))
    if len(scene["pairs"]) > 1:
        rotated = dict(scene)
        rotated["pairs"] = scene["pairs"][1:] + scene["pairs"][:1]
        out.append(("pair_record_order_rotated", rotated))
    swapped = dict(scene)
    swapped["pairs"] = [
        {"colour": pair["colour"], "bars": list(reversed(pair["bars"]))}
        for pair in scene["pairs"]
    ]
    out.append(("bar_record_order_reversed", swapped))
    return out


def render(scene: dict) -> Image.Image:
    """Rasterise the scene with flat integer rectangles and no antialiasing."""

    background = BACKGROUNDS[int(scene["background"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), background)
    draw = ImageDraw.Draw(image)
    for sx, sy in scene["specks"]:
        draw.rectangle((sx, sy, sx + SPECK - 1, sy + SPECK - 1), fill=SPECK_RGB)
    for pair in scene["pairs"]:
        fill = BAR_RGB[pair["colour"]]
        for bar in pair["bars"]:
            draw.rectangle(bar_box(bar), fill=fill)
    return image
