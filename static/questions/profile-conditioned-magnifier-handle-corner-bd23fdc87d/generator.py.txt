"""Deterministic matched magnifier scenes for visible-prior conflict."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SCALE = 4
CORNERS = ("top-left", "top-right", "bottom-left", "bottom-right")
BASE_ANGLES = {
    "top-left": 225.0,
    "top-right": 315.0,
    "bottom-left": 135.0,
    "bottom-right": 45.0,
}
BLUE = (31, 112, 205)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0x6D4147)
    corner = CORNERS[seed % 4]
    return {
        "corner": corner,
        "center_x": 192 + rng.randint(-12, 12),
        "center_y": 192 + rng.randint(-12, 12),
        "ring_radius": rng.randint(82, 98),
        "handle_length": rng.randint(70, 88),
        "handle_width": rng.randint(22, 29),
        "angle_offset_deg": round(rng.uniform(-11.0, 11.0), 4),
        "ring_tint": rng.choice(([45, 51, 61], [56, 48, 66], [39, 59, 58])),
        "background_tint": rng.choice(("#f5f1e8", "#eef3f5", "#f4eff4")),
        "draw_order": ["handle", "ring"] if rng.randrange(2) else ["ring", "handle"],
    }


def _unit(degrees: float) -> tuple[float, float]:
    angle = math.radians(degrees)
    return math.cos(angle), math.sin(angle)


def _bar_polygon(
    cx: float, cy: float, start: float, end: float, width: float, degrees: float
) -> list[tuple[float, float]]:
    ux, uy = _unit(degrees)
    px, py = -uy, ux
    half = width / 2.0
    return [
        (cx + ux * start + px * half, cy + uy * start + py * half),
        (cx + ux * start - px * half, cy + uy * start - py * half),
        (cx + ux * end - px * half, cy + uy * end - py * half),
        (cx + ux * end + px * half, cy + uy * end + py * half),
    ]


def render(scene: dict) -> Image.Image:
    cx = float(scene["center_x"]) * SCALE
    cy = float(scene["center_y"]) * SCALE
    radius = float(scene["ring_radius"]) * SCALE
    length = float(scene["handle_length"]) * SCALE
    width = float(scene["handle_width"]) * SCALE
    degrees = BASE_ANGLES[str(scene["corner"])] + float(scene["angle_offset_deg"])
    ring_tint = tuple(scene["ring_tint"])
    background = scene["background_tint"]
    order = list(scene["draw_order"])

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), background)
    draw = ImageDraw.Draw(image)

    def draw_handle() -> None:
        # The blue evidence is wholly outside the ring. A neutral collar added
        # afterward bridges the tiny construction gap.
        polygon = _bar_polygon(cx, cy, radius + 4 * SCALE, radius + length, width, degrees)
        draw.polygon(polygon, fill=BLUE)
        ux, uy = _unit(degrees)
        end_x, end_y = cx + ux * (radius + length), cy + uy * (radius + length)
        cap = width / 2.0
        draw.ellipse((end_x - cap, end_y - cap, end_x + cap, end_y + cap), fill=BLUE)

    def draw_ring() -> None:
        outer = (cx - radius, cy - radius, cx + radius, cy + radius)
        inner_radius = radius - 14 * SCALE
        inner = (
            cx - inner_radius,
            cy - inner_radius,
            cx + inner_radius,
            cy + inner_radius,
        )
        draw.ellipse(outer, fill=ring_tint)
        draw.ellipse(inner, fill=(224, 238, 243))
        # Two small translucent-looking glints make the object read as a lens.
        glint = (203, 223, 231)
        draw.arc(inner, 205, 285, fill=glint, width=3 * SCALE)

    for item in order:
        if item == "handle":
            draw_handle()
        elif item == "ring":
            draw_ring()
        else:
            raise ValueError("invalid draw_order")

    # This final collar makes the separate ring and handle visibly attached and
    # overwrites their nearest edges, so reversing their nonoverlapping draw
    # order is a genuine pixel-identical latent alias.
    collar = _bar_polygon(cx, cy, radius - 3 * SCALE, radius + 10 * SCALE, width + 6 * SCALE, degrees)
    draw.polygon(collar, fill=ring_tint)
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def _direction(scene: dict) -> float:
    return (BASE_ANGLES[str(scene["corner"])] + float(scene["angle_offset_deg"])) % 360.0


def analytic_gold(scene: dict) -> str:
    degrees = _direction(scene)
    if 0.0 < degrees < 90.0:
        return "bottom-right"
    if 90.0 < degrees < 180.0:
        return "bottom-left"
    if 180.0 < degrees < 270.0:
        return "top-left"
    return "top-right"


def margin(scene: dict) -> float:
    degrees = _direction(scene)
    return min(degrees % 90.0, 90.0 - degrees % 90.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_nonoverlapping_construction_order", twin)]
