"""Independent image-only inverse for ring-attached blue handle direction."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _evidence(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    blue = (
        (pixels[..., 2] > 130)
        & (pixels[..., 2] > pixels[..., 0] * 1.35)
        & (pixels[..., 2] > pixels[..., 1] * 1.25)
        & (pixels[..., 0] < 90)
    )
    dark = (
        (pixels[..., 0] < 95)
        & (pixels[..., 1] < 95)
        & (pixels[..., 2] < 105)
        & ~blue
    )
    return blue, dark


def decision_from_image(image: Image.Image) -> str:
    blue, dark = _evidence(image)
    by, bx = np.nonzero(blue)
    ry, rx = np.nonzero(dark)
    if bx.size < 800:
        raise ValueError("blue handle evidence is absent or incomplete")
    if rx.size < 1800:
        raise ValueError("dark ring evidence is absent or incomplete")

    # Median annulus coordinates resist the small off-center collar. The blue
    # centroid is then compared to this independently recovered ring center.
    cx, cy = float(np.median(rx)), float(np.median(ry))
    hx, hy = float(np.median(bx)), float(np.median(by))
    dx, dy = hx - cx, hy - cy
    if math.hypot(dx, dy) < 55.0:
        raise ValueError("handle displacement is too small")
    angle = math.degrees(math.atan2(dy, dx)) % 360.0
    boundary_margin = min(angle % 90.0, 90.0 - angle % 90.0)
    if boundary_margin < 15.0:
        raise ValueError("handle direction is ambiguous at pixel resolution")
    if 0.0 < angle < 90.0:
        return "bottom-right"
    if 90.0 < angle < 180.0:
        return "bottom-left"
    if 180.0 < angle < 270.0:
        return "top-left"
    return "top-right"
