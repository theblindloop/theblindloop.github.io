"""Deterministic renderer and analytic arm for Paired Pocket Parity."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
OUTER_SHORT = 60
OUTER_LONG = 72
HOLE_CROSS = 24
TOTAL_HOLE_RUN = 30
MIN_CLEARANCE = 8.0

# Consecutive seeds alternate label and cover all ordered count pairs.
COUNT_PAIRS = (
    (1, 1), (1, 2), (1, 3),
    (2, 1), (2, 2), (2, 3),
    (3, 1), (3, 2), (3, 3),
)


def _rect(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    size: tuple[int, int],
    fill: str,
) -> None:
    """Draw exactly width*height pixels, without antialias ambiguity."""

    cx, cy = center
    width, height = size
    x0 = cx - width // 2
    y0 = cy - height // 2
    draw.rectangle((x0, y0, x0 + width - 1, y0 + height - 1), fill=fill)


def _hole_centers(center: tuple[int, int], count: int, axis: str) -> list[tuple[int, int]]:
    cx, cy = center
    offsets = {1: (0,), 2: (-15, 15), 3: (-22, 0, 22)}[count]
    if axis == "vertical":
        return [(cx, cy + offset) for offset in offsets]
    if axis == "horizontal":
        return [(cx + offset, cy) for offset in offsets]
    raise ValueError(f"unsupported hole_axis={axis!r}")


def _draw_emblem(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    count: int,
    axis: str,
    ink: str,
    background: str,
) -> None:
    if count not in {1, 2, 3}:
        raise ValueError(f"unsupported pocket count {count!r}")
    if axis == "vertical":
        outer_size = (OUTER_SHORT, OUTER_LONG)
        hole_size = (HOLE_CROSS, TOTAL_HOLE_RUN // count)
    elif axis == "horizontal":
        outer_size = (OUTER_LONG, OUTER_SHORT)
        hole_size = (TOTAL_HOLE_RUN // count, HOLE_CROSS)
    else:
        raise ValueError(f"unsupported hole_axis={axis!r}")
    _rect(draw, center, outer_size, ink)
    for hole_center in _hole_centers(center, count, axis):
        _rect(draw, hole_center, hole_size, background)


def _sample_decorations(
    rng: random.Random,
    centers: list[tuple[int, int]],
) -> list[list[int]]:
    decorations: list[list[int]] = []
    attempts = 0
    while len(decorations) < 8 and attempts < 500:
        attempts += 1
        x = rng.randrange(30, CANVAS - 30)
        y = rng.randrange(30, CANVAS - 30)
        if min(math.dist((x, y), center) for center in centers) < 82:
            continue
        if any(math.dist((x, y), (old_x, old_y)) < 28 for old_x, old_y, _ in decorations):
            continue
        decorations.append([x, y, rng.randrange(2)])
    if len(decorations) != 8:
        raise RuntimeError("could not place disjoint decorations")
    return decorations


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x50C6E7)
    pocket_counts = list(COUNT_PAIRS[int(seed) % len(COUNT_PAIRS)])
    pair_axis = rng.choice(("horizontal", "vertical"))
    cross = rng.choice((192, 256, 320))
    if pair_axis == "horizontal":
        centers = [(144, cross), (368, cross)]
    else:
        centers = [(cross, 144), (cross, 368)]
    if rng.randrange(2):
        centers.reverse()
    hole_axis = rng.choice(("horizontal", "vertical"))
    ink = rng.choice(("#173f5f", "#49345f", "#205447", "#4c3d27"))
    decoration_color = rng.choice(("#d8dee2", "#d2dadd", "#e0dadd"))
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": "#ffffff",
        "ink": ink,
        "decoration_color": decoration_color,
        "pocket_counts": pocket_counts,
        "centers": [list(center) for center in centers],
        "hole_axis": hole_axis,
        "decorations": _sample_decorations(rng, centers),
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = str(scene["background"])
    ink = str(scene["ink"])
    decoration_color = str(scene["decoration_color"])
    pocket_counts = [int(value) for value in scene["pocket_counts"]]
    centers = [tuple(int(value) for value in center) for center in scene["centers"]]
    hole_axis = str(scene["hole_axis"])
    decorations = list(scene["decorations"])
    if len(pocket_counts) != 2 or len(centers) != 2:
        raise ValueError("exactly two emblems are required")

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for x, y, kind in decorations:
        cx, cy = int(x), int(y)
        if int(kind) == 0:
            draw.rectangle((cx - 4, cy - 4, cx + 3, cy + 3), outline=decoration_color, width=2)
        else:
            draw.rectangle((cx - 1, cy - 5, cx, cy + 4), fill=decoration_color)
            draw.rectangle((cx - 5, cy - 1, cx + 4, cy), fill=decoration_color)
    for center, count in zip(centers, pocket_counts, strict=True):
        _draw_emblem(draw, center, count, hole_axis, ink, background)
    return image


def analytic_gold(scene: dict) -> str:
    pocket_counts = [int(value) for value in scene["pocket_counts"]]
    return "yes" if sum(pocket_counts) % 2 == 1 else "no"


def _count_clearance(count: int) -> float:
    return {1: 18.0, 2: 14.0, 3: 9.0}[count]


def margin(scene: dict) -> float:
    pocket_counts = [int(value) for value in scene["pocket_counts"]]
    centers = [tuple(int(value) for value in center) for center in scene["centers"]]
    width = int(scene["width"])
    height = int(scene["height"])
    hole_axis = str(scene["hole_axis"])
    half_x = OUTER_SHORT // 2 if hole_axis == "vertical" else OUTER_LONG // 2
    half_y = OUTER_LONG // 2 if hole_axis == "vertical" else OUTER_SHORT // 2
    topology_clearance = min(_count_clearance(count) for count in pocket_counts)
    component_clearance = math.dist(centers[0], centers[1]) - 2.0 * max(half_x, half_y)
    border_clearance = min(
        min(cx - half_x, width - (cx + half_x), cy - half_y, height - (cy + half_y))
        for cx, cy in centers
    )
    return float(min(topology_clearance, component_clearance, border_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
