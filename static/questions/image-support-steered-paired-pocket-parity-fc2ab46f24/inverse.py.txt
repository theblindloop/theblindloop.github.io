"""Independent image-only inverse arm for Paired Pocket Parity."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the raster no longer exposes the declared robust topology."""


def _dark_mask(image: Image.Image) -> np.ndarray:
    """Separate dark emblem ink from white and pale-gray decoration pixels."""

    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    return np.max(array, axis=2) < 155


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    remaining = {tuple(int(value) for value in point) for point in np.argwhere(mask)}
    components: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        components.append(component)
    return components


def _hole_count(component: list[tuple[int, int]]) -> int:
    ys = [point[0] for point in component]
    xs = [point[1] for point in component]
    y0, y1 = min(ys) - 1, max(ys) + 2
    x0, x1 = min(xs) - 1, max(xs) + 2
    if y0 < 0 or x0 < 0:
        raise OracleAbstention("dark component touches the image border")
    local = np.zeros((y1 - y0, x1 - x0), dtype=bool)
    for y, x in component:
        local[y - y0, x - x0] = True
    holes = 0
    height, width = local.shape
    for region in _components(~local):
        touches_border = any(
            y in {0, height - 1} or x in {0, width - 1} for y, x in region
        )
        if not touches_border and len(region) >= 120:
            holes += 1
    return holes


def decision_from_image(image: Image.Image) -> str:
    mask = _dark_mask(image)
    foreground = [component for component in _components(mask) if len(component) >= 120]
    if len(foreground) != 2:
        raise OracleAbstention(f"expected two dark emblems, found {len(foreground)}")
    if any(len(component) < 2800 for component in foreground):
        raise OracleAbstention("an emblem is materially erased or fragmented")
    counts = [_hole_count(component) for component in foreground]
    if any(count not in {1, 2, 3} for count in counts):
        raise OracleAbstention(f"unsupported pocket topology {sorted(counts)}")
    return "yes" if sum(counts) % 2 == 1 else "no"
