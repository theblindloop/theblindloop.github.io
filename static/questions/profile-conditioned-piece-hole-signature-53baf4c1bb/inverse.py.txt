"""Pixel-only inverse arm for the piece/hole signature world.

Input is one rendered RGB image and nothing else.  The arm splits the ink into
its four colour families by chroma alone, counts each family's 8-connected ink
blobs, counts the 4-connected white pockets that the image border cannot reach
inside each family, and returns the colour whose (blobs, pockets) pair repeats
the grey/black family's pair.  It imports no renderer, prompt, generator,
verifier, or gold code and never sees a scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

INK_LEVEL = 200  # a pixel is ink when its darkest channel is below this
NEUTRAL_SPREAD = 45  # channel spread below which ink counts as grey/black
MIN_BLOB_AREA = 250
MIN_POCKET_AREA = 14
CHROMATIC = ("red", "green", "blue")


def _flood(seed: np.ndarray, allowed: np.ndarray) -> np.ndarray:
    """4-connected reachability inside `allowed`, seeded by `seed`."""
    current = seed & allowed
    height, width = allowed.shape
    while True:
        before = int(current.sum())
        for col in range(1, width):
            current[:, col] |= current[:, col - 1] & allowed[:, col]
        for col in range(width - 2, -1, -1):
            current[:, col] |= current[:, col + 1] & allowed[:, col]
        for row in range(1, height):
            current[row] |= current[row - 1] & allowed[row]
        for row in range(height - 2, -1, -1):
            current[row] |= current[row + 1] & allowed[row]
        if int(current.sum()) == before:
            return current


def _components(mask: np.ndarray, min_area: int) -> list[np.ndarray]:
    remaining = mask.copy()
    blobs: list[np.ndarray] = []
    while True:
        hits = np.argwhere(remaining)
        if hits.size == 0:
            return blobs
        seed = np.zeros_like(remaining)
        seed[hits[0][0], hits[0][1]] = True
        blob = _flood(seed, remaining)
        if int(blob.sum()) >= min_area:
            blobs.append(blob)
        remaining &= ~blob


def _color_masks(image: Image.Image) -> dict[str, np.ndarray]:
    array = np.asarray(image.convert("RGB")).astype(np.int16)
    darkest = array.min(axis=2)
    brightest = array.max(axis=2)
    ink = darkest < INK_LEVEL
    neutral = ink & ((brightest - darkest) < NEUTRAL_SPREAD)
    chromatic = ink & ~neutral
    channel = np.argmax(array, axis=2)
    masks = {"neutral": neutral}
    for index, name in enumerate(CHROMATIC):
        masks[name] = chromatic & (channel == index)
    return masks


def _signature(mask: np.ndarray) -> tuple[int, int] | None:
    """(blob count, enclosed white pocket count) for one colour family."""
    rows, cols = np.nonzero(mask)
    if rows.size == 0:
        return None
    window = mask[rows.min() : rows.max() + 1, cols.min() : cols.max() + 1]
    window = np.pad(window, 2, constant_values=False)
    blobs = _components(window, MIN_BLOB_AREA)
    if not blobs:
        return None
    solid = np.zeros_like(window)
    for blob in blobs:
        solid |= blob
    outside = np.zeros_like(window)
    outside[0, :] = True
    outside[-1, :] = True
    outside[:, 0] = True
    outside[:, -1] = True
    reachable = _flood(outside, ~solid)
    pockets = _components((~solid) & (~reachable), MIN_POCKET_AREA)
    return len(blobs), len(pockets)


def decision_from_image(image: Image.Image) -> str:
    """Return the matching colour name, or 'none' when the match is not unique."""
    masks = _color_masks(image)
    reference = _signature(masks["neutral"])
    if reference is None:
        return "none"
    hits = [name for name in CHROMATIC if _signature(masks[name]) == reference]
    if len(hits) == 1:
        return hits[0]
    return "none"
