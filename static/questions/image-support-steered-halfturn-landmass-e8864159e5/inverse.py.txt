"""Independent image-only inverse arm for the half-turn landmass world.

This module deliberately duplicates the public visual specification instead of
importing any rendering or gold code. It discovers the colored component, infers
its 11x11 lattice from the raster, validates whole-cell occupancy and constant
area, then compares that recovered occupancy field with its half-turn.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

BOARD_N = 11
EXPECTED_OCCUPIED = 73
MIN_CELL = 28
MAX_CELL = 38


class OracleAbstention(ValueError):
    """The raster is outside the unambiguous public rendering contract."""


@dataclass(frozen=True)
class Measurement:
    decision: str
    mismatch_cells: int
    occupied_cells: int
    cell_size: int
    origin: tuple[int, int]


def _colored_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    channel_range = values.max(axis=2) - values.min(axis=2)
    # Every allowed landmass color is strongly chromatic; the warm-white background
    # and gray board lines are nearly neutral. This rule is palette-name agnostic.
    mask = channel_range >= 45
    if int(mask.sum()) < EXPECTED_OCCUPIED * MIN_CELL * MIN_CELL:
        raise OracleAbstention("too little colored area")
    return mask


def _connected(occupancy: np.ndarray) -> bool:
    points = np.argwhere(occupancy)
    if not len(points):
        return False
    start = tuple(int(value) for value in points[0])
    seen = {start}
    stack = [start]
    while stack:
        row, col = stack.pop()
        for neighbor in ((row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)):
            nr, nc = neighbor
            if 0 <= nr < BOARD_N and 0 <= nc < BOARD_N and occupancy[nr, nc] and neighbor not in seen:
                seen.add(neighbor)
                stack.append(neighbor)
    return len(seen) == int(occupancy.sum())


def measure_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 256:
        raise OracleAbstention("unsupported raster dimensions")
    mask = _colored_mask(array)
    ys, xs = np.nonzero(mask)
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if width != height or width % BOARD_N or not (MIN_CELL <= width // BOARD_N <= MAX_CELL):
        raise OracleAbstention("colored bounding box does not define the public 11x11 board")
    cell = width // BOARD_N

    occupancy = np.zeros((BOARD_N, BOARD_N), dtype=bool)
    for row in range(BOARD_N):
        for col in range(BOARD_N):
            tile = mask[y0 + row * cell : y0 + (row + 1) * cell, x0 + col * cell : x0 + (col + 1) * cell]
            fraction = float(tile.mean())
            if fraction >= 0.995:
                occupancy[row, col] = True
            elif fraction > 0.005:
                raise OracleAbstention("partial colored cell")

    count = int(occupancy.sum())
    if count != EXPECTED_OCCUPIED:
        raise OracleAbstention(f"expected {EXPECTED_OCCUPIED} whole colored cells, found {count}")
    expected_pixels = EXPECTED_OCCUPIED * cell * cell
    if int(mask.sum()) != expected_pixels:
        raise OracleAbstention("colored pixel area is inconsistent with whole cells")
    if not _connected(occupancy):
        raise OracleAbstention("colored cells do not form one connected landmass")
    mismatch = int(np.count_nonzero(occupancy != occupancy[::-1, ::-1]))
    if mismatch == 2:
        raise OracleAbstention("one-orbit boundary case")
    return Measurement(
        decision="yes" if mismatch == 0 else "no",
        mismatch_cells=mismatch,
        occupied_cells=count,
        cell_size=cell,
        origin=(x0, y0),
    )


def decision_from_image(image: Image.Image) -> str:
    return measure_image(image).decision
