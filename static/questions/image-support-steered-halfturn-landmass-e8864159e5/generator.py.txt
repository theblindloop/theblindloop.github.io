"""Deterministic renderer and analytic arm for the half-turn landmass world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
GRID_SIZE = 11
FILLED_CELLS = 73
PALETTE = (
    (22, 137, 145),
    (104, 75, 173),
    (214, 105, 38),
    (38, 94, 177),
)
BACKGROUND = (248, 246, 239)
GRID_COLOR = (218, 215, 207)


def _rotate_cell(cell: tuple[int, int], n: int) -> tuple[int, int]:
    return (n - 1 - cell[0], n - 1 - cell[1])


def _orbit_pairs(cells: Iterable[tuple[int, int]], n: int) -> list[tuple[tuple[int, int], tuple[int, int]]]:
    remaining = set(cells)
    pairs: list[tuple[tuple[int, int], tuple[int, int]]] = []
    while remaining:
        first = min(remaining)
        second = _rotate_cell(first, n)
        if first == second:
            raise ValueError("unexpected singleton orbit")
        remaining.remove(first)
        remaining.remove(second)
        pairs.append((first, second))
    return pairs


def _base_symmetric_mask(rng: random.Random) -> tuple[set[tuple[int, int]], list, list]:
    # A 7x7 central slab supplies substantial connected interior support.
    occupied = {(row, col) for row in range(2, 9) for col in range(2, 9)}
    ring_cells = {
        (row, col)
        for row in range(GRID_SIZE)
        for col in range(GRID_SIZE)
        if ((row in (1, 9) and 2 <= col <= 8) or (col in (1, 9) and 2 <= row <= 8))
    }
    ring_pairs = _orbit_pairs(ring_cells, GRID_SIZE)
    required_cells = {(1, 5), (9, 5), (5, 1), (5, 9)}
    required_pairs = [pair for pair in ring_pairs if set(pair) & required_cells]
    optional_pairs = [pair for pair in ring_pairs if pair not in required_pairs]
    rng.shuffle(optional_pairs)
    chosen_optional = optional_pairs[:8]
    unchosen_optional = optional_pairs[8:]
    for pair in required_pairs + chosen_optional:
        occupied.update(pair)

    # Opposite tips fix a wide, square bounding box; the required ring cells bridge them.
    tip_pairs = [((0, 5), (10, 5)), ((5, 0), (5, 10))]
    for pair in tip_pairs:
        occupied.update(pair)
    if len(occupied) != FILLED_CELLS:
        raise AssertionError(f"construction produced {len(occupied)} cells")
    return occupied, chosen_optional, unchosen_optional


def _halfturn_mismatch(occupied_cells: Iterable[Iterable[int]], grid_size: int) -> int:
    occupied = {tuple(int(value) for value in cell) for cell in occupied_cells}
    rotated = {_rotate_cell(cell, grid_size) for cell in occupied}
    return len(occupied.symmetric_difference(rotated))


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Even seeds are symmetric; odd seeds are not."""
    rng = random.Random(int(seed) ^ 0x5A17_180)
    occupied, selected_pairs, unselected_pairs = _base_symmetric_mask(rng)
    symmetric = int(seed) % 2 == 0
    asymmetry_level = 0 if symmetric else 1 + ((abs(int(seed)) // 2) % 3)
    if not symmetric:
        rng.shuffle(selected_pairs)
        rng.shuffle(unselected_pairs)
        for index in range(asymmetry_level):
            remove_pair = selected_pairs[index]
            add_pair = unselected_pairs[index]
            removed = remove_pair[rng.randrange(2)]
            added = add_pair[rng.randrange(2)]
            occupied.remove(removed)
            occupied.add(added)

    cells = [[row, col] for row, col in occupied]
    rng.shuffle(cells)  # draw order is a genuine latent alias
    cell_size = rng.choice((30, 32, 34, 36))
    extent = GRID_SIZE * cell_size
    dx = rng.choice((-10, -5, 0, 5, 10))
    dy = rng.choice((-10, -5, 0, 5, 10))
    origin_x = (CANVAS_SIZE - extent) // 2 + dx
    origin_y = (CANVAS_SIZE - extent) // 2 + dy
    return {
        "grid_size": GRID_SIZE,
        "occupied_cells": cells,
        "cell_size": cell_size,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "fill_rgb": list(rng.choice(PALETTE)),
        "background_rgb": list(BACKGROUND),
        "grid_rgb": list(GRID_COLOR),
    }


def render(scene: dict) -> Image.Image:
    n = int(scene["grid_size"])
    cell = int(scene["cell_size"])
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), tuple(scene["background_rgb"]))
    draw = ImageDraw.Draw(image)
    extent = n * cell
    grid_color = tuple(scene["grid_rgb"])
    for index in range(n + 1):
        x = ox + index * cell
        y = oy + index * cell
        draw.line((x, oy, x, oy + extent), fill=grid_color, width=1)
        draw.line((ox, y, ox + extent, y), fill=grid_color, width=1)
    fill = tuple(scene["fill_rgb"])
    for row, col in scene["occupied_cells"]:
        x0, y0 = ox + int(col) * cell, oy + int(row) * cell
        draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=fill)
    return image


def analytic_gold(scene: dict) -> str:
    mismatch = _halfturn_mismatch(scene["occupied_cells"], int(scene["grid_size"]))
    return "yes" if mismatch == 0 else "no"


def margin(scene: dict) -> float:
    """Runner-up gap in cells around the midpoint between realizable classes."""
    mismatch = _halfturn_mismatch(scene["occupied_cells"], int(scene["grid_size"]))
    return 2.0 if mismatch == 0 else float(mismatch - 2)


def is_quarantined(scene: dict) -> bool:
    mismatch = _halfturn_mismatch(scene["occupied_cells"], int(scene["grid_size"]))
    return mismatch not in (0, 4, 8, 12) or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["occupied_cells"] = list(reversed([list(cell) for cell in scene["occupied_cells"]]))
    return [("reverse_unordered_cell_draw_order", twin)]
