"""Independent PNG-only inverse arm for the iterated cell-parity relay.

Recovers the nine cell ON/OFF states and the number of transition rounds S
directly from the final raster (structure-mask column bands for the cells and a
teal connected-component count for the dot key), then iterates the neighbour
relay rule S times and returns whether the final dark-cell count is odd.  Uses
no renderer geometry, scene parameters, prompt, or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N_CELLS = 9
MAX_STEP = 3


def _structure_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    grays = chroma <= 12
    notwhite = (255 - arr.mean(axis=2)) >= 10
    return grays & notwhite


def _teal_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[:, :, 0].astype(np.int16)
    g = arr[:, :, 1].astype(np.int16)
    b = arr[:, :, 2].astype(np.int16)
    return (g > 90) & (r < 80) & (b > 90) & ((b - r) > 25)


def _components(mask: np.ndarray, minsize: int) -> int:
    pixels = set(zip(*np.nonzero(mask)))
    count = 0
    while pixels:
        start = pixels.pop()
        stack = [start]
        size = 1
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    size += 1
                    stack.append((ny, nx))
        if size >= minsize:
            count += 1
    return count


def _onestep(cells: list[int]) -> list[int]:
    n = len(cells)
    out = list(cells)
    for i in range(n):
        if cells[i] == 1:
            out[i] = 1
        else:
            left = cells[i - 1] if i > 0 else 0
            right = cells[i + 1] if i < n - 1 else 0
            out[i] = 1 if (left + right) == 1 else 0
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    struct = _structure_mask(arr)
    walls = np.where(struct.any(axis=1))[0]
    if len(walls) == 0:
        raise ValueError("no cell structure found in image")
    row_center = int((walls.min() + walls.max()) // 2)
    band = struct[row_center - 40:row_center + 41, :]
    cols = band.any(axis=0)
    trans = np.diff(cols.astype(np.int8))
    starts = np.flatnonzero(trans == 1)
    ends = np.flatnonzero(trans == -1) + 1
    if cols[0]:
        starts = np.concatenate(([0], starts))
    if cols[-1]:
        ends = np.concatenate((ends, [len(cols)]))
    pairs = [(int(s), int(e)) for s, e in zip(starts, ends) if int(e) - int(s) >= 20]
    pairs.sort()
    if len(pairs) != N_CELLS:
        raise ValueError(f"expected {N_CELLS} cells, found {len(pairs)}")
    bits: list[int] = []
    for start, end in pairs:
        x = (start + end) // 2
        px = arr[row_center, x].astype(np.int16)
        bits.append(1 if float(px.mean()) < 120 else 0)
    steps = _components(_teal_mask(arr), 20)
    if steps < 1 or steps > MAX_STEP:
        raise ValueError(f"invalid round count {steps}")
    current = bits
    for _ in range(steps):
        current = _onestep(current)
    return "yes" if sum(current) % 2 == 1 else "no"
