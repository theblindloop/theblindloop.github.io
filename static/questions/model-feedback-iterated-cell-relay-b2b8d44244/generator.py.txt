"""Deterministic renderer and analytic arm for the iterated cell-parity relay.

A single horizontal line of nine square cells is drawn.  Each cell is filled
either dark (an ON cell) or pale (an OFF cell).  A small teal dot-key below the
row shows the number of transition rounds S to apply (one, two, or three dots).

The counterfactual is a repeated whole-line transition rule that depends on the
*current* state: a dark cell stays dark; a light cell turns dark only when
exactly one of its two neighbours (an end cell has just one neighbour) is dark,
otherwise it stays light.  This rule is applied S times, converting each round's
row into the next round's row.  The question asks whether the final number of
dark cells is odd.  The after-state is never drawn; it must be recovered by
iterating the visible initial row through S rounds of the rule.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 720
N_CELLS = 9
CELL = 56
GAP = 14
ROW_Y = 240
DOT_Y = 430
DOT_R = 16
DOT_SP = 44
BACKGROUND = (250, 250, 250)
OUTLINE = (128, 128, 128)
ON_FILL = (35, 35, 35)
OFF_FILL = (232, 232, 232)
DOT_COLOR = (0, 150, 150)
MAX_STEP = 3


def _row_left() -> int:
    total = N_CELLS * CELL + (N_CELLS - 1) * GAP
    return (CANVAS - total) // 2


def step(cells: list[int]) -> list[int]:
    """One round of the neighbour-relay rule on a row of bits."""

    n = len(cells)
    out = list(cells)
    for i in range(n):
        if cells[i] == 1:
            out[i] = 1
        else:
            left = cells[i - 1] if i > 0 else 0
            right = cells[i + 1] if i < n - 1 else 0
            out[i] = 1 if (left + right) == 1 else 0
    return out


def _simulate(cells: list[int], steps: int) -> list[int]:
    current = list(cells)
    for _ in range(steps):
        current = step(current)
    return current


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cells = [rng.randint(0, 1) for _ in range(N_CELLS)]
    steps = rng.randint(1, MAX_STEP)
    return {
        "canvas_size": CANVAS,
        "cells": cells,
        "steps": steps,
        "cell_px": CELL,
        "gap_px": GAP,
        "row_y": ROW_Y,
        "dot_y": DOT_Y,
        "dot_r": DOT_R,
        "dot_spacing": DOT_SP,
    }


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    cells = [int(value) for value in scene["cells"]]
    steps = int(scene["steps"])
    cell = int(scene["cell_px"])
    gap = int(scene["gap_px"])
    row_y = int(scene["row_y"])
    dot_y = int(scene["dot_y"])
    dot_r = int(scene["dot_r"])
    dot_spacing = int(scene["dot_spacing"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    left = (size - (N_CELLS * cell + (N_CELLS - 1) * gap)) // 2
    for index, value in enumerate(cells):
        x = left + index * (cell + gap)
        draw.rectangle(
            (x, row_y, x + cell, row_y + cell),
            fill=ON_FILL if value == 1 else OFF_FILL,
            outline=OUTLINE,
            width=3,
        )
    total_dots = steps * dot_spacing
    x0 = (size - total_dots) // 2 + dot_spacing // 2
    for k in range(steps):
        cx = x0 + k * dot_spacing
        draw.ellipse(
            (cx - dot_r, dot_y - dot_r, cx + dot_r, dot_y + dot_r),
            fill=DOT_COLOR,
        )
    return image


def analytic_gold(scene: dict) -> str:
    cells = [int(value) for value in scene["cells"]]
    steps = int(scene["steps"])
    final = _simulate(cells, steps)
    return "yes" if sum(final) % 2 == 1 else "no"


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    cells = scene["cells"]
    steps = int(scene["steps"])
    if len(cells) != N_CELLS or steps < 1 or steps > MAX_STEP:
        return True
    if any(value not in (0, 1) for value in cells):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
