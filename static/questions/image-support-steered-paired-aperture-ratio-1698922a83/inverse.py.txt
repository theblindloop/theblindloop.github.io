"""Pixel-only inverse arm for the two colored U apertures.

This module intentionally contains its own palette and structural measurement.
It accepts only a raster image and never imports scene or generation code.
"""

from __future__ import annotations

import numpy as np

CORAL = np.array((225, 84, 84), dtype=np.uint8)
BLUE = np.array((48, 116, 184), dtype=np.uint8)


def _aperture_ratio(array: np.ndarray, color: np.ndarray) -> float:
    mask = np.all(array == color, axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise ValueError("colored gauge is absent or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    local = mask[y0 : y1 + 1, x0 : x1 + 1]
    row_counts = local.sum(axis=1)
    outer_width = local.shape[1]
    outer_height = local.shape[0]
    wall_count = int(row_counts[0])
    if wall_count < 6 or wall_count % 2:
        raise ValueError("gauge mouth walls are malformed")
    thickness = wall_count // 2
    inner_width = outer_width - 2 * thickness
    inner_depth = outer_height - thickness
    if inner_width < 8 or inner_depth < 10:
        raise ValueError("gauge aperture is too small")
    expected_rows = np.concatenate(
        (
            np.full(inner_depth, 2 * thickness, dtype=np.int64),
            np.full(thickness, outer_width, dtype=np.int64),
        )
    )
    if row_counts.shape != expected_rows.shape or not np.array_equal(row_counts, expected_rows):
        raise ValueError("gauge is not a complete crisp U")
    if not np.all(local[:inner_depth, :thickness]):
        raise ValueError("left wall is incomplete")
    if not np.all(local[:inner_depth, -thickness:]):
        raise ValueError("right wall is incomplete")
    if np.any(local[:inner_depth, thickness:-thickness]):
        raise ValueError("clear aperture is obstructed")
    return inner_width / inner_depth


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    coral_ratio = _aperture_ratio(array, CORAL)
    blue_ratio = _aperture_ratio(array, BLUE)
    if abs(coral_ratio - blue_ratio) < 1e-12:
        raise ValueError("equal aperture ratios have no unique answer")
    return "coral" if coral_ratio > blue_ratio else "blue"
