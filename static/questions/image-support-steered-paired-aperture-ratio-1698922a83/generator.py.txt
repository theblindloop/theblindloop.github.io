"""Deterministic latent sampler and crisp Pillow renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CORAL = (225, 84, 84)
BLUE = (48, 116, 184)
PALE = (207, 212, 218)
WHITE = (255, 255, 255)
QUARANTINE_MARGIN = 0.12


def _ratio(width: int, depth: int) -> float:
    return width / depth


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair, including deterministic near-boundary cases."""
    rng = random.Random(int(seed))
    wanted = "coral" if seed % 2 == 0 else "blue"
    near = seed % 7 == 0
    chosen = None
    for _ in range(50_000):
        a = (rng.randint(12, 22), rng.randint(16, 24))
        b = (rng.randint(12, 22), rng.randint(16, 24))
        gap = abs(_ratio(*a) - _ratio(*b))
        if (0.045 <= gap < 0.105) if near else (0.18 <= gap <= 0.58):
            chosen = (a, b)
            break
    if chosen is None:
        raise RuntimeError("could not sample separated aperture ratios")
    larger, smaller = sorted(chosen, key=lambda pair: _ratio(*pair), reverse=True)
    coral, blue = (larger, smaller) if wanted == "coral" else (smaller, larger)

    cell_col = rng.randint(1, 5)
    cell_row = rng.randint(1, 5)
    diagonal_sign = rng.choice((-1, 1))
    if cell_row + diagonal_sign < 1 or cell_row + diagonal_sign > 6:
        diagonal_sign *= -1
    color_order = rng.randint(0, 1)

    occupied = {
        (cell_col, cell_row),
        (cell_col + 1, cell_row + diagonal_sign),
    }
    distractors: list[list[int]] = []
    candidates = [(c, r) for r in range(1, 7) for c in range(1, 7) if (c, r) not in occupied]
    rng.shuffle(candidates)
    for c, r in candidates[:6]:
        distractors.append([64 * c + 32, 64 * r + 32, rng.randint(0, 1)])
    return {
        "width": 512,
        "height": 512,
        "cell_col": cell_col,
        "cell_row": cell_row,
        "diagonal_sign": diagonal_sign,
        "color_order": color_order,
        "thickness": 4,
        "coral_inner_width": coral[0],
        "coral_inner_depth": coral[1],
        "blue_inner_width": blue[0],
        "blue_inner_depth": blue[1],
        "distractors": distractors,
    }


def _draw_u(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    inner_width: int,
    inner_depth: int,
    thickness: int,
    color: tuple[int, int, int],
) -> None:
    outer_width = inner_width + 2 * thickness
    outer_height = inner_depth + thickness
    x0 = center[0] - outer_width // 2
    y0 = center[1] - outer_height // 2
    x1 = x0 + outer_width - 1
    y1 = y0 + outer_height - 1
    draw.rectangle((x0, y0, x0 + thickness - 1, y1), fill=color)
    draw.rectangle((x1 - thickness + 1, y0, x1, y1), fill=color)
    draw.rectangle((x0, y0 + inner_depth, x1, y1), fill=color)


def _draw_distractors(draw: ImageDraw.ImageDraw, distractors: Iterable[list[int]]) -> None:
    for x, y, kind in distractors:
        if kind == 0:
            draw.rectangle((x - 5, y - 5, x + 5, y + 5), outline=PALE, width=2)
        else:
            draw.ellipse((x - 5, y - 5, x + 5, y + 5), outline=PALE, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), WHITE)
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, scene["distractors"])
    p0 = (64 * scene["cell_col"] + 32, 64 * scene["cell_row"] + 32)
    p1 = (p0[0] + 64, p0[1] + 64 * scene["diagonal_sign"])
    positions = (p0, p1) if scene["color_order"] == 0 else (p1, p0)
    _draw_u(
        draw,
        positions[0],
        scene["coral_inner_width"],
        scene["coral_inner_depth"],
        scene["thickness"],
        CORAL,
    )
    _draw_u(
        draw,
        positions[1],
        scene["blue_inner_width"],
        scene["blue_inner_depth"],
        scene["thickness"],
        BLUE,
    )
    return image


def analytic_gold(scene: dict) -> str:
    coral_ratio = scene["coral_inner_width"] / scene["coral_inner_depth"]
    blue_ratio = scene["blue_inner_width"] / scene["blue_inner_depth"]
    return "coral" if coral_ratio > blue_ratio else "blue"


def margin(scene: dict) -> float:
    coral_ratio = scene["coral_inner_width"] / scene["coral_inner_depth"]
    blue_ratio = scene["blue_inner_width"] / scene["blue_inner_depth"]
    return abs(coral_ratio - blue_ratio)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
