"""Deterministic scene sampler and raster renderer for the QR-like world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CORNERS = ("top-left", "top-right", "bottom-left", "bottom-right")

_PALETTES = (
    ((35, 45, 58), (20, 112, 184), (224, 229, 235), (250, 251, 248)),
    ((43, 42, 48), (176, 55, 118), (232, 226, 218), (252, 250, 246)),
    ((32, 48, 45), (16, 137, 104), (220, 230, 226), (250, 252, 249)),
    ((45, 40, 52), (126, 72, 190), (230, 225, 235), (252, 250, 253)),
)


def _style_rng(seed: int) -> random.Random:
    # Four consecutive seeds are a matched answer block with identical nuisance style.
    return random.Random(0x51A9_0000 + seed // 4)


def sample_scene(seed: int) -> dict:
    rng = _style_rng(seed)
    missing = CORNERS[seed % 4]
    unit = rng.choice((12, 13, 14, 15))
    tile_extent = 25 * unit
    offset_x = (WIDTH - tile_extent) // 2 + rng.randint(-14, 14)
    offset_y = (HEIGHT - tile_extent) // 2 + rng.randint(-14, 14)
    ink, accent, background, tile = rng.choice(_PALETTES)

    # Reserve every possible finder footprint so module texture is identical within
    # a matched block and never acts as a proxy for the omitted corner.
    anchors = ((2, 2), (16, 2), (2, 16), (16, 16))
    allowed = []
    for gy in range(2, 23):
        for gx in range(2, 23):
            if any(ax - 1 <= gx <= ax + 7 and ay - 1 <= gy <= ay + 7 for ax, ay in anchors):
                continue
            allowed.append((gx, gy))
    rng.shuffle(allowed)
    modules = sorted(allowed[:34])
    return {
        "missing_corner": missing,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "unit": unit,
        "ink": list(ink),
        "accent": list(accent),
        "background": list(background),
        "tile": list(tile),
        "modules": [list(item) for item in modules],
    }


def _rect(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], fill: tuple[int, int, int]) -> None:
    draw.rectangle(box, fill=fill)


def render(scene: dict) -> Image.Image:
    scale = 2
    width, height = WIDTH * scale, HEIGHT * scale
    background = tuple(scene["background"])
    tile_color = tuple(scene["tile"])
    ink = tuple(scene["ink"])
    accent = tuple(scene["accent"])
    unit = int(scene["unit"]) * scale
    ox = int(scene["offset_x"]) * scale
    oy = int(scene["offset_y"]) * scale
    missing = str(scene["missing_corner"])
    modules = scene["modules"]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    extent = 25 * unit
    shadow = 5 * scale
    draw.rounded_rectangle(
        (ox + shadow, oy + shadow, ox + extent + shadow, oy + extent + shadow),
        radius=2 * unit,
        fill=(175, 181, 188),
    )
    draw.rounded_rectangle((ox, oy, ox + extent, oy + extent), radius=2 * unit, fill=tile_color)

    # Quiet-zone frame hints at a printed tile without changing the four corners.
    draw.rectangle((ox + unit, oy + unit, ox + extent - unit, oy + extent - unit), outline=ink, width=max(2, unit // 5))

    positions = {
        "top-left": (2, 2),
        "top-right": (16, 2),
        "bottom-left": (2, 16),
        "bottom-right": (16, 16),
    }
    for name, (gx, gy) in positions.items():
        if name == missing:
            continue
        x0, y0 = ox + gx * unit, oy + gy * unit
        _rect(draw, (x0, y0, x0 + 7 * unit - 1, y0 + 7 * unit - 1), accent)
        _rect(draw, (x0 + unit, y0 + unit, x0 + 6 * unit - 1, y0 + 6 * unit - 1), tile_color)
        _rect(draw, (x0 + 2 * unit, y0 + 2 * unit, x0 + 5 * unit - 1, y0 + 5 * unit - 1), accent)

    for pair in modules:
        gx, gy = int(pair[0]), int(pair[1])
        x0, y0 = ox + gx * unit, oy + gy * unit
        _rect(draw, (x0, y0, x0 + unit - 1, y0 + unit - 1), ink)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["missing_corner"])


def margin(scene: dict) -> float:
    # Ideal colored foreground area of one 7x7 / 5x5 / 3x3 finder target.
    unit = float(scene["unit"])
    return 33.0 * unit * unit


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2400.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["modules"] = list(reversed(twin["modules"]))
    return [("reverse_module_serialization", twin)]
