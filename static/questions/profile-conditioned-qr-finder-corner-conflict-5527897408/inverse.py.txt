"""Independent inverse arm. It receives only final RGB pixels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_ALL_CORNERS = {"top-left", "top-right", "bottom-left", "bottom-right"}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for 4-connected true components."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    starts = np.argwhere(mask)
    for sy, sx in starts:
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((area, x0, y0, x1, y1))
    return output


def _finder_components(image: Image.Image) -> list[tuple[float, float, float]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    saturation_mask = (chroma >= 48) & (arr.min(axis=2) < 190)
    comps = _components(saturation_mask)
    plausible = []
    for area, x0, y0, x1, y1 in comps:
        bw, bh = x1 - x0 + 1, y1 - y0 + 1
        if area < 900 or bw < 70 or bh < 70:
            continue
        if not (0.88 <= bw / bh <= 1.12):
            continue
        # A finder has a saturated central 3/7 square as well as its outer ring.
        cx0, cx1 = x0 + round(2 * bw / 7), x0 + round(5 * bw / 7)
        cy0, cy1 = y0 + round(2 * bh / 7), y0 + round(5 * bh / 7)
        core = saturation_mask[cy0:cy1, cx0:cx1]
        if core.size == 0 or float(core.mean()) < 0.72:
            continue
        plausible.append(((x0 + x1) / 2.0, (y0 + y1) / 2.0, float(area)))
    plausible.sort(key=lambda item: item[2], reverse=True)
    return plausible[:3]


def decision_from_image(image: Image.Image) -> str:
    finders = _finder_components(image)
    if len(finders) != 3:
        raise ValueError("abstain: expected exactly three complete finder targets")
    areas = [item[2] for item in finders]
    if min(areas) < 0.80 * max(areas):
        raise ValueError("abstain: inconsistent finder-target evidence")
    xs = [item[0] for item in finders]
    ys = [item[1] for item in finders]
    if max(xs) - min(xs) < 120 or max(ys) - min(ys) < 120:
        raise ValueError("abstain: finder layout has no stable two-axis separation")
    xmid = (min(xs) + max(xs)) / 2.0
    ymid = (min(ys) + max(ys)) / 2.0
    occupied = set()
    for x, y, _ in finders:
        vertical = "top" if y < ymid else "bottom"
        horizontal = "left" if x < xmid else "right"
        occupied.add(f"{vertical}-{horizontal}")
    missing = _ALL_CORNERS - occupied
    if len(occupied) != 3 or len(missing) != 1:
        raise ValueError("abstain: corner incidence is ambiguous")
    return next(iter(missing))
