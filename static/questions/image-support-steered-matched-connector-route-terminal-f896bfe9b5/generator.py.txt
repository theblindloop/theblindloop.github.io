"""Deterministic renderer and analytic answer for the matched-tab route world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CANONICAL_WIDTH = 320.0
CANONICAL_HEIGHT = 300.0
ROUTE_RGB = (37, 53, 70)
GREEN_RGB = (39, 169, 104)
AMBER_RGB = (226, 151, 34)
VIOLET_RGB = (132, 86, 190)
CONNECTOR_RGB = {
    "cyan": (39, 157, 207),
    "magenta": (215, 76, 147),
}
TERMINAL_RGB = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
MARKER_RADIUS = 13.0
ROUTE_WIDTH = 9
MARGIN_THRESHOLD = 34.0


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _polyline_length(points: list[tuple[float, float]]) -> float:
    return sum(_distance(a, b) for a, b in zip(points, points[1:]))


def _transform_point(
    point: list[float] | tuple[float, float], transform: dict[str, Any]
) -> tuple[float, float]:
    x, y = float(point[0]), float(point[1])
    if bool(transform["mirror_x"]):
        x = CANONICAL_WIDTH - x
    if bool(transform["flip_y"]):
        y = CANONICAL_HEIGHT - y
    return (x + float(transform["dx"]), y + float(transform["dy"]))


def _piece_points(scene: dict[str, Any], piece: dict[str, Any]) -> list[tuple[float, float]]:
    transform = scene["layout_transform"]
    return [_transform_point(point, transform) for point in piece["points"]]


def _route_totals(scene: dict[str, Any]) -> tuple[dict[str, float], dict[str, str]]:
    """Assemble pieces by their visible connector labels and measure their paths."""

    pieces = scene["route_pieces"]
    connector_colors = scene["connector_colors"]
    terminal_assignment = scene["terminal_assignment"]
    # Reading the transform here makes translation/reflection part of the
    # rendered, label-bearing scene rather than a hidden gold-only nuisance.
    transform = scene["layout_transform"]
    if not isinstance(transform, dict):
        raise ValueError("layout_transform must be an object")
    totals: dict[str, float] = {}
    terminals: dict[str, str] = {}
    for piece in pieces:
        piece_id = piece["piece_id"]
        connector = connector_colors[piece_id]
        points = [_transform_point(point, transform) for point in piece["points"]]
        totals[connector] = totals.get(connector, 0.0) + _polyline_length(points)
        if piece["role"] == "terminal":
            terminals[connector] = terminal_assignment[connector]
    if set(totals) != {"cyan", "magenta"} or set(terminals) != set(totals):
        raise ValueError("scene must contain two complete matched connector routes")
    return totals, terminals


def _route_points(
    base_y: float, amplitude: float, sign: float, rng: random.Random
) -> tuple[list[list[float]], list[list[float]]]:
    """Create two monotone-x pieces with a gap between their matching tabs."""

    # The repeated x coordinates and alternating vertical offsets make the
    # long route visibly longer by accumulated path length, while both routes
    # occupy the same regional bay and retain clean raster centerlines.
    jitter = [rng.uniform(-3.0, 3.0) for _ in range(6)]
    start = [
        [0.0, base_y],
        [25.0, base_y + sign * amplitude * 0.55 + jitter[0]],
        [50.0, base_y - sign * amplitude * 0.95 + jitter[1]],
        [75.0, base_y + sign * amplitude * 0.40 + jitter[2]],
    ]
    terminal = [
        [245.0, base_y + sign * amplitude * 0.35 + jitter[3]],
        [270.0, base_y - sign * amplitude * 0.90 + jitter[4]],
        [295.0, base_y + sign * amplitude * 0.62 + jitter[5]],
        [320.0, base_y - sign * amplitude * 0.18],
    ]
    return start, terminal


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene; every answer-bearing field is rendered."""

    rng = random.Random(int(seed))
    long_route = rng.randrange(2)
    near_boundary = int(seed) % 13 == 0
    pieces: list[dict[str, Any]] = []
    connector_colors: dict[str, str] = {}
    connector_names = ["cyan", "magenta"]
    for route_index, connector in enumerate(connector_names):
        if near_boundary:
            amplitude = 18.0 if route_index == long_route else 16.0
        else:
            amplitude = 38.0 if route_index == long_route else 7.0
        sign = -1.0 if rng.random() < 0.5 else 1.0
        base_y = 76.0 if route_index == 0 else 224.0
        start_points, terminal_points = _route_points(base_y, amplitude, sign, rng)
        start_id = f"r{route_index}_start"
        terminal_id = f"r{route_index}_terminal"
        pieces.extend(
            [
                {"piece_id": start_id, "role": "start", "points": start_points},
                {"piece_id": terminal_id, "role": "terminal", "points": terminal_points},
            ]
        )
        connector_colors[start_id] = connector
        connector_colors[terminal_id] = connector

    rng.shuffle(pieces)
    terminal_assignment = {
        "cyan": "amber" if rng.randrange(2) == 0 else "violet",
        "magenta": "violet",
    }
    if terminal_assignment["cyan"] == "violet":
        terminal_assignment["magenta"] = "amber"

    transform = {
        "dx": 96 + rng.randrange(-9, 10),
        "dy": 112 + rng.randrange(-9, 10),
        "mirror_x": bool(rng.randrange(2)),
        "flip_y": bool(rng.randrange(2)),
    }
    return {
        "route_pieces": pieces,
        "connector_colors": connector_colors,
        "terminal_assignment": terminal_assignment,
        "layout_transform": transform,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields; antialias through fixed supersampling."""

    supersample = 4
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    def scaled(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
        return [(x * supersample, y * supersample) for x, y in points]

    for piece in scene["route_pieces"]:
        points = _piece_points(scene, piece)
        draw.line(scaled(points), fill=ROUTE_RGB, width=ROUTE_WIDTH * supersample, joint="curve")

    # Solid markers are painted last, so the inverse can identify endpoints by
    # color while each dark piece remains a separate connected component.
    for piece in scene["route_pieces"]:
        points = _piece_points(scene, piece)
        connector = scene["connector_colors"][piece["piece_id"]]
        if piece["role"] == "start":
            endpoint = points[0]
            endpoint_rgb = GREEN_RGB
            connector_point = points[-1]
        else:
            endpoint = points[-1]
            endpoint_rgb = TERMINAL_RGB[scene["terminal_assignment"][connector]]
            connector_point = points[0]
        for center, color in ((endpoint, endpoint_rgb), (connector_point, CONNECTOR_RGB[connector])):
            cx, cy = center[0] * supersample, center[1] * supersample
            radius = MARKER_RADIUS * supersample
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    totals, terminals = _route_totals(scene)
    winner = max(sorted(totals), key=lambda key: (totals[key], key))
    return terminals[winner]


def margin(scene: dict[str, Any]) -> float:
    totals, _ = _route_totals(scene)
    values = sorted(totals.values())
    return float(values[-1] - values[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MARGIN_THRESHOLD


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # This scene representation has no unused identity slots: connector labels
    # define the only valid binding, and all geometry/labels are rasterized.
    return []
