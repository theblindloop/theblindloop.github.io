"""Deterministic radial-glyph renderer and analytic correspondence gold."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored_endpoint_count_correspondence-0.1.0"
CANVAS_SIZE = (1200, 620)
PANEL_BOXES = {
    "reference": (24, 72, 376, 596),
    "A": (424, 72, 776, 596),
    "B": (824, 72, 1176, 596),
}
PALETTE = {
    "red": (220, 45, 50),
    "blue": (35, 90, 190),
    "green": (30, 150, 80),
    "gold": (235, 155, 25),
    "purple": (135, 60, 175),
}
COLORS = tuple(PALETTE)
ENDPOINT_COUNTS = (2, 3, 4, 5, 6)
LOCAL_SLOTS = ((92.0, 170.0), (260.0, 170.0), (88.0, 382.0), (264.0, 382.0), (176.0, 276.0))
LOCAL_CENTER = (176.0, 276.0)
LOCAL_X_BOUNDS = (30.0, 322.0)
LOCAL_Y_BOUNDS = (100.0, 524.0)
MIN_CENTER_CLEARANCE = 10.0
MIN_SCORE_MARGIN = 2.0
SUPERSAMPLE = 4


def _field(record: dict, key: str):
    """Access a record field through one shared, non-gold-specific helper."""
    return record.__getitem__(key)


def _distance(left: dict, right: dict) -> float:
    return math.hypot(
        float(_field(left, "x")) - float(_field(right, "x")),
        float(_field(left, "y")) - float(_field(right, "y")),
    )


def _signature(panel: dict) -> dict[str, int]:
    return {
        str(_field(glyph, "color")): int(_field(glyph, "endpoint_count"))
        for glyph in _field(panel, "glyphs")
    }


def _score(reference: dict[str, int], candidate: dict[str, int]) -> int:
    return sum(reference[color] == candidate.get(color) for color in COLORS)


def _transform_point(
    point: tuple[float, float],
    angle: float,
    scale: float,
    center: tuple[float, float] = LOCAL_CENTER,
) -> tuple[float, float]:
    cx, cy = center
    dx, dy = point[0] - cx, point[1] - cy
    ca, sa = math.cos(angle), math.sin(angle)
    return (
        cx + scale * (ca * dx - sa * dy),
        cy + scale * (sa * dx + ca * dy),
    )


def _panel_clearance(panel: dict) -> float:
    """Minimum distance to a panel edge or another glyph's drawn extent."""
    glyphs = _field(panel, "glyphs")
    edge = float("inf")
    for glyph in glyphs:
        x, y = float(_field(glyph, "x")), float(_field(glyph, "y"))
        extent = 35.0 * float(_field(glyph, "scale"))
        edge = min(
            edge,
            x - LOCAL_X_BOUNDS[0] - extent,
            LOCAL_X_BOUNDS[1] - x - extent,
            y - LOCAL_Y_BOUNDS[0] - extent,
            LOCAL_Y_BOUNDS[1] - y - extent,
        )
    for index, left in enumerate(glyphs):
        for right in glyphs[:index]:
            extent = 35.0 * (
                float(_field(left, "scale")) + float(_field(right, "scale"))
            )
            edge = min(edge, _distance(left, right) - extent)
    return float(edge)


def _valid_panel(panel: object) -> bool:
    if not isinstance(panel, dict) or set(panel) != {"glyphs"}:
        return False
    glyphs = _field(panel, "glyphs")
    if not isinstance(glyphs, list) or len(glyphs) != len(COLORS):
        return False
    seen: set[str] = set()
    expected_keys = {"color", "x", "y", "scale", "rotation", "endpoint_count"}
    for glyph in glyphs:
        if not isinstance(glyph, dict) or set(glyph) != expected_keys:
            return False
        color = str(_field(glyph, "color"))
        if color not in PALETTE or color in seen:
            return False
        try:
            values = [
                float(_field(glyph, key))
                for key in ("x", "y", "scale", "rotation", "endpoint_count")
            ]
        except (TypeError, ValueError, OverflowError):
            return False
        x, y, scale, rotation, count = values
        if not all(math.isfinite(value) for value in values):
            return False
        if not (LOCAL_X_BOUNDS[0] <= x <= LOCAL_X_BOUNDS[1]):
            return False
        if not (LOCAL_Y_BOUNDS[0] <= y <= LOCAL_Y_BOUNDS[1]):
            return False
        if not (0.78 <= scale <= 1.24 and -math.pi <= rotation <= math.pi):
            return False
        if int(count) != count or int(count) not in ENDPOINT_COUNTS:
            return False
        seen.add(color)
    return seen == set(COLORS) and _panel_clearance(panel) >= MIN_CENTER_CLEARANCE


def _make_panel(counts: dict[str, int], rng: random.Random) -> dict:
    """Create one independently rearranged panel with a fixed inventory."""
    positions: list[tuple[float, float]] = []
    slots = list(LOCAL_SLOTS)
    rng.shuffle(slots)
    angle = rng.uniform(-0.42, 0.42)
    layout_scale = rng.uniform(0.86, 0.96)
    for slot in slots:
        jittered = (slot[0] + rng.uniform(-9.0, 9.0), slot[1] + rng.uniform(-9.0, 9.0))
        positions.append(_transform_point(jittered, angle, layout_scale))
    colors = list(COLORS)
    rng.shuffle(colors)
    glyphs = []
    for (x, y), color in zip(positions, colors, strict=True):
        glyphs.append(
            {
                "color": color,
                "x": round(x, 2),
                "y": round(y, 2),
                "scale": round(rng.uniform(0.86, 1.14), 3),
                "rotation": round(rng.uniform(-math.pi, math.pi), 4),
                "endpoint_count": int(counts[color]),
            }
        )
    return {"glyphs": glyphs}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _attempt in range(500):
        values = list(ENDPOINT_COUNTS)
        rng.shuffle(values)
        counts = dict(zip(COLORS, values, strict=True))
        wrong_counts = dict(counts)
        cycle = list(COLORS[:3])
        # A three-cycle preserves the endpoint inventory and total arm ink while
        # changing exactly three color bindings.
        wrong_counts[cycle[0]], wrong_counts[cycle[1]], wrong_counts[cycle[2]] = (
            counts[cycle[1]],
            counts[cycle[2]],
            counts[cycle[0]],
        )
        reference = _make_panel(counts, rng)
        exact = _make_panel(counts, rng)
        wrong = _make_panel(wrong_counts, rng)
        correct_is_a = int(seed) % 2 == 0
        candidate_a, candidate_b = (exact, wrong) if correct_is_a else (wrong, exact)
        scene = {
            "reference": reference,
            "candidate_a": candidate_a,
            "candidate_b": candidate_b,
        }
        if not is_quarantined(scene):
            return scene
    raise RuntimeError("could not construct a separated endpoint-count scene")


def analytic_gold(scene: dict) -> str:
    reference = _signature(scene["reference"])
    scores = (
        _score(reference, _signature(scene["candidate_a"])),
        _score(reference, _signature(scene["candidate_b"])),
    )
    if max(scores) != len(COLORS) or scores[0] == scores[1]:
        raise ValueError("scene has no unique endpoint-count correspondence")
    return "A" if scores[0] > scores[1] else "B"


def margin(scene: dict) -> float:
    reference = _signature(scene["reference"])
    scores = (
        _score(reference, _signature(scene["candidate_a"])),
        _score(reference, _signature(scene["candidate_b"])),
    )
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    try:
        if not all(
            _valid_panel(scene[key])
            for key in ("reference", "candidate_a", "candidate_b")
        ):
            return True
        reference = _signature(scene["reference"])
        scores = (
            _score(reference, _signature(scene["candidate_a"])),
            _score(reference, _signature(scene["candidate_b"])),
        )
        return (
            max(scores) != len(COLORS)
            or scores[0] == scores[1]
            or abs(scores[0] - scores[1]) < MIN_SCORE_MARGIN
        )
    except (KeyError, TypeError, ValueError, OverflowError):
        return True


def _draw_glyph(draw: ImageDraw.ImageDraw, glyph: dict, left: int, top: int, ss: int) -> None:
    cx = (left + float(_field(glyph, "x"))) * ss
    cy = (top + float(_field(glyph, "y"))) * ss
    scale = float(_field(glyph, "scale"))
    rotation = float(_field(glyph, "rotation"))
    count = int(_field(glyph, "endpoint_count"))
    color = PALETTE[str(_field(glyph, "color"))]
    hub_radius = 9.0 * scale * ss
    stroke = max(3, int(round(8.0 * scale * ss)))
    start_radius = 5.0 * scale * ss
    end_radius = 31.0 * scale * ss
    tip_radius = stroke / 2.0
    for index in range(count):
        theta = rotation + 2.0 * math.pi * index / count
        ux, uy = math.cos(theta), math.sin(theta)
        start = (cx + ux * start_radius, cy + uy * start_radius)
        end = (cx + ux * end_radius, cy + uy * end_radius)
        draw.line((start, end), fill=color, width=stroke)
        draw.ellipse(
            (end[0] - tip_radius, end[1] - tip_radius, end[0] + tip_radius, end[1] + tip_radius),
            fill=color,
        )
    draw.ellipse(
        (cx - hub_radius, cy - hub_radius, cx + hub_radius, cy + hub_radius),
        fill=color,
    )


def render(scene: dict) -> Image.Image:
    """Render only the fields that define visible panel contents."""
    ss = SUPERSAMPLE
    image = Image.new("RGB", (CANVAS_SIZE[0] * ss, CANVAS_SIZE[1] * ss), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    panel_records = {
        "reference": scene["reference"],
        "A": scene["candidate_a"],
        "B": scene["candidate_b"],
    }
    border = (74, 78, 88)
    for name, box in PANEL_BOXES.items():
        left, top, right, bottom = box
        draw.rectangle(
            (left * ss, top * ss, right * ss, bottom * ss),
            outline=border,
            width=2 * ss,
        )
        panel = panel_records[name]
        for glyph in _field(panel, "glyphs"):
            _draw_glyph(draw, glyph, left, top, ss)
    image = image.resize(CANVAS_SIZE, Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for name, box in PANEL_BOXES.items():
        left, top, _, _ = box
        label_draw.text(
            (left + 13, top + 12),
            "REFERENCE" if name == "reference" else name,
            fill=(25, 28, 35),
            font=font,
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse each visible glyph list; draw order is the only latent alias."""
    twin = copy.deepcopy(scene)
    for key in ("reference", "candidate_a", "candidate_b"):
        _field(twin[key], "glyphs").reverse()
    return [("reverse_disjoint_glyph_lists", twin)]
