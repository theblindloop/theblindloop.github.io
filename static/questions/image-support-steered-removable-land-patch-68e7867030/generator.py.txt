"""Latent sampler, analytic arm, and pixel renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 384
BACKGROUND = (250, 249, 246)
LAND = (29, 35, 43)
PATCH = (244, 181, 45)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    scale = rng.choice([1, 1, 1, 2])
    # The scale-2 badge remains compact relative to the canvas.
    badge_span = 84 * scale
    return {
        "topology": "cut" if int(seed) % 2 == 0 else "bypass",
        "origin_x": rng.randint(22, CANVAS - badge_span - 22),
        "origin_y": rng.randint(22, CANVAS - badge_span - 22),
        "rotation": rng.choice([0, 90, 180, 270]),
        "scale": scale,
        "tab_side": rng.choice(["near", "far"]),
        "canvas_width": CANVAS,
        "canvas_height": CANVAS,
    }


def _badge(scene: dict) -> Image.Image:
    s = int(scene["scale"])
    badge = Image.new("RGBA", (84 * s, 68 * s), (0, 0, 0, 0))
    draw = ImageDraw.Draw(badge)
    # Inclusive Pillow rectangles are avoided by using explicit x2-1/y2-1.
    def rect(box: tuple[int, int, int, int], color: tuple[int, int, int]) -> None:
        x0, y0, x1, y1 = (v * s for v in box)
        draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=(*color, 255))

    rect((0, 8, 28, 60), LAND)
    rect((56, 8, 84, 60), LAND)
    if scene["topology"] == "bypass":
        rect((28, 46, 56, 60), LAND)
    else:
        # Equalize black ink with the bypass class using a tab attached to one lobe.
        if scene["tab_side"] == "near":
            rect((0, 0, 28, 8), LAND)
            rect((0, 60, 28, 64), LAND)
        else:
            rect((56, 0, 84, 8), LAND)
            rect((56, 60, 84, 64), LAND)
    rect((28, 16, 56, 48), PATCH)
    rotation = int(scene["rotation"])
    if rotation == 90:
        badge = badge.transpose(Image.Transpose.ROTATE_90)
    elif rotation == 180:
        badge = badge.transpose(Image.Transpose.ROTATE_180)
    elif rotation == 270:
        badge = badge.transpose(Image.Transpose.ROTATE_270)
    elif rotation != 0:
        raise ValueError("rotation must be a right angle")
    return badge


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    badge = _badge(scene)
    image.paste(badge, (int(scene["origin_x"]), int(scene["origin_y"])), badge)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["topology"] == "cut" else "no"


def margin(scene: dict) -> float:
    # Minimum nominal neck/attachment thickness in final pixels.
    return float(12 * int(scene["scale"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
