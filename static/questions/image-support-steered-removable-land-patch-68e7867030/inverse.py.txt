"""Independent pixel-only inverse arm.

This module deliberately knows only the public colors and topological rule. It
does not import or inspect renderer state, manifests, prompts, or filenames.
"""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        q = deque([(sy, sx)])
        seen[sy, sx] = True
        size = 0
        while q:
            y, x = q.popleft()
            size += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Broad color classes tolerate ordinary lossless decoding and small RGB jitter.
    yellow = (arr[:, :, 0] > 190) & (arr[:, :, 1] > 120) & (arr[:, :, 2] < 105)
    black = (arr.max(axis=2) < 92) & (arr.mean(axis=2) < 70)
    yellow_components = _components(yellow)
    patch_area = yellow_components[0] if len(yellow_components) == 1 else 0
    intact_patch = 880 <= patch_area <= 910 or 3550 <= patch_area <= 3620
    if len(yellow_components) != 1 or not intact_patch:
        raise OracleAbstention("missing, fragmented, or damaged removable patch")
    black_area = int(black.sum())
    if not (2200 <= black_area <= 13500):
        raise OracleAbstention("land area outside the supported badge range")
    black_components = _components(black)
    if len(black_components) not in (1, 2) or min(black_components) < 700:
        raise OracleAbstention("remaining land is damaged or has an unsupported topology")
    return "yes" if len(black_components) == 2 else "no"
