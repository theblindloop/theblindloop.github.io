"""Separating-loop-count world renderer.

Three closed black rings (loops) are drawn on a white canvas, well separated so
their interiors do not overlap. Two small colored dots (green and red) appear
somewhere on the canvas.  A loop SEPARATES the pair when exactly one of the two
dots lies in its interior (the loop has the green dot on one side and the red
dot on the other).  A loop does NOT separate them when both dots lie on the same
side of it (both inside or both outside).

Decision: "how many" of the three loops separate the green and red dots; an
integer in {0, 1, 2, 3}.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 400,
        "height": 400,
        "background": "white",
        "loops": [[cx, cy, r], ...],   # three disjoint closed black ring loops
        "loop_color": "black",
        "loop_width": 10.0,
        "dots":    [{"color": "green", "x": .., "y": ..},
                    {"color": "red",   "x": .., "y": ..}],
        "dot_radius": 8.0,
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "separating_count-0.4.0"

DOT_COLORS = {
    "green": (0, 128, 0),
    "red": (255, 0, 0),
}

LOOP_WIDTH = 10.0
LOOP_HALF = LOOP_WIDTH / 2.0
DOT_RADIUS = 8.0

MIN_R = 34.0
MAX_R = 72.0
DOT_CLEAR = 11.0        # dot centroid must clear every loop rim by this many px
DOT_SEP = 24.0          # min center distance between the two dots
LOOP_GAP = 16.0         # min center-to-center gap between neighbouring loops
CANVAS_MARGIN = 18.0
QUARANTINE_MIN = 7.0    # below this the inside/outside reading is fragile


def _inside_canvas(x: float, y: float, width: int, height: int) -> bool:
    return (
        CANVAS_MARGIN < x < width - CANVAS_MARGIN
        and CANVAS_MARGIN < y < height - CANVAS_MARGIN
    )


def _dot_clearance(dot: dict, loops: list[list[float]]) -> float:
    """Min over every (dot, loop) of |dist(dot, loop_center) - loop_radius|."""
    best: float = float("inf")
    for (cx, cy, r) in loops:
        d = math.hypot(dot["x"] - cx, dot["y"] - cy)
        best = min(best, abs(d - r))
    return best


def _place_dot(
    rng: random.Random,
    loops: list[list[float]],
    existing: list[dict],
    width: int,
    height: int,
) -> dict | None:
    for _ in range(1600):
        x = rng.uniform(CANVAS_MARGIN, width - CANVAS_MARGIN)
        y = rng.uniform(CANVAS_MARGIN, height - CANVAS_MARGIN)
        dot = {"x": x, "y": y}
        if _dot_clearance(dot, loops) < DOT_CLEAR:
            continue
        if any(math.hypot(x - o["x"], y - o["y"]) < DOT_SEP for o in existing):
            continue
        return dot
    return None


def _sample_loops(rng: random.Random, width: int, height: int) -> list[list[float]] | None:
    for _attempt in range(2000):
        cx = width / 2.0
        cy = height / 2.0
        base = rng.uniform(68.0, 90.0)
        angles = sorted(rng.uniform(0.0, 2.0 * math.pi) for _ in range(3))
        radii = [rng.uniform(MIN_R, MAX_R) for _ in range(3)]
        centers = []
        for a in angles:
            px = cx + base * math.cos(a)
            py = cy + base * math.sin(a)
            centers.append((px, py))
        for (px, py), r in zip(centers, radii):
            if not (
                _inside_canvas(px - r, py, width, height)
                and _inside_canvas(px + r, py, width, height)
                and _inside_canvas(px, py - r, width, height)
                and _inside_canvas(px, py + r, width, height)
            ):
                break
        else:
            ok = True
            for i in range(3):
                for j in range(i + 1, 3):
                    d = math.hypot(centers[i][0] - centers[j][0], centers[i][1] - centers[j][1])
                    if d < radii[i] + radii[j] + LOOP_GAP:
                        ok = False
                        break
                if not ok:
                    break
            if ok:
                return [
                    [centers[i][0], centers[i][1], float(radii[i])] for i in range(3)
                ]
    return None


def sample_scene(rng: random.Random, width: int = 400, height: int = 400) -> dict:
    """Sample a valid scene with three loops and two dots."""
    for _attempt in range(200):
        loops = _sample_loops(rng, width, height)
        if loops is None:
            continue
        g = _place_dot(rng, loops, [], width, height)
        if g is None:
            continue
        g["color"] = "green"
        r = _place_dot(rng, loops, [g], width, height)
        if r is None:
            continue
        r["color"] = "red"
        scene = {
            "width": width,
            "height": height,
            "background": "white",
            "loops": loops,
            "loop_color": "black",
            "loop_width": LOOP_WIDTH,
            "dots": [g, r],
            "dot_radius": DOT_RADIUS,
        }
        return scene
    return _fallback_scene()


def _fallback_scene() -> dict:
    loops = [
        [200.0, 118.0, 44.0],
        [120.0, 260.0, 44.0],
        [280.0, 260.0, 44.0],
    ]
    dots = [
        {"color": "green", "x": 200.0, "y": 200.0},
        {"color": "red", "x": 200.0, "y": 330.0},
    ]
    return {
        "width": 400,
        "height": 400,
        "background": "white",
        "loops": loops,
        "loop_color": "black",
        "loop_width": LOOP_WIDTH,
        "dots": dots,
        "dot_radius": DOT_RADIUS,
    }


def render(scene: dict, supersample: int = 1) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    loop_w = max(1, int(round(float(scene["loop_width"]) * ss)))
    dot_r = float(scene["dot_radius"]) * ss
    for (ccx, ccy, r) in scene["loops"]:
        box = [(ccx - r) * ss, (ccy - r) * ss, (ccx + r) * ss, (ccy + r) * ss]
        draw.ellipse(box, outline=scene.get("loop_color", "black"), width=loop_w)
    for dot in scene["dots"]:
        (dx, dy) = (float(dot["x"]), float(dot["y"]))
        dbox = [(dx - dot_r) * ss, (dy - dot_r) * ss, (dx + dot_r) * ss, (dy + dot_r) * ss]
        draw.ellipse(dbox, fill=DOT_COLORS[dot["color"]])
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def _loop_separation(scene: dict, loop: list[float], dots: dict) -> bool:
    (cx, cy, r) = (float(loop[0]), float(loop[1]), float(loop[2]))
    g_in = math.hypot(dots["green"]["x"] - cx, dots["green"]["y"] - cy) < r
    r_in = math.hypot(dots["red"]["x"] - cx, dots["red"]["y"] - cy) < r
    return g_in != r_in


def analytic_gold(scene: dict) -> str:
    """Return how many of the three loops separate the green and red dots."""
    dots = {dot["color"]: dot for dot in scene["dots"]}
    count = sum(1 for loop in scene["loops"] if _loop_separation(scene, loop, dots))
    return str(count)


def margin(scene: dict) -> float:
    """Robustness margin: the smallest distance between any dot and any loop rim."""
    best: float = float("inf")
    for dot in scene["dots"]:
        best = min(best, _dot_clearance(dot, scene["loops"]))
    if not scene["dots"] or math.isinf(best):
        return float("inf")
    return max(0.0, float(best))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render-invariant symmetry: every stored loop and dot
    coordinate changes the picture, and the decision is a deterministic function
    of the pixels. Declared not_applicable."""
    return iter(())
