"""Independent pixel-only inverse arm for the separating-loop-count world.

Receives ONLY the final PNG. It never imports or reads the renderer, the analytic
scene, prompts, or any renderer state. It recovers the decision by:

1. marking the two dot colors and taking each dot's centroid;
2. finding every connected black component (one per closed loop ring) and, for
   each, using the component's centroid as the loop center and the mean pixel
   distance as the loop radius;
3. for each loop deciding whether exactly one of the two dots is inside it;
4. returning how many loops separate the pair (as a string '0'..'3').
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "separating_count-oracle-0.4.0"

_DOT_RGB = {
    "green": (0, 128, 0),
    "red": (255, 0, 0),
}
_LOOP_MAX = 90


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, returned as a list of masks."""
    h, w = mask.shape
    labels = np.zeros_like(mask, dtype=np.int32)
    current = 0
    for y in range(h):
        for x in range(w):
            if not mask[y, x] or labels[y, x]:
                continue
            current += 1
            labels[y, x] = current
            stack = [(x, y)]
            while stack:
                cx, cy = stack.pop()
                for nx, ny in ((cx + 1, cy), (cx - 1, cy), (cx, cy + 1), (cx, cy - 1)):
                    if (
                        0 <= nx < w
                        and 0 <= ny < h
                        and mask[ny, nx]
                        and labels[ny, nx] == 0
                    ):
                        labels[ny, nx] = current
                        stack.append((nx, ny))
    comps: list[np.ndarray] = []
    for label in range(1, current + 1):
        comps.append(labels == label)
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    dots: dict[str, tuple[float, float]] = {}
    for name, (r, g, b) in _DOT_RGB.items():
        mask = (arr[..., 0] == r) & (arr[..., 1] == g) & (arr[..., 2] == b)
        c = _centroid(mask)
        if c is not None:
            dots[name] = c

    black = (arr[..., 0] <= _LOOP_MAX) & (arr[..., 1] <= _LOOP_MAX) & (arr[..., 2] <= _LOOP_MAX)
    loops: list[tuple[float, float, float]] = []
    for comp in _components(black):
        ys, xs = np.nonzero(comp)
        if len(xs) < 8:
            continue
        center_x = float(xs.mean())
        center_y = float(ys.mean())
        dists = np.hypot(xs.astype(float) - center_x, ys.astype(float) - center_y)
        loops.append((center_x, center_y, float(dists.mean())))

    if "green" not in dots or "red" not in dots:
        return "0"

    count = 0
    for (cx, cy, r) in loops:
        g_in = math.hypot(dots["green"][0] - cx, dots["green"][1] - cy) < r
        r_in = math.hypot(dots["red"][0] - cx, dots["red"][1] - cy) < r
        if g_in != r_in:
            count += 1
    return str(count)
