"""Pixel-only inverse arm for the layer-peel reveal world.

The oracle receives one RGB image and nothing else. It thresholds the image at
the midpoint of its own darkest and lightest tones, finds the bordered panel as
the ink component with the largest bounding box, measures the border thickness,
rectifies the eleven-by-eleven cell lattice, reads every cell as dark or pale,
executes the declared operation - delete every dark cell that touches a pale
cell or the lattice edge, corners included - and classifies the surviving cells
against the declared figure alphabet with a reported runner-up margin. It
abstains by raising whenever the panel, the lattice, a cell tone or the figure
decision is ambiguous.
"""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

ORACLE_VERSION = "peel_layer_reveal-oracle-0.1.0"

CELLS = 11
MIN_PANEL_PX = 200
MAX_PANEL_PX = 340
MIN_FRAME_PX = 3
MAX_FRAME_PX = 10
MAX_FIGURE_ERRORS = 0
MIN_FIGURE_GAP = 5


def _templates() -> dict[str, np.ndarray]:
    def build(cells: set[tuple[int, int]]) -> np.ndarray:
        grid = np.zeros((CELLS, CELLS), dtype=np.uint8)
        for row, col in cells:
            grid[row, col] = 1
        return grid

    def row_run(row: int, col0: int, col1: int) -> set[tuple[int, int]]:
        return {(row, col) for col in range(col0, col1 + 1)}

    def col_run(col: int, row0: int, row1: int) -> set[tuple[int, int]]:
        return {(row, col) for row in range(row0, row1 + 1)}

    return {
        "bars": build(row_run(2, 2, 6) | row_run(6, 4, 8)),
        "corner": build(col_run(3, 4, 8) | row_run(4, 4, 8)),
        "pipes": build(col_run(3, 2, 6) | col_run(7, 4, 8)),
        "tee": build(row_run(3, 3, 7) | col_run(5, 4, 8)),
    }


TEMPLATES = _templates()


def _ink_mask(array: np.ndarray) -> np.ndarray:
    luma = 0.299 * array[:, :, 0] + 0.587 * array[:, :, 1] + 0.114 * array[:, :, 2]
    darkest = float(np.percentile(luma, 0.2))
    lightest = float(np.percentile(luma, 99.8))
    if lightest - darkest < 20.0:
        raise ValueError("image has no usable ink-to-paper contrast")
    return luma < 0.5 * (darkest + lightest)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    """Bounding boxes of 4-connected ink components, widest box first."""

    seen = np.zeros(mask.shape, dtype=bool)
    boxes: list[tuple[int, int, int, int, int]] = []
    height, width = mask.shape
    for start_y, start_x in np.argwhere(mask):
        if seen[start_y, start_x]:
            continue
        stack = [(int(start_y), int(start_x))]
        seen[start_y, start_x] = True
        y0 = y1 = int(start_y)
        x0 = x1 = int(start_x)
        while stack:
            cy, cx = stack.pop()
            y0, y1 = min(y0, cy), max(y1, cy)
            x0, x1 = min(x0, cx), max(x1, cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        boxes.append((max(y1 - y0, x1 - x0), y0, x0, y1, x1))
    if not boxes:
        raise ValueError("image contains no ink")
    boxes.sort(reverse=True)
    return [(y0, x0, y1, x1) for _, y0, x0, y1, x1 in boxes]


def _panel(mask: np.ndarray) -> tuple[int, int, int, int]:
    """The bordered panel: the ink component with the largest bounding box."""

    y0, x0, y1, x1 = _components(mask)[0]
    height = y1 - y0 + 1
    width = x1 - x0 + 1
    if height != width:
        raise ValueError("largest ink component is not square")
    if not (MIN_PANEL_PX <= height <= MAX_PANEL_PX):
        raise ValueError("largest ink component is not panel sized")
    return (y0, x0, y1, x1)


def _frame_thickness(mask: np.ndarray, box: tuple[int, int, int, int]) -> int:
    y0, x0, y1, x1 = box
    window = mask[y0:y1 + 1, x0:x1 + 1]
    runs = []
    for lines in (window, window[:, ::-1], window.T, window.T[:, ::-1]):
        for line in lines:
            run = 0
            while run < line.size and line[run]:
                run += 1
            if run:
                runs.append(int(run))
    if not runs:
        raise ValueError("panel has no measurable border")
    thickness = min(runs)
    if not (MIN_FRAME_PX <= thickness <= MAX_FRAME_PX):
        raise ValueError("panel border thickness is out of range")
    side = y1 - y0 + 1
    if (side - 2 * thickness) % CELLS != 0:
        raise ValueError("panel interior does not divide into an 11x11 lattice")
    return thickness


def _read_cells(mask: np.ndarray, box: tuple[int, int, int, int], frame: int) -> np.ndarray:
    y0, x0, y1, x1 = box
    cell = ((y1 - y0 + 1) - 2 * frame) // CELLS
    if cell < 10:
        raise ValueError("panel cells are too small to read")
    grid = np.zeros((CELLS, CELLS), dtype=np.uint8)
    for row in range(CELLS):
        for col in range(CELLS):
            cy0 = y0 + frame + row * cell
            cx0 = x0 + frame + col * cell
            patch = mask[cy0 + 2:cy0 + cell - 2, cx0 + 2:cx0 + cell - 2]
            if patch.size == 0:
                raise ValueError("panel cell patch is empty")
            share = float(patch.mean())
            if 0.2 < share < 0.8:
                raise ValueError("panel cell tone is ambiguous")
            grid[row, col] = 1 if share >= 0.8 else 0
    return grid


def _peel(grid: np.ndarray) -> np.ndarray:
    """The declared operation: keep only cells whose eight neighbours are dark."""

    padded = np.zeros((CELLS + 2, CELLS + 2), dtype=np.uint8)
    padded[1:-1, 1:-1] = grid
    survives = np.ones((CELLS, CELLS), dtype=np.uint8)
    for drow, dcol in itertools.product((-1, 0, 1), repeat=2):
        survives &= padded[1 + drow:CELLS + 1 + drow, 1 + dcol:CELLS + 1 + dcol]
    return (survives & grid).astype(np.uint8)


def decision_from_image(image: Image.Image) -> str:
    """Recover the revealed figure from the final pixels alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _ink_mask(array)
    box = _panel(mask)
    frame = _frame_thickness(mask, box)
    drawn = _read_cells(mask, box, frame)
    if int(drawn.sum()) < 20:
        raise ValueError("panel carries too little ink to peel")
    survivors = _peel(drawn)
    scores = sorted(
        (int(np.abs(survivors.astype(int) - template.astype(int)).sum()), name)
        for name, template in TEMPLATES.items()
    )
    if scores[0][0] > MAX_FIGURE_ERRORS:
        raise ValueError("peeled pattern matches no declared figure")
    if scores[1][0] - scores[0][0] < MIN_FIGURE_GAP:
        raise ValueError("peeled pattern is ambiguous between two figures")
    return scores[0][1]
