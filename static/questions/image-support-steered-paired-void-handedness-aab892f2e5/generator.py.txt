"""Deterministic renderer for paired negative-space handedness."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
PANEL = 160
GRID = 5
CELL = 24
INSET = (PANEL - GRID * CELL) // 2
BACKGROUND = (248, 246, 240)
PALE = BACKGROUND
PALETTES = (
    (35, 69, 105),
    (91, 52, 99),
    (31, 96, 85),
    (120, 66, 38),
)


def _normalize(cells: Iterable[tuple[int, int]]) -> frozenset[tuple[int, int]]:
    points = {(int(x), int(y)) for x, y in cells}
    if not points:
        raise ValueError("empty polyomino")
    min_x = min(x for x, _ in points)
    min_y = min(y for _, y in points)
    return frozenset((x - min_x, y - min_y) for x, y in points)


def _rotate(cells: Iterable[tuple[int, int]], turns: int = 1) -> frozenset[tuple[int, int]]:
    result = _normalize(cells)
    for _ in range(int(turns) % 4):
        result = _normalize((-y, x) for x, y in result)
    return result


def _reflect(cells: Iterable[tuple[int, int]]) -> frozenset[tuple[int, int]]:
    return _normalize((-x, y) for x, y in cells)


def _orbit(cells: Iterable[tuple[int, int]]) -> tuple[frozenset[tuple[int, int]], ...]:
    return tuple({_rotate(cells, turns) for turns in range(4)})


def _chirality_margin(cells: Iterable[tuple[int, int]]) -> int:
    direct = _orbit(cells)
    reflected = _orbit(_reflect(cells))
    return min(len(a.symmetric_difference(b)) for a in direct for b in reflected)


def _random_chiral_polyomino(rng: random.Random) -> list[list[int]]:
    for _ in range(10000):
        cells = {(0, 0)}
        while len(cells) < 8:
            x, y = rng.choice(sorted(cells))
            dx, dy = rng.choice(((1, 0), (-1, 0), (0, 1), (0, -1)))
            cells.add((x + dx, y + dy))
        shape = _normalize(cells)
        width = max(x for x, _ in shape) + 1
        height = max(y for _, y in shape) + 1
        if width > GRID or height > GRID or width < 3 or height < 3:
            continue
        if max(width, height) < 4:
            continue
        if _chirality_margin(shape) < 4:
            continue
        return [[x, y] for x, y in sorted(shape)]
    raise RuntimeError("failed to sample a well-separated chiral polyomino")


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; parity controls only the answer class."""
    rng = random.Random((int(seed) << 17) ^ 0x5A17C3)
    relation = "same" if int(seed) % 2 == 0 else "mirrored"
    return {
        "base_cells": _random_chiral_polyomino(rng),
        "relation": relation,
        "left_turns": rng.randrange(4),
        "right_turns": rng.randrange(4),
        "palette_index": rng.randrange(len(PALETTES)),
        "jitter_x": rng.randint(-6, 6),
        "jitter_y": rng.randint(-24, 24),
    }


def _pair(scene: dict) -> tuple[frozenset[tuple[int, int]], frozenset[tuple[int, int]]]:
    base = _normalize(tuple(map(tuple, scene["base_cells"])))
    right_base = base if scene["relation"] == "same" else _reflect(base)
    return _rotate(base, scene["left_turns"]), _rotate(right_base, scene["right_turns"])


def _center_in_grid(cells: Iterable[tuple[int, int]]) -> frozenset[tuple[int, int]]:
    shape = _normalize(cells)
    width = max(x for x, _ in shape) + 1
    height = max(y for _, y in shape) + 1
    ox = (GRID - width) // 2
    oy = (GRID - height) // 2
    return frozenset((x + ox, y + oy) for x, y in shape)


def render(scene: dict) -> Image.Image:
    """Render two distant dark tiles whose background complements are the shapes."""
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    ink = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    x_positions = (20 + int(scene["jitter_x"]), 332 + int(scene["jitter_x"]))
    top = 176 + int(scene["jitter_y"])
    left_shape, right_shape = _pair(scene)

    for panel_x, shape in zip(x_positions, (left_shape, right_shape), strict=True):
        box = (panel_x * scale, top * scale, (panel_x + PANEL) * scale, (top + PANEL) * scale)
        draw.rounded_rectangle(box, radius=9 * scale, fill=ink)
        for gx, gy in _center_in_grid(shape):
            x0 = (panel_x + INSET + gx * CELL) * scale
            y0 = (top + INSET + gy * CELL) * scale
            draw.rectangle((x0, y0, x0 + CELL * scale, y0 + CELL * scale), fill=PALE)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    base = _normalize(tuple(map(tuple, scene["base_cells"])))
    if _chirality_margin(base) <= 0:
        raise ValueError("analytic label is undefined for an achiral cutout")
    relation = str(scene["relation"])
    if relation not in {"same", "mirrored"}:
        raise ValueError("invalid relation")
    return relation


def margin(scene: dict) -> float:
    base = _normalize(tuple(map(tuple, scene["base_cells"])))
    return float(_chirality_margin(base))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["left_turns"] = int(twin["left_turns"]) + 4
    twin["right_turns"] = int(twin["right_turns"]) + 4
    return [("full_turn_alias", twin)]
