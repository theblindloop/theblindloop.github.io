"""Independent image-only inverse for paired void handedness.

This module deliberately duplicates the public visual rules.  It receives only
the final RGB image, finds the two large ink components from border-relative
contrast, reconstructs each eight-cell background complement, and compares the
rotation and reflection orbits.
"""

from __future__ import annotations

from typing import Iterable

import numpy as np
from PIL import Image

GRID = 5
EXPECTED_HOLES = 8


class OracleAbstention(ValueError):
    pass


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(indices) - 1]
    return [(int(indices[a]), int(indices[b]) + 1) for a, b in zip(starts, ends, strict=True)]


def _normalize(cells: Iterable[tuple[int, int]]) -> frozenset[tuple[int, int]]:
    points = {(int(x), int(y)) for x, y in cells}
    if not points:
        raise OracleAbstention("empty cutout")
    x0 = min(x for x, _ in points)
    y0 = min(y for _, y in points)
    return frozenset((x - x0, y - y0) for x, y in points)


def _rotate(cells: Iterable[tuple[int, int]], turns: int) -> frozenset[tuple[int, int]]:
    result = _normalize(cells)
    for _ in range(int(turns) % 4):
        result = _normalize((-y, x) for x, y in result)
    return result


def _reflect(cells: Iterable[tuple[int, int]]) -> frozenset[tuple[int, int]]:
    return _normalize((-x, y) for x, y in cells)


def _orbit(cells: Iterable[tuple[int, int]]) -> set[frozenset[tuple[int, int]]]:
    return {_rotate(cells, turns) for turns in range(4)}


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)), axis=0
    )
    return np.median(border, axis=0)


def _decode_panel(
    array: np.ndarray, background: np.ndarray, mask: np.ndarray, x0: int, x1: int
) -> frozenset[tuple[int, int]]:
    local = mask[:, x0:x1]
    rows = np.flatnonzero(local.sum(axis=1) > 12)
    if not len(rows):
        raise OracleAbstention("missing tile")
    y0, y1 = int(rows[0]), int(rows[-1]) + 1
    width, height = x1 - x0, y1 - y0
    if not (154 <= width <= 166 and 154 <= height <= 166):
        raise OracleAbstention("damaged tile bounds")
    fill = float(mask[y0:y1, x0:x1].mean())
    if not (0.76 <= fill <= 0.87):
        raise OracleAbstention("damaged tile area")

    cell = width * 24.0 / 160.0
    inset_x = width * 20.0 / 160.0
    inset_y = height * 20.0 / 160.0
    holes: set[tuple[int, int]] = set()
    for gy in range(GRID):
        for gx in range(GRID):
            cx = int(round(x0 + inset_x + (gx + 0.5) * cell))
            cy = int(round(y0 + inset_y + (gy + 0.5) * cell))
            patch = array[max(0, cy - 2):cy + 3, max(0, cx - 2):cx + 3].astype(np.float32)
            distance = float(np.linalg.norm(patch.mean(axis=(0, 1)) - background))
            if distance < 28.0:
                holes.add((gx, gy))
    if len(holes) != EXPECTED_HOLES:
        raise OracleAbstention(f"expected eight cutout blocks, recovered {len(holes)}")
    return _normalize(holes)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 64:
        raise OracleAbstention("unsupported raster")
    background = _background(array).astype(np.float32)
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    mask = distance > 55.0
    column_runs = [run for run in _runs(mask.sum(axis=0) > 12) if run[1] - run[0] >= 40]
    if len(column_runs) != 2:
        raise OracleAbstention(f"expected two distant tiles, found {len(column_runs)}")
    left = _decode_panel(array, background, mask, *column_runs[0])
    right = _decode_panel(array, background, mask, *column_runs[1])

    direct = _orbit(left)
    reflected = _orbit(_reflect(left))
    in_direct = right in direct
    in_reflected = right in reflected
    if in_direct == in_reflected:
        raise OracleAbstention("cutouts are damaged or reflection-ambiguous")
    return "same" if in_direct else "mirrored"
