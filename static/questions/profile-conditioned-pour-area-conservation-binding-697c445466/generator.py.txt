"""Renderer and analytic gold for the pour / conserved-area correspondence world.

Top panel: four open vessels of different interior widths, each holding a solid
block of coloured liquid. Bottom panel: four measuring tubes of one common
width, each holding the poured liquid of exactly one vessel in a single gray, in
a permuted left-to-right order, with a black triangle marking one tube.

Because pouring conserves area, tube height equals vessel_width * vessel_height
divided by the common tube width. Scenes are rejected unless the height-only
heuristics (matching by height rank and matching by absolute height) both point
at the wrong vessel for the queried tube.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 540

BACKGROUND = (246, 246, 242)
WALL = (70, 70, 70)
GRAY_FILL = (120, 120, 120)
DIVIDER = (160, 160, 160)
MARKER = (0, 0, 0)

COLOUR_NAMES = ("red", "blue", "green", "orange")
COLOUR_RGB = {
    "red": (214, 44, 44),
    "blue": (40, 84, 206),
    "green": (28, 150, 74),
    "orange": (240, 152, 20),
}

TUBE_WIDTH = 40
WALL_THICK = 4

SLOT_CENTRES = (110, 265, 420, 575)
TOP_FLOOR_Y = 250
TOP_CEIL_Y = 56
BOTTOM_FLOOR_Y = 500
BOTTOM_CEIL_Y = 292
DIVIDER_Y = 266

VESSEL_WIDTHS = (20, 30, 40, 50, 60)
MIN_TUBE_H = 30
MAX_TUBE_H = 200
QUARANTINE_GAP = 14.0


def _ranks_desc(values: list[int]) -> list[int]:
    order = sorted(range(len(values)), key=lambda i: -values[i])
    ranks = [0] * len(values)
    for position, index in enumerate(order):
        ranks[index] = position
    return ranks


def sample_scene(seed: int) -> dict:
    """Deterministically sample one balanced, shortcut-resistant scene."""
    rng = random.Random(seed * 7919 + 13)
    for _ in range(20000):
        widths = rng.sample(VESSEL_WIDTHS, 4)
        heights = [rng.randrange(24, 164, 4) for _ in range(4)]
        if len(set(heights)) != 4:
            continue
        areas = [w * h for w, h in zip(widths, heights)]
        if len(set(areas)) != 4:
            continue
        if any(a % TUBE_WIDTH for a in areas):
            continue
        tube_h = [a // TUBE_WIDTH for a in areas]
        if not all(MIN_TUBE_H <= t <= MAX_TUBE_H for t in tube_h):
            continue
        if min(
            abs(tube_h[i] - tube_h[j]) for i in range(4) for j in range(i + 1, 4)
        ) < 3:
            continue

        colours = rng.sample(list(COLOUR_NAMES), 4)
        desired = rng.choice(COLOUR_NAMES)
        target = colours.index(desired)

        # The queried vessel must not be readable off one-dimensional height.
        if widths[target] == TUBE_WIDTH:
            continue
        if _ranks_desc(heights)[target] == _ranks_desc(tube_h)[target]:
            continue
        naive = min(range(4), key=lambda j: (abs(heights[j] - tube_h[target]), j))
        if naive == target:
            continue

        gap = min(abs(tube_h[j] - tube_h[target]) for j in range(4) if j != target)
        tight = rng.random() < 0.18
        if tight:
            if not 3 <= gap <= 10:
                continue
        elif gap < QUARANTINE_GAP:
            continue

        marked_slot = rng.randrange(4)
        others = [j for j in range(4) if j != target]
        rng.shuffle(others)
        tube_order = [0, 0, 0, 0]
        tube_order[marked_slot] = target
        for slot in (s for s in range(4) if s != marked_slot):
            tube_order[slot] = others.pop()

        return {
            "widths": widths,
            "heights": heights,
            "colours": colours,
            "tube_order": tube_order,
            "marked_slot": marked_slot,
            "tight_mode": bool(tight),
            "tube_width": TUBE_WIDTH,
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _tube_heights(scene: dict) -> list[int]:
    tw = int(scene["tube_width"])
    return [w * h // tw for w, h in zip(scene["widths"], scene["heights"])]


def _target_vessel(scene: dict) -> int:
    return int(scene["tube_order"][int(scene["marked_slot"])])


def analytic_gold(scene: dict) -> str:
    """Colour of the vessel that was poured into the marked tube."""
    return str(scene["colours"][_target_vessel(scene)])


def margin(scene: dict) -> float:
    """Runner-up area gap, expressed in tube-height pixels."""
    tube_h = _tube_heights(scene)
    target = _target_vessel(scene)
    return float(min(abs(tube_h[j] - tube_h[target]) for j in range(4) if j != target))


def is_quarantined(scene: dict) -> bool:
    """Ambiguous when a second vessel nearly reproduces the marked tube's area."""
    return bool(margin(scene) < QUARANTINE_GAP)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every field the gold reads is drawn into the raster."""
    return []


def _draw_container(
    draw: ImageDraw.ImageDraw,
    centre: int,
    inner_w: int,
    floor_y: int,
    ceil_y: int,
    fill_h: int,
    fill_rgb: tuple[int, int, int],
) -> None:
    x0 = centre - inner_w // 2
    x1 = x0 + inner_w - 1
    draw.rectangle([x0 - WALL_THICK, ceil_y, x0 - 1, floor_y], fill=WALL)
    draw.rectangle([x1 + 1, ceil_y, x1 + WALL_THICK, floor_y], fill=WALL)
    draw.rectangle(
        [x0 - WALL_THICK, floor_y + 1, x1 + WALL_THICK, floor_y + WALL_THICK], fill=WALL
    )
    draw.rectangle([x0, floor_y - fill_h + 1, x1, floor_y], fill=fill_rgb)


def render(scene: dict) -> Image.Image:
    """Deterministic raster of one scene."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rectangle([40, DIVIDER_Y, WIDTH - 41, DIVIDER_Y + 1], fill=DIVIDER)

    for slot in range(4):
        _draw_container(
            draw,
            SLOT_CENTRES[slot],
            int(scene["widths"][slot]),
            TOP_FLOOR_Y,
            TOP_CEIL_Y,
            int(scene["heights"][slot]),
            COLOUR_RGB[str(scene["colours"][slot])],
        )

    tube_h = _tube_heights(scene)
    tw = int(scene["tube_width"])
    for slot in range(4):
        vessel = int(scene["tube_order"][slot])
        _draw_container(
            draw,
            SLOT_CENTRES[slot],
            tw,
            BOTTOM_FLOOR_Y,
            BOTTOM_CEIL_Y,
            tube_h[vessel],
            GRAY_FILL,
        )

    cx = SLOT_CENTRES[int(scene["marked_slot"])]
    draw.polygon(
        [(cx - 13, BOTTOM_CEIL_Y - 20), (cx + 13, BOTTOM_CEIL_Y - 20), (cx, BOTTOM_CEIL_Y - 4)],
        fill=MARKER,
    )
    return image
