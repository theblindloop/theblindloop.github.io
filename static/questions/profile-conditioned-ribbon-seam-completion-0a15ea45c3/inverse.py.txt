"""Independent final-PNG inverse arm for two-seam ribbon correspondence."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = np.array(
    [
        (220, 72, 72),
        (31, 153, 145),
        (118, 83, 176),
    ],
    dtype=np.float32,
)


class OracleAbstention(ValueError):
    pass


def _color_masks(image: Image.Image) -> list[np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.sqrt(((array[:, :, None, :] - _PALETTE[None, None, :, :]) ** 2).sum(axis=3))
    nearest = distances.argmin(axis=2)
    minimum = distances.min(axis=2)
    # Neutral antialiased text/borders can lie moderately near violet in RGB
    # distance, so require the visible ribbon inks' strong channel spread too.
    chroma = array.max(axis=2) - array.min(axis=2)
    return [(nearest == index) & (minimum < 82.0) & (chroma > 55.0) for index in range(3)]


def _largest_gap_split(values: np.ndarray, minimum_gap: int) -> float:
    unique = np.unique(values)
    if len(unique) < 2:
        raise OracleAbstention("insufficient colored support")
    gaps = np.diff(unique)
    index = int(gaps.argmax())
    if gaps[index] < minimum_gap:
        raise OracleAbstention("no visible structural gap")
    return float(unique[index] + unique[index + 1]) / 2.0


def _runs_from_values(values: np.ndarray, minimum_gap: int) -> list[tuple[int, int]]:
    unique = np.unique(values)
    if not len(unique):
        return []
    cuts = np.flatnonzero(np.diff(unique) >= minimum_gap)
    starts = [0, *(cuts + 1).tolist()]
    ends = [*cuts.tolist(), len(unique) - 1]
    return [(int(unique[start]), int(unique[end])) for start, end in zip(starts, ends)]


def _endpoint_y(mask: np.ndarray, region: np.ndarray, side: str) -> float:
    ys, xs = np.where(mask & region)
    if len(xs) < 20:
        raise OracleAbstention("colored endpoint missing")
    edge = xs.min() if side == "left" else xs.max()
    close = np.abs(xs - edge) <= 9
    if close.sum() < 8:
        raise OracleAbstention("endpoint support too small")
    return float(np.median(ys[close]))


def _ranks(values: list[float]) -> list[int]:
    order = np.argsort(np.asarray(values))
    sorted_values = np.asarray(values)[order]
    if np.min(np.diff(sorted_values)) < 20.0:
        raise OracleAbstention("seam endpoints are too close")
    ranks = np.empty(len(values), dtype=int)
    ranks[order] = np.arange(len(values))
    return ranks.tolist()


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B/C solely from palette pixels and their inferred groups."""
    masks = _color_masks(image)
    colored = masks[0] | masks[1] | masks[2]
    all_y, all_x = np.where(colored)
    if len(all_x) < 500:
        raise OracleAbstention("not enough ribbon ink")
    horizontal_split = _largest_gap_split(all_y, 45)
    yy, xx = np.indices(colored.shape)
    top = yy < horizontal_split
    lower = yy > horizontal_split

    top_y, top_x = np.where(colored & top)
    vertical_split = _largest_gap_split(top_x, 80)
    top_left_region = top & (xx < vertical_split)
    top_right_region = top & (xx > vertical_split)
    reference_left = _ranks([_endpoint_y(mask, top_left_region, "right") for mask in masks])
    reference_right = _ranks([_endpoint_y(mask, top_right_region, "left") for mask in masks])

    lower_y, lower_x = np.where(colored & lower)
    option_runs = _runs_from_values(lower_x, 35)
    if len(option_runs) != 3:
        raise OracleAbstention(f"expected three candidate patches, found {len(option_runs)}")

    scores = []
    for x0, x1 in option_runs:
        option_region = lower & (xx >= x0) & (xx <= x1)
        left_ranks = _ranks([_endpoint_y(mask, option_region, "left") for mask in masks])
        right_ranks = _ranks([_endpoint_y(mask, option_region, "right") for mask in masks])
        score = sum(a == b for a, b in zip(left_ranks, reference_left))
        score += sum(a == b for a, b in zip(right_ranks, reference_right))
        scores.append(score)
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    ordered = sorted(scores, reverse=True)
    if len(winners) != 1 or best != 6 or best - ordered[1] < 2:
        raise OracleAbstention(f"ambiguous seam scores {scores}")
    return chr(ord("A") + winners[0])
