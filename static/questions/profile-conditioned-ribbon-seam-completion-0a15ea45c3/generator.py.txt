"""Deterministic latent sampler, raster renderer, and analytic arm."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 650
SCALE = 2
COLORS = {
    "coral": (220, 72, 72),
    "teal": (31, 153, 145),
    "violet": (118, 83, 176),
}
COLOR_NAMES = tuple(COLORS)
INK = (36, 43, 52)
BOARD = (238, 242, 246)
PATCH = (246, 248, 250)
BACKGROUND = (255, 253, 249)


def sample_scene(seed: int) -> dict:
    """Sample balanced visible orders and a randomized answer label."""
    rng = random.Random(int(seed))
    left_order = list(COLOR_NAMES)
    right_order = list(COLOR_NAMES)
    outer_left_order = [0, 1, 2]
    outer_right_order = [0, 1, 2]
    option_kinds = ["both", "left", "right"]
    rng.shuffle(left_order)
    rng.shuffle(right_order)
    rng.shuffle(outer_left_order)
    rng.shuffle(outer_right_order)
    rng.shuffle(option_kinds)
    return {
        "left_order": left_order,
        "right_order": right_order,
        "outer_left_order": outer_left_order,
        "outer_right_order": outer_right_order,
        "option_kinds": option_kinds,
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size * SCALE)


def _bezier(
    start: tuple[float, float],
    control1: tuple[float, float],
    control2: tuple[float, float],
    end: tuple[float, float],
    n: int = 80,
) -> list[tuple[float, float]]:
    points = []
    for index in range(n + 1):
        t = index / n
        u = 1.0 - t
        x = u**3 * start[0] + 3 * u * u * t * control1[0] + 3 * u * t * t * control2[0] + t**3 * end[0]
        y = u**3 * start[1] + 3 * u * u * t * control1[1] + 3 * u * t * t * control2[1] + t**3 * end[1]
        points.append((x * SCALE, y * SCALE))
    return points


def _draw_curve(
    draw: ImageDraw.ImageDraw,
    start: tuple[float, float],
    end: tuple[float, float],
    color: tuple[int, int, int],
    horizontal_run: float,
) -> None:
    c1 = (start[0] + horizontal_run, start[1])
    c2 = (end[0] - horizontal_run, end[1])
    draw.line(_bezier(start, c1, c2, end), fill=color, width=10 * SCALE, joint="curve")


def _position_by_color(order: Iterable[str]) -> dict[str, int]:
    return {color: rank for rank, color in enumerate(order)}


def render(scene: dict) -> Image.Image:
    """Render the complete public scene to a final RGB raster."""
    left_order = list(scene["left_order"])
    right_order = list(scene["right_order"])
    outer_left_order = list(scene["outer_left_order"])
    outer_right_order = list(scene["outer_right_order"])
    option_kinds = list(scene["option_kinds"])

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((450 * SCALE, 25 * SCALE), "RIBBON SEAM COMPLETION", fill=INK, font=_font(24, True), anchor="ma")
    draw.text((62 * SCALE, 62 * SCALE), "REFERENCE WITH MISSING WINDOW", fill=INK, font=_font(14, True), anchor="la")

    board_box = (60 * SCALE, 82 * SCALE, 840 * SCALE, 330 * SCALE)
    draw.rounded_rectangle(board_box, radius=10 * SCALE, fill=BOARD, outline=INK, width=3 * SCALE)
    window_box = (360 * SCALE, 112 * SCALE, 540 * SCALE, 300 * SCALE)
    draw.rectangle(window_box, fill=BACKGROUND, outline=INK, width=3 * SCALE)
    draw.text((450 * SCALE, 196 * SCALE), "?", fill=(151, 158, 168), font=_font(42, True), anchor="mm")

    seam_levels = [145, 206, 267]
    outer_levels = [135, 206, 277]
    left_rank = _position_by_color(left_order)
    right_rank = _position_by_color(right_order)
    for color_index, color_name in enumerate(COLOR_NAMES):
        left_outer = outer_levels[outer_left_order[color_index]]
        left_seam = seam_levels[left_rank[color_name]]
        _draw_curve(draw, (88, left_outer), (360, left_seam), COLORS[color_name], 92)
        right_seam = seam_levels[right_rank[color_name]]
        right_outer = outer_levels[outer_right_order[color_index]]
        _draw_curve(draw, (540, right_seam), (812, right_outer), COLORS[color_name], 92)

    draw.text((62 * SCALE, 370 * SCALE), "CANDIDATE PATCHES", fill=INK, font=_font(14, True), anchor="la")
    patch_x = [70, 355, 640]
    patch_y0, patch_y1 = 430, 612
    local_levels = [458, 521, 584]
    deranged_left = left_order[1:] + left_order[:1]
    deranged_right = right_order[1:] + right_order[:1]
    for option_index, (x0, kind) in enumerate(zip(patch_x, option_kinds)):
        x1 = x0 + 190
        label = chr(ord("A") + option_index)
        draw.text(((x0 + x1) / 2 * SCALE, 401 * SCALE), label, fill=INK, font=_font(24, True), anchor="mm")
        draw.rounded_rectangle(
            (x0 * SCALE, patch_y0 * SCALE, x1 * SCALE, patch_y1 * SCALE),
            radius=7 * SCALE,
            fill=PATCH,
            outline=INK,
            width=3 * SCALE,
        )
        option_left = left_order if kind in {"both", "left"} else deranged_left
        option_right = right_order if kind in {"both", "right"} else deranged_right
        option_left_rank = _position_by_color(option_left)
        option_right_rank = _position_by_color(option_right)
        for color_name in COLOR_NAMES:
            y_left = local_levels[option_left_rank[color_name]]
            y_right = local_levels[option_right_rank[color_name]]
            _draw_curve(draw, (x0, y_left), (x1, y_right), COLORS[color_name], 58)

    draw.text(
        (450 * SCALE, 632 * SCALE),
        "Match color and height at BOTH vertical seams.",
        fill=INK,
        font=_font(13),
        anchor="ms",
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return chr(ord("A") + list(scene["option_kinds"]).index("both"))


def margin(scene: dict) -> float:
    """Best minus runner-up count over six seam endpoint agreements."""
    kinds = list(scene["option_kinds"])
    scores = [6.0 if kind == "both" else 3.0 for kind in kinds]
    scores.sort(reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
