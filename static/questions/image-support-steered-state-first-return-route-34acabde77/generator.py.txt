"""Renderer and analytic arm for a connected state-route recurrence question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-first-return-route-0.1.0"
CANVAS = (640, 480)
STATION_COUNT = 11
STATE_LABELS = ("A", "B", "C", "D")
PALETTE = {
    "background": (249, 251, 253),
    "route": (36, 61, 84),
    "operation": (218, 96, 57),
    "state_a": (191, 69, 68),
    "state_b": (41, 149, 94),
    "state_c": (65, 105, 190),
    "state_d": (132, 83, 169),
    "neutral": (169, 177, 187),
    "text": (29, 40, 52),
}


def _colors(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    return {
        name: tuple(int(channel) for channel in value)
        for name, value in scene["palette"].items()
    }


def _trace(initial_state: int, operations: list[int]) -> list[int]:
    state = int(initial_state) % 4
    trace = [state]
    for operation in operations:
        state = (state + int(operation)) % 4
        trace.append(state)
    return trace


def _first_return(initial_state: int, operations: list[int]) -> int | None:
    trace = _trace(initial_state, operations)
    departed = False
    for station, state in enumerate(trace[1:], start=1):
        if state != trace[0]:
            departed = True
        elif departed:
            return station
    return None


def _route_points(rng: random.Random) -> list[list[int]]:
    """Keep the route inside a medium horizontal band and away from the canvas edge."""

    xs = [140, 176, 212, 248, 284, 320, 356, 392, 428, 464, 500, 536, 572]
    base_y = [270, 246, 292, 250, 286, 244, 290, 252, 286, 246, 292, 254, 270]
    # A phase changes the y placement without changing the regional route footprint.
    phase = rng.randrange(3)
    offsets = (-7, 0, 7)
    return [[x, y + offsets[phase]] for x, y in zip(xs, base_y, strict=True)]


def _distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(48, 58), (592, 60), (50, 418), (590, 416), (82, 120), (558, 118)]
    rng.shuffle(anchors)
    return [
        {
            "kind": index % 2,
            "x": x + rng.randint(-7, 7),
            "y": y + rng.randint(-7, 7),
            "size": rng.randint(8, 12),
        }
        for index, (x, y) in enumerate(anchors[:4])
    ]


def _sample_operations(rng: random.Random, target: int) -> list[int]:
    """Construct a random walk with its first return at the requested station."""

    for _ in range(1000):
        initial = rng.randrange(4)
        operations: list[int] = [rng.choice((-1, 1))]
        state = (initial + operations[0]) % 4
        # Avoid returning to the initial state before the target while allowing
        # varied intermediate excursions through the other three states.
        for _station in range(2, target):
            allowed = [step for step in (-1, 0, 1) if (state + step) % 4 != initial]
            step = rng.choice(allowed)
            operations.append(step)
            state = (state + step) % 4
        if state not in ((initial + 1) % 4, (initial - 1) % 4):
            continue
        operations.append(-1 if state == (initial + 1) % 4 else 1)
        operations.extend(rng.choice((-1, 0, 1)) for _ in range(STATION_COUNT - len(operations)))
        if _first_return(initial, operations) == target:
            # The initial state is sampled separately from the operation walk;
            # return it through a private attribute-free pair below.
            return [initial, *operations]
    raise RuntimeError(f"could not construct first return at station {target}")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a deterministic, balanced scene with a first return in stations 2..11."""

    rng = random.Random(int(seed))
    target = 2 + (int(seed) % 10)
    encoded = _sample_operations(rng, target)
    initial_state = int(encoded[0])
    operations = [int(value) for value in encoded[1:]]
    return {
        "initial_state": initial_state,
        "operations": operations,
        "route_points": _route_points(rng),
        "distractors": _distractors(rng),
        "palette": {name: list(value) for name, value in PALETTE.items()},
        "route_width": 7,
        "marker_size": 12,
        "canvas": list(CANVAS),
    }


def _draw_operation(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    operation: int,
    radius: int,
    color: tuple[int, int, int],
) -> None:
    x, y = center
    if operation == 1:
        draw.polygon(
            [(x - radius, y - radius), (x - radius, y + radius), (x + radius, y)],
            fill=color,
        )
    elif operation == -1:
        draw.polygon(
            [(x + radius, y - radius), (x + radius, y + radius), (x - radius, y)],
            fill=color,
        )
    else:
        draw.rectangle(
            (x - radius, y - radius, x + radius, y + radius),
            outline=color,
            width=4,
        )


def _state_color(state: int, colors: dict[str, tuple[int, int, int]]) -> tuple[int, int, int]:
    names = ("state_a", "state_b", "state_c", "state_d")
    return colors[names[int(state) % 4]]


def _draw_state_badge(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    state: int,
    radius: int,
    colors: dict[str, tuple[int, int, int]],
    font: ImageFont.ImageFont,
) -> None:
    x, y = center
    draw.ellipse(
        (x - radius, y - radius, x + radius, y + radius),
        fill=_state_color(state, colors),
        outline=colors["route"],
        width=3,
    )
    label = STATE_LABELS[int(state) % 4]
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((x - tw / 2, y - th / 2 - 1), label, fill=colors["background"], font=font)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one connected route with all answer-bearing fields visible."""

    width, height = (int(value) for value in scene["canvas"])
    colors = _colors(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)
        else:
            draw.rectangle((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)

    points = [tuple(int(value) for value in point) for point in scene["route_points"]]
    draw.line(points, fill=colors["route"], width=int(scene["route_width"]), joint="curve")
    marker_radius = int(scene["marker_size"])
    for index, (point, operation) in enumerate(zip(points[1:-1], scene["operations"], strict=True), start=1):
        center = [int(point[0]), int(point[1])]
        _draw_operation(draw, center, int(operation), marker_radius, colors["operation"])
        label = str(index)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw = bbox[2] - bbox[0]
        draw.text((center[0] - tw / 2, center[1] - marker_radius - 18), label, fill=colors["text"], font=font)

    _draw_state_badge(draw, [int(value) for value in points[0]], int(scene["initial_state"]), marker_radius + 8, colors, font)
    end_x, end_y = points[-1]
    draw.ellipse((end_x - 10, end_y - 10, end_x + 10, end_y + 10), outline=colors["neutral"], width=3)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the first later station that revisits the marked initial state."""

    answer = _first_return(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    if answer is None:
        raise ValueError("scene has no later return to its starting state")
    return str(answer)


def margin(scene: dict[str, Any]) -> float:
    """A one-sided evidence margin: number of stations before the first return."""

    answer = int(analytic_gold(scene))
    return round(float(answer - 1), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    if len(scene.get("operations", [])) != STATION_COUNT:
        return True
    if len(scene.get("route_points", [])) != STATION_COUNT + 2:
        return True
    try:
        answer = int(analytic_gold(scene))
    except (TypeError, ValueError):
        return True
    return answer < 2 or answer > STATION_COUNT or margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # All fields that determine the label are directly rendered; no pixel-identical
    # latent alias is claimed for this parametrization.
    return []
