"""Deterministic renderer for visible outward-point counting."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SUPERSAMPLE = 4

PALETTES = (
    ((34, 91, 168), (247, 242, 226)),
    ((176, 55, 76), (238, 246, 242)),
    ((39, 126, 105), (248, 237, 231)),
    ((111, 67, 156), (245, 244, 225)),
    ((187, 104, 28), (232, 243, 249)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a matched control/counterfactual scene from a numeric seed."""
    rng = random.Random(int(seed))
    foreground, background = PALETTES[rng.randrange(len(PALETTES))]
    # The answer cycles independently of the nuisance draws and includes the
    # familiar five-point control as well as four counterfactual identities.
    point_count = 4 + (int(seed) % 5)
    return {
        "point_count": point_count,
        "rotation": rng.uniform(-math.pi, math.pi),
        "outer_radius": rng.uniform(230.0, 235.0),
        "inner_ratio": rng.uniform(0.55, 0.62),
        "center_x": rng.uniform(254.0, 258.0),
        "center_y": rng.uniform(254.0, 258.0),
        "foreground": list(foreground),
        "background": list(background),
        "width": WIDTH,
        "height": HEIGHT,
    }


def _vertices(scene: dict[str, Any], scale: float = 1.0) -> list[tuple[float, float]]:
    count = int(scene["point_count"])
    rotation = float(scene["rotation"])
    outer = float(scene["outer_radius"])
    inner = outer * float(scene["inner_ratio"])
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    vertices = []
    for index in range(2 * count):
        radius = outer if index % 2 == 0 else inner
        angle = rotation + index * math.pi / count
        vertices.append(((cx + radius * math.cos(angle)) * scale,
                         (cy + radius * math.sin(angle)) * scale))
    return vertices


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one connected, high-fill radial polygon on a plain field."""
    width, height = int(scene["width"]), int(scene["height"])
    foreground = tuple(int(v) for v in scene["foreground"])
    background = tuple(int(v) for v in scene["background"])
    large = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(large)
    draw.polygon(_vertices(scene, SUPERSAMPLE), fill=foreground)
    return large.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["point_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Minimum radial prominence of each visible outer tip in canvas pixels."""
    return float(scene["outer_radius"]) * (1.0 - float(scene["inner_ratio"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 84.0 or float(scene["outer_radius"]) < 225.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # Pose is represented canonically rather than modulo rotational aliases.
    return []
