"""Pixel-only inverse for a single filled radial polygon.

This module intentionally knows neither the renderer nor any scene fields.  It
segments the dominant region and counts prominent maxima in its radial contour.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    output = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        output[np.asarray(ys), np.asarray(xs)] = True
    return output


def _smooth_circular(values: np.ndarray, radius: int) -> np.ndarray:
    extended = np.concatenate((values[-radius:], values, values[:radius]))
    kernel = np.ones(2 * radius + 1, dtype=np.float64) / (2 * radius + 1)
    return np.convolve(extended, kernel, mode="valid")


def _has_hard_erase_edge(array: np.ndarray, background: np.ndarray) -> bool:
    """Recognize long non-antialiased cuts through otherwise smooth artwork."""
    comparisons = ((array[:, :-1], array[:, 1:], True),
                   (array[:-1, :], array[1:, :], False))
    for first, second, vertical in comparisons:
        first_bg = np.linalg.norm(first - background, axis=2)
        second_bg = np.linalg.norm(second - background, axis=2)
        jump = np.linalg.norm(first - second, axis=2)
        hard = (jump > 100.0) & (((first_bg < 3.0) & (second_bg > 80.0)) |
                                 ((second_bg < 3.0) & (first_bg > 80.0)))
        lines = hard.T if vertical else hard
        for line in lines:
            run = 0
            for value in line:
                run = run + 1 if value else 0
                if run >= 12:
                    return True
    return False


def decision_from_image(image: Image.Image) -> str:
    """Return the visible number of sharp outward points, or abstain by error."""
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or min(array.shape[:2]) < 128:
        raise ValueError("image is too small")
    border = np.concatenate((array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
                             array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    if _has_hard_erase_edge(array, background):
        raise ValueError("contour contains a hard rectangular erasure edge")
    distance = np.linalg.norm(array - background, axis=2)
    component = _largest_component(distance > 45.0)
    area = int(component.sum())
    height, width = component.shape
    if not (0.16 * height * width < area < 0.58 * height * width):
        raise ValueError("no valid dominant filled region")
    ys, xs = np.nonzero(component)
    box_w, box_h = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
    if min(box_w, box_h) < 320 or max(box_w, box_h) > min(height, width) - 20:
        raise ValueError("dominant region has invalid extent")
    # The area centroid is stable under palette and antialiasing changes.
    cx, cy = float(xs.mean()), float(ys.mean())
    dx, dy = xs.astype(np.float64) - cx, ys.astype(np.float64) - cy
    radii = np.hypot(dx, dy)
    angles = np.mod(np.arctan2(dy, dx), 2.0 * np.pi)
    bins = 1440
    indices = np.floor(angles * bins / (2.0 * np.pi)).astype(int) % bins
    profile = np.zeros(bins, dtype=np.float64)
    np.maximum.at(profile, indices, radii)
    # Fill rare empty angular bins, then smooth only raster-scale noise.
    for _ in range(4):
        empty = profile == 0
        if not empty.any():
            break
        profile[empty] = (np.roll(profile, 1)[empty] + np.roll(profile, -1)[empty]) / 2.0
    # A valid badge star is filled continuously from its centre to its boundary
    # at every angle. Missing interior ink is therefore evidence erasure, even
    # when the remaining outer tips happen to retain the same count.
    grid_y, grid_x = np.indices(component.shape)
    grid_dx, grid_dy = grid_x - cx, grid_y - cy
    grid_r = np.hypot(grid_dx, grid_dy)
    grid_a = np.mod(np.arctan2(grid_dy, grid_dx), 2.0 * np.pi)
    grid_i = np.floor(grid_a * bins / (2.0 * np.pi)).astype(int) % bins
    expected_interior = (grid_r > 5.0) & (grid_r < profile[grid_i] - 7.0)
    missing_interior = int(np.count_nonzero(expected_interior & ~component))
    if missing_interior > max(80, int(area * 0.002)):
        raise ValueError("filled radial region has erased interior support")
    profile = _smooth_circular(profile, 5)
    # Identify local maxima, consolidate each peak neighbourhood, and require
    # enough prominence over valleys on both sides to count as an outward tip.
    raw = np.flatnonzero((profile > np.roll(profile, 1)) & (profile >= np.roll(profile, -1)))
    candidates: list[tuple[float, int]] = []
    window = 85  # under half the separation even for the eight-point case
    for index in raw:
        left = profile[(index - window) % bins]
        right = profile[(index + window) % bins]
        prominence = float(profile[index] - max(left, right))
        if prominence > 24.0:
            candidates.append((prominence, int(index)))
    selected: list[tuple[float, int]] = []
    for prominence, index in sorted(candidates, reverse=True):
        if all(min((index - prior) % bins, (prior - index) % bins) >= 105 for _, prior in selected):
            selected.append((prominence, index))
    count = len(selected)
    if count not in range(4, 9):
        raise ValueError("ambiguous radial peak count")
    strengths = sorted((value for value, _ in selected), reverse=True)
    if min(strengths) < 30.0:
        raise ValueError("weak outer-point prominence")
    return str(count)
