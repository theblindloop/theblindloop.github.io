"""Independent pixel-only inverse for event-splice intervention search."""
from __future__ import annotations
import numpy as np
from PIL import Image

CANVAS=(1040,700); BOARD_W,BOARD_H=1000,660; BORDER=(55,64,72); CANCEL=(204,43,55); INK=(37,43,49)
REGISTERS=("A","B","C","D"); OPTIONS=("amber","cyan","violet")
BADGES={(198,82,52):"A",(22,139,150):"B",(121,75,180):"C",(65,104,174):"D"}
OPTION_RGB={(232,145,23):"amber",(13,151,184):"cyan",(139,72,181):"violet"}
SHAPES_BY_RGB={(42,124,202):"circle",(38,157,91):"triangle",(230,158,26):"star",(205,69,119):"diamond"}
SHAPES=("circle","triangle","star","diamond"); PAIRS=tuple((a,b) for i,a in enumerate(REGISTERS) for b in REGISTERS[i+1:])
class OracleError(ValueError): pass

def _patch(a,x,y,r=2):
    p=a[y-r:y+r+1,x-r:x+r+1]; colors={tuple(int(v) for v in row) for row in p.reshape(-1,3)}
    if len(colors)!=1: raise OracleError("a required marker is damaged")
    return next(iter(colors))
def _shape(a,x,y):
    c=_patch(a,x,y)
    if c not in SHAPES_BY_RGB: raise OracleError("unknown shape")
    return SHAPES_BY_RGB[c]
def _badge(a,x,y):
    c=_patch(a,x-18,y)
    if c not in BADGES: raise OracleError("unknown badge")
    return BADGES[c]
def _dependencies(pairs):
    last={}; deps=[]
    for i,pair in enumerate(pairs):
        deps.append({last[s] for s in pair if s in last})
        for s in pair: last[s]=i
    return deps
def _removed(pairs,cancel):
    deps=_dependencies(pairs); removed={cancel}; depth={cancel:0}
    for i in range(cancel+1,len(pairs)):
        hit=deps[i]&removed
        if hit: removed.add(i); depth[i]=1+max(depth[p] for p in hit)
    return removed,depth
def _run(initial,pairs,skipped):
    state=list(initial)
    for i,pair in enumerate(pairs):
        if i in skipped: continue
        x,y=(REGISTERS.index(s) for s in pair); state[x],state[y]=state[y],state[x]
    return state
def _world(initial,pairs,patch,position,cancel,target):
    augmented=list(pairs[:position])+[patch]+list(pairs[position:]); cancel_aug=cancel+(1 if position<=cancel else 0); removed,depth=_removed(augmented,cancel_aug)
    outcome=_run(initial,augmented,removed)[REGISTERS.index(target)]
    removed_original={i for i in range(len(pairs)) if i+(1 if i>=position else 0) in removed}
    return outcome,(position in removed,tuple(sorted(removed_original))),depth

def decision_from_image(image:Image.Image)->str:
    a=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if (a.shape[1],a.shape[0])!=CANVAS: raise OracleError("unexpected canvas")
    mask=np.all(a==np.asarray(BORDER,dtype=np.uint8),axis=2); pts=np.argwhere(mask)
    if not len(pts): raise OracleError("board absent")
    top,left=pts.min(axis=0); bottom,right=pts.max(axis=0); ax,ay=int(left),int(top)
    if int(right-left+1)!=BOARD_W or int(bottom-top+1)!=BOARD_H: raise OracleError("wrong board extent")
    if not (np.all(mask[ay,ax:ax+BOARD_W]) and np.all(mask[ay+BOARD_H-1,ax:ax+BOARD_W]) and np.all(mask[ay:ay+BOARD_H,ax]) and np.all(mask[ay:ay+BOARD_H,ax+BOARD_W-1])): raise OracleError("broken border")
    initial=[_shape(a,ax+x+16,ay+98) for x in (145,285,425,565)]
    if sorted(initial)!=sorted(SHAPES): raise OracleError("bad start")
    target=_badge(a,ax+773,ay+94); goal=_shape(a,ax+930,ay+94); patch=(_badge(a,ax+450,ay+263),_badge(a,ax+550,ay+263))
    if patch not in PAIRS: raise OracleError("bad patch")
    pairs=[]; selectors=[]
    for i in range(7):
        x,y=ax+39+i*133,ay+337; selectors.append(_patch(a,x+43,y+28)); pair=(_badge(a,x+56,y+91),_badge(a,x+56,y+187))
        if pair not in PAIRS: raise OracleError("bad pair")
        pairs.append(pair)
    selected=[i for i,c in enumerate(selectors) if c==CANCEL]
    if len(selected)!=1 or any(c not in {CANCEL,INK} for c in selectors): raise OracleError("bad cancellation")
    cancel=selected[0]; located={}
    for pos in range(1,7):
        c=_patch(a,ax+29+133*pos,ay+324)
        if c in OPTION_RGB: located[OPTION_RGB[c]]=pos
        elif c!=(246,248,244): raise OracleError("damaged insertion gap")
    if set(located)!=set(OPTIONS): raise OracleError("bad insertion markers")
    positions=[located[c] for c in OPTIONS]
    if cancel not in range(2,5) or len(set(positions))!=3 or any(p not in range(1,7) for p in positions): raise OracleError("invalid gaps")
    worlds=[_world(initial,pairs,patch,p,cancel,target) for p in positions]; outcomes=[w[0] for w in worlds]; signatures=[w[1] for w in worlds]
    if len(set(outcomes))!=3 or len(set(signatures))!=3 or outcomes.count(goal)!=1: raise OracleError("nonunique result")
    if not all(max(w[2].values())>=2 and len(w[2])>=3 for w in worlds): raise OracleError("shallow cascade")
    baseline=_run(initial,pairs,_removed(pairs,cancel)[0])[REGISTERS.index(target)]; winner=outcomes.index(goal)
    if goal==baseline or signatures[winner][0] or not any(i>=positions[winner] for i in signatures[winner][1]): raise OracleError("shortcut")
    return OPTIONS[winner]
