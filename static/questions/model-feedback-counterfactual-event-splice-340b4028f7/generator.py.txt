"""Renderer for inserting one event before causal pruning and replay."""
from __future__ import annotations
import math, random
from PIL import Image, ImageDraw, ImageFont

CANVAS=(1040,700); BOARD_W,BOARD_H=1000,660
REGISTERS=("A","B","C","D"); SHAPES=("circle","triangle","star","diamond"); OPTIONS=("amber","cyan","violet")
PAIRS=tuple((a,b) for i,a in enumerate(REGISTERS) for b in REGISTERS[i+1:])
RGB={"white":(255,255,255),"board":(246,248,244),"card":(255,255,252),"ink":(37,43,49),"muted":(105,112,118),"border":(55,64,72),"cancel":(204,43,55),"target":(196,119,16),"patch":(24,137,85),"amber":(232,145,23),"cyan":(13,151,184),"violet":(139,72,181),"A":(198,82,52),"B":(22,139,150),"C":(121,75,180),"D":(65,104,174),"circle":(42,124,202),"triangle":(38,157,91),"star":(230,158,26),"diamond":(205,69,119)}
FONT="/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"; BOLD="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _dependencies(pairs):
    last={}; deps=[]
    for i,pair in enumerate(pairs):
        deps.append({last[s] for s in pair if s in last})
        for s in pair: last[s]=i
    return deps

def _removed(pairs,cancel):
    deps=_dependencies(pairs); removed={cancel}; depth={cancel:0}
    for i in range(cancel+1,len(pairs)):
        hit=deps[i]&removed
        if hit: removed.add(i); depth[i]=1+max(depth[p] for p in hit)
    return removed,depth

def _run(initial,pairs,skipped):
    state=list(initial)
    for i,pair in enumerate(pairs):
        if i in skipped: continue
        a,b=(REGISTERS.index(s) for s in pair); state[a],state[b]=state[b],state[a]
    return state

def _splice(pairs,patch,position,cancel):
    augmented=list(pairs[:position])+[patch]+list(pairs[position:]); cancel_aug=cancel+(1 if position<=cancel else 0)
    return augmented,cancel_aug

def _world(initial,pairs,patch,position,cancel,target):
    augmented,cancel_aug=_splice(pairs,patch,position,cancel); removed,depth=_removed(augmented,cancel_aug)
    outcome=_run(initial,augmented,removed)[REGISTERS.index(target)]
    removed_original={i for i in range(len(pairs)) if i+(1 if i>=position else 0) in removed}
    return outcome,(position in removed,tuple(sorted(removed_original))),depth

def sample_scene(seed:int)->dict:
    rng=random.Random((int(seed)^0xE71A5C3D)&0xffffffffffffffff); wanted=int(seed)%3
    for _ in range(500000):
        initial=list(SHAPES); rng.shuffle(initial); pairs=[list(rng.choice(PAIRS)) for _ in range(7)]
        patch=list(rng.choice(PAIRS)); cancel=rng.randrange(2,5); choices=list(range(1,7)); rng.shuffle(choices); positions=choices[:3]; target=rng.choice(REGISTERS)
        worlds=[_world(initial,pairs,patch,p,cancel,target) for p in positions]; outcomes=[w[0] for w in worlds]; signatures=[w[1] for w in worlds]
        if len(set(outcomes))!=3 or len(set(signatures))!=3: continue
        if not all(max(w[2].values())>=2 and len(w[2])>=3 for w in worlds): continue
        baseline=_run(initial,pairs,_removed(pairs,cancel)[0])[REGISTERS.index(target)]; goal=outcomes[wanted]
        if goal==baseline or signatures[wanted][0] or not any(i>=positions[wanted] for i in signatures[wanted][1]): continue
        return {"initial":initial,"pairs":pairs,"patch_pair":patch,"cancel_card":cancel,"insert_positions":positions,"target_slot":target,"goal_shape":goal,"anchor_x":rng.randint(18,22),"anchor_y":rng.randint(18,22)}
    raise RuntimeError("could not sample a unique causal event-splice scene")

def _validate(scene):
    initial=[str(v) for v in scene["initial"]]; pairs=[[str(v) for v in p] for p in scene["pairs"]]; patch=[str(v) for v in scene["patch_pair"]]
    cancel=int(scene["cancel_card"]); positions=[int(v) for v in scene["insert_positions"]]; target=str(scene["target_slot"]); goal=str(scene["goal_shape"])
    if sorted(initial)!=sorted(SHAPES) or len(pairs)!=7: raise ValueError("invalid start or card count")
    if any(len(p)!=2 or tuple(p) not in PAIRS for p in pairs+[patch]): raise ValueError("invalid swap")
    if cancel not in range(2,5) or target not in REGISTERS or goal not in SHAPES: raise ValueError("invalid selector")
    if len(positions)!=3 or len(set(positions))!=3 or any(p not in range(1,7) for p in positions): raise ValueError("invalid insertion gaps")
    worlds=[_world(initial,pairs,patch,p,cancel,target) for p in positions]; outcomes=[w[0] for w in worlds]; signatures=[w[1] for w in worlds]
    if len(set(outcomes))!=3 or len(set(signatures))!=3 or outcomes.count(goal)!=1: raise ValueError("nonunique splice result")
    if not all(max(w[2].values())>=2 and len(w[2])>=3 for w in worlds): raise ValueError("shallow cascade")
    baseline=_run(initial,pairs,_removed(pairs,cancel)[0])[REGISTERS.index(target)]; winner=outcomes.index(goal)
    if goal==baseline or signatures[winner][0] or not any(i>=positions[winner] for i in signatures[winner][1]): raise ValueError("shortcut")

def analytic_gold(scene):
    _validate(scene); outcomes=[_world(scene["initial"],scene["pairs"],scene["patch_pair"],p,int(scene["cancel_card"]),str(scene["target_slot"]))[0] for p in scene["insert_positions"]]
    return OPTIONS[outcomes.index(str(scene["goal_shape"]))]
def margin(scene):
    try: return 1.0 if analytic_gold(scene) in OPTIONS else 0.0
    except (KeyError,TypeError,ValueError): return 0.0
def is_quarantined(scene): return margin(scene)<1.0
def latent_symmetries(scene): return []
def _font(n,bold=False): return ImageFont.truetype(BOLD if bold else FONT,n)
def _text(d,xy,s,n,fill=None,bold=False,anchor="mm"): d.text(xy,s,font=_font(n,bold),fill=fill or RGB["ink"],anchor=anchor)
def _shape(d,c,k,scale=20):
    x,y=c; color=RGB[k]
    if k=="circle": d.ellipse((x-scale,y-scale,x+scale,y+scale),fill=color,outline=RGB["ink"],width=2)
    elif k=="triangle":
        p=((x,y-scale-2),(x-scale-2,y+scale),(x+scale+2,y+scale)); d.polygon(p,fill=color); d.line(p+(p[0],),fill=RGB["ink"],width=2)
    elif k=="diamond":
        p=((x,y-scale-2),(x-scale-2,y),(x,y+scale+2),(x+scale+2,y)); d.polygon(p,fill=color); d.line(p+(p[0],),fill=RGB["ink"],width=2)
    elif k=="star":
        p=[]
        for i in range(10):
            r=scale+2 if i%2==0 else scale*.45; a=-math.pi/2+i*math.pi/5; p.append((x+r*math.cos(a),y+r*math.sin(a)))
        d.polygon(p,fill=color); d.line(p+[p[0]],fill=RGB["ink"],width=2)
    else: raise ValueError(k)
def _badge(d,c,label,large=False):
    x,y=c; w,h=(52,38) if large else (42,31); d.rounded_rectangle((x-w//2,y-h//2,x+w//2,y+h//2),radius=6,fill=RGB[label],outline=RGB["ink"],width=2); _text(d,c,label,20 if large else 17,fill=RGB["white"],bold=True)

def render(scene):
    _validate(scene); ax,ay=int(scene["anchor_x"]),int(scene["anchor_y"]); im=Image.new("RGB",CANVAS,RGB["white"]); d=ImageDraw.Draw(im)
    d.rectangle((ax,ay,ax+BOARD_W-1,ay+BOARD_H-1),fill=RGB["board"],outline=RGB["border"],width=4)
    _text(d,(ax+500,ay+24),"INSERT ONE PATCH, THEN PRUNE",22,bold=True); _text(d,(ax+57,ay+58),"START",16,bold=True)
    for label,kind,x in zip(REGISTERS,scene["initial"],(145,285,425,565),strict=True):
        d.rounded_rectangle((ax+x-60,ay+43,ax+x+60,ay+130),radius=9,fill=RGB["card"],outline=RGB[label],width=4); _badge(d,(ax+x-36,ay+66),label); _shape(d,(ax+x+16,ay+98),str(kind),20)
    _text(d,(ax+715,ay+56),"TARGET",15,fill=RGB["target"],bold=True); _badge(d,(ax+773,ay+94),str(scene["target_slot"]),True); _text(d,(ax+865,ay+56),"GOAL",15,fill=RGB["target"],bold=True); _shape(d,(ax+930,ay+94),str(scene["goal_shape"]),23)
    d.rounded_rectangle((ax+34,ay+143,ax+966,ay+303),radius=10,fill=(255,252,238),outline=RGB["target"],width=3)
    _text(d,(ax+500,ay+165),"Try PATCH at one colored gap. Rebuild dependencies on that spliced row.",16,bold=True); _text(d,(ax+500,ay+194),"Each swap depends on the latest earlier swap touching either box.",15); _text(d,(ax+500,ay+223),"Cancel red X; repeatedly cancel every dependent swap.",16,fill=RGB["cancel"],bold=True)
    _text(d,(ax+365,ay+263),"PATCH",15,fill=RGB["patch"],bold=True); _badge(d,(ax+450,ay+263),str(scene["patch_pair"][0]),True); _text(d,(ax+500,ay+263),"↕",24,bold=True); _badge(d,(ax+550,ay+263),str(scene["patch_pair"][1]),True); _text(d,(ax+760,ay+263),"Replay survivors → GOAL in TARGET?",15,bold=True)
    gap_colors={p:OPTIONS[i] for i,p in enumerate(scene["insert_positions"])}; cancel=int(scene["cancel_card"])
    for i,pair in enumerate(scene["pairs"]):
        x,y=ax+39+i*133,ay+337; selected=i==cancel; outline=RGB["cancel"] if selected else RGB["border"]
        d.rounded_rectangle((x,y,x+112,y+282),radius=9,fill=RGB["card"],outline=outline,width=5 if selected else 3); d.ellipse((x+38,y+10,x+74,y+46),fill=RGB["cancel"] if selected else RGB["ink"]); _text(d,(x+56,y+28),str(i+1),17,fill=RGB["white"],bold=True)
        _badge(d,(x+56,y+91),str(pair[0]),True); _text(d,(x+56,y+139),"↕",27,bold=True); _badge(d,(x+56,y+187),str(pair[1]),True); _text(d,(x+56,y+246),"X" if selected else "SWAP",34 if selected else 13,fill=RGB["cancel"] if selected else RGB["muted"],bold=True)
    for pos,color in gap_colors.items():
        gx=ax+29+133*pos; d.ellipse((gx-9,ay+315,gx+9,ay+333),fill=RGB[color],outline=RGB["ink"],width=2); d.polygon(((gx-7,ay+333),(gx+7,ay+333),(gx,ay+341)),fill=RGB[color])
    return im
