"""Arena field-flip renderer (episode 14).

A single thick, connected, area-like arena spans the whole canvas. Its
perimeter outline colour declares an initial binary state (blue = OFF,
orange = ON). Scattered across the arena's interior at widely separated
locations are red toggle dots; each interior toggle flips the tracked state.
Red dots placed on the thin background margin outside the arena are decoys and
do nothing. The final state is the initial state XOR the parity of the interior
toggle count. The answer aggregates evidence from many distant parts of one
connected region.

The decisive evidence is one connected high-fill region spanning wide, so the
inverse oracle must read the whole arena (outline + interior toggles) to answer.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
RADIUS = 12

# Palette
BG = (255, 255, 255)
ARENA = (125, 125, 140)
OUT_OFF = (40, 80, 220)     # blue  -> initial OFF
OUT_ON = (235, 150, 30)     # orange-> initial ON
TOGGLE = (210, 40, 50)      # red

OUTLINE_WIDTH = 12


def _edge_pad(inset: float) -> int:
    return int(inset * CANVAS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    s0 = rng.randint(0, 1)
    inset = rng.uniform(0.05, 0.075)
    pad = _edge_pad(inset)
    a0 = pad
    a1 = CANVAS - pad
    inner = 72
    n_toggle = rng.randint(0, 4)
    n_decoy = rng.randint(0, 3)

    toggles: list[list[float]] = []
    attempts = 0
    while len(toggles) < n_toggle and attempts < 2000:
        attempts += 1
        x = rng.uniform(a0 + inner, a1 - inner)
        y = rng.uniform(a0 + inner, a1 - inner)
        if all(math.hypot(x - tx, y - ty) > 46 for tx, ty in toggles):
            toggles.append([round(x, 1), round(y, 1)])

    decoys: list[list[float]] = []
    attempts = 0
    # Decoys live on the background margin, clearly outside the arena (clearance
    # from the arena inner edge so inside/outside membership is unambiguous).
    inner_edge = pad - 34
    while len(decoys) < n_decoy and attempts < 2000:
        attempts += 1
        side = rng.randint(0, 3)
        t = rng.uniform(0.06, 0.94)
        if side == 0:
            x, y = t * CANVAS, inner_edge
        elif side == 1:
            x, y = t * CANVAS, CANVAS - inner_edge
        elif side == 2:
            x, y = inner_edge, t * CANVAS
        else:
            x, y = CANVAS - inner_edge, t * CANVAS
        # keep on canvas
        if not (RADIUS + 6 <= x <= CANVAS - RADIUS - 6 and RADIUS + 6 <= y <= CANVAS - RADIUS - 6):
            continue
        if all(math.hypot(x - dx, y - dy) > 40 for dx, dy in decoys):
            decoys.append([round(x, 1), round(y, 1)])

    return {
        "s0": s0,
        "toggles": toggles,
        "decoys": decoys,
        "inset": round(inset, 4),
        "canvas": CANVAS,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas"], scene["canvas"]), BG)
    draw = ImageDraw.Draw(img)
    pad = _edge_pad(scene["inset"])
    a0, a1 = pad, scene["canvas"] - pad
    draw.rectangle([a0, a0, a1, a1], fill=ARENA)
    outline = OUT_ON if scene["s0"] == 1 else OUT_OFF
    draw.rectangle([a0, a0, a1, a1], outline=outline, width=OUTLINE_WIDTH)
    for x, y in scene["toggles"]:
        draw.ellipse([x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS], fill=TOGGLE)
    for x, y in scene["decoys"]:
        draw.ellipse([x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS], fill=TOGGLE)
    return img


def _interior_toggle_count(scene: dict) -> int:
    return len(scene["toggles"])


def analytic_gold(scene: dict) -> str:
    final = (scene["s0"] + (_interior_toggle_count(scene) % 2)) % 2
    return "yes" if final == 1 else "no"


def margin(scene: dict) -> float:
    return float(10 + _interior_toggle_count(scene))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    return []
