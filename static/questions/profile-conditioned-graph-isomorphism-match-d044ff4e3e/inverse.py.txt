"""Independent pixel-only graph reconstruction and isomorphism decision."""

from __future__ import annotations

import itertools
import math

import numpy as np


def _components(mask):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]; seen[y, x] = True; pixels = []
        while stack:
            cy, cx = stack.pop(); pixels.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; stack.append((ny, nx))
        if len(pixels) >= 80:
            found.append((sum(x for y, x in pixels) / len(pixels),
                          sum(y for y, x in pixels) / len(pixels), len(pixels)))
    return found


def _edge_present(dark, a, b):
    ax, ay = a; bx, by = b
    hits = 0; total = 0
    for t in np.linspace(0.10, 0.90, 65):
        x = int(round(ax + t * (bx - ax))); y = int(round(ay + t * (by - ay)))
        y0, y1 = max(0, y - 2), min(dark.shape[0], y + 3)
        x0, x1 = max(0, x - 2), min(dark.shape[1], x + 3)
        hits += bool(dark[y0:y1, x0:x1].any()); total += 1
    return hits / total > 0.82


def _isomorphic(left, right):
    if len(left) != len(right):
        return False
    aa = {i: set() for i in range(8)}; bb = {i: set() for i in range(8)}
    for u, v in left: aa[u].add(v); aa[v].add(u)
    for u, v in right: bb[u].add(v); bb[v].add(u)
    if sorted(map(len, aa.values())) != sorted(map(len, bb.values())):
        return False
    order = sorted(range(8), key=lambda u: (-len(aa[u]), -sum(len(aa[v]) for v in aa[u])))
    mapping, used = {}, set()
    def search(k):
        if k == 8: return True
        u = order[k]
        for v in range(8):
            if v in used or len(aa[u]) != len(bb[v]): continue
            if any(((x in aa[u]) != (mapping[x] in bb[v])) for x in mapping): continue
            mapping[u] = v; used.add(v)
            if search(k + 1): return True
            used.remove(v); del mapping[u]
        return False
    return search(0)


def decision_from_image(image):
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    target = np.array([16, 185, 210], dtype=np.int16)
    cyan = np.max(np.abs(rgb - target), axis=2) <= 35
    nodes = sorted(_components(cyan))
    if len(nodes) != 32:
        raise ValueError(f"expected 32 visible stations, found {len(nodes)}")
    nodes.sort(key=lambda item: item[0])
    panels = [nodes[index * 8:(index + 1) * 8] for index in range(4)]
    if any(max(p[i][0] for i in range(8)) - min(p[i][0] for i in range(8)) > 210 for p in panels):
        raise ValueError("station panels are ambiguous")
    dark = (rgb[:, :, 0] < 105) & (rgb[:, :, 1] < 115) & (rgb[:, :, 2] < 125)
    graphs = []
    for panel in panels:
        points = [(x, y) for x, y, area in panel]
        edges = set()
        for u, v in itertools.combinations(range(8), 2):
            if _edge_present(dark, points[u], points[v]):
                edges.add((u, v))
        if len(edges) != 12:
            raise ValueError(f"expected 12 links in a panel, found {len(edges)}")
        graphs.append(edges)
    hits = [_isomorphic(graphs[0], graph) for graph in graphs[1:]]
    if sum(hits) != 1:
        raise ValueError("image does not expose a unique isomorphic candidate")
    return ("left", "center", "right")[hits.index(True)]
