"""Deterministic visual world for reversing a three-pair registry."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1200
BACKGROUND = (248, 250, 252)
INK = (35, 43, 54)
TARGET_GREEN = (22, 150, 82)
KEY_GRAY = (107, 114, 128)
PAIR_GRAY = (188, 194, 202)
TAG_RGB = {
    "amber": (217, 119, 6),
    "cyan": (8, 145, 178),
    "violet": (124, 58, 237),
}
TAGS = tuple(TAG_RGB)

# The four small registries sit in separated canvas corners. Each is isolated
# by more than two cells of the public 8x8 overlapping-ablation grid.
REFERENCE_CENTER = (240, 240)
PROPOSAL_CENTERS = ((960, 240), (240, 960), (960, 960))
CARD_HALF_W = 72
CARD_HALF_H = 84
ROW_OFFSETS = (-48, 0, 48)


def _permutation(rng: random.Random) -> list[int]:
    values = [0, 1, 2]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    """Sample a bijection, two full-inventory distractors, rows, and tags."""

    seed = int(seed)
    rng = random.Random(seed ^ 0x1A2B3C7D)
    reference_map = _permutation(rng)  # dark source -> green target
    inverse = [0, 0, 0]  # green target -> dark source
    for source, target in enumerate(reference_map):
        inverse[target] = source

    # Both distractors remain bijections and have exactly the same local ink.
    distractor_a = [inverse[(target + 1) % 3] for target in range(3)]
    distractor_b = [inverse[(target + 2) % 3] for target in range(3)]
    if rng.randrange(2):
        distractor_a, distractor_b = distractor_b, distractor_a

    correct_position = (seed // 3) % 3
    proposals: list[list[int] | None] = [None, None, None]
    proposals[correct_position] = inverse
    other_positions = [index for index in range(3) if index != correct_position]
    proposals[other_positions[0]] = distractor_a
    proposals[other_positions[1]] = distractor_b

    # Answer color and panel position vary independently across every 9 seeds.
    answer_tag = TAGS[seed % 3]
    proposal_tags: list[str | None] = [None, None, None]
    proposal_tags[correct_position] = answer_tag
    other_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(other_tags)
    proposal_tags[other_positions[0]] = other_tags[0]
    proposal_tags[other_positions[1]] = other_tags[1]

    return {
        "reference_map": reference_map,
        "proposals": proposals,
        "row_orders": {
            "reference": _permutation(rng),
            "proposals": [_permutation(rng) for _ in range(3)],
        },
        "proposal_tags": proposal_tags,
    }


def _draw_source(draw: ImageDraw.ImageDraw, code: int, x: int, y: int) -> None:
    if code == 0:
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=INK)
    elif code == 1:
        draw.rectangle((x - 13, y - 13, x + 13, y + 13), fill=INK)
    elif code == 2:
        draw.polygon(((x, y - 15), (x - 15, y + 13), (x + 15, y + 13)), fill=INK)
    else:
        raise ValueError("unknown source glyph")


def _draw_target(draw: ImageDraw.ImageDraw, code: int, x: int, y: int) -> None:
    if code == 0:
        draw.rounded_rectangle((x - 16, y - 6, x + 16, y + 6), radius=3, fill=TARGET_GREEN)
    elif code == 1:
        draw.rounded_rectangle((x - 6, y - 16, x + 6, y + 16), radius=3, fill=TARGET_GREEN)
    elif code == 2:
        draw.polygon(((x, y - 16), (x - 16, y), (x, y + 16), (x + 16, y)), fill=TARGET_GREEN)
    else:
        raise ValueError("unknown target glyph")


def _frame(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]) -> None:
    cx, cy = center
    draw.rectangle(
        (cx - CARD_HALF_W, cy - CARD_HALF_H, cx + CARD_HALF_W, cy + CARD_HALF_H),
        outline=color,
        width=5,
    )


def _pair_mark(draw: ImageDraw.ImageDraw, x: int, y: int) -> None:
    draw.line((x - 8, y, x + 8, y), fill=PAIR_GRAY, width=3)
    draw.ellipse((x - 2, y - 2, x + 2, y + 2), fill=PAIR_GRAY)


def _draw_reference(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = REFERENCE_CENTER
    _frame(draw, (cx, cy), KEY_GRAY)
    for row, source in enumerate(scene["row_orders"]["reference"]):
        y = cy + ROW_OFFSETS[row]
        _draw_source(draw, source, cx - 42, y)
        _pair_mark(draw, cx, y)
        _draw_target(draw, scene["reference_map"][source], cx + 42, y)


def _draw_proposal(scene: dict, draw: ImageDraw.ImageDraw, index: int) -> None:
    cx, cy = PROPOSAL_CENTERS[index]
    _frame(draw, (cx, cy), TAG_RGB[scene["proposal_tags"][index]])
    for row, target in enumerate(scene["row_orders"]["proposals"][index]):
        y = cy + ROW_OFFSETS[row]
        _draw_target(draw, target, cx - 42, y)
        _pair_mark(draw, cx, y)
        _draw_source(draw, scene["proposals"][index][target], cx + 42, y)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_reference(scene, draw)
    for index in range(3):
        _draw_proposal(scene, draw, index)
    return image


def _inverse_relation(scene: dict) -> list[int]:
    inverse = [0, 0, 0]
    for source, target in enumerate(scene["reference_map"]):
        inverse[target] = source
    return inverse


def analytic_gold(scene: dict) -> str:
    inverse = _inverse_relation(scene)
    matches = [
        index for index, proposal in enumerate(scene["proposals"])
        if list(proposal) == inverse
    ]
    if len(matches) != 1:
        raise ValueError("scene has no unique inverse proposal")
    return scene["proposal_tags"][matches[0]]


def margin(scene: dict) -> float:
    inverse = _inverse_relation(scene)
    distances = [
        sum(int(proposal[target] != inverse[target]) for target in range(3))
        for proposal in scene["proposals"]
        if list(proposal) != inverse
    ]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        reference = list(scene["reference_map"])
        proposals = [list(value) for value in scene["proposals"]]
        tags = list(scene["proposal_tags"])
        orders = scene["row_orders"]
        if sorted(reference) != [0, 1, 2]:
            return True
        if len(proposals) != 3 or any(sorted(value) != [0, 1, 2] for value in proposals):
            return True
        if sorted(tags) != sorted(TAGS):
            return True
        if sorted(orders["reference"]) != [0, 1, 2]:
            return True
        if len(orders["proposals"]) != 3 or any(
            sorted(value) != [0, 1, 2] for value in orders["proposals"]
        ):
            return True
        inverse = _inverse_relation(scene)
        if sum(list(proposal) == inverse for proposal in proposals) != 1:
            return True
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
