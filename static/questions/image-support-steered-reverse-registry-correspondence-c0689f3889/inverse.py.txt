"""Independent pixel-only inverse for Reverse Registry Correspondence.

Only the public palette and glyph vocabulary are encoded here. The routine
discovers all four frames, classifies local endpoint glyphs, reconstructs the
gray relation, reverses its typed endpoints, and compares every proposal.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

INK = np.array((35, 43, 54), dtype=np.int32)
TARGET_GREEN = np.array((22, 150, 82), dtype=np.int32)
KEY_GRAY = np.array((107, 114, 128), dtype=np.int32)
TAG_COLORS = {
    "amber": np.array((217, 119, 6), dtype=np.int32),
    "cyan": np.array((8, 145, 178), dtype=np.int32),
    "violet": np.array((124, 58, 237), dtype=np.int32),
}


@dataclass(frozen=True)
class Component:
    x0: int
    y0: int
    x1: int
    y1: int
    area: int
    cx: float
    cy: float

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1


class OracleAbstention(ValueError):
    """The raster does not expose one complete, unique visual decision."""


def _mask(array: np.ndarray, color: np.ndarray, tolerance: float = 8.0) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum_area: int = 1) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum_area:
            output.append(
                Component(
                    min(xs), min(ys), max(xs), max(ys), len(xs),
                    float(sum(xs) / len(xs)), float(sum(ys) / len(ys)),
                )
            )
    return output


def _shifted_components(
    mask: np.ndarray, minimum_area: int, *, offset_x: int = 0, offset_y: int = 0
) -> list[Component]:
    points = np.argwhere(mask)
    if not len(points):
        return []
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    local = _components(mask[y0 : y1 + 1, x0 : x1 + 1], minimum_area)
    return [
        Component(
            item.x0 + int(x0) + offset_x,
            item.y0 + int(y0) + offset_y,
            item.x1 + int(x0) + offset_x,
            item.y1 + int(y0) + offset_y,
            item.area,
            item.cx + int(x0) + offset_x,
            item.cy + int(y0) + offset_y,
        )
        for item in local
    ]


def _large_frame(array: np.ndarray, color: np.ndarray) -> Component:
    raw = _mask(array, color)
    candidates = [
        item for item in _shifted_components(raw, 400)
        if item.width > 100 and item.height > 120
    ]
    valid: list[Component] = []
    for item in candidates:
        crop = raw[item.y0 : item.y1 + 1, item.x0 : item.x1 + 1]
        if min(
            float(crop[:5].mean()), float(crop[-5:].mean()),
            float(crop[:, :5].mean()), float(crop[:, -5:].mean()),
        ) >= 0.96:
            valid.append(item)
    if len(valid) != 1:
        raise OracleAbstention(f"expected one intact frame of a palette color, found {len(valid)}")
    return valid[0]


def _glyph_components(
    array: np.ndarray, color: np.ndarray, frame: Component, minimum: int
) -> list[Component]:
    x0, y0 = frame.x0 + 8, frame.y0 + 8
    x1, y1 = frame.x1 - 7, frame.y1 - 7
    crop = array[y0:y1, x0:x1]
    return _shifted_components(_mask(crop, color), minimum, offset_x=x0, offset_y=y0)


def _source_code(item: Component) -> int:
    fill = item.area / float(item.width * item.height)
    if fill > 0.94:
        return 1
    if fill > 0.68:
        return 0
    if 0.42 < fill < 0.68:
        return 2
    raise OracleAbstention("ambiguous dark source glyph")


def _target_code(item: Component) -> int:
    ratio = item.width / float(item.height)
    if ratio > 1.70:
        return 0
    if ratio < 0.59:
        return 1
    fill = item.area / float(item.width * item.height)
    if 0.42 < fill < 0.68:
        return 2
    raise OracleAbstention("ambiguous green target glyph")


def _validate_bijection(mapping: dict[int, int]) -> dict[int, int]:
    if set(mapping) != {0, 1, 2} or set(mapping.values()) != {0, 1, 2}:
        raise OracleAbstention("registry is not a complete bijection")
    return mapping


def _parse_reference(array: np.ndarray, frame: Component) -> dict[int, int]:
    sources = sorted(_glyph_components(array, INK, frame, 120), key=lambda item: item.cy)
    targets = sorted(_glyph_components(array, TARGET_GREEN, frame, 100), key=lambda item: item.cy)
    if len(sources) != 3 or len(targets) != 3:
        raise OracleAbstention("gray registry lacks three complete rows")
    mapping: dict[int, int] = {}
    for source, target in zip(sources, targets, strict=True):
        if abs(source.cy - target.cy) > 7.0 or source.cx >= target.cx:
            raise OracleAbstention("gray registry endpoint roles or rows are unclear")
        code = _source_code(source)
        if code in mapping:
            raise OracleAbstention("gray registry repeats a source")
        mapping[code] = _target_code(target)
    return _validate_bijection(mapping)


def _parse_proposal(array: np.ndarray, frame: Component) -> dict[int, int]:
    targets = sorted(_glyph_components(array, TARGET_GREEN, frame, 100), key=lambda item: item.cy)
    sources = sorted(_glyph_components(array, INK, frame, 120), key=lambda item: item.cy)
    if len(sources) != 3 or len(targets) != 3:
        raise OracleAbstention("colored registry lacks three complete rows")
    mapping: dict[int, int] = {}
    for target, source in zip(targets, sources, strict=True):
        if abs(source.cy - target.cy) > 7.0 or target.cx >= source.cx:
            raise OracleAbstention("colored registry endpoint roles or rows are unclear")
        code = _target_code(target)
        if code in mapping:
            raise OracleAbstention("colored registry repeats a target")
        mapping[code] = _source_code(source)
    return _validate_bijection(mapping)


def decision_from_image(image: Image.Image) -> str:
    """Return the matched frame color using pixels alone, or abstain."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0).astype(np.int32)
    delta = array.astype(np.int32) - background
    foreground = np.sum(delta * delta, axis=2) > 12 * 12
    points = np.argwhere(foreground)
    if not len(points):
        raise OracleAbstention("image has no visible registries")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    pad = 5
    y0, x0 = max(0, int(y0) - pad), max(0, int(x0) - pad)
    y1 = min(array.shape[0], int(y1) + pad + 1)
    x1 = min(array.shape[1], int(x1) + pad + 1)
    array = array[y0:y1, x0:x1]

    reference = _parse_reference(array, _large_frame(array, KEY_GRAY))
    inverse = {target: source for source, target in reference.items()}
    proposals = {
        tag: _parse_proposal(array, _large_frame(array, color))
        for tag, color in TAG_COLORS.items()
    }
    matches = [tag for tag, proposal in proposals.items() if proposal == inverse]
    if len(matches) != 1:
        raise OracleAbstention("inverse relation has no unique proposal")
    return matches[0]
