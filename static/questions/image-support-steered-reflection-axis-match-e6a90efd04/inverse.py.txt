"""Independent PNG-only inverse arm for reflection_axis_match."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image


def _colorfulness(rgb: tuple[int, int, int]) -> int:
    return max(rgb) - min(rgb)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    colors = Counter(map(tuple, array.reshape(-1, 3).tolist()))
    specimen_choices = [
        (count, color)
        for color, count in colors.items()
        if count > 300 and _colorfulness(color) > 45
    ]
    if not specimen_choices:
        return "abstain"
    _, specimen = max(specimen_choices)
    mask = np.all(array == np.asarray(specimen, dtype=np.uint8), axis=2)

    gold = (
        (array[:, :, 0] > 200)
        & (array[:, :, 1] > 125)
        & (array[:, :, 1] < 210)
        & (array[:, :, 2] < 70)
    )
    gy, gx = np.nonzero(gold)
    if len(gx) < 20:
        return "abstain"
    cx = int(round(float(gx.mean())))
    cy = int(round(float(gy.mean())))

    ys, xs = np.nonzero(mask)
    if len(xs) < 300:
        return "abstain"
    radius = int(max(cx - xs.min(), xs.max() - cx, cy - ys.min(), ys.max() - cy))
    if radius < 10 or cx - radius < 0 or cy - radius < 0:
        return "abstain"
    if cx + radius >= mask.shape[1] or cy + radius >= mask.shape[0]:
        return "abstain"
    local = mask[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
    left_right_mismatch = int(np.count_nonzero(local != np.fliplr(local)))
    top_bottom_mismatch = int(np.count_nonzero(local != np.flipud(local)))
    difference = abs(left_right_mismatch - top_bottom_mismatch)
    if difference < 100:
        return "abstain"
    return "left-right" if left_right_mismatch < top_bottom_mismatch else "top-bottom"
