"""Latent sampler, renderer, and analytic arm for reflection_axis_match."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw


PALETTES = (
    ((18, 136, 156), (242, 178, 24)),
    ((31, 142, 102), (239, 167, 22)),
    ((48, 112, 181), (245, 184, 28)),
    ((132, 76, 181), (238, 171, 18)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    specimen, marker = PALETTES[rng.randrange(len(PALETTES))]
    # Keep the compact cue near, but not fixed to, one of several safe raster-grid
    # intersections.  The small jitter preserves translation variation while
    # ensuring finite-resolution erasures sample a two-dimensional neighborhood.
    center_x = rng.choice((192, 256, 320)) + rng.randint(-6, 6)
    center_y = rng.choice((192, 256, 320)) + rng.randint(-6, 6)
    core_half = rng.randint(29, 35)
    extension = rng.randint(19, 28)
    half_span = rng.randint(12, core_half - 8)
    axis = "left-right" if int(seed) % 2 == 0 else "top-bottom"
    side = -1 if rng.randrange(2) == 0 else 1

    distractors = []
    used_quadrants = rng.sample(range(4), 3)
    for quadrant in used_quadrants:
        x_lo, x_hi = ((34, 112) if quadrant % 2 == 0 else (400, 478))
        y_lo, y_hi = ((36, 114) if quadrant < 2 else (398, 476))
        distractors.append(
            {
                "x": rng.randint(x_lo, x_hi),
                "y": rng.randint(y_lo, y_hi),
                "radius": rng.randint(5, 11),
                "shade": rng.randint(196, 225),
            }
        )
    return {
        "canvas": 512,
        "center_x": center_x,
        "center_y": center_y,
        "core_half": core_half,
        "extension": extension,
        "half_span": half_span,
        "axis": axis,
        "side": side,
        "teal_rgb": list(specimen),
        "gold_rgb": list(marker),
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    core = int(scene["core_half"])
    extension = int(scene["extension"])
    half_span = int(scene["half_span"])
    axis = str(scene["axis"])
    side = int(scene["side"])
    specimen = tuple(int(value) for value in scene["teal_rgb"])
    marker = tuple(int(value) for value in scene["gold_rgb"])

    image = Image.new("RGB", (canvas, canvas), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    for item in scene["distractors"]:
        x = int(item["x"])
        y = int(item["y"])
        radius = int(item["radius"])
        shade = int(item["shade"])
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=(shade, shade, shade),
        )

    draw.rectangle((cx - core, cy - core, cx + core, cy + core), fill=specimen)
    if axis == "left-right":
        if side < 0:
            box = (cx - half_span, cy - core - extension, cx + half_span, cy - core - 1)
        else:
            box = (cx - half_span, cy + core + 1, cx + half_span, cy + core + extension)
    else:
        if side < 0:
            box = (cx - core - extension, cy - half_span, cx - core - 1, cy + half_span)
        else:
            box = (cx + core + 1, cy - half_span, cx + core + extension, cy + half_span)
    draw.rectangle(box, fill=specimen)

    marker_radius = 5
    draw.ellipse(
        (cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius),
        fill=marker,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["axis"])


def margin(scene: dict) -> float:
    return float(2 * int(scene["extension"]) * (2 * int(scene["half_span"]) + 1))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 100.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_nonoverlapping_distractor_order", twin)]
