"""Render a compact two-courtyard tile with one intervention shutter."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

BACKGROUND = (248, 248, 244)
TEALS = ((25, 137, 130), (47, 119, 166))
YELLOW = (246, 184, 42)
GRAY = (169, 174, 178)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 384
    pocket = rng.choice((22, 24, 26))
    rim = rng.choice((8, 9, 10))
    cx = rng.choice((96, 144, 192, 240, 288)) + rng.randint(-3, 3)
    cy = rng.choice((96, 144, 192, 240, 288)) + rng.randint(-3, 3)
    distractors = []
    while len(distractors) < 6:
        x, y = rng.randint(18, width - 19), rng.randint(18, height - 19)
        if abs(x - cx) > 72 or abs(y - cy) > 72:
            distractors.append([x, y, rng.choice((2, 3, 4))])
    return {
        "width": width,
        "height": height,
        "cx": cx,
        "cy": cy,
        "fuses": bool(int(seed) % 2 == 0),
        "rotation": rng.randrange(4),
        "pocket": pocket,
        "rim": rim,
        "palette": rng.randrange(len(TEALS)),
        "distractors": distractors,
    }


def _tile(scene: dict) -> Image.Image:
    pocket, rim = int(scene["pocket"]), int(scene["rim"])
    teal = TEALS[int(scene["palette"]) % len(TEALS)]
    width, height = 2 * pocket + 3 * rim, pocket + 2 * rim
    tile = Image.new("RGB", (width, height), teal)
    draw = ImageDraw.Draw(tile)
    y0, y1 = rim, rim + pocket - 1
    left_x0, left_x1 = rim, rim + pocket - 1
    right_x0 = 2 * rim + pocket
    right_x1 = right_x0 + pocket - 1
    draw.rectangle((left_x0, y0, left_x1, y1), fill=BACKGROUND)
    draw.rectangle((right_x0, y0, right_x1, y1), fill=BACKGROUND)
    if bool(scene["fuses"]):
        draw.rectangle((rim + pocket, y0, 2 * rim + pocket - 1, y1), fill=YELLOW)
    else:
        draw.rectangle((left_x0, 0, left_x1, rim - 1), fill=YELLOW)
    rotation = int(scene["rotation"]) % 4
    for _ in range(rotation):
        tile = tile.transpose(Image.Transpose.ROTATE_90)
    return tile


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["distractors"]:
        draw.rectangle((int(x) - int(radius), int(y) - int(radius), int(x) + int(radius), int(y) + int(radius)), fill=GRAY)
    tile = _tile(scene)
    cx, cy = int(scene["cx"]), int(scene["cy"])
    image.paste(tile, (cx - tile.width // 2, cy - tile.height // 2))
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["fuses"]) else "no"


def margin(scene: dict) -> float:
    return float(min(int(scene["rim"]), int(scene["pocket"])))


def is_quarantined(scene: dict) -> bool:
    return int(scene["rim"]) < 7 or int(scene["pocket"]) < 18


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene
    return []
