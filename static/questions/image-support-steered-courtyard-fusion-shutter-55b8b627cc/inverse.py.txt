"""Independent pixel-only topology oracle.

The inverse arm segments colors, labels four-neighbor components from runs,
classifies white components by border reachability, and inspects which distinct
pre-intervention components share an edge with the yellow shutter. It does not
import or use renderer state.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _label_components(mask: np.ndarray) -> tuple[np.ndarray, list[tuple[int, bool]]]:
    """Return dense labels and (area, touches_border) using run union-find."""
    height, width = mask.shape
    parent: list[int] = []
    area: list[int] = []
    border: list[bool] = []
    runs: list[tuple[int, int, int, int]] = []
    previous: list[tuple[int, int, int]] = []

    def root(index: int) -> int:
        while parent[index] != index:
            parent[index] = parent[parent[index]]
            index = parent[index]
        return index

    def join(left: int, right: int) -> int:
        left, right = root(left), root(right)
        if left == right:
            return left
        if area[left] < area[right]:
            left, right = right, left
        parent[right] = left
        area[left] += area[right]
        border[left] = border[left] or border[right]
        return left

    for y in range(height):
        edges = np.flatnonzero(np.diff(np.r_[False, mask[y], False].astype(np.int8)))
        current: list[tuple[int, int, int]] = []
        for x0, x1 in edges.reshape(-1, 2):
            index = len(parent)
            parent.append(index)
            area.append(int(x1 - x0))
            border.append(y in (0, height - 1) or int(x0) == 0 or int(x1) == width)
            current.append((int(x0), int(x1), index))
            runs.append((y, int(x0), int(x1), index))
        p = 0
        for x0, x1, index in current:
            while p < len(previous) and previous[p][1] <= x0:
                p += 1
            q = p
            while q < len(previous) and previous[q][0] < x1:
                join(index, previous[q][2])
                q += 1
        previous = current

    root_to_label: dict[int, int] = {}
    info: list[tuple[int, bool]] = []
    for index in range(len(parent)):
        representative = root(index)
        if representative not in root_to_label:
            root_to_label[representative] = len(info)
            info.append((area[representative], border[representative]))
    labels = np.full(mask.shape, -1, dtype=np.int32)
    for y, x0, x1, index in runs:
        labels[y, x0:x1] = root_to_label[root(index)]
    return labels, info


def _adjacent_labels(mask: np.ndarray, labels: np.ndarray) -> set[int]:
    ids: set[int] = set()
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and labels[ny, nx] >= 0:
                ids.add(int(labels[ny, nx]))
    return ids


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 128:
        raise ValueError("unexpected canvas")
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    yellow = (red > 205) & (green > 125) & (green < 225) & (blue < 100) & ((red - blue) > 120)
    teal = (red < 105) & (green > 95) & (blue > 80) & ((green - red) > 25)

    _, yellow_info = _label_components(yellow)
    yellow_groups = [item for item in yellow_info if item[0] >= 100]
    if len(yellow_groups) != 1 or not 150 <= yellow_groups[0][0] <= 280:
        raise ValueError("expected one intact thick yellow shutter")

    union = teal | yellow
    _, union_info = _label_components(union)
    land = [item for item in union_info if item[0] >= 500]
    if len(land) != 1 or land[0][1] or not 1500 <= land[0][0] <= 3800:
        raise ValueError("expected one compact solid wall tile")

    before_labels, before_info = _label_components(~union)
    before_large = {index for index, (size, _) in enumerate(before_info) if size >= 100}
    before_bounded = {index for index in before_large if not before_info[index][1]}
    before_border = {index for index in before_large if before_info[index][1]}
    if len(before_bounded) != 2 or len(before_border) != 1:
        raise ValueError("expected exactly two enclosed courtyards before intervention")

    after_labels, after_info = _label_components(~teal)
    after_large = {index for index, (size, _) in enumerate(after_info) if size >= 100}
    after_bounded = {index for index in after_large if not after_info[index][1]}
    after_border = {index for index in after_large if after_info[index][1]}
    if len(after_bounded) != 1 or len(after_border) != 1:
        raise ValueError("intervention must leave exactly one enclosed courtyard")

    adjacent = _adjacent_labels(yellow, before_labels) & before_large
    bounded_neighbors = adjacent & before_bounded
    border_neighbors = adjacent & before_border
    if len(bounded_neighbors) == 2 and not border_neighbors:
        return "yes"
    if len(bounded_neighbors) == 1 and len(border_neighbors) == 1:
        return "no"
    raise ValueError("shutter does not exhibit either declared component transition")
