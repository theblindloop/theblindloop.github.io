"""Independent pixel-only inverse for the fixed-lag state-slot world.

Only the public raster geometry and bitmap signatures are duplicated here. The
module does not import renderer, prompts, generate, verify, or analytic-gold
code and receives no latent scene data.
"""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 7
IDENTITY_COUNT = 4
STATE_COUNT = 3
LABELS = ("A", "B", "C", "D")
CANVAS = (1940, 820)
INK = (31, 48, 61)
PANEL_X0 = 24
PANEL_TOP = 222
PANEL_WIDTH = 260
PANEL_GAP = 8
SLOT_ORIGINS = ((14, 88), (134, 88), (14, 332), (134, 332))
LABEL_X_OFFSET = 14
LABEL_Y_OFFSET = 44
LABEL_SCALE = 3
STATE_X_OFFSET = 76
STATE_Y_OFFSET = 43
STATE_BOX_SIZE = 26
KEY_XS = (470, 720, 970)
KEY_Y = 112

LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)


def _label_points(label: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * LABEL_SCALE + dx, y * LABEL_SCALE + dy)
        for y, row in enumerate(LABEL_PATTERNS[label])
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(LABEL_SCALE)
        for dy in range(LABEL_SCALE)
    )


LABEL_POINTS = tuple(_label_points(label) for label in range(IDENTITY_COUNT))


def _read_label(frame: Image.Image, origin: tuple[int, int]) -> int:
    ox, oy = origin
    observed = frozenset(
        (x, y)
        for x in range(15)
        for y in range(21)
        if frame.getpixel((ox + x, oy + y)) == INK
    )
    matches = [label for label, points in enumerate(LABEL_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("identity glyph is missing, altered, or ambiguous")
    return matches[0]


def _decode_key(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    mapping: dict[tuple[int, int, int], int] = {}
    for state, key_x in enumerate(KEY_XS):
        color = frame.getpixel((key_x + STATE_BOX_SIZE // 2, KEY_Y + STATE_BOX_SIZE // 2))
        if color in mapping:
            raise ValueError("state key repeats a palette color")
        mapping[color] = state
    if len(mapping) != STATE_COUNT or set(mapping.values()) != set(range(STATE_COUNT)):
        raise ValueError("state key does not expose three keyed colors")
    return mapping


def decision_from_image(image: Image.Image) -> str:
    """Decode the PNG and return the unique two-step joint-echo winner."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_key = _decode_key(frame)
    state_tables: list[dict[int, int]] = []
    position_tables: list[dict[int, int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        states: dict[int, int] = {}
        positions: dict[int, int] = {}
        for slot, (slot_x, slot_y) in enumerate(SLOT_ORIGINS):
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            label = _read_label(frame, (card_x + LABEL_X_OFFSET, card_y + LABEL_Y_OFFSET))
            if label in states:
                raise ValueError("a panel repeats one identity")
            color = frame.getpixel(
                (
                    card_x + STATE_X_OFFSET + STATE_BOX_SIZE // 2,
                    card_y + STATE_Y_OFFSET + STATE_BOX_SIZE // 2,
                )
            )
            if color not in state_key:
                raise ValueError("tile state swatch is missing or not keyed")
            states[label] = state_key[color]
            positions[label] = slot
        if set(states) != set(range(IDENTITY_COUNT)):
            raise ValueError("a panel does not contain all four identities")
        if set(positions.values()) != set(range(IDENTITY_COUNT)):
            raise ValueError("a panel does not contain four distinct slots")
        state_tables.append(states)
        position_tables.append(positions)

    scores = [
        sum(
            state_tables[panel][label] == state_tables[panel + 2][label]
            and position_tables[panel][label] == position_tables[panel + 2][label]
            for panel in range(PANEL_COUNT - 2)
        )
        for label in range(IDENTITY_COUNT)
    ]
    ordered = sorted(scores, reverse=True)
    winners = [label for label, score in enumerate(scores) if score == ordered[0]]
    if len(winners) != 1 or ordered[0] - ordered[1] < 2:
        raise ValueError("image does not expose one unique two-step echo winner")
    return LABELS[winners[0]]
