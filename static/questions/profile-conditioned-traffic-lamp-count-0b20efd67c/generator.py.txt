"""Deterministic raster renderer for the traffic-lamp count question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4

LAMP_PALETTE = (
    (219, 54, 52),
    (38, 125, 221),
    (39, 173, 105),
    (235, 139, 39),
    (155, 76, 202),
    (24, 159, 174),
    (211, 179, 35),
)
HOUSING_PALETTE = ((38, 47, 57), (49, 54, 62), (43, 56, 58), (55, 48, 61))
BACKGROUND_PALETTE = ((247, 249, 246), (255, 252, 245), (243, 247, 250), (249, 246, 251))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene; the count is independent of palette and pose."""

    rng = random.Random(int(seed))
    lamp_count = rng.choice((2, 3, 4, 5))
    housing_width = rng.uniform(154.0, 188.0)
    housing_height = rng.uniform(390.0, 420.0)
    lens_radius = rng.uniform(20.5, 26.0)

    # A small minority sit close to the finite-resolution boundary.  They remain
    # visibly separable, but are explicitly quarantined by is_quarantined().
    near_boundary = rng.random() < 0.18
    if near_boundary:
        lens_radius = rng.uniform(19.5, 22.5)
        lens_gap = rng.uniform(4.5, 7.0)
        housing_height = 100.0 + 2.0 * lens_radius * lamp_count + lens_gap * (lamp_count - 1)
    else:
        usable = housing_height - 100.0
        max_radius = min(26.0, (usable - 12.0 * (lamp_count - 1)) / (2.0 * lamp_count))
        lens_radius = min(lens_radius, max_radius)
        lens_gap = (usable - 2.0 * lens_radius * lamp_count) / (lamp_count - 1)

    center_x = rng.uniform(102.0, 410.0)
    center_y = rng.uniform(housing_height / 2.0 + 13.0, HEIGHT - housing_height / 2.0 - 13.0)

    return {
        "lamp_count": lamp_count,
        "center": [round(center_x, 3), round(center_y, 3)],
        "housing_width": round(housing_width, 3),
        "housing_height": round(housing_height, 3),
        "lens_radius": round(lens_radius, 3),
        "lens_gap": round(lens_gap, 3),
        "lamp_color": list(rng.choice(LAMP_PALETTE)),
        "housing_color": list(rng.choice(HOUSING_PALETTE)),
        "background": list(rng.choice(BACKGROUND_PALETTE)),
        "mount_offset": round(rng.uniform(-12.0, 12.0), 3),
        "label_text": "SIGNAL",
    }


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[float, ...]:
    return tuple(value * scale for value in box)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a single signal; all lenses share one sampled, saturated colour."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx, cy = (float(value) for value in scene["center"])
    hw = float(scene["housing_width"])
    hh = float(scene["housing_height"])
    radius = float(scene["lens_radius"])
    gap = float(scene["lens_gap"])
    housing = tuple(scene["housing_color"])
    lamp = tuple(scene["lamp_color"])

    # A neutral rear bracket and foot provide object context without adding a
    # second saturated object for the image-only component counter to mistake.
    mx = cx + float(scene["mount_offset"])
    top = cy - hh / 2.0
    bottom = cy + hh / 2.0
    draw.rounded_rectangle(
        _scaled_box((mx - 17.0, top - 30.0, mx + 17.0, top + 18.0), ss),
        radius=7 * ss,
        fill=(112, 120, 124),
    )
    draw.rounded_rectangle(
        _scaled_box((mx - 34.0, bottom - 3.0, mx + 34.0, bottom + 20.0), ss),
        radius=6 * ss,
        fill=(112, 120, 124),
    )

    body_box = (cx - hw / 2.0, top, cx + hw / 2.0, bottom)
    draw.rounded_rectangle(
        _scaled_box(body_box, ss),
        radius=18 * ss,
        fill=housing,
        outline=(15, 22, 29),
        width=5 * ss,
    )
    draw.text(
        (cx * ss, (top + 27.0) * ss),
        str(scene["label_text"]),
        fill=(194, 202, 204),
        anchor="mm",
        font=_font(12 * ss),
    )

    first_y = top + 50.0 + radius
    step = 2.0 * radius + gap
    for index in range(int(scene["lamp_count"])):
        y = first_y + index * step
        box = (cx - radius, y - radius, cx + radius, y + radius)
        draw.ellipse(_scaled_box(box, ss), fill=lamp, outline=(12, 22, 28), width=4 * ss)
        # A restrained neutral glint keeps the lens reading as a lens without
        # creating a disconnected colour component.
        glint = radius * 0.22
        draw.ellipse(
            _scaled_box((cx - radius * 0.42, y - radius * 0.46, cx - radius * 0.42 + glint, y - radius * 0.46 + glint), ss),
            fill=(225, 232, 230),
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["lamp_count"])


def margin(scene: dict[str, Any]) -> float:
    """Smallest visible clearance relevant to separating adjacent lenses."""

    return float(min(float(scene["lens_gap"]), float(scene["lens_radius"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return float(scene["lens_gap"]) < 8.0 or float(scene["lens_radius"]) < 18.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No sampled latent transform is pixel-identical under this renderer."""

    return []
