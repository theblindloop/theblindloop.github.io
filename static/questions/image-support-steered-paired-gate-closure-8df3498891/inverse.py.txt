"""Independent PNG-only oracle for paired boundary completion.

The arm segments the two colored component sets and the black construction,
reconstructs the thick rectangular boundary from its visible extrema, then
independently inserts each candidate pair and flood-fills the central chamber.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CYAN = np.array([0, 156, 184], dtype=np.int16)
_ORANGE = np.array([230, 112, 24], dtype=np.int16)
_BLACK = np.array([28, 30, 34], dtype=np.int16)
_COLOR_TOL = 48.0


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        area = 0
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            area += 1
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((area, min(xs), min(ys), max(xs), max(ys)))
    return output


def _two_bars(mask: np.ndarray, name: str) -> list[tuple[int, int, int, int, int]]:
    parts = [part for part in _components(mask) if part[0] >= 80]
    if len(parts) != 2 or any(not 120 <= part[0] <= 240 for part in parts):
        raise ValueError(f"expected two intact {name} bars")
    return parts


def _reaches_border(barrier: np.ndarray, start: tuple[int, int]) -> bool:
    height, width = barrier.shape
    sy, sx = start
    if not (0 <= sy < height and 0 <= sx < width) or barrier[sy, sx]:
        raise ValueError("central white chamber seed is obstructed")
    seen = np.zeros_like(barrier, dtype=bool)
    queue = deque([(sy, sx)])
    seen[sy, sx] = True
    while queue:
        y, x = queue.popleft()
        if y == 0 or x == 0 or y == height - 1 or x == width - 1:
            return True
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if not barrier[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 200:
        raise ValueError("raster is too small")
    cyan = np.linalg.norm(array - _CYAN, axis=2) <= _COLOR_TOL
    orange = np.linalg.norm(array - _ORANGE, axis=2) <= _COLOR_TOL
    black = np.linalg.norm(array - _BLACK, axis=2) <= _COLOR_TOL
    cyan_parts = _two_bars(cyan, "cyan")
    orange_parts = _two_bars(orange, "orange")
    if int(black.sum()) < 4500:
        raise ValueError("substantial black boundary is missing")

    ys, xs = np.nonzero(black)
    left, right, top, bottom = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (105 <= right - left <= 150 and 105 <= bottom - top <= 150):
        raise ValueError("black boundary extents are invalid")
    thickness = 13
    half = thickness // 2
    cx, cy = (left + right) // 2, (top + bottom) // 2

    # Reconstruct thick outer walls from visible extrema. Colored pixels on an
    # outer edge are then cut back out, leaving exactly the visible candidate
    # openings. This makes the topology arm robust to small raster damage while
    # retaining direct dependence on every colored component.
    base = black.copy()
    base[top : top + thickness, left : right + 1] = True
    base[bottom - thickness + 1 : bottom + 1, left : right + 1] = True
    base[top : bottom + 1, left : left + thickness] = True
    base[top : bottom + 1, right - thickness + 1 : right + 1] = True
    colored = cyan | orange
    outer_band = np.zeros_like(base)
    outer_band[top : top + thickness, left : right + 1] = True
    outer_band[bottom - thickness + 1 : bottom + 1, left : right + 1] = True
    base[colored & outer_band] = False

    decisions = {}
    for name, mask, parts in (("cyan", cyan, cyan_parts), ("orange", orange, orange_parts)):
        barrier = base | mask
        # At least one component must participate in a visible black adjacency;
        # this rejects floating recolors unrelated to the construction.
        adjacency = 0
        for _, x0, y0, x1, y1 in parts:
            halo = black[max(0, y0 - 2) : y1 + 3, max(0, x0 - 2) : x1 + 3]
            adjacency += int(bool(halo.any()))
        if adjacency != 2:
            raise ValueError(f"{name} pair lacks wall adjacency")
        decisions[name] = not _reaches_border(barrier, (cy, cx))
    winners = [name for name, closes in decisions.items() if closes]
    if len(winners) != 1:
        raise ValueError("counterfactuals do not yield one closing color")
    return winners[0]
