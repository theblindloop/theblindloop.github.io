"""Deterministic renderer for paired counterfactual boundary completion."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 320
HEIGHT = 320
BACKGROUND = [247, 247, 242]
BLACK = [28, 30, 34]
CYAN = [0, 156, 184]
ORANGE = [230, 112, 24]
GRAY = [163, 166, 168]
WALL_HALF = 6
BAR_HALF_X = 7
MIN_MARGIN = 10


def _box(cx: int, cy: int) -> list[int]:
    return [cx - BAR_HALF_X, cy - WALL_HALF, cx + BAR_HALF_X, cy + WALL_HALF]


def _rotate_box(box: list[int]) -> list[int]:
    x0, y0, x1, y1 = box
    return [WIDTH - 1 - x1, HEIGHT - 1 - y1, WIDTH - 1 - x0, HEIGHT - 1 - y0]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    x0 = rng.choice([80, 160])
    y0 = 100
    x1, y1 = x0 + 120, y0 + 120
    gap_bars = [_box(x0 + 20, y0), _box(x0 + 100, y0)]
    rail_bars = [_box(x0 + 20, y0 + 80), _box(x0 + 100, y0 + 80)]
    frame = [x0, y0, x1, y1]

    # Half the scenes invert the entire local construction. This randomizes
    # which frame edge carries the true gaps while preserving thick geometry.
    if rng.randrange(2):
        frame = _rotate_box(frame)
        gap_bars = [_rotate_box(box) for box in gap_bars]
        rail_bars = [_rotate_box(box) for box in rail_bars]

    closing_is_cyan = (int(seed) % 2) == 0
    cyan_bars = gap_bars if closing_is_cyan else rail_bars
    orange_bars = rail_bars if closing_is_cyan else gap_bars

    # Remote gray marks vary but are never used by either decision arm.
    distractor_slots = [[18, 24, 25, 31], [292, 35, 299, 42], [22, 284, 29, 291]]
    rng.shuffle(distractor_slots)
    distractors = sorted(distractor_slots[: 1 + rng.randrange(3)])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "wall_color": list(BLACK),
        "cyan": list(CYAN),
        "orange": list(ORANGE),
        "frame_bounds": frame,
        "cyan_bars": sorted([list(box) for box in cyan_bars]),
        "orange_bars": sorted([list(box) for box in orange_bars]),
        "distractors": distractors,
    }


def _draw_frame(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    x0, y0, x1, y1 = scene["frame_bounds"]
    half = WALL_HALF
    all_bars = scene["cyan_bars"] + scene["orange_bars"]
    edge_bars = []
    for box in all_bars:
        _, by0, _, by1 = box
        cy = (by0 + by1) // 2
        if abs(cy - y0) <= 1 or abs(cy - y1) <= 1:
            edge_bars.append(box)
    if len(edge_bars) != 2:
        raise ValueError("exactly two colored bars must occupy one outer edge")
    edge_y = (edge_bars[0][1] + edge_bars[0][3]) // 2
    centers = sorted((box[0] + box[2]) // 2 for box in edge_bars)

    # Complete side walls and the opposite horizontal wall.
    draw.rectangle([x0 - half, y0 - half, x0 + half, y1 + half], fill=tuple(scene["wall_color"]))
    draw.rectangle([x1 - half, y0 - half, x1 + half, y1 + half], fill=tuple(scene["wall_color"]))
    opposite_y = y1 if abs(edge_y - y0) <= 1 else y0
    draw.rectangle([x0 - half, opposite_y - half, x1 + half, opposite_y + half], fill=tuple(scene["wall_color"]))

    # Broken outer edge: two equal openings occupied by the true candidate pair.
    cursor = x0 - half
    for cx in centers:
        draw.rectangle([cursor, edge_y - half, cx - BAR_HALF_X - 1, edge_y + half], fill=tuple(scene["wall_color"]))
        cursor = cx + BAR_HALF_X + 1
    draw.rectangle([cursor, edge_y - half, x1 + half, edge_y + half], fill=tuple(scene["wall_color"]))

    # A visually matched inner rail holds the other colored pair. Completing it
    # divides the interior but cannot seal the central chamber from the outside.
    rail_boxes = [box for box in all_bars if box not in edge_bars]
    rail_y = (rail_boxes[0][1] + rail_boxes[0][3]) // 2
    rail_centers = sorted((box[0] + box[2]) // 2 for box in rail_boxes)
    cursor = x0 - half
    for cx in rail_centers:
        draw.rectangle([cursor, rail_y - half, cx - BAR_HALF_X - 1, rail_y + half], fill=tuple(scene["wall_color"]))
        cursor = cx + BAR_HALF_X + 1
    draw.rectangle([cursor, rail_y - half, x1 + half, rail_y + half], fill=tuple(scene["wall_color"]))


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    if len(scene["frame_bounds"]) != 4:
        raise ValueError("frame_bounds must contain four coordinates")
    _draw_frame(draw, scene)
    for box in scene["cyan_bars"]:
        draw.rectangle(box, fill=tuple(scene["cyan"]))
    for box in scene["orange_bars"]:
        draw.rectangle(box, fill=tuple(scene["orange"]))
    for box in scene["distractors"]:
        draw.ellipse(box, fill=tuple(GRAY))
    return image


def _bars_on_outer_edge(frame: list[int], bars: list[list[int]]) -> bool:
    _, y0, _, y1 = frame
    centers = [(box[1] + box[3]) // 2 for box in bars]
    return len(centers) == 2 and (all(abs(y - y0) <= 1 for y in centers) or all(abs(y - y1) <= 1 for y in centers))


def analytic_gold(scene: dict[str, Any]) -> str:
    frame = scene["frame_bounds"]
    cyan_closes = _bars_on_outer_edge(frame, scene["cyan_bars"])
    orange_closes = _bars_on_outer_edge(frame, scene["orange_bars"])
    if cyan_closes == orange_closes:
        raise ValueError("scene must have exactly one closing color")
    return "cyan" if cyan_closes else "orange"


def margin(scene: dict[str, Any]) -> float:
    # The minimum visible thickness/contact redundancy in the construction.
    widths = [box[3] - box[1] + 1 for box in scene["cyan_bars"] + scene["orange_bars"]]
    return float(min(widths))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["cyan_bars"] = list(reversed(twin["cyan_bars"]))
    twin["orange_bars"] = list(reversed(twin["orange_bars"]))
    return [("reverse_same_color_pair_storage", twin)]
