"""Deterministic renderer and analytic arm for step-plateau measurement."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "blue": (35, 104, 180),
    "orange": (230, 112, 28),
    "green": (32, 145, 82),
}
PLOT_X0 = 150
PLOT_X1 = 650
PANEL_CENTERS = (92, 239, 386)
PANEL_HALF_HEIGHT = 55


def _partition_with_max(rng: random.Random, maximum: int) -> list[int]:
    """Five positive integer spans totaling 500, with one unique prescribed max."""
    remainder = PLOT_X1 - PLOT_X0 - maximum
    for _ in range(2000):
        cuts = sorted(rng.sample(range(56, remainder - 55), 3))
        parts = [cuts[0], cuts[1] - cuts[0], cuts[2] - cuts[1], remainder - cuts[2]]
        if min(parts) >= 48 and max(parts) <= maximum - 8:
            insert_at = rng.randrange(5)
            parts.insert(insert_at, maximum)
            return parts
    raise RuntimeError("could not construct plateau partition")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(COLORS)
    winner = names[int(seed) % len(names)]
    others = [name for name in names if name != winner]
    rng.shuffle(others)

    winner_max = rng.randint(176, 198)
    gap = rng.randint(14, 42)
    runner_max = winner_max - gap
    third_max = runner_max - rng.randint(12, 25)
    assigned = {winner: winner_max, others[0]: runner_max, others[1]: third_max}
    widths = {name: _partition_with_max(rng, assigned[name]) for name in names}

    panel_order = names[:]
    rng.shuffle(panel_order)
    step_levels = {}
    offsets = [-38, -19, 0, 19, 38]
    for panel_index, name in enumerate(panel_order):
        levels = [PANEL_CENTERS[panel_index] + value for value in offsets]
        rng.shuffle(levels)
        step_levels[name] = levels

    return {
        "width": 720,
        "height": 480,
        "panel_order": panel_order,
        "plateau_widths": widths,
        "step_levels": step_levels,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    panel_order = list(scene["panel_order"])
    plateau_widths = scene["plateau_widths"]
    step_levels = scene["step_levels"]
    image = Image.new("RGB", (width, height), (250, 251, 253))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((18, 12), "STEP RESPONSE: compare longest horizontal plateaus", fill=(25, 31, 39), font=font)
    for panel_index, name in enumerate(panel_order):
        center = PANEL_CENTERS[panel_index]
        top, bottom = center - PANEL_HALF_HEIGHT, center + PANEL_HALF_HEIGHT
        draw.rectangle((PLOT_X0, top, PLOT_X1, bottom), fill=(255, 255, 255), outline=(185, 191, 199), width=1)
        for gx in range(PLOT_X0 + 100, PLOT_X1, 100):
            draw.line((gx, top + 1, gx, bottom - 1), fill=(232, 235, 239), width=1)
        draw.text((76, center - 5), name.upper(), fill=(28, 34, 42), font=font)
        draw.ellipse((50, center - 6, 62, center + 6), fill=COLORS[name])

        x = PLOT_X0
        widths = [int(value) for value in plateau_widths[name]]
        levels = [int(value) for value in step_levels[name]]
        for index, span in enumerate(widths):
            x_next = x + span
            draw.line((x, levels[index], x_next, levels[index]), fill=COLORS[name], width=7)
            if index + 1 < len(widths):
                draw.line((x_next, levels[index], x_next, levels[index + 1]), fill=COLORS[name], width=7)
            x = x_next

    draw.text((PLOT_X0, 459), "shared horizontal scale", fill=(75, 82, 91), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    widths = scene["plateau_widths"]
    scores = {name: max(int(value) for value in values) for name, values in widths.items()}
    ordered = sorted(scores, key=lambda name: (-scores[name], name))
    if scores[ordered[0]] == scores[ordered[1]]:
        raise ValueError("ambiguous plateau maximum")
    return ordered[0]


def margin(scene: dict) -> float:
    widths = scene["plateau_widths"]
    values = sorted((max(int(value) for value in spans) for spans in widths.values()), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
