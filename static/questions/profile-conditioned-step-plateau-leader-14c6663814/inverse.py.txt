"""Independent PNG-only inverse arm.

The oracle knows only the public palette and decision rule. It segments each
curve color and measures the longest contiguous horizontal color run in any
image row. Vertical transition strokes contribute only their stroke width and
cannot win against a valid plateau.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "blue": np.array((35, 104, 180), dtype=np.int16),
    "orange": np.array((230, 112, 28), dtype=np.int16),
    "green": np.array((32, 145, 82), dtype=np.int16),
}


def _longest_run(row: np.ndarray) -> int:
    padded = np.concatenate(([False], row.astype(bool), [False]))
    edges = np.flatnonzero(padded[1:] != padded[:-1])
    return int(np.max(edges[1::2] - edges[::2])) if len(edges) else 0


def decision_from_image(image: Image.Image) -> str:
    # int32 prevents squared channel distances from overflowing.
    rgb = np.asarray(image.convert("RGB"), dtype=np.int32)
    scores = {}
    for name, target in PALETTE.items():
        distance2 = np.sum((rgb - target.astype(np.int32)) ** 2, axis=2)
        mask = distance2 <= 28 * 28
        if int(mask.sum()) < 500:
            raise ValueError(f"missing or corrupted {name} curve")
        scores[name] = max(_longest_run(mask[y]) for y in range(mask.shape[0]))
    ordered = sorted(scores, key=lambda name: (-scores[name], name))
    if scores[ordered[0]] - scores[ordered[1]] < 8:
        raise ValueError("ambiguous pixel plateau maximum")
    return ordered[0]
