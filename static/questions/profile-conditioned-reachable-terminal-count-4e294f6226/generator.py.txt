"""Tracing world: how many colored terminals are reachable from the start ring.

A winding blue network stroke (the ``main`` polyline) is always a single
connected component and carries an orange start ring.  A short, separate blue
``island`` stroke is a second component separated by a clearly visible white
gap.  Two colored terminals (red and green) are each placed either on the main
network or on the island.  The decision is the NUMBER of those two colored
terminals that are reachable from the start by following only the blue lines --
i.e. how many terminal dots lie on the same connected blue component as the
ring (0, 1 or 2).
"""

from __future__ import annotations

import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "reachable_terminal_count-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20
START_R = 5
TERM_R = 4
MARGIN_PAD = 46

GAP_MIN = 28.0
MARKER_SEP = 24.0

SUBSTRATE = "blue"
TERMINAL_COLORS = ["red", "green"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points, s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _sample_pts(segments, step: float = 2.0) -> list[tuple[float, float]]:
    out = []
    for a, b in segments:
        d = math.dist(a, b)
        n = max(1, int(d / step))
        for i in range(n + 1):
            t = i / n
            out.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
    return out


def _polyline_segments(points):
    return list(zip(points, points[1:]))


def _min_clearance(main_pts, island_pts) -> float:
    best = float("inf")
    for p in island_pts:
        for q in main_pts:
            d = math.dist(p, q)
            if d < best:
                best = d
    return best


def _rejection_loop(rng: random.Random):
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        n_verts = rng.randint(4, 6)
        seg_len = 90.0
        ys = [rng.uniform(lo, hi)]
        xs = [rng.uniform(lo, hi)]
        for _ in range(n_verts):
            xs.append(xs[-1] + rng.uniform(0.45, 1.0) * seg_len)
            ys.append(rng.uniform(lo, hi))
        if xs[-1] > w - MARGIN_PAD:
            fold = (xs[-1] - (w - MARGIN_PAD))
            xs = [x - max(0.0, fold * (i / n_verts)) for i, x in enumerate(xs)]
        main = _pts(list(zip(xs, ys)))
        total = _path_length(main)

        s_start = rng.uniform(4, 0.55 * total)
        start = _point_at(main, s_start)

        for _ in range(60):
            base = (rng.uniform(lo, hi), rng.uniform(lo, hi))
            ang = rng.uniform(0.0, 2 * math.pi)
            ilen = rng.uniform(60, 100)
            tip = (base[0] + ilen * math.cos(ang), base[1] + ilen * math.sin(ang))
            clamp = lambda v, a, b: max(a, min(b, v))
            tip = (clamp(tip[0], lo, hi), clamp(tip[1], lo, hi))
            island = [base, tip]
            gap = _min_clearance(
                _sample_pts(_polyline_segments(main)), _sample_pts(_polyline_segments(island))
            )
            if gap >= GAP_MIN + 3.0:
                break
        else:
            continue

        ilen = _path_length(island)
        # Candidate terminal points: pick k on main, (2-k) on island.
        k = rng.randint(0, 2)
        main_cands = [_point_at(main, rng.uniform(0.6 * total, total - 4)) for _ in range(2)]
        island_cands = [_point_at(island, ilen * rng.uniform(0.25, 0.75)) for _ in range(2)]

        # Build final positions: first k on main, remaining 2-k on island.
        positions = []
        on_main = []
        if k == 2:
            positions = [main_cands[0], main_cands[1]]
            on_main = [True, True]
        elif k == 1:
            positions = [main_cands[0], island_cands[0]]
            on_main = [True, False]
        else:
            positions = [island_cands[0], island_cands[1]]
            on_main = [False, False]

        def _good(mk):
            px, py = mk
            return 20 < px < w - 20 and 20 < py < h - 20

        if not (_good(start) and all(_good(p) for p in positions)):
            continue
        all_mk = [start] + positions
        ok = True
        for i in range(len(all_mk)):
            for j in range(i + 1, len(all_mk)):
                if math.dist(all_mk[i], all_mk[j]) < MARKER_SEP:
                    ok = False
        if not ok:
            continue

        answer_count = k
        scene = {
            "canvas_w": w,
            "canvas_h": h,
            "main": main,
            "island_pts": island,
            "start": start,
            "term_pos": positions,
            "term_colors": list(TERMINAL_COLORS),
            "gap": gap,
            "answer_count": answer_count,
        }
        return scene
    raise RuntimeError("could not satisfy isolation constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    w = int(scene["canvas_w"])
    h = int(scene["canvas_h"])
    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    main = _pts(scene["main"])
    island = _pts(scene["island_pts"])
    _draw_polyline(draw, main, COLOR_BLUE, STROKE)
    _draw_polyline(draw, island, COLOR_BLUE, STROKE)
    # Re-read the answer count (validating the scene) so the rendered-field
    # audit sees that this field is part of the visible contract.
    count = int(scene["answer_count"])
    assert count in (0, 1, 2)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN}
    for pos, name in zip(scene["term_pos"], scene["term_colors"]):
        px, py = float(pos[0]), float(pos[1])
        r = TERM_R
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[name])
    sx, sy = float(scene["start"][0]), float(scene["start"][1])
    draw.ellipse([sx - START_R, sy - START_R, sx + START_R, sy + START_R], fill=COLOR_ORANGE)
    return img


def analytic_gold(scene: dict) -> str:
    return str(int(scene["answer_count"]))


def margin(scene: dict) -> float:
    main = _pts(scene["main"])
    island = _pts(scene["island_pts"])
    return _min_clearance(
        _sample_pts(_polyline_segments(main)), _sample_pts(_polyline_segments(island))
    )


def is_quarantined(scene: dict) -> bool:
    if float(scene.get("gap", 0.0)) < GAP_MIN:
        return True
    start = _pts([scene["start"]])[0]
    positions = [_pts([p])[0] for p in scene["term_pos"]]
    all_mk = [start] + positions
    for i in range(len(all_mk)):
        for j in range(i + 1, len(all_mk)):
            if math.dist(all_mk[i], all_mk[j]) < MARKER_SEP:
                return True
    w, h = CANVAS
    for mx, my in all_mk:
        if not (20 <= mx <= w - 20 and 20 <= my <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
