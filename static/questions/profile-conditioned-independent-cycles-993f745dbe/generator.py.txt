"""Independent-cycles (cyclomatic number) world renderer.

The image shows a network of thick black links joining round black nodes on a
white background. The network is a disjoint union of components: some are closed
rings (regular polygons) that enclose a white pocket, and some are open tree
paths that enclose no pocket.

Decision : the number of independent cycles (closed loops) in the network,
which equals the planar cyclomatic number E - V + C. Because components are
disjoint, this equals the number of closed-ring components, which in turn
equals the number of white pockets fully sealed by the network ink.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "components": [ {"nodes": [[x,y],...], "closed": bool}, ... ],
        "width": 400,
        "height": 400,
        "background": "white",
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "independent_cycles-0.4.0"

# Layout constants (pixels). Everything is kept comfortably robust so ordinary
# antialiasing can never open a ring or merge two components.
NODE_RADIUS = 7.0        # filled node disk radius
LINK_WIDTH = 7.0         # link stroke width
COMPONENT_GAP = 26.0     # minimum clearance between component extents
RING_POCKET_MIN = 12.0   # minimum ring inner pocket radius
QUARANTINE_MIN = 6.0     # below this clearance the scene is ambiguous
MAX_CYCLES = 3


def _ring_nodes(cx: float, cy: float, radius: float, nv: int, rot: float) -> list[list[float]]:
    return [
        [cx + radius * math.cos(rot + 2.0 * math.pi * i / nv),
         cy + radius * math.sin(rot + 2.0 * math.pi * i / nv)]
        for i in range(nv)
    ]


def _extent(nodes: list[list[float]]) -> tuple[float, float, float]:
    """Centroid and bounding radius of a component's nodes plus node disks."""
    n = len(nodes)
    mx = sum(p[0] for p in nodes) / n
    my = sum(p[1] for p in nodes) / n
    r = max(math.hypot(p[0] - mx, p[1] - my) for p in nodes) + NODE_RADIUS
    return (mx, my, r)


def sample_scene(
    rng: random.Random,
    n_cycles: int | None = None,
    width: int = 400,
    height: int = 400,
) -> dict:
    """Sample a network with 0..MAX_CYCLES independent cycles, robustly placed."""
    if n_cycles is None:
        n_cycles = rng.randint(0, MAX_CYCLES)
    n_cycles = int(max(0, min(MAX_CYCLES, n_cycles)))
    n_trees = rng.randint(1, 2)

    components: list[dict] = []
    centers: list[tuple[float, float, float]] = []

    def place(cx: float, cy: float, radius: float) -> bool:
        for (ox, oy, orad) in centers:
            if math.hypot(cx - ox, cy - oy) < radius + orad + COMPONENT_GAP:
                return False
        if radius + 8.0 < cx < width - radius - 8.0 and radius + 8.0 < cy < height - radius - 8.0:
            centers.append((cx, cy, radius))
            return True
        return False

    for _ in range(n_cycles):
        ring_r = rng.uniform(33.0, 42.0)
        nv = int(rng.randint(6, 8))
        rot = rng.random() * 2.0 * math.pi
        for _ in range(400):
            cx = rng.uniform(60.0, width - 60.0)
            cy = rng.uniform(60.0, height - 60.0)
            nodes = _ring_nodes(cx, cy, ring_r, nv, rot)
            ex, ey, er = _extent(nodes)
            if place(ex, ey, er):
                components.append({"nodes": nodes, "closed": True})
                break

    for _ in range(n_trees):
        m = int(rng.randint(3, 5))
        for _ in range(400):
            cx = rng.uniform(70.0, width - 70.0)
            cy = rng.uniform(70.0, height - 70.0)
            length = rng.uniform(55.0, 85.0)
            ang = rng.random() * 2.0 * math.pi
            step = length / (m - 1)
            nodes = [
                [
                    cx - math.cos(ang) * length / 2.0 + math.cos(ang) * step * i,
                    cy - math.sin(ang) * length / 2.0 + math.sin(ang) * step * i,
                ]
                for i in range(m)
            ]
            ex, ey, er = _extent(nodes)
            if place(ex, ey, er):
                components.append({"nodes": nodes, "closed": False})
                break

    return {
        "components": components,
        "width": width,
        "height": height,
        "background": "white",
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    for comp in scene["components"]:
        nodes = comp["nodes"]
        n = len(nodes)
        if comp["closed"]:
            pairs = [(i, (i + 1) % n) for i in range(n)]
        else:
            pairs = [(i, i + 1) for i in range(n - 1)]
        for (a, b) in pairs:
            ax, ay = nodes[a]
            bx, by = nodes[b]
            draw.line(
                [(ax * ss, ay * ss), (bx * ss, by * ss)],
                fill="black",
                width=int(LINK_WIDTH * ss),
            )
        for (x, y) in nodes:
            draw.ellipse(
                [(x - NODE_RADIUS) * ss, (y - NODE_RADIUS) * ss,
                 (x + NODE_RADIUS) * ss, (y + NODE_RADIUS) * ss],
                fill="black",
            )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> int:
    """Number of independent cycles = number of closed-ring components."""
    return sum(1 for comp in scene["components"] if comp["closed"])


def margin(scene: dict) -> float:
    """Robustness margin: the minimum clearance between any two ink features.

    Small values denote scenes whose recovered cycle count (the sealed-pocket
    count) is fragile at pixel scale and is therefore quarantined.
    """
    width = float(scene["width"])
    height = float(scene["height"])
    centers: list[tuple[float, float, float]] = []
    for comp in scene["components"]:
        ex, ey, er = _extent(comp["nodes"])
        centers.append((ex, ey, er))
    if not centers:
        return float(min(width, height)) / 2.0
    best: float = float("inf")
    for (ex, ey, er) in centers:
        best = min(best, ex - er, width - ex - er, ey - er, height - ey - er)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            (ax, ay, ar) = centers[i]
            (bx, by, br) = centers[j]
            best = min(best, math.hypot(ax - bx, ay - by) - ar - br)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose distinguishing clearance is too thin to trust."""
    if margin(scene) < QUARANTINE_MIN:
        return True
    return False


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render symmetry: every node coordinate directly changes the
    picture and the answer. Rotating the whole scene would change every stored
    coordinate and is not a label-preserving pixel-identical transform of this
    rooted image; declared not_applicable.
    """
    return iter(())
