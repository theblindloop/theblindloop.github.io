"""Renderer and analytic label for exact enclosure-depth matching.

The public latent scene contains only visible rings and colored marker disks.
The analytic label counts, for each marker center, how many ring centerlines
strictly contain it.  No answer or role field is stored in the scene.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "enclosure_depth_match-renderer-0.1.0"
WIDTH, HEIGHT = 760, 520
SUPERSAMPLE = 4
BACKGROUND = (249, 248, 244)
INK = (31, 32, 37)
COLORS: dict[str, tuple[int, int, int]] = {
    "red": (211, 66, 72),
    "green": (54, 157, 93),
    "blue": (68, 105, 198),
    "orange": (224, 132, 46),
}
REFERENCE = "green"
CANDIDATES = ("red", "blue", "orange")
QUARANTINE_CLEARANCE = 10.0
SAMPLING_CLEARANCE = 16.0


def _depth_at(x: float, y: float, rings: list[dict[str, Any]]) -> int:
    return sum(
        math.hypot(x - float(ring["cx"]), y - float(ring["cy"])) < float(ring["radius"])
        for ring in rings
    )


def _ring_radii(scene: dict[str, Any]) -> list[float]:
    return sorted(float(ring["radius"]) for ring in scene["rings"])


def _boundary_clearance(scene: dict[str, Any]) -> float:
    values = []
    for marker in scene["markers"]:
        for ring in scene["rings"]:
            distance = math.hypot(
                float(marker["x"]) - float(ring["cx"]),
                float(marker["y"]) - float(ring["cy"]),
            )
            values.append(abs(distance - float(ring["radius"])))
    radii = _ring_radii(scene)
    strokes = sorted(float(ring["stroke"]) for ring in scene["rings"])
    for left, right, left_stroke, right_stroke in zip(
        radii, radii[1:], strokes, strokes[1:]
    ):
        values.append(right - left - 0.5 * (left_stroke + right_stroke))
    return float(min(values)) if values else 0.0


def _valid_marker_positions(
    markers: list[dict[str, Any]],
    width: int,
    height: int,
) -> bool:
    for marker in markers:
        radius = float(marker["radius"])
        if not (radius + 4.0 <= float(marker["x"]) <= width - radius - 4.0):
            return False
        if not (radius + 4.0 <= float(marker["y"]) <= height - radius - 4.0):
            return False
    for index, first in enumerate(markers):
        for second in markers[index + 1 :]:
            spacing = math.hypot(
                float(first["x"]) - float(second["x"]),
                float(first["y"]) - float(second["y"]),
            )
            if spacing < float(first["radius"]) + float(second["radius"]) + 22.0:
                return False
    return True


def _radial_interval(radii: list[float], depth: int) -> tuple[float, float]:
    """Return a safe radial interval for an exact concentric-ring depth."""

    count = len(radii)
    if depth == 0:
        return radii[-1] + SAMPLING_CLEARANCE + 10.0, radii[-1] + 42.0
    if depth == count:
        return 12.0, radii[0] - SAMPLING_CLEARANCE - 10.0
    lower = radii[count - depth - 1] + SAMPLING_CLEARANCE + 4.0
    upper = radii[count - depth] - SAMPLING_CLEARANCE - 4.0
    return lower, upper


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, high-clearance scene from one integer seed."""

    rng = random.Random(int(seed))
    width, height = WIDTH, HEIGHT
    for _ in range(5000):
        ring_count = rng.choice((3, 4))
        inner = rng.uniform(50.0, 58.0) if ring_count == 4 else rng.uniform(58.0, 68.0)
        gap_floor, gap_ceil = (42.0, 50.0) if ring_count == 4 else (48.0, 58.0)
        radii = [inner]
        for _index in range(1, ring_count):
            radii.append(radii[-1] + rng.uniform(gap_floor, gap_ceil))
        outer = radii[-1]
        if outer > 205.0:
            continue
        cx = width / 2.0 + rng.uniform(-22.0, 22.0)
        cy = height / 2.0 + rng.uniform(-14.0, 14.0)
        strokes = [rng.uniform(6.0, 9.0) for _index in range(ring_count)]
        rings = [
            {"cx": cx, "cy": cy, "radius": radius, "stroke": stroke}
            for radius, stroke in zip(radii, strokes)
        ]
        if min(
            radii[index + 1] - radii[index] - 0.5 * (strokes[index] + strokes[index + 1])
            for index in range(ring_count - 1)
        ) < 28.0:
            continue

        reference_depth = rng.randrange(ring_count + 1)
        other_depths = [depth for depth in range(ring_count + 1) if depth != reference_depth]
        candidate_depths = rng.sample(other_depths, 3)
        answer_color = CANDIDATES[int(seed) % len(CANDIDATES)]
        depth_by_color = {REFERENCE: reference_depth}
        for color, depth in zip(CANDIDATES, candidate_depths):
            depth_by_color[color] = reference_depth if color == answer_color else depth

        colors = [REFERENCE, *CANDIDATES]
        rng.shuffle(colors)
        markers: list[dict[str, Any]] = []
        for color in colors:
            low, high = _radial_interval(radii, depth_by_color[color])
            if high <= low + 2.0:
                break
            radial = rng.uniform(low, high)
            angle = rng.uniform(0.0, 2.0 * math.pi)
            markers.append(
                {
                    "color": color,
                    "x": cx + radial * math.cos(angle),
                    "y": cy + radial * math.sin(angle),
                    "radius": rng.uniform(8.5, 14.0),
                }
            )
        if len(markers) != 4 or not _valid_marker_positions(markers, width, height):
            continue
        scene = {
            "canvas": [width, height],
            "ring_count": ring_count,
            "rings": rings,
            "markers": markers,
        }
        if _boundary_clearance(scene) < SAMPLING_CLEARANCE:
            continue
        if analytic_gold(scene) not in CANDIDATES:
            continue
        return scene
    raise RuntimeError("could not sample a separated exact-depth scene")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the public scene fields into a deterministic RGB PNG."""

    width, height = (int(value) for value in scene["canvas"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for ring in scene["rings"]:
        cx = float(ring["cx"]) * scale
        cy = float(ring["cy"]) * scale
        radius = float(ring["radius"]) * scale
        box = (cx - radius, cy - radius, cx + radius, cy + radius)
        draw.ellipse(
            box,
            outline=INK,
            width=max(2, int(round(float(ring["stroke"]) * scale))),
        )
    for marker in scene["markers"]:
        cx = float(marker["x"]) * scale
        cy = float(marker["y"]) * scale
        radius = float(marker["radius"]) * scale
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=COLORS[str(marker["color"])],
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique candidate with the reference marker's exact depth."""

    markers = {str(marker["color"]): marker for marker in scene["markers"]}
    if set(markers) != {REFERENCE, *CANDIDATES}:
        raise ValueError(f"scene must contain exactly the four marker colors: {sorted(markers)}")
    reference = markers[REFERENCE]
    reference_depth = _depth_at(float(reference["x"]), float(reference["y"]), scene["rings"])
    matches = [
        color
        for color in CANDIDATES
        if _depth_at(float(markers[color]["x"]), float(markers[color]["y"]), scene["rings"])
        == reference_depth
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one depth match, got {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest marker-to-ring or ring-to-ring clearance in pixels."""

    return _boundary_clearance(scene)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are claimed: all label inputs are visible geometry."""

    return []
