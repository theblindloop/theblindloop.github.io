"""Deterministic scene sampling and rendering for common_cover_cells.

One graduated rail carries evenly spaced ticks that cut it into unit cells.
Five coloured bars stand above the rail, each covering the stretch of rail
directly beneath it. The common stretch is the intersection of all five covered
stretches: it starts at the rightmost left end and ends at the leftmost right
end, so it is delimited by two different bars and is drawn nowhere. The answer
is how many unit cells lie completely inside that undrawn stretch.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 560
X0 = 60
UNIT = 44
CELLS = 20
X1 = X0 + CELLS * UNIT
RAIL_Y = 486
TICK_HALF = 13
RAIL_RGB = (70, 70, 70)
DECOY_RGB = (198, 198, 198)
BACKGROUND = (255, 255, 255)
ROW_Y = (60, 138, 216, 294, 372)
BAR_H = 34
DECOY_TOP, DECOY_BOTTOM = 48, 412

COLOR_NAMES = ("red", "green", "blue", "orange", "purple")
COLOR_RGB = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}

MIN_FRACTION = 0.16
MAX_FRACTION = 0.84
QUARANTINE_MARGIN = 0.20
COUNTS = (2, 3, 4, 5)
LOW_CELL_RANGE = (4, 11)
SPREAD_BASE = 3.6
SPREAD_SLOPE = 0.6


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def _fraction(rng: random.Random) -> float:
    """A within-cell offset that never lands near a tick."""

    return rng.uniform(MIN_FRACTION, MAX_FRACTION)


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene with a controlled cell count."""

    rng = random.Random(f"common_cover_cells:{int(seed)}")
    count = rng.choice(COUNTS)
    slack = rng.choice((1, 2))
    low = rng.randint(*LOW_CELL_RANGE)
    spread = SPREAD_BASE - SPREAD_SLOPE * count
    start = low + _fraction(rng)
    high = low + 1 + count
    finish = high + _fraction(rng)

    inner_left = low + 1 - slack
    inner_right = high + slack
    left_lo = inner_left - spread
    right_hi = inner_right + spread

    def outer_left() -> float:
        return rng.uniform(left_lo, inner_left - MIN_FRACTION)

    def outer_right() -> float:
        return rng.uniform(inner_right + MIN_FRACTION, right_hi)

    names = list(COLOR_NAMES)
    rng.shuffle(names)
    opener, closer = names[0], names[1]
    rows = list(range(len(ROW_Y)))
    rng.shuffle(rows)

    bars: list[dict] = []
    for index, name in enumerate(COLOR_NAMES):
        left = start if name == opener else outer_left()
        right = finish if name == closer else outer_right()
        bars.append(
            {"color": name, "row": rows[index], "l": float(left), "r": float(right)}
        )

    free = [bar for bar in bars if bar["color"] not in (opener, closer)]
    decoys = [
        float(rng.choice(free)["l"]),
        float(rng.choice(free)["r"]),
    ]
    order = list(range(len(bars)))
    rng.shuffle(order)
    return {"cells": CELLS, "bars": [bars[i] for i in order], "decoys": decoys}


# --------------------------------------------------------------------------- #
# gold
# --------------------------------------------------------------------------- #
def _stretch(scene: dict) -> tuple[float, float]:
    """The undrawn common stretch in cell units."""

    start = max(float(bar["l"]) for bar in scene["bars"])
    finish = min(float(bar["r"]) for bar in scene["bars"])
    return start, finish


def analytic_gold(scene: dict) -> str:
    """Number of unit cells lying completely inside the common stretch."""

    start, finish = _stretch(scene)
    return str(max(0, math.floor(finish) - math.ceil(start)))


def margin(scene: dict) -> float:
    """Cell-unit distance to the nearest configuration with a different count."""

    start, finish = _stretch(scene)
    gaps = []
    for value in (start, finish):
        gaps.append(min(value - math.floor(value), math.ceil(value) - value))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes are excluded from headline claims."""

    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Nontrivial latent changes that leave the raster byte-identical."""

    bars = [dict(bar) for bar in scene["bars"]]
    reversed_scene = {
        "cells": scene["cells"],
        "bars": list(reversed(bars)),
        "decoys": list(scene["decoys"]),
    }
    rotated_scene = {
        "cells": scene["cells"],
        "bars": bars[2:] + bars[:2],
        "decoys": list(scene["decoys"]),
    }
    swapped_decoys = {
        "cells": scene["cells"],
        "bars": bars,
        "decoys": list(reversed(list(scene["decoys"]))),
    }
    return [
        ("bar_record_order_reversed", reversed_scene),
        ("bar_record_order_rotated", rotated_scene),
        ("decoy_order_reversed", swapped_decoys),
    ]


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _px(value: float) -> int:
    return int(round(X0 + float(value) * UNIT))


def render(scene: dict) -> Image.Image:
    """Rasterise one scene; the common stretch itself is never drawn."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rectangle([X0, RAIL_Y - 1, X1, RAIL_Y + 1], fill=RAIL_RGB)
    for index in range(int(scene["cells"]) + 1):
        x = X0 + index * UNIT
        draw.rectangle([x - 1, RAIL_Y - TICK_HALF, x + 1, RAIL_Y + TICK_HALF], fill=RAIL_RGB)

    for value in scene["decoys"]:
        x = _px(value)
        draw.rectangle([x - 1, DECOY_TOP, x + 1, DECOY_BOTTOM], fill=DECOY_RGB)

    for bar in scene["bars"]:
        top = ROW_Y[int(bar["row"])]
        draw.rectangle(
            [_px(bar["l"]), top, _px(bar["r"]), top + BAR_H - 1],
            fill=COLOR_RGB[bar["color"]],
        )
    return image
