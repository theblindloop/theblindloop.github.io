"""Pixel-only decision arm for common_cover_cells.

Receives the final PNG and nothing else: no scene, no renderer constants beyond
the published palette, no analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_PALETTE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
_COLOR_TOLERANCE = 30
_DARK = 120
_MIN_TICKS = 8
_MIN_BAR_PIXELS = 2000
_MAX_BAR_ROWS = 60
_TICK_SCAN_OFFSET = 8
_MIN_CLEARANCE = 4.0
_SPACING_SLACK = 3.0


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    """Inclusive index ranges of consecutive true entries."""

    out: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _ticks(dark: np.ndarray) -> list[float] | None:
    """Centres of the graduation ticks recovered from the rail band."""

    per_row = dark.sum(axis=1)
    rail_row = int(np.argmax(per_row))
    scan = rail_row - _TICK_SCAN_OFFSET
    if scan < 0 or per_row[rail_row] < 200:
        return None
    centres = [(lo + hi) / 2.0 for lo, hi in _runs(dark[scan]) if hi - lo <= 6]
    if len(centres) < _MIN_TICKS:
        return None
    steps = [b - a for a, b in zip(centres, centres[1:])]
    if not steps or max(steps) - min(steps) > _SPACING_SLACK:
        return None
    return centres


def _extent(pixels: np.ndarray, colour: tuple[int, int, int]):
    """Horizontal extent of one coloured bar, or None when it is unusable."""

    target = np.array(colour, dtype=np.int16)
    mask = (np.abs(pixels.astype(np.int16) - target) <= _COLOR_TOLERANCE).all(axis=2)
    if int(mask.sum()) < _MIN_BAR_PIXELS:
        return None
    rows = np.flatnonzero(mask.any(axis=1))
    cols = np.flatnonzero(mask.any(axis=0))
    if rows[-1] - rows[0] + 1 > _MAX_BAR_ROWS:
        return None
    return float(cols[0]), float(cols[-1])


def decision_from_image(image: Image.Image) -> str:
    """Count the unit cells lying completely inside the common stretch."""

    pixels = np.asarray(image.convert("RGB"))
    dark = (pixels < _DARK).all(axis=2)
    centres = _ticks(dark)
    if centres is None:
        return ABSTAIN

    starts: list[float] = []
    finishes: list[float] = []
    for colour in _PALETTE.values():
        extent = _extent(pixels, colour)
        if extent is None:
            return ABSTAIN
        starts.append(extent[0])
        finishes.append(extent[1])
    if len(starts) != len(_PALETTE):
        return ABSTAIN

    start = max(starts)
    finish = min(finishes)
    if finish <= start:
        return ABSTAIN
    for edge in (start, finish):
        if min(abs(edge - centre) for centre in centres) < _MIN_CLEARANCE:
            return ABSTAIN

    inside = [index for index, centre in enumerate(centres) if start <= centre <= finish]
    if not inside:
        return "0"
    return str(max(0, inside[-1] - inside[0]))
