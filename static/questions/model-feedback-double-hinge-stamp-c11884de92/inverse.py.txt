"""Image-only inverse for the ordered double-hinge task.

Only exact raster colors are used to recover the token, target, hinge centers,
and the signed direction of each curved arrow.  The latent scene and renderer
are intentionally not imported.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 1020
HEIGHT = 740
PANEL_W = 480
PANEL_H = 340
MARGIN_X = 20
MARGIN_Y = 20
GAP_X = 20
GAP_Y = 20

SOURCE = (28, 65, 104)
TARGET_FILL = (255, 243, 211)
HINGE1_ARC = (20, 157, 190)
HINGE1_CORE = (5, 76, 116)
HINGE2_ARC = (224, 151, 38)
HINGE2_CORE = (126, 79, 10)


def _origin(slot: int) -> tuple[int, int]:
    return (MARGIN_X + (slot % 2) * (PANEL_W + GAP_X), MARGIN_Y + (slot // 2) * (PANEL_H + GAP_Y))


def _exact(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(color, dtype=np.uint8), axis=2)


def _points(mask: np.ndarray) -> np.ndarray:
    y, x = np.nonzero(mask)
    if len(x) == 0:
        raise ValueError("empty raster component")
    return np.column_stack((x.astype(np.float64), y.astype(np.float64)))


def _center(mask: np.ndarray) -> np.ndarray:
    return _points(mask).mean(axis=0)


def _turn_direction(mask: np.ndarray, center: np.ndarray) -> str:
    points = _points(mask)
    radii = np.linalg.norm(points - center[None, :], axis=1)
    points = points[radii > 15.0]
    if len(points) < 20:
        raise ValueError("missing curved turn")
    return "cw" if float((points[:, 1] - center[1]).mean()) < 0.0 else "ccw"


def _quarter(points: np.ndarray, pivot: np.ndarray, direction: str) -> np.ndarray:
    px, py = np.rint(pivot).astype(np.int64)
    x = np.rint(points[:, 0]).astype(np.int64)
    y = np.rint(points[:, 1]).astype(np.int64)
    dx, dy = x - px, y - py
    if direction == "cw":
        return np.column_stack((px - dy, py + dx))
    if direction == "ccw":
        return np.column_stack((px + dy, py - dx))
    raise ValueError(direction)


def _transformed_mask(source: np.ndarray, hinge1: np.ndarray, turn1: str, hinge2: np.ndarray, turn2: str) -> set[tuple[int, int]]:
    points = _points(source)
    points = _quarter(points, hinge1, turn1)
    points = _quarter(points, hinge2, turn2)
    height, width = source.shape
    inside = (
        (points[:, 0] >= 0)
        & (points[:, 0] < width)
        & (points[:, 1] >= 0)
        & (points[:, 1] < height)
    )
    return {(int(x), int(y)) for x, y in points[inside]}


def _overlap_score(predicted: set[tuple[int, int]], target: np.ndarray) -> float:
    target_points = {(int(x), int(y)) for x, y in np.column_stack(np.nonzero(target))[:, ::-1]}
    intersection = len(predicted & target_points)
    union = len(predicted | target_points)
    return intersection / union if union else -1.0


def _panel_score(arr: np.ndarray, slot: int) -> float:
    x0, y0 = _origin(slot)
    crop = arr[y0 : y0 + PANEL_H + 1, x0 : x0 + PANEL_W + 1]
    source = _exact(crop, SOURCE)
    target = _exact(crop, TARGET_FILL)
    hinge1_arc = _exact(crop, HINGE1_ARC)
    hinge1_core = _exact(crop, HINGE1_CORE)
    hinge2_arc = _exact(crop, HINGE2_ARC)
    hinge2_core = _exact(crop, HINGE2_CORE)
    if source.sum() < 500 or target.sum() < 500:
        raise ValueError("missing token or target")
    if hinge1_arc.sum() < 80 or hinge2_arc.sum() < 80 or hinge1_core.sum() < 20 or hinge2_core.sum() < 20:
        raise ValueError("missing hinge control")
    center1 = _center(hinge1_core)
    center2 = _center(hinge2_core)
    turn1 = _turn_direction(hinge1_arc, center1)
    turn2 = _turn_direction(hinge2_arc, center2)
    predicted = _transformed_mask(source, center1, turn1, center2, turn2)
    if not predicted:
        raise ValueError("empty transformed token")
    target_center = _center(target)
    predicted_center = np.asarray(
        [sum(point[0] for point in predicted) / len(predicted), sum(point[1] for point in predicted) / len(predicted)]
    )
    center_error = float(np.linalg.norm(predicted_center - target_center))
    return _overlap_score(predicted, target) - 0.0015 * center_error


def decision_from_image(image: Image.Image) -> str:
    """Select the tile with the best complete-silhouette match."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected image size {arr.shape[:2]}")
    scores = [_panel_score(arr, slot) for slot in range(4)]
    winner = max(range(4), key=lambda index: scores[index])
    return f"tile-{chr(65 + winner)}"
