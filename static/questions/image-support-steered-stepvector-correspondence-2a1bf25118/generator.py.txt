"""Deterministic scene sampler and renderer for Right-Step Tracker.

The latent scene contains only visible fields.  Analytic gold is derived from
the two rendered shape-to-cell maps and the rendered destination color map.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 900
HEIGHT = 430
BACKGROUND = [248, 246, 240]
SHAPES = ("circle", "square", "triangle")
ANSWER_COLORS = ("amber", "cyan", "violet")

RGB = {
    "dark": (35, 40, 48),
    "grid": (166, 174, 186),
    "accent": (218, 221, 224),
    "amber": (222, 145, 0),
    "cyan": (0, 154, 188),
    "violet": (130, 83, 190),
}

LEFT_X = 42
RIGHT_X = 562
TOP_Y = 78
PITCH = 74
CELL = 62


def _score(before: list[int], after: list[int]) -> int:
    """Manhattan distance of a cell displacement from (row 0, column +1)."""
    dr = after[0] - before[0]
    dc = after[1] - before[1]
    return abs(dr) + abs(dc - 1)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, unambiguous three-shape correspondence scene."""
    rng = random.Random(int(seed))
    target = SHAPES[rng.randrange(len(SHAPES))]
    desired_margin = 1 + (abs(int(seed)) % 3)

    # Resample visible placements until the requested runner-up gap is present.
    for _ in range(500):
        target_before = [rng.randrange(4), rng.randrange(3)]
        before_pool = [[r, c] for r in range(4) for c in range(4)]
        before_pool.remove(target_before)
        rng.shuffle(before_pool)
        before_cells: dict[str, list[int]] = {target: target_before}
        others = [shape for shape in SHAPES if shape != target]
        before_cells[others[0]] = before_pool[0]
        before_cells[others[1]] = before_pool[1]

        target_after = [target_before[0], target_before[1] + 1]
        choices: dict[str, list[list[int]]] = {}
        for shape in others:
            options = [
                [r, c]
                for r in range(4)
                for c in range(4)
                if [r, c] != target_after
                and _score(before_cells[shape], [r, c]) >= desired_margin
            ]
            rng.shuffle(options)
            choices[shape] = options

        pairs = [
            (a, b)
            for a in choices[others[0]]
            for b in choices[others[1]]
            if a != b
            and min(
                _score(before_cells[others[0]], a),
                _score(before_cells[others[1]], b),
            )
            == desired_margin
        ]
        if not pairs:
            continue
        a, b = rng.choice(pairs)
        after_cells = {target: target_after, others[0]: a, others[1]: b}
        break
    else:  # pragma: no cover - defensive, the finite construction is roomy.
        raise RuntimeError("could not sample a separated displacement scene")

    # Consecutive generator seeds exactly counterbalance answer identity.
    answer_color = ANSWER_COLORS[abs(int(seed)) % len(ANSWER_COLORS)]
    remaining_colors = [color for color in ANSWER_COLORS if color != answer_color]
    rng.shuffle(remaining_colors)
    tag_colors = {
        target: answer_color,
        others[0]: remaining_colors[0],
        others[1]: remaining_colors[1],
    }

    return {
        "before_cells": {shape: list(before_cells[shape]) for shape in SHAPES},
        "after_cells": {shape: list(after_cells[shape]) for shape in SHAPES},
        "tag_colors": {shape: tag_colors[shape] for shape in SHAPES},
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }


def _cell_box(origin_x: int, row: int, col: int) -> tuple[int, int, int, int]:
    x0 = origin_x + col * PITCH
    y0 = TOP_Y + row * PITCH
    return (x0, y0, x0 + CELL, y0 + CELL)


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, cx: int, cy: int) -> None:
    dark = RGB["dark"]
    if shape == "circle":
        draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=dark)
    elif shape == "square":
        draw.rectangle((cx - 15, cy - 15, cx + 15, cy + 15), fill=dark)
    elif shape == "triangle":
        draw.polygon(((cx, cy - 18), (cx - 18, cy + 16), (cx + 18, cy + 16)), fill=dark)
    else:  # pragma: no cover
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render from visible scene fields only, with crisp separable palette ink."""
    before = scene["before_cells"]
    after = scene["after_cells"]
    tags = scene["tag_colors"]
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background"])

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)

    # Every cell is an independent outline component.  The gap between cells is
    # visible evidence for the discrete coordinate system.
    for origin_x in (LEFT_X, RIGHT_X):
        for row in range(4):
            for col in range(4):
                draw.rectangle(_cell_box(origin_x, row, col), outline=RGB["grid"], width=3)

    # Short disconnected header motifs distinguish BEFORE (one dash) from AFTER
    # (two dashes) without requiring OCR; the question also declares left/right.
    draw.rounded_rectangle((132, 38, 232, 50), radius=5, fill=RGB["accent"])
    draw.rounded_rectangle((624, 38, 694, 50), radius=5, fill=RGB["accent"])
    draw.rounded_rectangle((710, 38, 780, 50), radius=5, fill=RGB["accent"])

    for shape in SHAPES:
        row, col = before[shape]
        x0, y0, x1, y1 = _cell_box(LEFT_X, int(row), int(col))
        _draw_shape(draw, shape, (x0 + x1) // 2, (y0 + y1) // 2)

        row, col = after[shape]
        x0, y0, x1, y1 = _cell_box(RIGHT_X, int(row), int(col))
        _draw_shape(draw, shape, (x0 + x1) // 2, (y0 + y1) // 2)
        # Equal-area, spatially separate color tabs; each destination shape has one.
        draw.rounded_rectangle((x1 - 17, y0 + 6, x1 - 5, y0 + 17), radius=2, fill=RGB[tags[shape]])

    return image


def _facts(scene: dict[str, Any]) -> tuple[dict, dict, dict, list[str]]:
    before = scene["before_cells"]
    after = scene["after_cells"]
    tags = scene["tag_colors"]
    matches = [shape for shape in SHAPES if _score(before[shape], after[shape]) == 0]
    return before, after, tags, matches


def analytic_gold(scene: dict[str, Any]) -> str:
    before = scene["before_cells"]
    after = scene["after_cells"]
    tags = scene["tag_colors"]
    matches = [shape for shape in SHAPES if _score(before[shape], after[shape]) == 0]
    if len(matches) != 1:
        raise ValueError("requested displacement is not unique")
    answer = tags[matches[0]]
    if answer not in ANSWER_COLORS:
        raise ValueError("invalid answer color")
    return answer


def margin(scene: dict[str, Any]) -> float:
    before, after, _tags, matches = _facts(scene)
    if len(matches) != 1:
        return 0.0
    runner_up = min(_score(before[shape], after[shape]) for shape in SHAPES if shape != matches[0])
    return float(runner_up)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        before, after, tags, matches = _facts(scene)
        return not (
            len(matches) == 1
            and len({tuple(before[s]) for s in SHAPES}) == 3
            and len({tuple(after[s]) for s in SHAPES}) == 3
            and set(tags) == set(SHAPES)
            and set(tags.values()) == set(ANSWER_COLORS)
            and all(len(before[s]) == 2 and len(after[s]) == 2 for s in SHAPES)
            and all(0 <= v < 4 for s in SHAPES for v in (*before[s], *after[s]))
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases: every declared scene field is visually expressed."""
    return []
