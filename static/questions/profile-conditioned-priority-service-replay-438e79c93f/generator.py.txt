"""Deterministic sampler, analytic replay, and Pillow renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1040, 640
LABELS = ("A", "B", "C", "D")
BASE_COLORS = ("#4C78A8", "#E45756", "#54A24B", "#F2CF5B")
CARD_FILL = "#F4F6F8"
INK = "#17212B"
MUTED = "#5B6773"


def _replay(scene: dict) -> tuple[list[str], list[str]]:
    waiting = set(scene["initial"])
    priority = list(scene["priority_order"])
    served: list[str] = []
    for operation in scene["operations"]:
        if operation["kind"] == "add":
            waiting.add(operation["label"])
        elif operation["kind"] == "serve":
            if not waiting:
                raise ValueError("SERVE encountered an empty waiting set")
            chosen = next(label for label in priority if label in waiting)
            waiting.remove(chosen)
            served.append(chosen)
        else:
            raise ValueError("unknown operation kind")
    ordered = [label for label in priority if label in waiting]
    if len(ordered) < 2:
        raise ValueError("final waiting set must have a runner-up")
    return ordered, served


def analytic_gold(scene: dict) -> str:
    return _replay(scene)[0][0]


def margin(scene: dict) -> float:
    ordered, _ = _replay(scene)
    priority = list(scene["priority_order"])
    return float(priority.index(ordered[1]) - priority.index(ordered[0]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def sample_scene(seed: int) -> dict:
    """Condition on a balanced answer while keeping counts and operation types fixed."""
    target = LABELS[seed % len(LABELS)]
    target_priority_position = (seed // len(LABELS)) % 2
    rng = random.Random(seed ^ 0x6A09E667)
    for _ in range(20000):
        priority = list(LABELS)
        rng.shuffle(priority)
        if priority.index(target) != target_priority_position:
            continue
        initial = rng.sample(list(LABELS), 3)
        waiting = set(initial)
        operations: list[dict] = []
        adds_left = 4
        serves_left = 4
        valid = True
        while adds_left or serves_left:
            kinds = []
            if adds_left and len(waiting) < 4:
                kinds.append("add")
            if serves_left and waiting:
                kinds.append("serve")
            if not kinds:
                valid = False
                break
            kind = rng.choice(kinds)
            if kind == "add":
                label = rng.choice([label for label in LABELS if label not in waiting])
                waiting.add(label)
                operations.append({"kind": "add", "label": label})
                adds_left -= 1
            else:
                chosen = next(label for label in priority if label in waiting)
                waiting.remove(chosen)
                operations.append({"kind": "serve"})
                serves_left -= 1
        if not valid or len(waiting) != 3:
            continue
        ordered = [label for label in priority if label in waiting]
        if ordered[0] != target:
            continue
        colors = list(BASE_COLORS)
        rng.shuffle(colors)
        shift = [rng.randint(-8, 8), rng.randint(-6, 6)]
        candidate = {
            "initial": initial,
            "operations": operations,
            "priority_order": priority,
            "palette": colors,
            "shift": shift,
        }
        _replay(candidate)
        return candidate
    raise RuntimeError(f"could not sample balanced scene for seed {seed}")


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size=size)


def _chip(draw: ImageDraw.ImageDraw, x: int, y: int, color: str, radius: int = 20) -> None:
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color, outline=INK, width=3)


def render(scene: dict) -> Image.Image:
    initial = list(scene["initial"])
    operations = list(scene["operations"])
    priority = list(scene["priority_order"])
    palette = list(scene["palette"])
    dx, dy = scene["shift"]
    if sorted(initial) not in (["A", "B", "C"], ["A", "B", "D"], ["A", "C", "D"], ["B", "C", "D"]):
        raise ValueError("initial rack must contain three distinct labels")
    if sorted(priority) != list(LABELS) or len(palette) != 4:
        raise ValueError("invalid priority or palette")
    color_for = dict(zip(LABELS, palette))
    image = Image.new("RGB", (WIDTH, HEIGHT), "#FFFFFF")
    draw = ImageDraw.Draw(image)
    title_font = _font(28, True)
    section_font = _font(17, True)
    label_font = _font(18, True)
    small_font = _font(14, True)
    icon_font = _font(36, True)

    def p(x: int, y: int) -> tuple[int, int]:
        return x + int(dx), y + int(dy)

    draw.text(p(520, 34), "PRIORITY SERVICE LEDGER", fill=INK, font=title_font, anchor="mm")
    draw.line((*p(70, 61), *p(970, 61)), fill="#D8DEE5", width=2)

    draw.text(p(105, 108), "COLOR LEGEND", fill=MUTED, font=section_font, anchor="lm")
    legend_x = [430, 520, 610, 700]
    for index, label in enumerate(LABELS):
        x, y = p(legend_x[index], 102)
        _chip(draw, x, y, color_for[label])
        draw.text(p(legend_x[index], 137), label, fill=INK, font=label_font, anchor="mm")

    draw.text(p(105, 198), "PRIORITY", fill=MUTED, font=section_font, anchor="lm")
    draw.text(p(273, 198), "FIRST", fill=INK, font=small_font, anchor="mm")
    priority_x = [400, 500, 600, 700]
    draw.line((*p(344, 192), *p(756, 192)), fill=INK, width=3)
    draw.polygon([p(756, 192), p(741, 184), p(741, 200)], fill=INK)
    draw.text(p(824, 198), "LAST", fill=INK, font=small_font, anchor="mm")
    for x0, label in zip(priority_x, priority):
        x, y = p(x0, 192)
        _chip(draw, x, y, color_for[label])

    draw.rounded_rectangle((*p(75, 238), *p(965, 330)), radius=14, fill="#FAFBFC", outline="#C7D0D9", width=2)
    draw.text(p(105, 284), "START WAITING", fill=MUTED, font=section_font, anchor="lm")
    initial_x = [470, 570, 670]
    for x0, label in zip(initial_x, initial):
        x, y = p(x0, 284)
        _chip(draw, x, y, color_for[label], radius=23)

    draw.text(p(74, 367), "PROCESS CARDS LEFT TO RIGHT", fill=MUTED, font=section_font, anchor="lm")
    card_w, card_h, gap = 103, 190, 10
    left, top = 63, 390
    for index, operation in enumerate(operations):
        x0 = left + index * (card_w + gap)
        x1 = x0 + card_w
        draw.rounded_rectangle((*p(x0, top), *p(x1, top + card_h)), radius=10, fill=CARD_FILL, outline="#8E9AA6", width=2)
        draw.text(p(x0 + card_w // 2, top + 18), str(index + 1), fill=MUTED, font=small_font, anchor="mm")
        draw.line((*p(x0 + 14, top + 36), *p(x1 - 14, top + 36)), fill="#CDD4DB", width=2)
        if operation["kind"] == "add":
            label = operation["label"]
            draw.text(p(x0 + card_w // 2, top + 72), "+", fill=INK, font=icon_font, anchor="mm")
            x, y = p(x0 + card_w // 2, top + 132)
            _chip(draw, x, y, color_for[label], radius=20)
            draw.text(p(x0 + card_w // 2, top + 168), "ADD", fill=MUTED, font=small_font, anchor="mm")
        else:
            draw.text(p(x0 + card_w // 2, top + 92), "−", fill=INK, font=icon_font, anchor="mm")
            draw.text(p(x0 + card_w // 2, top + 142), "SERVE", fill=MUTED, font=small_font, anchor="mm")
    return image


def palette_equivalent(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["palette"] = twin["palette"][1:] + twin["palette"][:1]
    return twin


def translated_equivalent(scene: dict, dx: int = 5, dy: int = -4) -> dict:
    twin = deepcopy(scene)
    twin["shift"] = [scene["shift"][0] + dx, scene["shift"][1] + dy]
    return twin
