"""Renderer for a visible one-token counterfactual transfer.

The image contains two labelled trays of large navy tokens.  One token has a
gold ring, and a gold arrow leaves that token and points into the other tray.
The question asks for the larger tray *after* that move, so the gold cue is an
intervention rather than a decorative attribute of the initial scene.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "counterfactual_tray_transfer-0.1.0"

WIDTH = 720
HEIGHT = 520
BACKGROUND = (255, 255, 255)
TRAY_FILL = (244, 247, 251)
TRAY_BORDER = (54, 70, 92)
TOKEN = (31, 79, 137)
GOLD = (232, 157, 24)
INK = (38, 48, 62)


def sample_scene(seed: int) -> dict:
    """Return one deterministic, balanced, non-tied scene."""

    rng = random.Random((int(seed) * 6364136223846793005 + 1442695040888963407) & ((1 << 64) - 1))
    source_side = "left" if rng.randrange(2) == 0 else "right"
    final_left = rng.randrange(2) == 0
    level = rng.randrange(3)

    # The four cases keep the post-move answer one token away from a tie.  In
    # half of them the move reverses the initial majority, so an initial-count
    # heuristic is systematically wrong on a visible subset.
    if source_side == "left":
        if final_left:
            left_count, right_count = 4 + level, 1 + level
        else:
            left_count, right_count = 3 + level, 2 + level
    else:
        if final_left:
            left_count, right_count = 2 + level, 3 + level
        else:
            left_count, right_count = 1 + level, 4 + level

    source_count = left_count if source_side == "left" else right_count
    return {
        "left_count": int(left_count),
        "right_count": int(right_count),
        "source_side": source_side,
        "selected_index": int(rng.randrange(source_count)),
        "layout_variant": int(rng.randrange(3)),
        "width": WIDTH,
        "height": HEIGHT,
        "tray_width": 250,
        "tray_height": 270,
        "token_radius": 21 + int(rng.randrange(2)),
        "token_gap": 72 + int(rng.randrange(5)),
    }


def _tray_boxes(scene: dict) -> tuple[tuple[float, float, float, float], tuple[float, float, float, float]]:
    width = float(scene["width"])
    height = float(scene["height"])
    tray_width = float(scene["tray_width"])
    tray_height = float(scene["tray_height"])
    left_x = 52.0
    right_x = width - 52.0 - tray_width
    top = (height - tray_height) / 2.0 + 40.0
    return (
        (left_x, top, left_x + tray_width, top + tray_height),
        (right_x, top, right_x + tray_width, top + tray_height),
    )


def _slot_centres(box: tuple[float, float, float, float], scene: dict) -> list[tuple[float, float]]:
    x0, y0, _, _ = box
    gap = float(scene["token_gap"])
    variant = int(scene["layout_variant"])
    centres: list[tuple[float, float]] = []
    for row in range(2):
        for column in range(3):
            x = x0 + 62.0 + column * gap
            y = y0 + 82.0 + row * 94.0
            if variant == 1:
                x += (6.0 if column == 1 else -3.0 if column == 2 else 0.0)
                y += 4.0 if column % 2 else -2.0
            elif variant == 2:
                x += -5.0 if row == 0 else 5.0
                y += 3.0 if column == 0 else -3.0
            centres.append((x, y))
    return centres


def _selected_centre(scene: dict, boxes: tuple[tuple[float, float, float, float], tuple[float, float, float, float]]) -> tuple[float, float]:
    side_index = 0 if scene["source_side"] == "left" else 1
    return _slot_centres(boxes[side_index], scene)[int(scene["selected_index"])]


def _draw_arrow(draw: ImageDraw.ImageDraw, scene: dict, boxes: tuple[tuple[float, float, float, float], tuple[float, float, float, float]]) -> None:
    source_index = 0 if scene["source_side"] == "left" else 1
    target_index = 1 - source_index
    source = _selected_centre(scene, boxes)
    target_box = boxes[target_index]
    target_x = (target_box[0] + target_box[2]) / 2.0
    lane_y = target_box[1] - 58.0
    endpoint_y = target_box[1] - 8.0
    points = [source, (source[0], lane_y), (target_x, lane_y), (target_x, endpoint_y)]
    draw.line(points, fill=GOLD, width=6, joint="curve")
    draw.polygon(
        [(target_x - 13.0, endpoint_y - 18.0), (target_x + 13.0, endpoint_y - 18.0), (target_x, endpoint_y + 2.0)],
        fill=GOLD,
    )


def render(scene: dict) -> Image.Image:
    """Render the initial scene, including the visible transfer intervention."""

    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    boxes = _tray_boxes(scene)
    radius = float(scene["token_radius"])

    for box in boxes:
        draw.rounded_rectangle(box, radius=18, fill=TRAY_FILL, outline=TRAY_BORDER, width=5)

    _draw_arrow(draw, scene, boxes)

    font = ImageFont.load_default()
    left_box, right_box = boxes
    draw.text(((left_box[0] + left_box[2]) / 2.0, left_box[1] - 25.0), "LEFT TRAY", fill=INK, anchor="mm", font=font)
    draw.text(((right_box[0] + right_box[2]) / 2.0, right_box[1] - 25.0), "RIGHT TRAY", fill=INK, anchor="mm", font=font)
    action_box = (width / 2.0 - 32.0, 24.0, width / 2.0 + 32.0, 50.0)
    draw.rounded_rectangle(action_box, radius=8, fill=GOLD)
    draw.text((width / 2.0, 37.0), "MOVE", fill=(255, 255, 255), anchor="mm", font=font)

    counts = (int(scene["left_count"]), int(scene["right_count"]))
    selected_side = 0 if scene["source_side"] == "left" else 1
    selected_index = int(scene["selected_index"])
    for side, box in enumerate(boxes):
        centres = _slot_centres(box, scene)
        for index in range(counts[side]):
            cx, cy = centres[index]
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=TOKEN)
            if side == selected_side and index == selected_index:
                ring_radius = radius + 8.0
                draw.ellipse(
                    (cx - ring_radius, cy - ring_radius, cx + ring_radius, cy + ring_radius),
                    outline=GOLD,
                    width=5,
                )

    return image


def _post_move_counts(scene: dict) -> tuple[int, int]:
    left = int(scene["left_count"])
    right = int(scene["right_count"])
    if scene["source_side"] == "left":
        left -= 1
        right += 1
    else:
        left += 1
        right -= 1
    return left, right


def analytic_gold(scene: dict) -> str:
    """Return the fuller tray after applying the visible one-token move."""

    left, right = _post_move_counts(scene)
    if left > right:
        return "left"
    if right > left:
        return "right"
    return "tie"


def margin(scene: dict) -> float:
    """Absolute post-move count difference; zero is the visible tie boundary."""

    left, right = _post_move_counts(scene)
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There is no declared render-invariant latent reparameterization."""

    return []
