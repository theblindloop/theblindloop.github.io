"""Independent image-only inverse for the tray-transfer world.

This module deliberately knows nothing about latent scenes or the renderer. It
segments the navy token interiors, counts their connected components on each
side of the image, and locates the gold ring by looking for gold pixels around
exactly one token. It then performs the stated one-token transfer on those
measured counts.
"""

from __future__ import annotations

from collections import deque

import numpy as np


ORACLE_VERSION = "counterfactual_tray_transfer-oracle-0.1.0"
ABSTAIN = "abstain"

_TOKEN = np.array((31, 79, 137), dtype=np.int16)
_GOLD = np.array((232, 157, 24), dtype=np.int16)


def _near(arr: np.ndarray, colour: np.ndarray, tolerance: int = 6) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - colour), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 450) -> list[tuple[int, int, int, int, int]]:
    """Return (pixel count, y0, x0, y1, x1) for sizeable 8-connected blobs."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        queue = deque([(start_y, start_x)])
        seen[start_y, start_x] = True
        size = 0
        y0 = y1 = start_y
        x0 = x1 = start_x
        while queue:
            y, x = queue.popleft()
            size += 1
            y0 = min(y0, y)
            y1 = max(y1, y)
            x0 = min(x0, x)
            x1 = max(x1, x)
            for ny, nx in (
                (y - 1, x),
                (y + 1, x),
                (y, x - 1),
                (y, x + 1),
                (y - 1, x - 1),
                (y - 1, x + 1),
                (y + 1, x - 1),
                (y + 1, x + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if size >= minimum:
            output.append((size, y0, x0, y1, x1))
    return output


def _gold_score(gold: np.ndarray, component: tuple[int, int, int, int, int]) -> int:
    _, y0, x0, y1, x1 = component
    pad = 13
    crop = gold[max(0, y0 - pad):y1 + pad + 1, max(0, x0 - pad):x1 + pad + 1]
    return int(crop.sum())


def decision_from_image(image) -> str:
    """Measure the initial trays, apply the marked move, and return left/right/tie."""

    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 32 or array.shape[1] < 32:
        return ABSTAIN

    token_mask = _near(array, _TOKEN)
    gold_mask = _near(array, _GOLD)
    tokens = _components(token_mask)
    if not tokens:
        return ABSTAIN

    midpoint = array.shape[1] / 2.0
    left_tokens = [part for part in tokens if (part[2] + part[4]) / 2.0 < midpoint]
    right_tokens = [part for part in tokens if (part[2] + part[4]) / 2.0 > midpoint]
    if not left_tokens or not right_tokens:
        return ABSTAIN

    scored = sorted((_gold_score(gold_mask, part), part) for part in tokens)
    if scored[-1][0] < 40 or (len(scored) > 1 and scored[-1][0] == scored[-2][0]):
        return ABSTAIN
    selected = scored[-1][1]
    source_left = (selected[2] + selected[4]) / 2.0 < midpoint

    left = len(left_tokens)
    right = len(right_tokens)
    if source_left:
        left -= 1
        right += 1
    else:
        left += 1
        right -= 1
    if left > right:
        return "left"
    if right > left:
        return "right"
    return "tie"
