"""Deterministic scene sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 640
BG = (246, 241, 228)
NAVY = (27, 47, 73)
CREAM = (244, 216, 160)
CYAN = (54, 196, 204)
GOLD = (239, 183, 54)
LAVENDER = (183, 153, 232)
TAG_RGB = {
    "red": (220, 72, 72),
    "green": (62, 169, 96),
    "blue": (67, 113, 218),
}

# Each connected polyomino fits inside a 4x5 zone. The varied boundary shapes,
# not their area alone, determine the ledger.
SHAPES = (
    ((1, 2),),
    ((1, 1), (1, 2)),
    ((1, 1), (1, 2), (1, 3)),
    ((0, 1), (1, 1), (2, 1), (2, 2)),
    ((0, 1), (1, 0), (1, 1), (1, 2), (2, 1)),
    ((0, 0), (1, 0), (2, 0), (2, 1), (2, 2)),
    ((0, 1), (1, 0), (1, 1), (2, 1), (2, 2), (3, 2)),
    ((0, 0), (0, 1), (1, 1), (2, 1), (2, 2), (2, 3)),
)
ZONE_OFFSETS = ((0, 0), (0, 6), (5, 0), (5, 6))


def _grid_from_shapes(shape_ids: Iterable[int]) -> list[str]:
    grid = [[0] * 11 for _ in range(9)]
    for shape_id, (row0, col0) in zip(shape_ids, ZONE_OFFSETS):
        for row, col in SHAPES[shape_id]:
            grid[row0 + row][col0 + col] = 1
    return ["".join(str(value) for value in row) for row in grid]


def _components(source: list[str]) -> list[list[tuple[int, int]]]:
    on = {(r, c) for r, row in enumerate(source) for c, value in enumerate(row) if value == "1"}
    parts = []
    while on:
        start = min(on)
        on.remove(start)
        stack = [start]
        part = []
        while stack:
            cell = stack.pop()
            part.append(cell)
            r, c = cell
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in on:
                    on.remove(nxt)
                    stack.append(nxt)
        parts.append(part)
    return parts


def _perimeter(part: list[tuple[int, int]]) -> int:
    cells = set(part)
    return sum(
        (r + dr, c + dc) not in cells
        for r, c in cells
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1))
    )


def _target(source: list[str]) -> list[int]:
    return sorted(_perimeter(part) for part in _components(source))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    shape_ids = [rng.randrange(len(SHAPES)) for _ in range(4)]
    source = _grid_from_shapes(shape_ids)
    target = _target(source)
    alternatives = [
        list(values)
        for values in itertools.combinations_with_replacement(range(4, 15, 2), 4)
        if sum(values) == sum(target) and list(values) != target
    ]
    if len(alternatives) < 2:
        # Deterministically force a varied source; this branch is visible in pixels.
        source = _grid_from_shapes((0, 1, 4, 6))
        target = _target(source)
        alternatives = [
            list(values)
            for values in itertools.combinations_with_replacement(range(4, 15, 2), 4)
            if sum(values) == sum(target) and list(values) != target
        ]
    rng.shuffle(alternatives)
    target_row = (seed // 3) % 3
    candidates = [alternatives[0], alternatives[1]]
    candidates.insert(target_row, target)
    correct_color = ("red", "green", "blue")[seed % 3]
    other_colors = [color for color in ("red", "green", "blue") if color != correct_color]
    rng.shuffle(other_colors)
    tags = other_colors
    tags.insert(target_row, correct_color)
    return {"source": source, "candidates": candidates, "tags": tags}


def analytic_gold(scene: dict) -> str:
    target = _target(scene["source"])
    matches = [i for i, values in enumerate(scene["candidates"]) if list(values) == target]
    if len(matches) != 1:
        raise ValueError("ledger match is not unique")
    return scene["tags"][matches[0]]


def margin(scene: dict) -> float:
    target = _target(scene["source"])
    distances = sorted(sum(abs(a - b) for a, b in zip(target, values)) for values in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        target = _target(scene["source"])
        matches = sum(list(values) == target for values in scene["candidates"])
        return len(_components(scene["source"])) != 4 or matches != 1 or margin(scene) < 4
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.rectangle((24, 28, 935, 611), fill=NAVY)

    source = scene["source"]
    cell, gap = 36, 5
    x0, y0 = 72, 120
    grid_w = 11 * cell + 10 * gap
    grid_h = 9 * cell + 8 * gap
    draw.rectangle((56, 102, x0 + grid_w + 15, y0 + grid_h + 17), outline=GOLD, width=7)
    for row in range(9):
        for col in range(11):
            x = x0 + col * (cell + gap)
            y = y0 + row * (cell + gap)
            fill = CYAN if source[row][col] == "1" else CREAM
            draw.rectangle((x, y, x + cell - 1, y + cell - 1), fill=fill)

    bottoms = (216, 387, 558)
    stack_xs = (650, 712, 774, 836)
    tile_w, tile_h, tile_gap = 30, 7, 2
    for values, tag, bottom in zip(scene["candidates"], scene["tags"], bottoms):
        tag_y = bottom - 63
        draw.rectangle((574, tag_y - 15, 604, tag_y + 15), fill=TAG_RGB[tag])
        for x, count in zip(stack_xs, values):
            for level in range(int(count)):
                y2 = bottom - level * (tile_h + tile_gap)
                draw.rectangle((x, y2 - tile_h + 1, x + tile_w - 1, y2), fill=LAVENDER)
    return image
