"""Independent pixel-only inverse for the island-perimeter atlas."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BG = (246, 241, 228)
NAVY = (27, 47, 73)
CREAM = (244, 216, 160)
CYAN = (54, 196, 204)
GOLD = (239, 183, 54)
LAVENDER = (183, 153, 232)
TAG_RGB = {
    "red": (220, 72, 72),
    "green": (62, 169, 96),
    "blue": (67, 113, 218),
}
PALETTE = (BG, NAVY, CREAM, CYAN, GOLD, LAVENDER, *TAG_RGB.values())


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int, float, float]]:
    seen = np.zeros(mask.shape, dtype=bool)
    found = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        found.append((min(xs), min(ys), max(xs), max(ys), len(points), sum(xs) / len(xs), sum(ys) / len(ys)))
    return found


def _cluster(values: list[float], tolerance: float = 3.0) -> list[float]:
    groups: list[list[float]] = []
    for value in sorted(values):
        if not groups or value - sum(groups[-1]) / len(groups[-1]) > tolerance:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [sum(group) / len(group) for group in groups]


def _nearest(value: float, centers: list[float]) -> int:
    return min(range(len(centers)), key=lambda i: abs(value - centers[i]))


def _grid_perimeters(cells: list[tuple[float, float, bool]]) -> list[int]:
    xs = _cluster([x for x, _, _ in cells])
    ys = _cluster([y for _, y, _ in cells])
    if len(xs) != 11 or len(ys) != 9 or len(cells) != 99:
        raise ValueError("source grid could not be recovered")
    on = {(_nearest(y, ys), _nearest(x, xs)) for x, y, state in cells if state}
    parts = []
    while on:
        start = min(on)
        on.remove(start)
        stack = [start]
        part = []
        while stack:
            cell = stack.pop()
            part.append(cell)
            r, c = cell
            for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if nxt in on:
                    on.remove(nxt)
                    stack.append(nxt)
        parts.append(part)
    if len(parts) != 4:
        raise ValueError("expected four visible cyan islands")
    answer = []
    for part in parts:
        occupied = set(part)
        answer.append(sum((r + dr, c + dc) not in occupied for r, c in occupied for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1))))
    return sorted(answer)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    colors = np.array(PALETTE, dtype=np.int32)
    distances = ((arr[:, :, None, :] - colors[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    if int(distances.min(axis=2).max()) != 0:
        raise ValueError("atlas contains an unknown or corrupted color")

    navy = labels == 1
    navy_points = np.argwhere(navy)
    if not len(navy_points):
        raise ValueError("navy atlas missing")
    y0, x0 = navy_points.min(axis=0)
    y1, x1 = navy_points.max(axis=0)
    if (x1 - x0 + 1) * (y1 - y0 + 1) < image.width * image.height * 0.75:
        raise ValueError("navy atlas does not span the display")
    if np.any(labels[y0 : y1 + 1, x0 : x1 + 1] == 0):
        raise ValueError("background-colored hole tears the atlas")

    source_cells = []
    for label, is_cyan in ((2, False), (3, True)):
        for bx0, by0, bx1, by1, area, cx, cy in _components(labels == label):
            width, height = bx1 - bx0 + 1, by1 - by0 + 1
            if 30 <= width <= 42 and 30 <= height <= 42 and area >= 900:
                source_cells.append((cx, cy, is_cyan))
    target = _grid_perimeters(source_cells)

    tags = []
    for offset, color in enumerate(("red", "green", "blue"), start=6):
        pieces = [part for part in _components(labels == offset) if part[4] >= 500]
        if len(pieces) != 1:
            raise ValueError(f"could not recover {color} tag")
        tags.append((pieces[0][6], color))
    tags.sort()

    tiles = [part for part in _components(labels == 5) if 150 <= part[4] <= 300]
    if not tiles:
        raise ValueError("no ledger tiles recovered")
    ledgers = []
    for tag_y, color in tags:
        row_tiles = [part for part in tiles if abs(part[6] - tag_y) < 68]
        x_centers = _cluster([part[5] for part in row_tiles], tolerance=4.0)
        if len(x_centers) != 4:
            raise ValueError("candidate ledger does not have four stacks")
        counts = [0, 0, 0, 0]
        for part in row_tiles:
            counts[_nearest(part[5], x_centers)] += 1
        ledgers.append((counts, color))
    matches = [color for counts, color in ledgers if counts == target]
    if len(matches) != 1:
        raise ValueError("pixel ledger match is not unique")
    return matches[0]
