"""Independent inverse arm: final RGB pixels in, border-color decision out."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

DOT_RGB = {
    0: (138, 78, 184),
    1: (18, 158, 145),
    2: (238, 123, 38),
    3: (211, 63, 126),
    4: (102, 166, 42),
    5: (45, 181, 220),
}
BORDER_RGB = {
    "reference": (105, 112, 122),
    "red": (205, 52, 55),
    "blue": (48, 91, 190),
    "gold": (218, 160, 30),
}
INK = (30, 33, 38)


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _border_box(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    components = [item for item in _components(mask) if item[2] - item[0] > 250 and item[3] - item[1] > 220]
    if len(components) != 1:
        raise OracleError("missing or ambiguous panel border")
    return components[0][:4]


def _centroid(arr: np.ndarray, rgb: tuple[int, int, int], box: tuple[int, int, int, int]) -> tuple[float, float]:
    x0, y0, x1, y1 = box
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    ys, xs = np.nonzero(np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2))
    if len(xs) < 150:
        raise OracleError("colored leaf missing")
    return float(xs.mean() + x0), float(ys.mean() + y0)


def _signature(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple:
    centers = {color_id: _centroid(arr, rgb, box) for color_id, rgb in DOT_RGB.items()}
    x0, y0, x1, y1 = box
    ink_crop = np.all(arr[y0 : y1 + 1, x0 : x1 + 1] == np.asarray(INK, dtype=np.uint8), axis=2)
    frames = []
    for left, top, right, bottom, count in _components(ink_crop):
        width, height = right - left, bottom - top
        if width > 100 and height > 50 and count > 500:
            global_box = (left + x0, top + y0, right + x0, bottom + y0)
            members = tuple(sorted(
                color_id
                for color_id, (cx, cy) in centers.items()
                if global_box[0] < cx < global_box[2] and global_box[1] < cy < global_box[3]
            ))
            if len(members) in (2, 3):
                frames.append(members)
    triples = sorted(set(members for members in frames if len(members) == 3))
    pairs = sorted(set(members for members in frames if len(members) == 2))
    if len(triples) != 2 or len(pairs) != 2:
        raise OracleError("could not recover two outer and two inner enclosures")
    result = []
    for triple in triples:
        nested = [pair for pair in pairs if set(pair) < set(triple)]
        if len(nested) != 1:
            raise OracleError("enclosure hierarchy is ambiguous")
        result.append((triple, nested[0]))
    return tuple(sorted(result))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    boxes = {label: _border_box(arr, rgb) for label, rgb in BORDER_RGB.items()}
    reference = _signature(arr, boxes["reference"])
    matches = [label for label in ("red", "blue", "gold") if _signature(arr, boxes[label]) == reference]
    if len(matches) != 1:
        raise OracleError("raster does not contain one unique matching hierarchy")
    return matches[0]
