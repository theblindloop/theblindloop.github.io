"""Deterministic scene sampler and Pillow renderer."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1000, 720
DOT_RGB = {
    0: (138, 78, 184),   # purple
    1: (18, 158, 145),   # teal
    2: (238, 123, 38),   # orange
    3: (211, 63, 126),   # pink
    4: (102, 166, 42),   # green
    5: (45, 181, 220),   # cyan
}
BORDER_RGB = {
    "reference": (105, 112, 122),
    "red": (205, 52, 55),
    "blue": (48, 91, 190),
    "gold": (218, 160, 30),
}
SLOTS = {
    0: (350, 35),
    1: (30, 410),
    2: (350, 410),
    3: (670, 410),
}


def _canonical_groups(groups: list[dict]) -> tuple:
    items = []
    for group in groups:
        triple = tuple(sorted(int(value) for value in group["triple"]))
        pair = tuple(sorted(int(value) for value in group["pair"]))
        items.append((triple, pair))
    return tuple(sorted(items))


def _relation_vector(groups: list[dict]) -> tuple[int, ...]:
    pair_sets = [set(group["pair"]) for group in groups]
    triple_sets = [set(group["triple"]) for group in groups]
    values = []
    for left in range(6):
        for right in range(left + 1, 6):
            if any({left, right} <= pair for pair in pair_sets):
                values.append(2)
            elif any({left, right} <= triple for triple in triple_sets):
                values.append(1)
            else:
                values.append(0)
    return tuple(values)


def _similarity(left: list[dict], right: list[dict]) -> int:
    return sum(a == b for a, b in zip(_relation_vector(left), _relation_vector(right)))


def _panel(role: str, label: str, slot: int, groups: list[dict], rng: random.Random) -> dict:
    copied = copy.deepcopy(groups)
    rng.shuffle(copied)
    for group in copied:
        rng.shuffle(group["triple"])
        rng.shuffle(group["pair"])
    return {
        "role": role,
        "label": label,
        "slot": slot,
        "groups": copied,
        "inner_sides": [rng.randrange(2), rng.randrange(2)],
        "pair_orders": [rng.randrange(2), rng.randrange(2)],
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(range(6))
    rng.shuffle(colors)
    reference = [
        {"triple": colors[:3], "pair": colors[:2]},
        {"triple": colors[3:], "pair": colors[3:5]},
    ]
    inner_mutation = copy.deepcopy(reference)
    inner_mutation[0]["pair"] = [reference[0]["pair"][0], reference[0]["triple"][2]]
    outer_mutation = copy.deepcopy(reference)
    outer_mutation[0]["triple"][2], outer_mutation[1]["triple"][2] = (
        outer_mutation[1]["triple"][2],
        outer_mutation[0]["triple"][2],
    )

    correct_label = ("red", "blue", "gold")[int(seed) % 3]
    structures = {
        correct_label: reference,
    }
    remaining = [label for label in ("red", "blue", "gold") if label != correct_label]
    structures[remaining[0]] = inner_mutation
    structures[remaining[1]] = outer_mutation
    option_labels = ["red", "blue", "gold"]
    rng.shuffle(option_labels)

    panels = [_panel("reference", "reference", 0, reference, rng)]
    for slot, label in enumerate(option_labels, 1):
        panels.append(_panel("option", label, slot, structures[label], rng))
    return {"panels": panels}


def _draw_panel(draw: ImageDraw.ImageDraw, panel: dict) -> None:
    x, y = SLOTS[int(panel["slot"])]
    border = BORDER_RGB[panel["label"]]
    draw.rounded_rectangle((x, y, x + 300, y + 260), radius=16, outline=border, width=9)
    groups = panel["groups"]
    sides = panel["inner_sides"]
    orders = panel["pair_orders"]
    for row, group in enumerate(groups):
        top = y + 30 + row * 112
        draw.rounded_rectangle(
            (x + 22, top, x + 278, top + 94), radius=25, outline=(30, 33, 38), width=6
        )
        side = int(sides[row])
        if side == 0:
            inner = (x + 37, top + 13, x + 186, top + 81)
            centers = [(x + 78, top + 47), (x + 145, top + 47), (x + 238, top + 47)]
        else:
            inner = (x + 114, top + 13, x + 263, top + 81)
            centers = [(x + 62, top + 47), (x + 155, top + 47), (x + 222, top + 47)]
        draw.rounded_rectangle(inner, radius=22, outline=(30, 33, 38), width=6)
        pair = list(group["pair"])
        if int(orders[row]):
            pair.reverse()
        single = next(value for value in group["triple"] if value not in set(group["pair"]))
        ordered = pair + [single] if side == 0 else [single] + pair
        for color_id, (cx, cy) in zip(ordered, centers):
            rgb = DOT_RGB[int(color_id)]
            draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), fill=rgb, outline=(255, 255, 255), width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 248, 243))
    draw = ImageDraw.Draw(image)
    for panel in sorted(scene["panels"], key=lambda item: int(item["slot"])):
        _draw_panel(draw, panel)
    return image


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    reference = next(panel for panel in panels if panel.get("role") == "reference")
    target = _canonical_groups(reference.get("groups"))
    matches = [
        panel.get("label")
        for panel in panels
        if panel.get("role") == "option" and _canonical_groups(panel.get("groups")) == target
    ]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one hierarchy match")
    return str(matches[0])


def margin(scene: dict) -> float:
    panels = scene["panels"]
    reference = next(panel for panel in panels if panel.get("role") == "reference")
    scores = sorted(
        (_similarity(reference.get("groups"), panel.get("groups")) for panel in panels if panel.get("role") == "option"),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0
    except (KeyError, ValueError, StopIteration, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return [("panel_serialization_reversal", {"panels": list(reversed(copy.deepcopy(scene["panels"])))})]
