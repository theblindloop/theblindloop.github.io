"""Pixel-only inverse arm for peak_to_trough_drop.

Receives one rasterized picture and nothing else: no latent parameters, no
analytic gold, no renderer, prompt, generator or verifier code.  It segments
each coloured trace, recovers one vertical position per occupied column, scans
those positions left to right while carrying the running highest point seen so
far, measures the deepest fall from that running high to a later lower point,
and names the colour whose fall is deepest.  It abstains when the evidence is
not clean.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

TRACE_RGB = {
    "red": (208, 30, 38),
    "green": (16, 138, 60),
    "blue": (30, 66, 208),
}
CHANNEL_TOLERANCE = 30
MIN_TRACE_PIXELS = 600
MIN_COLUMNS = 200
MIN_GAP_PX = 6.0
MIN_FALL_PX = 6.0
ABSTAIN = "abstain"


def _profile(pixels: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray | None:
    """One representative row per occupied column, or ``None`` if unusable."""

    target = np.array(rgb, dtype=np.int16)
    mask = (np.abs(pixels.astype(np.int16) - target) <= CHANNEL_TOLERANCE).all(axis=2)
    if int(mask.sum()) < MIN_TRACE_PIXELS:
        return None
    columns = np.flatnonzero(mask.any(axis=0))
    if columns.size < MIN_COLUMNS:
        return None
    left, right = int(columns[0]), int(columns[-1])
    if columns.size != right - left + 1:
        return None
    rows = np.arange(mask.shape[0])
    profile = np.empty(columns.size, dtype=np.float64)
    for offset, column in enumerate(range(left, right + 1)):
        present = rows[mask[:, column]]
        profile[offset] = float(np.median(present))
    return profile


def _deepest_fall(profile: np.ndarray) -> float:
    """Largest increase in row index from an earlier column to a later one.

    Rows grow downwards, so a fall in the plotted quantity is an increase in
    row index relative to the highest point seen so far on the left.
    """

    running_high = np.minimum.accumulate(profile)
    return float(np.max(profile - running_high))


def decision_from_image(image: Image.Image) -> str:
    """Return the trace colour with the deepest ordered fall, or ``abstain``."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    falls: dict[str, float] = {}
    for name, rgb in TRACE_RGB.items():
        profile = _profile(pixels, rgb)
        if profile is None:
            return ABSTAIN
        fall = _deepest_fall(profile)
        if fall < MIN_FALL_PX:
            return ABSTAIN
        falls[name] = fall

    ranked = sorted(falls.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] - ranked[1][1] < MIN_GAP_PX:
        return ABSTAIN
    return ranked[0][0]
