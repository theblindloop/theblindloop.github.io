"""Independent RGB-only inverse arm for three-way socket connectivity."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1200, 820
INK = np.array((38, 44, 52), dtype=np.int16)
START_COLOR = np.array((30, 143, 86), dtype=np.int16)
TERMINAL_COLOR = np.array((224, 146, 28), dtype=np.int16)
COLORS = {
    "red": np.array((211, 61, 67), dtype=np.int16),
    "blue": np.array((45, 100, 205), dtype=np.int16),
    "orange": np.array((231, 116, 31), dtype=np.int16),
    "violet": np.array((137, 75, 186), dtype=np.int16),
    "teal": np.array((25, 151, 157), dtype=np.int16),
}
IDENTITIES = tuple(f"{name}_{shape}" for name in COLORS for shape in ("circle", "square"))
CARD_CENTERS = tuple((200 + 400 * column, 225 + 240 * row) for row in range(3) for column in range(3))
PORT_OFFSETS = ((-145, -38), (-145, 38), (145, -38), (145, 38))
STRAIGHT = ((0, 2), (1, 3))
CROSS = ((0, 3), (1, 2))
MARKERS = {"START": (0, 0), "A": (1, 0), "B": (2, 0), "C": (3, 0), "D": (4, 0), "E": (5, 0)}
ANSWERS = ("A", "B", "C", "D", "E")


class OracleAbstention(ValueError):
    pass


def _near(arr: np.ndarray, color: np.ndarray, threshold: float) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=-1, dtype=np.int64) < threshold * threshold


def _patch_score(arr: np.ndarray, x: int, y: int, color: np.ndarray, radius: int = 3) -> int:
    return int(_near(arr[y - radius:y + radius + 1, x - radius:x + radius + 1], color, 58.0).sum())


def _wire_mode(arr: np.ndarray, cx: int, cy: int) -> str:
    straight_points = ((-75, -38), (75, -38), (-75, 38), (75, 38))
    cross_points = ((-75, -20), (75, 20), (-75, 20), (75, -20))
    straight_score = sum(_patch_score(arr, cx + dx, cy + dy, INK, 4) for dx, dy in straight_points)
    cross_score = sum(_patch_score(arr, cx + dx, cy + dy, INK, 4) for dx, dy in cross_points)
    if max(straight_score, cross_score) < 190 or abs(straight_score - cross_score) < 120:
        raise OracleAbstention(f"ambiguous cord pairing scores {straight_score}/{cross_score}")
    return "straight" if straight_score > cross_score else "cross"


def _decode_badge(arr: np.ndarray, x: int, y: int) -> str:
    patch = arr[y - 29:y + 30, x - 29:x + 30]
    scores = {name: int(_near(patch, color, 55.0).sum()) for name, color in COLORS.items()}
    name, score = max(scores.items(), key=lambda item: item[1])
    if score < 500:
        raise OracleAbstention(f"missing socket color near {(x, y)}: {scores}")
    color = COLORS[name]

    def colored(dx: int, dy: int) -> bool:
        return bool(_near(arr[y + dy:y + dy + 1, x + dx:x + dx + 1], color, 55.0)[0, 0])

    shape = "square" if sum(colored(dx, dy) for dx in (-15, 15) for dy in (-15, 15)) >= 3 else "circle"
    return f"{name}_{shape}"


def _components(adjacency: dict[tuple[int, int], set[tuple[int, int]]]) -> list[set[tuple[int, int]]]:
    unseen = set(adjacency)
    groups = []
    while unseen:
        root = min(unseen)
        group = {root}
        queue = deque([root])
        while queue:
            node = queue.popleft()
            for neighbor in adjacency[node] - group:
                group.add(neighbor)
                queue.append(neighbor)
        unseen -= group
        groups.append(group)
    return groups


def _distance(adjacency: dict[tuple[int, int], set[tuple[int, int]]], start: tuple[int, int], finish: tuple[int, int]) -> int:
    distance = {start: 0}
    queue = deque([start])
    while queue:
        node = queue.popleft()
        if node == finish:
            return distance[node]
        for neighbor in adjacency[node] - distance.keys():
            distance[neighbor] = distance[node] + 1
            queue.append(neighbor)
    raise OracleAbstention("marked ports are disconnected")


class _Union:
    def __init__(self) -> None:
        self.parent = list(range(9))

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def join(self, left: int, right: int) -> None:
        a, b = self.find(left), self.find(right)
        if a != b:
            self.parent[max(a, b)] = min(a, b)


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)
    if int(_near(arr[115:810], INK, 62.0).sum()) < 24000:
        raise OracleAbstention("plate boundaries or cords are missing")
    reverse_markers = {slot: label for label, slot in MARKERS.items()}
    plates = []
    counts: Counter[str] = Counter()
    for plate_index, (cx, cy) in enumerate(CARD_CENTERS):
        mode = _wire_mode(arr, cx, cy)
        ports = []
        for port, (dx, dy) in enumerate(PORT_OFFSETS):
            x, y = cx + dx, cy + dy
            marker = reverse_markers.get((plate_index, port))
            if marker is not None:
                color = START_COLOR if marker == "START" else TERMINAL_COLOR
                patch = arr[y - 25:y + 26, x - 45:x + 46]
                if int(_near(patch, color, 58.0).sum()) < 650:
                    raise OracleAbstention(f"marker {marker} is missing")
                ports.append(f"marker:{marker}")
            else:
                identity = _decode_badge(arr, x, y)
                counts[identity] += 1
                ports.append(identity)
        plates.append({"ports": ports, "wires": list(STRAIGHT if mode == "straight" else CROSS)})
    if counts != Counter({identity: 3 for identity in IDENTITIES}):
        raise OracleAbstention(f"socket inventory is malformed: {dict(counts)}")
    adjacency = {(plate, port): set() for plate in range(9) for port in range(4)}
    locations: dict[str, list[tuple[int, int]]] = {identity: [] for identity in IDENTITIES}
    for plate_index, plate in enumerate(plates):
        for left, right in plate["wires"]:
            a, b = (plate_index, left), (plate_index, right)
            adjacency[a].add(b)
            adjacency[b].add(a)
        for port, value in enumerate(plate["ports"]):
            if not value.startswith("marker:"):
                locations[value].append((plate_index, port))
    coarse = _Union()
    for identity, group in locations.items():
        if len(group) != 3 or len({plate for plate, _ in group}) != 3:
            raise OracleAbstention(f"malformed three-way socket {identity}: {group}")
        coarse.join(group[0][0], group[1][0])
        coarse.join(group[0][0], group[2][0])
        for index, left in enumerate(group):
            for right in group[index + 1:]:
                adjacency[left].add(right)
                adjacency[right].add(left)
    if len({coarse.find(index) for index in range(9)}) != 1:
        raise OracleAbstention("coarse plate graph is not connected")
    groups = _components(adjacency)
    if sorted(map(len, groups)) != [16, 20]:
        raise OracleAbstention(f"noncanonical fine components: {sorted(map(len, groups))}")
    if sorted(sum(slot in MARKERS.values() for slot in group) for group in groups) != [2, 4]:
        raise OracleAbstention("component marker inventory is malformed")
    start_group = next(group for group in groups if MARKERS["START"] in group)
    answers = [label for label in ANSWERS if MARKERS[label] in start_group]
    if len(answers) != 1:
        raise OracleAbstention(f"non-unique START terminal: {answers}")
    if _distance(adjacency, MARKERS["START"], MARKERS[answers[0]]) < 9:
        raise OracleAbstention("START terminal has a noncanonical short certificate")
    return answers[0]
