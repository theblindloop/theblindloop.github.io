"""Deterministic cyclic connectivity world built from three-occurrence sockets."""

from __future__ import annotations

import random
from collections import Counter, deque
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 1200, 820, 2
BACKGROUND = (238, 241, 245)
PAPER = (255, 255, 255)
INK = (38, 44, 52)
MUTED = (166, 174, 184)
START_COLOR = (30, 143, 86)
TERMINAL_COLOR = (224, 146, 28)
BADGE_COLORS = {
    "red": (211, 61, 67),
    "blue": (45, 100, 205),
    "orange": (231, 116, 31),
    "violet": (137, 75, 186),
    "teal": (25, 151, 157),
}
BADGE_ORDER = tuple(f"{name}_{shape}" for name in BADGE_COLORS for shape in ("circle", "square"))
CARD_CENTERS = tuple((200 + 400 * column, 225 + 240 * row) for row in range(3) for column in range(3))
PORT_OFFSETS = ((-145, -38), (-145, 38), (145, -38), (145, 38))
STRAIGHT = ((0, 2), (1, 3))
CROSS = ((0, 3), (1, 2))
MARKER_SLOTS = {"START": (0, 0), "A": (1, 0), "B": (2, 0), "C": (3, 0), "D": (4, 0), "E": (5, 0)}
ANSWERS = ("A", "B", "C", "D", "E")


class _Union:
    def __init__(self, n: int) -> None:
        self.parent = list(range(n))

    def find(self, value: int) -> int:
        while self.parent[value] != value:
            self.parent[value] = self.parent[self.parent[value]]
            value = self.parent[value]
        return value

    def join(self, left: int, right: int) -> None:
        a, b = self.find(left), self.find(right)
        if a != b:
            self.parent[max(a, b)] = min(a, b)


def _wire_pairs(mode: str) -> tuple[tuple[int, int], tuple[int, int]]:
    return STRAIGHT if mode == "straight" else CROSS


def _hyper_connected(segment_ids: set[int], groups: list[list[tuple[int, int, int]]]) -> bool:
    adjacency = {node: set() for node in segment_ids}
    for group in groups:
        nodes = [stub[0] for stub in group]
        for left in nodes:
            adjacency[left].update(right for right in nodes if right != left)
    reached = {min(segment_ids)}
    queue = deque(reached)
    while queue:
        node = queue.popleft()
        for neighbor in adjacency[node] - reached:
            reached.add(neighbor)
            queue.append(neighbor)
    return reached == segment_ids


def _coarse_connected(groups: list[list[tuple[int, int]]]) -> bool:
    union = _Union(9)
    for group in groups:
        plates = [slot[0] for slot in group]
        union.join(plates[0], plates[1])
        union.join(plates[0], plates[2])
    return len({union.find(index) for index in range(9)}) == 1


def _partition_stubs(rng: random.Random, segment_ids: set[int], segments: list[list[tuple[int, int]]], marker_slots: set[tuple[int, int]]) -> list[list[tuple[int, int, int]]] | None:
    stubs = [(segment, plate, port) for segment in segment_ids for plate, port in segments[segment]
             if (plate, port) not in marker_slots]
    if len(stubs) % 3:
        return None
    for _ in range(400):
        rng.shuffle(stubs)
        groups = [stubs[index:index + 3] for index in range(0, len(stubs), 3)]
        if any(len({stub[0] for stub in group}) != 3 or len({stub[1] for stub in group}) != 3
               for group in groups):
            continue
        if _hyper_connected(segment_ids, groups):
            return groups
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ANSWERS[int(seed) % len(ANSWERS)]
    for _ in range(20000):
        modes = [rng.choice(("straight", "cross")) for _ in range(9)]
        if len(set(modes)) == 1:
            modes[int(seed) % 9] = "cross" if modes[0] == "straight" else "straight"
        segments: list[list[tuple[int, int]]] = []
        slot_to_segment: dict[tuple[int, int], int] = {}
        for plate, mode in enumerate(modes):
            for pair in _wire_pairs(mode):
                segment = [(plate, pair[0]), (plate, pair[1])]
                segment_id = len(segments)
                segments.append(segment)
                for slot in segment:
                    slot_to_segment[slot] = segment_id
        special = {label: slot_to_segment[slot] for label, slot in MARKER_SLOTS.items()}
        if len(set(special.values())) != 6:
            continue
        ordinary = [node for node in range(18) if node not in special.values()]
        rng.shuffle(ordinary)
        start_nodes = {special["START"], special[target], *ordinary[:8]}
        other_nodes = set(range(18)) - start_nodes
        start_groups = _partition_stubs(rng, start_nodes, segments, set(MARKER_SLOTS.values()))
        other_groups = _partition_stubs(rng, other_nodes, segments, set(MARKER_SLOTS.values()))
        if start_groups is None or other_groups is None:
            continue
        raw_groups = start_groups + other_groups
        slot_groups = [[(plate, port) for _, plate, port in group] for group in raw_groups]
        if len(slot_groups) != 10 or not _coarse_connected(slot_groups):
            continue
        identities = list(BADGE_ORDER)
        rng.shuffle(identities)
        plates = [{"ports": [None] * 4, "wires": [list(pair) for pair in _wire_pairs(mode)]}
                  for mode in modes]
        for label, (plate, port) in MARKER_SLOTS.items():
            plates[plate]["ports"][port] = f"marker:{label}"
        for identity, group in zip(identities, slot_groups, strict=True):
            for plate, port in group:
                plates[plate]["ports"][port] = identity
        scene = {"plates": plates}
        if not is_quarantined(scene) and analytic_gold(scene) == target:
            return scene
    raise RuntimeError("could not sample a valid triplet-socket network")


def _adjacency(scene: dict) -> tuple[dict[tuple[int, int], set[tuple[int, int]]], list[list[tuple[int, int]]]]:
    adjacency = {(plate, port): set() for plate in range(9) for port in range(4)}
    locations: dict[str, list[tuple[int, int]]] = {}
    for plate_index, plate in enumerate(scene["plates"]):
        for left, right in plate["wires"]:
            a, b = (plate_index, int(left)), (plate_index, int(right))
            adjacency[a].add(b)
            adjacency[b].add(a)
        for port, value in enumerate(plate["ports"]):
            if not str(value).startswith("marker:"):
                locations.setdefault(str(value), []).append((plate_index, port))
    groups = []
    for identity in BADGE_ORDER:
        group = locations.get(identity, [])
        if len(group) != 3:
            raise ValueError(f"socket {identity} occurs {len(group)} times")
        groups.append(group)
        for index, left in enumerate(group):
            for right in group[index + 1:]:
                adjacency[left].add(right)
                adjacency[right].add(left)
    return adjacency, groups


def _components(scene: dict) -> list[set[tuple[int, int]]]:
    adjacency, _ = _adjacency(scene)
    unseen = set(adjacency)
    groups = []
    while unseen:
        root = min(unseen)
        group = {root}
        queue = deque([root])
        while queue:
            node = queue.popleft()
            for neighbor in adjacency[node] - group:
                group.add(neighbor)
                queue.append(neighbor)
        unseen -= group
        groups.append(group)
    return groups


def _port_distance(scene: dict, start: tuple[int, int], finish: tuple[int, int]) -> int:
    adjacency, _ = _adjacency(scene)
    distance = {start: 0}
    queue = deque([start])
    while queue:
        node = queue.popleft()
        if node == finish:
            return distance[node]
        for neighbor in adjacency[node] - distance.keys():
            distance[neighbor] = distance[node] + 1
            queue.append(neighbor)
    raise ValueError("ports are disconnected")


def analytic_gold(scene: dict) -> str:
    start_group = next(group for group in _components(scene) if MARKER_SLOTS["START"] in group)
    answers = [label for label in ANSWERS if MARKER_SLOTS[label] in start_group]
    if len(answers) != 1:
        raise ValueError(f"expected one START terminal, got {answers}")
    return answers[0]


def margin(scene: dict) -> float:
    _ = analytic_gold(scene)
    return 18.0


def is_quarantined(scene: dict) -> bool:
    try:
        if set(scene) != {"plates"} or not isinstance(scene["plates"], list) or len(scene["plates"]) != 9:
            return True
        markers = {}
        counts: Counter[str] = Counter()
        for plate_index, plate in enumerate(scene["plates"]):
            if set(plate) != {"ports", "wires"} or len(plate["ports"]) != 4 or len(plate["wires"]) != 2:
                return True
            normalized = {tuple(sorted(map(int, pair))) for pair in plate["wires"]}
            if normalized not in ({tuple(sorted(pair)) for pair in STRAIGHT}, {tuple(sorted(pair)) for pair in CROSS}):
                return True
            for port, value in enumerate(plate["ports"]):
                value = str(value)
                if value.startswith("marker:"):
                    label = value[7:]
                    if label in markers:
                        return True
                    markers[label] = (plate_index, port)
                else:
                    counts[value] += 1
        if markers != MARKER_SLOTS or counts != Counter({identity: 3 for identity in BADGE_ORDER}):
            return True
        _, socket_groups = _adjacency(scene)
        if any(len({plate for plate, _ in group}) != 3 for group in socket_groups):
            return True
        groups = _components(scene)
        if sorted(map(len, groups)) != [16, 20]:
            return True
        if sorted(sum(slot in MARKER_SLOTS.values() for slot in group) for group in groups) != [2, 4]:
            return True
        if not _coarse_connected(socket_groups):
            return True
        target = analytic_gold(scene)
        # Four remote socket junctions means nine alternating port-graph edges.
        # This blocks an accidental one- or two-junction certificate.
        if _port_distance(scene, MARKER_SLOTS["START"], MARKER_SLOTS[target]) < 9:
            return True
        return target not in ANSWERS
    except (KeyError, TypeError, ValueError, IndexError, StopIteration):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for plate in twin["plates"]:
        plate["wires"].reverse()
    return [("reverse_unordered_cord_serialization", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _p(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _b(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _draw_badge(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: str) -> None:
    color_name, shape = identity.split("_", 1)
    color = BADGE_COLORS[color_name]
    x, y = center
    if shape == "circle":
        draw.ellipse(_b((x - 23, y - 23, x + 23, y + 23)), fill=color, outline=INK, width=4 * SCALE)
    else:
        draw.rounded_rectangle(_b((x - 23, y - 23, x + 23, y + 23)), radius=3 * SCALE,
                               fill=color, outline=INK, width=4 * SCALE)


def _draw_marker(draw: ImageDraw.ImageDraw, center: tuple[int, int], label: str) -> None:
    x, y = center
    if label == "START":
        draw.rounded_rectangle(_b((x - 42, y - 20, x + 42, y + 20)), radius=15 * SCALE,
                               fill=START_COLOR, outline=INK, width=3 * SCALE)
        draw.text(_p((x, y - 1)), "START", anchor="mm", fill=PAPER, font=_font(12, True))
    else:
        draw.ellipse(_b((x - 23, y - 23, x + 23, y + 23)), fill=TERMINAL_COLOR,
                     outline=INK, width=4 * SCALE)
        draw.text(_p((x, y - 1)), label, anchor="mm", fill=PAPER, font=_font(21, True))


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(_b((20, 14, 1180, 108)), radius=15 * SCALE, fill=PAPER, outline=MUTED, width=2 * SCALE)
    draw.text(_p((42, 27)), "THREE-WAY SOCKETS", fill=INK, font=_font(24, True))
    draw.text(_p((42, 67)), "each color + shape occurs 3 times; all 3 join", fill=INK, font=_font(15))
    draw.text(_p((600, 35)), "black cords pair ports", fill=INK, font=_font(19, True))
    draw.text(_p((600, 70)), "white bridge = cross, not join", fill=INK, font=_font(15))
    draw.text(_p((1010, 51)), "START  →  ?", anchor="mm", fill=START_COLOR, font=_font(22, True))
    reverse_markers = {slot: label for label, slot in MARKER_SLOTS.items()}
    for plate_index, (cx, cy) in enumerate(CARD_CENTERS):
        draw.rounded_rectangle(_b((cx - 180, cy - 100, cx + 180, cy + 100)), radius=15 * SCALE,
                               fill=PAPER, outline=MUTED, width=3 * SCALE)
        draw.text(_p((cx, cy - 82)), str(plate_index + 1), anchor="mm", fill=MUTED, font=_font(12, True))
        centers = [(cx + dx, cy + dy) for dx, dy in PORT_OFFSETS]
        normalized = {tuple(sorted(map(int, pair))) for pair in scene["plates"][plate_index]["wires"]}
        if normalized == {tuple(sorted(pair)) for pair in STRAIGHT}:
            draw.line([_p(centers[0]), _p(centers[2])], fill=INK, width=11 * SCALE)
            draw.line([_p(centers[1]), _p(centers[3])], fill=INK, width=11 * SCALE)
        else:
            draw.line([_p(centers[0]), _p(centers[3])], fill=INK, width=11 * SCALE)
            draw.line([_p(centers[1]), _p(centers[2])], fill=PAPER, width=25 * SCALE)
            draw.line([_p(centers[1]), _p(centers[2])], fill=INK, width=11 * SCALE)
        for port, center in enumerate(centers):
            marker = reverse_markers.get((plate_index, port))
            if marker is None:
                _draw_badge(draw, center, str(scene["plates"][plate_index]["ports"][port]))
            else:
                _draw_marker(draw, center, marker)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
