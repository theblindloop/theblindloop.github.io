"""Deterministic sampler, renderer, and analytic arm."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont


ANSWERS = ("red", "blue", "orange")
RGB = {
    "red": (218, 66, 58),
    "blue": (50, 105, 210),
    "orange": (239, 139, 31),
    "green": (35, 164, 88),
    "cyan": (29, 177, 190),
    "yellow": (246, 190, 32),
}


def _longest_alternation(sequence: list[str]) -> int:
    best = current = 1 if sequence else 0
    for left, right in zip(sequence, sequence[1:]):
        current = current + 1 if left != right else 1
        best = max(best, current)
    return best


def _point_at_fraction(points: list[list[int]], fraction: float) -> list[int]:
    lengths = [abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])]
    target = float(sum(lengths)) * fraction
    for (a, b), length in zip(zip(points, points[1:]), lengths):
        if target <= length:
            ratio = target / float(length)
            return [int(round(a[0] + ratio * (b[0] - a[0]))), int(round(a[1] + ratio * (b[1] - a[1])))]
        target -= length
    return list(points[-1])


def _route_points(slot: int, mirrored: bool, jitter: int) -> list[list[int]]:
    left = 25 + 310 * slot
    local = [
        [55, 54], [244, 54], [244, 135], [72, 135], [72, 220],
        [242, 220], [242, 305], [74, 305], [74, 390], [242, 390], [242, 516],
    ]
    if mirrored:
        local = [[280 - x, y] for x, y in local]
    return [[left + x + jitter, y] for x, y in local]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner_color = ANSWERS[int(seed) % 3]
    remaining = [color for color in ANSWERS if color != winner_color]
    rng.shuffle(remaining)
    profile_color = {"winner": winner_color, "runner": remaining[0], "loser": remaining[1]}

    sequences = {
        "winner": ["cyan", "yellow", "cyan", "yellow", "cyan", "yellow"],
        "runner": ["cyan", "yellow", "cyan", "yellow", "yellow", "cyan"],
        "loser": ["cyan", "cyan", "yellow", "yellow", "cyan", "yellow"],
    }
    if rng.randrange(2):
        sequences = {key: ["yellow" if value == "cyan" else "cyan" for value in seq] for key, seq in sequences.items()}
    for key in list(sequences):
        if rng.randrange(2):
            sequences[key] = list(reversed(sequences[key]))

    profiles = ["winner", "runner", "loser"]
    rng.shuffle(profiles)
    routes = []
    for slot, profile in enumerate(profiles):
        points = _route_points(slot, bool(rng.randrange(2)), rng.randint(-3, 3))
        checkpoints = [_point_at_fraction(points, fraction / 7.0) for fraction in range(1, 7)]
        routes.append({
            "points": points,
            "checkpoints": checkpoints,
            "checkpoint_colors": sequences[profile],
            "terminal_color": profile_color[profile],
        })
    rng.shuffle(routes)
    return {
        "canvas": [960, 600],
        "routes": routes,
        "style": {
            "background": [248, 247, 243],
            "ink": [25, 30, 36],
            "line_width": 9,
            "marker_radius": 14,
        },
    }


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    style = scene["style"]
    background = tuple(style["background"])
    ink = tuple(style["ink"])
    line_width = int(style["line_width"])
    radius = int(style["marker_radius"])
    routes = scene["routes"]
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for slot in range(3):
        left = 18 + 310 * slot
        draw.rounded_rectangle((left, 18, left + 294, 572), radius=12, outline=(196, 199, 202), width=2)
    for route in routes:
        draw.line([tuple(point) for point in route["points"]], fill=ink, width=line_width, joint="curve")
    for route in routes:
        for point, color in zip(route["checkpoints"], route["checkpoint_colors"]):
            x, y = point
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=RGB[color], outline=ink, width=3)
        sx, sy = route["points"][0]
        tx, ty = route["points"][-1]
        draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=RGB["green"], outline=ink, width=3)
        draw.ellipse((tx - radius, ty - radius, tx + radius, ty + radius), fill=RGB[route["terminal_color"]], outline=ink, width=3)
        font = ImageFont.load_default(size=13)
        draw.text((sx, sy), "S", fill=(255, 255, 255), font=font, anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    scores = [(_longest_alternation(route["checkpoint_colors"]), str(route["terminal_color"])) for route in scene["routes"]]
    scores.sort(reverse=True)
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = sorted((_longest_alternation(route["checkpoint_colors"]) for route in scene["routes"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    routes = scene.get("routes", [])
    if len(routes) != 3 or any(len(route.get("checkpoints", [])) != 6 for route in routes):
        return True
    sequences = [route.get("checkpoint_colors", []) for route in routes]
    if any(sorted(sequence) != ["cyan"] * 3 + ["yellow"] * 3 for sequence in sequences):
        return True
    scores = sorted((_longest_alternation(sequence) for sequence in sequences), reverse=True)
    return scores[0] == scores[1] or scores[0] - scores[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["routes"] = list(reversed(twin["routes"]))
    return [("reverse_disjoint_route_storage", twin)]
