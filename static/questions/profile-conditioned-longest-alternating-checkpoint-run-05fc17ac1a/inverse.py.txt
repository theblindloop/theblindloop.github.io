"""Independent pixel-only inverse arm for ordered checkpoint runs."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


PALETTE = {
    "green": np.array([35, 164, 88], dtype=np.int16),
    "red": np.array([218, 66, 58], dtype=np.int16),
    "blue": np.array([50, 105, 210], dtype=np.int16),
    "orange": np.array([239, 139, 31], dtype=np.int16),
    "cyan": np.array([29, 177, 190], dtype=np.int16),
    "yellow": np.array([246, 190, 32], dtype=np.int16),
}
ANSWERS = ("red", "blue", "orange")


def _mask(array: np.ndarray, name: str) -> np.ndarray:
    delta = np.abs(array.astype(np.int16) - PALETTE[name])
    return np.max(delta, axis=2) <= 28


def _components(mask: np.ndarray, minimum: int = 25) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= minimum:
            components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _flood(traversable: np.ndarray, source: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    height, width = traversable.shape
    reached = np.zeros_like(traversable, dtype=bool)
    distance = np.full((height, width), -1, dtype=np.int32)
    queue = deque()
    for y, x in source.tolist():
        reached[y, x] = True
        distance[y, x] = 0
        queue.append((int(y), int(x)))
    while queue:
        y, x = queue.popleft()
        next_distance = int(distance[y, x]) + 1
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dy or dx):
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and traversable[ny, nx] and not reached[ny, nx]:
                    reached[ny, nx] = True
                    distance[ny, nx] = next_distance
                    queue.append((ny, nx))
    return reached, distance


def _longest(sequence: list[str]) -> int:
    best = current = 1 if sequence else 0
    for left, right in zip(sequence, sequence[1:]):
        current = current + 1 if left != right else 1
        best = max(best, current)
    return best


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = {name: _mask(array, name) for name in PALETTE}
    starts = _components(masks["green"], minimum=80)
    terminals = {name: _components(masks[name], minimum=80) for name in ANSWERS}
    checkpoints = []
    for name in ("cyan", "yellow"):
        checkpoints.extend((name, component) for component in _components(masks[name], minimum=80))
    if len(starts) != 3:
        raise ValueError(f"expected three green START disks, found {len(starts)}")
    if any(len(terminals[name]) != 1 for name in ANSWERS):
        raise ValueError("each terminal color must appear exactly once")
    if len(checkpoints) != 18:
        raise ValueError(f"expected eighteen checkpoint disks, found {len(checkpoints)}")

    dark = np.max(array, axis=2) <= 100
    traversable = dark.copy()
    for color_mask in masks.values():
        traversable |= color_mask

    scored = []
    for start in starts:
        reached, distance = _flood(traversable, start)
        terminal_hits = []
        for name in ANSWERS:
            component = terminals[name][0]
            if bool(np.any(reached[component[:, 0], component[:, 1]])):
                terminal_hits.append(name)
        if len(terminal_hits) != 1:
            raise ValueError("a START route does not reach exactly one terminal")
        ordered = []
        for name, component in checkpoints:
            values = distance[component[:, 0], component[:, 1]]
            values = values[values >= 0]
            if values.size:
                ordered.append((int(values.min()), name))
        if len(ordered) != 6:
            raise ValueError("a route does not contain exactly six checkpoints")
        ordered.sort()
        scored.append((_longest([name for _distance, name in ordered]), terminal_hits[0]))
    scored.sort(reverse=True)
    if scored[0][0] - scored[1][0] < 2:
        raise ValueError("runner-up alternating-run margin is ambiguous")
    return scored[0][1]
