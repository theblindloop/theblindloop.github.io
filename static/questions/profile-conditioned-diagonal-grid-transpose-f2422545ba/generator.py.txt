"""Deterministic renderer for the corner-anchored transpose world."""

from __future__ import annotations

import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

LABELS = ("flag", "hook", "stair", "fork")
PATTERNS = {
    "flag": ("11000", "11110", "11000", "01000", "00000"),
    "hook": ("10000", "10000", "10000", "10100", "11110"),
    "stair": ("11000", "01100", "00110", "00011", "00001"),
    "fork": ("10001", "01010", "00100", "00100", "01110"),
}
PALETTES = (
    {"background": (250, 248, 240), "empty": (238, 236, 228), "ink": (24, 28, 35)},
    {"background": (246, 250, 252), "empty": (232, 240, 244), "ink": (38, 31, 48)},
    {"background": (252, 247, 250), "empty": (242, 235, 240), "ink": (21, 38, 36)},
)
CYAN = (0, 174, 205)
MAGENTA = (218, 38, 139)
YELLOW = (244, 184, 0)
GRIDLINE = (168, 174, 178)
DISTRACTOR_COLORS = ((218, 83, 68), (47, 151, 93), (231, 132, 38), (87, 104, 187))


def _logical_pattern(label: str) -> np.ndarray:
    return np.array([[ch == "1" for ch in row] for row in PATTERNS[label]], dtype=bool)


def _noisy_pattern(scene: dict) -> np.ndarray:
    mask = _logical_pattern(scene["target"])
    for row, col in scene["noise_remove"]:
        mask[int(row), int(col)] = False
    for row, col in scene["noise_add"]:
        mask[int(row), int(col)] = True
    return mask


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = LABELS[int(seed) % len(LABELS)]
    base = _logical_pattern(target)
    noise_count = rng.choice((0, 0, 1, 1))
    filled = list(zip(*np.where(base)))
    empty = list(zip(*np.where(~base)))
    rng.shuffle(filled)
    rng.shuffle(empty)
    cell = rng.randint(13, 17)
    left = rng.randint(72, 384 - 72 - 10 * cell)
    top = rng.randint(72, 384 - 72 - 10 * cell)
    distractors = []
    zones = ((20, 48, 38, 125), (20, 48, 255, 346), (336, 364, 38, 125),
             (336, 364, 255, 346), (128, 256, 18, 42))
    for index, (x0, x1, y0, y1) in enumerate(zones):
        x, y = rng.randint(x0, x1), rng.randint(y0, y1)
        distractors.append([x, y, rng.randint(4, 8), index % len(DISTRACTOR_COLORS)])
    return {
        "target": target,
        "cell": cell,
        "left": left,
        "top": top,
        "palette": rng.randrange(len(PALETTES)),
        "distractors": distractors,
        "noise_remove": [[int(r), int(c)] for r, c in filled[:noise_count]],
        "noise_add": [[int(r), int(c)] for r, c in empty[:noise_count]],
    }


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"])]
    image = Image.new("RGB", (384, 384), palette["background"])
    draw = ImageDraw.Draw(image)
    for x, y, radius, color_index in scene["distractors"]:
        color = DISTRACTOR_COLORS[int(color_index) % len(DISTRACTOR_COLORS)]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    # The displayed occupancy is the transpose of the intended post-operation glyph.
    display = np.repeat(np.repeat(_noisy_pattern(scene), 2, axis=0), 2, axis=1).T
    cell = int(scene["cell"])
    left, top = int(scene["left"]), int(scene["top"])
    size = 10 * cell
    for row in range(10):
        for col in range(10):
            x0, y0 = left + col * cell, top + row * cell
            fill = palette["ink"] if display[row, col] else palette["empty"]
            draw.rectangle((x0 + 1, y0 + 1, x0 + cell - 2, y0 + cell - 2), fill=fill)
    for index in range(11):
        x = left + index * cell
        y = top + index * cell
        draw.line((x, top, x, top + size), fill=GRIDLINE, width=1)
        draw.line((left, y, left + size, y), fill=GRIDLINE, width=1)
    draw.rectangle((left - 4, top - 4, left + size + 4, top + size + 4), outline=CYAN, width=3)
    draw.rectangle((left - 12, top - 12, left - 6, top - 6), fill=MAGENTA)
    draw.rectangle((left + size + 6, top + size + 6, left + size + 12, top + size + 12), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["target"])


def _distances(scene: dict) -> list[int]:
    observed = _noisy_pattern(scene)
    return sorted(int(np.count_nonzero(observed != _logical_pattern(label))) for label in LABELS)


def margin(scene: dict) -> float:
    distances = _distances(scene)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return int(scene["cell"]) < 11 or margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_distractor_draw_order", twin)]
