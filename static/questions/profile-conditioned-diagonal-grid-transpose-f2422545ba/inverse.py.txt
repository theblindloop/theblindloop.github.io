"""Independent pixel-only inverse for the transpose world.

This module intentionally duplicates only the public glyph vocabulary.  It has
no access to renderer state, scene dictionaries, filenames, or analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CYAN = np.array((0, 174, 205), dtype=np.int16)
MAGENTA = np.array((218, 38, 139), dtype=np.int16)
YELLOW = np.array((244, 184, 0), dtype=np.int16)
PATTERNS = {
    "flag": ("11000", "11110", "11000", "01000", "00000"),
    "hook": ("10000", "10000", "10000", "10100", "11110"),
    "stair": ("11000", "01100", "00110", "00011", "00001"),
    "fork": ("10001", "01010", "00100", "00100", "01110"),
}


def _near(rgb: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(rgb.astype(np.int16) - color), axis=2) <= tolerance


def _template(rows: tuple[str, ...]) -> np.ndarray:
    return np.array([[ch == "1" for ch in row] for row in rows], dtype=bool)


def decision_and_margin_from_image(image: Image.Image) -> tuple[str, float]:
    rgb = np.asarray(image.convert("RGB"))
    cyan = _near(rgb, CYAN)
    ys, xs = np.where(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame is missing")
    min_x, max_x = int(xs.min()), int(xs.max())
    min_y, max_y = int(ys.min()), int(ys.max())
    if abs((max_x - min_x) - (max_y - min_y)) > 2:
        raise ValueError("cyan frame is not square")
    left, top = min_x + 4, min_y + 4
    # Pillow's inclusive rectangle endpoints make the outer cyan extent span+9.
    span = max_x - min_x + 1 - 9
    if span < 100 or span % 10:
        raise ValueError("frame does not contain a stable ten-cell grid")
    cell = span // 10

    magenta_y, magenta_x = np.where(_near(rgb, MAGENTA))
    yellow_y, yellow_x = np.where(_near(rgb, YELLOW))
    if len(magenta_x) < 20 or len(yellow_x) < 20:
        raise ValueError("diagonal endpoint fiducials are missing")
    if magenta_x.mean() >= left or magenta_y.mean() >= top:
        raise ValueError("magenta endpoint does not anchor the top-left")
    if yellow_x.mean() <= left + span or yellow_y.mean() <= top + span:
        raise ValueError("yellow endpoint does not anchor the bottom-right")

    displayed = np.zeros((10, 10), dtype=bool)
    for row in range(10):
        for col in range(10):
            x0 = left + col * cell + max(2, cell // 4)
            x1 = left + (col + 1) * cell - max(2, cell // 4)
            y0 = top + row * cell + max(2, cell // 4)
            y1 = top + (row + 1) * cell - max(2, cell // 4)
            patch = rgb[y0:y1, x0:x1].astype(float)
            luminance = patch.mean(axis=(0, 1)).dot(np.array((0.299, 0.587, 0.114)))
            displayed[row, col] = luminance < 120.0

    recovered = displayed.T.reshape(5, 2, 5, 2).mean(axis=(1, 3)) >= 0.75
    scored = sorted(
        (int(np.count_nonzero(recovered != _template(rows))), label)
        for label, rows in PATTERNS.items()
    )
    gap = float(scored[1][0] - scored[0][0])
    if gap < 3.0:
        raise ValueError(f"ambiguous glyph; runner-up gap={gap:.1f}")
    return scored[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return decision_and_margin_from_image(image)[0]
