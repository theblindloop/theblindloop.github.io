"""Independent image-only inverse arm for paired-vector homecoming.

The arm locates saturated connected components without knowing their absolute
position or submitted palette, estimates each triangular glyph's pointing
direction from its pixel centroid relative to its own bounding box, orders the
two glyphs by their detected x centroids, and replays their unit displacements.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _colored_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        [
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    spread = array.max(axis=2) - array.min(axis=2)
    contrast = np.linalg.norm(array - background, axis=2)
    return (spread >= 48) & (contrast >= 58) & (array.max(axis=2) <= 235)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = {tuple(map(int, item)) for item in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        points = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    points.append(neighbor)
        if len(points) >= 70:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _direction(points: np.ndarray) -> tuple[str, float, tuple[float, float], float]:
    ys = points[:, 0]
    xs = points[:, 1]
    center = (float(xs.mean()), float(ys.mean()))
    box_center = ((float(xs.min()) + float(xs.max())) / 2.0, (float(ys.min()) + float(ys.max())) / 2.0)
    dx = center[0] - box_center[0]
    dy = center[1] - box_center[1]
    dominant = max(abs(dx), abs(dy))
    secondary = min(abs(dx), abs(dy))
    confidence = dominant - secondary
    width = float(xs.max() - xs.min() + 1)
    height = float(ys.max() - ys.min() + 1)
    if not (16 <= width <= 34 and 16 <= height <= 34):
        raise OracleAbstention("colored component is not a supported arrow size")
    if dominant < 2.0 or confidence < 1.5:
        raise OracleAbstention("triangle direction has insufficient runner-up separation")
    if abs(dx) > abs(dy):
        direction = "right" if dx < 0 else "left"
    else:
        direction = "down" if dy < 0 else "up"
    return direction, confidence, center, float(len(points))


def _delta(direction: str) -> tuple[int, int]:
    return {
        "up": (0, -1),
        "right": (1, 0),
        "down": (0, 1),
        "left": (-1, 0),
    }[direction]


def decision_from_image(image: Image.Image) -> str:
    groups = _components(_colored_mask(image))
    if len(groups) != 2:
        raise OracleAbstention(f"expected two colored move glyphs, found {len(groups)}")
    measured = [_direction(group) for group in groups]
    measured.sort(key=lambda item: item[2][0])
    if measured[1][2][0] - measured[0][2][0] < 3.0 * max(math.sqrt(item[3]) for item in measured):
        raise OracleAbstention("move glyphs are not visibly separated left to right")
    if abs(measured[0][2][1] - measured[1][2][1]) > 18.0:
        raise OracleAbstention("move glyphs do not form one ordered row")
    dx1, dy1 = _delta(measured[0][0])
    dx2, dy2 = _delta(measured[1][0])
    return "home" if (dx1 + dx2, dy1 + dy2) == (0, 0) else "away"
