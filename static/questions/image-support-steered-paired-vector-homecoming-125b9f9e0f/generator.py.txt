"""Deterministic renderer and analytic arm for paired-vector homecoming."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
DIRECTIONS = ("up", "right", "down", "left")
ACCENTS = ("#1769aa", "#a83e82", "#168b68")
BACKGROUNDS = ("#ffffff", "#fffdf7")
PALE_INK = "#c9cdd2"
MID_INK = "#747b84"


def _opposite(direction: str) -> str:
    return DIRECTIONS[(DIRECTIONS.index(direction) + 2) % 4]


def _clockwise(direction: str) -> str:
    return DIRECTIONS[(DIRECTIONS.index(direction) + 1) % 4]


def _delta(direction: str) -> tuple[int, int]:
    return {
        "up": (0, -1),
        "right": (1, 0),
        "down": (0, 1),
        "left": (-1, 0),
    }[direction]


def sample_scene(seed: int) -> dict:
    """Return one balanced scene; adjacent integer seeds share every nuisance."""

    pair = seed // 2
    first = DIRECTIONS[pair % 4]
    returns_home = seed % 2 == 0
    if returns_home:
        second = _opposite(first)
    else:
        # Alternate same-direction and quarter-turn away cases. Both remain away,
        # while their runner-up angular margins differ.
        second = first if (pair // 4) % 2 else _clockwise(first)

    nuisance_rng = random.Random(pair * 1_000_003 + 41)
    decorations = ["circle", "square", "diamond", "bar"]
    nuisance_rng.shuffle(decorations)
    row_cell = 1 + (pair * 5 + pair // 3) % 6
    left_cell = 1 + (pair * 3 + pair // 5) % 4
    home_side = "above" if nuisance_rng.randrange(2) else "below"
    return {
        "step1": first,
        "step2": second,
        "left_cell": left_cell,
        "row_cell": row_cell,
        "accent": ACCENTS[pair % len(ACCENTS)],
        "glyph_size": 10 + pair % 4,
        "home_side": home_side,
        "decorations": decorations,
        "background": BACKGROUNDS[(pair // 3) % len(BACKGROUNDS)],
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
    }


def _triangle_points(
    center: tuple[float, float], direction: str, radius: float
) -> list[tuple[float, float]]:
    cx, cy = center
    if direction == "right":
        return [(cx - radius, cy - radius), (cx + radius, cy), (cx - radius, cy + radius)]
    if direction == "left":
        return [(cx + radius, cy - radius), (cx - radius, cy), (cx + radius, cy + radius)]
    if direction == "up":
        return [(cx - radius, cy + radius), (cx, cy - radius), (cx + radius, cy + radius)]
    if direction == "down":
        return [(cx - radius, cy - radius), (cx, cy + radius), (cx + radius, cy - radius)]
    raise ValueError(f"unsupported direction {direction!r}")


def _scaled(points: Iterable[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def _draw_decoration(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    kind: str,
    scale: int,
) -> None:
    cx, cy = center
    radius = 9
    box = tuple(round(value * scale) for value in (cx - radius, cy - radius, cx + radius, cy + radius))
    width = 2 * scale
    if kind == "circle":
        draw.ellipse(box, outline=PALE_INK, width=width)
    elif kind == "square":
        draw.rectangle(box, outline=PALE_INK, width=width)
    elif kind == "diamond":
        draw.polygon(
            _scaled([(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)], scale),
            outline=PALE_INK,
            width=width,
        )
    else:
        draw.line(_scaled([(cx - radius, cy), (cx + radius, cy)], scale), fill=PALE_INK, width=width)


def render(scene: dict) -> Image.Image:
    """Render only public scene fields into a 512px RGB question image."""

    # Direct accesses to the two label inputs make the rendered-field audit
    # explicit. All remaining accesses are visible nuisance parameters.
    first = scene["step1"]
    second = scene["step2"]
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    background = str(scene["background"])
    accent = str(scene["accent"])
    radius = int(scene["glyph_size"])
    left_cell = int(scene["left_cell"])
    row_cell = int(scene["row_cell"])
    home_side = str(scene["home_side"])
    decorations = list(scene["decorations"])

    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    cell_w = width / 8.0
    cell_h = height / 8.0
    first_center = ((left_cell + 0.5) * cell_w, (row_cell + 0.5) * cell_h)
    second_center = ((left_cell + 2.5) * cell_w, (row_cell + 0.5) * cell_h)
    mid_x = (first_center[0] + second_center[0]) / 2.0
    home_y = first_center[1] + (-39 if home_side == "above" else 39)

    # A local pale card makes the sequence easy for a human to group. Its ink is
    # intentionally outside the saturated mask used by the inverse arm.
    card = (
        round((first_center[0] - 24) * scale),
        round((min(first_center[1], home_y) - 18) * scale),
        round((second_center[0] + 24) * scale),
        round((max(first_center[1], home_y) + 18) * scale),
    )
    draw.rounded_rectangle(card, radius=12 * scale, outline="#e1e3e6", width=2 * scale)

    # HOME is visible but constant and nondiagnostic; only the arrows determine
    # whether the accumulated displacement cancels.
    home_box = tuple(
        round(value * scale)
        for value in (mid_x - 8, home_y - 8, mid_x + 8, home_y + 8)
    )
    draw.ellipse(home_box, fill=background, outline=MID_INK, width=3 * scale)
    draw.ellipse(
        tuple(round(value * scale) for value in (mid_x - 2, home_y - 2, mid_x + 2, home_y + 2)),
        fill=MID_INK,
    )

    # Tiny pale ticks reinforce left-to-right reading without carrying direction.
    tick_y = first_center[1] + (-22 if home_side == "below" else 22)
    for center in (first_center, second_center):
        draw.line(
            _scaled([(center[0] - 5, tick_y), (center[0] + 5, tick_y)], scale),
            fill=PALE_INK,
            width=2 * scale,
        )

    draw.polygon(_scaled(_triangle_points(first_center, first, radius), scale), fill=accent)
    draw.polygon(_scaled(_triangle_points(second_center, second, radius), scale), fill=accent)

    remote = [(48, 48), (width - 48, 48), (48, height - 48), (width - 48, height - 48)]
    for center, kind in zip(remote, decorations, strict=True):
        _draw_decoration(draw, center, kind, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    dx1, dy1 = _delta(scene["step1"])
    dx2, dy2 = _delta(scene["step2"])
    return "home" if (dx1 + dx2, dy1 + dy2) == (0, 0) else "away"


def margin(scene: dict) -> float:
    """Angular runner-up margin around the 135-degree categorical boundary."""

    first = DIRECTIONS.index(scene["step1"])
    second = DIRECTIONS.index(scene["step2"])
    quarter_turns = abs(first - second)
    separation = min(quarter_turns, 4 - quarter_turns) * 90.0
    return abs(separation - 135.0)


def is_quarantined(scene: dict) -> bool:
    return int(scene["glyph_size"]) < 10 or margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
