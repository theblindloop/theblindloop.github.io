"""Independent inverse arm using PNG pixels and the public shape rule only."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _foreground(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[0] < 96 or rgb.shape[1] < 96:
        raise ValueError("canvas is too small")
    corners = np.stack((rgb[0, 0], rgb[0, -1], rgb[-1, 0], rgb[-1, -1]))
    background = np.median(corners, axis=0)
    if np.max(np.linalg.norm(corners - background, axis=1)) > 8:
        raise ValueError("canvas corners disagree")
    distance = np.linalg.norm(rgb - background, axis=2)
    return distance > 55.0


def _one_component(mask: np.ndarray) -> np.ndarray:
    ys, xs = np.nonzero(mask)
    if len(xs) < 250:
        raise ValueError("too little foreground")
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    if x0 == 0 or y0 == 0 or x1 == mask.shape[1] - 1 or y1 == mask.shape[0] - 1:
        raise ValueError("foreground touches border")
    crop = mask[y0:y1 + 1, x0:x1 + 1]
    seen = np.zeros_like(crop, dtype=bool)
    start = tuple(np.argwhere(crop)[0])
    stack = [start]
    seen[start] = True
    reached = 0
    while stack:
        y, x = stack.pop()
        reached += 1
        for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= yy < crop.shape[0] and 0 <= xx < crop.shape[1] and crop[yy, xx] and not seen[yy, xx]:
                seen[yy, xx] = True
                stack.append((yy, xx))
    if reached != int(crop.sum()):
        raise ValueError("foreground is not one connected component")
    return crop


def _constant_runs(values: np.ndarray) -> list[tuple[int, int, int]]:
    runs = []
    start = 0
    for i in range(1, len(values) + 1):
        if i == len(values) or values[i] != values[start]:
            runs.append((start, i, int(values[start])))
            start = i
    return runs


def decision_from_image(image: Image.Image) -> str:
    try:
        crop = _one_component(_foreground(image))
        counts = crop.sum(axis=0).astype(int)
        if np.any(counts == 0):
            raise ValueError("component does not span every bounding-box column")
        runs = _constant_runs(counts)
        if len(runs) != 3:
            raise ValueError("expected exactly lobe, neck, lobe column runs")
        left_run, neck_run, right_run = runs
        if neck_run[2] >= min(left_run[2], right_run[2]):
            raise ValueError("middle run is not a neck")
        if neck_run[1] - neck_run[0] < 4 or min(left_run[1] - left_run[0], right_run[1] - right_run[0]) < 8:
            raise ValueError("profile spans are too narrow")
        # Each run must be a solid rectangle and the neck must join both lobes.
        for start, stop, count in runs:
            block = crop[:, start:stop]
            row_counts = block.sum(axis=1)
            occupied = row_counts > 0
            if int(occupied.sum()) != count or not np.all(row_counts[occupied] == stop - start):
                raise ValueError("a profile run is not a solid rectangle")
            indices = np.flatnonzero(occupied)
            if len(indices) == 0 or indices[-1] - indices[0] + 1 != len(indices):
                raise ValueError("a profile run is vertically fragmented")
        split = neck_run[0] + (neck_run[1] - neck_run[0]) // 2
        if (neck_run[1] - neck_run[0]) % 2:
            raise ValueError("neck midpoint falls between unequal pixel halves")
        left_area = int(crop[:, :split].sum())
        right_area = int(crop[:, split:].sum())
        delta = left_area - right_area
        if abs(delta) < 20:
            return "abstain"
        return "left" if delta > 0 else "right"
    except (IndexError, TypeError, ValueError):
        return "abstain"
