"""Deterministic compact two-lobe specimen renderer."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

CANVAS = 512
BACKGROUND = (247, 246, 242)
PALETTE = (
    (20, 132, 139),
    (47, 91, 166),
    (157, 67, 100),
    (116, 79, 151),
)
QUARANTINE_MARGIN = 20


def _geometry_pairs(side: str, band: int) -> list[tuple[int, int, int, int]]:
    dims = [(w, h) for w in range(8, 13, 2) for h in range(16, 31, 2)]
    limits = ((21, 50), (51, 100), (101, 160), (161, 260))[band]
    pairs = []
    for lw, lh in dims:
        for rw, rh in dims:
            delta = lw * lh - rw * rh
            if limits[0] <= abs(delta) <= limits[1] and ((delta > 0) == (side == "left")):
                pairs.append((lw, lh, rw, rh))
    return pairs


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    side = "left" if seed % 2 == 0 else "right"
    band = (seed // 2) % 4
    lw, lh, rw, rh = rng.choice(_geometry_pairs(side, band))
    neck_w = rng.choice((4, 6))
    neck_h = rng.choice((6, 8, 10))
    total_w = lw + neck_w + rw
    max_h = max(lh, rh)
    # The specimen varies across the canvas but stays within one 32x32 half-cell
    # of the public overlapping 8x8 ablation grid.
    left = rng.choice((65, 129, 193, 257, 321, 385)) + rng.randint(0, 1)
    top = rng.choice((65, 129, 193, 257, 321, 385)) + rng.randint(0, 1)
    return {
        "canvas_w": CANVAS,
        "canvas_h": CANVAS,
        "left": left,
        "top": top,
        "left_w": lw,
        "left_h": lh,
        "right_w": rw,
        "right_h": rh,
        "neck_w": neck_w,
        "neck_h": neck_h,
        "ink": list(rng.choice(PALETTE)),
    }


def _validate(scene: dict) -> None:
    required = {
        "canvas_w", "canvas_h", "left", "top", "left_w", "left_h",
        "right_w", "right_h", "neck_w", "neck_h", "ink",
    }
    if set(scene) != required:
        raise ValueError("scene fields do not match the public latent contract")
    values = [scene[k] for k in required - {"ink"}]
    if not all(isinstance(v, int) for v in values):
        raise ValueError("geometry and canvas values must be integers")
    if len(scene["ink"]) != 3 or not all(isinstance(v, int) and 0 <= v <= 255 for v in scene["ink"]):
        raise ValueError("ink must be an RGB triplet")
    if min(scene["left_w"], scene["right_w"]) < 8 or scene["neck_w"] < 4:
        raise ValueError("horizontal spans are too small")
    if scene["neck_w"] % 2 or scene["neck_h"] >= min(scene["left_h"], scene["right_h"]):
        raise ValueError("neck must be even-width and visibly narrower than both lobes")
    total_w = scene["left_w"] + scene["neck_w"] + scene["right_w"]
    max_h = max(scene["left_h"], scene["right_h"])
    if scene["left"] < 1 or scene["top"] < 1:
        raise ValueError("specimen touches the top or left border")
    if scene["left"] + total_w >= scene["canvas_w"] or scene["top"] + max_h >= scene["canvas_h"]:
        raise ValueError("specimen touches the bottom or right border")


def render(scene: dict) -> Image.Image:
    _validate(scene)
    width, height = scene["canvas_w"], scene["canvas_h"]
    raster = np.empty((height, width, 3), dtype=np.uint8)
    raster[:] = BACKGROUND
    ink = np.asarray(scene["ink"], dtype=np.uint8)
    x0 = scene["left"]
    x1 = x0 + scene["left_w"]
    x2 = x1 + scene["neck_w"]
    x3 = x2 + scene["right_w"]
    center_y = scene["top"] + max(scene["left_h"], scene["right_h"]) // 2
    ly0 = center_y - scene["left_h"] // 2
    ry0 = center_y - scene["right_h"] // 2
    ny0 = center_y - scene["neck_h"] // 2
    raster[ly0:ly0 + scene["left_h"], x0:x1] = ink
    raster[ny0:ny0 + scene["neck_h"], x1:x2] = ink
    raster[ry0:ry0 + scene["right_h"], x2:x3] = ink
    return Image.fromarray(raster, mode="RGB")


def _signed_area_difference(scene: dict) -> int:
    return scene["left_w"] * scene["left_h"] - scene["right_w"] * scene["right_h"]


def analytic_gold(scene: dict) -> str:
    _validate(scene)
    delta = _signed_area_difference(scene)
    if delta == 0:
        return "abstain"
    return "left" if delta > 0 else "right"


def margin(scene: dict) -> float:
    _validate(scene)
    return float(abs(_signed_area_difference(scene)))


def is_quarantined(scene: dict) -> bool:
    try:
        _validate(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _validate(scene)
    return []
