"""Deterministic sampler and raster renderer for nested carrier exchanges."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1240, 650
ANSWERS = ("LEFT", "MIDDLE", "RIGHT")
PALETTES = (
    ((52, 126, 192), (48, 154, 104), (203, 83, 145)),
    ((38, 108, 174), (31, 137, 88), (180, 69, 130)),
)
ORANGE_CARD = (248, 221, 183)
PURPLE_CARD = (225, 211, 243)
PARCEL_FILL = (250, 246, 220)
DOCK_FILL = (235, 239, 240)
BACKGROUND = (248, 248, 246)
DARK = (35, 40, 45)
RED = (211, 55, 62)

_PIPS = {
    1: ((0, 0),),
    2: ((-1, -1), (1, 1)),
    3: ((-1, -1), (0, 0), (1, 1)),
}


def _replay(
    cart_order: list[int], parcel_by_cart: list[int], operations: list[dict]
) -> tuple[list[int], list[int]]:
    """Replay independent dock->cart and cart->parcel bindings.

    Orange events name persistent cart identities. Purple events name current
    dock positions, so earlier cart moves determine which cart contents they touch.
    """
    order = [int(value) for value in cart_order]
    contents = [int(value) for value in parcel_by_cart]
    for operation in operations:
        kind = operation["kind"]
        a, b = int(operation["a"]), int(operation["b"])
        if kind == "cart":
            dock_a, dock_b = order.index(a), order.index(b)
            order[dock_a], order[dock_b] = order[dock_b], order[dock_a]
        elif kind == "parcel":
            cart_a, cart_b = order[a], order[b]
            contents[cart_a], contents[cart_b] = contents[cart_b], contents[cart_a]
        else:
            raise ValueError(f"unknown operation kind {kind!r}")
    return order, contents


def sample_scene(seed: int) -> dict:
    """Balance all three answers in every consecutive trio of seeds."""
    seed = int(seed)
    answer_dock = seed % 3
    group = seed // 3
    rng = random.Random(0xCA771E + group * 104729)

    cart_order = [0, 1, 2]
    parcel_by_cart = [1, 2, 3]
    rng.shuffle(cart_order)
    rng.shuffle(parcel_by_cart)

    pairs = [(0, 1), (0, 2), (1, 2)]
    operations = [
        {"kind": kind, "a": pair[0], "b": pair[1]}
        for kind in ("cart", "parcel")
        for pair in pairs
    ]
    extra_pair = rng.choice(pairs)
    operations.append(
        {"kind": rng.choice(("cart", "parcel")), "a": extra_pair[0], "b": extra_pair[1]}
    )
    rng.shuffle(operations)

    final_order, final_contents = _replay(cart_order, parcel_by_cart, operations)
    cart_at_answer = final_order[answer_dock]
    target_parcel = final_contents[cart_at_answer]
    return {
        "cart_order": cart_order,
        "parcel_by_cart": parcel_by_cart,
        "operations": operations,
        "target_parcel": target_parcel,
        "palette_variant": group % 2,
        "layout_variant": (group // 2) % 2,
    }


def analytic_gold(scene: dict) -> str:
    order, contents = _replay(
        scene["cart_order"], scene["parcel_by_cart"], scene["operations"]
    )
    target = int(scene["target_parcel"])
    target_cart = contents.index(target)
    return ANSWERS[order.index(target_cart)]


def margin(scene: dict) -> float:
    layout = int(scene["layout_variant"])
    palette = PALETTES[int(scene["palette_variant"])]
    card_separation = math.dist(ORANGE_CARD, PURPLE_CARD) / 5.0
    identity_separation = min(
        math.dist(palette[a], palette[b]) for a in range(3) for b in range(a + 1, 3)
    ) / 5.0
    card_gap_clearance = float((22 + 4 * layout) - 8)
    return float(min(card_separation, identity_separation, card_gap_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _validate(scene: dict) -> None:
    if sorted(int(value) for value in scene["cart_order"]) != [0, 1, 2]:
        raise ValueError("cart_order must be a permutation of 0,1,2")
    if sorted(int(value) for value in scene["parcel_by_cart"]) != [1, 2, 3]:
        raise ValueError("parcel_by_cart must be a permutation of 1,2,3")
    operations = scene["operations"]
    if not isinstance(operations, list) or len(operations) != 7:
        raise ValueError("exactly seven operations are required")
    for operation in operations:
        if set(operation) != {"kind", "a", "b"}:
            raise ValueError("operation fields are invalid")
        a, b = int(operation["a"]), int(operation["b"])
        if operation["kind"] not in {"cart", "parcel"} or a == b or not {a, b} <= {0, 1, 2}:
            raise ValueError("operation operands are invalid")
    if int(scene["target_parcel"]) not in (1, 2, 3):
        raise ValueError("target_parcel must be 1, 2, or 3")
    if int(scene["palette_variant"]) not in (0, 1):
        raise ValueError("unknown palette variant")
    if int(scene["layout_variant"]) not in (0, 1):
        raise ValueError("unknown layout variant")


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in values)  # type: ignore[return-value]


def _draw_pips(
    draw: ImageDraw.ImageDraw,
    value: int,
    center: tuple[float, float],
    spacing: float,
    radius: float,
    scale: int,
) -> None:
    for dx, dy in _PIPS[value]:
        cx, cy = center[0] + dx * spacing, center[1] + dy * spacing
        draw.ellipse(_box((cx - radius, cy - radius, cx + radius, cy + radius), scale), fill=DARK)


def _draw_cart(
    draw: ImageDraw.ImageDraw,
    center_x: float,
    body_y: float,
    color: tuple[int, int, int],
    scale: int,
    *,
    miniature: bool,
) -> None:
    if miniature:
        half_w, height, radius = 25, 30, 5
        wheel_r = 4
        draw.rounded_rectangle(
            _box((center_x - half_w, body_y, center_x + half_w, body_y + height), scale),
            radius=radius * scale,
            fill=color,
            outline=DARK,
            width=2 * scale,
        )
        for wheel_x in (center_x - 14, center_x + 14):
            draw.ellipse(
                _box((wheel_x - wheel_r, body_y + height - 1, wheel_x + wheel_r, body_y + height + 2 * wheel_r - 1), scale),
                fill=DARK,
            )
    else:
        half_w, height, radius = 95, 90, 11
        wheel_r = 15
        draw.rounded_rectangle(
            _box((center_x - half_w, body_y, center_x + half_w, body_y + height), scale),
            radius=radius * scale,
            fill=color,
            outline=DARK,
            width=3 * scale,
        )
        for wheel_x in (center_x - 55, center_x + 55):
            draw.ellipse(
                _box((wheel_x - wheel_r, body_y + height - 2, wheel_x + wheel_r, body_y + height + 2 * wheel_r - 2), scale),
                fill=DARK,
            )


def _draw_exchange_arrow(
    draw: ImageDraw.ImageDraw, x_left: float, x_right: float, y: float, scale: int
) -> None:
    draw.line(
        [(round(x_left * scale), round(y * scale)), (round(x_right * scale), round(y * scale))],
        fill=(92, 94, 96),
        width=3 * scale,
    )
    head = 5
    draw.polygon(
        [
            (round(x_left * scale), round(y * scale)),
            (round((x_left + head) * scale), round((y - head) * scale)),
            (round((x_left + head) * scale), round((y + head) * scale)),
        ],
        fill=(92, 94, 96),
    )
    draw.polygon(
        [
            (round(x_right * scale), round(y * scale)),
            (round((x_right - head) * scale), round((y - head) * scale)),
            (round((x_right - head) * scale), round((y + head) * scale)),
        ],
        fill=(92, 94, 96),
    )


def render(scene: dict) -> Image.Image:
    _validate(scene)
    cart_order = [int(value) for value in scene["cart_order"]]
    parcel_by_cart = [int(value) for value in scene["parcel_by_cart"]]
    operations = list(scene["operations"])
    target_parcel = int(scene["target_parcel"])
    palette_variant = int(scene["palette_variant"])
    layout_variant = int(scene["layout_variant"])
    palette = PALETTES[palette_variant]

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    shift_x = 8 if layout_variant else 0
    shift_y = 7 if layout_variant else 0
    dock_centers = (210 + shift_x, 620 + shift_x, 1030 + shift_x)
    dock_top, dock_bottom = 48 + shift_y, 315 + shift_y
    for dock_index, center_x in enumerate(dock_centers):
        draw.rounded_rectangle(
            _box((center_x - 142, dock_top, center_x + 142, dock_bottom), scale),
            radius=14 * scale,
            fill=DOCK_FILL,
            outline=(116, 126, 132),
            width=3 * scale,
        )
        # A pip-coded dock badge makes the purple card reference frame explicit.
        draw.rounded_rectangle(
            _box((center_x - 36, dock_bottom - 47, center_x + 36, dock_bottom - 9), scale),
            radius=8 * scale,
            fill=PARCEL_FILL,
            outline=(96, 91, 76),
            width=2 * scale,
        )
        _draw_pips(draw, dock_index + 1, (center_x, dock_bottom - 28), 11, 4.3, scale)

    for dock_index, cart_id in enumerate(cart_order):
        center_x = dock_centers[dock_index]
        _draw_cart(draw, center_x, 137 + shift_y, palette[cart_id], scale, miniature=False)
        parcel = parcel_by_cart[cart_id]
        parcel_box = (center_x - 39, 92 + shift_y, center_x + 39, 168 + shift_y)
        draw.rounded_rectangle(
            _box(parcel_box, scale),
            radius=9 * scale,
            fill=PARCEL_FILL,
            outline=DARK,
            width=3 * scale,
        )
        if parcel == target_parcel:
            draw.rounded_rectangle(
                _box((parcel_box[0] - 8, parcel_box[1] - 8, parcel_box[2] + 8, parcel_box[3] + 8), scale),
                radius=13 * scale,
                outline=RED,
                width=5 * scale,
            )
        _draw_pips(draw, parcel, (center_x, 130 + shift_y), 14, 6.0, scale)

    card_w, card_h = 150, 184
    card_gap = 22 + 4 * layout_variant
    total_width = 7 * card_w + 6 * card_gap
    start_x = (WIDTH - total_width) / 2
    card_y = 390 - 5 * layout_variant
    for index, operation in enumerate(operations):
        x0 = start_x + index * (card_w + card_gap)
        kind = operation["kind"]
        fill = ORANGE_CARD if kind == "cart" else PURPLE_CARD
        draw.rounded_rectangle(
            _box((x0, card_y, x0 + card_w, card_y + card_h), scale),
            radius=12 * scale,
            fill=fill,
            outline=(83, 77, 70),
            width=3 * scale,
        )
        left_x, right_x = x0 + 42, x0 + card_w - 42
        a, b = int(operation["a"]), int(operation["b"])
        if kind == "cart":
            _draw_cart(draw, left_x, card_y + 57, palette[a], scale, miniature=True)
            _draw_cart(draw, right_x, card_y + 57, palette[b], scale, miniature=True)
            _draw_exchange_arrow(draw, left_x + 27, right_x - 27, card_y + 121, scale)
            # Two dark wheel marks at the footer reinforce whole-carrier motion.
            for footer_x in (x0 + 64, x0 + 86):
                draw.ellipse(_box((footer_x - 6, card_y + 146, footer_x + 6, card_y + 158), scale), fill=DARK)
        else:
            for value, token_x in ((a + 1, left_x), (b + 1, right_x)):
                draw.rounded_rectangle(
                    _box((token_x - 25, card_y + 54, token_x + 25, card_y + 108), scale),
                    radius=9 * scale,
                    fill=PARCEL_FILL,
                    outline=DARK,
                    width=2 * scale,
                )
                _draw_pips(draw, value, (token_x, card_y + 81), 10, 4.5, scale)
            _draw_exchange_arrow(draw, left_x + 27, right_x - 27, card_y + 132, scale)
            # A small parcel outline at the footer reinforces content-only motion.
            draw.rectangle(
                _box((x0 + 65, card_y + 146, x0 + 85, card_y + 164), scale),
                outline=DARK,
                width=3 * scale,
            )

        if index < 6:
            arrow_x = x0 + card_w + card_gap / 2
            arrow_y = card_y + card_h / 2
            draw.line(
                [(round((arrow_x - 6) * scale), round(arrow_y * scale)),
                 (round((arrow_x + 5) * scale), round(arrow_y * scale))],
                fill=(102, 102, 99), width=3 * scale,
            )
            draw.polygon(
                [(round((arrow_x + 6) * scale), round(arrow_y * scale)),
                 (round(arrow_x * scale), round((arrow_y - 6) * scale)),
                 (round(arrow_x * scale), round((arrow_y + 6) * scale))],
                fill=(102, 102, 99),
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
