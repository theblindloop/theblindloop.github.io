"""Independent pixel-only decision path for net_reaches_more_pads.

Receives a PIL image and nothing else: it segments the four painted materials,
labels the dark wire into connected components, attaches each source square and
each pad to the wire component it touches, and compares supplied pad counts.
"""

from __future__ import annotations

from collections import deque

import numpy as np

REFERENCE = {
    "wire": (38, 38, 42),
    "pad": (240, 142, 24),
    "green": (26, 148, 74),
    "purple": (128, 58, 178),
    "background": (255, 255, 255),
}
MATCH_TOLERANCE = 90.0
DILATE = 3


def _classify(image) -> dict[str, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    names = list(REFERENCE)
    refs = np.array([REFERENCE[n] for n in names], dtype=np.int16)
    d = np.linalg.norm(arr[:, :, None, :] - refs[None, None, :, :], axis=3)
    best = np.argmin(d, axis=2)
    close = np.min(d, axis=2) <= MATCH_TOLERANCE
    return {n: (best == i) & close for i, n in enumerate(names)}


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    labels = np.zeros(mask.shape, dtype=np.int32)
    h, w = mask.shape
    current = 0
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if labels[sy, sx]:
            continue
        current += 1
        labels[sy, sx] = current
        queue = deque([(sy, sx)])
        while queue:
            y, x = queue.popleft()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = current
                    queue.append((ny, nx))
    return labels, current


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    out = mask.copy()
    for _ in range(radius):
        grown = out.copy()
        grown[1:, :] |= out[:-1, :]
        grown[:-1, :] |= out[1:, :]
        grown[:, 1:] |= out[:, :-1]
        grown[:, :-1] |= out[:, 1:]
        out = grown
    return out


def _touching_wire_labels(mask: np.ndarray, wire_labels: np.ndarray) -> set[int]:
    near = _dilate(mask, DILATE)
    found = np.unique(wire_labels[near])
    return {int(v) for v in found if v > 0}


def decision_from_image(image) -> str:
    masks = _classify(image)
    wire_labels, _ = _label(masks["wire"])
    pad_labels, n_pads = _label(masks["pad"])

    pad_nets = [
        _touching_wire_labels(pad_labels == pad, wire_labels) for pad in range(1, n_pads + 1)
    ]

    supplied: dict[str, int] = {}
    for colour in ("green", "purple"):
        source_labels, n_sources = _label(masks[colour])
        if n_sources != 1:
            return "abstain"
        nets = _touching_wire_labels(source_labels > 0, wire_labels)
        supplied[colour] = sum(1 for pad in pad_nets if pad & nets) if nets else 0

    if supplied.get("green", 0) == supplied.get("purple", 0):
        return "abstain"
    return "green" if supplied["green"] > supplied["purple"] else "purple"
