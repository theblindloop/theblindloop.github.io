"""Independent image-only inverse for the two-register state board."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 920
REGISTER_NAMES = ("A", "C", "V")
REGISTER_COLORS = np.array(
    [
        (245, 158, 11),
        (6, 182, 212),
        (139, 92, 246),
    ],
    dtype=np.int16,
)
START_CENTERS = ((135, 235), (300, 235))
COMMAND_X = (150, 335, 520, 705, 890, 1075, 1260)
COMMAND_Y = 530
COMMAND_COLOR_OFFSET = -28
COMMAND_SIGN_OFFSET = 33


def _nearest_color(rgb: np.ndarray) -> int:
    distances = ((REGISTER_COLORS.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=1)
    value = int(np.argmin(distances))
    if int(distances[value]) > 1800:
        raise ValueError("register color is missing or unreadable")
    return value


def _event_from_card(arr: np.ndarray, x: int) -> tuple[int, int]:
    color_center = (x + COMMAND_COLOR_OFFSET, COMMAND_Y)
    color = _nearest_color(arr[color_center[1], color_center[0]])
    sx = x + COMMAND_SIGN_OFFSET
    crop = arr[COMMAND_Y - 24:COMMAND_Y + 25, sx - 24:sx + 25]
    ink = np.all(crop == np.array((25, 31, 42), dtype=np.int16), axis=2)
    vertical = int(ink[:, 20:29].sum())
    if vertical > 200:
        return color, 1
    if int(ink.sum()) > 150:
        return color, -1
    raise ValueError("command sign is missing or unreadable")


def decision_from_image(image: Image.Image) -> str:
    """Recover the terminal ordered pair from raster pixels only."""
    image = image.convert("RGB")
    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {image.size}")
    arr = np.asarray(image, dtype=np.int16)
    left = _nearest_color(arr[START_CENTERS[0][1], START_CENTERS[0][0]])
    right = _nearest_color(arr[START_CENTERS[1][1], START_CENTERS[1][0]])
    for x in COMMAND_X:
        color, direction = _event_from_card(arr, x)
        if color == left:
            right = (right + direction) % 3
        else:
            left = (left - direction) % 3
    return REGISTER_NAMES[left] + REGISTER_NAMES[right]
