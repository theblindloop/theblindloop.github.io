"""Renderer and analytic world for a two-register conditional trace."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 920
BACKGROUND = (248, 250, 252)
INK = (25, 31, 42)
MUTED = (88, 98, 113)
PANEL = (255, 255, 255)
LINE = (199, 206, 216)
REGISTER_COLORS = (
    (245, 158, 11),   # amber
    (6, 182, 212),    # cyan
    (139, 92, 246),   # violet
)
REGISTER_NAMES = ("A", "C", "V")
REGISTER_WORDS = ("amber", "cyan", "violet")

START_LEFT_CENTER = (135, 235)
START_RIGHT_CENTER = (300, 235)
COMMAND_X = (150, 335, 520, 705, 890, 1075, 1260)
COMMAND_Y = 530
COMMAND_COLOR_OFFSET = -28
COMMAND_SIGN_OFFSET = 33
MODE_KEY_X = (575, 805, 1035)
MODE_KEY_Y = (725, 795, 865)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[int, int, list[tuple[int, int]]]:
    left = int(scene["start_left"])
    right = int(scene["start_right"])
    raw_commands = scene["commands"]
    commands = [(int(item[0]), int(item[1])) for item in raw_commands]
    if not 0 <= left < 3 or not 0 <= right < 3:
        raise ValueError("start registers must be amber, cyan, or violet")
    if len(commands) != 7:
        raise ValueError("commands must contain seven cards")
    if any(color not in range(3) or direction not in (-1, 1) for color, direction in commands):
        raise ValueError("command colors or signs are invalid")
    return left, right, commands


def _trace(left: int, right: int, commands: list[tuple[int, int]]) -> list[tuple[int, int]]:
    path = [(left, right)]
    for color, direction in commands:
        if color == left:
            right = (right + direction) % 3
        else:
            left = (left - direction) % 3
        path.append((left, right))
    return path


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced terminal pairs with both branch outcomes represented."""
    seed = int(seed)
    target = seed % 9
    for attempt in range(20000):
        rng = random.Random(seed * 1_000_003 + attempt * 91_879 + 29)
        left = rng.randrange(3)
        right = rng.randrange(3)
        commands = [(rng.randrange(3), rng.choice((-1, 1))) for _ in range(7)]
        path = _trace(left, right, commands)
        matches = sum(
            color == before_left
            for (before_left, _), (color, _) in zip(path[:-1], commands, strict=True)
        )
        if (
            path[-1][0] * 3 + path[-1][1] == target
            and len({color for color, _ in commands}) == 3
            and {-1, 1}.issubset({direction for _, direction in commands})
            and 2 <= matches <= 5
            and len(set(path)) >= 4
        ):
            return {
                "start_left": left,
                "start_right": right,
                "commands": [[color, direction] for color, direction in commands],
            }
    raise RuntimeError(f"could not sample a balanced register trace for seed {seed}")


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=LINE, width=2)


def _chip(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int, radius: int) -> None:
    cx, cy = center
    draw.ellipse(
        (cx - radius, cy - radius, cx + radius, cy + radius),
        fill=REGISTER_COLORS[value],
        outline=INK,
        width=3,
    )


def _sign(draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: int, half: int = 21) -> None:
    cx, cy = center
    draw.rectangle((cx - half, cy - 5, cx + half, cy + 5), fill=INK)
    if direction == 1:
        draw.rectangle((cx - 5, cy - half, cx + 5, cy + half), fill=INK)


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    draw.line((*start, *end), fill=LINE, width=5)
    ex, ey = end
    draw.polygon([(ex, ey), (ex - 14, ey - 8), (ex - 14, ey + 8)], fill=LINE)


def _pair_key_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], left: int, right: int) -> None:
    cx, cy = center
    draw.rounded_rectangle((cx - 100, cy - 25, cx + 100, cy + 25), radius=12, fill=(247, 249, 252), outline=LINE, width=2)
    _chip(draw, (cx - 50, cy), left, 17)
    _chip(draw, (cx + 50, cy), right, 17)
    draw.text((cx, cy + 1), REGISTER_NAMES[left] + REGISTER_NAMES[right], font=_font(19, True), fill=INK, anchor="mm")


def render(scene: dict[str, Any]) -> Image.Image:
    left, right, commands = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 24), "TWO-REGISTER STATE BOARD", font=_font(31, True), fill=INK)
    draw.text(
        (40, 64),
        "Each command branches on the left register; carry the resulting pair forward.",
        font=_font(17),
        fill=MUTED,
    )

    _panel(draw, (35, 105, 405, 360))
    draw.text((58, 130), "START PAIR", font=_font(20, True), fill=INK)
    draw.text((58, 164), "left register", font=_font(15), fill=MUTED)
    draw.text((223, 164), "right register", font=_font(15), fill=MUTED)
    _chip(draw, START_LEFT_CENTER, left, 48)
    _chip(draw, START_RIGHT_CENTER, right, 48)
    draw.text((START_LEFT_CENTER[0], 305), REGISTER_WORDS[left].upper(), font=_font(16, True), fill=INK, anchor="mm")
    draw.text((START_RIGHT_CENTER[0], 305), REGISTER_WORDS[right].upper(), font=_font(16, True), fill=INK, anchor="mm")
    draw.text((58, 340), "keep the order: L then R", font=_font(15), fill=MUTED)

    _panel(draw, (435, 105, 1365, 360))
    draw.text((460, 130), "COLOR CYCLE AND BRANCH RULE", font=_font(20, True), fill=INK)
    draw.text((460, 166), "forward: +     backward: −", font=_font(16), fill=MUTED)
    cycle_x = (645, 805, 965)
    for index, value in enumerate(range(3)):
        _chip(draw, (cycle_x[index], 225), value, 30)
        draw.text((cycle_x[index], 225), REGISTER_NAMES[value], font=_font(18, True), fill=(255, 255, 255), anchor="mm")
        if index < 2:
            _arrow(draw, (cycle_x[index] + 38, 225), (cycle_x[index + 1] - 38, 225))
    _arrow(draw, (cycle_x[2] - 38, 275), (cycle_x[0] + 38, 275))
    draw.text((1075, 216), "amber", font=_font(15, True), fill=INK)
    draw.text((1075, 242), "cyan", font=_font(15, True), fill=INK)
    draw.text((1075, 268), "violet", font=_font(15, True), fill=INK)
    draw.text((460, 308), "MATCH L  →  move R by the card sign", font=_font(16, True), fill=INK)
    draw.text((460, 335), "MISS L   →  move L opposite the card sign", font=_font(16, True), fill=INK)

    _panel(draw, (35, 390, 1365, 635))
    draw.text((58, 414), "COMMAND ORDER", font=_font(20, True), fill=INK)
    draw.text((250, 417), "read cards 1 through 7 from left to right", font=_font(16), fill=MUTED)
    for number, (x, (color, direction)) in enumerate(zip(COMMAND_X, commands, strict=True), start=1):
        draw.rounded_rectangle((x - 78, 455, x + 78, 605), radius=14, fill=(247, 249, 252), outline=LINE, width=2)
        draw.ellipse((x - 62, 466, x - 34, 494), fill=INK)
        draw.text((x - 48, 480), str(number), font=_font(13, True), fill=(255, 255, 255), anchor="mm")
        _chip(draw, (x + COMMAND_COLOR_OFFSET, COMMAND_Y), color, 28)
        _sign(draw, (x + COMMAND_SIGN_OFFSET, COMMAND_Y), direction)
        draw.text((x, 585), "color + sign", font=_font(12), fill=MUTED, anchor="mm")

    _panel(draw, (35, 665, 1365, 890))
    draw.text((58, 690), "FINAL PAIR KEY", font=_font(20, True), fill=INK)
    draw.text((255, 692), "report the matching pair code", font=_font(16), fill=MUTED)
    for row, left_value in enumerate(range(3)):
        for column, right_value in enumerate(range(3)):
            _pair_key_cell(draw, (MODE_KEY_X[column], MODE_KEY_Y[row]), left_value, right_value)
    draw.text((58, 858), "code order is always left register, then right register", font=_font(15), fill=MUTED)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    left, right, commands = _validate_scene(scene)
    final_left, final_right = _trace(left, right, commands)[-1]
    return REGISTER_NAMES[final_left] + REGISTER_NAMES[final_right]


def margin(scene: dict[str, Any]) -> float:
    _validate_scene(scene)
    return 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    _validate_scene(scene)
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
