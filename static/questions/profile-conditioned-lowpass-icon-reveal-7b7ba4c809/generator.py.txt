"""lowpass_icon_reveal: hybrid low-pass icon reveal world.

Declared transform: low-pass average (blur) the image to destroy the
high-frequency per-pixel speckle, leaving only the coarse two-region structure.
A single block glyph (the letter A or the letter B) is carried entirely in the
low-frequency mean of the noise: inside the glyph the per-pixel speckle is
centred on a dark mean, outside on a light mean. Viewed as raw salt-and-pepper
specks the answer is invisible; only the declared blur exposes the glyph.

Latent scene z = {noise, glyph, scale, pos_r, pos_c}. label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "lowpass_icon_reveal-0.1.0"

W = 256
H = 256

GR = 9          # template rows
GC = 7          # template cols


def template_array(glyph: str) -> np.ndarray:
    """Block templates on a GR x GC grid.

    A = a thick rising diagonal slash "/",
    B = a thick falling diagonal slash "\\".
    Each is a solid diagonal band three cells wide, so the orientation survives
    low-pass smoothing and coarse resampling and is distinguished structurally.
    """
    grid = np.zeros((GR, GC), dtype=bool)
    thickness = 1
    for row in range(GR):
        col_main = (GC - 1) * (GR - 1 - row) // (GR - 1)
        for t in range(-thickness, thickness + 1):
            c = col_main + t
            if 0 <= c < GC:
                if glyph == "A":
                    grid[row, c] = True
                else:
                    grid[row, GC - 1 - c] = True
    return grid

# Mean luminance anchors: glyph region is dark, background is light.
GLYPH_MEAN = 60.0
BG_MEAN = 196.0
SWING = 80.0    # half-width of uniform per-pixel speckle amplitude

Q_TOL = 0.15    # quarantine threshold on the analytic A/B correlation margin


def build_scene(noise: int, glyph: str, scale: int, pos_r: int, pos_c: int) -> dict:
    return {
        "noise": int(noise),
        "glyph": glyph,
        "scale": int(scale),
        "pos_r": int(pos_r),
        "pos_c": int(pos_c),
    }


def sample_scene(seed: int) -> dict:
    """One deterministic scene from a scalar seed (interface requirement)."""
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    scale = int(rng.choice([12, 14, 16]))
    glyph_h = GR * scale
    glyph_w = GC * scale
    max_r = H - glyph_h - 24
    max_c = W - glyph_w - 24
    if max_r <= 0 or max_c <= 0:
        max_r = max_c = 1
    pos_r = int(rng.integers(18, max(max_r, 19)))
    pos_c = int(rng.integers(18, max(max_c, 19)))
    return build_scene(noise=int(seed), glyph=glyph, scale=scale, pos_r=pos_r, pos_c=pos_c)


def glyph_mask(scene: dict) -> np.ndarray:
    """(H, W) boolean ink mask of the glyph, at the declared scale/position."""
    scale = int(scene.get("scale", 14))
    row0 = int(scene.get("pos_r", 18))
    col0 = int(scene.get("pos_c", 18))
    cells = template_array(scene["glyph"])
    block = np.repeat(np.repeat(cells, scale, axis=0), scale, axis=1)
    mask = np.zeros((H, W), dtype=bool)
    r1 = min(H, row0 + block.shape[0])
    c1 = min(W, col0 + block.shape[1])
    mask[row0:r1, col0:c1] = block[: r1 - row0, : c1 - col0]
    return mask


def base_image(scene: dict) -> np.ndarray:
    """(H, W, 3) uint8 RGB with the glyph region at GLYPH_MEAN, else BG_MEAN."""
    ink = glyph_mask(scene)
    arr = np.where(
        ink[:, :, None],
        np.full(3, GLYPH_MEAN, dtype=np.float64),
        np.full(3, BG_MEAN, dtype=np.float64),
    )
    return arr


def render(scene: dict) -> Image.Image:
    """Render the noisy raster. Deterministic given the latent scene."""
    scene = dict(scene)
    rng = np.random.default_rng(int(scene.get("noise", 0)) & 0xFFFFFFFF)
    base = base_image(scene)
    noise = rng.uniform(-SWING, SWING, size=(H, W, 1))
    pixels = np.clip(base + noise, 0, 255).astype(np.uint8)
    return Image.fromarray(pixels, "RGB")


def margin(scene: dict) -> float:
    """Analytic A/B template correlation margin, deterministic and scene-free
    of noise (only the declared glyph matters)."""
    chosen = template_array(scene["glyph"])
    other = template_array("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_chosen = float((chosen & chosen).sum()) / max(1, c_filled)
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(corr_chosen - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the random speckle seed, the answer
    glyph, the block scale, the placement row and column) directly changes the
    rendered pixels, so the latent-to-raster map is injective and declared empty."""
    return iter(())
