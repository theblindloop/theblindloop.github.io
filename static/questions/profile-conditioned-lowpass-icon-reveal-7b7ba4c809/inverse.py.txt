"""Independent pixel-only inverse arm for lowpass_icon_reveal.

Receives ONLY the final PNG. It re-discovers the revealed symbol by applying
the declared low-pass (box-average) transform to destroy the per-pixel speckle,
thresholding the smoothed two-region field with an automatic (Otsu) histogram
cut, cropping the surviving dark diagonal-band silhouette, and classifying its
orientation against the two public candidates: a rising slash "/" (A) versus a
falling slash "\\" (B).

The slash-orientation signature is structurally robust: it is a coarse diagonal
band whose tilt survives low-pass smoothing and coarse resampling.

No scene data, no analytic gold, no renderer import.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageFilter

ORACLE_VERSION = "lowpass_icon_reveal-oracle-0.1.0"

GR = 9
GC = 7
BLUR_RADIUS = 4


def _otsu_threshold(values: np.ndarray) -> float:
    """Two-class histogram threshold maximizing between-class variance (Otsu)."""
    hist, edges = np.histogram(values, bins=256, range=(0.0, 255.0))
    counts = hist.astype(np.float64)
    total = counts.sum()
    if total <= 0:
        return 127.5
    weighted = edges[:-1] + 0.5
    sum_total = (counts * weighted).sum()
    w_0 = 0.0
    sum_0 = 0.0
    best = 0.0
    best_thresh = 127.5
    for index in range(256):
        w_0 += counts[index]
        if w_0 == 0:
            continue
        w_1 = total - w_0
        if w_1 == 0:
            break
        sum_0 += counts[index] * weighted[index]
        m_0 = sum_0 / w_0
        m_1 = (sum_total - sum_0) / w_1
        between = w_0 * w_1 * (m_0 - m_1) ** 2
        if between > best:
            best = between
            best_thresh = weighted[index]
    return best_thresh


def _orientation_scores(binary: np.ndarray) -> dict[str, float]:
    """Score how much recovered ink sits on the rising "/" vs falling "\\" band.

    For each grid cell a band-weight w(row, col) is 1 near the main diagonal
    (A, "/", rising from bottom-left to top-right) and 0 there for the anti
    diagonal (B). The A score uses the main band, the B score the anti band.
    """
    rows, cols = binary.shape
    main = np.zeros((rows, cols), dtype=np.float64)
    anti = np.zeros((rows, cols), dtype=np.float64)
    for r in range(rows):
        c_main = (cols - 1) * (rows - 1 - r) // (rows - 1)
        for t in range(-1, 2):
            c = c_main + t
            if 0 <= c < cols:
                main[r, c] = 1.0
                anti[r, cols - 1 - c] = 1.0
    mass = float(binary.sum())
    if mass <= 0:
        return {"A": 0.0, "B": 0.0}
    return {
        "A": float((binary * main).sum()) / mass,
        "B": float((binary * anti).sum()) / mass,
    }


def decision_from_image(image: Image.Image) -> str:
    gray = np.asarray(image.convert("L"), dtype=np.float32)
    blurred = np.asarray(
        Image.fromarray(gray.astype(np.uint8)).filter(ImageFilter.BoxBlur(BLUR_RADIUS)),
        dtype=np.float32,
    )
    thresh = _otsu_threshold(blurred)
    ink = blurred < thresh  # dark survivors = the smoothed glyph silhouette
    if ink.sum() == 0:
        return "A"
    coords = np.argwhere(ink)
    r0, c0 = coords.min(axis=0)
    r1, c1 = coords.max(axis=0)
    crop = ink[r0 : r1 + 1, c0 : c1 + 1].astype(np.uint8) * 255
    if crop.size == 0:
        return "A"
    resized = np.asarray(
        Image.fromarray(crop).resize((GC, GR), Image.Resampling.BILINEAR), dtype=np.float32
    )
    binary = resized > 127.5
    scores = _orientation_scores(binary)
    return "A" if scores["A"] >= scores["B"] else "B"
