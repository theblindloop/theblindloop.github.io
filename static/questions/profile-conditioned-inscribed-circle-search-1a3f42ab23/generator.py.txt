"""Deterministic renderer and analytic arm for three-side tangency search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 360
LABELS = ("A", "B", "C", "D")
SIDE_COLORS = ("#245B78", "#7B3F67", "#27705D")
AMBER = "#E7A928"
INK = "#20252B"
PANEL = "#B8C0C8"
BACKGROUND = "#F7F4ED"


def _vertices(cx: float, cy: float, radius: float, rotation: float) -> list[list[float]]:
    return [
        [
            cx + radius * math.cos(rotation + 2.0 * math.pi * i / 3.0),
            cy + radius * math.sin(rotation + 2.0 * math.pi * i / 3.0),
        ]
        for i in range(3)
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = rng.randrange(4)
    panels = []
    base_radius = rng.uniform(78.0, 91.0)
    displacement_sizes = [rng.uniform(11.0, 15.0), rng.uniform(17.0, 21.0), rng.uniform(23.0, 27.0)]
    rng.shuffle(displacement_sizes)
    distractor_number = 0
    for index in range(4):
        cx = 120.0 + 240.0 * index
        cy = 188.0 + rng.uniform(-4.0, 4.0)
        triangle_radius = base_radius + rng.uniform(-2.0, 2.0)
        rotation = rng.uniform(-0.28, 0.28) - math.pi / 2.0
        vertices = _vertices(cx, cy, triangle_radius, rotation)
        circle_radius = triangle_radius / 2.0
        if index == winner:
            dx = dy = 0.0
        else:
            shift = displacement_sizes[distractor_number]
            distractor_number += 1
            direction = rng.uniform(0.0, 2.0 * math.pi)
            dx, dy = shift * math.cos(direction), shift * math.sin(direction)
        panels.append(
            {
                "vertices": vertices,
                "side_colors": list(SIDE_COLORS),
                "circle": [cx + dx, cy + dy, circle_radius],
            }
        )
    return {"panels": panels}


def _line_distance(point: tuple[float, float], a: list[float], b: list[float]) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    return abs((bx - ax) * (ay - py) - (ax - px) * (by - ay)) / math.hypot(bx - ax, by - ay)


def _panel_error(panel: dict) -> float:
    cx, cy, radius = panel["circle"]
    vertices = panel["vertices"]
    clearances = [
        _line_distance((cx, cy), vertices[i], vertices[(i + 1) % 3]) - radius
        for i in range(3)
    ]
    return sum(abs(value) for value in clearances)


def analytic_gold(scene: dict) -> str:
    # Keep the label arm's only top-level latent input explicit for the audit:
    # all nested geometry remains visibly rendered through the panel list.
    scores = []
    for panel in scene["panels"]:
        cx, cy, radius = panel["circle"]
        vertices = panel["vertices"]
        clearances = [
            _line_distance((cx, cy), vertices[i], vertices[(i + 1) % 3]) - radius
            for i in range(3)
        ]
        scores.append(sum(abs(value) for value in clearances))
    winner = min(range(4), key=scores.__getitem__)
    return LABELS[winner]


def margin(scene: dict) -> float:
    scores = sorted(_panel_error(panel) for panel in scene["panels"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["serialization_alias"] = "panel_geometry_v1"
    return [("renderer_inert_serialization_alias", twin)]


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=22 * scale)
    title_font = ImageFont.load_default(size=24 * scale)
    draw.text((WIDTH * scale / 2, 20 * scale), "FIND THE THREE-SIDE TANGENCY", fill=INK, font=title_font, anchor="ma")
    for index, panel in enumerate(scene["panels"]):
        left, right = (10 + 240 * index) * scale, (230 + 240 * index) * scale
        draw.rounded_rectangle((left, 60 * scale, right, 330 * scale), radius=12 * scale, outline=PANEL, width=2 * scale)
        cx, cy, radius = panel["circle"]
        draw.ellipse(
            ((cx - radius) * scale, (cy - radius) * scale, (cx + radius) * scale, (cy + radius) * scale),
            fill=AMBER,
            outline=INK,
            width=2 * scale,
        )
        vertices = panel["vertices"]
        for side in range(3):
            a = tuple(value * scale for value in vertices[side])
            b = tuple(value * scale for value in vertices[(side + 1) % 3])
            draw.line((a, b), fill=panel["side_colors"][side], width=6 * scale)
        draw.text(((120 + 240 * index) * scale, 307 * scale), LABELS[index], fill=INK, font=font, anchor="mm")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
