"""Pixel-only inverse arm. This module intentionally has no world imports."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
SIDE_RGB = ((36, 91, 120), (123, 63, 103), (39, 112, 93))
AMBER_RGB = (231, 169, 40)


def _mask(rgb: np.ndarray, color: tuple[int, int, int], tolerance: float = 20.0) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _fit_line(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("side missing")
    points = np.column_stack((xs.astype(float), ys.astype(float)))
    center = points.mean(axis=0)
    covariance = np.cov(points - center, rowvar=False)
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    return center, direction


def _distance(point: np.ndarray, line_center: np.ndarray, direction: np.ndarray) -> float:
    delta = point - line_center
    return abs(float(delta[0] * direction[1] - delta[1] * direction[0]))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.shape[1] < 400 or rgb.shape[0] < 150:
        return "abstain"
    scores = []
    panel_width = rgb.shape[1] / 4.0
    for index in range(4):
        x0, x1 = int(index * panel_width), int((index + 1) * panel_width)
        crop = rgb[int(rgb.shape[0] * 0.18):int(rgb.shape[0] * 0.82), x0:x1]
        amber = _mask(crop, AMBER_RGB, 64.0)
        ys, xs = np.nonzero(amber)
        if len(xs) < 300:
            return "abstain"
        center = np.array([xs.mean(), ys.mean()])
        radius = math.sqrt(len(xs) / math.pi)
        residuals = []
        side_points = []
        for color in SIDE_RGB:
            side_mask = _mask(crop, color)
            sy, sx = np.nonzero(side_mask)
            side_points.append(np.column_stack((sx.astype(float), sy.astype(float))))
            line_center, direction = _fit_line(side_mask)
            residuals.append(abs(_distance(center, line_center, direction) - radius))
        triangle_center = np.concatenate(side_points, axis=0).mean(axis=0)
        # For an equilateral triangle the equal-length side-pixel centroid is its
        # incenter.  The residual term independently checks the three contacts.
        scores.append(float(np.linalg.norm(center - triangle_center)) + 0.1 * sum(residuals))
    order = np.argsort(scores)
    if float(scores[int(order[1])] - scores[int(order[0])]) < 3.0:
        return "abstain"
    return LABELS[int(order[0])]
