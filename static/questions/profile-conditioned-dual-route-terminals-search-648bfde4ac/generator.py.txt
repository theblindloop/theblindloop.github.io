"""Deterministic scene sampler and raster renderer for dual-route search."""

from __future__ import annotations

import itertools
import random
from collections import deque

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")
GRID_EDGES = tuple(
    sorted(
        [(r * 3 + c, r * 3 + c + 1) for r in range(3) for c in range(2)]
        + [(r * 3 + c, (r + 1) * 3 + c) for r in range(2) for c in range(3)]
    )
)


def _connected(edges: tuple[tuple[int, int], ...]) -> bool:
    adjacency = {node: set() for node in range(9)}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    reached = {0}
    pending = [0]
    while pending:
        reached.update(adjacency[pending.pop()] - reached)
        pending.extend(adjacency[node] - reached for node in ())
        # The explicit scan keeps the helper compact and deterministic.
        for node in tuple(reached):
            for neighbor in adjacency[node]:
                if neighbor not in reached:
                    reached.add(neighbor)
                    pending.append(neighbor)
    return len(reached) == 9


def _all_paths(edges: tuple[tuple[int, int], ...], start: int, goal: int) -> list[tuple[int, ...]]:
    adjacency = {node: set() for node in range(9)}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    paths: list[tuple[int, ...]] = []

    def visit(node: int, path: tuple[int, ...]) -> None:
        if node == goal:
            paths.append(path)
            return
        for neighbor in sorted(adjacency[node]):
            if neighbor not in path:
                visit(neighbor, path + (neighbor,))

    visit(start, (start,))
    return paths


def _route_score(panel: dict) -> int:
    edges = tuple(tuple(edge) for edge in panel["edges"])
    start, goal = panel["terminals"]
    paths = _all_paths(edges, start, goal)
    for first, second in itertools.combinations(paths, 2):
        first_edges = {tuple(sorted(pair)) for pair in zip(first, first[1:])}
        second_edges = {tuple(sorted(pair)) for pair in zip(second, second[1:])}
        if not (set(first[1:-1]) & set(second[1:-1])) and not (
            first_edges & second_edges
        ):
            return 2
    return 1 if paths else 0


CONNECTED_UNICYCLIC = tuple(
    edges for edges in itertools.combinations(GRID_EDGES, 9) if _connected(edges)
)


def sample_scene(seed: int) -> dict:
    """Sample four equal-ink unicyclic graphs with one unique dual-route panel."""

    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    terminal_pairs = [
        pair
        for pair in itertools.combinations(range(9), 2)
        if abs(pair[0] // 3 - pair[1] // 3) + abs(pair[0] % 3 - pair[1] % 3) >= 2
    ]
    panels = []
    used: set[tuple[tuple[int, int], ...]] = set()
    for index in range(4):
        want = 2 if index == answer_index else 1
        options = []
        pairs = list(terminal_pairs)
        rng.shuffle(pairs)
        for terminals in pairs:
            for edges in CONNECTED_UNICYCLIC:
                panel = {"edges": [list(edge) for edge in edges], "terminals": list(terminals)}
                if edges not in used and _route_score(panel) == want:
                    options.append((terminals, edges))
            if len(options) >= 20:
                break
        terminals, edges = rng.choice(options)
        used.add(edges)
        panels.append(
            {
                "edges": [list(edge) for edge in edges],
                "terminals": list(terminals),
            }
        )
    return {
        "edges": [panel["edges"] for panel in panels],
        "terminals": [panel["terminals"] for panel in panels],
    }


def _panels(scene: dict) -> list[dict]:
    return [
        {"edges": edges, "terminals": terminals}
        for edges, terminals in zip(scene["edges"], scene["terminals"])
    ]


def analytic_gold(scene: dict) -> str:
    scores = [_route_score(panel) for panel in _panels(scene)]
    winners = [index for index, score in enumerate(scores) if score == 2]
    return LABELS[winners[0]] if len(winners) == 1 else "abstain"


def margin(scene: dict) -> float:
    scores = sorted((_route_score(panel) for panel in _panels(scene)), reverse=True)
    return float(scores[0] - scores[1]) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (1100, 520), "white")
    draw = ImageDraw.Draw(image)
    draw.text((550, 30), "FIND TWO ROUTES THAT SHARE ONLY THEIR COLORED ENDS", fill=(28, 40, 46), font=_font(22, True), anchor="mm")

    dark = (38, 50, 56)
    neutral = (94, 145, 160)
    magenta = (216, 27, 96)
    amber = (244, 163, 0)
    for panel_index, panel in enumerate(_panels(scene)):
        left = 18 + panel_index * 270
        draw.rounded_rectangle((left, 68, left + 254, 455), radius=15, fill=(248, 250, 251), outline=(174, 188, 194), width=3)
        positions = {
            node: (left + 47 + (node % 3) * 80, 135 + (node // 3) * 105)
            for node in range(9)
        }
        for first, second in panel["edges"]:
            draw.line((positions[first], positions[second]), fill=dark, width=9)
        magenta_node, amber_node = panel["terminals"]
        for node, center in positions.items():
            fill = magenta if node == magenta_node else amber if node == amber_node else neutral
            x, y = center
            draw.ellipse((x - 17, y - 17, x + 17, y + 17), fill=fill, outline=dark, width=4)
        draw.text((left + 127, 478), LABELS[panel_index], fill=dark, font=_font(25, True), anchor="mm")
    return image
