"""Independent image-only inverse arm for the dual-route panels."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 80) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        pending = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while pending:
            cy, cx = pending.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    pending.append((ny, nx))
        if len(points) >= minimum:
            ys = np.fromiter((point[0] for point in points), dtype=float)
            xs = np.fromiter((point[1] for point in points), dtype=float)
            output.append((float(xs.mean()), float(ys.mean()), len(points)))
    return output


def _near_color(array: np.ndarray, rgb: tuple[int, int, int], radius: float = 28.0) -> np.ndarray:
    # int32 prevents squared RGB distances from overflowing.
    delta = array.astype(np.int32) - np.array(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _route_score(edges: set[tuple[int, int]], start: int, goal: int, count: int) -> int:
    adjacency = {node: set() for node in range(count)}
    for left, right in edges:
        adjacency[left].add(right)
        adjacency[right].add(left)
    paths = []

    def visit(node: int, path: tuple[int, ...]) -> None:
        if node == goal:
            paths.append(path)
            return
        for neighbor in sorted(adjacency[node]):
            if neighbor not in path:
                visit(neighbor, path + (neighbor,))

    visit(start, (start,))
    for first, second in itertools.combinations(paths, 2):
        edge_a = {tuple(sorted(pair)) for pair in zip(first, first[1:])}
        edge_b = {tuple(sorted(pair)) for pair in zip(second, second[1:])}
        if not (set(first[1:-1]) & set(second[1:-1])) and not (edge_a & edge_b):
            return 2
    return 1 if paths else 0


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    kinds = (
        ("neutral", (94, 145, 160)),
        ("magenta", (216, 27, 96)),
        ("amber", (244, 163, 0)),
    )
    nodes = []
    for kind, color in kinds:
        for x, y, area in _components(_near_color(array, color)):
            nodes.append({"x": x, "y": y, "kind": kind, "area": area})
    if len(nodes) != 36:
        return "abstain"

    nodes.sort(key=lambda item: item["x"])
    panels = [nodes[index : index + 9] for index in range(0, 36, 9)]
    dark = _near_color(array, (38, 50, 56), radius=32.0)
    scores = []
    for panel in panels:
        if sum(item["kind"] == "magenta" for item in panel) != 1 or sum(item["kind"] == "amber" for item in panel) != 1:
            return "abstain"
        ordered = sorted(panel, key=lambda item: (item["y"], item["x"]))
        index_of = {id(item): index for index, item in enumerate(ordered)}
        rows = [sorted(ordered[offset : offset + 3], key=lambda item: item["x"]) for offset in (0, 3, 6)]
        columns = [[rows[row][column] for row in range(3)] for column in range(3)]
        adjacent = []
        for row in rows:
            adjacent.extend(zip(row, row[1:]))
        for column in columns:
            adjacent.extend(zip(column, column[1:]))
        edges: set[tuple[int, int]] = set()
        for first, second in adjacent:
            mx = int(round((first["x"] + second["x"]) / 2))
            my = int(round((first["y"] + second["y"]) / 2))
            patch = dark[max(0, my - 4) : my + 5, max(0, mx - 4) : mx + 5]
            if patch.size and float(patch.mean()) > 0.35:
                edges.add(tuple(sorted((index_of[id(first)], index_of[id(second)]))))
        if len(edges) != 9:
            return "abstain"
        start = next(index for index, item in enumerate(ordered) if item["kind"] == "magenta")
        goal = next(index for index, item in enumerate(ordered) if item["kind"] == "amber")
        scores.append(_route_score(edges, start, goal, 9))
    winners = [index for index, score in enumerate(scores) if score == 2]
    return ("A", "B", "C", "D")[winners[0]] if len(winners) == 1 else "abstain"
