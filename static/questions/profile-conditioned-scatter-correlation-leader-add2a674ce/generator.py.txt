"""Deterministic scatterplot renderer and analytic decision arm."""

from __future__ import annotations

import math
import random
from itertools import combinations

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 460
LEFT, TOP, RIGHT, BOTTOM = 72, 64, 612, 402
RADIUS = 6
COLORS = {"blue": (31, 119, 180), "orange": (230, 105, 20)}
ORDER = ("blue", "orange")
QUARANTINE_GAP = 0.10


def _corr(points: list[list[float]] | list[tuple[float, float]]) -> float:
    n = len(points)
    xs = [float(p[0]) for p in points]
    ys = [float(p[1]) for p in points]
    mx, my = sum(xs) / n, sum(ys) / n
    xx = sum((x - mx) ** 2 for x in xs)
    yy = sum((y - my) ** 2 for y in ys)
    if xx <= 0 or yy <= 0:
        raise ValueError("correlation requires nonzero horizontal and vertical variance")
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / math.sqrt(xx * yy)


def _pixel_points(points: list[list[float]]) -> list[tuple[float, float]]:
    span_x = RIGHT - LEFT
    span_y = BOTTOM - TOP
    return [
        (round(LEFT + float(x) * span_x), round(BOTTOM - float(y) * span_y))
        for x, y in points
    ]


def _normalize(values: list[float], low: float = 0.10, high: float = 0.90) -> list[float]:
    lo, hi = min(values), max(values)
    if hi - lo < 1e-8:
        raise ValueError("degenerate values")
    return [low + (v - lo) * (high - low) / (hi - lo) for v in values]


def _cloud(rng: random.Random, noise: float) -> list[list[float]]:
    xs = [i / 8 + rng.uniform(-0.025, 0.025) for i in range(9)]
    rng.shuffle(xs)
    sign = rng.choice((-1.0, 1.0))
    slope = rng.uniform(0.7, 1.6) * sign
    ys = [slope * (x - 0.5) + rng.gauss(0.0, noise) for x in xs]
    nx, ny = _normalize(xs), _normalize(ys)
    return [[round(x, 6), round(y, 6)] for x, y in zip(nx, ny)]


def _clear(series: dict[str, list[list[float]]]) -> bool:
    all_points: list[tuple[float, float]] = []
    for name in ORDER:
        points = _pixel_points(series[name])
        if any(math.dist(a, b) < 2 * RADIUS + 5 for a, b in combinations(points, 2)):
            return False
        all_points.extend(points)
    return not any(math.dist(a, b) < 2 * RADIUS + 4 for a, b in combinations(all_points, 2))


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes; every sixth seed is a quarantined near-boundary probe."""
    rng = random.Random(int(seed))
    desired = ORDER[int(seed) & 1]
    near = int(seed) % 6 == 0
    target = (0.035, 0.085) if near else (0.20, 0.46)
    for _ in range(12000):
        strong = _cloud(rng, rng.uniform(0.06, 0.19))
        weak = _cloud(rng, rng.uniform(0.28, 0.85))
        values = [abs(_corr(_pixel_points(p))) for p in (strong, weak)]
        gap = values[0] - values[1]
        if not (target[0] <= gap <= target[1]):
            continue
        other = ORDER[1 - ORDER.index(desired)]
        series = {desired: strong, other: weak}
        if _clear(series):
            return {"series": series}
    raise RuntimeError(f"could not sample a robust scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    series = scene["series"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.rectangle((LEFT, TOP, RIGHT, BOTTOM), fill=(255, 255, 255), outline=(48, 53, 58), width=2)
    for i in range(1, 5):
        x = round(LEFT + i * (RIGHT - LEFT) / 5)
        y = round(TOP + i * (BOTTOM - TOP) / 5)
        draw.line((x, TOP + 1, x, BOTTOM - 1), fill=(225, 228, 230), width=1)
        draw.line((LEFT + 1, y, RIGHT - 1, y), fill=(225, 228, 230), width=1)
    draw.text((LEFT, 23), "TWO-SERIES SCATTERPLOT", fill=(35, 39, 42), font=font)
    draw.text((286, 427), "horizontal coordinate", fill=(55, 60, 64), font=font)
    draw.text((10, 214), "vertical", fill=(55, 60, 64), font=font)
    for name in ORDER:
        color = COLORS[name]
        for x, y in _pixel_points(series[name]):
            draw.ellipse((x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    series = scene["series"]
    scores = {name: abs(_corr(_pixel_points(series[name]))) for name in ORDER}
    return max(ORDER, key=lambda name: scores[name])


def margin(scene: dict) -> float:
    series = scene["series"]
    scores = sorted((abs(_corr(_pixel_points(series[name]))) for name in ORDER), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    series = scene["series"]
    twin = {"series": {name: list(reversed(series[name])) for name in ORDER}}
    return [("reverse_point_storage_order", twin)]
