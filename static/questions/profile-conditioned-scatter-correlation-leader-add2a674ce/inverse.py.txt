"""Independent inverse arm: only the raster enters decision_from_image."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

PALETTE = {"blue": (31, 119, 180), "orange": (230, 105, 20)}
ORDER = ("blue", "orange")


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(group) >= 40:
            groups.append(group)
    return groups


def _centers(image: Image.Image, rgb: tuple[int, int, int]) -> list[tuple[float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    target = np.asarray(rgb, dtype=np.int16)
    mask = np.max(np.abs(array - target), axis=2) <= 2
    groups = _components(mask)
    if len(groups) != 9:
        raise ValueError(f"expected nine isolated markers, found {len(groups)}")
    centers = []
    for group in groups:
        ys = [p[0] for p in group]
        xs = [p[1] for p in group]
        centers.append((sum(xs) / len(xs), sum(ys) / len(ys)))
    return centers


def _abs_corr(points: list[tuple[float, float]]) -> float:
    n = len(points)
    mx = sum(x for x, _ in points) / n
    my = sum(y for _, y in points) / n
    xx = sum((x - mx) ** 2 for x, _ in points)
    yy = sum((y - my) ** 2 for _, y in points)
    if xx <= 0 or yy <= 0:
        raise ValueError("degenerate marker coordinates")
    cov = sum((x - mx) * (y - my) for x, y in points)
    return abs(cov / math.sqrt(xx * yy))


def decision_from_image(image: Image.Image) -> str:
    scores = {name: _abs_corr(_centers(image, PALETTE[name])) for name in ORDER}
    if abs(scores["blue"] - scores["orange"]) < 0.01:
        raise ValueError("pixel correlation comparison is too close to resolve")
    return max(ORDER, key=lambda name: scores[name])
