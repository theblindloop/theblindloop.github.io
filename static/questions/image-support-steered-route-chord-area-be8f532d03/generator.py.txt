"""Deterministic local-route world with an endpoint-chord signed-area decision."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 768
PALETTES = [
    ((19, 151, 180), (250, 248, 242)),
    ((28, 160, 145), (248, 250, 247)),
    ((44, 139, 179), (251, 249, 245)),
]


def _signed_mean(points: list[list[float]]) -> float:
    """Mean (above-positive) vertical displacement from the endpoint chord."""
    x0, y0 = points[0]
    x1, y1 = points[-1]
    area = 0.0
    for (xa, ya), (xb, yb) in zip(points, points[1:]):
        ca = y0 + (y1 - y0) * (xa - x0) / (x1 - x0)
        cb = y0 + (y1 - y0) * (xb - x0) / (x1 - x0)
        area += 0.5 * ((ca - ya) + (cb - yb)) * (xb - xa)
    return area / (x1 - x0)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    route_w = rng.choice([140, 146, 152])
    x0 = rng.randint(120, CANVAS - route_w - 120)
    # Tile-center placement keeps the small trace's ablation footprint thin,
    # while the row itself and the horizontal position remain randomized.
    base_y = rng.choice([144, 240, 336, 432, 528, 624])
    end_delta = rng.randint(-6, 6)
    sign = 1 if int(seed) % 2 == 0 else -1
    target = sign * rng.uniform(6.2, 7.2)

    # A large zero-mean oscillation blocks single-peak and all-on-one-side cues.
    raw = [
        rng.uniform(8, 13),
        -rng.uniform(8, 13),
        rng.uniform(8, 13),
        -rng.uniform(8, 13),
        rng.uniform(8, 13),
    ]
    rng.shuffle(raw)
    centered = [value - sum(raw) / len(raw) for value in raw]
    shift = target * 6.0 / 5.0
    above_offsets = [0.0] + [value + shift for value in centered] + [0.0]
    xs = [x0 + route_w * index / 6.0 for index in range(7)]
    points = []
    for index, (x, offset) in enumerate(zip(xs, above_offsets)):
        chord_y = base_y + end_delta * index / 6.0
        points.append([round(x, 3), round(chord_y - offset, 3)])
    route_color, background = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": list(background),
        "route_color": list(route_color),
        "stroke": rng.choice([7, 8, 9]),
        "points": points,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    points = [tuple(point) for point in scene["points"]]
    color = tuple(scene["route_color"])
    stroke = int(scene["stroke"])
    draw.line(points, fill=color, width=stroke, joint="curve")
    radius = stroke / 2.0
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return "above" if _signed_mean(scene["points"]) > 0 else "below"


def margin(scene: dict) -> float:
    return round(abs(_signed_mean(scene["points"])), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
