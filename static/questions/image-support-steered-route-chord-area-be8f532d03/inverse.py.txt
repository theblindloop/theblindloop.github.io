"""Independent inverse arm: infer signed route/chord area from RGB pixels only."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _route_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    # All renderer palettes are saturated cyan/blue; neutral paper never passes.
    return (green > red + 45) & (blue > red + 55) & (green > 105) & (blue > 115)


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        component = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    if len(best) < 450:
        raise ValueError("route ink is absent or too incomplete")
    output = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*best)
    output[np.asarray(ys), np.asarray(xs)] = True
    if len(best) < 0.93 * int(mask.sum()):
        raise ValueError("route was split into material fragments")
    return output


def _signed_pixel_mean(image: Image.Image) -> float:
    mask = _largest_component(_route_mask(image))
    ys, xs = np.nonzero(mask)
    xmin, xmax = int(xs.min()), int(xs.max())
    if xmax - xmin < 105:
        raise ValueError("route horizontal span is incomplete")
    center = {}
    for x in range(xmin, xmax + 1):
        column = np.flatnonzero(mask[:, x])
        if len(column):
            center[x] = float(np.median(column))
    if len(center) < 0.94 * (xmax - xmin + 1):
        raise ValueError("route centerline contains a material gap")
    edge = 3
    left_values = [center[x] for x in range(xmin, min(xmax + 1, xmin + edge + 1)) if x in center]
    right_values = [center[x] for x in range(max(xmin, xmax - edge), xmax + 1) if x in center]
    if not left_values or not right_values:
        raise ValueError("route endpoints cannot be recovered")
    y0 = float(np.median(left_values))
    y1 = float(np.median(right_values))
    values = []
    for x in range(xmin + 2, xmax - 1):
        if x in center:
            chord = y0 + (y1 - y0) * (x - xmin) / (xmax - xmin)
            values.append(chord - center[x])
    if len(values) < 100:
        raise ValueError("too little intact route interior")
    return float(np.mean(values))


def decision_from_image(image: Image.Image) -> str:
    signed = _signed_pixel_mean(image)
    if abs(signed) <= 3.0:
        raise ValueError("pixel measurement is too close to the signed-area boundary")
    return "above" if signed > 0 else "below"
