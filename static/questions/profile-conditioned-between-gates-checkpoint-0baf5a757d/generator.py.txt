"""Deterministic latent scenes and raster renderer for an ordered-route task."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = [250, 249, 246]
INK = (28, 32, 38)
PALETTE = {
    "red": (220, 55, 55),
    "blue": (52, 103, 210),
    "orange": (235, 135, 35),
    "green": (35, 170, 90),
    "purple": (145, 70, 185),
    "yellow": (230, 190, 35),
    "cyan": (30, 170, 185),
}
CHECKPOINTS = ("red", "blue", "orange")
QUARANTINE_CLEARANCE = 55.0


def _lengths(points: list[list[float]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for a, b in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(a, b))
    return cumulative, cumulative[-1]


def _at_distance(points: list[list[float]], distance: float) -> tuple[float, float]:
    cumulative, total = _lengths(points)
    distance = min(max(float(distance), 0.0), total)
    for i in range(len(points) - 1):
        if distance <= cumulative[i + 1] + 1e-9:
            span = cumulative[i + 1] - cumulative[i]
            t = 0.0 if span == 0 else (distance - cumulative[i]) / span
            return (
                points[i][0] + t * (points[i + 1][0] - points[i][0]),
                points[i][1] + t * (points[i + 1][1] - points[i][1]),
            )
    return tuple(points[-1])


def _transform(point: list[float], code: int) -> list[float]:
    """Apply one of eight exact square-canvas dihedral transforms."""
    x, y = point
    if code >= 4:
        x = WIDTH - x
        code -= 4
    for _ in range(code):
        x, y = WIDTH - y, x
    return [round(x, 3), round(y, 3)]


def _base_route(rng: random.Random) -> list[list[float]]:
    left = rng.randint(48, 72)
    right = rng.randint(440, 464)
    top = rng.randint(48, 64)
    gaps = [rng.randint(88, 98) for _ in range(4)]
    ys = [top]
    for gap in gaps:
        ys.append(ys[-1] + gap)
    if ys[-1] > 464:
        shift = ys[-1] - 464
        ys = [value - shift for value in ys]
    points: list[list[float]] = []
    for row, y in enumerate(ys):
        a, b = (left, right) if row % 2 == 0 else (right, left)
        if row == 0:
            points.append([a, y])
        points.append([b, y])
        if row < len(ys) - 1:
            points.append([b, ys[row + 1]])
    return points


def _marker(kind: str, color: str, at: float) -> dict:
    return {"kind": kind, "color": color, "at": round(float(at), 3)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    points = [_transform(point, int(seed) % 8) for point in _base_route(rng)]
    _, total = _lengths(points)

    gate_1 = total * rng.uniform(0.30, 0.34)
    gate_2 = total * rng.uniform(0.66, 0.70)
    before = total * rng.uniform(0.11, 0.20)
    after = total * rng.uniform(0.80, 0.89)

    near_boundary = int(seed) % 11 == 0
    if near_boundary:
        clearance = rng.uniform(42.0, 51.0)
        target = gate_1 + clearance if (int(seed) // 11) % 2 == 0 else gate_2 - clearance
    else:
        target = rng.uniform(gate_1 + 0.30 * (gate_2 - gate_1), gate_1 + 0.70 * (gate_2 - gate_1))

    answer_index = int(seed) % len(CHECKPOINTS)
    target_color = CHECKPOINTS[answer_index]
    other = [color for color in CHECKPOINTS if color != target_color]
    if (int(seed) // 3) % 2:
        other.reverse()

    items = [
        _marker("start", "green", 0.0),
        _marker("checkpoint", other[0], before),
        _marker("gate", "yellow", gate_1),
        _marker("checkpoint", target_color, target),
        _marker("gate", "cyan", gate_2),
        _marker("checkpoint", other[1], after),
        _marker("finish", "purple", total),
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "points": points,
        "stroke": 10,
        "items": items,
    }


def _draw_ring(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: int, width: int) -> None:
    x, y = center
    box = [x - radius, y - radius, x + radius, y + radius]
    draw.ellipse(box, outline=color, width=width)


def _draw_diamond(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = center
    radius = 17 * 3
    points = [(x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)]
    draw.line(points + [points[0]], fill=color, width=8 * 3, joint="curve")


def render(scene: dict) -> Image.Image:
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    points = [tuple(value * scale for value in point) for point in scene["points"]]
    stroke = int(scene["stroke"])
    items = scene["items"]

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    draw.line(points, fill=INK, width=stroke * scale, joint="curve")

    route_points = scene["points"]
    for item in items:
        x, y = _at_distance(route_points, item["at"])
        center = (x * scale, y * scale)
        color = PALETTE[item["color"]]
        if item["kind"] == "gate":
            _draw_diamond(draw, center, color)
        elif item["kind"] in {"start", "finish"}:
            _draw_ring(draw, center, color, 19 * scale, 8 * scale)
        else:
            _draw_ring(draw, center, color, 14 * scale, 7 * scale)

    # Preserve an explicit continuous centerline through every outlined marker.
    draw.line(points, fill=INK, width=3 * scale, joint="curve")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _ordered_items(scene: dict) -> list[dict]:
    return sorted(scene["items"], key=lambda item: float(item["at"]))


def analytic_gold(scene: dict) -> str:
    ordered = _ordered_items(scene)
    yellow = next(float(item["at"]) for item in ordered if item["color"] == "yellow")
    cyan = next(float(item["at"]) for item in ordered if item["color"] == "cyan")
    inside = [
        item["color"]
        for item in ordered
        if item["kind"] == "checkpoint" and yellow < float(item["at"]) < cyan
    ]
    if len(inside) != 1:
        raise ValueError(f"expected exactly one checkpoint between gates, got {inside}")
    return inside[0]


def _scores(scene: dict) -> list[tuple[float, str]]:
    ordered = _ordered_items(scene)
    yellow = next(float(item["at"]) for item in ordered if item["color"] == "yellow")
    cyan = next(float(item["at"]) for item in ordered if item["color"] == "cyan")
    return sorted(
        (
            min(float(item["at"]) - yellow, cyan - float(item["at"])),
            item["color"],
        )
        for item in ordered
        if item["kind"] == "checkpoint"
    )


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return float(scores[-1][0] - scores[-2][0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return bool(scores[-1][0] <= QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
