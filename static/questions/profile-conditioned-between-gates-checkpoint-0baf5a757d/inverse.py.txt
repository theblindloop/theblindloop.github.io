"""Image-only inverse arm: trace black pixels from START and order marker centroids."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 55, 55),
    "blue": (52, 103, 210),
    "orange": (235, 135, 35),
    "green": (35, 170, 90),
    "yellow": (230, 190, 35),
    "cyan": (30, 170, 185),
}
CHECKPOINTS = ("red", "blue", "orange")


def _centroid(array: np.ndarray, color: tuple[int, int, int]) -> tuple[float, float]:
    target = np.asarray(color, dtype=np.int32)
    delta = array.astype(np.int32) - target
    mask = np.sum(delta * delta, axis=2) < 45 * 45
    ys, xs = np.nonzero(mask)
    if len(xs) < 25:
        raise ValueError("required colored marker is absent or erased")
    return float(xs.mean()), float(ys.mean())


def _route_mask(array: np.ndarray) -> np.ndarray:
    return np.max(array, axis=2) < 85


def _nearest(mask: np.ndarray, point: tuple[float, float]) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise ValueError("black route is absent or too small")
    squared = (xs - point[0]) ** 2 + (ys - point[1]) ** 2
    index = int(np.argmin(squared))
    if float(squared[index]) > 15.0 ** 2:
        raise ValueError("marker does not touch the route centerline")
    return int(ys[index]), int(xs[index])


def _flood_distance(mask: np.ndarray, source: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    distance[source] = 0
    queue: deque[tuple[int, int]] = deque([source])
    while queue:
        y, x = queue.popleft()
        nd = int(distance[y, x]) + 1
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = nd
                queue.append((ny, nx))
    return distance


def _marker_distance(distance: np.ndarray, mask: np.ndarray, point: tuple[float, float]) -> int:
    y, x = _nearest(mask, point)
    value = int(distance[y, x])
    if value < 0:
        raise ValueError("route is broken between START and a marker")
    return value


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    centers = {name: _centroid(array, PALETTE[name]) for name in PALETTE}
    route = _route_mask(array)
    source = _nearest(route, centers["green"])
    distances = _flood_distance(route, source)
    positions = {
        name: _marker_distance(distances, route, centers[name])
        for name in ("yellow", "cyan", *CHECKPOINTS)
    }
    yellow, cyan = positions["yellow"], positions["cyan"]
    if cyan - yellow < 40:
        raise ValueError("gate order is absent or ambiguous")
    inside = [name for name in CHECKPOINTS if yellow + 3 < positions[name] < cyan - 3]
    if len(inside) != 1:
        raise ValueError(f"expected one checkpoint between gates, recovered {inside}")
    return inside[0]
