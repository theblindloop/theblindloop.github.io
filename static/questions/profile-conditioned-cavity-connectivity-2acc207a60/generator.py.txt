"""Deterministic colored-band renderer for a negative-space connectivity task."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "cavity_connectivity-0.1.0"
CANVAS = (256, 192)
BACKGROUND = (255, 255, 255)
PALETTE = (
    (26, 77, 122),
    (112, 48, 96),
    (36, 112, 78),
    (153, 77, 28),
    (67, 58, 128),
)


def _point(cx: float, cy: float, rx: float, ry: float, angle: float) -> tuple[float, float]:
    return (cx + rx * math.cos(angle), cy + ry * math.sin(angle))


def _ring_points(scene: dict, inner: bool) -> list[tuple[float, float]]:
    cx = float(scene["cx"])
    cy = float(scene["cy"])
    rx = float(scene["outer_rx"])
    ry = float(scene["outer_ry"])
    band = float(scene["band_width"])
    rotation = float(scene["rotation"])
    sides = int(scene["sides"])
    jitter = scene["radial_jitter"]
    points = []
    scale = 0.58 if inner else 1.0
    inset = band if inner else 0.0
    for index in range(sides):
        angle = rotation + 2.0 * math.pi * index / sides
        perturbation = float(jitter[index]) * scale
        points.append(
            _point(
                cx,
                cy,
                max(18.0, rx - inset + perturbation),
                max(16.0, ry - inset + perturbation),
                angle,
            )
        )
    return points


def _draw_decorations(draw: ImageDraw.ImageDraw, scene: dict, color: tuple[int, int, int]) -> None:
    for decoration in scene["decorations"]:
        x = float(decoration["x"])
        y = float(decoration["y"])
        radius = float(decoration["radius"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def sample_scene(seed: int) -> dict:
    """Sample one balanced, rendered-label scene from an integer seed."""

    # Alternating the label by scene seed guarantees both classes in each public
    # stress batch; all visible nuisances come from a separate RNG stream.
    label = "sealed" if seed % 2 == 0 else "open"
    rng = random.Random((seed + 7919) * 1009)
    cx = rng.randint(82, 174)
    cy = rng.randint(66, 126)
    outer_rx = rng.randint(48, 66)
    outer_ry = rng.randint(38, 54)
    band_width = rng.randint(8, 12)
    if label == "sealed":
        gap_width = 0
    elif seed % 7 == 0:
        gap_width = rng.choice((1, 2))
    else:
        gap_width = rng.randint(7, 16)
    gap_angle = rng.uniform(-math.pi, math.pi)
    rotation = rng.uniform(-math.pi / 10.0, math.pi / 10.0)
    sides = rng.choice((8, 9, 10, 12))
    radial_jitter = [round(rng.uniform(-3.5, 3.5), 3) for _ in range(sides)]
    color = PALETTE[rng.randrange(len(PALETTE))]

    decorations = []
    for _ in range(rng.randint(1, 4)):
        for _attempt in range(80):
            x = rng.uniform(15.0, CANVAS[0] - 15.0)
            y = rng.uniform(14.0, CANVAS[1] - 14.0)
            radius = rng.uniform(2.5, 5.5)
            normalized = ((x - cx) / (outer_rx + 14.0)) ** 2 + (
                (y - cy) / (outer_ry + 14.0)
            ) ** 2
            if normalized > 1.0:
                decorations.append(
                    {"x": round(x, 3), "y": round(y, 3), "radius": round(radius, 3)}
                )
                break

    return {
        "label": label,
        "cx": cx,
        "cy": cy,
        "outer_rx": outer_rx,
        "outer_ry": outer_ry,
        "band_width": band_width,
        "gap_width": gap_width,
        "gap_angle": round(gap_angle, 6),
        "rotation": round(rotation, 6),
        "sides": sides,
        "radial_jitter": radial_jitter,
        "fg": list(color),
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    foreground = tuple(int(value) for value in scene["fg"])
    draw.polygon(_ring_points(scene, inner=False), fill=foreground)
    draw.polygon(_ring_points(scene, inner=True), fill=BACKGROUND)

    # Consume the declared label in the renderer as well as in analytic gold;
    # this makes the answer-bearing input observable to the rendered-field audit.
    label = scene["label"]
    gap_width = 0 if label == "sealed" else int(scene["gap_width"])
    if gap_width > 0:
        cx = float(scene["cx"])
        cy = float(scene["cy"])
        radius = max(float(scene["outer_rx"]), float(scene["outer_ry"])) + 34.0
        angle = float(scene["gap_angle"])
        half_angle = gap_width / (2.0 * radius) + 0.008
        left = _point(cx, cy, radius, radius, angle - half_angle)
        right = _point(cx, cy, radius, radius, angle + half_angle)
        draw.polygon([(cx, cy), left, right], fill=BACKGROUND)

    _draw_decorations(draw, scene, foreground)
    return image


def analytic_gold(scene: dict) -> str:
    # Deliberately only the rendered topological label is answer-bearing.
    return "sealed" if scene["label"] == "sealed" else "open"


def margin(scene: dict) -> float:
    # Positive is a closed-band clearance; negative is the visible opening width.
    if scene["label"] == "sealed":
        return float(scene["band_width"])
    return -float(scene["gap_width"])


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # This world declares no latent aliases: pose, contour, palette, and label
    # are all rendered, and the evaluator treats the mode as not_applicable.
    return []
