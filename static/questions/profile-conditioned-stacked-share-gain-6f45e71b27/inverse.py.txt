"""Independent inverse arm using only final RGB pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = ("red", "blue", "green", "orange")
RGB = {
    "red": (214, 55, 55),
    "blue": (47, 105, 190),
    "green": (43, 151, 82),
    "orange": (232, 137, 38),
}


def _column_runs(active: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(active)
    if len(indices) == 0:
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(indices) - 1]
    return [(int(indices[a]), int(indices[b])) for a, b in zip(starts, ends)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = {
        color: np.all(array == np.asarray(RGB[color], dtype=np.uint8), axis=2)
        for color in COLORS
    }
    colored = np.logical_or.reduce(tuple(masks.values()))
    runs = [run for run in _column_runs(colored.any(axis=0)) if run[1] - run[0] + 1 >= 40]
    if len(runs) != 2:
        raise ValueError(f"expected two colored bar runs, found {len(runs)}")
    runs.sort()

    shares: list[list[float]] = []
    for x0, x1 in runs:
        heights: list[int] = []
        for color in COLORS:
            local = masks[color][:, x0 : x1 + 1]
            row_counts = np.count_nonzero(local, axis=1)
            rows = int(np.count_nonzero(row_counts >= max(20, int(0.75 * (x1 - x0 + 1)))))
            if rows < 8:
                raise ValueError(f"missing or too-small {color} segment")
            heights.append(rows)
        total = sum(heights)
        if total < 100:
            raise ValueError("colored bar is too small to measure")
        shares.append([height / total for height in heights])

    gains = [shares[1][index] - shares[0][index] for index in range(4)]
    order = sorted(range(4), key=lambda index: (-gains[index], index))
    if gains[order[0]] - gains[order[1]] < 0.075:
        raise ValueError("ambiguous share-gain winner")
    return COLORS[order[0]]
