"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
COLORS = ("red", "blue", "green", "orange")
RGB = {
    "red": (214, 55, 55),
    "blue": (47, 105, 190),
    "green": (43, 151, 82),
    "orange": (232, 137, 38),
}
BACKGROUND = (248, 249, 251)
INK = (35, 40, 48)
GRID = (215, 219, 225)
DIVIDER = (255, 255, 255)
UNIT_PERCENT = 5
QUARANTINE_PP = 10.0


def _composition(rng: random.Random) -> list[int]:
    """Return four positive five-percent units summing to twenty."""
    values = [2, 2, 2, 2]
    for _ in range(12):
        values[rng.randrange(4)] += 1
    return values


def _rank_desc(values: list[int], index: int) -> int:
    order = sorted(range(4), key=lambda item: (-values[item], item))
    return order.index(index)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    winner = int(seed) % 4
    # Alternate whether the answer is the largest or second-largest segment in B,
    # defeating a terminal-size shortcut while keeping a generous gain margin.
    desired_right_rank = (int(seed) // 4) % 2

    for _ in range(50000):
        left = _composition(rng)
        right = _composition(rng)
        if len(set(right)) != 4:
            continue
        gains = [right[i] - left[i] for i in range(4)]
        ranked = sorted(range(4), key=lambda item: (-gains[item], item))
        if ranked[0] != winner:
            continue
        if gains[ranked[0]] - gains[ranked[1]] < 3:
            continue
        if _rank_desc(right, winner) != desired_right_rank:
            continue
        break
    else:
        raise RuntimeError("unable to sample a separated composition")

    stack_left = list(range(4))
    stack_right = list(range(4))
    rng.shuffle(stack_left)
    rng.shuffle(stack_right)
    if stack_right == stack_left:
        stack_right = stack_right[1:] + stack_right[:1]

    tall_left = ((int(seed) // 2) % 2) == 0
    return {
        "shares_left": left,
        "shares_right": right,
        "stack_left": stack_left,
        "stack_right": stack_right,
        "height_left": 300 if tall_left else 200,
        "height_right": 200 if tall_left else 300,
        "shift_x": rng.randrange(-18, 19),
        "shift_y": rng.randrange(-8, 9),
    }


def _gains(scene: dict[str, Any]) -> list[int]:
    left = scene["shares_left"]
    right = scene["shares_right"]
    return [int(right[i]) - int(left[i]) for i in range(4)]


def analytic_gold(scene: dict[str, Any]) -> str:
    gains = _gains(scene)
    return COLORS[max(range(4), key=lambda index: (gains[index], -index))]


def margin(scene: dict[str, Any]) -> float:
    ordered = sorted(_gains(scene), reverse=True)
    return float((ordered[0] - ordered[1]) * UNIT_PERCENT)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_PP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _draw_bar(
    draw: ImageDraw.ImageDraw,
    *,
    x: int,
    baseline: int,
    width: int,
    colored_height: int,
    shares: list[int],
    order: list[int],
) -> None:
    scale = colored_height // 20
    gap = 4
    cursor = baseline
    for position, category in enumerate(order):
        segment_height = int(shares[category]) * scale
        top = cursor - segment_height + 1
        draw.rectangle(
            (x, top, x + width - 1, cursor),
            fill=RGB[COLORS[category]],
        )
        cursor = top - gap
    outline_top = cursor + gap - 1
    draw.rectangle(
        (x - 4, outline_top - 4, x + width + 3, baseline + 4),
        outline=INK,
        width=3,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    left = [int(value) for value in scene["shares_left"]]
    right = [int(value) for value in scene["shares_right"]]
    order_left = [int(value) for value in scene["stack_left"]]
    order_right = [int(value) for value in scene["stack_right"]]
    height_left = int(scene["height_left"])
    height_right = int(scene["height_right"])
    shift_x = int(scene["shift_x"])
    shift_y = int(scene["shift_y"])

    baseline = 404 + shift_y
    x_left = 152 + shift_x
    x_right = 370 + shift_x
    bar_width = 116

    draw.text((210, 20), "COMPOSITION BY CONDITION", fill=INK, font=font)
    for y in (104 + shift_y, 204 + shift_y, 304 + shift_y, 404 + shift_y):
        draw.line((78, y, 558, y), fill=GRID, width=1)

    _draw_bar(
        draw,
        x=x_left,
        baseline=baseline,
        width=bar_width,
        colored_height=height_left,
        shares=left,
        order=order_left,
    )
    _draw_bar(
        draw,
        x=x_right,
        baseline=baseline,
        width=bar_width,
        colored_height=height_right,
        shares=right,
        order=order_right,
    )

    draw.text((x_left + 53, baseline + 14), "A", fill=INK, font=font)
    draw.text((x_right + 53, baseline + 14), "B", fill=INK, font=font)

    legend_x = 134
    legend_y = 450
    for index, color in enumerate(COLORS):
        x = legend_x + index * 100
        draw.rectangle((x, legend_y, x + 11, legend_y + 11), fill=RGB[color])
        draw.text((x + 16, legend_y + 1), color, fill=INK, font=font)
    return image
