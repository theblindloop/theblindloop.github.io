"""Independent final-PNG inverse arm for three-line concurrence."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not expose three stable detached line components."""


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = array.max(axis=2) - array.min(axis=2)
    darkness = 255.0 - array.mean(axis=2)
    return (chroma >= 42) & (darkness >= 35)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        sy, sx = int(raw_y), int(raw_x)
        if seen[sy, sx]:
            continue
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[float, float]] = []
        while queue:
            y, x = queue.popleft()
            points.append((float(x) + 0.5, float(y) + 0.5))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 80:
            output.append(np.asarray(points, dtype=np.float64))
    return sorted(output, key=lambda points: (float(points[:, 0].mean()), float(points[:, 1].mean())))


def _fit_line(points: np.ndarray) -> tuple[np.ndarray, np.ndarray, float]:
    center = points.mean(axis=0)
    centered = points - center
    covariance = np.cov(centered.T)
    values, vectors = np.linalg.eigh(covariance)
    if not np.all(np.isfinite(values)) or values[1] <= 0.0 or values[0] <= 0.0:
        raise OracleAbstention("a bar fit is degenerate")
    elongation = float(values[1] / values[0])
    direction = vectors[:, 1]
    along = centered @ direction
    length = float(np.percentile(along, 99) - np.percentile(along, 1))
    if elongation < 12.0 or not 34.0 <= length <= 65.0:
        raise OracleAbstention("a chromatic component is not a resolvable short bar")
    return center, direction / np.linalg.norm(direction), length


def _cross(left: np.ndarray, right: np.ndarray) -> float:
    return float(left[0] * right[1] - left[1] * right[0])


def _intersection(left: tuple[np.ndarray, np.ndarray, float], right: tuple[np.ndarray, np.ndarray, float]) -> np.ndarray:
    p, d, _ = left
    q, e, _ = right
    denominator = _cross(d, e)
    if abs(denominator) < 0.42:
        raise OracleAbstention("two inferred axes are too nearly parallel")
    return p + (_cross(q - p, e) / denominator) * d


def _intersection_score(image: Image.Image) -> float:
    components = _components(_ink_mask(image))
    if len(components) != 3:
        raise OracleAbstention(f"expected three detached bars, found {len(components)}")
    lines = [_fit_line(component) for component in components]
    intersections = [
        _intersection(lines[left], lines[right])
        for left, right in ((0, 1), (0, 2), (1, 2))
    ]
    spread = max(float(np.linalg.norm(a - b)) for a in intersections for b in intersections)
    centers = np.asarray([line[0] for line in lines])
    constellation_center = centers.mean(axis=0)
    radius = float(np.median(np.linalg.norm(centers - constellation_center, axis=1)))
    if not 85.0 <= radius <= 140.0:
        raise OracleAbstention("the bars do not form a compact, well-spaced triad")
    return spread / radius


def decision_from_image(image: Image.Image) -> str:
    """Classify only the structure reconstructed from final RGB pixels."""

    score = _intersection_score(image)
    if score <= 0.16:
        return "one_point"
    if score >= 0.44:
        return "triangle"
    raise OracleAbstention(f"intersection spread {score:.3f} lies in the ambiguity band")
