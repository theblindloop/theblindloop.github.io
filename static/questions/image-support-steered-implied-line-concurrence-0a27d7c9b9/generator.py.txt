"""Deterministic renderer and analytic arm for three-line concurrence."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS_SIZE = 768
CELL = CANVAS_SIZE // 8
BASE_RADIUS = 120.0
TRIANGLE_TURN_DEG = 10.0
INK_PALETTE = (
    (27, 96, 174),
    (174, 51, 83),
    (22, 132, 91),
    (113, 69, 171),
    (191, 91, 25),
)
BACKGROUNDS = ((250, 249, 245), (247, 250, 252), (251, 247, 251))
DECORATION = (214, 216, 214)


def _anchor_center(phase: int, row: int, col: int) -> tuple[float, float]:
    """Center a three-cell triad while its bars land near nonadjacent cell centers."""

    x0 = col * CELL + CELL / 2.0
    y0 = row * CELL + CELL / 2.0
    if phase == 0:
        return x0 + CELL, y0 + 0.75 * CELL
    if phase == 1:
        return x0 + CELL, y0 + 1.25 * CELL
    if phase == 2:
        return x0 + 0.75 * CELL, y0 + CELL
    return x0 + 1.25 * CELL, y0 + CELL


def _canonical_offsets(phase: int) -> list[tuple[float, float]]:
    if phase == 0:
        return [(-96.0, -72.0), (96.0, -72.0), (0.0, 120.0)]
    if phase == 1:
        return [(-96.0, 72.0), (96.0, 72.0), (0.0, -120.0)]
    if phase == 2:
        return [(-72.0, -96.0), (-72.0, 96.0), (120.0, 0.0)]
    return [(72.0, -96.0), (72.0, 96.0), (-120.0, 0.0)]


def sample_scene(seed: int) -> dict:
    """Sample balanced modes; even seeds concur and odd seeds form a triangle."""

    rng = random.Random(int(seed))
    phase = rng.randrange(4)
    row = rng.randrange(1, 5)
    col = rng.randrange(1, 5)
    center_x, center_y = _anchor_center(phase, row, col)
    if int(seed) % 2 == 0:
        line_mode = "one_point"
    else:
        line_mode = rng.choice(("triangle_cw", "triangle_ccw"))
    order = [0, 1, 2]
    rng.shuffle(order)
    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "layout_phase": phase,
        "layout_scale": rng.choice((0.94, 1.0, 1.06)),
        "line_mode": line_mode,
        "bar_length": rng.choice((44.0, 48.0, 52.0)),
        "stroke_width": rng.choice((7, 8, 9)),
        "ink_rgb": list(rng.choice(INK_PALETTE)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "draw_order": order,
        "decoration_phase": rng.randrange(29),
    }


def _bar_centers(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["layout_scale"])
    return [
        (cx + scale * dx, cy + scale * dy)
        for dx, dy in _canonical_offsets(int(scene["layout_phase"]))
    ]


def _turn_radians(scene: dict) -> float:
    mode = str(scene["line_mode"])
    if mode == "one_point":
        return 0.0
    if mode == "triangle_cw":
        return math.radians(TRIANGLE_TURN_DEG)
    if mode == "triangle_ccw":
        return -math.radians(TRIANGLE_TURN_DEG)
    raise ValueError(f"unknown line_mode {mode!r}")


def _bar_geometry(scene: dict, index: int) -> tuple[np.ndarray, np.ndarray, tuple[float, float], tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    center = np.asarray(_bar_centers(scene)[index], dtype=np.float64)
    inward = math.atan2(cy - center[1], cx - center[0]) + _turn_radians(scene)
    direction = np.asarray([math.cos(inward), math.sin(inward)], dtype=np.float64)
    half = float(scene["bar_length"]) / 2.0
    left = center - half * direction
    right = center + half * direction
    return center, direction, (float(left[0]), float(left[1])), (float(right[0]), float(right[1]))


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    background = tuple(int(value) for value in scene["background_rgb"])
    image = Image.new("RGB", (size, size), background)
    draw = ImageDraw.Draw(image)
    phase = int(scene["decoration_phase"])
    # Low-chroma peripheral crosses are visible nuisance marks but are ignored
    # by the chroma-based inverse arm and never encode the answer.
    for index in range(9):
        x = 30 + ((index * 181 + phase * 37) % (size - 60))
        y = 34 + ((index * 233 + phase * 53) % (size - 68))
        if abs(x - float(scene["center_x"])) < 175 and abs(y - float(scene["center_y"])) < 175:
            x = 34 if index % 2 == 0 else size - 34
        draw.line((x - 6, y, x + 6, y), fill=DECORATION, width=2)
        draw.line((x, y - 6, x, y + 6), fill=DECORATION, width=2)
    ink = tuple(int(value) for value in scene["ink_rgb"])
    width = int(scene["stroke_width"])
    radius = width // 2
    for raw_index in scene["draw_order"]:
        _, _, first, second = _bar_geometry(scene, int(raw_index))
        draw.line((first, second), fill=ink, width=width)
        for x, y in (first, second):
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ink)
    return image


def _cross(left: np.ndarray, right: np.ndarray) -> float:
    return float(left[0] * right[1] - left[1] * right[0])


def _pair_intersection(p: np.ndarray, d: np.ndarray, q: np.ndarray, e: np.ndarray) -> np.ndarray:
    denominator = _cross(d, e)
    if abs(denominator) < 1e-9:
        raise ValueError("parallel analytic axes")
    return p + (_cross(q - p, e) / denominator) * d


def _intersection_score(scene: dict) -> float:
    lines = [(_bar_geometry(scene, index)[0], _bar_geometry(scene, index)[1]) for index in range(3)]
    points = [
        _pair_intersection(lines[left][0], lines[left][1], lines[right][0], lines[right][1])
        for left, right in ((0, 1), (0, 2), (1, 2))
    ]
    spread = max(float(np.linalg.norm(a - b)) for a in points for b in points)
    return spread / (BASE_RADIUS * float(scene["layout_scale"]))


def analytic_gold(scene: dict) -> str:
    mode = str(scene["line_mode"])
    if mode == "one_point":
        return "one_point"
    if mode in {"triangle_cw", "triangle_ccw"}:
        return "triangle"
    raise ValueError(f"unknown line_mode {mode!r}")


def margin(scene: dict) -> float:
    """Normalized runner-up clearance from the 0.30 concurrence boundary."""

    return abs(_intersection_score(scene) - 0.30)


def is_quarantined(scene: dict) -> bool:
    try:
        score = _intersection_score(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return str(scene.get("line_mode")) not in {"one_point", "triangle_cw", "triangle_ccw"} or 0.16 < score < 0.44


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["draw_order"] = list(reversed([int(value) for value in scene["draw_order"]]))
    return [("reverse_disjoint_bar_draw_order", twin)]
