"""Independent image-only inverse arm."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((222, 68, 76)),
    "green": np.array((40, 170, 105)),
    "blue": np.array((55, 112, 214)),
    "yellow": np.array((242, 190, 45)),
}


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while q:
            yy, xx = q.pop()
            points.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(points) > 1200:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            found.append((len(points), sum(xs) / len(xs), sum(ys) / len(ys)))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    tokens = []
    for name, rgb in PALETTE.items():
        delta = np.max(np.abs(array - rgb), axis=2)
        for area, cx, cy in _components(delta <= 8):
            if area > 1200:
                tokens.append((cy, cx, name, area))
    if len(tokens) < 16:
        raise ValueError("not enough colored tile components")
    ys = sorted(t[0] for t in tokens)
    gaps = [(ys[i + 1] - ys[i], i) for i in range(len(ys) - 1)]
    gap, cut = max(gaps)
    if gap < 100:
        raise ValueError("ambiguous row/card separation")
    split = (ys[cut] + ys[cut + 1]) / 2
    upper = sorted((t for t in tokens if t[0] < split), key=lambda t: t[1])
    lower = sorted((t for t in tokens if t[0] > split), key=lambda t: t[1])
    if not (4 <= len(upper) <= 6 and len(lower) == 12):
        raise ValueError("unexpected tile counts")
    word = [t[2] for t in upper]
    for token in (t[2] for t in lower):
        if word and word[-1] == token:
            word.pop()
        else:
            word.append(token)
    if not word:
        raise ValueError("empty final row")
    return word[-1]
