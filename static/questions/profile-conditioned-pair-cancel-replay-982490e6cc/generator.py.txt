"""Deterministic raster world for adjacent-pair cancellation."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "red": (222, 68, 76),
    "green": (40, 170, 105),
    "blue": (55, 112, 214),
    "yellow": (242, 190, 45),
}
NAMES = tuple(COLORS)


def _reduce(initial: list[str], cards: list[list[str]]) -> list[str]:
    word = list(initial)
    for card in cards:
        for token in card:
            if word and word[-1] == token:
                word.pop()
            else:
                word.append(token)
    return word


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = NAMES[seed % len(NAMES)]
    for _ in range(10000):
        initial = [rng.choice(NAMES) for _ in range(rng.randint(4, 6))]
        cards = [[rng.choice(NAMES), rng.choice(NAMES)] for _ in range(6)]
        final = _reduce(initial, cards)
        flat = [token for card in cards for token in card]
        cancellations = (len(initial) + len(flat) - len(final)) // 2
        if final and final[-1] == target and cancellations >= 3 and len(set(initial + flat)) >= 3:
            return {"initial": initial, "cards": cards}
    raise RuntimeError("unable to sample balanced scene")


def render(scene: dict) -> Image.Image:
    initial = list(scene["initial"])
    cards = list(scene["cards"])
    image = Image.new("RGB", (1200, 700), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    ink = (35, 42, 50)

    draw.rounded_rectangle((55, 55, 1145, 305), radius=24, fill=(255, 255, 255), outline=ink, width=5)
    draw.text((85, 78), "START ROW", fill=ink, font=font)
    draw.line((935, 180, 1080, 180), fill=ink, width=8)
    draw.polygon([(1080, 180), (1048, 158), (1048, 202)], fill=ink)
    tile = 92
    gap = 24
    x0 = 110
    for i, name in enumerate(initial):
        x = x0 + i * (tile + gap)
        draw.rounded_rectangle((x, 135, x + tile, 227), radius=14, fill=COLORS[name], outline=ink, width=5)

    draw.text((65, 350), "ADD CARDS  -  READ LEFT TO RIGHT", fill=ink, font=font)
    card_w, card_h, card_gap = 158, 220, 25
    cx0 = 62
    for i, card in enumerate(cards):
        x = cx0 + i * (card_w + card_gap)
        draw.rounded_rectangle((x, 385, x + card_w, 605), radius=18, fill=(255, 255, 255), outline=ink, width=5)
        draw.text((x + 65, 410), str(i + 1), fill=ink, font=font)
        for j, name in enumerate(card):
            tx = x + 14 + j * 70
            draw.rounded_rectangle((tx, 478, tx + 60, 538), radius=11, fill=COLORS[name], outline=ink, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    return _reduce(list(scene["initial"]), list(scene["cards"]))[-1]


def margin(scene: dict) -> float:
    # Minimum designed empty-space separation between colored components.
    _ = scene["initial"], scene["cards"]
    return 24.0


def is_quarantined(scene: dict) -> bool:
    initial, cards = list(scene["initial"]), list(scene["cards"])
    return not initial or len(cards) != 6 or any(len(card) != 2 for card in cards) or not _reduce(initial, cards)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
