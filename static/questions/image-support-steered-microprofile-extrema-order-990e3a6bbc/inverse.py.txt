"""Independent pixel-only inverse for Micro-profile Extrema Order.

Only the final RGB image is accepted. The inverse segments cool chromatic ink,
checks that it is one intact short trace, reconstructs its centerline column by
column, finds the unique high and low flat extrema, and compares their x order.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _trace_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    chroma = array.max(axis=2) - array.min(axis=2)
    return (
        (chroma >= 58)
        & (red <= 55)
        & (green >= 105)
        & (((green - red) >= 72) | ((blue - red) >= 88))
    )


def _component_size(mask: np.ndarray, start: tuple[int, int]) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    stack = [start]
    seen[start] = True
    count = 0
    while stack:
        y, x = stack.pop()
        count += 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                stack.append((ny, nx))
    return count


def _groups(values: np.ndarray) -> list[np.ndarray]:
    if not len(values):
        return []
    cuts = np.flatnonzero(np.diff(values) > 1) + 1
    return [group for group in np.split(values, cuts) if len(group)]


def _measure(image: Image.Image) -> tuple[float, float]:
    mask = _trace_mask(image)
    points = np.argwhere(mask)
    if not 470 <= len(points) <= 900:
        raise ValueError("teal mini-profile is missing or materially erased")
    first = tuple(int(value) for value in points[0])
    if _component_size(mask, first) != int(mask.sum()):
        raise ValueError("teal mini-profile is severed")
    ys, xs = np.nonzero(mask)
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not 109 <= x1 - x0 + 1 <= 117:
        raise ValueError("mini-profile horizontal extent is malformed")
    if not 30 <= y1 - y0 + 1 <= 38:
        raise ValueError("mini-profile vertical extent is malformed")

    columns = np.arange(x0, x1 + 1)
    centerline = np.asarray([np.median(np.flatnonzero(mask[:, x])) for x in columns])
    if not np.isfinite(centerline).all():
        raise ValueError("mini-profile has missing horizontal samples")
    high_level, low_level = float(centerline.min()), float(centerline.max())
    if low_level - high_level < 24:
        raise ValueError("visible extrema are insufficiently separated")
    high_groups = _groups(columns[centerline <= high_level + 0.5])
    low_groups = _groups(columns[centerline >= low_level - 0.5])
    if not high_groups or not low_groups:
        raise ValueError("extreme plateaus cannot be recovered")
    high = max(high_groups, key=len)
    low = max(low_groups, key=len)
    if len(high) < 9 or len(low) < 9:
        raise ValueError("an extreme plateau is too short")
    if sum(len(group) >= 5 for group in high_groups) != 1 or sum(len(group) >= 5 for group in low_groups) != 1:
        raise ValueError("extreme plateau is not unique")
    high_x, low_x = float(np.median(high)), float(np.median(low))
    if min(high_x, low_x) - x0 < 14 or x1 - max(high_x, low_x) < 14:
        raise ValueError("an extreme is not visibly interior")
    if abs(high_x - low_x) < 14:
        raise ValueError("pixel-measured extreme order is ambiguous")
    return high_x, low_x


def decision_from_image(image: Image.Image) -> str:
    high_x, low_x = _measure(image)
    return "peak" if high_x < low_x else "trough"
