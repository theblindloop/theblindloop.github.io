"""Deterministic renderer and analytic arm for Micro-profile Extrema Order."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 600
QUARANTINE_PX = 26.0
TRACE_PALETTE = (
    (10, 132, 150),
    (27, 118, 169),
    (14, 145, 126),
)
BACKGROUND_PALETTE = (
    (249, 248, 244),
    (247, 249, 248),
    (250, 247, 246),
)
INK = (46, 51, 55)
MUTED = (122, 126, 127)
FRAME = (205, 207, 203)
X_OFFSETS = (-56, -44, -32, -20, -8, 4, 16, 28, 40, 52, 56)
X_CENTERS = (200, 300, 400, 500, 600)
Y_CENTERS = (112, 187, 262, 337, 412, 487)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _ordered_points(scene: dict[str, Any]) -> list[tuple[int, int]]:
    points = [tuple(int(axis) for axis in point) for point in scene["points"]]
    if len(points) != len(X_OFFSETS) or len({point[0] for point in points}) != len(points):
        raise ValueError("profile must have eleven distinct horizontal samples")
    points.sort(key=lambda point: point[0])
    if any(b[0] <= a[0] for a, b in zip(points, points[1:], strict=False)):
        raise ValueError("profile samples must have strict horizontal order")
    return points


def _extrema_centers(scene: dict[str, Any]) -> tuple[float, float]:
    points = _ordered_points(scene)
    ys = [point[1] for point in points]
    high_y, low_y = min(ys), max(ys)
    high_xs = [x for x, y in points if y == high_y]
    low_xs = [x for x, y in points if y == low_y]
    if len(high_xs) != 2 or len(low_xs) != 2:
        raise ValueError("profile must expose one two-sample plateau at each extreme")
    if high_xs[1] - high_xs[0] != 12 or low_xs[1] - low_xs[0] != 12:
        raise ValueError("extreme plateaus must have the declared visible width")
    if high_y >= low_y - 20:
        raise ValueError("vertical extreme separation is too small")
    high_x = sum(high_xs) / 2.0
    low_x = sum(low_xs) / 2.0
    if high_x == low_x:
        raise ValueError("extreme order is tied")
    return high_x, low_x


def _base_scene(group: int) -> dict[str, Any]:
    rng = random.Random(0xE17E_2026 + group * 1_000_003)
    center_x = X_CENTERS[group % len(X_CENTERS)]
    center_y = Y_CENTERS[(group // len(X_CENTERS)) % len(Y_CENTERS)]
    values = [rng.randint(-5, 5) for _ in X_OFFSETS]
    near_boundary = group % 7 == 0
    if near_boundary:
        high_pair, low_pair = (4, 5), (6, 7)
    else:
        high_pair, low_pair = (2, 3), (7, 8)
    for index in high_pair:
        values[index] = -14
    for index in low_pair:
        values[index] = 14
    points = [[center_x + dx, center_y + dy] for dx, dy in zip(X_OFFSETS, values, strict=True)]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "points": points,
        "trace_rgb": list(TRACE_PALETTE[group % len(TRACE_PALETTE)]),
        "background_rgb": list(BACKGROUND_PALETTE[(group // 3) % len(BACKGROUND_PALETTE)]),
    }
    high_x, low_x = _extrema_centers(scene)
    expected = 24.0 if near_boundary else 60.0
    if high_x >= low_x or abs(low_x - high_x) != expected:
        raise AssertionError("base profile extrema changed")
    return scene


def sample_scene(seed: int) -> dict[str, Any]:
    """Return horizontally mirrored answer pairs with matched style and position."""

    seed = int(seed)
    group = seed // 2
    scene = _base_scene(group)
    if seed % 2:
        points = _ordered_points(scene)
        center_x = (points[0][0] + points[-1][0]) // 2
        scene = dict(scene)
        scene["points"] = [[2 * center_x - x, y] for x, y in points]
    return scene


def analytic_gold(scene: dict[str, Any]) -> str:
    high_x, low_x = _extrema_centers(scene)
    return "peak" if high_x < low_x else "trough"


def margin(scene: dict[str, Any]) -> float:
    high_x, low_x = _extrema_centers(scene)
    return float(abs(high_x - low_x))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["points"] = list(reversed([list(point) for point in scene["points"]]))
    return [("reverse_sample_storage", twin)]


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    points = _ordered_points(scene)
    trace = tuple(int(value) for value in scene["trace_rgb"])
    background = tuple(int(value) for value in scene["background_rgb"])
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("unexpected canvas size")
    _extrema_centers(scene)
    if not all(130 <= x <= 670 and 90 <= y <= 510 for x, y in points):
        raise ValueError("profile leaves the safe canvas")
    if not all(0 <= channel <= 255 for color in (trace, background) for channel in color):
        raise ValueError("RGB channel outside range")

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    center_x = (points[0][0] + points[-1][0]) // 2
    center_y = round(sum(point[1] for point in points) / len(points))
    draw.rounded_rectangle(
        (center_x - 76, center_y - 36, center_x + 76, center_y + 36),
        radius=9,
        fill=(253, 253, 251),
        outline=FRAME,
        width=2,
    )
    draw.line(points, fill=trace, width=5, joint="curve")
    draw.text(
        (center_x, center_y + 27),
        "LEFT  →  RIGHT",
        fill=MUTED,
        font=_font(10, bold=True),
        anchor="mm",
    )
    draw.text(
        (width // 2, 34),
        "ORDER OF EXTREMES",
        fill=INK,
        font=_font(22, bold=True),
        anchor="mm",
    )
    return image
