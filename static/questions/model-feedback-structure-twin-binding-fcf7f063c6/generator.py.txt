"""Structure-twin binding world.

Two panels draw the same nine-dot, thirteen-link network in two unrelated
layouts.  One dot on the left carries a black ring; four dots on the right are
coloured.  The ring's counterpart under the unique structural correspondence is
exactly one of the coloured dots, and its colour is the decision.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 1600
HEIGHT = 900
N_VERTICES = 9
N_EDGES = 13

BG = (246, 245, 241)
FRAME = (176, 174, 168)
EDGE = (58, 62, 72)
NODE = (152, 160, 172)
RING = (12, 12, 14)

COLOUR_NAMES = ("red", "blue", "green", "orange")
COLOUR_RGB = {
    "red": (206, 46, 40),
    "blue": (34, 78, 200),
    "green": (26, 136, 60),
    "orange": (232, 142, 16),
}

NODE_R = 28
RING_R = 42
EDGE_W = 7
FRAME_W = 4
DIVIDER_X = 800

LEFT_BOX = (85, 95, 705, 805)
RIGHT_BOX = (895, 95, 1515, 805)

MIN_CENTRE_DIST = 132.0
MIN_NODE_EDGE_DIST = 50.0
MIN_INCIDENT_ANGLE_DEG = 20.0
MIN_CROSS_ANGLE_DEG = 22.0
MIN_CROSS_VERTEX_DIST = 65.0
MIN_SIGHT_GAP = 22.0
QUARANTINE_GAP = 30.0

_EXCLUDE_R = 46.0


# --------------------------------------------------------------------------
# graph helpers
# --------------------------------------------------------------------------
def _adjacency(n: int, edges) -> list[set[int]]:
    adj: list[set[int]] = [set() for _ in range(n)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    return adj


def _connected(n: int, adj) -> bool:
    seen = {0}
    stack = [0]
    while stack:
        x = stack.pop()
        for y in adj[x]:
            if y not in seen:
                seen.add(y)
                stack.append(y)
    return len(seen) == n


def count_isomorphisms(n: int, adj_a, adj_b, limit: int = 2) -> int:
    """Count isomorphisms adj_a -> adj_b, stopping at ``limit``."""
    deg_a = [len(adj_a[i]) for i in range(n)]
    deg_b = [len(adj_b[i]) for i in range(n)]
    if sorted(deg_a) != sorted(deg_b):
        return 0
    order = sorted(range(n), key=lambda i: -deg_a[i])
    mapping: dict[int, int] = {}
    used = [False] * n
    found = 0

    def backtrack(pos: int) -> None:
        nonlocal found
        if found >= limit:
            return
        if pos == n:
            found += 1
            return
        u = order[pos]
        for w in range(n):
            if used[w] or deg_b[w] != deg_a[u]:
                continue
            ok = True
            for x, y in mapping.items():
                if (x in adj_a[u]) != (y in adj_b[w]):
                    ok = False
                    break
            if not ok:
                continue
            mapping[u] = w
            used[w] = True
            backtrack(pos + 1)
            del mapping[u]
            used[w] = False
            if found >= limit:
                return

    backtrack(0)
    return found


def _wl_history(n: int, adj, rounds: int = 5) -> list[list[int]]:
    colours = [len(adj[i]) for i in range(n)]
    history = [list(colours)]
    for _ in range(rounds):
        keys = [
            (colours[i], tuple(sorted(colours[j] for j in adj[i])))
            for i in range(n)
        ]
        table = {key: index for index, key in enumerate(sorted(set(keys)))}
        colours = [table[key] for key in keys]
        history.append(list(colours))
    return history


def _rounds_needed(history, marked: int, decoys) -> int:
    for depth, colours in enumerate(history):
        if all(colours[d] != colours[marked] for d in decoys):
            return depth
    return len(history)


# --------------------------------------------------------------------------
# geometry helpers
# --------------------------------------------------------------------------
def _point_segment_distance(p, a, b) -> float:
    px, py = p
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    length2 = dx * dx + dy * dy
    if length2 <= 0.0:
        return math.hypot(px - ax, py - ay)
    t = ((px - ax) * dx + (py - ay) * dy) / length2
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _angle(a, b) -> float:
    return math.atan2(b[1] - a[1], b[0] - a[0])


def _angle_gap_deg(t1: float, t2: float) -> float:
    d = abs(t1 - t2) % (2.0 * math.pi)
    if d > math.pi:
        d = 2.0 * math.pi - d
    return math.degrees(d)


def _segment_crossing(a, b, c, d):
    r = (b[0] - a[0], b[1] - a[1])
    s = (d[0] - c[0], d[1] - c[1])
    den = r[0] * s[1] - r[1] * s[0]
    if abs(den) < 1e-9:
        return None
    t = ((c[0] - a[0]) * s[1] - (c[1] - a[1]) * s[0]) / den
    u = ((c[0] - a[0]) * r[1] - (c[1] - a[1]) * r[0]) / den
    if 0.0 < t < 1.0 and 0.0 < u < 1.0:
        return (a[0] + t * r[0], a[1] + t * r[1])
    return None


def _sight_gap(u: int, v: int, positions, edges) -> float:
    """Widest background disc that fits on the open sight line from u to v."""
    a = positions[u]
    b = positions[v]
    length = math.hypot(b[0] - a[0], b[1] - a[1])
    if length <= 2.0 * _EXCLUDE_R:
        return 0.0
    lo = _EXCLUDE_R / length
    hi = 1.0 - lo
    steps = 64
    best = 0.0
    for k in range(steps + 1):
        t = lo + (hi - lo) * k / steps
        p = (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))
        clear = min(
            _point_segment_distance(p, positions[x], positions[y]) - EDGE_W / 2.0
            for x, y in edges
        )
        for w in range(len(positions)):
            clear = min(clear, math.hypot(p[0] - positions[w][0], p[1] - positions[w][1]) - RING_R)
        best = max(best, clear)
    return 2.0 * max(0.0, best)


def _layout_ok(positions, edges) -> tuple[bool, float]:
    n = len(positions)
    for i in range(n):
        for j in range(i + 1, n):
            if math.hypot(positions[i][0] - positions[j][0], positions[i][1] - positions[j][1]) < MIN_CENTRE_DIST:
                return False, 0.0
    edge_set = {tuple(sorted(e)) for e in edges}
    for u, v in edge_set:
        for w in range(n):
            if w in (u, v):
                continue
            if _point_segment_distance(positions[w], positions[u], positions[v]) < MIN_NODE_EDGE_DIST:
                return False, 0.0
    adj = _adjacency(n, edge_set)
    for w in range(n):
        angles = sorted(_angle(positions[w], positions[x]) for x in adj[w])
        for i in range(len(angles)):
            for j in range(i + 1, len(angles)):
                if _angle_gap_deg(angles[i], angles[j]) < MIN_INCIDENT_ANGLE_DEG:
                    return False, 0.0
    listed = sorted(edge_set)
    for i in range(len(listed)):
        for j in range(i + 1, len(listed)):
            u1, v1 = listed[i]
            u2, v2 = listed[j]
            if {u1, v1} & {u2, v2}:
                continue
            cross = _segment_crossing(positions[u1], positions[v1], positions[u2], positions[v2])
            if cross is None:
                continue
            if _angle_gap_deg(_angle(positions[u1], positions[v1]), _angle(positions[u2], positions[v2])) < MIN_CROSS_ANGLE_DEG:
                return False, 0.0
            for w in range(n):
                if math.hypot(cross[0] - positions[w][0], cross[1] - positions[w][1]) < MIN_CROSS_VERTEX_DIST:
                    return False, 0.0
    worst = float("inf")
    for i in range(n):
        for j in range(i + 1, n):
            if (i, j) in edge_set:
                continue
            worst = min(worst, _sight_gap(i, j, positions, listed))
    if worst < MIN_SIGHT_GAP:
        return False, 0.0
    return True, worst


def _cheap_penalty(positions, listed, adj) -> float:
    n = len(positions)
    pen = 0.0
    for i in range(n):
        for j in range(i + 1, n):
            d = math.hypot(positions[i][0] - positions[j][0], positions[i][1] - positions[j][1])
            if d < MIN_CENTRE_DIST:
                pen += MIN_CENTRE_DIST - d
    for u, v in listed:
        for w in range(n):
            if w in (u, v):
                continue
            d = _point_segment_distance(positions[w], positions[u], positions[v])
            if d < MIN_NODE_EDGE_DIST:
                pen += 2.0 * (MIN_NODE_EDGE_DIST - d)
    for w in range(n):
        angles = sorted(_angle(positions[w], positions[x]) for x in adj[w])
        for i in range(len(angles)):
            for j in range(i + 1, len(angles)):
                gap = _angle_gap_deg(angles[i], angles[j])
                if gap < MIN_INCIDENT_ANGLE_DEG:
                    pen += 3.0 * (MIN_INCIDENT_ANGLE_DEG - gap)
    for i in range(len(listed)):
        for j in range(i + 1, len(listed)):
            u1, v1 = listed[i]
            u2, v2 = listed[j]
            if {u1, v1} & {u2, v2}:
                continue
            cross = _segment_crossing(positions[u1], positions[v1], positions[u2], positions[v2])
            if cross is None:
                continue
            ang = _angle_gap_deg(
                _angle(positions[u1], positions[v1]), _angle(positions[u2], positions[v2])
            )
            if ang < MIN_CROSS_ANGLE_DEG:
                pen += 3.0 * (MIN_CROSS_ANGLE_DEG - ang)
            for w in range(n):
                d = math.hypot(cross[0] - positions[w][0], cross[1] - positions[w][1])
                if d < MIN_CROSS_VERTEX_DIST:
                    pen += 2.0 * (MIN_CROSS_VERTEX_DIST - d)
    return pen


def _sample_layout(rng: random.Random, box, edges):
    x0, y0, x1, y1 = box
    listed = sorted({tuple(sorted(e)) for e in edges})
    adj = _adjacency(N_VERTICES, listed)
    for _ in range(24):
        positions = [
            (float(rng.randint(x0, x1)), float(rng.randint(y0, y1)))
            for _ in range(N_VERTICES)
        ]
        pen = _cheap_penalty(positions, listed, adj)
        for step in range(2600):
            if pen <= 0.0:
                break
            w = rng.randrange(N_VERTICES)
            if step % 3 == 0:
                trial = (float(rng.randint(x0, x1)), float(rng.randint(y0, y1)))
            else:
                sigma = 90.0 if step < 1200 else 35.0
                trial = (
                    float(min(x1, max(x0, round(positions[w][0] + rng.gauss(0.0, sigma))))),
                    float(min(y1, max(y0, round(positions[w][1] + rng.gauss(0.0, sigma))))),
                )
            keep = positions[w]
            positions[w] = trial
            new_pen = _cheap_penalty(positions, listed, adj)
            if new_pen <= pen:
                pen = new_pen
            else:
                positions[w] = keep
        if pen > 0.0:
            continue
        good, gap = _layout_ok(positions, edges)
        if good:
            return positions, gap
    return None, 0.0


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _sample_graph(rng: random.Random):
    pairs = [(i, j) for i in range(N_VERTICES) for j in range(i + 1, N_VERTICES)]
    for _ in range(4000):
        edges = sorted(rng.sample(pairs, N_EDGES))
        adj = _adjacency(N_VERTICES, edges)
        degrees = [len(a) for a in adj]
        if min(degrees) < 2 or max(degrees) > 5:
            continue
        if not _connected(N_VERTICES, adj):
            continue
        classes: dict[int, list[int]] = {}
        for i, d in enumerate(degrees):
            classes.setdefault(d, []).append(i)
        usable = [c for c in classes.values() if len(c) >= 4]
        if not usable:
            continue
        if count_isomorphisms(N_VERTICES, adj, adj, limit=2) != 1:
            continue
        history = _wl_history(N_VERTICES, adj)
        best = None
        for group in usable:
            for marked in group:
                others = [x for x in group if x != marked]
                for a in range(len(others)):
                    for b in range(a + 1, len(others)):
                        for c in range(b + 1, len(others)):
                            decoys = (others[a], others[b], others[c])
                            depth = _rounds_needed(history, marked, decoys)
                            if best is None or depth > best[0]:
                                best = (depth, marked, decoys)
        if best is None or best[0] < 2:
            continue
        return edges, best[1], list(best[2]), best[0]
    raise RuntimeError("graph sampling failed")


def _normalised(positions, box):
    x0, y0, x1, y1 = box
    return [
        ((p[0] - x0) / float(x1 - x0), (p[1] - y0) / float(y1 - y0))
        for p in positions
    ]


def _position_shortcut_blocked(left_pos, right_pos, marked, candidates) -> bool:
    nl = _normalised(left_pos, LEFT_BOX)
    nr = _normalised(right_pos, RIGHT_BOX)
    anchor = nl[marked]
    dists = {
        c: math.hypot(nr[c][0] - anchor[0], nr[c][1] - anchor[1])
        for c in candidates
    }
    closer = sum(1 for c in candidates if dists[c] < dists[marked])
    if closer < 2:
        return False
    left_rank_x = sorted(range(N_VERTICES), key=lambda i: left_pos[i][0]).index(marked)
    right_rank_x = sorted(range(N_VERTICES), key=lambda i: right_pos[i][0]).index(marked)
    if left_rank_x == right_rank_x:
        return False
    left_rank_y = sorted(range(N_VERTICES), key=lambda i: left_pos[i][1]).index(marked)
    right_rank_y = sorted(range(N_VERTICES), key=lambda i: right_pos[i][1]).index(marked)
    return left_rank_y != right_rank_y


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 2654435761) % (2 ** 61 - 1))
    for _ in range(200):
        edges, marked, decoys, depth = _sample_graph(rng)
        left_pos, left_gap = _sample_layout(rng, LEFT_BOX, edges)
        if left_pos is None:
            continue
        right_pos, right_gap = _sample_layout(rng, RIGHT_BOX, edges)
        if right_pos is None:
            continue
        candidates = [marked] + list(decoys)
        if not _position_shortcut_blocked(left_pos, right_pos, marked, candidates):
            continue
        names = list(COLOUR_NAMES)
        random.Random((int(seed) ^ 0x5BF03635) & 0xFFFFFFFF).shuffle(names)
        coloured = [[int(v), names[k]] for k, v in enumerate(candidates)]
        coloured.sort(key=lambda item: item[0])
        return {
            "canvas": [WIDTH, HEIGHT],
            "n": N_VERTICES,
            "edges": [[int(u), int(v)] for u, v in edges],
            "left_pos": [[float(p[0]), float(p[1])] for p in left_pos],
            "right_pos": [[float(p[0]), float(p[1])] for p in right_pos],
            "marked": int(marked),
            "coloured": coloured,
            "sight_gap": float(round(min(left_gap, right_gap), 4)),
            "refinement_depth": int(depth),
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _draw_panel(draw, positions, edges, marked, colour_of) -> None:
    for u, v in edges:
        draw.line(
            [tuple(int(c) for c in positions[u]), tuple(int(c) for c in positions[v])],
            fill=EDGE,
            width=EDGE_W,
        )
    if marked is not None:
        cx, cy = (int(positions[marked][0]), int(positions[marked][1]))
        draw.ellipse([cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R], fill=RING)
    for index, p in enumerate(positions):
        cx, cy = int(p[0]), int(p[1])
        fill = COLOUR_RGB[colour_of[index]] if index in colour_of else NODE
        draw.ellipse([cx - NODE_R, cy - NODE_R, cx + NODE_R, cy + NODE_R], fill=fill)


def render(scene: dict) -> Image.Image:
    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    draw.rectangle([30, 30, DIVIDER_X - 40, height - 31], outline=FRAME, width=FRAME_W)
    draw.rectangle([DIVIDER_X + 40, 30, width - 31, height - 31], outline=FRAME, width=FRAME_W)
    draw.line([(DIVIDER_X, 30), (DIVIDER_X, height - 31)], fill=FRAME, width=6)
    edges = [(int(u), int(v)) for u, v in scene["edges"]]
    left_pos = [(float(p[0]), float(p[1])) for p in scene["left_pos"]]
    right_pos = [(float(p[0]), float(p[1])) for p in scene["right_pos"]]
    _draw_panel(draw, left_pos, edges, int(scene["marked"]), {})
    colour_of = {int(v): name for v, name in scene["coloured"]}
    _draw_panel(draw, right_pos, edges, None, colour_of)
    return image


# --------------------------------------------------------------------------
# labels and gates
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    marked = int(scene["marked"])
    for vertex, name in scene["coloured"]:
        if int(vertex) == marked:
            return str(name)
    raise ValueError("marked vertex carries no colour in the right panel")


def margin(scene: dict) -> float:
    return float(scene["sight_gap"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_edges = dict(scene)
    reversed_edges["edges"] = list(reversed(scene["edges"]))
    swapped = dict(scene)
    swapped["edges"] = [[int(v), int(u)] for u, v in scene["edges"]]
    rotated = dict(scene)
    coloured = list(scene["coloured"])
    rotated["coloured"] = coloured[1:] + coloured[:1]
    return [
        ("edge_list_reversed", reversed_edges),
        ("edge_endpoints_swapped", swapped),
        ("coloured_list_rotated", rotated),
    ]
