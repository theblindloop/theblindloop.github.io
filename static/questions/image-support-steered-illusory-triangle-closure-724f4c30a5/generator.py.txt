"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

SIZE = 512
SCALE = 4
BACKGROUND = (247, 245, 239)
PALETTES = ((38, 91, 156), (33, 122, 112), (112, 73, 156), (177, 70, 79))
HALF_MOUTH = math.radians(37.0)
QUARANTINE_COSINE = 0.25


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    center_cell_x = rng.choice((2, 3, 4, 5))
    center_cell_y = rng.choice((2, 3, 4, 5))
    upright = bool(rng.randrange(2))
    if upright:
        offsets = ((0, -1), (-1, 1), (1, 1))
    else:
        offsets = ((-1, -1), (1, -1), (0, 1))
    centers = [
        [32 + 64 * (center_cell_x + dx), 32 + 64 * (center_cell_y + dy)]
        for dx, dy in offsets
    ]
    centroid_x = sum(point[0] for point in centers) / 3.0
    centroid_y = sum(point[1] for point in centers) / 3.0
    angles = [
        math.atan2(centroid_y - y, centroid_x - x) + math.radians(rng.uniform(-3.0, 3.0))
        for x, y in centers
    ]
    closed = int(seed) % 2 == 0
    if not closed:
        broken_index = rng.randrange(3)
        angles[broken_index] += math.pi
    return {
        "centers": centers,
        "mouth_angles": angles,
        "radius": rng.choice((23, 24, 25)),
        "color": list(rng.choice(PALETTES)),
    }


def _alignment_scores(scene: dict) -> list[float]:
    centers = scene["centers"]
    angles = scene["mouth_angles"]
    centroid_x = sum(float(point[0]) for point in centers) / len(centers)
    centroid_y = sum(float(point[1]) for point in centers) / len(centers)
    scores = []
    for (x, y), angle in zip(centers, angles, strict=True):
        dx = centroid_x - float(x)
        dy = centroid_y - float(y)
        length = math.hypot(dx, dy)
        scores.append((math.cos(float(angle)) * dx + math.sin(float(angle)) * dy) / length)
    return scores


def analytic_gold(scene: dict) -> str:
    return "closed" if min(_alignment_scores(scene)) > 0.0 else "broken"


def margin(scene: dict) -> float:
    """Absolute weakest alignment: the runner-up gap to the zero boundary."""
    return min(abs(score) for score in _alignment_scores(scene))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_COSINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    centers = scene["centers"]
    angles = scene["mouth_angles"]
    radius = int(scene["radius"])
    color = tuple(int(value) for value in scene["color"])
    canvas = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    for (x, y), angle in zip(centers, angles, strict=True):
        cx, cy, rr = int(x) * SCALE, int(y) * SCALE, radius * SCALE
        draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=color)
        pad = 2 * SCALE
        start = math.degrees(float(angle) - HALF_MOUTH)
        end = math.degrees(float(angle) + HALF_MOUTH)
        draw.pieslice(
            (cx - rr - pad, cy - rr - pad, cx + rr + pad, cy + rr + pad),
            start=start,
            end=end,
            fill=BACKGROUND,
        )
    return canvas.resize((SIZE, SIZE), Image.Resampling.LANCZOS)
