"""Independent image-only inverse for illusory triangle closure."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 180:
            groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _mouth_vector(mask: np.ndarray, points: np.ndarray) -> tuple[float, float, float, float]:
    ys = points[:, 0]
    xs = points[:, 1]
    cx = (float(xs.min()) + float(xs.max())) / 2.0
    cy = (float(ys.min()) + float(ys.max())) / 2.0
    radius = (float(xs.max() - xs.min()) + float(ys.max() - ys.min())) / 4.0
    if radius < 13.0:
        raise ValueError("an inducer is too truncated to invert")
    y0 = max(0, int(math.floor(cy - radius)))
    y1 = min(mask.shape[0], int(math.ceil(cy + radius + 1)))
    x0 = max(0, int(math.floor(cx - radius)))
    x1 = min(mask.shape[1], int(math.ceil(cx + radius + 1)))
    yy, xx = np.mgrid[y0:y1, x0:x1]
    dx = xx.astype(float) - cx
    dy = yy.astype(float) - cy
    interior = dx * dx + dy * dy <= (0.68 * radius) ** 2
    missing = interior & (~mask[y0:y1, x0:x1])
    if int(missing.sum()) < 18:
        raise ValueError("missing wedge cannot be reconstructed")
    vx = float(dx[missing].mean())
    vy = float(dy[missing].mean())
    if math.hypot(vx, vy) < 2.0:
        raise ValueError("missing wedge direction is ambiguous")
    return cx, cy, vx, vy


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3), array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array.astype(float) - background.astype(float), axis=2)
    mask = distance > 48.0
    groups = _components(mask)
    if len(groups) != 3:
        raise ValueError(f"expected exactly three visible inducers, found {len(groups)}")
    estimates = [_mouth_vector(mask, group) for group in groups]
    center_x = sum(item[0] for item in estimates) / 3.0
    center_y = sum(item[1] for item in estimates) / 3.0
    scores = []
    for cx, cy, vx, vy in estimates:
        tx, ty = center_x - cx, center_y - cy
        scores.append((vx * tx + vy * ty) / (math.hypot(vx, vy) * math.hypot(tx, ty)))
    weakest = min(scores)
    if min(abs(score) for score in scores) < 0.18:
        raise ValueError("mouth-to-center relation is raster-ambiguous")
    return "closed" if weakest > 0.0 else "broken"
