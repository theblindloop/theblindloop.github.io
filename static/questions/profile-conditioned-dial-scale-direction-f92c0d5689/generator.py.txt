"""Forward arm for `dial_scale_direction`.

A gauge dial is drawn as a grey rim with a grey hub and a grey pointer.  Five
groups of coloured dots sit on a common circle just inside the rim; the groups
show one, two, three, four and five dots.  Reading them in ascending order
walks around the rim in exactly one rotational direction, and that direction -
clockwise or counter-clockwise as seen on the page - is the answer.

The canonical prior is that dials and clock faces ascend clockwise.  Half the
sampled scenes contradict it.  The prior is recorded for analysis only and is
never used as gold.
"""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "dial_scale_direction-renderer-0.1.0"

WIDTH = 480
HEIGHT = 480

VALUES = (1, 2, 3, 4, 5)
DIRECTIONS = ("clockwise", "counterclockwise")
CANONICAL_DECISION = "clockwise"

QUARANTINE_MARGIN = 6.0
MIN_SAMPLED_MARGIN = 3.0
MIN_STEP_DEG = 20.0
MAX_STEP_DEG = 160.0

WHITE = (255, 255, 255)
FRAME_COLOR = (150, 150, 150)
RIM_COLOR = (150, 150, 150)
NEEDLE_COLOR = (110, 110, 110)
HUB_COLOR = (150, 150, 150)

PALETTE = (
    (200, 40, 40),
    (30, 90, 200),
    (25, 140, 60),
    (150, 45, 160),
    (200, 110, 10),
    (20, 130, 150),
)


def _wrap(deg: float) -> float:
    """Signed angular difference in (-180, 180]."""
    d = (deg + 180.0) % 360.0 - 180.0
    return 180.0 if d == -180.0 else d


def _half_width(value: int, gap: float) -> float:
    """Half the angular extent a group of `value` dots occupies."""
    return (value - 1) / 2.0 * gap


def _edge_gaps(scene: dict) -> list:
    """Blank arc, in degrees, between each neighbouring pair of dot groups."""
    gap = scene["dot_gap_deg"]
    items = sorted((float(a) % 360.0, int(v)) for v, a in scene["groups"])
    out = []
    n = len(items)
    for i in range(n):
        a, va = items[i]
        b, vb = items[(i + 1) % n]
        d = (b - a) % 360.0
        out.append(d - _half_width(va, gap) - _half_width(vb, gap))
    return out


def _steps(scene: dict) -> list:
    """Signed angular step from the group showing v dots to the one showing v+1."""
    by_value = {int(v): float(a) for v, a in scene["groups"]}
    return [_wrap(by_value[v + 1] - by_value[v]) for v in (1, 2, 3, 4)]


def _draft(rng: random.Random) -> dict:
    ring_r = rng.randint(168, 186)
    group_r = ring_r - rng.randint(18, 30)
    phi0 = round(rng.uniform(0.0, 360.0), 3)
    sense = rng.choice((1, -1))
    jitter = [round(rng.uniform(-22.0, 22.0), 3) for _ in range(5)]
    groups = [
        [i + 1, round((phi0 + sense * (72.0 * i + jitter[i])) % 360.0, 3)]
        for i in range(5)
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cx": 240 + rng.randint(-12, 12),
        "cy": 240 + rng.randint(-12, 12),
        "ring_r": ring_r,
        "ring_w": rng.randint(3, 5),
        "group_r": group_r,
        "dot_r": rng.randint(6, 8),
        "dot_gap_deg": round(rng.uniform(7.5, 10.0), 3),
        "groups": groups,
        "needle_deg": round(rng.uniform(0.0, 360.0), 3),
        "needle_len": round(group_r * rng.uniform(0.45, 0.62), 2),
        "hub_r": rng.randint(7, 10),
        "dot_color": list(PALETTE[rng.randrange(len(PALETTE))]),
    }


def _legal(scene: dict) -> bool:
    gap = scene["dot_gap_deg"]
    if min(_edge_gaps(scene)) - gap < MIN_SAMPLED_MARGIN:
        return False
    arc_px = math.radians(gap) * scene["group_r"]
    if arc_px < 2 * scene["dot_r"] + 8:
        return False
    steps = _steps(scene)
    if not (all(s > 0 for s in steps) or all(s < 0 for s in steps)):
        return False
    return all(MIN_STEP_DEG <= abs(s) <= MAX_STEP_DEG for s in steps)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(4000):
        scene = _draft(rng)
        if _legal(scene):
            return scene
    raise RuntimeError(f"no legal scene for seed {seed}")


def _dot_positions(scene: dict) -> list:
    gap = scene["dot_gap_deg"]
    out = []
    for value, ang in scene["groups"]:
        value = int(value)
        for j in range(value):
            a = float(ang) + (j - (value - 1) / 2.0) * gap
            r = math.radians(a)
            out.append(
                (
                    scene["cx"] + scene["group_r"] * math.cos(r),
                    scene["cy"] + scene["group_r"] * math.sin(r),
                )
            )
    return out


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), WHITE)
    draw = ImageDraw.Draw(img)
    w, h = scene["width"], scene["height"]

    draw.rectangle([10, 10, w - 11, h - 11], outline=FRAME_COLOR, width=2)
    for x0, y0, x1, y1 in (
        (24, 24, 44, 24),
        (w - 45, 24, w - 25, 24),
        (24, h - 25, 44, h - 25),
        (w - 45, h - 25, w - 25, h - 25),
    ):
        draw.line([x0, y0, x1, y1], fill=FRAME_COLOR, width=2)

    cx, cy = scene["cx"], scene["cy"]
    rr = scene["ring_r"]
    draw.ellipse([cx - rr, cy - rr, cx + rr, cy + rr], outline=RIM_COLOR, width=scene["ring_w"])

    a = math.radians(scene["needle_deg"])
    ex = round(cx + scene["needle_len"] * math.cos(a))
    ey = round(cy + scene["needle_len"] * math.sin(a))
    draw.line([cx, cy, ex, ey], fill=NEEDLE_COLOR, width=3)
    hr = scene["hub_r"]
    draw.ellipse([cx - hr, cy - hr, cx + hr, cy + hr], fill=HUB_COLOR)

    dr = scene["dot_r"]
    color = tuple(int(c) for c in scene["dot_color"])
    for x, y in _dot_positions(scene):
        draw.ellipse([round(x - dr), round(y - dr), round(x + dr), round(y + dr)], fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    steps = _steps(scene)
    if all(s > 0 for s in steps):
        return "clockwise"
    if all(s < 0 for s in steps):
        return "counterclockwise"
    raise ValueError("scene has no consistent ascending direction")


def margin(scene: dict) -> float:
    """Smallest visible separation, in degrees, that protects the reading.

    Two visible quantities can fail.  The blank arc between neighbouring groups
    can shrink to the blank arc inside a group, at which point the marks stop
    being five readable groups; and an ascending step can approach the
    antipode, at which point 'the way round' stops being defined.  The margin is
    the smaller of the two, clamped at zero.
    """
    excess = min(_edge_gaps(scene)) - scene["dot_gap_deg"]
    antipode = min(180.0 - abs(s) for s in _steps(scene))
    return round(max(0.0, min(excess, antipode)), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every field is painted into the raster."""
    return []


def runner_up(scene: dict) -> str:
    return "counterclockwise" if analytic_gold(scene) == "clockwise" else "clockwise"


def _copy(scene: dict) -> dict:
    out = dict(scene)
    out["groups"] = [[int(v), float(a)] for v, a in scene["groups"]]
    out["dot_color"] = list(scene["dot_color"])
    return out


def translate(scene: dict, dx: int, dy: int) -> dict:
    out = _copy(scene)
    out["cx"] += dx
    out["cy"] += dy
    return out


def recolor(scene: dict) -> dict:
    out = _copy(scene)
    current = tuple(scene["dot_color"])
    for c in PALETTE:
        if c != current:
            out["dot_color"] = list(c)
            break
    return out


def move_needle(scene: dict, seed: int) -> dict:
    """Nuisance: the pointer is decoration and never indicates the answer."""
    out = _copy(scene)
    rng = random.Random(seed ^ 0x5EED)
    out["needle_deg"] = round(rng.uniform(0.0, 360.0), 3)
    out["needle_len"] = round(scene["group_r"] * rng.uniform(0.45, 0.62), 2)
    return out


def rescale(scene: dict, factor: float = 0.9) -> dict:
    out = _copy(scene)
    out["ring_r"] = int(round(scene["ring_r"] * factor))
    out["group_r"] = int(round(scene["group_r"] * factor))
    out["dot_r"] = max(5, int(round(scene["dot_r"] * factor)))
    out["hub_r"] = max(5, int(round(scene["hub_r"] * factor)))
    out["needle_len"] = round(scene["needle_len"] * factor, 2)
    return out


def set_direction(scene: dict, direction: str) -> dict:
    """Declared label-transforming edit: reflect the layout about the one-dot mark."""
    if direction not in DIRECTIONS:
        raise ValueError(f"unknown direction: {direction}")
    if analytic_gold(scene) == direction:
        return _copy(scene)
    out = _copy(scene)
    anchor = next(float(a) for v, a in scene["groups"] if int(v) == 1)
    out["groups"] = [
        [int(v), round((2.0 * anchor - float(a)) % 360.0, 3)] for v, a in scene["groups"]
    ]
    return out


def reverse_direction(scene: dict) -> dict:
    return set_direction(scene, runner_up(scene))


def collapse_margin(scene: dict) -> dict:
    """Widen every group until the tightest between-group arc is smaller than
    the arc inside a group: the five marks stop being separable."""
    out = _copy(scene)
    gap = scene["dot_gap_deg"]
    items = sorted((float(a) % 360.0, int(v)) for v, a in scene["groups"])
    best = None
    n = len(items)
    for i in range(n):
        a, va = items[i]
        b, vb = items[(i + 1) % n]
        d = (b - a) % 360.0
        k = (va - 1) / 2.0 + (vb - 1) / 2.0
        edge = d - k * gap
        if best is None or edge < best[0]:
            best = (edge, d, k)
    _, d, k = best
    out["dot_gap_deg"] = round(d / (k + 0.35), 4)
    return out


def ink_pixels(image: Image.Image) -> int:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    return int((chroma >= 40).sum())
