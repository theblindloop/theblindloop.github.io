"""Deterministic renderer and analytic label for the red-region component world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 3
TARGET_RGB = (220, 45, 55)
DISTRACTOR_RGB = {
    "blue": (45, 105, 210),
    "green": (45, 165, 100),
    "purple": (125, 65, 190),
    "cyan": (35, 170, 190),
}
QUARANTINE_MARGIN = 18.0


def _bound_radius(region: dict) -> float:
    return 1.12 * max(float(region["rx"]), float(region["ry"]))


def _all_regions(scene: dict) -> list[dict]:
    return list(scene["components"]) + list(scene["distractors"])


def _points(region: dict, scale: float = 1.0) -> list[tuple[float, float]]:
    cx = float(region["cx"]) * scale
    cy = float(region["cy"]) * scale
    rx = float(region["rx"]) * scale
    ry = float(region["ry"]) * scale
    angle = math.radians(float(region["angle"]))
    phase = float(region["phase"])
    vertices = int(region["vertices"])
    result = []
    for index in range(vertices):
        t = 2.0 * math.pi * index / vertices
        ripple = 1.0 + 0.08 * math.sin(3.0 * t + phase) + 0.045 * math.cos(2.0 * t - phase)
        x0 = rx * ripple * math.cos(t)
        y0 = ry * ripple * math.sin(t)
        x = cx + x0 * math.cos(angle) - y0 * math.sin(angle)
        y = cy + x0 * math.sin(angle) + y0 * math.cos(angle)
        result.append((x, y))
    return result


def _draw_regions(draw: ImageDraw.ImageDraw, regions: Iterable[dict], color_for: str) -> None:
    for region in regions:
        if color_for == "red":
            color = TARGET_RGB
        else:
            color = DISTRACTOR_RGB[str(region["color"])]
        polygon = [(round(x), round(y)) for x, y in _points(region, SUPERSAMPLE)]
        draw.polygon(polygon, fill=color)


def sample_scene(seed: int) -> dict:
    """Sample a scene whose visible red component count is 1, 2, or 3."""

    rng = random.Random(int(seed))
    target_count = 1 + (int(seed) % 3)
    near_boundary = (int(seed) // 3) % 9 == 0
    requested_clearance = 11.5 if near_boundary else rng.uniform(24.0, 37.0)

    # Keep total red area overlapping across classes: more components are smaller.
    target_bases = {1: (60.0, 40.0), 2: (43.0, 30.0), 3: (34.0, 25.0)}
    components: list[dict] = []
    for _ in range(target_count):
        base_rx, base_ry = target_bases[target_count]
        components.append(
            {
                "cx": 0.0,
                "cy": 0.0,
                "rx": base_rx * rng.uniform(0.87, 1.13),
                "ry": base_ry * rng.uniform(0.87, 1.13),
                "angle": rng.uniform(0.0, 180.0),
                "phase": rng.uniform(0.0, 2.0 * math.pi),
                "vertices": rng.choice((11, 12, 13, 14)),
            }
        )

    distractors: list[dict] = []
    distractor_count = 2 + rng.randrange(4)
    colors = list(DISTRACTOR_RGB)
    for _ in range(distractor_count):
        distractors.append(
            {
                "cx": 0.0,
                "cy": 0.0,
                "rx": rng.uniform(18.0, 34.0),
                "ry": rng.uniform(14.0, 29.0),
                "angle": rng.uniform(0.0, 180.0),
                "phase": rng.uniform(0.0, 2.0 * math.pi),
                "vertices": rng.choice((9, 10, 11, 12)),
                "color": colors[rng.randrange(len(colors))],
            }
        )

    placed: list[dict] = []
    for region in components + distractors:
        radius = _bound_radius(region)
        placed_ok = False
        for _attempt in range(2500):
            region["cx"] = rng.uniform(radius + 8.0, WIDTH - radius - 8.0)
            region["cy"] = rng.uniform(radius + 8.0, HEIGHT - radius - 8.0)
            if all(
                math.hypot(region["cx"] - other["cx"], region["cy"] - other["cy"])
                >= radius + _bound_radius(other) + requested_clearance
                for other in placed
            ):
                placed.append(region)
                placed_ok = True
                break
        if not placed_ok:
            raise RuntimeError("could not pack randomized regions")

    # The top-level fields are deliberately limited to rendered scene inputs.
    return {"components": components, "distractors": distractors}


def render(scene: dict) -> Image.Image:
    high = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), "white")
    draw = ImageDraw.Draw(high)
    _draw_regions(draw, scene["distractors"], "distractor")
    _draw_regions(draw, scene["components"], "red")
    return high.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """The analytic label uses only the visible target-region list cardinality."""

    return str(len(scene["components"]))


def margin(scene: dict) -> float:
    """Return the smallest estimated geometric clearance in pixels."""

    regions = _all_regions(scene)
    clearances = []
    for region in regions:
        radius = _bound_radius(region)
        clearances.extend(
            (
                float(region["cx"]) - radius,
                WIDTH - float(region["cx"]) - radius,
                float(region["cy"]) - radius,
                HEIGHT - float(region["cy"]) - radius,
            )
        )
    for index, left in enumerate(regions):
        for right in regions[index + 1 :]:
            clearances.append(
                math.hypot(float(left["cx"]) - float(right["cx"]), float(left["cy"]) - float(right["cy"]))
                - _bound_radius(left)
                - _bound_radius(right)
            )
    return round(min(clearances), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """List-order permutations are pixel-identical because all regions are separated."""

    twin = copy.deepcopy(scene)
    twin["components"] = list(reversed(twin["components"]))
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_render_list_order", twin)]
