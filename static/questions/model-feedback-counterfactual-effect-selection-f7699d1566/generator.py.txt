"""Deterministic visual world for selecting an intervention by its causal effect."""

from __future__ import annotations

import math
import random
from itertools import product
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 800
SCALE = 3

SOURCE_CENTERS = {
    "S": (70, 86),
    "T": (276, 86),
    "U": (484, 86),
    "V": (690, 86),
}
GATE_CENTERS = {
    "J": (145, 205),
    "K": (380, 205),
    "L": (615, 205),
    "M": (260, 330),
    "N": (500, 330),
    "R": (380, 455),
    "P": (260, 570),
    "Q": (500, 570),
}
EDGES = (
    ("S", "J"), ("T", "J"), ("T", "K"), ("U", "K"),
    ("U", "L"), ("V", "L"), ("J", "M"), ("K", "M"),
    ("K", "N"), ("L", "N"), ("M", "R"), ("N", "R"),
    ("M", "P"), ("R", "P"), ("R", "Q"), ("N", "Q"),
)
GATE_IDS = tuple(GATE_CENTERS)
GATE_TYPES = ("XOR", "AND", "OR")
ELIGIBLE_NODES = ("J", "K", "L", "M")
OPTIONS = ("A", "B", "C", "D")
CARD_LEFTS = (12, 201, 390, 579)

GATE_COLORS = {
    "XOR": "#2474D8",
    "AND": "#F2A900",
    "OR": "#8754C8",
}
NODE_COLORS = {
    "J": "#D84A5B",
    "K": "#00A3A3",
    "L": "#3A9D4D",
    "M": "#D13FA5",
}
INK = "#17212B"
RED = "#D5262F"
GRAY = "#687582"
PALE = "#F4F6F8"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _gate(kind: str, left: int, right: int) -> int:
    if kind == "XOR":
        return int(left != right)
    if kind == "AND":
        return int(left == 1 and right == 1)
    if kind == "OR":
        return int(left == 1 or right == 1)
    raise ValueError(f"unknown gate type {kind!r}")


def _evaluate(
    scene: dict[str, Any], intervention_node: str | None = None,
    forced_bit: int | None = None,
) -> dict[str, int]:
    source = {name: int(value) for name, value in zip("STUV", scene["source_bits"])}
    kinds = {str(name): str(value) for name, value in scene["gate_types"].items()}
    if intervention_node is not None:
        if intervention_node not in ELIGIBLE_NODES or forced_bit not in (0, 1):
            raise ValueError("invalid intervention")

    values = dict(source)
    values["J"] = _gate(kinds["J"], values["S"], values["T"])
    values["K"] = _gate(kinds["K"], values["T"], values["U"])
    values["L"] = _gate(kinds["L"], values["U"], values["V"])
    if intervention_node in ("J", "K", "L"):
        values[intervention_node] = int(forced_bit)
    values["M"] = _gate(kinds["M"], values["J"], values["K"])
    if intervention_node == "M":
        values["M"] = int(forced_bit)
    values["N"] = _gate(kinds["N"], values["K"], values["L"])
    values["R"] = _gate(kinds["R"], values["M"], values["N"])
    values["P"] = _gate(kinds["P"], values["M"], values["R"])
    values["Q"] = _gate(kinds["Q"], values["R"], values["N"])
    return values


def _effects(scene: dict[str, Any]) -> dict[str, tuple[int, int]]:
    factual = _evaluate(scene)
    effects: dict[str, tuple[int, int]] = {}
    for node, forced in zip(scene["option_nodes"], scene["forced_bits"]):
        changed = _evaluate(scene, str(node), int(forced))
        effects[str(node)] = (
            changed["P"] ^ factual["P"],
            changed["Q"] ^ factual["Q"],
        )
    return effects


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    required_effects = set(product((0, 1), repeat=2))
    for _ in range(50000):
        source_bits = [rng.randrange(2) for _ in SOURCE_CENTERS]
        gate_types = {name: rng.choice(GATE_TYPES) for name in GATE_IDS}
        base: dict[str, Any] = {
            "source_bits": source_bits,
            "gate_types": gate_types,
            "option_nodes": list(ELIGIBLE_NODES),
            "forced_bits": [0, 0, 0, 0],
        }
        factual = _evaluate(base)
        forced_by_node = {node: 1 - factual[node] for node in ELIGIBLE_NODES}
        effects = {}
        for node in ELIGIBLE_NODES:
            changed = _evaluate(base, node, forced_by_node[node])
            effects[node] = (
                changed["P"] ^ factual["P"],
                changed["Q"] ^ factual["Q"],
            )
        if (
            set(source_bits) == {0, 1}
            and set(gate_types.values()) == set(GATE_TYPES)
            and set(effects.values()) == required_effects
        ):
            option_nodes = list(ELIGIBLE_NODES)
            rng.shuffle(option_nodes)
            return {
                "source_bits": source_bits,
                "gate_types": gate_types,
                "option_nodes": option_nodes,
                "forced_bits": [forced_by_node[node] for node in option_nodes],
            }
    raise RuntimeError("could not sample four distinct causal effects")


def analytic_gold(scene: dict[str, Any]) -> str:
    effects = _effects(scene)
    matches = [
        option for option, node in zip(OPTIONS, scene["option_nodes"])
        if effects[str(node)] == (1, 0)
    ]
    if len(matches) != 1:
        raise ValueError("scene has no unique P-only intervention")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        source = list(scene["source_bits"])
        kinds = dict(scene["gate_types"])
        nodes = list(scene["option_nodes"])
        forced = list(scene["forced_bits"])
        if (
            len(source) != 4
            or any(type(value) is not int or value not in (0, 1) for value in source)
            or set(source) != {0, 1}
            or set(kinds) != set(GATE_IDS)
            or any(value not in GATE_TYPES for value in kinds.values())
            or set(kinds.values()) != set(GATE_TYPES)
            or len(nodes) != 4
            or set(nodes) != set(ELIGIBLE_NODES)
            or len(forced) != 4
            or any(type(value) is not int or value not in (0, 1) for value in forced)
        ):
            return True
        factual = _evaluate(scene)
        if any(bit != 1 - factual[node] for node, bit in zip(nodes, forced)):
            return True
        effects = _effects(scene)
        return set(effects.values()) != set(product((0, 1), repeat=2))
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _center_text(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], text: str,
    font: ImageFont.FreeTypeFont, fill: str,
) -> None:
    cx, cy = center
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (int(round(cx * SCALE - width / 2)), int(round(cy * SCALE - height / 2 - SCALE))),
        text, fill=fill, font=font,
    )


def _arrow(
    draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int],
    start_pad: float, end_pad: float,
) -> None:
    dx = float(end[0] - start[0])
    dy = float(end[1] - start[1])
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    p0 = (start[0] + ux * start_pad, start[1] + uy * start_pad)
    tip = (end[0] - ux * end_pad, end[1] - uy * end_pad)
    base = (tip[0] - ux * 11, tip[1] - uy * 11)
    normal = (-uy, ux)
    left = (base[0] + normal[0] * 6, base[1] + normal[1] * 6)
    right = (base[0] - normal[0] * 6, base[1] - normal[1] * 6)
    sp = lambda p: (int(round(p[0] * SCALE)), int(round(p[1] * SCALE)))
    draw.line((*sp(p0), *sp(base)), fill=INK, width=4 * SCALE)
    draw.polygon([sp(tip), sp(left), sp(right)], fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    source_bits = [int(value) for value in scene["source_bits"]]
    gate_types = {str(name): str(value) for name, value in scene["gate_types"].items()}
    option_nodes = [str(value) for value in scene["option_nodes"]]
    forced_bits = [int(value) for value in scene["forced_bits"]]

    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(canvas)
    sx = lambda value: int(round(value * SCALE))

    draw.text((sx(16), sx(12)), "START", fill=GRAY, font=_font(15))
    draw.text((sx(100), sx(12)), "DARK = 1   LIGHT = 0", fill=GRAY, font=_font(13))
    for x, kind, meaning in ((365, "XOR", "different"), (500, "AND", "both"), (625, "OR", "either")):
        draw.rounded_rectangle(
            (sx(x), sx(10), sx(x + 42), sx(36)), radius=sx(5),
            fill=GATE_COLORS[kind], outline=INK, width=sx(2),
        )
        _center_text(draw, (x + 21, 23), kind, _font(8), INK if kind == "AND" else "white")
        draw.text((sx(x + 47), sx(14)), meaning, fill=GRAY, font=_font(9))

    centers = {**SOURCE_CENTERS, **GATE_CENTERS}
    for parent, child in EDGES:
        _arrow(draw, centers[parent], centers[child], 28.0 if parent in SOURCE_CENTERS else 39.0, 40.0)

    for (name, (cx, cy)), bit in zip(SOURCE_CENTERS.items(), source_bits):
        half = 25
        draw.rounded_rectangle(
            (sx(cx - half), sx(cy - half), sx(cx + half), sx(cy + half)),
            radius=sx(7), fill=INK if bit else "white", outline=INK, width=sx(4),
        )
        _center_text(draw, (cx, cy - 41), name, _font(15), INK)

    for name, (cx, cy) in GATE_CENTERS.items():
        kind = gate_types[name]
        if name in NODE_COLORS:
            draw.ellipse(
                (sx(cx - 39), sx(cy - 39), sx(cx + 39), sx(cy + 39)),
                fill=NODE_COLORS[name], outline=INK, width=sx(2),
            )
        draw.ellipse(
            (sx(cx - 31), sx(cy - 31), sx(cx + 31), sx(cy + 31)),
            fill=GATE_COLORS[kind], outline=INK, width=sx(3),
        )
        text_fill = INK if kind == "AND" else "white"
        _center_text(draw, (cx, cy - 11), name, _font(14), text_fill)
        _center_text(draw, (cx, cy + 10), kind, _font(9), text_fill)

    _center_text(draw, (380, 620), "TRY ONE CARD  ·  RESET BEFORE EACH", _font(15), RED)
    for option, left, node, bit in zip(OPTIONS, CARD_LEFTS, option_nodes, forced_bits):
        top, right, bottom = 650, left + 169, 780
        color = NODE_COLORS[node]
        draw.rounded_rectangle(
            (sx(left), sx(top), sx(right), sx(bottom)), radius=sx(12),
            fill=PALE, outline=color, width=sx(7),
        )
        _center_text(draw, (left + 25, 680), option, _font(22), INK)
        _center_text(draw, (left + 70, 724), f"{node} =", _font(18), color)
        cx, cy, half = left + 132, 724, 25
        draw.rounded_rectangle(
            (sx(cx - half), sx(cy - half), sx(cx + half), sx(cy + half)),
            radius=sx(7), fill=INK if bit else "white", outline=color, width=sx(6),
        )

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
