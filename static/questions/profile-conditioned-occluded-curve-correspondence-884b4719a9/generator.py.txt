"""Occluded-curve correspondence: sampling, rendering, analytic gold and margin.

A single smooth trunk curve is interrupted by an opaque vertical band.  Three
coloured curves leave the band's far edge; exactly one of them is the trunk's
continuation.  Nothing in the image marks the answer locally: the decision is
the correspondence between one contour fragment and three candidate fragments,
resolved by continuing the trunk's height and direction across the gap.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 480
LEFT_X0 = 120
BAND_X0 = 300
BAND_X1 = 392
RIGHT_X1 = 580
LINE_PX = 5
BG_COLOR = "#f4f3ef"
BAND_COLOR = "#8b909a"
TRUNK_COLOR = "#161616"
CURVE_COLORS = {"red": "#d62728", "green": "#2ca02c", "blue": "#1f77b4"}
Y_LO = 34.0
Y_HI = HEIGHT - 34.0
MIN_SEPARATION_PX = 26.0
QUARANTINE_PX = 34.0
NEAR_BOUNDARY_EVERY = 7


def _hex(value: str) -> tuple[int, int, int]:
    return tuple(int(value[index : index + 2], 16) for index in (1, 3, 5))


def _trunk_y(trunk: list[float], x: float) -> float:
    y_e, slope, curv = trunk
    dx = x - BAND_X0
    return y_e + slope * dx + 0.5 * curv * dx * dx


def _curve_y(entry_y: float, slope: float, curv: float, x: float) -> float:
    dx = x - BAND_X1
    return entry_y + slope * dx + 0.5 * curv * dx * dx


def _predicted(trunk: list[float]) -> tuple[float, float]:
    """Height and direction of the trunk where it would leave the band."""
    y_e, slope, curv = trunk
    width = float(BAND_X1 - BAND_X0)
    return y_e + slope * width + 0.5 * curv * width * width, slope + curv * width


def _score(trunk: list[float], curve: list) -> float:
    predicted_y, predicted_slope = _predicted(trunk)
    width = float(BAND_X1 - BAND_X0)
    return abs(float(curve[1]) - predicted_y) + 0.5 * width * abs(float(curve[2]) - predicted_slope)


def _scores(scene: dict) -> list[tuple[float, str]]:
    trunk = [float(value) for value in scene["trunk"]]
    return sorted((_score(trunk, curve), str(curve[0])) for curve in scene["right_curves"])


def _in_frame(values: list[float]) -> bool:
    return all(Y_LO <= value <= Y_HI for value in values)


def _samples(curve: list) -> list[float]:
    return [
        _curve_y(float(curve[1]), float(curve[2]), float(curve[3]), float(x))
        for x in range(BAND_X1, RIGHT_X1 + 1, 4)
    ]


def sample_scene(seed: int) -> dict:
    """Deterministically sample one occluded-continuation scene."""
    rng = random.Random(seed)
    near_boundary = (seed % NEAR_BOUNDARY_EVERY) == 3
    width = float(BAND_X1 - BAND_X0)
    for _ in range(6000):
        y_e = rng.uniform(180.0, 300.0)
        slope = rng.choice((-1.0, 1.0)) * rng.uniform(0.42, 0.62)
        curv = rng.choice((-1.0, 1.0)) * rng.uniform(0.0004, 0.0016)
        trunk = [y_e, slope, curv]
        true_y, true_slope = _predicted(trunk)
        # Decoy 1 keeps the height at which the trunk met the band: a reader who
        # continues horizontally instead of following the direction is always wrong.
        flat_slope = -slope * rng.uniform(0.45, 0.95)
        flat = ["", y_e, flat_slope, rng.choice((-1.0, 1.0)) * rng.uniform(0.0004, 0.0016)]
        # Decoy 2 runs parallel to the true continuation: direction alone never decides.
        offset = rng.choice((-1.0, 1.0)) * (
            rng.uniform(30.5, 33.4) if near_boundary else rng.uniform(44.0, 72.0)
        )
        parallel = ["", true_y + offset, true_slope, curv]
        true_curve = ["", true_y, true_slope, curv]
        colours = list(CURVE_COLORS)
        rng.shuffle(colours)
        for curve, colour in zip((true_curve, flat, parallel), colours):
            curve[0] = colour
        right = [true_curve, flat, parallel]
        rng.shuffle(right)

        trunk_samples = [_trunk_y(trunk, float(x)) for x in range(LEFT_X0, BAND_X0 + 1, 4)]
        if not _in_frame(trunk_samples):
            continue
        tracks = [_samples(curve) for curve in right]
        if not all(_in_frame(track) for track in tracks):
            continue
        gaps = [
            min(abs(a - b) for a, b in zip(tracks[i], tracks[j]))
            for i in range(3)
            for j in range(i + 1, 3)
        ]
        if min(gaps) < MIN_SEPARATION_PX:
            continue
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "bg_color": BG_COLOR,
            "band_color": BAND_COLOR,
            "trunk_color": TRUNK_COLOR,
            "band_x0": BAND_X0,
            "band_x1": BAND_X1,
            "left_x0": LEFT_X0,
            "right_x1": RIGHT_X1,
            "line_px": LINE_PX,
            "trunk": [round(value, 6) for value in trunk],
            "right_curves": [
                [curve[0], round(curve[1], 6), round(curve[2], 6), round(curve[3], 6)]
                for curve in right
            ],
        }
        ranked = _scores(scene)
        gap = ranked[1][0] - ranked[0][0]
        if ranked[0][0] > 1e-2:
            continue
        if near_boundary and not (MIN_SEPARATION_PX + 4.0 <= gap < QUARANTINE_PX):
            continue
        if not near_boundary and gap < QUARANTINE_PX + 6.0:
            continue
        return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Paint the scene; the band is opaque and no two curves ever overlap."""
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), _hex(scene["bg_color"]))
    draw = ImageDraw.Draw(image)
    band_x0 = int(scene["band_x0"])
    band_x1 = int(scene["band_x1"])
    draw.rectangle(
        [band_x0, 0, band_x1 - 1, int(scene["height"]) - 1], fill=_hex(scene["band_color"])
    )
    trunk = [float(value) for value in scene["trunk"]]
    width_px = int(scene["line_px"])
    draw.line(
        [(x, _trunk_y(trunk, float(x))) for x in range(int(scene["left_x0"]), band_x0 + 1)],
        fill=_hex(scene["trunk_color"]),
        width=width_px,
    )
    for colour, entry_y, slope, curv in scene["right_curves"]:
        draw.line(
            [
                (x, _curve_y(float(entry_y), float(slope), float(curv), float(x)))
                for x in range(band_x1, int(scene["right_x1"]) + 1)
            ],
            fill=_hex(CURVE_COLORS[colour]),
            width=width_px,
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Colour of the curve that continues the trunk's height and direction."""
    return _scores(scene)[0][1]


def margin(scene: dict) -> float:
    """Top-gap: how much worse the best decoy fits the trunk's continuation."""
    ranked = _scores(scene)
    return round(ranked[1][0] - ranked[0][0], 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_PX)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The three right-hand curves are an unordered set: reordering is pixel-identical."""
    curves = [list(curve) for curve in scene["right_curves"]]
    rotated = dict(scene)
    rotated["right_curves"] = curves[1:] + curves[:1]
    reversed_scene = dict(scene)
    reversed_scene["right_curves"] = list(reversed(curves))
    return [("rotate_right_curve_order", rotated), ("reverse_right_curve_order", reversed_scene)]
