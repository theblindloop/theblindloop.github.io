"""Image-only inverse arm: rebuild the continuation graph from pixels.

Input is one rendered PNG and the public question contract.  Nothing else: no
latent scene, no renderer, no gold.  The arm locates the opaque band, fits the
trunk fragment left of it, extrapolates a single smooth continuation to the
band's far edge, fits each coloured fragment where it emerges, and returns the
colour whose height-and-direction mismatch is smallest.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_BAND = (139, 144, 154)
_TRUNK = (22, 22, 22)
_COLOURS = {"red": (214, 39, 40), "green": (44, 160, 44), "blue": (31, 119, 180)}
_TOL = 40
_ABSTAIN = "abstain"


def _mask(array: np.ndarray, colour: tuple[int, int, int], tol: int = _TOL) -> np.ndarray:
    reference = np.array(colour, dtype=np.int16)
    return (np.abs(array.astype(np.int16) - reference).max(axis=2) <= tol)


def _column_track(mask: np.ndarray, columns: range) -> tuple[list[float], list[float]]:
    xs: list[float] = []
    ys: list[float] = []
    for x in columns:
        rows = np.flatnonzero(mask[:, x])
        if rows.size:
            xs.append(float(x))
            ys.append(float(rows.mean()))
    return xs, ys


def _fit(xs: list[float], ys: list[float], origin: float) -> tuple[float, float] | None:
    if len(xs) < 24:
        return None
    shifted = np.asarray(xs, dtype=float) - origin
    coefficients = np.polyfit(shifted, np.asarray(ys, dtype=float), 2)
    return float(coefficients[2]), float(coefficients[1])


def decision_from_image(image: Image.Image) -> str:
    """Return "red", "green", "blue", or "abstain" when the evidence is absent."""
    array = np.asarray(image.convert("RGB"))
    height, width = array.shape[0], array.shape[1]
    band_columns = np.flatnonzero(_mask(array, _BAND).sum(axis=0) > 0.5 * height)
    if band_columns.size < 8:
        return _ABSTAIN
    band_x0 = int(band_columns[0])
    band_x1 = int(band_columns[-1]) + 1
    band_width = float(band_x1 - band_x0)

    trunk_mask = _mask(array, _TRUNK)
    trunk_mask[:, band_x0 - 4 :] = False
    trunk = _fit(*_column_track(trunk_mask, range(0, max(band_x0 - 4, 1))), origin=float(band_x1))
    if trunk is None:
        return _ABSTAIN
    predicted_y, predicted_slope = trunk

    scores: list[tuple[float, str]] = []
    for name, colour in _COLOURS.items():
        mask = _mask(array, colour)
        mask[:, : band_x1 + 3] = False
        columns = range(band_x1 + 3, min(band_x1 + 140, width))
        fitted = _fit(*_column_track(mask, columns), origin=float(band_x1))
        if fitted is None:
            continue
        entry_y, slope = fitted
        scores.append(
            (
                abs(entry_y - predicted_y) + 0.5 * band_width * abs(slope - predicted_slope),
                name,
            )
        )
    if len(scores) < 2:
        return _ABSTAIN
    scores.sort()
    if scores[1][0] - scores[0][0] < 8.0:
        return _ABSTAIN
    return scores[0][1]
