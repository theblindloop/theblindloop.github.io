"""Deterministic renderer for a prior-conflict chest drawer-count world.

The image contains one familiar chest with five possible horizontal drawer
slots.  A visible drawer front is a wide colored panel with one centered warm
handle.  ``drawer_count`` is the only analytic label input; the handle and the
front are both rendered from it, while slot order, colors, placement, and
scale are nuisance variation.
"""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SLOT_COUNT = 5
SUPERSAMPLE = 4

BACKGROUND_SWATCHES = (
    (247, 249, 246),
    (244, 247, 250),
    (250, 247, 242),
)
CHEST_SWATCHES = (
    (224, 231, 232),
    (231, 229, 224),
    (220, 231, 226),
)
FRAME_SWATCHES = (
    (48, 67, 91),
    (55, 72, 96),
    (62, 76, 95),
)
DRAWER_SWATCHES = (
    (174, 211, 220),
    (189, 205, 232),
    (191, 222, 199),
    (216, 195, 143),
    (222, 184, 205),
)
HANDLE_SWATCHES = (
    (88, 48, 24),
    (104, 54, 27),
    (76, 45, 34),
    (96, 58, 42),
)
LABEL_COLOR = (43, 59, 80)


def _rgb(value: list[int] | tuple[int, int, int]) -> tuple[int, int, int]:
    return int(value[0]), int(value[1]), int(value[2])


def _box(values: tuple[float, float, float, float], ss: int) -> list[int]:
    return [int(round(value * ss)) for value in values]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one deterministic scene with two through five visible drawers."""

    rng = random.Random(int(seed))
    drawer_count = 2 + (abs(int(seed)) % 4)
    compact = (abs(int(seed)) % 13) == 0

    body_left = round(rng.uniform(112.0, 136.0), 3)
    body_width = round(rng.uniform(250.0, 286.0), 3)
    body_top = round(rng.uniform(136.0, 151.0), 3)
    if compact:
        row_height = round(rng.uniform(29.0, 32.0), 3)
        row_gap = round(rng.uniform(6.0, 8.0), 3)
    else:
        row_height = round(rng.uniform(36.0, 45.0), 3)
        row_gap = round(rng.uniform(10.0, 15.0), 3)
    body_right = body_left + body_width
    body_bottom = body_top + SLOT_COUNT * row_height + (SLOT_COUNT - 1) * row_gap

    active_slots = sorted(rng.sample(range(SLOT_COUNT), drawer_count))
    drawer_fills = [list(rng.choice(DRAWER_SWATCHES)) for _ in range(SLOT_COUNT)]
    return {
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
        "background": list(rng.choice(BACKGROUND_SWATCHES)),
        "chest_fill": list(rng.choice(CHEST_SWATCHES)),
        "frame_fill": list(rng.choice(FRAME_SWATCHES)),
        "drawer_outline": list(rng.choice(FRAME_SWATCHES)),
        "handle_color": list(rng.choice(HANDLE_SWATCHES)),
        "label_color": list(LABEL_COLOR),
        "label_text": "CHEST",
        "body_left": body_left,
        "body_top": body_top,
        "body_width": round(body_width, 3),
        "row_height": row_height,
        "row_gap": row_gap,
        "drawer_count": drawer_count,
        "active_slots": active_slots,
        "drawer_fills": drawer_fills,
        "compact": compact,
    }


def counterfactual_scene(scene: dict[str, Any]) -> dict[str, Any]:
    """Toggle one visible slot while keeping the matched scene styling fixed."""

    twin = copy.deepcopy(scene)
    active = set(int(value) for value in twin["active_slots"])
    if int(twin["drawer_count"]) < SLOT_COUNT:
        inactive = [index for index in range(SLOT_COUNT) if index not in active]
        active.add(inactive[0])
    else:
        active.remove(max(active))
    twin["active_slots"] = sorted(active)
    twin["drawer_count"] = len(active)
    return twin


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the scene with supersampling for clean, inspectable boundaries."""

    ss = SUPERSAMPLE
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    image = Image.new("RGB", (width * ss, height * ss), _rgb(scene["background"]))
    draw = ImageDraw.Draw(image)

    left = float(scene["body_left"])
    top = float(scene["body_top"])
    body_width = float(scene["body_width"])
    right = left + body_width
    row_height = float(scene["row_height"])
    row_gap = float(scene["row_gap"])
    bottom = top + SLOT_COUNT * row_height + (SLOT_COUNT - 1) * row_gap
    frame = _rgb(scene["frame_fill"])
    outline = _rgb(scene["drawer_outline"])

    draw.text((int(24 * ss), int(42 * ss)), str(scene["label_text"]), fill=_rgb(scene["label_color"]), font=ImageFont.load_default())
    draw.rounded_rectangle(
        _box((left - 24.0, top - 21.0, right + 24.0, bottom + 30.0), ss),
        radius=22 * ss,
        fill=frame,
    )
    draw.rounded_rectangle(
        _box((left, top, right, bottom), ss),
        radius=13 * ss,
        fill=_rgb(scene["chest_fill"]),
        outline=outline,
        width=2 * ss,
    )

    active_slots = set(int(value) for value in scene["active_slots"][: int(scene["drawer_count"])])
    drawer_left = left + 17.0
    drawer_right = right - 17.0
    handle_x = (left + right) / 2.0
    for index in range(SLOT_COUNT):
        drawer_top = top + index * (row_height + row_gap)
        drawer_bottom = drawer_top + row_height
        if index in active_slots:
            draw.rounded_rectangle(
                _box((drawer_left, drawer_top, drawer_right, drawer_bottom), ss),
                radius=6 * ss,
                fill=_rgb(scene["drawer_fills"][index]),
                outline=outline,
                width=2 * ss,
            )
            handle_y = (drawer_top + drawer_bottom) / 2.0
            draw.rounded_rectangle(
                _box((handle_x - 15.0, handle_y - 4.5, handle_x + 15.0, handle_y + 4.5), ss),
                radius=4 * ss,
                fill=_rgb(scene["handle_color"]),
            )
        else:
            # Empty slots remain visible as pale recesses, but have no handle.
            draw.line(
                [(int(drawer_left * ss), int(((drawer_top + drawer_bottom) / 2.0) * ss)),
                 (int(drawer_right * ss), int(((drawer_top + drawer_bottom) / 2.0) * ss))],
                fill=outline,
                width=max(1, ss),
            )

    foot_y = bottom + 29.0
    draw.rounded_rectangle(_box((left + 26.0, bottom + 17.0, left + 66.0, foot_y), ss), radius=5 * ss, fill=frame)
    draw.rounded_rectangle(_box((right - 66.0, bottom + 17.0, right - 26.0, foot_y), ss), radius=5 * ss, fill=frame)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible drawer-front count from the sole label input."""

    return str(int(scene["drawer_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Report the smallest handle/slot clearance in pixels."""

    row_height = float(scene["row_height"])
    row_gap = float(scene["row_gap"])
    body_width = float(scene["body_width"])
    # The handle height is 9 px and the front has 17 px side clearance.
    values = [row_height - 12.0, row_height + row_gap - 9.0, body_width - 42.0]
    return round(min(values), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Quarantine compact rows whose finite-resolution separation is small."""

    return margin(scene) < 20.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No unclaimed render-invariant latent transform is used in this world."""

    return []
