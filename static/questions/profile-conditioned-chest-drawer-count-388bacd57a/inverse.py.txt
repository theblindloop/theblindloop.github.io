"""Independent PNG-only inverse for the chest drawer-count question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "chest-drawer-count-oracle-0.1.0"


def _components(mask: np.ndarray, minimum: int = 55) -> list[tuple[np.ndarray, np.ndarray]]:
    """Return connected components without using any scene or renderer state."""

    height, width = mask.shape
    remaining = mask.copy()
    components: list[tuple[np.ndarray, np.ndarray]] = []
    while remaining.any():
        y0, x0 = np.argwhere(remaining)[0]
        queue = deque([(int(y0), int(x0))])
        remaining[y0, x0] = False
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    yy = y + dy
                    xx = x + dx
                    if 0 <= yy < height and 0 <= xx < width and remaining[yy, xx]:
                        remaining[yy, xx] = False
                        queue.append((yy, xx))
        if len(ys) >= minimum:
            components.append((np.asarray(ys), np.asarray(xs)))
    return components


def decision_from_image(image: Image.Image) -> str:
    """Count isolated warm handle components from the final raster only.

    The renderer uses navy for the frame, feet, and label, while handles are
    the only small warm-dark marks.  A broad color predicate accepts the
    public handle palette and antialiased interiors, and geometric filters
    reject text, frame strokes, and the large drawer fills.
    """

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    red = arr[..., 0]
    green = arr[..., 1]
    blue = arr[..., 2]
    warm_dark = (
        (red > 50.0)
        & (green > 20.0)
        & (blue < 75.0)
        & (red > green * 1.22)
        & (green > blue * 1.08)
    )

    count = 0
    for ys, xs in _components(warm_dark, minimum=55):
        x_span = int(xs.max() - xs.min() + 1)
        y_span = int(ys.max() - ys.min() + 1)
        area = len(xs)
        aspect = x_span / max(1, y_span)
        center_y = float(ys.mean())
        if 10 <= x_span <= 42 and 4 <= y_span <= 20 and 1.2 <= aspect <= 9.0 and 95.0 <= center_y <= 430.0:
            count += 1
    return str(count)
