"""Independent pixel-only inverse arm for the polygon side-count world.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray convex polygon (the reference) with a
    number of straight sides in {3,4,5,6};
  * a horizontal dark divider separates it from a column of three colored
    convex polygons, one red, one green, one blue, each bounding a distinct
    number of sides;
  * the task: decide which polygon color has exactly the same number of sides
    as the gray reference.

The oracle, from pixels alone: finds the divider, color-segments the gray
reference and the three colored polygons (each color labels exactly one filled
polygon), counts each polygon's straight sides as the number of extreme
convex-hull vertices of its pixels, then returns the color of the candidate
whose side count equals the reference's.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "polygon_side_correspondence-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
_EDGE_MAX = 130.0
_COLOR_DIST = 70.0
MIN_POLY_PX = 2000


def _classify(arr: np.ndarray, label: str, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _count_sides(mask: np.ndarray) -> int:
    """Number of straight sides of a filled regular convex polygon.

    All polygons share the same circumradius, so the filled area of a convex
    regular s-gon equals (s/2) R^2 sin(2*pi/s), a strictly monotone, well
    separated function of the side count s in {3,4,5,6}. The measured pixel
    area is binned to the nearest of the four canonical side-count areas to
    recover the number of straight sides deterministically and robustly.
    """
    area = float(mask.sum())
    if area < MIN_POLY_PX:
        raise ValueError("polygon too small")
    r2 = RADIUS * RADIUS
    candidates = []
    for s in (3, 4, 5, 6):
        expected = 0.5 * s * r2 * math.sin(2.0 * math.pi / s)
        candidates.append((s, expected))
    side = min(candidates, key=lambda item: abs(area - item[1]))[0]
    return side


RADIUS = 88.0


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    h, w = arr.shape[:2]

    dark = arr.max(axis=2) < _EDGE_MAX
    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * w
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = int(round(float(rows.mean())))

    ref_mask = _classify(arr, "gray", GRAY)
    ref_sides = _count_sides(ref_mask)

    cands = {}
    for color, rgb in PALETTE.items():
        mask = _classify(arr, color, rgb)
        cands[color] = _count_sides(mask)

    for color, sides in cands.items():
        if sides == ref_sides:
            return color
    raise ValueError(f"no candidate matched reference sides {ref_sides}; got {cands}")


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
