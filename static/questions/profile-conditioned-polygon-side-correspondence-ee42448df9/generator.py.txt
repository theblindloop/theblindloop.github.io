"""Polygon side-count correspondence renderer (paper-semantic family).

A cross-panel shape-correspondence task. A neutral gray convex polygon is shown
at the top (the reference); below a divider sit three colored convex polygons
A, B, C, each bound to a distinct color (red / green / blue) and each with a
distinct number of straight sides. Exactly one candidate has the same number of
sides as the reference. The question asks for that candidate's color.

The pixel-only inverse arm must, for every panel, recover the convex polygon,
count its straight sides (its convex-hull corner count), establish the
cross-panel correspondence by side-count equality, and only then read out the
bound color of the single matched polygon.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_sides : number of sides of the neutral gray reference polygon (3..6).
    cand_sides: length-3 list of candidate side counts; exactly one equals
                ``ref_sides``, the other two are distinct and different.
    cand_colors: the fixed 3-color palette bound one-to-one to the candidates.
                 Each candidate polygon is drawn in its bound color.
    poly_rot   : a per-polygon rotation (degrees) applied to every polygon so
                 that side count is the only shape cue and absolute vertex
                 orientation is counterbalanced.
    width, height, divider_y, ref_center, cand_centers, radius : fixed geometry.

The analytic gold is the color of the candidate whose side count equals the
reference's. Because the reference side count, each candidate side count, and
the bound colors are all rendered verbatim, the gold is recoverable from the
pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "polygon_side_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (80, 80, 80)
GRAY = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

WIDTH = 900
HEIGHT = 760
DIVIDER_Y = 200
DIVIDER_THICK = 3
RADIUS = 88.0
REF_CENTER = (450, 108)
CAND_Y = [316, 468, 620]
CAND_X = 450

_MARGIN_FLOOR = 1.0  # side counts are distinct integers, so margin >= 1 always


def _regular_polygon(center, radius: float, sides: int, angle_deg: float) -> list[tuple[float, float]]:
    """Return a regular convex polygon with ``sides`` straight sides."""
    cx, cy = center
    pts = []
    for k in range(sides):
        a = math.radians(angle_deg + 360.0 * k / sides)
        pts.append((cx + radius * math.cos(a), cy + radius * math.sin(a)))
    return pts


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_sides = rng.choice([3, 4, 5, 6])
    pool = [3, 4, 5, 6]
    pool.remove(ref_sides)
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_sides = [ref_sides] + distractors
    rng.shuffle(cand_sides)

    rotations = [rng.uniform(0.0, 12.0) for _ in range(4)]

    return {
        "ref_sides": ref_sides,
        "cand_sides": cand_sides,
        "cand_colors": list(PALETTE.keys()),
        "poly_rot": rotations,
        "ref_center": list(REF_CENTER),
        "cand_centers": [[CAND_X, y] for y in CAND_Y],
        "radius": RADIUS,
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_polygon(draw, center, sides: int, angle_deg: float, color) -> None:
    draw.polygon(_regular_polygon(center, RADIUS, sides, angle_deg), fill=color)


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution (crisp edges)."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_polygon(
        draw, tuple(scene["ref_center"]), scene["ref_sides"], scene["poly_rot"][0], GRAY
    )
    for i, center in enumerate(scene["cand_centers"]):
        color = scene["cand_colors"][i]
        _draw_polygon(
            draw, tuple(center), scene["cand_sides"][i], scene["poly_rot"][1 + i], PALETTE[color]
        )
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the candidate with the reference sides."""
    sides = scene["cand_sides"]
    idx = sides.index(scene["ref_sides"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute side-count difference between the
    reference polygon and any distractor candidate. Counts are distinct
    integers, so this is at least 1.0 for every produced scene."""
    ref = scene["ref_sides"]
    diffs = [
        abs(ref - s) for s in scene["cand_sides"] if s != ref
    ]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene (never produced: distinct side counts
    always place every distractor at least one side away)."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference side count, each
    candidate side count, and the bound colors) is rendered verbatim, so the
    raster uniquely determines the gold; no pixel-identical transform exists."""
    return []
