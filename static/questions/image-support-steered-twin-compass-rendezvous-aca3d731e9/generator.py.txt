"""Sampler and renderer for two independent compass-state histories."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
GOLD = (242, 190, 65)
CENTER = (57, 92, 145)
DARK = (31, 39, 53)
PALE = (226, 230, 235)
PALETTES = (
    ((55, 184, 190), (231, 112, 103)),
    ((43, 169, 183), (224, 123, 91)),
    ((68, 178, 168), (222, 105, 126)),
)


def _apply(state: int, operations: str) -> int:
    for operation in operations:
        if operation == "C":
            state = (state + 1) % 4
        elif operation == "M":
            state = (0, 3, 2, 1)[state]
        else:
            raise ValueError("operations must contain C or M")
    return state


def _start_for(target: int, operations: str) -> int:
    matches = [state for state in range(4) if _apply(state, operations) == target]
    if len(matches) != 1:
        raise ValueError("operation history is not invertible")
    return matches[0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    histories = ("CCM", "CMC", "MCC", "CMM", "MCM", "MMC")
    left_moves = rng.choice(histories)
    right_moves = rng.choice(histories)
    left_start = rng.randrange(4)
    left_final = _apply(left_start, left_moves)
    should_match = bool(rng.randrange(2))
    right_final = left_final if should_match else rng.choice([v for v in range(4) if v != left_final])
    right_start = _start_for(right_final, right_moves)
    return {
        "left_start": left_start,
        "right_start": right_start,
        "left_moves": left_moves,
        "right_moves": right_moves,
        "layout_variant": [rng.randint(-5, 5), rng.randint(-5, 5)],
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def analytic_gold(scene: dict) -> str:
    left = _apply(int(scene["left_start"]), str(scene["left_moves"]))
    right = _apply(int(scene["right_start"]), str(scene["right_moves"]))
    return "yes" if left == right else "no"


def margin(scene: dict) -> float:
    # One discrete visible cue can change a terminal state; the runner-up class
    # is therefore one valid cue mutation away.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_compass(draw: ImageDraw.ImageDraw, cx: int, cy: int, start: int) -> None:
    radius = 38
    positions = ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy))
    draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), fill=CENTER)
    for state, (x, y) in enumerate(positions):
        fill = GOLD if state == start else PALE
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=fill, outline=DARK, width=3)


def render(scene: dict) -> Image.Image:
    left_start = int(scene["left_start"])
    right_start = int(scene["right_start"])
    left_moves = str(scene["left_moves"])
    right_moves = str(scene["right_moves"])
    shift_x, shift_y = (int(v) for v in scene["layout_variant"])
    palette = PALETTES[int(scene["palette_variant"])]
    if left_start not in range(4) or right_start not in range(4):
        raise ValueError("start states must be in 0..3")
    if any(len(moves) != 3 or any(v not in "CM" for v in moves) for moves in (left_moves, right_moves)):
        raise ValueError("each history must contain three C/M operations")

    image = Image.new("RGB", (CANVAS, CANVAS), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    title = ImageFont.truetype(str(FONT_PATH), 22)
    label = ImageFont.truetype(str(FONT_PATH), 13)
    chip_font = ImageFont.truetype(str(FONT_PATH), 18)
    draw.text((CANVAS // 2, 43), "TWIN COMPASS RENDEZVOUS", fill=DARK, anchor="mm", font=title)
    draw.text((CANVAS // 2, 74), "GOLD = START    READ EACH STACK  ↓", fill=(71, 77, 88), anchor="mm", font=label)

    centers = (125 + shift_x, 387 + shift_x)
    cy = 178 + shift_y
    for side, (cx, start, operations) in enumerate(
        ((centers[0], left_start, left_moves), (centers[1], right_start, right_moves))
    ):
        draw.text((cx, 111 + shift_y), "LEFT" if side == 0 else "RIGHT", fill=DARK, anchor="mm", font=label)
        _draw_compass(draw, cx, cy, start)
        for index, operation in enumerate(operations):
            chip_y = 258 + shift_y + index * 44
            fill = palette[0 if operation == "C" else 1]
            draw.rounded_rectangle((cx - 34, chip_y - 14, cx + 34, chip_y + 14), radius=7, fill=fill)
            draw.text((cx, chip_y - 1), "C" if operation == "C" else "M", fill=DARK, anchor="mm", font=chip_font)

    draw.text((CANVAS // 2, 426), "CYAN C: 1 STEP CLOCKWISE", fill=palette[0], anchor="mm", font=label)
    draw.text((CANVAS // 2, 449), "CORAL M: MIRROR LEFT ↔ RIGHT", fill=palette[1], anchor="mm", font=label)
    draw.text((CANVAS // 2, 481), "SAME FINAL STATE?", fill=DARK, anchor="mm", font=title)
    return image
