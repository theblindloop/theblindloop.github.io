"""Independent PNG-only inverse for twin compass rendezvous."""

from __future__ import annotations

import numpy as np
from PIL import Image

_GOLD = np.array((242, 190, 65), dtype=np.int16)
_CENTER = np.array((57, 92, 145), dtype=np.int16)
_CYANS = np.array(((55, 184, 190), (43, 169, 183), (68, 178, 168)), dtype=np.int16)
_CORALS = np.array(((231, 112, 103), (224, 123, 91), (222, 105, 126)), dtype=np.int16)


def _mask_near(array: np.ndarray, colors: np.ndarray, tolerance: float = 24.0) -> np.ndarray:
    colors = np.atleast_2d(colors)
    delta = array[:, :, None, :].astype(np.int32) - colors[None, None, :, :].astype(np.int32)
    return np.min(np.sum(delta * delta, axis=3), axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int, maximum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            py, px = stack.pop()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if minimum <= len(points) <= maximum:
            values = np.asarray(points, dtype=np.float64)
            output.append((float(values[:, 1].mean()), float(values[:, 0].mean()), len(points)))
    return output


def _state(gold: tuple[float, float, int], center: tuple[float, float, int]) -> int:
    dx, dy = gold[0] - center[0], gold[1] - center[1]
    distance = float(np.hypot(dx, dy))
    if not 32.0 <= distance <= 44.0:
        raise ValueError("start marker is not on a compass state")
    if abs(dx) > abs(dy) * 1.8:
        return 1 if dx > 0 else 3
    if abs(dy) > abs(dx) * 1.8:
        return 2 if dy > 0 else 0
    raise ValueError("ambiguous compass start")


def _apply(state: int, operations: list[str]) -> int:
    for operation in operations:
        if operation == "C":
            state = (state + 1) % 4
        else:
            state = (0, 3, 2, 1)[state]
    return state


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 400 or array.shape[1] < 400:
        raise ValueError("invalid canvas")
    centers = sorted(_components(_mask_near(array, _CENTER), 120, 260), key=lambda p: p[0])
    starts = sorted(_components(_mask_near(array, _GOLD), 280, 650), key=lambda p: p[0])
    cyan = _components(_mask_near(array, _CYANS), 1200, 2400)
    coral = _components(_mask_near(array, _CORALS), 1200, 2400)
    if len(centers) != 2 or len(starts) != 2:
        raise ValueError("both complete compass starts are required")
    if len(cyan) + len(coral) != 6:
        raise ValueError("exactly six complete move chips are required")
    if not 220.0 <= centers[1][0] - centers[0][0] <= 280.0:
        raise ValueError("compass separation is invalid")

    finals = []
    tagged = [(x, y, "C") for x, y, _ in cyan] + [(x, y, "M") for x, y, _ in coral]
    for center, start in zip(centers, starts, strict=True):
        if abs(start[0] - center[0]) > 46.0:
            raise ValueError("start is not bound to its compass")
        local = sorted((item for item in tagged if abs(item[0] - center[0]) < 50.0), key=lambda item: item[1])
        if len(local) != 3 or any(local[i + 1][1] - local[i][1] < 35.0 for i in range(2)):
            raise ValueError("each compass requires three ordered move chips")
        finals.append(_apply(_state(start, center), [item[2] for item in local]))
    return "yes" if finals[0] == finals[1] else "no"
