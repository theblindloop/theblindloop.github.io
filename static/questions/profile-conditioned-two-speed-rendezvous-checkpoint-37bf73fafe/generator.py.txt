"""Deterministic renderer and analytic arm for two-speed route rendezvous."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 460
BACKGROUND = "#fbfaf6"
LINK = "#26364a"
OUTLINE = "#101820"
GRAY = "#aeb8c2"
COLORS = {
    "green": "#20a464",
    "red": "#db3b4b",
    "orange": "#f29f30",
    "purple": "#8b5bd6",
    "blue": "#268bd2",
}
ANSWER_COLORS = ("orange", "purple", "blue")


def _base_nodes() -> list[list[int]]:
    xs = [110, 270, 430, 590]
    ys = [82, 176, 270, 364]
    points: list[list[int]] = []
    for row, y in enumerate(ys):
        order = xs if row % 2 == 0 else list(reversed(xs))
        points.extend([[x, y] for x in order])
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    links = (9, 12, 15)[seed % 3]
    nodes = _base_nodes()[: links + 1]

    # Layout nuisance variation changes physical position/orientation, not graph order.
    dx = rng.choice((-22, -11, 0, 11, 22))
    dy = rng.choice((-10, 0, 10))
    mirror_x = bool((seed // 3) % 2)
    mirror_y = bool((seed // 7) % 2)
    for point in nodes:
        x, y = point
        if mirror_x:
            x = WIDTH - x
        if mirror_y:
            y = HEIGHT - y
        point[:] = [x + dx, y + dy]

    edges = [[i, i + 1] for i in range(links)]
    meeting = 2 * links // 3
    allowed = [i for i in range(1, links) if i != meeting]
    # Always include one closest runner-up (discrete mismatch 3), plus a remote foil.
    near = meeting + rng.choice((-1, 1))
    if near not in allowed:
        near = meeting - 1
    remote_pool = [i for i in allowed if i != near and abs(3 * i - 2 * links) >= 6]
    remote = rng.choice(remote_pool)
    checkpoint_indices = [meeting, near, remote]

    # Six permutations across consecutive seeds balance answer identity and foil colors.
    permutations = (
        ("orange", "purple", "blue"),
        ("purple", "blue", "orange"),
        ("blue", "orange", "purple"),
        ("orange", "blue", "purple"),
        ("purple", "orange", "blue"),
        ("blue", "purple", "orange"),
    )
    palette_order = permutations[seed % len(permutations)]
    checkpoints = {color: index for color, index in zip(palette_order, checkpoint_indices)}
    return {
        "nodes": nodes,
        "edges": edges,
        "start_green": 0,
        "start_red": links,
        "checkpoints": checkpoints,
    }


def _distances(node_count: int, edges: list[list[int]], source: int) -> list[int]:
    graph = [[] for _ in range(node_count)]
    for left, right in edges:
        graph[left].append(right)
        graph[right].append(left)
    distances = [-1] * node_count
    distances[source] = 0
    queue = deque([source])
    while queue:
        node = queue.popleft()
        for other in graph[node]:
            if distances[other] < 0:
                distances[other] = distances[node] + 1
                queue.append(other)
    return distances


def _scores(scene: dict) -> dict[str, int]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    green = scene["start_green"]
    red = scene["start_red"]
    checkpoints = scene["checkpoints"]
    from_green = _distances(len(nodes), edges, green)
    from_red = _distances(len(nodes), edges, red)
    return {
        color: abs(from_green[int(index)] - 2 * from_red[int(index)])
        for color, index in checkpoints.items()
    }


def analytic_gold(scene: dict) -> str:
    # Keep the accessed keys explicit for the protocol's rendered-field audit.
    _ = scene["nodes"], scene["edges"], scene["start_green"], scene["start_red"], scene["checkpoints"]
    ranked = sorted(_scores(scene).items(), key=lambda item: (item[1], item[0]))
    return ranked[0][0]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values())
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    values = sorted(_scores(scene).values())
    return values[0] != 0 or values[1] - values[0] < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_serialization", twin)]


def _disk(draw: ImageDraw.ImageDraw, xy: tuple[int, int], radius: int, fill: str, width: int = 3) -> None:
    x, y = xy
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=OUTLINE, width=width)


def render(scene: dict) -> Image.Image:
    nodes = [tuple(map(int, point)) for point in scene["nodes"]]
    edges = scene["edges"]
    green = int(scene["start_green"])
    red = int(scene["start_red"])
    checkpoints = {str(color): int(index) for color, index in scene["checkpoints"].items()}

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        draw.line((nodes[int(left)], nodes[int(right)]), fill=LINK, width=8)

    by_index = {index: color for color, index in checkpoints.items()}
    for index, point in enumerate(nodes):
        if index == green:
            _disk(draw, point, 18, COLORS["green"], width=4)
            # Two visible pips encode two links per round.
            x, y = point
            draw.ellipse((x - 9, y - 3, x - 3, y + 3), fill="white")
            draw.ellipse((x + 3, y - 3, x + 9, y + 3), fill="white")
        elif index == red:
            _disk(draw, point, 18, COLORS["red"], width=4)
            x, y = point
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill="white")
        elif index in by_index:
            _disk(draw, point, 16, COLORS[by_index[index]], width=4)
        else:
            _disk(draw, point, 11, GRAY, width=3)
    return image
