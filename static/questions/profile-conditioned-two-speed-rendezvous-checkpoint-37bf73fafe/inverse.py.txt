"""Independent image-only inverse arm.

It segments node fills and dark-link components from the final RGB raster,
assigns each link component to its nearest two node centers, reconstructs the
graph, and compares bidirectional hop clocks. It has no scene input.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "green": (32, 164, 100),
    "red": (219, 59, 75),
    "orange": (242, 159, 48),
    "purple": (139, 91, 214),
    "blue": (38, 139, 210),
    "gray": (174, 184, 194),
    "link": (38, 54, 74),
}


def _components(mask: np.ndarray, minimum: int = 8) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum:
            groups.append(np.asarray(pixels, dtype=np.int32))
    return groups


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)


def _one_center(arr: np.ndarray, name: str) -> tuple[float, float]:
    groups = _components(_mask(arr, PALETTE[name]), minimum=40)
    if len(groups) != 1:
        raise ValueError(f"expected one {name} node, found {len(groups)}")
    yx = groups[0].mean(axis=0)
    return float(yx[1]), float(yx[0])


def _all_nodes(arr: np.ndarray) -> tuple[list[tuple[float, float]], dict[str, int]]:
    nodes: list[tuple[float, float]] = []
    labels: dict[str, int] = {}
    for name in ("green", "red", "orange", "purple", "blue"):
        center = _one_center(arr, name)
        labels[name] = len(nodes)
        nodes.append(center)
    gray_groups = _components(_mask(arr, PALETTE["gray"]), minimum=40)
    for group in gray_groups:
        yx = group.mean(axis=0)
        nodes.append((float(yx[1]), float(yx[0])))
    return nodes, labels


def _recover_edges(arr: np.ndarray, nodes: list[tuple[float, float]]) -> list[tuple[int, int]]:
    link_groups = _components(_mask(arr, PALETTE["link"]), minimum=40)
    edges: set[tuple[int, int]] = set()
    for group in link_groups:
        # Distance from each detected node center to the closest pixel of this
        # independently segmented straight-link component.
        ys = group[:, 0].astype(float)
        xs = group[:, 1].astype(float)
        distances = []
        for index, (cx, cy) in enumerate(nodes):
            distances.append((float(np.min((xs - cx) ** 2 + (ys - cy) ** 2)), index))
        distances.sort()
        left, right = sorted((distances[0][1], distances[1][1]))
        if distances[1][0] > 32.0 ** 2 or left == right:
            raise ValueError("link component has no two nearby node endpoints")
        edges.add((left, right))
    if len(edges) != len(nodes) - 1:
        raise ValueError(f"recovered {len(edges)} links for {len(nodes)} chain nodes")
    return sorted(edges)


def _bfs(size: int, edges: list[tuple[int, int]], source: int) -> list[int]:
    graph = [[] for _ in range(size)]
    for left, right in edges:
        graph[left].append(right)
        graph[right].append(left)
    distances = [-1] * size
    distances[source] = 0
    queue = deque([source])
    while queue:
        node = queue.popleft()
        for other in graph[node]:
            if distances[other] < 0:
                distances[other] = distances[node] + 1
                queue.append(other)
    if any(value < 0 for value in distances):
        raise ValueError("segmented graph is disconnected")
    return distances


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    nodes, labels = _all_nodes(arr)
    edges = _recover_edges(arr, nodes)
    from_green = _bfs(len(nodes), edges, labels["green"])
    from_red = _bfs(len(nodes), edges, labels["red"])
    scores = {
        color: abs(from_green[labels[color]] - 2 * from_red[labels[color]])
        for color in ("orange", "purple", "blue")
    }
    ranked = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    if ranked[0][1] != 0 or ranked[1][1] - ranked[0][1] < 3:
        return "__abstain__"
    return ranked[0][0]
