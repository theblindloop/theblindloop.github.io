"""Independent image-only inverse for paired route multiplicity.

The oracle segments colored rings and dark strokes, labels stroke components,
and counts only components that reach both a blue and an orange ring in the
same window.  It accepts no scene data and contains no renderer import.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BLUE = np.array((37, 99, 235), dtype=np.int16)
ORANGE = np.array((234, 88, 12), dtype=np.int16)
DARK = np.array((31, 41, 55), dtype=np.int16)


class OracleAbstention(ValueError):
    pass


def _near(arr: np.ndarray, color: np.ndarray, radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = deque([(sy, sx)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int16))
    return output


def _ring_centers(mask: np.ndarray, name: str) -> list[tuple[float, float, int]]:
    groups = _components(mask, minimum=180)
    if len(groups) != 2:
        raise OracleAbstention(f"expected two intact {name} rings, found {len(groups)}")
    centers = []
    for pts in groups:
        centers.append((float(pts[:, 1].mean()), float(pts[:, 0].mean()), len(pts)))
    centers.sort()
    if centers[1][0] - centers[0][0] < 120:
        raise OracleAbstention(f"{name} rings are not separated into two windows")
    return centers


def _touches(points: np.ndarray, center: tuple[float, float, int]) -> bool:
    cx, cy = center[:2]
    dy = points[:, 0].astype(np.float32) - cy
    dx = points[:, 1].astype(np.float32) - cx
    return bool(np.min(dx * dx + dy * dy) <= 21.0 * 21.0)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[:2] != (512, 512):
        raise OracleAbstention(f"unexpected raster size {arr.shape[:2]}")
    blue = _ring_centers(_near(arr, BLUE, 28.0), "blue")
    orange = _ring_centers(_near(arr, ORANGE, 28.0), "orange")
    for index in range(2):
        if abs(blue[index][0] - orange[index][0]) > 14:
            raise OracleAbstention("start and finish rings do not form vertical windows")

    strokes = _components(_near(arr, DARK, 24.0), minimum=80)
    counts = [0, 0]
    associated = [0, 0]
    for points in strokes:
        mean_x = float(points[:, 1].mean())
        panel = 0 if abs(mean_x - blue[0][0]) < abs(mean_x - blue[1][0]) else 1
        reaches_blue = _touches(points, blue[panel])
        reaches_orange = _touches(points, orange[panel])
        if reaches_blue or reaches_orange:
            associated[panel] += 1
        if reaches_blue and reaches_orange:
            counts[panel] += 1

    if associated != [3, 3]:
        raise OracleAbstention(f"expected three one-or-two-ended strokes per window, got {associated}")
    if any(value not in (1, 2) for value in counts) or counts[0] == counts[1]:
        raise OracleAbstention(f"route counts are unsupported or tied: {counts}")
    return "left" if counts[0] > counts[1] else "right"
