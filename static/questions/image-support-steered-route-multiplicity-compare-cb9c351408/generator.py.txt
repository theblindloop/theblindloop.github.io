"""Deterministic renderer for paired route-multiplicity comparison."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 250, 252)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (210, 218, 228)
ROUTE = (31, 41, 55)
BLUE = (37, 99, 235)
ORANGE = (234, 88, 12)
PANEL_CENTERS = (145, 367)
START_Y = 194
FINISH_Y = 318
MARKER_RADIUS = 18
ROUTE_WIDTH = 6
LANES = (-1, 0, 1)


def _routes(rng: random.Random, connecting: int) -> list[dict]:
    full_lanes = set(rng.sample(list(LANES), connecting))
    remaining = [lane for lane in LANES if lane not in full_lanes]
    rng.shuffle(remaining)
    routes: list[dict] = []
    for lane in LANES:
        if lane in full_lanes:
            kind = "full"
        else:
            # A one-ended stroke has almost the same ink as a full route but a
            # clearly visible gap at one marker.  Both kinds occur across scenes.
            kind = "start" if remaining.index(lane) % 2 == 0 else "finish"
            if len(remaining) == 1 and rng.randrange(2):
                kind = "finish"
        routes.append(
            {
                "lane": lane,
                "kind": kind,
                "bend_a": rng.choice([-3, -2, 2, 3]),
                "bend_b": rng.choice([-3, -2, 2, 3]),
            }
        )
    rng.shuffle(routes)
    return routes


def sample_scene(seed: int) -> dict:
    """Sample balanced left/right answers with randomized lane ownership."""

    rng = random.Random(int(seed))
    left_more = int(seed) % 2 == 0
    left_count, right_count = ((2, 1) if left_more else (1, 2))
    return {
        "left_routes": _routes(rng, left_count),
        "right_routes": _routes(rng, right_count),
    }


def _attachment_y(dx: int, top: bool) -> int:
    radial = int(round(math.sqrt(MARKER_RADIUS * MARKER_RADIUS - dx * dx)))
    return START_Y + radial if top else FINISH_Y - radial


def _points(cx: int, route: dict) -> list[tuple[int, int]]:
    lane = int(route["lane"])
    attach_dx = lane * 12
    lane_x = cx + lane * 24
    upper = (cx + attach_dx, _attachment_y(attach_dx, True))
    lower = (cx + attach_dx, _attachment_y(attach_dx, False))
    middle = [
        (lane_x, 229),
        (lane_x + int(route["bend_a"]), 256),
        (lane_x + int(route["bend_b"]), 283),
    ]
    kind = route["kind"]
    if kind == "full":
        return [upper, *middle, lower]
    if kind == "start":
        # The short sideways foot roughly equalizes ink while leaving a gap.
        return [upper, *middle, (lane_x + 9 * (-1 if lane > 0 else 1), 294)]
    if kind == "finish":
        return [(lane_x + 9 * (1 if lane < 0 else -1), 218), *middle, lower]
    raise ValueError(f"unknown route kind {kind!r}")


def _draw_marker(draw: ImageDraw.ImageDraw, cx: int, cy: int, color: tuple[int, int, int]) -> None:
    r = MARKER_RADIUS
    draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=PANEL_FILL, outline=color, width=6)


def _draw_panel(draw: ImageDraw.ImageDraw, cx: int, routes: list[dict]) -> None:
    draw.rounded_rectangle(
        (cx - 50, 166, cx + 50, 346), radius=18, fill=PANEL_FILL, outline=PANEL_EDGE, width=3
    )
    for route in routes:
        draw.line(_points(cx, route), fill=ROUTE, width=ROUTE_WIDTH, joint="curve")
    _draw_marker(draw, cx, START_Y, BLUE)
    _draw_marker(draw, cx, FINISH_Y, ORANGE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, PANEL_CENTERS[0], scene["left_routes"])
    _draw_panel(draw, PANEL_CENTERS[1], scene["right_routes"])
    return image


def _connecting(routes: list[dict]) -> int:
    return sum(item["kind"] == "full" for item in routes)


def analytic_gold(scene: dict) -> str:
    left = _connecting(scene["left_routes"])
    right = _connecting(scene["right_routes"])
    if left == right:
        raise ValueError("tied route counts are quarantined and have no forced answer")
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    return float(abs(_connecting(scene["left_routes"]) - _connecting(scene["right_routes"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Route draw-order is a genuine pixel-identical latent alias."""

    twin = copy.deepcopy(scene)
    twin["left_routes"] = list(reversed(twin["left_routes"]))
    twin["right_routes"] = list(reversed(twin["right_routes"]))
    return [("reverse_same_color_route_draw_order", twin)]
