"""Independent image-only inverse arm for panel-half exchange.

This module deliberately contains its own palette and image parsing.  It has no
access to scene data or forward-world modules.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CELL_COLORS = np.array(((25, 157, 190), (198, 65, 145), (65, 160, 88), (231, 139, 45)), dtype=np.uint8)
BORDERS = {
    "red": (215, 63, 55),
    "blue": (50, 99, 201),
    "gold": (220, 166, 32),
}
REFERENCE_BORDER = (105, 112, 122)


def _bbox_for_color(a: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    ys, xs = np.where(np.all(a == np.array(rgb, dtype=np.uint8), axis=2))
    if len(xs) < 100:
        raise ValueError("required card border is absent")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indexes = np.flatnonzero(values)
    if len(indexes) == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = last = int(indexes[0])
    for value in indexes[1:]:
        value = int(value)
        if value != last + 1:
            groups.append((start, last))
            start = value
        last = value
    groups.append((start, last))
    return groups


def _read_board(a: np.ndarray, bbox: tuple[int, int, int, int]) -> tuple[tuple[int, ...], ...]:
    x0, y0, x1, y1 = bbox
    crop = a[y0 + 5 : y1 - 4, x0 + 5 : x1 - 4]
    classified = np.full(crop.shape[:2], -1, dtype=np.int8)
    for index, rgb in enumerate(CELL_COLORS):
        classified[np.all(crop == rgb, axis=2)] = index

    col_runs = _runs(np.sum(classified >= 0, axis=0) > 40)
    row_runs = _runs(np.sum(classified >= 0, axis=1) > 80)
    if len(col_runs) != 6 or len(row_runs) != 4:
        raise ValueError(f"could not recover 4x6 cell lattice: {len(row_runs)}x{len(col_runs)}")
    board = []
    for ya, yb in row_runs:
        row = []
        for xa, xb in col_runs:
            patch = classified[ya : yb + 1, xa : xb + 1]
            counts = np.bincount(patch[patch >= 0], minlength=4)
            if int(counts.max()) < 100:
                raise ValueError("cell color evidence is too weak")
            row.append(int(np.argmax(counts)))
        board.append(tuple(row))
    return tuple(board)


def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"), dtype=np.uint8)
    reference = _read_board(a, _bbox_for_color(a, REFERENCE_BORDER))
    target = tuple(row[3:] + row[:3] for row in reference)
    matches = []
    for name, rgb in BORDERS.items():
        option = _read_board(a, _bbox_for_color(a, rgb))
        if option == target:
            matches.append(name)
    if len(matches) != 1:
        raise ValueError("image does not contain one uniquely matching option")
    return matches[0]
