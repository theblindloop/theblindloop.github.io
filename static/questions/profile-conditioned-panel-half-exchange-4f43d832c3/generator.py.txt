"""Deterministic scene sampling and rendering for panel-half exchange."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 520
ROWS, COLS, HALF = 4, 6, 3
CELL, GAP = 27, 3
CARD_W, CARD_H = 202, 142

INK = (35, 40, 48)
MUTED = (108, 117, 128)
WHITE = (250, 250, 247)
GRID = (226, 229, 226)
REFERENCE_BORDER = (105, 112, 122)
CELL_COLORS = (
    (25, 157, 190),
    (198, 65, 145),
    (65, 160, 88),
    (231, 139, 45),
)
BORDER_RGB = {
    "red": (215, 63, 55),
    "blue": (50, 99, 201),
    "gold": (220, 166, 32),
}


def _exchange(board: list[list[int]]) -> list[list[int]]:
    return [row[HALF:] + row[:HALF] for row in board]


def _hamming(a: list[list[int]], b: list[list[int]]) -> int:
    return sum(x != y for ra, rb in zip(a, b) for x, y in zip(ra, rb))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        values = list(range(4)) * 6
        rng.shuffle(values)
        reference = [values[r * COLS : (r + 1) * COLS] for r in range(ROWS)]
        target = _exchange(reference)

        # Both distractors retain the exact color histogram.  Each makes a
        # plausible but explicitly forbidden turn of one exchanged block.
        d1 = [row[:HALF][::-1] + row[HALF:] for row in target]
        d2 = deepcopy(target)
        right = [row[HALF:] for row in target][::-1]
        for r in range(ROWS):
            d2[r][HALF:] = right[r]
        if min(_hamming(target, d1), _hamming(target, d2), _hamming(d1, d2)) >= 4:
            break

    options = [d1, d2]
    target_slot = seed % 3
    options.insert(target_slot, target)
    borders = ["red", "blue", "gold"]
    rng.shuffle(borders)
    return {"reference": reference, "options": options, "borders": borders}


def _target_index(scene: dict) -> int:
    target = _exchange(scene["reference"])
    matches = [i for i, option in enumerate(scene["options"]) if option == target]
    if len(matches) != 1:
        raise ValueError("scene must contain exactly one exchanged board")
    # Access borders here as part of the complete label computation.
    if len(scene["borders"]) != 3:
        raise ValueError("scene must contain three borders")
    return matches[0]


def analytic_gold(scene: dict) -> str:
    return scene["borders"][_target_index(scene)]


def margin(scene: dict) -> float:
    target = _exchange(scene["reference"])
    winner = _target_index(scene)
    return float(min(_hamming(target, option) for i, option in enumerate(scene["options"]) if i != winner))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0
    except (ValueError, KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _draw_card(draw: ImageDraw.ImageDraw, x: int, y: int, board: list[list[int]], border: tuple[int, int, int], reference: bool = False) -> None:
    draw.rounded_rectangle((x, y, x + CARD_W - 1, y + CARD_H - 1), radius=8, fill=WHITE, outline=border, width=5)
    gx = x + (CARD_W - (COLS * CELL - GAP)) // 2
    gy = y + (CARD_H - (ROWS * CELL - GAP)) // 2
    draw.rectangle((gx - 4, gy - 4, gx + COLS * CELL - GAP + 3, gy + ROWS * CELL - GAP + 3), fill=GRID)
    for r in range(ROWS):
        for c in range(COLS):
            x0, y0 = gx + c * CELL, gy + r * CELL
            draw.rectangle((x0, y0, x0 + CELL - GAP - 1, y0 + CELL - GAP - 1), fill=CELL_COLORS[board[r][c]])
    if reference:
        cut_x = gx + HALF * CELL - 2
        for yy in range(y - 10, y + CARD_H + 10, 12):
            draw.line((cut_x, yy, cut_x, min(yy + 6, y + CARD_H + 9)), fill=INK, width=3)


def render(scene: dict) -> Image.Image:
    reference = scene["reference"]
    options = scene["options"]
    borders = scene["borders"]
    if len(options) != 3 or len(borders) != 3:
        raise ValueError("three options and three borders are required")

    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    title_font, label_font, small_font = _font(22, True), _font(16, True), _font(14)
    _center_text(draw, (WIDTH // 2, 25), "EXCHANGE THE TWO HALVES", title_font, INK)
    _center_text(draw, (WIDTH // 2, 54), "Keep every half upright and in the same internal order", small_font, MUTED)
    _center_text(draw, (WIDTH // 2, 78), "REFERENCE", label_font, INK)
    _draw_card(draw, 279, 94, reference, REFERENCE_BORDER, reference=True)

    # Visible block-motion cue: arrows cross without implying a flip.
    draw.line((315, 259, 443, 259), fill=INK, width=3)
    draw.polygon(((315, 259), (327, 252), (327, 266)), fill=INK)
    draw.polygon(((443, 259), (431, 252), (431, 266)), fill=INK)
    _center_text(draw, (WIDTH // 2, 286), "RESULT OPTIONS", label_font, INK)

    for x, board, name in zip((40, 279, 518), options, borders):
        _draw_card(draw, x, 316, board, BORDER_RGB[name])
    _center_text(draw, (WIDTH // 2, 489), "Answer with the border color: red, blue, or gold", small_font, MUTED)
    return image
