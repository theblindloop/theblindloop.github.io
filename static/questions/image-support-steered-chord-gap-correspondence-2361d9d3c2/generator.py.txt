"""Deterministic analytic scene and Pillow renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = (250, 250, 248)
PANEL_FILL = (244, 247, 249)
REFERENCE_FRAME = (105, 114, 126)
INK = (28, 33, 40)

TAGS = ("amber", "cyan", "violet")
TAG_RGB = {
    "amber": (225, 151, 32),
    "cyan": (25, 158, 181),
    "violet": (132, 82, 190),
}
TRACKS = ("red", "green", "blue")
TRACK_RGB = {
    "red": (205, 63, 69),
    "green": (47, 151, 91),
    "blue": (50, 103, 190),
}
SHAPES = ("triangle", "circle", "square")

# Every permutation has aggregate |source lane - destination lane| == 4.
# Consequently all three orthogonally routed panels have the same total centerline length.
ROUTE_PERMUTATIONS = ((2, 1, 0), (1, 2, 0), (2, 0, 1))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    left = list(SHAPES)
    rng.shuffle(left)
    required = ROUTE_PERMUTATIONS[int(seed) % len(ROUTE_PERMUTATIONS)]
    right: list[str | None] = [None, None, None]
    for source, destination in enumerate(required):
        right[destination] = left[source]

    mappings = [list(item) for item in ROUTE_PERMUTATIONS]
    rng.shuffle(mappings)
    target_tag = TAGS[int(seed) % len(TAGS)]
    remaining_tags = [tag for tag in TAGS if tag != target_tag]
    rng.shuffle(remaining_tags)
    panels = []
    other_index = 0
    for slot, mapping in enumerate(mappings):
        if tuple(mapping) == tuple(required):
            tag = target_tag
        else:
            tag = remaining_tags[other_index]
            other_index += 1
        colors = list(TRACKS)
        rng.shuffle(colors)
        panels.append(
            {"slot": slot, "tag": tag, "mapping": mapping, "track_colors": colors}
        )
    # Storage order is deliberately independent of the visible slot.
    rng.shuffle(panels)
    return {
        "left_order": left,
        "right_order": [str(value) for value in right],
        "candidate_panels": panels,
    }


def _shape(draw: ImageDraw.ImageDraw, kind: str, x: int, y: int) -> None:
    if kind == "triangle":
        draw.polygon([(x, y - 10), (x - 10, y + 9), (x + 10, y + 9)], fill=INK)
    elif kind == "circle":
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=INK)
    elif kind == "square":
        draw.rectangle((x - 10, y - 10, x + 10, y + 10), fill=INK)
    else:
        raise ValueError(f"unknown shape {kind!r}")


def _reference_bank(
    draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], order: list[str], side: str
) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=PANEL_FILL, outline=REFERENCE_FRAME, width=5)
    lanes = (y0 + 22, y0 + 58, y0 + 94)
    for lane, kind in zip(lanes, order, strict=True):
        if side == "left":
            _shape(draw, kind, x0 + 22, lane)
            draw.line((x0 + 34, lane, x1, lane), fill=INK, width=6)
        else:
            draw.line((x0, lane, x1 - 34, lane), fill=INK, width=6)
            _shape(draw, kind, x1 - 22, lane)


def _candidate_panel(draw: ImageDraw.ImageDraw, panel: dict) -> None:
    slot = int(panel["slot"])
    x0 = (232, 346, 460)[slot]
    y0 = 550
    x1, y1 = x0 + 108, y0 + 112
    tag = str(panel["tag"])
    draw.rectangle((x0, y0, x1, y1), fill=PANEL_FILL, outline=TAG_RGB[tag], width=6)
    # A redundant solid tag tab makes color readout stable under ordinary rasterization.
    draw.rounded_rectangle((x0 + 36, y1 - 15, x0 + 72, y1 - 4), radius=3, fill=TAG_RGB[tag])
    lanes = (y0 + 22, y0 + 50, y0 + 78)
    mapping = [int(value) for value in panel["mapping"]]
    colors = [str(value) for value in panel["track_colors"]]
    for source in range(3):
        destination = mapping[source]
        sy, ey = lanes[source], lanes[destination]
        bend_x = x0 + (32, 54, 76)[source]
        points = [(x0 + 6, sy), (bend_x, sy), (bend_x, ey), (x1 - 6, ey)]
        # Pale casing keeps crossings visually traceable; colored cores carry identity.
        draw.line(points, fill=PANEL_FILL, width=13, joint="curve")
        draw.line(points, fill=TRACK_RGB[colors[source]], width=7, joint="curve")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    left = [str(value) for value in scene["left_order"]]
    right = [str(value) for value in scene["right_order"]]
    panels = sorted(scene["candidate_panels"], key=lambda item: int(item["slot"]))
    _reference_bank(draw, (250, 135, 345, 250), left, "left")
    _reference_bank(draw, (455, 135, 550, 250), right, "right")
    # Window lips state the two seams without joining the decisive components.
    draw.line((345, 129, 345, 256), fill=REFERENCE_FRAME, width=3)
    draw.line((455, 129, 455, 256), fill=REFERENCE_FRAME, width=3)
    for panel in panels:
        _candidate_panel(draw, panel)
    return image


def _required_mapping(scene: dict) -> tuple[int, int, int]:
    left = [str(value) for value in scene["left_order"]]
    right = [str(value) for value in scene["right_order"]]
    return tuple(right.index(kind) for kind in left)


def analytic_gold(scene: dict) -> str:
    left = [str(value) for value in scene["left_order"]]
    right = [str(value) for value in scene["right_order"]]
    required = tuple(right.index(kind) for kind in left)
    matches = [
        str(panel["tag"])
        for panel in scene["candidate_panels"]
        if tuple(int(value) for value in panel["mapping"]) == required
    ]
    if len(matches) != 1:
        raise ValueError("scene has no unique completing patch")
    return matches[0]


def margin(scene: dict) -> float:
    required = _required_mapping(scene)
    distances = sorted(
        sum(int(a != b) for a, b in zip(required, panel["mapping"], strict=True))
        for panel in scene["candidate_panels"]
    )
    if not distances or distances[0] != 0:
        return 0.0
    return float(distances[1]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        tags = [str(panel["tag"]) for panel in scene["candidate_panels"]]
        slots = [int(panel["slot"]) for panel in scene["candidate_panels"]]
        mappings = [tuple(int(v) for v in panel["mapping"]) for panel in scene["candidate_panels"]]
        valid = (
            sorted(scene["left_order"]) == sorted(SHAPES)
            and sorted(scene["right_order"]) == sorted(SHAPES)
            and sorted(tags) == sorted(TAGS)
            and sorted(slots) == [0, 1, 2]
            and len(set(mappings)) == 3
            and analytic_gold(scene) in TAGS
            and margin(scene) >= 2.0
        )
    except (KeyError, TypeError, ValueError):
        return True
    return not valid


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["candidate_panels"] = list(reversed(twin["candidate_panels"]))
    return [("reverse_unordered_panel_records", twin)]
