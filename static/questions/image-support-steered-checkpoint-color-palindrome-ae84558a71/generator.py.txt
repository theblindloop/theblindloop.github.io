"""Deterministic visual world for route-ordered checkpoint palindrome testing."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 512
BG = (248, 247, 242)
NAVY = (31, 52, 77)
CYAN = (0, 184, 204)
ORANGE = (239, 139, 32)
GREEN = (43, 168, 92)
RED = (218, 67, 74)
PALE = (222, 225, 224)


def _palindrome(colors: list[str]) -> bool:
    return colors == list(reversed(colors))


def _point_at(points: list[list[int]], distance: float) -> tuple[float, float]:
    remaining = distance
    for left, right in zip(points, points[1:]):
        length = abs(right[0] - left[0]) + abs(right[1] - left[1])
        if remaining <= length:
            ratio = remaining / length
            return (
                left[0] + ratio * (right[0] - left[0]),
                left[1] + ratio * (right[1] - left[1]),
            )
        remaining -= length
    return tuple(points[-1])


def _route_length(points: list[list[int]]) -> float:
    return float(sum(abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    want_yes = int(seed) % 2 == 0
    eligible: list[list[str]] = []
    for cyan_positions in itertools.combinations(range(8), 4):
        cyan = set(cyan_positions)
        colors = ["cyan" if i in cyan else "orange" for i in range(8)]
        if _palindrome(colors) == want_yes:
            eligible.append(colors)
    colors = list(eligible[rng.randrange(len(eligible))])
    if rng.random() < 0.5:
        colors.reverse()

    # A thin, non-self-crossing trace spans nearly the full canvas. Geometry jitter
    # is independent of the label and preserves wide clearance between segments.
    base = [(62, 72), (185, 210), (310, 84), (435, 304), (560, 168), (704, 438)]
    points = [[x + rng.randint(-8, 8), y + rng.randint(-8, 8)] for x, y in base]
    return {"points": points, "checkpoint_colors": colors}


def render(scene: dict) -> Image.Image:
    points = [[int(x), int(y)] for x, y in scene["points"]]
    colors = list(scene["checkpoint_colors"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for y in (126, 252, 378):
        draw.line((34, y, 734, y), fill=PALE, width=1)
    draw.line([tuple(point) for point in points], fill=NAVY, width=9, joint="curve")

    total = _route_length(points)
    fractions = (0.075, 0.195, 0.315, 0.435, 0.565, 0.685, 0.805, 0.925)
    palette = {"cyan": CYAN, "orange": ORANGE}
    for fraction, color_name in zip(fractions, colors):
        x, y = _point_at(points, total * fraction)
        radius = 13
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette[color_name])
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=(255, 255, 255))

    start, finish = tuple(points[0]), tuple(points[-1])
    for center, color in ((start, GREEN), (finish, RED)):
        x, y = center
        draw.ellipse((x - 19, y - 19, x + 19, y + 19), fill=color)
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=BG)

    font = ImageFont.load_default()
    sx, sy = start
    fx, fy = finish
    draw.text((sx + 24, sy - 24), "START", fill=GREEN, font=font)
    draw.text((fx - 56, fy + 22), "FINISH", fill=RED, font=font)
    draw.text((287, 14), "TRACE ALL 8 CHECKPOINTS", fill=NAVY, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _palindrome(list(scene["checkpoint_colors"])) else "no"


def margin(scene: dict) -> float:
    colors = list(scene["checkpoint_colors"])
    mismatches = sum(a != b for a, b in zip(colors[:4], reversed(colors[4:])))
    # Positive runner-up margin: number of mirrored symbol substitutions needed,
    # combined with the fixed 30-pixel-plus marker separation reserve.
    return float(30 + max(1, mismatches))


def is_quarantined(scene: dict) -> bool:
    points = scene.get("points", [])
    colors = scene.get("checkpoint_colors", [])
    return (
        len(points) != 6
        or len(colors) != 8
        or colors.count("cyan") != 4
        or colors.count("orange") != 4
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
