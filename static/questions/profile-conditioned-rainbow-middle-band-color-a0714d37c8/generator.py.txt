"""Deterministic renderer and analytic label for the rainbow-band world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "rainbow_middle_band_color-renderer-0.1.0"
WIDTH = 512
HEIGHT = 512
SCALE = 3

COLORS = ("red", "orange", "yellow", "green", "blue")
RGB = {
    "red": (211, 57, 63),
    "orange": (230, 126, 38),
    "yellow": (235, 190, 52),
    "green": (57, 157, 91),
    "blue": (57, 116, 190),
}
BACKGROUND = (248, 246, 240)
CARD = (255, 253, 248)
INK = (38, 50, 65)
SHADOW = (218, 215, 207)


def sample_scene(seed: int) -> dict:
    """Sample a visible card while cycling the answer identity across seeds."""

    # Separate the answer cycle from nuisance sampling.  In each block of five
    # seeds the card geometry is held fixed while the middle color cycles, so
    # the counterfactuals do not let panel size or placement predict the label.
    style_rng = random.Random((seed // len(COLORS)) + 918273)
    order_rng = random.Random(seed * 7919 + 17)
    middle = COLORS[seed % len(COLORS)]
    remaining = [name for name in COLORS if name != middle]
    order_rng.shuffle(remaining)
    order = remaining[:2] + [middle] + remaining[2:]

    # A small deterministic near-boundary slice supplies explicit quarantine
    # examples without making stress scenes ambiguous to the pixel inverse.
    style_block = seed // len(COLORS)
    if style_block % 13 == 0:
        band_height = style_rng.randint(14, 19)
        separator = 1
    else:
        band_height = style_rng.randint(24, 48)
        separator = style_rng.randint(3, 8)

    card_width = style_rng.randint(254, 326)
    card_x = style_rng.randint(82, WIDTH - card_width - 82)
    card_y = style_rng.randint(142, 178)
    label_y = card_y - style_rng.randint(27, 36)
    corner_radius = style_rng.randint(12, 22)
    return {
        "stripe_order": order,
        "card_x": card_x,
        "card_y": card_y,
        "card_width": card_width,
        "band_height": band_height,
        "separator": separator,
        "corner_radius": corner_radius,
        "label_y": label_y,
    }


def _card_height(scene: dict) -> int:
    return 84 + 5 * int(scene["band_height"]) + 4 * int(scene["separator"])


def render(scene: dict) -> Image.Image:
    """Render only the visible scene fields into a fixed RGB PNG-sized image."""

    width = WIDTH * SCALE
    height = HEIGHT * SCALE
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    card_x = int(scene["card_x"])
    card_y = int(scene["card_y"])
    card_width = int(scene["card_width"])
    band_height = int(scene["band_height"])
    separator = int(scene["separator"])
    corner_radius = int(scene["corner_radius"])
    label_y = int(scene["label_y"])
    card_height = _card_height(scene)

    def box(x0: int, y0: int, x1: int, y1: int) -> tuple[int, int, int, int]:
        return (x0 * SCALE, y0 * SCALE, x1 * SCALE, y1 * SCALE)

    draw.rounded_rectangle(
        box(card_x + 5, card_y + 6, card_x + card_width + 5, card_y + card_height + 6),
        radius=(corner_radius + 2) * SCALE,
        fill=SHADOW,
    )
    draw.rounded_rectangle(
        box(card_x, card_y, card_x + card_width, card_y + card_height),
        radius=corner_radius * SCALE,
        fill=CARD,
        outline=INK,
        width=3 * SCALE,
    )

    inner_x0 = card_x + 26
    inner_x1 = card_x + card_width - 26
    stripe_y = card_y + 49
    order = scene["stripe_order"]
    for index, color_name in enumerate(order):
        y0 = stripe_y + index * (band_height + separator)
        y1 = y0 + band_height
        draw.rectangle(box(inner_x0, y0, inner_x1, y1), fill=RGB[color_name])

    # The title is a fixed semantic cue, not an answer-bearing field.  Drawing
    # it after downsampling keeps the bundled font-independent raster stable.
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    final_draw = ImageDraw.Draw(image)
    final_draw.text((card_x + 25, label_y), "RAINBOW", fill=INK, font=ImageFont.load_default())
    return image


def analytic_gold(scene: dict) -> str:
    """Closed-form label: the visible third member of the ordered five-band set."""

    return str(scene["stripe_order"][2])


def margin(scene: dict) -> float:
    """Positive raster-verifiability margin in pixels."""

    return float(min(int(scene["band_height"]), 2 * int(scene["separator"]) + 1))


def is_quarantined(scene: dict) -> bool:
    """Exclude thin bands or near-zero separators from headline claims."""

    return (
        int(scene["band_height"]) < 20
        or int(scene["separator"]) < 2
        or int(scene["card_width"]) < 220
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No untested render-invariant aliases are claimed for this world."""

    return []
