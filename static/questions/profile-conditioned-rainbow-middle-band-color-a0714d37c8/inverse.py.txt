"""Independent pixel-only inverse for the rainbow-band raster.

This module intentionally owns its palette and geometry-free measurement.  It
does not import the renderer or consume a latent scene.  Each answer color is
located by nearest-color segmentation, then its visible pixel centroid gives
the ordinal band position.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLOR_NAMES = ("red", "orange", "yellow", "green", "blue")
COLOR_RGB = (
    (211, 57, 63),
    (230, 126, 38),
    (235, 190, 52),
    (57, 157, 91),
    (57, 116, 190),
)
BACKGROUND_RGB = (248, 246, 240)
INK_RGB = (38, 50, 65)


def _nearest_labels(image: Image.Image) -> np.ndarray:
    """Assign each raster pixel to the nearest fixed visible palette color."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    palette = np.asarray((*COLOR_RGB, BACKGROUND_RGB, INK_RGB), dtype=np.int32)
    distances = ((pixels[:, :, None, :] - palette[None, None, :, :]) ** 2).sum(axis=3)
    return np.argmin(distances, axis=2)


def decision_from_image(image: Image.Image) -> str:
    """Return the color at the visible median vertical band, from pixels alone."""

    labels = _nearest_labels(image)
    centroids = []
    for index, name in enumerate(COLOR_NAMES):
        ys = np.nonzero(labels == index)[0]
        if len(ys) < 100:
            raise ValueError(f"color {name} has too few visible pixels")
        centroids.append((float(ys.mean()), name))
    centroids.sort(key=lambda item: item[0])
    return centroids[2][1]
