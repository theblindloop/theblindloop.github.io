"""key_xor_null_reveal: legend-keyed XOR nulling reveal world.

Declared transform: XOR the noise field with an externally rendered noise KEY.
The image has, on the left, a 32-by-32 grid of dark/light cells that is pure
noise (no symbol visible), and on the right a magenta-framed box that shows the
pure-noise KEY that was used to hide the symbol. The hidden symbol was combined
with the field by bitwise XOR: every field cell where the symbol is "on" had its
noise shade flipped to the opposite value. Undoing the declared cipher means
taking the logical XOR of the left field with the key grid; the matching noise
cancels and the clean abstract symbol remains (a filled circle for A or a filled
square for B).

The raw left field alone (or the key alone) is uniform random noise in which no
region reveals the glyph; the answer appears only after executing the declared
XOR-nulling with the explicitly rendered key.

Latent scene z = {glyph, key, invert}. label = glyph. The key noise is sampled
per-scene from the seed; the glyph is the only label.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "key_xor_null_reveal-0.1.0"

NCELL = 32
FIELD_CELL = 8      # field rendered at 8 px per cell -> 256x256
FIELD = NCELL * FIELD_CELL
KEY_CELL = 3        # key box rendered at 3 px per cell -> 96x96
KEY = NCELL * KEY_CELL
BORDER = 4          # magenta frame thickness around the key box
KEY_X0 = 284
KEY_Y0 = 20
W = 470
H = 300
NULL_TOL = 25.0     # quarantine threshold (cells) on the A/B reconstruction gap

MAGENTA = (255, 0, 255)


def glyph_mask(glyph: str) -> np.ndarray:
    """(NCELL, NCELL) bool: True = symbol ink."""
    m = np.zeros((NCELL, NCELL), dtype=bool)
    i, j = np.indices((NCELL, NCELL))
    cy = cx = (NCELL - 1) / 2.0
    if glyph == "A":
        r = 8.0
        m[(i - cy) ** 2 + (j - cx) ** 2 <= r * r] = True
    else:
        half = 7.0
        m[(np.abs(i - cy) <= half) & (np.abs(j - cx) <= half)] = True
    return m


def sample_key(seed: int) -> np.ndarray:
    rng = np.random.default_rng((int(seed) & 0xFFFFFFFF) ^ 0x5A5A5A5A)
    return rng.integers(0, 2, size=(NCELL, NCELL), dtype=np.uint8) > 0


def build_scene(glyph: str, key: np.ndarray, invert: bool) -> dict:
    return {"glyph": glyph, "key": key.copy(), "invert": bool(invert)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    key = rng.integers(0, 2, size=(NCELL, NCELL), dtype=np.uint8) > 0
    return build_scene(glyph=glyph, key=key, invert=bool(int(seed) % 4 == 0))


def _as_bool_keys(key) -> np.ndarray:
    return np.asarray(key, dtype=bool)


def _field_bits(scene: dict) -> np.ndarray:
    glyph = scene["glyph"]
    key = _as_bool_keys(scene["key"])
    field = key ^ glyph_mask(glyph)
    if scene.get("invert"):
        field = ~field
    return field


def _key_bits(scene: dict) -> np.ndarray:
    key = _as_bool_keys(scene["key"])
    if scene.get("invert"):
        key = ~key
    return key


def render(scene: dict) -> Image.Image:
    arr = np.full((H, W, 3), 255, dtype=np.uint8)
    field = _field_bits(scene)
    for r in range(NCELL):
        for c in range(NCELL):
            color = 0 if field[r, c] else 255
            arr[r * FIELD_CELL:(r + 1) * FIELD_CELL,
                c * FIELD_CELL:(c + 1) * FIELD_CELL] = color
    key = _key_bits(scene)
    x0, y0 = KEY_X0, KEY_Y0
    inner = arr[y0 + BORDER:y0 + BORDER + KEY, x0 + BORDER:x0 + BORDER + KEY]
    for r in range(NCELL):
        for c in range(NCELL):
            color = 0 if key[r, c] else 255
            inner[r * KEY_CELL:(r + 1) * KEY_CELL,
                  c * KEY_CELL:(c + 1) * KEY_CELL] = color
    arr[y0:y0 + BORDER, x0:x0 + BORDER + KEY] = MAGENTA
    arr[y0 + BORDER + KEY:y0 + BORDER + KEY + BORDER, x0:x0 + BORDER + KEY] = MAGENTA
    arr[y0:y0 + BORDER + KEY + BORDER, x0:x0 + BORDER] = MAGENTA
    arr[y0:y0 + BORDER + KEY + BORDER, x0 + BORDER + KEY:x0 + BORDER + KEY + BORDER] = MAGENTA
    return Image.fromarray(arr, "RGB")


def _extract(luma: np.ndarray, cells: int, cell_px: int) -> np.ndarray:
    cells_px = cells * cell_px
    block = luma[:cells_px, :cells_px].reshape(cells, cell_px, cells, cell_px)
    return block.mean(axis=(1, 3)) < 128


def reconstruct(scene: dict) -> np.ndarray:
    field = _field_bits(scene)
    key = _key_bits(scene)
    rec = field ^ key
    if rec.sum() > (NCELL * NCELL) / 2:
        rec = ~rec
    return rec


def margin(scene: dict) -> float:
    rec = reconstruct(scene)
    ga = glyph_mask("A")
    gb = glyph_mask("B")
    diffA = float(np.count_nonzero(rec != ga))
    diffB = float(np.count_nonzero(rec != gb))
    return round(diffB - diffA, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < NULL_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field directly changes the rendered pixels
    (glyph flips the field bits, key sets the noise, invert toggles all shades),
    so no two distinct latent vectors reproduce the same raster with different
    labels. Declared empty."""
    return iter(())
