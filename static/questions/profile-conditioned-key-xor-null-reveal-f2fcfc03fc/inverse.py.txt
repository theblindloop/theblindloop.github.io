"""Independent pixel-only inverse arm for key_xor_null_reveal.

Receives ONLY the final PNG. It locates the magenta-framed key box, reads the
32-by-32 key grid it displays, reads the main noise field (left, 32x32), and
executes the declared cipher by taking the logical XOR of the field with the key,
which nulls the matching noise to expose the hidden symbol. It then classifies the
recovered glyph against the two declared reference symbols (A = filled circle,
B = filled square) and returns the answer. It reads no scene data and no analytic
gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "key_xor_null_reveal-oracle-0.1.0"

NCELL = 32
FIELD_CELL = 8
FIELD = NCELL * FIELD_CELL
KEY_CELL = 3
KEY = NCELL * KEY_CELL
BORDER = 4
KEY_X0 = 284
KEY_Y0 = 20

MAGENTA = (255, 0, 255)

_A: np.ndarray | None = None
_B: np.ndarray | None = None


def _glyph_mask(glyph: str) -> np.ndarray:
    m = np.zeros((NCELL, NCELL), dtype=bool)
    i, j = np.indices((NCELL, NCELL))
    cy = cx = (NCELL - 1) / 2.0
    if glyph == "A":
        m[(i - cy) ** 2 + (j - cx) ** 2 <= 64.0] = True
    else:
        m[(np.abs(i - cy) <= 7.0) & (np.abs(j - cx) <= 7.0)] = True
    return m


def _ref(glyph: str) -> np.ndarray:
    global _A, _B
    if glyph == "A":
        if _A is None:
            _A = _glyph_mask("A")
        return _A
    if _B is None:
        _B = _glyph_mask("B")
    return _B


def _locate_key_box(rgb: np.ndarray) -> tuple[int, int] | None:
    magenta = (rgb[:, :, 0].astype(int) > 200) & (
        rgb[:, :, 1].astype(int) < 80
    ) & (rgb[:, :, 2].astype(int) > 200)
    ys, xs = np.nonzero(magenta)
    if ys.size == 0:
        return None
    ymin, ymax = int(ys.min()), int(ys.max())
    xmin, xmax = int(xs.min()), int(xs.max())
    y0 = ymin + BORDER
    x0 = xmin + BORDER
    if ymax - ymin < 2 * BORDER or xmax - xmin < 2 * BORDER:
        return None
    return x0, y0


def _bits(luma: np.ndarray, y0: int, x0: int, cell_px: int) -> np.ndarray:
    span = NCELL * cell_px
    block = luma[y0:y0 + span, x0:x0 + span].reshape(NCELL, cell_px, NCELL, cell_px)
    return block.mean(axis=(1, 3)) < 128


def reconstruct(image: Image.Image) -> np.ndarray | None:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    luma = np.asarray(image.convert("L"), dtype=np.uint8)
    loc = _locate_key_box(rgb)
    if loc is None:
        return None
    kx, ky = loc
    field = _bits(luma, 0, 0, FIELD_CELL)
    key = _bits(luma, ky, kx, KEY_CELL)
    rec = field ^ key
    if rec.sum() > (NCELL * NCELL) / 2:
        rec = ~rec
    return rec


def decision_from_image(image: Image.Image) -> str:
    rec = reconstruct(image)
    if rec is None:
        return "ABSTAIN"
    diffA = int(np.count_nonzero(rec != _ref("A")))
    diffB = int(np.count_nonzero(rec != _ref("B")))
    if diffA == diffB:
        return "ABSTAIN"
    return "A" if diffA < diffB else "B"


def reveal_margin(image: Image.Image) -> float:
    rec = reconstruct(image)
    if rec is None:
        return 0.0
    diffA = float(np.count_nonzero(rec != _ref("A")))
    diffB = float(np.count_nonzero(rec != _ref("B")))
    return abs(diffB - diffA)
