"""Deterministic renderer and analytic arm for one-junction loop cuts."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512

# Six paired classes.  Within every pair, the YES and NO graphs have the same
# twelve vertices, thirteen unit-grid links, and complete sorted degree list.
# The analytic answer is never stored in a scene; it is recomputed from edges.
TEMPLATE_PAIRS = (
    (
        ((0,1),(0,4),(1,2),(1,5),(2,3),(2,6),(4,5),(5,6),(5,9),(6,7),(6,10),(7,11),(8,9)),
        ((0,1),(0,4),(1,5),(2,3),(2,6),(3,7),(4,5),(4,8),(5,6),(5,9),(6,7),(6,10),(7,11)),
    ),
    (
        ((0,1),(0,4),(1,2),(1,5),(2,3),(2,6),(3,7),(4,5),(4,8),(5,6),(5,9),(6,10),(7,11)),
        ((0,4),(1,2),(2,3),(2,6),(3,7),(4,5),(4,8),(5,6),(5,9),(6,7),(6,10),(7,11),(8,9)),
    ),
    (
        ((0,1),(0,4),(1,2),(1,5),(2,3),(2,6),(3,7),(5,6),(5,9),(6,7),(7,11),(8,9),(9,10)),
        ((0,1),(0,4),(1,2),(1,5),(2,3),(2,6),(3,7),(4,5),(4,8),(5,9),(6,7),(6,10),(7,11)),
    ),
    (
        ((0,1),(0,4),(1,5),(2,3),(2,6),(4,5),(4,8),(5,6),(5,9),(6,7),(6,10),(7,11),(8,9)),
        ((0,1),(0,4),(1,5),(2,3),(2,6),(3,7),(4,5),(5,6),(5,9),(6,7),(6,10),(7,11),(8,9)),

        ((0,1),(0,4),(1,5),(2,3),(2,6),(3,7),(4,5),(4,8),(5,6),(6,7),(6,10),(7,11),(8,9)),

        ((0,1),(0,4),(1,2),(1,5),(2,3),(2,6),(3,7),(4,5),(4,8),(5,6),(6,10),(7,11),(8,9)),

NODE_COLORS = ("#1677c8", "#d86613", "#7a42b8", "#178f73", "#c43b65", "#476bd1")
LINK_COLORS = ("#20242b", "#303238", "#17242c")
BACKGROUNDS = ("#ffffff", "#faf9f6", "#f7f9fc")


def _contains_cycle(edges: Iterable[tuple[int, int]], removed: int | None = None) -> bool:
    vertices = set()
    adjacency: dict[int, list[int]] = {}
    for a, b in edges:
        vertices.update((a, b))
        if a == removed or b == removed:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    vertices.discard(removed)
    seen: set[int] = set()

    def visit(vertex: int, parent: int) -> bool:
        seen.add(vertex)
        for neighbor in adjacency.get(vertex, ()):
            if neighbor == parent:
                continue
            if neighbor in seen or visit(neighbor, vertex):
                return True
        return False

    return any(vertex not in seen and visit(vertex, -1) for vertex in vertices)


def _one_vertex_hits_cycles(scene: dict) -> bool:
    edges = [tuple(edge) for edge in scene["edges"]]
    vertices = sorted({value for edge in edges for value in edge})
    return _contains_cycle(edges) and any(not _contains_cycle(edges, vertex) for vertex in vertices)


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    paired_seed = seed // 2
    rng = random.Random(paired_seed ^ 0x5A17C3)
    pair_index = paired_seed % len(TEMPLATE_PAIRS)
    # Even seeds are YES and odd seeds are NO. Consecutive scenes share every
    # nuisance choice and matched-degree template group.
    template = TEMPLATE_PAIRS[pair_index][0 if seed % 2 == 0 else 1]
    spacing = rng.choice((82, 88, 94))
    turns = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    dx, dy = rng.randint(-10, 10), rng.randint(-10, 10)
    nodes = []
    for index in range(12):
        row, column = divmod(index, 4)
        x = (column - 1.5) * spacing
        y = (row - 1.0) * spacing
        if reflect:
            x = -x
        for _ in range(turns):
            x, y = -y, x
        nodes.append([int(round(CANVAS / 2 + dx + x)), int(round(CANVAS / 2 + dy + y))])
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in template],
        "node_radius": rng.choice((16, 18)),
        "link_width": rng.choice((9, 11)),
        "node_color": rng.choice(NODE_COLORS),
        "link_color": rng.choice(LINK_COLORS),
        "background": rng.choice(BACKGROUNDS),
        "canvas_size": [CANVAS, CANVAS],
    }


def render(scene: dict) -> Image.Image:
    width, height = [int(value) for value in scene["canvas_size"]]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    nodes = [(int(x), int(y)) for x, y in scene["nodes"]]
    for a, b in scene["edges"]:
        x1, y1 = nodes[int(a)]
        x2, y2 = nodes[int(b)]
        draw.line(
            (x1 * scale, y1 * scale, x2 * scale, y2 * scale),
            fill=scene["link_color"],
            width=int(scene["link_width"]) * scale,
        )
    radius = int(scene["node_radius"]) * scale
    for x, y in nodes:
        cx, cy = x * scale, y * scale
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=scene["node_color"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if _one_vertex_hits_cycles(scene) else "no"


def margin(scene: dict) -> float:
    """Runner-up margin: minimum single-link edits needed to flip the decision."""
    edges = {tuple(sorted(edge)) for edge in scene["edges"]}
    vertices = sorted({value for edge in edges for value in edge})
    original = _contains_cycle(edges) and any(not _contains_cycle(edges, v) for v in vertices)
    for a in vertices:
        for b in vertices:
            if a >= b:
                continue
            changed = set(edges)
            edge = (a, b)
            if edge in changed:
                changed.remove(edge)
            else:
                changed.add(edge)
            value = _contains_cycle(changed) and any(
                not _contains_cycle(changed, v) for v in vertices
            )
            if value != original:
                return 1.0
    return 2.0


def is_quarantined(scene: dict) -> bool:
    nodes = scene.get("nodes", [])
    edges = {tuple(sorted(edge)) for edge in scene.get("edges", [])}
    if len(nodes) != 12 or len(edges) != 13:
        return True
    if int(scene.get("node_radius", 0)) < 14 or int(scene.get("link_width", 0)) < 8:
        return True
    if any(a == b or not (0 <= a < 12 and 0 <= b < 12) for a, b in edges):
        return True
    minimum = min(
        math.hypot(nodes[a][0] - nodes[b][0], nodes[a][1] - nodes[b][1])
        for a in range(12) for b in range(a + 1, 12)
    )
    return minimum < 75


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = [[b, a] for a, b in reversed(scene["edges"])]
    return [("reverse_edge_serialization", twin)]
