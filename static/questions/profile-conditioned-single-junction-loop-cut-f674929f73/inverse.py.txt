"""Independent inverse arm: recover the graph from final RGB pixels only."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros(mask.shape, dtype=np.uint8)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if visited[y0, x0]:
            continue
        visited[y0, x0] = 1
        stack = [(y0, x0)]
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = 1
                    stack.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _recover_graph(image: Image.Image) -> tuple[int, set[tuple[int, int]]]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    chroma = high - low
    colored = (chroma > 55) & (high > 115) & (low < 225)
    dark = (high < 110) & (chroma < 35)
    node_parts = _components(colored, 250)
    link_parts = _components(dark, 80)
    if len(node_parts) != 12:
        raise ValueError(f"expected 12 colored junctions, recovered {len(node_parts)}")
    if len(link_parts) != 13:
        raise ValueError(f"expected 13 separated dark links, recovered {len(link_parts)}")
    centers = np.asarray(
        [[part[:, 1].mean(), part[:, 0].mean()] for part in node_parts], dtype=np.float64
    )
    edges: set[tuple[int, int]] = set()
    for part in link_parts:
        xy = part[:, ::-1]
        distances = ((xy[:, None, :] - centers[None, :, :]) ** 2).sum(axis=2).min(axis=0)
        nearest = np.argsort(distances)[:2]
        if distances[nearest[1]] > 38.0**2:
            raise ValueError("a dark component does not terminate at two visible junctions")
        edge = tuple(sorted((int(nearest[0]), int(nearest[1]))))
        if edge[0] == edge[1] or edge in edges:
            raise ValueError("duplicate or self-link recovered from pixels")
        edges.add(edge)
    if len(edges) != 13:
        raise ValueError("link components do not encode thirteen unique incidences")
    return len(centers), edges


def _is_acyclic(vertex_count: int, edges: set[tuple[int, int]], removed: int) -> bool:
    parent = list(range(vertex_count))

    def root(value: int) -> int:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    for a, b in edges:
        if a == removed or b == removed:
            continue
        ra, rb = root(a), root(b)
        if ra == rb:
            return False
        parent[ra] = rb
    return True


def decision_from_image(image: Image.Image) -> str:
    vertex_count, edges = _recover_graph(image)
    # Thirteen edges on twelve vertices guarantee at least one cycle here, but
    # we still explicitly reject a malformed acyclic inverse.
    if _is_acyclic(vertex_count, edges, -1):
        raise ValueError("recovered graph unexpectedly has no loop")
    return "yes" if any(_is_acyclic(vertex_count, edges, v) for v in range(vertex_count)) else "no"
