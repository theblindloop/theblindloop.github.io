"""Independent pixel-only inverse arm for alternating_station_cycle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _dilate(mask: np.ndarray, iterations: int) -> np.ndarray:
    result = mask.copy()
    for _ in range(iterations):
        expanded = result.copy()
        expanded[1:] |= result[:-1]
        expanded[:-1] |= result[1:]
        expanded[:, 1:] |= result[:, :-1]
        expanded[:, :-1] |= result[:, 1:]
        result = expanded
    return result


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    groups = []
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = [(sy, sx)]
        points = []
        while queue:
            y, x = queue.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        if len(points) >= minimum:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _has_large_enclosure(rail: np.ndarray) -> bool:
    # Pool to a 128x128 obstacle map. This is independent of renderer geometry
    # and deliberately tolerant of antialiasing and ordinary PNG decoding.
    h, w = rail.shape
    factor = 4
    hh, ww = h // factor, w // factor
    obstacle = rail[: hh * factor, : ww * factor].reshape(hh, factor, ww, factor).any(axis=(1, 3))
    free = ~obstacle
    reachable = np.zeros_like(free)
    q: deque[tuple[int, int]] = deque()
    for x in range(ww):
        if free[0, x]: q.append((0, x)); reachable[0, x] = True
        if free[-1, x] and not reachable[-1, x]: q.append((hh - 1, x)); reachable[-1, x] = True
    for y in range(hh):
        if free[y, 0] and not reachable[y, 0]: q.append((y, 0)); reachable[y, 0] = True
        if free[y, -1] and not reachable[y, -1]: q.append((y, ww - 1)); reachable[y, ww - 1] = True
    while q:
        y, x = q.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < hh and 0 <= nx < ww and free[ny, nx] and not reachable[ny, nx]:
                reachable[ny, nx] = True
                q.append((ny, nx))
    return int((free & ~reachable).sum()) >= 300


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[0] < 128 or array.shape[1] < 128:
        raise ValueError("image too small for route recovery")
    rail = (array[:, :, 0] < 80) & (array[:, :, 1] < 85) & (array[:, :, 2] < 90)
    yellow = (array[:, :, 0] > 205) & (array[:, :, 1] > 135) & (array[:, :, 1] < 220) & (array[:, :, 2] < 100)
    if int(rail.sum()) < 2500 or not _has_large_enclosure(rail):
        raise ValueError("no intact widely spanning closed railway loop")
    # Seven-pixel closing rejoins the two yellow lobes split by a 7-9px dark stroke.
    marker_groups = _components(_dilate(yellow, 7), minimum=120)
    # Off-route distractors are explicitly irrelevant, so their erasure must not
    # make the inverse arm abstain. Only an implausible explosion of fragments is
    # rejected here; damage to an on-route marker also cuts the underlying rail
    # and is caught by the closed-enclosure check above.
    if len(marker_groups) > 14:
        raise ValueError("station markers have fragmented under corruption")
    on_route = 0
    for group in marker_groups:
        cy, cx = group.mean(axis=0)
        yy, xx = np.ogrid[: rail.shape[0], : rail.shape[1]]
        radius2 = (yy - cy) ** 2 + (xx - cx) ** 2
        crossing_ink = int((rail & (radius2 >= 15 ** 2) & (radius2 <= 25 ** 2)).sum())
        if crossing_ink >= 35:
            on_route += 1
    if not 5 <= on_route <= 10:
        raise ValueError("on-route station count is outside the declared visual world")
    return "yes" if on_route % 2 == 0 else "no"
