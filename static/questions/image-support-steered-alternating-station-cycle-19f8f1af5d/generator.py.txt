"""Deterministic renderer and analytic arm for alternating_station_cycle."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

BACKGROUND = (248, 245, 237)
RAIL = (32, 37, 43)
YELLOW = (242, 182, 50)
STATION_EDGE = (183, 119, 25)


def _point_on_closed_polyline(points: list[list[float]], fraction: float) -> list[float]:
    edges = list(zip(points, points[1:] + points[:1]))
    lengths = [math.dist(a, b) for a, b in edges]
    target = (fraction % 1.0) * sum(lengths)
    for (a, b), length in zip(edges, lengths):
        if target <= length:
            t = target / length
            return [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])]
        target -= length
    return list(points[0])


def _segment_distance(point: list[float], a: list[float], b: list[float]) -> float:
    vx, vy = b[0] - a[0], b[1] - a[1]
    wx, wy = point[0] - a[0], point[1] - a[1]
    denom = vx * vx + vy * vy
    t = 0.0 if denom == 0 else max(0.0, min(1.0, (wx * vx + wy * vy) / denom))
    return math.hypot(point[0] - (a[0] + t * vx), point[1] - (a[1] + t * vy))


def _distance_to_route(point: list[float], route: list[list[float]]) -> float:
    return min(_segment_distance(point, a, b) for a, b in zip(route, route[1:] + route[:1]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # A long, narrow closed capsule keeps the decisive support a single thin
    # distributed trace rather than an area-filling enclosure.
    base = [
        [43, 256], [54, 230], [82, 216], [190, 211], [322, 215],
        [430, 219], [462, 237], [469, 263], [456, 286], [425, 297],
        [315, 301], [186, 298], [78, 293], [51, 280],
    ]
    dx, dy = rng.randint(-5, 5), rng.randint(-5, 5)
    mirror = rng.choice([False, True])
    route = []
    for x, y in base:
        x = 512 - x if mirror else x
        route.append([x + dx + rng.randint(-3, 3), y + dy + rng.randint(-3, 3)])

    route_count = rng.choice([5, 6, 7, 8, 9, 10])
    phase = rng.uniform(0.0, 1.0)
    route_stations = [
        _point_on_closed_polyline(route, phase + index / route_count)
        for index in range(route_count)
    ]
    interior_pool = [
        [130, 105], [251, 98], [378, 112], [168, 155], [335, 157],
        [132, 365], [252, 355], [382, 370], [255, 420],
    ]
    rng.shuffle(interior_pool)
    distractors = []
    for x, y in interior_pool[: 12 - route_count]:
        x = 512 - x if mirror else x
        distractors.append([x + dx + rng.randint(-4, 4), y + dy + rng.randint(-4, 4)])
    return {
        "width": 512,
        "height": 512,
        "route": route,
        "route_stations": route_stations,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    route = [[float(x), float(y)] for x, y in scene["route"]]
    route_stations = [[float(x), float(y)] for x, y in scene["route_stations"]]
    distractors = [[float(x), float(y)] for x, y in scene["distractors"]]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # All station disks are drawn first. The rail then passes visibly through
    # route stations; equal short slashes give off-route stations comparable ink.
    for x, y in route_stations + distractors:
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=YELLOW, outline=STATION_EDGE, width=2)
    rail_points = [(round(x), round(y)) for x, y in route]
    draw.line(rail_points + [rail_points[0]], fill=RAIL, width=9, joint="curve")
    for x, y in distractors:
        draw.line((x - 6, y, x + 6, y), fill=RAIL, width=7)
    return image


def analytic_gold(scene: dict) -> str:
    route_stations = scene["route_stations"]
    return "yes" if len(route_stations) % 2 == 0 else "no"


def margin(scene: dict) -> float:
    route = scene["route"]
    route_stations = scene["route_stations"]
    distractors = scene["distractors"]
    off_clearance = min((_distance_to_route(p, route) - 24.0 for p in distractors), default=40.0)
    all_points = route_stations + distractors
    spacing = min(
        (math.dist(a, b) - 30.0 for i, a in enumerate(all_points) for b in all_points[i + 1 :]),
        default=40.0,
    )
    return float(min(20.0, off_clearance, spacing))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_distractor_draw_order", twin)]
