"""Renderer and analytic arm for the ribbon-width blur world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route-width-blur-0.1.0"
WIDTH = 640
HEIGHT = 480
N_SAMPLES = 32
ROUTE_SPAN = 320.0
BACKGROUND = (248, 248, 244)

# Both labels have 16 bright/wide cells.  They differ in spatial arrangement,
# so total ribbon ink, endpoint identity, and one-dimensional counts do not
# answer the question.
BOW = tuple(int(bit) for bit in "10011001011001100110011010011001")
HARP = tuple(int(bit) for bit in "11100111100000011000000111100111")
TEMPLATES = {"A": BOW, "B": HARP}
DIFF_POSITIONS = tuple(i for i, (a, b) in enumerate(zip(BOW, HARP)) if a != b)

PALETTES = (
    ((22, 44, 82), (211, 55, 62), (42, 145, 89), (234, 239, 242)),
    ((27, 49, 89), (204, 61, 58), (36, 139, 94), (237, 241, 240)),
    ((20, 42, 78), (219, 68, 57), (47, 151, 91), (232, 238, 244)),
)


def _blur(values: tuple[int, ...] | list[int]) -> tuple[float, ...]:
    """Apply the public three-sample box blur with edge replication."""

    if len(values) != N_SAMPLES:
        raise ValueError("width code must contain 32 samples")
    result: list[float] = []
    for index in range(N_SAMPLES):
        left = values[max(0, index - 1)]
        middle = values[index]
        right = values[min(N_SAMPLES - 1, index + 1)]
        result.append((float(left) + float(middle) + float(right)) / 3.0)
    return tuple(result)


def _blurred_bits(code: tuple[int, ...] | list[int]) -> tuple[int, ...]:
    return tuple(int(value >= 0.5) for value in _blur(tuple(code)))


def _hamming(code: tuple[int, ...], template: tuple[int, ...]) -> int:
    return sum(left != right for left, right in zip(code, template))


def _classify(code: tuple[int, ...]) -> str:
    blurred = _blurred_bits(code)
    distance_a = _hamming(blurred, BOW)
    distance_b = _hamming(blurred, HARP)
    return "A" if distance_a <= distance_b else "B"


def sample_scene(seed: int) -> dict:
    """Sample a balanced route, independent pose, and visible width nuisance."""

    rng = random.Random(int(seed))
    label = "A" if int(seed) % 2 == 0 else "B"
    code = list(TEMPLATES[label])
    # A sparse near-boundary population makes the displayed decision boundary
    # observable without making the ordinary stress population ambiguous.
    if int(seed) % 13 == 0:
        flips = min(12, len(DIFF_POSITIONS) // 2)
        for index in rng.sample(DIFF_POSITIONS, flips):
            code[index] = 1 - code[index]
    return {
        "code": code,
        "route_style": int(rng.randrange(4)),
        "x0": int(rng.choice((132, 148, 164, 180))),
        "base_y": int(rng.choice((205, 225, 245, 265))),
        "reverse": bool(rng.randrange(2)),
        "width_jitter": [int(rng.choice((-1, 0, 0, 1))) for _ in range(N_SAMPLES)],
        "palette_variant": int(rng.randrange(len(PALETTES))),
        "window_variant": int(rng.randrange(3)),
    }


def _centerline(scene: dict) -> list[tuple[float, float]]:
    style = int(scene["route_style"]) % 4
    base_y = float(scene["base_y"])
    values: list[tuple[float, float]] = []
    for index in range(N_SAMPLES + 1):
        t = index / float(N_SAMPLES)
        if style == 0:
            offset = 54.0 * math.sin(math.pi * t)
        elif style == 1:
            offset = -54.0 * math.sin(math.pi * t)
        elif style == 2:
            offset = 40.0 * math.sin(2.0 * math.pi * t + 0.35)
        else:
            offset = 36.0 * math.sin(math.pi * t) + 16.0 * math.sin(2.0 * math.pi * t + 0.4)
        x = float(scene["x0"]) + ROUTE_SPAN * t
        values.append((x, base_y + offset))
    if bool(scene["reverse"]):
        values.reverse()
    return values


def _route_geometry(scene: dict) -> tuple[list[tuple[float, float]], list[int]]:
    points = _centerline(scene)
    code = tuple(int(value) for value in scene["code"])
    jitter = tuple(int(value) for value in scene["width_jitter"])
    if len(code) != N_SAMPLES or len(jitter) != N_SAMPLES:
        raise ValueError("route scene has the wrong number of sections")
    widths = [8 + 6 * bit + jitter[index] for index, bit in enumerate(code)]
    return points, widths


def render(scene: dict) -> Image.Image:
    """Render one antialiased RGB route image."""

    scale = 4
    navy, red, green, window = PALETTES[int(scene["palette_variant"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    x0 = float(scene["x0"])
    x1 = x0 + ROUTE_SPAN
    y0 = float(scene["base_y"])
    window_margin = 28.0 + 4.0 * (int(scene["window_variant"]) % 3)
    box = (
        int(round((x0 - window_margin) * scale)),
        int(round((y0 - 92.0) * scale)),
        int(round((x1 + window_margin) * scale)),
        int(round((y0 + 92.0) * scale)),
    )
    draw.rounded_rectangle(box, radius=16 * scale, fill=window)

    points, widths = _route_geometry(scene)
    scaled = [(int(round(x * scale)), int(round(y * scale))) for x, y in points]
    for index, width in enumerate(widths):
        p0, p1 = scaled[index], scaled[index + 1]
        stroke = max(1, int(round(float(width) * scale)))
        draw.line((p0, p1), fill=navy, width=stroke)
        # Rounded joins keep the route one connected component as its width
        # changes, while the section midpoint remains a clean width sample.
        # A small bridge is enough because the two stroked segments share the
        # same endpoint.  Using the maximum width here would fill the narrow
        # sections with a sequence of oversized beads and erase the signal.
        radius = min(width, widths[min(index + 1, N_SAMPLES - 1)]) * scale / 2.0
        for px, py in (p0, p1):
            draw.ellipse(
                (int(round(px - radius)), int(round(py - radius)),
                 int(round(px + radius)), int(round(py + radius))),
                fill=navy,
            )

    # Keep the endpoint markers visibly distinct without covering the middle
    # of the first/last width section at the sampling column.
    cap_radius = 4 * scale
    start = scaled[0]
    end = scaled[-1]
    draw.ellipse(
        (start[0] - cap_radius, start[1] - cap_radius,
         start[0] + cap_radius, start[1] + cap_radius),
        fill=red,
    )
    draw.ellipse(
        (end[0] - cap_radius, end[1] - cap_radius,
         end[0] + cap_radius, end[1] + cap_radius),
        fill=green,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    code = tuple(int(value) for value in scene["code"])
    if len(code) != N_SAMPLES or any(value not in (0, 1) for value in code):
        raise ValueError("width code must contain 32 binary samples")
    return _classify(code)


def margin(scene: dict) -> float:
    code = tuple(int(value) for value in scene["code"])
    blurred = _blurred_bits(code)
    return float(abs(_hamming(blurred, BOW) - _hamming(blurred, HARP)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No unrendered aliases are claimed.  All nuisance fields are visible in
    # the final pixels and the code is the only analytic label input.
    return []
