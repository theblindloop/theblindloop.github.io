"""Pixel-only inverse arm for `stamp_piece_hole_balance`.

It receives one rendered PNG and nothing else: no scene, no block size, no stamp
origin, no analytic gold.  It segments the dark stamp, recovers its block
lattice from pixels, rebuilds the occupancy map, splits the ink into edge-joined
pieces, decides which pale patches are shut inside the ink, binds every shut
patch to the piece that rings it, and reports whether some piece holds none.  A
raster that does not show the declared blocky stamp is answered `abstain`.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK = 96
NEIGHBOURS = ((1, 0), (-1, 0), (0, 1), (0, -1))
DIAGONALS = ((1, 1), (1, -1), (-1, 1), (-1, -1))
MIN_GRID = 5
MAX_GRID = 16
MIN_BLOCK = 6
MAX_BLOCK = 24


def _components(mask, want):
    rows, cols = len(mask), len(mask[0])
    seen = [[False] * cols for _ in range(rows)]
    out = []
    for r in range(rows):
        for c in range(cols):
            if mask[r][c] == want and not seen[r][c]:
                stack = [(r, c)]
                seen[r][c] = True
                group = []
                while stack:
                    y, x = stack.pop()
                    group.append((y, x))
                    for dy, dx in NEIGHBOURS:
                        ny, nx = y + dy, x + dx
                        if 0 <= ny < rows and 0 <= nx < cols:
                            if not seen[ny][nx] and mask[ny][nx] == want:
                                seen[ny][nx] = True
                                stack.append((ny, nx))
                out.append(group)
    return out


def _read_grid(occupancy):
    """Return each piece's shut-hole tally when the structure is the declared one."""
    rows, cols = len(occupancy), len(occupancy[0])
    if not (MIN_GRID <= rows <= MAX_GRID and MIN_GRID <= cols <= MAX_GRID):
        return None
    pieces = _components(occupancy, 1)
    if len(pieces) < 2 or len(pieces) > 6 or any(len(p) < 4 for p in pieces):
        return None
    label = {}
    for index, group in enumerate(pieces):
        for cell in group:
            label[cell] = index
    for (y, x), index in label.items():
        for dy, dx in DIAGONALS:
            other = (y + dy, x + dx)
            if other in label and label[other] != index:
                return None
    shut, open_cells = [], set()
    for group in _components(occupancy, 0):
        if all(0 < y < rows - 1 and 0 < x < cols - 1 for y, x in group):
            shut.append(set(group))
        else:
            open_cells |= set(group)
    if not shut or len(shut) > 8:
        return None
    shares = [0] * len(pieces)
    for index, group in enumerate(shut):
        touching = set()
        for y, x in group:
            for dy, dx in NEIGHBOURS:
                other = (y + dy, x + dx)
                if other in label:
                    touching.add(label[other])
        if len(touching) != 1:
            return None
        shares[touching.copy().pop()] += 1
        for y, x in group:
            for dy, dx in DIAGONALS:
                other = (y + dy, x + dx)
                if other in open_cells:
                    return None
                for rival_index, rival in enumerate(shut):
                    if rival_index != index and other in rival:
                        return None
    return shares


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB")).astype(np.int16)
    dark = (rgb[:, :, 0] < DARK) & (rgb[:, :, 1] < DARK) & (rgb[:, :, 2] < DARK)
    if not dark.any():
        return "abstain"
    ys, xs = np.nonzero(dark)
    top, bottom = int(ys.min()), int(ys.max())
    left, right = int(xs.min()), int(xs.max())
    height = bottom - top + 1
    width = right - left + 1
    patch = dark[top : bottom + 1, left : right + 1]
    for block in range(MAX_BLOCK, MIN_BLOCK - 1, -1):
        if height % block or width % block:
            continue
        rows = height // block
        cols = width // block
        if not (MIN_GRID <= rows <= MAX_GRID and MIN_GRID <= cols <= MAX_GRID):
            continue
        folded = patch.reshape(rows, block, cols, block)
        totals = folded.sum(axis=(1, 3))
        area = block * block
        if not np.all((totals == 0) | (totals == area)):
            continue
        occupancy = (totals == area).astype(int).tolist()
        shares = _read_grid(occupancy)
        if shares is None:
            continue
        return "all" if min(shares) > 0 else "bare"
    return "abstain"
