"""Deterministic renderer for the Running-Parity Carpet world.

The analytic arm operates on three explicit visible fields.  It never derives
the answer from image measurements; the independent oracle does that in
``oracle.py`` from the final PNG alone.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 720
N = 7

BACKGROUND = (250, 248, 243)
PLAQUE = (226, 220, 201)
DARK = (45, 50, 58)
LIGHT = (245, 241, 226)
SOURCE_TAG = (105, 116, 128)
TAG_RGB = {
    "amber": (230, 156, 31),
    "cyan": (25, 177, 194),
    "violet": (139, 83, 205),
}
TAG_NAMES = tuple(TAG_RGB)

BOARD_POSITIONS = ((18, 18), (365, 18), (18, 365), (365, 365))
BOARD_SIZE = 337
GRID_OFFSET = 12
CELL = 44
TILE_INSET = 2
TAG_LEFT = 12
TAG_RIGHT = 324
TAG_TOP = 320
TAG_BOTTOM = 329


def _matrix(value: object, *, name: str) -> list[list[int]]:
    if not isinstance(value, list) or len(value) != N:
        raise ValueError(f"{name} must have {N} rows")
    result: list[list[int]] = []
    for row in value:
        if not isinstance(row, list) or len(row) != N:
            raise ValueError(f"{name} rows must have {N} cells")
        if any(bit not in (0, 1) for bit in row):
            raise ValueError(f"{name} cells must be binary")
        result.append([int(bit) for bit in row])
    return result


def _running_parity(source: list[list[int]]) -> list[list[int]]:
    output: list[list[int]] = []
    for row in source:
        state = 0
        emitted = []
        for bit in row:
            state ^= bit
            emitted.append(state)
        output.append(emitted)
    return output


def _flat(matrix: list[list[int]]) -> list[int]:
    return [bit for row in matrix for bit in row]


def _reshape(values: Iterable[int]) -> list[list[int]]:
    flat = list(values)
    if len(flat) != N * N:
        raise ValueError("wrong flattened matrix size")
    return [flat[index : index + N] for index in range(0, N * N, N)]


def _mutate_equal_ink(
    target: list[list[int]], rng: random.Random, prior: set[tuple[int, ...]]
) -> list[list[int]]:
    flat = _flat(target)
    ones = [index for index, bit in enumerate(flat) if bit]
    zeros = [index for index, bit in enumerate(flat) if not bit]
    if len(ones) < 3 or len(zeros) < 3:
        raise ValueError("target lacks enough bits for equal-ink distractors")
    for _ in range(200):
        changed = flat.copy()
        for index in rng.sample(ones, 3):
            changed[index] = 0
        for index in rng.sample(zeros, 3):
            changed[index] = 1
        key = tuple(changed)
        if key not in prior:
            prior.add(key)
            return _reshape(changed)
    raise RuntimeError("could not construct a distinct distractor")


def sample_scene(seed: int) -> dict:
    """Sample a balanced visible scene from one integer seed."""

    rng = random.Random(int(seed) ^ 0x52A91D73)
    for _ in range(400):
        dark_indices = set(rng.sample(range(N * N), 24))
        source = [
            [int(row * N + column in dark_indices) for column in range(N)]
            for row in range(N)
        ]
        row_counts = [sum(row) for row in source]
        target = _running_parity(source)
        target_ink = sum(_flat(target))
        if all(2 <= count <= 5 for count in row_counts) and 13 <= target_ink <= 36:
            break
    else:
        raise RuntimeError("could not sample a well-conditioned source")

    seen = {tuple(_flat(target))}
    distractor_a = _mutate_equal_ink(target, rng, seen)
    distractor_b = _mutate_equal_ink(target, rng, seen)

    answer_index = int(seed) % 3
    answer_tag = TAG_NAMES[answer_index]
    correct_position = (answer_index + (int(seed) // 3)) % 3
    candidates: list[list[list[int]]] = [distractor_a, distractor_b]
    candidates.insert(correct_position, target)

    remaining_tags = [tag for tag in TAG_NAMES if tag != answer_tag]
    if (int(seed) // 9) % 2:
        remaining_tags.reverse()
    tag_colors = [""] * 3
    tag_colors[correct_position] = answer_tag
    cursor = 0
    for position in range(3):
        if position != correct_position:
            tag_colors[position] = remaining_tags[cursor]
            cursor += 1

    scene = {
        "source": source,
        "candidates": candidates,
        "tag_colors": tag_colors,
    }
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a quarantined scene")
    return scene


def _scene_parts(scene: dict) -> tuple[list[list[int]], list[list[list[int]]], list[str]]:
    source = _matrix(scene["source"], name="source")
    raw_candidates = scene["candidates"]
    if not isinstance(raw_candidates, list) or len(raw_candidates) != 3:
        raise ValueError("candidates must contain exactly three matrices")
    candidates = [
        _matrix(candidate, name=f"candidates[{index}]")
        for index, candidate in enumerate(raw_candidates)
    ]
    raw_tags = scene["tag_colors"]
    if (
        not isinstance(raw_tags, list)
        or len(raw_tags) != 3
        or set(raw_tags) != set(TAG_NAMES)
    ):
        raise ValueError("tag_colors must be a permutation of the answer tags")
    return source, candidates, [str(tag) for tag in raw_tags]


def analytic_gold(scene: dict) -> str:
    source = _matrix(scene["source"], name="source")
    raw_candidates = scene["candidates"]
    raw_tags = scene["tag_colors"]
    if not isinstance(raw_candidates, list) or not isinstance(raw_tags, list):
        raise ValueError("candidate and tag fields must be lists")
    candidates = [
        _matrix(candidate, name=f"candidates[{index}]")
        for index, candidate in enumerate(raw_candidates)
    ]
    if len(candidates) != 3 or len(raw_tags) != 3 or set(raw_tags) != set(TAG_NAMES):
        raise ValueError("scene must contain three uniquely tagged proposals")
    target = _running_parity(source)
    matches = [index for index, candidate in enumerate(candidates) if candidate == target]
    if len(matches) != 1:
        raise ValueError("scene does not have one unique exact proposal")
    return str(raw_tags[matches[0]])


def margin(scene: dict) -> float:
    source, candidates, _ = _scene_parts(scene)
    target = _flat(_running_parity(source))
    distances = [
        sum(left != right for left, right in zip(target, _flat(candidate), strict=True))
        for candidate in candidates
    ]
    nonzero = [distance for distance in distances if distance]
    return float(min(nonzero)) if nonzero else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        source, candidates, tags = _scene_parts(scene)
        target = _running_parity(source)
        matches = sum(candidate == target for candidate in candidates)
        equal_ink = len({sum(_flat(candidate)) for candidate in candidates}) == 1
        return not (
            matches == 1
            and set(tags) == set(TAG_NAMES)
            and equal_ink
            and margin(scene) >= 4.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_board(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    matrix: list[list[int]],
    tag_rgb: tuple[int, int, int],
) -> None:
    x0, y0 = origin
    draw.rectangle((x0, y0, x0 + BOARD_SIZE - 1, y0 + BOARD_SIZE - 1), fill=DARK)
    for row in range(N):
        for column in range(N):
            left = x0 + GRID_OFFSET + column * CELL + TILE_INSET
            top = y0 + GRID_OFFSET + row * CELL + TILE_INSET
            right = x0 + GRID_OFFSET + (column + 1) * CELL - TILE_INSET - 1
            bottom = y0 + GRID_OFFSET + (row + 1) * CELL - TILE_INSET - 1
            fill = DARK if matrix[row][column] else LIGHT
            draw.rectangle((left, top, right, bottom), fill=fill)
    draw.rectangle(
        (x0 + TAG_LEFT, y0 + TAG_TOP, x0 + TAG_RIGHT, y0 + TAG_BOTTOM),
        fill=tag_rgb,
    )


def render(scene: dict) -> Image.Image:
    source = _matrix(scene["source"], name="source")
    raw_candidates = scene["candidates"]
    raw_tags = scene["tag_colors"]
    if not isinstance(raw_candidates, list) or not isinstance(raw_tags, list):
        raise ValueError("candidate and tag fields must be lists")
    candidates = [
        _matrix(candidate, name=f"candidates[{index}]")
        for index, candidate in enumerate(raw_candidates)
    ]
    if len(candidates) != 3 or len(raw_tags) != 3 or set(raw_tags) != set(TAG_NAMES):
        raise ValueError("scene must contain three uniquely tagged proposals")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((8, 8, WIDTH - 9, HEIGHT - 9), fill=PLAQUE)
    _draw_board(draw, BOARD_POSITIONS[0], source, SOURCE_TAG)
    for index, candidate in enumerate(candidates):
        _draw_board(
            draw,
            BOARD_POSITIONS[index + 1],
            candidate,
            TAG_RGB[str(raw_tags[index])],
        )
    return image
