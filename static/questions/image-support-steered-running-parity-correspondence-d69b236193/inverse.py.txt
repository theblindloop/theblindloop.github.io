"""Independent PNG-only inverse arm for Running-Parity Carpet.

This module deliberately duplicates no renderer imports and receives no scene.
It segments the public palette, discovers each board relative to its tag, reads
every cell, carries row-prefix parity, solves the full assignment, and finally
returns the matching tag name.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = 720
N = 7

BACKGROUND = np.array((250, 248, 243), dtype=np.uint8)
PLAQUE = np.array((226, 220, 201), dtype=np.uint8)
DARK = np.array((45, 50, 58), dtype=np.uint8)
LIGHT = np.array((245, 241, 226), dtype=np.uint8)
TAG_PALETTE = {
    "source": np.array((105, 116, 128), dtype=np.uint8),
    "amber": np.array((230, 156, 31), dtype=np.uint8),
    "cyan": np.array((25, 177, 194), dtype=np.uint8),
    "violet": np.array((139, 83, 205), dtype=np.uint8),
}


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(array == color, axis=2)


def _rect_for_color(array: np.ndarray, color: np.ndarray, name: str) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(_mask(array, color))
    if not len(xs):
        raise ValueError(f"missing {name} tag")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if len(xs) != (x1 - x0 + 1) * (y1 - y0 + 1):
        raise ValueError(f"{name} tag is not one solid rectangle")
    return x0, y0, x1, y1


def _discover_board(
    array: np.ndarray, dark: np.ndarray, tag_rect: tuple[int, int, int, int], tag: str
) -> tuple[tuple[int, int, int, int], list[list[int]]]:
    tx0, ty0, tx1, ty1 = tag_rect
    tag_width = tx1 - tx0 + 1
    tag_height = ty1 - ty0 + 1
    if tag_width < 250 or not 6 <= tag_height <= 20:
        raise ValueError(f"{tag} tag has implausible dimensions")

    center_x = (tx0 + tx1) // 2
    search_half = int(round(tag_width * 0.55))
    sx0 = max(0, center_x - search_half)
    sx1 = min(array.shape[1] - 1, center_x + search_half)
    # The tag sits along the board's lower rim.  A tag-width-scaled window
    # reaches the board's upper rim without leaking into the board above.
    sy0 = max(0, ty0 - int(round(tag_width * 1.03)))
    sy1 = min(array.shape[0] - 1, ty1 + tag_height)
    local_y, local_x = np.nonzero(dark[sy0 : sy1 + 1, sx0 : sx1 + 1])
    if not len(local_x):
        raise ValueError(f"cannot find the board carrying {tag}")
    x0, x1 = sx0 + int(local_x.min()), sx0 + int(local_x.max())
    y0, y1 = sy0 + int(local_y.min()), sy0 + int(local_y.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if abs(width - height) > 2 or not 300 <= width <= 370:
        raise ValueError(f"{tag} board is not a well-formed square")

    board = array[y0 : y1 + 1, x0 : x1 + 1]
    allowed = _mask(board, DARK) | _mask(board, LIGHT) | _mask(board, TAG_PALETTE[tag])
    if not bool(np.all(allowed)):
        raise ValueError(f"{tag} board contains erased or foreign pixels")

    scale = width / 337.0
    offset = int(round(12 * scale))
    cell = int(round(44 * scale))
    inset = max(1, int(round(2 * scale)))
    decoded: list[list[int]] = []
    for row in range(N):
        values = []
        for column in range(N):
            left = offset + column * cell + inset
            top = offset + row * cell + inset
            right = offset + (column + 1) * cell - inset
            bottom = offset + (row + 1) * cell - inset
            tile = board[top:bottom, left:right]
            if tile.size == 0:
                raise ValueError("empty decoded tile")
            dark_fraction = float(np.mean(_mask(tile, DARK)))
            light_fraction = float(np.mean(_mask(tile, LIGHT)))
            if dark_fraction >= 0.995 and light_fraction <= 0.005:
                values.append(1)
            elif light_fraction >= 0.995 and dark_fraction <= 0.005:
                values.append(0)
            else:
                raise ValueError(f"{tag} board has an ambiguous or erased tile")
        decoded.append(values)
    return (x0, y0, x1, y1), decoded


def _running_parity(source: list[list[int]]) -> list[list[int]]:
    output: list[list[int]] = []
    for row in source:
        state = 0
        emitted = []
        for bit in row:
            state = (state + bit) % 2
            emitted.append(state)
        output.append(emitted)
    return output


def decision_from_image(image: Image.Image) -> str:
    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    rgb = image.convert("RGB")
    if rgb.size != (CANVAS, CANVAS):
        raise ValueError("unexpected raster dimensions")
    array = np.asarray(rgb, dtype=np.uint8)

    border = np.concatenate(
        (array[:5].reshape(-1, 3), array[-5:].reshape(-1, 3), array[:, :5].reshape(-1, 3), array[:, -5:].reshape(-1, 3)),
        axis=0,
    )
    if not bool(np.all(border == BACKGROUND)):
        raise ValueError("outer raster boundary is corrupted")
    allowed_colors = [BACKGROUND, PLAQUE, DARK, LIGHT, *TAG_PALETTE.values()]
    allowed = np.zeros(array.shape[:2], dtype=bool)
    for color in allowed_colors:
        allowed |= _mask(array, color)
    if not bool(np.all(allowed)):
        raise ValueError("raster contains a color outside the public palette")

    tag_rects = {
        name: _rect_for_color(array, color, name) for name, color in TAG_PALETTE.items()
    }
    dimensions = {(x1 - x0 + 1, y1 - y0 + 1) for x0, y0, x1, y1 in tag_rects.values()}
    if len(dimensions) != 1:
        raise ValueError("tag strips do not share one visible size")

    dark = _mask(array, DARK)
    boards: dict[str, tuple[tuple[int, int, int, int], list[list[int]]]] = {}
    for tag, rect in tag_rects.items():
        boards[tag] = _discover_board(array, dark, rect, tag)
    board_rects = [value[0] for value in boards.values()]
    if len(set(board_rects)) != 4:
        raise ValueError("tags do not identify four distinct boards")
    sizes = {(x1 - x0 + 1, y1 - y0 + 1) for x0, y0, x1, y1 in board_rects}
    if len(sizes) != 1:
        raise ValueError("the four discovered boards have unequal geometry")

    source_rect = boards["source"][0]
    source_center = ((source_rect[0] + source_rect[2]) / 2, (source_rect[1] + source_rect[3]) / 2)
    other_centers = [
        ((rect[0] + rect[2]) / 2, (rect[1] + rect[3]) / 2)
        for name, (rect, _) in boards.items()
        if name != "source"
    ]
    if any(source_center[0] >= center[0] and source_center[1] >= center[1] for center in other_centers):
        raise ValueError("neutral source is not the upper-left board")

    source = boards["source"][1]
    target = _running_parity(source)
    proposal_ink = {sum(sum(row) for row in boards[tag][1]) for tag in ("amber", "cyan", "violet")}
    if len(proposal_ink) != 1:
        raise ValueError("proposal mosaics leak the answer through total ink")
    matches = [tag for tag in ("amber", "cyan", "violet") if boards[tag][1] == target]
    if len(matches) != 1:
        raise ValueError("pixel assignment is not unique")
    return matches[0]
