"""Independent pixel-only inverse arm for the maze-path-panel world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or renderer state. The decision is recovered from the
raster alone:

1. classify each pixel as WALL (dark, the black maze walls / panel borders) or
   OPEN (white corridor or the colored dots);
2. detect the green dots and red dots by color and take each connected blob's
   centroid, sorted left-to-right so that index i pairs the green dot in panel
   i with the red dot in panel i;
3. flood-fill the OPEN pixels from each green centroid and report the panel
   connected exactly when its red centroid lies in the same OPEN component.

Exactly one panel is connected, so the answer is its left/center/right label.
"""

from __future__ import annotations

from PIL import Image

import numpy as np

ORACLE_VERSION = "maze_path_panels-oracle-0.4.0"

_WALL_MAX = 100.0
_COLOR_TOLERANCE = 90.0
_MIN_DOT_AREA = 8
_LABELS = ("left", "center", "right")
_GREEN = np.array([50.0, 185.0, 60.0])
_RED = np.array([230.0, 45.0, 45.0])


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: float) -> np.ndarray:
    diff = arr.astype(np.float32) - target
    return np.sqrt((diff * diff).sum(axis=-1)) < tolerance


def _components(mask: np.ndarray) -> tuple[list[tuple[float, float]], int]:
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    current = 1
    centers: list[tuple[float, float]] = []
    stack: list[tuple[int, int]] = []
    areas: list[int] = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and labels[y, x] == 0:
                labels[y, x] = current
                stack.append((x, y))
                sx, sy, cnt = 0.0, 0.0, 0
                while stack:
                    px, py = stack.pop()
                    sx += px
                    sy += py
                    cnt += 1
                    for nx, ny in ((px + 1, py), (px - 1, py), (px, py + 1), (px, py - 1)):
                        if 0 <= nx < w and 0 <= ny < h and mask[ny, nx] and labels[ny, nx] == 0:
                            labels[ny, nx] = current
                            stack.append((nx, ny))
                centers.append((sx / cnt, sy / cnt))
                areas.append(cnt)
                current += 1
    kept: list[tuple[float, float]] = []
    for idx, area in enumerate(areas):
        if area >= _MIN_DOT_AREA:
            kept.append(centers[idx])
    return kept, current - 1


def _dot_centers(arr: np.ndarray, target: np.ndarray) -> list[tuple[float, float]]:
    mask = _color_mask(arr, target, _COLOR_TOLERANCE)
    centers, _ = _components(mask)
    centers.sort(key=lambda p: p[0])
    return centers


def _flood_reaches(open_mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    h, w = open_mask.shape
    reached = np.zeros((h, w), dtype=bool)
    sx, sy = int(round(start[0])), int(round(start[1]))
    sx = max(0, min(w - 1, sx))
    sy = max(0, min(h - 1, sy))
    if not open_mask[sy, sx]:
        return reached
    stack = [(sx, sy)]
    reached[sy, sx] = True
    while stack:
        px, py = stack.pop()
        for nx, ny in ((px + 1, py), (px - 1, py), (px, py + 1), (px, py - 1)):
            if 0 <= nx < w and 0 <= ny < h and open_mask[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                stack.append((nx, ny))
    return reached


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    open_mask = arr.max(axis=-1) >= _WALL_MAX
    greens = _dot_centers(arr, _GREEN)
    reds = _dot_centers(arr, _RED)
    n = min(len(greens), len(reds))
    if n == 0:
        return _LABELS[0]
    for i in range(n):
        reached = _flood_reaches(open_mask, greens[i])
        rx, ry = int(round(reds[i][0])), int(round(reds[i][1]))
        rx = max(0, min(open_mask.shape[1] - 1, rx))
        ry = max(0, min(open_mask.shape[0] - 1, ry))
        if reached[ry, rx]:
            return _LABELS[i]
    return _LABELS[0]
