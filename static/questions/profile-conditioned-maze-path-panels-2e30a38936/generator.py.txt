"""Maze-path panel world renderer.

Three maze panels sit side by side on a white background. Each panel is a
black-bordered rectangle of black wall cells on white, containing exactly one
green dot (start) and one red dot (goal). A panel is CONNECTING exactly when a
4-connected open (non-wall) corridor joins its green dot to its red dot.
Exactly one of the three panels is connecting.

Decision : which panel (left / center / right) has an open corridor from its
green dot to its red dot.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": W, "height": H,
        "cs": cell size, "border": border thickness,
        "nc": grid columns, "nr": grid rows,
        "inner_origins": [[ix, iy], ...] per panel,
        "dot_r": dot radius,
        "cells": [panel][row][col] bool (wall=True),
        "g": [panel] = [col, row] green-dot cell,
        "r": [panel] = [col, row] red-dot cell,
        "connected_panel": 0/1/2,
    }
"""

from __future__ import annotations

import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "maze_path_panels-0.4.0"

WHITE = (255, 255, 255)
BLACK = (25, 25, 25)
GREEN = (50, 185, 60)
RED = (230, 45, 45)

_WIDTH = 900
_HEIGHT = 230
_CS = 34
_BORDER = 4
_NC = 6
_NR = 5
_GAP = 42
_DOT_R = 7
_PANEL_LABELS = ("left", "center", "right")


def _layout():
    pw = _NC * _CS + 2 * _BORDER
    ph = _NR * _CS + 2 * _BORDER
    total_w = 3 * pw + 2 * _GAP
    left0 = (_WIDTH - total_w) // 2
    top0 = (_HEIGHT - ph) // 2
    origins = []
    for i in range(3):
        px = left0 + i * (pw + _GAP)
        py = top0
        origins.append([px + _BORDER, py + _BORDER])
    return pw, ph, origins


def _reachable(cells, start, goal):
    nr = len(cells)
    nc = len(cells[0])
    sc, sr = start
    gc, gr = goal
    if cells[sr][sc]:
        return False
    seen = {(sr, sc)}
    stack = [(sr, sc)]
    while stack:
        r, c = stack.pop()
        if (r, c) == (gr, gc):
            return True
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nr2, nc2 = r + dr, c + dc
            if 0 <= nr2 < nr and 0 <= nc2 < nc and not cells[nr2][nc2] and (nr2, nc2) not in seen:
                seen.add((nr2, nc2))
                stack.append((nr2, nc2))
    return (gr, gc) in seen


def sample_scene(
    rng: random.Random,
    width: int = _WIDTH,
    height: int = _HEIGHT,
) -> dict:
    nr = _NR
    nc = _NC
    _, _, origins = _layout()

    cells: list = []
    g: list = []
    r: list = []
    connected = rng.randrange(3)

    for panel in range(3):
        grid = [[False] * nc for _ in range(nr)]
        if panel == connected:
            sc = rng.randrange(nc)
            sr = rng.randrange(nr)
            gc = rng.randrange(nc)
            gr = rng.randrange(nr)
            while (gc, gr) == (sc, sr):
                gc = rng.randrange(nc)
                gr = rng.randrange(nr)
            cur = [sc, sr]
            protected = {(sr, sc)}
            grid[sr][sc] = False
            while (cur[0], cur[1]) != (gc, gr):
                dcol = (gc > cur[0]) - (gc < cur[0])
                drow = (gr > cur[1]) - (gr < cur[1])
                moves = []
                if dcol:
                    moves.append((dcol, 0))
                if drow:
                    moves.append((0, drow))
                if not moves:
                    moves = [(1, 0), (-1, 0), (0, 1), (0, -1)]
                dc, dr = rng.choice(moves)
                cur[0] += dc
                cur[1] += dr
                grid[cur[1]][cur[0]] = False
                protected.add((cur[1], cur[0]))
            protected.add((gr, gc))
            prob = rng.uniform(0.38, 0.55)
            for rr in range(nr):
                for cc in range(nc):
                    if (rr, cc) in protected:
                        continue
                    grid[rr][cc] = rng.random() < prob
            g.append([sc, sr])
            r.append([gc, gr])
        else:
            cut_col = rng.randrange(1, nc - 1)
            sc = rng.randrange(0, cut_col)
            sr = rng.randrange(nr)
            gc = rng.randrange(cut_col + 1, nc)
            gr = rng.randrange(nr)
            grid[sr][sc] = False
            grid[gr][gc] = False
            prob = rng.uniform(0.38, 0.55)
            for rr in range(nr):
                for cc in range(nc):
                    if cc == cut_col:
                        grid[rr][cc] = True
                    elif (rr, cc) in {(sr, sc), (gr, gc)}:
                        grid[rr][cc] = False
                    else:
                        grid[rr][cc] = rng.random() < prob
            g.append([sc, sr])
            r.append([gc, gr])
        cells.append(grid)

    return {
        "width": width,
        "height": height,
        "cs": _CS,
        "border": _BORDER,
        "nc": nc,
        "nr": nr,
        "inner_origins": origins,
        "dot_r": _DOT_R,
        "cells": cells,
        "g": g,
        "r": r,
        "connected_panel": connected,
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    cs = int(scene["cs"])
    border = int(scene["border"])
    nc = int(scene["nc"])
    nr = int(scene["nr"])
    origins = scene["inner_origins"]
    dot_r = int(scene["dot_r"])
    cells = scene["cells"]
    g = scene["g"]
    r = scene["r"]

    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), WHITE)
    draw = ImageDraw.Draw(img)
    for panel in range(3):
        ix, iy = origins[panel]
        pw = nc * cs + 2 * border
        ph = nr * cs + 2 * border
        draw.rectangle(
            [ix * ss - border * ss, iy * ss - border * ss,
             ix * ss + (nc * cs + border) * ss, iy * ss + (nr * cs + border) * ss],
            fill=BLACK,
        )
        draw.rectangle(
            [ix * ss, iy * ss, (ix + nc * cs) * ss, (iy + nr * cs) * ss], fill=WHITE
        )
        for rr in range(nr):
            for cc in range(nc):
                if cells[panel][rr][cc]:
                    x0 = ix + cc * cs
                    y0 = iy + rr * cs
                    draw.rectangle(
                        [x0 * ss, y0 * ss, (x0 + cs) * ss, (y0 + cs) * ss], fill=BLACK
                    )
    for panel in range(3):
        ix, iy = origins[panel]
        gc, gr = g[panel]
        rc, rr = r[panel]
        gx = ix + gc * cs + cs / 2.0
        gy = iy + gr * cs + cs / 2.0
        rx = ix + rc * cs + cs / 2.0
        ry = iy + rr * cs + cs / 2.0
        ddr = dot_r * ss
        draw.ellipse(
            [int(gx * ss) - ddr, int(gy * ss) - ddr, int(gx * ss) + ddr, int(gy * ss) + ddr],
            fill=GREEN,
        )
        draw.ellipse(
            [int(rx * ss) - ddr, int(ry * ss) - ddr, int(rx * ss) + ddr, int(ry * ss) + ddr],
            fill=RED,
        )
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    cells = scene["cells"]
    g = scene["g"]
    r = scene["r"]
    for panel in range(len(cells)):
        if _reachable(cells[panel], g[panel], r[panel]):
            return _PANEL_LABELS[panel]
    return _PANEL_LABELS[0]


def _dot_wall_clearance(cs: float, dot_r: float, grid, origin, cr: tuple) -> float:
    """Nearest distance from the dot center to any wall-pixel square, minus radius."""
    cc, rr = cr
    ix, iy = origin
    cx = ix + cc * cs + cs / 2.0
    cy = iy + rr * cs + cs / 2.0
    best = float("inf")
    for row in range(len(grid)):
        for col in range(len(grid[0])):
            if not grid[row][col]:
                continue
            x0 = ix + col * cs
            y0 = iy + row * cs
            dx = max(x0 - cx, 0.0, cx - (x0 + cs))
            dy = max(y0 - cy, 0.0, cy - (y0 + cs))
            best = min(best, math_hypot(dx, dy))
    return best - dot_r


def margin(scene: dict) -> float:
    cs = float(scene["cs"])
    dot_r = float(scene["dot_r"])
    best = float("inf")
    for panel in range(3):
        origin = scene["inner_origins"][panel]
        grid = scene["cells"][panel]
        best = min(best, _dot_wall_clearance(cs, dot_r, grid, origin, tuple(scene["g"][panel])))
        best = min(best, _dot_wall_clearance(cs, dot_r, grid, origin, tuple(scene["r"][panel])))
    return float(best)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> Iterator[tuple[str, dict]]:
    return iter(())


def math_hypot(a: float, b: float) -> float:
    import math

    return math.hypot(a, b)
