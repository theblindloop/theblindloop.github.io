"""Deterministic renderer and analytic arm for Paired Cut-Station Agreement.

The scene contains two widely separated station graphs.  Within each graph the
red station has degree two.  Removing that vertex either does or does not split
the remaining stations.  Gold compares those two independently computed cut
effects.  The two graph classes deliberately share their node positions, node
count, edge count, red degree, degree multiset, and total Euclidean edge length.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480
BACKGROUND = (250, 248, 242)
DARK = (29, 50, 72)
RED = (207, 55, 61)
STATION_FILL = (222, 233, 240)
STATION_RADIUS = 11
STROKE_WIDTH = 10
OUTLINE_WIDTH = 3
SUPERSAMPLE = 3

# Shared seven-station embedding.  Vertex 0 is the marked station.  The two
# final edges determine its cut effect while preserving all cheap graph counts.
BASE_NODES = (
    (0, 0),
    (-28, 0),
    (-55, -55),
    (-55, 55),
    (28, 0),
    (55, -55),
    (55, 55),
)
SHARED_EDGES = ((0, 1), (0, 4), (1, 2), (1, 3), (4, 5), (4, 6))
CUT_EDGES = ((2, 3), (5, 6))
NONCUT_EDGES = ((2, 5), (3, 6))


def _transformed_nodes(
    center: tuple[int, int], quarter_turns: int, mirror: bool
) -> list[list[int]]:
    result: list[list[int]] = []
    for source_x, source_y in BASE_NODES:
        x, y = source_x, source_y
        if mirror:
            x = -x
        for _ in range(quarter_turns % 4):
            x, y = -y, x
        result.append([center[0] + x, center[1] + y])
    return result


def _panel(center: tuple[int, int], is_cut: bool, rng: random.Random) -> dict:
    edges = list(SHARED_EDGES + (CUT_EDGES if is_cut else NONCUT_EDGES))
    # Edge order affects neither pixels (all strokes share one color) nor gold,
    # but shuffling prevents the serialized scene from encoding a fixed suffix.
    rng.shuffle(edges)
    return {
        "nodes": _transformed_nodes(center, rng.randrange(4), bool(rng.randrange(2))),
        "edges": [list(edge) for edge in edges],
        "red": 0,
        "station_radius": STATION_RADIUS,
        "stroke_width": STROKE_WIDTH,
        "outline_width": OUTLINE_WIDTH,
    }


def sample_scene(seed: int) -> dict:
    """Sample one balanced, fully visible scene from an integer seed.

    Consecutive seeds cycle through cut/cut, cut/non-cut, non-cut/cut, and
    non-cut/non-cut.  Thus every 24-scene public stress run is exactly balanced
    for the equality decision and for panel-local cut status.
    """

    rng = random.Random(int(seed))
    pattern = int(seed) % 4
    statuses = ((True, True), (True, False), (False, True), (False, False))[pattern]
    if rng.randrange(2):
        base_centers = ((105, 100), (535, 380))
    else:
        base_centers = ((105, 380), (535, 100))
    centers = [
        (x + rng.randint(-8, 8), y + rng.randint(-8, 8)) for x, y in base_centers
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "dark": list(DARK),
        "red": list(RED),
        "station_fill": list(STATION_FILL),
        "panels": [_panel(centers[i], statuses[i], rng) for i in range(2)],
    }


def _remaining_connected(panel: dict) -> bool:
    nodes = panel["nodes"]
    removed = int(panel["red"])
    remaining = [index for index in range(len(nodes)) if index != removed]
    adjacency = {index: set() for index in remaining}
    for raw_a, raw_b in panel["edges"]:
        a, b = int(raw_a), int(raw_b)
        if a == removed or b == removed:
            continue
        adjacency[a].add(b)
        adjacency[b].add(a)
    seen = {remaining[0]}
    stack = [remaining[0]]
    while stack:
        current = stack.pop()
        for neighbor in adjacency[current] - seen:
            seen.add(neighbor)
            stack.append(neighbor)
    return len(seen) == len(remaining)


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    effects: list[bool] = []
    # Keep the analytic scene access explicit here.  This is intentionally
    # independent of any raster measurement and lets the public rendered-field
    # audit see that ``panels`` is the sole label input.
    for panel in panels:
        node_count = len(panel["nodes"])
        removed = int(panel["red"])
        remaining = [index for index in range(node_count) if index != removed]
        adjacency = {index: set() for index in remaining}
        for raw_a, raw_b in panel["edges"]:
            a, b = int(raw_a), int(raw_b)
            if a == removed or b == removed:
                continue
            adjacency[a].add(b)
            adjacency[b].add(a)
        seen = {remaining[0]}
        stack = [remaining[0]]
        while stack:
            current = stack.pop()
            for neighbor in adjacency[current] - seen:
                seen.add(neighbor)
                stack.append(neighbor)
        effects.append(len(seen) == len(remaining))
    return "yes" if effects[0] == effects[1] else "no"


def _point_segment_distance(
    point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def margin(scene: dict) -> float:
    """Return the minimum visible topology-preservation clearance in pixels."""

    clearances: list[float] = []
    for panel in scene["panels"]:
        nodes = [tuple(map(float, point)) for point in panel["nodes"]]
        radius = float(panel["station_radius"])
        half_stroke = float(panel["stroke_width"]) / 2.0
        clearances.append(half_stroke)
        for i, first in enumerate(nodes):
            for second in nodes[i + 1 :]:
                clearances.append(math.dist(first, second) - 2.0 * radius)
        for raw_a, raw_b in panel["edges"]:
            a, b = int(raw_a), int(raw_b)
            for index, point in enumerate(nodes):
                if index not in (a, b):
                    clearances.append(
                        _point_segment_distance(point, nodes[a], nodes[b])
                        - radius
                        - half_stroke
                    )
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The submitted latent-alias mode is not_applicable: transformations are
    # sampled into the visible node coordinates rather than retained as aliases.
    return []


def _scaled_points(points: Iterable[Iterable[int]], scale: int) -> list[tuple[int, int]]:
    return [(round(float(x) * scale), round(float(y) * scale)) for x, y in points]


def render(scene: dict) -> Image.Image:
    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    background = tuple(scene.get("background", BACKGROUND))
    dark = tuple(scene.get("dark", DARK))
    red = tuple(scene.get("red", RED))
    station_fill = tuple(scene.get("station_fill", STATION_FILL))
    panels = scene["panels"]

    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    for panel in panels:
        nodes = _scaled_points(panel["nodes"], scale)
        line_width = max(1, round(float(panel["stroke_width"]) * scale))
        for raw_a, raw_b in panel["edges"]:
            draw.line([nodes[int(raw_a)], nodes[int(raw_b)]], fill=dark, width=line_width)

        radius = round(float(panel["station_radius"]) * scale)
        outline_width = max(1, round(float(panel["outline_width"]) * scale))
        marked = int(panel["red"])
        for index, (x, y) in enumerate(nodes):
            box = (x - radius, y - radius, x + radius, y + radius)
            fill = red if index == marked else station_fill
            draw.ellipse(box, fill=fill, outline=dark, width=outline_width)

    return image.resize((width, height), Image.Resampling.LANCZOS)
