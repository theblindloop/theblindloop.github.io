from __future__ import annotations

import math

import numpy as np
from PIL import Image

TEAL = np.array((0, 137, 145), dtype=float)
AMBER = np.array((232, 142, 28), dtype=float)


def _fit_segment(points: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    if len(points) < 80:
        raise ValueError("colored segment missing")
    center = points.mean(axis=0)
    _, _, vh = np.linalg.svd(points - center, full_matrices=False)
    axis = vh[0]
    projections = (points - center) @ axis
    lo, hi = np.percentile(projections, (0.5, 99.5))
    return center + lo * axis, center + hi * axis


def _score(teal_points: np.ndarray, amber_points: np.ndarray) -> float:
    t0, t1 = _fit_segment(teal_points)
    a0, a1 = _fit_segment(amber_points)
    av = a1 - a0
    length = float(np.linalg.norm(av))
    scores = []
    for endpoint in (t0, t1):
        raw_t = float(np.dot(endpoint - a0, av) / np.dot(av, av))
        clipped = min(1.0, max(0.0, raw_t))
        nearest = a0 + clipped * av
        distance = float(np.linalg.norm(endpoint - nearest))
        continuation = min(max(raw_t, 0.0), max(1.0 - raw_t, 0.0)) * length
        outside = 80.0 if not 0.0 < raw_t < 1.0 else 0.0
        scores.append(distance + max(0.0, 30.0 - continuation) + outside)
    return min(scores)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=float)
    h, w = array.shape[:2]
    masks = {
        "teal": np.linalg.norm(array - TEAL, axis=2) < 35.0,
        "amber": np.linalg.norm(array - AMBER, axis=2) < 35.0,
    }
    regions = ((0, h // 2, 0, w // 2), (0, h // 2, w // 2, w), (h // 2, h, 0, w // 2), (h // 2, h, w // 2, w))
    scores = []
    for y0, y1, x0, x1 in regions:
        fitted = {}
        for name, mask in masks.items():
            yy, xx = np.nonzero(mask[y0:y1, x0:x1])
            fitted[name] = np.column_stack((xx + x0, yy + y0)).astype(float)
        scores.append(_score(fitted["teal"], fitted["amber"]))
    order = np.argsort(scores)
    if float(scores[int(order[1])] - scores[int(order[0])]) < 7.0:
        raise ValueError("ambiguous junction decision")
    return "ABCD"[int(order[0])]
