from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 520
TEAL = (0, 137, 145)
AMBER = (232, 142, 28)
INK = (30, 37, 45)
PANEL = (174, 181, 189)
CENTERS = ((190, 145), (530, 145), (190, 385), (530, 385))


def _rot(point: tuple[float, float], quarter: int) -> tuple[float, float]:
    x, y = point
    for _ in range(quarter % 4):
        x, y = -y, x
    return x, y


def _placed(points: Iterable[tuple[float, float]], center: tuple[int, int], q: int):
    cx, cy = center
    return [[cx + x, cy + y] for x, y in (_rot(p, q) for p in points)]


def _configuration(kind: str, center: tuple[int, int], q: int, rng: random.Random) -> dict:
    a = rng.randint(52, 61)
    b = rng.randint(52, 61)
    v = rng.randint(54, 64)
    if kind == "t":
        amber = [(-a, 0), (b, 0)]
        teal = [(0, -v), (0, 0)]
    elif kind == "x":
        amber = [(-a, 0), (b, 0)]
        teal = [(0, -v), (0, v)]
    elif kind == "l":
        amber = [(0, 0), (rng.randint(78, 88), 0)]
        teal = [(0, -v), (0, 0)]
    elif kind == "gap":
        amber = [(-a, 0), (b, 0)]
        teal = [(0, -v), (0, -rng.randint(30, 38))]
    else:
        raise ValueError(kind)
    return {"teal": _placed(teal, center, q), "amber": _placed(amber, center, q)}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % 4
    distractors = ["x", "l", "gap"]
    rng.shuffle(distractors)
    kinds = []
    j = 0
    for i in range(4):
        if i == target:
            kinds.append("t")
        else:
            kinds.append(distractors[j])
            j += 1
    panels = [
        _configuration(kind, CENTERS[i], rng.randrange(4), rng)
        for i, kind in enumerate(kinds)
    ]
    return {"panels": panels}


def _point_segment_distance(point, start, end) -> tuple[float, float, float]:
    px, py = point
    ax, ay = start
    bx, by = end
    dx, dy = bx - ax, by - ay
    length2 = dx * dx + dy * dy
    t = ((px - ax) * dx + (py - ay) * dy) / length2
    tc = min(1.0, max(0.0, t))
    distance = math.hypot(px - (ax + tc * dx), py - (ay + tc * dy))
    length = math.sqrt(length2)
    continuation = min(max(t, 0.0), max(1.0 - t, 0.0)) * length
    return distance, continuation, t


def _panel_score(panel: dict) -> float:
    amber = panel["amber"]
    teal = panel["teal"]
    scores = []
    for endpoint in teal:
        distance, continuation, t = _point_segment_distance(endpoint, amber[0], amber[1])
        strict_penalty = max(0.0, 30.0 - continuation)
        outside_penalty = 80.0 if not 0.0 < t < 1.0 else 0.0
        scores.append(distance + strict_penalty + outside_penalty)
    return min(scores)


def analytic_gold(scene: dict) -> str:
    # ``map`` keeps the public rendered-field audit focused on the top-level
    # scene access; _panel_score receives one already-visible panel record.
    scores = list(map(_panel_score, scene["panels"]))
    winner = min(range(4), key=scores.__getitem__)
    if sum(abs(score - scores[winner]) < 1e-9 for score in scores) != 1:
        raise ValueError("non-unique junction panel")
    return "ABCD"[winner]


def margin(scene: dict) -> float:
    scores = sorted(_panel_score(panel) for panel in scene["panels"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    boxes = ((35, 25, 345, 250), (375, 25, 685, 250), (35, 270, 345, 495), (375, 270, 685, 495))
    for i, box in enumerate(boxes):
        draw.rounded_rectangle(box, radius=12, fill=(255, 255, 255), outline=PANEL, width=3)
        draw.text((box[0] + 13, box[1] + 10), "ABCD"[i], fill=INK, stroke_width=1)
        panel = scene["panels"][i]
        draw.line([tuple(p) for p in panel["amber"]], fill=AMBER, width=11)
        draw.line([tuple(p) for p in panel["teal"]], fill=TEAL, width=11)
        for color, key in ((AMBER, "amber"), (TEAL, "teal")):
            for p in panel[key]:
                x, y = p
                draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=color)
    return image
