"""Independent pixel-only inverse arm for the rung-removal-replay world.

Given only the final PNG, this module

1. locates the vertical ladder lines as the dark-gray columns carrying a tall
   stack of ink;
2. probes the midpoint column of every neighbouring pair of lines and reads
   off the rungs that cross it, splitting them into blue rungs and the single
   red rung;
3. locates the pure-black start arrow and snaps it to the nearest line;
4. replays the descent with the red rung deleted and reports the bottom slot.

It imports no renderer / prompts / generate / verify / analytic-gold code and
never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _near(arr: np.ndarray, color: tuple[int, int, int], tol: int) -> np.ndarray:
    diff = np.abs(arr.astype(int) - np.array(color, dtype=int))
    return (diff <= tol).all(axis=-1)


def _runs(column: np.ndarray) -> list[float]:
    """Centres of the contiguous True runs down a boolean column."""
    centres: list[float] = []
    start = None
    for y, flag in enumerate(column):
        if flag and start is None:
            start = y
        elif not flag and start is not None:
            centres.append((start + y - 1) / 2.0)
            start = None
    if start is not None:
        centres.append((start + len(column) - 1) / 2.0)
    return centres


def _line_columns(gray: np.ndarray) -> list[int]:
    counts = gray.sum(axis=0)
    threshold = max(40, int(0.35 * gray.shape[0]))
    columns = [int(x) for x in np.nonzero(counts >= threshold)[0]]
    groups: list[list[int]] = []
    for x in columns:
        if groups and x - groups[-1][-1] <= 3:
            groups[-1].append(x)
        else:
            groups.append([x])
    return [int(round(sum(group) / len(group))) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    """Return the slot label reached after the red rung is deleted."""
    arr = np.asarray(image.convert("RGB"))
    gray = _near(arr, (72, 72, 78), 30)
    blue = _near(arr, (38, 92, 176), 40)
    red = _near(arr, (198, 44, 44), 40)
    black = (arr.astype(int).sum(axis=-1) <= 24)

    lines = _line_columns(gray)
    if len(lines) < 2:
        return "slot-1"

    rungs: list[tuple[float, int, bool]] = []  # (y, left line index, is_red)
    for i in range(len(lines) - 1):
        xm = int(round((lines[i] + lines[i + 1]) / 2.0))
        for y in _runs(blue[:, xm]):
            rungs.append((float(y), i, False))
        for y in _runs(red[:, xm]):
            rungs.append((float(y), i, True))

    ys, xs = np.nonzero(black)
    if xs.size:
        marker_x = float(xs.mean())
        current = int(min(range(len(lines)), key=lambda i: abs(lines[i] - marker_x)))
    else:
        current = 0

    for y, left, is_red in sorted(rungs, key=lambda item: (item[0], item[1])):
        if is_red:
            continue
        if current == left:
            current = left + 1
        elif current == left + 1:
            current = left
    return f"slot-{current + 1}"
