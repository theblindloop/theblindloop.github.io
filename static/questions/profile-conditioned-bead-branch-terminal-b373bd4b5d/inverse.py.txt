"""Independent image-only inverse for the bead-guided route tree."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

PALETTE = {
    "green": (32, 157, 88),
    "gray": (126, 132, 139),
    "cyan": (0, 169, 206),
    "orange": (235, 126, 36),
    "purple": (143, 82, 190),
    "blue": (48, 103, 210),
    "red": (210, 55, 63),
}


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], radius: float = 42.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        sx = sy = n = 0
        while stack:
            yy, xx = stack.pop()
            sx += xx
            sy += yy
            n += 1
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if n >= minimum:
            out.append((sx / n, sy / n, n))
    return out


def _distance_to_segment(p, a, b) -> tuple[float, float]:
    vx, vy = b[0] - a[0], b[1] - a[1]
    denom = vx * vx + vy * vy
    t = ((p[0] - a[0]) * vx + (p[1] - a[1]) * vy) / denom
    q = (a[0] + t * vx, a[1] + t * vy)
    return math.dist(p, q), t


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    starts = _components(_mask(arr, PALETTE["green"]), 120)
    junctions = _components(_mask(arr, PALETTE["gray"]), 120)
    beads = _components(_mask(arr, PALETTE["cyan"]), 50)
    terminals = []
    for name in ("orange", "purple", "blue", "red"):
        comps = _components(_mask(arr, PALETTE[name]), 150)
        if len(comps) != 1:
            raise ValueError(f"expected one {name} terminal")
        terminals.append((comps[0][0], comps[0][1], name))
    if len(starts) != 1 or len(junctions) != 3 or len(beads) != 12:
        raise ValueError("required start, junction, or bead markers are missing")

    start = starts[0][:2]
    js = sorted((c[:2] for c in junctions), key=lambda p: p[0])
    root = js[0]
    children = sorted(js[1:], key=lambda p: p[1])
    leaves = sorted(terminals, key=lambda p: p[1])
    if not (start[0] < root[0] < min(p[0] for p in children) < min(p[0] for p in leaves)):
        raise ValueError("route levels are not separable")

    edges = [
        (root, children[0]), (root, children[1]),
        (children[0], leaves[0][:2]), (children[0], leaves[1][:2]),
        (children[1], leaves[2][:2]), (children[1], leaves[3][:2]),
    ]
    counts = [0] * len(edges)
    for bead in (c[:2] for c in beads):
        candidates = []
        for idx, (a, b) in enumerate(edges):
            dist, t = _distance_to_segment(bead, a, b)
            if 0.18 < t < 0.82:
                candidates.append((dist, idx))
        candidates.sort()
        if not candidates or candidates[0][0] > 16:
            raise ValueError("bead is not owned by a visible branch")
        if len(candidates) > 1 and candidates[1][0] - candidates[0][0] < 12:
            raise ValueError("ambiguous bead ownership")
        counts[candidates[0][1]] += 1

    if sorted(counts[0:2]) != [1, 3] or sorted(counts[2:4]) != [1, 3] or sorted(counts[4:6]) != [1, 3]:
        raise ValueError(f"invalid outgoing bead counts {counts}")
    first = 0 if counts[0] > counts[1] else 1
    base = 2 if first == 0 else 4
    second = 0 if counts[base] > counts[base + 1] else 1
    return str(leaves[first * 2 + second][2])
