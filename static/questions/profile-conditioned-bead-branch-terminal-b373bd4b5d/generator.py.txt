"""Deterministic renderer for a bead-guided route tree."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BG = (250, 249, 246)
DARK = (35, 42, 49)
GRAY = (126, 132, 139)
GREEN = (32, 157, 88)
CYAN = (0, 169, 206)
COLORS = {
    "orange": (235, 126, 36),
    "purple": (143, 82, 190),
    "blue": (48, 103, 210),
    "red": (210, 55, 63),
}


def _chosen_leaf(bead_sides: list[int]) -> int:
    first = int(bead_sides[0])
    second = int(bead_sides[1 + first])
    return first * 2 + second


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    bead_sides = [rng.randrange(2) for _ in range(3)]
    chosen = _chosen_leaf(bead_sides)
    wanted = list(COLORS)[int(seed) % 4]
    remaining = [c for c in COLORS if c != wanted]
    rng.shuffle(remaining)
    terminal_colors = [None] * 4
    terminal_colors[chosen] = wanted
    for i in range(4):
        if terminal_colors[i] is None:
            terminal_colors[i] = remaining.pop()

    base = [
        (66, 256), (164, 256), (286, 142), (286, 370),
        (444, 58), (444, 208), (444, 304), (444, 454),
    ]
    nodes = []
    for index, (x, y) in enumerate(base):
        # Keep levels unmistakable while varying geometry and all route lengths.
        jx = rng.randint(-5, 5) if index else rng.randint(-3, 3)
        jy = rng.randint(-6, 6)
        nodes.append([x + jx, y + jy])
    return {
        "terminal_colors": terminal_colors,
        "bead_sides": bead_sides,
        "nodes": nodes,
        "seed": int(seed),
    }


def analytic_gold(scene: dict) -> str:
    index = _chosen_leaf(scene["bead_sides"])
    return str(scene["terminal_colors"][index])


def margin(scene: dict) -> float:
    # At every choice, the winning branch has three beads and the runner-up one.
    return 2.0


def is_quarantined(scene: dict) -> bool:
    points = [tuple(map(float, p)) for p in scene["nodes"]]
    return any(math.dist(a, b) < 70 for i, a in enumerate(points) for b in points[i + 1 :])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _point(a: tuple[float, float], b: tuple[float, float], t: float) -> tuple[float, float]:
    return (a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1]))


def render(scene: dict) -> Image.Image:
    terminal_colors = list(scene["terminal_colors"])
    bead_sides = [int(v) for v in scene["bead_sides"]]
    nodes = [tuple(map(float, p)) for p in scene["nodes"]]
    _ = int(scene["seed"])  # The sampled seed is visibly instantiated through nodes.

    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BG)
    draw = ImageDraw.Draw(image)

    def xy(p):
        return (round(p[0] * scale), round(p[1] * scale))

    edges = [(0, 1), (1, 2), (1, 3), (2, 4), (2, 5), (3, 6), (3, 7)]
    for a, b in edges:
        draw.line([xy(nodes[a]), xy(nodes[b])], fill=DARK, width=8 * scale)

    branch_pairs = [((1, 2), (1, 3)), ((2, 4), (2, 5)), ((3, 6), (3, 7))]
    for junction, pair in enumerate(branch_pairs):
        winner = bead_sides[junction]
        for side, (a, b) in enumerate(pair):
            count = 3 if side == winner else 1
            ts = (0.38, 0.50, 0.62) if count == 3 else (0.50,)
            for t in ts:
                x, y = _point(nodes[a], nodes[b], t)
                r = 7 * scale
                cx, cy = round(x * scale), round(y * scale)
                draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=CYAN, outline=BG, width=2 * scale)

    # Root start: a green ring with a white center.
    sx, sy = xy(nodes[0])
    draw.ellipse((sx - 17 * scale, sy - 17 * scale, sx + 17 * scale, sy + 17 * scale), fill=GREEN)
    draw.ellipse((sx - 8 * scale, sy - 8 * scale, sx + 8 * scale, sy + 8 * scale), fill=BG)

    for idx in (1, 2, 3):
        x, y = xy(nodes[idx])
        r = 10 * scale
        draw.ellipse((x - r, y - r, x + r, y + r), fill=GRAY, outline=DARK, width=2 * scale)

    for leaf, node_index in enumerate((4, 5, 6, 7)):
        x, y = xy(nodes[node_index])
        r = 16 * scale
        draw.rectangle((x - r, y - r, x + r, y + r), fill=COLORS[terminal_colors[leaf]], outline=DARK, width=3 * scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
