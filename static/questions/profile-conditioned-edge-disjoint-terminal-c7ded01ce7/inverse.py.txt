"""Independent image-only inverse for edge-disjoint-route images."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

COLOR_REFERENCES = {
    "green": ((41, 164, 98), (30, 150, 105)),
    "orange": ((239, 143, 45), (225, 132, 38)),
    "purple": ((139, 92, 246), (125, 83, 225)),
    "blue": ((47, 127, 230), (38, 116, 210)),
    "gray": ((205, 211, 220), (214, 218, 226)),
    "ink": ((37, 42, 52), (28, 35, 46)),
}


def _mask_for(array: np.ndarray, name: str, tolerance: float = 12.0) -> np.ndarray:
    pixels = array.astype(np.int32)
    masks = []
    for reference in COLOR_REFERENCES[name]:
        ref = np.asarray(reference, dtype=np.int32)
        masks.append(np.sqrt(np.sum((pixels - ref) ** 2, axis=2)) <= tolerance)
    return np.logical_or.reduce(masks)


def _components(mask: np.ndarray, minimum: int = 40) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        coords = []
        while queue:
            cy, cx = queue.pop()
            coords.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(coords) >= minimum:
            ys = np.fromiter((p[0] for p in coords), dtype=float)
            xs = np.fromiter((p[1] for p in coords), dtype=float)
            result.append((float(xs.mean()), float(ys.mean()), len(coords)))
    return result


def _line_occupancy(ink: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> float:
    samples = 120
    hits = 0
    used = 0
    for index in range(10, samples - 9):
        t = index / (samples - 1)
        x = int(round(a[0] + t * (b[0] - a[0])))
        y = int(round(a[1] + t * (b[1] - a[1])))
        y0, y1 = max(0, y - 3), min(ink.shape[0], y + 4)
        x0, x1 = max(0, x - 3), min(ink.shape[1], x + 4)
        hits += bool(ink[y0:y1, x0:x1].any())
        used += 1
    return hits / used


def _reachable(adjacency: list[set[int]], start: int, goal: int, removed: tuple[int, int] | None = None) -> bool:
    queue = deque([start])
    seen = {start}
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for neighbor in adjacency[node]:
            if removed is not None and {node, neighbor} == set(removed):
                continue
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def _has_two_edge_disjoint(adjacency: list[set[int]], start: int, goal: int) -> bool:
    edges = [(a, b) for a, neighbors in enumerate(adjacency) for b in neighbors if a < b]
    if not _reachable(adjacency, start, goal):
        return False
    return all(_reachable(adjacency, start, goal, edge) for edge in edges)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    green = _components(_mask_for(array, "green"), minimum=100)
    gray = _components(_mask_for(array, "gray"), minimum=80)
    terminals = []
    for name in ("orange", "purple", "blue"):
        parts = _components(_mask_for(array, name), minimum=100)
        if len(parts) != 1:
            raise ValueError(f"expected one {name} terminal")
        terminals.append((name, parts[0]))
    if len(green) != 3 or len(gray) != 12:
        raise ValueError("marker segmentation failed")

    ink = _mask_for(array, "ink", tolerance=16.0)
    winners = []
    for start_component in green:
        start = (start_component[0], start_component[1])
        name, terminal_component = min(terminals, key=lambda item: abs(item[1][1] - start[1]))
        terminal = (terminal_component[0], terminal_component[1])
        row_gray = sorted(gray, key=lambda item: abs(item[1] - start[1]))[:4]
        if max(abs(item[1] - start[1]) for item in row_gray) > 140:
            raise ValueError("row grouping failed")
        nodes = [start, terminal] + [(item[0], item[1]) for item in row_gray]
        adjacency = [set() for _ in nodes]
        for a in range(len(nodes)):
            for b in range(a + 1, len(nodes)):
                occupancy = _line_occupancy(ink, nodes[a], nodes[b])
                if occupancy > 0.78:
                    adjacency[a].add(b)
                    adjacency[b].add(a)
        edge_count = sum(len(neighbors) for neighbors in adjacency) // 2
        degrees = sorted(len(neighbors) for neighbors in adjacency)
        if edge_count != 7 or degrees != [2, 2, 2, 2, 3, 3]:
            raise ValueError(f"invalid reconstructed graph: edges={edge_count}, degrees={degrees}")
        if _has_two_edge_disjoint(adjacency, 0, 1):
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"expected one edge-redundant row, found {winners}")
    return winners[0]
