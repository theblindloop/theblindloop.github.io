"""Scene sampler and raster renderer for dial_zero_offset.

Four round dials stand in a row. Every dial carries a red zero mark on its rim
and a black needle from its hub. A dial's READING is the clockwise sweep, in
degrees, from that dial's own red zero mark round to its own black needle.
The dials do not share a zero mark, so the absolute direction a needle points
in carries no answer information: the reading has to be measured against the
per-dial origin. The decision names the ring colour of the dial with the
largest reading.

Every latent field is drawn. The needle tip and zero-mark tip are stored as
integer pixel offsets from the dial centre, so the analytic gold is a function
of geometry that is literally on the canvas: there is no sub-pixel latent that
the raster fails to expose.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "dial_zero_offset-renderer-1.0.0"

WIDTH = 760
HEIGHT = 220
RADIUS = 62
NEEDLE_LEN = 50          # needle runs hub -> 0.81 R
ZERO_INNER = 0.60        # red zero mark spans 0.60 R -> 1.00 R
CENTERS = [(100 + 180 * i, 110) for i in range(4)]

BACKGROUND = (255, 255, 255)
NEEDLE_RGB = (0, 0, 0)
ZERO_RGB = (220, 0, 0)
DISTRACTOR_RGB = (150, 150, 150)

RING_COLORS: dict[str, tuple[int, int, int]] = {
    "blue": (0, 90, 200),
    "green": (0, 140, 60),
    "purple": (140, 60, 180),
    "orange": (230, 130, 0),
}
COLOR_NAMES = ("blue", "green", "purple", "orange")

QUARANTINE_DEGREES = 12.0
MIN_READING = 22.0
MAX_READING = 338.0


# --- geometry helpers ---------------------------------------------------------


def _bearing(dx: float, dy: float) -> float:
    """Screen-space angle in degrees; increasing it sweeps clockwise on screen."""
    return math.degrees(math.atan2(dy, dx)) % 360.0


def _tip(angle_deg: float, length: float) -> list[int]:
    return [
        int(round(length * math.cos(math.radians(angle_deg)))),
        int(round(length * math.sin(math.radians(angle_deg)))),
    ]


def readings(scene: dict) -> list[float]:
    """Clockwise sweep from each dial's own zero mark to its own needle."""
    out = []
    for needle, zero in zip(scene["needle_tips"], scene["zero_tips"]):
        out.append(
            (_bearing(needle[0], needle[1]) - _bearing(zero[0], zero[1])) % 360.0
        )
    return out


def analytic_gold(scene: dict) -> str:
    values = readings(scene)
    best = max(range(4), key=lambda i: values[i])
    return scene["ring_names"][best]


def margin(scene: dict) -> float:
    ordered = sorted(readings(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_DEGREES)


def naive_bearing_choice(scene: dict) -> str:
    """The shortcut answer: ignore the zero marks, take the needle that points
    furthest clockwise from straight up."""
    scores = [
        (_bearing(tip[0], tip[1]) + 90.0) % 360.0 for tip in scene["needle_tips"]
    ]
    return scene["ring_names"][max(range(4), key=lambda i: scores[i])]


def congruent(scene: dict) -> bool:
    """True when the raw-bearing shortcut happens to agree with the answer."""
    return naive_bearing_choice(scene) == analytic_gold(scene)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every field is drawn and gold reads integer pixel
    offsets that the raster exposes directly."""
    return []


# --- sampling -----------------------------------------------------------------


def _draft(rng: random.Random, forced_ring: str) -> dict | None:
    answer_slot = rng.randrange(4)
    want_congruent = rng.random() < 0.30
    want_quarantine = rng.random() < 0.18

    top = rng.uniform(170.0, 330.0)
    gap = rng.uniform(5.5, 10.5) if want_quarantine else rng.uniform(18.0, 70.0)
    second = top - gap
    if second < 70.0:
        return None
    rest_hi = second - 16.0
    if rest_hi < MIN_READING + 10.0:
        return None
    others = [rng.uniform(MIN_READING + 4.0, rest_hi) for _ in range(2)]
    if abs(others[0] - others[1]) < 8.0:
        return None

    slots = [i for i in range(4) if i != answer_slot]
    rng.shuffle(slots)
    values = [0.0] * 4
    values[answer_slot] = top
    values[slots[0]] = second
    values[slots[1]] = others[0]
    values[slots[2]] = others[1]

    zeros = None
    for _ in range(80):
        trial = [rng.uniform(0.0, 360.0) for _ in range(4)]
        needles = [(trial[i] + values[i]) % 360.0 for i in range(4)]
        scores = [(a + 90.0) % 360.0 for a in needles]
        naive = max(range(4), key=lambda i: scores[i])
        if (naive == answer_slot) == want_congruent:
            zeros = trial
            break
    if zeros is None:
        return None

    # The answering ring colour is fixed by the seed, so the four colours are
    # exactly balanced across a generated dataset and colour identity cannot be
    # read off any other cue.
    spare = [name for name in COLOR_NAMES if name != forced_ring]
    rng.shuffle(spare)
    ring_names = [None] * 4
    ring_names[answer_slot] = forced_ring
    for slot, name in zip([i for i in range(4) if i != answer_slot], spare):
        ring_names[slot] = name
    scene = {
        "centers": [list(point) for point in CENTERS],
        "radius": RADIUS,
        "needle_len": NEEDLE_LEN,
        "ring_names": ring_names,
        "needle_tips": [
            _tip((zeros[i] + values[i]) % 360.0, NEEDLE_LEN) for i in range(4)
        ],
        "zero_tips": [_tip(zeros[i], RADIUS) for i in range(4)],
        "distractor_tips": [
            _tip(rng.uniform(0.0, 360.0), 0.34 * RADIUS) for _ in range(4)
        ],
        "offset": [rng.randint(-8, 8), rng.randint(-6, 6)],
    }

    # Re-read gold off the quantized integer tips, then police the sample.
    quantized = readings(scene)
    if min(quantized) < MIN_READING or max(quantized) > MAX_READING:
        return None
    ordered = sorted(quantized, reverse=True)
    if ordered[0] - ordered[1] < 5.0:          # unresolvable at raster scale: reject
        return None
    if quantized.index(ordered[0]) != answer_slot:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    forced_ring = COLOR_NAMES[seed % len(COLOR_NAMES)]
    for _ in range(600):
        scene = _draft(rng, forced_ring)
        if scene is not None:
            return scene
    raise RuntimeError(f"sampler failed to find a clean scene for seed {seed}")


# --- rendering ----------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    off_x, off_y = scene["offset"]
    radius = scene["radius"]

    for index in range(4):
        cx = scene["centers"][index][0] + off_x
        cy = scene["centers"][index][1] + off_y
        ring = RING_COLORS[scene["ring_names"][index]]
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            outline=ring,
            width=4,
        )

        dtip = scene["distractor_tips"][index]
        draw.rectangle(
            [cx + dtip[0] - 3, cy + dtip[1] - 3, cx + dtip[0] + 3, cy + dtip[1] + 3],
            fill=DISTRACTOR_RGB,
        )

        ntip = scene["needle_tips"][index]
        draw.line([cx, cy, cx + ntip[0], cy + ntip[1]], fill=NEEDLE_RGB, width=5)
        draw.ellipse([cx - 4, cy - 4, cx + 4, cy + 4], fill=NEEDLE_RGB)

        ztip = scene["zero_tips"][index]
        draw.line(
            [
                cx + int(round(ZERO_INNER * ztip[0])),
                cy + int(round(ZERO_INNER * ztip[1])),
                cx + ztip[0],
                cy + ztip[1],
            ],
            fill=ZERO_RGB,
            width=5,
        )
    return image
