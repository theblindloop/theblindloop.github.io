"""Pixel-only inverse arm for dial_zero_offset.

Receives one rendered PNG and nothing else. It rediscovers the dials from ink:
each coloured ring gives a centre and a radius, the black radial stroke inside
that ring gives the needle direction, and the red rim stroke gives that dial's
zero direction. The reading is the clockwise sweep from zero to needle, and the
answer is the ring colour with the largest reading. Abstains whenever the
evidence is incomplete or the top two readings are too close to separate.

Imports no renderer, prompt, generator, verifier or gold code and never sees a
latent scene.
"""

from __future__ import annotations

import math

from PIL import Image

ORACLE_VERSION = "dial_zero_offset-oracle-1.0.0"

_RINGS: dict[str, tuple[int, int, int]] = {
    "blue": (0, 90, 200),
    "green": (0, 140, 60),
    "purple": (140, 60, 180),
    "orange": (230, 130, 0),
}
_ZERO = (220, 0, 0)
_NEEDLE = (0, 0, 0)
_TOLERANCE = 30
_DECISION_MARGIN = 3.0
ABSTAIN = "abstain"


def _mask(pixels, width: int, height: int, target) -> list[tuple[int, int]]:
    tr, tg, tb = target
    found = []
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y]
            if abs(r - tr) <= _TOLERANCE and abs(g - tg) <= _TOLERANCE and abs(b - tb) <= _TOLERANCE:
                found.append((x, y))
    return found


def _circular_mean(points, cx: float, cy: float) -> float:
    sx = sy = 0.0
    for x, y in points:
        angle = math.atan2(y - cy, x - cx)
        sx += math.cos(angle)
        sy += math.sin(angle)
    return math.degrees(math.atan2(sy, sx)) % 360.0


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    dials = []
    for name in _RINGS:
        ring = _mask(pixels, width, height, _RINGS[name])
        if len(ring) < 200:
            return ABSTAIN
        cx = sum(x for x, _ in ring) / len(ring)
        cy = sum(y for _, y in ring) / len(ring)
        radius = sum(math.hypot(x - cx, y - cy) for x, y in ring) / len(ring)
        if radius < 20.0:
            return ABSTAIN
        dials.append((name, cx, cy, radius))

    black = _mask(pixels, width, height, _NEEDLE)
    red = _mask(pixels, width, height, _ZERO)

    values = {}
    for name, cx, cy, radius in dials:
        needle = [
            point
            for point in black
            if 0.34 * radius <= math.hypot(point[0] - cx, point[1] - cy) <= 0.95 * radius
        ]
        zero = [
            point
            for point in red
            if 0.50 * radius <= math.hypot(point[0] - cx, point[1] - cy) <= 1.18 * radius
        ]
        if len(needle) < 40 or len(zero) < 40:
            return ABSTAIN
        # A stray stroke from a neighbouring dial would smear the circular mean;
        # require both strokes to be tight, radial and single-armed.
        needle_angle = _circular_mean(needle, cx, cy)
        zero_angle = _circular_mean(zero, cx, cy)
        for points, mean in ((needle, needle_angle), (zero, zero_angle)):
            for x, y in points:
                delta = abs(((math.degrees(math.atan2(y - cy, x - cx)) - mean + 180.0) % 360.0) - 180.0)
                if delta > 25.0:
                    return ABSTAIN
        values[name] = (needle_angle - zero_angle) % 360.0

    ordered = sorted(values.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < _DECISION_MARGIN:
        return ABSTAIN
    return ordered[0][0]
