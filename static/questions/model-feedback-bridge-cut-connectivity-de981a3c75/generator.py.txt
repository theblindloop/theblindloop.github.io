"""Deterministic removable-bridge connectivity world."""

from __future__ import annotations

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 1080, 760, 2
BACKGROUND = (244, 243, 238)
PAPER = (255, 255, 255)
INK = (35, 42, 50)
MUTED = (184, 190, 195)
START = (42, 151, 92)
TERMINAL = (221, 154, 36)
PLUG_COLORS = {
    "red": (211, 61, 69),
    "blue": (47, 105, 202),
    "orange": (230, 126, 35),
    "violet": (135, 78, 186),
    "teal": (34, 151, 151),
    "pink": (211, 79, 142),
}
TERMINALS = (("T0", "A", 215), ("T1", "B", 335), ("T2", "C", 515), ("T3", "D", 635))


def _edge(a: str, b: str, points: list[tuple[int, int]], plug: str | None = None,
          plug_leg: int | None = None) -> dict:
    return {"a": a, "b": b, "points": [list(p) for p in points],
            "plug": plug, "plug_leg": plug_leg}


def _base_edges() -> list[dict]:
    # A hierarchical tree is deliberately routed as a winding loom.  The extra
    # branches are genuine dead ends, not decoration or hidden metadata.
    return [
        _edge("S", "R", [(105, 435), (230, 435)]),
        _edge("R", "U", [(230, 435), (282, 435), (282, 285), (390, 285)], "group_u", 2),
        _edge("R", "L", [(230, 435), (332, 435), (332, 575), (390, 575)], "group_l", 2),
        _edge("U", "T0", [(390, 285), (470, 285), (470, 195), (720, 195),
                            (720, 215), (960, 215)], "leaf_0", 5),
        _edge("U", "T1", [(390, 285), (520, 285), (520, 375), (695, 375),

        _edge("L", "T2", [(390, 575), (490, 575), (490, 485), (710, 485),
                            (710, 515), (960, 515)], "leaf_2", 5),
        _edge("L", "T3", [(390, 575), (545, 575), (545, 665), (690, 665),
                            (690, 635), (960, 635)], "leaf_3", 5),
        _edge("R", "X0", [(230, 435), (185, 435), (185, 330), (125, 330)]),

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    roles = ("group_u", "group_l", "leaf_0", "leaf_1", "leaf_2", "leaf_3")
    names = list(PLUG_COLORS)
    rng.shuffle(names)
    plug_symbols = {role: name for role, name in zip(roles, names, strict=True)}
    survivor = int(seed) % 4
    cut_roles = {
        0: ("group_l", "leaf_1"),
        1: ("group_l", "leaf_0"),
        2: ("group_u", "leaf_3"),
        3: ("group_u", "leaf_2"),
    }[survivor]
    cut_symbols = [plug_symbols[role] for role in cut_roles]
    rng.shuffle(cut_symbols)
    edges = _base_edges()
    rng.shuffle(edges)
    return {"edges": edges, "plug_symbols": plug_symbols, "cut_symbols": cut_symbols}


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    symbols = scene["plug_symbols"]
    cuts = set(scene["cut_symbols"])
    graph: dict[str, list[str]] = {}
    for edge in edges:
        role = edge["plug"]
        if role is not None and symbols[role] in cuts:
            continue
        a, b = str(edge["a"]), str(edge["b"])
        graph.setdefault(a, []).append(b)
        graph.setdefault(b, []).append(a)
    reached, queue = {"S"}, deque(["S"])
    while queue:
        here = queue.popleft()
        for nxt in graph.get(here, []):
            if nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    surviving = [label for node, label, _ in TERMINALS if node in reached]
    if len(surviving) != 1:
        raise ValueError(f"expected one reachable terminal, got {surviving}")
    return surviving[0]


def margin(scene: dict) -> float:
    _ = analytic_gold(scene)
    return 34.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["cut_symbols"]) != 2 or len(set(scene["plug_symbols"].values())) != 6:
            return True
        analytic_gold(scene)
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_same_ink_edge_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _p(point: tuple[int, int] | list[int]) -> tuple[int, int]:
    return int(point[0]) * SCALE, int(point[1]) * SCALE


def _b(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(v * SCALE for v in box)  # type: ignore[return-value]


def _plug_segment(points: list[list[int]], leg: int) -> tuple[tuple[int, int], tuple[int, int]]:
    p, q = points[leg - 1], points[leg]
    cx, cy = (p[0] + q[0]) / 2, (p[1] + q[1]) / 2
    half = 24
    if p[1] == q[1]:
        return (int(cx - half), int(cy)), (int(cx + half), int(cy))
    return (int(cx), int(cy - half)), (int(cx), int(cy + half))


def render(scene: dict) -> Image.Image:
    edges = scene["edges"]
    symbols = scene["plug_symbols"]
    cuts = scene["cut_symbols"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(_b((18, 14, 1062, 126)), radius=16 * SCALE, fill=PAPER,
                           outline=MUTED, width=2 * SCALE)
    draw.text(_p((38, 31)), "CUT", fill=INK, font=_font(25, True))
    draw.text(_p((38, 72)), "remove both matching bridge-plugs", fill=INK, font=_font(16))
    for index, name in enumerate(cuts):
        x = 420 + index * 205
        draw.line([_p((x, 70)), _p((x + 116, 70))], fill=INK, width=8 * SCALE)
        draw.line([_p((x + 34, 70)), _p((x + 82, 70))], fill=PLUG_COLORS[name],
                  width=22 * SCALE)
        draw.text(_p((x + 132, 56)), name.upper(), fill=INK, font=_font(17, True))
    draw.text(_p((816, 40)), "black dots = junctions", fill=INK, font=_font(15))

    draw.rounded_rectangle(_b((24, 142, 1056, 728)), radius=18 * SCALE, fill=PAPER,
                           outline=MUTED, width=2 * SCALE)

    # All ordinary edges first, then opaque plugs replace short edge intervals.
    for edge in edges:
        draw.line([_p(p) for p in edge["points"]], fill=INK, width=8 * SCALE, joint="curve")
    for edge in edges:
        role = edge["plug"]
        if role is None:
            continue
        p, q = _plug_segment(edge["points"], int(edge["plug_leg"]))
        draw.line([_p(p), _p(q)], fill=PLUG_COLORS[symbols[role]], width=22 * SCALE)
        for cap in (p, q):
            draw.ellipse(_b((cap[0] - 5, cap[1] - 5, cap[0] + 5, cap[1] + 5)), fill=INK)

    for x, y in ((230, 435), (390, 285), (390, 575)):
        draw.ellipse(_b((x - 10, y - 10, x + 10, y + 10)), fill=INK)

    draw.ellipse(_b((68, 417, 104, 453)), fill=START, outline=INK, width=3 * SCALE)
    draw.text(_p((34, 378)), "START", fill=START, font=_font(19, True))
    for _, label, y in TERMINALS:
        draw.ellipse(_b((960, y - 14, 988, y + 14)), fill=PAPER, outline=TERMINAL,
                     width=6 * SCALE)
        draw.text(_p((1002, y - 19)), label, fill=INK, font=_font(27, True))

    # Visible dead-end caps make endpoint status explicit.
    for x, y in ((125, 330), (105, 585), (590, 160), (305, 225),
                 (600, 680), (300, 635), (650, 245), (660, 545)):
        draw.ellipse(_b((x - 7, y - 7, x + 7, y + 7)), fill=PAPER, outline=INK, width=3 * SCALE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
