"""Independent PNG-only inverse arm using keyed color deletion and components."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1080, 760
INK = np.array((35, 42, 50), dtype=np.int16)
COLORS = {
    "red": np.array((211, 61, 69), dtype=np.int16),
    "blue": np.array((47, 105, 202), dtype=np.int16),
    "orange": np.array((230, 126, 35), dtype=np.int16),
    "violet": np.array((135, 78, 186), dtype=np.int16),
    "teal": np.array((34, 151, 151), dtype=np.int16),
    "pink": np.array((211, 79, 142), dtype=np.int16),
}
ANCHORS = (("A", 215), ("B", 335), ("C", 515), ("D", 635))


class OracleAbstention(ValueError):
    pass


def _near(arr: np.ndarray, color: np.ndarray, threshold: float = 58.0) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2, dtype=np.int64)) < threshold


def _component(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    sx, sy = start
    if not mask[sy, sx]:
        raise OracleAbstention("START route is missing")
    seen = np.zeros(mask.shape, dtype=bool)
    seen[sy, sx] = True
    queue: deque[tuple[int, int]] = deque([(sx, sy)])
    height, width = mask.shape
    while queue:
        x, y = queue.popleft()
        for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((nx, ny))
    return seen


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)
    cut: list[str] = []
    for name, color in COLORS.items():
        # Only the CUT card is inspected here; every network plug lies below it.
        if int(_near(arr[20:118], color).sum()) > 450:
            cut.append(name)
    if len(cut) != 2:
        raise OracleAbstention(f"expected two CUT swatches, recovered {cut}")

    crop = arr[145:710, 96:970]
    walkable = _near(crop, INK, 68.0)
    for name, color in COLORS.items():
        network_pixels = _near(crop, color)
        if int(network_pixels.sum()) < 220:
            raise OracleAbstention(f"network plug {name} is missing")
        if name not in cut:
            walkable |= network_pixels

    # Coordinates are translated into the network crop.  They are sampled on
    # the dark route, before the green/amber endpoint markers.
    reached = _component(walkable, (112 - 96, 435 - 145))
    answers: list[str] = []
    for label, y in ANCHORS:
        patch = reached[y - 145 - 5:y - 145 + 6, 945 - 96:960 - 96]
        if int(patch.sum()) > 8:
            answers.append(label)
    if len(answers) != 1:
        raise OracleAbstention(f"expected one reachable terminal, recovered {answers}")
    return answers[0]
