"""Declared in-place derotation reassembly reveal world.

A single solid ``A`` (apex-up) or ``B`` (apex-down) triangle is drawn across a
3x3 grid. Each cell is then rotated about its own center by a random multiple
of 90 degrees, scrambling the glyph into nine rotated fragments. A small
distinct-colored key dot, drawn at a cell's bottom-left corner before rotation,
marks that cell's upright orientation (it lands in the corner where the cell's
bottom-left edge currently sits). The declared transform is to rotate every cell
so its key dot returns to bottom-left, reassembling the upright glyph; the
answer is whether the reassembled triangle points up (A) or down (B).

The transform is an in-place quarter-turn derotation (a discrete 2D reassembly
by per-tile orientation), distinct from the rectilinear rank-collation of the
panel worlds: no tile changes grid position, only its in-place angular phase.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N = 3                    # grid is N x N cells
CELL = 90                # side of one cell in pixels
G = N * CELL             # side of the whole grid
X0 = 60
Y0 = 60
BORDER = 3               # frame outline width
RENDERER_VERSION = "quad_derotate_reveal-0.1.0"

FRAME = "rgb(0,70,140)"
DOT = "rgb(230,0,220)"
INK = "black"
BG = "white"


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    direction = rng.choice(["A", "B"])
    rotations = [[rng.randrange(4) for _ in range(N)] for _ in range(N)]
    return {
        "direction": direction,
        "cell": CELL,
        "n": N,
        "x0": X0,
        "y0": Y0,
        "rotations": rotations,
        "seed": seed,
    }


def _tri_points(scene: dict) -> list[tuple[float, float]]:
    g = G
    x0, y0 = X0, Y0
    if scene["direction"] == "A":
        # apex at top center, base along the bottom edge
        return [(x0 + 0.5 * g, y0), (x0 + 0.05 * g, y0 + g), (x0 + 0.95 * g, y0 + g)]
    # apex at bottom center, base along the top edge
    return [(x0 + 0.5 * g, y0 + g), (x0 + 0.05 * g, y0), (x0 + 0.95 * g, y0)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (512, 512), BG)
    d = ImageDraw.Draw(img)
    d.polygon(_tri_points(scene), fill=INK)

    # key dot at the bottom-left corner of each cell, drawn before rotation
    for r in range(N):
        for c in range(N):
            cx = X0 + c * CELL
            cy = Y0 + r * CELL
            d.rectangle([cx + 8, cy + CELL - 18, cx + 18, cy + CELL - 8], fill=DOT)

    # rotate each cell in place by a random multiple of 90 degrees (lossless)
    for r in range(N):
        for c in range(N):
            k = scene["rotations"][r][c]
            if k == 0:
                continue
            box = (X0 + c * CELL, Y0 + r * CELL, X0 + (c + 1) * CELL, Y0 + (r + 1) * CELL)
            crop = img.crop(box).rotate(k * 90, expand=False)
            img.paste(crop, box)

    d = ImageDraw.Draw(img)
    d.rectangle(
        [X0 - BORDER, Y0 - BORDER, X0 + G + BORDER, Y0 + G + BORDER],
        outline=FRAME,
        width=BORDER,
    )
    return img


def analytic_gold(scene: dict) -> str:
    return scene["direction"]


def margin(scene: dict) -> float:
    # analytic verifiability margin: the triangle centroid sits G/6 pixels off
    # the grid centre in the vertical direction.
    return float(N * CELL) / 6.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list:
    # no pixel-identical label-preserving transform for not_applicable mode
    return []
