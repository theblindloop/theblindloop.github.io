"""Pixel-only inverse arm for the declared derotation reveal.

Receives only the rendered PNG. It finds the framed 3x3 grid, detects each
cell's key-dot corner to recover that cell's in-place quarter-turn, rotates the
cell back to upright (reassembling the glyph), and then reads whether the
reassembled solid triangle points up (A) or down (B) from its ink centroid.
No scene, latent, renderer, or gold data is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 3
CELL = 90
G = N * CELL


def _mask(arr, color, tol):
    return np.abs(arr - np.array(color, dtype=np.float32)).sum(axis=2) < tol


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    arr = np.asarray(image, dtype=np.float32)

    # locokate the grid by its blue frame, then tight inside the border line
    fm = _mask(arr, (0, 70, 140), 70)
    ys, xs = np.nonzero(fm)
    if len(ys) == 0:
        raise ValueError("no grid frame found in image")
    ox = int(xs.min()) + 3
    oy = int(ys.min()) + 3

    ink = _mask(arr, (0, 0, 0), 80)

    # reassemble the upright glyph cell by cell
    out = np.zeros((G, G), dtype=bool)
    for r in range(N):
        for c in range(N):
            row0 = oy + r * CELL
            row1 = row0 + CELL
            col0 = ox + c * CELL
            col1 = col0 + CELL
            sub = ink[row0:row1, col0:col1]

            dm = _mask(arr[row0:row1, col0:col1, :], (230, 0, 220), 80)
            dys, dxs = np.nonzero(dm)
            if len(dys) == 0:
                k = 0
            else:
                cy = dys.mean()
                cx = dxs.mean()
                top = cy < CELL / 2.0
                left = cx < CELL / 2.0
                if top and left:
                    corner = 0  # top-left  -> was 180deg from bottom-left
                elif top and not left:
                    corner = 1  # top-right -> was 90deg
                elif not top and not left:
                    corner = 2  # bottom-right -> was 0deg from a 90 turn
                else:
                    corner = 3  # bottom-left -> upright
                k = {0: 2, 1: 1, 2: 0, 3: 0}[corner]
            out[r * CELL:(r + 1) * CELL, c * CELL:(c + 1) * CELL] = np.rot90(sub, k=k)

    ys, xs = np.nonzero(out)
    cy = ys.mean() / G
    # apex-up (A) centroid is in the lower half; apex-down (B) in the upper half
    return "B" if cy < 0.5 else "A"
