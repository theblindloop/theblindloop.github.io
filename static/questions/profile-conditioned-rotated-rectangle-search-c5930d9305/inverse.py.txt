"""Independent image-only inverse for rotated rectangle search.

This module knows only the public ink color and question contract. It discovers
dot components, groups them into panels by visible horizontal separation, and
tests the rectangle diagonal theorem. It imports no world-side module.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

TEAL_RGB = np.array((8, 127, 140), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        sx = sy = count = 0
        while queue:
            yy, xx = queue.popleft()
            sx += xx
            sy += yy
            count += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if count >= 60:
            found.append((sx / count, sy / count, count))
    return found


def _residual(points: list[tuple[float, float]]) -> float:
    pairings = (((0, 1), (2, 3)), ((0, 2), (1, 3)), ((0, 3), (1, 2)))
    best = float("inf")
    for (a, b), (c, d) in pairings:
        p, q, r, s = points[a], points[b], points[c], points[d]
        midpoint_gap = math.hypot((p[0] + q[0]) - (r[0] + s[0]), (p[1] + q[1]) - (r[1] + s[1]))
        length_gap = abs(math.hypot(p[0] - q[0], p[1] - q[1]) - math.hypot(r[0] - s[0], r[1] - s[1]))
        best = min(best, math.hypot(midpoint_gap, length_gap))
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = np.linalg.norm(arr - TEAL_RGB, axis=2) < 45.0
    centers = sorted(_components(mask), key=lambda item: item[0])
    if len(centers) != 16:
        raise ValueError(f"expected 16 teal dot components, found {len(centers)}")
    panels = [[(x, y) for x, y, _ in centers[i : i + 4]] for i in range(0, 16, 4)]
    if any(len(panel) != 4 for panel in panels):
        raise ValueError("could not enumerate four four-dot panels")
    scores = [_residual(panel) for panel in panels]
    order = sorted(range(4), key=lambda i: scores[i])
    if scores[order[1]] - scores[order[0]] < 3.0:
        raise ValueError("ambiguous rectangle residual")
    return "ABCD"[order[0]]
