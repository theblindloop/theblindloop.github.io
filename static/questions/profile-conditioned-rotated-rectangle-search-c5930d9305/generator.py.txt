"""Deterministic latent sampler, analytic decision, and Pillow renderer."""

from __future__ import annotations

import math
import random
from itertools import combinations

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 260
PANEL_CENTERS = (105, 285, 465, 645)
PANEL_TOP, PANEL_BOTTOM = 55, 220
DOT_RADIUS = 8
TEAL = "#087f8c"
PANEL_INK = "#5b6470"
BACKGROUND = "#f4f1e8"
QUESTION = (
    "Which labeled panel has four teal dots at the corners of a rectangle? "
    "The rectangle may be rotated. Panels A-D run left to right."
)


def _rectangle_points(cx: int, cy: int, rng: random.Random) -> list[list[int]]:
    # Integer Pythagorean directions make the latent target an exact rotated rectangle.
    ux, uy, den = rng.choice(((4, 3, 5), (3, 4, 5), (12, 5, 13), (5, 12, 13)))
    if den == 5:
        a = 5 * rng.randint(8, 10)
        b = 5 * rng.randint(5, 7)
    else:
        a = 13 * rng.randint(3, 4)
        b = 26
    vx, vy = -uy, ux
    points = []
    for su, sv in ((-1, -1), (-1, 1), (1, 1), (1, -1)):
        x = cx + (su * a * ux + sv * b * vx) // den
        y = cy + (su * a * uy + sv * b * vy) // den
        points.append([x, y])
    return points


def _pairings() -> tuple[tuple[tuple[int, int], tuple[int, int]], ...]:
    return (((0, 1), (2, 3)), ((0, 2), (1, 3)), ((0, 3), (1, 2)))


def _rectangle_residual(points: list[list[float]]) -> float:
    """Zero iff some pairing is equal, mutually bisecting diagonals."""
    best = float("inf")
    for (a, b), (c, d) in _pairings():
        p, q, r, s = points[a], points[b], points[c], points[d]
        midpoint_gap = math.hypot((p[0] + q[0]) - (r[0] + s[0]), (p[1] + q[1]) - (r[1] + s[1]))
        length_gap = abs(math.hypot(p[0] - q[0], p[1] - q[1]) - math.hypot(r[0] - s[0], r[1] - s[1]))
        best = min(best, math.hypot(midpoint_gap, length_gap))
    return best


def _scores(scene: dict) -> list[float]:
    return [_rectangle_residual(panel) for panel in scene["panels"]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % 4
    panels: list[list[list[int]]] = []
    for index, cx in enumerate(PANEL_CENTERS):
        cy = rng.randint(132, 143)
        points = _rectangle_points(cx + rng.randint(-4, 4), cy, rng)
        if index != target:
            # Displace one corner. Direction and magnitude vary independently of
            # answer position; every panel retains four identical marks.
            vertex = rng.randrange(4)
            angle = rng.random() * 2.0 * math.pi
            # A deterministic minority of scenes sits just above quarantine so
            # the gallery and stress sets exercise the observable boundary.
            displacement = rng.randint(8, 11) if seed % 6 == 0 else rng.randint(14, 27)
            points[vertex][0] += int(round(displacement * math.cos(angle)))
            points[vertex][1] += int(round(displacement * math.sin(angle)))
        rng.shuffle(points)
        panels.append(points)

    # Extremely rare perturbations can approximate another rectangle. Increase
    # their displaced corner deterministically until the visible margin is safe.
    scene = {"panels": panels}
    for _ in range(8):
        scores = sorted(_scores(scene))
        if scores[1] - scores[0] >= 7.0:
            break
        loser = min((i for i in range(4) if i != target), key=lambda i: _scores(scene)[i])
        scene["panels"][loser][0][0] += 4
    return scene


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=18)
    title_font = ImageFont.load_default(size=17)
    draw.text((WIDTH // 2, 24), "FIND THE RECTANGLE", fill="#263238", anchor="mm", font=title_font)
    for index, points in enumerate(panels):
        cx = PANEL_CENTERS[index]
        draw.rounded_rectangle((cx - 82, PANEL_TOP, cx + 82, PANEL_BOTTOM), radius=9, fill="#ffffff", outline=PANEL_INK, width=2)
        draw.text((cx, 238), "ABCD"[index], fill="#263238", anchor="mm", font=font)
        for x, y in points:
            draw.ellipse((x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS), fill=TEAL)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return "ABCD"[min(range(4), key=lambda i: scores[i])]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return scores[1] - scores[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    panels = scene["panels"]
    twin = {"panels": [list(reversed(panel)) for panel in panels]}
    return [("reverse_dot_storage_order", twin)]
