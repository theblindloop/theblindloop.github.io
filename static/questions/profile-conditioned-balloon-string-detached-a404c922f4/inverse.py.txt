"""Pixel-only inverse arm for balloon_string_detached.

Receives the final PNG and nothing else: no scene, no renderer, no gold. It
segments the chromatic balloon ink by column runs, finds each balloon's lowest
painted row, walks down the column under it to the first dark achromatic string
pixel, and calls the string loose when the white band between them exceeds the
declared width. Abstains when the ink is missing or the reading is impossible.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "balloon_string_detached-oracle-0.1.0"
ABSTAIN = "abstain"

CHROMA_MIN = 55  # balloon ink is chromatic; frame, ticks and strings are grey
STRING_MAX_V = 130  # string ink is dark; the page frame is light grey
BAND = 3  # half-width of the column band probed under each balloon
GAP_T = 8.0
MAX_DROP = 90  # rows searched below a balloon for its string
MIN_RUN_W = 12
MIN_RUN_PX = 220
MAX_BALLOONS = 8


def _masks(image: Image.Image):
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    hi, lo = arr.max(axis=-1), arr.min(axis=-1)
    chroma = (hi - lo) >= CHROMA_MIN
    string = ((hi - lo) < CHROMA_MIN) & (hi <= STRING_MAX_V)
    return chroma, string


def _column_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    cols = mask.any(axis=0)
    runs, start = [], None
    for x, on in enumerate(cols):
        if on and start is None:
            start = x
        elif not on and start is not None:
            runs.append((start, x - 1))
            start = None
    if start is not None:
        runs.append((start, len(cols) - 1))
    return runs


def measure_gaps(image: Image.Image):
    """Visible white band, in pixels, under each balloon. None when unreadable."""
    chroma, string = _masks(image)
    runs = [r for r in _column_runs(chroma) if r[1] - r[0] + 1 >= MIN_RUN_W]
    runs = [r for r in runs if int(chroma[:, r[0] : r[1] + 1].sum()) >= MIN_RUN_PX]
    if not runs or len(runs) > MAX_BALLOONS:
        return None
    h = chroma.shape[0]
    gaps = []
    for x0, x1 in runs:
        cx = (x0 + x1) // 2
        lo_c, hi_c = max(0, cx - BAND), min(chroma.shape[1], cx + BAND + 1)
        band = chroma[:, lo_c:hi_c]
        rows = np.flatnonzero(band.any(axis=1))
        if rows.size == 0:
            return None
        bottom = int(rows[-1])
        sband = string[bottom + 1 : min(h, bottom + 1 + MAX_DROP), lo_c:hi_c]
        srows = np.flatnonzero(sband.any(axis=1))
        if srows.size == 0:
            return None  # no string under this balloon: nothing to judge
        gaps.append(float(srows[0]))
    return gaps


def decision_from_image(image: Image.Image) -> str:
    gaps = measure_gaps(image)
    if gaps is None:
        return ABSTAIN
    loose = sum(1 for g in gaps if g > GAP_T)
    if loose > 3:
        return ABSTAIN
    return str(loose)


def pixel_margin(image: Image.Image) -> float:
    gaps = measure_gaps(image)
    if gaps is None:
        return 0.0
    return round(min(abs(g - GAP_T) for g in gaps), 3)
