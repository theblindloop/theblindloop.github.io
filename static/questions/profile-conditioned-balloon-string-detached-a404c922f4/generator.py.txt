"""Scene sampler and renderer for balloon_string_detached.

Each image shows a row of balloons; every balloon has exactly one straight
string hanging below it. Some strings are TIED (their top touches the balloon)
and some are LOOSE (a clear band of white page separates the top of the string
from the balloon). The answer is how many strings are loose. The canonical
prior -- balloons are tied to their strings -- always predicts zero.

Every latent field is painted into the raster. The only field feeding analytic
gold is the per-balloon visible gap.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "balloon_string_detached-renderer-0.1.0"

SS = 3  # supersampling factor
WIDTH, HEIGHT = 540, 360

# A string counts as loose when the white band above it is wider than this many
# page pixels. The sampler never emits a gap inside (0, MIN_LOOSE_GAP), so the
# evidence is always either a touching string or a band at least 10 px wide.
GAP_T = 8.0
MIN_LOOSE_GAP = 10
MAX_LOOSE_GAP = 34
ATTACH_OVERLAP = 10  # tied strings start this far up inside the balloon

QUARANTINE_MARGIN = 4.0

N_MIN, N_MAX = 4, 6
K_MIN, K_MAX = 0, 3

FRAME = (205, 205, 205)
TICK = (170, 170, 170)
STRING = (60, 60, 60)
BG = (255, 255, 255)

PALETTE = [
    (206, 58, 62),
    (44, 106, 190),
    (232, 152, 40),
    (58, 148, 96),
    (150, 78, 178),
    (28, 158, 176),
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    n = N_MIN + (seed // 7) % (N_MAX - N_MIN + 1)
    k = K_MIN + seed % (K_MAX - K_MIN + 1)

    rx = rng.randint(20, 27)
    ry = rx + rng.randint(2, 7)
    string_len = rng.randint(66, 92)
    string_w = rng.choice([3, 4])

    loose_idx = sorted(rng.sample(range(n), k))
    colors = [list(PALETTE[(rng.randrange(len(PALETTE)) + i) % len(PALETTE)]) for i in range(n)]

    span = WIDTH - 96
    step = span / n
    balloons = []
    for i in range(n):
        cx = int(56 + step * (i + 0.5) + rng.randint(-6, 6))
        cy = int(96 + rng.randint(-18, 26))
        gap = rng.randint(MIN_LOOSE_GAP, MAX_LOOSE_GAP) if i in loose_idx else 0
        balloons.append(
            {"cx": cx, "cy": cy, "rx": rx, "ry": ry, "gap": gap, "color": colors[i]}
        )

    decor = [rng.randint(12, 22) for _ in range(4)]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "balloons": balloons,
        "string_len": string_len,
        "string_width": string_w,
        "decor": decor,
    }


def analytic_gold(scene: dict) -> str:
    return str(sum(1 for b in scene["balloons"] if float(b["gap"]) > GAP_T))


def margin(scene: dict) -> float:
    """Pixels of white band separating the reading from the runner-up count."""
    return round(min(abs(float(b["gap"]) - GAP_T) for b in scene["balloons"]), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every field is painted and recoverable."""
    return []


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    img = Image.new("RGB", (w * SS, h * SS), BG)
    d = ImageDraw.Draw(img)

    d.rectangle([6 * SS, 6 * SS, (w - 7) * SS, (h - 7) * SS], outline=FRAME, width=2 * SS)
    for i, off in enumerate(scene["decor"]):
        x = (14 + off) if i % 2 == 0 else (w - 15 - off)
        y = 14 if i < 2 else h - 15
        d.line([x * SS, y * SS, x * SS, (y + 9) * SS], fill=TICK, width=2 * SS)

    sw = int(scene["string_width"])
    slen = int(scene["string_len"])
    # strings first, balloons on top: a tied string is occluded by its balloon,
    # so the painted evidence is exactly the visible white band.
    for b in scene["balloons"]:
        cx, bottom, gap = int(b["cx"]), int(b["cy"]) + int(b["ry"]), int(b["gap"])
        top = bottom - ATTACH_OVERLAP if gap == 0 else bottom + gap
        d.line([cx * SS, top * SS, cx * SS, (top + slen) * SS], fill=STRING, width=sw * SS)
    for b in scene["balloons"]:
        cx, cy, rx, ry = int(b["cx"]), int(b["cy"]), int(b["rx"]), int(b["ry"])
        d.ellipse(
            [(cx - rx) * SS, (cy - ry) * SS, (cx + rx) * SS, (cy + ry) * SS],
            fill=tuple(b["color"]),
        )
    return img.resize((w, h), Image.LANCZOS)


def ink_pixels(image: Image.Image) -> int:
    import numpy as np

    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return int((arr.min(axis=-1) < 235).sum())


# ---------------------------------------------------------------- transforms
def translate(scene: dict, dx: int, dy: int) -> dict:
    out = dict(scene)
    out["balloons"] = [dict(b, cx=int(b["cx"]) + dx, cy=int(b["cy"]) + dy) for b in scene["balloons"]]
    return out


def recolor(scene: dict) -> dict:
    out = dict(scene)
    out["balloons"] = [
        dict(b, color=list(PALETTE[(PALETTE.index(tuple(b["color"])) + 3) % len(PALETTE)]))
        for b in scene["balloons"]
    ]
    return out


def resize_balloons(scene: dict, drx: int) -> dict:
    out = dict(scene)
    out["balloons"] = [
        dict(b, rx=int(b["rx"]) + drx, ry=int(b["ry"]) + drx) for b in scene["balloons"]
    ]
    return out


def permute_positions(scene: dict) -> dict:
    """Move the loose strings onto different balloons: the count is unchanged."""
    out = dict(scene)
    gaps = [int(b["gap"]) for b in scene["balloons"]]
    gaps = gaps[1:] + gaps[:1]
    out["balloons"] = [dict(b, gap=g) for b, g in zip(scene["balloons"], gaps)]
    return out


def restring(scene: dict, dlen: int) -> dict:
    out = dict(scene)
    out["string_len"] = int(scene["string_len"]) + dlen
    return out


def set_loose_count(scene: dict, k: int, gap: int = 22) -> dict:
    """Label-transforming edit: repaint with exactly k loose strings."""
    out = dict(scene)
    out["balloons"] = [
        dict(b, gap=(gap if i < k else 0)) for i, b in enumerate(scene["balloons"])
    ]
    return out
