"""Deterministic renderer and analytic arm for Cyclic Ternary Difference."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 480
N, CELL, GRID = 7, 16, 112
BACKGROUND = (249, 247, 241)
INK = (37, 43, 54)
MUTED = (103, 109, 119)
FRAME = (91, 65, 135)
GOLD = (232, 170, 45)

PALETTES = (
    ((241, 235, 218), (145, 156, 168), (38, 49, 70)),
    ((232, 242, 238), (130, 157, 151), (31, 55, 58)),
    ((244, 234, 239), (158, 140, 151), (55, 37, 49)),
    ((240, 239, 224), (157, 149, 116), (48, 46, 35)),
)


def _templates() -> dict[str, list[list[int]]]:
    out: dict[str, list[list[int]]] = {}
    x = [[int(r == c or r + c == 6) for c in range(N)] for r in range(N)]
    plus = [[int(r == 3 or c == 3) for c in range(N)] for r in range(N)]
    tee = [[int(r == 0 or c == 3) for c in range(N)] for r in range(N)]
    ell = [[int(c == 0 or r == 6) for c in range(N)] for r in range(N)]
    out["X"], out["plus"], out["T"], out["L"] = x, plus, tee, ell
    return out


TEMPLATES = _templates()
SYMBOLS = tuple(TEMPLATES)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    target = TEMPLATES[symbol]
    share_two = [[rng.randrange(3) for _ in range(N)] for _ in range(N)]
    difference = [
        [1 if target[r][c] else rng.choice((0, 2)) for c in range(N)]
        for r in range(N)
    ]
    share_one = [
        [(share_two[r][c] + difference[r][c]) % 3 for c in range(N)]
        for r in range(N)
    ]
    return {
        "symbol": symbol,
        "share_one": share_one,
        "share_two": share_two,
        "panel_order": rng.choice(("one_left", "two_left")),
        "palette": [list(rgb) for rgb in rng.choice(PALETTES)],
        "offset_x": rng.randint(-1, 1),
        "offset_y": rng.randint(-4, 4),
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    x0: int,
    y0: int,
    values: list[list[int]],
    palette: list[list[int]],
    marker_count: int,
) -> None:
    draw.rounded_rectangle(
        (x0 - 7, y0 - 7, x0 + GRID + 6, y0 + GRID + 6),
        radius=7,
        fill=FRAME,
    )
    draw.rectangle((x0, y0, x0 + GRID - 1, y0 + GRID - 1), fill=BACKGROUND)
    for r in range(N):
        for c in range(N):
            xx, yy = x0 + c * CELL, y0 + r * CELL
            draw.rectangle(
                (xx + 1, yy + 1, xx + CELL - 2, yy + CELL - 2),
                fill=tuple(palette[values[r][c]]),
            )
    center = x0 + GRID // 2
    dot_gap = 18
    start = center - (marker_count - 1) * dot_gap // 2
    for index in range(marker_count):
        cx = start + index * dot_gap
        draw.ellipse((cx - 6, y0 - 31, cx + 6, y0 - 19), fill=GOLD, outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font, body_font, small_font = _font(27, True), _font(17), _font(15, True)
    draw.text((WIDTH // 2, 22), "CYCLIC TERNARY DIFFERENCE", fill=INK, font=title_font, anchor="ma")
    draw.text(
        (WIDTH // 2, 62),
        "pale = 0   mid = 1   dark = 2     • minus •• (mod 3)     keep remainder 1",
        fill=MUTED,
        font=body_font,
        anchor="ma",
    )
    palette = scene["palette"]
    legend_palette = sorted(palette, key=lambda rgb: sum(rgb), reverse=True)
    for value, color in enumerate(legend_palette):
        sx = 295 + value * 46
        draw.rounded_rectangle((sx, 96, sx + 28, 118), radius=3, fill=tuple(color), outline=INK, width=1)
        draw.text((sx + 14, 121), str(value), fill=INK, font=small_font, anchor="ma")

    y0 = 188 + int(scene["offset_y"])
    left = 127 + int(scene["offset_x"])
    right = 470 + int(scene["offset_x"])
    if scene["panel_order"] == "one_left":
        panels = ((left, scene["share_one"], 1), (right, scene["share_two"], 2))
    else:
        panels = ((left, scene["share_two"], 2), (right, scene["share_one"], 1))
    for x0, values, count in panels:
        _draw_panel(draw, x0, y0, values, palette, count)

    draw.text(
        (WIDTH // 2, 397),
        "Subtract matching squares with wraparound; mark only results equal to 1.",
        fill=INK,
        font=body_font,
        anchor="ma",
    )
    draw.text((WIDTH // 2, 430), "X     +     T     L", fill=FRAME, font=title_font, anchor="ma")
    return image


def _decode(scene: dict) -> list[list[int]]:
    first = scene["share_one"]
    second = scene["share_two"]
    return [[int((first[r][c] - second[r][c]) % 3 == 1) for c in range(N)] for r in range(N)]


def _rank(mask: list[list[int]]) -> list[tuple[int, str]]:
    return sorted(
        (
            sum(mask[r][c] != template[r][c] for r in range(N) for c in range(N)),
            name,
        )
        for name, template in TEMPLATES.items()
    )


def analytic_gold(scene: dict) -> str:
    return _rank(_decode(scene))[0][1]


def margin(scene: dict) -> float:
    ranked = _rank(_decode(scene))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["share_one"] = [[(v + 1) % 3 for v in row] for row in scene["share_one"]]
    twin["share_two"] = [[(v + 1) % 3 for v in row] for row in scene["share_two"]]
    old = scene["palette"]
    twin["palette"] = [copy.deepcopy(old[(index - 1) % 3]) for index in range(3)]
    return [("increment_digits_counterrotate_palette", twin)]
