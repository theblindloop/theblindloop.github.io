"""Independent raster-only inverse for Cyclic Ternary Difference."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

FRAME = np.array((91, 65, 135), dtype=np.int16)
GOLD = np.array((232, 170, 45), dtype=np.int16)
N = 7


def _components(mask: np.ndarray, minimum: int = 1) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for yy, xx in zip(*np.nonzero(mask)):
        if seen[yy, xx]:
            continue
        queue = deque([(int(yy), int(xx))])
        seen[yy, xx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            result.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return result


def _panels(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    distance = np.max(np.abs(array.astype(np.int16) - FRAME), axis=2)
    comps = _components(distance <= 2, minimum=300)
    boxes = [(x0, y0, x1, y1) for x0, y0, x1, y1, _ in comps if 105 <= x1 - x0 <= 135 and 105 <= y1 - y0 <= 135]
    if len(boxes) != 2:
        raise ValueError("expected two intact violet panel frames")
    return sorted(boxes)


def _marker_count(array: np.ndarray, box: tuple[int, int, int, int]) -> int:
    x0, y0, x1, _ = box
    crop = array[max(0, y0 - 34):y0, max(0, x0 - 4):min(array.shape[1], x1 + 5)]
    dist = np.max(np.abs(crop.astype(np.int16) - GOLD), axis=2)
    return len(_components(dist <= 3, minimum=20))


def _cell_colors(array: np.ndarray, box: tuple[int, int, int, int]) -> list[list[np.ndarray]]:
    x0, y0, x1, y1 = box
    inner_x0, inner_y0 = x0 + 7, y0 + 7
    step_x = (x1 - x0 - 13) / N
    step_y = (y1 - y0 - 13) / N
    cells = []
    for r in range(N):
        row = []
        for c in range(N):
            cx = int(round(inner_x0 + (c + 0.5) * step_x))
            cy = int(round(inner_y0 + (r + 0.5) * step_y))
            patch = array[cy - 3:cy + 4, cx - 3:cx + 4]
            if patch.shape[:2] != (7, 7):
                raise ValueError("truncated panel cell")
            row.append(np.median(patch.reshape(-1, 3), axis=0))
        cells.append(row)
    return cells


def _decode_tones(all_cells: list[list[list[np.ndarray]]]) -> list[list[list[int]]]:
    flat = np.array([color for panel in all_cells for row in panel for color in row], dtype=float)
    luminance = flat @ np.array((0.2126, 0.7152, 0.0722))
    centers = np.array((np.min(luminance), np.median(luminance), np.max(luminance)), dtype=float)
    for _ in range(12):
        labels = np.argmin(np.abs(luminance[:, None] - centers[None, :]), axis=1)
        new = np.array([np.mean(luminance[labels == k]) if np.any(labels == k) else centers[k] for k in range(3)])
        if np.allclose(new, centers):
            break
        centers = new
    order = np.argsort(centers)[::-1]
    rank = {int(cluster): code for code, cluster in enumerate(order)}
    labels = np.argmin(np.abs(luminance[:, None] - centers[None, :]), axis=1)
    if min(abs(centers[i] - centers[j]) for i in range(3) for j in range(i)) < 28:
        raise ValueError("tone levels are not separated")
    decoded = []
    cursor = 0
    for _ in all_cells:
        panel = []
        for _r in range(N):
            panel.append([rank[int(labels[cursor + c])] for c in range(N)])
            cursor += N
        decoded.append(panel)
    return decoded


def _templates() -> dict[str, list[list[int]]]:
    return {
        "X": [[int(r == c or r + c == 6) for c in range(N)] for r in range(N)],
        "plus": [[int(r == 3 or c == 3) for c in range(N)] for r in range(N)],
        "T": [[int(r == 0 or c == 3) for c in range(N)] for r in range(N)],
        "L": [[int(c == 0 or r == 6) for c in range(N)] for r in range(N)],
    }


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    boxes = _panels(array)
    counts = [_marker_count(array, box) for box in boxes]
    if sorted(counts) != [1, 2]:
        raise ValueError("panel dot fiducials are missing or ambiguous")
    decoded = _decode_tones([_cell_colors(array, box) for box in boxes])
    one = decoded[counts.index(1)]
    two = decoded[counts.index(2)]
    mask = [[int((one[r][c] - two[r][c]) % 3 == 1) for c in range(N)] for r in range(N)]
    ranked = sorted(
        (sum(mask[r][c] != templ[r][c] for r in range(N) for c in range(N)), name)
        for name, templ in _templates().items()
    )
    if ranked[0][0] > 3 or ranked[1][0] - ranked[0][0] < 4:
        raise ValueError("transformed raster has insufficient template margin")
    return ranked[0][1]
