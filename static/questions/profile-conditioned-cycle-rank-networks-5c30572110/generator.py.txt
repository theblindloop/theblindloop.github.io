"""Deterministic renderer and analytic graph invariant for one candidate.

The public scene contains three disjoint five-node planar graphs.  Every graph
has a five-edge cycle plus zero, one, or two non-crossing chords, so its cycle
rank is respectively 1, 2, or 3.  Nodes are colored; links are neutral.  The
pixel oracle therefore has to recover both kinds of visual primitive.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "cycle_rank_networks-0.1.0"
WIDTH, HEIGHT = 1080, 420
BACKGROUND = (248, 248, 244)
CHARCOAL = (47, 54, 64)
COLORS = {
    "red": (214, 67, 73),
    "blue": (62, 103, 193),
    "green": (54, 157, 94),
}
SLOTS = (180.0, 540.0, 900.0)
NODE_RADIUS = 17.0
EDGE_WIDTH = 8
SUPERSAMPLE = 3


def _edge(a: int, b: int) -> list[int]:
    return [min(a, b), max(a, b)]


def _edges_for(anchor: int, rank: int) -> list[list[int]]:
    edges = [_edge(index, (index + 1) % 5) for index in range(5)]
    if rank >= 2:
        edges.append(_edge(anchor, (anchor + 2) % 5))
    if rank >= 3:
        edges.append(_edge(anchor, (anchor + 3) % 5))
    return edges


def _node_positions(
    cx: float, cy: float, rx: float, ry: float, phase: float
) -> list[list[float]]:
    return [
        [
            float(cx + rx * math.cos(phase + 2.0 * math.pi * index / 5.0)),
            float(cy + ry * math.sin(phase + 2.0 * math.pi * index / 5.0)),
        ]
        for index in range(5)
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene from one integer seed."""

    rng = random.Random(int(seed))
    ranks = [1, 2, 3]
    colors = list(COLORS)
    rng.shuffle(ranks)
    rng.shuffle(colors)
    networks: list[dict[str, Any]] = []
    for slot, color, rank in zip(SLOTS, colors, ranks):
        cx = slot + rng.uniform(-18.0, 18.0)
        cy = 210.0 + rng.uniform(-12.0, 12.0)
        rx = rng.uniform(82.0, 95.0)
        ry = rng.uniform(74.0, 87.0)
        phase = rng.uniform(-0.25, 0.25)
        anchor = rng.randrange(5)
        nodes = _node_positions(cx, cy, rx, ry, phase)
        networks.append(
            {
                "color": color,
                "nodes": nodes,
                "edges": _edges_for(anchor, rank),
                "anchor": anchor,
            }
        )
    return {"networks": networks}


def _scaled(point: list[float], scale: int) -> tuple[int, int]:
    return int(round(point[0] * scale)), int(round(point[1] * scale))


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields into a deterministic RGB image."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for network in scene["networks"]:
        nodes = network["nodes"]
        for left, right in network["edges"]:
            draw.line(
                [_scaled(nodes[left], scale), _scaled(nodes[right], scale)],
                fill=CHARCOAL,
                width=EDGE_WIDTH * scale,
            )
        fill = COLORS[network["color"]]
        radius = int(round(NODE_RADIUS * scale))
        for node in nodes:
            x, y = _scaled(node, scale)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _component_count(node_count: int, edges: list[list[int]]) -> int:
    parent = list(range(node_count))

    def find(value: int) -> int:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    for left, right in edges:
        a, b = find(int(left)), find(int(right))
        if a != b:
            parent[b] = a
    return len({find(index) for index in range(node_count)})


def _rank(network: dict[str, Any]) -> int:
    nodes = network["nodes"]
    edges = network["edges"]
    return len(edges) - len(nodes) + _component_count(len(nodes), edges)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the color of the unique rank-two visible graph."""

    # Keep the analytic answer's only top-level scene input explicit.  The
    # nested nodes and edges are part of the rendered networks object rather
    # than hidden answer fields.
    matches = [
        network["color"]
        for network in scene["networks"]
        if len(network["edges"]) - len(network["nodes"]) + 1 == 2
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one rank-two network, found {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest node/link clearance in the rendered geometry."""

    clearances: list[float] = []
    for network in scene["networks"]:
        nodes = network["nodes"]
        for index in range(len(nodes)):
            for other in range(index + 1, len(nodes)):
                distance = math.dist(nodes[index], nodes[other])
                clearances.append(distance - 2.0 * NODE_RADIUS)
        for left, right in network["edges"]:
            clearances.append(math.dist(nodes[left], nodes[right]) - 2.0 * NODE_RADIUS)
    return float(min(clearances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No render-invariant latent aliases are claimed for this scene encoding."""

    return []
