"""Deterministic RGB renderer and analytic label for the link-depth world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


WIDTH = 640
HEIGHT = 480
SCALE = 3


PALETTES = (
    {
        "background": (248, 249, 246),
        "route": (35, 48, 63),
        "green": (37, 174, 104),
        "red": (221, 68, 73),
        "blue": (57, 111, 218),
        "gold": (232, 163, 52),
        "purple": (174, 75, 180),
        "distractor": (204, 213, 218),
    },
    {
        "background": (247, 249, 252),
        "route": (42, 53, 70),
        "green": (29, 166, 113),
        "red": (224, 75, 78),
        "blue": (65, 105, 222),
        "gold": (235, 170, 59),
        "purple": (183, 80, 169),
        "distractor": (202, 211, 219),
    },
    {
        "background": (250, 247, 244),
        "route": (48, 48, 57),
        "green": (41, 171, 104),
        "red": (218, 70, 76),
        "blue": (55, 113, 207),
        "gold": (230, 158, 47),
        "purple": (169, 76, 177),
        "distractor": (210, 207, 214),
    },
)


def _palette(palette_id: int) -> dict[str, tuple[int, int, int]]:
    return PALETTES[int(palette_id) % len(PALETTES)]


def _transform(point: Iterable[float], scene: dict) -> tuple[float, float]:
    x, y = float(point[0]), float(point[1])
    cx, cy = float(scene["width"]) / 2.0, float(scene["height"]) / 2.0
    dx, dy = x - cx, y - cy
    rotation = int(scene["rotation"]) % 360
    if rotation == 90:
        dx, dy = -dy, dx
    elif rotation == 180:
        dx, dy = -dx, -dy
    elif rotation == 270:
        dx, dy = dy, -dx
    tx, ty = scene["translation"]
    return cx + dx + float(tx), cy + dy + float(ty)


def _points(scene: dict, points: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    return [_transform(point, scene) for point in points]


def _branch_points(
    start: list[float],
    anchor: list[float],
    end: list[float],
    links: int,
    wobble: float,
    phase: float,
) -> list[list[float]]:
    points: list[list[float]] = [list(start), list(anchor)]
    ax, ay = anchor
    ex, ey = end
    intermediate = max(0, links - 2)
    for index in range(1, intermediate + 1):
        t = index / float(intermediate + 1)
        curve = math.sin(math.pi * t) ** 2
        x = ax + (ex - ax) * t + 0.18 * wobble * math.sin(2.0 * math.pi * t + phase)
        y = ay + (ey - ay) * t + wobble * curve
        points.append([round(x, 2), round(y, 2)])
    points.append(list(end))
    return points


def sample_scene(seed: int) -> dict:
    """Sample one balanced rooted tree from an integer seed."""

    rng = random.Random(int(seed))
    palette_id = rng.randrange(len(PALETTES))
    rotation = rng.choice((0, 90, 180, 270))
    translation = [rng.randint(-8, 8), rng.randint(-6, 6)]

    source = [92.0, 240.0]
    fork = [258.0 + rng.randint(-10, 10), 240.0 + rng.randint(-7, 7)]
    ends = [
        [548.0 + rng.randint(-8, 8), 104.0 + rng.randint(-7, 7)],
        [548.0 + rng.randint(-8, 8), 240.0 + rng.randint(-10, 10)],
        [548.0 + rng.randint(-8, 8), 376.0 + rng.randint(-7, 7)],
    ]
    anchors = [
        [344.0 + rng.randint(-6, 6), 150.0 + rng.randint(-6, 6)],
        [344.0 + rng.randint(-6, 6), 252.0 + rng.randint(-6, 6)],
        [344.0 + rng.randint(-6, 6), 330.0 + rng.randint(-6, 6)],
    ]

    branch_depths = [2, 3, 4]
    rng.shuffle(branch_depths)
    deep_branch = branch_depths.index(max(branch_depths))

    labels = ["red", "blue", "gold"]
    color_rng = random.Random(int(seed) ^ 0x5EED123)
    target_color = labels[color_rng.randrange(len(labels))]
    terminal_colors: list[str | None] = [None, None, None]
    terminal_colors[deep_branch] = target_color
    remaining = [label for label in labels if label != target_color]
    rng.shuffle(remaining)
    for branch_index in range(3):
        if terminal_colors[branch_index] is None:
            terminal_colors[branch_index] = remaining.pop()

    branch_paths = []
    for branch_index, end in enumerate(ends):
        wobble = float(rng.choice((12, 15, 18, 21)))
        phase = rng.uniform(-0.4, 0.4)
        branch_paths.append(
            _branch_points(
                fork,
                anchors[branch_index],
                end,
                int(branch_depths[branch_index]),
                wobble,
                phase,
            )
        )

    junction_centers: list[list[float]] = [list(fork)]
    for path in branch_paths:
        junction_centers.extend([list(point) for point in path[1:-1]])

    distractors = [
        [[104.0, 72.0], [176.0 + rng.randint(-9, 9), 58.0], [242.0, 78.0]],
        [[378.0, 425.0], [465.0 + rng.randint(-10, 10), 410.0], [582.0, 430.0]],
    ]
    palette = _palette(palette_id)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(palette["background"]),
        "route_color": list(palette["route"]),
        "junction_color": list(palette["purple"]),
        "source_color": "green",
        "terminal_colors": terminal_colors,
        "source": source,
        "trunk_path": [source, fork],
        "branch_paths": branch_paths,
        "junction_centers": junction_centers,
        "stroke_width": 8,
        "junction_radius": rng.choice((12, 13)),
        "source_radius": rng.choice((16, 17, 18)),
        "terminal_radii": [rng.choice((15, 16, 17, 18)) for _ in range(3)],
        "rotation": rotation,
        "translation": translation,
        "palette_id": palette_id,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    palette = _palette(scene["palette_id"])
    scale = SCALE
    image = Image.new(
        "RGB",
        (int(scene["width"]) * scale, int(scene["height"]) * scale),
        tuple(scene["background"]),
    )
    draw = ImageDraw.Draw(image)

    def scaled(point: Iterable[float]) -> tuple[int, int]:
        return (round(float(point[0]) * scale), round(float(point[1]) * scale))

    def line(points: Iterable[Iterable[float]], fill: tuple[int, int, int], width: int) -> None:
        draw.line(
            [scaled(point) for point in _points(scene, points)],
            fill=fill,
            width=int(width) * scale,
            joint="curve",
        )

    def marker(center: Iterable[float], radius: float, fill: tuple[int, int, int]) -> None:
        cx, cy = scaled(_transform(center, scene))
        rr = round(float(radius) * scale)
        draw.ellipse(
            (cx - rr, cy - rr, cx + rr, cy + rr),
            fill=fill,
            outline=(255, 255, 255),
            width=2 * scale,
        )

    for distractor in scene["distractors"]:
        line(distractor, tuple(palette["distractor"]), 4)

    route = tuple(scene["route_color"])
    line(scene["trunk_path"], route, int(scene["stroke_width"]))
    for path in scene["branch_paths"]:
        line(path, route, int(scene["stroke_width"]))

    junction_fill = tuple(scene["junction_color"])
    for center in scene["junction_centers"]:
        marker(center, float(scene["junction_radius"]), junction_fill)

    marker(scene["source"], float(scene["source_radius"]), tuple(palette[scene["source_color"]]))
    for branch_index, path in enumerate(scene["branch_paths"]):
        marker(
            path[-1],
            float(scene["terminal_radii"][branch_index]),
            tuple(palette[str(scene["terminal_colors"][branch_index])]),
        )

    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    counts = [len(path) - 1 for path in scene["branch_paths"]]
    winner = max(range(len(counts)), key=counts.__getitem__)
    return str(scene["terminal_colors"][winner])


def margin(scene: dict) -> float:
    counts = sorted((len(path) - 1 for path in scene["branch_paths"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
