"""Sampler and renderer for a cross-world frozen-value intervention."""

from __future__ import annotations

import random
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

SIZE = (768, 640)
BACKGROUND = (247, 245, 238); INK = (35, 42, 52); CARD = (255, 255, 252)
BORDER = (105, 113, 124); ONE = (30, 55, 78); ZERO = (229, 232, 226)
XOR = (126, 73, 163); AND = (28, 126, 125)
NOW = (238, 163, 38); SAMPLE = (35, 137, 190); PATCH = (91, 77, 166); PLAIN = (238, 239, 235)
INPUT_X = (12, 203, 394, 585); GATE_X = (8, 264, 520); GATE_Y = (166, 302)
OUTPUT_X = (5, 198, 391, 584); OUTPUT_Y = 466
DECISIONS = ("output-A", "output-B", "output-C", "output-D")
GATE_PARENTS = (("I1", "I2"), ("I2", "I3"), ("I3", "I4"), ("G1", "G2"), ("G2", "G3"), ("G1", "G3"))
OUTPUT_PARENTS = (("G4", "G5"), ("G4", "G6"), ("G5", "G6"), ("G4", "G3"))


def _font(size: int) -> ImageFont.ImageFont:
    path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path), size=size) if path.exists() else ImageFont.load_default()


def _apply(op: str, a: int, b: int) -> int:
    if op == "XOR": return int(bool(a) != bool(b))
    if op == "AND": return int(bool(a) and bool(b))
    raise ValueError(op)


def _evaluate_overrides(inputs: list[int], operations: list[str], overrides: dict[int, int]) -> tuple[list[int], list[int], list[int]]:
    """Return gates, outputs, and raw values arriving at the twelve sockets."""
    if len(inputs) != 4 or len(operations) != 10: raise ValueError("wrong circuit size")
    if any(not 0 <= int(slot) < 12 or int(value) not in (0, 1) for slot, value in overrides.items()):
        raise ValueError("invalid override")
    values = {f"I{i + 1}": int(value) for i, value in enumerate(inputs)}
    gates: list[int] = []; sockets: list[int] = []
    for index, (left, right) in enumerate(GATE_PARENTS):
        raw_a, raw_b = values[left], values[right]
        sockets.extend((raw_a, raw_b))
        a = overrides.get(2 * index, raw_a); b = overrides.get(2 * index + 1, raw_b)
        value = _apply(operations[index], a, b)
        gates.append(value); values[f"G{index + 1}"] = value
    outputs = [_apply(operations[6 + i], values[a], values[b]) for i, (a, b) in enumerate(OUTPUT_PARENTS)]
    return gates, outputs, sockets


def evaluate(inputs: list[int], operations: list[str], clamp_slot: int | None = None,
             forced_value: int | None = None) -> tuple[list[int], list[int]]:
    overrides = {} if clamp_slot is None else {int(clamp_slot): int(forced_value)}
    gates, outputs, _sockets = _evaluate_overrides(inputs, operations, overrides)
    return gates, outputs


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x46524F5A); wanted = int(seed) % 4
    for _ in range(350000):
        inputs = [rng.randrange(2) for _ in range(4)]
        operations = [rng.choice(("XOR", "AND")) for _ in range(10)]
        forced_value = rng.randrange(2); now_slot = rng.randrange(6)
        _ng, natural_outputs, natural_sockets = _evaluate_overrides(inputs, operations, {})
        current_gates, current_outputs, current_sockets = _evaluate_overrides(inputs, operations, {now_slot: forced_value})
        if current_outputs == natural_outputs or len(set(current_outputs)) < 2: continue
        sample_choices = [slot for slot in range(6, 12)
                          if current_sockets[slot] != natural_sockets[slot]
                          and current_sockets[slot] != forced_value]
        if not sample_choices: continue
        sample_slot = rng.choice(sample_choices); frozen = current_sockets[sample_slot]
        patch_choices = [slot for slot in range(12) if slot not in (now_slot, sample_slot)]
        rng.shuffle(patch_choices)
        for patch_slot in patch_choices:
            future_gates, future_outputs, _ = _evaluate_overrides(inputs, operations, {patch_slot: frozen})
            changed = [i for i, (a, b) in enumerate(zip(current_outputs, future_outputs)) if a != b]
            if changed != [wanted]: continue
            _wg, wrong_literal, _ = _evaluate_overrides(inputs, operations, {patch_slot: forced_value})
            wrong_changed = [i for i, (a, b) in enumerate(zip(current_outputs, wrong_literal)) if a != b]
            if len(wrong_changed) != 1 or wrong_changed == changed: continue
            _kg, wrong_keep, _ = _evaluate_overrides(inputs, operations, {now_slot: forced_value, patch_slot: frozen})
            keep_changed = [i for i, (a, b) in enumerate(zip(current_outputs, wrong_keep)) if a != b]
            if len(keep_changed) != 1 or keep_changed == changed: continue
            if sum(a != b for a, b in zip(current_gates, future_gates)) < 2: continue
            return {"inputs": inputs, "operations": operations, "forced_value": forced_value,
                    "now_slot": now_slot, "sample_slot": sample_slot, "patch_slot": patch_slot,
                    "current_gates": current_gates, "current_outputs": current_outputs}
    raise RuntimeError("could not sample balanced frozen-sample scene")


def _validate(scene: dict) -> tuple[list[int], list[str], int, int, int, int, list[int], list[int]]:
    inputs = [int(v) for v in scene["inputs"]]; operations = [str(v) for v in scene["operations"]]
    forced = int(scene["forced_value"]); now = int(scene["now_slot"])
    sample = int(scene["sample_slot"]); patch = int(scene["patch_slot"])
    gates = [int(v) for v in scene["current_gates"]]; outputs = [int(v) for v in scene["current_outputs"]]
    if len(inputs) != 4 or len(operations) != 10 or len(gates) != 6 or len(outputs) != 4: raise ValueError("wrong size")
    if any(v not in (0, 1) for v in inputs + gates + outputs + [forced]): raise ValueError("nonbinary")
    if any(op not in ("XOR", "AND") for op in operations): raise ValueError("bad operation")
    if len({now, sample, patch}) != 3 or any(not 0 <= slot < 12 for slot in (now, sample, patch)): raise ValueError("bad sockets")
    expected_gates, expected_outputs, _ = _evaluate_overrides(inputs, operations, {now: forced})
    if (gates, outputs) != (expected_gates, expected_outputs): raise ValueError("inconsistent current state")
    return inputs, operations, forced, now, sample, patch, gates, outputs


def analytic_gold(scene: dict) -> str:
    try:
        inputs, operations, forced, now, sample, patch, _gates, current = _validate(scene)
        _cg, _co, current_sockets = _evaluate_overrides(inputs, operations, {now: forced})
        frozen = current_sockets[sample]
        _future_gates, future, _ = _evaluate_overrides(inputs, operations, {patch: frozen})
        changed = [i for i, (a, b) in enumerate(zip(current, future)) if a != b]
        return DECISIONS[changed[0]] if len(changed) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError): return "abstain"


def margin(scene: dict) -> float: return 1.0 if analytic_gold(scene) in DECISIONS else 0.0
def is_quarantined(scene: dict) -> bool: return analytic_gold(scene) == "abstain"
def latent_symmetries(scene: dict) -> list[tuple[str, dict]]: return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _lamp(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int, radius: int = 20) -> None:
    x, y = center; draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=ONE if value else ZERO, outline=INK, width=3)
    _centered(draw, center, str(value), _font(max(16, radius+2)), (255,255,255) if value else INK)


def _operation(draw: ImageDraw.ImageDraw, box, op: str) -> None:
    draw.rounded_rectangle(box, radius=9, fill=XOR if op == "XOR" else AND)
    _centered(draw, ((box[0]+box[2])//2, (box[1]+box[3])//2), op, _font(16), (255,255,255))


def render(scene: dict) -> Image.Image:
    inputs, operations, forced, now, sample, patch, gates, outputs = _validate(scene)
    image = Image.new("RGB", SIZE, BACKGROUND); draw = ImageDraw.Draw(image)
    _centered(draw, (305, 24), "DISPLAYED CURRENT CIRCUIT", _font(22), INK)
    draw.text((596, 8), "NOW BIT", font=_font(15), fill=INK); _lamp(draw, (721, 28), forced, 18)
    _centered(draw, (384, 53), "amber NOW     blue SAMPLE     violet PATCH", _font(17), INK)
    for i, (x, value) in enumerate(zip(INPUT_X, inputs)):
        draw.rounded_rectangle((x,66,x+171,137), radius=12, fill=CARD, outline=BORDER, width=3)
        draw.text((x+16,86), f"INPUT I{i+1}", font=_font(19), fill=INK); _lamp(draw, (x+142,102), value)
    for i, ((left, right), value) in enumerate(zip(GATE_PARENTS, gates)):
        x, y = GATE_X[i%3], GATE_Y[i//3]
        draw.rectangle((x,y,x+240,y+115), fill=CARD, outline=BORDER, width=3); draw.text((x+10,y+10), f"G{i+1}", font=_font(19), fill=INK)
        for side, label in enumerate((left,right)):
            slot = 2*i+side; sx = x+(12 if side==0 else 140)
            color = NOW if slot==now else SAMPLE if slot==sample else PATCH if slot==patch else PLAIN
            draw.rounded_rectangle((sx,y+50,sx+48,y+96), radius=7, fill=color, outline=INK, width=2)
            _centered(draw, (sx+24,y+73), label, _font(17), (255,255,255) if slot in (sample,patch) else INK)
            if slot in (now,sample,patch):
                role = "NOW" if slot==now else "SAMPLE" if slot==sample else "PATCH"
                _centered(draw, (sx+24,y+42), role, _font(10), color)
        _operation(draw, (x+69,y+53,x+133,y+93), operations[i]); draw.text((x+191,y+66), "=", font=_font(18), fill=INK)
        _lamp(draw, (x+220,y+74), value, 18)
    draw.line((18,438,750,438), fill=BORDER, width=3); _centered(draw, (384,450), "DISPLAYED CURRENT OUTPUTS", _font(17), INK)
    for i, ((left,right), value) in enumerate(zip(OUTPUT_PARENTS, outputs)):
        x,y=OUTPUT_X[i],OUTPUT_Y; draw.rounded_rectangle((x,y,x+179,y+143),radius=12,fill=CARD,outline=BORDER,width=3)
        draw.text((x+12,y+12),f"OUTPUT {chr(65+i)}",font=_font(18),fill=INK); draw.text((x+10,y+69),left,font=_font(15),fill=INK)
        _operation(draw,(x+48,y+54,x+113,y+94),operations[6+i]); draw.text((x+121,y+69),right,font=_font(15),fill=INK)
        _lamp(draw,(x+89,y+118),value,18)
    return image
