"""Independent fixed-layout pixel inverse for the frozen-sample patch."""

from __future__ import annotations
import numpy as np
from PIL import Image

BACKGROUND=(247,245,238); ONE=(30,55,78); ZERO=(229,232,226); XOR=(126,73,163); AND=(28,126,125)
NOW=(238,163,38); SAMPLE=(35,137,190); PATCH=(91,77,166); PLAIN=(238,239,235)
INPUT_X=(12,203,394,585); GATE_X=(8,264,520); GATE_Y=(166,302); OUTPUT_X=(5,198,391,584); OUTPUT_Y=466
DECISIONS=("output-A","output-B","output-C","output-D")
GATE_PARENTS=(("I1","I2"),("I2","I3"),("I3","I4"),("G1","G2"),("G2","G3"),("G1","G3"))
OUTPUT_PARENTS=(("G4","G5"),("G4","G6"),("G5","G6"),("G4","G3"))


def _near(pixel, color, tolerance=4):
    return bool(np.max(np.abs(pixel.astype(np.int16)-np.array(color,dtype=np.int16)))<=tolerance)


def _bit(arr, x, y):
    pixel=arr[y,x+10]
    if _near(pixel,ONE): return 1
    if _near(pixel,ZERO): return 0
    return None


def _apply(op,a,b): return int(bool(a)!=bool(b)) if op=="XOR" else int(bool(a) and bool(b))


def _evaluate(inputs, operations, overrides):
    values={f"I{i+1}":int(v) for i,v in enumerate(inputs)}; gates=[]; sockets=[]
    for i,(left,right) in enumerate(GATE_PARENTS):
        raw_a,raw_b=values[left],values[right]; sockets.extend((raw_a,raw_b))
        a=overrides.get(2*i,raw_a); b=overrides.get(2*i+1,raw_b)
        value=_apply(operations[i],a,b); gates.append(value); values[f"G{i+1}"]=value
    outputs=[_apply(operations[6+i],values[a],values[b]) for i,(a,b) in enumerate(OUTPUT_PARENTS)]
    return gates,outputs,sockets


def decision_from_image(image: Image.Image) -> str:
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(640,768,3) or not _near(arr[630,760],BACKGROUND): return "abstain"
    inputs=[_bit(arr,x+142,102) for x in INPUT_X]; forced=_bit(arr,721,28)
    if forced is None or any(v is None for v in inputs): return "abstain"
    operations=[]; role_hits={"now":[],"sample":[],"patch":[]}; displayed_gates=[]
    for i in range(6):
        x,y=GATE_X[i%3],GATE_Y[i//3]
        pixel=arr[y+56,x+72]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        value=_bit(arr,x+220,y+74)
        if value is None: return "abstain"
        displayed_gates.append(value)
        for side in range(2):
            sx=x+(12 if side==0 else 140); mark=arr[y+55,sx+5]; slot=2*i+side
            if _near(mark,NOW): role_hits["now"].append(slot)
            elif _near(mark,SAMPLE): role_hits["sample"].append(slot)
            elif _near(mark,PATCH): role_hits["patch"].append(slot)
            elif not _near(mark,PLAIN): return "abstain"
    displayed_outputs=[]
    for x in OUTPUT_X:
        pixel=arr[OUTPUT_Y+57,x+51]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        value=_bit(arr,x+89,OUTPUT_Y+118)
        if value is None: return "abstain"
        displayed_outputs.append(value)
    if any(len(role_hits[role])!=1 for role in role_hits): return "abstain"
    now,sample,patch=(role_hits[role][0] for role in ("now","sample","patch"))
    if len({now,sample,patch})!=3: return "abstain"
    current_gates,current_outputs,current_sockets=_evaluate(inputs,operations,{now:forced})
    if current_gates!=displayed_gates or current_outputs!=displayed_outputs: return "abstain"
    frozen=current_sockets[sample]
    _future_gates,future_outputs,_future_sockets=_evaluate(inputs,operations,{patch:frozen})
    changed=[i for i,(a,b) in enumerate(zip(displayed_outputs,future_outputs)) if a!=b]
    return DECISIONS[changed[0]] if len(changed)==1 else "abstain"
