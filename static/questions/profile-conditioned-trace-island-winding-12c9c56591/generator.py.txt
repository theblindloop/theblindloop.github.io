"""Deterministic renderer and analytic reference for the island-winding world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 560
HEIGHT = 440
ISLAND_FILL = (214, 225, 232)
ISLAND_OUTLINE = (116, 125, 135)
GREEN = (45, 174, 99)
RED = (222, 76, 76)
WHITE = (255, 255, 255)


def _point(cx: float, cy: float, radius: float, angle: float) -> tuple[float, float]:
    return (cx + radius * math.cos(angle), cy + radius * math.sin(angle))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winding_sign = 1 if rng.randrange(2) == 0 else -1
    near_boundary = rng.random() < 0.22
    sweep = rng.uniform(1.22, 1.72) if near_boundary else rng.uniform(2.25, 4.65)
    cx = rng.uniform(220.0, 340.0)
    cy = rng.uniform(205.0, 235.0)
    island_radius = rng.uniform(48.0, 66.0)
    path_radius = island_radius + rng.uniform(34.0, 58.0)
    start_angle = rng.uniform(-math.pi, math.pi)
    phase = rng.uniform(-math.pi, math.pi)
    arc_count = rng.randint(10, 14)

    arc_points: list[tuple[float, float]] = []
    for index in range(arc_count + 1):
        fraction = index / arc_count
        angle = start_angle + winding_sign * sweep * fraction
        radius = path_radius + 5.0 * math.sin(math.pi * fraction + phase)
        arc_points.append(_point(cx, cy, radius, angle))

    start_radius = path_radius + rng.uniform(48.0, 66.0)
    terminal_radius = path_radius + rng.uniform(48.0, 66.0)
    path_points = [
        _point(cx, cy, start_radius, start_angle),
        _point(cx, cy, path_radius, start_angle),
        *arc_points[1:-1],
        _point(cx, cy, path_radius, start_angle + winding_sign * sweep),
        _point(cx, cy, terminal_radius, start_angle + winding_sign * sweep),
    ]

    # Keep all declared fields JSON-native and make the label sign itself
    # redundant with the rendered centerline rather than hidden from pixels.
    return {
        "winding_sign": winding_sign,
        "island_x": round(cx, 3),
        "island_y": round(cy, 3),
        "island_radius": round(island_radius, 3),
        "path_points": [[round(x, 3), round(y, 3)] for x, y in path_points],
        "start_color": list(GREEN),
        "terminal_color": list(RED),
        "stroke_width": rng.randint(7, 9),
        "port_radius": rng.randint(12, 16),
        "background": list(rng.choice([(250, 249, 245), (246, 249, 251), (250, 247, 243)])),
        "path_color": list(rng.choice([(28, 49, 74), (35, 52, 70), (25, 62, 65)])),
        "near_boundary": near_boundary,
    }


def _signed_turn(points: Iterable[Iterable[float]], cx: float, cy: float) -> float:
    vectors = [(float(point[0]) - cx, float(point[1]) - cy) for point in points]
    total = 0.0
    for first, second in zip(vectors, vectors[1:]):
        cross = first[0] * second[1] - first[1] * second[0]
        dot = first[0] * second[0] + first[1] * second[1]
        total += math.atan2(cross, dot)
    return total


def analytic_gold(scene: dict) -> str:
    turn = _signed_turn(scene["path_points"], float(scene["island_x"]), float(scene["island_y"]))
    return "clockwise" if turn > 0.0 else "counterclockwise"


def margin(scene: dict) -> float:
    turn = abs(_signed_turn(scene["path_points"], float(scene["island_x"]), float(scene["island_y"])))
    return turn - 1.8


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx = float(scene["island_x"])
    cy = float(scene["island_y"])
    radius = float(scene["island_radius"])
    island_box = (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius))
    draw.ellipse(island_box, fill=ISLAND_FILL, outline=ISLAND_OUTLINE, width=3)

    points = [(round(float(point[0])), round(float(point[1]))) for point in scene["path_points"]]
    draw.line(
        points,
        fill=tuple(scene["path_color"]),
        width=int(scene["stroke_width"]),
        joint="curve",
    )

    port_radius = int(scene["port_radius"])
    for point, color in ((points[0], tuple(scene["start_color"])), (points[-1], tuple(scene["terminal_color"]))):
        x, y = point
        halo = port_radius + 4
        draw.ellipse((x - halo, y - halo, x + halo, y + halo), fill=WHITE)
        draw.ellipse((x - port_radius, y - port_radius, x + port_radius, y + port_radius), fill=color)
        draw.ellipse(
            (x - port_radius, y - port_radius, x + port_radius, y + port_radius),
            outline=WHITE,
            width=2,
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
