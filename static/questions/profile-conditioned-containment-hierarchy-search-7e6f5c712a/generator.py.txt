"""Deterministic renderer for colored-circle containment hierarchy search."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 640
LABELS = ("A", "B", "C", "D")
COLOR_NAMES = ("red", "green", "blue", "purple")
PALETTE = {
    "red": (220, 55, 65),
    "green": (35, 155, 95),
    "blue": (50, 105, 220),
    "purple": (150, 70, 185),
}

# (color, x, y, radius), in panel-local coordinates. Every template contains
# the same four colors and equal stroke width; only the containment relation differs.
TEMPLATES = {
    "chain4": [
        ("red", 0, 0, 78), ("green", 0, 0, 57),
        ("blue", 0, 0, 38), ("purple", 0, 0, 19),
    ],
    "fork_nested": [
        ("red", 0, 0, 79), ("green", -30, 0, 41),
        ("blue", 40, 8, 25), ("purple", -30, 0, 18),
    ],
    "three_children": [
        ("red", 0, 0, 79), ("green", -34, -18, 21),
        ("blue", 34, -18, 21), ("purple", 0, 34, 21),
    ],
    "two_pairs": [
        ("red", -46, 0, 41), ("green", -46, 0, 19),
        ("blue", 46, 0, 41), ("purple", 46, 0, 19),
    ],
    "chain3_free": [
        ("red", -28, 0, 57), ("green", -28, 0, 37),
        ("blue", -28, 0, 18), ("purple", 57, 0, 22),
    ],
    "fork_free": [
        ("red", -27, 0, 59), ("green", -51, 0, 20),
        ("blue", 1, 0, 20), ("purple", 57, 0, 22),
    ],
}
CODES = tuple(TEMPLATES)


def _relations(code: str) -> frozenset[tuple[str, str]]:
    """Directed transitive full-outline containment pairs (inner, outer)."""
    circles = TEMPLATES[code]
    result: set[tuple[str, str]] = set()
    for inner, ix, iy, ir in circles:
        for outer, ox, oy, oradius in circles:
            if inner == outer:
                continue
            if math.hypot(ix - ox, iy - oy) + ir < oradius:
                result.add((inner, outer))
    return frozenset(result)


RELATIONS = {code: _relations(code) for code in CODES}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_code = CODES[(int(seed) // 4) % len(CODES)]
    answer_index = int(seed) % 4
    distractors = [code for code in CODES if code != reference_code]
    rng.shuffle(distractors)
    candidate_codes = distractors[:3]
    candidate_codes.insert(answer_index, reference_code)
    variants = []
    for _ in range(5):
        variants.append(
            {
                "quarter_turns": rng.randrange(4),
                "reflect": bool(rng.randrange(2)),
                "scale": round(rng.uniform(0.90, 1.0), 4),
                "dx": rng.randint(-4, 4),
                "dy": rng.randint(-4, 4),
            }
        )
    return {
        "reference_code": reference_code,
        "candidate_codes": candidate_codes,
        "layout_variants": variants,
    }


def _transform(x: float, y: float, variant: dict) -> tuple[float, float]:
    if variant["reflect"]:
        x = -x
    for _ in range(int(variant["quarter_turns"]) % 4):
        x, y = -y, x
    scale = float(variant["scale"])
    return x * scale + float(variant["dx"]), y * scale + float(variant["dy"])


def _font(size: int) -> ImageFont.ImageFont:
    return ImageFont.load_default(size=size)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    code: str,
    variant: dict,
    *,
    scale_factor: int,
) -> None:
    cx, cy = center
    for color, x, y, radius in TEMPLATES[code]:
        tx, ty = _transform(x, y, variant)
        rr = radius * float(variant["scale"])
        box = (
            int((cx + tx - rr) * scale_factor),
            int((cy + ty - rr) * scale_factor),
            int((cx + tx + rr) * scale_factor),
            int((cy + ty + rr) * scale_factor),
        )
        draw.ellipse(box, outline=PALETTE[color], width=6 * scale_factor)


def render(scene: dict) -> Image.Image:
    # Access every declared scene field in the visible renderer.
    reference_code = scene["reference_code"]
    candidate_codes = list(scene["candidate_codes"])
    variants = list(scene["layout_variants"])
    if reference_code not in TEMPLATES or len(candidate_codes) != 4 or len(variants) != 5:
        raise ValueError("invalid scene")

    s = 2
    image = Image.new("RGB", (WIDTH * s, HEIGHT * s), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    title_font = _font(25 * s)
    label_font = _font(28 * s)
    small_font = _font(18 * s)
    draw.text((WIDTH * s // 2, 20 * s), "MATCH THE CONTAINMENT HIERARCHY", fill=(30, 35, 42), font=title_font, anchor="ma")
    draw.text((WIDTH * s // 2, 54 * s), "Same colors + same circle-within-circle relationships", fill=(70, 75, 82), font=small_font, anchor="ma")

    ref_center = (500, 185)
    draw.rounded_rectangle((392 * s, 84 * s, 608 * s, 292 * s), radius=14 * s, fill=(255, 255, 255), outline=(185, 188, 194), width=2 * s)
    draw.text((500 * s, 96 * s), "REFERENCE", fill=(35, 38, 44), font=small_font, anchor="ma")
    _draw_panel(draw, ref_center, reference_code, variants[0], scale_factor=s)

    centers = ((130, 445), (375, 445), (625, 445), (870, 445))
    for index, (center, code) in enumerate(zip(centers, candidate_codes)):
        cx, _ = center
        draw.rounded_rectangle(((cx - 108) * s, 334 * s, (cx + 108) * s, 567 * s), radius=14 * s, fill=(255, 255, 255), outline=(185, 188, 194), width=2 * s)
        _draw_panel(draw, center, code, variants[index + 1], scale_factor=s)
        draw.text((cx * s, 576 * s), LABELS[index], fill=(25, 28, 34), font=label_font, anchor="ma")

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_code"]
    candidates = scene["candidate_codes"]
    matches = [index for index, code in enumerate(candidates) if RELATIONS[code] == RELATIONS[reference]]
    if len(matches) != 1:
        raise ValueError("scene has no unique containment match")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    reference = RELATIONS[scene["reference_code"]]
    distances = sorted(len(reference.symmetric_difference(RELATIONS[code])) for code in scene["candidate_codes"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0
    except (KeyError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Pixel-identical latent aliases are intentionally not part of this representation.
    return []
