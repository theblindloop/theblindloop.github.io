"""Independent image-only inverse for the colored containment hierarchy."""

from __future__ import annotations

import hashlib
import math

import numpy as np
from PIL import Image

_LABELS = ("A", "B", "C", "D")
_COLORS = {
    "red": (220, 55, 65),
    "green": (35, 155, 95),
    "blue": (50, 105, 220),
    "purple": (150, 70, 185),
}
_CACHE: dict[bytes, str] = {}


def _components(mask: np.ndarray) -> list[tuple[float, float, float, int]]:
    height, width = mask.shape
    remaining = set((int(y) * width + int(x)) for y, x in np.argwhere(mask))
    found = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            xs.append(x)
            ys.append(y)
            for ny in range(max(0, y - 1), min(height, y + 2)):
                base = ny * width
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    neighbor = base + nx
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        if len(xs) >= 180:
            xmin, xmax = min(xs), max(xs)
            ymin, ymax = min(ys), max(ys)
            cx = (xmin + xmax) / 2.0
            cy = (ymin + ymax) / 2.0
            radius = ((xmax - xmin) + (ymax - ymin)) / 4.0
            if 12.0 <= radius <= 90.0:
                found.append((cx, cy, radius, len(xs)))
    return found


def _measure_circles(image: Image.Image) -> list[dict]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    circles: list[dict] = []
    for name, rgb in _COLORS.items():
        delta = arr - np.array(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) < 52 * 52
        comps = _components(mask)
        if len(comps) != 5:
            raise ValueError(f"expected five {name} circle outlines, found {len(comps)}")
        for cx, cy, radius, area in comps:
            circles.append({"color": name, "cx": cx, "cy": cy, "r": radius, "area": area})
    return circles


def _groups(circles: list[dict]) -> tuple[list[dict], list[list[dict]]]:
    ordered_y = sorted(circles, key=lambda circle: circle["cy"])
    gaps = [ordered_y[i + 1]["cy"] - ordered_y[i]["cy"] for i in range(len(ordered_y) - 1)]
    split = max(range(len(gaps)), key=gaps.__getitem__) + 1
    top, bottom = ordered_y[:split], ordered_y[split:]
    if len(top) != 4 or len(bottom) != 16:
        raise ValueError("could not separate reference and candidate rows")
    bottom.sort(key=lambda circle: circle["cx"])
    candidates = [bottom[index:index + 4] for index in range(0, 16, 4)]
    if any({circle["color"] for circle in group} != set(_COLORS) for group in [top, *candidates]):
        raise ValueError("panel color inventory is incomplete")
    return top, candidates


def _relation(group: list[dict]) -> frozenset[tuple[str, str]]:
    result: set[tuple[str, str]] = set()
    for inner in group:
        for outer in group:
            if inner["color"] == outer["color"]:
                continue
            distance = math.hypot(inner["cx"] - outer["cx"], inner["cy"] - outer["cy"])
            # The 4 px inset is a public raster safety allowance, far below the
            # renderer's minimum contained-circle clearance.
            if distance + inner["r"] < outer["r"] - 4.0:
                result.add((inner["color"], outer["color"]))
    return frozenset(result)


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    key = hashlib.sha256(rgb.tobytes()).digest()
    if key in _CACHE:
        return _CACHE[key]
    reference_group, candidate_groups = _groups(_measure_circles(rgb))
    reference = _relation(reference_group)
    distances = [len(reference.symmetric_difference(_relation(group))) for group in candidate_groups]
    best = min(distances)
    winners = [index for index, value in enumerate(distances) if value == best]
    if best != 0 or len(winners) != 1:
        raise ValueError("image has no unique exact containment-hierarchy match")
    decision = _LABELS[winners[0]]
    if len(_CACHE) > 512:
        _CACHE.clear()
    _CACHE[key] = decision
    return decision
