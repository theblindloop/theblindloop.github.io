"""Deterministic sampler and Pillow renderer for illusory containment."""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw


QUESTION = (
    "Among the six black disks, four right-angle white cutouts face one another and "
    "their straight edges imply one closed quadrilateral; the other two do not complete "
    "it. Which colored dot lies inside the implied quadrilateral? Answer red or blue."
)

CORNER_CUTOUT = {"tl": "se", "tr": "sw", "br": "nw", "bl": "ne"}


def _valid_quads(inducers: list[dict], tolerance: float = 3.0) -> list[tuple[dict, ...]]:
    by_cut = {name: [item for item in inducers if item["cutout"] == name] for name in ("se", "sw", "nw", "ne")}
    quads = []
    for tl, tr, br, bl in itertools.product(by_cut["se"], by_cut["sw"], by_cut["nw"], by_cut["ne"]):
        if len({id(tl), id(tr), id(br), id(bl)}) < 4:
            continue
        x0 = (tl["center"][0] + bl["center"][0]) / 2.0
        x1 = (tr["center"][0] + br["center"][0]) / 2.0
        y0 = (tl["center"][1] + tr["center"][1]) / 2.0
        y1 = (bl["center"][1] + br["center"][1]) / 2.0
        aligned = max(
            abs(tl["center"][0] - bl["center"][0]),
            abs(tr["center"][0] - br["center"][0]),
            abs(tl["center"][1] - tr["center"][1]),
            abs(bl["center"][1] - br["center"][1]),
        ) <= tolerance
        ordered = x1 - x0 >= 120 and y1 - y0 >= 120
        if aligned and ordered:
            quads.append((tl, tr, br, bl))
    return quads


def _bounds(inducers: list[dict]) -> tuple[float, float, float, float]:
    quads = _valid_quads(inducers)
    if len(quads) != 1:
        raise ValueError(f"expected one implied quadrilateral, found {len(quads)}")
    tl, tr, br, bl = quads[0]
    return (
        (tl["center"][0] + bl["center"][0]) / 2.0,
        (tl["center"][1] + tr["center"][1]) / 2.0,
        (tr["center"][0] + br["center"][0]) / 2.0,
        (bl["center"][1] + br["center"][1]) / 2.0,
    )


def _score(point: list[float], bounds: tuple[float, float, float, float]) -> float:
    x, y = point
    left, top, right, bottom = bounds
    return min(x - left, right - x, y - top, bottom - y)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    left = rng.randint(145, 180)
    top = rng.randint(140, 175)
    right = left + rng.randint(165, 205)
    bottom = top + rng.randint(160, 200)
    radius = rng.randint(29, 33)

    inducers = [
        {"center": [left, top], "radius": radius, "cutout": "se"},
        {"center": [right, top], "radius": radius, "cutout": "sw"},
        {"center": [right, bottom], "radius": radius, "cutout": "nw"},
        {"center": [left, bottom], "radius": radius, "cutout": "ne"},
    ]
    # The two extra inducers have the same ink as the four useful ones, but their
    # centers cannot align into a second closed quadrilateral.
    inducers.extend([
        {"center": [rng.randint(58, 82), rng.randint(62, 92)], "radius": radius, "cutout": rng.choice(("se", "sw", "nw", "ne"))},
        {"center": [rng.randint(430, 454), rng.randint(420, 450)], "radius": radius, "cutout": rng.choice(("se", "sw", "nw", "ne"))},
    ])
    rng.shuffle(inducers)

    outside_side = int(seed) % 4
    inside_side = (int(seed) // 4) % 4
    mid_x = rng.uniform(left + 0.38 * (right - left), left + 0.62 * (right - left))
    mid_y = rng.uniform(top + 0.38 * (bottom - top), top + 0.62 * (bottom - top))
    inside_gap = rng.randint(24, 39)
    outside_gap = rng.randint(22, 37)
    inside_positions = (
        [left + inside_gap, mid_y], [mid_x, top + inside_gap],
        [right - inside_gap, mid_y], [mid_x, bottom - inside_gap],
    )
    outside_positions = (
        [left - outside_gap, mid_y], [mid_x, top - outside_gap],
        [right + outside_gap, mid_y], [mid_x, bottom + outside_gap],
    )
    inside_color = "red" if int(seed) % 2 == 0 else "blue"
    outside_color = "blue" if inside_color == "red" else "red"
    probes = [
        {"color": inside_color, "center": [round(value, 3) for value in inside_positions[inside_side]], "radius": 10},
        {"color": outside_color, "center": [round(value, 3) for value in outside_positions[outside_side]], "radius": 10},
    ]
    rng.shuffle(probes)
    return {
        "inducers": inducers,
        "probes": probes,
        "palette": {
            "background": [rng.randint(246, 252)] * 3,
            "inducer": [rng.randint(20, 42)] * 3,
            "red": [220 + rng.randint(0, 18), 49 + rng.randint(0, 16), 52 + rng.randint(0, 15)],
            "blue": [45 + rng.randint(0, 16), 92 + rng.randint(0, 18), 202 + rng.randint(0, 28)],
        },
        "canvas": [512, 512],
    }


def analytic_gold(scene: dict) -> str:
    bounds = _bounds(scene["inducers"])
    scores = {probe["color"]: _score(probe["center"], bounds) for probe in scene["probes"]}
    inside = [color for color, score in scores.items() if score > 0]
    if len(inside) != 1 or set(scores) != {"red", "blue"}:
        raise ValueError("scene must have exactly one colored probe inside")
    return inside[0]


def margin(scene: dict) -> float:
    bounds = _bounds(scene["inducers"])
    scores = sorted((_score(probe["center"], bounds) for probe in scene["probes"]), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    try:
        bounds = _bounds(scene["inducers"])
        scores = [_score(probe["center"], bounds) for probe in scene["probes"]]
        return len([score for score in scores if score > 0]) != 1 or min(abs(score) for score in scores) < 12 or margin(scene) < 24
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["inducers"] = list(reversed(twin["inducers"]))
    twin["probes"] = list(reversed(twin["probes"]))
    return [("reverse_nonoverlapping_draw_order", twin)]


def render(scene: dict) -> Image.Image:
    width, height = (int(value) for value in scene["canvas"])
    palette = scene["palette"]
    background = tuple(palette["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for inducer in scene["inducers"]:
        cx, cy = (round(value) for value in inducer["center"])
        radius = int(inducer["radius"])
        draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=tuple(palette["inducer"]))
        dx = 1 if "e" in inducer["cutout"] else -1
        dy = 1 if "s" in inducer["cutout"] else -1
        x1 = cx + dx * (radius + 2)
        y1 = cy + dy * (radius + 2)
        draw.rectangle([min(cx, x1), min(cy, y1), max(cx, x1), max(cy, y1)], fill=background)
    for probe in scene["probes"]:
        cx, cy = (round(value) for value in probe["center"])
        radius = int(probe["radius"])
        draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=tuple(palette[probe["color"]]))
    return image
