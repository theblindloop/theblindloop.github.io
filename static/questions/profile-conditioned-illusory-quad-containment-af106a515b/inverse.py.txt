"""Independent pixel-only inverse for the implied-quadrilateral relation."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        group = []
        while queue:
            yy, xx = queue.popleft()
            group.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _recover_inducer(group: list[tuple[int, int]]) -> dict:
    ys = np.array([point[0] for point in group], dtype=float)
    xs = np.array([point[1] for point in group], dtype=float)
    cx = (float(xs.min()) + float(xs.max())) / 2.0
    cy = (float(ys.min()) + float(ys.max())) / 2.0
    counts = {
        "nw": int(np.sum((xs < cx) & (ys < cy))),
        "ne": int(np.sum((xs > cx) & (ys < cy))),
        "se": int(np.sum((xs > cx) & (ys > cy))),
        "sw": int(np.sum((xs < cx) & (ys > cy))),
    }
    ordered = sorted(counts.items(), key=lambda item: item[1])
    if ordered[0][1] > 0.28 * ordered[-1][1] or ordered[1][1] < 0.65 * ordered[-1][1]:
        raise ValueError("black disk lacks one clear right-angle cutout")
    return {"center": [cx, cy], "cutout": ordered[0][0]}


def _find_bounds(inducers: list[dict]) -> tuple[float, float, float, float]:
    by_cut = {name: [item for item in inducers if item["cutout"] == name] for name in ("se", "sw", "nw", "ne")}
    quads = []
    for tl, tr, br, bl in itertools.product(by_cut["se"], by_cut["sw"], by_cut["nw"], by_cut["ne"]):
        if len({id(tl), id(tr), id(br), id(bl)}) < 4:
            continue
        x0 = (tl["center"][0] + bl["center"][0]) / 2.0
        x1 = (tr["center"][0] + br["center"][0]) / 2.0
        y0 = (tl["center"][1] + tr["center"][1]) / 2.0
        y1 = (bl["center"][1] + br["center"][1]) / 2.0
        aligned = max(
            abs(tl["center"][0] - bl["center"][0]), abs(tr["center"][0] - br["center"][0]),
            abs(tl["center"][1] - tr["center"][1]), abs(bl["center"][1] - br["center"][1]),
        ) <= 5.0
        if aligned and x1 - x0 >= 115 and y1 - y0 >= 115:
            quads.append((x0, y0, x1, y1))
    if len(quads) != 1:
        raise ValueError(f"expected one completed quadrilateral, found {len(quads)}")
    return quads[0]


def _centroid(groups: list[list[tuple[int, int]]], label: str) -> tuple[float, float]:
    if len(groups) != 1:
        raise ValueError(f"expected one {label} probe, found {len(groups)}")
    group = groups[0]
    return (sum(x for y, x in group) / len(group), sum(y for y, x in group) / len(group))


def _score(point: tuple[float, float], bounds: tuple[float, float, float, float]) -> float:
    x, y = point
    left, top, right, bottom = bounds
    return min(x - left, right - x, y - top, bottom - y)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = np.max(array, axis=2) < 75
    red = (array[:, :, 0] > 170) & (array[:, :, 0] > array[:, :, 1] + 90) & (array[:, :, 0] > array[:, :, 2] + 80)
    blue = (array[:, :, 2] > 150) & (array[:, :, 2] > array[:, :, 0] + 90) & (array[:, :, 2] > array[:, :, 1] + 80)

    black_groups = [group for group in _components(black) if 1500 <= len(group) <= 3000]
    if len(black_groups) != 6:
        raise ValueError(f"expected six cutout disks, found {len(black_groups)}")
    inducers = [_recover_inducer(group) for group in black_groups]
    bounds = _find_bounds(inducers)
    probes = {
        "red": _centroid([group for group in _components(red) if 220 <= len(group) <= 500], "red"),
        "blue": _centroid([group for group in _components(blue) if 220 <= len(group) <= 500], "blue"),
    }
    scores = {color: _score(point, bounds) for color, point in probes.items()}
    inside = [color for color, score in scores.items() if score > 0]
    if len(inside) != 1 or min(abs(score) for score in scores.values()) < 8:
        raise ValueError("pixel containment evidence is ambiguous")
    return inside[0]
