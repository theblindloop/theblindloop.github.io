"""Independent PNG-only inverse for the ternary maximin task."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "blue": (35, 103, 205),
    "orange": (224, 112, 30),
    "green": (24, 151, 91),
}
_FRAME = (72, 82, 94)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    output: list[np.ndarray] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append(np.asarray(pixels, dtype=float))
    return output


def _dot_center(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    components = [component for component in _components(mask) if len(component) > 300]
    if len(components) != 1 or not 450 <= len(components[0]) <= 650:
        raise ValueError("expected one intact colored sample dot")
    ys = components[0][:, 0]
    xs = components[0][:, 1]
    if max(xs) - min(xs) < 24 or max(ys) - min(ys) < 24:
        raise ValueError("colored sample is too small")
    return float(xs.mean()), float(ys.mean())


def _fit_frame(arr: np.ndarray) -> tuple[tuple[float, float, float], ...]:
    mask = np.all(arr == np.asarray(_FRAME, dtype=np.uint8), axis=2)
    components = [component for component in _components(mask) if len(component) > 2500]
    if len(components) != 1:
        raise ValueError("triangular frame is missing or broken")
    pixels = components[0]
    ys = pixels[:, 0]
    xs = pixels[:, 1]
    min_y, max_y = float(ys.min()), float(ys.max())
    center_x = float(np.median(xs[ys < min_y + 18]))
    if not (360 <= max_y - min_y <= 395 and 420 <= xs.max() - xs.min() <= 450):
        raise ValueError("frame extent is implausible")

    # Each fitted line is ax + by + c = 0. Separate thick edge pixels by position.
    base = pixels[ys > max_y - 10]
    left = pixels[(xs < center_x - 8) & (ys < max_y - 12)]
    right = pixels[(xs > center_x + 8) & (ys < max_y - 12)]
    if min(len(base), len(left), len(right)) < 500:
        raise ValueError("a frame side is unresolved")

    base_y = float(base[:, 0].mean())
    lines: list[tuple[float, float, float]] = [(0.0, 1.0, -base_y)]
    for edge in (left, right):
        x = edge[:, 1]
        y = edge[:, 0]
        slope, intercept = np.polyfit(x, y, 1)
        lines.append((float(slope), -1.0, float(intercept)))
    return tuple(lines)


def _distance(point: tuple[float, float], line: tuple[float, float, float]) -> float:
    x, y = point
    a, b, c = line
    return abs(a * x + b * y + c) / math.hypot(a, b)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (620, 700):
        raise ValueError("unexpected canvas dimensions")
    lines = _fit_frame(arr)
    scores = {
        color: min(_distance(_dot_center(arr, rgb), line) for line in lines)
        for color, rgb in _COLORS.items()
    }
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 20.0:
        raise ValueError("top maximin scores are unresolved")
    return ordered[0][0]
