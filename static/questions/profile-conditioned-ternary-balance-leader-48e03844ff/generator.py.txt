"""Deterministic Pillow renderer for a ternary maximin measurement task."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLORS = {
    "blue": (35, 103, 205),
    "orange": (224, 112, 30),
    "green": (24, 151, 91),
}
BACKGROUND = (247, 249, 251)
PANEL = (255, 255, 255)
FRAME = (72, 82, 94)
GRID = (211, 217, 224)
TEXT = (31, 37, 45)


def _point_from_weights(triangle: list[list[int]], weights: list[float]) -> list[int]:
    # weights correspond to the visible top, left, and right vertices.
    x = sum(float(w) * int(vertex[0]) for w, vertex in zip(weights, triangle, strict=True))
    y = sum(float(w) * int(vertex[1]) for w, vertex in zip(weights, triangle, strict=True))
    return [int(round(x)), int(round(y))]


def _distance(point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    return abs((bx - ax) * (ay - py) - (ax - px) * (by - ay)) / math.hypot(bx - ax, by - ay)


def _scores(scene: dict[str, Any]) -> dict[str, float]:
    vertices = [tuple(map(float, vertex)) for vertex in scene["triangle"]]
    sides = ((vertices[0], vertices[1]), (vertices[1], vertices[2]), (vertices[2], vertices[0]))
    return {
        str(item["color"]): min(_distance(tuple(map(float, item["center"])), a, b) for a, b in sides)
        for item in scene["points"]
    }


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    dx = rng.randint(-14, 14)
    dy = rng.randint(-8, 8)
    triangle = [[350 + dx, 142 + dy], [132 + dx, 520 + dy], [568 + dx, 520 + dy]]

    colors = list(COLORS)
    winner = colors[int(seed) % len(colors)]
    other_colors = [color for color in colors if color != winner]
    rng.shuffle(other_colors)
    roles = {winner: "high", other_colors[0]: "mid", other_colors[1]: "low"}
    templates = {
        "high": [0.34, 0.31, 0.35],
        "mid": [0.57, 0.17, 0.26],
        "low": [0.68, 0.09, 0.23],
    }
    points: list[dict[str, Any]] = []
    for color in colors:
        for _attempt in range(60):
            weights = templates[roles[color]].copy()
            rng.shuffle(weights)
            # Small zero-sum perturbation varies geometry without weakening separation.
            delta = rng.choice((-0.012, -0.006, 0.006, 0.012))
            weights[0] += delta
            weights[1] -= delta
            center = _point_from_weights(triangle, weights)
            if all(math.dist(center, item["center"]) >= 58.0 for item in points):
                points.append({"color": color, "center": center})
                break
        else:
            raise RuntimeError("could not place separated ternary samples")
    rng.shuffle(points)
    return {
        "width": 700,
        "height": 620,
        "triangle": triangle,
        "points": points,
        "point_radius": 13,
        "line_width": 5,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = _scores(scene)
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if len(ordered) != 3 or ordered[0][1] - ordered[1][1] < 1e-12:
        return "abstain"
    return ordered[0][0]


def margin(scene: dict[str, Any]) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    scores = _scores(scene)
    centers = [tuple(map(float, item["center"])) for item in scene["points"]]
    min_pair_gap = min(
        math.dist(centers[i], centers[j]) - 2 * int(scene["point_radius"])
        for i in range(3) for j in range(i + 1, 3)
    )
    return (
        analytic_gold(scene) == "abstain"
        or margin(scene) < 30.0
        or min(scores.values()) < 24.0
        or min_pair_gap < 18.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    transformed = copy.deepcopy(scene)
    transformed["points"] = list(reversed(transformed["points"]))
    return [("reverse_nonoverlapping_point_records", transformed)]


def _font(name: str, size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font("DejaVuSans-Bold.ttf", 19)
    note_font = _font("DejaVuSans.ttf", 13)
    label_font = _font("DejaVuSans-Bold.ttf", 14)
    draw.text((34, 20), "TERNARY COMPOSITION BALANCE", fill=TEXT, font=title_font)
    draw.text((34, 49), "Distance to the nearest side", fill=TEXT, font=note_font)
    draw.rounded_rectangle((75, 88, 625, 580), radius=12, fill=PANEL, outline=GRID, width=2)

    top, left, right = [[int(v) for v in vertex] for vertex in scene["triangle"]]
    # Pale parallels make the measurement structure chart-like without entering the oracle frame mask.
    for fraction in (0.25, 0.50, 0.75):
        y = int(round(top[1] + fraction * (left[1] - top[1])))
        xa = int(round(top[0] + fraction * (left[0] - top[0])))
        xb = int(round(top[0] + fraction * (right[0] - top[0])))
        draw.line((xa, y, xb, y), fill=GRID, width=1)
    width = int(scene["line_width"])
    draw.line((tuple(top), tuple(left), tuple(right), tuple(top)), fill=FRAME, width=width, joint="curve")

    draw.text((top[0] - 19, top[1] - 30), "A", fill=TEXT, font=label_font)
    draw.text((left[0] - 28, left[1] + 7), "B", fill=TEXT, font=label_font)
    draw.text((right[0] + 13, right[1] + 7), "C", fill=TEXT, font=label_font)
    radius = int(scene["point_radius"])
    for item in scene["points"]:
        x, y = map(int, item["center"])
        color = str(item["color"])
        draw.ellipse((x - radius - 2, y - radius - 2, x + radius + 2, y + radius + 2), fill=PANEL)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[color])
    return image
