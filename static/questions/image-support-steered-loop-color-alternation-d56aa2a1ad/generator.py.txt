"""Deterministic renderer and analytic arm for Alternating Loop Stations."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
GRID = 48
RING_RADIUS = 48
MARKER_RADIUS = 9
BACKGROUND = (248, 247, 243)
NAVY = (31, 55, 78)
ORANGE = (230, 126, 34)
PURPLE = (132, 74, 177)
GRAY = (218, 222, 226)
PALETTE = {"orange": ORANGE, "purple": PURPLE}
SLOTS = ((0, -RING_RADIUS), (RING_RADIUS, 0), (0, RING_RADIUS), (-RING_RADIUS, 0))


def _rotate(values: list[str], amount: int) -> list[str]:
    amount %= len(values)
    return values[amount:] + values[:amount]


def sample_scene(seed: int) -> dict:
    """Sample a balanced cyclic pattern with translated, nondiagnostic clutter."""
    rng = random.Random(int(seed))
    center = [
        (rng.choice((2, 3, 4, 5)) + 0.5) * GRID,
        (rng.choice((2, 3, 4, 5)) + 0.5) * GRID,
    ]
    if int(seed) % 2 == 0:
        base = ["orange", "purple", "orange", "purple"]
    else:
        base = ["orange", "orange", "purple", "purple"]
    station_colors = _rotate(base, rng.randrange(4))

    distractors = []
    forbidden_x0, forbidden_y0 = center[0] - 76, center[1] - 76
    forbidden_x1, forbidden_y1 = center[0] + 76, center[1] + 76
    attempts = 0
    while len(distractors) < 6 and attempts < 500:
        attempts += 1
        x = rng.randrange(22, WIDTH - 21)
        y = rng.randrange(22, HEIGHT - 21)
        if forbidden_x0 <= x <= forbidden_x1 and forbidden_y0 <= y <= forbidden_y1:
            continue
        if any((x - item["x"]) ** 2 + (y - item["y"]) ** 2 < 28**2 for item in distractors):
            continue
        kind = "dot" if len(distractors) % 2 == 0 else "dash"
        distractors.append({"kind": kind, "x": x, "y": y, "size": 4 + rng.randrange(3)})
    distractors.sort(key=lambda item: (item["y"], item["x"], item["kind"]))
    return {
        "center": [int(center[0]), int(center[1])],
        "station_colors": station_colors,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    """Render a small closed station trail on a roomy nuisance-filled canvas."""
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    cx, cy = scene["center"]

    for item in scene["distractors"]:
        x, y, size = item["x"] * scale, item["y"] * scale, item["size"] * scale
        if item["kind"] == "dot":
            draw.ellipse((x - size, y - size, x + size, y + size), fill=GRAY)
        else:
            draw.rounded_rectangle(
                (x - 2 * size, y - size // 2, x + 2 * size, y + size // 2),
                radius=max(1, size // 2),
                fill=GRAY,
            )

    box = (
        (cx - RING_RADIUS) * scale,
        (cy - RING_RADIUS) * scale,
        (cx + RING_RADIUS) * scale,
        (cy + RING_RADIUS) * scale,
    )
    draw.ellipse(box, outline=NAVY, width=6 * scale)
    for (dx, dy), color_name in zip(SLOTS, scene["station_colors"], strict=True):
        x, y = (cx + dx) * scale, (cy + dy) * scale
        radius = MARKER_RADIUS * scale
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=PALETTE[color_name],
            outline=NAVY,
            width=2 * scale,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    colors = scene["station_colors"]
    alternating = all(colors[index] != colors[(index + 1) % 4] for index in range(4))
    return "yes" if alternating else "no"


def margin(scene: dict) -> float:
    """Angular clearance to an order ambiguity, in degrees."""
    _ = scene["station_colors"]
    return 90.0


def is_quarantined(scene: dict) -> bool:
    colors = scene["station_colors"]
    return len(colors) != 4 or colors.count("orange") != 2 or colors.count("purple") != 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Distractor enumeration is a natural render-invariant latent alias."""
    twin = deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_distractor_enumeration", twin)]
