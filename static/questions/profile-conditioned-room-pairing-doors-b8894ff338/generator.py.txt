"""Deterministic room-map renderer and analytic perfect-matching arm."""

from __future__ import annotations

import copy
import itertools
import random
from functools import lru_cache

from PIL import Image, ImageDraw

SIZE = 512
ROWS = 2
COLS = 4
ROOM_COUNT = ROWS * COLS
DOOR_COUNT = 7
TARGET_DEGREES = (1, 1, 1, 1, 2, 2, 3, 3)
PALETTES = (
    {"background": [249, 248, 244], "wall": [42, 50, 60], "door": [234, 126, 28]},
    {"background": [247, 250, 246], "wall": [54, 57, 67], "door": [224, 104, 30]},
    {"background": [250, 247, 243], "wall": [37, 48, 58], "door": [240, 142, 36]},
    {"background": [246, 248, 250], "wall": [50, 55, 65], "door": [230, 116, 24]},
)


def _all_edges() -> tuple[tuple[int, int, int, int], ...]:
    edges: list[tuple[int, int, int, int]] = []
    for row in range(ROWS):
        for col in range(COLS):
            if col + 1 < COLS:
                edges.append((row, col, row, col + 1))
            if row + 1 < ROWS:
                edges.append((row, col, row + 1, col))
    return tuple(edges)


ALL_EDGES = _all_edges()


def _canonical_edge(edge: list[int] | tuple[int, ...]) -> tuple[int, int, int, int]:
    if len(edge) != 4:
        raise ValueError("a doorway edge needs two cell endpoints")
    first = (int(edge[0]), int(edge[1]))
    second = (int(edge[2]), int(edge[3]))
    if first > second:
        first, second = second, first
    value = (first[0], first[1], second[0], second[1])
    if value not in ALL_EDGES:
        raise ValueError(f"invalid room adjacency: {edge!r}")
    return value


def _edge_set(edges: list[list[int]] | tuple[tuple[int, ...], ...]) -> tuple[tuple[int, int, int, int], ...]:
    values = tuple(sorted({_canonical_edge(edge) for edge in edges}))
    if len(values) != len(edges):
        raise ValueError("duplicate doorway adjacency")
    return values


def _vertices() -> tuple[tuple[int, int], ...]:
    return tuple((row, col) for row in range(ROWS) for col in range(COLS))


VERTICES = _vertices()


def _adjacency(edges: list[list[int]] | tuple[tuple[int, ...], ...]) -> dict[tuple[int, int], set[tuple[int, int]]]:
    result = {vertex: set() for vertex in VERTICES}
    for edge in _edge_set(edges):
        left = (edge[0], edge[1])
        right = (edge[2], edge[3])
        result[left].add(right)
        result[right].add(left)
    return result


def _degree_signature(edges: list[list[int]] | tuple[tuple[int, ...], ...]) -> tuple[int, ...]:
    return tuple(sorted(len(neighbors) for neighbors in _adjacency(edges).values()))


def _component_sizes(edges: list[list[int]] | tuple[tuple[int, ...], ...]) -> tuple[int, ...]:
    adjacency = _adjacency(edges)
    unseen = set(VERTICES)
    sizes: list[int] = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        stack = [start]
        size = 0
        while stack:
            vertex = stack.pop()
            size += 1
            for neighbor in adjacency[vertex]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    stack.append(neighbor)
        sizes.append(size)
    return tuple(sorted(sizes))


def _has_perfect_matching(edges: list[list[int]] | tuple[tuple[int, ...], ...]) -> bool:
    adjacency = _adjacency(edges)

    @lru_cache(maxsize=None)
    def search(remaining: tuple[tuple[int, int], ...]) -> bool:
        if not remaining:
            return True
        active = set(remaining)
        vertex = min(active, key=lambda item: (len(adjacency[item] & active), item))
        for neighbor in sorted(adjacency[vertex] & active):
            reduced = tuple(sorted(active - {vertex, neighbor}))
            if search(reduced):
                return True
        return False

    return search(VERTICES)


def _valid_pools() -> dict[str, tuple[tuple[tuple[int, int, int, int], ...], ...]]:
    pools: dict[str, list[tuple[tuple[int, int, int, int], ...]]] = {"yes": [], "no": []}
    for chosen in itertools.combinations(ALL_EDGES, DOOR_COUNT):
        if _degree_signature(chosen) != TARGET_DEGREES:
            continue
        if _component_sizes(chosen) != (ROOM_COUNT,):
            continue
        pools["yes" if _has_perfect_matching(chosen) else "no"].append(chosen)
    if not pools["yes"] or not pools["no"]:
        raise RuntimeError("balanced perfect-matching graph pools are unavailable")
    return {key: tuple(values) for key, values in pools.items()}


GRAPH_POOLS = _valid_pools()


def _jittered_axes(rng: random.Random) -> tuple[list[int], list[int]]:
    dx = rng.randint(-8, 8)
    dy = rng.randint(-8, 8)
    xs = [54 + dx, 154 + dx + rng.randint(-4, 4), 256 + dx + rng.randint(-4, 4),
          358 + dx + rng.randint(-4, 4), 458 + dx]
    ys = [140 + dy, 256 + dy + rng.randint(-5, 5), 372 + dy]
    return xs, ys


def sample_scene(seed: int) -> dict:
    """Sample balanced yes/no trees with matched counts and degree sequence."""

    seed = int(seed)
    rng = random.Random(seed)
    target = "yes" if seed % 2 == 0 else "no"
    pool = GRAPH_POOLS[target]
    chosen = pool[rng.randrange(len(pool))]
    xs, ys = _jittered_axes(rng)
    return {
        "door_edges": [list(edge) for edge in chosen],
        "xs": xs,
        "ys": ys,
        "wall_width": rng.choice([14, 16]),
        "door_length": rng.choice([42, 46, 50]),
        "palette": copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
        "transform": (seed // 2) % 8,
    }


def _grid_edges() -> tuple[list[tuple[str, int, int]], list[tuple[str, int, int]]]:
    outer: list[tuple[str, int, int]] = []
    internal: list[tuple[str, int, int]] = []
    for row in range(ROWS + 1):
        for col in range(COLS):
            (outer if row in {0, ROWS} else internal).append(("h", row, col))
    for row in range(ROWS):
        for col in range(COLS + 1):
            (outer if col in {0, COLS} else internal).append(("v", row, col))
    return outer, internal


def _wall_box(edge: tuple[str, int, int], xs: list[int], ys: list[int], width: int) -> tuple[int, int, int, int]:
    kind, row, col = edge
    half = width // 2
    if kind == "h":
        return xs[col] - half, ys[row] - half, xs[col + 1] + half, ys[row] + half
    return xs[col] - half, ys[row] - half, xs[col] + half, ys[row + 1] + half


def _door_wall(edge: tuple[int, int, int, int]) -> tuple[str, int, int]:
    row, col, next_row, next_col = edge
    if row == next_row:
        return "v", row, max(col, next_col)
    return "h", max(row, next_row), col


def _door_box(edge: tuple[int, int, int, int], xs: list[int], ys: list[int], width: int,
              length: int) -> tuple[int, int, int, int]:
    kind, row, col = _door_wall(edge)
    half_width = width // 2
    half_length = length // 2
    if kind == "h":
        center = (xs[col] + xs[col + 1]) // 2
        return center - half_length, ys[row] - half_width, center + half_length, ys[row] + half_width
    center = (ys[row] + ys[row + 1]) // 2
    return xs[col] - half_width, center - half_length, xs[col] + half_width, center + half_length


def _apply_transform(image: Image.Image, code: int) -> Image.Image:
    operations = (
        None,
        Image.Transpose.ROTATE_90,
        Image.Transpose.ROTATE_180,
        Image.Transpose.ROTATE_270,
        Image.Transpose.FLIP_LEFT_RIGHT,
        Image.Transpose.FLIP_TOP_BOTTOM,
        Image.Transpose.TRANSPOSE,
        Image.Transpose.TRANSVERSE,
    )
    operation = operations[int(code) % len(operations)]
    return image if operation is None else image.transpose(operation)


def render(scene: dict) -> Image.Image:
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    width = int(scene["wall_width"])
    length = int(scene["door_length"])
    palette = scene["palette"]
    if len(xs) != COLS + 1 or len(ys) != ROWS + 1:
        raise ValueError("grid axes have invalid lengths")
    doors = _edge_set(scene["door_edges"])
    image = Image.new("RGB", (SIZE, SIZE), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    outer, internal = _grid_edges()
    for edge in outer + internal:
        draw.rectangle(_wall_box(edge, xs, ys, width), fill=tuple(palette["wall"]))
    for edge in doors:
        draw.rectangle(_door_box(edge, xs, ys, width, length), fill=tuple(palette["door"]))
    return _apply_transform(image, int(scene["transform"]))


def analytic_gold(scene: dict) -> str:
    return "yes" if _has_perfect_matching(scene["door_edges"]) else "no"


@lru_cache(maxsize=128)
def _edit_margin(edges: tuple[tuple[int, int, int, int], ...]) -> int:
    original = _has_perfect_matching(edges)
    base = set(edges)
    for distance in range(1, len(ALL_EDGES) + 1):
        for alternate_mask in range(1 << len(ALL_EDGES)):
            alternate = {edge for index, edge in enumerate(ALL_EDGES) if alternate_mask & (1 << index)}
            if len(base ^ alternate) != distance:
                continue
            if _has_perfect_matching(tuple(sorted(alternate))) != original:
                return distance
    raise RuntimeError("no opposite matching decision exists")


def margin(scene: dict) -> float:
    """Minimum number of visible doorway edits needed to change the answer."""

    return float(_edit_margin(_edge_set(scene["door_edges"])))


def raster_clearance(scene: dict) -> float:
    xs = [int(value) for value in scene["xs"]]
    ys = [int(value) for value in scene["ys"]]
    width = float(scene["wall_width"])
    length = float(scene["door_length"])
    shortest = min(
        min(right - left for left, right in zip(xs, xs[1:])),
        min(right - left for left, right in zip(ys, ys[1:])),
    )
    endpoint_clearance = (shortest - length) / 2.0 - width / 2.0
    return float(min(width / 2.0, endpoint_clearance))


def is_quarantined(scene: dict) -> bool:
    try:
        edges = _edge_set(scene["door_edges"])
        return (
            len(edges) != DOOR_COUNT
            or _degree_signature(edges) != TARGET_DEGREES
            or _component_sizes(edges) != (ROOM_COUNT,)
            or raster_clearance(scene) < 6.0
            or margin(scene) < 1.0
        )
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["door_edges"] = [
        [edge[2], edge[3], edge[0], edge[1]] for edge in reversed(twin["door_edges"])
    ]
    return [("undirected_serialization", twin)]


def graph_stats(scene: dict) -> dict:
    edges = _edge_set(scene["door_edges"])
    return {
        "rooms": ROOM_COUNT,
        "doors": len(edges),
        "degree_signature": list(_degree_signature(edges)),
        "component_sizes": list(_component_sizes(edges)),
        "cycle_rank": len(edges) - ROOM_COUNT + len(_component_sizes(edges)),
    }
