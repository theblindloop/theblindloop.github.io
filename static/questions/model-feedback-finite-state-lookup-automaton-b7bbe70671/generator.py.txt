"""Renderer and analytic arm for an input-driven finite-state table process."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1600
HEIGHT = 720

BACKGROUND = (245, 248, 251)
PANEL = (255, 255, 255)
FRAME = (102, 117, 133)
INK = (35, 45, 60)
MUTED = (132, 145, 158)
GRID = (213, 221, 229)
STATE_COLORS = (
    (229, 157, 30),   # amber
    (27, 164, 201),   # cyan
    (128, 75, 177),   # violet
    (101, 171, 73),   # lime
)
STATE_NAMES = ("amber", "cyan", "violet", "lime")
INPUT_COLORS = (
    (208, 77, 66),    # red
    (47, 113, 197),   # blue
    (31, 145, 132),   # teal
)
INPUT_NAMES = ("red", "blue", "teal")

RULE_X = (390, 550, 710)
RULE_Y = (250, 350, 450, 550)
START_CENTER = (930, 370)
INPUT_Y = 370
INPUT_X = tuple(1100 + 51 * index for index in range(9))

STATE_LABELS = dict(zip(range(4), STATE_NAMES))


def _apply(state: int, rules: list[list[int]], event: int) -> int:
    return int(rules[int(state)][int(event)])


def _final_state(scene: dict) -> int:
    state = int(scene["start"])
    rules = scene["rules"]
    for event in scene["events"]:
        state = _apply(state, rules, int(event))
    return state


def sample_scene(seed: int) -> dict:
    """Sample a visibly varied table and event stream with balanced final classes."""
    rng = random.Random(int(seed))
    desired = int(seed) % 4
    for _ in range(20_000):
        rules = [[rng.randrange(4) for _ in range(3)] for _ in range(4)]
        flat = {value for row in rules for value in row}
        if flat != set(range(4)):
            continue
        if any(len(set(row)) < 2 for row in rules):
            continue
        if any(len({rules[row][column] for row in range(4)}) < 2 for column in range(3)):
            continue
        events = [rng.randrange(3) for _ in range(9)]
        if set(events) != set(range(3)):
            continue
        start = rng.randrange(4)
        scene = {"start": start, "rules": rules, "events": events}
        if _final_state(scene) == desired:
            return scene
    raise RuntimeError("could not sample a balanced finite-state scene")


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _state_marker(draw: ImageDraw.ImageDraw, center: tuple[int, int], state: int, radius: int = 27) -> None:
    cx, cy = center
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=STATE_COLORS[int(state)], outline=INK, width=3)


def _panel(draw: ImageDraw.ImageDraw, rect: tuple[int, int, int, int], title: str, font, subtitle: str | None = None) -> None:
    draw.rounded_rectangle(rect, radius=18, fill=PANEL, outline=FRAME, width=3)
    x0, y0, _, _ = rect
    draw.text((x0 + 22, y0 + 18), title, font=font, fill=INK)
    if subtitle:
        draw.text((x0 + 22, y0 + 47), subtitle, font=_font(14), fill=MUTED)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(24)
    label_font = _font(17)
    small_font = _font(14)
    badge_font = _font(16)

    draw.rectangle((0, 0, WIDTH, 67), fill=(231, 237, 242))
    draw.text((30, 21), "STATE TRANSITION WORKBENCH", font=title_font, fill=INK)
    draw.text((1280, 26), "read the table, then carry the state", font=small_font, fill=MUTED)

    _panel(
        draw,
        (28, 90, 795, 650),
        "RULEBOOK",
        title_font,
        "row = current state    column = input    disk = next state",
    )
    draw.text((69, 168), "CURRENT", font=small_font, fill=MUTED)
    draw.text((69, 188), "STATE", font=small_font, fill=MUTED)
    draw.text((308, 111), "NEXT INPUT", font=small_font, fill=MUTED)

    for column, x in enumerate(RULE_X):
        draw.rounded_rectangle((x - 39, 145, x + 39, 189), radius=9, fill=INPUT_COLORS[column], outline=INK, width=2)
        _center_text(draw, (x, 167), chr(ord("A") + column), badge_font, PANEL)

    for row, y in enumerate(RULE_Y):
        _state_marker(draw, (145, y), row, radius=25)
        draw.text((188, y - 10), STATE_NAMES[row].upper(), font=label_font, fill=INK)
        for column, x in enumerate(RULE_X):
            draw.rounded_rectangle((x - 58, y - 39, x + 58, y + 39), radius=11, fill=(249, 251, 253), outline=GRID, width=2)
            _state_marker(draw, (x, y), int(scene["rules"][row][column]), radius=25)

    draw.line((277, 207, 277, 596), fill=GRID, width=2)
    draw.line((310, 207, 763, 207), fill=GRID, width=2)
    draw.text((309, 612), "A / B / C are the three input columns", font=small_font, fill=MUTED)

    _panel(draw, (840, 180, 1020, 560), "START", title_font, "initial state")
    _state_marker(draw, START_CENTER, int(scene["start"]), radius=54)
    _center_text(draw, (930, 468), STATE_NAMES[int(scene["start"])].upper(), label_font, INK)

    _panel(draw, (1060, 90, 1572, 650), "INPUT STREAM", title_font, "read left to right")
    draw.text((1080, 187), "A", font=small_font, fill=INPUT_COLORS[0])
    draw.text((1110, 187), "red", font=small_font, fill=MUTED)
    draw.text((1165, 187), "B", font=small_font, fill=INPUT_COLORS[1])
    draw.text((1195, 187), "blue", font=small_font, fill=MUTED)
    draw.text((1255, 187), "C", font=small_font, fill=INPUT_COLORS[2])
    draw.text((1285, 187), "teal", font=small_font, fill=MUTED)
    draw.line((1080, 218, 1550, 218), fill=GRID, width=2)

    for index, event in enumerate(scene["events"]):
        x = INPUT_X[index]
        draw.rounded_rectangle((x - 21, INPUT_Y - 46, x + 21, INPUT_Y + 46), radius=9, fill=INPUT_COLORS[int(event)], outline=INK, width=2)
        _center_text(draw, (x, INPUT_Y + 15), chr(ord("A") + int(event)), badge_font, PANEL)
        _center_text(draw, (x, INPUT_Y - 69), str(index + 1), small_font, FRAME)

    draw.line((1087, 487, 1540, 487), fill=GRID, width=2)
    draw.text((1080, 508), "each badge selects one column", font=small_font, fill=MUTED)
    draw.text((1080, 536), "the cell's disk becomes the next row", font=small_font, fill=MUTED)

    return image


def analytic_gold(scene: dict) -> str:
    return STATE_LABELS[_final_state(scene)]


def margin(scene: dict) -> float:
    # Fixed spatial clearance of the decision-bearing disks from cell borders.
    return 28.0


def is_quarantined(scene: dict) -> bool:
    if not isinstance(scene, dict):
        return True
    if len(scene.get("rules", [])) != 4 or len(scene.get("events", [])) != 9:
        return True
    if any(len(row) != 3 for row in scene.get("rules", [])):
        return True
    if not 0 <= int(scene.get("start", -1)) < 4:
        return True
    return any(int(value) not in range(4) for row in scene["rules"] for value in row) or any(
        int(value) not in range(3) for value in scene["events"]
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
