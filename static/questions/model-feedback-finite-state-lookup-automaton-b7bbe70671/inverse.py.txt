"""Independent pixel-only inverse for the finite-state lookup page."""

from __future__ import annotations

from PIL import Image

WIDTH = 1600
HEIGHT = 720
RULE_X = (390, 550, 710)
RULE_Y = (250, 350, 450, 550)
START_CENTER = (930, 370)
INPUT_Y = 370
INPUT_X = tuple(1100 + 51 * index for index in range(9))
STATE_COLORS = (
    (229, 157, 30),
    (27, 164, 201),
    (128, 75, 177),
    (101, 171, 73),
)
INPUT_COLORS = (
    (208, 77, 66),
    (47, 113, 197),
    (31, 145, 132),
)
STATE_LABELS = ("amber", "cyan", "violet", "lime")


def _palette_index(pixel: tuple[int, int, int], palette: tuple[tuple[int, int, int], ...]) -> int:
    for index, color in enumerate(palette):
        if tuple(pixel) == color:
            return index
    raise ValueError(f"unreadable color {pixel}")


def decision_from_image(image: Image.Image) -> str:
    """Recover the table and input stream from fixed exact-color raster marks."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")

    start = _palette_index(rgb.getpixel(START_CENTER), STATE_COLORS)
    rules: list[list[int]] = []
    for y in RULE_Y:
        row = []
        for x in RULE_X:
            row.append(_palette_index(rgb.getpixel((x, y)), STATE_COLORS))
        rules.append(row)

    events = []
    for x in INPUT_X:
        # The badge letters sit low in each card; this offset is an untouched fill pixel.
        events.append(_palette_index(rgb.getpixel((x + 14, INPUT_Y - 20)), INPUT_COLORS))

    state = start
    for event in events:
        state = rules[state][event]
    return STATE_LABELS[state]
