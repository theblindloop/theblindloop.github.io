"""Deterministic scene sampler and raster renderer for the thermometer conflict world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

_BACKGROUND = (248, 249, 246)
_TUBE_PALETTES = ("#315f78", "#386f68", "#565b8b", "#3d6b86")
_BULB_PALETTES = ("#d83b46", "#c93649", "#e14c3f", "#cf4057")
_ACCENTS = ("#2aa198", "#7b61a8", "#2f80b7", "#558b55")

_DIGITS = {
    "0": (
        "11111",
        "10001",
        "10001",
        "10001",
        "10001",
        "10001",
        "11111",
    ),
    "1": (
        "00100",
        "01100",
        "00100",
        "00100",
        "00100",
        "00100",
        "11111",
    ),
}


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle through all label/end relations."""
    rng = random.Random(int(seed))
    bulb_end = "top" if (int(seed) // 2) % 2 == 0 else "bottom"
    answer_is_zero = int(seed) % 2 == 0
    zero_end = bulb_end if answer_is_zero else ("bottom" if bulb_end == "top" else "top")
    label_side = "left" if (int(seed) // 4) % 2 == 0 else "right"
    palette_index = (int(seed) // 8) % len(_TUBE_PALETTES)
    center_x = rng.randint(238, 274)
    top_y = rng.randint(112, 140)
    bottom_y = rng.randint(372, 400)
    tube_width = rng.choice((28, 30, 32))
    bulb_radius = rng.choice((25, 27, 29))
    marks = [
        {
            "x": 38 + rng.randint(0, 20),
            "y": 38 + rng.randint(0, 18),
            "radius": rng.choice((7, 8, 9)),
            "color": _ACCENTS[(palette_index + 1) % len(_ACCENTS)],
        },
        {
            "x": 454 + rng.randint(0, 20),
            "y": 446 + rng.randint(0, 18),
            "radius": rng.choice((7, 8, 9)),
            "color": _ACCENTS[(palette_index + 2) % len(_ACCENTS)],
        },
    ]
    if rng.randrange(2):
        marks.reverse()
    return {
        "width": 512,
        "height": 512,
        "center_x": center_x,
        "top_y": top_y,
        "bottom_y": bottom_y,
        "tube_width": tube_width,
        "bulb_radius": bulb_radius,
        "bulb_end": bulb_end,
        "zero_end": zero_end,
        "label_side": label_side,
        "tube_color": _TUBE_PALETTES[palette_index],
        "bulb_color": _BULB_PALETTES[(int(seed) // 32) % len(_BULB_PALETTES)],
        "ink_color": "#202530",
        "panel_color": "#edf0e9",
        "corner_marks": marks,
    }


def _scaled_box(box: tuple[int, int, int, int], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(value * scale) for value in box)


def _endpoint_y(scene: dict, end: str) -> int:
    return int(scene["top_y"] if end == "top" else scene["bottom_y"])


def _panel_box(scene: dict, end: str) -> tuple[int, int, int, int]:
    center_x = int(scene["center_x"])
    center_y = _endpoint_y(scene, end)
    if scene["label_side"] == "left":
        x0, x1 = center_x - 190, center_x - 62
    else:
        x0, x1 = center_x + 62, center_x + 190
    return (x0, center_y - 30, x1, center_y + 30)


def _draw_digit(
    draw: ImageDraw.ImageDraw,
    digit: str,
    x: int,
    y: int,
    block: int,
    scale: int,
    color: str,
) -> None:
    for row, pattern in enumerate(_DIGITS[digit]):
        for column, value in enumerate(pattern):
            if value == "1":
                left = (x + column * block) * scale
                top = (y + row * block) * scale
                right = (x + (column + 1) * block) * scale - 1
                bottom = (y + (row + 1) * block) * scale - 1
                draw.rectangle((left, top, right, bottom), fill=color)


def _draw_number(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: int,
    center_y: int,
    scale: int,
    color: str,
) -> None:
    block = 5
    glyph_width = 5 * block
    gap = 5
    total_width = len(text) * glyph_width + (len(text) - 1) * gap
    x = int(round(center_x - total_width / 2))
    y = int(round(center_y - (7 * block) / 2))
    for digit in text:
        _draw_digit(draw, digit, x, y, block, scale, color)
        x += glyph_width + gap


def render(scene: dict) -> Image.Image:
    """Render the scene to an antialiased RGB image."""
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    center_x = int(scene["center_x"])
    top_y = int(scene["top_y"])
    bottom_y = int(scene["bottom_y"])
    tube_width = int(scene["tube_width"])
    bulb_radius = int(scene["bulb_radius"])
    bulb_end = scene["bulb_end"]
    zero_end = scene["zero_end"]
    label_side = scene["label_side"]
    tube_color = scene["tube_color"]
    bulb_color = scene["bulb_color"]
    ink_color = scene["ink_color"]
    panel_color = scene["panel_color"]
    corner_marks = scene["corner_marks"]

    image = Image.new("RGB", (width * scale, height * scale), _BACKGROUND)
    draw = ImageDraw.Draw(image)

    for mark in corner_marks:
        x = int(mark["x"]) * scale
        y = int(mark["y"]) * scale
        radius = int(mark["radius"]) * scale
        draw.polygon(
            ((x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)),
            fill=mark["color"],
        )

    for end in ("top", "bottom"):
        panel = _panel_box(scene, end)
        draw.rounded_rectangle(
            _scaled_box(panel, scale),
            radius=10 * scale,
            fill=panel_color,
        )
        label = "0" if end == zero_end else "100"
        px0, py0, px1, py1 = panel
        _draw_number(draw, label, (px0 + px1) // 2, (py0 + py1) // 2, scale, ink_color)

    half = tube_width // 2
    tube_box = (center_x - half, top_y - 10, center_x + half, bottom_y + 10)
    draw.rounded_rectangle(
        _scaled_box(tube_box, scale),
        radius=half * scale,
        fill="#f4f5f0",
        outline=tube_color,
        width=5 * scale,
    )
    midpoint = (top_y + bottom_y) // 2
    bulb_y = top_y if bulb_end == "top" else bottom_y
    draw.line(
        ((center_x * scale, bulb_y * scale), (center_x * scale, midpoint * scale)),
        fill=bulb_color,
        width=11 * scale,
    )
    draw.ellipse(
        _scaled_box(
            (
                center_x - bulb_radius,
                bulb_y - bulb_radius,
                center_x + bulb_radius,
                bulb_y + bulb_radius,
            ),
            scale,
        ),
        fill=bulb_color,
        outline=tube_color,
        width=4 * scale,
    )

    tick_x0 = center_x + half + 8 if label_side == "left" else center_x - half - 8
    tick_x1 = tick_x0 + 18 if label_side == "left" else tick_x0 - 18
    for y in (top_y, bottom_y):
        draw.line(
            ((tick_x0 * scale, y * scale), (tick_x1 * scale, y * scale)),
            fill=tube_color,
            width=4 * scale,
        )

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Return the printed label occupying the same endpoint as the bulb."""
    return "0" if scene["bulb_end"] == scene["zero_end"] else "100"


def margin(scene: dict) -> float:
    """Runner-up label distance minus winning label distance in logical pixels."""
    bulb_y = float(_endpoint_y(scene, scene["bulb_end"]))
    bulb_x = float(scene["center_x"])
    panel0 = _panel_box(scene, scene["zero_end"])
    other_end = "bottom" if scene["zero_end"] == "top" else "top"
    panel100 = _panel_box(scene, other_end)
    centers = {
        "0": ((panel0[0] + panel0[2]) / 2.0, (panel0[1] + panel0[3]) / 2.0),
        "100": ((panel100[0] + panel100[2]) / 2.0, (panel100[1] + panel100[3]) / 2.0),
    }
    distances = {
        key: math.hypot(x - bulb_x, y - bulb_y) for key, (x, y) in centers.items()
    }
    ordered = sorted(distances.values())
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 80.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse disjoint decoration order, a real latent alias with identical pixels."""
    twin = deepcopy(scene)
    twin["corner_marks"] = list(reversed(twin["corner_marks"]))
    return [("reverse_disjoint_corner_mark_order", twin)]
