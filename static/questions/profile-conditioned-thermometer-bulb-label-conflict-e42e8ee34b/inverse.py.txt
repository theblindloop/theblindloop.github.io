"""Independent image-only inverse for bulb-to-printed-label binding.

This module intentionally imports no world implementation module. It finds the red
bulb by color and row width, finds connected dark numeral components, groups those
components into the two endpoint labels, distinguishes one component (0) from three
components (100), and compares measured centroid distances.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the requested binding is not recoverable with a safe margin."""


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[dict] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = np.fromiter((point[0] for point in points), dtype=np.float64)
        xs = np.fromiter((point[1] for point in points), dtype=np.float64)
        components.append(
            {
                "area": len(points),
                "cx": float(xs.mean()),
                "cy": float(ys.mean()),
                "width": int(xs.max() - xs.min() + 1),
                "height": int(ys.max() - ys.min() + 1),
            }
        )
    return components


def _measure(image: Image.Image) -> dict:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[2] != 3 or min(rgb.shape[:2]) < 64:
        raise OracleAbstention("unsupported or too-small raster")

    red = (rgb[:, :, 0] > 155) & (rgb[:, :, 0] > rgb[:, :, 1] * 1.65) & (
        rgb[:, :, 0] > rgb[:, :, 2] * 1.45
    )
    row_widths = red.sum(axis=1)
    maximum = int(row_widths.max(initial=0))
    if maximum < 30:
        raise OracleAbstention("red round bulb is absent")
    broad_rows = np.flatnonzero(row_widths >= maximum * 0.92)
    if broad_rows.size == 0:
        raise OracleAbstention("red bulb center is unresolved")
    bulb_y = float(broad_rows.mean())
    bulb_pixels = np.argwhere(red[broad_rows, :])
    if bulb_pixels.size == 0:
        raise OracleAbstention("red bulb pixels are absent")
    bulb_x = float(bulb_pixels[:, 1].mean())

    dark = (rgb[:, :, 0] < 74) & (rgb[:, :, 1] < 74) & (rgb[:, :, 2] < 74)
    digits = [
        component
        for component in _components(dark)
        if component["area"] >= 80
        and 4 <= component["width"] <= 32
        and 20 <= component["height"] <= 42
    ]
    if len(digits) != 4:
        raise OracleAbstention(f"expected four numeral components, found {len(digits)}")
    digits.sort(key=lambda component: (component["cy"], component["cx"]))
    gaps = [digits[index + 1]["cy"] - digits[index]["cy"] for index in range(3)]
    split = int(np.argmax(gaps)) + 1
    top_group, bottom_group = digits[:split], digits[split:]
    if sorted((len(top_group), len(bottom_group))) != [1, 3] or max(gaps) < 100:
        raise OracleAbstention("endpoint numeral groups are ambiguous")

    def center(group: list[dict]) -> tuple[float, float]:
        return (
            float(sum(component["cx"] for component in group) / len(group)),
            float(sum(component["cy"] for component in group) / len(group)),
        )

    groups = {
        "0" if len(top_group) == 1 else "100": center(top_group),
        "0" if len(bottom_group) == 1 else "100": center(bottom_group),
    }
    distances = {
        label: math.hypot(cx - bulb_x, cy - bulb_y) for label, (cx, cy) in groups.items()
    }
    ranked = sorted(distances, key=distances.get)
    measured_margin = float(distances[ranked[1]] - distances[ranked[0]])
    if measured_margin < 60.0:
        raise OracleAbstention("nearest-label distance gap is ambiguous")
    return {
        "decision": ranked[0],
        "margin": measured_margin,
        "bulb_center": (bulb_x, bulb_y),
        "label_centers": groups,
    }


def decision_from_image(image: Image.Image) -> str:
    """Return ``0`` or ``100`` using only raster evidence."""
    return str(_measure(image)["decision"])
