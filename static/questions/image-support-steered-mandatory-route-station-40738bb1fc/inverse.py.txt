"""Independent pixel-only oracle for terminal-relative station separation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = np.asarray([
    (29, 34, 40),
    (233, 103, 28),
    (43, 104, 196),
    (0, 164, 184),
    (246, 194, 48),
], dtype=np.int32)
COLOR_TOLERANCE_SQ = 42 * 42
DELETE_RADIUS = 16


def _masks(image: Image.Image) -> list[np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    return [distances[:, :, index] <= COLOR_TOLERANCE_SQ for index in range(len(PALETTE))]


def _centroid(mask: np.ndarray, name: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 300:
        raise ValueError(f"missing or damaged {name} station")
    y, x = np.rint(points.mean(axis=0)).astype(int)
    return int(y), int(x)


def _component(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    sy, sx = start
    if not (0 <= sy < height and 0 <= sx < width and mask[sy, sx]):
        raise ValueError("terminal center is not on the visible network")
    seen = np.zeros_like(mask, dtype=bool)
    seen[sy, sx] = True
    queue = deque([(sy, sx)])
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return seen


def _without_station(network: np.ndarray, center: tuple[int, int]) -> np.ndarray:
    y, x = center
    yy, xx = np.ogrid[:network.shape[0], :network.shape[1]]
    keep = (yy - y) ** 2 + (xx - x) ** 2 > DELETE_RADIUS ** 2
    return network & keep


def decision_from_image(image: Image.Image) -> str:
    black, orange_mask, blue_mask, cyan_mask, yellow_mask = _masks(image)
    orange = _centroid(orange_mask, "orange")
    blue = _centroid(blue_mask, "blue")
    cyan = _centroid(cyan_mask, "cyan")
    yellow = _centroid(yellow_mask, "yellow")
    network = black | orange_mask | blue_mask | cyan_mask | yellow_mask

    # The prompt declares one connected network. Enforcing that visible rule
    # makes damage to any interior branch an explicit abstention rather than an
    # accidental answer based on a surviving local cue.
    original = _component(network, orange)
    if int(original.sum()) != int(network.sum()) or not original[blue]:
        raise ValueError("visible tracks are not one connected network")

    separates = {}
    for name, center in (("cyan", cyan), ("yellow", yellow)):
        cut = _without_station(network, center)
        reached = _component(cut, orange)
        separates[name] = not bool(reached[blue])
    if sum(separates.values()) != 1:
        raise ValueError("both or neither colored stations separate the terminals")
    return "cyan" if separates["cyan"] else "yellow"
