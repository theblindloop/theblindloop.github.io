"""Deterministic rail-tree sampler and renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = (248, 247, 242)
RAIL = (29, 34, 40)
ORANGE = (233, 103, 28)
BLUE = (43, 104, 196)
CYAN = (0, 164, 184)
YELLOW = (246, 194, 48)
RAIL_WIDTH = 9
MARKER_RADIUS = 13
CANVAS = 416


def _transform(point: list[int], turn: int, reflect: bool, dx: int, dy: int) -> list[int]:
    x, y = point[0] - 208, point[1] - 208
    if reflect:
        x = -x
    for _ in range(turn % 4):
        x, y = -y, x
    return [x + 208 + dx, y + 208 + dy]


def _distance_to_vertices(point: list[int], paths: list[list[list[int]]]) -> float:
    distances = []
    for path in paths:
        distances.extend(math.dist(point, vertex) for vertex in (path[0], path[-1]))
        distances.extend(math.dist(point, vertex) for vertex in path[1:-1])
    return min(distances)


def _on_segment(point: list[int], a: list[int], b: list[int]) -> bool:
    px, py = point
    ax, ay = a
    bx, by = b
    cross = (px - ax) * (by - ay) - (py - ay) * (bx - ax)
    return cross == 0 and min(ax, bx) <= px <= max(ax, bx) and min(ay, by) <= py <= max(ay, by)


def _on_path(point: list[int], path: list[list[int]]) -> bool:
    return any(_on_segment(point, a, b) for a, b in zip(path, path[1:]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))

    # The canonical tree has one winding trunk and one bent side branch.  The
    # two station slots are both in straight interiors, far from endpoints and
    # the shared junction.  Eight rigid transforms counterbalance image axes.
    main = [[100, 165], [175, 165], [220, 210], [310, 210]]
    branch = [[175, 165], [175, 285]]
    main_slot = [265, 210]
    branch_slot = [175, 225]

    turn = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    dx = rng.randint(-9, 9)
    dy = rng.randint(-9, 9)
    main = [_transform(p, turn, reflect, dx, dy) for p in main]
    branch = [_transform(p, turn, reflect, dx, dy) for p in branch]
    main_slot = _transform(main_slot, turn, reflect, dx, dy)
    branch_slot = _transform(branch_slot, turn, reflect, dx, dy)

    endpoints = [list(main[0]), list(main[-1])]
    if rng.randrange(2):
        endpoints.reverse()
    cyan_is_mandatory = int(seed) % 2 == 0
    cyan = list(main_slot if cyan_is_mandatory else branch_slot)
    yellow = list(branch_slot if cyan_is_mandatory else main_slot)
    return {
        "main_path": main,
        "branch_path": branch,
        "orange": endpoints[0],
        "blue": endpoints[1],
        "cyan": cyan,
        "yellow": yellow,
        "width": CANVAS,
        "height": CANVAS,
    }


def _draw_path(draw: ImageDraw.ImageDraw, path: list[list[int]]) -> None:
    points = [tuple(point) for point in path]
    draw.line(points, fill=RAIL, width=RAIL_WIDTH, joint="curve")
    radius = RAIL_WIDTH // 2
    for x, y in points[1:-1]:
        draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=RAIL)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_path(draw, scene["main_path"])
    _draw_path(draw, scene["branch_path"])
    for center, color in (
        (scene["cyan"], CYAN),
        (scene["yellow"], YELLOW),
        (scene["orange"], ORANGE),
        (scene["blue"], BLUE),
    ):
        x, y = center
        r = MARKER_RADIUS
        draw.ellipse((x-r, y-r, x+r, y+r), fill=color)
        draw.ellipse((x-2, y-2, x+2, y+2), fill=RAIL)
    return image


def analytic_gold(scene: dict) -> str:
    main = scene["main_path"]
    branch = scene["branch_path"]
    cyan = scene["cyan"]
    yellow = scene["yellow"]
    cyan_main = _on_path(cyan, main) and not _on_path(cyan, branch)
    yellow_main = _on_path(yellow, main) and not _on_path(yellow, branch)
    if cyan_main == yellow_main:
        raise ValueError("exactly one colored station must occupy the trunk interior")
    return "cyan" if cyan_main else "yellow"


def margin(scene: dict) -> float:
    main = scene["main_path"]
    branch = scene["branch_path"]
    # Clearance to every endpoint/junction is the one-sided deletion margin.
    return min(
        _distance_to_vertices(scene["cyan"], [main, branch]),
        _distance_to_vertices(scene["yellow"], [main, branch]),
    )


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["width"]) != CANVAS or int(scene["height"]) != CANVAS:
            return True
        if scene["branch_path"][0] not in scene["main_path"]:
            return True
        if scene["orange"] not in (scene["main_path"][0], scene["main_path"][-1]):
            return True
        if scene["blue"] not in (scene["main_path"][0], scene["main_path"][-1]):
            return True
        return margin(scene) < 40.0 or analytic_gold(scene) not in ("cyan", "yellow")
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
