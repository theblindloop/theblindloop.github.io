"""Renderer and analytic gold for a two-panel heatmap row-reversal question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "heatmap_column_reversal-0.1.0"
WIDTH = 1040
HEIGHT = 620
BACKGROUND = (247, 249, 246)
PLOT_FILL = (255, 255, 255)
AXIS_INK = (54, 64, 72)
GRID_INK = (236, 238, 235)
COLD = (39, 89, 171)
HOT = (222, 71, 45)
LABELS = ("A", "B", "C", "D")
GRID_LEFTS = (156, 636)
GRID_TOP = 150
CELL_W = 66
CELL_H = 70
ROWS = 4
COLS = 5
GRID_WIDTH = CELL_W * COLS
GRID_HEIGHT = CELL_H * ROWS
GRID_BOTTOM = GRID_TOP + GRID_HEIGHT
ROW_LABEL_SCALE = 3
HEADER_SCALE = 5
ROW_LABEL_X_OFFSET = -28
# The header glyph ends immediately above the panel frame, so its pixels are
# not contaminated by the frame's horizontal outline.
HEADER_Y = 67
MIN_MEAN_GAP = 0.10

GLYPHS: dict[str, tuple[str, ...]] = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "P": ("11110", "10001", "10001", "11110", "10000", "10000", "10000"),
    "Q": ("01110", "10001", "10001", "10001", "10101", "10010", "01101"),
}


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _draw_bitmap(
    draw: ImageDraw.ImageDraw,
    x: int,
    y: int,
    label: str,
    scale: int,
    fill: tuple[int, int, int] = AXIS_INK,
) -> None:
    for row, pattern in enumerate(GLYPHS[label]):
        for col, bit in enumerate(pattern):
            if bit == "1":
                draw.rectangle(
                    (
                        x + col * scale,
                        y + row * scale,
                        x + (col + 1) * scale - 1,
                        y + (row + 1) * scale - 1,
                    ),
                    fill=fill,
                )


def _value_color(value: float) -> tuple[int, int, int]:
    t = max(0.0, min(1.0, float(value)))
    return tuple(round(COLD[i] * (1.0 - t) + HOT[i] * t) for i in range(3))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with the answer identity and row position decoupled."""

    seed = int(seed)
    rng = random.Random(seed)
    answer_label = LABELS[seed % len(LABELS)]
    answer_row = (seed // len(LABELS)) % len(LABELS)
    remaining_labels = [label for label in LABELS if label != answer_label]
    rng.shuffle(remaining_labels)
    row_labels = [""] * ROWS
    row_labels[answer_row] = answer_label
    cursor = 0
    for row in range(ROWS):
        if row != answer_row:
            row_labels[row] = remaining_labels[cursor]
            cursor += 1

    p_levels = [0.29, 0.48, 0.67]
    q_levels = [0.74, 0.53, 0.33]
    rng.shuffle(p_levels)
    rng.shuffle(q_levels)
    p_values: list[list[float]] = []
    q_values: list[list[float]] = []
    for row in range(ROWS):
        if row == answer_row:
            p_base, q_base = 0.86, 0.12
        else:
            p_base = p_levels.pop()
            q_base = q_levels.pop()
        p_row: list[float] = []
        q_row: list[float] = []
        for col in range(COLS):
            column_wave = (col - 2) * 0.004
            p_row.append(round(max(0.04, min(0.96, p_base + column_wave + rng.uniform(-0.018, 0.018))), 3))
            q_row.append(round(max(0.04, min(0.96, q_base - column_wave + rng.uniform(-0.018, 0.018))), 3))
        p_values.append(p_row)
        q_values.append(q_row)

    return {
        "p_values": p_values,
        "q_values": q_values,
        "row_labels": row_labels,
        "panel_order": "QP" if rng.randrange(2) else "PQ",
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel_name: str,
    values: list[list[float]],
    row_labels: list[str],
    grid_left: int,
) -> None:
    frame_left = grid_left - 55
    frame_top = GRID_TOP - 48
    frame_right = grid_left + GRID_WIDTH + 20
    frame_bottom = GRID_BOTTOM + 26
    draw.rectangle((frame_left, frame_top, frame_right, frame_bottom), fill=PLOT_FILL, outline=AXIS_INK, width=2)
    _draw_bitmap(draw, grid_left + 4, HEADER_Y, panel_name, HEADER_SCALE)
    draw.text((grid_left + 38, HEADER_Y + 10), "panel", fill=AXIS_INK, font=_font())
    for row in range(ROWS):
        row_top = GRID_TOP + row * CELL_H
        label_y = row_top + (CELL_H - 7 * ROW_LABEL_SCALE) // 2
        _draw_bitmap(draw, grid_left + ROW_LABEL_X_OFFSET, label_y, row_labels[row], ROW_LABEL_SCALE)
        for col in range(COLS):
            left = grid_left + col * CELL_W
            color = _value_color(values[row][col])
            draw.rectangle(
                (left, row_top, left + CELL_W - 1, row_top + CELL_H - 1),
                fill=color,
                outline=GRID_INK,
                width=2,
            )
        draw.text((grid_left + GRID_WIDTH + 5, row_top + 28), str(row + 1), fill=AXIS_INK, font=_font())
    for col in range(COLS):
        draw.text((grid_left + col * CELL_W + 28, GRID_BOTTOM + 7), str(col + 1), fill=AXIS_INK, font=_font())
    draw.text((grid_left + GRID_WIDTH // 2 - 18, GRID_BOTTOM + 18), "columns", fill=AXIS_INK, font=_font())


def _draw_colorbar(draw: ImageDraw.ImageDraw) -> None:
    left, top, width, height = 360, 515, 320, 22
    for index in range(width):
        draw.line((left + index, top, left + index, top + height), fill=_value_color(index / max(1, width - 1)), width=1)
    draw.rectangle((left, top, left + width, top + height), outline=AXIS_INK, width=1)
    draw.text((left - 2, top + 28), "cold", fill=AXIS_INK, font=_font())
    draw.text((left + width - 23, top + 28), "hot", fill=AXIS_INK, font=_font())
    draw.text((left + 106, top - 18), "shared heat scale", fill=AXIS_INK, font=_font())


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((35, 28), "Cross-panel heatmap row reversal", fill=AXIS_INK, font=_font())
    draw.text((35, 48), "Mean each labeled row across five cells", fill=AXIS_INK, font=_font())
    panels = {"P": scene["p_values"], "Q": scene["q_values"]}
    order = str(scene["panel_order"])
    for slot, name in enumerate(order):
        _draw_panel(draw, name, panels[name], scene["row_labels"], GRID_LEFTS[slot])
    _draw_colorbar(draw)
    return image


def _row_means(values: list[list[float]]) -> list[float]:
    return [sum(float(value) for value in row) / len(row) for row in values]


def _winner_indices(values: list[float], *, highest: bool) -> list[int]:
    if not values:
        return []
    target = max(values) if highest else min(values)
    return [index for index, value in enumerate(values) if abs(value - target) <= 1e-9]


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the row label that is highest in P and lowest in Q."""

    p_means = _row_means(scene["p_values"])
    q_means = _row_means(scene["q_values"])
    row_labels = scene["row_labels"]
    if len(p_means) != ROWS or len(q_means) != ROWS or len(row_labels) != ROWS:
        return "quarantine"
    if len(set(row_labels)) != ROWS or any(label not in LABELS for label in row_labels):
        return "quarantine"
    p_winners = _winner_indices(p_means, highest=True)
    q_winners = _winner_indices(q_means, highest=False)
    if len(p_winners) != 1 or len(q_winners) != 1 or p_winners[0] != q_winners[0]:
        return "quarantine"
    return str(row_labels[p_winners[0]])


def margin(scene: dict[str, Any]) -> float:
    """Smallest top-two row-mean separation in the two panels."""

    p_means = sorted(_row_means(scene["p_values"]), reverse=True)
    q_means = sorted(_row_means(scene["q_values"]))
    if len(p_means) < 2 or len(q_means) < 2:
        return -1.0
    return min(p_means[0] - p_means[1], q_means[1] - q_means[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    values = [
        float(value)
        for matrix_key in ("p_values", "q_values")
        for row in scene.get(matrix_key, [])
        for value in row
    ]
    return (
        analytic_gold(scene) == "quarantine"
        or margin(scene) < MIN_MEAN_GAP
        or len(values) != ROWS * COLS * 2
        or any(value < 0.0 or value > 1.0 for value in values)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent alias is declared; all label inputs are rendered directly."""

    return []
