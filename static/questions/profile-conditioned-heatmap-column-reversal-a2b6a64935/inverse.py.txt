"""Independent pixel-only inverse for the heatmap row-reversal world."""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

WIDTH = 1040
HEIGHT = 620
AXIS_INK = (54, 64, 72)
COLD = np.array((39, 89, 171), dtype=np.float64)
HOT = np.array((222, 71, 45), dtype=np.float64)
GRID_LEFTS = (156, 636)
GRID_TOP = 150
CELL_W = 66
CELL_H = 70
ROWS = 4
COLS = 5
ROW_LABEL_SCALE = 3
HEADER_SCALE = 5
ROW_LABEL_X_OFFSET = -28
HEADER_Y = 67
LABELS = ("A", "B", "C", "D")
GLYPHS: dict[str, tuple[str, ...]] = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "P": ("11110", "10001", "10001", "11110", "10000", "10000", "10000"),
    "Q": ("01110", "10001", "10001", "10001", "10101", "10010", "01101"),
}


def _pattern_mask(label: str, scale: int) -> np.ndarray:
    return np.repeat(np.repeat(np.array([[bit == "1" for bit in row] for row in GLYPHS[label]], dtype=bool), scale, axis=0), scale, axis=1)


def _ink_mask(crop: np.ndarray) -> np.ndarray:
    target = np.array(AXIS_INK, dtype=np.int16)
    return np.linalg.norm(crop.astype(np.int16) - target, axis=2) < 12


def _read_glyph(arr: np.ndarray, x: int, y: int, label_set: tuple[str, ...], scale: int) -> str | None:
    height, width = 7 * scale, 5 * scale
    if x < 0 or y < 0 or x + width > arr.shape[1] or y + height > arr.shape[0]:
        return None
    crop = arr[y : y + height, x : x + width]
    observed = _ink_mask(crop)
    choices = []
    for label in label_set:
        expected = _pattern_mask(label, scale)
        choices.append((int(np.count_nonzero(observed != expected)), label))
    score, label = min(choices)
    return label if score <= max(2, scale * 2) else None


def _panel_slots(arr: np.ndarray) -> tuple[int, int] | None:
    labels = []
    for slot, grid_left in enumerate(GRID_LEFTS):
        labels.append(_read_glyph(arr, grid_left + 4, HEADER_Y, ("P", "Q"), HEADER_SCALE))
    if labels.count("P") != 1 or labels.count("Q") != 1:
        return None
    return labels.index("P"), labels.index("Q")


def _row_labels(arr: np.ndarray, slot: int) -> list[str] | None:
    grid_left = GRID_LEFTS[slot]
    labels: list[str] = []
    for row in range(ROWS):
        row_top = GRID_TOP + row * CELL_H
        y = row_top + (CELL_H - 7 * ROW_LABEL_SCALE) // 2
        label = _read_glyph(arr, grid_left + ROW_LABEL_X_OFFSET, y, LABELS, ROW_LABEL_SCALE)
        if label is None:
            return None
        labels.append(label)
    return labels if len(set(labels)) == ROWS else None


def _temperature(rgb: np.ndarray) -> float:
    vector = HOT - COLD
    value = np.asarray(rgb, dtype=np.float64) - COLD
    return float(np.dot(value, vector) / np.dot(vector, vector))


def _panel_means(arr: np.ndarray, slot: int) -> list[float]:
    grid_left = GRID_LEFTS[slot]
    means = []
    for row in range(ROWS):
        values = []
        for col in range(COLS):
            x = grid_left + col * CELL_W + CELL_W // 2
            y = GRID_TOP + row * CELL_H + CELL_H // 2
            patch = arr[y - 2 : y + 3, x - 2 : x + 3]
            values.append(_temperature(patch.mean(axis=(0, 1))))
        means.append(sum(values) / len(values))
    return means


def decision_from_image(image: Image.Image) -> str:
    """Recover the row label from the final raster without scene access."""

    try:
        if image.size != (WIDTH, HEIGHT):
            return "quarantine"
        arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
        slots = _panel_slots(arr)
        if slots is None:
            return "quarantine"
        p_slot, q_slot = slots
        labels = _row_labels(arr, p_slot)
        if labels is None:
            return "quarantine"
        p_means = _panel_means(arr, p_slot)
        q_means = _panel_means(arr, q_slot)
        p_order = sorted(range(ROWS), key=lambda row: p_means[row], reverse=True)
        q_order = sorted(range(ROWS), key=lambda row: q_means[row])
        if p_means[p_order[0]] - p_means[p_order[1]] < 0.10:
            return "quarantine"
        if q_means[q_order[1]] - q_means[q_order[0]] < 0.10:
            return "quarantine"
        if p_order[0] != q_order[0]:
            return "quarantine"
        return labels[p_order[0]]
    except (IndexError, TypeError, ValueError):
        return "quarantine"
