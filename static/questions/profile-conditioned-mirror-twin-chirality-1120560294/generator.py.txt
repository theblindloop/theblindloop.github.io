"""Latent-scene sampler and RGB renderer for the mirror-twin chirality world.

A grey vertical bar splits a white canvas. One coloured step-block shape (five
columns of integer height) stands on each side of the bar, on a shared grey
baseline. Both instances share colour, cell size, column multiset, total area
and bounding box; their placements are exactly symmetric about the bar. The only
thing that differs between the two answer classes is the HANDEDNESS of the
right-hand shape: either its column heights run in the reversed order (a true
reflection of its twin) or in the same order (an unflipped copy).

The familiar prior under test is that something drawn beside a mirror bar is its
reflection. The canonical variant satisfies it; the counterfactual variant
contradicts it with identical wording, palette, scale, ink and layout.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mirror_twin_chirality-renderer-0.1.0"

WIDTH = 448
HEIGHT = 448
SS = 3                      # supersampling factor; BOX downsample is exact
N_COLS = 5
BAND_HALF = 4               # half width of the grey mirror bar, in output px
PIP_SIZE = 11               # side of a decorative grey pip square
QUARANTINE_MARGIN = 12.0    # chirality percent below which a scene is quarantined
MIN_SAMPLED_MARGIN = 6.0    # sampler never emits a shape weaker than this

WHITE = (255, 255, 255)
BAND_GREY = (122, 122, 122)
PIP_GREY = (88, 88, 88)
BASE_GREY = (150, 150, 150)

# Saturated fills only: every entry has chroma (max-min) >= 130, so the pixel
# oracle can separate shape ink from every achromatic decoration.
PALETTE = (
    (211, 47, 47),
    (25, 118, 210),
    (56, 142, 60),
    (245, 124, 0),
    (123, 31, 162),
    (0, 151, 167),
    (194, 24, 91),
    (85, 139, 47),
)


def chirality(heights: list[int]) -> float:
    """Percent mismatch between the skyline and its left-right reversal.

    This is the drawn asymmetry of the shape: 0 for a palindromic skyline (the
    two answer classes would then be the same picture) and large when the tall
    end is clearly on one side.
    """
    rev = list(heights)[::-1]
    inter = sum(min(a, b) for a, b in zip(heights, rev))
    union = sum(max(a, b) for a, b in zip(heights, rev))
    return round(100.0 * (1.0 - inter / float(union)), 6)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        if rng.random() < 0.30:
            # Near-boundary draw: perturb a palindrome by one cell so that the
            # visible asymmetry, and with it the decision margin, is small.
            a, b, c = rng.randint(1, 6), rng.randint(1, 6), rng.randint(1, 6)
            heights = [a, b, c, b, a]
            i = rng.randrange(N_COLS)
            heights[i] = max(1, min(6, heights[i] + rng.choice([-1, 1])))
        else:
            heights = [rng.randint(1, 6) for _ in range(N_COLS)]
        if heights == heights[::-1]:
            continue
        if chirality(heights) < MIN_SAMPLED_MARGIN:
            continue
        break
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cell": rng.choice([22, 24, 26, 28]),
        "heights": heights,
        "mirrored": rng.random() < 0.5,
        "color": list(PALETTE[rng.randrange(len(PALETTE))]),
        "band_x": rng.randint(214, 234),
        "baseline": rng.randint(300, 336),
        "gap": rng.randint(26, 46),
        "pip_offsets": sorted(rng.sample(range(48, 400, 8), 5)),
    }


def analytic_gold(scene: dict) -> str:
    """Gold depends on exactly one scene field, and that field is drawn."""
    return "mirrored" if scene["mirrored"] else "copied"


def margin(scene: dict) -> float:
    """Visible decision margin: the shape's own drawn asymmetry, in percent."""
    return chirality(list(scene["heights"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every field is rendered (see candidate.json)."""
    return []


def _skyline_points(heights: list[int], x0: int, baseline: int, cell: int) -> list[tuple[int, int]]:
    pts = [(x0, baseline)]
    for i, h in enumerate(heights):
        x_left = x0 + i * cell
        x_right = x_left + cell
        top = baseline - h * cell
        pts.append((x_left, top))
        pts.append((x_right, top))
    pts.append((x0 + len(heights) * cell, baseline))
    return pts


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    cell = int(scene["cell"])
    band_x = int(scene["band_x"])
    baseline = int(scene["baseline"])
    gap = int(scene["gap"])
    heights = [int(v) for v in scene["heights"]]
    right_heights = heights[::-1] if scene["mirrored"] else list(heights)
    color = tuple(int(c) for c in scene["color"])

    img = Image.new("RGB", (w * SS, h * SS), WHITE)
    d = ImageDraw.Draw(img)

    def rect(x0: int, y0: int, x1: int, y1: int, fill: tuple[int, int, int]) -> None:
        d.rectangle([x0 * SS, y0 * SS, x1 * SS - 1, y1 * SS - 1], fill=fill)

    # Fixed decoration: baseline, mirror bar, and achromatic pips on the bar.
    rect(24, baseline, w - 24, baseline + 3, BASE_GREY)
    rect(band_x - BAND_HALF, 30, band_x + BAND_HALF, h - 30, BAND_GREY)
    for off in scene["pip_offsets"]:
        off = int(off)
        rect(band_x - PIP_SIZE // 2, off, band_x - PIP_SIZE // 2 + PIP_SIZE,
             off + PIP_SIZE, PIP_GREY)

    span = N_COLS * cell
    left_x0 = band_x - BAND_HALF - gap - span
    right_x0 = band_x + BAND_HALF + gap
    for x0, hs in ((left_x0, heights), (right_x0, right_heights)):
        pts = [(x * SS, y * SS) for x, y in _skyline_points(hs, x0, baseline, cell)]
        d.polygon(pts, fill=color)

    return img.resize((w, h), Image.BOX)


# --- gold-preserving nuisance transforms (used by verify.py and tests) --------

def repalette(scene: dict, k: int) -> dict:
    idx = PALETTE.index(tuple(int(c) for c in scene["color"]))
    alt = dict(scene)
    alt["color"] = list(PALETTE[(idx + k) % len(PALETTE)])
    return alt


def permute_pips(scene: dict, seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    alt = dict(scene)
    alt["pip_offsets"] = sorted(rng.sample(range(48, 400, 8), 5))
    return alt


def translate(scene: dict, dx: int, dy: int) -> dict:
    """Translate the whole configuration (bar, baseline and both shapes)."""
    alt = dict(scene)
    alt["band_x"] = int(scene["band_x"]) + dx
    alt["baseline"] = int(scene["baseline"]) + dy
    return alt


# --- label-transforming transform -------------------------------------------

def toggle_handedness(scene: dict) -> dict:
    alt = dict(scene)
    alt["mirrored"] = not scene["mirrored"]
    return alt


def expected_toggled_gold(decision: str) -> str:
    if decision not in ("mirrored", "copied"):
        raise ValueError(f"unknown decision: {decision}")
    return "copied" if decision == "mirrored" else "mirrored"
