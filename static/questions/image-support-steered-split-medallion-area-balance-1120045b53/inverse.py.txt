"""Independent inverse arm using only pixels of the final PNG."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AmbiguousImage(ValueError):
    pass


def _largest_component(mask: np.ndarray) -> tuple[int, tuple[int, int, int, int]]:
    """Return area and bounding box of the largest 4-connected component."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best_area = 0
    best_box = (0, 0, 0, 0)
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            yy, xx = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, xx), max(max_x, xx)
            min_y, max_y = min(min_y, yy), max(max_y, yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area > best_area:
            best_area = area
            best_box = (min_x, min_y, max_x + 1, max_y + 1)
    return best_area, best_box


def _measure(image: Image.Image) -> tuple[int, int, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or min(arr.shape[:2]) < 64:
        raise AmbiguousImage("image is too small")

    maximum = arr.max(axis=2)
    minimum = arr.min(axis=2)
    saturation = (maximum - minimum) / np.maximum(maximum, 1.0)
    colored = saturation > 0.28
    colored_area, box = _largest_component(colored)
    if colored_area < 12000:
        raise AmbiguousImage("no substantial colored medallion interior")
    x0, y0, x1, y1 = box
    span_x, span_y = x1 - x0, y1 - y0
    if not (150 <= span_x <= 230 and 150 <= span_y <= 230):
        raise AmbiguousImage("colored interior has implausible extent")
    if abs(span_x - span_y) > 10:
        raise AmbiguousImage("colored interior is not circular in extent")

    # A complete disk has area / bounding-square area near pi/4. This integrity
    # check makes erasure of a substantial interior patch cause abstention.
    fill_ratio = colored_area / float(span_x * span_y)
    if not 0.755 <= fill_ratio <= 0.805:
        raise AmbiguousImage("medallion interior is incomplete")

    values = arr[colored]
    luminance = values @ np.array([0.2126, 0.7152, 0.0722], dtype=np.float32)
    lo = float(np.percentile(luminance, 10))
    hi = float(np.percentile(luminance, 90))
    if hi - lo < 55.0:
        raise AmbiguousImage("dark and light regions are not separable")
    threshold = (lo + hi) / 2.0
    luma_map = arr @ np.array([0.2126, 0.7152, 0.0722], dtype=np.float32)
    dark_mask = colored & (luma_map < threshold)
    light_mask = colored & ~dark_mask
    dark_area, _ = _largest_component(dark_mask)
    light_area, _ = _largest_component(light_mask)
    if dark_area < 3500 or light_area < 3500:
        raise AmbiguousImage("one colored region is missing or too small")
    # Antialiased outline pixels can form tiny hue fragments; only the largest
    # connected component of each named region contributes to visible area.
    imbalance = abs(dark_area - light_area) / float(dark_area + light_area)
    return dark_area, light_area, imbalance


def decision_from_image(image: Image.Image) -> str:
    """Report raster equality; structural corruption raises AmbiguousImage."""
    _, _, imbalance = _measure(image)
    return "yes" if imbalance <= 0.025 else "no"
