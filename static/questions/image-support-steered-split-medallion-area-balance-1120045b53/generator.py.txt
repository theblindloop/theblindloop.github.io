"""Deterministic visual world for a visibly balanced or biased medallion."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


CANVAS = 512
SUPER_SAMPLE = 4
PALETTES = (
    ((42, 72, 156), (246, 188, 65)),
    ((31, 103, 111), (246, 166, 122)),
    ((93, 55, 143), (238, 191, 83)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced classes plus both directions and rare quarantined offsets."""
    rng = random.Random(int(seed))
    kind = int(seed) % 6
    radius = rng.randint(92, 108)
    if kind in (0, 3):
        offset = 0
    elif kind == 5:
        offset = rng.choice((-1, 1)) * rng.randint(4, 7)
    else:
        offset = rng.choice((-1, 1)) * rng.randint(20, 32)
    dark, light = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "width": CANVAS,
        "height": CANVAS,
        "center_x": rng.randint(224, 288),
        "center_y": rng.randint(224, 288),
        "radius": radius,
        "partition_offset": offset,
        "partition_side": rng.choice(("right", "left")),
        "dark_rgb": list(dark),
        "light_rgb": list(light),
        "background_rgb": [247, 246, 242],
        "outline_rgb": [35, 38, 43],
        "outline_width": 5,
    }


def _signed_partition(scene: dict[str, Any]) -> int:
    side = 1 if scene["partition_side"] == "right" else -1
    return side * int(scene["partition_offset"])


def render(scene: dict[str, Any]) -> Image.Image:
    """Render complementary colored regions clipped to one circular interior."""
    ss = SUPER_SAMPLE
    width, height = int(scene["width"]), int(scene["height"])
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    radius = int(scene["radius"])
    outline_width = int(scene["outline_width"])
    background = tuple(scene["background_rgb"])
    outline = tuple(scene["outline_rgb"])
    dark = tuple(scene["dark_rgb"])
    light = tuple(scene["light_rgb"])

    image = Image.new("RGB", (width * ss, height * ss), background)
    disk_mask = Image.new("L", image.size, 0)
    mask_draw = ImageDraw.Draw(disk_mask)
    inner_radius = radius - outline_width
    inner_box = tuple(
        int(value * ss)
        for value in (
            cx - inner_radius,
            cy - inner_radius,
            cx + inner_radius,
            cy + inner_radius,
        )
    )
    mask_draw.ellipse(inner_box, fill=255)

    split_x = (cx + _signed_partition(scene)) * ss
    colored = Image.new("RGB", image.size, light)
    colored_draw = ImageDraw.Draw(colored)
    colored_draw.rectangle((0, 0, split_x - 1, height * ss), fill=dark)
    image.paste(colored, (0, 0), disk_mask)

    draw = ImageDraw.Draw(image)
    outer_box = tuple(
        int(value * ss)
        for value in (cx - radius, cy - radius, cx + radius, cy + radius)
    )
    draw.ellipse(outer_box, outline=outline, width=outline_width * ss)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _offset_fraction(scene: dict[str, Any]) -> float:
    return abs(float(scene["partition_offset"])) / float(scene["radius"])


def analytic_gold(scene: dict[str, Any]) -> str:
    """Equality is exact in the latent world; near offsets are quarantined."""
    return "yes" if _offset_fraction(scene) == 0.0 else "no"


def margin(scene: dict[str, Any]) -> float:
    """Visible partition displacement as a fraction of medallion radius."""
    fraction = _offset_fraction(scene)
    return 0.12 if fraction == 0.0 else fraction


def is_quarantined(scene: dict[str, Any]) -> bool:
    fraction = _offset_fraction(scene)
    return 0.0 < fraction < 0.10


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Flip the redundant side/sign encoding without changing any pixel."""
    twin = dict(scene)
    twin["partition_side"] = "left" if scene["partition_side"] == "right" else "right"
    twin["partition_offset"] = -int(scene["partition_offset"])
    return [("side_sign_reencoding", twin)]
