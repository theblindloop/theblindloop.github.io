"""Independent PNG-only inverse for trajectory directness.

The inverse knows only the public palette and question contract. It segments
colored marker disks, recovers their centers, orders the five centers from left
to right, and recomputes the declared dimensionless ratio.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (210, 55, 65),
    "blue": (42, 99, 184),
    "green": (31, 137, 91),
    "orange": (226, 120, 32),
}
_COLOR_TOL = 8
_MIN_MARKER_AREA = 120
_MAX_MARKER_AREA = 260
_ABSTAIN_GAP = 0.015


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    ys, xs = np.nonzero(mask)
    remaining = {int(y) * width + int(x) for y, x in zip(ys, xs)}
    groups: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        group: list[tuple[int, int]] = []
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            group.append((y, x))
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        groups.append(group)
    return groups


def _marker_centers(image: Image.Image, rgb: tuple[int, int, int]) -> list[tuple[float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    target = np.asarray(rgb, dtype=np.int16)
    mask = np.max(np.abs(array - target), axis=2) <= _COLOR_TOL
    centers = []
    debris = 0
    for component in _components(mask):
        area = len(component)
        if _MIN_MARKER_AREA <= area <= _MAX_MARKER_AREA:
            ys = [item[0] for item in component]
            xs = [item[1] for item in component]
            centers.append((sum(xs) / area, sum(ys) / area))
        else:
            debris += area
    if debris > 12:
        raise ValueError(f"color mask contains {debris} unsupported pixels")
    if len(centers) != 5:
        raise ValueError(f"expected five separated markers, recovered {len(centers)}")
    centers.sort(key=lambda point: point[0])
    if min(b[0] - a[0] for a, b in zip(centers, centers[1:])) < 25:
        raise ValueError("marker ordering is not visually resolved")
    return centers


def _ratio(points: list[tuple[float, float]]) -> float:
    endpoint = math.dist(points[0], points[-1])
    total = sum(math.dist(a, b) for a, b in zip(points, points[1:]))
    if total <= 0:
        raise ValueError("degenerate trajectory")
    return endpoint / total


def decision_from_image(image: Image.Image) -> str:
    ratios = {name: _ratio(_marker_centers(image, rgb)) for name, rgb in _COLORS.items()}
    ordered = sorted(ratios.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < _ABSTAIN_GAP:
        raise ValueError("top two pixel-derived ratios are ambiguous")
    return ordered[0][0]
