"""Deterministic Pillow renderer for the trajectory-directness world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 620
COLORS: dict[str, tuple[int, int, int]] = {
    "red": (210, 55, 65),
    "blue": (42, 99, 184),
    "green": (31, 137, 91),
    "orange": (226, 120, 32),
}
COLOR_NAMES = tuple(COLORS)
PANELS = (
    (30, 105, 365, 330),
    (395, 105, 730, 330),
    (30, 360, 365, 585),
    (395, 360, 730, 585),
)
QUARANTINE_GAP = 0.030


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = f"/usr/share/fonts/truetype/dejavu/{name}"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _metric(points: list[list[int]] | list[tuple[int, int]]) -> tuple[float, float, float]:
    endpoint = math.dist(points[0], points[-1])
    path_length = sum(math.dist(a, b) for a, b in zip(points, points[1:]))
    return endpoint / path_length, endpoint, path_length


def _role_points(rng: random.Random, role: str) -> list[tuple[int, int]]:
    """Make one local five-point path with a controlled but randomized shape.

    The four roles jointly defeat two tempting shortcuts: the direct role is
    neither the shortest path nor the path with the widest endpoint span.
    """
    if role == "direct":
        span = rng.randint(224, 238)
        delta = rng.randint(-28, 28)
        deviations = [0, rng.randint(-3, 3), rng.randint(-3, 3), rng.randint(-3, 3), 0]
    elif role == "short_hump":
        span = rng.randint(172, 186)
        delta = rng.randint(-12, 12)
        amp = rng.randint(33, 45) * rng.choice((-1, 1))
        deviations = [0, int(0.65 * amp), amp, int(0.65 * amp), 0]
    elif role == "wide_zigzag":
        span = rng.randint(254, 266)
        delta = rng.randint(-16, 16)
        amp = rng.randint(28, 39) * rng.choice((-1, 1))
        deviations = [0, amp, -amp, amp, 0]
    elif role == "rough_zigzag":
        span = rng.randint(205, 222)
        delta = rng.randint(-20, 20)
        amp = rng.randint(48, 62) * rng.choice((-1, 1))
        deviations = [0, amp, -amp, amp, 0]
    else:
        raise ValueError(f"unknown role {role!r}")

    start_x = rng.randint(31, 43)
    xs = [start_x]
    for i in range(1, 4):
        xs.append(round(start_x + span * i / 4 + rng.randint(-4, 4)))
    xs.append(start_x + span)
    y0 = rng.randint(102, 123)
    points = []
    for i, (x, deviation) in enumerate(zip(xs, deviations)):
        baseline = y0 + delta * i / 4
        points.append((int(x), int(round(baseline + deviation))))
    return points


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    roles = ("direct", "short_hump", "wide_zigzag", "rough_zigzag")

    # The templates have broad headroom, but retain a deterministic rejection
    # loop so future harmless jitter changes cannot admit a shortcut or tie.
    for _ in range(500):
        local = {role: _role_points(rng, role) for role in roles}
        stats = {role: _metric(local[role]) for role in roles}
        ranked = sorted(roles, key=lambda role: stats[role][0], reverse=True)
        gap = stats[ranked[0]][0] - stats[ranked[1]][0]
        winner = ranked[0]
        lengths = {role: stats[role][2] for role in roles}
        endpoints = {role: stats[role][1] for role in roles}
        if (
            winner == "direct"
            and gap >= 0.045
            and lengths[winner] > min(lengths.values()) + 3.0
            and endpoints[winner] < max(endpoints.values()) - 8.0
        ):
            break
    else:
        raise RuntimeError("could not construct a separated, shortcut-controlled scene")

    winning_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    other_colors = [color for color in COLOR_NAMES if color != winning_color]
    rng.shuffle(other_colors)
    role_color = {"direct": winning_color}
    role_color.update(dict(zip(roles[1:], other_colors)))

    winning_panel = (int(seed) // len(COLOR_NAMES)) % len(PANELS)
    other_panels = [index for index in range(len(PANELS)) if index != winning_panel]
    rng.shuffle(other_panels)
    role_panel = {"direct": winning_panel}
    role_panel.update(dict(zip(roles[1:], other_panels)))

    paths = []
    for role in roles:
        panel = role_panel[role]
        x0, y0, _, _ = PANELS[panel]
        global_points = [[x0 + x, y0 + y] for x, y in local[role]]
        paths.append({"color": role_color[role], "points": global_points})
    rng.shuffle(paths)
    return {"paths": paths}


def _ratios(scene: dict[str, Any]) -> dict[str, float]:
    return {path["color"]: _metric(path["points"])[0] for path in scene["paths"]}


def analytic_gold(scene: dict[str, Any]) -> str:
    ratios = _ratios(scene)
    return max(ratios, key=ratios.get)


def margin(scene: dict[str, Any]) -> float:
    ordered = sorted(_ratios(scene).values(), reverse=True)
    return ordered[0] - ordered[1]


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = {"paths": list(reversed(scene["paths"]))}
    return [("reverse_disjoint_draw_order", twin)]


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 249, 252))
    draw = ImageDraw.Draw(image)
    draw.rectangle((0, 0, WIDTH - 1, 82), fill=(25, 34, 49))
    draw.text((30, 15), "TRAJECTORY DIRECTNESS", font=_font(25, True), fill=(255, 255, 255))
    draw.text(
        (31, 50),
        "ratio = endpoint distance / sum of four segment lengths",
        font=_font(15),
        fill=(205, 214, 227),
    )

    for index, (x0, y0, x1, y1) in enumerate(PANELS):
        draw.rounded_rectangle((x0, y0, x1, y1), radius=10, fill=(255, 255, 255), outline=(205, 212, 222), width=2)
        draw.text((x0 + 14, y0 + 12), chr(ord("A") + index), font=_font(16, True), fill=(71, 79, 91))
        draw.line((x0 + 24, y1 - 26, x1 - 24, y1 - 26), fill=(229, 233, 239), width=1)

    for path in scene["paths"]:
        points = [tuple(point) for point in path["points"]]
        draw.line(points, fill=(151, 160, 174), width=4, joint="curve")
        rgb = COLORS[path["color"]]
        for x, y in points:
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=(255, 255, 255))
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=rgb)

    draw.text((31, 598), "First and last markers are the leftmost and rightmost in each panel.", font=_font(13), fill=(91, 100, 114))
    return image
