"""Scene sampler and renderer for a left-to-right cumulative parity reveal."""

from __future__ import annotations

import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

N = 12
CELL = 18
GRID_PX = N * CELL
CYAN = (0, 174, 199)
MAGENTA = (220, 45, 135)
YELLOW = (244, 190, 35)
DECISIONS = ("diamond", "hourglass", "left fan", "right fan")


def _templates() -> dict[str, np.ndarray]:
    """Equal-ink glyphs whose row intervals also have equal encoded ink."""
    widths = {
        "diamond": (2, 4, 6, 8, 8, 6, 4, 2),
        "hourglass": (8, 6, 4, 2, 2, 4, 6, 8),
        "left fan": (2, 4, 6, 8, 8, 6, 4, 2),
        "right fan": (2, 4, 6, 8, 8, 6, 4, 2),
    }
    result: dict[str, np.ndarray] = {}
    for name, row_widths in widths.items():
        mask = np.zeros((N, N), dtype=np.uint8)
        for offset, width in enumerate(row_widths):
            if name in ("diamond", "hourglass"):
                left = (N - width) // 2
            elif name == "left fan":
                left = 1
            else:
                left = N - width - 1
            mask[offset + 2, left:left + width] = 1
        result[name] = mask
    return result


TEMPLATES = _templates()
CARRIER = np.fromfunction(lambda r, c: (r + c) % 2 == 0, (N, N), dtype=int).astype(np.uint8)


def _prefix_parity(cells: np.ndarray) -> np.ndarray:
    return np.bitwise_xor.accumulate(cells.astype(np.uint8), axis=1)


def _encode(target: np.ndarray) -> np.ndarray:
    source = np.zeros_like(target)
    source[:, 0] = target[:, 0]
    source[:, 1:] = target[:, 1:] ^ target[:, :-1]
    return source


def _scores(transformed: np.ndarray) -> list[tuple[int, str]]:
    return sorted(
        (int(np.count_nonzero(transformed ^ template)), name)
        for name, template in TEMPLATES.items()
    )


def _decoded(cells: list[list[int]]) -> tuple[str, float]:
    cleared = np.asarray(cells, dtype=np.uint8) ^ CARRIER
    scores = _scores(_prefix_parity(cleared))
    return scores[0][1], float(scores[1][0] - scores[0][0])


def _balanced_perturb(source: np.ndarray, label: str, rng: random.Random, level: int) -> np.ndarray:
    """Swap source bits, preserving total visible ink and the intended class."""
    current = source.copy()
    for _ in range(level):
        ones = list(zip(*np.nonzero(current)))
        zeros = list(zip(*np.nonzero(1 - current)))
        rng.shuffle(ones)
        rng.shuffle(zeros)
        choices: list[tuple[float, int, int, int, int, np.ndarray]] = []
        for r1, c1 in ones[:16]:
            for r0, c0 in zeros[:30]:
                if CARRIER[r1, c1] != CARRIER[r0, c0]:
                    continue
                trial = current.copy()
                trial[r1, c1] = 0
                trial[r0, c0] = 1
                decision, gap = _decoded((trial ^ CARRIER).tolist())
                if decision == label and gap > 2:
                    choices.append((gap, r1, c1, r0, c0, trial))
        if not choices:
            break
        choices.sort(key=lambda item: (item[0], item[1], item[2], item[3], item[4]))
        band = choices[: max(1, min(8, len(choices)))]
        current = band[rng.randrange(len(band))][-1]
    return current


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = DECISIONS[int(seed) % len(DECISIONS)]
    source = _encode(TEMPLATES[label])
    source = _balanced_perturb(source, label, rng, int(seed) % 3)
    displayed = source ^ CARRIER
    palettes = [
        ([22, 25, 29], [250, 249, 245]),
        ([45, 32, 60], [246, 250, 248]),
        ([20, 51, 58], [250, 247, 242]),
    ]
    ink, background = palettes[(int(seed) // 4) % len(palettes)]
    decorations = [
        {"x": 390, "y": 80, "color": [232, 91, 69], "kind": "diamond"},
        {"x": 408, "y": 238, "color": [92, 113, 210], "kind": "circle"},
    ]
    if rng.randrange(2):
        decorations.reverse()
    return {
        "cells": displayed.astype(int).tolist(),
        "grid_x": 66 + 12 * rng.randrange(4),
        "grid_y": 72 + 9 * rng.randrange(3),
        "ink": ink,
        "background": background,
        "decorations": decorations,
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, decorations: list[dict]) -> None:
    for item in sorted(decorations, key=lambda value: (value["x"], value["y"], value["kind"])):
        x, y = int(item["x"]), int(item["y"])
        color = tuple(int(value) for value in item["color"])
        if item["kind"] == "diamond":
            draw.polygon([(x, y - 10), (x + 10, y), (x, y + 10), (x - 10, y)], fill=color)
        else:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)


def render(scene: dict) -> Image.Image:
    cells = np.asarray(scene["cells"], dtype=np.uint8)
    x, y = int(scene["grid_x"]), int(scene["grid_y"])
    ink = tuple(int(value) for value in scene["ink"])
    image = Image.new("RGB", (480, 360), tuple(int(value) for value in scene["background"]))
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene["decorations"])
    draw.rectangle((x - 6, y - 6, x + GRID_PX + 5, y + GRID_PX + 5), fill=CYAN)
    draw.rectangle((x, y, x + GRID_PX - 1, y + GRID_PX - 1), fill=(255, 255, 255))
    for r in range(N):
        for c in range(N):
            left, top = x + c * CELL, y + r * CELL
            if cells[r, c]:
                draw.rectangle((left + 2, top + 2, left + CELL - 3, top + CELL - 3), fill=ink)
    for k in range(N + 1):
        draw.line((x + k * CELL, y, x + k * CELL, y + GRID_PX), fill=(204, 207, 207), width=1)
        draw.line((x, y + k * CELL, x + GRID_PX, y + k * CELL), fill=(204, 207, 207), width=1)
    # The magenta arrow marks the row start; yellow marks the far end.
    draw.polygon([(x - 25, y + 9), (x - 10, y), (x - 10, y + 18)], fill=MAGENTA)
    draw.rectangle((x + GRID_PX + 9, y + 2, x + GRID_PX + 22, y + 16), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return _decoded(scene["cells"])[0]


def margin(scene: dict) -> float:
    return _decoded(scene["cells"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("decoration_serialization_order", twin)]
