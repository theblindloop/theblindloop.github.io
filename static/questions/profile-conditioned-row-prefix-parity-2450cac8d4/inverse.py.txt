"""Independent PNG-only inverse arm for cumulative row parity."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 12


def _templates() -> dict[str, np.ndarray]:
    specifications = {
        "diamond": ((2, 4, 6, 8, 8, 6, 4, 2), "center"),
        "hourglass": ((8, 6, 4, 2, 2, 4, 6, 8), "center"),
        "left fan": ((2, 4, 6, 8, 8, 6, 4, 2), "left"),
        "right fan": ((2, 4, 6, 8, 8, 6, 4, 2), "right"),
    }
    result: dict[str, np.ndarray] = {}
    for name, (widths, alignment) in specifications.items():
        mask = np.zeros((N, N), dtype=np.uint8)
        for offset, width in enumerate(widths):
            if alignment == "center":
                left = (N - width) // 2
            elif alignment == "left":
                left = 1
            else:
                left = N - width - 1
            mask[offset + 2, left:left + width] = 1
        result[name] = mask
    return result


TEMPLATES = _templates()
CARRIER = np.fromfunction(lambda r, c: (r + c) % 2 == 0, (N, N), dtype=int).astype(np.uint8)


def _read_cells(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (
        (array[:, :, 0] < 35)
        & (array[:, :, 1] > 145)
        & (array[:, :, 1] < 205)
        & (array[:, :, 2] > 170)
    )
    ys, xs = np.nonzero(cyan)
    if len(xs) < 700:
        raise ValueError("cyan grid frame is missing")
    min_x, max_x = int(xs.min()), int(xs.max())
    min_y, max_y = int(ys.min()), int(ys.max())
    width, height = max_x - min_x + 1, max_y - min_y + 1
    if not (224 <= width <= 232 and 224 <= height <= 232):
        raise ValueError("cyan frame has invalid dimensions")
    x0, y0 = min_x + 6, min_y + 6
    cell_w, cell_h = (width - 12) / N, (height - 12) / N
    cells = np.zeros((N, N), dtype=np.uint8)
    for r in range(N):
        for c in range(N):
            cx = int(round(x0 + (c + 0.5) * cell_w))
            cy = int(round(y0 + (r + 0.5) * cell_h))
            patch = array[cy - 3:cy + 4, cx - 3:cx + 4]
            cells[r, c] = int(float(patch.mean()) < 120.0)
    return cells


def decision_from_image(image: Image.Image) -> str:
    cleared = _read_cells(image) ^ CARRIER
    transformed = np.bitwise_xor.accumulate(cleared, axis=1)
    scores = sorted(
        (int(np.count_nonzero(transformed ^ template)), name)
        for name, template in TEMPLATES.items()
    )
    if scores[0][0] == scores[1][0]:
        raise ValueError("symbol classification is tied")
    return scores[0][1]
