"""Pixel-only inverse arm for the route-first-waypoint world.

Receives only the final PNG. Segments the dark route, the orange start ring, and
the three colored waypoint disks; then walks the connected route outward from the
orange start, recording the order in which each colored waypoint is met. The
first waypoint met (smallest outward distance) is the answer. If the route is
broken so that no waypoint can be reached from the start, the scene is
ambiguous and the oracle abstains.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CP_COLORS = {
    "red": (220, 50, 50),
    "green": (45, 170, 70),
    "blue": (40, 95, 220),
}
RING_COLOR = (230, 120, 20)
_BG = (255, 255, 255)
_COLOR_TOL = 55
_FG_TOL = 60


def _mask_for(arr, target, tol):
    diff = np.abs(arr.astype(np.int16) - np.array(target, dtype=np.int16))
    return np.all(diff <= tol, axis=2)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    h, w = arr.shape[:2]

    fg = np.any(np.abs(arr.astype(np.int16) - np.array(_BG, dtype=np.int16)) > _FG_TOL, axis=2)

    ring = _mask_for(arr, RING_COLOR, _COLOR_TOL)
    cp_masks = {color: _mask_for(arr, rgb, _COLOR_TOL) for color, rgb in CP_COLORS.items()}

    ring_pts = np.argwhere(ring)
    if not len(ring_pts):
        raise ValueError("no start ring found")
    sy, sx = ring_pts.mean(axis=0)
    start = (int(round(sy)), int(round(sx)))
    if not fg[start[0], start[1]]:
        start = _nearest_fg(start, fg)

    # outward BFS over the connected foreground from the ring center.
    dist = np.full((h, w), -1, dtype=np.int32)
    dq = deque()
    if fg[start[0], start[1]]:
        dist[start] = 0
        dq.append(start)
    reached = {}
    for color, mask in cp_masks.items():
        reached[color] = None
    while dq:
        cy, cx = dq.popleft()
        d = dist[cy, cx]
        for ny, nx in (
            (cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1),
            (cy - 1, cx - 1), (cy - 1, cx + 1), (cy + 1, cx - 1), (cy + 1, cx + 1),
        ):
            if 0 <= ny < h and 0 <= nx < w and fg[ny, nx] and dist[ny, nx] < 0:
                dist[ny, nx] = d + 1
                dq.append((ny, nx))
        # record waypoint distances directly from path pixels adjacently
        for color, mask in cp_masks.items():
            if reached[color] is None and mask[cy, cx]:
                reached[color] = d

    best = None
    best_color = None
    for color in CP_COLORS:
        if reached[color] is not None:
            if best is None or reached[color] < best:
                best = reached[color]
                best_color = color
    if best_color is None:
        raise ValueError("no waypoint reachable from start along the route")
    return best_color


def _nearest_fg(start, fg):
    h, w = fg.shape
    best = None
    for dy in range(-6, 7):
        for dx in range(-6, 7):
            y, x = start[0] + dy, start[1] + dx
            if 0 <= y < h and 0 <= x < w and fg[y, x]:
                best = (y, x)
                break
        if best is not None:
            break
    return best if best is not None else start
