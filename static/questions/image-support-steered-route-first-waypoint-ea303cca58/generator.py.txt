"""Render a single connected orthogonal staircase route carrying colored waypoints.

Latent scene z = {points, waypoints, width, height}. The route is one connected
thin orthogonal polyline (a staircase) confined to a medium central region. A set
of colored waypoint disks (red/green/blue) sit at ordered interior points along
the route. A neutral orange ring marks the start of the route.

The decided quantity is *which colored waypoint is met first when walking along
the route from the orange ringed start*. This depends on following the connected
route in order, so every interior portion of the route from start to the first
waypoint is decisive: erasing any such portion breaks the reachability that
determines the first checkpoint. That decisive support is one connected thin
route contained within one medium-sized region (regional + pathlike).
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route_first_waypoint-0.1.0"

WIDTH = 512
HEIGHT = 512
BG = (255, 255, 255)
PATH_COLOR = (28, 38, 92)        # dark navy route
RING_COLOR = (230, 120, 20)      # orange start ring
RING_RADIUS = 22
RING_WIDTH = 6
STROKE = 16                       # thick route stroke (px)
CP_RADIUS = 13                    # waypoint disk radius (px)
CP_COLORS = {
    "red": (220, 50, 50),
    "green": (45, 170, 70),
    "blue": (40, 95, 220),
}

MIN_STEP = 42.0
MAX_STEP = 120.0
QUARANTINE_MARGIN = 26.0


def _polyline_length(points) -> float:
    total = 0.0
    for i in range(len(points) - 1):
        total += math.dist(points[i], points[i + 1])
    return total


def _point_at_fraction(points, fraction: float):
    """Return the coordinates of a point a given fraction along the polyline."""
    lengths = [
        math.dist(points[i], points[i + 1]) for i in range(len(points) - 1)
    ]
    total = sum(lengths)
    target = fraction * total
    acc = 0.0
    for i, length in enumerate(lengths):
        if acc + length >= target - 1e-9:
            t = (target - acc) / length if length > 1e-9 else 0.0
            return (
                points[i][0] + t * (points[i + 1][0] - points[i][0]),
                points[i][1] + t * (points[i + 1][1] - points[i][1]),
            )
        acc += length
    return points[-1]


def _min_nonadjacent_gap(points) -> float:
    """Minimum perpendicular separation between non-adjacent parallel runs.

    A monotone staircase with positive axis-aligned steps keeps consecutive
    parallel runs apart by at least the intervening step length. This is the
    verifiability margin: small gaps mean bends/waypoints could visually merge.
    """
    best = float("inf")
    for i in range(len(points) - 1):
        best = min(best, math.dist(points[i], points[i + 1]))
    return best


def _sample_route(rng: random.Random):
    """Build a monotone orthogonal staircase confined to a medium central box."""
    x0, y0, x1, y1 = 70, 64, 446, 448
    bends = rng.randint(3, 5)
    n = bends + 1  # number of segments
    sx = rng.uniform(x0 + 26, x0 + 66)
    sy = rng.uniform(y1 - 90, y1 - 40)
    pts = [(sx, sy)]
    cx, cy = sx, sy
    for k in range(n):
        length = rng.uniform(MIN_STEP, MAX_STEP)
        if k % 2 == 0:  # horizontal run, rightward
            nx = min(x1 - 24, cx + length)
            pts.append((nx, cy))
            cx, cy = nx, cy
        else:  # vertical run, upward
            ny = max(y0 + 24, cy - length)
            pts.append((cx, ny))
            cx, cy = cx, ny
    # ensure strictly monotone increase in x (no degenerate horizontal)
    return pts


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    points = _sample_route(rng)
    total = _polyline_length(points)
    # ordered along-path fractions for the three waypoints.
    fracs = [
        rng.uniform(0.66, 0.73),
        rng.uniform(0.82, 0.89),
        rng.uniform(0.955, 1.0),
    ]
    colors = list(CP_COLORS.keys())
    rng.shuffle(colors)
    waypoints = [{"color": colors[i], "frac": fracs[i]} for i in range(3)]
    return {
        "points": [[round(x, 2), round(y, 2)] for x, y in points],
        "waypoints": waypoints,
        "total_length": round(total, 2),
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    points = [(x, y) for x, y in scene["points"]]
    # route
    draw.line(points, fill=PATH_COLOR, width=STROKE, joint="curve")
    # start ring at route start
    sx, sy = points[0]
    ring = [(sx - RING_RADIUS, sy - RING_RADIUS), (sx + RING_RADIUS, sy + RING_RADIUS)]
    draw.ellipse(ring, outline=RING_COLOR, width=RING_WIDTH)
    # waypoint disks placed along the route (drawn on top of the route)
    total = float(scene["total_length"])
    for wp in scene["waypoints"]:
        cx, cy = _point_at_fraction(points, min(1.0, wp["frac"]))
        color = CP_COLORS[wp["color"]]
        box = [(cx - CP_RADIUS, cy - CP_RADIUS), (cx + CP_RADIUS, cy + CP_RADIUS)]
        draw.ellipse(box, fill=color)
    return img


def _first_waypoint(scene: dict) -> dict:
    return min(scene["waypoints"], key=lambda wp: wp["frac"])


def analytic_gold(scene: dict) -> str:
    return _first_waypoint(scene)["color"]


def margin(scene: dict) -> float:
    return _min_nonadjacent_gap(scene["points"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    return iter(())


if __name__ == "__main__":
    for s in range(6):
        sc = sample_scene(101 + s)
        img = render(sc)
        img.save(f"/workspace/scratch/preview_{s}.png")
        print(s, "gold=", analytic_gold(sc), "margin=", round(margin(sc), 1),
              "quarantined=", is_quarantined(sc), "pts=", len(sc["points"]))
