"""Orientation-defined texture patch: which coloured dot lies inside it?

Latent scene z = {canvas, palette, grid_seed, spacing, jitter, stroke geometry,
base_angle_deg, patch (cx, cy, r), markers}. The image I = R(z) is a dense,
uniform field of identical short grey strokes. Every stroke outside the patch
carries the same tilt ``base_angle_deg``; every stroke whose centre falls inside
the patch disk is turned by exactly 90 degrees. The patch has no outline, no
shading and no density change: stroke count per unit area, stroke length, stroke
width, ink colour and total ink are identical inside and out, so the region
exists only as a second-order (orientation) statistic of the texture.

Four coloured dots are painted on top; exactly one of them lies inside the
patch. The declared rule (stated verbatim in every prompt family) is

    answer = colour of the unique marker m with ||m - patch_centre|| < patch_r

and the patch itself is recoverable only by segmenting the orientation field,
so the answer is a property of GLOBAL texture organisation, not of any local
mark. Every scene contains the same four colours, the same four dot sizes and
the same stroke field, so colour, ink, count and size carry no information.

Margin: m = min over the four markers of | ||marker - centre|| - r | in pixels,
i.e. the distance to the invisible patch boundary. m -> 0 is the ambiguity
event (a marker sitting on the rim of the orientation-defined region).
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "texture_orientation_patch-0.1.0"

CANVAS = 640
SPACING = 30  # texture grid pitch, identical inside and outside the patch
JITTER = 4.0  # positional jitter so the field is not a visible lattice
STROKE_LEN = 15
STROKE_WIDTH = 3
GRID_MARGIN = 26

COLOR_NAMES = ("red", "green", "blue", "orange")
MARKER_RGB = {
    "red": "#d2192b",
    "green": "#1f9b3c",
    "blue": "#2050d8",
    "orange": "#f08a12",
}
MARKER_R = 11
MARKER_HALO = 16  # background-coloured ring: identical for all four markers

BG_COLORS = ("#f6f4ee", "#ffffff", "#f1f4f8")
STROKE_COLORS = ("#3c3c3c", "#444444", "#38383e")

PATCH_R_MIN, PATCH_R_MAX = 96, 132
PATCH_CENTRE_LO, PATCH_CENTRE_HI = 190, 450

OUTSIDE_CLEAR = 42.0  # non-answer markers stay this far outside the rim
INSIDE_CLEAR_LO, INSIDE_CLEAR_HI = 30.0, 0.62  # normal scenes: r*0.62 >= gap >= 30
NEAR_LO, NEAR_HI = 5.0, 13.0  # deliberate near-rim scenes
MARKER_SEP = 62.0  # markers never crowd or touch each other
BOUND_LO, BOUND_HI = 42, CANVAS - 42

QUARANTINE_PX = 14.0  # |d - r| < 14 px  =>  quarantined
BOUNDARY_EVERY = 6  # one in six sampled scenes targets the rim


def _strokes(scene: dict) -> list[tuple[float, float, float]]:
    """Rebuild the deterministic stroke field: (x, y, angle_deg) per stroke."""
    rng = random.Random(scene["grid_seed"])
    spacing = scene["spacing"]
    jitter = scene["jitter"]
    base = float(scene["base_angle_deg"])
    cx, cy, radius = scene["patch_cx"], scene["patch_cy"], scene["patch_r"]
    out: list[tuple[float, float, float]] = []
    coords = list(range(GRID_MARGIN, scene["width"] - GRID_MARGIN + 1, spacing))
    for gy in coords:
        for gx in coords:
            x = gx + rng.uniform(-jitter, jitter)
            y = gy + rng.uniform(-jitter, jitter)
            inside = math.hypot(x - cx, y - cy) < radius
            out.append((x, y, base + 90.0 if inside else base))
    return out


def _signed_gaps(scene: dict) -> list[float]:
    cx, cy, radius = scene["patch_cx"], scene["patch_cy"], scene["patch_r"]
    return [math.hypot(mx - cx, my - cy) - radius for mx, my, _ in scene["markers"]]


def _inside_index(scene: dict) -> int:
    inside = [i for i, gap in enumerate(_signed_gaps(scene)) if gap < 0.0]
    if len(inside) != 1:
        raise ValueError(f"scene must have exactly one interior marker, got {len(inside)}")
    return inside[0]


def _try_markers(rng: random.Random, scene: dict, answer: str, boundary: bool):
    cx, cy, radius = scene["patch_cx"], scene["patch_cy"], scene["patch_r"]
    if boundary:
        gap = rng.uniform(NEAR_LO, NEAR_HI)
    else:
        gap = rng.uniform(INSIDE_CLEAR_LO, radius * INSIDE_CLEAR_HI)
    placed: list[tuple[float, float, str]] = []
    for _ in range(400):
        angle = rng.uniform(0.0, 2.0 * math.pi)
        x, y = cx + (radius - gap) * math.cos(angle), cy + (radius - gap) * math.sin(angle)
        if BOUND_LO <= x <= BOUND_HI and BOUND_LO <= y <= BOUND_HI:
            placed.append((round(x, 4), round(y, 4), answer))
            break
    if not placed:
        return None
    for name in COLOR_NAMES:
        if name == answer:
            continue
        seated = False
        for _ in range(500):
            x = rng.uniform(BOUND_LO, BOUND_HI)
            y = rng.uniform(BOUND_LO, BOUND_HI)
            if math.hypot(x - cx, y - cy) - radius < OUTSIDE_CLEAR:
                continue
            if any(math.hypot(x - px, y - py) < MARKER_SEP for px, py, _ in placed):
                continue
            placed.append((round(x, 4), round(y, 4), name))
            seated = True
            break
        if not seated:
            return None
    rng.shuffle(placed)
    return placed


def sample_scene(seed: int) -> dict:
    """Deterministically sample one latent scene."""
    rng = random.Random(seed * 6151 + 29)
    boundary = (seed % BOUNDARY_EVERY) == 2
    answer = COLOR_NAMES[seed % len(COLOR_NAMES)]
    for _ in range(240):
        scene = {
            "width": CANVAS,
            "height": CANVAS,
            "bg_color": BG_COLORS[rng.randrange(len(BG_COLORS))],
            "stroke_color": STROKE_COLORS[rng.randrange(len(STROKE_COLORS))],
            "grid_seed": rng.randrange(1, 10_000_000),
            "spacing": SPACING,
            "jitter": JITTER,
            "stroke_len": STROKE_LEN,
            "stroke_width": STROKE_WIDTH,
            "base_angle_deg": round(rng.uniform(0.0, 180.0), 4),
            "patch_cx": round(rng.uniform(PATCH_CENTRE_LO, PATCH_CENTRE_HI), 4),
            "patch_cy": round(rng.uniform(PATCH_CENTRE_LO, PATCH_CENTRE_HI), 4),
            "patch_r": round(rng.uniform(PATCH_R_MIN, PATCH_R_MAX), 4),
            "markers": [],
        }
        markers = _try_markers(rng, scene, answer, boundary)
        if markers is None:
            continue
        scene["markers"] = markers
        try:
            if analytic_gold(scene) != answer:
                continue
        except ValueError:
            continue
        gap = margin(scene)
        if boundary and gap >= QUARANTINE_PX:
            continue
        if not boundary and gap < QUARANTINE_PX:
            continue
        return scene
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def render(scene: dict, supersample: int = 3) -> Image.Image:
    """I = R(z): the texture field first, then the four coloured markers."""
    width, height = scene["width"], scene["height"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), scene["bg_color"])
    draw = ImageDraw.Draw(img)
    half = scene["stroke_len"] / 2.0
    for x, y, angle in _strokes(scene):
        radians = math.radians(angle)
        dx, dy = half * math.cos(radians), half * math.sin(radians)
        draw.line(
            [((x - dx) * ss, (y - dy) * ss), ((x + dx) * ss, (y + dy) * ss)],
            fill=scene["stroke_color"],
            width=scene["stroke_width"] * ss,
        )
    for mx, my, name in scene["markers"]:
        draw.ellipse(
            [(mx - MARKER_HALO) * ss, (my - MARKER_HALO) * ss,
             (mx + MARKER_HALO) * ss, (my + MARKER_HALO) * ss],
            fill=scene["bg_color"],
        )
        draw.ellipse(
            [(mx - MARKER_R) * ss, (my - MARKER_R) * ss,
             (mx + MARKER_R) * ss, (my + MARKER_R) * ss],
            fill=MARKER_RGB[name],
        )
    return img.resize((width, height), Image.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Colour name of the unique marker inside the orientation-defined patch."""
    return scene["markers"][_inside_index(scene)][2]


def margin(scene: dict) -> float:
    """m = min_i | ||marker_i - centre|| - r |: distance to the invisible rim."""
    return round(min(abs(gap) for gap in _signed_gaps(scene)), 6)


def is_quarantined(scene: dict) -> bool:
    """A marker within 14 px of the patch rim is not resolvable at this scale."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict):
    """Declared pixel-identical latent aliases.

    The marker list is an unordered set: markers are pairwise separated by at
    least 62 px, which exceeds twice the 16 px halo radius, so no two marker
    stamps overlap and any permutation of ``markers`` paints exactly the same
    pixels. The gold is the colour of the interior marker, which is a property
    of the set, so it is permutation-invariant by construction.
    """
    markers = scene["markers"]
    rotated = dict(scene, markers=markers[1:] + markers[:1])
    reversed_scene = dict(scene, markers=list(reversed(markers)))
    return [("marker_order_rotation", rotated), ("marker_order_reversal", reversed_scene)]
