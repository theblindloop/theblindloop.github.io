"""Pixel-only inverse arm for pair_sum_reference.

Receives a rendered PNG and nothing else. It segments the five saturated bar
colors, recovers the baseline from the bar feet, recovers the unit pitch from
the long gray gridline rows, reads the dashed black target level, converts every
extent into whole gridline units, and reports the unique pair of colors whose
units add up to the target level. It abstains on any structural surprise.

It imports no renderer, prompt, generator, verifier or gold code and never sees
scene data.
"""

from __future__ import annotations

ORACLE_VERSION = "pair_sum_reference-oracle-1.0.0"
ABSTAIN = "abstain"

_PALETTE = {
    (220, 40, 40): "red",
    (240, 140, 20): "orange",
    (30, 150, 60): "green",
    (40, 80, 220): "blue",
    (140, 60, 190): "purple",
}
_GRID = (170, 170, 170)
_BLACK = (0, 0, 0)
_WHITE = (255, 255, 255)

_MIN_GRID_RUN = 120
_MIN_DASH_RUN = 60
_UNIT_TOLERANCE = 0.12


def decision_from_image(image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    data = list(rgb.getdata())

    blobs: dict = {}
    gray_rows = [0] * height
    dark_rows = [0] * height
    for index, value in enumerate(data):
        if value == _WHITE:
            continue
        row = index // width
        if value == _GRID:
            gray_rows[row] += 1
            continue
        if value == _BLACK:
            dark_rows[row] += 1
            continue
        name = _PALETTE.get(value)
        if name is None:
            continue
        column = index - row * width
        record = blobs.get(name)
        if record is None:
            blobs[name] = [1, column, column, row, row]
        else:
            record[0] += 1
            if column < record[1]:
                record[1] = column
            if column > record[2]:
                record[2] = column
            if row < record[3]:
                record[3] = row
            if row > record[4]:
                record[4] = row

    if len(blobs) != len(_PALETTE):
        return ABSTAIN
    for record in blobs.values():
        area = (record[2] - record[1] + 1) * (record[4] - record[3] + 1)
        if record[0] != area or area < 200:
            return ABSTAIN

    feet = {record[4] for record in blobs.values()}
    if len(feet) != 1:
        return ABSTAIN
    baseline = feet.pop() + 1

    gridlines = [row for row, count in enumerate(gray_rows) if count >= _MIN_GRID_RUN]
    if len(gridlines) < 4:
        return ABSTAIN
    pitches = {gridlines[i + 1] - gridlines[i] for i in range(len(gridlines) - 1)}
    if len(pitches) != 1:
        return ABSTAIN
    pitch = pitches.pop()
    if pitch < 8 or baseline not in gridlines:
        return ABSTAIN

    dashes = [row for row, count in enumerate(dark_rows) if count >= _MIN_DASH_RUN]
    if not dashes or len(dashes) > 4:
        return ABSTAIN
    if dashes[-1] - dashes[0] != len(dashes) - 1:
        return ABSTAIN
    level = (baseline - dashes[0]) / pitch
    if abs(level - round(level)) > _UNIT_TOLERANCE or round(level) < 2:
        return ABSTAIN
    target = int(round(level))

    units: dict = {}
    for name, record in blobs.items():
        value = (baseline - record[3]) / pitch
        if abs(value - round(value)) > _UNIT_TOLERANCE or round(value) < 1:
            return ABSTAIN
        units[name] = int(round(value))

    names = sorted(units)
    hits = [
        (a, b)
        for i, a in enumerate(names)
        for b in names[i + 1 :]
        if units[a] + units[b] == target
    ]
    if len(hits) != 1:
        return ABSTAIN
    a, b = hits[0]
    if units[a] == units[b]:
        return ABSTAIN
    tall, short = (a, b) if units[a] > units[b] else (b, a)
    return f"{tall}+{short}"
