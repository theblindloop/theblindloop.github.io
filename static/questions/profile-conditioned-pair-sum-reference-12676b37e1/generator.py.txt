"""Scene sampler, renderer and analytic arm for pair_sum_reference.

Five colored bars stand on one baseline over gray gridlines that are exactly
one unit apart. A dashed black rule marks a target level in those same units.
Exactly one pair of bars has values adding up to the target level; the gold
label names that pair, taller bar first.

Every drawn quantity is an exact integer multiple of the gridline pitch, so the
raster is an injective image of the label inputs and the pixel arm can recover
the same integers the analytic arm uses.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pair_sum_reference-1.0.0"

WIDTH, HEIGHT = 560, 380
N_BARS = 5
COLORS = ("red", "orange", "green", "blue", "purple")
RGB = {
    "red": (220, 40, 40),
    "orange": (240, 140, 20),
    "green": (30, 150, 60),
    "blue": (40, 80, 220),
    "purple": (140, 60, 190),
}
GRID_RGB = (170, 170, 170)
INK_RGB = (0, 0, 0)

BASE_Y = 330
SLOT_X = 46
SLOT_STEP = 96
PLOT_LEFT = 24
PLOT_RIGHT = 536
TOP_LIMIT = 16

MAX_TARGET_UNITS = 24
QUARANTINE_UNITS = 2.0

_PAIRS = tuple((i, j) for i in range(N_BARS) for j in range(i + 1, N_BARS))


def sample_scene(seed: int) -> dict:
    """Sample one scene. The answer pair's height ranks are drawn uniformly, so
    "the two tallest bars" and "the tallest bar is in the pair" are deliberately
    weak cues."""
    rng = random.Random(seed)
    # The answer pair's height ranks are drawn once, before the rejection loop,
    # so admissibility filtering cannot bias the answer towards extreme pairs.
    low, high = _PAIRS[rng.randrange(len(_PAIRS))]
    # The quarantine band is an explicit design draw, not a by-product of
    # rejection: roughly one scene in eight is sampled at a one-unit margin.
    want_quarantine = rng.random() < 0.12
    heights = None
    target = 0
    # The first pass honours the quarantine draw; the relaxed pass accepts any
    # nonzero margin so that every rank pair stays reachable.
    for relaxed in (False, True):
        for _ in range(20000):
            trial = sorted(rng.sample(range(2, 16), N_BARS))
            value = trial[low] + trial[high]
            if value > MAX_TARGET_UNITS or value in trial:
                # The dashed level must never coincide with a bar top, so the
                # rule is never a touching case and never masks a bar's top.
                continue
            closest = min(
                abs(trial[a] + trial[b] - value)
                for (a, b) in _PAIRS
                if (a, b) != (low, high)
            )
            if closest == 0:
                continue
            if not relaxed:
                if want_quarantine and closest != 1:
                    continue
                if not want_quarantine and closest < 2:
                    continue
            heights, target = trial, value
            break
        if heights is not None:
            break
    if heights is None:  # pragma: no cover - unreachable for the seeds in use
        raise RuntimeError("scene sampler failed to find an admissible scene")

    order = list(range(N_BARS))
    rng.shuffle(order)
    slot_colors = list(COLORS)
    rng.shuffle(slot_colors)
    return {
        "heights_units": [heights[k] for k in order],
        "slot_colors": slot_colors,
        "target_units": target,
        "widths_px": [rng.randrange(44, 63, 2) for _ in range(N_BARS)],
        "px_per_unit": rng.choice([10, 11, 12]),
        "x_jitter": [rng.randrange(0, 11) for _ in range(N_BARS)],
        "offset": [rng.randrange(-8, 9), rng.randrange(-6, 7)],
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    dx, dy = scene["offset"]
    pitch = int(scene["px_per_unit"])
    base = BASE_Y + dy
    left, right = PLOT_LEFT + dx, PLOT_RIGHT + dx

    step = 0
    while base - step * pitch >= TOP_LIMIT:
        y = base - step * pitch
        draw.rectangle([left, y, right, y], fill=GRID_RGB)
        step += 1

    # The dashed target rule is drawn before the bars, so bars occlude it and
    # every bar top stays a clean color edge whatever the target level is.
    y_ref = base - int(scene["target_units"]) * pitch
    x = left
    while x <= right:
        draw.rectangle([x, y_ref, min(x + 13, right), y_ref + 1], fill=INK_RGB)
        x += 24

    for slot in range(N_BARS):
        x = SLOT_X + dx + slot * SLOT_STEP + scene["x_jitter"][slot]
        width = scene["widths_px"][slot]
        tall = scene["heights_units"][slot] * pitch
        draw.rectangle(
            [x, base - tall, x + width - 1, base - 1],
            fill=RGB[scene["slot_colors"][slot]],
        )
    return image


def _answer_pair(scene: dict) -> tuple[int, int]:
    heights = scene["heights_units"]
    target = int(scene["target_units"])
    hits = [(a, b) for (a, b) in _PAIRS if heights[a] + heights[b] == target]
    if len(hits) != 1:
        raise ValueError("scene does not carry exactly one qualifying pair")
    a, b = hits[0]
    if heights[a] == heights[b]:
        raise ValueError("qualifying pair has no taller bar")
    return (a, b) if heights[a] > heights[b] else (b, a)


def analytic_gold(scene: dict) -> str:
    tall, short = _answer_pair(scene)
    return f"{scene['slot_colors'][tall]}+{scene['slot_colors'][short]}"


def margin(scene: dict) -> float:
    """Runner-up decision margin: how far, in gridline units, the closest rival
    pair sits from the drawn target level."""
    heights = scene["heights_units"]
    target = int(scene["target_units"])
    tall, short = _answer_pair(scene)
    answer = {tall, short}
    return float(
        min(
            abs(heights[a] + heights[b] - target)
            for (a, b) in _PAIRS
            if {a, b} != answer
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_UNITS


def latent_symmetries(scene: dict) -> list:
    """No latent alias exists: see candidate.json latent_alias.justification."""
    return []


def cue_congruence(scene: dict) -> dict:
    """Diagnostics for cheap shortcuts that could mimic the answer."""
    heights = scene["heights_units"]
    tall, short = _answer_pair(scene)
    ranked = sorted(range(N_BARS), key=lambda k: heights[k])
    return {
        "two_tallest": {tall, short} == {ranked[-1], ranked[-2]},
        "tallest_included": ranked[-1] in (tall, short),
        "adjacent_slots": abs(tall - short) == 1,
        "leftmost_included": 0 in (tall, short),
    }


def totals_units(scene: dict) -> dict:
    tall, short = _answer_pair(scene)
    return {
        "target_units": int(scene["target_units"]),
        "tall_slot": tall,
        "short_slot": short,
        "heights_units": list(scene["heights_units"]),
    }
