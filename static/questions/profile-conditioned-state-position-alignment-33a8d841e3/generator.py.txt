"""Deterministic raster world for state-to-position alignment."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state_position_alignment-0.1.0"

PANEL_COUNT = 5
IDENTITY_COUNT = 3
STATE_COUNT = 3
LABELS = ("A", "B", "C")
STATE_NAMES = ("LOW", "MID", "HIGH")
LANE_NAMES = ("TOP", "MIDDLE", "BOTTOM")
CANVAS = (1840, 760)

BACKGROUND = (247, 249, 250)
INK = (28, 47, 63)
MUTED = (85, 103, 116)
OUTLINE = (151, 168, 178)
CARD = (255, 255, 255)
STATE_OUTLINE = (38, 58, 72)
IDENTITY_INK = (30, 58, 91)
STATE_COLORS = (
    (63, 135, 216),   # LOW
    (238, 160, 54),   # MID
    (211, 76, 86),    # HIGH
)
PANEL_TINT_PALETTE = (
    (237, 245, 249),
    (244, 241, 249),
    (240, 248, 241),
    (250, 245, 235),
    (240, 247, 247),
    (249, 240, 244),
)

PANEL_X0 = 24
PANEL_TOP = 170
PANEL_WIDTH = 340
PANEL_HEIGHT = 550
PANEL_GAP = 18
LANE_Y_OFFSETS = (112, 272, 432)
CARD_X0 = 28
CARD_WIDTH = 284
CARD_HEIGHT = 104
IDENTITY_X_OFFSET = 94
STATE_X_OFFSET = 226
SHAPE_RADIUS = 25
STATE_RADIUS = 20


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ) if bold else (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
    )
    for name in names:
        if Path(name).is_file():
            return ImageFont.truetype(name, size)
    return ImageFont.load_default()


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    text: str,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (center[0] - (box[2] - box[0]) / 2, center[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _validate_rgb(value: object, name: str) -> tuple[int, int, int]:
    if not isinstance(value, list) or len(value) != 3:
        raise ValueError(f"{name} must be an RGB list")
    channels = tuple(int(channel) for channel in value)
    if any(channel not in range(256) for channel in channels):
        raise ValueError(f"{name} has an invalid channel")
    return channels


def _core_tables(scene: dict) -> tuple[list[list[int]], list[list[int]]]:
    states_raw = scene["panel_states"]
    positions_raw = scene["panel_positions"]
    if not isinstance(states_raw, list) or len(states_raw) != PANEL_COUNT:
        raise ValueError("panel_states must contain five panels")
    if not isinstance(positions_raw, list) or len(positions_raw) != PANEL_COUNT:
        raise ValueError("panel_positions must contain five panels")
    states: list[list[int]] = []
    positions: list[list[int]] = []
    for state_panel, position_panel in zip(states_raw, positions_raw):
        if not isinstance(state_panel, list) or len(state_panel) != IDENTITY_COUNT:
            raise ValueError("each state panel must contain three identities")
        if not isinstance(position_panel, list) or len(position_panel) != IDENTITY_COUNT:
            raise ValueError("each position panel must contain three identities")
        state_values = [int(value) for value in state_panel]
        position_values = [int(value) for value in position_panel]
        if any(value not in range(STATE_COUNT) for value in state_values):
            raise ValueError("states must use LOW=0, MID=1, HIGH=2")
        if sorted(position_values) != list(range(IDENTITY_COUNT)):
            raise ValueError("each position panel must be a lane permutation")
        states.append(state_values)
        positions.append(position_values)
    return states, positions


def _alignment_scores(scene: dict) -> list[int]:
    states, positions = _core_tables(scene)
    return [
        sum(states[panel][label] == STATE_COUNT - 1 - positions[panel][label] for panel in range(PANEL_COUNT))
        for label in range(IDENTITY_COUNT)
    ]


def _shape(draw: ImageDraw.ImageDraw, center: tuple[float, float], label: int, radius: int) -> None:
    cx, cy = center
    if label == 0:
        draw.polygon(
            ((cx, cy - radius), (cx - radius, cy + radius), (cx + radius, cy + radius)),
            fill=IDENTITY_INK,
        )
    elif label == 1:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=IDENTITY_INK)
    else:
        draw.rectangle((cx - radius, cy - radius, cx + radius, cy + radius), fill=IDENTITY_INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, font: ImageFont.ImageFont, small: ImageFont.ImageFont) -> None:
    draw.text((28, 91), "VISIBLE STATE KEY", font=font, fill=INK)
    for state, x in enumerate((730, 965, 1200)):
        draw.ellipse(
            (x, 83, x + 34, 117),
            fill=STATE_COLORS[state],
            outline=STATE_OUTLINE,
            width=2,
        )
        _centered_text(draw, (x + 80, 100), STATE_NAMES[state], small, INK)
    draw.text((1450, 91), "VERTICAL RANK", font=font, fill=INK)
    draw.text((1450, 115), "HIGH = top   MID = middle   LOW = bottom", font=small, fill=MUTED)


def _draw_identity_key(draw: ImageDraw.ImageDraw, font: ImageFont.ImageFont, small: ImageFont.ImageFont) -> None:
    draw.text((28, 18), "STATE / POSITION ALIGNMENT", font=font, fill=INK)
    draw.text((28, 54), "Count each label's panels where its state badge matches its vertical slot.", font=small, fill=MUTED)
    draw.text((520, 20), "VISIBLE IDENTITY KEY", font=font, fill=INK)
    for label, x in enumerate((760, 850, 940)):
        _shape(draw, (x, 35), label, 13)
        _centered_text(draw, (x + 28, 35), LABELS[label], small, INK)


def render(scene: dict) -> Image.Image:
    states, positions = _core_tables(scene)
    tints_raw = scene["panel_tints"]
    if not isinstance(tints_raw, list) or len(tints_raw) != PANEL_COUNT:
        raise ValueError("panel_tints must contain five RGB colors")
    tints = [_validate_rgb(color, "panel tint") for color in tints_raw]
    marker_scale = float(scene["marker_scale"])
    if not 0.85 <= marker_scale <= 1.15:
        raise ValueError("marker_scale is outside the supported visible range")

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(26, bold=True)
    section_font = _font(15, bold=True)
    body_font = _font(13)
    label_font = _font(17, bold=True)
    tiny_font = _font(11)
    _draw_identity_key(draw, title_font, body_font)
    _draw_state_key(draw, section_font, body_font)

    for panel in range(PANEL_COUNT - 1):
        left = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP) + PANEL_WIDTH + 5
        right = PANEL_X0 + (panel + 1) * (PANEL_WIDTH + PANEL_GAP) - 5
        y = PANEL_TOP - 23
        draw.line((left, y, right - 10, y), fill=OUTLINE, width=2)
        draw.polygon(((right - 10, y - 5), (right, y), (right - 10, y + 5)), fill=OUTLINE)

    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            radius=12,
            fill=tints[panel],
            outline=OUTLINE,
            width=2,
        )
        draw.text((panel_x + 18, PANEL_TOP + 15), f"P{panel + 1}", font=section_font, fill=INK)
        draw.text((panel_x + 100, PANEL_TOP + 18), "IDENTITY / STATE", font=tiny_font, fill=MUTED)
        draw.line(
            (panel_x + 16, PANEL_TOP + 52, panel_x + PANEL_WIDTH - 16, PANEL_TOP + 52),
            fill=OUTLINE,
            width=1,
        )
        label_at_lane = [0] * IDENTITY_COUNT
        for label, lane in enumerate(positions[panel]):
            label_at_lane[lane] = label
        for lane, offset in enumerate(LANE_Y_OFFSETS):
            card_y = PANEL_TOP + offset
            draw.rounded_rectangle(
                (panel_x + CARD_X0, card_y - CARD_HEIGHT // 2, panel_x + CARD_X0 + CARD_WIDTH, card_y + CARD_HEIGHT // 2),
                radius=10,
                fill=CARD,
                outline=(193, 204, 210),
                width=1,
            )
            draw.text((panel_x + CARD_X0 + 10, card_y - 40), LANE_NAMES[lane], font=tiny_font, fill=MUTED)
            label = label_at_lane[lane]
            radius = max(20, round(SHAPE_RADIUS * marker_scale))
            _shape(draw, (panel_x + CARD_X0 + IDENTITY_X_OFFSET, card_y + 7), label, radius)
            _centered_text(draw, (panel_x + CARD_X0 + 132, card_y + 7), LABELS[label], label_font, INK)
            state = states[panel][label]
            state_radius = max(16, round(STATE_RADIUS * marker_scale))
            state_x = panel_x + CARD_X0 + STATE_X_OFFSET
            state_y = card_y + 7
            draw.ellipse(
                (state_x - state_radius, state_y - state_radius, state_x + state_radius, state_y + state_radius),
                fill=STATE_COLORS[state],
                outline=STATE_OUTLINE,
                width=2,
            )
            _centered_text(draw, (panel_x + CARD_X0 + 256, card_y + 7), STATE_NAMES[state], tiny_font, MUTED)
            draw.line(
                (panel_x + CARD_X0 + 12, card_y + 37, panel_x + CARD_X0 + CARD_WIDTH - 12, card_y + 37),
                fill=(221, 228, 232),
                width=1,
            )
            draw.text((panel_x + CARD_X0 + 12, card_y + 43), "badge agrees with slot?", font=tiny_font, fill=MUTED)
    return image


def _random_state_for_match(rng: random.Random, expected: int, probability: float) -> int:
    if rng.random() < probability:
        return expected
    choices = [value for value in range(STATE_COUNT) if value != expected]
    return rng.choice(choices)


def _sample_tables(rng: random.Random, winner: int, force_quarantine: bool) -> tuple[list[list[int]], list[list[int]]]:
    for _ in range(5000):
        positions = [rng.sample(range(IDENTITY_COUNT), IDENTITY_COUNT) for _ in range(PANEL_COUNT)]
        states: list[list[int]] = []
        for panel in range(PANEL_COUNT):
            state_panel = []
            for label in range(IDENTITY_COUNT):
                expected = STATE_COUNT - 1 - positions[panel][label]
                probability = 0.80 if label == winner else 0.27
                state_panel.append(_random_state_for_match(rng, expected, probability))
            states.append(state_panel)
        scores = [
            sum(states[panel][label] == STATE_COUNT - 1 - positions[panel][label] for panel in range(PANEL_COUNT))
            for label in range(IDENTITY_COUNT)
        ]
        ordered = sorted(scores, reverse=True)
        gap = ordered[0] - ordered[1]
        unique = scores.count(ordered[0]) == 1
        if force_quarantine:
            if not (not unique or gap == 1):
                continue
        elif not (unique and gap >= 2 and scores[winner] == ordered[0]):
            continue
        return states, positions
    raise RuntimeError("could not sample a state-position table with the requested boundary")


def sample_scene(seed: int, force_quarantine: bool = False) -> dict:
    """Sample a deterministic scene; default scenes have a two-count lead."""
    seed = int(seed)
    logic_rng = random.Random(seed * 1_000_003 + 17)
    style_rng = random.Random((seed + 104_729) * 1_000_033 + 31)
    winner = seed % IDENTITY_COUNT
    states, positions = _sample_tables(logic_rng, winner, bool(force_quarantine))
    scene = {
        "panel_states": states,
        "panel_positions": positions,
        "panel_tints": [list(style_rng.choice(PANEL_TINT_PALETTE)) for _ in range(PANEL_COUNT)],
        "marker_scale": style_rng.choice((0.92, 1.0, 1.08)),
    }
    if not force_quarantine and analytic_gold(scene) != LABELS[winner]:
        raise AssertionError("sampler failed to create the requested alignment winner")
    return scene


def analytic_gold(scene: dict) -> str:
    scores = _alignment_scores(scene)
    best = max(scores)
    winner = scores.index(best)
    return LABELS[winner]


def margin(scene: dict) -> float:
    ordered = sorted(_alignment_scores(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _alignment_scores(scene)
        ordered = sorted(scores, reverse=True)
        return scores.count(ordered[0]) != 1 or ordered[0] - ordered[1] < 2
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant aliases are claimed for the fully keyed scene."""
    return []
