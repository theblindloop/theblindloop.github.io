"""Independent image-only inverse for the state-position alignment world.

This module deliberately repeats only the public canvas contract.  It does not
import the renderer, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_CANVAS = (1840, 760)
_PANEL_COUNT = 5
_IDENTITY_COUNT = 3
_STATE_COUNT = 3
_LABELS = ("A", "B", "C")
_STATE_COLORS = np.array(
    (
        (63, 135, 216),
        (238, 160, 54),
        (211, 76, 86),
    ),
    dtype=np.int16,
)
_IDENTITY_INK = np.array((30, 58, 91), dtype=np.int16)
_PANEL_X0 = 24
_PANEL_TOP = 170
_PANEL_WIDTH = 340
_PANEL_GAP = 18
_LANE_Y_OFFSETS = (112, 272, 432)
_CARD_X0 = 28
_IDENTITY_X_OFFSET = 94
_STATE_X_OFFSET = 226


def _nearest_state(pixel: np.ndarray) -> int:
    distances = np.linalg.norm(_STATE_COLORS - pixel.astype(np.int16), axis=1)
    state = int(np.argmin(distances))
    if float(distances[state]) > 18.0:
        raise ValueError("state badge is missing or unreadable")
    return state


def _shape_label(patch: np.ndarray) -> int:
    distances = np.linalg.norm(patch.astype(np.int16) - _IDENTITY_INK, axis=2)
    mask = distances <= 10.0
    ys, xs = np.where(mask)
    if len(xs) < 180:
        raise ValueError("identity glyph is missing")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if width < 30 or height < 30 or width > 62 or height > 62:
        raise ValueError("identity glyph bounds are ambiguous")
    fill_ratio = float(len(xs)) / float(width * height)
    if fill_ratio < 0.64:
        return 0  # triangle = A
    if fill_ratio < 0.90:
        return 1  # circle = B
    return 2  # square = C


def _read_panel(image: np.ndarray, panel: int) -> list[tuple[int, int, int]]:
    panel_x = _PANEL_X0 + panel * (_PANEL_WIDTH + _PANEL_GAP)
    records: list[tuple[int, int, int]] = []
    seen: set[int] = set()
    for lane, offset in enumerate(_LANE_Y_OFFSETS):
        y = _PANEL_TOP + offset + 7
        identity_x = panel_x + _CARD_X0 + _IDENTITY_X_OFFSET
        state_x = panel_x + _CARD_X0 + _STATE_X_OFFSET
        patch = image[y - 31 : y + 32, identity_x - 31 : identity_x + 32]
        if patch.shape[:2] != (63, 63):
            raise ValueError("panel geometry is incomplete")
        label = _shape_label(patch)
        if label in seen:
            raise ValueError("duplicate identity in panel")
        seen.add(label)
        state = _nearest_state(image[y, state_x])
        records.append((label, lane, state))
    if seen != {0, 1, 2}:
        raise ValueError("panel does not expose all three identities")
    return records


def decision_from_image(image: Image.Image) -> str:
    """Recover the top state-position alignment scorer from the raster alone."""
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if tuple(rgb.shape[1::-1]) != _CANVAS:
        raise ValueError(f"unexpected image size {tuple(rgb.shape[1::-1])}")
    scores = [0, 0, 0]
    for panel in range(_PANEL_COUNT):
        for label, lane, state in _read_panel(rgb, panel):
            expected = _STATE_COUNT - 1 - lane
            if state == expected:
                scores[label] += 1
    best = max(scores)
    return _LABELS[scores.index(best)]
