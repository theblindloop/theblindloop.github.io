"""Deterministic renderer for topology revealed only by a dot-lattice phase shift."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw


def _grow(rng: random.Random, size: int, bounds: tuple[int, int, int, int]) -> set[tuple[int, int]]:
    c0, c1, r0, r1 = bounds
    cells = {(rng.randint(c0, c1), rng.randint(r0, r1))}
    while len(cells) < size:
        frontier = []
        for c, r in cells:
            for dc, dr in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                q = (c + dc, r + dr)
                if c0 <= q[0] <= c1 and r0 <= q[1] <= r1 and q not in cells:
                    frontier.append(q)
        cells.add(rng.choice(frontier))
    return cells


def _component_count(cells: set[tuple[int, int]]) -> int:
    unseen = set(cells)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        todo = [start]
        while todo:
            c, r = todo.pop()
            for q in ((c + 1, r), (c - 1, r), (c, r + 1), (c, r - 1)):
                if q in unseen:
                    unseen.remove(q)
                    todo.append(q)
    return count


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_count = 1 if int(seed) % 2 == 0 else 2
    if answer_count == 1:
        cells = _grow(rng, 24, (3, 17, 2, 12))
    else:
        if rng.random() < 0.5:
            a = _grow(rng, 12, (2, 7, 2, 12))
            b = _grow(rng, 12, (13, 18, 2, 12))
        else:
            a = _grow(rng, 12, (2, 18, 2, 5))
            b = _grow(rng, 12, (2, 18, 9, 12))
        cells = a | b
    assert len(cells) == 24 and _component_count(cells) == answer_count
    dark_palettes = [
        ("#17324d", "#fbfaf4"),
        ("#37245b", "#fffaf0"),
        ("#123b35", "#fffdf7"),
    ]
    ink, background = dark_palettes[rng.randrange(len(dark_palettes))]
    return {
        "width": 672,
        "height": 504,
        "rows": 15,
        "cols": 21,
        "pitch": 28,
        "origin_x": rng.randint(48, 62),
        "origin_y": rng.randint(48, 62),
        "shift": rng.choice([7, 8, 9]),
        "radius": rng.choice([4, 5]),
        "ink": ink,
        "background": background,
        "shifted_cells": [[c, r] for c, r in sorted(cells)],
        "component_count": answer_count,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    pitch = int(scene["pitch"])
    ox = int(scene["origin_x"])
    oy = int(scene["origin_y"])
    shift = int(scene["shift"])
    radius = int(scene["radius"])
    ink = str(scene["ink"])
    background = str(scene["background"])
    shifted = {tuple(map(int, q)) for q in scene["shifted_cells"]}
    declared_count = int(scene["component_count"])
    if declared_count not in (1, 2) or _component_count(shifted) != declared_count:
        raise ValueError("component_count disagrees with shifted_cells")
    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)
    for r in range(rows):
        for c in range(cols):
            delta = shift if (c, r) in shifted else 0
            x = (ox + c * pitch + delta) * ss
            y = (oy + r * pitch + delta) * ss
            rr = radius * ss
            draw.ellipse((x - rr, y - rr, x + rr, y + rr), fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    value = int(scene["component_count"])
    if value not in (1, 2):
        raise ValueError("answer must be one or two")
    return "one" if value == 1 else "two"


def margin(scene: dict) -> float:
    # Pixel separation between the displaced phase and the conservative 4 px
    # ambiguity band used by the independent inverse arm.
    return float(scene["shift"]) - 4.0


def is_quarantined(scene: dict) -> bool:
    shifted = {tuple(map(int, q)) for q in scene["shifted_cells"]}
    return (
        int(scene["shift"]) <= 4
        or len(shifted) != 24
        or _component_count(shifted) != int(scene["component_count"])
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
