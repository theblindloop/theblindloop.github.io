"""Pixel-only inverse arm for a two-phase square dot lattice.

The implementation estimates dot centroids, the dominant square-lattice period
and its two coordinate phases, then counts four-neighbor components among dots
whose x and y residuals share the displaced phase.  It imports no world code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class AmbiguousImage(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        todo = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while todo:
            yy, xx = todo.pop()
            pts.append((yy, xx))
            for y2, x2 in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= y2 < h and 0 <= x2 < w and mask[y2, x2] and not seen[y2, x2]:
                    seen[y2, x2] = True
                    todo.append((y2, x2))
        if len(pts) >= 20:
            out.append(np.asarray(pts, dtype=float))
    return out


def _ink_centroids(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float32)
    # Ink is dark and the background is near white in every public palette.
    luminance = 0.2126 * rgb[:, :, 0] + 0.7152 * rgb[:, :, 1] + 0.0722 * rgb[:, :, 2]
    comps = _components(luminance < 150.0)
    if len(comps) < 100:
        raise AmbiguousImage("too few dot components to establish a lattice")
    centers = np.asarray([[p[:, 1].mean(), p[:, 0].mean()] for p in comps], dtype=float)
    return centers


def _wrap(values: np.ndarray, period: float) -> np.ndarray:
    return (values + period / 2.0) % period - period / 2.0


def _fit_axis(values: np.ndarray) -> tuple[float, float, int]:
    best = None
    for period in range(24, 33):
        for phase in range(period):
            residual = np.abs(_wrap(values - phase, float(period)))
            score = int(np.count_nonzero(residual < 1.25))
            loss = float(np.sort(residual)[: max(1, int(0.85 * len(residual)))].mean())
            key = (score, -loss)
            if best is None or key > best[0]:
                best = (key, float(period), float(phase), score)
    assert best is not None
    _, period, phase, score = best
    if score < 0.80 * len(values):
        raise AmbiguousImage("no dominant periodic phase")
    return period, phase, score


def _count_cell_components(cells: set[tuple[int, int]]) -> int:
    unseen = set(cells)
    count = 0
    while unseen:
        count += 1
        todo = [unseen.pop()]
        while todo:
            c, r = todo.pop()
            for q in ((c + 1, r), (c - 1, r), (c, r + 1), (c, r - 1)):
                if q in unseen:
                    unseen.remove(q)
                    todo.append(q)
    return count


def decision_from_image(image: Image.Image) -> str:
    centers = _ink_centroids(image)
    px, phx, _ = _fit_axis(centers[:, 0])
    py, phy, _ = _fit_axis(centers[:, 1])
    if abs(px - py) > 0.5:
        raise AmbiguousImage("the inferred texture is not a square lattice")
    pitch = (px + py) / 2.0
    rx = _wrap(centers[:, 0] - phx, pitch)
    ry = _wrap(centers[:, 1] - phy, pitch)
    shifted_mask = (np.abs(rx) > 4.0) & (np.abs(ry) > 4.0)
    shifted = centers[shifted_mask]
    if not 18 <= len(shifted) <= 30:
        raise AmbiguousImage(f"shifted phase has implausible support: {len(shifted)} dots")
    # Every shifted dot should share one coherent diagonal residual, well away
    # from both zero and the half-period wrap boundary.
    sx = float(np.median(rx[shifted_mask]))
    sy = float(np.median(ry[shifted_mask]))
    if sx * sy <= 0 or min(abs(sx), abs(sy)) <= 4.0:
        raise AmbiguousImage("shifted dots do not share a diagonal phase")
    if np.max(np.abs(rx[shifted_mask] - sx)) > 1.5 or np.max(np.abs(ry[shifted_mask] - sy)) > 1.5:
        raise AmbiguousImage("shifted phase is not coherent")
    cells = {
        (int(round((x - phx) / pitch)), int(round((y - phy) / pitch)))
        for x, y in shifted
    }
    if len(cells) != len(shifted):
        raise AmbiguousImage("multiple dots map to one lattice cell")
    count = _count_cell_components(cells)
    if count not in (1, 2):
        raise AmbiguousImage(f"recovered {count} shifted components")
    return "one" if count == 1 else "two"
