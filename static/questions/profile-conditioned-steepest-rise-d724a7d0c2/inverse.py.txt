"""Independent pixel-only inverse arm for the steepest-rise world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures which colored stroke rises
the most. It shares no geometry with renderer.render() and never imports the
renderer, prompts, generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "steepest_rise-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background).

    argmin leaves no unclassified antialias band: edge blends split at their
    ~50% point, which is dominated by the background, so the labeled stroke
    masks stay clean regardless of any anti-aliasing the renderer introduced.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = (
        arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    )
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _rise_from_mask(mask: np.ndarray) -> float:
    """Vertical rise of a stroke estimated by a least-squares line fit.

    Fitting the median centerline through the thick stroke recovers the slope.
    Because the horizontal width is identical for every stroke, the slope
    magnitude orders the rises identically (|slope| = |rise| / W).
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 10:
        return -1.0
    slope = np.polyfit(xs.astype(np.float64), ys.astype(np.float64), 1)[0]
    return float(abs(slope))


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the steepest (largest-|rise|) stroke from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_slope = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        slope = _rise_from_mask(mask)
        if slope > best_slope:
            best_slope = slope
            best_color = color
    return best_color
