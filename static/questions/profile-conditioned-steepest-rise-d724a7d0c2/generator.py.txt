"""Steepest-rise selection world: which colored segment has the largest vertical rise.

Latent scene z carries a set of colored line segments that all span the SAME
horizontal width W on a light panel. The task is a pure measurement: find the
segment whose vertical difference between its higher and lower end is largest
(i.e. the steepest). Because every segment shares the same horizontal width,
``|slope|`` and ``|rise|`` order the segments identically, so the answer is
recovered from the raster by fitting each colored stroke and ranking |slope|.

This world is a *measurement* of a recovered scalar quantity (a slope/rise rank)
and it is a set-argmax selection, counterbalancing which color holds the extremum
and which spatial column each segment occupies.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "steepest_rise-0.1.0"

# Fixed palette shared with oracle.py (kept as a plain list so both modules can
# agree on the named answer set without importing each other).
PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
SEGMENT_WIDTH_PX = 6
SHARED_WIDTH = 140  # every segment spans this many horizontal pixels

# Minimum top-two |rise| gap required for a decidable selection (pixels).
MIN_MARGIN_PX = 16.0
# Below this top-two |rise| gap the scene is quarantined as near-boundary yet
# still decidable (well above pixel-fit noise).
QUARANTINE_PX = 20.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum.

    Returns a plain dict (JSON-serialisable) that both render() and the manifest
    understand. ``segments`` is a list of {"color", "x0", "y0", "x1", "y1"}.
    """
    rng = random.Random(seed * 1009 + 7)
    ncols = len(PALETTE)
    col_width = WIDTH // ncols
    # Three distinct rises, strongest at least MIN_MARGIN_PX above the runner-up.
    while True:
        rises = [rng.uniform(70.0, 140.0) for _ in range(ncols)]
        rises_sorted = sorted(rises, reverse=True)
        if rises_sorted[0] - rises_sorted[1] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    columns = list(range(ncols))
    rng.shuffle(columns)
    base_y = rng.uniform(150.0, 190.0)
    segments = []
    for color, col, rise in zip(colors, columns, rises):
        x0 = col * col_width + (col_width - SHARED_WIDTH) / 2.0
        y0 = base_y - rise / 2.0
        y1 = base_y + rise / 2.0
        segments.append({"color": color, "x0": x0, "y0": y0, "x1": x0 + SHARED_WIDTH, "y1": y1})
    return {"segments": segments, "width": WIDTH, "height": HEIGHT, "background": "light"}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored strokes rise from a shared baseline row."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for seg in scene["segments"]:
        draw.line(
            [(seg["x0"], seg["y0"]), (seg["x1"], seg["y1"])],
            fill=COLOR_RGB[seg["color"]],
            width=SEGMENT_WIDTH_PX,
        )
    return img


def _steepest_rise(segments: list[dict]) -> tuple[dict, float, float]:
    """Return (segment, its |rise|, top-two margin) from the analytic scene."""
    keyed = [(seg, abs(seg["y1"] - seg["y0"])) for seg in segments]
    keyed.sort(key=lambda item: item[1], reverse=True)
    top, second = keyed[0], keyed[1]
    margin = top[1] - second[1]
    return top[0], top[1], margin


def analytic_gold(scene: dict) -> str:
    """The color of the segment with the largest vertical |rise|."""
    top, _rise, _margin = _steepest_rise(scene["segments"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two |rise| gap in pixels."""
    _top, _rise, margin_px = _steepest_rise(scene["segments"])
    return float(margin_px)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two |rise| gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Swapping two segments' colors would change which answer is correct, and any
    global flip/rotation turns the raster into a different image. Every latent
    field (the segment geometry and labels) is directly visible in the render,
    so the parametrisation has no hidden label alias.
    """
    return []
