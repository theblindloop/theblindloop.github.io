"""Independent pixel-only inverse for contour closure.

This module intentionally contains its own palette and flood-fill logic. It
accepts only a PIL image and never imports the renderer, prompt, generator, or
verifier. For each named color it extracts the largest stroke component and
counts background components not connected to a padded crop border. A closed
contour has an isolated interior component; the unique open contour does not.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array([220.0, 55.0, 55.0]),
    "green": np.array([45.0, 155.0, 85.0]),
    "blue": np.array([55.0, 90.0, 215.0]),
}
WHITE = np.array([255.0, 255.0, 255.0])
COLOR_NAMES = ("red", "green", "blue")
COLOR_THRESHOLD = 155.0


def _color_mask(arr: np.ndarray, color: str) -> np.ndarray:
    target = PALETTE[color]
    target_distance = np.linalg.norm(arr - target, axis=2)
    white_distance = np.linalg.norm(arr - WHITE, axis=2)
    other_distances = [
        np.linalg.norm(arr - PALETTE[name], axis=2)
        for name in COLOR_NAMES
        if name != color
    ]
    nearest_other = np.minimum.reduce(other_distances)
    return (
        (target_distance < COLOR_THRESHOLD)
        & (target_distance + 1.0 < white_distance)
        & (target_distance <= nearest_other)
    )


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        for cy, cx in queue:
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(best):
            best = component
    component_mask = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        component_mask[np.array(ys), np.array(xs)] = True
    return component_mask


def _has_isolated_interior(component: np.ndarray) -> bool:
    ys, xs = np.nonzero(component)
    if len(xs) < 250:
        raise ValueError("colored contour component is too small")
    y0, y1 = max(0, int(ys.min()) - 5), min(component.shape[0], int(ys.max()) + 6)
    x0, x1 = max(0, int(xs.min()) - 5), min(component.shape[1], int(xs.max()) + 6)
    barrier = component[y0:y1, x0:x1]
    free = ~barrier
    seen = np.zeros_like(free, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    h, w = free.shape
    for x in range(w):
        if free[0, x]:
            seen[0, x] = True
            queue.append((0, x))
        if free[h - 1, x] and not seen[h - 1, x]:
            seen[h - 1, x] = True
            queue.append((h - 1, x))
    for y in range(h):
        if free[y, 0] and not seen[y, 0]:
            seen[y, 0] = True
            queue.append((y, 0))
        if free[y, w - 1] and not seen[y, w - 1]:
            seen[y, w - 1] = True
            queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and free[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    isolated = int(np.count_nonzero(free & ~seen))
    return isolated >= 400


def decision_from_image(image: Image.Image) -> str:
    """Return the unique color whose contour does not enclose an interior."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    open_colors: list[str] = []
    for color in COLOR_NAMES:
        component = _largest_component(_color_mask(arr, color))
        if not _has_isolated_interior(component):
            open_colors.append(color)
    if len(open_colors) != 1:
        raise ValueError(f"expected one open contour, recovered {open_colors}")
    return open_colors[0]
