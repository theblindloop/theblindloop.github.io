"""Renderer for the contour-closure selection world.

The scene is deliberately plain line art: three separated colored contours are
drawn on a white raster. Exactly one contour omits a wide arc. The analytic label
is the color name assigned to that opening; the independent oracle recovers it
from the raster by testing whether each contour encloses an isolated background
component.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "contour-closure-selection-0.1.0"
SUPERSAMPLE = 4
COLOR_RGB = {
    "red": (220, 55, 55),
    "green": (45, 155, 85),
    "blue": (55, 90, 215),
}
COLOR_NAMES = ("red", "green", "blue")
BACKGROUND = (255, 255, 255)
OPEN_GAP_RANGE = (26.0, 66.0)


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene from an integer seed.

    Label, color order, geometry, and gap direction are sampled by separate
    deterministic streams so the answer is not tied to a panel slot or size.
    """

    label_rng = random.Random(seed * 1000003 + 17)
    layout_rng = random.Random(seed * 1000003 + 101)
    width, height = 600, 240
    colors = list(COLOR_NAMES)
    layout_rng.shuffle(colors)
    open_color = COLOR_NAMES[label_rng.randrange(len(COLOR_NAMES))]
    x_slots = (105.0, 300.0, 495.0)
    panel_centers = [
        [x + layout_rng.uniform(-10.0, 10.0), 120.0 + layout_rng.uniform(-9.0, 9.0)]
        for x in x_slots
    ]
    radii = [layout_rng.uniform(53.0, 69.0) for _ in colors]
    strokes = [layout_rng.uniform(7.0, 10.0) for _ in colors]
    gap_centers = [layout_rng.uniform(0.0, 360.0) for _ in colors]
    gap_degrees = layout_rng.uniform(*OPEN_GAP_RANGE)
    return {
        "width": width,
        "height": height,
        "panel_centers": panel_centers,
        "radii": radii,
        "strokes": strokes,
        "colors": colors,
        "open_color": open_color,
        "gap_centers": gap_centers,
        "gap_degrees": gap_degrees,
    }


def _contour_points(
    cx: float,
    cy: float,
    radius: float,
    open_contour: bool,
    gap_center: float,
    gap_degrees: float,
    samples: int = 420,
) -> list[tuple[float, float]]:
    if open_contour:
        start = math.radians(gap_center + gap_degrees / 2.0)
        span = math.radians(360.0 - gap_degrees)
        angles = [start + span * i / (samples - 1) for i in range(samples)]
    else:
        angles = [2.0 * math.pi * i / (samples - 1) for i in range(samples)]
    return [(cx + radius * math.cos(a), cy + radius * math.sin(a)) for a in angles]


def _draw_contour(
    draw: ImageDraw.ImageDraw,
    center: list[float],
    radius: float,
    stroke: float,
    color: str,
    open_contour: bool,
    gap_center: float,
    gap_degrees: float,
) -> None:
    points = _contour_points(
        center[0] * SUPERSAMPLE,
        center[1] * SUPERSAMPLE,
        radius * SUPERSAMPLE,
        open_contour,
        gap_center,
        gap_degrees,
    )
    draw.line(
        points,
        fill=COLOR_RGB[color],
        width=max(1, int(round(stroke * SUPERSAMPLE))),
        joint="curve",
    )


def render(scene: dict) -> Image.Image:
    """Render only the visible scene fields into a deterministic RGB PNG image."""

    width = int(scene["width"])
    height = int(scene["height"])
    panel_centers = scene["panel_centers"]
    radii = scene["radii"]
    strokes = scene["strokes"]
    colors = scene["colors"]
    open_color = scene["open_color"]
    gap_centers = scene["gap_centers"]
    gap_degrees = float(scene["gap_degrees"])
    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for index, color in enumerate(colors):
        _draw_contour(
            draw,
            panel_centers[index],
            float(radii[index]),
            float(strokes[index]),
            color,
            color == open_color,
            float(gap_centers[index]),
            gap_degrees,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Analytic decision from the label input, not from any raster measurement."""

    return str(scene["open_color"])


def margin(scene: dict) -> float:
    """Return the smallest visible clearance relevant to closure."""

    colors = scene["colors"]
    open_index = colors.index(scene["open_color"])
    open_radius = float(scene["radii"][open_index])
    open_stroke = float(scene["strokes"][open_index])
    gap = math.radians(float(scene["gap_degrees"]))
    endpoint_chord = 2.0 * open_radius * math.sin(gap / 2.0)
    contour_clearance = min(
        float(radius) - float(stroke) / 2.0
        for radius, stroke in zip(scene["radii"], scene["strokes"])
    )
    return round(min(endpoint_chord - open_stroke, contour_clearance), 6)


def is_quarantined(scene: dict) -> bool:
    """Exclude contours whose opening is too close to stroke-scale ambiguity."""

    return margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unrendered label-preserving latent aliases in this scene."""

    return []
