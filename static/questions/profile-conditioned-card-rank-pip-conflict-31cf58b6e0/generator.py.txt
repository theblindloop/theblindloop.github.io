"""Deterministic Pillow renderer for a visible playing-card consistency conflict."""

from __future__ import annotations

import hashlib
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 520
HEIGHT = 440
RANKS = (3, 4, 5, 6)
LAYOUTS = ("wave_left", "wave_right")
SUPERSAMPLE = 3

# Each tuple is background, card, outline, pip, digit, badge. Palette selection
# is pair-level style and therefore independent of the yes/no answer.
PALETTES = (
    ("#e9eef4", "#fffdf8", "#202933", "#2459c4", "#20252c", "#f2ca63"),
    ("#f0ece7", "#fffefe", "#30252a", "#bd3155", "#28242a", "#72cdb9"),
    ("#edf0e8", "#fffdf9", "#20312c", "#168263", "#202a27", "#d49be9"),
    ("#eeeaf2", "#fffefa", "#2e2637", "#7145bd", "#28222f", "#eea36f"),
)

SEGMENTS = {
    3: frozenset("abcdg"),
    4: frozenset("bcfg"),
    5: frozenset("acdfg"),
    6: frozenset("acdefg"),
}


def _rng(pair_id: int, channel: str) -> random.Random:
    payload = f"card-rank-pips-v1|{int(pair_id)}|{channel}".encode("utf-8")
    return random.Random(int.from_bytes(hashlib.sha256(payload).digest()[:8], "big"))


def sample_scene(seed: int) -> dict:
    """Return paired canonical/counterfactual scenes with matched visual style."""
    seed = int(seed)
    pair_id = seed // 2
    rank_index = pair_id % len(RANKS)
    printed_rank = RANKS[rank_index]
    matches = seed % 2 == 0
    direction = 1 if (pair_id // len(RANKS)) % 2 == 0 else -1
    pip_count = printed_rank if matches else RANKS[(rank_index + direction) % len(RANKS)]

    style = _rng(pair_id, "style")
    card_width = style.choice((304, 320, 336))
    card_height = style.choice((368, 376, 384))
    x_offset = style.randint(62, WIDTH - card_width - 62)
    y_offset = style.randint(22, HEIGHT - card_height - 18)
    pip_radius = style.choice((16, 17, 18))
    pip_layout = style.choice(LAYOUTS)
    digit_thickness = style.choice((10, 12)) if pair_id % 9 == 0 else style.choice((15, 17))
    background, card_fill, outline, pip_ink, digit_ink, badge_fill = style.choice(PALETTES)

    order = list(range(pip_count))
    _rng(pair_id, f"paint-{pip_count}").shuffle(order)
    return {
        "printed_rank": printed_rank,
        "pip_count": pip_count,
        "background": background,
        "card_fill": card_fill,
        "outline": outline,
        "pip_ink": pip_ink,
        "digit_ink": digit_ink,
        "x_offset": x_offset,
        "y_offset": y_offset,
        "card_width": card_width,
        "card_height": card_height,
        "pip_radius": pip_radius,
        "pip_layout": pip_layout,
        "pip_paint_order": order,
        "badge_fill": badge_fill,
        "digit_thickness": digit_thickness,
    }


def _pip_centers(scene: dict) -> list[tuple[float, float]]:
    count = int(scene["pip_count"])
    patterns = {
        3: ((0.50, 0.20), (0.50, 0.50), (0.50, 0.80)),
        4: ((0.34, 0.24), (0.66, 0.24), (0.34, 0.76), (0.66, 0.76)),
        5: ((0.34, 0.22), (0.66, 0.22), (0.50, 0.50), (0.34, 0.78), (0.66, 0.78)),
        6: ((0.34, 0.20), (0.66, 0.20), (0.34, 0.50), (0.66, 0.50), (0.34, 0.80), (0.66, 0.80)),
    }
    if count not in patterns:
        raise ValueError("pip_count must be one of 3, 4, 5, or 6")
    x = float(scene["x_offset"])
    y = float(scene["y_offset"])
    width = float(scene["card_width"])
    height = float(scene["card_height"])
    top = y + 145.0
    usable_height = height - 171.0
    direction = -1.0 if scene["pip_layout"] == "wave_left" else 1.0
    centers = []
    for index, (nx, ny) in enumerate(patterns[count]):
        wave = direction * (6.0 if index % 2 == 0 else -6.0)
        centers.append((x + nx * width + wave, top + ny * usable_height))
    return centers


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SUPERSAMPLE)) for value in box)  # type: ignore[return-value]


def _draw_digit(
    draw: ImageDraw.ImageDraw,
    rank: int,
    badge: tuple[float, float, float, float],
    thickness: int,
    ink: str,
) -> None:
    bx0, by0, bx1, by1 = badge
    left, right = bx0 + 19, bx1 - 19
    top, bottom = by0 + 14, by1 - 14
    middle = (top + bottom) / 2.0
    t = float(thickness)
    boxes = {
        "a": (left + t / 2, top, right - t / 2, top + t),
        "b": (right - t, top + t / 2, right, middle - t / 2),
        "c": (right - t, middle + t / 2, right, bottom - t / 2),
        "d": (left + t / 2, bottom - t, right - t / 2, bottom),
        "e": (left, middle + t / 2, left + t, bottom - t / 2),
        "f": (left, top + t / 2, left + t, middle - t / 2),
        "g": (left + t / 2, middle - t / 2, right - t / 2, middle + t / 2),
    }
    for name in SEGMENTS[rank]:
        draw.rounded_rectangle(_scaled_box(boxes[name]), radius=max(2, thickness * SUPERSAMPLE // 3), fill=ink)


def render(scene: dict) -> Image.Image:
    rank = int(scene["printed_rank"])
    count = int(scene["pip_count"])
    if rank not in SEGMENTS or count not in RANKS:
        raise ValueError("visible rank and pip count must be in 3..6")
    x = int(scene["x_offset"])
    y = int(scene["y_offset"])
    width = int(scene["card_width"])
    height = int(scene["card_height"])
    radius = int(scene["pip_radius"])
    if not (0 <= x and 0 <= y and x + width < WIDTH and y + height < HEIGHT):
        raise ValueError("card does not fit on the canvas")
    order = [int(value) for value in scene["pip_paint_order"]]
    if sorted(order) != list(range(count)):
        raise ValueError("pip_paint_order must permute every visible pip")
    if scene["pip_layout"] not in LAYOUTS:
        raise ValueError("unknown pip layout")

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), scene["background"])
    draw = ImageDraw.Draw(image)
    card = (x, y, x + width, y + height)
    draw.rounded_rectangle(
        _scaled_box(card),
        radius=24 * SUPERSAMPLE,
        fill=scene["card_fill"],
        outline=scene["outline"],
        width=5 * SUPERSAMPLE,
    )
    badge = (x + 22, y + 20, x + 114, y + 138)
    draw.rounded_rectangle(
        _scaled_box(badge),
        radius=13 * SUPERSAMPLE,
        fill=scene["badge_fill"],
        outline=scene["outline"],
        width=4 * SUPERSAMPLE,
    )
    _draw_digit(
        draw,
        rank,
        badge,
        int(scene["digit_thickness"]),
        scene["digit_ink"],
    )

    centers = _pip_centers(scene)
    for index in order:
        cx, cy = centers[index]
        draw.ellipse(
            _scaled_box((cx - radius, cy - radius, cx + radius, cy + radius)),
            fill=scene["pip_ink"],
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    printed_rank = int(scene["printed_rank"])
    pip_count = int(scene["pip_count"])
    return "yes" if printed_rank == pip_count else "no"


def margin(scene: dict) -> float:
    centers = _pip_centers(scene)
    radius = float(scene["pip_radius"])
    separation = min(
        math.dist(first, second) - 2.0 * radius
        for index, first in enumerate(centers)
        for second in centers[index + 1 :]
    )
    return float(min(separation, float(scene["digit_thickness"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["pip_paint_order"] = list(reversed(scene["pip_paint_order"]))
    return [("reverse_disjoint_same_ink_pip_paint_order", twin)]
