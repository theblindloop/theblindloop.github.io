"""Independent PNG-only decoder for rank/pip equality.

The inverse arm receives no latent scene. It segments high-chroma regions,
classifies the one large rounded badge separately from circular pip components,
and decodes seven normalized stroke-occupancy zones inside that visible badge.
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class AmbiguousRasterError(ValueError):
    """The public card evidence cannot be decoded unambiguously."""


@dataclass(frozen=True)
class Component:
    area: int
    x0: int
    y0: int
    x1: int
    y1: int

    @property
    def width(self) -> int:
        return self.x1 - self.x0

    @property
    def height(self) -> int:
        return self.y1 - self.y0


def _components(mask: np.ndarray) -> list[Component]:
    work = mask.copy()
    components: list[Component] = []
    height, width = work.shape
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(sy, sx)])
        area = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while queue:
            y, x = queue.popleft()
            area += 1
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        components.append(Component(area, x0, y0, x1 + 1, y1 + 1))
    return components


def _segment_bits(arr: np.ndarray, badge: Component) -> frozenset[str]:
    dark = arr.max(axis=2) < 112
    zones = {
        "a": (0.50, 0.14),
        "b": (0.75, 0.32),
        "c": (0.75, 0.68),
        "d": (0.50, 0.86),
        "e": (0.25, 0.68),
        "f": (0.25, 0.32),
        "g": (0.50, 0.50),
    }
    present = set()
    half_width = max(3, int(round(badge.width * 0.045)))
    half_height = max(3, int(round(badge.height * 0.035)))
    for name, (nx, ny) in zones.items():
        cx = int(round(badge.x0 + nx * (badge.width - 1)))
        cy = int(round(badge.y0 + ny * (badge.height - 1)))
        patch = dark[
            max(0, cy - half_height) : min(dark.shape[0], cy + half_height + 1),
            max(0, cx - half_width) : min(dark.shape[1], cx + half_width + 1),
        ]
        if patch.size and float(patch.mean()) >= 0.40:
            present.add(name)
    return frozenset(present)


def _measure(image: Image.Image) -> tuple[int, int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 240 or arr.shape[1] < 280:
        raise AmbiguousRasterError("raster is too small for the public card contract")
    high = arr.max(axis=2).astype(np.int16)
    low = arr.min(axis=2).astype(np.int16)
    chromatic = (high - low) > 58
    components = [component for component in _components(chromatic) if component.area >= 240]

    badges = [
        component
        for component in components
        if component.area >= 3500 and component.width >= 65 and component.height >= 85
    ]
    if len(badges) != 1:
        raise AmbiguousRasterError(f"expected one visible colored rank badge, found {len(badges)}")
    badge = badges[0]

    pips = []
    for component in components:
        if component == badge:
            continue
        center_x = (component.x0 + component.x1) / 2.0
        center_y = (component.y0 + component.y1) / 2.0
        if badge.x0 <= center_x <= badge.x1 and badge.y0 <= center_y <= badge.y1:
            # A dark seven-segment stroke may enclose a small island of badge
            # fill. It is visibly inside the recovered badge, never a center pip.
            continue
        ratio = component.width / max(component.height, 1)
        fill = component.area / max(component.width * component.height, 1)
        if 22 <= component.width <= 48 and 22 <= component.height <= 48 and 0.78 <= ratio <= 1.28 and 0.58 <= fill <= 0.90:
            pips.append(component)
        elif component.area >= 350:
            raise AmbiguousRasterError("unclassified colored evidence is visible")
    if len(pips) not in (3, 4, 5, 6):
        raise AmbiguousRasterError(f"visible pip count {len(pips)} is outside 3..6")

    patterns = {
        frozenset("abcdg"): 3,
        frozenset("bcfg"): 4,
        frozenset("acdfg"): 5,
        frozenset("acdefg"): 6,
    }
    bits = _segment_bits(arr, badge)
    if bits not in patterns:
        raise AmbiguousRasterError(f"badge strokes do not encode one supported rank: {sorted(bits)}")
    return patterns[bits], len(pips)


def decision_from_image(image: Image.Image) -> str:
    printed_rank, pip_count = _measure(image)
    return "yes" if printed_rank == pip_count else "no"
