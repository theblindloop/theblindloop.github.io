"""Central-symmetry world: recover a global point-reflection Gestalt.

A small set (four) of colored dots is drawn on a white canvas. The decision is
the global Gestalt property of central (point) symmetry of the color-carrying
dot configuration about its own centroid: rotating the whole pattern 180
degrees about the set centroid maps every dot onto its mirrored partner.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
decision from the final PNG alone, from the four dot centroids.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
RADIUS = 8
BACKGROUND = (255, 255, 255)
# Saturated, mutually distinct ink colors (none is close to the background).
PALETTE = [
    (215, 40, 40),
    (40, 110, 220),
    (40, 170, 80),
    (230, 150, 20),
    (150, 60, 180),
    (40, 190, 190),
]

# Center sampled in a region that keeps every dot (up to _R_MAX away) inside.
_CX_LO = 150
_CX_HI = 363
_R_MIN = 85
_R_MAX = 145

# Central-symmetry decision thresholds on the max reflection residual (px).
YES_TOL = 3.0
NO_MIN = 8.0
# Asymmetric break: displacement magnitude of the one broken dot (px).
DISP_MIN = 26
DISP_MAX = 46


def _color(cidx: int) -> tuple[int, int, int]:
    return PALETTE[cidx % len(PALETTE)]


def _symmetric_residual(points: list) -> float:
    """Max, over each dot, of its distance to the nearest reflected partner.

    Reflect every centroid about the set centroid; a dot is matched when its
    reflected position lands on another dot. The residual is the worst
    mismatched dot. Exactly 0 for a centrally symmetric dot set.
    """
    pts = np.asarray(points, dtype=float)
    center = pts.mean(axis=0)
    reflected = 2.0 * center - pts
    worst = 0.0
    for i in range(len(pts)):
        d = np.hypot(reflected[i, 0] - pts[:, 0], reflected[i, 1] - pts[:, 1])
        worst = max(worst, float(d.min()))
    return worst


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    cx = int(rng.integers(_CX_LO, _CX_HI))
    cy = int(rng.integers(_CX_LO, _CX_HI))

    # Two mirror pairs on roughly perpendicular axes so the four dots stay
    # far apart (within-pair separation is 2*r >= 170px; cross-pair distance
    # is at least ~100px), and a displaced dot cannot collide with a sibling.
    a = rng.uniform(0.0, 2.0 * math.pi)
    b = a + math.pi * 0.5 + rng.uniform(-0.3, 0.3)
    r1 = int(rng.integers(_R_MIN, _R_MAX + 1))
    r2 = int(rng.integers(_R_MIN, _R_MAX + 1))

    raw = [
        (cx + r1 * math.cos(a), cy + r1 * math.sin(a)),
        (cx - r1 * math.cos(a), cy - r1 * math.sin(a)),
        (cx + r2 * math.cos(b), cy + r2 * math.sin(b)),
        (cx - r2 * math.cos(b), cy - r2 * math.sin(b)),
    ]
    points = [[int(round(x)), int(round(y))] for (x, y) in raw]

    colors = list(int(i) for i in rng.permutation(len(PALETTE))[:4])

    symmetric = bool(rng.integers(0, 2))
    if not symmetric:
        # Displace one dot so the reflected pairing fails across the set.
        k = int(rng.integers(0, 4))
        ox, oy = points[k][0], points[k][1]
        vmag = int(rng.integers(DISP_MIN, DISP_MAX + 1))
        best = None
        best_score = 0.0
        for _attempt in range(300):
            ang = rng.uniform(0.0, 2.0 * math.pi)
            nx = min(WIDTH - 14, max(14, round(ox + vmag * math.cos(ang))))
            ny = min(HEIGHT - 14, max(14, round(oy + vmag * math.sin(ang))))
            length = math.hypot(nx - ox, ny - oy)
            if length < 20:
                continue
            candidate = list(points[:k]) + list(points[k + 1 :]) + [[nx, ny]]
            spacing = min(
                math.hypot(candidate[i][0] - candidate[j][0], candidate[i][1] - candidate[j][1])
                for i in range(4)
                for j in range(i + 1, 4)
            )
            if spacing >= 30:
                points[k][0], points[k][1] = nx, ny
                break
            if spacing > best_score:
                best_score = spacing
                best = (nx, ny)
        else:
            # Fall back to the least-colliding valid-length placement.
            if best is not None:
                points[k][0], points[k][1] = best

    return {"points": points, "colors": colors}


def _draw_scene(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (x, y), cidx in zip(scene["points"], scene["colors"]):
        d.ellipse([x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS], fill=_color(cidx))
    return img


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    return _draw_scene(scene)


def residual(scene: dict) -> float:
    """Max reflection residual of the dot set (the structural decision margin)."""
    return _symmetric_residual(scene["points"])


def analytic_gold(scene: dict) -> str:
    """Structural gold: is the dot set centrally symmetric about its centroid?"""
    if _symmetric_residual(scene["points"]) <= YES_TOL:
        return "yes"
    return "no"


def margin(scene: dict) -> float:
    """Commitment magnitude: the max reflection residual in pixels."""
    return float(_symmetric_residual(scene["points"]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine the ambiguous residual band between YES_TOL and NO_MIN."""
    r = _symmetric_residual(scene["points"])
    return YES_TOL < r < NO_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
