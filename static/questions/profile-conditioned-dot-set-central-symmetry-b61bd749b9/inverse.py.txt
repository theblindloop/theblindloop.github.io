"""Independent pixel-only inverse arm for the central-symmetry world.

Recovers the central-symmetry decision from the final PNG alone. It never sees
the latent scene, analytic gold, or renderer code. It detects the four colored
dot centroids by nearest-palette-color segmentation, then decides whether the
recovered dot configuration is centrally symmetric about its own centroid.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = [
    (215, 40, 40),
    (40, 110, 220),
    (40, 170, 80),
    (230, 150, 20),
    (150, 60, 180),
    (40, 190, 190),
]
BACKGROUND = (255, 255, 255)
# Independent threshold on the recovered max reflection residual (px).
RESIDUAL_TOL = 5.0


def _centroids(arr: np.ndarray, palette: list[tuple]) -> list[tuple[float, float]]:
    """Centroid of every non-empty palette-color mask."""
    centers = []
    for color in palette:
        target = np.array(color, dtype=np.float32)
        mask = np.linalg.norm(arr - target, axis=-1) < 48.0
        ys, xs = np.nonzero(mask)
        if len(ys) >= 5:
            centers.append((float(xs.mean()) + 0.5, float(ys.mean()) + 0.5))
    return centers


def _residual(centers: list[tuple[float, float]]) -> float:
    """Max over each dot of its distance to the nearest reflected partner."""
    pts = np.asarray(centers, dtype=float)
    center = pts.mean(axis=0)
    reflected = 2.0 * center - pts
    worst = 0.0
    for i in range(len(pts)):
        d = np.hypot(reflected[i, 0] - pts[:, 0], reflected[i, 1] - pts[:, 1])
        worst = max(worst, float(d.min()))
    return worst


def decision_from_image(image: Image.Image) -> str:
    """Centrally symmetric (yes) or not (no) about the dot set's centroid."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    centers = _centroids(arr, PALETTE)
    if len(centers) != 4:
        # Could not recover exactly four well-formed dots: abstain is not
        # expressible, so treat an unrecoverable configuration as asymmetric.
        return "no"
    if _residual(centers) <= RESIDUAL_TOL:
        return "yes"
    return "no"
