"""Pixel-only inverse arm for strand_more_ends.

Input is a rendered PNG and nothing else: no scene, no latent parameters, no
renderer, prompt, generator, verifier or gold code.  The arm segments the four
painted colours, groups every inked pixel into connected strand systems,
counts the black square terminals inside each system, finds the ring colour
each system carries, and reports the ring colour of the system with strictly
more terminals.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "strand_more_ends-oracle-0.1.0"
ABSTAIN = "abstain"

_BACKGROUND = (255, 255, 255)
_STRAND = (90, 90, 90)
_TERMINAL = (17, 17, 17)
_RINGS = {"blue": (30, 90, 220), "orange": (235, 130, 20)}
_FOREIGN_TOLERANCE = 100


def _exact(arr: np.ndarray, rgb: tuple) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    """Eight-connected component labelling over a boolean mask."""
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    current = 0
    ys, xs = np.nonzero(mask)
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if labels[y0, x0]:
            continue
        current += 1
        stack = [(y0, x0)]
        labels[y0, x0] = current
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not labels[ny, nx]:
                        labels[ny, nx] = current
                        stack.append((ny, nx))
    return labels, current


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    background = _exact(arr, _BACKGROUND)
    strand = _exact(arr, _STRAND)
    terminal = _exact(arr, _TERMINAL)
    rings = {name: _exact(arr, rgb) for name, rgb in _RINGS.items()}

    known = background | strand | terminal
    for m in rings.values():
        known = known | m
    if int(np.count_nonzero(~known)) > _FOREIGN_TOLERANCE:
        return ABSTAIN  # the raster is not this world's palette

    inked = ~background & known
    if not inked.any():
        return ABSTAIN

    labels, n_components = _label(inked)
    if n_components == 0:
        return ABSTAIN

    tally: dict[int, dict] = {}
    for comp in range(1, n_components + 1):
        comp_mask = labels == comp
        colours = [name for name, m in rings.items() if bool((m & comp_mask).any())]
        _, n_terminals = _label(terminal & comp_mask)
        tally[comp] = {"rings": colours, "terminals": n_terminals}

    ported = [v for v in tally.values() if v["rings"]]
    if len(ported) != 2:
        return ABSTAIN  # not exactly two ringed strand systems
    if any(len(v["rings"]) != 1 for v in ported):
        return ABSTAIN  # a system carries more than one ring colour
    first, second = ported
    if first["rings"][0] == second["rings"][0]:
        return ABSTAIN
    if first["terminals"] == second["terminals"]:
        return ABSTAIN  # tie: no strictly larger system
    if first["terminals"] == 0 or second["terminals"] == 0:
        return ABSTAIN
    winner = first if first["terminals"] > second["terminals"] else second
    return winner["rings"][0]
