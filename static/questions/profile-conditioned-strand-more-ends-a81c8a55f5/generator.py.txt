"""Scene sampler, renderer, analytic gold and margin for strand_more_ends.

A scene holds two strand systems drawn on a cell lattice.  Each system is a
tree of lattice cells: its root cell carries a coloured ring (the port) and
every other degree-one cell carries a small black square (a terminal).  The two
systems never share a cell, so their strokes neither touch nor cross.  The
decision is which ringed port belongs to the system that ends at more squares.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "strand_more_ends-renderer-0.1.0"

CANVAS = 600
GRID = 8
CELL_PX = 68
MARGIN_PX = 28
STROKE_PX = 5
MARKER_PX = 13
RING_R = 12
RING_W = 4

PALETTE = {
    "background": (255, 255, 255),
    "strand": (90, 90, 90),
    "terminal": (17, 17, 17),
    "blue": (30, 90, 220),
    "orange": (235, 130, 20),
}

RING_COLORS = ("blue", "orange")


def _center(cell: list[int] | tuple[int, int]) -> tuple[float, float]:
    i, j = cell
    return (
        MARGIN_PX + i * CELL_PX + CELL_PX / 2.0,
        MARGIN_PX + j * CELL_PX + CELL_PX / 2.0,
    )


def _neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    i, j = cell
    out = []
    for di, dj in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        a, b = i + di, j + dj
        if 0 <= a < GRID and 0 <= b < GRID:
            out.append((a, b))
    return out


def _grow(rng: random.Random, blocked: set, n_nodes: int) -> tuple[list, list] | None:
    """Grow a random lattice tree of `n_nodes` cells avoiding `blocked`."""
    free = [(i, j) for i in range(GRID) for j in range(GRID) if (i, j) not in blocked]
    if len(free) < n_nodes:
        return None
    root = rng.choice(free)
    nodes = {root}
    edges: list[tuple] = []
    while len(nodes) < n_nodes:
        options = [
            (u, v)
            for u in nodes
            for v in _neighbours(u)
            if v not in nodes and v not in blocked
        ]
        if not options:
            return None
        u, v = options[rng.randrange(len(options))]
        nodes.add(v)
        edges.append((u, v))
    return [root, sorted(nodes)], edges


def _degrees(root: tuple, edges: list) -> dict:
    deg: dict = {tuple(root): 0}
    for u, v in edges:
        deg[tuple(u)] = deg.get(tuple(u), 0) + 1
        deg[tuple(v)] = deg.get(tuple(v), 0) + 1
    return deg


def _leaves(root, edges: list) -> list[tuple]:
    """Terminal cells: degree-one cells other than the ringed root cell."""
    root = tuple(root)
    deg = _degrees(root, edges)
    return sorted(c for c, d in deg.items() if d == 1 and c != root)


def _try_system(rng: random.Random, blocked: set, n_nodes: int, want_leaves: int):
    for _ in range(60):
        grown = _grow(rng, blocked, n_nodes)
        if grown is None:
            continue
        (root, cells), edges = grown
        system = {"root": list(root), "edges": [[list(u), list(v)] for u, v in edges]}
        if len(_leaves(system["root"], system["edges"])) == want_leaves:
            system["cells"] = [list(c) for c in cells]
            return system
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(1_000_003 + 7919 * int(seed))
    for _ in range(400):
        win_leaves = rng.choice((3, 4, 5))
        diff = rng.choice((1, 2))
        lose_leaves = win_leaves - diff
        if lose_leaves < 2:
            continue
        # Anti-correlate the answer with how much strand ink a system spends.
        ink_conflict = rng.random() < 0.5
        if ink_conflict:
            win_nodes = win_leaves + 1
            lose_nodes = max(lose_leaves + 1, win_nodes + rng.choice((1, 2, 3)))
        else:
            lose_nodes = lose_leaves + 1
            win_nodes = max(win_leaves + 1, lose_nodes + rng.choice((1, 2, 3)))
        if ink_conflict and not win_nodes < lose_nodes:
            continue
        if not ink_conflict and not win_nodes > lose_nodes:
            continue

        winner_first = rng.random() < 0.5
        win_color = RING_COLORS[0] if rng.random() < 0.5 else RING_COLORS[1]
        lose_color = RING_COLORS[1] if win_color == RING_COLORS[0] else RING_COLORS[0]

        order = [
            (win_nodes, win_leaves, win_color),
            (lose_nodes, lose_leaves, lose_color),
        ]
        if not winner_first:
            order.reverse()

        blocked: set = set()
        systems = []
        ok = True
        for n_nodes, want_leaves, color in order:
            system = _try_system(rng, blocked, n_nodes, want_leaves)
            if system is None:
                ok = False
                break
            for cell in system["cells"]:
                c = (cell[0], cell[1])
                blocked.add(c)
                # keep a one-cell buffer so no two systems run side by side
                for nb in _neighbours(c):
                    blocked.add(nb)
            systems.append((color, system["root"], system["edges"]))
        if not ok:
            continue

        scene = {
            "canvas": CANVAS,
            "grid": GRID,
            "cell_px": CELL_PX,
            "margin_px": MARGIN_PX,
            "stroke_px": STROKE_PX,
            "marker_px": MARKER_PX,
            "ring_r": RING_R,
            "ring_w": RING_W,
            "ring_colors": [s[0] for s in systems],
            "roots": [s[1] for s in systems],
            "edges": [s[2] for s in systems],
        }
        if _clearance(scene) < 10.0:
            continue
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


def _segments(edges: list) -> list[tuple]:
    return [(_center(u), _center(v)) for u, v in edges]


def _seg_distance(a: tuple, b: tuple) -> float:
    (x1, y1), (x2, y2) = a
    (x3, y3), (x4, y4) = b
    best = 1e18
    for px, py, qa, qb in (
        (x1, y1, (x3, y3), (x4, y4)),
        (x2, y2, (x3, y3), (x4, y4)),
        (x3, y3, (x1, y1), (x2, y2)),
        (x4, y4, (x1, y1), (x2, y2)),
    ):
        (ax, ay), (bx, by) = qa, qb
        dx, dy = bx - ax, by - ay
        den = dx * dx + dy * dy
        t = 0.0 if den == 0 else max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / den))
        cx, cy = ax + t * dx, ay + t * dy
        best = min(best, ((px - cx) ** 2 + (py - cy) ** 2) ** 0.5)
    return best


def _clearance(scene: dict) -> float:
    """Conservative lower bound on the painted gap between the two systems."""
    a, b = scene["edges"]
    pad = scene["stroke_px"] + max(scene["marker_px"] / 2.0, scene["ring_r"] + scene["ring_w"] / 2.0)
    best = 1e18
    for sa in _segments(a):
        for sb in _segments(b):
            best = min(best, _seg_distance(sa, sb))
    return best - pad


def terminal_counts(scene: dict) -> dict:
    colors, roots, edges = scene["ring_colors"], scene["roots"], scene["edges"]
    return {c: len(_leaves(r, e)) for c, r, e in zip(colors, roots, edges)}


def analytic_gold(scene: dict) -> str:
    counts = terminal_counts(scene)
    blue, orange = counts["blue"], counts["orange"]
    if blue == orange:
        return "tie"
    return "blue" if blue > orange else "orange"


def margin(scene: dict) -> float:
    counts = terminal_counts(scene)
    diff = abs(counts["blue"] - counts["orange"])
    return round(min(_clearance(scene), 8.0 * diff), 6)


def is_quarantined(scene: dict) -> bool:
    counts = terminal_counts(scene)
    if counts["blue"] == counts["orange"]:
        return True
    return _clearance(scene) < 10.0


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["canvas"], scene["canvas"]), PALETTE["background"])
    draw = ImageDraw.Draw(img)
    for edges in scene["edges"]:
        for u, v in edges:
            draw.line([_center(u), _center(v)], fill=PALETTE["strand"], width=scene["stroke_px"])
    half = scene["marker_px"] / 2.0
    for root, edges in zip(scene["roots"], scene["edges"]):
        for cell in _leaves(root, edges):
            cx, cy = _center(cell)
            draw.rectangle(
                [cx - half, cy - half, cx + half, cy + half], fill=PALETTE["terminal"]
            )
    for root, color in zip(scene["roots"], scene["ring_colors"]):
        cx, cy = _center(root)
        r = scene["ring_r"]
        draw.ellipse(
            [cx - r, cy - r, cx + r, cy + r],
            outline=PALETTE[color],
            width=scene["ring_w"],
        )
    return img


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, label-preserving rewrites of the latent record."""
    out: list[tuple[str, dict]] = []

    flipped = _copy(scene)
    flipped["edges"] = [[[e[1], e[0]] for e in reversed(edges)] for edges in flipped["edges"]]
    out.append(("reverse_edge_list_and_endpoint_order", flipped))

    swapped = _copy(scene)
    for key in ("ring_colors", "roots", "edges"):
        swapped[key] = list(reversed(swapped[key]))
    out.append(("reverse_system_list", swapped))
    return out


def label_transforming_symmetries(scene: dict) -> list[tuple[str, dict, str]]:
    """Declared transforms that change the label in an exactly specified way."""
    recoloured = _copy(scene)
    recoloured["ring_colors"] = [
        "orange" if c == "blue" else "blue" for c in recoloured["ring_colors"]
    ]
    gold = analytic_gold(scene)
    expect = "orange" if gold == "blue" else "blue"
    return [("swap_ring_colours", recoloured, expect)]


def _copy(scene: dict) -> dict:
    out = dict(scene)
    out["ring_colors"] = list(scene["ring_colors"])
    out["roots"] = [list(r) for r in scene["roots"]]
    out["edges"] = [[[list(e[0]), list(e[1])] for e in edges] for edges in scene["edges"]]
    return out
