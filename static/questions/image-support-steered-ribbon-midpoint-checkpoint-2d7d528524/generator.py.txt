"""Renderer and analytic arm for the broad-ribbon midpoint task.

The only scene fields are the visible route polyline and two visible marker
fractions along its arclength.  Rendering deliberately uses one thick,
connected ribbon, so the route is an area-like object rather than a thin line.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 3
RIBBON_COLOR = (35, 86, 148)
SPINE_COLOR = (103, 151, 211)
BACKGROUND = (255, 255, 255)
AMBER = (244, 154, 42)
VIOLET = (176, 72, 182)
GREEN = (38, 168, 104)
RED = (222, 67, 64)
BLACK = (24, 30, 38)
RIBBON_WIDTH = 220.0
CHECKPOINT_RADIUS = 21.0
PORT_RADIUS = 20.0


def _points(scene: dict) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in scene["route_points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(cumulative[-1] + math.hypot(second[0] - first[0], second[1] - first[1]))
    return cumulative, cumulative[-1]


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span <= 0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return (x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0))
    return points[-1]


def sample_scene(seed: int) -> dict:
    """Sample a broad, monotone-x ribbon and balanced checkpoint labels."""

    rng = random.Random(int(seed))
    # The ribbon is intentionally broad enough that grid ablations see one
    # connected area-like support, while its end ports remain several pixels
    # inside the canvas border for a stable background estimate.
    left = rng.uniform(120.0, 130.0)
    right = rng.uniform(382.0, 392.0)
    baseline = rng.uniform(246.0, 272.0)
    amplitude = rng.uniform(58.0, 78.0)
    phase = rng.uniform(-0.45, 0.45)
    frequency = rng.uniform(0.88, 1.16)
    points: list[list[float]] = []
    for index in range( nine := 9 ):
        t = index / (nine - 1)
        x = left + (right - left) * t
        y = baseline + amplitude * math.sin(2.0 * math.pi * frequency * t + phase)
        y += rng.uniform(-7.0, 7.0) * math.sin(math.pi * t)
        points.append([round(x, 4), round(y, 4)])

    # The two fractions are deliberately on opposite sides of the midpoint,
    # with a gap large enough to survive raster centerline error.  Swapping the
    # labels is independent of route geometry and balances the answer classes.
    if rng.randrange(2) == 0:
        checkpoint_fractions = {"amber": 0.30, "violet": 0.84}
    else:
        checkpoint_fractions = {"amber": 0.84, "violet": 0.30}
    return {"route_points": points, "checkpoint_fractions": checkpoint_fractions}


def render(scene: dict) -> Image.Image:
    points = _points(scene)
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    scaled = [(round(x * SCALE), round(y * SCALE)) for x, y in points]
    # One large rounded body is the filled ribbon support.  The ordered route
    # remains visible as its contrasting spine; the body keeps that route a
    # single compact area-like component across widely separated positions.
    draw.rounded_rectangle(
        (28 * SCALE, 70 * SCALE, 484 * SCALE, 442 * SCALE),
        radius=48 * SCALE,
        fill=RIBBON_COLOR,
    )

    # A narrow contrasting spine makes the ordered route legible in the
    # raster, while the surrounding filled ribbon remains the decisive,
    # area-like support object.
    spine_width = round(11.0 * SCALE)
    spine_radius = spine_width // 2
    draw.line(scaled, fill=SPINE_COLOR, width=spine_width, joint="curve")
    for x, y in scaled:
        draw.ellipse((x - spine_radius, y - spine_radius, x + spine_radius, y + spine_radius), fill=SPINE_COLOR)

    start = scaled[0]
    finish = scaled[-1]
    port_radius = round(PORT_RADIUS * SCALE)
    for center, color in ((start, GREEN), (finish, RED)):
        x, y = center
        draw.ellipse(
            (x - port_radius - 3 * SCALE, y - port_radius - 3 * SCALE,
             x + port_radius + 3 * SCALE, y + port_radius + 3 * SCALE),
            fill=BLACK,
        )
        draw.ellipse(
            (x - port_radius, y - port_radius, x + port_radius, y + port_radius),
            fill=color,
        )

    marker_specs = (("amber", AMBER), ("violet", VIOLET))
    marker_radius = round(CHECKPOINT_RADIUS * SCALE)
    for name, color in marker_specs:
        cx, cy = _point_at_fraction(points, float(scene["checkpoint_fractions"][name]))
        x, y = round(cx * SCALE), round(cy * SCALE)
        outer = marker_radius + 4 * SCALE
        draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=BLACK)
        draw.ellipse((x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius), fill=color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    _, total = _route_lengths(_points(scene))
    fractions = scene["checkpoint_fractions"]
    amber_gap = abs(float(fractions["amber"]) - 0.5) * total
    violet_gap = abs(float(fractions["violet"]) - 0.5) * total
    if abs(amber_gap - violet_gap) < 1e-9:
        raise ValueError("midpoint tie is not a valid scene")
    return "amber" if amber_gap < violet_gap else "violet"


def margin(scene: dict) -> float:
    _, total = _route_lengths(_points(scene))
    fractions = scene["checkpoint_fractions"]
    amber_gap = abs(float(fractions["amber"]) - 0.5) * total
    violet_gap = abs(float(fractions["violet"]) - 0.5) * total
    return abs(amber_gap - violet_gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All label-bearing latent fields are visibly rendered; there are no
    # unobservable aliases to declare for this injective scene representation.
    return []
