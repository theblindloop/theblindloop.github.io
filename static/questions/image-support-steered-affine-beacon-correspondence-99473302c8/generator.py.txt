"""Deterministic visual world for landmark-defined affine point transfer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

CANVAS = 960
BACKGROUND = (246, 249, 252)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (196, 207, 219)
TITLE_GRAY = (100, 116, 139)
INK = (30, 41, 59)
QUERY_MAGENTA = (219, 39, 119)
TAG_RGB = {
    "amber": (217, 119, 6),
    "cyan": (8, 145, 178),
    "violet": (124, 58, 237),
}
TAGS = tuple(TAG_RGB)

# These three well-separated barycentric locations appear in every target.
# Their colors, the source query location, panel handedness, and geometry vary.
BEACON_WEIGHTS = (
    (0.64, 0.18, 0.18),
    (0.18, 0.64, 0.18),
    (0.18, 0.18, 0.64),
)


def _weighted(points: list[list[int]], weights: tuple[float, float, float]) -> list[int]:
    return [
        int(round(sum(weights[index] * points[index][axis] for index in range(3))))
        for axis in range(2)
    ]


def _landmark_triangle(rng: random.Random, *, right: bool) -> list[list[int]]:
    """Return circle, square, triangle coordinates in one compact panel."""

    if right:
        top_x = 715 + rng.randint(-5, 5)
        low = [660 + rng.randint(-4, 4), 770 + rng.randint(-4, 4)]
    else:
        top_x = 245 + rng.randint(-5, 5)
        low = [190 + rng.randint(-4, 4), 300 + rng.randint(-4, 4)]
    if rng.randrange(2):
        low.reverse()
    return [
        [top_x, 320 + rng.randint(-8, 8)],
        [low[0], 596 + rng.randint(-8, 8)],
        [low[1], 606 + rng.randint(-8, 8)],
    ]


def sample_scene(seed: int) -> dict:
    """Sample two nondegenerate triangles and independently bound target colors."""

    seed = int(seed)
    rng = random.Random(seed ^ 0xA771B34C)
    left = _landmark_triangle(rng, right=False)
    right = _landmark_triangle(rng, right=True)
    query_index = (seed // 3) % 3
    query = _weighted(left, BEACON_WEIGHTS[query_index])
    candidates = [_weighted(right, weights) for weights in BEACON_WEIGHTS]

    answer_tag = TAGS[seed % 3]
    candidate_tags: list[str | None] = [None, None, None]
    candidate_tags[query_index] = answer_tag
    remaining_positions = [index for index in range(3) if index != query_index]
    remaining_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(remaining_tags)
    for index, tag in zip(remaining_positions, remaining_tags, strict=True):
        candidate_tags[index] = tag

    return {
        "left_landmarks": left,
        "right_landmarks": right,
        "query_point": query,
        "candidate_points": candidates,
        "candidate_tags": candidate_tags,
        "landmark_size": (21, 24)[rng.randrange(2)],
    }


def _draw_landmark(
    draw: ImageDraw.ImageDraw, code: int, point: list[int], size: int
) -> None:
    x, y = point
    if code == 0:
        draw.ellipse((x - size, y - size, x + size, y + size), fill=INK)
    elif code == 1:
        draw.rectangle((x - size, y - size, x + size, y + size), fill=INK)
    elif code == 2:
        # The polygon is centered at (x, y), so its measured centroid is a
        # faithful landmark rather than a renderer-only vertex convention.
        upper = 4 * size // 3
        lower = 2 * size // 3
        draw.polygon(((x, y - upper), (x - size, y + lower), (x + size, y + lower)), fill=INK)
    else:
        raise ValueError("unknown landmark code")


def _draw_dot(draw: ImageDraw.ImageDraw, point: list[int], color: tuple[int, int, int]) -> None:
    x, y = point
    radius = 13
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    # Direct key reads keep the rendered-field audit explicit.
    left = scene["left_landmarks"]
    right = scene["right_landmarks"]
    query = scene["query_point"]
    candidate_points = scene["candidate_points"]
    candidate_tags = scene["candidate_tags"]
    size = int(scene["landmark_size"])

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((120, 230, 370, 700), radius=22, fill=PANEL_FILL, outline=PANEL_EDGE, width=4)
    draw.rounded_rectangle((590, 230, 840, 700), radius=22, fill=PANEL_FILL, outline=PANEL_EDGE, width=4)
    font = ImageFont.load_default(size=22)
    draw.text((245, 262), "SOURCE", fill=TITLE_GRAY, anchor="mm", font=font)
    draw.text((715, 262), "WARPED", fill=TITLE_GRAY, anchor="mm", font=font)
    for code, point in enumerate(left):
        _draw_landmark(draw, code, point, size)
    for code, point in enumerate(right):
        _draw_landmark(draw, code, point, size)
    _draw_dot(draw, query, QUERY_MAGENTA)
    for point, tag in zip(candidate_points, candidate_tags, strict=True):
        _draw_dot(draw, point, TAG_RGB[tag])
    return image


def _barycentric(point: list[int], triangle: list[list[int]]) -> tuple[float, float, float]:
    (x0, y0), (x1, y1), (x2, y2) = triangle
    x, y = point
    denominator = (y1 - y2) * (x0 - x2) + (x2 - x1) * (y0 - y2)
    if abs(denominator) < 1e-9:
        raise ValueError("degenerate landmark triangle")
    w0 = ((y1 - y2) * (x - x2) + (x2 - x1) * (y - y2)) / denominator
    w1 = ((y2 - y0) * (x - x2) + (x0 - x2) * (y - y2)) / denominator
    return w0, w1, 1.0 - w0 - w1


def _distance_ranking(scene: dict) -> list[tuple[float, int]]:
    weights = _barycentric(scene["query_point"], scene["left_landmarks"])
    target = (
        sum(weights[index] * scene["right_landmarks"][index][0] for index in range(3)),
        sum(weights[index] * scene["right_landmarks"][index][1] for index in range(3)),
    )
    return sorted(
        (
            math.dist(target, scene["candidate_points"][index]),
            index,
        )
        for index in range(3)
    )


def analytic_gold(scene: dict) -> str:
    ranking = _distance_ranking(scene)
    if len(ranking) != 3 or ranking[1][0] - ranking[0][0] <= 1e-9:
        raise ValueError("scene has no unique nearest target dot")
    return scene["candidate_tags"][ranking[0][1]]


def margin(scene: dict) -> float:
    ranking = _distance_ranking(scene)
    return float(ranking[1][0] - ranking[0][0])


def _triangle_area(points: list[list[int]]) -> float:
    return abs(
        (points[1][0] - points[0][0]) * (points[2][1] - points[0][1])
        - (points[2][0] - points[0][0]) * (points[1][1] - points[0][1])
    ) / 2.0


def is_quarantined(scene: dict) -> bool:
    try:
        left = scene["left_landmarks"]
        right = scene["right_landmarks"]
        query = scene["query_point"]
        points = scene["candidate_points"]
        tags = scene["candidate_tags"]
        size = int(scene["landmark_size"])
        if len(left) != 3 or len(right) != 3 or len(points) != 3:
            return True
        if sorted(tags) != sorted(TAGS) or size not in {21, 24}:
            return True
        if _triangle_area(left) < 9000 or _triangle_area(right) < 9000:
            return True
        weights = _barycentric(query, left)
        if min(weights) < 0.12 or max(weights) > 0.70:
            return True
        ranking = _distance_ranking(scene)
        if ranking[0][0] > 8.0 or ranking[1][0] - ranking[0][0] < 40.0:
            return True
        if min(math.dist(points[a], points[b]) for a in range(3) for b in range(a)) < 38.0:
            return True
        return analytic_gold(scene) not in TAGS
    except (KeyError, TypeError, ValueError, IndexError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["candidate_points"] = twin["candidate_points"][1:] + twin["candidate_points"][:1]
    twin["candidate_tags"] = twin["candidate_tags"][1:] + twin["candidate_tags"][:1]
    return [("rotate_candidate_record_order", twin)]
