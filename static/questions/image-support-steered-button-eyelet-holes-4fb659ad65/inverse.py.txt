"""Pixel-only inverse arm: count the openings inside the one coloured disc.

Receives a PNG and nothing else. It knows only the public question contract:
exactly one item is strongly coloured, every other mark is grey, and a hole is a
background-coloured opening lying wholly inside the coloured disc.

Method: chroma segmentation (max-min channel spread) isolates the coloured ink;
the largest 4-connected chromatic component is the button; the holes are the
background-labelled 4-connected components enclosed by that component, found by
flooding the complement inward from a padded border. No renderer state, latent
scene, or analytic gold is consulted.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "button_eyelet_holes-oracle-0.1.0"

CHROMA_THRESHOLD = 46  # grey clutter and white paper have chroma 0
MIN_BUTTON_AREA = 900  # px; smaller chromatic blobs are noise
MIN_HOLE_AREA = 22  # px; below this an opening is not legible
MAX_HOLE_ECCENTRICITY = 0.55  # a legible opening is not a hairline crack
ANSWERS = ("2", "3", "4", "5")
ABSTAIN = "abstain"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """4-connected components of a boolean mask, as index arrays."""
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    out: list[np.ndarray] = []
    current = 0
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if labels[sy, sx]:
            continue
        current += 1
        stack = deque([(sy, sx)])
        labels[sy, sx] = current
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = current
                    stack.append((ny, nx))
        out.append(np.array(pixels, dtype=np.int32))
    return out


def _largest_chromatic_component(arr: np.ndarray) -> np.ndarray | None:
    hi = arr.max(axis=-1).astype(np.int16)
    lo = arr.min(axis=-1).astype(np.int16)
    chroma = (hi - lo) >= CHROMA_THRESHOLD
    if not chroma.any():
        return None
    best = None
    best_size = 0
    for pixels in _components(chroma):
        if len(pixels) > best_size:
            best, best_size = pixels, len(pixels)
    if best is None or best_size < MIN_BUTTON_AREA:
        return None
    mask = np.zeros(arr.shape[:2], dtype=bool)
    mask[best[:, 0], best[:, 1]] = True
    return mask


def _enclosed_regions(mask: np.ndarray) -> list[np.ndarray]:
    """Background components fully surrounded by the mask, inside its bbox."""
    ys, xs = np.nonzero(mask)
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    window = mask[y0 : y1 + 1, x0 : x1 + 1]
    padded = np.zeros((window.shape[0] + 2, window.shape[1] + 2), dtype=bool)
    padded[1:-1, 1:-1] = window
    hole_space = ~padded

    h, w = hole_space.shape
    outside = np.zeros_like(hole_space)
    stack = deque([(0, 0)])
    outside[0, 0] = True
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and hole_space[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                stack.append((ny, nx))
    return _components(hole_space & ~outside)


def decision_from_image(image: Image.Image) -> str:
    """Number of openings inside the coloured disc, from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mask = _largest_chromatic_component(arr)
    if mask is None:
        return ABSTAIN
    holes = []
    for pixels in _enclosed_regions(mask):
        if len(pixels) < MIN_HOLE_AREA:
            return ABSTAIN  # an illegible speck: refuse rather than guess
        span_y = pixels[:, 0].max() - pixels[:, 0].min() + 1
        span_x = pixels[:, 1].max() - pixels[:, 1].min() + 1
        if min(span_y, span_x) / max(span_y, span_x) < MAX_HOLE_ECCENTRICITY:
            return ABSTAIN  # a crack between merging holes, not an opening
        holes.append(pixels)
    answer = str(len(holes))
    return answer if answer in ANSWERS else ABSTAIN
