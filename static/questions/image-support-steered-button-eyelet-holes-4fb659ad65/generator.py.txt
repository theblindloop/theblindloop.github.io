"""Button-eyelet world: how many holes are punched through the one coloured button?

The decisive evidence is a single small, thick, connected coloured disc. Its
answer is the number of enclosed background openings (holes) inside that disc --
a topological property of one compact region, not a property of any separate
mark. Everything else on the canvas is achromatic sewing-kit clutter, including
grey decoy buttons that carry a different number of holes.

The familiar canonical prior ("a button has four holes") is recorded by the
generator for analysis only; gold is always the number of holes actually punched.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
SS = 3  # supersampling factor; BOX downsample keeps interiors flat

BACKGROUND = (255, 255, 255)
HOLE_RGB = BACKGROUND

# Saturated button colours. Every decoration is achromatic, so "the coloured
# disc" is an unambiguous, wording-declared selector.
BUTTON_COLORS = (
    (198, 32, 46),
    (28, 72, 190),
    (22, 132, 62),
    (128, 40, 162),
    (222, 118, 12),
    (0, 138, 150),
)
GREYS = ((120, 120, 120), (152, 152, 152), (96, 96, 96), (176, 176, 176))

HOLE_COUNTS = (2, 3, 4, 5)
CANONICAL_PRIOR = 4  # analysis only; never gold

MIN_VISIBLE_HOLE_DIAMETER = 8.0
QUARANTINE_MARGIN = 4.0
DECOR_CLEARANCE = 12.0


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _hole_ring(radius: float, count: int, hole_r: float, ring: float, phase: float,
               cx: float, cy: float) -> list[list[float]]:
    holes = []
    for k in range(count):
        a = phase + 2.0 * math.pi * k / count
        holes.append(
            [
                round(cx + ring * math.cos(a), 4),
                round(cy + ring * math.sin(a), 4),
                round(hole_r, 4),
            ]
        )
    return holes


def sample_scene(seed: int) -> dict:
    rng = random.Random(0x9E37 ^ (seed * 2654435761 & 0xFFFFFFFF))
    count = HOLE_COUNTS[seed % len(HOLE_COUNTS)]
    # Radius, hole-area fraction and colour are sampled independently of the
    # answer; ink area is pi*R^2*(1-f), so total ink carries no count signal.
    radius = round(rng.uniform(44.0, 58.0), 4)
    frac = round(rng.uniform(0.10, 0.20), 5)
    hole_r = radius * math.sqrt(frac / count)
    colour = BUTTON_COLORS[rng.randrange(len(BUTTON_COLORS))]

    pad = radius + 16.0
    cx = round(rng.uniform(pad, CANVAS - pad), 4)
    cy = round(rng.uniform(pad, CANVAS - pad), 4)

    # Feasible ring radii: holes must clear each other and the rim.
    sep = 2.0 * math.sin(math.pi / count) if count > 1 else 0.0
    ring_hi = radius - 7.0 - hole_r
    tight = rng.random() < 0.28
    gap = rng.uniform(2.6, 8.5) if tight else rng.uniform(9.0, 26.0)
    ring_lo = (2.0 * hole_r + gap) / sep
    if ring_lo > ring_hi:
        ring_lo = (2.0 * hole_r + 2.6) / sep
    ring = round(min(max(ring_lo, 0.0), ring_hi), 4)
    phase = round(rng.uniform(0.0, 2.0 * math.pi), 6)

    holes = _hole_ring(radius, count, hole_r, ring, phase, cx, cy)
    rng.shuffle(holes)

    scene = {
        "width": CANVAS,
        "height": CANVAS,
        "button_center": [cx, cy],
        "button_radius": radius,
        "button_color": list(colour),
        "rim_shade": round(rng.uniform(0.55, 0.70), 4),
        "rim_width": round(rng.uniform(4.0, 8.0), 4),
        "holes": holes,
        "decor": _sample_decor(rng, cx, cy, radius),
    }
    return scene


def _sample_decor(rng: random.Random, bx: float, by: float, br: float) -> list[dict]:
    """Achromatic clutter, including decoy buttons with their own hole counts."""
    items: list[dict] = []
    placed: list[tuple[float, float, float]] = [(bx, by, br + DECOR_CLEARANCE)]
    attempts = 0
    wanted = rng.randrange(4, 8)
    while len(items) < wanted and attempts < 400:
        attempts += 1
        kind = "button" if rng.random() < 0.55 else "spool"
        size = rng.uniform(20.0, 42.0) if kind == "button" else rng.uniform(14.0, 34.0)
        x = rng.uniform(size + 6.0, CANVAS - size - 6.0)
        y = rng.uniform(size + 6.0, CANVAS - size - 6.0)
        if any(math.hypot(x - px, y - py) < size + pr + 8.0 for px, py, pr in placed):
            continue
        placed.append((x, y, size))
        grey = GREYS[rng.randrange(len(GREYS))]
        if kind == "button":
            n = rng.choice((2, 3, 4, 4, 5, 6))
            hr = size * math.sqrt(0.14 / n)
            ring = min(size - 6.0 - hr, max((2.0 * hr + 7.0) / (2.0 * math.sin(math.pi / n)), 0.0))
            items.append(
                {
                    "kind": "button",
                    "center": [round(x, 4), round(y, 4)],
                    "size": round(size, 4),
                    "grey": list(grey),
                    "holes": _hole_ring(size, n, hr, ring, rng.uniform(0, 6.28), x, y),
                }
            )
        else:
            items.append(
                {
                    "kind": "spool",
                    "center": [round(x, 4), round(y, 4)],
                    "size": round(size, 4),
                    "grey": list(grey),
                    "holes": [],
                }
            )
    return items


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def _ellipse(draw: ImageDraw.ImageDraw, x: float, y: float, r: float, fill) -> None:
    draw.ellipse([(x - r) * SS, (y - r) * SS, (x + r) * SS, (y + r) * SS], fill=fill)


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    img = Image.new("RGB", (w * SS, h * SS), BACKGROUND)
    draw = ImageDraw.Draw(img)

    for item in scene["decor"]:
        gx, gy = item["center"]
        size = item["size"]
        grey = tuple(item["grey"])
        if item["kind"] == "button":
            _ellipse(draw, gx, gy, size, grey)
            _ellipse(draw, gx, gy, size * 0.74, tuple(int(c * 1.18) if c * 1.18 < 236 else 236 for c in grey))
        else:
            draw.rounded_rectangle(
                [(gx - size) * SS, (gy - size * 0.62) * SS, (gx + size) * SS, (gy + size * 0.62) * SS],
                radius=int(size * 0.3 * SS),
                fill=grey,
            )
        for hx, hy, hr in item["holes"]:
            _ellipse(draw, hx, hy, hr, HOLE_RGB)

    cx, cy = scene["button_center"]
    radius = scene["button_radius"]
    colour = tuple(scene["button_color"])
    rim = tuple(max(0, int(c * scene["rim_shade"])) for c in colour)
    _ellipse(draw, cx, cy, radius, rim)
    _ellipse(draw, cx, cy, radius - scene["rim_width"], colour)
    for hx, hy, hr in scene["holes"]:
        _ellipse(draw, hx, hy, hr, HOLE_RGB)

    return img.resize((w, h), Image.BOX)


# --------------------------------------------------------------------------
# analytic gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """The answer is the number of openings punched through the coloured disc."""
    return str(len(scene["holes"]))


def margin(scene: dict) -> float:
    """Distance (px) to the nearest configuration change of the visible count.

    The count changes if two holes merge, if a hole breaks the rim, or if a hole
    shrinks below visibility; the runner-up decision is therefore one hole more
    or one hole fewer, and this is the clearance to that event.
    """
    cx, cy = scene["button_center"]
    radius = float(scene["button_radius"])
    holes = scene["holes"]
    terms = [2.0 * min(h[2] for h in holes) - MIN_VISIBLE_HOLE_DIAMETER]
    for hx, hy, hr in holes:
        terms.append(radius - (math.hypot(hx - cx, hy - cy) + hr))
    for i in range(len(holes)):
        for j in range(i + 1, len(holes)):
            ax, ay, ar = holes[i]
            bx, by, br = holes[j]
            terms.append(math.hypot(ax - bx, ay - by) - ar - br)
    for item in scene["decor"]:
        gx, gy = item["center"]
        terms.append(math.hypot(gx - cx, gy - cy) - item["size"] - radius)
    return round(float(min(terms)), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The stored order of the hole list is a genuine, unobservable alias.

    Holes are pairwise disjoint and painted in one colour, so any permutation of
    the list renders to the identical raster while the latent record differs.
    """
    out: list[tuple[str, dict]] = []
    holes = scene["holes"]
    if len(holes) < 2:
        return out
    reversed_scene = dict(scene)
    reversed_scene["holes"] = list(reversed(holes))
    out.append(("hole_order_reversed", reversed_scene))
    rotated = dict(scene)
    rotated["holes"] = holes[1:] + holes[:1]
    out.append(("hole_order_rotated", rotated))
    return out
