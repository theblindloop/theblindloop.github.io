"""Renderer for a structural intervention on last-writer provenance."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

CANVAS = (960, 620)
BOARD_W, BOARD_H = 920, 580
REGISTERS = ("A", "B", "C", "D")
PAIRS = tuple((a, b) for i, a in enumerate(REGISTERS) for b in REGISTERS[i + 1 :])
RGB = {
    "white": (255, 255, 255), "board": (246, 248, 244), "card": (255, 255, 252),
    "ink": (37, 43, 49), "muted": (105, 112, 118), "border": (55, 64, 72),
    "intervene": (204, 43, 55), "new": (226, 158, 24),
    "A": (198, 82, 52), "B": (22, 139, 150), "C": (121, 75, 180), "D": (65, 104, 174),
}
FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _dependencies(pairs: list[list[str]]) -> list[set[int]]:
    last: dict[str, int] = {}
    deps: list[set[int]] = []
    for index, pair in enumerate(pairs):
        parents = {last[box] for box in pair if box in last}
        deps.append(parents)
        for box in pair:
            last[box] = index
    return deps


def _descendants(pairs: list[list[str]], root: int) -> tuple[set[int], dict[int, int]]:
    deps = _dependencies(pairs)
    descendants = {root}
    depth = {root: 0}
    for index in range(root + 1, len(pairs)):
        hit = deps[index] & descendants
        if hit:
            descendants.add(index)
            depth[index] = 1 + max(depth[parent] for parent in hit)
    return descendants - {root}, depth


def _analysis(scene: dict) -> tuple[set[int], set[int], set[int], dict[int, int]]:
    pairs = [[str(v) for v in pair] for pair in scene["pairs"]]
    root = int(scene["intervention_card"])
    changed = [list(pair) for pair in pairs]
    changed[root] = [str(v) for v in scene["replacement_pair"]]
    before, depth = _descendants(pairs, root)
    after, _ = _descendants(changed, root)
    return before, after, before - after, depth


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xA11CE5EED) & 0xFFFFFFFFFFFFFFFF)
    wanted = 4 + int(seed) % 3
    for _ in range(200000):
        pairs = [list(rng.choice(PAIRS)) for _ in range(7)]
        root = rng.randrange(1, 3)
        replacement = list(rng.choice([pair for pair in PAIRS if list(pair) != pairs[root]]))
        scene = {
            "pairs": pairs, "intervention_card": root, "replacement_pair": replacement,
            "anchor_x": rng.randint(18, 22), "anchor_y": rng.randint(18, 22),
        }
        before, after, lost, depth = _analysis(scene)
        gained = after - before
        if lost != {wanted} or not gained:
            continue
        _validate(scene)
        return scene
    raise RuntimeError("could not sample a deep provenance-rewiring scene")


def _validate(scene: dict) -> None:
    pairs = [[str(v) for v in pair] for pair in scene["pairs"]]
    root = int(scene["intervention_card"])
    replacement = [str(v) for v in scene["replacement_pair"]]
    if len(pairs) != 7 or root not in (1, 2):
        raise ValueError("invalid card count or intervention position")
    if any(tuple(pair) not in PAIRS for pair in pairs) or tuple(replacement) not in PAIRS:
        raise ValueError("invalid box pair")
    if replacement == pairs[root]:
        raise ValueError("replacement must change the red card")
    before, after, lost, depth = _analysis(scene)
    if len(lost) != 1 or next(iter(lost)) not in {4, 5, 6}:
        raise ValueError("scene lacks a unique eligible lost descendant")
    target = next(iter(lost))
    if not (after - before):
        raise ValueError("rewiring does not both lose and gain descendants")


def analytic_gold(scene: dict) -> str:
    _validate(scene)
    _, _, lost, _ = _analysis(scene)
    return str(next(iter(lost)) + 1)


def margin(scene: dict) -> float:
    try:
        return 1.0 if analytic_gold(scene) in {"5", "6", "7"} else 0.0
    except (KeyError, TypeError, ValueError):
        return 0.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False):
    return ImageFont.truetype(BOLD if bold else FONT, size)


def _text(draw, xy, value, size, *, fill=None, bold=False, anchor="mm"):
    draw.text(xy, value, font=_font(size, bold), fill=fill or RGB["ink"], anchor=anchor)


def _badge(draw, center, label: str, large: bool = False) -> None:
    x, y = center
    w, h = ((52, 38) if large else (42, 31))
    draw.rounded_rectangle((x-w//2, y-h//2, x+w//2, y+h//2), radius=6,
                           fill=RGB[label], outline=RGB["ink"], width=2)
    _text(draw, center, label, 20 if large else 17, fill=RGB["white"], bold=True)


def render(scene: dict) -> Image.Image:
    _validate(scene)
    ax, ay = int(scene["anchor_x"]), int(scene["anchor_y"])
    image = Image.new("RGB", CANVAS, RGB["white"])
    draw = ImageDraw.Draw(image)
    draw.rectangle((ax, ay, ax+BOARD_W-1, ay+BOARD_H-1), fill=RGB["board"], outline=RGB["border"], width=4)
    _text(draw, (ax+460, ay+24), "REWIRE THE DEPENDENCY GRAPH", 22, bold=True)
    draw.rounded_rectangle((ax+42, ay+48, ax+878, ay+172), radius=10,
                           fill=(255, 252, 238), outline=RGB["new"], width=3)
    _text(draw, (ax+460, ay+70), "For EACH box, look LEFT for the latest earlier card touching that box.", 16, bold=True)
    _text(draw, (ax+460, ay+100), "Those card(s) are dependencies. Follow them repeatedly for descendants.", 16)
    _text(draw, (ax+460, ay+132), "Compare BEFORE (OLD red card) with AFTER (NEW red card).", 17, fill=RGB["intervene"], bold=True)
    _text(draw, (ax+460, ay+158), "Which of cards 5, 6, 7 leaves the red card's descendants?", 16, bold=True)
    root = int(scene["intervention_card"])
    for index, pair in enumerate(scene["pairs"]):
        x, y = ax+20+index*128, ay+192
        selected = index == root
        outline = RGB["intervene"] if selected else RGB["border"]
        draw.rounded_rectangle((x, y, x+112, y+250), radius=9, fill=RGB["card"], outline=outline,
                               width=5 if selected else 3)
        draw.ellipse((x+38, y+10, x+74, y+46), fill=RGB["intervene"] if selected else RGB["ink"])
        _text(draw, (x+56, y+28), str(index+1), 17, fill=RGB["white"], bold=True)
        if selected:
            _text(draw, (x+56, y+59), "OLD", 13, fill=RGB["intervene"], bold=True)
        _badge(draw, (x+56, y+96), str(pair[0]), large=True)
        _text(draw, (x+56, y+142), "↕", 27, bold=True)
        _badge(draw, (x+56, y+188), str(pair[1]), large=True)
        _text(draw, (x+56, y+229), "TOUCH", 13, fill=RGB["muted"], bold=True)
    draw.rounded_rectangle((ax+210, ay+458, ax+710, ay+558), radius=10,
                           fill=(255, 249, 224), outline=RGB["new"], width=4)
    _text(draw, (ax+298, ay+481), "WRENCH", 16, fill=RGB["intervene"], bold=True)
    _text(draw, (ax+298, ay+521), f"CARD {root+1}: NEW", 18, fill=RGB["intervene"], bold=True)
    _badge(draw, (ax+470, ay+520), str(scene["replacement_pair"][0]), large=True)
    _text(draw, (ax+530, ay+520), "↕", 27, bold=True)
    _badge(draw, (ax+590, ay+520), str(scene["replacement_pair"][1]), large=True)
    return image
