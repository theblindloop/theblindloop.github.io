"""Independent pixel-only inverse for provenance rewiring."""
from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (960, 620)
BOARD_W, BOARD_H = 920, 580
BORDER = (55, 64, 72)
RED = (204, 43, 55)
INK = (37, 43, 49)
REGISTERS = ("A", "B", "C", "D")
BADGES = {(198, 82, 52): "A", (22, 139, 150): "B", (121, 75, 180): "C", (65, 104, 174): "D"}
PAIRS = tuple((a, b) for i, a in enumerate(REGISTERS) for b in REGISTERS[i + 1 :])


class OracleError(ValueError):
    pass


def _patch(array: np.ndarray, x: int, y: int, radius: int = 2) -> tuple[int, int, int]:
    patch = array[y-radius:y+radius+1, x-radius:x+radius+1]
    colors = {tuple(int(v) for v in row) for row in patch.reshape(-1, 3)}
    if len(colors) != 1:
        raise OracleError("a required marker is damaged")
    return next(iter(colors))


def _badge(array: np.ndarray, x: int, y: int) -> str:
    color = _patch(array, x-18, y)
    if color not in BADGES:
        raise OracleError("unknown box badge")
    return BADGES[color]


def _dependencies(pairs: list[tuple[str, str]]) -> list[set[int]]:
    last: dict[str, int] = {}
    deps = []
    for index, pair in enumerate(pairs):
        parents = {last[box] for box in pair if box in last}
        deps.append(parents)
        for box in pair:
            last[box] = index
    return deps


def _descendants(pairs: list[tuple[str, str]], root: int) -> tuple[set[int], dict[int, int]]:
    deps = _dependencies(pairs)
    descendants = {root}
    depth = {root: 0}
    for index in range(root+1, len(pairs)):
        hit = deps[index] & descendants
        if hit:
            descendants.add(index)
            depth[index] = 1 + max(depth[parent] for parent in hit)
    return descendants - {root}, depth


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if (array.shape[1], array.shape[0]) != CANVAS:
        raise OracleError("unexpected canvas size")
    mask = np.all(array == np.asarray(BORDER, dtype=np.uint8), axis=2)
    points = np.argwhere(mask)
    if not len(points):
        raise OracleError("board absent")
    top, left = points.min(axis=0); bottom, right = points.max(axis=0)
    if int(right-left+1) != BOARD_W or int(bottom-top+1) != BOARD_H:
        raise OracleError("wrong board extent")
    ax, ay = int(left), int(top)
    if not (np.all(mask[ay, ax:ax+BOARD_W]) and np.all(mask[ay+BOARD_H-1, ax:ax+BOARD_W])
            and np.all(mask[ay:ay+BOARD_H, ax]) and np.all(mask[ay:ay+BOARD_H, ax+BOARD_W-1])):
        raise OracleError("broken board border")
    pairs: list[tuple[str, str]] = []
    selectors = []
    for index in range(7):
        x, y = ax+20+index*128, ay+192
        selectors.append(_patch(array, x+43, y+28))
        pair = (_badge(array, x+56, y+96), _badge(array, x+56, y+188))
        if pair not in PAIRS:
            raise OracleError("malformed original pair")
        pairs.append(pair)
    selected = [index for index, color in enumerate(selectors) if color == RED]
    if len(selected) != 1 or any(color not in {RED, INK} for color in selectors):
        raise OracleError("malformed intervention selector")
    root = selected[0]
    if root not in (1, 2):
        raise OracleError("intervention outside supported positions")
    replacement = (_badge(array, ax+470, ay+520), _badge(array, ax+590, ay+520))
    if replacement not in PAIRS or replacement == pairs[root]:
        raise OracleError("malformed replacement pair")
    changed = list(pairs)
    changed[root] = replacement
    before, depth = _descendants(pairs, root)
    after, _ = _descendants(changed, root)
    lost = before - after
    if len(lost) != 1 or next(iter(lost)) not in {4, 5, 6}:
        raise OracleError("no unique eligible lost descendant")
    target = next(iter(lost))
    if not (after - before):
        raise OracleError("rewiring does not both lose and gain descendants")
    return str(target + 1)
