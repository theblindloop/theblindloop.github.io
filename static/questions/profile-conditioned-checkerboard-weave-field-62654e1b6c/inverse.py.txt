"""Independent pixel-only inverse for the checkerboard-weave world.

This module intentionally knows only the public color palette and the visual
question. It segments each color, measures connected bar components, estimates
each bar's unoriented line angle with PCA, bins bars into their visible rows and
columns, and scores alternation along both axes.
"""

from __future__ import annotations

import math
from collections import deque

from PIL import Image


PALETTE = {
    "red": (220, 70, 70),
    "blue": (55, 105, 215),
    "green": (55, 165, 95),
}


def _mask(image: Image.Image, target: tuple[int, int, int]) -> list[list[bool]]:
    rgb = image.convert("RGB")
    pixels = rgb.load()
    width, height = rgb.size
    result = [[False] * width for _ in range(height)]
    for y in range(height):
        row = result[y]
        for x in range(width):
            r, g, b = pixels[x, y]
            distance = (r - target[0]) ** 2 + (g - target[1]) ** 2 + (b - target[2]) ** 2
            row[x] = distance <= 18 * 18
    return result


def _components(mask: list[list[bool]]) -> list[list[tuple[int, int]]]:
    height = len(mask)
    width = len(mask[0]) if height else 0
    seen = [[False] * width for _ in range(height)]
    found = []
    for y in range(height):
        for x in range(width):
            if not mask[y][x] or seen[y][x]:
                continue
            seen[y][x] = True
            queue = deque([(x, y)])
            component = []
            while queue:
                px, py = queue.popleft()
                component.append((px, py))
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny][nx] and not seen[ny][nx]:
                        seen[ny][nx] = True
                        queue.append((nx, ny))
            if len(component) >= 18:
                found.append(component)
    return found


def _centroid(component: list[tuple[int, int]]) -> tuple[float, float]:
    return (
        sum(x for x, _ in component) / len(component),
        sum(y for _, y in component) / len(component),
    )


def _line_angle(component: list[tuple[int, int]]) -> float:
    cx, cy = _centroid(component)
    xx = yy = xy = 0.0
    for x, y in component:
        dx = x - cx
        dy = y - cy
        xx += dx * dx
        yy += dy * dy
        xy += dx * dy
    angle = 0.5 * math.degrees(math.atan2(2.0 * xy, xx - yy))
    while angle <= -90.0:
        angle += 180.0
    while angle > 90.0:
        angle -= 180.0
    return angle


def _field_score(components: list[list[tuple[int, int]]]) -> float | None:
    if len(components) != 36:
        return None
    points = [(_centroid(component), _line_angle(component)) for component in components]
    points.sort(key=lambda item: (item[0][1], item[0][0]))
    rows = []
    for start in range(0, 36, 6):
        row = sorted(points[start : start + 6], key=lambda item: item[0][0])
        rows.append([1 if angle >= 0.0 else 0 for (_, angle) in row])
    horizontal_total = 0
    horizontal_changes = 0
    vertical_total = 0
    vertical_changes = 0
    for row in rows:
        for left, right in zip(row, row[1:]):
            horizontal_total += 1
            horizontal_changes += int(left != right)
    for row_a, row_b in zip(rows, rows[1:]):
        for upper, lower in zip(row_a, row_b):
            vertical_total += 1
            vertical_changes += int(upper != lower)
    if not horizontal_total or not vertical_total:
        return None
    horizontal = horizontal_changes / horizontal_total
    vertical = vertical_changes / vertical_total
    return horizontal + vertical


def decision_from_image(image: Image.Image) -> str:
    """Return the color with the strongest two-axis alternation, or ambiguity."""

    scores = []
    for color, target in PALETTE.items():
        score = _field_score(_components(_mask(image, target)))
        if score is None:
            return "ambiguous"
        scores.append((score, color))
    scores.sort(key=lambda item: (item[0], item[1]), reverse=True)
    if len(scores) < 2 or scores[0][0] - scores[1][0] < 0.20:
        return "ambiguous"
    return scores[0][1]
