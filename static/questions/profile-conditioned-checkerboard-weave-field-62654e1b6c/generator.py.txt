"""Renderer and analytic target for the checkerboard-weave question world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


RENDERER_VERSION = "checkerboard_weave_field-0.1.0"
COLORS = {
    "red": (220, 70, 70),
    "blue": (55, 105, 215),
    "green": (55, 165, 95),
}
PATTERNS = ("checkerboard", "row_stripes", "column_stripes")


def _bit(pattern: str, row: int, col: int, phase: int) -> int:
    if pattern == "checkerboard":
        return (row + col + phase) % 2
    if pattern == "row_stripes":
        return (row + phase) % 2
    if pattern == "column_stripes":
        return (col + phase) % 2
    raise ValueError(f"unknown pattern {pattern!r}")


def sample_scene(seed: int) -> dict:
    """Return one balanced randomized scene from an integer seed."""

    rng = random.Random(int(seed))
    colors = ["red", "blue", "green"]
    answer_color = colors[int(seed) % len(colors)]
    decoy_patterns = ["row_stripes", "column_stripes"]
    rng.shuffle(decoy_patterns)
    pattern_for_color = {answer_color: "checkerboard"}
    other_colors = [color for color in colors if color != answer_color]
    for color, pattern in zip(other_colors, decoy_patterns):
        pattern_for_color[color] = pattern

    slot_x = [10, 215, 420]
    panels = []
    color_order = list(colors)
    rng.shuffle(color_order)
    for base_x, color in zip(slot_x, color_order):
        panels.append(
            {
                "color": color,
                "pattern": pattern_for_color[color],
                "phase": rng.randrange(2),
                "x": base_x + rng.randint(-5, 5),
                "y": 18 + rng.randint(-3, 3),
                "spacing_x": round(rng.uniform(30.0, 33.0), 3),
                "spacing_y": round(rng.uniform(34.0, 37.0), 3),
                "bar_length": round(rng.uniform(18.0, 22.0), 3),
                "stroke_width": rng.choice([4, 5]),
                "angle_center": round(rng.uniform(-8.0, 8.0), 3),
                "angle_delta": round(rng.uniform(25.0, 31.0), 3),
            }
        )
    # The list order has no visual meaning. Keeping it randomized gives the
    # tested panel-record permutation a genuine, visible-set interpretation.
    rng.shuffle(panels)
    return {
        "width": 660,
        "height": 260,
        "rows": 6,
        "cols": 6,
        "background": "white",
        "panels": panels,
    }


def _bar_center(panel: dict, row: int, col: int) -> tuple[float, float]:
    return (
        float(panel["x"]) + 20.0 + col * float(panel["spacing_x"]),
        float(panel["y"]) + 22.0 + row * float(panel["spacing_y"]),
    )


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields into a deterministic RGB raster."""

    width = int(scene["width"])
    height = int(scene["height"])
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    image = Image.new("RGB", (width, height), scene["background"])
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        color = COLORS[panel["color"]]
        phase = int(panel["phase"])
        for row in range(rows):
            for col in range(cols):
                bit = _bit(panel["pattern"], row, col, phase)
                angle = float(panel["angle_center"]) + (
                    float(panel["angle_delta"]) if bit else -float(panel["angle_delta"])
                )
                radians = math.radians(angle)
                cx, cy = _bar_center(panel, row, col)
                dx = 0.5 * float(panel["bar_length"]) * math.cos(radians)
                dy = 0.5 * float(panel["bar_length"]) * math.sin(radians)
                draw.line(
                    [(round(cx - dx), round(cy - dy)), (round(cx + dx), round(cy + dy))],
                    fill=color,
                    width=int(panel["stroke_width"]),
                )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the color bound to the checkerboard pattern."""

    for panel in scene["panels"]:
        if panel["pattern"] == "checkerboard":
            return str(panel["color"])
    raise ValueError("scene has no checkerboard panel")


def _axis_score(pattern: str) -> tuple[float, float]:
    """The exact visible neighbor-alternation rates for the three patterns."""

    if pattern == "checkerboard":
        return (1.0, 1.0)
    if pattern == "row_stripes":
        return (0.0, 1.0)
    if pattern == "column_stripes":
        return (1.0, 0.0)
    raise ValueError(f"unknown pattern {pattern!r}")


def margin(scene: dict) -> float:
    """Return the best-versus-runner-up global coherence gap."""

    scores = []
    for panel in scene["panels"]:
        horizontal, vertical = _axis_score(panel["pattern"])
        scores.append((horizontal + vertical, str(panel["color"])))
    values = sorted((score for score, _ in scores), reverse=True)
    if len(values) < 2:
        return 0.0
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    """Exclude only unresolved structural ties from headline evidence."""

    return margin(scene) < 0.20


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse unordered panel records; the non-overlapping raster is unchanged."""

    twin = dict(scene)
    twin["panels"] = list(reversed(scene["panels"]))
    return [("panel_record_permutation", twin)]
