"""Deterministic Pillow world for synchronous local binary evolution."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 500
WHITE = (255, 255, 255)
INK = (25, 34, 49)
BLUE = (31, 91, 214)
TILE = (235, 239, 244)
ORANGE = (239, 125, 32)
CYAN = (37, 190, 207)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size)


def _replay(states: list[int], steps: int) -> list[int]:
    current = list(states)
    for _ in range(steps):
        old = current
        current = [
            (old[i - 1] if i else 0) ^ (old[i + 1] if i + 1 < len(old) else 0)
            for i in range(len(old))
        ]
    return current


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    query_index = rng.randrange(5)
    step_count = 1 + rng.randrange(4)
    desired = "ON" if int(seed) % 2 == 0 else "OFF"
    preferred_count = 2 + rng.randrange(2)
    matching = []
    fallback = []
    for mask in range(32):
        states = [(mask >> i) & 1 for i in range(5)]
        got = "ON" if _replay(states, step_count)[query_index] else "OFF"
        if got == desired:
            fallback.append(states)
            if sum(states) == preferred_count:
                matching.append(states)
    pool = matching or fallback
    initial_states = list(pool[rng.randrange(len(pool))])
    return {
        "initial_states": initial_states,
        "query_index": query_index,
        "step_count": step_count,
    }


def render(scene: dict) -> Image.Image:
    initial_states = [int(v) for v in scene["initial_states"]]
    query_index = int(scene["query_index"])
    step_count = int(scene["step_count"])
    if len(initial_states) != 5 or any(v not in (0, 1) for v in initial_states):
        raise ValueError("initial_states must contain five bits")
    if not 0 <= query_index < 5 or not 1 <= step_count <= 4:
        raise ValueError("query index or step count out of range")

    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.text((40, 28), "SYNCHRONOUS NEIGHBORS", fill=INK, font=_font(28, True))
    draw.text((40, 76), "START", fill=INK, font=_font(22, True))

    centers = [220, 360, 500, 640, 780]
    y = 205
    for i, (x, state) in enumerate(zip(centers, initial_states)):
        draw.rounded_rectangle((x - 48, y - 48, x + 48, y + 48), radius=12, fill=TILE)
        if i == query_index:
            draw.ellipse((x - 41, y - 41, x + 41, y + 41), outline=ORANGE, width=7)
        draw.ellipse(
            (x - 29, y - 29, x + 29, y + 29),
            fill=BLUE if state else WHITE,
            outline=INK,
            width=5,
        )
        draw.text((x - 10, y + 62), str(i + 1), fill=INK, font=_font(18, True))

    draw.text((40, 315), "APPLY LEFT TO RIGHT", fill=INK, font=_font(18, True))
    card_w, gap = 148, 24
    total = step_count * card_w + (step_count - 1) * gap
    left = (WIDTH - total) // 2
    for j in range(step_count):
        x0 = left + j * (card_w + gap)
        draw.rounded_rectangle((x0, 350, x0 + card_w, 425), radius=12, fill=CYAN, outline=INK, width=4)
        draw.text((x0 + 38, 371), "STEP", fill=INK, font=_font(23, True))
    draw.text((40, 452), "All lamps update together from the previous row.", fill=INK, font=_font(17))
    return image


def analytic_gold(scene: dict) -> str:
    initial_states = [int(v) for v in scene["initial_states"]]
    query_index = int(scene["query_index"])
    step_count = int(scene["step_count"])
    return "ON" if _replay(initial_states, step_count)[query_index] else "OFF"


def margin(scene: dict) -> float:
    # One-sided raster confidence: the narrowest decisive role mark is 7 px wide.
    _ = scene
    return 7.0


def is_quarantined(scene: dict) -> bool:
    try:
        states = scene["initial_states"]
        query = scene["query_index"]
        steps = scene["step_count"]
        return not (
            isinstance(states, list)
            and len(states) == 5
            and all(v in (0, 1) for v in states)
            and isinstance(query, int)
            and 0 <= query < 5
            and isinstance(steps, int)
            and 1 <= steps <= 4
        )
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene
    return []
