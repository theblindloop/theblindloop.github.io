"""Independent image-only inverse arm.

It discovers role-colored connected regions and their relative centers. It has
no access to latent scenes or the forward renderer.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

TILE = (235, 239, 244)
BLUE = (31, 91, 214)
ORANGE = (239, 125, 32)
CYAN = (37, 190, 207)


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        todo = deque([(int(y), int(x))])
        seen[y, x] = True
        count = 0
        minx = maxx = int(x)
        miny = maxy = int(y)
        while todo:
            cy, cx = todo.popleft()
            count += 1
            minx, maxx = min(minx, cx), max(maxx, cx)
            miny, maxy = min(miny, cy), max(maxy, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        if count >= minimum:
            found.append((minx, miny, maxx, maxy, count))
    return found


def _mask(rgb: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(rgb == np.asarray(color, dtype=np.uint8), axis=2)


def _horizontal_regions(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Group broad role-colored columns even when an overdrawn ring splits fill."""
    active = np.nonzero(mask.sum(axis=0) > 20)[0]
    if not len(active):
        return []
    runs = []
    start = previous = int(active[0])
    for value in active[1:]:
        value = int(value)
        if value > previous + 1:
            runs.append((start, previous))
            start = value
        previous = value
    runs.append((start, previous))
    regions = []
    for minx, maxx in runs:
        ys, xs = np.nonzero(mask[:, minx : maxx + 1])
        if maxx - minx >= 60 and len(ys) >= 1000:
            regions.append((minx, int(ys.min()), maxx, int(ys.max()), int(len(ys))))
    return regions


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    tiles = _horizontal_regions(_mask(rgb, TILE))
    if len(tiles) != 5:
        raise ValueError(f"expected five lamp tiles, found {len(tiles)}")
    tiles.sort(key=lambda box: (box[0] + box[2]) / 2)

    blue = _mask(rgb, BLUE)
    states = []
    centers = []
    for minx, miny, maxx, maxy, _ in tiles:
        centers.append(((minx + maxx) / 2, (miny + maxy) / 2))
        states.append(1 if int(blue[miny : maxy + 1, minx : maxx + 1].sum()) > 500 else 0)

    rings = _components(_mask(rgb, ORANGE), 500)
    if len(rings) != 1:
        raise ValueError(f"expected one query ring, found {len(rings)}")
    ring = rings[0]
    rc = ((ring[0] + ring[2]) / 2, (ring[1] + ring[3]) / 2)
    distances = [(cx - rc[0]) ** 2 + (cy - rc[1]) ** 2 for cx, cy in centers]
    query = int(np.argmin(distances))
    ordered = sorted(distances)
    if len(ordered) < 2 or ordered[1] - ordered[0] < 400:
        raise ValueError("query ring is not uniquely bound to a lamp")

    cards = _components(_mask(rgb, CYAN), 1000)
    steps = len(cards)
    if not 1 <= steps <= 4:
        raise ValueError(f"expected one to four STEP cards, found {steps}")

    current = states
    for _ in range(steps):
        old = current
        current = [
            (old[i - 1] if i else 0) ^ (old[i + 1] if i + 1 < len(old) else 0)
            for i in range(len(old))
        ]
    return "ON" if current[query] else "OFF"
