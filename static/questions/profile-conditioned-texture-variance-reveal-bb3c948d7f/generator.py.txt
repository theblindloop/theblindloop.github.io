"""texture_variance_reveal: local-variance (texture-energy) glyph reveal.

Declared transform: every 16x16 lattice block has the SAME mean brightness, but
blocks that lie inside a coarse diagonal-band glyph carry a strong checkerboard
texture (a high second-moment luminance modulation) while blocks outside the
band carry only a low-dispersion gray-noise texture. The declared operation
computes the local luminance VARIANCE of every block (a second-order, per-block
texture-energy statistic); that map is flat-mean but high where the band was
drawn, which reconstructs a clean diagonal band whose rising (A) or falling (B)
orientation is the answer. In the raw raster any single block, row, column, or
local patch reads as gray noise whose mean is constant across the whole field,
so the glyph is not readable from any local brightness, position, or count cue;
only the declared per-block variance (texture energy) exposes it.

The glyph is a diagonal band: A = rising '/' (anti-diagonal, r+c ~ const),
B = falling '\\' (main diagonal, r ~ c).

Latent scene z = {seedc (glyph/class seed), glyph, seedn (noise seed)}.
label = glyph.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "texture_variance_reveal-0.1.0"

W = 512
H = 512
NGRID = 32          # blocks per side
CELL = W // NGRID   # 16
BASE = 128.0
NOISE_SIGMA = 22.0  # low-dispersion gray noise, present on every block
CHECK_AMP = 48.0    # checkerboard amplitude added only inside the band
BAND_WIDTH = 5      # half-width of the diagonal band in block units
Q_TOL = 0.15


def band_masks() -> dict[str, np.ndarray]:
    """(NGRID, NGRID) bool masks: A rising '/', B falling '\\'."""
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def build_scene(seedc: int, glyph: str, seedn: int) -> dict:
    return {"seedc": int(seedc), "glyph": glyph, "seedn": int(seedn)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    return build_scene(seedc=int(seed), glyph=glyph, seedn=int(seed) + 91)


def _glyph_mask(glyph: str) -> np.ndarray:
    return band_masks()[glyph]


def render_matrix(scene: dict) -> np.ndarray:
    """(H, W) float luminance raster on which the block variance is later read.

    Every pixel carries an independent low-dispersion gray noise so that, seen
    at block granularity, the mean of every 16x16 block is ~BASE regardless of
    glyph membership. Where the block lies inside the band, a zero-mean
    checkerboard modulation is added at PIXEL resolution, which raises only the
    block's local second moment (variance) and leaves its mean unchanged; so the
    whole field is flat-mean gray noise and the band is exposed only by the
    declared per-block variance operator."""
    inside_mask = _glyph_mask(scene["glyph"]).astype(np.float64)
    rng = np.random.default_rng(int(scene["seedn"]) & 0xFFFFFFFF)
    # per-pixel low-dispersion gray noise, mean BASE
    pix = rng.normal(BASE, NOISE_SIGMA, size=(W, H))
    # zero-mean checkerboard at pixel resolution, present only inside the band
    yy, xx = np.indices((W, H))
    check_sign = np.where(((yy + xx) % 2) == 0, 1.0, -1.0)
    # expand the inside band mask from block to pixel resolution
    inside_pix = np.repeat(np.repeat(inside_mask, CELL, axis=0), CELL, axis=1)
    pix = pix + CHECK_AMP * check_sign * inside_pix
    return pix[:W, :H]


def render(scene: dict) -> Image.Image:
    arr = render_matrix(scene)
    arr = np.clip(arr, 0, 255)
    canvas = np.repeat(arr[:, :, None], 3, axis=2).astype(np.uint8)
    canvas = canvas[:W, :H]
    return Image.fromarray(canvas, "RGB")


def margin(scene: dict) -> float:
    chosen = _glyph_mask(scene["glyph"])
    other = _glyph_mask("B" if scene["glyph"] == "A" else "A")
    o_filled = int(other.sum())
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(1.0 - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field (the class seed, the answer glyph, and
    the noise seed) directly changes the rendered pixels, so no distinct latent
    vector reproduces the same image. Declared empty."""
    return iter(())
