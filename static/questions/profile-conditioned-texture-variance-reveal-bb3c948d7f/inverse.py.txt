"""Independent pixel-only inverse arm for texture_variance_reveal.

Receives ONLY the final PNG. It:
  1. reads the luminance of every pixel;
  2. computes the local luminance VARIANCE of every 16x16 block (the declared
     second-order texture-energy operator);
  3. correlates that variance map against the two declared diagonal-band
     templates A (rising) and B (falling);
  4. reports the template whose inside region carries the higher relative
     variance contrast.

No scene data, no analytic gold, no renderer imports.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "texture_variance_reveal-oracle-0.1.0"

W = 512
NGRID = 32
CELL = W // NGRID
BAND_WIDTH = 5


def _band_masks() -> dict[str, np.ndarray]:
    r, c = np.indices((NGRID, NGRID))
    mask_a = np.abs((r + c) - (NGRID - 1)) <= BAND_WIDTH
    mask_b = np.abs(r - c) <= BAND_WIDTH
    return {"A": mask_a, "B": mask_b}


def _block_variances(image: Image.Image) -> np.ndarray:
    """(NGRID, NGRID) map of the local luminance variance of every block."""
    arr = np.asarray(image.convert("L"), dtype=np.float64)
    out = np.zeros((NGRID, NGRID), dtype=np.float64)
    for i in range(NGRID):
        for j in range(NGRID):
            block = arr[i * CELL:(i + 1) * CELL, j * CELL:(j + 1) * CELL]
            out[i, j] = block.var()
    return out


def _contrast(var_map: np.ndarray, mask: np.ndarray) -> float:
    inside = var_map[mask].mean()
    outside = var_map[~mask].mean()
    return float(inside - outside)


def decision_from_image(image: Image.Image) -> str:
    var_map = _block_variances(image)
    masks = _band_masks()
    sa = _contrast(var_map, masks["A"])
    sb = _contrast(var_map, masks["B"])
    return "A" if sa >= sb else "B"


def reveal_margin(image: Image.Image) -> float:
    var_map = _block_variances(image)
    masks = _band_masks()
    sa = _contrast(var_map, masks["A"])
    sb = _contrast(var_map, masks["B"])
    return abs(sa - sb)
