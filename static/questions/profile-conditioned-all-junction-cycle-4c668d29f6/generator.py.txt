"""Deterministic analytic and raster arms for the all-junction-cycle world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


CANVAS_SIZE = 512
BACKGROUND = [248, 248, 244]
LINK_COLOR = (38, 42, 48)
NODE_COLORS = [
    [42, 111, 190],
    [39, 139, 91],
    [126, 78, 184],
    [184, 91, 38],
]

# A pentagonal prism.  It is cubic and has an obvious (analytic, not styled)
# ten-vertex cycle around the two pentagons and two of their matching spokes.
_SPANNING_NODES = [
    (256, 62), (437, 197), (368, 410), (144, 410), (75, 197),
    (256, 164), (343, 228), (310, 330), (202, 330), (169, 228),
]
_SPANNING_EDGES = [
    (0, 1), (1, 2), (2, 3), (3, 4), (4, 0),
    (5, 6), (6, 7), (7, 8), (8, 9), (9, 5),
    (0, 5), (1, 6), (2, 7), (3, 8), (4, 9),
]

# Two subdivided planar K4 graphs joined by one bridge.  Subdivision vertices
# 4 and 9 have degree two within their lobes and degree three after adding the
# bridge.  The bridge cannot belong to a cycle, so no cycle spans all vertices.
# Counts and the complete degree sequence match the positive class exactly.
_BLOCKED_NODES = [
    (55, 256), (175, 90), (175, 422), (123, 225), (210, 256),
    (457, 256), (337, 90), (337, 422), (389, 287), (302, 256),
]
_BLOCKED_EDGES = [
    (0, 1), (0, 2), (0, 3), (1, 3), (2, 3), (1, 4), (4, 2),
    (5, 6), (5, 7), (5, 8), (6, 8), (7, 8), (6, 9), (9, 7),
    (4, 9),
]


def _transform_nodes(points: list[tuple[int, int]], seed: int) -> list[list[int]]:
    """Apply class-blind dihedral, scale, translation, and small jitter nuisances."""
    rng = random.Random(seed ^ 0xC1C1E5)
    quarter_turns = (seed // 2) % 4
    reflect = bool((seed // 8) % 2)
    scale = (0.92, 0.96, 1.0)[(seed // 16) % 3]
    tx = rng.randint(-4, 4)
    ty = rng.randint(-4, 4)
    transformed: list[list[int]] = []
    for x, y in points:
        dx = x - 256
        dy = y - 256
        if reflect:
            dx = -dx
        for _ in range(quarter_turns):
            dx, dy = -dy, dx
        transformed.append([
            int(round(256 + scale * dx + tx + rng.randint(-2, 2))),
            int(round(256 + scale * dy + ty + rng.randint(-2, 2))),
        ])
    return transformed


def sample_scene(seed: int) -> dict:
    """Alternate labels while varying every declared visual nuisance class-blindly."""
    has_spanning_cycle = bool(seed % 2)
    base_nodes = _SPANNING_NODES if has_spanning_cycle else _BLOCKED_NODES
    base_edges = _SPANNING_EDGES if has_spanning_cycle else _BLOCKED_EDGES
    edges = [list(edge) for edge in base_edges]
    random.Random(seed ^ 0xA11CE).shuffle(edges)
    return {
        "nodes": _transform_nodes(base_nodes, seed),
        "edges": edges,
        "node_radius": (14, 15, 16)[(seed // 6) % 3],
        "link_width": (7, 9, 11, 13)[(seed // 2) % 4],
        "node_color": NODE_COLORS[(seed // 8) % len(NODE_COLORS)],
        "background": BACKGROUND,
        "canvas_size": CANVAS_SIZE,
    }


def render(scene: dict) -> Image.Image:
    """Draw straight links first and opaque circular junctions second."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    node_radius = int(scene["node_radius"])
    link_width = int(scene["link_width"])
    node_color = tuple(scene["node_color"])
    background = tuple(scene["background"])
    canvas_size = int(scene["canvas_size"])
    ss = 3
    image = Image.new("RGB", (canvas_size * ss, canvas_size * ss), background)
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        first = tuple(int(value * ss) for value in nodes[a])
        second = tuple(int(value * ss) for value in nodes[b])
        draw.line([first, second], fill=LINK_COLOR, width=link_width * ss)
    for x, y in nodes:
        draw.ellipse(
            [
                (x - node_radius) * ss,
                (y - node_radius) * ss,
                (x + node_radius) * ss,
                (y + node_radius) * ss,
            ],
            fill=node_color,
            outline=LINK_COLOR,
            width=2 * ss,
        )
    return image.resize((canvas_size, canvas_size), Image.Resampling.LANCZOS)


def _has_spanning_cycle(nodes: list, edges: list) -> bool:
    """Exact Hamiltonian-cycle predicate for a small undirected simple graph."""
    count = len(nodes)
    adjacency = [set() for _ in range(count)]
    for raw_a, raw_b in edges:
        a, b = int(raw_a), int(raw_b)
        if a == b or not (0 <= a < count and 0 <= b < count):
            raise ValueError("invalid graph edge")
        adjacency[a].add(b)
        adjacency[b].add(a)
    if count < 3 or any(not neighbors for neighbors in adjacency):
        return False

    start = 0

    def search(current: int, visited: frozenset[int]) -> bool:
        if len(visited) == count:
            return start in adjacency[current]
        return any(
            search(neighbor, visited | {neighbor})
            for neighbor in adjacency[current]
            if neighbor not in visited
        )

    return search(start, frozenset({start}))


def analytic_gold(scene: dict) -> str:
    return "yes" if _has_spanning_cycle(scene["nodes"], scene["edges"]) else "no"


def _point_segment_distance(
    point: tuple[float, float], first: tuple[float, float], second: tuple[float, float]
) -> float:
    px, py = point
    ax, ay = first
    bx, by = second
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def margin(scene: dict) -> float:
    """Raster headroom from thinning and nonincident node/link confusion."""
    nodes = [tuple(map(float, point)) for point in scene["nodes"]]
    radius = float(scene["node_radius"])
    width = float(scene["link_width"])
    clearances = [width - 5.0, radius - 10.0]
    for a, b in scene["edges"]:
        for index, point in enumerate(nodes):
            if index not in (a, b):
                clearances.append(
                    _point_segment_distance(point, nodes[a], nodes[b]) - radius - width / 2.0
                )
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_serialized_edge_order", twin)]
