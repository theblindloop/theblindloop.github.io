"""Independent final-PNG inverse for the all-junction-cycle question.

Only the public visual contract is encoded here: ten saturated circular
junctions, fifteen straight dark links, and degree three at every junction.
The inverse locates junction components, scores every possible straight
corridor, requires a separated 15-link choice, and then performs its own
vertex-spanning-cycle search.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


class AmbiguousRasterError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 180:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _junctions_and_radius(arr: np.ndarray) -> tuple[list[tuple[float, float]], float]:
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    saturated = (high - low > 48) & (high > 90) & (high < 235) & (low < 190)
    groups = _components(saturated)
    if len(groups) != 10:
        raise AmbiguousRasterError(f"expected 10 colored junctions, recovered {len(groups)}")
    centers: list[tuple[float, float]] = []
    radii: list[float] = []
    for points in groups:
        ys, xs = points[:, 0], points[:, 1]
        centers.append((float(xs.mean()), float(ys.mean())))
        radii.append(math.sqrt(len(points) / math.pi))
    return centers, float(np.median(radii))


def _corridor_fraction(
    dark: np.ndarray,
    first: tuple[float, float],
    second: tuple[float, float],
    endpoint_exclusion: float,
) -> float:
    x1, y1 = first
    x2, y2 = second
    distance = math.hypot(x2 - x1, y2 - y1)
    usable = distance - 2 * endpoint_exclusion
    if usable < 12:
        return 0.0
    samples = max(24, int(usable))
    start = endpoint_exclusion / distance
    stop = 1.0 - start
    hits = 0
    height, width = dark.shape
    for t in np.linspace(start, stop, samples):
        x = int(round(x1 + t * (x2 - x1)))
        y = int(round(y1 + t * (y2 - y1)))
        y0, y1w = max(0, y - 2), min(height, y + 3)
        x0, x1w = max(0, x - 2), min(width, x + 3)
        hits += bool(dark[y0:y1w, x0:x1w].any())
    return hits / samples


def _recover_graph(image: Image.Image) -> tuple[list[set[int]], float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    centers, radius = _junctions_and_radius(arr)
    channel_spread = arr.max(axis=2) - arr.min(axis=2)
    dark = (arr.max(axis=2) < 115) & (channel_spread < 35)
    exclusion = max(17.0, radius + 7.0)
    scores: list[tuple[float, int, int]] = []
    for a in range(len(centers)):
        for b in range(a + 1, len(centers)):
            scores.append((
                _corridor_fraction(dark, centers[a], centers[b], exclusion), a, b
            ))
    scores.sort(reverse=True)
    selected_floor = scores[14][0]
    runner_up = scores[15][0]
    decision_margin = selected_floor - runner_up
    if selected_floor < 0.88 or runner_up > 0.70 or decision_margin < 0.20:
        raise AmbiguousRasterError(
            "uncertain link recovery: "
            f"15th={selected_floor:.3f}, runner-up={runner_up:.3f}, gap={decision_margin:.3f}"
        )
    adjacency = [set() for _ in centers]
    for _, a, b in scores[:15]:
        adjacency[a].add(b)
        adjacency[b].add(a)
    degrees = [len(items) for items in adjacency]
    if degrees != [3] * 10:
        raise AmbiguousRasterError(f"expected degree three at every junction, got {degrees}")
    reached = {0}
    queue = [0]
    for vertex in queue:
        for neighbor in adjacency[vertex]:
            if neighbor not in reached:
                reached.add(neighbor)
                queue.append(neighbor)
    if len(reached) != 10:
        raise AmbiguousRasterError("recovered graph is not connected")
    return adjacency, float(decision_margin)


def _has_spanning_cycle(adjacency: list[set[int]]) -> bool:
    count = len(adjacency)
    start = 0

    def search(current: int, visited: frozenset[int]) -> bool:
        if len(visited) == count:
            return start in adjacency[current]
        return any(
            search(neighbor, visited | {neighbor})
            for neighbor in adjacency[current]
            if neighbor not in visited
        )

    return search(start, frozenset({start}))


def decision_and_margin_from_image(image: Image.Image) -> tuple[str, float]:
    adjacency, runner_up_margin = _recover_graph(image)
    decision = "yes" if _has_spanning_cycle(adjacency) else "no"
    return decision, runner_up_margin


def decision_from_image(image: Image.Image) -> str:
    return decision_and_margin_from_image(image)[0]
