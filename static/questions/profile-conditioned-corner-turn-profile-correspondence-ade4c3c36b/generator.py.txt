"""Deterministic renderer and analytic gold for local-turn correspondence."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 520
SUPERSAMPLE = 3
SCALE = 0.72
PANEL_CENTERS = ((160, 292), (480, 292), (800, 292))
LINE_COLOR = (35, 38, 48)
BACKGROUND = (255, 255, 255)
PANEL_FILL = (249, 251, 253)
PANEL_EDGE = (218, 224, 232)
MARKER_RADIUS = 15
LINE_WIDTH = 8
QUARANTINE_CROSS = 2600.0
COLORS = (
    ("red", (218, 65, 70)),
    ("blue", (55, 105, 220)),
    ("green", (45, 165, 92)),
    ("orange", (235, 135, 40)),
    ("purple", (145, 75, 190)),
    ("cyan", (35, 175, 205)),
    ("magenta", (215, 65, 155)),
    ("gold", (222, 180, 42)),
)
PALETTE = dict(COLORS)


def _base_vertices(inset: float) -> list[list[float]]:
    # Counter-clockwise outline with two inward notches at indices 3 and 7.
    return [
        [-140.0, -100.0],
        [0.0, -125.0],
        [140.0, -100.0],
        [float(inset), -30.0],
        [140.0, 100.0],
        [0.0, 125.0],
        [-140.0, 100.0],
        [-float(inset), 20.0],
    ]


def _turn_signature(
    vertices: Iterable[Iterable[float]], colors: Iterable[str]
) -> tuple[dict[str, str], float]:
    points = [(float(p[0]), float(p[1])) for p in vertices]
    labels = list(colors)
    if len(points) != 8 or len(labels) != 8 or len(set(labels)) != 8:
        raise ValueError("a scene must contain eight unique colored vertices")
    area2 = sum(
        points[i][0] * points[(i + 1) % 8][1]
        - points[(i + 1) % 8][0] * points[i][1]
        for i in range(8)
    )
    if abs(area2) < 1e-9:
        raise ValueError("degenerate polygon")
    orientation = 1.0 if area2 > 0 else -1.0
    signature: dict[str, str] = {}
    clearances: list[float] = []
    for i, color in enumerate(labels):
        prev = points[(i - 1) % 8]
        here = points[i]
        nxt = points[(i + 1) % 8]
        cross = (here[0] - prev[0]) * (nxt[1] - here[1]) - (
            here[1] - prev[1]
        ) * (nxt[0] - here[0])
        clearances.append(abs(cross))
        signature[color] = "convex" if cross * orientation > 0 else "reflex"
    return signature, min(clearances)


def _mismatch(left: dict[str, str], right: dict[str, str]) -> int:
    return sum(left[color] != right.get(color) for color in left)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # Use two adjacent residues so the generator's alternating answer labels
    # place near-boundary examples in both classes rather than only one.
    if seed % 10 in (0, 1):
        inset = rng.uniform(129.0, 135.0)
    else:
        inset = rng.uniform(68.0, 118.0)
    vertices = _base_vertices(inset)
    colors = [name for name, _rgb in COLORS]
    rng.shuffle(colors)
    decoy = list(colors)
    decoy[0], decoy[3] = decoy[3], decoy[0]
    match_side = "A" if seed % 2 == 0 else "B"
    candidate_a = list(colors) if match_side == "A" else decoy
    candidate_b = decoy if match_side == "A" else list(colors)
    ref_rotation = rng.randrange(4)
    candidate_rotation = (ref_rotation + rng.choice((1, 2, 3))) % 4
    return {
        "vertices": vertices,
        "reference_colors": colors,
        "candidate_a_colors": candidate_a,
        "candidate_b_colors": candidate_b,
        "ref_rotation": ref_rotation,
        "candidate_rotation": candidate_rotation,
        "notch_inset": inset,
    }


def analytic_gold(scene: dict) -> str:
    reference, _ = _turn_signature(scene["vertices"], scene["reference_colors"])
    candidate_a, _ = _turn_signature(scene["vertices"], scene["candidate_a_colors"])
    candidate_b, _ = _turn_signature(scene["vertices"], scene["candidate_b_colors"])
    score_a = _mismatch(reference, candidate_a)
    score_b = _mismatch(reference, candidate_b)
    return "A" if score_a <= score_b else "B"


def margin(scene: dict) -> float:
    reference, _ = _turn_signature(scene["vertices"], scene["reference_colors"])
    candidate_a, _ = _turn_signature(scene["vertices"], scene["candidate_a_colors"])
    candidate_b, _ = _turn_signature(scene["vertices"], scene["candidate_b_colors"])
    score_a = _mismatch(reference, candidate_a)
    score_b = _mismatch(reference, candidate_b)
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    _signature, clearance = _turn_signature(scene["vertices"], scene["reference_colors"])
    return clearance < QUARANTINE_CROSS or margin(scene) <= 0.0


def _rotate(point: tuple[float, float], quarter_turns: int) -> tuple[float, float]:
    x, y = point
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return x, y


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    vertices: list[list[float]],
    colors: list[str],
    quarter_turns: int,
    supersample: int,
) -> None:
    cx, cy = center

    def screen(point: tuple[float, float]) -> tuple[int, int]:
        x, y = _rotate(point, quarter_turns)
        return (round((cx + SCALE * x) * supersample), round((cy + SCALE * y) * supersample))

    points = [screen((float(x), float(y))) for x, y in vertices]
    draw.line(
        points + [points[0]],
        fill=LINE_COLOR,
        width=LINE_WIDTH * supersample,
        joint="curve",
    )
    for point, color in zip(points, colors):
        radius = MARKER_RADIUS * supersample
        draw.ellipse(
            [point[0] - radius, point[1] - radius, point[0] + radius, point[1] + radius],
            fill=PALETTE[color],
            outline=(255, 255, 255),
            width=2 * supersample,
        )


def render(scene: dict) -> Image.Image:
    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for cx, cy in PANEL_CENTERS:
        draw.rounded_rectangle(
            [(cx - 145) * ss, 40 * ss, (cx + 145) * ss, 480 * ss],
            radius=16 * ss,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2 * ss,
        )
    _draw_panel(
        draw,
        PANEL_CENTERS[0],
        scene["vertices"],
        scene["reference_colors"],
        int(scene["ref_rotation"]),
        ss,
    )
    _draw_panel(
        draw,
        PANEL_CENTERS[1],
        scene["vertices"],
        scene["candidate_a_colors"],
        int(scene["candidate_rotation"]),
        ss,
    )
    _draw_panel(
        draw,
        PANEL_CENTERS[2],
        scene["vertices"],
        scene["candidate_b_colors"],
        int(scene["candidate_rotation"]),
        ss,
    )
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    labels = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for text, (cx, _cy) in zip(("REFERENCE", "A", "B"), PANEL_CENTERS):
        bbox = labels.textbbox((0, 0), text, font=font)
        labels.text((cx - (bbox[2] - bbox[0]) / 2, 52), text, fill=LINE_COLOR, font=font)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    for key in (
        "vertices",
        "reference_colors",
        "candidate_a_colors",
        "candidate_b_colors",
    ):
        values = list(scene[key])
        twin[key] = values[1:] + values[:1]
    return [("cyclic_vertex_origin_shift", twin)]
