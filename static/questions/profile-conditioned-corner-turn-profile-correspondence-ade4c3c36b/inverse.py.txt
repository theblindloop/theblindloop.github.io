"""Independent pixel-only inverse for the colored turn-profile world.

This module deliberately knows only the public raster contract: three panels,
eight uniquely colored markers per panel, and one closed outline. It recovers
marker centroids from RGB pixels, orders them around each panel centroid, and
computes local polygon turn signs. It does not import the renderer or any
latent-scene code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (218, 65, 70),
    "blue": (55, 105, 220),
    "green": (45, 165, 92),
    "orange": (235, 135, 40),
    "purple": (145, 75, 190),
    "cyan": (35, 175, 205),
    "magenta": (215, 65, 155),
    "gold": (222, 180, 42),
}


def _marker_centers(image: Image.Image) -> list[tuple[str, tuple[float, float]]]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    records: list[tuple[str, tuple[float, float]]] = []
    names = list(PALETTE)
    targets = np.asarray([PALETTE[name] for name in names], dtype=np.int32)
    deltas = pixels[:, :, None, :] - targets[None, None, :, :]
    distances2 = (deltas * deltas).sum(axis=3)
    nearest = np.argmin(distances2, axis=2)
    for color_index, name in enumerate(names):
        # Keep a compact interior core.  A larger radius would admit dark
        # antialiased outline pixels that happen to be near a palette color.
        mask = (nearest == color_index) & (distances2[:, :, color_index] < 50.0**2)
        ys, xs = np.nonzero(mask)
        if len(xs) < 30:
            raise ValueError(f"missing marker color {name}")
        unique_x = np.unique(xs)
        cuts = np.where(np.diff(unique_x) > 70)[0] + 1
        x_segments = np.split(unique_x, cuts)
        if len(x_segments) != 3:
            raise ValueError(f"expected three separated {name} markers")
        for segment in x_segments:
            selected = np.isin(xs, segment)
            if int(selected.sum()) < 30:
                raise ValueError(f"small {name} marker component")
            records.append((name, (float(xs[selected].mean()), float(ys[selected].mean()))))
    if len(records) != 24:
        raise ValueError("expected 24 colored marker components")
    return records


def _turn_signature(group: list[tuple[str, tuple[float, float]]]) -> dict[str, int]:
    if len(group) != 8:
        raise ValueError("each panel must contain eight colored markers")
    cx = sum(point[0] for _name, point in group) / 8.0
    cy = sum(point[1] for _name, point in group) / 8.0
    ordered = sorted(
        group,
        key=lambda item: math.atan2(item[1][1] - cy, item[1][0] - cx),
    )
    points = [point for _name, point in ordered]
    names = [name for name, _point in ordered]
    area2 = sum(
        points[i][0] * points[(i + 1) % 8][1]
        - points[(i + 1) % 8][0] * points[i][1]
        for i in range(8)
    )
    if abs(area2) < 100.0:
        raise ValueError("panel polygon is degenerate")
    orientation = 1.0 if area2 > 0 else -1.0
    signature: dict[str, int] = {}
    for i, name in enumerate(names):
        previous = points[(i - 1) % 8]
        here = points[i]
        following = points[(i + 1) % 8]
        cross = (here[0] - previous[0]) * (following[1] - here[1]) - (
            here[1] - previous[1]
        ) * (following[0] - here[0])
        if abs(cross) < 25.0:
            raise ValueError("corner turn is below pixel decision floor")
        signature[name] = 1 if cross * orientation > 0 else -1
    return signature


def _mismatch(left: dict[str, int], right: dict[str, int]) -> int:
    return sum(left[name] != right.get(name) for name in left)


def decision_from_image(image: Image.Image) -> str:
    records = _marker_centers(image)
    records.sort(key=lambda item: item[1][0])
    groups = [records[index : index + 8] for index in range(0, 24, 8)]
    if any(max(point[0] for _name, point in group) - min(point[0] for _name, point in group) > 260 for group in groups):
        raise ValueError("panel markers are not spatially grouped")
    reference = _turn_signature(groups[0])
    candidate_a = _turn_signature(groups[1])
    candidate_b = _turn_signature(groups[2])
    score_a = _mismatch(reference, candidate_a)
    score_b = _mismatch(reference, candidate_b)
    if score_a == score_b:
        raise ValueError("candidate correspondence is ambiguous")
    return "A" if score_a < score_b else "B"
