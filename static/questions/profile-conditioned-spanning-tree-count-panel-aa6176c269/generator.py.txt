"""Render three degree-matched station graphs for a spanning-tree count task."""

from __future__ import annotations

import itertools
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 340
PANEL_X = (20, 335, 650)
PANEL_Y = 20
PANEL_SIZE = 290
CENTER = 140
NODE_R = 12
FRAME = (25, 28, 32)
LINK = (103, 112, 124)
NODE = (37, 168, 154)
BACKGROUND = (250, 249, 246)

# All templates have n=6, m=8 and degree multiset (1,2,3,3,3,4).
EDGES = (
    ((0, 1), (0, 2), (0, 3), (0, 4), (1, 2), (1, 3), (2, 3), (4, 5)),
    ((0, 1), (0, 2), (0, 3), (0, 4), (1, 2), (1, 3), (2, 4), (3, 5)),
    ((0, 1), (0, 2), (0, 3), (0, 4), (1, 2), (1, 3), (2, 5), (3, 5)),
)
POINTS = (
    ((245, 192), (43, 172), (226, 48), (142, 156), (89, 211), (197, 226)),
    ((146, 185), (162, 66), (243, 225), (50, 146), (54, 229), (110, 43)),
    ((234, 146), (103, 88), (222, 36), (106, 243), (195, 242), (64, 52)),
)
PERMUTATIONS = tuple(itertools.permutations(range(3)))


def _transform_point(point: tuple[int, int], code: int) -> tuple[int, int]:
    x, y = point[0] - CENTER, point[1] - CENTER
    if code >= 4:
        x = -x
    for _ in range(code % 4):
        x, y = -y, x
    return x + CENTER, y + CENTER


def _connected(n: int, edges: tuple[tuple[int, int], ...]) -> bool:
    adjacency = [set() for _ in range(n)]
    for u, v in edges:
        adjacency[u].add(v)
        adjacency[v].add(u)
    seen, stack = {0}, [0]
    while stack:
        for v in adjacency[stack.pop()]:
            if v not in seen:
                seen.add(v)
                stack.append(v)
    return len(seen) == n


def _tree_count(edges: tuple[tuple[int, int], ...]) -> int:
    return sum(_connected(6, subset) for subset in itertools.combinations(edges, 5))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    order = list(PERMUTATIONS[seed % len(PERMUTATIONS)])
    transforms = [rng.randrange(8) for _ in range(3)]
    panels = [
        {"points": [list(p) for p in POINTS[i]], "edges": [list(e) for e in EDGES[i]]}
        for i in range(3)
    ]
    return {"panels": panels, "panel_order": order, "transform": transforms}


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    order = scene["panel_order"]
    transforms = scene["transform"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot, template_id in enumerate(order):
        ox, oy = PANEL_X[slot], PANEL_Y
        draw.rounded_rectangle(
            (ox, oy, ox + PANEL_SIZE, oy + PANEL_SIZE), radius=8, outline=FRAME, width=3
        )
        panel = panels[template_id]
        points = [_transform_point(tuple(p), transforms[slot]) for p in panel["points"]]
        points = [(ox + 5 + x, oy + 5 + y) for x, y in points]
        for u, v in panel["edges"]:
            draw.line((points[u], points[v]), fill=LINK, width=7)
        for x, y in points:
            draw.ellipse((x - NODE_R, y - NODE_R, x + NODE_R, y + NODE_R), fill=NODE,
                         outline=FRAME, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    order = scene["panel_order"]
    return ("left", "center", "right")[order.index(1)]


def margin(scene: dict) -> float:
    # Counts are 16, 21, 24; the runner-up is 3 counts from the declared target.
    return float(min(abs(_tree_count(EDGES[i]) - 21) for i in (0, 2)))


def is_quarantined(scene: dict) -> bool:
    # The finite vetted template set has wide clearances and no crossings.
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
