"""Independent inverse arm: recover graphs from RGB pixels and count spanning trees."""

from __future__ import annotations

import itertools
import math
import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        stack = [(int(y), int(x))]
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum:
            output.append(np.asarray(pixels, dtype=np.int32))
    return output


def _near_color(arr: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _panel_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    dark = np.max(arr, axis=2) < 70
    boxes = []
    for component in _components(dark, 500):
        ys, xs = component[:, 0], component[:, 1]
        x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
        if x1 - x0 > 220 and y1 - y0 > 220:
            boxes.append((x0, y0, x1, y1))
    boxes.sort()
    if len(boxes) != 3:
        raise ValueError(f"expected three visible panel frames, found {len(boxes)}")
    return boxes


def _stations(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    cyan = _near_color(crop, (37, 168, 154), 45)
    centers = []
    for component in _components(cyan, 120):
        centers.append((float(component[:, 1].mean() + x0), float(component[:, 0].mean() + y0)))
    if len(centers) != 6:
        raise ValueError(f"expected six visible stations, found {len(centers)}")
    return centers


def _has_link(gray: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    # Test a narrow raster tube along the centerline, excluding station interiors.
    hits = 0
    total = 0
    for t in np.linspace(0.16, 0.84, 55):
        x = a[0] * (1 - t) + b[0] * t
        y = a[1] * (1 - t) + b[1] * t
        xi, yi = int(round(x)), int(round(y))
        patch = gray[max(0, yi - 2):yi + 3, max(0, xi - 2):xi + 3]
        hits += bool(patch.size and patch.any())
        total += 1
    return hits / total >= 0.87


def _edges(arr: np.ndarray, centers: list[tuple[float, float]]) -> tuple[tuple[int, int], ...]:
    gray = _near_color(arr, (103, 112, 124), 35)
    edges = tuple((u, v) for u, v in itertools.combinations(range(6), 2)
                  if _has_link(gray, centers[u], centers[v]))
    if len(edges) != 8:
        raise ValueError(f"expected eight visible links, found {len(edges)}")
    return edges


def _connected(edges: tuple[tuple[int, int], ...]) -> bool:
    adjacency = [set() for _ in range(6)]
    for u, v in edges:
        adjacency[u].add(v)
        adjacency[v].add(u)
    seen, stack = {0}, [0]
    while stack:
        for v in adjacency[stack.pop()]:
            if v not in seen:
                seen.add(v)
                stack.append(v)
    return len(seen) == 6


def _tree_count(edges: tuple[tuple[int, int], ...]) -> int:
    return sum(_connected(subset) for subset in itertools.combinations(edges, 5))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    counts = []
    for box in _panel_boxes(arr):
        centers = _stations(arr, box)
        counts.append(_tree_count(_edges(arr, centers)))
    if counts.count(21) != 1:
        raise ValueError(f"expected unique 21-tree panel, measured {counts}")
    return ("left", "center", "right")[counts.index(21)]
