"""Deterministic scene sampler and Pillow renderer."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

ROWS = COLS = 4
FILLED = 8
BLUE = (42, 111, 187)
DARK = (31, 42, 53)
PALE = (245, 248, 251)
WHITE = (255, 255, 255)
LABELS = ("A", "B", "C", "D")


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _signature(board: list[list[int]]) -> tuple[int, ...]:
    row_counts = tuple(sum(row) for row in board)
    col_counts = tuple(sum(board[r][c] for r in range(ROWS)) for c in range(COLS))
    return row_counts + col_counts


def _valid_board(board: list[list[int]]) -> bool:
    sig = _signature(board)
    return sum(sig[:ROWS]) == FILLED and all(0 < value < 4 for value in sig)


def _random_board(rng: random.Random) -> list[list[int]]:
    while True:
        positions = rng.sample(range(ROWS * COLS), FILLED)
        board = [[0] * COLS for _ in range(ROWS)]
        for position in positions:
            board[position // COLS][position % COLS] = 1
        if _valid_board(board):
            return board


def _switched_copy(board: list[list[int]], rng: random.Random):
    switches = []
    for r1 in range(ROWS):
        for r2 in range(r1 + 1, ROWS):
            for c1 in range(COLS):
                for c2 in range(c1 + 1, COLS):
                    block = (board[r1][c1], board[r1][c2], board[r2][c1], board[r2][c2])
                    if block in ((1, 0, 0, 1), (0, 1, 1, 0)):
                        switches.append((r1, r2, c1, c2))
    if not switches:
        return None
    r1, r2, c1, c2 = rng.choice(switches)
    result = deepcopy(board)
    result[r1][c1], result[r1][c2] = result[r1][c2], result[r1][c1]
    result[r2][c1], result[r2][c2] = result[r2][c2], result[r2][c1]
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        reference = _random_board(rng)
        target = _switched_copy(reference, rng)
        if target is not None and target != reference:
            break
    candidates = []
    answer_index = int(seed) % 4
    distractors = []
    seen = {tuple(map(tuple, reference)), tuple(map(tuple, target))}
    reference_signature = _signature(reference)
    while len(distractors) < 3:
        board = _random_board(rng)
        frozen = tuple(map(tuple, board))
        if frozen in seen or _signature(board) == reference_signature:
            continue
        seen.add(frozen)
        distractors.append(board)
    iterator = iter(distractors)
    for index in range(4):
        candidates.append(target if index == answer_index else next(iterator))
    return {"reference": reference, "candidates": candidates}


def _draw_board(draw: ImageDraw.ImageDraw, board, left: int, top: int, size: int = 164):
    cell = size // 4
    draw.rectangle((left, top, left + size, top + size), fill=WHITE)
    for row in range(ROWS):
        for col in range(COLS):
            if board[row][col]:
                pad = 4
                x0 = left + col * cell + pad
                y0 = top + row * cell + pad
                x1 = left + (col + 1) * cell - pad
                y1 = top + (row + 1) * cell - pad
                draw.rectangle((x0, y0, x1, y1), fill=BLUE)
    for index in range(5):
        x = left + index * cell
        y = top + index * cell
        draw.line((x, top, x, top + size), fill=DARK, width=3)
        draw.line((left, y, left + size, y), fill=DARK, width=3)


def render(scene: dict) -> Image.Image:
    reference = scene["reference"]
    candidates = scene["candidates"]
    image = Image.new("RGB", (1200, 470), PALE)
    draw = ImageDraw.Draw(image)
    title_font = _font(27, True)
    body_font = _font(18)
    label_font = _font(23, True)
    draw.text((600, 20), "MATCH THE ROW + COLUMN COUNT SIGNATURE", fill=DARK,
              font=title_font, anchor="ma")
    draw.text((600, 59), "Count blue cells in each row (top→bottom) and each column (left→right).",
              fill=DARK, font=body_font, anchor="ma")
    draw.text((600, 84), "Choose the candidate with both ordered count lists matching the REFERENCE.",
              fill=DARK, font=body_font, anchor="ma")
    positions = [35, 282, 502, 722, 942]
    top = 155
    draw.text((positions[0] + 82, 131), "REFERENCE", fill=DARK, font=_font(18, True), anchor="ms")
    _draw_board(draw, reference, positions[0], top)
    for index, board in enumerate(candidates):
        left = positions[index + 1]
        _draw_board(draw, board, left, top)
        draw.text((left + 82, top + 203), LABELS[index], fill=DARK, font=label_font, anchor="mm")
    draw.rounded_rectangle((21, 122, 220, 378), radius=12, outline=(130, 143, 156), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidates = scene["candidates"]
    target = _signature(reference)
    matches = [index for index, board in enumerate(candidates) if _signature(board) == target]
    if len(matches) != 1:
        raise ValueError("scene does not have a unique projection-signature match")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    reference = scene["reference"]
    candidates = scene["candidates"]
    target = _signature(reference)
    distances = sorted(sum(abs(a - b) for a, b in zip(_signature(board), target)) for board in candidates)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0
    except (ValueError, IndexError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
