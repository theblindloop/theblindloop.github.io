"""Independent inverse arm using only raster pixels.

It detects the five visible grids from their dark line projections, samples the
blue occupancy inside each detected cell, and compares ordered line-sum vectors.
No renderer state, latent scene, metadata, or fixed board coordinates are read.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(values):
    groups = []
    start = previous = None
    for value in values:
        value = int(value)
        if start is None:
            start = previous = value
        elif value == previous + 1:
            previous = value
        else:
            groups.append((start + previous) / 2.0)
            start = previous = value
    if start is not None:
        groups.append((start + previous) / 2.0)
    return groups


def _occupancy(rgb, x_lines, y_lines):
    board = []
    for row in range(4):
        values = []
        for col in range(4):
            cx = int(round((x_lines[col] + x_lines[col + 1]) / 2))
            cy = int(round((y_lines[row] + y_lines[row + 1]) / 2))
            patch = rgb[max(0, cy - 5):cy + 6, max(0, cx - 5):cx + 6]
            blue = (patch[:, :, 2] > 140) & (patch[:, :, 0] < 100) & (patch[:, :, 1] < 160)
            values.append(1 if blue.mean() > 0.55 else 0)
        board.append(values)
    return board


def _signature(board):
    return tuple(sum(row) for row in board) + tuple(sum(board[r][c] for r in range(4)) for c in range(4))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = (rgb[:, :, 0] < 85) & (rgb[:, :, 1] < 85) & (rgb[:, :, 2] < 85)
    x_centers = _runs(np.flatnonzero(dark.sum(axis=0) > 105))
    y_centers = _runs(np.flatnonzero(dark.sum(axis=1) > 580))
    if len(x_centers) != 25 or len(y_centers) != 5:
        raise ValueError(f"could not detect five 4x4 grids: {len(x_centers)} vertical, {len(y_centers)} horizontal")
    boards = [_occupancy(rgb, x_centers[index * 5:(index + 1) * 5], y_centers) for index in range(5)]
    reference_signature = _signature(boards[0])
    distances = [sum(abs(a - b) for a, b in zip(_signature(board), reference_signature)) for board in boards[1:]]
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("pixel evidence does not identify one signature match")
    runner = sorted(distances)[1]
    if runner < 2:
        raise ValueError("pixel decision has insufficient runner-up margin")
    return ("A", "B", "C", "D")[matches[0]]
