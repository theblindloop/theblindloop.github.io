"""Independent pixel-only inverse for guarded color-register rewrites."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1220
HEIGHT = 700
CARD_X = (295, 505, 715, 925, 925, 715, 505, 295)
CARD_Y = (90, 90, 90, 90, 370, 370, 370, 370)
START_CENTERS = ((78, 260), (180, 260), (78, 445), (180, 445))
RULE_CENTERS = tuple((x + 55, y + 112, x + 135, y + 112) for x, y in zip(CARD_X, CARD_Y))
STATE_COLORS = (
    np.array([235, 157, 39], dtype=np.uint8),
    np.array([34, 167, 190], dtype=np.uint8),
    np.array([139, 85, 190], dtype=np.uint8),
    np.array([112, 176, 73], dtype=np.uint8),
)
INK = np.array([34, 44, 58], dtype=np.uint8)


def _pixel(arr: np.ndarray, center: tuple[int, int]) -> np.ndarray:
    x, y = center
    return arr[y, x]


def _read_state(arr: np.ndarray, center: tuple[int, int]) -> int:
    pixel = _pixel(arr, center)
    for index, color in enumerate(STATE_COLORS):
        if np.array_equal(pixel, color):
            return index
    raise ValueError("state badge is unreadable")


def _apply(state: int, source: int, target: int) -> int:
    return target if state == source else state


def decision_from_image(image: Image.Image) -> str:
    """Read the marked START bay and eight FROM/TO badge pairs, then execute."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb)
    initial_candidates = []
    for cx, cy in START_CENTERS:
        # The token diamond has an exact ink pixel just above the colored chip.
        initial_candidates.append(np.array_equal(arr[cy - 39, cx], INK))
    if sum(initial_candidates) != 1:
        raise ValueError("START token marker is unreadable")
    state = initial_candidates.index(True)
    for source_center_x, source_center_y, target_center_x, target_center_y in RULE_CENTERS:
        source = _read_state(arr, (source_center_x, source_center_y))
        target = _read_state(arr, (target_center_x, target_center_y))
        if source == target:
            raise ValueError("rewrite card has identical FROM and TO")
        state = _apply(state, source, target)
    return ("amber", "cyan", "violet", "lime")[state]
