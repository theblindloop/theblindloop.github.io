"""Renderer and analytic guarded-register state-transition arm."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1220
HEIGHT = 700
START_RECT = (24, 150, 254, 570)
CARD_RECTS = tuple(
    (x, y, x + 190, y + 205)
    for x, y in (
        (295 + 210 * (index if index < 4 else 7 - index), 90 if index < 4 else 370)
        for index in range(8)
    )
)
START_CENTERS = ((78, 260), (180, 260), (78, 445), (180, 445))
RULE_CENTERS = tuple(
    (rect[0] + 55, rect[1] + 112, rect[0] + 135, rect[1] + 112)
    for rect in CARD_RECTS
)

BACKGROUND = (244, 247, 250)
PANEL = (255, 255, 255)
FRAME = (116, 130, 145)
INK = (34, 44, 58)
CONNECTOR = (157, 169, 181)
STATE_COLORS = ((235, 157, 39), (34, 167, 190), (139, 85, 190), (112, 176, 73))
STATE_NAMES = ("amber", "cyan", "violet", "lime")


def _apply(state: int, rule: tuple[int, int] | list[int]) -> int:
    source, target = (int(rule[0]), int(rule[1]))
    return target if state == source else state


def _valid_rule(source: int, target: int) -> bool:
    return 0 <= source < len(STATE_COLORS) and 0 <= target < len(STATE_COLORS) and source != target


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _attempt in range(256):
        initial = rng.randrange(4)
        rules: list[list[int]] = []
        current = initial
        fired = 0
        visited = {current}
        for _ in range(8):
            source = rng.randrange(4)
            target = rng.randrange(3)
            if target >= source:
                target += 1
            rules.append([source, target])
            next_state = _apply(current, (source, target))
            fired += int(next_state != current)
            current = next_state
            visited.add(current)
        if 2 <= fired <= 6 and len(visited) >= 3:
            return {"initial_state": initial, "rules": rules}
    # This deterministic fallback is also a valid, visibly nontrivial process.
    return {
        "initial_state": 0,
        "rules": [[0, 1], [2, 3], [1, 2], [3, 0], [1, 3], [0, 2], [2, 1], [3, 2]],
    }


def _font() -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 18)
    except OSError:
        return ImageFont.load_default()


def _arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = tip
    draw.polygon(((x, y), (x - 12, y - 8), (x - 12, y + 8)), fill=color)


def _left_arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = tip
    draw.polygon(((x, y), (x + 12, y - 8), (x + 12, y + 8)), fill=color)


def _down_arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], color: tuple[int, int, int]) -> None:
    x, y = tip
    draw.polygon(((x, y), (x - 8, y - 12), (x + 8, y - 12)), fill=color)


def _connector(draw: ImageDraw.ImageDraw, start: tuple[float, float], end: tuple[float, float]) -> None:
    draw.line((start, end), fill=CONNECTOR, width=4)
    _arrowhead(draw, end, CONNECTOR)


def _draw_state_chip(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    size: int,
    outline_width: int = 2,
) -> None:
    cx, cy = center
    half = size // 2
    draw.rounded_rectangle(
        (cx - half, cy - half, cx + half, cy + half),
        radius=9,
        fill=STATE_COLORS[state],
        outline=INK,
        width=outline_width,
    )


def _draw_token_marker(draw: ImageDraw.ImageDraw, center: tuple[float, float]) -> None:
    cx, cy = center
    draw.polygon(
        ((cx, cy - 49), (cx + 10, cy - 39), (cx, cy - 29), (cx - 10, cy - 39)),
        fill=INK,
    )


def _draw_order_connectors(draw: ImageDraw.ImageDraw) -> None:
    # The numbered cards form a clear top-row then bottom-row zig-zag.
    for index in range(3):
        left = CARD_RECTS[index]
        right = CARD_RECTS[index + 1]
        _connector(draw, (left[2] + 5, 196), (right[0] - 5, 196))
    top_last = CARD_RECTS[3]
    bottom_first = CARD_RECTS[4]
    draw.line(
        (top_last[0] + 95, top_last[3] + 5, bottom_first[0] + 95, bottom_first[1] - 5),
        fill=CONNECTOR,
        width=4,
    )
    _down_arrowhead(draw, (bottom_first[0] + 95, bottom_first[1] - 5), CONNECTOR)
    for index in range(4, 7):
        left = CARD_RECTS[index]
        right = CARD_RECTS[index + 1]
        draw.line((left[0] - 5, 476, right[2] + 5, 476), fill=CONNECTOR, width=4)
        _left_arrowhead(draw, (right[2] + 5, 476), CONNECTOR)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = _font()

    _draw_order_connectors(draw)
    draw.rounded_rectangle(START_RECT, radius=18, fill=PANEL, outline=FRAME, width=3)
    draw.text((48, 176), "START", fill=INK, font=font)
    draw.text((48, 203), "one token", fill=FRAME, font=font)
    for index, center in enumerate(START_CENTERS):
        _draw_state_chip(draw, center, index, 54, outline_width=3)
        draw.text((center[0] - 20, center[1] + 37), STATE_NAMES[index], fill=INK, font=font)
        if int(scene["initial_state"]) == index:
            _draw_token_marker(draw, center)
    draw.text((47, 520), "colored bays", fill=FRAME, font=font)

    for index, rect in enumerate(CARD_RECTS):
        x0, y0, x1, y1 = rect
        draw.rounded_rectangle(rect, radius=16, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 14, y0 + 14), str(index + 1), fill=FRAME, font=font)
        draw.text((x0 + 39, y0 + 18), "FROM", fill=FRAME, font=font)
        draw.text((x0 + 117, y0 + 18), "TO", fill=FRAME, font=font)
        source, target = (int(scene["rules"][index][0]), int(scene["rules"][index][1]))
        source_center, target_center = RULE_CENTERS[index][:2], RULE_CENTERS[index][2:]
        _draw_state_chip(draw, source_center, source, 50, outline_width=2)
        _draw_state_chip(draw, target_center, target, 50, outline_width=2)
        y = source_center[1]
        draw.line((source_center[0] + 27, y, target_center[0] - 27, y), fill=INK, width=6)
        _arrowhead(draw, (target_center[0] - 25, y), INK)
        draw.text((x0 + 58, y0 + 166), "IF MATCHED", fill=FRAME, font=font)
    draw.text((296, 40), "RULE CARDS — FOLLOW THE NUMBERS", fill=INK, font=font)
    return image


def _final_state(scene: dict) -> int:
    state = int(scene["initial_state"])
    for rule in scene["rules"]:
        state = _apply(state, rule)
    return state


def analytic_gold(scene: dict) -> str:
    return STATE_NAMES[_final_state(scene)]


def margin(scene: dict) -> float:
    # The answer boundary is discrete; this is the deliberate badge clearance.
    return 28.0


def is_quarantined(scene: dict) -> bool:
    if not isinstance(scene.get("initial_state"), int) or not 0 <= scene["initial_state"] < 4:
        return True
    rules = scene.get("rules")
    if not isinstance(rules, list) or len(rules) != 8:
        return True
    return any(
        not isinstance(rule, (list, tuple))
        or len(rule) != 2
        or not all(isinstance(value, int) for value in rule)
        or not _valid_rule(int(rule[0]), int(rule[1]))
        for rule in rules
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
