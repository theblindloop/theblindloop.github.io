"""Raster renderer for the remote-notch pair relation."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


CANVAS = (512, 512)
SUPERSAMPLE = 4
NOTCH_HALF_ANGLE = 0.34
NOTCH_INNER_RATIO = 0.48
PALETTE = ("#17324d", "#8c3b2f", "#416b4e", "#5b477e", "#9a641d")


def _angle_delta(angle_a: float, angle_b: float) -> float:
    """Smallest unsigned angular separation in [0, pi]."""
    raw = (float(angle_b) - float(angle_a)) % (2.0 * math.pi)
    return min(raw, 2.0 * math.pi - raw)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair with randomized visible nuisance parameters."""
    rng = random.Random(int(seed))
    opposite = int(seed) % 2 == 0
    base = rng.uniform(-math.pi, math.pi)
    angle_a = base + rng.uniform(-0.10, 0.10)
    angle_b = base + (math.pi if opposite else 0.0) + rng.uniform(-0.10, 0.10)
    radius = rng.uniform(74.0, 78.0)
    return {
        "angle_a": round(angle_a, 6),
        "angle_b": round(angle_b, 6),
        "cx_a": round(rng.uniform(84.0, 92.0), 3),
        "cy_a": round(rng.uniform(222.0, 290.0), 3),
        "cx_b": round(rng.uniform(420.0, 428.0), 3),
        "cy_b": round(rng.uniform(222.0, 290.0), 3),
        "radius": round(radius, 3),
        "color": rng.choice(PALETTE),
    }


def _notched_medallion(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    radius: float,
    angle: float,
    color: str,
    supersample: int,
) -> None:
    """Draw a connected filled disk with one deep, wide triangular cut."""
    s = float(supersample)
    center = (cx * s, cy * s)
    r = radius * s
    draw.ellipse((center[0] - r, center[1] - r, center[0] + r, center[1] + r), fill=color)

    inner = radius * NOTCH_INNER_RATIO
    outer = radius * 1.18
    points = []
    for distance, theta in (
        (inner, angle - NOTCH_HALF_ANGLE),
        (outer, angle - NOTCH_HALF_ANGLE),
        (outer, angle + NOTCH_HALF_ANGLE),
        (inner, angle + NOTCH_HALF_ANGLE),
    ):
        points.append(
            (
                (cx + distance * math.cos(theta)) * s,
                (cy + distance * math.sin(theta)) * s,
            )
        )
    draw.polygon(points, fill="white")


def render(scene: dict[str, Any]) -> Image.Image:
    """Render exactly two separated, filled, notched medallions."""
    width, height = CANVAS
    ss = SUPERSAMPLE
    image = Image.new("RGB", (width * ss, height * ss), "white")
    draw = ImageDraw.Draw(image)
    color = str(scene["color"])
    radius = float(scene["radius"])
    _notched_medallion(
        draw,
        float(scene["cx_a"]),
        float(scene["cy_a"]),
        radius,
        float(scene["angle_a"]),
        color,
        ss,
    )
    _notched_medallion(
        draw,
        float(scene["cx_b"]),
        float(scene["cy_b"]),
        radius,
        float(scene["angle_b"]),
        color,
        ss,
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """The label is the visible pairwise angular relation."""
    return "yes" if _angle_delta(scene["angle_a"], scene["angle_b"]) > math.pi / 2.0 else "no"


def margin(scene: dict[str, Any]) -> float:
    """Distance from the right-angle decision boundary, in radians."""
    delta = _angle_delta(scene["angle_a"], scene["angle_b"])
    return round(abs(delta - math.pi / 2.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Near-right-angle pairs are finite-resolution boundary cases."""
    return margin(scene) < 0.18


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No distinct latent state is declared pixel-identical."""
    return []
