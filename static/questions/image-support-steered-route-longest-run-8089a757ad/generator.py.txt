"""Deterministic visual world for ordered run-length tracing."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 768, 512
BG = (248, 247, 242)
NAVY = (31, 52, 77)
CYAN = (0, 184, 204)
ORANGE = (239, 139, 32)
GREEN = (43, 168, 92)
RED = (218, 67, 74)
PALE = (222, 225, 224)


def _longest_run(colors: list[str]) -> int:
    best = current = 0
    for value in colors:
        current = current + 1 if value == "cyan" else 0
        best = max(best, current)
    return best


def _point_at(points: list[list[int]], distance: float) -> tuple[float, float]:
    remaining = distance
    for left, right in zip(points, points[1:]):
        length = abs(right[0] - left[0]) + abs(right[1] - left[1])
        if remaining <= length:
            ratio = remaining / length
            return (
                left[0] + ratio * (right[0] - left[0]),
                left[1] + ratio * (right[1] - left[1]),
            )
        remaining -= length
    return tuple(points[-1])


def _route_length(points: list[list[int]]) -> float:
    return float(sum(abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 3 + 1
    eligible: list[list[str]] = []
    for cyan_positions in itertools.combinations(range(10), 5):
        cyan_set = set(cyan_positions)
        colors = ["cyan" if i in cyan_set else "orange" for i in range(10)]
        if _longest_run(colors) == target:
            eligible.append(colors)
    colors = eligible[rng.randrange(len(eligible))]
    if rng.random() < 0.5:
        colors = list(reversed(colors))

    base_points = [(66, 68), (190, 205), (315, 88), (440, 302), (565, 175), (700, 440)]
    points = [[x + rng.randint(-9, 9), y + rng.randint(-9, 9)] for x, y in base_points]
    return {"points": points, "checkpoint_colors": colors}


def render(scene: dict) -> Image.Image:
    points = [[int(x), int(y)] for x, y in scene["points"]]
    colors = list(scene["checkpoint_colors"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for y in (126, 252, 378):
        draw.line((34, y, 734, y), fill=PALE, width=1)
    draw.line([tuple(point) for point in points], fill=NAVY, width=9, joint="curve")

    total = _route_length(points)
    # Ten equal-size markers span the route while leaving visible trace between them.
    fractions = (0.055, 0.145, 0.235, 0.325, 0.415, 0.585, 0.675, 0.765, 0.855, 0.945)
    palette = {"cyan": CYAN, "orange": ORANGE}
    for fraction, color_name in zip(fractions, colors):
        x, y = _point_at(points, total * fraction)
        radius = 13
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette[color_name])
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=(255, 255, 255))

    start = tuple(points[0])
    finish = tuple(points[-1])
    for center, color in ((start, GREEN), (finish, RED)):
        x, y = center
        draw.ellipse((x - 19, y - 19, x + 19, y + 19), fill=color)
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=BG)

    font = ImageFont.load_default()
    sx, sy = start
    fx, fy = finish
    draw.text((sx + 24, sy - 24), "START", fill=GREEN, font=font)
    draw.text((fx - 56, fy + 22), "FINISH", fill=RED, font=font)
    draw.text((292, 14), "FOLLOW THE ROUTE", fill=NAVY, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    return str(_longest_run(list(scene["checkpoint_colors"])))


def margin(scene: dict) -> float:
    colors = list(scene["checkpoint_colors"])
    target = _longest_run(colors)
    # Discrete run boundaries plus generous raster separation.
    transitions = sum(a != b for a, b in zip(colors, colors[1:]))
    return float(40 + 4 * target + transitions)


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    colors = scene["checkpoint_colors"]
    return len(points) != 6 or len(colors) != 10 or _longest_run(list(colors)) not in (1, 2, 3)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
