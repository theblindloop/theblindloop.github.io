"""Independent image-only inverse arm for the checkpoint route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((31, 52, 77), dtype=np.int16)
CYAN = np.array((0, 184, 204), dtype=np.int16)
ORANGE = np.array((239, 139, 32), dtype=np.int16)
GREEN = np.array((43, 168, 92), dtype=np.int16)
RED = np.array((218, 67, 74), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 16) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 25) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            component.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) >= minimum:
            result.append(component)
    return result


def _distance_map(traversable: np.ndarray, starts: np.ndarray) -> np.ndarray:
    height, width = traversable.shape
    distances = np.full((height, width), -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for y, x in zip(*np.nonzero(starts & traversable)):
        distances[y, x] = 0
        queue.append((int(y), int(x)))
    while queue:
        y, x = queue.popleft()
        value = int(distances[y, x]) + 1
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < height and 0 <= nx < width and traversable[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = value
                queue.append((ny, nx))
    return distances


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"))
    masks = {
        "navy": _near(pixels, NAVY),
        "cyan": _near(pixels, CYAN),
        "orange": _near(pixels, ORANGE),
        "green": _near(pixels, GREEN),
        "red": _near(pixels, RED),
    }
    cyan_components = _components(masks["cyan"])
    orange_components = _components(masks["orange"])
    green_components = _components(masks["green"])
    red_components = _components(masks["red"])
    if not (len(cyan_components) == 5 and len(orange_components) == 5):
        return "abstain"
    if len(green_components) != 1 or len(red_components) != 1:
        return "abstain"

    traversable = masks["navy"] | masks["cyan"] | masks["orange"] | masks["green"] | masks["red"]
    distances = _distance_map(traversable, masks["green"])
    if not np.any((distances >= 0) & masks["red"]):
        return "abstain"

    ordered: list[tuple[int, str]] = []
    for label, components in (("cyan", cyan_components), ("orange", orange_components)):
        for component in components:
            values = [int(distances[y, x]) for y, x in component if distances[y, x] >= 0]
            if not values:
                return "abstain"
            ordered.append((min(values), label))
    ordered.sort()
    if len({distance for distance, _ in ordered}) != 10:
        return "abstain"

    best = current = 0
    for _, label in ordered:
        current = current + 1 if label == "cyan" else 0
        best = max(best, current)
    return str(best) if best in (1, 2, 3) else "abstain"
