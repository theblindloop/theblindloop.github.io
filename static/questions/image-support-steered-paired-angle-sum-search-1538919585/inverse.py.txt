"""Independent image-only inverse arm for paired-angle search."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (214, 57, 66),
    "blue": (48, 105, 205),
    "green": (34, 145, 91),
    "purple": (145, 72, 184),
    "gold": (222, 158, 36),
}


class OracleAbstention(ValueError):
    pass


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= 48 * 48


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    remaining = set(zip(ys.tolist(), xs.tolist(), strict=True))
    groups = []
    height, width = mask.shape
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        if len(points) >= 180:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _opening_angle(points: np.ndarray) -> float:
    if not 1050 <= len(points) <= 1950:
        raise OracleAbstention(f"angle component has implausible ink area {len(points)}")
    ys = points[:, 0]
    xs = points[:, 1]
    x0, x1 = float(xs.min()), float(xs.max())
    if not 56 <= x1 - x0 <= 88:
        raise OracleAbstention("angle component has implausible ray extent")
    columns = []
    for x in range(int(x0) + 15, int(x1) - 7):
        values = ys[xs == x]
        if len(values):
            columns.append((float(x), float(values.min()), float(values.max())))
    if len(columns) < 28:
        raise OracleAbstention("too little intact ray support for a line fit")
    values = np.asarray(columns, dtype=np.float64)
    upper = float(np.polyfit(values[:, 0], values[:, 1], 1)[0])
    lower = float(np.polyfit(values[:, 0], values[:, 2], 1)[0])
    opening = math.degrees(math.atan(abs(upper)) + math.atan(abs(lower)))
    if not 12.0 <= opening <= 108.0:
        raise OracleAbstention(f"recovered opening is implausible: {opening:.2f}")
    return opening


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    measured: dict[str, list[tuple[float, float]]] = {}
    for color in (*COLORS, "gold"):
        groups = _components(_mask(array, RGB[color]))
        expected = 1 if color == "gold" else 2
        if len(groups) != expected:
            raise OracleAbstention(
                f"expected {expected} detached {color} angle component(s), found {len(groups)}"
            )
        measured[color] = [(_opening_angle(group), float(group[:, 1].mean())) for group in groups]

    reference = measured["gold"][0][0]
    errors = []
    for color in COLORS:
        parts = sorted(measured[color], key=lambda item: item[1])
        if parts[1][1] - parts[0][1] < image.width * 0.55:
            raise OracleAbstention(f"{color} evidence is not split across the two fields")
        errors.append((abs(parts[0][0] + parts[1][0] - reference), color))
    errors.sort()
    if errors[0][0] > 4.0:
        raise OracleAbstention(f"no color matches the reference closely enough ({errors[0][0]:.2f})")
    if errors[1][0] - errors[0][0] < 5.0:
        raise OracleAbstention("angle-sum winner lacks a runner-up margin")
    return errors[0][1]
