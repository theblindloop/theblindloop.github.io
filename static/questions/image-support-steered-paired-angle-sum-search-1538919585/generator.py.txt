"""Deterministic latent sampler and Pillow renderer for paired-angle search."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

COLORS = ("red", "blue", "green", "purple")
RGB = {
    "red": (214, 57, 66),
    "blue": (48, 105, 205),
    "green": (34, 145, 91),
    "purple": (145, 72, 184),
    "gold": (222, 158, 36),
}
BACKGROUND = (248, 247, 242)
INK = (45, 49, 58)
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = COLORS[int(seed) % len(COLORS)]
    reference = (80, 90, 100)[(int(seed) // len(COLORS)) % 3]
    shift = (reference - 80) // 2
    values = [20 + shift, 30 + shift, 50 + shift, 60 + shift]
    rng.shuffle(values)
    left_angles = dict(zip(COLORS, values, strict=True))

    valid = []
    for perm in itertools.permutations(values):
        right = dict(zip(COLORS, perm, strict=True))
        matches = [
            color
            for color in COLORS
            if left_angles[color] + right[color] == reference
        ]
        errors = sorted(abs(left_angles[c] + right[c] - reference) for c in COLORS)
        if matches == [target] and errors[1] >= 10:
            valid.append(right)
    if not valid:
        raise RuntimeError("could not construct a unique complementary permutation")
    right_angles = dict(rng.choice(valid))
    left_order = list(COLORS)
    right_order = list(COLORS)
    rng.shuffle(left_order)
    rng.shuffle(right_order)
    return {
        "width": 800,
        "height": 680,
        "reference_angle": reference,
        "left_angles": left_angles,
        "right_angles": right_angles,
        "left_order": left_order,
        "right_order": right_order,
    }


def _font(path: str, size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(path, size=size)


def _angle(draw: ImageDraw.ImageDraw, vertex: tuple[int, int], opening: float,
           color: tuple[int, int, int], *, length: int, width: int) -> None:
    x, y = vertex
    half = math.radians(opening / 2.0)
    dx = length * math.cos(half)
    dy = length * math.sin(half)
    points = [(round(x + dx), round(y - dy)), (x, y), (round(x + dx), round(y + dy))]
    draw.line(points, fill=color, width=width, joint="curve")
    radius = width // 2
    for px, py in points:
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=color)


def render(scene: dict) -> Image.Image:
    scale = 2
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def box(bounds, **kwargs):
        draw.rounded_rectangle(tuple(int(v * scale) for v in bounds), radius=18 * scale, **kwargs)

    panel_fill = (255, 255, 252)
    panel_line = (218, 216, 207)
    box((20, 215, 185, 655), fill=panel_fill, outline=panel_line, width=2 * scale)
    box((615, 215, 780, 655), fill=panel_fill, outline=panel_line, width=2 * scale)
    box((305, 24, 495, 190), fill=(255, 252, 239), outline=(228, 203, 135), width=2 * scale)

    bold = _font(FONT_BOLD, 18 * scale)
    small = _font(FONT_BOLD, 15 * scale)
    draw.text((55 * scale, 232 * scale), "LEFT SET", font=small, fill=INK)
    draw.text((645 * scale, 232 * scale), "RIGHT SET", font=small, fill=INK)
    draw.text((328 * scale, 43 * scale), "GOLD REFERENCE", font=bold, fill=INK)
    draw.text((385 * scale, 315 * scale), "+", font=_font(FONT_BOLD, 38 * scale), fill=(115, 117, 120))

    _angle(
        draw,
        (357 * scale, 126 * scale),
        float(scene["reference_angle"]),
        RGB["gold"],
        length=75 * scale,
        width=10 * scale,
    )
    rows = (300, 405, 510, 615)
    for row, color in zip(rows, scene["left_order"], strict=True):
        _angle(draw, (58 * scale, row * scale), float(scene["left_angles"][color]),
               RGB[color], length=75 * scale, width=10 * scale)
    for row, color in zip(rows, scene["right_order"], strict=True):
        _angle(draw, (653 * scale, row * scale), float(scene["right_angles"][color]),
               RGB[color], length=75 * scale, width=10 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _errors(scene: dict) -> list[tuple[float, str]]:
    reference = float(scene["reference_angle"])
    return sorted(
        (abs(float(scene["left_angles"][color]) + float(scene["right_angles"][color]) - reference), color)
        for color in COLORS
    )


def analytic_gold(scene: dict) -> str:
    return _errors(scene)[0][1]


def margin(scene: dict) -> float:
    errors = _errors(scene)
    return float(errors[1][0] - errors[0][0])


def is_quarantined(scene: dict) -> bool:
    errors = _errors(scene)
    return bool(errors[0][0] > 0.0 or errors[1][0] - errors[0][0] < 9.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
