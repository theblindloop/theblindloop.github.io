"""Pixel-only inverse. This module deliberately imports no world-side code."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "orange": np.array((230, 126, 34), dtype=np.int16),
    "green": np.array((39, 174, 96), dtype=np.int16),
    "purple": np.array((142, 68, 173), dtype=np.int16),
}


def _components(mask: np.ndarray, minimum: int) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        component: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            component.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) >= minimum:
            output.append(component)
    return output


def _centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    result = []
    for comp in _components(mask, 80):
        result.append((sum(x for y, x in comp) / len(comp), sum(y for y, x in comp) / len(comp)))
    return result


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = np.all(rgb < 70, axis=2)
    loop_components = _components(black, 250)
    if len(loop_components) != 9:
        raise ValueError(f"expected 9 closed loop components, found {len(loop_components)}")
    boxes = []
    for comp in loop_components:
        ys = [p[0] for p in comp]
        xs = [p[1] for p in comp]
        boxes.append((min(xs), min(ys), max(xs), max(ys)))
    common_counts: dict[str, int] = {}
    for name, target in PALETTE.items():
        distance = np.max(np.abs(rgb - target), axis=2)
        centers = _centroids(distance <= 8)
        if len(centers) != 2:
            raise ValueError(f"expected two {name} markers, found {len(centers)}")
        count = 0
        for left, top, right, bottom in boxes:
            if all(left < x < right and top < y < bottom for x, y in centers):
                count += 1
        common_counts[name] = count
    winners = [name for name, count in common_counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"no unique two-common-loop pair: {common_counts}")
    return winners[0]
