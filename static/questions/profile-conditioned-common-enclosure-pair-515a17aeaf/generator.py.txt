"""Deterministic latent sampler and renderer for common-enclosure pairs."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

COLORS = {
    "orange": (230, 126, 34),
    "green": (39, 174, 96),
    "purple": (142, 68, 173),
}


def _rect(cx: int, cy: int, half_w: int, half_h: int) -> list[int]:
    return [cx - half_w, cy - half_h, cx + half_w, cy + half_h]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    rng.shuffle(colors)
    depths = [1, 2, 3]
    rng.shuffle(depths)
    xs = [120, 360, 600]
    markers: list[dict] = []
    loops: list[list[int]] = []
    for slot, (cx, depth, color) in enumerate(zip(xs, depths, colors)):
        cy = 204 + rng.choice([-8, 0, 8])
        # Every panel has exactly three loops. Only their common-enclosure
        # membership differs: d common loops plus 3-d single-dot loops.
        common_sizes = [(92, 154), (76, 128), (60, 102)]
        for half_w, half_h in common_sizes[:depth]:
            loops.append(_rect(cx, cy, half_w, half_h))
        left_first = rng.choice([False, True])
        single_centers = [cx - 30, cx + 30]
        if left_first:
            single_centers.reverse()
        for sx in single_centers[: 3 - depth]:
            loops.append(_rect(sx, cy, 17, 26))
        markers.extend(
            [
                {"color": color, "x": cx - 30, "y": cy},
                {"color": color, "x": cx + 30, "y": cy},
            ]
        )
    return {"width": 720, "height": 400, "loops": loops, "markers": markers}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    for box in scene["loops"]:
        draw.rounded_rectangle(tuple(box), radius=10, outline=(20, 24, 28), width=7)
    for marker in scene["markers"]:
        x, y = int(marker["x"]), int(marker["y"])
        color = COLORS[marker["color"]]
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=color, outline=(255, 255, 255), width=2)
    return image


def _inside(box: list[int], marker: dict) -> bool:
    return box[0] < marker["x"] < box[2] and box[1] < marker["y"] < box[3]


def analytic_gold(scene: dict) -> str:
    loops = scene["loops"]
    markers = scene["markers"]
    by_color: dict[str, list[dict]] = {}
    for marker in markers:
        by_color.setdefault(marker["color"], []).append(marker)
    matches = []
    for color, pair in by_color.items():
        common = sum(_inside(box, pair[0]) and _inside(box, pair[1]) for box in loops)
        if common == 2:
            matches.append(color)
    if len(matches) != 1:
        raise ValueError(f"expected one pair with two common enclosures, got {matches}")
    return matches[0]


def margin(scene: dict) -> float:
    loops = scene["loops"]
    markers = scene["markers"]
    by_color: dict[str, list[dict]] = {}
    for marker in markers:
        by_color.setdefault(marker["color"], []).append(marker)
    distances = sorted(
        abs(sum(_inside(box, pair[0]) and _inside(box, pair[1]) for box in loops) - 2)
        for pair in by_color.values()
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    # Generated centers have >=13 px clearance from every loop stroke and
    # separate loop strokes have >=9 px of background between them.
    for marker in scene["markers"]:
        x, y = marker["x"], marker["y"]
        for left, top, right, bottom in scene["loops"]:
            if left - 12 <= x <= right + 12 and top - 12 <= y <= bottom + 12:
                edge_gap = min(abs(x - left), abs(x - right), abs(y - top), abs(y - bottom))
                if edge_gap < 12:
                    return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
