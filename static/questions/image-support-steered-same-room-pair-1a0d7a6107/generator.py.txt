"""Renderer for comparing room connectivity of two disconnected color pairs."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 320
HEIGHT = 320
BACKGROUND = [248, 247, 241]
BLACK = [27, 29, 33]
CYAN = [0, 151, 181]
ORANGE = [228, 105, 25]
GRAY = [169, 170, 166]
WALL = 13
DOT_R = 13
MIN_MARGIN = 13


def _rot(point: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(quarter_turns % 4):
        x, y = WIDTH - 1 - y, x
    return x, y


def _rot_box(box: list[int], quarter_turns: int) -> list[int]:
    x0, y0, x1, y1 = box
    points = [_rot((x0, y0), quarter_turns), _rot((x1, y0), quarter_turns),
              _rot((x1, y1), quarter_turns), _rot((x0, y1), quarter_turns)]
    return [min(x for x, _ in points), min(y for _, y in points),
            max(x for x, _ in points), max(y for _, y in points)]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    turns = rng.randrange(4)
    # The upper two chambers are separated.  The lower half is one room.
    disconnected = [(105, 105), (215, 105)]
    connected = [(105, 215), (215, 215)]
    cyan_wins = int(seed) % 2 == 0
    cyan_points = connected if cyan_wins else disconnected
    orange_points = disconnected if cyan_wins else connected
    frame = [55, 55, 265, 265]
    divider = [55, 153, 265, 166]
    branch = [153, 55, 166, 166]
    distractor_slots = [[18, 22, 25, 29], [292, 25, 299, 32], [21, 289, 28, 296]]
    rng.shuffle(distractor_slots)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "wall_color": list(BLACK),
        "cyan": list(CYAN),
        "orange": list(ORANGE),
        "frame_bounds": _rot_box(frame, turns),
        "divider": _rot_box(divider, turns),
        "branch": _rot_box(branch, turns),
        "cyan_points": sorted([list(_rot(p, turns)) for p in cyan_points]),
        "orange_points": sorted([list(_rot(p, turns)) for p in orange_points]),
        "distractors": sorted(distractor_slots[:1 + rng.randrange(3)]),
        "quarter_turns": turns,
    }


def _dot(draw: ImageDraw.ImageDraw, point: list[int], color: list[int]) -> None:
    x, y = point
    draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=tuple(color))


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    x0, y0, x1, y1 = scene["frame_bounds"]
    color = tuple(scene["wall_color"])
    half = WALL // 2
    draw.rectangle([x0, y0, x1, y0 + WALL - 1], fill=color)
    draw.rectangle([x0, y1 - WALL + 1, x1, y1], fill=color)
    draw.rectangle([x0, y0, x0 + WALL - 1, y1], fill=color)
    draw.rectangle([x1 - WALL + 1, y0, x1, y1], fill=color)
    draw.rectangle(scene["divider"], fill=color)
    draw.rectangle(scene["branch"], fill=color)
    for point in scene["cyan_points"]:
        _dot(draw, point, scene["cyan"])
    for point in scene["orange_points"]:
        _dot(draw, point, scene["orange"])
    for box in scene["distractors"]:
        draw.ellipse(box, fill=tuple(GRAY))
    return image


def _is_connected_pair(scene: dict[str, Any], points: list[list[int]]) -> bool:
    divider = scene["divider"]
    branch = scene["branch"]
    horizontal = (divider[2] - divider[0]) > (divider[3] - divider[1])
    if horizontal:
        split = (divider[1] + divider[3]) / 2.0
        branch_side = ((branch[1] + branch[3]) / 2.0) < split
        return all((point[1] < split) != branch_side for point in points)
    split = (divider[0] + divider[2]) / 2.0
    branch_side = ((branch[0] + branch[2]) / 2.0) < split
    return all((point[0] < split) != branch_side for point in points)


def analytic_gold(scene: dict[str, Any]) -> str:
    cyan_connected = _is_connected_pair(scene, scene["cyan_points"])
    orange_connected = _is_connected_pair(scene, scene["orange_points"])
    if cyan_connected == orange_connected:
        raise ValueError("exactly one color pair must share a room")
    return "cyan" if cyan_connected else "orange"


def margin(scene: dict[str, Any]) -> float:
    return float(MIN_MARGIN)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < MIN_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["cyan_points"] = list(reversed(twin["cyan_points"]))
    twin["orange_points"] = list(reversed(twin["orange_points"]))
    return [("reverse_same_color_point_storage", twin)]
