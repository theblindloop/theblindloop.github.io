"""Independent PNG-only room-connectivity oracle."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CYAN = np.array([0, 151, 181], dtype=np.int16)
_ORANGE = np.array([228, 105, 25], dtype=np.int16)
_BLACK = np.array([27, 29, 33], dtype=np.int16)
_TOL = 42.0


def _components(mask: np.ndarray) -> list[tuple[int, int, int]]:
    seen = np.zeros_like(mask, dtype=bool)
    parts: list[tuple[int, int, int]] = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 250:
            parts.append((len(xs), int(round(sum(xs) / len(xs))), int(round(sum(ys) / len(ys)))))
    return parts


def _pair(mask: np.ndarray, name: str) -> list[tuple[int, int]]:
    parts = _components(mask)
    if len(parts) != 2 or any(not 250 <= area <= 650 for area, _, _ in parts):
        raise ValueError(f"expected two intact {name} tokens")
    return [(x, y) for _, x, y in parts]


def _connected(barrier: np.ndarray, first: tuple[int, int], second: tuple[int, int]) -> bool:
    height, width = barrier.shape
    sx, sy = first
    tx, ty = second
    seen = np.zeros_like(barrier, dtype=bool)
    queue = deque([(sy, sx)])
    seen[sy, sx] = True
    while queue:
        y, x = queue.popleft()
        if (x - tx) ** 2 + (y - ty) ** 2 <= 4:
            return True
        for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if 0 <= ny < height and 0 <= nx < width and not barrier[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return False


def _reconstruct_walls(black: np.ndarray) -> np.ndarray:
    """Bridge localized raster loss along visibly supported thick wall runs."""
    repaired = black.copy()
    row_counts = black.sum(axis=1)
    col_counts = black.sum(axis=0)
    # Long horizontal/vertical wall bands remain identifiable from their ink
    # projection even when one overlapping ablation tile removes a segment.
    def bridge(line: np.ndarray) -> None:
        indices = np.flatnonzero(line)
        for left, right in zip(indices[:-1], indices[1:]):
            gap = int(right - left - 1)
            if 0 < gap <= 65:
                line[int(left): int(right) + 1] = True

    for y in np.flatnonzero(row_counts >= 70):
        bridge(repaired[y])
    for x in np.flatnonzero(col_counts >= 70):
        bridge(repaired[:, x])
    return repaired


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (320, 320):
        raise ValueError("expected a 320 by 320 raster")
    cyan_mask = np.linalg.norm(array - _CYAN, axis=2) <= _TOL
    orange_mask = np.linalg.norm(array - _ORANGE, axis=2) <= _TOL
    black = np.linalg.norm(array - _BLACK, axis=2) <= _TOL
    cyan = _pair(cyan_mask, "cyan")
    orange = _pair(orange_mask, "orange")
    # A low integrity floor rejects blank or wholesale-corrupted rasters without
    # turning total ink into a decision shortcut. Local wall erasure is judged
    # by its actual connectivity consequence below.
    if int(black.sum()) < 5000:
        raise ValueError("substantial floor-plan walls are missing")
    barrier = _reconstruct_walls(black)
    outcomes = {
        "cyan": _connected(barrier, cyan[0], cyan[1]),
        "orange": _connected(barrier, orange[0], orange[1]),
    }
    winners = [name for name, value in outcomes.items() if value]
    if len(winners) != 1:
        raise ValueError("image does not expose exactly one same-room pair")
    return winners[0]
