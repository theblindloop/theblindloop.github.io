"""Independent raster-only inverse for paired-vector parallelogram area."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class MeasurementError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    components: list[np.ndarray] = []
    height, width = work.shape
    for y, x in np.argwhere(work):
        y, x = int(y), int(x)
        if not work[y, x]:
            continue
        work[y, x] = False
        queue = deque([(y, x)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                        work[ny, nx] = False
                        queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.float64))
    return components


def _centers(mask: np.ndarray, kind: str) -> list[tuple[float, float]]:
    centers: list[tuple[float, float]] = []
    for points in _components(mask):
        count = len(points)
        ys, xs = points[:, 0], points[:, 1]
        bh = float(ys.max() - ys.min() + 1)
        bw = float(xs.max() - xs.min() + 1)
        fill = count / (bh * bw)
        if kind == "endpoint":
            if 135 <= count <= 280 and 0.72 <= bw / bh <= 1.38 and fill > 0.66:
                centers.append((float(xs.mean()), float(ys.mean())))
        elif kind == "origin":
            if 120 <= count <= 230 and 0.72 <= bw / bh <= 1.38 and fill > 0.65:
                centers.append((float(xs.mean()), float(ys.mean())))
    return centers


def _reading_order(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    if len(points) != 4:
        raise MeasurementError(f"expected four panel origins, found {len(points)}")
    by_y = sorted(points, key=lambda p: p[1])
    top = sorted(by_y[:2], key=lambda p: p[0])
    bottom = sorted(by_y[2:], key=lambda p: p[0])
    return top + bottom


def _nearest_unique(
    origins: list[tuple[float, float]], endpoints: list[tuple[float, float]], role: str
) -> list[tuple[float, float]]:
    if len(endpoints) != 4:
        raise MeasurementError(f"expected four {role} endpoints, found {len(endpoints)}")
    result: list[tuple[float, float]] = []
    unused = set(range(4))
    for ox, oy in origins:
        index = min(unused, key=lambda i: (endpoints[i][0] - ox) ** 2 + (endpoints[i][1] - oy) ** 2)
        distance2 = (endpoints[index][0] - ox) ** 2 + (endpoints[index][1] - oy) ** 2
        if not 35.0**2 <= distance2 <= 105.0**2:
            raise MeasurementError(f"{role} endpoint is outside the measurable vector range")
        result.append(endpoints[index])
        unused.remove(index)
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = arr[..., 0], arr[..., 1], arr[..., 2]
    blue_mask = (blue > red + 55) & (blue > green + 35) & (blue > 150)
    orange_mask = (red > green + 55) & (green > blue + 45) & (red > 180)
    neutral_spread = arr.max(axis=2) - arr.min(axis=2)
    origin_mask = (neutral_spread <= 7) & (red >= 65) & (red <= 95)

    origins = _reading_order(_centers(origin_mask, "origin"))
    blue_ends = _nearest_unique(origins, _centers(blue_mask, "endpoint"), "blue")
    orange_ends = _nearest_unique(origins, _centers(orange_mask, "endpoint"), "orange")
    scores: list[float] = []
    for (ox, oy), (bx, by), (tx, ty) in zip(origins, blue_ends, orange_ends, strict=True):
        ux, uy = bx - ox, by - oy
        vx, vy = tx - ox, ty - oy
        scores.append(abs(ux * vy - uy * vx))
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] - scores[order[1]] < 120.0:
        raise MeasurementError("pixel cross-area runner-up gap is ambiguous")
    return ("A", "B", "C", "D")[order[0]]
