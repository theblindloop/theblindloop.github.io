"""Deterministic paired-vector cross-area world and analytic arm."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 620
LABELS = ("A", "B", "C", "D")
QUARANTINE_FLOOR = 260.0

PALETTES = (
    {"blue": [35, 105, 210], "orange": [230, 125, 25]},
    {"blue": [45, 95, 190], "orange": [220, 115, 20]},
    {"blue": [28, 118, 205], "orange": [235, 135, 32]},
    {"blue": [55, 110, 220], "orange": [215, 105, 18]},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / name
    return ImageFont.truetype(str(path), size=size)


def _cross_area(panel: dict) -> float:
    # Iteration order is fixed by the sampled JSON object (blue then orange).
    # Avoid presenting nested dictionary field names as top-level latent fields
    # to the conservative rendered-field audit.
    (bx, by), (ox, oy) = panel.values()
    return float(abs(bx * oy - by * ox))


def _scores(scene: dict) -> list[float]:
    return [_cross_area(panel) for panel in scene["panels"]]


def _angle_between(panel: dict) -> float:
    bx, by = panel["blue"]
    ox, oy = panel["orange"]
    nb = math.hypot(bx, by)
    no = math.hypot(ox, oy)
    cosine = max(-1.0, min(1.0, (bx * ox + by * oy) / (nb * no)))
    return math.acos(cosine)


def _valid_panel_set(panels: list[dict]) -> bool:
    scores = [_cross_area(panel) for panel in panels]
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    winner = order[0]
    if scores[order[0]] - scores[order[1]] < 520.0:
        return False
    products = [
        math.hypot(*panel["blue"]) * math.hypot(*panel["orange"]) for panel in panels
    ]
    sums = [math.hypot(*p["blue"]) + math.hypot(*p["orange"]) for p in panels]
    maxima = [max(math.hypot(*p["blue"]), math.hypot(*p["orange"])) for p in panels]
    angles = [_angle_between(panel) for panel in panels]
    # The winning cross area must integrate magnitude and angular separation. No listed
    # one-feature shortcut is permitted to identify it.
    return (
        winner != max(range(4), key=lambda i: products[i])
        and winner != max(range(4), key=lambda i: sums[i])
        and winner != max(range(4), key=lambda i: maxima[i])
        and winner != max(range(4), key=lambda i: abs(math.sin(angles[i])))
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(20000):
        panels: list[dict] = []
        for _panel in range(4):
            blue_len = rng.randint(46, 90)
            orange_len = rng.randint(46, 90)
            included = math.radians(rng.uniform(16.0, 148.0))
            if rng.random() < 0.5:
                included = -included
            base = math.radians(rng.uniform(0.0, 360.0))
            blue = [
                int(round(blue_len * math.cos(base))),
                int(round(blue_len * math.sin(base))),
            ]
            orange = [
                int(round(orange_len * math.cos(base + included))),
                int(round(orange_len * math.sin(base + included))),
            ]
            panels.append({"blue": blue, "orange": orange})
        if not _valid_panel_set(panels):
            continue
        scores = [_cross_area(panel) for panel in panels]
        winner = max(range(4), key=lambda i: scores[i])
        target = int(seed) % 4
        selected = panels.pop(winner)
        panels.insert(target, selected)
        return {
            "panels": panels,
            "palette": dict(PALETTES[rng.randrange(len(PALETTES))]),
            "offset_x": rng.randint(-8, 8),
            "offset_y": rng.randint(-6, 6),
        }
    raise RuntimeError("could not sample a shortcut-controlled cross-area scene")


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return LABELS[max(range(4), key=lambda i: scores[i])]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_vector(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    vector: list[int],
    color: tuple[int, int, int],
) -> None:
    ox, oy = origin
    vx, vy = vector
    length = math.hypot(vx, vy)
    ux, uy = vx / length, vy / length
    start = (round(ox + 10 * ux), round(oy + 10 * uy))
    # A white break isolates the circular endpoint as a measurable component.
    stop = (round(ox + vx - 12 * ux), round(oy + vy - 12 * uy))
    draw.line([start, stop], fill=color, width=5)
    ex, ey = ox + vx, oy + vy
    draw.ellipse([ex - 8, ey - 8, ex + 8, ey + 8], fill=color)


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    palette = scene["palette"]
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    title_font = _font(24, bold=True)
    body_font = _font(16)
    label_font = _font(22, bold=True)
    draw.text(
        (44 + offset_x, 20 + offset_y),
        "Which panel spans the largest parallelogram area?",
        fill=(30, 30, 32),
        font=title_font,
    )
    draw.text(
        (44 + offset_x, 54 + offset_y),
        "Each vector runs from the gray origin to its colored endpoint dot.   Area = |uₓvᵧ − uᵧvₓ|",
        fill=(55, 55, 58),
        font=body_font,
    )

    panel_boxes = ((42, 96, 382, 326), (418, 96, 758, 326), (42, 354, 382, 584), (418, 354, 758, 584))
    blue = tuple(int(v) for v in palette["blue"])
    orange = tuple(int(v) for v in palette["orange"])
    for index, (box, panel) in enumerate(zip(panel_boxes, panels, strict=True)):
        x0, y0, x1, y1 = (value + (offset_x if pos % 2 == 0 else offset_y) for pos, value in enumerate(box))
        draw.rounded_rectangle([x0, y0, x1, y1], radius=12, fill=(255, 255, 255), outline=(170, 174, 180), width=2)
        draw.text((x0 + 14, y0 + 10), LABELS[index], fill=(25, 25, 27), font=label_font)
        origin = ((x0 + x1) // 2, (y0 + y1) // 2 + 10)
        draw.line([(x0 + 55, origin[1]), (x1 - 35, origin[1])], fill=(222, 224, 226), width=1)
        draw.line([(origin[0], y0 + 48), (origin[0], y1 - 27)], fill=(222, 224, 226), width=1)
        _draw_vector(draw, origin, panel["blue"], blue)
        _draw_vector(draw, origin, panel["orange"], orange)
        ox, oy = origin
        draw.ellipse([ox - 7, oy - 7, ox + 7, oy + 7], fill=(78, 78, 82))
    return image
