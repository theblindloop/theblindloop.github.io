"""Image-only oracle for hub_clockwise_wire.

Receives one rendered PNG and nothing else.  It segments the dark hub and the
grey strokes, finds where a stroke meets each hub side, walks each stroke's
centreline (continuing straight through crossings, turning at genuine corners)
until it arrives at a coloured square, and then reports the colour attached one
hub side clockwise of the stroke that ends at the green square.
"""

from __future__ import annotations

ORACLE_VERSION = "hub_clockwise_wire-oracle-0.1.0"
ABSTAIN = "unknown"

_SIDES = ("top", "right", "bottom", "left")
_DIRS = {"top": (0, -1), "right": (1, 0), "bottom": (0, 1), "left": (-1, 0)}
_PALETTE = {
    "green": (34, 160, 60),
    "red": (215, 45, 45),
    "blue": (45, 90, 215),
    "orange": (240, 140, 20),
}
_WIRE = (110, 110, 110)
_WIRE_TOL = 45
_DARK = 70
_STROKE = 5


def _near(px, ref, tol) -> bool:
    return all(abs(int(px[i]) - ref[i]) <= tol for i in range(3))


def _classify(px):
    for name, ref in _PALETTE.items():
        if _near(px, ref, 45):
            return name
    return None


class _Field:
    def __init__(self, image):
        self.w, self.h = image.size
        self.px = image.convert("RGB").load()

    def wire(self, x, y) -> bool:
        if not (0 <= x < self.w and 0 <= y < self.h):
            return False
        return _near(self.px[x, y], _WIRE, _WIRE_TOL)

    def dark(self, x, y) -> bool:
        p = self.px[x, y]
        return p[0] < _DARK and p[1] < _DARK and p[2] < _DARK


def _hub_box(field: _Field):
    xs, ys = [], []
    for y in range(0, field.h, 2):
        for x in range(0, field.w, 2):
            if field.dark(x, y):
                xs.append(x)
                ys.append(y)
    if len(xs) < 100:
        return None
    x0, x1, y0, y1 = min(xs), max(xs), min(ys), max(ys)
    if not (30 < x1 - x0 < 200) or abs((x1 - x0) - (y1 - y0)) > 12:
        return None
    return x0, y0, x1, y1


def _recentre(field: _Field, x, y, dx, dy):
    """Snap back onto the stroke centreline across the direction of travel."""
    if dx:
        lo = hi = y
        while field.wire(x, lo - 1):
            lo -= 1
        while field.wire(x, hi + 1):
            hi += 1
        if hi - lo + 1 <= _STROKE + 2:
            y = (lo + hi) // 2
    else:
        lo = hi = x
        while field.wire(lo - 1, y):
            lo -= 1
        while field.wire(hi + 1, y):
            hi += 1
        if hi - lo + 1 <= _STROKE + 2:
            x = (lo + hi) // 2
    return x, y


def _trace(field: _Field, start, direction):
    x, y = start
    dx, dy = direction
    for _ in range(20000):
        if field.wire(x + dx, y + dy):
            x, y = x + dx, y + dy
            x, y = _recentre(field, x, y, dx, dy)
            continue
        # end of the current straight run: step back to the corner centre
        cx, cy = x - 2 * dx, y - 2 * dy
        turns = [(-dy, dx), (dy, -dx)]
        options = [t for t in turns if field.wire(cx + t[0] * 4, cy + t[1] * 4)]
        if len(options) == 1:
            dx, dy = options[0]
            x, y = _recentre(field, cx, cy, dx, dy)
            continue
        if options:
            return None                      # ambiguous branch: refuse to guess
        for k in range(3, 20):               # the stroke has run into a square
            px = field.px[min(max(x + dx * k, 0), field.w - 1),
                          min(max(y + dy * k, 0), field.h - 1)]
            name = _classify(px)
            if name:
                return name
        return None
    return None


def _attachment(field: _Field, box, side):
    x0, y0, x1, y1 = box
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    half = (x1 - x0) // 2
    off = 4
    run = []
    if side in ("top", "bottom"):
        y = y0 - off if side == "top" else y1 + off
        for x in range(cx - half, cx + half + 1):
            if field.wire(x, y):
                run.append(x)
        if not run or max(run) - min(run) + 1 != len(run):
            return None
        return ((min(run) + max(run)) // 2, y)
    x = x0 - off if side == "left" else x1 + off
    for y in range(cy - half, cy + half + 1):
        if field.wire(x, y):
            run.append(y)
    if not run or max(run) - min(run) + 1 != len(run):
        return None
    return (x, (min(run) + max(run)) // 2)


def decision_from_image(image) -> str:
    try:
        field = _Field(image)
    except Exception:
        return ABSTAIN
    box = _hub_box(field)
    if box is None:
        return ABSTAIN
    by_side = {}
    for side in _SIDES:
        start = _attachment(field, box, side)
        if start is None:
            return ABSTAIN
        color = _trace(field, start, _DIRS[side])
        if color is None or color in by_side.values():
            return ABSTAIN
        by_side[side] = color
    if "green" not in by_side.values():
        return ABSTAIN
    green_side = next(s for s, c in by_side.items() if c == "green")
    answer = by_side[_SIDES[(_SIDES.index(green_side) + 1) % 4]]
    return answer if answer in ("red", "blue", "orange") else ABSTAIN
