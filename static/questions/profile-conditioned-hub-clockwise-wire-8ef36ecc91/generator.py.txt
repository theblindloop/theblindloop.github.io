"""Renderer for hub_clockwise_wire.

Four identical grey wires leave a black square hub, one from each side, wander
through a nest of rectangular channels crossing one another at right angles and
end at four coloured terminal squares.  The decision is the colour of the wire
that attaches to the hub side immediately clockwise of the side where the wire
coming from the GREEN square attaches.

Nothing in the picture marks a wire's identity: wire colour, thickness and
style are constant, so the attachment order around the hub can only be
recovered by following each stroke through its crossings.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "hub_clockwise_wire-renderer-0.1.0"

CANVAS = 720
CENTER = 360
HUB_HALF = 40
WIRE_HALF = 2                      # stroke width 5
RING_OFFSETS = (65, 105, 145, 185)  # channel radii, offset from the hub edge
TERMINAL_RADIUS = 320
TERMINAL_HALF = 17
EXIT_LATERALS = (-85, -55, 55, 85)
ATTACH_LIMIT = 24                   # |lateral| of a comfortable attachment
NEAR_CORNER_LIMITS = (27, 34)       # |lateral| of a deliberate near-corner one
NEAR_CORNER_RATE = 0.2

BACKGROUND = (245, 245, 245)
WIRE_COLOR = (110, 110, 110)
HUB_COLOR = (20, 20, 20)

SIDES = ("top", "right", "bottom", "left")   # clockwise on screen
START_COLOR = "green"
CHOICE_COLORS = ("red", "blue", "orange")
TERMINAL_COLORS = {
    "green": (34, 160, 60),
    "red": (215, 45, 45),
    "blue": (45, 90, 215),
    "orange": (240, 140, 20),
}

QUARANTINE_MARGIN = 14.0


# ------------------------------------------------------------------ geometry
def _point(side: str, radius: float, lateral: float) -> tuple[float, float]:
    """A point at `radius` out from the hub centre on `side`, offset laterally."""
    if side == "top":
        return (CENTER + lateral, CENTER - radius)
    if side == "right":
        return (CENTER + radius, CENTER + lateral)
    if side == "bottom":
        return (CENTER + lateral, CENTER + radius)
    if side == "left":
        return (CENTER - radius, CENTER + lateral)
    raise KeyError(side)


def _corner(side_a: str, side_b: str, radius: float) -> tuple[float, float]:
    pair = frozenset((side_a, side_b))
    if pair == frozenset(("top", "right")):
        return (CENTER + radius, CENTER - radius)
    if pair == frozenset(("right", "bottom")):
        return (CENTER + radius, CENTER + radius)
    if pair == frozenset(("bottom", "left")):
        return (CENTER - radius, CENTER + radius)
    if pair == frozenset(("left", "top")):
        return (CENTER - radius, CENTER - radius)
    raise KeyError(pair)


def _route(attach_side: str, attach_lateral: int, ring: int,
           exit_side: str, exit_lateral: int, clockwise: bool) -> list[list[float]]:
    """Axis-aligned waypoints: hub edge -> channel -> around -> terminal."""
    pts = [list(_point(attach_side, HUB_HALF, attach_lateral)),
           list(_point(attach_side, ring, attach_lateral))]
    step = 1 if clockwise else -1
    cur = SIDES.index(attach_side)
    target = SIDES.index(exit_side)
    while cur != target:
        nxt = (cur + step) % 4
        pts.append(list(_corner(SIDES[cur], SIDES[nxt], ring)))
        cur = nxt
    pts.append(list(_point(exit_side, ring, exit_lateral)))
    pts.append(list(_point(exit_side, TERMINAL_RADIUS, exit_lateral)))
    return pts


# ------------------------------------------------------------------ sampling
def sample_scene(seed: int) -> dict:
    rng = random.Random(f"hub_clockwise_wire:{int(seed)}")
    rings = list(RING_OFFSETS)
    rng.shuffle(rings)
    exit_sides = list(SIDES)
    rng.shuffle(exit_sides)
    colors = list(TERMINAL_COLORS)
    rng.shuffle(colors)

    laterals = [rng.randint(-ATTACH_LIMIT, ATTACH_LIMIT) for _ in SIDES]
    if rng.random() < NEAR_CORNER_RATE:
        # deliberately park one attachment just short of a hub corner, so the
        # deterministic sweep contains genuine near-boundary scenes
        pick = rng.randrange(4)
        laterals[pick] = rng.choice((-1, 1)) * rng.randint(*NEAR_CORNER_LIMITS)

    wires = []
    for i, attach_side in enumerate(SIDES):
        exit_side = exit_sides[i]
        ring = HUB_HALF + rings[i]
        attach_lateral = laterals[i]
        exit_lateral = rng.choice(EXIT_LATERALS)
        clockwise = rng.random() < 0.5
        wires.append(
            {
                "color": colors[i],
                "attach_side": attach_side,
                "attach_lateral": attach_lateral,
                "ring": ring,
                "exit_side": exit_side,
                "exit_lateral": exit_lateral,
                "clockwise": clockwise,
                "path": _route(attach_side, attach_lateral, ring,
                               exit_side, exit_lateral, clockwise),
            }
        )
    rng.shuffle(wires)
    return {
        "seed": int(seed),
        "canvas": CANVAS,
        "background": list(BACKGROUND),
        "center": [CENTER, CENTER],
        "hub_half": HUB_HALF,
        "wire_half": WIRE_HALF,
        "wire_color": list(WIRE_COLOR),
        "hub_color": list(HUB_COLOR),
        "terminal_half": TERMINAL_HALF,
        "wires": wires,
    }


# ----------------------------------------------------------------- rendering
def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])),
                    tuple(scene["background"]))
    draw = ImageDraw.Draw(img)
    h = float(scene["wire_half"])
    for wire in scene["wires"]:
        pts = wire["path"]
        for (x1, y1), (x2, y2) in zip(pts, pts[1:]):
            draw.rectangle(
                [min(x1, x2) - h, min(y1, y2) - h, max(x1, x2) + h, max(y1, y2) + h],
                fill=tuple(scene["wire_color"]),
            )
    cx, cy = scene["center"]
    hh = float(scene["hub_half"])
    draw.rectangle([cx - hh, cy - hh, cx + hh, cy + hh], fill=tuple(scene["hub_color"]))
    th = float(scene["terminal_half"])
    for wire in scene["wires"]:
        x, y = wire["path"][-1]
        draw.rectangle([x - th, y - th, x + th, y + th],
                       fill=TERMINAL_COLORS[wire["color"]])
    return img


# --------------------------------------------------------------------- gold
def _wire_by_color(scene: dict, color: str) -> dict:
    for wire in scene["wires"]:
        if wire["color"] == color:
            return wire
    raise KeyError(color)


def _wire_by_attach_side(scene: dict, side: str) -> dict:
    for wire in scene["wires"]:
        if wire["attach_side"] == side:
            return wire
    raise KeyError(side)


def successor_side(scene: dict) -> str:
    start = _wire_by_color(scene, START_COLOR)["attach_side"]
    return SIDES[(SIDES.index(start) + 1) % 4]


def analytic_gold(scene: dict) -> str:
    """Colour of the wire attaching one hub side clockwise of the green wire."""
    return _wire_by_attach_side(scene, successor_side(scene))["color"]


def margin(scene: dict) -> float:
    """Smallest distance, in px, from any attachment to the hub corner it
    would have to cross for its side membership -- and hence the clockwise
    ordering -- to change."""
    return float(min(scene["hub_half"] - abs(w["attach_lateral"]) for w in scene["wires"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


# --------------------------------------------------------------- symmetries
def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, label-preserving rewrites of the latent record."""
    out: list[tuple[str, dict]] = []

    rotated = dict(scene)
    rotated["wires"] = scene["wires"][1:] + scene["wires"][:1]
    out.append(("wire_record_reordering", rotated))

    refined = dict(scene)
    wires = []
    for wire in scene["wires"]:
        pts = wire["path"]
        dense = [list(pts[0])]
        for (x1, y1), (x2, y2) in zip(pts, pts[1:]):
            dense.append([(x1 + x2) / 2.0, (y1 + y2) / 2.0])
            dense.append([x2, y2])
        clone = dict(wire)
        clone["path"] = dense
        wires.append(clone)
    refined["wires"] = wires
    out.append(("collinear_waypoint_refinement", refined))
    return out
