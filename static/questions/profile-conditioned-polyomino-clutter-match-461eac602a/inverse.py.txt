"""Pixel-only inverse arm; intentionally independent of renderer and scene code."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = "ABCD"


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, np.ndarray]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(x0), int(y0))])
        seen[y0, x0] = True
        pixels = []
        while queue:
            x, y = queue.popleft()
            pixels.append((x, y))
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((nx, ny))
        if len(pixels) < minimum:
            continue
        xs = [p[0] for p in pixels]
        ys = [p[1] for p in pixels]
        left, top, right, bottom = min(xs), min(ys), max(xs), max(ys)
        crop = np.zeros((bottom - top + 1, right - left + 1), dtype=bool)
        for x, y in pixels:
            crop[y - top, x - left] = True
        result.append((left, top, right, bottom, crop))
    return result


def _same_shape(left: np.ndarray, right: np.ndarray) -> bool:
    return left.shape == right.shape and bool(np.array_equal(left, right))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Shape inks are deliberately dark under every allowed palette choice.
    ink_mask = np.max(rgb, axis=2) < 82
    shapes = _components(ink_mask, minimum=250)
    if len(shapes) != 13:
        raise ValueError(f"expected 13 connected block-shapes, found {len(shapes)}")

    # Frame ink is cyan-blue and much brighter than every shape ink.
    frame_mask = (rgb[:, :, 0] >= 35) & (rgb[:, :, 0] <= 75) & (rgb[:, :, 1] >= 105) & (rgb[:, :, 1] <= 145) & (rgb[:, :, 2] >= 135) & (rgb[:, :, 2] <= 175)
    frames = [part for part in _components(frame_mask, minimum=500) if part[2] - part[0] > 200 and part[3] - part[1] > 130]
    if len(frames) != 5:
        raise ValueError(f"expected five visible frames, found {len(frames)}")
    frames.sort(key=lambda part: ((part[1] + part[3]) / 2, (part[0] + part[2]) / 2))
    reference_frame = frames[0]
    panel_frames = sorted(frames[1:], key=lambda part: ((part[1] + part[3]) / 2, (part[0] + part[2]) / 2))

    def contained(shape, frame) -> bool:
        cx = (shape[0] + shape[2]) / 2
        cy = (shape[1] + shape[3]) / 2
        return frame[0] < cx < frame[2] and frame[1] < cy < frame[3]

    reference_shapes = [shape for shape in shapes if contained(shape, reference_frame)]
    if len(reference_shapes) != 1:
        raise ValueError("reference frame must contain exactly one shape")
    reference = reference_shapes[0][4]
    winning_panels = []
    for index, frame in enumerate(panel_frames):
        members = [shape for shape in shapes if contained(shape, frame)]
        if len(members) != 3:
            raise ValueError("each candidate panel must contain exactly three shapes")
        if any(_same_shape(reference, member[4]) for member in members):
            winning_panels.append(index)
    if len(winning_panels) != 1:
        raise ValueError("image does not contain one unique matching panel")
    return LABELS[winning_panels[0]]
