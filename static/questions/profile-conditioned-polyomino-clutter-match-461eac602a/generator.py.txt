"""Deterministic renderer for exact polyomino search among structured clutter."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 700
LABELS = "ABCD"
INKS = [(22, 24, 29), (58, 25, 48), (23, 57, 54), (55, 40, 22)]
FRAME = (55, 126, 156)
BACKGROUND = (248, 247, 243)


def _norm(shape: Iterable[Iterable[int]]) -> tuple[tuple[int, int], ...]:
    cells = [(int(p[0]), int(p[1])) for p in shape]
    min_x = min(x for x, _ in cells)
    min_y = min(y for _, y in cells)
    return tuple(sorted((x - min_x, y - min_y) for x, y in cells))


def _shape_pool() -> list[tuple[tuple[int, int], ...]]:
    """Create a stable catalog of connected seven-cell silhouettes."""
    rng = random.Random(731941)
    found: set[tuple[tuple[int, int], ...]] = set()
    while len(found) < 96:
        cells = {(0, 0)}
        while len(cells) < 7:
            x, y = rng.choice(sorted(cells))
            dx, dy = rng.choice(((1, 0), (-1, 0), (0, 1), (0, -1)))
            cells.add((x + dx, y + dy))
        shape = _norm(cells)
        w = max(x for x, _ in shape) + 1
        h = max(y for _, y in shape) + 1
        if w <= 5 and h <= 5:
            found.add(shape)
    return sorted(found)


SHAPES = _shape_pool()


def _shape_distance(left: Iterable[Iterable[int]], right: Iterable[Iterable[int]]) -> int:
    return len(set(_norm(left)) ^ set(_norm(right)))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = SHAPES[rng.randrange(len(SHAPES))]
    eligible = [shape for shape in SHAPES if _shape_distance(reference, shape) >= 4]
    rng.shuffle(eligible)
    distractors = eligible[:11]
    answer_index = int(seed) % 4
    target_slot = rng.randrange(3)
    panels: list[list[list[list[int]]]] = []
    cursor = 0
    for panel_index in range(4):
        shapes = []
        for slot in range(3):
            if panel_index == answer_index and slot == target_slot:
                shape = reference
            else:
                shape = distractors[cursor]
                cursor += 1
            shapes.append([[x, y] for x, y in shape])
        rng.shuffle(shapes)
        panels.append(shapes)
    return {
        "reference_shape": [[x, y] for x, y in reference],
        "panels": panels,
        "palette": {"ink": list(INKS[rng.randrange(len(INKS))]), "frame": list(FRAME), "background": list(BACKGROUND)},
        "cell_size": 10,
        "layout_dx": rng.randint(-8, 8),
        "layout_dy": rng.randint(-5, 5),
    }


def _draw_shape(draw: ImageDraw.ImageDraw, shape: Iterable[Iterable[int]], center: tuple[int, int], cell: int, fill: tuple[int, int, int]) -> None:
    normalized = _norm(shape)
    width = (max(x for x, _ in normalized) + 1) * cell
    height = (max(y for _, y in normalized) + 1) * cell
    left = int(center[0] - width / 2)
    top = int(center[1] - height / 2)
    for x, y in normalized:
        draw.rectangle((left + x * cell, top + y * cell, left + (x + 1) * cell - 1, top + (y + 1) * cell - 1), fill=fill)


def render(scene: dict) -> Image.Image:
    reference = scene["reference_shape"]
    panels = scene["panels"]
    palette = scene["palette"]
    cell = int(scene["cell_size"])
    dx = int(scene["layout_dx"])
    dy = int(scene["layout_dy"])
    ink = tuple(palette["ink"])
    frame = tuple(palette["frame"])
    background = tuple(palette["background"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=18)
    title_font = ImageFont.load_default(size=19)

    ref_box = (250 + dx, 45 + dy, 510 + dx, 215 + dy)
    draw.rectangle(ref_box, outline=frame, width=3)
    draw.text((ref_box[0] + 76, ref_box[1] + 10), "REFERENCE", fill=frame, font=title_font)
    _draw_shape(draw, reference, ((ref_box[0] + ref_box[2]) // 2, ref_box[1] + 104), cell, ink)

    boxes = [
        (25 + dx, 255 + dy, 365 + dx, 430 + dy),
        (395 + dx, 255 + dy, 735 + dx, 430 + dy),
        (25 + dx, 475 + dy, 365 + dx, 650 + dy),
        (395 + dx, 475 + dy, 735 + dx, 650 + dy),
    ]
    for index, (box, shapes) in enumerate(zip(boxes, panels)):
        draw.rectangle(box, outline=frame, width=3)
        draw.text((box[0] + 12, box[1] + 9), LABELS[index], fill=frame, font=font)
        centers = (box[0] + 70, box[0] + 170, box[0] + 270)
        for center_x, shape in zip(centers, shapes):
            _draw_shape(draw, shape, (center_x, box[1] + 105), cell, ink)
    return image


def _winner_and_distances(scene: dict) -> tuple[int, list[int]]:
    reference = scene["reference_shape"]
    panels = scene["panels"]
    distances = [min(_shape_distance(reference, shape) for shape in panel) for panel in panels]
    winners = [index for index, value in enumerate(distances) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one matching panel")
    return winners[0], distances


def analytic_gold(scene: dict) -> str:
    winner, _ = _winner_and_distances(scene)
    return LABELS[winner]


def margin(scene: dict) -> float:
    winner, distances = _winner_and_distances(scene)
    return float(min(value for index, value in enumerate(distances) if index != winner))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 4.0
    except (ValueError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
