"""Pixel-only inverse arm for `slotted_slab_piece_count`.

The arm receives one rasterised page and nothing else: no scene, no lattice
order, no cell pitch, no origin, no slot list and no analytic gold. It segments
the near-black slab by darkness alone, recovers the cell pitch from the ink
itself by testing which square subdivision of the ink bounding box leaves every
cell either wholly inked or wholly blank, rebuilds the occupancy lattice, and
counts its connected components under edge adjacency. It abstains whenever the
page does not state a clean slotted-slab configuration: no ink, a non-square ink
bounding box, no consistent cell pitch, a disagreement between edge adjacency and
corner adjacency, or more than three pieces.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

DARK_LIMIT = 90
MIN_PITCH = 6
MIN_ORDER = 5
MAX_ORDER = 12
MAX_PIECES = 3
ABSTAIN = "ambiguous"


def _occupancy(mask: np.ndarray):
    """Recover the occupancy lattice of the ink, or None when the page is unclear."""
    rows = np.flatnonzero(mask.any(axis=1))
    columns = np.flatnonzero(mask.any(axis=0))
    if not len(rows) or not len(columns):
        return None
    top, bottom = int(rows[0]), int(rows[-1])
    left, right = int(columns[0]), int(columns[-1])
    height, width = bottom - top + 1, right - left + 1
    if height != width:
        return None
    side = height
    block = mask[top : top + side, left : left + side]
    for order in range(MIN_ORDER, MAX_ORDER + 1):
        if side % order:
            continue
        pitch = side // order
        if pitch < MIN_PITCH:
            continue
        counts = block.reshape(order, pitch, order, pitch).sum(axis=(1, 3))
        if not np.all((counts == 0) | (counts == pitch * pitch)):
            continue
        return counts == pitch * pitch
    return None


def _piece_count(occupied: np.ndarray, diagonal: bool) -> int:
    order = occupied.shape[0]
    seen = np.zeros_like(occupied, dtype=bool)
    steps = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    if diagonal:
        steps += [(-1, -1), (-1, 1), (1, -1), (1, 1)]
    pieces = 0
    for row in range(order):
        for column in range(order):
            if not occupied[row, column] or seen[row, column]:
                continue
            pieces += 1
            stack = [(row, column)]
            seen[row, column] = True
            while stack:
                y, x = stack.pop()
                for dy, dx in steps:
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < order and 0 <= nx < order and occupied[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
    return pieces


def decision_from_image(image) -> str:
    frame = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = (
        (frame[:, :, 0] < DARK_LIMIT) & (frame[:, :, 1] < DARK_LIMIT) & (frame[:, :, 2] < DARK_LIMIT)
    )
    if not mask.any():
        return ABSTAIN
    occupied = _occupancy(mask)
    if occupied is None:
        return ABSTAIN
    edgewise = _piece_count(occupied, diagonal=False)
    cornerwise = _piece_count(occupied, diagonal=True)
    if edgewise != cornerwise:
        return ABSTAIN
    if edgewise < 1 or edgewise > MAX_PIECES:
        return ABSTAIN
    return str(edgewise)


def load_decision(path: str) -> str:
    return decision_from_image(Image.open(path))
