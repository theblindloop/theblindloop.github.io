"""Scene sampler, raster renderer and analytic gold for `slotted_slab_piece_count`.

One near-black slab is drawn as a filled 9x9 lattice of square cells. Four white
slots are cut along whole lattice lines, each starting at one border of the slab.
A slot that reaches the far border severs the slab there; a slot that stops short
leaves a bridge of cells at its far end. Because slot ends can also miss each
other between neighbouring lines, the number of pieces is not the number of
severing slots: it is the number of connected components of the remaining cells
under edge adjacency.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
GRID = 9
SLAB_COLOR = (24, 26, 32)
PAPER = (255, 255, 255)
SLOT_COUNT = 4
REMOVED_CELLS = 26
MIN_PIECE_CELLS = 6
QUARANTINE_PIXELS = 12.0
BAND = 128
DECOR_COLORS = (
    (64, 116, 214),
    (232, 140, 44),
    (48, 152, 96),
    (150, 90, 200),
    (160, 160, 160),
    (60, 180, 190),
)


def _neighbours(cell):
    row, column = cell
    return ((row - 1, column), (row + 1, column), (row, column - 1), (row, column + 1))


def _diagonals(cell):
    row, column = cell
    return ((row - 1, column - 1), (row - 1, column + 1), (row + 1, column - 1), (row + 1, column + 1))


def _pieces(grid, removed):
    """Connected components of the remaining lattice cells under edge adjacency."""
    occupied = {
        (row, column)
        for row in range(grid)
        for column in range(grid)
        if [row, column] not in removed
    }
    seen = set()
    groups = []
    for cell in sorted(occupied):
        if cell in seen:
            continue
        stack = [cell]
        seen.add(cell)
        group = []
        while stack:
            current = stack.pop()
            group.append(current)
            for other in _neighbours(current):
                if other in occupied and other not in seen:
                    seen.add(other)
                    stack.append(other)
        groups.append(group)
    return groups


def _corner_only_contact(grid, removed):
    """True when two cells touch only at a corner, which the raster cannot state."""
    groups = _pieces(grid, removed)
    label = {}
    for index, group in enumerate(groups):
        for cell in group:
            label[cell] = index
    for cell, index in label.items():
        for other in _diagonals(cell):
            if other in label and label[other] != index:
                return True
    return False


def _line_cells(line, side, length, grid):
    if side == 0:
        span = range(0, length)
    else:
        span = range(grid - length, grid)
    return [[line, column] for column in span]


def _transpose(cells):
    return [[column, row] for row, column in cells]


def _lengths(rng, count, total, low, high):
    for _ in range(200):
        draw = [rng.randint(low, high) for _ in range(count - 1)]
        last = total - sum(draw)
        if low <= last <= high:
            draw.append(last)
            rng.shuffle(draw)
            return draw
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(1000003 + 7919 * int(seed))
    target = int(seed) % 3 + 1
    for _ in range(6000):
        cell = rng.choice((11, 12, 12, 13, 13))
        vertical = rng.random() < 0.5
        severing = {1: 0, 2: rng.choice((0, 1, 1, 1)), 3: rng.choice((1, 2, 2, 2))}[target]
        lines = rng.sample(range(1, GRID - 1), SLOT_COUNT)
        rng.shuffle(lines)
        blind = SLOT_COUNT - severing
        blind_total = REMOVED_CELLS - severing * GRID
        if blind_total < 2 * blind or blind_total > (GRID - 1) * blind:
            continue
        drawn = _lengths(rng, blind, blind_total, 2, GRID - 1)
        if drawn is None:
            continue
        lengths = [GRID] * severing + drawn
        removed = []
        for line, length in zip(lines, lengths):
            side = rng.randint(0, 1)
            removed.extend(_line_cells(line, side, length, GRID))
        if vertical:
            removed = _transpose(removed)
        removed = sorted(removed)
        if len(removed) != REMOVED_CELLS:
            continue
        groups = _pieces(GRID, removed)
        if len(groups) != target:
            continue
        if min(len(group) for group in groups) < MIN_PIECE_CELLS:
            continue
        if _corner_only_contact(GRID, removed):
            continue
        span = GRID * cell
        origin = [_axis_origin(rng, span), _axis_origin(rng, span)]
        decor = _sample_decor(rng, origin, span)
        return {
            "canvas": CANVAS,
            "grid": GRID,
            "cell": cell,
            "origin": origin,
            "removed": removed,
            "decor": decor,
        }
    raise RuntimeError(f"no valid slab configuration for seed {seed}")


def _axis_origin(rng, span):
    """Place the block so that its decisive support stays one small local patch.

    The block is kept inside one quarter-page band of each axis, uniformly jittered
    within that band and independently per axis, so it appears in any corner region
    of the page at a position that varies from scene to scene while never spreading
    its evidence across the page.
    """
    low = (2, max(2, BAND - span))
    high = (CANVAS - 128, CANVAS - span - 2)
    start, stop = low if rng.random() < 0.5 else high
    return rng.randint(start, max(start, stop))


def _sample_decor(rng, origin, span):
    keep_out = (origin[0] - 26, origin[1] - 26, origin[0] + span + 26, origin[1] + span + 26)
    items = []
    boxes = []
    attempts = 0
    wanted = rng.randint(3, 6)
    while len(items) < wanted and attempts < 400:
        attempts += 1
        width = rng.randint(22, 62)
        height = rng.randint(22, 62)
        x0 = rng.randint(12, CANVAS - 12 - width)
        y0 = rng.randint(12, CANVAS - 12 - height)
        box = (x0, y0, x0 + width, y0 + height)
        if box[0] < keep_out[2] and keep_out[0] < box[2] and box[1] < keep_out[3] and keep_out[1] < box[3]:
            continue
        if any(
            box[0] - 6 < other[2] and other[0] - 6 < box[2] and box[1] - 6 < other[3] and other[1] - 6 < box[3]
            for other in boxes
        ):
            continue
        boxes.append(box)
        items.append(
            [
                rng.choice(("ring", "frame", "block", "bar")),
                box[0],
                box[1],
                box[2],
                box[3],
                rng.randrange(len(DECOR_COLORS)),
                rng.randint(3, 6),
            ]
        )
    return items


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    grid = int(scene["grid"])
    cell = int(scene["cell"])
    origin = scene["origin"]
    removed = [tuple(item) for item in scene["removed"]]
    image = Image.new("RGB", (canvas, canvas), PAPER)
    draw = ImageDraw.Draw(image)
    for item in scene["decor"]:
        kind, x0, y0, x1, y1, color_index, width = item
        color = DECOR_COLORS[int(color_index) % len(DECOR_COLORS)]
        if kind == "ring":
            draw.ellipse([x0, y0, x1, y1], outline=color, width=int(width))
        elif kind == "frame":
            draw.rectangle([x0, y0, x1, y1], outline=color, width=int(width))
        elif kind == "block":
            draw.rectangle([x0, y0, x1, y1], fill=color)
        else:
            middle = (y0 + y1) // 2
            draw.rectangle([x0, middle - int(width), x1, middle + int(width)], fill=color)
    for row in range(grid):
        for column in range(grid):
            if (row, column) in removed:
                continue
            x0 = origin[0] + column * cell
            y0 = origin[1] + row * cell
            draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=SLAB_COLOR)
    return image


def analytic_gold(scene: dict) -> str:
    """Number of separate slab pieces, computed from the lattice alone."""
    groups = _pieces(int(scene["grid"]), scene["removed"])
    return str(len(groups))


def _weakest_cells(grid, removed):
    """Weakest decisive width, in lattice cells, of the drawn configuration."""
    occupied = [
        {column for column in range(grid) if [row, column] not in removed} for row in range(grid)
    ]
    occupied_columns = [
        {row for row in range(grid) if [row, column] not in removed} for column in range(grid)
    ]
    weakest = grid
    for runs in (occupied, occupied_columns):
        for index in range(grid - 1):
            first, second = runs[index], runs[index + 1]
            shared = first & second
            if shared:
                weakest = min(weakest, len(shared))
            elif first and second:
                gap = min(abs(a - b) for a in first for b in second)
                weakest = min(weakest, gap)
    return max(1, weakest)


def margin(scene: dict) -> float:
    return float(int(scene["cell"]) * _weakest_cells(int(scene["grid"]), scene["removed"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PIXELS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent rewrites that leave both the raster and the gold untouched."""
    reordered = dict(scene)
    reordered["removed"] = list(reversed(scene["removed"]))
    duplicated = dict(scene)
    duplicated["removed"] = list(scene["removed"]) + list(scene["removed"])
    return [("reversed_slot_cell_order", reordered), ("duplicated_slot_cells", duplicated)]
