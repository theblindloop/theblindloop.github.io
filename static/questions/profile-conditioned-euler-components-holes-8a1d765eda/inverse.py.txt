"""Independent pixel-only inverse for foreground components and light holes."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


@dataclass(frozen=True)
class TopologyMeasurement:
    components: int
    holes: int
    ink_pixels: int
    hole_areas: tuple[int, ...]


def _regions(mask: np.ndarray) -> list[tuple[int, bool]]:
    height, width = mask.shape
    remaining = mask.copy()
    regions: list[tuple[int, bool]] = []
    for flat in np.flatnonzero(remaining):
        y, x = divmod(int(flat), width)
        if not remaining[y, x]:
            continue
        remaining[y, x] = False
        queue = deque([(x, y)])
        area = 0
        touches_edge = False
        while queue:
            px, py = queue.popleft()
            area += 1
            touches_edge |= px == 0 or py == 0 or px == width - 1 or py == height - 1
            if px and remaining[py, px - 1]:
                remaining[py, px - 1] = False
                queue.append((px - 1, py))
            if px + 1 < width and remaining[py, px + 1]:
                remaining[py, px + 1] = False
                queue.append((px + 1, py))
            if py and remaining[py - 1, px]:
                remaining[py - 1, px] = False
                queue.append((px, py - 1))
            if py + 1 < height and remaining[py + 1, px]:
                remaining[py + 1, px] = False
                queue.append((px, py + 1))
        regions.append((area, touches_edge))
    return regions


def _dark_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    luminance = 0.2126 * pixels[:, :, 0] + 0.7152 * pixels[:, :, 1] + 0.0722 * pixels[:, :, 2]
    return luminance < 120.0


def measure_topology(image: Image.Image) -> TopologyMeasurement:
    dark = _dark_mask(image)
    ink_pixels = int(dark.sum())
    if ink_pixels < max(64, dark.size // 200):
        raise ValueError("dark foreground is absent or too small")
    if ink_pixels > dark.size * 0.8:
        raise ValueError("dark foreground occupies too much of the canvas")
    foreground = _regions(dark)
    background = _regions(~dark)
    hole_areas = tuple(sorted(area for area, touches_edge in background if not touches_edge))
    if hole_areas and min(hole_areas) < 64:
        raise ValueError("an enclosed light region is below the robust area floor")
    return TopologyMeasurement(len(foreground), len(hole_areas), ink_pixels, hole_areas)


def decision_from_image(image: Image.Image) -> str:
    measurement = measure_topology(image)
    return "equal" if measurement.components == measurement.holes else "different"
