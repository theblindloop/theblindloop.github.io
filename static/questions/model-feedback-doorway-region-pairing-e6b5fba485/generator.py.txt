"""Doorway region pairing: four mazes of rooms, eight dots, one same-region colour pair.

The scene is a 6 by 6 lattice of square rooms.  Every room boundary carries a
thick wall; a boundary that is a *doorway* is drawn with a central gap instead of
a solid run.  The doorways cut the lattice into exactly four regions, and each
region holds exactly two dots, so dot count carries no information.  Exactly one
region holds two dots of the same colour, and that colour is the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1000
MARGIN_PX = 50
CELLS = 6
CELL_PX = 150
WALL_PX = 12
DOT_R = 30

BACKGROUND = (247, 246, 242)
WALL = (28, 30, 34)

COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (214, 40, 40),
    "blue": (32, 82, 214),
    "green": (26, 140, 62),
    "purple": (140, 48, 196),
}
COLOUR_ORDER = ("red", "blue", "green", "purple")

GAP_MIN = 48
GAP_MAX = 78
QUARANTINE_GAP = 56


# --------------------------------------------------------------------------- #
# lattice helpers
# --------------------------------------------------------------------------- #
def _neighbours(cell: tuple[int, int]) -> list[tuple[int, int]]:
    row, col = cell
    out = []
    for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        nr, nc = row + dr, col + dc
        if 0 <= nr < CELLS and 0 <= nc < CELLS:
            out.append((nr, nc))
    return out


def _edge_key(a: tuple[int, int], b: tuple[int, int]) -> tuple[int, int, int, int]:
    first, second = sorted((a, b))
    return (first[0], first[1], second[0], second[1])


def _grow_regions(rng: random.Random) -> dict[tuple[int, int], int] | None:
    """Partition the lattice into four connected regions of nine cells each."""
    all_cells = [(r, c) for r in range(CELLS) for c in range(CELLS)]
    seeds = rng.sample(all_cells, 4)
    owner: dict[tuple[int, int], int] = {cell: index for index, cell in enumerate(seeds)}
    frontier: dict[int, list[tuple[int, int]]] = {
        index: [n for n in _neighbours(cell)] for index, cell in enumerate(seeds)
    }
    sizes = [1, 1, 1, 1]
    for _ in range(len(all_cells) * 8):
        if sum(sizes) == len(all_cells):
            break
        hungry = [i for i in range(4) if sizes[i] < 9 and frontier[i]]
        if not hungry:
            return None
        index = rng.choice(hungry)
        options = [cell for cell in frontier[index] if cell not in owner]
        if not options:
            frontier[index] = []
            continue
        pick = rng.choice(options)
        owner[pick] = index
        sizes[index] += 1
        frontier[index] = [cell for cell in frontier[index] if cell not in owner]
        frontier[index].extend(n for n in _neighbours(pick) if n not in owner)
    if sum(sizes) != len(all_cells) or sorted(sizes) != [9, 9, 9, 9]:
        return None
    return owner


def _spanning_doorways(
    rng: random.Random, owner: dict[tuple[int, int], int]
) -> list[tuple[int, int, int, int]]:
    """One random spanning tree per region; its edges become the doorways."""
    doorways: list[tuple[int, int, int, int]] = []
    for index in range(4):
        cells = sorted(cell for cell, who in owner.items() if who == index)
        start = cells[rng.randrange(len(cells))]
        visited = {start}
        stack = [start]
        while stack:
            current = stack[rng.randrange(len(stack))] if rng.random() < 0.35 else stack[-1]
            options = [
                n
                for n in _neighbours(current)
                if owner[n] == index and n not in visited
            ]
            if not options:
                stack.remove(current)
                continue
            nxt = rng.choice(options)
            visited.add(nxt)
            doorways.append(_edge_key(current, nxt))
            stack.append(nxt)
        if len(visited) != len(cells):
            return []
    return sorted(doorways)


def _region_paths(
    doorways: set[tuple[int, int, int, int]], cells: list[tuple[int, int]]
) -> dict[tuple[int, int], dict[tuple[int, int], int]]:
    """All-pairs doorway-graph distances inside one region."""
    dist: dict[tuple[int, int], dict[tuple[int, int], int]] = {}
    for source in cells:
        seen = {source: 0}
        queue = [source]
        head = 0
        while head < len(queue):
            current = queue[head]
            head += 1
            for n in _neighbours(current):
                if n in seen or _edge_key(current, n) not in doorways:
                    continue
                seen[n] = seen[current] + 1
                queue.append(n)
        dist[source] = seen
    return dist


def _manhattan(a: tuple[int, int], b: tuple[int, int]) -> int:
    return abs(a[0] - b[0]) + abs(a[1] - b[1])


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    for _ in range(600):
        owner = _grow_regions(rng)
        if owner is None:
            continue
        doorways = _spanning_doorways(rng, owner)
        if not doorways:
            continue
        door_set = set(doorways)
        regions = [
            sorted(cell for cell, who in owner.items() if who == index) for index in range(4)
        ]
        order = list(range(4))
        rng.shuffle(order)
        home = order[0]
        others = order[1:]
        dist = _region_paths(door_set, regions[home])
        far_pairs = [
            (a, b)
            for a in regions[home]
            for b in regions[home]
            if a < b and dist[a].get(b, 0) >= 5 and _manhattan(a, b) >= 3
        ]
        if not far_pairs:
            continue
        colours = list(COLOUR_ORDER)
        rng.shuffle(colours)
        target = colours[0]
        rest = colours[1:]
        pair = far_pairs[rng.randrange(len(far_pairs))]
        target_span = _manhattan(pair[0], pair[1])
        placement: dict[str, list[tuple[int, int]]] = {target: [pair[0], pair[1]]}
        # each remaining region holds two dots of two different colours
        slots = [(rest[0], rest[1]), (rest[1], rest[2]), (rest[2], rest[0])]
        for colour in rest:
            placement[colour] = []
        ok = True
        for region_index, (first, second) in zip(others, slots):
            cells = regions[region_index]
            picked = rng.sample(cells, 2)
            placement[first].append(picked[0])
            placement[second].append(picked[1])
        spans = {}
        for colour in rest:
            if len(placement[colour]) != 2:
                ok = False
                break
            spans[colour] = _manhattan(placement[colour][0], placement[colour][1])
        if not ok:
            continue
        # defeat the "closest pair" and the "farthest pair" shortcuts
        if not any(span <= 2 for span in spans.values()):
            continue
        if sum(1 for span in spans.values() if span >= 3) < 2:
            continue
        if not any(span >= target_span for span in spans.values()):
            continue
        gap = rng.randrange(GAP_MIN, GAP_MAX + 1, 2)
        scene = {
            "canvas": CANVAS,
            "cells": CELLS,
            "cell_px": CELL_PX,
            "wall_px": WALL_PX,
            "gap_px": int(gap),
            "doorways": [list(edge) for edge in doorways],
            "dots": {
                colour: [list(placement[colour][0]), list(placement[colour][1])]
                for colour in COLOUR_ORDER
            },
            "draw_order": [COLOUR_ORDER.index(c) for c in colours],
        }
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _origin(scene: dict) -> int:
    return (int(scene["canvas"]) - int(scene["cells"]) * int(scene["cell_px"])) // 2


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    cells = int(scene["cells"])
    step = int(scene["cell_px"])
    half = int(scene["wall_px"]) // 2
    gap = int(scene["gap_px"])
    off = _origin(scene)
    doorways = {tuple(int(v) for v in edge) for edge in scene["doorways"]}

    image = Image.new("RGB", (canvas, canvas), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def horizontal(row: int, col: int, open_gap: bool) -> None:
        """Wall on the boundary above room (row, col)."""
        y = off + row * step
        x0 = off + col * step
        x1 = x0 + step
        if open_gap:
            mid = (x0 + x1) // 2
            draw.rectangle([x0, y - half, mid - gap // 2, y + half - 1], fill=WALL)
            draw.rectangle([mid + gap // 2, y - half, x1 - 1, y + half - 1], fill=WALL)
        else:
            draw.rectangle([x0, y - half, x1 - 1, y + half - 1], fill=WALL)

    def vertical(row: int, col: int, open_gap: bool) -> None:
        """Wall on the boundary left of room (row, col)."""
        x = off + col * step
        y0 = off + row * step
        y1 = y0 + step
        if open_gap:
            mid = (y0 + y1) // 2
            draw.rectangle([x - half, y0, x + half - 1, mid - gap // 2], fill=WALL)
            draw.rectangle([x - half, mid + gap // 2, x + half - 1, y1 - 1], fill=WALL)
        else:
            draw.rectangle([x - half, y0, x + half - 1, y1 - 1], fill=WALL)

    for row in range(cells):
        for col in range(cells + 1):
            if col == 0 or col == cells:
                vertical(row, col, False)
            else:
                edge = _edge_key((row, col - 1), (row, col))
                vertical(row, col, edge in doorways)
    for col in range(cells):
        for row in range(cells + 1):
            if row == 0 or row == cells:
                horizontal(row, col, False)
            else:
                edge = _edge_key((row - 1, col), (row, col))
                horizontal(row, col, edge in doorways)
    # square junction blocks so no doorway leaks around a wall end
    for row in range(cells + 1):
        for col in range(cells + 1):
            x = off + col * step
            y = off + row * step
            draw.rectangle([x - half, y - half, x + half - 1, y + half - 1], fill=WALL)

    painted = [COLOUR_ORDER[int(index)] for index in scene["draw_order"]]
    for colour in painted:
        rgb = COLOURS[colour]
        for cell in scene["dots"][colour]:
            row, col = int(cell[0]), int(cell[1])
            cx = off + col * step + step // 2
            cy = off + row * step + step // 2
            draw.ellipse([cx - DOT_R, cy - DOT_R, cx + DOT_R, cy + DOT_R], fill=rgb)
    return image


# --------------------------------------------------------------------------- #
# analytic gold
# --------------------------------------------------------------------------- #
def _components(scene: dict) -> dict[tuple[int, int], int]:
    cells = int(scene["cells"])
    doorways = {tuple(int(v) for v in edge) for edge in scene["doorways"]}
    label: dict[tuple[int, int], int] = {}
    nxt = 0
    for row in range(cells):
        for col in range(cells):
            start = (row, col)
            if start in label:
                continue
            queue = [start]
            label[start] = nxt
            head = 0
            while head < len(queue):
                current = queue[head]
                head += 1
                for n in _neighbours(current):
                    if n in label or _edge_key(current, n) not in doorways:
                        continue
                    label[n] = nxt
                    queue.append(n)
            nxt += 1
    return label


def analytic_gold(scene: dict) -> str:
    label = _components(scene)
    winners = []
    for colour in COLOUR_ORDER:
        first, second = scene["dots"][colour]
        a = (int(first[0]), int(first[1]))
        b = (int(second[0]), int(second[1]))
        if label[a] == label[b]:
            winners.append(colour)
    if len(winners) != 1:
        raise ValueError(f"scene does not have exactly one same-region colour: {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    """Smallest pixel dimension that separates a doorway from a solid wall."""
    gap = int(scene["gap_px"])
    stub = (int(scene["cell_px"]) - gap) // 2
    return float(min(gap, stub * 2))


def is_quarantined(scene: dict) -> bool:
    return int(scene["gap_px"]) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_doors = dict(scene)
    reversed_doors["doorways"] = list(reversed(scene["doorways"]))
    swapped_dots = dict(scene)
    swapped_dots["dots"] = {
        colour: [list(pair[1]), list(pair[0])] for colour, pair in scene["dots"].items()
    }
    shuffled_draw = dict(scene)
    shuffled_draw["draw_order"] = list(reversed(scene["draw_order"]))
    return [
        ("reverse_doorway_serialisation", reversed_doors),
        ("swap_dot_endpoints", swapped_dots),
        ("reverse_draw_order", shuffled_draw),
    ]
