"""Pixel-only oracle: which colour has both of its dots in one walled region.

Receives a rendered RGB image and nothing else.  It segments the dark wall ink,
floods the free space between the walls, and reports the colour whose two dot
blobs land in the same flooded region.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BLOCK = 5
WALL_MAX = 110


class OracleAbstention(ValueError):
    """Raised when the raster does not present a readable scene."""


DOT_COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (214, 40, 40),
    "blue": (32, 82, 214),
    "green": (26, 140, 62),
    "purple": (140, 48, 196),
}


def _dot_centres(pixels: np.ndarray, rgb: tuple[int, int, int]) -> list[tuple[int, int]]:
    """Centroids of the connected blobs of one exact flat colour."""
    mask = np.all(pixels == np.array(rgb, dtype=np.int16), axis=2)
    if not mask.any():
        return []
    rows, cols = np.nonzero(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    centres: list[tuple[int, int]] = []
    for start_row, start_col in zip(rows.tolist(), cols.tolist()):
        if seen[start_row, start_col]:
            continue
        queue = deque([(start_row, start_col)])
        seen[start_row, start_col] = True
        blob: list[tuple[int, int]] = []
        while queue:
            row, col = queue.popleft()
            blob.append((row, col))
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nr, nc = row + dr, col + dc
                if 0 <= nr < mask.shape[0] and 0 <= nc < mask.shape[1]:
                    if mask[nr, nc] and not seen[nr, nc]:
                        seen[nr, nc] = True
                        queue.append((nr, nc))
        if len(blob) < 200:
            continue
        centres.append(
            (
                int(round(sum(p[0] for p in blob) / len(blob))),
                int(round(sum(p[1] for p in blob) / len(blob))),
            )
        )
    return sorted(centres)


def _free_blocks(pixels: np.ndarray) -> np.ndarray:
    """Coarse free-space mask: a block is free when it holds no wall ink."""
    wall = np.all(pixels <= WALL_MAX, axis=2)
    height, width = wall.shape
    rows = (height // BLOCK) * BLOCK
    cols = (width // BLOCK) * BLOCK
    folded = wall[:rows, :cols].reshape(rows // BLOCK, BLOCK, cols // BLOCK, BLOCK)
    return ~folded.any(axis=(1, 3))


def _flood_labels(free: np.ndarray) -> np.ndarray:
    labels = np.zeros(free.shape, dtype=np.int32)
    current = 0
    height, width = free.shape
    for row in range(height):
        for col in range(width):
            if not free[row, col] or labels[row, col]:
                continue
            current += 1
            queue = deque([(row, col)])
            labels[row, col] = current
            while queue:
                r, c = queue.popleft()
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < height and 0 <= nc < width:
                        if free[nr, nc] and not labels[nr, nc]:
                            labels[nr, nc] = current
                            queue.append((nr, nc))
    return labels


def _label_at(labels: np.ndarray, centre: tuple[int, int]) -> int:
    row = min(centre[0] // BLOCK, labels.shape[0] - 1)
    col = min(centre[1] // BLOCK, labels.shape[1] - 1)
    if labels[row, col]:
        return int(labels[row, col])
    for radius in range(1, 12):
        lo_r, hi_r = max(0, row - radius), min(labels.shape[0], row + radius + 1)
        lo_c, hi_c = max(0, col - radius), min(labels.shape[1], col + radius + 1)
        window = labels[lo_r:hi_r, lo_c:hi_c]
        hits = window[window > 0]
        if hits.size:
            return int(hits.flat[0])
    raise OracleAbstention("dot centre is not in any free region")


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    labels = _flood_labels(_free_blocks(pixels))
    winners: list[str] = []
    for colour, rgb in sorted(DOT_COLOURS.items()):
        centres = _dot_centres(pixels, rgb)
        if len(centres) != 2:
            raise OracleAbstention(f"expected two {colour} dots, found {len(centres)}")
        first = _label_at(labels, centres[0])
        second = _label_at(labels, centres[1])
        if first == second:
            winners.append(colour)
    if len(winners) != 1:
        raise OracleAbstention(f"image does not show exactly one same-region colour: {winners}")
    return winners[0]
