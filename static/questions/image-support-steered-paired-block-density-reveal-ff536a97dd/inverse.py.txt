from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 15
OUT = 5
TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _frame_bbox(arr: np.ndarray, target: tuple[int, int, int]) -> tuple[int, int, int, int]:
    color = np.array(target, dtype=np.int16)
    delta = np.abs(arr.astype(np.int16) - color).max(axis=2)
    ys, xs = np.where(delta <= 8)
    if len(xs) < 100:
        raise ValueError("colored panel fiducial is missing")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _read_panel(arr: np.ndarray, frame: tuple[int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = _frame_bbox(arr, frame)
    # Six-pixel frame surrounds the grid; infer its geometry solely from pixels.
    ix0, iy0, ix1, iy1 = x0 + 6, y0 + 6, x1 - 6, y1 - 6
    width, height = ix1 - ix0 + 1, iy1 - iy0 + 1
    if width != height or width < 90 or width % GRID:
        raise ValueError("panel fiducial has invalid geometry")
    cell = width // GRID
    cells = [[0] * GRID for _ in range(GRID)]
    for r in range(GRID):
        for c in range(GRID):
            # Sample a small patch away from all drawn grid lines.
            cx = ix0 + c * cell + cell // 2
            cy = iy0 + r * cell + cell // 2
            patch = arr[cy - 1:cy + 2, cx - 1:cx + 2].astype(np.float32)
            if patch.shape != (3, 3, 3):
                raise ValueError("microcell sample lies outside image")
            luminance = float((0.2126 * patch[..., 0] + 0.7152 * patch[..., 1] +
                               0.0722 * patch[..., 2]).mean())
            cells[r][c] = int(luminance < 125.0)
    return cells


def analyze_image(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"))
    left = _read_panel(arr, (18, 154, 154))
    right = _read_panel(arr, (229, 107, 93))
    grid = [[0] * OUT for _ in range(OUT)]
    local_margins = []
    for br in range(OUT):
        for bc in range(OUT):
            lc = sum(left[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3))
            rc = sum(right[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3))
            if lc == rc:
                raise ValueError("equal-count block pair is quarantined")
            grid[br][bc] = int(lc > rc)
            local_margins.append(abs(lc - rc))
    distances = sorted((sum(grid[r][c] != int(template[r][c])
                            for r in range(OUT) for c in range(OUT)), name)
                       for name, template in TEMPLATES.items())
    if distances[0][0] != 0 or distances[0][0] == distances[1][0]:
        raise ValueError("decoded grid is not an exact unambiguous symbol")
    return {
        "decision": distances[0][1],
        "margin": float(min(min(local_margins), distances[1][0] - distances[0][0])),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
