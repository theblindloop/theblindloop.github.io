from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
GRID = 15
BLOCK = 3
OUT = 5

TEMPLATES = {
    "X": (
        "10001", "01010", "00100", "01010", "10001",
    ),
    "plus": (
        "00100", "00100", "11111", "00100", "00100",
    ),
    "T": (
        "11111", "00100", "00100", "00100", "00100",
    ),
    "L": (
        "10000", "10000", "10000", "10000", "11111",
    ),
}
LABELS = tuple(TEMPLATES)


def _bits(label: str) -> list[list[int]]:
    return [[int(v) for v in row] for row in TEMPLATES[label]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = LABELS[int(seed) % len(LABELS)]
    target = _bits(label)
    left = [[0] * GRID for _ in range(GRID)]
    right = [[0] * GRID for _ in range(GRID)]
    # Each matching pair always has nine dark microcells in total. A target
    # bit is encoded by a 6-vs-3 local count, never by a fixed micro-position.
    for br in range(OUT):
        for bc in range(OUT):
            lc = 6 if target[br][bc] else 3
            rc = 9 - lc
            lp = list(range(9))
            rp = list(range(9))
            rng.shuffle(lp)
            rng.shuffle(rp)
            for k in lp[:lc]:
                left[br * 3 + k // 3][bc * 3 + k % 3] = 1
            for k in rp[:rc]:
                right[br * 3 + k // 3][bc * 3 + k % 3] = 1

    cell = 7
    panel = GRID * cell
    # Keep an ablation-stable empty column between the detached panels while
    # confining their joint span to a central, non-edge-crossing region.
    left_x = 100 + rng.randint(-2, 2)
    right_x = 348 + rng.randint(-2, 2)
    base_y = (HEIGHT - panel) // 2 + rng.randint(-10, 10)
    return {
        "left_cells": left,
        "right_cells": right,
        "left_origin": [left_x, base_y],
        "right_origin": [right_x, base_y],
        "cell_size": cell,
        "dark_color": "#233044",
        "light_color": "#f5f1e7",
        "background_color": "#fffdf8",
    }


def _validate_cells(cells: object) -> list[list[int]]:
    if not isinstance(cells, list) or len(cells) != GRID:
        raise ValueError("panel must have 15 rows")
    clean: list[list[int]] = []
    for row in cells:
        if not isinstance(row, list) or len(row) != GRID or any(v not in (0, 1) for v in row):
            raise ValueError("panel rows must contain 15 binary cells")
        clean.append([int(v) for v in row])
    return clean


def _draw_panel(draw: ImageDraw.ImageDraw, origin: list[int], cell: int,
                cells: list[list[int]], frame: str, dark: str, light: str) -> None:
    x0, y0 = map(int, origin)
    side = GRID * cell
    border = 6
    draw.rectangle((x0 - border, y0 - border, x0 + side + border - 1,
                    y0 + side + border - 1), fill=frame)
    for r in range(GRID):
        for c in range(GRID):
            x = x0 + c * cell
            y = y0 + r * cell
            fill = dark if cells[r][c] else light
            draw.rectangle((x, y, x + cell - 1, y + cell - 1), fill=fill)
    # Thin microcell lines and heavier 3x3 block boundaries make the declared
    # aggregation units directly observable.
    for k in range(GRID + 1):
        pos = k * cell
        width = 3 if k % BLOCK == 0 else 1
        draw.line((x0 + pos, y0, x0 + pos, y0 + side), fill="#8d96a0", width=width)
        draw.line((x0, y0 + pos, x0 + side, y0 + pos), fill="#8d96a0", width=width)


def render(scene: dict) -> Image.Image:
    left = _validate_cells(scene["left_cells"])
    right = _validate_cells(scene["right_cells"])
    cell = int(scene["cell_size"])
    dark = str(scene["dark_color"])
    light = str(scene["light_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), str(scene["background_color"]))
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, scene["left_origin"], cell, left, "#129a9a", dark, light)
    _draw_panel(draw, scene["right_origin"], cell, right, "#e56b5d", dark, light)
    return image


def _comparison_grid(scene: dict) -> list[list[int]]:
    left = _validate_cells(scene["left_cells"])
    right = _validate_cells(scene["right_cells"])
    out = [[0] * OUT for _ in range(OUT)]
    for br in range(OUT):
        for bc in range(OUT):
            lcount = sum(left[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3))
            rcount = sum(right[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3))
            out[br][bc] = int(lcount > rcount)
    return out


def _distances(grid: list[list[int]]) -> list[tuple[int, str]]:
    return sorted((sum(grid[r][c] != int(TEMPLATES[name][r][c])
                       for r in range(OUT) for c in range(OUT)), name)
                  for name in LABELS)


def analytic_gold(scene: dict) -> str:
    # Keep scene access explicit for the protected rendered-field audit.
    local = {"left_cells": scene["left_cells"], "right_cells": scene["right_cells"]}
    best = _distances(_comparison_grid(local))
    if best[0][0] != 0 or best[0][0] == best[1][0]:
        raise ValueError("comparison grid is not an unambiguous declared symbol")
    return best[0][1]


def margin(scene: dict) -> float:
    left = _validate_cells(scene["left_cells"])
    right = _validate_cells(scene["right_cells"])
    count_margin = min(abs(sum(left[br * 3 + r][bc * 3 + c]
                                   for r in range(3) for c in range(3)) -
                               sum(right[br * 3 + r][bc * 3 + c]
                                   for r in range(3) for c in range(3)))
                       for br in range(OUT) for bc in range(OUT))
    distances = _distances(_comparison_grid(scene))
    runner_gap = distances[1][0] - distances[0][0]
    return float(min(count_margin, runner_gap))


def is_quarantined(scene: dict) -> bool:
    try:
        left = _validate_cells(scene["left_cells"])
        right = _validate_cells(scene["right_cells"])
        if any(sum(left[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3)) ==
               sum(right[br * 3 + r][bc * 3 + c] for r in range(3) for c in range(3))
               for br in range(OUT) for bc in range(OUT)):
            return True
        distances = _distances(_comparison_grid(scene))
        return distances[0][0] != 0 or distances[0][0] == distances[1][0] or margin(scene) < 2
    except (TypeError, ValueError, KeyError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
