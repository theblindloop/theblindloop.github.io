"""Independent final-PNG inverse for affine triangle point transfer."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ANCHOR_RGB = {
    "cyan": (18, 160, 190),
    "orange": (235, 116, 35),
    "purple": (132, 79, 184),
}
CANDIDATE_RGB = {
    "red": (210, 51, 58),
    "blue": (46, 91, 190),
    "gold": (218, 160, 27),
}
RING_RGB = (218, 54, 150)
ANCHOR_ORDER = ("cyan", "orange", "purple")


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys)), len(xs)))
    return result


def _centers(arr: np.ndarray, rgb: tuple[int, int, int], minimum: int) -> list[tuple[float, float]]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    return [(x, y) for x, y, area in _components(mask) if area >= minimum]


def _barycentric(point: tuple[float, float], anchors: dict[str, tuple[float, float]]) -> tuple[float, float, float]:
    ax, ay = anchors["cyan"]
    bx, by = anchors["orange"]
    cx, cy = anchors["purple"]
    px, py = point
    determinant = (by - cy) * (ax - cx) + (cx - bx) * (ay - cy)
    if abs(determinant) < 1000:
        raise OracleError("pixel anchors are degenerate")
    wa = ((by - cy) * (px - cx) + (cx - bx) * (py - cy)) / determinant
    wb = ((cy - ay) * (px - cx) + (ax - cx) * (py - cy)) / determinant
    return (wa, wb, 1.0 - wa - wb)


def _distance(a: tuple[float, ...], b: tuple[float, ...]) -> float:
    return math.sqrt(sum((x - y) ** 2 for x, y in zip(a, b)))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    left, right = {}, {}
    for name, rgb in ANCHOR_RGB.items():
        centers = _centers(arr, rgb, 300)
        if len(centers) != 2:
            raise OracleError(f"expected two {name} anchors")
        left_centers = [center for center in centers if center[0] < arr.shape[1] / 2]
        right_centers = [center for center in centers if center[0] > arr.shape[1] / 2]
        if len(left_centers) != 1 or len(right_centers) != 1:
            raise OracleError(f"ambiguous {name} anchor panels")
        left[name], right[name] = left_centers[0], right_centers[0]
    rings = _centers(arr, RING_RGB, 60)
    if len(rings) != 1 or rings[0][0] >= arr.shape[1] / 2:
        raise OracleError("reference ring is missing or ambiguous")
    target = _barycentric(rings[0], left)
    distances = {}
    for label, rgb in CANDIDATE_RGB.items():
        centers = _centers(arr, rgb, 250)
        if len(centers) != 1 or centers[0][0] <= arr.shape[1] / 2:
            raise OracleError(f"expected one right-side {label} candidate")
        distances[label] = _distance(target, _barycentric(centers[0], right))
    ordered = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    if ordered[1][1] - ordered[0][1] < 0.12:
        raise OracleError("affine match has insufficient pixel margin")
    return ordered[0][0]
