"""Panel-parity overlay world.

Three 7x7 cell panels are drawn side by side. Superimposing them and keeping
the cells that are filled in an odd number of panels reveals one of five named
abstract symbols. No single panel carries the symbol: two panels are sampled as
unbiased coin flips and the third is forced so that the three-way parity equals
the (slightly perturbed) target symbol.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 7
NCELL = GRID * GRID
NPANEL = 3

SYMBOL_ART: dict[str, tuple[str, ...]] = {
    "arrow": (
        "...#...",
        "..###..",
        ".#.#.#.",
        "#..#..#",
        "...#...",
        "...#...",
        "...#...",
    ),
    "hourglass": (
        "#######",
        ".#####.",
        "..###..",
        "...#...",
        "..###..",
        ".#####.",
        "#######",
    ),
    "chevron": (
        ".......",
        "#.....#",
        "##...##",
        ".##.##.",
        "..###..",
        "...#...",
        ".......",
    ),
    "ell": (
        "#......",
        "#......",
        "#......",
        "#......",
        "#......",
        "#......",
        "#######",
    ),
    "ring": (
        ".#####.",
        ".#...#.",
        ".#...#.",
        ".#...#.",
        ".#...#.",
        ".#...#.",
        ".#####.",
    ),
}

SYMBOLS: tuple[str, ...] = tuple(sorted(SYMBOL_ART))

TEMPLATES: dict[str, tuple[int, ...]] = {
    name: tuple(1 if ch == "#" else 0 for row in art for ch in row)
    for name, art in SYMBOL_ART.items()
}

BACKGROUNDS = [(246, 243, 236), (238, 242, 247), (243, 238, 244), (236, 245, 239)]
FRAMES = [(38, 44, 58), (56, 40, 34), (30, 52, 48), (48, 36, 60)]
INKS = [(24, 62, 128), (132, 34, 44), (26, 92, 60), (94, 48, 120), (40, 40, 44)]

CELL = 26
BORDER = 3
INSET = 3
PANEL = GRID * CELL + 2 * BORDER
GAP = 34
MARGIN_PX = 30
QUARANTINE_MARGIN = 2.0


def _bits_to_grid(bits: tuple[int, ...]) -> list[list[int]]:
    return [list(bits[r * GRID : (r + 1) * GRID]) for r in range(GRID)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    n_flips = rng.choice([0, 1, 2, 2, 3])
    flips = sorted(rng.sample(range(NCELL), n_flips))
    p1 = [rng.randint(0, 1) for _ in range(NCELL)]
    p2 = [rng.randint(0, 1) for _ in range(NCELL)]
    return {
        "symbol": symbol,
        "flips": flips,
        "p1": p1,
        "p2": p2,
        "bg": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "frame": list(FRAMES[rng.randrange(len(FRAMES))]),
        "ink": list(INKS[rng.randrange(len(INKS))]),
        "dx": [rng.randint(-5, 5) for _ in range(NPANEL)],
        "dy": [rng.randint(-5, 5) for _ in range(NPANEL)],
        "cell": CELL,
        "border": BORDER,
        "gap": GAP,
        "pad": MARGIN_PX,
    }


def _target_bits(scene: dict) -> tuple[int, ...]:
    """The pattern the declared parity overlay produces: symbol XOR flips."""
    bits = list(TEMPLATES[scene["symbol"]])
    for index in scene["flips"]:
        bits[index] ^= 1
    return tuple(bits)


def _panels(scene: dict) -> list[tuple[int, ...]]:
    """Three panels whose cellwise parity is exactly ``_target_bits``."""
    target = _target_bits(scene)
    p1 = tuple(int(b) & 1 for b in scene["p1"])
    p2 = tuple(int(b) & 1 for b in scene["p2"])
    p3 = tuple(target[i] ^ p1[i] ^ p2[i] for i in range(NCELL))
    return [p1, p2, p3]


def _distances(bits: tuple[int, ...]) -> list[tuple[float, str]]:
    scored = [
        (float(sum(a ^ b for a, b in zip(bits, TEMPLATES[name]))), name)
        for name in SYMBOLS
    ]
    scored.sort(key=lambda item: (item[0], SYMBOLS.index(item[1])))
    return scored


def analytic_gold(scene: dict) -> str:
    return _distances(_target_bits(scene))[0][1]


def margin(scene: dict) -> float:
    scored = _distances(_target_bits(scene))
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def _naive_readable(scene: dict) -> int:
    """Smallest cell mismatch reachable without performing the parity overlay.

    Compares the revealed pattern against every single panel and against the
    union and intersection of the three panels; a small value here would mean
    the symbol leaks without the declared transform.
    """
    target = _target_bits(scene)
    panels = _panels(scene)
    union = tuple(p1 | p2 | p3 for p1, p2, p3 in zip(*panels))
    inter = tuple(p1 & p2 & p3 for p1, p2, p3 in zip(*panels))
    return min(
        sum(a ^ b for a, b in zip(target, cand))
        for cand in [*panels, union, inter]
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared pixel-identical, gold-preserving latent aliases."""
    reordered = dict(scene)
    reordered["flips"] = list(reversed(scene["flips"]))
    cancelled = dict(scene)
    cancelled["flips"] = list(scene["flips"]) + [7, 7]
    return [
        ("flips_reordered", reordered),
        ("flip_pair_cancellation", cancelled),
    ]


def _canvas_size(scene: dict) -> tuple[int, int]:
    pad = int(scene["pad"])
    panel = GRID * int(scene["cell"]) + 2 * int(scene["border"])
    width = 2 * pad + NPANEL * panel + (NPANEL - 1) * int(scene["gap"])
    height = 2 * pad + panel
    return width, height


def render(scene: dict) -> Image.Image:
    cell = int(scene["cell"])
    border = int(scene["border"])
    gap = int(scene["gap"])
    pad = int(scene["pad"])
    panel = GRID * cell + 2 * border
    width, height = _canvas_size(scene)
    image = Image.new("RGB", (width, height), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    frame = tuple(scene["frame"])
    ink = tuple(scene["ink"])
    for index, bits in enumerate(_panels(scene)):
        x0 = pad + index * (panel + gap) + int(scene["dx"][index])
        y0 = pad + int(scene["dy"][index])
        draw.rectangle([x0, y0, x0 + panel - 1, y0 + panel - 1], outline=frame, width=border)
        for row in range(GRID):
            for col in range(GRID):
                if not bits[row * GRID + col]:
                    continue
                cx = x0 + border + col * cell
                cy = y0 + border + row * cell
                draw.rectangle(
                    [cx + INSET, cy + INSET, cx + cell - 1 - INSET, cy + cell - 1 - INSET],
                    fill=ink,
                )
    return image
