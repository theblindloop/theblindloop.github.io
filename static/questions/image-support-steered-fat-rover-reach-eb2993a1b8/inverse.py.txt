"""Image-only inverse arm for the clearance-gated corridor world.

The oracle receives one rasterized PNG and nothing else.  It never imports the
renderer, the prompt module, the generator, the verifier or any analytic-gold
code, and it never sees a latent scene.

Method (an inverse of the forward construction, implemented independently):

0. abstain unless the raster actually shows an intact corridor network — one
   connected non-background structure enclosing four interior background
   pockets — because the published traversal rule presupposes it;
1. label every pixel by nearest palette color, so antialiased edges split at the
   ~50% mix which sits on the true silhouette;
2. the *free space* is every non-background pixel (corridor ink plus the rover
   body and the marker dots that are painted on top of it);
3. measure the rover radius from its own ink area, ``rho = sqrt(area / pi)``;
4. morphologically erode the free space by a disk of that radius — this is the
   configuration space in which the rover center may legally sit;
5. run connected components over the eroded set with a run-length union-find and
   report, for each marker color, whether its centroid lies in the same
   component as the rover centroid.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "fat_rover_reach-oracle-0.1.0"

_BACKGROUND = (255, 255, 255)
_CORRIDOR = (58, 102, 158)
_ROVER = (25, 25, 25)
_MARKERS = (
    ("red", (211, 47, 47)),
    ("orange", (245, 140, 20)),
    ("green", (46, 160, 67)),
)
_PALETTE = [_BACKGROUND, _CORRIDOR, _ROVER] + [rgb for _, rgb in _MARKERS]

_MIN_ROVER_PIXELS = 40
_MIN_MARKER_PIXELS = 20

# The published question presupposes an intact corridor network: one connected
# blue structure whose corridors enclose exactly four interior background
# pockets (plus the surrounding page).  If the raster does not show that
# structure the traversal rule has no referent, so the inverse arm abstains
# instead of guessing from a mutilated picture.
_EXPECTED_FOREGROUND_REGIONS = 1
_EXPECTED_BACKGROUND_POCKETS = 5
_MIN_REGION_PIXELS = 64


class OracleReadError(ValueError):
    """The raster does not contain a measurable corridor scene."""


def _label(image):
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    stack = np.stack(
        [((arr - np.array(rgb, dtype=np.float32)) ** 2).sum(axis=-1) for rgb in _PALETTE],
        axis=-1,
    )
    return stack.argmin(axis=-1)


def _shift(array, dy, dx):
    height, width = array.shape
    out = np.zeros_like(array)
    i0, i1 = max(0, -dy), min(height, height - dy)
    j0, j1 = max(0, -dx), min(width, width - dx)
    if i0 < i1 and j0 < j1:
        out[i0:i1, j0:j1] = array[i0 + dy : i1 + dy, j0 + dx : j1 + dx]
    return out


def _erode_disk(mask, radius):
    """Erode ``mask`` by a discrete disk of the given (fractional) radius."""

    span = int(math.floor(radius))
    horizontal = [mask]
    for width in range(1, span + 1):
        horizontal.append(horizontal[width - 1] & _shift(mask, 0, -width) & _shift(mask, 0, width))
    eroded = None
    for dy in range(-span, span + 1):
        half = int(math.floor(math.sqrt(max(radius * radius - dy * dy, 0.0))))
        layer = _shift(horizontal[min(half, span)], dy, 0)
        eroded = layer if eroded is None else (eroded & layer)
    return mask if eroded is None else eroded


def _run_components(mask):
    """Label 4-connected components using run-length union-find.

    Returns ``(runs_per_row, root_of_run)`` where ``runs_per_row[y]`` is a list
    of ``(x_start, x_end_exclusive, run_index)`` tuples.
    """

    height = mask.shape[0]
    parent: list[int] = []

    def find(node):
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    def union(left, right):
        left, right = find(left), find(right)
        if left != right:
            parent[left] = right

    rows: list[list[tuple[int, int, int]]] = []
    previous: list[tuple[int, int, int]] = []
    for y in range(height):
        flags = mask[y].astype(np.int8)
        edges = np.diff(np.concatenate(([0], flags, [0])))
        starts = np.flatnonzero(edges == 1)
        stops = np.flatnonzero(edges == -1)
        current: list[tuple[int, int, int]] = []
        for start, stop in zip(starts.tolist(), stops.tolist()):
            index = len(parent)
            parent.append(index)
            for prev_start, prev_stop, prev_index in previous:
                if prev_start < stop and start < prev_stop:
                    union(index, prev_index)
            current.append((start, stop, index))
        rows.append(current)
        previous = current
    return rows, find


def _component_at(rows, find, x, y):
    if y < 0 or y >= len(rows):
        return None
    for start, stop, index in rows[y]:
        if start <= x < stop:
            return find(index)
    return None


def _count_regions(mask, min_pixels=_MIN_REGION_PIXELS):
    """Number of 4-connected components of a boolean mask, ignoring specks.

    Sub-pixel antialiasing at a shallow capsule junction can leave a single
    stray pixel; a region smaller than ``min_pixels`` is such a speck and is not
    a structural pocket.
    """

    rows, find = _run_components(mask)
    sizes: dict[int, int] = {}
    for row in rows:
        for start, stop, index in row:
            root = find(index)
            sizes[root] = sizes.get(root, 0) + (stop - start)
    return sum(1 for size in sizes.values() if size >= min_pixels)


def _centroid(mask):
    ys, xs = np.nonzero(mask)
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    """Recover the reachability profile of the three dots from pixels alone."""

    labels = _label(image)
    free = labels != 0
    if _count_regions(free) != _EXPECTED_FOREGROUND_REGIONS:
        raise OracleReadError("the corridor structure is not a single connected region")
    if _count_regions(~free) != _EXPECTED_BACKGROUND_POCKETS:
        raise OracleReadError("the corridor structure does not enclose four interior pockets")
    rover_mask = labels == 2
    rover_pixels = int(rover_mask.sum())
    if rover_pixels < _MIN_ROVER_PIXELS:
        raise OracleReadError("no rover body found in the raster")
    radius = math.sqrt(rover_pixels / math.pi)
    rover_x, rover_y = _centroid(rover_mask)

    configuration = _erode_disk(free, radius)
    rows, find = _run_components(configuration)
    home = _component_at(rows, find, int(round(rover_x)), int(round(rover_y)))

    flags = []
    for offset, (color, _rgb) in enumerate(_MARKERS):
        marker_mask = labels == (3 + offset)
        if int(marker_mask.sum()) < _MIN_MARKER_PIXELS:
            raise OracleReadError(f"marker {color} is not visible in the raster")
        mx, my = _centroid(marker_mask)
        target = _component_at(rows, find, int(round(mx)), int(round(my)))
        reachable = home is not None and target is not None and target == home
        flags.append(f"{color}={1 if reachable else 0}")
    return "|".join(flags)
