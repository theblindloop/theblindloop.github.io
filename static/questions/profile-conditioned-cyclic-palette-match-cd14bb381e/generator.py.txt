"""Deterministic scene sampler and raster renderer."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1100, 430
SYMBOLS = ("p", "q", "r", "s")
RGB_POOL = ((220, 55, 55), (35, 145, 85), (45, 105, 210), (235, 145, 30))
LABELS = ("A", "B", "C", "D")


def _canon(order: list[str] | tuple[str, ...]) -> tuple[str, ...]:
    values = tuple(order)
    return min(values[index:] + values[:index] for index in range(len(values)))


def _alignment(reference: list[str], candidate: list[str]) -> int:
    return max(
        sum(reference[pos] == candidate[(pos + shift) % 4] for pos in range(4))
        for shift in range(4)
    )


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    reference = list(SYMBOLS)
    rng.shuffle(reference)
    target_index = int(seed) % 4

    cyclic_classes: dict[tuple[str, ...], list[str]] = {}
    import itertools

    for perm in itertools.permutations(SYMBOLS):
        cyclic_classes.setdefault(_canon(perm), list(perm))
    wrong = [list(key) for key in sorted(cyclic_classes) if key != _canon(reference)]
    rng.shuffle(wrong)
    orders: list[list[str]] = []
    wrong_index = 0
    for index in range(4):
        if index == target_index:
            shift = rng.randrange(4)
            orders.append(reference[shift:] + reference[:shift])
        else:
            item = wrong[wrong_index]
            wrong_index += 1
            shift = rng.randrange(4)
            orders.append(item[shift:] + item[:shift])

    visible = list(RGB_POOL)
    rng.shuffle(visible)
    palette = {symbol: list(rgb) for symbol, rgb in zip(SYMBOLS, visible)}
    phases = [rng.randrange(0, 360, 15) for _ in range(5)]
    return {"orders": [reference, *orders], "phases": phases, "palette": palette}


def analytic_gold(scene: dict[str, Any]) -> str:
    orders = scene["orders"]
    reference = _canon(orders[0])
    matches = [index for index, order in enumerate(orders[1:]) if _canon(order) == reference]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one cyclic match")
    return LABELS[matches[0]]


def margin(scene: dict[str, Any]) -> float:
    orders = scene["orders"]
    scores = sorted(
        (_alignment(orders[0], order) for order in orders[1:]), reverse=True
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < 1.5
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    paths = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf",
    )
    for path in paths:
        try:
            return ImageFont.truetype(path, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    x = xy[0] - (box[2] - box[0]) / 2
    y = xy[1] - (box[3] - box[1]) / 2 - box[1]
    draw.text((x, y), text, fill=(35, 38, 45), font=font)


def _draw_cycle(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    order: list[str],
    phase: int,
    palette: dict[str, list[int]],
) -> None:
    cx, cy = center
    radius = 61
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), outline=(175, 180, 188), width=3)
    for slot, symbol in enumerate(order):
        angle = math.radians(phase + 90 * slot)
        x = int(round(cx + radius * math.cos(angle)))
        y = int(round(cy + radius * math.sin(angle)))
        marker_radius = 14
        draw.ellipse(
            (x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius),
            fill=tuple(palette[symbol]),
            outline=(28, 31, 38),
            width=3,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    orders = scene["orders"]
    phases = scene["phases"]
    palette = scene["palette"]
    if len(orders) != 5 or len(phases) != 5:
        raise ValueError("expected one reference and four candidate cycles")

    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    label_font = _font(24)
    small_font = _font(16)
    _centered(draw, (WIDTH / 2, 31), "MATCH THE CLOCKWISE COLOR ORDER", title_font)
    _centered(draw, (WIDTH / 2, 62), "Rotation is allowed • mirror reversal is not", small_font)

    panels = [(25, 90, 235, 375)]
    panels.extend((280 + 202 * i, 90, 465 + 202 * i, 375) for i in range(4))
    for index, box in enumerate(panels):
        fill = (238, 241, 244) if index == 0 else (253, 253, 251)
        draw.rounded_rectangle(box, radius=13, fill=fill, outline=(92, 99, 110), width=3)
        cx = (box[0] + box[2]) // 2
        cy = 231
        _draw_cycle(draw, (cx, cy), orders[index], int(phases[index]), palette)
        caption = "REFERENCE" if index == 0 else LABELS[index - 1]
        _centered(draw, (cx, 341), caption, label_font)
    return image
