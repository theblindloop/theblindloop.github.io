"""Independent inverse arm using only pixels and generic component geometry."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
VISIBLE_COLORS = (
    (220, 55, 55),
    (35, 145, 85),
    (45, 105, 210),
    (235, 145, 30),
)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 150:
            components.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return components


def _canon(values: list[int]) -> tuple[int, ...]:
    seq = tuple(values)
    return min(seq[index:] + seq[:index] for index in range(len(seq)))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    points: list[tuple[float, float, int]] = []
    for token, color in enumerate(VISIBLE_COLORS):
        target = np.asarray(color, dtype=np.int16)
        distance = np.max(np.abs(array - target), axis=2)
        for x, y, area in _components(distance <= 3):
            if 250 <= area <= 900:
                points.append((x, y, token))
    if len(points) != 20:
        raise ValueError(f"expected 20 visible marker components, found {len(points)}")

    # Regions do not overlap horizontally. Enumerate them from the component
    # ordering itself, without using renderer coordinates or panel dimensions.
    ordered = sorted(points, key=lambda item: item[0])
    groups = [ordered[index : index + 4] for index in range(0, 20, 4)]
    if len(groups) != 5 or any(len(group) != 4 for group in groups):
        raise ValueError("could not enumerate five four-marker regions")
    groups.sort(key=lambda group: sum(point[0] for point in group) / len(group))

    cycles: list[tuple[int, ...]] = []
    for group in groups:
        cx = sum(point[0] for point in group) / 4
        cy = sum(point[1] for point in group) / 4
        clockwise = sorted(group, key=lambda point: math.atan2(point[1] - cy, point[0] - cx))
        cycles.append(_canon([point[2] for point in clockwise]))
    matches = [index for index, cycle in enumerate(cycles[1:]) if cycle == cycles[0]]
    if len(matches) != 1:
        raise ValueError(f"expected one matching candidate, found {len(matches)}")
    return LABELS[matches[0]]
