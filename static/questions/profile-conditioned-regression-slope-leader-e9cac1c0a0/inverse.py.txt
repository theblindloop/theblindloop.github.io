"""Independent PNG-only inverse arm for regression_slope_leader."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public visual palette is duplicated here; no renderer or latent code is imported.
_NAMES = ("red", "blue", "green", "orange")
_PALETTE = np.asarray([
    (220, 52, 61),
    (45, 105, 200),
    (33, 160, 95),
    (240, 135, 35),
    (248, 249, 252),
    (42, 48, 58),
    (215, 220, 228),
], dtype=np.int16)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = {tuple(v) for v in np.argwhere(mask).tolist()}
    found: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        pixels = [start]
        while queue:
            y, x = queue.popleft()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in remaining:
                        remaining.remove(neighbor)
                        queue.append(neighbor)
                        pixels.append(neighbor)
        found.append(np.asarray(pixels, dtype=float))
    return found


def _centers_for_color(arr: np.ndarray, label: int) -> np.ndarray:
    delta = arr.astype(np.int32) - _PALETTE[label].astype(np.int32)
    # An absolute chroma ball excludes gray antialiasing that can be nearest to
    # green while retaining the solid and inner antialiased portion of each dot.
    mask = np.sum(delta * delta, axis=2) < 65 * 65
    components = [part for part in _components(mask) if 55 <= len(part) <= 260]
    if len(components) != 7:
        raise ValueError(f"expected seven isolated point marks for {_NAMES[label]}, found {len(components)}")
    # Component arrays are (row=y, column=x); return Cartesian (x, y-screen).
    return np.asarray([[part[:, 1].mean(), part[:, 0].mean()] for part in components], dtype=float)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    slopes: dict[str, float] = {}
    for index, name in enumerate(_NAMES):
        pts = _centers_for_color(arr, index)
        x = pts[:, 0]
        y_up = -pts[:, 1]
        denom = float(np.dot(x - x.mean(), x - x.mean()))
        if denom < 1.0:
            raise ValueError(f"degenerate horizontal spread for {name}")
        slopes[name] = float(np.dot(x - x.mean(), y_up - y_up.mean()) / denom)
    ranking = sorted(slopes, key=slopes.get, reverse=True)
    if slopes[ranking[0]] - slopes[ranking[1]] < 0.015:
        raise ValueError("pixel slope decision is ambiguous")
    return ranking[0]
