"""Deterministic latent sampler, renderer, and analytic arm."""

from __future__ import annotations

import copy
import random

import numpy as np
from PIL import Image, ImageDraw, ImageFont

COLOR_NAMES = ("red", "blue", "green", "orange")
RGB = {
    "red": (220, 52, 61),
    "blue": (45, 105, 200),
    "green": (33, 160, 95),
    "orange": (240, 135, 35),
}
BACKGROUND = (248, 249, 252)
INK = (42, 48, 58)
GRID = (215, 220, 228)
QUARANTINE_GAP = 0.12


def _orthogonal_residuals(rng: random.Random, xs: list[float]) -> np.ndarray:
    """Small residuals with zero mean and zero sample covariance with x."""
    x = np.asarray(xs, dtype=float)
    r = np.asarray([rng.uniform(-1.0, 1.0) for _ in xs], dtype=float)
    r -= r.mean()
    xc = x - x.mean()
    r -= xc * float(np.dot(r, xc) / np.dot(xc, xc))
    peak = float(np.max(np.abs(r)))
    return r / peak * rng.uniform(0.018, 0.030) if peak else r


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLOR_NAMES)
    winner = colors[int(seed) % len(colors)]
    # Every sixth scene deliberately approaches the visible equality boundary.
    slope_bank = [-0.58, -0.12, 0.43, 0.51] if int(seed) % 6 == 0 else [-0.61, -0.11, 0.27, 0.67]
    remaining_colors = [c for c in colors if c != winner]
    remaining_slopes = slope_bank[:-1]
    rng.shuffle(remaining_colors)
    rng.shuffle(remaining_slopes)
    assigned = dict(zip(remaining_colors, remaining_slopes, strict=True))
    assigned[winner] = slope_bank[-1]

    points: dict[str, list[list[float]]] = {}
    for color in colors:
        xs = [0.14 + i * 0.12 + rng.uniform(-0.010, 0.010) for i in range(7)]
        residual = _orthogonal_residuals(rng, xs)
        intercept = rng.uniform(0.47, 0.53)
        ys = [intercept + assigned[color] * (x - 0.5) + float(e) for x, e in zip(xs, residual, strict=True)]
        if not all(0.08 < y < 0.92 for y in ys):
            raise RuntimeError("sampled point escaped the visible plotting range")
        points[color] = [[round(x, 7), round(y, 7)] for x, y in zip(xs, ys, strict=True)]

    panel_order = colors[:]
    rng.shuffle(panel_order)
    return {
        "canvas": [640, 590],
        "point_radius": 6,
        "panel_order": panel_order,
        "points": points,
    }


def _slopes(scene: dict) -> dict[str, float]:
    values: dict[str, float] = {}
    for color in scene["panel_order"]:
        pts = np.asarray(scene["points"][color], dtype=float)
        x = pts[:, 0]
        y = pts[:, 1]
        values[color] = float(np.dot(x - x.mean(), y - y.mean()) / np.dot(x - x.mean(), x - x.mean()))
    return values


def analytic_gold(scene: dict) -> str:
    slopes = _slopes(scene)
    return max(scene["panel_order"], key=lambda color: slopes[color])


def margin(scene: dict) -> float:
    ranked = sorted(_slopes(scene).values(), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size=size)


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float], **kwargs) -> None:
        draw.rectangle(tuple(int(round(v * scale)) for v in values), **kwargs)

    title_font = _font(17 * scale)
    label_font = _font(15 * scale)
    draw.text((width * scale // 2, 20 * scale), "EQUAL-SCALE SCATTER PANELS", fill=INK,
              font=title_font, anchor="ma")

    panels = [(65, 70), (365, 70), (65, 335), (365, 335)]
    plot_size = 210
    labels = ("A", "B", "C", "D")
    for panel_index, (left, top) in enumerate(panels):
        color = scene["panel_order"][panel_index]
        # Visible common grid and square frame make the shared scale explicit.
        for frac in (0.25, 0.50, 0.75):
            gx = int(round((left + frac * plot_size) * scale))
            gy = int(round((top + frac * plot_size) * scale))
            draw.line((gx, top * scale, gx, (top + plot_size) * scale), fill=GRID, width=2 * scale)
            draw.line((left * scale, gy, (left + plot_size) * scale, gy), fill=GRID, width=2 * scale)
        box((left, top, left + plot_size, top + plot_size), outline=INK, width=2 * scale)
        draw.text(((left - 19) * scale, (top + 3) * scale), labels[panel_index], fill=INK,
                  font=label_font, anchor="ma")
        # Small positive-axis arrowheads disambiguate conventional x-right/y-up axes.
        draw.polygon([
            ((left + plot_size) * scale, (top + plot_size) * scale),
            ((left + plot_size - 8) * scale, (top + plot_size - 4) * scale),
            ((left + plot_size - 8) * scale, (top + plot_size + 4) * scale),
        ], fill=INK)
        draw.polygon([
            (left * scale, top * scale),
            ((left - 4) * scale, (top + 8) * scale),
            ((left + 4) * scale, (top + 8) * scale),
        ], fill=INK)
        radius = int(scene["point_radius"]) * scale
        for x, y in scene["points"][color]:
            px = int(round((left + float(x) * plot_size) * scale))
            py = int(round((top + (1.0 - float(y)) * plot_size) * scale))
            draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=RGB[color])

    return image.resize((width, height), Image.Resampling.LANCZOS)
