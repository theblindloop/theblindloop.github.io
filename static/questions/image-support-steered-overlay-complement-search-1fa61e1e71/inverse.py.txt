"""Independent pixel-only inverse arm for exact-overlay complement search.

The implementation discovers the eight visible grid components, groups them by
their raster arrangement, classifies all cell centers by color, and enumerates
all four candidate pairs. It receives neither scene data nor renderer state.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_GRID_RGB = np.array((55, 65, 78), dtype=np.int16)
_CYAN_RGB = np.array((18, 166, 184), dtype=np.int16)
_GOLD_RGB = np.array((226, 157, 25), dtype=np.int16)
_ANSWERS = ("upper-left", "upper-right", "lower-left", "lower-right")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _read_grid(array: np.ndarray, box: tuple[int, int, int, int, int]) -> tuple[str, list[int]]:
    x0, y0, x1, y1, _ = box
    width, height = x1 - x0, y1 - y0
    if not (62 <= width <= 66 and 62 <= height <= 66):
        raise ValueError("malformed grid geometry")
    values = []
    kinds = []
    for r in range(4):
        for c in range(4):
            cx = round(x0 + (c + 0.5) * width / 4)
            cy = round(y0 + (r + 0.5) * height / 4)
            patch = array[cy - 3:cy + 4, cx - 3:cx + 4].astype(np.int16)
            rgb = np.median(patch.reshape(-1, 3), axis=0)
            dc = float(np.linalg.norm(rgb - _CYAN_RGB))
            dg = float(np.linalg.norm(rgb - _GOLD_RGB))
            if min(dc, dg) < 45.0:
                values.append(1)
                kinds.append("cyan" if dc < dg else "gold")
            else:
                values.append(0)
    if sum(values) != 8 or len(set(kinds)) != 1:
        raise ValueError("grid must contain exactly eight cells of one color")
    return kinds[0], values


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 128:
        raise ValueError("unsupported raster")
    dark_distance = np.linalg.norm(array.astype(np.int16) - _GRID_RGB, axis=2)
    boxes = []
    for component in _components(dark_distance < 18.0):
        x0, y0, x1, y1, area = component
        if 62 <= x1 - x0 <= 66 and 62 <= y1 - y0 <= 66 and area >= 650:
            boxes.append(component)
    if len(boxes) != 8:
        raise ValueError(f"expected eight complete grids, found {len(boxes)}")

    boxes.sort(key=lambda item: ((item[1] + item[3]) / 2, (item[0] + item[2]) / 2))
    if ((boxes[4][1] + boxes[4][3]) - (boxes[3][1] + boxes[3][3])) / 2 < 80:
        raise ValueError("candidate rows are not visibly separated")

    scores = []
    for row_index, row_boxes in enumerate((boxes[:4], boxes[4:])):
        row_boxes = sorted(row_boxes, key=lambda item: item[0] + item[2])
        for col_index in range(2):
            left = _read_grid(array, row_boxes[2 * col_index])
            right = _read_grid(array, row_boxes[2 * col_index + 1])
            if left[0] != "cyan" or right[0] != "gold":
                raise ValueError("each panel must visibly pair cyan then gold")
            score = sum(abs(1 - (a + b)) for a, b in zip(left[1], right[1], strict=True))
            scores.append((score, row_index * 2 + col_index))
    scores.sort()
    if scores[0][0] != 0 or scores[1][0] < 4:
        raise ValueError("no unique exact-cover winner with required runner-up margin")
    return _ANSWERS[scores[0][1]]
