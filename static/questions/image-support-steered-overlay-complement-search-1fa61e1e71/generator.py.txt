"""Deterministic scene sampler and raster renderer for exact-overlay search."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 768
BACKGROUND = (248, 249, 247)
CYAN = (18, 166, 184)
GOLD = (226, 157, 25)
GRID = (55, 65, 78)
FRAME = (205, 211, 215)
CELL = 16
SIDE = 4
GRID_PX = CELL * SIDE
ANSWERS = ("upper-left", "upper-right", "lower-left", "lower-right")

# Candidate pairs occupy two separated horizontal bands in a central region.
_BASES = ((213, 190), (411, 190), (213, 514), (411, 514))


def _valid_mask(mask: list[int]) -> bool:
    return (
        sum(mask) == 8
        and all(any(mask[r * SIDE + c] for c in range(SIDE)) for r in range(SIDE))
        and all(any(mask[r * SIDE + c] for r in range(SIDE)) for c in range(SIDE))
    )


def _random_mask(rng: random.Random) -> list[int]:
    while True:
        ones = set(rng.sample(range(16), 8))
        mask = [int(i in ones) for i in range(16)]
        if _valid_mask(mask):
            return mask


def _near_complement(rng: random.Random, cyan: list[int], swaps: int) -> list[int]:
    gold = [1 - value for value in cyan]
    remove = rng.sample([i for i, value in enumerate(gold) if value], swaps)
    add = rng.sample([i for i, value in enumerate(gold) if not value], swaps)
    for i in remove:
        gold[i] = 0
    for i in add:
        gold[i] = 1
    return gold


def _score(panel: dict) -> int:
    return sum(abs(1 - (a + b)) for a, b in zip(panel["cyan"], panel["gold"], strict=True))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner_slot = int(seed) % 4
    swap_counts = [2, 3, 4]
    rng.shuffle(swap_counts)
    distractor_index = 0
    panels = []
    for slot in range(4):
        cyan = _random_mask(rng)
        if slot == winner_slot:
            gold = [1 - value for value in cyan]
        else:
            # Both colors retain eight cells. Each swap creates one visible gap
            # and one visible overlap, so cheap total-ink cues remain constant.
            gold = _near_complement(rng, cyan, swap_counts[distractor_index])
            distractor_index += 1
        panels.append({"slot": slot, "cyan": cyan, "gold": gold})
    # Serialized order has no visual or semantic meaning; slots determine display.
    rng.shuffle(panels)
    return {"panels": panels}


def _draw_grid(draw: ImageDraw.ImageDraw, x0: int, y0: int, mask: list[int], color) -> None:
    for r in range(SIDE):
        for c in range(SIDE):
            if mask[r * SIDE + c]:
                draw.rectangle(
                    (x0 + c * CELL + 2, y0 + r * CELL + 2,
                     x0 + (c + 1) * CELL - 2, y0 + (r + 1) * CELL - 2),
                    fill=color,
                )
    for k in range(SIDE + 1):
        draw.line((x0 + k * CELL, y0, x0 + k * CELL, y0 + GRID_PX), fill=GRID, width=3)
        draw.line((x0, y0 + k * CELL, x0 + GRID_PX, y0 + k * CELL), fill=GRID, width=3)


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for panel in panels:
        slot = int(panel["slot"])
        x0, y0 = _BASES[slot]
        draw.rounded_rectangle((x0 - 12, y0 - 16, x0 + 156, y0 + 80), radius=12,
                               outline=FRAME, width=3)
        _draw_grid(draw, x0, y0, panel["cyan"], CYAN)
        _draw_grid(draw, x0 + 80, y0, panel["gold"], GOLD)
        # A small neutral plus sign makes the intended pair grouping explicit.
        draw.line((x0 + 70, y0 + 27, x0 + 74, y0 + 37), fill=FRAME, width=3)
        draw.line((x0 + 66, y0 + 32, x0 + 78, y0 + 32), fill=FRAME, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    # Kept inline so the rendered-field audit sees only the one top-level scene
    # input (``panels``); cyan/gold are nested fields of each visible panel.
    scored = sorted(
        (
            sum(
                abs(1 - (a + b))
                for a, b in zip(panel["cyan"], panel["gold"], strict=True)
            ),
            int(panel["slot"]),
        )
        for panel in panels
    )
    if scored[0][0] != 0 or scored[1][0] == 0:
        raise ValueError("scene does not contain a unique exact cover")
    return ANSWERS[scored[0][1]]


def margin(scene: dict) -> float:
    scores = sorted(_score(panel) for panel in scene["panels"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_score(panel) for panel in scene["panels"])
    return not (scores[0] == 0 and scores.count(0) == 1 and scores[1] - scores[0] >= 4)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("reverse_serialized_panel_records", twin)]
