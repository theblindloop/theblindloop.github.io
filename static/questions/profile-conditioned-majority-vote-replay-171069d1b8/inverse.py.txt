"""Independent image-only inverse for the majority-vote lamp replay."""

from __future__ import annotations

import numpy as np
from PIL import Image

ON = (246, 192, 35)
OFF = (128, 83, 181)
ADDRESS = (43, 111, 190)
CHECK = (32, 160, 122)
QUERY = (216, 52, 132)


def _near(rgb: np.ndarray, color: tuple[int, int, int], radius: int = 10) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray, minimum: int = 60):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                yy, xx = y + dy, x + dx
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    stack.append((yy, xx))
        if len(xs) >= minimum:
            found.append({
                "area": len(xs), "x0": min(xs), "x1": max(xs),
                "y0": min(ys), "y1": max(ys),
                "cx": float(sum(xs)) / len(xs), "cy": float(sum(ys)) / len(ys),
            })
    return found


def _split_vertical(items: list[dict]) -> tuple[list[dict], list[dict]]:
    ordered = sorted(items, key=lambda item: item["cy"])
    if len(ordered) < 2:
        raise ValueError("address layers are missing")
    gaps = [ordered[index + 1]["cy"] - ordered[index]["cy"] for index in range(len(ordered) - 1)]
    cut = int(np.argmax(gaps))
    if gaps[cut] < 5 * np.median([item["y1"] - item["y0"] + 1 for item in ordered]):
        raise ValueError("address layers do not separate")
    return ordered[:cut + 1], ordered[cut + 1:]


def _group_horizontal(items: list[dict]) -> list[list[dict]]:
    ordered = sorted(items, key=lambda item: item["cx"])
    if not ordered:
        return []
    typical_width = float(np.median([item["x1"] - item["x0"] + 1 for item in ordered]))
    groups = [[ordered[0]]]
    for item in ordered[1:]:
        if item["cx"] - groups[-1][-1]["cx"] > 4.0 * typical_width:
            groups.append([item])
        else:
            groups[-1].append(item)
    return groups


def _group_centers(groups: list[list[dict]]) -> list[float]:
    return [sum(item["cx"] for item in group) / len(group) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.ndim != 3 or rgb.shape[0] < 600 or rgb.shape[1] < 800:
        raise ValueError("raster is too small")

    address_marks = _components(_near(rgb, ADDRESS), 100)
    initial_marks, card_marks = _split_vertical(address_marks)
    initial_groups = _group_horizontal(initial_marks)
    card_groups = _group_horizontal(card_marks)
    if len(initial_groups) != 4 or [len(group) for group in initial_groups] != [1, 2, 3, 4]:
        raise ValueError("initial lamp identities are unresolved")
    if len(card_groups) != 5 or any(not 1 <= len(group) <= 4 for group in card_groups):
        raise ValueError("vote-card addresses are unresolved")

    check_groups = _group_horizontal(_components(_near(rgb, CHECK), 90))
    if len(check_groups) != 5:
        raise ValueError("verification strips are unresolved")
    card_counts = [len(group) for group in card_groups]
    check_counts = [len(group) for group in check_groups]
    if card_counts != check_counts:
        raise ValueError("address/check evidence disagrees")

    state_items = []
    for state, color in ((True, ON), (False, OFF)):
        for item in _components(_near(rgb, color), 1500):
            item = dict(item)
            item["state"] = state
            state_items.append(item)
    state_items.sort(key=lambda item: item["cx"])
    if len(state_items) != 4:
        raise ValueError("starting lamp states are unresolved")
    lamp_centers = _group_centers(initial_groups)
    typical_spacing = float(np.median(np.diff(lamp_centers)))
    if any(abs(item["cx"] - lamp_centers[index]) > typical_spacing / 5 for index, item in enumerate(state_items)):
        raise ValueError("lamp labels and states are misaligned")

    query_marks = _components(_near(rgb, QUERY), 500)
    if len(query_marks) != 1:
        raise ValueError("query ring is unresolved")
    query_x = query_marks[0]["cx"]
    distances = sorted((abs(query_x - x), index) for index, x in enumerate(lamp_centers))
    if distances[0][0] > typical_spacing / 5 or distances[1][0] - distances[0][0] < typical_spacing / 2:
        raise ValueError("query identity is ambiguous")
    query = distances[0][1]

    values = [bool(item["state"]) for item in state_items]
    for count in card_counts:
        target = count - 1
        values[target] = sum(values[index] for index in range(4) if index != target) >= 2
    return "ON" if values[query] else "OFF"
