"""Deterministic visual world for asynchronous majority-vote lamp updates."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

PAPERS = {"warm": (248, 249, 246), "ivory": (248, 249, 246)}
INK = (31, 38, 48)
ON = (246, 192, 35)
OFF = (128, 83, 181)
ADDRESS = (43, 111, 190)
CHECK = (32, 160, 122)
QUERY = (216, 52, 132)


def _font(size: int, bold: bool = False):
    face = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{face}", size)


def _replay(initial: list[bool], operations: list[int]) -> list[bool]:
    values = list(initial)
    for target in operations:
        other_on = sum(values[index] for index in range(4) if index != target)
        values[target] = other_on >= 2
    return values


def _change_count(initial: list[bool], operations: list[int]) -> int:
    values = list(initial)
    changes = 0
    for target in operations:
        replacement = sum(values[index] for index in range(4) if index != target) >= 2
        changes += replacement != values[target]
        values[target] = replacement
    return changes


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with an order-sensitive five-card replay."""
    seed = int(seed)
    desired = seed % 2 == 0
    query = seed % 4
    for attempt in range(20000):
        rng = random.Random(seed * 10009 + attempt * 137)
        initial = [bool(rng.randrange(2)) for _ in range(4)]
        if sum(initial) not in (1, 2, 3):
            continue
        operations = [rng.randrange(4) for _ in range(5)]
        result = _replay(initial, operations)[query]
        if result != desired or query not in operations:
            continue
        if len(set(operations)) < 3 or _change_count(initial, operations) < 2:
            continue
        if _replay(initial, list(reversed(operations)))[query] == result:
            continue
        return {
            "initial": initial,
            "operations": operations,
            "query": query,
            "paper": "warm" if rng.randrange(2) == 0 else "ivory",
            "width": 1000,
            "height": 700,
        }
    raise RuntimeError("could not sample a balanced majority-vote scene")


def analytic_gold(scene: dict) -> str:
    result = _replay(scene["initial"], scene["operations"])[int(scene["query"])]
    return "ON" if result else "OFF"


def margin(scene: dict) -> float:
    # Minimum designed edge separation between independently counted address marks.
    _ = (scene["initial"], scene["operations"], scene["query"])
    return 14.0


def is_quarantined(scene: dict) -> bool:
    try:
        return not (
            len(scene["initial"]) == 4
            and len(scene["operations"]) == 5
            and all(isinstance(value, bool) for value in scene["initial"])
            and all(0 <= int(value) < 4 for value in scene["operations"])
            and 0 <= int(scene["query"]) < 4
            and int(scene["width"]) >= 800
            and int(scene["height"]) >= 600
            and margin(scene) >= 10.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["paper"] = "ivory" if scene["paper"] == "warm" else "warm"
    return [("paper_name_alias", twin)]


def _dot_pattern(count: int, center_x: int, center_y: int) -> list[tuple[int, int]]:
    patterns = {
        1: [(0, 0)],
        2: [(-14, 0), (14, 0)],
        3: [(-18, 0), (0, 0), (18, 0)],
        4: [(-16, -11), (16, -11), (-16, 11), (16, 11)],
    }
    return [(center_x + dx, center_y + dy) for dx, dy in patterns[count]]


def _draw_address(draw: ImageDraw.ImageDraw, count: int, cx: int, cy: int) -> None:
    for x, y in _dot_pattern(count, cx, cy):
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=ADDRESS)


def _draw_check_strip(draw: ImageDraw.ImageDraw, count: int, cx: int, cy: int) -> None:
    gap = 16
    start = cx - gap * (count - 1) / 2
    for index in range(count):
        x = int(round(start + index * gap))
        draw.rounded_rectangle((x - 5, cy - 8, x + 5, cy + 8), radius=2, fill=CHECK)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), PAPERS[scene["paper"]])
    draw = ImageDraw.Draw(image)
    title = _font(28, True)
    heading = _font(17, True)
    small = _font(13, True)

    draw.text((38, 22), "MAJORITY VOTE LAMP REPLAY", font=title, fill=INK)
    draw.rounded_rectangle((30, 70, 970, 292), radius=18, fill=(255, 255, 255), outline=INK, width=3)
    draw.text((50, 87), "STARTING LAMPS  •  MAGENTA RING = QUERY", font=heading, fill=INK)

    lamp_xs = (145, 380, 615, 850)
    for identity, x in enumerate(lamp_xs):
        _draw_address(draw, identity + 1, x, 142)
        state = bool(scene["initial"][identity])
        if identity == int(scene["query"]):
            draw.ellipse((x - 43, 174, x + 43, 260), outline=QUERY, width=7)
        draw.ellipse((x - 30, 187, x + 30, 247), fill=ON if state else OFF, outline=INK, width=3)
        word = "ON" if state else "OFF"
        bbox = draw.textbbox((0, 0), word, font=small)
        draw.text((x - (bbox[2] - bbox[0]) / 2, 264), word, font=small, fill=INK)

    draw.rounded_rectangle((30, 312, 970, 674), radius=18, fill=(255, 255, 255), outline=INK, width=3)
    draw.text((50, 330), "VOTE CARDS  •  READ LEFT → RIGHT", font=heading, fill=INK)
    centers = (110, 305, 500, 695, 890)
    for number, (cx, target) in enumerate(zip(centers, scene["operations"]), start=1):
        draw.rounded_rectangle((cx - 82, 378, cx + 82, 645), radius=13, fill=(244, 247, 250), outline=(91, 99, 110), width=3)
        draw.text((cx - 66, 391), str(number), font=small, fill=(75, 82, 92))
        draw.text((cx - 34, 422), "VOTE", font=_font(15, True), fill=INK)
        _draw_address(draw, int(target) + 1, cx, 485)
        draw.line((cx - 44, 530, cx + 44, 530), fill=(159, 168, 178), width=2)
        draw.text((cx - 34, 548), "CHECK", font=_font(11, True), fill=(55, 117, 94))
        _draw_check_strip(draw, int(target) + 1, cx, 590)
    return image
