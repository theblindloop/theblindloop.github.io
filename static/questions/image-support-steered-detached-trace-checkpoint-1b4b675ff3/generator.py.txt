"""Renderer and analytic arm for a compact two-trace ordering question."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE = (30, 82, 140)
GREEN = (35, 169, 101)
RED = (220, 65, 61)
AMBER = (239, 151, 33)
VIOLET = (173, 72, 187)
BLACK = (23, 29, 38)


def _points(trace: dict) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in trace["points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(
            cumulative[-1]
            + math.hypot(second[0] - first[0], second[1] - first[1])
        )
    return cumulative, cumulative[-1]


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span <= 0.0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return (x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0))
    return points[-1]


def _marker_color(label: str) -> tuple[int, int, int]:
    if label == "amber":
        return AMBER
    if label == "violet":
        return VIOLET
    raise ValueError(f"unknown checkpoint label: {label}")


def _sample_trace(
    rng: random.Random,
    *,
    x0: float,
    y0: float,
    width: float,
    fraction: float,
    label: str,
    reverse: bool,
) -> dict:
    amplitude = rng.uniform(4.5, 8.0)
    phase = rng.uniform(-0.7, 0.7)
    points: list[list[float]] = []
    count = 5
    for index in range(count):
        t = index / (count - 1)
        x = x0 + width * t
        y = y0 + amplitude * math.sin(math.pi * (1.1 * t + phase))
        y += rng.uniform(-2.5, 2.5) * math.sin(math.pi * t)
        points.append([round(x, 4), round(y, 4)])
    if reverse:
        points.reverse()
    return {
        "points": points,
        "checkpoint_fraction": round(float(fraction), 4),
        "checkpoint": label,
        "trace_width": round(rng.uniform(8.5, 10.5), 3),
        "marker_radius": round(rng.uniform(7.0, 9.0), 3),
        "port_radius": round(rng.uniform(8.0, 10.0), 3),
    }


def sample_scene(seed: int) -> dict:
    """Sample two nearby but disconnected traces with balanced answer classes."""

    rng = random.Random(int(seed))
    # The two traces occupy a compact diagonal inset.  The seed controls only
    # visible geometry and label assignment; answer classes alternate across
    # the generator's odd stride while remaining independent of layout.
    # Keep the two routes in separated ablation rows (tile rows 2 and 4 on
    # the protected 8-by-8 probe) while their combined span stays focal.
    x0 = rng.uniform(145.0, 155.0)
    y0 = rng.uniform(158.0, 164.0)
    width = rng.uniform(52.0, 60.0)
    diagonal_x = rng.uniform(57.0, 64.0)
    diagonal_y = rng.uniform(126.0, 134.0)
    early_label = "amber" if int(seed) % 2 == 0 else "violet"
    late_label = "violet" if early_label == "amber" else "amber"
    early_fraction = rng.uniform(0.22, 0.28)
    late_fraction = rng.uniform(0.72, 0.78)

    first = _sample_trace(
        rng,
        x0=x0,
        y0=y0,
        width=width,
        fraction=early_fraction,
        label=early_label,
        reverse=bool(rng.randrange(2)),
    )
    second = _sample_trace(
        rng,
        x0=x0 + diagonal_x,
        y0=y0 + diagonal_y,
        width=width * rng.uniform(0.96, 1.04),
        fraction=late_fraction,
        label=late_label,
        reverse=bool(rng.randrange(2)),
    )
    traces = [first, second]
    if rng.randrange(2):
        traces.reverse()
    return {"traces": traces}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for trace in scene["traces"]:
        points = _points(trace)
        scaled = [(round(x * SCALE), round(y * SCALE)) for x, y in points]
        route_width = round(float(trace["trace_width"]) * SCALE)
        route_radius = max(1, route_width // 2)
        draw.line(scaled, fill=ROUTE, width=route_width, joint="curve")
        for x, y in scaled:
            draw.ellipse(
                (x - route_radius, y - route_radius, x + route_radius, y + route_radius),
                fill=ROUTE,
            )

        checkpoint = _point_at_fraction(points, float(trace["checkpoint_fraction"]))
        cx, cy = round(checkpoint[0] * SCALE), round(checkpoint[1] * SCALE)
        marker_radius = round(float(trace["marker_radius"]) * SCALE)
        outer = marker_radius + 4 * SCALE
        color = _marker_color(str(trace["checkpoint"]))
        # Draw the colored ring before a second centerline pass.  This keeps
        # the route a connected raster component while retaining a
        # color-segmentable ring around the checkpoint.
        draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), outline=BLACK, width=3 * SCALE)
        draw.ellipse(
            (cx - marker_radius, cy - marker_radius, cx + marker_radius, cy + marker_radius),
            outline=color,
            width=3 * SCALE,
        )
        draw.line(scaled, fill=ROUTE, width=route_width, joint="curve")
        for x, y in scaled:
            draw.ellipse(
                (x - route_radius, y - route_radius, x + route_radius, y + route_radius),
                fill=ROUTE,
            )

        port_radius = round(float(trace["port_radius"]) * SCALE)
        for center, color in ((scaled[0], GREEN), (scaled[-1], RED)):
            x, y = center
            outer = port_radius + 3 * SCALE
            draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=BLACK)
            draw.ellipse(
                (x - port_radius, y - port_radius, x + port_radius, y + port_radius),
                fill=color,
            )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    traces = scene["traces"]
    if len(traces) != 2:
        raise ValueError("exactly two traces are required")
    labels = {str(trace["checkpoint"]): float(trace["checkpoint_fraction"]) for trace in traces}
    if set(labels) != {"amber", "violet"}:
        raise ValueError("traces must contain one amber and one violet checkpoint")
    gap = abs(labels["amber"] - labels["violet"])
    if gap < 1e-9:
        raise ValueError("checkpoint order tie is not a valid scene")
    return "amber" if labels["amber"] < labels["violet"] else "violet"


def margin(scene: dict) -> float:
    traces = scene["traces"]
    labels = {str(trace["checkpoint"]): float(trace["checkpoint_fraction"]) for trace in traces}
    return abs(labels["amber"] - labels["violet"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.16


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every label-bearing trace field is rendered.  There are no hidden
    # aliases or nontrivial pixel-identical transforms to exercise.
    return []
