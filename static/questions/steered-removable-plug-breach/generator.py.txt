"""Deterministic compact topology world: counterfactual plug erasure."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


CANVAS_SIZE = 512
PALETTE = {
    "background": [248, 250, 252],
    "barrier": [22, 50, 78],
    "plug": [245, 158, 11],
    "decoration": [203, 213, 225],
}


def _nonoverlapping_decorations(rng: random.Random, cx: int, cy: int) -> list[dict]:
    items: list[dict] = []
    attempts = 0
    while len(items) < 6 and attempts < 500:
        attempts += 1
        x, y = rng.randint(45, 467), rng.randint(45, 467)
        radius = rng.choice([5, 6, 7])
        if abs(x - cx) < 82 and abs(y - cy) < 82:
            continue
        if any((x - item["center"][0]) ** 2 + (y - item["center"][1]) ** 2 < 30 ** 2 for item in items):
            continue
        items.append({"center": [x, y], "radius": radius})
    if len(items) != 6:
        raise RuntimeError("could not place disjoint decorations")
    return items


def sample_scene(seed: int) -> dict:
    """Return one answer-balanced scene with a translated, quarter-turned glyph."""

    seed = int(seed)
    rng = random.Random(seed)
    breach = seed % 2 == 0
    rotation = (seed // 2) % 4
    cx, cy = rng.randint(118, 394), rng.randint(118, 394)
    # The white polygon cuts a narrow neck from the top edge into a broad chamber.
    pocket = [
        [-14, -47], [14, -47], [14, -14], [32, -14],
        [32, 32], [-32, 32], [-32, -14], [-14, -14],
    ]
    latch = [-14, 4, 14, 12] if breach else [-14, -24, 14, -16]
    return {
        "canvas_size": CANVAS_SIZE,
        "center": [cx, cy],
        "rotation": rotation,
        "breach_after_erasure": breach,
        "outer_box_local": [-46, -46, 46, 46],
        "pocket_polygon_local": pocket,
        "plug_box_local": [-14, -48, 14, -29],
        "latch_box_local": latch,
        "decorations": _nonoverlapping_decorations(rng, cx, cy),
        "palette": copy.deepcopy(PALETTE),
        "clearance": 8.0,
    }


def _rotate(x: float, y: float, quarter_turns: int) -> tuple[float, float]:
    quarter_turns %= 4
    if quarter_turns == 0:
        return x, y
    if quarter_turns == 1:
        return -y, x
    if quarter_turns == 2:
        return -x, -y
    return y, -x


def _polygon_from_box(box: list[int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    return [[x0, y0], [x1, y0], [x1, y1], [x0, y1]]


def _global_polygon(scene: dict, points: list[list[int]], scale: int) -> list[tuple[int, int]]:
    cx, cy = scene["center"]
    rotation = int(scene["rotation"])
    output = []
    for x, y in points:
        rx, ry = _rotate(float(x), float(y), rotation)
        output.append((round((cx + rx) * scale), round((cy + ry) * scale)))
    return output


def render(scene: dict) -> Image.Image:
    """Render every decision-bearing field with 3x supersampling."""

    size = int(scene["canvas_size"])
    palette = scene["palette"]
    _ = bool(scene["breach_after_erasure"])  # rendered through the latch placement
    scale = 3
    image = Image.new("RGB", (size * scale, size * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    draw.polygon(
        _global_polygon(scene, _polygon_from_box(scene["outer_box_local"]), scale),
        fill=tuple(palette["barrier"]),
    )
    draw.polygon(
        _global_polygon(scene, scene["pocket_polygon_local"], scale),
        fill=tuple(palette["background"]),
    )
    draw.polygon(
        _global_polygon(scene, _polygon_from_box(scene["plug_box_local"]), scale),
        fill=tuple(palette["plug"]),
    )
    draw.polygon(
        _global_polygon(scene, _polygon_from_box(scene["latch_box_local"]), scale),
        fill=tuple(palette["barrier"]),
    )
    decoration = tuple(palette["decoration"])
    for item in scene["decorations"]:
        x, y = item["center"]
        radius = int(item["radius"])
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale,
             (x + radius) * scale, (y + radius) * scale),
            fill=decoration,
        )
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["breach_after_erasure"]) else "no"


def margin(scene: dict) -> float:
    return float(scene["clearance"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_draw_order", twin)]
