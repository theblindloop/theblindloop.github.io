"""Independent pixel-only inverse for the removable-plug topology predicate."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


_COLORS = np.asarray(
    [
        [248, 250, 252],  # background
        [22, 50, 78],     # navy barrier
        [245, 158, 11],   # amber removable plug
        [203, 213, 225],  # ignored decorations
    ],
    dtype=np.int32,
)
_BACKGROUND, _BARRIER, _PLUG, _DECORATION = range(4)
_FACTOR = 2


class OracleAbstention(ValueError):
    """The raster no longer exposes a safe, unambiguous topology instance."""


@dataclass(frozen=True)
class PixelMeasurement:
    decision: str
    runner_up_margin_px: float
    baseline_pocket_area: int
    plug_area: int


def _nearest_labels(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    if array.shape != (512, 512, 3):
        raise OracleAbstention(f"expected a 512x512 RGB raster, got {array.shape}")
    distance = ((array[:, :, None, :] - _COLORS[None, None, :, :]) ** 2).sum(axis=3)
    return distance.argmin(axis=2).astype(np.uint8)


def _block_any(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    blocks = mask.reshape(height // _FACTOR, _FACTOR, width // _FACTOR, _FACTOR)
    return blocks.mean(axis=(1, 3)) >= 0.25


def _components(mask: np.ndarray) -> tuple[np.ndarray, dict[int, int]]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    areas: dict[int, int] = {}
    component = 0
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if labels[y, x]:
            continue
        component += 1
        labels[y, x] = component
        queue = deque([(y, x)])
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (0 <= ny < height and 0 <= nx < width and mask[ny, nx]
                        and labels[ny, nx] == 0):
                    labels[ny, nx] = component
                    queue.append((ny, nx))
        areas[component] = area
    return labels, areas


def _border_ids(labels: np.ndarray) -> set[int]:
    values = set(int(value) for value in labels[0, :])
    values.update(int(value) for value in labels[-1, :])
    values.update(int(value) for value in labels[:, 0])
    values.update(int(value) for value in labels[:, -1])
    values.discard(0)
    return values


def measure(image: Image.Image) -> PixelMeasurement:
    """Recover the counterfactual result from pixels and report a runner-up margin."""

    labels = _nearest_labels(image)
    raw_plug = labels == _PLUG
    plug_area = int(raw_plug.sum())
    if not 430 <= plug_area <= 640:
        raise OracleAbstention(f"amber plug area is missing or unsafe: {plug_area}")
    plug_points = np.argwhere(raw_plug)
    ph = int(plug_points[:, 0].max() - plug_points[:, 0].min() + 1)
    pw = int(plug_points[:, 1].max() - plug_points[:, 1].min() + 1)
    if min(ph, pw) < 16 or max(ph, pw) > 32:
        raise OracleAbstention("amber plug is not the expected compact removable patch")

    navy = _block_any(labels == _BARRIER)
    amber = _block_any(raw_plug)
    navy_area = int(navy.sum())
    if not 900 <= navy_area <= 1450:
        raise OracleAbstention(f"navy barrier area is missing or unsafe: {navy_area}")
    foreground = navy | amber
    points = np.argwhere(foreground)
    if not len(points):
        raise OracleAbstention("compact topology glyph is absent")
    height = int(points[:, 0].max() - points[:, 0].min() + 1)
    width = int(points[:, 1].max() - points[:, 1].min() + 1)
    if not (42 <= height <= 50 and 42 <= width <= 50):
        raise OracleAbstention(f"topology glyph extent is corrupted: {(width, height)}")

    before_labels, before_areas = _components(~foreground)
    before_border = _border_ids(before_labels)
    bounded = [(key, area) for key, area in before_areas.items()
               if key not in before_border and area >= 300]
    if len(bounded) != 1:
        raise OracleAbstention(f"expected one enclosed white pocket, recovered {len(bounded)}")
    pocket_id, pocket_area = bounded[0]
    if not 620 <= pocket_area <= 900:
        raise OracleAbstention(f"enclosed pocket area is unsafe: {pocket_area}")
    pocket_points = np.argwhere(before_labels == pocket_id)
    reference_y, reference_x = map(int, pocket_points[len(pocket_points) // 2])

    # Counterfactual intervention: amber becomes white; navy remains a barrier.
    after_labels, _ = _components(~navy)
    after_border = _border_ids(after_labels)
    after_id = int(after_labels[reference_y, reference_x])
    if after_id == 0:
        raise OracleAbstention("pocket reference was swallowed by a corrupted barrier")
    opened = after_id in after_border

    # A conservative pixel margin: the smaller visible plug dimension is the
    # runner-up separation from a missing/degenerate patch decision.
    runner_up = float(min(ph, pw))
    if runner_up < 12.0:
        raise OracleAbstention("counterfactual decision margin is too small")
    return PixelMeasurement(
        decision="yes" if opened else "no",
        runner_up_margin_px=runner_up,
        baseline_pocket_area=int(pocket_area),
        plug_area=plug_area,
    )


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff amber erasure merges the bounded pocket with the border."""

    return measure(image).decision
