"""panel_collate_reveal: marked-panel reassembly reveal world.

Declared transform: reassemble the nine marked panels in their declared rank
order. A block letter A or B is partitioned into a 3x3 grid of nine rectangular
tiles. Each tile is drawn on its own square panel that is outlined in blue, and
each panel carries a small blue rank dot near its lower edge; the left-to-right
position of that dot declares the tile's destination slot in the final 3x3
grid (leftmost dot -> top-left cell, proceeding left-to-right, top-to-bottom).
The nine panels are then scattered to random canvas positions and the tile-to-
position assignment is shuffled. Read directly, the scattered fragments look
like undecipherable noise; only collating them by their declared rank order
reconstructs the flat letter, whose identity is the answer.

Latent scene z = {seed, glyph, slot_map, positions}. label = glyph.

The independent inverse arm receives only the final PNG, segments the blue
panel frames and the blue rank dots, reads each dot's rank, reassembles the
tiles into the 3x3 grid, and classifies the reconstructed letter against the
two block candidates A and B.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "panel_collate_reveal-0.1.0"

W = 420
H = 420

G_ROWS = 12
G_COLS = 12
TILES_PER_AXIS = 3          # 3x3 reassembly grid
CELLS_PER_TILE = G_ROWS // TILES_PER_AXIS   # 4
Q_TOL = 0.40

PANEL = 84                  # panel side length (px)
BORDER = 2                  # blue border width (px)
RANK_DOT_RADIUS = 4
PAD = 34
JIT = 10

BLACK = (0, 0, 0)
WHITE = (255, 255, 255)
FRAME_COLOR = (40, 90, 220)
GRAY = (205, 205, 205)

# 12 x 12 block-glyph templates.
TEMPLATE_A = [
    list("............"),
    list("....##......"),
    list("...####....."),
    list("..##..##...."),
    list("..##..##...."),
    list(".##....##..."),
    list(".##....##..."),
    list("##########.."),
    list("##......##.."),
    list("##......##.."),
    list("............"),
    list("............"),
]
TEMPLATE_B = [
    list("............"),
    list(".#########.."),
    list(".##.....##.."),
    list(".##.....##.."),
    list(".##.....##.."),
    list(".#########.."),
    list(".##.....##.."),
    list(".##.....##.."),
    list(".##.....##.."),
    list(".#########.."),
    list("............"),
    list("............"),
]
TEMPLATES = {"A": TEMPLATE_A, "B": TEMPLATE_B}


def _template_mask(glyph: str) -> np.ndarray:
    return np.array([[c == "#" for c in row] for row in TEMPLATES[glyph]], dtype=bool)


def _tile_mask(template: np.ndarray, slot: int) -> np.ndarray:
    rb = (slot // TILES_PER_AXIS) * CELLS_PER_TILE
    cb = (slot % TILES_PER_AXIS) * CELLS_PER_TILE
    return template[rb:rb + CELLS_PER_TILE, cb:cb + CELLS_PER_TILE]


def build_scene(seed: int, glyph: str) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    slot_map = list(range(9))
    rng.shuffle(slot_map)
    spacing = (W - 2 * PAD) / TILES_PER_AXIS
    positions = []
    for m in range(9):
        row = m // TILES_PER_AXIS
        col = m % TILES_PER_AXIS
        cx = PAD + spacing * (col + 0.5) + float(rng.uniform(-JIT, JIT))
        cy = PAD + spacing * (row + 0.5) + float(rng.uniform(-JIT, JIT))
        positions.append([int(round(cx - PANEL / 2.0)), int(round(cy - PANEL / 2.0))])
    return {
        "seed": int(seed),
        "glyph": glyph,
        "slot_map": list(slot_map),
        "positions": positions,
    }


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if int(rng.integers(0, 2)) else "B"
    return build_scene(seed, glyph)


def render(scene: dict) -> Image.Image:
    template = _template_mask(scene["glyph"])
    image = Image.new("RGB", (W, H), WHITE)
    draw = ImageDraw.Draw(image)
    for m in range(9):
        slot = scene["slot_map"][m]
        x, y = scene["positions"][m]
        draw.rectangle([x, y, x + PANEL - 1, y + PANEL - 1],
                       outline=FRAME_COLOR, width=BORDER)
        mask = _tile_mask(template, slot)
        cell = PANEL / CELLS_PER_TILE
        for a in range(CELLS_PER_TILE):
            for b in range(CELLS_PER_TILE):
                if mask[a, b]:
                    left = x + b * cell
                    top = y + a * cell
                    draw.rectangle(
                        [left, top, left + cell, top + cell], fill=BLACK)
        inner_left = x + BORDER
        inner_right = x + PANEL - BORDER
        inner_width = inner_right - inner_left
        frac = (slot + 0.5) / 9.0
        dot_cx = inner_left + frac * inner_width
        dot_cy = y + PANEL - BORDER - 6
        draw.ellipse([dot_cx - RANK_DOT_RADIUS, dot_cy - RANK_DOT_RADIUS,
                      dot_cx + RANK_DOT_RADIUS, dot_cy + RANK_DOT_RADIUS],
                     fill=FRAME_COLOR)
    return image


def assemble_grid(scene: dict) -> np.ndarray:
    """Analytic reassembled 12x12 letter (== the glyph template)."""
    return _template_mask(scene["glyph"])


def margin(scene: dict) -> float:
    chosen = assemble_grid(scene)
    other = _template_mask("B" if scene["glyph"] == "A" else "A")
    c_filled = int(chosen.sum())
    o_filled = int(other.sum())
    corr_chosen = float((chosen & chosen).sum()) / max(1, c_filled)
    corr_other = float((chosen & other).sum()) / max(1, o_filled)
    return round(corr_chosen - corr_other, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < Q_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field - the per-scene seed, the declared
    glyph, the shuffled tile-to-position mapping, and the panel positions -
    directly changes the raster (the shuffled mapping relocates the ink tiles
    and their rank dots, the positions shift every drawn panel), so no two
    distinct latent vectors reproduce the same image and no render-invariant
    transform exists. Declared empty."""
    return iter(())
