"""Color evenness comparison world.

Latent state z = 4 vertical bars (a pair red, a pair blue) standing on a shared
baseline. Each bar is an axis-aligned filled rectangle of a single solid color.
The decision is: which color's two bars are MORE EVEN in height (closer to each
other in length).

    red_even  = |h_red_a  - h_red_b|   (px)
    blue_even = |h_blue_a - h_blue_b|  (px)
    margin    = blue_even - red_even   (px)

with margin > 0 => the red bars are more even, margin < 0 => blue bars more even.
The margin is a continuous length difference (in pixels); the boundary event
(margin = 0) is the two colors being equally even, directly readable from the
four visible bar lengths.

The four bars are interleaved as RED, BLUE, RED, BLUE so same-color bars are
never adjacent (each colour forms two clean connected components) and a model
cannot shortcut on any single bar or on spatial adjacency. Positions are
randomized so no latent fixes the answer.
"""

from __future__ import annotations

import dataclasses
import math
import random
from dataclasses import dataclass

from PIL import Image, ImageDraw

RENDERER_VERSION = "sum_height_color_compare-0.1.0"

BASE_Y = 460.0
BAR_WIDTH_RANGE = (28.0, 46.0)
HEIGHT_RANGE = (40.0, 110.0)
GAP_RANGE = (18.0, 46.0)
COLORS = {"red": (200, 40, 40), "blue": (40, 60, 210)}


@dataclass(frozen=True)
class Bar:
    x_left: float
    y_top: float
    width: float
    height: float
    color: str


@dataclass(frozen=True)
class SumHeightScene:
    bars: tuple  # four Bar, alternating RED BLUE RED BLUE
    width: int = 512
    height: int = 512
    background: str = "white"

    @property
    def red_total(self) -> float:
        return sum(bar.height for bar in self.bars if bar.color == "red")

    @property
    def blue_total(self) -> float:
        return sum(bar.height for bar in self.bars if bar.color == "blue")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors_order = ("red", "blue", "red", "blue")
    for _ in range(10_000):
        heights = [rng.uniform(*HEIGHT_RANGE) for _ in range(4)]
        red_even = abs(heights[0] - heights[2])
        blue_even = abs(heights[1] - heights[3])
        if abs(blue_even - red_even) < QUARANTINE_MARGIN_PX:
            continue  # the within-colour gaps are a sub-resolvable near-tie
        x = rng.uniform(40.0, 90.0)
        bars = []
        for i, color in enumerate(colors_order):
            width = rng.uniform(*BAR_WIDTH_RANGE)
            bars.append(
                {
                    "x_left": x,
                    "y_top": BASE_Y - heights[i],
                    "width": width,
                    "height": heights[i],
                    "color": color,
                }
            )
            x += width + rng.uniform(*GAP_RANGE)
        return {
            "bars": bars,
            "width": 512,
            "height": 512,
            "background": "white",
        }
    raise RuntimeError(f"could not find off-quarantine scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), scene["background"])
    draw = ImageDraw.Draw(img)
    for bar in scene["bars"]:
        draw.rectangle(
            [
                bar["x_left"],
                bar["y_top"],
                bar["x_left"] + bar["width"],
                BASE_Y,
            ],
            fill=COLORS[bar["color"]],
        )
    return img


def _color_heights(scene: dict, color: str) -> tuple:
    return tuple(bar["height"] for bar in scene["bars"] if bar["color"] == color)


def _evenness(scene: dict, color: str) -> float:
    h1, h2 = _color_heights(scene, color)
    return abs(h1 - h2)


def analytic_gold(scene: dict) -> str:
    red_even = _evenness(scene, "red")
    blue_even = _evenness(scene, "blue")
    return "red" if red_even < blue_even else "blue"


def margin(scene: dict) -> float:
    return _evenness(scene, "blue") - _evenness(scene, "red")


QUARANTINE_MARGIN_PX = 8.0


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < QUARANTINE_MARGIN_PX


def latent_symmetries(scene: dict) -> list:
    return []
