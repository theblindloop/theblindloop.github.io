"""Independent pixel-only inverse arm for the colour-total height compare world.

Receives only the final PNG (never the scene, renderer, prompts or gold). It
segments the image by nearest palette colour, splits each colour into connected
components (each axis-aligned bar is one solid component), measures each
component's bounding-box height, sums the heights per colour, and returns the
colour with the greater total height.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageColor

ORACLE_VERSION = "sum_height_color_compare-oracle-0.1.0"

_PALETTE = {
    "red": ImageColor.getrgb("rgb(200,40,40)"),
    "blue": ImageColor.getrgb("rgb(40,60,210)"),
    "white": (255, 255, 255),
}
_ORDER = ("red", "blue", "white")


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in _PALETTE.values()],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _component_heights(mask: np.ndarray) -> list:
    h, w = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    heights = []
    for y0 in range(h):
        for x0 in range(w):
            if not mask[y0, x0] or visited[y0, x0]:
                continue
            stack = [(y0, x0)]
            visited[y0, x0] = True
            min_y = max_y = y0
            while stack:
                y, x = stack.pop()
                if y < min_y:
                    min_y = y
                if y > max_y:
                    max_y = y
                for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
            heights.append(float(max_y - min_y + 1))
    return sorted(heights)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    red_heights = _component_heights(labels == 0)
    blue_heights = _component_heights(labels == 1)
    red_even = abs(red_heights[1] - red_heights[0])
    blue_even = abs(blue_heights[1] - blue_heights[0])
    return "red" if red_even < blue_even else "blue"
