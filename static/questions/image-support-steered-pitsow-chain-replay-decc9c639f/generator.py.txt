"""Renderer and analytic gold for the `pitsow_chain_replay` world.

One small nine-pit block is drawn somewhere on the canvas. Each pit carries a
distinct solid floor colour and holds zero to three black pips. Exactly one pit
is ringed in white. The scene is replayed by relay sowing in reading order: the
ringed pit is emptied into the pits that follow it, one pip per pit, wrapping
round; if the last pip falls into a pit that already held pips, that whole pit
is lifted and sowing continues from there. The gold answer is the floor colour
of the pit where the relay finally stops.

All drawing uses flat, unantialiased integer rectangle fills, so re-rendering a
stored scene reproduces the stored raster byte for byte.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pitsow_chain_replay-renderer-0.1.0"

WIDTH = 960
HEIGHT = 720

BACKGROUND = (250, 250, 250)
DECOR = (214, 214, 214)
FRAME = (12, 12, 12)
BED = (232, 232, 232)
PIP = (30, 30, 30)
RING = (255, 255, 255)

# Block geometry (all integers; the oracle re-derives the grid from the pixels).
PIT = 44
CELL_GAP = 6
PAD = 8
FRAME_W = 4
GRID = 3 * PIT + 2 * CELL_GAP
BLOCK = GRID + 2 * PAD + 2 * FRAME_W

PIP_SIZE = 7
RING_INSET = 2
RING_W = 3
# Interior of a pit that is free of the ring bars.
FREE0 = RING_INSET + RING_W
FREE_SPAN = PIT - 2 * FREE0

COLOUR_NAMES = (
    "red",
    "green",
    "blue",
    "yellow",
    "orange",
    "purple",
    "teal",
    "pink",
    "brown",
)

COLOUR_RGB = {
    "red": (214, 38, 38),
    "green": (34, 150, 58),
    "blue": (36, 78, 200),
    "yellow": (238, 200, 20),
    "orange": (240, 124, 20),
    "purple": (128, 50, 178),
    "teal": (20, 158, 158),
    "pink": (240, 130, 190),
    "brown": (140, 86, 44),
}

MAX_DROPS = 200
QUARANTINE_MARGIN = 1.5


# --------------------------------------------------------------------------
# replay
# --------------------------------------------------------------------------


def _replay(pips: list, ring: int) -> tuple:
    """Relay sowing. Returns (final_pit, rounds, drops) or (-1, rounds, drops)."""
    board = list(pips)
    here = ring
    rounds = 0
    drops = 0
    while True:
        hand = board[here]
        board[here] = 0
        rounds += 1
        if hand <= 0:
            return -1, rounds, drops
        while hand > 0:
            here = (here + 1) % 9
            board[here] += 1
            hand -= 1
            drops += 1
            if drops > MAX_DROPS:
                return -1, rounds, drops
        if board[here] == 1:
            return here, rounds, drops


def _pip_gap_geometry(pip_gap: int) -> tuple:
    """Return (row_width, side_clearance) for a pip gap."""
    row = 3 * PIP_SIZE + 2 * pip_gap
    return row, (FREE_SPAN - row) // 2


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------


def _draft(rng: random.Random, tight: bool) -> dict:
    pip_gap = 5 if tight else rng.choice([3, 4])
    order = list(COLOUR_NAMES)
    rng.shuffle(order)
    pips = [rng.choice([0, 1, 1, 2, 2, 3]) for _ in range(9)]
    ring = rng.randrange(9)
    if pips[ring] == 0:
        pips[ring] = rng.choice([1, 2, 3])
    ox = rng.randrange(16, WIDTH - BLOCK - 16)
    oy = rng.randrange(16, HEIGHT - BLOCK - 16)
    decor = []
    forbid = (ox - 10, oy - 10, ox + BLOCK + 10, oy + BLOCK + 10)
    for _ in range(60):
        if len(decor) >= 9:
            break
        w = rng.randrange(14, 70)
        h = rng.randrange(4, 40)
        x = rng.randrange(6, WIDTH - w - 6)
        y = rng.randrange(6, HEIGHT - h - 6)
        if x + w >= forbid[0] and x <= forbid[2] and y + h >= forbid[1] and y <= forbid[3]:
            continue
        decor.append([x, y, w, h])
    return {
        "origin": [ox, oy],
        "pit_colours": order,
        "pips": pips,
        "ring": ring,
        "pip_gap": pip_gap,
        "decor": decor,
    }


def _acceptable(scene: dict) -> bool:
    pips = scene["pips"]
    ring = scene["ring"]
    total = sum(pips)
    if not 8 <= total <= 16:
        return False
    if pips[ring] < 1:
        return False
    final, rounds, _ = _replay(pips, ring)
    if final < 0:
        return False
    if rounds < 2:
        # a one-round answer would not require a relay
        return False
    if final == ring:
        return False
    if final == (ring + 1) % 9:
        # kills the "the pit just after the ring" shortcut
        return False
    if len(scene["decor"]) < 4:
        return False
    return True


def sample_scene(seed: int) -> dict:
    """Deterministically sample one accepted scene for `seed`."""
    tight = (seed % 8) == 3
    for attempt in range(4000):
        rng = random.Random((seed * 7919 + attempt * 104729) & 0x7FFFFFFF)
        scene = _draft(rng, tight)
        if _acceptable(scene):
            return scene
    raise RuntimeError(f"no acceptable scene for seed {seed}")


# --------------------------------------------------------------------------
# gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------


def analytic_gold(scene: dict) -> str:
    final, _, _ = _replay(list(scene["pips"]), int(scene["ring"]))
    if final < 0:
        raise ValueError("scene does not resolve to a stopping pit")
    return scene["pit_colours"][final]


def margin(scene: dict) -> float:
    """Smallest decisive clearance in pixels inside the block."""
    _, side = _pip_gap_geometry(int(scene["pip_gap"]))
    vertical = (FREE_SPAN - PIP_SIZE) // 2
    return float(min(int(scene["pip_gap"]), side, vertical, RING_W))


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_MARGIN:
        return True
    final, _, _ = _replay(list(scene["pips"]), int(scene["ring"]))
    return final < 0


def latent_symmetries(scene: dict) -> list:
    """No latent aliases: every label input is drawn as flat colour."""
    return []


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def _pit_origin(ox: int, oy: int, index: int) -> tuple:
    row, col = divmod(index, 3)
    x = ox + FRAME_W + PAD + col * (PIT + CELL_GAP)
    y = oy + FRAME_W + PAD + row * (PIT + CELL_GAP)
    return x, y


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, w, h in scene["decor"]:
        draw.rectangle([x, y, x + w - 1, y + h - 1], fill=DECOR)

    ox, oy = scene["origin"]
    draw.rectangle([ox, oy, ox + BLOCK - 1, oy + BLOCK - 1], fill=FRAME)
    draw.rectangle(
        [ox + FRAME_W, oy + FRAME_W, ox + BLOCK - 1 - FRAME_W, oy + BLOCK - 1 - FRAME_W],
        fill=BED,
    )

    pip_gap = int(scene["pip_gap"])
    row_w, side = _pip_gap_geometry(pip_gap)
    vertical = (FREE_SPAN - PIP_SIZE) // 2

    for index in range(9):
        px, py = _pit_origin(ox, oy, index)
        draw.rectangle([px, py, px + PIT - 1, py + PIT - 1], fill=COLOUR_RGB[scene["pit_colours"][index]])
        if index == int(scene["ring"]):
            a = px + RING_INSET
            b = py + RING_INSET
            c = px + PIT - 1 - RING_INSET
            d = py + PIT - 1 - RING_INSET
            draw.rectangle([a, b, c, b + RING_W - 1], fill=RING)
            draw.rectangle([a, d - RING_W + 1, c, d], fill=RING)
            draw.rectangle([a, b, a + RING_W - 1, d], fill=RING)
            draw.rectangle([c - RING_W + 1, b, c, d], fill=RING)
        count = int(scene["pips"][index])
        for k in range(count):
            qx = px + FREE0 + side + k * (PIP_SIZE + pip_gap)
            qy = py + FREE0 + vertical
            draw.rectangle([qx, qy, qx + PIP_SIZE - 1, qy + PIP_SIZE - 1], fill=PIP)

    return image
