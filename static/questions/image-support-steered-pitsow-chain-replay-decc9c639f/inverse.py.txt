"""Pixel-only oracle for `pitsow_chain_replay`.

Receives a rendered RGB image and nothing else. It segments the nine coloured
pits from saturation alone, recovers each pit's floor colour, its pip count and
which pit carries the white ring, replays the declared relay sowing, and reports
the floor colour of the stopping pit.

It imports no renderer, prompt, generator, verifier or analytic-gold code and
never sees latent scene data.
"""

from __future__ import annotations

from PIL import Image

ORACLE_VERSION = "pitsow_chain_replay-oracle-0.1.0"

_NAMES = (
    "red",
    "green",
    "blue",
    "yellow",
    "orange",
    "purple",
    "teal",
    "pink",
    "brown",
)

_REFERENCE = {
    "red": (214, 38, 38),
    "green": (34, 150, 58),
    "blue": (36, 78, 200),
    "yellow": (238, 200, 20),
    "orange": (240, 124, 20),
    "purple": (128, 50, 178),
    "teal": (20, 158, 158),
    "pink": (240, 130, 190),
    "brown": (140, 86, 44),
}

_SATURATION = 40
_MAX_DROPS = 200
ABSTAIN = "abstain"


def _runs(flags: list) -> list:
    out = []
    start = None
    for i, on in enumerate(flags):
        if on and start is None:
            start = i
        elif not on and start is not None:
            out.append((start, i - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _classify(rgb: tuple) -> str:
    best = None
    best_d = None
    for name, ref in _REFERENCE.items():
        d = sum((a - b) ** 2 for a, b in zip(rgb, ref))
        if best_d is None or d < best_d:
            best_d = d
            best = name
    return best


def _replay(pips: list, ring: int):
    board = list(pips)
    here = ring
    drops = 0
    while True:
        hand = board[here]
        board[here] = 0
        if hand <= 0:
            return -1
        while hand > 0:
            here = (here + 1) % 9
            board[here] += 1
            hand -= 1
            drops += 1
            if drops > _MAX_DROPS:
                return -1
        if board[here] == 1:
            return here


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    w, h = rgb.size
    px = rgb.load()

    colourful = [[False] * w for _ in range(h)]
    col_hits = [0] * w
    row_hits = [0] * h
    for y in range(h):
        row = colourful[y]
        for x in range(w):
            r, g, b = px[x, y]
            if max(r, g, b) - min(r, g, b) > _SATURATION:
                row[x] = True
                col_hits[x] += 1
                row_hits[y] += 1
    if not any(col_hits):
        return ABSTAIN

    x_runs = [r for r in _runs([c > 0 for c in col_hits]) if r[1] - r[0] >= 8]
    y_runs = [r for r in _runs([c > 0 for c in row_hits]) if r[1] - r[0] >= 8]
    if len(x_runs) != 3 or len(y_runs) != 3:
        return ABSTAIN

    colours = []
    pip_counts = []
    ring_index = -1
    for ri, (y0, y1) in enumerate(y_runs):
        for ci, (x0, x1) in enumerate(x_runs):
            tally = {}
            dark = 0
            white = 0
            for y in range(y0, y1 + 1):
                for x in range(x0, x1 + 1):
                    r, g, b = px[x, y]
                    if max(r, g, b) - min(r, g, b) > _SATURATION:
                        tally[(r, g, b)] = tally.get((r, g, b), 0) + 1
                    elif r > 245 and g > 245 and b > 245:
                        white += 1
                    elif r < 70 and g < 70 and b < 70:
                        dark += 1
            if not tally:
                return ABSTAIN
            dominant = max(tally.items(), key=lambda kv: (kv[1], kv[0]))[0]
            colours.append(_classify(dominant))
            side = int(round(_pip_side(dark)))
            if side < 0:
                return ABSTAIN
            pip_counts.append(side)
            if white > 120:
                if ring_index >= 0:
                    return ABSTAIN
                ring_index = ri * 3 + ci

    if ring_index < 0:
        return ABSTAIN
    if len(set(colours)) != 9:
        return ABSTAIN
    if pip_counts[ring_index] < 1:
        return ABSTAIN

    final = _replay(pip_counts, ring_index)
    if final < 0:
        return ABSTAIN
    return colours[final]


def _pip_side(dark_pixels: int) -> float:
    """Pips are equal solid squares; recover how many were drawn."""
    if dark_pixels == 0:
        return 0.0
    for count in (1, 2, 3):
        for size in range(5, 12):
            if dark_pixels == count * size * size:
                return float(count)
    # fall back on the modal pip area of 7x7
    approx = dark_pixels / 49.0
    nearest = round(approx)
    if abs(approx - nearest) < 0.12 and 0 <= nearest <= 3:
        return float(nearest)
    return -1.0
