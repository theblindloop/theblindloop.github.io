"""Prior-conflict clock-face world: count visible hour marks.

The latent scene holds a clock face with ``n`` evenly spaced hour marks. A
familiar canonical prior asserts that a clock has TWELVE hour marks; the
counterfactual renders a non-canonical count and the decision must be read
from the visible marks alone, never from the prior. The pixel oracle counts
the actual mark blobs.

Latent alias mode is ``not_applicable``: every latent field is directly
reflected in visible marks or geometry (face size, rotation, count), so the
rendering is injective in the latent scene and there is no render-invariant
alias. The canonical-prior name "clock" is a matched-control/counterfactual
manipulation of the *question*, not of any latent field.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
CX = 256.0
CY = 256.0
FACE_RADIUS = 200.0
RIM_RADIUS = 216.0
RIM_WIDTH = 18
# Marks are drawn inward from the rim onto the face.
TICK_INNER_OFFSET = 46.0
TICK_OUTER_OFFSET = 12.0
TICK_WIDTH = 13

# Supported off-quarantine counts. The canonical prior is 12; 8 and 10 are the
# counterfactuals that contradict it. 12, 10, 8 all remain far enough apart at
# face radius ~200px that each mark is an isolated blob (>= 30deg gaps).
SUPPORTED_N = (12, 10, 8)

# A count is quarantined when the marks are so crowded that adjacent blobs can
# touch and be miscounted at pixel scale. At the smallest supported gap
# (360/12 = 30deg) with marks drawn inward, blobs are separated by ~80px, far
# above this threshold, so supported counts are never ambiguous.
QUAR_MIN_GAP_DEG = 26.0

BACKGROUND = (255, 255, 255)
FACE_COLOR = (214, 228, 240)   # pale blue face
RIM_COLOR = (122, 122, 134)    # gray bezel
TICK_COLOR = (12, 18, 44)      # dark navy marks


def sample_scene(seed: int) -> dict:
    """Deterministic scene rgba from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # start_angle and face_radius vary independently of n so that neither
    # position nor object size predicts the count (cheap-cue counterbalance).
    start_angle = rng.uniform(0.0, 360.0)
    face_radius = FACE_RADIUS + rng.uniform(-5.0, 7.0)
    tick_in = face_radius - TICK_INNER_OFFSET
    tick_out = face_radius - TICK_OUTER_OFFSET
    return {
        "n": n,
        "start_angle": round(start_angle, 6),
        "face_radius": round(face_radius, 3),
        "tick_in": round(tick_in, 3),
        "tick_out": round(tick_out, 3),
        "canvas": CANVAS,
    }


def _pol(angle_deg: float, radius: float) -> tuple[float, float]:
    import math

    rad = math.radians(angle_deg)
    return (CX + radius * math.cos(rad), CY + radius * math.sin(rad))


def render(scene: dict) -> Image.Image:
    """Render the clock face to an RGB image."""
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), BACKGROUND)
    draw = ImageDraw.Draw(img)
    face_r = float(scene["face_radius"])
    rim_r = face_r + (RIM_RADIUS - FACE_RADIUS)
    # Bezel ring, then the face on top, then the marks.
    draw.ellipse(
        [CX - rim_r, CY - rim_r, CX + rim_r, CY + rim_r], fill=RIM_COLOR
    )
    draw.ellipse([CX - face_r, CY - face_r, CX + face_r, CY + face_r], fill=FACE_COLOR)
    n = int(scene["n"])
    tick_in = float(scene["tick_in"])
    tick_out = float(scene["tick_out"])
    step = 360.0 / n
    for k in range(n):
        angle = float(scene["start_angle"]) + k * step
        pin = _pol(angle, tick_in)
        pout = _pol(angle, tick_out)
        draw.line([pin, pout], fill=TICK_COLOR, width=TICK_WIDTH)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible count of hour marks, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = smallest angular gap between adjacent marks."""
    return 360.0 / float(scene["n"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine crowded counts that are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < QUAR_MIN_GAP_DEG


def latent_symmetries(scene: dict):
    """not_applicable: empty, the latent rendering is injective."""
    return []
