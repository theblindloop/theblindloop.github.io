"""Independent PNG-only inverse for the every-link terminal task."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "red": (224, 58, 62),
    "blue": (55, 116, 224),
    "orange": (238, 143, 47),
    "green": (36, 166, 91),
    "node": (207, 219, 231),
    "ink": (25, 28, 33),
}


def _components(mask: np.ndarray, minimum: int = 70) -> list[np.ndarray]:
    """Return large 8-connected components as arrays of (y, x) pixels."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (0 <= ny < height and 0 <= nx < width and mask[ny, nx]
                            and not seen[ny, nx]):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centers(arr: np.ndarray, color: tuple[int, int, int]) -> list[tuple[float, float]]:
    target = np.asarray(color, dtype=np.int16)
    distance = np.linalg.norm(arr.astype(np.int16) - target, axis=2)
    centers = []
    for component in _components(distance < 24):
        span_y, span_x = np.ptp(component, axis=0) + 1
        if 12 <= span_x <= 30 and 12 <= span_y <= 30:
            centers.append((float(component[:, 1].mean()), float(component[:, 0].mean())))
    return centers


def _point_segment_distance(
    point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> tuple[float, float]:
    px, py = point
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    denominator = vx * vx + vy * vy
    if denominator == 0:
        return math.inf, 0.0
    fraction = ((px - ax) * vx + (py - ay) * vy) / denominator
    qx, qy = ax + fraction * vx, ay + fraction * vy
    return math.hypot(px - qx, py - qy), fraction


def _stroke_present(
    dark: np.ndarray,
    a: tuple[float, float],
    b: tuple[float, float],
    all_nodes: list[tuple[float, float]],
) -> bool:
    distance = math.dist(a, b)
    if not 34.0 <= distance <= 132.0:
        return False
    for point in all_nodes:
        if point == a or point == b:
            continue
        clearance, fraction = _point_segment_distance(point, a, b)
        if 0.12 < fraction < 0.88 and clearance < 19.0:
            return False
    trim = 17.0 / distance
    samples = np.linspace(trim, 1.0 - trim, max(20, int(distance)))
    hits = 0
    height, width = dark.shape
    for fraction in samples:
        x = int(round(a[0] + fraction * (b[0] - a[0])))
        y = int(round(a[1] + fraction * (b[1] - a[1])))
        y0, y1 = max(0, y - 2), min(height, y + 3)
        x0, x1 = max(0, x - 2), min(width, x + 3)
        hits += bool(dark[y0:y1, x0:x1].any())
    return hits / len(samples) >= 0.84


def _connected_all(adjacency: list[set[int]]) -> bool:
    if not adjacency:
        return False
    seen = {0}
    queue = deque([0])
    while queue:
        current = queue.popleft()
        for neighbor in adjacency[current] - seen:
            seen.add(neighbor)
            queue.append(neighbor)
    return len(seen) == len(adjacency)


def _prescribed_euler(adjacency: list[set[int]], start: int, terminal: int) -> bool:
    if not _connected_all(adjacency):
        return False
    odd = {index for index, neighbors in enumerate(adjacency) if len(neighbors) % 2}
    return odd == {start, terminal}


def decision_from_image(image: Image.Image) -> str:
    """Recover all three visible graphs and return the Euler-trail terminal color."""
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 600:
        raise ValueError("unexpected raster dimensions")
    starts = _centers(arr, _PALETTE["green"])
    terminals = [(color, center) for color in ("red", "blue", "orange")
                 for center in _centers(arr, _PALETTE[color])]
    ordinary = _centers(arr, _PALETTE["node"])
    if len(starts) != 3 or len(terminals) != 3 or len(ordinary) != 21:
        raise ValueError(
            f"expected 3 starts, 3 terminals, and 21 ordinary nodes; found "
            f"{len(starts)}, {len(terminals)}, {len(ordinary)}"
        )

    starts.sort(key=lambda point: point[0])
    terminals.sort(key=lambda item: item[1][0])
    panel_centers = [(starts[index][0] + terminals[index][1][0]) / 2.0
                     for index in range(3)]
    groups: list[list[tuple[float, float]]] = [[], [], []]
    for point in ordinary:
        slot = min(range(3), key=lambda index: abs(point[0] - panel_centers[index]))
        groups[slot].append(point)
    if [len(group) for group in groups] != [7, 7, 7]:
        raise ValueError(f"ordinary-node panel split is ambiguous: {[len(group) for group in groups]}")

    ink = np.asarray(_PALETTE["ink"], dtype=np.int16)
    dark = np.linalg.norm(arr.astype(np.int16) - ink, axis=2) < 58
    winners: list[str] = []
    for slot in range(3):
        nodes = [starts[slot], terminals[slot][1], *groups[slot]]
        adjacency = [set() for _ in nodes]
        for left in range(len(nodes)):
            for right in range(left + 1, len(nodes)):
                if _stroke_present(dark, nodes[left], nodes[right], nodes):
                    adjacency[left].add(right)
                    adjacency[right].add(left)
        if sum(len(neighbors) for neighbors in adjacency) != 20:
            raise ValueError(f"panel {slot} does not resolve to exactly ten links")
        if len(adjacency[0]) != 3 or len(adjacency[1]) != 3:
            raise ValueError(f"panel {slot} endpoint degrees are not both three")
        if _prescribed_euler(adjacency, 0, 1):
            winners.append(terminals[slot][0])
    if len(winners) != 1:
        raise ValueError(f"expected one prescribed Euler-trail raster graph, found {winners}")
    return winners[0]
