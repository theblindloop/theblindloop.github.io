"""Deterministic renderer and analytic arm for an every-link network walk."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 840
HEIGHT = 360
COLORS = ("red", "blue", "orange")
RGB = {
    "red": (224, 58, 62),
    "blue": (55, 116, 224),
    "orange": (238, 143, 47),
    "green": (36, 166, 91),
    "node": (207, 219, 231),
    "ink": (25, 28, 33),
    "border": (190, 198, 207),
    "background": (255, 255, 255),
}


def _adjacency(nodes: list[dict], edges: list[list[str]]) -> dict[str, set[str]]:
    adjacency = {str(node["id"]): set() for node in nodes}
    for left, right in edges:
        if left not in adjacency or right not in adjacency or left == right:
            raise ValueError("edge must join two declared distinct nodes")
        adjacency[left].add(right)
        adjacency[right].add(left)
    return adjacency


def _connected(adjacency: dict[str, set[str]]) -> bool:
    if not adjacency:
        return False
    start = next(iter(adjacency))
    seen = {start}
    queue = deque([start])
    while queue:
        current = queue.popleft()
        for neighbor in adjacency[current] - seen:
            seen.add(neighbor)
            queue.append(neighbor)
    return len(seen) == len(adjacency)


def _euler_defect(nodes: list[dict], edges: list[list[str]]) -> int:
    """Parity violations for a walk prescribed to start and finish at the markers."""
    adjacency = _adjacency(nodes, edges)
    if not _connected(adjacency):
        return 100
    starts = [str(node["id"]) for node in nodes if node["role"] == "start"]
    terminals = [str(node["id"]) for node in nodes if node["role"] == "terminal"]
    if len(starts) != 1 or len(terminals) != 1:
        return 100
    prescribed_odd = {starts[0], terminals[0]}
    return sum((len(neighbors) % 2 == 1) != (node_id in prescribed_odd)
               for node_id, neighbors in adjacency.items())


def _base_panel(kind: str, color: str, slot: int, mirror_x: bool, mirror_y: bool) -> dict:
    # The common graph is three internally disjoint s-to-t paths. Each losing
    # graph swaps one 50-pixel link for a perpendicular 50-pixel link. Thus all
    # panels have nine nodes, ten links, degree-three endpoints, and equal ink.
    coords = {
        "s": (30, 180), "t": (230, 180),
        "a": (80, 130), "b": (130, 130),
        "c": (80, 180), "d": (130, 180), "e": (180, 180),
        "f": (80, 230), "g": (180, 230),
    }
    edges = [
        ["s", "a"], ["a", "b"], ["b", "t"],
        ["s", "c"], ["c", "d"], ["d", "e"], ["e", "t"],
        ["s", "f"], ["f", "g"], ["g", "t"],
    ]
    if kind == "upper_switch":
        edges.remove(["d", "e"])
        edges.append(["d", "b"])
    elif kind == "lower_switch":
        edges.remove(["c", "d"])
        edges.append(["c", "a"])
    elif kind != "euler":
        raise ValueError(f"unknown panel kind {kind!r}")

    x_offset = 10 + slot * 275
    nodes = []
    for node_id, (x, y) in coords.items():
        if mirror_x:
            x = 260 - x
        if mirror_y:
            y = 360 - y
        role = "start" if node_id == "s" else "terminal" if node_id == "t" else "node"
        fill = "green" if role == "start" else color if role == "terminal" else "node"
        nodes.append({"id": node_id, "x": x_offset + x, "y": y, "role": role, "fill": fill})
    return {"slot": slot, "kind": kind, "terminal_color": color, "nodes": nodes, "edges": edges}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xE013A11)
    answer_color = COLORS[int(seed) % 3]
    target_slot = (int(seed) // 3) % 3
    other_colors = [color for color in COLORS if color != answer_color]
    rng.shuffle(other_colors)
    slot_colors = []
    for slot in range(3):
        slot_colors.append(answer_color if slot == target_slot else other_colors.pop())

    losing_kinds = iter(("upper_switch", "lower_switch"))
    panels = []
    for slot in range(3):
        kind = "euler" if slot == target_slot else next(losing_kinds)
        panels.append(_base_panel(kind, slot_colors[slot], slot,
                                  bool(rng.getrandbits(1)), bool(rng.getrandbits(1))))
    return {"width": WIDTH, "height": HEIGHT, "panels": panels}


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), RGB["background"])
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        slot = int(panel["slot"])
        x0 = (10 + slot * 275) * scale
        draw.rounded_rectangle(
            (x0 + 2 * scale, 16 * scale, x0 + 264 * scale, 344 * scale),
            radius=12 * scale, outline=RGB["border"], width=2 * scale,
        )
        lookup = {node["id"]: node for node in panel["nodes"]}
        for left, right in panel["edges"]:
            a, b = lookup[left], lookup[right]
            draw.line((a["x"] * scale, a["y"] * scale,
                       b["x"] * scale, b["y"] * scale),
                      fill=RGB["ink"], width=7 * scale)
        for node in panel["nodes"]:
            radius = 13
            box = ((node["x"] - radius) * scale, (node["y"] - radius) * scale,
                   (node["x"] + radius) * scale, (node["y"] + radius) * scale)
            draw.ellipse(box, fill=RGB[node["fill"]], outline=RGB["ink"], width=3 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    winners = [panel["terminal_color"] for panel in scene["panels"]
               if _euler_defect(panel["nodes"], panel["edges"]) == 0]
    if len(winners) != 1 or winners[0] not in COLORS:
        raise ValueError(f"expected one prescribed Euler-trail panel, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    defects = sorted(_euler_defect(panel["nodes"], panel["edges"])
                     for panel in scene["panels"])
    if len(defects) != 3:
        return 0.0
    return float(defects[1] - defects[0])


def is_quarantined(scene: dict) -> bool:
    try:
        panels = scene["panels"]
        if len(panels) != 3 or margin(scene) < 2.0:
            return True
        lengths = []
        for panel in panels:
            lookup = {node["id"]: node for node in panel["nodes"]}
            if len(lookup) != 9 or len(panel["edges"]) != 10:
                return True
            if len(_adjacency(panel["nodes"], panel["edges"])["s"]) != 3:
                return True
            if len(_adjacency(panel["nodes"], panel["edges"])["t"]) != 3:
                return True
            lengths.append(sum(math.dist((lookup[a]["x"], lookup[a]["y"]),
                                         (lookup[b]["x"], lookup[b]["y"]))
                               for a, b in panel["edges"]))
        return max(lengths) - min(lengths) > 1e-6
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"].reverse()
    return [("reverse_edge_storage", twin)]
