"""Pixel-only inverse arm for excursion_area_balance.

Receives one RGB raster and nothing else.  It segments the blue reference line
and the dark trace by colour, recovers the reference row, checks that the trace
is a single-valued curve that runs uncut from one end of the sheet to the other,
integrates the signed offset of the trace column by column, and compares the
trapped area above the reference line with the trapped area below it.  A trace
that is cut, truncated, doubled or unreadable yields an abstention instead of a
guess.
"""

from __future__ import annotations

import numpy as np

ORACLE_VERSION = "excursion_area_balance-oracle-0.1.0"
ABSTAIN = "abstain"

TRACE_MAX = 60
BASE_MIN_BLUE = 110
BASE_MAX_RED = 100
BASE_MAX_GREEN = 130
STROKE_HALF = 1
MIN_BASELINE_FRACTION = 0.35
MIN_SPAN_FRACTION = 0.85
MAX_COLUMN_RUN = 9
MIN_TOTAL_AREA = 600.0
ABSTAIN_GAP = 400.0


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    height, width = array.shape[0], array.shape[1]
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    trace = array.max(axis=2) < TRACE_MAX
    reference = (blue > BASE_MIN_BLUE) & (red < BASE_MAX_RED) & (green < BASE_MAX_GREEN)

    row_counts = reference.sum(axis=1)
    baseline = int(np.argmax(row_counts))
    if int(row_counts[baseline]) < MIN_BASELINE_FRACTION * width:
        return ABSTAIN

    columns = np.flatnonzero(trace.any(axis=0))
    if columns.size == 0:
        return ABSTAIN
    first = int(columns[0])
    last = int(columns[-1])
    if last - first + 1 != int(columns.size):
        return ABSTAIN
    if last - first + 1 < MIN_SPAN_FRACTION * width:
        return ABSTAIN

    above = 0.0
    below = 0.0
    for x in range(first, last + 1):
        rows = np.flatnonzero(trace[:, x])
        run = int(rows[-1] - rows[0]) + 1
        if run != int(rows.size) or run > MAX_COLUMN_RUN:
            return ABSTAIN
        high = baseline - int(rows[0])
        low = baseline - int(rows[-1])
        if low > 0:
            above += (low + high) / 2.0
        elif high < 0:
            below += -(low + high) / 2.0
    for end in (first, last):
        rows = np.flatnonzero(trace[:, end])
        if abs(baseline - int(rows[0])) > STROKE_HALF + 1:
            return ABSTAIN
        if abs(baseline - int(rows[-1])) > STROKE_HALF + 1:
            return ABSTAIN
    if above + below < MIN_TOTAL_AREA:
        return ABSTAIN
    if abs(above - below) < ABSTAIN_GAP:
        return ABSTAIN
    return "above" if above > below else "below"
