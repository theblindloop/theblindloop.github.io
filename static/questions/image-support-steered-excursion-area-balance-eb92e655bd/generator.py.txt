"""Renderer for excursion_area_balance.

One thin dark trace runs the full width of a pale sheet and crosses a straight
blue reference line.  Between its two ends it makes several separate bulges,
each of which leaves the reference line, reaches an extremum, and returns to
it.  The decision is whether the paper area trapped between the trace and the
reference line is larger on the above side or on the below side, so it is a
signed integral of one connected thin trace over the whole sheet width.

Latent state z stores the sheet extent, the reference-line row, the left end of
the trace, the signed per-column offsets of the trace from the reference line,
the stroke half-thickness, the pale edge ticks, and the pale decoration.  The
analytic gold reads the per-column offsets and nothing else.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "excursion_area_balance-0.1.0"

WIDTH = 1000
HEIGHT = 600
X0 = 40
SPAN = 920
LEAD = 26

BG = (247, 246, 242)
BASE = (40, 72, 150)
TRACE = (16, 16, 16)
TICK = (196, 196, 192)
DECOR = (
    (214, 178, 120),
    (150, 190, 168),
    (196, 168, 206),
    (170, 196, 214),
    (206, 196, 176),
)

STROKE_HALF = 1
MAX_AMPLITUDE = 42.0
MIN_AMPLITUDE = 20.0
MAX_STEP = 3
MIN_EXCURSION_AREA = 2600.0
MARGIN_FLOOR = 1700.0
QUARANTINE_MARGIN = 1950.0
NECESSITY_SLACK = 350.0


def _r(value: float) -> int:
    """Deterministic half-up rounding shared by the gold and the rasteriser."""

    return int(math.floor(float(value) + 0.5))


def _column_runs(offsets: list) -> list:
    """Signed extreme offsets of the drawn ink of every trace column."""

    ints = [_r(value) for value in offsets]
    runs = []
    for index, high in enumerate(ints):
        low = ints[index - 1] if index else high
        if low <= high:
            runs.append((low, high))
        else:
            runs.append((high, low))
    return runs


def _areas(offsets: list) -> tuple:
    """Trapped paper area above and below the reference line, in pixels."""

    above = 0.0
    below = 0.0
    for low, high in _column_runs(offsets):
        if low - STROKE_HALF > 0:
            above += (low + high) / 2.0
        elif high + STROKE_HALF < 0:
            below += -(low + high) / 2.0
    return above, below


def analytic_gold(scene: dict) -> str:
    above, below = _areas(scene["offsets"])
    return "above" if above > below else "below"


def margin(scene: dict) -> float:
    above, below = _areas(scene["offsets"])
    return float(abs(above - below))


def is_quarantined(scene: dict) -> bool:
    above, below = _areas(scene["offsets"])
    return bool(abs(above - below) <= QUARANTINE_MARGIN)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    for tick in scene["ticks"]:
        draw.line([(tick[0], tick[1]), (tick[0], tick[2])], fill=TICK, width=1)
    for item in scene["decor"]:
        box = [item[1], item[2], item[3], item[4]]
        colour = tuple(item[5])
        if item[0] == "rect":
            draw.rectangle(box, outline=colour, width=2)
        elif item[0] == "fill":
            draw.rectangle(box, fill=colour)
        else:
            draw.ellipse(box, outline=colour, width=2)
    baseline = int(scene["baseline_y"])
    draw.line([(18, baseline), (width - 19, baseline)], fill=BASE, width=1)
    half = int(scene["stroke"])
    left = int(scene["x0"])
    pixels = image.load()
    for index, (low, high) in enumerate(_column_runs(scene["offsets"])):
        x = left + index
        for offset in range(low - half, high + half + 1):
            pixels[x, baseline - offset] = TRACE
    return image


def latent_symmetries(scene: dict) -> list:
    """Sub-pixel offset aliases: a different latent, the identical raster."""

    transforms = []
    for name, shift in (("subpixel_shift_up", 0.12), ("subpixel_shift_down", -0.12)):
        twin = dict(scene)
        twin["offsets"] = [float(value) + shift for value in scene["offsets"]]
        transforms.append((name, twin))
    return transforms


def _snap(value: float) -> float:
    """Quantise to a drawn row while keeping a sub-pixel alias band."""

    rounded = float(_r(value))
    return rounded + 0.25 if rounded >= 0 else rounded - 0.25


def _excursion_area(offsets: list, lo: int, hi: int) -> float:
    runs = _column_runs(offsets)
    total = 0.0
    for index in range(lo, hi):
        low, high = runs[index]
        if low - STROKE_HALF > 0 or high + STROKE_HALF < 0:
            total += abs(low + high) / 2.0
    return total


def _attempt(rng, want_above, bit_amp, bit_wid, bit_first):
    count = rng.choice((4, 6))
    signs = [1] * (count // 2) + [-1] * (count // 2)
    rng.shuffle(signs)
    if (signs[0] == 1) != bit_first:
        return None
    gap = rng.randint(16, 34)
    usable = SPAN - 2 * LEAD - gap * (count - 1)
    weights = [rng.uniform(1.0, 1.65) for _ in range(count)]
    total_weight = sum(weights)
    widths = [max(60, int(usable * weight / total_weight)) for weight in weights]
    if sum(widths) > usable:
        return None
    amps = []
    for width in widths:
        ceiling = min(MAX_AMPLITUDE, 0.8 * width)
        if ceiling <= MIN_AMPLITUDE + 2.0:
            return None
        amps.append(rng.uniform(MIN_AMPLITUDE + 2.0, ceiling))

    win = 1 if want_above else -1
    raw = [0.63662 * amp * width for amp, width in zip(amps, widths)]
    win_sum = sum(raw[j] for j in range(count) if signs[j] == win)
    lose_sum = sum(raw[j] for j in range(count) if signs[j] != win)
    target = rng.uniform(MARGIN_FLOOR + 60.0, 3100.0)
    scale = (lose_sum + target) / win_sum
    if not 0.45 <= scale <= 2.2:
        return None
    amps = [amp * scale if signs[j] == win else amp for j, amp in enumerate(amps)]
    for amp, width in zip(amps, widths):
        if amp < MIN_AMPLITUDE or amp > min(MAX_AMPLITUDE, 0.8 * width):
            return None

    max_above = max(amps[j] for j in range(count) if signs[j] == 1)
    max_below = max(amps[j] for j in range(count) if signs[j] == -1)
    if max_above == max_below:
        return None
    if ((max_above > max_below) == want_above) == bit_amp:
        return None
    width_above = sum(widths[j] for j in range(count) if signs[j] == 1)
    width_below = sum(widths[j] for j in range(count) if signs[j] == -1)
    if width_above == width_below:
        return None
    if ((width_above > width_below) == want_above) == bit_wid:
        return None

    offsets = [0.0] * SPAN
    spans = []
    cursor = LEAD
    for j in range(count):
        width = widths[j]
        for step in range(width):
            offsets[cursor + step] = signs[j] * amps[j] * math.sin(
                math.pi * (step + 0.5) / width
            )
        spans.append((cursor, cursor + width, signs[j]))
        cursor += width + gap
    offsets = [_snap(value) for value in offsets]

    runs = _column_runs(offsets)
    for low, high in runs:
        if high - low > MAX_STEP:
            return None
    above, below = _areas(offsets)
    if (above > below) != want_above:
        return None
    gap_area = abs(above - below)
    if gap_area < MARGIN_FLOOR:
        return None
    for lo, hi, sign in spans:
        area = _excursion_area(offsets, lo, hi)
        if area < MIN_EXCURSION_AREA:
            return None
        if sign == win and area < gap_area + NECESSITY_SLACK:
            return None

    baseline = rng.randint(264, 276)
    ticks = []
    for _ in range(rng.randint(16, 28)):
        ticks.append([rng.randint(24, WIDTH - 25), 16, rng.randint(26, 40)])
    for _ in range(rng.randint(16, 28)):
        ticks.append([rng.randint(24, WIDTH - 25), rng.randint(HEIGHT - 41, HEIGHT - 27),
                      HEIGHT - 17])
    decor = []
    for _ in range(rng.randint(4, 8)):
        box_w = rng.randint(30, 110)
        box_h = rng.randint(16, 44)
        dx = rng.randint(26, WIDTH - 26 - box_w)
        if rng.random() < 0.5:
            dy = rng.randint(52, 196 - box_h)
        else:
            dy = rng.randint(344, HEIGHT - 56 - box_h)
        kind = rng.choice(("rect", "ellipse", "fill"))
        decor.append([kind, dx, dy, dx + box_w, dy + box_h, list(rng.choice(DECOR))])

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "baseline_y": baseline,
        "x0": X0,
        "offsets": offsets,
        "stroke": STROKE_HALF,
        "ticks": ticks,
        "decor": decor,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 104729 + 17)
    want_above = ((seed // 8) % 2) == 0
    bit_amp = (seed % 2) == 1
    bit_wid = ((seed // 2) % 2) == 1
    bit_first = ((seed // 4) % 2) == 1
    for _ in range(900):
        scene = _attempt(rng, want_above, bit_amp, bit_wid, bit_first)
        if scene is not None:
            return scene
    raise RuntimeError(f"no legible excursion profile for seed {seed}")
