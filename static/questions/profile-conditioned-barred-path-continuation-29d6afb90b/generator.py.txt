"""Deterministic renderer for the cross-occluder continuation world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


RENDERER_VERSION = "barred_path_continuation-0.1.0"
CANVAS = (360, 300)
SUPERSAMPLE = 3
COLORS = ("red", "blue", "green")
PALETTE = {
    "red": (204, 61, 68),
    "blue": (52, 101, 190),
    "green": (48, 146, 103),
}


def _curve_y(scene: dict, lane: int, x: float, offset: float = 0.0) -> float:
    start = float(scene["x_start"])
    end = float(scene["x_end"])
    t = (float(x) - start) / (end - start)
    v = t - 0.5
    coefficients = scene["lane_coefficients"][lane]
    shape = (
        float(coefficients[0]) * v
        + float(coefficients[1]) * (v * v - 0.08333333333333333)
        + float(coefficients[2]) * (v * v * v - 0.125 * v)
    )
    return float(scene["lane_centers"][lane]) + float(scene["path_scale"]) * shape + offset


def sample_scene(seed: int) -> dict:
    """Sample balanced lanes, with target identity independent of lane and color order."""

    rng = random.Random((int(seed) + 7919) * 1009)
    target_color = COLORS[int(seed) % len(COLORS)]
    lane_colors = list(COLORS)
    rng.shuffle(lane_colors)
    lane_draw_order = list(range(3))
    rng.shuffle(lane_draw_order)
    lane_centers = [
        round(59.0 + rng.uniform(-5.0, 5.0), 3),
        round(150.0 + rng.uniform(-5.0, 5.0), 3),
        round(241.0 + rng.uniform(-5.0, 5.0), 3),
    ]
    coefficients = [
        [
            round(rng.uniform(-0.68, 0.68), 5),
            round(rng.uniform(-0.62, 0.62), 5),
            round(rng.uniform(-0.54, 0.54), 5),
        ]
        for _ in range(3)
    ]
    target_lane = lane_colors.index(target_color)
    near_boundary = int(seed) % 11 == 0
    offsets = []
    for lane in range(3):
        if lane == target_lane:
            offsets.append(0.0)
        else:
            magnitude = rng.uniform(10.5, 13.2) if near_boundary else rng.uniform(23.0, 31.0)
            offsets.append(round(magnitude * (-1.0 if rng.randrange(2) else 1.0), 3))
    return {
        "target_color": target_color,
        "lane_colors": lane_colors,
        "lane_draw_order": lane_draw_order,
        "lane_centers": lane_centers,
        "lane_coefficients": coefficients,
        "path_scale": round(rng.uniform(31.0, 38.0), 3),
        "decoy_offsets": offsets,
        "x_start": round(rng.uniform(29.0, 39.0), 3),
        "x_end": round(rng.uniform(321.0, 331.0), 3),
        "panel_left": round(rng.uniform(143.0, 151.0), 3),
        "panel_right": round(rng.uniform(209.0, 217.0), 3),
        "panel_top": 18.0,
        "panel_bottom": 282.0,
        "bar_positions": [
            round(151.0 + 13.0 * index + rng.uniform(-1.0, 1.0), 3)
            for index in range(5)
        ],
        "bar_width": round(rng.uniform(7.0, 9.0), 3),
        "stroke_width": round(rng.uniform(5.0, 6.4), 3),
        "foreground_palette": {name: list(PALETTE[name]) for name in COLORS},
        "panel_palette": {"fill": [230, 232, 228], "bar": [83, 92, 91]},
        "background": [248, 249, 245],
    }


def render(scene: dict) -> Image.Image:
    """Paint full paths first, then hide their middle behind the opaque panel."""

    image = Image.new(
        "RGB",
        (CANVAS[0] * SUPERSAMPLE, CANVAS[1] * SUPERSAMPLE),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    scale = SUPERSAMPLE
    width = max(1, int(round(float(scene["stroke_width"]) * scale)))
    for lane in scene["lane_draw_order"]:
        color_name = str(scene["lane_colors"][lane])
        color = tuple(int(value) for value in scene["foreground_palette"][color_name])
        # Read the answer-bearing field in the renderer as well as in gold.  A
        # changed target color therefore changes which visible right fragment
        # is the unshifted continuation; target_color is not metadata-only.
        offset = (
            0.0
            if color_name == str(scene["target_color"])
            else float(scene["decoy_offsets"][lane])
        )
        points = [
            (
                int(round(x * scale)),
                int(
                    round(
                        _curve_y(
                            scene,
                            lane,
                            x,
                            offset if x > float(scene["panel_right"]) else 0.0,
                        )
                        * scale
                    )
                ),
            )
            for x in range(int(math.floor(float(scene["x_start"]))), int(math.ceil(float(scene["x_end"]))) + 1)
        ]
        draw.line(points, fill=color, width=width, joint="curve")

    left = int(round(float(scene["panel_left"]) * scale))
    right = int(round(float(scene["panel_right"]) * scale))
    top = int(round(float(scene["panel_top"]) * scale))
    bottom = int(round(float(scene["panel_bottom"]) * scale))
    panel_fill = tuple(int(value) for value in scene["panel_palette"]["fill"])
    bar_fill = tuple(int(value) for value in scene["panel_palette"]["bar"])
    draw.rectangle((left, top, right, bottom), fill=panel_fill)
    bar_width = int(round(float(scene["bar_width"]) * scale))
    for position in scene["bar_positions"]:
        x = int(round(float(position) * scale))
        draw.rectangle((x, top, x + bar_width, bottom), fill=bar_fill)
    return image.resize(CANVAS, Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The sole analytic label input is the visible target color."""

    return str(scene["target_color"])


def margin(scene: dict) -> float:
    """Runner-up cross-panel offset gap in pixels."""

    target = str(scene["target_color"])
    target_lane = list(scene["lane_colors"]).index(target)
    candidates = [
        abs(float(offset))
        for lane, offset in enumerate(scene["decoy_offsets"])
        if lane != target_lane
    ]
    return round(min(candidates), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["lane_draw_order"] = list(reversed(scene["lane_draw_order"]))
    return [("reverse_nonoverlapping_lane_draw_order", twin)]
