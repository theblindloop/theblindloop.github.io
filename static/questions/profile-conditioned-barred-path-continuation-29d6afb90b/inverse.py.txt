"""Independent PNG-only inverse for barred path continuation."""

from __future__ import annotations

import numpy as np
from PIL import Image


_COLORS = {
    "red": np.asarray((204.0, 61.0, 68.0), dtype=np.float32),
    "blue": np.asarray((52.0, 101.0, 190.0), dtype=np.float32),
    "green": np.asarray((48.0, 146.0, 103.0), dtype=np.float32),
}


def _centerline(image: Image.Image, color: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    all_colors = np.stack(list(_COLORS.values()), axis=0)
    all_distances = np.sqrt(((array[:, :, None, :] - all_colors[None, None, :, :]) ** 2).sum(axis=3))
    color_index = next(
        index for index, candidate in enumerate(all_colors) if np.array_equal(candidate, color)
    )
    distance = all_distances[:, :, color_index]
    nearest = np.argmin(all_distances, axis=2)
    # The striped panel is intentionally gray and is close to the green
    # foreground in Euclidean RGB distance.  Chroma is the pixel-only way to
    # reject that neutral occluder without consulting its declared geometry.
    chroma = array.max(axis=2) - array.min(axis=2)
    mask = (nearest == color_index) & (distance < 92.0) & (chroma > 50.0)
    # The public question fixes one central opaque panel.  Exclude a generous
    # border around it: antialiasing at a neutral panel edge can blend with a
    # neighboring colored stroke, but no answer-bearing centerline is visible
    # inside this interval.
    columns = np.arange(array.shape[1])[None, :]
    mask &= (columns < 135) | (columns > 225)
    ys: list[float] = []
    xs: list[float] = []
    for x in range(array.shape[1]):
        columns = np.flatnonzero(mask[:, x])
        if columns.size:
            xs.append(float(x))
            ys.append(float(np.median(columns)))
    return np.asarray(xs, dtype=np.float64), np.asarray(ys, dtype=np.float64)


def _continuation_residual(xs: np.ndarray, ys: np.ndarray) -> float:
    if xs.size < 40:
        return float("inf")
    center = (float(xs.min()) + float(xs.max())) * 0.5
    span = max(float(xs.max()) - float(xs.min()), 1.0)
    u = (xs - center) / span
    design = np.column_stack((np.ones_like(u), u, u * u, u * u * u))
    coefficients, _, _, _ = np.linalg.lstsq(design, ys, rcond=None)
    residuals = ys - design @ coefficients
    # A small fixed trimming removes only the two antialiased end caps; the
    # central missing interval is already absent from the fitted pixels.
    return float(np.sqrt(np.mean(np.square(residuals))))


def decision_from_image(image: Image.Image) -> str:
    """Choose the color whose two visible pieces share one cubic centerline."""

    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    scores = []
    for name, color in _COLORS.items():
        xs, ys = _centerline(image, color)
        scores.append((_continuation_residual(xs, ys), name))
    scores.sort(key=lambda item: (item[0], item[1]))
    return scores[0][1]
