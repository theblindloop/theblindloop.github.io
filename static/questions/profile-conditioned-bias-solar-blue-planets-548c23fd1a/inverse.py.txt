"""Scene-free inverse arm: count saturated blue raster components only."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _component_areas(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    areas: list[int] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        areas.append(area)
    return areas


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    mask = (red < 92) & (green > 92) & (green < 194) & (blue > 178) & ((blue - red) > 95)
    count = sum(area >= 45 for area in _component_areas(mask))
    return str(count) if 1 <= count <= 4 else "abstain"
