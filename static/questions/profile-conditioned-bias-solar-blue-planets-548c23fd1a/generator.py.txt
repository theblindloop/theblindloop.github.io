"""Deterministic latent sampler and raster renderer for a prior-conflict solar system."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

SIZE = 512
SCALE = 3
BLUE = (38, 142, 235)
NONBLUE = (
    (229, 87, 74),
    (91, 190, 112),
    (241, 151, 55),
    (174, 112, 211),
    (225, 206, 151),
    (198, 102, 139),
)
THEMES = (
    {"background": (8, 17, 38), "orbit": (112, 126, 146), "sun": (251, 195, 52)},
    {"background": (20, 12, 42), "orbit": (132, 123, 151), "sun": (255, 184, 45)},
)
ORBIT_RADII = (55, 88, 121, 154, 187, 220)
BLUE_RADIUS_PATTERNS = {
    1: ((11,), (12,), (13,)),
    2: ((8, 8), (8, 9), (9, 9)),
    3: ((6, 7, 7), (7, 7, 7), (7, 7, 8)),
    4: ((5, 6, 6, 6), (6, 6, 6, 6), (6, 6, 6, 7)),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    blue_count = 1 + (int(seed) % 4)
    blue_slots = set(rng.sample(range(6), blue_count))
    blue_radii = list(rng.choice(BLUE_RADIUS_PATTERNS[blue_count]))
    rng.shuffle(blue_radii)
    colors = []
    radii = []
    blue_index = 0
    other_palette = list(NONBLUE)
    rng.shuffle(other_palette)
    other_index = 0
    for slot in range(6):
        if slot in blue_slots:
            colors.append(list(BLUE))
            radii.append(blue_radii[blue_index])
            blue_index += 1
        else:
            colors.append(list(other_palette[other_index]))
            radii.append(rng.randint(7, 12))
            other_index += 1
    return {
        "blue_count": blue_count,
        "orbit_radii": list(ORBIT_RADII),
        "planet_angles": [rng.randrange(0, 360) for _ in range(6)],
        "planet_radii": radii,
        "planet_colors": colors,
        "angle_alias": rng.randrange(0, 360),
        "palette_theme": (int(seed) // 4) % len(THEMES),
    }


def _scaled_box(cx: float, cy: float, radius: float) -> tuple[int, int, int, int]:
    return tuple(round(value * SCALE) for value in (cx - radius, cy - radius, cx + radius, cy + radius))


def render(scene: dict) -> Image.Image:
    theme = THEMES[int(scene["palette_theme"]) % len(THEMES)]
    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), theme["background"])
    draw = ImageDraw.Draw(image)
    center = SIZE / 2
    orbit_radii = [int(value) for value in scene["orbit_radii"]]
    angles = [int(value) for value in scene["planet_angles"]]
    radii = [int(value) for value in scene["planet_radii"]]
    colors = [tuple(value) for value in scene["planet_colors"]]
    alias = int(scene["angle_alias"]) % 360
    declared_count = int(scene["blue_count"])
    if len(orbit_radii) != 6 or len(angles) != 6 or len(radii) != 6 or len(colors) != 6:
        raise ValueError("scene must contain six visible orbits and planets")
    if sum(color == BLUE for color in colors) != declared_count:
        raise ValueError("blue_count must match the rendered blue planet colors")

    for radius in orbit_radii:
        box = _scaled_box(center, center, radius)
        draw.ellipse(box, outline=theme["orbit"], width=2 * SCALE)

    draw.ellipse(_scaled_box(center, center, 21), fill=theme["sun"], outline=(255, 223, 104), width=SCALE)

    for orbit, angle, radius, color in zip(orbit_radii, angles, radii, colors):
        theta = math.radians((angle + alias) % 360)
        cx = center + orbit * math.cos(theta)
        cy = center + orbit * math.sin(theta)
        draw.ellipse(_scaled_box(cx, cy, radius), fill=color, outline=(241, 245, 249), width=SCALE)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["blue_count"]))


def margin(scene: dict) -> float:
    targets = [
        int(radius)
        for radius, color in zip(scene["planet_radii"], scene["planet_colors"])
        if tuple(color) == BLUE
    ]
    return float(2 * min(targets))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["angle_alias"] = int(twin["angle_alias"]) + 360
    return [("add_full_turn_to_angle_alias", twin)]
