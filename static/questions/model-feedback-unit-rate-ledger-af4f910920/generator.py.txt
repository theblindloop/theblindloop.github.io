"""Unit-rate ledger world.

Four horizontal ledger rows are drawn on white. Each row carries

* a solid colored swatch that names the row (blue, green, orange, purple),
* a colored bar laid over a ruler of evenly spaced black tick marks whose
  spacing is chosen independently for that row, and
* a run of identical black squares.

The bar always begins at the ruler's first tick and ends exactly on one later
tick, so the bar's length expressed in that row's own tick steps is an exact
integer ``k``. The reported quantity is the product of that step count with the
number of black squares in the row. Because the tick pitch differs per row, the
bar's pixel length is a deliberately misleading cue: every scene is sampled so
that the longest drawn bar, the largest step count, the largest square count and
the widest ruler each belong to a row that is *not* the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1040
HEIGHT = 720

_BG = (255, 255, 255)
_INK = (0, 0, 0)

COLOR_NAMES = ("blue", "green", "orange", "purple")

COLORS = {
    "blue": (30, 80, 200),
    "green": (25, 150, 60),
    "orange": (240, 130, 20),
    "purple": (150, 45, 190),
}

ROW_COUNT = 4
BAND_TOP = 30
BAND_HEIGHT = 168

SWATCH_X = 28
SWATCH_SIZE = 56

TRACK_X = 132
TICK_WIDTH = 6
TICK_HEIGHT = 30
TICK_TOP_OFFSET = 8
BAR_THICKNESS = 26
BAR_TOP_OFFSET = 30

SQUARE_X = 720
SQUARE_SIZE = 38

SPACINGS = (34, 44, 54, 64)
QUARANTINE_MARGIN = 11.0


def _row_center(index: int) -> int:
    return BAND_TOP + index * BAND_HEIGHT + BAND_HEIGHT // 2


def _argmax_unique(values):
    """Return the index of a strict maximum, or ``None`` when it is tied."""
    best = max(values)
    winners = [i for i, v in enumerate(values) if v == best]
    return winners[0] if len(winners) == 1 else None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(60000):
        spacings = list(SPACINGS)
        rng.shuffle(spacings)
        names = list(COLOR_NAMES)
        rng.shuffle(names)
        ks = [rng.randint(2, 6) for _ in range(ROW_COUNT)]
        ns = [rng.randint(2, 5) for _ in range(ROW_COUNT)]

        products = [k * n for k, n in zip(ks, ns)]
        winner = _argmax_unique(products)
        if winner is None:
            continue
        ordered = sorted(products, reverse=True)
        gap = ordered[0] - ordered[1]
        if gap < 1:
            continue
        # Product gaps of one step are genuine near-boundary scenes and are kept
        # only occasionally, so the quarantined tail stays represented in the
        # gallery without dominating the dataset.
        if gap == 1 and rng.random() >= 0.15:
            continue

        # Every naive single-cue reading must point somewhere else.
        lengths = [k * s for k, s in zip(ks, spacings)]
        decoys = [
            _argmax_unique(lengths),
            _argmax_unique(ns),
            _argmax_unique(ks),
            _argmax_unique(spacings),
        ]
        if any(d is None or d == winner for d in decoys):
            continue

        return {
            "colors": names,
            "spacings": spacings,
            "steps": ks,
            "squares": ns,
            "jitters": [rng.randint(0, 14) for _ in range(ROW_COUNT)],
            "square_gaps": [rng.choice([20, 24, 28]) for _ in range(ROW_COUNT)],
        }
    raise RuntimeError("scene rejection sampling failed")


def analytic_gold(scene: dict) -> str:
    """Reads only ``steps``, ``squares`` and ``colors``: the declared label inputs."""
    products = [k * n for k, n in zip(scene["steps"], scene["squares"])]
    winner = _argmax_unique(products)
    if winner is None:
        raise ValueError("scene has no unique largest product")
    return scene["colors"][winner]


def margin(scene: dict) -> float:
    products = [k * n for k, n in zip(scene["steps"], scene["squares"])]
    ordered = sorted(products, reverse=True)
    product_gap = ordered[0] - ordered[1]

    tick_clearance = min(s - TICK_WIDTH for s in scene["spacings"])
    square_clearance = min(scene["square_gaps"])
    # Vertical clearance between the tallest mark of one row and the next row.
    row_clearance = BAND_HEIGHT - (BAR_TOP_OFFSET + TICK_TOP_OFFSET + TICK_HEIGHT)
    return float(
        min(
            tick_clearance,
            square_clearance,
            row_clearance,
            8.0 * product_gap,
        )
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """No latent alias exists: every sampled field is drawn directly."""
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BG)
    draw = ImageDraw.Draw(image)

    for index in range(len(scene["colors"])):
        cy = _row_center(index)
        rgb = COLORS[scene["colors"][index]]

        draw.rectangle(
            [SWATCH_X, cy - SWATCH_SIZE // 2, SWATCH_X + SWATCH_SIZE - 1, cy + SWATCH_SIZE // 2 - 1],
            fill=rgb,
        )

        x0 = TRACK_X + scene["jitters"][index]
        spacing = scene["spacings"][index]
        steps = scene["steps"][index]
        for i in range(steps + 3):
            cx = x0 + i * spacing
            draw.rectangle(
                [
                    cx - TICK_WIDTH // 2,
                    cy + TICK_TOP_OFFSET,
                    cx - TICK_WIDTH // 2 + TICK_WIDTH - 1,
                    cy + TICK_TOP_OFFSET + TICK_HEIGHT - 1,
                ],
                fill=_INK,
            )

        draw.rectangle(
            [
                x0 - TICK_WIDTH // 2,
                cy - BAR_TOP_OFFSET,
                x0 + steps * spacing - TICK_WIDTH // 2 + TICK_WIDTH - 1,
                cy - BAR_TOP_OFFSET + BAR_THICKNESS - 1,
            ],
            fill=rgb,
        )

        sx = SQUARE_X
        for _ in range(scene["squares"][index]):
            draw.rectangle(
                [sx, cy - SQUARE_SIZE // 2, sx + SQUARE_SIZE - 1, cy + SQUARE_SIZE // 2 - 1],
                fill=_INK,
            )
            sx += SQUARE_SIZE + scene["square_gaps"][index]

    return image
