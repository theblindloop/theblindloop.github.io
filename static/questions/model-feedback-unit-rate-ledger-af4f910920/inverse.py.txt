"""Scene-free pixel oracle for the unit-rate ledger world.

Receives only a rendered PNG. It segments the four colored rows, locates each
row's bar, counts the black ruler ticks the bar actually spans, counts the black
squares in the same row, and reports the color of the row with the largest
product. It never imports the renderer, the prompts, the generator, the verifier
or any analytic-gold code, and it never sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# The rendered palette is public information carried by the image itself.
_PALETTE = {
    "blue": (30, 80, 200),
    "green": (25, 150, 60),
    "orange": (240, 130, 20),
    "purple": (150, 45, 190),
}

_INK_MAX = 40          # a pixel this dark in every channel is ruler/square ink
_MIN_SQUARE_WIDTH = 20  # ticks are far narrower than squares
_TICK_TOLERANCE = 4     # slack when testing whether a tick sits under the bar


def _column_runs(present: np.ndarray) -> list:
    """Split a boolean column-presence vector into (start, end) inclusive runs."""
    runs = []
    start = None
    for x, flag in enumerate(present):
        if flag and start is None:
            start = x
        elif not flag and start is not None:
            runs.append((start, x - 1))
            start = None
    if start is not None:
        runs.append((start, len(present) - 1))
    return runs


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    height, width, _ = arr.shape
    ink = (arr[:, :, 0] <= _INK_MAX) & (arr[:, :, 1] <= _INK_MAX) & (arr[:, :, 2] <= _INK_MAX)

    scored = []
    for name, rgb in _PALETTE.items():
        mask = (
            (arr[:, :, 0] == rgb[0])
            & (arr[:, :, 1] == rgb[1])
            & (arr[:, :, 2] == rgb[2])
        )
        if not mask.any():
            raise ValueError(f"color {name} is absent from the image")

        runs = _column_runs(mask.any(axis=0))
        bar_x0, bar_x1 = max(runs, key=lambda run: run[1] - run[0])
        bar_rows = np.nonzero(mask[:, bar_x0 : bar_x1 + 1].any(axis=1))[0]
        bar_top = int(bar_rows.min())

        band_lo = max(0, bar_top - 20)
        band_hi = min(height, bar_top + 110)
        band = ink[band_lo:band_hi, :]

        squares = 0
        ticks_under_bar = 0
        for x0, x1 in _column_runs(band.any(axis=0)):
            if x1 - x0 + 1 >= _MIN_SQUARE_WIDTH:
                squares += 1
                continue
            center = 0.5 * (x0 + x1)
            if bar_x0 - _TICK_TOLERANCE <= center <= bar_x1 + _TICK_TOLERANCE:
                ticks_under_bar += 1

        steps = ticks_under_bar - 1
        if steps < 1 or squares < 1:
            raise ValueError(f"row {name} could not be measured")
        scored.append((steps * squares, name))

    best = max(score for score, _ in scored)
    winners = sorted(name for score, name in scored if score == best)
    if len(winners) != 1:
        raise ValueError("no unique largest product in the image")
    return winners[0]
