"""Balanced barbell-railway scenes and analytic graph decision."""

from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

WIDTH, HEIGHT, SCALE = 640, 480, 3
BG = (248, 246, 239)
TRACK = (92, 108, 120)
JUNCTION = (28, 32, 36)
ORANGE = (232, 105, 32)
BLUE = (38, 112, 218)

BASE_NODES = [
    (190, 205), (270, 205), (270, 275), (190, 275),
    (370, 205), (450, 205), (450, 275), (370, 275),
]
BASE_EDGES = [[0, 1], [1, 2], [2, 3], [3, 0], [4, 5], [5, 6], [6, 7], [7, 4], [2, 7]]


def _transform(points: list[tuple[int, int]], quarter_turns: int, mirror: bool, dx: int, dy: int):
    out = []
    cx, cy = WIDTH // 2, HEIGHT // 2
    for x, y in points:
        x, y = x - cx, y - cy
        if mirror:
            x = -x
        for _ in range(quarter_turns):
            x, y = -y, x
        out.append([int(x + cx + dx), int(y + cy + dy)])
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    yes = (seed % 2 == 0)
    turns = rng.randrange(4)
    # Keep translations modest enough that the graph remains a regional object.
    dx = rng.randrange(-24, 25)
    dy = rng.randrange(-18, 19)
    nodes = _transform(BASE_NODES, turns, bool(rng.randrange(2)), dx, dy)
    left = [0, 1, 2, 3]
    right = [4, 5, 6, 7]
    orange_loop = left if rng.randrange(2) == 0 else right
    orange = rng.choice(orange_loop)
    if yes:
        blue = rng.choice([n for n in orange_loop if n != orange])
    else:
        other = right if orange_loop is left else left
        blue = rng.choice(other)
    return {
        "nodes": nodes,
        "edges": [edge[:] for edge in BASE_EDGES],
        "terminal_nodes": [orange, blue],
    }


def _reachable(edges: list[list[int]], start: int, goal: int, omitted: int | None = None) -> bool:
    adjacency: dict[int, list[int]] = {}
    for index, (a, b) in enumerate(edges):
        if index == omitted:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    seen = {start}
    queue = deque([start])
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for nxt in adjacency.get(node, []):
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    start, goal = scene["terminal_nodes"]
    if not _reachable(edges, start, goal):
        return "no"
    return "yes" if all(_reachable(edges, start, goal, i) for i in range(len(edges))) else "no"


def margin(scene: dict) -> float:
    # Both supported classes are one discrete edge-connectivity unit from the boundary.
    edges = scene["edges"]
    start, goal = scene["terminal_nodes"]
    cut_edges = sum(not _reachable(edges, start, goal, i) for i in range(len(edges)))
    return 1.0 if cut_edges == 0 else float(cut_edges)


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    if len(nodes) != 8 or len(edges) != 9:
        return True
    for a, b in edges:
        x1, y1 = nodes[a]
        x2, y2 = nodes[b]
        if ((x2 - x1) ** 2 + (y2 - y1) ** 2) ** 0.5 < 55:
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BG)
    draw = ImageDraw.Draw(image)
    nodes = [(int(x * SCALE), int(y * SCALE)) for x, y in scene["nodes"]]
    for a, b in scene["edges"]:
        draw.line([nodes[a], nodes[b]], fill=TRACK, width=14 * SCALE)
    terminal_map = {
        int(scene["terminal_nodes"][0]): ORANGE,
        int(scene["terminal_nodes"][1]): BLUE,
    }
    for index, (x, y) in enumerate(nodes):
        radius = (15 if index in terminal_map else 14) * SCALE
        color = terminal_map.get(index, JUNCTION)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
