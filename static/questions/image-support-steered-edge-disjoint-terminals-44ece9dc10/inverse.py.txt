"""Independent pixel-only recovery of the visible railway graph."""

from __future__ import annotations

from collections import deque
import math
import numpy as np
from PIL import Image

PALETTE = np.asarray([
    (248, 246, 239), (92, 108, 120), (28, 32, 36), (232, 105, 32), (38, 112, 218)
], dtype=np.float32)


def _components(mask: np.ndarray, minimum: int = 80):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        pixels = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= minimum:
            pts = np.asarray(pixels, dtype=np.float64)
            output.append((len(pixels), (float(pts[:, 1].mean()), float(pts[:, 0].mean()))))
    return output


def _near_other_node(a, b, point, radius=22.0):
    ax, ay = a
    bx, by = b
    px, py = point
    vx, vy = bx - ax, by - ay
    length2 = vx * vx + vy * vy
    if length2 <= 0:
        return False
    t = ((px - ax) * vx + (py - ay) * vy) / length2
    if not 0.12 < t < 0.88:
        return False
    qx, qy = ax + t * vx, ay + t * vy
    return math.hypot(px - qx, py - qy) < radius


def _has_track(gray: np.ndarray, a, b, all_centers) -> bool:
    if any(_near_other_node(a, b, p) for p in all_centers if p != a and p != b):
        return False
    distance = math.dist(a, b)
    samples = max(12, int(distance / 3))
    hits = 0
    used = 0
    for k in range(samples + 1):
        t = k / samples
        if t * distance < 19 or (1 - t) * distance < 19:
            continue
        x = int(round(a[0] * (1 - t) + b[0] * t))
        y = int(round(a[1] * (1 - t) + b[1] * t))
        y0, y1 = max(0, y - 4), min(gray.shape[0], y + 5)
        x0, x1 = max(0, x - 4), min(gray.shape[1], x + 5)
        used += 1
        hits += bool(gray[y0:y1, x0:x1].mean() >= 0.42)
    return used >= 8 and hits / used >= 0.86


def _reachable(edges, start, goal, omitted=None):
    adjacency = {i: [] for i in range(8)}
    for index, (a, b) in enumerate(edges):
        if index == omitted:
            continue
        adjacency[a].append(b)
        adjacency[b].append(a)
    queue, seen = deque([start]), {start}
    while queue:
        node = queue.popleft()
        if node == goal:
            return True
        for nxt in adjacency[node]:
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = ((array[:, :, None, :] - PALETTE[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)
    gray = labels == 1
    black = _components(labels == 2, 180)
    orange = _components(labels == 3, 180)
    blue = _components(labels == 4, 180)
    if len(black) != 6 or len(orange) != 1 or len(blue) != 1:
        raise ValueError("expected six junctions and two terminals")
    centers = [item[1] for item in black] + [orange[0][1], blue[0][1]]
    start, goal = 6, 7
    edges = []
    for a in range(8):
        for b in range(a + 1, 8):
            if _has_track(gray, centers[a], centers[b], centers):
                edges.append((a, b))
    if len(edges) != 9 or not _reachable(edges, start, goal):
        raise ValueError(f"ambiguous railway graph with {len(edges)} tracks")
    return "yes" if all(_reachable(edges, start, goal, i) for i in range(len(edges))) else "no"
