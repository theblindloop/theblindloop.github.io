"""Deterministic renderer for a two-dimensional spiral-raster reveal."""

from __future__ import annotations

import math
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 440
TILE_SIZE = 240
MARKER_OFFSET = 18
MARKER_RADIUS = 8
SYMBOLS = ("arch", "coil", "kite", "trident")

MARKER_PALETTES = (
    ((213, 58, 61), (39, 142, 207), (67, 166, 98), (221, 157, 42)),
    ((121, 76, 190), (218, 91, 48), (32, 158, 151), (198, 174, 39)),
    ((31, 137, 168), (210, 69, 103), (102, 155, 59), (224, 125, 42)),
    ((196, 72, 155), (44, 151, 92), (226, 132, 49), (55, 106, 196)),
)


def _spiral_path(size: int) -> list[tuple[int, int]]:
    """Return the clockwise outside-in path named by the public question."""
    path: list[tuple[int, int]] = []
    top, left, bottom, right = 0, 0, size - 1, size - 1
    while top <= bottom and left <= right:
        for col in range(left, right + 1):
            path.append((top, col))
        top += 1
        for row in range(top, bottom + 1):
            path.append((row, right))
        right -= 1
        if top <= bottom:
            for col in range(right, left - 1, -1):
                path.append((bottom, col))
            bottom -= 1
        if left <= right:
            for row in range(bottom, top - 1, -1):
                path.append((row, left))
            left += 1
    if len(path) != size * size:
        raise AssertionError("spiral path does not cover the square exactly")
    return path


def _draw_glyph(symbol: str, dx: int, dy: int) -> np.ndarray:
    """Draw one broad asymmetric glyph into canonical raster coordinates."""
    glyph = Image.new("L", (TILE_SIZE, TILE_SIZE), 0)
    draw = ImageDraw.Draw(glyph)
    width = 13

    def points(values: Iterable[tuple[float, float]]) -> list[tuple[int, int]]:
        return [(int(round(x + dx)), int(round(y + dy))) for x, y in values]

    def line(values: Iterable[tuple[float, float]]) -> None:
        draw.line(points(values), fill=255, width=width, joint="curve")

    if symbol == "arch":
        line([(88, 178), (88, 104), (101, 79), (126, 64), (153, 63), (177, 78), (184, 105)])
        line([(88, 178), (184, 178)])
    elif symbol == "coil":
        line([(173, 75), (141, 61), (108, 68), (94, 95), (101, 122), (128, 136),
              (158, 130), (169, 108), (158, 91), (129, 89), (112, 103), (119, 119),
              (143, 121), (178, 159)])
    elif symbol == "kite":
        line([(128, 55), (179, 118), (131, 181), (82, 119), (128, 55)])
        line([(131, 181), (131, 207), (158, 207)])
    elif symbol == "trident":
        line([(130, 180), (130, 79), (91, 101), (91, 72)])
        line([(130, 112), (169, 101), (169, 72)])
        line([(130, 79), (130, 54)])
        line([(84, 72), (98, 72), (98, 94)])
        line([(162, 72), (176, 72), (176, 94)])
        line([(107, 180), (153, 180)])
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(glyph, dtype=np.float32) / 255.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed))
    seed = int(seed)
    strength_levels = (40.0, 52.0, 64.0, 76.0)
    return {
        "seed": seed,
        "symbol": SYMBOLS[seed % len(SYMBOLS)],
        "tile_x": int(rng.integers(70, 280)),
        "tile_y": int(rng.integers(70, 150)),
        "carrier_phase": int(rng.integers(0, 8)),
        "carrier_amplitude": float(rng.integers(18, 26)),
        "signal_strength": strength_levels[(seed // len(SYMBOLS)) % len(strength_levels)],
        "base_level": int(rng.integers(182, 204)),
        "symbol_dx": int(rng.integers(-12, 13)),
        "symbol_dy": int(rng.integers(-12, 13)),
        "marker_palette_id": int(rng.integers(0, len(MARKER_PALETTES))),
    }


def _canonical_tile(scene: dict) -> np.ndarray:
    yy, xx = np.indices((TILE_SIZE, TILE_SIZE), dtype=np.float32)
    phase = (int(scene["carrier_phase"]) % 8) / 8.0
    carrier = float(scene["carrier_amplitude"]) * np.sin(
        2.0 * math.pi * (xx / 4.0 + yy / 6.0 + phase)
    )
    mask = _draw_glyph(str(scene["symbol"]), int(scene["symbol_dx"]), int(scene["symbol_dy"]))
    gray = float(scene["base_level"]) - float(scene["signal_strength"]) * mask + carrier
    return np.clip(np.rint(gray), 0, 255).astype(np.uint8)


def _spiral_store(canonical: np.ndarray) -> np.ndarray:
    """Lay a row-major recorded stream into clockwise spiral positions."""
    stored = np.empty_like(canonical)
    stream = canonical.reshape(-1)
    for index, (row, col) in enumerate(_spiral_path(canonical.shape[0])):
        stored[row, col] = stream[index]
    return stored


def render(scene: dict) -> Image.Image:
    canvas = np.full((HEIGHT, WIDTH, 3), 247, dtype=np.uint8)
    tile = _spiral_store(_canonical_tile(scene))
    x = int(scene["tile_x"])
    y = int(scene["tile_y"])
    canvas[y : y + TILE_SIZE, x : x + TILE_SIZE] = tile[:, :, None]

    image = Image.fromarray(canvas, mode="RGB")
    draw = ImageDraw.Draw(image)
    palette = MARKER_PALETTES[int(scene["marker_palette_id"]) % len(MARKER_PALETTES)]
    centers = (
        (x - MARKER_OFFSET, y - MARKER_OFFSET),
        (x + TILE_SIZE + MARKER_OFFSET, y - MARKER_OFFSET),
        (x - MARKER_OFFSET, y + TILE_SIZE + MARKER_OFFSET),
        (x + TILE_SIZE + MARKER_OFFSET, y + TILE_SIZE + MARKER_OFFSET),
    )
    for (cx, cy), color in zip(centers, palette):
        draw.ellipse(
            (cx - MARKER_RADIUS, cy - MARKER_RADIUS, cx + MARKER_RADIUS, cy + MARKER_RADIUS),
            fill=color,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    return float(scene["signal_strength"]) - float(scene["carrier_amplitude"]) - 18.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["carrier_phase"] = int(scene["carrier_phase"]) + 8
    return [("carrier_period_alias", twin)]
