"""Independent image-only inverse for the inward spiral raster world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image, ImageDraw

TILE_SIZE = 240
MARKER_OFFSET = 18
SYMBOLS = ("arch", "coil", "kite", "trident")


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components = []
    for sy, sx in zip(*np.nonzero(mask)):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x
            sum_y += y
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 30:
            components.append((count, sum_x / count, sum_y / count))
    return sorted(components, reverse=True)


def _locate_tile(rgb: np.ndarray) -> np.ndarray:
    spread = rgb.max(axis=2).astype(np.int16) - rgb.min(axis=2).astype(np.int16)
    mean = rgb.mean(axis=2)
    components = _components((spread > 32) & (mean < 235))
    if len(components) < 4:
        raise ValueError("four colored corner marks are required")
    marks = components[:4]
    xs = [item[1] for item in marks]
    ys = [item[2] for item in marks]
    x0 = int(round(min(xs) + MARKER_OFFSET))
    x1 = int(round(max(xs) - MARKER_OFFSET))
    y0 = int(round(min(ys) + MARKER_OFFSET))
    y1 = int(round(max(ys) - MARKER_OFFSET))
    if x1 - x0 != TILE_SIZE or y1 - y0 != TILE_SIZE:
        raise ValueError("colored marks do not define a 240 pixel square")
    tile = rgb[y0:y1, x0:x1].mean(axis=2).astype(np.float32)
    if tile.shape != (TILE_SIZE, TILE_SIZE):
        raise ValueError("marked square crop has the wrong shape")
    return tile


def _spiral_path(size: int) -> list[tuple[int, int]]:
    path: list[tuple[int, int]] = []
    top, left, bottom, right = 0, 0, size - 1, size - 1
    while top <= bottom and left <= right:
        for col in range(left, right + 1):
            path.append((top, col))
        top += 1
        for row in range(top, bottom + 1):
            path.append((row, right))
        right -= 1
        if top <= bottom:
            for col in range(right, left - 1, -1):
                path.append((bottom, col))
            bottom -= 1
        if left <= right:
            for row in range(bottom, top - 1, -1):
                path.append((row, left))
            left += 1
    if len(path) != size * size:
        raise ValueError("spiral path does not cover the square exactly")
    return path


def _restore_spiral(stored: np.ndarray) -> np.ndarray:
    """Read the displayed spiral positions back into the row-major stream."""
    restored = np.empty_like(stored)
    stream = restored.reshape(-1)
    for index, (row, col) in enumerate(_spiral_path(stored.shape[0])):
        stream[index] = stored[row, col]
    return restored


def _box_average(values: np.ndarray, radius: int = 4) -> np.ndarray:
    padded = np.pad(values, ((radius, radius), (radius, radius)), mode="edge")
    integral = np.pad(padded, ((1, 0), (1, 0)), mode="constant")
    integral = integral.cumsum(axis=0).cumsum(axis=1)
    size = 2 * radius + 1
    return (
        integral[size:, size:]
        - integral[:-size, size:]
        - integral[size:, :-size]
        + integral[:-size, :-size]
    ) / float(size * size)


def _glyph(symbol: str, dx: int = 0, dy: int = 0) -> np.ndarray:
    image = Image.new("L", (TILE_SIZE, TILE_SIZE), 0)
    draw = ImageDraw.Draw(image)
    width = 13

    def points(values: list[tuple[float, float]]) -> list[tuple[int, int]]:
        return [(int(round(x + dx)), int(round(y + dy))) for x, y in values]

    def line(values: list[tuple[float, float]]) -> None:
        draw.line(points(values), fill=255, width=width, joint="curve")

    if symbol == "arch":
        line([(88, 178), (88, 104), (101, 79), (126, 64), (153, 63), (177, 78), (184, 105)])
        line([(88, 178), (184, 178)])
    elif symbol == "coil":
        line([(173, 75), (141, 61), (108, 68), (94, 95), (101, 122), (128, 136),
              (158, 130), (169, 108), (158, 91), (129, 89), (112, 103), (119, 119),
              (143, 121), (178, 159)])
    elif symbol == "kite":
        line([(128, 55), (179, 118), (131, 181), (82, 119), (128, 55)])
        line([(131, 181), (131, 207), (158, 207)])
    elif symbol == "trident":
        line([(130, 180), (130, 79), (91, 101), (91, 72)])
        line([(130, 112), (169, 101), (169, 72)])
        line([(130, 79), (130, 54)])
        line([(84, 72), (98, 72), (98, 94)])
        line([(162, 72), (176, 72), (176, 94)])
        line([(107, 180), (153, 180)])
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return np.asarray(image, dtype=np.float32) / 255.0


def _scores(restored_blur: np.ndarray) -> list[tuple[float, str]]:
    high = float(np.percentile(restored_blur, 95))
    low = float(np.percentile(restored_blur, 5))
    scale = max(high - low, 1.0)
    observed = np.clip((high - restored_blur) / scale, 0.0, 1.0)
    observed = observed[32:208:2, 32:208:2]
    scores = []
    for symbol in SYMBOLS:
        best = float("inf")
        for dx in range(-14, 15, 2):
            for dy in range(-14, 15, 2):
                template = _glyph(symbol, dx, dy)[32:208:2, 32:208:2]
                error = float(np.mean((observed - template) ** 2))
                if error < best:
                    best = error
        scores.append((best, symbol))
    return sorted(scores)


def _restored(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float32)
    tile = _locate_tile(rgb)
    return _box_average(_restore_spiral(tile))


def confidence_margin(image: Image.Image) -> float:
    blurred = _restored(image)
    if float(np.percentile(blurred, 95) - np.percentile(blurred, 5)) < 8.0:
        raise ValueError("restored raster has no measurable symbol contrast")
    scores = _scores(blurred)
    if len(scores) < 2:
        raise ValueError("not enough symbol templates")
    return float(scores[1][0] - scores[0][0])


def decision_from_image(image: Image.Image) -> str:
    """Recover one symbol using only the final RGB image."""
    blurred = _restored(image)
    if float(np.percentile(blurred, 95) - np.percentile(blurred, 5)) < 8.0:
        raise ValueError("restored raster has no measurable symbol contrast")
    scores = _scores(blurred)
    if len(scores) < 2 or scores[1][0] - scores[0][0] < 0.0015:
        raise ValueError("ambiguous restored symbol")
    return scores[0][1]
