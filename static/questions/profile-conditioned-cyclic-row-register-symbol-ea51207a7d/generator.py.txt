"""Deterministic latent sampler and raster renderer for cyclic row registration."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

N = 9
SYMBOLS = ("slash", "backslash", "left_chevron", "right_chevron")

# One payload tile per row.  All four classes therefore have identical target
# ink, row occupancy, component count, and unregistered-position marginals.
TEMPLATES: dict[str, tuple[int, ...]] = {
    "slash": (8, 7, 6, 5, 4, 3, 2, 1, 0),
    "backslash": (0, 1, 2, 3, 4, 5, 6, 7, 8),
    "left_chevron": (8, 8, 7, 6, 1, 6, 7, 8, 8),
    "right_chevron": (0, 0, 1, 2, 7, 2, 1, 0, 0),
}

PALETTES = (
    {"payload": (29, 57, 92), "marker": (231, 105, 32), "rail": (214, 166, 42)},
    {"payload": (34, 72, 108), "marker": (220, 91, 37), "rail": (202, 151, 32)},
)
BACKGROUNDS = ((248, 246, 239), (239, 246, 247))
GRID = (137, 148, 158)
BAND_A = (229, 233, 232)
BAND_B = (218, 225, 226)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene using only the supplied integer seed."""
    rng = random.Random((int(seed) ^ 0x5EED_C1C1) & 0xFFFFFFFFFFFFFFFF)
    block = int(seed) // 4
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "shifts": [rng.randrange(N) for _ in range(N)],
        "cell": (30, 32, 34, 36)[block % 4],
        "origin_x": (82, 88, 94)[block % 3],
        "origin_y": (52, 58, 64)[(block // 3) % 3],
        "palette_index": block % len(PALETTES),
        "background_index": (block // 2) % len(BACKGROUNDS),
    }


def _payload_columns(scene: dict) -> tuple[int, ...]:
    # Reading symbol here makes the label input visibly consequential; the
    # protected field audit follows this helper from render().
    return TEMPLATES[scene["symbol"]]


def render(scene: dict) -> Image.Image:
    """Render only public visual evidence; all shifts are periodic modulo nine."""
    symbol_columns = _payload_columns(scene)
    shifts = [int(value) % N for value in scene["shifts"]]
    if len(shifts) != N:
        raise ValueError("exactly nine row phases are required")
    cell = int(scene["cell"])
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    palette = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    background = BACKGROUNDS[int(scene["background_index"]) % len(BACKGROUNDS)]

    image = Image.new("RGB", (540, 440), background)
    draw = ImageDraw.Draw(image)
    board_right = ox + N * cell
    board_bottom = oy + N * cell

    # Alternating strip fill makes row boundaries easy to see without carrying
    # any answer information.
    for row in range(N):
        y0 = oy + row * cell
        draw.rectangle(
            (ox, y0, board_right, y0 + cell),
            fill=BAND_A if row % 2 == 0 else BAND_B,
        )

    # Fixed neutral grid is also the oracle's visible geometric calibration.
    for column in range(N + 1):
        x = ox + column * cell
        draw.line((x, oy, x, board_bottom), fill=GRID, width=2)
    for row in range(N + 1):
        y = oy + row * cell
        draw.line((ox, y, board_right, y), fill=GRID, width=2)

    # The gold rail visibly names the destination (the first cyclic cell).
    rail_x = ox - max(7, cell // 4)
    draw.rounded_rectangle(
        (rail_x - 3, oy, rail_x + 3, board_bottom), radius=3, fill=palette["rail"]
    )
    for row in range(N):
        cy = oy + row * cell + int(round(cell * 0.27))
        draw.line((rail_x, cy, ox - 2, cy), fill=palette["rail"], width=2)

    for row, (phase, canonical_column) in enumerate(zip(shifts, symbol_columns)):
        y0 = oy + row * cell
        marker_column = phase
        payload_column = (canonical_column + phase) % N

        # Marker lane: a compact upward orange triangle.
        mx = ox + marker_column * cell + cell // 2
        my = y0 + int(round(cell * 0.27))
        marker_half = max(4, cell // 6)
        draw.polygon(
            ((mx, my - marker_half), (mx - marker_half, my + marker_half),
             (mx + marker_half, my + marker_half)),
            fill=palette["marker"],
        )

        # Payload lane: a navy square, vertically separate even on marker overlap.
        px = ox + payload_column * cell + cell // 2
        py = y0 + int(round(cell * 0.70))
        half = max(5, cell // 5)
        draw.rounded_rectangle((px - half, py - half, px + half, py + half),
                               radius=2, fill=palette["payload"])

    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def margin(scene: dict) -> float:
    """Runner-up Hamming gap for a perfectly registered latent template."""
    target = TEMPLATES[scene["symbol"]]
    rival_scores = [
        sum(left != right for left, right in zip(target, candidate))
        for name, candidate in TEMPLATES.items()
        if name != scene["symbol"]
    ]
    return float(min(rival_scores))


def is_quarantined(scene: dict) -> bool:
    # Valid generator scenes have an eight-row or larger template gap and robust
    # cell sizes.  Image corruption is handled by oracle abstention (exception).
    return margin(scene) < 4.0 or int(scene["cell"]) < 24


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the genuine periodic alias in the phase parametrization."""
    twin = deepcopy(scene)
    twin["shifts"] = [int(value) + N for value in scene["shifts"]]
    return [("add_one_full_cycle_to_every_phase", twin)]
