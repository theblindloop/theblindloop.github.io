"""Renderer and analytic arm for the skip-step badge trail."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 600
BADGE_COUNT = 12
ANSWER_COLORS = ("cyan", "amber", "rose")
RGB = {
    "cyan": (25, 171, 190),
    "amber": (235, 166, 35),
    "rose": (211, 68, 112),
}
BACKGROUNDS = ((248, 247, 242), (244, 248, 251), (251, 246, 249))
ROUTE_COLORS = ((89, 70, 155), (77, 73, 154), (98, 65, 147))
INK = (38, 42, 54)
START = (42, 157, 82)
END = (215, 61, 52)

# Distances along a 1080-pixel hairpin. None falls on a corner.
STATION_DISTANCES = (55, 140, 230, 315, 445, 530, 615, 700, 815, 885, 950, 1020)


def _terminal_index(commands: Iterable[int]) -> int:
    values = [int(value) for value in commands]
    if len(values) != BADGE_COUNT or any(value not in (1, 2) for value in values):
        raise ValueError("commands must be twelve one- or two-step instructions")
    index = 0
    while index + values[index] < len(values):
        index += values[index]
    return index


def _visited_indices(commands: list[int]) -> list[int]:
    visited = [0]
    index = 0
    while index + commands[index] < len(commands):
        index += commands[index]
        visited.append(index)
    return visited


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, class-balanced scene from an integer seed."""

    rng = random.Random(int(seed) ^ 0x5A17C3)
    for _ in range(10_000):
        commands = [rng.choice((1, 2)) for _ in range(BADGE_COUNT)]
        visited = _visited_indices(commands)
        skipped = BADGE_COUNT - len(visited)
        if min(commands.count(1), commands.count(2)) >= 4 and len(visited) >= 7 and skipped >= 3:
            break
    else:
        raise RuntimeError("could not sample a well-conditioned jump program")

    # Every image contains exactly four badges of each answer color. The desired
    # terminal class cycles with the public seed, then is installed only by a swap.
    colors = [color for color in ANSWER_COLORS for _ in range(4)]
    rng.shuffle(colors)
    terminal = _terminal_index(commands)
    target = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    if colors[terminal] != target:
        replacement = next(i for i, color in enumerate(colors) if color == target and i != terminal)
        colors[terminal], colors[replacement] = colors[replacement], colors[terminal]

    return {
        "commands": commands,
        "colors": colors,
        "route_variant": (int(seed) // 3) % 4,
        "style_variant": (int(seed) // 12) % 3,
    }


def analytic_gold(scene: dict) -> str:
    terminal = _terminal_index(scene["commands"])
    return str(scene["colors"][terminal])


def _edit_margin(commands: list[int], colors: list[str]) -> int:
    original = colors[_terminal_index(commands)]
    for distance in (1, 2):
        for indices in itertools.combinations(range(BADGE_COUNT), distance):
            changed = list(commands)
            for index in indices:
                changed[index] = 3 - changed[index]
            if colors[_terminal_index(changed)] != original:
                return distance
    return 3


def margin(scene: dict) -> float:
    """Runner-up margin: minimum command-symbol edits changing the answer."""

    return float(_edit_margin(list(scene["commands"]), list(scene["colors"])))


def is_quarantined(scene: dict) -> bool:
    try:
        commands = list(scene["commands"])
        colors = list(scene["colors"])
        return not (
            len(commands) == BADGE_COUNT
            and len(colors) == BADGE_COUNT
            and all(value in (1, 2) for value in commands)
            and all(value in ANSWER_COLORS for value in colors)
            and all(colors.count(color) == 4 for color in ANSWER_COLORS)
            and min(commands.count(1), commands.count(2)) >= 4
            and margin(scene) >= 1.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)
    except OSError:
        return ImageFont.load_default()


def _route_points(variant: int) -> list[tuple[int, int]]:
    points = [(220, 270), (580, 270), (580, 320), (260, 320), (260, 370), (560, 370)]
    value = int(variant) % 4
    if value & 1:
        points = [(WIDTH - x, y) for x, y in points]
    if value & 2:
        points = [(x, 640 - y) for x, y in points]
    return points


def _style(variant: int) -> tuple[tuple[int, int, int], tuple[int, int, int]]:
    value = int(variant) % 3
    return BACKGROUNDS[value], ROUTE_COLORS[value]


def _point_at(points: list[tuple[int, int]], distance: float) -> tuple[int, int]:
    remaining = float(distance)
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        length = abs(x1 - x0) + abs(y1 - y0)
        if remaining <= length:
            fraction = remaining / length
            return (round(x0 + (x1 - x0) * fraction), round(y0 + (y1 - y0) * fraction))
        remaining -= length
    return points[-1]


def _draw_badge(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    command: int,
    color: tuple[int, int, int],
    route_color: tuple[int, int, int],
) -> None:
    x, y = center
    radius = 15
    if command == 1:
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            fill=color,
            outline=route_color,
            width=4,
        )
    else:
        diamond = [(x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)]
        draw.polygon(diamond, fill=color)
        draw.line(diamond + [diamond[0]], fill=route_color, width=4, joint="curve")


def render(scene: dict) -> Image.Image:
    commands = [int(value) for value in scene["commands"]]
    colors = [str(value) for value in scene["colors"]]
    points = _route_points(scene["route_variant"])
    background, route_color = _style(scene["style_variant"])
    if len(commands) != BADGE_COUNT or len(colors) != BADGE_COUNT:
        raise ValueError("a scene must contain twelve commands and colors")

    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((30, 24, 770, 112), radius=20, fill=(255, 255, 255), outline=(219, 216, 226), width=2)
    draw.text((55, 39), "SKIP-STEP BADGE TRAIL", font=_font(23, True), fill=INK)
    draw.text(
        (55, 75),
        "Begin at START. Obey only badges you land on; stop before an overshoot.",
        font=_font(18),
        fill=INK,
    )

    draw.line(points, fill=route_color, width=7, joint="curve")
    stations = [_point_at(points, distance) for distance in STATION_DISTANCES]
    for center, command, color in zip(stations, commands, colors):
        _draw_badge(draw, center, command, RGB[color], route_color)

    for center, fill in ((points[0], START), (points[-1], END)):
        x, y = center
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=fill, outline=route_color, width=4)

    sx, sy = points[0]
    ex, ey = points[-1]
    draw.text((sx - 34, sy - 46), "START", font=_font(15, True), fill=INK)
    draw.text((ex - 20, ey + 25), "END", font=_font(15, True), fill=INK)

    draw.rounded_rectangle((145, 455, 655, 545), radius=18, fill=(255, 255, 255), outline=(219, 216, 226), width=2)
    neutral = (189, 191, 199)
    _draw_badge(draw, (205, 500), 1, neutral, route_color)
    draw.text((228, 487), "jump 1", font=_font(18, True), fill=INK)
    _draw_badge(draw, (390, 500), 2, neutral, route_color)
    draw.text((413, 487), "jump 2", font=_font(18, True), fill=INK)
    draw.text((520, 487), "fill = answer", font=_font(18, True), fill=INK)
    return image
