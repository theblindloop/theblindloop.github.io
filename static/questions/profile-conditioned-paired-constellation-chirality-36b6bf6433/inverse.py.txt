"""Independent image-only inverse arm for paired dot constellations."""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

N_POINTS = 7
_PERMUTATIONS = np.asarray(list(itertools.permutations(range(N_POINTS))), dtype=np.int16)


def _component_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 50:
            centers.append((float(np.mean(xs)), float(np.mean(ys))))
    return centers


def _signatures(points: np.ndarray) -> np.ndarray:
    distances = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=2)
    scale = float(np.median(distances[np.triu_indices(len(points), 1)]))
    if scale <= 0:
        raise ValueError("degenerate point set")
    return np.sort(distances, axis=1)[:, 1:] / scale


def _match(left: np.ndarray, right: np.ndarray) -> np.ndarray:
    a = _signatures(left)
    b = _signatures(right)
    cost = ((a[:, None, :] - b[None, :, :]) ** 2).sum(axis=2)
    scores = cost[np.arange(N_POINTS)[None, :], _PERMUTATIONS].sum(axis=1)
    return _PERMUTATIONS[int(np.argmin(scores))]


def _fit_error(left: np.ndarray, right: np.ndarray, determinant: float) -> float:
    left = left - left.mean(axis=0)
    right = right - right.mean(axis=0)
    left /= math.sqrt(float((left * left).sum() / len(left)))
    right /= math.sqrt(float((right * right).sum() / len(right)))
    u, _, vt = np.linalg.svd(left.T @ right)
    base_det = 1.0 if np.linalg.det(u @ vt) >= 0 else -1.0
    correction = np.eye(2)
    correction[-1, -1] = determinant * base_det
    residual = left @ (u @ correction @ vt) - right
    return math.sqrt(float((residual * residual).sum() / len(left)))


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    black = (pixels[:, :, 0] < 80) & (pixels[:, :, 1] < 80) & (pixels[:, :, 2] < 80)
    centers = _component_centers(black)
    middle = image.width / 2.0
    left = np.asarray(sorted((point for point in centers if point[0] < middle)), dtype=np.float64)
    right = np.asarray(sorted((point for point in centers if point[0] > middle)), dtype=np.float64)
    if left.shape != (N_POINTS, 2) or right.shape != (N_POINTS, 2):
        raise ValueError(f"expected {N_POINTS} dot components per side, found {len(left)} and {len(right)}")
    right = right[_match(left, right)]
    rotation_error = _fit_error(left, right, 1.0)
    mirror_error = _fit_error(left, right, -1.0)
    return "rotated" if rotation_error < mirror_error else "mirrored"
