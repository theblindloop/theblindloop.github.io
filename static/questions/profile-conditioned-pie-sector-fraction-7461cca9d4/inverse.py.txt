"""Independent pixel-only inverse arm for the pie-sector-fraction world.

This module receives ONLY the rasterized PNG (plus the world's fixed color naming
convention) and re-measures which colored sector spans the largest central angle.
It shares no geometry with renderer.render() and never imports the renderer,
prompts, generator, verifier, or any analytical gold code.

Because all three sectors live in the same disk (identical radius), each sector's
on-screen pixel count is directly proportional to its central angle. The oracle
therefore labels every non-background pixel by its nearest palette color and
returns the color owning the most colored pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "pie_sector_fraction-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (210, 40, 45),
    "blue": (40, 80, 200),
    "green": (35, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (sector colors + background).

    The neutral gray hub and any antialias blends stay dominated by either a
    sector color or the background, so the labelled masks stay clean.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = (
        arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    )
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the sector with the most colored pixels (largest angle)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    counts = []
    for idx, color in enumerate(PALETTE):
        counts.append((int(np.count_nonzero(labels == idx)), color))
    counts.sort(key=lambda item: item[0], reverse=True)
    return counts[0][1]
