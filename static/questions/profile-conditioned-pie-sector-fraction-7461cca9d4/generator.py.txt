"""Pie-sector fraction world: which colored sector spans the largest central angle.

Latent scene z carries a single filled pie chart: a full disk partitioned into
three colored sectors (red, blue, green) that together tile all 360 degrees.
The task is a measurement of a recovered scalar quantity -- the central angle
(size) of each sector -- and a set-argmax selection: find the color whose
sector subtends the LARGEST central angle (the biggest slice). Because all
sectors share the same disk radius, the angular fraction equals the on-screen
pixel-area fraction, so the answer is recovered from the raster by
color-segmenting the disk and counting each sector's colored pixels.

This world counterbalances which color owns the largest sector, the order in
which the sectors are laid out around the circle, and the absolute angular
position of the largest slice.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "pie_sector_fraction-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (210, 40, 45),
    "blue": (40, 80, 200),
    "green": (35, 150, 70),
}
BACKGROUND = (250, 250, 250)
HUB_RGB = (195, 195, 195)

WIDTH = 480
HEIGHT = 340
CENTER = (240, 170)
RADIUS = 135
HUB_RADIUS = 10

# Minimum top-two sweep (central-angle) gap required for a decidable selection.
MIN_MARGIN_DEG = 4.0
# Below this top-two sweep gap the scene is flagged as a near-boundary,
# finite-resolution case (still decidable, well above pixel-count noise).
QUARANTINE_DEG = 10.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic pie scene with a guaranteed, counterbalanced largest sector.

    Returns a dict with ``sectors`` (list of {"color","sweep","start"}) where the
    three sweeps sum to 360 degrees.
    """
    rng = random.Random(seed * 1049 + 11)
    while True:
        raw = [rng.uniform(6.0, 340.0) for _ in range(3)]
        total = sum(raw)
        sweeps = [x * 360.0 / total for x in raw]
        ordered = sorted(sweeps, reverse=True)
        if ordered[0] - ordered[1] >= MIN_MARGIN_DEG:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    base_offset = rng.uniform(0.0, 360.0)
    sectors = []
    start = base_offset
    for color, sweep in zip(colors, sweeps):
        sectors.append({"color": color, "sweep": float(sweep), "start": start})
        start += sweep
    return {
        "sectors": sectors,
        "width": WIDTH,
        "height": HEIGHT,
        "center": list(CENTER),
        "radius": RADIUS,
        "hub_radius": HUB_RADIUS,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored sectors tile a disk around a neutral hub."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    cx, cy = scene["center"]
    radius = scene["radius"]
    bbox = [cx - radius, cy - radius, cx + radius, cy + radius]
    for sector in scene["sectors"]:
        draw.pieslice(
            bbox,
            start=sector["start"],
            end=sector["start"] + sector["sweep"],
            fill=COLOR_RGB[sector["color"]],
        )
    hr = scene["hub_radius"]
    draw.ellipse([cx - hr, cy - hr, cx + hr, cy + hr], fill=HUB_RGB)
    return img


def _largest_sector(sectors: list[dict]) -> tuple[dict, float]:
    """Return (largest sector, top-two central-angle gap in degrees)."""
    ordered = sorted(sectors, key=lambda s: s["sweep"], reverse=True)
    gap = ordered[0]["sweep"] - ordered[1]["sweep"]
    return ordered[0], gap


def analytic_gold(scene: dict) -> str:
    """The color of the sector with the largest central angle (sweep)."""
    sectors = scene["sectors"]
    largest, _gap = _largest_sector(sectors)
    return largest["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two central-angle gap in degrees."""
    sectors = scene["sectors"]
    _largest, gap = _largest_sector(sectors)
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two sweep gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_DEG


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Swapping two sectors' color labels changes which color is the largest and
    therefore changes the answer, and any global rotation or reflection of the
    disk turns the raster into a distinct image (and can move which color is
    largest in a way that changes the gold). Every latent field (the sector
    colors and their sweeps/angles) is directly rendered as visible ink, so the
    parametrisation has no hidden label alias.
    """
    return []
