"""Renderer and analytic arm for Scaled-Fraction Tag Transfer."""
from __future__ import annotations

import copy
import random
from math import gcd
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 680
BG = (255, 255, 255)
CARD = (244, 247, 252)
INK = (35, 43, 54)
TEAL = (31, 145, 142)
TAGS = ("red", "blue", "gold")
TAG_RGB = {
    "red": (218, 62, 67),
    "blue": (52, 108, 196),
    "gold": (224, 166, 45),
}

# Single-digit reduced references with at least one nontrivial single-digit scale.
REFERENCE_SCALES = (
    ((1, 2), (2, 3, 4)),
    ((1, 3), (2, 3)),
    ((2, 3), (2, 3)),
    ((1, 4), (2,)),
    ((3, 4), (2,)),
)

SEGMENTS = {
    "a": (10, 0, 46, 8),
    "b": (48, 10, 56, 30),
    "c": (48, 42, 56, 62),
    "d": (10, 64, 46, 72),
    "e": (0, 42, 8, 62),
    "f": (0, 10, 8, 30),
    "g": (10, 32, 46, 40),
}
DIGIT_SEGMENTS = {
    1: "bc", 2: "abdeg", 3: "abcdg", 4: "bcfg", 5: "acdfg",
    6: "acdefg", 7: "abc", 8: "abcdefg", 9: "abcdfg",
}


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    reference, valid_scales = REFERENCE_SCALES[rng.randrange(len(REFERENCE_SCALES))]
    scale = valid_scales[rng.randrange(len(valid_scales))]
    numerator, denominator = reference[0] * scale, reference[1] * scale

    den_steps = [step for step in (-1, 1) if 1 <= denominator + step <= 9]
    num_steps = [step for step in (-1, 1) if 1 <= numerator + step <= 9]
    rng.shuffle(den_steps)
    rng.shuffle(num_steps)
    distractors = [(numerator, denominator + den_steps[0]), (numerator + num_steps[0], denominator)]
    rng.shuffle(distractors)

    match_index = seed % 3
    candidates = list(distractors)
    candidates.insert(match_index, (numerator, denominator))

    answer_tag = TAGS[(seed // 3) % 3]
    other_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(other_tags)
    tag_order = list(other_tags)
    tag_order.insert(match_index, answer_tag)
    return {
        "seed": seed,
        "reference": list(reference),
        "candidates": [list(pair) for pair in candidates],
        "tag_order": tag_order,
        "stroke_order": list(SEGMENTS),
    }


def _draw_digit(draw: ImageDraw.ImageDraw, digit: int, center_x: int, top: int, order: list[str]) -> None:
    present = set(DIGIT_SEGMENTS[int(digit)])
    for name in order:
        if name in present:
            x0, y0, x1, y1 = SEGMENTS[name]
            draw.rounded_rectangle((center_x - 28 + x0, top + y0, center_x - 28 + x1, top + y1), radius=2, fill=INK)


def _draw_fraction(draw: ImageDraw.ImageDraw, pair: list[int], center_x: int, bar_y: int, order: list[str]) -> None:
    _draw_digit(draw, int(pair[0]), center_x, bar_y - 96, order)
    draw.rounded_rectangle((center_x - 52, bar_y - 5, center_x + 52, bar_y + 5), radius=4, fill=INK)
    _draw_digit(draw, int(pair[1]), center_x, bar_y + 20, order)


def render(scene: dict) -> Image.Image:
    reference = scene["reference"]
    candidates = scene["candidates"]
    tag_order = scene["tag_order"]
    stroke_order = scene["stroke_order"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=23)
    draw.text((480, 23), "REFERENCE", fill=TEAL, anchor="mm", font=font)
    ref_box = (330, 48, 630, 292)
    draw.rounded_rectangle(ref_box, radius=18, fill=CARD, outline=TEAL, width=6)
    _draw_fraction(draw, reference, 480, 174, stroke_order)

    draw.text((480, 330), "TAGGED CANDIDATES", fill=INK, anchor="mm", font=font)
    boxes = ((30, 360, 300, 655), (345, 360, 615, 655), (660, 360, 930, 655))
    for box, pair, tag in zip(boxes, candidates, tag_order):
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=INK, width=5)
        cx = (box[0] + box[2]) // 2
        _draw_fraction(draw, pair, cx, 482, stroke_order)
        draw.ellipse((cx - 25, 610, cx + 25, 660), fill=TAG_RGB[tag], outline=INK, width=3)
    return image


def _reduced(pair: list[int]) -> tuple[int, int]:
    g = gcd(int(pair[0]), int(pair[1]))
    return int(pair[0]) // g, int(pair[1]) // g


def _scores(scene: dict) -> list[int]:
    reference = scene["reference"]
    candidates = scene["candidates"]
    _ = scene["tag_order"]
    target = _reduced(reference)
    return [2 if _reduced(pair) == target else 1 if pair[0] == reference[0] or pair[1] == reference[1] else 0 for pair in candidates]


def analytic_gold(scene: dict) -> str:
    reference = scene["reference"]
    candidates = scene["candidates"]
    tag_order = scene["tag_order"]
    target = _reduced(reference)
    matches = [index for index, pair in enumerate(candidates) if _reduced(pair) == target]
    if len(matches) != 1:
        return "abstain"
    return tag_order[matches[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    reference = scene["reference"]
    candidates = scene["candidates"]
    _ = scene["tag_order"]
    target = _reduced(reference)
    return sum(_reduced(pair) == target for pair in candidates) != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["stroke_order"] = list(reversed(twin["stroke_order"]))
    return [("reverse_disjoint_segment_draw_order", twin)]
