"""Independent PNG-only inverse for the annulus hole/rim comparison."""

from __future__ import annotations

import numpy as np
from PIL import Image


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate((array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    mask = np.linalg.norm(array - background, axis=2) > 38.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 260:
        raise ValueError("missing or materially erased annulus")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (25 <= width <= 31 and 25 <= height <= 31 and abs(width - height) <= 1):
        raise ValueError("invalid outer circular boundary")
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    if mask[cy, cx]:
        raise ValueError("central hole is absent or corrupted")

    # Measure four directly visible center-to-ink transitions. Pillow's ellipse
    # rasterization includes both endpoints, so the median transition distance
    # is the inner radius plus one and is converted back below.
    transitions = []
    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        found = None
        for distance in range(1, 13):
            if mask[cy + dy * distance, cx + dx * distance]:
                found = distance
                break
        if found is None:
            raise ValueError("open or damaged central hole")
        transitions.append(found)
    if max(transitions) - min(transitions) > 1:
        raise ValueError("noncircular inner boundary")
    inner_radius = float(np.median(transitions) - 1.0)
    outer_radius = float((width - 1) / 2.0)
    signed = 3.0 * inner_radius - outer_radius
    if abs(signed) < 3.0:
        raise ValueError("raster measurement is too close to equality")
    return "yes" if signed > 0.0 else "no"
