"""Deterministic sampler and renderer for a compact annular measurement glyph."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    outer_radius = rng.choice((13, 14, 15))
    positive = int(seed) % 2 == 0
    if positive:
        inner_radius = rng.choice((7, 8))
    else:
        inner_radius = rng.choice((2, 3))
    # Centers coincide with fine-grid cell centers but vary widely on canvas.
    # With R < 16 the whole cue belongs only to the centered overlapping tile.
    center_x = rng.choice((96, 160, 224, 288, 352, 416))
    center_y = rng.choice((96, 160, 224, 288, 352, 416))
    inks = ([22, 126, 137], [201, 82, 94], [76, 99, 173], [195, 118, 30])
    return {
        "width": 512,
        "height": 512,
        "center_x": center_x,
        "center_y": center_y,
        "outer_radius": outer_radius,
        "inner_radius": inner_radius,
        "background": [248, 249, 251],
        "ink": list(rng.choice(inks)),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx, cy = int(scene["center_x"]), int(scene["center_y"])
    outer, inner = int(scene["outer_radius"]), int(scene["inner_radius"])
    draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=tuple(scene["ink"]))
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=tuple(scene["background"]))
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if 2 * int(scene["inner_radius"]) > int(scene["outer_radius"]) - int(scene["inner_radius"]) else "no"


def margin(scene: dict) -> float:
    return float(3 * int(scene["inner_radius"]) - int(scene["outer_radius"]))


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
