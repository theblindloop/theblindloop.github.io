"""Deterministic renderer and analytic side for paired heatmap contrast."""

from __future__ import annotations

import random
from typing import Sequence

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "adjacent_heat_contrast-renderer-0.1.0"
WIDTH = 1040
HEIGHT = 600
SCALE = 2
ROWS = 4
COLUMNS = 6
LEVELS = 16
PANEL_BOXES = {"A": (55, 72, 500, 548), "B": (540, 72, 985, 548)}
GRID_BOXES = {"A": (132, 158, 452, 398), "B": (617, 158, 937, 398)}
LEGEND_BOX = (354, 470, 686, 488)
QUARANTINE_GAP = 2.0
_HEAT_STOPS = ((27, 69, 142), (35, 169, 190), (250, 214, 64))
_LABEL_FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
_TEXT_FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _heat_color(level: int) -> tuple[int, int, int]:
    """Map one of the public discrete heat levels to the shared RGB scale."""

    value = max(0, min(LEVELS - 1, int(level))) / float(LEVELS - 1)
    if value <= 0.5:
        local = value * 2.0
        left, right = _HEAT_STOPS[0], _HEAT_STOPS[1]
    else:
        local = (value - 0.5) * 2.0
        left, right = _HEAT_STOPS[1], _HEAT_STOPS[2]
    return tuple(round(left[index] + local * (right[index] - left[index])) for index in range(3))


PALETTE = tuple(_heat_color(level) for level in range(LEVELS))


def _font(path: str, size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(path, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _score(matrix: Sequence[Sequence[int]]) -> float:
    differences = [
        abs(int(row[column + 1]) - int(row[column]))
        for row in matrix
        for column in range(COLUMNS - 1)
    ]
    return sum(differences) / float(len(differences))


def _random_permutation(rng: random.Random, values: Sequence[int]) -> list[list[int]]:
    order = list(values)
    rng.shuffle(order)
    return [order[row * COLUMNS : (row + 1) * COLUMNS] for row in range(ROWS)]


def _sample_pair(rng: random.Random, near: bool) -> tuple[list[list[int]], list[list[int]]]:
    """Find two arrangements of one multiset with a controlled score gap."""

    for _ in range(120):
        values = [rng.randrange(LEVELS) for _ in range(ROWS * COLUMNS)]
        if max(values) - min(values) < 9:
            values[0], values[-1] = 0, LEVELS - 1
        candidates: list[tuple[float, list[list[int]]]] = []
        for _ in range(260):
            matrix = _random_permutation(rng, values)
            candidates.append((_score(matrix), matrix))
        candidates.sort(key=lambda item: item[0])
        if near:
            for low_index, (low_score, low_matrix) in enumerate(candidates):
                for high_score, high_matrix in candidates[low_index + 1 : low_index + 25]:
                    if 0.75 <= high_score - low_score < QUARANTINE_GAP:
                        return low_matrix, high_matrix
        else:
            low_score, low_matrix = candidates[0]
            high_score, high_matrix = candidates[-1]
            if high_score - low_score >= 3.0:
                return low_matrix, high_matrix
    raise RuntimeError("could not sample a controlled heat-contrast pair")


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair with matched heat histograms."""

    rng = random.Random(int(seed))
    near = int(seed) % 13 == 0
    low, high = _sample_pair(rng, near=near)
    if rng.randrange(2) == 0:
        matrix_a, matrix_b = high, low
    else:
        matrix_a, matrix_b = low, high
    # Independent row shuffles are label-preserving for the all-row average.
    for matrix in (matrix_a, matrix_b):
        rng.shuffle(matrix)
        if rng.randrange(2):
            for row in matrix:
                row.reverse()
    return {
        "matrix_a": [[int(value) for value in row] for row in matrix_a],
        "matrix_b": [[int(value) for value in row] for row in matrix_b],
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel: str,
    matrix: Sequence[Sequence[int]],
    label_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    text_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
) -> None:
    px0, py0, px1, py1 = PANEL_BOXES[panel]
    gx0, gy0, gx1, gy1 = GRID_BOXES[panel]
    draw.rounded_rectangle(
        (px0 * SCALE, py0 * SCALE, px1 * SCALE, py1 * SCALE),
        radius=14 * SCALE,
        fill=(249, 251, 254),
        outline=(190, 199, 211),
        width=2 * SCALE,
    )
    title = f"Panel {panel}"
    title_box = draw.textbbox((0, 0), title, font=label_font)
    draw.text(
        (((px0 + px1) * SCALE - (title_box[2] - title_box[0])) // 2, 96 * SCALE),
        title,
        fill=(35, 47, 62),
        font=label_font,
    )
    cell_width = (gx1 - gx0) / COLUMNS
    cell_height = (gy1 - gy0) / ROWS
    for row in range(ROWS):
        cy = gy0 + (row + 0.5) * cell_height
        row_label = f"R{row + 1}"
        box = draw.textbbox((0, 0), row_label, font=text_font)
        draw.text(
            (round((gx0 - 42) * SCALE - (box[2] - box[0]) / 2), round((cy - 8) * SCALE)),
            row_label,
            fill=(83, 96, 111),
            font=text_font,
        )
        for column in range(COLUMNS):
            left = gx0 + column * cell_width
            top = gy0 + row * cell_height
            right = gx0 + (column + 1) * cell_width
            bottom = gy0 + (row + 1) * cell_height
            draw.rectangle(
                (round(left * SCALE), round(top * SCALE), round(right * SCALE), round(bottom * SCALE)),
                fill=PALETTE[int(matrix[row][column])],
                outline=(255, 255, 255),
                width=2 * SCALE,
            )
    for column in range(COLUMNS):
        cx = gx0 + (column + 0.5) * cell_width
        label = str(column + 1)
        box = draw.textbbox((0, 0), label, font=text_font)
        draw.text(
            (round(cx * SCALE - (box[2] - box[0]) / 2), round((gy1 + 13) * SCALE)),
            label,
            fill=(83, 96, 111),
            font=text_font,
        )


def _draw_legend(draw: ImageDraw.ImageDraw, text_font: ImageFont.FreeTypeFont | ImageFont.ImageFont) -> None:
    x0, y0, x1, y1 = LEGEND_BOX
    cell_width = (x1 - x0) / LEVELS
    for level, color in enumerate(PALETTE):
        draw.rectangle(
            (round((x0 + level * cell_width) * SCALE), round(y0 * SCALE), round((x0 + (level + 1) * cell_width) * SCALE), round(y1 * SCALE)),
            fill=color,
        )
    draw.rectangle((x0 * SCALE, y0 * SCALE, x1 * SCALE, y1 * SCALE), outline=(70, 80, 94), width=SCALE)
    draw.text((round((x0 - 38) * SCALE), round((y0 - 2) * SCALE)), "low", fill=(83, 96, 111), font=text_font)
    draw.text((round((x1 + 8) * SCALE), round((y0 - 2) * SCALE)), "high", fill=(83, 96, 111), font=text_font)
    draw.text((round(475 * SCALE), round((y1 + 9) * SCALE)), "shared heat scale", fill=(83, 96, 111), font=text_font)


def render(scene: dict) -> Image.Image:
    """Render the visible paired heatmaps as an RGB PNG image."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    label_font = _font(_LABEL_FONT, 25)
    text_font = _font(_TEXT_FONT, 14)
    _draw_panel(draw, "A", scene["matrix_a"], label_font, text_font)
    _draw_panel(draw, "B", scene["matrix_b"], label_font, text_font)
    _draw_legend(draw, text_font)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the panel with the larger mean adjacent-column heat difference."""

    score_a = _score(scene["matrix_a"])
    score_b = _score(scene["matrix_b"])
    if score_a == score_b:
        return "tie"
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    """Signed heat-level difference: A score minus B score."""

    return _score(scene["matrix_a"]) - _score(scene["matrix_b"])


def is_quarantined(scene: dict) -> bool:
    """Mark exact and finite-resolution near ties as unusable headline cases."""

    return abs(margin(scene)) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden render-invariant latent alias is declared for this world."""

    return []
