"""Independent image-only inverse for paired heatmap contrast.

The decoder repeats only the public page geometry and shared color-scale values.
It receives a final RGB image, samples the center of each visible cell, decodes
the nearest shared-scale level, and recomputes the declared local statistic.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


WIDTH = 1040
HEIGHT = 600
ROWS = 4
COLUMNS = 6
LEVELS = 16
GRID_BOXES = ((132, 158, 452, 398), (617, 158, 937, 398))
QUARANTINE_GAP = 2.0
HEAT_STOPS = ((27, 69, 142), (35, 169, 190), (250, 214, 64))


def _palette() -> np.ndarray:
    values = []
    for level in range(LEVELS):
        fraction = level / float(LEVELS - 1)
        if fraction <= 0.5:
            amount = fraction * 2.0
            left, right = HEAT_STOPS[0], HEAT_STOPS[1]
        else:
            amount = (fraction - 0.5) * 2.0
            left, right = HEAT_STOPS[1], HEAT_STOPS[2]
        values.append([round(left[index] + amount * (right[index] - left[index])) for index in range(3)])
    return np.asarray(values, dtype=np.float32)


_PALETTE = _palette()


def _decode_panel(rgb: np.ndarray, bounds: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = bounds
    cell_width = (x1 - x0) / COLUMNS
    cell_height = (y1 - y0) / ROWS
    matrix: list[list[int]] = []
    for row in range(ROWS):
        values: list[int] = []
        for column in range(COLUMNS):
            cx = round(x0 + (column + 0.5) * cell_width)
            cy = round(y0 + (row + 0.5) * cell_height)
            patch = rgb[max(0, cy - 2) : cy + 3, max(0, cx - 2) : cx + 3]
            if patch.size == 0:
                raise ValueError("cell center is outside the image")
            pixel = np.median(patch.reshape(-1, 3), axis=0).astype(np.float32)
            distances = np.linalg.norm(_PALETTE - pixel[None, :], axis=1)
            level = int(np.argmin(distances))
            if float(distances[level]) > 38.0:
                raise ValueError("cell is missing or not on the declared shared heat scale")
            values.append(level)
        matrix.append(values)
    return matrix


def _score(matrix: list[list[int]]) -> float:
    differences = [
        abs(matrix[row][column + 1] - matrix[row][column])
        for row in range(ROWS)
        for column in range(COLUMNS - 1)
    ]
    return sum(differences) / float(len(differences))


def decision_from_image(image: Image.Image) -> str:
    """Recover A, B, or an exact tie from the final raster alone."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected image size {image.size}")
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    matrix_a = _decode_panel(rgb, GRID_BOXES[0])
    matrix_b = _decode_panel(rgb, GRID_BOXES[1])
    score_a = _score(matrix_a)
    score_b = _score(matrix_b)
    if score_a == score_b:
        return "tie"
    if abs(score_a - score_b) < 0.5:
        raise ValueError("decoded contrast values are too close to classify")
    return "A" if score_a > score_b else "B"
