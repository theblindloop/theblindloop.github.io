"""Renderer and analytic gold for color-bound collinearity correspondence.

Each visible panel contains six colored disks.  Exactly one triple of centers
is collinear.  The answer is the candidate whose set of three line colors
matches the reference.  Record order is intentionally irrelevant and is used
as the declared render-invariant latent alias.
"""

from __future__ import annotations

import copy
import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored_collinearity_signature_correspondence-0.1.0"
WIDTH = 1200
HEIGHT = 520
BACKGROUND = (250, 251, 253)
CARD = (238, 243, 248)
BORDER = (166, 178, 192)
INK = (35, 45, 58)
DOT_RADIUS = 16
COLORS = ("red", "blue", "green", "gold", "purple", "orange")
PALETTE = {
    "red": (220, 58, 62),
    "blue": (45, 100, 190),
    "green": (35, 154, 85),
    "gold": (225, 155, 31),
    "purple": (137, 76, 173),
    "orange": (230, 105, 42),
}
PANEL_BOXES = {
    "reference": (24, 88, 376, 480),
    "candidate_a": (424, 88, 776, 480),
    "candidate_b": (824, 88, 1176, 480),
}
PANEL_TITLES = {"reference": "REFERENCE", "candidate_a": "A", "candidate_b": "B"}
PANEL_CENTERS = {
    key: ((box[0] + box[2]) // 2, 290) for key, box in PANEL_BOXES.items()
}

# The first three offsets form one line.  The remaining points make every
# other triple comfortably non-collinear; the smallest accidental perpendicular
# distance is much larger than the pixel tolerance used by both arms.
BASE_OFFSETS = (
    (-115, -85),
    (-5, -25),
    (105, 35),
    (-105, 85),
    (15, 75),
    (115, -75),
)
COLLINEAR_TOLERANCE = 3.5


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _rotate_offset(offset: tuple[int, int], quarter_turns: int, reflected: bool) -> tuple[int, int]:
    x, y = offset
    if reflected:
        x = -x
    for _ in range(quarter_turns % 4):
        x, y = y, -x
    return x, y


def _points_for_panel(
    panel_key: str,
    assignment: list[str],
    quarter_turns: int,
    reflected: bool,
    rng: random.Random,
) -> list[dict]:
    center_x, center_y = PANEL_CENTERS[panel_key]
    records = []
    for offset, color in zip(BASE_OFFSETS, assignment, strict=True):
        dx, dy = _rotate_offset(offset, quarter_turns, reflected)
        # A record contains only visible fields.  The tiny presentation jitter
        # is an integer translation shared by the whole panel and is rendered.
        records.append(
            {
                "color": color,
                "x": int(center_x + dx),
                "y": int(center_y + dy),
                "radius": DOT_RADIUS,
            }
        )
    rng.shuffle(records)
    return records


def _make_assignment_with_line_set(
    reference_assignment: list[str],
    target_set: set[str],
    rng: random.Random,
) -> list[str]:
    line_positions = [0, 1, 2]
    other_positions = [3, 4, 5]
    line_colors = list(target_set)
    other_colors = [color for color in COLORS if color not in target_set]
    rng.shuffle(line_colors)
    rng.shuffle(other_colors)
    assignment = list(reference_assignment)
    for position, color in zip(line_positions, line_colors, strict=True):
        assignment[position] = color
    for position, color in zip(other_positions, other_colors, strict=True):
        assignment[position] = color
    return assignment


def _wrong_line_set(target_set: set[str], rng: random.Random) -> set[str]:
    keep = list(target_set)
    outside = [color for color in COLORS if color not in target_set]
    rng.shuffle(keep)
    rng.shuffle(outside)
    return {keep[0], keep[1], outside[0]}


def _line_triples(panel: list[dict]) -> list[tuple[str, str, str]]:
    by_color = {str(record["color"]): (float(record["x"]), float(record["y"])) for record in panel}
    triples = []
    for colors in itertools.combinations(sorted(by_color), 3):
        a, b, c = (by_color[color] for color in colors)
        cross = abs((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]))
        scale = max(
            math.hypot(b[0] - a[0], b[1] - a[1]),
            math.hypot(c[0] - a[0], c[1] - a[1]),
            math.hypot(c[0] - b[0], c[1] - b[1]),
        )
        if scale > 0 and cross / scale <= COLLINEAR_TOLERANCE:
            triples.append(colors)
    return triples


def _line_signature(panel: list[dict]) -> frozenset[str]:
    triples = _line_triples(panel)
    if len(triples) != 1:
        return frozenset()
    return frozenset(triples[0])


def _membership_agreement(reference: frozenset[str], candidate: frozenset[str]) -> int:
    if len(reference) != 3 or len(candidate) != 3:
        return -1
    return sum((color in reference) == (color in candidate) for color in COLORS)


def _draw_panel(draw: ImageDraw.ImageDraw, panel_key: str, panel: list[dict]) -> None:
    left, top, right, bottom = PANEL_BOXES[panel_key]
    draw.rounded_rectangle((left, top, right, bottom), radius=18, fill=CARD, outline=BORDER, width=3)
    for record in panel:
        x = int(record["x"])
        y = int(record["y"])
        radius = int(record["radius"])
        color = PALETTE[str(record["color"])]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    """Render only the three visible panel records from a JSON scene."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(20)
    for panel_key in ("reference", "candidate_a", "candidate_b"):
        left, _top, right, _bottom = PANEL_BOXES[panel_key]
        title = PANEL_TITLES[panel_key]
        bbox = draw.textbbox((0, 0), title, font=title_font)
        draw.text(((left + right - (bbox[2] - bbox[0])) // 2, 34), title, fill=INK, font=title_font)
    # Keep the three label inputs explicit for the rendered-field audit.
    _draw_panel(draw, "reference", scene["reference"])
    _draw_panel(draw, "candidate_a", scene["candidate_a"])
    _draw_panel(draw, "candidate_b", scene["candidate_b"])
    return image


def sample_scene(seed: int) -> dict:
    """Sample a deterministic, balanced, non-quarantined scene."""

    rng = random.Random(int(seed))
    reference_assignment = list(COLORS)
    rng.shuffle(reference_assignment)
    target_set = set(reference_assignment[:3])
    correct_assignment = _make_assignment_with_line_set(reference_assignment, target_set, rng)
    wrong_set = _wrong_line_set(target_set, rng)
    wrong_assignment = _make_assignment_with_line_set(reference_assignment, wrong_set, rng)

    reference = _points_for_panel("reference", reference_assignment, 0, False, rng)
    correct_key = "candidate_a" if int(seed) % 2 == 0 else "candidate_b"
    transform_a = (rng.randrange(4), bool(rng.randrange(2)))
    transform_b = (rng.randrange(4), bool(rng.randrange(2)))
    candidate_a = _points_for_panel(
        "candidate_a",
        correct_assignment if correct_key == "candidate_a" else wrong_assignment,
        transform_a[0],
        transform_a[1],
        rng,
    )
    candidate_b = _points_for_panel(
        "candidate_b",
        correct_assignment if correct_key == "candidate_b" else wrong_assignment,
        transform_b[0],
        transform_b[1],
        rng,
    )
    return {"reference": reference, "candidate_a": candidate_a, "candidate_b": candidate_b}


def analytic_gold(scene: dict) -> str:
    """Compare complete visible color-to-line-membership signatures."""

    reference = _line_signature(scene["reference"])
    score_a = _membership_agreement(reference, _line_signature(scene["candidate_a"]))
    score_b = _membership_agreement(reference, _line_signature(scene["candidate_b"]))
    if score_a < 0 or score_b < 0 or score_a == score_b:
        return "AMBIGUOUS"
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    """Return the runner-up membership-agreement gap."""

    reference = _line_signature(scene["reference"])
    score_a = _membership_agreement(reference, _line_signature(scene["candidate_a"]))
    score_b = _membership_agreement(reference, _line_signature(scene["candidate_b"]))
    return float(abs(score_a - score_b))


def _valid_panel(panel_key: str, panel: object) -> bool:
    if not isinstance(panel, list) or len(panel) != len(COLORS):
        return False
    if {record.get("color") for record in panel if isinstance(record, dict)} != set(COLORS):
        return False
    left, top, right, bottom = PANEL_BOXES[panel_key]
    centers = []
    for record in panel:
        if not isinstance(record, dict):
            return False
        try:
            x = int(record["x"])
            y = int(record["y"])
            radius = int(record["radius"])
        except (KeyError, TypeError, ValueError):
            return False
        if radius != DOT_RADIUS or not (left + radius + 10 <= x <= right - radius - 10):
            return False
        if not (top + radius + 20 <= y <= bottom - radius - 20):
            return False
        centers.append((x, y))
    for a, b in itertools.combinations(centers, 2):
        if math.hypot(a[0] - b[0], a[1] - b[1]) <= 2 * DOT_RADIUS + 8:
            return False
    return len(_line_triples(panel)) == 1


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, clipped, non-unique, or tied scenes."""

    panels = (("reference", scene.get("reference")), ("candidate_a", scene.get("candidate_a")), ("candidate_b", scene.get("candidate_b")))
    if any(not _valid_panel(key, panel) for key, panel in panels):
        return True
    reference = _line_signature(scene["reference"])
    score_a = _membership_agreement(reference, _line_signature(scene["candidate_a"]))
    score_b = _membership_agreement(reference, _line_signature(scene["candidate_b"]))
    return score_a < 0 or score_b < 0 or score_a == score_b


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse unordered records without changing pixels or analytic gold."""

    twin = copy.deepcopy(scene)
    for key in ("reference", "candidate_a", "candidate_b"):
        twin[key] = list(reversed(twin[key]))
    return [("reverse_unordered_panel_records", twin)]
