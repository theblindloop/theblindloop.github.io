"""Independent image-only inverse for collinearity-signature correspondence.

This module repeats only the public raster contract: three fixed cards contain
one disk of each of six public colors.  It does not import or inspect the
renderer, prompts, scene dictionaries, manifests, or analytic gold.
"""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image


PALETTE = {
    "red": (220, 58, 62),
    "blue": (45, 100, 190),
    "green": (35, 154, 85),
    "gold": (225, 155, 31),
    "purple": (137, 76, 173),
    "orange": (230, 105, 42),
}
COLORS = tuple(PALETTE)
PANEL_BOXES = {
    "reference": (24, 88, 376, 480),
    "A": (424, 88, 776, 480),
    "B": (824, 88, 1176, 480),
}
LINE_TOLERANCE = 3.5
MIN_COMPONENT_PIXELS = 250


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    """Return four-connected components of an exact color mask."""

    height, width = mask.shape
    pending = mask.copy()
    components = []
    ys, xs = np.nonzero(pending)
    for start_y, start_x in zip(ys.tolist(), xs.tolist(), strict=True):
        if not pending[start_y, start_x]:
            continue
        pending[start_y, start_x] = False
        stack = [(start_y, start_x)]
        component = []
        while stack:
            y, x = stack.pop()
            component.append((y, x))
            for next_y, next_x in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= next_y < height and 0 <= next_x < width and pending[next_y, next_x]:
                    pending[next_y, next_x] = False
                    stack.append((next_y, next_x))
        components.append(component)
    return components


def _panel_centers(pixels: np.ndarray, box: tuple[int, int, int, int]) -> dict[str, tuple[float, float]]:
    left, top, right, bottom = box
    crop = pixels[top:bottom, left:right]
    centers = {}
    for color, rgb in PALETTE.items():
        mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
        components = [part for part in _components(mask) if len(part) >= MIN_COMPONENT_PIXELS]
        if len(components) != 1:
            raise ValueError(f"{color}: expected one isolated disk, got {len(components)}")
        ys = np.asarray([point[0] for point in components[0]], dtype=np.float64)
        xs = np.asarray([point[1] for point in components[0]], dtype=np.float64)
        centers[color] = (float(xs.mean() + left), float(ys.mean() + top))
    return centers


def _line_signature(centers: dict[str, tuple[float, float]]) -> frozenset[str]:
    triples = []
    for colors in itertools.combinations(COLORS, 3):
        a, b, c = (centers[color] for color in colors)
        cross = abs((b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]))
        scale = max(
            math.hypot(b[0] - a[0], b[1] - a[1]),
            math.hypot(c[0] - a[0], c[1] - a[1]),
            math.hypot(c[0] - b[0], c[1] - b[1]),
        )
        if scale > 0 and cross / scale <= LINE_TOLERANCE:
            triples.append(colors)
    if len(triples) != 1:
        raise ValueError(f"expected one collinear triple, got {len(triples)}")
    return frozenset(triples[0])


def _agreement(reference: frozenset[str], candidate: frozenset[str]) -> int:
    return sum((color in reference) == (color in candidate) for color in COLORS)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only the final RGB raster."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if pixels.shape[0] < 480 or pixels.shape[1] < 1176:
        raise ValueError(f"unexpected raster size {pixels.shape[:2]}")
    reference = _line_signature(_panel_centers(pixels, PANEL_BOXES["reference"]))
    candidate_a = _line_signature(_panel_centers(pixels, PANEL_BOXES["A"]))
    candidate_b = _line_signature(_panel_centers(pixels, PANEL_BOXES["B"]))
    score_a = _agreement(reference, candidate_a)
    score_b = _agreement(reference, candidate_b)
    if score_a == score_b:
        raise ValueError("collinearity correspondence is tied")
    return "A" if score_a > score_b else "B"
