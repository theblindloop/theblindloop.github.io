"""Independent pixel-only inverse arm for the labeled key-row task.

The arm discovers the background and key extents from the raster, then reads
each printed 5-by-7 glyph. It receives neither scene data nor coordinates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_EXPECTED_LABELS = ("A", "S", "D", "F", "G", "H", "J", "K")
_PATTERNS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "S": ("11111", "10000", "10000", "11111", "00001", "00001", "11111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "G": ("01110", "10001", "10000", "10111", "10001", "10001", "01110"),
    "H": ("10001", "10001", "10001", "11111", "10001", "10001", "10001"),
    "J": ("11111", "00100", "00100", "00100", "00100", "10100", "11000"),
    "K": ("10001", "10010", "10100", "11000", "10100", "10010", "10001"),
}
_TEMPLATES = {
    label: np.array([[value == "1" for value in row] for row in rows], dtype=bool)
    for label, rows in _PATTERNS.items()
}


class AmbiguousRasterError(ValueError):
    """The visible evidence does not support one exact label decision."""


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(values.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    stops = np.flatnonzero(changes == -1)
    return [(int(left), int(right)) for left, right in zip(starts, stops, strict=True)]


def _read_glyph(crop: np.ndarray) -> str:
    dark = crop.max(axis=2) < 90
    ys, xs = np.nonzero(dark)
    if len(xs) == 0:
        raise AmbiguousRasterError("a key has no readable dark glyph")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    glyph = dark[y0:y1, x0:x1]
    height, width = glyph.shape
    if width < 20 or height < 28:
        raise AmbiguousRasterError("a glyph is below the declared readable scale")

    sampled = np.zeros((7, 5), dtype=bool)
    for row in range(7):
        top = round(row * height / 7)
        bottom = round((row + 1) * height / 7)
        for column in range(5):
            left = round(column * width / 5)
            right = round((column + 1) * width / 5)
            cell = glyph[top:bottom, left:right]
            sampled[row, column] = bool(cell.size and cell.mean() >= 0.75)

    distances = sorted(
        (int(np.count_nonzero(sampled != template)), label)
        for label, template in _TEMPLATES.items()
    )
    if distances[0][0] != 0 or distances[1][0] <= distances[0][0]:
        raise AmbiguousRasterError(
            f"glyph has no exact unique match (best distances {distances[:2]})"
        )
    return distances[0][1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.ndim != 3 or arr.shape[0] < 100 or arr.shape[1] < 300:
        raise AmbiguousRasterError("raster is too small for the public key-row contract")

    corner_pixels = np.concatenate(
        (
            arr[:6, :6].reshape(-1, 3),
            arr[:6, -6:].reshape(-1, 3),
            arr[-6:, :6].reshape(-1, 3),
            arr[-6:, -6:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(corner_pixels, axis=0)
    foreground = np.linalg.norm(arr - background, axis=2) > 24.0
    x_runs = [(a, b) for a, b in _runs(foreground.any(axis=0)) if b - a >= 40]
    if len(x_runs) != 8:
        raise AmbiguousRasterError(f"expected eight separated visible keys, found {len(x_runs)}")

    labels: list[str] = []
    for x0, x1 in x_runs:
        local = foreground[:, x0:x1]
        ys = np.flatnonzero(local.any(axis=1))
        if len(ys) == 0:
            raise AmbiguousRasterError("empty key extent")
        y0, y1 = int(ys.min()), int(ys.max()) + 1
        key = arr[y0:y1, x0:x1]
        inset_x = max(6, int(key.shape[1] * 0.09))
        inset_y = max(6, int(key.shape[0] * 0.09))
        interior = key[inset_y:-inset_y, inset_x:-inset_x]
        labels.append(_read_glyph(interior))

    if sorted(labels) != sorted(_EXPECTED_LABELS):
        raise AmbiguousRasterError(f"visible glyph inventory is invalid: {labels}")
    anchor = labels.index("F")
    if anchor + 1 >= len(labels):
        raise AmbiguousRasterError("the visible F key has no right neighbor")
    decision = labels[anchor + 1]
    if decision not in {"G", "H", "J", "K"}:
        raise AmbiguousRasterError(f"right-neighbor label {decision!r} is outside the answer set")
    return decision
