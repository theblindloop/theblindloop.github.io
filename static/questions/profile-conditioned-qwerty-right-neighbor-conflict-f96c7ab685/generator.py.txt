"""Deterministic renderer for a visible QWERTY-prior conflict."""

from __future__ import annotations

import hashlib
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 240
LABELS = ("A", "S", "D", "F", "G", "H", "J", "K")
ANSWERS = ("G", "H", "J", "K")

# A deliberately simple public block alphabet. Every glyph occupies its full
# 5-by-7 bounding grid, which permits a separate raster arm to recognize it
# after discovering key boxes rather than receiving font state or coordinates.
GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "S": ("11111", "10000", "10000", "11111", "00001", "00001", "11111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
    "F": ("11111", "10000", "10000", "11110", "10000", "10000", "10000"),
    "G": ("01110", "10001", "10000", "10111", "10001", "10001", "01110"),
    "H": ("10001", "10001", "10001", "11111", "10001", "10001", "10001"),
    "J": ("11111", "00100", "00100", "00100", "00100", "10100", "11000"),
    "K": ("10001", "10010", "10100", "11000", "10100", "10010", "10001"),
}

PALETTE = (
    "#f3b7b7",
    "#b9d8f4",
    "#bce6ca",
    "#f5d39a",
    "#d2c1ef",
    "#f1bfe1",
    "#b9e5e4",
    "#e7dfa9",
)
BACKGROUNDS = ("#ffffff", "#f7f8fb", "#fbf8f2")


def _rng(seed: int, channel: str) -> random.Random:
    payload = f"qwerty-neighbor-v1|{int(seed)}|{channel}".encode("utf-8")
    return random.Random(int.from_bytes(hashlib.sha256(payload).digest()[:8], "big"))


def sample_scene(seed: int) -> dict:
    """Sample balanced answer identity and independently randomized styling."""
    seed = int(seed)
    answer = ANSWERS[seed % len(ANSWERS)]

    order_rng = _rng(seed, "order")
    remaining = [label for label in ANSWERS if label != answer]
    # Some G-answer scenes are exact QWERTY controls. All other scenes retain
    # the same label inventory and wording but permute the visible suffix.
    if answer == "G" and (seed // 4) % 2 == 0:
        remaining = ["H", "J", "K"]
    else:
        order_rng.shuffle(remaining)
    label_order = ["A", "S", "D", "F", answer, *remaining]

    palette_rng = _rng(seed, "palette")
    fill_colors = list(PALETTE)
    palette_rng.shuffle(fill_colors)

    boundary_rng = _rng(seed, "boundary")
    near_boundary = boundary_rng.randrange(7) == 0
    style_rng = _rng(seed, "style")
    if near_boundary:
        key_width, key_height, gap, glyph_cell = 58, 82, 6, 5
    else:
        key_width = style_rng.choice((66, 70, 74))
        key_height = style_rng.choice((94, 100, 106))
        gap = style_rng.choice((10, 12, 14))
        glyph_cell = style_rng.choice((6, 7, 8))

    total_width = 8 * key_width + 7 * gap
    x_offset = style_rng.randint(20, WIDTH - total_width - 20)
    y_offset = style_rng.randint(44, HEIGHT - key_height - 34)
    paint_order = list(range(8))
    _rng(seed, "paint").shuffle(paint_order)

    return {
        "label_order": label_order,
        "fill_colors": fill_colors,
        "background": style_rng.choice(BACKGROUNDS),
        "x_offset": x_offset,
        "y_offset": y_offset,
        "key_width": key_width,
        "key_height": key_height,
        "gap": gap,
        "glyph_cell": glyph_cell,
        "paint_order": paint_order,
    }


def _draw_glyph(draw: ImageDraw.ImageDraw, label: str, x: int, y: int, cell: int) -> None:
    for row, bits in enumerate(GLYPHS[label]):
        for column, bit in enumerate(bits):
            if bit == "1":
                left = x + column * cell
                top = y + row * cell
                draw.rectangle(
                    (left, top, left + cell - 1, top + cell - 1),
                    fill=(24, 27, 32),
                )


def render(scene: dict) -> Image.Image:
    labels = scene["label_order"]
    fills = scene["fill_colors"]
    background = scene["background"]
    x_offset = int(scene["x_offset"])
    y_offset = int(scene["y_offset"])
    key_width = int(scene["key_width"])
    key_height = int(scene["key_height"])
    gap = int(scene["gap"])
    glyph_cell = int(scene["glyph_cell"])
    paint_order = scene["paint_order"]

    if sorted(labels) != sorted(LABELS) or len(fills) != 8:
        raise ValueError("scene must contain the fixed eight-label inventory")
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    for index in paint_order:
        index = int(index)
        x = x_offset + index * (key_width + gap)
        y = y_offset
        draw.rectangle(
            (x, y, x + key_width - 1, y + key_height - 1),
            fill=fills[index],
            outline=(24, 27, 32),
            width=4,
        )
        glyph_width = 5 * glyph_cell
        glyph_height = 7 * glyph_cell
        gx = x + (key_width - glyph_width) // 2
        gy = y + (key_height - glyph_height) // 2
        _draw_glyph(draw, labels[index], gx, gy, glyph_cell)
    return image


def analytic_gold(scene: dict) -> str:
    labels = scene["label_order"]
    try:
        anchor = labels.index("F")
    except ValueError as exc:
        raise ValueError("visible label order has no F anchor") from exc
    if anchor + 1 >= len(labels):
        raise ValueError("F has no key on its right")
    return str(labels[anchor + 1])


def margin(scene: dict) -> float:
    """Gap between nearest and runner-up right-key distances, in pixels."""
    return float(int(scene["key_width"]) + int(scene["gap"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 68.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["paint_order"] = list(reversed(scene["paint_order"]))
    return [("reverse_nonoverlapping_key_paint_order", twin)]
