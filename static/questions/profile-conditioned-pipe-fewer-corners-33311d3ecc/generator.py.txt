"""Renderer for pipe_fewer_corners.

Two gray pipes are drawn as thick rectilinear polylines on a lattice. Each pipe
runs from an end sitting inside a black ring to an end covered by a colored
square (one orange, one purple). Every bend is exactly a right angle, so the
number of turns of a pipe is a discrete, visible, threshold-free quantity.

The decision is the color of the square that terminates the pipe with fewer
right-angle turns. Total drawn pipe length is equalized between the two pipes up
to +/- one lattice unit, with the sign of that residual drawn independently of
the answer, so neither total ink nor total length carries the label.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pipe_fewer_corners-renderer-0.1.0"

CANVAS = 720
ORIGIN = 36
PITCH = 48
GRID_N = 14
STROKE_W = 9
PIPE_RGB = (74, 74, 74)
RING_RGB = (15, 15, 15)
RING_R = 15
RING_W = 4
SQUARE_HALF = 15
BG_RGB = (255, 255, 255)
TERMINAL_RGB = {"orange": (235, 125, 20), "purple": (135, 55, 195)}

MIN_RUN = 2
MAX_RUN = 6
MIN_TURNS = 3
MAX_TURNS = 9
QUARANTINE_TURN_GAP = 2.0
MIN_CELL_CLEARANCE = 2


def _partition(rng: random.Random, total: int, parts: int) -> list[int] | None:
    """Split `total` lattice units into `parts` run lengths in [MIN_RUN, MAX_RUN]."""
    if total < MIN_RUN * parts or total > MAX_RUN * parts:
        return None
    vals = [MIN_RUN] * parts
    rem = total - MIN_RUN * parts
    idx = list(range(parts))
    while rem > 0:
        rng.shuffle(idx)
        moved = False
        for i in idx:
            if vals[i] < MAX_RUN:
                vals[i] += 1
                rem -= 1
                moved = True
                if rem == 0:
                    break
        if not moved:
            return None
    return vals


def _cells_of(nodes: list[list[int]]) -> list[tuple[int, int]]:
    """Every lattice cell visited by the polyline, in travel order."""
    cells = [(nodes[0][0], nodes[0][1])]
    for a, b in zip(nodes, nodes[1:]):
        dx = (b[0] > a[0]) - (b[0] < a[0])
        dy = (b[1] > a[1]) - (b[1] < a[1])
        n = max(abs(b[0] - a[0]), abs(b[1] - a[1]))
        for s in range(1, n + 1):
            cells.append((a[0] + dx * s, a[1] + dy * s))
    return cells


def _self_clear(cells: list[tuple[int, int]]) -> bool:
    """Self-avoiding, and non-neighbouring stretches keep >= 2 cells apart."""
    if len(set(cells)) != len(cells):
        return False
    for i in range(len(cells)):
        for j in range(i + 3, len(cells)):
            ax, ay = cells[i]
            bx, by = cells[j]
            if max(abs(ax - bx), abs(ay - by)) < MIN_CELL_CLEARANCE:
                return False
    return True


def _pair_clearance(a: list[tuple[int, int]], b: list[tuple[int, int]]) -> int:
    return min(max(abs(ax - bx), abs(ay - by)) for ax, ay in a for bx, by in b)


def _build_path(rng: random.Random, turns: int, total: int) -> list[list[int]] | None:
    lens = _partition(rng, total, turns + 1)
    if lens is None:
        return None
    horiz_first = rng.random() < 0.5
    cx, cy = rng.randrange(GRID_N), rng.randrange(GRID_N)
    nodes = [[cx, cy]]
    for r, run in enumerate(lens):
        horiz = ((r % 2 == 0) == horiz_first)
        options = [+1, -1]
        rng.shuffle(options)
        placed = False
        for sign in options:
            nx = cx + sign * run if horiz else cx
            ny = cy if horiz else cy + sign * run
            if 0 <= nx < GRID_N and 0 <= ny < GRID_N:
                cx, cy = nx, ny
                nodes.append([cx, cy])
                placed = True
                break
        if not placed:
            return None
    if len(nodes) != turns + 2:
        return None
    if not _self_clear(_cells_of(nodes)):
        return None
    return nodes


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) * 7919 + 13) & 0xFFFFFFFF)
    for _ in range(4000):
        gap = rng.choice([1, 2, 2, 3, 3, 4])
        turns_low = rng.randint(MIN_TURNS, MAX_TURNS - gap)
        turns_high = turns_low + gap
        lo_total = MIN_RUN * (turns_high + 1)
        hi_total = MAX_RUN * (turns_low + 1)
        if hi_total < max(lo_total, 20):
            continue
        total_a = rng.randint(max(lo_total, 20), min(hi_total, 30))
        total_b = total_a + rng.choice([-1, 0, 1])
        if not (lo_total <= total_b <= hi_total):
            total_b = total_a
        fewer_index = rng.randrange(2)
        turns = [0, 0]
        turns[fewer_index] = turns_low
        turns[1 - fewer_index] = turns_high
        totals = [0, 0]
        totals[fewer_index] = total_a
        totals[1 - fewer_index] = total_b

        nodes_a = _build_path(rng, turns[0], totals[0])
        if nodes_a is None:
            continue
        cells_a = _cells_of(nodes_a)
        nodes_b = None
        for _ in range(120):
            trial = _build_path(rng, turns[1], totals[1])
            if trial is None:
                continue
            if _pair_clearance(cells_a, _cells_of(trial)) >= MIN_CELL_CLEARANCE:
                nodes_b = trial
                break
        if nodes_b is None:
            continue

        orange_index = int(seed) % 2
        names = ["purple", "purple"]
        names[orange_index] = "orange"
        paths = []
        for idx, nodes in enumerate((nodes_a, nodes_b)):
            paths.append(
                {
                    "terminal": names[idx],
                    "terminal_rgb": list(TERMINAL_RGB[names[idx]]),
                    "length_cells": totals[idx],
                }
            )
        return {
            "seed": int(seed),
            "canvas": [CANVAS, CANVAS],
            "background": list(BG_RGB),
            "origin": ORIGIN,
            "pitch": PITCH,
            "grid_n": GRID_N,
            "stroke_width": STROKE_W,
            "pipe_color": list(PIPE_RGB),
            "ring_color": list(RING_RGB),
            "ring_radius": RING_R,
            "ring_width": RING_W,
            "square_half": SQUARE_HALF,
            "nodes": [nodes_a, nodes_b],
            "paths": paths,
            "draw_order": [0, 1] if rng.random() < 0.5 else [1, 0],
        }
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _px(scene: dict, cell) -> tuple[int, int]:
    return (
        scene["origin"] + scene["pitch"] * int(cell[0]),
        scene["origin"] + scene["pitch"] * int(cell[1]),
    )


def render(scene: dict) -> Image.Image:
    w, h = scene["canvas"]
    img = Image.new("RGB", (w, h), tuple(scene["background"]))
    dr = ImageDraw.Draw(img)
    order = scene["draw_order"]
    for i in order:
        x, y = _px(scene, scene["nodes"][i][0])
        r = scene["ring_radius"]
        dr.ellipse(
            [x - r, y - r, x + r, y + r],
            outline=tuple(scene["ring_color"]),
            width=scene["ring_width"],
        )
    for i in order:
        pts = [_px(scene, c) for c in scene["nodes"][i]]
        dr.line(
            pts,
            fill=tuple(scene["pipe_color"]),
            width=scene["stroke_width"],
            joint="curve",
        )
    for i in order:
        x, y = _px(scene, scene["nodes"][i][-1])
        s = scene["square_half"]
        dr.rectangle([x - s, y - s, x + s, y + s], fill=tuple(scene["paths"][i]["terminal_rgb"]))
    return img


def _turns(scene: dict, index: int) -> int:
    return len(scene["nodes"][index]) - 2


def analytic_gold(scene: dict) -> str:
    paths = scene["paths"]
    first, second = _turns(scene, 0), _turns(scene, 1)
    return paths[0]["terminal"] if first <= second else paths[1]["terminal"]


def margin(scene: dict) -> float:
    return float(abs(_turns(scene, 0) - _turns(scene, 1)))


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_TURN_GAP:
        return True
    cells = [_cells_of(n) for n in scene["nodes"]]
    if _pair_clearance(cells[0], cells[1]) < MIN_CELL_CLEARANCE:
        return True
    return any(not _self_clear(c) for c in cells)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every label input is a drawn field."""
    return []
