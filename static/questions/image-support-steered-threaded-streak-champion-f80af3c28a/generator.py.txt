"""Deterministic sampler and raster renderer for threaded streak tracking."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BACKGROUND = (250, 249, 245)
INK = (38, 40, 46)
START = (42, 167, 88)
END = (215, 58, 69)
ROUTE_PALETTES = ((37, 144, 164), (116, 88, 188), (213, 119, 42))
SHAPES = ("circle", "triangle", "square")
BEAD_COUNT = 12


def _runs(sequence: list[str] | tuple[str, ...]) -> list[tuple[str, int, int]]:
    output: list[tuple[str, int, int]] = []
    for index, shape in enumerate(sequence):
        if output and output[-1][0] == shape:
            prior_shape, length, _ = output[-1]
            output[-1] = (prior_shape, length + 1, index)
        else:
            output.append((str(shape), 1, index))
    return output


def _scores(sequence: list[str] | tuple[str, ...]) -> dict[str, int]:
    """Best lexicographic (run length, completion index) score per shape."""
    stride = len(sequence) + 1
    scores = {shape: -1 for shape in SHAPES}
    for shape, length, completion in _runs(sequence):
        scores[shape] = max(scores[shape], length * stride + completion)
    return scores


def _decision(sequence: list[str] | tuple[str, ...]) -> str:
    scores = _scores(sequence)
    ordered = sorted(scores, key=lambda shape: (scores[shape], shape), reverse=True)
    if scores[ordered[0]] == scores[ordered[1]]:
        raise ValueError("streak champion is ambiguous")
    return ordered[0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = SHAPES[int(seed) % len(SHAPES)]
    multiset = [shape for shape in SHAPES for _ in range(4)]
    sequence: list[str] | None = None
    for _ in range(20000):
        trial = list(multiset)
        rng.shuffle(trial)
        if _decision(trial) != target:
            continue
        if _decision(list(reversed(trial))) == target:
            continue
        if trial[-1] == target:
            continue
        sequence = trial
        break
    if sequence is None:
        raise RuntimeError("could not sample a balanced order-live streak sequence")

    width, height = 960, 480
    left = rng.choice((48, 56, 64))
    right = width - left
    route_y = 240 + rng.randrange(-3, 4)
    slot_xs = sorted(112 + index * 67 + rng.randrange(-7, 8) for index in range(BEAD_COUNT))
    bead_offsets = [rng.choice((-20, 20)) for _ in range(BEAD_COUNT)]
    return {
        "width": width,
        "height": height,
        "left": left,
        "right": right,
        "route_y": route_y,
        "slot_xs": slot_xs,
        "bead_offsets": bead_offsets,
        "sequence": sequence,
        "route_color": list(ROUTE_PALETTES[rng.randrange(len(ROUTE_PALETTES))]),
    }


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, x: int, y: int) -> None:
    if shape == "circle":
        draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=INK)
    elif shape == "triangle":
        draw.polygon(((x, y - 15), (x - 16, y + 15), (x + 16, y + 15)), fill=INK)
    elif shape == "square":
        draw.rectangle((x - 12, y - 12, x + 12, y + 12), fill=INK)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def _route_height(x: int, left: int, right: int, route_y: int) -> int:
    phase = (float(x) - left) / max(1.0, float(right - left))
    return int(round(route_y + 4.0 * math.sin(phase * 4.0 * math.pi)))


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    left, right = int(scene["left"]), int(scene["right"])
    route_y = int(scene["route_y"])
    slot_xs = [int(value) for value in scene["slot_xs"]]
    offsets = [int(value) for value in scene["bead_offsets"]]
    sequence = [str(value) for value in scene["sequence"]]
    route_color = tuple(int(value) for value in scene["route_color"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)

    route = [(x, _route_height(x, left, right, route_y)) for x in range(left, right + 1, 6)]
    if route[-1][0] != right:
        route.append((right, _route_height(right, left, right, route_y)))
    draw.line(route, fill=route_color, width=9, joint="curve")

    for index, shape in enumerate(sequence):
        x = slot_xs[index]
        y_route = _route_height(x, left, right, route_y)
        y = y_route + offsets[index]
        draw.line((x, y_route, x, y), fill=route_color, width=5)
        _draw_shape(draw, shape, x, y)

    radius = 17
    start_y = _route_height(left, left, right, route_y)
    end_y = _route_height(right, left, right, route_y)
    draw.ellipse((left - radius, start_y - radius, left + radius, start_y + radius),
                 outline=START, width=7)
    draw.ellipse((right - radius, end_y - radius, right + radius, end_y + radius),
                 outline=END, width=7)
    return image


def analytic_gold(scene: dict) -> str:
    return _decision([str(value) for value in scene["sequence"]])


def margin(scene: dict) -> float:
    scores = sorted(_scores([str(value) for value in scene["sequence"]]).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        sequence = [str(value) for value in scene["sequence"]]
        return not (
            len(sequence) == BEAD_COUNT
            and all(sequence.count(shape) == 4 for shape in SHAPES)
            and isinstance(scene["route_y"], int)
            and len(scene["slot_xs"]) == BEAD_COUNT
            and len(scene["bead_offsets"]) == BEAD_COUNT
            and margin(scene) >= 1.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene["sequence"]
    return []
