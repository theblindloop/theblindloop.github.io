"""Independent image-only inverse arm for the threaded streak raster."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

START_RGB = (42, 167, 88)
END_RGB = (215, 58, 69)
SHAPES = ("circle", "triangle", "square")


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[dict] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x - 1), (y - 1, x), (y - 1, x + 1),
                           (y, x - 1), (y, x + 1),
                           (y + 1, x - 1), (y + 1, x), (y + 1, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = np.asarray([point[0] for point in points])
        xs = np.asarray([point[1] for point in points])
        output.append({
            "area": len(points),
            "x0": int(xs.min()), "x1": int(xs.max()),
            "y0": int(ys.min()), "y1": int(ys.max()),
            "cx": float(xs.mean()), "cy": float(ys.mean()),
        })
    return output


def _shape(component: dict) -> str:
    width = component["x1"] - component["x0"] + 1
    height = component["y1"] - component["y0"] + 1
    fill = component["area"] / float(width * height)
    if not 22 <= width <= 36 or not 22 <= height <= 36:
        raise ValueError("bead geometry is outside the public glyph bounds")
    if fill >= 0.90:
        return "square"
    if fill >= 0.66:
        return "circle"
    if fill >= 0.42:
        return "triangle"
    raise ValueError("bead shape is ambiguous")


def _distance_map(mask: np.ndarray, sources: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    distance = np.full(mask.shape, -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for y, x in sources:
        y, x = int(y), int(x)
        if mask[y, x] and distance[y, x] < 0:
            distance[y, x] = 0
            queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        value = int(distance[y, x]) + 1
        for ny, nx in ((y - 1, x - 1), (y - 1, x), (y - 1, x + 1),
                       (y, x - 1), (y, x + 1),
                       (y + 1, x - 1), (y + 1, x), (y + 1, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = value
                queue.append((ny, nx))
    return distance


def _winner(sequence: list[str]) -> str:
    stride = len(sequence) + 1
    scores = {shape: -1 for shape in SHAPES}
    current = ""
    length = 0
    for index, shape in enumerate(sequence):
        if shape == current:
            length += 1
        else:
            current, length = shape, 1
        scores[shape] = max(scores[shape], length * stride + index)
    ranking = sorted(SHAPES, key=lambda shape: (scores[shape], shape), reverse=True)
    if scores[ranking[0]] == scores[ranking[1]]:
        raise ValueError("no unique champion")
    return ranking[0]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 200:
        raise ValueError("raster is too small")

    dark = np.max(array, axis=2) < 90
    beads = [component for component in _components(dark) if 350 <= component["area"] <= 750]
    if len(beads) != 12:
        raise ValueError(f"exactly 12 intact shape beads are required, found {len(beads)}")

    maximum = array.max(axis=2).astype(np.int16)
    minimum = array.min(axis=2).astype(np.int16)
    colored = (maximum - minimum >= 55) & (maximum >= 110) & (minimum <= 180)
    green = np.all(array == np.asarray(START_RGB, dtype=np.uint8), axis=2)
    red = np.all(array == np.asarray(END_RGB, dtype=np.uint8), axis=2)
    if int(green.sum()) < 100 or int(red.sum()) < 100:
        raise ValueError("green start or red end ring is missing")
    distance = _distance_map(colored, np.argwhere(green))
    if not np.any(distance[red] >= 0):
        raise ValueError("the colored thread is not continuous from green to red")

    route_y, route_x = np.argwhere(distance >= 0).T
    ordered: list[tuple[int, str]] = []
    for component in beads:
        squared = (route_x.astype(np.float64) - component["cx"]) ** 2 + (
            route_y.astype(np.float64) - component["cy"]
        ) ** 2
        nearest = int(np.argmin(squared))
        if float(squared[nearest]) > 24.0 ** 2:
            raise ValueError("a bead is detached from the thread")
        route_distance = int(distance[route_y[nearest], route_x[nearest]])
        ordered.append((route_distance, _shape(component)))
    ordered.sort(key=lambda item: item[0])
    if any(ordered[index + 1][0] - ordered[index][0] < 35 for index in range(11)):
        raise ValueError("bead order along the thread is ambiguous")
    return _winner([shape for _, shape in ordered])
