"""Deterministic renderer for the matched-color stitch world."""
from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

PATTERNS = {
    "ladder": (0, 1, 2, 3, 4, 5),
    "three Xs": (1, 0, 3, 2, 5, 4),
    "starburst": (5, 4, 3, 2, 1, 0),
    "braid": (2, 4, 0, 5, 1, 3),
}

PALETTE = (
    "#dc2d2d", "#289646", "#2d5adc", "#be32be", "#eb8219", "#14a5af"
)


def _matching(pairs: list[dict]) -> tuple[int, ...]:
    result = [-1] * 6
    for item in pairs:
        result[int(item["left"])] = int(item["right"])
    return tuple(result)


def _scores(pairs: list[dict]) -> list[tuple[int, str]]:
    matching = _matching(pairs)
    return sorted(
        [(sum(a == b for a, b in zip(matching, template)), name)
         for name, template in PATTERNS.items()],
        reverse=True,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    name = tuple(PATTERNS)[int(seed) % len(PATTERNS)]
    colors = list(PALETTE)
    rng.shuffle(colors)
    pair_records = [
        {"left": left, "right": right, "color": colors[left]}
        for left, right in enumerate(PATTERNS[name])
    ]
    rng.shuffle(pair_records)
    left_x = rng.randint(126, 158)
    rail_width = rng.randint(218, 252)
    row_gap = rng.randint(43, 48)
    top_y = rng.randint(102, 126)
    return {
        "pairs": pair_records,
        "left_x": left_x,
        "right_x": left_x + rail_width,
        "top_y": top_y,
        "row_gap": row_gap,
        "radius": rng.randint(12, 15),
        "rail_color": rng.choice(["#64748b", "#596579", "#6b7280"]),
        "background": rng.choice(["#ffffff", "#f8fafc", "#fffdf8"]),
    }


def render(scene: dict) -> Image.Image:
    # Explicit subscripts keep the rendered-field audit transparent.
    pairs = scene["pairs"]
    left_x = int(scene["left_x"])
    right_x = int(scene["right_x"])
    top_y = int(scene["top_y"])
    row_gap = int(scene["row_gap"])
    radius = int(scene["radius"])
    rail_color = scene["rail_color"]
    background = scene["background"]
    image = Image.new("RGB", (512, 440), background)
    draw = ImageDraw.Draw(image)
    bottom_y = top_y + 5 * row_gap
    # The two visible rails make the left/right grouping explicit.
    draw.line((left_x, top_y - 34, left_x, bottom_y + 34), fill=rail_color, width=7)
    draw.line((right_x, top_y - 34, right_x, bottom_y + 34), fill=rail_color, width=7)
    draw.line((left_x - 22, top_y - 34, left_x + 22, top_y - 34), fill=rail_color, width=7)
    draw.line((right_x - 22, top_y - 34, right_x + 22, top_y - 34), fill=rail_color, width=7)
    for item in pairs:
        color = item["color"]
        for x, row in ((left_x, int(item["left"])), (right_x, int(item["right"]))):
            y = top_y + row * row_gap
            draw.ellipse((x-radius-2, y-radius-2, x+radius+2, y+radius+2), fill="#172033")
            draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["pairs"])
    if scores[0][0] != 6:
        raise ValueError("pairing is not one of the declared stitch templates")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = _scores(scene["pairs"])
    return float(scores[0][0] - scores[1][0])


def is_quarantined(scene: dict) -> bool:
    pairs = scene["pairs"]
    colors = [item.get("color") for item in pairs]
    valid_matching = sorted(item.get("left") for item in pairs) == list(range(6))
    valid_matching &= sorted(item.get("right") for item in pairs) == list(range(6))
    geometry_ok = int(scene["right_x"]) - int(scene["left_x"]) > 180
    geometry_ok &= int(scene["row_gap"]) > 2 * int(scene["radius"]) + 8
    return not (len(set(colors)) == 6 and valid_matching and geometry_ok and margin(scene) >= 2)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["pairs"] = list(reversed(twin["pairs"]))
    return [("reverse_pair_serialization", twin)]
