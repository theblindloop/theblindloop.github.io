"""Independent pixel-only inverse arm for colored rail stitches."""
from __future__ import annotations

from collections import Counter, deque
import numpy as np
from PIL import Image

TEMPLATES = {
    "ladder": (0, 1, 2, 3, 4, 5),
    "three Xs": (1, 0, 3, 2, 5, 4),
    "starburst": (5, 4, 3, 2, 1, 0),
    "braid": (2, 4, 0, 5, 1, 3),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx); ys.append(yy)
            for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)):
                ny, nx = yy+dy, xx+dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 80:
            found.append((float(sum(xs))/len(xs), float(sum(ys))/len(ys), len(xs)))
    return found


def _recover_matching(image: Image.Image) -> tuple[int, ...]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flat = arr.reshape(-1, 3)
    counts = Counter(map(tuple, flat.tolist()))
    # Dot interiors are the only large, exactly flat, strongly saturated regions.
    colors = [rgb for rgb, count in counts.items()
              if count >= 160 and max(rgb) - min(rgb) >= 80 and max(rgb) >= 130]
    groups = []
    for rgb in colors:
        comps = _components(np.all(arr == np.array(rgb, dtype=np.uint8), axis=2))
        if len(comps) == 2:
            groups.append((rgb, comps))
    if len(groups) != 6:
        raise ValueError(f"expected six recoverable color pairs, found {len(groups)}")
    centers = [point for _, comps in groups for point in comps]
    split = (min(p[0] for p in centers) + max(p[0] for p in centers)) / 2.0
    left_points = sorted([p for p in centers if p[0] < split], key=lambda p: p[1])
    right_points = sorted([p for p in centers if p[0] > split], key=lambda p: p[1])
    if len(left_points) != 6 or len(right_points) != 6:
        raise ValueError("color endpoints do not form two six-dot rails")

    def nearest_rank(point, ordered):
        distances = [abs(point[1] - candidate[1]) for candidate in ordered]
        rank = int(np.argmin(distances))
        if distances[rank] > 8:
            raise ValueError("dot could not be assigned to a rail row")
        return rank

    matching = [-1] * 6
    for _, comps in groups:
        a, b = comps
        left, right = (a, b) if a[0] < b[0] else (b, a)
        li = nearest_rank(left, left_points)
        ri = nearest_rank(right, right_points)
        matching[li] = ri
    if sorted(matching) != list(range(6)):
        raise ValueError("recovered colors do not define a perfect matching")
    return tuple(matching)


def decision_details(image: Image.Image) -> tuple[str, float]:
    matching = _recover_matching(image)
    scores = sorted(
        [(sum(a == b for a, b in zip(matching, template)), name)
         for name, template in TEMPLATES.items()], reverse=True
    )
    if scores[0][0] != 6 or scores[0][0] - scores[1][0] < 2:
        raise ValueError("ambiguous stitch template")
    return scores[0][1], float(scores[0][0] - scores[1][0])


def decision_from_image(image: Image.Image) -> str:
    return decision_details(image)[0]
