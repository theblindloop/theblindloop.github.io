"""Independent pixel-only inverse arm.

Receives only the final PNG. It segments the gray probe bar on top and the
colored candidate bars below by color, recovers each bar's orientation (tilt
angle of its principal axis) from its pixel mask, finds the candidate whose
orientation best matches the probe's, and returns that candidate's color name.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
TOL = 90


def _mask(image: np.ndarray, color: tuple) -> np.ndarray:
    return np.abs(image.astype(int) - np.array(color)).sum(axis=2) < TOL


def _principal_angle(mask: np.ndarray) -> float | None:
    ys, xs = np.nonzero(mask)
    if xs.size < 40:
        return None
    n = xs.size
    cx = xs.mean()
    cy = ys.mean()
    x = xs - cx
    y = ys - cy
    sxx = float((x * x).sum())
    syy = float((y * y).sum())
    sxy = float((x * y).sum())
    theta = 0.5 * math.atan2(2.0 * sxy, sxx - syy)
    return math.degrees(theta) % 180.0


def _angle_dist(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB")).astype(int)
    probe_angle = _principal_angle(_mask(rgb, GRAY))
    if probe_angle is None:
        return "unknown"
    best_color = "unknown"
    best_delta = float("inf")
    for name, color in PALETTE.items():
        angle = _principal_angle(_mask(rgb, color))
        if angle is None:
            continue
        delta = _angle_dist(angle, probe_angle)
        if delta < best_delta:
            best_delta = delta
            best_color = name
    return best_color
