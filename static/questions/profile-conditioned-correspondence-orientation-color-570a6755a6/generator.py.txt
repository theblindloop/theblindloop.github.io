"""Cross-panel orientation-correspondence world: match a probe bar's tilt angle
to one colored candidate bar, then report the color of the matched object."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 440
PROBE_CENTER = (320, 140)
CAND_CENTERS = [(165, 330), (320, 330), (475, 330)]

GRAY = (150, 150, 150)
PALETTE = {
    "red": (215, 60, 60),
    "green": (55, 180, 75),
    "blue": (65, 100, 215),
}
COLOR_NAMES = ["red", "green", "blue"]

BAR_LEN = 135
BAR_W = 46
ANGLE_POOL = [12, 40, 68, 96, 124, 152]
MIN_GAP = 14.0


def _rotated_bar(center: tuple, angle_deg: float, color: tuple) -> Image.Image:
    pad = int(math.hypot(BAR_LEN, BAR_W)) + 6
    layer = Image.new("RGBA", (2 * pad, 2 * pad), (0, 0, 0, 0))
    draw = ImageDraw.Draw(layer)
    x0 = pad - BAR_LEN // 2
    y0 = pad - BAR_W // 2
    draw.rectangle([x0, y0, x0 + BAR_LEN, y0 + BAR_W], fill=tuple(color) + (255,))
    return layer.rotate(angle_deg, resample=Image.BICUBIC)


def _paste_bar(img: Image.Image, center: tuple, angle: float, color: tuple) -> None:
    rot = _rotated_bar(center, angle, color)
    alpha = rot.split()[3]
    img.paste(
        rot.convert("RGB"),
        (center[0] - rot.width // 2, center[1] - rot.height // 2),
        alpha,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    angles = rng.sample(ANGLE_POOL, 3)
    probe_angle = rng.choice(angles)
    positions = list(CAND_CENTERS)
    rng.shuffle(positions)
    return {
        "probe_angle": probe_angle,
        "candidate_colors": colors,
        "candidate_angles": angles,
        "candidate_positions": positions,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), (252, 252, 252))
    _paste_bar(img, PROBE_CENTER, scene["probe_angle"], GRAY)
    for color_name, angle, pos in zip(
        scene["candidate_colors"],
        scene["candidate_angles"],
        scene["candidate_positions"],
    ):
        _paste_bar(img, pos, angle, PALETTE[color_name])
    return img


def _angle_dist(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def analytic_gold(scene: dict) -> str:
    pa = scene["probe_angle"]
    return scene["candidate_colors"][scene["candidate_angles"].index(pa)]


def margin(scene: dict) -> float:
    pa = scene["probe_angle"]
    distances = sorted(
        _angle_dist(pa, angle) for angle in scene["candidate_angles"] if angle != pa
    )
    return float(distances[0]) if distances else float("inf")


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_GAP


def latent_symmetries(scene: dict) -> list:
    return []
