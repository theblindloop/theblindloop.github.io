"""Deterministic renderer for a visibly fiducialed polar tile transform."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw, ImageFont

GRID_N = 8
IMAGE_SIZE = (760, 760)

# Each pattern is the union of two permutations. Consequently every pattern has
# exactly two dark cells in every row and every column (16 total), preventing
# radial totals, angular totals, and total ink from identifying the answer.
_COLUMN_PAIRS = (
    tuple((r, 7 - r) for r in range(8)),
    tuple(zip((3, 2, 1, 0, 0, 1, 2, 3), (4, 5, 6, 7, 7, 6, 5, 4))),
    tuple(zip((0, 2, 4, 6, 7, 5, 3, 1), (7, 5, 3, 1, 0, 2, 4, 6))),
    tuple(zip((1, 0, 2, 3, 4, 5, 6, 7), tuple(7 - r for r in range(8)))),
)


def _make_patterns() -> tuple[np.ndarray, ...]:
    patterns = []
    for pairs in _COLUMN_PAIRS:
        mask = np.zeros((GRID_N, GRID_N), dtype=bool)
        for row, columns in enumerate(pairs):
            mask[row, list(columns)] = True
        assert np.all(mask.sum(axis=0) == 2)
        assert np.all(mask.sum(axis=1) == 2)
        patterns.append(mask)
    return tuple(patterns)


PATTERNS = _make_patterns()

PALETTES = (
    ((31, 48, 73), (238, 241, 237)),
    ((53, 45, 91), (241, 238, 230)),
    ((24, 73, 66), (235, 242, 238)),
    ((74, 43, 52), (243, 239, 234)),
)
BACKGROUNDS = ((250, 248, 242), (244, 249, 250), (249, 246, 251))
GRID_COLOR = (190, 198, 201)
GOLD = (231, 171, 48)
CYAN = (30, 169, 193)
CORAL = (218, 72, 126)
CARD_FILL = (252, 252, 249)
CARD_EDGE = (207, 211, 210)


def sample_scene(seed: int) -> dict:
    """Sample a JSON-serializable scene from one integer seed."""
    rng = random.Random((int(seed) << 1) ^ 0x51A7D3)
    reference_order = list(range(4))
    rng.shuffle(reference_order)
    return {
        "pattern_id": int(seed) % 4,
        "reference_order": reference_order,
        "center_x": rng.randint(350, 410),
        "center_y": rng.randint(274, 310),
        "inner_radius": rng.randint(48, 60),
        "band_width": rng.randint(18, 21),
        "seam_angle_deg": rng.randint(0, 359),
        "palette_index": rng.randrange(len(PALETTES)),
        "canvas_variant": rng.randrange(len(BACKGROUNDS)),
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.load_default(size=size)
    except TypeError:  # compatibility with older Pillow releases
        return ImageFont.load_default()


def _draw_annulus(scene: dict, image: Image.Image, foreground: tuple[int, int, int], empty: tuple[int, int, int]) -> None:
    arr = np.asarray(image).copy()
    height, width = arr.shape[:2]
    yy, xx = np.ogrid[:height, :width]
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    inner = float(scene["inner_radius"])
    band = float(scene["band_width"])
    outer = inner + GRID_N * band
    seam = math.radians(float(scene["seam_angle_deg"]) % 360.0)
    dx = xx - cx
    dy = yy - cy
    radius = np.hypot(dx, dy)
    theta = np.mod(np.arctan2(dy, dx), 2.0 * math.pi)
    relative = np.mod(theta - seam, 2.0 * math.pi)
    inside = (radius >= inner) & (radius < outer)
    rows = np.clip(np.floor((outer - radius) / band).astype(int), 0, GRID_N - 1)
    columns = np.clip(
        np.floor(relative / (2.0 * math.pi / GRID_N)).astype(int), 0, GRID_N - 1
    )
    pattern = PATTERNS[int(scene["pattern_id"]) % len(PATTERNS)]
    occupied = inside & pattern[rows, columns]
    arr[inside] = empty
    arr[occupied] = foreground

    # Visibly expose all polar cell boundaries without obscuring cell centers.
    radial_phase = np.mod(outer - radius, band)
    radial_distance = np.minimum(radial_phase, band - radial_phase)
    sector = 2.0 * math.pi / GRID_N
    angular_phase = np.mod(relative, sector)
    angular_distance = np.minimum(angular_phase, sector - angular_phase) * radius
    separators = inside & ((radial_distance < 1.15) | (angular_distance < 1.15))
    arr[separators] = GRID_COLOR
    image.paste(Image.fromarray(arr, mode="RGB"))

    draw = ImageDraw.Draw(image)
    draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), outline=GRID_COLOR, width=2)
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), outline=GRID_COLOR, width=2)

    # A rectangular radial spoke has unambiguous radial endpoints; the inverse
    # can infer inner radius, outer radius, and phase solely from its pixels.
    ux, uy = math.cos(seam), math.sin(seam)
    px, py = -uy * 3.5, ux * 3.5
    spoke = [
        (cx + inner * ux + px, cy + inner * uy + py),
        (cx + outer * ux + px, cy + outer * uy + py),
        (cx + outer * ux - px, cy + outer * uy - py),
        (cx + inner * ux - px, cy + inner * uy - py),
    ]
    draw.polygon(spoke, fill=GOLD)
    draw.ellipse((cx - 12, cy - 12, cx + 12, cy + 12), fill=(255, 255, 255))
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=CYAN)


def _draw_references(scene: dict, image: Image.Image, foreground: tuple[int, int, int], empty: tuple[int, int, int]) -> None:
    draw = ImageDraw.Draw(image)
    order = list(scene["reference_order"])
    grid_xs = (38, 218, 398, 578)
    grid_y = 610
    cell = 12
    inner_size = GRID_N * cell
    frame = 4
    font = _font(26)
    for card_index, grid_x in enumerate(grid_xs):
        draw.rounded_rectangle(
            (grid_x - 17, grid_y - 54, grid_x + inner_size + 21, grid_y + inner_size + 18),
            radius=10,
            fill=CARD_FILL,
            outline=CARD_EDGE,
            width=2,
        )
        draw.text(
            (grid_x + inner_size / 2, grid_y - 34),
            "ABCD"[card_index],
            fill=(42, 47, 53),
            anchor="mm",
            font=font,
        )
        draw.rectangle(
            (grid_x, grid_y, grid_x + inner_size + 2 * frame, grid_y + inner_size + 2 * frame),
            fill=CORAL,
        )
        pattern = PATTERNS[int(order[card_index]) % len(PATTERNS)]
        for row in range(GRID_N):
            for column in range(GRID_N):
                x0 = grid_x + frame + column * cell
                y0 = grid_y + frame + row * cell
                color = foreground if pattern[row, column] else empty
                draw.rectangle(
                    (x0, y0, x0 + cell - 1, y0 + cell - 1),
                    fill=color,
                    outline=GRID_COLOR,
                    width=1,
                )


def render(scene: dict) -> Image.Image:
    """Render one RGB scene deterministically."""
    palette = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    background = BACKGROUNDS[int(scene["canvas_variant"]) % len(BACKGROUNDS)]
    foreground, empty = palette
    image = Image.new("RGB", IMAGE_SIZE, background)
    _draw_annulus(scene, image, foreground, empty)
    _draw_references(scene, image, foreground, empty)
    return image


def analytic_gold(scene: dict) -> str:
    pattern_id = int(scene["pattern_id"])
    reference_order = list(scene["reference_order"])
    return "ABCD"[reference_order.index(pattern_id)]


def _pattern_margin(pattern_id: int) -> float:
    chosen = PATTERNS[int(pattern_id) % len(PATTERNS)]
    distances = sorted(
        int(np.count_nonzero(chosen != candidate)) for candidate in PATTERNS
    )
    return float(distances[1] - distances[0])


def margin(scene: dict) -> float:
    """Runner-up Hamming-distance gap in recovered-grid cells."""
    return _pattern_margin(int(scene["pattern_id"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette_index"] = int(scene["palette_index"]) + len(PALETTES)
    return [("palette_period", twin)]
