"""Join-board pip relay world.

Two disconnected white panels and one framed token square sit inside a single
medium-sized region of a decorated board.  The TAG panel binds each of four
input colours to a pip count; the SLOT panel binds each pip count to one of
four output colours.  The answer is obtained by composing the two bindings
through the shared pip-count alphabet.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "joinboard_pip_relay_bind-renderer-0.1.0"

CANVAS = 680
BG = (238, 230, 212)
PANEL_FILL = (252, 252, 250)
PANEL_EDGE = (58, 60, 68)
INK = (18, 18, 20)

PANEL_W = 160
PANEL_H = 188
ROW_H = 46
SLOT_DX = 190
SLOT_DY = 86

TAG_NAMES = ("red", "blue", "green", "purple")
SLOT_NAMES = ("orange", "teal", "pink", "brown")

ANCHORS = {
    "red": (200, 40, 40),
    "blue": (40, 70, 200),
    "green": (35, 150, 60),
    "purple": (150, 50, 190),
    "orange": (240, 150, 30),
    "teal": (20, 160, 160),
    "pink": (240, 110, 180),
    "brown": (120, 80, 35),
}

QUARANTINE_MARGIN = 24.0


def _classify(rgb) -> tuple[str, float]:
    """Nearest palette anchor and the nearest/second-nearest distance gap."""
    ranked = sorted(
        (math.dist(rgb, anchor), name) for name, anchor in ANCHORS.items()
    )
    return ranked[0][1], ranked[1][0] - ranked[0][0]


def _jitter(rng: random.Random, name: str, amp: float) -> list[int]:
    base = ANCHORS[name]
    scale = amp
    for _ in range(6):
        rgb = [
            max(0, min(255, int(round(base[i] + rng.uniform(-scale, scale)))))
            for i in range(3)
        ]
        # Colour blocks must stay visibly non-black so that they are never
        # confused with the ink pips drawn beside them.
        if _classify(rgb)[0] == name and max(rgb) >= 110:
            return rgb
        scale *= 0.5
    return list(base)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    # Counterbalanced factors.  Every block of four consecutive seeds visits
    # each value of each factor exactly once, under four independent
    # permutations, so answer colour, answer row, token colour and token row
    # are balanced and mutually decorrelated.
    block, position = divmod(seed, 4)

    def _factor(tag: int) -> int:
        order = [0, 1, 2, 3]
        random.Random(block * 1000003 + tag * 7717).shuffle(order)
        return order[position]

    answer_name = SLOT_NAMES[_factor(1)]
    slot_row = _factor(2)
    token_name = TAG_NAMES[_factor(3)]
    tag_row = _factor(4)

    tag_colours = list(TAG_NAMES)
    rng.shuffle(tag_colours)
    swap = tag_colours.index(token_name)
    tag_colours[swap], tag_colours[tag_row] = tag_colours[tag_row], tag_colours[swap]

    slot_colours = list(SLOT_NAMES)
    rng.shuffle(slot_colours)
    swap = slot_colours.index(answer_name)
    slot_colours[swap], slot_colours[slot_row] = (
        slot_colours[slot_row],
        slot_colours[swap],
    )

    tag_pips = [1, 2, 3, 4]
    rng.shuffle(tag_pips)
    key_pip = tag_pips[tag_row]

    slot_pips = [1, 2, 3, 4]
    rng.shuffle(slot_pips)
    swap = slot_pips.index(key_pip)
    slot_pips[swap], slot_pips[slot_row] = slot_pips[slot_row], slot_pips[swap]

    amp = rng.uniform(3.0, 52.0)
    tag_rgb = [_jitter(rng, name, amp) for name in tag_colours]
    slot_rgb = [_jitter(rng, name, amp) for name in slot_colours]
    token_rgb = _jitter(rng, token_name, amp)

    origin_x = rng.randint(40, CANVAS - (SLOT_DX + PANEL_W) - 40)
    origin_y = rng.randint(40, CANVAS - (SLOT_DY + PANEL_H) - 40)
    token_x = origin_x + rng.randint(10, 86)
    token_y = origin_y + SLOT_DY + PANEL_H - rng.randint(38, 66)

    apparatus = (
        origin_x - 16,
        origin_y - 16,
        origin_x + SLOT_DX + PANEL_W + 16,
        origin_y + SLOT_DY + PANEL_H + 16,
    )

    decor = []
    attempts = 0
    while len(decor) < 44 and attempts < 900:
        attempts += 1
        size = rng.randint(10, 22)
        dx = rng.randint(8, CANVAS - size - 8)
        dy = rng.randint(8, CANVAS - size - 8)
        if (
            dx + size >= apparatus[0]
            and dx <= apparatus[2]
            and dy + size >= apparatus[1]
            and dy <= apparatus[3]
        ):
            continue
        name = rng.choice(TAG_NAMES + SLOT_NAMES)
        base = ANCHORS[name]
        if rng.random() < 0.68:  # pale wash, never near-white in every channel
            fill = [int(round(0.45 * base[i] + 0.55 * 255)) for i in range(3)]
        else:
            fill = list(base)
        decor.append({"x": dx, "y": dy, "size": size, "fill": fill})

    scene = {
        "seed": seed,
        "origin": [origin_x, origin_y],
        "token": {"x": token_x, "y": token_y, "rgb": token_rgb},
        "tag_rows": [
            {"rgb": tag_rgb[i], "pips": tag_pips[i]} for i in range(4)
        ],
        "slot_rows": [
            {"rgb": slot_rgb[i], "pips": slot_pips[i]} for i in range(4)
        ],
        "decor": decor,
    }
    return scene


def _panel_boxes(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    ox, oy = scene["origin"]
    return (ox, oy), (ox + SLOT_DX, oy + SLOT_DY)


def _row_centre(py: int, index: int) -> int:
    return py + 26 + ROW_H * index


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BG)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        x, y, s = item["x"], item["y"], item["size"]
        draw.rectangle([x, y, x + s, y + s], fill=tuple(item["fill"]))

    (tx, ty), (sx, sy) = _panel_boxes(scene)
    for px, py in ((tx, ty), (sx, sy)):
        draw.rectangle(
            [px, py, px + PANEL_W, py + PANEL_H], fill=PANEL_FILL, outline=PANEL_EDGE, width=3
        )

    for index, row in enumerate(scene["tag_rows"]):
        cy = _row_centre(ty, index)
        draw.rectangle([tx + 19, cy - 15, tx + 49, cy + 15], fill=tuple(row["rgb"]))
        _pips(draw, tx + 108, cy, row["pips"])

    for index, row in enumerate(scene["slot_rows"]):
        cy = _row_centre(sy, index)
        _pips(draw, sx + 54, cy, row["pips"])
        draw.rectangle([sx + 104, cy - 12, sx + 148, cy + 12], fill=tuple(row["rgb"]))

    token = scene["token"]
    x, y = token["x"], token["y"]
    draw.rectangle([x, y, x + 38, y + 38], fill=INK)
    draw.rectangle([x + 6, y + 6, x + 32, y + 32], fill=tuple(token["rgb"]))
    return image


def _pips(draw: ImageDraw.ImageDraw, cx: int, cy: int, count: int) -> None:
    step = 18
    start = cx - (count - 1) * step // 2
    for i in range(count):
        x = start + i * step
        draw.ellipse([x - 6, cy - 6, x + 6, cy + 6], fill=INK)


def _readout(scene: dict) -> dict:
    tag = [(_classify(r["rgb"])[0], r["pips"]) for r in scene["tag_rows"]]
    slot = [(_classify(r["rgb"])[0], r["pips"]) for r in scene["slot_rows"]]
    token = _classify(scene["token"]["rgb"])[0]
    key_pip = next(pips for name, pips in tag if name == token)
    slot_index = next(i for i, (_, pips) in enumerate(slot) if pips == key_pip)
    return {
        "token": token,
        "key_pip": key_pip,
        "slot_index": slot_index,
        "answer": slot[slot_index][0],
    }


def analytic_gold(scene: dict) -> str:
    read = _readout(scene)
    return f"{read['answer']}|row{read['slot_index'] + 1}"


def margin(scene: dict) -> float:
    gaps = [_classify(scene["token"]["rgb"])[1]]
    for row in scene["tag_rows"] + scene["slot_rows"]:
        gaps.append(_classify(row["rgb"])[1])
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: the raster is a bijective drawing of the scene."""
    return []
