"""Independent PNG-only inverse for the equal-distance hub panel."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

CANVAS_SIZE = (1040, 380)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (12, 18, 264, 362),
    (270, 18, 522, 362),
    (528, 18, 780, 362),
    (786, 18, 1038, 362),
)


class OracleAmbiguous(ValueError):
    """Raised when the marker geometry is absent or does not have a clear winner."""


def _mask_for_panel(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    crop = array[y0 + 58 : y1 - 16, x0 + 25 : x1 - 25]
    return np.max(crop, axis=2) < 110


def _components(mask: np.ndarray, x_offset: int, y_offset: int) -> list[tuple[float, float]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            queue: deque[tuple[int, int]] = deque([(x, y)])
            visited[y, x] = True
            pixels: list[tuple[int, int]] = []
            while queue:
                cx, cy = queue.popleft()
                pixels.append((cx, cy))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((nx, ny))
            if len(pixels) < 180:
                raise OracleAmbiguous(f"unexpected dark component of size {len(pixels)}")
            centers.append(
                (
                    x_offset + sum(px for px, _ in pixels) / len(pixels) + 0.5,
                    y_offset + sum(py for _, py in pixels) / len(pixels) + 0.5,
                )
            )
    return centers


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _panel_error(points: list[tuple[float, float]]) -> float:
    if len(points) != 4:
        raise OracleAmbiguous(f"expected four marker components, received {len(points)}")
    errors = []
    for index, hub in enumerate(points):
        distances = [_distance(hub, peer) for peer_index, peer in enumerate(points) if peer_index != index]
        errors.append(max(distances) - min(distances))
    return min(errors)


def decision_from_image(image: Image.Image) -> str:
    """Measure all four marker sets from the final raster and return its panel label."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    errors: list[float] = []
    for box in PANEL_BOXES:
        x0, y0, x1, y1 = box
        centers = _components(_mask_for_panel(array, box), x0 + 25, y0 + 58)
        errors.append(_panel_error(centers))
    order = sorted(range(4), key=lambda index: errors[index])
    if errors[order[1]] - errors[order[0]] < 8.0 or errors[order[0]] > 8.0:
        raise OracleAmbiguous(f"equal-distance scores are not separated: {errors}")
    return LABELS[order[0]]
