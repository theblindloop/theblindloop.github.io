"""Renderer and analytic scorer for the equal-distance hub search."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "equidistant_hub_panel_search-renderer-0.1.0"
CANVAS_SIZE = (1040, 380)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (12, 18, 264, 362),
    (270, 18, 522, 362),
    (528, 18, 780, 362),
    (786, 18, 1038, 362),
)
BACKGROUND = (248, 249, 247)
PANEL_FILL = (255, 255, 255)
PANEL_EDGE = (183, 190, 194)
LABEL_INK = (48, 58, 64)
MARKER_INK = (27, 53, 71)


def _distance(a: Iterable[float], b: Iterable[float]) -> float:
    ax, ay = a
    bx, by = b
    return math.hypot(ax - bx, ay - by)


def _hub_errors(points: list[list[float]] | list[tuple[float, float]]) -> list[float]:
    if len(points) != 4:
        raise ValueError("each panel must contain exactly four marker centers")
    errors: list[float] = []
    for hub_index, hub in enumerate(points):
        distances = [
            _distance(hub, point)
            for peer_index, point in enumerate(points)
            if peer_index != hub_index
        ]
        errors.append(max(distances) - min(distances))
    return errors


def _panel_error(points: list[list[float]] | list[tuple[float, float]]) -> float:
    """The smallest radius spread after trying every marker as the hub."""

    return min(_hub_errors(points))


def _inside(point: tuple[float, float], bounds: tuple[float, float, float, float]) -> bool:
    x, y = point
    left, top, right, bottom = bounds
    return left <= x <= right and top <= y <= bottom


def _separated(points: list[tuple[float, float]], minimum: float = 45.0) -> bool:
    return all(_distance(points[i], points[j]) >= minimum for i in range(4) for j in range(i))


def _sample_target(
    rng: random.Random, box: tuple[int, int, int, int]
) -> list[list[float]]:
    x0, y0, x1, y1 = box
    bounds = (x0 + 48.0, y0 + 104.0, x1 - 48.0, y1 - 58.0)
    for _ in range(5000):
        hub = (rng.uniform(x0 + 68.0, x1 - 68.0), rng.uniform(y0 + 142.0, y1 - 88.0))
        radius = rng.uniform(61.0, 79.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        offsets = (0.0, 2.0 + rng.uniform(-0.18, 0.18), 4.15 + rng.uniform(-0.18, 0.18))
        peers = [
            (hub[0] + radius * math.cos(phase + offset), hub[1] + radius * math.sin(phase + offset))
            for offset in offsets
        ]
        points = [hub, *peers]
        if all(_inside(point, bounds) for point in points) and _separated(points):
            return [[float(x), float(y)] for x, y in points]
    raise RuntimeError("could not sample a well-separated equal-radius hub")


def _sample_distractor(
    rng: random.Random, box: tuple[int, int, int, int]
) -> list[list[float]]:
    x0, y0, x1, y1 = box
    bounds = (x0 + 48.0, y0 + 104.0, x1 - 48.0, y1 - 58.0)
    for _ in range(5000):
        points: list[tuple[float, float]] = []
        for _ in range(4):
            for _ in range(500):
                point = (rng.uniform(bounds[0], bounds[2]), rng.uniform(bounds[1], bounds[3]))
                if all(_distance(point, previous) >= 50.0 for previous in points):
                    points.append(point)
                    break
            else:
                break
        if len(points) == 4 and _panel_error(points) >= 18.0:
            return [[float(x), float(y)] for x, y in points]
    raise RuntimeError("could not sample a non-tied distractor panel")


def sample_scene(seed: int) -> dict:
    """Sample one deterministic four-panel scene; target position is balanced by seed."""

    rng = random.Random(int(seed))
    target_index = int(seed) % 4
    radius = rng.choice((10, 11, 12, 13))
    panels: list[dict] = []
    for index, box in enumerate(PANEL_BOXES):
        points = _sample_target(rng, box) if index == target_index else _sample_distractor(rng, box)
        panels.append({"label": LABELS[index], "points": points, "radius": radius})
    scene = {"panels": panels}
    errors = [_panel_error(panel["points"]) for panel in panels]
    if errors[target_index] > 1e-7 or min(errors[:target_index] + errors[target_index + 1 :]) < 18.0:
        raise AssertionError(f"invalid sampled margin: {errors}")
    return scene


def render(scene: dict) -> Image.Image:
    """Render only visible panels, labels, and identical dark marker disks."""

    image = Image.new("RGB", CANVAS_SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for panel, box in zip(scene["panels"], PANEL_BOXES, strict=True):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle((x0, y0, x1 - 1, y1 - 1), radius=12, fill=PANEL_FILL, outline=PANEL_EDGE, width=2)
        draw.text((x0 + 12, y0 + 12), str(panel["label"]), font=font, fill=LABEL_INK)
        marker_radius = int(panel["radius"])
        for raw_x, raw_y in panel["points"]:
            x, y = float(raw_x), float(raw_y)
            draw.ellipse(
                (x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius),
                fill=MARKER_INK,
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Choose the unique panel with the smallest equal-radius error."""

    errors = [_panel_error(panel["points"]) for panel in scene["panels"]]
    best = min(errors)
    if errors.count(best) != 1:
        raise ValueError(f"equal-distance hub is not unique: {errors}")
    return str(scene["panels"][errors.index(best)]["label"])


def margin(scene: dict) -> float:
    """Continuous runner-up margin: second-smallest radius spread minus the best."""

    errors = sorted(_panel_error(panel["points"]) for panel in scene["panels"])
    if len(errors) != 4:
        raise ValueError("need four panel scores")
    return float(errors[1] - errors[0])


def is_quarantined(scene: dict) -> bool:
    errors = [_panel_error(panel["points"]) for panel in scene["panels"]]
    return errors.count(min(errors)) != 1 or margin(scene) < 18.0 or min(errors) > 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; the sole top-level field is rendered and scored."""

    return []
