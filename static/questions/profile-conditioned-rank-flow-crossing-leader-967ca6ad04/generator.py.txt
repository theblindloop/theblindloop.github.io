"""Deterministic Pillow renderer for pairwise rank-reversal aggregation."""

from __future__ import annotations

import itertools
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

PANEL_RGB = {
    "blue": (36, 105, 184),
    "orange": (230, 126, 34),
    "green": (35, 148, 88),
}
ITEM_RGB = {
    "violet": (126, 71, 181),
    "cyan": (20, 155, 184),
    "gold": (214, 164, 24),
    "rose": (206, 75, 122),
    "brown": (142, 92, 55),
}
BACKGROUND = (247, 248, 250)
PAPER = (255, 255, 255)
GRID = (217, 221, 226)
INK = (43, 47, 53)
ITEMS = tuple(ITEM_RGB)
PANEL_X = (24, 282, 540)
PANEL_TOP = 66
PANEL_BOTTOM = 360
PANEL_WIDTH = 216
LEFT_DX = 27
RIGHT_DX = 189
LEVELS = (132, 176, 220, 264, 308)


def _inversions(values: tuple[int, ...]) -> int:
    return sum(values[i] > values[j] for i in range(5) for j in range(i + 1, 5))


PERMUTATIONS_BY_COUNT: dict[int, list[tuple[int, ...]]] = {value: [] for value in range(11)}
for _permutation in itertools.permutations(range(5)):
    PERMUTATIONS_BY_COUNT[_inversions(_permutation)].append(_permutation)


def _reversal_count(left: dict[str, Any], right: dict[str, Any]) -> int:
    return sum(
        (float(left[a]) - float(left[b])) * (float(right[a]) - float(right[b])) < 0
        for index, a in enumerate(ITEMS)
        for b in ITEMS[index + 1 :]
    )


def _crossings(panel: dict[str, Any]) -> list[tuple[float, float]]:
    left = panel["left_y"]
    right = panel["right_y"]
    points: list[tuple[float, float]] = []
    for index, a in enumerate(ITEMS):
        for b in ITEMS[index + 1 :]:
            da0 = float(left[a]) - float(left[b])
            da1 = float(right[a]) - float(right[b])
            if da0 * da1 >= 0:
                continue
            t = da0 / (da0 - da1)
            y = float(left[a]) + t * (float(right[a]) - float(left[a]))
            points.append((t, y))
    return points


def _resolved(panel: dict[str, Any]) -> bool:
    left_values = sorted(float(value) for value in panel["left_y"].values())
    right_values = sorted(float(value) for value in panel["right_y"].values())
    if min(b - a for a, b in zip(left_values, left_values[1:])) < 30:
        return False
    if min(b - a for a, b in zip(right_values, right_values[1:])) < 30:
        return False
    points = _crossings(panel)
    if any(t < 0.055 or t > 0.945 for t, _ in points):
        return False
    for index, (ta, ya) in enumerate(points):
        xa = ta * (RIGHT_DX - LEFT_DX)
        for tb, yb in points[index + 1 :]:
            xb = tb * (RIGHT_DX - LEFT_DX)
            if math.hypot(xa - xb, ya - yb) < 6.5:
                return False
    return True


def _panel_for_count(rng: random.Random, label: str, count: int) -> dict[str, Any]:
    for _ in range(1200):
        left_order = list(ITEMS)
        rng.shuffle(left_order)
        relative = rng.choice(PERMUTATIONS_BY_COUNT[count])
        right_order = [left_order[index] for index in relative]
        left_y = {item: LEVELS[position] + rng.randint(-5, 5) for position, item in enumerate(left_order)}
        right_y = {item: LEVELS[position] + rng.randint(-5, 5) for position, item in enumerate(right_order)}
        panel = {"label": label, "left_y": left_y, "right_y": right_y}
        if _reversal_count(panel["left_y"], panel["right_y"]) == count and _resolved(panel):
            return panel
    raise RuntimeError(f"could not sample a resolved panel with {count} reversals")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    labels = list(PANEL_RGB)
    winner = labels[int(seed) % len(labels)]
    winner_count = rng.choice((7, 8))
    runner_count = winner_count - 1
    loser_count = rng.randint(1, max(1, runner_count - 2))
    counts = {winner: winner_count}
    others = [label for label in labels if label != winner]
    rng.shuffle(others)
    counts[others[0]] = runner_count
    counts[others[1]] = loser_count
    rng.shuffle(labels)
    panels = [_panel_for_count(rng, label, counts[label]) for label in labels]
    return {
        "width": 780,
        "height": 430,
        "panels": panels,
        "item_colors": {name: list(rgb) for name, rgb in ITEM_RGB.items()},
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    panels = scene["panels"]
    counts = [_reversal_count(panel["left_y"], panel["right_y"]) for panel in panels]
    best = max(counts)
    if counts.count(best) != 1:
        return "abstain"
    return str(panels[counts.index(best)]["label"])


def margin(scene: dict[str, Any]) -> float:
    counts = sorted((_reversal_count(panel["left_y"], panel["right_y"]) for panel in scene["panels"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    panels = scene["panels"]
    counts = [_reversal_count(panel["left_y"], panel["right_y"]) for panel in panels]
    return counts.count(max(counts)) != 1 or any(not _resolved(panel) for panel in panels)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    panels = scene["panels"]
    item_colors = {name: tuple(int(channel) for channel in rgb) for name, rgb in scene["item_colors"].items()}
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 17 * scale)
        panel_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 14 * scale)
        label_font = ImageFont.truetype("DejaVuSans.ttf", 12 * scale)
    except OSError:
        title_font = panel_font = label_font = ImageFont.load_default()

    draw.text((24 * scale, 18 * scale), "Pairwise rank reversals from A to B", fill=INK, font=title_font)
    for panel_index, (left, panel) in enumerate(zip(PANEL_X, panels, strict=True)):
        right = left + PANEL_WIDTH
        panel_rgb = PANEL_RGB[str(panel["label"])]
        box = tuple(value * scale for value in (left, PANEL_TOP, right, PANEL_BOTTOM))
        draw.rectangle(box, fill=PAPER, outline=panel_rgb, width=4 * scale)
        draw.rectangle((left * scale, PANEL_TOP * scale, right * scale, 98 * scale), fill=panel_rgb)
        draw.text(((left + 70) * scale, 73 * scale), str(panel["label"]), fill=PAPER, font=panel_font)
        x0, x1 = left + LEFT_DX, left + RIGHT_DX
        draw.line((x0 * scale, 118 * scale, x0 * scale, 325 * scale), fill=GRID, width=2 * scale)
        draw.line((x1 * scale, 118 * scale, x1 * scale, 325 * scale), fill=GRID, width=2 * scale)
        draw.text(((x0 - 5) * scale, 332 * scale), "A", fill=INK, font=label_font)
        draw.text(((x1 - 5) * scale, 332 * scale), "B", fill=INK, font=label_font)
        for item in ITEMS:
            y0 = float(panel["left_y"][item])
            y1 = float(panel["right_y"][item])
            rgb = item_colors[item]
            draw.line((x0 * scale, y0 * scale, x1 * scale, y1 * scale), fill=rgb, width=4 * scale)
        radius = 6 * scale
        for item in ITEMS:
            rgb = item_colors[item]
            for x, y in ((x0, float(panel["left_y"][item])), (x1, float(panel["right_y"][item]))):
                draw.ellipse((x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius), fill=rgb, outline=PAPER, width=1 * scale)

    draw.text((24 * scale, 388 * scale), "Each crossing represents one pair whose top-to-bottom order reverses.", fill=INK, font=label_font)
    return image.resize((width, height), Image.Resampling.LANCZOS)
