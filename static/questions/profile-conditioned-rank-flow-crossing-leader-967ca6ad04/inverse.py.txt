"""Independent raster-only inverse for colored rank-flow panels."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PANELS = {
    "blue": (36, 105, 184),
    "orange": (230, 126, 34),
    "green": (35, 148, 88),
}
_ITEMS = {
    "violet": (126, 71, 181),
    "cyan": (20, 155, 184),
    "gold": (214, 164, 24),
    "rose": (206, 75, 122),
    "brown": (142, 92, 55),
}


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 48.0) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.int32)
    delta = arr.astype(np.int32) - target
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _panel_box(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int, int, int]:
    mask = _near(arr, rgb, 42.0)
    dense_rows = np.flatnonzero(np.sum(mask, axis=1) > 145)
    if len(dense_rows) < 20:
        raise ValueError("colored panel frame is missing")
    top = int(np.min(dense_rows))
    header_rows = dense_rows[dense_rows <= top + 36]
    ys, xs = np.nonzero(mask & np.isin(np.indices(mask.shape)[0], header_rows))
    left, right = int(np.min(xs)), int(np.max(xs))
    edge_band = mask[:, max(0, left - 2) : left + 4] | mask[:, right - 3 : min(mask.shape[1], right + 3)]
    bottom_rows = np.nonzero(np.any(edge_band, axis=1))[0]
    bottom = int(np.max(bottom_rows))
    if right - left < 190 or bottom - top < 270:
        raise ValueError("colored panel frame is malformed")
    return left, top, right, bottom


def _endpoint_y(arr: np.ndarray, rgb: tuple[int, int, int], box: tuple[int, int, int, int]) -> tuple[float, float]:
    left, top, right, bottom = box
    mask = _near(arr, rgb, 48.0)
    ys, xs = np.nonzero(mask & (np.indices(mask.shape)[1] > left + 8) & (np.indices(mask.shape)[1] < right - 8) & (np.indices(mask.shape)[0] > top + 45) & (np.indices(mask.shape)[0] < bottom - 20))
    if len(xs) < 260:
        raise ValueError("item trajectory is missing")
    xmin, xmax = int(np.min(xs)), int(np.max(xs))
    if xmax - xmin < 145:
        raise ValueError("item trajectory span is too short")
    left_y = float(np.median(ys[xs <= xmin + 9]))
    right_y = float(np.median(ys[xs >= xmax - 9]))
    return left_y, right_y


def _count_reversals(endpoints: dict[str, tuple[float, float]]) -> int:
    names = tuple(endpoints)
    for side in (0, 1):
        ordered = sorted(endpoints[name][side] for name in names)
        if min(b - a for a, b in zip(ordered, ordered[1:])) < 24:
            raise ValueError("endpoint ranks are not resolved")
    return sum(
        (endpoints[a][0] - endpoints[b][0]) * (endpoints[a][1] - endpoints[b][1]) < 0
        for index, a in enumerate(names)
        for b in names[index + 1 :]
    )


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    counts: dict[str, int] = {}
    for label, panel_rgb in _PANELS.items():
        box = _panel_box(arr, panel_rgb)
        endpoints = {name: _endpoint_y(arr, rgb, box) for name, rgb in _ITEMS.items()}
        counts[label] = _count_reversals(endpoints)
    best = max(counts.values())
    winners = [label for label, value in counts.items() if value == best]
    if len(winners) != 1:
        raise ValueError("top reversal count is tied")
    return winners[0]
