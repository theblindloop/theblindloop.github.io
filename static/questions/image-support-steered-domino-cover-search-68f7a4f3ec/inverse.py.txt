"""Independent pixel-only inverse arm for domino-coverable island search."""

from __future__ import annotations

from typing import Iterable

import numpy as np
from PIL import Image

# Public visual color contract, deliberately duplicated rather than imported.
_COLOR_NAMES = ("red", "blue", "green", "purple")
_COLOR_PROTOTYPES = {
    "red": ((210, 66, 62), (190, 52, 55), (226, 82, 67)),
    "blue": ((48, 112, 190), (62, 132, 204), (46, 96, 170)),
    "green": ((47, 145, 91), (35, 129, 82), (63, 160, 102)),
    "purple": ((132, 78, 167), (151, 91, 179), (117, 66, 151)),
}
_BACKGROUND_PROTOTYPES = ((248, 247, 242), (250, 248, 244), (246, 248, 245), (255, 255, 255))


def _category_labels(array: np.ndarray) -> np.ndarray:
    categories = ("background",) + _COLOR_NAMES
    prototypes = {"background": _BACKGROUND_PROTOTYPES, **_COLOR_PROTOTYPES}
    distances = []
    pixels = array.astype(np.int32)
    for category in categories:
        variants = np.asarray(prototypes[category], dtype=np.int32)
        squared = ((pixels[:, :, None, :] - variants[None, None, :, :]) ** 2).sum(axis=3)
        distances.append(squared.min(axis=2))
    return np.stack(distances, axis=2).argmin(axis=2)


def _connected(cells: set[tuple[int, int]]) -> bool:
    if not cells:
        return False
    seen = {next(iter(cells))}
    pending = list(seen)
    while pending:
        x, y = pending.pop()
        for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if neighbor in cells and neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return seen == cells


def _recover_cells(mask: np.ndarray) -> set[tuple[int, int]] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) < 8000:
        return None
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    cell = int(round((width + height) / 10.0))
    if not 27 <= cell <= 35:
        return None
    if abs(width - 5 * cell) > 1 or abs(height - 5 * cell) > 1:
        return None

    occupied: set[tuple[int, int]] = set()
    radius = max(2, cell // 8)
    for gy in range(5):
        for gx in range(5):
            cx = x0 + gx * cell + cell // 2
            cy = y0 + gy * cell + cell // 2
            patch = mask[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
            if patch.size and float(patch.mean()) >= 0.80:
                occupied.add((gx, gy))
    if len(occupied) != 16 or not _connected(occupied):
        return None
    expected_area = 16 * cell * cell
    if abs(int(mask.sum()) - expected_area) > max(20, int(0.025 * expected_area)):
        return None
    return occupied


def _maximum_dominoes(cells: Iterable[tuple[int, int]]) -> int:
    occupied = set(cells)
    left = sorted(point for point in occupied if (point[0] + point[1]) % 2 == 0)
    matched: dict[tuple[int, int], tuple[int, int]] = {}

    def augment(point: tuple[int, int], seen: set[tuple[int, int]]) -> bool:
        x, y = point
        for neighbor in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
            if neighbor not in occupied or neighbor in seen:
                continue
            seen.add(neighbor)
            if neighbor not in matched or augment(matched[neighbor], seen):
                matched[neighbor] = point
                return True
        return False

    return sum(1 for point in left if augment(point, set()))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("image is too small for the public lattice contract")
    labels = _category_labels(array)
    scores: dict[str, int] = {}
    for label_index, color in enumerate(_COLOR_NAMES, start=1):
        cells = _recover_cells(labels == label_index)
        if cells is not None:
            scores[color] = _maximum_dominoes(cells)
    winners = [color for color, score in scores.items() if score == 8]
    if len(winners) != 1:
        raise ValueError("pixels do not expose one unique complete domino cover")
    return winners[0]
