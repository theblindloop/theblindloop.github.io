"""Deterministic raster world for domino-coverable island search.

Every candidate is a connected 16-cell subset of a visible 5 by 5 lattice.
All templates share cell count, bounding extent, perimeter, bipartite color
counts, adjacency-degree multiset, and sorted row/column counts.  The target
alone has a perfect matching in the edge-adjacency graph.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = (512, 512)
COLOR_NAMES = ("red", "blue", "green", "purple")
PALETTES = (
    {
        "red": (210, 66, 62),
        "blue": (48, 112, 190),
        "green": (47, 145, 91),
        "purple": (132, 78, 167),
    },
    {
        "red": (190, 52, 55),
        "blue": (62, 132, 204),
        "green": (35, 129, 82),
        "purple": (151, 91, 179),
    },
    {
        "red": (226, 82, 67),
        "blue": (46, 96, 170),
        "green": (63, 160, 102),
        "purple": (117, 66, 151),
    },
)
BACKGROUNDS = ((248, 247, 242), (250, 248, 244), (246, 248, 245))

# Four tileable and four non-tileable templates from one exact-stat stratum.
# '#' is an occupied cell.  Every template is connected and hole-free.
_TILEABLE_TEXT = (
    ("####.", "..##.", "#####", ".##.#", ".##.."),
    (".##..", "..#.#", "..###", "#####", "###.#"),
    (".#..#", "##.##", "#####", ".##.#", ".##.."),
    ("####.", "#####", "##..#", ".##..", "##..."),
)
_NONTILEABLE_TEXT = (
    (".###.", "...##", "..##.", "####.", "#####"),
    ("...##", "#..##", "#.###", "#####", ".#..#"),
    ("##...", "##..#", "###.#", "#####", "#..#."),
    ("#####", "####.", "..##.", "...##", ".###."),
)


def _parse(rows: Iterable[str]) -> tuple[tuple[int, int], ...]:
    return tuple(
        (x, y)
        for y, row in enumerate(rows)
        for x, value in enumerate(row)
        if value == "#"
    )


TILEABLE_TEMPLATES = tuple(_parse(rows) for rows in _TILEABLE_TEXT)
NONTILEABLE_TEMPLATES = tuple(_parse(rows) for rows in _NONTILEABLE_TEXT)


def _transform(cells: Iterable[tuple[int, int]], code: int) -> list[list[int]]:
    """Apply one of the eight dihedral transforms of the 5 by 5 lattice."""

    output = []
    reflect = code >= 4
    turns = code % 4
    for x0, y0 in cells:
        x, y = (4 - x0, y0) if reflect else (x0, y0)
        for _ in range(turns):
            x, y = 4 - y, x
        output.append([int(x), int(y)])
    return sorted(output, key=lambda point: (point[1], point[0]))


def _maximum_dominoes(cells: Iterable[Iterable[int]]) -> int:
    occupied = {tuple(map(int, point)) for point in cells}
    left = sorted(point for point in occupied if (point[0] + point[1]) % 2 == 0)
    matched: dict[tuple[int, int], tuple[int, int]] = {}

    def augment(point: tuple[int, int], seen: set[tuple[int, int]]) -> bool:
        x, y = point
        for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if neighbor not in occupied or neighbor in seen:
                continue
            seen.add(neighbor)
            if neighbor not in matched or augment(matched[neighbor], seen):
                matched[neighbor] = point
                return True
        return False

    return sum(1 for point in left if augment(point, set()))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = COLOR_NAMES[int(seed) % 4]
    target_slot = (int(seed) % 4 + (int(seed) // 4) % 4) % 4

    colors: list[str | None] = [None, None, None, None]
    colors[target_slot] = answer_color
    remaining_colors = [color for color in COLOR_NAMES if color != answer_color]
    rng.shuffle(remaining_colors)
    for slot, color in zip(
        (index for index in range(4) if index != target_slot),
        remaining_colors,
        strict=True,
    ):
        colors[slot] = color

    target = rng.choice(TILEABLE_TEMPLATES)
    distractors = rng.sample(list(NONTILEABLE_TEMPLATES), 3)
    shapes: list[list[list[int]] | None] = [None, None, None, None]
    shapes[target_slot] = _transform(target, rng.randrange(8))
    for slot, template in zip(
        (index for index in range(4) if index != target_slot),
        distractors,
        strict=True,
    ):
        shapes[slot] = _transform(template, rng.randrange(8))

    cell_size = 30 + rng.randrange(3)
    dx, dy = rng.randint(-5, 5), rng.randint(-5, 5)
    centers = ((128, 139), (384, 139), (128, 373), (384, 373))
    half = (5 * cell_size) // 2
    origins = [[cx - half + dx, cy - half + dy] for cx, cy in centers]
    palette_variant = rng.randrange(len(PALETTES))
    return {
        "canvas": list(CANVAS),
        "background": list(BACKGROUNDS[palette_variant]),
        "cell_size": cell_size,
        "origins": origins,
        "colors": [str(color) for color in colors],
        "shapes": [shape for shape in shapes],
        "palette_variant": palette_variant,
    }


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas"])
    background = tuple(map(int, scene["background"]))
    cell_size = int(scene["cell_size"])
    origins = scene["origins"]
    colors = scene["colors"]
    shapes = scene["shapes"]
    palette = PALETTES[int(scene["palette_variant"])]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    for origin, color, cells in zip(origins, colors, shapes, strict=True):
        ox, oy = map(int, origin)
        fill = palette[str(color)]
        seam = tuple(round(0.84 * channel + 0.16 * bg) for channel, bg in zip(fill, background))
        for cell_x, cell_y in cells:
            x0 = ox + int(cell_x) * cell_size
            y0 = oy + int(cell_y) * cell_size
            box = (x0, y0, x0 + cell_size - 1, y0 + cell_size - 1)
            draw.rectangle(box, fill=fill)
            draw.rectangle(box, outline=seam, width=1)
    return image


def analytic_gold(scene: dict) -> str:
    scores = [_maximum_dominoes(cells) for cells in scene["shapes"]]
    winners = [index for index, score in enumerate(scores) if score == 8]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one perfectly coverable island")
    return str(scene["colors"][winners[0]])


def margin(scene: dict) -> float:
    scores = sorted((_maximum_dominoes(cells) for cells in scene["shapes"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) <= 0.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
