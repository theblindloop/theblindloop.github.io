"""Scene sampler, raster renderer and analytic gold for gap_fit_binding.

Four horizontal rails of equal drawn length each carry a row of solid blocks in
that rail's own colour.  A separate black BAR is drawn above them.  A rail can
accept the bar when one of its empty stretches, measured along the rail and
including the stretches at the two rail ends, is wider than the bar.  Exactly
one rail can accept it.  The decided quantity is therefore the widest piece of
negative space on each rail, compared with a separately drawn reference width.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 880
HEIGHT = 620

RAIL_X0 = 70
RAIL_X1 = 810
RAIL_LEN = RAIL_X1 - RAIL_X0

LANE_Y = (168, 278, 388, 498)
RAIL_THICK = 5
BLOCK_H = 46
CAP_H = 12

BAR_Y0 = 58
BAR_H = 30

COLOR_NAMES = ("red", "green", "blue", "orange")
COLOR_RGB = {
    "red": (214, 42, 42),
    "green": (30, 140, 60),
    "blue": (36, 86, 200),
    "orange": (232, 140, 20),
}
RAIL_RGB = (110, 110, 110)
BAR_RGB = (20, 20, 20)
BACKGROUND = (255, 255, 255)

MIN_BLOCK = 34
MAX_BLOCK = 300
MIN_GAP = 10
GAP_HEADROOM = 12

BAR_MIN, BAR_MAX = 100, 132
CLEARANCE_MIN, CLEARANCE_MAX = 20, 42
TIGHT_MIN, TIGHT_MAX = 14, 19
TIGHT_RATE = 0.3
FREE_TOTALS = (200, 220, 240, 260)

MIN_CLEARANCE = 14.0
QUARANTINE_MARGIN = 20.0
MARGIN_CAP = 40.0


# ---------------------------------------------------------------- sampling


def _bounded_composition(rng: random.Random, total: int, parts: int, low: int, high: int):
    """Split ``total`` into ``parts`` integers, each within ``[low, high]``."""

    if parts * low > total or parts * high < total:
        return None
    values = []
    remaining = total
    for index in range(parts):
        left = parts - index - 1
        lo = max(low, remaining - left * high)
        hi = min(high, remaining - left * low)
        if lo > hi:
            return None
        value = rng.randint(lo, hi)
        values.append(value)
        remaining -= value
    rng.shuffle(values)
    return values


def _build_rail(rng: random.Random, widest: int, free_total: int):
    """Lay out one rail whose widest empty stretch is exactly ``widest``."""

    rest = free_total - widest
    high = widest - GAP_HEADROOM
    if high < MIN_GAP or rest < MIN_GAP:
        return None
    feasible = [n for n in (2, 3, 4, 5) if n * MIN_GAP <= rest <= n * high]
    if not feasible:
        return None
    count = feasible[rng.randrange(len(feasible))]
    small = _bounded_composition(rng, rest, count, MIN_GAP, high)
    if small is None:
        return None
    blocks = _bounded_composition(
        rng, RAIL_LEN - free_total, count, MIN_BLOCK, MAX_BLOCK
    )
    if blocks is None:
        return None
    gaps = list(small)
    gaps.insert(rng.randrange(count + 1), widest)
    cursor = RAIL_X0
    placed = []
    for index, gap in enumerate(gaps):
        cursor += gap
        if index < count:
            placed.append([cursor, cursor + blocks[index]])
            cursor += blocks[index]
    if cursor != RAIL_X1:
        return None
    return placed


def sample_scene(seed: int) -> dict:
    """Sample one deterministic scene."""

    rng = random.Random(int(seed) * 7919 + 13)
    answer = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    for _ in range(400):
        bar_w = rng.randrange(BAR_MIN, BAR_MAX + 1)
        lanes = list(range(4))
        rng.shuffle(lanes)
        totals = list(FREE_TOTALS)
        rng.shuffle(totals)
        tight = rng.randrange(4) if rng.random() < TIGHT_RATE else -1
        blocks = []
        ok = True
        for index, name in enumerate(COLOR_NAMES):
            if index == tight:
                slack = rng.randrange(TIGHT_MIN, TIGHT_MAX + 1)
            else:
                slack = rng.randrange(CLEARANCE_MIN, CLEARANCE_MAX + 1)
            widest = bar_w + slack if name == answer else bar_w - slack
            placed = _build_rail(rng, widest, totals[index])
            if placed is None:
                ok = False
                break
            blocks.append(placed)
        if not ok:
            continue
        bar_x = rng.randrange(RAIL_X0, RAIL_X1 - bar_w + 1)
        scene = {
            "bar_width": bar_w,
            "bar_x": bar_x,
            "colors": list(COLOR_NAMES),
            "lanes": lanes,
            "blocks": blocks,
        }
        if analytic_gold(scene) != answer:
            continue
        if margin(scene) < MIN_CLEARANCE:
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


# ---------------------------------------------------------------- geometry


def _gaps(blocks) -> list[int]:
    """Widths of the empty stretches of a rail, rail ends included."""

    edges = sorted((int(a), int(b)) for a, b in blocks)
    gaps = []
    cursor = RAIL_X0
    for start, end in edges:
        gaps.append(start - cursor)
        cursor = end
    gaps.append(RAIL_X1 - cursor)
    return gaps


def _widest_gap(blocks) -> int:
    return max(_gaps(blocks))


def analytic_gold(scene: dict) -> str:
    """Colour of the one rail with an empty stretch wider than the bar."""

    bar = int(scene["bar_width"])
    fits = [
        name
        for name, blocks in zip(scene["colors"], scene["blocks"])
        if _widest_gap(blocks) > bar
    ]
    if len(fits) != 1:
        raise ValueError("scene does not have exactly one accepting rail")
    return fits[0]


def margin(scene: dict) -> float:
    """Smallest distance in pixels between a widest gap and the bar width."""

    bar = int(scene["bar_width"])
    closest = min(abs(_widest_gap(blocks) - bar) for blocks in scene["blocks"])
    return float(min(float(closest), MARGIN_CAP))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabelings that must preserve the answer."""

    def copy_with(colors, lanes, blocks):
        twin = dict(scene)
        twin["colors"] = colors
        twin["lanes"] = lanes
        twin["blocks"] = blocks
        return twin

    order = list(range(len(scene["colors"])))
    rotated = order[1:] + order[:1]
    flipped = list(reversed(order))
    return [
        (
            "reverse_block_order_within_each_rail",
            copy_with(
                list(scene["colors"]),
                list(scene["lanes"]),
                [[list(b) for b in reversed(rail)] for rail in scene["blocks"]],
            ),
        ),
        (
            "rotate_rail_storage_order",
            copy_with(
                [scene["colors"][i] for i in rotated],
                [scene["lanes"][i] for i in rotated],
                [scene["blocks"][i] for i in rotated],
            ),
        ),
        (
            "reverse_rail_storage_order",
            copy_with(
                [scene["colors"][i] for i in flipped],
                [scene["lanes"][i] for i in flipped],
                [scene["blocks"][i] for i in flipped],
            ),
        ),
    ]


# ---------------------------------------------------------------- rendering


def render(scene: dict) -> Image.Image:
    """Rasterise a scene with axis-aligned, anti-aliasing-free primitives."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    bar_x = int(scene["bar_x"])
    bar_w = int(scene["bar_width"])
    draw.rectangle(
        [bar_x, BAR_Y0, bar_x + bar_w - 1, BAR_Y0 + BAR_H - 1], fill=BAR_RGB
    )

    rails = sorted(
        zip(scene["lanes"], scene["colors"], scene["blocks"]),
        key=lambda rail: int(rail[0]),
    )
    for lane, color, rail_blocks in rails:
        y = LANE_Y[int(lane)]
        draw.rectangle([RAIL_X0, y, RAIL_X1 - 1, y + RAIL_THICK - 1], fill=RAIL_RGB)
        draw.rectangle([RAIL_X0, y, RAIL_X0 + 2, y + CAP_H - 1], fill=RAIL_RGB)
        draw.rectangle([RAIL_X1 - 3, y, RAIL_X1 - 1, y + CAP_H - 1], fill=RAIL_RGB)
        fill = COLOR_RGB[color]
        for start, end in sorted(
            ([int(a), int(b)] for a, b in rail_blocks), key=lambda p: p[0]
        ):
            draw.rectangle([start, y - BLOCK_H, end - 1, y - 1], fill=fill)
    return image
