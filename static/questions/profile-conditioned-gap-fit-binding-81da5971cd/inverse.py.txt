"""Pixel-only decision arm for gap_fit_binding.

It receives a rasterised picture and nothing else: no latent scene, no renderer
constants and no analytic gold.  It recovers the rail span from the grey rails,
the reference width from the black bar, the occupied columns of each coloured
rail from colour segmentation, and then measures the widest empty column run of
each rail.  It abstains whenever the evidence is damaged or too close to call.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_TARGETS = {
    "red": (214, 42, 42),
    "green": (30, 140, 60),
    "blue": (36, 86, 200),
    "orange": (232, 140, 20),
}
_RAIL = (110, 110, 110)
_BAR = (20, 20, 20)

_TOLERANCE = 26
_MIN_COLOUR_PIXELS = 400
_MIN_RAIL_COLUMNS = 400
_MIN_BAR_PIXELS = 200
_MAX_LANE_ROWS = 70
_MIN_DECIDABLE_GAP = 6.0


def _mask(array: np.ndarray, rgb) -> np.ndarray:
    diff = np.abs(array.astype(np.int16) - np.array(rgb, dtype=np.int16))
    return diff.max(axis=2) <= _TOLERANCE


def _longest_false_run(flags: np.ndarray) -> int:
    best = run = 0
    for value in flags:
        if value:
            run = 0
        else:
            run += 1
            if run > best:
                best = run
    return best


def decision_from_image(image: Image.Image) -> str:
    """Name the one rail whose widest empty stretch exceeds the bar width."""

    array = np.asarray(image.convert("RGB"))

    rail = _mask(array, _RAIL)
    rail_columns = np.flatnonzero(rail.any(axis=0))
    if rail_columns.size < _MIN_RAIL_COLUMNS:
        return ABSTAIN
    left, right = int(rail_columns[0]), int(rail_columns[-1]) + 1

    bar = _mask(array, _BAR)
    if int(bar.sum()) < _MIN_BAR_PIXELS:
        return ABSTAIN
    bar_columns = np.flatnonzero(bar.any(axis=0))
    reference = int(bar_columns[-1]) - int(bar_columns[0]) + 1

    widest: dict[str, int] = {}
    for name, rgb in _TARGETS.items():
        blocks = _mask(array, rgb)
        if int(blocks.sum()) < _MIN_COLOUR_PIXELS:
            return ABSTAIN
        rows = np.flatnonzero(blocks.any(axis=1))
        if int(rows[-1]) - int(rows[0]) + 1 > _MAX_LANE_ROWS:
            return ABSTAIN
        occupied = blocks.any(axis=0)[left:right]
        if not occupied.any():
            return ABSTAIN
        widest[name] = _longest_false_run(occupied)

    closest = min(abs(value - reference) for value in widest.values())
    if closest < _MIN_DECIDABLE_GAP:
        return ABSTAIN
    accepting = [name for name, value in widest.items() if value > reference]
    if len(accepting) != 1:
        return ABSTAIN
    return accepting[0]
