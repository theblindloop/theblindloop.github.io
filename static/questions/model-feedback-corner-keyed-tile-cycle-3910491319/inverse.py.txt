"""Independent PNG-only inverse for the corner-keyed tile cycle."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (900, 700)
CELL, GAP, BOARD_X, BOARD_Y = 120, 14, 466, 220
TURNS = 7
DARK = np.array((24, 59, 86), dtype=np.uint8)
LIGHT = np.array((248, 246, 238), dtype=np.uint8)
GOLD = np.array((213, 141, 22), dtype=np.uint8)
CUE = np.array((0, 169, 199), dtype=np.uint8)
IDENTITIES = np.array(
    (
        (209, 73, 91), (224, 122, 31), (42, 157, 112),
        (37, 137, 189), (115, 87, 178), (198, 78, 139),
        (0, 137, 123), (139, 90, 43), (116, 130, 32),
    ),
    dtype=np.float32,
)
BLOCKS = (
    (0, 1, 4, 3),
    (1, 2, 5, 4),
    (4, 5, 8, 7),
    (3, 4, 7, 6),
)


class OracleError(ValueError):
    pass


def _origin(index: int) -> tuple[int, int]:
    row, column = divmod(index, 3)
    return BOARD_X + column * (CELL + GAP), BOARD_Y + row * (CELL + GAP)


def _read_identity(array: np.ndarray, index: int) -> int:
    left, top = _origin(index)
    patch = array[top + 3 : top + 11, left + 42 : left + 78].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    distances = np.linalg.norm(IDENTITIES - mean, axis=1)
    order = np.argsort(distances)
    if float(distances[order[0]]) > 6.0 or float(distances[order[1]] - distances[order[0]]) < 20.0:
        raise OracleError("missing or ambiguous identity frame")
    return int(order[0])


def _read_bit(array: np.ndarray, index: int) -> int:
    left, top = _origin(index)
    patch = array[top + 17 : top + 103, left + 17 : left + 103]
    dark_count = int(np.all(patch == DARK, axis=2).sum())
    light_count = int(np.all(patch == LIGHT, axis=2).sum())
    if dark_count > 4300 and light_count < 50:
        return 1
    if light_count > 4300 and dark_count < 50:
        return 0
    raise OracleError("tile has no clean binary face")


def _read_corner(array: np.ndarray, index: int) -> int:
    left, top = _origin(index)
    centres = ((29, 29), (91, 29), (91, 91), (29, 91))
    counts = []
    for cx, cy in centres:
        patch = array[top + cy - 12 : top + cy + 13, left + cx - 12 : left + cx + 13]
        counts.append(int(np.all(patch == CUE, axis=2).sum()))
    matches = [corner for corner, count in enumerate(counts) if count > 150]
    if len(matches) != 1 or any(count > 8 for corner, count in enumerate(counts) if corner != matches[0]):
        raise OracleError("missing or duplicated corner cue")
    return matches[0]


def _validate_reader(array: np.ndarray) -> None:
    left = BOARD_X + CELL + GAP
    top = BOARD_Y + CELL + GAP
    upper = array[top - 7 : top, left + 25 : left + CELL - 25]
    lower = array[top + CELL + 1 : top + CELL + 8, left + 25 : left + CELL - 25]
    count = int(np.all(upper == GOLD, axis=2).sum() + np.all(lower == GOLD, axis=2).sum())
    if count < 700:
        raise OracleError("fixed reader outline is missing")


def _read_board(array: np.ndarray) -> list[tuple[int, int, int]]:
    board = [
        (_read_identity(array, index), _read_bit(array, index), _read_corner(array, index))
        for index in range(9)
    ]
    if {tile[0] for tile in board} != set(range(9)):
        raise OracleError("board does not contain nine distinct identities")
    return board


def _advance(board: list[tuple[int, int, int]]) -> list[tuple[int, int, int]]:
    current = list(board)
    identity, old_face, corner = current[4]
    current[4] = (identity, 1 - old_face, corner)
    positions = BLOCKS[corner]
    prior = [current[position] for position in positions]
    destinations = (
        (positions[1], positions[2], positions[3], positions[0])
        if old_face == 0
        else (positions[3], positions[0], positions[1], positions[2])
    )
    for value, destination in zip(prior, destinations):
        current[destination] = value
    return current


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_reader(array)
    board = _read_board(array)
    for _ in range(TURNS):
        board = _advance(board)
    return str(board[4][0] + 1)
