"""Deterministic sampler, renderer, and analytic keyed-cycle machine."""

from __future__ import annotations

import random
from typing import Iterable, NamedTuple

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 700
BOARD_N, TURNS = 3, 7
CELL, GAP, BOARD_X, BOARD_Y = 120, 14, 466, 220
BACKGROUND = "#ffffff"
DARK = "#183b56"
LIGHT = "#f8f6ee"
NAVY = "#334d5c"
GOLD = "#d58d16"
CUE = "#00a9c7"
IDENTITY_COLORS = (
    "#d1495b", "#e07a1f", "#2a9d70", "#2589bd", "#7357b2",
    "#c64e8b", "#00897b", "#8b5a2b", "#748220",
)
FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

# Positions are top-left, top-right, bottom-right, bottom-left. The controller
# at board index 4 belongs to every block.
BLOCKS = (
    (0, 1, 4, 3),
    (1, 2, 5, 4),
    (4, 5, 8, 7),
    (3, 4, 7, 6),
)


class Tile(NamedTuple):
    identity: int
    bit: int
    corner: int


Board = tuple[Tile, ...]


def _tile(value: object) -> Tile:
    if not isinstance(value, (list, tuple)) or len(value) != 3:
        raise ValueError("each tile needs identity, face, and corner")
    identity, bit, corner = (int(item) for item in value)
    if identity not in range(9) or bit not in (0, 1) or corner not in range(4):
        raise ValueError("invalid tile attribute")
    return Tile(identity, bit, corner)


def _board(values: Iterable[object]) -> Board:
    result = tuple(_tile(value) for value in values)
    if len(result) != 9 or {tile.identity for tile in result} != set(range(9)):
        raise ValueError("board must contain each identity exactly once")
    return result


def _as_json(board: Board) -> list[list[int]]:
    return [[tile.identity, tile.bit, tile.corner] for tile in board]


def _turn(board: Board) -> Board:
    current = list(_board(_as_json(board)))
    controller = current[4]
    current[4] = Tile(controller.identity, 1 - controller.bit, controller.corner)
    positions = BLOCKS[controller.corner]
    old = [current[index] for index in positions]
    if controller.bit == 0:  # old LIGHT: clockwise
        for source, destination in zip(old, (positions[1], positions[2], positions[3], positions[0])):
            current[destination] = source
    else:  # old DARK: counterclockwise
        for source, destination in zip(old, (positions[3], positions[0], positions[1], positions[2])):
            current[destination] = source
    return tuple(current)


def _evolve(board: Board, turns: int = TURNS) -> Board:
    result = _board(_as_json(board))
    for _ in range(int(turns)):
        result = _turn(result)
    return result


def _trace(board: Board) -> list[Board]:
    states = [_board(_as_json(board))]
    for _ in range(TURNS):
        states.append(_turn(states[-1]))
    return states


def _valid_scene(scene: dict) -> Board:
    if set(scene) != {"start_tiles"}:
        raise ValueError("scene fields do not match declared label inputs")
    return _board(scene["start_tiles"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(4000):
        identities = list(range(9))
        rng.shuffle(identities)
        bits = [rng.randrange(2) for _ in range(9)]
        corners = [rng.randrange(4) for _ in range(9)]
        start = tuple(
            Tile(identity, bit, corner)
            for identity, bit, corner in zip(identities, bits, corners)
        )
        states = _trace(start)
        controllers = [state[4] for state in states[:-1]]
        if (
            len({tile.identity for tile in controllers}) >= 5
            and len({tile.corner for tile in controllers}) >= 3
            and {tile.bit for tile in controllers} == {0, 1}
            and states[-1][4].identity != start[4].identity
        ):
            return {"start_tiles": _as_json(start)}
    raise RuntimeError("could not sample a transition-rich keyed-cycle board")


def analytic_gold(scene: dict) -> str:
    board = _valid_scene(scene)
    return str(_evolve(board)[4].identity + 1)


def margin(scene: dict) -> float:
    _valid_scene(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        _valid_scene(scene)
        return False
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _valid_scene(scene)
    return []


def _font(size: int, *, bold: bool = True) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_BOLD if bold else FONT_REGULAR, size=size)


def _draw_tile(draw: ImageDraw.ImageDraw, tile: Tile, left: int, top: int) -> None:
    draw.rounded_rectangle(
        (left, top, left + CELL, top + CELL), radius=10,
        fill=IDENTITY_COLORS[tile.identity],
    )
    draw.rounded_rectangle(
        (left + 14, top + 14, left + CELL - 14, top + CELL - 14),
        radius=6, fill=DARK if tile.bit else LIGHT, outline=NAVY, width=2,
    )
    draw.text(
        (left + CELL // 2, top + CELL // 2 + 2), str(tile.identity + 1),
        fill="white" if tile.bit else NAVY, font=_font(34), anchor="mm",
    )
    cue_centres = ((29, 29), (91, 29), (91, 91), (29, 91))
    cx, cy = cue_centres[tile.corner]
    draw.ellipse(
        (left + cx - 9, top + cy - 9, left + cx + 9, top + cy + 9),
        fill=CUE, outline=NAVY, width=2,
    )


def render(scene: dict) -> Image.Image:
    board = _valid_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 28), "CORNER-KEYED TILE CYCLE", fill=NAVY, font=_font(28), anchor="mm")
    draw.text(
        (WIDTH // 2, 66), "7 TURNS  •  READ OLD FACE  •  TILES SLIDE WITHOUT TURNING",
        fill=NAVY, font=_font(17), anchor="mm",
    )

    draw.rounded_rectangle((34, 118, 414, 610), radius=18, fill="#f3f6f7", outline="#cbd5da", width=3)
    draw.text((58, 146), "EACH TURN", fill=GOLD, font=_font(24), anchor="lm")
    instructions = (
        "1   Read the tile in READER",
        "2   Its cyan dot chooses the",
        "     2 by 2 block toward that corner",
        "3   Flip that reader tile",
        "4   Cycle all four tiles:",
    )
    for line, y in zip(instructions, (194, 242, 278, 334, 390)):
        draw.text((58, y), line, fill=NAVY, font=_font(18), anchor="lm")
    draw.rounded_rectangle((58, 430, 91, 463), radius=5, fill=LIGHT, outline=NAVY, width=2)
    draw.text((108, 446), "old LIGHT  →  CLOCKWISE", fill=NAVY, font=_font(17), anchor="lm")
    draw.rounded_rectangle((58, 484, 91, 517), radius=5, fill=DARK, outline=NAVY, width=2)
    draw.text((108, 500), "old DARK  →  COUNTERCLOCKWISE", fill=NAVY, font=_font(17), anchor="lm")
    draw.text((58, 558), "Cyan dot stays bound to its tile.", fill=CUE, font=_font(17), anchor="lm")
    draw.text((58, 584), "Gold outline stays at the center.", fill=GOLD, font=_font(17), anchor="lm")

    draw.text((BOARD_X + 187, 188), "FIXED READER", fill=GOLD, font=_font(22), anchor="mm")
    for index, tile in enumerate(board):
        row, column = divmod(index, 3)
        left = BOARD_X + column * (CELL + GAP)
        top = BOARD_Y + row * (CELL + GAP)
        _draw_tile(draw, tile, left, top)

    centre_left = BOARD_X + CELL + GAP
    centre_top = BOARD_Y + CELL + GAP
    draw.rounded_rectangle(
        (centre_left - 7, centre_top - 7, centre_left + CELL + 7, centre_top + CELL + 7),
        radius=15, outline=GOLD, width=7,
    )
    draw.text(
        (BOARD_X + 187, 650), "Which numbered tile is here after turn 7?",
        fill=NAVY, font=_font(20), anchor="mm",
    )
    return image
