"""Deterministic sampler and renderer for two-axis punch-pattern unfolding."""

from __future__ import annotations

import copy
import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 720
SCALE = 3
BACKGROUND = (249, 247, 240)
PAPER = (255, 255, 252)
FOLDED = (226, 239, 244)
INK = (25, 29, 35)
MUTED = (142, 147, 151)
REFERENCE_FRAME = (166, 63, 132)
CANDIDATE_FRAME = (39, 112, 142)
LABEL = (51, 55, 61)
LABELS = ("A", "B", "C")


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _point_set(points: Iterable[Iterable[int]]) -> set[tuple[int, int]]:
    return {(int(point[0]), int(point[1])) for point in points}


def _closure(points: Iterable[Iterable[int]]) -> set[tuple[int, int]]:
    return {
        (sx * abs(x), sy * abs(y))
        for x, y in _point_set(points)
        for sx in (-1, 1)
        for sy in (-1, 1)
    }


def _target(scene: dict) -> set[tuple[int, int]]:
    return _closure(scene["quarter_holes"])


def _shuffle_points(rng: random.Random, points: set[tuple[int, int]]) -> list[list[int]]:
    values = list(points)
    rng.shuffle(values)
    return [[x, y] for x, y in values]


def sample_scene(seed: int) -> dict:
    """Sample three unique x/y bindings and two projection-matched distractors."""
    rng = random.Random(int(seed))
    xs = rng.sample((1, 2, 3, 4), 3)
    ys = rng.sample((1, 2, 3, 4), 3)
    rng.shuffle(ys)
    quarter = {(x, y) for x, y in zip(xs, ys, strict=True)}

    # Both distractors retain the exact x and y marginals.  Only the binding of
    # one distance-from-vertical to one distance-from-horizontal changes.
    swapped_y = [ys[1], ys[0], ys[2]]
    cycled_y = [ys[1], ys[2], ys[0]]
    distractor_quarters = [
        {(x, y) for x, y in zip(xs, swapped_y, strict=True)},
        {(x, y) for x, y in zip(xs, cycled_y, strict=True)},
    ]
    distractors = [_closure(points) for points in distractor_quarters]
    rng.shuffle(distractors)

    answer_index = int(seed) % 3
    candidates: list[set[tuple[int, int]]] = []
    cursor = 0
    for index in range(3):
        if index == answer_index:
            candidates.append(_closure(quarter))
        else:
            candidates.append(distractors[cursor])
            cursor += 1
    return {
        "quarter_holes": _shuffle_points(rng, quarter),
        "candidate_holes": [_shuffle_points(rng, points) for points in candidates],
    }


def analytic_gold(scene: dict) -> str:
    target = _target(scene)
    matches = [
        index
        for index, points in enumerate(scene["candidate_holes"])
        if _point_set(points) == target
    ]
    if len(matches) != 1:
        raise ValueError("scene does not contain one exact unfolded pattern")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    target = _target(scene)
    distances = sorted(
        len(target.symmetric_difference(_point_set(points)))
        for points in scene["candidate_holes"]
    )
    if len(distances) != 3 or distances[0] != 0:
        return 0.0
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 8.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["quarter_holes"] = list(reversed(twin["quarter_holes"]))
    twin["candidate_holes"] = [list(reversed(points)) for points in twin["candidate_holes"]]
    return [("reverse_hole_serialization", twin)]


def _centered(
    draw: ImageDraw.ImageDraw,
    xy: tuple[float, float],
    text: str,
    font: ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
        text,
        font=font,
        fill=fill,
    )


def _dashed(
    draw: ImageDraw.ImageDraw,
    start: tuple[int, int],
    end: tuple[int, int],
    fill: tuple[int, int, int],
    width: int,
) -> None:
    x0, y0 = start
    x1, y1 = end
    if x0 == x1:
        for y in range(y0, y1, 14 * SCALE):
            draw.line((x0, y, x1, min(y + 7 * SCALE, y1)), fill=fill, width=width)
    else:
        for x in range(x0, x1, 14 * SCALE):
            draw.line((x, y0, min(x + 7 * SCALE, x1), y1), fill=fill, width=width)


def _hole_xy(x: int, y: int, left: int, top: int, size: int) -> tuple[int, int]:
    center_x = (left + size / 2) * SCALE
    center_y = (top + size / 2) * SCALE
    return (
        int(round(center_x + x * size * SCALE / 10)),
        int(round(center_y - y * size * SCALE / 10)),
    )


def _draw_sheet(
    draw: ImageDraw.ImageDraw,
    points: Iterable[Iterable[int]],
    left: int,
    top: int,
    size: int,
    frame: tuple[int, int, int],
    folded_quarter: bool,
) -> None:
    x0, y0, x1, y1 = (value * SCALE for value in (left, top, left + size, top + size))
    draw.rectangle((x0, y0, x1, y1), fill=PAPER, outline=frame, width=5 * SCALE)
    cx, cy = (left + size / 2) * SCALE, (top + size / 2) * SCALE
    if folded_quarter:
        draw.rectangle((cx + 2 * SCALE, y0 + 5 * SCALE, x1 - 5 * SCALE, cy - 2 * SCALE), fill=FOLDED)
    _dashed(draw, (int(cx), y0 + 7 * SCALE), (int(cx), y1 - 7 * SCALE), MUTED, 2 * SCALE)
    _dashed(draw, (x0 + 7 * SCALE, int(cy)), (x1 - 7 * SCALE, int(cy)), MUTED, 2 * SCALE)
    for x, y in _point_set(points):
        px, py = _hole_xy(x, y, left, top, size)
        radius = 7 * SCALE
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title = _font(23, bold=True)
    label = _font(18, bold=True)
    small = _font(15)

    _centered(draw, (WIDTH * SCALE / 2, 28 * SCALE), "UNFOLD ACROSS VERTICAL, THEN HORIZONTAL", title, LABEL)
    _centered(draw, (WIDTH * SCALE / 2, 66 * SCALE), "FOLDED QUARTER — punched face is shaded", small, REFERENCE_FRAME)
    _draw_sheet(draw, scene["quarter_holes"], 355, 84, 250, REFERENCE_FRAME, True)

    # Colored arrows make the two declared reflection axes explicit without
    # introducing answer-specific marks.
    draw.line((338 * SCALE, 145 * SCALE, 338 * SCALE, 270 * SCALE), fill=REFERENCE_FRAME, width=4 * SCALE)
    draw.polygon(((338 * SCALE, 276 * SCALE), (330 * SCALE, 262 * SCALE), (346 * SCALE, 262 * SCALE)), fill=REFERENCE_FRAME)
    draw.line((420 * SCALE, 350 * SCALE, 545 * SCALE, 350 * SCALE), fill=REFERENCE_FRAME, width=4 * SCALE)
    draw.polygon(((551 * SCALE, 350 * SCALE), (537 * SCALE, 342 * SCALE), (537 * SCALE, 358 * SCALE)), fill=REFERENCE_FRAME)

    _centered(draw, (WIDTH * SCALE / 2, 389 * SCALE), "FULL-SHEET CANDIDATES", label, CANDIDATE_FRAME)
    for index, left in enumerate((45, 370, 695)):
        _draw_sheet(draw, scene["candidate_holes"][index], left, 415, 220, CANDIDATE_FRAME, False)
        _centered(draw, ((left + 110) * SCALE, 667 * SCALE), LABELS[index], label, LABEL)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
