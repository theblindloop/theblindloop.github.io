"""Independent pixel-only inverse arm for the concyclic-consistency world.

Recovers every dark dot centre from the final raster alone (by connected-
component centroid extraction), fits the single best common circle to the whole
constellation by least squares, and decides whether all the dot centres lie on
that one circle's perimeter (yes) or are scattered so no single circle passes
through all of them (no). It receives only the image, so the answer is
recoverable from the pixels alone -- the implicit global circle the whole set
either shares in full or lacks -- with no latent scene input.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

MIN_PIXELS = 6
MIN_DOTS = 3
RESIDUAL_THRESHOLD = 3.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    labelled = np.zeros((h, w), dtype=bool)
    components = []
    for sy, sx in np.argwhere(mask):
        if labelled[sy, sx]:
            continue
        comp = []
        stack = [(int(sy), int(sx))]
        labelled[sy, sx] = True
        while stack:
            y, x = stack.pop()
            comp.append((y, x))
            for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not labelled[ny, nx]:
                    labelled[ny, nx] = True
                    stack.append((ny, nx))
        if len(comp) >= MIN_PIXELS:
            components.append(np.asarray(comp, dtype=float))
    return components


def _centroids(mask: np.ndarray) -> np.ndarray:
    centroids = []
    for comp in _components(mask):
        centroids.append(comp.mean(axis=0))
    if not centroids:
        return np.zeros((0, 2))
    return np.asarray(centroids)


def _fit_circle(points: np.ndarray) -> tuple[float, float, float]:
    x = points[:, 0]
    y = points[:, 1]
    n = len(points)
    xm = x.mean()
    ym = y.mean()
    zx = x - xm
    zy = y - ym
    zr2 = zx * zx + zy * zy
    a = np.array(
        [
            [(zx * zx).sum(), (zx * zy).sum()],
            [(zx * zy).sum(), (zy * zy).sum()],
        ]
    )
    b = np.array([0.5 * (zx * zr2).sum(), 0.5 * (zy * zr2).sum()])
    try:
        sol = np.linalg.solve(a, b)
    except np.linalg.LinAlgError:
        return float(xm), float(ym), 1.0
    cx = xm + sol[0]
    cy = ym + sol[1]
    radius = float(np.sqrt(((zx - sol[0]) ** 2 + (zy - sol[1]) ** 2).mean()))
    return float(cx), float(cy), radius


def _circular_residual(mask: np.ndarray) -> float:
    pts = _centroids(mask)
    if len(pts) < MIN_DOTS:
        return float(RESIDUAL_THRESHOLD * 4.0)
    cx, cy, r = _fit_circle(pts)
    dists = np.hypot(pts[:, 0] - cx, pts[:, 1] - cy)
    return float(((dists - r) ** 2).mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    mask = (arr[..., 0].astype(int) < 150) & (arr[..., 1].astype(int) < 150)
    residual = _circular_residual(mask)
    return "yes" if residual < RESIDUAL_THRESHOLD else "no"
