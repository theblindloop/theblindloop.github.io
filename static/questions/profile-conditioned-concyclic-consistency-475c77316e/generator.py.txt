"""Concyclic-consistency world: a global circle Gestalt of a dot constellation.

A set of small dark dot disks is scattered across a white canvas. In the
"concyclic" state every dot centre lies exactly on the perimeter of a single
common circle, so together the dots trace one round ring. In the "nonconcyclic"
state the same number of dots is scattered so that no single circle passes
through all of their centres. The figure to reveal is the implicit common circle
the whole constellation either shares in full or fails to share at all -- a
global circular-arrangement Gestalt, not any single dot, its count, colour, or
absolute position.

The renderer owns scene sampling, rasterization, and the structural analytic
gold (the boolean concyclic flag). The independent pixel-only inverse arm
(``oracle.py``) reconstructs every dot centre from the raster, fits the
best single circle to the whole set by least squares, and measures how far the
dots deviate from one shared perimeter.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
DOT_RADIUS = 4          # filled disk radius in pixels
DOT_COLOUR = (30, 40, 90)
BACKGROUND = (255, 255, 255)
MIN_PIXELS = 6          # minimum connected-component size to count as a dot
MIN_DOTS = 3            # minimum dot centres required for a circle fit
RESIDUAL_THRESHOLD = 3.0   # mean squared radial deviation from best-fit circle (px^2)
MARGIN_QUARANTINE = 1.5    # ambiguity band around the residual threshold (px^2)
_MARGIN = 45            # keep dots away from the canvas border


def _fit_circle(points: np.ndarray) -> tuple[float, float, float]:
    """Kasa least-squares circle fit; returns (cx, cy, radius)."""
    x = points[:, 0]
    y = points[:, 1]
    n = len(points)
    xm = x.mean()
    ym = y.mean()
    zx = x - xm
    zy = y - ym
    zr2 = zx * zx + zy * zy
    a = np.array(
        [
            [(zx * zx).sum(), (zx * zy).sum()],
            [(zx * zy).sum(), (zy * zy).sum()],
        ]
    )
    b = np.array([0.5 * (zx * zr2).sum(), 0.5 * (zy * zr2).sum()])
    try:
        sol = np.linalg.solve(a, b)
    except np.linalg.LinAlgError:
        return float(xm), float(ym), 1.0
    cx = xm + sol[0]
    cy = ym + sol[1]
    radius = float(np.sqrt(((zx - sol[0]) ** 2 + (zy - sol[1]) ** 2).mean()))
    return float(cx), float(cy), radius


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    concyclic = bool(rng.integers(0, 2))
    cx = 192 + float(rng.uniform(-25.0, 25.0))
    cy = 192 + float(rng.uniform(-25.0, 25.0))
    n = int(rng.integers(8, 14))
    points = []
    if concyclic:
        radius = float(rng.uniform(78.0, 118.0))
        base = float(rng.uniform(0.0, 2.0 * np.pi))
        for i in range(n):
            ang = base + 2.0 * np.pi * (i + float(rng.uniform(-0.12, 0.12))) / n
            points.append((cx + radius * float(np.cos(ang)), cy + radius * float(np.sin(ang))))
    else:
        radius = float(rng.uniform(70.0, 95.0))
        for _ in range(n):
            rr = radius * float(np.sqrt(rng.uniform(0.15, 1.0)))
            ang = float(rng.uniform(0.0, 2.0 * np.pi))
            px = cx + rr * float(np.cos(ang))
            py = cy + rr * float(np.sin(ang))
            points.append((px, py))
    return {
        "concyclic": concyclic,
        "n": n,
        "cx": round(cx, 4),
        "cy": round(cy, 4),
        "points": [[round(px, 4), round(py, 4)] for (px, py) in points],
    }


def _draw_scene(scene: dict) -> Image.Image:
    if scene["concyclic"] not in (True, False):
        raise ValueError(f"unknown concyclic flag {scene.get('concyclic')!r}")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for px, py in scene["points"]:
        draw.ellipse(
            [px - DOT_RADIUS, py - DOT_RADIUS, px + DOT_RADIUS, py + DOT_RADIUS],
            fill=DOT_COLOUR,
        )
    return image


def render(scene: dict) -> Image.Image:
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["concyclic"] else "no"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    labelled = np.zeros((h, w), dtype=bool)
    components = []
    for sy, sx in np.argwhere(mask):
        if labelled[sy, sx]:
            continue
        comp = []
        stack = [(int(sy), int(sx))]
        labelled[sy, sx] = True
        while stack:
            y, x = stack.pop()
            comp.append((y, x))
            for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not labelled[ny, nx]:
                    labelled[ny, nx] = True
                    stack.append((ny, nx))
        if len(comp) >= MIN_PIXELS:
            components.append(np.asarray(comp, dtype=float))
    return components


def _centroids(mask: np.ndarray) -> np.ndarray:
    centroids = []
    for comp in _components(mask):
        centroids.append(comp.mean(axis=0))
    if not centroids:
        return np.zeros((0, 2))
    return np.asarray(centroids)


def _circular_residual(mask: np.ndarray) -> float:
    pts = _centroids(mask)
    if len(pts) < MIN_DOTS:
        return float(RESIDUAL_THRESHOLD * 4.0)
    cx, cy, r = _fit_circle(pts)
    dists = np.hypot(pts[:, 0] - cx, pts[:, 1] - cy)
    return float(((dists - r) ** 2).mean())


def margin(scene: dict) -> float:
    arr = np.asarray(_draw_scene(scene).convert("RGB"))
    mask = (arr[..., 0].astype(int) < 150) & (arr[..., 1].astype(int) < 150)
    residual = _circular_residual(mask)
    return float(RESIDUAL_THRESHOLD - residual)


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
