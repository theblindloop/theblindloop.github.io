"""Deterministic latent sampler and raster renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLORS = ("red", "blue", "green", "orange")
RGB = {
    "red": (220, 50, 47),
    "blue": (38, 139, 210),
    "green": (54, 160, 90),
    "orange": (245, 130, 32),
}
DARK = (38, 50, 56)
BAND = (217, 222, 226)
BAND_EDGE = (142, 151, 158)
CURVE_SCALE = 82.0
QUARANTINE_GAP = 4.0


def _curve_y(coefficients: dict, x: float, center_x: float) -> float:
    t = (x - center_x) / CURVE_SCALE
    return coefficients["a"] * t * t + coefficients["b"] * t + coefficients["c"]


def _continuation_scores(scene: dict) -> list[tuple[float, str]]:
    scores: list[tuple[float, str]] = []
    for pair in scene["pairs"]:
        left = pair["left"]
        right = pair["right"]
        da = left["a"] - right["a"]
        db = left["b"] - right["b"]
        dc = left["c"] - right["c"]
        score = math.sqrt(da * da + 0.36 * db * db + 0.015625 * dc * dc)
        scores.append((score, pair["color"]))
    return sorted(scores)


def sample_scene(seed: int) -> dict:
    """Sample a balanced four-way continuation scene from one integer seed."""
    rng = random.Random(int(seed))
    width = 512
    height = 512
    answer_color = COLORS[int(seed) % len(COLORS)]
    answer_lane = (int(seed) // len(COLORS)) % len(COLORS)

    remaining_colors = [color for color in COLORS if color != answer_color]
    remaining_lanes = [lane for lane in range(4) if lane != answer_lane]
    rng.shuffle(remaining_colors)
    rng.shuffle(remaining_lanes)
    color_by_lane = {answer_lane: answer_color}
    color_by_lane.update(dict(zip(remaining_lanes, remaining_colors, strict=True)))

    curvature_choices = (-6.0, -4.0, 4.0, 6.0)
    pairs = []
    for lane in range(4):
        color = color_by_lane[lane]
        left_a = rng.choice(curvature_choices)
        if color == answer_color:
            right_a = left_a
        else:
            alternatives = [value for value in curvature_choices if abs(value - left_a) >= 8.0]
            right_a = rng.choice(alternatives)
        shared_b = round(rng.uniform(-2.5, 2.5), 6)
        shared_c = round(100.0 + 104.0 * lane + rng.uniform(-4.0, 4.0), 6)
        pairs.append(
            {
                "color": color,
                "left": {"a": left_a, "b": shared_b, "c": shared_c},
                "right": {"a": right_a, "b": shared_b, "c": shared_c},
            }
        )

    return {
        "width": width,
        "height": height,
        "background": [250, 249, 246],
        "band_left": 218,
        "band_right": 294,
        "pairs": pairs,
    }


def render(scene: dict) -> Image.Image:
    """Render only visible evidence; no analytic label or metadata is embedded."""
    ss = 3
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    band_left = int(scene["band_left"])
    band_right = int(scene["band_right"])
    center_x = (band_left + band_right) / 2.0
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    def points(coefficients: dict, start: int, stop: int) -> list[tuple[int, int]]:
        return [
            (x * ss, round(_curve_y(coefficients, x, center_x) * ss))
            for x in range(start, stop + 1)
        ]

    for pair in scene["pairs"]:
        draw.line(points(pair["left"], 64, band_left + 8), fill=DARK, width=5 * ss)
        draw.line(points(pair["right"], band_right - 8, 448), fill=DARK, width=5 * ss)

    draw.rectangle(
        (band_left * ss, 24 * ss, band_right * ss, (height - 24) * ss),
        fill=BAND,
        outline=BAND_EDGE,
        width=2 * ss,
    )

    marker_positions = (band_left - 10, band_right + 10)
    for pair in scene["pairs"]:
        for side, x in zip(("left", "right"), marker_positions, strict=True):
            y = _curve_y(pair[side], x, center_x)
            outer = 9 * ss
            inner = 6 * ss
            cx = round(x * ss)
            cy = round(y * ss)
            draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=(250, 249, 246))
            draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=RGB[pair["color"]])

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return _continuation_scores(scene)[0][1]


def margin(scene: dict) -> float:
    scores = _continuation_scores(scene)
    return scores[1][0] - scores[0][0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["pairs"] = list(reversed(twin["pairs"]))
    return [("reverse_pair_storage_order", twin)]
