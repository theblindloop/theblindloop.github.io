"""Independent inverse arm operating on the final RGB raster only."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 139, 210),
    "green": (54, 160, 90),
    "orange": (245, 130, 32),
}
_DARK = np.array((38, 50, 56), dtype=np.int16)
_BAND = np.array((217, 222, 226), dtype=np.int16)


def _distance_sq(arr: np.ndarray, rgb: np.ndarray) -> np.ndarray:
    delta = arr.astype(np.int32) - rgb.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32)


def _band_bounds(arr: np.ndarray) -> tuple[int, int]:
    mask = _distance_sq(arr, _BAND) < 18 * 18
    columns = np.flatnonzero(mask.sum(axis=0) > arr.shape[0] * 0.70)
    if len(columns) < 30:
        raise ValueError("opaque gray band not recoverable from pixels")
    return int(columns[0]), int(columns[-1])


def _runs(ys: np.ndarray) -> list[np.ndarray]:
    if len(ys) == 0:
        return []
    cuts = np.flatnonzero(np.diff(ys) > 1) + 1
    return list(np.split(ys, cuts))


def _fit_ranked_curves(
    dark: np.ndarray,
    x_start: int,
    x_stop: int,
    center_x: float,
    scale: float,
) -> list[np.ndarray]:
    traces: list[list[tuple[float, float]]] = [[] for _ in range(4)]
    for x in range(max(0, x_start), min(dark.shape[1], x_stop)):
        groups = [group for group in _runs(np.flatnonzero(dark[:, x])) if 2 <= len(group) <= 16]
        if len(groups) != 4:
            continue
        centers = sorted(float(group.mean()) for group in groups)
        for rank, y in enumerate(centers):
            traces[rank].append((float(x), y))

    coefficients: list[np.ndarray] = []
    for rank, trace in enumerate(traces):
        if len(trace) < 48:
            raise ValueError(f"curve rank {rank} has only {len(trace)} usable columns")
        values = np.asarray(trace, dtype=np.float64)
        t = (values[:, 0] - center_x) / scale
        design = np.column_stack((t * t, t, np.ones_like(t)))
        fit, *_ = np.linalg.lstsq(design, values[:, 1], rcond=None)
        coefficients.append(fit)
    return coefficients


def _marker_centers(arr: np.ndarray, center_x: float) -> dict[str, tuple[tuple[float, float], tuple[float, float]]]:
    result = {}
    x_grid = np.broadcast_to(np.arange(arr.shape[1]), arr.shape[:2])
    for name, value in PALETTE.items():
        mask = _distance_sq(arr, np.array(value, dtype=np.int16)) < 62 * 62
        halves = []
        for side_mask in (mask & (x_grid < center_x), mask & (x_grid > center_x)):
            ys, xs = np.nonzero(side_mask)
            if len(xs) < 24:
                raise ValueError(f"missing {name} marker on one side of the band")
            halves.append((float(xs.mean()), float(ys.mean())))
        result[name] = (halves[0], halves[1])
    return result


def _bind_markers(
    fits: list[np.ndarray],
    markers: dict[str, tuple[tuple[float, float], tuple[float, float]]],
    side: int,
    center_x: float,
    scale: float,
) -> dict[str, np.ndarray]:
    bound = {}
    used: set[int] = set()
    for color, marker_pair in markers.items():
        x, y = marker_pair[side]
        t = (x - center_x) / scale
        predicted = [float(coeff[0] * t * t + coeff[1] * t + coeff[2]) for coeff in fits]
        rank = min(range(4), key=lambda index: abs(predicted[index] - y))
        if rank in used or abs(predicted[rank] - y) > 18.0:
            raise ValueError("colored markers do not bind uniquely to visible curves")
        used.add(rank)
        bound[color] = fits[rank]
    if len(bound) != 4:
        raise ValueError("not all four colors could be bound")
    return bound


def continuation_scores_from_image(image: Image.Image) -> dict[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 128 or arr.shape[1] < 128:
        raise ValueError("image is too small for curve recovery")
    band_left, band_right = _band_bounds(arr)
    center_x = (band_left + band_right) / 2.0
    scale = arr.shape[1] / 6.0
    dark = _distance_sq(arr, _DARK) < 58 * 58
    left_fits = _fit_ranked_curves(dark, 0, band_left - 12, center_x, scale)
    right_fits = _fit_ranked_curves(dark, band_right + 12, arr.shape[1], center_x, scale)
    markers = _marker_centers(arr, center_x)
    left_by_color = _bind_markers(left_fits, markers, 0, center_x, scale)
    right_by_color = _bind_markers(right_fits, markers, 1, center_x, scale)

    scores = {}
    for color in PALETTE:
        delta = left_by_color[color] - right_by_color[color]
        scores[color] = math.sqrt(
            float(delta[0] ** 2 + 0.36 * delta[1] ** 2 + 0.015625 * delta[2] ** 2)
        )
    return scores


def decision_from_image(image: Image.Image) -> str:
    scores = continuation_scores_from_image(image)
    return min(scores, key=scores.get)
