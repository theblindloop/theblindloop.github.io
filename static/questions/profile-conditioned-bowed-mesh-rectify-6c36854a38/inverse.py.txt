"""Independent raster-only inverse for the bowed mesh world."""
from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

N = 9
NAMES = ("plus", "X", "diamond", "square")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        q = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while q:
            yy, xx = q.popleft()
            xs.append(xx); ys.append(yy)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = yy + dy, xx + dx
                    if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        q.append((ny, nx))
        if len(xs) >= 18:
            out.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return out


def _single(mask: np.ndarray, label: str) -> tuple[float, float]:
    parts = _components(mask)
    if len(parts) != 1:
        raise ValueError(f"expected one {label} corner, found {len(parts)}")
    return parts[0][0], parts[0][1]


def _template(name: str) -> np.ndarray:
    # Independently specified public glyph prototypes; no renderer state is read.
    a = np.zeros((9, 9), dtype=np.uint8)
    if name == "plus":
        for k in range(9): a[4, k] = a[k, 4] = 1
    elif name == "X":
        for k in range(9): a[k, k] = a[k, 8-k] = 1
    elif name == "diamond":
        for r in range(9):
            for c in range(9):
                if abs(r-4) + abs(c-4) == 4: a[r, c] = 1
        a[4, 4] = 1
    elif name == "square":
        for r in range(2, 7):
            for c in range(2, 7):
                if r in (2, 6) or c in (2, 6): a[r, c] = 1
        a[4, 4] = 1
    return a


def reconstruct_cells(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    magenta = (r > 175) & (b > 125) & (g < 105) & ((r-g) > 90)
    internal = [(x, y) for x, y, area in _components(magenta) if 35 <= area <= 180]
    if len(internal) != 96:
        raise ValueError(f"mesh fiducials missing or merged: found {len(internal)} of 96")

    yellow = _single((r > 190) & (g > 135) & (b < 90), "yellow")
    orange = _single((r > 200) & (g < 135) & (b < 105), "orange")
    purple = _single((r > 90) & (r < 185) & (g < 115) & (b > 145), "purple")
    green = _single((r < 95) & (g > 125) & (b < 145), "green")
    points = internal + [yellow, orange, purple, green]
    points.sort(key=lambda p: p[1])
    rows = []
    for i in range(10):
        row = sorted(points[i*10:(i+1)*10], key=lambda p: p[0])
        if len(row) != 10:
            raise ValueError("incomplete fiducial row")
        rows.append(row)
    # Corner colors fix orientation and reject a plausible but wrong lattice order.
    for got, want in ((rows[0][0], yellow), (rows[0][-1], orange),
                      (rows[-1][-1], purple), (rows[-1][0], green)):
        if np.hypot(got[0]-want[0], got[1]-want[1]) > 3:
            raise ValueError("corner orientation inconsistent with declared order")

    out = np.zeros((9, 9), dtype=np.uint8)
    for rr in range(9):
        for cc in range(9):
            quad = [rows[rr][cc], rows[rr][cc+1], rows[rr+1][cc+1], rows[rr+1][cc]]
            cx = int(round(sum(p[0] for p in quad) / 4))
            cy = int(round(sum(p[1] for p in quad) / 4))
            patch = arr[max(0,cy-4):cy+5, max(0,cx-4):cx+5]
            if patch.size == 0:
                raise ValueError("cell sample falls outside raster")
            dark = np.all(patch < 105, axis=2)
            out[rr, cc] = int(float(dark.mean()) > 0.55)
    return out


def decision_and_margin(image: Image.Image) -> tuple[str, float]:
    cells = reconstruct_cells(image)
    scores = sorted((int(np.count_nonzero(cells != _template(name))), name) for name in NAMES)
    if len(scores) < 2 or scores[0][0] == scores[1][0]:
        raise ValueError("ambiguous rectified glyph")
    return scores[0][1], float(scores[1][0] - scores[0][0])


def decision_from_image(image: Image.Image) -> str:
    return decision_and_margin(image)[0]
