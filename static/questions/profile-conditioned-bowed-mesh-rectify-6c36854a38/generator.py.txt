"""Renderer for a glyph carried by a smoothly bowed 9x9 fiducial mesh."""
from __future__ import annotations

import math
import random
from typing import Iterable

import numpy as np
from PIL import Image, ImageDraw

SYMBOLS = ("plus", "X", "diamond", "square")
N = 9


def _template(name: str) -> np.ndarray:
    a = np.zeros((N, N), dtype=np.uint8)
    if name == "plus":
        a[4, :] = 1
        a[:, 4] = 1
    elif name == "X":
        for i in range(N):
            a[i, i] = a[i, N - 1 - i] = 1
    elif name == "diamond":
        for r in range(N):
            for c in range(N):
                if abs(r - 4) + abs(c - 4) == 4:
                    a[r, c] = 1
        a[4, 4] = 1
    elif name == "square":
        for r in range(2, 7):
            for c in range(2, 7):
                if r in (2, 6) or c in (2, 6):
                    a[r, c] = 1
        a[4, 4] = 1
    else:
        raise KeyError(name)
    return a


def _observed_cells(scene: dict) -> np.ndarray:
    a = _template(scene["symbol"]).copy()
    for r, c in scene["noise_cells"]:
        a[int(r), int(c)] ^= 1
    return a


def _scores(scene: dict) -> dict[str, int]:
    cells = _observed_cells(scene)
    return {name: int(np.count_nonzero(cells != _template(name))) for name in SYMBOLS}


def _node(scene: dict, r: int, c: int) -> tuple[float, float]:
    s = float(scene["cell_size"])
    u = c / N
    v = r / N
    x = float(scene["offset_x"]) + c * s
    y = float(scene["offset_y"]) + r * s
    # Two nonseparable smooth bows.  Bounds keep rows and columns ordered while
    # making a single global projective rectification insufficient.
    x += float(scene["x_amp"]) * math.sin(2 * math.pi * v + float(scene["phase_x"])) * (
        0.48 + 0.52 * math.sin(math.pi * u)
    )
    y += float(scene["y_amp"]) * math.sin(2 * math.pi * u + float(scene["phase_y"])) * (
        0.42 + 0.58 * math.sin(math.pi * v)
    )
    return x, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5EEDB0)
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    desired_flips = (0, 2, 5, 8)[(int(seed) // 4) % 4]
    base = _template(symbol)
    noise_cells: list[list[int]] = []
    for _ in range(200):
        picks = rng.sample(range(N * N), desired_flips)
        trial = base.copy()
        for p in picks:
            trial[p // N, p % N] ^= 1
        scores = sorted(
            (int(np.count_nonzero(trial != _template(name))), name) for name in SYMBOLS
        )
        if scores[0][1] == symbol and scores[1][0] - scores[0][0] >= 4:
            noise_cells = [[p // N, p % N] for p in sorted(picks)]
            break
    if desired_flips and not noise_cells:
        raise RuntimeError("could not sample an unambiguous corrupted glyph")

    distractors = []
    for i in range(16):
        side = i % 4
        if side == 0:
            x, y = rng.uniform(15, 38), rng.uniform(45, 595)
        elif side == 1:
            x, y = rng.uniform(602, 625), rng.uniform(45, 595)
        elif side == 2:
            x, y = rng.uniform(45, 595), rng.uniform(15, 38)
        else:
            x, y = rng.uniform(45, 595), rng.uniform(602, 625)
        distractors.append([round(x, 3), round(y, 3), rng.randint(2, 5)])

    return {
        "symbol": symbol,
        "x_amp": round(rng.uniform(64.0, 78.0), 6),
        "y_amp": round(rng.uniform(10.0, 16.0), 6),
        "phase_x": round(rng.uniform(0.0, 2 * math.pi), 7),
        "phase_y": round(rng.uniform(0.0, 2 * math.pi), 7),
        "cell_size": round(rng.uniform(43.5, 46.0), 6),
        "offset_x": round(rng.uniform(112.0, 124.0), 6),
        "offset_y": round(rng.uniform(111.0, 121.0), 6),
        "palette_variant": rng.randrange(3),
        "noise_cells": noise_cells,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    ss = 2
    img = Image.new("RGB", (640 * ss, 640 * ss), (250, 249, 245))
    d = ImageDraw.Draw(img)
    palettes = [
        ((25, 31, 38), (239, 239, 231)),
        ((48, 24, 34), (242, 238, 227)),
        ((22, 46, 38), (236, 240, 232)),
    ]
    ink, pale = palettes[int(scene["palette_variant"])]
    cells = _observed_cells(scene)
    nodes = [[_node(scene, r, c) for c in range(N + 1)] for r in range(N + 1)]

    def pts(values: Iterable[tuple[float, float]]) -> list[tuple[float, float]]:
        return [(round(x * ss), round(y * ss)) for x, y in values]

    for r in range(N):
        for c in range(N):
            poly = [nodes[r][c], nodes[r][c + 1], nodes[r + 1][c + 1], nodes[r + 1][c]]
            d.polygon(pts(poly), fill=ink if cells[r, c] else pale)

    line_color = (54, 174, 202)
    for r in range(N + 1):
        d.line(pts(nodes[r]), fill=line_color, width=3 * ss, joint="curve")
    for c in range(N + 1):
        d.line(pts(nodes[r][c] for r in range(N + 1)), fill=line_color, width=3 * ss)

    magenta = (222, 43, 180)
    corners = {(0, 0): (246, 197, 29), (0, N): (246, 88, 34),
               (N, N): (132, 63, 211), (N, 0): (35, 181, 88)}
    for r in range(N + 1):
        for c in range(N + 1):
            x, y = nodes[r][c]
            radius = 10 if (r, c) in corners else 5
            color = corners.get((r, c), magenta)
            d.ellipse(((x-radius)*ss, (y-radius)*ss, (x+radius)*ss, (y+radius)*ss), fill=color)

    for x, y, radius in scene["distractors"]:
        d.ellipse(((x-radius)*ss, (y-radius)*ss, (x+radius)*ss, (y+radius)*ss),
                  fill=(174, 178, 176))
    return img.resize((640, 640), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values())
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
