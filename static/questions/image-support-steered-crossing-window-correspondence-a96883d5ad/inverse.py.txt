"""Independent RGB-only inverse for the connected crossing-window question.

The inverse first isolates the single thread component and verifies that its
route is present throughout the distributed fields.  It then samples the
opposite-slope diagonal in each fixed window to distinguish an X crossing from
the parallel near-miss, reverses candidate window order, and compares codes.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

_THREAD = np.asarray((19, 97, 133), dtype=np.float32)
_FIELD_X0 = (70, 360, 650)
_FIELD_BASE = (250, 350, 450)
_WINDOW_OFFSET = 34
_WINDOW_STEP = 65
_WINDOW_COUNT = 4
_COLOR_RADIUS = 64.0
_ROUTE_HALF_BAND = 62
_NEGATIVE_SLOPE = -42.0 / 34.0


def _thread_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.sqrt(np.sum((array - _THREAD) ** 2, axis=2))
    return distance < _COLOR_RADIUS


def _single_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 180:
            components.append(points)
    if len(components) != 1:
        raise ValueError("thread is not one connected trace")
    component = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*components[0], strict=True)
    component[np.asarray(ys), np.asarray(xs)] = True
    return component


def _route_coverage(mask: np.ndarray) -> None:
    """Reject an erasure anywhere along the long field portions of the thread."""

    height, width = mask.shape
    for x0, base in zip(_FIELD_X0, _FIELD_BASE, strict=True):
        start = x0 + _WINDOW_OFFSET - 31
        end = x0 + _WINDOW_OFFSET + (_WINDOW_COUNT - 1) * _WINDOW_STEP + 31
        top = max(0, base - _ROUTE_HALF_BAND)
        bottom = min(height, base + _ROUTE_HALF_BAND + 1)
        for x in range(start, end + 1, 3):
            left = max(0, x - 1)
            right = min(width, x + 2)
            if not np.any(mask[top:bottom, left:right]):
                raise ValueError("thread route has an erased longitudinal section")


def _line_score(mask: np.ndarray, center_x: int, base: int, slope: float) -> int:
    score = 0
    for dx in range(-13, 14):
        x = center_x + dx
        y = int(round(base + slope * dx))
        y0 = max(0, y - 3)
        y1 = min(mask.shape[0], y + 4)
        x0 = max(0, x - 3)
        x1 = min(mask.shape[1], x + 4)
        if np.any(mask[y0:y1, x0:x1]):
            score += 1
    return score


def _field_code(mask: np.ndarray, x0: int, base: int) -> tuple[bool, ...]:
    bits: list[bool] = []
    for index in range(_WINDOW_COUNT):
        center = x0 + _WINDOW_OFFSET + index * _WINDOW_STEP
        positive = _line_score(mask, center, base, -_NEGATIVE_SLOPE)
        negative = _line_score(mask, center, base, _NEGATIVE_SLOPE)
        if positive < 16:
            raise ValueError("window thread is missing or ambiguous")
        # The near-miss's outgoing leg passes close to the opposite diagonal
        # near one end of the window.  Requiring almost the full diagonal
        # score separates that local proximity from a genuine X.
        if negative >= 22:
            bits.append(True)
        elif negative <= 18:
            bits.append(False)
        else:
            raise ValueError("self-crossing window is raster-ambiguous")
    return tuple(bits)


def _score(reference: tuple[bool, ...], candidate: tuple[bool, ...]) -> int:
    transformed = tuple(reversed(candidate))
    return sum(left == right for left, right in zip(reference, transformed, strict=True))


def decision_from_image(image: Image.Image) -> str:
    """Return ``A`` or ``B`` using only the supplied final raster."""

    mask = _single_component(_thread_mask(image))
    if mask.ndim != 2 or mask.shape[1] < 900 or mask.shape[0] < 560:
        raise ValueError("unexpected raster dimensions")
    _route_coverage(mask)
    reference = _field_code(mask, _FIELD_X0[0], _FIELD_BASE[0])
    candidate_a = _field_code(mask, _FIELD_X0[1], _FIELD_BASE[1])
    candidate_b = _field_code(mask, _FIELD_X0[2], _FIELD_BASE[2])
    score_a = _score(reference, candidate_a)
    score_b = _score(reference, candidate_b)
    if score_a == score_b or not math.isfinite(float(score_a + score_b)):
        raise ValueError("candidate crossing signatures are tied or non-finite")
    return "A" if score_a > score_b else "B"
