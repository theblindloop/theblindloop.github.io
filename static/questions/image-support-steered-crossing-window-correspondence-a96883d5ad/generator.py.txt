"""Renderer for the connected self-crossing-window correspondence world."""

from __future__ import annotations

import copy
import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "crossing-window-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 600
SCALE = 4
PAPER = (249, 247, 241)
PANEL = (237, 242, 240)
GUIDE = (193, 207, 208)
SOFT = (218, 226, 224)
TEXT = (42, 61, 74)
THREAD = (19, 97, 133)

FIELD_X0 = (70, 360, 650)
FIELD_BASE = (250, 350, 450)
WINDOW_CENTERS_OFFSET = 34
WINDOW_STEP = 65
WINDOW_COUNT = 4
QUARANTINE_MARGIN = 0.25


def _font(size: int):
    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size * SCALE)
    return ImageFont.load_default()


def _slot_codes(scene: dict) -> dict[str, tuple[bool, ...]]:
    return {
        str(record["slot"]): tuple(bool(value) for value in record["code"])
        for record in scene["candidate_codes"]
    }


def _score(reference: tuple[bool, ...], candidate: tuple[bool, ...]) -> int:
    transformed = tuple(reversed(candidate))
    return sum(left == right for left, right in zip(reference, transformed, strict=True))


def sample_scene(seed: int) -> dict:
    """Sample balanced, equal-weight reference and candidate codes."""

    rng = random.Random(int(seed))
    all_codes = [tuple(bool(bit) for bit in bits) for bits in itertools.product((0, 1), repeat=4)]
    balanced = [code for code in all_codes if sum(code) == 2]
    reference = list(rng.choice(balanced))
    # Avoid a palindromic reference so the visible half-turn is exercised.
    while tuple(reference) == tuple(reversed(reference)):
        reference = list(rng.choice(balanced))
    target = tuple(reversed(reference))
    wrong_pool = [code for code in balanced if code != target]
    wrong = list(rng.choice(wrong_pool))
    matching_slot = "A" if rng.randrange(2) == 0 else "B"
    records = [
        {"slot": "A", "code": list(target if matching_slot == "A" else wrong)},
        {"slot": "B", "code": list(target if matching_slot == "B" else wrong)},
    ]
    rng.shuffle(records)
    return {
        "reference_code": [bool(value) for value in reference],
        "candidate_codes": records,
    }


def analytic_gold(scene: dict) -> str:
    reference = tuple(bool(value) for value in scene["reference_code"])
    codes = _slot_codes(scene)
    scores = {slot: _score(reference, codes[slot]) for slot in ("A", "B")}
    if scores["A"] == scores["B"]:
        raise ValueError("candidate crossing signatures are tied")
    return "A" if scores["A"] > scores["B"] else "B"


def margin(scene: dict) -> float:
    reference = tuple(bool(value) for value in scene["reference_code"])
    codes = _slot_codes(scene)
    scores = [_score(reference, codes[slot]) for slot in ("A", "B")]
    return abs(scores[0] - scores[1]) / float(WINDOW_COUNT)


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < QUARANTINE_MARGIN
    except (KeyError, TypeError, ValueError):
        return True


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in box)


def _section_points(x0: int, base: int, code: tuple[bool, ...]) -> list[tuple[float, float]]:
    """Build one connected route section from four self-crossing modules.

    A true window contains two diagonals with opposite slope.  A false window
    keeps the same upper diagonal but replaces the return diagonal by a nearby
    parallel leg, so both cases have the same endpoints and a visible near-miss.
    """

    points: list[tuple[float, float]] = []
    for index, crossed in enumerate(code):
        center = float(x0 + WINDOW_CENTERS_OFFSET + index * WINDOW_STEP)
        module = [
            (center - 31.0, float(base)),
            (center - 17.0, float(base - 42)),
            (center + 17.0, float(base + 42)),
        ]
        if crossed:
            module.extend(
                [
                    (center + 17.0, float(base - 42)),
                    (center - 17.0, float(base + 42)),
                ]
            )
        else:
            module.extend(
                [
                    (center + 17.0, float(base + 17)),
                    (center - 17.0, float(base - 7)),
                ]
            )
        module.append((center + 31.0, float(base)))
        points.extend(module if not points else module)
    return points


def _draw_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    draw.text((int(xy[0] * SCALE), int(xy[1] * SCALE)), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    """Render one deterministic antialiased RGB raster."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    body_font = _font(15)
    label_font = _font(19)
    tiny_font = _font(11)

    _draw_text(draw, (40, 26), "CROSSING WINDOW CORRESPONDENCE", title_font, TEXT)
    _draw_text(
        draw,
        (40, 66),
        "Follow the single thread; rotate candidates 180 degrees before comparing crossings.",
        body_font,
        TEXT,
    )
    draw.rounded_rectangle(
        _scaled_box((42, 104, 918, 558)),
        radius=15 * SCALE,
        outline=GUIDE,
        width=2 * SCALE,
    )

    labels = ("R  REFERENCE", "A  CANDIDATE", "B  CANDIDATE")
    codes = _slot_codes(scene)
    sections = [
        _section_points(FIELD_X0[0], FIELD_BASE[0], tuple(bool(v) for v in scene["reference_code"])),
        _section_points(FIELD_X0[1], FIELD_BASE[1], codes["A"]),
        _section_points(FIELD_X0[2], FIELD_BASE[2], codes["B"]),
    ]
    for field_index, (x0, base, label) in enumerate(zip(FIELD_X0, FIELD_BASE, labels, strict=True)):
        left = x0 - 34
        right = x0 + WINDOW_CENTERS_OFFSET + (WINDOW_COUNT - 1) * WINDOW_STEP + 34
        top = base - 75
        bottom = base + 75
        draw.rounded_rectangle(
            _scaled_box((left, top, right, bottom)),
            radius=10 * SCALE,
            fill=PANEL,
            outline=SOFT,
            width=2 * SCALE,
        )
        bbox = draw.textbbox((0, 0), label, font=label_font)
        text_width = bbox[2] - bbox[0]
        _draw_text(draw, ((left + right) / 2 - text_width / (2 * SCALE), top + 12), label, label_font, TEXT)
        for index in range(WINDOW_COUNT):
            center = x0 + WINDOW_CENTERS_OFFSET + index * WINDOW_STEP
            draw.rounded_rectangle(
                _scaled_box((center - 25, base - 58, center + 25, base + 58)),
                radius=7 * SCALE,
                outline=GUIDE,
                width=1 * SCALE,
            )
            draw.line(
                [(int(center * SCALE), int((base - 53) * SCALE)), (int(center * SCALE), int((base + 53) * SCALE))],
                fill=SOFT,
                width=1 * SCALE,
            )
        if field_index < 2:
            divider = right + 15
            draw.line(
                [(int(divider * SCALE), int((top - 10) * SCALE)), (int(divider * SCALE), int((bottom + 10) * SCALE))],
                fill=GUIDE,
                width=1 * SCALE,
            )

    # The route sections share their endpoints through diagonal connectors;
    # drawing the complete point chain makes one physical trace across R, A, B.
    route_points: list[tuple[float, float]] = []
    for section in sections:
        route_points.extend(section)
    draw.line(
        [(int(x * SCALE), int(y * SCALE)) for x, y in route_points],
        fill=THREAD,
        width=8 * SCALE,
        joint="curve",
    )
    radius = 4.0
    for x, y in (route_points[0], route_points[-1]):
        draw.ellipse(_scaled_box((x - radius, y - radius, x + radius, y + radius)), fill=THREAD)

    _draw_text(
        draw,
        (40, 572),
        "A crossing is an X where the thread intersects itself; a near-miss is not a crossing.",
        tiny_font,
        TEXT,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the render-invariant permutation of slot-addressed records."""

    twin = copy.deepcopy(scene)
    twin["candidate_codes"] = list(reversed(twin["candidate_codes"]))
    return [("candidate_record_order", twin)]
