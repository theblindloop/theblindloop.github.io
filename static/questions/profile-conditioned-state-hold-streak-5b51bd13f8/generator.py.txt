"""Longest same-slot hold streak across an ordered panel sequence.

A vertical timeline of KPANEL=5 panels reads top to bottom. Every panel shows
the same horizontal track divided into NSLOT=6 slot columns by thin vertical
separators. Three tokens -- one red, one green, one blue -- each keep their own
fixed colour and occupy exactly one slot in every panel; the three tokens always
sit in three distinct slots, so no two tokens overlap.

A token's "hold streak" is the longest run of consecutive panels over which it
stays in the SAME slot without moving. The question is: which coloured token
stays in the same slot for the LONGEST consecutive run of panels across the
ordered sequence? The answer is that token's colour (one of the three token
colours).

The decision is a run-length argmax over each token's own temporal position
history, so no single panel, and no size / ink / colour-brightness / slot
intensity cue, determines it: it is the longest contiguous stretch of
self-consistency for one moving token, which must be recovered by replaying the
per-token slot sequence in panel order.

The latent scene stores ``positions``: a KPANEL x NCOL list where
``positions[panel][token]`` is the slot index (0..5) that token occupies in that
panel. The token index maps to a fixed colour via the module constant COLORS.
Every latent field is render-evident, so no nontrivial render-invariant latent
symmetry exists.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

KPANEL = 5
NSLOT = 6
NCOL = 3

LEFT_M = 46
TOP_M = 40
SLOT_W = 84
SLOT_H = 88
STRIP_H = 26
TOKEN_R = 15

BG = (255, 255, 255)
SEP = (168, 172, 180)

COLOR_NAMES = ["red", "green", "blue"]
COLORS = {
    "red": (215, 40, 40),
    "green": (40, 165, 70),
    "blue": (45, 80, 215),
}

WIDTH = LEFT_M + NSLOT * SLOT_W + LEFT_M
HEIGHT = TOP_M + KPANEL * SLOT_H + (KPANEL - 1) * STRIP_H + 40


def _panel_top(k: int) -> int:
    return TOP_M + k * (SLOT_H + STRIP_H)


def _slot_x(s: int) -> int:
    return LEFT_M + SLOT_W // 2 + s * SLOT_W


def _hold_streaks(positions: list) -> list[int]:
    """Per-token longest run of consecutive panels with an unchanged slot."""
    streaks = []
    for c in range(NCOL):
        best = 1
        cur = 1
        for k in range(1, KPANEL):
            if positions[k][c] == positions[k - 1][c]:
                cur += 1
                best = max(best, cur)
            else:
                cur = 1
        streaks.append(best)
    return streaks


def sample_scene(seed: int) -> dict:
    """Deterministic scene with a strict unique longest hold-streak winner."""
    rng = random.Random(seed * 1049 + 3)
    winner = rng.randrange(NCOL)
    run = rng.choice([2, 3])
    h0 = rng.randrange(0, KPANEL - run + 1)
    held = rng.randrange(NSLOT)

    wslots = []
    for k in range(KPANEL):
        if h0 <= k < h0 + run:
            wslots.append(held)
        else:
            s = rng.randrange(NSLOT - 1)
            if s >= held:
                s += 1
            wslots.append(s)

    others = [[0] * KPANEL, [0] * KPANEL]
    prev = [None, None]
    for k in range(KPANEL):
        taken = {wslots[k]}
        picks = []
        for j in range(2):
            cand = [s for s in range(NSLOT) if s not in taken and s != prev[j]]
            if not cand:
                cand = [s for s in range(NSLOT) if s not in taken]
            s = rng.choice(cand)
            taken.add(s)
            picks.append(s)
        others[0][k], others[1][k] = picks
        prev = picks

    positions = []
    for k in range(KPANEL):
        row = [None] * NCOL
        row[winner] = wslots[k]
        oi = 0
        for c in range(NCOL):
            if c == winner:
                continue
            row[c] = others[oi][k]
            oi += 1
        positions.append(row)

    scene = {"positions": positions}
    assert analytic_gold(scene) == COLOR_NAMES[winner]
    assert not is_quarantined(scene)
    return scene


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]

    for s in range(NSLOT + 1):
        x = LEFT_M + s * SLOT_W
        draw.line([x, 0, x, HEIGHT], fill=SEP, width=2)

    for k in range(KPANEL):
        top = _panel_top(k)
        cy = top + SLOT_H // 2
        for c in range(NCOL):
            slot = positions[k][c]
            cx = _slot_x(slot)
            draw.ellipse(
                [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R],
                fill=COLORS[COLOR_NAMES[c]],
            )
    return img


def analytic_gold(scene: dict) -> str:
    streaks = _hold_streaks(scene["positions"])
    best = max(streaks)
    winner = streaks.index(best)
    return COLOR_NAMES[winner]


def margin(scene: dict) -> float:
    streaks = _hold_streaks(scene["positions"])
    best = max(streaks)
    return float(best - (sorted(streaks)[-2]))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists.

    ``positions`` fixes the exact drawn slot of every uniquely-coloured token in
    every panel. Because token identity is carried by a fixed unique colour and
    the slot columns are shared and fixed, any change to ``positions`` changes
    the raster and any permutation of the token colours changes the raster. No
    reparametrization yields pixel-identical renders with a different gold, so
    no alias symmetry is declared.
    """
    return []
