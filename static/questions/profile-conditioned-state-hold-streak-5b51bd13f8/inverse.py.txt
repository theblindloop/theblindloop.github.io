"""Independent pixel-only inverse arm for the longest-hold-streak world.

Receives only the final PNG. From the raster alone it:
  1. labels every pixel against the fixed palette (background, slot separators,
     and the three token colours);
  2. finds the slot boundary columns from the full-height vertical separator
     lines, giving the shared slot grid;
  3. for each token colour, recovers each of its disks as a contiguous vertical
     run of that colour, sorts the disks by row to recover the ordered panel
     sequence, and quantizes each disk's horizontal centre to a slot index;
  4. computes each token's longest run of consecutive panels with an unchanged
     slot and returns the colour of the token with the longest hold streak.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
SEP = (168, 172, 180)
COLOR_NAMES = ["red", "green", "blue"]
COLORS = {
    "red": (215, 40, 40),
    "green": (40, 165, 70),
    "blue": (45, 80, 215),
}

PALETTE = [BG, SEP, COLORS["red"], COLORS["green"], COLORS["blue"]]
_BG, _SEP, _R, _G, _B = range(5)
_TOKEN_LABEL_TO_COLOR = {_R: "red", _G: "green", _B: "blue"}

KPANEL = 5
NCOL = 3
SLOT_W = 84
HEIGHT_TOL = 0.5


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _slot_boundaries(labels: np.ndarray) -> list[int]:
    """Column x-positions of the full-height vertical slot separators."""
    col_count = (labels == _SEP).sum(axis=0)
    cols = np.nonzero(col_count > labels.shape[0] * HEIGHT_TOL)[0]
    if len(cols) < 2:
        raise UnreadableImageError("could not recover slot separator columns")
    merged = [cols[0]]
    for c in cols[1:]:
        if c - merged[-1] > 2:
            merged.append(c)
    return merged


def _slot_index(boundaries: list[int], cx: float) -> int:
    idx = int(np.searchsorted(boundaries, cx))
    slot = idx - 1
    if slot < 0 or slot >= (len(boundaries) - 1):
        raise UnreadableImageError("token centre outside recovered slot grid")
    return slot


def _colour_disks(labels: np.ndarray, token_label: int) -> list[tuple[float, float, int]]:
    """Return (cx, cy, size) for each contiguous vertical run of one token colour,
    sorted by row (top to bottom = panel order)."""
    mask = labels == token_label
    row_has = mask.any(axis=1)
    disks = []
    y = 0
    h, w = mask.shape
    while y < h:
        if not row_has[y]:
            y += 1
            continue
        y0 = y
        while y < h and row_has[y]:
            y += 1
        ys, xs = np.nonzero(mask[y0:y, :])
        disks.append((float(xs.mean()), float((y0 + y - 1) / 2.0), int((y - y0) * mask[y0:y, :].sum())))
    if not disks or len(disks) != KPANEL:
        raise UnreadableImageError(f"token colour recovered {len(disks)} disks, expected {KPANEL}")
    disks.sort(key=lambda d: d[1])
    return disks


def _hold_streaks(positions: list) -> list[int]:
    streaks = []
    for c in range(NCOL):
        best = 1
        cur = 1
        for k in range(1, KPANEL):
            if positions[k][c] == positions[k - 1][c]:
                cur += 1
                best = max(best, cur)
            else:
                cur = 1
        streaks.append(best)
    return streaks


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)

    boundaries = _slot_boundaries(labels)

    positions = [[None] * NCOL for _ in range(KPANEL)]
    for token_label in (_R, _G, _B):
        color_name = _TOKEN_LABEL_TO_COLOR[token_label]
        disks = _colour_disks(labels, token_label)
        for k, (cx, cy, _size) in enumerate(disks):
            positions[k][COLOR_NAMES.index(color_name)] = _slot_index(boundaries, cx)

    streaks = _hold_streaks(positions)
    best = max(streaks)
    winner = streaks.index(best)
    return COLOR_NAMES[winner]
