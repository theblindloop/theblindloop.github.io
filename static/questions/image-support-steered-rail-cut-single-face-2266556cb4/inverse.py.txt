"""Independent image-only oracle for post-deletion enclosed-face count."""

from __future__ import annotations

import numpy as np
from PIL import Image

_BLACK = np.array([39, 44, 53], dtype=np.int16)
_YELLOW = np.array([244, 184, 34], dtype=np.int16)
_TOLERANCE = 30.0


def _near(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.linalg.norm(array.astype(np.int16) - color, axis=2) <= _TOLERANCE


def _components(mask: np.ndarray) -> list[tuple[int, bool]]:
    parents: list[int] = []
    sizes: list[int] = []
    edge_flags: list[bool] = []
    prior: list[tuple[int, int, int]] = []
    height, width = mask.shape

    def representative(item: int) -> int:
        while parents[item] != item:
            parents[item] = parents[parents[item]]
            item = parents[item]
        return item

    def join(a: int, b: int) -> None:
        a, b = representative(a), representative(b)
        if a != b:
            parents[b] = a

    for y in range(height):
        row = np.pad(mask[y], (1, 1), constant_values=False)
        transitions = np.flatnonzero(row[:-1] != row[1:])
        present: list[tuple[int, int, int]] = []
        for start, stop in zip(transitions[::2], transitions[1::2], strict=True):
            label = len(parents)
            parents.append(label)
            sizes.append(int(stop - start))
            edge_flags.append(y == 0 or y == height - 1 or start == 0 or stop == width)
            for old_start, old_stop, old_label in prior:
                if start < old_stop and old_start < stop:
                    join(label, old_label)
            present.append((int(start), int(stop), label))
        prior = present
    combined: dict[int, list[int | bool]] = {}
    for label, size in enumerate(sizes):
        root = representative(label)
        record = combined.setdefault(root, [0, False])
        record[0] = int(record[0]) + size
        record[1] = bool(record[1]) or edge_flags[label]
    return [(int(size), bool(edge)) for size, edge in combined.values()]


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if pixels.shape != (340, 420, 3):
        raise ValueError("unexpected canvas dimensions")
    black = _near(pixels, _BLACK)
    yellow = _near(pixels, _YELLOW)
    foreground = black | yellow
    if int(yellow.sum()) != 50 * 18:
        raise ValueError("missing or malformed marked rail section")
    foreground_area = int(foreground.sum())
    if foreground_area != 10440:
        raise ValueError("missing or extraneous railway ink")
    foreground_components = _components(foreground)
    if len(foreground_components) != 1:
        raise ValueError("the visible railway must be connected")
    enclosed = [size for size, border in _components(~black) if not border]
    if any(size < 1000 for size in enclosed):
        raise ValueError("undersized or ambiguous courtyard")
    return "yes" if len(enclosed) == 1 else "no"
