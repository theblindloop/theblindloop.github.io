"""Deterministic sampler and renderer for post-deletion railway faces."""

from __future__ import annotations

import copy
import random
from typing import Any

import numpy as np
from PIL import Image

WIDTH = 420
HEIGHT = 340
TRACK = 18
BLACK = [39, 44, 53]
YELLOW = [244, 184, 34]
BACKGROUND = [250, 249, 245]


def _segment(a: tuple[int, int], b: tuple[int, int], thickness: int = TRACK) -> list[int]:
    """Return a half-open solid rectangle centered on an orthogonal segment."""
    x0, y0 = a
    x1, y1 = b
    half = thickness // 2
    if y0 == y1:
        return [min(x0, x1) - half, y0 - half, max(x0, x1) + half, y0 + half]
    if x0 == x1:
        return [x0 - half, min(y0, y1) - half, x0 + half, max(y0, y1) + half]
    raise ValueError("rail segments must be orthogonal")


def _transform_rect(
    rect: list[int], horizontal: bool, vertical: bool, dx: int, dy: int
) -> list[int]:
    x0, y0, x1, y1 = rect
    if horizontal:
        x0, x1 = WIDTH - x1, WIDTH - x0
    if vertical:
        y0, y1 = HEIGHT - y1, HEIGHT - y0
    return [x0 + dx, y0 + dy, x1 + dx, y1 + dy]


def _mask(scene: dict[str, Any], key: str) -> np.ndarray:
    mask = np.zeros((int(scene["height"]), int(scene["width"])), dtype=bool)
    for x0, y0, x1, y1 in scene[key]:
        mask[int(y0):int(y1), int(x0):int(x1)] = True
    return mask


def _component_stats(mask: np.ndarray) -> list[tuple[int, bool]]:
    """Four-connected component areas and border incidence via row-run union."""
    parents: list[int] = []
    areas: list[int] = []
    borders: list[bool] = []
    previous: list[tuple[int, int, int]] = []
    height, width = mask.shape

    def root(label: int) -> int:
        while parents[label] != label:
            parents[label] = parents[parents[label]]
            label = parents[label]
        return label

    def union(left: int, right: int) -> None:
        left, right = root(left), root(right)
        if left != right:
            parents[right] = left

    for y, row in enumerate(mask):
        padded = np.pad(row, (1, 1), constant_values=False)
        edges = np.flatnonzero(padded[1:] != padded[:-1])
        current: list[tuple[int, int, int]] = []
        for start, end in zip(edges[::2], edges[1::2], strict=True):
            label = len(parents)
            parents.append(label)
            areas.append(int(end - start))
            borders.append(y in (0, height - 1) or start == 0 or end == width)
            for old_start, old_end, old_label in previous:
                if old_start < end and start < old_end:
                    union(label, old_label)
            current.append((int(start), int(end), label))
        previous = current
    totals: dict[int, list[int | bool]] = {}
    for label, area in enumerate(areas):
        representative = root(label)
        item = totals.setdefault(representative, [0, False])
        item[0] = int(item[0]) + area
        item[1] = bool(item[1]) or borders[label]
    return sorted(((int(a), bool(b)) for a, b in totals.values()), reverse=True)


def _bounded_white_areas(rail: np.ndarray) -> list[int]:
    return sorted(
        [area for area, touches_border in _component_stats(~rail) if not touches_border],
        reverse=True,
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Build a balanced ladder: deletion leaves zero, one, or two faces."""
    seed = int(seed)
    rng = random.Random(seed)
    positive = seed % 2 == 0
    if positive:
        rung_count = 3
    else:
        rung_count = 2 if (seed // 2) % 2 == 0 else 4

    xs = (92, 168, 244, 320)
    # A long, shallow interior chain: its decisive support is regional and
    # elongated rather than canvas-spanning or area-filling.
    top, bottom = 150, 190
    rails: list[list[int]] = []
    for left, right in zip(xs[:-1], xs[1:], strict=True):
        rails.append(_segment((left, top), (right, top)))
        rails.append(_segment((left, bottom), (right, bottom)))
    for index, x in enumerate(xs):
        if index < rung_count:
            rails.append(_segment((x, top), (x, bottom)))
        else:
            # A shorter outward tail replaces each absent rung. Its unique ink
            # equals a rung's after accounting for one versus two junctions.
            rails.append(_segment((x, top), (x, top - (bottom - top - TRACK))))

    yellow = [_segment((xs[0] + 22, top), (xs[1] - 22, top))]
    horizontal = bool((seed // 4) % 2)
    vertical = bool((seed // 8) % 2)
    dx = rng.choice((-10, -5, 0, 5, 10))
    dy = rng.choice((-2, 0, 2))
    rails = [_transform_rect(r, horizontal, vertical, dx, dy) for r in rails]
    yellow = [_transform_rect(r, horizontal, vertical, dx, dy) for r in yellow]
    if rng.random() < 0.5:
        rails.reverse()
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "rail_rectangles": rails,
        "yellow_rectangles": yellow,
        "black": list(BLACK),
        "yellow": list(YELLOW),
        "background": list(BACKGROUND),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    rail = _mask(scene, "rail_rectangles")
    yellow = _mask(scene, "yellow_rectangles")
    if np.any(yellow & ~rail):
        raise ValueError("the marked section must lie within the railway")
    array = np.empty((int(scene["height"]), int(scene["width"]), 3), dtype=np.uint8)
    array[:] = np.asarray(scene["background"], dtype=np.uint8)
    array[rail] = np.asarray(scene["black"], dtype=np.uint8)
    array[yellow] = np.asarray(scene["yellow"], dtype=np.uint8)
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    remaining = _mask(scene, "rail_rectangles") & ~_mask(scene, "yellow_rectangles")
    return "yes" if len(_bounded_white_areas(remaining)) == 1 else "no"


def margin(scene: dict[str, Any]) -> float:
    remaining = _mask(scene, "rail_rectangles") & ~_mask(scene, "yellow_rectangles")
    holes = _bounded_white_areas(remaining)
    smallest_hole_clearance = min((area ** 0.5 for area in holes), default=TRACK)
    return float(min(TRACK, smallest_hole_clearance))


def is_quarantined(scene: dict[str, Any]) -> bool:
    rail = _mask(scene, "rail_rectangles")
    yellow = _mask(scene, "yellow_rectangles")
    initial_holes = _bounded_white_areas(rail)
    return (
        int(yellow.sum()) != 50 * TRACK
        or np.any(yellow & ~rail)
        or len(_component_stats(rail)) != 1
        or not initial_holes
        or min(initial_holes) < 1000
        or margin(scene) < 12.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["rail_rectangles"] = list(reversed(twin["rail_rectangles"]))
    twin["yellow_rectangles"] = list(reversed(twin["yellow_rectangles"]))
    return [("reverse_union_draw_order", twin)]
