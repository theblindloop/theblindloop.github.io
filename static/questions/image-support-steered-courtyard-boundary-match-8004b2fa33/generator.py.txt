"""Deterministic renderer and analytic arm for courtyard boundary matching."""

from __future__ import annotations

import random
from typing import Any

import numpy as np
from PIL import Image

CANVAS = 512
PALETTE = {
    "background": [246, 243, 234],
    "teal": [35, 126, 132],
    "yellow": [244, 194, 46],
    "cyan": [24, 183, 214],
    "orange": [230, 102, 38],
}


def _overlap(a0: int, a1: int, b0: int, b1: int) -> int:
    return max(0, min(a1, b1) - max(a0, b0))


def _adjacent(rect: list[int], hole: list[int]) -> bool:
    x0, y0, x1, y1 = rect
    hx0, hy0, hx1, hy1 = hole
    vertical = (x1 == hx0 or x0 == hx1) and _overlap(y0, y1, hy0, hy1) > 0
    horizontal = (y1 == hy0 or y0 == hy1) and _overlap(x0, x1, hx0, hx1) > 0
    return vertical or horizontal


def _patch_rect(hole: list[int], side: str, length: int, thickness: int) -> list[int]:
    x0, y0, x1, y1 = hole
    if side in {"top", "bottom"}:
        cx = (x0 + x1) // 2
        lo, hi = cx - length // 2, cx + (length - length // 2)
        return [lo, y0 - thickness, hi, y0] if side == "top" else [lo, y1, hi, y1 + thickness]
    cy = (y0 + y1) // 2
    lo, hi = cy - length // 2, cy + (length - length // 2)
    return [x0 - thickness, lo, x0, hi] if side == "left" else [x1, lo, x1 + thickness, hi]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced geometry. Seed parity counterbalances answer identity."""

    rng = random.Random(int(seed))
    board_size = 216
    x0 = 148 + rng.randrange(0, 5)
    y0 = 148 + rng.randrange(0, 5)
    x1, y1 = x0 + board_size, y0 + board_size
    outer_x = 27 + rng.randrange(0, 3)
    outer_y = 29 + rng.randrange(0, 4)
    middle = 30 + rng.randrange(0, 3)
    usable = board_size - 2 * outer_x - middle
    left_width = usable // 2 + rng.randrange(-3, 4)
    right_width = usable - left_width
    holes = [
        [x0 + outer_x, y0 + outer_y, x0 + outer_x + left_width, y1 - outer_y],
        [x1 - outer_x - right_width, y0 + outer_y, x1 - outer_x, y1 - outer_y],
    ]

    match_color = "cyan" if int(seed) % 2 == 0 else "orange"
    other_color = "orange" if match_color == "cyan" else "cyan"
    reference_hole = rng.randrange(2)
    safe_sides = (("top", "bottom", "left"), ("top", "bottom", "right"))
    same_sides = rng.sample(list(safe_sides[reference_hole]), 2)
    other_side = rng.choice(safe_sides[1 - reference_hole])
    length = 30
    thickness = 14
    patches = [
        {"color": "yellow", "rect": _patch_rect(holes[reference_hole], same_sides[0], length, thickness)},
        {"color": match_color, "rect": _patch_rect(holes[reference_hole], same_sides[1], length, thickness)},
        {"color": other_color, "rect": _patch_rect(holes[1 - reference_hole], other_side, length, thickness)},
    ]
    rng.shuffle(patches)
    return {
        "canvas": [CANVAS, CANVAS],
        "board_rect": [x0, y0, x1, y1],
        "holes": holes,
        "patches": patches,
        "rotation": rng.randrange(4),
        "palette": {key: list(value) for key, value in PALETTE.items()},
    }


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = (int(v) for v in scene["canvas"])
    colors = scene["palette"]
    array = np.empty((height, width, 3), dtype=np.uint8)
    array[:] = np.asarray(colors["background"], dtype=np.uint8)
    x0, y0, x1, y1 = (int(v) for v in scene["board_rect"])
    array[y0:y1, x0:x1] = np.asarray(colors["teal"], dtype=np.uint8)
    for hole in scene["holes"]:
        hx0, hy0, hx1, hy1 = (int(v) for v in hole)
        array[hy0:hy1, hx0:hx1] = np.asarray(colors["background"], dtype=np.uint8)
    for patch in scene["patches"]:
        px0, py0, px1, py1 = (int(v) for v in patch["rect"])
        array[py0:py1, px0:px1] = np.asarray(colors[patch["color"]], dtype=np.uint8)
    array = np.rot90(array, int(scene["rotation"])).copy()
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    holes = scene["holes"]
    patches = scene["patches"]
    by_color: dict[str, int] = {}
    for patch in patches:
        hits = [index for index, hole in enumerate(holes) if _adjacent(patch["rect"], hole)]
        if len(hits) != 1:
            raise ValueError("every patch must edge-touch exactly one courtyard")
        by_color[str(patch["color"])] = hits[0]
    matches = [color for color in ("cyan", "orange") if by_color[color] == by_color["yellow"]]
    if len(matches) != 1:
        raise ValueError("exactly one candidate must match the reference courtyard")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    x0, y0, x1, y1 = (int(v) for v in scene["board_rect"])
    holes = [[int(v) for v in hole] for hole in scene["holes"]]
    patch_widths = [min(p["rect"][2] - p["rect"][0], p["rect"][3] - p["rect"][1]) for p in scene["patches"]]
    walls = [
        holes[0][0] - x0,
        holes[0][1] - y0,
        y1 - holes[0][3],
        holes[1][0] - holes[0][2],
        x1 - holes[1][2],
    ]
    return float(min(patch_widths + walls))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
