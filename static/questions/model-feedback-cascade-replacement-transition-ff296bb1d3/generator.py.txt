"""Cascade colour-replacement world (state-transition reasoning).

A single mutable state holds one colour. A KEY legend states a replacement
rule: for each source colour it shows the target colour it turns into under one
application (a function over the four-colour alphabet). A row of STEPS dots
tells the solver how many times to apply that rule in sequence. Starting from
the START disc colour, the solver iterates the replacement rule exactly K times
(K = the number of STEPS dots), each application changing the single in-motion
state, and answers with the final colour. The decision is the state reached
after the given number of discrete state transitions, not a static reachability
landing in a pre-drawn network.

The renderer owns scene sampling, rasterization and the structural analytic
gold (the final colour after iterating the KEY rule K times). The independent
pixel-only inverse arm (``oracle.py``) recovers the same decision from the final
PNG alone by reading the KEY mapping, the START colour and the dot count.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 560
HEIGHT = 640
BACKGROUND = (255, 255, 255)
GREY = (85, 85, 85)
LABEL = (40, 40, 40)

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
    (242, 146, 30),   # orange
]
COLOR_NAMES = ["red", "blue", "green", "orange"]

# KEY legend geometry: four rows (source order = COLOR order), top to bottom.
KEYX_SRC = 90
KEYX_TGT = 260
KEYY0 = 130
ROW_STEP = 62
SQUARE = 34

# START disc
START_CX = 150
START_CY = 470
START_R = 26

# STEPS dots
STEPS_Y = 545
STEPS_X0 = 170
STEPS_SPACING = 40
DOT_R = 9

K_LO, K_HI = 2, 5

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)
    # key_rows[i] = index of the target colour for source colour i.
    key_rows = [int(rng.integers(0, 4)) for _ in range(4)]
    start_index = int(rng.integers(0, 4))
    k = int(rng.integers(K_LO, K_HI + 1))
    return {
        "key_rows": key_rows,
        "start_index": start_index,
        "k": k,
    }


def _apply(colour: int, key_rows: list[int], steps: int) -> int:
    state = colour
    for _ in range(steps):
        state = key_rows[state]
    return state


def analytic_gold(scene: dict) -> str:
    """Structural gold: the final colour after ``k`` replacement applications."""
    final_index = _apply(scene["start_index"], scene["key_rows"], scene["k"])
    return COLOR_NAMES[final_index]


def margin(scene: dict) -> float:
    """The transition is fully deterministic; the decision is unambiguous."""
    return 1.0


def is_quarantined(scene: dict) -> bool:
    """No illegibility: colours are fixed, K is 2..5, geometry is fixed."""
    return False


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font_big = ImageFont.truetype(_FONT_PATH, 26)
        font_small = ImageFont.truetype(_FONT_PATH, 20)
    except OSError:
        font_big = ImageFont.load_default()
        font_small = ImageFont.load_default()

    title = "Replacement KEY"
    draw.text((40, 80), title, font=font_big, fill=LABEL)

    key_rows = scene["key_rows"]
    for row in range(4):
        y = KEYY0 + row * ROW_STEP
        src_colour = COLORS[row]
        tgt_colour = COLORS[key_rows[row]]
        draw.rectangle([KEYX_SRC, y, KEYX_SRC + SQUARE, y + SQUARE], fill=src_colour)
        draw.rectangle([KEYX_TGT, y, KEYX_TGT + SQUARE, y + SQUARE], fill=tgt_colour)
        arrow_x0 = KEYX_SRC + SQUARE + 10
        arrow_x1 = KEYX_TGT - 10
        mid_y = y + SQUARE // 2
        draw.line([arrow_x0, mid_y, arrow_x1, mid_y], fill=GREY, width=4)
        draw.polygon(
            [(arrow_x1, mid_y), (arrow_x1 - 14, mid_y - 9), (arrow_x1 - 14, mid_y + 9)],
            fill=GREY,
        )

    start_colour = COLORS[scene["start_index"]]
    draw.text((40, START_CY - 55), "START", font=font_big, fill=LABEL)
    draw.ellipse(
        [
            START_CX - START_R,
            START_CY - START_R,
            START_CX + START_R,
            START_CY + START_R,
        ],
        fill=start_colour,
    )

    draw.text((40, STEPS_Y - 24), "STEPS:", font=font_small, fill=LABEL)
    for i in range(scene["k"]):
        cx = STEPS_X0 + i * STEPS_SPACING
        draw.ellipse([cx - DOT_R, STEPS_Y - DOT_R, cx + DOT_R, STEPS_Y + DOT_R], fill=GREY)

    return img


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
