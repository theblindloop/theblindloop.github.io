"""Independent pixel-only inverse arm for the cascade colour-replacement world.

Recovers the decision (the final colour after iterating the KEY replacement rule
``k`` times from the START colour, where ``k`` is the number of STEPS dots) from
the final rasterized PNG alone. It reads the four KEY mapping rows by sampling
the fixed source/target squares, reads the START disc colour, counts the STEPS
dots by scanning the grey row band, applies the discovered mapping ``k`` times,
and returns the resulting colour name. No renderer / prompts / generate /
verify / analytic-gold code is imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

COLORS = [
    (222, 55, 60),    # red
    (45, 95, 215),    # blue
    (50, 160, 72),    # green
    (242, 146, 30),   # orange
]
COLOR_NAMES = ["red", "blue", "green", "orange"]
GREY = (85, 85, 85)

KEYX_SRC, KEYX_TGT, SQUARE = 90, 260, 34
KEYY0, ROW_STEP = 130, 62
START_CX, START_CY = 150, 470
STEPS_Y = 545
DOT_MIN_GAP = 6


def _nearest_colour(rgb: tuple[int, int, int]) -> int:
    best = 0
    best_d = float("inf")
    for index, colour in enumerate(COLORS):
        d = sum((rgb[i] - colour[i]) ** 2 for i in range(3))
        if d < best_d:
            best_d = d
            best = index
    return best


def _pixel(arr: np.ndarray, x: int, y: int) -> tuple[int, int, int]:
    return (int(arr[y, x, 0]), int(arr[y, x, 1]), int(arr[y, x, 2]))


def _count_dots(arr: np.ndarray) -> int:
    mask = np.all(arr == GREY, axis=-1)
    band = mask[STEPS_Y - 1 : STEPS_Y + 2]
    cols = np.any(band, axis=0)
    runs: list[int] = []
    in_run = False
    last = None
    for x in range(cols.shape[0]):
        if cols[x]:
            if not in_run or (last is not None and x - last > DOT_MIN_GAP):
                runs.append(x)
                in_run = True
            last = x
        else:
            in_run = False
    return len(runs)


def decision_from_image(image: Image.Image) -> str:
    """Return the final colour name after iterating the KEY rule K times."""
    arr = np.asarray(image.convert("RGB"))

    key_rows: list[int] = []
    for row in range(4):
        y0 = KEYY0 + row * ROW_STEP
        src_idx = _nearest_colour(_pixel(arr, KEYX_SRC + SQUARE // 2, y0 + SQUARE // 2))
        tgt_rgb = _pixel(arr, KEYX_TGT + SQUARE // 2, y0 + SQUARE // 2)
        key_rows.append(_nearest_colour(tgt_rgb))
        _ = src_idx  # source row order is fixed by the world's colour order

    start_index = _nearest_colour(_pixel(arr, START_CX, START_CY))
    k = _count_dots(arr)
    if k <= 0:
        k = 1

    state = start_index
    for _ in range(k):
        state = key_rows[state]
    return COLOR_NAMES[state]
