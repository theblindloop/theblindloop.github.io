"""Independent pixel-only inverse for the paired aperture relation.

It receives only the final raster. Saturated ink components reveal the ten
broken rings; an algebraic circle fit recovers each center and the largest
unpainted angular run recovers each aperture direction.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    output = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        if len(points) >= 500:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _fit_aperture(points_yx: np.ndarray) -> tuple[float, float, float]:
    y = points_yx[:, 0] + 0.5
    x = points_yx[:, 1] + 0.5
    design = np.column_stack((2.0 * x, 2.0 * y, np.ones_like(x)))
    target = x * x + y * y
    cx, cy, _ = np.linalg.lstsq(design, target, rcond=None)[0]
    angles = (np.degrees(np.arctan2(y - cy, x - cx)) + 360.0) % 360.0
    bins = np.zeros(360, dtype=bool)
    bins[np.floor(angles).astype(int) % 360] = True
    doubled = np.concatenate((bins, bins))
    best_start = best_len = run_start = run_len = 0
    for index, occupied in enumerate(doubled):
        if not occupied:
            if run_len == 0:
                run_start = index
            run_len += 1
            if run_len > best_len and run_len <= 360:
                best_start, best_len = run_start, run_len
        else:
            run_len = 0
    if best_len < 25:
        raise ValueError("ring aperture is not visible")
    gap = (best_start + (best_len - 1) / 2.0) % 360.0
    return float(cx), float(cy), float(gap)


def _delta(left: tuple[float, float, float], right: tuple[float, float, float]) -> float:
    return (right[2] - left[2]) % 360.0


def _distance(a: float, b: float) -> float:
    d = (a - b) % 360.0
    return min(d, 360.0 - d)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    mask = (chroma > 55) & (arr.max(axis=2) < 245)
    components = _components(mask)
    if len(components) != 10:
        raise ValueError(f"expected ten broken rings, found {len(components)}")
    rings = [_fit_aperture(component) for component in components]
    rings.sort(key=lambda item: item[1])
    reference = sorted(rings[:2], key=lambda item: item[0])
    candidates = sorted(rings[2:], key=lambda item: item[0])
    ref_delta = _delta(reference[0], reference[1])
    scores = []
    for index in range(4):
        pair = candidates[index * 2:index * 2 + 2]
        scores.append(_distance(_delta(pair[0], pair[1]), ref_delta))
    order = np.argsort(scores)
    if scores[int(order[0])] > 8.0 or scores[int(order[1])] - scores[int(order[0])] < 12.0:
        raise ValueError(f"ambiguous aperture relation scores {scores}")
    return LABELS[int(order[0])]
