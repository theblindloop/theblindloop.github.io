"""Deterministic broken-ring relation world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 520
BG = (250, 249, 245)
INK = (45, 48, 54)
MUTED = (132, 136, 142)
PALETTES = ("#1769aa", "#00897b", "#7b3fb0", "#c4512d")
LABELS = ("A", "B", "C", "D")
GAP_WIDTH = 48


def _circ_distance(a: float, b: float) -> float:
    d = (a - b) % 360.0
    return min(d, 360.0 - d)


def _pair_delta(pair: list[int]) -> float:
    return (float(pair[1]) - float(pair[0])) % 360.0


def _scores(scene: dict) -> list[float]:
    reference = _pair_delta(scene["reference_angles"])
    logical = [_circ_distance(_pair_delta(pair), reference) for pair in scene["candidate_angles"]]
    return [logical[index] for index in scene["candidate_order"]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_delta = rng.choice((60, 90, 120))
    reference_left = rng.randrange(0, 24) * 15
    deltas = [
        reference_delta,
        (360 - reference_delta) % 360,
        (reference_delta + 30) % 360,
        (reference_delta - 30) % 360,
    ]
    candidate_angles = []
    for delta in deltas:
        left = rng.randrange(0, 24) * 15
        candidate_angles.append([left, left + delta])
    order = [0, 1, 2, 3]
    rng.shuffle(order)
    return {
        "reference_angles": [reference_left, reference_left + reference_delta],
        "candidate_angles": candidate_angles,
        "candidate_order": order,
        "palette": rng.choice(PALETTES),
    }


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winner = min(range(4), key=lambda index: scores[index])
    return LABELS[winner]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return bool(scores[0] > 1e-9 or scores[1] - scores[0] < 20.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["reference_angles"] = [angle + 360 for angle in twin["reference_angles"]]
    twin["candidate_angles"] = [
        [angle + 360 for angle in pair] for pair in twin["candidate_angles"]
    ]
    return [("full_turn_angle_alias", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _draw_ring(draw: ImageDraw.ImageDraw, cx: int, cy: int, radius: int, gap: float, color: str) -> None:
    box = (cx - radius, cy - radius, cx + radius, cy + radius)
    start = (float(gap) + GAP_WIDTH / 2.0) % 360.0
    end = start + 360.0 - GAP_WIDTH
    draw.arc(box, start=start, end=end, fill=color, width=9)


def _draw_pair(draw: ImageDraw.ImageDraw, centers: tuple[int, int], cy: int, radius: int, angles: list[int], color: str) -> None:
    _draw_ring(draw, centers[0], cy, radius, angles[0], color)
    _draw_ring(draw, centers[1], cy, radius, angles[1], color)


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BG)
    draw = ImageDraw.Draw(image)

    def line(points, **kwargs):
        draw.line([(int(x * scale), int(y * scale)) for x, y in points], **kwargs)

    color = scene["palette"]
    title_font = _font(18 * scale, True)
    label_font = _font(20 * scale, True)
    small_font = _font(15 * scale, True)
    draw.text((450 * scale, 28 * scale), "REFERENCE", fill=INK, font=title_font, anchor="mm")
    _draw_pair(draw, (400 * scale, 500 * scale), 135 * scale, 38 * scale,
               scene["reference_angles"], color)
    line(((52, 215), (848, 215)), fill=MUTED, width=2 * scale)
    draw.text((450 * scale, 240 * scale), "CANDIDATES", fill=MUTED, font=small_font, anchor="mm")

    panel_centers = (117, 339, 561, 783)
    for display_index, panel_center in enumerate(panel_centers):
        logical_index = scene["candidate_order"][display_index]
        x0, x1 = panel_center - 103, panel_center + 103
        draw.rounded_rectangle(
            (x0 * scale, 270 * scale, x1 * scale, 485 * scale),
            radius=10 * scale,
            outline=(190, 193, 197),
            width=2 * scale,
        )
        _draw_pair(
            draw,
            ((panel_center - 43) * scale, (panel_center + 43) * scale),
            370 * scale,
            34 * scale,
            scene["candidate_angles"][logical_index],
            color,
        )
        draw.text(
            (panel_center * scale, 456 * scale),
            LABELS[display_index],
            fill=INK,
            font=label_font,
            anchor="mm",
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
