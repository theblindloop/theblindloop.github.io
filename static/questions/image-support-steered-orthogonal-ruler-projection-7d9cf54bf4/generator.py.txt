"""Deterministic renderer for the orthogonal-ruler projection world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

SIZE = 512
N = 7
CARD_W, CARD_H = 112, 56
CARD_X = (74, 330)
CARD_Y = 228
PALETTES = (
    ((25, 150, 135), (226, 126, 46)),
    ((48, 112, 196), (208, 72, 118)),
    ((116, 86, 190), (38, 158, 92)),
    ((18, 142, 190), (214, 92, 48)),
)
BACKGROUNDS = ((248, 246, 239), (244, 247, 245), (247, 244, 248))


def _grid(rows: Iterable[int], cols: Iterable[int]) -> list[list[int]]:
    row_set, col_set = set(rows), set(cols)
    return [[int(y in row_set or x in col_set) for x in range(N)] for y in range(N)]


def _templates() -> dict[str, list[list[list[int]]]]:
    return {
        "plus": [_grid([3], [3])],
        "tee": [
            _grid([0], [3]), _grid([6], [3]),
            _grid([3], [0]), _grid([3], [6]),
        ],
        "corner": [_grid([r], [c]) for r in (0, 6) for c in (0, 6)],
    }


def _scores(scene: dict) -> dict[str, int]:
    observed = _grid(scene["row_tabs"], scene["col_tabs"])
    scores: dict[str, int] = {}
    for name, variants in _templates().items():
        scores[name] = min(
            sum(a != b for row_a, row_b in zip(observed, target) for a, b in zip(row_a, row_b))
            for target in variants
        )
    return scores


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    category = int(seed) % 4
    if category == 0:  # exact centered cross
        row_tabs, col_tabs = [3], [3]
    elif category == 1:  # one centered axis and one edge axis
        edge = rng.choice((0, 6))
        if rng.randrange(2):
            row_tabs, col_tabs = [edge], [3]
        else:
            row_tabs, col_tabs = [3], [edge]
    elif category == 2:  # two edge axes
        row_tabs, col_tabs = [rng.choice((0, 6))], [rng.choice((0, 6))]
    else:  # exactly between plus and tee: explicit quarantined tie
        near_edge = rng.choice((1, 5))
        if rng.randrange(2):
            row_tabs, col_tabs = [near_edge], [3]
        else:
            row_tabs, col_tabs = [3], [near_edge]
    return {
        "row_tabs": row_tabs,
        "col_tabs": col_tabs,
        "swap_cards": bool(rng.randrange(2)),
        "quarter_turn": rng.randrange(4),
        "offset_x": rng.choice((-1, 0, 1)),
        "offset_y": rng.choice((-1, 0, 1)),
        "palette_index": rng.randrange(len(PALETTES)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "tab_inset": rng.choice((1, 2)),
    }


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    ordered = sorted((value, name) for name, value in scores.items())
    return "ambiguous" if ordered[0][0] == ordered[1][0] else ordered[0][1]


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values())
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "ambiguous" or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_card(
    image: Image.Image,
    *,
    x: int,
    y: int,
    role: str,
    active: int,
    color: tuple[int, int, int],
    tab_inset: int,
) -> None:
    draw = ImageDraw.Draw(image)
    frame = (53, 61, 69)
    card = (224, 228, 230)
    draw.rectangle((x, y, x + CARD_W, y + CARD_H), fill=card, outline=frame, width=2)
    marker = (x + 15, y + 28)
    if role == "row":
        draw.ellipse(
            (marker[0] - 8, marker[1] - 8, marker[0] + 8, marker[1] + 8),
            outline=color,
            width=3,
        )
    else:
        draw.polygon(
            ((marker[0], marker[1] - 9), (marker[0] - 8, marker[1] + 8),
             (marker[0] + 8, marker[1] + 8)),
            fill=color,
        )
    for index in range(N):
        cx = x + 36 + index * 10
        draw.rectangle((cx - 4, y + 16, cx + 4, y + 40), outline=frame, width=1)
        if index == active:
            half = 4 - tab_inset
            draw.rectangle((cx - half, y + 18 + tab_inset, cx + half, y + 38 - tab_inset), fill=color)


def render(scene: dict) -> Image.Image:
    background = BACKGROUNDS[int(scene["background_index"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (SIZE, SIZE), background)
    palette = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    row_x, col_x = CARD_X
    if bool(scene["swap_cards"]):
        row_x, col_x = col_x, row_x
    dx, dy = int(scene["offset_x"]), int(scene["offset_y"])
    _draw_card(
        image, x=row_x + dx, y=CARD_Y + dy, role="row",
        active=int(scene["row_tabs"][0]), color=palette[0], tab_inset=int(scene["tab_inset"]),
    )
    _draw_card(
        image, x=col_x + dx, y=CARD_Y + dy, role="column",
        active=int(scene["col_tabs"][0]), color=palette[1], tab_inset=int(scene["tab_inset"]),
    )
    turns = int(scene["quarter_turn"]) % 4
    for _ in range(turns):
        image = image.transpose(Image.Transpose.ROTATE_90)
    return image
