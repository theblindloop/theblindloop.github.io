"""Independent image-only inverse for marker-spacing variation.

This module deliberately repeats the small amount of raster measurement needed
to answer the question. It does not import the renderer or any scene-producing
module and receives only a PIL image.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        stack = [(start_y, start_x)]
        seen[start_y, start_x] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _row_centers(array: np.ndarray, lo: int, hi: int) -> list[float]:
    sub = array[:, lo:hi]
    maximum = sub.max(axis=2)
    minimum = sub.min(axis=2)
    # Colored interiors are saturated; gray borders, guides, and black labels
    # are intentionally excluded from this mask.
    mask = (maximum - minimum > 38) & (maximum < 250) & (minimum < 245)
    found: list[float] = []
    for component in _components(mask):
        if len(component) < 120:
            continue
        y0, x0 = component.min(axis=0)
        y1, x1 = component.max(axis=0)
        width = int(x1 - x0 + 1)
        height = int(y1 - y0 + 1)
        if not (14 <= width <= 42 and 14 <= height <= 42):
            continue
        ratio = width / height
        if 0.65 <= ratio <= 1.55:
            found.append(float(component[:, 1].mean() + lo))
    return sorted(found)


def _variation(centers: list[float]) -> float:
    if len(centers) != 5:
        raise ValueError("expected five visible marker components in each row")
    gaps = [b - a for a, b in zip(centers, centers[1:])]
    if min(gaps) <= 0:
        raise ValueError("marker centroids are not strictly ordered")
    mean = sum(gaps) / 4.0
    variance = sum((gap - mean) ** 2 for gap in gaps) / 4.0
    return math.sqrt(variance) / mean


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from saturated marker components in the raster alone."""

    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[1] < 200:
        raise ValueError("image has no usable paired marker rows")
    split = rgb.shape[1] // 2
    centers_a = _row_centers(rgb, 0, split)
    centers_b = _row_centers(rgb, split, rgb.shape[1])
    score_a = _variation(centers_a)
    score_b = _variation(centers_b)
    if abs(score_a - score_b) < 0.018:
        raise ValueError("spacing variation is below the raster decision margin")
    return "A" if score_a > score_b else "B"
