"""Deterministic renderer and analytic contract for marker-spacing variation."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "marker-spacing-variation-0.1.0"
SCALE = 3
CANVAS = (920, 520)
PANEL_X = (40, 480)
PANEL_WIDTH = 400
PANEL_TOP = 70
PANEL_BOTTOM = 450
PALETTE = (
    (28, 92, 190),
    (205, 46, 58),
    (25, 138, 76),
    (226, 112, 24),
    (140, 55, 170),
    (12, 143, 155),
)


def _coefficient(points: list[int] | tuple[int, ...]) -> float:
    gaps = [float(b - a) for a, b in zip(points, points[1:])]
    mean = sum(gaps) / len(gaps)
    variance = sum((gap - mean) ** 2 for gap in gaps) / len(gaps)
    return math.sqrt(variance) / mean


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, fully visible paired marker-row scene.

    The winner is balanced by seed parity, while geometry, colors, marker sizes,
    translations, and row heights come from independent-looking random draws.
    Both rows use the same marker-radius multiset and the same total horizontal
    span, removing easy total-ink and total-width cues.
    """

    rng = random.Random(int(seed))
    winner = "A" if int(seed) % 2 == 0 else "B"

    even_deltas = [rng.randint(-3, 3) for _ in range(3)]
    even_deltas.append(-sum(even_deltas))
    even_gaps = [60 + delta for delta in even_deltas]
    # These patterns all sum to 240 and have no gap close to a marker diameter.
    uneven_options = (
        [42, 78, 48, 72],
        [46, 74, 52, 68],
        [40, 82, 50, 68],
        [44, 70, 56, 70],
    )
    uneven_gaps = list(rng.choice(uneven_options))
    rng.shuffle(even_gaps)
    rng.shuffle(uneven_gaps)

    def positions(start: int, gaps: list[int]) -> list[int]:
        values = [start]
        for gap in gaps:
            values.append(values[-1] + gap)
        return values

    start_a = rng.randint(105, 145)
    start_b = rng.randint(545, 585)
    if winner == "A":
        gaps_a, gaps_b = uneven_gaps, even_gaps
    else:
        gaps_a, gaps_b = even_gaps, uneven_gaps

    radii = [rng.randint(10, 14) for _ in range(5)]
    colors = [list(color) for color in rng.sample(PALETTE, 2)]
    row_y = [rng.randint(190, 260), rng.randint(190, 260)]

    return {
        "points_a": positions(start_a, gaps_a),
        "points_b": positions(start_b, gaps_b),
        "marker_colors": colors,
        "radii": radii,
        "row_y": row_y,
        "canvas": list(CANVAS),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary into an RGB PNG image."""

    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    image = Image.new("RGB", (width * SCALE, height * SCALE), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    def sc(value: float) -> int:
        return int(round(value * SCALE))

    border = (135, 142, 150)
    guide = (213, 218, 224)
    text_color = (45, 53, 62)
    font = ImageFont.load_default(size=22 * SCALE)

    for panel_index, panel_x in enumerate(PANEL_X):
        x0, x1 = panel_x, panel_x + PANEL_WIDTH
        draw.rounded_rectangle(
            (sc(x0), sc(PANEL_TOP), sc(x1), sc(PANEL_BOTTOM)),
            radius=sc(12),
            outline=border,
            width=sc(2),
        )
        label = "A" if panel_index == 0 else "B"
        draw.text((sc(x0 + 18), sc(PANEL_TOP + 16)), label, fill=text_color, font=font)
        y = int(scene["row_y"][panel_index])
        draw.line(
            (sc(x0 + 45), sc(y + 62), sc(x1 - 45), sc(y + 62)),
            fill=guide,
            width=sc(2),
        )
        points = scene["points_a"] if panel_index == 0 else scene["points_b"]
        color = tuple(int(value) for value in scene["marker_colors"][panel_index])
        for x, radius in zip(points, scene["radii"]):
            draw.ellipse(
                (
                    sc(int(x) - int(radius)),
                    sc(y - int(radius)),
                    sc(int(x) + int(radius)),
                    sc(y + int(radius)),
                ),
                fill=color,
            )
        for x in points:
            draw.line(
                (sc(int(x)), sc(y + 58), sc(int(x)), sc(y + 66)),
                fill=guide,
                width=sc(1),
            )

    draw.line((sc(460), sc(95), sc(460), sc(425)), fill=(230, 233, 236), width=sc(1))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Select the row with the larger normalized adjacent-gap variation."""

    score_a = _coefficient(scene["points_a"])
    score_b = _coefficient(scene["points_b"])
    return "A" if score_a > score_b else "B"


def margin(scene: dict[str, Any]) -> float:
    """Absolute coefficient gap; this is the runner-up decision margin."""

    return abs(_coefficient(scene["points_a"]) - _coefficient(scene["points_b"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject ties, fragile spacing, and markers that would touch a panel edge."""

    points_a = scene["points_a"]
    points_b = scene["points_b"]
    radii = [int(value) for value in scene["radii"]]
    if margin(scene) < 0.08:
        return True
    for points, panel_x in ((points_a, PANEL_X[0]), (points_b, PANEL_X[1])):
        if len(points) != 5:
            return True
        if any(b - a <= radii[i] + radii[i + 1] + 6 for i, (a, b) in enumerate(zip(points, points[1:]))):
            return True
        if min(points) - max(radii) <= panel_x + 8 or max(points) + max(radii) >= panel_x + PANEL_WIDTH - 8:
            return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared for this directly rendered parameterization."""

    return []
