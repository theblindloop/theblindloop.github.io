"""Independent image-only inverse for the exact-balance board."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 1100
HEIGHT = 720
ROW_Y = (190, 315, 440)
QUERY_Y = 615
COLORS = (
    (43, 104, 189),
    (207, 78, 68),
    (43, 148, 103),
    (145, 79, 176),
)
INK = np.array((33, 39, 48), dtype=np.int16)
QUERY_BLUE = np.array((43, 118, 174), dtype=np.int16)


class OracleError(ValueError):
    pass


def _near_mask(patch: np.ndarray, color: tuple[int, int, int] | np.ndarray, tolerance: int = 18) -> np.ndarray:
    target = np.asarray(color, dtype=np.int32)
    delta = patch.astype(np.int32) - target
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _fraction_near(array: np.ndarray, box: tuple[int, int, int, int], color: np.ndarray, tolerance: int = 20) -> float:
    x0, y0, x1, y1 = box
    patch = array[y0:y1, x0:x1]
    if patch.size == 0:
        return 0.0
    return float(np.mean(_near_mask(patch, color, tolerance)))


def _runs(active: np.ndarray) -> list[tuple[int, int]]:
    padded = np.concatenate((np.array([False]), active, np.array([False])))
    changes = np.flatnonzero(padded[1:] != padded[:-1])
    return [(int(changes[i]), int(changes[i + 1])) for i in range(0, len(changes), 2)]


def _read_bundle(array: np.ndarray, box: tuple[int, int, int, int], label: str) -> tuple[int, int, int, int]:
    x0, y0, x1, y1 = box
    patch = array[y0:y1, x0:x1]
    counts = []
    for symbol, color in enumerate(COLORS):
        mask = _near_mask(patch, color, 18)
        components = []
        for start, stop in _runs(np.sum(mask, axis=0) >= 8):
            width = stop - start
            area = int(np.sum(mask[:, start:stop]))
            if width < 22 or area < 500:
                raise OracleError(f"{label}: malformed color-{symbol} fragment")
            if width > 50 or area > 1800:
                raise OracleError(f"{label}: merged or oversized color-{symbol} token")
            components.append((start, stop))
        counts.append(len(components))
    result = tuple(counts)
    if any(count > 4 for count in result) or not 1 <= sum(result) <= 5:
        raise OracleError(f"{label}: invalid token count vector {result}")
    return result


def _determinant3(matrix: list[list[int]]) -> int:
    return (
        matrix[0][0] * (matrix[1][1] * matrix[2][2] - matrix[1][2] * matrix[2][1])
        - matrix[0][1] * (matrix[1][0] * matrix[2][2] - matrix[1][2] * matrix[2][0])
        + matrix[0][2] * (matrix[1][0] * matrix[2][1] - matrix[1][1] * matrix[2][0])
    )


def _recover_ratio(rows: list[tuple[tuple[int, ...], tuple[int, ...]]]) -> tuple[int, ...]:
    coefficients = [[left[i] - right[i] for i in range(4)] for left, right in rows]
    vector = []
    for omitted in range(4):
        square = [[row[i] for i in range(4) if i != omitted] for row in coefficients]
        vector.append((1 if omitted % 2 == 0 else -1) * _determinant3(square))
    if all(value < 0 for value in vector):
        vector = [-value for value in vector]
    if not all(value > 0 for value in vector):
        raise OracleError("the visible balances do not have one positive ratio")
    divisor = math.gcd(*vector)
    weights = tuple(value // divisor for value in vector)
    if sorted(weights) != [2, 3, 5, 7]:
        raise OracleError(f"unexpected visible weight ratio {weights}")
    for left, right in rows:
        if sum((left[i] - right[i]) * weights[i] for i in range(4)) != 0:
            raise OracleError("a visible premise row is not an exact balance")
    return weights


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleError(f"unexpected image size {rgb.size}")
    array = np.asarray(rgb)

    for y in ROW_Y:
        if _fraction_near(array, (536, y - 9, 565, y - 5), INK, 22) < 0.90:
            raise OracleError("an equality sign is missing or damaged")
    if _fraction_near(array, (300, 525, 800, 530), QUERY_BLUE, 22) < 0.55:
        raise OracleError("the QUESTION frame is missing or damaged")

    rows = []
    for index, y in enumerate(ROW_Y, start=1):
        left = _read_bundle(array, (100, y - 38, 470, y + 38), f"balance {index} left")
        right = _read_bundle(array, (630, y - 38, 1000, y + 38), f"balance {index} right")
        if left == right:
            raise OracleError("a premise repeats the same bundle")
        rows.append((left, right))
    weights = _recover_ratio(rows)

    left_query = _read_bundle(array, (100, QUERY_Y - 39, 470, QUERY_Y + 39), "query left")
    right_query = _read_bundle(array, (630, QUERY_Y - 39, 1000, QUERY_Y + 39), "query right")
    if sum(left_query) != sum(right_query):
        raise OracleError("query sides do not have equal cardinality")
    gap = sum((left_query[i] - right_query[i]) * weights[i] for i in range(4))
    if abs(gap) > 1:
        raise OracleError("query is outside the declared one-unit boundary")
    if gap > 0:
        return "left"
    if gap < 0:
        return "right"
    return "same"
