"""Independent raster-only inverse arm for convex-hull fill."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "blue": (42, 105, 178),
    "orange": (225, 112, 28),
    "green": (38, 151, 82),
}
NAMES = tuple(PALETTE)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        group: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            group.append((y, x))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(group) >= 45:
            groups.append(group)
    return groups


def _centers(pixels: np.ndarray, rgb: tuple[int, int, int], name: str) -> list[tuple[float, float]]:
    target = np.asarray(rgb, dtype=np.int16)
    mask = np.max(np.abs(pixels - target), axis=2) <= 2
    groups = _components(mask)
    if len(groups) != 8:
        raise ValueError(f"expected eight isolated {name} markers, found {len(groups)}")
    return [
        (sum(x for _, x in group) / len(group), sum(y for y, _ in group) / len(group))
        for group in groups
    ]


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    ordered = sorted(set(points))
    if len(ordered) < 3:
        raise ValueError("not enough distinct marker centers")
    lower: list[tuple[float, float]] = []
    for point in ordered:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 0:
            lower.pop()
        lower.append(point)
    upper: list[tuple[float, float]] = []
    for point in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 0:
            upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]


def _fill(points: list[tuple[float, float]]) -> float:
    xs = [p[0] for p in points]
    ys = [p[1] for p in points]
    rectangle = (max(xs) - min(xs)) * (max(ys) - min(ys))
    if rectangle <= 0:
        raise ValueError("degenerate marker bounding rectangle")
    hull = _hull(points)
    area = abs(sum(x1 * y2 - y1 * x2 for (x1, y1), (x2, y2) in zip(hull, hull[1:] + hull[:1]))) / 2.0
    return area / rectangle


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    values = {name: _fill(_centers(pixels, PALETTE[name], name)) for name in NAMES}
    ordered = sorted(NAMES, key=values.get, reverse=True)
    if values[ordered[0]] - values[ordered[1]] < 0.015:
        raise ValueError("pixel hull-fill comparison is too close to resolve")
    return ordered[0]
