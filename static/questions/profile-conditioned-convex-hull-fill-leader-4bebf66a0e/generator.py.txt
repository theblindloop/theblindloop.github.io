"""Deterministic renderer and analytic arm for convex_hull_fill_leader."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 480
COLORS = {
    "blue": (42, 105, 178),
    "orange": (225, 112, 28),
    "green": (38, 151, 82),
}
NAMES = tuple(COLORS)
PANELS = ((35, 82, 225, 414), (265, 82, 455, 414), (495, 82, 685, 414))
POINT_RADIUS = 7
QUARANTINE_GAP = 0.075


def _cross(o: tuple[float, float], a: tuple[float, float], b: tuple[float, float]) -> float:
    return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    ordered = sorted(set(points))
    if len(ordered) < 3:
        raise ValueError("a hull requires at least three distinct points")
    lower: list[tuple[float, float]] = []
    for point in ordered:
        while len(lower) >= 2 and _cross(lower[-2], lower[-1], point) <= 0:
            lower.pop()
        lower.append(point)
    upper: list[tuple[float, float]] = []
    for point in reversed(ordered):
        while len(upper) >= 2 and _cross(upper[-2], upper[-1], point) <= 0:
            upper.pop()
        upper.append(point)
    return lower[:-1] + upper[:-1]


def _polygon_area(points: list[tuple[float, float]]) -> float:
    return abs(sum(x1 * y2 - y1 * x2 for (x1, y1), (x2, y2) in zip(points, points[1:] + points[:1]))) / 2.0


def _score(points: list[tuple[float, float]] | list[list[float]]) -> float:
    values = [(float(x), float(y)) for x, y in points]
    xs = [p[0] for p in values]
    ys = [p[1] for p in values]
    rectangle = (max(xs) - min(xs)) * (max(ys) - min(ys))
    if rectangle <= 0:
        raise ValueError("degenerate bounding rectangle")
    return _polygon_area(_hull(values)) / rectangle


def _pixel_points(points: list[list[float]], panel_index: int) -> list[tuple[int, int]]:
    left, top, right, bottom = PANELS[panel_index]
    pad_x, pad_y = 20, 22
    return [
        (
            round(left + pad_x + float(x) * (right - left - 2 * pad_x)),
            round(bottom - pad_y - float(y) * (bottom - top - 2 * pad_y)),
        )
        for x, y in points
    ]


def _normalized_cloud(rng: random.Random) -> list[list[float]]:
    raw = [(rng.random(), rng.random()) for _ in range(8)]
    xs = [p[0] for p in raw]
    ys = [p[1] for p in raw]
    xlo, xhi = min(xs), max(xs)
    ylo, yhi = min(ys), max(ys)
    points = [
        [
            round(0.08 + 0.84 * (x - xlo) / (xhi - xlo), 6),
            round(0.08 + 0.84 * (y - ylo) / (yhi - ylo), 6),
        ]
        for x, y in raw
    ]
    pixel = _pixel_points(points, 0)
    if any(math.dist(pixel[i], pixel[j]) < 2 * POINT_RADIUS + 7 for i in range(8) for j in range(i + 1, 8)):
        raise ValueError("crowded cloud")
    return points


def _scores(series: dict[str, list[list[float]]]) -> dict[str, float]:
    # All panels share the same geometry, so one canonical panel removes position
    # from the analytic decision while retaining final-raster rounding.
    return {name: _score(_pixel_points(series[name], 0)) for name in NAMES}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = NAMES[int(seed) % len(NAMES)]
    near = int(seed) % 7 == 0
    gap_band = (0.025, 0.068) if near else (0.12, 0.28)
    for _ in range(30000):
        clouds: list[list[list[float]]] = []
        try:
            clouds = [_normalized_cloud(rng) for _ in NAMES]
        except ValueError:
            continue
        raw_scores = [_score(_pixel_points(cloud, 0)) for cloud in clouds]
        ranked = sorted(range(3), key=lambda i: raw_scores[i], reverse=True)
        gap = raw_scores[ranked[0]] - raw_scores[ranked[1]]
        if not gap_band[0] <= gap <= gap_band[1]:
            continue
        remaining_names = [name for name in NAMES if name != desired]
        rng.shuffle(remaining_names)
        assignment = {desired: clouds[ranked[0]]}
        assignment[remaining_names[0]] = clouds[ranked[1]]
        assignment[remaining_names[1]] = clouds[ranked[2]]
        panel_order = list(NAMES)
        rng.shuffle(panel_order)
        return {"series": assignment, "panel_order": panel_order}
    raise RuntimeError(f"could not sample hull-fill scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 249, 247))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((35, 22), "POINT-CLOUD HULL FILL", fill=(34, 39, 43), font=font)
    draw.text((35, 43), "fill = tightest-polygon area / axis-aligned bounding-rectangle area", fill=(70, 75, 79), font=font)
    for panel_index, name in enumerate(scene["panel_order"]):
        left, top, right, bottom = PANELS[panel_index]
        draw.rectangle((left, top, right, bottom), fill=(255, 255, 255), outline=(115, 121, 126), width=2)
        for step in range(1, 4):
            x = round(left + step * (right - left) / 4)
            y = round(top + step * (bottom - top) / 4)
            draw.line((x, top + 2, x, bottom - 2), fill=(229, 232, 233), width=1)
            draw.line((left + 2, y, right - 2, y), fill=(229, 232, 233), width=1)
        for x, y in _pixel_points(scene["series"][name], panel_index):
            draw.ellipse((x - POINT_RADIUS, y - POINT_RADIUS, x + POINT_RADIUS, y + POINT_RADIUS), fill=COLORS[name], outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    values = _scores(scene["series"])
    return max(NAMES, key=values.get)


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene["series"]).values(), reverse=True)
    return ordered[0] - ordered[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "series": {name: list(reversed(scene["series"][name])) for name in NAMES},
        "panel_order": list(scene["panel_order"]),
    }
    return [("reverse_point_storage_order", twin)]
