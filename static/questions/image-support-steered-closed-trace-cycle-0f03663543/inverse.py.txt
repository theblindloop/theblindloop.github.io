"""Independent pixel-only inverse for closed versus open dark routes.

The inverse does not import the renderer or any scene metadata. It thresholds
the dark ink, checks that the ink is one connected component, then builds
adjacency between the complementary background components and the trace. A
background component that cannot reach the crop border is a raster witness of
a return cycle in the trace.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _dark_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # The route is the only low-maximum-luminance ink. Colored dots have a
    # bright channel and are intentionally ignored as nuisance marks.
    return np.all(pixels < np.array([125, 125, 125], dtype=np.uint8), axis=2)


def _neighbors(y: int, x: int, height: int, width: int):
    if y:
        yield y - 1, x
    if y + 1 < height:
        yield y + 1, x
    if x:
        yield y, x - 1
    if x + 1 < width:
        yield y, x + 1


def _component_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        count += 1
        queue = deque([(y, x)])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in _neighbors(cy, cx, height, width):
                if mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return count


def _bounded_background_count(trace: np.ndarray) -> int:
    """Flood the outside, then count enclosed background components and adjacency."""

    free = ~trace
    height, width = free.shape
    outside = np.zeros_like(free, dtype=bool)
    starts = []
    for x in range(width):
        starts.extend(((0, x), (height - 1, x)))
    for y in range(height):
        starts.extend(((y, 0), (y, width - 1)))
    queue = deque()
    for y, x in starts:
        if free[y, x] and not outside[y, x]:
            outside[y, x] = True
            queue.append((y, x))
    while queue:
        cy, cx = queue.popleft()
        for ny, nx in _neighbors(cy, cx, height, width):
            if free[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))

    enclosed_seen = np.zeros_like(free, dtype=bool)
    enclosed = 0
    for y, x in np.argwhere(free & ~outside):
        y, x = int(y), int(x)
        if enclosed_seen[y, x]:
            continue
        enclosed += 1
        queue = deque([(y, x)])
        enclosed_seen[y, x] = True
        touches_trace = False
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in _neighbors(cy, cx, height, width):
                if trace[ny, nx]:
                    touches_trace = True
                elif free[ny, nx] and not enclosed_seen[ny, nx]:
                    enclosed_seen[ny, nx] = True
                    queue.append((ny, nx))
        if not touches_trace:
            raise ValueError("enclosed background component has no trace adjacency")
    return enclosed


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff the dark connected route contains a closed return cycle."""

    dark = _dark_mask(image)
    ys, xs = np.nonzero(dark)
    if not len(ys):
        raise ValueError("no dark route detected")
    if _component_count(dark) != 1:
        raise ValueError("dark route is not one connected component")
    pad = 16
    y0 = max(0, int(ys.min()) - pad)
    y1 = min(dark.shape[0], int(ys.max()) + pad + 1)
    x0 = max(0, int(xs.min()) - pad)
    x1 = min(dark.shape[1], int(xs.max()) + pad + 1)
    trace = dark[y0:y1, x0:x1]
    return "yes" if _bounded_background_count(trace) else "no"
