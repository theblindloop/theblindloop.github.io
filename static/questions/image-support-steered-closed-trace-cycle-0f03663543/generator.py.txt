"""Deterministic latent scenes and RGB rendering for the closed-trace world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = (255, 255, 255)
INK = (29, 43, 57)
RED = (216, 62, 70)
GREEN = (27, 158, 101)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a local trace and independently randomized nuisance geometry."""

    rng = random.Random(int(seed))
    # Alternating by the public scene seed keeps every stress gallery balanced.
    route_type = "closed" if int(seed) % 2 == 0 else "open"
    loop_width = rng.choice((148.0, 156.0, 164.0, 172.0))
    loop_height = rng.choice((20.0, 22.0, 24.0))
    stroke_width = rng.choice((8.0, 9.0))
    gap = rng.uniform(18.0, 29.0)
    # Positions sit in different coarse cells while the trace remains a small inset.
    col = rng.randrange(1, 7)
    row = rng.randrange(1, 7)
    offset_x = 100.0 * col + 50.0 + rng.uniform(-7.0, 7.0)
    offset_y = 100.0 * row + 50.0 + rng.uniform(-3.0, 3.0)
    green_dx = -(loop_width / 2.0 + rng.uniform(40.0, 52.0))
    green_dy = -(loop_height / 2.0 + rng.uniform(35.0, 48.0))
    red_dx = loop_width / 2.0 + rng.uniform(40.0, 52.0)
    red_dy = loop_height / 2.0 + rng.uniform(35.0, 48.0)
    return {
        "route_type": route_type,
        "offset_x": round(offset_x, 3),
        "offset_y": round(offset_y, 3),
        "loop_width": loop_width,
        "loop_height": loop_height,
        "stroke_width": stroke_width,
        "gap": round(gap, 3),
        "green_dx": round(green_dx, 3),
        "green_dy": round(green_dy, 3),
        "red_dx": round(red_dx, 3),
        "red_dy": round(red_dy, 3),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one compact route plus two nondiagnostic colored dots."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    x = float(scene["offset_x"])
    y = float(scene["offset_y"])
    half_w = float(scene["loop_width"]) / 2.0
    half_h = float(scene["loop_height"]) / 2.0
    width = int(round(float(scene["stroke_width"])))
    left, right = x - half_w, x + half_w
    top, bottom = y - half_h, y + half_h
    if scene["route_type"] == "closed":
        points = [(left, top), (right, top), (right, bottom), (left, bottom), (left, top)]
    elif scene["route_type"] == "open":
        gap = float(scene["gap"])
        points = [(left, top + gap), (left, bottom), (right, bottom), (right, top), (left, top)]
    else:
        raise ValueError(f"unknown route_type: {scene['route_type']!r}")
    draw.line(points, fill=INK, width=width, joint="curve")

    green_x = x + float(scene["green_dx"])
    green_y = y + float(scene["green_dy"])
    red_x = x + float(scene["red_dx"])
    red_y = y + float(scene["red_dy"])
    marker_radius = 11
    draw.ellipse(
        (green_x - marker_radius, green_y - marker_radius,
         green_x + marker_radius, green_y + marker_radius),
        fill=GREEN,
    )
    draw.ellipse(
        (red_x - marker_radius, red_y - marker_radius,
         red_x + marker_radius, red_y + marker_radius),
        fill=RED,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """The visible route class is the only analytic label input."""

    return "yes" if scene["route_type"] == "closed" else "no"


def margin(scene: dict[str, Any]) -> float:
    """One-sided pixel margin to the route's observable topology boundary."""

    x = float(scene["offset_x"])
    y = float(scene["offset_y"])
    half_w = float(scene["loop_width"]) / 2.0
    half_h = float(scene["loop_height"]) / 2.0
    stroke = float(scene["stroke_width"])
    edge_clearance = min(
        x - half_w - stroke - 2.0,
        WIDTH - x - half_w - stroke - 2.0,
        y - half_h - stroke - 2.0,
        HEIGHT - y - half_h - stroke - 2.0,
    )
    topology_clearance = float(scene["gap"]) - 8.0 if scene["route_type"] == "open" else stroke
    hole_clearance = half_h - stroke / 2.0
    return round(max(0.1, min(edge_clearance, topology_clearance, hole_clearance)), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude hairlines, near-closures, and border-near routes."""

    return (
        margin(scene) < 5.0
        or float(scene["gap"]) < 10.0
        or float(scene["stroke_width"]) < 7.0
        or float(scene["loop_height"]) < 18.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; all label inputs are exposed and rendered."""

    return []
