"""Renderer and analytic arm for a one-stroke network question."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    return {
        "width": 640,
        "height": 640,
        "center_x": rng.choice([200, 280, 360, 440]),
        "center_y": rng.choice([200, 280, 360, 440]),
        "quarter_turn": rng.randrange(4),
        "reflect": bool(rng.randrange(2)),
        "layout": "single_trail" if int(seed) % 2 == 0 else "requires_retrace",
        "half_width": rng.randint(50, 57),
        "half_height": rng.randint(15, 16),
        "edge_stroke": 4,
        "node_radius": 4,
        "edge_color": [53, 111, 166],
        "node_color": [26, 65, 105],
        "background": [248, 245, 238],
    }


def _transform(point: tuple[int, int], scene: dict) -> tuple[int, int]:
    x, y = point
    if scene["reflect"]:
        y = -y
    for _ in range(int(scene["quarter_turn"]) % 4):
        x, y = -y, x
    return x + int(scene["center_x"]), y + int(scene["center_y"])


def _graph(scene: dict) -> tuple[list[tuple[int, int]], list[tuple[int, int]]]:
    w = int(scene["half_width"])
    h = int(scene["half_height"])
    # A,B,C,D form a rectangle; E,F occupy the two positions just beyond B,C.
    points = [(-w, -h), (w, -h), (w, h), (-w, h), (w + h, -h), (w + h, h)]
    cycle = [(0, 1), (1, 2), (2, 3), (3, 0)]
    if scene["layout"] == "single_trail":
        # B-E-F is one two-edge tail: exactly two odd-degree nodes.
        extra = [(1, 4), (4, 5)]
    elif scene["layout"] == "requires_retrace":
        # B-E and C-F are separate tails: exactly four odd-degree nodes.
        extra = [(1, 4), (2, 5)]
    else:
        raise ValueError("unknown graph layout")
    return [_transform(point, scene) for point in points], cycle + extra


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    nodes, edges = _graph(scene)
    edge_color = tuple(int(value) for value in scene["edge_color"])
    for left, right in edges:
        draw.line((nodes[left], nodes[right]), fill=edge_color, width=int(scene["edge_stroke"]))
    node_color = tuple(int(value) for value in scene["node_color"])
    radius = int(scene["node_radius"])
    for x, y in nodes:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=node_color)
    return image


def analytic_gold(scene: dict) -> str:
    layout = scene["layout"]
    if layout == "single_trail":
        return "possible"
    if layout == "requires_retrace":
        return "impossible"
    raise ValueError("unknown graph layout")


def margin(scene: dict) -> float:
    # Hypothesis scores are -|odd_count-2| and -|odd_count-4|; their gap is 2.
    return 2.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
