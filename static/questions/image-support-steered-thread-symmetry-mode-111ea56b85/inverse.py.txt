"""Independent pixel-only graph reconstruction and Euler-trail classifier."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


EDGE = np.asarray([53.0, 111.0, 166.0])
NODE = np.asarray([26.0, 65.0, 105.0])


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    groups = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            groups.append(np.asarray(points, dtype=np.float64))
    return groups


def _between(value: float, left: float, right: float) -> bool:
    return min(left, right) + 3 < value < max(left, right) - 3


def _visible_link(mask: np.ndarray, first: np.ndarray, second: np.ndarray) -> bool:
    # Node coordinates are (x,y). Only consecutive collinear nodes can be adjacent.
    distance = float(np.linalg.norm(second - first))
    samples = max(12, round(distance))
    hits = 0
    for alpha in np.linspace(0.25, 0.75, samples):
        point = first * (1.0 - alpha) + second * alpha
        x, y = int(round(point[0])), int(round(point[1]))
        patch = mask[max(0, y - 1):y + 2, max(0, x - 1):x + 2]
        hits += bool(patch.any())
    return hits / samples >= 0.70


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    edge_mask = np.linalg.norm(array.astype(np.float32) - EDGE, axis=2) <= 30.0
    node_mask = np.linalg.norm(array.astype(np.float32) - NODE, axis=2) <= 24.0
    full_mask = edge_mask | node_mask
    if len(_components(full_mask, minimum=20)) != 1:
        raise ValueError("network is not one connected component")
    node_groups = _components(node_mask, minimum=40)
    if len(node_groups) != 6:
        raise ValueError("expected exactly six visible junction nodes")
    centers = [group[:, ::-1].mean(axis=0) for group in node_groups]
    edges = set()
    for i, first in enumerate(centers):
        for j in range(i + 1, len(centers)):
            second = centers[j]
            horizontal = abs(first[1] - second[1]) <= 2.0
            vertical = abs(first[0] - second[0]) <= 2.0
            if not (horizontal or vertical):
                continue
            blocked = False
            for k, middle in enumerate(centers):
                if k in (i, j):
                    continue
                if horizontal and abs(middle[1] - first[1]) <= 2.0 and _between(middle[0], first[0], second[0]):
                    blocked = True
                if vertical and abs(middle[0] - first[0]) <= 2.0 and _between(middle[1], first[1], second[1]):
                    blocked = True
            if not blocked and _visible_link(edge_mask, first, second):
                edges.add((i, j))
    if len(edges) != 6:
        raise ValueError("expected exactly six complete network edges")
    degrees = [0] * 6
    adjacency = [set() for _ in range(6)]
    for left, right in edges:
        degrees[left] += 1
        degrees[right] += 1
        adjacency[left].add(right)
        adjacency[right].add(left)
    reached = {0}
    queue = deque([0])
    while queue:
        here = queue.popleft()
        for there in adjacency[here]:
            if there not in reached:
                reached.add(there)
                queue.append(there)
    if len(reached) != 6:
        raise ValueError("reconstructed graph is disconnected")
    odd = sum(degree % 2 for degree in degrees)
    if odd == 2:
        return "possible"
    if odd == 4:
        return "impossible"
    raise ValueError("odd-degree count is outside the declared classes")
