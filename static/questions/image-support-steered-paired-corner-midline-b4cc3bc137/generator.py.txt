"""Renderer and analytic arm for Paired-Corner Midline Reveal."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("M", "W", "U", "Z")

# Nine reconstructed midline heights.  All are centered and then scaled; the
# displayed source is a much larger alternating zigzag around these values.
TEMPLATES = {
    "M": (1.0, .55, .1, -.35, .25, -.35, .1, .55, 1.0),
    "W": (-1.0, -.55, -.1, .35, -.25, .35, -.1, -.55, -1.0),
    "U": (-1.0, -.45, .1, .65, 1.0, .65, .1, -.45, -1.0),
    "Z": (-.85, -.85, -.55, -.25, .05, .35, .65, .85, .85),
}

PALETTES = (
    {"bg": (250, 248, 242), "ink": (28, 48, 72), "node": (181, 62, 132)},
    {"bg": (244, 249, 250), "ink": (46, 40, 73), "node": (195, 70, 126)},
)
AMBER = (238, 158, 42)
TEAL = (42, 174, 165)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xB17D1E)
    return {
        "symbol": SYMBOLS[int(seed) % 4],
        "x0": rng.randint(69, 79),
        "y0": rng.randint(245, 267),
        "x_step": rng.choice((43, 44, 45)),
        "glyph_scale": rng.choice((72, 76, 80)),
        "pair_offset": rng.choice((31, 34, 37)),
        "reverse": bool(rng.getrandbits(1)),
        "palette": rng.randrange(2),
        "stroke": rng.choice((5, 6)),
        "decor_seed": rng.randrange(256),
    }


def _midpoints(scene: dict) -> list[tuple[float, float]]:
    values = TEMPLATES[str(scene["symbol"])]
    return [
        (float(scene["x0"]) + i * float(scene["x_step"]),
         float(scene["y0"]) + float(scene["glyph_scale"]) * value)
        for i, value in enumerate(values)
    ]


def _route_points(scene: dict) -> list[tuple[float, float]]:
    points: list[tuple[float, float]] = []
    offset = float(scene["pair_offset"])
    # Alternating which member comes first prevents a visible upper/lower rail,
    # while the average of each consecutive pair remains exactly the glyph.
    for i, (x, y) in enumerate(_midpoints(scene)):
        sign = -1.0 if i % 2 == 0 else 1.0
        points.extend(((x, y + sign * offset), (x, y - sign * offset)))
    if bool(scene["reverse"]):
        points.reverse()
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), palette["bg"])
    draw = ImageDraw.Draw(image)

    # Sparse pale decorations are deliberately remote and label-independent.
    rng = random.Random((int(scene["decor_seed"]) % 256) + 7001)
    for _ in range(13):
        x, y = rng.randint(28, 484), rng.choice((rng.randint(28, 78), rng.randint(434, 484)))
        r = rng.randint(2, 4)
        draw.ellipse(((x-r)*scale, (y-r)*scale, (x+r)*scale, (y+r)*scale), fill=(218, 218, 215))

    pts = [(round(x*scale), round(y*scale)) for x, y in _route_points(scene)]
    width = int(scene["stroke"]) * scale
    draw.line(pts, fill=palette["ink"], width=width, joint="curve")

    # Every corner is visibly ringed, but the dark under-route and dark outlines
    # remain one connected component.  The inverse must validate that component.
    for x, y in pts:
        r = 7 * scale
        draw.ellipse((x-r, y-r, x+r, y+r), fill=palette["node"], outline=palette["ink"], width=2*scale)
        q = 2 * scale
        draw.ellipse((x-q, y-q, x+q, y+q), fill=palette["bg"])

    start, end = pts[0], pts[-1]
    # Caps sit outside the route ends.  Their shapes and colors declare order.
    neighbor = pts[1]
    dx, dy = start[0]-neighbor[0], start[1]-neighbor[1]
    d = math.hypot(dx, dy)
    sx, sy = start[0] + 16*scale*dx/d, start[1] + 16*scale*dy/d
    draw.line((start[0], start[1], sx, sy), fill=palette["ink"], width=width)
    draw.polygon(((sx-7*scale, sy+7*scale), (sx-7*scale, sy-7*scale), (sx+8*scale, sy)), fill=AMBER, outline=palette["ink"])
    neighbor = pts[-2]
    dx, dy = end[0]-neighbor[0], end[1]-neighbor[1]
    d = math.hypot(dx, dy)
    ex, ey = end[0] + 16*scale*dx/d, end[1] + 16*scale*dy/d
    draw.line((end[0], end[1], ex, ey), fill=palette["ink"], width=width)
    r = 7*scale
    draw.ellipse((ex-r, ey-r, ex+r, ey+r), fill=TEAL, outline=palette["ink"], width=scale)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _template_margin(symbol: str) -> float:
    source = TEMPLATES[symbol]
    # Centered shape distance; used only as an analytic runner-up margin.
    centered = [v - sum(source)/len(source) for v in source]
    scores = []
    for other, values in TEMPLATES.items():
        if other == symbol:
            continue
        oc = [v - sum(values)/len(values) for v in values]
        scores.append(math.sqrt(sum((a-b)**2 for a, b in zip(centered, oc))))
    return min(scores)


def margin(scene: dict) -> float:
    return round(_template_margin(str(scene["symbol"])), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decor_seed"] = int(twin["decor_seed"]) + 256
    return [("low_byte_decoration_alias", twin)]
