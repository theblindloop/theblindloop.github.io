"""Deterministic scene sampler, Pillow renderer, and analytic arm."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 420
LABELS = ("A", "B", "C", "D")
BACKGROUND = "#F7F4EC"
INK = "#222930"
PANEL = "#AAB4BE"
BASE_PALETTE = ("#148B8F", "#E49A18", "#B93D71", "#3A69BE")
SHAPES = ("circle", "square")


def _winner_tokens() -> list[dict]:
    return [{"color": color, "shape": shape} for color in range(4) for shape in SHAPES]


def _distractor_tokens(circle_color: int, square_color: int) -> list[dict]:
    tokens = []
    for color in range(4):
        if color == circle_color:
            shapes = ("circle", "circle")
        elif color == square_color:
            shapes = ("square", "square")
        else:
            shapes = SHAPES
        tokens.extend({"color": color, "shape": shape} for shape in shapes)
    return tokens


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = rng.randrange(4)
    distractor_specs = [(a, b) for a in range(4) for b in range(4) if a != b]
    rng.shuffle(distractor_specs)
    panel_tokens = []
    panel_positions = []
    base_slots = [(43, 151), (92, 151), (141, 151), (190, 151),
                  (43, 254), (92, 254), (141, 254), (190, 254)]
    for panel_index in range(4):
        if panel_index == winner:
            tokens = _winner_tokens()
        else:
            tokens = _distractor_tokens(*distractor_specs.pop())
        rng.shuffle(tokens)
        slots = list(base_slots)
        rng.shuffle(slots)
        positions = [[x + rng.uniform(-5, 5), y + rng.uniform(-6, 6)] for x, y in slots]
        panel_tokens.append(tokens)
        panel_positions.append(positions)
    palette = list(BASE_PALETTE)
    rng.shuffle(palette)
    return {
        "panel_tokens": panel_tokens,
        "panel_positions": panel_positions,
        "palette": palette,
        "serialization_alias": "color_shape_table_v1",
    }


def _panel_score(tokens: list[dict]) -> int:
    return sum(
        sorted(token["shape"] for token in tokens if token["color"] == color)
        == ["circle", "square"]
        for color in range(4)
    )


def analytic_gold(scene: dict) -> str:
    scores = [_panel_score(tokens) for tokens in scene["panel_tokens"]]
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    return LABELS[winners[0]] if len(winners) == 1 and best == 4 else "abstain"


def margin(scene: dict) -> float:
    scores = sorted((_panel_score(tokens) for tokens in scene["panel_tokens"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["serialization_alias"] = "equivalent_color_shape_table"
    return [("serialization_alias", twin)]


def render(scene: dict) -> Image.Image:
    # Access the declared inert alias so the complete scene surface is audited.
    _serialization_alias = scene["serialization_alias"]
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = ImageFont.load_default(size=24 * scale)
    label_font = ImageFont.load_default(size=23 * scale)
    note_font = ImageFont.load_default(size=15 * scale)
    draw.text((WIDTH * scale / 2, 17 * scale), "PAIR EVERY COLOR WITH BOTH SHAPES", fill=INK, font=title_font, anchor="ma")
    draw.text((WIDTH * scale / 2, 48 * scale), "one circle + one square per color", fill=INK, font=note_font, anchor="ma")
    for panel_index in range(4):
        offset = 8 + 248 * panel_index
        draw.rounded_rectangle(
            (offset * scale, 76 * scale, (offset + 232) * scale, 382 * scale),
            radius=11 * scale, outline=PANEL, width=2 * scale,
        )
        for token, position in zip(scene["panel_tokens"][panel_index], scene["panel_positions"][panel_index]):
            local_x, y = position
            x = offset + local_x
            half = 17
            bounds = ((x - half) * scale, (y - half) * scale,
                      (x + half) * scale, (y + half) * scale)
            fill = scene["palette"][token["color"]]
            if token["shape"] == "circle":
                draw.ellipse(bounds, fill=fill)
            elif token["shape"] == "square":
                draw.rectangle(bounds, fill=fill)
            else:
                raise ValueError("unknown marker shape")
        draw.text(((offset + 116) * scale, 352 * scale), LABELS[panel_index], fill=INK, font=label_font, anchor="mm")
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
