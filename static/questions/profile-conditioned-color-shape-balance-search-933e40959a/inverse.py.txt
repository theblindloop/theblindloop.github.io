"""Independent raster-only inverse arm; imports no candidate world module."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
COLOR_RGB = ((20, 139, 143), (228, 154, 24), (185, 61, 113), (58, 105, 190))


def _near(rgb: np.ndarray, target: tuple[int, int, int], tolerance: float = 48.0) -> np.ndarray:
    delta = rgb.astype(np.int32) - np.asarray(target, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 300) -> list[tuple[np.ndarray, np.ndarray]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        ys, xs = [], []
        while stack:
            y, x = stack.pop()
            ys.append(y)
            xs.append(x)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            found.append((np.asarray(ys), np.asarray(xs)))
    return found


def _shape(ys: np.ndarray, xs: np.ndarray) -> str:
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if min(width, height) < 27 or max(width, height) > 40 or abs(width - height) > 3:
        raise ValueError("malformed marker bounds")
    fill_ratio = len(xs) / float(width * height)
    if 0.70 <= fill_ratio <= 0.86:
        return "circle"
    if 0.92 <= fill_ratio <= 1.01:
        return "square"
    raise ValueError("ambiguous marker shape")


def _split_panels(objects: list[tuple[float, float, int, str]]) -> list[list[tuple[float, float, int, str]]]:
    ordered = sorted(objects, key=lambda item: item[0])
    gaps = [ordered[index + 1][0] - ordered[index][0] for index in range(len(ordered) - 1)]
    boundaries = sorted(sorted(range(len(gaps)), key=lambda index: gaps[index], reverse=True)[:3])
    cuts = [0] + [index + 1 for index in boundaries] + [len(ordered)]
    panels = [ordered[cuts[index]:cuts[index + 1]] for index in range(4)]
    if any(len(panel) != 8 for panel in panels):
        raise ValueError("panel grouping failed")
    boundary_gaps = [gaps[index] for index in boundaries]
    inner_gaps = [gap for index, gap in enumerate(gaps) if index not in boundaries]
    if min(boundary_gaps) < max(inner_gaps) * 1.25:
        raise ValueError("panel boundaries ambiguous")
    return panels


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    if rgb.shape[1] < 700 or rgb.shape[0] < 280:
        return "abstain"
    masks = [_near(rgb, color) for color in COLOR_RGB]
    union = np.logical_or.reduce(masks)
    blobs = _components(union)
    if len(blobs) != 32:
        return "abstain"
    objects = []
    try:
        for ys, xs in blobs:
            mean = rgb[ys, xs].astype(float).mean(axis=0)
            color = int(np.argmin([np.linalg.norm(mean - np.asarray(target)) for target in COLOR_RGB]))
            objects.append((float(xs.mean()), float(ys.mean()), color, _shape(ys, xs)))
        panels = _split_panels(objects)
    except ValueError:
        return "abstain"
    scores = []
    for panel in panels:
        if sum(shape == "circle" for _, _, _, shape in panel) != 4:
            return "abstain"
        counts = []
        score = 0
        for color in range(4):
            shapes = sorted(shape for _, _, item_color, shape in panel if item_color == color)
            counts.append(len(shapes))
            score += shapes == ["circle", "square"]
        if counts != [2, 2, 2, 2]:
            return "abstain"
        scores.append(score)
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if len(winners) != 1 or best != 4 or best - sorted(scores, reverse=True)[1] < 1:
        return "abstain"
    return LABELS[winners[0]]
