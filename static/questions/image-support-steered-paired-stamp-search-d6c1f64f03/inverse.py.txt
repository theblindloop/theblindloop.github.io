"""Independent pixel-only inverse for paired microstamp equality."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

TEAL = np.array((15, 135, 145), dtype=np.int16)
GOLD = np.array((232, 167, 35), dtype=np.int16)
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 600:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _decode(array: np.ndarray, points: np.ndarray) -> tuple[tuple[int, ...], tuple[float, float]]:
    ys, xs = points[:, 0], points[:, 1]
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (26 <= x1 - x0 + 1 <= 30 and 26 <= y1 - y0 + 1 <= 30):
        raise ValueError("malformed microstamp extent")
    pattern = []
    for row in range(2):
        for col in range(2):
            cx = round(x0 + (col + 0.5) * (x1 - x0 + 1) / 2 - 0.5)
            cy = round(y0 + (row + 0.5) * (y1 - y0 + 1) / 2 - 0.5)
            pixel = array[cy, cx].astype(np.int16)
            d_teal = float(np.linalg.norm(pixel - TEAL))
            d_gold = float(np.linalg.norm(pixel - GOLD))
            if min(d_teal, d_gold) > 35.0:
                raise ValueError("microstamp cell has no recognized ink")
            pattern.append(0 if d_teal < d_gold else 1)
    return tuple(pattern), ((x0 + x1) / 2.0, (y0 + y1) / 2.0)


def _spatial_groups(stamps: list[tuple[tuple[int, ...], tuple[float, float]]]):
    pending = set(range(len(stamps)))
    groups = []
    while pending:
        root = pending.pop()
        group = {root}
        frontier = [root]
        while frontier:
            current = frontier.pop()
            cx, cy = stamps[current][1]
            linked = []
            for other in pending:
                ox, oy = stamps[other][1]
                if math.hypot(cx - ox, cy - oy) < 112.0:
                    linked.append(other)
            for other in linked:
                pending.remove(other)
                group.add(other)
                frontier.append(other)
        groups.append(sorted(group))
    if len(groups) != 4 or any(len(group) > 2 for group in groups):
        raise ValueError("could not enumerate four badge regions")
    return groups


def _label_groups(groups, stamps):
    positioned = []
    for group in groups:
        xs = [stamps[index][1][0] for index in group]
        ys = [stamps[index][1][1] for index in group]
        positioned.append((sum(xs) / len(xs), sum(ys) / len(ys), group))
    positioned.sort(key=lambda item: item[1])
    top = sorted(positioned[:2], key=lambda item: item[0])
    bottom = sorted(positioned[2:], key=lambda item: item[0])
    return {tuple(item[2]): LABELS[index] for index, item in enumerate(top + bottom)}


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    signed = array.astype(np.int16)
    teal_distance = np.linalg.norm(signed - TEAL, axis=2)
    gold_distance = np.linalg.norm(signed - GOLD, axis=2)
    saturated = (teal_distance < 35.0) | (gold_distance < 35.0)
    raw = _components(saturated)
    if len(raw) not in (7, 8):
        raise ValueError(f"expected seven or eight visible stamps, found {len(raw)}")
    stamps = [_decode(array, component) for component in raw]
    groups = _spatial_groups(stamps)
    labels = _label_groups(groups, stamps)
    winners = []
    runner_distances = []
    for group in groups:
        if len(group) != 2:
            continue
        left, right = (stamps[index][0] for index in group)
        distance = sum(a != b for a, b in zip(left, right, strict=True))
        if distance == 0:
            winners.append(labels[tuple(group)])
        else:
            runner_distances.append(distance)
    if len(winners) != 1 or not runner_distances or min(runner_distances) < 2:
        raise ValueError("no unique equality winner with a two-cell runner-up margin")
    return winners[0]
