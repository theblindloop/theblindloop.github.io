"""Independent pixel-only inverse for the branch-and-readout world."""

from __future__ import annotations

from collections import deque

import numpy as np


ORACLE_VERSION = "counterfactual_branch_readout-oracle-0.1.0"
ABSTAIN = "abstain"

_TOKEN = np.array((31, 79, 137), dtype=np.int16)
_GOLD = np.array((232, 157, 24), dtype=np.int16)
_READOUTS = {
    "red": np.array((216, 59, 68), dtype=np.int16),
    "blue": np.array((48, 105, 194), dtype=np.int16),
    "green": np.array((45, 157, 94), dtype=np.int16),
    "violet": np.array((139, 78, 190), dtype=np.int16),
}


def _near(array: np.ndarray, colour: np.ndarray, tolerance: int = 6) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - colour), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 600) -> list[tuple[int, int, int, int, int]]:
    """Return sizeable 8-connected components as size,y0,x0,y1,x1."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    ys, xs = np.nonzero(mask)
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if seen[start_y, start_x]:
            continue
        queue = deque([(start_y, start_x)])
        seen[start_y, start_x] = True
        size = 0
        y0 = y1 = start_y
        x0 = x1 = start_x
        while queue:
            y, x = queue.popleft()
            size += 1
            y0 = min(y0, y)
            y1 = max(y1, y)
            x0 = min(x0, x)
            x1 = max(x1, x)
            for ny, nx in (
                (y - 1, x),
                (y + 1, x),
                (y, x - 1),
                (y, x + 1),
                (y - 1, x - 1),
                (y - 1, x + 1),
                (y + 1, x - 1),
                (y + 1, x + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if size >= minimum:
            output.append((size, y0, x0, y1, x1))
    return output


def _ring_score(gold_mask: np.ndarray, component: tuple[int, int, int, int, int]) -> int:
    """Count gold pixels in the annulus where the broad source ring is drawn."""

    _, y0, x0, y1, x1 = component
    cx = (x0 + x1) / 2.0
    cy = (y0 + y1) / 2.0
    pad = 15
    top = max(0, y0 - pad)
    bottom = min(gold_mask.shape[0], y1 + pad + 1)
    left = max(0, x0 - pad)
    right = min(gold_mask.shape[1], x1 + pad + 1)
    yy, xx = np.ogrid[top:bottom, left:right]
    radius = np.sqrt((xx - cx) ** 2 + (yy - cy) ** 2)
    annulus = (radius >= 26.0) & (radius <= 41.0)
    return int(np.count_nonzero(gold_mask[top:bottom, left:right] & annulus))


def _readout_colour(array: np.ndarray, left: int, right: int) -> str:
    """Find the dominant large answer-colour chip in a lower panel crop."""

    top = int(array.shape[0] * 0.70)
    bottom = int(array.shape[0] * 0.93)
    crop = array[top:bottom, max(0, left):min(array.shape[1], right)]
    scores = {
        name: int(np.count_nonzero(_near(crop, colour)))
        for name, colour in _READOUTS.items()
    }
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if not ranked or ranked[0][1] < 5000:
        return ABSTAIN
    if len(ranked) > 1 and ranked[0][1] == ranked[1][1]:
        return ABSTAIN
    return ranked[0][0]


def decision_from_image(image) -> str:
    """Count tokens, apply the visible move, then read the selected chip."""

    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        return ABSTAIN

    token_mask = _near(array, _TOKEN)
    tokens = _components(token_mask)
    if not tokens:
        return ABSTAIN

    midpoint = array.shape[1] / 2.0
    left_tokens = [part for part in tokens if (part[2] + part[4]) / 2.0 < midpoint]
    right_tokens = [part for part in tokens if (part[2] + part[4]) / 2.0 > midpoint]
    if not left_tokens or not right_tokens:
        return ABSTAIN

    gold_mask = _near(array, _GOLD)
    scored = sorted((_ring_score(gold_mask, part), part) for part in tokens)
    if scored[-1][0] < 40 or (len(scored) > 1 and scored[-1][0] == scored[-2][0]):
        return ABSTAIN
    source_left = (scored[-1][1][2] + scored[-1][1][4]) / 2.0 < midpoint

    left = len(left_tokens)
    right = len(right_tokens)
    if source_left:
        left -= 1
        right += 1
    else:
        left += 1
        right -= 1
    if left == right:
        return "tie"

    left_readout = _readout_colour(array, 0, int(midpoint))
    right_readout = _readout_colour(array, int(midpoint), array.shape[1])
    if left_readout == ABSTAIN or right_readout == ABSTAIN:
        return ABSTAIN
    return left_readout if left > right else right_readout
