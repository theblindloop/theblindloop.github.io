"""Renderer for a post-action branch-and-readout visual question.

The displayed state has two labelled banks of large navy tokens.  One token is
marked with a gold ring and a gold arrow shows that it moves to the other bank.
Each bank has an independent cyan-framed colour readout.  The answer is the
readout belonging to whichever bank is fuller after the pictured move.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "counterfactual_branch_readout-0.1.0"

WIDTH = 1100
HEIGHT = 700
BACKGROUND = (255, 255, 255)
PANEL_FILL = (244, 247, 251)
PANEL_BORDER = (54, 70, 92)
TOKEN = (31, 79, 137)
GOLD = (232, 157, 24)
CYAN = (20, 174, 196)
INK = (38, 48, 62)
SUBTLE = (92, 105, 122)

ANSWER_COLORS = ("red", "blue", "green", "violet")
READOUT_COLORS = {
    "red": (216, 59, 68),
    "blue": (48, 105, 194),
    "green": (45, 157, 94),
    "violet": (139, 78, 190),
}


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene with a unique post-move fuller bank."""

    rng = random.Random(
        (int(seed) * 6364136223846793005 + 1442695040888963407) & ((1 << 64) - 1)
    )
    source_side = "left" if rng.randrange(2) == 0 else "right"
    final_left = rng.randrange(2) == 0
    level = rng.randrange(3)

    # The four cases keep the post-move answer one token away from a tie.  In
    # two cases the intervention reverses the initial majority, so reading the
    # initially fuller bank selects the deliberately wrong readout.
    if source_side == "left":
        if final_left:
            left_count, right_count = 4 + level, 1 + level
        else:
            left_count, right_count = 3 + level, 2 + level
    else:
        if final_left:
            left_count, right_count = 2 + level, 3 + level
        else:
            left_count, right_count = 1 + level, 4 + level

    post_left, post_right = _post_move_counts_raw(left_count, right_count, source_side)
    active_side = "left" if post_left > post_right else "right"
    target_color = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    other_color = ANSWER_COLORS[
        (ANSWER_COLORS.index(target_color) + 1 + rng.randrange(3)) % len(ANSWER_COLORS)
    ]
    if active_side == "left":
        left_readout, right_readout = target_color, other_color
    else:
        left_readout, right_readout = other_color, target_color

    source_count = left_count if source_side == "left" else right_count
    return {
        "left_count": int(left_count),
        "right_count": int(right_count),
        "source_side": source_side,
        "selected_index": int(rng.randrange(source_count)),
        "left_readout": left_readout,
        "right_readout": right_readout,
        "layout_variant": int(rng.randrange(3)),
        "width": WIDTH,
        "height": HEIGHT,
        "tray_width": 445,
        "tray_height": 505,
        "token_radius": 23 + int(rng.randrange(3)),
        "token_gap": 102 + int(rng.randrange(5)),
    }


def _post_move_counts_raw(left: int, right: int, source_side: str) -> tuple[int, int]:
    if source_side == "left":
        return int(left) - 1, int(right) + 1
    return int(left) + 1, int(right) - 1


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    paths = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    )
    path = paths[0 if bold else 1]
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _panel_boxes(scene: dict) -> tuple[tuple[float, float, float, float], tuple[float, float, float, float]]:
    width = float(scene["width"])
    height = float(scene["height"])
    panel_width = float(scene["tray_width"])
    panel_height = float(scene["tray_height"])
    gap = width - 2.0 * panel_width - 2.0 * 55.0
    left_x = 55.0
    right_x = left_x + panel_width + gap
    top = height - panel_height - 50.0
    return (
        (left_x, top, left_x + panel_width, top + panel_height),
        (right_x, top, right_x + panel_width, top + panel_height),
    )


def _slot_centres(
    box: tuple[float, float, float, float], scene: dict
) -> list[tuple[float, float]]:
    x0, y0, _, _ = box
    gap = float(scene["token_gap"])
    variant = int(scene["layout_variant"])
    centres: list[tuple[float, float]] = []
    for row in range(2):
        for column in range(3):
            x = x0 + 82.0 + column * gap
            y = y0 + 132.0 + row * 102.0
            if variant == 1:
                x += (5.0 if column == 1 else -3.0 if column == 2 else 0.0)
                y += 4.0 if column % 2 else -3.0
            elif variant == 2:
                x += -4.0 if row == 0 else 5.0
                y += 4.0 if column == 0 else -3.0
            centres.append((x, y))
    return centres


def _readout_box(
    box: tuple[float, float, float, float], scene: dict
) -> tuple[float, float, float, float]:
    x0, y0, x1, y1 = box
    readout_width = min(300.0, float(scene["tray_width"]) - 82.0)
    readout_height = 78.0
    cx = (x0 + x1) / 2.0
    cy = y1 - 74.0
    return (
        cx - readout_width / 2.0,
        cy - readout_height / 2.0,
        cx + readout_width / 2.0,
        cy + readout_height / 2.0,
    )


def _selected_centre(
    scene: dict,
    boxes: tuple[tuple[float, float, float, float], tuple[float, float, float, float]],
) -> tuple[float, float]:
    side_index = 0 if scene["source_side"] == "left" else 1
    return _slot_centres(boxes[side_index], scene)[int(scene["selected_index"])]


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    boxes: tuple[tuple[float, float, float, float], tuple[float, float, float, float]],
) -> None:
    source_index = 0 if scene["source_side"] == "left" else 1
    target_index = 1 - source_index
    source = _selected_centre(scene, boxes)
    target_box = boxes[target_index]
    target_x = (target_box[0] + target_box[2]) / 2.0
    lane_y = 91.0
    endpoint_y = target_box[1] - 8.0
    points = [source, (source[0], lane_y), (target_x, lane_y), (target_x, endpoint_y)]
    draw.line(points, fill=GOLD, width=8, joint="curve")
    draw.polygon(
        [
            (target_x - 16.0, endpoint_y - 20.0),
            (target_x + 16.0, endpoint_y - 20.0),
            (target_x, endpoint_y + 3.0),
        ],
        fill=GOLD,
    )


def _draw_center_badge(draw: ImageDraw.ImageDraw, scene: dict) -> None:
    cx = float(scene["width"]) / 2.0
    badge = (cx - 45.0, 99.0, cx + 45.0, 137.0)
    draw.rounded_rectangle(badge, radius=10, fill=GOLD)
    draw.text(
        (cx, 118.0),
        "MOVE",
        fill=(255, 255, 255),
        anchor="mm",
        font=_font(17, bold=True),
    )


def render(scene: dict) -> Image.Image:
    """Render the visible pre-action banks, cues, and independent readouts."""

    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    boxes = _panel_boxes(scene)
    radius = float(scene["token_radius"])

    draw.text(
        (width / 2.0, 18.0),
        "COUNTERFACTUAL BANK SWITCH",
        fill=INK,
        anchor="ma",
        font=_font(25, bold=True),
    )
    draw.text(
        (width / 2.0, 78.0),
        "Banks show the state before the marked move",
        fill=SUBTLE,
        anchor="ma",
        font=_font(16),
    )

    for row, box in enumerate(boxes):
        draw.rounded_rectangle(box, radius=24, fill=PANEL_FILL, outline=PANEL_BORDER, width=5)
        label = "LEFT BANK" if row == 0 else "RIGHT BANK"
        draw.text(
            ((box[0] + box[2]) / 2.0, box[1] + 30.0),
            label,
            fill=INK,
            anchor="mm",
            font=_font(19, bold=True),
        )
        draw.text(
            ((box[0] + box[2]) / 2.0, box[1] + 57.0),
            "BEFORE MOVE",
            fill=SUBTLE,
            anchor="mm",
            font=_font(12, bold=True),
        )

    _draw_arrow(draw, scene, boxes)
    _draw_center_badge(draw, scene)

    selected_side = 0 if scene["source_side"] == "left" else 1
    selected_index = int(scene["selected_index"])
    counts = (int(scene["left_count"]), int(scene["right_count"]))
    readouts = (scene["left_readout"], scene["right_readout"])

    # Readout chips are independent downstream data.  Their cyan frames make
    # the branch targets visible without encoding any answer in metadata.
    for side, box in enumerate(boxes):
        readout = _readout_box(box, scene)
        draw.text(
            ((box[0] + box[2]) / 2.0, readout[1] - 17.0),
            "READOUT",
            fill=CYAN,
            anchor="mm",
            font=_font(13, bold=True),
        )
        draw.rounded_rectangle(
            readout,
            radius=13,
            fill=READOUT_COLORS[readouts[side]],
            outline=CYAN,
            width=7,
        )

    # Gold ring is laid down before token fills, leaving a clean annulus around
    # exactly one large navy token.  Tokens then cover the action line at their
    # centres, which makes the source cue easy to localise from pixels.
    source_center = _selected_centre(scene, boxes)
    ring_radius = radius + 10.0
    draw.ellipse(
        (
            source_center[0] - ring_radius,
            source_center[1] - ring_radius,
            source_center[0] + ring_radius,
            source_center[1] + ring_radius,
        ),
        outline=GOLD,
        width=7,
    )

    for side, box in enumerate(boxes):
        centres = _slot_centres(box, scene)
        for index in range(counts[side]):
            cx, cy = centres[index]
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                fill=TOKEN,
            )
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                outline=(255, 255, 255),
                width=2,
            )

    return image


def _post_move_counts(scene: dict) -> tuple[int, int]:
    return _post_move_counts_raw(
        int(scene["left_count"]), int(scene["right_count"]), scene["source_side"]
    )


def analytic_gold(scene: dict) -> str:
    """Move the marked token, branch on the fuller bank, and read its colour."""

    left, right = _post_move_counts(scene)
    if left > right:
        return scene["left_readout"]
    if right > left:
        return scene["right_readout"]
    return "tie"


def margin(scene: dict) -> float:
    """Absolute post-move count difference; zero is the branch boundary."""

    left, right = _post_move_counts(scene)
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) == 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant latent reparameterization is intended."""

    return []
