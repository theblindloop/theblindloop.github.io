"""Eccentric-dot oddball quadrant search world.

The image is a 2x2 grid of quadrants split by a thin light divider. Each
quadrant holds one light-gray filled disk with a small dark dot at the disk
center. Exactly ONE of the four disks has its dark dot clearly displaced
off-center; the other three disks have their dot exactly at the disk center.

The decision is the quadrant whose dot is off-center (the unique disk where the
dot is NOT concentric with the disk). Every attribute that decides the answer
(which quadrant's dot displacement is large) is drawn in the raster; nothing
else in the scene is the answer.
"""

from __future__ import annotations

import math

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
DISK = (170, 170, 170)
DOT = (25, 25, 25)
DIV = (235, 235, 235)
MG = SIZE // 2  # 240

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, SIZE // 4),
    "TR": (3 * SIZE // 4, SIZE // 4),
    "BL": (SIZE // 4, 3 * SIZE // 4),
    "BR": (3 * SIZE // 4, 3 * SIZE // 4),
}

WIN_OFF_MIN = 16.0
WIN_OFF_MAX = 24.0
CENTER_OFF_MAX = 0.0
RADIUS_MIN = 46.0
RADIUS_MAX = 52.0
DOT_RADIUS = 6.0
_QUARANTINE_THRESH = 12.0


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    offsets = {}
    directions = {}
    for sector in SECTORS:
        if sector == winner:
            offsets[sector] = float(rng.uniform(WIN_OFF_MIN, WIN_OFF_MAX))
        else:
            offsets[sector] = CENTER_OFF_MAX
        directions[sector] = float(rng.uniform(0.0, 360.0))
    radius = float(rng.uniform(RADIUS_MIN, RADIUS_MAX))
    dot_radius = DOT_RADIUS
    return {
        "offsets": offsets,
        "directions": directions,
        "radius": radius,
        "dot_radius": dot_radius,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    draw.line([MG, 0, MG, SIZE], fill=DIV, width=2)
    draw.line([0, MG, SIZE, MG], fill=DIV, width=2)
    r = scene["radius"]
    dr = scene["dot_radius"]
    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        box = [cx - r, cy - r, cx + r, cy + r]
        draw.ellipse(box, fill=DISK)
        off = scene["offsets"][sector]
        ang = math.radians(scene["directions"][sector])
        dx = off * math.cos(ang)
        dy = off * math.sin(ang)
        dbox = [cx + dx - dr, cy + dy - dr, cx + dx + dr, cy + dy + dr]
        draw.ellipse(dbox, fill=DOT)
    return img


def analytic_gold(scene: dict) -> str:
    order = sorted(SECTORS, key=lambda s: scene["offsets"][s], reverse=True)
    winner = order[0]
    if scene["offsets"][winner] <= scene["offsets"][order[1]]:
        raise RuntimeError("no unique off-center dot found")
    return winner


def margin(scene: dict) -> float:
    order = sorted(SECTORS, key=lambda s: scene["offsets"][s], reverse=True)
    return float(scene["offsets"][order[0]] - scene["offsets"][order[1]])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < _QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
