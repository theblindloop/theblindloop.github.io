"""Independent pixel-only inverse arm for eccentric_dot_oddball.

Receives only the final RGB image. It separates each of the four quadrant
cells, thresholds the light-gray filled disk and the dark center dot, computes
the centroid of each, and measures the distance from the dot to the disk
center. The quadrant whose dark dot is displaced farthest from its disk center
holds the off-center dot. Imports nothing from renderer/prompts/generate/verify.
"""

from __future__ import annotations

import numpy as np

SIZE = 480
MG = SIZE // 2

DOT_MAX_SUM = 200
DISK_MIN_SUM = 400
DISK_MAX_SUM = 600

SECTORS = ("TL", "TR", "BL", "BR")
_CELL_X = {"TL": (0, MG), "TR": (MG, SIZE), "BL": (0, MG), "BR": (MG, SIZE)}
_CELL_Y = {"TL": (0, MG), "TR": (0, MG), "BL": (MG, SIZE), "BR": (MG, SIZE)}


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return None
    return float(xs.mean()), float(ys.mean())


def _cell_offset(dot: np.ndarray, disk: np.ndarray) -> float:
    disk_c = _centroid(disk)
    dot_c = _centroid(dot)
    if disk_c is None or dot_c is None:
        return 0.0
    return float(np.hypot(dot_c[0] - disk_c[0], dot_c[1] - disk_c[1]))


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    total = arr.sum(axis=2)
    dark = total < DOT_MAX_SUM
    disk = (total >= DISK_MIN_SUM) & (total <= DISK_MAX_SUM)
    best = None
    best_off = -1.0
    for sector in SECTORS:
        x0, x1 = _CELL_X[sector]
        y0, y1 = _CELL_Y[sector]
        off = _cell_offset(dark[y0:y1, x0:x1], disk[y0:y1, x0:x1])
        if off > best_off:
            best_off = off
            best = sector
    return best if best is not None else SECTORS[0]
