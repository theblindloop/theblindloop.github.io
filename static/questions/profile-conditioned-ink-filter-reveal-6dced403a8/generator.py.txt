"""Renderer for the ink-filter separation world.

A card carries a 9x9 lattice of round printed marks. Every mark is printed with
one or more of three process inks (cyan, magenta, yellow) and shows their
subtractive mixture. Three different symbol stencils are overprinted on the same
lattice, one per ink. A single large colour patch above the lattice - the filter -
declares which ink is read: a red filter darkens marks containing cyan, a green
filter darkens marks containing magenta, a blue filter darkens marks containing
yellow. The marks that stay dark under the shown filter form one of five symbols.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ink_filter_reveal-renderer-0.1.0"

GRID = 9
SYMBOLS = ("arch", "arrow", "ladder", "ring", "vee")
FILTERS = ("red", "green", "blue")
# filter -> index of the RGB channel that the filter passes; a mark looks dark in
# that channel exactly when its printing contains the ink named below.
FILTER_INK = {"red": "cyan", "green": "magenta", "blue": "yellow"}
INK_ORDER = ("cyan", "magenta", "yellow")

WIDTH, HEIGHT = 560, 660
SWATCH_W, SWATCH_H = 104, 46
FID = 15  # side of the square corner fiducials


def _templates() -> dict[str, frozenset[tuple[int, int]]]:
    ring = {(r, c) for r in range(GRID) for c in range(GRID) if max(abs(r - 4), abs(c - 4)) == 3}
    arch = {(r, c) for c in range(GRID) for r in (abs(c - 4), abs(c - 4) + 1)}
    vee = {(r, c) for c in range(GRID) for r in (8 - abs(c - 4), 7 - abs(c - 4))}
    ladder = {(r, c) for r in range(1, 8) for c in (2, 6)}
    ladder |= {(r, c) for r in (1, 4, 7) for c in range(2, 7)}
    arrow = {(r, 4) for r in range(2, 9)}
    arrow |= {(1 + d, 4 + s * d) for d in (1, 2, 3) for s in (-1, 1)}
    arrow |= {(8, c) for c in (2, 3, 5, 6)}
    return {
        "arch": frozenset(arch),
        "arrow": frozenset(arrow),
        "ladder": frozenset(ladder),
        "ring": frozenset(ring),
        "vee": frozenset(vee),
    }


TEMPLATES = _templates()


def _cells(scene: dict, key: str) -> set[tuple[int, int]]:
    """Dark-cell set for one filter: its stencil toggled by the declared flips."""
    cells = set(TEMPLATES[scene["assign"][key]])
    for r, c in scene["flips"][key]:
        cell = (int(r), int(c))
        cells.symmetric_difference_update({cell})
    return cells


def _distances(cells: set[tuple[int, int]]) -> list[tuple[int, str]]:
    return sorted((len(cells ^ set(t)), name) for name, t in TEMPLATES.items())


def _flip_plan(rng: random.Random, symbol: str, near_boundary: bool) -> list[list[int]]:
    base = set(TEMPLATES[symbol])
    if near_boundary:
        rival = rng.choice([s for s in SYMBOLS if s != symbol])
        diff = sorted(base ^ set(TEMPLATES[rival]))
        rng.shuffle(diff)
        take = max(2, len(diff) // 2 - rng.randint(1, 3))
        chosen = diff[:take]
    else:
        pool = [(r, c) for r in range(GRID) for c in range(GRID)]
        rng.shuffle(pool)
        chosen = pool[: rng.randint(2, 3)]
    return [[int(r), int(c)] for r, c in chosen]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    order = list(SYMBOLS)
    rng.shuffle(order)
    assign = {f: order[i] for i, f in enumerate(FILTERS)}
    shown = rng.choice(FILTERS)
    near = rng.random() < 0.25
    flips = {f: _flip_plan(rng, assign[f], near and f == shown) for f in FILTERS}
    cell = rng.choice([40, 42, 44])
    x0 = rng.randint(48, WIDTH - 48 - GRID * cell)
    y0 = rng.randint(150, HEIGHT - 60 - GRID * cell)
    scene = {
        "assign": assign,
        "filter": shown,
        "flips": flips,
        "geom": {"x0": x0, "y0": y0, "cell": cell},
        "swatch": {
            "x": rng.randint(60, WIDTH - 60 - SWATCH_W),
            "y": rng.randint(40, 92),
        },
        "palette": {
            "low": rng.randint(20, 38),
            "high": rng.randint(216, 236),
            "bg": [rng.randint(242, 250), rng.randint(242, 248), rng.randint(236, 244)],
            "line": [rng.randint(198, 214)] * 3,
        },
        "mark_jitter": [
            [rng.randint(-4, 4), rng.randint(-4, 4), rng.randint(-4, 4)]
            for _ in range(GRID * GRID)
        ],
        "radius": rng.choice([12, 13]),
        "draw_order": list(range(GRID * GRID)),
    }
    rng.shuffle(scene["draw_order"])
    return scene


def _ink_map(scene: dict) -> dict[tuple[int, int], tuple[bool, bool, bool]]:
    per_ink = {FILTER_INK[f]: _cells(scene, f) for f in FILTERS}
    marks: dict[tuple[int, int], tuple[bool, bool, bool]] = {}
    for r in range(GRID):
        for c in range(GRID):
            flags = tuple((r, c) in per_ink[ink] for ink in INK_ORDER)
            if any(flags):
                marks[(r, c)] = flags  # type: ignore[assignment]
    return marks


def _swatch_rgb(name: str) -> tuple[int, int, int]:
    return {"red": (214, 44, 44), "green": (40, 168, 66), "blue": (48, 72, 214)}[name]


def render(scene: dict) -> Image.Image:
    pal = scene["palette"]
    geom = scene["geom"]
    cell = int(geom["cell"])
    x0, y0 = int(geom["x0"]), int(geom["y0"])
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(pal["bg"]))
    draw = ImageDraw.Draw(image)

    sw = scene["swatch"]
    draw.rectangle(
        [sw["x"], sw["y"], sw["x"] + SWATCH_W, sw["y"] + SWATCH_H],
        fill=_swatch_rgb(scene["filter"]),
        outline=tuple(pal["line"]),
        width=3,
    )

    span = GRID * cell
    for i in range(GRID + 1):
        draw.line([x0, y0 + i * cell, x0 + span, y0 + i * cell], fill=tuple(pal["line"]), width=1)
        draw.line([x0 + i * cell, y0, x0 + i * cell, y0 + span], fill=tuple(pal["line"]), width=1)

    marks = _ink_map(scene)
    low, high = int(pal["low"]), int(pal["high"])
    radius = int(scene["radius"])
    for index in scene["draw_order"]:
        r, c = divmod(int(index), GRID)
        flags = marks.get((r, c))
        if flags is None:
            continue
        jitter = scene["mark_jitter"][r * GRID + c]
        rgb = tuple(
            max(0, min(255, (low if flags[k] else high) + int(jitter[k]))) for k in range(3)
        )
        cx = x0 + c * cell + cell // 2
        cy = y0 + r * cell + cell // 2
        draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=rgb)

    for fx, fy in ((x0, y0), (x0 + span, y0), (x0, y0 + span), (x0 + span, y0 + span)):
        draw.rectangle(
            [fx - FID // 2, fy - FID // 2, fx + FID // 2, fy + FID // 2], fill=(18, 18, 18)
        )
    return image


def analytic_gold(scene: dict) -> str:
    return _distances(_cells(scene, scene["filter"]))[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(_cells(scene, scene["filter"]))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared aliases: pixel-identical re-encodings that must preserve gold."""
    out: list[tuple[str, dict]] = []

    reversed_order = dict(scene)
    reversed_order["draw_order"] = list(reversed(scene["draw_order"]))
    out.append(("draw_order_reversed", reversed_order))

    sorted_order = dict(scene)
    sorted_order["draw_order"] = sorted(scene["draw_order"])
    out.append(("draw_order_sorted", sorted_order))

    flipped = dict(scene)
    flipped["flips"] = {k: list(reversed(v)) for k, v in scene["flips"].items()}
    out.append(("flip_list_reversed", flipped))
    return out


def _naive_readable(scene: dict) -> int:
    """Diagnostic: template distance of the filter-blind union of every mark."""
    union = set(_ink_map(scene))
    return int(_distances(union)[0][0])
