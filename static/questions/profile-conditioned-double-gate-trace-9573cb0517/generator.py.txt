"""Deterministic latent sampler and Pillow renderer.

Each scene has three separated winding routes.  Across the complete image each
gate color appears exactly six times, so a global color count cannot reveal the
answer.  On the green-to-purple route, the three colors occur one, two, and
three times in a seed-balanced assignment.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 480
COLORS = ("red", "blue", "orange")
RGB = {
    "red": (218, 55, 50),
    "blue": (43, 104, 190),
    "orange": (238, 137, 24),
    "green": (42, 154, 88),
    "purple": (132, 75, 180),
}
_BAND_Y = (90, 240, 390)
_X = (62, 154, 248, 342, 436, 530, 658)
_GATE_FRACTIONS = (0.105, 0.265, 0.425, 0.585, 0.745, 0.895)


def _point_at_fraction(points: list[list[float]], fraction: float) -> list[float]:
    lengths = [math.dist(a, b) for a, b in zip(points, points[1:])]
    wanted = fraction * sum(lengths)
    walked = 0.0
    for a, b, length in zip(points, points[1:], lengths):
        if walked + length >= wanted:
            t = (wanted - walked) / length
            return [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])]
        walked += length
    return list(points[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_row = (int(seed) // 3) % 3
    answer_index = int(seed) % 3

    paths: list[list[list[float]]] = []
    for row, base_y in enumerate(_BAND_Y):
        local = random.Random((int(seed) + 1) * 1009 + row * 9176)
        offsets = [0]
        for _ in range(len(_X) - 2):
            offsets.append(local.choice((-24, -16, -8, 8, 16, 24)))
        offsets.append(0)
        paths.append([[float(x), float(base_y + dy)] for x, dy in zip(_X, offsets)])

    other = [index for index in range(3) if index != answer_index]
    rng.shuffle(other)
    base_counts = [0, 0, 0]
    base_counts[answer_index] = 2
    base_counts[other[0]] = 1
    base_counts[other[1]] = 3

    gates: list[list[dict]] = []
    for row in range(3):
        shift = (row - target_row) % 3
        counts = [base_counts[(color_index - shift) % 3] for color_index in range(3)]
        sequence: list[str] = []
        for color_index, count in enumerate(counts):
            sequence.extend([COLORS[color_index]] * count)
        local = random.Random((int(seed) + 7) * 7919 + row * 104729)
        local.shuffle(sequence)
        row_gates = []
        for fraction, color in zip(_GATE_FRACTIONS, sequence):
            row_gates.append({"center": _point_at_fraction(paths[row], fraction), "color": color})
        gates.append(row_gates)

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "target_row": target_row,
        "paths": paths,
        "gates": gates,
    }


def _draw_scaled_line(draw: ImageDraw.ImageDraw, points: list[list[float]], scale: int) -> None:
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    width = 7 * scale
    for a, b in zip(scaled, scaled[1:]):
        draw.line((a, b), fill=(24, 27, 30), width=width)
    radius = width // 2
    for x, y in scaled[1:-1]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=(24, 27, 30))


def render(scene: dict) -> Image.Image:
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), (250, 249, 246))
    draw = ImageDraw.Draw(image)

    for divider_y in (165, 315):
        draw.line((28 * scale, divider_y * scale, (width - 28) * scale, divider_y * scale),
                  fill=(224, 222, 216), width=scale)

    paths = scene["paths"]
    target_row = int(scene["target_row"])
    for points in paths:
        _draw_scaled_line(draw, points, scale)

    for row_gates in scene["gates"]:
        for gate in row_gates:
            x, y = gate["center"]
            radius = 13 * scale
            box = (
                round(x * scale) - radius,
                round(y * scale) - radius,
                round(x * scale) + radius,
                round(y * scale) + radius,
            )
            draw.ellipse(box, outline=RGB[gate["color"]], width=6 * scale)

    for row, points in enumerate(paths):
        for endpoint_index in (0, -1):
            x, y = points[endpoint_index]
            radius = 16 * scale
            if row == target_row:
                fill = RGB["green"] if endpoint_index == 0 else RGB["purple"]
                outline = (24, 27, 30)
            else:
                fill = (188, 187, 181)
                outline = (105, 104, 100)
            box = (
                round(x * scale) - radius,
                round(y * scale) - radius,
                round(x * scale) + radius,
                round(y * scale) + radius,
            )
            draw.ellipse(box, fill=fill, outline=outline, width=2 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    counts = {color: 0 for color in COLORS}
    for gate in scene["gates"][int(scene["target_row"])]:
        counts[gate["color"]] += 1
    winners = [color for color, count in counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"target route lacks a unique twice-crossed color: {counts}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = {color: 0 for color in COLORS}
    for gate in scene["gates"][int(scene["target_row"])]:
        counts[gate["color"]] += 1
    distances = sorted(abs(value - 2) for value in counts.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["width"]) != WIDTH or int(scene["height"]) != HEIGHT:
            return True
        if len(scene["paths"]) != 3 or len(scene["gates"]) != 3:
            return True
        if any(len(path) != len(_X) for path in scene["paths"]):
            return True
        if any(len(row) != 6 for row in scene["gates"]):
            return True
        totals = {color: 0 for color in COLORS}
        for row in scene["gates"]:
            for gate in row:
                totals[gate["color"]] += 1
        if any(totals[color] != 6 for color in COLORS):
            return True
        target = {color: 0 for color in COLORS}
        for gate in scene["gates"][int(scene["target_row"])]:
            target[gate["color"]] += 1
        return sorted(target.values()) != [1, 2, 3]
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    distractor = next(row for row in range(3) if row != int(scene["target_row"]))
    twin["paths"][distractor] = list(reversed(twin["paths"][distractor]))
    return [("reverse_undirected_distractor_serialization", twin)]


def translated(scene: dict, dx: float, dy: float) -> dict:
    """Label-preserving geometric transform used by candidate tests."""
    twin = copy.deepcopy(scene)
    for path in twin["paths"]:
        for point in path:
            point[0] += dx
            point[1] += dy
    for row in twin["gates"]:
        for gate in row:
            gate["center"][0] += dx
            gate["center"][1] += dy
    return twin


def cycle_gate_labels(scene: dict) -> dict:
    """Declared label-transforming color permutation for test coverage."""
    twin = copy.deepcopy(scene)
    mapping = {"red": "blue", "blue": "orange", "orange": "red"}
    for row in twin["gates"]:
        for gate in row:
            gate["color"] = mapping[gate["color"]]
    return twin
