"""Independent image-only inverse for blue bar orientations."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {tuple(v) for v in np.argwhere(mask)}
    groups: list[np.ndarray] = []
    while points:
        start = points.pop()
        stack = [start]
        group = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    neighbor = (y + dy, x + dx)
                    if neighbor in points:
                        points.remove(neighbor)
                        stack.append(neighbor)
                        group.append(neighbor)
        if len(group) >= 180:
            groups.append(np.asarray(group, dtype=float))
    return groups


def _slant(component: np.ndarray) -> int:
    xy = component[:, [1, 0]]
    xy -= xy.mean(axis=0)
    covariance = xy.T @ xy
    values, vectors = np.linalg.eigh(covariance)
    axis = vectors[:, int(np.argmax(values))]
    anisotropy = float(values.max() / max(values.min(), 1.0))
    if anisotropy < 5.0:
        raise ValueError("non-bar component")
    return 1 if float(axis[0] * axis[1]) > 0 else -1


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    blue = (b - r > 55) & (b - g > 20) & (r < 130) & (g < 155) & (b < 190)
    components = _components(blue)
    if len(components) != 19:
        raise ValueError(f"expected 19 visible bars, found {len(components)}")
    ordered = sorted(components, key=lambda comp: float(comp[:, 0].mean()))
    reference_component = ordered[0]
    field = ordered[1:]
    ref_y = float(reference_component[:, 0].mean())
    if any(float(comp[:, 0].mean()) <= ref_y + 40 for comp in field):
        raise ValueError("reference is not visually separated")
    reference_slant = _slant(reference_component)
    midpoint = image.width / 2.0
    left = [comp for comp in field if float(comp[:, 1].mean()) < midpoint]
    right = [comp for comp in field if float(comp[:, 1].mean()) > midpoint]
    if len(left) != 9 or len(right) != 9:
        raise ValueError("fields do not contain nine bars each")
    left_count = sum(_slant(comp) == reference_slant for comp in left)
    right_count = sum(_slant(comp) == reference_slant for comp in right)
    if left_count == right_count:
        raise ValueError("pixel tallies tie")
    return "LEFT" if left_count > right_count else "RIGHT"
