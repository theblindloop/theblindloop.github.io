"""Deterministic renderer for reference-guided comparative orientation counting."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
SCALE = 3
INK = (28, 91, 130)
DARK = (45, 52, 58)
MID = (140, 148, 154)
PAPER = (250, 249, 246)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_slant = 1 if rng.randrange(2) else -1
    lower = 2 + rng.randrange(3)
    gap = 2 + rng.randrange(2)
    higher = lower + gap
    if higher > 7:
        lower, higher = 4, 7
    left_wins = (int(seed) % 2 == 0)
    left_match = higher if left_wins else lower
    right_match = lower if left_wins else higher

    def slants(matches: int) -> list[int]:
        values = [reference_slant] * matches + [-reference_slant] * (9 - matches)
        rng.shuffle(values)
        return values

    jitter = [[rng.randint(-5, 5), rng.randint(-5, 5)] for _ in range(18)]
    return {
        "reference_slant": reference_slant,
        "left_slants": slants(left_match),
        "right_slants": slants(right_match),
        "jitter": jitter,
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _bar(draw: ImageDraw.ImageDraw, center: tuple[int, int], slant: int, length: int, width: int) -> None:
    cx, cy = center[0] * SCALE, center[1] * SCALE
    angle = math.radians(35 * slant)
    dx, dy = math.cos(angle) * length * SCALE / 2, math.sin(angle) * length * SCALE / 2
    draw.line((cx - dx, cy - dy, cx + dx, cy + dy), fill=INK, width=width * SCALE)


def render(scene: dict) -> Image.Image:
    reference = int(scene["reference_slant"])
    left = [int(v) for v in scene["left_slants"]]
    right = [int(v) for v in scene["right_slants"]]
    jitter = scene["jitter"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAPER)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((238*SCALE, 20*SCALE, 402*SCALE, 122*SCALE), radius=10*SCALE,
                           outline=MID, width=2*SCALE, fill=(255, 255, 255))
    label = "REFERENCE"
    font = _font(14)
    box = draw.textbbox((0, 0), label, font=font)
    draw.text(((320*SCALE - (box[2]-box[0])/2), 29*SCALE), label, fill=DARK, font=font)
    _bar(draw, (320, 84), reference, 66, 12)

    panels = [(32, 148, 306, 446), (334, 148, 608, 446)]
    for bounds in panels:
        draw.rounded_rectangle(tuple(v*SCALE for v in bounds), radius=12*SCALE,
                               outline=MID, width=2*SCALE, fill=(255, 255, 255))
    font_big = _font(17)
    draw.text((137*SCALE, 160*SCALE), "LEFT", fill=DARK, font=font_big)
    draw.text((435*SCALE, 160*SCALE), "RIGHT", fill=DARK, font=font_big)

    xs_left, xs_right = [91, 169, 247], [393, 471, 549]
    ys = [235, 315, 395]
    centers = [(x, y) for y in ys for x in xs_left] + [(x, y) for y in ys for x in xs_right]
    for index, (center, slant) in enumerate(zip(centers, left + right)):
        jx, jy = jitter[index]
        _bar(draw, (center[0] + int(jx), center[1] + int(jy)), int(slant), 52, 11)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    reference = int(scene["reference_slant"])
    left_count = sum(int(v) == reference for v in scene["left_slants"])
    right_count = sum(int(v) == reference for v in scene["right_slants"])
    if left_count == right_count:
        raise ValueError("tied reference-parallel tallies")
    return "LEFT" if left_count > right_count else "RIGHT"


def margin(scene: dict) -> float:
    reference = int(scene["reference_slant"])
    left_count = sum(int(v) == reference for v in scene["left_slants"])
    right_count = sum(int(v) == reference for v in scene["right_slants"])
    return float(abs(left_count - right_count))


def is_quarantined(scene: dict) -> bool:
    reference = int(scene["reference_slant"])
    left_count = sum(int(v) == reference for v in scene["left_slants"])
    right_count = sum(int(v) == reference for v in scene["right_slants"])
    return abs(left_count - right_count) < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
