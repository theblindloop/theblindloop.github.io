"""Independent PNG-only inverse for phase-keyed block decimation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 16
OUTPUT_GRID = 8
LABELS = ("cup", "arch", "bolt", "crown")
CYAN = (24, 156, 170)
GOLD = (242, 178, 47)


def _references() -> dict[str, set[tuple[int, int]]]:
    cup = (
        {(1, y) for y in range(1, 7)}
        | {(6, y) for y in range(1, 7)}
        | {(x, 6) for x in range(2, 6)}
    )
    arch = (
        {(1, y) for y in range(1, 7)}
        | {(6, y) for y in range(1, 7)}
        | {(x, 1) for x in range(2, 6)}
    )
    bolt = (
        {(x, 1) for x in range(1, 7)}
        | {(6, 2), (5, 3), (4, 4), (3, 5), (2, 6)}
        | {(x, 6) for x in range(1, 7)}
    )
    crown = (
        {(x, 6) for x in range(1, 7)}
        | {(1, 4), (1, 5), (6, 4), (6, 5)}
        | {(1, 3), (2, 1), (3, 3), (4, 3), (5, 1), (6, 3)}
    )
    return {"cup": cup, "arch": arch, "bolt": bolt, "crown": crown}


REFERENCES = _references()


def _near(pixels: np.ndarray, rgb: tuple[int, int, int], radius: int) -> np.ndarray:
    delta = pixels.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) < radius * radius


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, x0, y0, x1, y1) for 4-connected mask components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[tuple[int, int, int, int, int]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        x0 = x1 = int(x)
        y0 = y1 = int(y)
        while queue:
            cy, cx = queue.popleft()
            area += 1
            x0, x1 = min(x0, cx), max(x1, cx)
            y0, y1 = min(y0, cy), max(y1, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area >= 120:
            result.append((area, x0, y0, x1, y1))
    return result


def _find_frames(pixels: np.ndarray) -> tuple[tuple[int, int, int, int], tuple[int, int, int, int]]:
    components = _components(_near(pixels, CYAN, 46))
    if len(components) != 2:
        raise ValueError("the two required cyan frames were not uniquely recovered")
    ranked = sorted(components, reverse=True)
    _, mx0, my0, mx1, my1 = ranked[0]
    _, kx0, ky0, kx1, ky1 = ranked[1]
    main_w, main_h = mx1 - mx0 + 1, my1 - my0 + 1
    key_w, key_h = kx1 - kx0 + 1, ky1 - ky0 + 1
    if not (300 <= main_w <= 350 and 300 <= main_h <= 350):
        raise ValueError("large cyan frame does not bound a 16-by-16 mosaic")
    if not (65 <= key_w <= 90 and 65 <= key_h <= 90):
        raise ValueError("small cyan frame does not bound a 2-by-2 key")
    return (mx0, my0, mx1, my1), (kx0, ky0, kx1, ky1)


def _phase_from_gold(pixels: np.ndarray, key: tuple[int, int, int, int]) -> tuple[int, int]:
    mask = _near(pixels, GOLD, 48)
    ys, xs = np.nonzero(mask)
    if len(xs) < 150:
        raise ValueError("the gold phase dot is missing")
    x, y = float(xs.mean()), float(ys.mean())
    x0, y0, x1, y1 = key
    if not (x0 < x < x1 and y0 < y < y1):
        raise ValueError("the gold phase dot is outside the visible key")
    return (int(y >= (y0 + y1) / 2), int(x >= (x0 + x1) / 2))


def _read_cells(pixels: np.ndarray, main: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = main
    width, height = x1 - x0 + 1, y1 - y0 + 1
    cells: list[list[int]] = []
    for row in range(GRID):
        values: list[int] = []
        for column in range(GRID):
            cx = x0 + (column + 0.5) * width / GRID
            cy = y0 + (row + 0.5) * height / GRID
            ix, iy = int(round(cx)), int(round(cy))
            patch = pixels[max(0, iy - 2): iy + 3, max(0, ix - 2): ix + 3].astype(float)
            color = np.median(patch.reshape(-1, 3), axis=0)
            luminance = 0.2126 * color[0] + 0.7152 * color[1] + 0.0722 * color[2]
            values.append(int(luminance < 155.0))
        cells.append(values)
    return cells


def analyze_image(image: Image.Image) -> tuple[str, float]:
    pixels = np.asarray(image.convert("RGB"))
    main, key = _find_frames(pixels)
    phase_row, phase_col = _phase_from_gold(pixels, key)
    cells = _read_cells(pixels, main)
    occupied = {
        (column, row)
        for row in range(OUTPUT_GRID)
        for column in range(OUTPUT_GRID)
        if cells[2 * row + phase_row][2 * column + phase_col]
    }
    if len(occupied) != 16:
        raise ValueError("the selected field does not have the declared equal-ink count")
    ranked = sorted((len(occupied ^ REFERENCES[label]), label) for label in LABELS)
    gap = float(ranked[1][0] - ranked[0][0])
    if gap <= 0:
        raise ValueError("the recovered symbol is tied at the decision boundary")
    return ranked[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return analyze_image(image)[0]
