"""Deterministic renderer for phase-keyed block decimation."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (640, 480)
GRID = 16
OUTPUT_GRID = 8
CELL = 20
FRAME = 6
KEY_CELL = 32
NAMES = ("cup", "arch", "bolt", "crown")

CYAN = (24, 156, 170)
GOLD = (242, 178, 47)
GRID_LINE = (188, 190, 184)
BLOCK_LINE = (105, 124, 132)
PALETTES = (
    ((25, 48, 76), (239, 234, 219), (249, 247, 241)),
    ((86, 39, 60), (239, 232, 216), (248, 246, 239)),
    ((31, 73, 56), (235, 235, 220), (250, 248, 242)),
    ((59, 49, 89), (238, 232, 218), (248, 247, 241)),
)


def _glyphs() -> dict[str, set[tuple[int, int]]]:
    cup = (
        {(1, y) for y in range(1, 7)}
        | {(6, y) for y in range(1, 7)}
        | {(x, 6) for x in range(2, 6)}
    )
    arch = (
        {(1, y) for y in range(1, 7)}
        | {(6, y) for y in range(1, 7)}
        | {(x, 1) for x in range(2, 6)}
    )
    bolt = (
        {(x, 1) for x in range(1, 7)}
        | {(6, 2), (5, 3), (4, 4), (3, 5), (2, 6)}
        | {(x, 6) for x in range(1, 7)}
    )
    crown = (
        {(x, 6) for x in range(1, 7)}
        | {(1, 4), (1, 5), (6, 4), (6, 5)}
        | {(1, 3), (2, 1), (3, 3), (4, 3), (5, 1), (6, 3)}
    )
    return {"cup": cup, "arch": arch, "bolt": bolt, "crown": crown}


GLYPHS = _glyphs()


def _selected(scene: dict) -> set[tuple[int, int]]:
    row_phase = int(scene["phase_row"])
    column_phase = int(scene["phase_col"])
    return {
        (column, row)
        for row in range(OUTPUT_GRID)
        for column in range(OUTPUT_GRID)
        if int(scene["cells"][2 * row + row_phase][2 * column + column_phase])
    }


def _distances(scene: dict) -> list[tuple[int, str]]:
    observed = _selected(scene)
    return sorted((len(observed ^ GLYPHS[name]), name) for name in NAMES)


def analytic_gold(scene: dict) -> str:
    """Derive the label from only rendered cell and phase fields."""
    return _distances(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def sample_scene(seed: int) -> dict:
    """Balance labels, phase, palette, and total ink over consecutive seeds."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x9E3779B1)
    target_index = seed % len(NAMES)
    target = NAMES[target_index]
    points = set(GLYPHS[target])

    # One group in six approaches the visible template boundary. A paired
    # removal/addition preserves the selected and total ink count exactly.
    if (seed // 4) % 6 == 0:
        runner = NAMES[(target_index + 1) % len(NAMES)]
        target_only = sorted(points - GLYPHS[runner])
        runner_only = sorted(GLYPHS[runner] - points)
        if target_only and runner_only:
            points.remove(rng.choice(target_only))
            points.add(rng.choice(runner_only))

    phase_index = ((seed // 4) + target_index) % 4
    phase_row, phase_col = divmod(phase_index, 2)
    cells = [[0 for _ in range(GRID)] for _ in range(GRID)]
    for column, row in points:
        cells[2 * row + phase_row][2 * column + phase_col] = 1

    nuisance_slots = [
        (row, column)
        for row in range(GRID)
        for column in range(GRID)
        if (row % 2, column % 2) != (phase_row, phase_col)
    ]
    # Exactly half of the 192 nuisance cells are filled for every class.
    for row, column in rng.sample(nuisance_slots, 96):
        cells[row][column] = 1

    palette = PALETTES[((seed // 4) + 2 * target_index) % len(PALETTES)]
    scene = {
        "cells": cells,
        "phase_row": phase_row,
        "phase_col": phase_col,
        "dark_rgb": list(palette[0]),
        "light_rgb": list(palette[1]),
        "background_rgb": list(palette[2]),
        "main_x": 238 + rng.randint(-14, 14),
        "main_y": 70 + rng.randint(-14, 14),
        "key_x": 55 + rng.randint(-12, 12),
        "key_y": 112 + rng.randint(-30, 30),
    }
    if analytic_gold(scene) != target or len(_selected(scene)) != 16:
        raise RuntimeError("balanced construction crossed its intended class")
    return scene


def render(scene: dict) -> Image.Image:
    """Draw the noisy mosaic, block grouping, and visible phase key."""
    scale = 2
    image = Image.new(
        "RGB", (CANVAS[0] * scale, CANVAS[1] * scale), tuple(scene["background_rgb"])
    )
    draw = ImageDraw.Draw(image)

    def box(coords: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
        return tuple(int(value * scale) for value in coords)

    main_x, main_y = int(scene["main_x"]), int(scene["main_y"])
    outer = 2 * FRAME + GRID * CELL
    draw.rectangle(box((main_x, main_y, main_x + outer - 1, main_y + outer - 1)), fill=CYAN)
    grid_x, grid_y = main_x + FRAME, main_y + FRAME
    for row in range(GRID):
        for column in range(GRID):
            color = scene["dark_rgb"] if scene["cells"][row][column] else scene["light_rgb"]
            x0, y0 = grid_x + column * CELL, grid_y + row * CELL
            draw.rectangle(box((x0, y0, x0 + CELL - 1, y0 + CELL - 1)), fill=tuple(color))

    for index in range(1, GRID):
        width = (3 if index % 2 == 0 else 1) * scale
        color = BLOCK_LINE if index % 2 == 0 else GRID_LINE
        x = (grid_x + index * CELL) * scale
        y = (grid_y + index * CELL) * scale
        draw.line((x, grid_y * scale, x, (grid_y + GRID * CELL) * scale), fill=color, width=width)
        draw.line((grid_x * scale, y, (grid_x + GRID * CELL) * scale, y), fill=color, width=width)

    key_x, key_y = int(scene["key_x"]), int(scene["key_y"])
    key_outer = 2 * FRAME + 2 * KEY_CELL
    draw.rectangle(box((key_x, key_y, key_x + key_outer - 1, key_y + key_outer - 1)), fill=CYAN)
    key_grid_x, key_grid_y = key_x + FRAME, key_y + FRAME
    for row in range(2):
        for column in range(2):
            x0, y0 = key_grid_x + column * KEY_CELL, key_grid_y + row * KEY_CELL
            draw.rectangle(
                box((x0, y0, x0 + KEY_CELL - 1, y0 + KEY_CELL - 1)),
                fill=tuple(scene["light_rgb"]),
            )
    draw.line(
        ((key_grid_x + KEY_CELL) * scale, key_grid_y * scale,
         (key_grid_x + KEY_CELL) * scale, (key_grid_y + 2 * KEY_CELL) * scale),
        fill=BLOCK_LINE, width=2 * scale,
    )
    draw.line(
        (key_grid_x * scale, (key_grid_y + KEY_CELL) * scale,
         (key_grid_x + 2 * KEY_CELL) * scale, (key_grid_y + KEY_CELL) * scale),
        fill=BLOCK_LINE, width=2 * scale,
    )
    dot_x = key_grid_x + (int(scene["phase_col"]) + 0.5) * KEY_CELL
    dot_y = key_grid_y + (int(scene["phase_row"]) + 0.5) * KEY_CELL
    radius = 10
    draw.ellipse(
        box((int(dot_x - radius), int(dot_y - radius), int(dot_x + radius), int(dot_y + radius))),
        fill=GOLD,
        outline=(255, 250, 230),
        width=2 * scale,
    )
    return image.resize(CANVAS, Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
