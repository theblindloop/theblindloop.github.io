"""Deterministic Pillow renderer for a framed piano panel with raised black keys."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

SIZE = 640
SUPERSAMPLE = 3
BACKGROUND = (246, 244, 238)
KEY_DARK = (25, 29, 36)
KEY_HIGHLIGHT = (47, 52, 61)
WHITE_KEY = (244, 242, 232)
SEAM = (139, 143, 148)
PALETTES = (
    (39, 112, 196),
    (186, 70, 91),
    (26, 145, 119),
    (147, 82, 181),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5049414E)
    return {
        "black_key_count": 4 + (int(seed) % 3),
        "panel_x": rng.randint(75, 85),
        "panel_y": rng.randint(180, 200),
        "panel_width": rng.randint(480, 490),
        "panel_height": rng.randint(244, 270),
        "key_phase": rng.uniform(-0.34, 0.34),
        "accent_index": rng.randrange(len(PALETTES)),
        "screw_side": rng.randrange(2),
        "render_nonce": rng.randrange(1_000_000),
    }


def _geometry(scene: dict) -> tuple[float, float, float, float, float, float, list[float]]:
    x = float(scene["panel_x"]) + 0.0 * int(scene["render_nonce"])
    y = float(scene["panel_y"])
    width = float(scene["panel_width"])
    height = float(scene["panel_height"])
    count = int(scene["black_key_count"])
    phase = float(scene["key_phase"])
    inner_left = x + 25.0
    inner_right = x + width - 25.0
    left_count = count // 2
    right_count = count - left_count
    key_width = 150.0 / count
    bank_step = key_width + 16.0
    left_anchor = inner_left + 22.0 + 4.0 * phase
    right_anchor = inner_right - 22.0 + 4.0 * phase
    centers = [left_anchor + bank_step * index for index in range(left_count)]
    centers += [right_anchor - bank_step * index for index in reversed(range(right_count))]
    key_height = 0.57 * height
    return x, y, width, height, key_width, key_height, centers


def render(scene: dict) -> Image.Image:
    x, y, width, height, key_width, key_height, centers = _geometry(scene)
    accent = PALETTES[int(scene["accent_index"]) % len(PALETTES)]
    scale = SUPERSAMPLE
    canvas = Image.new("RGB", (SIZE * scale, SIZE * scale), BACKGROUND)
    draw = ImageDraw.Draw(canvas)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(round(value * scale) for value in values)

    # A restrained shadow and chromatic bezel establish one recognizable keyboard panel.
    draw.rounded_rectangle(box((x + 8, y + 10, x + width + 8, y + height + 10)), radius=18 * scale,
                           fill=(192, 188, 179))
    draw.rounded_rectangle(box((x, y, x + width, y + height)), radius=18 * scale, fill=accent)
    inset = 15.0
    keyboard = (x + inset, y + inset, x + width - inset, y + height - inset)
    draw.rectangle(box(keyboard), fill=WHITE_KEY)

    # Seven equal long white keys are context, not answer units.
    white_width = (keyboard[2] - keyboard[0]) / 7.0
    for index in range(1, 7):
        sx = keyboard[0] + white_width * index
        draw.line((round(sx * scale), round(keyboard[1] * scale), round(sx * scale),
                   round(keyboard[3] * scale)), fill=SEAM, width=2 * scale)
    lip_y = keyboard[3] - 22.0
    draw.line((round(keyboard[0] * scale), round(lip_y * scale), round(keyboard[2] * scale),
               round(lip_y * scale)), fill=(184, 181, 173), width=2 * scale)

    # Total black-key width is fixed at 150 px across answer classes, limiting total-ink cues.
    top = y + inset - 1.0
    bottom = top + key_height
    for center in centers:
        left = center - key_width / 2.0
        right = center + key_width / 2.0
        draw.rounded_rectangle(box((left, top, right, bottom)), radius=4 * scale, fill=KEY_DARK)
        # A small inset highlight makes each insert visibly raised but stays in the dark mask.
        draw.rounded_rectangle(box((left + 4, top + 5, right - 4, bottom - 10)),
                               radius=2 * scale, fill=KEY_HIGHLIGHT)
        draw.rectangle(box((left + 2, bottom - 11, right - 2, bottom)), fill=KEY_DARK)

    # Two colored screws are answer-independent distractors and never enter the dark mask.
    screw_y = y + height - 8.0
    offsets = (34.0, width - 34.0) if int(scene["screw_side"]) == 0 else (43.0, width - 43.0)
    screw_color = tuple(min(255, channel + 55) for channel in accent)
    for offset in offsets:
        cx = x + offset
        r = 5.0
        draw.ellipse(box((cx - r, screw_y - r, cx + r, screw_y + r)), fill=screw_color)

    return canvas.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["black_key_count"]))


def margin(scene: dict) -> float:
    _, _, _, _, key_width, _, centers = _geometry(scene)
    if len(centers) < 2:
        return 0.0
    return float(min(b - a - key_width for a, b in zip(centers, centers[1:])))


def is_quarantined(scene: dict) -> bool:
    x, y, width, height, key_width, key_height, _ = _geometry(scene)
    count = int(scene["black_key_count"])
    return (
        count not in {4, 5, 6}
        or margin(scene) < 14.0
        or key_width < 20.0
        or key_height < 125.0
        or min(x, y, SIZE - x - width, SIZE - y - height) < 65.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["render_nonce"] = int(scene["render_nonce"]) + 1
    return [("render_nonce_alias", twin)]
