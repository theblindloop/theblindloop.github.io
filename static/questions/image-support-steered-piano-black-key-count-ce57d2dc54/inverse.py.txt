"""Independent image-only inverse for the number of raised black piano keys."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if visited[y, x]:
            continue
        queue = deque([(y, x)])
        visited[y, x] = True
        area = 0
        x0 = x1 = x
        y0 = y1 = y
        while queue:
            cy, cx = queue.popleft()
            area += 1
            x0 = min(x0, cx)
            x1 = max(x1, cx)
            y0 = min(y0, cy)
            y1 = max(y1, cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        output.append((area, x0, y0, x1, y1))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low

    # The bezel may change palette; this check only establishes a substantial bounded panel.
    colored = (chroma > 38) & (low < 225)
    colored_points = np.argwhere(colored)
    if len(colored_points) < 6500:
        raise ValueError("missing substantial colored keyboard bezel")
    cy0, cx0 = colored_points.min(axis=0)
    cy1, cx1 = colored_points.max(axis=0)
    if not (460 <= cx1 - cx0 + 1 <= 520 and 220 <= cy1 - cy0 + 1 <= 300):
        raise ValueError("colored keyboard panel has invalid extent")
    if min(cx0, cy0, array.shape[1] - 1 - cx1, array.shape[0] - 1 - cy1) < 65:
        raise ValueError("keyboard panel approaches image boundary")

    # Only neutral, nearly black pixels can be raised black-key evidence.
    dark = (high < 76) & (chroma < 28)
    raw = _components(dark)
    keys: list[tuple[int, int, int, int, int]] = []
    for component in raw:
        area, x0, y0, x1, y1 = component
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        fill = area / float(width * height)
        if area >= 2100 and 20 <= width <= 43 and 125 <= height <= 165 and height / width >= 3.2 and fill >= 0.82:
            keys.append(component)

    count = len(keys)
    if count not in {4, 5, 6}:
        raise ValueError(f"ambiguous black-key count: {count}")
    centers_x = sorted((component[1] + component[3]) / 2.0 for component in keys)
    tops = [component[2] for component in keys]
    bottoms = [component[4] for component in keys]
    if max(tops) - min(tops) > 5 or max(bottoms) - min(bottoms) > 5:
        raise ValueError("dark inserts are not aligned like piano keys")
    gaps = [b - a for a, b in zip(centers_x, centers_x[1:])]
    large_gaps = [gap for gap in gaps if gap > 75]
    small_gaps = [gap for gap in gaps if gap <= 75]
    if len(large_gaps) != 1 or not (180 <= large_gaps[0] <= 340) or not small_gaps or min(small_gaps) < 38 or max(small_gaps) > 56:
        raise ValueError("dark inserts lack clean key spacing")
    # Every qualifying key must lie inside the detected regional bezel.
    if any(x0 <= cx0 + 10 or x1 >= cx1 - 10 or y0 < cy0 or y1 >= cy1 - 45
           for _, x0, y0, x1, y1 in keys):
        raise ValueError("dark component is not bound to the keyboard panel")
    return str(count)
