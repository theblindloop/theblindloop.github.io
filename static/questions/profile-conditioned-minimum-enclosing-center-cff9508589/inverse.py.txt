"""Independent image-only inverse for the minimax enclosure decision."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

DARK_RGB = (38, 42, 51)
COLOR_RGB = {
    "red": (214, 47, 53),
    "blue": (36, 116, 198),
    "green": (25, 140, 85),
    "orange": (223, 123, 32),
}
BACKGROUND_RGB = (247, 246, 241)
MIN_DARK_AREA = 55
MIN_RING_AREA = 90
ABSTAIN_MARGIN_PX = 3.0


def _components(mask: np.ndarray, minimum_area: int) -> list[tuple[float, float, int]]:
    """Return 8-connected component centroids without geometry side inputs."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while queue:
            current_y, current_x = queue.popleft()
            count += 1
            sum_x += current_x + 0.5
            sum_y += current_y + 0.5
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    next_y = current_y + dy
                    next_x = current_x + dx
                    if (
                        0 <= next_y < height
                        and 0 <= next_x < width
                        and mask[next_y, next_x]
                        and not seen[next_y, next_x]
                    ):
                        seen[next_y, next_x] = True
                        queue.append((next_y, next_x))
        if count >= minimum_area:
            components.append((sum_x / count, sum_y / count, count))
    return components


def _pixel_labels(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    palette = np.asarray(
        [DARK_RGB, *COLOR_RGB.values(), BACKGROUND_RGB], dtype=np.int16
    )
    differences = array[:, :, None, :] - palette[None, None, :, :]
    squared = np.sum(differences.astype(np.int32) ** 2, axis=3)
    labels = np.argmin(squared, axis=2)
    nearest = np.take_along_axis(squared, labels[:, :, None], axis=2)[:, :, 0]
    labels[nearest > 85 * 85] = -1
    return labels


def measure(image: Image.Image) -> dict:
    """Recover marks, candidate radii, decision, and runner-up margin."""
    labels = _pixel_labels(image)
    dark = _components(labels == 0, MIN_DARK_AREA)
    if len(dark) < 3:
        raise ValueError(f"need at least three dark dots, found {len(dark)}")
    points = [(x, y) for x, y, _ in dark]

    candidates: dict[str, tuple[float, float]] = {}
    for label, color in enumerate(COLOR_RGB, start=1):
        found = _components(labels == label, MIN_RING_AREA)
        if len(found) != 1:
            raise ValueError(f"expected one {color} candidate ring, found {len(found)}")
        candidates[color] = (found[0][0], found[0][1])

    scores = []
    for color, center in candidates.items():
        farthest = max(math.hypot(center[0] - x, center[1] - y) for x, y in points)
        scores.append((farthest, color))
    scores.sort()
    return {
        "decision": scores[0][1],
        "margin": scores[1][0] - scores[0][0],
        "scores": {color: score for score, color in scores},
        "point_count": len(points),
        "candidate_centers": candidates,
    }


def decision_from_image(image: Image.Image) -> str:
    """Return the minimax-center color from pixels alone, or abstain on a tie."""
    measurement = measure(image)
    if measurement["margin"] < ABSTAIN_MARGIN_PX:
        raise ValueError(
            f"ambiguous raster: runner-up margin {measurement['margin']:.3f}px"
        )
    return str(measurement["decision"])
