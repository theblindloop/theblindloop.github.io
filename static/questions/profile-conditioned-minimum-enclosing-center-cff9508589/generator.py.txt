"""Deterministic latent sampler and renderer for a minimax enclosure world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = "#f7f6f1"
DARK = "#262a33"
COLORS = {
    "red": "#d62f35",
    "blue": "#2474c6",
    "green": "#198c55",
    "orange": "#df7b20",
}
COLOR_NAMES = tuple(COLORS)
DOT_RADIUS = 6.0
RING_RADIUS = 11.0
RING_WIDTH = 5
QUARANTINE_MARGIN = 8.0


def _distance(left: list[float], right: list[float]) -> float:
    return math.hypot(left[0] - right[0], left[1] - right[1])


def _candidate_scores(scene: dict) -> list[tuple[float, str]]:
    points = scene["points"]
    candidates = scene["candidates"]
    return sorted(
        (max(_distance(candidate["center"], point) for point in points), candidate["color"])
        for candidate in candidates
    )


def analytic_gold(scene: dict) -> str:
    """Color of the visible candidate with the lowest farthest-dot radius."""
    return _candidate_scores(scene)[0][1]


def margin(scene: dict) -> float:
    """Runner-up minus winning minimax radius, in scene pixels."""
    scores = _candidate_scores(scene)
    return scores[1][0] - scores[0][0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _far_enough(point: list[float], others: list[list[float]], clearance: float) -> bool:
    return all(_distance(point, other) >= clearance for other in others)


def sample_scene(seed: int) -> dict:
    """Sample an asymmetric cloud whose true minimum enclosing circle is visible.

    Three boundary dots form an acute triangle on one circle. Therefore that
    circle, rather than a diameter circle from only two dots, is the unique
    minimum enclosure. Interior marks are deliberately irregular so a centroid
    shortcut is unreliable. Answer color cycles with the integer seed.
    """
    rng = random.Random(int(seed))
    answer_color = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]

    for _ in range(500):
        cx = rng.uniform(238.0, 274.0)
        cy = rng.uniform(238.0, 274.0)
        radius = rng.uniform(132.0, 151.0)
        base = rng.uniform(0.0, 2.0 * math.pi)
        # All three cyclic angular gaps stay below 180 degrees, so the support
        # triangle is acute and its circumcircle is the minimum enclosure.
        offsets_deg = (0.0, rng.uniform(106.0, 121.0), rng.uniform(235.0, 249.0))
        support = [
            [
                round(cx + radius * math.cos(base + math.radians(offset)), 3),
                round(cy + radius * math.sin(base + math.radians(offset)), 3),
            ]
            for offset in offsets_deg
        ]

        phase = rng.uniform(0.0, 2.0 * math.pi)
        centers = [[round(cx, 3), round(cy, 3)]]
        for index in range(3):
            angle = phase + index * 2.0 * math.pi / 3.0 + rng.uniform(-0.14, 0.14)
            distance = rng.uniform(45.0, 70.0)
            centers.append(
                [
                    round(cx + distance * math.cos(angle), 3),
                    round(cy + distance * math.sin(angle), 3),
                ]
            )
        if any(not _far_enough(center, centers[:index], 31.0) for index, center in enumerate(centers)):
            continue

        points = list(support)
        interior_count = 9 + (int(seed) % 5)
        attempts = 0
        while len(points) < interior_count + 3 and attempts < 3000:
            attempts += 1
            angle = rng.uniform(0.0, 2.0 * math.pi)
            # Area-uniform but bounded well inside the support circle.
            radial = math.sqrt(rng.uniform(0.12, 0.68)) * radius
            point = [
                round(cx + radial * math.cos(angle), 3),
                round(cy + radial * math.sin(angle), 3),
            ]
            if not _far_enough(point, points, 19.0):
                continue
            if not _far_enough(point, centers, 25.0):
                continue
            points.append(point)
        if len(points) != interior_count + 3:
            continue

        remaining = [name for name in COLOR_NAMES if name != answer_color]
        rng.shuffle(remaining)
        candidates = [{"color": answer_color, "center": centers[0]}]
        candidates.extend(
            {"color": color, "center": center}
            for color, center in zip(remaining, centers[1:], strict=True)
        )
        rng.shuffle(candidates)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "points": points,
            "candidates": candidates,
        }
        if analytic_gold(scene) == answer_color and margin(scene) >= 13.0:
            return scene
    raise RuntimeError(f"could not sample a separated minimax scene for seed {seed}")


def _ellipse_box(center: list[float], radius: float, scale: int) -> tuple[float, ...]:
    x, y = center
    return (
        (x - radius) * scale,
        (y - radius) * scale,
        (x + radius) * scale,
        (y + radius) * scale,
    )


def render(scene: dict) -> Image.Image:
    """Render only the declared point and candidate sets, with antialiasing."""
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for point in scene["points"]:
        draw.ellipse(_ellipse_box(point, DOT_RADIUS, scale), fill=DARK)
    for candidate in scene["candidates"]:
        draw.ellipse(
            _ellipse_box(candidate["center"], RING_RADIUS, scale),
            outline=COLORS[candidate["color"]],
            width=RING_WIDTH * scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the unordered-set alias without changing draw-order pixels."""
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    twin["candidates"] = list(reversed(twin["candidates"]))
    return [("reverse_unordered_sets", twin)]
