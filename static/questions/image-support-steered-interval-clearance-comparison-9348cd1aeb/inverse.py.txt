"""Independent pixel-only inverse for the separated interval-clearance task."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the raster does not contain four intact, bindable intervals."""


@dataclass(frozen=True)
class Component:
    x0: int
    y0: int
    x1: int
    y1: int
    area: int

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1

    @property
    def cx(self) -> float:
        return (self.x0 + self.x1) / 2.0


def _chromatic_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    return (spread >= 55) & (array.max(axis=2) <= 245) & (array.min(axis=2) <= 205)


def _components(mask: np.ndarray) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for start_y, start_x in np.argwhere(mask):
        sy, sx = int(start_y), int(start_x)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 500:
            output.append(Component(min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _validate_interval(mask: np.ndarray, item: Component) -> None:
    if not 64 <= item.width <= 134 or item.height != 45:
        raise OracleAbstention(
            f"invalid capped-interval extent {item.width}x{item.height}"
        )
    region = mask[item.y0 : item.y1 + 1, item.x0 : item.x1 + 1]
    expected = np.zeros_like(region)
    center = item.height // 2
    expected[center - 6 : center + 7, :] = True
    expected[:, :7] = True
    expected[:, -7:] = True
    if not np.array_equal(region, expected):
        raise OracleAbstention("a capped interval is clipped, erased, or malformed")


def _measure_clearances(image: Image.Image) -> tuple[int, int]:
    mask = _chromatic_mask(image)
    components = sorted(_components(mask), key=lambda item: item.cx)
    if len(components) != 4:
        raise OracleAbstention(f"expected four intact colored intervals, found {len(components)}")
    for item in components:
        _validate_interval(mask, item)
    local_left = components[1].cx - components[0].cx
    panel_gap = components[2].cx - components[1].cx
    local_right = components[3].cx - components[2].cx
    if panel_gap <= 1.45 * max(local_left, local_right):
        raise OracleAbstention("the two distant interval pairs cannot be bound reliably")
    left_gap = components[1].x0 - components[0].x1 - 1
    right_gap = components[3].x0 - components[2].x1 - 1
    if not 12 <= min(left_gap, right_gap) or max(left_gap, right_gap) > 100:
        raise OracleAbstention("a recovered inner clearance is outside the supported range")
    return left_gap, right_gap


def decision_from_image(image: Image.Image) -> str:
    """Return LEFT or RIGHT from rasterized interval bounding boxes alone."""

    left_gap, right_gap = _measure_clearances(image)
    difference = left_gap - right_gap
    if abs(difference) < 4:
        raise OracleAbstention("the two raster clearances are indistinguishable")
    return "LEFT" if difference > 0 else "RIGHT"
