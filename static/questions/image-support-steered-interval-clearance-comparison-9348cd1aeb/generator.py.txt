"""Deterministic renderer and analytic arm for interval_clearance_comparison."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 500
BACKGROUND = (248, 248, 245)
INK = (54, 59, 65)
FRAME = (116, 121, 127)
GUIDE = (205, 207, 205)
LEFT_FRAME = (18, 36, 420, 224)
RIGHT_FRAME = (540, 276, 942, 464)
CAP_HALF_HEIGHT = 22
CAP_HALF_WIDTH = 3
STEM_HALF_HEIGHT = 6
QUARANTINE_MARGIN = 14.0
PALETTE = (
    (20, 117, 210),
    (224, 78, 54),
    (18, 154, 102),
    (145, 75, 188),
    (202, 51, 124),
    (6, 150, 172),
    (225, 142, 13),
    (74, 98, 211),
)


def _clearance(intervals: list[list[int]]) -> int:
    return int(intervals[1][0]) - int(intervals[0][1]) - 1


def _make_pair(rng: random.Random, gap: int, widths: tuple[int, int]) -> list[list[int]]:
    first_width, second_width = (int(value) for value in widths)
    start = rng.randint(27, 43)
    first_end = start + first_width - 1
    second_start = first_end + int(gap) + 1
    second_end = second_start + second_width - 1
    if second_end > 374:
        raise ValueError("interval pair exceeds panel interior")
    return [[start, first_end], [second_start, second_end]]


def sample_scene(seed: int) -> dict:
    """Sample balanced answers while counterbalancing colored-ink and outer-span cues."""

    rng = random.Random(int(seed))
    winner_is_left = int(seed) % 2 == 0
    cue_aligns = (int(seed) // 2) % 2 == 0
    differences = (7, 13, 21, 31, 43, 53)
    difference = differences[int(seed) % len(differences)]
    loser_gap = rng.randint(18, 36)
    winner_gap = loser_gap + difference

    for _ in range(30_000):
        winner_widths = (rng.randint(58, 128), rng.randint(58, 128))
        loser_widths = (rng.randint(58, 128), rng.randint(58, 128))
        winner_ink = sum(winner_widths)
        loser_ink = sum(loser_widths)
        if cue_aligns:
            cue_ok = winner_ink > loser_ink + 18
        else:
            # This stronger opposition also reverses the total outer span cue.
            cue_ok = winner_ink + difference + 12 < loser_ink
        if not cue_ok:
            continue
        try:
            winner_pair = _make_pair(rng, winner_gap, winner_widths)
            loser_pair = _make_pair(rng, loser_gap, loser_widths)
        except ValueError:
            continue
        break
    else:
        raise RuntimeError(f"could not counterbalance interval scene for seed {seed}")

    if winner_is_left:
        left_intervals, right_intervals = winner_pair, loser_pair
    else:
        left_intervals, right_intervals = loser_pair, winner_pair

    colors = list(PALETTE)
    rng.shuffle(colors)
    return {
        "left_intervals": left_intervals,
        "right_intervals": right_intervals,
        "left_colors": [list(colors[0]), list(colors[1])],
        "right_colors": [list(colors[2]), list(colors[3])],
        "panel_offsets": [
            [rng.randint(-4, 4), rng.randint(-4, 4)],
            [rng.randint(-4, 4), rng.randint(-4, 4)],
        ],
    }


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.load_default(size=size)
    except TypeError:
        return ImageFont.load_default()


def _draw_interval(
    draw: ImageDraw.ImageDraw,
    x0: int,
    x1: int,
    y: int,
    color: tuple[int, int, int],
) -> None:
    draw.rectangle(
        (x0, y - STEM_HALF_HEIGHT, x1, y + STEM_HALF_HEIGHT),
        fill=color,
    )
    for x in (x0, x1):
        draw.rectangle(
            (x - CAP_HALF_WIDTH, y - CAP_HALF_HEIGHT, x + CAP_HALF_WIDTH, y + CAP_HALF_HEIGHT),
            fill=color,
        )


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    frame: tuple[int, int, int, int],
    intervals: list[list[int]],
    colors: list[list[int]],
    offset: list[int],
    label: str,
) -> None:
    dx, dy = (int(offset[0]), int(offset[1]))
    fx0, fy0, fx1, fy1 = frame
    fx0, fx1, fy0, fy1 = fx0 + dx, fx1 + dx, fy0 + dy, fy1 + dy
    draw.rounded_rectangle((fx0, fy0, fx1, fy1), radius=9, outline=FRAME, width=3)
    draw.text(((fx0 + fx1) // 2, fy0 + 25), label, fill=INK, anchor="mm", font=_font(24))
    row_y = (fy0 + fy1) // 2 + 20
    draw.line((fx0 + 20, row_y, fx1 - 20, row_y), fill=GUIDE, width=1)
    for endpoints, color_values in zip(intervals, colors):
        x0 = fx0 + int(endpoints[0])
        x1 = fx0 + int(endpoints[1])
        _draw_interval(draw, x0, x1, row_y, tuple(int(value) for value in color_values))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(
        draw,
        LEFT_FRAME,
        scene["left_intervals"],
        scene["left_colors"],
        scene["panel_offsets"][0],
        "LEFT",
    )
    _draw_panel(
        draw,
        RIGHT_FRAME,
        scene["right_intervals"],
        scene["right_colors"],
        scene["panel_offsets"][1],
        "RIGHT",
    )
    return image


def analytic_gold(scene: dict) -> str:
    left_gap = _clearance(scene["left_intervals"])
    right_gap = _clearance(scene["right_intervals"])
    return "LEFT" if left_gap > right_gap else "RIGHT"


def margin(scene: dict) -> float:
    return float(abs(_clearance(scene["left_intervals"]) - _clearance(scene["right_intervals"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
