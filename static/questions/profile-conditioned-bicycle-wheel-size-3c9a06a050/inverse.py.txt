"""Pixel-only inverse arm for `bicycle_wheel_size`.

Receives one rasterised PNG and nothing else: no latent scene, no renderer,
prompt, generator, verifier or gold code.  It segments the black wheel ink
(dark and colourless, so the coloured frame, the coloured spots, the ochre
ground line and the grey page decoration are all excluded), keeps the two large
connected wheel bodies, measures each body's drawn diameter from its bounding
box, and applies the tolerance rule stated in the question: the wheels read as
the same size when their diameters differ by less than a tenth of the larger
diameter, otherwise the larger side is reported.

It abstains whenever the raster does not show two intact round wheel bodies, or
when the measured difference sits inside the unreadable band around the stated
tolerance.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "bicycle_wheel_size-oracle-0.1.0"
ABSTAIN = "abstain"

DARK_MAX = 90  # every channel this dark or darker
CHROMA_MAX = 40  # and near-colourless: black ink, not coloured ink
MIN_WHEEL_AREA = 400
MIN_DIAMETER = 50.0
ROUNDNESS_TOL = 0.18  # |h - w| / max(h, w) for an intact wheel body
RATIO_THRESHOLD = 0.10  # the tolerance the question declares
READ_BAND_PX = 2.0  # measured distance to the tolerance below which we abstain


def _components(mask: np.ndarray) -> list:
    """4-connected components of a boolean mask as (area, r0, r1, c0, c1)."""
    h, w = mask.shape
    flat = mask.reshape(-1).copy()
    out = []
    for start in np.flatnonzero(flat):
        if not flat[start]:
            continue
        stack = [int(start)]
        flat[start] = False
        area = 0
        r0 = r1 = int(start) // w
        c0 = c1 = int(start) % w
        while stack:
            p = stack.pop()
            area += 1
            r, c = divmod(p, w)
            if r < r0:
                r0 = r
            if r > r1:
                r1 = r
            if c < c0:
                c0 = c
            if c > c1:
                c1 = c
            if c > 0 and flat[p - 1]:
                flat[p - 1] = False
                stack.append(p - 1)
            if c < w - 1 and flat[p + 1]:
                flat[p + 1] = False
                stack.append(p + 1)
            if r > 0 and flat[p - w]:
                flat[p - w] = False
                stack.append(p - w)
            if r < h - 1 and flat[p + w]:
                flat[p + w] = False
                stack.append(p + w)
        out.append((area, r0, r1, c0, c1))
    return out


def decision_from_image(image: "Image.Image") -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    ink = (arr.max(axis=-1) <= DARK_MAX) & (chroma <= CHROMA_MAX)
    if int(ink.sum()) < 2 * MIN_WHEEL_AREA:
        return ABSTAIN

    bodies = [c for c in _components(ink) if c[0] >= MIN_WHEEL_AREA]
    if len(bodies) != 2:
        return ABSTAIN

    measured = []
    for area, r0, r1, c0, c1 in bodies:
        height = float(r1 - r0 + 1)
        width = float(c1 - c0 + 1)
        if max(height, width) < MIN_DIAMETER:
            return ABSTAIN
        if abs(height - width) > ROUNDNESS_TOL * max(height, width):
            return ABSTAIN  # not an intact round wheel body
        measured.append(((c0 + c1) / 2.0, (height + width) / 2.0))

    measured.sort(key=lambda t: t[0])
    d_left, d_right = measured[0][1], measured[1][1]
    diff = d_left - d_right
    tolerance = RATIO_THRESHOLD * max(d_left, d_right)
    if abs(abs(diff) - tolerance) < READ_BAND_PX:
        return ABSTAIN
    if abs(diff) < tolerance:
        return "same"
    return "left" if diff > 0 else "right"
