"""Deterministic triangular edge-rolling scene and analytic solution."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1120
HEIGHT = 760
PATH_LEN = 8
H = math.sqrt(3.0) / 2.0
TRACK_BOX = (32, 118, 804, 726)
MAX_SIDE = 116.0
LABELS = ("red", "green", "violet")
PIP_COLORS = {
    "red": (220, 55, 68),
    "green": (30, 158, 104),
    "violet": (132, 72, 184),
}
STEP_FILLS = (
    (224, 240, 250), (211, 232, 248), (199, 225, 246), (186, 217, 243),
    (174, 209, 239), (163, 201, 234), (151, 192, 228), (139, 183, 221),
)
GOLD = (242, 177, 45)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)


def _vertices(cell: list) -> tuple[tuple[int, int], ...]:
    i, j, orientation = int(cell[0]), int(cell[1]), str(cell[2])
    if orientation == "u":
        return ((i, j), (i + 1, j), (i, j + 1))
    if orientation == "d":
        return ((i + 1, j), (i + 1, j + 1), (i, j + 1))
    raise ValueError(f"unknown orientation {orientation!r}")


def _edges(cell: list) -> set[frozenset[tuple[int, int]]]:
    a, b, c = _vertices(cell)
    return {frozenset((a, b)), frozenset((b, c)), frozenset((c, a))}


def _candidate_cells(cell: list) -> list[list]:
    i, j = int(cell[0]), int(cell[1])
    candidates = []
    own_edges = _edges(cell)
    for ii in range(i - 2, i + 3):
        for jj in range(j - 2, j + 3):
            for orientation in ("u", "d"):
                other = [ii, jj, orientation]
                if other != cell and own_edges & _edges(other):
                    candidates.append(other)
    candidates.sort(key=lambda item: (item[0], item[1], item[2]))
    return candidates


def _edge_kind(first: list, second: list) -> tuple[int, int]:
    shared = set(_vertices(first)) & set(_vertices(second))
    if len(shared) != 2:
        raise ValueError("cells do not share one edge")
    a, b = sorted(shared)
    delta = (abs(a[0] - b[0]), abs(a[1] - b[1]))
    return delta


def _sample_path(rng: random.Random) -> list[list]:
    for attempt in range(300):
        start = [0, 0, "u" if (attempt + rng.randrange(2)) % 2 == 0 else "d"]
        path = [start]
        while len(path) < PATH_LEN:
            choices = _candidate_cells(path[-1])
            rng.shuffle(choices)
            accepted = []
            for nxt in choices:
                if nxt in path:
                    continue
                # An induced path: the new face shares no edge with an earlier
                # face except the face from which it is entered.
                if any(_edges(nxt) & _edges(old) for old in path[:-1]):
                    continue
                accepted.append(nxt)
            if not accepted:
                break
            path.append(accepted[0])
        if len(path) != PATH_LEN:
            continue
        kinds = {_edge_kind(a, b) for a, b in zip(path, path[1:])}
        if len(kinds) == 3:
            return path
    raise RuntimeError("could not construct a three-axis induced path")


def _propagate(path: list[list], corner_colors: list[dict]) -> dict[tuple[int, int], str]:
    state = {tuple(item["vertex"]): str(item["color"]) for item in corner_colors}
    if set(state) != set(_vertices(path[0])) or set(state.values()) != set(LABELS):
        raise ValueError("corner colors must bijectively label the start vertices")
    for current, nxt in zip(path, path[1:]):
        old_vertices = set(_vertices(current))
        new_vertices = set(_vertices(nxt))
        shared = old_vertices & new_vertices
        if len(shared) != 2:
            raise ValueError("consecutive triangles must share one full edge")
        old_tip = next(iter(old_vertices - shared))
        new_tip = next(iter(new_vertices - shared))
        state = {vertex: state[vertex] for vertex in shared} | {new_tip: state[old_tip]}
    return state


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    path = _sample_path(rng)
    start_vertices = list(_vertices(path[0]))
    colors = list(LABELS)
    rng.shuffle(colors)
    corner_colors = [
        {"vertex": list(vertex), "color": color}
        for vertex, color in zip(start_vertices, colors)
    ]
    final_state = _propagate(path, corner_colors)
    desired = LABELS[int(seed) % len(LABELS)]
    star_vertex = next(vertex for vertex, color in final_state.items() if color == desired)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "path": path,
        "corner_colors": corner_colors,
        "star_vertex": list(star_vertex),
    }


def _logical(vertex: tuple[int, int]) -> tuple[float, float]:
    i, j = vertex
    return i + 0.5 * j, H * j


def _layout(scene: dict) -> tuple[float, float, float]:
    points = [_logical(v) for cell in scene["path"] for v in _vertices(cell)]
    min_x, max_x = min(x for x, _ in points), max(x for x, _ in points)
    min_y, max_y = min(y for _, y in points), max(y for _, y in points)
    left, top, right, bottom = TRACK_BOX
    span_x = max(max_x - min_x, 1e-6)
    span_y = max(max_y - min_y, 1e-6)
    side = min(MAX_SIDE, (right - left - 70) / span_x, (bottom - top - 70) / span_y)
    content_w, content_h = span_x * side, span_y * side
    ox = left + (right - left - content_w) / 2.0 - min_x * side
    oy = top + (bottom - top - content_h) / 2.0 - min_y * side
    return ox, oy, side


def _pixel(vertex: tuple[int, int], layout: tuple[float, float, float]) -> tuple[int, int]:
    ox, oy, side = layout
    x, y = _logical(vertex)
    return round(ox + side * x), round(oy + side * y)


def _centroid(cell: list, layout: tuple[float, float, float]) -> tuple[float, float]:
    points = [_pixel(v, layout) for v in _vertices(cell)]
    return sum(x for x, _ in points) / 3.0, sum(y for _, y in points) / 3.0


def _star_points(cx: float, cy: float, outer: float, inner: float) -> list[tuple[int, int]]:
    points = []
    for index in range(10):
        radius = outer if index % 2 == 0 else inner
        angle = -math.pi / 2.0 + index * math.pi / 5.0
        points.append((round(cx + radius * math.cos(angle)), round(cy + radius * math.sin(angle))))
    return points


def render(scene: dict) -> Image.Image:
    if int(scene["width"]) != WIDTH or int(scene["height"]) != HEIGHT:
        raise ValueError("unexpected canvas dimensions")
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 248, 244))
    draw = ImageDraw.Draw(image)
    layout = _layout(scene)
    side = layout[2]

    draw.text((WIDTH // 2, 24), "ROLL THE STAMP — EDGE BY EDGE", font=_font(27, True), fill=(31, 38, 49), anchor="ma")
    draw.text((48, 78), "Follow the numbered triangular cells.", font=_font(19), fill=(67, 76, 88))

    polygons = []
    for index, cell in enumerate(scene["path"]):
        polygon = [_pixel(v, layout) for v in _vertices(cell)]
        polygons.append(polygon)
        draw.polygon(polygon, fill=STEP_FILLS[index])
    for polygon in polygons:
        draw.line(polygon + [polygon[0]], fill=(48, 58, 72), width=5, joint="curve")

    for index, cell in enumerate(scene["path"], 1):
        cx, cy = _centroid(cell, layout)
        radius = max(15, round(side * 0.17))
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=(252, 253, 250), outline=(48, 58, 72), width=3)
        draw.text((cx, cy - 1), str(index), font=_font(max(18, round(side * 0.20)), True), fill=(32, 40, 52), anchor="mm")

    # Direction arrows make the ordering redundant with the large numerals.
    centers = [_centroid(cell, layout) for cell in scene["path"]]
    for first, second in zip(centers, centers[1:]):
        mx, my = (first[0] + second[0]) / 2.0, (first[1] + second[1]) / 2.0
        dx, dy = second[0] - first[0], second[1] - first[1]
        length = math.hypot(dx, dy)
        ux, uy = dx / length, dy / length
        px, py = -uy, ux
        tip = (round(mx + ux * 15), round(my + uy * 15))
        draw.polygon([
            tip,
            (round(mx - ux * 8 + px * 7), round(my - uy * 8 + py * 7)),
            (round(mx - ux * 8 - px * 7), round(my - uy * 8 - py * 7)),
        ], fill=(45, 89, 126))

    pip_radius = max(13, round(side * 0.14))
    for item in scene["corner_colors"]:
        x, y = _pixel(tuple(item["vertex"]), layout)
        color = PIP_COLORS[str(item["color"])]
        draw.ellipse((x - pip_radius - 3, y - pip_radius - 3, x + pip_radius + 3, y + pip_radius + 3), fill=(255, 255, 252), outline=(37, 43, 52), width=2)
        draw.ellipse((x - pip_radius, y - pip_radius, x + pip_radius, y + pip_radius), fill=color)

    sx, sy = _pixel(tuple(scene["star_vertex"]), layout)
    star = _star_points(sx, sy, pip_radius + 7, max(7, pip_radius * 0.46))
    draw.polygon(star, fill=GOLD, outline=(37, 43, 52))
    draw.line(star + [star[0]], fill=(37, 43, 52), width=3, joint="curve")

    # A compact explanatory card uses no answer-color pixels, keeping raster inversion clean.
    draw.rounded_rectangle((826, 122, 1092, 698), radius=20, fill=(255, 253, 246), outline=(101, 92, 72), width=4)
    draw.text((959, 148), "HOW TO ROLL", font=_font(22, True), fill=(43, 45, 48), anchor="ma")
    draw.text((850, 202), "1", font=_font(22, True), fill=(45, 89, 126))
    draw.text((884, 202), "Use the full", font=_font(17), fill=(50, 54, 60))
    draw.text((884, 229), "shared edge as the hinge.", font=_font(15), fill=(50, 54, 60))
    draw.line((864, 292, 1050, 292), fill=(48, 58, 72), width=8)
    draw.ellipse((885, 277, 915, 307), fill=(116, 151, 174), outline=(37, 43, 52), width=2)
    draw.ellipse((999, 277, 1029, 307), fill=(164, 123, 171), outline=(37, 43, 52), width=2)
    draw.text((850, 342), "2", font=_font(22, True), fill=(45, 89, 126))
    draw.text((884, 342), "Those two corners", font=_font(17), fill=(50, 54, 60))
    draw.text((884, 369), "stay on the hinge edge.", font=_font(15), fill=(50, 54, 60))
    draw.arc((888, 411, 1036, 535), 190, 520, fill=(45, 89, 126), width=7)
    draw.polygon([(1037, 462), (1018, 451), (1021, 474)], fill=(45, 89, 126))
    draw.text((850, 562), "3", font=_font(22, True), fill=(45, 89, 126))
    draw.text((884, 562), "The third corner", font=_font(17), fill=(50, 54, 60))
    draw.text((884, 589), "swings into the next cell.", font=_font(15), fill=(50, 54, 60))
    draw.text((959, 651), "Which color reaches ★ ?", font=_font(18, True), fill=(43, 45, 48), anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    final_state = _propagate(list(scene["path"]), list(scene["corner_colors"]))
    star = tuple(scene["star_vertex"])
    if star not in final_state:
        raise ValueError("star is not a final-cell vertex")
    return final_state[star]


def margin(scene: dict) -> float:
    return float(_layout(scene)[2])


def is_quarantined(scene: dict) -> bool:
    try:
        path = list(scene["path"])
        if len(path) != PATH_LEN or len({tuple(c) for c in path}) != PATH_LEN:
            return True
        for index, (first, second) in enumerate(zip(path, path[1:])):
            if len(set(_vertices(first)) & set(_vertices(second))) != 2:
                return True
            if any(_edges(second) & _edges(old) for old in path[:index]):
                return True
        if len({_edge_kind(a, b) for a, b in zip(path, path[1:])}) != 3:
            return True
        if set(tuple(item["vertex"]) for item in scene["corner_colors"]) != set(_vertices(path[0])):
            return True
        if set(item["color"] for item in scene["corner_colors"]) != set(LABELS):
            return True
        if tuple(scene["star_vertex"]) not in set(_vertices(path[-1])):
            return True
        return margin(scene) < 82.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    altered = copy.deepcopy(scene)
    altered["corner_colors"] = list(reversed(altered["corner_colors"]))
    return [("reverse_serialized_corner_records", altered)]
