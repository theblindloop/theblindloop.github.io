"""Independent pixel-only inverse for cumulative histogram mass."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {"blue": (43, 103, 210), "orange": (230, 116, 39)}


def _near(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 45.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _boxes(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    work = mask.copy()
    height, width = work.shape
    output: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(work), strict=True):
        if not work[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        work[y0, x0] = False
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                        work[ny, nx] = False
                        queue.append((ny, nx))
        output.append((len(xs), min(xs), max(xs), min(ys), max(ys)))
    return output


def _bar_heights(arr: np.ndarray, rgb: tuple[int, int, int]) -> list[int]:
    # Bar components are the only large, tall regions of a series color. Panel
    # position and title text are not assumed; the recovered boxes are ordered
    # by their own horizontal centroids.
    bars = []
    for area, x0, x1, y0, y1 in _boxes(_near(arr, rgb)):
        box_width, box_height = x1 - x0 + 1, y1 - y0 + 1
        if area > 500 and box_width >= 20 and box_height >= 40 and y1 > 315:
            bars.append((x0, box_height))
    bars.sort()
    if len(bars) != 7:
        raise ValueError(f"expected seven separated bars, recovered {len(bars)}")
    if min(b[0] - a[0] for a, b in zip(bars, bars[1:])) < 25:
        raise ValueError("bar order is not resolvable")
    return [height for _, height in bars]


def _half_mass_bin(values: list[int]) -> int:
    total = sum(values)
    running = 0
    for index, value in enumerate(values):
        running += value
        if 2 * running >= total:
            if abs(2 * running - total) < 8:
                raise ValueError("cumulative mass is too close to half")
            return index
    raise ValueError("invalid empty histogram")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    bins = {name: _half_mass_bin(_bar_heights(arr, rgb)) for name, rgb in _COLORS.items()}
    if bins["blue"] == bins["orange"]:
        raise ValueError("half-mass bins are tied")
    return min(bins, key=bins.get)
