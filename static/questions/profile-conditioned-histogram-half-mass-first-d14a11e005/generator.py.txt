"""Deterministic Pillow world for normalized cumulative histogram mass."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

COLOR_RGB = {"blue": (43, 103, 210), "orange": (230, 116, 39)}
BACKGROUND = (249, 249, 247)
PLOT = (255, 255, 255)
GRID = (220, 224, 229)
INK = (42, 47, 54)
TOTAL = 420
N_BINS = 7


def _profile(rng: random.Random, crossing: int) -> list[int]:
    """Return seven positive integer heights summing to TOTAL.

    ``crossing`` is the zero-based first bin where cumulative mass exceeds half.
    Rejection sampling is deterministic and keeps every cumulative sum visibly
    away from the half-total boundary.
    """
    for _ in range(20000):
        cuts = sorted(rng.sample(range(28, TOTAL - 27), N_BINS - 1))
        values = [cuts[0], *[b - a for a, b in zip(cuts, cuts[1:])], TOTAL - cuts[-1]]
        if min(values) < 26 or max(values) > 108:
            continue
        cumulative = []
        running = 0
        for value in values:
            running += value
            cumulative.append(running)
        found = next(index for index, value in enumerate(cumulative) if 2 * value >= TOTAL)
        if found == crossing and min(abs(2 * value - TOTAL) for value in cumulative) >= 24:
            return values
    raise RuntimeError("could not sample a separated histogram profile")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    colors = ["blue", "orange"]
    winner = int(seed) % 2
    crossings = [0, 0]
    # One panel reaches half in bin 3 and the other in bin 5. Equal total mass
    # removes total colored ink as an answer cue.
    crossings[winner] = 2
    crossings[1 - winner] = 4
    heights = [_profile(rng, crossing) for crossing in crossings]
    panel_order = colors.copy()
    rng.shuffle(panel_order)
    return {
        "width": 700,
        "height": 430,
        "colors": colors,
        "heights": heights,
        "panel_order": panel_order,
    }


def _half_mass_bin(values: list[int]) -> int:
    total = sum(values)
    running = 0
    for index, value in enumerate(values):
        running += value
        if 2 * running >= total:
            return index
    raise AssertionError("positive histogram must reach its total")


def analytic_gold(scene: dict[str, Any]) -> str:
    bins = [_half_mass_bin([int(v) for v in row]) for row in scene["heights"]]
    if bins[0] == bins[1]:
        return "abstain"
    return str(scene["colors"][bins.index(min(bins))])


def margin(scene: dict[str, Any]) -> float:
    bins = sorted(_half_mass_bin([int(v) for v in row]) for row in scene["heights"])
    return float(bins[1] - bins[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    rows = [[int(v) for v in row] for row in scene["heights"]]
    bins = [_half_mass_bin(row) for row in rows]
    boundary_clearance = min(
        abs(2 * sum(row[: index + 1]) - sum(row))
        for row in rows
        for index in range(len(row))
    )
    return bins[0] == bins[1] or boundary_clearance < 20 or min(map(min, rows)) < 20


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    colors = [str(value) for value in scene["colors"]]
    heights = [[int(value) for value in row] for row in scene["heights"]]
    panel_order = [str(value) for value in scene["panel_order"]]
    by_color = dict(zip(colors, heights, strict=True))

    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    try:
        title_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 16 * scale)
        label_font = ImageFont.truetype("DejaVuSans.ttf", 12 * scale)
        panel_font = ImageFont.truetype("DejaVuSans-Bold.ttf", 14 * scale)
    except OSError:
        title_font = label_font = panel_font = ImageFont.load_default()

    draw.text((42 * scale, 17 * scale), "Cumulative mass of two equal-total histograms", fill=INK, font=title_font)
    panel_lefts = [54, 374]
    plot_top, baseline = 74, 322
    bar_width, gap = 27, 8
    for left, color in zip(panel_lefts, panel_order, strict=True):
        right = left + 272
        draw.rectangle((left * scale, plot_top * scale, right * scale, baseline * scale), fill=PLOT, outline=GRID, width=2 * scale)
        draw.text(((left + 4) * scale, 48 * scale), f"{color.capitalize()} histogram", fill=COLOR_RGB[color], font=panel_font)
        x0 = left + 17
        for index, value in enumerate(by_color[color]):
            x = x0 + index * (bar_width + gap)
            top = baseline - value * 2
            draw.rectangle((x * scale, top * scale, (x + bar_width) * scale, baseline * scale), fill=COLOR_RGB[color])
            label = str(index + 1)
            draw.text(((x + 9) * scale, 330 * scale), label, fill=INK, font=label_font)
        draw.line((left * scale, baseline * scale, right * scale, baseline * scale), fill=INK, width=2 * scale)
        draw.text(((left + 106) * scale, 354 * scale), "bin", fill=INK, font=label_font)

    draw.text((150 * scale, 392 * scale), "Each panel has the same total bar area", fill=INK, font=label_font)
    return image.resize((width, height), Image.Resampling.LANCZOS)
