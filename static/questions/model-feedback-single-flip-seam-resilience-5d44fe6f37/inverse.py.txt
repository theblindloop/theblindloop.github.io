"""Independent image-only inverse for single-flip seam resilience."""

from __future__ import annotations
from collections import deque
import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1020, 650
INK = (42, 47, 55)
SAME_FILL = (207, 232, 219)
FLIP_FILL = (242, 215, 221)
COLORS = {"amber": (232, 156, 37), "cyan": (31, 163, 190), "violet": (132, 79, 188)}
PATCH_BASES = (25, 365, 705)
GRID_X = (18, 130, 242)
GRID_Y = (160, 280, 400, 520)


class OracleError(ValueError):
    pass


def _node(p: int, r: int, c: int) -> str:
    return f"P{p}r{r}c{c}"


def _xy(node: str) -> tuple[int, int]:
    return PATCH_BASES[int(node[1])] + GRID_X[int(node[5])], GRID_Y[int(node[3])]


def _mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 15) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16)-np.asarray(color, dtype=np.int16)), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    remaining = set(map(tuple, np.argwhere(mask))); found = []
    while remaining:
        start = remaining.pop(); stack = [start]; points = [start]
        while stack:
            y, x = stack.pop()
            for nxt in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= nxt[0] < mask.shape[0] and 0 <= nxt[1] < mask.shape[1] and nxt in remaining:
                    remaining.remove(nxt); stack.append(nxt); points.append(nxt)
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _nearest_node(x: float, y: float) -> str:
    nodes = [_node(p,r,c) for p in range(3) for r in range(4) for c in range(3)]
    ranked = sorted(((x-_xy(n)[0])**2+(y-_xy(n)[1])**2,n) for n in nodes)
    if ranked[0][0] > 8**2:
        raise OracleError(f"colored mark is off lattice at {(x,y)}")
    return ranked[0][1]


def _recover_terminals(arr: np.ndarray) -> dict[str, list[str]]:
    terminals = {}
    for color, rgb in COLORS.items():
        parts = _components(_mask(arr, rgb), 350)
        if len(parts) != 2:
            raise OracleError(f"expected two {color} terminals, found {len(parts)}")
        located = [_nearest_node(float(part[:,1].mean()), float(part[:,0].mean())) for part in parts]
        if len(set(located)) != 2:
            raise OracleError(f"duplicate {color} terminal")
        terminals[color] = sorted(located)
    if len({n for pair in terminals.values() for n in pair}) != 6:
        raise OracleError("terminal colors overlap")
    return terminals


def _coverage(mask: np.ndarray, a: tuple[int,int], b: tuple[int,int]) -> float:
    hits = 0
    for t in np.linspace(.25, .75, 48):
        x = int(round(a[0]+t*(b[0]-a[0]))); y = int(round(a[1]+t*(b[1]-a[1])))
        hits += bool(mask[max(0,y-3):y+4, max(0,x-3):x+4].any())
    return hits/48


def _recover_patches(arr: np.ndarray) -> list[dict]:
    ink = _mask(arr, INK, 22); patches = []
    for p in range(3):
        edges = []
        for r in range(4):
            for c in range(2):
                a,b=_node(p,r,c),_node(p,r,c+1)
                if _coverage(ink,_xy(a),_xy(b)) >= .82: edges.append(sorted((a,b)))
        for r in range(3):
            for c in range(3):
                a,b=_node(p,r,c),_node(p,r+1,c)
                if _coverage(ink,_xy(a),_xy(b)) >= .82: edges.append(sorted((a,b)))
        patches.append({"name": chr(65+p), "edges": edges})
    if sum(len(p["edges"]) for p in patches) != 21:
        raise OracleError("implausible local link count")
    return patches


def _recover_seams(arr: np.ndarray) -> list[dict]:
    same, flip, seams = _mask(arr,SAME_FILL,14), _mask(arr,FLIP_FILL,14), []
    for left,cx in enumerate((325,665)):
        region=np.s_[140:555,cx-24:cx+25]
        counts=(int(same[region].sum()),int(flip[region].sum()))
        if max(counts)<10000 or min(counts)>80:
            raise OracleError(f"damaged seam indicator {left}")
        seams.append({"left":left,"right":left+1,"mode":"same" if counts[0]>counts[1] else "flip"})
    return seams


def _adjacency(patches: list[dict], seams: list[dict], toggled: int) -> dict[str,list[str]]:
    adjacency={_node(p,r,c):[] for p in range(3) for r in range(4) for c in range(3)}
    for patch in patches:
        for a,b in patch["edges"]: adjacency[a].append(b); adjacency[b].append(a)
    for seam in seams:
        left,mode=seam["left"],seam["mode"]
        if left==toggled: mode="flip" if mode=="same" else "same"
        for r in range(4):
            rr=r if mode=="same" else 3-r
            a,b=_node(left,r,2),_node(left+1,rr,0)
            adjacency[a].append(b); adjacency[b].append(a)
    return adjacency


def _connected(patches: list[dict], seams: list[dict], pair: list[str], toggled: int) -> bool:
    adjacency=_adjacency(patches,seams,toggled); goal=pair[1]; queue=deque([pair[0]]); seen={pair[0]}
    while queue:
        here=queue.popleft()
        if here==goal: return True
        for nxt in adjacency[here]:
            if nxt not in seen: seen.add(nxt); queue.append(nxt)
    return False


def decision_from_image(image: Image.Image) -> str:
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape[:2] != (HEIGHT,WIDTH):
        raise OracleError(f"unexpected image size {image.size}")
    terminals=_recover_terminals(arr); patches=_recover_patches(arr); seams=_recover_seams(arr)
    answers=[color for color,pair in terminals.items()
             if all(_connected(patches,seams,pair,toggled) for toggled in (0,1))]
    if len(answers)!=1:
        raise OracleError(f"expected one toggle-resilient color pair, found {answers}")
    return answers[0]
