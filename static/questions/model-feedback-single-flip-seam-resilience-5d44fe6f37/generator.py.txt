"""Deterministic oriented-seam graph with counterfactual seam toggles."""

from __future__ import annotations

from collections import deque
import copy
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1020, 650
BACKGROUND = (250, 248, 244)
INK = (42, 47, 55)
JOINT = (218, 230, 241)
PANEL = (164, 176, 189)
LABEL = (67, 76, 88)
SAME_FILL = (207, 232, 219)
FLIP_FILL = (242, 215, 221)
SOCKET = (185, 201, 218)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
COLOR_NAMES = tuple(COLORS)
PATCH_BASES = (25, 365, 705)
GRID_X = (18, 130, 242)
GRID_Y = (160, 280, 400, 520)

# A fixed audited topology. Independent row reflections of each patch create
# eight presentations while preserving both single-seam-toggle outcomes.
BASE_PATCH_EDGES = (
    (("P0r0c0", "P0r0c1"), ("P0r0c1", "P0r0c2"),
     ("P0r3c1", "P0r3c2"), ("P0r0c0", "P0r1c0"),
     ("P0r1c2", "P0r2c2"), ("P0r2c2", "P0r3c2")),
    (("P1r0c1", "P1r0c2"), ("P1r2c1", "P1r2c2"),
     ("P1r3c0", "P1r3c1"), ("P1r3c1", "P1r3c2"),
     ("P1r0c0", "P1r1c0"), ("P1r0c1", "P1r1c1"),
     ("P1r1c0", "P1r2c0"), ("P1r1c1", "P1r2c1"),
     ("P1r1c2", "P1r2c2")),
    (("P2r0c0", "P2r0c1"), ("P2r2c0", "P2r2c1"),
     ("P2r3c0", "P2r3c1"), ("P2r3c1", "P2r3c2"),
     ("P2r0c0", "P2r1c0"), ("P2r0c1", "P2r1c1")),
)
BASE_PAIRS = (
    ("P0r0c0", "P2r3c1"),  # connected after either seam toggle
    ("P0r3c1", "P2r3c2"),  # fails when the left seam toggles
    ("P0r0c1", "P2r0c1"),  # fails when the right seam toggles
)


def _node(patch: int, row: int, col: int) -> str:
    return f"P{patch}r{row}c{col}"


def _decode(node_id: str) -> tuple[int, int, int]:
    return int(node_id[1]), int(node_id[3]), int(node_id[5])


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _all_nodes() -> list[str]:
    return [_node(p, r, c) for p in range(3) for r in range(4) for c in range(3)]


def _adjacency(scene: dict, toggled: int | None = None) -> dict[str, list[str]]:
    adjacency = {node: [] for node in _all_nodes()}
    for patch in scene["patches"]:
        for a, b in patch["edges"]:
            adjacency[a].append(b)
            adjacency[b].append(a)
    for seam in scene["seams"]:
        left = int(seam["left"])
        mode = str(seam["mode"])
        if toggled == left:
            mode = "flip" if mode == "same" else "same"
        for row in range(4):
            other = row if mode == "same" else 3 - row
            a, b = _node(left, row, 2), _node(left + 1, other, 0)
            adjacency[a].append(b)
            adjacency[b].append(a)
    return adjacency


def _connected(scene: dict, start: str, goal: str, toggled: int | None = None) -> bool:
    adjacency = _adjacency(scene, toggled)
    reached, queue = {start}, deque([start])
    while queue:
        here = queue.popleft()
        if here == goal:
            return True
        for nxt in adjacency[here]:
            if nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    return False


def _toggle_results(scene: dict, pair: list[str] | tuple[str, str]) -> tuple[bool, bool]:
    return tuple(_connected(scene, pair[0], pair[1], seam) for seam in (0, 1))


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    flips = tuple(bool((seed // 6) >> patch & 1) for patch in range(3))

    def transform(node_id: str) -> str:
        patch, row, col = _decode(node_id)
        return _node(patch, 3 - row if flips[patch] else row, col)

    patches = []
    for patch, base_edges in enumerate(BASE_PATCH_EDGES):
        edges = sorted(_edge(transform(a), transform(b)) for a, b in base_edges)
        patches.append({"name": chr(65 + patch), "edges": edges})
    base_modes = ("same", "flip")
    seams = []
    for left, base_mode in enumerate(base_modes):
        parity = (base_mode == "flip") ^ flips[left] ^ flips[left + 1]
        seams.append({"left": left, "right": left + 1,
                      "mode": "flip" if parity else "same"})
    pairs = [sorted(transform(node) for node in pair) for pair in BASE_PAIRS]
    answer = COLOR_NAMES[seed % 3]
    wrong = [color for color in COLOR_NAMES if color != answer]
    if (seed // 3) % 2:
        wrong.reverse()
    scene = {"patches": patches, "seams": seams,
             "terminals": {answer: pairs[0], wrong[0]: pairs[1], wrong[1]: pairs[2]}}
    if is_quarantined(scene):
        raise RuntimeError(f"constructed invalid scene for seed {seed}")
    return scene


def _answer(scene: dict) -> str:
    answers = [color for color, pair in scene["terminals"].items()
               if _toggle_results(scene, pair) == (True, True)]
    if len(answers) != 1:
        raise ValueError(f"expected one toggle-resilient pair, got {answers}")
    return str(answers[0])


def analytic_gold(scene: dict) -> str:
    answer = _answer(scene)
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    return answer


def margin(scene: dict) -> float:
    _answer(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["patches"]) != 3 or len(scene["seams"]) != 2:
            return True
        patches = sorted(scene["patches"], key=lambda p: str(p["name"]))
        if [p["name"] for p in patches] != ["A", "B", "C"]:
            return True
        nodes, local_edges = set(_all_nodes()), []
        for patch_index, patch in enumerate(patches):
            for edge in patch["edges"]:
                if len(edge) != 2 or edge[0] not in nodes or edge[1] not in nodes:
                    return True
                p1, r1, c1 = _decode(edge[0]); p2, r2, c2 = _decode(edge[1])
                if p1 != patch_index or p2 != patch_index or abs(r1-r2)+abs(c1-c2) != 1:
                    return True
                local_edges.append(tuple(sorted(edge)))
        if len(local_edges) != 21 or len(local_edges) != len(set(local_edges)):
            return True
        seams = sorted(scene["seams"], key=lambda s: int(s["left"]))
        if [(s["left"], s["right"]) for s in seams] != [(0, 1), (1, 2)]:
            return True
        if any(s["mode"] not in {"same", "flip"} for s in seams):
            return True
        if set(scene["terminals"]) != set(COLOR_NAMES):
            return True
        marked = [node for pair in scene["terminals"].values() for node in pair]
        if any(len(pair) != 2 for pair in scene["terminals"].values()):
            return True
        if len(marked) != 6 or len(set(marked)) != 6 or not set(marked) <= nodes:
            return True
        if any({_decode(a)[0], _decode(b)[0]} != {0, 2}
               for a, b in scene["terminals"].values()):
            return True
        adjacency = _adjacency(scene)
        if any(not adjacency[node] for node in marked):
            return True
        if not all(_connected(scene, *pair) for pair in scene["terminals"].values()):
            return True
        results = {color: _toggle_results(scene, pair)
                   for color, pair in scene["terminals"].items()}
        answer = _answer(scene)
        if results[answer] != (True, True):
            return True
        if sorted(results[color] for color in COLOR_NAMES if color != answer) != [(False, True), (True, False)]:
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def _font(size: int):
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _xy(node_id: str) -> tuple[int, int]:
    patch, row, col = _decode(node_id)
    return PATCH_BASES[patch] + GRID_X[col], GRID_Y[row]


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH*scale, HEIGHT*scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    patches = {str(p["name"]): p for p in scene["patches"]}
    for patch_index, base in enumerate(PATCH_BASES):
        draw.rounded_rectangle((base*scale, 105*scale, (base+260)*scale, 575*scale),
                               radius=15*scale, outline=PANEL, width=3*scale)
        draw.text(((base+130)*scale, 69*scale), f"PATCH {chr(65+patch_index)}",
                  fill=LABEL, anchor="mm", font=_font(22*scale))
        for edge in patches[chr(65+patch_index)]["edges"]:
            a, b = sorted(edge)
            draw.line([tuple(v*scale for v in _xy(a)), tuple(v*scale for v in _xy(b))],
                      fill=INK, width=7*scale)
    terminal_of = {node: color for color, pair in scene["terminals"].items() for node in pair}
    for node_id in sorted(_all_nodes()):
        x, y = _xy(node_id); patch, _, col = _decode(node_id)
        radius = 18 if node_id in terminal_of else 11
        socket = (patch < 2 and col == 2) or (patch > 0 and col == 0)
        fill = COLORS[terminal_of[node_id]] if node_id in terminal_of else (SOCKET if socket else JOINT)
        if socket and node_id not in terminal_of:
            draw.rounded_rectangle(((x-radius)*scale, (y-radius)*scale,
                                    (x+radius)*scale, (y+radius)*scale),
                                   radius=4*scale, fill=fill, outline=INK, width=3*scale)
        else:
            draw.ellipse(((x-radius)*scale, (y-radius)*scale,
                          (x+radius)*scale, (y+radius)*scale),
                         fill=fill, outline=INK, width=3*scale)
    for seam in sorted(scene["seams"], key=lambda s: int(s["left"])):
        left, mode = int(seam["left"]), str(seam["mode"])
        cx = 325 + 340*left
        fill = SAME_FILL if mode == "same" else FLIP_FILL
        draw.rounded_rectangle(((cx-27)*scale, 112*scale, (cx+27)*scale, 568*scale),
                               radius=12*scale, fill=fill, outline=PANEL, width=2*scale)
        draw.rounded_rectangle(((cx-43)*scale, 82*scale, (cx+43)*scale, 130*scale),
                               radius=9*scale, fill=fill, outline=INK, width=2*scale)
        draw.text((cx*scale, 99*scale), mode.upper(), fill=INK,
                  anchor="mm", font=_font(16*scale))
        draw.text((cx*scale, 119*scale), "↑ ↑" if mode == "same" else "↑ ↓",
                  fill=INK, anchor="mm", font=_font(15*scale))
        for row in range(4):
            other = row if mode == "same" else 3-row
            y1, y2 = GRID_Y[row], GRID_Y[other]
            draw.line(((cx-18)*scale, y1*scale, (cx+18)*scale, y2*scale),
                      fill=LABEL, width=3*scale)
            draw.ellipse(((cx-22)*scale, (y1-4)*scale, (cx-14)*scale, (y1+4)*scale), fill=INK)
            draw.ellipse(((cx+14)*scale, (y2-4)*scale, (cx+22)*scale, (y2+4)*scale), fill=INK)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["patches"] = list(reversed(twin["patches"]))
    for patch in twin["patches"]:
        patch["edges"] = list(reversed(patch["edges"]))
    twin["seams"] = list(reversed(twin["seams"]))
    twin["terminals"] = {k: list(reversed(v)) for k, v in reversed(list(twin["terminals"].items()))}
    return [("reverse_record_orders", twin)]
