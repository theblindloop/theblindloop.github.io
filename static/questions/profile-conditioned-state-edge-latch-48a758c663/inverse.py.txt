"""Independent pixel-only inverse arm for the state-edge-latch world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette and recovers the horizontal signal
     bar's left/right box edges;
  2. groups the box top/bottom outline rows into one vertical band per panel,
     and samples each bar interior to recover the ON/OFF signal sequence in
     panel order (top to bottom);
  3. reads the magenta ringed monitor in the top panel to recover the starting
     bit (filled interior -> ON, hollow -> OFF);
  4. toggles the starting bit once per rising (OFF-to-ON) edge across the signal
     sequence and returns the final ON/OFF state.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
BOX = (95, 95, 95)
ONFILL = (55, 55, 55)
RING = (210, 0, 210)

PALETTE = [BG, BOX, ONFILL, RING]
_BG, _BOX, _ON, _RING = range(4)

PANELS = 5


class UnreadableImageError(ValueError):
    """The raster does not expose the expected panel structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _cluster(sorted_vals: list[int], tol: int) -> list[int]:
    if not sorted_vals:
        return []
    out = [[sorted_vals[0]]]
    for v in sorted_vals[1:]:
        if v - out[-1][-1] > tol:
            out.append([v])
        else:
            out[-1].append(v)
    return [r[len(r) // 2] for r in out]


def _is_dark(arr: np.ndarray, y: int, x: int) -> bool:
    return bool(np.all(np.abs(arr[y, x] - np.array(ONFILL, dtype=np.float32)) < 40))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    h, w = labels.shape

    box_ys, box_xs = np.nonzero(labels == _BOX)
    if len(box_xs) == 0:
        raise UnreadableImageError("no panel box found")
    x0 = int(box_xs.min())
    x1 = int(box_xs.max())

    row_counts = np.bincount(box_ys, minlength=h)
    edge_rows = [
        int(y) for y in range(h) if row_counts[y] > (x1 - x0) * 0.5
    ]
    edges = _cluster(sorted(edge_rows), 8)
    if len(edges) != 2 * PANELS:
        raise UnreadableImageError(
            f"expected {2 * PANELS} box edges, found {len(edges)}"
        )

    mid_x = (x0 + x1) // 2
    signal = []
    for k in range(PANELS):
        top = edges[2 * k]
        bot = edges[2 * k + 1]
        cy = (top + bot) // 2
        signal.append(_is_dark(arr, int(cy), int(mid_x)))

    ring_pts = list(zip(*np.nonzero(labels == _RING)))
    if not ring_pts:
        raise UnreadableImageError("no magenta ring found")
    rc_x = sum(p[1] for p in ring_pts) // len(ring_pts)
    rc_y = sum(p[0] for p in ring_pts) // len(ring_pts)
    start_bit = _is_dark(arr, int(rc_y), int(rc_x))

    flips = 0
    for k in range(1, PANELS):
        if (not signal[k - 1]) and signal[k]:
            flips += 1
    final = bool(start_bit) ^ (flips % 2 == 1)
    return "ON" if final else "OFF"
