"""Edge-triggered latch register: track a single bit across an ordered timeline.

The image is a vertical timeline of PANELS=5 panels read top to bottom. Each
panel shows one horizontal signal bar that is either ON (dark filled) or OFF
(hollow). A magenta ring beside the first (top) panel marks a bit whose starting
state is shown there: a dark filled interior means the bit starts ON, a hollow
interior means it starts OFF.

The bit's state changes only when the signal makes a rising edge, i.e., when the
signal is OFF in one panel and ON in the following panel; otherwise the bit keeps
its current state. The decision is the final ON/OFF state of the bit in the
bottom panel, obtained by toggling the shown starting bit once for each counted
rising edge across the whole ordered signal sequence. The scene below guarantees
exactly one consistent replay because every signal sample and the starting bit
are explicit and unambiguous.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

PANELS = 5

MARGIN = 60
PANEL_H = 90
GAP = 40
BAR_W = 320
BAR_H = 60
BAR_TOP = 15
RING_CX_OFFSET = 90
RING_R = 20
INNER_R = 12

WIDTH = MARGIN + BAR_W + 160 + MARGIN
HEIGHT = MARGIN + PANELS * PANEL_H + (PANELS - 1) * GAP + MARGIN

BG = (255, 255, 255)
BOX = (95, 95, 95)
ONFILL = (55, 55, 55)
RING = (210, 0, 210)

BAR_X0 = MARGIN
BAR_X1 = MARGIN + BAR_W


def _panel_top(k: int) -> int:
    return MARGIN + k * (PANEL_H + GAP)


def _bar_rect(k: int) -> tuple[int, int, int, int]:
    top = _panel_top(k) + BAR_TOP
    return (BAR_X0, top, BAR_X1, top + BAR_H)


def _ring_center(k: int) -> tuple[int, int]:
    return (BAR_X1 + RING_CX_OFFSET, _panel_top(k) + PANEL_H // 2)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    signal = [rng.random() < 0.5 for _ in range(PANELS)]
    start_bit = rng.random() < 0.5
    return {"signal": signal, "start_bit": start_bit}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    signal = scene["signal"]
    start_bit = scene["start_bit"]

    for k in range(PANELS):
        x0, y0, x1, y1 = _bar_rect(k)
        draw.rectangle([x0, y0, x1, y1], outline=BOX, width=3)
        if signal[k]:
            draw.rectangle([x0 + 6, y0 + 6, x1 - 6, y1 - 6], fill=ONFILL)

    cx, cy = _ring_center(0)
    draw.ellipse(
        [cx - RING_R, cy - RING_R, cx + RING_R, cy + RING_R],
        outline=RING,
        width=5,
    )
    if start_bit:
        draw.ellipse(
            [cx - INNER_R, cy - INNER_R, cx + INNER_R, cy + INNER_R], fill=ONFILL
        )
    return img


def analytic_gold(scene: dict) -> str:
    signal = scene["signal"]
    start_bit = bool(scene["start_bit"])
    flips = 0
    for k in range(1, len(signal)):
        if (not signal[k - 1]) and signal[k]:
            flips += 1
    final = start_bit ^ (flips % 2 == 1)
    return "ON" if final else "OFF"


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No render-invariant latent symmetry: ``signal`` fixes the drawn
    filled/empty bars in every panel and ``start_bit`` fixes the filled/empty
    interior of the ringed monitor, so any change to either field changes the
    raster and no alias exists."""
    return []
