"""Deterministic oriented-seam connectivity sampler and renderer."""

from __future__ import annotations

from collections import deque
import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1020, 650
BACKGROUND = (250, 248, 244)
INK = (42, 47, 55)
JOINT = (218, 230, 241)
PANEL = (164, 176, 189)
LABEL = (67, 76, 88)
SAME_FILL = (207, 232, 219)
FLIP_FILL = (242, 215, 221)
SOCKET = (185, 201, 218)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
COLOR_NAMES = tuple(COLORS)
PATCH_BASES = (25, 365, 705)
GRID_X = (18, 130, 242)
GRID_Y = (160, 280, 400, 520)


def _node(patch: int, row: int, col: int) -> str:
    return f"P{patch}r{row}c{col}"


def _decode(node_id: str) -> tuple[int, int, int]:
    return int(node_id[1]), int(node_id[3]), int(node_id[5])


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _all_nodes() -> list[str]:
    return [_node(p, r, c) for p in range(3) for r in range(4) for c in range(3)]


def _graph(scene: dict) -> tuple[dict[str, list[str]], list[set[str]]]:
    nodes = _all_nodes()
    adjacency = {node: [] for node in nodes}
    for patch in scene["patches"]:
        for a, b in patch["edges"]:
            adjacency[a].append(b)
            adjacency[b].append(a)
    for seam in scene["seams"]:
        left = int(seam["left"])
        mode = str(seam["mode"])
        for row in range(4):
            other = row if mode == "same" else 3 - row
            a, b = _node(left, row, 2), _node(left + 1, other, 0)
            adjacency[a].append(b)
            adjacency[b].append(a)
    remaining, parts = set(nodes), []
    while remaining:
        start = min(remaining)
        reached, queue = {start}, deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here]:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        parts.append(reached)
        remaining -= reached
    return adjacency, parts


def _choose_terminals(rng: random.Random, parts: list[set[str]], adjacency: dict[str, list[str]], answer: str) -> dict[str, list[str]] | None:
    spanning = [
        part for part in parts
        if len(part) >= 5
        and any(_decode(n)[0] == 0 and _decode(n)[2] == 0 for n in part)
        and any(_decode(n)[0] == 2 and _decode(n)[2] == 2 for n in part)
    ]
    rng.shuffle(spanning)
    for answer_part in spanning:
        starts = [n for n in answer_part if _decode(n)[0] == 0 and _decode(n)[2] == 0 and adjacency[n]]
        goals = [n for n in answer_part if _decode(n)[0] == 2 and _decode(n)[2] == 2 and adjacency[n]]
        rng.shuffle(starts)
        rng.shuffle(goals)
        other_parts = [p for p in parts if p is not answer_part and len(p) >= 2]
        rng.shuffle(other_parts)
        for first_part, second_part in itertools.combinations(other_parts, 2):
            pairs = [
                (a, b) for a in first_part for b in second_part
                if adjacency[a] and adjacency[b] and _decode(a)[0] != _decode(b)[0]
            ]
            rng.shuffle(pairs)
            for pair1, pair2 in itertools.combinations(pairs, 2):
                chosen = [starts[0], goals[0], *pair1, *pair2]
                if len(set(chosen)) != 6:
                    continue
                wrong = [color for color in COLOR_NAMES if color != answer]
                rng.shuffle(wrong)
                return {
                    answer: sorted((starts[0], goals[0])),
                    wrong[0]: sorted(pair1),
                    wrong[1]: sorted(pair2),
                }
    return None


def sample_scene(seed: int) -> dict:
    """Sample sparse patch graphs with one answer path spanning both seams."""

    rng = random.Random(int(seed))
    answer = COLOR_NAMES[int(seed) % 3]
    for _ in range(6000):
        patches = []
        for patch in range(3):
            edges: list[list[str]] = []
            for row in range(4):
                for col in range(2):
                    if rng.random() < 0.28:
                        edges.append(_edge(_node(patch, row, col), _node(patch, row, col + 1)))
            for row in range(3):
                for col in range(3):
                    if rng.random() < 0.28:
                        edges.append(_edge(_node(patch, row, col), _node(patch, row + 1, col)))
            patches.append({"name": chr(65 + patch), "edges": edges})
        seams = [
            {"left": 0, "right": 1, "mode": "flip" if rng.randrange(2) else "same"},
            {"left": 1, "right": 2, "mode": "flip" if rng.randrange(2) else "same"},
        ]
        scene = {"patches": patches, "seams": seams, "terminals": {}}
        adjacency, parts = _graph(scene)
        terminals = _choose_terminals(rng, parts, adjacency, answer)
        if terminals is not None:
            scene["terminals"] = terminals
            return scene
    raise RuntimeError(f"could not sample valid scene for seed {seed}")


def _answer(scene: dict) -> str:
    _, parts = _graph(scene)
    answers = []
    for color, pair in scene["terminals"].items():
        if len(pair) == 2 and any(pair[0] in part and pair[1] in part for part in parts):
            answers.append(str(color))
    if len(answers) != 1:
        raise ValueError(f"expected one connected pair, got {answers}")
    return answers[0]


def analytic_gold(scene: dict) -> str:
    answer = _answer(scene)
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    return answer


def margin(scene: dict) -> float:
    _answer(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["patches"]) != 3 or len(scene["seams"]) != 2:
            return True
        if {p["name"] for p in scene["patches"]} != {"A", "B", "C"}:
            return True
        nodes = set(_all_nodes())
        local_edges = []
        for patch_index, patch in enumerate(sorted(scene["patches"], key=lambda p: p["name"])):
            for edge in patch["edges"]:
                if len(edge) != 2 or edge[0] not in nodes or edge[1] not in nodes:
                    return True
                p1, r1, c1 = _decode(edge[0])
                p2, r2, c2 = _decode(edge[1])
                if p1 != patch_index or p2 != patch_index or abs(r1-r2) + abs(c1-c2) != 1:
                    return True
                local_edges.append(tuple(sorted(edge)))
        if len(local_edges) != len(set(local_edges)) or not 8 <= len(local_edges) <= 35:
            return True
        seams = sorted(scene["seams"], key=lambda s: int(s["left"]))
        if [(s["left"], s["right"]) for s in seams] != [(0, 1), (1, 2)]:
            return True
        if any(s["mode"] not in {"same", "flip"} for s in seams):
            return True
        if set(scene["terminals"]) != set(COLOR_NAMES):
            return True
        marked = [n for pair in scene["terminals"].values() for n in pair]
        if any(len(pair) != 2 for pair in scene["terminals"].values()):
            return True
        if len(marked) != 6 or len(set(marked)) != 6 or not set(marked) <= nodes:
            return True
        adjacency, parts = _graph(scene)
        if any(not adjacency[node] for node in marked):
            return True
        answer = _answer(scene)
        a, b = scene["terminals"][answer]
        if {_decode(a)[0], _decode(b)[0]} != {0, 2}:
            return True
        containing = next(part for part in parts if a in part)
        if len(containing) < 5:
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _xy(node_id: str) -> tuple[int, int]:
    patch, row, col = _decode(node_id)
    return PATCH_BASES[patch] + GRID_X[col], GRID_Y[row]


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    patches = {str(patch["name"]): patch for patch in scene["patches"]}
    for patch_index, base in enumerate(PATCH_BASES):
        draw.rounded_rectangle((base*scale, 105*scale, (base+260)*scale, 575*scale),
                               radius=15*scale, outline=PANEL, width=3*scale)
        draw.text(((base+130)*scale, 69*scale), f"PATCH {chr(65+patch_index)}",
                  fill=LABEL, anchor="mm", font=_font(22*scale))
        for a, b in sorted(_edge(*edge) for edge in patches[chr(65+patch_index)]["edges"]):
            draw.line([tuple(v*scale for v in _xy(a)), tuple(v*scale for v in _xy(b))],
                      fill=INK, width=7*scale)

    terminal_of = {node: color for color, pair in scene["terminals"].items() for node in pair}
    for node_id in sorted(_all_nodes()):
        x, y = _xy(node_id)
        patch, _, col = _decode(node_id)
        radius = 18 if node_id in terminal_of else 11
        fill = COLORS[terminal_of[node_id]] if node_id in terminal_of else (SOCKET if (patch < 2 and col == 2) or (patch > 0 and col == 0) else JOINT)
        if fill == SOCKET:
            draw.rounded_rectangle(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale),
                                   radius=4*scale, fill=fill, outline=INK, width=3*scale)
        else:
            draw.ellipse(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale),
                         fill=fill, outline=INK, width=3*scale)

    for seam in sorted(scene["seams"], key=lambda s: int(s["left"])):
        left = int(seam["left"])
        cx = 325 + 340 * left
        mode = str(seam["mode"])
        fill = SAME_FILL if mode == "same" else FLIP_FILL
        draw.rounded_rectangle(((cx-27)*scale, 112*scale, (cx+27)*scale, 568*scale),
                               radius=12*scale, fill=fill, outline=PANEL, width=2*scale)
        draw.rounded_rectangle(((cx-43)*scale, 82*scale, (cx+43)*scale, 130*scale),
                               radius=9*scale, fill=fill, outline=INK, width=2*scale)
        draw.text((cx*scale, 99*scale), mode.upper(), fill=INK, anchor="mm", font=_font(16*scale))
        draw.text((cx*scale, 119*scale), "↑ ↑" if mode == "same" else "↑ ↓",
                  fill=INK, anchor="mm", font=_font(15*scale))
        for row in range(4):
            other = row if mode == "same" else 3-row
            y1, y2 = GRID_Y[row], GRID_Y[other]
            draw.line(((cx-18)*scale, y1*scale, (cx+18)*scale, y2*scale), fill=LABEL, width=3*scale)
            draw.ellipse(((cx-22)*scale, (y1-4)*scale, (cx-14)*scale, (y1+4)*scale), fill=INK)
            draw.ellipse(((cx+14)*scale, (y2-4)*scale, (cx+22)*scale, (y2+4)*scale), fill=INK)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["patches"] = list(reversed(twin["patches"]))
    for patch in twin["patches"]:
        patch["edges"] = list(reversed(patch["edges"]))
    twin["seams"] = list(reversed(twin["seams"]))
    twin["terminals"] = {k: list(reversed(v)) for k, v in reversed(list(twin["terminals"].items()))}
    return [("reverse_record_orders", twin)]
