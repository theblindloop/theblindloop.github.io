"""Deterministic sampler and raster renderer for the mirror-axis symmetry search world.

Every scene is a row of five framed panels.  Each panel carries one drawn dashed
axis at its own tilt and six identical dots.  In exactly one panel the six dot
centres are an exact mirror image of themselves across that panel's own drawn
axis; in every other panel one single dot has been pushed off its mirror place.
"""

from __future__ import annotations

import hashlib
import math
import random

from PIL import Image, ImageDraw

WIDTH = 976
HEIGHT = 270
N_PANELS = 5
PANEL_W = 180
PANEL_H = 230
MARGIN_X = 14
MARGIN_Y = 20
PANEL_GAP = 12
FRAME_PX = 2

DOTS_PER_PANEL = 6
PAIRS_PER_PANEL = 3
DOT_R = 7
INTERIOR_PAD = 15
MIN_AXIS_DIST = 16.0
MAX_AXIS_DIST = 62.0
MIN_DOT_SEP = 23.0
DISP_MIN = 12.0
DISP_MAX = 19.0
MAX_TILT_DEG = 38.0

SYMMETRY_TOL = 4.0
QUARANTINE_MARGIN = 8.0

DASH_ON = 9.0
DASH_OFF = 7.0
AXIS_WIDTH = 3

BACKGROUND = (255, 255, 255)
FRAME_COLOR = (62, 62, 62)
AXIS_COLOR = (168, 168, 168)
PALETTES = (
    (188, 32, 46),
    (26, 84, 180),
    (18, 122, 72),
    (146, 52, 158),
    (198, 104, 10),
    (16, 124, 150),
)


def _panel_box(index: int) -> tuple:
    x0 = MARGIN_X + index * (PANEL_W + PANEL_GAP)
    return (x0, MARGIN_Y, x0 + PANEL_W, MARGIN_Y + PANEL_H)


def _axis_direction(angle_deg: float) -> tuple:
    angle = math.radians(angle_deg)
    return (math.sin(angle), math.cos(angle))


def _signed_offset(point, centre, direction) -> float:
    return (point[0] - centre[0]) * direction[1] - (point[1] - centre[1]) * direction[0]


def _reflect(point, centre, direction) -> tuple:
    offset = _signed_offset(point, centre, direction)
    normal = (direction[1], -direction[0])
    return (point[0] - 2.0 * offset * normal[0], point[1] - 2.0 * offset * normal[1])


def panel_asymmetry(dots: list, axis: dict) -> float:
    """Largest distance from a reflected dot centre to the nearest dot centre."""
    centre = (axis["px"], axis["py"])
    direction = _axis_direction(axis["angle_deg"])
    dots = [(float(p[0]), float(p[1])) for p in dots]
    worst = 0.0
    for point in dots:
        mirror = _reflect(point, centre, direction)
        nearest = min(math.hypot(mirror[0] - q[0], mirror[1] - q[1]) for q in dots)
        worst = max(worst, nearest)
    return worst


def _sample_panel(rng: random.Random, box: tuple, symmetric: bool) -> dict:
    x0, y0, x1, y1 = box
    lo_x, hi_x = x0 + FRAME_PX + INTERIOR_PAD, x1 - FRAME_PX - INTERIOR_PAD
    lo_y, hi_y = y0 + FRAME_PX + INTERIOR_PAD, y1 - FRAME_PX - INTERIOR_PAD
    for _ in range(600):
        angle = rng.uniform(-MAX_TILT_DEG, MAX_TILT_DEG)
        direction = _axis_direction(angle)
        centre = (
            (x0 + x1) / 2.0 + rng.uniform(-14.0, 14.0),
            (y0 + y1) / 2.0 + rng.uniform(-14.0, 14.0),
        )
        dots = []
        complete = True
        for _pair in range(PAIRS_PER_PANEL):
            placed = False
            for _attempt in range(400):
                point = (rng.uniform(lo_x, hi_x), rng.uniform(lo_y, hi_y))
                offset = _signed_offset(point, centre, direction)
                if not MIN_AXIS_DIST <= abs(offset) <= MAX_AXIS_DIST:
                    continue
                mirror = _reflect(point, centre, direction)
                if not (lo_x <= mirror[0] <= hi_x and lo_y <= mirror[1] <= hi_y):
                    continue
                if math.dist(point, mirror) < MIN_DOT_SEP:
                    continue
                clash = False
                for other in dots:
                    if math.dist(point, other) < MIN_DOT_SEP:
                        clash = True
                    if math.dist(mirror, other) < MIN_DOT_SEP:
                        clash = True
                if clash:
                    continue
                dots.append(point)
                dots.append(mirror)
                placed = True
                break
            if not placed:
                complete = False
                break
        if not complete:
            continue
        rng.shuffle(dots)
        panel = {
            "axis": {
                "px": round(centre[0], 4),
                "py": round(centre[1], 4),
                "angle_deg": round(angle, 4),
            },
            "dots": [[round(p[0], 4), round(p[1], 4)] for p in dots],
        }
        if symmetric:
            if panel_asymmetry(panel["dots"], panel["axis"]) > 1e-3:
                continue
            return panel
        for _attempt in range(400):
            index = rng.randrange(DOTS_PER_PANEL)
            heading = rng.uniform(0.0, 2.0 * math.pi)
            distance = rng.uniform(DISP_MIN, DISP_MAX)
            moved = (
                panel["dots"][index][0] + distance * math.cos(heading),
                panel["dots"][index][1] + distance * math.sin(heading),
            )
            if not (lo_x <= moved[0] <= hi_x and lo_y <= moved[1] <= hi_y):
                continue
            if abs(_signed_offset(moved, centre, direction)) < MIN_AXIS_DIST:
                continue
            clash = False
            for other_index, other in enumerate(panel["dots"]):
                if other_index != index and math.dist(moved, other) < MIN_DOT_SEP:
                    clash = True
            if clash:
                continue
            trial = {
                "axis": dict(panel["axis"]),
                "dots": [list(p) for p in panel["dots"]],
            }
            trial["dots"][index] = [round(moved[0], 4), round(moved[1], 4)]
            residual = panel_asymmetry(trial["dots"], trial["axis"])
            if DISP_MIN <= residual <= DISP_MAX + 2.0:
                return trial
    raise RuntimeError("panel sampling failed to converge")


def sample_scene(seed: int) -> dict:
    digest = hashlib.sha256(f"mirror_axis_panel|{int(seed)}".encode("utf-8")).hexdigest()
    rng = random.Random(digest)
    answer = rng.randrange(N_PANELS)
    panels = [
        _sample_panel(rng, _panel_box(index), index == answer)
        for index in range(N_PANELS)
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "palette": rng.randrange(len(PALETTES)),
        "panels": panels,
    }


def _draw_axis(draw: ImageDraw.ImageDraw, panel: dict, box: tuple) -> None:
    x0, y0, x1, y1 = box
    lo_x, hi_x = x0 + FRAME_PX + 2, x1 - FRAME_PX - 2
    lo_y, hi_y = y0 + FRAME_PX + 2, y1 - FRAME_PX - 2
    centre = (panel["axis"]["px"], panel["axis"]["py"])
    direction = _axis_direction(panel["axis"]["angle_deg"])
    t_lo, t_hi = -1.0e9, 1.0e9
    for lo, hi, base, step in (
        (lo_x, hi_x, centre[0], direction[0]),
        (lo_y, hi_y, centre[1], direction[1]),
    ):
        if abs(step) < 1e-9:
            if not lo <= base <= hi:
                return
            continue
        first = (lo - base) / step
        second = (hi - base) / step
        t_lo = max(t_lo, min(first, second))
        t_hi = min(t_hi, max(first, second))
    if t_hi <= t_lo:
        return
    position = t_lo
    while position < t_hi:
        end = min(position + DASH_ON, t_hi)
        draw.line(
            [
                (centre[0] + position * direction[0], centre[1] + position * direction[1]),
                (centre[0] + end * direction[0], centre[1] + end * direction[1]),
            ],
            fill=AXIS_COLOR,
            width=AXIS_WIDTH,
        )
        position = end + DASH_OFF


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    colour = PALETTES[int(scene["palette"]) % len(PALETTES)]
    for index, panel in enumerate(scene["panels"]):
        box = _panel_box(index)
        draw.rectangle(list(box), outline=FRAME_COLOR, width=FRAME_PX)
        _draw_axis(draw, panel, box)
        for point in panel["dots"]:
            draw.ellipse(
                [
                    point[0] - DOT_R,
                    point[1] - DOT_R,
                    point[0] + DOT_R,
                    point[1] + DOT_R,
                ],
                fill=colour,
            )
    return image


def _asymmetries(scene: dict) -> list:
    values = []
    for panel in scene["panels"]:
        values.append(panel_asymmetry(panel["dots"], panel["axis"]))
    return values


def margin(scene: dict) -> float:
    values = sorted(_asymmetries(scene))
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    values = _asymmetries(scene)
    ordered = sorted(values)
    return bool(ordered[0] > SYMMETRY_TOL or (ordered[1] - ordered[0]) < QUARANTINE_MARGIN)


def analytic_gold(scene: dict) -> str:
    if is_quarantined(scene):
        return "ambiguous"
    values = _asymmetries(scene)
    return f"panel-{min(range(len(values)), key=lambda i: values[i]) + 1}"


def latent_symmetries(scene: dict) -> list:
    """Declared latent aliases: pixel-identical rewrites of the same scene."""
    transforms = []

    shifted = {
        "width": scene["width"],
        "height": scene["height"],
        "palette": int(scene["palette"]) + len(PALETTES),
        "panels": [
            {"axis": dict(p["axis"]), "dots": [list(d) for d in p["dots"]]}
            for p in scene["panels"]
        ],
    }
    transforms.append(("palette_period", shifted))

    reversed_dots = {
        "width": scene["width"],
        "height": scene["height"],
        "palette": scene["palette"],
        "panels": [
            {"axis": dict(p["axis"]), "dots": [list(d) for d in reversed(p["dots"])]}
            for p in scene["panels"]
        ],
    }
    transforms.append(("dot_order_reversal", reversed_dots))

    rotated_dots = {
        "width": scene["width"],
        "height": scene["height"],
        "palette": scene["palette"],
        "panels": [
            {
                "axis": dict(p["axis"]),
                "dots": [list(d) for d in p["dots"][2:] + p["dots"][:2]],
            }
            for p in scene["panels"]
        ],
    }
    transforms.append(("dot_order_rotation", rotated_dots))
    return transforms
