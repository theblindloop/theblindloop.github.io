"""Row/column grid-toggle state tracker (state tracking).

The image shows one static 3x3 grid of binary cells: each cell is either FILLED
(a solid dark square) or HOLLOW (white with a thin outline). A thick orange ring
marks exactly one cell -- the TARGET cell whose final state is asked about. No
other cell matters; the rest are counterbalancing distractors that never change.

To the right of the grid, a row of three operator boxes is read left to right.
Each box holds exactly one colored disk in one of three slots (left / middle /
right). A RED disk means "toggle the row of that index" and a BLUE disk means
"toggle the column of that index"; the slot (1, 2 or 3) gives the row/column
index. "Toggle" flips the fill state (filled <-> hollow) of every cell in that
row/column.

The latent scene stores ``grid`` (9 fill bits in row-major order), ``target``
(the index 0..8 of the ringed cell) and ``ops`` (a length-3 list of (axis,
index) pairs, axis in {"row","col"}, index in 0..2). The decision is the final
fill state of the ringed cell, which equals its initial fill XOR the parity of
the number of declared toggles whose row/column passes through it. Because the
final grid is never drawn, the answer can only be obtained by replaying the
operators from the ringed initial cell -- a genuine sequential composition of a
row operator and a column operator (joint parity).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 900
HEIGHT = 320

CS = 56              # cell size (px)
GPAD = 24            # padding between the cell block and the panel outline
GRID_X = 80          # x of first cell's left edge
GRID_Y = 24          # y of first cell's top edge
CELLS = 3

OPS_Y = 252          # top of the operator boxes
OPS_BOX_W = 120
OPS_BOX_H = 54
OPS_X0 = 60
OPS_SPACING = 250
SLOTS = 3
DISK_R = 12

BG = (255, 255, 255)
GRID_BORDER = (190, 190, 190)
INK = (35, 35, 35)
RING = (240, 150, 20)
ROW_COLOR = (205, 40, 40)
COL_COLOR = (45, 70, 210)

STATE_LABELS = {0: "outline", 1: "filled"}
OP_COUNT = 3


def cell_panel_box():
    x0 = GRID_X - GPAD
    y0 = GRID_Y - GPAD
    x1 = GRID_X + CELLS * CS + GPAD
    y1 = GRID_Y + CELLS * CS + GPAD
    return (x0, y0, x1, y1)


def cell_center(idx: int) -> tuple[int, int]:
    r, c = divmod(idx, CELLS)
    return (int(GRID_X + CS * (c + 0.5)), int(GRID_Y + CS * (r + 0.5)))


def op_box_center(op_index: int) -> tuple[int, int]:
    left = OPS_X0 + op_index * OPS_SPACING
    return (int(left + OPS_BOX_W * 0.5), int(OPS_Y + OPS_BOX_H * 0.5))


def slot_center(op_index: int, slot: int) -> tuple[int, int]:
    left = OPS_X0 + op_index * OPS_SPACING
    x = int(left + OPS_BOX_W * (slot + 0.5) / SLOTS)
    return (x, op_box_center(op_index)[1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    grid = [rng.randint(0, 1) for _ in range(CELLS * CELLS)]
    target = rng.randrange(CELLS * CELLS)
    ops = []
    for _ in range(OP_COUNT):
        axis = rng.choice(["row", "col"])
        index = rng.randrange(CELLS)
        ops.append((axis, index))
    return {"grid": grid, "target": target, "ops": ops}


def _apply_ops(grid, target, ops):
    r0, c0 = divmod(target, CELLS)
    state = grid[target]
    for axis, index in ops:
        hit = (axis == "row" and index == r0) or (axis == "col" and index == c0)
        if hit:
            state = 1 - state
    return state


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    grid = scene["grid"]
    target = scene["target"]
    ops = scene["ops"]

    panel = cell_panel_box()
    draw.rectangle(panel, outline=GRID_BORDER, width=3)

    for idx, fill in enumerate(grid):
        r, c = divmod(idx, CELLS)
        x0 = GRID_X + c * CS
        y0 = GRID_Y + r * CS
        x1 = x0 + CS
        y1 = y0 + CS
        if fill:
            draw.rectangle([x0, y0, x1, y1], fill=INK, outline=GRID_BORDER, width=2)
        else:
            draw.rectangle([x0, y0, x1, y1], fill=BG, outline=GRID_BORDER, width=2)

    cx, cy = cell_center(target)
    inset = 8
    draw.rectangle(
        [cx - CS / 2 + inset, cy - CS / 2 + inset, cx + CS / 2 - inset, cy + CS / 2 - inset],
        outline=RING,
        width=5,
    )

    for i, (axis, index) in enumerate(ops):
        left = OPS_X0 + i * OPS_SPACING
        draw.rectangle(
            [left, OPS_Y, left + OPS_BOX_W, OPS_Y + OPS_BOX_H], outline=GRID_BORDER, width=3
        )
        color = ROW_COLOR if axis == "row" else COL_COLOR
        sx, sy = slot_center(i, index)
        draw.ellipse([sx - DISK_R, sy - DISK_R, sx + DISK_R, sy + DISK_R], fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    state = _apply_ops(scene["grid"], scene["target"], scene["ops"])
    return STATE_LABELS[state]


def margin(scene: dict) -> float:
    return 12.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry.

    Every latent field is directly render-evident: ``grid`` fixes each drawn
    cell's fill, ``target`` fixes which cell the orange ring surrounds, and
    ``ops`` fixes the three drawn operator disks (colour and slot). No
    reparametrization yields a pixel-identical image with a different gold,
    because the target fill and every operator are visible in the raster, so
    there is no nontrivial render-invariant alias.
    """
    return []
