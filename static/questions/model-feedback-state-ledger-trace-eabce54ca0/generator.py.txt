"""Renderer and analytic world for an ordered finite-state visual trace."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1200
HEIGHT = 760
BACKGROUND = (250, 251, 253)
INK = (25, 31, 42)
MUTED = (88, 98, 113)
PANEL = (255, 255, 255)
STATE_NAMES = ("alpha", "beta", "gamma", "delta")
STATE_COLORS = (
    (37, 99, 235),
    (217, 119, 6),
    (22, 163, 74),
    (190, 24, 93),
)
EVENT_COLORS = (
    (8, 145, 178),
    (124, 58, 237),
    (220, 38, 38),
)
EVENT_NAMES = ("triangle", "diamond", "hexagon")

START_CENTER = (170, 220)
TABLE_EVENT_X = (685, 850, 1015)
TABLE_STATE_Y = (190, 235, 280, 325)
SEQUENCE_X = (120, 275, 430, 585, 740, 895, 1050)
SEQUENCE_Y = 435
MODE_KEY_X = (175, 425, 675, 925)
MODE_KEY_Y = 625


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    candidates = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    )
    for path in candidates:
        try:
            return ImageFont.truetype(path, size=size)
        except OSError:
            pass
    return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[int, list[int], list[list[int]]]:
    start = int(scene["start_state"])
    events = [int(value) for value in scene["event_sequence"]]
    table = [[int(value) for value in row] for row in scene["transition_table"]]
    if not 0 <= start < 4:
        raise ValueError("start_state must be one of four states")
    if len(events) != 7 or any(not 0 <= value < 3 for value in events):
        raise ValueError("event_sequence must contain seven valid events")
    if len(table) != 4 or any(len(row) != 3 for row in table):
        raise ValueError("transition_table must be four by three")
    if any(value < 0 or value >= 4 for row in table for value in row):
        raise ValueError("transition table contains an invalid state")
    return start, events, table


def _trace(start: int, events: list[int], table: list[list[int]]) -> list[int]:
    state = start
    path = [state]
    for event in events:
        state = table[state][event]
        path.append(state)
    return path


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a randomized table and event stream with balanced terminal labels."""
    seed = int(seed)
    target = seed % 4
    for attempt in range(2000):
        rng = random.Random(seed * 1_000_003 + attempt * 91_879 + 17)
        table = [[rng.randrange(4) for _ in range(3)] for _ in range(4)]
        events = [rng.randrange(3) for _ in range(7)]
        start = rng.randrange(4)
        path = _trace(start, events, table)
        if (
            path[-1] == target
            and len(set(events)) == 3
            and len(set(path)) >= 3
            and all(len(set(row)) >= 2 for row in table)
        ):
            return {
                "start_state": start,
                "event_sequence": events,
                "transition_table": table,
            }
    raise RuntimeError(f"could not sample a balanced state trace for seed {seed}")


def _draw_state_chip(draw: ImageDraw.ImageDraw, center: tuple[int, int], state: int, radius: int) -> None:
    cx, cy = center
    draw.ellipse(
        (cx - radius, cy - radius, cx + radius, cy + radius),
        fill=STATE_COLORS[state],
        outline=INK,
        width=2,
    )


def _event_points(cx: int, cy: int, event: int, radius: int) -> list[tuple[int, int]]:
    if event == 0:
        return [(cx, cy - radius), (cx + radius, cy + radius), (cx - radius, cy + radius)]
    if event == 1:
        return [(cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)]
    return [
        (cx + radius, cy),
        (cx + radius // 2, cy - radius),
        (cx - radius // 2, cy - radius),
        (cx - radius, cy),
        (cx - radius // 2, cy + radius),
        (cx + radius // 2, cy + radius),
    ]


def _draw_event_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], event: int, radius: int) -> None:
    draw.polygon(_event_points(center[0], center[1], event, radius), fill=EVENT_COLORS[event], outline=INK)


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=(205, 211, 220), width=2)


def render(scene: dict[str, Any]) -> Image.Image:
    start, events, table = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 25), "MODE TRANSITION BOARD", font=_font(30, True), fill=INK)
    draw.text(
        (40, 64),
        "A state chip changes after each event; carry the new row forward.",
        font=_font(17),
        fill=MUTED,
    )

    _panel(draw, (35, 100, 325, 340))
    draw.text((58, 126), "GOLD START MODE", font=_font(19, True), fill=INK)
    _draw_state_chip(draw, START_CENTER, start, 46)
    draw.text((228, 202), STATE_NAMES[start].upper(), font=_font(22, True), fill=INK)
    draw.text((228, 232), "first row", font=_font(16), fill=MUTED)

    _panel(draw, (365, 100, 1165, 340))
    draw.text((390, 126), "TRANSITION TABLE", font=_font(19, True), fill=INK)
    draw.text((390, 151), "current mode", font=_font(14), fill=MUTED)
    draw.text((620, 151), "next event", font=_font(14), fill=MUTED)

    grid_left, grid_right, grid_top, grid_bottom = 575, 1095, 166, 342
    for x in (grid_left, 748, 932, grid_right):
        draw.line((x, grid_top, x, grid_bottom), fill=(198, 204, 214), width=2)
    for y in (grid_top, 210, 254, 298, grid_bottom):
        draw.line((grid_left, y, grid_right, y), fill=(198, 204, 214), width=2)
    for x, event in zip(TABLE_EVENT_X, range(3), strict=True):
        _draw_event_icon(draw, (x, 148), event, 16)
    for state, y in enumerate(TABLE_STATE_Y):
        _draw_state_chip(draw, (415, y), state, 14)
        draw.text((445, y - 11), STATE_NAMES[state], font=_font(17, True), fill=INK)
        for x, value in zip(TABLE_EVENT_X, table[state], strict=True):
            _draw_state_chip(draw, (x, y), value, 17)

    _panel(draw, (35, 370, 1165, 500))
    draw.text((58, 391), "EVENT ORDER", font=_font(19, True), fill=INK)
    draw.text((220, 393), "read the numbered icons from left to right", font=_font(16), fill=MUTED)
    for number, (x, event) in enumerate(zip(SEQUENCE_X, events, strict=True), start=1):
        draw.rounded_rectangle((x - 58, 420, x + 58, 486), radius=12, fill=(247, 249, 252), outline=(190, 198, 210), width=2)
        draw.ellipse((x - 48, 427, x - 26, 449), fill=INK)
        draw.text((x - 37, 438), str(number), font=_font(12, True), fill=(255, 255, 255), anchor="mm")
        _draw_event_icon(draw, (x + 12, SEQUENCE_Y), event, 19)

    _panel(draw, (35, 530, 1165, 710))
    draw.text((58, 552), "MODE KEY", font=_font(19, True), fill=INK)
    draw.text(
        (220, 554),
        "after event 7, report the name beside the final chip",
        font=_font(16),
        fill=MUTED,
    )
    for x, state in zip(MODE_KEY_X, range(4), strict=True):
        _draw_state_chip(draw, (x, MODE_KEY_Y), state, 30)
        draw.text((x - 60, 672), STATE_NAMES[state].upper(), font=_font(16, True), fill=INK)
    draw.text(
        (58, 690),
        "table rule: current row + event column = next mode",
        font=_font(15),
        fill=MUTED,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start, events, table = _validate_scene(scene)
    return STATE_NAMES[_trace(start, events, table)[-1]]


def margin(scene: dict[str, Any]) -> float:
    _validate_scene(scene)
    return 17.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    _validate_scene(scene)
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
