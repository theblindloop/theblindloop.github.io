"""Independent image-only inverse for the mode transition board."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1200
HEIGHT = 760
STATE_NAMES = ("alpha", "beta", "gamma", "delta")
STATE_COLORS = np.array(
    [
        (37, 99, 235),
        (217, 119, 6),
        (22, 163, 74),
        (190, 24, 93),
    ],
    dtype=np.int16,
)
EVENT_COLORS = np.array(
    [
        (8, 145, 178),
        (124, 58, 237),
        (220, 38, 38),
    ],
    dtype=np.int16,
)
START_CENTER = (170, 220)
TABLE_EVENT_X = (685, 850, 1015)
TABLE_STATE_Y = (190, 235, 280, 325)
SEQUENCE_X = (120, 275, 430, 585, 740, 895, 1050)
SEQUENCE_Y = 435


def _nearest_state(rgb: np.ndarray) -> int:
    distances = ((STATE_COLORS.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=1)
    value = int(np.argmin(distances))
    if int(distances[value]) > 1800:
        raise ValueError("state chip is missing or unreadable")
    return value


def _nearest_event(rgb: np.ndarray) -> int:
    distances = ((EVENT_COLORS.astype(np.int32) - rgb.astype(np.int32)) ** 2).sum(axis=1)
    value = int(np.argmin(distances))
    if int(distances[value]) > 1800:
        raise ValueError("event icon is missing or unreadable")
    return value


def _event_from_card(arr: np.ndarray, x: int) -> int:
    crop = arr[405:475, x - 45 : x + 50]
    colors = EVENT_COLORS.reshape(3, 1, 1, 3)
    matches = np.all(crop[None, ...] == colors, axis=3)
    counts = matches.reshape(3, -1).sum(axis=1)
    event = int(np.argmax(counts))
    if int(counts[event]) < 120:
        raise ValueError("event card has no legible colored icon")
    return event


def decision_from_image(image: Image.Image) -> str:
    """Recover the terminal mode from pixels only."""
    image = image.convert("RGB")
    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected board size {image.size}")
    arr = np.asarray(image, dtype=np.int16)
    start = _nearest_state(arr[START_CENTER[1], START_CENTER[0]])
    events = [_event_from_card(arr, x) for x in SEQUENCE_X]
    table = []
    for y in TABLE_STATE_Y:
        row = [_nearest_state(arr[y, x]) for x in TABLE_EVENT_X]
        table.append(row)
    state = start
    for event in events:
        state = table[state][event]
    return STATE_NAMES[state]
