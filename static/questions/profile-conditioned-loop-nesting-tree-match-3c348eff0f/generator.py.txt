"""Deterministic renderer and analytic arm for loop-nesting tree matching."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


WIDTH = 680
HEIGHT = 680
BACKGROUND = [247, 245, 239]
DARK = (31, 38, 45)
FRAME_COLORS = {
    "gray": (112, 118, 124),
    "blue": (35, 116, 178),
    "orange": (225, 125, 35),
    "purple": (137, 86, 171),
}
ANSWER_COLORS = ("blue", "orange", "purple")
FRAME_WIDTH = 8
LOOP_WIDTH = 6
PANEL_SIZE = 300
SLOTS = ((24, 24), (356, 24), (24, 356), (356, 356))
PATTERNS = ((3, 0, 0), (2, 1, 0), (1, 1, 1))


def _transform_box(box: list[int], quarter_turns: int) -> list[int]:
    x0, y0, x1, y1 = box
    points = [(x0, y0), (x0, y1), (x1, y0), (x1, y1)]
    for _ in range(quarter_turns % 4):
        points = [(PANEL_SIZE - y, x) for x, y in points]
    return [
        min(x for x, _ in points),
        min(y for _, y in points),
        max(x for x, _ in points),
        max(y for _, y in points),
    ]


def _make_loops(origin: tuple[int, int], pattern_index: int, rng: random.Random) -> list[list[int]]:
    """Make seven rectangles with level counts 1/3/3 and one of three tree shapes."""
    counts = list(PATTERNS[pattern_index])
    rng.shuffle(counts)
    loops: list[list[int]] = [[22, 22, 278, 278]]
    child_xs = (44, 121, 198)
    for child_index, left in enumerate(child_xs):
        child = [left, 48, left + 58, 252]
        loops.append(child)
        count = counts[child_index]
        centers = {0: (), 1: (150,), 2: (107, 193), 3: (85, 150, 215)}[count]
        for center in centers:
            loops.append([left + 16, center - 18, left + 42, center + 18])

    turns = rng.randrange(4)
    ox, oy = origin
    transformed = []
    for box in loops:
        local = _transform_box(box, turns)
        transformed.append([local[0] + ox, local[1] + oy, local[2] + ox, local[3] + oy])
    rng.shuffle(transformed)  # nuisance draw order, not a topology label
    return transformed


def _contains(outer: list[int], inner: list[int]) -> bool:
    return outer[0] < inner[0] and outer[1] < inner[1] and outer[2] > inner[2] and outer[3] > inner[3]


def _forest_signature(boxes: list[list[int]]) -> tuple[str, ...]:
    parents: list[int | None] = []
    for i, box in enumerate(boxes):
        containers = [j for j, other in enumerate(boxes) if i != j and _contains(other, box)]
        parent = min(containers, key=lambda j: (boxes[j][2] - boxes[j][0]) * (boxes[j][3] - boxes[j][1])) if containers else None
        parents.append(parent)
    children = {i: [] for i in range(len(boxes))}
    roots = []
    for i, parent in enumerate(parents):
        if parent is None:
            roots.append(i)
        else:
            children[parent].append(i)

    def encode(node: int) -> str:
        return "(" + "".join(sorted(encode(child) for child in children[node])) + ")"

    return tuple(sorted(encode(root) for root in roots))


def _decision_from_panels(panels: list[dict]) -> str:
    reference = [panel for panel in panels if panel["frame_color"] == "gray"]
    if len(reference) != 1:
        raise ValueError("scene must contain exactly one gray reference")
    reference_signature = _forest_signature(reference[0]["loops"])
    matches = [
        panel["frame_color"]
        for panel in panels
        if panel["frame_color"] != "gray" and _forest_signature(panel["loops"]) == reference_signature
    ]
    if len(matches) != 1 or matches[0] not in ANSWER_COLORS:
        raise ValueError(f"expected one colored tree match, found {matches}")
    return matches[0]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_pattern = (int(seed) // 3) % len(PATTERNS)
    answer_color = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    remaining_patterns = [index for index in range(len(PATTERNS)) if index != reference_pattern]
    rng.shuffle(remaining_patterns)
    pattern_by_color = {answer_color: reference_pattern}
    for color, pattern in zip((c for c in ANSWER_COLORS if c != answer_color), remaining_patterns):
        pattern_by_color[color] = pattern

    identities = ["gray", *ANSWER_COLORS]
    rng.shuffle(identities)
    panels = []
    for identity, origin in zip(identities, SLOTS):
        pattern = reference_pattern if identity == "gray" else pattern_by_color[identity]
        ox, oy = origin
        panels.append(
            {
                "frame_color": identity,
                "frame": [ox, oy, ox + PANEL_SIZE, oy + PANEL_SIZE],
                "loops": _make_loops(origin, pattern, rng),
            }
        )
    rng.shuffle(panels)
    return {"width": WIDTH, "height": HEIGHT, "background": list(BACKGROUND), "panels": panels}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        draw.rectangle(panel["frame"], outline=FRAME_COLORS[panel["frame_color"]], width=FRAME_WIDTH)
        for box in panel["loops"]:
            draw.rectangle(box, outline=DARK, width=LOOP_WIDTH)
    return image


def analytic_gold(scene: dict) -> str:
    return _decision_from_panels(scene["panels"])


def _box_clearance(a: list[int], b: list[int]) -> float:
    if _contains(a, b):
        return float(min(b[0] - a[0], b[1] - a[1], a[2] - b[2], a[3] - b[3]) - LOOP_WIDTH)
    if _contains(b, a):
        return _box_clearance(b, a)
    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return -1.0
    return float(math.hypot(dx, dy))


def _raster_clearance(panels: list[dict]) -> float:
    clearances = []
    for panel in panels:
        boxes = panel["loops"]
        if len(boxes) != 7:
            return -1.0
        clearances.extend(_box_clearance(boxes[i], boxes[j]) for i in range(len(boxes)) for j in range(i + 1, len(boxes)))
    return min(clearances) if clearances else -1.0


def margin(scene: dict) -> float:
    """Runner-up gap capped by the narrowest visible topological clearance."""
    panels = scene["panels"]
    reference = next(panel for panel in panels if panel["frame_color"] == "gray")
    ref_signature = _forest_signature(reference["loops"])
    candidate_signatures = [_forest_signature(p["loops"]) for p in panels if p["frame_color"] != "gray"]
    runner_up_gap = 1.0 if candidate_signatures.count(ref_signature) == 1 else 0.0
    return float(min(_raster_clearance(panels), 8.0 * runner_up_gap))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 7.5 or analytic_gold(scene) not in ANSWER_COLORS
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["loops"].reverse()
    return [("reverse_all_loop_draw_orders", twin)]
