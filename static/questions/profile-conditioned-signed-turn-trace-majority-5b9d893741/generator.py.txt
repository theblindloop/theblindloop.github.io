"""Raster renderer and analytic contract for signed turn aggregation."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


RENDERER_VERSION = "signed_turn_trace_majority-0.1.0"
WIDTH = 720
HEIGHT = 560

BACKGROUND_PALETTE = (
    (247, 249, 246),
    (250, 247, 241),
    (246, 246, 250),
)
ROUTE_PALETTE = (
    (28, 52, 73),
    (38, 63, 82),
    (42, 50, 75),
)
MARKER_PALETTE = (
    (127, 76, 170),
    (146, 81, 158),
    (105, 84, 174),
)
DISTRACTOR_PALETTE = (
    (202, 207, 205),
    (205, 203, 211),
    (211, 207, 198),
)
START_COLOR = (44, 145, 91)
TERMINAL_COLOR = (214, 75, 83)

# Seven segments, six explicitly marked internal vertices.  In screen
# coordinates a positive dy1*dx2-dx1*dy2 is a left turn for travel from the
# green endpoint to the red endpoint.
LEFT_PATH = (
    (140.0, 300.0),
    (200.0, 420.0),
    (260.0, 432.0),
    (320.0, 348.0),
    (380.0, 408.0),
    (440.0, 390.0),
    (500.0, 270.0),
    (560.0, 270.0),
)
RIGHT_PATH = tuple((x, 560.0 - y) for x, y in LEFT_PATH)

DISTRACTOR_PATHS = (
    ((90.0, 92.0), (180.0, 116.0), (270.0, 91.0)),
    ((420.0, 91.0), (510.0, 116.0), (630.0, 92.0)),
    ((90.0, 472.0), (180.0, 447.0), (270.0, 474.0)),
    ((420.0, 474.0), (520.0, 447.0), (630.0, 472.0)),
)


def _transform_point(point: tuple[float, float], angle: int, offset: tuple[float, float]) -> list[float]:
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    dx, dy = point[0] - cx, point[1] - cy
    radians = math.radians(angle)
    cosine, sine = math.cos(radians), math.sin(radians)
    return [
        round(cx + dx * cosine - dy * sine + offset[0], 3),
        round(cy + dx * sine + dy * cosine + offset[1], 3),
    ]


def _turn_counts(points: list[list[float]] | tuple[tuple[float, float], ...]) -> tuple[int, int]:
    left = 0
    right = 0
    for index in range(1, len(points) - 1):
        x0, y0 = float(points[index - 1][0]), float(points[index - 1][1])
        x1, y1 = float(points[index][0]), float(points[index][1])
        x2, y2 = float(points[index + 1][0]), float(points[index + 1][1])
        cross = (y1 - y0) * (x2 - x1) - (x1 - x0) * (y2 - y1)
        if abs(cross) < 1e-6:
            raise ValueError("unmarked route vertex is not a directional turn")
        if cross > 0.0:
            left += 1
        else:
            right += 1
    return left, right


def sample_scene(seed: int) -> dict:
    """Sample balanced, transformed paths with independent visible nuisances."""

    seed = int(seed)
    rng = random.Random(seed)
    base = LEFT_PATH if seed % 2 == 0 else RIGHT_PATH
    angle = rng.choice((0, 90, 180, 270))
    offset = (rng.uniform(-14.0, 14.0), rng.uniform(-12.0, 12.0))
    path = [_transform_point(point, angle, offset) for point in base]
    distractors = [
        [_transform_point(point, angle, offset) for point in stroke]
        for stroke in DISTRACTOR_PATHS
    ]
    rng.shuffle(distractors)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": rng.choice(BACKGROUND_PALETTE),
        "route_color": rng.choice(ROUTE_PALETTE),
        "marker_color": rng.choice(MARKER_PALETTE),
        "start_color": START_COLOR,
        "terminal_color": TERMINAL_COLOR,
        "distractor_color": rng.choice(DISTRACTOR_PALETTE),
        "route_width": 11,
        "marker_radius": 14,
        "marker_width": 4,
        "path": path,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    for stroke in scene["distractors"]:
        draw.line(
            [(float(point[0]), float(point[1])) for point in stroke],
            fill=tuple(scene["distractor_color"]),
            width=5,
            joint="curve",
        )

    path = [(float(point[0]), float(point[1])) for point in scene["path"]]
    draw.line(
        path,
        fill=tuple(scene["route_color"]),
        width=int(scene["route_width"]),
        joint="curve",
    )

    radius = int(scene["marker_radius"])
    marker_width = int(scene["marker_width"])
    for x, y in path[1:-1]:
        draw.ellipse(
            (x - radius, y - radius, x + radius, y + radius),
            outline=tuple(scene["marker_color"]),
            width=marker_width,
        )

    endpoint_radius = radius + 4
    for (x, y), color in (
        (path[0], tuple(scene["start_color"])),
        (path[-1], tuple(scene["terminal_color"])),
    ):
        draw.ellipse(
            (x - endpoint_radius, y - endpoint_radius, x + endpoint_radius, y + endpoint_radius),
            fill=color,
            outline=(238, 239, 237),
            width=3,
        )
    return image


def analytic_gold(scene: dict) -> str:
    left, right = _turn_counts(scene["path"])
    if left == right:
        raise ValueError("turn-count tie is outside the question boundary")
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    left, right = _turn_counts(scene["path"])
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("distractor_order", twin)]
