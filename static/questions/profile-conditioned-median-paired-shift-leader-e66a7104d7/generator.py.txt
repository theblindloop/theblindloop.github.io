"""Deterministic sampler and raster renderer for median paired shifts."""

from __future__ import annotations

import random
import statistics
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 940, 640
SCALE = 3
QUARANTINE_PX = 4.0

PALETTES = (
    {"blue": "#1769aa", "orange": "#e66b15"},
    {"blue": "#225ea8", "orange": "#d97706"},
    {"blue": "#2166ac", "orange": "#e08214"},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _medians(shifts: list[list[int]]) -> list[float]:
    return [float(statistics.median(panel)) for panel in shifts]


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a balanced scene with mean, maximum, and direction-count shortcuts blocked."""

    seed = int(seed)
    rng = random.Random(seed)
    winner = seed % 4
    runner = (winner + 1 + (seed // 4) % 3) % 4
    near = seed % 11 == 0

    winner_pattern = [-60, 28, 32, 38, 52] if near else [-60, 34, 40, 46, 54]
    patterns = [
        winner_pattern,
        [-22, 26, 30, 36, 60] if near else [-22, 24, 28, 34, 60],
        [-56, -48, 18, 50, 58],
        [-60, -32, -20, 44, 56],
    ]
    assigned: list[list[int] | None] = [None] * 4
    assigned[winner] = patterns[0]
    assigned[runner] = patterns[1]
    leftovers = [patterns[2][:], patterns[3][:]]
    rng.shuffle(leftovers)
    for panel in range(4):
        if assigned[panel] is None:
            assigned[panel] = leftovers.pop()

    shifts: list[list[int]] = []
    blue_values: list[list[int]] = []
    for panel_pattern in assigned:
        values = list(panel_pattern or [])
        rng.shuffle(values)
        shifts.append(values)
        bases: list[int] = []
        for shift in values:
            low = max(20, 20 - shift)
            high = min(160, 160 - shift)
            bases.append(rng.randint(low, high))
        blue_values.append(bases)

    return {
        "shifts": shifts,
        "blue_values": blue_values,
        "palette": seed % len(PALETTES),
        "offset": [rng.randint(-7, 7), rng.randint(-5, 5)],
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = _medians(scene["shifts"])
    return "ABCD"[max(range(4), key=lambda index: scores[index])]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(_medians(scene["shifts"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _layout(panel: int, offset: list[int]) -> tuple[int, int, int, int, int, list[int]]:
    row, col = divmod(panel, 2)
    ox, oy = offset
    panel_left = 43 + 447 * col + ox
    panel_top = 54 + 276 * row + oy
    plot_left = panel_left + 64
    plot_right = panel_left + 399
    plot_top = panel_top + 34
    plot_bottom = panel_top + 214
    x_values = [plot_left + 37 + 65 * index for index in range(5)]
    return panel_left, panel_top, plot_top, plot_bottom, plot_right, x_values


def render(scene: dict[str, Any]) -> Image.Image:
    shifts = scene["shifts"]
    blue_values = scene["blue_values"]
    palette = PALETTES[int(scene["palette"])]
    offset = scene["offset"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "#ffffff")
    draw = ImageDraw.Draw(image)

    def line(points: list[tuple[int, int]], fill: str, width: int = 1) -> None:
        draw.line([(x * SCALE, y * SCALE) for x, y in points], fill=fill, width=width * SCALE)

    def text(x: int, y: int, value: str, *, size: int = 13, bold: bool = False, fill: str = "#252a31") -> None:
        draw.text((x * SCALE, y * SCALE), value, font=_font(size, bold), fill=fill)

    text(265, 8, "MEDIAN OF FIVE SIGNED PAIRED SHIFTS", size=16, bold=True, fill="#30343b")
    text(394, 31, "UPWARD = POSITIVE", size=10, bold=True, fill="#59616b")

    for panel in range(4):
        panel_left, panel_top, plot_top, plot_bottom, plot_right, x_values = _layout(panel, offset)
        plot_left = x_values[0] - 37
        text(panel_left + 5, panel_top + 2, "ABCD"[panel], size=18, bold=True)
        for tick in (0, 45, 90, 135, 180):
            y = plot_bottom - tick
            line([(plot_left, y), (plot_right, y)], "#e3e6e9", 1)
        line([(plot_left, plot_top), (plot_left, plot_bottom), (plot_right, plot_bottom)], "#69717a", 2)
        line([(plot_left, plot_top), (plot_right, plot_top), (plot_right, plot_bottom)], "#b8bec5", 1)

        for checkpoint, x in enumerate(x_values):
            line([(x, plot_top), (x, plot_bottom)], "#f0f1f3", 1)
            text(x - 4, plot_bottom + 8, str(checkpoint + 1), size=10, fill="#5e6670")
            blue_value = int(blue_values[panel][checkpoint])
            orange_value = blue_value + int(shifts[panel][checkpoint])
            blue_y = plot_bottom - blue_value
            orange_y = plot_bottom - orange_value
            line([(x, blue_y), (x, orange_y)], "#aab0b7", 2)
            for y, color in ((blue_y, palette["blue"]), (orange_y, palette["orange"])):
                radius = 7
                draw.ellipse(
                    ((x - radius) * SCALE, (y - radius) * SCALE, (x + radius) * SCALE, (y + radius) * SCALE),
                    fill=color,
                    outline=color,
                    width=SCALE,
                )
        text(plot_left + 118, plot_bottom + 27, "CHECKPOINT", size=10, bold=True, fill="#59616b")

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
