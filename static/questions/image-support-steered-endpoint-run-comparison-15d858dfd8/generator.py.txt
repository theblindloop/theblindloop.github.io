"""Deterministic renderer for endpoint-adjacent run comparison."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 384


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1000003 + 9176)
    # Swapping a and d preserves total ink and both endpoint positions.
    a, d = ((48, 24) if seed % 2 == 0 else (24, 48))
    # A thin 72x10 hairpin keeps decisive support local and path-shaped.
    base = [(0, 0), (a, 0), (a, 10), (72, 10)]
    rotation = rng.randrange(4)
    transformed = []
    for x, y in base:
        if rotation == 0:
            u, v = x, y
        elif rotation == 1:
            u, v = -y, x
        elif rotation == 2:
            u, v = -x, -y
        else:
            u, v = y, -x
        transformed.append((u, v))
    min_x = min(x for x, _ in transformed)
    max_x = max(x for x, _ in transformed)
    min_y = min(y for _, y in transformed)
    max_y = max(y for _, y in transformed)
    centers = (72, 120, 168, 216, 264, 312)
    if (max_x - min_x) > (max_y - min_y):
        left = rng.randint(72, SIZE - 72 - (max_x - min_x))
        top = rng.choice(centers) - (max_y - min_y) // 2
    else:
        left = rng.choice(centers) - (max_x - min_x) // 2
        top = rng.randint(72, SIZE - 72 - (max_y - min_y))
    points = [[x - min_x + left, y - min_y + top] for x, y in transformed]
    return {
        "points": points,
        "start_color": [35, 170, 92],
        "finish_color": [220, 58, 72],
        "stroke_color": [31, 58, 88],
        "background_color": [248, 246, 239],
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), tuple(scene["background_color"]))
    draw = ImageDraw.Draw(image)
    points = [tuple(p) for p in scene["points"]]
    draw.line(points, fill=tuple(scene["stroke_color"]), width=5, joint="curve")
    radius = 6
    for point, color in ((points[0], scene["start_color"]), (points[-1], scene["finish_color"])):
        x, y = point
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=tuple(color))
    return image


def _run_lengths(scene: dict) -> tuple[float, float]:
    points = scene["points"]
    first = abs(points[1][0] - points[0][0]) + abs(points[1][1] - points[0][1])
    final = abs(points[-1][0] - points[-2][0]) + abs(points[-1][1] - points[-2][1])
    return float(first), float(final)


def analytic_gold(scene: dict) -> str:
    first, final = _run_lengths(scene)
    return "first" if first > final else "final"


def margin(scene: dict) -> float:
    first, final = _run_lengths(scene)
    return abs(first - final)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
