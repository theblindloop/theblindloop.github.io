"""Pixel-only inverse arm; it has no access to renderer or latent state."""
from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _near(array: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 48.0) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _centroid(mask: np.ndarray) -> tuple[int, int]:
    pts = np.argwhere(mask)
    if len(pts) < 40:
        raise ValueError("endpoint marker missing")
    y, x = np.rint(pts.mean(axis=0)).astype(int)
    return int(x), int(y)


def _connected(mask: np.ndarray, start: tuple[int, int], finish: tuple[int, int]) -> bool:
    sx, sy = start
    fx, fy = finish
    if not mask[sy, sx] or not mask[fy, fx]:
        return False
    seen = np.zeros(mask.shape, dtype=bool)
    queue = deque([(sy, sx)])
    seen[sy, sx] = True
    while queue:
        y, x = queue.popleft()
        if abs(x - fx) <= 1 and abs(y - fy) <= 1:
            return True
        for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
            if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] and mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return False


def _ray_length(navy: np.ndarray, center: tuple[int, int]) -> float:
    x, y = center
    height, width = navy.shape
    candidates: list[tuple[int, int, int]] = []
    for dx, dy in ((1,0),(-1,0),(0,1),(0,-1)):
        evidence = 0
        for step in range(9, 24):
            xx, yy = x + dx * step, y + dy * step
            if not (2 <= xx < width-2 and 2 <= yy < height-2):
                continue
            if dx:
                evidence += int(navy[yy-2:yy+3, xx].sum())
            else:
                evidence += int(navy[yy, xx-2:xx+3].sum())
        candidates.append((evidence, dx, dy))
    evidence, dx, dy = max(candidates)
    if evidence < 45:
        raise ValueError("endpoint direction is ambiguous")
    last = 0
    misses = 0
    for step in range(9, 120):
        xx, yy = x + dx * step, y + dy * step
        if not (2 <= xx < width-2 and 2 <= yy < height-2):
            break
        occupied = bool(navy[yy-2:yy+3, xx].any()) if dx else bool(navy[yy, xx-2:xx+3].any())
        if occupied:
            last = step
            misses = 0
        else:
            misses += 1
            if misses >= 4:
                break
    if last < 24:
        raise ValueError("endpoint run too short")
    return float(last)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    navy = _near(array, (31, 58, 88), 32.0)
    green = _near(array, (35, 170, 92), 38.0)
    red = _near(array, (220, 58, 72), 38.0)
    start = _centroid(green)
    finish = _centroid(red)
    traversable = navy | green | red
    if not _connected(traversable, start, finish):
        raise ValueError("route is not continuously traceable")
    first = _ray_length(navy, start)
    final = _ray_length(navy, finish)
    if abs(first - final) < 12.0:
        raise ValueError("run comparison is within quarantine")
    return "first" if first > final else "final"
