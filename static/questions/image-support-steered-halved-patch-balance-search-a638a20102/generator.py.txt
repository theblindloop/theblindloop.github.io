"""Renderer for the halved-patch area-balance search world.

Four solid cell-built patches sit in one tight two-by-two cluster on an
otherwise empty page. A short coloured bar cuts each patch into two parts.
Every patch is built from the same number of unit cells, so all four hold the
same total dark area; only the way that area is divided by the bar differs.
Exactly one patch is cut into two parts of equal dark area.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "halved_patch_balance_search-renderer-0.1.0"

CELL = 10                 # side of one unit cell in pixels
GRID_C = 7                # cell columns available to one patch
GRID_R = 6                # cell rows available to one patch
AREA_CELLS = 20           # every patch is built from exactly this many cells
GAP = 4                   # width of the cut bar, inserted between the parts
SLOT_PITCH_X = 92         # centre-to-centre spacing of the two cluster columns
SLOT_PITCH_Y = 82         # centre-to-centre spacing of the two cluster rows
BOX_W = GRID_C * CELL + GAP
BOX_H = GRID_R * CELL + GAP
CLUSTER_W = SLOT_PITCH_X + BOX_W
CLUSTER_H = SLOT_PITCH_Y + BOX_H
JITTER = 3                # per-patch placement jitter inside its cluster slot
SPECK = 3                 # side of one background speck
CLEARANCE = 16            # specks stay this far outside the cluster box
CANVAS_W = 768
CANVAS_H = 576
MARGIN = 22               # keeps the cluster away from the page edge

WORDS = ("upper-left", "upper-right", "lower-left", "lower-right")
QUARANTINE_MARGIN = 4 * CELL * CELL

PALETTES = (
    ((246, 244, 238), (38, 42, 50), (188, 72, 56)),
    ((238, 242, 246), (44, 38, 46), (36, 108, 172)),
    ((243, 240, 232), (34, 48, 42), (152, 96, 24)),
    ((240, 240, 244), (48, 40, 40), (28, 118, 108)),
)


def _grow(region: set, seed_cell: tuple, count: int, rng: random.Random) -> list:
    """Grow a compactness-biased connected cell blob inside ``region``."""

    blob = {seed_cell}
    while len(blob) < count:
        frontier: dict = {}
        for (cc, rr) in blob:
            for dc, dr in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                cand = (cc + dc, rr + dr)
                if cand in blob or cand not in region:
                    continue
                touching = sum(
                    1
                    for ec, er in ((1, 0), (-1, 0), (0, 1), (0, -1))
                    if (cand[0] + ec, cand[1] + er) in blob
                )
                frontier[cand] = 1 + 3 * touching
        if not frontier:
            return []
        cells = sorted(frontier)
        weights = [frontier[cell] for cell in cells]
        blob.add(rng.choices(cells, weights=weights, k=1)[0])
    return sorted(blob)


def _build_patch(rng: random.Random, slot: int, n_a: int, n_b: int) -> dict | None:
    """One patch whose part before the bar holds ``n_a`` cells and after it ``n_b``."""

    options = []
    for cut in range(2, GRID_C - 1):
        if n_a <= GRID_R * cut and n_b <= GRID_R * (GRID_C - cut):
            options.append(("vertical", cut))
    for cut in range(2, GRID_R - 1):
        if n_a <= GRID_C * cut and n_b <= GRID_C * (GRID_R - cut):
            options.append(("horizontal", cut))
    if not options:
        return None
    orientation, cut = rng.choice(options)

    if orientation == "vertical":
        region_a = {(c, r) for c in range(cut) for r in range(GRID_R)}
        region_b = {(c, r) for c in range(cut, GRID_C) for r in range(GRID_R)}
        touch = rng.randrange(GRID_R)
        seed_a, seed_b = (cut - 1, touch), (cut, touch)
    else:
        region_a = {(c, r) for c in range(GRID_C) for r in range(cut)}
        region_b = {(c, r) for c in range(GRID_C) for r in range(cut, GRID_R)}
        touch = rng.randrange(GRID_C)
        seed_a, seed_b = (touch, cut - 1), (touch, cut)

    cells_a = _grow(region_a, seed_a, n_a, rng)
    cells_b = _grow(region_b, seed_b, n_b, rng)
    if not cells_a or not cells_b:
        return None
    return {
        "slot": slot,
        "orientation": orientation,
        "cut": cut,
        "cells_a": [list(cell) for cell in cells_a],
        "cells_b": [list(cell) for cell in cells_b],
        "jx": rng.randint(-JITTER, JITTER),
        "jy": rng.randint(-JITTER, JITTER),
    }


def patch_geometry(scene: dict, patch: dict) -> dict:
    """Pixel geometry of one patch: its cell rectangles and its cut bar."""

    slot = int(patch["slot"])
    row, col = divmod(slot, 2)
    px = scene["origin_x"] + col * SLOT_PITCH_X + int(patch["jx"])
    py = scene["origin_y"] + row * SLOT_PITCH_Y + int(patch["jy"])
    cut = int(patch["cut"])
    rects = []
    if patch["orientation"] == "vertical":
        for cc, rr in patch["cells_a"]:
            rects.append((px + cc * CELL, py + rr * CELL))
        for cc, rr in patch["cells_b"]:
            rects.append((px + cc * CELL + GAP, py + rr * CELL))
    else:
        for cc, rr in patch["cells_a"]:
            rects.append((px + cc * CELL, py + rr * CELL))
        for cc, rr in patch["cells_b"]:
            rects.append((px + cc * CELL, py + rr * CELL + GAP))
    x0 = min(x for x, _ in rects)
    x1 = max(x for x, _ in rects) + CELL
    y0 = min(y for _, y in rects)
    y1 = max(y for _, y in rects) + CELL
    if patch["orientation"] == "vertical":
        bar = (px + cut * CELL, y0 - 4, px + cut * CELL + GAP, y1 + 4)
    else:
        bar = (x0 - 4, py + cut * CELL, x1 + 4, py + cut * CELL + GAP)
    return {"rects": rects, "bar": bar, "ink_box": (x0, y0, x1, y1)}


def _extent_offset(scene: dict, patch: dict) -> float:
    """How far the bar sits from the middle of the patch outline, as a fraction."""

    geom = patch_geometry(scene, patch)
    x0, y0, x1, y1 = geom["ink_box"]
    bx0, by0, bx1, by1 = geom["bar"]
    if patch["orientation"] == "vertical":
        span = x1 - x0
        centre = 0.5 * (bx0 + bx1) - x0
    else:
        span = y1 - y0
        centre = 0.5 * (by0 + by1) - y0
    return abs(centre / span - 0.5)


def _sample_patches(rng: random.Random, winner: int) -> list | None:
    imbalance_floor = rng.choice([2, 4, 4, 6])
    choices = [value for value in (2, 4, 6, 8) if value >= imbalance_floor]
    imbalances = [imbalance_floor] + [rng.choice(choices) for _ in range(2)]
    rng.shuffle(imbalances)
    patches = []
    queue = list(imbalances)
    for slot in range(4):
        if slot == winner:
            n_a = n_b = AREA_CELLS // 2
        else:
            gap = queue.pop()
            n_a = (AREA_CELLS - gap) // 2
            n_b = AREA_CELLS - n_a
            if rng.random() < 0.5:
                n_a, n_b = n_b, n_a
        patch = _build_patch(rng, slot, n_a, n_b)
        if patch is None:
            return None
        patches.append(patch)
    return patches


def sample_scene(seed: int) -> dict:
    """Deterministic scene: one two-by-two cluster of cut patches plus specks."""

    rng = random.Random(seed * 7919 + 13)
    winner = seed % 4
    scene = {
        "width": CANVAS_W,
        "height": CANVAS_H,
        "palette": rng.randrange(len(PALETTES)),
        "origin_x": rng.randint(MARGIN, CANVAS_W - CLUSTER_W - MARGIN),
        "origin_y": rng.randint(MARGIN, CANVAS_H - CLUSTER_H - MARGIN),
        "patches": [],
        "specks": [],
    }
    for _ in range(400):
        patches = _sample_patches(rng, winner)
        if patches is None:
            continue
        scene["patches"] = patches
        offsets = [_extent_offset(scene, patch) for patch in patches]
        # The naive "the bar looks centred in the outline" reading must be wrong.
        if min(range(4), key=lambda i: offsets[i]) == winner:
            continue
        if offsets[winner] < 0.03:
            continue
        break
    else:  # pragma: no cover - the loop succeeds well inside 400 attempts
        raise RuntimeError("failed to sample a counterbalanced scene")

    keep = (
        scene["origin_x"] - CLEARANCE,
        scene["origin_y"] - CLEARANCE,
        scene["origin_x"] + CLUSTER_W + CLEARANCE,
        scene["origin_y"] + CLUSTER_H + CLEARANCE,
    )
    specks: list = []
    while len(specks) < rng.randint(6, 14):
        sx = rng.randint(8, CANVAS_W - 8 - SPECK)
        sy = rng.randint(8, CANVAS_H - 8 - SPECK)
        if keep[0] <= sx + SPECK and sx <= keep[2] and keep[1] <= sy + SPECK and sy <= keep[3]:
            continue
        if any(abs(sx - ox) < 3 * SPECK and abs(sy - oy) < 3 * SPECK for ox, oy in specks):
            continue
        specks.append([sx, sy])
    scene["specks"] = specks
    return scene


def part_areas(patch: dict) -> tuple:
    """Dark pixel area of the part before and after the bar."""

    unit = CELL * CELL
    return len(patch["cells_a"]) * unit, len(patch["cells_b"]) * unit


def analytic_gold(scene: dict) -> str:
    """Cluster position of the only patch cut into two equal dark areas."""

    balanced = [
        patch for patch in scene["patches"] if len(set(map(len, (patch["cells_a"], patch["cells_b"])))) == 1
    ]
    if len(balanced) != 1:
        raise ValueError("scene does not have a unique balanced patch")
    return WORDS[int(balanced[0]["slot"])]


def margin(scene: dict) -> float:
    """Smallest competing area imbalance, in dark pixels."""

    gaps = [abs(a - b) for a, b in (part_areas(patch) for patch in scene["patches"])]
    competing = [gap for gap in gaps if gap > 0]
    if len(competing) != len(gaps) - 1:
        raise ValueError("scene does not have exactly one balanced patch")
    return float(min(competing))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes where a second patch is nearly balanced too."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical, answer-preserving rewrites of the latent scene."""

    out = []
    if len(scene["specks"]) > 1:
        flipped = dict(scene)
        flipped["specks"] = list(reversed(scene["specks"]))
        out.append(("speck_order_reversed", flipped))
    if len(scene["patches"]) > 1:
        rotated = dict(scene)
        rotated["patches"] = scene["patches"][1:] + scene["patches"][:1]
        out.append(("patch_record_order_rotated", rotated))
    return out


def render(scene: dict) -> Image.Image:
    """Rasterise the scene. Drawing order never overlaps two marks."""

    background, ink, divider = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), background)
    draw = ImageDraw.Draw(image)
    for sx, sy in scene["specks"]:
        draw.rectangle((sx, sy, sx + SPECK - 1, sy + SPECK - 1), fill=ink)
    for patch in scene["patches"]:
        geom = patch_geometry(scene, patch)
        for x, y in geom["rects"]:
            draw.rectangle((x, y, x + CELL - 1, y + CELL - 1), fill=ink)
        bx0, by0, bx1, by1 = geom["bar"]
        draw.rectangle((bx0, by0, bx1 - 1, by1 - 1), fill=divider)
    return image
