"""Independent image-only inverse for additive span balance.

This module deliberately duplicates the small amount of layout knowledge needed
to locate the four panels. It never imports renderer, prompts, or scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 760
PANEL_BOXES = (
    (35, 75, 685, 345),
    (715, 75, 1365, 345),
    (35, 415, 685, 685),
    (715, 415, 1365, 685),
)
BAR_COLORS = (
    (208, 60, 60),
    (49, 101, 194),
    (220, 162, 40),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
BALANCE_TOLERANCE = 6
MIN_BAR_PIXELS = 500


def _width_of_color(arr: np.ndarray, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> int:
    x0, y0, x1, y1 = box
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < MIN_BAR_PIXELS:
        raise ValueError("colored span is missing or too small")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if height < 25 or width < 50:
        raise ValueError("colored span geometry is unreadable")
    # A single solid bar has one filled horizontal row extent for every color.
    row_counts = mask.sum(axis=1)
    if int(np.count_nonzero(row_counts >= 25)) < 25:
        raise ValueError("colored span is not a solid bar")
    return width


def decision_from_image(image: Image.Image) -> str:
    """Measure each panel's three colored bounding boxes and solve the equation."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb, dtype=np.uint8)
    residuals: list[int] = []
    for box in PANEL_BOXES:
        red, blue, gold = (
            _width_of_color(arr, box, color) for color in BAR_COLORS
        )
        residuals.append(red + blue - gold)
    matches = [index for index, residual in enumerate(residuals) if abs(residual) <= BALANCE_TOLERANCE]
    if len(matches) != 1:
        raise ValueError(f"expected one additive match, found {matches}")
    return PANEL_LABELS[matches[0]]
