"""Renderer and analytic arm for a three-span additive relation.

The answer is carried by a small arithmetic program over visible bar widths:
for each panel, red_width + blue_width - gold_width.  The PNG uses exact RGB
rectangles so an independent inverse can recover those widths without scene
metadata.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 760

BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
FRAME = (102, 116, 134)
INK = (34, 43, 56)
GUIDE = (218, 225, 233)
RED = (208, 60, 60)
BLUE = (49, 101, 194)
GOLD = (220, 162, 40)

PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
PANEL_BOXES = (
    (35, 75, 685, 345),
    (715, 75, 1365, 345),
    (35, 415, 685, 685),
    (715, 415, 1365, 685),
)
BAR_X_OFFSET = 148
BAR_Y_OFFSETS = (102, 162, 222)
BAR_HEIGHT = 30
BAR_CAPACITY = 480
BALANCE_TOLERANCE = 6.0
TARGET_HEADROOM = 2.0
DECOY_CLEARANCE = 18.0

_BAR_NAMES = ("red", "blue", "gold")
_BAR_COLORS = {"red": RED, "blue": BLUE, "gold": GOLD}


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    """Sample four large-bar panels with exactly one additive match."""
    rng = random.Random(int(seed))
    target_index = rng.randrange(len(PANEL_LABELS))
    panels: list[dict[str, int]] = []
    for index in range(len(PANEL_LABELS)):
        red = rng.randint(84, 150)
        blue = rng.randint(82, 148)
        base = red + blue
        if index == target_index:
            gold = base
        else:
            # Both signs occur among the three decoys.  The smallest deviation
            # is still well outside the pixel-resolution quarantine band.
            sign = -1 if ((index + target_index) % 2 == 0) else 1
            deviation = rng.randint(28, 76)
            gold = base + sign * deviation
        if not 70 <= gold <= BAR_CAPACITY:
            raise AssertionError("sampled bar escaped the panel capacity")
        panels.append({"red": red, "blue": blue, "gold": gold})
    return {"panels": panels}


def _residuals(panels: Iterable[dict]) -> list[float]:
    return [
        float(panel["red"] + panel["blue"] - panel["gold"])
        for panel in panels
    ]


def _matches(panels: Iterable[dict]) -> list[int]:
    return [
        index
        for index, residual in enumerate(_residuals(panels))
        if abs(residual) <= BALANCE_TOLERANCE
    ]


def render(scene: dict) -> Image.Image:
    """Render four framed panels with common-baseline colored spans."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27)
    panel_font = _font(28)
    row_font = _font(21)
    draw.text((40, 18), "Add the red and blue spans; compare with gold", fill=INK, font=title_font)
    draw.text((1070, 25), "A    B    C    D", fill=FRAME, font=row_font)

    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 28, y0 + 22), PANEL_LABELS[index][-1], fill=INK, font=panel_font)
        bar_x = x0 + BAR_X_OFFSET
        for name, y_offset in zip(_BAR_NAMES, BAR_Y_OFFSETS):
            y = y0 + y_offset
            draw.line((bar_x, y + BAR_HEIGHT + 12, bar_x + BAR_CAPACITY, y + BAR_HEIGHT + 12), fill=GUIDE, width=2)
            draw.text((x0 + 50, y + 2), name[0].upper(), fill=INK, font=row_font)
            width = int(panel[name])
            draw.rectangle(
                (bar_x, y, bar_x + width - 1, y + BAR_HEIGHT - 1),
                fill=_BAR_COLORS[name],
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique panel whose gold span balances two addends."""
    panels = scene["panels"]
    matches = _matches(panels)
    if len(matches) != 1:
        raise ValueError(f"expected one additive match, found {matches}")
    return PANEL_LABELS[matches[0]]


def margin(scene: dict) -> float:
    """Return the smallest signed clearance from the declared balance bands."""
    residuals = [abs(value) for value in _residuals(scene["panels"])]
    if len(residuals) != len(PANEL_LABELS):
        return -1.0
    ordered = sorted(residuals)
    return min(
        BALANCE_TOLERANCE - ordered[0],
        ordered[1] - BALANCE_TOLERANCE,
    )


def is_quarantined(scene: dict) -> bool:
    """Exclude scenes without one exact target and a separated runner-up."""
    try:
        residuals = [abs(value) for value in _residuals(scene["panels"])]
    except (KeyError, TypeError, ValueError):
        return True
    if len(residuals) != len(PANEL_LABELS):
        return True
    matches = [value for value in residuals if value <= BALANCE_TOLERANCE]
    if len(matches) != 1:
        return True
    target = matches[0]
    decoys = [value for value in residuals if value != target]
    return target > TARGET_HEADROOM or not decoys or min(decoys) < DECOY_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden interchangeable fields in this panel encoding."""
    return []
