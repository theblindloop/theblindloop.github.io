"""Deterministic latent sampler, raster renderer, and analytic graph answer."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = "#ffffff"
UNDERLAY = "#6b7280"
CENTERLINE = "#111827"
GREEN = "#16a34a"
BLUE = "#2563eb"
COLORS = {"orange": "#f97316", "purple": "#8b5cf6", "red": "#ef4444"}

_NODES = {
    "r": (50, 256),
    "a": (140, 256),
    "b": (250, 256),
    "c": (360, 256),
    "ref": (462, 256),
    "l0": (430, 70),
    "l1": (60, 442),
    "l2": (220, 150),
}

_EDGES = [
    ("r", "a", [(50, 256), (140, 256)]),
    ("a", "b", [(140, 256), (250, 256)]),
    ("b", "c", [(250, 256), (360, 256)]),
    ("c", "ref", [(360, 256), (462, 256)]),
    ("a", "l0", [(140, 256), (140, 70), (430, 70)]),
    ("b", "l1", [(250, 256), (250, 442), (60, 442)]),
    ("c", "l2", [(360, 256), (360, 150), (220, 150)]),
]


def _transform_point(point: tuple[int, int], code: int) -> list[int]:
    x, y = point
    x -= 256
    y -= 256
    if code >= 4:
        x = -x
    for _ in range(code % 4):
        x, y = -y, x
    return [x + 256, y + 256]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = ("orange", "purple", "red")[int(seed) % 3]
    remaining = [color for color in COLORS if color != winner]
    rng.shuffle(remaining)
    assignment = {"l0": remaining[0], "l1": remaining[1], "l2": winner}
    code = int(seed) % 8
    nodes = {name: _transform_point(point, code) for name, point in _NODES.items()}
    edges = [
        {"a": a, "b": b, "points": [_transform_point(point, code) for point in points]}
        for a, b, points in _EDGES
    ]
    return {
        "canvas": [CANVAS, CANVAS],
        "nodes": nodes,
        "edges": edges,
        "root_node": "r",
        "reference_node": "ref",
        "candidate_nodes": ["l0", "l1", "l2"],
        "candidate_colors": assignment,
    }


def _edge_length(nodes: dict, edge: dict) -> float:
    points = edge["points"]
    if list(nodes[edge["a"]]) != list(points[0]) or list(nodes[edge["b"]]) != list(points[-1]):
        raise ValueError("edge endpoints disagree with nodes")
    return sum(
        math.hypot(q[0] - p[0], q[1] - p[1]) for p, q in zip(points, points[1:])
    )


def _node_path(scene: dict, target: str) -> list[tuple[str, str, float]]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    root = scene["root_node"]
    adjacency: dict[str, list[tuple[str, float]]] = {name: [] for name in nodes}
    for edge in edges:
        length = _edge_length(nodes, edge)
        a, b = edge["a"], edge["b"]
        adjacency[a].append((b, length))
        adjacency[b].append((a, length))
    pending = deque([(root, [])])
    seen = {root}
    while pending:
        node, path = pending.popleft()
        if node == target:
            return path
        for nxt, length in adjacency[node]:
            if nxt not in seen:
                seen.add(nxt)
                pending.append((nxt, path + [(node, nxt, length)]))
    raise ValueError(f"unreachable node {target}")


def _shared_lengths(scene: dict) -> dict[str, float]:
    reference = scene["reference_node"]
    candidates = scene["candidate_nodes"]
    colors = scene["candidate_colors"]
    ref_path = _node_path(scene, reference)
    ref_keys = {frozenset((a, b)): length for a, b, length in ref_path}
    values: dict[str, float] = {}
    for candidate in candidates:
        path = _node_path(scene, candidate)
        shared = sum(length for a, b, length in path if frozenset((a, b)) in ref_keys)
        values[colors[candidate]] = shared
    return values


def analytic_gold(scene: dict) -> str:
    values = _shared_lengths(scene)
    return max(values, key=values.get)


def margin(scene: dict) -> float:
    values = sorted(_shared_lengths(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_square(draw: ImageDraw.ImageDraw, center: list[int], fill: str) -> None:
    x, y = center
    radius = 13
    draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=fill, outline="#ffffff", width=2)


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    nodes = scene["nodes"]
    edges = scene["edges"]
    root = scene["root_node"]
    reference = scene["reference_node"]
    candidates = scene["candidate_nodes"]
    colors = scene["candidate_colors"]
    image = Image.new("RGB", (int(width), int(height)), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for edge in edges:
        draw.line([tuple(point) for point in edge["points"]], fill=UNDERLAY, width=9, joint="curve")
    for edge in edges:
        draw.line([tuple(point) for point in edge["points"]], fill=CENTERLINE, width=1)
    rx, ry = nodes[root]
    draw.ellipse((rx - 15, ry - 15, rx + 15, ry + 15), fill=BACKGROUND, outline=GREEN, width=6)
    bx, by = nodes[reference]
    diamond = [(bx, by - 16), (bx + 16, by), (bx, by + 16), (bx - 16, by)]
    draw.polygon(diamond, fill=BLUE, outline="#ffffff")
    for candidate in candidates:
        _draw_square(draw, nodes[candidate], COLORS[colors[candidate]])
    return image
