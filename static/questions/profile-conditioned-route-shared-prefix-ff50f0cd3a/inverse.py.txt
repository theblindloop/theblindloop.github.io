"""Independent image-only inverse arm for shared route prefixes."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "green": (22, 163, 74),
    "blue": (37, 99, 235),
    "orange": (249, 115, 22),
    "purple": (139, 92, 246),
    "red": (239, 68, 68),
}
_ANSWERS = ("orange", "purple", "red")


def _centroid(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[int, int]:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    mask = np.sum(delta * delta, axis=2) <= 25 * 25
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("required colored marker is missing")
    return int(round(float(xs.mean()))), int(round(float(ys.mean())))


def _nearest(mask: np.ndarray, point: tuple[int, int]) -> tuple[int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("dark centerline is missing")
    px, py = point
    index = int(np.argmin((xs - px) ** 2 + (ys - py) ** 2))
    return int(xs[index]), int(ys[index])


def _path(mask: np.ndarray, start: tuple[int, int], goal: tuple[int, int]) -> list[tuple[int, int]]:
    width = mask.shape[1]
    start_i = start[1] * width + start[0]
    goal_i = goal[1] * width + goal[0]
    queue = deque([start_i])
    parent = {start_i: -1}
    while queue:
        current = queue.popleft()
        if current == goal_i:
            break
        y, x = divmod(current, width)
        for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if 0 <= ny < mask.shape[0] and 0 <= nx < width and mask[ny, nx]:
                nxt = ny * width + nx
                if nxt not in parent:
                    parent[nxt] = current
                    queue.append(nxt)
    if goal_i not in parent:
        raise ValueError("marker is not connected to the dark centerline")
    result = []
    current = goal_i
    while current != -1:
        y, x = divmod(current, width)
        result.append((x, y))
        current = parent[current]
    result.reverse()
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    marker = {name: _centroid(arr, rgb) for name, rgb in _PALETTE.items()}
    dark = np.max(arr, axis=2) < 45
    anchors = {name: _nearest(dark, point) for name, point in marker.items()}
    reference_path = set(_path(dark, anchors["green"], anchors["blue"]))
    scores = {}
    for color in _ANSWERS:
        candidate_path = _path(dark, anchors["green"], anchors[color])
        scores[color] = sum(pixel in reference_path for pixel in candidate_path)
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 20:
        raise ValueError("ambiguous shared-prefix margin")
    return ordered[0][0]
