"""Deterministic latent sampler and renderer for cable-midpoint tracing."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
BACKGROUND = (250, 249, 245)
CABLE = (24, 50, 74)
GREEN = (31, 157, 105)
ORANGE = (235, 126, 34)
PURPLE = (132, 80, 190)
END_RED = (207, 59, 73)
PALE = (194, 199, 202)

# A long, non-self-intersecting orthogonal cable. Its folds make Euclidean
# midpoint and along-cable midpoint materially different.
BASE_ROUTE = [
    [70, 80], [430, 80], [430, 180], [120, 180],
    [120, 280], [400, 280], [400, 390], [70, 390],
]
BASE_DISTRACTORS = [
    [[230, 127], [335, 127]],
    [[205, 230], [310, 230]],
    [[190, 340], [285, 340]],
]

# Each pair has one vertex substantially closer than the other to half of the
# total arclength. Across pairs every internal region can be decisive.
CHECKPOINT_PAIRS = ((2, 4), (3, 5), (1, 5), (2, 6))


def _pose(point: list[int], code: int) -> list[int]:
    """Apply one of the eight rigid square symmetries."""
    x, y = point
    if code >= 4:
        x = SIZE - x
    for _ in range(code % 4):
        x, y = SIZE - y, x
    return [int(x), int(y)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    pose = rng.randrange(8)
    route = [_pose(point, pose) for point in BASE_ROUTE]
    distractors = [[_pose(point, pose) for point in line] for line in BASE_DISTRACTORS]
    rng.shuffle(distractors)
    first, second = CHECKPOINT_PAIRS[int(seed) % len(CHECKPOINT_PAIRS)]
    # Palette identity flips every complete four-pair cycle, independently of
    # the rigid pose. Every public 24-scene batch therefore has 12 per answer.
    if (int(seed) // len(CHECKPOINT_PAIRS)) % 2 == 0:
        orange_index, purple_index = first, second
    else:
        orange_index, purple_index = second, first
    return {
        "route_points": route,
        "start_index": 0,
        "end_index": len(route) - 1,
        "orange_index": orange_index,
        "purple_index": purple_index,
        "transform": pose,
        "distractors": distractors,
        "stroke_width": 10 + 2 * rng.randrange(2),
    }


def _arc_distance(points: list[list[int]], start: int, target: int) -> float:
    lo, hi = sorted((int(start), int(target)))
    return sum(
        math.hypot(points[i + 1][0] - points[i][0], points[i + 1][1] - points[i][1])
        for i in range(lo, hi)
    )


def _midpoint_offsets(scene: dict) -> dict[str, float]:
    points = scene["route_points"]
    start, end = int(scene["start_index"]), int(scene["end_index"])
    half = _arc_distance(points, start, end) / 2.0
    return {
        "orange": abs(_arc_distance(points, start, int(scene["orange_index"])) - half),
        "purple": abs(_arc_distance(points, start, int(scene["purple_index"])) - half),
    }


def analytic_gold(scene: dict) -> str:
    offsets = _midpoint_offsets(scene)
    return "orange" if offsets["orange"] < offsets["purple"] else "purple"


def margin(scene: dict) -> float:
    offsets = _midpoint_offsets(scene)
    return float(abs(offsets["orange"] - offsets["purple"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 35.0


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def pts(values: list[list[int]]) -> list[tuple[int, int]]:
        return [(int(x * scale), int(y * scale)) for x, y in values]

    for fragment in scene["distractors"]:
        draw.line(pts(fragment), fill=PALE, width=6 * scale)

    route = scene["route_points"]
    width = int(scene["stroke_width"]) * scale
    # Sort primitives so route-parameter reversal is pixel-identical.
    segments = sorted(
        (list(a), list(b)) if tuple(a) <= tuple(b) else (list(b), list(a))
        for a, b in zip(route, route[1:])
    )
    for first, second in segments:
        draw.line(pts([first, second]), fill=CABLE, width=width)
    radius = width // 2
    for x, y in pts(sorted(route)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=CABLE)

    marker_radius = 15 * scale
    for index, color in ((scene["orange_index"], ORANGE), (scene["purple_index"], PURPLE)):
        x, y = pts([route[int(index)]])[0]
        draw.ellipse((x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius), fill=color)

    # Start and end rings expose the full interval whose midpoint is requested.
    for index, color in ((scene["start_index"], GREEN), (scene["end_index"], END_RED)):
        x, y = pts([route[int(index)]])[0]
        outer, inner = 18 * scale, 9 * scale
        draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=color)
        draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=CABLE)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse only the latent parameterization, preserving visible roles."""
    twin = deepcopy(scene)
    count = len(scene["route_points"])
    twin["route_points"] = list(reversed(scene["route_points"]))
    for key in ("start_index", "end_index", "orange_index", "purple_index"):
        twin[key] = count - 1 - int(scene[key])
    return [("reverse_route_parameterization", twin)]
