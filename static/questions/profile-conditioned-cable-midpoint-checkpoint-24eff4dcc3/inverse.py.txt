"""Independent PNG-only inverse arm for along-cable midpoint comparison."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "cable": np.array([24, 50, 74], dtype=np.int32),
    "green": np.array([31, 157, 105], dtype=np.int32),
    "orange": np.array([235, 126, 34], dtype=np.int32),
    "purple": np.array([132, 80, 190], dtype=np.int32),
    "end": np.array([207, 59, 73], dtype=np.int32),
}


def _near(pixels: np.ndarray, color: np.ndarray, radius: int = 45) -> np.ndarray:
    delta = pixels.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= radius * radius


def _distances(walkable: np.ndarray, sources: np.ndarray) -> np.ndarray:
    height, width = walkable.shape
    distance = np.full((height, width), -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    ys, xs = np.nonzero(sources)
    for y, x in zip(ys.tolist(), xs.tolist()):
        distance[y, x] = 0
        queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        nd = int(distance[y, x]) + 1
        for dy, dx in ((-1, 0), (0, -1), (0, 1), (1, 0)):
            yy, xx = y + dy, x + dx
            if 0 <= yy < height and 0 <= xx < width and walkable[yy, xx] and distance[yy, xx] < 0:
                distance[yy, xx] = nd
                queue.append((yy, xx))
    return distance


def _measurement(image: Image.Image) -> tuple[str, float]:
    pixels = np.asarray(image.convert("RGB"))
    masks = {name: _near(pixels, color) for name, color in _COLORS.items()}
    for name in ("green", "orange", "purple", "end"):
        if int(masks[name].sum()) < 80:
            raise ValueError(f"missing {name} marker")

    walkable = np.logical_or.reduce(tuple(masks.values()))
    distance = _distances(walkable, masks["green"])
    endpoint = distance[masks["end"]]
    endpoint = endpoint[endpoint >= 0]
    if endpoint.size == 0:
        raise ValueError("red endpoint is disconnected from green start")
    total = float(endpoint.min())
    if total < 500:
        raise ValueError("cable is missing or implausibly short")
    half = total / 2.0

    offsets: dict[str, float] = {}
    for name in ("orange", "purple"):
        reached = distance[masks[name]]
        reached = reached[reached >= 0]
        if reached.size == 0:
            raise ValueError(f"{name} checkpoint is disconnected")
        offsets[name] = abs(float(reached.min()) - half)
    gap = abs(offsets["orange"] - offsets["purple"])
    if gap < 25.0:
        raise ValueError("ambiguous midpoint-offset comparison")
    return ("orange" if offsets["orange"] < offsets["purple"] else "purple"), float(gap)


def decision_from_image(image: Image.Image) -> str:
    return _measurement(image)[0]


def measured_margin(image: Image.Image) -> float:
    return _measurement(image)[1]
