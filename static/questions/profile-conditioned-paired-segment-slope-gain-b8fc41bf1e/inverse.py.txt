"""Independent raster-only inverse for paired segment slope gain."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Duplicated public palette and panel split; no renderer or latent data is read.
_PALETTE = {
    "blue": np.array((35, 105, 210), dtype=np.int32),
    "orange": np.array((235, 120, 20), dtype=np.int32),
    "green": np.array((20, 155, 95), dtype=np.int32),
}
_SPLIT = 360
_COLOR_DISTANCE = 58.0
_MIN_PIXELS = 450
_MIN_X_SPAN = 110
_ABSTAIN_GAIN_GAP = 0.095


def _fit_upward_slope(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if len(xs) < _MIN_PIXELS:
        raise ValueError("abstain: colored segment missing or too small")
    if int(xs.max()) - int(xs.min()) < _MIN_X_SPAN:
        raise ValueError("abstain: colored segment horizontal run is too short")
    x = xs.astype(np.float64)
    y = ys.astype(np.float64)
    centered = x - x.mean()
    denom = float(np.dot(centered, centered))
    if denom <= 0:
        raise ValueError("abstain: degenerate segment")
    raster_slope = float(np.dot(centered, y - y.mean()) / denom)
    return -raster_slope


def decision_from_image(image: Image.Image) -> str:
    # int32 prevents overflow when squaring channel differences.
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    h, w, _ = arr.shape
    if w < 600 or h < 350:
        raise ValueError("abstain: unexpected raster dimensions")
    gains: list[tuple[float, str]] = []
    for color, rgb in _PALETTE.items():
        dist = np.sqrt(np.sum((arr - rgb) ** 2, axis=2))
        mask = dist <= _COLOR_DISTANCE
        a = mask.copy()
        a[:, _SPLIT:] = False
        b = mask.copy()
        b[:, :_SPLIT] = False
        gains.append((_fit_upward_slope(b) - _fit_upward_slope(a), color))
    gains.sort(reverse=True)
    if gains[0][0] - gains[1][0] < _ABSTAIN_GAIN_GAP:
        raise ValueError("abstain: top two pixel-measured gains are too close")
    return gains[0][1]
