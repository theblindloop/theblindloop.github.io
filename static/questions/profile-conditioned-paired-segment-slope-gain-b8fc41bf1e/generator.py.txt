"""Deterministic paired-panel colored-segment world."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 460
PANEL_X = {"a": (54, 342), "b": (378, 666)}
PANEL_Y = (72, 418)
COLORS = {
    "blue": (35, 105, 210),
    "orange": (235, 120, 20),
    "green": (20, 155, 95),
}
QUARANTINE_GAP = 0.12


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, comfortably separated slope-gain comparison."""
    rng = random.Random(int(seed))
    names = list(COLORS)
    winner = names[int(seed) % len(names)]
    rows = [0, 1, 2]
    rng.shuffle(rows)
    series = []
    for color, row in zip(names, rows, strict=True):
        a_slope = rng.uniform(-0.16, 0.08)
        gain = rng.uniform(0.24, 0.31) if color == winner else rng.uniform(-0.14, 0.055)
        b_slope = a_slope + gain
        # Independent horizontal spans prevent raw endpoint displacement from
        # being a valid proxy for rise/run.
        a_run = rng.randint(142, 204)
        b_run = rng.randint(142, 204)
        a_x0 = rng.randint(78, 104)
        b_x0 = rng.randint(400, 426)
        row_center = (132, 245, 358)[row]
        series.append(
            {
                "color": color,
                "row": row,
                "a": _endpoints(a_x0, a_run, row_center, a_slope),
                "b": _endpoints(b_x0, b_run, row_center, b_slope),
            }
        )
    return {"series": series}


def _endpoints(x0: int, run: int, center_y: int, upward_slope: float) -> list[list[float]]:
    rise = upward_slope * run
    # Raster y decreases as the visually plotted value rises.
    return [[float(x0), center_y + rise / 2.0], [float(x0 + run), center_y - rise / 2.0]]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only information explicitly recoverable from the final pixels."""
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 248))
    draw = ImageDraw.Draw(image)

    def rect(box, **kwargs):
        draw.rectangle(tuple(int(v * scale) for v in box), **kwargs)

    for key in ("a", "b"):
        x0, x1 = PANEL_X[key]
        y0, y1 = PANEL_Y
        rect((x0, y0, x1, y1), fill=(255, 255, 255), outline=(45, 48, 52), width=2 * scale)
        # Pale row guides aid binding without participating in the oracle mask.
        for yc in (132, 245, 358):
            draw.line((x0 * scale, yc * scale, x1 * scale, yc * scale), fill=(225, 228, 230), width=1 * scale)
        label = "Panel A" if key == "a" else "Panel B"
        draw.text(((x0 + 94) * scale, 24 * scale), label, fill=(30, 32, 35), font=_font(22 * scale, True))

    for item in scene["series"]:
        rgb = COLORS[item["color"]]
        for key in ("a", "b"):
            (x0, y0), (x1, y1) = item[key]
            pts = (round(x0 * scale), round(y0 * scale), round(x1 * scale), round(y1 * scale))
            draw.line(pts, fill=rgb, width=7 * scale)
            radius = 5 * scale
            for x, y in ((x0, y0), (x1, y1)):
                cx, cy = round(x * scale), round(y * scale)
                draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=rgb)

    draw.text((45 * scale, 432 * scale), "All segments run left to right", fill=(75, 78, 82), font=_font(15 * scale))
    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def _slope(endpoints: list[list[float]]) -> float:
    (x0, y0), (x1, y1) = endpoints
    return -(y1 - y0) / (x1 - x0)


def _gains(scene: dict[str, Any]) -> list[tuple[float, str]]:
    return sorted(
        [(_slope(item["b"]) - _slope(item["a"]), item["color"]) for item in scene["series"]],
        reverse=True,
    )


def analytic_gold(scene: dict[str, Any]) -> str:
    return _gains(scene)[0][1]


def margin(scene: dict[str, Any]) -> float:
    gains = _gains(scene)
    return gains[0][0] - gains[1][0]


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) < QUARANTINE_GAP:
        return True
    return any(item[key][1][0] - item[key][0][0] < 100 for item in scene["series"] for key in ("a", "b"))


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["series"] = list(reversed(twin["series"]))
    return [("reverse_unordered_series", twin)]
