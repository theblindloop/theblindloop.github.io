"""Deterministic room-partition renderer and analytic arm."""

from __future__ import annotations

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
GRID_N = 5
GRID_LEFT = 56
CELL = 80
GRID_RIGHT = GRID_LEFT + GRID_N * CELL
WALL = 9
MARKER_RADIUS = 14
PALETTE = {
    "orange": "#e85d04",
    "green": "#168b3f",
    "purple": "#7749c2",
}


def _connected(layout: list[list[int]], region: int) -> bool:
    cells = {(r, c) for r in range(GRID_N) for c in range(GRID_N) if layout[r][c] == region}
    if not cells:
        return False
    seen = {next(iter(cells))}
    queue = deque(seen)
    while queue:
        r, c = queue.popleft()
        for nxt in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if nxt in cells and nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen == cells


def _adjacency(layout: list[list[int]]) -> dict[int, set[int]]:
    regions = {value for row in layout for value in row}
    graph = {region: set() for region in regions}
    for r in range(GRID_N):
        for c in range(GRID_N):
            here = layout[r][c]
            for rr, cc in ((r + 1, c), (r, c + 1)):
                if rr < GRID_N and cc < GRID_N:
                    there = layout[rr][cc]
                    if here != there:
                        graph[here].add(there)
                        graph[there].add(here)
    return graph


def _make_layout(rng: random.Random) -> list[list[int]]:
    """Make a connected Manhattan-Voronoi partition with a degree-3 room."""
    cells = [(r, c) for r in range(GRID_N) for c in range(GRID_N)]
    for _ in range(5000):
        seeds = rng.sample(cells, 7)
        priority = list(range(7))
        rng.shuffle(priority)
        rank = {region: index for index, region in enumerate(priority)}
        layout: list[list[int]] = []
        for r in range(GRID_N):
            row = []
            for c in range(GRID_N):
                distances = [abs(r - sr) + abs(c - sc) for sr, sc in seeds]
                nearest = min(distances)
                tied = [i for i, distance in enumerate(distances) if distance == nearest]
                row.append(min(tied, key=rank.__getitem__))
            layout.append(row)
        if not all(_connected(layout, region) for region in range(7)):
            continue
        degrees = {region: len(neighbors) for region, neighbors in _adjacency(layout).items()}
        if any(value == 3 for value in degrees.values()) and sum(value != 3 for value in degrees.values()) >= 2:
            return layout
    raise RuntimeError("could not sample a valid connected partition")


def _rotated(layout: list[list[int]], turns: int) -> list[list[int]]:
    result = deepcopy(layout)
    for _ in range(turns % 4):
        result = [list(row) for row in zip(*result[::-1])]
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    layout = _rotated(_make_layout(rng), rng.randrange(4))
    graph = _adjacency(layout)
    targets = [region for region, neighbors in graph.items() if len(neighbors) == 3]
    distractors = [region for region, neighbors in graph.items() if len(neighbors) != 3]
    target_region = rng.choice(targets)
    # Prefer different distractor degrees to avoid a shared cheap predicate.
    rng.shuffle(distractors)
    chosen = None
    for left in distractors:
        for right in distractors:
            if left != right and len(graph[left]) != len(graph[right]):
                chosen = [left, right]
                break
        if chosen:
            break
    if chosen is None:
        chosen = distractors[:2]
    colors = list(PALETTE)
    target_color = colors[int(seed) % len(colors)]
    other_colors = [color for color in colors if color != target_color]
    rng.shuffle(other_colors)
    assignment = {target_color: target_region, other_colors[0]: chosen[0], other_colors[1]: chosen[1]}
    markers = {}
    for color, region in assignment.items():
        options = [(r, c) for r in range(GRID_N) for c in range(GRID_N) if layout[r][c] == region]
        markers[color] = list(rng.choice(options))
    return {"layout": layout, "markers": markers}


def render(scene: dict) -> Image.Image:
    layout = scene["layout"]
    markers = scene["markers"]
    image = Image.new("RGB", (SIZE, SIZE), "#e8edf2")
    draw = ImageDraw.Draw(image)
    draw.rectangle((GRID_LEFT, GRID_LEFT, GRID_RIGHT, GRID_RIGHT), fill="white")
    # Exterior wall.
    draw.rectangle((GRID_LEFT, GRID_LEFT, GRID_RIGHT, GRID_RIGHT), outline="#111111", width=WALL)
    # Only boundaries between distinct region IDs are walls; omitted boundaries
    # visibly join grid cells into one chamber.
    half = WALL // 2
    for r in range(GRID_N):
        for c in range(1, GRID_N):
            if layout[r][c - 1] != layout[r][c]:
                x = GRID_LEFT + c * CELL
                y0 = GRID_LEFT + r * CELL
                y1 = y0 + CELL
                draw.rectangle((x - half, y0 - half, x + half, y1 + half), fill="#111111")
    for r in range(1, GRID_N):
        for c in range(GRID_N):
            if layout[r - 1][c] != layout[r][c]:
                y = GRID_LEFT + r * CELL
                x0 = GRID_LEFT + c * CELL
                x1 = x0 + CELL
                draw.rectangle((x0 - half, y - half, x1 + half, y + half), fill="#111111")
    for color, (r, c) in markers.items():
        x = GRID_LEFT + c * CELL + CELL // 2
        y = GRID_LEFT + r * CELL + CELL // 2
        radius = MARKER_RADIUS
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALETTE[color], outline="white", width=2)
    return image


def analytic_gold(scene: dict) -> str:
    layout = scene["layout"]
    markers = scene["markers"]
    graph = _adjacency(layout)
    matches = [color for color, (r, c) in markers.items() if len(graph[layout[r][c]]) == 3]
    if len(matches) != 1:
        return "abstain"
    return matches[0]


def margin(scene: dict) -> float:
    layout = scene["layout"]
    markers = scene["markers"]
    graph = _adjacency(layout)
    gaps = [abs(len(graph[layout[r][c]]) - 3) for r, c in markers.values() if len(graph[layout[r][c]]) != 3]
    return float(min(gaps)) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every declared field is visible; region IDs have no standalone field and
    # are embedded in the visibly rendered partition, so no alias is claimed.
    return []
