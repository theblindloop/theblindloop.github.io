"""Two-panel landmark-correspondence world.

Upper panel: four colored dots forming a constellation.
Lower panel: the same constellation after an orientation-preserving similarity
transform (rotation + uniform scale + translation), drawn in neutral gray with
exactly one dot circled in black.

The decision is the color of the circled landmark. It is fixed by the unique
landmark assignment that aligns the two constellations; the runner-up
assignment supplies a continuous, visible margin.
"""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH = 560
HEIGHT = 440
DIVIDER_Y = 220
TOP_CENTER = (280.0, 112.0)
BOTTOM_CENTER = (280.0, 332.0)

BACKGROUND = (255, 255, 255)
DIVIDER_COLOR = (40, 40, 40)
GRAY = (150, 150, 150)
RING_COLOR = (0, 0, 0)

COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (214, 40, 40),
    "blue": (40, 80, 200),
    "green": (34, 139, 64),
    "orange": (240, 150, 20),
}

DOT_RADIUS = 11
RING_RADIUS = 19
RING_WIDTH = 3

QUARANTINE_MARGIN = 8.0
MIN_MARGIN = 3.0
MIN_PAIR_DISTANCE = 34.0
PANEL_PAD = 30.0


def _similarity_residual(source: list[list[float]], target: list[list[float]]) -> float:
    """RMS residual of the best orientation-preserving similarity source -> target."""
    n = len(source)
    sx = sum(p[0] for p in source) / n
    sy = sum(p[1] for p in source) / n
    tx = sum(p[0] for p in target) / n
    ty = sum(p[1] for p in target) / n
    a = [(p[0] - sx, p[1] - sy) for p in source]
    b = [(p[0] - tx, p[1] - ty) for p in target]
    num_re = sum(a[i][0] * b[i][0] + a[i][1] * b[i][1] for i in range(n))
    num_im = sum(a[i][0] * b[i][1] - a[i][1] * b[i][0] for i in range(n))
    den = sum(a[i][0] * a[i][0] + a[i][1] * a[i][1] for i in range(n))
    if den <= 0.0:
        return 0.0
    kr = num_re / den
    ki = num_im / den
    total = 0.0
    for i in range(n):
        px = kr * a[i][0] - ki * a[i][1]
        py = ki * a[i][0] + kr * a[i][1]
        total += (px - b[i][0]) ** 2 + (py - b[i][1]) ** 2
    return math.sqrt(total / n)


def _mean_pair_distance(points: list[list[float]]) -> float:
    pairs = list(itertools.combinations(range(len(points)), 2))
    total = 0.0
    for i, j in pairs:
        total += math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1])
    return total / len(pairs)


def assignment_margin(top: list[list[float]], bottom: list[list[float]]) -> float:
    """Runner-up misfit as a percentage of the mean inter-dot spacing."""
    n = len(top)
    runner_up = None
    for perm in itertools.permutations(range(n)):
        if perm == tuple(range(n)):
            continue
        residual = _similarity_residual(top, [bottom[k] for k in perm])
        if runner_up is None or residual < runner_up:
            runner_up = residual
    scale = _mean_pair_distance(bottom)
    if scale <= 0.0:
        return 0.0
    return 100.0 * float(runner_up) / scale


def _base_offsets(rng: random.Random, near_symmetric: bool) -> list[list[float]]:
    if near_symmetric:
        radius = rng.uniform(58.0, 72.0)
        phase = rng.uniform(0.0, math.pi / 2.0)
        jitter = rng.uniform(0.010, 0.055) * radius
        points = []
        for k in range(4):
            angle = phase + k * math.pi / 2.0 + rng.uniform(-0.05, 0.05)
            points.append(
                [
                    radius * math.cos(angle) + rng.uniform(-jitter, jitter),
                    radius * math.sin(angle) + rng.uniform(-jitter, jitter),
                ]
            )
        return points
    points: list[list[float]] = []
    while len(points) < 4:
        angle = rng.uniform(0.0, 2.0 * math.pi)
        radius = math.sqrt(rng.uniform(0.05, 1.0)) * 72.0
        cand = [radius * math.cos(angle), radius * math.sin(angle)]
        if all(math.hypot(cand[0] - p[0], cand[1] - p[1]) >= MIN_PAIR_DISTANCE for p in points):
            points.append(cand)
    return points


def _panel_points(
    offsets: list[list[float]], center: tuple[float, float], theta: float, scale: float
) -> list[list[float]]:
    cos_t = math.cos(theta)
    sin_t = math.sin(theta)
    out = []
    for x, y in offsets:
        out.append(
            [
                center[0] + scale * (cos_t * x - sin_t * y),
                center[1] + scale * (sin_t * x + cos_t * y),
            ]
        )
    return out


def _fits(points: list[list[float]], y_low: float, y_high: float) -> bool:
    pad = PANEL_PAD
    return all(
        pad <= p[0] <= WIDTH - pad and y_low + pad <= p[1] <= y_high - pad for p in points
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    while True:
        near_symmetric = rng.random() < 0.20
        offsets = _base_offsets(rng, near_symmetric)
        theta = rng.uniform(0.0, 2.0 * math.pi)
        scale = rng.uniform(0.72, 1.25)
        top = _panel_points(offsets, TOP_CENTER, 0.0, 1.0)
        bottom = _panel_points(offsets, BOTTOM_CENTER, theta, scale)
        if not _fits(top, 0.0, DIVIDER_Y) or not _fits(bottom, DIVIDER_Y, HEIGHT):
            continue
        if min(
            math.hypot(bottom[i][0] - bottom[j][0], bottom[i][1] - bottom[j][1])
            for i, j in itertools.combinations(range(4), 2)
        ) < 2.4 * RING_RADIUS:
            continue
        margin_value = assignment_margin(top, bottom)
        if margin_value < MIN_MARGIN:
            continue
        colors = list(COLOR_NAMES)
        rng.shuffle(colors)
        return {
            "base_points": [[round(v, 6) for v in p] for p in offsets],
            "colors": colors,
            "query_index": rng.randrange(4),
            "theta": round(theta, 9),
            "scale": round(scale, 9),
            "top_center": [TOP_CENTER[0], TOP_CENTER[1]],
            "bottom_center": [BOTTOM_CENTER[0], BOTTOM_CENTER[1]],
            "dot_radius": DOT_RADIUS,
            "ring_radius": RING_RADIUS,
            "near_symmetric": bool(near_symmetric),
        }


def scene_panels(scene: dict) -> tuple[list[list[float]], list[list[float]]]:
    offsets = [[float(p[0]), float(p[1])] for p in scene["base_points"]]
    top_center = (float(scene["top_center"][0]), float(scene["top_center"][1]))
    bottom_center = (float(scene["bottom_center"][0]), float(scene["bottom_center"][1]))
    top = _panel_points(offsets, top_center, 0.0, 1.0)
    bottom = _panel_points(offsets, bottom_center, float(scene["theta"]), float(scene["scale"]))
    return top, bottom


def _disk(draw: ImageDraw.ImageDraw, center: list[float], radius: float, fill) -> None:
    cx = round(center[0])
    cy = round(center[1])
    r = int(radius)
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=fill)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, DIVIDER_Y - 1, WIDTH - 1, DIVIDER_Y + 1], fill=DIVIDER_COLOR)
    top, bottom = scene_panels(scene)
    radius = float(scene["dot_radius"])
    ring_radius = float(scene["ring_radius"])
    for index, point in enumerate(top):
        _disk(draw, point, radius, COLOR_RGB[scene["colors"][index]])
    for point in bottom:
        _disk(draw, point, radius, GRAY)
    marked = bottom[int(scene["query_index"])]
    cx = round(marked[0])
    cy = round(marked[1])
    r = int(ring_radius)
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], outline=RING_COLOR, width=RING_WIDTH)
    return image


def analytic_gold(scene: dict) -> str:
    """Color bound to the landmark that the similarity map sends to the circled dot."""
    return str(scene["colors"][int(scene["query_index"])])


def margin(scene: dict) -> float:
    top, bottom = scene_panels(scene)
    return round(assignment_margin(top, bottom), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every label input is drawn into the raster."""
    return []
