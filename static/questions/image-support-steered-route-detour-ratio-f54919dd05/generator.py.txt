"""Deterministic route-tortuosity world and analytic arm."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 360
BG = (250, 249, 246)
CYAN = (24, 158, 181)
CORAL = (226, 82, 86)
BLUE = (61, 96, 203)


def _length(points: list[list[float]]) -> float:
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:]))


def _signed_margin(points: list[list[float]]) -> float:
    chord = math.hypot(points[-1][0] - points[0][0], points[-1][1] - points[0][1])
    return _length(points) - 2.0 * chord


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    xs = [28.0, 101.0, 174.0, 247.0, 320.0, 393.0, 466.0, 539.0, 612.0]
    start_y = float(rng.randint(145, 215))
    end_y = float(rng.randint(145, 215))
    # Alternating target classes are guaranteed in every consecutive stress set.
    yes = int(seed) % 2 == 0
    if int(seed) % 6 in (0, 1):
        target_ratio = 2.045 if yes else 1.955
    else:
        target_ratio = rng.uniform(2.18, 2.52) if yes else rng.uniform(1.48, 1.82)

    raw = [0.0]
    for index in range(1, len(xs) - 1):
        sign = -1.0 if index % 2 else 1.0
        raw.append(sign * rng.uniform(0.55, 1.0))
    raw.append(0.0)

    def make(scale: float) -> list[list[float]]:
        pts = []
        for index, x in enumerate(xs):
            baseline = start_y + (end_y - start_y) * index / (len(xs) - 1)
            pts.append([x, baseline + scale * raw[index]])
        return pts

    low, high = 0.0, 150.0
    for _ in range(40):
        mid = (low + high) / 2.0
        pts = make(mid)
        ratio = _length(pts) / math.hypot(xs[-1] - xs[0], end_y - start_y)
        if ratio < target_ratio:
            low = mid
        else:
            high = mid
    points = [[round(x, 3), round(y, 3)] for x, y in make((low + high) / 2.0)]
    return {"points": points}


def render(scene: dict) -> Image.Image:
    points = [(float(x), float(y)) for x, y in scene["points"]]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    # A restrained frame gives the question a scientific measurement-figure feel
    # without contributing to the answer.
    draw.rounded_rectangle((12, 12, WIDTH - 13, HEIGHT - 13), radius=12, outline=(216, 217, 213), width=2)
    draw.line(points, fill=CYAN, width=7, joint="curve")
    radius = 11
    for (x, y), color in ((points[0], CORAL), (points[-1], BLUE)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color, outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _signed_margin(scene["points"]) > 0.0 else "no"


def margin(scene: dict) -> float:
    return abs(_signed_margin(scene["points"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
