"""Independent image-only inverse arm for route_detour_ratio."""

from __future__ import annotations

import math
from PIL import Image


def _mask_pixels(image: Image.Image, target: tuple[int, int, int], tolerance: int) -> list[tuple[int, int]]:
    rgb = image.convert("RGB")
    out = []
    tr, tg, tb = target
    for y in range(rgb.height):
        for x in range(rgb.width):
            r, g, b = rgb.getpixel((x, y))
            if (r - tr) ** 2 + (g - tg) ** 2 + (b - tb) ** 2 <= tolerance * tolerance:
                out.append((x, y))
    return out


def _centroid(pixels: list[tuple[int, int]]) -> tuple[float, float]:
    if len(pixels) < 40:
        raise ValueError("endpoint marker is missing")
    return (sum(x for x, _ in pixels) / len(pixels), sum(y for _, y in pixels) / len(pixels))


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    coral = _centroid(_mask_pixels(rgb, (226, 82, 86), 28))
    blue = _centroid(_mask_pixels(rgb, (61, 96, 203), 28))
    cyan = _mask_pixels(rgb, (24, 158, 181), 32)
    if len(cyan) < 1800:
        raise ValueError("route ink is missing")

    columns: dict[int, list[int]] = {}
    for x, y in cyan:
        columns.setdefault(x, []).append(y)
    xs = sorted(columns)
    if len(xs) < 500:
        raise ValueError("route does not span the figure")
    if max(b - a for a, b in zip(xs, xs[1:])) > 12:
        raise ValueError("route has an unresolved gap")

    centers = [(float(x), sum(columns[x]) / len(columns[x])) for x in xs]
    # The endpoint dots visibly cover a small portion of the route. Reinsert their
    # independently segmented centers, then smooth only raster stair-stepping.
    centers = [coral] + centers + [blue]
    route_length = 0.0
    for i in range(1, len(centers)):
        dx = centers[i][0] - centers[i - 1][0]
        dy = centers[i][1] - centers[i - 1][1]
        route_length += math.hypot(dx, dy)
    chord = math.hypot(blue[0] - coral[0], blue[1] - coral[1])
    if chord < 500.0:
        raise ValueError("endpoint scale is invalid")
    return "yes" if route_length > 2.0 * chord else "no"
