"""Scene sampler, rasteriser and analytic gold for the stepwise pen-trail world.

A pen walks a 7x7 grid under a printed list of one-cell arrow moves. The trail of
inked cells is the answer-bearing pattern; the grid itself is printed empty, so the
symbol exists nowhere in the raster until the declared walk is executed.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 7
TILE_COLS = 8
CELL = 44
PITCH = 58
ARROW = 42
CANVAS = (520, 640)

DIRECTIONS: dict[str, tuple[int, int]] = {"U": (-1, 0), "D": (1, 0), "L": (0, -1), "R": (0, 1)}

SYMBOL_NAMES: tuple[str, ...] = ("bracket", "ell", "plus", "stairs", "tee")

PALETTES: tuple[dict[str, list[int]], ...] = (
    {
        "bg": [246, 244, 238],
        "plate": [255, 255, 255],
        "rule": [176, 180, 186],
        "frame": [46, 74, 122],
        "ink": [28, 32, 40],
        "mark": [212, 96, 32],
    },
    {
        "bg": [238, 243, 245],
        "plate": [252, 252, 249],
        "rule": [166, 176, 178],
        "frame": [22, 96, 88],
        "ink": [36, 26, 46],
        "mark": [196, 40, 96],
    },
    {
        "bg": [243, 240, 246],
        "plate": [250, 249, 255],
        "rule": [182, 172, 190],
        "frame": [104, 52, 132],
        "ink": [24, 38, 30],
        "mark": [214, 148, 24],
    },
)


def _template_cells(name: str) -> frozenset[tuple[int, int]]:
    if name == "plus":
        cells = {(3, c) for c in range(GRID)} | {(r, 3) for r in range(GRID)}
    elif name == "tee":
        cells = {(0, c) for c in range(GRID)} | {(r, 3) for r in range(1, GRID)}
    elif name == "ell":
        cells = {(r, 0) for r in range(GRID)} | {(6, c) for c in range(1, GRID)}
    elif name == "bracket":
        cells = (
            {(0, c) for c in range(5)} | {(r, 4) for r in range(1, 4)} | {(4, c) for c in range(5)}
        )
    elif name == "stairs":
        cells = set()
        row, col = 6, 0
        while True:
            cells.add((row, col))
            if row == 0 and col == 6:
                break
            if len(cells) % 2 == 1:
                col += 1
            else:
                row -= 1
    else:
        raise ValueError(f"unknown symbol {name!r}")
    return frozenset(cells)


TEMPLATES: dict[str, frozenset[tuple[int, int]]] = {n: _template_cells(n) for n in SYMBOL_NAMES}


def _bits_from_template(name: str) -> frozenset[tuple[int, int]]:
    return TEMPLATES[name]


def _tour(cells: frozenset[tuple[int, int]], start: tuple[int, int], rng: random.Random):
    """Depth-first covering walk of a connected cell set: 2*(n-1) one-cell moves."""
    seen = {start}
    order = [start]

    def visit(cell: tuple[int, int]) -> None:
        neighbours = [
            (cell[0] + dr, cell[1] + dc)
            for dr, dc in DIRECTIONS.values()
            if (cell[0] + dr, cell[1] + dc) in cells
        ]
        rng.shuffle(neighbours)
        for nxt in neighbours:
            if nxt in seen:
                continue
            seen.add(nxt)
            order.append(nxt)
            visit(nxt)
            order.append(cell)

    visit(start)
    return order


def _add_detours(order: list, cells: frozenset, rng: random.Random, k: int) -> list:
    """Insert k out-and-back excursions into cells outside the covered set."""
    used: set[tuple[int, int]] = set()
    for _ in range(k):
        options = []
        for index, cell in enumerate(order):
            for dr, dc in DIRECTIONS.values():
                spot = (cell[0] + dr, cell[1] + dc)
                if (
                    0 <= spot[0] < GRID
                    and 0 <= spot[1] < GRID
                    and spot not in cells
                    and spot not in used
                ):
                    options.append((index, spot))
        if not options:
            break
        index, spot = options[rng.randrange(len(options))]
        used.add(spot)
        order = order[: index + 1] + [spot, order[index]] + order[index + 1 :]
    return order


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = SYMBOL_NAMES[rng.randrange(len(SYMBOL_NAMES))]
    cells = TEMPLATES[target]
    start = sorted(cells)[rng.randrange(len(cells))]
    order = _tour(cells, start, rng)
    detours = rng.choice([0, 0, 0, 1, 1, 2, 2, 3, 4])
    order = _add_detours(order, cells, rng, detours)
    moves = []
    for before, after in zip(order, order[1:]):
        step = (after[0] - before[0], after[1] - before[1])
        moves.append(next(key for key, value in DIRECTIONS.items() if value == step))
    palette = PALETTES[rng.randrange(len(PALETTES))]
    scene = {
        "start": [start[0], start[1]],
        "moves": moves,
        "grid_x": 100 + rng.randrange(-18, 19),
        "grid_y": 300 + rng.randrange(-10, 11),
        "strip_x": 30 + rng.randrange(-8, 9),
        "strip_y": 22 + rng.randrange(-6, 7),
    }
    scene.update({key: list(value) for key, value in palette.items()})
    return scene


def trail_cells(scene: dict) -> frozenset[tuple[int, int]]:
    """The declared transform: run the printed move list and collect inked cells."""
    row, col = int(scene["start"][0]), int(scene["start"][1])
    inked = {(row, col)}
    for move in scene["moves"]:
        dr, dc = DIRECTIONS[str(move).upper()]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID and 0 <= col < GRID):
            raise ValueError("declared walk left the printed grid")
        inked.add((row, col))
    return frozenset(inked)


def _distances(inked: frozenset) -> list[tuple[int, str]]:
    return sorted((len(inked ^ TEMPLATES[name]), name) for name in SYMBOL_NAMES)


def analytic_gold(scene: dict) -> str:
    return _distances(trail_cells(scene))[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(trail_cells(scene))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def _naive_read_distance(scene: dict) -> int:
    """Distance between the true trail and an order-blind read of the same arrows."""
    moves = sorted(str(move).upper() for move in scene["moves"])
    row, col = int(scene["start"][0]), int(scene["start"][1])
    inked = {(row, col)}
    for move in moves:
        dr, dc = DIRECTIONS[move]
        row = min(GRID - 1, max(0, row + dr))
        col = min(GRID - 1, max(0, col + dc))
        inked.add((row, col))
    return len(inked ^ trail_cells(scene))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Label-preserving latent aliases that must be pixel-identical."""
    lowered = dict(scene)
    lowered["moves"] = [str(move).lower() for move in scene["moves"]]
    floated = dict(scene)
    floated["start"] = [float(scene["start"][0]), float(scene["start"][1])]
    recast = dict(scene)
    for key in ("bg", "plate", "rule", "frame", "ink", "mark"):
        recast[key] = [float(channel) for channel in scene[key]]
    return [
        ("lowercase_moves", lowered),
        ("float_start", floated),
        ("float_channels", recast),
    ]


def _rgb(value) -> tuple[int, int, int]:
    return (int(value[0]), int(value[1]), int(value[2]))


def _arrow_polygon(x: int, y: int, direction: str) -> list[tuple[int, int]]:
    half = ARROW // 2
    cx, cy = x + half, y + half
    if direction == "R":
        return [(cx - half, cy - half), (cx - half, cy + half), (cx + half, cy)]
    if direction == "L":
        return [(cx + half, cy - half), (cx + half, cy + half), (cx - half, cy)]
    if direction == "D":
        return [(cx - half, cy - half), (cx + half, cy - half), (cx, cy + half)]
    return [(cx - half, cy + half), (cx + half, cy + half), (cx, cy - half)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, _rgb(scene["bg"]))
    draw = ImageDraw.Draw(image)

    strip_x, strip_y = int(scene["strip_x"]), int(scene["strip_y"])
    for index, move in enumerate(scene["moves"]):
        row, col = divmod(index, TILE_COLS)
        x = strip_x + col * PITCH
        y = strip_y + row * PITCH
        draw.polygon(_arrow_polygon(x, y, str(move).upper()), fill=_rgb(scene["ink"]))

    gx, gy = int(scene["grid_x"]), int(scene["grid_y"])
    span = GRID * CELL
    draw.rectangle([gx - 7, gy - 7, gx + span + 6, gy + span + 6], fill=_rgb(scene["frame"]))
    draw.rectangle([gx, gy, gx + span - 1, gy + span - 1], fill=_rgb(scene["rule"]))
    for row in range(GRID):
        for col in range(GRID):
            x0 = gx + col * CELL + 2
            y0 = gy + row * CELL + 2
            draw.rectangle([x0, y0, x0 + CELL - 5, y0 + CELL - 5], fill=_rgb(scene["plate"]))
    srow, scol = int(scene["start"][0]), int(scene["start"][1])
    mx = gx + scol * CELL + 11
    my = gy + srow * CELL + 11
    draw.rectangle(
        [mx, my, mx + CELL - 23, my + CELL - 23], outline=_rgb(scene["mark"]), width=4
    )
    return image
