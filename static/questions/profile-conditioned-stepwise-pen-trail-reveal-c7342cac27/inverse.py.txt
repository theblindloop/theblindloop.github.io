"""Image-only inverse arm for the stepwise pen-trail world.

Receives one rasterised scene and the public question contract only. It locates the
printed arrows and the start marker, executes the declared one-cell walk on the 7x7
grid, and matches the inked trail to the nearest named symbol template.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

GRID = 7
SYMBOL_NAMES: tuple[str, ...] = ("bracket", "ell", "plus", "stairs", "tee")


def _templates() -> dict[str, frozenset]:
    plus = {(3, c) for c in range(GRID)} | {(r, 3) for r in range(GRID)}
    tee = {(0, c) for c in range(GRID)} | {(r, 3) for r in range(1, GRID)}
    ell = {(r, 0) for r in range(GRID)} | {(6, c) for c in range(1, GRID)}
    bracket = {(0, c) for c in range(5)} | {(r, 4) for r in range(1, 4)} | {(4, c) for c in range(5)}
    stairs = set()
    row, col = 6, 0
    while True:
        stairs.add((row, col))
        if (row, col) == (0, 6):
            break
        if len(stairs) % 2 == 1:
            col += 1
        else:
            row -= 1
    return {
        "bracket": frozenset(bracket),
        "ell": frozenset(ell),
        "plus": frozenset(plus),
        "stairs": frozenset(stairs),
        "tee": frozenset(tee),
    }


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[dict]:
    """Connected components (4-connectivity) via row runs and union-find."""
    height, width = mask.shape
    parent: list[int] = []

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    runs: list[tuple[int, int, int, int]] = []  # y, x0, x1, label
    previous: list[tuple[int, int, int]] = []
    for y in range(height):
        row = mask[y].astype(np.int8)
        edges = np.flatnonzero(np.diff(np.concatenate(([0], row, [0]))))
        current: list[tuple[int, int, int]] = []
        for start, stop in zip(edges[0::2], edges[1::2]):
            label = len(parent)
            parent.append(label)
            for py0, py1, plabel in previous:
                if start < py1 and py0 < stop:
                    union(label, plabel)
            current.append((int(start), int(stop), label))
            runs.append((y, int(start), int(stop), label))
        previous = current

    groups: dict[int, dict] = {}
    for y, x0, x1, label in runs:
        root = find(label)
        area = x1 - x0
        stats = groups.setdefault(
            root,
            {"area": 0, "sx": 0.0, "sy": 0.0, "x0": x0, "x1": x1, "y0": y, "y1": y},
        )
        stats["area"] += area
        stats["sx"] += (x0 + x1 - 1) * area / 2.0
        stats["sy"] += float(y) * area
        stats["x0"] = min(stats["x0"], x0)
        stats["x1"] = max(stats["x1"], x1)
        stats["y0"] = min(stats["y0"], y)
        stats["y1"] = max(stats["y1"], y)
    out = []
    for stats in groups.values():
        stats["cx"] = stats["sx"] / stats["area"]
        stats["cy"] = stats["sy"] / stats["area"]
        out.append(stats)
    return out


def _colour_layers(arr: np.ndarray) -> list[tuple[tuple[int, int, int], np.ndarray, int]]:
    flat = arr.reshape(-1, 3)
    codes = (flat[:, 0].astype(np.int64) << 16) | (flat[:, 1].astype(np.int64) << 8) | flat[:, 2]
    values, counts = np.unique(codes, return_counts=True)
    layers = []
    for value, count in zip(values, counts):
        colour = (int(value >> 16) & 255, int(value >> 8) & 255, int(value) & 255)
        layers.append((colour, (codes == value).reshape(arr.shape[:2]), int(count)))
    layers.sort(key=lambda item: -item[2])
    return layers


def _read_arrows(components: list[dict]) -> tuple[list[str], float] | None:
    heights = sorted(stats["y1"] - stats["y0"] + 1 for stats in components)
    typical = heights[len(heights) // 2]
    if typical < 8:
        return None
    ordered = sorted(components, key=lambda stats: stats["cy"])
    rows: list[list[dict]] = [[ordered[0]]]
    for stats in ordered[1:]:
        if stats["cy"] - rows[-1][-1]["cy"] > 0.6 * typical:
            rows.append([stats])
        else:
            rows[-1].append(stats)
    moves: list[str] = []
    confidence = 1e9
    for row in rows:
        for stats in sorted(row, key=lambda item: item["cx"]):
            width = stats["x1"] - stats["x0"]
            height = stats["y1"] - stats["y0"] + 1
            dx = stats["cx"] - (stats["x0"] + stats["x1"] - 1) / 2.0
            dy = stats["cy"] - (stats["y0"] + stats["y1"]) / 2.0
            major, minor = (abs(dx), abs(dy)) if abs(dx) >= abs(dy) else (abs(dy), abs(dx))
            scale = max(width, height, 1)
            if major < 0.06 * scale or major < 2.0 * max(minor, 1e-6):
                return None
            confidence = min(confidence, major / scale)
            if abs(dx) >= abs(dy):
                moves.append("R" if dx < 0 else "L")
            else:
                moves.append("D" if dy < 0 else "U")
    return moves, confidence


def decide(image: Image.Image) -> dict:
    arr = np.asarray(image.convert("RGB"))
    layers = _colour_layers(arr)
    if len(layers) < 4:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "too few printed colours"}

    scored = []
    for colour, mask, count in layers[1:]:
        if count < 40:
            continue
        parts = _components(mask)
        scored.append((colour, mask, count, parts))
    if len(scored) < 3:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "not enough printed layers"}

    scored.sort(key=lambda item: -len(item[3]))
    plate = scored[0]
    arrows = scored[1]
    if len(plate[3]) < 40 or not 6 <= len(arrows[3]) <= 40:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "grid or arrow layer is missing"}

    cells = plate[3]
    gx0 = min(stats["x0"] for stats in cells)
    gx1 = max(stats["x1"] for stats in cells)
    gy0 = min(stats["y0"] for stats in cells)
    gy1 = max(stats["y1"] for stats in cells)
    pitch_x = (gx1 - gx0) / GRID
    pitch_y = (gy1 + 1 - gy0) / GRID
    if pitch_x < 8 or pitch_y < 8 or abs(pitch_x - pitch_y) > 0.25 * pitch_x:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "grid geometry is unstable"}

    marker = None
    for colour, mask, count, parts in scored[2:]:
        span = max(stats["x1"] for stats in parts) - min(stats["x0"] for stats in parts)
        tall = max(stats["y1"] for stats in parts) - min(stats["y0"] for stats in parts)
        if span > 2.2 * pitch_x or tall > 2.2 * pitch_y:
            continue
        if not (gx0 - pitch_x <= parts[0]["cx"] <= gx1 + pitch_x):
            continue
        if not (gy0 - pitch_y <= parts[0]["cy"] <= gy1 + pitch_y):
            continue
        centre_x = sum(s["cx"] * s["area"] for s in parts) / sum(s["area"] for s in parts)
        centre_y = sum(s["cy"] * s["area"] for s in parts) / sum(s["area"] for s in parts)
        if marker is None or count < marker[2]:
            marker = (centre_x, centre_y, count)
    if marker is None:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "no start marker"}

    col = int((marker[0] - gx0 + 2) // pitch_x)
    row = int((marker[1] - gy0 + 2) // pitch_y)
    if not (0 <= row < GRID and 0 <= col < GRID):
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "start marker outside the grid"}

    read = _read_arrows(arrows[3])
    if read is None:
        return {"decision": ABSTAIN, "margin": 0.0, "reason": "arrow directions are ambiguous"}
    moves, confidence = read

    steps = {"U": (-1, 0), "D": (1, 0), "L": (0, -1), "R": (0, 1)}
    inked = {(row, col)}
    for move in moves:
        dr, dc = steps[move]
        row, col = row + dr, col + dc
        if not (0 <= row < GRID and 0 <= col < GRID):
            return {"decision": ABSTAIN, "margin": 0.0, "reason": "walk left the printed grid"}
        inked.add((row, col))

    ranked = sorted((len(inked ^ TEMPLATES[name]), name) for name in SYMBOL_NAMES)
    return {
        "decision": ranked[0][1],
        "margin": float(ranked[1][0] - ranked[0][0]),
        "arrow_confidence": float(confidence),
        "moves_read": len(moves),
        "reason": "ok",
    }


def decision_from_image(image) -> str:
    return decide(image)["decision"]


def margin_from_image(image) -> float:
    return float(decide(image)["margin"])
