"""Renderer and analytic arm for the proposed-link cycle question."""

from __future__ import annotations

from collections import deque
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 360
BG = (250, 249, 245)
FRAME = (196, 199, 201)
OUTLINE = (31, 35, 39)
LINK = (105, 112, 120)
CYAN = (38, 176, 190)
GOLD = (230, 153, 30)
LABELS = ("left", "center", "right")

# Regular-hexagon vertex indices. Every pattern is P3 + P3, uses two side
# lengths and two sqrt(3)-side lengths, and has the same degree sequence.
WINNER = {"edges": [[0, 1], [1, 3], [2, 4], [4, 5]], "proposal": [0, 3]}
LOSER_A = {"edges": [[0, 2], [2, 1], [3, 5], [5, 4]], "proposal": [0, 3]}
LOSER_B = {"edges": [[0, 2], [0, 5], [3, 1], [3, 4]], "proposal": [0, 3]}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    panels = [WINNER, LOSER_A, LOSER_B]
    panels = [{"edges": [list(e) for e in p["edges"]], "proposal": list(p["proposal"])} for p in panels]
    rng.shuffle(panels)
    # Reflecting vertex placement preserves all segment lengths and avoids a
    # single screen-orientation cue; the graph remains entirely explicit.
    if rng.randrange(2):
        remap = {0: 3, 1: 2, 2: 1, 3: 0, 4: 5, 5: 4}
        for panel in panels:
            panel["edges"] = [[remap[a], remap[b]] for a, b in panel["edges"]]
            panel["proposal"] = [remap[a] for a in panel["proposal"]]
    return {"panels": panels, "offset": [rng.randint(-4, 4), rng.randint(-7, 7)]}


def _positions(panel_index: int, offset: list[int]) -> list[list[int]]:
    cx = 120 + 240 * panel_index + offset[0]
    cy = 184 + offset[1]
    # Exact integer approximation of a regular hexagon, starting at top.
    return [[cx, cy - 92], [cx + 80, cy - 46], [cx + 80, cy + 46],
            [cx, cy + 92], [cx - 80, cy + 46], [cx - 80, cy - 46]]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for i, panel in enumerate(scene["panels"]):
        positions = _positions(i, scene["offset"])
        x0 = 12 + 240 * i + scene["offset"][0]
        draw.rounded_rectangle((x0, 36 + scene["offset"][1], x0 + 216, 326 + scene["offset"][1]),
                               radius=13, outline=FRAME, width=3)
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=OUTLINE, width=12)
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=LINK, width=8)
        a, b = panel["proposal"]
        draw.line([tuple(positions[a]), tuple(positions[b])], fill=OUTLINE, width=14)
        draw.line([tuple(positions[a]), tuple(positions[b])], fill=GOLD, width=9)
        for x, y in positions:
            draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=OUTLINE)
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=CYAN)
    return image


def _connected(edges: list[list[int]], start: int, goal: int) -> bool:
    adjacency = {i: set() for i in range(6)}
    for a, b in edges:
        adjacency[a].add(b); adjacency[b].add(a)
    seen = {start}; queue = deque([start])
    while queue:
        v = queue.popleft()
        for n in adjacency[v]:
            if n not in seen:
                seen.add(n); queue.append(n)
    return goal in seen


def analytic_gold(scene: dict) -> str:
    winners = []
    for label, panel in zip(LABELS, scene["panels"]):
        if _connected(panel["edges"], *panel["proposal"]):
            winners.append(label)
    if len(winners) != 1:
        raise ValueError("scene must have exactly one cycle-closing proposal")
    return winners[0]


def margin(scene: dict) -> float:
    # One edge insertion is the exact observable discrete topology boundary.
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["panels"]) != 3 or len(scene["offset"]) != 2:
            return True
        results = []
        for panel in scene["panels"]:
            if len(panel["edges"]) != 4 or len(panel["proposal"]) != 2:
                return True
            vertices = {v for e in panel["edges"] for v in e}
            if vertices != set(range(6)) or panel["proposal"] in panel["edges"] or list(reversed(panel["proposal"])) in panel["edges"]:
                return True
            # Four edges on six vertices must form exactly two acyclic components.
            adjacency = {v: set() for v in range(6)}
            for a, b in panel["edges"]:
                adjacency[a].add(b); adjacency[b].add(a)
            unseen = set(range(6)); components = 0
            while unseen:
                components += 1; start = unseen.pop(); queue = [start]
                while queue:
                    v = queue.pop()
                    for n in adjacency[v]:
                        if n in unseen:
                            unseen.remove(n); queue.append(n)
            if components != 2:
                return True
            results.append(_connected(panel["edges"], *panel["proposal"]))
        return sum(results) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "panels": [{"edges": [list(e) for e in reversed(p["edges"])], "proposal": list(p["proposal"])}
                   for p in scene["panels"]],
        "offset": list(scene["offset"]),
    }
    return [("reverse_gray_edge_serialization", twin)]
