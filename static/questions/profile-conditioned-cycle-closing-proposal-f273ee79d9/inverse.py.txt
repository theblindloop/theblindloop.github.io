"""Independent pixel-only inverse arm for proposed-link cycle closure."""

from __future__ import annotations

from collections import deque
import numpy as np

CYAN = np.array([38, 176, 190])
GOLD = np.array([230, 153, 30])
LINK = np.array([105, 112, 120])


def _components(mask: np.ndarray, minimum: int = 70) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]; seen[y, x] = True; xs = []; ys = []
        while stack:
            cy, cx = stack.pop(); xs.append(cx); ys.append(cy)
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; stack.append((ny, nx))
        if len(xs) >= minimum:
            result.append((sum(xs)/len(xs), sum(ys)/len(ys), len(xs)))
    return result


def _has_band(rgb: np.ndarray, p: tuple[float, float], q: tuple[float, float], color: np.ndarray) -> bool:
    x1, y1 = p; x2, y2 = q
    distance = float(np.hypot(x2-x1, y2-y1))
    if distance < 45:
        return False
    hits = 0; total = 0
    for t in np.linspace(18.0/distance, 1.0-18.0/distance, max(18, int(distance-34))):
        x = int(round(x1 + t*(x2-x1))); y = int(round(y1 + t*(y2-y1)))
        patch = rgb[max(0,y-2):y+3, max(0,x-2):x+3]
        if patch.size:
            delta = np.max(np.abs(patch.astype(int) - color), axis=2)
            hits += int(np.any(delta <= 5)); total += 1
    return total > 0 and hits / total > 0.76


def _connected(adjacency: list[set[int]], start: int, goal: int) -> bool:
    seen = {start}; queue = deque([start])
    while queue:
        v = queue.popleft()
        for n in adjacency[v]:
            if n not in seen:
                seen.add(n); queue.append(n)
    return goal in seen


def decision_from_image(image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan_delta = np.max(np.abs(rgb.astype(int) - CYAN), axis=2)
    nodes = sorted(_components(cyan_delta <= 5), key=lambda item: item[0])
    if len(nodes) != 18:
        raise ValueError("could not recover eighteen stations")
    # The three framed panels occupy fixed thirds, but no filename, metadata,
    # labels, or latent coordinates are consulted.
    groups = [[], [], []]
    for x, y, _ in nodes:
        index = min(2, max(0, int(x // (rgb.shape[1] / 3))))
        groups[index].append((x, y))
    if any(len(group) != 6 for group in groups):
        raise ValueError("could not separate the three station panels")
    winners = []
    for panel_index, points in enumerate(groups):
        adjacency = [set() for _ in points]
        proposals = []
        gray_count = 0
        for i in range(6):
            for j in range(i+1, 6):
                if _has_band(rgb, points[i], points[j], LINK):
                    adjacency[i].add(j); adjacency[j].add(i); gray_count += 1
                if _has_band(rgb, points[i], points[j], GOLD):
                    proposals.append((i, j))
        if gray_count != 4 or len(proposals) != 1:
            raise ValueError("ambiguous gray forest or proposed link")
        if _connected(adjacency, *proposals[0]):
            winners.append(("left", "center", "right")[panel_index])
    if len(winners) != 1:
        raise ValueError("pixel graph has no unique cycle-closing proposal")
    return winners[0]
