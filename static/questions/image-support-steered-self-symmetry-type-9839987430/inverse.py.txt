"""Pixel-only inverse arm for the self-symmetry-type world.

Input is the final PNG and nothing else: no latent scene, no renderer import,
no analytic gold. The arm reconstructs the declared global evidence (the dark
region's own support and its centre of mass), then scores the three declared
isometries about that centre and applies the declared decision rule.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "self_symmetry_type-oracle-0.1.0"

DARK_LUMA = 150.0
ACCEPT = 0.96
GAP = 0.08
MIN_AREA_FRACTION = 0.04
MAX_AREA_FRACTION = 0.55
AMBIGUOUS = "ambiguous"


def _figure_support(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    luma = 0.299 * array[:, :, 0] + 0.587 * array[:, :, 1] + 0.114 * array[:, :, 2]
    return luma < DARK_LUMA


def _reflect_horizontally(support: np.ndarray, cx: int) -> np.ndarray:
    out = np.zeros_like(support)
    columns = 2 * cx - np.arange(support.shape[1])
    inside = (columns >= 0) & (columns < support.shape[1])
    out[:, inside] = support[:, columns[inside]]
    return out


def _reflect_vertically(support: np.ndarray, cy: int) -> np.ndarray:
    out = np.zeros_like(support)
    rows = 2 * cy - np.arange(support.shape[0])
    inside = (rows >= 0) & (rows < support.shape[0])
    out[inside, :] = support[rows[inside], :]
    return out


def _overlap(a: np.ndarray, b: np.ndarray) -> float:
    both = int(np.count_nonzero(np.logical_and(a, b)))
    either = int(np.count_nonzero(np.logical_or(a, b)))
    return 0.0 if either == 0 else both / either


def decision_from_image(image: Image.Image) -> str:
    support = _figure_support(image)
    pixels = int(np.count_nonzero(support))
    total = support.shape[0] * support.shape[1]
    if pixels < MIN_AREA_FRACTION * total or pixels > MAX_AREA_FRACTION * total:
        return AMBIGUOUS
    rows, columns = np.nonzero(support)
    cx = int(round(float(columns.mean())))
    cy = int(round(float(rows.mean())))
    mirrored_lr = _reflect_horizontally(support, cx)
    scored = {
        "vertical_mirror": _overlap(support, mirrored_lr),
        "horizontal_mirror": _overlap(support, _reflect_vertically(support, cy)),
        "point_rotation": _overlap(support, _reflect_vertically(mirrored_lr, cy)),
    }
    ordered = sorted(scored.items(), key=lambda item: (-item[1], item[0]))
    winner, top = ordered[0]
    runner_up = ordered[1][1]
    if top < ACCEPT or top - runner_up < GAP:
        return AMBIGUOUS
    return winner


def decision_scores(image: Image.Image) -> dict:
    """Diagnostic helper (not used by the protected probe)."""
    support = _figure_support(image)
    if not int(np.count_nonzero(support)):
        return {}
    rows, columns = np.nonzero(support)
    cx = int(round(float(columns.mean())))
    cy = int(round(float(rows.mean())))
    mirrored_lr = _reflect_horizontally(support, cx)
    return {
        "vertical_mirror": _overlap(support, mirrored_lr),
        "horizontal_mirror": _overlap(support, _reflect_vertically(support, cy)),
        "point_rotation": _overlap(support, _reflect_vertically(mirrored_lr, cy)),
    }
