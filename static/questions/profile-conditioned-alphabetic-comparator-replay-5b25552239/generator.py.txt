"""Deterministic visual world for conditional adjacent-comparator replay."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 620
BACKGROUND = (247, 249, 252)
INK = (32, 42, 56)
MUTED = (103, 116, 137)
CARD_BORDER = (183, 193, 207)
PIP_COLOR = (111, 45, 168)
DIRECTION_COLOR = (230, 86, 30)
RING_COLOR = (220, 38, 38)
TILE_COLORS = (
    (48, 143, 224),
    (44, 176, 112),
    (245, 183, 36),
    (36, 184, 196),
)
LABELS = ("A", "B", "C", "D")

# Public 5x7 block alphabet. Each row is left-to-right; 1 means visible dark ink.
GLYPHS: dict[str, tuple[str, ...]] = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _font(size: int, bold: bool = True) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _replay(scene: dict[str, Any]) -> list[str]:
    state = list(scene["initial_order"])
    for operation in scene["operations"]:
        left = int(operation["pair"]) - 1
        later_on_left = operation["later_side"] == "left"
        should_swap = (state[left] < state[left + 1]) if later_on_left else (state[left] > state[left + 1])
        if should_swap:
            state[left], state[left + 1] = state[left + 1], state[left]
    return state


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one fully visible state program from an integer seed."""
    rng = random.Random(int(seed))
    initial = list(LABELS)
    rng.shuffle(initial)
    pairs = [1, 1, 2, 2, 3, 3]
    sides = ["left", "left", "left", "right", "right", "right"]
    rng.shuffle(pairs)
    rng.shuffle(sides)
    operations = [
        {"pair": pair, "later_side": side}
        for pair, side in zip(pairs, sides, strict=True)
    ]
    palette = list(TILE_COLORS)
    rng.shuffle(palette)
    return {
        "initial_order": initial,
        "operations": operations,
        "target_slot": rng.randrange(4),
        "color_map": {label: list(color) for label, color in zip(LABELS, palette, strict=True)},
        "glyph_scale": rng.choice((7, 8, 9)),
        "layout_shift": [rng.randint(-12, 12), rng.randint(-5, 5)],
        "card_jitter": [[rng.randint(-3, 3), rng.randint(-4, 4)] for _ in range(6)],
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the label occupying the visibly queried slot after replay."""
    final = _replay(scene)
    return final[int(scene["target_slot"])]


def _glyph_hamming(a: str, b: str) -> int:
    return sum(x != y for ra, rb in zip(GLYPHS[a], GLYPHS[b], strict=True) for x, y in zip(ra, rb, strict=True))


def margin(scene: dict[str, Any]) -> float:
    """Pixel disagreement to the nearest valid glyph or dot-count runner-up."""
    scale = int(scene["glyph_scale"])
    glyph_gap = min(
        _glyph_hamming(a, b) * scale * scale
        for i, a in enumerate(LABELS)
        for b in LABELS[i + 1 :]
    )
    pip_gap = math.pi * (scale + 3) ** 2
    return float(round(min(glyph_gap, pip_gap), 6))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 180.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _draw_centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font: Any, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_glyph(draw: ImageDraw.ImageDraw, center: tuple[int, int], label: str, scale: int) -> None:
    pattern = GLYPHS[label]
    x0 = int(round(center[0] - 2.5 * scale))
    y0 = int(round(center[1] - 3.5 * scale))
    for row, bits in enumerate(pattern):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (x0 + col * scale, y0 + row * scale, x0 + (col + 1) * scale - 1, y0 + (row + 1) * scale - 1),
                    fill=INK,
                )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible fields; the inverse arm receives this raster alone."""
    initial = list(scene["initial_order"])
    operations = [dict(value) for value in scene["operations"]]
    target_slot = int(scene["target_slot"])
    color_map = scene["color_map"]
    scale = int(scene["glyph_scale"])
    shift_x, shift_y = (int(value) for value in scene["layout_shift"])
    jitter = [[int(v) for v in pair] for pair in scene["card_jitter"]]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((24, 18, WIDTH - 24, HEIGHT - 18), radius=24, fill=(255, 255, 255), outline=(222, 228, 236), width=2)
    draw.text((52, 39), "STARTING SLOTS", font=_font(22), fill=INK)
    draw.text((52, 69), "Keep the letter identities attached as the cards are replayed.", font=_font(15, False), fill=MUTED)

    centers_x = [170 + 220 * index + shift_x for index in range(4)]
    tile_y = 163 + shift_y
    tile_size = 100
    for slot, (x, label) in enumerate(zip(centers_x, initial, strict=True)):
        _draw_centered_text(draw, (x, 91 + shift_y), str(slot + 1), _font(22), MUTED)
        half = tile_size // 2
        box = (x - half, tile_y - half, x + half, tile_y + half)
        if slot == target_slot:
            draw.rounded_rectangle((box[0] - 11, box[1] - 11, box[2] + 11, box[3] + 11), radius=22, outline=RING_COLOR, width=7)
        draw.rounded_rectangle(box, radius=18, fill=tuple(color_map[label]), outline=INK, width=4)
        _draw_glyph(draw, (x, tile_y), label, scale)

    draw.line((52, 250, WIDTH - 52, 250), fill=(222, 228, 236), width=2)
    draw.text((52, 269), "COMPARATOR CARDS", font=_font(22), fill=INK)
    draw.text((52, 299), "Read left to right. Put the later letter on the side indicated by each orange arrow.", font=_font(15, False), fill=MUTED)

    card_w, card_h, gap = 126, 154, 24
    start_x = 52
    card_y = 344
    for index, operation in enumerate(operations):
        jx, jy = jitter[index]
        left = start_x + index * (card_w + gap) + jx
        top = card_y + jy
        right, bottom = left + card_w, top + card_h
        draw.rounded_rectangle((left, top, right, bottom), radius=16, fill=(250, 251, 253), outline=CARD_BORDER, width=3)
        _draw_centered_text(draw, ((left + right) / 2, top + 25), f"STEP {index + 1}", _font(12), MUTED)
        pair = int(operation["pair"])
        dot_y = top + 65
        radius = scale + 3
        spacing = 30
        for dot in range(pair):
            dot_x = (left + right) / 2 + (dot - (pair - 1) / 2) * spacing
            draw.ellipse((dot_x - radius, dot_y - radius, dot_x + radius, dot_y + radius), fill=PIP_COLOR)
        arrow_y = top + 103
        arrow_left, arrow_right = (left + right) / 2 - 24, (left + right) / 2 + 24
        draw.rectangle((arrow_left, arrow_y - 4, arrow_right, arrow_y + 4), fill=DIRECTION_COLOR)
        if operation["later_side"] == "right":
            draw.polygon(((arrow_right + 10, arrow_y), (arrow_right - 4, arrow_y - 13), (arrow_right - 4, arrow_y + 13)), fill=DIRECTION_COLOR)
        else:
            draw.polygon(((arrow_left - 10, arrow_y), (arrow_left + 4, arrow_y - 13), (arrow_left + 4, arrow_y + 13)), fill=DIRECTION_COLOR)
        _draw_centered_text(draw, ((left + right) / 2, top + 135), f"{pair} ↔ {pair + 1}", _font(15), INK)
        if index < 5:
            arrow_x = right + gap / 2
            draw.polygon(((arrow_x - 5, top + 77 - 7), (arrow_x + 6, top + 77), (arrow_x - 5, top + 77 + 7)), fill=(146, 157, 174))

    legend_y = 549
    draw.rounded_rectangle((52, legend_y, WIDTH - 52, legend_y + 42), radius=12, fill=(245, 241, 250))
    _draw_centered_text(draw, (WIDTH / 2, legend_y + 21), "Dots select the pair     •     Orange arrow points toward the alphabetically later tile", _font(15), INK)
    return image
