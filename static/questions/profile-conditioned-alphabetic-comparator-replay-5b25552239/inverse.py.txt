"""Independent image-only inverse for Alphabetic Comparator Replay."""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

_TILE_COLORS = (
    (48, 143, 224),
    (44, 176, 112),
    (245, 183, 36),
    (36, 184, 196),
)
_PIP_COLOR = (111, 45, 168)
_DIRECTION_COLOR = (230, 86, 30)
_RING_COLOR = (220, 38, 38)
_INK = (32, 42, 56)
_GLYPHS: dict[str, tuple[str, ...]] = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _near(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 3) -> np.ndarray:
    target = np.asarray(color, dtype=np.int16)
    return np.max(np.abs(arr.astype(np.int16) - target), axis=2) <= tolerance


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    if values.size == 0:
        return []
    ordered = np.unique(values.astype(int))
    result: list[tuple[int, int]] = []
    start = previous = int(ordered[0])
    for value in ordered[1:]:
        value = int(value)
        if value != previous + 1:
            result.append((start, previous))
            start = value
        previous = value
    result.append((start, previous))
    return result


def _tile_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    boxes: list[tuple[int, int, int, int]] = []
    for color in _TILE_COLORS:
        ys, xs = np.nonzero(_near(arr, color))
        if len(xs) < 5000:
            raise ValueError("missing or damaged colored tile")
        boxes.append((int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())))
    boxes.sort(key=lambda box: (box[0] + box[2]) / 2)
    if len(boxes) != 4 or any(box[2] - box[0] < 90 or box[3] - box[1] < 90 for box in boxes):
        raise ValueError("could not recover four full tiles")
    return boxes


def _classify_glyph(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[str, int]:
    x0, y0, x1, y1 = box
    pad = 12
    crop = _near(arr[y0 + pad : y1 - pad + 1, x0 + pad : x1 - pad + 1], _INK, tolerance=5)
    ys, xs = np.nonzero(crop)
    if len(xs) < 250:
        raise ValueError("letter glyph is missing")
    gx0, gx1, gy0, gy1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = gx1 - gx0 + 1, gy1 - gy0 + 1
    if width % 5 or height % 7 or width < 30 or height < 42:
        raise ValueError("letter glyph grid is malformed")
    cell_w, cell_h = width // 5, height // 7
    observed: list[str] = []
    for row in range(7):
        bits = []
        for col in range(5):
            cell = crop[gy0 + row * cell_h : gy0 + (row + 1) * cell_h, gx0 + col * cell_w : gx0 + (col + 1) * cell_w]
            bits.append("1" if float(cell.mean()) > 0.72 else "0")
        observed.append("".join(bits))
    scores = {
        label: sum(a != b for ra, rb in zip(observed, pattern, strict=True) for a, b in zip(ra, rb, strict=True))
        for label, pattern in _GLYPHS.items()
    }
    ranked = sorted(scores.items(), key=lambda item: (item[1], item[0]))
    if ranked[0][1] != 0 or ranked[1][1] - ranked[0][1] < 3:
        raise ValueError("letter glyph is ambiguous")
    return ranked[0][0], ranked[1][1] - ranked[0][1]


def _operations(arr: np.ndarray) -> list[dict[str, Any]]:
    ys, xs = np.nonzero(_near(arr, _PIP_COLOR))
    if len(xs) < 200:
        raise ValueError("comparator dots are missing")
    dot_runs = _runs(xs)
    dot_centers = [(lo + hi) / 2 for lo, hi in dot_runs if hi - lo >= 16]
    if not dot_centers:
        raise ValueError("no complete comparator dots")
    groups: list[list[float]] = [[dot_centers[0]]]
    for center in dot_centers[1:]:
        if center - groups[-1][-1] > 55:
            groups.append([center])
        else:
            groups[-1].append(center)
    pairs = [len(group) for group in groups]
    if len(pairs) != 6 or any(value not in (1, 2, 3) for value in pairs):
        raise ValueError("six unambiguous comparator cards were not recovered")
    if sorted(pairs) != [1, 1, 2, 2, 3, 3]:
        raise ValueError("comparator-card multiset is invalid")
    direction_mask = _near(arr, _DIRECTION_COLOR)
    direction_ys, direction_xs = np.nonzero(direction_mask)
    direction_runs = [(lo, hi) for lo, hi in _runs(direction_xs) if hi - lo >= 45]
    if len(direction_runs) != 6:
        raise ValueError("six direction arrows were not recovered")
    sides: list[str] = []
    for lo, hi in direction_runs:
        mid = (lo + hi) // 2
        left_ink = int(direction_mask[:, lo : mid - 5].sum())
        right_ink = int(direction_mask[:, mid + 6 : hi + 1].sum())
        if abs(left_ink - right_ink) < 40:
            raise ValueError("direction arrow is ambiguous")
        sides.append("left" if left_ink > right_ink else "right")
    if sorted(sides) != ["left", "left", "left", "right", "right", "right"]:
        raise ValueError("direction-arrow multiset is invalid")
    return [{"pair": pair, "later_side": side} for pair, side in zip(pairs, sides, strict=True)]


def _parse(image: Image.Image) -> tuple[list[str], list[dict[str, Any]], int, int]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 400 or arr.shape[1] < 700:
        raise ValueError("raster is too small")
    boxes = _tile_boxes(arr)
    classified = [_classify_glyph(arr, box) for box in boxes]
    labels = [item[0] for item in classified]
    if sorted(labels) != ["A", "B", "C", "D"]:
        raise ValueError("tile labels are not a permutation of A-D")
    ring_y, ring_x = np.nonzero(_near(arr, _RING_COLOR))
    if len(ring_x) < 800:
        raise ValueError("queried-slot ring is missing")
    ring_center = float(ring_x.mean())
    centers = [(box[0] + box[2]) / 2 for box in boxes]
    distances = sorted((abs(center - ring_center), index) for index, center in enumerate(centers))
    if distances[1][0] - distances[0][0] < 40:
        raise ValueError("queried slot is ambiguous")
    return labels, _operations(arr), distances[0][1], min(item[1] for item in classified)


def decision_from_image(image: Image.Image) -> str:
    """Recover, replay, and answer using only visible pixels."""
    labels, operations, target_slot, glyph_margin = _parse(image)
    if glyph_margin < 3:
        raise ValueError("runner-up glyph margin is too small")
    state = list(labels)
    for operation in operations:
        left = int(operation["pair"]) - 1
        later_on_left = operation["later_side"] == "left"
        should_swap = (state[left] < state[left + 1]) if later_on_left else (state[left] > state[left + 1])
        if should_swap:
            state[left], state[left + 1] = state[left + 1], state[left]
    return state[target_slot]
