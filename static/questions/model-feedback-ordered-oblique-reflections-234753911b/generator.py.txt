"""Deterministic geometric world: four ordered point reflections."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 720
FIELD = (42.0, 126.0, 918.0, 676.0)
BG = (247, 247, 242)
INK = (35, 40, 48)
START = (244, 184, 38)
LINE_COLORS = [(206, 67, 62), (47, 104, 188), (39, 143, 91), (135, 78, 177)]
CUP_COLORS = {"A": (229, 119, 45), "B": (35, 163, 184), "C": (214, 72, 137), "D": (119, 91, 68)}
LABELS = ("A", "B", "C", "D")


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _reflect(point, line):
    px, py = float(point[0]), float(point[1])
    ax, ay = map(float, line["a"])
    bx, by = map(float, line["b"])
    dx, dy = bx - ax, by - ay
    scale = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
    qx, qy = ax + scale * dx, ay + scale * dy
    return [2.0 * qx - px, 2.0 * qy - py]


def _clip_line(mid, direction):
    x0, y0 = mid
    dx, dy = direction
    xmin, ymin, xmax, ymax = FIELD
    points = []
    if abs(dx) > 1e-9:
        for x in (xmin, xmax):
            t = (x - x0) / dx
            y = y0 + t * dy
            if ymin - 1e-7 <= y <= ymax + 1e-7:
                points.append((x, y))
    if abs(dy) > 1e-9:
        for y in (ymin, ymax):
            t = (y - y0) / dy
            x = x0 + t * dx
            if xmin - 1e-7 <= x <= xmax + 1e-7:
                points.append((x, y))
    unique = []
    for point in points:
        if not any(math.dist(point, prior) < 1e-5 for prior in unique):
            unique.append(point)
    if len(unique) != 2:
        raise ValueError("line did not clip field twice")
    return [list(unique[0]), list(unique[1])]


def _apply(lines, start, order):
    point = list(start)
    for index in order:
        point = _reflect(point, lines[index])
    return point


def _inside(point, pad=52.0):
    xmin, ymin, xmax, ymax = FIELD
    return xmin + pad <= point[0] <= xmax - pad and ymin + pad <= point[1] <= ymax - pad


def _point_line_distance(point, line):
    ax, ay = line["a"]
    bx, by = line["b"]
    return abs((bx - ax) * (ay - point[1]) - (ax - point[0]) * (by - ay)) / math.hypot(bx - ax, by - ay)


def _nearly_coincident(left, right):
    lax, lay = left["a"]
    lbx, lby = left["b"]
    rax, ray = right["a"]
    rbx, rby = right["b"]
    ldx, ldy = lbx-lax, lby-lay
    rdx, rdy = rbx-rax, rby-ray
    cross = abs(ldx*rdy-ldy*rdx) / (math.hypot(ldx,ldy)*math.hypot(rdx,rdy))
    separation = _point_line_distance(right["a"], left)
    return cross < 0.16 and separation < 28


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = LABELS[int(seed) % len(LABELS)]
    for _ in range(12000):
        current = [rng.uniform(180, 780), rng.uniform(210, 590)]
        start = list(current)
        lines = []
        good = True
        for number, color in enumerate(LINE_COLORS, 1):
            angle = rng.uniform(0, 2 * math.pi)
            step = rng.uniform(76, 136)
            nxt = [current[0] + step * math.cos(angle), current[1] + step * math.sin(angle)]
            if not _inside(nxt, 64):
                good = False
                break
            mid = [(current[0] + nxt[0]) / 2, (current[1] + nxt[1]) / 2]
            direction = [-math.sin(angle), math.cos(angle)]
            a, b = _clip_line(mid, direction)
            new_line = {"number": number, "color": list(color), "a": a, "b": b}
            if any(_nearly_coincident(new_line, prior) for prior in lines):
                good = False
                break
            lines.append(new_line)
            current = nxt
        if not good:
            continue
        variants = [
            _apply(lines, start, (0, 1, 2, 3)),
            _apply(lines, start, (3, 2, 1, 0)),
            _apply(lines, start, (0, 2, 1, 3)),
            _apply(lines, start, (0, 2, 3)),
        ]
        if not all(_inside(point, 42) for point in variants):
            continue
        if min(math.dist(variants[i], variants[j]) for i in range(4) for j in range(i)) < 78:
            continue
        if min(math.dist(start, point) for point in variants) < 68:
            continue
        if any(_point_line_distance(point, line) < 25 for point in variants for line in lines):
            continue
        distractors = variants[1:]
        rng.shuffle(distractors)
        cups = []
        cursor = 0
        for label in LABELS:
            point = variants[0] if label == answer else distractors[cursor]
            cursor += int(label != answer)
            cups.append({"label": label, "color": list(CUP_COLORS[label]), "center": point})
        return {"start": start, "lines": lines, "cups": cups}
    raise RuntimeError("could not sample a separated reflection scene")


def _sequence_point(scene):
    return _apply(scene["lines"], scene["start"], (0, 1, 2, 3))


def analytic_gold(scene: dict) -> str:
    endpoint = _sequence_point(scene)
    cups = {record["label"]: record["center"] for record in scene["cups"]}
    return min(cups, key=lambda label: math.dist(endpoint, cups[label]))


def margin(scene: dict) -> float:
    endpoint = _sequence_point(scene)
    distances = sorted(math.dist(endpoint, record["center"]) for record in scene["cups"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    endpoint = _sequence_point(scene)
    cup_points = [record["center"] for record in scene["cups"]]
    return (
        not all(_inside(point, 34) for point in [scene["start"], *cup_points])
        or min(math.dist(cup_points[i], cup_points[j]) for i in range(4) for j in range(i)) < 70
        or min(math.dist(endpoint, point) for point in cup_points) > 2
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cups"] = list(reversed(twin["cups"]))
    return [("reverse_label_keyed_cup_serialization", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 14, 942, 102), radius=18, fill=(255, 255, 255), outline=(180, 184, 188), width=2)
    draw.text((42, 30), "REFLECT THE BEAD", font=_font(25, True), fill=INK)
    draw.text((42, 64), "mirror order", font=_font(17), fill=(92, 98, 105))
    x = 225
    for index, color in enumerate(LINE_COLORS, 1):
        draw.line((x, 64, x + 38, 64), fill=color, width=9)
        draw.text((x + 47, 48), str(index), font=_font(24, True), fill=INK)
        if index < 4:
            draw.text((x + 78, 49), "→", font=_font(23, True), fill=(110, 115, 120))
        x += 128
    draw.rounded_rectangle(FIELD, radius=12, fill=(255, 255, 252), outline=(161, 166, 171), width=3)
    for record in sorted(scene["lines"], key=lambda item: item["number"]):
        a, b = tuple(record["a"]), tuple(record["b"])
        color = tuple(record["color"])
        draw.line((*a, *b), fill=color, width=9)
        t = 0.13 + 0.12 * (record["number"] % 3)
        bx = a[0] * (1 - t) + b[0] * t
        by = a[1] * (1 - t) + b[1] * t
        draw.ellipse((bx - 18, by - 18, bx + 18, by + 18), fill=(255, 255, 255), outline=INK, width=3)
        text = str(record["number"])
        box = draw.textbbox((0, 0), text, font=_font(21, True))
        draw.text((bx - (box[2]-box[0])/2, by - (box[3]-box[1])/2 - 2), text, font=_font(21, True), fill=INK)
    sx, sy = scene["start"]
    draw.ellipse((sx - 20, sy - 20, sx + 20, sy + 20), fill=(255,255,255), outline=INK, width=3)
    draw.ellipse((sx - 13, sy - 13, sx + 13, sy + 13), fill=START)
    for cup in sorted(scene["cups"], key=lambda item: item["label"]):
        cx, cy = cup["center"]
        draw.ellipse((cx - 28, cy - 28, cx + 28, cy + 28), fill=(255,255,255), outline=INK, width=3)
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=tuple(cup["color"]))
        label = cup["label"]
        box = draw.textbbox((0, 0), label, font=_font(22, True))
        draw.text((cx - (box[2]-box[0])/2, cy - (box[3]-box[1])/2 - 2), label, font=_font(22, True), fill=(255,255,255))
    return image
