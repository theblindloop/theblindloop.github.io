"""Image-only inverse for ordered oblique reflections."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

START_COLOR = np.array((244, 184, 38), dtype=np.float64)
LINE_COLORS = [np.array(c, dtype=np.float64) for c in ((206,67,62), (47,104,188), (39,143,91), (135,78,177))]
CUP_COLORS = {
    "A": np.array((229,119,45), dtype=np.float64),
    "B": np.array((35,163,184), dtype=np.float64),
    "C": np.array((214,72,137), dtype=np.float64),
    "D": np.array((119,91,68), dtype=np.float64),
}


def _mask(arr, color, tolerance=38.0):
    return np.linalg.norm(arr.astype(np.float64) - color, axis=2) <= tolerance


def _centroid(arr, color, minimum):
    ys, xs = np.nonzero(_mask(arr, color))
    if len(xs) < minimum:
        raise ValueError("required colored mark is missing")
    return np.array([xs.mean(), ys.mean()], dtype=np.float64)


def _fit_line(arr, color):
    ys, xs = np.nonzero(_mask(arr, color))
    # The header swatch is above y=110; only field pixels define the mirror.
    keep = ys >= 120
    pts = np.column_stack((xs[keep], ys[keep])).astype(np.float64)
    if len(pts) < 1500:
        raise ValueError("mirror line is missing or too short")
    center = pts.mean(axis=0)
    _, _, vt = np.linalg.svd(pts - center, full_matrices=False)
    direction = vt[0]
    residual = np.abs((pts - center) @ np.array([-direction[1], direction[0]]))
    if float(np.quantile(residual, 0.9)) > 7.0:
        raise ValueError("mirror is not a clean line")
    return center, direction


def _reflect(point, fitted):
    center, direction = fitted
    projection = center + np.dot(point - center, direction) * direction
    return 2 * projection - point


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.shape[:2] != (720, 960):
        raise ValueError("unexpected canvas size")
    point = _centroid(arr, START_COLOR, 300)
    for color in LINE_COLORS:
        point = _reflect(point, _fit_line(arr, color))
    cups = {label: _centroid(arr, color, 450) for label, color in CUP_COLORS.items()}
    distances = {label: float(np.linalg.norm(point - center)) for label, center in cups.items()}
    ordered = sorted(distances, key=distances.get)
    if distances[ordered[0]] > 8.0 or distances[ordered[1]] - distances[ordered[0]] < 45.0:
        raise ValueError("reflected point does not identify one cup")
    return ordered[0]
