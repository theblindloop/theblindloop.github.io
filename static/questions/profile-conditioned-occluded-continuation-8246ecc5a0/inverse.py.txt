"""Independent pixel-only inverse arm for the occluded-continuation world.

Recovers the whole symbol from the final PNG alone by performing amodal
continuation: for each image column, the visible symbol fragments are stitched
together vertically across the occluding bar bands, reconstructing the full
silhouette. The result is classified over {circle, square, diamond, triangle}
from reconstructed-silhouette geometry (bounding-box fill ratio and vertical
centroid offset).

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

_SYMBOL_COLOR = (30, 60, 140)
_NEAR = 20
_SQUARE_MIN = 0.90
_CIRCLE_MIN = 0.68
_OFFSET_MIN = 0.08


def _symbol_mask(arr: np.ndarray) -> np.ndarray:
    same = (
        (abs(arr[..., 0].astype(int) - _SYMBOL_COLOR[0]) <= _NEAR)
        & (abs(arr[..., 1].astype(int) - _SYMBOL_COLOR[1]) <= _NEAR)
        & (abs(arr[..., 2].astype(int) - _SYMBOL_COLOR[2]) <= _NEAR)
    )
    return same


def _continue(mask: np.ndarray) -> np.ndarray:
    """Fill each column between its topmost and bottommost visible pixel."""
    out = np.zeros_like(mask)
    for x in range(mask.shape[1]):
        ys = np.where(mask[:, x])[0]
        if ys.size == 0:
            continue
        out[ys.min(): ys.max() + 1, x] = True
    return out


def _features(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.where(mask)
    if xs.size == 0:
        return 0.0, 0.0
    h = int(ys.max()) - int(ys.min()) + 1
    w = int(xs.max()) - int(xs.min()) + 1
    fill = float(mask.sum()) / float(h * w)
    centre_y = (int(ys.min()) + int(ys.max())) / 2.0
    offset = abs(float(ys.mean()) - centre_y) / h
    return fill, offset


def decision_from_image(image: Image.Image) -> str:
    """Return the continued symbol shape recovered from the raster alone."""
    arr = np.asarray(image.convert("RGB")).astype(int)
    mask = _symbol_mask(arr)
    continued = _continue(mask)
    fill, offset = _features(continued)
    if fill >= _SQUARE_MIN:
        return "square"
    if fill >= _CIRCLE_MIN:
        return "circle"
    return "triangle" if offset > _OFFSET_MIN else "diamond"
