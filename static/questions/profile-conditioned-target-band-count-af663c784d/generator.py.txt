"""Concentric-target renderer.

One filled target is painted on white paper: a solid centre disc surrounded by
concentric rings of alternating colour.  The answer is how many of those solid
bands are actually painted, against the canonical prior that a target face has
five coloured zones.

Every latent field is painted into the raster, the ring count is the only field
that feeds analytic gold, and the outer radius, the two ink colours, the centre
position, the individual band thicknesses and the achromatic decoration are all
sampled independently of the count, so painted ink area, colour and page
position carry no information about the answer.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "target_band_count-renderer-0.1.0"

WIDTH = 620
HEIGHT = 620

N_MIN = 3
N_MAX = 7

MIN_BAND_PX = 12          # sampler floor: no sub-resolution band is ever painted
QUARANTINE_PX = 16.0      # near-boundary bands are quarantined
DECOR_GREY = (150, 150, 150)
PAPER = (255, 255, 255)

# Saturated, pairwise well separated ink colours.
PALETTE: dict[str, tuple[int, int, int]] = {
    "crimson": (198, 26, 46),
    "teal": (0, 128, 132),
    "navy": (30, 52, 168),
    "orange": (224, 116, 12),
    "forest": (24, 122, 48),
    "purple": (126, 34, 156),
    "magenta": (196, 30, 140),
    "olive": (118, 118, 20),
}


def _dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return float(np.sqrt(sum((int(x) - int(y)) ** 2 for x, y in zip(a, b))))


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene.

    The band count is a balanced deterministic function of the seed, so the five
    classes are exactly equifrequent over consecutive generator scenes and the
    canonical five-band class is a 1-in-5 minority.
    """
    rng = random.Random(seed)
    n_bands = N_MIN + (seed % 5)

    outer_r = rng.randint(215, 255)          # sampled independently of n_bands
    thin = rng.randint(MIN_BAND_PX, 26)      # the band that carries the margin
    thin_idx = rng.randrange(n_bands)

    widths = [thin + 4] * n_bands
    widths[thin_idx] = thin
    leftover = outer_r - sum(widths)
    others = [i for i in range(n_bands) if i != thin_idx]
    for _ in range(leftover):
        widths[rng.choice(others)] += 1

    names = list(PALETTE)
    first = rng.choice(names)
    second = rng.choice([c for c in names if _dist(PALETTE[c], PALETTE[first]) >= 120.0])

    cx = WIDTH // 2 + rng.randint(-18, 18)
    cy = HEIGHT // 2 + rng.randint(-18, 18)
    decor = [rng.randint(0, 14) for _ in range(4)]

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cx": cx,
        "cy": cy,
        "n_bands": n_bands,
        "widths": widths,          # centre band first, outward
        "colors": [first, second],  # bands alternate first, second, first, ...
        "decor": decor,
    }


def _band_widths(scene: dict) -> list[int]:
    """The thicknesses actually painted: n_bands of them, cycling the list."""
    ws = [int(w) for w in scene["widths"]]
    n = int(scene["n_bands"])
    return [ws[k % len(ws)] for k in range(n)]


def _radii(scene: dict) -> list[int]:
    out, acc = [], 0
    for w in _band_widths(scene):
        acc += int(w)
        out.append(acc)
    return out


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (int(scene["width"]), int(scene["height"])), PAPER)
    d = ImageDraw.Draw(img)

    w, h = int(scene["width"]), int(scene["height"])
    d.rectangle([12, 12, w - 13, h - 13], outline=DECOR_GREY, width=3)
    o = [int(v) for v in scene["decor"]]
    d.line([30 + o[0], 30, 30 + o[0], 54], fill=DECOR_GREY, width=3)
    d.line([w - 31 - o[1], 30, w - 31 - o[1], 54], fill=DECOR_GREY, width=3)
    d.line([30 + o[2], h - 55, 30 + o[2], h - 31], fill=DECOR_GREY, width=3)
    d.line([w - 31 - o[3], h - 55, w - 31 - o[3], h - 31], fill=DECOR_GREY, width=3)

    cx, cy = int(scene["cx"]), int(scene["cy"])
    pal = [PALETTE[c] for c in scene["colors"]]
    radii = _radii(scene)
    for k in range(len(radii) - 1, -1, -1):   # outermost band first
        r = radii[k]
        d.ellipse([cx - r, cy - r, cx + r, cy + r], fill=pal[k % 2])
    return img


def analytic_gold(scene: dict) -> str:
    return str(int(scene["n_bands"]))


def margin(scene: dict) -> float:
    """Thickness in pixels of the thinnest painted band.

    That band is the whole evidence separating the count from the runner-up
    count one lower: when its thickness reaches zero it stops being painted and
    its two neighbours merge into a single band.
    """
    return float(min(_band_widths(scene)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every field is painted and recoverable from the PNG."""
    return []


# --- gold-preserving nuisance perturbations -------------------------------


def translate(scene: dict, dx: int, dy: int) -> dict:
    out = dict(scene)
    out["cx"] = int(scene["cx"]) + int(dx)
    out["cy"] = int(scene["cy"]) + int(dy)
    return out


def recolor(scene: dict) -> dict:
    out = dict(scene)
    names = list(PALETTE)
    a = names[(names.index(scene["colors"][0]) + 3) % len(names)]
    b = next(
        c for c in names[::-1]
        if _dist(PALETTE[c], PALETTE[a]) >= 120.0 and c != a
    )
    out["colors"] = [a, b]
    return out


def permute_decor(scene: dict, seed: int) -> dict:
    out = dict(scene)
    rng = random.Random(int(seed) + 7)
    out["decor"] = [rng.randint(0, 14) for _ in range(4)]
    return out


def rescale(scene: dict) -> dict:
    """Shrink every band by the same factor: the count cannot change."""
    out = dict(scene)
    out["widths"] = [max(MIN_BAND_PX, int(round(int(w) * 0.8))) for w in scene["widths"]]
    return out


# --- declared label-transforming edits -------------------------------------


def set_band_count(scene: dict, n: int) -> dict:
    """Repaint the same target with n equally thick bands: gold becomes n."""
    out = dict(scene)
    total = sum(int(w) for w in scene["widths"])
    base = total // int(n)
    widths = [base] * int(n)
    widths[-1] += total - base * int(n)
    out["n_bands"] = int(n)
    out["widths"] = widths
    return out


def drop_thinnest(scene: dict) -> dict:
    """Collapse the thinnest band to nothing: the count falls to the runner-up."""
    widths = [int(w) for w in scene["widths"]]
    k = widths.index(min(widths))
    lost = widths.pop(k)
    if not widths:
        return dict(scene)
    widths[min(k, len(widths) - 1)] += lost
    out = dict(scene)
    out["widths"] = widths
    out["n_bands"] = len(widths)
    return out


def ink_pixels(image: Image.Image) -> int:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    return int((chroma >= 40).sum())
