"""Independent pixel-only inverse arm.

Receives the final PNG and nothing else: no latent scene, no renderer, no
prompts, no analytic gold, no filenames and no metadata.  It recovers the number
of painted bands by walking rays outwards from the centre of the chromatic ink
and counting solid colour runs, i.e. by measuring radial nesting depth directly
in the raster.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "target_band_count-oracle-0.1.0"
ABSTAIN = "abstain"

_CHROMA_MIN = 40          # separates ink from white paper and grey decoration
_COLOR_STEP = 45          # colour distance that counts as a band boundary
_MIN_RUN_PX = 4           # ignore stair-stepping at a jagged circle edge
_MIN_INK_PX = 2000
_N_RAYS = 72
_AGREE_FRAC = 0.6
_LO, _HI = 2, 9           # readings outside this range are not a target


def _ink_mask(arr: np.ndarray) -> np.ndarray:
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    return chroma >= _CHROMA_MIN


def _ray_runs(arr: np.ndarray, mask: np.ndarray, cy: int, cx: int,
              dy: float, dx: float, r_max: int) -> int:
    h, w = mask.shape
    steps = np.arange(0, r_max + 1)
    ys = np.rint(cy + dy * steps).astype(np.int64)
    xs = np.rint(cx + dx * steps).astype(np.int64)
    inside = (ys >= 0) & (ys < h) & (xs >= 0) & (xs < w)
    if not inside[0]:
        return 0
    stop = int(np.argmin(inside)) if not inside.all() else len(steps)
    ys, xs = ys[:stop], xs[:stop]
    ok = mask[ys, xs]
    if ok.size == 0 or not ok[0]:
        return 0
    end = int(np.argmin(ok)) if not ok.all() else ok.size
    cols = arr[ys[:end], xs[:end]].astype(np.int32)
    if cols.shape[0] < _MIN_RUN_PX:
        return 0

    # split the ray into maximal runs of one colour
    diff = np.sqrt(((cols[1:] - cols[:-1]) ** 2).sum(axis=1))
    bounds = [0] + [int(i) + 1 for i in np.nonzero(diff > _COLOR_STEP)[0]] + [cols.shape[0]]
    runs = [(bounds[i], bounds[i + 1]) for i in range(len(bounds) - 1)]
    return sum(1 for a, b in runs if b - a >= _MIN_RUN_PX)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = _ink_mask(arr)
    if int(mask.sum()) < _MIN_INK_PX:
        return ABSTAIN

    ys, xs = np.nonzero(mask)
    cy, cx = int(round(ys.mean())), int(round(xs.mean()))
    r_max = int(max(ys.max() - ys.min(), xs.max() - xs.min())) + 4

    counts: dict[int, int] = {}
    for k in range(_N_RAYS):
        th = 2.0 * np.pi * k / _N_RAYS
        n = _ray_runs(arr, mask, cy, cx, float(np.sin(th)), float(np.cos(th)), r_max)
        counts[n] = counts.get(n, 0) + 1

    best = max(counts, key=lambda n: (counts[n], -n))
    if counts[best] < _AGREE_FRAC * _N_RAYS:
        return ABSTAIN
    if not (_LO <= best <= _HI):
        return ABSTAIN
    return str(best)
