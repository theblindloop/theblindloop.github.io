"""Independent image-only inverse arm for repeated-block visual search.

The oracle knows only the public visual grammar: four horizontal bands, eight
chromatic tiles per band, and a four-way top-to-bottom answer alphabet. It does
not use renderer coordinates, colors, scene data, filenames, or metadata.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(flags)
    if len(indices) == 0:
        return []
    cuts = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[0, cuts + 1]
    ends = np.r_[cuts, len(indices) - 1]
    return [(int(indices[a]), int(indices[b]) + 1) for a, b in zip(starts, ends, strict=True)]


def _cluster_colors(colors: list[np.ndarray]) -> list[int]:
    centers: list[np.ndarray] = []
    labels: list[int] = []
    for color in colors:
        if not centers:
            centers.append(color)
            labels.append(0)
            continue
        distances = [float(np.linalg.norm(color - center)) for center in centers]
        nearest = int(np.argmin(distances))
        if distances[nearest] <= 28.0:
            labels.append(nearest)
        else:
            centers.append(color)
            labels.append(len(centers) - 1)
    if len(centers) != 4:
        raise ValueError(f"expected four tile colors, recovered {len(centers)}")
    return labels


def _read_rows(image: Image.Image) -> list[list[int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = arr.max(axis=2)
    low = arr.min(axis=2)
    chromatic = (high - low > 55) & (high < 248) & (low < 225)

    y_bands = _runs(chromatic.sum(axis=1) >= 120)
    y_bands = [(a, b) for a, b in y_bands if b - a >= 25]
    if len(y_bands) != 4:
        raise ValueError(f"expected four colored rows, recovered {len(y_bands)}")

    boxes: list[tuple[int, int, int, int]] = []
    for y0, y1 in y_bands:
        min_column_ink = max(12, int((y1 - y0) * 0.55))
        x_bands = _runs(chromatic[y0:y1].sum(axis=0) >= min_column_ink)
        x_bands = [(a, b) for a, b in x_bands if b - a >= 25]
        if len(x_bands) != 8:
            raise ValueError(f"expected eight tiles in a row, recovered {len(x_bands)}")
        boxes.extend((x0, y0, x1, y1) for x0, x1 in x_bands)

    colors: list[np.ndarray] = []
    for x0, y0, x1, y1 in boxes:
        inset_x = max(3, (x1 - x0) // 5)
        inset_y = max(3, (y1 - y0) // 5)
        crop = arr[y0 + inset_y:y1 - inset_y, x0 + inset_x:x1 - inset_x]
        if crop.size == 0:
            raise ValueError("tile interior vanished")
        colors.append(np.median(crop.reshape(-1, 3), axis=0).astype(np.float64))

    ids = _cluster_colors(colors)
    rows = [ids[index:index + 8] for index in range(0, 32, 8)]
    expected = {0, 1, 2, 3}
    for row in rows:
        if set(row[:4]) != expected or set(row[4:]) != expected:
            raise ValueError("each half-row must contain every recovered color exactly once")
        if any(row.count(color) != 2 for color in expected):
            raise ValueError("row color inventory is not balanced")
    return rows


def decision_from_image(image: Image.Image) -> str:
    rows = _read_rows(image)
    scores = [sum(a == b for a, b in zip(row[:4], row[4:], strict=True)) for row in rows]
    winners = [index for index, score in enumerate(scores) if score == 4]
    if len(winners) != 1:
        raise ValueError(f"expected one exact repeated block, scores={scores}")
    ordered = sorted(scores, reverse=True)
    if ordered[0] <= ordered[1]:
        raise ValueError(f"ambiguous winner, scores={scores}")
    return LABELS[winners[0]]
