"""Deterministic renderer for repeated-block visual search."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 520
LABELS = ("A", "B", "C", "D")
SYMBOLS = ("w", "x", "y", "z")
BASE_COLORS = (
    (42, 104, 198),   # blue
    (232, 116, 36),   # orange
    (18, 154, 126),   # teal
    (139, 82, 183),   # purple
)

PANEL_Y = (70, 180, 290, 400)
PANEL_X0 = 75
PANEL_X1 = 642
PANEL_H = 80
TILE_X0 = 105
TILE_Y_INSET = 18
TILE_SIZE = 44
TILE_PITCH = 57
BLOCK_GAP = 30


def _second_block(first: list[str], matches: int, rng: random.Random) -> list[str]:
    """Return a permutation of ``first`` with exactly ``matches`` fixed positions."""
    if matches == 4:
        return list(first)
    if matches == 2:
        out = list(first)
        i, j = rng.sample(range(4), 2)
        out[i], out[j] = out[j], out[i]
        return out
    if matches == 1:
        fixed = rng.randrange(4)
        moving = [i for i in range(4) if i != fixed]
        if rng.random() < 0.5:
            moving = [moving[0], moving[2], moving[1]]
        out = list(first)
        for src, dst in zip(moving, moving[1:] + moving[:1], strict=True):
            out[dst] = first[src]
        return out
    if matches == 0:
        shift = rng.choice((1, 3))
        return [first[(i + shift) % 4] for i in range(4)]
    raise ValueError(f"unsupported match count {matches}")


def _scores(rows: Iterable[Iterable[str]]) -> list[int]:
    values = [list(row) for row in rows]
    if len(values) != 4 or any(len(row) != 8 for row in values):
        raise ValueError("scene must contain four rows of eight tiles")
    return [sum(a == b for a, b in zip(row[:4], row[4:], strict=True)) for row in values]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. The answer position cycles exactly with the seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    distractor_scores = [2, 1, 0]
    rng.shuffle(distractor_scores)
    assigned: list[int] = []
    for index in range(4):
        assigned.append(4 if index == winner else distractor_scores.pop())

    rows: list[list[str]] = []
    for score in assigned:
        first = list(SYMBOLS)
        rng.shuffle(first)
        rows.append(first + _second_block(first, score, rng))

    colors = list(BASE_COLORS)
    rng.shuffle(colors)
    palette = {symbol: list(color) for symbol, color in zip(SYMBOLS, colors, strict=True)}
    return {
        "rows": rows,
        "palette": palette,
        "offset_x": rng.randint(-7, 7),
        "offset_y": rng.randint(-5, 5),
    }


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene["rows"])
    winners = [index for index, score in enumerate(scores) if score == 4]
    if len(winners) != 1:
        raise ValueError("scene has no unique exact repeated block")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene["rows"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = _scores(scene["rows"])
    except ValueError:
        return True
    return scores.count(4) != 1 or sorted(scores, reverse=True)[0] <= sorted(scores, reverse=True)[1]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)


def _tile_x(index: int) -> int:
    return TILE_X0 + index * TILE_PITCH + (BLOCK_GAP if index >= 4 else 0)


def render(scene: dict) -> Image.Image:
    rows = scene["rows"]
    palette = scene["palette"]
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    _scores(rows)
    if set(palette) != set(SYMBOLS):
        raise ValueError("palette must define the four symbols")

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    label_font = _font(27 * scale)
    small_font = _font(15 * scale)

    draw.text((205 * scale, (18 + oy) * scale), "FIRST BLOCK", fill=(82, 82, 82),
              font=small_font, anchor="mm")
    draw.text((482 * scale, (18 + oy) * scale), "SECOND BLOCK", fill=(82, 82, 82),
              font=small_font, anchor="mm")

    for row_index, row in enumerate(rows):
        py = PANEL_Y[row_index] + oy
        draw.rounded_rectangle(
            ((PANEL_X0 + ox) * scale, py * scale,
             (PANEL_X1 + ox) * scale, (py + PANEL_H) * scale),
            radius=10 * scale,
            fill=(255, 255, 255),
            outline=(188, 190, 194),
            width=2 * scale,
        )
        draw.text(((45 + ox) * scale, (py + PANEL_H / 2) * scale), LABELS[row_index],
                  fill=(28, 31, 36), font=label_font, anchor="mm")
        divider_x = 342 + ox
        draw.line((divider_x * scale, (py + 10) * scale,
                   divider_x * scale, (py + PANEL_H - 10) * scale),
                  fill=(205, 207, 211), width=2 * scale)
        ty = py + TILE_Y_INSET
        for tile_index, symbol in enumerate(row):
            tx = _tile_x(tile_index) + ox
            color = tuple(int(value) for value in palette[symbol])
            draw.rounded_rectangle(
                (tx * scale, ty * scale, (tx + TILE_SIZE) * scale, (ty + TILE_SIZE) * scale),
                radius=5 * scale,
                fill=color,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def with_palette_permutation(scene: dict, permutation: tuple[int, int, int, int]) -> dict:
    """Public nuisance transform used by tests; it changes pixels but preserves the label."""
    twin = copy.deepcopy(scene)
    old = [tuple(scene["palette"][symbol]) for symbol in SYMBOLS]
    twin["palette"] = {
        symbol: list(old[permutation[index]]) for index, symbol in enumerate(SYMBOLS)
    }
    return twin
