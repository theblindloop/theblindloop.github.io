"""Independent pixel-only inverse arm for the normalized fractional-location
correspondence world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (this file's own specification):

  * the image contains four horizontal ``track``+``marker`` panels stacked top
    to bottom in horizontal bands; the top one is the ``reference`` panel, the
    next three are candidate panels A, B, C;
  * each panel is a thin light-gray horizontal line (the track) of some length,
    with a single solid filled disk (the marker) sitting on it; the marker is
    dark gray on the reference panel and a saturated color (red, green, blue)
    on the candidate panels;
  * the marker sits at a normalized fractional position along its own track;
  * the task: decide which candidate marker sits at the same normalized
    fractional position (marker center x minus track-left, divided by track
    length) as the reference marker.

The oracle recovers, from pixels alone: each panel's track left/right endpoints
from the gray track line, the marker center x from the non-track mark, the
normalized fraction for every panel, and finally which candidate share the
reference's fraction.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "track_fraction_correspondence-oracle-0.1.0"

_WHITE_DIST = 35.0
_TRACK_DIST = 35.0
_MIN_PX = 300
_TOL = 0.08
_TRACK_RGB = np.array([200.0, 200.0, 200.0])
_WHITE = np.array([255.0, 255.0, 255.0])


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    comps = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(x, y)])
                visited[y, x] = True
                cur = []
                while queue:
                    cx, cy = queue.popleft()
                    cur.append((cx, cy))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx_, ny_ = cx + dx, cy + dy
                        if 0 <= nx_ < w and 0 <= ny_ < h and mask[ny_, nx_] and not visited[ny_, nx_]:
                            visited[ny_, nx_] = True
                            queue.append((nx_, ny_))
                comps.append(np.array(cur, dtype=np.intp))
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    white_dist = np.linalg.norm(arr - _WHITE, axis=-1)
    track_dist = np.linalg.norm(arr - _TRACK_RGB, axis=-1)
    non_bg = ~(white_dist < _WHITE_DIST)

    comps = [c for c in _connected_components(non_bg) if len(c) >= _MIN_PX]
    if len(comps) != 4:
        raise ValueError(f"expected 4 panels, got {len(comps)}")

    # Separate each panel into its gray track line (endpoints) and its marker.
    track_pix = track_dist[non_bg] < _TRACK_DIST

    fractions = []
    centers = []
    for comp in comps:
        pts = comp
        mask = np.zeros(non_bg.shape, dtype=bool)
        mask[pts[:, 1], pts[:, 0]] = True
        comp_track = mask & (track_dist < _TRACK_DIST)
        comp_marker = mask & ~(track_dist < _TRACK_DIST)
        if not comp_track.any() or not comp_marker.any():
            raise ValueError("panel is missing a separable track or marker")
        ty, tx = np.nonzero(comp_track)
        my, mx = np.nonzero(comp_marker)
        left = float(tx.min())
        right = float(tx.max())
        length = right - left
        if length <= 0:
            raise ValueError("degenerate track length")
        marker_center = float(mx.mean())
        fractions.append((marker_center - left) / length)
        centers.append(float(comp.mean(axis=0)[1]))

    order = sorted(range(4), key=lambda i: centers[i])
    by_y = [fractions[i] for i in order]
    ref_frac = by_y[0]
    for k in range(1, 4):
        if abs(by_y[k] - ref_frac) < _TOL:
            return "ABC"[k - 1]
    raise ValueError("no candidate reproduced the reference marker fraction")


if __name__ == "__main__":
    import sys

    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
