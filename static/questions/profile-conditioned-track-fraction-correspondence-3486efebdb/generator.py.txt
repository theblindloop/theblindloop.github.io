"""Cross-panel normalized fractional-location correspondence renderer.

Four panels are stacked vertically: the top ``reference`` panel, then three
candidate panels labeled A, B, C. Every panel shows one horizontal ``track`` (a
thin gray horizontal line of some length) with a single ``marker`` (a filled
disk) sitting directly on it. The marker is placed at a *fractional* position
along its own track: fraction ``p = (marker_x - track_left) / track_length``,
where ``p`` takes one of three discrete class values {0.25, 0.50, 0.75}
(one-quarter, one-half, three-quarters along the track).

The reference panel's marker sits at a randomized fraction ``p_ref``. Exactly
one candidate (the ``match`` slot, randomized to A, B, or C) places its marker
at the *same* normalized fraction ``p_ref``; the other two candidates place
their markers at the two remaining distinct fractions. The task asks which
candidate's marker sits at the same fractional position along its own track as
the reference's marker.

Crucially, every panel has a *different, randomized* track length and a
different random horizontal track offset, so the absolute pixel x of a marker
carries no information: only the marker's position normalized by its own track
length identifies the matching candidate. This deliberately breaks the cheap
"absolute position" and "size / total ink" cues and forces a proportional
(relative-location) read.

Latent state z (every field is rendered - none hidden):

    panels : ordered list of four panel dicts [reference, A, B, C]; each
             {left: int track-left x, length: int track pixel length,
              fraction: one of {0.25, 0.5, 0.75}, yc: int band center}.
    width  : raster width.
    height : raster height.

The analytic gold is the label (A/B/C) of the candidate whose normalized marker
fraction equals the reference's. Because the track's length and left offset, and
the marker's centered disk, are all drawn verbatim, the normalized fraction is
recoverable from pixels alone (marker center x minus track-left, divided by
track length); no gold-only field exists.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "track_fraction_correspondence-0.1.0"

BACKGROUND = (255, 255, 255)
TRACK_COLOR = (200, 200, 200)
REF_MARKER_COLOR = (55, 55, 55)

CANDIDATE_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}

WIDTH = 900
HEIGHT = 1000
BAND_CY = [160, 400, 640, 880]   # reference, A, B, C top to bottom
TRACK_THICKNESS = 5
MARKER_RADIUS = 16
MIN_LENGTH = 340
MAX_LENGTH = 460
MIN_LEFT = 90
MAX_LEFT = 170
FRACTIONS = [0.25, 0.5, 0.75]     # three discrete normalized classes
_MARGIN_FLOOR = 30.0


def _class_fraction(fraction: float) -> float:
    best = min(FRACTIONS, key=lambda f: abs(f - fraction))
    return best


def _build_panel(rng: random.Random, yc: int, fraction: float) -> dict:
    length = rng.randint(MIN_LENGTH, MAX_LENGTH)
    left = rng.randint(MIN_LEFT, MAX_LEFT)
    return {
        "left": left,
        "length": length,
        "fraction": _class_fraction(fraction),
        "yc": yc,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref_frac = rng.choice(FRACTIONS)
    match_slot = rng.randrange(3)
    other = [f for f in FRACTIONS if f != ref_frac]
    # randomize which two distinct fractions the non-matching candidates use
    rng.shuffle(other)
    cand_fracs = []
    for i in range(3):
        if i == match_slot:
            cand_fracs.append(ref_frac)
        else:
            cand_fracs.append(other.pop())
    panels = [_build_panel(rng, BAND_CY[0], ref_frac)]
    for i in range(3):
        panels.append(_build_panel(rng, BAND_CY[i + 1], cand_fracs[i]))
    return {
        "panels": panels,
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for i, panel in enumerate(scene["panels"]):
        if i == 0:
            marker_color = REF_MARKER_COLOR
        else:
            marker_color = CANDIDATE_PALETTE[["red", "green", "blue"][i - 1]]
        left = panel["left"]
        right = left + panel["length"]
        yc = panel["yc"]
        draw.line((left, yc, right, yc), fill=TRACK_COLOR, width=TRACK_THICKNESS)
        mx = left + panel["fraction"] * panel["length"]
        draw.ellipse(
            (mx - MARKER_RADIUS, yc - MARKER_RADIUS, mx + MARKER_RADIUS, yc + MARKER_RADIUS),
            fill=marker_color,
        )
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate whose normalized marker
    fraction equals the reference panel's marker fraction."""
    panels = scene["panels"]
    ref_frac = panels[0]["fraction"]
    for i in range(1, 4):
        if panels[i]["fraction"] == ref_frac:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference marker fraction")


def margin(scene: dict) -> float:
    """One-sided verifiability margin: minimum, over every panel, of the pixel
    distance from that panel's marker center to the nearest fraction-class
    boundary within its track. Class centers are 0.25 apart, so each marker is
    0.125 * track_length from a boundary; with length >= MIN_LENGTH the margin
    is always at least 0.125 * 340 = 42.5 pixels.""" 
    panels = scene["panels"]
    return float(min(0.125 * p["length"] for p in panels))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous class boundary. Fraction classes
    are 0.25 apart with boundaries at the midpoints (0.375, 0.625), so a marker
    is at least 0.125 * length >= 42.5 pixels clear of a boundary; the sampler
    never produces an ambiguous scene at this resolution."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field, the marker fraction of
    each panel (encoded by the marker's normalized position along the drawn
    track), is rendered verbatim, so the raster uniquely determines each
    panel's normalized fraction and the matching candidate; no pixel-identical
    transform is declared."""
    return []
