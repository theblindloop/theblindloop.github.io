"""Independent image-only inverse for the checkpoint-rank figure.

The inverse intentionally contains no renderer import and receives no scene.
It identifies A/B/C by the visible legend swatches, samples each colored marker
at the six fixed labeled checkpoints, and returns the unique middle-rank label.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1000
HEIGHT = 650
PLOT_TOP = 165
PLOT_BOTTOM = 555
CHECKPOINTS = (180, 329, 478, 627, 776, 925)
LEGEND_ROWS = {"A": 52, "B": 84, "C": 116}
ROLES = ("A", "B", "C")


def _legend_colors(arr: np.ndarray) -> dict[str, np.ndarray]:
    colors: dict[str, np.ndarray] = {}
    for role, row in LEGEND_ROWS.items():
        # The center of each visible swatch is a solid pixel, unlike the nearby
        # antialiased text.  A small crop is used to remain stable to PNG edits.
        crop = arr[row - 6 : row + 7, 84:105]
        colors[role] = crop[crop.shape[0] // 2, crop.shape[1] // 2].astype(np.float32)
    if len({tuple(color.astype(int)) for color in colors.values()}) != 3:
        raise ValueError("legend does not expose three distinct trace colors")
    return colors


def _checkpoint_y(mask: np.ndarray, x: int) -> float:
    x0 = max(0, x - 6)
    x1 = min(mask.shape[1], x + 7)
    crop = mask[PLOT_TOP : PLOT_BOTTOM + 1, x0:x1]
    ys = np.nonzero(crop)[0] + PLOT_TOP
    if len(ys) < 8:
        raise ValueError(f"trace marker missing at x={x}")
    return float(np.median(ys))


def decision_from_image(image: Image.Image) -> str:
    """Recover the checkpoint where legend trace A is vertically between B/C."""

    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected image size {arr.shape[:2][::-1]}")
    colors = _legend_colors(arr)
    heights = {role: [] for role in ROLES}
    for role in ROLES:
        distance = np.linalg.norm(arr - colors[role], axis=-1)
        mask = distance < 18.0
        heights[role] = [_checkpoint_y(mask, x) for x in CHECKPOINTS]

    found = []
    for index in range(len(CHECKPOINTS)):
        a = heights["A"][index]
        b = heights["B"][index]
        c = heights["C"][index]
        if min(b, c) < a < max(b, c):
            found.append(index)
    if len(found) != 1:
        raise ValueError(f"expected one middle-rank checkpoint, recovered {len(found)}")
    return f"T{found[0] + 1}"
