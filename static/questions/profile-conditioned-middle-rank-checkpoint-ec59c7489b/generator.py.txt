"""Renderer and analytic world for a three-trace checkpoint-rank question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "middle_rank_checkpoint-0.1.0"
WIDTH = 1000
HEIGHT = 650
BACKGROUND = (247, 249, 246)
PLOT_FILL = (255, 255, 255)
AXIS_INK = (54, 64, 72)
GRID_INK = (221, 227, 224)
BASELINE_INK = (117, 126, 132)
PLOT_LEFT = 140
PLOT_RIGHT = 950
PLOT_TOP = 165
PLOT_BOTTOM = 555
PLOT_HEIGHT = PLOT_BOTTOM - PLOT_TOP
CHECKPOINTS = (180, 329, 478, 627, 776, 925)
LEGEND_ROWS = {"A": 52, "B": 84, "C": 116}
PALETTE = (
    (211, 69, 83),
    (31, 142, 154),
    (116, 82, 177),
    (224, 128, 34),
    (45, 103, 185),
    (55, 150, 91),
)
ROLES = ("A", "B", "C")
MIN_GAP_PX = 24.0


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, legible scene.

    The seed residue selects the answer checkpoint.  Absolute levels, the
    above/below status at non-answer checkpoints, palette assignment, baseline,
    and marker radius are independently varied so the answer is not a color or
    total-ink cue.
    """

    rng = random.Random(int(seed))
    winner = int(seed) % len(CHECKPOINTS)
    levels = [rng.uniform(0.46, 0.54) for _ in CHECKPOINTS]
    values = {role: [] for role in ROLES}
    for index, base in enumerate(levels):
        if index == winner:
            offsets = (0.0, 0.20, -0.20)  # A is strictly between B and C.
        elif (index + int(seed)) % 2:
            offsets = (0.22, -0.20, -0.05)  # A is above both.
        else:
            offsets = (-0.22, 0.20, 0.05)  # A is below both.
        for role, offset in zip(ROLES, offsets, strict=True):
            values[role].append(round(base + offset + rng.uniform(-0.012, 0.012), 4))

    palette_values = rng.sample(list(PALETTE), k=3)
    palette = {role: list(color) for role, color in zip(ROLES, palette_values, strict=True)}
    return {
        "a_values": values["A"],
        "b_values": values["B"],
        "c_values": values["C"],
        "palette": palette,
        "baseline": round(rng.uniform(0.08, 0.15), 4),
        "point_radius": 7 + rng.randrange(3),
    }


def _y_for_value(value: float) -> int:
    return round(PLOT_BOTTOM - float(value) * PLOT_HEIGHT)


def _draw_dashed_horizontal(
    draw: ImageDraw.ImageDraw,
    y: int,
    fill: tuple[int, int, int],
    width: int = 2,
) -> None:
    for start in range(PLOT_LEFT + 8, PLOT_RIGHT - 8, 24):
        draw.line((start, y, min(start + 12, PLOT_RIGHT - 8), y), fill=fill, width=width)


def _draw_legend(draw: ImageDraw.ImageDraw, palette: dict[str, Any]) -> None:
    font = _font()
    draw.text((48, 22), "Trace identity", fill=AXIS_INK, font=font)
    for role in ROLES:
        y = LEGEND_ROWS[role]
        color = tuple(int(value) for value in palette[role])
        draw.rectangle((78, y - 8, 110, y + 8), fill=color)
        draw.text((124, y - 6), role, fill=AXIS_INK, font=font)


def _draw_plot_frame(draw: ImageDraw.ImageDraw, baseline: float) -> None:
    font = _font()
    draw.rectangle((PLOT_LEFT, PLOT_TOP, PLOT_RIGHT, PLOT_BOTTOM), fill=PLOT_FILL, outline=AXIS_INK, width=2)
    for level in (0.2, 0.4, 0.6, 0.8):
        y = _y_for_value(level)
        draw.line((PLOT_LEFT, y, PLOT_RIGHT, y), fill=GRID_INK, width=1)
    _draw_dashed_horizontal(draw, _y_for_value(baseline), BASELINE_INK, width=2)
    draw.text((PLOT_LEFT - 2, PLOT_BOTTOM + 22), "height", fill=AXIS_INK, font=font)
    for index, x in enumerate(CHECKPOINTS, start=1):
        draw.line((x, PLOT_TOP, x, PLOT_BOTTOM), fill=(239, 242, 240), width=1)
        draw.text((x - 9, PLOT_BOTTOM + 8), f"T{index}", fill=AXIS_INK, font=font)


def _draw_trace(
    draw: ImageDraw.ImageDraw,
    values: list[float],
    color: tuple[int, int, int],
    point_radius: int,
) -> None:
    points = [(x, _y_for_value(value)) for x, value in zip(CHECKPOINTS, values, strict=True)]
    draw.line(points, fill=color, width=5)
    for x, y in points:
        draw.ellipse(
            (x - point_radius, y - point_radius, x + point_radius, y + point_radius),
            fill=color,
            outline=BACKGROUND,
            width=1,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_legend(draw, scene["palette"])
    _draw_plot_frame(draw, float(scene["baseline"]))
    trace_data = (
        ("A", scene["a_values"]),
        ("B", scene["b_values"]),
        ("C", scene["c_values"]),
    )
    for role, values in trace_data:
        _draw_trace(
            draw,
            [float(value) for value in values],
            tuple(int(value) for value in scene["palette"][role]),
            int(scene["point_radius"]),
        )
    return image


def _middle_indices(scene: dict[str, Any]) -> list[int]:
    a_values = scene["a_values"]
    b_values = scene["b_values"]
    c_values = scene["c_values"]
    found: list[int] = []
    for index in range(len(CHECKPOINTS)):
        a = float(a_values[index])
        b = float(b_values[index])
        c = float(c_values[index])
        if min(b, c) < a < max(b, c):
            found.append(index)
    return found


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique checkpoint where A has the middle height."""

    found = _middle_indices(scene)
    if len(found) != 1:
        return "quarantine"
    return f"T{found[0] + 1}"


def margin(scene: dict[str, Any]) -> float:
    """Smallest visible vertical separation between any pair at any checkpoint."""

    a_values = scene["a_values"]
    b_values = scene["b_values"]
    c_values = scene["c_values"]
    gaps = []
    for index in range(len(CHECKPOINTS)):
        values = [float(a_values[index]), float(b_values[index]), float(c_values[index])]
        gaps.extend(abs(left - right) * PLOT_HEIGHT for left, right in ((values[0], values[1]), (values[0], values[2]), (values[1], values[2])))
    return min(gaps) if gaps else -1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    values = [
        float(value)
        for role in ROLES
        for value in scene[f"{role.lower()}_values"]
    ]
    return (
        analytic_gold(scene) == "quarantine"
        or margin(scene) < MIN_GAP_PX
        or min(values) < 0.15
        or max(values) > 0.85
        or not 5 <= int(scene["point_radius"]) <= 12
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
