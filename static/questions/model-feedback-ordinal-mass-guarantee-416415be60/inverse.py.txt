"""Independent RGB-only inverse for the ambiguous mass-inequality world."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

PALETTE = (
    (201, 67, 64),
    (48, 111, 176),
    (43, 147, 103),
    (222, 153, 35),
    (137, 88, 174),
    (27, 153, 164),
)
INK = (35, 42, 54)
CLUE_RECTS = (
    (20, 132, 295, 252),
    (315, 132, 590, 252),
    (610, 132, 885, 252),
    (905, 132, 1180, 252),
    (167, 272, 442, 392),
    (462, 272, 737, 392),
    (757, 272, 1032, 392),
)
OPTION_RECTS = (
    (30, 472, 585, 692),
    (615, 472, 1170, 692),
    (30, 720, 585, 940),
    (615, 720, 1170, 940),
)
ASSIGNMENTS = np.asarray(list(itertools.permutations(range(1, 10), 6)), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    found: list[tuple[int, float, float]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            xs.append(x)
            ys.append(y)
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        found.append((len(xs), sum(xs) / len(xs), sum(ys) / len(ys)))
    return found


def _read_difference(arr: np.ndarray, rect: tuple[int, int, int, int], clue: bool) -> np.ndarray | None:
    x0, y0, x1, y1 = rect
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    center = (x1 - x0) / 2
    vector = np.zeros(6, dtype=np.int16)
    token_count = 0
    side_count = {-1: 0, 1: 0}
    expected_y = 82 if clue else 132
    for color_index, color in enumerate(PALETTE):
        mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
        for size, x, y in _components(mask):
            if not 560 <= size <= 760 or abs(y - expected_y) > 1.0 or abs(x - center) < 28:
                return None
            side = -1 if x < center else 1
            vector[color_index] += -side
            side_count[side] += 1
            token_count += 1
    if token_count != 4 or side_count != {-1: 2, 1: 2}:
        return None
    return vector


def _structure_ok(arr: np.ndarray) -> bool:
    if arr.shape != (960, 1200, 3):
        return False
    ink = np.asarray(INK, dtype=np.uint8)
    border_points = [(x0, (y0 + y1) // 2) for x0, y0, _x1, y1 in CLUE_RECTS + OPTION_RECTS]
    return all(np.array_equal(arr[y, x], ink) for x, y in border_points)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if not _structure_ok(arr):
        return "abstain"
    clue_rows = []
    for rect in CLUE_RECTS:
        vector = _read_difference(arr, rect, True)
        if vector is None:
            return "abstain"
        clue_rows.append(vector)
    clue_matrix = np.stack(clue_rows)
    feasible = ASSIGNMENTS[np.all(ASSIGNMENTS @ clue_matrix.T > 0, axis=1)]
    if not 2 <= len(feasible) <= 5000:
        return "abstain"
    matches: list[int] = []
    for index, rect in enumerate(OPTION_RECTS, 1):
        vector = _read_difference(arr, rect, False)
        if vector is None:
            return "abstain"
        values = feasible @ vector
        if np.all(values > 0):
            matches.append(index)
        elif not (np.any(values > 0) and np.any(values < 0)):
            return "abstain"
    return f"test-{matches[0]}" if len(matches) == 1 else "abstain"
