"""Deterministic sampler and renderer for invariant inequality reasoning."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 960
SIZE = (WIDTH, HEIGHT)
BACKGROUND = (247, 245, 238)
CARD = (255, 253, 248)
CLUE_CARD = (237, 246, 250)
INK = (35, 42, 54)
MUTED = (103, 110, 120)
COLORS = (
    (201, 67, 64),
    (48, 111, 176),
    (43, 147, 103),
    (222, 153, 35),
    (137, 88, 174),
    (27, 153, 164),
)
DECISIONS = ("test-1", "test-2", "test-3", "test-4")

# A spec is (left roles, right roles). All bundles contain two equal-size tokens.
CLUE_SPECS = (
    ((3, 5), (0, 1)),
    ((3, 4), (1, 2)),
    ((1, 3), (4, 5)),
    ((0, 4), (2, 5)),
    ((2, 3), (0, 1)),
    ((1, 4), (0, 5)),
    ((0, 3), (1, 2)),
)

# The first comparison is true in all 864 feasible calibrations. Each other
# comparison changes sign within that same feasible set.
BASE_TEST_SPECS = (
    ((3, 4), (0, 1)),
    ((1, 4), (0, 3)),
    ((0, 3), (2, 4)),
    ((4, 5), (0, 1)),
)

CLUE_RECTS = (
    (20, 132, 295, 252),
    (315, 132, 590, 252),
    (610, 132, 885, 252),
    (905, 132, 1180, 252),
    (167, 272, 442, 392),
    (462, 272, 737, 392),
    (757, 272, 1032, 392),
)
OPTION_RECTS = (
    (30, 472, 585, 692),
    (615, 472, 1170, 692),
    (30, 720, 585, 940),
    (615, 720, 1170, 940),
)

_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_SMALL = ImageFont.truetype(_BOLD_PATH, 16)
FONT_MEDIUM = ImageFont.truetype(_BOLD_PATH, 25)
FONT_LARGE = ImageFont.truetype(_BOLD_PATH, 38)


def _difference(spec: tuple[tuple[int, ...], tuple[int, ...]], masses: tuple[int, ...]) -> int:
    left, right = spec
    return sum(masses[index] for index in left) - sum(masses[index] for index in right)


FEASIBLE_MASSES = tuple(
    masses
    for masses in itertools.permutations(range(1, 10), 6)
    if all(_difference(spec, masses) > 0 for spec in CLUE_SPECS)
)


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    role_colors = list(range(6))
    rng.shuffle(role_colors)
    distractors = list(BASE_TEST_SPECS[1:])
    rng.shuffle(distractors)
    answer_slot = seed % 4
    options = distractors[:]
    options.insert(answer_slot, BASE_TEST_SPECS[0])
    return {
        "role_colors": role_colors,
        "option_specs": [[list(left), list(right)] for left, right in options],
    }


def _canonical(spec: tuple[tuple[int, ...], tuple[int, ...]]) -> tuple[tuple[int, ...], tuple[int, ...]]:
    return tuple(sorted(spec[0])), tuple(sorted(spec[1]))


def _validated(scene: dict) -> tuple[list[int], tuple[tuple[tuple[int, ...], tuple[int, ...]], ...]]:
    role_colors = [int(value) for value in scene["role_colors"]]
    raw_options = scene["option_specs"]
    if sorted(role_colors) != list(range(6)):
        raise ValueError("role_colors must be a permutation of six colors")
    if not isinstance(raw_options, list) or len(raw_options) != 4:
        raise ValueError("four test specs are required")
    options = []
    for raw in raw_options:
        if not isinstance(raw, list) or len(raw) != 2:
            raise ValueError("each test needs left and right bundles")
        left = tuple(int(value) for value in raw[0])
        right = tuple(int(value) for value in raw[1])
        if len(left) != 2 or len(right) != 2:
            raise ValueError("each bundle must contain two tokens")
        if any(value not in range(6) for value in left + right):
            raise ValueError("invalid role")
        if len(set(left + right)) != 4:
            raise ValueError("test bundles must use four distinct roles")
        options.append((left, right))
    expected = sorted(_canonical(spec) for spec in BASE_TEST_SPECS)
    if sorted(_canonical(spec) for spec in options) != expected:
        raise ValueError("test specs must be the fixed comparison set")
    return role_colors, tuple(options)


def _guaranteed(spec: tuple[tuple[int, ...], tuple[int, ...]]) -> bool:
    return bool(FEASIBLE_MASSES) and all(_difference(spec, masses) > 0 for masses in FEASIBLE_MASSES)


def analytic_gold(scene: dict) -> str:
    try:
        _role_colors, options = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return "abstain"
    matches = [index for index, spec in enumerate(options) if _guaranteed(spec)]
    return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    try:
        _role_colors, options = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return 0.0
    guaranteed = [spec for spec in options if _guaranteed(spec)]
    distractors = [spec for spec in options if not _guaranteed(spec)]
    if len(guaranteed) != 1 or len(distractors) != 3:
        return 0.0
    positive_gap = min(_difference(guaranteed[0], masses) for masses in FEASIBLE_MASSES)
    counterexample_gaps = [-min(_difference(spec, masses) for masses in FEASIBLE_MASSES) for spec in distractors]
    return float(min([positive_gap, *counterexample_gaps]))


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.FreeTypeFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _token(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int], radius: int = 18) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color, outline=INK, width=3)
    draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=(255, 255, 255))


def _draw_card(draw: ImageDraw.ImageDraw, rect: tuple[int, int, int, int], spec: tuple[tuple[int, ...], tuple[int, ...]], role_colors: list[int], label: str, clue: bool) -> None:
    x0, y0, x1, y1 = rect
    draw.rounded_rectangle(rect, radius=15, fill=CLUE_CARD if clue else CARD, outline=INK, width=4)
    pill_width = 82 if clue else 100
    draw.rounded_rectangle((x0 + 10, y0 + 9, x0 + 10 + pill_width, y0 + 39), radius=8, fill=INK)
    _centered_text(draw, (x0 + 10 + pill_width // 2, y0 + 24), label, FONT_SMALL, (255, 255, 255))
    cx = (x0 + x1) // 2
    cy = y0 + (82 if clue else 132)
    if clue:
        offsets = (-77, -41, 41, 77)
    else:
        offsets = (-150, -94, 94, 150)
        _centered_text(draw, (cx - 122, y0 + 73), "LEFT", FONT_SMALL, MUTED)
        _centered_text(draw, (cx + 122, y0 + 73), "RIGHT", FONT_SMALL, MUTED)
        draw.line((cx - 177, cy + 26, cx - 67, cy + 26), fill=MUTED, width=3)
        draw.line((cx + 67, cy + 26, cx + 177, cy + 26), fill=MUTED, width=3)
    left, right = spec
    for role, offset in zip(left + right, offsets):
        _token(draw, (cx + offset, cy), COLORS[role_colors[role]])
    symbol = ">" if clue else "?"
    _centered_text(draw, (cx, cy), symbol, FONT_LARGE, INK)


def render(scene: dict) -> Image.Image:
    role_colors, options = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 27), "DIFFERENT WHOLE-NUMBER MASSES 1–9", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 64), "EVERY CLUE SAYS:  LEFT TOTAL  >  RIGHT TOTAL", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 103), "SEVEN CLUES", FONT_SMALL, MUTED)
    for index, (rect, spec) in enumerate(zip(CLUE_RECTS, CLUE_SPECS), 1):
        _draw_card(draw, rect, spec, role_colors, f"CLUE {index}", True)
    _centered_text(draw, (WIDTH // 2, 432), "WHICH TEST MUST ALSO HAVE LEFT HEAVIER?", FONT_MEDIUM, INK)
    for index, (rect, spec) in enumerate(zip(OPTION_RECTS, options), 1):
        _draw_card(draw, rect, spec, role_colors, f"TEST {index}", False)
    return image
