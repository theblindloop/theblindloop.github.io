"""Deterministic two-panel modular binary-addition world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
GRID = 7
CELL = 13
PANEL = 109
BACKGROUND = (248, 247, 242)
DARK, LIGHT = 55, 224
FRAME_PALETTES = (
    ((20, 145, 166), (229, 116, 30)),
    ((229, 116, 30), (20, 145, 166)),
    ((100, 79, 168), (35, 158, 104)),
    ((35, 158, 104), (100, 79, 168)),
)
SYMBOLS = ("X", "plus", "T", "L")
BIT_COUNT = GRID * GRID
MODULUS = 1 << BIT_COUNT


def _template(name: str) -> list[list[int]]:
    cells = [[0 for _ in range(GRID)] for _ in range(GRID)]
    for row in range(GRID):
        for col in range(GRID):
            if name == "X":
                on = row == col or row + col == GRID - 1
            elif name == "plus":
                on = row == GRID // 2 or col == GRID // 2
            elif name == "T":
                on = row == 0 or col == GRID // 2
            elif name == "L":
                on = col == 0 or row == GRID - 1
            else:
                raise ValueError(f"unknown symbol {name!r}")
            cells[row][col] = int(on)
    return cells


def _grid_to_int(bits: list[list[int]]) -> int:
    value = 0
    for row in bits:
        for bit in row:
            value = (value << 1) | int(bit)
    return value


def _int_to_grid(value: int) -> list[list[int]]:
    flat = [int((value >> (BIT_COUNT - 1 - i)) & 1) for i in range(BIT_COUNT)]
    return [flat[r * GRID : (r + 1) * GRID] for r in range(GRID)]


def _classify(bits: list[list[int]]) -> tuple[str, int]:
    scored = []
    for name in SYMBOLS:
        target = _template(name)
        distance = sum(bits[r][c] != target[r][c] for r in range(GRID) for c in range(GRID))
        scored.append((distance, name))
    scored.sort()
    return scored[0][1], int(scored[1][0] - scored[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = _grid_to_int(_template(SYMBOLS[int(seed) % len(SYMBOLS)]))
    # Fix each panel at 24 dark cells so answer identity cannot be read from
    # either panel's count or total ink. Rejection is deterministic and cheap.
    indices = list(range(BIT_COUNT))
    for _ in range(10000):
        chosen = rng.sample(indices, 24)
        left_value = sum(1 << i for i in chosen)
        right_value = (target - left_value) % MODULUS
        if right_value.bit_count() == 24:
            break
    else:
        raise RuntimeError("could not counterbalance binary addends")
    if rng.randrange(2):
        left_value, right_value = right_value, left_value
    palette = FRAME_PALETTES[rng.randrange(len(FRAME_PALETTES))]
    return {
        "left_bits": _int_to_grid(left_value),
        "right_bits": _int_to_grid(right_value),
        "frame_colors": [list(palette[0]), list(palette[1])],
        "offset_x": rng.randint(-2, 2),
        "offset_y": rng.randint(-2, 2),
    }


def _draw_panel(draw: ImageDraw.ImageDraw, x: int, y: int, bits: list, color: Iterable[int]) -> None:
    draw.rectangle((x, y, x + PANEL - 1, y + PANEL - 1), fill=tuple(int(v) for v in color))
    draw.rectangle((x + 8, y + 8, x + PANEL - 9, y + PANEL - 9), fill=(238, 236, 229))
    for row in range(GRID):
        for col in range(GRID):
            x0, y0 = x + 9 + col * CELL, y + 9 + row * CELL
            tone = DARK if int(bits[row][col]) else LIGHT
            draw.rectangle((x0, y0, x0 + 11, y0 + 11), fill=(tone, tone, tone))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    _draw_panel(draw, 188 + ox, 100 + oy, scene["left_bits"], scene["frame_colors"][0])
    _draw_panel(draw, 352 + ox, 277 + oy, scene["right_bits"], scene["frame_colors"][1])
    return image


def _sum_bits(scene: dict) -> list[list[int]]:
    value = (_grid_to_int(scene["left_bits"]) + _grid_to_int(scene["right_bits"])) % MODULUS
    return _int_to_grid(value)


def analytic_gold(scene: dict) -> str:
    return _classify(_sum_bits(scene))[0]


def margin(scene: dict) -> float:
    return float(_classify(_sum_bits(scene))[1])


def is_quarantined(scene: dict) -> bool:
    return _classify(_sum_bits(scene))[1] < 3


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
