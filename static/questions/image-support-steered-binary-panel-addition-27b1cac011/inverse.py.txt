"""Independent pixel-only inverse for two-panel modular binary addition."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GRID = 7
BIT_COUNT = GRID * GRID
MODULUS = 1 << BIT_COUNT
SYMBOLS = ("X", "plus", "T", "L")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue, xs, ys = deque([(sy, sx)]), [], []
        seen[sy, sx] = True
        while queue:
            y, x = queue.popleft()
            xs.append(x); ys.append(y)
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; queue.append((ny,nx))
        if len(xs) >= 500:
            found.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return found


def _read_panel(array: np.ndarray, box: tuple[int, int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1, count = box
    if not (107 <= x1-x0 <= 110 and 107 <= y1-y0 <= 110 and count >= 1800):
        raise ValueError("missing or damaged panel fiducial")
    px, py = (x1-x0+1-18)/GRID, (y1-y0+1-18)/GRID
    bits = []
    for row in range(GRID):
        line = []
        for col in range(GRID):
            cx = int(round(x0 + 9 + col*px + px/2 - .5))
            cy = int(round(y0 + 9 + row*py + py/2 - .5))
            patch = array[cy-3:cy+4, cx-3:cx+4].astype(np.float32)
            if patch.shape != (7,7,3) or float(np.max(np.ptp(patch,axis=2))) > 8:
                raise ValueError("missing or non-gray cell evidence")
            tone = float(np.median(np.mean(patch,axis=2)))
            if tone < 85: line.append(1)
            elif tone > 185: line.append(0)
            else: raise ValueError("ambiguous binary cell tone")
        bits.append(line)
    return bits


def _as_int(bits: list[list[int]]) -> int:
    value = 0
    for row in bits:
        for bit in row: value = (value << 1) | bit
    return value


def _template(name: str) -> list[int]:
    out = []
    for row in range(GRID):
        for col in range(GRID):
            if name == "X": on = row == col or row+col == GRID-1
            elif name == "plus": on = row == GRID//2 or col == GRID//2
            elif name == "T": on = row == 0 or col == GRID//2
            else: on = col == 0 or row == GRID-1
            out.append(int(on))
    return out


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    boxes = _components(spread > 45)
    if len(boxes) != 2:
        raise ValueError(f"expected two visible panel fiducials, found {len(boxes)}")
    boxes.sort(key=lambda item: item[0])
    left, right = _read_panel(array, boxes[0]), _read_panel(array, boxes[1])
    summed = (_as_int(left) + _as_int(right)) % MODULUS
    result = [int((summed >> (BIT_COUNT-1-i)) & 1) for i in range(BIT_COUNT)]
    scored = sorted((sum(a != b for a,b in zip(result,_template(name))), name) for name in SYMBOLS)
    if scored[1][0] - scored[0][0] < 3:
        raise ValueError("ambiguous symbol classification")
    return scored[0][1]
