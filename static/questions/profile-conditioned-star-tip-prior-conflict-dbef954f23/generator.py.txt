"""Deterministic latent sampler and raster renderer for the star-tip world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 384
SUPERSAMPLE = 4
INKS = (
    (35, 72, 154),
    (171, 48, 58),
    (39, 111, 76),
    (103, 62, 145),
    (20, 105, 119),
    (174, 83, 24),
)
BACKGROUNDS = (
    (250, 248, 242),
    (242, 248, 251),
    (249, 244, 249),
    (244, 249, 243),
)
QUARANTINE_PROMINENCE_PX = 31.5


def sample_scene(seed: int) -> dict:
    """Return one scene; consecutive triplets share all non-label nuisances."""
    tip_count = 4 + int(seed) % 3
    rng = random.Random(int(seed) // 3)
    base_radius = rng.uniform(106.0, 124.0)
    # Equalize polygon area across tip counts while leaving ample size overlap.
    reference_factor = 5.0 * math.sin(math.pi / 5.0)
    count_factor = tip_count * math.sin(math.pi / tip_count)
    outer_radius = base_radius * math.sqrt(reference_factor / count_factor)
    return {
        "tip_count": tip_count,
        "rotation_deg": rng.uniform(-102.0, -78.0),
        "center_x": rng.uniform(177.0, 207.0),
        "center_y": rng.uniform(177.0, 207.0),
        "outer_radius": round(outer_radius, 6),
        "inner_ratio": round(rng.uniform(0.40, 0.74), 6),
        "color_index": rng.randrange(len(INKS)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
    }


def _vertices(scene: dict, scale: float = 1.0) -> list[tuple[float, float]]:
    count = int(scene["tip_count"])
    rotation = math.radians(float(scene["rotation_deg"]))
    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    outer = float(scene["outer_radius"]) * scale
    inner = outer * float(scene["inner_ratio"])
    points = []
    for index in range(2 * count):
        angle = rotation + index * math.pi / count
        radius = outer if index % 2 == 0 else inner
        points.append((cx + radius * math.cos(angle), cy + radius * math.sin(angle)))
    return points


def render(scene: dict) -> Image.Image:
    """Rasterize one solid alternating-radius polygon with antialiasing."""
    count = int(scene["tip_count"])
    if count not in (4, 5, 6):
        raise ValueError("tip_count must be 4, 5, or 6")
    # Read every declared style field explicitly so the rendered-field audit can
    # verify there are no hidden answer inputs.
    _ = (
        float(scene["rotation_deg"]),
        float(scene["center_x"]),
        float(scene["center_y"]),
        float(scene["outer_radius"]),
        float(scene["inner_ratio"]),
    )
    ink = INKS[int(scene["color_index"])]
    background = BACKGROUNDS[int(scene["background_index"])]
    large = Image.new("RGB", (CANVAS * SUPERSAMPLE, CANVAS * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(large)
    draw.polygon(_vertices(scene, SUPERSAMPLE), fill=ink)
    return large.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["tip_count"]))


def margin(scene: dict) -> float:
    """One-sided tip-to-notch radial prominence, in output pixels."""
    outer = float(scene["outer_radius"])
    prominence = outer * (1.0 - float(scene["inner_ratio"]))
    angular_clearance = outer * math.sin(math.pi / int(scene["tip_count"]))
    return float(min(prominence, angular_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PROMINENCE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All latent fields alter pixels; aliases are therefore not applicable.
    return []
