"""Independent image-only inverse for counting tips on one star silhouette."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


class AmbiguousRaster(ValueError):
    """Raised when the raster does not contain one measurable star silhouette."""


def _foreground_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or min(array.shape[:2]) < 32:
        raise AmbiguousRaster("image is too small")
    band = max(3, min(array.shape[:2]) // 40)
    border = np.concatenate(
        (
            array[:band].reshape(-1, 3),
            array[-band:].reshape(-1, 3),
            array[:, :band].reshape(-1, 3),
            array[:, -band:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array - background, axis=2)
    mask = distance > 42.0
    area = int(mask.sum())
    if area < 1800 or area > 0.55 * mask.size:
        raise AmbiguousRaster("foreground area is outside the supported range")
    ys, xs = np.nonzero(mask)
    if xs.min() < band or ys.min() < band or xs.max() >= mask.shape[1] - band or ys.max() >= mask.shape[0] - band:
        raise AmbiguousRaster("foreground touches the border")
    return mask


def _circular_smooth(values: np.ndarray, radius: int) -> np.ndarray:
    padded = np.concatenate((values[-radius:], values, values[:radius]))
    kernel = np.ones(2 * radius + 1, dtype=np.float64) / (2 * radius + 1)
    return np.convolve(padded, kernel, mode="valid")


def _radial_profile(mask: np.ndarray, bins: int = 720) -> np.ndarray:
    ys, xs = np.nonzero(mask)
    cx = float(xs.mean())
    cy = float(ys.mean())
    dx = xs.astype(np.float64) + 0.5 - cx
    dy = ys.astype(np.float64) + 0.5 - cy
    radius = np.hypot(dx, dy)
    angles = (np.arctan2(dy, dx) + 2.0 * math.pi) % (2.0 * math.pi)
    indices = np.floor(angles * bins / (2.0 * math.pi)).astype(np.int64) % bins
    profile = np.zeros(bins, dtype=np.float64)
    np.maximum.at(profile, indices, radius)
    good = np.flatnonzero(profile > 0)
    if len(good) < bins * 0.92:
        raise AmbiguousRaster("angular boundary coverage is incomplete")
    if len(good) != bins:
        extended_x = np.concatenate((good - bins, good, good + bins))
        extended_y = np.concatenate((profile[good], profile[good], profile[good]))
        missing = np.flatnonzero(profile == 0)
        profile[missing] = np.interp(missing, extended_x, extended_y)
    return _circular_smooth(profile, 5)


def analyze_image(image: Image.Image) -> tuple[str, float]:
    """Return (tip count, top-vs-runner-up Fourier amplitude gap)."""
    profile = _radial_profile(_foreground_mask(image))
    centered = profile - profile.mean()
    theta = 2.0 * math.pi * np.arange(len(profile), dtype=np.float64) / len(profile)
    amplitudes = {
        count: float(abs(np.sum(centered * np.exp(-1j * count * theta))) * 2.0 / len(profile))
        for count in (4, 5, 6)
    }
    ranked = sorted(amplitudes.items(), key=lambda item: (-item[1], item[0]))
    decision, strength = ranked[0]
    runner_up = ranked[1][1]
    gap = strength - runner_up
    if strength < 2.0 or gap < 0.55:
        raise AmbiguousRaster("no stable 4/5/6-tip periodicity")
    return str(decision), float(gap)


def decision_from_image(image: Image.Image) -> str:
    return analyze_image(image)[0]
