"""Renderer and analytic state replay for the Turn-and-Flip Walker."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 700
BG = (250, 249, 246)
DARK_INK = (30, 36, 45)
LIGHT_TILE = (242, 211, 105)
DARK_TILE = (57, 91, 128)
MAGENTA = (201, 43, 132)
CYAN = (35, 166, 190)
HEADINGS = ("up", "right", "down", "left")


def _replay(grid: list[list[int]], row: int, col: int, heading: int, steps: int) -> int:
    board = [line[:] for line in grid]
    vectors = ((-1, 0), (0, 1), (1, 0), (0, -1))
    for _ in range(steps):
        heading = (heading + (1 if board[row][col] == 0 else -1)) % 4
        board[row][col] ^= 1
        dr, dc = vectors[heading]
        row = (row + dr) % 4
        col = (col + dc) % 4
    return heading


def sample_scene(seed: int) -> dict:
    """Balance final heading by seed while holding light/dark tile ink constant."""
    rng = random.Random(int(seed) * 104729 + 4099)
    target = int(seed) % 4
    for _ in range(20000):
        states = [0] * 8 + [1] * 8
        rng.shuffle(states)
        grid = [states[index:index + 4] for index in range(0, 16, 4)]
        row, col, heading = rng.randrange(4), rng.randrange(4), rng.randrange(4)
        if _replay(grid, row, col, heading, 7) == target:
            return {"grid": grid, "start_row": row, "start_col": col,
                    "start_heading": heading, "step_count": 7}
    raise RuntimeError("failed to sample balanced turn-and-flip scene")


def _font(size: int, bold: bool = False):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int,
            fill=DARK_INK, bold: bool = False) -> None:
    font = _font(size, bold)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
              text, fill=fill, font=font)


def _arrow_points(cx: int, cy: int, heading: int) -> list[tuple[int, int]]:
    # Base arrow points right. Its broad triangular head has more ink on the tip side.
    base = [(-34, -8), (2, -8), (2, -20), (35, 0),
            (2, 20), (2, 8), (-34, 8)]
    points = []
    for x, y in base:
        # Base points right (heading index 1); rotate to the requested compass index.
        for _ in range((heading - 1) % 4):
            x, y = -y, x
        points.append((cx + x, cy + y))
    return points


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    _center(draw, (WIDTH // 2, 29), "TURN-AND-FLIP WALKER", 26, bold=True)
    _center(draw, (WIDTH // 2, 61), "LIGHT: turn right   •   DARK: turn left   •   then flip and move", 17)

    left, top, cell, gap = 250, 91, 100, 5
    draw.rounded_rectangle((left - 12, top - 12, left + 4 * cell + 12, top + 4 * cell + 12),
                           radius=12, fill=DARK_INK)
    for row in range(4):
        for col in range(4):
            x0, y0 = left + col * cell + gap, top + row * cell + gap
            color = DARK_TILE if int(scene["grid"][row][col]) else LIGHT_TILE
            draw.rectangle((x0, y0, x0 + cell - 2 * gap - 1, y0 + cell - 2 * gap - 1), fill=color)

    cx = left + int(scene["start_col"]) * cell + cell // 2
    cy = top + int(scene["start_row"]) * cell + cell // 2
    draw.polygon(_arrow_points(cx, cy, int(scene["start_heading"])), fill=MAGENTA,
                 outline=(255, 255, 255), width=2)
    _center(draw, (160, 275), "START", 18, bold=True)
    draw.line((190, 275, 229, 275), fill=MAGENTA, width=5)

    _center(draw, (WIDTH // 2, 535), "STEP CARDS  —  READ LEFT TO RIGHT", 17, bold=True)
    count = int(scene["step_count"])
    card_xs = [144 + 102 * index for index in range(count)]
    for index, x in enumerate(card_xs, start=1):
        draw.rounded_rectangle((x - 37, 566, x + 37, 638), radius=12,
                               fill=(255, 255, 255), outline=(215, 220, 225), width=3)
        draw.ellipse((x - 19, 580, x + 19, 618), fill=CYAN)
        _center(draw, (x, 598), str(index), 18, fill=(255, 255, 255), bold=True)
        _center(draw, (x, 661), "STEP", 13, bold=True)
    return image


def analytic_gold(scene: dict) -> str:
    result = _replay([[int(value) for value in row] for row in scene["grid"]],
                     int(scene["start_row"]), int(scene["start_col"]),
                     int(scene["start_heading"]), int(scene["step_count"]))
    return HEADINGS[result]


def margin(scene: dict) -> float:
    # Conservative minimum channel separation used by the independent masks.
    return 38.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
