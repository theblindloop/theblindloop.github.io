from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 640
BACKGROUND = (250, 250, 246)
NAVY = (25, 45, 74)
CREAM = (238, 224, 179)
CYAN = (39, 190, 202)
GOLD = (236, 175, 41)
TAG_COLORS = {
    "red": (219, 66, 60),
    "green": (55, 160, 91),
    "blue": (60, 105, 211),
}


def _rotate_pos(row: int, col: int) -> tuple[int, int]:
    return col, 3 - row


def _orbits() -> list[list[tuple[int, int]]]:
    remaining = {(r, c) for r in range(4) for c in range(4)}
    groups = []
    while remaining:
        start = min(remaining)
        group = []
        point = start
        for _ in range(4):
            group.append(point)
            point = _rotate_pos(*point)
        group = sorted(set(group))
        groups.append(group)
        remaining -= set(group)
    return groups


ORBITS = _orbits()


def _target_from_source(source: list[list[int]]) -> list[list[int]]:
    result = [[0] * 4 for _ in range(4)]
    for group in ORBITS:
        value = int(sum(source[r][c] for r, c in group) == 2)
        for r, c in group:
            result[r][c] = value
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    source = [[0] * 4 for _ in range(4)]
    statuses = [rng.randrange(2) for _ in ORBITS]
    if len(set(statuses)) == 1:
        statuses[seed % len(statuses)] ^= 1
    for status, group in zip(statuses, ORBITS, strict=True):
        if status:
            count = 2
        else:
            count = rng.choice((0, 1, 3, 4))
        for r, c in rng.sample(group, count):
            source[r][c] = 1

    target = _target_from_source(source)
    distractors = []
    flip_order = list(range(len(ORBITS)))
    rng.shuffle(flip_order)
    for number in (1, 2):
        grid = deepcopy(target)
        for orbit_index in flip_order[:number]:
            for r, c in ORBITS[orbit_index]:
                grid[r][c] ^= 1
        distractors.append(grid)

    target_slot = seed % 3
    patterns = [None, None, None]
    patterns[target_slot] = target
    other_slots = [index for index in range(3) if index != target_slot]
    for slot, grid in zip(other_slots, distractors, strict=True):
        patterns[slot] = grid
    base_tags = ["red", "green", "blue"]
    shift = (seed // 3) % 3
    tags = base_tags[shift:] + base_tags[:shift]
    return {"source": source, "candidate_patterns": patterns, "tags": tags}


def _draw_quadrant(draw: ImageDraw.ImageDraw, qx: int, qy: int, tag, grid) -> None:
    size = 272
    draw.rectangle((qx, qy, qx + size - 1, qy + size - 1), fill=NAVY)
    draw.rectangle((qx + 8, qy + 8, qx + 263, qy + 47), fill=tag)
    for row in range(4):
        for col in range(4):
            x0 = qx + 8 + col * 64
            y0 = qy + 48 + row * 54
            fill = CYAN if grid[row][col] else CREAM
            draw.rectangle((x0, y0, x0 + 63, y0 + 53), fill=fill)
            draw.rectangle((x0, y0, x0 + 63, y0 + 53), outline=NAVY, width=5)


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    candidates = scene["candidate_patterns"]
    tags = scene["tags"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    left = top = 48
    _draw_quadrant(draw, left, top, GOLD, source)
    positions = ((left + 272, top), (left, top + 272), (left + 272, top + 272))
    for (qx, qy), tag, grid in zip(positions, tags, candidates, strict=True):
        _draw_quadrant(draw, qx, qy, TAG_COLORS[tag], grid)
    draw.rectangle((left, top, left + 543, top + 543), outline=NAVY, width=8)
    return image


def analytic_gold(scene: dict) -> str:
    target = _target_from_source(scene["source"])
    matches = [i for i, grid in enumerate(scene["candidate_patterns"]) if grid == target]
    if len(matches) != 1:
        raise ValueError("scene does not have a unique matching candidate")
    return scene["tags"][matches[0]]


def margin(scene: dict) -> float:
    target = _target_from_source(scene["source"])
    distances = sorted(
        sum(a != b for row_a, row_b in zip(grid, target, strict=True) for a, b in zip(row_a, row_b, strict=True))
        for grid in scene["candidate_patterns"]
    )
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
