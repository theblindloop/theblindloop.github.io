from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = np.array((250, 250, 246), dtype=np.int32)
NAVY = np.array((25, 45, 74), dtype=np.int32)
CREAM = np.array((238, 224, 179), dtype=np.int32)
CYAN = np.array((39, 190, 202), dtype=np.int32)
TAGS = {
    "red": np.array((219, 66, 60), dtype=np.int32),
    "green": np.array((55, 160, 91), dtype=np.int32),
    "blue": np.array((60, 105, 211), dtype=np.int32),
}
PALETTE = np.stack((NAVY, CREAM, CYAN, *TAGS.values(), np.array((236, 175, 41))))


def _nearest(color: np.ndarray, palette: np.ndarray) -> int:
    return int(np.argmin(np.sum((palette - color.astype(np.int32)) ** 2, axis=1)))


def _rotate_pos(row: int, col: int) -> tuple[int, int]:
    return col, 3 - row


def _orbits() -> list[list[tuple[int, int]]]:
    remaining = {(r, c) for r in range(4) for c in range(4)}
    groups = []
    while remaining:
        point = min(remaining)
        group = []
        for _ in range(4):
            group.append(point)
            point = _rotate_pos(*point)
        group = sorted(set(group))
        groups.append(group)
        remaining -= set(group)
    return groups


ORBITS = _orbits()


def _read_grid(arr: np.ndarray, qx: float, qy: float, qsize: float) -> list[list[int]]:
    grid = []
    for row in range(4):
        values = []
        for col in range(4):
            cx = int(round(qx + qsize * ((8 + 64 * (col + 0.5)) / 272)))
            cy = int(round(qy + qsize * ((48 + 54 * (row + 0.5)) / 272)))
            patch = arr[cy - 3:cy + 4, cx - 3:cx + 4].mean(axis=(0, 1))
            label = _nearest(patch, np.stack((CREAM, CYAN)))
            if np.min(np.sqrt(np.sum((np.stack((CREAM, CYAN)) - patch) ** 2, axis=1))) > 30:
                raise ValueError("cell center is unreadable")
            values.append(label)
        grid.append(values)
    return grid


def _target(source: list[list[int]]) -> list[list[int]]:
    result = [[0] * 4 for _ in range(4)]
    for group in ORBITS:
        value = int(sum(source[r][c] for r, c in group) == 2)
        for r, c in group:
            result[r][c] = value
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    if arr.ndim != 3 or min(arr.shape[:2]) < 300:
        raise ValueError("image is too small")
    delta = np.sqrt(np.sum((arr - BACKGROUND) ** 2, axis=2))
    ys, xs = np.nonzero(delta > 35)
    if len(xs) < 10000:
        raise ValueError("carpet is missing")
    x0, x1, y0, y1 = xs.min(), xs.max() + 1, ys.min(), ys.max() + 1
    width, height = x1 - x0, y1 - y0
    if min(width, height) < 400 or abs(width - height) > 0.03 * max(width, height):
        raise ValueError("carpet bounds are malformed")
    carpet = arr[y0:y1, x0:x1]
    background_fraction = float(np.mean(np.sqrt(np.sum((carpet - BACKGROUND) ** 2, axis=2)) < 20))
    if background_fraction > 0.003:
        raise ValueError("connected carpet has an erased region")

    q = width / 2.0
    source = _read_grid(arr, x0, y0, q)
    positions = ((x0 + q, y0), (x0, y0 + q), (x0 + q, y0 + q))
    candidates = [_read_grid(arr, qx, qy, q) for qx, qy in positions]
    tag_palette = np.stack(tuple(TAGS.values()))
    tag_names = tuple(TAGS)
    tags = []
    for qx, qy in positions:
        patch = arr[int(qy + q * 0.06):int(qy + q * 0.14), int(qx + q * 0.18):int(qx + q * 0.82)]
        mean = patch.mean(axis=(0, 1))
        index = _nearest(mean, tag_palette)
        if np.min(np.sqrt(np.sum((tag_palette - mean) ** 2, axis=1))) > 30:
            raise ValueError("candidate tag is unreadable")
        tags.append(tag_names[index])
    if len(set(tags)) != 3:
        raise ValueError("candidate tags are not unique")
    target = _target(source)
    matches = [index for index, grid in enumerate(candidates) if grid == target]
    if len(matches) != 1:
        raise ValueError("image has no unique match")
    distances = sorted(sum(a != b for ra, rb in zip(grid, target) for a, b in zip(ra, rb)) for grid in candidates)
    if distances[1] - distances[0] < 4:
        raise ValueError("decision margin is ambiguous")
    return tags[matches[0]]
