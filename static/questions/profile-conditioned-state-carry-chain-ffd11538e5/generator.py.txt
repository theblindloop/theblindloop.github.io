"""Binary carry-increment register: track the target cell's final bit.

A horizontal 4-bit register is drawn once at its START value; the rightmost cell
is the least-significant bit (ones place) and each cell one column to the left
doubles the place value (position p contributes 2^p).  An orange ring marks the
TARGET cell.  To the lower right, K green up-triangle glyphs declare K successive
increment-by-one operations, applied left to right.  The register is never drawn
again, so the final value must be recovered by replaying the increments with
carry propagation:

    v0   = sum(bits[p] * 2^p)          initial 4-bit value (0..15)
    vf   = (v0 + K) mod 16             final value after K increments
    bit  = (vf >> target_pos) & 1      final state of the ringed target cell

The decision is whether the ringed cell finishes FILLED (solid disk, bit 1) or
OUTLINE (hollow ring, bit 0).  The gold depends on carry-aware modular addition,
not on any single frame: a single +1 can ripple a carry leftward flipping every
trailing 1 to 0 before settling on the first 0, so the target bit can flip even
when it lies far from the ones place.

Rendering is fully parameter-evaluated from the latent scene; no hidden state.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "state_carry_chain-0.1.0"

CANVAS_W = 700
CANVAS_H = 360
CELL_Y = 150
SLOT_X = [150, 268, 386, 504]  # left to right == MSB (position 3) .. LSB (position 0)
CHIP_R = 26
OUTLINE_WIDTH = 6

FILLED_COLOR = (66, 66, 66)
EMPTY_COLOR = "white"
OUTLINE_COLOR = (40, 40, 40)
RING_COLOR = (255, 150, 0)
OP_COLOR = (0, 165, 0)
OP_Y = 295
OP_X = [160, 270, 380, 490]
OP_HALF = 22
OP_H = 46


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) % (2**32))
    bits = [rng.randint(0, 1) for _ in range(4)]
    target_pos = rng.randint(0, 3)
    increments = rng.randint(1, 4)
    return {
        "bits": bits,
        "target_pos": target_pos,
        "increments": increments,
    }


def _value(scene: dict) -> int:
    return sum(scene["bits"][p] * (1 << p) for p in range(4))


def _final_bit(scene: dict) -> int:
    v0 = _value(scene)
    k = scene["increments"]
    vf = (v0 + k) % 16
    return (vf >> scene["target_pos"]) & 1


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), "white")
    draw = ImageDraw.Draw(image)

    bits = scene["bits"]
    for i in range(4):
        p = 3 - i
        x = SLOT_X[i]
        if bits[p]:
            draw.ellipse(
                [x - CHIP_R, CELL_Y - CHIP_R, x + CHIP_R, CELL_Y + CHIP_R],
                fill=FILLED_COLOR,
                outline=OUTLINE_COLOR,
                width=4,
            )
        else:
            draw.ellipse(
                [x - CHIP_R, CELL_Y - CHIP_R, x + CHIP_R, CELL_Y + CHIP_R],
                fill=EMPTY_COLOR,
                outline=OUTLINE_COLOR,
                width=OUTLINE_WIDTH,
            )

    target_i = 3 - scene["target_pos"]
    tx = SLOT_X[target_i]
    rr = CHIP_R + 14
    draw.ellipse(
        [tx - rr, CELL_Y - rr, tx + rr, CELL_Y + rr],
        outline=RING_COLOR,
        width=7,
    )

    k = scene["increments"]
    for j in range(k):
        x = OP_X[j]
        draw.polygon(
            [
                (x, OP_Y - OP_H // 2),
                (x - OP_HALF, OP_Y + OP_H // 2),
                (x + OP_HALF, OP_Y + OP_H // 2),
            ],
            fill=OP_COLOR,
        )

    return image


def analytic_gold(scene: dict) -> str:
    return "filled" if _final_bit(scene) == 1 else "outline"


def margin(scene: dict) -> float:
    """Steps of +1 until the final target bit next flips (always >= 1)."""
    v0 = _value(scene)
    vf = (v0 + scene["increments"]) % 16
    p = scene["target_pos"]
    period = 1 << (p + 1)
    r = vf % period
    half = 1 << p
    dist = (period - r) if r >= half else (half - r)
    return float(max(1, dist))


def is_quarantined(scene: dict) -> bool:
    return (_value(scene) + scene["increments"]) >= 16


def latent_symmetries(scene: dict) -> list:
    return []
