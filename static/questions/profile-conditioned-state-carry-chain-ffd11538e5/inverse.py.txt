"""Independent pixel-only inverse arm for the carry-increment register.

Recovers the decision from the final PNG alone.  Layout constants are duplicated
here (not imported) so this module never reads the latent scene, the analytic
gold, or the renderer.  Steps:

1. Count the green up-triangle glyphs -> K increments.
2. Locate the orange ring -> the TARGET cell (bit position).
3. Read each register cell's centre patch: filled (dark) = 1, hollow (white
   centre with dark rim) = 0 -> initial bits.
4. Replay K increments as carry-aware modular addition, then report the bit
   that lands at the target cell as 'filled' or 'outline'.
"""

from __future__ import annotations

import numpy as np

CANVAS_W = 700
CANVAS_H = 360
CELL_Y = 150
SLOT_X = [150, 268, 386, 504]  # left to right == MSB (position 3) .. LSB (position 0)
CHIP_R = 26
PATCH_R = 10

GREEN = (0, 165, 0)
RING = (255, 150, 0)
FILLED = (66, 66, 66)


def _connected_components(mask: np.ndarray) -> int:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not seen[y, x]:
                count += 1
                stack = [(x, y)]
                seen[y, x] = True
                while stack:
                    cx, cy = stack.pop()
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx, ny = cx + dx, cy + dy
                        if (
                            0 <= nx < w
                            and 0 <= ny < h
                            and mask[ny, nx]
                            and not seen[ny, nx]
                        ):
                            seen[ny, nx] = True
                            stack.append((nx, ny))
    return count


def decision_from_image(image) -> str:
    rgb = image.convert("RGB")
    arr = np.asarray(rgb, dtype=int)

    green = (arr[:, :, 1] > 100) & (arr[:, :, 0] < 90) & (arr[:, :, 2] < 90)
    k = _connected_components(green)

    orange = (arr[:, :, 0] > 200) & (arr[:, :, 1] > 80) & (arr[:, :, 2] < 90)
    ys, xs = np.nonzero(orange)
    target_cx = int(round(float(xs.mean()))) if xs.size else 0
    target_i = min(range(4), key=lambda i: abs(SLOT_X[i] - target_cx))
    target_pos = 3 - target_i

    bits = [0] * 4
    for i in range(4):
        x = SLOT_X[i]
        patch = arr[
            CELL_Y - PATCH_R : CELL_Y + PATCH_R + 1,
            x - PATCH_R : x + PATCH_R + 1,
        ]
        dark = (patch[:, :, 0] < 120) & (patch[:, :, 1] < 120) & (patch[:, :, 2] < 120)
        bits[3 - i] = 1 if dark.mean() > 0.4 else 0

    v0 = sum(bits[p] * (1 << p) for p in range(4))
    vf = (v0 + k) % 16
    final_bit = (vf >> target_pos) & 1
    return "filled" if final_bit == 1 else "outline"
