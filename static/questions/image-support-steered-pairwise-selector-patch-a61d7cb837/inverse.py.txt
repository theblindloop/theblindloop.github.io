from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 25:
            found.append((len(xs), float(np.mean(xs)), float(np.mean(ys))))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]

    # Cyan bead rims are the only saturated blue-green marks in the visual world.
    cyan = (blue > red + 75) & (green > red + 70) & (green > 125)
    rims = [(area, x, y) for area, x, y in _components(cyan) if 80 <= area <= 360]
    if len(rims) != 6:
        raise ValueError(f"expected six intact bead rims, found {len(rims)}")

    xs = sorted(x for _, x, _ in rims)
    split_gap = max(xs[i + 1] - xs[i] for i in range(5))
    if split_gap < 140:
        raise ValueError("two disconnected bead columns are not resolvable")
    midpoint = (xs[2] + xs[3]) / 2.0
    left = sorted((item for item in rims if item[1] < midpoint), key=lambda item: item[2])
    right = sorted((item for item in rims if item[1] > midpoint), key=lambda item: item[2])
    if len(left) != 3 or len(right) != 3:
        raise ValueError("expected three beads in each column")

    def value(item: tuple[int, float, float]) -> int:
        _, x, y = item
        xi, yi = int(round(x)), int(round(y))
        patch = array[yi - 4:yi + 5, xi - 4:xi + 5]
        if patch.shape[:2] != (9, 9):
            raise ValueError("bead center clipped")
        luminance = float(patch.mean())
        if luminance < 115:
            return 1
        if luminance > 205:
            return 0
        raise ValueError("ambiguous bead fill")

    rows = [[value(left[index]), value(right[index])] for index in range(3)]

    # Gold pixels locate the start selector; the title/instructions contain no gold.
    gold = (red > 160) & (green > 105) & (green < 205) & (blue < 90)
    gold_parts = [(area, x, y) for area, x, y in _components(gold) if area >= 25]
    if not gold_parts:
        raise ValueError("START marker missing")
    marker_x = sum(area * x for area, x, _ in gold_parts) / sum(area for area, _, _ in gold_parts)
    left_x = float(np.mean([item[1] for item in left]))
    right_x = float(np.mean([item[1] for item in right]))
    if abs(marker_x - left_x) + 35 < abs(marker_x - right_x):
        active = 0
    elif abs(marker_x - right_x) + 35 < abs(marker_x - left_x):
        active = 1
    else:
        raise ValueError("START marker does not select one column")

    lamp = False
    for left_value, right_value in rows:
        if (left_value, right_value)[active]:
            lamp = not lamp
        if left_value == right_value:
            active = 1 - active
    return "yes" if lamp else "no"
