from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 640
BACKGROUND = (247, 244, 236)
PAPER = (255, 253, 248)
INK = (37, 43, 53)
MUTED = (104, 109, 116)
GOLD = (218, 164, 35)
PALETTES = (
    {"ring": (22, 157, 190), "on": (35, 48, 67)},
    {"ring": (29, 174, 198), "on": (48, 42, 66)},
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size=size)


def _simulate(start_side: int, rows: list[list[int]]) -> bool:
    active = int(start_side)
    lamp = False
    for left, right in rows:
        if (left, right)[active]:
            lamp = not lamp
        if left == right:
            active = 1 - active
    return lamp


def _pools() -> dict[int, tuple[tuple[int, tuple[tuple[int, int], ...]], ...]]:
    output: dict[int, list] = {0: [], 1: []}
    for start_side in (0, 1):
        for bits in itertools.product((0, 1), repeat=6):
            if sum(bits) != 3:
                continue
            rows = tuple((bits[i], bits[i + 1]) for i in range(0, 6, 2))
            output[int(_simulate(start_side, [list(row) for row in rows]))].append((start_side, rows))
    return {key: tuple(value) for key, value in output.items()}


POOLS = _pools()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) & 1
    start_side, rows = POOLS[target][rng.randrange(len(POOLS[target]))]
    offsets = ((-8, -6), (8, -6), (-8, 6), (8, 6))
    dx, dy = offsets[rng.randrange(len(offsets))]
    return {
        "start_side": start_side,
        "rows": [list(row) for row in rows],
        "origin": [250 + dx, 306 + dy],
        "palette": rng.randrange(len(PALETTES)),
    }


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _bead(draw: ImageDraw.ImageDraw, x: int, y: int, value: int, palette: dict) -> None:
    radius = 11
    fill = palette["on"] if value else PAPER
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=palette["ring"], width=4)


def render(scene: dict) -> Image.Image:
    left_x, center_y = (int(value) for value in scene["origin"])
    right_x = left_x + 200
    rows = [[int(value) for value in row] for row in scene["rows"]]
    start_side = int(scene["start_side"])
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered(draw, (WIDTH // 2, 54), "PAIRWISE SELECTOR PATCH", _font(27, True), INK)
    _centered(draw, (WIDTH // 2, 92), "Read rows top to bottom.  Lamp starts OFF.", _font(18), MUTED)
    _centered(draw, (WIDTH // 2, 120), "1) use ACTIVE bead   2) matching pair switches ACTIVE", _font(18), MUTED)

    # Remote neutral marks add presentation variation but encode no scene state.
    for x in range(76, 725, 93):
        draw.ellipse((x - 3, 176, x + 3, 182), fill=(201, 198, 190))
    draw.line((82, 465, 718, 465), fill=(217, 213, 204), width=2)

    ys = (center_y - 24, center_y, center_y + 24)
    for row_index, (y, row) in enumerate(zip(ys, rows), start=1):
        _centered(draw, (left_x - 38, y), str(row_index), _font(13, True), MUTED)
        _bead(draw, left_x, y, row[0], palette)
        _bead(draw, right_x, y, row[1], palette)

    # The start marker is detached from both bead banks and points at the active column.
    marker_x = (left_x, right_x)[start_side]
    marker_y = center_y - 61
    draw.polygon(((marker_x, marker_y + 12), (marker_x - 9, marker_y - 3), (marker_x + 9, marker_y - 3)), fill=GOLD)
    _centered(draw, (marker_x, marker_y - 18), "START", _font(13, True), GOLD)
    _centered(draw, (left_x, center_y + 62), "LEFT", _font(14, True), INK)
    _centered(draw, (right_x, center_y + 62), "RIGHT", _font(14, True), INK)

    _centered(draw, (WIDTH // 2, 506), "Dark ACTIVE bead flips lamp; then a MATCH switches columns.", _font(18), INK)
    _centered(draw, (WIDTH // 2, 552), "Is the lamp ON after row 3?   YES / NO", _font(21, True), INK)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if _simulate(int(scene["start_side"]), [list(row) for row in scene["rows"]]) else "no"


def margin(scene: dict) -> float:
    return 40.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        side = int(scene["start_side"])
        rows = [[int(value) for value in row] for row in scene["rows"]]
        origin = [int(value) for value in scene["origin"]]
        palette = int(scene["palette"])
    except (KeyError, TypeError, ValueError):
        return True
    return not (
        side in (0, 1)
        and len(rows) == 3
        and all(len(row) == 2 and set(row) <= {0, 1} for row in rows)
        and sum(sum(row) for row in rows) == 3
        and origin[0] in (242, 258)
        and origin[1] in (300, 312)
        and palette >= 0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette"] = int(scene["palette"]) + len(PALETTES)
    return [("palette_encoding_alias", twin)]
