"""Deterministic four-island visual world for orthogonal key correspondence."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 1200
BACKGROUND = (248, 250, 252)
INK = (35, 43, 54)
DOT_BLUE = (37, 99, 235)
QUERY_MAGENTA = (219, 39, 119)
KEY_GRAY = (107, 114, 128)
GRID_TEAL = (13, 148, 136)
TAG_RGB = {
    "amber": (217, 119, 6),
    "cyan": (8, 145, 178),
    "violet": (124, 58, 237),
}
TAGS = tuple(TAG_RGB)
SHAPES = ("circle", "square", "triangle")
COUNTS = (1, 2, 3)

# Four compact cards occupy opposite corners with a blank, non-decisive center.
QUERY_CENTER = (180, 180)
COLUMN_CENTER = (900, 180)
ROW_CENTER = (180, 900)
GRID_CENTER = (900, 900)


def _shuffled(rng: random.Random, values: tuple) -> list:
    output = list(values)
    rng.shuffle(output)
    return output


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed ^ 0x6C8E9CF5)
    shape_order = _shuffled(rng, SHAPES)
    count_order = _shuffled(rng, COUNTS)
    query_shape = rng.choice(SHAPES)
    query_count = rng.choice(COUNTS)
    row = count_order.index(query_count)
    column = shape_order.index(query_shape)

    # Start from a Latin square so every answer color has identical total ink
    # and occurs once in every row and column. Pin only the selected base value
    # to seed % 3; randomly exchange the two nonanswers.
    base_at_query = (row + column) % 3
    desired = seed % 3
    remaining_tags = [index for index in range(3) if index != desired]
    rng.shuffle(remaining_tags)
    color_map: list[int | None] = [None, None, None]
    color_map[base_at_query] = desired
    other_base = [index for index in range(3) if index != base_at_query]
    color_map[other_base[0]] = remaining_tags[0]
    color_map[other_base[1]] = remaining_tags[1]
    grid = [
        [TAGS[color_map[(r + c) % 3]] for c in range(3)]
        for r in range(3)
    ]
    return {
        "shape_order": shape_order,
        "count_order": count_order,
        "grid": grid,
        "query_shape": query_shape,
        "query_count": query_count,
    }


def _frame(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    half_width: int,
    half_height: int,
    color: tuple[int, int, int],
) -> None:
    cx, cy = center
    draw.rectangle(
        (cx - half_width, cy - half_height, cx + half_width, cy + half_height),
        outline=color,
        width=6,
    )


def _glyph(draw: ImageDraw.ImageDraw, name: str, x: int, y: int, size: int = 18) -> None:
    if name == "circle":
        draw.ellipse((x - size, y - size, x + size, y + size), fill=INK)
    elif name == "square":
        draw.rectangle((x - size, y - size, x + size, y + size), fill=INK)
    elif name == "triangle":
        draw.polygon(
            ((x, y - size - 2), (x - size - 2, y + size), (x + size + 2, y + size)),
            fill=INK,
        )
    else:
        raise ValueError("unknown glyph")


def _dot_group(draw: ImageDraw.ImageDraw, count: int, x: int, y: int) -> None:
    if count not in COUNTS:
        raise ValueError("invalid dot count")
    spacing = 20
    start = x - spacing * (count - 1) // 2
    for index in range(count):
        cx = start + spacing * index
        draw.ellipse((cx - 7, y - 7, cx + 7, y + 7), fill=DOT_BLUE)


def _draw_query(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = QUERY_CENTER
    _frame(draw, QUERY_CENTER, 90, 60, QUERY_MAGENTA)
    _glyph(draw, scene["query_shape"], cx - 38, cy, 18)
    _dot_group(draw, scene["query_count"], cx + 38, cy)


def _draw_column_key(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = COLUMN_CENTER
    _frame(draw, COLUMN_CENTER, 140, 60, KEY_GRAY)
    for offset, name in zip((-82, 0, 82), scene["shape_order"], strict=True):
        _glyph(draw, name, cx + offset, cy, 18)


def _draw_row_key(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = ROW_CENTER
    _frame(draw, ROW_CENTER, 60, 140, KEY_GRAY)
    for offset, count in zip((-82, 0, 82), scene["count_order"], strict=True):
        _dot_group(draw, count, cx, cy + offset)


def _draw_grid(scene: dict, draw: ImageDraw.ImageDraw) -> None:
    cx, cy = GRID_CENTER
    _frame(draw, GRID_CENTER, 130, 130, GRID_TEAL)
    for row, values in enumerate(scene["grid"]):
        for column, tag in enumerate(values):
            x = cx + (-72, 0, 72)[column]
            y = cy + (-72, 0, 72)[row]
            draw.rectangle((x - 25, y - 25, x + 25, y + 25), fill=TAG_RGB[tag])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_query(scene, draw)
    _draw_column_key(scene, draw)
    _draw_row_key(scene, draw)
    _draw_grid(scene, draw)
    return image


def analytic_gold(scene: dict) -> str:
    shape_order = list(scene["shape_order"])
    count_order = list(scene["count_order"])
    grid = scene["grid"]
    query_shape = scene["query_shape"]
    query_count = scene["query_count"]
    column = shape_order.index(query_shape)
    row = count_order.index(query_count)
    return str(grid[row][column])


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        shapes = list(scene["shape_order"])
        counts = list(scene["count_order"])
        grid = [list(row) for row in scene["grid"]]
        query_shape = scene["query_shape"]
        query_count = scene["query_count"]
        if sorted(shapes) != sorted(SHAPES) or sorted(counts) != list(COUNTS):
            return True
        if query_shape not in SHAPES or query_count not in COUNTS:
            return True
        if len(grid) != 3 or any(len(row) != 3 for row in grid):
            return True
        if any(sorted(row) != sorted(TAGS) for row in grid):
            return True
        for column in range(3):
            if sorted(grid[row][column] for row in range(3)) != sorted(TAGS):
                return True
        return analytic_gold(scene) not in TAGS
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
