"""Deterministic matched-panel bridge-incidence world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = (960, 360)
PANEL_X = (20, 340, 660)
PANEL_Y = 34
PANEL_SIZE = 280
LABELS = ("left", "center", "right")

# One planar graph is repeated in all panels.  Nodes 0, 5, and 7 all have
# degree four but are incident to respectively 0, 2, and 4 bridges.
BASE_NODES = [
    [108, 142],  # 0: wheel center, 0 bridges
    [78, 64],    # 1: outer a
    [204, 140],  # 2: outer b
    [164, 218],  # 3: outer c
    [66, 216],   # 4: outer d
    [158, 70],   # 5: subdivided rim station, 2 bridges
    [196, 28],   # 6: leaf at station 5
    [230, 70],   # 7: tree hub, 4 bridges
    [254, 28],   # 8: hub leaf
    [270, 74],   # 9: hub leaf
    [250, 120],  # 10: hub leaf
]
BASE_EDGES = [
    [1, 5], [5, 2], [2, 3], [3, 4], [4, 1],
    [0, 1], [0, 2], [0, 3], [0, 4],
    [5, 6], [5, 7], [7, 8], [7, 9], [7, 10],
]
ROLE_NODE = {0: 0, 2: 5, 4: 7}


def _transform(points: list[list[int]], code: int) -> list[list[int]]:
    out = []
    for x, y in points:
        x -= 140
        y -= 140
        if code >= 4:
            x = -x
        for _ in range(code % 4):
            x, y = -y, x
        out.append([x + 140, y + 140])
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    roles = [0, 2, 4]
    rng.shuffle(roles)
    transform = rng.randrange(8)
    palettes = [
        {"background": [248, 249, 246], "panel": [255, 255, 255], "border": [184, 196, 208], "edge": [83, 94, 105], "node": [22, 181, 199], "highlight": [244, 174, 50]},
        {"background": [250, 250, 247], "panel": [255, 255, 255], "border": [190, 199, 210], "edge": [91, 99, 109], "node": [18, 174, 196], "highlight": [238, 165, 43]},
    ]
    panels = []
    for role in roles:
        panels.append({
            "bridge_count": role,
            "nodes": _transform(copy.deepcopy(BASE_NODES), transform),
            "edges": copy.deepcopy(BASE_EDGES),
            "highlighted": ROLE_NODE[role],
        })
    return {
        "canvas": list(CANVAS),
        "panels": panels,
        "palette": copy.deepcopy(palettes[rng.randrange(2)]),
        "style": {"edge_width": 7, "node_radius": 11, "panel_x": list(PANEL_X), "panel_y": PANEL_Y, "panel_size": PANEL_SIZE},
    }


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    style = scene["style"]
    image = Image.new("RGB", tuple(scene["canvas"]), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    for panel_index, panel in enumerate(scene["panels"]):
        ox, oy = style["panel_x"][panel_index], style["panel_y"]
        size = style["panel_size"]
        draw.rounded_rectangle((ox, oy, ox + size, oy + size), radius=12, fill=tuple(palette["panel"]), outline=tuple(palette["border"]), width=3)
        points = [(ox + x, oy + y) for x, y in panel["nodes"]]
        for a, b in panel["edges"]:
            draw.line((points[a], points[b]), fill=tuple(palette["edge"]), width=style["edge_width"])
        radius = style["node_radius"]
        for index, (x, y) in enumerate(points):
            if index == panel["highlighted"]:
                fill, outline = tuple(palette["highlight"]), (133, 82, 5)
            else:
                fill, outline = tuple(palette["node"]), (5, 92, 109)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=outline, width=2)
    return image


def _bridge_count(panel: dict) -> int:
    n = len(panel["nodes"])
    edges = [tuple(edge) for edge in panel["edges"]]
    incident = 0
    for removed, (a, b) in enumerate(edges):
        seen = {a}
        stack = [a]
        while stack:
            u = stack.pop()
            for index, (x, y) in enumerate(edges):
                if index == removed:
                    continue
                v = y if x == u else x if y == u else None
                if v is not None and v not in seen:
                    seen.add(v)
                    stack.append(v)
        if b not in seen and panel["highlighted"] in (a, b):
            incident += 1
    return incident


def analytic_gold(scene: dict) -> str:
    # The sampled role is analytic gold; _bridge_count independently validates
    # that role in margin/quarantine, while the PNG-only oracle reconstructs it.
    hits = [i for i, panel in enumerate(scene["panels"]) if panel["bridge_count"] == 2]
    return LABELS[hits[0]] if len(hits) == 1 else "abstain"


def margin(scene: dict) -> float:
    counts = [_bridge_count(panel) for panel in scene["panels"]]
    return float(min(abs(value - 2) for value in counts if value != 2)) if counts.count(2) == 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    if tuple(scene.get("canvas", ())) != CANVAS or len(scene.get("panels", [])) != 3:
        return True
    counts = []
    for panel in scene["panels"]:
        if len(panel.get("nodes", [])) != 11 or len(panel.get("edges", [])) != 14:
            return True
        if not (0 <= panel.get("highlighted", -1) < 11):
            return True
        degree = sum(panel["highlighted"] in edge for edge in panel["edges"])
        if degree != 4:
            return True
        counts.append(_bridge_count(panel))
    return sorted(counts) != [0, 2, 4]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"] = [[b, a] for a, b in reversed(panel["edges"])]
    return [("reverse_undirected_edge_serialization", twin)]
