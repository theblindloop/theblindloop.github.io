"""Image-only decision arm.

Receives a PNG and nothing else. It rebuilds the constellation from the raster
(background estimate, connected blobs, blob centres, blob hue), organises the
centres into their global convex outline, and reports which coloured blob is a
corner of that outline.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
_COLOURS = {
    "red": (198, 46, 42),
    "green": (30, 138, 62),
    "blue": (36, 66, 190),
    "orange": (226, 132, 18),
}
_NEUTRAL = (108, 110, 116)
_MIN_BLOB_PX = 40
_SEPARATION_PX = 1.0


def _blobs(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        group: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            group.append((y, x))
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        groups.append(group)
    return groups


def _hull(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    pts = sorted(set(points))
    if len(pts) <= 2:
        return list(pts)

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower: list[tuple[float, float]] = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper: list[tuple[float, float]] = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _dist_segment(p, a, b) -> float:
    dx, dy = b[0] - a[0], b[1] - a[1]
    length = dx * dx + dy * dy
    if length <= 1e-12:
        return math.hypot(p[0] - a[0], p[1] - a[1])
    t = max(0.0, min(1.0, ((p[0] - a[0]) * dx + (p[1] - a[1]) * dy) / length))
    return math.hypot(p[0] - (a[0] + t * dx), p[1] - (a[1] + t * dy))


def _outside(p, poly) -> float:
    if len(poly) < 3:
        return 0.0
    for i in range(len(poly)):
        a, b = poly[i], poly[(i + 1) % len(poly)]
        if (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0]) < -1e-9:
            return min(_dist_segment(p, poly[j], poly[(j + 1) % len(poly)]) for j in range(len(poly)))
    return 0.0


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    flat = rgb.reshape(-1, 3)
    values, counts = np.unique(flat, axis=0, return_counts=True)
    paper = values[int(np.argmax(counts))].astype(np.int16)
    ink = np.abs(rgb - paper.reshape(1, 1, 3)).sum(axis=2) > 60

    centres: list[tuple[float, float]] = []
    labels: list[str] = []
    for group in _blobs(ink):
        if len(group) < _MIN_BLOB_PX:
            continue
        ys = np.array([p[0] for p in group])
        xs = np.array([p[1] for p in group])
        core = np.abs(rgb[ys, xs] - paper.reshape(1, 3)).sum(axis=1)
        keep = core >= 0.8 * core.max()
        tone = rgb[ys[keep], xs[keep]].mean(axis=0)
        best, best_gap = None, None
        for name, ref in list(_COLOURS.items()) + [("grey", _NEUTRAL)]:
            gap = float(np.abs(tone - np.array(ref)).sum())
            if best_gap is None or gap < best_gap:
                best, best_gap = name, gap
        centres.append((float(xs.mean()), float(ys.mean())))
        labels.append(best)

    if len(centres) < 8 or sum(1 for name in labels if name != "grey") != 4:
        return ABSTAIN

    scores: list[tuple[float, str]] = []
    for index, name in enumerate(labels):
        if name == "grey":
            continue
        others = [q for j, q in enumerate(centres) if j != index]
        scores.append((_outside(centres[index], _hull(others)), name))
    scores.sort(reverse=True)
    if scores[0][0] < _SEPARATION_PX or scores[1][0] >= _SEPARATION_PX:
        return ABSTAIN
    return scores[0][1]
