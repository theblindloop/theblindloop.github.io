"""Independent image-only inverse for the one-shot turnstile trace."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

CYAN = np.array((25, 170, 190), dtype=np.int16)
CORAL = np.array((235, 95, 80), dtype=np.int16)
GOLD = np.array((242, 190, 40), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[np.ndarray] = []
    height, width = mask.shape
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        out.append(np.asarray(points, dtype=np.int32))
    return out


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan = _mask(array, CYAN)
    coral = _mask(array, CORAL)
    command_mask = cyan | coral
    pieces = [part for part in _components(command_mask) if len(part) >= 150]
    if len(pieces) != 6:
        raise ValueError("expected six intact command spans")

    commands = []
    for part in pieces:
        ys, xs = part[:, 0], part[:, 1]
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        if width != 17 or height != 13 or len(part) != 221:
            raise ValueError("damaged or ambiguous command span")
        cy = float(np.mean(ys))
        cx = float(np.mean(xs))
        c_count = int(cyan[ys, xs].sum())
        r_count = int(coral[ys, xs].sum())
        if max(c_count, r_count) != len(part) or min(c_count, r_count) != 0:
            raise ValueError("mixed command color")
        commands.append((cx, cy, "load" if c_count else "use"))
    if sum(op == "load" for _, _, op in commands) != 3:
        raise ValueError("command balance is not three and three")
    if max(cy for _, cy, _ in commands) - min(cy for _, cy, _ in commands) > 1.0:
        raise ValueError("command spans do not form one thin trace")

    gold_points = np.argwhere(_mask(array, GOLD))
    if len(gold_points) < 120:
        raise ValueError("round gold start is missing or damaged")
    gold_x = float(np.mean(gold_points[:, 1]))
    commands.sort(key=lambda item: item[0])
    if gold_x > float(np.mean([item[0] for item in commands])):
        commands.reverse()

    key_full = False
    lamp_on = False
    for _, _, op in commands:
        if op == "load":
            key_full = True
        elif key_full:
            lamp_on = not lamp_on
            key_full = False
    return "on" if lamp_on else "off"
