"""Render a short ordered command trace with a consumable one-shot key."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SEGMENT_WIDTH = 18
SEGMENT_HEIGHT = 14
CYAN = (25, 170, 190)
CORAL = (235, 95, 80)
GOLD = (242, 190, 40)
INK = (35, 42, 52)
BACKGROUND = (250, 249, 246)


def _replay(ops: list[str]) -> str:
    key_full = False
    lamp_on = False
    for op in ops:
        if op == "load":
            key_full = True
        elif op == "use" and key_full:
            lamp_on = not lamp_on
            key_full = False
    return "on" if lamp_on else "off"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "on" if int(seed) % 2 else "off"
    base = ["load"] * 3 + ["use"] * 3
    for _ in range(200):
        ops = base[:]
        rng.shuffle(ops)
        if _replay(ops) == desired:
            break
    else:  # pragma: no cover - finite six-item permutation space
        raise RuntimeError("could not sample requested balanced state")
    direction = rng.choice([-1, 1])
    # Scene positions vary across the canvas while keeping the short trace well
    # inside one high-resolution ablation row. Direction-aware offsets ensure
    # the gold start plus all commands occupy three adjacent ablation columns.
    column_band = rng.choice([1, 2, 3, 4])
    x = 64 * column_band + (52 if direction == 1 else 32)
    y = rng.choice([89, 153, 217, 281, 345, 409])
    return {
        "ops": ops,
        "x": x,
        "y": y,
        "direction": direction,
    }


def render(scene: dict) -> Image.Image:
    ops = list(scene["ops"])
    x = int(scene["x"])
    y = int(scene["y"])
    direction = int(scene["direction"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    total = SEGMENT_WIDTH * len(ops)
    cy = y + SEGMENT_HEIGHT // 2

    # One dark backbone joins the start, every command span, and the arrow.
    draw.line((x - 13, cy, x + total + 13, cy), fill=INK, width=5)
    for index, op in enumerate(ops):
        spatial = index if direction == 1 else len(ops) - 1 - index
        left = x + spatial * SEGMENT_WIDTH
        color = CYAN if op == "load" else CORAL
        draw.rectangle(
            (left + 1, y + 1, left + SEGMENT_WIDTH - 1, y + SEGMENT_HEIGHT - 1),
            fill=color,
        )
    draw.rectangle((x, y, x + total, y + SEGMENT_HEIGHT), outline=INK, width=1)
    for boundary in range(1, len(ops)):
        bx = x + boundary * SEGMENT_WIDTH
        draw.line((bx, y, bx, y + SEGMENT_HEIGHT), fill=INK, width=1)

    if direction == 1:
        start_x = x - 12
        tip_x = x + total + 15
        draw.polygon(
            [(tip_x, cy), (tip_x - 10, cy - 7), (tip_x - 10, cy + 7)], fill=INK
        )
    else:
        start_x = x + total + 12
        tip_x = x - 15
        draw.polygon(
            [(tip_x, cy), (tip_x + 10, cy - 7), (tip_x + 10, cy + 7)], fill=INK
        )
    draw.ellipse((start_x - 8, cy - 8, start_x + 8, cy + 8), fill=GOLD, outline=INK, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(list(scene["ops"]))


def margin(scene: dict) -> float:
    # Nearest competing parse requires losing at least half a command span.
    return float(SEGMENT_WIDTH // 2)


def is_quarantined(scene: dict) -> bool:
    return not (
        len(scene.get("ops", [])) == 6
        and scene.get("ops", []).count("load") == 3
        and scene.get("ops", []).count("use") == 3
        and scene.get("direction") in (-1, 1)
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
