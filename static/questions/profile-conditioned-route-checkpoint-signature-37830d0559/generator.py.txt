"""Deterministic renderer and analytic arm for ordered checkpoint cables."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
SCALE = 3
TARGET_ORDER = ("orange", "blue", "purple")
TERMINAL_NAMES = ("red", "green", "yellow")

COLORS = {
    "background": (250, 249, 246),
    "dark": (38, 43, 50),
    "gray": (135, 143, 151),
    "orange": (238, 126, 34),
    "blue": (38, 112, 214),
    "purple": (143, 73, 188),
    "red": (214, 55, 62),
    "green": (42, 155, 88),
    "yellow": (232, 184, 36),
    "white": (255, 255, 255),
}


def _route_geometry(rng: random.Random, base_y: int) -> tuple[list[list[int]], list[list[int]]]:
    offsets = [-24, -12, 12, 24]
    rng.shuffle(offsets)
    y1, y2, y3, y4 = [base_y + value for value in offsets]
    points = [
        [52, base_y], [112, base_y], [112, y1], [216, y1],
        [216, y2], [262, y2], [262, y3], [370, y3],
        [370, y4], [430, y4], [430, y2], [536, y2],
        [536, y1], [588, y1],
    ]
    checkpoints = [[164, y1], [316, y3], [482, y2]]
    return points, checkpoints


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    match_route = (int(seed) // 3) % 3
    desired_terminal = TERMINAL_NAMES[int(seed) % 3]

    non_target = [
        list(order)
        for order in itertools.permutations(TARGET_ORDER)
        if tuple(order) != TARGET_ORDER
    ]
    rng.shuffle(non_target)
    route_orders: list[list[str]] = []
    distractor_cursor = 0
    for route_index in range(3):
        if route_index == match_route:
            route_orders.append(list(TARGET_ORDER))
        else:
            route_orders.append(non_target[distractor_cursor])
            distractor_cursor += 1

    remaining = [name for name in TERMINAL_NAMES if name != desired_terminal]
    rng.shuffle(remaining)
    terminal_colors: list[str] = []
    for route_index in range(3):
        terminal_colors.append(
            desired_terminal if route_index == match_route else remaining.pop()
        )

    bases = [92, 240, 388]
    route_points: list[list[list[int]]] = []
    checkpoint_points: list[list[list[int]]] = []
    for base in bases:
        points, markers = _route_geometry(rng, base)
        route_points.append(points)
        checkpoint_points.append(markers)

    ring_sides = [rng.choice(("left", "right")) for _ in range(3)]
    return {
        "route_orders": route_orders,
        "terminal_colors": terminal_colors,
        "route_points": route_points,
        "checkpoint_points": checkpoint_points,
        "ring_sides": ring_sides,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _scaled_points(points: list[list[int]]) -> list[tuple[int, int]]:
    return [(int(x) * SCALE, int(y) * SCALE) for x, y in points]


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    route_orders = scene["route_orders"]
    terminal_colors = scene["terminal_colors"]
    route_points = scene["route_points"]
    checkpoint_points = scene["checkpoint_points"]
    ring_sides = scene["ring_sides"]

    image = Image.new("RGB", (width * SCALE, height * SCALE), COLORS["background"])
    draw = ImageDraw.Draw(image)
    for points in route_points:
        draw.line(_scaled_points(points), fill=COLORS["dark"], width=7 * SCALE, joint="curve")

    for route_index in range(3):
        physical_order = list(route_orders[route_index])
        if ring_sides[route_index] == "right":
            physical_order.reverse()
        for (x, y), color_name in zip(checkpoint_points[route_index], physical_order):
            box = [(x - 11) * SCALE, (y - 11) * SCALE, (x + 11) * SCALE, (y + 11) * SCALE]
            draw.ellipse(box, fill=COLORS[color_name], outline=COLORS["dark"], width=2 * SCALE)

        left = route_points[route_index][0]
        right = route_points[route_index][-1]
        ring_point, terminal_point = (left, right)
        if ring_sides[route_index] == "right":
            ring_point, terminal_point = right, left

        rx, ry = ring_point
        outer = [(rx - 15) * SCALE, (ry - 15) * SCALE, (rx + 15) * SCALE, (ry + 15) * SCALE]
        inner = [(rx - 8) * SCALE, (ry - 8) * SCALE, (rx + 8) * SCALE, (ry + 8) * SCALE]
        draw.ellipse(outer, fill=COLORS["gray"], outline=COLORS["dark"], width=2 * SCALE)
        draw.ellipse(inner, fill=COLORS["background"])

        tx, ty = terminal_point
        square = [(tx - 14) * SCALE, (ty - 14) * SCALE, (tx + 14) * SCALE, (ty + 14) * SCALE]
        draw.rectangle(square, fill=COLORS[terminal_colors[route_index]], outline=COLORS["dark"], width=2 * SCALE)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    orders = scene["route_orders"]
    terminals = scene["terminal_colors"]
    matches = [index for index, order in enumerate(orders) if tuple(order) == TARGET_ORDER]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one target checkpoint sequence")
    return str(terminals[matches[0]])


def _sequence_gap(scene: dict) -> float:
    orders = scene["route_orders"]
    mismatches = [
        sum(a != b for a, b in zip(order, TARGET_ORDER))
        for order in orders
        if tuple(order) != TARGET_ORDER
    ]
    return float(min(mismatches, default=0) * 12)


def _band_clearance(scene: dict) -> float:
    ranges = []
    for points in scene["route_points"]:
        ys = [float(point[1]) for point in points]
        ranges.append((min(ys) - 16.0, max(ys) + 16.0))
    ranges.sort()
    return min(ranges[i + 1][0] - ranges[i][1] for i in range(len(ranges) - 1))


def margin(scene: dict) -> float:
    return float(min(_sequence_gap(scene), _band_clearance(scene)))


def is_quarantined(scene: dict) -> bool:
    orders = scene["route_orders"]
    unique = sum(tuple(order) == TARGET_ORDER for order in orders) == 1
    return (not unique) or margin(scene) < 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
