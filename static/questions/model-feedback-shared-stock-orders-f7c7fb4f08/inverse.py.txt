"""Independent image-only inverse for shared-stock orders."""

from __future__ import annotations

import numpy as np
from PIL import Image

TOKEN_COLORS = ((24, 137, 181), (146, 75, 163), (228, 130, 30), (52, 153, 91))
STAR_COLOR = (238, 185, 45)
RUN_COLOR = (72, 79, 89)
RECIPE_X = (22, 336, 650)
ORDER_X = (18, 254, 490, 726)
ROW_Y = (470, 545, 620)


def _components(mask: np.ndarray) -> int | None:
    h, w = mask.shape
    remaining = set(int(v) for v in np.flatnonzero(mask))
    sizes = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        size = 0
        while stack:
            v = stack.pop(); y, x = divmod(v, w); size += 1
            for n in (v-1, v+1, v-w, v+w):
                if n in remaining:
                    ny, nx = divmod(n, w)
                    if abs(nx-x) + abs(ny-y) == 1:
                        remaining.remove(n); stack.append(n)
        if size >= 70:
            sizes.append(size)
    if any(not 120 <= size <= 190 for size in sizes):
        return None
    return len(sizes)


def _count_color(arr: np.ndarray, box, color) -> int | None:
    x0,y0,x1,y1 = box
    crop = arr[y0:y1,x0:x1]
    delta = np.max(np.abs(crop.astype(np.int16)-np.asarray(color,dtype=np.int16)),axis=2)
    return _components(delta <= 2)


def _counts(arr: np.ndarray, box) -> tuple[int,...] | None:
    values = tuple(_count_color(arr, box, color) for color in TOKEN_COLORS)
    return None if any(v is None for v in values) else values


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (720,960,3):
        return "abstain"
    recipes = []
    for x0 in RECIPE_X:
        costs = _counts(arr, (x0+6,92,x0+152,194))
        stars = _count_color(arr, (x0+196,92,x0+282,194), STAR_COLOR)
        if costs is None or stars is None or sum(v>0 for v in costs) != 2 or not 3 <= stars <= 7:
            return "abstain"
        recipes.append((costs,stars))
    stock = _counts(arr,(145,246,815,336))
    if stock is None:
        return "abstain"
    orders=[]
    for x0 in ORDER_X:
        plan=[]
        for cy in ROW_Y:
            value=_count_color(arr,(x0+50,cy-28,x0+206,cy+28),RUN_COLOR)
            if value is None or not 0 <= value <= 4:
                return "abstain"
            plan.append(value)
        orders.append(tuple(plan))
    scored=[]
    for i,plan in enumerate(orders):
        used=tuple(sum(plan[r]*recipes[r][0][c] for r in range(3)) for c in range(4))
        if all(u<=s for u,s in zip(used,stock)):
            scored.append((i,sum(plan[r]*recipes[r][1] for r in range(3))))
    if not scored:
        return "abstain"
    top=max(score for _,score in scored)
    winners=[i for i,score in scored if score==top]
    return f"order-{chr(65+winners[0])}" if len(winners)==1 else "abstain"
