"""Deterministic renderer for competing integer allocations over shared stock."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

SIZE = (960, 720)
BACKGROUND = (246, 244, 238)
PANEL = (255, 254, 250)
INK = (38, 42, 48)
MUTED = (214, 211, 203)
ACCENT = (178, 76, 54)
TOKEN_COLORS = ((24, 137, 181), (146, 75, 163), (228, 130, 30), (52, 153, 91))
STAR_COLOR = (238, 185, 45)
RUN_COLOR = (72, 79, 89)
RECIPE_COLORS = ((208, 73, 73), (29, 151, 151), (128, 92, 184))
DECISIONS = ("order-A", "order-B", "order-C", "order-D")
RECIPE_X = (22, 336, 650)
ORDER_X = (18, 254, 490, 726)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _usage(plan: tuple[int, ...] | list[int], recipes: tuple[tuple[tuple[int, ...], int], ...] | list) -> tuple[int, ...]:
    return tuple(sum(int(plan[r]) * int(recipes[r][0][c]) for r in range(3)) for c in range(4))


def _reward(plan: tuple[int, ...] | list[int], recipes: tuple[tuple[tuple[int, ...], int], ...] | list) -> int:
    return sum(int(plan[r]) * int(recipes[r][1]) for r in range(3))


def _feasible(plan: tuple[int, ...] | list[int], stock: tuple[int, ...] | list[int], recipes) -> bool:
    return all(used <= have for used, have in zip(_usage(plan, recipes), stock))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(500):
        recipes = []
        pairs = [(0, 1), (1, 2), (2, 3)]
        rng.shuffle(pairs)
        for a, b in pairs:
            costs = [0, 0, 0, 0]
            costs[a], costs[b] = rng.randint(1, 3), rng.randint(1, 3)
            recipes.append((tuple(costs), rng.randint(3, 7)))
        recipes_t = tuple(recipes)
        stock = tuple(rng.randint(5, 10) for _ in range(4))
        plans = [p for p in itertools.product(range(5), repeat=3) if 2 <= sum(p) <= 9]
        feasible = [p for p in plans if _feasible(p, stock, recipes_t)]
        feasible.sort(key=lambda p: (_reward(p, recipes_t), p), reverse=True)
        if len(feasible) < 4:
            continue
        best_reward = _reward(feasible[0], recipes_t)
        best = [p for p in feasible if _reward(p, recipes_t) == best_reward]
        lower = [p for p in feasible if _reward(p, recipes_t) <= best_reward - 2]
        higher_bad = [p for p in plans if not _feasible(p, stock, recipes_t) and _reward(p, recipes_t) > best_reward]
        if len(best) != 1 or not lower or len(higher_bad) < 2:
            continue
        rng.shuffle(lower)
        rng.shuffle(higher_bad)
        shown = [best[0], lower[0], higher_bad[0], higher_bad[1]]
        if len(set(shown)) != 4:
            continue
        rng.shuffle(shown[1:])
        winner = int(seed) % 4
        shown[0], shown[winner] = shown[winner], shown[0]
        return {
            "stock_counts": list(stock),
            "recipes": [{"costs": list(costs), "stars": stars} for costs, stars in recipes_t],
            "order_runs": [list(plan) for plan in shown],
            "layout_variant": rng.randrange(32),
        }
    raise RuntimeError("unable to sample a valid shared-stock scene")


def _validated(scene: dict):
    stock = tuple(int(x) for x in scene["stock_counts"])
    raw_recipes = scene["recipes"]
    recipes = tuple((tuple(int(x) for x in item["costs"]), int(item["stars"])) for item in raw_recipes)
    orders = tuple(tuple(int(x) for x in plan) for plan in scene["order_runs"])
    variant = int(scene["layout_variant"])
    if len(stock) != 4 or len(recipes) != 3 or len(orders) != 4:
        raise ValueError("wrong collection size")
    if any(len(costs) != 4 or sum(v > 0 for v in costs) != 2 for costs, _ in recipes):
        raise ValueError("each recipe needs two resources")
    if any(not 0 <= x <= 12 for x in stock) or any(not 0 <= x <= 3 for costs, _ in recipes for x in costs):
        raise ValueError("resource count out of range")
    if any(not 3 <= stars <= 7 for _, stars in recipes) or any(len(p) != 3 or any(not 0 <= x <= 4 for x in p) for p in orders):
        raise ValueError("reward or run count out of range")
    if not 0 <= variant < 32:
        raise ValueError("bad layout variant")
    return stock, recipes, orders, variant


def analytic_gold(scene: dict) -> str:
    try:
        stock, recipes, orders, _ = _validated(scene)
        feasible = [(i, _reward(plan, recipes)) for i, plan in enumerate(orders) if _feasible(plan, stock, recipes)]
        if not feasible:
            return "abstain"
        top = max(score for _, score in feasible)
        winners = [i for i, score in feasible if score == top]
        return DECISIONS[winners[0]] if len(winners) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    stock, recipes, orders, _ = _validated(scene)
    scores = sorted((_reward(p, recipes) for p in orders if _feasible(p, stock, recipes)), reverse=True)
    return float(scores[0] - scores[1]) if len(scores) >= 2 and scores[0] > scores[1] else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        stock, recipes, orders, _ = _validated(scene)
        feasible_count = sum(_feasible(p, stock, recipes) for p in orders)
        winner = analytic_gold(scene)
        winner_score = max((_reward(p, recipes) for p in orders if _feasible(p, stock, recipes)), default=-1)
        has_high_infeasible = any(not _feasible(p, stock, recipes) and _reward(p, recipes) > winner_score for p in orders)
        return winner == "abstain" or margin(scene) < 2 or feasible_count < 2 or len(set(orders)) != 4 or not has_high_infeasible
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _disks(draw: ImageDraw.ImageDraw, counts, box, columns: int, reverse: bool, colors=TOKEN_COLORS, radius: int = 7) -> None:
    x0, y0, x1, y1 = box
    kinds = [k for k, count in enumerate(counts) for _ in range(int(count))]
    if reverse:
        kinds.reverse()
    gap = 20
    rows = max(1, (len(kinds) + columns - 1) // columns)
    shown_cols = min(columns, max(1, len(kinds)))
    sx = (x0 + x1) // 2 - (shown_cols - 1) * gap // 2
    sy = (y0 + y1) // 2 - (rows - 1) * gap // 2
    for i, kind in enumerate(kinds):
        cx, cy = sx + (i % columns) * gap, sy + (i // columns) * gap
        draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), fill=colors[kind], outline=INK, width=1)


def _same_disks(draw, count: int, box, columns: int, color, reverse: bool = False) -> None:
    _disks(draw, [count], box, columns, reverse, colors=(color,), radius=7)


def render(scene: dict) -> Image.Image:
    stock, recipes, orders, variant = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title, label, small = _font(23), _font(17), _font(14)
    draw.text((22, 12), "RECIPES: TOKENS USED → STARS EARNED", font=title, fill=INK)
    for i, ((costs, stars), x0) in enumerate(zip(recipes, RECIPE_X)):
        draw.rounded_rectangle((x0, 48, x0+288, 202), radius=11, fill=PANEL, outline=RECIPE_COLORS[i], width=4)
        draw.text((x0+12, 57), f"RECIPE {i+1}", font=small, fill=INK)
        _disks(draw, costs, (x0+8, 96, x0+150, 190), 6, bool((variant >> i) & 1))
        draw.line((x0+153, 144, x0+194, 144), fill=ACCENT, width=5)
        draw.polygon(((x0+194,144),(x0+181,134),(x0+181,154)), fill=ACCENT)
        _same_disks(draw, stars, (x0+198, 96, x0+280, 190), 4, STAR_COLOR, bool((variant >> (i+3)) & 1))
    draw.text((22, 216), "START STOCK", font=label, fill=INK)
    draw.rounded_rectangle((22, 242, 938, 340), radius=11, fill=PANEL, outline=ACCENT, width=4)
    _disks(draw, stock, (150, 248, 810, 334), 14, bool(variant & 16))
    draw.text((22, 356), "ORDERS: ONE MARK = ONE RECIPE RUN", font=label, fill=INK)
    row_y = (470, 545, 620)
    for oi, (plan, x0) in enumerate(zip(orders, ORDER_X)):
        draw.rounded_rectangle((x0, 386, x0+216, 700), radius=11, fill=PANEL, outline=INK, width=3)
        draw.text((x0+66, 398), f"order {chr(65+oi)}", font=small, fill=INK)
        draw.line((x0+10, 430, x0+206, 430), fill=MUTED, width=2)
        for ri, (runs, cy) in enumerate(zip(plan, row_y)):
            draw.ellipse((x0+13,cy-14,x0+41,cy+14), fill=RECIPE_COLORS[ri], outline=INK, width=2)
            tw = draw.textbbox((0,0), str(ri+1), font=small)[2]
            draw.text((x0+27-tw//2, cy-10), str(ri+1), font=small, fill=PANEL)
            _same_disks(draw, runs, (x0+54, cy-25, x0+202, cy+25), 5, RUN_COLOR, bool((variant+oi+ri)&1))
            if ri < 2:
                draw.line((x0+10, cy+36, x0+206, cy+36), fill=MUTED, width=1)
    return image
