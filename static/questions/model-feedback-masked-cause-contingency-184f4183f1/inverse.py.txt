"""Independent fixed-layout pixel inverse for masked causal contingency."""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = (247, 245, 238)
ON = (31, 61, 86)
OFF = (229, 232, 226)
AND = (25, 126, 124)
OR = (202, 93, 54)
XOR = (124, 75, 164)
PAIR_COLORS = ((43, 119, 176), (220, 125, 31), (73, 152, 94), (184, 75, 112))
PAIR_X = (12, 203, 394, 585)
SECOND_X = (126, 510)
DECISIONS = ("cause-A", "cause-B", "cause-C", "cause-D")


def _near(pixel, color, tolerance: int = 4) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _read_bit(arr, x: int, y: int) -> int | None:
    pixel = arr[y, x + 9]
    if _near(pixel, ON):
        return 1
    if _near(pixel, OFF):
        return 0
    return None


def _read_op(pixel) -> str | None:
    for name, color in (("AND", AND), ("OR", OR), ("XOR", XOR)):
        if _near(pixel, color):
            return name
    return None


def _apply(op: str, a: int, b: int) -> int:
    if op == "AND":
        return int(bool(a) and bool(b))
    if op == "OR":
        return int(bool(a) or bool(b))
    return int(bool(a) != bool(b))


def _evaluate(bits: list[int], ops: list[str]) -> list[int]:
    first = [_apply(ops[i], bits[2 * i], bits[2 * i + 1]) for i in range(4)]
    second = [_apply(ops[4], first[0], first[1]), _apply(ops[5], first[2], first[3])]
    return first + second + [_apply(ops[6], second[0], second[1])]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (640, 768, 3) or not _near(arr[635, 760], BACKGROUND):
        return "abstain"
    pair_labels: list[int] = []
    bits: list[int] = []
    ops: list[str] = []
    shown_nodes: list[int] = []
    for physical, x in enumerate(PAIR_X):
        border = arr[80, x + 2]
        matches = [i for i, color in enumerate(PAIR_COLORS) if _near(border, color)]
        if len(matches) != 1:
            return "abstain"
        pair_labels.append(matches[0])
        for sx in (x + 49, x + 122):
            bit = _read_bit(arr, sx, 116)
            if bit is None:
                return "abstain"
            bits.append(bit)
        op = _read_op(arr[211, x + 73 - 35])
        node = _read_bit(arr, x + 73 + 75, 211)
        if op is None or node is None:
            return "abstain"
        ops.append(op)
        shown_nodes.append(node)
    for gx in SECOND_X:
        op = _read_op(arr[354, gx - 35])
        node = _read_bit(arr, gx + 75, 354)
        if op is None or node is None:
            return "abstain"
        ops.append(op)
        shown_nodes.append(node)
    op = _read_op(arr[459, 309 - 35])
    target = _read_bit(arr, 309 + 75, 459)
    if op is None or target is None:
        return "abstain"
    ops.append(op)
    shown_nodes.append(target)
    if bits != [1] * 8 or sorted(pair_labels) != list(range(4)) or _evaluate(bits, ops) != shown_nodes:
        return "abstain"
    actual = shown_nodes[-1]
    qualifying: list[int] = []
    for physical in range(4):
        cause, backup = 2 * physical, 2 * physical + 1
        worlds: list[int] = []
        for off in ((cause,), (backup,), (cause, backup)):
            changed = bits.copy()
            for index in off:
                changed[index] = 0
            worlds.append(_evaluate(changed, ops)[-1])
        if worlds[0] == actual and worlds[1] == actual and worlds[2] != actual:
            qualifying.append(physical)
    return DECISIONS[pair_labels[qualifying[0]]] if len(qualifying) == 1 else "abstain"
