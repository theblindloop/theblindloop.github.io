"""Deterministic causal-pair world and high-contrast raster renderer."""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

SIZE = (768, 640)
BACKGROUND = (247, 245, 238)
INK = (32, 40, 50)
CARD = (255, 255, 252)
BORDER = (104, 113, 124)
ON = (31, 61, 86)
OFF = (229, 232, 226)
AND = (25, 126, 124)
OR = (202, 93, 54)
XOR = (124, 75, 164)
PAIR_COLORS = ((43, 119, 176), (220, 125, 31), (73, 152, 94), (184, 75, 112))
OPS = ("AND", "OR", "XOR")
DECISIONS = ("cause-A", "cause-B", "cause-C", "cause-D")

PAIR_X = (12, 203, 394, 585)
PAIR_W = 171
SOURCE_Y = 116
FIRST_GATE_Y = 211
SECOND_X = (126, 510)
SECOND_Y = 354
TARGET_XY = (384, 535)


def _font(size: int) -> ImageFont.ImageFont:
    path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path), size=size) if path.exists() else ImageFont.load_default()


def _apply(op: str, a: int, b: int) -> int:
    if op == "AND":
        return int(bool(a) and bool(b))
    if op == "OR":
        return int(bool(a) or bool(b))
    if op == "XOR":
        return int(bool(a) != bool(b))
    raise ValueError(op)


def evaluate(source_bits: list[int], operations: list[str]) -> list[int]:
    if len(source_bits) != 8 or len(operations) != 7:
        raise ValueError("wrong circuit size")
    if any(v not in (0, 1) for v in source_bits) or any(op not in OPS for op in operations):
        raise ValueError("invalid circuit values")
    first = [_apply(operations[i], source_bits[2 * i], source_bits[2 * i + 1]) for i in range(4)]
    second = [
        _apply(operations[4], first[0], first[1]),
        _apply(operations[5], first[2], first[3]),
    ]
    return first + second + [_apply(operations[6], second[0], second[1])]


def _qualifying_physical_pairs(operations: list[str]) -> list[int]:
    actual = [1] * 8
    target = evaluate(actual, operations)[-1]
    qualifying: list[int] = []
    for physical in range(4):
        cause, backup = 2 * physical, 2 * physical + 1
        worlds = []
        for off in ((cause,), (backup,), (cause, backup)):
            bits = actual.copy()
            for index in off:
                bits[index] = 0
            worlds.append(evaluate(bits, operations)[-1])
        if worlds[0] == target and worlds[1] == target and worlds[2] != target:
            qualifying.append(physical)
    return qualifying


_VALID_OPERATION_SETS = [
    list(ops) for ops in itertools.product(OPS, repeat=7) if len(_qualifying_physical_pairs(list(ops))) == 1
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x43415553)
    wanted_label = int(seed) % 4
    operations = rng.choice(_VALID_OPERATION_SETS)
    gold_physical = _qualifying_physical_pairs(operations)[0]
    remaining_labels = [label for label in range(4) if label != wanted_label]
    rng.shuffle(remaining_labels)
    pair_labels = [-1] * 4
    pair_labels[gold_physical] = wanted_label
    for physical, label in zip((p for p in range(4) if p != gold_physical), remaining_labels):
        pair_labels[physical] = label
    actual_nodes = evaluate([1] * 8, operations)
    return {
        "source_bits": [1] * 8,
        "operations": operations,
        "pair_labels": pair_labels,
        "actual_nodes": actual_nodes,
        "actual_target": actual_nodes[-1],
    }


def _validate(scene: dict) -> tuple[list[int], list[str], list[int], list[int], int]:
    source_bits = [int(v) for v in scene["source_bits"]]
    operations = [str(v) for v in scene["operations"]]
    pair_labels = [int(v) for v in scene["pair_labels"]]
    actual_nodes = [int(v) for v in scene["actual_nodes"]]
    actual_target = int(scene["actual_target"])
    if source_bits != [1] * 8 or len(operations) != 7 or any(op not in OPS for op in operations):
        raise ValueError("bad source or operation data")
    if sorted(pair_labels) != list(range(4)):
        raise ValueError("pair labels must be a permutation")
    expected = evaluate(source_bits, operations)
    if actual_nodes != expected or len(actual_nodes) != 7 or actual_target != expected[-1]:
        raise ValueError("inconsistent displayed actual state")
    return source_bits, operations, pair_labels, actual_nodes, actual_target


def analytic_gold(scene: dict) -> str:
    try:
        _bits, operations, pair_labels, _nodes, _target = _validate(scene)
        physical = _qualifying_physical_pairs(operations)
        return DECISIONS[pair_labels[physical[0]]] if len(physical) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _lamp(draw: ImageDraw.ImageDraw, xy: tuple[int, int], bit: int, radius: int = 20) -> None:
    x, y = xy
    fill = ON if bit else OFF
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=INK, width=3)
    _center(draw, xy, str(bit), _font(radius + 2), (255, 255, 255) if bit else INK)


def _gate(draw: ImageDraw.ImageDraw, xy: tuple[int, int], op: str, node_bit: int, label: str) -> None:
    x, y = xy
    color = {"AND": AND, "OR": OR, "XOR": XOR}[op]
    draw.rounded_rectangle((x - 48, y - 23, x + 48, y + 23), radius=8, fill=color, outline=INK, width=2)
    _center(draw, (x, y), op, _font(17), (255, 255, 255))
    draw.text((x - 47, y + 28), label, font=_font(13), fill=INK)
    _lamp(draw, (x + 75, y), node_bit, 17)


def render(scene: dict) -> Image.Image:
    source_bits, operations, pair_labels, nodes, target = _validate(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _center(draw, (384, 22), "ACTUAL CAUSAL CIRCUIT", _font(23), INK)
    _center(draw, (384, 49), "all CAUSE and BACKUP sources are ON", _font(16), INK)
    for physical, x in enumerate(PAIR_X):
        label = pair_labels[physical]
        color = PAIR_COLORS[label]
        draw.rounded_rectangle((x, 70, x + PAIR_W, 285), radius=12, fill=CARD, outline=color, width=5)
        _center(draw, (x + PAIR_W // 2, 88), f"PAIR {chr(65 + label)}", _font(18), color)
        cause_xy, backup_xy = (x + 49, SOURCE_Y), (x + 122, SOURCE_Y)
        _lamp(draw, cause_xy, source_bits[2 * physical], 18)
        _lamp(draw, backup_xy, source_bits[2 * physical + 1], 18)
        _center(draw, (cause_xy[0], 151), f"CAUSE {chr(65 + label)}", _font(12), color)
        _center(draw, (backup_xy[0], 151), "BACKUP", _font(12), color)
        draw.line((cause_xy[0], 171, x + 85, 188), fill=INK, width=3)
        draw.line((backup_xy[0], 171, x + 85, 188), fill=INK, width=3)
        _gate(draw, (x + 73, FIRST_GATE_Y), operations[physical], nodes[physical], f"N{physical + 1}")
    for branch in range(2):
        left = PAIR_X[2 * branch] + 73
        right = PAIR_X[2 * branch + 1] + 73
        gx = SECOND_X[branch]
        draw.line((left + 75, 228, gx, SECOND_Y - 28), fill=INK, width=4)
        draw.line((right + 75, 228, gx, SECOND_Y - 28), fill=INK, width=4)
        _gate(draw, (gx, SECOND_Y), operations[4 + branch], nodes[4 + branch], f"N{5 + branch}")
    draw.line((SECOND_X[0] + 75, SECOND_Y, TARGET_XY[0] - 50, 455), fill=INK, width=4)
    draw.line((SECOND_X[1] + 75, SECOND_Y, TARGET_XY[0] + 50, 455), fill=INK, width=4)
    _gate(draw, (TARGET_XY[0] - 75, 459), operations[6], target, "TARGET")
    draw.rounded_rectangle((77, 500, 691, 624), radius=12, fill=(239, 236, 249), outline=XOR, width=3)
    _center(draw, (384, 518), "MASKED CAUSE TEST", _font(18), XOR)
    _center(draw, (384, 546), "CAUSE off alone: TARGET stays", _font(17), INK)
    _center(draw, (384, 573), "BACKUP off alone: TARGET stays", _font(17), INK)
    _center(draw, (384, 601), "CAUSE + BACKUP off: TARGET flips", _font(17), INK)
    return image
