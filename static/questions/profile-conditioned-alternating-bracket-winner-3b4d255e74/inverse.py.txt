"""Pixel-only decision arm for alternating_bracket_winner.

Receives a rendered PNG and nothing else: it segments the eight bar colours,
measures each bar's drawn height above the common baseline, orders the bars by
their horizontal position and replays the declared alternating knockout.  It
imports no renderer, prompt, generator, verifier or gold code and never sees a
scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (214, 40, 40),
    "green": (34, 150, 60),
    "blue": (40, 80, 200),
    "orange": (245, 150, 25),
    "purple": (140, 55, 175),
    "brown": (130, 80, 35),
    "pink": (240, 120, 190),
    "cyan": (30, 175, 185),
}

TOLERANCE = 34
MIN_PIXELS = 2000
MIN_BAR_WIDTH, MAX_BAR_WIDTH = 45, 85
BASELINE_SLACK = 3
ABSTAIN_GAP = 6.0
ABSTAIN = "abstain"


def _blob(mask: np.ndarray):
    ys, xs = np.nonzero(mask)
    if ys.size < MIN_PIXELS:
        return None
    return int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max()), int(ys.size)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    found = []
    for name, rgb in PALETTE.items():
        delta = np.abs(arr - np.array(rgb, dtype=np.int16)).max(axis=2)
        blob = _blob(delta <= TOLERANCE)
        if blob is None:
            return ABSTAIN
        x0, x1, y0, y1, count = blob
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        if not (MIN_BAR_WIDTH <= width <= MAX_BAR_WIDTH):
            return ABSTAIN
        if count < 0.9 * width * height:
            return ABSTAIN
        found.append({"name": name, "x": (x0 + x1) / 2.0, "bottom": y1, "height": float(height)})

    if len(found) != len(PALETTE):
        return ABSTAIN
    bottoms = [bar["bottom"] for bar in found]
    if max(bottoms) - min(bottoms) > BASELINE_SLACK:
        return ABSTAIN

    found.sort(key=lambda bar: bar["x"])
    for left, right in zip(found, found[1:]):
        if right["x"] - left["x"] < MIN_BAR_WIDTH:
            return ABSTAIN

    idx = list(range(len(found)))
    for rule in ("taller", "shorter", "taller"):
        nxt = []
        for i in range(0, len(idx), 2):
            a, b = idx[i], idx[i + 1]
            if abs(found[a]["height"] - found[b]["height"]) < ABSTAIN_GAP:
                return ABSTAIN
            taller = a if found[a]["height"] > found[b]["height"] else b
            shorter = a if found[a]["height"] < found[b]["height"] else b
            nxt.append(taller if rule == "taller" else shorter)
        idx = nxt
    return str(found[idx[0]]["name"])
