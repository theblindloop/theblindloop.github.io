"""Scene sampler, Pillow renderer and analytic gold for alternating_bracket_winner.

A knockout bracket over eight bars where the advancement rule alternates:
round one keeps the TALLER bar, round two keeps the SHORTER survivor, the final
keeps the TALLER of the last two.  The champion is therefore neither an
extremum nor any fixed order statistic of the eight drawn heights.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 560
BASELINE_Y = 500
BAR_WIDTH = 62
BAR_GAP = 30
LEFT_MARGIN = 97
MIN_H, MAX_H = 70, 320
MIN_SEP = 15
QUARANTINE_MARGIN = 20.0

COLOURS = ("red", "green", "blue", "orange", "purple", "brown", "pink", "cyan")
RGB = {
    "red": (214, 40, 40),
    "green": (34, 150, 60),
    "blue": (40, 80, 200),
    "orange": (245, 150, 25),
    "purple": (140, 55, 175),
    "brown": (130, 80, 35),
    "pink": (240, 120, 190),
    "cyan": (30, 175, 185),
}

BG = (255, 255, 255)
FRAME = (70, 70, 70)
RULE = (223, 223, 223)
BRACKET = (125, 125, 125)
BASE = (25, 25, 25)

ROUND_RULES = ("taller", "shorter", "taller")
SWAPPED_RULES = ("shorter", "taller", "shorter")

BR_Y1, BR_Y2, BR_Y3, BR_STUB = 140, 95, 55, 160


def slot_x0(slot: int) -> int:
    return LEFT_MARGIN + slot * (BAR_WIDTH + BAR_GAP)


def slot_centre(slot: int) -> int:
    return slot_x0(slot) + BAR_WIDTH // 2


def _heights(rng: random.Random) -> list[int]:
    """Eight heights, shuffled, pairwise at least MIN_SEP pixels apart."""

    span = MAX_H - MIN_H + 1 - 7 * (MIN_SEP - 1)
    picks = sorted(rng.sample(range(span), 8))
    vals = [MIN_H + picks[i] + i * (MIN_SEP - 1) for i in range(8)]
    rng.shuffle(vals)
    return vals


def _matches(heights: list[int], rules: tuple[str, ...]) -> list[tuple[int, int]]:
    played: list[tuple[int, int]] = []
    idx = list(range(len(heights)))
    for rule in rules:
        nxt = []
        for i in range(0, len(idx), 2):
            a, b = idx[i], idx[i + 1]
            played.append((a, b))
            taller = a if heights[a] > heights[b] else b
            shorter = a if heights[a] < heights[b] else b
            nxt.append(taller if rule == "taller" else shorter)
        idx = nxt
    return played


def _winner(heights: list[int], rules: tuple[str, ...]) -> int:
    idx = list(range(len(heights)))
    for rule in rules:
        nxt = []
        for i in range(0, len(idx), 2):
            a, b = idx[i], idx[i + 1]
            taller = a if heights[a] > heights[b] else b
            shorter = a if heights[a] < heights[b] else b
            nxt.append(taller if rule == "taller" else shorter)
        idx = nxt
    return idx[0]


def _by_slot(scene: dict) -> list[int]:
    heights = [0] * 8
    for bar in scene["bars"]:
        heights[int(bar["slot"])] = int(bar["height"])
    return heights


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    while True:
        heights = _heights(rng)
        win = _winner(heights, ROUND_RULES)
        order = sorted(range(8), key=lambda i: -heights[i])
        rank = order.index(win)
        if rank not in (2, 3, 4, 5):
            continue
        if win == _winner(heights, SWAPPED_RULES):
            continue
        gap = min(abs(heights[a] - heights[b]) for a, b in _matches(heights, ROUND_RULES))
        if gap < MIN_SEP:
            continue
        break
    palette = list(COLOURS)
    rng.shuffle(palette)
    bars = [
        {"slot": slot, "colour": palette[slot], "height": heights[slot]}
        for slot in range(8)
    ]
    rng.shuffle(bars)
    rules = sorted(rng.sample(range(210, 471), 3))
    return {
        "canvas": [WIDTH, HEIGHT],
        "baseline_y": BASELINE_Y,
        "bar_width": BAR_WIDTH,
        "bars": bars,
        "rules": rules,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["canvas"][0]), int(scene["canvas"][1])), BG)
    draw = ImageDraw.Draw(image)
    base_y = int(scene["baseline_y"])
    width = int(scene["bar_width"])

    draw.rectangle([12, 12, image.width - 13, image.height - 13], outline=FRAME, width=2)
    for y in scene["rules"]:
        draw.line([(40, int(y)), (image.width - 41, int(y))], fill=RULE, width=2)

    for slot in range(0, 8, 2):
        left, right = slot_centre(slot), slot_centre(slot + 1)
        draw.line([(left, BR_Y1), (right, BR_Y1)], fill=BRACKET, width=2)
        draw.line([(left, BR_Y1), (left, BR_STUB)], fill=BRACKET, width=2)
        draw.line([(right, BR_Y1), (right, BR_STUB)], fill=BRACKET, width=2)
    mids1 = [(slot_centre(s) + slot_centre(s + 1)) // 2 for s in range(0, 8, 2)]
    for i in range(0, 4, 2):
        left, right = mids1[i], mids1[i + 1]
        draw.line([(left, BR_Y2), (right, BR_Y2)], fill=BRACKET, width=2)
        draw.line([(left, BR_Y2), (left, BR_Y1)], fill=BRACKET, width=2)
        draw.line([(right, BR_Y2), (right, BR_Y1)], fill=BRACKET, width=2)
    mids2 = [(mids1[0] + mids1[1]) // 2, (mids1[2] + mids1[3]) // 2]
    draw.line([(mids2[0], BR_Y3), (mids2[1], BR_Y3)], fill=BRACKET, width=2)
    draw.line([(mids2[0], BR_Y3), (mids2[0], BR_Y2)], fill=BRACKET, width=2)
    draw.line([(mids2[1], BR_Y3), (mids2[1], BR_Y2)], fill=BRACKET, width=2)
    top = (mids2[0] + mids2[1]) // 2
    draw.line([(top, BR_Y3), (top, BR_Y3 - 16)], fill=BRACKET, width=2)

    draw.line([(30, base_y + 3), (image.width - 31, base_y + 3)], fill=BASE, width=3)
    for bar in scene["bars"]:
        x0 = slot_x0(int(bar["slot"]))
        height = int(bar["height"])
        draw.rectangle(
            [x0, base_y - height + 1, x0 + width - 1, base_y],
            fill=RGB[str(bar["colour"])],
        )
    return image


def analytic_gold(scene: dict) -> str:
    heights = _by_slot(scene)
    win = _winner(heights, ROUND_RULES)
    for bar in scene["bars"]:
        if int(bar["slot"]) == win:
            return str(bar["colour"])
    raise ValueError("scene has no bar in the winning slot")


def margin(scene: dict) -> float:
    """Smallest change of one bar's drawn height that hands the final to another colour."""

    heights = _by_slot(scene)
    win = _winner(heights, ROUND_RULES)
    best = float(max(heights))
    for i in range(8):
        for j in range(8):
            if i == j:
                continue
            for target in (heights[j] - 1, heights[j] + 1):
                probe = list(heights)
                probe[i] = target
                if _winner(probe, ROUND_RULES) != win:
                    best = min(best, float(abs(target - heights[i])))
    return best


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent reorderings that leave the raster byte-identical."""

    bars = list(scene["bars"])
    out: list[tuple[str, dict]] = []
    out.append(("reverse_bar_list", {**scene, "bars": list(reversed(bars))}))
    out.append(("rotate_bar_list_by_three", {**scene, "bars": bars[3:] + bars[:3]}))
    out.append(("reverse_rule_list", {**scene, "rules": list(reversed(scene["rules"]))}))
    out.append(
        (
            "sort_bar_list_by_height",
            {**scene, "bars": sorted(bars, key=lambda b: int(b["height"]))},
        )
    )
    return out
