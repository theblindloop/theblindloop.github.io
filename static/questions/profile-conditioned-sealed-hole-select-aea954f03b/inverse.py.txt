"""Independent pixel-only inverse arm for the sealed_hole_select world.

Receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state.

The sealed loop is a fully closed colored ring whose interior white is a sealed
white region (a hole) with no connection to the surrounding white through the
ring pixels; the open loop is a gapped ring arc whose interior white connects
to the surrounding white through its gap. So the sealed white pixels -- the
white that a flood fill from the image border can never reach -- are exactly
the interior of the sealed ring. We isolate those unreachable white pixels and
report the color of the ring band that encloses them.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "sealed_hole_select-oracle-0.4.0"

_RING_RGB = {"red": (255, 0, 0), "blue": (0, 0, 255)}
_DILATE = 3


def _sealed_white_mask(arr: np.ndarray) -> np.ndarray:
    white = (arr[..., 0] > 240) & (arr[..., 1] > 240) & (arr[..., 2] > 240)
    height, width = white.shape
    seen = np.zeros((height, width), dtype=bool)
    queue: deque[tuple[int, int]] = deque()

    for y in range(height):
        for x in range(width):
            if white[y, x]:
                seen[y, x] = True
                queue.append((x, y))
                break
        if queue:
            break

    while queue:
        x, y = queue.popleft()
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nx, ny = x + dx, y + dy
            if 0 <= nx < width and 0 <= ny < height and white[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((nx, ny))

    return white & ~seen


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    sealed = _sealed_white_mask(arr)
    ys, xs = np.nonzero(sealed)
    if len(ys) == 0:
        return "red"

    height, width = sealed.shape
    counts: dict[str, int] = {"red": 0, "blue": 0}
    for y, x in zip(ys, xs):
        lo_y = max(0, y - _DILATE)
        hi_y = min(height, y + _DILATE + 1)
        lo_x = max(0, x - _DILATE)
        hi_x = min(width, x + _DILATE + 1)
        window = arr[lo_y:hi_y, lo_x:hi_x]
        red = int(((window[..., 0] == 255) & (window[..., 1] == 0) & (window[..., 2] == 0)).sum())
        blue = int(
            ((window[..., 0] == 0) & (window[..., 1] == 0) & (window[..., 2] == 255)).sum()
        )
        counts["red"] += red
        counts["blue"] += blue

    if counts["blue"] > counts["red"]:
        return "blue"
    return "red"
