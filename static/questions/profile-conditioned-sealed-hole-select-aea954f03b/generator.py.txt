"""Sealed-hole region-connectivity world renderer.

Two colored loops are drawn on a flat white canvas. The SEALED loop is a fully
closed ring; the white disk strictly inside its band is a sealed white region
(a hole) that has NO connection to the surrounding white through the loop
pixels. The OPEN loop is a wide ring arc with one large visible gap; the white
region inside its curvature is NOT sealed -- it is connected to the
surrounding white through that gap. The independent arm must recover which
colored loop is the sealed one purely from the pixels.

Decision: which colored ring (red or blue) is the SEALED loop -- the fully
closed ring that encloses a white region with no opening to the outside.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 400,
        "height": 400,
        "background": "white",
        "stroke": 11.0,
        "sealed_color": "red"|"blue",    # color of the fully closed ring
        "sealed_center": [cx, cy],
        "sealed_radius": r,
        "open_color": "red"|"blue",      # color of the gapped ring arc
        "open_center": [cx, cy],
        "open_radius": r,
        "gap_rad": g,                    # angular size of the open-ring gap
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "sealed_hole_select-0.4.0"

RING_COLORS = {"red": (255, 0, 0), "blue": (0, 0, 255)}
WHITE = (255, 255, 255)

WIDTH = 400
HEIGHT = 400
STROKE = 11.0
GAP_RAD = 1.19
QUARANTINE_MIN = 6.0
SEALED_RADIUS = (38.0, 54.0)
OPEN_RADIUS = (38.0, 54.0)


def _rng_scene(rng: random.Random) -> dict:
    sealed_color = rng.choice(["red", "blue"])
    open_color = "blue" if sealed_color == "red" else "red"

    sealed_radius = rng.uniform(*SEALED_RADIUS)
    open_radius = rng.uniform(*OPEN_RADIUS)

    half = STROKE / 2.0
    pad = 8.0
    # sealed ring first, anywhere comfortably inside
    for _attempt in range(400):
        scx = rng.uniform(sealed_radius + half + pad, WIDTH - (sealed_radius + half + pad))
        scy = rng.uniform(sealed_radius + half + pad, HEIGHT - (sealed_radius + half + pad))
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "background": "white",
            "stroke": STROKE,
            "sealed_color": sealed_color,
            "sealed_center": [scx, scy],
            "sealed_radius": sealed_radius,
            "open_color": open_color,
            "open_center": [0.0, 0.0],
            "open_radius": open_radius,
            "gap_rad": GAP_RAD,
        }
        # find an open-ring center that keeps the two rings clearly apart
        min_center_dist = sealed_radius + open_radius + STROKE + QUARANTINE_MIN
        for _inner in range(200):
            ocx = rng.uniform(open_radius + half + pad, WIDTH - (open_radius + half + pad))
            ocy = rng.uniform(open_radius + half + pad, HEIGHT - (open_radius + half + pad))
            if math.hypot(ocx - scx, ocy - scy) < min_center_dist:
                continue
            scene["open_center"] = [ocx, ocy]
            if not is_quarantined(scene):
                return scene
    return _fallback_scene()


def sample_scene(rng: random.Random, width: int = WIDTH, height: int = HEIGHT) -> dict:
    return _rng_scene(rng)


def _fallback_scene() -> dict:
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
        "stroke": STROKE,
        "sealed_color": "red",
        "sealed_center": [150.0, 200.0],
        "sealed_radius": 46.0,
        "open_color": "blue",
        "open_center": [255.0, 200.0],
        "open_radius": 46.0,
        "gap_rad": GAP_RAD,
    }


def render(scene: dict, supersample: int = 1) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    stroke = max(1, int(round(float(scene["stroke"]) * ss)))

    scx, scy = (float(v) * ss for v in scene["sealed_center"])
    srad = float(scene["sealed_radius"]) * ss
    draw.ellipse(
        [scx - srad, scy - srad, scx + srad, scy + srad],
        outline=RING_COLORS[scene["sealed_color"]],
        width=stroke,
    )

    ocx, ocy = (float(v) * ss for v in scene["open_center"])
    orad = float(scene["open_radius"]) * ss
    gap = float(scene["gap_rad"]) * ss
    # Draw the open ring as an arc covering 2*pi - gap, centred on a random start.
    gap_start = 0.0  # radians; angular position of the gap start
    gap_start_deg = math.degrees(gap_start)
    gap_deg = math.degrees(gap)
    draw.arc(
        [ocx - orad, ocy - orad, ocx + orad, ocy + orad],
        start=gap_start_deg + gap_deg,
        end=gap_start_deg + 360.0,
        fill=RING_COLORS[scene["open_color"]],
        width=stroke,
    )

    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def margin(scene: dict) -> float:
    """Robustness margin in pixels: minimum visual clearance fixing the answer.

    The boundary between the two labels is closed (sealed) versus open. The
    answer is stable as long as (a) the sealed ring is a really closed ring, so
    its interior white stays a separate sealed component; (b) the open ring has
    a wide gap, so its interior white reaches the surrounding white; and (c)
    both rings stay clearly inside the canvas and clearly apart from each
    other. We report the minimum of the painterly clearances that guarantee
    all three.
    """
    half = float(scene["stroke"]) / 2.0
    scx, scy = (float(v) for v in scene["sealed_center"])
    ocx, ocy = (float(v) for v in scene["open_center"])
    srad = float(scene["sealed_radius"])
    orad = float(scene["open_radius"])
    w = float(scene["width"])
    h = float(scene["height"])
    gap_px = float(scene["gap_rad"]) * orad

    inter = math.hypot(ocx - scx, ocy - scy) - (srad + orad + float(scene["stroke"]))
    canvas = min(
        scx - srad - half,
        w - (scx + srad) - half,
        scy - srad - half,
        h - (scy + srad) - half,
        ocx - orad - half,
        w - (ocx + orad) - half,
        ocy - orad - half,
        h - (ocy + orad) - half,
    )
    # opening width of the gapped ring and how far the arc endpoints sit apart
    open_gap_clear = min(gap_px / 3.0, 60.0)
    return max(0.0, float(min(inter, canvas, open_gap_clear)) - 2.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN


def analytic_gold(scene: dict) -> str:
    """Return the color of the sealed (fully closed) colored ring."""
    return str(scene["sealed_color"])


def latent_symmetries(scene: dict) -> Iterator[dict]:
    return iter(())
