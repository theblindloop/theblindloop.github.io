"""Deterministic scene sampler and raster renderer."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
BG = (248, 247, 242)
INK = (28, 31, 38)
PALETTE = {
    "cyan": (0, 166, 205),
    "magenta": (218, 55, 157),
    "red": (218, 48, 56),
    "gold": (232, 163, 25),
    "violet": (123, 72, 194),
    "green": (38, 158, 82),
}
CANDIDATE_COLORS = ("red", "gold", "violet", "green")


def _lengths(points: list[list[int]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for left, right in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.hypot(right[0] - left[0], right[1] - left[1]))
    return cumulative, cumulative[-1]


def _point_at(points: list[list[int]], distance: float) -> tuple[float, float]:
    cumulative, total = _lengths(points)
    distance = max(0.0, min(total, distance))
    for index in range(len(points) - 1):
        if distance <= cumulative[index + 1]:
            span = cumulative[index + 1] - cumulative[index]
            fraction = 0.0 if span == 0 else (distance - cumulative[index]) / span
            x = points[index][0] + fraction * (points[index + 1][0] - points[index][0])
            y = points[index][1] + fraction * (points[index + 1][1] - points[index][1])
            return x, y
    return float(points[-1][0]), float(points[-1][1])


def _transform_point(point: tuple[float, float], transform: int) -> tuple[float, float]:
    x, y = point
    mode = transform % 4
    if mode & 1:
        x = WIDTH - x
    if mode & 2:
        y = HEIGHT - y
    return x, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    xs = [48, 170, 286, 402, 518, 588]
    ys = [48, 108, 174, 238, 306, 372]
    xs = [value + rng.randint(-8, 8) for value in xs]
    ys = [value + rng.randint(-7, 7) for value in ys]
    points = [
        [xs[0], ys[0]], [xs[1], ys[0]],
        [xs[1], ys[1]], [xs[2], ys[1]],
        [xs[2], ys[2]], [xs[3], ys[2]],
        [xs[3], ys[3]], [xs[4], ys[3]],
        [xs[4], ys[4]], [xs[5], ys[4]],
        [xs[5], ys[5]],
    ]
    _, total = _lengths(points)
    low = rng.choice((0.15, 0.19, 0.23))
    midlow = rng.choice((0.31, 0.35, 0.39))
    midhigh = rng.choice((0.63, 0.68, 0.73))
    distances = [total * low, total * midlow, total * 0.5, total * midhigh]
    rng.shuffle(distances)
    colors = list(CANDIDATE_COLORS)
    rng.shuffle(colors)
    if rng.randrange(2):
        points = list(reversed(points))
        distances = [total - value for value in distances]
    return {
        "points": points,
        "checkpoint_distances": [round(value, 6) for value in distances],
        "checkpoint_colors": colors,
        "transform": rng.randrange(4),
    }


def render(scene: dict) -> Image.Image:
    points = scene["points"]
    distances = scene["checkpoint_distances"]
    colors = scene["checkpoint_colors"]
    transform = int(scene["transform"]) % 4
    route = [_transform_point((float(x), float(y)), transform) for x, y in points]
    checkpoints = [_transform_point(_point_at(points, float(value)), transform) for value in distances]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)

    def disk(center: tuple[float, float], radius: int, color: tuple[int, int, int]) -> None:
        x, y = center
        draw.ellipse((round(x-radius), round(y-radius), round(x+radius), round(y+radius)), fill=color)

    disk(route[0], 17, PALETTE["cyan"])
    disk(route[-1], 17, PALETTE["magenta"])
    for center, name in zip(checkpoints, colors):
        disk(center, 15, PALETTE[name])
    # Colored markers are painted first; the route is restored last so its
    # centerline remains one unbroken component through every marker.
    for center, name in zip(checkpoints, colors):
        x, y = center
        draw.ellipse((round(x-12), round(y-12), round(x+12), round(y+12)), fill=PALETTE[name])
    draw.line(route, fill=INK, width=9, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    distances = [float(value) for value in scene["checkpoint_distances"]]
    colors = list(scene["checkpoint_colors"])
    transform = int(scene["transform"])
    if not 0 <= transform % 4 < 4 or len(colors) != 4:
        raise ValueError("invalid scene")
    _, total = _lengths(points)
    scores = [abs(2.0 * value - total) for value in distances]
    winner = min(range(4), key=scores.__getitem__)
    if scores.count(scores[winner]) != 1:
        raise ValueError("ambiguous midpoint")
    return colors[winner]


def margin(scene: dict) -> float:
    points = scene["points"]
    distances = [float(value) for value in scene["checkpoint_distances"]]
    colors = list(scene["checkpoint_colors"])
    transform = int(scene["transform"])
    if len(colors) != 4 or transform < 0:
        raise ValueError("invalid scene")
    _, total = _lengths(points)
    scores = sorted(abs(2.0 * value - total) for value in distances)
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    distances = scene["checkpoint_distances"]
    colors = scene["checkpoint_colors"]
    transform = scene["transform"]
    _, total = _lengths(points)
    return bool(len(distances) != 4 or len(colors) != 4 or int(transform) < 0 or margin(scene) < 0.12 * total)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["transform"] = int(twin["transform"]) + 4
    return [("periodic_transform_alias", twin)]
