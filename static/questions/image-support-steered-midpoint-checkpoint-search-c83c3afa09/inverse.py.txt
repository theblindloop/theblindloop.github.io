"""Independent image-only inverse arm using color masks and graph geodesics."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = np.array((28, 31, 38), dtype=np.int16)
COLORS = {
    "cyan": np.array((0, 166, 205), dtype=np.int16),
    "magenta": np.array((218, 55, 157), dtype=np.int16),
    "red": np.array((218, 48, 56), dtype=np.int16),
    "gold": np.array((232, 163, 25), dtype=np.int16),
    "violet": np.array((123, 72, 194), dtype=np.int16),
    "green": np.array((38, 158, 82), dtype=np.int16),
}
CANDIDATES = ("red", "gold", "violet", "green")


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, name: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 80:
        raise ValueError(f"missing {name} marker")
    y, x = np.mean(points, axis=0)
    return int(round(y)), int(round(x))


def _component(mask: np.ndarray, origin: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    oy, ox = origin
    points = np.argwhere(mask)
    if not len(points):
        raise ValueError("missing route")
    nearest = points[np.argmin((points[:, 0]-oy)**2 + (points[:, 1]-ox)**2)]
    start = (int(nearest[0]), int(nearest[1]))
    seen = np.zeros_like(mask, dtype=bool)
    queue = deque([start])
    seen[start] = True
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny,nx] and not seen[ny,nx]:
                seen[ny,nx] = True
                queue.append((ny,nx))
    return seen


def _distances(component: np.ndarray, origin: tuple[int, int]) -> np.ndarray:
    height, width = component.shape
    oy, ox = origin
    points = np.argwhere(component)
    nearest = points[np.argmin((points[:,0]-oy)**2 + (points[:,1]-ox)**2)]
    start = (int(nearest[0]), int(nearest[1]))
    dist = np.full((height, width), -1, dtype=np.int32)
    dist[start] = 0
    queue = deque([start])
    while queue:
        y, x = queue.popleft()
        next_distance = dist[y,x] + 1
        for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
            if 0 <= ny < height and 0 <= nx < width and component[ny,nx] and dist[ny,nx] < 0:
                dist[ny,nx] = next_distance
                queue.append((ny,nx))
    return dist


def _value_near(distance: np.ndarray, center: tuple[int, int]) -> int:
    y, x = center
    y0, y1 = max(0, y-8), min(distance.shape[0], y+9)
    x0, x1 = max(0, x-8), min(distance.shape[1], x+9)
    values = distance[y0:y1, x0:x1]
    values = values[values >= 0]
    if not len(values):
        raise ValueError("marker is detached from route")
    return int(np.median(values))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 300:
        raise ValueError("invalid raster")
    centers = {name: _centroid(_mask(array, color), name) for name, color in COLORS.items()}
    black = _mask(array, INK, tolerance=10)
    route = _component(black, centers["cyan"])
    # Both endpoint plugs and all candidate plugs must share the same connected trace.
    for name in ("magenta", *CANDIDATES):
        local = _value_near(np.where(route, 0, -1), centers[name])
        if local < 0:
            raise ValueError("route is disconnected")
    from_start = _distances(route, centers["cyan"])
    from_end = _distances(route, centers["magenta"])
    scores = {}
    for name in CANDIDATES:
        a = _value_near(from_start, centers[name])
        b = _value_near(from_end, centers[name])
        scores[name] = abs(a-b)
    ranked = sorted(scores, key=lambda name: (scores[name], name))
    if scores[ranked[1]] - scores[ranked[0]] < 120:
        raise ValueError("ambiguous midpoint checkpoint")
    return ranked[0]
