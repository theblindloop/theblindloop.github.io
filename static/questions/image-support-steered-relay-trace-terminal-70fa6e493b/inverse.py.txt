"""Independent image-only inverse: recover a marker-component attachment graph."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

COLORS = {
    "navy": np.array((29, 48, 78), dtype=np.int16),
    "green": np.array((35, 171, 92), dtype=np.int16),
    "violet": np.array((119, 89, 183), dtype=np.int16),
    "cyan": np.array((25, 172, 211), dtype=np.int16),
    "orange": np.array((239, 126, 45), dtype=np.int16),
    "magenta": np.array((201, 58, 143), dtype=np.int16),
    "yellow": np.array((241, 190, 45), dtype=np.int16),
}
TOLERANCE = 8


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= TOLERANCE


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> np.ndarray:
    return component.astype(np.float64).mean(axis=0)


def _near(component: np.ndarray, marker: np.ndarray, radius: float = 22.0) -> bool:
    center = _centroid(marker)
    delta = component.astype(np.float64) - center
    return float(np.min(np.sum(delta * delta, axis=1))) <= radius * radius


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 600:
        raise ValueError("unexpected raster")

    trails = _components(_mask(array, COLORS["navy"]), 180)
    markers = {
        name: _components(_mask(array, COLORS[name]), 120)
        for name in ("green", "violet", "cyan", "orange", "magenta", "yellow")
    }
    expected_counts = {"green": 1, "violet": 1, "cyan": 2, "orange": 2, "magenta": 1, "yellow": 1}
    if len(trails) != 4:
        raise ValueError("expected four disconnected trails")
    for name, count in expected_counts.items():
        if len(markers[name]) != count:
            raise ValueError(f"expected {count} {name} marker(s)")

    stage_one_candidates = [trail for trail in trails if _near(trail, markers["green"][0])]
    if len(stage_one_candidates) != 1:
        raise ValueError("green start must identify exactly one trail")
    stage_one = stage_one_candidates[0]

    relay_hits: list[tuple[str, np.ndarray]] = []
    for relay in ("cyan", "orange"):
        relay_hits.extend((relay, marker) for marker in markers[relay] if _near(stage_one, marker))
    if len(relay_hits) != 1:
        raise ValueError("first trace must reach exactly one relay")
    relay_color, used_relay_marker = relay_hits[0]

    other_relay_markers = [
        marker for marker in markers[relay_color]
        if marker is not used_relay_marker and not _near(stage_one, marker)
    ]
    if len(other_relay_markers) != 1:
        raise ValueError("matching relay marker is not unique")
    stage_two_candidates = [
        trail for trail in trails
        if trail is not stage_one and _near(trail, other_relay_markers[0])
    ]
    if len(stage_two_candidates) != 1:
        raise ValueError("matching relay must identify one separate trail")
    stage_two = stage_two_candidates[0]

    finish_hits: list[str] = []
    for finish in ("magenta", "yellow"):
        finish_hits.extend(finish for marker in markers[finish] if _near(stage_two, marker))
    if len(finish_hits) != 1:
        raise ValueError("second trace must reach exactly one finish")
    return finish_hits[0]
