"""Deterministic renderer and independent analytic arm for a two-stage trace relay."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 520
BACKGROUND = (250, 248, 242)
NAVY = (29, 48, 78)
GREEN = (35, 171, 92)
VIOLET = (119, 89, 183)
CYAN = (25, 172, 211)
ORANGE = (239, 126, 45)
MAGENTA = (201, 58, 143)
YELLOW = (241, 190, 45)
STROKE = 11
MARKER_R = 15
QUARANTINE_MARGIN = 28.0

COLOR_RGB = {
    "green": GREEN,
    "violet": VIOLET,
    "cyan": CYAN,
    "orange": ORANGE,
    "magenta": MAGENTA,
    "yellow": YELLOW,
}


def _bent_path(a: list[int], b: list[int], variant: int) -> list[list[int]]:
    """Create a clear four-segment trace between fixed endpoints."""
    x0, y0 = a
    x1, y1 = b
    dx = x1 - x0
    sign = -1 if variant & 1 else 1
    amp = 18 + 6 * ((variant // 2) % 3)
    return [
        [x0, y0],
        [x0 + dx // 4, y0 + sign * amp],
        [x0 + dx // 2, (y0 + y1) // 2 - sign * (amp // 2)],
        [x0 + 3 * dx // 4, y1 + sign * (amp // 2)],
        [x1, y1],
    ]


def _trail(
    start: list[int],
    end: list[int],
    start_color: str,
    end_color: str,
    variant: int,
    role: str,
) -> dict[str, Any]:
    return {
        "points": _bent_path(start, end, variant),
        "start": start,
        "end": end,
        "start_color": start_color,
        "end_color": end_color,
        "role": role,
    }


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    selected_relay = "cyan" if rng.randrange(2) == 0 else "orange"
    other_relay = "orange" if selected_relay == "cyan" else "cyan"
    selected_finish = "magenta" if (int(seed) & 1) == 0 else "yellow"
    other_finish = "yellow" if selected_finish == "magenta" else "magenta"

    left_rows = [155 + rng.choice((-8, 0, 8)), 357 + rng.choice((-8, 0, 8))]
    if rng.randrange(2):
        left_rows.reverse()
    right_rows = [145 + rng.choice((-8, 0, 8)), 365 + rng.choice((-8, 0, 8))]
    selected_right_row = right_rows[rng.randrange(2)]
    other_right_row = right_rows[1] if selected_right_row == right_rows[0] else right_rows[0]

    left_reverse = bool(rng.randrange(2))
    lx0, lx1 = (315, 75) if left_reverse else (75, 315)
    right_reverse = bool(rng.randrange(2))
    rx0, rx1 = (835, 555) if right_reverse else (555, 835)

    trails = [
        _trail(
            [lx0, left_rows[0]], [lx1, left_rows[0]], "green", selected_relay,
            rng.randrange(6), "stage_one",
        ),
        _trail(
            [lx0, left_rows[1]], [lx1, left_rows[1]], "violet", other_relay,
            rng.randrange(6), "left_distractor",
        ),
        _trail(
            [rx0, selected_right_row], [rx1, selected_right_row], selected_relay,
            selected_finish, rng.randrange(6), "stage_two",
        ),
        _trail(
            [rx0, other_right_row], [rx1, other_right_row], other_relay,
            other_finish, rng.randrange(6), "right_distractor",
        ),
    ]
    return {"trails": trails}


def _marker(draw: ImageDraw.ImageDraw, center: list[int], color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse(
        (x - MARKER_R, y - MARKER_R, x + MARKER_R, y + MARKER_R),
        fill=color,
        outline=NAVY,
        width=3,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for trail in scene["trails"]:
        draw.line([tuple(point) for point in trail["points"]], fill=NAVY, width=STROKE, joint="curve")
    for trail in scene["trails"]:
        _marker(draw, trail["start"], COLOR_RGB[trail["start_color"]])
        _marker(draw, trail["end"], COLOR_RGB[trail["end_color"]])
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    trails = scene["trails"]
    stage_one = [trail for trail in trails if trail["start_color"] == "green"]
    if len(stage_one) != 1:
        raise ValueError("scene must contain exactly one green-started trail")
    relay = stage_one[0]["end_color"]
    stage_two = [trail for trail in trails if trail["start_color"] == relay]
    if len(stage_two) != 1:
        raise ValueError("relay must select exactly one second-stage trail")
    finish = stage_two[0]["end_color"]
    if finish not in {"magenta", "yellow"}:
        raise ValueError("invalid finish")
    return finish


def _segment_distance(a: list[int], b: list[int]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def margin(scene: dict[str, Any]) -> float:
    """One-sided raster clearance between marker centers on distinct trails."""
    endpoints = [point for trail in scene["trails"] for point in (trail["start"], trail["end"])]
    closest = min(
        _segment_distance(endpoints[i], endpoints[j])
        for i in range(len(endpoints))
        for j in range(i + 1, len(endpoints))
        if endpoints[i] != endpoints[j]
    )
    return float(closest - 2 * MARKER_R - STROKE)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return len(scene.get("trails", [])) != 4 or margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
