"""Deterministic scene sampler, analytic hull decision, and Pillow renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 270
PANEL_CENTERS = (105, 285, 465, 645)
PANEL_TOP, PANEL_BOTTOM = 58, 226
DOT_RADIUS = 8
TEAL = "#087f8c"
PANEL_INK = "#5b6470"
BACKGROUND = "#f4f1e8"
QUESTION = (
    "Which labeled panel has one teal dot strictly INSIDE the triangle formed "
    "by its other three teal dots? A-D run left to right."
)


def _cross(a: list[float], b: list[float], p: list[float]) -> float:
    return (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])


def _point_triangle_clearance(point: list[float], triangle: list[list[float]]) -> float:
    """Positive signed edge clearance iff point is strictly inside triangle."""
    cx = sum(p[0] for p in triangle) / 3.0
    cy = sum(p[1] for p in triangle) / 3.0
    ordered = sorted(triangle, key=lambda p: math.atan2(p[1] - cy, p[0] - cx))
    area2 = _cross(ordered[0], ordered[1], ordered[2])
    if area2 < 0:
        ordered.reverse()
    distances = []
    for index, a in enumerate(ordered):
        b = ordered[(index + 1) % 3]
        edge_length = math.hypot(b[0] - a[0], b[1] - a[1])
        distances.append(_cross(a, b, point) / edge_length)
    return min(distances)


def _panel_score(points: list[list[float]]) -> float:
    scores = []
    for index, point in enumerate(points):
        triangle = [p for j, p in enumerate(points) if j != index]
        scores.append(_point_triangle_clearance(point, triangle))
    return max(scores)


def _scores(scene: dict) -> list[float]:
    return [_panel_score(panel) for panel in scene["panels"]]


def _transform_points(
    local: list[tuple[float, float]], cx: int, cy: int, angle: float, scale_x: float, scale_y: float
) -> list[list[int]]:
    ca, sa = math.cos(angle), math.sin(angle)
    points = []
    for x, y in local:
        xx, yy = x * scale_x, y * scale_y
        points.append([int(round(cx + xx * ca - yy * sa)), int(round(cy + xx * sa + yy * ca))])
    return points


def _target_points(cx: int, cy: int, rng: random.Random, near: bool) -> list[list[int]]:
    # A broad triangle plus a strict interior point. In near-boundary scenes the
    # interior point is deliberately placed just inside one visible hull edge.
    outer = [(-0.95, 0.70), (0.98, 0.65), (0.02, -1.0)]
    if near:
        # Convex combination with only 0.10 weight on the opposite vertex.
        u = rng.uniform(0.35, 0.55)
        local_inner = (
            0.10 * outer[2][0] + 0.90 * (u * outer[0][0] + (1.0 - u) * outer[1][0]),
            0.10 * outer[2][1] + 0.90 * (u * outer[0][1] + (1.0 - u) * outer[1][1]),
        )
    else:
        weights = [rng.uniform(0.22, 0.42) for _ in range(3)]
        total = sum(weights)
        weights = [w / total for w in weights]
        local_inner = (
            sum(weights[i] * outer[i][0] for i in range(3)),
            sum(weights[i] * outer[i][1] for i in range(3)),
        )
    local = outer + [local_inner]
    return _transform_points(
        local,
        cx + rng.randint(-3, 3),
        cy + rng.randint(-3, 3),
        rng.uniform(-0.45, 0.45),
        rng.uniform(49.0, 58.0),
        rng.uniform(48.0, 57.0),
    )


def _convex_points(cx: int, cy: int, rng: random.Random) -> list[list[int]]:
    # Four strictly convex points. Independent anisotropy and rotation prevent
    # the distractors from collapsing to a single canonical square template.
    skew = rng.uniform(-0.14, 0.14)
    local = [(-0.94, -0.72 + skew), (0.86, -0.87), (1.0, 0.68 - skew), (-0.82, 0.91)]
    return _transform_points(
        local,
        cx + rng.randint(-4, 4),
        cy + rng.randint(-4, 4),
        rng.uniform(-0.55, 0.55),
        rng.uniform(46.0, 57.0),
        rng.uniform(42.0, 53.0),
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % 4
    panels: list[list[list[int]]] = []
    for index, cx in enumerate(PANEL_CENTERS):
        if index == target:
            points = _target_points(cx, 143, rng, near=(seed % 7 == 0))
        else:
            points = _convex_points(cx, 143, rng)
        rng.shuffle(points)
        panels.append(points)
    scene = {"panels": panels, "serialization_alias": seed % 2}
    if margin(scene) < 8.0:
        raise RuntimeError(f"unsafe sampled scene for seed {seed}")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = ImageFont.load_default(size=18)
    title_font = ImageFont.load_default(size=17)
    draw.text((WIDTH // 2, 25), "FIND THE INTERIOR DOT", fill="#263238", anchor="mm", font=title_font)
    for index, points in enumerate(scene["panels"]):
        cx = PANEL_CENTERS[index]
        draw.rounded_rectangle(
            (cx - 82, PANEL_TOP, cx + 82, PANEL_BOTTOM),
            radius=9,
            fill="#ffffff",
            outline=PANEL_INK,
            width=2,
        )
        draw.text((cx, 247), "ABCD"[index], fill="#263238", anchor="mm", font=label_font)
        for x, y in points:
            draw.ellipse((x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS), fill=TEAL)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] <= 0:
        raise ValueError("scene has no strict interior point")
    return "ABCD"[order[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene), reverse=True)
    return scores[0] < 5.0 or scores[0] - scores[1] < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "panels": [list(reversed(panel)) for panel in scene["panels"]],
        "serialization_alias": 1 - int(scene["serialization_alias"]),
    }
    return [("reverse_storage_order_and_flip_alias", twin)]
