"""Independent image-only inverse for four-dot interior-point search.

Only the public teal ink and geometric rule are used. No renderer, prompt,
generator, verifier, scene, metadata, or analytic-gold module is imported.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

TEAL_RGB = np.array((8, 127, 140), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        sx = sy = count = 0
        while queue:
            y, x = queue.popleft()
            sx += x
            sy += y
            count += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if count >= 80:
            found.append((sx / count, sy / count, count))
    return found


def _oriented_edge(a: tuple[float, float], b: tuple[float, float], p: tuple[float, float]) -> float:
    numerator = (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])
    return numerator / math.hypot(b[0] - a[0], b[1] - a[1])


def _inside_strength(point: tuple[float, float], tri: list[tuple[float, float]]) -> float:
    center = (sum(p[0] for p in tri) / 3.0, sum(p[1] for p in tri) / 3.0)
    ordered = sorted(tri, key=lambda p: math.atan2(p[1] - center[1], p[0] - center[0]))
    signed_area = (
        (ordered[1][0] - ordered[0][0]) * (ordered[2][1] - ordered[0][1])
        - (ordered[1][1] - ordered[0][1]) * (ordered[2][0] - ordered[0][0])
    )
    if signed_area < 0:
        ordered = list(reversed(ordered))
    return min(_oriented_edge(ordered[i], ordered[(i + 1) % 3], point) for i in range(3))


def _set_score(points: list[tuple[float, float]]) -> float:
    strengths = []
    for index, point in enumerate(points):
        triangle = [other for j, other in enumerate(points) if j != index]
        strengths.append(_inside_strength(point, triangle))
    return max(strengths)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    mask = np.linalg.norm(arr - TEAL_RGB, axis=2) < 45.0
    components = sorted(_components(mask), key=lambda item: item[0])
    if len(components) != 16:
        raise ValueError(f"expected 16 teal dots, found {len(components)}")
    panels = [[(x, y) for x, y, _ in components[start : start + 4]] for start in range(0, 16, 4)]
    scores = [_set_score(panel) for panel in panels]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] < 4.0:
        raise ValueError("no panel has a safely interior dot")
    if scores[order[0]] - scores[order[1]] < 6.0:
        raise ValueError("ambiguous interior-point winner")
    return "ABCD"[order[0]]
