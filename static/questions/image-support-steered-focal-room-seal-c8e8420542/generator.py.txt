"""Deterministic raster world for a focal planar face-merging question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "focal_room_seal-0.1.0"
BACKGROUND = (255, 255, 255)
WALL = (38, 49, 60)
REMOTE_INK = (226, 232, 238)
PALETTE = (
    (221, 67, 58),
    (38, 128, 218),
    (24, 165, 137),
    (181, 67, 151),
)
CANVAS = (512, 512)
SUPERSAMPLE = 4


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced planar topologies and randomized rendered nuisance fields."""

    rng = random.Random(int(seed))
    bucket = int(seed) % 8
    topology = "sealed" if bucket in {0, 1, 4, 5} else "bypass"
    tight = bucket in {0, 3, 6}
    cell_x = rng.randint(1, 6)
    cell_y = rng.randint(1, 6)
    origin_x = (cell_x + 0.5) * 64.0 + rng.uniform(-2.5, 2.5)
    origin_y = (cell_y + 0.5) * 64.0 + rng.uniform(-2.5, 2.5)
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "origin_x": round(origin_x, 3),
        "origin_y": round(origin_y, 3),
        "topology": topology,
        "orientation": rng.randrange(4),
        "frame_side": 88.0 if tight else 96.0,
        "stroke_width": 6.0 if tight else 8.0,
        "divider_width": 6.0 if tight else 8.0,
        "gate_half": 8.0 if tight else 12.0,
        "bypass_gap": 7.0 if tight else 16.0,
        "node_color": rng.choice(PALETTE),
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _rotate(point: tuple[float, float], orientation: int) -> tuple[float, float]:
    x, y = point
    turn = int(orientation) % 4
    if turn == 0:
        return x, y
    if turn == 1:
        return -y, x
    if turn == 2:
        return -x, -y
    return y, -x


def _point(scene: dict[str, Any], local: tuple[float, float], scale: int) -> tuple[int, int]:
    rx, ry = _rotate(local, int(scene["orientation"]))
    return (
        int(round((float(scene["origin_x"]) + rx) * scale)),
        int(round((float(scene["origin_y"]) + ry) * scale)),
    )


def _line(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    points: list[tuple[float, float]],
    fill: tuple[int, int, int],
    width: float,
    scale: int,
) -> None:
    draw.line(
        [_point(scene, point, scale) for point in points],
        fill=fill,
        width=max(1, int(round(float(width) * scale))),
        joint="curve",
    )


def _draw_remote_marks(draw: ImageDraw.ImageDraw, scene: dict[str, Any], scale: int) -> None:
    """Add pale remote marks that are deliberately ignored by the pixel oracle."""

    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((34, 34), (478, 34), (34, 478), (478, 478)):
        x = base_x + rng.randint(-7, 7)
        y = base_y + rng.randint(-7, 7)
        radius = (2 + rng.randrange(2)) * scale
        draw.ellipse(
            (x * scale - radius, y * scale - radius, x * scale + radius, y * scale + radius),
            fill=REMOTE_INK,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the compact room glyph into a final RGB PNG."""

    width = int(scene["width"])
    height = int(scene["height"])
    orientation = int(scene["orientation"])
    frame_side = float(scene["frame_side"])
    stroke_width = float(scene["stroke_width"])
    divider_width = float(scene["divider_width"])
    gate_half = float(scene["gate_half"])
    bypass_gap = float(scene["bypass_gap"])
    topology = str(scene["topology"])
    node_color = tuple(int(value) for value in scene["node_color"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, scene, scale)

    half = frame_side / 2.0
    # A closed, single connected outer wall encloses the two white chambers.
    _line(
        draw,
        scene,
        [(-half, -half), (half, -half), (half, half), (-half, half), (-half, -half)],
        WALL,
        stroke_width,
        scale,
    )

    # The horizontal divider is complete in the sealed class. In the bypass
    # class its right endpoint visibly stops short of the outer wall.
    right_end = half if topology == "sealed" else half - bypass_gap
    _line(draw, scene, [(-half, 0.0), (-gate_half, 0.0)], WALL, divider_width, scale)
    _line(draw, scene, [(gate_half, 0.0), (right_end, 0.0)], WALL, divider_width, scale)

    # The saturated tile is the only material across the central divider gap.
    # It is painted last, but its endpoints overlap the dark wall endpoints.
    _line(
        draw,
        scene,
        [(-gate_half, 0.0), (gate_half, 0.0)],
        node_color,
        divider_width,
        scale,
    )

    # Keep orientation in the direct render path so the visible transform is
    # explicit to the latent-field audit.
    _ = orientation
    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the single label input: whether the tile is the sole separator."""

    return "creates_connection" if str(scene["topology"]) == "sealed" else "has_bypass"


def _clearance(scene: dict[str, Any]) -> float:
    half = float(scene["frame_side"]) / 2.0
    divider = float(scene["divider_width"])
    gate = float(scene["gate_half"])
    side_clearance = half - gate - divider / 2.0
    if str(scene["topology"]) == "bypass":
        boundary_clearance = float(scene["bypass_gap"])
    else:
        boundary_clearance = gate - 2.0
    return min(side_clearance, boundary_clearance)


def margin(scene: dict[str, Any]) -> float:
    clearance = _clearance(scene)
    return clearance if analytic_gold(scene) == "creates_connection" else -clearance


def is_quarantined(scene: dict[str, Any]) -> bool:
    return _clearance(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No intentionally pixel-identical alternate latent transforms are declared."""

    del scene
    return []
