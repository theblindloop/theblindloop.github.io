"""Independent image-only oracle for the focal planar face-merging question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    points = np.argwhere(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    found: list[np.ndarray] = []
    for y, x in points:
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = [(y, x)]
        seen[y, x] = True
        values: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            values.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(values) >= minimum:
            found.append(np.asarray(values, dtype=np.int32))
    return found


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    if radius <= 0:
        return mask.copy()
    height, width = mask.shape
    padded = np.pad(mask, radius, mode="constant")
    output = np.zeros_like(mask)
    for dy in range(2 * radius + 1):
        for dx in range(2 * radius + 1):
            output |= padded[dy : dy + height, dx : dx + width]
    return output


def _colored_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    high = values.max(axis=2)
    low = values.min(axis=2)
    return (high - low >= 55) & (high >= 100) & (low <= 240)


def _white_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    return (values.min(axis=2) >= 242) & ((values.max(axis=2) - values.min(axis=2)) <= 22)


def _enclosed_white_faces(white: np.ndarray, minimum: int = 30) -> int:
    """Count white components not connected to the crop boundary."""

    height, width = white.shape
    outside = np.zeros_like(white, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        for y in (0, height - 1):
            if white[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if white[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < height and 0 <= nx < width and white[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))

    enclosed = white & ~outside
    return len(_components(enclosed, minimum=minimum))


def decision_from_image(image: Image.Image) -> str:
    """Infer face connectivity from the PNG, without renderer state."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    colored = _colored_mask(array)
    markers = _components(colored, minimum=24)
    if len(markers) != 1:
        raise ValueError(f"expected one saturated wall tile, found {len(markers)}")
    marker = markers[0]
    center_y = int(round(float(marker[:, 0].mean())))
    center_x = int(round(float(marker[:, 1].mean())))
    half = 76
    height, width = array.shape[:2]
    y0 = max(0, center_y - half)
    y1 = min(height, center_y + half + 1)
    x0 = max(0, center_x - half)
    x1 = min(width, center_x + half + 1)
    crop = array[y0:y1, x0:x1]
    crop_color = colored[y0:y1, x0:x1]

    before = _enclosed_white_faces(_white_mask(crop))
    erased = crop.copy()
    erased[_dilate(crop_color, radius=2)] = (255, 255, 255)
    after = _enclosed_white_faces(_white_mask(erased))
    if before < 1 or after < 1:
        raise ValueError("foreground support is too damaged for a face-merge decision")
    if before >= 2 and after < before:
        return "creates_connection"
    if before == 1 and after == 1:
        return "has_bypass"
    raise ValueError(f"ambiguous enclosed-face counts before={before} after={after}")
