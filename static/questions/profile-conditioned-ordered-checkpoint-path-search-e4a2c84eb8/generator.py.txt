"""Deterministic latent sampler and raster renderer.

The gold uses the stored visible bead order.  The image-only inverse arm does
not import this module and instead traces raster connectivity from cyan.
"""

from __future__ import annotations

import copy
import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 610
LABELS = ("A", "B", "C", "D")
TARGET_ORDER = ("red", "orange", "blue")

BACKGROUND = (232, 238, 245)
CARD_FILL = (255, 255, 255)
BORDER = (118, 130, 145)
PATH = (34, 42, 53)
COLORS = {
    "cyan": (25, 181, 190),
    "red": (218, 57, 49),
    "orange": (242, 146, 32),
    "blue": (51, 101, 211),
}

CARD_BOXES = (
    (30, 50, 330, 275),
    (390, 50, 690, 275),
    (30, 335, 330, 560),
    (390, 335, 690, 560),
)

# Four non-self-intersecting paths with distinct screen orientations.  Slots are
# points 1..3; point 0 is the cyan start and point 4 is the plain far endpoint.
TEMPLATES = (
    ((48, 133), (96, 76), (146, 154), (201, 87), (252, 145)),
    ((252, 118), (207, 176), (155, 88), (99, 161), (48, 103)),
    ((150, 43), (87, 86), (207, 126), (101, 171), (180, 207)),
    ((132, 207), (210, 166), (91, 126), (205, 82), (120, 43)),
)


def _global_points(box: tuple[int, int, int, int], template: Iterable[tuple[int, int]], dx: int, dy: int) -> list[list[int]]:
    x0, y0, _, _ = box
    return [[x0 + x + dx, y0 + y + dy] for x, y in template]


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; consecutive seeds cycle the winning label."""
    rng = random.Random(int(seed))
    winner_index = int(seed) % len(LABELS)
    distractors = [
        list(order)
        for order in itertools.permutations(TARGET_ORDER)
        if tuple(order) != TARGET_ORDER
    ]
    rng.shuffle(distractors)
    orders: list[list[str]] = []
    next_distractor = 0
    for index in range(4):
        if index == winner_index:
            orders.append(list(TARGET_ORDER))
        else:
            orders.append(distractors[next_distractor])
            next_distractor += 1

    template_ids = list(range(4))
    rng.shuffle(template_ids)
    cards = []
    for index, (label, box) in enumerate(zip(LABELS, CARD_BOXES, strict=True)):
        # One small whole-path translation is independent of target identity.
        dx, dy = rng.randint(-5, 5), rng.randint(-4, 4)
        points = _global_points(box, TEMPLATES[template_ids[index]], dx, dy)
        cards.append(
            {
                "label": label,
                "box": list(box),
                "segments": [[points[j], points[j + 1]] for j in range(4)],
                "start": points[0],
                "slots": points[1:4],
                "order": orders[index],
            }
        )
    return {"width": WIDTH, "height": HEIGHT, "cards": cards}


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _scaled_point(point: list[int] | tuple[int, int], scale: int) -> tuple[int, int]:
    return int(point[0] * scale), int(point[1] * scale)


def _disk(draw: ImageDraw.ImageDraw, point: list[int], radius: float, fill: tuple[int, int, int], scale: int) -> None:
    x, y = _scaled_point(point, scale)
    r = int(round(radius * scale))
    draw.ellipse((x - r, y - r, x + r, y + r), fill=fill)


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = _font(24 * scale)
    for card in scene["cards"]:
        box = tuple(int(value * scale) for value in card["box"])
        draw.rounded_rectangle(
            box,
            radius=17 * scale,
            fill=CARD_FILL,
            outline=BORDER,
            width=3 * scale,
        )
        x0, y0, _, _ = card["box"]
        draw.text((int((x0 + 13) * scale), int((y0 + 8) * scale)), card["label"], fill=PATH, font=font)

        for segment in card["segments"]:
            draw.line(
                [_scaled_point(segment[0], scale), _scaled_point(segment[1], scale)],
                fill=PATH,
                width=9 * scale,
            )
        # Sorted joint painting makes segment-list order a true raster alias.
        joints = sorted({tuple(point) for segment in card["segments"] for point in segment})
        for point in joints:
            _disk(draw, list(point), 4.5, PATH, scale)
        _disk(draw, card["start"], 14.0, COLORS["cyan"], scale)
        for point, color_name in zip(card["slots"], card["order"], strict=True):
            _disk(draw, point, 12.5, COLORS[color_name], scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def _scores(scene: dict) -> dict[str, int]:
    return {
        card["label"]: sum(
            visible == expected
            for visible, expected in zip(card["order"], TARGET_ORDER, strict=True)
        )
        for card in scene["cards"]
    }


def analytic_gold(scene: dict) -> str:
    winners = [card["label"] for card in scene["cards"] if tuple(card["order"]) == TARGET_ORDER]
    if len(winners) != 1:
        raise ValueError(f"expected one ordered-path winner, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    if margin(scene) < 1.0:
        return True
    for card in scene["cards"]:
        checkpoints = [card["start"], *card["slots"]]
        for left, right in zip(checkpoints, checkpoints[1:]):
            if math.dist(left, right) < 38.0:
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for card in twin["cards"]:
        card["segments"] = list(reversed(card["segments"]))
    return [("reverse_undirected_segment_storage", twin)]
