"""Independent pixel-only inverse for the state-change-load world.

This module duplicates only the public raster geometry and bitmap glyphs. It
does not import the renderer, prompt, generator, verifier, or analytic gold.
"""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 6
TILE_COUNT = 4
STATE_COUNT = 3
STEP_LABELS = tuple(f"P{index}->P{index + 1}" for index in range(1, PANEL_COUNT))
CANVAS = (1940, 760)
STATE_INK = (44, 101, 132)
IDENTITY_INK = (36, 51, 64)
PANEL_X0 = 28
PANEL_WIDTH = 300
PANEL_GAP = 12
ROW_TOP = 252
ROW_STEP = 78
LABEL_X_OFFSET = 34
LABEL_Y_OFFSET = 20
LABEL_SCALE = 3
STATE_BOX_X_OFFSET = 214
STATE_BOX_Y_OFFSET = 20
STATE_BOX_SIZE = 20
KEY_XS = (435, 700, 965)
KEY_PATTERN_Y = 98

LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)


def _label_points(label_index: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * LABEL_SCALE + dx, y * LABEL_SCALE + dy)
        for y, row in enumerate(LABEL_PATTERNS[label_index])
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(LABEL_SCALE)
        for dy in range(LABEL_SCALE)
    )


LABEL_POINTS = tuple(_label_points(index) for index in range(TILE_COUNT))

STATE_CODE_PATTERNS = (
    ("01110", "10001", "10011", "10101", "11001", "10001", "01110"),
    ("00100", "01100", "00100", "00100", "00100", "00100", "01110"),
    ("01110", "10001", "00001", "00010", "00100", "01000", "11111"),
)
STATE_CODE_SCALE = 2
STATE_CODE_POINTS = tuple(
    frozenset(
        (x * STATE_CODE_SCALE + dx, y * STATE_CODE_SCALE + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(STATE_CODE_SCALE)
        for dy in range(STATE_CODE_SCALE)
    )
    for pattern in STATE_CODE_PATTERNS
)
STATE_CODE_X_OFFSET = 28
STATE_CODE_Y_OFFSET = 3


def _state_points() -> tuple[frozenset[tuple[int, int]], ...]:
    low = {(x, y) for x in range(6, 14) for y in range(6, 14)}
    mid = {
        (x, y)
        for x0 in (1, 15)
        for y0 in (1, 15)
        for x in range(x0, x0 + 4)
        for y in range(y0, y0 + 4)
    }
    lengths = (4, 8, 12, 16, 12, 8, 4)
    high: set[tuple[int, int]] = set()
    for row, length in enumerate(lengths):
        left = (20 - length) // 2
        high.update((x, row + 6) for x in range(left, left + length))
    points = (frozenset(low), frozenset(mid), frozenset(high))
    if any(len(value) != 64 for value in points):
        raise AssertionError("state glyphs must have equal ink area")
    return points


STATE_POINTS = _state_points()


def _read_label(frame: Image.Image, origin: tuple[int, int]) -> int:
    ox, oy = origin
    observed = frozenset(
        (x, y)
        for x in range(15)
        for y in range(21)
        if frame.getpixel((ox + x, oy + y)) == IDENTITY_INK
    )
    matches = [index for index, points in enumerate(LABEL_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("identity glyph is missing, altered, or ambiguous")
    return matches[0]


def _read_state(frame: Image.Image, origin: tuple[int, int]) -> frozenset[tuple[int, int]]:
    ox, oy = origin
    return frozenset(
        (x, y)
        for x in range(STATE_BOX_SIZE)
        for y in range(STATE_BOX_SIZE)
        if frame.getpixel((ox + x, oy + y)) == STATE_INK
    )


def _decode_key(frame: Image.Image) -> dict[frozenset[tuple[int, int]], int]:
    mapping: dict[frozenset[tuple[int, int]], int] = {}
    for x in KEY_XS:
        signature = _read_state(frame, (x, KEY_PATTERN_Y))
        if signature not in STATE_POINTS or signature in mapping:
            raise ValueError("state key is missing or ambiguous")
        ox = x + STATE_CODE_X_OFFSET
        oy = KEY_PATTERN_Y + STATE_CODE_Y_OFFSET
        observed = frozenset(
            (dx, dy)
            for dx in range(10)
            for dy in range(14)
            if frame.getpixel((ox + dx, oy + dy)) == IDENTITY_INK
        )
        codes = [code for code, points in enumerate(STATE_CODE_POINTS) if observed == points]
        if len(codes) != 1:
            raise ValueError("state key code is missing or ambiguous")
        mapping[signature] = codes[0]
    if len(mapping) != STATE_COUNT:
        raise ValueError("state key does not expose three distinct glyphs")
    return mapping


def decision_from_image(image: Image.Image) -> str:
    """Decode identity/state glyphs from one final PNG and choose a step."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_key = _decode_key(frame)
    tables: list[dict[int, int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        table: dict[int, int] = {}
        for row in range(TILE_COUNT):
            top = ROW_TOP + row * ROW_STEP
            label = _read_label(frame, (panel_x + LABEL_X_OFFSET, top + LABEL_Y_OFFSET))
            if label in table:
                raise ValueError("a panel repeats one identity")
            signature = _read_state(
                frame,
                (panel_x + STATE_BOX_X_OFFSET, top + STATE_BOX_Y_OFFSET),
            )
            if signature not in state_key:
                raise ValueError("state glyph is missing or not declared in the key")
            table[label] = state_key[signature]
        if set(table) != set(range(TILE_COUNT)):
            raise ValueError("a panel does not contain all four identities")
        tables.append(table)

    counts = [
        sum(tables[step - 1][label] != tables[step][label] for label in range(TILE_COUNT))
        for step in range(1, PANEL_COUNT)
    ]
    best = max(counts)
    winners = [index for index, value in enumerate(counts) if value == best]
    if len(winners) != 1:
        raise ValueError("image does not expose one unique change-load winner")
    return STEP_LABELS[winners[0]]
