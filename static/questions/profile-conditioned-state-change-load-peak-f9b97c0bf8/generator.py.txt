def _label_points(label_index: int) -> frozenset[tuple[int, int]]:
    pattern = LABEL_PATTERNS[label_index]
    return frozenset(
        (x * LABEL_SCALE + dx, y * LABEL_SCALE + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(LABEL_SCALE)
        for dy in range(LABEL_SCALE)
    )


LABEL_POINTS = tuple(_label_points(index) for index in range(TILE_COUNT))

STATE_CODE_PATTERNS: tuple[tuple[str, ...], ...] = (
    ("01110", "10001", "10011", "10101", "11001", "10001", "01110"),
    ("00100", "01100", "00100", "00100", "00100", "00100", "01110"),
    ("01110", "10001", "00001", "00010", "00100", "01000", "11111"),
)
STATE_CODE_SCALE = 2


def _state_code_points(code: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * STATE_CODE_SCALE + dx, y * STATE_CODE_SCALE + dy)
        for y, row in enumerate(STATE_CODE_PATTERNS[code])
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(STATE_CODE_SCALE)
        for dy in range(STATE_CODE_SCALE)
    )


STATE_CODE_POINTS = tuple(_state_code_points(index) for index in range(STATE_COUNT))
STATE_CODE_X_OFFSET = 28
STATE_CODE_Y_OFFSET = 3


def _state_points() -> tuple[frozenset[tuple[int, int]], ...]:
    # Every state marker uses exactly 64 ink pixels. The different spatial
    # signatures, not color area, carry the state identity.
    low = {(x, y) for x in range(6, 14) for y in range(6, 14)}
    mid = {
        (x, y)
        for x0 in (1, 15)
        for y0 in (1, 15)
        for x in range(x0, x0 + 4)
        for y in range(y0, y0 + 4)
    }
    lengths = (4, 8, 12, 16, 12, 8, 4)
    high: set[tuple[int, int]] = set()
    for row, length in enumerate(lengths):
        left = (20 - length) // 2
        high.update((x, row + 6) for x in range(left, left + length))
    points = (frozenset(low), frozenset(mid), frozenset(high))
    if any(len(value) != 64 for value in points):
        raise AssertionError("state glyphs must have equal ink area")
    return points


STATE_POINTS = _state_points()


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ) if bold else (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
    )
    for name in names:
        if Path(name).is_file():
            return ImageFont.truetype(name, size)
    return ImageFont.load_default()


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    text: str,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (center[0] - (box[2] - box[0]) / 2, center[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _draw_points(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    points: frozenset[tuple[int, int]],
    fill: tuple[int, int, int],
) -> None:
    ox, oy = origin
    for x, y in points:
        draw.point((ox + x, oy + y), fill=fill)


def _validate_states(scene: dict) -> list[list[int]]:
    states = scene.get("states")
    if not isinstance(states, list) or len(states) != PANEL_COUNT:
        raise ValueError("states must contain six panels")
    normalized: list[list[int]] = []
    for panel in states:
        if not isinstance(panel, list) or len(panel) != TILE_COUNT:
            raise ValueError("each state panel must contain four labels")
        values = [int(value) for value in panel]
        if any(value not in range(STATE_COUNT) for value in values):
            raise ValueError("states must use LOW=0, MID=1, HIGH=2")
        normalized.append(values)
    return normalized


def _change_counts(states: list[list[int]]) -> list[int]:
    return [
        sum(states[step - 1][label] != states[step][label] for label in range(TILE_COUNT))
        for step in range(1, PANEL_COUNT)
    ]


def sample_scene(seed: int) -> dict:
    """Sample one scene with a unique change-load winner."""
    seed = int(seed)
    logic_rng = random.Random(seed * 1_000_003 + 17)
    style_rng = random.Random((seed + 104_729) * 1_000_033 + 31)
    requested_winner = seed % (PANEL_COUNT - 1)

    for _ in range(20_000):
        states = [
            [logic_rng.randrange(STATE_COUNT) for _ in range(TILE_COUNT)]
            for _ in range(PANEL_COUNT)
        ]
        counts = _change_counts(states)
        best = max(counts)
        winners = [index for index, value in enumerate(counts) if value == best]
        if len(winners) == 1 and winners[0] == requested_winner:
            break
    else:
        raise RuntimeError("could not sample a unique requested change-load winner")

    row_orders: list[list[int]] = []
    for _panel in range(PANEL_COUNT):
        order = list(range(TILE_COUNT))
        style_rng.shuffle(order)
        row_orders.append(order)
    state_glyph_order = list(range(STATE_COUNT))
    style_rng.shuffle(state_glyph_order)
    scene = {
        "states": states,
        "row_orders": row_orders,
        "state_glyph_order": state_glyph_order,
        "panel_tint": style_rng.randrange(len(PANEL_TINTS)),
    }
    if analytic_gold(scene) != STEP_LABELS[requested_winner]:
        raise AssertionError("sampler failed to create requested winner")
    return scene


def render(scene: dict) -> Image.Image:
    states = _validate_states(scene)
    row_orders = scene.get("row_orders")
    if not isinstance(row_orders, list) or len(row_orders) != PANEL_COUNT:
        raise ValueError("row_orders must contain six permutations")
    glyph_order = [int(value) for value in scene.get("state_glyph_order", [])]
    if sorted(glyph_order) != list(range(STATE_COUNT)):
        raise ValueError("state_glyph_order must be a permutation")
    tint = PANEL_TINTS[int(scene.get("panel_tint", 0)) % len(PANEL_TINTS)]

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(26, bold=True)
    section_font = _font(16, bold=True)
    body_font = _font(15)
    small_font = _font(13)
    label_font = _font(18, bold=True)

    draw.text((28, 18), "STATE CHANGE LOAD", font=title_font, fill=INK)
    draw.text(
        (28, 54),
        "Follow the same four identity glyphs across the ordered checkpoints.",
        font=small_font,
        fill=MUTED,
    )
    draw.text((28, 82), "VISIBLE STATE KEY", font=section_font, fill=INK)
    state_to_slot = {state: slot for slot, state in enumerate(glyph_order)}
    for state in range(STATE_COUNT):
        key_origin = (KEY_XS[state_to_slot[state]], KEY_PATTERN_Y)
        draw.rectangle(
            (
                key_origin[0] - 1,
                key_origin[1] - 1,
                key_origin[0] + STATE_BOX_SIZE,
                key_origin[1] + STATE_BOX_SIZE,
            ),
            outline=OUTLINE,
            width=1,
        )
        _draw_points(draw, key_origin, STATE_POINTS[state], STATE_INK)
        _draw_points(
            draw,
            (key_origin[0] + STATE_CODE_X_OFFSET, key_origin[1] + STATE_CODE_Y_OFFSET),
            STATE_CODE_POINTS[state],
            IDENTITY_INK,
        )
        _centered_text(
            draw,
            (key_origin[0] + 67, key_origin[1] + 10),
            STATE_NAMES[state],
            body_font,
            INK,
        )

    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH, PANEL_TOP + PANEL_HEIGHT),
            radius=12,
            fill=tint,
            outline=OUTLINE,
            width=2,
        )
        draw.text((panel_x + 16, PANEL_TOP + 14), f"P{panel + 1}", font=section_font, fill=INK)
        draw.text((panel_x + 205, PANEL_TOP + 17), "ID / STATE", font=small_font, fill=MUTED)
        draw.line(
            (panel_x + 16, PANEL_TOP + 55, panel_x + PANEL_WIDTH - 16, PANEL_TOP + 55),
            fill=OUTLINE,
            width=1,
        )
        for row, label_index in enumerate(row_orders[panel]):
            label_index = int(label_index)
            if label_index not in range(TILE_COUNT):
                raise ValueError("row order contains an invalid identity")
            top = ROW_TOP + row * ROW_STEP
            draw.rounded_rectangle(
                (
                    panel_x + CARD_X_OFFSET,
                    top,
                    panel_x + CARD_X_OFFSET + CARD_WIDTH,
                    top + CARD_HEIGHT,
                ),
                radius=8,
                fill=CARD,
                outline=OUTLINE,
                width=1,
            )
            label_origin = (panel_x + LABEL_X_OFFSET, top + LABEL_Y_OFFSET)
            _draw_points(draw, label_origin, LABEL_POINTS[label_index], IDENTITY_INK)
            draw.text(
                (panel_x + 58, top + 18),
                LABELS[label_index],
                font=label_font,
                fill=IDENTITY_INK,
            )
            draw.line(
                (panel_x + 158, top + 10, panel_x + 158, top + CARD_HEIGHT - 10),
                fill=(215, 224, 230),
                width=1,
            )
            state = states[panel][label_index]
            state_origin = (panel_x + STATE_BOX_X_OFFSET, top + STATE_BOX_Y_OFFSET)
            draw.rectangle(
                (
                    state_origin[0] - 1,
                    state_origin[1] - 1,
                    state_origin[0] + STATE_BOX_SIZE,
                    state_origin[1] + STATE_BOX_SIZE,
                ),
                outline=OUTLINE,
                width=1,
            )
            _draw_points(draw, state_origin, STATE_POINTS[state], STATE_INK)

    for step, label in enumerate(STEP_LABELS):
        left = PANEL_X0 + step * (PANEL_WIDTH + PANEL_GAP) + PANEL_WIDTH
        right = PANEL_X0 + (step + 1) * (PANEL_WIDTH + PANEL_GAP)
        center = (left + right) / 2
        y = PANEL_TOP + PANEL_HEIGHT + 18
        draw.line((left + 4, y, right - 6, y), fill=OUTLINE, width=2)
        draw.polygon(((right - 6, y), (right - 14, y - 5), (right - 14, y + 5)), fill=OUTLINE)
        _centered_text(draw, (center, y + 18), label, small_font, MUTED)

    draw.text(
        (28, 735),
        "A state change means the visible LOW/MID/HIGH glyph differs for the same identity.",
        font=small_font,
        fill=MUTED,
    )
    return image


def analytic_gold(scene: dict) -> str:
    states = _validate_states(scene)
    counts = _change_counts(states)
    best = max(counts)
    winners = [index for index, value in enumerate(counts) if value == best]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique change-load winner")
    return STEP_LABELS[winners[0]]


def margin(scene: dict) -> float:
    values = sorted(_change_counts(_validate_states(scene)), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    values = _change_counts(_validate_states(scene))
    best = max(values)
    return values.count(best) != 1 or margin(scene) <= 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; all analytic label inputs are visible."""
    return []
