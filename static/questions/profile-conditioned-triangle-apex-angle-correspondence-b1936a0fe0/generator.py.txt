"""Triangle apex-angle correspondence renderer (paper-semantic correspondence).

A cross-panel shape-correspondence task. A single ``reference`` isosceles
triangle (drawn in neutral gray, apex pointing up) is shown in the top panel.
The bottom panel shows three ``candidate`` isosceles triangles (red, green,
blue), also apex-up, each with a *different* apex angle. Exactly one candidate
has the same apex angle as the reference (it is the same object's shape,
re-arranged across the divider). The question asks: after matching by apex
angle, what is the color of that matched candidate?

The matching attribute is the shape (apex angle) of the triangle, not its color,
position, or size; the reference is neutral gray so color alone cannot give the
answer. The answer attribute (candidate color) is read only after the shape
correspondence is established.

Latent state z (all exposed scene fields are rendered -- nothing hidden):

    ref_b, ref_h : reference triangle base width / height (pixels) -> its apex angle
    ref_cx, ref_cy: reference triangle apex position
    cand_configs: per-candidate [base, height] -> each candidate's apex angle
    cand_colors : per-candidate color name (red/green/blue), the bound attribute
    cand_centers: [x, y] apex of each candidate triangle
    width, height: canvas

The analytic gold is derived from these rendered fields: find the candidate
whose [base, height] equals the reference's (same apex angle), then report its
color. Because every such field is drawn into the raster (base and height set
the visible triangle extent), the gold is recoverable from pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "triangle_apex_angle-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)

COLOR_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (230, 0, 0),
    "green": (0, 180, 0),
    "blue": (0, 0, 235),
}
REF_GRAY = (130, 130, 130)

# (base, height) pixel extents; each gives a distinct apex angle for an
# apex-up isosceles triangle. Latent apex angles and pixel-measured angles are
# listed for the audit of separation.
ANGLE_MENU: list[tuple[int, int]] = [
    (100, 140),
    (140, 140),
    (180, 140),
    (220, 140),
    (260, 140),
]

# Geometry (public visual spec; constant across scenes for counterbalance).
WIDTH = 760
HEIGHT = 440
DIVIDER_Y = 230
DIVIDER_THICK = 3
REF_CX = 380
REF_TOP = 60
CAND_BASE_Y = 400
SLOTS = [180, 380, 580]

_MARGIN_FLOOR = 8.0


def apex_angle(b: int, h: int) -> float:
    """Apex angle (degrees) of an apex-up isosceles triangle of base b, height h."""
    return 2.0 * math.degrees(math.atan((b / 2.0) / h))


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref = rng.choice(ANGLE_MENU)
    others = [c for c in ANGLE_MENU if c != ref]
    c1 = rng.choice(others)
    others2 = [c for c in others if c != c1]
    c2 = rng.choice(others2)
    cfgs = [ref, c1, c2]

    colors = ["red", "green", "blue"]
    matched_color = colors[seed % 3]
    other_colors = [c for c in colors if c != matched_color]

    triples = list(zip(cfgs, [matched_color] + other_colors))
    rng.shuffle(triples)
    cand_configs = [list(t[0]) for t in triples]
    cand_colors = [t[1] for t in triples]

    return {
        "ref_b": ref[0],
        "ref_h": ref[1],
        "ref_cx": REF_CX,
        "ref_cy": REF_TOP,
        "cand_configs": cand_configs,
        "cand_colors": cand_colors,
        "cand_centers": [[x, CAND_BASE_Y] for x in SLOTS],
        "width": WIDTH,
        "height": HEIGHT,
    }


def _triangle_points(cx: int, top: int, b: int, h: int) -> list[tuple[int, int]]:
    left = cx - (b - 1) // 2
    right = left + (b - 1)
    base_y = top + (h - 1)
    return [(cx, top), (left, base_y), (right, base_y)]


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution (crisp edges)."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    # Reference panel: a neutral gray apex-up isosceles triangle.
    draw.polygon(
        _triangle_points(int(scene["ref_cx"]), int(scene["ref_cy"]), scene["ref_b"], scene["ref_h"]),
        fill=REF_GRAY,
    )
    # Candidate panel: three apex-up isosceles triangles in red/green/blue.
    for cfg, color, center in zip(
        scene["cand_configs"], scene["cand_colors"], scene["cand_centers"], strict=True
    ):
        b, hh = int(cfg[0]), int(cfg[1])
        x, top = int(center[0]), int(center[1]) - (hh - 1)
        draw.polygon(_triangle_points(x, top, b, hh), fill=COLOR_PALETTE[color])

    draw.rectangle([0, DIVIDER_Y, w, DIVIDER_Y + DIVIDER_THICK], fill=DIVIDER)
    return img


def _matched_index(scene: dict) -> int:
    rb, rh = scene["ref_b"], scene["ref_h"]
    for i, cfg in enumerate(scene["cand_configs"]):
        if int(cfg[0]) == rb and int(cfg[1]) == rh:
            return i
    return 0


def analytic_gold(scene: dict) -> str:
    """Decision: the color of the candidate whose apex angle matches the reference."""
    return scene["cand_colors"][_matched_index(scene)]


def margin(scene: dict) -> float:
    """Runner-up margin: angular separation (degrees) from the reference apex
    angle to the nearest non-matching candidate. Large -> unambiguous match."""
    idx = _matched_index(scene)
    ref = apex_angle(scene["ref_b"], scene["ref_h"])
    ds = []
    for i, cfg in enumerate(scene["cand_configs"]):
        if i != idx:
            ds.append(abs(apex_angle(int(cfg[0]), int(cfg[1])) - ref))
    return float(min(ds)) if ds else 999.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous (near-equal angle) scene; the
    menu's adjacent apex angles are separated by >10 degrees, so the sampler
    never produces one."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (base, height per triangle,
    colors) is rendered verbatim, so the raster uniquely determines the gold;
    no pixel-identical transform is declared."""
    return []
