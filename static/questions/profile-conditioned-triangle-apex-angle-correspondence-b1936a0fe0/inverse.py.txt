"""Independent pixel-only inverse arm for the triangle-apex-angle world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (duplicated here as the oracle's own specification):

  * a neutral gray apex-up isosceles triangle (the reference) sits in the upper
    panel, above a dark horizontal divider line;
  * a row of three apex-up isosceles triangles (red, green, blue) sits below the
    divider, one per slot;
  * each triangle's apex angle is set by its base width and height;
  * the task: find the candidate whose apex angle equals the reference's, then
    report that candidate's color.

The oracle recovers, from pixels alone: the divider, the reference triangle's
apex angle (from its bounding box), each candidate's color and apex angle, and
then returns the color of the single candidate whose measured apex angle is
closest to the reference's. Angles are separated by far more than pixel noise,
so the nearest match is unambiguous.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "triangle_apex_angle-oracle-0.1.0"

DIVIDER = (70, 70, 70)
REF_GRAY = (130, 130, 130)
COLOR_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (230, 0, 0),
    "green": (0, 180, 0),
    "blue": (0, 0, 235),
}
_ORDER = ["red", "green", "blue"]
MIN_SEPARATION_DEG = 3.0


def _angle_from_mask(mask: np.ndarray) -> float | None:
    """Apex angle of an apex-up isosceles triangle from its pixel bounding box."""
    idx = np.argwhere(mask)
    if idx.shape[0] == 0:
        return None
    ymin, xmin = idx.min(axis=0)
    ymax, xmax = idx.max(axis=0)
    bw = xmax - xmin + 1
    bh = ymax - ymin + 1
    if bh <= 0 or bw <= 0:
        return None
    return 2.0 * math.degrees(math.atan((bw / 2.0) / bh))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    h, w = arr.shape[:2]

    # Locate the horizontal divider from its near-black rows.
    divider = np.all(arr == np.array(DIVIDER), axis=-1)
    rowcount = divider.sum(axis=1)
    rows = np.where(rowcount > 0.5 * w)[0]
    if rows.size == 0:
        raise ValueError("no divider line: cannot separate panels")
    divider_y = int(round(float(rows.mean())))

    # Reference gray triangle (upper panel).
    ref_mask = np.all(arr[:divider_y] == np.array(REF_GRAY), axis=-1)
    ref_angle = _angle_from_mask(ref_mask)
    if ref_angle is None:
        raise ValueError("reference triangle not found")

    # Candidates (lower panel), one mask per palette color.
    best = None
    best_gap = float("inf")
    second_gap = float("inf")
    for name in _ORDER:
        mask = np.all(arr[divider_y + 1:] == np.array(COLOR_PALETTE[name]), axis=-1)
        angle = _angle_from_mask(mask)
        if angle is None:
            continue
        gap = abs(angle - ref_angle)
        if best is None or gap < best_gap:
            second_gap = best_gap if best is not None else second_gap
            best = name
            best_gap = gap
        elif gap < second_gap:
            second_gap = gap

    if best is None:
        raise ValueError("no candidate triangle found")
    if second_gap < MIN_SEPARATION_DEG:
        raise ValueError("apex-angle match is ambiguous at pixel scale")
    return best
