"""Independent image-only inverse for disk-to-bar clearance."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_PANELS = ((38, 426), (474, 862))
_Y_WINDOW = (82, 350)
_MIN_COMPONENT = 120
_MIN_DECISION_GAP = 4.0


def _colored_mask(rgb: np.ndarray) -> np.ndarray:
    values = rgb.astype(np.float32) / 255.0
    high = values.max(axis=2)
    low = values.min(axis=2)
    saturation = (high - low) / np.maximum(high, 1e-6)
    # The two objects use saturated inks; gray frames and labels are excluded.
    return (saturation > 0.22) & (high < 0.99) & (low < 0.90)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return area and inclusive local bbox for each 4-connected ink component."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        area = 0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            cy, cx = queue.popleft()
            area += 1
            min_x = min(min_x, cx)
            max_x = max(max_x, cx)
            min_y = min(min_y, cy)
            max_y = max(max_y, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area >= _MIN_COMPONENT:
            found.append((area, min_x, min_y, max_x, max_y))
    return found


def _panel_gap(mask: np.ndarray, panel: tuple[int, int]) -> float:
    x0, x1 = panel
    y0, y1 = _Y_WINDOW
    components = _components(mask[y0:y1, x0:x1])
    if len(components) != 2:
        raise ValueError(f"expected two colored components, found {len(components)}")
    disk, bar = sorted(
        components,
        key=lambda item: (item[4] - item[2] + 1, item[0]),
        reverse=True,
    )
    disk_height = disk[4] - disk[2] + 1
    bar_height = bar[4] - bar[2] + 1
    bar_width = bar[3] - bar[1] + 1
    disk_width = disk[3] - disk[1] + 1
    if disk_height < 2.0 * bar_height or bar_width < 1.5 * disk_width:
        raise ValueError("disk/bar component geometry is ambiguous")
    disk_bottom = y0 + disk[4]
    bar_top = y0 + bar[2]
    if bar_top <= disk_bottom:
        raise ValueError("disk and bar are touching or overlapping")
    return float(bar_top - disk_bottom)


def decision_from_image(image: Image.Image) -> str:
    """Measure both raster gaps and choose the panel with the smaller one."""

    rgb = np.asarray(image.convert("RGB"))
    mask = _colored_mask(rgb)
    gap_a = _panel_gap(mask, _PANELS[0])
    gap_b = _panel_gap(mask, _PANELS[1])
    if abs(gap_a - gap_b) < _MIN_DECISION_GAP:
        raise ValueError("raster gap estimates are too close to classify")
    return "A" if gap_a < gap_b else "B"
