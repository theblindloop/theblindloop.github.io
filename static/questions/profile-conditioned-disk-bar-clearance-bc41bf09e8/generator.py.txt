"""Deterministic paired disk-to-bar clearance renderer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 900
HEIGHT = 440
SUPERSAMPLE = 4
PANEL_BOUNDS = ((38, 426), (474, 862))
PALETTE = ("#d1495b", "#0077b6", "#2a9d8f", "#e09f3e", "#7b2cbf", "#f15bb5")
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size * SUPERSAMPLE)


def _clearance(disk_y: float, radius: float, bar_y: float, bar_height: float) -> float:
    """Return the exposed vertical distance between the two visible edges."""

    return (bar_y - bar_height / 2.0) - (disk_y + radius)


def _sample_panel(rng: random.Random, panel: tuple[int, int], gap: float) -> dict:
    x0, x1 = panel
    disk_x = rng.uniform(x0 + 132.0, x1 - 132.0)
    disk_y = rng.uniform(126.0, 172.0)
    radius = rng.uniform(34.0, 50.0)
    bar_height = rng.uniform(16.0, 24.0)
    bar_width = rng.uniform(190.0, 260.0)
    bar_offset = rng.uniform(-8.0, 8.0)
    bar_x0 = disk_x - bar_width / 2.0 + bar_offset
    bar_x1 = disk_x + bar_width / 2.0 + bar_offset
    bar_y = disk_y + radius + gap + bar_height / 2.0
    disk_color, bar_color = rng.sample(list(PALETTE), 2)
    return {
        "disk_x": round(disk_x, 6),
        "disk_y": round(disk_y, 6),
        "radius": round(radius, 6),
        "bar_x0": round(bar_x0, 6),
        "bar_x1": round(bar_x1, 6),
        "bar_y": round(bar_y, 6),
        "bar_height": round(bar_height, 6),
        "disk_color": disk_color,
        "bar_color": bar_color,
    }


def sample_scene(seed: int) -> dict:
    """Sample balanced positive and near-boundary paired clearances."""

    rng = random.Random(int(seed))
    small_gap = rng.uniform(18.0, 32.0)
    if rng.random() < 0.22:
        difference = rng.uniform(5.0, 10.5)
    else:
        difference = rng.uniform(28.0, 62.0)
    large_gap = small_gap + difference
    small_panel = rng.randrange(2)
    gap_a, gap_b = (
        (small_gap, large_gap) if small_panel == 0 else (large_gap, small_gap)
    )
    panel_a = _sample_panel(rng, PANEL_BOUNDS[0], gap_a)
    panel_b = _sample_panel(rng, PANEL_BOUNDS[1], gap_b)
    return {
        "a_disk_x": panel_a["disk_x"],
        "a_disk_y": panel_a["disk_y"],
        "a_radius": panel_a["radius"],
        "a_bar_x0": panel_a["bar_x0"],
        "a_bar_x1": panel_a["bar_x1"],
        "a_bar_y": panel_a["bar_y"],
        "a_bar_height": panel_a["bar_height"],
        "a_disk_color": panel_a["disk_color"],
        "a_bar_color": panel_a["bar_color"],
        "b_disk_x": panel_b["disk_x"],
        "b_disk_y": panel_b["disk_y"],
        "b_radius": panel_b["radius"],
        "b_bar_x0": panel_b["bar_x0"],
        "b_bar_x1": panel_b["bar_x1"],
        "b_bar_y": panel_b["bar_y"],
        "b_bar_height": panel_b["bar_height"],
        "b_disk_color": panel_b["disk_color"],
        "b_bar_color": panel_b["bar_color"],
    }


def render(scene: dict) -> Image.Image:
    """Render two framed panels with one disk above one horizontal bar each."""

    a_disk_x = float(scene["a_disk_x"])
    a_disk_y = float(scene["a_disk_y"])
    a_radius = float(scene["a_radius"])
    a_bar_x0 = float(scene["a_bar_x0"])
    a_bar_x1 = float(scene["a_bar_x1"])
    a_bar_y = float(scene["a_bar_y"])
    a_bar_height = float(scene["a_bar_height"])
    a_disk_color = str(scene["a_disk_color"])
    a_bar_color = str(scene["a_bar_color"])
    b_disk_x = float(scene["b_disk_x"])
    b_disk_y = float(scene["b_disk_y"])
    b_radius = float(scene["b_radius"])
    b_bar_x0 = float(scene["b_bar_x0"])
    b_bar_x1 = float(scene["b_bar_x1"])
    b_bar_y = float(scene["b_bar_y"])
    b_bar_height = float(scene["b_bar_height"])
    b_disk_color = str(scene["b_disk_color"])
    b_bar_color = str(scene["b_bar_color"])

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    title_font = _font(25)
    small_font = _font(15)
    for (x0, x1), label in zip(PANEL_BOUNDS, ("A", "B")):
        draw.rounded_rectangle(
            (x0 * scale, 34 * scale, x1 * scale, 408 * scale),
            radius=10 * scale,
            fill="#fbfcfe",
            outline="#c7cdd4",
            width=2 * scale,
        )
        bbox = draw.textbbox((0, 0), label, font=title_font)
        label_width = bbox[2] - bbox[0]
        draw.text(
            (((x0 + x1) * scale - label_width) / 2.0, 48 * scale),
            label,
            font=title_font,
            fill="#27313b",
        )
    draw.text((58 * scale, 382 * scale), "disk", font=small_font, fill="#66717d")
    draw.text((494 * scale, 382 * scale), "disk", font=small_font, fill="#66717d")

    draw.ellipse(
        ((a_disk_x - a_radius) * scale, (a_disk_y - a_radius) * scale,
         (a_disk_x + a_radius) * scale, (a_disk_y + a_radius) * scale),
        fill=a_disk_color,
    )
    draw.rectangle(
        (a_bar_x0 * scale, (a_bar_y - a_bar_height / 2.0) * scale,
         a_bar_x1 * scale, (a_bar_y + a_bar_height / 2.0) * scale),
        fill=a_bar_color,
    )
    draw.ellipse(
        ((b_disk_x - b_radius) * scale, (b_disk_y - b_radius) * scale,
         (b_disk_x + b_radius) * scale, (b_disk_y + b_radius) * scale),
        fill=b_disk_color,
    )
    draw.rectangle(
        (b_bar_x0 * scale, (b_bar_y - b_bar_height / 2.0) * scale,
         b_bar_x1 * scale, (b_bar_y + b_bar_height / 2.0) * scale),
        fill=b_bar_color,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Choose the panel with the smaller analytic vertical clearance."""

    clearance_a = _clearance(
        float(scene["a_disk_y"]),
        float(scene["a_radius"]),
        float(scene["a_bar_y"]),
        float(scene["a_bar_height"]),
    )
    clearance_b = _clearance(
        float(scene["b_disk_y"]),
        float(scene["b_radius"]),
        float(scene["b_bar_y"]),
        float(scene["b_bar_height"]),
    )
    return "A" if clearance_a < clearance_b else "B"


def margin(scene: dict) -> float:
    """Positive margin means panel A has the smaller clearance."""

    clearance_a = _clearance(
        float(scene["a_disk_y"]), float(scene["a_radius"]),
        float(scene["a_bar_y"]), float(scene["a_bar_height"]),
    )
    clearance_b = _clearance(
        float(scene["b_disk_y"]), float(scene["b_radius"]),
        float(scene["b_bar_y"]), float(scene["b_bar_height"]),
    )
    return clearance_b - clearance_a


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < 14.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant latent relabeling is claimed."""

    return []
