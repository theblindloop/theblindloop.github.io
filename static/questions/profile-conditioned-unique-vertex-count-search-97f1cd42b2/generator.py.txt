"""Unique-vertex-count search world.

Four equal-size dark convex regular polygons are drawn, one in each quadrant of
the image. Three of them share the same number of sides (same vertex count),
while the fourth polygon has a different number of sides. The decision is the
quadrant containing the polygon whose number of sides is unique. All geometry
and the decision are visible in the raster; rendering is deterministic.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
INK = (30, 30, 30)
R = 80  # circumradius of every polygon, in pixels

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 120, "TR": 360, "BL": 120, "BR": 360}
SECTOR_CY = {"TL": 120, "TR": 120, "BL": 360, "BR": 360}

SIDE_COUNTS = (3, 4, 5, 6)
MIN_DIFF = 2  # minimum |target_sides - distractor_sides|
QUARANTINE_THRESH = 2.0


def _polygon_verts(n: int, cx: int, cy: int) -> list[tuple[float, float]]:
    return [
        (cx + R * np.cos(2.0 * np.pi * k / n), cy + R * np.sin(2.0 * np.pi * k / n))
        for k in range(n)
    ]


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    target = int(SIDE_COUNTS[rng.integers(0, len(SIDE_COUNTS))])
    offsets = [o for o in (-3, -2, 2, 3) if target + o in SIDE_COUNTS]
    distractor = target + int(offsets[rng.integers(0, len(offsets))])

    winner = SECTORS[int(rng.integers(0, len(SECTORS)))]
    candidates = [
        {
            "sector": sector,
            "count": target if sector == winner else distractor,
            "cx": SECTOR_CX[sector],
            "cy": SECTOR_CY[sector],
        }
        for sector in SECTORS
    ]
    return {"candidates": candidates}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)
    for cand in scene["candidates"]:
        draw.polygon(_polygon_verts(cand["count"], cand["cx"], cand["cy"]), fill=INK)
    return img


def _decision(scene: dict) -> tuple[str, float]:
    freq: dict[int, int] = {}
    for cand in scene["candidates"]:
        freq[cand["count"]] = freq.get(cand["count"], 0) + 1
    unique = [cand["sector"] for cand in scene["candidates"] if freq[cand["count"]] == 1]
    if len(unique) != 1:
        return "TL", 0.0
    winner = unique[0]
    winner_count = next(cand["count"] for cand in scene["candidates"] if cand["sector"] == winner)
    others = [
        cand["count"] for cand in scene["candidates"] if cand["sector"] != winner
    ]
    margin = min(abs(winner_count - other) for other in others)
    return winner, float(margin)


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
