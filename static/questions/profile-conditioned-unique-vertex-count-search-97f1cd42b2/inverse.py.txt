"""Independent pixel-only inverse arm for unique_vertex_count_search.

Receives only a PIL image and the public question contract. It separates the
image into four quadrant windows, extracts the dark polygon pixels in each, and
counts each polygon's vertices (number of sides) via its convex hull, keeping
only corners with a sharp interior angle. It then returns the quadrant whose
vertex count occurs exactly once among the four.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CX = {"TL": 120, "TR": 360, "BL": 120, "BR": 360}
SECTOR_CY = {"TL": 120, "TR": 120, "BL": 360, "BR": 360}
WIN = 92
CORNER_ANGLE = 150.0


def _convex_hull(pts: list[tuple[int, int]]) -> list[tuple[int, int]]:
    pts = sorted(set(pts))
    if len(pts) <= 1:
        return pts

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    return lower[:-1] + upper[:-1]


def _vertex_count(arr, cx: int, cy: int) -> int:
    x0, x1 = cx - WIN, cx + WIN
    y0, y1 = cy - WIN, cy + WIN
    region = arr[y0:y1, x0:x1]
    dark = np.min(region, axis=2) < 150
    ys, xs = np.nonzero(dark)
    if len(xs) < 20:
        return 0
    hull = _convex_hull(list(zip(xs.tolist(), ys.tolist())))
    n = len(hull)
    if n < 3:
        return 0
    corners = 0
    for i in range(n):
        a = np.asarray(hull[(i - 1) % n], dtype=float)
        b = np.asarray(hull[i], dtype=float)
        c = np.asarray(hull[(i + 1) % n], dtype=float)
        v1 = a - b
        v2 = c - b
        n1 = np.linalg.norm(v1)
        n2 = np.linalg.norm(v2)
        if n1 == 0 or n2 == 0:
            continue
        cosv = float(np.clip(np.dot(v1, v2) / (n1 * n2), -1.0, 1.0))
        ang = float(np.degrees(np.arccos(cosv)))
        if ang < CORNER_ANGLE:
            corners += 1
    return corners


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    counts = {sector: _vertex_count(arr, SECTOR_CX[sector], SECTOR_CY[sector]) for sector in SECTORS}
    values = list(counts.values())
    unique_value = next((v for v in values if values.count(v) == 1), None)
    if unique_value is None:
        return "NONE"
    for sector in SECTORS:
        if counts[sector] == unique_value:
            return sector
    return "NONE"
