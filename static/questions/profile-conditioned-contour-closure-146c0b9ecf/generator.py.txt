from __future__ import annotations

import math
import random

WIDTH = 400
HEIGHT = 400
CENTER = (200, 200)
BACKGROUND = (255, 255, 255)
INK = (28, 28, 28)

# Closure decision is read from the gap sizes between consecutive contour
# segments. A ring whose every consecutive facing gap stays at or below
# T_JOIN closes into one loop; an open drawing has one conspicuous facing gap
# at or above T_BREAK. BOUNDARY is the decision midpoint used by the margin.
T_JOIN = 14
T_BREAK = 26
BOUNDARY = 20.0
QUAR_T = 3.0
QUARANTINE_MARGIN = QUAR_T

SMALL_LO = 4
SMALL_HI = 10
BIG_LO = 32
BIG_HI = 54


def _build_vertices(scene: dict) -> list[tuple[float, float]]:
    n = int(scene["n_segments"])
    cx, cy = scene["center"]
    base = float(scene["base_angle"])
    r = float(scene["R"])
    step = 2.0 * math.pi / n
    return [
        (cx + r * math.cos(base + k * step), cy + r * math.sin(base + k * step))
        for k in range(n)
    ]


def _build_segments(scene: dict) -> list[tuple[float, float, float, float]]:
    n = int(scene["n_segments"])
    gaps = scene["gaps"]
    verts = _build_vertices(scene)
    segments = []
    for i in range(n):
        x1, y1 = verts[i]
        x2, y2 = verts[(i + 1) % n]
        dx, dy = x2 - x1, y2 - y1
        length = math.hypot(dx, dy)
        ux, uy = dx / length, dy / length
        start_x = x1 + gaps[i] * ux
        start_y = y1 + gaps[i] * uy
        end_x = x2 - gaps[(i + 1) % n] * ux
        end_y = y2 - gaps[(i + 1) % n] * uy
        segments.append((start_x, start_y, end_x, end_y))
    return segments


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    is_open = (seed % 2 == 0)
    n = rng.randint(5, 9)
    cx = rng.randint(185, 215)
    cy = rng.randint(185, 215)
    base_angle = rng.uniform(0.0, 2.0 * math.pi)
    radius = rng.uniform(105.0, 125.0)
    gaps = [rng.randint(SMALL_LO, SMALL_HI) for _ in range(n)]
    if is_open:
        j = rng.randrange(n)
        gaps[j] = rng.randint(BIG_LO, BIG_HI)
    return {
        "n_segments": n,
        "center": (cx, cy),
        "base_angle": base_angle,
        "R": radius,
        "gaps": gaps,
    }


def render(scene: dict):
    from PIL import Image, ImageDraw

    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for (x1, y1, x2, y2) in _build_segments(scene):
        draw.line([x1, y1, x2, y2], fill=INK, width=8)
    return img


def analytic_gold(scene: dict) -> str:
    gaps = scene["gaps"]
    return "open" if max(gaps) >= T_BREAK else "closed"


def margin(scene: dict) -> float:
    gaps = scene["gaps"]
    return abs(BOUNDARY - float(max(gaps)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUAR_T


def latent_symmetries(scene: dict) -> list:
    return []
