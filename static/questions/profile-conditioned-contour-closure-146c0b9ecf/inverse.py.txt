from __future__ import annotations

import math

import numpy as np

# Image-only inverse arm. It receives only the final PNG raster and reconstructs
# the contour topology: every dark segment is isolated, its two facing cap
# centres are recovered by principal-axis projection, and the segments are
# joined wherever any facing cap pair lands within JOIN_PX. The whole set forms
# one closed loop exactly when every recovered segment has two neighbours and
# the graph is a single cycle; an open drawing instead gives a single broken
# chain with exactly two dangling ends.
JOIN_PX = 32.0


def _label_components(mask):
    h, w = mask.shape
    labeled = __import__("numpy").zeros((h, w), dtype="int32")
    current = 0
    comps = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and labeled[y, x] == 0:
                current += 1
                stack = [(y, x)]
                labeled[y, x] = current
                pts = []
                while stack:
                    cy, cx = stack.pop()
                    pts.append((cx, cy))
                    for dy in (-1, 0, 1):
                        for dx in (-1, 0, 1):
                            if dx == 0 and dy == 0:
                                continue
                            ny, nx = cy + dy, cx + dx
                            if 0 <= ny < h and 0 <= nx < w:
                                if mask[ny, nx] and labeled[ny, nx] == 0:
                                    labeled[ny, nx] = current
                                    stack.append((ny, nx))
                comps.append(pts)
    return comps


def _endpoints(pts):
    arr = np.array(pts, dtype=np.float64)
    c = arr.mean(axis=0)
    centered = arr - c
    if centered.shape[0] < 6:
        return [tuple(arr[0]), tuple(arr[-1])]
    cov = centered.T @ centered
    evals, evecs = np.linalg.eigh(cov)
    u = evecs[:, -1]
    if evals[-1] <= 1e-9:
        return [tuple(arr[0]), tuple(arr[-1])]
    t = centered @ u
    thresh_p = np.percentile(t, 98.0)
    thresh_n = np.percentile(t, 2.0)
    pos = centered[t >= thresh_p]
    neg = centered[t <= thresh_n]
    a = neg.mean(axis=0) if neg.shape[0] else centered.min(axis=0)
    b = pos.mean(axis=0) if pos.shape[0] else centered.max(axis=0)
    return [tuple(c + a), tuple(c + b)]


def _joined(first, second) -> bool:
    for ei in first:
        for ej in second:
            if math.hypot(ei[0] - ej[0], ei[1] - ej[1]) <= JOIN_PX:
                return True
    return False


def decision_from_image(image):
    arr = np.array(image.convert("RGB"), dtype=np.int16)
    mask = (arr[:, :, 0] < 128) & (arr[:, :, 1] < 128) & (arr[:, :, 2] < 128)
    comps = _label_components(mask)
    if len(comps) < 3:
        return "none"
    endps = [_endpoints(c) for c in comps]
    m = len(comps)
    adj = [[] for _ in range(m)]
    for i in range(m):
        for j in range(i + 1, m):
            if _joined(endps[i], endps[j]):
                adj[i].append(j)
                adj[j].append(i)
    deg = [len(adj[i]) for i in range(m)]
    seen = set()
    stack = [0]
    seen.add(0)
    while stack:
        cur = stack.pop()
        for nb in adj[cur]:
            if nb not in seen:
                seen.add(nb)
                stack.append(nb)
    if len(seen) != m:
        return "none"
    if all(d == 2 for d in deg):
        return "closed"
    if deg.count(1) == 2 and all(d in (1, 2) for d in deg):
        return "open"
    return "none"
