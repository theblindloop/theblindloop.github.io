"""Deterministic latent sampler and renderer for Single-Pocket Seal."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image

WIDTH = 320
HEIGHT = 320
CELL = 40
OUTER = 30
BACKGROUND = (250, 250, 247)
TEAL = (9, 110, 115)
PALE_GRAY = (218, 221, 218)


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; consecutive integer seeds alternate classes."""
    rng = random.Random(int(seed))
    pocket_count = 1 if int(seed) % 2 == 0 else 2
    cell_x = rng.randint(1, 6)
    cell_y = rng.randint(1, 6)
    hole_axis = "horizontal" if rng.randrange(2) == 0 else "vertical"
    cx = cell_x * CELL + CELL // 2
    cy = cell_y * CELL + CELL // 2
    decorations: list[list[int]] = []
    for _ in range(8):
        for _attempt in range(200):
            x = rng.randint(14, WIDTH - 20)
            y = rng.randint(14, HEIGHT - 20)
            w = rng.randint(4, 9)
            h = rng.randint(4, 9)
            if abs((x + w / 2) - cx) > 55 or abs((y + h / 2) - cy) > 55:
                decorations.append([x, y, w, h])
                break
    return {
        "pocket_count": pocket_count,
        "cell_x": cell_x,
        "cell_y": cell_y,
        "hole_axis": hole_axis,
        "decorations": decorations,
    }


def _paint_seal(array: np.ndarray, scene: dict) -> None:
    cx = int(scene["cell_x"]) * CELL + CELL // 2
    cy = int(scene["cell_y"]) * CELL + CELL // 2
    count = int(scene["pocket_count"])
    axis = str(scene["hole_axis"])
    half = OUTER // 2
    array[cy - half : cy + half, cx - half : cx + half] = TEAL

    # Both classes erase exactly 144 pixels. The one-pocket class uses one
    # 12x12 chamber; the two-pocket class uses two 6x12 chambers separated by
    # a four-pixel teal wall. Thus outer extent, total ink, and cutout area match.
    if count == 1:
        array[cy - 6 : cy + 6, cx - 6 : cx + 6] = BACKGROUND
    elif count == 2 and axis == "horizontal":
        array[cy - 6 : cy + 6, cx - 8 : cx - 2] = BACKGROUND
        array[cy - 6 : cy + 6, cx + 2 : cx + 8] = BACKGROUND
    elif count == 2 and axis == "vertical":
        array[cy - 8 : cy - 2, cx - 6 : cx + 6] = BACKGROUND
        array[cy + 2 : cy + 8, cx - 6 : cx + 6] = BACKGROUND
    else:
        raise ValueError("unsupported pocket configuration")


def render(scene: dict) -> Image.Image:
    # Direct access keeps the public rendered-field audit explicit: this label
    # input is validated here and then consumed by the painting helper.
    if int(scene["pocket_count"]) not in (1, 2):
        raise ValueError("pocket_count must be one or two")
    array = np.full((HEIGHT, WIDTH, 3), BACKGROUND, dtype=np.uint8)
    for raw in scene["decorations"]:
        x, y, w, h = (int(value) for value in raw)
        array[y : y + h, x : x + w] = PALE_GRAY
    _paint_seal(array, scene)
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["pocket_count"]) == 1 else "no"


def margin(scene: dict) -> float:
    # Minimum thickness of a topology-preserving teal wall in pixels.
    return 7.0 if int(scene["pocket_count"]) == 1 else 4.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
