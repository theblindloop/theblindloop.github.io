"""Independent PNG-only inverse arm for Single-Pocket Seal.

The inverse segments teal by color, validates one compact connected component,
then labels the four-connected complement in a padded crop. Components that do
not reach the crop border are the seal's enclosed white pockets.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

TEAL = np.array((9, 110, 115), dtype=np.int16)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        iy, ix = int(y), int(x)
        if seen[iy, ix]:
            continue
        seen[iy, ix] = True
        queue = deque([(iy, ix)])
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _count_holes(teal: np.ndarray) -> int:
    points = np.argwhere(teal)
    if len(points) < 740:
        raise ValueError("teal seal is missing or materially damaged")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width = int(x1 - x0 + 1)
    height = int(y1 - y0 + 1)
    if not (29 <= width <= 31 and 29 <= height <= 31):
        raise ValueError("teal component has an invalid compact extent")
    groups = _components(teal)
    if len(groups) != 1 or len(groups[0]) < 740:
        raise ValueError("teal seal is not one robust connected component")

    crop = teal[max(0, y0 - 1) : min(teal.shape[0], y1 + 2), max(0, x0 - 1) : min(teal.shape[1], x1 + 2)]
    complement_groups = _components(~crop)
    holes = 0
    ch, cw = crop.shape
    for group in complement_groups:
        touches_border = any(y in (0, ch - 1) or x in (0, cw - 1) for y, x in group)
        if not touches_border:
            if len(group) < 40:
                raise ValueError("sub-resolution enclosed component")
            holes += 1
    if holes not in (1, 2):
        raise ValueError(f"unexpected bounded-component count: {holes}")
    return holes


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    distance = np.max(np.abs(array - TEAL), axis=2)
    teal = distance <= 24
    return "yes" if _count_holes(teal) == 1 else "no"
