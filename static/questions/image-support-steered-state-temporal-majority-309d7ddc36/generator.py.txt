"""Deterministic renderer and analytic arm for tracked-state majority scenes."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "state-temporal-majority-0.1.0"
WIDTH = 512
HEIGHT = 512
CARD_W = 62
CARD_H = 112
PANEL_SLOTS = ((145, 118), (220, 118), (295, 118), (180, 285), (255, 285))
ROW_OFFSETS = (36, 56, 76, 96)
BACKGROUND = (255, 255, 255)
CARD_FILL = (246, 248, 250)
BORDER = (40, 48, 62)
HEADER_RULE = (190, 198, 208)
DIGIT = (38, 54, 78)
RING = (226, 44, 145)

PALETTE = (
    (35, 102, 180),
    (234, 120, 35),
    (39, 150, 110),
    (123, 81, 170),
    (210, 70, 70),
    (70, 115, 145),
)

DIGITS = {
    1: ("00100", "01100", "00100", "00100", "00100", "00100", "01110"),
    2: ("01110", "10001", "00001", "00010", "00100", "01000", "11111"),
    3: ("11110", "00001", "00001", "01110", "00001", "00001", "11110"),
    4: ("00010", "00110", "01010", "10010", "11111", "00010", "00010"),
    5: ("11111", "10000", "10000", "11110", "00001", "00001", "11110"),
}

LETTERS = {
    "A": ("01110", "10001", "10001", "11111", "10001"),
    "B": ("11110", "10001", "11110", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "11110"),
}


def _bitmap(draw: ImageDraw.ImageDraw, x: int, y: int, pattern: tuple[str, ...], scale: int, fill: tuple[int, int, int]) -> None:
    for row, line in enumerate(pattern):
        for col, value in enumerate(line):
            if value == "1":
                draw.rectangle(
                    (x + col * scale, y + row * scale, x + (col + 1) * scale - 1, y + (row + 1) * scale - 1),
                    fill=fill,
                )


def _gold_from_states(states: list[list[str]], tracked_row: int) -> str:
    circle_count = sum(snapshot[tracked_row] == "circle" for snapshot in states)
    diamond_count = len(states) - circle_count
    if circle_count == diamond_count:
        raise ValueError("temporal majority is tied")
    return "circle" if circle_count > diamond_count else "diamond"


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene with answer-independent global shape totals."""
    rng = random.Random(int(seed))
    tracked_row = rng.randrange(4)
    circle_majority = int(seed) % 2 == 0
    if circle_majority:
        tracked_circle_count = 3 + rng.randrange(2)
    else:
        tracked_circle_count = 1 + rng.randrange(2)

    states = [["diamond" for _ in range(4)] for _ in range(5)]
    tracked_positions = list(range(5))
    rng.shuffle(tracked_positions)
    for index in tracked_positions[:tracked_circle_count]:
        states[index][tracked_row] = "circle"

    other_slots = [(snapshot, row) for snapshot in range(5) for row in range(4) if row != tracked_row]
    rng.shuffle(other_slots)
    other_circle_count = 10 - tracked_circle_count
    for snapshot, row in other_slots[:other_circle_count]:
        states[snapshot][row] = "circle"

    panel_order = [1, 2, 3, 4, 5]
    rng.shuffle(panel_order)
    glyph_colors = [
        [list(rng.choice(PALETTE)) for _ in range(4)]
        for _ in range(5)
    ]
    return {
        "states": states,
        "tracked_row": tracked_row,
        "panel_order": panel_order,
        "glyph_colors": glyph_colors,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    states = scene["states"]
    tracked_row = int(scene["tracked_row"])
    panel_order = scene["panel_order"]
    glyph_colors = scene["glyph_colors"]

    for slot_index, snapshot_number in enumerate(panel_order):
        x, y = PANEL_SLOTS[slot_index]
        snapshot_index = int(snapshot_number) - 1
        draw.rectangle((x, y, x + CARD_W - 1, y + CARD_H - 1), fill=CARD_FILL, outline=BORDER, width=2)
        _bitmap(draw, x + 8, y + 5, DIGITS[int(snapshot_number)], 2, DIGIT)
        draw.line((x + 6, y + 22, x + CARD_W - 7, y + 22), fill=HEADER_RULE, width=1)
        for row, offset in enumerate(ROW_OFFSETS):
            _bitmap(draw, x + 8, y + offset - 3, LETTERS["ABCD"[row]], 1, DIGIT)
            cx = x + 39
            cy = y + offset
            color = tuple(int(value) for value in glyph_colors[snapshot_index][row])
            if states[snapshot_index][row] == "circle":
                draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=color)
            else:
                draw.polygon(((cx, cy - 10), (cx + 10, cy), (cx, cy + 10), (cx - 10, cy)), fill=color)
            if int(snapshot_number) == 1 and row == tracked_row:
                draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), outline=RING, width=2)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return _gold_from_states(scene["states"], int(scene["tracked_row"]))


def margin(scene: dict[str, Any]) -> float:
    states = scene["states"]
    tracked_row = int(scene["tracked_row"])
    circle_count = sum(snapshot[tracked_row] == "circle" for snapshot in states)
    return float(abs(2 * circle_count - len(states)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    states = scene["states"]
    tracked_row = int(scene["tracked_row"])
    if len(states) != 5 or not 0 <= tracked_row < 4:
        return True
    try:
        return margin(scene) <= 0.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
