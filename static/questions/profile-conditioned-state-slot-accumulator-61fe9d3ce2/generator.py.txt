"""Ordered-slot displacement accumulator (state tracking).

Five panels are shown ordered left to right. Each panel draws a horizontal
track with five evenly spaced slots (numbered 0-4 left to right, marked by
divider ticks) and a single marker token sitting in exactly one slot. An
orange ring marks the marker in the leftmost (initial) panel.

Between every pair of consecutive panels an operator glyph is shown:
a red right-pointing triangle means "advance the marker one slot (+1)" and
a blue left-pointing triangle means "retreat the marker one slot (-1)".

The latent scene stores only ``start`` (the initial slot), ``arrows`` (the
four per-step +1/-1 operators) and ``positions`` (the resulting marker slot
in each panel, already consistent with ``start`` and ``arrows``). The decision
is the marker's final slot, i.e. the last element of ``positions``.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1010
HEIGHT = 230
PANEL_COUNT = 5
N_SLOTS = 5
PANEL_W = 150
PANEL_H = 160
GAP = 50
MARGIN_LEFT = 30
PITCH = PANEL_W + GAP
CENTER_Y = HEIGHT // 2
TOKEN_R = 9
TRACK_Y_OFF = 42
TOKEN_Y = CENTER_Y - 8
TICK_H = 10
GLYPH_R = 14

BG = (255, 255, 255)
BORDER = (205, 205, 205)
INK = (0, 0, 0)
TRACK_COLOR = (90, 90, 90)
RING_MARK = (240, 150, 20)
ADVANCE_COLOR = (210, 40, 40)
RETREAT_COLOR = (40, 60, 210)


def _panel_left(index: int) -> int:
    return MARGIN_LEFT + PITCH * index


def _slot_center(panel_left: int, slot: int) -> int:
    return int(panel_left + PANEL_W * (slot + 0.5) / N_SLOTS)


def _track_y(panel_top: int) -> int:
    return panel_top + TRACK_Y_OFF


def _trajectory(start: int, arrows: list[int]) -> list[int]:
    pos = start
    out = [start]
    for arrow in arrows:
        pos += arrow
        out.append(pos)
    return out


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed, keeping the marker on track."""
    rng = random.Random(seed)
    while True:
        start = rng.randint(0, N_SLOTS - 1)
        arrows = [rng.choice([-1, 1]) for _ in range(PANEL_COUNT - 1)]
        positions = _trajectory(start, arrows)
        if all(0 <= p < N_SLOTS for p in positions):
            return {"start": start, "arrows": arrows, "positions": positions}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    positions = scene["positions"]
    arrows = scene["arrows"]

    for i in range(PANEL_COUNT):
        left = _panel_left(i)
        top = CENTER_Y - PANEL_H // 2
        draw.rectangle([left, top, left + PANEL_W, top + PANEL_H], outline=BORDER, width=3)
        ty = _track_y(top)
        draw.line([left + 12, ty, left + PANEL_W - 12, ty], fill=TRACK_COLOR, width=3)
        for s in range(N_SLOTS + 1):
            tx = int(left + PANEL_W * s / N_SLOTS)
            draw.line([tx, ty - TICK_H // 2, tx, ty + TICK_H // 2], fill=TRACK_COLOR, width=2)
        slot = positions[i]
        cx = _slot_center(left, slot)
        cy = TOKEN_Y
        draw.ellipse(
            [cx - TOKEN_R, cy - TOKEN_R, cx + TOKEN_R, cy + TOKEN_R], fill=INK, outline=INK
        )

    left0 = _panel_left(0)
    cx0 = _slot_center(left0, positions[0])
    draw.ellipse(
        [cx0 - TOKEN_R - 7, TOKEN_Y - TOKEN_R - 7, cx0 + TOKEN_R + 7, TOKEN_Y + TOKEN_R + 7],
        outline=RING_MARK,
        width=4,
    )

    for i, arrow in enumerate(arrows):
        ax = MARGIN_LEFT + PITCH * i + PANEL_W + GAP // 2
        ay = CENTER_Y
        color = ADVANCE_COLOR if arrow == 1 else RETREAT_COLOR
        r = GLYPH_R
        if arrow == 1:
            points = [(ax - r, ay - r), (ax + r, ay), (ax - r, ay + r)]
        else:
            points = [(ax + r, ay - r), (ax - r, ay), (ax + r, ay + r)]
        draw.polygon(points, fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["positions"][-1])


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry: ``start`` fixes the
    marked token's slot, ``arrows`` fix the drawn triangle glyphs, and
    ``positions`` fix the marker's slot in every panel. Reordering any of these
    would move the marker or the glyphs and change the pixels."""
    return []
