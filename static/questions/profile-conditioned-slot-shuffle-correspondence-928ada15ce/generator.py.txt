"""Deterministic renderer and analytic arm for the slot-shuffle world."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1120, 800
SCALE = 2
BACKGROUND = (248, 247, 243)
INK = (24, 27, 32)
MUTED = (92, 99, 110)
CARD_BORDER = (31, 56, 88)
CARD_FILL = (255, 255, 252)
PATTERN_TILE = (235, 231, 219)
COLORS = {
    "red": (214, 68, 69),
    "green": (42, 157, 105),
    "blue": (55, 106, 204),
    "orange": (232, 145, 39),
}
LABELS = ("A", "B", "C")
PATTERNS = ("p0", "p1", "p2", "p3")
PATTERN_BITS = {
    "p0": (0, 1, 3, 8),
    "p1": (0, 2, 4, 7),
    "p2": (1, 3, 5, 6),
    "p3": (2, 3, 4, 8),
}
CARD_W, CARD_H = 360, 100


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _nontrivial_permutations() -> tuple[tuple[int, ...], ...]:
    return tuple(p for p in itertools.permutations(range(4)) if p != (0, 1, 2, 3))


def _target_colors(scene: dict) -> list[str]:
    before = scene["pattern_before"]
    after = scene["pattern_after"]
    colors = scene["color_input"]
    if sorted(before) != sorted(PATTERNS) or sorted(after) != sorted(PATTERNS):
        raise ValueError("pattern rows must each contain the four unique public patterns")
    if sorted(colors) != sorted(COLORS):
        raise ValueError("input must contain each public color exactly once")
    source_slots = [before.index(pattern) for pattern in after]
    return [colors[index] for index in source_slots]


def _candidate_scores(scene: dict) -> list[int]:
    target = _target_colors(scene)
    candidates = scene["candidates"]
    return [sum(left == right for left, right in zip(row, target)) for row in candidates]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, unambiguous analogical permutation scene."""
    rng = random.Random(int(seed))
    before = list(PATTERNS)
    rng.shuffle(before)
    permutation = rng.choice(_nontrivial_permutations())
    after = [before[index] for index in permutation]

    color_input = list(COLORS)
    rng.shuffle(color_input)
    target = [color_input[index] for index in permutation]

    pairs = list(itertools.combinations(range(4), 2))
    rng.shuffle(pairs)
    distractors: list[list[str]] = []
    for left, right in pairs:
        row = list(target)
        row[left], row[right] = row[right], row[left]
        if row not in distractors:
            distractors.append(row)
        if len(distractors) == 2:
            break

    correct_index = int(seed) % 3
    candidates = list(distractors)
    candidates.insert(correct_index, target)
    scene = {
        "pattern_before": before,
        "pattern_after": after,
        "color_input": color_input,
        "candidates": candidates,
    }
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a quarantined scene")
    return scene


def analytic_gold(scene: dict) -> str:
    """Infer the permutation from the example and select its unique output row."""
    target = _target_colors(scene)
    candidates = scene["candidates"]
    matches = [index for index, row in enumerate(candidates) if list(row) == target]
    if len(matches) != 1:
        raise ValueError("the transported output must occur exactly once")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    """Best-minus-runner-up number of correctly transported slots."""
    scores = sorted(_candidate_scores(scene), reverse=True)
    return float(scores[0] - scores[1]) if len(scores) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        candidates = scene["candidates"]
        if len(candidates) != 3:
            return True
        if any(sorted(row) != sorted(COLORS) for row in candidates):
            return True
        if len({tuple(row) for row in candidates}) != 3:
            return True
        _ = analytic_gold(scene)
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no pixel-identical aliases in the declared scene representation."""
    return []


def _scaled_box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)


def _draw_pattern(draw: ImageDraw.ImageDraw, cx: int, cy: int, pattern: str) -> None:
    draw.rounded_rectangle(
        _scaled_box((cx - 31, cy - 31, cx + 31, cy + 31)),
        radius=8 * SCALE,
        fill=PATTERN_TILE,
        outline=(184, 177, 158),
        width=2 * SCALE,
    )
    bit = 12
    gap = 4
    total = bit * 3 + gap * 2
    start_x = cx - total // 2
    start_y = cy - total // 2
    for index in PATTERN_BITS[pattern]:
        row, col = divmod(index, 3)
        x0 = start_x + col * (bit + gap)
        y0 = start_y + row * (bit + gap)
        draw.rounded_rectangle(
            _scaled_box((x0, y0, x0 + bit, y0 + bit)),
            radius=2 * SCALE,
            fill=INK,
        )


def _draw_color(draw: ImageDraw.ImageDraw, cx: int, cy: int, color: str) -> None:
    draw.rounded_rectangle(
        _scaled_box((cx - 29, cy - 29, cx + 29, cy + 29)),
        radius=12 * SCALE,
        fill=COLORS[color],
        outline=(255, 255, 255),
        width=3 * SCALE,
    )


def _draw_card(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    items: Iterable[str],
    *,
    kind: str,
) -> None:
    x, y = xy
    draw.rectangle(_scaled_box((x, y, x + CARD_W, y + CARD_H)), fill=CARD_FILL)
    draw.rectangle(
        _scaled_box((x, y, x + CARD_W, y + CARD_H)),
        outline=CARD_BORDER,
        width=5 * SCALE,
    )
    slot_w = CARD_W / 4
    for slot in range(1, 4):
        sx = round(x + slot * slot_w)
        draw.line(
            [(sx * SCALE, y * SCALE), (sx * SCALE, (y + CARD_H) * SCALE)],
            fill=CARD_BORDER,
            width=3 * SCALE,
        )
    for slot, item in enumerate(items):
        cx = round(x + (slot + 0.5) * slot_w)
        cy = y + CARD_H // 2
        if kind == "pattern":
            _draw_pattern(draw, cx, cy, item)
        else:
            _draw_color(draw, cx, cy, item)


def _text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    value: str,
    size: int,
    *,
    bold: bool = False,
    fill: tuple[int, int, int] = INK,
    anchor: str = "mm",
) -> None:
    draw.text(
        (xy[0] * SCALE, xy[1] * SCALE),
        value,
        font=_font(size, bold),
        fill=fill,
        anchor=anchor,
    )


def render(scene: dict) -> Image.Image:
    """Render the complete question world from declared scene fields."""
    pattern_before = scene["pattern_before"]
    pattern_after = scene["pattern_after"]
    color_input = scene["color_input"]
    candidates = scene["candidates"]

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    _text(draw, (560, 38), "LEARN THE SLOT SHUFFLE", 25, bold=True)
    _text(
        draw,
        (560, 72),
        "Match each pattern from BEFORE to AFTER, then apply the same rearrangement below.",
        15,
        fill=MUTED,
    )
    _text(draw, (270, 119), "EXAMPLE · BEFORE", 16, bold=True)
    _text(draw, (910, 119), "EXAMPLE · AFTER", 16, bold=True)
    _draw_card(draw, (90, 140), pattern_before, kind="pattern")
    _draw_card(draw, (730, 140), pattern_after, kind="pattern")
    _text(draw, (560, 190), "→", 44, bold=True, fill=MUTED)

    draw.line(
        [(55 * SCALE, 292 * SCALE), (1065 * SCALE, 292 * SCALE)],
        fill=(210, 207, 197),
        width=2 * SCALE,
    )
    _text(draw, (270, 337), "APPLY · BEFORE", 16, bold=True)
    _draw_card(draw, (90, 360), color_input, kind="color")
    _text(draw, (520, 410), "→", 44, bold=True, fill=MUTED)

    _text(draw, (850, 317), "WHICH AFTER ROW?", 16, bold=True)
    candidate_y = (340, 480, 620)
    for label, row, y in zip(LABELS, candidates, candidate_y):
        draw.ellipse(
            _scaled_box((575, y + 31, 613, y + 69)),
            fill=(255, 255, 252),
            outline=INK,
            width=3 * SCALE,
        )
        _text(draw, (594, y + 50), label, 18, bold=True)
        _draw_card(draw, (650, y), row, kind="color")

    _text(
        draw,
        (560, 775),
        "Tiles keep their own pattern or color; only their slots change.",
        14,
        fill=MUTED,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
