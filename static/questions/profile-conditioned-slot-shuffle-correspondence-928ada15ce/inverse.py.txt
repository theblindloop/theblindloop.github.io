"""Independent pixel-only inverse for the slot-shuffle image.

The code discovers the six navy card frames, matches pattern masks between the
worked example, reads saturated tile colors, and compares the three outputs.
It receives no scene or renderer state.
"""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

CARD_BORDER_RGB = np.array((31, 56, 88), dtype=np.int16)
PALETTE = {
    "red": np.array((214, 68, 69), dtype=np.int16),
    "green": np.array((42, 157, 105), dtype=np.int16),
    "blue": np.array((55, 106, 204), dtype=np.int16),
    "orange": np.array((232, 145, 39), dtype=np.int16),
}
LABELS = ("A", "B", "C")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    remaining = set(np.flatnonzero(mask).tolist())
    found: list[tuple[int, int, int, int, int]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        x0 = x1 = start % width
        y0 = y1 = start // width
        count = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            count += 1
            x0, x1 = min(x0, x), max(x1, x)
            y0, y1 = min(y0, y), max(y1, y)
            for other in (value - 1, value + 1, value - width, value + width):
                if other in remaining:
                    oy, ox = divmod(other, width)
                    if abs(oy - y) + abs(ox - x) == 1:
                        remaining.remove(other)
                        stack.append(other)
        found.append((x0, y0, x1, y1, count))
    return found


def _find_cards(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    distance = np.linalg.norm(arr.astype(np.int16) - CARD_BORDER_RGB, axis=2)
    mask = distance < 18
    components = _components(mask)
    frames = []
    for x0, y0, x1, y1, count in components:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if width > arr.shape[1] * 0.24 and height > arr.shape[0] * 0.08 and count > 900:
            frames.append((x0, y0, x1, y1))
    if len(frames) != 6:
        raise ValueError(f"expected six visible card frames, found {len(frames)}")
    return frames


def _roles(frames: list[tuple[int, int, int, int]]):
    ordered = sorted(frames, key=lambda box: (box[0] + box[2]) / 2)
    gaps = [
        (ordered[index + 1][0] + ordered[index + 1][2]) / 2
        - (ordered[index][0] + ordered[index][2]) / 2
        for index in range(len(ordered) - 1)
    ]
    split = int(np.argmax(gaps)) + 1
    left, right = ordered[:split], ordered[split:]
    if len(left) != 2 or len(right) != 4:
        raise ValueError("card layout does not form two left and four right cards")
    left.sort(key=lambda box: (box[1] + box[3]) / 2)
    right.sort(key=lambda box: (box[1] + box[3]) / 2)
    return left[0], right[0], left[1], right[1:]


def _cell_crops(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[np.ndarray]:
    x0, y0, x1, y1 = box
    width = x1 - x0 + 1
    crops = []
    for index in range(4):
        left = round(x0 + index * width / 4) + 9
        right = round(x0 + (index + 1) * width / 4) - 9
        top, bottom = y0 + 10, y1 - 9
        if right <= left or bottom <= top:
            raise ValueError("card slots are too small to read")
        crops.append(arr[top:bottom, left:right])
    return crops


def _pattern_masks(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[np.ndarray]:
    masks = []
    for crop in _cell_crops(arr, box):
        dark = np.max(crop, axis=2) < 80
        # Normalize using nearest-neighbor sampling without importing world code.
        normalized = Image.fromarray((dark * 255).astype(np.uint8)).resize((64, 64), Image.Resampling.NEAREST)
        masks.append(np.asarray(normalized) > 127)
    return masks


def _infer_permutation(
    before_masks: list[np.ndarray], after_masks: list[np.ndarray]
) -> tuple[tuple[int, ...], float]:
    costs = np.array(
        [
            [np.count_nonzero(after ^ before) for before in before_masks]
            for after in after_masks
        ],
        dtype=float,
    )
    ranked = []
    for permutation in itertools.permutations(range(4)):
        ranked.append((sum(costs[out, src] for out, src in enumerate(permutation)), permutation))
    ranked.sort(key=lambda item: item[0])
    best_cost, best = ranked[0]
    gap = ranked[1][0] - best_cost
    # The two cards can land on different antialiasing phases after the final
    # downsample, so identical glyphs retain a small edge-pixel cost.  The
    # assignment separation is the stronger ambiguity test.
    if best_cost > 400 or gap < 120:
        raise ValueError(f"pattern correspondence is ambiguous (cost={best_cost}, gap={gap})")
    return best, float(gap)


def _read_colors(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[str]:
    result = []
    for crop in _cell_crops(arr, box):
        pixels = crop.reshape(-1, 3).astype(np.int16)
        counts = []
        for name, rgb in PALETTE.items():
            distances = np.linalg.norm(pixels - rgb, axis=1)
            counts.append((int(np.count_nonzero(distances < 38)), name))
        counts.sort(reverse=True)
        if counts[0][0] < 800 or counts[0][0] < counts[1][0] * 5:
            raise ValueError("a colored slot is missing or ambiguous")
        result.append(counts[0][1])
    if len(set(result)) != 4:
        raise ValueError("each row must visibly contain four different colors")
    return result


def _measure(image: Image.Image) -> tuple[str, dict]:
    arr = np.asarray(image.convert("RGB"))
    before_box, after_box, input_box, candidate_boxes = _roles(_find_cards(arr))
    before_masks = _pattern_masks(arr, before_box)
    after_masks = _pattern_masks(arr, after_box)
    permutation, pattern_gap = _infer_permutation(before_masks, after_masks)
    input_colors = _read_colors(arr, input_box)
    target = [input_colors[index] for index in permutation]
    candidate_rows = [_read_colors(arr, box) for box in candidate_boxes]
    scores = [sum(left == right for left, right in zip(row, target)) for row in candidate_rows]
    order = sorted(range(3), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] != 4 or scores[order[0]] - scores[order[1]] < 2:
        raise ValueError(f"candidate decision is ambiguous (scores={scores})")
    return LABELS[order[0]], {
        "pattern_assignment_gap_pixels": pattern_gap,
        "candidate_scores": scores,
        "runner_up_gap_slots": scores[order[0]] - scores[order[1]],
    }


def decision_from_image(image: Image.Image) -> str:
    """Return A, B, or C using only the final rasterized pixels."""
    return _measure(image)[0]
