"""Independent image-only inverse for bounded white chambers."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _outside(open_mask: np.ndarray) -> np.ndarray:
    height, width = open_mask.shape
    seen = np.zeros_like(open_mask, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(width):
        if open_mask[0, x] and not seen[0, x]:
            seen[0, x] = True
            queue.append((0, x))
        if open_mask[height - 1, x] and not seen[height - 1, x]:
            seen[height - 1, x] = True
            queue.append((height - 1, x))
    for y in range(height):
        if open_mask[y, 0] and not seen[y, 0]:
            seen[y, 0] = True
            queue.append((y, 0))
        if open_mask[y, width - 1] and not seen[y, width - 1]:
            seen[y, width - 1] = True
            queue.append((y, width - 1))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and open_mask[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return seen


def decision_from_image(image: Image.Image) -> str:
    """Count non-black components that do not touch the PNG boundary."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if pixels.ndim != 3 or pixels.shape[0] < 3 or pixels.shape[1] < 3:
        return "0"
    wall = np.max(pixels, axis=2) < 100
    open_mask = ~wall
    outside = _outside(open_mask)
    enclosed_seen = outside.copy()
    count = 0
    height, width = open_mask.shape
    for y in range(height):
        for x in range(width):
            if not open_mask[y, x] or enclosed_seen[y, x]:
                continue
            count += 1
            queue: deque[tuple[int, int]] = deque([(y, x)])
            enclosed_seen[y, x] = True
            while queue:
                cy, cx = queue.popleft()
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and open_mask[ny, nx]
                        and not enclosed_seen[ny, nx]
                    ):
                        enclosed_seen[ny, nx] = True
                        queue.append((ny, nx))
    return str(count)
