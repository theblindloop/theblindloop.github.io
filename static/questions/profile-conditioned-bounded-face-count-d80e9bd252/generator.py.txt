"""Deterministic renderer and analytic target for the bounded-face world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


CANVAS = (512, 512)
WALL = (27, 31, 38)
BRIGHT_DOTS = ((226, 79, 99), (54, 137, 214), (39, 164, 111))


def _chamber_bounds(
    frame: list[int], thickness: int, partitions: list[list[int]]
) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = frame
    left = x0 + thickness
    right = x1 - thickness
    centers = sorted((int((part[0] + part[2]) / 2) for part in partitions))
    bounds: list[tuple[int, int]] = []
    previous = left
    for center in centers:
        bounds.append((previous + 12, center - thickness // 2 - 12))
        previous = center + thickness // 2
    bounds.append((previous + 12, right - 12))
    return bounds


def _add_decorations(
    rng: random.Random,
    frame: list[int],
    thickness: int,
    partitions: list[list[int]],
) -> list[list[int]]:
    """Add isolated black strokes to keep ink and layout from identifying count."""

    x0, y0, x1, y1 = frame
    inner_left, inner_top = x0 + thickness, y0 + thickness
    inner_right, inner_bottom = x1 - thickness, y1 - thickness
    base_area = (
        (x1 - x0 + 1) * (y1 - y0 + 1)
        - (inner_right - inner_left + 1) * (inner_bottom - inner_top + 1)
    )
    base_area += sum(
        (part[2] - part[0] + 1) * (part[3] - part[1] + 1) for part in partitions
    )
    target_area = 30000 + rng.randint(-500, 500)
    remaining = max(0, target_area - base_area)
    bounds = _chamber_bounds(frame, thickness, partitions)
    decorations: list[list[int]] = []
    slots = [(column, row) for column in range(len(bounds)) for row in range(14)]
    rng.shuffle(slots)
    for slot_index, (column, row) in enumerate(slots):
        if remaining <= 0:
            break
        left, right = bounds[column]
        chamber_width = right - left
        if chamber_width < 60:
            continue
        max_length = max(28, chamber_width - 18)
        slots_left = max(1, len(slots) - slot_index)
        desired = max(28, int(remaining / max(1, slots_left) / 7))
        length = min(max_length, max(28, desired + rng.randint(-18, 18)))
        length = min(length, max_length)
        center = rng.randint(left + length // 2, right - length // 2)
        y = inner_top + 18 + row * 22 + rng.randint(-2, 2)
        if y + 4 >= inner_bottom - 4:
            continue
        x_start = center - int(length // 2)
        x_end = center + int(length // 2)
        decorations.append([x_start, y, x_end, y])
        remaining -= (x_end - x_start + 1) * 7

    return decorations


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a safe scene; counts are balanced by consecutive integer seeds."""

    rng = random.Random(int(seed))
    chamber_count = 1 + (int(seed) % 3)
    width, height = CANVAS
    x0 = rng.randint(44, 68)
    y0 = rng.randint(48, 70)
    x1 = rng.randint(444, 468)
    y1 = rng.randint(430, 460)
    thickness = rng.choice((12, 14, 16))
    frame = [x0, y0, x1, y1]
    inner_left, inner_top = x0 + thickness, y0 + thickness
    inner_right, inner_bottom = x1 - thickness, y1 - thickness
    usable = inner_right - inner_left

    partition_centers: list[int] = []
    if chamber_count == 2:
        partition_centers = [inner_left + usable // 2 + rng.randint(-10, 10)]
    elif chamber_count == 3:
        partition_centers = [
            inner_left + usable // 3 + rng.randint(-9, 9),
            inner_left + 2 * usable // 3 + rng.randint(-9, 9),
        ]
    partitions = [
        [center - thickness // 2, inner_top - 3, center + thickness // 2, inner_bottom + 3]
        for center in partition_centers
    ]
    decorations = _add_decorations(rng, frame, thickness, partitions)
    bounds = _chamber_bounds(frame, thickness, partitions)

    colors = list(BRIGHT_DOTS)
    rng.shuffle(colors)
    dots: list[dict[str, Any]] = []
    for index in range(3):
        column = rng.randrange(len(bounds))
        left, right = bounds[column]
        x = rng.randint(left + 14, right - 14)
        y = rng.randint(inner_top + 26, inner_bottom - 26)
        dots.append({"x": x, "y": y, "r": 8, "color": list(colors[index])})

    return {
        "canvas": [width, height],
        "frame": frame,
        "frame_thickness": thickness,
        "partitions": partitions,
        "decorations": decorations,
        "dots": dots,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary, with no hidden answer mark."""

    width, height = (int(value) for value in scene["canvas"])
    image = Image.new("RGB", (width, height), "white")
    draw = ImageDraw.Draw(image)
    frame = tuple(int(value) for value in scene["frame"])
    thickness = int(scene["frame_thickness"])
    draw.rectangle(frame, fill=WALL)
    x0, y0, x1, y1 = frame
    inner = (x0 + thickness, y0 + thickness, x1 - thickness, y1 - thickness)
    draw.rectangle(inner, fill="white")
    for part in scene["partitions"]:
        draw.rectangle(tuple(int(value) for value in part), fill=WALL)
    for decoration in scene["decorations"]:
        x_start, y, x_end, _ = (int(value) for value in decoration)
        draw.rectangle((x_start, y - 3, x_end, y + 3), fill=WALL)
    for dot in scene["dots"]:
        x, y, radius = int(dot["x"]), int(dot["y"]), int(dot["r"])
        color = tuple(int(value) for value in dot["color"])
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """The visible separator list determines the number of enclosed chambers."""

    return str(len(scene["partitions"]) + 1)


def margin(scene: dict[str, Any]) -> float:
    """Positive clearance from the discrete topological boundary."""

    frame = [int(value) for value in scene["frame"]]
    thickness = int(scene["frame_thickness"])
    bounds = _chamber_bounds(frame, thickness, scene["partitions"])
    minimum_width = min(right - left for left, right in bounds)
    return float(min(thickness - 4, minimum_width - 60, 24))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject only scenes with a thin wall or a chamber too close to collapse."""

    frame = [int(value) for value in scene["frame"]]
    width, height = (int(value) for value in scene["canvas"])
    x0, y0, x1, y1 = frame
    if min(x0, y0, width - 1 - x1, height - 1 - y1) < 24:
        return True
    return margin(scene) < 7.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Reverse an unordered list of equal-colored décor strokes."""

    twin = dict(scene)
    twin["decorations"] = list(reversed(scene["decorations"]))
    return [("decor_order_reversal", twin)]
