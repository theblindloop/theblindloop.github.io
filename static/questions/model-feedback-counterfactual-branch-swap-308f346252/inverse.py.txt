"""Independent pixel-only inverse for the counterfactual branch-swap board."""
from __future__ import annotations
import numpy as np
from PIL import Image

CANVAS=(920,700); BOARD_W,BOARD_H=880,660; BORDER=(55,64,72)
REGISTERS=("A","B","C","D")
BADGES={(198,82,52):"A",(22,139,150):"B",(121,75,180):"C",(65,104,174):"D"}
SHAPES={(42,124,202):"circle",(38,157,91):"triangle",(230,158,26):"star",(205,69,119):"diamond"}

class OracleError(ValueError): pass

def _patch(array: np.ndarray,x:int,y:int,radius:int=2)->tuple[int,int,int]:
    patch=array[y-radius:y+radius+1,x-radius:x+radius+1]
    colors={tuple(int(v) for v in row) for row in patch.reshape(-1,3)}
    if len(colors)!=1: raise OracleError("a required marker is damaged")
    return next(iter(colors))

def _shape(array,x,y):
    color=_patch(array,x,y)
    if color not in SHAPES: raise OracleError("unknown shape marker")
    return SHAPES[color]

def _badge(array,cx,y,large=False):
    color=_patch(array,cx-(18 if large else 14),y)
    if color not in BADGES: raise OracleError("unknown register badge")
    return BADGES[color]

def _run(initial,cards,intervention):
    state=list(initial)
    if intervention is not None: state[REGISTERS.index(intervention[0])]=intervention[1]
    choices=[]
    for test_slot,test_shape,yes,no in cards:
        hit=state[REGISTERS.index(test_slot)]==test_shape; pair=yes if hit else no
        i,j=REGISTERS.index(pair[0]),REGISTERS.index(pair[1]); state[i],state[j]=state[j],state[i]; choices.append(hit)
    return state,choices

def decision_from_image(image: Image.Image)->str:
    array=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if (array.shape[1],array.shape[0])!=CANVAS: raise OracleError("unexpected canvas size")
    mask=np.all(array==np.asarray(BORDER,dtype=np.uint8),axis=2); points=np.argwhere(mask)
    if not len(points): raise OracleError("board absent")
    top,left=points.min(axis=0); bottom,right=points.max(axis=0)
    if int(right-left+1)!=BOARD_W or int(bottom-top+1)!=BOARD_H: raise OracleError("wrong board extent")
    ax,ay=int(left),int(top)
    if not (np.all(mask[ay,ax:ax+BOARD_W]) and np.all(mask[ay+BOARD_H-1,ax:ax+BOARD_W]) and np.all(mask[ay:ay+BOARD_H,ax]) and np.all(mask[ay:ay+BOARD_H,ax+BOARD_W-1])):
        raise OracleError("broken board border")
    initial=[_shape(array,ax+x,ay+128) for x in (110,330,550,770)]
    if sorted(initial)!=sorted(SHAPES.values()): raise OracleError("invalid initial inventory")
    intervention_slot=_badge(array,ax+355,ay+210,large=True); replacement=_shape(array,ax+535,ay+210)
    if replacement==initial[REGISTERS.index(intervention_slot)]: raise OracleError("intervention changes nothing")
    cards=[]; card_xs=(25,315,605); card_ys=(285,455)
    for index in range(6):
        x,y=ax+card_xs[index%3],ay+card_ys[index//3]
        test_slot=_badge(array,x+93,y+49); test_shape=_shape(array,x+165,y+49)
        yes=(_badge(array,x+105,y+94),_badge(array,x+190,y+94)); no=(_badge(array,x+105,y+132),_badge(array,x+190,y+132))
        if yes[0]==yes[1] or no[0]==no[1] or yes==no: raise OracleError("malformed branch swap")
        cards.append((test_slot,test_shape,yes,no))
    factual,fc=_run(initial,cards,None); counterfactual,cc=_run(initial,cards,(intervention_slot,replacement))
    if factual[3]==counterfactual[3] or sum(a!=b for a,b in zip(fc,cc,strict=True))<3:
        raise OracleError("intervention is not causally deep")
    answer=counterfactual[3]
    if answer not in SHAPES.values(): raise OracleError("unknown answer")
    return answer
