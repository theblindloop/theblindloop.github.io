"""Deterministic renderer for a counterfactual conditional-swap program."""
from __future__ import annotations
import math
import random
from PIL import Image, ImageDraw, ImageFont

CANVAS = (920, 700)
BOARD_W, BOARD_H = 880, 660
REGISTERS = ("A", "B", "C", "D")
ANSWERS = ("circle", "triangle", "star", "diamond")
RGB = {
    "white": (255,255,255), "board": (246,248,244), "card": (255,255,252),
    "ink": (37,43,49), "muted": (100,108,114), "border": (55,64,72),
    "intervene": (204,43,55), "yes": (31,133,83), "no": (185,73,48),
    "A": (198,82,52), "B": (22,139,150), "C": (121,75,180), "D": (65,104,174),
    "circle": (42,124,202), "triangle": (38,157,91), "star": (230,158,26), "diamond": (205,69,119),
}
FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _run(initial: list[str], cards: list[dict], intervention: tuple[int,str] | None) -> tuple[list[str],list[bool]]:
    state = list(initial)
    if intervention is not None:
        state[intervention[0]] = intervention[1]
    choices = []
    for card in cards:
        hit = state[REGISTERS.index(card["test_slot"])] == card["test_shape"]
        pair = card["yes"] if hit else card["no"]
        i, j = REGISTERS.index(pair[0]), REGISTERS.index(pair[1])
        state[i], state[j] = state[j], state[i]
        choices.append(hit)
    return state, choices

def _worlds(scene: dict) -> tuple[list[str],list[str],list[bool],list[bool]]:
    initial = [str(v) for v in scene["initial"]]
    cards = [dict(c) for c in scene["cards"]]
    slot = REGISTERS.index(str(scene["intervention_slot"]))
    factual, fc = _run(initial, cards, None)
    counterfactual, cc = _run(initial, cards, (slot, str(scene["replacement_shape"])))
    return factual, counterfactual, fc, cc

def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xC0FA17A1) & 0xFFFFFFFFFFFFFFFF)
    wanted = ANSWERS[int(seed) % 4]
    pairs = [(a,b) for i,a in enumerate(REGISTERS) for b in REGISTERS[i+1:]]
    for _ in range(300000):
        initial = list(ANSWERS); rng.shuffle(initial)
        intervention_index = rng.randrange(4)
        replacement = rng.choice([s for s in ANSWERS if s != initial[intervention_index]])
        cards = []
        for _card in range(6):
            yes = rng.choice(pairs); no = rng.choice([p for p in pairs if p != yes])
            cards.append({"test_slot": rng.choice(REGISTERS), "test_shape": rng.choice(ANSWERS), "yes": list(yes), "no": list(no)})
        probe = {"initial": initial, "intervention_slot": REGISTERS[intervention_index], "replacement_shape": replacement, "cards": cards, "anchor_x": 0, "anchor_y": 0}
        factual, counterfactual, fc, cc = _worlds(probe)
        if counterfactual[3] == wanted and factual[3] != wanted and sum(a != b for a,b in zip(fc,cc,strict=True)) >= 3:
            return {**probe, "anchor_x": rng.randint(18,22), "anchor_y": rng.randint(18,22)}
    raise RuntimeError("could not sample a causally deep conditional-swap program")

def analytic_gold(scene: dict) -> str:
    initial = [str(v) for v in scene["initial"]]; cards = [dict(c) for c in scene["cards"]]
    if len(initial) != 4 or sorted(initial) != sorted(ANSWERS) or len(cards) != 6:
        raise ValueError("invalid initial row or card count")
    slot, replacement = str(scene["intervention_slot"]), str(scene["replacement_shape"])
    if slot not in REGISTERS or replacement not in ANSWERS or replacement == initial[REGISTERS.index(slot)]:
        raise ValueError("invalid intervention")
    for card in cards:
        if card.get("test_slot") not in REGISTERS or card.get("test_shape") not in ANSWERS:
            raise ValueError("invalid predicate")
        yes, no = card.get("yes"), card.get("no")
        if not isinstance(yes,list) or not isinstance(no,list) or len(yes)!=2 or len(no)!=2:
            raise ValueError("invalid swap")
        if any(v not in REGISTERS for v in yes+no) or yes[0]==yes[1] or no[0]==no[1] or yes==no:
            raise ValueError("invalid swap pair")
    factual, counterfactual, fc, cc = _worlds(scene)
    if factual[3] == counterfactual[3] or sum(a != b for a,b in zip(fc,cc,strict=True)) < 3:
        raise ValueError("intervention is not causally deep")
    return counterfactual[3]

def margin(scene: dict) -> float:
    try: return 1.0 if analytic_gold(scene) in ANSWERS else 0.0
    except (KeyError,TypeError,ValueError): return 0.0
def is_quarantined(scene: dict) -> bool: return margin(scene) < 1.0
def latent_symmetries(scene: dict) -> list[tuple[str,dict]]: return []
def _font(size: int, bold: bool=False): return ImageFont.truetype(BOLD if bold else FONT, size)
def _text(draw, xy, value, size, *, fill=None, bold=False, anchor="mm"): draw.text(xy,value,font=_font(size,bold),fill=fill or RGB["ink"],anchor=anchor)

def _shape(draw, center, kind: str, scale: int=18) -> None:
    x,y=center; color=RGB[kind]
    if kind=="circle": draw.ellipse((x-scale,y-scale,x+scale,y+scale),fill=color,outline=RGB["ink"],width=2)
    elif kind=="triangle":
        pts=((x,y-scale-2),(x-scale-2,y+scale),(x+scale+2,y+scale)); draw.polygon(pts,fill=color); draw.line(pts+(pts[0],),fill=RGB["ink"],width=2)
    elif kind=="diamond":
        pts=((x,y-scale-2),(x-scale-2,y),(x,y+scale+2),(x+scale+2,y)); draw.polygon(pts,fill=color); draw.line(pts+(pts[0],),fill=RGB["ink"],width=2)
    elif kind=="star":
        pts=[]
        for i in range(10):
            radius=scale+2 if i%2==0 else scale*.45; angle=-math.pi/2+i*math.pi/5
            pts.append((x+radius*math.cos(angle),y+radius*math.sin(angle)))
        draw.polygon(pts,fill=color); draw.line(pts+[pts[0]],fill=RGB["ink"],width=2,joint="curve")
    else: raise ValueError(kind)

def _badge(draw, center, label: str, small: bool=False) -> None:
    x,y=center; w,h=((38,27) if small else (48,34))
    draw.rounded_rectangle((x-w//2,y-h//2,x+w//2,y+h//2),radius=6,fill=RGB[label],outline=RGB["ink"],width=2)
    _text(draw,center,label,16 if small else 19,fill=RGB["white"],bold=True)

def render(scene: dict) -> Image.Image:
    analytic_gold(scene); ax,ay=int(scene["anchor_x"]),int(scene["anchor_y"])
    image=Image.new("RGB",CANVAS,RGB["white"]); draw=ImageDraw.Draw(image)
    draw.rectangle((ax,ay,ax+BOARD_W-1,ay+BOARD_H-1),fill=RGB["board"],outline=RGB["border"],width=4)
    _text(draw,(ax+440,ay+24),"WHAT-IF SWAP PROGRAM",22,bold=True); _text(draw,(ax+440,ay+50),"Start tokens",16,fill=RGB["muted"],bold=True)
    for label,kind,xoff in zip(REGISTERS,scene["initial"],(110,330,550,770),strict=True):
        x=ax+xoff; draw.rounded_rectangle((x-75,ay+67,x+75,ay+157),radius=9,fill=RGB["card"],outline=RGB[label],width=4)
        _badge(draw,(x-45,ay+91),label,small=True); _shape(draw,(x,ay+128),str(kind),20)
    draw.rounded_rectangle((ax+100,ay+174,ax+780,ay+247),radius=10,fill=(255,243,243),outline=RGB["intervene"],width=4)
    _text(draw,(ax+183,ay+210),"BEFORE CARD 1",16,fill=RGB["intervene"],bold=True); _badge(draw,(ax+355,ay+210),str(scene["intervention_slot"]))
    _text(draw,(ax+430,ay+210),"SET TO",17,bold=True); _shape(draw,(ax+535,ay+210),str(scene["replacement_shape"]),21)
    _text(draw,(ax+650,ay+210),"RED WRENCH",14,fill=RGB["intervene"],bold=True)
    _text(draw,(ax+440,ay+267),"For each card: test first, then do exactly one swap",16,bold=True)
    card_xs,card_ys=(25,315,605),(285,455)
    for index,card in enumerate(scene["cards"]):
        x,y=ax+card_xs[index%3],ay+card_ys[index//3]
        draw.rounded_rectangle((x,y,x+250,y+150),radius=9,fill=RGB["card"],outline=RGB["border"],width=3)
        draw.ellipse((x+8,y+8,x+38,y+38),fill=RGB["ink"]); _text(draw,(x+23,y+23),str(index+1),16,fill=RGB["white"],bold=True)
        _text(draw,(x+54,y+49),"IF",15,bold=True); _badge(draw,(x+93,y+49),str(card["test_slot"]),small=True); _text(draw,(x+126,y+49),"=",18,bold=True); _shape(draw,(x+165,y+49),str(card["test_shape"]),15)
        _text(draw,(x+39,y+94),"YES",14,fill=RGB["yes"],bold=True); _badge(draw,(x+105,y+94),str(card["yes"][0]),small=True); _text(draw,(x+145,y+94),"↔",22,bold=True); _badge(draw,(x+190,y+94),str(card["yes"][1]),small=True)
        _text(draw,(x+39,y+132),"NO",14,fill=RGB["no"],bold=True); _badge(draw,(x+105,y+132),str(card["no"][0]),small=True); _text(draw,(x+145,y+132),"↔",22,bold=True); _badge(draw,(x+190,y+132),str(card["no"][1]),small=True)
    return image
