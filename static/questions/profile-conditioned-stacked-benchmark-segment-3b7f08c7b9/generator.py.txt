from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 420
BAR_LEFT, BAR_TOP, BAR_WIDTH, BAR_HEIGHT = 100, 170, 520, 80
COLORS = ("blue", "green", "magenta", "orange")
RGB = {
    "blue": (37, 99, 235),
    "green": (22, 163, 74),
    "magenta": (192, 38, 211),
    "orange": (234, 88, 12),
}


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = f"/usr/share/fonts/truetype/dejavu/{name}"
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    order = list(COLORS)
    rng.shuffle(order)
    widths = [100, 100, 100, 100]
    for _ in range(BAR_WIDTH - sum(widths)):
        widths[rng.randrange(4)] += 1
    target_color = COLORS[int(seed) % len(COLORS)]
    target_index = order.index(target_color)
    left = BAR_LEFT + sum(widths[:target_index])
    right = left + widths[target_index]
    benchmark_x = rng.randint(left + 25, right - 25)
    return {
        "segment_order": order,
        "widths": widths,
        "benchmark_x": benchmark_x,
    }


def _layout(scene: dict[str, Any]) -> list[tuple[str, int, int]]:
    order = list(scene["segment_order"])
    widths = [int(v) for v in scene["widths"]]
    benchmark_x = int(scene["benchmark_x"])
    if sorted(order) != sorted(COLORS) or len(widths) != 4 or sum(widths) != BAR_WIDTH:
        raise ValueError("invalid stacked-bar scene")
    if not BAR_LEFT < benchmark_x < BAR_LEFT + BAR_WIDTH:
        raise ValueError("benchmark outside bar")
    result = []
    x = BAR_LEFT
    for color, width in zip(order, widths, strict=True):
        if width < 40:
            raise ValueError("segment too narrow")
        result.append((color, x, x + width))
        x += width
    return result


def analytic_gold(scene: dict[str, Any]) -> str:
    benchmark_x = int(scene["benchmark_x"])
    order = list(scene["segment_order"])
    widths = [int(v) for v in scene["widths"]]
    x = BAR_LEFT
    for color, width in zip(order, widths, strict=True):
        if x < benchmark_x < x + width:
            return color
        x += width
    raise ValueError("benchmark is not inside a unique segment")


def margin(scene: dict[str, Any]) -> float:
    benchmark_x = int(scene["benchmark_x"])
    boundaries = [BAR_LEFT]
    x = BAR_LEFT
    for width in scene["widths"]:
        x += int(width)
        boundaries.append(x)
    return float(min(abs(benchmark_x - boundary) for boundary in boundaries))


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        _layout(scene)
        return margin(scene) < 12.0
    except (TypeError, ValueError, KeyError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    layout = _layout(scene)
    benchmark_x = int(scene["benchmark_x"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 250, 252))
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 48), "STACKED COMPOSITION", fill=(51, 65, 85),
              font=_font(26, True), anchor="mm")
    draw.text((WIDTH // 2, 88), "Dashed benchmark", fill=(100, 116, 139),
              font=_font(17), anchor="mm")

    for color, left, right in layout:
        draw.rectangle((left, BAR_TOP, right - 1, BAR_TOP + BAR_HEIGHT - 1), fill=RGB[color])
        draw.text(((left + right) // 2, BAR_TOP + BAR_HEIGHT // 2), color.upper(),
                  fill=(255, 255, 255), font=_font(16, True), anchor="mm")
    for _, _, right in layout[:-1]:
        draw.rectangle((right - 2, BAR_TOP, right + 1, BAR_TOP + BAR_HEIGHT - 1),
                       fill=(248, 250, 252))

    for y in range(122, 292, 18):
        draw.rectangle((benchmark_x - 1, y, benchmark_x + 1, min(y + 10, 292)), fill=(0, 0, 0))
    draw.polygon([(benchmark_x, 154), (benchmark_x - 9, 138), (benchmark_x + 9, 138)],
                 fill=(0, 0, 0))
    draw.text((BAR_LEFT, 284), "0%", fill=(51, 65, 85), font=_font(16), anchor="ma")
    draw.text((BAR_LEFT + BAR_WIDTH, 284), "100%", fill=(51, 65, 85), font=_font(16), anchor="ma")
    draw.text((WIDTH // 2, 344), "Locate the benchmark relative to the visible segment joins.",
              fill=(71, 85, 105), font=_font(17), anchor="mm")
    return image
