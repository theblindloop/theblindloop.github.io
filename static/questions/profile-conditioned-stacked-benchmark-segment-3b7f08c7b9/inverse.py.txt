from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (37, 99, 235),
    "green": (22, 163, 74),
    "magenta": (192, 38, 211),
    "orange": (234, 88, 12),
}


def decision_from_image(image: Image.Image) -> str:
    """Recover the containing segment using pixels only."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    h, w, _ = arr.shape
    if h < 300 or w < 500:
        raise ValueError("image too small")

    # Pure-black ink in the chart band is reserved for the dashed benchmark.
    band = arr[110:300]
    black = np.max(band, axis=2) <= 8
    ys, xs = np.nonzero(black)
    if len(xs) < 30:
        raise ValueError("benchmark not found")
    counts = np.bincount(xs, minlength=w)
    benchmark_x = int(np.argmax(counts))
    if counts[benchmark_x] < 20:
        raise ValueError("benchmark line is unstable")

    containing: list[str] = []
    for name, rgb in _PALETTE.items():
        distance = np.max(np.abs(arr - np.array(rgb, dtype=np.int16)), axis=2)
        mask = distance <= 8
        yy, xx = np.nonzero(mask)
        if len(xx) < 1000:
            raise ValueError(f"missing segment {name}")
        x0, x1 = int(xx.min()), int(xx.max())
        y0, y1 = int(yy.min()), int(yy.max())
        if y1 - y0 < 50 or x1 - x0 < 35:
            raise ValueError(f"malformed segment {name}")
        if x0 < benchmark_x < x1:
            containing.append(name)
    if len(containing) != 1:
        raise ValueError(f"benchmark has {len(containing)} containing segments")
    return containing[0]
