"""Deterministic latent sampler and Pillow renderer for heartbeat excursions."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = (640, 420)
TRACE_PALETTE = (
    (28, 72, 124),
    (67, 48, 118),
    (22, 91, 84),
    (43, 65, 91),
)
GUIDE_PALETTE = (
    (210, 52, 62),
    (198, 64, 43),
    (220, 76, 49),
)


def _rng(seed: int, salt: int) -> random.Random:
    return random.Random((int(seed) * 0x9E3779B1 + salt) & 0xFFFFFFFFFFFFFFFF)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one matched-style control/counterfactual scene.

    ``excursion_count == 1`` is the familiar one-upward-spike control. Counts
    2--4 are counterfactuals. Palette, translation, scale, stroke width, and
    slot order are sampled on seed channels that do not use the answer value.
    Every scene still contains four equal-complexity deflections: slots not
    counted above the guide are reflected downward, limiting total-ink cues.
    """

    seed = int(seed)
    count = 1 + (seed % 4)
    layout_rng = _rng(seed, 0xA17E)
    order_rng = _rng(seed, 0xB42D)

    active = list(range(4))
    order_rng.shuffle(active)
    active_set = set(active[:count])
    states = ["up" if index in active_set else "down" for index in range(4)]

    slot_width = 82 + 2 * layout_rng.randrange(5)
    x_start = 134 + layout_rng.randint(-10, 10)
    guide_y = 188 + layout_rng.randint(-8, 8)
    baseline_gap = 21 + layout_rng.randint(0, 9)
    baseline_y = guide_y + baseline_gap
    clearance = 23 + layout_rng.randint(0, 13)
    amplitude = baseline_gap + clearance
    stroke_width = 8 + layout_rng.randrange(3)

    points: list[list[int]] = [[x_start - 16, baseline_y]]
    for slot, state in enumerate(states):
        left = x_start + slot * slot_width
        apex_y = baseline_y - amplitude if state == "up" else baseline_y + amplitude
        local = (
            [left, baseline_y],
            [left + round(0.17 * slot_width), baseline_y - 5],
            [left + round(0.31 * slot_width), baseline_y + 6],
            [left + round(0.50 * slot_width), apex_y],
            [left + round(0.69 * slot_width), baseline_y + 6],
            [left + round(0.83 * slot_width), baseline_y - 5],
            [left + slot_width, baseline_y],
        )
        for point in local:
            if point != points[-1]:
                points.append(point)
    points.append([x_start + 4 * slot_width + 16, baseline_y])

    return {
        "excursion_count": count,
        "slot_states": states,
        "trace_points": points,
        "guide_y": guide_y,
        "trace_color": list(TRACE_PALETTE[(seed // 4) % len(TRACE_PALETTE)]),
        "guide_color": list(GUIDE_PALETTE[(seed // 16) % len(GUIDE_PALETTE)]),
        "stroke_width": stroke_width,
        "canvas_width": CANVAS[0],
        "canvas_height": CANVAS[1],
    }


def _scaled_points(points: list[list[int]], scale: int) -> list[tuple[int, int]]:
    return [(int(x) * scale, int(y) * scale) for x, y in points]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only public scene fields; no answer annotation is painted."""

    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    points = [[int(x), int(y)] for x, y in scene["trace_points"]]
    # A visual polyline is undirected. Canonicalize its serialization so the
    # two equivalent endpoint-to-endpoint point orders rasterize identically.
    if points and points[0][0] > points[-1][0]:
        points.reverse()
    guide_y = int(scene["guide_y"])
    trace_color = tuple(int(value) for value in scene["trace_color"])
    guide_color = tuple(int(value) for value in scene["guide_color"])
    stroke_width = int(scene["stroke_width"])
    states = list(scene["slot_states"])
    declared_count = int(scene["excursion_count"])
    if sum(state == "up" for state in states) != declared_count:
        raise ValueError("slot states disagree with excursion_count")
    if width != 640 or height != 420 or len(points) < 8:
        raise ValueError("unsupported canvas or trace")

    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), (235, 239, 243))
    draw = ImageDraw.Draw(image)

    card = tuple(value * scale for value in (112, 60, 528, 368))
    draw.rounded_rectangle(
        card,
        radius=18 * scale,
        fill=(249, 250, 247),
        outline=(137, 143, 149),
        width=2 * scale,
    )
    for x in (160, 240, 320, 400, 480):
        draw.line(
            [(x * scale, 78 * scale), (x * scale, 350 * scale)],
            fill=(229, 233, 230),
            width=1 * scale,
        )
    for y in (120, 160, 200, 240, 280, 320):
        draw.line(
            [(130 * scale, y * scale), (510 * scale, y * scale)],
            fill=(229, 233, 230),
            width=1 * scale,
        )

    draw.line(
        [(138 * scale, guide_y * scale), (502 * scale, guide_y * scale)],
        fill=guide_color,
        width=3 * scale,
    )

    scaled = _scaled_points(points, scale)
    draw.line(scaled, fill=trace_color, width=stroke_width * scale, joint="curve")
    radius = stroke_width * scale // 2
    # Explicit round joints make the path independent of point ordering.
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=trace_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["excursion_count"]))


def margin(scene: dict[str, Any]) -> float:
    """Runner-up clearance in pixels from an up/down slot to the guide."""

    guide_y = float(scene["guide_y"])
    half_stroke = float(scene["stroke_width"]) / 2.0
    states = list(scene["slot_states"])
    points = scene["trace_points"]
    # Each seven-point slot contributes its apex at offsets 4, 10, 16, 22
    # after the entry point and de-duplication used by the sampler.
    apex_indices = (4, 10, 16, 22)
    clearances = []
    for state, index in zip(states, apex_indices, strict=True):
        apex_y = float(points[index][1])
        if state == "up":
            clearances.append(guide_y - apex_y - half_stroke)
        else:
            # The closest non-counted shoulder is five pixels above baseline.
            slot_start = 1 + apex_indices.index(index) * 6
            shoulder_y = min(float(points[slot_start + 1][1]), float(points[slot_start + 5][1]))
            clearances.append(shoulder_y - guide_y - half_stroke)
    return float(min(clearances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 16.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["trace_points"] = list(reversed(twin["trace_points"]))
    return [("reverse_undirected_trace_point_order", twin)]
