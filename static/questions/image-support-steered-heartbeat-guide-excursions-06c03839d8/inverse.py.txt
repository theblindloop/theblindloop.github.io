"""Independent image-only inverse arm for heartbeat guide excursions.

This module deliberately knows only the public raster specification: a warm
horizontal guide and one dark, cool continuous trace. It imports no renderer,
scene, prompt, generator, verifier, or analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not expose a single unambiguous trace/guide decision."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    outputs: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = y + dy, x + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        outputs.append(np.asarray(pixels, dtype=np.int32))
    return outputs


def _cool_trace_mask(array: np.ndarray) -> np.ndarray:
    rgb = array.astype(np.int16)
    red, green, blue = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    chroma = np.maximum.reduce((red, green, blue)) - np.minimum.reduce((red, green, blue))
    return (
        (np.maximum.reduce((red, green, blue)) < 165)
        & (chroma > 22)
        & ((blue > red + 14) | (green > red + 18))
    )


def _warm_guide_mask(array: np.ndarray) -> np.ndarray:
    rgb = array.astype(np.int16)
    red, green, blue = rgb[..., 0], rgb[..., 1], rgb[..., 2]
    return (red > 155) & (red > green + 65) & (red > blue + 48)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    height, width = array.shape[:2]
    if width < 320 or height < 200:
        raise OracleAbstention("canvas is too small")

    guide = _warm_guide_mask(array)
    gy, gx = np.nonzero(guide)
    if len(gy) < 350 or int(gx.max() - gx.min()) < int(0.46 * width):
        raise OracleAbstention("the visible guide is missing or truncated")
    guide_y = float(np.median(gy))
    if float(np.percentile(gy, 90) - np.percentile(gy, 10)) > 5.0:
        raise OracleAbstention("warm pixels do not form one horizontal guide")

    trace = _cool_trace_mask(array)
    components = [component for component in _components(trace) if len(component) >= 35]
    if len(components) != 1:
        raise OracleAbstention("trace is absent, disconnected, or multiply connected")
    component = components[0]
    y_span = int(component[:, 0].max() - component[:, 0].min())
    x_span = int(component[:, 1].max() - component[:, 1].min())
    if not (0.56 * width <= x_span <= 0.68 * width) or y_span < 0.14 * height:
        raise OracleAbstention("trace extent is inconsistent with a complete monitor path")

    dominant = np.zeros((height, width), dtype=bool)
    dominant[component[:, 0], component[:, 1]] = True
    above = dominant & (np.arange(height)[:, None] < guide_y - 2.0)
    excursions = []
    for candidate in _components(above):
        if len(candidate) < 90:
            continue
        candidate_height = int(candidate[:, 0].max() - candidate[:, 0].min())
        candidate_width = int(candidate[:, 1].max() - candidate[:, 1].min())
        if candidate_height >= 18 and candidate_width >= 12:
            excursions.append(candidate)
    count = len(excursions)
    if count not in (1, 2, 3, 4):
        raise OracleAbstention(f"recovered unsupported excursion count {count}")
    return str(count)
