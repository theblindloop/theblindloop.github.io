"""Deterministic renderer for the two-pip die alignment world.

The familiar control is a conventional diagonal two-pip face.  Its matched
counterfactual uses the same face, palette, position, pip count, pip size, pair
distance, and total ink, but the two pips share a row or column.
"""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
ALIGNMENT_THRESHOLD_PX = 32.0
QUARANTINE_BAND_PX = 8.0
PAIR_HALF_DIAGONAL_PX = 48
PAIR_HALF_AXIAL_PX = 68
SUPERSAMPLE = 3

_PIP_PALETTE = (
    (37, 99, 235),
    (14, 146, 136),
    (200, 48, 118),
    (234, 88, 12),
)
_FACE_PALETTE = (
    (252, 250, 244),
    (247, 250, 252),
    (252, 247, 242),
    (246, 249, 245),
)
_CENTER_VALUES = (164, 228, 292, 356)


def _style_rng(seed: int) -> random.Random:
    # Adjacent even/odd seeds form an explicitly matched pair.
    return random.Random((seed // 2) ^ 0xD1E5A11)


def sample_scene(seed: int) -> dict:
    """Sample one scene; even seeds are controls and odd seeds counterfactuals."""

    rng = _style_rng(int(seed))
    diagonal = int(seed) % 2 == 0
    cx = rng.choice(_CENTER_VALUES)
    cy = rng.choice(_CENTER_VALUES)
    pip_radius = rng.choice((13, 14, 15))
    pip_rgb = list(rng.choice(_PIP_PALETTE))
    face_rgb = list(rng.choice(_FACE_PALETTE))
    face_size = rng.choice((208, 212, 216))
    shadow_offset = [rng.choice((5, 6, 7)), rng.choice((6, 7, 8))]

    # Small neutral marks make remote pixels visibly non-diagnostic.  Sample
    # them before the class-specific layout branch so matched pairs retain the
    # exact same decoration.
    decor_pool = [
        [30, 34, 3],
        [478, 39, 4],
        [35, 472, 4],
        [475, 474, 3],
        [70, 256, 3],
        [442, 256, 3],
    ]
    rng.shuffle(decor_pool)

    if diagonal:
        slope = rng.choice((-1, 1))
        centers = [
            [cx - PAIR_HALF_DIAGONAL_PX, cy - slope * PAIR_HALF_DIAGONAL_PX],
            [cx + PAIR_HALF_DIAGONAL_PX, cy + slope * PAIR_HALF_DIAGONAL_PX],
        ]
    elif rng.random() < 0.5:
        centers = [
            [cx - PAIR_HALF_AXIAL_PX, cy],
            [cx + PAIR_HALF_AXIAL_PX, cy],
        ]
    else:
        centers = [
            [cx, cy - PAIR_HALF_AXIAL_PX],
            [cx, cy + PAIR_HALF_AXIAL_PX],
        ]
    if rng.random() < 0.5:
        centers.reverse()

    return {
        "canvas_size": [CANVAS_SIZE, CANVAS_SIZE],
        "face_center": [cx, cy],
        "face_size": face_size,
        "face_rgb": face_rgb,
        "outline_rgb": [70, 73, 78],
        "pip_centers": centers,
        "pip_radius": pip_radius,
        "pip_rgb": pip_rgb,
        "shadow_offset": shadow_offset,
        "decor_marks": decor_pool[:4],
    }


def _minor_axis_separation(scene: dict) -> float:
    (x1, y1), (x2, y2) = scene["pip_centers"]
    return float(min(abs(x2 - x1), abs(y2 - y1)))


def analytic_gold(scene: dict) -> str:
    return "yes" if _minor_axis_separation(scene) > ALIGNMENT_THRESHOLD_PX else "no"


def margin(scene: dict) -> float:
    """Absolute runner-up margin to the minor-axis alignment boundary, in px."""

    return float(abs(_minor_axis_separation(scene) - ALIGNMENT_THRESHOLD_PX))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_BAND_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise the order alias of the two indistinguishable pips."""

    twin = deepcopy(scene)
    twin["pip_centers"] = list(reversed(twin["pip_centers"]))
    return [("reverse_identical_pip_order", twin)]


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def render(scene: dict) -> Image.Image:
    """Render the latent scene to a 512x512 RGB PNG-ready image."""

    width, height = (int(value) for value in scene["canvas_size"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), (244, 241, 234))
    draw = ImageDraw.Draw(image)

    # A restrained paper texture, deterministic and achromatic.
    for y in range(18, height, 28):
        shade = 238 + ((y // 28) % 3)
        draw.line(
            [(0, y * scale), (width * scale, y * scale)],
            fill=(shade, shade, shade),
            width=1,
        )

    for x, y, radius in scene["decor_marks"]:
        box = (x - radius, y - radius, x + radius, y + radius)
        draw.ellipse(_scaled_box(box, scale), fill=(177, 177, 174))

    cx, cy = scene["face_center"]
    face_size = float(scene["face_size"])
    half = face_size / 2.0
    dx, dy = scene["shadow_offset"]
    shadow_box = (cx - half + dx, cy - half + dy, cx + half + dx, cy + half + dy)
    draw.rounded_rectangle(
        _scaled_box(shadow_box, scale),
        radius=25 * scale,
        fill=(207, 203, 194),
    )
    face_box = (cx - half, cy - half, cx + half, cy + half)
    draw.rounded_rectangle(
        _scaled_box(face_box, scale),
        radius=25 * scale,
        fill=tuple(scene["face_rgb"]),
        outline=tuple(scene["outline_rgb"]),
        width=3 * scale,
    )

    # Neutral recessed slot marks evoke the usual 3x3 die lattice without
    # being read by the chromatic inverse arm.
    for ox in (-48, 0, 48):
        for oy in (-48, 0, 48):
            r = 3
            box = (cx + ox - r, cy + oy - r, cx + ox + r, cy + oy + r)
            draw.ellipse(_scaled_box(box, scale), fill=(218, 216, 210))

    pip_color = tuple(scene["pip_rgb"])
    pip_radius = float(scene["pip_radius"])
    for px, py in scene["pip_centers"]:
        box = (px - pip_radius, py - pip_radius, px + pip_radius, py + pip_radius)
        draw.ellipse(_scaled_box(box, scale), fill=pip_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)
