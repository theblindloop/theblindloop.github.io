"""Independent pixel-only inverse arm for two-pip alignment.

Only the final raster is accepted.  Saturated connected components are located
without knowledge of the renderer palette, face position, pip size, or latent
placement.  The two recovered centroids directly determine the answer.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ALIGNMENT_THRESHOLD_PX = 32.0
MIN_COMPONENT_AREA = 220
MAX_COMPONENT_AREA = 1100


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return []
    y0, y1 = max(0, int(ys.min()) - 1), min(mask.shape[0], int(ys.max()) + 2)
    x0, x1 = max(0, int(xs.min()) - 1), min(mask.shape[1], int(xs.max()) + 2)
    sub = mask[y0:y1, x0:x1]
    seen = np.zeros_like(sub, dtype=bool)
    result: list[np.ndarray] = []
    for sy, sx in np.argwhere(sub):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy + y0, cx + x0))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if (
                    0 <= ny < sub.shape[0]
                    and 0 <= nx < sub.shape[1]
                    and sub[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(pixels, dtype=np.float64))
    return result


def _pip_centroids(image: Image.Image) -> list[tuple[float, float]]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    maximum = rgb.max(axis=2)
    minimum = rgb.min(axis=2)
    chroma = maximum - minimum
    # All scenery is achromatic; pips are the only high-chroma regions.  The
    # rule deliberately does not enumerate or import renderer colors.
    mask = (chroma >= 72) & (maximum >= 90)
    components = [part for part in _components(mask) if len(part) >= MIN_COMPONENT_AREA]
    if len(components) != 2:
        raise OracleAbstention(f"expected exactly two visible colored pips, found {len(components)}")
    centroids: list[tuple[float, float]] = []
    areas = []
    for part in components:
        area = len(part)
        if not MIN_COMPONENT_AREA <= area <= MAX_COMPONENT_AREA:
            raise OracleAbstention(f"colored component area {area} is outside the pip range")
        areas.append(area)
        cy, cx = part.mean(axis=0)
        centroids.append((float(cx), float(cy)))
    if max(areas) / min(areas) > 1.28:
        raise OracleAbstention("the two colored components have incompatible visible areas")
    return centroids


def decision_from_image(image: Image.Image) -> str:
    """Return yes iff the two visible pip centroids differ on both axes."""

    (x1, y1), (x2, y2) = _pip_centroids(image)
    minor_axis_separation = min(abs(x2 - x1), abs(y2 - y1))
    return "yes" if minor_axis_separation > ALIGNMENT_THRESHOLD_PX else "no"
