"""Independent image-only oracle for marked-edge connectivity.

The oracle segments the four panel interiors, finds the green and amber disks
and cyan collar in each, erases a disk around the collar, and flood-fills the
remaining dark ink.  It never imports or receives the latent scene.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BOXES = (
    (40, 95, 680, 390),
    (720, 95, 1360, 390),
    (40, 430, 680, 725),
    (720, 430, 1360, 725),
)
_INK = np.array([28, 38, 52], dtype=np.float32)
_GREEN = np.array([28, 166, 91], dtype=np.float32)
_AMBER = np.array([224, 145, 27], dtype=np.float32)
_CYAN = np.array([16, 164, 190], dtype=np.float32)
_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: float = 55.0) -> np.ndarray:
    return np.linalg.norm(arr - color.reshape(1, 1, 3), axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 8:
        raise ValueError("required colored mark is missing")
    return float(xs.mean()), float(ys.mean())


def _terminal_seeds(dark: np.ndarray, center: tuple[float, float]) -> list[tuple[int, int]]:
    yy, xx = np.indices(dark.shape)
    near = dark & ((xx - center[0]) ** 2 + (yy - center[1]) ** 2 <= 35.0 ** 2)
    points = np.argwhere(near)
    if len(points) == 0:
        raise ValueError("terminal has no adjacent dark route pixels")
    return [(int(y), int(x)) for y, x in points]


def _reachable(dark: np.ndarray, seeds: list[tuple[int, int]], target: list[tuple[int, int]]) -> bool:
    height, width = dark.shape
    seen = np.zeros_like(dark, dtype=bool)
    queue = deque()
    for y, x in seeds:
        if dark[y, x] and not seen[y, x]:
            seen[y, x] = True
            queue.append((y, x))
    target_set = set(target)
    while queue:
        y, x = queue.popleft()
        if (y, x) in target_set:
            return True
        for ny, nx in (
            (y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1),
            (y - 1, x - 1), (y - 1, x + 1), (y + 1, x - 1), (y + 1, x + 1),
        ):
            if 0 <= ny < height and 0 <= nx < width and dark[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                queue.append((ny, nx))
    return False


def _panel_survives(arr: np.ndarray, box: tuple[int, int, int, int]) -> bool:
    x0, y0, x1, y1 = box
    sub = arr[y0 + 4:y1 - 4, x0 + 4:x1 - 4]
    dark = _mask(sub, _INK, 42.0)
    green = _centroid(_mask(sub, _GREEN, 65.0))
    amber = _centroid(_mask(sub, _AMBER, 65.0))
    collar_pixels = _mask(sub, _CYAN, 70.0)
    collar = _centroid(collar_pixels)

    yy, xx = np.indices(dark.shape)
    dark[((xx - collar[0]) ** 2 + (yy - collar[1]) ** 2) <= 29.0 ** 2] = False
    start = _terminal_seeds(dark, green)
    finish = _terminal_seeds(dark, amber)
    return _reachable(dark, start, finish)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    winners = [index for index, box in enumerate(_BOXES) if _panel_survives(arr, box)]
    if len(winners) != 1:
        return "ambiguous"
    return _LABELS[winners[0]]
