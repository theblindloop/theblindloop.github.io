"""Renderer and analytic arm for an edge-cut connectivity question.

Every panel is a small planar graph with colored terminal disks.  The cyan
collar identifies an edge whose pixels are removed by the question.  Exactly
one panel has an alternate route around its marked edge.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 820
PANEL_BOXES = (
    (40, 95, 680, 390),
    (720, 95, 1360, 390),
    (40, 430, 680, 725),
    (720, 430, 1360, 725),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")

BACKGROUND = (247, 249, 252)
PANEL_FILL = (255, 255, 255)
FRAME = (105, 119, 137)
INK = (28, 38, 52)
GREEN = (28, 166, 91)
AMBER = (224, 145, 27)
CYAN = (16, 164, 190)

EDGE_WIDTH = 14
NODE_RADIUS = 10
TERMINAL_RADIUS = 13
TERMINAL_HALO = 19
CUT_COLLAR_RADIUS = 23
CUT_COLLAR_WIDTH = 5
CUT_RADIUS = 29.0


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _base_graph(panel: dict) -> tuple[dict[str, tuple[float, float]], list[tuple[str, str]], int]:
    """Return local node coordinates, edges, and the marked edge index."""
    if panel["kind"] == "resilient":
        if int(panel["variant"]) == 0:
            nodes = {
                "s": (-260, 0), "g": (260, 0),
                "u1": (-170, -88), "u2": (-25, -88), "u3": (155, -88),
                "l1": (-170, 88), "l2": (-25, 88), "l3": (155, 88),
            }
        else:
            nodes = {
                "s": (-260, 0), "g": (260, 0),
                "u1": (-190, -68), "u2": (-58, -106), "u3": (132, -64),
                "l1": (-190, 68), "l2": (-58, 106), "l3": (132, 64),
            }
        edges = [
            ("s", "u1"), ("u1", "u2"), ("u2", "u3"), ("u3", "g"),
            ("s", "l1"), ("l1", "l2"), ("l2", "l3"), ("l3", "g"),
        ]
        marked = 1 if int(panel["mark_slot"]) == 0 else 5
        return nodes, edges, marked

    if int(panel["variant"]) == 0:
        nodes = {
            "s": (-260, 0), "g": (260, 0),
            "a": (-182, -88), "b": (-182, 88), "m": (-45, 0), "n": (45, 0),
            "c": (182, -88), "d": (182, 88),
        }
    else:
        nodes = {
            "s": (-260, 0), "g": (260, 0),
            "a": (-188, -68), "b": (-188, 68), "m": (-55, 0), "n": (55, 0),
            "c": (188, -68), "d": (188, 68),
        }
    # Two diamonds are joined by one central bridge.  Deleting that bridge
    # separates the terminals even though each side has a visible cycle.
    edges = [
        ("s", "a"), ("a", "m"), ("s", "b"), ("b", "m"), ("m", "n"),
        ("n", "c"), ("c", "g"), ("n", "d"), ("d", "g"),
    ]
    return nodes, edges, 4


def _panel_geometry(panel: dict, box: tuple[int, int, int, int]) -> tuple[dict, list[tuple[tuple[int, int], tuple[int, int]]], tuple[int, int]]:
    x0, y0, x1, y1 = box
    center = ((x0 + x1) / 2.0, (y0 + y1) / 2.0)
    nodes, edges, marked = _base_graph(panel)

    def transform(point: tuple[float, float]) -> tuple[int, int]:
        x, y = point
        if bool(panel["flip_x"]):
            x = -x
        if bool(panel["flip_y"]):
            y = -y
        x += float(panel["offset_x"])
        y += float(panel["offset_y"])
        return (round(center[0] + x), round(center[1] + y))

    points = {name: transform(point) for name, point in nodes.items()}
    segments = [(points[a], points[b]) for a, b in edges]
    ax, bx = segments[marked]
    marker = (round((ax[0] + bx[0]) / 2), round((ax[1] + bx[1]) / 2))
    return points, segments, marker


def sample_scene(seed: int) -> dict:
    """Sample one scene with one resilient panel, balanced over panel position."""
    rng = random.Random(int(seed) * 7919 + 13)
    winner = int(seed) % len(PANEL_BOXES)
    panels = []
    for index in range(len(PANEL_BOXES)):
        panels.append(
            {
                "kind": "resilient" if index == winner else "bottleneck",
                "variant": rng.randrange(2),
                "mark_slot": rng.randrange(2),
                "flip_x": bool(rng.randrange(2)),
                "flip_y": bool(rng.randrange(2)),
                "offset_x": rng.randint(-9, 9),
                "offset_y": rng.randint(-7, 7),
            }
        )
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((42, 25), "Edge-cut connectivity", fill=INK, font=_font(29))
    draw.text(
        (950, 31),
        "A      B      C      D",
        fill=FRAME,
        font=_font(21),
    )
    for index, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=16, fill=PANEL_FILL, outline=FRAME, width=3)
        draw.text((x0 + 21, y0 + 13), PANEL_LABELS[index][-1], fill=INK, font=_font(25))

        points, segments, marker = _panel_geometry(scene["panels"][index], box)
        for first, second in segments:
            draw.line((first, second), fill=INK, width=EDGE_WIDTH)
        for point in points.values():
            px, py = point
            draw.ellipse(
                (px - NODE_RADIUS, py - NODE_RADIUS, px + NODE_RADIUS, py + NODE_RADIUS),
                fill=INK,
            )

        mx, my = marker
        draw.ellipse(
            (mx - CUT_COLLAR_RADIUS, my - CUT_COLLAR_RADIUS,
             mx + CUT_COLLAR_RADIUS, my + CUT_COLLAR_RADIUS),
            outline=CYAN,
            width=CUT_COLLAR_WIDTH,
        )

        for name, point, color in (
            ("start", points["s"], GREEN),
            ("finish", points["g"], AMBER),
        ):
            px, py = point
            draw.ellipse(
                (px - TERMINAL_HALO, py - TERMINAL_HALO,
                 px + TERMINAL_HALO, py + TERMINAL_HALO),
                fill=PANEL_FILL,
            )
            draw.ellipse(
                (px - TERMINAL_RADIUS, py - TERMINAL_RADIUS,
                 px + TERMINAL_RADIUS, py + TERMINAL_RADIUS),
                fill=color,
                outline=PANEL_FILL,
                width=3,
            )
    return image


def analytic_gold(scene: dict) -> str:
    winners = [index for index, panel in enumerate(scene["panels"]) if panel["kind"] == "resilient"]
    if len(winners) != 1:
        raise ValueError(f"expected one resilient panel, got {winners}")
    return PANEL_LABELS[winners[0]]


def margin(scene: dict) -> float:
    """Minimum clearance between a cut disk and the nearest marked-edge node."""
    clearances = []
    for panel, box in zip(scene["panels"], PANEL_BOXES):
        _, segments, _ = _panel_geometry(panel, box)
        _, _, marked = _base_graph(panel)
        first, second = segments[marked]
        length = math.hypot(second[0] - first[0], second[1] - first[1])
        clearances.append(length / 2.0 - CUT_RADIUS)
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    try:
        winners = sum(panel["kind"] == "resilient" for panel in scene["panels"])
        return winners != 1 or margin(scene) < 12.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """All panel fields are rendered; this encoding has no latent aliases."""
    return []
