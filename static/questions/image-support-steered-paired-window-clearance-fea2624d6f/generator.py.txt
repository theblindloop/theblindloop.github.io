"""Renderer for the paired nested-window clearance question.

The visible scene contains two disconnected, equally sized colored plaques.  A
large background-colored aperture is cut into each plaque at a sampled vertical
offset.  The analytic label compares the two visible top-rim clearances.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "paired_window_clearance-0.1.0"
BACKGROUND = (247, 244, 235)
AMBER = (231, 153, 43)
TEAL = (30, 145, 145)
COLORS = {"amber": AMBER, "teal": TEAL}
QUARANTINE_PX = 12.0


def _beveled_rect(cx: float, cy: float, width: float, height: float, bevel: float) -> list[tuple[float, float]]:
    """Return a closed, gently beveled rectangle in clockwise order."""

    left = cx - width / 2.0
    right = cx + width / 2.0
    top = cy - height / 2.0
    bottom = cy + height / 2.0
    b = min(bevel, width / 3.0, height / 3.0)
    return [
        (left + b, top),
        (right - b, top),
        (right, top + b),
        (right, bottom - b),
        (right - b, bottom),
        (left + b, bottom),
        (left, bottom - b),
        (left, top + b),
    ]


def _scaled(points: list[tuple[float, float]], scale: int) -> list[tuple[float, float]]:
    return [(x * scale, y * scale) for x, y in points]


def _draw_plaque(
    draw: ImageDraw.ImageDraw,
    *,
    cx: float,
    cy: float,
    plaque_width: float,
    plaque_height: float,
    corner: float,
    hole_width: float,
    hole_height: float,
    clearance: float,
    color: tuple[int, int, int],
    scale: int,
) -> None:
    outer = _beveled_rect(cx, cy, plaque_width, plaque_height, corner)
    draw.polygon(_scaled(outer, scale), fill=color)

    outer_top = cy - plaque_height / 2.0
    hole_cy = outer_top + clearance + hole_height / 2.0
    hole_bevel = min(corner * 0.58, hole_width / 3.0, hole_height / 3.0)
    hole = _beveled_rect(cx, hole_cy, hole_width, hole_height, hole_bevel)
    draw.polygon(_scaled(hole, scale), fill=BACKGROUND)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, comfortably answerable scene from one integer seed."""

    rng = random.Random(int(seed))
    width, height = 560, 420
    plaque_width = rng.randint(90, 98)
    plaque_height = rng.randint(154, 166)
    corner = rng.randint(14, 20)
    hole_width = rng.randint(36, 44)
    hole_height = rng.randint(52, 62)
    left_cx = rng.randint(136, 144)
    separation = rng.randint(274, 282)
    right_cx = left_cx + separation
    center_y = rng.randint(202, 218)

    base = rng.uniform(39.0, 52.0)
    delta = rng.uniform(20.0, 34.0)
    if rng.random() < 0.5:
        amber_clearance = base - delta / 2.0
        teal_clearance = base + delta / 2.0
    else:
        amber_clearance = base + delta / 2.0
        teal_clearance = base - delta / 2.0

    left_color = "amber" if rng.random() < 0.5 else "teal"
    return {
        "width": width,
        "height": height,
        "left_cx": left_cx,
        "right_cx": right_cx,
        "center_y": center_y,
        "plaque_width": plaque_width,
        "plaque_height": plaque_height,
        "corner": corner,
        "hole_width": hole_width,
        "hole_height": hole_height,
        "amber_clearance": round(amber_clearance, 3),
        "teal_clearance": round(teal_clearance, 3),
        "left_color": left_color,
    }


def render(scene: dict) -> Image.Image:
    """Render a final RGB PNG-sized image using only the scene fields."""

    supersample = 4
    image = Image.new(
        "RGB",
        (int(scene["width"]) * supersample, int(scene["height"]) * supersample),
        BACKGROUND,
    )
    draw = ImageDraw.Draw(image)
    left_color = scene["left_color"]
    right_color = "teal" if left_color == "amber" else "amber"
    common = {
        "cy": float(scene["center_y"]),
        "plaque_width": float(scene["plaque_width"]),
        "plaque_height": float(scene["plaque_height"]),
        "corner": float(scene["corner"]),
        "hole_width": float(scene["hole_width"]),
        "hole_height": float(scene["hole_height"]),
        "scale": supersample,
    }
    _draw_plaque(
        draw,
        cx=float(scene["left_cx"]),
        clearance=float(scene["amber_clearance"] if left_color == "amber" else scene["teal_clearance"]),
        color=COLORS[left_color],
        **common,
    )
    _draw_plaque(
        draw,
        cx=float(scene["right_cx"]),
        clearance=float(scene["teal_clearance"] if left_color == "amber" else scene["amber_clearance"]),
        color=COLORS[right_color],
        **common,
    )
    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Select the color whose visible top clearance is smaller."""

    amber = float(scene["amber_clearance"])
    teal = float(scene["teal_clearance"])
    return "amber" if amber < teal else "teal"


def margin(scene: dict) -> float:
    """Return the runner-up gap between the two visible clearance values."""

    return abs(float(scene["amber_clearance"]) - float(scene["teal_clearance"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This minimal scene has no distinct pixel-identical latent transforms."""

    return []
