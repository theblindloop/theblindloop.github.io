"""Independent image-only inverse arm for graph endpoint parity."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

PALETTE = {
    "green": (25, 165, 95),
    "red": (220, 65, 65),
    "orange": (235, 140, 35),
    "purple": (145, 80, 190),
    "gray": (198, 203, 207),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            cy, cx = stack.pop()
            pts.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pts) >= 80:
            result.append(np.asarray(pts))
    return result


def _node_centers(arr: np.ndarray):
    centers = []
    for name, rgb in PALETTE.items():
        distance = np.linalg.norm(arr.astype(np.int16) - np.asarray(rgb), axis=2)
        for component in _components(distance < 42):
            y, x = component.mean(axis=0)
            centers.append((float(x), float(y), name))
    if len(centers) != 9:
        raise ValueError(f"expected 9 visible nodes, found {len(centers)}")
    return centers


def _has_intermediate(a, b, nodes) -> bool:
    ax, ay, _ = a
    bx, by, _ = b
    vx, vy = bx - ax, by - ay
    length2 = vx * vx + vy * vy
    for node in nodes:
        if node is a or node is b:
            continue
        px, py, _ = node
        t = ((px - ax) * vx + (py - ay) * vy) / length2
        if 0.08 < t < 0.92:
            d = math.hypot(px - (ax + t * vx), py - (ay + t * vy))
            if d < 19.0:
                return True
    return False


def _linked(a, b, nodes, dark: np.ndarray) -> bool:
    if _has_intermediate(a, b, nodes):
        return False
    ax, ay, _ = a
    bx, by, _ = b
    distance = math.hypot(bx - ax, by - ay)
    if distance > 180:
        return False
    samples = max(24, int(distance))
    hits = 0
    total = 0
    for t in np.linspace(0.13, 0.87, samples):
        x = int(round(ax + t * (bx - ax)))
        y = int(round(ay + t * (by - ay)))
        patch = dark[max(0, y - 2): y + 3, max(0, x - 2): x + 3]
        hits += bool(patch.any())
        total += 1
    return hits / total > 0.94


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    nodes = _node_centers(arr)
    dark = (arr[:, :, 0] < 90) & (arr[:, :, 1] < 90) & (arr[:, :, 2] < 100)
    degrees = [0] * len(nodes)
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            if _linked(nodes[i], nodes[j], nodes, dark):
                degrees[i] += 1
                degrees[j] += 1
    green = [i for i, node in enumerate(nodes) if node[2] == "green"]
    if len(green) != 1 or degrees[green[0]] % 2 != 1:
        raise ValueError("green start is not a unique visible odd-degree node")
    choices = [
        node[2] for i, node in enumerate(nodes)
        if node[2] in {"red", "orange", "purple"} and degrees[i] % 2 == 1
    ]
    if len(choices) != 1:
        raise ValueError(f"expected one colored odd endpoint, found {choices}")
    return choices[0]
