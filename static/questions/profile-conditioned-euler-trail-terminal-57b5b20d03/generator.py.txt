"""Renderer and analytic arm for the every-link trail endpoint world."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

SIZE = 512
SS = 3
BG = (248, 247, 242)
DARK = (35, 40, 48)
GRAY = (198, 203, 207)
COLORS = {
    "green": (25, 165, 95),
    "red": (220, 65, 65),
    "orange": (235, 140, 35),
    "purple": (145, 80, 190),
}


def _transform(points: list[list[int]], turn: int, flip: bool) -> list[list[int]]:
    result = []
    for x, y in points:
        x, y = x - 256, y - 256
        if flip:
            y = -y
        for _ in range(turn):
            x, y = -y, x
        result.append([x + 256, y + 256])
    return result


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    # A path from node 0 to node 4 with one triangular cycle attached at
    # node 1 and another at node 3. Only 0 and 4 have odd degree.
    base_nodes = [
        [52, 256], [140, 256], [230, 256], [320, 256], [460, 256],
        [150, 120], [230, 120], [330, 390], [410, 390],
    ]
    edges = [
        [0, 1], [1, 2], [2, 3], [3, 4],
        [1, 5], [5, 6], [6, 1],
        [3, 7], [7, 8], [8, 3],
    ]
    nodes = _transform(base_nodes, rng.randrange(4), bool(rng.randrange(2)))
    names = ["red", "orange", "purple"]
    answer = names[seed % 3]
    others = [name for name in names if name != answer]
    rng.shuffle(others)
    # Terminal colors are visible bindings: node 4 is the odd endpoint; nodes
    # 5 and 7 are even-degree distractors on cycles.
    terminal_colors = {"4": answer, "5": others[0], "7": others[1]}
    return {"nodes": nodes, "edges": edges, "terminal_colors": terminal_colors}


def _degrees(scene: dict) -> list[int]:
    edges = scene["edges"]
    terminal_colors = scene["terminal_colors"]
    # Access terminal_colors here as an explicit validation of label bindings.
    if set(terminal_colors.values()) != {"red", "orange", "purple"}:
        raise ValueError("terminal colors must be a permutation")
    degrees = [0] * 9
    for a, b in edges:
        degrees[a] += 1
        degrees[b] += 1
    return degrees


def analytic_gold(scene: dict) -> str:
    degrees = _degrees(scene)
    terminal_colors = scene["terminal_colors"]
    odd_colored = [terminal_colors[str(i)] for i in map(int, terminal_colors) if degrees[i] % 2]
    if len(odd_colored) != 1 or degrees[0] % 2 != 1:
        raise ValueError("scene does not have the declared unique Euler endpoint")
    return odd_colored[0]


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    terminal_colors = scene["terminal_colors"]
    image = Image.new("RGB", (SIZE * SS, SIZE * SS), BG)
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        p = tuple(v * SS for v in nodes[a])
        q = tuple(v * SS for v in nodes[b])
        draw.line([p, q], fill=DARK, width=8 * SS)
    for i, (x, y) in enumerate(nodes):
        color = COLORS["green"] if i == 0 else COLORS.get(terminal_colors.get(str(i)), GRAY)
        r = 14 * SS
        box = ((x * SS - r, y * SS - r), (x * SS + r, y * SS + r))
        draw.ellipse(box, fill=color, outline=DARK, width=3 * SS)
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def _point_segment_distance(p, a, b) -> float:
    px, py = p
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    t = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / (vx * vx + vy * vy)))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def margin(scene: dict) -> float:
    nodes = scene["nodes"]
    edges = scene["edges"]
    edge_set = {tuple(sorted(edge)) for edge in edges}
    values = [math.dist(nodes[a], nodes[b]) - 34.0 for a in range(9) for b in range(a + 1, 9)]
    for a, b in edges:
        for i, point in enumerate(nodes):
            if i not in (a, b):
                values.append(_point_segment_distance(point, nodes[a], nodes[b]) - 18.0)
    # Separation from a hypothetical false direct link through an intermediate node.
    for a in range(9):
        for b in range(a + 1, 9):
            if (a, b) not in edge_set:
                values.append(math.dist(nodes[a], nodes[b]) - 34.0)
    return round(min(values), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
