"""Independent pixel-only inverse arm for Bendprint Correspondence."""

from __future__ import annotations

import numpy as np
from PIL import Image

REFERENCE = (28, 31, 36)
PALETTE = {
    "amber": (224, 137, 26),
    "cyan": (20, 151, 181),
    "violet": (139, 82, 190),
}


class OracleAbstention(ValueError):
    """The raster does not expose a unique, intact correspondence."""


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2)


def _component_count(mask: np.ndarray) -> int:
    points = {tuple(int(value) for value in point) for point in np.argwhere(mask)}
    count = 0
    while points:
        count += 1
        stack = [points.pop()]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in points:
                    points.remove(neighbor)
                    stack.append(neighbor)
    return count


def _trim(mask: np.ndarray) -> np.ndarray:
    ys, xs = np.where(mask)
    if not len(xs):
        raise OracleAbstention("a required colored path is absent")
    return mask[ys.min() : ys.max() + 1, xs.min() : xs.max() + 1]


def _variants(mask: np.ndarray) -> list[np.ndarray]:
    output = []
    for reflected in (False, True):
        base = np.fliplr(mask) if reflected else mask
        for turns in range(4):
            value = _trim(np.rot90(base, turns))
            if not any(value.shape == old.shape and np.array_equal(value, old) for old in output):
                output.append(value)
    return output


def _center_pad(mask: np.ndarray, height: int, width: int) -> np.ndarray:
    output = np.zeros((height, width), dtype=bool)
    y = (height - mask.shape[0]) // 2
    x = (width - mask.shape[1]) // 2
    output[y : y + mask.shape[0], x : x + mask.shape[1]] = mask
    return output


def _iou(left: np.ndarray, right: np.ndarray) -> float:
    height = max(left.shape[0], right.shape[0])
    width = max(left.shape[1], right.shape[1])
    a = _center_pad(left, height, width)
    b = _center_pad(right, height, width)
    union = int(np.logical_or(a, b).sum())
    return float(np.logical_and(a, b).sum() / union) if union else 0.0


def _similarity(reference: np.ndarray, candidate: np.ndarray) -> float:
    return max(_iou(reference, variant) for variant in _variants(candidate))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    masks = {"reference": _mask(array, REFERENCE)}
    masks.update({name: _mask(array, rgb) for name, rgb in PALETTE.items()})

    counts = []
    cropped = {}
    for name, mask in masks.items():
        count = int(mask.sum())
        if count < 50:
            raise OracleAbstention(f"{name} path is missing or materially erased")
        if _component_count(mask) != 1:
            raise OracleAbstention(f"{name} path is not one connected trace")
        counts.append(count)
        cropped[name] = _trim(mask)
    if max(counts) - min(counts) > 3:
        raise OracleAbstention("equal-ink path counterbalance was corrupted")

    reference = cropped.pop("reference")
    scores = sorted(
        ((_similarity(reference, mask), name) for name, mask in cropped.items()),
        reverse=True,
    )
    best_score, best_name = scores[0]
    runner_score = scores[1][0]
    if best_score < 0.985:
        raise OracleAbstention(f"no exact transformed path match (score={best_score:.3f})")
    if best_score - runner_score < 0.08:
        raise OracleAbstention("best and runner-up correspondences are ambiguous")
    return best_name
