"""Renderer and analytic arm for cyclic tile-order correspondence.

The visual decision is whether four individually patterned tiles retain their
cyclic order after being rearranged from a row into a ring.  All tile patterns
have equal ink count, and all tiles in a scene share one palette, so neither
colour nor total ink identifies the answer.
"""

from __future__ import annotations

import itertools
import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 3

PALETTES = {
    "ocean": ((205, 235, 241), (31, 105, 122)),
    "sand": ((244, 226, 178), (142, 94, 25)),
    "lavender": ((226, 211, 240), (104, 67, 137)),
}


def _hamming(left: str, right: str) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def _build_codebook() -> tuple[str, ...]:
    """A deterministic constant-weight code with pairwise distance >= 4."""

    codes: list[str] = []
    for occupied in itertools.combinations(range(9), 4):
        code = "".join("1" if index in occupied else "0" for index in range(9))
        if all(_hamming(code, prior) >= 4 for prior in codes):
            codes.append(code)
    return tuple(codes)


CODEBOOK = _build_codebook()


def _cyclic_equal(left: list[str], right: list[str]) -> bool:
    if len(left) != len(right):
        return False
    doubled = left + left
    return any(doubled[offset : offset + len(left)] == right for offset in range(len(left)))


def _assignment_gap(top_codes: list[str], bottom_codes: list[str]) -> float:
    scores = []
    for assignment in itertools.permutations(range(4)):
        score = sum(
            _hamming(bottom_codes[bottom_index], top_codes[top_index])
            for bottom_index, top_index in enumerate(assignment)
        )
        scores.append(score)
    scores.sort()
    return float(scores[1] - scores[0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    chosen = rng.sample(CODEBOOK, 4)
    rng.shuffle(chosen)
    top_codes = list(chosen)

    positive = int(seed) % 2 == 0
    if positive:
        shift = rng.randrange(4)
        bottom_codes = top_codes[shift:] + top_codes[:shift]
    else:
        alternatives = [
            list(order)
            for order in itertools.permutations(top_codes)
            if not _cyclic_equal(top_codes, list(order))
        ]
        bottom_codes = rng.choice(alternatives)

    top_y = rng.randint(125, 135)
    top_centers = [
        [74 + 121 * index + rng.randint(-5, 5), top_y + rng.randint(-3, 3)]
        for index in range(4)
    ]
    ring_center = [256 + rng.randint(-5, 5), 369 + rng.randint(-4, 4)]
    ring_radius = rng.randint(82, 87)
    base_angle = rng.uniform(0.0, 2.0 * math.pi)
    bottom_angles = [
        base_angle + index * math.pi / 2.0 + rng.uniform(-0.045, 0.045)
        for index in range(4)
    ]

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "top_codes": top_codes,
        "bottom_codes": bottom_codes,
        "top_centers": top_centers,
        "top_sizes": [rng.randint(58, 66) for _ in range(4)],
        "ring_center": ring_center,
        "ring_radius": ring_radius,
        "bottom_angles": bottom_angles,
        "bottom_sizes": [rng.randint(58, 66) for _ in range(4)],
        "palette": rng.choice(sorted(PALETTES)),
    }


def analytic_gold(scene: dict) -> str:
    return "yes" if _cyclic_equal(scene["top_codes"], scene["bottom_codes"]) else "no"


def margin(scene: dict) -> float:
    """Runner-up assignment cost gap, measured in mismatched grid cells."""

    return _assignment_gap(scene["top_codes"], scene["bottom_codes"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-index the ring records without changing any code/position pairing."""

    twin = dict(scene)
    for field in ("bottom_codes", "bottom_angles", "bottom_sizes"):
        values = list(scene[field])
        twin[field] = values[1:] + values[:1]
    return [("cyclic_reindex_ring_records", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    filename = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / filename
    try:
        return ImageFont.truetype(str(path), size=size)
    except OSError:
        return ImageFont.load_default()


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    size: float,
    code: str,
    fill: tuple[int, int, int],
    border: tuple[int, int, int],
    scale: int,
) -> None:
    cx, cy = center
    half = size / 2.0
    box = tuple(int(round(value * scale)) for value in (cx - half, cy - half, cx + half, cy + half))
    draw.rounded_rectangle(
        box,
        radius=int(round(size * 0.16 * scale)),
        fill=fill,
        outline=border,
        width=3 * scale,
    )
    cell_side = size * 0.125
    step = size * 0.185
    ink = (25, 30, 37)
    for index, bit in enumerate(code):
        if bit != "1":
            continue
        row, col = divmod(index, 3)
        gx = cx + (col - 1) * step
        gy = cy + (row - 1) * step
        cell_box = tuple(
            int(round(value * scale))
            for value in (
                gx - cell_side / 2,
                gy - cell_side / 2,
                gx + cell_side / 2,
                gy + cell_side / 2,
            )
        )
        draw.rounded_rectangle(cell_box, radius=max(1, scale), fill=ink)


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), (250, 249, 247))
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(int(round(value * scale)) for value in values)

    draw.rounded_rectangle(
        box((17, 45, width - 17, 207)),
        radius=16 * scale,
        fill=(255, 255, 255),
        outline=(203, 207, 210),
        width=2 * scale,
    )
    draw.rounded_rectangle(
        box((17, 218, width - 17, height - 13)),
        radius=16 * scale,
        fill=(255, 255, 255),
        outline=(203, 207, 210),
        width=2 * scale,
    )

    heading = _font(13 * scale, bold=True)
    draw.text((29 * scale, 58 * scale), "REFERENCE CYCLE  ·  LEFT TO RIGHT", fill=(70, 75, 81), font=heading)
    draw.text((29 * scale, 230 * scale), "TEST RING  ·  CLOCKWISE", fill=(70, 75, 81), font=heading)

    # A neutral direction guide; it is intentionally not part of tile segmentation.
    cx, cy = scene["ring_center"]
    guide_r = 38
    draw.arc(
        box((cx - guide_r, cy - guide_r, cx + guide_r, cy + guide_r)),
        start=205,
        end=505,
        fill=(118, 123, 128),
        width=3 * scale,
    )
    theta = math.radians(505)
    tip = (cx + guide_r * math.cos(theta), cy + guide_r * math.sin(theta))
    tangent = theta + math.pi / 2
    points = [
        (tip[0], tip[1]),
        (tip[0] - 10 * math.cos(tangent - 0.55), tip[1] - 10 * math.sin(tangent - 0.55)),
        (tip[0] - 10 * math.cos(tangent + 0.55), tip[1] - 10 * math.sin(tangent + 0.55)),
    ]
    draw.polygon([(int(x * scale), int(y * scale)) for x, y in points], fill=(118, 123, 128))

    fill, border = PALETTES[scene["palette"]]
    for code, center, size in zip(
        scene["top_codes"], scene["top_centers"], scene["top_sizes"], strict=True
    ):
        _draw_tile(draw, tuple(center), float(size), code, fill, border, scale)

    rcx, rcy = scene["ring_center"]
    for code, angle, size in zip(
        scene["bottom_codes"], scene["bottom_angles"], scene["bottom_sizes"], strict=True
    ):
        center = (
            rcx + scene["ring_radius"] * math.cos(float(angle)),
            rcy + scene["ring_radius"] * math.sin(float(angle)),
        )
        _draw_tile(draw, center, float(size), code, fill, border, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)
