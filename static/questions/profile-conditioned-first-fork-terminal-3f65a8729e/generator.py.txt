"""Deterministic scene sampler and RGB renderer for the first-fork world."""

from __future__ import annotations

import heapq
import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 720
SCALE = 3
CENTER = (480.0, 360.0)
ROUTE = (39, 53, 70)
BACKGROUND = (248, 247, 243)
PALE = (192, 200, 207)
PALETTE = {
    "green": (43, 151, 91),
    "red": (211, 69, 70),
    "blue": (53, 103, 193),
    "gold": (215, 153, 35),
}


def _transform(point: tuple[float, float], scale: float, angle: float, mirror: bool, tx: float, ty: float) -> list[float]:
    x = (point[0] - CENTER[0]) * scale
    y = (point[1] - CENTER[1]) * scale
    if mirror:
        y = -y
    c = math.cos(angle)
    s = math.sin(angle)
    return [round(CENTER[0] + c * x - s * y + tx, 3), round(CENTER[1] + s * x + c * y + ty, 3)]


def _polyline_length(points: list[list[float]]) -> float:
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:]))


def _adjacency(scene: dict[str, Any]) -> dict[str, list[tuple[str, float]]]:
    graph: dict[str, list[tuple[str, float]]] = {}
    for edge in scene["edges"]:
        u, v = edge["u"], edge["v"]
        length = _polyline_length(edge["points"])
        graph.setdefault(u, []).append((v, length))
        graph.setdefault(v, []).append((u, length))
    return graph


def _distances(scene: dict[str, Any]) -> dict[str, float]:
    graph = _adjacency(scene)
    distances = {"source": 0.0}
    queue: list[tuple[float, str]] = [(0.0, "source")]
    while queue:
        distance, node = heapq.heappop(queue)
        if distance != distances.get(node):
            continue
        for neighbor, weight in graph.get(node, []):
            candidate = distance + weight
            if candidate < distances.get(neighbor, float("inf")):
                distances[neighbor] = candidate
                heapq.heappush(queue, (candidate, neighbor))
    return distances


def _component(graph: dict[str, list[tuple[str, float]]], start: str, removed: str) -> set[str]:
    seen: set[str] = set()
    stack = [start]
    while stack:
        node = stack.pop()
        if node == removed or node in seen:
            continue
        seen.add(node)
        stack.extend(neighbor for neighbor, _ in graph.get(node, []) if neighbor not in seen and neighbor != removed)
    return seen


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    scale = 0.86 + 0.07 * rng.random()
    angle = (int(seed) % 4) * math.pi / 2.0 + rng.choice((-0.035, 0.0, 0.035))
    mirror = bool((int(seed) // 4) % 2)
    tx = rng.uniform(-24.0, 24.0)
    ty = rng.uniform(-18.0, 18.0)

    base = {
        "source": (125.0, 360.0),
        "fork1": (350.0, 360.0),
        "fork2": (590.0, 360.0),
        "single": (505.0, 145.0),
        "upper": (820.0, 165.0),
        "lower": (820.0, 555.0),
        "single_elbow": (408.0, 242.0),
        "upper_elbow": (696.0, 212.0),
        "lower_elbow": (696.0, 508.0),
    }
    points = {
        key: _transform(value, scale, angle, mirror, tx, ty) for key, value in base.items()
    }

    colors = ["red", "blue", "gold"]
    single_color = colors[int(seed) % 3]
    remainder = [color for color in colors if color != single_color]
    if int(seed) % 2:
        remainder.reverse()
    terminal_colors = {"single": single_color, "upper": remainder[0], "lower": remainder[1]}

    edges = [
        {"u": "source", "v": "fork1", "points": [points["source"], points["fork1"]]},
        {"u": "fork1", "v": "single", "points": [points["fork1"], points["single_elbow"], points["single"]]},
        {"u": "fork1", "v": "fork2", "points": [points["fork1"], points["fork2"]]},
        {"u": "fork2", "v": "upper", "points": [points["fork2"], points["upper_elbow"], points["upper"]]},
        {"u": "fork2", "v": "lower", "points": [points["fork2"], points["lower_elbow"], points["lower"]]},
    ]

    distractor_bases = [
        ((150.0, 110.0), (280.0, 95.0)),
        ((695.0, 92.0), (875.0, 112.0)),
        ((150.0, 620.0), (290.0, 600.0)),
        ((690.0, 640.0), (870.0, 615.0)),
        ((75.0, 205.0), (155.0, 175.0)),
    ]
    distractors = [
        [_transform(a, scale, angle, mirror, tx, ty), _transform(b, scale, angle, mirror, tx, ty)]
        for a, b in distractor_bases
    ]

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "route_color": list(ROUTE),
        "route_width": 11,
        "terminal_radius": 24,
        "source": {"id": "source", "center": points["source"], "color": "green"},
        "forks": [
            {"id": "fork1", "center": points["fork1"]},
            {"id": "fork2", "center": points["fork2"]},
        ],
        "terminals": [
            {"id": "single", "center": points["single"], "color": terminal_colors["single"]},
            {"id": "upper", "center": points["upper"], "color": terminal_colors["upper"]},
            {"id": "lower", "center": points["lower"], "color": terminal_colors["lower"]},
        ],
        "edges": edges,
        "distractors": distractors,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    factor = SCALE
    size = (int(scene["width"]) * factor, int(scene["height"]) * factor)
    image = Image.new("RGB", size, tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    def xy(point: list[float]) -> tuple[int, int]:
        return (round(point[0] * factor), round(point[1] * factor))

    route_color = tuple(scene["route_color"])
    width = int(scene["route_width"]) * factor
    for distractor in scene["distractors"]:
        draw.line([xy(point) for point in distractor], fill=PALE, width=max(2, factor * 2))
    for edge in scene["edges"]:
        draw.line([xy(point) for point in edge["points"]], fill=route_color, width=width, joint="curve")

    # Touching the hubs with the same ink makes the intended graph boundary explicit
    # without introducing a new marker symbol.
    for fork in scene["forks"]:
        center = xy(fork["center"])
        draw.ellipse((center[0] - factor * 3, center[1] - factor * 3, center[0] + factor * 3, center[1] + factor * 3), fill=route_color)

    radius = int(scene["terminal_radius"]) * factor
    for marker in [scene["source"], *scene["terminals"]]:
        center = xy(marker["center"])
        halo = radius + factor * 7
        draw.ellipse((center[0] - halo, center[1] - halo, center[0] + halo, center[1] + halo), fill=tuple(scene["background"]))
        color = PALETTE[marker["color"]]
        draw.ellipse((center[0] - radius, center[1] - radius, center[0] + radius, center[1] + radius), fill=color)

    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    graph = _adjacency(scene)
    distances = _distances(scene)
    fork_ids = [fork["id"] for fork in scene["forks"]]
    first_fork = min(fork_ids, key=lambda node: distances.get(node, float("inf")))
    terminal_colors = {terminal["id"]: terminal["color"] for terminal in scene["terminals"]}
    for neighbor, _ in graph.get(first_fork, []):
        component = _component(graph, neighbor, first_fork)
        terminal_ids = [node for node in component if node in terminal_colors]
        if len(terminal_ids) == 1:
            return terminal_colors[terminal_ids[0]]
    return "unknown"


def margin(scene: dict[str, Any]) -> float:
    distances = _distances(scene)
    fork_distances = sorted(distances.get(fork["id"], 0.0) for fork in scene["forks"])
    return round(fork_distances[1] - fork_distances[0], 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    if len(scene.get("forks", [])) != 2 or len(scene.get("terminals", [])) != 3 or len(scene.get("edges", [])) != 5:
        return True
    if margin(scene) < 80.0:
        return True
    centers = [scene["source"]["center"]] + [terminal["center"] for terminal in scene["terminals"]]
    for i, first in enumerate(centers):
        for second in centers[i + 1 :]:
            if math.dist(first, second) < 55.0:
                return True
    return analytic_gold(scene) == "unknown"


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
