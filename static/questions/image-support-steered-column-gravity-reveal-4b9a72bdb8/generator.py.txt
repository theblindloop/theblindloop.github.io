"""Deterministic latent sampler and Pillow renderer for a gravity card."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = 512
GRID = 9
PAD = 12
FRAME_WIDTH = 5
BACKGROUND = (250, 249, 245)
PLATE = (222, 239, 234)
FRAME = (24, 169, 181)
MAGENTA = (205, 45, 139)
GOLD = (232, 174, 35)
GRID_LINE = (174, 207, 202)
INKS = ((36, 48, 72), (76, 43, 91), (34, 78, 61))
DISTRACTOR_COLORS = ((187, 196, 197), (199, 188, 179), (181, 196, 181))

NAMES = ("crown", "cup", "stair")
TEMPLATES = {
    "crown": (3, 5, 7, 5, 3, 5, 7, 5, 3),
    "cup": (7, 6, 4, 3, 3, 3, 4, 6, 7),
    "stair": (1, 2, 3, 4, 5, 6, 7, 7, 8),
}


def _column_heights(cells: list[list[int]]) -> tuple[int, ...]:
    if len(cells) != GRID or any(len(row) != GRID for row in cells):
        raise ValueError("cells must be a 9 by 9 matrix")
    return tuple(sum(int(bool(cells[row][col])) for row in range(GRID)) for col in range(GRID))


def _rank(heights: tuple[int, ...]) -> list[tuple[int, str]]:
    return sorted(
        (sum(abs(value - target) for value, target in zip(heights, template)), name)
        for name, template in TEMPLATES.items()
    )


def _profile_for(seed: int, name: str, difficulty: int) -> tuple[int, ...]:
    rng = random.Random((seed * 0x9E3779B1) ^ 0xA17C93D5)
    for _attempt in range(6000):
        heights = list(TEMPLATES[name])
        steps = (
            rng.randint(0, 1)
            if difficulty == 0
            else rng.randint(3, 5)
            if difficulty == 1
            else rng.randint(6, 11)
        )
        for _ in range(steps):
            donors = [index for index, value in enumerate(heights) if value > 1]
            recipients = [index for index, value in enumerate(heights) if value < 8]
            donor = rng.choice(donors)
            recipient = rng.choice(recipients)
            if donor != recipient:
                heights[donor] -= 1
                heights[recipient] += 1
        ranked = _rank(tuple(heights))
        gap = ranked[1][0] - ranked[0][0]
        accepted = (
            ranked[0][1] == name
            and (
                (difficulty == 0 and gap >= 14)
                or (difficulty == 1 and 6 <= gap <= 12)
                or (difficulty == 2 and gap == 2)
            )
        )
        if accepted:
            return tuple(heights)
    raise RuntimeError("could not synthesize a calibrated gravity profile")


def sample_scene(seed: int) -> dict:
    """Sample balanced classes with independently scrambled vertical positions."""

    name = NAMES[seed % len(NAMES)]
    difficulty = (seed // len(NAMES)) % 3
    heights = _profile_for(seed, name, difficulty)
    rng = random.Random((seed * 0x85EBCA6B) ^ 0xC2B2AE35)
    cells = [[0 for _ in range(GRID)] for _ in range(GRID)]
    for col, height in enumerate(heights):
        for row in rng.sample(range(GRID), height):
            cells[row][col] = 1

    anchors = ((53, 70), (82, 430), (430, 76), (426, 420), (61, 260), (447, 257))
    distractors = []
    for index, (x, y) in enumerate(anchors):
        distractors.append(
            {
                "x": x + rng.randint(-5, 5),
                "y": y + rng.randint(-5, 5),
                "size": 5 + 2 * rng.randrange(3),
                "color": (index + rng.randrange(3)) % len(DISTRACTOR_COLORS),
            }
        )
    return {
        "cells": cells,
        "rotation": rng.randrange(4),
        "cell_size": (18, 20)[rng.randrange(2)],
        "offset_x": rng.randint(-5, 5),
        "offset_y": rng.randint(-5, 5),
        "palette": rng.randrange(len(INKS)),
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    rotation = int(scene["rotation"]) % 4
    cell_size = int(scene["cell_size"])
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    ink = INKS[int(scene["palette"]) % len(INKS)]
    distractors = scene["distractors"]
    if cell_size not in (18, 20):
        raise ValueError("unsupported cell size")

    board_size = 2 * PAD + GRID * cell_size
    board = Image.new("RGB", (board_size, board_size), PLATE)
    draw = ImageDraw.Draw(board)
    draw.rectangle((0, 0, board_size - 1, board_size - 1), outline=FRAME, width=FRAME_WIDTH)
    for index in range(GRID + 1):
        coordinate = PAD + index * cell_size
        draw.line((PAD, coordinate, board_size - PAD, coordinate), fill=GRID_LINE, width=1)
        draw.line((coordinate, PAD, coordinate, board_size - PAD), fill=GRID_LINE, width=1)
    for row in range(GRID):
        for col in range(GRID):
            if cells[row][col]:
                x0 = PAD + col * cell_size + 3
                y0 = PAD + row * cell_size + 3
                x1 = PAD + (col + 1) * cell_size - 3
                y1 = PAD + (row + 1) * cell_size - 3
                draw.rectangle((x0, y0, x1, y1), fill=ink)
    draw.rectangle((6, 6, 10, 10), fill=MAGENTA)
    draw.rectangle((board_size - 11, 6, board_size - 7, 10), fill=GOLD)

    for _ in range(rotation):
        board = board.transpose(Image.Transpose.ROTATE_270)

    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    image_draw = ImageDraw.Draw(image)
    for mark in distractors:
        color = DISTRACTOR_COLORS[int(mark["color"]) % len(DISTRACTOR_COLORS)]
        radius = int(mark["size"])
        x, y = int(mark["x"]), int(mark["y"])
        image_draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=2)
    x0 = (CANVAS - board_size) // 2 + offset_x
    y0 = (CANVAS - board_size) // 2 + offset_y
    image.paste(board, (x0, y0))
    return image


def analytic_gold(scene: dict) -> str:
    ranked = _rank(_column_heights(scene["cells"]))
    return "ambiguous" if ranked[0][0] == ranked[1][0] else ranked[0][1]


def margin(scene: dict) -> float:
    ranked = _rank(_column_heights(scene["cells"]))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_disjoint_distractor_draw_order", twin)]
