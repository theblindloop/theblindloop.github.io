"""Independent PNG-only inverse for the declared column-gravity operation."""

from __future__ import annotations

import numpy as np
from PIL import Image

FRAME_RGB = np.array((24, 169, 181), dtype=np.int16)
MAGENTA_RGB = np.array((205, 45, 139), dtype=np.int16)
GOLD_RGB = np.array((232, 174, 35), dtype=np.int16)
GRID_RGB = np.array((174, 207, 202), dtype=np.int16)
PLATE_RGB = np.array((222, 239, 234), dtype=np.float64)

SHAPES = {
    "crown": (3, 5, 7, 5, 3, 5, 7, 5, 3),
    "cup": (7, 6, 4, 3, 3, 3, 4, 6, 7),
    "stair": (1, 2, 3, 4, 5, 6, 7, 7, 8),
}


def _near(array: np.ndarray, color: np.ndarray, tolerance: float) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2, dtype=np.int64)) <= tolerance


def _center(mask: np.ndarray, name: str) -> tuple[float, float]:
    points = np.argwhere(mask)
    if len(points) < 8:
        raise ValueError(f"missing {name} orientation fiducial")
    return float(points[:, 1].mean()), float(points[:, 0].mean())


def _line_coordinates(mask: np.ndarray, axis: int, board_size: int) -> list[int]:
    counts = mask.sum(axis=axis)
    raw = np.flatnonzero(counts >= board_size * 0.24).tolist()
    groups: list[list[int]] = []
    for coordinate in raw:
        if groups and coordinate <= groups[-1][-1] + 1:
            groups[-1].append(coordinate)
        else:
            groups.append([coordinate])
    centers = [int(round(sum(group) / len(group))) for group in groups]
    if len(centers) != 10:
        raise ValueError("the visible 9 by 9 grid is incomplete")
    gaps = np.diff(centers)
    if gaps.min() < 14 or gaps.max() - gaps.min() > 2:
        raise ValueError("the visible grid spacing is inconsistent")
    return centers


def _rectified_cells(image: Image.Image) -> list[list[int]]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    frame = _near(array, FRAME_RGB, 12.0)
    points = np.argwhere(frame)
    if len(points) < 250:
        raise ValueError("cyan card frame is missing")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width, height = int(x1 - x0 + 1), int(y1 - y0 + 1)
    if abs(width - height) > 1 or not 180 <= width <= 210:
        raise ValueError("cyan card frame has invalid bounds")
    crop = image.convert("RGB").crop((int(x0), int(y0), int(x1) + 1, int(y1) + 1))
    card = np.asarray(crop, dtype=np.uint8)
    magenta_center = _center(_near(card, MAGENTA_RGB, 14.0), "magenta")
    gold_center = _center(_near(card, GOLD_RGB, 14.0), "gold")
    dx = gold_center[0] - magenta_center[0]
    dy = gold_center[1] - magenta_center[1]
    if abs(dx) >= abs(dy) and dx > 0:
        quarter_turns = 0
    elif abs(dy) > abs(dx) and dy > 0:
        quarter_turns = 1
    elif abs(dx) >= abs(dy) and dx < 0:
        quarter_turns = 2
    elif abs(dy) > abs(dx) and dy < 0:
        quarter_turns = 3
    else:
        raise ValueError("orientation fiducials are ambiguous")
    for _ in range(quarter_turns):
        crop = crop.transpose(Image.Transpose.ROTATE_90)
    card = np.asarray(crop, dtype=np.uint8)
    size = card.shape[0]
    magenta_center = _center(_near(card, MAGENTA_RGB, 14.0), "magenta")
    gold_center = _center(_near(card, GOLD_RGB, 14.0), "gold")
    if not (
        magenta_center[0] < size * 0.25
        and magenta_center[1] < size * 0.25
        and gold_center[0] > size * 0.75
        and gold_center[1] < size * 0.25
    ):
        raise ValueError("card could not be rectified from its fiducials")

    grid_mask = _near(card, GRID_RGB, 14.0)
    xs = _line_coordinates(grid_mask, axis=0, board_size=size)
    ys = _line_coordinates(grid_mask, axis=1, board_size=size)
    cells: list[list[int]] = []
    for row in range(9):
        decoded_row = []
        for col in range(9):
            cx = (xs[col] + xs[col + 1]) // 2
            cy = (ys[row] + ys[row + 1]) // 2
            radius = max(3, min(xs[col + 1] - xs[col], ys[row + 1] - ys[row]) // 5)
            patch = card[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1]
            mean = patch.astype(np.float64).mean(axis=(0, 1))
            if float(mean.mean()) < 145.0:
                decoded_row.append(1)
            elif float(np.linalg.norm(mean - PLATE_RGB)) <= 12.0:
                decoded_row.append(0)
            else:
                raise ValueError("a visible grid cell is erased or unreadable")
        cells.append(decoded_row)
    return cells


def decision_from_image(image: Image.Image) -> str:
    cells = _rectified_cells(image)
    heights = tuple(sum(cells[row][col] for row in range(9)) for col in range(9))
    ranked = sorted(
        (sum(abs(value - target) for value, target in zip(heights, shape)), name)
        for name, shape in SHAPES.items()
    )
    return "ambiguous" if ranked[0][0] == ranked[1][0] else ranked[0][1]
