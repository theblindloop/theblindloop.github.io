"""Independent pixel-only inverse arm for the partial-sum-ratio world.

Recovers the decision (which candidate reproduces the reference's composed
partial-sum ratio?) from the final rasterized PNG alone. It detects the
engineered three-section bars by their exact indigo, turquoise, and amber
colours, measures, within each panel, the widths of the indigo (left) section
and the turquoise (middle) section in columns plus the total bar width, forms
the composed ratio Q = (left + middle)/(middle + right) where right is the
residual amber width, and returns the label of the candidate whose measured Q
matches the reference's.

The three panels occupy fixed rows (renderer geometry), so the arm reads the
decision from the image pixels alone and never receives the latent scene. No
renderer / prompts / generate / verify / analytic-gold code is imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

INDIGO = (63, 72, 204)
TURQUOISE = (0, 188, 168)
AMBER = (255, 176, 32)
BAR_HEIGHT = 38
PANEL_TOPS = {"reference": 150, "A": 320, "B": 460}


def _composed_q_in_band(arr: np.ndarray, top: int) -> tuple[float, bool]:
    """Composed partial-sum ratio of the three-section bar within a row band."""
    band = arr[top : top + BAR_HEIGHT]
    indigo = (band == INDIGO).all(axis=-1)
    turquoise = (band == TURQUOISE).all(axis=-1)
    amber = (band == AMBER).all(axis=-1)
    ink = indigo | turquoise | amber
    cols = np.any(ink, axis=0)
    xs = np.nonzero(cols)[0]
    if xs.size == 0:
        return 1.0, False
    min_x = int(xs[0])
    max_x = int(xs[-1])
    total = max_x - min_x + 1
    left = int(np.count_nonzero(np.any(indigo[:, min_x : max_x + 1], axis=0)))
    mid = int(np.count_nonzero(np.any(turquoise[:, min_x : max_x + 1], axis=0)))
    right = int(np.count_nonzero(np.any(amber[:, min_x : max_x + 1], axis=0)))
    if total <= 0 or (mid + right) <= 0:
        return 1.0, False
    return float((left + mid) / (mid + right)), True


def decision_from_image(image: Image.Image) -> str:
    """Return A or B: which candidate reproduces the reference's composed ratio."""
    arr = np.asarray(image.convert("RGB"))
    q_ref, ok_ref = _composed_q_in_band(arr, PANEL_TOPS["reference"])
    q_a, ok_a = _composed_q_in_band(arr, PANEL_TOPS["A"])
    q_b, ok_b = _composed_q_in_band(arr, PANEL_TOPS["B"])
    if not (ok_ref and ok_a and ok_b):
        return "A"
    da = abs(q_a - q_ref)
    db = abs(q_b - q_ref)
    if da <= db:
        return "A"
    return "B"
