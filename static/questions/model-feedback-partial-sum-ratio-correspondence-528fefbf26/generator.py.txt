"""Partial-sum-ratio correspondence world.

Three horizontal three-section bars are drawn on a white canvas. Every bar is
divided by two vertical boundaries into, from left to right, an indigo section,
a turquoise middle section, and an amber section. Writing the left, middle and
right section widths as fractions (``l``, ``m``, ``r = 1 - l - m``) of the bar's
own length, each bar carries a *composed* derived quantitative relation

    Q = (l + m) / (m + r)  =  (l + m) / (1 - l)

i.e. the ratio of the two left-most spans (indigo plus turquoise) to the two
right-most spans (turquoise plus amber). The reference bar fixes ``Q``. Two
candidate bars, ``A`` and ``B``, are drawn at different total widths and
positions. Exactly one candidate reproduces the reference's composed value
``Q``; the other does not.

The correct candidate reproduces ``Q`` while its *simpler*, single-span
middle-over-left ratio ``m/l`` is deliberately made to differ from the
reference's, and the wrong (decoy) candidate is constructed so that its simple
middle-over-left ratio ``m/l`` *matches* the reference exactly but its composed
value ``Q`` differs clearly. This is the one causal change away from the parent
``segment_ratio_correspondence``: the parent decides on a single-span ratio
``middle/left`` and matches it across uniformly-scaled bars; here the decision
moves to a ratio built from *sums* of two spans each, and the decoy is tuned so
that any solver reusing the parent's single-span middle-over-left routine is
systematically misled toward the wrong candidate, while the composed partial-sum
comparison still selects the unique correct one.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 560
BACKGROUND = (255, 255, 255)
INDIGO = (63, 72, 204)
TURQUOISE = (0, 188, 168)
AMBER = (255, 176, 32)
LABEL = (35, 35, 35)

BAR_HEIGHT = 38
PANEL_TOPS = {"reference": 150, "A": 320, "B": 460}

REF_W_LO, REF_W_HI = 220, 300
CAND_W_LO, CAND_W_HI = 170, 300

Q_LO, Q_HI = 0.40, 1.20
DELTA_Q_LO, DELTA_Q_HI = 0.30, 0.55
DELTA_P_LO = 0.15
QUAR_MIN_MARGIN = 0.30
MIN_SECTION_FRAC = 0.12

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def composed_ratio(left_frac: float, mid_frac: float) -> float:
    """Composed partial-sum ratio: (left + middle) / (middle + right).

    Right is the residual of the full bar, ``1 - left - middle``.
    """
    r = 1.0 - left_frac - mid_frac
    return float((left_frac + mid_frac) / (mid_frac + r))


def simple_ratio(left_frac: float, mid_frac: float) -> float:
    """The parent's single-span middle-over-left ratio (kept as a distractor cue)."""
    return float(mid_frac / left_frac)


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)

    ref_w = float(rng.integers(REF_W_LO, REF_W_HI + 1))
    a_w = float(rng.integers(CAND_W_LO, CAND_W_HI + 1))
    b_w = float(rng.integers(CAND_W_LO, CAND_W_HI + 1))
    ref_cx = float(rng.uniform(0.35, 0.65) * WIDTH)
    a_cx = float(rng.uniform(0.30, 0.70) * WIDTH)
    b_cx = float(rng.uniform(0.30, 0.70) * WIDTH)

    while True:
        ref_l = float(rng.uniform(0.16, 0.34))
        ref_m = float(rng.uniform(0.16, 0.34))
        q_ref = composed_ratio(ref_l, ref_m)
        if 1.0 - ref_l - ref_m >= 0.18 and Q_LO <= q_ref <= Q_HI:
            break
    p_ref = simple_ratio(ref_l, ref_m)

    def _legible(l: float, m: float) -> bool:
        return (
            l >= MIN_SECTION_FRAC
            and m >= MIN_SECTION_FRAC
            and (1.0 - l - m) >= MIN_SECTION_FRAC
            and 1.0 - l - m >= 0.15
        )

    # Correct candidate: reproduces Q_ref but breaks the simple m/l cue.
    c_l = c_m = None
    for _ in range(4000):
        l_c = float(rng.uniform(0.14, 0.42))
        m_c = q_ref * (1.0 - l_c) - l_c
        if not _legible(l_c, m_c):
            continue
        if abs(simple_ratio(l_c, m_c) - p_ref) < DELTA_P_LO:
            continue
        c_l, c_m = l_c, m_c
        break
    if c_l is None:
        raise AssertionError("could not place correct candidate")

    # Decoy: matches the simple middle-over-left cue exactly, differs in Q.
    w_l = w_m = None
    for _ in range(4000):
        l_w = float(rng.uniform(0.14, 0.42))
        m_w = p_ref * l_w
        if not _legible(l_w, m_w):
            continue
        q_w = composed_ratio(l_w, m_w)
        if abs(q_w - q_ref) < DELTA_Q_LO:
            continue
        w_l, w_m = l_w, m_w
        break
    if w_l is None:
        raise AssertionError("could not place decoy candidate")

    correct_index = int(rng.integers(0, 2))
    if correct_index == 0:
        a_l, a_m, b_l, b_m = c_l, c_m, w_l, w_m
    else:
        a_l, a_m, b_l, b_m = w_l, w_m, c_l, c_m

    return {
        "reference_l": float(round(ref_l, 4)),
        "reference_m": float(round(ref_m, 4)),
        "a_l": float(round(a_l, 4)),
        "a_m": float(round(a_m, 4)),
        "b_l": float(round(b_l, 4)),
        "b_m": float(round(b_m, 4)),
        "ref_width": ref_w,
        "ref_cx": ref_cx,
        "a_width": a_w,
        "a_cx": a_cx,
        "b_width": b_w,
        "b_cx": b_cx,
        "correct_index": correct_index,
    }


def _bar_box(center_x: float, width: float, top: int, height: int) -> tuple[float, float, int, int]:
    x0 = center_x - width / 2.0
    x1 = center_x + width / 2.0
    return x0, x1, top, top + height


def _draw_bar(draw: ImageDraw.ImageDraw, cx: float, width: float, top: int, lf: float, mf: float) -> None:
    x0, x1, top, bottom = _bar_box(cx, width, top, BAR_HEIGHT)
    l_end = int(round(x0 + lf * width))
    m_end = int(round(x0 + (lf + mf) * width))
    x1i = int(round(x0 + width))
    draw.rectangle([x0, top, l_end, bottom], fill=INDIGO)
    draw.rectangle([l_end + 1, top, m_end, bottom], fill=TURQUOISE)
    draw.rectangle([m_end + 1, top, x1i, bottom], fill=AMBER)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype(_FONT_PATH, 26)
    except OSError:
        font = ImageFont.load_default()

    draw.text((12, 116), "REFERENCE", font=font, fill=LABEL)
    _draw_bar(
        draw, scene["ref_cx"], scene["ref_width"], PANEL_TOPS["reference"],
        scene["reference_l"], scene["reference_m"],
    )
    draw.text((12, PANEL_TOPS["A"] + 6), "A", font=font, fill=LABEL)
    _draw_bar(
        draw, scene["a_cx"], scene["a_width"], PANEL_TOPS["A"],
        scene["a_l"], scene["a_m"],
    )
    draw.text((12, PANEL_TOPS["B"] + 6), "B", font=font, fill=LABEL)
    _draw_bar(
        draw, scene["b_cx"], scene["b_width"], PANEL_TOPS["B"],
        scene["b_l"], scene["b_m"],
    )
    return img


def analytic_gold(scene: dict) -> str:
    q_ref = composed_ratio(scene["reference_l"], scene["reference_m"])
    q_a = composed_ratio(scene["a_l"], scene["a_m"])
    q_b = composed_ratio(scene["b_l"], scene["b_m"])
    da = abs(q_a - q_ref)
    db = abs(q_b - q_ref)
    return "A" if da <= db else "B"


def margin(scene: dict) -> float:
    q_a = composed_ratio(scene["a_l"], scene["a_m"])
    q_b = composed_ratio(scene["b_l"], scene["b_m"])
    return float(abs(q_a - q_b))


def is_quarantined(scene: dict) -> bool:
    q_ref = composed_ratio(scene["reference_l"], scene["reference_m"])
    if not (Q_LO <= q_ref <= Q_HI):
        return True
    if margin(scene) < QUAR_MIN_MARGIN:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
