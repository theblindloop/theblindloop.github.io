"""Quarter-turn stack world.

A square panel carries a 7x7 lattice of cells.  Many cells are inked, but the
inked pattern by itself is not the answer.  The question declares one raster
operation: rotate the panel a quarter turn, a half turn and a three-quarter
turn about its own centre, stack the four views, and keep only the cells that
are dark in every view.  That intersection - and nothing visible in any single
view - forms one of five named abstract symbols.

Construction guarantees the intersection exactly: cells of the intended symbol
are painted in every one of their four rotational positions, while every
distractor orbit is deliberately broken, i.e. at most three of its four
positions are inked, so no distractor can survive the stack.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N = 7  # lattice side; odd so the centre cell is fixed by the quarter turn
CENTER = N // 2
NCELLS = N * N
INK_CELLS = 28  # fixed painted-cell budget: total ink carries no answer signal


def rot90(r: int, c: int) -> tuple[int, int]:
    """One quarter turn of the lattice about its centre."""

    return (c, N - 1 - r)


def orbit(r: int, c: int) -> tuple[int, ...]:
    """The rotational orbit of a cell, as sorted flat indices."""

    cells = set()
    for _ in range(4):
        cells.add(r * N + c)
        r, c = rot90(r, c)
    return tuple(sorted(cells))


ORBITS: tuple[tuple[int, ...], ...] = tuple(
    sorted({orbit(r, c) for r in range(N) for c in range(N)})
)


def _cells(predicate) -> frozenset[int]:
    return frozenset(
        r * N + c for r in range(N) for c in range(N) if predicate(r - CENTER, c - CENTER)
    )


# ---------------------------------------------------------------- templates --
# Every template is a union of whole rotational orbits, so each is invariant
# under the declared quarter turn.
TEMPLATES: dict[str, frozenset[int]] = {
    "plus": _cells(lambda dr, dc: dr == 0 or dc == 0),
    "ex": _cells(lambda dr, dc: abs(dr) == abs(dc)),
    "ring": _cells(lambda dr, dc: max(abs(dr), abs(dc)) == 2),
    "diamond": _cells(lambda dr, dc: abs(dr) + abs(dc) == 3),
    "corners": _cells(lambda dr, dc: min(abs(dr), abs(dc)) >= 2),
}
SYMBOLS: tuple[str, ...] = tuple(sorted(TEMPLATES))

for _name, _cellset in TEMPLATES.items():
    assert frozenset(rot90(i // N, i % N)[0] * N + rot90(i // N, i % N)[1] for i in _cellset) == _cellset, _name


def hamming(a: frozenset[int], b: frozenset[int]) -> int:
    return len(a ^ b)


MIN_TEMPLATE_GAP = min(
    hamming(TEMPLATES[a], TEMPLATES[b])
    for i, a in enumerate(SYMBOLS)
    for b in SYMBOLS[i + 1 :]
)
assert MIN_TEMPLATE_GAP >= 12, MIN_TEMPLATE_GAP

# ------------------------------------------------------------------ palette --
# Paper, rule and ink are near-neutral and always widely separated in
# luminance; the corner fiducials are the only strongly saturated colour.  The
# pixel arm only uses that ordering, never a specific hue.
PALETTES = [
    {"bg": (244, 243, 240), "line": (178, 180, 182), "ink": (40, 42, 46)},
    {"bg": (238, 240, 243), "line": (172, 176, 180), "ink": (46, 44, 40)},
    {"bg": (243, 240, 243), "line": (182, 178, 182), "ink": (34, 40, 44)},
    {"bg": (240, 243, 238), "line": (176, 180, 174), "ink": (48, 44, 38)},
]
ACCENTS = [(226, 96, 24), (28, 150, 196), (206, 42, 128), (46, 158, 72)]

MARGIN_PX = 34
FID = 9  # half-side of a corner fiducial square


def _luma(rgb) -> float:
    return 0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]


# -------------------------------------------------------------------- scene --
def _revealed_from_painted(painted: frozenset[int]) -> frozenset[int]:
    """The declared transform: intersection of the four quarter-turn views."""

    survivors = set()
    for index in painted:
        r, c = divmod(index, N)
        if all(cell in painted for cell in orbit(r, c)):
            survivors.add(index)
    return frozenset(survivors)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    # Answer identity is cycled against the seed, so it is independent of every
    # visual draw below (palette, pitch, jitter, accent hue, noise, ink layout).
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    revealed = set(TEMPLATES[symbol])

    # A third of the scenes carry one extra surviving orbit, which moves the
    # revealed pattern four cells off its template and shrinks the runner-up gap.
    free = [o for o in ORBITS if not set(o) & revealed and len(o) == 4]
    rng.shuffle(free)
    if rng.random() < 0.34 and free:
        revealed |= set(free.pop())

    painted = set(revealed)
    # Every remaining orbit is broken: at most three of its four positions are
    # inked, so it cannot survive the stack no matter how dense the ink gets.
    # Ink is topped up orbit by orbit until the fixed budget is exactly met, so
    # total ink is the same for every scene and every symbol.
    quota = {orb: rng.randint(1, 3) for orb in free}
    while len(painted) < INK_CELLS:
        progressed = False
        for orb in free:
            if len(painted) >= INK_CELLS:
                break
            room = [c for c in orb if c not in painted]
            if len(orb) - len(room) >= min(3, quota[orb]):
                continue
            painted.add(rng.choice(sorted(room)))
            progressed = True
        if not progressed:
            quota = {orb: 3 for orb in free}
            if all(sum(c in painted for c in orb) >= 3 for orb in free):
                raise RuntimeError("cannot reach the ink budget with broken orbits only")
    assert len(painted) == INK_CELLS
    assert _revealed_from_painted(frozenset(painted)) == frozenset(revealed)

    order = sorted(painted)
    rng.shuffle(order)  # list order is a pure alias; see latent_symmetries
    return {
        "painted": order,
        "palette": rng.randrange(len(PALETTES)),
        "accent": rng.randrange(len(ACCENTS)),
        "cell": rng.choice([34, 38, 42]),
        "jitter_x": rng.randrange(0, 16),
        "jitter_y": rng.randrange(0, 16),
        "noise": rng.choice([0, 1, 2, 3]),
        "noise_seed": rng.randrange(1, 10_000),
    }


def painted_cells(scene: dict) -> frozenset[int]:
    return frozenset(int(i) for i in scene["painted"])


def revealed_cells(scene: dict) -> frozenset[int]:
    return _revealed_from_painted(painted_cells(scene))


def _ranked(scene: dict) -> list[tuple[int, str]]:
    revealed = revealed_cells(scene)
    return sorted((hamming(revealed, TEMPLATES[s]), s) for s in SYMBOLS)


# ------------------------------------------------------------------- layout --
def _layout(scene: dict) -> dict:
    cell = int(scene["cell"])
    x0 = MARGIN_PX + int(scene["jitter_x"])
    y0 = MARGIN_PX + int(scene["jitter_y"])
    side = N * cell
    size = (x0 + side + MARGIN_PX + 16, y0 + side + MARGIN_PX + 16)
    return {"cell": cell, "x0": x0, "y0": y0, "side": side, "size": size}


def render(scene: dict) -> Image.Image:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    accent = ACCENTS[int(scene["accent"]) % len(ACCENTS)]
    lay = _layout(scene)
    cell, x0, y0, side = lay["cell"], lay["x0"], lay["y0"], lay["side"]
    painted = painted_cells(scene)

    image = Image.new("RGB", lay["size"], palette["bg"])
    draw = ImageDraw.Draw(image)
    for r in range(N):
        for c in range(N):
            x = x0 + c * cell
            y = y0 + r * cell
            draw.rectangle(
                [x, y, x + cell, y + cell],
                fill=palette["ink"] if (r * N + c) in painted else palette["bg"],
                outline=palette["line"],
                width=2,
            )
    draw.rectangle([x0, y0, x0 + side, y0 + side], outline=palette["line"], width=3)
    # Four corner fiducials, centred exactly on the panel's corners.
    for px, py in ((x0, y0), (x0 + side, y0), (x0, y0 + side), (x0 + side, y0 + side)):
        draw.rectangle([px - FID, py - FID, px + FID, py + FID], fill=accent)

    noise = int(scene["noise"])
    if noise:
        rng = random.Random(int(scene["noise_seed"]))
        pixels = image.load()
        width, height = image.size
        for y in range(height):
            for x in range(width):
                r, g, b = pixels[x, y]
                d = rng.randint(-noise, noise)
                pixels[x, y] = (
                    min(255, max(0, r + d)),
                    min(255, max(0, g + d)),
                    min(255, max(0, b + d)),
                )
    return image


# --------------------------------------------------------------------- gold --
def analytic_gold(scene: dict) -> str:
    return _ranked(scene)[0][1]


def _legibility(scene: dict) -> float:
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    bg = _luma(palette["bg"])
    ink = _luma(palette["ink"])
    line = _luma(palette["line"])
    thresh = (bg + ink) / 2.0
    sep = min(bg - line, line - thresh, thresh - ink)
    return sep / (1.0 + 0.5 * int(scene["noise"]))


def margin(scene: dict) -> float:
    ranked = _ranked(scene)
    gap = float(ranked[1][0] - ranked[0][0])
    factor = max(0.0, min(1.0, _legibility(scene) / 24.0))
    return round(gap * factor, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical re-encodings of one and the same scene."""

    reordered = dict(scene)
    reordered["painted"] = sorted(scene["painted"], reverse=True)
    sorted_order = dict(scene)
    sorted_order["painted"] = sorted(scene["painted"])
    palette_modulus = dict(scene)
    palette_modulus["palette"] = int(scene["palette"]) + len(PALETTES)
    accent_modulus = dict(scene)
    accent_modulus["accent"] = int(scene["accent"]) + len(ACCENTS)
    return [
        ("painted_list_order_descending", reordered),
        ("painted_list_order_ascending", sorted_order),
        ("palette_index_modulus", palette_modulus),
        ("accent_index_modulus", accent_modulus),
    ]
