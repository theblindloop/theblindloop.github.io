"""Pixel-only inverse arm for the quarter-turn stack world.

Input is one rasterised PNG plus the public question contract; nothing else.
The arm locates the four saturated corner marks, rebuilds the 7x7 lattice they
delimit, reads each cell's fill, executes the declared quarter-turn stack on
that raster evidence, and matches the surviving cells against the five symbols
named in the question.  It abstains when the marks are unreadable, when the
lattice has no legible ink/paper separation, or when two symbols are tied.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"
SIDE = 7

# The five symbols named in the question, transcribed independently from their
# verbal descriptions onto a 7x7 lattice.
ART: dict[str, tuple[str, ...]] = {
    "plus": (
        "...#...",
        "...#...",
        "...#...",
        "#######",
        "...#...",
        "...#...",
        "...#...",
    ),
    "ex": (
        "#.....#",
        ".#...#.",
        "..#.#..",
        "...#...",
        "..#.#..",
        ".#...#.",
        "#.....#",
    ),
    "ring": (
        ".......",
        ".#####.",
        ".#...#.",
        ".#...#.",
        ".#...#.",
        ".#####.",
        ".......",
    ),
    "diamond": (
        "...#...",
        "..#.#..",
        ".#...#.",
        "#.....#",
        ".#...#.",
        "..#.#..",
        "...#...",
    ),
    "corners": (
        "##...##",
        "##...##",
        ".......",
        ".......",
        ".......",
        "##...##",
        "##...##",
    ),
}

TEMPLATES: dict[str, frozenset[int]] = {
    name: frozenset(
        r * SIDE + c for r, row in enumerate(rows) for c, ch in enumerate(row) if ch == "#"
    )
    for name, rows in ART.items()
}


def _corner_marks(arr: np.ndarray):
    """Centroids of the four saturated corner marks, or None."""

    high = arr.max(axis=2).astype(np.int16)
    low = arr.min(axis=2).astype(np.int16)
    mask = (high - low) > 60
    ys, xs = np.nonzero(mask)
    if xs.size < 200:
        return None
    mid_x = 0.5 * (float(xs.min()) + float(xs.max()))
    mid_y = 0.5 * (float(ys.min()) + float(ys.max()))
    centres = {}
    for key, sel in (
        ("tl", (xs < mid_x) & (ys < mid_y)),
        ("tr", (xs >= mid_x) & (ys < mid_y)),
        ("bl", (xs < mid_x) & (ys >= mid_y)),
        ("br", (xs >= mid_x) & (ys >= mid_y)),
    ):
        if sel.sum() < 40:
            return None
        centres[key] = (float(xs[sel].mean()), float(ys[sel].mean()))
    return centres


def _read_cells(arr: np.ndarray, centres) -> np.ndarray | None:
    x0 = 0.5 * (centres["tl"][0] + centres["bl"][0])
    x1 = 0.5 * (centres["tr"][0] + centres["br"][0])
    y0 = 0.5 * (centres["tl"][1] + centres["tr"][1])
    y1 = 0.5 * (centres["bl"][1] + centres["br"][1])
    width = (x1 - x0) / SIDE
    height = (y1 - y0) / SIDE
    if width < 8.0 or height < 8.0:
        return None
    luma = arr[:, :, 0] * 0.299 + arr[:, :, 1] * 0.587 + arr[:, :, 2] * 0.114
    rows, cols = luma.shape
    means = np.zeros((SIDE, SIDE), dtype=np.float64)
    for r in range(SIDE):
        for c in range(SIDE):
            cx = x0 + (c + 0.5) * width
            cy = y0 + (r + 0.5) * height
            rx = max(2.0, 0.18 * width)
            ry = max(2.0, 0.18 * height)
            xa, xb = int(round(cx - rx)), int(round(cx + rx)) + 1
            ya, yb = int(round(cy - ry)), int(round(cy + ry)) + 1
            if xa < 0 or ya < 0 or xb > cols or yb > rows:
                return None
            means[r, c] = float(luma[ya:yb, xa:xb].mean())
    if means.max() - means.min() < 40.0:
        return None
    threshold = 0.5 * (means.max() + means.min())
    return means < threshold


def _stack(dark: np.ndarray) -> frozenset[int]:
    """Keep only cells dark in the original and in all three quarter turns."""

    survivors = dark.copy()
    turned = dark
    for _ in range(3):
        turned = np.rot90(turned, k=-1)
        survivors &= turned
    return frozenset(int(r * SIDE + c) for r in range(SIDE) for c in range(SIDE) if survivors[r, c])


def decide(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centres = _corner_marks(arr)
    if centres is None:
        return ABSTAIN, 0.0
    dark = _read_cells(arr, centres)
    if dark is None:
        return ABSTAIN, 0.0
    revealed = _stack(dark)
    ranked = sorted((len(revealed ^ cells), name) for name, cells in TEMPLATES.items())
    gap = float(ranked[1][0] - ranked[0][0])
    if gap < 1.0:
        return ABSTAIN, gap
    return ranked[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return decide(image)[0]


def margin_from_image(image: Image.Image) -> float:
    return decide(image)[1]
