"""Render two balance premises and three proposed universal consequences."""

from __future__ import annotations

from functools import lru_cache
from itertools import product
import math
import random

from PIL import Image, ImageDraw

CANVAS = (1200, 900)
ANSWERS = ("amber", "cyan", "violet")
WEIGHT_FAMILIES = (
    (2, 3, 5, 7, 11), (3, 4, 6, 9, 11), (2, 5, 7, 8, 13),
    (3, 5, 8, 10, 13), (4, 6, 7, 11, 13), (2, 4, 7, 9, 12),
)
RGB = {
    "background": (255, 255, 255), "panel": (247, 249, 252), "ink": (31, 38, 46),
    "red": (210, 61, 67), "blue": (38, 111, 187), "green": (47, 151, 91),
    "purple": (126, 73, 181), "orange": (218, 103, 44),
    "amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190),
}
SYMBOL_COLORS = ("red", "blue", "green", "purple", "orange")
PREMISE_Y = (45, 220)
PROPOSAL_Y = (420, 575, 730)
LEFT_BOX = (35, 515)
RIGHT_BOX = (685, 1165)
ROW_H = 130


def _rank(rows: list[tuple[int, ...]] | tuple[tuple[int, ...], ...]) -> int:
    a = [[float(v) for v in row] for row in rows]
    rank = 0
    for col in range(5):
        pivot = next((i for i in range(rank, len(a)) if abs(a[i][col]) > 1e-8), None)
        if pivot is None:
            continue
        a[rank], a[pivot] = a[pivot], a[rank]
        scale = a[rank][col]
        a[rank] = [v / scale for v in a[rank]]
        for i in range(len(a)):
            if i != rank and abs(a[i][col]) > 1e-8:
                scale = a[i][col]
                a[i] = [v - scale * q for v, q in zip(a[i], a[rank])]
        rank += 1
    return rank


def _canonical(values: tuple[int, ...]) -> tuple[int, ...]:
    divisor = math.gcd(*(abs(v) for v in values if v))
    if divisor:
        values = tuple(v // divisor for v in values)
    first = next((v for v in values if v), 0)
    return tuple(-v for v in values) if first < 0 else values


def _sides(relation: tuple[int, ...]) -> tuple[list[int], list[int]]:
    return ([max(v, 0) for v in relation], [max(-v, 0) for v in relation])


def _compact(relation: tuple[int, ...]) -> bool:
    left, right = _sides(relation)
    return 2 <= sum(left) <= 6 and 2 <= sum(right) <= 6 and max(map(abs, relation)) <= 3


@lru_cache(maxsize=None)
def _relation_pool(weights: tuple[int, ...]) -> tuple[tuple[int, ...], ...]:
    found = set()
    for relation in product(range(-3, 4), repeat=5):
        if relation == (0, 0, 0, 0, 0):
            continue
        if sum(a * b for a, b in zip(relation, weights)) != 0:
            continue
        canonical = _canonical(relation)
        if _compact(canonical):
            found.add(canonical)
    return tuple(sorted(found))


def _build_relations(seed: int) -> tuple[list[list[list[int]]], list[list[list[int]]]]:
    rng = random.Random((int(seed) ^ 0xE17A11ED) & 0xFFFFFFFFFFFFFFFF)
    weights = WEIGHT_FAMILIES[int(seed) % len(WEIGHT_FAMILIES)]
    pool = list(_relation_pool(weights))
    rng.shuffle(pool)
    coefficients = ((1, 1), (1, -1), (2, 1), (1, 2), (2, -1), (1, -2))
    choice = None
    for first in pool:
        for second in pool:
            if second == first or _rank([first, second]) != 2:
                continue
            for a, b in coefficients:
                target = _canonical(tuple(a * x + b * y for x, y in zip(first, second)))
                if target in (first, second) or not _compact(target):
                    continue
                outsiders = [r for r in pool if r != target and _rank([first, second, r]) == 3]
                if len(outsiders) < 2:
                    continue
                rng.shuffle(outsiders)
                outsiders.sort(key=lambda r: sum(abs(x - y) for x, y in zip(r, target)))
                choice = (first, second, target, outsiders[0], outsiders[1])
                break
            if choice:
                break
        if choice:
            break
    if choice is None:
        raise RuntimeError("could not construct entailment scene")
    first, second, target, wrong_a, wrong_b = choice
    premises = []
    for relation in (first, second):
        left, right = _sides(relation)
        if rng.randrange(2):
            left, right = right, left
        premises.append([left, right])
    rng.shuffle(premises)
    proposals = []
    for relation in (target, wrong_a, wrong_b):
        left, right = _sides(relation)
        if rng.randrange(2):
            left, right = right, left
        proposals.append([left, right])
    return premises, proposals


def _relation(equation: list[list[int]]) -> tuple[int, ...]:
    if len(equation) != 2 or any(len(side) != 5 for side in equation):
        raise ValueError("malformed equation")
    return tuple(int(a) - int(b) for a, b in zip(equation[0], equation[1]))


def sample_scene(seed: int) -> dict:
    premises, raw_proposals = _build_relations(int(seed))
    target = raw_proposals[0]
    wrong = raw_proposals[1:]
    target_slot = int(seed) % 3
    proposals = list(wrong)
    proposals.insert(target_slot, target)
    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [c for c in ANSWERS if c != answer_color]
    rng = random.Random((int(seed) ^ 0xA115A11C) & 0xFFFFFFFFFFFFFFFF)
    rng.shuffle(other_colors)
    tag_colors: list[str | None] = [None, None, None]
    tag_colors[target_slot] = answer_color
    for slot in range(3):
        if tag_colors[slot] is None:
            tag_colors[slot] = other_colors.pop()
    scene = {"premises": premises, "proposals": proposals, "tag_colors": tag_colors}
    if analytic_gold(scene) != answer_color:
        raise RuntimeError("constructed answer binding is inconsistent")
    return scene


def _symbol(draw: ImageDraw.ImageDraw, kind: int, cx: int, cy: int) -> None:
    color = RGB[SYMBOL_COLORS[kind]]
    if kind == 0:
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=color, outline=RGB["ink"], width=3)
    elif kind == 1:
        draw.rectangle((cx - 19, cy - 19, cx + 19, cy + 19), fill=color, outline=RGB["ink"], width=3)
    elif kind == 2:
        points = ((cx, cy - 23), (cx - 23, cy + 19), (cx + 23, cy + 19))
        draw.polygon(points, fill=color)
        draw.line(points + (points[0],), fill=RGB["ink"], width=3, joint="curve")
    elif kind == 3:
        points = ((cx, cy - 24), (cx - 24, cy), (cx, cy + 24), (cx + 24, cy))
        draw.polygon(points, fill=color)
        draw.line(points + (points[0],), fill=RGB["ink"], width=3, joint="curve")
    else:
        points = ((cx - 20, cy - 12), (cx, cy - 23), (cx + 20, cy - 12),
                  (cx + 20, cy + 12), (cx, cy + 23), (cx - 20, cy + 12))
        draw.polygon(points, fill=color)
        draw.line(points + (points[0],), fill=RGB["ink"], width=3, joint="curve")


def _group(draw: ImageDraw.ImageDraw, vector: list[int], box: tuple[int, int, int, int]) -> None:
    items = [kind for kind, count in enumerate(vector) for _ in range(int(count))]
    x0, y0, x1, y1 = box
    spacing = 64
    start = (x0 + x1 - spacing * (len(items) - 1)) // 2
    for index, kind in enumerate(items):
        _symbol(draw, kind, start + index * spacing, (y0 + y1) // 2)


def _equals(draw: ImageDraw.ImageDraw, y: int) -> None:
    cy = y + ROW_H // 2
    draw.line((568, cy - 11, 632, cy - 11), fill=RGB["ink"], width=7)
    draw.line((568, cy + 11, 632, cy + 11), fill=RGB["ink"], width=7)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    for y, equation in zip(PREMISE_Y, scene["premises"]):
        draw.rounded_rectangle((LEFT_BOX[0], y, LEFT_BOX[1], y + ROW_H), radius=18,
                               fill=RGB["panel"], outline=RGB["ink"], width=4)
        draw.rounded_rectangle((RIGHT_BOX[0], y, RIGHT_BOX[1], y + ROW_H), radius=18,
                               fill=RGB["panel"], outline=RGB["ink"], width=4)
        _group(draw, equation[0], (55, y + 10, 495, y + ROW_H - 10))
        _group(draw, equation[1], (705, y + 10, 1145, y + ROW_H - 10))
        _equals(draw, y)
    for y, equation, color in zip(PROPOSAL_Y, scene["proposals"], scene["tag_colors"]):
        draw.rounded_rectangle((35, y, 1165, y + ROW_H), radius=18,
                               fill=RGB["panel"], outline=RGB[str(color)], width=10)
        _group(draw, equation[0], (55, y + 10, 515, y + ROW_H - 10))
        _group(draw, equation[1], (685, y + 10, 1145, y + ROW_H - 10))
        _equals(draw, y)
    return image


def _entailed(premises: list[tuple[int, ...]], proposal: tuple[int, ...]) -> bool:
    return _rank(premises + [proposal]) == _rank(premises)


def analytic_gold(scene: dict) -> str:
    premises = [_relation(eq) for eq in scene["premises"]]
    if _rank(premises) != 2:
        raise ValueError("premises must be independent")
    matches = [i for i, eq in enumerate(scene["proposals"]) if _entailed(premises, _relation(eq))]
    if len(matches) != 1 or set(scene["tag_colors"]) != set(ANSWERS):
        raise ValueError("scene has no unique entailed proposal")
    return str(scene["tag_colors"][matches[0]])


def _minor3(rows: list[tuple[int, ...]], cols: tuple[int, int, int]) -> int:
    a, b, c = ([row[i] for i in cols] for row in rows)
    return (a[0] * (b[1] * c[2] - b[2] * c[1])
            - a[1] * (b[0] * c[2] - b[2] * c[0])
            + a[2] * (b[0] * c[1] - b[1] * c[0]))


def margin(scene: dict) -> float:
    premises = [_relation(eq) for eq in scene["premises"]]
    gaps = []
    for equation in scene["proposals"]:
        relation = _relation(equation)
        if not _entailed(premises, relation):
            gaps.append(max(abs(_minor3(premises + [relation], cols))
                            for cols in ((0, 1, 2), (0, 1, 3), (0, 1, 4), (0, 2, 3),
                                         (0, 2, 4), (0, 3, 4), (1, 2, 3), (1, 2, 4),
                                         (1, 3, 4), (2, 3, 4))))
    return float(min(gaps)) if len(gaps) == 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1 or analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
