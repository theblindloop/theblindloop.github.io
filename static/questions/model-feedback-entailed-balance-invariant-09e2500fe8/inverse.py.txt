"""Independent image-only decoder for balance entailment."""

from __future__ import annotations

from fractions import Fraction

import numpy as np
from PIL import Image

CANVAS = (1200, 900)
INK = (31, 38, 46)
PANEL = (247, 249, 252)
SYMBOL_RGB = ((210, 61, 67), (38, 111, 187), (47, 151, 91), (126, 73, 181), (218, 103, 44))
TAG_RGB = {"amber": (226, 151, 28), "cyan": (18, 157, 181), "violet": (141, 77, 190)}
ALLOWED = {(255, 255, 255), PANEL, INK, *SYMBOL_RGB, *TAG_RGB.values()}
PREMISE_Y = (45, 220)
PROPOSAL_Y = (420, 575, 730)


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return out


def _read_group(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[int]:
    x0, y0, x1, y1 = box
    vector = []
    for color in SYMBOL_RGB:
        mask = np.all(arr[y0:y1, x0:x1] == color, axis=2)
        components = [c for c in _components(mask) if c[4] > 400]
        if any(not (500 <= c[4] <= 1500 and 30 <= c[2] - c[0] + 1 <= 45
                    and 30 <= c[3] - c[1] + 1 <= 46) for c in components):
            raise OracleError("malformed symbol")
        if int(mask.sum()) - sum(c[4] for c in components) > 10:
            raise OracleError("damaged symbol fragments")
        vector.append(len(components))
    if not 2 <= sum(vector) <= 6:
        raise OracleError("invalid group size")
    return vector


def _rank(rows: list[list[int]]) -> int:
    a = [[Fraction(v) for v in row] for row in rows]
    rank = 0
    for col in range(5):
        pivot = next((i for i in range(rank, len(a)) if a[i][col]), None)
        if pivot is None:
            continue
        a[rank], a[pivot] = a[pivot], a[rank]
        scale = a[rank][col]
        a[rank] = [v / scale for v in a[rank]]
        for i in range(len(a)):
            if i != rank and a[i][col]:
                scale = a[i][col]
                a[i] = [v - scale * q for v, q in zip(a[i], a[rank])]
        rank += 1
    return rank


def _check_equals(arr: np.ndarray, y: int) -> None:
    cy = y + 65
    count = int(np.all(arr[cy - 18:cy + 19, 560:640] == INK, axis=2).sum())
    if count < 700:
        raise OracleError("equality mark is missing")


def _relation(left: list[int], right: list[int]) -> list[int]:
    return [a - b for a, b in zip(left, right)]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas")
    colors = rgb.getcolors(maxcolors=32)
    if colors is None or not {tuple(color) for _, color in colors} <= ALLOWED:
        raise OracleError("unknown raster color")
    arr = np.asarray(rgb)
    premises = []
    for y in PREMISE_Y:
        left = _read_group(arr, (55, y + 10, 495, y + 120))
        right = _read_group(arr, (705, y + 10, 1145, y + 120))
        _check_equals(arr, y)
        premises.append(_relation(left, right))
    if _rank(premises) != 2:
        raise OracleError("premises are not independent")
    matches, seen = [], set()
    for y in PROPOSAL_Y:
        left = _read_group(arr, (55, y + 10, 515, y + 120))
        right = _read_group(arr, (685, y + 10, 1145, y + 120))
        _check_equals(arr, y)
        color_counts = {name: int(np.all(arr[y:y + 11, 350:850] == color, axis=2).sum())
                        for name, color in TAG_RGB.items()}
        names = [name for name, count in color_counts.items() if count > 2000]
        if len(names) != 1 or names[0] in seen:
            raise OracleError("proposal border is invalid")
        seen.add(names[0])
        relation = _relation(left, right)
        if _rank(premises + [relation]) == 2:
            matches.append(names[0])
    if seen != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique entailed proposal")
    return matches[0]
