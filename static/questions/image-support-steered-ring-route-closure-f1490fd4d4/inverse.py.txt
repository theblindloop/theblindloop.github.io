"""Independent pixel-only inverse for the topology of the blue route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _blue_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    return (b > 145) & (b - r > 80) & (b - g > 35) & (g > 65) & (g < 175)


def _components(mask: np.ndarray) -> list[int]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y, x in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y, x]:
            continue
        q = [(int(y), int(x))]
        seen[y, x] = True
        size = 0
        while q:
            yy, xx = q.pop()
            size += 1
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _enclosed_background_area(route: np.ndarray) -> int:
    """Return background pixels unreachable from the image border."""
    h, w = route.shape
    free = ~route
    seen = np.zeros_like(route, dtype=bool)
    q: deque[tuple[int, int]] = deque()
    for x in range(w):
        for y in (0, h - 1):
            if free[y, x] and not seen[y, x]:
                seen[y, x] = True
                q.append((y, x))
    for y in range(h):
        for x in (0, w - 1):
            if free[y, x] and not seen[y, x]:
                seen[y, x] = True
                q.append((y, x))
    while q:
        y, x = q.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and free[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                q.append((ny, nx))
    return int(np.count_nonzero(free & ~seen))


def decision_from_image(image: Image.Image) -> str:
    route = _blue_mask(image)
    area = int(route.sum())
    if area < 2500:
        raise ValueError("route ink is missing or materially erased")
    ys, xs = np.nonzero(route)
    if int(xs.max() - xs.min()) < 260 or int(ys.max() - ys.min()) < 220:
        raise ValueError("route span is too small or cropped")
    # Topology is measured on conservative 4x4 occupancy blocks.  The sampled
    # route is at least 12 px thick and every open gap exceeds 100 px, so this
    # preserves both connectivity classes while keeping repeated ablation probes
    # inexpensive and independent of any renderer coordinates.
    h, w = route.shape
    block = 4
    topology = route[: h - h % block, : w - w % block].reshape(
        h // block, block, w // block, block
    ).any(axis=(1, 3))
    large = [size for size in _components(topology) if size >= 5]
    if len(large) != 1:
        raise ValueError("route is not one connected visible component")
    return "yes" if _enclosed_background_area(topology) >= 300 else "no"
