"""Deterministic scene sampler and renderer for visible route closure."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

CANVAS = 512
SCALE = 4


def sample_scene(seed: int) -> dict[str, Any]:
    """Return a balanced, randomized closed-loop or open-arc route scene."""
    rng = random.Random(int(seed))
    closed = (int(seed) % 2) == 0
    cx = rng.randint(250, 262)
    cy = rng.randint(285, 299)
    rx = rng.randint(158, 184) if closed else rng.randint(176, 198)
    ry = rng.randint(132, 157) if closed else rng.randint(147, 172)
    rotation = rng.uniform(-math.pi, math.pi)
    if closed:
        start = rotation
        sweep = 2.0 * math.pi
    else:
        gap = math.radians(rng.uniform(58.0, 92.0))
        start = rotation + gap / 2.0
        sweep = 2.0 * math.pi - gap
    count = 144 if closed else 121
    points: list[list[float]] = []
    phase = rng.uniform(0.0, 2.0 * math.pi)
    for i in range(count):
        t = start + sweep * i / (count if closed else count - 1)
        wobble = 1.0 + 0.035 * math.sin(3.0 * t + phase)
        points.append([cx + rx * wobble * math.cos(t), cy + ry * wobble * math.sin(t)])
    return {
        "closed": closed,
        "points": points,
        "stroke_width": rng.choice([12, 13, 14, 15, 16]),
        "route_color": rng.choice([[20, 105, 205], [25, 115, 220], [15, 125, 210]]),
        "label_color": [52, 57, 64],
        "background": rng.choice([[250, 248, 242], [248, 250, 252], [252, 249, 246]]),
        "width": CANVAS,
        "height": CANVAS,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only declared visible scene fields; no answer-specific annotation."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background"])
    route_color = tuple(scene["route_color"])
    label_color = tuple(scene["label_color"])
    stroke_width = int(scene["stroke_width"])
    points = [(float(p[0]) * SCALE, float(p[1]) * SCALE) for p in scene["points"]]
    if bool(scene["closed"]):
        points.append(points[0])

    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)
    font = ImageFont.truetype("DejaVuSans-Bold.ttf", 25 * SCALE)
    label = "RING ROUTE"
    box = draw.textbbox((0, 0), label, font=font)
    label_w = box[2] - box[0]
    draw.text(((width * SCALE - label_w) / 2, 20 * SCALE), label, font=font, fill=label_color)

    scaled_width = stroke_width * SCALE
    draw.line(points, fill=route_color, width=scaled_width, joint="curve")
    radius = scaled_width / 2.0
    for x, y in points:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=route_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if bool(scene["closed"]) else "no"


def margin(scene: dict[str, Any]) -> float:
    """One-sided visible clearance: stroke thickness or open-end separation."""
    width = float(scene["stroke_width"])
    if bool(scene["closed"]):
        return width
    first, last = scene["points"][0], scene["points"][-1]
    return math.hypot(float(first[0]) - float(last[0]), float(first[1]) - float(last[1])) - width


def is_quarantined(scene: dict[str, Any]) -> bool:
    if float(scene["stroke_width"]) < 8.0:
        return True
    return (not bool(scene["closed"])) and margin(scene) < 50.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
