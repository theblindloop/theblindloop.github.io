"""Scene-free pixel inverse arm for count_out_elimination_ring.

Reads only the rendered RGB image.  It re-finds the row of six or seven
coloured discs, the
black marker triangle above the row and the black pips on the bottom card, then
replays the count-out from the marked disc and returns the colour of the disc
that is never taken away.

This module imports no world code and receives no latent scene.  Its palette,
its band split and its replay rule are stated here independently; everything it
uses is a property of the drawn strokes.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "count_out_elimination_ring-oracle-0.1.0"

UNREADABLE = "unreadable"

# Palette of the drawn discs, declared independently of the drawing code.
PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (206, 48, 42),
    "orange": (235, 122, 24),
    "yellow": (222, 190, 36),
    "green": (36, 142, 72),
    "teal": (26, 150, 156),
    "blue": (46, 86, 196),
    "purple": (128, 60, 168),
}

COLOUR_TOL = 40          # per-channel tolerance when matching a disc colour
MIN_DISC_PIXELS = 400    # a drawn disc is far larger than this
DARK_SUM = 170           # near-black ink: the marker triangle and the pips
MIN_DARK_PIXELS = 30
DISC_COUNT_RANGE = (6, 7)
K_RANGE = (2, 5)


def _colour_centres(arr: np.ndarray) -> list[tuple[float, str]]:
    """Centre x of every drawn disc, tagged with its palette colour name."""
    found: list[tuple[float, str]] = []
    for name, rgb in PALETTE.items():
        diff = np.abs(arr.astype(np.int16) - np.array(rgb, dtype=np.int16))
        mask = np.all(diff <= COLOUR_TOL, axis=2)
        if int(mask.sum()) < MIN_DISC_PIXELS:
            continue
        ys, xs = np.nonzero(mask)
        found.append((float(xs.mean()), name))
    return sorted(found)


def _column_runs(mask: np.ndarray) -> list[tuple[int, int]]:
    """Maximal runs of columns that contain at least one set pixel."""
    cols = np.nonzero(mask.any(axis=0))[0]
    runs: list[tuple[int, int]] = []
    if cols.size == 0:
        return runs
    start = prev = int(cols[0])
    for c in cols[1:]:
        c = int(c)
        if c > prev + 1:
            runs.append((start, prev))
            start = c
        prev = c
    runs.append((start, prev))
    return runs


def _survivor_position(n: int, start: int, k: int) -> int:
    alive = list(range(n))
    idx = start
    while len(alive) > 1:
        idx = (idx + k - 1) % len(alive)
        alive.pop(idx)
        idx %= len(alive)
    return alive[0]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    h = arr.shape[0]

    discs = _colour_centres(arr)
    if not (DISC_COUNT_RANGE[0] <= len(discs) <= DISC_COUNT_RANGE[1]):
        return UNREADABLE
    xs = [x for x, _ in discs]
    names = [n for _, n in discs]

    dark = arr.astype(np.int32).sum(axis=2) <= DARK_SUM

    # The disc row is the horizontal band that holds the coloured discs; the
    # marker is the dark ink above it and the pips are the dark ink below it.
    colour_rows = np.zeros(h, dtype=bool)
    for name, rgb in PALETTE.items():
        diff = np.abs(arr.astype(np.int16) - np.array(rgb, dtype=np.int16))
        m = np.all(diff <= COLOUR_TOL, axis=2)
        if int(m.sum()) >= MIN_DISC_PIXELS:
            colour_rows |= m.any(axis=1)
    rows = np.nonzero(colour_rows)[0]
    if rows.size == 0:
        return UNREADABLE
    row_top, row_bottom = int(rows.min()), int(rows.max())

    above = dark.copy()
    above[row_top:, :] = False
    if int(above.sum()) < MIN_DARK_PIXELS:
        return UNREADABLE
    marker_x = float(np.nonzero(above)[1].mean())

    below = dark.copy()
    below[: row_bottom + 1, :] = False
    if int(below.sum()) < MIN_DARK_PIXELS:
        return UNREADABLE
    pip_runs = [r for r in _column_runs(below) if (r[1] - r[0]) >= 4]
    k = len(pip_runs)
    if not (K_RANGE[0] <= k <= K_RANGE[1]):
        return UNREADABLE

    start = min(range(len(xs)), key=lambda i: abs(marker_x - xs[i]))
    return names[_survivor_position(len(xs), start, k)]
