"""Renderer, analytic gold and margin for count_out_elimination_ring.

Six or seven differently coloured discs stand in a row.  The row is read as a closed
ring: counting runs rightwards and, after the rightmost disc still standing, it
continues at the leftmost disc still standing.  A black triangle above the row
marks the disc that is counted as one at the very first count.  A cream card at
the bottom carries a number of black pips: that count is the step length k.

Repeatedly count k discs, starting from the marked disc counted as one and
counting only discs still standing; the k-th disc counted is taken away.  The
next count starts at the disc still standing that follows the one just taken
away, counted as one.  Discs go until exactly one is left.

The decision is the colour of the disc that is never taken away.  Only the
starting row is drawn: no intermediate ring and no final disc is drawn.

The scene is a plain JSON-able dict.  Discs live in an unordered bag: each disc
carries its own drawn position, so the bag may be permuted without changing the
raster or the answer.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "count_out_elimination_ring-renderer-0.1.0"

# ---------------------------------------------------------------- palette ---
COLOURS: dict[str, tuple[int, int, int]] = {
    "red": (206, 48, 42),
    "orange": (235, 122, 24),
    "yellow": (222, 190, 36),
    "green": (36, 142, 72),
    "teal": (26, 150, 156),
    "blue": (46, 86, 196),
    "purple": (128, 60, 168),
}
COLOUR_NAMES = ("red", "orange", "yellow", "green", "teal", "blue", "purple")

BG = (245, 247, 250)
CARD_FILL = (250, 238, 205)
CARD_EDGE = (176, 162, 128)
INK = (15, 15, 15)
ARROW_GREY = (150, 150, 150)

# ----------------------------------------------------------------- layout ---
W, H = 760, 320
N_DISC_CHOICES = (6, 7)
MAX_DISCS = 7
PITCH = 96
DISC_R = 26
ROW_Y = 118
HALF_PITCH = PITCH / 2.0
JITTER_MAX = 30.0
QUARANTINE_MARGIN = 8.0

MARK_Y = ROW_Y - DISC_R - 20                    # tip of the marker triangle
ARROW_Y = ROW_Y + DISC_R + 22

CARD_W, CARD_H = 260, 74
CARD_X0 = (W - CARD_W) // 2
CARD_Y0 = 218
PIP_R = 8
PIP_PITCH = 34

K_CHOICES = (2, 3, 4, 5)


def _row_x0(n: int) -> float:
    """Centre of the leftmost disc: the row of n discs is centred on the page."""
    return float((W - (n - 1) * PITCH) // 2)


def _disc_centre(n: int, i: int) -> float:
    return _row_x0(n) + i * PITCH


# --------------------------------------------------------------- readings ---
def _ordered_discs(scene: dict[str, Any]) -> list[dict[str, Any]]:
    return sorted(scene["discs"], key=lambda c: int(c["pos"]))


def _colours_in_drawn_order(scene: dict[str, Any]) -> list[str]:
    return [str(c["colour"]) for c in _ordered_discs(scene)]


def _n_discs(scene: dict[str, Any]) -> int:
    return int(scene["n_discs"])


def _marker_x(scene: dict[str, Any]) -> float:
    n = _n_discs(scene)
    return _disc_centre(n, int(scene["start_slot"])) + float(scene["jitter"])


def _read_start_slot(scene: dict[str, Any]) -> int:
    """The declared reading: the disc whose centre is nearest the marker."""
    mx = _marker_x(scene)
    n = _n_discs(scene)
    best, best_d = 0, None
    for i in range(n):
        d = abs(mx - _disc_centre(n, i))
        if best_d is None or d < best_d:
            best, best_d = i, d
    return best


def _survivor_position(n: int, start: int, k: int) -> int:
    """Count-out with step k from `start`, returning the last standing index."""
    alive = list(range(n))
    idx = start                       # index inside `alive` counted as one
    while len(alive) > 1:
        idx = (idx + k - 1) % len(alive)
        alive.pop(idx)
        idx %= len(alive)
    return alive[0]


def _removal_order(scene: dict[str, Any]) -> list[int]:
    """Drawn positions of the discs taken away, in the order they go."""
    order: list[int] = []
    alive = list(range(_n_discs(scene)))
    idx = _read_start_slot(scene)
    k = int(scene["k"])
    while len(alive) > 1:
        idx = (idx + k - 1) % len(alive)
        order.append(alive.pop(idx))
        idx %= len(alive)
    return order


# --------------------------------------------------------------- sampling ---
def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    target_answer = COLOUR_NAMES[seed % len(COLOUR_NAMES)]
    near_boundary = (seed % 7) == 3

    relax_after = 4000
    for attempt in range(8000):
        k = rng.choice(K_CHOICES)
        n_discs = rng.choice(N_DISC_CHOICES)
        order = rng.sample(list(COLOUR_NAMES), n_discs)
        start_slot = rng.randrange(n_discs)
        if near_boundary:
            mag = rng.uniform(HALF_PITCH - 7.0, HALF_PITCH - 1.5)
            jitter = round(mag * rng.choice([-1.0, 1.0]), 4)
        else:
            jitter = round(rng.uniform(-JITTER_MAX, JITTER_MAX), 4)

        discs = [{"pos": i, "colour": order[i]} for i in range(n_discs)]
        rng.shuffle(discs)            # the bag order carries nothing
        scene = {
            "canvas": [W, H],
            "n_discs": n_discs,
            "k": k,
            "start_slot": start_slot,
            "jitter": jitter,
            "discs": discs,
        }
        read_start = _read_start_slot(scene)
        pos = _survivor_position(n_discs, read_start, k)
        answer = order[pos]
        if answer != target_answer:
            continue
        if attempt < relax_after:
            if pos == read_start:
                continue              # "the marked disc survives"
            if pos in (0, n_discs - 1):
                continue              # "the leftmost / rightmost disc survives"
            if pos == (read_start + k - 1) % n_discs:
                continue              # "the first disc counted out"
            if abs(pos - read_start) == 1:
                continue              # "a neighbour of the marker"
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


# ------------------------------------------------------------------ gold ----
def analytic_gold(scene: dict[str, Any]) -> str:
    order = _colours_in_drawn_order(scene)
    pos = _survivor_position(_n_discs(scene), _read_start_slot(scene), int(scene["k"]))
    return order[pos]


def margin(scene: dict[str, Any]) -> float:
    return round(float(HALF_PITCH - abs(float(scene["jitter"]))), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Permutations of the disc bag: pixel-identical and answer preserving."""
    bag = list(scene["discs"])
    out: list[tuple[str, dict[str, Any]]] = []
    for name, perm in (
        ("disc_bag_reversed", list(reversed(bag))),
        ("disc_bag_rotated", bag[1:] + bag[:1]),
        ("disc_bag_pos_sorted", sorted(bag, key=lambda c: int(c["pos"]))),
    ):
        alt = dict(scene)
        alt["discs"] = [dict(c) for c in perm]
        out.append((name, alt))
    return out


# ---------------------------------------------------------------- render ----
def render(scene: dict[str, Any]) -> Image.Image:
    img = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(img)

    n = _n_discs(scene)
    for disc in _ordered_discs(scene):
        cx = _disc_centre(n, int(disc["pos"]))
        d.ellipse(
            [cx - DISC_R, ROW_Y - DISC_R, cx + DISC_R, ROW_Y + DISC_R],
            fill=COLOURS[str(disc["colour"])],
        )

    mx = int(round(_marker_x(scene)))
    d.polygon(
        [(mx, MARK_Y + 16), (mx - 12, MARK_Y), (mx + 12, MARK_Y)],
        fill=INK,
    )

    ax0 = _row_x0(n) - DISC_R - 8
    ax1 = _row_x0(n) + (n - 1) * PITCH + DISC_R + 8
    d.line([ax0, ARROW_Y, ax1, ARROW_Y], fill=ARROW_GREY, width=4)
    d.polygon(
        [(ax1 + 13, ARROW_Y), (ax1 - 5, ARROW_Y - 9), (ax1 - 5, ARROW_Y + 9)],
        fill=ARROW_GREY,
    )

    d.rectangle(
        [CARD_X0, CARD_Y0, CARD_X0 + CARD_W, CARD_Y0 + CARD_H],
        fill=CARD_FILL,
        outline=CARD_EDGE,
        width=2,
    )
    k = int(scene["k"])
    pcy = CARD_Y0 + CARD_H / 2.0
    px0 = CARD_X0 + CARD_W / 2.0 - (k - 1) * PIP_PITCH / 2.0
    for j in range(k):
        pcx = px0 + j * PIP_PITCH
        d.ellipse(
            [pcx - PIP_R, pcy - PIP_R, pcx + PIP_R, pcy + PIP_R],
            fill=INK,
        )
    return img
