"""Renderer for the rail-conjunction reveal world.

Two rails of nine square pads each are drawn far apart on the canvas: one
upright rail near a left or right margin and one flat rail near the top or
bottom margin.  Exactly three pads of each rail are dark.  The declared
operation marks every cell of a nine-by-nine field whose row index is a dark
pad of the upright rail *and* whose column index is a dark pad of the flat
rail, i.e. the outer product of the two visible pad patterns.  The figure the
marked cells form is drawn nowhere in the image: each rail alone leaves the
answer two-way ambiguous, and a single dark pad is worth a whole line of the
rebuilt field.

All fills are flat and hard edged, so every pad is exactly one of two rendered
tones and the classification boundary is the midpoint of two tones that are
both rendered inside every rail.
"""

from __future__ import annotations

from PIL import Image, ImageDraw
import numpy as np

RENDERER_VERSION = "rail_conjunction_reveal-renderer-0.1.0"

CANVAS = 720
PAD_PX = 34
PITCH_PX = 52
RAIL_LEN = 8 * PITCH_PX + PAD_PX  # 450
QUARANTINE_CLEARANCE = 34.0

FIGURES = ("solid_block", "dot_lattice", "upright_bars", "flat_bars")

# (row pattern type, column pattern type) -> figure
_FIGURE_BY_TYPES = {
    ("tight", "tight"): "solid_block",
    ("spread", "spread"): "dot_lattice",
    ("tight", "spread"): "upright_bars",
    ("spread", "tight"): "flat_bars",
}

_SPREAD_SETS = tuple(
    (a, b, c)
    for a in range(9)
    for b in range(9)
    for c in range(9)
    if a + 3 <= b and b + 3 <= c
)
_TIGHT_SETS = tuple((a, a + 1, a + 2) for a in range(7))


def _luma(rgb: tuple[int, int, int]) -> float:
    return 0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]


def _tone(value: int, tint: tuple[int, int, int]) -> tuple[int, int, int]:
    return tuple(int(min(255, max(0, value + t))) for t in tint)


def _pattern_type(indices: tuple[int, ...]) -> str:
    gaps = [indices[1] - indices[0], indices[2] - indices[1]]
    if gaps == [1, 1]:
        return "tight"
    if min(gaps) >= 3:
        return "spread"
    raise ValueError(f"pad set {indices} is neither tight nor spread")


def _rect_gap(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> int:
    """Separation in pixels between two axis-aligned boxes (0 when overlapping)."""

    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return 0
    return int(max(dx, dy))


def sample_scene(seed: int) -> dict:
    """Sample one balanced, counterbalanced latent scene."""

    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    figure = FIGURES[(int(seed) * 3) % len(FIGURES)]
    row_type, col_type = next(
        types for types, name in _FIGURE_BY_TYPES.items() if name == figure
    )
    row_pool = _TIGHT_SETS if row_type == "tight" else _SPREAD_SETS
    col_pool = _TIGHT_SETS if col_type == "tight" else _SPREAD_SETS
    row_pads = [int(v) for v in row_pool[int(rng.integers(len(row_pool)))]]
    col_pads = [int(v) for v in col_pool[int(rng.integers(len(col_pool)))]]

    pale = int(rng.integers(198, 223))
    clearance = float(np.round(float(rng.uniform(26.0, 92.0)), 3))
    ink = int(round(pale - 2.0 * clearance))
    if ink < 12:
        ink = 12
        clearance = float(np.round((pale - ink) / 2.0, 3))
    tint = tuple(int(v) for v in rng.integers(-6, 7, size=3))

    upright_side = "left" if rng.integers(2) == 0 else "right"
    flat_side = "top" if rng.integers(2) == 0 else "bottom"
    margin = 46
    upright_x = margin if upright_side == "left" else CANVAS - margin - PAD_PX
    flat_y = margin if flat_side == "top" else CANVAS - margin - PAD_PX

    lo, hi = 40, CANVAS - RAIL_LEN - 40
    upright_y = flat_x = 0
    for _ in range(400):
        upright_y = int(rng.integers(lo, hi + 1))
        flat_x = int(rng.integers(lo, hi + 1))
        up_box = (upright_x, upright_y, upright_x + PAD_PX, upright_y + RAIL_LEN)
        flat_box = (flat_x, flat_y, flat_x + RAIL_LEN, flat_y + PAD_PX)
        if _rect_gap(up_box, flat_box) >= 28:
            break
    else:  # pragma: no cover - the search always succeeds for these geometries
        raise RuntimeError("could not place the two rails apart")

    up_box = (upright_x, upright_y, upright_x + PAD_PX, upright_y + RAIL_LEN)
    flat_box = (flat_x, flat_y, flat_x + RAIL_LEN, flat_y + PAD_PX)

    distractors: list[dict] = []
    wanted = int(rng.integers(4, 8))
    guard = 0
    while len(distractors) < wanted and guard < 900:
        guard += 1
        kind = ("dot", "hbar", "vbar")[int(rng.integers(3))]
        w, h = {"dot": (11, 11), "hbar": (23, 8), "vbar": (8, 23)}[kind]
        x = int(rng.integers(24, CANVAS - 24 - w))
        y = int(rng.integers(24, CANVAS - 24 - h))
        box = (x, y, x + w, y + h)
        if _rect_gap(box, up_box) < 34 or _rect_gap(box, flat_box) < 34:
            continue
        if any(_rect_gap(box, tuple(d["box"])) < 26 for d in distractors):
            continue
        distractors.append({"kind": kind, "box": [int(v) for v in box],
                            "tone": "ink" if rng.integers(2) == 0 else "pale"})

    return {
        "canvas": CANVAS,
        "pad_px": PAD_PX,
        "pitch_px": PITCH_PX,
        "figure": figure,
        "row_pads": row_pads,
        "col_pads": col_pads,
        "upright_side": upright_side,
        "flat_side": flat_side,
        "upright_origin": [int(upright_x), int(upright_y)],
        "flat_origin": [int(flat_x), int(flat_y)],
        "pale_value": pale,
        "ink_value": ink,
        "background_value": 250,
        "tint": [int(v) for v in tint],
        "clearance": clearance,
        "distractors": distractors,
    }


def _pad_box(scene: dict, rail: str, index: int) -> tuple[int, int, int, int]:
    pad = int(scene["pad_px"])
    pitch = int(scene["pitch_px"])
    if rail == "upright":
        x, y = scene["upright_origin"]
        return (int(x), int(y) + index * pitch, int(x) + pad, int(y) + index * pitch + pad)
    x, y = scene["flat_origin"]
    return (int(x) + index * pitch, int(y), int(x) + index * pitch + pad, int(y) + pad)


def render(scene: dict) -> Image.Image:
    """Render the scene as flat, hard-edged blocks."""

    tint = tuple(int(v) for v in scene["tint"])
    background = _tone(int(scene["background_value"]), tint)
    ink = _tone(int(scene["ink_value"]), tint)
    pale = _tone(int(scene["pale_value"]), tint)
    size = int(scene["canvas"])
    image = Image.new("RGB", (size, size), background)
    draw = ImageDraw.Draw(image)

    for spec in scene["distractors"]:
        x0, y0, x1, y1 = (int(v) for v in spec["box"])
        colour = ink if spec["tone"] == "ink" else pale
        if spec["kind"] == "dot":
            draw.polygon(
                [(x0 + (x1 - x0) // 2, y0), (x1, y0 + (y1 - y0) // 2),
                 (x0 + (x1 - x0) // 2, y1), (x0, y0 + (y1 - y0) // 2)],
                fill=colour,
            )
        else:
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=colour)

    for rail, marked in (("upright", scene["row_pads"]), ("flat", scene["col_pads"])):
        dark = set(int(v) for v in marked)
        for index in range(9):
            x0, y0, x1, y1 = _pad_box(scene, rail, index)
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=ink if index in dark else pale)
    return image


def analytic_gold(scene: dict) -> str:
    """Figure formed by the conjunction of the two visible pad patterns."""

    row_type = _pattern_type(tuple(sorted(int(v) for v in scene["row_pads"])))
    col_type = _pattern_type(tuple(sorted(int(v) for v in scene["col_pads"])))
    return _FIGURE_BY_TYPES[(row_type, col_type)]


def conjunction_field(scene: dict) -> np.ndarray:
    """The nine-by-nine field the declared operation rebuilds (never rendered)."""

    field = np.zeros((9, 9), dtype=bool)
    for r in scene["row_pads"]:
        for c in scene["col_pads"]:
            field[int(r), int(c)] = True
    return field


def margin(scene: dict) -> float:
    """Per-pad tone clearance in luma units: half the pale-to-ink luma gap."""

    tint = tuple(int(v) for v in scene["tint"])
    return float(
        round((_luma(_tone(int(scene["pale_value"]), tint))
               - _luma(_tone(int(scene["ink_value"]), tint))) / 2.0, 6)
    )


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered."""

    return []
