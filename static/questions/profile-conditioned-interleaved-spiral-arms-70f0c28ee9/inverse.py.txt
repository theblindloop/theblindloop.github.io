"""Pixel-only inverse arm; this module has no access to latent renderer state."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = np.fromiter((p[0] for p in pixels), dtype=float)
            xs = np.fromiter((p[1] for p in pixels), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(pixels)))
    return found


def measure_phase_scores(image: Image.Image) -> dict[int, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = np.all(array < 82, axis=2)
    gray = (
        (array.min(axis=2) > 105)
        & (array.max(axis=2) < 185)
        & ((array.max(axis=2) - array.min(axis=2)) < 16)
    )
    dots = _components(dark, minimum=35)
    centers = _components(gray, minimum=80)
    if len(dots) != 24:
        raise ValueError(f"expected 24 black dot components, found {len(dots)}")
    if len(centers) != 1:
        raise ValueError(f"expected one gray center component, found {len(centers)}")
    cx, cy, _ = centers[0]
    xy = np.asarray([(x, y) for x, y, _ in dots], dtype=float)
    dx = xy[:, 0] - cx
    dy = xy[:, 1] - cy
    radii = np.hypot(dx, dy)
    normalized = (radii - radii.min()) / max(float(np.ptp(radii)), 1e-9)
    angles = np.arctan2(dy, dx)
    trial_slopes = np.linspace(-6.5, 6.5, 521)
    answer: dict[int, float] = {}
    for multiplicity in (2, 3):
        residual = multiplicity * (
            angles[None, :] - trial_slopes[:, None] * normalized[None, :]
        )
        resultants = np.abs(np.cos(residual).mean(axis=1) + 1j * np.sin(residual).mean(axis=1))
        answer[multiplicity] = float(resultants.max())
    return answer


def decision_from_image(image: Image.Image) -> str:
    scores = measure_phase_scores(image)
    if not all(math.isfinite(value) for value in scores.values()):
        raise ValueError("non-finite phase score")
    return "two" if scores[2] > scores[3] else "three"
