"""Deterministic renderer for a two-versus-three interleaved spiral field."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 520
HEIGHT = 520
POINT_COUNT = 24
QUARANTINE_GAP = 0.20


def _phase_scores(scene: dict) -> dict[int, float]:
    """Global periodic coherence after factoring out radial progression."""
    points = np.asarray(scene["points"], dtype=float)
    center = np.asarray(scene["center"], dtype=float)
    delta = points - center
    radii = np.hypot(delta[:, 0], delta[:, 1])
    radial_phase = (radii - radii.min()) / max(float(np.ptp(radii)), 1e-9)
    angles = np.arctan2(delta[:, 1], delta[:, 0])
    slopes = np.linspace(-6.5, 6.5, 521)
    scores: dict[int, float] = {}
    for arms in (2, 3):
        phases = arms * (angles[None, :] - slopes[:, None] * radial_phase[None, :])
        coherence = np.abs(np.exp(1j * phases).mean(axis=1))
        scores[arms] = float(coherence.max())
    return scores


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    arms = 2 if int(seed) % 2 else 3
    center = [rng.randint(245, 275), rng.randint(245, 275)]
    direction = rng.choice((-1.0, 1.0))
    phase = rng.uniform(0.0, 2.0 * math.pi)
    slope = direction * rng.uniform(3.75, 5.15)
    per_arm = POINT_COUNT // arms
    points: list[list[float]] = []
    for arm in range(arms):
        for step in range(per_arm):
            t = (step + 0.35) / (per_arm - 0.30)
            radius = 43.0 + 169.0 * t + rng.uniform(-1.4, 1.4)
            angle = phase + slope * t + 2.0 * math.pi * arm / arms
            angle += rng.uniform(-0.022, 0.022)
            points.append(
                [
                    round(center[0] + radius * math.cos(angle), 4),
                    round(center[1] + radius * math.sin(angle), 4),
                ]
            )
    rng.shuffle(points)
    return {
        "points": points,
        "center": center,
        "dot_radius": 5,
        "center_radius": 9,
        "width": WIDTH,
        "height": HEIGHT,
        "background": [248, 247, 243],
        "point_color": [25, 28, 31],
        "center_color": [142, 146, 151],
    }


def render(scene: dict) -> Image.Image:
    scale = 3
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    dot_radius = float(scene["dot_radius"]) * scale
    for x, y in scene["points"]:
        box = (
            x * scale - dot_radius,
            y * scale - dot_radius,
            x * scale + dot_radius,
            y * scale + dot_radius,
        )
        draw.ellipse(box, fill=tuple(scene["point_color"]))
    cx, cy = scene["center"]
    center_radius = float(scene["center_radius"]) * scale
    draw.ellipse(
        (
            cx * scale - center_radius,
            cy * scale - center_radius,
            cx * scale + center_radius,
            cy * scale + center_radius,
        ),
        fill=tuple(scene["center_color"]),
    )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _phase_scores(scene)
    return "two" if scores[2] > scores[3] else "three"


def margin(scene: dict) -> float:
    scores = _phase_scores(scene)
    return abs(scores[2] - scores[3])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["points"] = list(reversed(scene["points"]))
    return [("reverse_identical_dot_draw_order", twin)]
