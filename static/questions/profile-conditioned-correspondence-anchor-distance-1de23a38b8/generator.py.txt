"""Anchor-distance near/far correspondence renderer (paper-semantic family).

A cross-panel *anchor-distance (near/far)* correspondence task. Each panel
shows one gray ``anchor`` dot and one colored ``target`` dot. The reference
panel's colored target is placed either clearly *near* its anchor (small
center-to-center distance) or clearly *far* from it (large distance) -
randomized. Three candidate panels A, B, C each show their own anchor and one
colored target (re-arranged elsewhere in the raster with random target colors,
positions, and near/far side). Exactly one candidate reproduces the reference's
near/far *distance relationship*; the other two have the opposite class. The
question asks which candidate A, B, or C reproduces the same near/far distance
relationship as the reference.

This is *not* a within-panel "which anchor is closer" comparison (nearest_square)
and *not* an exact inter-disk spacing match (disk_spacing_correspondence): here
each panel contributes a single normalized boolean distance class (near vs far
relative to that panel's own anchor), and the decision is which candidate
transfers the reference's randomized near/far class across the re-arranged
panels.

The pixel-only inverse arm must, in every panel, recover the gray anchor and the
colored target, measure the center-to-center distance between them, classify it
as near or far by an absolute threshold, and then match the candidate whose
class equals the reference's.

Latent state z (every scene field is rendered - none hidden):

    panels : ordered list of four panels [reference, A, B, C]; each panel is
             {anchor: [x, y], target: [x, y]}. The target's distance class
             (near/far) is encoded in the anchor-target distance.
    width  : raster width.
    height : raster height.

The analytic gold is the label (A/B/C) of the candidate panel whose anchor-target
near/far class equals the reference's. Because panels (all anchor and target
positions) are drawn verbatim, the gold is recoverable from pixels alone; no
gold-only field exists.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "correspondence_anchor_distance-0.1.0"

BACKGROUND = (255, 255, 255)
ANCHOR_COLOR = (95, 95, 95)

TARGET_PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
    "orange": (245, 140, 20),
}

# Public visual geometry (constant across scenes for ink/position counterbalance).
WIDTH = 900
HEIGHT = 1000
BAND_CY = [160, 400, 640, 880]   # reference, A, B, C top to bottom
RADIUS = 16
ANCHOR_X = 260                    # anchor horizontal position per panel
NEAR_MIN = 45
NEAR_MAX = 75
FAR_MIN = 185
FAR_MAX = 215
THRESH = 130.0                    # near/far classifier threshold (px center distance)

_MARGIN_FLOOR = 2.0


def _distance(panel: dict) -> float:
    ax, ay = panel["anchor"]
    tx, ty = panel["target"]
    return math.hypot(tx - ax, ty - ay)


def _state(panel: dict) -> str:
    return "near" if _distance(panel) < THRESH else "far"


def _build_panel(rng: random.Random, yc: float, near: bool) -> dict:
    direction = rng.choice((-1, 1))
    if near:
        d = rng.uniform(NEAR_MIN, NEAR_MAX)
    else:
        d = rng.uniform(FAR_MIN, FAR_MAX)
    ax = ANCHOR_X + rng.randint(-20, 20)
    return {
        "anchor": [round(ax), round(yc)],
        "target": [round(ax + direction * d), round(yc)],
        "target_color": rng.choice(list(TARGET_PALETTE)),
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_near = rng.random() < 0.5
    match_slot = rng.randrange(3)
    cand_states = [not ref_near, not ref_near, not ref_near]
    cand_states[match_slot] = ref_near

    panels = [_build_panel(rng, BAND_CY[0], ref_near)]
    for i in range(3):
        panels.append(_build_panel(rng, BAND_CY[i + 1], cand_states[i]))

    return {
        "panels": panels,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_dot(draw, pos, color, radius: int = RADIUS) -> None:
    x, y = int(pos[0]), int(pos[1])
    draw.ellipse([x - radius, y - radius, x + radius, y + radius], fill=color)


def render(scene: dict) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)

    for panel in scene["panels"]:
        _draw_dot(draw, panel["anchor"], ANCHOR_COLOR)
        _draw_dot(draw, panel["target"], TARGET_PALETTE[panel["target_color"]])
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: label (A/B/C) of the candidate panel whose anchor-target near/far
    relationship matches the reference panel's state."""
    panels = scene["panels"]
    states = []
    for p in panels:
        ax, ay = p["anchor"]
        tx, ty = p["target"]
        d = math.hypot(tx - ax, ty - ay)
        states.append("near" if d < THRESH else "far")
    ref_state = states[0]
    for i in range(1, 4):
        if states[i] == ref_state:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference anchor-distance relationship")


def margin(scene: dict) -> float:
    """One-sided verifiability margin: minimum, over every panel, of the distance
    from that panel's anchor-target distance to the near/far boundary (THRESH)."""
    return float(min(abs(_distance(p) - THRESH) for p in scene["panels"]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only a genuinely ambiguous scene. Near targets sit at most
    NEAR_MAX (75) px from the anchor while far targets sit at least FAR_MIN
    (185) px from it; the anchor-target distance to the 130 px threshold is always
    >= 55 px, so ambiguous scenes are never produced by the sampler."""
    return margin(scene) < _MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (all anchor and target
    positions) is rendered verbatim, so the raster uniquely determines each
    panel's near/far anchor-distance class and the matching candidate; no
    pixel-identical transform is declared."""
    return []
