"""Independent pixel-only inverse arm for the anchor-distance world.

Receives ONLY the final rasterized PNG and the public question contract (the
visual spec). It never imports the renderer, prompt, generator, verifier, or
analytic-gold code, and never reads scene data.

Public visual spec (this file's own specification):

  * the image contains four horizontal panels stacked top to bottom: the top
    one is the ``reference`` panel; the next three are candidate panels A, B, C;
  * every panel shows one dark-gray ``anchor`` disk and exactly one saturated
    colored ``target`` disk; the question is whether the colored target lies
    *near* the gray anchor (small center-to-center distance) or *far* from it
    (large distance);
  * the task: decide which candidate panel (A, B, or C) reproduces the
    reference's near/far distance relationship (its target near or far from its
    own anchor) - the candidate whose anchor-target distance class matches the
    reference's.

The oracle recovers, from pixels alone: every disk and its color, the gray
anchor and the colored target of each panel, the center-to-center anchor-target
distance, each panel's near/far class, and finally which candidate matches the
reference's class.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "correspondence_anchor_distance-oracle-0.1.0"

_BACKGROUND_DIST = 30.0
_GRAY_MAX_CH = 150.0
_GRAY_RANGE = 55.0
_MIN_PX = 100
_THRESH = 130.0
_GROUP_TOL = 60.0


def _is_background(rgb: np.ndarray) -> np.ndarray:
    return np.linalg.norm(rgb - np.array([255.0, 255.0, 255.0]), axis=-1) < _BACKGROUND_DIST


def _is_gray(rgb: np.ndarray) -> bool:
    r, g, b = (float(v) for v in rgb)
    return max(r, g, b) < _GRAY_MAX_CH and max(r, g, b) - min(r, g, b) < _GRAY_RANGE


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    comps = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(x, y)])
                visited[y, x] = True
                cur = []
                while queue:
                    cx, cy = queue.popleft()
                    cur.append((cx, cy))
                    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                        nx_, ny_ = cx + dx, cy + dy
                        if 0 <= nx_ < w and 0 <= ny_ < h and mask[ny_, nx_] and not visited[ny_, nx_]:
                            visited[ny_, nx_] = True
                            queue.append((nx_, ny_))
                comps.append(np.array(cur, dtype=np.float32))
    return comps


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    mask = ~_is_background(arr)
    comps = [c for c in _connected_components(mask) if len(c) >= _MIN_PX]
    if len(comps) != 8:
        raise ValueError(f"expected 8 disks, got {len(comps)}")

    disks = []
    for comp in comps:
        cx, cy = comp.mean(axis=0)
        xs = comp[:, 0].astype(int)
        ys = comp[:, 1].astype(int)
        mean_color = arr[ys, xs].mean(axis=0)
        disks.append({"y": float(cy), "x": float(cx), "color": mean_color})

    disks.sort(key=lambda d: d["y"])
    panels: list[list[dict]] = []
    for disk in disks:
        if panels and abs(disk["y"] - panels[-1][0]["y"]) < _GROUP_TOL:
            panels[-1].append(disk)
        else:
            panels.append([disk])
    if len(panels) != 4:
        raise ValueError(f"expected 4 panels, got {len(panels)}")

    states = []
    for panel in panels:
        anchors = [d for d in panel if _is_gray(d["color"])]
        targets = [d for d in panel if not _is_gray(d["color"])]
        if len(anchors) != 1 or len(targets) != 1:
            raise ValueError("panel must contain exactly one anchor and one target")
        dist = float(np.hypot(targets[0]["x"] - anchors[0]["x"], targets[0]["y"] - anchors[0]["y"]))
        states.append("near" if dist < _THRESH else "far")

    ref_state = states[0]
    for i in range(1, 4):
        if states[i] == ref_state:
            return "ABC"[i - 1]
    raise ValueError("no candidate reproduced the reference anchor-distance relationship")


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
