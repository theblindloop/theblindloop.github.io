"""Independent pixel-only inverse arm for the thickest-stroke world.

This module receives ONLY the rasterized PNG (plus the public palette, which is
the world's fixed naming convention) and re-measures which colored stroke has
the greatest perpendicular line width / ink thickness. It shares no geometry
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "thickest_stroke-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background).

    argmin leaves no unclassified antialias band: edge blends split at their
    ~50% point, which is dominated by the background, so the labeled stroke
    masks stay clean regardless of any anti-aliasing the renderer introduced.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _thickness_from_mask(mask: np.ndarray) -> float:
    """Recover the perpendicular ink width (line thickness) of a straight stroke.

    A long straight stroke has mask area ``length * thickness``. The length is
    the extent of the mask projected onto its dominant (principal) axis, so the
    perpendicular thickness is ``area / length``. This is a scalar width
    measurement recovered purely from the pixel mask, independent of the
    stroke's absolute orientation.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 10:
        return -1.0
    xy = np.stack([xs.astype(np.float64), ys.astype(np.float64)], axis=1)
    mu = xy.mean(axis=0)
    cov = np.cov((xy - mu).T)
    w, v = np.linalg.eigh(cov)
    axis = v[:, int(np.argmax(w))]
    proj = (xy - mu) @ axis
    length = float(proj.max() - proj.min())
    if length <= 0.0:
        return -1.0
    return float(xy.shape[0]) / length


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the thickest (largest line-width) stroke from pixels alone."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_thickness = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        thickness = _thickness_from_mask(mask)
        if thickness > best_thickness:
            best_thickness = thickness
            best_color = color
    return best_color
