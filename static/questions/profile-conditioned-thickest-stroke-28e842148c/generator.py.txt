"""Thickest-stroke selection world: which colored segment has the greatest line width.

Latent scene z carries a set of colored line segments that all share the SAME
true length L on a light panel but differ in their stroke thickness (line width).
The task is a pure measurement: recover the perpendicular ink width of every
stroke by fitting its colored mask, then select the color of the widest stroke.

This world is a *measurement* of a recovered scalar quantity (a perpendicular
line width / ink thickness), and it is a set-argmax selection, counterbalancing
which color owns the maximum width, which spatial column each stroke occupies,
and each stroke's arbitrary orientation.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "thickest_stroke-0.1.0"

# Fixed palette shared with oracle.py (kept as a plain list so both modules can
# agree on the named answer set without importing each other).
PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
LENGTH = 140.0            # every segment has the same true end-to-end length
COL_CENTERS = (80.0, 240.0, 400.0)
WIDTH_RANGE = (5, 21)     # integer stroke widths sampled within this range

# Minimum top-two line-width gap required for a decidable selection (pixels).
MIN_MARGIN_PX = 8.0
# Below this top-two width gap the scene is quarantined as near-boundary yet
# still decidable (well above pixel-fit noise).
QUARANTINE_PX = 6.0


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum.

    Returns a plain dict (JSON-serialisable) that both render() and the manifest
    understand. ``segments`` is a list of {"color", "x0", "y0", "x1", "y1", "width"}.
    """
    rng = random.Random(seed * 1009 + 7)
    ncols = len(PALETTE)
    # Three distinct integer widths, widest at least MIN_MARGIN_PX over runner-up.
    while True:
        widths = sorted(rng.sample(range(WIDTH_RANGE[0], WIDTH_RANGE[1] + 1), ncols))
        if widths[-1] - widths[-2] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    columns = list(range(ncols))
    rng.shuffle(columns)
    segments = []
    for color, column, width in zip(colors, columns, widths):
        cx = COL_CENTERS[column]
        cy = rng.uniform(130.0, 210.0)
        theta = rng.uniform(-0.9, 0.9)
        half = LENGTH / 2.0
        dx = half * math.cos(theta)
        dy = half * math.sin(theta)
        segments.append(
            {
                "color": color,
                "x0": round(cx - dx, 2),
                "y0": round(cy - dy, 2),
                "x1": round(cx + dx, 2),
                "y1": round(cy + dy, 2),
                "width": int(width),
            }
        )
    return {"segments": segments, "width": WIDTH, "height": HEIGHT, "background": "light"}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored strokes of equal length but distinct widths."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for seg in scene["segments"]:
        draw.line(
            [(seg["x0"], seg["y0"]), (seg["x1"], seg["y1"])],
            fill=COLOR_RGB[seg["color"]],
            width=seg["width"],
        )
    return img


def _thickest_stroke(segments: list[dict]) -> tuple[dict, int, float]:
    """Return (segment, its width, top-two width gap) from the analytic scene."""
    keyed = sorted(segments, key=lambda seg: -seg["width"])
    top, second = keyed[0], keyed[1]
    margin = top["width"] - second["width"]
    return top, top["width"], float(margin)


def analytic_gold(scene: dict) -> str:
    """The color of the segment with the greatest line width."""
    top, _width, _margin = _thickest_stroke(scene["segments"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two line-width gap in pixels."""
    _top, _width, margin_px = _thickest_stroke(scene["segments"])
    return float(margin_px)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two line-width gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each segment's color label, geometry, and width) is
    directly rendered as visible ink. Swapping two segments' widths changes which
    color is correct, and any global flip/rotation turns the raster into a
    different image. There is no hidden field that could change the gold without
    changing the raster.
    """
    return []
