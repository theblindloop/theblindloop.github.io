"""Independent pixel-only inverse arm for the texture-energy silhouette world.

Reconstructs the local mark-density field from the final raster alone: it
thresholds the dark marks, smooths them into a density field with a small local
window, finds the single dense interior region that does not touch the image
border, and classifies its silhouette into one of
{circle, square, diamond, triangle} from bounding-box fill ratio and vertical
centroid offset.

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations

import numpy as np
from collections import deque
from PIL import Image

WINDOW = 11
DENSITY_THRESHOLD = 0.30
_SQUARE_MIN = 0.90
_CIRCLE_MIN = 0.68
_OFFSET_MIN = 0.08


def _box_sum(a: np.ndarray, win: int) -> np.ndarray:
    h, w = a.shape
    pad = np.zeros((h + 1, w + 1))
    pad[1:, 1:] = np.cumsum(np.cumsum(a, axis=0), axis=1)
    return pad[:-win, :-win] + pad[win:, win:] - pad[:-win, win:] - pad[win:, :-win]


def _largest_interior(init: np.ndarray) -> np.ndarray:
    lab = np.zeros_like(init, bool)
    best = np.zeros_like(init, bool)
    h, w = init.shape
    for sy, sx in np.argwhere(init):
        if lab[sy, sx]:
            continue
        comp = np.zeros_like(init, bool)
        stack = [(int(sy), int(sx))]
        lab[sy, sx] = True
        while stack:
            y, x = stack.pop()
            comp[y, x] = True
            for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                if 0 <= ny < h and 0 <= nx < w and init[ny, nx] and not lab[ny, nx]:
                    lab[ny, nx] = True
                    stack.append((ny, nx))
        if comp.sum() > best.sum():
            best = comp
    return best


def _dense_silhouette(arr: np.ndarray) -> np.ndarray:
    mask = (np.asarray(arr).astype(int) < 120).astype(float)
    h, w = mask.shape
    dens = _box_sum(mask, WINDOW) / (WINDOW * WINDOW)
    half = WINDOW // 2
    dens = np.pad(dens, ((half, WINDOW - 1 - half), (half, WINDOW - 1 - half)),
                  mode="edge")[:h, :w]
    thresh = dens > DENSITY_THRESHOLD
    thresh[0, :] = False
    thresh[-1, :] = False
    thresh[:, 0] = False
    thresh[:, -1] = False
    return _largest_interior(thresh)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(int)[..., 0]
    mask = _dense_silhouette(arr)
    ys, xs = np.where(mask)
    if xs.size == 0:
        return "circle"
    th = int(ys.max()) - int(ys.min()) + 1
    tw = int(xs.max()) - int(xs.min()) + 1
    fill = float(mask.sum()) / float(th * tw)
    if fill >= _SQUARE_MIN:
        return "square"
    if fill >= _CIRCLE_MIN:
        return "circle"
    centre = (int(ys.min()) + int(ys.max())) / 2.0
    offset = abs(float(ys.mean()) - centre) / th
    return "triangle" if offset > _OFFSET_MIN else "diamond"
