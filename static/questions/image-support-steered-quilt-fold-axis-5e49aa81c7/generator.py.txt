from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

N = 12
SIZE = 512
CELL = 32
PALETTES = [
    [[31, 111, 139], [239, 155, 59]],
    [[70, 82, 156], [222, 91, 112]],
    [[28, 135, 105], [235, 178, 54]],
    [[116, 76, 148], [58, 153, 178]],
]


def _errors(cells):
    lr = sum(cells[r][c] != cells[r][N - 1 - c] for r in range(N) for c in range(N // 2))
    tb = sum(cells[r][c] != cells[N - 1 - r][c] for r in range(N // 2) for c in range(N))
    return lr, tb


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    axis = "left-to-right" if seed % 2 == 0 else "top-to-bottom"
    while True:
        cells = [[0] * N for _ in range(N)]
        if axis == "left-to-right":
            for r in range(N):
                for c in range(N // 2):
                    cells[r][c] = cells[r][N - 1 - c] = rng.randrange(2)
        else:
            for r in range(N // 2):
                for c in range(N):
                    cells[r][c] = cells[N - 1 - r][c] = rng.randrange(2)
        lr, tb = _errors(cells)
        if max(lr, tb) >= 20:
            break
    palette = deepcopy(rng.choice(PALETTES))
    if rng.randrange(2):
        palette.reverse()
    offset = [rng.randint(54, 74), rng.randint(54, 74)]
    return {"cells": cells, "axis": axis, "palette": palette, "offset": offset}


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    palette = scene["palette"]
    x0, y0 = scene["offset"]
    _declared_axis = scene["axis"]
    image = Image.new("RGB", (SIZE, SIZE), (246, 242, 232))
    draw = ImageDraw.Draw(image)
    # Adjacent filled squares share edges: the evidence is one dense area, not islands.
    for r in range(N):
        for c in range(N):
            x, y = x0 + c * CELL, y0 + r * CELL
            draw.rectangle((x, y, x + CELL - 1, y + CELL - 1), fill=tuple(palette[cells[r][c]]))
    draw.rectangle((x0 - 5, y0 - 5, x0 + N * CELL + 4, y0 + N * CELL + 4), outline=(45, 42, 38), width=5)
    return image


def analytic_gold(scene: dict) -> str:
    cells = scene["cells"]
    axis = scene["axis"]
    lr, tb = _errors(cells)
    measured = "left-to-right" if lr < tb else "top-to-bottom"
    if measured != axis or min(lr, tb) != 0:
        raise ValueError("latent field violates declared fold")
    return measured


def margin(scene: dict) -> float:
    lr, tb = _errors(scene["cells"])
    return float(abs(lr - tb))


def is_quarantined(scene: dict) -> bool:
    lr, tb = _errors(scene["cells"])
    return min(lr, tb) != 0 or abs(lr - tb) < 12


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["cells"] = [[1 - v for v in row] for row in scene["cells"]]
    twin["palette"] = list(reversed(scene["palette"]))
    return [("invert_indices_and_reverse_palette", twin)]
