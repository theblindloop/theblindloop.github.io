from __future__ import annotations

import numpy as np
from PIL import Image


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.shape[0] < 100 or arr.shape[1] < 100:
        raise ValueError("image too small")
    bg = np.median(np.concatenate([arr[:12, :12].reshape(-1, 3), arr[-12:, -12:].reshape(-1, 3)]), axis=0)
    dist = np.max(np.abs(arr - bg), axis=2)
    mask = dist > 35
    ys, xs = np.where(mask)
    if len(xs) < 20000:
        raise ValueError("no dense swatch")
    x0, x1, y0, y1 = xs.min(), xs.max(), ys.min(), ys.max()
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    # Remove the dark frame by retaining colorful pixels, then compare paired colors.
    colorful = crop.max(axis=2) - crop.min(axis=2) > 35
    # A seamless quilt has a calibrated ~0.95 colorful fill inside its frame.
    # Missing a 1/8-canvas ablation chunk is evidence erasure, not a smaller quilt.
    if colorful.mean() < 0.94:
        raise ValueError("swatch interior incomplete")
    def score(flipped):
        valid = colorful & np.flip(colorful, axis=flipped)
        delta = np.max(np.abs(crop - np.flip(crop, axis=flipped)), axis=2)
        return float(delta[valid].mean()) if valid.any() else 1e9
    lr = score(1)
    tb = score(0)
    if abs(lr - tb) < 8 or min(lr, tb) > 8:
        raise ValueError("ambiguous fold")
    return "left-to-right" if lr < tb else "top-to-bottom"
