"""Deterministic renderer and analytic arm for trace_bend_leader."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 420
PALETTE = (
    ("red", (220, 55, 65)),
    ("blue", (45, 105, 210)),
    ("green", (35, 155, 90)),
)
QUARANTINE_GAP = 8.0


def _bend(trace: dict[str, Any]) -> float:
    points = trace["points"]
    y0, y1, y2 = (float(point[1]) for point in points)
    return abs((y2 - y1) - (y1 - y0))


def sample_scene(seed: int) -> dict:
    """Sample a balanced, well-separated scene from one integer seed."""
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    runner = (winner + 1 + rng.randrange(2)) % 3
    remaining = next(index for index in range(3) if index not in (winner, runner))

    bends = [0.0, 0.0, 0.0]
    bends[winner] = float(rng.randint(50, 64))
    bends[runner] = float(rng.randint(24, 34))
    bends[remaining] = float(rng.randint(6, 16))

    x0 = rng.randint(125, 155)
    dx = rng.randint(165, 180)
    xs = [x0, x0 + dx, x0 + 2 * dx]
    bands = [92, 210, 328]
    rng.shuffle(bands)

    traces = []
    for index, (name, rgb) in enumerate(PALETTE):
        first_step = rng.randint(-12, 12)
        signed_bend = bends[index] * rng.choice((-1, 1))
        raw = [0.0, float(first_step), float(2 * first_step) + signed_bend]
        center = sum(raw) / 3.0
        ys = [round(bands[index] + value - center, 3) for value in raw]
        traces.append(
            {
                "name": name,
                "rgb": list(rgb),
                "points": [[float(x), y] for x, y in zip(xs, ys)],
            }
        )
    return {"traces": traces}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.rectangle((65, 35, 590, 370), fill=(255, 255, 255), outline=(55, 60, 65), width=2)
    all_points = [point for trace in scene["traces"] for point in trace["points"]]
    checkpoint_xs = sorted({int(round(point[0])) for point in all_points})
    for number, x in enumerate(checkpoint_xs, start=1):
        for y in range(42, 366, 12):
            draw.line((x, y, x, min(y + 5, 366)), fill=(205, 208, 210), width=1)
        draw.text((x - 8, 382), f"C{number}", fill=(40, 43, 46), font=font)
    draw.text((70, 13), "Three equally spaced checkpoints", fill=(35, 38, 42), font=font)
    draw.text((475, 13), "vertical measure", fill=(90, 92, 95), font=font)

    for trace in scene["traces"]:
        rgb = tuple(trace["rgb"])
        pale = tuple(int(round(0.30 * channel + 0.70 * 255)) for channel in rgb)
        points = [(int(round(x)), int(round(y))) for x, y in trace["points"]]
        draw.line(points, fill=pale, width=4)
        for x, y in points:
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=rgb, outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    traces = scene["traces"]
    return str(max(traces, key=_bend)["name"])


def margin(scene: dict) -> float:
    values = sorted((_bend(trace) for trace in scene["traces"]), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
