"""Independent PNG-only inverse arm for trace_bend_leader."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = (
    ("red", (220, 55, 65)),
    ("blue", (45, 105, 210)),
    ("green", (35, 155, 90)),
)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 80:
            groups.append(np.asarray(pixels, dtype=float))
    return groups


def decision_from_image(image: Image.Image) -> str:
    # int32 keeps squared RGB distances from overflowing for pale connectors.
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    bends: dict[str, float] = {}
    for name, rgb in PALETTE:
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) <= 28 * 28
        groups = _components(mask)
        if len(groups) != 3:
            raise ValueError(f"expected three isolated {name} markers, found {len(groups)}")
        centers = []
        for group in groups:
            centers.append((float(group[:, 1].mean()), float(group[:, 0].mean())))
        centers.sort()
        if min(centers[i + 1][0] - centers[i][0] for i in range(2)) < 40:
            raise ValueError(f"{name} checkpoint markers are not horizontally separated")
        y0, y1, y2 = (center[1] for center in centers)
        bends[name] = abs((y2 - y1) - (y1 - y0))
    ranked = sorted(bends, key=bends.get, reverse=True)
    if bends[ranked[0]] - bends[ranked[1]] < 4.0:
        raise ValueError("pixel decision is ambiguous")
    return ranked[0]
