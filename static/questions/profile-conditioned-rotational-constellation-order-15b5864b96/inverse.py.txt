"""Independent pixel-only inverse arm for rotational constellation order."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 35:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _centers(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Estimate the background from border pixels, then segment by chromatic
    # distance. This admits all renderer palettes without knowing them.
    border = np.concatenate([arr[0], arr[-1], arr[:, 0], arr[:, -1]], axis=0)
    bg = np.median(border, axis=0)
    dist = np.sqrt(((arr.astype(float) - bg) ** 2).sum(axis=2))
    comps = _components(dist > 45.0)
    if len(comps) != 18:
        raise ValueError(f"expected 18 visible dot components, found {len(comps)}")
    areas = np.asarray([area for _, _, area in comps], dtype=float)
    if areas.max() / areas.min() > 1.8:
        raise ValueError("dot components are not uniform")
    return np.asarray([[x, y] for x, y, _ in comps], dtype=float)


def _score(points: np.ndarray, angle_deg: float) -> float:
    center = points.mean(axis=0)
    q = points - center
    a = math.radians(angle_deg)
    rotation = np.array([[math.cos(a), -math.sin(a)], [math.sin(a), math.cos(a)]])
    turned = q @ rotation.T + center
    distances = np.sqrt(((turned[:, None, :] - points[None, :, :]) ** 2).sum(axis=2))
    raw = 0.5 * (distances.min(axis=1).mean() + distances.min(axis=0).mean())
    scale = max(float(np.sqrt((q * q).sum(axis=1)).mean()), 1.0)
    return float(raw / scale)


def decision_scores_from_image(image: Image.Image) -> dict[str, float]:
    points = _centers(image)
    return {"180": _score(points, 180.0), "120": _score(points, 120.0)}


def decision_from_image(image: Image.Image) -> str:
    scores = decision_scores_from_image(image)
    ordered = sorted(scores, key=scores.get)
    if scores[ordered[1]] - scores[ordered[0]] < 0.06:
        raise ValueError("rotational evidence is ambiguous")
    return ordered[0]
