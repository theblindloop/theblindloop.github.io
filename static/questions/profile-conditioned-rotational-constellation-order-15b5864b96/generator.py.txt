"""Renderer and analytic arm for rotational_constellation_order."""

from __future__ import annotations

import math
import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS = 420
COUNT = 18
QUARANTINE_GAP = 0.075


def _score(points: list[list[float]], angle_deg: float) -> float:
    p = np.asarray(points, dtype=float)
    center = p.mean(axis=0)
    q = p - center
    a = math.radians(angle_deg)
    rot = np.array([[math.cos(a), -math.sin(a)], [math.sin(a), math.cos(a)]])
    turned = q @ rot.T + center
    distances = np.sqrt(((turned[:, None, :] - p[None, :, :]) ** 2).sum(axis=2))
    # Symmetric nearest-neighbor error. True orbit members make the winning
    # score zero; the sampler rejects near-cross-symmetric constellations.
    raw = 0.5 * (distances.min(axis=1).mean() + distances.min(axis=0).mean())
    scale = max(float(np.sqrt((q * q).sum(axis=1)).mean()), 1.0)
    return float(raw / scale)


def _gap(points: list[list[float]]) -> float:
    s180 = _score(points, 180.0)
    s120 = _score(points, 120.0)
    return abs(s180 - s120)


def _separated(points: list[list[float]], clearance: float) -> bool:
    p = np.asarray(points, dtype=float)
    d = np.sqrt(((p[:, None, :] - p[None, :, :]) ** 2).sum(axis=2))
    d += np.eye(len(p)) * 1e6
    return float(d.min()) >= clearance


def _orbit_points(rng: random.Random, mode: str, center: tuple[float, float]) -> list[list[float]]:
    order = 2 if mode == "180" else 3
    orbit_count = COUNT // order
    global_turn = rng.uniform(0.0, 2.0 * math.pi)
    points: list[list[float]] = []
    for i in range(orbit_count):
        # Stratified radii avoid dense center clutter while leaving phase and
        # radius non-diagnostic of class identity.
        band = i % 3
        radius = rng.uniform(58 + 38 * band, 84 + 38 * band)
        phase = rng.uniform(0.0, 2.0 * math.pi) + global_turn
        for k in range(order):
            a = phase + 2.0 * math.pi * k / order
            points.append([
                center[0] + radius * math.cos(a),
                center[1] + radius * math.sin(a),
            ])
    rng.shuffle(points)
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    mode = "180" if int(seed) % 2 == 0 else "120"
    dot_radius = rng.choice([6, 7, 8])
    inks = ["#17233b", "#24352b", "#38252d", "#202020"]
    backgrounds = ["#ffffff", "#faf9f5", "#f7fafc"]
    for _ in range(500):
        center = (rng.uniform(192, 228), rng.uniform(192, 228))
        points = _orbit_points(rng, mode, center)
        if _separated(points, 2 * dot_radius + 7) and _gap(points) >= 0.16:
            return {
                "mode": mode,
                "points": [[round(x, 4), round(y, 4)] for x, y in points],
                "dot_radius": dot_radius,
                "ink": rng.choice(inks),
                "background": rng.choice(backgrounds),
                "canvas": CANVAS,
            }
    raise RuntimeError("could not sample a well-separated rotational constellation")


def render(scene: dict) -> Image.Image:
    mode = scene["mode"]
    if mode not in {"180", "120"}:
        raise ValueError("mode must be 180 or 120")
    points = scene["points"]
    radius = int(scene["dot_radius"])
    ink = scene["ink"]
    background = scene["background"]
    canvas = int(scene["canvas"])
    scale = 3
    image = Image.new("RGB", (canvas * scale, canvas * scale), background)
    draw = ImageDraw.Draw(image)
    for x, y in points:
        box = ((x - radius) * scale, (y - radius) * scale,
               (x + radius) * scale, (y + radius) * scale)
        draw.ellipse(box, fill=ink)
    return image.resize((canvas, canvas), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return scene["mode"]


def margin(scene: dict) -> float:
    return round(_gap(scene["points"]), 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
