"""Deterministic renderer for ordered cyclic row/column token shifts."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1100, 690
BACKGROUND = (246, 248, 252)
PAPER = (255, 255, 255)
INK = (30, 40, 54)
MUTED = (98, 111, 132)
CARD_BORDER = (174, 184, 199)
START_BG = (237, 241, 247)
MINI_BG = (221, 227, 236)
TOKEN = (91, 103, 121)
PALE_TOKEN = (190, 198, 210)
LABELS = tuple("ABCDEFGHI")

# Ring and operation-band colours change together, but their meaning is carried
# by geometry and panel role rather than by a particular RGB name.
PALETTES = (
    ((218, 39, 82), (38, 126, 218)),
    ((135, 52, 205), (232, 112, 25)),
    ((13, 151, 137), (204, 48, 151)),
)


def _font(size: int, bold: bool = True) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _apply_one(cell: int, operation: dict[str, Any]) -> int:
    row, col = divmod(int(cell), 3)
    axis = str(operation["axis"])
    index = int(operation["index"])
    if axis == "row" and row == index:
        col = (col + 1) % 3
    elif axis == "col" and col == index:
        row = (row + 1) % 3
    return row * 3 + col


def _replay(scene: dict[str, Any]) -> int:
    cell = int(scene["initial_cell"])
    for operation in scene["operations"]:
        cell = _apply_one(cell, operation)
    return cell


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, fully visible six-card permutation program."""
    seed = int(seed)
    rng = random.Random(seed)
    operations = [
        *({"axis": "row", "index": index} for index in range(3)),
        *({"axis": "col", "index": index} for index in range(3)),
    ]
    rng.shuffle(operations)

    # The composition is bijective. Choose the unique starting cell that maps
    # to seed mod 9, which gives exact answer balance over consecutive seeds.
    desired = seed % 9
    initial = next(
        cell
        for cell in range(9)
        if _replay({"initial_cell": cell, "operations": operations}) == desired
    )
    return {
        "initial_cell": initial,
        "operations": operations,
        "palette_variant": rng.randrange(len(PALETTES)),
        "layout_dx": rng.randint(-12, 12),
        "layout_dy": rng.randint(-7, 7),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the row-major label reached by the marked token."""
    return LABELS[_replay(scene)]


def margin(scene: dict[str, Any]) -> float:
    """One-sided pixel margin to a neighboring cell/index interpretation."""
    # Cell centers are 76 px apart in the start grid and 28 px apart in cards.
    # A band center is therefore 14 px from the closest index boundary; the
    # 10 px threshold leaves four pixels of certified headroom.
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _centered_text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[float, float],
    text: str,
    font: Any,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    axis: str,
) -> None:
    x, y = center
    if axis == "row":
        draw.line((x - 20, y, x + 17, y), fill=INK, width=5)
        draw.polygon(((x + 25, y), (x + 12, y - 9), (x + 12, y + 9)), fill=INK)
    else:
        draw.line((x, y - 20, x, y + 17), fill=INK, width=5)
        draw.polygon(((x, y + 25), (x - 9, y + 12), (x + 9, y + 12)), fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the public state program; every scene field affects pixels."""
    initial = int(scene["initial_cell"])
    operations = [dict(operation) for operation in scene["operations"]]
    palette_variant = int(scene["palette_variant"])
    dx = int(scene["layout_dx"])
    dy = int(scene["layout_dy"])
    ring_color, band_color = PALETTES[palette_variant]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        (22, 17, WIDTH - 22, HEIGHT - 17),
        radius=24,
        fill=PAPER,
        outline=(220, 226, 235),
        width=2,
    )
    draw.text((50, 36), "GRID-SHIFT TOKEN", font=_font(24), fill=INK)
    draw.text(
        (50, 70),
        "Track the ringed token. Horizontal bands shift right; vertical bands shift down.",
        font=_font(16, False),
        fill=MUTED,
    )

    # Large initial 3x3 grid.
    grid_size = 228
    cell = grid_size // 3
    gx = WIDTH // 2 - grid_size // 2 + dx
    gy = 116 + dy
    draw.text((gx - 118, gy + 91), "START", font=_font(20), fill=INK)
    draw.rectangle((gx, gy, gx + grid_size, gy + grid_size), fill=START_BG, outline=INK, width=4)
    for step in (1, 2):
        draw.line((gx + step * cell, gy, gx + step * cell, gy + grid_size), fill=INK, width=3)
        draw.line((gx, gy + step * cell, gx + grid_size, gy + step * cell), fill=INK, width=3)
    for index, label in enumerate(LABELS):
        row, col = divmod(index, 3)
        cx = gx + col * cell + cell // 2
        cy = gy + row * cell + cell // 2
        _centered_text(draw, (gx + col * cell + 14, gy + row * cell + 14), label, _font(14), MUTED)
        radius = 12
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=TOKEN if index == initial else PALE_TOKEN)
        if index == initial:
            ring_r = 28
            draw.ellipse((cx - ring_r, cy - ring_r, cx + ring_r, cy + ring_r), outline=ring_color, width=7)

    draw.line((50, 374, WIDTH - 50, 374), fill=(220, 226, 235), width=2)
    draw.text((50, 389), "SHIFT CARDS — READ LEFT TO RIGHT", font=_font(20), fill=INK)

    card_w, card_h, gap = 144, 211, 24
    total_w = 6 * card_w + 5 * gap
    start_x = (WIDTH - total_w) // 2 + dx
    top = 430 + dy
    mini = 84
    mini_cell = mini // 3
    for step, operation in enumerate(operations):
        left = start_x + step * (card_w + gap)
        right, bottom = left + card_w, top + card_h
        draw.rounded_rectangle(
            (left, top, right, bottom),
            radius=15,
            fill=(249, 251, 253),
            outline=CARD_BORDER,
            width=3,
        )
        _centered_text(draw, ((left + right) / 2, top + 21), str(step + 1), _font(14), MUTED)
        mx = left + (card_w - mini) // 2
        my = top + 43
        draw.rectangle((mx, my, mx + mini, my + mini), fill=MINI_BG)
        axis = str(operation["axis"])
        index = int(operation["index"])
        if axis == "row":
            y0 = my + index * mini_cell + 4
            draw.rectangle((mx + 2, y0, mx + mini - 2, y0 + mini_cell - 8), fill=band_color)
        elif axis == "col":
            x0 = mx + index * mini_cell + 4
            draw.rectangle((x0, my + 2, x0 + mini_cell - 8, my + mini - 2), fill=band_color)
        else:
            raise ValueError(f"unknown axis {axis!r}")
        draw.rectangle((mx, my, mx + mini, my + mini), outline=INK, width=3)
        for split in (1, 2):
            draw.line((mx + split * mini_cell, my, mx + split * mini_cell, my + mini), fill=INK, width=2)
            draw.line((mx, my + split * mini_cell, mx + mini, my + split * mini_cell), fill=INK, width=2)
        _draw_arrow(draw, ((left + right) // 2, top + 161), axis)
        _centered_text(
            draw,
            ((left + right) / 2, top + 193),
            "ROW RIGHT" if axis == "row" else "COLUMN DOWN",
            _font(11),
            MUTED,
        )
        if step < 5:
            arrow_x = right + gap // 2
            draw.polygon(((arrow_x - 5, top + 104 - 7), (arrow_x + 7, top + 104), (arrow_x - 5, top + 104 + 7)), fill=(139, 151, 169))
    return image
