from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray):
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        todo = [(y, x)]
        seen[y, x] = True
        points = []
        while todo:
            cy, cx = todo.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    todo.append((ny, nx))
        result.append(points)
    return result


def _identity_from_pips(array: np.ndarray, cx: float, cy: float) -> str:
    yy, xx = np.ogrid[:array.shape[0], :array.shape[1]]
    core = (xx - cx) ** 2 + (yy - cy) ** 2 <= 16 ** 2
    dark = (array[:, :, 0] < 65) & (array[:, :, 1] < 75) & (array[:, :, 2] < 90) & core
    blobs = []
    for points in _components(dark):
        if 15 <= len(points) <= 80:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            if max(xs) - min(xs) <= 9 and max(ys) - min(ys) <= 9:
                blobs.append(points)
    count = len(blobs)
    if count not in (1, 2, 3):
        raise ValueError("pip identity is not uniquely readable")
    return "ABC"[count - 1]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    # Bead interiors are large, saturated connected regions; identity comes only
    # from their one/two/three dark pips, not from renderer state or palette.
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    saturated = (high - low > 55) & (high > 120) & (low < 205)
    bead_centers = []
    for points in _components(saturated):
        if not 750 <= len(points) <= 1700:
            continue
        ys = np.array([p[0] for p in points])
        xs = np.array([p[1] for p in points])
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        if 35 <= width <= 45 and 35 <= height <= 45:
            bead_centers.append((float(xs.mean()), float(ys.mean())))
    if len(bead_centers) != 11:
        raise ValueError(f"expected 11 readable beads, found {len(bead_centers)}")
    bead_centers.sort()
    identities = [_identity_from_pips(array, x, y) for x, y in bead_centers]
    if sorted(identities[:3]) != ["A", "B", "C"]:
        raise ValueError("START is not a visible identity permutation")

    # Validate the connected evidence path relative to detected bead geometry.
    median_y = int(round(float(np.median([y for _, y in bead_centers]))))
    x0 = max(0, int(bead_centers[0][0]) - 32)
    x1 = min(array.shape[1], int(bead_centers[-1][0]) + 70)
    y0 = max(0, median_y - 8)
    y1 = min(array.shape[0], median_y + 9)
    band = array[y0:y1, x0:x1]
    nonbackground = np.linalg.norm(band.astype(np.int16) - np.array([248, 247, 242]), axis=2) > 45
    covered_columns = np.mean(nonbackground.any(axis=0))
    if covered_columns < 0.94:
        raise ValueError("connected wire is broken")

    order = identities[:3]
    for item in identities[3:]:
        order.remove(item)
        order.append(item)
    return order[0]
