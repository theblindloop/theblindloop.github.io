from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1024, 384
BACKGROUND = (248, 247, 242)
INK = (27, 39, 54)
GOLD = (225, 167, 42)
PALETTES = (
    ((31, 174, 181), (231, 103, 88), (113, 91, 181)),
    ((39, 152, 210), (232, 127, 44), (87, 171, 105)),
    ((194, 82, 132), (36, 164, 148), (226, 147, 36)),
)


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _replay(initial_order: list[str], touches: list[str]) -> list[str]:
    order = list(initial_order)
    for item in touches:
        order.remove(item)
        order.append(item)
    return order


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    identities = ["A", "B", "C"]
    initial = identities[:]
    rng.shuffle(initial)
    target = identities[int(seed) % 3]
    # Rejection sampling balances final identity while retaining varied histories.
    for _ in range(20000):
        touches = [rng.choice(identities) for _ in range(8)]
        if len(set(touches)) != 3:
            continue
        if _replay(initial, touches)[0] != target:
            continue
        # Avoid degenerate monochrome tails and make each identity recur.
        if min(touches.count(x) for x in identities) < 2:
            continue
        break
    else:
        raise RuntimeError("could not sample a balanced touch history")
    base = list(PALETTES[rng.randrange(len(PALETTES))])
    rng.shuffle(base)
    palette = {identity: list(base[i]) for i, identity in enumerate(identities)}
    return {
        "initial_order": initial,
        "touches": touches,
        "palette": palette,
        "y_offset": rng.randint(-14, 14),
    }


def _bead(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: str,
          palette: dict, start: bool) -> None:
    x, y = center
    if start:
        draw.ellipse((x - 29, y - 29, x + 29, y + 29), fill=GOLD)
    draw.ellipse((x - 25, y - 25, x + 25, y + 25), fill=INK)
    color = tuple(palette[identity])
    draw.ellipse((x - 20, y - 20, x + 20, y + 20), fill=color)
    pip_positions = {
        "A": [(0, 0)],
        "B": [(-7, 0), (7, 0)],
        "C": [(-9, 5), (0, -7), (9, 5)],
    }
    for dx, dy in pip_positions[identity]:
        draw.ellipse((x + dx - 3, y + dy - 3, x + dx + 3, y + dy + 3), fill=INK)


def render(scene: dict) -> Image.Image:
    initial = list(scene["initial_order"])
    touches = list(scene["touches"])
    palette = dict(scene["palette"])
    y_offset = int(scene["y_offset"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    y = 205 + y_offset
    xs = [76 + 82 * i for i in range(11)]

    draw.text((42, 28), "RECENCY TOUCH WIRE", fill=INK, font=_font(28, True))
    draw.text((42, 67), "gold beads: OLDEST  →  NEWEST", fill=INK, font=_font(20, True))
    draw.text((470, 67), "then TOUCH each bead →", fill=INK, font=_font(20, True))

    # A single unbroken trace is laid down first, so every bead belongs to it.
    draw.line((45, y, 954, y), fill=INK, width=7)
    draw.ellipse((37, y - 8, 53, y + 8), fill=GOLD, outline=INK, width=3)
    draw.polygon([(954, y - 15), (986, y), (954, y + 15)], fill=INK)

    sequence = initial + touches
    for index, identity in enumerate(sequence):
        _bead(draw, (xs[index], y), identity, palette, start=index < 3)
        if index == 2:
            draw.line((xs[index] + 35, y - 40, xs[index] + 35, y + 40), fill=GOLD, width=4)

    draw.text((44, y + 55), "START order", fill=INK, font=_font(18, True))
    draw.text((354, y + 55), "touch history", fill=INK, font=_font(18, True))
    draw.text((42, 326), "Identity code:  A = •     B = ••     C = •••", fill=INK, font=_font(21, True))
    return image


def analytic_gold(scene: dict) -> str:
    initial = list(scene["initial_order"])
    touches = list(scene["touches"])
    return _replay(initial, touches)[0]


def margin(scene: dict) -> float:
    # Discrete pip counts and a unique permutation state provide a one-class gap.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        initial = list(scene["initial_order"])
        touches = list(scene["touches"])
        return sorted(initial) != ["A", "B", "C"] or len(touches) != 8 or any(
            item not in {"A", "B", "C"} for item in touches
        )
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
