from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))]); seen[y, x] = True
        xs, ys = [], []
        while q:
            yy, xx = q.popleft(); xs.append(xx); ys.append(yy)
            for dy, dx in ((1,0),(-1,0),(0,1),(0,-1)):
                ny, nx = yy+dy, xx+dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; q.append((ny, nx))
        out.append((len(xs), min(xs), min(ys), max(xs), max(ys)))
    return out


def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"))
    # Dark frame pixels; antialiasing is safely excluded by a strict threshold.
    dark = (a[:,:,0] < 90) & (a[:,:,1] < 90) & (a[:,:,2] < 100)
    ys, xs = np.nonzero(dark)
    if len(xs) < 500:
        raise ValueError("sampling frames absent")
    # Two frames are separated by the canvas midpoint. Their inner dimensions
    # are recovered from the extreme dark border coordinates.
    mid = a.shape[1] // 2
    boxes = []
    for side in (xs < mid, xs > mid):
        sx, sy = xs[side], ys[side]
        if len(sx) < 200:
            raise ValueError("one sampling frame absent")
        boxes.append((int(sx.min()), int(sy.min()), int(sx.max()), int(sy.max())))

    blue = (a[:,:,2] > 135) & (a[:,:,0] < 90) & (a[:,:,1] > 65) & (a[:,:,1] < 175)
    surface = (a[:,:,0] > 170) & (a[:,:,0] < 235) & (a[:,:,1] > 200) & (a[:,:,2] > 190) & (a[:,:,2] < 245)
    comps = [c for c in _components(blue) if 55 <= c[0] <= 180 and 7 <= c[3]-c[1] <= 16 and 7 <= c[4]-c[2] <= 16]
    counts = [0, 0]
    for size, x0, y0, x1, y1 in comps:
        cx, cy = (x0+x1)/2, (y0+y1)/2
        for i, (bx0, by0, bx1, by1) in enumerate(boxes):
            if bx0 < cx < bx1 and by0 < cy < by1:
                counts[i] += 1
    if min(counts) < 1:
        raise ValueError("point evidence absent")
    areas = []
    for bx0, by0, bx1, by1 in boxes:
        # Require an intact visible sampling surface. A removed raster tile in
        # either remote operand causes abstention rather than silently turning
        # into a smaller valid window.
        interior = (surface | blue)[by0+4:by1-3, bx0+4:bx1-3]
        if interior.size == 0 or float(interior.mean()) < 0.92:
            raise ValueError("sampling surface incomplete")
        areas.append((bx1-bx0-7) * (by1-by0-7))
    delta = counts[0] / areas[0] - counts[1] / areas[1]
    if abs(delta) < 0.00012:
        raise ValueError("density comparison ambiguous")
    return "left" if delta > 0 else "right"
