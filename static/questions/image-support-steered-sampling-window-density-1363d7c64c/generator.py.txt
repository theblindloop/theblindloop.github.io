from __future__ import annotations

import random
from PIL import Image, ImageDraw

BG = (250, 250, 247)
INK = (35, 39, 47)
BLUE = (24, 116, 205)
GOLD = (232, 174, 36)
SURFACE = (207, 232, 220)
SCALE = 3


def _area(box: list[int]) -> int:
    # Four-pixel border on each side leaves this visible interior area.
    return (box[2] - box[0] - 8) * (box[3] - box[1] - 8)


def _density_cross(scene: dict) -> int:
    return len(scene["left_points"]) * _area(scene["right_box"]) - len(scene["right_points"]) * _area(scene["left_box"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    widths = [108, 126, 144]
    heights = [108, 126, 144]
    # Rejection sampling balances side and forces a clear density gap. It also
    # often makes the higher raw count have the lower density.
    for _ in range(10000):
        lw, lh = rng.choice(widths), rng.choice(heights)
        rw, rh = rng.choice(widths), rng.choice(heights)
        la, ra = lw * lh, rw * rh
        ln, rn = rng.randint(5, 13), rng.randint(5, 13)
        cross = ln * ra - rn * la
        if abs(cross) < 2.0 * la * ra / 10000.0:
            continue
        target = "left" if seed % 2 == 0 else "right"
        if (cross > 0) != (target == "left"):
            ln, rn = rn, ln
            cross = ln * ra - rn * la
        if (cross > 0) != (target == "left"):
            continue
        if abs(cross) < 2.0 * la * ra / 10000.0:
            continue
        # Ensure count alone is misleading or tied for roughly half the scenes.
        if seed % 3 != 0 and ((ln > rn) == (cross > 0)) and ln != rn:
            continue
        break
    else:
        raise RuntimeError("could not sample robust density comparison")

    ly = 184 + rng.randint(-12, 12)
    ry = 184 + rng.randint(-12, 12)
    left_box = [30 + rng.randint(-3, 3), ly, 30 + lw, ly + lh]
    right_box = [482 - rw + rng.randint(-3, 3), ry, 482 + rng.randint(-3, 3), ry + rh]
    # Anchor the right edge after harmless position jitter.
    left_box[2] = left_box[0] + lw
    right_box[2] = right_box[0] + rw

    def place(box: list[int], n: int) -> list[list[int]]:
        points: list[list[int]] = []
        for _ in range(n):
            for _attempt in range(1000):
                x = rng.randint(box[0] + 13, box[2] - 13)
                y = rng.randint(box[1] + 13, box[3] - 13)
                if all((x-px) ** 2 + (y-py) ** 2 >= 17 ** 2 for px, py in points):
                    points.append([x, y])
                    break
            else:
                raise RuntimeError("point packing failed")
        return sorted(points)

    return {
        "canvas_width": 512,
        "canvas_height": 512,
        "left_box": left_box,
        "right_box": right_box,
        "left_points": place(left_box, ln),
        "right_points": place(right_box, rn),
    }


def render(scene: dict) -> Image.Image:
    w, h = scene["canvas_width"], scene["canvas_height"]
    image = Image.new("RGB", (w * SCALE, h * SCALE), BG)
    draw = ImageDraw.Draw(image)
    # Small gold tabs make the two separate frames visually explicit without
    # encoding the answer; both sides receive identical tabs.
    for box in (scene["left_box"], scene["right_box"]):
        b = [v * SCALE for v in box]
        draw.rectangle(b, fill=SURFACE, outline=INK, width=4 * SCALE)
        cx = (box[0] + box[2]) // 2
        draw.rectangle([(cx-13)*SCALE, (box[1]-5)*SCALE, (cx+13)*SCALE, (box[1]+1)*SCALE], fill=GOLD)
    for points in (scene["left_points"], scene["right_points"]):
        for x, y in points:
            r = 6
            draw.ellipse([(x-r)*SCALE, (y-r)*SCALE, (x+r)*SCALE, (y+r)*SCALE], fill=BLUE)
    return image.resize((w, h), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    # These are exactly the scene fields that determine the declared label.
    return "left" if len(scene["left_points"]) * _area(scene["right_box"]) > len(scene["right_points"]) * _area(scene["left_box"]) else "right"


def margin(scene: dict) -> float:
    la, ra = _area(scene["left_box"]), _area(scene["right_box"])
    return abs(len(scene["left_points"]) / la - len(scene["right_points"]) / ra) * 10000.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
