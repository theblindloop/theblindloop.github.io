"""Deterministic renderer and analytic state replay for compass turn cards."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 420
BG = (248, 247, 243)
DARK = (38, 43, 52)
MUTED = (117, 124, 135)
RED = (211, 55, 48)
BLUE = (35, 105, 190)
CARD = (255, 255, 255)
DIRECTIONS = ("north", "east", "south", "west")


def sample_scene(seed: int) -> dict:
    """Sample a balanced, randomized sequence with a known final heading."""
    rng = random.Random(int(seed) ^ 0x4C52504C)
    target = int(seed) % 4
    n_turns = 3 + ((int(seed) // 4) % 3)
    turns = ["R" if rng.getrandbits(1) else "L" for _ in range(n_turns)]
    net = sum(1 if turn == "R" else -1 for turn in turns)
    initial = DIRECTIONS[(target - net) % 4]
    return {
        "initial": initial,
        "turns": turns,
        "pointer_jitter": round(rng.uniform(-8.0, 8.0), 3),
        "card_tilts": [round(rng.uniform(-2.2, 2.2), 3) for _ in turns],
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _draw_glyph(image: Image.Image, center: tuple[int, int], letter: str, tilt: float) -> None:
    glyph = Image.new("RGBA", (64, 92), (0, 0, 0, 0))
    pen = ImageDraw.Draw(glyph)
    if letter == "L":
        pen.line([(17, 14), (17, 73), (49, 73)], fill=(*BLUE, 255), width=11, joint="curve")
    else:
        pen.line([(16, 77), (16, 14), (43, 14), (50, 21), (50, 39), (43, 46), (16, 46)], fill=(*BLUE, 255), width=10, joint="curve")
        pen.line([(34, 46), (52, 77)], fill=(*BLUE, 255), width=10)
    rotated = glyph.rotate(tilt, resample=Image.Resampling.BICUBIC, expand=False)
    image.alpha_composite(rotated, (int(center[0] - 32), int(center[1] - 46)))


def render(scene: dict) -> Image.Image:
    """Render only visible evidence: the initial pointer and ordered cards."""
    initial = scene["initial"]
    turns = list(scene["turns"])
    jitter = float(scene["pointer_jitter"])
    tilts = list(scene["card_tilts"])
    image = Image.new("RGBA", (WIDTH, HEIGHT), (*BG, 255))
    draw = ImageDraw.Draw(image)

    draw.text((28, 18), "COMPASS TURN REPLAY", font=_font(25, True), fill=DARK)
    draw.text((28, 52), "Start at the red pointer. Apply the blue cards left to right.", font=_font(16), fill=MUTED)

    cx, cy, radius = 142, 227, 86
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), outline=DARK, width=5)
    draw.line((cx - radius + 12, cy, cx + radius - 12, cy), fill=MUTED, width=3)
    draw.line((cx, cy - radius + 12, cx, cy + radius - 12), fill=MUTED, width=3)
    for text, pos in (("N", (cx, cy - radius - 20)), ("E", (cx + radius + 20, cy)), ("S", (cx, cy + radius + 20)), ("W", (cx - radius - 20, cy))):
        _centered(draw, pos, text, _font(18, True), DARK)

    base_angle = {"east": 0.0, "south": 90.0, "west": 180.0, "north": 270.0}[initial]
    angle = math.radians(base_angle + jitter)
    ux, uy = math.cos(angle), math.sin(angle)
    tip = (cx + 66 * ux, cy + 66 * uy)
    draw.line((cx, cy, tip[0], tip[1]), fill=RED, width=13)
    px, py = -uy, ux
    triangle = [
        (cx + 77 * ux, cy + 77 * uy),
        (cx + 54 * ux + 12 * px, cy + 54 * uy + 12 * py),
        (cx + 54 * ux - 12 * px, cy + 54 * uy - 12 * py),
    ]
    draw.polygon(triangle, fill=RED)
    draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=RED)

    draw.text((270, 102), "TURN CARDS", font=_font(18, True), fill=DARK)
    draw.text((270, 128), "L = 90° left     R = 90° right", font=_font(15), fill=MUTED)
    n = len(turns)
    left, right = 285, 680
    xs = [left + (right - left) * (i + 0.5) / n for i in range(n)]
    card_y = 239
    for i, (x, turn) in enumerate(zip(xs, turns, strict=True)):
        draw.rounded_rectangle((x - 34, card_y - 61, x + 34, card_y + 61), radius=10, fill=CARD, outline=DARK, width=3)
        _draw_glyph(image, (round(x), card_y), turn, float(tilts[i]))
        if i + 1 < n:
            nx = xs[i + 1]
            mid = (x + nx) / 2
            draw.line((x + 39, card_y, nx - 42, card_y), fill=MUTED, width=3)
            draw.polygon([(nx - 42, card_y), (nx - 50, card_y - 6), (nx - 50, card_y + 6)], fill=MUTED)
    draw.text((270, 326), "No final pointer is shown — replay every card.", font=_font(15, True), fill=DARK)
    return image.convert("RGB")


def analytic_gold(scene: dict) -> str:
    index = DIRECTIONS.index(scene["initial"])
    for turn in scene["turns"]:
        index = (index + (1 if turn == "R" else -1)) % 4
    return DIRECTIONS[index]


def margin(scene: dict) -> float:
    pointer_clearance = 45.0 - abs(float(scene["pointer_jitter"]))
    glyph_clearance = 20.0 - max(abs(float(value)) for value in scene["card_tilts"])
    return round(min(pointer_clearance, glyph_clearance), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
