"""Independent image-only inverse arm for compass turn replay."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_DIRS = ("north", "east", "south", "west")


class AmbiguousImage(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 8:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _compass_center(arr: np.ndarray) -> tuple[float, float]:
    # Neutral dark ink is intentionally separate from the red pointer and blue cards.
    dark = (arr[..., 0] < 85) & (arr[..., 1] < 90) & (arr[..., 2] < 100)
    candidates = []
    for pts in _components(dark):
        ys, xs = pts[:, 0], pts[:, 1]
        bw, bh = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
        if bw > 120 and bh > 120:
            candidates.append((len(pts), (float(xs.min() + xs.max()) / 2, float(ys.min() + ys.max()) / 2)))
    if len(candidates) != 1:
        raise AmbiguousImage("could not isolate one compass frame")
    return max(candidates)[1]


def _initial_heading(arr: np.ndarray, center: tuple[float, float]) -> int:
    red = (arr[..., 0] > 155) & (arr[..., 1] < 115) & (arr[..., 2] < 105)
    ys, xs = np.nonzero(red)
    if len(xs) < 250:
        raise AmbiguousImage("red pointer is missing or too small")
    dx, dy = float(xs.mean()) - center[0], float(ys.mean()) - center[1]
    if max(abs(dx), abs(dy)) < 12 or abs(abs(dx) - abs(dy)) < 6:
        raise AmbiguousImage("pointer is too close to a directional boundary")
    if abs(dx) > abs(dy):
        return 1 if dx > 0 else 3
    return 2 if dy > 0 else 0


def _turn_cards(arr: np.ndarray) -> list[str]:
    blue = (arr[..., 2] > 125) & (arr[..., 0] < 95) & (arr[..., 1] < 155)
    glyphs = []
    for pts in _components(blue):
        ys, xs = pts[:, 0], pts[:, 1]
        bw, bh = int(xs.max() - xs.min() + 1), int(ys.max() - ys.min() + 1)
        if len(pts) < 180 or bw < 20 or bh < 42:
            continue
        x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
        # R has substantial ink in its upper-right quadrant; L has none there.
        upper_right = np.count_nonzero(blue[y0 : y0 + max(1, int(0.52 * (y1 - y0 + 1))), x0 + int(0.55 * (x1 - x0 + 1)) : x1 + 1])
        ratio = upper_right / len(pts)
        if ratio > 0.11:
            letter = "R"
        elif ratio < 0.055:
            letter = "L"
        else:
            raise AmbiguousImage("turn glyph has insufficient L/R margin")
        glyphs.append((float(xs.mean()), letter))
    glyphs.sort()
    if not 3 <= len(glyphs) <= 5:
        raise AmbiguousImage(f"expected three to five separated turn cards, found {len(glyphs)}")
    return [letter for _, letter in glyphs]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or min(arr.shape[:2]) < 100:
        raise AmbiguousImage("unsupported raster")
    center = _compass_center(arr)
    state = _initial_heading(arr, center)
    for turn in _turn_cards(arr):
        state = (state + (1 if turn == "R" else -1)) % 4
    return _DIRS[state]
