"""Cord-walk turn echo: one thin staircase cord whose two stretches carry the
same left/right turn word up to one disagreement.

A single slate cord runs from a black square CAP, through a black square RELAY
that sits on it, out to a plain end. The stretch before the relay and the
stretch after it have the same number of corners, and the list of left/right
turns taken while walking cap -> relay equals the list taken while walking
relay -> plain end at every corner but one. A coloured bead lies under the cord
at each corner of the second stretch; the answer is the bead colour at the one
corner where the second turn list disagrees with the first.

The two stretches leave the relay in different absolute directions, so the
comparison is intrinsic (turns relative to the direction of travel) and cannot
be done by overlaying two visually aligned rows.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "cordwalk_turn_echo_bind-renderer-0.1.0"

CANVAS = 480
QUARANTINE_MARGIN = 18.0

CREAM = (247, 243, 232)
SLATE = (84, 94, 110)
BLACK = (18, 18, 20)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (208, 54, 46),
    "orange": (226, 124, 32),
    "yellow": (222, 190, 40),
    "green": (48, 150, 66),
    "blue": (44, 96, 196),
    "purple": (140, 62, 180),
}
COLOUR_NAMES = tuple(sorted(PALETTE))

CORD_W = 5
BEAD = 15
CAP = 13
RELAY = 13
TURNS = 4

_DIRS = ((1, 0), (0, 1), (-1, 0), (0, -1))  # E, S, W, N in image coordinates


def _turn(direction: int, letter: str) -> int:
    return (direction + 1) % 4 if letter == "R" else (direction - 1) % 4


def _lattice(scene: dict) -> dict:
    """Return lattice geometry: polyline vertices, relay point, bead corners."""
    d = int(scene["start_dir"])
    x = y = 0
    pts = [(x, y)]
    visited = [(x, y)]

    def walk(n: int) -> None:
        nonlocal x, y
        dx, dy = _DIRS[d]
        for _ in range(n):
            x += dx
            y += dy
            visited.append((x, y))
        pts.append((x, y))

    runs_a = scene["runs_a"]
    runs_b = scene["runs_b"]
    wa = scene["wa"]
    wb = scene["wb"]

    for i in range(TURNS):
        walk(runs_a[i])
        d = _turn(d, wa[i])
    walk(scene["la_end"])
    # the relay sits on the straight run A4 -> B1, la_end units after A4
    relay_vertex_index = len(pts) - 1
    relay_pt = pts[-1]
    walk(scene["lb_start"])
    for i in range(TURNS):
        d = _turn(d, wb[i])
        walk(runs_b[i])
    # merge the collinear vertex created at the relay
    merged = pts[:relay_vertex_index] + pts[relay_vertex_index + 1 :]
    bead_pts = merged[relay_vertex_index + 0 : relay_vertex_index + TURNS]
    return {
        "vertices": merged,
        "relay": relay_pt,
        "beads": bead_pts,
        "visited": visited,
        "cap": pts[0],
    }


def _px(scene: dict, p: tuple[int, int]) -> tuple[int, int]:
    return (scene["ox"] + p[0] * scene["unit"], scene["oy"] + p[1] * scene["unit"])


def _self_avoiding(visited: list[tuple[int, int]]) -> bool:
    seen: dict[tuple[int, int], int] = {}
    for i, p in enumerate(visited):
        for q, j in seen.items():
            if abs(q[0] - p[0]) <= 1 and abs(q[1] - p[1]) <= 1 and i - j > 2:
                return False
        seen[p] = i
    return True


def _cell_footprint(scene: dict, geom: dict, cell: int = 60) -> set[tuple[int, int]]:
    """Coarse cells (default 8x8 grid) touched by the drawn cord structure."""
    cells: set[tuple[int, int]] = set()
    verts = [_px(scene, v) for v in geom["vertices"]]
    for a, b in zip(verts, verts[1:]):
        steps = max(abs(b[0] - a[0]), abs(b[1] - a[1]))
        for s in range(steps + 1):
            x = a[0] + (b[0] - a[0]) * s // max(steps, 1)
            y = a[1] + (b[1] - a[1]) * s // max(steps, 1)
            cells.add((x // cell, y // cell))
    return cells


def _footprint_ok(scene: dict, geom: dict, strict: bool = True) -> bool:
    """Keep the decisive support one elongated, sparsely filled chain of cells."""
    cells = _cell_footprint(scene, geom)
    xs = [c[0] for c in cells]
    ys = [c[1] for c in cells]
    w = max(xs) - min(xs) + 1
    h = max(ys) - min(ys) + 1
    if not (4 <= max(w, h) <= 6 and 2 <= min(w, h) <= 4):
        return False
    if max(w, h) < (1.4 if strict else 1.2) * min(w, h):
        return False
    return len(cells) <= (0.46 if strict else 0.56) * w * h


def _run_lengths(scene: dict) -> list[int]:
    return (
        list(scene["runs_a"])
        + [scene["la_end"], scene["lb_start"]]
        + list(scene["runs_b"])
    )


def _forbidden_mask(scene: dict, geom: dict) -> list[list[bool]]:
    """Coarse 8-px occupancy grid marking where decoration may not go."""
    cells = CANVAS // 8
    grid = [[False] * cells for _ in range(cells)]

    def block(cx: int, cy: int, r: int) -> None:
        for gy in range(max(0, (cy - r) // 8), min(cells - 1, (cy + r) // 8) + 1):
            for gx in range(max(0, (cx - r) // 8), min(cells - 1, (cx + r) // 8) + 1):
                grid[gy][gx] = True

    verts = [_px(scene, v) for v in geom["vertices"]]
    for a, b in zip(verts, verts[1:]):
        steps = max(abs(b[0] - a[0]), abs(b[1] - a[1])) // 4 + 1
        for s in range(steps + 1):
            cx = a[0] + (b[0] - a[0]) * s // steps
            cy = a[1] + (b[1] - a[1]) * s // steps
            block(cx, cy, 14)
    for b in geom["beads"]:
        bx, by = _px(scene, b)
        block(bx, by, 28)
    for key in ("relay", "cap"):
        cx, cy = _px(scene, geom[key])
        block(cx, cy, 22)
    return grid


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed * 7919 + 13)
    boundary = (seed % 7) == 3
    target_colour = COLOUR_NAMES[seed % len(COLOUR_NAMES)]

    for attempt in range(6000):
        strict = attempt < 3500
        if boundary:
            unit = rng.randint(16, 17)
            runs_a = [rng.randint(2, 3) for _ in range(TURNS)]
            runs_b = [rng.randint(2, 3) for _ in range(TURNS)]
            la_end = rng.randint(2, 3)
            lb_start = rng.randint(2, 3)
            runs_a[rng.randrange(TURNS)] = 1
        else:
            unit = rng.randint(24, 30)
            runs_a = [rng.randint(1, 2) for _ in range(TURNS)]
            runs_b = [rng.randint(2, 3) for _ in range(TURNS)]
            la_end = rng.randint(2, 3)
            lb_start = rng.randint(2, 3)

        wa = [rng.choice("LR") for _ in range(TURNS)]
        m = rng.randrange(TURNS)
        wb = list(wa)
        wb[m] = "L" if wa[m] == "R" else "R"

        colours = [target_colour]
        pool = [c for c in COLOUR_NAMES if c != target_colour]
        rng.shuffle(pool)
        colours.extend(pool[: TURNS - 1])
        rng.shuffle(colours)
        # place the target colour at the disagreeing corner
        swap_at = colours.index(target_colour)
        colours[swap_at], colours[m] = colours[m], colours[swap_at]

        scene: dict[str, Any] = {
            "unit": unit,
            "ox": 0,
            "oy": 0,
            "start_dir": rng.randrange(4),
            "runs_a": runs_a,
            "runs_b": runs_b,
            "la_end": la_end,
            "lb_start": lb_start,
            "wa": wa,
            "wb": wb,
            "m": m,
            "colours": colours,
            "decor": [],
        }
        # beads must never crowd each other: keep every bead-bearing run wide
        if min(runs_b) * unit < 30:
            continue
        geom = _lattice(scene)
        if not _self_avoiding(geom["visited"]):
            continue

        xs = [p[0] for p in geom["visited"]]
        ys = [p[1] for p in geom["visited"]]
        w = (max(xs) - min(xs)) * unit
        h = (max(ys) - min(ys)) * unit
        if not (115 <= min(w, h) and max(w, h) <= 275):
            continue
        pad = 46
        if w + 2 * pad > CANVAS or h + 2 * pad > CANVAS:
            continue
        slack_x = CANVAS - 2 * pad - w
        slack_y = CANVAS - 2 * pad - h
        scene["ox"] = pad + rng.randint(0, slack_x) - min(xs) * unit
        scene["oy"] = pad + rng.randint(0, slack_y) - min(ys) * unit

        geom = _lattice(scene)
        if not _footprint_ok(scene, geom, strict):
            continue
        scene["decor"] = _sample_decor(rng, scene, geom)
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


def _sample_decor(rng: random.Random, scene: dict, geom: dict) -> list[list[Any]]:
    grid = _forbidden_mask(scene, geom)
    cells = CANVAS // 8
    wanted: list[str] = []
    for name in COLOUR_NAMES:
        wanted.extend([name] * 3)
    wanted.extend(["slate"] * 4)
    wanted.extend(["black"] * 4)
    rng.shuffle(wanted)

    placed: list[list[Any]] = []
    for name in wanted:
        for _ in range(120):
            size = rng.randint(7, 10)
            cx = rng.randint(14, CANVAS - 15)
            cy = rng.randint(14, CANVAS - 15)
            gx0 = max(0, (cx - size) // 8)
            gx1 = min(cells - 1, (cx + size) // 8)
            gy0 = max(0, (cy - size) // 8)
            gy1 = min(cells - 1, (cy + size) // 8)
            if any(grid[gy][gx] for gy in range(gy0, gy1 + 1) for gx in range(gx0, gx1 + 1)):
                continue
            clash = False
            for px, py, ps, _n in placed:
                if abs(px - cx) < size + ps + 6 and abs(py - cy) < size + ps + 6:
                    clash = True
                    break
            if clash:
                continue
            placed.append([cx, cy, size, name])
            break
    return placed


def _decor_colour(name: str) -> tuple[int, int, int]:
    if name == "slate":
        return SLATE
    if name == "black":
        return BLACK
    return PALETTE[name]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), CREAM)
    draw = ImageDraw.Draw(img)
    geom = _lattice(scene)

    for cx, cy, size, name in scene["decor"]:
        h = size // 2
        draw.rectangle([cx - h, cy - h, cx + h, cy + h], fill=_decor_colour(name))

    h = BEAD // 2
    for bead, name in zip(geom["beads"], scene["colours"]):
        bx, by = _px(scene, bead)
        draw.rectangle([bx - h, by - h, bx + h, by + h], fill=PALETTE[name])

    hw = CORD_W // 2
    verts = [_px(scene, v) for v in geom["vertices"]]
    for a, b in zip(verts, verts[1:]):
        x0, x1 = sorted((a[0], b[0]))
        y0, y1 = sorted((a[1], b[1]))
        draw.rectangle([x0 - hw, y0 - hw, x1 + hw, y1 + hw], fill=SLATE)

    rx, ry = _px(scene, geom["relay"])
    hr = RELAY // 2
    draw.rectangle([rx - hr, ry - hr, rx + hr, ry + hr], fill=BLACK)

    cx, cy = _px(scene, geom["cap"])
    hc = CAP // 2
    draw.rectangle([cx - hc, cy - hc, cx + hc, cy + hc], fill=BLACK)
    return img


def echo_break_index(scene: dict) -> int:
    """Index of the single corner at which the two rendered turn words differ."""
    diffs = [i for i, (u, v) in enumerate(zip(scene["wa"], scene["wb"])) if u != v]
    if len(diffs) != 1:
        raise ValueError("the two turn words must disagree at exactly one corner")
    return diffs[0]


def analytic_gold(scene: dict) -> str:
    return str(scene["colours"][echo_break_index(scene)])


def margin(scene: dict) -> float:
    return float(min(_run_lengths(scene)) * scene["unit"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    out: list[tuple[str, dict]] = []
    decor = list(scene["decor"])
    if len(decor) > 1:
        rotated = dict(scene)
        rotated["decor"] = decor[1:] + decor[:1]
        out.append(("decor_list_rotation", rotated))
        reversed_scene = dict(scene)
        reversed_scene["decor"] = list(reversed(decor))
        out.append(("decor_list_reversal", reversed_scene))
    return out
