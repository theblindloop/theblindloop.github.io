"""Independent pixel-only inverse arm for the removable-seal question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 62, 62), dtype=np.int16),
    "green": np.array((42, 166, 91), dtype=np.int16),
    "blue": np.array((55, 112, 210), dtype=np.int16),
}
YELLOW = np.array((242, 190, 38), dtype=np.int16)


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr - target), axis=2) <= tolerance


def _flood_reaches_border(open_floor: np.ndarray, start: tuple[int, int]) -> bool:
    height, width = open_floor.shape
    sx, sy = start
    if not (0 <= sx < width and 0 <= sy < height and open_floor[sy, sx]):
        return False
    seen = np.zeros(open_floor.shape, dtype=bool)
    seen[sy, sx] = True
    pending = deque([(sx, sy)])
    while pending:
        x, y = pending.popleft()
        if x == 0 or y == 0 or x == width - 1 or y == height - 1:
            return True
        for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if 0 <= nx < width and 0 <= ny < height and open_floor[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                pending.append((nx, ny))
    return False


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    if arr.shape[:2] != (480, 640):
        raise ValueError(f"unexpected image dimensions {arr.shape[:2]}")

    marker_mask = _color_mask(arr, YELLOW)
    ys, xs = np.nonzero(marker_mask)
    if not 250 <= len(xs) <= 700:
        raise ValueError("yellow marker is missing or malformed")
    start = (int(round(xs.mean())), int(round(ys.mean())))

    seal_masks = {}
    for name, rgb in PALETTE.items():
        mask = _color_mask(arr, rgb)
        if not 700 <= int(mask.sum()) <= 1500:
            raise ValueError(f"{name} seal is missing or malformed")
        seal_masks[name] = mask

    # White/off-white pixels are traversable. The yellow marker and the one
    # candidate seal being removed are explicitly converted to floor.
    base_floor = np.all(arr >= 238, axis=2) | marker_mask
    winners = []
    for name, mask in seal_masks.items():
        opened = base_floor | mask
        if _flood_reaches_border(opened, start):
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"expected one pixel-derived escape seal, found {winners}")
    return winners[0]
