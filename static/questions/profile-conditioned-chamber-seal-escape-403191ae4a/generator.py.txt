"""Render thick-walled chambers with removable colored barrier seals."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 250, 247)
WALL = (42, 47, 52)
MARKER = (242, 190, 38)
DECORATION = (188, 197, 204)
WALL_WIDTH = 19
COLORS = {
    "red": (220, 62, 62),
    "green": (42, 166, 91),
    "blue": (55, 112, 210),
}


def _transform_point(point: tuple[int, int], turn: bool, flip_y: bool, dx: int, dy: int) -> list[int]:
    x, y = point
    if turn:
        x, y = WIDTH - x, HEIGHT - y
    if flip_y:
        y = HEIGHT - y
    return [x + dx, y + dy]


def _transform_segment(segment: list[tuple[int, int]], turn: bool, flip_y: bool, dx: int, dy: int) -> list[list[int]]:
    return [_transform_point(point, turn, flip_y, dx, dy) for point in segment]


def sample_scene(seed: int) -> dict:
    """Return a deterministic, balanced scene for one integer seed."""

    rng = random.Random(seed)
    turn = bool((seed // 3) % 2)
    flip_y = bool((seed // 6) % 2)
    dx = rng.randint(-9, 9)
    dy = rng.randint(-7, 7)

    # The large chamber contains the marker and a partition with a generous
    # permanent doorway. The small chamber is unrelated to the marker.
    base_walls = [
        [(140, 90), (450, 90)],
        [(140, 390), (450, 390)],
        [(140, 90), (140, 390)],
        [(450, 90), (450, 390)],
        [(295, 90), (295, 276)],
        [(295, 344), (295, 390)],
        [(500, 160), (590, 160)],
        [(500, 310), (590, 310)],
        [(500, 160), (500, 310)],
        [(590, 160), (590, 310)],
    ]
    roles = [
        [(140, 205), (140, 265)],   # outer wall: opens marker chamber
        [(295, 155), (295, 215)],   # partition: chamber remains enclosed
        [(500, 205), (500, 265)],   # unrelated chamber wall
    ]
    names = ["red", "green", "blue"]
    shift = seed % 3
    seals = {
        names[(index + shift) % 3]: _transform_segment(segment, turn, flip_y, dx, dy)
        for index, segment in enumerate(roles)
    }
    decorations = [
        _transform_point((76, 116 + 34 * index), turn, flip_y, dx, dy)
        for index in range(2 + seed % 3)
    ]
    return {
        "walls": [_transform_segment(segment, turn, flip_y, dx, dy) for segment in base_walls],
        "seals": seals,
        "marker": _transform_point((210, 235), turn, flip_y, dx, dy),
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    walls = scene["walls"]
    seals = scene["seals"]
    marker = scene["marker"]
    decorations = scene["decorations"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for segment in walls:
        draw.line([tuple(segment[0]), tuple(segment[1])], fill=WALL, width=WALL_WIDTH)
    # Colored pixels replace the underlying black wall and are the exact visible
    # material removed by the question's counterfactual.
    for name, segment in seals.items():
        draw.line([tuple(segment[0]), tuple(segment[1])], fill=COLORS[name], width=WALL_WIDTH)
    mx, my = marker
    draw.ellipse((mx - 12, my - 12, mx + 12, my + 12), fill=MARKER)
    for x, y in decorations:
        draw.ellipse((x - 6, y - 6, x + 6, y + 6), fill=DECORATION, outline=WALL, width=2)
    return image


def _barrier_mask(scene: dict, removed: str) -> Image.Image:
    walls = scene["walls"]
    seals = scene["seals"]
    mask = Image.new("1", (WIDTH, HEIGHT), 0)
    draw = ImageDraw.Draw(mask)
    for segment in walls:
        draw.line([tuple(segment[0]), tuple(segment[1])], fill=1, width=WALL_WIDTH)
    for name, segment in seals.items():
        draw.line(
            [tuple(segment[0]), tuple(segment[1])],
            fill=0 if name == removed else 1,
            width=WALL_WIDTH,
        )
    return mask


def _marker_reaches_border(scene: dict, removed: str) -> bool:
    marker = scene["marker"]
    blocked = _barrier_mask(scene, removed).load()
    sx, sy = int(marker[0]), int(marker[1])
    if blocked[sx, sy]:
        return False
    pending = deque([(sx, sy)])
    seen = {(sx, sy)}
    while pending:
        x, y = pending.popleft()
        if x == 0 or y == 0 or x == WIDTH - 1 or y == HEIGHT - 1:
            return True
        for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            px, py = point
            if 0 <= px < WIDTH and 0 <= py < HEIGHT and point not in seen and not blocked[px, py]:
                seen.add(point)
                pending.append(point)
    return False


def analytic_gold(scene: dict) -> str:
    # Explicit accesses are intentional: these are exactly the visible latent
    # inputs allowed to influence the analytic label.
    scene["walls"]
    scene["seals"]
    scene["marker"]
    winners = [name for name in COLORS if _marker_reaches_border(scene, name)]
    if len(winners) != 1:
        raise ValueError(f"expected one escape-opening seal, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    lengths = [math.dist(segment[0], segment[1]) for segment in scene["seals"].values()]
    # Opening length beyond wall thickness is conservative clearance for a
    # four-connected raster path through the deleted seal.
    return float(min(lengths) - WALL_WIDTH)


def is_quarantined(scene: dict) -> bool:
    try:
        if analytic_gold(scene) not in COLORS or margin(scene) < 35:
            return True
        mx, my = scene["marker"]
        if not (28 <= mx < WIDTH - 28 and 28 <= my < HEIGHT - 28):
            return True
        return any(math.dist(scene["marker"], endpoint) < 30 for segment in scene["seals"].values() for endpoint in segment)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["walls"] = list(reversed(scene["walls"]))
    return [("reverse_same_color_wall_storage_order", twin)]
