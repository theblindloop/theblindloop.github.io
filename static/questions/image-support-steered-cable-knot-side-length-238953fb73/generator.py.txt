"""Renderer and analytic arm for the cable-knot side-length world.

One thin pure-black cable crosses the whole sheet from its left edge region to
its right edge region, single-valued in the horizontal direction.  A small red
ring is drawn centred on one interior bend of that cable and splits it into a
left part and a right part.  The decision is which part is longer *measured
along the cable*, so both widely separated halves of the same connected thin
stroke must be integrated before the question can be answered; the horizontal
position of the ring is counterbalanced against the answer by the sampler in
`generate.py`.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "cable_knot_side_length-renderer-0.1.0"

WIDTH = 980
HEIGHT = 430
X_START = 46
X_END = 934
Y_MIN = 74
Y_MAX = 356

CABLE_WIDTH = 3
RING_RADIUS = 9
RING_WIDTH = 2

CLEAR_FLOOR = 34.0        # no scene with a smaller length gap is ever emitted
QUARANTINE_GAP = 46.0     # below this the scene is flagged near-boundary

CABLE_INK = (14, 14, 16)
RING_COLOUR = (206, 42, 52)
SHEET_TONES = ((250, 249, 245), (247, 248, 250), (249, 247, 250))
FURNITURE = (142, 143, 150)
DECOY_COLOURS = (
    (150, 186, 214),
    (156, 200, 172),
    (196, 186, 214),
    (176, 198, 150),
    (146, 196, 202),
    (206, 186, 152),
)


def _knot_xs(rng: random.Random, count: int) -> list[int]:
    """Strictly increasing knot abscissae with jittered but bounded spacing."""
    weights = [rng.uniform(0.72, 1.32) for _ in range(count - 1)]
    total = sum(weights)
    span = X_END - X_START
    xs = [X_START]
    acc = 0.0
    for weight in weights[:-1]:
        acc += weight
        xs.append(int(round(X_START + span * acc / total)))
    xs.append(X_END)
    for index in range(1, len(xs)):          # keep every segment usably wide
        if xs[index] - xs[index - 1] < 34:
            xs[index] = xs[index - 1] + 34
    if xs[-1] != X_END:
        xs[-1] = max(X_END, xs[-2] + 34)
    return xs


def sample_scene(seed: int) -> dict:
    """Sample one deterministic scene from `seed`."""
    rng = random.Random((seed * 2654435761) % (2 ** 61 - 1))

    knots = rng.choice((11, 12, 13))
    xs = _knot_xs(rng, knots)
    segments = len(xs) - 1

    # One contiguous run of steep segments makes the two halves differ in cable
    # length in a way the horizontal split position cannot reveal.
    steep_len = rng.choice((3, 4, 5))
    steep_start = rng.randrange(0, segments - steep_len + 1)
    steep = set(range(steep_start, steep_start + steep_len))

    ys = [float(rng.uniform(150.0, 285.0))]
    sign = rng.choice((-1.0, 1.0))
    for index in range(segments):
        dx = xs[index + 1] - xs[index]
        slope = rng.uniform(1.45, 2.15) if index in steep else rng.uniform(0.12, 0.62)
        if rng.random() < 0.62:
            sign = -sign
        for _ in range(6):
            nxt = ys[-1] + sign * slope * dx
            if Y_MIN <= nxt <= Y_MAX:
                break
            sign = -sign
            nxt = ys[-1] + sign * slope * dx
            if Y_MIN <= nxt <= Y_MAX:
                break
            slope *= 0.6
        nxt = min(max(ys[-1] + sign * slope * dx, Y_MIN), Y_MAX)
        ys.append(nxt)

    nodes = [[int(x), int(round(y))] for x, y in zip(xs, ys)]
    knot_index = rng.randrange(3, segments - 2)

    decoys = []
    for _ in range(rng.choice((2, 3))):
        count = rng.choice((4, 5, 6))
        dxs = sorted(rng.randrange(30, WIDTH - 30) for _ in range(count))
        decoys.append(
            {
                "colour": list(DECOY_COLOURS[rng.randrange(len(DECOY_COLOURS))]),
                "width": rng.choice((2, 3)),
                "nodes": [[int(x), int(rng.randrange(52, HEIGHT - 40))] for x in dxs],
            }
        )

    tick_row = int(rng.choice((HEIGHT - 22, HEIGHT - 20)))
    ticks = [
        [int(x), tick_row, tick_row - int(rng.choice((7, 9, 11)))]
        for x in range(int(rng.choice((54, 62, 70))), WIDTH - 40, int(rng.choice((46, 54, 62))))
    ]

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "sheet": list(SHEET_TONES[rng.randrange(len(SHEET_TONES))]),
        "nodes": nodes,
        "cable_width": CABLE_WIDTH,
        "knot_index": knot_index,
        "ring_radius": RING_RADIUS,
        "ring_width": RING_WIDTH,
        "frame": [
            int(rng.choice((16, 20, 24))),
            int(rng.choice((16, 20, 24))),
            WIDTH - int(rng.choice((16, 20, 24))),
            HEIGHT - int(rng.choice((16, 20, 24))),
        ],
        "decoys": decoys,
        "ticks": ticks,
    }


def _segment_lengths(scene: dict) -> list[float]:
    nodes = scene["nodes"]
    return [
        math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(nodes, nodes[1:])
    ]


def side_lengths(scene: dict) -> tuple[float, float]:
    """Cable length left of the ring centre and right of it, in pixels."""
    lengths = _segment_lengths(scene)
    split = scene["knot_index"]
    return sum(lengths[:split]), sum(lengths[split:])


def analytic_gold(scene: dict) -> str:
    """"left" when the part left of the ring is the longer one, else "right"."""
    left, right = side_lengths(scene)
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    """Absolute difference in cable length between the two parts, in pixels."""
    left, right = side_lengths(scene)
    return abs(left - right)


def horizontal_cue(scene: dict) -> str:
    """Which part is wider horizontally: the cheap cue the answer must defy."""
    nodes = scene["nodes"]
    split = scene["knot_index"]
    left = nodes[split][0] - nodes[0][0]
    right = nodes[-1][0] - nodes[split][0]
    return "left" if left > right else "right"


def max_abs_slope(scene: dict) -> float:
    nodes = scene["nodes"]
    return max(
        abs(b[1] - a[1]) / max(b[0] - a[0], 1) for a, b in zip(nodes, nodes[1:])
    )


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose length gap is close to the finite-resolution floor."""
    return bool(margin(scene) < QUARANTINE_GAP or max_abs_slope(scene) > 2.35)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is stroked into the raster."""
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), tuple(scene["sheet"]))
    draw = ImageDraw.Draw(image)

    draw.rectangle([tuple(scene["frame"][:2]), tuple(scene["frame"][2:])],
                   outline=FURNITURE, width=1)
    for x, y0, y1 in scene["ticks"]:
        draw.line([(x, y0), (x, y1)], fill=FURNITURE, width=1)
    for decoy in scene["decoys"]:
        draw.line(
            [tuple(node) for node in decoy["nodes"]],
            fill=tuple(decoy["colour"]),
            width=decoy["width"],
            joint="curve",
        )

    knot = scene["nodes"][scene["knot_index"]]
    radius = scene["ring_radius"]
    draw.ellipse(
        [knot[0] - radius, knot[1] - radius, knot[0] + radius, knot[1] + radius],
        outline=RING_COLOUR,
        width=scene["ring_width"],
    )

    draw.line(
        [tuple(node) for node in scene["nodes"]],
        fill=CABLE_INK,
        width=scene["cable_width"],
        joint="curve",
    )
    return image
