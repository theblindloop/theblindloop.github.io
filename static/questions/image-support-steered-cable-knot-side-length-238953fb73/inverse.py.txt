"""Pixel-only inverse arm for the cable-knot side-length world.

The arm receives one rendered PNG and nothing else: no scene, no renderer code,
no analytic gold, no file names.  The procedure is

1. segment the pure-black ink by colour and keep its single largest 8-connected
   component; a second substantial black component is a degenerate raster;
2. recover one centre height per occupied column and require the column support
   to be gap-free and at least `MIN_COLUMNS` wide, so a cable that has been cut
   anywhere makes the arm abstain instead of answering from a fragment;
3. locate the red ring by colour and take its centroid as the split abscissa,
   rejecting a ring whose surviving pixels are not a small compact cluster;
4. accumulate chord length along the recovered centre line on each side of the
   split column and compare the two totals.

Both accumulations run over the whole traced stroke, so neither side can be
answered from a local crop.  Degenerate rasters return "unresolved".
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "cable_knot_side_length-oracle-0.1.0"

INK_MAX = 70            # every channel below this counts as cable ink
MIN_INK = 900           # a smaller component cannot be a usable cable
STRAY_INK = 40          # a second black component above this size is degenerate
MIN_COLUMNS = 840       # a usable cable spans at least this many image columns
RING_MIN = 26           # fewer surviving red pixels means no usable ring
RING_SPAN = 34          # a usable ring's red pixels fit inside this box
EDGE_CLEAR = 60         # the split must sit this far inside the cable's span
STEP = 3                # columns per chord when accumulating cable length
TIE = 6.0               # smaller measured length gaps are not resolvable


def _mask_points(array: np.ndarray) -> list[tuple[int, int]]:
    return [(int(x), int(y)) for y, x in np.argwhere(array)]


def _components(points: list[tuple[int, int]]) -> list[list[tuple[int, int]]]:
    remaining = set(points)
    groups: list[list[tuple[int, int]]] = []
    while remaining:
        start = next(iter(remaining))
        remaining.discard(start)
        queue = deque([start])
        group = [start]
        while queue:
            cx, cy = queue.popleft()
            for dx in (-1, 0, 1):
                for dy in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    neighbour = (cx + dx, cy + dy)
                    if neighbour in remaining:
                        remaining.discard(neighbour)
                        queue.append(neighbour)
                        group.append(neighbour)
        groups.append(group)
    groups.sort(key=len, reverse=True)
    return groups


def _centre_line(group: list[tuple[int, int]]) -> tuple[int, list[float]] | None:
    columns: dict[int, list[int]] = {}
    for x, y in group:
        columns.setdefault(x, []).append(y)
    xs = sorted(columns)
    if len(xs) < MIN_COLUMNS:
        return None
    if xs[-1] - xs[0] + 1 != len(xs):        # a gap anywhere in the support
        return None
    return xs[0], [sum(columns[x]) / len(columns[x]) for x in xs]


def _arc_length(line: list[float], lo: int, hi: int) -> float:
    """Chord length of the centre line between column offsets `lo` and `hi`."""
    total = 0.0
    cursor = lo
    while cursor < hi:
        nxt = min(cursor + STEP, hi)
        total += math.hypot(nxt - cursor, line[nxt] - line[cursor])
        cursor = nxt
    return total


def decision_from_image(image: Image.Image) -> str:
    """Return "left", "right", or "unresolved" from the pixels alone."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)

    ink = (array < INK_MAX).all(axis=2)
    points = _mask_points(ink)
    if len(points) < MIN_INK:
        return "unresolved"
    groups = _components(points)
    if len(groups) > 1 and len(groups[1]) > STRAY_INK:
        return "unresolved"
    if len(groups[0]) < MIN_INK:
        return "unresolved"
    traced = _centre_line(groups[0])
    if traced is None:
        return "unresolved"
    x0, line = traced

    red = (
        (array[:, :, 0] > 150)
        & (array[:, :, 1] < 110)
        & (array[:, :, 2] < 110)
        & (array[:, :, 0] - array[:, :, 1] > 60)
    )
    ring = _mask_points(red)
    if len(ring) < RING_MIN:
        return "unresolved"
    rxs = [x for x, _ in ring]
    rys = [y for _, y in ring]
    if max(rxs) - min(rxs) > RING_SPAN or max(rys) - min(rys) > RING_SPAN:
        return "unresolved"
    split = int(round(sum(rxs) / len(rxs))) - x0
    if not EDGE_CLEAR <= split <= len(line) - 1 - EDGE_CLEAR:
        return "unresolved"

    left = _arc_length(line, 0, split)
    right = _arc_length(line, split, len(line) - 1)
    if abs(left - right) < TIE:
        return "unresolved"
    return "left" if left > right else "right"
