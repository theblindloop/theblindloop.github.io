"""Deterministic scene sampling and integer raster rendering for unit_cut_remainder.

Each picture shows one black measuring stick and five horizontal rods that all
start on a common left edge.  Four rods are coloured and one is grey clutter.
A coloured rod is notionally chopped into as many whole stick lengths as will
fit; the decision is the colour of the rod whose *leftover* offcut is longest.
The leftover is a residue, not a magnitude: it is what integer division throws
away, so rod length, rod end position and the number of whole cuts are all
deliberately decorrelated from the answer.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "unit_cut_remainder-renderer-0.1.0"

WIDTH = 900
HEIGHT = 500
BACKGROUND = (255, 255, 255)

COLOR_NAMES = ("red", "green", "blue", "orange")
COLOR_RGB = {
    "red": (214, 32, 32),
    "green": (18, 146, 44),
    "blue": (28, 62, 214),
    "orange": (240, 142, 8),
}
GREY_RGB = (150, 150, 150)
STICK_RGB = (0, 0, 0)

STICK_Y = 62
STICK_HEIGHT = 18
CAP_OVERHANG = 9
CAP_WIDTH = 3
ROD_TOP = 150
ROD_HEIGHT = 22
SLOTS = 5

UNIT_MIN = 72
UNIT_MAX = 100
WHOLE_MIN = 1
WHOLE_MAX = 5
REMAINDER_LO_FRACTION = 0.20
REMAINDER_HI_FRACTION = 0.80
MIN_REMAINDER_GAP = 8
MIN_LENGTH_GAP = 10
MIN_CLEARANCE = 8.0
QUARANTINE_MARGIN = 12.0


def _remainder_bounds(unit: int) -> tuple[int, int]:
    return (
        int(math.ceil(REMAINDER_LO_FRACTION * unit)),
        int(math.floor(REMAINDER_HI_FRACTION * unit)),
    )


def _try_scene(rng: random.Random) -> dict | None:
    unit = rng.randrange(UNIT_MIN, UNIT_MAX + 1)
    low, high = _remainder_bounds(unit)
    remainders = sorted(rng.sample(range(low, high + 1), 4))
    for earlier, later in zip(remainders, remainders[1:]):
        if later - earlier < MIN_REMAINDER_GAP:
            return None
    rng.shuffle(remainders)
    wholes = [rng.randrange(WHOLE_MIN, WHOLE_MAX + 1) for _ in range(4)]
    lengths = [whole * unit + rest for whole, rest in zip(wholes, remainders)]
    ordered = sorted(lengths)
    for earlier, later in zip(ordered, ordered[1:]):
        if later - earlier < MIN_LENGTH_GAP:
            return None
    origin_x = rng.randrange(40, 96)
    unit_x = rng.randrange(origin_x, origin_x + 240)
    grey_slot = rng.randrange(SLOTS)
    grey_len = rng.randrange(unit + 24, 4 * unit + high)
    scene = {
        "unit": unit,
        "unit_x": unit_x,
        "origin_x": origin_x,
        "lengths": lengths,
        "rod_colors": rng.sample(list(COLOR_NAMES), 4),
        "grey_len": grey_len,
        "grey_slot": grey_slot,
        "row_gap": rng.randrange(60, 73),
    }
    if origin_x + max(max(lengths), grey_len) > WIDTH - 30:
        return None
    if unit_x + unit > WIDTH - 30:
        return None
    if margin(scene) < MIN_CLEARANCE:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one integer seed."""

    rng = random.Random(int(seed) * 7919 + 13)
    for _ in range(4000):
        scene = _try_scene(rng)
        if scene is not None:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def _remainders(scene: dict) -> list[int]:
    unit = int(scene["unit"])
    return [int(length) % unit for length in scene["lengths"]]


def analytic_gold(scene: dict) -> str:
    """Colour of the rod whose leftover offcut is the longest."""

    rests = _remainders(scene)
    best = max(range(4), key=lambda index: rests[index])
    return scene["rod_colors"][best]


def margin(scene: dict) -> float:
    """Smallest length change, in pixels, that could flip the answer.

    Two events can flip it: the runner-up leftover overtaking the winner, or a
    rod crossing a whole-unit boundary so that one more or one fewer whole cut
    fits and its leftover jumps by a full unit.
    """

    unit = int(scene["unit"])
    rests = _remainders(scene)
    ordered = sorted(rests, reverse=True)
    top_two = float(ordered[0] - ordered[1])
    wrap = float(min(min(rest, unit - rest) for rest in rests))
    return min(top_two, wrap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes are excluded from headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent alias exists; see candidate.json."""

    return []


def _slot_rows(scene: dict) -> list[int]:
    gap = int(scene["row_gap"])
    return [ROD_TOP + slot * gap for slot in range(SLOTS)]


def render(scene: dict) -> Image.Image:
    """Rasterize one scene with hard integer edges and no antialiasing."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    unit = int(scene["unit"])
    ux = int(scene["unit_x"])
    draw.rectangle(
        [ux, STICK_Y, ux + unit - 1, STICK_Y + STICK_HEIGHT - 1], fill=STICK_RGB
    )
    for cap_x in (ux, ux + unit - CAP_WIDTH):
        draw.rectangle(
            [
                cap_x,
                STICK_Y - CAP_OVERHANG,
                cap_x + CAP_WIDTH - 1,
                STICK_Y + STICK_HEIGHT - 1 + CAP_OVERHANG,
            ],
            fill=STICK_RGB,
        )

    rows = _slot_rows(scene)
    origin = int(scene["origin_x"])
    grey_slot = int(scene["grey_slot"])
    draw.rectangle(
        [origin, rows[grey_slot], origin + int(scene["grey_len"]) - 1,
         rows[grey_slot] + ROD_HEIGHT - 1],
        fill=GREY_RGB,
    )
    free = [slot for slot in range(SLOTS) if slot != grey_slot]
    for index, slot in enumerate(free):
        top = rows[slot]
        draw.rectangle(
            [origin, top, origin + int(scene["lengths"][index]) - 1, top + ROD_HEIGHT - 1],
            fill=COLOR_RGB[scene["rod_colors"][index]],
        )
    return image
