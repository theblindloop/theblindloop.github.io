"""Pixel-only inverse arm for unit_cut_remainder.

Receives one rasterized picture and nothing else: no latent parameters, no
analytic gold, no renderer, prompt, generator or verifier code.  It recovers
the black measuring stick's length, segments each coloured rod, measures each
rod's drawn length, divides by the stick length, and returns the colour with
the longest remainder.  It abstains when the evidence is not clean.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ROD_RGB = {
    "red": (214, 32, 32),
    "green": (18, 146, 44),
    "blue": (28, 62, 214),
    "orange": (240, 142, 8),
}
CHANNEL_TOLERANCE = 34
DARK_LEVEL = 45
MIN_STICK_PIXELS = 400
MIN_ROD_PIXELS = 600
MIN_REMAINDER_GAP = 4.0
MIN_WRAP_CLEARANCE = 4.0
ABSTAIN = "abstain"


def _column_span(mask: np.ndarray) -> tuple[int, int] | None:
    columns = np.flatnonzero(mask.any(axis=0))
    if columns.size == 0:
        return None
    left, right = int(columns[0]), int(columns[-1])
    if columns.size != right - left + 1:
        return None
    return left, right


def _stick_length(pixels: np.ndarray) -> int | None:
    mask = (pixels < DARK_LEVEL).all(axis=2)
    if int(mask.sum()) < MIN_STICK_PIXELS:
        return None
    span = _column_span(mask)
    if span is None:
        return None
    left, right = span
    width = right - left + 1
    if width < 24:
        return None
    return width


def _rod_length(pixels: np.ndarray, rgb: tuple[int, int, int]) -> int | None:
    target = np.array(rgb, dtype=np.int16)
    mask = (np.abs(pixels.astype(np.int16) - target) <= CHANNEL_TOLERANCE).all(axis=2)
    if int(mask.sum()) < MIN_ROD_PIXELS:
        return None
    span = _column_span(mask)
    if span is None:
        return None
    left, right = span
    return right - left + 1


def decision_from_image(image: Image.Image) -> str:
    """Return the rod colour with the longest leftover, or ``abstain``."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    unit = _stick_length(pixels)
    if unit is None:
        return ABSTAIN

    rests: dict[str, float] = {}
    for name, rgb in ROD_RGB.items():
        length = _rod_length(pixels, rgb)
        if length is None:
            return ABSTAIN
        rest = float(length % unit)
        if min(rest, unit - rest) < MIN_WRAP_CLEARANCE:
            return ABSTAIN
        rests[name] = rest

    ranked = sorted(rests.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] - ranked[1][1] < MIN_REMAINDER_GAP:
        return ABSTAIN
    return ranked[0][0]
