"""Independent PNG-only inverse arm for visible clover leaf bodies."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[int]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    sizes: list[int] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        size = 0
        while q:
            yy, xx = q.popleft()
            size += 1
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        sizes.append(size)
    return sizes


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    # The public rule says green bodies. This broad chromatic predicate covers
    # every permitted green palette but rejects brown, gold, gray, and white.
    green = (g >= 95) & (g >= r + 28) & (g >= b + 18) & (r <= 125)
    sizes = _components(green)
    bodies = [size for size in sizes if size >= 900]
    if len(bodies) not in (3, 4, 5) or min(bodies, default=0) < 900:
        raise ValueError("abstain: no unambiguous set of 3, 4, or 5 green leaf bodies")
    # Runner-up noise components must be far below the smallest accepted body.
    noise = sorted((size for size in sizes if size < 900), reverse=True)
    if noise and noise[0] > 250:
        raise ValueError("abstain: ambiguous secondary green component")
    return str(len(bodies))
