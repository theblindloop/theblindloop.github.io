"""Deterministic clover scene sampler and renderer."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

SIZE = 512
SS = 3
LEAF_PALETTES = ((38, 153, 84), (27, 145, 112), (91, 154, 42))
OUTLINE = (31, 47, 40)
BROWN = (112, 75, 45)
GOLD = (225, 169, 42)
BG = (248, 246, 238)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "leaf_count": (3, 4, 5)[seed % 3],
        "rotation_deg": rng.uniform(-28.0, 28.0),
        "scale": rng.uniform(0.92, 1.06),
        "offset_x": rng.randint(-18, 18),
        "offset_y": rng.randint(-12, 15),
        "palette_index": rng.randrange(len(LEAF_PALETTES)),
        "dot_phase": rng.uniform(0.0, 360.0),
        "dot_count": 2 + rng.randrange(4),
        "audit_alias": seed % 11,
    }


def _leaf_polygon(cx: float, cy: float, angle: float, length: float, width: float) -> list[tuple[float, float]]:
    # A smooth-looking pointed oval sampled from a polar leaf curve. The inward
    # point aims toward the center; rasterization is performed at 3x resolution.
    points = []
    ux, uy = math.cos(angle), math.sin(angle)
    vx, vy = -uy, ux
    for i in range(64):
        t = 2.0 * math.pi * i / 64.0
        along = 0.5 * length * math.cos(t)
        across = 0.5 * width * math.sin(t) * (0.78 + 0.22 * math.cos(t))
        points.append((cx + along * ux + across * vx, cy + along * uy + across * vy))
    return points


def render(scene: dict) -> Image.Image:
    count = int(scene["leaf_count"])
    rotation = math.radians(float(scene["rotation_deg"]))
    scale = float(scene["scale"])
    ox = float(scene["offset_x"])
    oy = float(scene["offset_y"])
    palette = LEAF_PALETTES[int(scene["palette_index"]) % len(LEAF_PALETTES)]
    dot_phase = math.radians(float(scene["dot_phase"]))
    dot_count = int(scene["dot_count"])

    image = Image.new("RGB", (SIZE * SS, SIZE * SS), BG)
    draw = ImageDraw.Draw(image)

    def poly(points, **kwargs):
        draw.polygon([(round(x * SS), round(y * SS)) for x, y in points], **kwargs)

    def line(points, **kwargs):
        draw.line([(round(x * SS), round(y * SS)) for x, y in points], **kwargs)

    def ellipse(box, **kwargs):
        draw.ellipse(tuple(round(v * SS) for v in box), **kwargs)

    # Neutral card and label make the object category explicit in every variant.
    draw.rounded_rectangle((20 * SS, 20 * SS, 492 * SS, 492 * SS), radius=22 * SS,
                           fill=(255, 255, 252), outline=(190, 185, 170), width=3 * SS)
    try:
        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", 28 * SS)
    except OSError:
        font = ImageFont.load_default()
    draw.text((256 * SS, 48 * SS), "FOUR-LEAF CLOVER", font=font, fill=(58, 65, 60), anchor="mm")

    center = (256.0 + ox, 273.0 + oy)
    # Count-adjusted size approximately equalizes total green ink across classes.
    leaf_length = 112.0 * scale * math.sqrt(4.0 / count)
    leaf_width = 76.0 * scale * math.sqrt(4.0 / count)
    radial = 92.0 * scale

    # Brown stalks visibly attach each leaf body to the named gold center but are
    # excluded from the target-color segmentation.
    for i in range(count):
        a = rotation - math.pi / 2.0 + 2.0 * math.pi * i / count
        p0 = (center[0] + 13 * math.cos(a), center[1] + 13 * math.sin(a))
        p1 = (center[0] + (radial - leaf_length * 0.42) * math.cos(a),
              center[1] + (radial - leaf_length * 0.42) * math.sin(a))
        line((p0, p1), fill=BROWN, width=9 * SS)

    for i in range(count):
        a = rotation - math.pi / 2.0 + 2.0 * math.pi * i / count
        cx = center[0] + radial * math.cos(a)
        cy = center[1] + radial * math.sin(a)
        points = _leaf_polygon(cx, cy, a, leaf_length, leaf_width)
        poly(points, fill=palette, outline=OUTLINE)
        line(points + [points[0]], fill=OUTLINE, width=3 * SS, joint="curve")
        # A short neutral vein reinforces leaf identity without splitting green.
        tip0 = (cx - 0.27 * leaf_length * math.cos(a), cy - 0.27 * leaf_length * math.sin(a))
        tip1 = (cx + 0.22 * leaf_length * math.cos(a), cy + 0.22 * leaf_length * math.sin(a))
        line((tip0, tip1), fill=(229, 235, 218), width=2 * SS)

    ellipse((center[0] - 19, center[1] - 19, center[0] + 19, center[1] + 19),
            fill=GOLD, outline=OUTLINE, width=3 * SS)

    # Count-independent gold decorations vary position and number and cannot be
    # confused with green target bodies.
    for j in range(dot_count):
        a = dot_phase + 2.0 * math.pi * j / dot_count
        rr = 172 + 8 * ((j % 2) * 2 - 1)
        x, y = center[0] + rr * math.cos(a), center[1] + rr * math.sin(a)
        ellipse((x - 4, y - 4, x + 4, y + 4), fill=GOLD)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["leaf_count"]))


def margin(scene: dict) -> float:
    count = int(scene["leaf_count"])
    scale = float(scene["scale"])
    width = 76.0 * scale * math.sqrt(4.0 / count)
    radial = 92.0 * scale
    center_spacing = 2.0 * radial * math.sin(math.pi / count)
    return round(center_spacing - width, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["audit_alias"] = (int(twin["audit_alias"]) + 1) % 11
    return [("audit_alias_increment", twin)]
