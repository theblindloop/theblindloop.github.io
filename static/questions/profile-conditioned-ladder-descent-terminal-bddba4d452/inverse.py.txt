"""Pixel-only oracle for ladder_descent_terminal.

Receives one RGB image and nothing else. It segments the gray ladder, recovers
the rail centre lines, recovers each rung as an adjacent rail pair at a height,
locates the green start dot and the four coloured terminal squares, and then
simulates the declared descent. It imports no renderer, prompt, generator,
verifier or gold code and never sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "ladder_descent_terminal-oracle-0.1.0"
ABSTAIN = "unknown"

_GRAY = (105, 105, 105)
_GREEN = (0, 150, 60)
_TERMINALS = {
    "blue": (30, 70, 200),
    "orange": (235, 130, 20),
    "purple": (140, 40, 170),
    "red": (215, 35, 35),
}
_TOL = 26


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    ref = np.array(rgb, dtype=np.int16)
    return (np.abs(arr.astype(np.int16) - ref).max(axis=2) <= _TOL)


def _runs(idx: np.ndarray) -> list[tuple[int, int]]:
    """Contiguous runs over a sorted index array, as inclusive (start, end)."""
    out: list[tuple[int, int]] = []
    for i in idx.tolist():
        if out and i == out[-1][1] + 1:
            out[-1] = (out[-1][0], i)
        else:
            out.append((i, i))
    return out


def _centroid(mask: np.ndarray) -> tuple[float, float] | None:
    ys, xs = np.nonzero(mask)
    if xs.size < 40:
        return None
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    h, w = arr.shape[:2]
    gray = _mask(arr, _GRAY)
    if gray.sum() < 500:
        return ABSTAIN

    # --- rails: image columns carrying a tall gray stroke -------------------
    col_counts = gray.sum(axis=0)
    tall = np.flatnonzero(col_counts > 0.4 * h)
    if tall.size == 0:
        return ABSTAIN
    rails = []
    for a, b in _runs(tall):
        if b - a > 20:
            return ABSTAIN  # a slab, not a rail
        rails.append(0.5 * (a + b))
    if len(rails) != 4:
        return ABSTAIN
    rail_x = [int(round(x)) for x in rails]

    # vertical extent of the ladder, from the first rail's own column
    rail_rows = np.flatnonzero(gray[:, rail_x[0]])
    if rail_rows.size < 50:
        return ABSTAIN
    top, bottom = int(rail_rows.min()), int(rail_rows.max())

    # --- rungs: rows where the gap between two adjacent rails is bridged ----
    rungs: list[dict] = []
    row_hits: list[tuple[int, int]] = []  # (row, left rail index)
    for y in range(top, bottom + 1):
        for i in range(3):
            mid = (rail_x[i] + rail_x[i + 1]) // 2
            span = gray[y, rail_x[i] : rail_x[i + 1] + 1]
            if gray[y, mid] and span.all():
                row_hits.append((y, i))
    by_pair: dict[int, list[int]] = {}
    for y, i in row_hits:
        by_pair.setdefault(i, []).append(y)
    for i, ys in by_pair.items():
        for a, b in _runs(np.array(sorted(ys))):
            if b - a > 20:
                return ABSTAIN  # implausibly thick bar
            rungs.append({"y": 0.5 * (a + b), "col": i})
    if not rungs:
        return ABSTAIN

    # order ambiguity: two rungs sharing a rail at indistinguishable heights
    rungs.sort(key=lambda r: r["y"])
    for a in range(len(rungs)):
        for b in range(a + 1, len(rungs)):
            ra, rb = rungs[a], rungs[b]
            if {ra["col"], ra["col"] + 1} & {rb["col"], rb["col"] + 1}:
                if abs(ra["y"] - rb["y"]) < 12:
                    return ABSTAIN

    # --- start dot and terminal squares ------------------------------------
    start = _centroid(_mask(arr, _GREEN))
    if start is None:
        return ABSTAIN
    start_col = min(range(4), key=lambda i: abs(rail_x[i] - start[0]))
    if abs(rail_x[start_col] - start[0]) > 40 or start[1] > top:
        return ABSTAIN

    terminals: dict[int, str] = {}
    for name, rgb in _TERMINALS.items():
        spot = _centroid(_mask(arr, rgb))
        if spot is None:
            return ABSTAIN
        col = min(range(4), key=lambda i: abs(rail_x[i] - spot[0]))
        if abs(rail_x[col] - spot[0]) > 40 or spot[1] < bottom or col in terminals:
            return ABSTAIN
        terminals[col] = name
    if len(terminals) != 4:
        return ABSTAIN

    # --- declared descent ---------------------------------------------------
    col = start_col
    for rung in rungs:
        if rung["col"] == col:
            col += 1
        elif rung["col"] + 1 == col:
            col -= 1
    return terminals[col]
