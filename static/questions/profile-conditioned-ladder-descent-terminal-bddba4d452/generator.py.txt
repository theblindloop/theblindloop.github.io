"""Renderer for ladder_descent_terminal.

A ladder lottery ("ghost leg"): four vertical rails, a set of horizontal rungs
each joining one adjacent pair of rails at its own height, a green dot on top of
one rail and four differently coloured squares under the four rails. The walker
starts at the green dot, moves downwards, and is forced to cross every rung it
meets, so the answer is the composition of the rung swaps in top-to-bottom
order rather than the identity of the rail the walker started on.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "ladder_descent_terminal-renderer-0.1.0"

CANVAS = (640, 520)
BACKGROUND = (255, 255, 255)
RAIL_COLOR = (105, 105, 105)
START_COLOR = (0, 150, 60)
N_COLS = 4
RAIL_X = (80, 240, 400, 560)
RAIL_TOP = 100
RAIL_BOTTOM = 440
STROKE = 5
START_Y = 70
START_R = 13
SQUARE_Y = 474
SQUARE_HALF = 14

MARK_COLORS = {
    "blue": (30, 70, 200),
    "orange": (235, 130, 20),
    "purple": (140, 40, 170),
    "red": (215, 35, 35),
}
COLOR_ORDER = ("blue", "orange", "purple", "red")

Y_MIN, Y_MAX = 140, 410
MIN_RUNG_GAP = 30.0        # pixel-separability floor between any two rungs
QUARANTINE_MARGIN = 34.0   # decision-relevant floor, in pixels


def _descend(rungs: list[dict], start_col: int) -> int:
    col = start_col
    for rung in sorted(rungs, key=lambda r: r["y"]):
        if rung["col"] == col:
            col += 1
        elif rung["col"] + 1 == col:
            col -= 1
    return col


def _boundary_gaps(rungs: list[dict], start_col: int, end_col: int) -> list[float]:
    """Vertical gaps of rung pairs whose order swap moves the walker elsewhere.

    Two rungs can only interact when they share a rail; swapping the heights of
    such a pair is exactly the visible event that flips the descent.
    """
    gaps: list[float] = []
    for a in range(len(rungs)):
        for b in range(a + 1, len(rungs)):
            ra, rb = rungs[a], rungs[b]
            if not {ra["col"], ra["col"] + 1} & {rb["col"], rb["col"] + 1}:
                continue
            swapped = [dict(r) for r in rungs]
            swapped[a]["y"], swapped[b]["y"] = rb["y"], ra["y"]
            if _descend(swapped, start_col) != end_col:
                gaps.append(abs(ra["y"] - rb["y"]))
    return gaps


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1000003 + 17)
    target_color = COLOR_ORDER[seed % len(COLOR_ORDER)]
    grid = list(range(Y_MIN, Y_MAX + 1, 5))

    for _ in range(4000):
        n_rungs = rng.randint(7, 10)
        ys: list[int] = []
        for _ in range(n_rungs):
            for _try in range(60):
                y = rng.choice(grid)
                if all(abs(y - other) >= MIN_RUNG_GAP for other in ys):
                    ys.append(y)
                    break
        if len(ys) != n_rungs:
            continue
        cols = [rng.randrange(N_COLS - 1) for _ in ys]
        if len(set(cols)) < N_COLS - 1:
            continue  # every gap between rails must carry at least one rung
        rungs = [{"y": y, "col": c} for y, c in zip(ys, cols)]
        rungs.sort(key=lambda r: r["y"])

        start_col = rng.randrange(N_COLS)
        end_col = _descend(rungs, start_col)
        if end_col == start_col:
            continue  # the answer must never be the rail standing below the start
        gaps = _boundary_gaps(rungs, start_col, end_col)
        if not gaps:
            continue  # rung order must genuinely decide the outcome
        margin = min(gaps)
        if margin < QUARANTINE_MARGIN:
            continue

        others = [c for c in COLOR_ORDER if c != target_color]
        rng.shuffle(others)
        terminals = []
        for col in range(N_COLS):
            terminals.append(target_color if col == end_col else others.pop())
        return {
            "canvas": list(CANVAS),
            "background": list(BACKGROUND),
            "rail_x": list(RAIL_X),
            "rail_top": RAIL_TOP,
            "rail_bottom": RAIL_BOTTOM,
            "stroke_width": STROKE,
            "rungs": rungs,
            "start_col": start_col,
            "start_y": START_Y,
            "start_radius": START_R,
            "terminals": [
                {"col": col, "color": color, "x": RAIL_X[col], "y": SQUARE_Y}
                for col, color in enumerate(terminals)
            ],
            "square_half": SQUARE_HALF,
            "rail_color": list(RAIL_COLOR),
            "start_color": list(START_COLOR),
        }
    raise RuntimeError(f"sampling failed for seed {seed}")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), tuple(scene["background"]))
    draw = ImageDraw.Draw(img)
    w = scene["stroke_width"]
    half = w // 2
    rail = tuple(scene["rail_color"])

    for x in scene["rail_x"]:
        draw.rectangle(
            [x - half, scene["rail_top"], x + half, scene["rail_bottom"]], fill=rail
        )
    for rung in scene["rungs"]:
        x0 = scene["rail_x"][rung["col"]]
        x1 = scene["rail_x"][rung["col"] + 1]
        draw.rectangle([x0 - half, rung["y"] - half, x1 + half, rung["y"] + half], fill=rail)

    sx = scene["rail_x"][scene["start_col"]]
    sy = scene["start_y"]
    r = scene["start_radius"]
    draw.ellipse([sx - r, sy - r, sx + r, sy + r], fill=tuple(scene["start_color"]))

    h = scene["square_half"]
    for term in scene["terminals"]:
        draw.rectangle(
            [term["x"] - h, term["y"] - h, term["x"] + h, term["y"] + h],
            fill=MARK_COLORS[term["color"]],
        )
    return img


def analytic_gold(scene: dict) -> str:
    end_col = _descend(scene["rungs"], scene["start_col"])
    for term in scene["terminals"]:
        if term["col"] == end_col:
            return term["color"]
    raise RuntimeError("no terminal on the ending rail")


def margin(scene: dict) -> float:
    end_col = _descend(scene["rungs"], scene["start_col"])
    gaps = _boundary_gaps(scene["rungs"], scene["start_col"], end_col)
    return float(min(gaps)) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_MARGIN:
        return True
    ys = sorted(r["y"] for r in scene["rungs"])
    return any(b - a < MIN_RUNG_GAP for a, b in zip(ys, ys[1:]))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every labelling field is painted into the raster."""
    return []
