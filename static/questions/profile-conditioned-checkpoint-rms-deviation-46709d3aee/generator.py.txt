"""Deterministic latent sampler and independent analytic arm."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 460
COLORS = ["blue", "orange", "green"]
RGB = {
    "blue": (31, 119, 180),
    "orange": (230, 111, 24),
    "green": (35, 150, 82),
}
MAGENTA = (180, 35, 125)
INK = (45, 48, 55)
GRID = (225, 228, 232)
SCALE = 4
BASE_X = [112, 232, 352, 472, 592]
BASELINE_Y = 230

# Every difficulty role has the same maximum absolute offset (12), blocking a
# single-farthest-marker shortcut. Their RMS values remain well separated.
ROLE_PATTERNS = [
    [12, 12, 11, 10, 9],
    [12, 7, 6, 5, 4],
    [12, 4, 3, 2, 1],
]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    role_for_color = [None, None, None]
    role_for_color[winner] = 0
    others = [index for index in range(3) if index != winner]
    if rng.randrange(2):
        others.reverse()
    role_for_color[others[0]] = 1
    role_for_color[others[1]] = 2

    values = []
    baseline = rng.randint(-30, 30)
    for color_index in range(3):
        offsets = list(ROLE_PATTERNS[role_for_color[color_index]])
        rng.shuffle(offsets)
        offsets = [value * (-1 if rng.randrange(2) else 1) for value in offsets]
        values.append([baseline + value for value in offsets])

    jitters = [-16, 0, 16]
    rng.shuffle(jitters)
    return {
        "colors": list(COLORS),
        "values": values,
        "baseline": baseline,
        "translate_x": rng.randint(-10, 10),
        "translate_y": rng.randint(-12, 12),
        "x_jitter": jitters,
    }


def _scores(scene: dict) -> list[float]:
    baseline = float(scene["baseline"])
    colors = scene["colors"]
    values = scene["values"]
    if len(colors) != 3 or len(values) != 3:
        raise ValueError("expected three color series")
    scores = []
    for series in values:
        if len(series) != 5:
            raise ValueError("expected five checkpoints per color")
        scores.append(math.sqrt(sum((float(value) - baseline) ** 2 for value in series) / 5.0))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    colors = scene["colors"]
    return str(colors[max(range(3), key=lambda index: scores[index])])


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene), reverse=True)
    return (ordered[0] - ordered[1]) * SCALE


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["baseline"] += 37
    twin["values"] = [[value + 37 for value in series] for series in twin["values"]]
    return [("shift_numeric_value_origin", twin)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    tx = int(scene["translate_x"])
    ty = int(scene["translate_y"])
    baseline = float(scene["baseline"])
    colors = scene["colors"]
    values = scene["values"]
    x_jitter = scene["x_jitter"]
    baseline_y = BASELINE_Y + ty

    # Light checkpoint guides carry no answer-specific styling.
    for x in BASE_X:
        draw.line((x + tx, 62 + ty, x + tx, 398 + ty), fill=GRID, width=1)
    # A visibly dashed reference, recovered by the inverse arm from magenta ink.
    for x0 in range(66 + tx, 656 + tx, 18):
        draw.line((x0, baseline_y, min(x0 + 10, 656 + tx), baseline_y), fill=MAGENTA, width=3)
    draw.line((66 + tx, 398 + ty, 656 + tx, 398 + ty), fill=INK, width=2)

    for color_index, color in enumerate(colors):
        rgb = RGB[color]
        for point_index, value in enumerate(values[color_index]):
            x = BASE_X[point_index] + tx + int(x_jitter[color_index])
            y = int(round(baseline_y - (float(value) - baseline) * SCALE))
            radius = 7
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=rgb)
    return image
