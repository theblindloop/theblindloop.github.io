"""Pixel-only inverse arm for the final raster.

This module deliberately has no dependency on the latent/analytic implementation.
"""

from __future__ import annotations

import math
from collections import deque

from PIL import Image

PALETTE = {
    "blue": (31, 119, 180),
    "orange": (230, 111, 24),
    "green": (35, 150, 82),
}
REFERENCE = (180, 35, 125)


def _components(mask: list[list[bool]]) -> list[tuple[int, float, float]]:
    height = len(mask)
    width = len(mask[0]) if height else 0
    seen = set()
    found = []
    for y in range(height):
        for x in range(width):
            if not mask[y][x] or (x, y) in seen:
                continue
            queue = deque([(x, y)])
            seen.add((x, y))
            count = 0
            sx = 0
            sy = 0
            while queue:
                px, py = queue.popleft()
                count += 1
                sx += px
                sy += py
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny][nx] and (nx, ny) not in seen:
                        seen.add((nx, ny))
                        queue.append((nx, ny))
            if count >= 80:
                found.append((count, sx / count, sy / count))
    return found


def _mask(pixels, width: int, height: int, target: tuple[int, int, int]):
    return [[pixels[x, y] == target for x in range(width)] for y in range(height)]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 100 or height < 100:
        raise ValueError("raster too small")
    pixels = rgb.load()

    reference_y = [y for y in range(height) for x in range(width) if pixels[x, y] == REFERENCE]
    if len(reference_y) < 100:
        raise ValueError("visible dashed baseline not recovered")
    baseline_y = sum(reference_y) / len(reference_y)

    scores = {}
    for name, target in PALETTE.items():
        components = _components(_mask(pixels, width, height, target))
        if len(components) != 5:
            raise ValueError(f"expected five {name} checkpoint components, found {len(components)}")
        centers = [component[2] for component in components]
        scores[name] = math.sqrt(sum((y - baseline_y) ** 2 for y in centers) / 5.0)
    ranked = sorted(scores, key=scores.get, reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 8.0:
        raise ValueError("ambiguous pixel decision margin")
    return ranked[0]
