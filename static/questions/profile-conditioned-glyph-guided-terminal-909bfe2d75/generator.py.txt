"""Analytic scene sampler and raster renderer for glyph-guided navigation."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

SIZE = 600
BG = (250, 249, 246)
DARK = (42, 47, 54)
GREEN = (35, 164, 88)
TERMINAL_RGB = {
    "orange": (234, 116, 28),
    "purple": (137, 73, 188),
    "blue": (42, 105, 207),
    "red": (205, 55, 63),
}
JUNCTION_RGB = {
    "cyan": (25, 183, 194),
    "yellow": (236, 180, 34),
}
COLORS = tuple(TERMINAL_RGB)

# Node order: start, root, upper child, lower child, four terminals.
POINTS = [
    (60, 300), (180, 300), (340, 170), (340, 430),
    (540, 90), (540, 250), (540, 350), (540, 510),
]
EDGES = [
    (0, 1, [(60, 300), (180, 300)]),
    (1, 2, [(180, 300), (180, 170), (340, 170)]),
    (1, 3, [(180, 300), (180, 430), (340, 430)]),
    (2, 4, [(340, 170), (340, 90), (540, 90)]),
    (2, 5, [(340, 170), (340, 250), (540, 250)]),
    (3, 6, [(340, 430), (340, 350), (540, 350)]),
    (3, 7, [(340, 430), (340, 510), (540, 510)]),
]


def _rotate_point(point: tuple[int, int], turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(turns % 4):
        x, y = SIZE - y, x
    return x, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    glyphs = [rng.choice(("circle", "square")) for _ in range(3)]
    junction_colors = [rng.choice(("cyan", "yellow")) for _ in range(3)]
    winner = 0 if glyphs[0] == "circle" else 2
    chosen_child = 1 if winner == 0 else 2
    winner += 0 if glyphs[chosen_child] == "circle" else 1
    terminal_colors = list(COLORS)
    rng.shuffle(terminal_colors)
    target = COLORS[seed % len(COLORS)]
    target_index = terminal_colors.index(target)
    terminal_colors[winner], terminal_colors[target_index] = (
        terminal_colors[target_index], terminal_colors[winner]
    )
    return {
        "rotation": (seed // 4) % 4,
        "junction_glyphs": glyphs,
        "junction_colors": junction_colors,
        "terminal_colors": terminal_colors,
    }


def analytic_gold(scene: dict) -> str:
    glyphs = scene["junction_glyphs"]
    terminal_colors = scene["terminal_colors"]
    leaf = 0 if glyphs[0] == "circle" else 2
    child = 1 if leaf == 0 else 2
    if glyphs[child] == "square":
        leaf += 1
    return terminal_colors[leaf]


def margin(scene: dict) -> float:
    # Every choice is an exact orthogonal fork; the runner-up angular gap is 180 degrees.
    return 180.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    rotation = int(scene["rotation"])
    glyphs = list(scene["junction_glyphs"])
    junction_colors = list(scene["junction_colors"])
    terminal_colors = list(scene["terminal_colors"])
    scale = 4
    image = Image.new("RGB", (SIZE * scale, SIZE * scale), BG)
    draw = ImageDraw.Draw(image)

    def pts(values):
        return [tuple(v * scale for v in _rotate_point(p, rotation)) for p in values]

    for _a, _b, polyline in EDGES:
        draw.line(pts(polyline), fill=DARK, width=9 * scale, joint="curve")

    transformed = [_rotate_point(p, rotation) for p in POINTS]

    def ellipse(center, radius, **kwargs):
        x, y = center
        draw.ellipse(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale), **kwargs)

    def rectangle(center, half, **kwargs):
        x, y = center
        draw.rectangle(((x-half)*scale, (y-half)*scale, (x+half)*scale, (y+half)*scale), **kwargs)

    # White moats separate incident dark links into independently recoverable components.
    ellipse(transformed[0], 23, fill=BG)
    ellipse(transformed[0], 18, fill=BG, outline=GREEN, width=6 * scale)

    for index, center in enumerate(transformed[1:4]):
        glyph = glyphs[index]
        color = JUNCTION_RGB[junction_colors[index]]
        if glyph == "circle":
            ellipse(center, 23, fill=BG)
            ellipse(center, 17, fill=color)
        else:
            rectangle(center, 23, fill=BG)
            rectangle(center, 16, fill=color)

    for index, center in enumerate(transformed[4:]):
        rectangle(center, 23, fill=BG)
        rectangle(center, 17, fill=TERMINAL_RGB[terminal_colors[index]])

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def rotated_invariance(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["rotation"] = (int(twin["rotation"]) + 1) % 4
    return twin
