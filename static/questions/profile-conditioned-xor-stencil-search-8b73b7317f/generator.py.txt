"""Deterministic renderer and analytic arm for XOR stencil search."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

GRID_N = 5
LABELS = ("A", "B", "C", "D")


def _valid_reference(bits: list[int]) -> bool:
    rows = [sum(bits[r * GRID_N : (r + 1) * GRID_N]) for r in range(GRID_N)]
    cols = [sum(bits[r * GRID_N + c] for r in range(GRID_N)) for c in range(GRID_N)]
    return all(1 <= value <= 4 for value in rows + cols)


def _changed_copy(reference: list[int], swaps: int, rng: random.Random) -> list[int]:
    ones = [i for i, value in enumerate(reference) if value]
    zeros = [i for i, value in enumerate(reference) if not value]
    out = list(reference)
    for index in rng.sample(ones, swaps):
        out[index] = 0
    for index in rng.sample(zeros, swaps):
        out[index] = 1
    return out


def _decompose(output: list[int], rng: random.Random) -> list[list[int]]:
    """Make two 12-cell stencils with six overlaps and the requested XOR."""
    ones = [i for i, value in enumerate(output) if value]
    zeros = [i for i, value in enumerate(output) if not value]
    shared = set(rng.sample(zeros, 6))
    first_only = set(rng.sample(ones, 6))
    second_only = set(ones) - first_only
    first = [int(i in shared or i in first_only) for i in range(GRID_N * GRID_N)]
    second = [int(i in shared or i in second_only) for i in range(GRID_N * GRID_N)]
    return [first, second]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        reference = [0] * (GRID_N * GRID_N)
        for index in rng.sample(range(GRID_N * GRID_N), 12):
            reference[index] = 1
        if _valid_reference(reference):
            break

    outputs = [list(reference)] + [_changed_copy(reference, k, rng) for k in (1, 2, 3)]
    target = int(seed) % 4
    ordered: list[list[int] | None] = [None] * 4
    ordered[target] = outputs[0]
    distractors = outputs[1:]
    rng.shuffle(distractors)
    for slot in range(4):
        if ordered[slot] is None:
            ordered[slot] = distractors.pop()
    pairs = [_decompose(output, rng) for output in ordered if output is not None]

    return {
        "reference": reference,
        "pairs": pairs,
        "palette": {
            "background": [249, 248, 244],
            "ink": [35, 42, 52],
            "cyan": [33, 158, 188],
            "amber": [230, 147, 36],
            "empty": [255, 255, 255],
        },
        "layout": {
            "canvas": [900, 750],
            "cell": 19,
            "reference_origin": [402, 88],
            "pair_origins": [[85, 300], [505, 300], [85, 545], [505, 545]],
            "pair_gap": 28,
        },
    }


def _xor(pair: list[list[int]]) -> list[int]:
    return [int(a != b) for a, b in zip(pair[0], pair[1])]


def _scores(scene: dict) -> list[int]:
    reference = scene["reference"]
    pairs = scene["pairs"]
    return [sum(a != b for a, b in zip(reference, _xor(pair))) for pair in pairs]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = min(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if len(winners) != 1:
        raise ValueError("scene has no unique best candidate")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = sorted(_scores(scene))
    return scores[0] != 0 or scores[1] - scores[0] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _draw_grid(
    draw: ImageDraw.ImageDraw,
    origin: Iterable[int],
    bits: list[int],
    cell: int,
    fill: tuple[int, int, int],
    ink: tuple[int, int, int],
    empty: tuple[int, int, int],
) -> None:
    x0, y0 = (int(value) for value in origin)
    extent = GRID_N * cell
    draw.rectangle((x0, y0, x0 + extent, y0 + extent), fill=empty)
    for r in range(GRID_N):
        for c in range(GRID_N):
            if bits[r * GRID_N + c]:
                left = x0 + c * cell + 2
                top = y0 + r * cell + 2
                draw.rectangle((left, top, left + cell - 4, top + cell - 4), fill=fill)
    for k in range(GRID_N + 1):
        x = x0 + k * cell
        y = y0 + k * cell
        draw.line((x, y0, x, y0 + extent), fill=ink, width=2)
        draw.line((x0, y, x0 + extent, y), fill=ink, width=2)


def render(scene: dict) -> Image.Image:
    palette = {key: tuple(value) for key, value in scene["palette"].items()}
    layout = scene["layout"]
    reference = scene["reference"]
    pairs = scene["pairs"]
    image = Image.new("RGB", tuple(layout["canvas"]), palette["background"])
    draw = ImageDraw.Draw(image)
    title_font = _font(24, bold=True)
    label_font = _font(22, bold=True)
    small_font = _font(16)
    draw.text((48, 24), "XOR STENCIL SEARCH", fill=palette["ink"], font=title_font)
    draw.text((342, 62), "REFERENCE", fill=palette["ink"], font=small_font)
    _draw_grid(
        draw,
        layout["reference_origin"],
        reference,
        int(layout["cell"]),
        palette["ink"],
        palette["ink"],
        palette["empty"],
    )
    draw.text(
        (532, 128),
        "filled when exactly one is filled",
        fill=palette["ink"],
        font=small_font,
    )
    for index, (origin, pair) in enumerate(zip(layout["pair_origins"], pairs)):
        x, y = origin
        cell = int(layout["cell"])
        grid_width = GRID_N * cell
        second_x = x + grid_width + int(layout["pair_gap"])
        draw.text((x - 38, y + 34), LABELS[index], fill=palette["ink"], font=label_font)
        _draw_grid(draw, (x, y), pair[0], cell, palette["cyan"], palette["ink"], palette["empty"])
        draw.text((x + grid_width + 8, y + 34), "⊕", fill=palette["ink"], font=label_font)
        _draw_grid(
            draw,
            (second_x, y),
            pair[1],
            cell,
            palette["amber"],
            palette["ink"],
            palette["empty"],
        )
        draw.text(
            (x, y + grid_width + 10),
            "one filled → filled    both filled → blank",
            fill=palette["ink"],
            font=small_font,
        )
    return image
