"""Independent pixel-only inverse arm for XOR stencil search."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _long_dark_segments(arr: np.ndarray) -> list[tuple[int, int, int]]:
    dark = (arr[:, :, 0] < 75) & (arr[:, :, 1] < 75) & (arr[:, :, 2] < 85)
    segments: list[tuple[int, int, int]] = []
    for y, row in enumerate(dark):
        padded = np.concatenate(([False], row, [False])).astype(np.int8)
        changes = np.diff(padded)
        starts = np.flatnonzero(changes == 1)
        ends = np.flatnonzero(changes == -1) - 1
        for start, end in zip(starts, ends):
            if 90 <= end - start + 1 <= 110:
                segments.append((int(start), int(end), y))
    return segments


def _grid_boxes(arr: np.ndarray) -> list[tuple[int, int, int, int]]:
    segments = _long_dark_segments(arr)
    boxes = []
    counts = Counter((x0, x1) for x0, x1, _ in segments)
    for x0, x1 in counts:
        ys = [y for a, b, y in segments if a == x0 and b == x1]
        clusters: list[list[int]] = []
        for y in sorted(ys):
            if not clusters or y - clusters[-1][-1] > 30:
                clusters.append([y])
            else:
                clusters[-1].append(y)
        for cluster in clusters:
            if len(cluster) >= 8 and 90 <= max(cluster) - min(cluster) <= 105:
                boxes.append((x0, min(cluster), x1, max(cluster)))
    # Duplicate line widths can create near-identical candidates; consolidate.
    boxes.sort(key=lambda box: (box[1], box[0]))
    unique: list[tuple[int, int, int, int]] = []
    for box in boxes:
        if not any(max(abs(a - b) for a, b in zip(box, old)) <= 2 for old in unique):
            unique.append(box)
    return unique


def _decode(arr: np.ndarray, box: tuple[int, int, int, int]) -> list[int]:
    x0, y0, x1, y1 = box
    cell_x = (x1 - x0) / 5.0
    cell_y = (y1 - y0) / 5.0
    bits = []
    for r in range(5):
        for c in range(5):
            x = int(round(x0 + (c + 0.5) * cell_x))
            y = int(round(y0 + (r + 0.5) * cell_y))
            pixel = arr[y, x].astype(np.int16)
            bits.append(int(np.linalg.norm(pixel - np.array([255, 255, 255])) > 80))
    return bits


def _measure(image: Image.Image) -> tuple[str, list[int]]:
    arr = np.asarray(image.convert("RGB"))
    boxes = _grid_boxes(arr)
    if len(boxes) != 9:
        raise ValueError(f"expected nine visible grids, found {len(boxes)}")
    boxes.sort(key=lambda box: (box[1], box[0]))
    reference = _decode(arr, boxes[0])
    candidate_boxes = boxes[1:]
    # The eight stencil grids occupy two visible rows; reading order pairs neighbors.
    candidate_boxes.sort(key=lambda box: (round(box[1] / 20), box[0]))
    scores = []
    for index in range(0, 8, 2):
        left = _decode(arr, candidate_boxes[index])
        right = _decode(arr, candidate_boxes[index + 1])
        output = [int(a != b) for a, b in zip(left, right)]
        scores.append(sum(a != b for a, b in zip(output, reference)))
    best = min(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if best != 0 or len(winners) != 1:
        raise ValueError(f"no unique exact XOR match; scores={scores}")
    return LABELS[winners[0]], scores


def decision_from_image(image: Image.Image) -> str:
    return _measure(image)[0]


def scores_from_image(image: Image.Image) -> list[int]:
    """Public diagnostic used by candidate tests; still accepts pixels only."""
    return _measure(image)[1]
