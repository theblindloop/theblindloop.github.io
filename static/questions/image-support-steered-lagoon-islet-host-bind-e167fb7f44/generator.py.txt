"""Renderer for the lagoon / islet host-binding correspondence world.

A cream board carries scattered decorative coloured squares and exactly one
medium LAGOON: a filled water-coloured rectangle whose whole interior is the
decisive region.

Inside the lagoon sit exactly three solid coloured ISLANDS, one per used
quadrant of the lagoon's interior. Two of the islands each contain one LAKE, a
water-coloured rectangular hole cut in the island. The remaining island has no
lake and instead carries two small coloured PATCHES painted on its body. Two
further patches of the same size family float directly on the lagoon water.

Exactly one of the two lakes contains a small coloured ISLET, and that islet is
the only region in the whole picture at containment depth four
(board -> lagoon -> island -> lake -> islet). Every other small patch sits at
depth two (on the water) or depth three (on an island's body), so patch size,
patch colour and patch count carry no information: only the containment chain
does.

The answer is the colour of the ISLAND that hosts the lake holding the islet.
The islet's own colour is always different from that answer, so the landmark
cannot be read off directly - the solver has to walk outwards from the islet
through its lake to its host island. Recovering that chain requires the whole
lagoon interior, which is one connected area-like region of medium span.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "lagoon_islet_host_bind-0.1.0"

CANVAS_W = 640
CANVAS_H = 640

CREAM = (238, 231, 210)
WATER = (108, 170, 202)

COLOR_NAMES = ("red", "blue", "green", "purple")
COLOR_RGB = {
    "red": (198, 56, 48),
    "blue": (72, 76, 208),
    "green": (44, 152, 66),
    "purple": (132, 54, 158),
}

LAGOON = 192          # lagoon side length in pixels
WATER_BORDER = 10     # water margin between the lagoon edge and its quadrants
SLOT = 86             # quadrant side length inside the lagoon
BOARD_MARGIN = 28     # smallest distance from the lagoon to the image edge

ISLAND_SIZES = (62, 66, 70)
LAKE_INSETS = (20, 24, 28)   # island_size - lake_size
ISLET_SIZES = (12, 16)
BODY_PATCH = 14
WATER_PATCHES = (12, 16)

QUARANTINE_MARGIN = 7.0

DECOR_COUNT = (13, 17)
DECOR_SIZES = (8, 10, 12, 14)


def _tight(seed: int) -> bool:
    """Deterministically force a near-boundary islet clearance on some seeds."""
    return seed % 6 == 3


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    lagoon_x = rng.randint(BOARD_MARGIN, CANVAS_W - LAGOON - BOARD_MARGIN)
    lagoon_y = rng.randint(BOARD_MARGIN, CANVAS_H - LAGOON - BOARD_MARGIN)

    # three of the four lagoon quadrants hold islands; the fourth stays water
    slots = [(0, 0), (1, 0), (0, 1), (1, 1)]
    rng.shuffle(slots)
    free_slot = slots[3]
    island_slots = sorted(slots[:3])

    # the answer colour cycles with the seed so every class is covered
    host_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    others = [name for name in COLOR_NAMES if name != host_color]
    rng.shuffle(others)
    lake_only_color, dry_color = others[0], others[1]

    host_index = rng.randrange(3)          # which island slot hosts the islet
    dry_index = rng.choice([i for i in range(3) if i != host_index])

    islands = []
    for index, slot in enumerate(island_slots):
        size = rng.choice(ISLAND_SIZES)
        free = SLOT - size
        offset = (rng.randint(6, free - 6), rng.randint(6, free - 6))
        if index == host_index:
            color, kind = host_color, "host"
        elif index == dry_index:
            color, kind = dry_color, "dry"
        else:
            color, kind = lake_only_color, "lake"
        islands.append(
            {"slot": list(slot), "size": size, "offset": list(offset), "color": color, "kind": kind}
        )

    tight = _tight(seed)
    clearance = None
    for island in islands:
        if island["kind"] == "dry":
            free = island["size"] - BODY_PATCH
            # a body patch must contrast with the island it is painted on
            visible = [name for name in COLOR_NAMES if name != island["color"]]
            island["patches"] = [
                {
                    "size": BODY_PATCH,
                    "offset": [rng.randint(9, 14), rng.randint(9, 14)],
                    "color": rng.choice(visible),
                },
                {
                    "size": BODY_PATCH,
                    "offset": [rng.randint(free - 14, free - 9), rng.randint(free - 14, free - 9)],
                    "color": rng.choice(visible),
                },
            ]
            continue

        inset = rng.choice(LAKE_INSETS)
        lake_size = island["size"] - inset
        lake_offset = [rng.randint(10, inset - 10), rng.randint(10, inset - 10)]
        island["lake"] = {"size": lake_size, "offset": lake_offset}
        island["patches"] = []
        if island["kind"] != "host":
            continue

        islet_size = rng.choice(ISLET_SIZES)
        span = lake_size - islet_size
        low = 5 if tight else 7
        high = 6 if tight else min(12, span // 2)
        clearance = rng.randint(low, max(low, high))
        islet_color = rng.choice([name for name in COLOR_NAMES if name != host_color])
        island["islet"] = {
            "size": islet_size,
            "offset": [clearance, rng.randint(clearance, span - clearance)],
            "color": islet_color,
        }

    # two patches floating on open lagoon water inside the unused quadrant
    water_patches = []
    for band in (0, 1):
        size = rng.choice(WATER_PATCHES)
        y_low = 9 if band == 0 else 50
        y_high = 17 if band == 0 else SLOT - size - 9
        water_patches.append(
            {
                "size": size,
                "offset": [rng.randint(9, SLOT - size - 9), rng.randint(y_low, y_high)],
                "color": rng.choice(COLOR_NAMES),
            }
        )

    decoration = _sample_decoration(rng, lagoon_x, lagoon_y)

    return {
        "lagoon": [lagoon_x, lagoon_y],
        "islands": islands,
        "free_slot": list(free_slot),
        "water_patches": water_patches,
        "decoration": decoration,
        "host_color": host_color,
        "islet_clearance": clearance,
        "tight": tight,
    }


def _sample_decoration(rng: random.Random, lagoon_x: int, lagoon_y: int) -> list[dict]:
    forbidden = (lagoon_x - 9, lagoon_y - 9, lagoon_x + LAGOON + 9, lagoon_y + LAGOON + 9)
    placed: list[tuple[int, int, int, int]] = []
    decoration: list[dict] = []
    wanted = rng.randint(*DECOR_COUNT)
    for _ in range(wanted * 30):
        if len(decoration) >= wanted:
            break
        size = rng.choice(DECOR_SIZES)
        x = rng.randint(10, CANVAS_W - size - 10)
        y = rng.randint(10, CANVAS_H - size - 10)
        box = (x, y, x + size, y + size)
        if box[0] < forbidden[2] and forbidden[0] < box[2] and box[1] < forbidden[3] and forbidden[1] < box[3]:
            continue
        clash = False
        for other in placed:
            if (
                box[0] - 7 < other[2]
                and other[0] < box[2] + 7
                and box[1] - 7 < other[3]
                and other[1] < box[3] + 7
            ):
                clash = True
                break
        if clash:
            continue
        placed.append(box)
        decoration.append({"box": [x, y, size], "color": rng.choice(COLOR_NAMES)})
    return decoration


def _island_origin(scene: dict, island: dict) -> tuple[int, int]:
    lx, ly = scene["lagoon"]
    sx, sy = island["slot"]
    return (
        lx + WATER_BORDER + sx * SLOT + island["offset"][0],
        ly + WATER_BORDER + sy * SLOT + island["offset"][1],
    )


def _rect(draw: ImageDraw.ImageDraw, x: int, y: int, w: int, h: int, rgb: tuple[int, int, int]) -> None:
    draw.rectangle([x, y, x + w - 1, y + h - 1], fill=rgb)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), CREAM)
    draw = ImageDraw.Draw(image)

    # decoration is consumed as a set: draw it in a canonical order
    for item in sorted(scene["decoration"], key=lambda d: (d["box"][1], d["box"][0], d["color"])):
        x, y, size = item["box"]
        _rect(draw, x, y, size, size, COLOR_RGB[item["color"]])

    lx, ly = scene["lagoon"]
    _rect(draw, lx, ly, LAGOON, LAGOON, WATER)

    fx, fy = scene["free_slot"]
    free_x = lx + WATER_BORDER + fx * SLOT
    free_y = ly + WATER_BORDER + fy * SLOT
    for patch in sorted(scene["water_patches"], key=lambda p: (p["offset"][1], p["offset"][0])):
        px, py = patch["offset"]
        _rect(draw, free_x + px, free_y + py, patch["size"], patch["size"], COLOR_RGB[patch["color"]])

    for island in sorted(scene["islands"], key=lambda i: (i["slot"][1], i["slot"][0])):
        ox, oy = _island_origin(scene, island)
        size = island["size"]
        _rect(draw, ox, oy, size, size, COLOR_RGB[island["color"]])
        for patch in sorted(island.get("patches", []), key=lambda p: (p["offset"][1], p["offset"][0])):
            px, py = patch["offset"]
            _rect(draw, ox + px, oy + py, patch["size"], patch["size"], COLOR_RGB[patch["color"]])
        lake = island.get("lake")
        if lake is None:
            continue
        kx, ky = lake["offset"]
        lake_x, lake_y = ox + kx, oy + ky
        _rect(draw, lake_x, lake_y, lake["size"], lake["size"], WATER)
        islet = island.get("islet")
        if islet is None:
            continue
        ix, iy = islet["offset"]
        _rect(draw, lake_x + ix, lake_y + iy, islet["size"], islet["size"], COLOR_RGB[islet["color"]])

    return image


def _host(scene: dict) -> dict:
    hosts = [island for island in scene["islands"] if island.get("islet") is not None]
    if len(hosts) != 1:
        raise ValueError("a scene must contain exactly one islet")
    return hosts[0]


def analytic_gold(scene: dict) -> str:
    """The colour of the island whose lake contains the depth-four islet."""
    return _host(scene)["color"]


def margin(scene: dict) -> float:
    """Smallest pixel clearance in the decisive containment chain.

    The runner-up reading is the other lake-bearing island: it would win if the
    islet stopped being strictly inside its lake. The controlling quantity is
    therefore the islet's smallest gap to the lake rim, capped by the lake's own
    smallest gap to its island rim.
    """
    host = _host(scene)
    islet, lake = host["islet"], host["lake"]
    span = lake["size"] - islet["size"]
    islet_gap = min(islet["offset"][0], islet["offset"][1], span - islet["offset"][0], span - islet["offset"][1])
    inset = host["size"] - lake["size"]
    lake_gap = min(lake["offset"][0], lake["offset"][1], inset - lake["offset"][0], inset - lake["offset"][1])
    return float(min(islet_gap, lake_gap))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Stored decoration and patch lists are consumed as sets.

    The renderer sorts every such list into a canonical drawing order, so
    reversing the stored order re-draws a byte-identical raster and cannot move
    the analytic gold.
    """
    reversed_islands = []
    for island in scene["islands"]:
        twin = dict(island)
        twin["patches"] = list(reversed(island.get("patches", [])))
        reversed_islands.append(twin)
    return [
        ("decoration_order", {**scene, "decoration": list(reversed(scene["decoration"]))}),
        ("water_patch_order", {**scene, "water_patches": list(reversed(scene["water_patches"]))}),
        ("island_and_body_patch_order", {**scene, "islands": list(reversed(reversed_islands))}),
    ]
