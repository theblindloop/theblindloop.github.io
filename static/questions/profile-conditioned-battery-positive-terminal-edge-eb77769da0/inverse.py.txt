"""Pixel-only inverse for the visible red positive-terminal side.

This module intentionally shares no imports, scene state, geometry constants, or
analytic-gold code with the renderer. It recovers two masks from the final RGB
raster: saturated red terminal pixels and the large pale neutral body interior.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("required visible region is absent")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (rgb[:, :, 0] > 155) & (rgb[:, :, 0] > rgb[:, :, 1] * 1.55) & (
        rgb[:, :, 0] > rgb[:, :, 2] * 1.45
    ) & (rgb[:, :, 1] < 125)
    # The body is deliberately pale and nearly neutral; this rule is based only
    # on visible pixels and tolerates antialiasing around its boundary.
    spread = rgb.max(axis=2) - rgb.min(axis=2)
    pale_body = (
        (rgb.mean(axis=2) > 185)
        & (rgb.mean(axis=2) < 242)
        & (spread < 24)
    )
    if int(red.sum()) < 450 or int(pale_body.sum()) < 9000:
        raise ValueError("terminal or battery body lacks sufficient visible evidence")

    rx, ry = _centroid(red)
    bx, by = _centroid(pale_body)
    scores = {
        "right": rx - bx,
        "left": bx - rx,
        "bottom": ry - by,
        "top": by - ry,
    }
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] < 65 or ordered[0][1] - ordered[1][1] < 55:
        raise ValueError("positive-terminal side is ambiguous")
    return ordered[0][0]
