"""Render a familiar battery icon whose visible polarity can oppose its prior."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
SCALE = 3
EDGES = ("top", "right", "bottom", "left")
BODY_FILL = (226, 230, 234)
OUTLINE = (38, 43, 52)
POSITIVE = (220, 48, 46)
NEGATIVE = (43, 103, 190)
BAR_PALETTES = (
    ((35, 163, 105), (238, 174, 40)),
    ((126, 87, 194), (35, 163, 105)),
    ((238, 174, 40), (126, 87, 194)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    edge = EDGES[int(seed) % len(EDGES)]
    # A base horizontal battery is rotated for vertical answers, preserving ink.
    long_side = rng.randrange(246, 291, 2)
    short_side = rng.randrange(142, 177, 2)
    if edge in ("left", "right"):
        body_size = [long_side, short_side]
    else:
        body_size = [short_side, long_side]
    center = [rng.randint(244, 268), rng.randint(244, 268)]
    terminal_depth = rng.randint(30, 42)
    terminal_span = rng.randrange(58, 81, 2)
    charge_count = 1 + (rng.randrange(4))
    charge_colors = list(BAR_PALETTES[rng.randrange(len(BAR_PALETTES))])
    if rng.random() < 0.5:
        charge_colors.reverse()
    return {
        "positive_edge": edge,
        "center": center,
        "body_size": body_size,
        "terminal_depth": terminal_depth,
        "terminal_span": terminal_span,
        "palette": [list(color) for color in charge_colors],
        "charge_bars": charge_count,
    }


def _scaled_box(box: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(round(value * SCALE) for value in box)  # type: ignore[return-value]


def _terminal_box(
    edge: str,
    center: tuple[float, float],
    body_size: tuple[float, float],
    depth: float,
    span: float,
) -> tuple[float, float, float, float]:
    cx, cy = center
    width, height = body_size
    left, top = cx - width / 2, cy - height / 2
    right, bottom = cx + width / 2, cy + height / 2
    if edge == "right":
        return right - 2, cy - span / 2, right + depth, cy + span / 2
    if edge == "left":
        return left - depth, cy - span / 2, left + 2, cy + span / 2
    if edge == "top":
        return cx - span / 2, top - depth, cx + span / 2, top + 2
    if edge == "bottom":
        return cx - span / 2, bottom - 2, cx + span / 2, bottom + depth
    raise ValueError(f"unknown edge {edge!r}")


def _opposite(edge: str) -> str:
    return EDGES[(EDGES.index(edge) + 2) % 4]


def render(scene: dict[str, Any]) -> Image.Image:
    edge = str(scene["positive_edge"])
    center = tuple(float(v) for v in scene["center"])
    body_size = tuple(float(v) for v in scene["body_size"])
    depth = float(scene["terminal_depth"])
    span = float(scene["terminal_span"])
    palette = [tuple(int(c) for c in color) for color in scene["palette"]]
    charge_bars = int(scene["charge_bars"])

    image = Image.new("RGB", (CANVAS * SCALE, CANVAS * SCALE), (250, 249, 247))
    draw = ImageDraw.Draw(image)
    cx, cy = center
    width, height = body_size
    body = (cx - width / 2, cy - height / 2, cx + width / 2, cy + height / 2)

    # Terminals are drawn first so their two-pixel overlap is hidden by the body.
    positive_box = _terminal_box(edge, center, body_size, depth, span)
    negative_box = _terminal_box(_opposite(edge), center, body_size, depth, span)
    draw.rounded_rectangle(_scaled_box(negative_box), radius=7 * SCALE, fill=NEGATIVE)
    draw.rounded_rectangle(_scaled_box(positive_box), radius=7 * SCALE, fill=POSITIVE)
    draw.rounded_rectangle(
        _scaled_box(body),
        radius=18 * SCALE,
        fill=BODY_FILL,
        outline=OUTLINE,
        width=8 * SCALE,
    )

    # Internal charge bars are centered and symmetric, so their count and palette
    # cannot identify which end is positive. Their long axis follows the body.
    inner_w, inner_h = width - 56, height - 54
    horizontal = width >= height
    gap = 10
    if horizontal:
        bar_w = min(42, (inner_w - gap * (charge_bars - 1)) / charge_bars)
        total = charge_bars * bar_w + (charge_bars - 1) * gap
        start = cx - total / 2
        for index in range(charge_bars):
            x0 = start + index * (bar_w + gap)
            box = (x0, cy - inner_h / 2, x0 + bar_w, cy + inner_h / 2)
            draw.rounded_rectangle(_scaled_box(box), radius=6 * SCALE, fill=palette[index % 2])
    else:
        bar_h = min(42, (inner_h - gap * (charge_bars - 1)) / charge_bars)
        total = charge_bars * bar_h + (charge_bars - 1) * gap
        start = cy - total / 2
        for index in range(charge_bars):
            y0 = start + index * (bar_h + gap)
            box = (cx - inner_w / 2, y0, cx + inner_w / 2, y0 + bar_h)
            draw.rounded_rectangle(_scaled_box(box), radius=6 * SCALE, fill=palette[index % 2])

    # A high-resolution + cutout makes the visible polarity explicit without
    # being required by the pixel inverse.
    px0, py0, px1, py1 = positive_box
    pcx, pcy = (px0 + px1) / 2, (py0 + py1) / 2
    arm = max(10, min(depth, span) * 0.27)
    thick = max(3, min(depth, span) * 0.10)
    draw.rectangle(_scaled_box((pcx - arm, pcy - thick, pcx + arm, pcy + thick)), fill="white")
    draw.rectangle(_scaled_box((pcx - thick, pcy - arm, pcx + thick, pcy + arm)), fill="white")

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["positive_edge"])


def margin(scene: dict[str, Any]) -> float:
    # The score gap between the intended axis and runner-up axis is governed by
    # terminal displacement from center; symmetric rotations preserve it.
    width, height = (float(v) for v in scene["body_size"])
    depth = float(scene["terminal_depth"])
    span = float(scene["terminal_span"])
    edge = str(scene["positive_edge"])
    axial = (width if edge in ("left", "right") else height) / 2 + depth / 2
    runner_up = span / 2
    return round(axial - runner_up, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 75.0 or int(scene["charge_bars"]) not in range(1, 5)


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
