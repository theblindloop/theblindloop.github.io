"""Renderer and analytic contract for the traced-fork turn-choice world.

The image contains one dark, non-crossing Y-shaped route.  A viewer must first
find the purple-ringed fork on the route from the green source, then compare the
absolute turn of the two outgoing arms with the direction of travel into that
fork.  The independent inverse performs the same operations from the raster;
it never receives this module's scene dictionary.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "junction_branch_turn_choice-0.1.0"
WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (248, 247, 242)
ROUTE = (39, 54, 73)
SOURCE = (44, 137, 91)
JUNCTION = (126, 89, 181)
GLYPH = (236, 171, 45)
DISTRACTOR = (207, 205, 198)
MARKER_FILL = (249, 247, 239)
TERMINAL_COLORS = (
    (218, 75, 83),
    (36, 150, 163),
    (224, 143, 45),
    (78, 119, 190),
)
QUARANTINE_GAP_DEG = 10.0
GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
}


def _unit(angle: float) -> tuple[float, float]:
    radians = math.radians(float(angle))
    return (math.cos(radians), math.sin(radians))


def _point_from(origin: tuple[float, float], angle: float, length: float) -> tuple[float, float]:
    ux, uy = _unit(angle)
    return (origin[0] + ux * float(length), origin[1] + uy * float(length))


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _sub(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    return (a[0] - b[0], a[1] - b[1])


def _point_segment_distance(
    point: tuple[float, float], a: tuple[float, float], b: tuple[float, float]
) -> float:
    ab = _sub(b, a)
    denom = ab[0] * ab[0] + ab[1] * ab[1]
    if denom <= 1e-12:
        return _distance(point, a)
    t = max(0.0, min(1.0, ((point[0] - a[0]) * ab[0] + (point[1] - a[1]) * ab[1]) / denom))
    q = (a[0] + t * ab[0], a[1] + t * ab[1])
    return _distance(point, q)


def _angle_delta(a: float, b: float) -> float:
    """Signed shortest difference a-b in degrees in [-180, 180)."""
    return (float(a) - float(b) + 180.0) % 360.0 - 180.0


def _turn_magnitudes(scene: dict) -> tuple[float, float]:
    incoming = float(scene["incoming_heading"])
    turn_a = abs(_angle_delta(float(scene["branch_a_heading"]), incoming))
    turn_b = abs(_angle_delta(float(scene["branch_b_heading"]), incoming))
    return turn_a, turn_b


def _main_segments(scene: dict) -> list[tuple[tuple[float, float], tuple[float, float]]]:
    source = (float(scene["source_center"][0]), float(scene["source_center"][1]))
    junction = (float(scene["junction_center"][0]), float(scene["junction_center"][1]))
    branch_a_end = _point_from(
        junction, float(scene["branch_a_heading"]), float(scene["branch_a_length"])
    )
    branch_b_end = _point_from(
        junction, float(scene["branch_b_heading"]), float(scene["branch_b_length"])
    )
    incoming_end = _point_from(
        source, float(scene["incoming_heading"]), float(scene["incoming_length"])
    )
    return [(source, incoming_end), (junction, branch_a_end), (junction, branch_b_end)]


def _distractor_is_clear(
    points: list[tuple[float, float]],
    main_segments: list[tuple[tuple[float, float], tuple[float, float]]],
    anchors: list[tuple[float, float]],
) -> bool:
    for a, b in zip(points, points[1:]):
        if any(_point_segment_distance(q, a, b) < 27.0 for seg in main_segments for q in seg):
            return False
        if any(_point_segment_distance(anchor, a, b) < 30.0 for anchor in anchors):
            return False
    return all(35.0 <= x <= 477.0 and 35.0 <= y <= 477.0 for x, y in points)


def _sample_distractors(
    rng: random.Random,
    main_segments: list[tuple[tuple[float, float], tuple[float, float]]],
    anchors: list[tuple[float, float]],
) -> list[list[list[float]]]:
    result: list[list[list[float]]] = []
    for _ in range(3):
        for _attempt in range(3000):
            x = rng.uniform(58.0, 454.0)
            y = rng.uniform(58.0, 454.0)
            heading = rng.uniform(0.0, 360.0)
            p0 = (x, y)
            p1 = _point_from(p0, heading + rng.uniform(-22.0, 22.0), rng.uniform(38.0, 68.0))
            p2 = _point_from(p1, heading + rng.uniform(-55.0, 55.0), rng.uniform(34.0, 62.0))
            points = [p0, p1, p2]
            if _distractor_is_clear(points, main_segments, anchors):
                result.append([[round(px, 3), round(py, 3)] for px, py in points])
                break
        else:
            raise RuntimeError("could not place a disconnected distractor")
    return result


def sample_scene(seed: int) -> dict:
    """Return one deterministic, balanced, render-robust scene."""
    rng = random.Random(int(seed))
    scene_index = int(seed) % 1000
    requested_winner = "A" if scene_index % 2 == 0 else "B"
    near_boundary = scene_index % 8 in (0, 1)

    for _attempt in range(5000):
        junction = (rng.uniform(162.0, 350.0), rng.uniform(162.0, 350.0))
        incoming_heading = rng.uniform(0.0, 360.0)
        incoming_length = rng.uniform(132.0, 176.0)
        source = _point_from(junction, incoming_heading + 180.0, incoming_length)

        small_turn = rng.uniform(22.0, 41.0)
        gap = rng.uniform(5.0, 9.0) if near_boundary else rng.uniform(18.0, 46.0)
        large_turn = small_turn + gap
        small_sign = rng.choice((-1.0, 1.0))
        large_sign = small_sign if rng.random() < 0.55 else -small_sign
        a_is_small = rng.random() < 0.5
        turn_a = small_turn if a_is_small else large_turn
        turn_b = large_turn if a_is_small else small_turn
        sign_a = small_sign if a_is_small else large_sign
        sign_b = large_sign if a_is_small else small_sign
        branch_a_heading = incoming_heading + sign_a * turn_a
        branch_b_heading = incoming_heading + sign_b * turn_b
        branch_a_length = rng.uniform(126.0, 176.0)
        branch_b_length = rng.uniform(126.0, 176.0)
        branch_a_end = _point_from(junction, branch_a_heading, branch_a_length)
        branch_b_end = _point_from(junction, branch_b_heading, branch_b_length)

        terminal_radius = rng.uniform(14.5, 17.0)
        source_radius = rng.uniform(17.0, 20.5)
        junction_radius = rng.uniform(17.0, 20.5)
        anchors = [source, junction, branch_a_end, branch_b_end]
        if any(not (42.0 <= x <= 470.0 and 42.0 <= y <= 470.0) for x, y in anchors):
            continue
        if _distance(source, junction) < source_radius + junction_radius + 70.0:
            continue
        if _distance(branch_a_end, branch_b_end) < 2.0 * terminal_radius + 28.0:
            continue
        if _distance(source, branch_a_end) < source_radius + terminal_radius + 28.0:
            continue
        if _distance(source, branch_b_end) < source_radius + terminal_radius + 28.0:
            continue

        terminal_colors = rng.sample(list(TERMINAL_COLORS), 2)
        # The winner identity is balanced independently of which geometric arm
        # is straighter.  Labels are drawn at the actual terminal centers.
        if a_is_small:
            assignment = {"branch_a": requested_winner, "branch_b": "B" if requested_winner == "A" else "A"}
        else:
            assignment = {"branch_b": requested_winner, "branch_a": "B" if requested_winner == "A" else "A"}

        provisional = {
            "source_center": [round(source[0], 3), round(source[1], 3)],
            "junction_center": [round(junction[0], 3), round(junction[1], 3)],
            "incoming_heading": round(incoming_heading % 360.0, 5),
            "incoming_length": round(incoming_length, 3),
            "branch_a_heading": round(branch_a_heading % 360.0, 5),
            "branch_b_heading": round(branch_b_heading % 360.0, 5),
            "branch_a_length": round(branch_a_length, 3),
            "branch_b_length": round(branch_b_length, 3),
            "terminal_label_assignment": assignment,
            "stroke_width": round(rng.uniform(7.0, 9.0), 3),
            "source_radius": round(source_radius, 3),
            "junction_radius": round(junction_radius, 3),
            "terminal_radius": round(terminal_radius, 3),
        }
        # Use the rounded scene geometry for the clearance check too, so every
        # manifest can be re-rendered exactly from its stored dictionary.
        main_segments = _main_segments({**provisional, "distractors": []})
        try:
            distractors = _sample_distractors(rng, main_segments, anchors)
        except RuntimeError:
            continue
        palette = {
            "background": list(BACKGROUND),
            "route": list(ROUTE),
            "source": list(SOURCE),
            "junction": list(JUNCTION),
            "glyph": list(GLYPH),
            "marker_fill": list(MARKER_FILL),
            "terminal_a": list(terminal_colors[0]),
            "terminal_b": list(terminal_colors[1]),
            "distractor": list(DISTRACTOR),
        }
        provisional["distractors"] = distractors
        provisional["palette"] = palette
        return {
            "width": WIDTH,
            "height": HEIGHT,
            **provisional,
        }
    raise RuntimeError("could not sample a legal traced fork")


def _scaled_points(points: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    return [(float(point[0]) * SCALE, float(point[1]) * SCALE) for point in points]


def _circle(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    radius: float,
    *,
    fill,
    outline,
    width: int,
) -> None:
    cx, cy = center[0] * SCALE, center[1] * SCALE
    r = float(radius) * SCALE
    draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=fill, outline=outline, width=max(1, round(width * SCALE)))


def _draw_glyph(draw: ImageDraw.ImageDraw, center: tuple[float, float], label: str, color) -> None:
    pattern = GLYPHS[label]
    cell = 3.0 * SCALE
    x0 = center[0] * SCALE - 2.5 * cell
    y0 = center[1] * SCALE - 3.5 * cell
    for row, line in enumerate(pattern):
        for col, bit in enumerate(line):
            if bit == "1":
                left = round(x0 + col * cell)
                top = round(y0 + row * cell)
                right = round(left + cell - 0.35 * SCALE)
                bottom = round(top + cell - 0.35 * SCALE)
                draw.rectangle((left, top, right, bottom), fill=color)


def render(scene: dict) -> Image.Image:
    """Render only from the serializable scene dictionary."""
    width, height = int(scene["width"]), int(scene["height"])
    palette = scene["palette"]
    background = tuple(int(value) for value in palette["background"])
    route_color = tuple(int(value) for value in palette["route"])
    source_color = tuple(int(value) for value in palette["source"])
    junction_color = tuple(int(value) for value in palette["junction"])
    glyph_color = tuple(int(value) for value in palette["glyph"])
    marker_fill = tuple(int(value) for value in palette["marker_fill"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)

    for distractor in scene["distractors"]:
        draw.line(
            _scaled_points(distractor),
            fill=tuple(int(value) for value in palette["distractor"]),
            width=2 * SCALE,
            joint="curve",
        )

    source = (float(scene["source_center"][0]), float(scene["source_center"][1]))
    junction = (float(scene["junction_center"][0]), float(scene["junction_center"][1]))
    incoming_end = _point_from(source, float(scene["incoming_heading"]), float(scene["incoming_length"]))
    branch_a_end = _point_from(junction, float(scene["branch_a_heading"]), float(scene["branch_a_length"]))
    branch_b_end = _point_from(junction, float(scene["branch_b_heading"]), float(scene["branch_b_length"]))
    route_width = max(1, round(float(scene["stroke_width"]) * SCALE))
    draw.line(_scaled_points([source, incoming_end]), fill=route_color, width=route_width, joint="curve")
    draw.line(_scaled_points([junction, branch_a_end]), fill=route_color, width=route_width, joint="curve")
    draw.line(_scaled_points([junction, branch_b_end]), fill=route_color, width=route_width, joint="curve")
    _circle(draw, junction, float(scene["junction_radius"]) * 0.42, fill=route_color, outline=route_color, width=1)

    _circle(
        draw,
        source,
        float(scene["source_radius"]),
        fill=marker_fill,
        outline=source_color,
        width=4,
    )
    _circle(
        draw,
        junction,
        float(scene["junction_radius"]),
        fill=marker_fill,
        outline=junction_color,
        width=4,
    )

    for branch_name, end in (("branch_a", branch_a_end), ("branch_b", branch_b_end)):
        marker_color = tuple(int(value) for value in palette[branch_name.replace("branch_", "terminal_")])
        label = str(scene["terminal_label_assignment"][branch_name])
        _circle(
            draw,
            end,
            float(scene["terminal_radius"]),
            fill=marker_fill,
            outline=marker_color,
            width=4,
        )
        _draw_glyph(draw, end, label, glyph_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the label on the branch with the smaller rendered turn."""
    turn_a, turn_b = _turn_magnitudes(scene)
    branch = "branch_a" if turn_a < turn_b else "branch_b"
    return str(scene["terminal_label_assignment"][branch])


def margin(scene: dict) -> float:
    """Continuous runner-up gap in degrees."""
    turn_a, turn_b = _turn_magnitudes(scene)
    return abs(turn_a - turn_b)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unrendered aliases in this explicit scene parametrization."""
    return []
