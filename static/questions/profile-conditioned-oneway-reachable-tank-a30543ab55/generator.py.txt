"""One-way lane network renderer for `oneway_reachable_tank`.

Latent scene: junction disks on a jittered 4x3 lattice, joined by straight gray
lanes.  Every lane carries exactly one black arrowhead at its midpoint that
declares the only direction of travel allowed on that lane.  One junction is
green; three junctions are colored tanks (blue, orange, purple).  The undirected
lane graph is always connected, so all three tanks are joined to green by lanes
and only the arrow directions decide which tank can actually be reached.

The analytic label reads only `node_roles` and `edges` (both drawn: roles are
painted as disk colors, edges are painted as lanes plus one oriented arrowhead).
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "oneway_reachable_tank-renderer-0.1.0"

BG = (255, 255, 255)
LANE = (176, 176, 176)
ARROW = (0, 0, 0)
NODE = (110, 110, 110)
ROLE_COLORS = {
    "gray": NODE,
    "green": (0, 150, 60),
    "blue": (30, 90, 220),
    "orange": (235, 130, 20),
    "purple": (140, 50, 170),
}
TANKS = ("blue", "orange", "purple")

COLS, ROWS = 4, 3
PITCH = 150
ORIGIN = (95, 95)
CANVAS = (640, 490)
NODE_RADIUS = 20
LANE_WIDTH = 9
ARROW_LEN = 26
ARROW_HALF = 11
N_EDGES = 13
MIN_ROUTE_EDGES = 3
QUARANTINE_PX = 8.0
SAMPLE_FLOOR_PX = 4.0
JITTER_NORMAL = 14.0
JITTER_TIGHT = 45.0


def _sites() -> list[tuple[int, int]]:
    return [(c, r) for r in range(ROWS) for c in range(COLS)]


def _lattice_adjacency() -> list[tuple[int, int]]:
    sites = _sites()
    index = {s: i for i, s in enumerate(sites)}
    pairs = []
    for (c, r), i in index.items():
        for dc, dr in ((1, 0), (0, 1)):
            j = index.get((c + dc, r + dr))
            if j is not None:
                pairs.append((i, min(j, i) if False else j))
    return [(min(a, b), max(a, b)) for a, b in pairs]


def _reachable(n: int, edges: list[list[int]], start: int) -> dict[int, int]:
    """Directed BFS: node -> number of lanes travelled from `start`."""
    out: dict[int, list[int]] = {i: [] for i in range(n)}
    for a, b in edges:
        out[a].append(b)
    dist = {start: 0}
    frontier = [start]
    while frontier:
        nxt = []
        for u in frontier:
            for v in out[u]:
                if v not in dist:
                    dist[v] = dist[u] + 1
                    nxt.append(v)
        frontier = nxt
    return dist


def _undirected_connected(n: int, edges: list[list[int]]) -> bool:
    nb: dict[int, list[int]] = {i: [] for i in range(n)}
    for a, b in edges:
        nb[a].append(b)
        nb[b].append(a)
    seen = {0}
    stack = [0]
    while stack:
        u = stack.pop()
        for v in nb[u]:
            if v not in seen:
                seen.add(v)
                stack.append(v)
    return len(seen) == n


def _random_connected_subset(rng: random.Random, n: int, adjacency, k: int):
    """Random spanning tree over the lattice, padded with extra lattice lanes."""
    nb: dict[int, list[int]] = {i: [] for i in range(n)}
    for a, b in adjacency:
        nb[a].append(b)
        nb[b].append(a)
    tree: list[tuple[int, int]] = []
    seen = {rng.randrange(n)}
    while len(seen) < n:
        frontier = [(u, v) for u in seen for v in nb[u] if v not in seen]
        u, v = rng.choice(frontier)
        tree.append((min(u, v), max(u, v)))
        seen.add(v)
    rest = [e for e in adjacency if e not in tree]
    rng.shuffle(rest)
    return tree + rest[: max(0, k - len(tree))]


def sample_scene(seed: int) -> dict:
    rng = random.Random((seed * 2654435761) % (2**32))
    sites = _sites()
    n = len(sites)
    adjacency = sorted(set(_lattice_adjacency()))
    tight = (seed % 8) == 7
    jitter = JITTER_TIGHT if tight else JITTER_NORMAL
    want_color = TANKS[seed % 3]
    want_rank = (seed // 3) % 3

    for attempt in range(60_000):
        relaxed = attempt > 30_000
        positions = []
        for c, r in sites:
            positions.append(
                [
                    ORIGIN[0] + PITCH * c + rng.uniform(-jitter, jitter),
                    ORIGIN[1] + PITCH * r + rng.uniform(-jitter, jitter),
                ]
            )
        slots = rng.sample(range(n), 4)
        roles = ["gray"] * n
        roles[slots[0]] = "green"
        for name, slot in zip(TANKS, slots[1:]):
            roles[slot] = name
        undirected = _random_connected_subset(rng, n, adjacency, N_EDGES)
        if not _undirected_connected(n, [list(e) for e in undirected]):
            continue
        edges = [[a, b] if rng.random() < 0.5 else [b, a] for a, b in undirected]
        green = roles.index("green")
        dist = _reachable(n, edges, green)
        hit = [t for t in TANKS if roles.index(t) in dist]
        if len(hit) != 1:
            continue
        answer = hit[0]
        if dist[roles.index(answer)] < MIN_ROUTE_EDGES:
            continue
        if answer != want_color:
            continue
        gp = positions[green]
        order = sorted(TANKS, key=lambda t: math.dist(gp, positions[roles.index(t)]))
        if not relaxed and order.index(answer) != want_rank:
            continue
        scene = {
            "canvas": list(CANVAS),
            "background": list(BG),
            "origin": list(ORIGIN),
            "pitch": PITCH,
            "grid": [COLS, ROWS],
            "jitter_regime": "tight" if tight else "normal",
            "node_positions": positions,
            "node_roles": roles,
            "edges": edges,
            "node_radius": NODE_RADIUS,
            "lane_width": LANE_WIDTH,
            "arrow_len": ARROW_LEN,
            "arrow_half": ARROW_HALF,
            "draw_order": "lanes_then_arrowheads_then_junction_disks",
        }
        # Never emit geometry the image-only arm cannot read: below this pixel
        # floor a lane, disk or arrowhead stops being separable and the inverse
        # arm abstains, so such layouts are resampled rather than shipped as
        # quarantined rows. The quarantine flag sits above this floor and marks
        # scenes that are readable but near the resolution boundary.
        if margin(scene) < SAMPLE_FLOOR_PX:
            continue
        return scene
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), tuple(scene["background"]))
    d = ImageDraw.Draw(img)
    pos = [tuple(p) for p in scene["node_positions"]]
    for a, b in scene["edges"]:
        d.line([pos[a], pos[b]], fill=LANE, width=scene["lane_width"])
    for a, b in scene["edges"]:
        d.polygon(_arrow_polygon(pos[a], pos[b], scene), fill=ARROW)
    r = scene["node_radius"]
    for (x, y), role in zip(pos, scene["node_roles"]):
        d.ellipse([x - r, y - r, x + r, y + r], fill=ROLE_COLORS[role])
    return img


def _arrow_polygon(p, q, scene) -> list[tuple[float, float]]:
    ux, uy = q[0] - p[0], q[1] - p[1]
    L = math.hypot(ux, uy)
    ux, uy = ux / L, uy / L
    mx, my = (p[0] + q[0]) / 2.0, (p[1] + q[1]) / 2.0
    half, wing = scene["arrow_len"] / 2.0, scene["arrow_half"]
    tip = (mx + ux * half, my + uy * half)
    base = (mx - ux * half, my - uy * half)
    return [
        tip,
        (base[0] - uy * wing, base[1] + ux * wing),
        (base[0] + uy * wing, base[1] - ux * wing),
    ]


def analytic_gold(scene: dict) -> str:
    roles = scene["node_roles"]
    dist = _reachable(len(roles), scene["edges"], roles.index("green"))
    hit = [t for t in TANKS if roles.index(t) in dist]
    if len(hit) != 1:
        raise ValueError("scene does not have exactly one reachable tank")
    return hit[0]


def _seg_dist(pt, a, b) -> float:
    ax, ay, bx, by = a[0], a[1], b[0], b[1]
    vx, vy = bx - ax, by - ay
    L2 = vx * vx + vy * vy
    t = 0.0 if L2 == 0 else max(0.0, min(1.0, ((pt[0] - ax) * vx + (pt[1] - ay) * vy) / L2))
    return math.dist(pt, (ax + t * vx, ay + t * vy))


def _seg_seg_dist(p1, p2, q1, q2) -> float:
    return min(
        _seg_dist(p1, q1, q2),
        _seg_dist(p2, q1, q2),
        _seg_dist(q1, p1, p2),
        _seg_dist(q2, p1, p2),
    )


def margin(scene: dict) -> float:
    """Smallest pixel clearance the image-only reading of the scene relies on.

    Terms: (a) the arrowhead's own direction cue, the gap in pixels between the
    tip-side and base-side reach of the triangle about its ink centroid;
    (b) clearance between each arrowhead and the two junction disks its lane
    ends at; (c) disk-disk clearance; (d) clearance between a disk and a lane it
    is not an endpoint of; (e) clearance between two lanes sharing no junction;
    (f) phantom-lane slack: for every junction pair joined by no lane, the widest
    clearly empty gap found along the middle of the straight line between them.
    """
    pos = [tuple(p) for p in scene["node_positions"]]
    r = scene["node_radius"]
    half_w = scene["lane_width"] / 2.0
    edges = [tuple(e) for e in scene["edges"]]
    und = {frozenset(e) for e in edges}
    terms = [scene["arrow_len"] / 3.0]

    for a, b in edges:
        L = math.dist(pos[a], pos[b])
        terms.append(L / 2.0 - scene["arrow_len"] / 2.0 - r)
    for i in range(len(pos)):
        for j in range(i + 1, len(pos)):
            terms.append(math.dist(pos[i], pos[j]) - 2 * r)
    for i, p in enumerate(pos):
        for a, b in edges:
            if i in (a, b):
                continue
            terms.append(_seg_dist(p, pos[a], pos[b]) - (r + half_w))
    for x, e1 in enumerate(edges):
        for e2 in edges[x + 1 :]:
            if set(e1) & set(e2):
                continue
            terms.append(_seg_seg_dist(pos[e1[0]], pos[e1[1]], pos[e2[0]], pos[e2[1]]) - 2 * half_w)

    for i in range(len(pos)):
        for j in range(i + 1, len(pos)):
            if frozenset((i, j)) in und:
                continue
            best = -1e9
            for k in range(41):
                t = 0.30 + 0.40 * k / 40.0
                pt = (
                    pos[i][0] + t * (pos[j][0] - pos[i][0]),
                    pos[i][1] + t * (pos[j][1] - pos[i][1]),
                )
                # A sample refutes the phantom lane either by lying clear of every
                # drawn lane (white page) or by lying well inside a junction disk,
                # which paints over the lane color and is not travellable ink.
                off_lane = min(_seg_dist(pt, pos[a], pos[b]) - half_w for a, b in edges)
                in_disk = max(r - math.dist(pt, p) for p in pos)
                best = max(best, max(off_lane, in_disk))
            terms.append(best)
    return float(min(terms))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent alias: every sampled field is painted, and the two label inputs
    (`node_roles`, `edges`) are painted as disk colors and as oriented lanes, so
    no unrendered parameter can move the label.  Declared empty on purpose."""
    return []
