"""Deterministic renderer and analytic arm for Boundary-Limited Level Replay."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1200
HEIGHT = 380
CAPACITY = 4
N_OPS = 8
PALETTES = (
    (35, 112, 181),
    (137, 75, 170),
    (0, 137, 123),
    (202, 78, 62),
)


def _replay(start: int, ops: list[str]) -> int:
    level = int(start)
    for op in ops:
        if op == "up":
            level = min(CAPACITY, level + 1)
        elif op == "down":
            level = max(0, level - 1)
        else:
            raise ValueError(f"unknown operation {op!r}")
    return level


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with four UP and four DOWN commands.

    Consecutive seeds cycle independently through desired final and initial
    levels.  Rejection occurs over the finite 70 command orderings only.
    """

    rng = random.Random(int(seed))
    target = int(seed) % (CAPACITY + 1)
    start = (int(seed) // (CAPACITY + 1)) % (CAPACITY + 1)
    candidates: list[list[str]] = []
    for up_positions in itertools.combinations(range(N_OPS), N_OPS // 2):
        up_set = set(up_positions)
        ops = ["up" if index in up_set else "down" for index in range(N_OPS)]
        if _replay(start, ops) == target:
            candidates.append(ops)
    ops = list(candidates[rng.randrange(len(candidates))])
    return {
        "start": start,
        "ops": ops,
        "palette": rng.randrange(len(PALETTES)),
        "shift": [rng.choice((-8, 0, 8)), rng.choice((-6, 0, 6))],
        "decoy_variant": rng.randrange(4),
    }


def _font(size: int) -> ImageFont.ImageFont:
    return ImageFont.load_default(size=size)


def render(scene: dict) -> Image.Image:
    """Render only observable fields from the scene dictionary."""

    if is_quarantined(scene):
        raise ValueError("cannot render malformed or ambiguous scene")
    start = int(scene["start"])
    ops = list(scene["ops"])
    ink = PALETTES[int(scene["palette"])]
    dx, dy = (int(value) for value in scene["shift"])
    decoy_variant = int(scene["decoy_variant"])

    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    dark = (39, 44, 49)
    mid = (113, 121, 128)
    pale = (229, 232, 232)
    white = (255, 255, 253)

    draw.rounded_rectangle((24 + dx, 66 + dy, 222 + dx, 326 + dy), 12, fill=white, outline=dark, width=3)
    draw.text((123 + dx, 84 + dy), "START LEVEL", font=_font(20), fill=dark, anchor="mm")
    draw.text((123 + dx, 113 + dy), "filled steps", font=_font(16), fill=mid, anchor="mm")

    cell_x0, cell_x1 = 91 + dx, 158 + dx
    bottom = 296 + dy
    cell_h, gap = 34, 8
    for level in range(CAPACITY):
        y1 = bottom - level * (cell_h + gap)
        y0 = y1 - cell_h
        draw.rounded_rectangle((cell_x0, y0, cell_x1, y1), 4, fill=white, outline=dark, width=2)
        if level < start:
            draw.rectangle((cell_x0 + 6, y0 + 6, cell_x1 - 6, y1 - 6), fill=ink)
    draw.text((76 + dx, bottom - 15), "0", font=_font(16), fill=mid, anchor="mm")
    draw.text((76 + dx, bottom - 3 * (cell_h + gap) - 17), "4", font=_font(16), fill=mid, anchor="mm")

    draw.text((710 + dx, 36 + dy), "READ THE 8 CARDS LEFT TO RIGHT", font=_font(24), fill=dark, anchor="mm")
    card_y0, card_y1 = 105 + dy, 283 + dy
    card_w, stride = 96, 116
    first_x = 248 + dx
    for index, op in enumerate(ops):
        x0 = first_x + index * stride
        x1 = x0 + card_w
        draw.rounded_rectangle((x0, card_y0, x1, card_y1), 10, fill=white, outline=dark, width=3)
        draw.text(((x0 + x1) // 2, card_y0 + 23), str(index + 1), font=_font(18), fill=mid, anchor="mm")
        cx = (x0 + x1) // 2
        top, bottom_tri, half = card_y0 + 61, card_y0 + 127, 27
        if op == "up":
            points = ((cx, top), (cx - half, bottom_tri), (cx + half, bottom_tri))
        else:
            points = ((cx - half, top), (cx + half, top), (cx, bottom_tri))
        draw.polygon(points, fill=ink)
        draw.text((cx, card_y1 - 22), "UP" if op == "up" else "DOWN", font=_font(14), fill=dark, anchor="mm")
        if index < len(ops) - 1:
            arrow_x = x1 + 10
            arrow_y = (card_y0 + card_y1) // 2
            draw.line((arrow_x, arrow_y, arrow_x + 10, arrow_y), fill=mid, width=2)
            draw.polygon(((arrow_x + 10, arrow_y), (arrow_x + 5, arrow_y - 4), (arrow_x + 5, arrow_y + 4)), fill=mid)

    # Neutral decorations change visible nuisance structure without adding
    # saturated pixels that could be mistaken for state evidence.
    patterns = (
        ((330, 337), (600, 337), (870, 337)),
        ((390, 337), (660, 337), (930, 337)),
        ((450, 337), (720, 337), (990, 337)),
        ((510, 337), (780, 337), (1050, 337)),
    )
    for x, y in patterns[decoy_variant]:
        draw.ellipse((x - 5 + dx, y - 5 + dy, x + 5 + dx, y + 5 + dy), fill=pale, outline=mid)
    return image


def analytic_gold(scene: dict) -> str:
    return str(_replay(int(scene["start"]), list(scene["ops"])))


def margin(scene: dict) -> float:
    """Runner-up visual decision margin in pixels.

    The closest operation alternative crosses the triangle vertical-centroid
    boundary at about eleven pixels; the gauge-block aspect boundary is farther.
    """

    return 0.0 if is_quarantined(scene) else 10.0


def is_quarantined(scene: dict) -> bool:
    try:
        start = scene.get("start")
        ops = scene.get("ops")
        palette = scene.get("palette")
        shift = scene.get("shift")
        decoy = scene.get("decoy_variant")
        return not (
            isinstance(start, int)
            and 0 <= start <= CAPACITY
            and isinstance(ops, list)
            and len(ops) == N_OPS
            and set(ops) <= {"up", "down"}
            and ops.count("up") == N_OPS // 2
            and ops.count("down") == N_OPS // 2
            and isinstance(palette, int)
            and 0 <= palette < len(PALETTES)
            and isinstance(shift, list)
            and len(shift) == 2
            and all(isinstance(value, int) and -12 <= value <= 12 for value in shift)
            and isinstance(decoy, int)
            and 0 <= decoy < 4
        )
    except (AttributeError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
