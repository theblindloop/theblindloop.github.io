"""Deterministic renderer for conditional one-puck interventions."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 700
PANEL_RECTS = (
    (35, 100, 455, 375),
    (505, 100, 925, 375),
    (35, 400, 455, 675),
    (505, 400, 925, 675),
)

PUCK_RADIUS = 24.0
NOSE_OFFSET = 25.0
TARGET_RADIUS = 18.0
GATE_MARGIN = 12.0
QUARANTINE_MARGIN = 10.0

BACKGROUND = (238, 242, 246)
PANEL_FILL = (250, 251, 253)
FRAME = (112, 126, 140)
FRAME_ACCENT = (205, 213, 220)
PUCK_BLUE = (45, 94, 178)
NOSE_GOLD = (242, 178, 39)
MOVE_GREEN = (26, 117, 91)
GATE_VIOLET = (126, 70, 176)
TARGET_RED = (196, 55, 76)
LABEL = (57, 70, 86)
WHITE = (255, 255, 255)

PANEL_LABELS = ("A", "B", "C", "D")
_BITMAPS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _vector(angle_deg: float, length: float = 1.0) -> tuple[float, float]:
    angle = math.radians(angle_deg)
    return length * math.cos(angle), length * math.sin(angle)


def _add(a: tuple[float, float], b: tuple[float, float]) -> tuple[float, float]:
    return a[0] + b[0], a[1] + b[1]


def _distance(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _post_center(panel: dict) -> tuple[float, float]:
    dx, dy = _vector(float(panel["move_angle_deg"]), float(panel["move_length"]))
    return float(panel["center_x"]) + dx, float(panel["center_y"]) + dy


def _nose_direction(panel: dict) -> tuple[float, float]:
    return _vector(float(panel["nose_angle_deg"]))


def _final_nose(panel: dict) -> tuple[float, float]:
    post = _post_center(panel)
    direction = _nose_direction(panel)
    flip = post[0] > float(panel["gate_x"])
    sign = -1.0 if flip else 1.0
    return post[0] + sign * NOSE_OFFSET * direction[0], post[1] + sign * NOSE_OFFSET * direction[1]


def _nose_score(panel: dict) -> float:
    return _distance(_final_nose(panel), (float(panel["target_x"]), float(panel["target_y"]))) - float(
        panel["target_radius"]
    )


def _inside(rect: tuple[int, int, int, int], point: tuple[float, float], pad: float) -> bool:
    x0, y0, x1, y1 = rect
    return x0 + pad < point[0] < x1 - pad and y0 + pad < point[1] < y1 - pad


def _choose_far_target(
    rng: random.Random,
    rect: tuple[int, int, int, int],
    center: tuple[float, float],
    post: tuple[float, float],
    direction: tuple[float, float],
) -> tuple[float, float] | None:
    x0, y0, x1, y1 = rect
    choices = [
        (x0 + 315.0, y0 + 58.0),
        (x0 + 315.0, y0 + 217.0),
        (x0 + 70.0, y0 + 58.0),
        (x0 + 70.0, y0 + 217.0),
    ]
    rng.shuffle(choices)
    for target in choices:
        if _distance(target, center) > 56.0 and _distance(target, post) > 42.0:
            return target
    for _ in range(100):
        target = (rng.uniform(x0 + 55.0, x1 - 55.0), rng.uniform(y0 + 52.0, y1 - 52.0))
        if _distance(target, center) > 56.0 and _distance(target, post) > 42.0:
            return target
    return None


def _sample_panel(rng: random.Random, rect: tuple[int, int, int, int], mode: str) -> dict:
    x0, y0, x1, y1 = rect
    nose_angles = (0.0, 90.0, 180.0, 270.0)
    for _ in range(4000):
        center = (rng.uniform(x0 + 82.0, x0 + 130.0), rng.uniform(y0 + 94.0, y0 + 185.0))
        move_angle = rng.uniform(-170.0, 170.0)
        move_length = rng.uniform(102.0, 138.0)
        move = _vector(move_angle, move_length)
        post = _add(center, move)
        if not _inside(rect, post, 66.0):
            continue
        nose_angle = rng.choice(nose_angles)
        direction = _vector(nose_angle)

        if mode == "winner":
            gate_x = post[0] - rng.uniform(34.0, 52.0)
            target = _add(post, (-NOSE_OFFSET * direction[0], -NOSE_OFFSET * direction[1]))
        elif mode == "true_trap":
            gate_x = post[0] - rng.uniform(34.0, 52.0)
            target = _add(post, (NOSE_OFFSET * direction[0], NOSE_OFFSET * direction[1]))
        elif mode == "false_trap":
            gate_x = post[0] + rng.uniform(34.0, 52.0)
            target = _add(post, (-NOSE_OFFSET * direction[0], -NOSE_OFFSET * direction[1]))
        else:
            gate_x = rng.uniform(x0 + 75.0, x1 - 75.0)
            if abs(post[0] - gate_x) < 34.0:
                continue
            target = _choose_far_target(rng, rect, center, post, direction)
            if target is None:
                continue

        if not (x0 + 70.0 < gate_x < x1 - 70.0):
            continue
        if abs(post[0] - gate_x) < GATE_MARGIN + 12.0 or abs(center[0] - gate_x) < PUCK_RADIUS + 8.0:
            continue
        if not _inside(rect, target, 42.0):
            continue
        if _distance(target, center) < 50.0:
            continue
        panel = {
            "center_x": round(center[0], 3),
            "center_y": round(center[1], 3),
            "move_angle_deg": round(move_angle, 6),
            "move_length": round(move_length, 6),
            "nose_angle_deg": nose_angle,
            "gate_x": round(gate_x, 3),
            "target_x": round(target[0], 3),
            "target_y": round(target[1], 3),
            "puck_radius": PUCK_RADIUS,
            "nose_offset": NOSE_OFFSET,
            "target_radius": TARGET_RADIUS,
        }
        if abs(_nose_score(panel)) < QUARANTINE_MARGIN:
            continue
        return panel
    raise RuntimeError(f"could not sample a clear {mode} panel")


def sample_scene(seed: int) -> dict:
    """Sample four panels with one post-intervention nose inside a bullseye."""

    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = []
    for index, rect in enumerate(PANEL_RECTS):
        if index == answer_index:
            mode = "winner"
        else:
            offset = (index - answer_index) % 3
            mode = "true_trap" if offset == 1 else "false_trap" if offset == 2 else "far"
        panels.append(_sample_panel(rng, rect, mode))
    scene = {"canvas_width": WIDTH, "canvas_height": HEIGHT, "panels": panels}
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a quarantined scene")
    return scene


def _draw_bitmap(draw: ImageDraw.ImageDraw, glyph: str, xy: tuple[int, int], scale: int = 4) -> None:
    x0, y0 = xy
    for row, bits in enumerate(_BITMAPS[glyph]):
        for col, bit in enumerate(bits):
            if bit == "1":
                draw.rectangle(
                    (
                        x0 + col * scale,
                        y0 + row * scale,
                        x0 + (col + 1) * scale - 1,
                        y0 + (row + 1) * scale - 1,
                    ),
                    fill=LABEL,
                )


def _draw_arrow(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    angle_deg: float,
    length: float,
) -> None:
    ux, uy = _vector(angle_deg)
    px, py = -uy, ux
    tail = _add(center, ((PUCK_RADIUS + 6.0) * ux, (PUCK_RADIUS + 6.0) * uy))
    tip = _add(center, (length * ux, length * uy))
    draw.line((tail, tip), fill=MOVE_GREEN, width=8)
    base = (tip[0] - 15.0 * ux, tip[1] - 15.0 * uy)
    left = (base[0] + 10.0 * px, base[1] + 10.0 * py)
    right = (base[0] - 10.0 * px, base[1] - 10.0 * py)
    draw.polygon((tip, left, right), fill=MOVE_GREEN)


def _draw_puck(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    nose_angle_deg: float,
) -> None:
    x, y = center
    draw.ellipse(
        (round(x - PUCK_RADIUS), round(y - PUCK_RADIUS), round(x + PUCK_RADIUS), round(y + PUCK_RADIUS)),
        fill=PUCK_BLUE,
        outline=WHITE,
        width=2,
    )
    ux, uy = _vector(nose_angle_deg)
    px, py = -uy, ux
    base = (x + 7.0 * ux, y + 7.0 * uy)
    tip = (x + 24.0 * ux, y + 24.0 * uy)
    left = (base[0] + 9.0 * px, base[1] + 9.0 * py)
    right = (base[0] - 9.0 * px, base[1] - 9.0 * py)
    draw.polygon((tip, left, right), fill=NOSE_GOLD)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((35, 42, 925, 45), fill=FRAME_ACCENT)
    for index, panel in enumerate(scene["panels"]):
        x0, y0, x1, y1 = PANEL_RECTS[index]
        draw.rounded_rectangle(
            (x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3
        )
        draw.rectangle((x0 + 14, y0 + 14, x0 + 44, y0 + 44), fill=WHITE, outline=FRAME_ACCENT, width=1)
        _draw_bitmap(draw, PANEL_LABELS[index], (x0 + 19, y0 + 17))

        gate_x = round(float(panel["gate_x"]))
        draw.line((gate_x, y0 + 48, gate_x, y1 - 48), fill=GATE_VIOLET, width=6)
        target_x = float(panel["target_x"])
        target_y = float(panel["target_y"])
        target_r = float(panel["target_radius"])
        draw.ellipse(
            (round(target_x - target_r), round(target_y - target_r), round(target_x + target_r), round(target_y + target_r)),
            outline=TARGET_RED,
            width=5,
        )

        center = (float(panel["center_x"]), float(panel["center_y"]))
        _draw_arrow(draw, center, float(panel["move_angle_deg"]), float(panel["move_length"]))
        _draw_puck(draw, center, float(panel["nose_angle_deg"]))
    return image


def analytic_gold(scene: dict) -> str:
    # Keep the analytic entrypoint's only scene-level read at ``panels``.  The
    # nested panel fields are label inputs of that collection, and the field
    # audit follows this entrypoint separately from the renderer helpers.
    scores = []
    for panel in scene["panels"]:
        angle = math.radians(float(panel["move_angle_deg"]))
        post_x = float(panel["center_x"]) + float(panel["move_length"]) * math.cos(angle)
        post_y = float(panel["center_y"]) + float(panel["move_length"]) * math.sin(angle)
        nose_angle = math.radians(float(panel["nose_angle_deg"]))
        flip = post_x > float(panel["gate_x"])
        sign = -1.0 if flip else 1.0
        nose_x = post_x + sign * float(panel["nose_offset"]) * math.cos(nose_angle)
        nose_y = post_y + sign * float(panel["nose_offset"]) * math.sin(nose_angle)
        target_distance = math.hypot(
            nose_x - float(panel["target_x"]), nose_y - float(panel["target_y"])
        )
        scores.append(target_distance - float(panel["target_radius"]))
    inside = [score < 0.0 for score in scores]
    index = inside.index(True) if sum(inside) == 1 else min(range(4), key=lambda i: scores[i])
    return f"panel-{chr(65 + index)}"


def margin(scene: dict) -> float:
    scores = [_nose_score(panel) for panel in scene["panels"]]
    gate_gaps = [abs(_post_center(panel)[0] - float(panel["gate_x"])) for panel in scene["panels"]]
    current_target_gaps = [
        _distance(
            (float(panel["center_x"]), float(panel["center_y"])),
            (float(panel["target_x"]), float(panel["target_y"])),
        )
        - float(panel["target_radius"])
        for panel in scene["panels"]
    ]
    return float(min([abs(value) for value in scores] + gate_gaps + current_target_gaps))


def is_quarantined(scene: dict) -> bool:
    scores = [_nose_score(panel) for panel in scene["panels"]]
    gate_gaps = [abs(_post_center(panel)[0] - float(panel["gate_x"])) for panel in scene["panels"]]
    current_target_gaps = [
        _distance(
            (float(panel["center_x"]), float(panel["center_y"])),
            (float(panel["target_x"]), float(panel["target_y"])),
        )
        - float(panel["target_radius"])
        for panel in scene["panels"]
    ]
    return (
        sum(value < 0.0 for value in scores) != 1
        or min(abs(value) for value in scores) < QUARANTINE_MARGIN
        or min(gate_gaps) < GATE_MARGIN
        or min(current_target_gaps) < 20.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no interchangeable latent slots or unused label-bearing fields."""

    return []
