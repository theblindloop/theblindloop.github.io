"""Peak-distance-from-start register accumulator (state tracking).

The image shows five panels side by side, read left to right. The leftmost
(panels slot 0) is the START panel: it shows a short horizontal rail divided
into five slots, slot 0 at the left to slot 4 at the right, with a single
magenta marker disk sitting on one slot. An orange ring surrounds that marker
and marks the register's START slot index (0, 1, 2, 3, or 4). To the right are
four OPERATOR panels, read left to right. Each operator panel holds one colored
triangle: a green right-pointing triangle means ADVANCE the register's value by
one (+1) and a red left-pointing triangle means RETREAT it by one (-1).

Replay the four operators in order starting from the ringed START slot value;
after each operator the register's value is start + (sum of applied steps). The
register's position is never drawn after the start panel. The task is to report
the MAXIMUM horizontal distance the register's value ever gets from its start
value across the whole replay, i.e. the largest value of |value - start| seen at
any point (including the start itself, where the distance is 0). The answer is
one of the integers 0, 1, 2, 3, or 4.

Because the reported value is an extremum over the whole hidden replayed
trajectory, it cannot be read from any single operator panel, from the start
panel alone, from the final value, or from a plain sum of the steps.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1180
HEIGHT = 300
PANEL_TOP = 60
PANEL_H = 140
PANEL_W = 200
START_X = 30
OP_STEP = 215
N_OP = 4
N_SLOTS = 5

BG = (255, 255, 255)
BORDER = (200, 200, 200)
MARKER = (200, 60, 200)
RING = (255, 150, 0)
GREEN = (40, 180, 90)
RED = (215, 60, 60)

MARKER_R = 16
RING_R = 27
RAIL_PAD = 36
SLOT_W = (PANEL_W - 2 * RAIL_PAD) / N_SLOTS
PANEL_MID = PANEL_TOP + PANEL_H // 2


def _panel_left(panel_index: int) -> int:
    return START_X + panel_index * OP_STEP


def _slot_center(panel_left: int, slot: int) -> tuple[int, int]:
    rail_left = panel_left + RAIL_PAD
    x = int(rail_left + SLOT_W * (slot + 0.5))
    return (x, PANEL_MID)


def _draw_triangle(draw: ImageDraw.ImageDraw, panel_left: int, direction: int, color) -> None:
    cx = panel_left + PANEL_W // 2
    cy = PANEL_MID
    if direction > 0:
        pts = [(cx - 22, cy - 26), (cx - 22, cy + 26), (cx + 22, cy)]
    else:
        pts = [(cx + 22, cy - 26), (cx + 22, cy + 26), (cx - 22, cy)]
    draw.polygon(pts, fill=color)


def _draw_rail(draw: ImageDraw.ImageDraw, panel_left: int) -> None:
    x0 = panel_left + RAIL_PAD - 6
    x1 = panel_left + PANEL_W - RAIL_PAD + 6
    y = PANEL_MID
    draw.line([x0, y, x1, y], fill=(120, 120, 120), width=2)
    for k in range(N_SLOTS):
        cx = _slot_center(panel_left, k)[0]
        draw.line([cx, y - 6, cx, y + 6], fill=(120, 120, 120), width=2)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start_slot = rng.randrange(N_SLOTS)
    moves = [rng.choice([1, -1]) for _ in range(N_OP)]
    return {"start_slot": start_slot, "moves": moves}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    start_slot = scene["start_slot"]
    moves = scene["moves"]

    for i in range(N_OP + 1):
        left = _panel_left(i)
        draw.rectangle([left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H], outline=BORDER, width=3)

    _draw_rail(draw, _panel_left(0))
    marker = _slot_center(_panel_left(0), start_slot)
    draw.ellipse(
        [marker[0] - MARKER_R, marker[1] - MARKER_R, marker[0] + MARKER_R, marker[1] + MARKER_R],
        fill=MARKER,
    )
    draw.ellipse(
        [marker[0] - RING_R, marker[1] - RING_R, marker[0] + RING_R, marker[1] + RING_R],
        outline=RING,
        width=5,
    )

    for i, move in enumerate(moves):
        _draw_triangle(draw, _panel_left(i + 1), move, GREEN if move > 0 else RED)
    return img


def analytic_gold(scene: dict) -> str:
    start = scene["start_slot"]
    moves = scene["moves"]
    val = start
    max_dist = 0
    for move in moves:
        val += move
        dist = abs(val - start)
        if dist > max_dist:
            max_dist = dist
    return str(max_dist)


def margin(scene: dict) -> float:
    """Discrete-boundary margin: the reported peak-distance is an exact integer
    class value whose gap to every other integer class is (at least) one slot,
    so every non-degenerate scene carries a fixed positive boundary separation."""
    return 2.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    return []
