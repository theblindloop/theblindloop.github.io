"""Independent pixel-only inverse arm for peak-distance-from-start.

Receives only the final PNG. From the raster alone it:
  1. locates the four operator triangles by green/red colour segmentation and
     recovers the ordered left-to-right sign sequence (green right-triangle =
     +1, red left-triangle = -1);
  2. locates the magenta start marker within the leftmost panel and reads the
     ringed START slot index from its position on the rail;
  3. replays the +1/-1 steps from the start and reports the maximum absolute
     distance the register's value ever reaches from the start value.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GREEN = (40, 180, 90)
RED = (215, 60, 60)
MAGENTA = (200, 60, 200)
N_OP = 4
N_SLOTS = 5
START_X = 30
RAIL_PAD = 36
PANEL_W = 200
SLOT_W = (PANEL_W - 2 * RAIL_PAD) / N_SLOTS


class UnreadableImageError(ValueError):
    """The raster does not expose the expected operator/start structure."""


def _mask(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return (arr[:, :, 0] == color[0]) & (arr[:, :, 1] == color[1]) & (arr[:, :, 2] == color[2])


def _components(mask: np.ndarray, min_size: int = 12) -> list[np.ndarray]:
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < mask.shape[0]
                        and 0 <= nx < mask.shape[1]
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= min_size:
            comps.append(points)
    return comps


def _centroid(points: list[tuple[int, int]]) -> tuple[float, float]:
    ys = [p[0] for p in points]
    xs = [p[1] for p in points]
    return float(np.mean(xs)), float(np.mean(ys))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))

    green_mask = _mask(arr, GREEN)
    red_mask = _mask(arr, RED)
    if not green_mask.any() and not red_mask.any():
        raise UnreadableImageError("no operator glyphs found")

    moves = []
    for mask, direction in ((green_mask, 1), (red_mask, -1)):
        for comp in _components(mask):
            cx, cy = _centroid(comp)
            moves.append((cx, direction))
    if len(moves) != N_OP:
        raise UnreadableImageError(f"expected {N_OP} operators; found {len(moves)}")
    moves.sort(key=lambda item: item[0])
    signs = [item[1] for item in moves]

    magenta_mask = _mask(arr, MAGENTA)
    comps = _components(magenta_mask)
    if not comps:
        raise UnreadableImageError("start marker not found")
    largest = max(comps, key=len)
    cx, cy = _centroid(largest)
    rail_left = START_X + RAIL_PAD
    start = int(round((cx - rail_left) / SLOT_W))
    if not (0 <= start < N_SLOTS):
        raise UnreadableImageError(f"start slot {start} outside rail")

    val = start
    max_dist = 0
    for step in signs:
        val += step
        dist = abs(val - start)
        if dist > max_dist:
            max_dist = dist
    return str(max_dist)
