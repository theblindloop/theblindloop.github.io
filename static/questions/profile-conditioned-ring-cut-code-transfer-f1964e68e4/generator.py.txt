"""Scene sampling, rendering, analytic gold, margin and aliases.

World: a ring of twelve gray spokes of unequal length and a straight row of the
same twelve spokes, cut open at an unknown spoke and drawn at a different
overall size. One ring spoke carries a black dot. The answer is the color of the
square under the matching bar in the row; the matching is fixed only by the
cyclic length profile, never by position, angle or color.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WORLD_VERSION = "ring_cut_code_transfer-renderer-0.1.0"

WIDTH = 640
HEIGHT = 700
SEPARATOR_Y = 300
SLOTS = 12
STEP_DEG = 360.0 / SLOTS

WHITE = (255, 255, 255)
STROKE = (70, 70, 70)
HUB = (120, 120, 120)
SEPARATOR = (160, 160, 160)
MARKER = (0, 0, 0)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (214, 39, 40),
    "blue": (31, 84, 190),
    "green": (23, 140, 60),
    "orange": (240, 150, 20),
}
COLOR_NAMES = ("red", "blue", "green", "orange")

BAR_HALF_WIDTH = 3
SQUARE_HALF = 11
BASELINE_Y = 212
SQUARE_TOP = 232
FIRST_X = 60
SLOT_DX = 44
RING_CX = 320
RING_CY = 505
RING_INNER = 70
HUB_RADIUS = 9
MARKER_GAP = 22
MARKER_RADIUS = 9

QUARANTINE_MARGIN = 10.0
MIN_SAMPLED_MARGIN = 5.0


def _profile_cost(strip: list[float], ring: list[float], shift: int) -> float:
    """Mean absolute difference of scale-normalised profiles under a cyclic shift.

    ``ring[i]`` is compared with ``strip[(i + shift) % SLOTS]``. Both profiles are
    divided by their own mean first, so an overall ring scale factor cancels.
    """
    strip_mean = sum(strip) / len(strip)
    ring_mean = sum(ring) / len(ring)
    total = 0.0
    for index, value in enumerate(ring):
        total += abs(value / ring_mean - strip[(index + shift) % SLOTS] / strip_mean)
    return total / SLOTS


def _alignment_costs(scene: dict) -> list[float]:
    strip = [float(value) for value in scene["lengths"]]
    offset = int(scene["cut_offset"])
    scale = float(scene["ring_scale"])
    ring = [scale * strip[(index + offset) % SLOTS] for index in range(SLOTS)]
    return [_profile_cost(strip, ring, shift) for shift in range(SLOTS)]


def _sample_lengths(rng: random.Random, mode: str) -> list[float]:
    if mode == "generic":
        return [round(rng.uniform(40.0, 110.0), 3) for _ in range(SLOTS)]
    period = rng.choice((4, 6))
    jitter = rng.uniform(1.5, 7.0)
    base = [rng.uniform(40.0, 110.0) for _ in range(period)]
    return [
        round(min(112.0, max(38.0, base[index % period] + rng.uniform(-jitter, jitter))), 3)
        for index in range(SLOTS)
    ]


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for ``seed``."""
    rng = random.Random(seed * 7919 + 13)
    target_color = COLOR_NAMES[seed % len(COLOR_NAMES)]
    for attempt in range(400):
        mode = "near_symmetric" if (seed + attempt) % 5 == 0 else "generic"
        lengths = _sample_lengths(rng, mode)
        colors = [name for name in COLOR_NAMES for _ in range(SLOTS // len(COLOR_NAMES))]
        rng.shuffle(colors)
        offset = rng.randrange(SLOTS)
        scale = round(rng.uniform(0.55, 0.80), 4)
        start_deg = float(rng.randrange(360))
        strip_index = rng.choice([i for i, name in enumerate(colors) if name == target_color])
        marker_index = (strip_index - offset) % SLOTS
        scene = {
            "lengths": lengths,
            "colors": colors,
            "cut_offset": offset,
            "ring_scale": scale,
            "ring_start_deg": start_deg,
            "marker_ring_index": marker_index,
            "profile_mode": mode,
        }
        costs = _alignment_costs(scene)
        runner_up = min(cost for shift, cost in enumerate(costs) if shift != offset)
        if 100.0 * runner_up >= MIN_SAMPLED_MARGIN:
            return scene
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def _ring_length(scene: dict, ring_index: int) -> float:
    lengths = scene["lengths"]
    offset = int(scene["cut_offset"])
    return float(scene["ring_scale"]) * float(lengths[(ring_index + offset) % SLOTS])


def _ring_angle(scene: dict, ring_index: int) -> float:
    # Whole-degree angles reduced modulo 360 keep re-indexed scenes bit-identical.
    return math.radians((float(scene["ring_start_deg"]) + STEP_DEG * ring_index) % 360.0)


def render(scene: dict) -> Image.Image:
    """Rasterise the scene; identical scenes give byte-identical images."""
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, SEPARATOR_Y, WIDTH - 1, SEPARATOR_Y + 2], fill=SEPARATOR)

    for index, length in enumerate(scene["lengths"]):
        x = FIRST_X + SLOT_DX * index
        top = int(round(BASELINE_Y - float(length)))
        draw.rectangle(
            [x - BAR_HALF_WIDTH, top, x + BAR_HALF_WIDTH, BASELINE_Y], fill=STROKE
        )
        color = PALETTE[scene["colors"][index]]
        draw.rectangle(
            [
                x - SQUARE_HALF,
                SQUARE_TOP,
                x + SQUARE_HALF,
                SQUARE_TOP + 2 * SQUARE_HALF,
            ],
            fill=color,
        )

    for ring_index in range(SLOTS):
        angle = _ring_angle(scene, ring_index)
        length = _ring_length(scene, ring_index)
        cos_a, sin_a = math.cos(angle), math.sin(angle)
        x0 = int(round(RING_CX + RING_INNER * cos_a))
        y0 = int(round(RING_CY + RING_INNER * sin_a))
        x1 = int(round(RING_CX + (RING_INNER + length) * cos_a))
        y1 = int(round(RING_CY + (RING_INNER + length) * sin_a))
        draw.line([x0, y0, x1, y1], fill=STROKE, width=2 * BAR_HALF_WIDTH + 1)

    draw.ellipse(
        [
            RING_CX - HUB_RADIUS,
            RING_CY - HUB_RADIUS,
            RING_CX + HUB_RADIUS,
            RING_CY + HUB_RADIUS,
        ],
        fill=HUB,
    )

    marker_index = int(scene["marker_ring_index"])
    angle = _ring_angle(scene, marker_index)
    radius = RING_INNER + _ring_length(scene, marker_index) + MARKER_GAP
    mx = int(round(RING_CX + radius * math.cos(angle)))
    my = int(round(RING_CY + radius * math.sin(angle)))
    draw.ellipse(
        [mx - MARKER_RADIUS, my - MARKER_RADIUS, mx + MARKER_RADIUS, my + MARKER_RADIUS],
        fill=MARKER,
    )
    return image


def analytic_gold(scene: dict) -> str:
    """Color of the square under the row bar that matches the marked spoke."""
    strip_index = (int(scene["marker_ring_index"]) + int(scene["cut_offset"])) % SLOTS
    return str(scene["colors"][strip_index])


def margin(scene: dict) -> float:
    """Runner-up alignment cost, in percent of the mean spoke length.

    The true cut offset costs exactly zero, so this is the gap between the
    correct cyclic alignment and the best competing one.
    """
    costs = _alignment_costs(scene)
    offset = int(scene["cut_offset"])
    runner_up = min(cost for shift, cost in enumerate(costs) if shift != offset)
    return round(100.0 * runner_up, 6)


def is_quarantined(scene: dict) -> bool:
    """Near-periodic profiles admit a second alignment and are excluded."""
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, label-preserving re-indexings of the ring.

    Where the ring's index origin sits is a bookkeeping choice: rotating the
    origin by whole spoke steps rewrites ``ring_start_deg``, ``cut_offset`` and
    ``marker_ring_index`` together and reproduces the same raster and gold.
    Start angles are whole degrees and the step is 30 degrees, so the rewritten
    angles are exactly equal floats, not merely close ones.
    """
    aliases = []
    for steps in (1, 5):
        alias = dict(scene)
        alias["ring_start_deg"] = float(scene["ring_start_deg"]) + STEP_DEG * steps
        alias["cut_offset"] = (int(scene["cut_offset"]) + steps) % SLOTS
        alias["marker_ring_index"] = (int(scene["marker_ring_index"]) - steps) % SLOTS
        aliases.append((f"ring_index_origin_shift_{steps}", alias))
    return aliases
