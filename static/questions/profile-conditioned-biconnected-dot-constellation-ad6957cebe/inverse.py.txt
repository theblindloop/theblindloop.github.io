"""Independent pixel-only inverse for the biconnected dot constellation.

This module intentionally reconstructs centers from raster color components and
reimplements the normalized proximity graph without importing renderer code or
reading the latent scene.
"""

from __future__ import annotations

import math
from statistics import median

import numpy as np
from PIL import Image


LABELS = ("red", "blue", "green")
EXPECTED_DOTS = 11
EDGE_FACTOR = 1.35
MIN_COMPONENT_AREA = 80


def _label_mask(arr: np.ndarray, label: str) -> np.ndarray:
    r = arr[:, :, 0].astype(np.int16)
    g = arr[:, :, 1].astype(np.int16)
    b = arr[:, :, 2].astype(np.int16)
    if label == "red":
        return (r > g + 55) & (r > b + 35)
    if label == "blue":
        return (b > r + 55) & (b > g + 25)
    return (g > r + 55) & (g > b + 35)


def _component_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    ys, xs = np.nonzero(mask)
    active = set(zip(ys.tolist(), xs.tolist()))
    centers: list[tuple[float, float]] = []
    while active:
        start = active.pop()
        stack = [start]
        count = 1
        total_x = start[1]
        total_y = start[0]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in active:
                    active.remove(neighbor)
                    stack.append(neighbor)
                    count += 1
                    total_x += neighbor[1]
                    total_y += neighbor[0]
        if count >= MIN_COMPONENT_AREA:
            centers.append((total_x / count, total_y / count))
    return centers


def _graph(points: list[tuple[float, float]]) -> list[list[bool]]:
    nearest = []
    distances: dict[tuple[int, int], float] = {}
    for i, left in enumerate(points):
        closest = float("inf")
        for j, right in enumerate(points):
            if i == j:
                continue
            distance = math.hypot(left[0] - right[0], left[1] - right[1])
            closest = min(closest, distance)
            distances[(min(i, j), max(i, j))] = distance
        nearest.append(closest)
    threshold = EDGE_FACTOR * median(nearest)
    graph = [[False] * len(points) for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            linked = distances[(i, j)] <= threshold
            graph[i][j] = graph[j][i] = linked
    return graph


def _connected_after(graph: list[list[bool]], removed: int | None) -> bool:
    remaining = [index for index in range(len(graph)) if index != removed]
    if not remaining:
        return True
    reached = {remaining[0]}
    pending = [remaining[0]]
    while pending:
        node = pending.pop()
        for neighbor, linked in enumerate(graph[node]):
            if linked and neighbor != removed and neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return len(reached) == len(remaining)


def _is_resilient(points: list[tuple[float, float]]) -> bool:
    graph = _graph(points)
    return _connected_after(graph, None) and all(
        _connected_after(graph, removed=index) for index in range(len(points))
    )


def decision_from_image(image: Image.Image) -> str:
    """Return the only hue whose recovered graph survives every one-dot deletion."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers = {}
    for label in LABELS:
        found = _component_centers(_label_mask(arr, label))
        if len(found) != EXPECTED_DOTS:
            raise ValueError(f"expected {EXPECTED_DOTS} {label} dots, found {len(found)}")
        centers[label] = found
    resilient = [label for label in LABELS if _is_resilient(centers[label])]
    if len(resilient) != 1:
        raise ValueError(f"expected one resilient color, found {resilient}")
    return resilient[0]
