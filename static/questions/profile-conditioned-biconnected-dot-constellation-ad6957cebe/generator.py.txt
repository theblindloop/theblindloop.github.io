"""Deterministic renderer and analytic contract for the dot-graph world.

The image contains only colored dots.  The declared visual rule turns each
color's dot centers into a normalized proximity graph; the selected graph is
the only one with no articulation dot.
"""

from __future__ import annotations

import math
import random
from statistics import median

from PIL import Image, ImageDraw


RENDERER_VERSION = "biconnected_dot_constellation-0.1.0"
COLORS = ("red", "blue", "green")
WIDTH = 900
HEIGHT = 780
DOT_COUNT = 11
RING_RADIUS = 112.0
EDGE_FACTOR = 1.35
EDGE_MARGIN_FLOOR = 5.0
DECISION_MARGIN_FLOOR = 0.5
BACKGROUND = (248, 247, 242)

PALETTE_STYLES = (
    {
        "red": (205, 55, 67),
        "blue": (48, 103, 176),
        "green": (47, 143, 82),
    },
    {
        "red": (191, 57, 86),
        "blue": (55, 91, 171),
        "green": (44, 139, 101),
    },
    {
        "red": (214, 72, 49),
        "blue": (48, 119, 184),
        "green": (73, 151, 75),
    },
)


def _pattern(gap_multiplier: float, phase: float) -> list[list[float]]:
    """Return eleven points on a circle, with an optional single open gap."""

    base = 2.0 * math.pi / DOT_COUNT
    long_gap = base * gap_multiplier
    short_gap = (2.0 * math.pi - long_gap) / (DOT_COUNT - 1)
    gaps = [long_gap] + [short_gap] * (DOT_COUNT - 1)
    angles = [phase]
    for gap_value in gaps[:-1]:
        angles.append(angles[-1] + gap_value)
    return [
        [round(RING_RADIUS * math.cos(angle), 5), round(RING_RADIUS * math.sin(angle), 5)]
        for angle in angles
    ]


def _role_for_color(scene: dict, color: str) -> str:
    """Assign the closed pattern from the visible target label at render time."""

    target = str(scene["target_color"])
    if color == target:
        return "closed"
    decoys = [name for name in COLORS if name != target]
    return "open_a" if color == decoys[0] else "open_b"


def _world_points(scene: dict, color: str) -> list[tuple[float, float]]:
    """Apply the visible lane pose to one local point constellation."""

    cx, cy, rotation, scale = (float(value) for value in scene["pose"][color])
    c = math.cos(rotation)
    s = math.sin(rotation)
    points = []
    for local_x, local_y in scene["patterns"][_role_for_color(scene, color)]:
        x = float(local_x) * float(scale)
        y = float(local_y) * float(scale)
        points.append((cx + c * x - s * y, cy + s * x + c * y))
    return points


def _proximity_graph(points: list[tuple[float, float]]) -> tuple[list[list[bool]], float, float]:
    nearest = []
    distances: dict[tuple[int, int], float] = {}
    for i, left in enumerate(points):
        best = float("inf")
        for j, right in enumerate(points):
            if i == j:
                continue
            distance = math.hypot(left[0] - right[0], left[1] - right[1])
            best = min(best, distance)
            distances[(min(i, j), max(i, j))] = distance
        nearest.append(best)
    threshold = EDGE_FACTOR * median(nearest)
    graph = [[False] * len(points) for _ in points]
    edge_margin = float("inf")
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            distance = distances[(i, j)]
            graph[i][j] = graph[j][i] = distance <= threshold
            edge_margin = min(edge_margin, abs(distance - threshold))
    return graph, threshold, edge_margin


def _connected(graph: list[list[bool]], removed: int | None = None) -> bool:
    remaining = [i for i in range(len(graph)) if i != removed]
    if not remaining:
        return True
    reached = {remaining[0]}
    pending = [remaining[0]]
    while pending:
        node = pending.pop()
        for neighbor, linked in enumerate(graph[node]):
            if linked and neighbor != removed and neighbor not in reached:
                reached.add(neighbor)
                pending.append(neighbor)
    return len(reached) == len(remaining)


def _robustness(points: list[tuple[float, float]]) -> tuple[float, float]:
    graph, _, edge_margin = _proximity_graph(points)
    if not _connected(graph):
        return 0.0, edge_margin
    survives_every_deletion = all(_connected(graph, removed=index) for index in range(len(points)))
    return (1.0 if survives_every_deletion else 0.0), edge_margin


def sample_scene(seed: int) -> dict:
    """Sample one scene; the seed deterministically counterbalances the label."""

    rng = random.Random(int(seed))
    target_color = COLORS[int(seed) % len(COLORS)]
    lane_centers = [130.0, 390.0, 650.0]
    lane_order = list(COLORS)
    rng.shuffle(lane_order)
    palette = {name: list(rgb) for name, rgb in rng.choice(PALETTE_STYLES).items()}
    dot_radius = round(rng.uniform(9.0, 10.5), 4)
    common_scale = round(rng.uniform(0.96, 1.04), 4)
    patterns = {
        "closed": _pattern(1.0, rng.uniform(-math.pi, math.pi)),
        "open_a": _pattern(1.82, rng.uniform(-math.pi, math.pi)),
        "open_b": _pattern(1.94, rng.uniform(-math.pi, math.pi)),
    }
    poses: dict[str, list[float]] = {}
    for lane_index, color in enumerate(lane_order):
        poses[color] = [
            round(450.0 + rng.uniform(-18.0, 18.0), 4),
            round(lane_centers[lane_index] + rng.uniform(-7.0, 7.0), 4),
            round(rng.uniform(-math.pi, math.pi), 5),
            common_scale,
        ]
    scene = {
        "target_color": target_color,
        "lane_order": lane_order,
        "palette": palette,
        "patterns": patterns,
        "pose": poses,
        "dot_radius": dot_radius,
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
    }
    # The construction is intentionally checked at the analytic boundary so a
    # future style edit cannot silently publish a tied or raster-fragile scene.
    if is_quarantined(scene):
        return sample_scene(int(seed) + 104729)
    return scene


def render(scene: dict) -> Image.Image:
    """Render colored dot centers with deterministic supersampling."""

    supersample = 4
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new(
        "RGB",
        (width * supersample, height * supersample),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"]) * supersample
    for color in scene["lane_order"]:
        rgb = tuple(int(value) for value in scene["palette"][color])
        for x, y in _world_points(scene, color):
            draw.ellipse(
                [
                    x * supersample - radius,
                    y * supersample - radius,
                    x * supersample + radius,
                    y * supersample + radius,
                ],
                fill=rgb,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """The label is the visible color assigned the resilient constellation."""

    return str(scene["target_color"])


def margin(scene: dict) -> float:
    """Top-vs-runner-up binary resilience score gap."""

    scores = [_robustness(_world_points(scene, color))[0] for color in COLORS]
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    """Reject finite-resolution graph ties or thresholds too near an edge."""

    scores = []
    for color in COLORS:
        points = _world_points(scene, color)
        if len(points) != DOT_COUNT:
            return True
        score, edge_margin = _robustness(points)
        scores.append(score)
        if edge_margin < EDGE_MARGIN_FLOOR:
            return True
    ordered = sorted(scores, reverse=True)
    return ordered[0] != 1.0 or ordered[0] - ordered[1] < DECISION_MARGIN_FLOOR


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared for this explicit parameterization."""

    return []
