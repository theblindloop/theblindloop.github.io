"""Deterministic latent sampler and raster renderer for dial-program replay."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 500
LABELS = ("A", "B", "C", "D")  # top, right, bottom, left
OPERATIONS = ("advance", "reflect")

_PALE_CHIP = (255, 250, 225)
_CARD_FILL = (235, 244, 252)
_MARKERS = ((220, 45, 55), (194, 48, 72))
_ADVANCE = ((238, 130, 34), (224, 102, 30))
_REFLECT = ((126, 72, 181), (146, 62, 172))
_GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


def _step(state: int, operation: str) -> int:
    if operation == "advance":
        return (state + 1) % 4
    if operation == "reflect":
        return (0, 3, 2, 1)[state]
    raise ValueError(f"unknown operation {operation!r}")


def _replay(initial_state: int, operations: list[str]) -> int:
    state = int(initial_state)
    for operation in operations:
        state = _step(state, operation)
    return state


def sample_scene(seed: int) -> dict:
    """Counterbalance answer identity while holding visual nuisances within seed quartets."""
    seed = int(seed)
    target = seed % 4
    group = seed // 4
    rng = random.Random(0xD1A1 + group * 7919)
    length = 5 + group % 3
    operations = ["advance", "reflect"]
    operations.extend(rng.choice(OPERATIONS) for _ in range(length - 2))

    # Each operation is bijective, so exactly one initial state reaches the
    # balanced target. This blocks sequence length, event count, ink, palette,
    # and layout from predicting the answer within each four-scene block.
    initial_state = next(s for s in range(4) if _replay(s, operations) == target)
    return {
        "initial_state": initial_state,
        "operations": operations,
        "palette_variant": group % 2,
        "layout_variant": (group // 2) % 2,
    }


def analytic_gold(scene: dict) -> str:
    return LABELS[_replay(scene["initial_state"], scene["operations"])]


def margin(scene: dict) -> float:
    """One-sided visual runner-up margin: minimum empty gap between event cards."""
    gap = 14.0 if int(scene["layout_variant"]) == 0 else 18.0
    palette = int(scene["palette_variant"])
    color_clearance = math.dist(_ADVANCE[palette], _REFLECT[palette]) / 8.0
    return float(min(gap, color_clearance))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)  # type: ignore[return-value]


def _bitmap(
    draw: ImageDraw.ImageDraw,
    label: str,
    center: tuple[float, float],
    scale: int,
    supersample: int,
) -> None:
    glyph = _GLYPHS[label]
    logical_w, logical_h = 5 * scale, 7 * scale
    x0 = center[0] - logical_w / 2
    y0 = center[1] - logical_h / 2
    for row, bits in enumerate(glyph):
        for col, bit in enumerate(bits):
            if bit == "1":
                box = (
                    x0 + col * scale,
                    y0 + row * scale,
                    x0 + (col + 1) * scale,
                    y0 + (row + 1) * scale,
                )
                draw.rectangle(_scaled_box(box, supersample), fill=(20, 24, 30))


def render(scene: dict) -> Image.Image:
    initial_state = int(scene["initial_state"])
    operations = list(scene["operations"])
    palette_variant = int(scene["palette_variant"])
    layout_variant = int(scene["layout_variant"])
    if initial_state not in range(4) or not 5 <= len(operations) <= 7:
        raise ValueError("scene is outside the public dial contract")
    if palette_variant not in (0, 1) or layout_variant not in (0, 1):
        raise ValueError("unknown visible variant")
    if any(op not in OPERATIONS for op in operations):
        raise ValueError("unknown operation")

    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), (248, 248, 246))
    draw = ImageDraw.Draw(image)
    dial_center = (176 + 8 * layout_variant, 246 - 8 * layout_variant)
    radius = 126
    stations = (
        (dial_center[0], dial_center[1] - radius),
        (dial_center[0] + radius, dial_center[1]),
        (dial_center[0], dial_center[1] + radius),
        (dial_center[0] - radius, dial_center[1]),
    )

    # All geometry is explicit in pixels: a static dial plus four labeled chips.
    draw.ellipse(
        _scaled_box((dial_center[0] - 92, dial_center[1] - 92,
                     dial_center[0] + 92, dial_center[1] + 92), ss),
        fill=(241, 242, 239), outline=(90, 97, 106), width=3 * ss,
    )
    for point in stations:
        draw.line(
            [(round(dial_center[0] * ss), round(dial_center[1] * ss)),
             (round(point[0] * ss), round(point[1] * ss))],
            fill=(157, 163, 170), width=4 * ss,
        )
    draw.ellipse(
        _scaled_box((dial_center[0] - 13, dial_center[1] - 13,
                     dial_center[0] + 13, dial_center[1] + 13), ss),
        fill=(82, 91, 102),
    )
    for label, (cx, cy) in zip(LABELS, stations, strict=True):
        draw.rounded_rectangle(
            _scaled_box((cx - 31, cy - 31, cx + 31, cy + 31), ss),
            radius=8 * ss, fill=_PALE_CHIP, outline=(28, 34, 42), width=3 * ss,
        )
        _bitmap(draw, label, (cx, cy), 5, ss)

    marked = stations[initial_state]
    draw.ellipse(
        _scaled_box((marked[0] - 39, marked[1] - 39,
                     marked[0] + 39, marked[1] + 39), ss),
        outline=_MARKERS[palette_variant], width=7 * ss,
    )

    # Ordered cards occupy a separate visual band and are linked by arrowheads.
    card_w, card_h = 58, 104
    gap = 14 if layout_variant == 0 else 18
    total = len(operations) * card_w + (len(operations) - 1) * gap
    start_x = 366 + (514 - total) / 2
    card_y = 176 if layout_variant == 0 else 244
    advance_color = _ADVANCE[palette_variant]
    reflect_color = _REFLECT[palette_variant]
    for index, operation in enumerate(operations):
        x0 = start_x + index * (card_w + gap)
        draw.rounded_rectangle(
            _scaled_box((x0, card_y, x0 + card_w, card_y + card_h), ss),
            radius=7 * ss, fill=_CARD_FILL, outline=(53, 65, 78), width=3 * ss,
        )
        cx, cy = x0 + card_w / 2, card_y + card_h / 2
        if operation == "advance":
            triangle = [(cx - 17, cy + 20), (cx, cy - 22), (cx + 17, cy + 20)]
            draw.polygon([(round(x * ss), round(y * ss)) for x, y in triangle], fill=advance_color)
        else:
            draw.rounded_rectangle(
                _scaled_box((cx - 8, cy - 29, cx + 8, cy + 29), ss),
                radius=4 * ss, fill=reflect_color,
            )
        if index + 1 < len(operations):
            ax = x0 + card_w + gap / 2
            draw.line(
                [(round((ax - 5) * ss), round(cy * ss)),
                 (round((ax + 4) * ss), round(cy * ss))],
                fill=(70, 78, 88), width=3 * ss,
            )
            draw.polygon(
                [(round((ax + 5) * ss), round(cy * ss)),
                 (round(ax * ss), round((cy - 5) * ss)),
                 (round(ax * ss), round((cy + 5) * ss))],
                fill=(70, 78, 88),
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
