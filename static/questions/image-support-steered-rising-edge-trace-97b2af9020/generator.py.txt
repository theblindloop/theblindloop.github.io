"""Deterministic renderer and analytic arm for Rising Edge Trace."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
BACKGROUND = (250, 249, 245)
DARK = (34, 42, 52)
GOLD = (232, 176, 43)
PALETTES = (
    {"H": (28, 181, 190), "L": (113, 74, 181)},
    {"H": (20, 164, 184), "L": (131, 78, 166)},
)


def _decision(commands: list[str]) -> str:
    signal = "L"
    lamp = False
    for command in commands:
        if signal == "L" and command == "H":
            lamp = not lamp
        signal = command
    return "yes" if lamp else "no"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "yes" if int(seed) % 2 == 0 else "no"
    while True:
        commands = [rng.choice(("H", "L")) for _ in range(6)]
        if _decision(commands) == desired and len(set(commands)) == 2 and any(
            commands[i] == commands[i - 1] for i in range(1, 6)
        ):
            break
    # Fixed within-cell offsets keep the short trace's ablation support a thin
    # 3-by-1 run, while the coarse grid cell is randomized over the page.
    origin = [rng.randint(1, 5) * 64 + 28 + rng.randint(-3, 3),
              rng.randint(1, 6) * 64 + 22 + rng.randint(-3, 3)]
    decorations = []
    for _ in range(5):
        decorations.append([rng.randint(24, 488), rng.randint(24, 488), rng.choice((2, 3))])
    return {
        "commands": commands,
        "origin": origin,
        "decorations": decorations,
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def render(scene: dict) -> Image.Image:
    commands = list(scene["commands"])
    x, y = map(int, scene["origin"])
    decorations = list(scene["decorations"])
    palette = PALETTES[int(scene["palette_variant"])]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for dx, dy, radius in decorations:
        draw.ellipse((dx - radius, dy - radius, dx + radius, dy + radius), fill=(218, 218, 212))

    span_w, span_h, gap = 16, 18, 2
    colored_length = len(commands) * span_w + (len(commands) - 1) * gap
    # One attached dark rail makes the visual object a single connected trace.
    draw.rounded_rectangle((x - 4, y - 4, x + colored_length + 4, y + span_h + 4),
                           radius=7, fill=DARK)
    for index, command in enumerate(commands):
        sx = x + index * (span_w + gap)
        draw.rectangle((sx, y, sx + span_w - 1, y + span_h - 1), fill=palette[command])
    # Round gold origin and attached arrow tip declare traversal direction.
    draw.ellipse((x - 12, y + 3, x - 2, y + 13), fill=GOLD, outline=DARK, width=2)
    tip_x = x + colored_length + 14
    mid_y = y + span_h // 2
    draw.polygon(((x + colored_length + 2, mid_y - 8), (tip_x, mid_y),
                  (x + colored_length + 2, mid_y + 8)), fill=GOLD, outline=DARK)
    return image


def analytic_gold(scene: dict) -> str:
    return _decision(list(scene["commands"]))


def margin(scene: dict) -> float:
    # Minimum colored-span width after the two-pixel visual separators.
    return 16.0


def is_quarantined(scene: dict) -> bool:
    return len(scene.get("commands", [])) != 6 or any(c not in {"H", "L"} for c in scene.get("commands", []))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_nonoverlapping_decoration_draw_order", twin)]
