"""Independent image-only inverse arm for Rising Edge Trace."""

from __future__ import annotations

import numpy as np
from PIL import Image

_HIGH = np.array(((28, 181, 190), (20, 164, 184)), dtype=np.int16)
_LOW = np.array(((113, 74, 181), (131, 78, 166)), dtype=np.int16)


def _mask_for(arr: np.ndarray, colors: np.ndarray) -> np.ndarray:
    return np.any(np.all(arr[:, :, None, :] == colors[None, None, :, :], axis=3), axis=2)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = _mask_for(arr, _HIGH)
    low = _mask_for(arr, _LOW)
    ink = high | low
    ys, xs = np.nonzero(ink)
    if len(xs) < 1400:
        raise ValueError("colored command trace is missing or damaged")
    y0, y1 = int(ys.min()), int(ys.max())
    if not 16 <= y1 - y0 + 1 <= 20:
        raise ValueError("command trace height is malformed")
    projection = ink.sum(axis=0)
    occupied = np.flatnonzero(projection > 0)
    runs: list[tuple[int, int]] = []
    for value in occupied:
        if not runs or value > runs[-1][1] + 1:
            runs.append((int(value), int(value)))
        else:
            runs[-1] = (runs[-1][0], int(value))
    if len(runs) != 6:
        raise ValueError("expected six separated command spans")
    commands: list[str] = []
    for left, right in runs:
        width = right - left + 1
        if not 14 <= width <= 17:
            raise ValueError("command span width is malformed")
        h_count = int(high[:, left:right + 1].sum())
        l_count = int(low[:, left:right + 1].sum())
        if max(h_count, l_count) < 240 or min(h_count, l_count) != 0:
            raise ValueError("command span color is ambiguous")
        commands.append("H" if h_count > l_count else "L")
    signal = "L"
    lamp = False
    for command in commands:
        if signal == "L" and command == "H":
            lamp = not lamp
        signal = command
    return "yes" if lamp else "no"
