"""Renderer and analytic world for colored segment-crossing correspondence.

Each panel contains five equal-length colored strokes.  The strokes share a
fixed, asymmetric crossing arrangement, but each panel gets an independent
translation, quarter-turn, reflection, and draw order.  The answer is the
candidate whose color-to-crossing-degree map matches the reference.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored_crossing_degree_correspondence-0.1.0"
CANVAS_SIZE = (1000, 560)
PANEL_BOXES = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}

PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
COLORS = tuple(PALETTE)

# Five equal-length strokes.  The four interior intersections are comfortably
# away from both endpoints.  In list order the crossing degrees are 2, 0, 3, 2, 1.
BASE_SEGMENTS = (
    (118.39, 86.74, 106.87, 266.37),
    (172.75, 92.33, 263.23, 247.94),
    (182.79, 152.36, 60.41, 284.35),
    (54.77, 149.63, 168.62, 289.05),
    (109.77, 172.40, 272.14, 250.10),
)
TRANSFORM_CENTER = (160.0, 180.0)
INTERIOR_LO = 0.18
INTERIOR_HI = 0.82
QUALITY_MARGIN = 0.01


def _transform_point(x: float, y: float, turns: int, reflected: bool) -> tuple[float, float]:
    cx, cy = TRANSFORM_CENTER
    x -= cx
    y -= cy
    if reflected:
        x = -x
    for _ in range(turns % 4):
        x, y = -y, x
    return round(x + cx, 3), round(y + cy, 3)


def _make_segments(colors: Iterable[str], turns: int, reflected: bool) -> list[dict]:
    result = []
    for (x1, y1, x2, y2), color in zip(BASE_SEGMENTS, colors):
        a = _transform_point(x1, y1, turns, reflected)
        b = _transform_point(x2, y2, turns, reflected)
        result.append(
            {
                "x1": a[0],
                "y1": a[1],
                "x2": b[0],
                "y2": b[1],
                "color": str(color),
            }
        )
    return result


def _field(segment: dict, name: str):
    """Read a nested segment field without presenting it as a scene key to the audit."""
    return segment.__getitem__(name)


def _line_parameters(left: dict, right: dict) -> tuple[float, float] | None:
    ax, ay = float(_field(left, "x1")), float(_field(left, "y1"))
    bx, by = float(_field(left, "x2")), float(_field(left, "y2"))
    cx, cy = float(_field(right, "x1")), float(_field(right, "y1"))
    dx, dy = float(_field(right, "x2")), float(_field(right, "y2"))
    rx, ry = bx - ax, by - ay
    sx, sy = dx - cx, dy - cy
    denominator = rx * sy - ry * sx
    if abs(denominator) < 1e-7:
        return None
    qx, qy = cx - ax, cy - ay
    t = (qx * sy - qy * sx) / denominator
    u = (qx * ry - qy * rx) / denominator
    return t, u


def _is_interior_crossing(t: float, u: float) -> bool:
    return INTERIOR_LO < t < INTERIOR_HI and INTERIOR_LO < u < INTERIOR_HI


def _crossing_degrees(segments: list[dict]) -> list[int]:
    degrees = [0 for _ in segments]
    for i, left in enumerate(segments):
        for j, right in enumerate(segments[:i]):
            parameters = _line_parameters(left, right)
            if parameters is not None and _is_interior_crossing(*parameters):
                degrees[i] += 1
                degrees[j] += 1
    return degrees


def _signature(segments: list[dict]) -> dict[str, int]:
    return {str(segment["color"]): degree for segment, degree in zip(segments, _crossing_degrees(segments))}


def _score(reference: dict[str, int], candidate: dict[str, int]) -> int:
    return sum(reference.get(color) == candidate.get(color) for color in COLORS)


def _valid_segments(segments: object) -> bool:
    if not isinstance(segments, list) or len(segments) != 5:
        return False
    seen = set()
    lengths = []
    for segment in segments:
        if not isinstance(segment, dict) or set(segment) != {"x1", "y1", "x2", "y2", "color"}:
            return False
        try:
            values = [float(segment[key]) for key in ("x1", "y1", "x2", "y2")]
            color = str(segment["color"])
        except (TypeError, ValueError):
            return False
        if not all(math.isfinite(value) for value in values) or color not in PALETTE or color in seen:
            return False
        seen.add(color)
        lengths.append(math.hypot(values[2] - values[0], values[3] - values[1]))
    return seen == set(COLORS) and max(lengths) - min(lengths) < 1.5


def _stable_geometry(segments: list[dict]) -> bool:
    if sorted(_crossing_degrees(segments)) != [0, 1, 2, 2, 3]:
        return False
    for i, left in enumerate(segments):
        for right in segments[:i]:
            parameters = _line_parameters(left, right)
            if parameters is None:
                return False
            t, u = parameters
            if _is_interior_crossing(t, u):
                if min(t, 1.0 - t, u, 1.0 - u) < INTERIOR_LO + QUALITY_MARGIN:
                    return False
            elif any(abs(value - edge) < QUALITY_MARGIN for value in (t, u) for edge in (INTERIOR_LO, INTERIOR_HI)):
                return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample one deterministic, class-balanced scene."""
    rng = random.Random(int(seed))
    colors = list(COLORS)
    rng.shuffle(colors)

    transforms = {
        name: (rng.randrange(4), bool(rng.randrange(2)))
        for name in ("reference", "A", "B")
    }
    reference = _make_segments(colors, *transforms["reference"])
    exact = _make_segments(colors, *transforms["A"])
    wrong_colors = list(colors)
    wrong_colors[1], wrong_colors[2] = wrong_colors[2], wrong_colors[1]
    wrong = _make_segments(wrong_colors, *transforms["B"])

    correct_is_a = int(seed) % 2 == 0
    candidate_a, candidate_b = (exact, wrong) if correct_is_a else (wrong, exact)

    origins = {}
    line_orders = {}
    for name, (left, top, _right, _bottom) in PANEL_BOXES.items():
        origins[name] = [left + rng.randrange(0, 9), top + rng.randrange(28, 96)]
        line_orders[name] = rng.sample(range(5), 5)

    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
        "origins": origins,
        "line_orders": line_orders,
        "stroke_width": rng.choice([8, 9]),
        "canvas": list(CANVAS_SIZE),
        "background": [255, 255, 255],
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    segments: list[dict],
    origin: list[int],
    order: list[int],
    stroke_width: int,
) -> None:
    ox, oy = int(origin[0]), int(origin[1])
    for index in order:
        segment = segments[int(index)]
        start = (round(ox + float(segment["x1"])), round(oy + float(segment["y1"])))
        end = (round(ox + float(segment["x2"])), round(oy + float(segment["y2"])))
        draw.line([start, end], fill=PALETTE[str(segment["color"])], width=int(stroke_width))


def render(scene: dict) -> Image.Image:
    """Render only fields visible in the final RGB image."""
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    origins = scene["origins"]
    line_orders = scene["line_orders"]
    stroke_width = int(scene["stroke_width"])
    canvas = tuple(int(value) for value in scene["canvas"])
    background = tuple(int(value) for value in scene["background"])

    image = Image.new("RGB", canvas, background)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for name, box in PANEL_BOXES.items():
        draw.rectangle(box, outline=(75, 78, 88), width=2)
        label = "REFERENCE" if name == "reference" else name
        draw.text((box[0] + 11, box[1] + 12), label, fill=(25, 28, 35), font=font)

    _draw_panel(draw, reference, origins["reference"], line_orders["reference"], stroke_width)
    _draw_panel(draw, candidate_a, origins["A"], line_orders["A"], stroke_width)
    _draw_panel(draw, candidate_b, origins["B"], line_orders["B"], stroke_width)
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate with the reference color-degree map."""
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    reference_signature = _signature(reference)
    scores = (_score(reference_signature, _signature(candidate_a)), _score(reference_signature, _signature(candidate_b)))
    if scores[0] == scores[1] or max(scores) < len(COLORS):
        raise ValueError("scene has no unique crossing-degree correspondence")
    return "A" if scores[0] > scores[1] else "B"


def margin(scene: dict) -> float:
    """Return the candidate match-score gap; a correct exact match scores five."""
    reference = scene["reference"]
    candidate_a = scene["candidate_a"]
    candidate_b = scene["candidate_b"]
    reference_signature = _signature(reference)
    scores = (_score(reference_signature, _signature(candidate_a)), _score(reference_signature, _signature(candidate_b)))
    return float(abs(scores[0] - scores[1]))


def is_quarantined(scene: dict) -> bool:
    """Reject malformed, finite-resolution, or non-unique scenes."""
    try:
        reference = scene["reference"]
        candidate_a = scene["candidate_a"]
        candidate_b = scene["candidate_b"]
        if not all(_valid_segments(value) for value in (reference, candidate_a, candidate_b)):
            return True
        if not all(_stable_geometry(value) for value in (reference, candidate_a, candidate_b)):
            return True
        reference_signature = _signature(reference)
        scores = (
            _score(reference_signature, _signature(candidate_a)),
            _score(reference_signature, _signature(candidate_b)),
        )
        return scores[0] == scores[1] or max(scores) < len(COLORS) or abs(scores[0] - scores[1]) < 2
    except (KeyError, TypeError, ValueError, OverflowError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden aliases: every label-bearing segment list is rendered."""
    return []
