"""Independent image-only inverse for crossing-degree correspondence.

The inverse sees only an RGB image.  It segments the five public palette colors
inside each panel, fits a finite stroke to every color mask with PCA, and counts
pairwise intersections whose fitted intersection lies inside both projected
stroke extents.  It does not import the renderer or access a latent scene.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


_PANEL_BOXES = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}
_PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
_COLORS = tuple(_PALETTE)
_MIN_PIXELS = 500
_INTERIOR_LO = 0.14
_INTERIOR_HI = 0.86


def _points_for_color(image: Image.Image, box: tuple[int, int, int, int], rgb: tuple[int, int, int]) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    left, top, right, bottom = box
    crop = array[top + 2 : bottom - 2, left + 2 : right - 2]
    mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
    ys, xs = np.where(mask)
    if len(xs) < _MIN_PIXELS:
        raise ValueError("missing or tiny colored stroke")
    return np.column_stack((xs + left + 2, ys + top + 2)).astype(float)


def _fit_segment(points: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    center = points.mean(axis=0)
    centered = points - center
    covariance = centered.T @ centered / max(1, len(points) - 1)
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    direction = direction / np.linalg.norm(direction)
    projections = centered @ direction
    low = center + direction * float(projections.min())
    high = center + direction * float(projections.max())
    return low, high


def _intersection_parameters(left: tuple[np.ndarray, np.ndarray], right: tuple[np.ndarray, np.ndarray]) -> tuple[float, float] | None:
    a, b = left
    c, d = right
    ray = b - a
    other = d - c
    denominator = float(ray[0] * other[1] - ray[1] * other[0])
    if abs(denominator) < 1e-7:
        return None
    delta = c - a
    t = float((delta[0] * other[1] - delta[1] * other[0]) / denominator)
    u = float((delta[0] * ray[1] - delta[1] * ray[0]) / denominator)
    return t, u


def _degrees(image: Image.Image, box: tuple[int, int, int, int]) -> dict[str, int]:
    fitted = {
        color: _fit_segment(_points_for_color(image, box, rgb))
        for color, rgb in _PALETTE.items()
    }
    degrees = {color: 0 for color in _COLORS}
    for index, left_color in enumerate(_COLORS):
        for right_color in _COLORS[:index]:
            parameters = _intersection_parameters(fitted[left_color], fitted[right_color])
            if parameters is None:
                continue
            t, u = parameters
            if _INTERIOR_LO < t < _INTERIOR_HI and _INTERIOR_LO < u < _INTERIOR_HI:
                degrees[left_color] += 1
                degrees[right_color] += 1
    return degrees


def decision_from_image(image: Image.Image) -> str:
    """Return A or B from pixels only."""
    reference = _degrees(image, _PANEL_BOXES["reference"])
    candidate_a = _degrees(image, _PANEL_BOXES["A"])
    candidate_b = _degrees(image, _PANEL_BOXES["B"])
    score_a = sum(reference[color] == candidate_a[color] for color in _COLORS)
    score_b = sum(reference[color] == candidate_b[color] for color in _COLORS)
    if score_a == score_b or max(score_a, score_b) != len(_COLORS):
        raise ValueError("image has no unique crossing-degree correspondence")
    return "A" if score_a > score_b else "B"
