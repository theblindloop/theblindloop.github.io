"""Deterministic paired-ordinal-strip renderer and analytic reference."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 520
COLORS = ("red", "teal", "violet")
MARKER_COLORS = {
    "red": [222, 70, 76],
    "teal": [22, 157, 149],
    "violet": [128, 88, 194],
}


def _positions(rng: random.Random, left: int, right: int) -> list[int]:
    """Return three separated x-centres with harmless independent jitter."""
    usable = right - left
    base = [
        left + int(usable * 0.08),
        left + int(usable * 0.50),
        left + int(usable * 0.92),
    ]
    values = [base[index] + rng.randint(-22, 22) for index in range(3)]
    values.sort()
    if values[1] - values[0] < 145:
        values[1] = values[0] + 145
    if values[2] - values[1] < 145:
        values[2] = values[1] + 145
    if values[2] > right - 50:
        shift = values[2] - (right - 50)
        values = [value - shift for value in values]
    return [int(value) for value in values]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced, finite-resolution-safe scene from an integer seed."""
    rng = random.Random(int(seed))
    upper_order = list(COLORS)
    rng.shuffle(upper_order)

    # A transposition of the two non-fixed colors creates exactly one fixed rank.
    fixed_color = COLORS[int(seed) % len(COLORS)]
    lower_order = list(upper_order)
    swap_indices = [index for index, color in enumerate(upper_order) if color != fixed_color]
    lower_order[swap_indices[0]], lower_order[swap_indices[1]] = (
        lower_order[swap_indices[1]],
        lower_order[swap_indices[0]],
    )

    strip_left = rng.choice((42, 50, 58))
    strip_right = WIDTH - rng.choice((42, 50, 58))
    upper_y = rng.choice((124, 144, 164))
    lower_y = rng.choice((356, 376, 396))
    upper_x_positions = _positions(rng, strip_left + 10, strip_right - 10)
    lower_x_positions = _positions(rng, strip_left + 10, strip_right - 10)

    guide_palette = ([184, 194, 205], [174, 187, 200], [194, 199, 210])
    return {
        "upper_order": upper_order,
        "lower_order": lower_order,
        "upper_x_positions": upper_x_positions,
        "lower_x_positions": lower_x_positions,
        "upper_y": int(upper_y),
        "lower_y": int(lower_y),
        "strip_left": int(strip_left),
        "strip_right": int(strip_right),
        "marker_half_size": int(rng.choice((22, 24, 26))),
        "marker_color_map": {name: list(value) for name, value in MARKER_COLORS.items()},
        "guide_color": list(rng.choice(guide_palette)),
        "background": [255, 255, 255],
        "stroke_width": int(rng.choice((2, 3, 4))),
    }


def _draw_strip(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    order: list[str],
    positions: list[int],
    y: int,
) -> None:
    left = int(scene["strip_left"])
    right = int(scene["strip_right"])
    guide = tuple(int(value) for value in scene["guide_color"])
    stroke = int(scene["stroke_width"])
    draw.rounded_rectangle(
        [left, y - 68, right, y + 68],
        radius=18,
        outline=guide,
        width=stroke,
    )
    draw.line([left + 30, y, right - 30, y], fill=guide, width=stroke)
    half = int(scene["marker_half_size"])
    color_map = scene["marker_color_map"]
    for name, x in zip(order, positions, strict=True):
        points = [(int(x), y - half), (int(x) + half, y), (int(x), y + half), (int(x) - half, y)]
        draw.polygon(points, fill=tuple(int(value) for value in color_map[name]))


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the public scene dictionary."""
    image = Image.new(
        "RGB",
        (WIDTH, HEIGHT),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    _draw_strip(
        draw,
        scene,
        scene["upper_order"],
        scene["upper_x_positions"],
        int(scene["upper_y"]),
    )
    _draw_strip(
        draw,
        scene,
        scene["lower_order"],
        scene["lower_x_positions"],
        int(scene["lower_y"]),
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the unique color whose ordinal rank is unchanged."""
    upper_order = scene["upper_order"]
    lower_order = scene["lower_order"]
    fixed = [
        color
        for index, color in enumerate(upper_order)
        if lower_order[index] == color
    ]
    if len(fixed) != 1:
        raise ValueError("scene does not contain exactly one fixed-rank color")
    return fixed[0]


def margin(scene: dict[str, Any]) -> float:
    """Continuous raster margin: smallest marker-edge clearance in either strip."""
    upper = scene["upper_x_positions"]
    lower = scene["lower_x_positions"]
    gaps = [
        int(upper[1]) - int(upper[0]),
        int(upper[2]) - int(upper[1]),
        int(lower[1]) - int(lower[0]),
        int(lower[2]) - int(lower[1]),
    ]
    return float(min(gaps) - 2 * int(scene["marker_half_size"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude rank ties and low-clearance marker arrangements."""
    try:
        gold = analytic_gold(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return gold not in COLORS or margin(scene) < 42.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """There are no declared render-invariant latent transforms in this world."""
    return []
