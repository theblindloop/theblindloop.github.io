"""Independent raster-only inverse for paired ordinal strips.

The measurement path knows only the final RGB image and a public palette. It
finds colored connected components, checks that every color occurs once in each
vertical strip, and compares within-strip centroid ranks.
"""

from __future__ import annotations

from typing import Iterable

import numpy as np
from PIL import Image

_COLORS = ("red", "teal", "violet")
_PALETTE = {
    "red": np.array([222.0, 70.0, 76.0]),
    "teal": np.array([22.0, 157.0, 149.0]),
    "violet": np.array([128.0, 88.0, 194.0]),
}
_COLOR_TOLERANCE = 55.0
_MIN_COMPONENT_AREA = 500
_MAX_COMPONENT_AREA = 1800


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    points = {tuple(int(value) for value in point) for point in np.argwhere(mask)}
    found: list[list[tuple[int, int]]] = []
    while points:
        seed = points.pop()
        stack = [seed]
        component = [seed]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    neighbour = (y + dy, x + dx)
                    if neighbour in points:
                        points.remove(neighbour)
                        stack.append(neighbour)
                        component.append(neighbour)
        found.append(component)
    return found


def _centroid(component: Iterable[tuple[int, int]]) -> tuple[float, float, int]:
    values = list(component)
    if not values:
        raise ValueError("empty color component")
    ys = np.array([value[0] for value in values], dtype=np.float64)
    xs = np.array([value[1] for value in values], dtype=np.float64)
    return float(xs.mean()), float(ys.mean()), len(values)


def _color_centers(array: np.ndarray, name: str) -> list[tuple[float, float]]:
    color = _PALETTE[name]
    distances = np.linalg.norm(array.astype(np.float64) - color, axis=2)
    mask = distances <= _COLOR_TOLERANCE
    components = _components(mask)
    centers: list[tuple[float, float]] = []
    for component in components:
        x, y, area = _centroid(component)
        if _MIN_COMPONENT_AREA <= area <= _MAX_COMPONENT_AREA:
            centers.append((x, y))
    if len(centers) != 2:
        raise ValueError(f"expected two complete {name} markers, found {len(centers)}")
    centers.sort(key=lambda point: point[1])
    if centers[1][1] - centers[0][1] < 100.0:
        raise ValueError(f"{name} markers do not form two separated strips")
    return centers


def decision_from_image(image: Image.Image) -> str:
    """Recover the fixed-rank color from the raster, with strict abstention."""
    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")

    centers = {name: _color_centers(array, name) for name in _COLORS}
    upper = {name: points[0] for name, points in centers.items()}
    lower = {name: points[1] for name, points in centers.items()}
    upper_order = sorted(_COLORS, key=lambda name: upper[name][0])
    lower_order = sorted(_COLORS, key=lambda name: lower[name][0])
    upper_rank = {name: index for index, name in enumerate(upper_order)}
    lower_rank = {name: index for index, name in enumerate(lower_order)}
    fixed = [name for name in _COLORS if upper_rank[name] == lower_rank[name]]
    if len(fixed) != 1:
        raise ValueError("raster does not expose a unique fixed rank")
    return fixed[0]
