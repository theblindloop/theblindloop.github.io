"""Independent inverse arm using only RGB pixels from the final PNG."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    remaining = {tuple(map(int, point)) for point in np.argwhere(mask)}
    found = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        if len(component) >= 35:
            found.append(component)
    return found


def _mask_near(array: np.ndarray, color: tuple[int, int, int], tolerance: float) -> np.ndarray:
    delta = array.astype(np.int16) - np.asarray(color, dtype=np.int16)
    return np.sqrt(np.sum(delta.astype(np.float32) ** 2, axis=2)) < tolerance


def _centers(mask: np.ndarray) -> list[tuple[float, float]]:
    components = _components(mask)
    centers = []
    for component in components:
        if not 90 <= len(component) <= 500:
            continue
        points = np.asarray(component, dtype=np.float64)
        centers.append((float(points[:, 1].mean()), float(points[:, 0].mean())))
    return centers


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan = _centers(_mask_near(array, (0, 151, 190), 72.0))
    orange = _centers(_mask_near(array, (225, 105, 35), 72.0))
    if len(cyan) != 2 or len(orange) != 2:
        raise ValueError("expected exactly two intact cyan and two intact orange checkpoints")

    # Reconstruct the dark route's center and an ordered angular centerline from
    # its pixels.  No renderer constants, scene state, or stored coordinates enter.
    dark = (
        (array[:, :, 0] < 85)
        & (array[:, :, 1] < 100)
        & (array[:, :, 2] < 125)
    )
    route_pixels = np.argwhere(dark)
    if len(route_pixels) < 1000:
        raise ValueError("closed route is not recoverable")
    cy, cx = np.median(route_pixels, axis=0)
    route_angles = np.mod(
        np.arctan2(route_pixels[:, 0] - cy, route_pixels[:, 1] - cx), 2.0 * math.pi
    )
    occupied, _ = np.histogram(route_angles, bins=48, range=(0.0, 2.0 * math.pi))
    if int(np.count_nonzero(occupied)) < 30:
        raise ValueError("route does not expose a closed ordered centerline")

    markers = [(x, y, "cyan") for x, y in cyan] + [(x, y, "orange") for x, y in orange]
    ordered = sorted(
        markers,
        key=lambda item: math.atan2(item[1] - float(cy), item[0] - float(cx)),
    )
    cyan_indices = [index for index, item in enumerate(ordered) if item[2] == "cyan"]
    gap = (cyan_indices[1] - cyan_indices[0]) % 4
    return "yes" if min(gap, 4 - gap) == 1 else "no"
