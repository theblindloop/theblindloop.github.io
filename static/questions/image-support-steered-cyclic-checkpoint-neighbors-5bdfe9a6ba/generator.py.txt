"""Deterministic compact closed-trace world rendered with Pillow."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 640
BACKGROUND = (250, 248, 242)
ROUTE = (31, 45, 68)
CYAN = (0, 151, 190)
ORANGE = (225, 105, 35)
DISTRACTOR = (205, 202, 194)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    centers = (200, 280, 360, 440)
    center = [rng.choice(centers), rng.choice(centers)]
    yes = int(seed) % 2 == 0
    if yes:
        start = rng.randrange(4)
        cyan_slots = sorted((start, (start + 1) % 4))
    else:
        start = rng.randrange(2)
        cyan_slots = [start, start + 2]
    offsets = [round(rng.uniform(-0.055, 0.055), 6) for _ in range(4)]
    distractors = []
    while len(distractors) < 10:
        x, y = rng.randint(35, 605), rng.randint(35, 605)
        if abs(x - center[0]) > 145 or abs(y - center[1]) > 135:
            distractors.append([x, y, rng.randrange(2)])
    return {
        "canvas_size": CANVAS,
        "center": center,
        "rotation": rng.randrange(4),
        "mirror": bool(rng.randrange(2)),
        "shape_variant": rng.randrange(3),
        "cyan_slots": cyan_slots,
        "slot_offsets": offsets,
        "distractors": distractors,
    }


def _transform(x: float, y: float, scene: dict) -> tuple[float, float]:
    if scene["mirror"]:
        x = -x
    for _ in range(int(scene["rotation"]) % 4):
        x, y = -y, x
    return x + float(scene["center"][0]), y + float(scene["center"][1])


def _local_point(theta: float, variant: int) -> tuple[float, float]:
    # A visibly closed, gently irregular centerline.  The deformation is radial,
    # so angular order remains injective while the route is not a canonical circle.
    wobble = (5.0 + 1.5 * variant) * math.sin((3 + variant) * theta + 0.4 * variant)
    return (96.0 + wobble) * math.cos(theta), (76.0 + 0.55 * wobble) * math.sin(theta)


def _route_points(scene: dict) -> list[tuple[float, float]]:
    return [
        _transform(*_local_point(2.0 * math.pi * i / 240.0, int(scene["shape_variant"])), scene)
        for i in range(240)
    ]


def _checkpoint_points(scene: dict) -> list[tuple[float, float]]:
    points = []
    for slot, offset in enumerate(scene["slot_offsets"]):
        theta = slot * math.pi / 2.0 + float(offset)
        points.append(_transform(*_local_point(theta, int(scene["shape_variant"])), scene))
    return points


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, kind in scene["distractors"]:
        if kind:
            draw.ellipse((x - 4, y - 4, x + 4, y + 4), outline=DISTRACTOR, width=2)
        else:
            draw.line((x - 5, y, x + 5, y), fill=DISTRACTOR, width=2)
            draw.line((x, y - 5, x, y + 5), fill=DISTRACTOR, width=2)
    route = _route_points(scene)
    draw.line(route + [route[0]], fill=ROUTE, width=11, joint="curve")
    cyan_slots = set(int(value) for value in scene["cyan_slots"])
    for slot, (x, y) in enumerate(_checkpoint_points(scene)):
        # A background moat keeps every checkpoint a disconnected decisive glyph.
        draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=BACKGROUND)
        color = CYAN if slot in cyan_slots else ORANGE
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=color)
        draw.ellipse((x - 10, y - 10, x + 10, y + 10), outline=(255, 255, 255), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    a, b = sorted(int(value) for value in scene["cyan_slots"])
    return "yes" if min((b - a) % 4, (a - b) % 4) == 1 else "no"


def margin(scene: dict) -> float:
    offsets = [float(value) for value in scene["slot_offsets"]]
    angles = sorted(slot * math.pi / 2.0 + offsets[slot] for slot in range(4))
    gaps = [angles[i + 1] - angles[i] for i in range(3)] + [2 * math.pi + angles[0] - angles[3]]
    # Clearance from checkpoint-order collision, expressed in route radians.
    return round(min(gaps), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.30


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
