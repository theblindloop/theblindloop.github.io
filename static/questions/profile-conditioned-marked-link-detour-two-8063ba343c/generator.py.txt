"""Deterministic renderer and analytic arm for marked_link_detour_two."""

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw

NODE_RADIUS = 14
PALETTES = (
    {
        "background": [249, 247, 240],
        "edge": [42, 48, 58],
        "node": [42, 137, 180],
        "green": [43, 157, 96],
        "purple": [142, 76, 180],
        "mark": [235, 121, 35],

    {

        "edge": [48, 45, 56],

        "purple": [135, 71, 174],
        "mark": [238, 127, 39],

def _edge_key(edge: list[int]) -> tuple[int, int]:
    a, b = int(edge[0]), int(edge[1])
    return (a, b) if a < b else (b, a)


def sample_scene(seed: int) -> dict:
    """Sample matched chorded cycles; only terminal-to-chord relation varies by class."""
    rng = random.Random(int(seed))
    positive = (int(seed) % 2 == 0)
    cx, cy = 224 + rng.randint(-8, 8), 224 + rng.randint(-8, 8)
    rx, ry = rng.randint(128, 140), rng.randint(121, 137)
    phase = rng.uniform(-math.pi, math.pi)
    direction = rng.choice((-1, 1))
    nodes = []
    for index in range(10):
        angle = phase + direction * 2.0 * math.pi * index / 10.0
        nodes.append([
            round(cx + rx * math.cos(angle)),
            round(cy + ry * math.sin(angle)),
        ])

    # Every scene is the same C10 plus a 0--3 chord. Terminal pairs are always
    # antipodal, never chord endpoints, and therefore have identical Euclidean
    # separation and identical local degree. For (4,9) the chord shortens the
    # route from five to three links; for (1,6)/(2,7) it does not shorten it.
    edges = [[index, (index + 1) % 10] for index in range(10)]
    marked = [0, 3]
    edges.append(marked[:])
    if positive:
        terminal_pair = [4, 9]
    else:
        terminal_pair = list(rng.choice(((1, 6), (2, 7))))
    if rng.randrange(2):
        terminal_pair.reverse()
    green_node, purple_node = terminal_pair

    # Relabel latent vertices and reorder edges independently of all pixels.
    order = list(range(10))
    rng.shuffle(order)
    inverse = {old: new for new, old in enumerate(order)}
    nodes = [nodes[old] for old in order]
    edges = [[inverse[a], inverse[b]] for a, b in edges]
    marked = [inverse[marked[0]], inverse[marked[1]]]
    green_node, purple_node = inverse[green_node], inverse[purple_node]
    rng.shuffle(edges)

    return {
        "nodes": nodes,
        "edges": edges,
        "marked_edge": marked,
        "green_node": green_node,
        "purple_node": purple_node,
        "palette": deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
    }


def render(scene: dict) -> Image.Image:
    """Draw separated link strokes behind solid circular junction rings."""
    scale = 4
    nodes = scene["nodes"]
    edges = scene["edges"]
    marked = _edge_key(scene["marked_edge"])
    green_node = int(scene["green_node"])
    purple_node = int(scene["purple_node"])
    palette = scene["palette"]
    image = Image.new(
        "RGB",
        (CANVAS * scale, CANVAS * scale),
        tuple(palette["background"]),
    )
    draw = ImageDraw.Draw(image)
    for edge in edges:
        a, b = int(edge[0]), int(edge[1])
        points = [
            tuple(value * scale for value in nodes[a]),
            tuple(value * scale for value in nodes[b]),
        ]
        color = palette["mark"] if _edge_key(edge) == marked else palette["edge"]
        draw.line(points, fill=tuple(color), width=8 * scale)

    radius = NODE_RADIUS * scale
    center_hole = 3 * scale
    for index, (x, y) in enumerate(nodes):
        if index == green_node:
            color = palette["green"]
        elif index == purple_node:
            color = palette["purple"]
        else:
            color = palette["node"]
        px, py = x * scale, y * scale
        draw.ellipse(
            (px - radius, py - radius, px + radius, py + radius),
            fill=tuple(color),
        )
        draw.ellipse(
            (px - center_hole, py - center_hole, px + center_hole, py + center_hole),
            fill=tuple(palette["background"]),
        )
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def _distance(
    edges: list[list[int]],
    start: int,
    target: int,
    omitted: tuple[int, int] | None = None,
) -> int | None:
    vertices = {int(value) for edge in edges for value in edge}
    adjacency = {vertex: set() for vertex in vertices}
    skipped = False
    for edge in edges:
        key = _edge_key(edge)
        if omitted is not None and key == omitted and not skipped:
            skipped = True
            continue
        a, b = int(edge[0]), int(edge[1])
        adjacency[a].add(b)
        adjacency[b].add(a)
    if start not in adjacency or target not in adjacency:
        return None
    queue = deque([(start, 0)])
    seen = {start}
    while queue:
        vertex, length = queue.popleft()
        if vertex == target:
            return length
        for neighbor in adjacency[vertex]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append((neighbor, length + 1))
    return None


def _shortest_lengths(scene: dict) -> tuple[int | None, int | None]:
    edges = scene["edges"]
    marked = _edge_key(scene["marked_edge"])
    green = int(scene["green_node"])
    purple = int(scene["purple_node"])
    return (
        _distance(edges, green, purple),
        _distance(edges, green, purple, marked),
    )


def analytic_gold(scene: dict) -> str:
    current, after = _shortest_lengths(scene)
    if current is None or after is None:
        return "no"
    return "yes" if after - current == 2 else "no"


def margin(scene: dict) -> float:
    """Runner-up gap in whole-link units around the declared increase of two."""
    current, after = _shortest_lengths(scene)
    if current is None or after is None:
        return 1.0
    return float(max(1, abs((after - current) - 2)))


def _point_segment_distance(point: list[int], a: list[int], b: list[int]) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    denominator = vx * vx + vy * vy
    if denominator == 0:
        return math.inf
    fraction = max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / denominator))
    qx, qy = ax + fraction * vx, ay + fraction * vy
    return math.hypot(px - qx, py - qy)


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    if len(nodes) != 10 or len(edges) != 11:
        return True
    if int(scene["green_node"]) == int(scene["purple_node"]):
        return True
    keys = [_edge_key(edge) for edge in edges]
    if len(set(keys)) != 11 or _edge_key(scene["marked_edge"]) not in keys:
        return True
    if min(
        math.dist(first, second)
        for index, first in enumerate(nodes)
        for second in nodes[index + 1 :]
    ) < 54:
        return True
    for edge in edges:
        a, b = int(edge[0]), int(edge[1])
        if math.dist(nodes[a], nodes[b]) < 58:
            return True
        for index, point in enumerate(nodes):
            if index not in (a, b) and _point_segment_distance(point, nodes[a], nodes[b]) < 30:
                return True
    current, after = _shortest_lengths(scene)
    return current is None or after is None


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_link_draw_list_order", twin)]
