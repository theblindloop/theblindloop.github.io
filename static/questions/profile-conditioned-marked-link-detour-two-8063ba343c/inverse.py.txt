"""Independent image-only inverse arm for marked_link_detour_two.

The arm segments colored junction rings and each exposed link stroke, binds the
strokes to recovered junction centers, and compares shortest paths before and
after omitting the uniquely orange stroke. It imports no candidate module and
receives no latent scene data.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
            ):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _center(component: np.ndarray) -> np.ndarray:
    return np.asarray([component[:, 1].mean(), component[:, 0].mean()])


def _bind_link(points: np.ndarray, centers: np.ndarray) -> tuple[int, int]:
    xy = points[:, ::-1]
    distances = []
    for center in centers:
        delta = xy - center
        distances.append(float(np.min(np.sum(delta * delta, axis=1))))
    order = np.argsort(distances)
    if len(order) < 2 or distances[int(order[1])] > 23.0**2:
        raise ValueError("link endpoints are not recoverable")
    first, second = int(order[0]), int(order[1])
    return (first, second) if first < second else (second, first)


def _bind_terminal(component: np.ndarray, centers: np.ndarray) -> int:
    point = _center(component)
    distances = np.sum((centers - point) ** 2, axis=1)
    index = int(np.argmin(distances))
    if float(distances[index]) > 3.0**2:
        raise ValueError("terminal marker does not bind to a junction")
    return index


def _distance(edges: list[tuple[int, int]], start: int, target: int) -> int | None:
    adjacency = {index: set() for index in range(10)}
    for first, second in edges:
        adjacency[first].add(second)
        adjacency[second].add(first)
    queue = deque([(start, 0)])
    seen = {start}
    while queue:
        vertex, length = queue.popleft()
        if vertex == target:
            return length
        for neighbor in adjacency[vertex]:
            if neighbor not in seen:
                seen.add(neighbor)
                queue.append((neighbor, length + 1))
    return None


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]

    orange = (
        (red > 175)
        & (red > green + 60)
        & (green > 65)
        & (green < 180)
        & (blue < 115)
    )
    green_terminal = (
        (red < 115)
        & (green > 120)
        & (green > red + 55)
        & (green > blue + 25)
        & ~orange
    )
    purple_terminal = (
        (red > 105)
        & (red < 190)
        & (blue > 135)
        & (blue > green + 45)
        & ~orange
    )
    neutral_node = (
        (red < 100)
        & (green > 100)
        & (blue > 145)
        & (blue > green + 22)
        & ~green_terminal
        & ~purple_terminal
    )
    node_mask = neutral_node | green_terminal | purple_terminal
    dark = (
        (red < 105)
        & (green < 105)
        & (blue < 120)
        & ~node_mask
        & ~orange
    )

    node_parts = _components(node_mask, 170)
    green_parts = _components(green_terminal, 170)
    purple_parts = _components(purple_terminal, 170)
    dark_parts = _components(dark, 100)
    orange_parts = _components(orange, 100)
    if len(node_parts) != 10:
        raise ValueError(f"expected 10 visible junctions, found {len(node_parts)}")
    if len(green_parts) != 1 or len(purple_parts) != 1:
        raise ValueError("green and purple terminal junctions must be unique")
    if len(dark_parts) != 10 or len(orange_parts) != 1:
        raise ValueError(
            f"expected 10 dark links and one orange link, found {len(dark_parts)} and {len(orange_parts)}"
        )

    centers = np.asarray([_center(component) for component in node_parts])
    green_index = _bind_terminal(green_parts[0], centers)
    purple_index = _bind_terminal(purple_parts[0], centers)
    if green_index == purple_index:
        raise ValueError("terminal colors collapse onto one junction")

    dark_edges = [_bind_link(component, centers) for component in dark_parts]
    marked_edge = _bind_link(orange_parts[0], centers)
    if len(set(dark_edges)) != 10 or marked_edge in dark_edges:
        raise ValueError("visible graph has duplicate or ambiguous links")
    current = _distance(dark_edges + [marked_edge], green_index, purple_index)
    after = _distance(dark_edges, green_index, purple_index)
    if current is None or after is None:
        return "no"
    return "yes" if after - current == 2 else "no"
