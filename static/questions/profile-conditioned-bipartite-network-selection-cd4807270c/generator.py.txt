"""Deterministic renderer and analytic world for bipartite-network selection."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 840, 360
CANVAS = (WIDTH, HEIGHT)
BACKGROUND = (255, 255, 255)
LINK = (46, 53, 64)
COLORS = {
    "red": (214, 75, 87),
    "blue": (61, 111, 208),
    "green": (50, 159, 103),
}
COLOR_NAMES = tuple(COLORS)
NODE_RADIUS = 13.0
LINK_WIDTH = 7.0
SUPERSAMPLE = 4
SLOT_CENTERS = ((140.0, 180.0), (420.0, 180.0), (700.0, 180.0))

# Every topology has six nodes, eight links, the same sorted degree sequence,
# and the same chord-length histogram on the regular hexagon.  The first is
# K(2,4), hence bipartite.  The other two contain visible odd cycles.
TOPOLOGIES: dict[str, tuple[tuple[int, int], ...]] = {
    "bipartite": (
        (0, 1), (0, 2), (0, 3), (0, 4),
        (1, 5), (2, 5), (3, 5), (4, 5),
    ),
    "odd_a": (
        (0, 1), (0, 2), (0, 3), (0, 4),
        (1, 2), (1, 4), (1, 5), (3, 5),
    ),
    "odd_b": (
        (0, 1), (0, 2), (0, 3), (0, 4),
        (1, 2), (2, 4), (2, 5), (3, 5),
    ),
}


def _base_nodes() -> tuple[tuple[float, float], ...]:
    return tuple(
        (
            math.cos(-math.pi / 2.0 + 2.0 * math.pi * index / 6.0),
            math.sin(-math.pi / 2.0 + 2.0 * math.pi * index / 6.0),
        )
        for index in range(6)
    )


def _degrees(node_count: int, edges: Iterable[tuple[int, int]]) -> list[int]:
    degrees = [0] * node_count
    for left, right in edges:
        degrees[left] += 1
        degrees[right] += 1
    return degrees


def _connected(node_count: int, edges: Iterable[tuple[int, int]]) -> bool:
    adjacency = [[] for _ in range(node_count)]
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    if not node_count:
        return False
    seen = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for neighbor in adjacency[node]:
            if neighbor not in seen:
                seen.add(neighbor)
                pending.append(neighbor)
    return len(seen) == node_count


def _is_bipartite(node_count: int, edge_values: Iterable[tuple[int, int]]) -> bool:
    edges = [tuple(edge) for edge in edge_values]
    if node_count != 6 or len(edges) != 8 or not _connected(node_count, edges):
        return False
    adjacency = [[] for _ in range(node_count)]
    for left, right in edges:
        if left == right or not (0 <= left < node_count and 0 <= right < node_count):
            return False
        adjacency[left].append(right)
        adjacency[right].append(left)
    labels: list[int | None] = [None] * node_count
    for start in range(node_count):
        if labels[start] is not None:
            continue
        labels[start] = 0
        pending = [start]
        while pending:
            node = pending.pop()
            for neighbor in adjacency[node]:
                if labels[neighbor] is None:
                    labels[neighbor] = 1 - labels[node]
                    pending.append(neighbor)
                elif labels[neighbor] == labels[node]:
                    return False
    return True


def _transformed_nodes(
    center: tuple[float, float], scale: float, rotation: float, skew: float
) -> list[list[float]]:
    cosine, sine = math.cos(rotation), math.sin(rotation)
    output: list[list[float]] = []
    for x, y in _base_nodes():
        xr = cosine * x - sine * y
        yr = sine * x + cosine * y
        # A small visible anisotropy prevents a fixed orientation shortcut while
        # keeping every node comfortably separated from every other node.
        xr *= 1.0 + skew
        yr *= 1.0 - skew * 0.55
        output.append(
            [round(center[0] + scale * xr, 3), round(center[1] + scale * yr, 3)]
        )
    return output


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; the seed is the only random input."""
    rng = random.Random(int(seed))
    winner = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    slots = list(SLOT_CENTERS)
    rng.shuffle(slots)
    scale = rng.uniform(67.0, 74.0)
    networks: list[dict] = []
    distractor_kinds = ["odd_a", "odd_b"]
    rng.shuffle(distractor_kinds)
    distractor_index = 0
    for color, center in zip(COLOR_NAMES, slots):
        kind = "bipartite" if color == winner else distractor_kinds[distractor_index]
        distractor_index += color != winner
        graph_center = (
            center[0] + rng.uniform(-7.0, 7.0),
            center[1] + rng.uniform(-7.0, 7.0),
        )
        networks.append(
            {
                "color": color,
                "nodes": _transformed_nodes(
                    graph_center,
                    scale,
                    rng.uniform(-math.pi, math.pi),
                    rng.uniform(-0.035, 0.035),
                ),
                "edges": [list(edge) for edge in TOPOLOGIES[kind]],
            }
        )
    rng.shuffle(networks)
    return {"networks": networks}


def _draw_network(
    draw: ImageDraw.ImageDraw,
    nodes: list[list[float]],
    edges: list[list[int]],
    color: tuple[int, int, int],
    ss: int,
) -> None:
    for left, right in edges:
        draw.line(
            [
                (round(nodes[left][0] * ss), round(nodes[left][1] * ss)),
                (round(nodes[right][0] * ss), round(nodes[right][1] * ss)),
            ],
            fill=LINK,
            width=round(LINK_WIDTH * ss),
        )
    radius = NODE_RADIUS * ss
    for x, y in nodes:
        cx, cy = x * ss, y * ss
        draw.ellipse(
            (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)),
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for network in scene["networks"]:
        _draw_network(draw, network["nodes"], network["edges"], COLORS[network["color"]], ss)
    return image.resize(CANVAS, Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    winners = []
    for network in scene["networks"]:
        edges = [tuple(edge) for edge in network["edges"]]
        if _is_bipartite(len(network["nodes"]), edges):
            winners.append(network["color"])
    if len(winners) != 1:
        raise ValueError(f"expected one bipartite winner, got {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(
        (
            1.0
            if _is_bipartite(len(network["nodes"]), [tuple(edge) for edge in network["edges"]])
            else 0.0
        )
        for network in scene["networks"]
    )
    if len(scores) < 2:
        return 0.0
    return float(scores[-1] - scores[-2])


def _minimum_node_separation(nodes: list[list[float]]) -> float:
    return min(
        math.dist(nodes[left], nodes[right])
        for left in range(len(nodes))
        for right in range(left + 1, len(nodes))
    )


def is_quarantined(scene: dict) -> bool:
    try:
        networks = scene["networks"]
        if len(networks) != 3:
            return True
        colors = [network["color"] for network in networks]
        if set(colors) != set(COLOR_NAMES):
            return True
        if margin(scene) < 1.0:
            return True
        for network in networks:
            nodes = network["nodes"]
            edges = [tuple(edge) for edge in network["edges"]]
            if len(nodes) != 6 or len(edges) != 8:
                return True
            if any(not (20.0 <= x < WIDTH - 20.0 and 20.0 <= y < HEIGHT - 20.0) for x, y in nodes):
                return True
            if _minimum_node_separation(nodes) < 34.0:
                return True
            if not _connected(6, edges):
                return True
            if any(left == right or not (0 <= left < 6 and 0 <= right < 6) for left, right in edges):
                return True
            if len(set(tuple(sorted(edge)) for edge in edges)) != 8:
                return True
            if min(math.dist(nodes[left], nodes[right]) for left, right in edges) < 42.0:
                return True
        return False
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
