"""Independent image-only inverse for bipartite network selection."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 840, 360
LINK_RGB = np.array([46.0, 53.0, 64.0])
NODE_COLORS = {
    "red": np.array([214.0, 75.0, 87.0]),
    "blue": np.array([61.0, 111.0, 208.0]),
    "green": np.array([50.0, 159.0, 103.0]),
}
SLOT_X = (140.0, 420.0, 700.0)
NODE_RADIUS = 13.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return coordinate arrays for separated filled node-color components."""
    ys, xs = np.nonzero(mask)
    available = set(zip(xs.tolist(), ys.tolist()))
    result: list[np.ndarray] = []
    while available:
        start = available.pop()
        queue = deque([start])
        points = [start]
        while queue:
            x, y = queue.popleft()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in available:
                    available.remove(neighbor)
                    queue.append(neighbor)
                    points.append(neighbor)
        if len(points) >= 120:
            result.append(np.asarray(points, dtype=np.float64))
    return result


def _node_centers(arr: np.ndarray, color: np.ndarray) -> list[tuple[float, float]]:
    distance = np.sqrt(np.sum((arr - color) ** 2, axis=2))
    mask = distance <= 78.0
    centers = []
    for component in _components(mask):
        centers.append((float(component[:, 0].mean()), float(component[:, 1].mean())))
    return centers


def _gray_link_mask(arr: np.ndarray) -> np.ndarray:
    spread = arr.max(axis=2) - arr.min(axis=2)
    darkness = arr.mean(axis=2)
    return (spread <= 25.0) & (darkness <= 155.0)


def _near_gray(mask: np.ndarray, x: float, y: float, radius: int = 2) -> bool:
    ix, iy = int(round(x)), int(round(y))
    left, right = max(0, ix - radius), min(mask.shape[1], ix + radius + 1)
    top, bottom = max(0, iy - radius), min(mask.shape[0], iy + radius + 1)
    return bool(mask[top:bottom, left:right].any())


def _edge_support(
    first: tuple[float, float], second: tuple[float, float], gray: np.ndarray
) -> float:
    dx, dy = second[0] - first[0], second[1] - first[1]
    length = math.hypot(dx, dy)
    if length < 40.0:
        return 0.0
    start = (NODE_RADIUS + 8.0) / length
    stop = 1.0 - start
    samples = max(32, int(length * 0.62))
    hits = 0
    for index in range(samples):
        t = start + (stop - start) * index / max(1, samples - 1)
        if _near_gray(gray, first[0] + t * dx, first[1] + t * dy):
            hits += 1
    return hits / samples


def _recover_edges(centers: list[tuple[float, float]], gray: np.ndarray) -> list[tuple[int, int]]:
    candidates = []
    for left in range(len(centers)):
        for right in range(left + 1, len(centers)):
            support = _edge_support(centers[left], centers[right], gray)
            candidates.append((support, left, right))
    candidates.sort(reverse=True)
    selected = [(left, right) for support, left, right in candidates if support >= 0.72]
    if len(selected) != 8:
        # A strict visible-link count is part of the public contract.  Keeping
        # only high-support candidates prevents a crossing from becoming an
        # invented edge, while the explicit failure quarantines weak rasters.
        raise ValueError(f"expected eight visible links, recovered {len(selected)}")
    return selected


def _bipartite(node_count: int, edges: list[tuple[int, int]]) -> bool:
    if node_count != 6 or len(edges) != 8:
        return False
    adjacency = [[] for _ in range(node_count)]
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    labels: list[int | None] = [None] * node_count
    seen = {0}
    queue = deque([0])
    while queue:
        node = queue.popleft()
        for neighbor in adjacency[node]:
            if labels[neighbor] is None:
                labels[neighbor] = 1 - (labels[node] or 0)
                queue.append(neighbor)
            elif labels[neighbor] == labels[node]:
                return False
            seen.add(neighbor)
    return len(seen) == node_count


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique bipartite colored network using only the RGB raster."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    gray = _gray_link_mask(arr)
    answers = []
    for name, color in NODE_COLORS.items():
        all_centers = _node_centers(arr, color)
        for slot_x in SLOT_X:
            centers = sorted(
                [center for center in all_centers if abs(center[0] - slot_x) < 115.0],
                key=lambda point: (point[1], point[0]),
            )
            if len(centers) != 6:
                continue
            edges = _recover_edges(centers, gray)
            if _bipartite(len(centers), edges):
                answers.append(name)
    if len(answers) != 1:
        raise ValueError(f"expected one bipartite network, recovered {answers}")
    return answers[0]
