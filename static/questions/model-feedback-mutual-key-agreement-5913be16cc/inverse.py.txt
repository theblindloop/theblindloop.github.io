"""Independent image-only inverse for the mutual-key agreement worksheet."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1040
HEIGHT = 740
KEY_TABLE_Y = 110
KEY_CELL_W = 78
KEY_CELL_H = 52
ROW_TABLE_X = 158
COLUMN_TABLE_X = 678
ROW_LABEL_X = 100
COLUMN_LABEL_X = 620
KEY_HEADER_Y = 78
KEY_ROW_CENTERS = (136, 188, 240, 292)
ROW_KEY_CENTERS_X = (197, 275, 353, 431)
COLUMN_KEY_CENTERS_X = (717, 795, 873, 951)
GRID_X = 660
GRID_Y = 423
GRID_CELL = 70
BOARD_CENTERS_X = (695, 765, 835, 905)
BOARD_CENTERS_Y = (458, 528, 598, 668)

ROW_SYMBOLS = {
    "1111100010001111",
    "1001011001101001",
    "1111010001000100",
    "1000110001100011",
}
COLUMN_STAMPS = {
    "1111100110011111",
    "0110011011110110",
    "1000010000100001",
    "0001011101111111",
}

BACKGROUND = (248, 247, 242)
STAMP_BG = (255, 255, 255)
INK = (28, 36, 49)
STAMP_INK = (24, 91, 139)
GUIDE = (91, 104, 119)
ROW_RGB = {
    "red": (211, 62, 57),
    "blue": (43, 101, 201),
    "green": (34, 151, 88),
    "orange": (230, 123, 27),
}
DISK_RGB = {
    "magenta": (191, 48, 143),
    "teal": (13, 153, 163),
    "gold": (224, 177, 31),
    "purple": (112, 66, 188),
}
ANSWER_RGB = {
    "amber": (244, 163, 27),
    "cyan": (27, 184, 207),
    "violet": (145, 82, 205),
    "lime": (116, 186, 51),
}


class OracleError(ValueError):
    """Raised when the raster does not encode one intact mutual-key worksheet."""


def _require_uniform(
    array: np.ndarray,
    left: int,
    top: int,
    right: int,
    bottom: int,
    color: tuple[int, int, int],
    what: str,
) -> None:
    patch = array[top:bottom, left:right]
    if patch.size == 0 or not np.all(patch == np.asarray(color, dtype=np.uint8)):
        raise OracleError(f"{what} is missing or damaged")


def _identify_uniform(
    array: np.ndarray,
    box: tuple[int, int, int, int],
    palette: dict[str, tuple[int, int, int]],
    what: str,
) -> str:
    left, top, right, bottom = box
    patch = array[top:bottom, left:right]
    matches = [
        name
        for name, color in palette.items()
        if patch.size and np.all(patch == np.asarray(color, dtype=np.uint8))
    ]
    if len(matches) != 1:
        raise OracleError(f"{what} has no unique declared color")
    return matches[0]


def _read_mosaic(
    array: np.ndarray,
    center_x: int,
    center_y: int,
    ink: tuple[int, int, int],
    alphabet: set[str],
    what: str,
) -> str:
    _require_uniform(array, center_x - 19, center_y - 23, center_x + 20, center_y - 21, GUIDE, f"{what} top frame")
    _require_uniform(array, center_x - 19, center_y + 22, center_x + 20, center_y + 24, GUIDE, f"{what} bottom frame")
    ink_value = np.asarray(ink, dtype=np.uint8)
    white_value = np.asarray(STAMP_BG, dtype=np.uint8)
    values: list[str] = []
    start_x = center_x - 17
    start_y = center_y - 17
    for row in range(4):
        for column in range(4):
            left = start_x + column * 9
            top = start_y + row * 9
            patch = array[top : top + 8, left : left + 8]
            if np.all(patch == ink_value):
                values.append("1")
            elif np.all(patch == white_value):
                values.append("0")
            else:
                raise OracleError(f"{what} has an ambiguous mosaic cell")
    bits = "".join(values)
    if bits not in alphabet:
        raise OracleError(f"{what} is not in its visible alphabet")
    return bits


def _validate_structure(array: np.ndarray) -> None:
    for x, y in ((0, 0), (WIDTH - 1, 0), (0, HEIGHT - 1), (WIDTH - 1, HEIGHT - 1)):
        if tuple(int(value) for value in array[y, x]) != BACKGROUND:
            raise OracleError("canvas corners do not contain the declared background")
    for table_x in (ROW_TABLE_X, COLUMN_TABLE_X):
        for row in range(4):
            for column in range(4):
                left = table_x + column * KEY_CELL_W
                top = KEY_TABLE_Y + row * KEY_CELL_H
                _require_uniform(array, left + 2, top, left + KEY_CELL_W - 1, top + 2, GUIDE, "key-table horizontal frame")
                _require_uniform(array, left, top + 2, left + 2, top + KEY_CELL_H - 1, GUIDE, "key-table vertical frame")
    for row in range(4):
        for column in range(4):
            left = GRID_X + column * GRID_CELL
            top = GRID_Y + row * GRID_CELL
            _require_uniform(array, left, top, left + GRID_CELL, top + 5, INK, "answer-grid horizontal frame")
            _require_uniform(array, left, top, left + 5, top + GRID_CELL, INK, "answer-grid vertical frame")


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected functions and solve their unique reciprocal pair."""

    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected a {WIDTH} by {HEIGHT} raster")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (HEIGHT, WIDTH, 3):
        raise OracleError("expected one RGB worksheet")
    _validate_structure(array)

    row_stamp_headers = [
        _read_mosaic(array, center_x, KEY_HEADER_Y, STAMP_INK, COLUMN_STAMPS, f"ROW KEY stamp header {column}")
        for column, center_x in enumerate(ROW_KEY_CENTERS_X)
    ]
    if set(row_stamp_headers) != COLUMN_STAMPS:
        raise OracleError("ROW KEY stamp headers are not a permutation")
    row_symbols: list[str] = []
    row_outputs: list[list[str]] = []
    for row, center_y in enumerate(KEY_ROW_CENTERS):
        row_symbols.append(_read_mosaic(array, ROW_LABEL_X, center_y, INK, ROW_SYMBOLS, f"ROW KEY symbol {row}"))
        outputs = [
            _identify_uniform(array, (center_x - 17, center_y - 8, center_x + 18, center_y + 9), ROW_RGB, f"ROW KEY output {row},{column}")
            for column, center_x in enumerate(ROW_KEY_CENTERS_X)
        ]
        if set(outputs) != set(ROW_RGB):
            raise OracleError("each ROW KEY row must be a color bijection")
        row_outputs.append(outputs)
    if set(row_symbols) != ROW_SYMBOLS:
        raise OracleError("ROW KEY symbols are not a permutation")

    column_color_headers = [
        _identify_uniform(array, (center_x - 17, KEY_HEADER_Y - 8, center_x + 18, KEY_HEADER_Y + 9), ROW_RGB, f"COLUMN KEY color header {column}")
        for column, center_x in enumerate(COLUMN_KEY_CENTERS_X)
    ]
    if set(column_color_headers) != set(ROW_RGB):
        raise OracleError("COLUMN KEY color headers are not a permutation")
    column_disks: list[str] = []
    column_outputs: list[list[str]] = []
    for row, center_y in enumerate(KEY_ROW_CENTERS):
        column_disks.append(_identify_uniform(array, (COLUMN_LABEL_X - 9, center_y - 9, COLUMN_LABEL_X + 10, center_y + 10), DISK_RGB, f"COLUMN KEY disk {row}"))
        outputs = [
            _read_mosaic(array, center_x, center_y, STAMP_INK, COLUMN_STAMPS, f"COLUMN KEY output {row},{column}")
            for column, center_x in enumerate(COLUMN_KEY_CENTERS_X)
        ]
        if set(outputs) != COLUMN_STAMPS:
            raise OracleError("each COLUMN KEY row must be a stamp bijection")
        column_outputs.append(outputs)
    if set(column_disks) != set(DISK_RGB):
        raise OracleError("COLUMN KEY disks are not a permutation")

    query_symbol = _read_mosaic(array, 145, 505, INK, ROW_SYMBOLS, "query symbol")
    query_disk = _identify_uniform(array, (364, 496, 383, 515), DISK_RGB, "query disk")
    symbol_row = row_symbols.index(query_symbol)
    disk_row = column_disks.index(query_disk)
    row_map = {stamp: row_outputs[symbol_row][column] for column, stamp in enumerate(row_stamp_headers)}
    column_map = {tag: column_outputs[disk_row][column] for column, tag in enumerate(column_color_headers)}
    pairs = [(tag, stamp) for stamp, tag in row_map.items() if column_map.get(tag) == stamp]
    if len(pairs) != 1:
        raise OracleError("selected key rows do not have one reciprocal pair")
    target_tag, target_stamp = pairs[0]

    board_columns = [
        _read_mosaic(array, center_x, 392, STAMP_INK, COLUMN_STAMPS, f"board column {column}")
        for column, center_x in enumerate(BOARD_CENTERS_X)
    ]
    if set(board_columns) != COLUMN_STAMPS:
        raise OracleError("board column headers are not a permutation")
    board_rows = [
        _identify_uniform(array, (608, center_y - 8, 643, center_y + 9), ROW_RGB, f"board row {row}")
        for row, center_y in enumerate(BOARD_CENTERS_Y)
    ]
    if set(board_rows) != set(ROW_RGB):
        raise OracleError("board row headers are not a permutation")
    cells: list[list[str]] = []
    for row, center_y in enumerate(BOARD_CENTERS_Y):
        values = [
            _identify_uniform(array, (center_x - 20, center_y - 20, center_x + 21, center_y + 21), ANSWER_RGB, f"board cell {row},{column}")
            for column, center_x in enumerate(BOARD_CENTERS_X)
        ]
        if set(values) != set(ANSWER_RGB):
            raise OracleError("each answer-grid row must contain all four answer colors")
        cells.append(values)
    return cells[board_rows.index(target_tag)][board_columns.index(target_stamp)]
