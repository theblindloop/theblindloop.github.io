WIDTH = 1040
HEIGHT = 740

KEY_TABLE_Y = 110
KEY_CELL_W = 78
KEY_CELL_H = 52
ROW_TABLE_X = 158
COLUMN_TABLE_X = 678
ROW_LABEL_X = 100
COLUMN_LABEL_X = 620
KEY_HEADER_Y = 78
KEY_ROW_CENTERS = tuple(KEY_TABLE_Y + row * KEY_CELL_H + KEY_CELL_H // 2 for row in range(4))
ROW_KEY_CENTERS_X = tuple(ROW_TABLE_X + column * KEY_CELL_W + KEY_CELL_W // 2 for column in range(4))
COLUMN_KEY_CENTERS_X = tuple(COLUMN_TABLE_X + column * KEY_CELL_W + KEY_CELL_W // 2 for column in range(4))

GRID_X = 660
GRID_Y = 423
GRID_CELL = 70
BOARD_CENTERS_X = tuple(GRID_X + column * GRID_CELL + GRID_CELL // 2 for column in range(4))
BOARD_CENTERS_Y = tuple(GRID_Y + row * GRID_CELL + GRID_CELL // 2 for row in range(4))

ROW_SYMBOLS = (
    "1111100010001111",  # open box
    "1001011001101001",  # crossed corners
    "1111010001000100",  # tee
    "1000110001100011",  # staircase
)
COLUMN_STAMPS = (
    "1111100110011111",  # ring
    "0110011011110110",  # plus
    "1000010000100001",  # diagonal
    "0001011101111111",  # wedge
)
ROW_TAGS = ("red", "blue", "green", "orange")
DISK_TAGS = ("magenta", "teal", "gold", "purple")
ANSWERS = ("amber", "cyan", "violet", "lime")

RGB = {
    "background": (248, 247, 242),
    "panel": (232, 236, 240),
    "panel_edge": (116, 130, 146),
    "stamp_bg": (255, 255, 255),
    "ink": (28, 36, 49),
    "stamp_ink": (24, 91, 139),
    "guide": (91, 104, 119),
    "accent": (210, 150, 28),
    "red": (211, 62, 57),
    "blue": (43, 101, 201),
    "green": (34, 151, 88),
    "orange": (230, 123, 27),
    "magenta": (191, 48, 143),
    "teal": (13, 153, 163),
    "gold": (224, 177, 31),
    "purple": (112, 66, 188),
    "amber": (244, 163, 27),
    "cyan": (27, 184, 207),
    "violet": (145, 82, 205),
    "lime": (116, 186, 51),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _shuffled(rng: random.Random, values: tuple[str, ...]) -> list[str]:
    result = list(values)
    rng.shuffle(result)
    return result


def _agreement_pairs(scene: dict) -> list[tuple[str, str]]:
    symbol_row = scene["row_symbols"].index(scene["query_symbol"])
    disk_row = scene["column_disks"].index(scene["query_disk"])
    row_map = {
        stamp: scene["row_outputs"][symbol_row][column]
        for column, stamp in enumerate(scene["row_stamp_headers"])
    }
    column_map = {
        tag: scene["column_outputs"][disk_row][column]
        for column, tag in enumerate(scene["column_color_headers"])
    }
    return [
        (tag, stamp)
        for stamp, tag in row_map.items()
        if column_map.get(tag) == stamp
    ]


def sample_scene(seed: int) -> dict:
    """Sample two conditional keys with exactly one reciprocal coordinate pair."""

    seed = int(seed)
    rng = random.Random((seed ^ 0x6D75A11E) & 0xFFFFFFFFFFFFFFFF)

    row_symbols = _shuffled(rng, ROW_SYMBOLS)
    row_stamp_headers = _shuffled(rng, COLUMN_STAMPS)
    column_disks = _shuffled(rng, DISK_TAGS)
    column_color_headers = _shuffled(rng, ROW_TAGS)

    query_symbol = ROW_SYMBOLS[(seed // 4 + seed // 29) % 4]
    query_disk = DISK_TAGS[(seed // 16 + seed // 7) % 4]
    target_stamp = COLUMN_STAMPS[(seed // 3 + seed // 23) % 4]

    selected_tags = _shuffled(rng, ROW_TAGS)
    row_function = dict(zip(COLUMN_STAMPS, selected_tags, strict=True))

    other_stamps = [stamp for stamp in COLUMN_STAMPS if stamp != target_stamp]
    rng.shuffle(other_stamps)
    composed = {target_stamp: target_stamp}
    for index, stamp in enumerate(other_stamps):
        composed[stamp] = other_stamps[(index + 1) % 3]
    column_function = {
        row_function[stamp]: composed[stamp]
        for stamp in COLUMN_STAMPS
    }

    row_outputs: list[list[str]] = []
    for symbol in row_symbols:
        function = row_function if symbol == query_symbol else dict(
            zip(COLUMN_STAMPS, _shuffled(rng, ROW_TAGS), strict=True)
        )
        row_outputs.append([function[stamp] for stamp in row_stamp_headers])

    column_outputs: list[list[str]] = []
    for disk in column_disks:
        function = column_function if disk == query_disk else dict(
            zip(ROW_TAGS, _shuffled(rng, COLUMN_STAMPS), strict=True)
        )
        column_outputs.append([function[tag] for tag in column_color_headers])

    target_tag = row_function[target_stamp]
    board_rows = _shuffled(rng, ROW_TAGS)
    board_columns = _shuffled(rng, COLUMN_STAMPS)
    target_row = board_rows.index(target_tag)
    target_column = board_columns.index(target_stamp)

    palette = _shuffled(rng, ANSWERS)
    desired = ANSWERS[seed % 4]
    offset = (palette.index(desired) - target_row - target_column) % 4
    cells = [
        [palette[(row + column + offset) % 4] for column in range(4)]
        for row in range(4)
    ]

    scene = {
        "row_symbols": row_symbols,
        "row_stamp_headers": row_stamp_headers,
        "row_outputs": row_outputs,
        "column_disks": column_disks,
        "column_color_headers": column_color_headers,
        "column_outputs": column_outputs,
        "board_rows": board_rows,
        "board_columns": board_columns,
        "cells": cells,
        "query_symbol": query_symbol,
        "query_disk": query_disk,
    }
    _validate_scene(scene)
    return scene


def _draw_mosaic(
    draw: ImageDraw.ImageDraw,
    center_x: int,
    center_y: int,
    bits: str,
    ink: tuple[int, int, int],
) -> None:
    draw.rectangle(
        (center_x - 23, center_y - 23, center_x + 23, center_y + 23),
        fill=RGB["stamp_bg"],
        outline=RGB["guide"],
        width=2,
    )
    start_x = center_x - 17
    start_y = center_y - 17
    for row in range(4):
        for column in range(4):
            if bits[row * 4 + column] == "1":
                left = start_x + column * 9
                top = start_y + row * 9
                draw.rectangle((left, top, left + 7, top + 7), fill=ink)


def _draw_color_bar(draw: ImageDraw.ImageDraw, center_x: int, center_y: int, name: str) -> None:
    draw.rectangle(
        (center_x - 24, center_y - 15, center_x + 24, center_y + 15),
        fill=RGB[name],
        outline=RGB["ink"],
        width=2,
    )


def _validate_scene(scene: dict) -> None:
    expected = {
        "row_symbols",
        "row_stamp_headers",
        "row_outputs",
        "column_disks",
        "column_color_headers",
        "column_outputs",
        "board_rows",
        "board_columns",
        "cells",
        "query_symbol",
        "query_disk",
    }
    if set(scene) != expected:
        raise ValueError("scene fields do not match the world schema")
    if list(sorted(map(str, scene["row_symbols"]))) != sorted(ROW_SYMBOLS):
        raise ValueError("ROW KEY symbols must be a permutation")
    if list(sorted(map(str, scene["row_stamp_headers"]))) != sorted(COLUMN_STAMPS):
        raise ValueError("ROW KEY stamp headers must be a permutation")
    if list(sorted(map(str, scene["column_disks"]))) != sorted(DISK_TAGS):
        raise ValueError("COLUMN KEY disks must be a permutation")
    if list(sorted(map(str, scene["column_color_headers"]))) != sorted(ROW_TAGS):
        raise ValueError("COLUMN KEY color headers must be a permutation")
    row_outputs = scene["row_outputs"]
    column_outputs = scene["column_outputs"]
    if not isinstance(row_outputs, list) or len(row_outputs) != 4:
        raise ValueError("ROW KEY must have four output rows")
    if any(not isinstance(row, list) or list(sorted(map(str, row))) != sorted(ROW_TAGS) for row in row_outputs):
        raise ValueError("each ROW KEY output row must be a color bijection")
    if not isinstance(column_outputs, list) or len(column_outputs) != 4:
        raise ValueError("COLUMN KEY must have four output rows")
    if any(not isinstance(row, list) or list(sorted(map(str, row))) != sorted(COLUMN_STAMPS) for row in column_outputs):
        raise ValueError("each COLUMN KEY output row must be a stamp bijection")
    if list(sorted(map(str, scene["board_rows"]))) != sorted(ROW_TAGS):
        raise ValueError("board row headers must be a permutation")
    if list(sorted(map(str, scene["board_columns"]))) != sorted(COLUMN_STAMPS):
        raise ValueError("board column headers must be a permutation")
    cells = scene["cells"]
    if not isinstance(cells, list) or len(cells) != 4:
        raise ValueError("board must have four rows")
    if any(not isinstance(row, list) or list(sorted(map(str, row))) != sorted(ANSWERS) for row in cells):
        raise ValueError("each board row must contain all four answer colors")
    if str(scene["query_symbol"]) not in ROW_SYMBOLS:
        raise ValueError("unknown query symbol")
    if str(scene["query_disk"]) not in DISK_TAGS:
        raise ValueError("unknown query disk")
    if len(_agreement_pairs(scene)) != 1:
        raise ValueError("selected keys must have exactly one reciprocal pair")


def _draw_table_grid(draw: ImageDraw.ImageDraw, x: int) -> None:
    for row in range(4):
        for column in range(4):
            left = x + column * KEY_CELL_W
            top = KEY_TABLE_Y + row * KEY_CELL_H
            draw.rectangle(
                (left, top, left + KEY_CELL_W, top + KEY_CELL_H),
                fill=RGB["stamp_bg"],
                outline=RGB["guide"],
                width=2,
            )


def render(scene: dict) -> Image.Image:
    """Render two broad conditional tables, the query, and the answer grid."""

    _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["background"])
    draw = ImageDraw.Draw(image)
    title = _font(19, bold=True)
    label = _font(15, bold=True)
    body = _font(17, bold=False)

    draw.rounded_rectangle((18, 26, 500, 338), radius=11, fill=RGB["panel"], outline=RGB["panel_edge"], width=3)
    draw.text((34, 38), "ROW KEY:  BLUE STAMP -> ROW COLOR", fill=RGB["ink"], font=title)
    _draw_table_grid(draw, ROW_TABLE_X)
    for column, stamp in enumerate(scene["row_stamp_headers"]):
        _draw_mosaic(draw, ROW_KEY_CENTERS_X[column], KEY_HEADER_Y, str(stamp), RGB["stamp_ink"])
    for row, symbol in enumerate(scene["row_symbols"]):
        center_y = KEY_ROW_CENTERS[row]
        _draw_mosaic(draw, ROW_LABEL_X, center_y, str(symbol), RGB["ink"])
        for column, tag in enumerate(scene["row_outputs"][row]):
            _draw_color_bar(draw, ROW_KEY_CENTERS_X[column], center_y, str(tag))

    draw.rounded_rectangle((538, 26, 1022, 338), radius=11, fill=RGB["panel"], outline=RGB["panel_edge"], width=3)
    draw.text((554, 38), "COLUMN KEY:  ROW COLOR -> BLUE STAMP", fill=RGB["ink"], font=title)
    _draw_table_grid(draw, COLUMN_TABLE_X)
    for column, tag in enumerate(scene["column_color_headers"]):
        _draw_color_bar(draw, COLUMN_KEY_CENTERS_X[column], KEY_HEADER_Y, str(tag))
    for row, disk in enumerate(scene["column_disks"]):
        center_y = KEY_ROW_CENTERS[row]
        draw.ellipse((COLUMN_LABEL_X - 20, center_y - 20, COLUMN_LABEL_X + 20, center_y + 20), fill=RGB[str(disk)], outline=RGB["ink"], width=3)
        for column, stamp in enumerate(scene["column_outputs"][row]):
            _draw_mosaic(draw, COLUMN_KEY_CENTERS_X[column], center_y, str(stamp), RGB["stamp_ink"])

    draw.rounded_rectangle((28, 376, 604, 706), radius=12, fill=RGB["panel"], outline=RGB["panel_edge"], width=3)
    draw.text((48, 392), "QUERY: use these two table rows", fill=RGB["ink"], font=title)
    draw.text((58, 438), "ROW KEY symbol", fill=RGB["ink"], font=label)
    _draw_mosaic(draw, 145, 505, str(scene["query_symbol"]), RGB["ink"])
    draw.text((294, 438), "COLUMN KEY disk", fill=RGB["ink"], font=label)
    draw.ellipse((350, 482, 396, 528), fill=RGB[str(scene["query_disk"])], outline=RGB["ink"], width=3)
    draw.text((230, 482), "+", fill=RGB["accent"], font=_font(34, bold=True))
    draw.text((55, 574), "Find the ONLY row color + blue stamp", fill=RGB["ink"], font=body)
    draw.text((55, 602), "that point to each other in both selected rows.", fill=RGB["ink"], font=body)
    draw.text((55, 650), "Then use that pair on the answer grid.", fill=RGB["ink"], font=body)

    draw.text((660, 343), "ANSWER GRID", fill=RGB["ink"], font=title)
    for column, stamp in enumerate(scene["board_columns"]):
        _draw_mosaic(draw, BOARD_CENTERS_X[column], 392, str(stamp), RGB["stamp_ink"])
    for row, tag in enumerate(scene["board_rows"]):
        _draw_color_bar(draw, 625, BOARD_CENTERS_Y[row], str(tag))
    for row in range(4):
        for column in range(4):
            left = GRID_X + column * GRID_CELL
            top = GRID_Y + row * GRID_CELL
            draw.rectangle((left, top, left + GRID_CELL - 1, top + GRID_CELL - 1), fill=RGB["ink"])
            draw.rectangle((left + 5, top + 5, left + GRID_CELL - 6, top + GRID_CELL - 6), fill=RGB[str(scene["cells"][row][column])])
    return image


def analytic_gold(scene: dict) -> str:
    """Solve reciprocal agreement, then bind the pair to board axes."""

    _validate_scene(scene)
    tag, stamp = _agreement_pairs(scene)[0]
    row = [str(value) for value in scene["board_rows"]].index(str(tag))
    column = [str(value) for value in scene["board_columns"]].index(str(stamp))
    return str(scene["cells"][row][column])


def _hamming(left: str, right: str) -> int:
    return sum(a != b for a, b in zip(left, right, strict=True))


def margin(scene: dict) -> float:
    """Return the smallest broad mark separation used by the inverse arm."""

    _validate_scene(scene)
    row_gap = min(_hamming(a, b) for index, a in enumerate(ROW_SYMBOLS) for b in ROW_SYMBOLS[index + 1 :])
    stamp_gap = min(_hamming(a, b) for index, a in enumerate(COLUMN_STAMPS) for b in COLUMN_STAMPS[index + 1 :])
    color_gap = min(
        math.dist(RGB[a], RGB[b]) / 20.0
        for names in (ROW_TAGS, DISK_TAGS, ANSWERS)
        for index, a in enumerate(names)
        for b in names[index + 1 :]
    )
    return float(min(row_gap, stamp_gap, color_gap))


def is_quarantined(scene: dict) -> bool:
    try:
        _validate_scene(scene)
        analytic_gold(scene)
        return margin(scene) < 3.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The ordered, fully rendered worksheet has no intended latent aliases."""

    return []
