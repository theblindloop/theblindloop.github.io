"""Independent image-only inverse for red-cable endpoint attribute binding."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster lacks a complete supported cable-to-terminal construction."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _largest_centroid(mask: np.ndarray, minimum: int) -> tuple[float, float] | None:
    pieces = [piece for piece in _components(mask) if len(piece) >= minimum]
    if not pieces:
        return None
    piece = max(pieces, key=len)
    return float(piece[:, 1].mean()), float(piece[:, 0].mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape != (512, 512, 3):
        raise OracleAbstention("expected a 512 by 512 RGB raster")
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    cable_mask = (red >= 155) & (red - green >= 85) & (red - blue >= 65)
    cable_pieces = _components(cable_mask)
    substantial = [piece for piece in cable_pieces if len(piece) >= 35]
    speckle = sum(len(piece) for piece in cable_pieces if len(piece) < 35)
    if speckle > 30 or len(substantial) != 1:
        raise OracleAbstention("red cable is fragmented or contaminated")
    cable = substantial[0]
    ys, xs = cable[:, 0], cable[:, 1]
    area = len(cable)
    span_x = int(xs.max() - xs.min() + 1)
    span_y = int(ys.max() - ys.min() + 1)
    if not 1900 <= area <= 4400:
        raise OracleAbstention(f"unsupported cable ink area {area}")
    if span_x < 375 or int(xs.min()) > 75 or int(xs.max()) < 450:
        raise OracleAbstention("cable does not span from the left plug to the battery")
    if not 65 <= span_y <= 185 or span_x / span_y < 2.0:
        raise OracleAbstention(f"cable is not one thin distributed route ({span_x}x{span_y})")

    green_mask = (green >= 115) & (green - red >= 45) & (green - blue >= 18)
    blue_mask = (blue >= 145) & (blue - red >= 45) & (blue - green >= 22)
    markers = {
        "plus": _largest_centroid(green_mask, 280),
        "minus": _largest_centroid(blue_mask, 280),
    }
    tip = cable[xs >= int(xs.max()) - 12]
    tip_xy = (float(tip[:, 1].mean()), float(tip[:, 0].mean()))
    distances = {}
    for label, center in markers.items():
        if center is not None:
            distances[label] = float(np.hypot(tip_xy[0] - center[0], tip_xy[1] - center[1]))
    if not distances:
        raise OracleAbstention("no visible labeled terminal near the cable endpoint")
    ordered = sorted(distances.items(), key=lambda item: item[1])
    label, best = ordered[0]
    runner_up = ordered[1][1] if len(ordered) > 1 else float("inf")
    if best > 23.0 or runner_up - best < 70.0:
        raise OracleAbstention("cable endpoint is not unambiguously bound to one terminal")
    return label
