"""Deterministic Pillow renderer for a red cable reaching a battery terminal."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
SUPERSAMPLE = 4
TERMINAL_RADIUS = 26
TOP_Y = 181
BOTTOM_Y = 331
START_X = 69
START_Y = 256
TERMINAL_X = 455

CABLE_REDS = ((210, 40, 58), (194, 36, 52), (226, 54, 49))
BACKGROUNDS = ((248, 247, 243), (244, 248, 247), (248, 245, 249))
BATTERY_TONES = ((218, 221, 220), (211, 217, 219), (220, 215, 221))
PLUS_GREEN = (34, 164, 88)
MINUS_BLUE = (39, 105, 210)
DARK = (54, 58, 61)


def sample_scene(seed: int) -> dict:
    """Adjacent seeds are exactly style-matched plus/minus counterfactuals."""

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0xCA61E7E2)
    return {
        "answer_label": "plus" if seed % 2 == 0 else "minus",
        "background_code": rng.randrange(len(BACKGROUNDS)),
        "cable_red_code": rng.randrange(len(CABLE_REDS)),
        "cable_width": 7 + rng.randrange(2),
        "canvas_size": CANVAS_SIZE,
        "decor_variant": rng.randrange(12),
        "offset_x": rng.randrange(-6, 7),
        "offset_y": rng.randrange(-9, 10),
        "path_variant": rng.randrange(8),
        "terminal_order": "plus_top" if rng.randrange(2) == 0 else "minus_top",
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def _terminal_positions(scene: dict) -> dict[str, tuple[float, float]]:
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    top = (TERMINAL_X + ox, TOP_Y + oy)
    bottom = (TERMINAL_X + ox, BOTTOM_Y + oy)
    if scene["terminal_order"] == "plus_top":
        return {"plus": top, "minus": bottom}
    return {"plus": bottom, "minus": top}


def _cable_points(scene: dict) -> list[tuple[float, float]]:
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    start_x, start_y = START_X + ox, START_Y + oy
    target_x, target_y = _terminal_positions(scene)[str(scene["answer_label"])]
    side = -1.0 if target_y < start_y else 1.0
    variant = int(scene["path_variant"])
    amplitude = 29.0 + 3.0 * (variant % 4)
    cycles = 1.15 + 0.10 * (variant // 4)
    phase = (variant % 4) * math.pi / 7.0
    points = []
    for index in range(241):
        t = index / 240.0
        smooth = t * t * (3.0 - 2.0 * t)
        wave = amplitude * math.sin(2.0 * math.pi * cycles * t + phase) * math.sin(math.pi * t)
        x = start_x + (target_x - start_x) * t
        y = start_y + side * (abs(target_y - start_y) * smooth + wave)
        points.append((x, y))
    return points


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    if size != CANVAS_SIZE:
        raise ValueError("this world renders a fixed 512px canvas")
    bg_code = int(scene["background_code"]) % len(BACKGROUNDS)
    background = BACKGROUNDS[bg_code]
    battery = BATTERY_TONES[bg_code]
    red = CABLE_REDS[int(scene["cable_red_code"]) % len(CABLE_REDS)]
    width = int(scene["cable_width"])
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    decor = int(scene["decor_variant"]) % 12
    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Low-chroma remote marks vary independently and cannot be cable or terminal cues.
    faint = tuple(max(0, channel - 22) for channel in background)
    for index in range(4):
        x = 104 + ((decor * 41 + index * 83) % 250)
        y = 42 if (decor + index) % 2 == 0 else 470
        draw.ellipse(_scaled_box((x - 4, y - 4, x + 4, y + 4), ss), fill=faint)

    # Plug at left and battery housing at right establish the familiar visual context.
    draw.rounded_rectangle(
        _scaled_box((24 + ox, 232 + oy, 78 + ox, 280 + oy), ss),
        radius=7 * ss,
        fill=DARK,
    )
    for pin_y in (243 + oy, 269 + oy):
        draw.rectangle(_scaled_box((14 + ox, pin_y - 3, 26 + ox, pin_y + 3), ss), fill=(105, 108, 109))
    draw.rounded_rectangle(
        _scaled_box((420 + ox, 126 + oy, 510 + ox, 386 + oy), ss),
        radius=18 * ss,
        fill=battery,
        outline=DARK,
        width=3 * ss,
    )

    positions = _terminal_positions(scene)
    for label, color in (("plus", PLUS_GREEN), ("minus", MINUS_BLUE)):
        cx, cy = positions[label]
        draw.ellipse(
            _scaled_box((cx - TERMINAL_RADIUS, cy - TERMINAL_RADIUS, cx + TERMINAL_RADIUS, cy + TERMINAL_RADIUS), ss),
            fill=color,
            outline=DARK,
            width=2 * ss,
        )
        # Explicit visible +/− glyph beside each color-coded terminal.
        gx = 493 + ox
        draw.line([(round((gx - 9) * ss), round(cy * ss)), (round((gx + 9) * ss), round(cy * ss))], fill=DARK, width=4 * ss)
        if label == "plus":
            draw.line([(round(gx * ss), round((cy - 9) * ss)), (round(gx * ss), round((cy + 9) * ss))], fill=DARK, width=4 * ss)

    points = [(round(x * ss), round(y * ss)) for x, y in _cable_points(scene)]
    draw.line(points, fill=red, width=width * ss, joint="curve")
    radius = width * ss / 2.0
    for x, y in (points[0], points[-1]):
        draw.ellipse((round(x - radius), round(y - radius), round(x + radius), round(y + radius)), fill=red)

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    answer = str(scene["answer_label"])
    if answer not in ("plus", "minus"):
        raise ValueError("answer_label must be plus or minus")
    return answer


def margin(scene: dict) -> float:
    positions = _terminal_positions(scene)
    separation = abs(positions["plus"][1] - positions["minus"][1])
    return float(separation - 2 * TERMINAL_RADIUS)


def is_quarantined(scene: dict) -> bool:
    clearance = TERMINAL_RADIUS - float(scene["cable_width"]) / 2.0
    return margin(scene) <= 18.0 or clearance <= 18.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["cable_red_code"] = int(twin["cable_red_code"]) + len(CABLE_REDS)
    return [("one_full_red_palette_cycle", twin)]
