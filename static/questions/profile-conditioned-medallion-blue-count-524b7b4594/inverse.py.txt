"""Independent raster inverse for the blue-planet medallion question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public palette duplicated deliberately.  The image path is the only input;
# this module has no import or data path to the renderer or scene generator.
BLUE_SWATCHES = np.asarray(
    ((38, 119, 214), (47, 137, 204), (52, 104, 190)), dtype=np.float32
)
NONBLUE_SWATCHES = np.asarray(
    ((207, 73, 73), (219, 151, 36), (151, 73, 168), (43, 154, 112), (212, 103, 48)),
    dtype=np.float32,
)
PLANET_SWATCHES = np.concatenate((BLUE_SWATCHES, NONBLUE_SWATCHES), axis=0)


def _components(mask: np.ndarray, minimum: int) -> list[int]:
    """Return areas of 8-connected pixel components above a size floor."""

    height, width = mask.shape
    remaining = mask.copy()
    areas: list[int] = []
    while remaining.any():
        y0, x0 = np.argwhere(remaining)[0]
        queue = deque([(int(y0), int(x0))])
        remaining[y0, x0] = False
        area = 0
        while queue:
            y, x = queue.popleft()
            area += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    yy = y + dy
                    xx = x + dx
                    if 0 <= yy < height and 0 <= xx < width and remaining[yy, xx]:
                        remaining[yy, xx] = False
                        queue.append((yy, xx))
        if area >= minimum:
            areas.append(area)
    return areas


def decision_from_image(image: Image.Image) -> str:
    """Recover blue cardinality from swatch components in the final PNG only."""

    pixels = np.asarray(image.convert("RGB"), dtype=np.float32)
    all_distances = np.linalg.norm(
        pixels[:, :, None, :] - PLANET_SWATCHES[None, None, :, :], axis=-1
    )
    blue_distances = np.linalg.norm(
        pixels[:, :, None, :] - BLUE_SWATCHES[None, None, :, :], axis=-1
    )
    # Interior pixels are exact swatches.  The threshold retains interiors and
    # a small antialiased rim, while excluding neutral frames and text.
    all_mask = all_distances.min(axis=-1) <= 44.0
    blue_mask = blue_distances.min(axis=-1) <= 44.0
    # A planetary disk remains comfortably above 1,000 swatch pixels even in
    # the quarantined small-radius scenes.  The decorative accent rule is only
    # a few hundred pixels and is intentionally rejected by this floor.
    all_components = _components(all_mask, minimum=1000)
    blue_components = _components(blue_mask, minimum=1000)

    # Six colored medallions are part of the public visual contract.  Requiring
    # them makes evidence erasure (for example a blank image) abstain instead
    # of silently becoming a plausible zero-blue answer.
    if len(all_components) != 6:
        raise ValueError("six stable planet medallions were not recovered")
    if len(blue_components) > 6:
        raise ValueError("blue component segmentation is ambiguous")
    return str(len(blue_components))
