"""Deterministic renderer for the visible blue-planet medallion question."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 600
SUPERSAMPLE = 4

# The target palette is deliberately limited to saturated blue swatches.  The
# other swatches are far from blue and are used only for visible distractor
# planets.  These constants are duplicated by oracle.py rather than imported.
BLUE_SWATCHES = (
    (38, 119, 214),
    (47, 137, 204),
    (52, 104, 190),
)
NONBLUE_SWATCHES = (
    (207, 73, 73),
    (219, 151, 36),
    (151, 73, 168),
    (43, 154, 112),
    (212, 103, 48),
)

STYLE_PALETTES = (
    {
        "background": (248, 250, 247),
        "card_fill": (255, 255, 252),
        "card_outline": (194, 201, 204),
        "ring_color": (125, 139, 145),
        "header_color": (46, 61, 70),
        "accent_color": (210, 126, 55),
    },
    {
        "background": (247, 249, 252),
        "card_fill": (253, 254, 255),
        "card_outline": (183, 196, 207),
        "ring_color": (116, 133, 148),
        "header_color": (43, 59, 76),
        "accent_color": (188, 113, 56),
    },
    {
        "background": (251, 248, 245),
        "card_fill": (255, 253, 249),
        "card_outline": (204, 194, 186),
        "ring_color": (137, 126, 118),
        "header_color": (71, 57, 49),
        "accent_color": (65, 139, 126),
    },
)

SLOT_CENTERS = (
    (170.0, 195.0),
    (480.0, 195.0),
    (790.0, 195.0),
    (170.0, 440.0),
    (480.0, 440.0),
    (790.0, 440.0),
)
SLOT_LABELS = ("A-1", "A-2", "A-3", "B-1", "B-2", "B-3")
CARD_WIDTH = 270.0
CARD_HEIGHT = 190.0


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _is_blue_fill(fill: list[int] | tuple[int, int, int]) -> bool:
    return tuple(int(value) for value in fill) in BLUE_SWATCHES


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample six framed medallions with answer-independent geometry."""

    rng = random.Random(int(seed))
    style = STYLE_PALETTES[rng.randrange(len(STYLE_PALETTES))]
    # Keep total object count and total disk radius fixed within a scene.  The
    # role permutation and palette choices carry the answer, not geometry.
    blue_count = rng.randint(1, 5)
    roles = [True] * blue_count + [False] * (len(SLOT_CENTERS) - blue_count)
    rng.shuffle(roles)
    radius = rng.randint(25, 29) if rng.random() < 0.14 else rng.randint(38, 48)

    planets: list[dict[str, Any]] = []
    for index, ((base_x, base_y), is_blue) in enumerate(zip(SLOT_CENTERS, roles)):
        fill = rng.choice(BLUE_SWATCHES if is_blue else NONBLUE_SWATCHES)
        planets.append(
            {
                "x": round(base_x + rng.uniform(-10.0, 10.0), 3),
                "y": round(base_y + rng.uniform(-7.0, 7.0), 3),
                "fill": list(fill),
                "label": SLOT_LABELS[index],
            }
        )

    return {
        "background": list(style["background"]),
        "card_fill": list(style["card_fill"]),
        "card_outline": list(style["card_outline"]),
        "ring_color": list(style["ring_color"]),
        "header_color": list(style["header_color"]),
        "accent_color": list(style["accent_color"]),
        "radius": int(radius),
        "header_text": "PLANETARY ARCHIVE",
        "planets": planets,
    }


def counterfactual_scene(scene: dict[str, Any]) -> dict[str, Any]:
    """Make a matched counterfactual by recoloring one blue medallion."""

    twin = copy.deepcopy(scene)
    for index, planet in enumerate(twin["planets"]):
        if _is_blue_fill(planet["fill"]):
            planet["fill"] = list(NONBLUE_SWATCHES[index % len(NONBLUE_SWATCHES)])
            break
    return twin


def _box(left: float, top: float, right: float, bottom: float) -> tuple[int, int, int, int]:
    return (
        round(left * SUPERSAMPLE),
        round(top * SUPERSAMPLE),
        round(right * SUPERSAMPLE),
        round(bottom * SUPERSAMPLE),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one RGB page of six visible planetary medallions."""

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    header_font = _font(28 * ss)
    small_font = _font(15 * ss)
    tiny_font = _font(12 * ss)

    draw.text(
        (WIDTH * ss // 2, 34 * ss),
        str(scene["header_text"]),
        fill=tuple(scene["header_color"]),
        anchor="mm",
        font=header_font,
    )
    draw.line(
        (390 * ss, 59 * ss, 570 * ss, 59 * ss),
        fill=tuple(scene["accent_color"]),
        width=3 * ss,
    )

    radius = float(scene["radius"])
    for planet in scene["planets"]:
        cx = float(planet["x"])
        cy = float(planet["y"])
        left = cx - CARD_WIDTH / 2.0
        top = cy - CARD_HEIGHT / 2.0
        right = cx + CARD_WIDTH / 2.0
        bottom = cy + CARD_HEIGHT / 2.0

        draw.rounded_rectangle(
            _box(left + 5.0, top + 6.0, right + 5.0, bottom + 6.0),
            radius=15 * ss,
            fill=(222, 226, 226),
        )
        draw.rounded_rectangle(
            _box(left, top, right, bottom),
            radius=15 * ss,
            fill=tuple(scene["card_fill"]),
            outline=tuple(scene["card_outline"]),
            width=2 * ss,
        )

        draw.text(
            ((left + 18.0) * ss, (top + 19.0) * ss),
            str(planet["label"]),
            fill=tuple(scene["header_color"]),
            font=small_font,
        )
        ring_radius = radius + 13.0
        draw.ellipse(
            _box(cx - ring_radius, cy - ring_radius, cx + ring_radius, cy + ring_radius),
            outline=tuple(scene["ring_color"]),
            width=3 * ss,
        )
        draw.ellipse(
            _box(cx - radius, cy - radius, cx + radius, cy + radius),
            fill=tuple(planet["fill"]),
        )

        # Every medallion has the same neutral highlight.  It adds a visible
        # object cue without creating a second answer-bearing color class.
        highlight_radius = max(4.0, radius * 0.16)
        highlight_x = cx - radius * 0.30
        highlight_y = cy - radius * 0.31
        draw.ellipse(
            _box(
                highlight_x - highlight_radius,
                highlight_y - highlight_radius,
                highlight_x + highlight_radius,
                highlight_y + highlight_radius,
            ),
            fill=(239, 244, 244),
        )
        draw.text(
            (cx * ss, (bottom - 18.0) * ss),
            "VISIBLE SPECIMEN",
            fill=tuple(scene["ring_color"]),
            anchor="mm",
            font=tiny_font,
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Count the visible disk fills that belong to the public blue palette."""

    return str(sum(_is_blue_fill(planet["fill"]) for planet in scene["planets"]))


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest disk-interior raster headroom in pixels."""

    # The neutral ring is 13 px outside the disk and the highlight leaves a
    # connected blue annulus.  A 16 px headroom is a conservative boundary.
    return round(float(scene["radius"]) - 16.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent pixel-identical aliases are claimed for this catalog."""

    return []
