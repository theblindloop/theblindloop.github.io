"""Independent PNG-only inverse arm for portal_region_fusion."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((218, 55, 65), dtype=np.int16),
    "green": np.array((34, 158, 91), dtype=np.int16),
    "blue": np.array((48, 103, 207), dtype=np.int16),
}


def _components(mask: np.ndarray) -> int:
    """Four-connected components via row runs and union-find."""
    parent: list[int] = []

    def new_label() -> int:
        label = len(parent)
        parent.append(label)
        return label

    def find(label: int) -> int:
        while parent[label] != label:
            parent[label] = parent[parent[label]]
            label = parent[label]
        return label

    def union(a: int, b: int) -> None:
        a, b = find(a), find(b)
        if a != b:
            parent[b] = a

    previous: list[tuple[int, int, int]] = []
    for row in mask:
        padded = np.pad(row.astype(np.int8), (1, 1))
        changes = np.diff(padded)
        starts = np.flatnonzero(changes == 1)
        ends = np.flatnonzero(changes == -1) - 1
        current: list[tuple[int, int, int]] = []
        for start, end in zip(starts.tolist(), ends.tolist()):
            overlaps = [label for pstart, pend, label in previous if pstart <= end and start <= pend]
            label = overlaps[0] if overlaps else new_label()
            for other in overlaps[1:]:
                union(label, other)
            current.append((start, end, label))
        previous = current
    return len({find(label) for label in range(len(parent))})


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    floor = np.all(pixels >= 252, axis=2)
    before = _components(floor)
    winners: list[str] = []
    for name, rgb in PALETTE.items():
        distance = np.max(np.abs(pixels - rgb), axis=2)
        gate = distance <= 3
        area = int(gate.sum())
        if not 900 <= area <= 1400:
            raise ValueError(f"{name} gate area is invalid: {area}")
        opened = floor | gate
        merged_regions = before - _components(opened) + 1
        if merged_regions == 3:
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"expected one three-region gate, got {winners}")
    return winners[0]
