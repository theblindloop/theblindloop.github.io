"""Deterministic raster world for a region-adjacency contraction question."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 300
BACKGROUND = (236, 240, 245)
INK = (24, 28, 34)
FLOOR = (255, 255, 255)
PALETTE = {
    "red": (218, 55, 65),
    "green": (34, 158, 91),
    "blue": (48, 103, 207),
}
COLORS = ("red", "green", "blue")
ARMS = ("N", "E", "S", "W")


def _region_count(open_arms: Iterable[str]) -> int:
    """Count four quadrant regions after the named wall-arm gaps are opened."""
    parent = list(range(4))  # NW, NE, SE, SW

    def find(x: int) -> int:
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a: int, b: int) -> None:
        a, b = find(a), find(b)
        if a != b:
            parent[b] = a

    adjacent = {"N": (0, 1), "E": (1, 2), "S": (2, 3), "W": (3, 0)}
    for arm in open_arms:
        union(*adjacent[arm])
    return len({find(i) for i in range(4)})


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 3
    target = COLORS[answer_index]
    distractors = [color for color in COLORS if color != target]
    rng.shuffle(distractors)

    # No gap -> 4 regions; one gap -> 3; two non-redundant gaps -> 2.
    one_arm = rng.choice(ARMS)
    start = rng.randrange(4)
    if rng.random() < 0.5:
        two_arms = [ARMS[start], ARMS[(start + 1) % 4]]
    else:
        two_arms = [ARMS[start], ARMS[(start + 2) % 4]]
    layouts = {
        target: [one_arm],
        distractors[0]: [],
        distractors[1]: two_arms,
    }
    panel_order = list(COLORS)
    rng.shuffle(panel_order)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "panel_order": panel_order,
        "colors": list(COLORS),
        "layouts": layouts,
    }


def _draw_panel(draw: ImageDraw.ImageDraw, x0: int, y0: int, color: str, open_arms: list[str]) -> None:
    size = 210
    frame = 10
    cx = x0 + size // 2
    cy = y0 + size // 2
    half_gate = 17
    half_wall = 6

    draw.rectangle((x0, y0, x0 + size - 1, y0 + size - 1), fill=INK)
    draw.rectangle(
        (x0 + frame, y0 + frame, x0 + size - frame - 1, y0 + size - frame - 1),
        fill=FLOOR,
    )

    # Four thick wall arms terminate flush against the colored central gate.
    draw.rectangle((cx - half_wall, y0 + frame, cx + half_wall - 1, cy - half_gate - 1), fill=INK)
    draw.rectangle((cx + half_gate, cy - half_wall, x0 + size - frame - 1, cy + half_wall - 1), fill=INK)
    draw.rectangle((cx - half_wall, cy + half_gate, cx + half_wall - 1, y0 + size - frame - 1), fill=INK)
    draw.rectangle((x0 + frame, cy - half_wall, cx - half_gate - 1, cy + half_wall - 1), fill=INK)

    # A broad, visible gap through an arm pre-connects its adjacent quadrants.
    gap_lo = 48
    gap_hi = 75
    gaps = {
        "N": (cx - half_wall, y0 + gap_lo, cx + half_wall - 1, y0 + gap_hi),
        "E": (x0 + size - gap_hi - 1, cy - half_wall, x0 + size - gap_lo - 1, cy + half_wall - 1),
        "S": (cx - half_wall, y0 + size - gap_hi - 1, cx + half_wall - 1, y0 + size - gap_lo - 1),
        "W": (x0 + gap_lo, cy - half_wall, x0 + gap_hi, cy + half_wall - 1),
    }
    for arm in open_arms:
        draw.rectangle(gaps[arm], fill=FLOOR)

    draw.rectangle(
        (cx - half_gate, cy - half_gate, cx + half_gate - 1, cy + half_gate - 1),
        fill=PALETTE[color],
    )


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    panel_order = scene["panel_order"]
    colors = scene["colors"]
    layouts = scene["layouts"]
    if set(panel_order) != set(colors):
        raise ValueError("panel_order and colors disagree")
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot, color in enumerate(panel_order):
        _draw_panel(draw, 20 + 235 * slot, 45, color, layouts[color])
    return image


def analytic_gold(scene: dict) -> str:
    colors = scene["colors"]
    layouts = scene["layouts"]
    winners = [color for color in colors if _region_count(layouts[color]) == 3]
    if len(winners) != 1:
        raise ValueError(f"expected one three-region portal, got {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    colors = scene["colors"]
    layouts = scene["layouts"]
    distances = sorted(abs(_region_count(layouts[color]) - 3) for color in colors)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        colors = scene["colors"]
        layouts = scene["layouts"]
        counts = sorted(_region_count(layouts[color]) for color in colors)
        return counts != [2, 3, 4] or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
