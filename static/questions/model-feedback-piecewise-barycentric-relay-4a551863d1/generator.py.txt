"""Deterministic two-card relay through visible piecewise-affine triangle charts."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1400, 740
SIZE = 11
START_X, START_Y = 18, 145
CARD_XS = (215, 805)
CARD_FROM_DX, CARD_TO_DX = 18, 372
MAIN_Y, MAIN_CELL, MAIN_GAP = 145, 15, 2
OPTION_XS = (48, 312, 576, 840, 1104)
OPTION_Y, OPTION_CELL, OPTION_GAP = 565, 11, 2

BACKGROUND = "#ffffff"
INK = "#20364a"
PAPER = "#faf8f1"
BORDER = "#8998a3"
PALE = "#dce4e8"
RED = "#c62828"
FAMILY_EDGES = ("#ad6f2e", "#087f78")
VERTEX_COLORS = (
    "#d84315", "#f9a825", "#8e24aa",
    "#00838f", "#1565c0", "#2e7d32",
)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

# Every positive triple is a possible barycentric numerator vector.  Template
# offsets are generated so their weighted sum is zero, making the visible bead
# cell the exact weighted point of all three visible vertices.
WEIGHTS = (
    (2, 1, 1), (1, 2, 1), (1, 1, 2),
    (3, 2, 1), (3, 1, 2), (2, 3, 1),
    (1, 3, 2), (2, 1, 3), (1, 2, 3),
)
CARD_FIELDS = ("from_triangles", "to_triangles")
ANCHOR_PAIRS = (
    ((3, 3), (7, 7)), ((3, 7), (7, 3)),
    ((2, 3), (8, 7)), ((2, 7), (8, 3)),
    ((3, 2), (7, 8)), ((3, 8), (7, 2)),
    ((2, 5), (8, 5)), ((5, 2), (5, 8)),
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _cross(a: tuple[int, int], b: tuple[int, int]) -> int:
    return a[0] * b[1] - a[1] * b[0]


def _det(triangle) -> int:
    a, b, c = triangle
    return _cross((b[0] - a[0], b[1] - a[1]), (c[0] - a[0], c[1] - a[1]))


def _barycentric_numerators(point, triangle) -> tuple[int, int, int, int]:
    a, b, c = triangle
    det = _det(triangle)
    n0 = _cross((b[0] - point[0], b[1] - point[1]),
                (c[0] - point[0], c[1] - point[1]))
    n1 = _cross((c[0] - point[0], c[1] - point[1]),
                (a[0] - point[0], a[1] - point[1]))
    n2 = _cross((a[0] - point[0], a[1] - point[1]),
                (b[0] - point[0], b[1] - point[1]))
    return n0, n1, n2, det


def _strictly_inside(point, triangle) -> bool:
    n0, n1, n2, det = _barycentric_numerators(point, triangle)
    return det != 0 and all(n * det > 0 for n in (n0, n1, n2))


def _map_point(point, source, target) -> tuple[int, int]:
    n0, n1, n2, det = _barycentric_numerators(point, source)
    if det == 0 or not all(n * det > 0 for n in (n0, n1, n2)):
        raise ValueError("point is not strictly inside its source triangle")
    raw_row = n0 * target[0][0] + n1 * target[1][0] + n2 * target[2][0]
    raw_col = n0 * target[0][1] + n1 * target[1][1] + n2 * target[2][1]
    if raw_row % det or raw_col % det:
        raise ValueError("affine image is not an exact grid cell")
    return raw_row // det, raw_col // det


def _point(raw: object) -> tuple[int, int]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a cell must be a two-integer list")
    row, col = int(raw[0]), int(raw[1])
    if [row, col] != raw or not (0 <= row < SIZE and 0 <= col < SIZE):
        raise ValueError("cell coordinate is invalid")
    return row, col


def _token(raw: object) -> tuple[tuple[int, int], tuple[int, int]]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a state needs navy then red bead")
    token = _point(raw[0]), _point(raw[1])
    if token[0] == token[1]:
        raise ValueError("the two beads must occupy different cells")
    return token


def _triangle(raw: object) -> tuple[tuple[int, int], ...]:
    if not isinstance(raw, list) or len(raw) != 3:
        raise ValueError("a triangle needs three ordered corners")
    triangle = tuple(_point(item) for item in raw)
    if len(set(triangle)) != 3 or abs(_det(triangle)) < 4:
        raise ValueError("triangle corners must be distinct and well separated")
    return triangle


def _triangle_pair(raw: object):
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("each panel needs two triangles")
    triangles = tuple(_triangle(item) for item in raw)
    if len(set(itertools.chain.from_iterable(triangles))) != 6:
        raise ValueError("the six corner markers in a panel must occupy distinct cells")
    return triangles


def _card(raw: object):
    if not isinstance(raw, dict) or set(raw) != set(CARD_FIELDS):
        raise ValueError("a card needs FROM and TO triangle panels")
    return tuple(_triangle_pair(raw[name]) for name in CARD_FIELDS)


def _parse_scene(scene: dict):
    if not isinstance(scene, dict) or set(scene) != {"start_points", "cards", "options"}:
        raise ValueError("scene fields must be start_points, cards, and options")
    start = _token(scene["start_points"])
    if not isinstance(scene["cards"], list) or len(scene["cards"]) != 2:
        raise ValueError("exactly two cards are required")
    cards = tuple(_card(raw) for raw in scene["cards"])
    if not isinstance(scene["options"], list) or len(scene["options"]) != 5:
        raise ValueError("exactly five options are required")
    options = tuple(_token(raw) for raw in scene["options"])
    if len(set(options)) != 5:
        raise ValueError("options must be distinct")
    return start, cards, options


def _apply_card(state, card):
    sources, targets = card
    output = []
    for point in state:
        hits = [index for index, triangle in enumerate(sources) if _strictly_inside(point, triangle)]
        if len(hits) != 1:
            raise ValueError("each bead must be strictly inside exactly one FROM triangle")
        mapped = _map_point(point, sources[hits[0]], targets[hits[0]])
        if not (0 <= mapped[0] < SIZE and 0 <= mapped[1] < SIZE):
            raise ValueError("mapped bead leaves the board")
        output.append(mapped)
    if output[0] == output[1]:
        raise ValueError("a card may not merge the beads")
    return tuple(output)


def _evolve(state, cards):
    current = state
    for card in cards:
        current = _apply_card(current, card)
    return current


def _shape_signature(offsets) -> tuple[int, int, int, int]:
    distances = []
    for a, b in ((0, 1), (1, 2), (2, 0)):
        dr = offsets[a][0] - offsets[b][0]
        dc = offsets[a][1] - offsets[b][1]
        distances.append(dr * dr + dc * dc)
    return (*sorted(distances), abs(_det(offsets)))


def _build_template_pools():
    offsets = tuple((row, col) for row in range(-2, 3) for col in range(-2, 3)
                    if (row, col) != (0, 0))
    pools = {}
    for weights in WEIGHTS:
        choices = []
        for triangle in itertools.permutations(offsets, 3):
            if (sum(weights[i] * triangle[i][0] for i in range(3)) != 0 or
                    sum(weights[i] * triangle[i][1] for i in range(3)) != 0):
                continue
            if len(set(triangle)) != 3 or abs(_det(triangle)) < 4:
                continue
            choices.append(triangle)
        if not choices:
            raise RuntimeError(f"no offset templates for {weights}")
        pools[weights] = tuple(choices)
    return pools


TEMPLATE_POOLS = _build_template_pools()


def _shift(point, offsets):
    return tuple((point[0] + offset[0], point[1] + offset[1]) for offset in offsets)


def _json_triangle(triangle):
    return [list(point) for point in triangle]


def _json_token(token):
    return [list(point) for point in token]


def _make_card(rng: random.Random, state, desired, assignment):
    for _ in range(1500):
        sources, targets = [None, None], [None, None]
        for group, point_index in enumerate(assignment):
            weights = rng.choice(WEIGHTS)
            source_offsets = rng.choice(TEMPLATE_POOLS[weights])
            candidates = [item for item in TEMPLATE_POOLS[weights]
                          if _shape_signature(item) != _shape_signature(source_offsets)]
            target_offsets = rng.choice(candidates)
            sources[group] = _shift(state[point_index], source_offsets)
            targets[group] = _shift(desired[point_index], target_offsets)
        if any(not all(0 <= r < SIZE and 0 <= c < SIZE for r, c in triangle)
               for triangle in sources + targets):
            continue
        if any(len(set(itertools.chain.from_iterable(panel))) != 6
               for panel in (sources, targets)):
            continue
        card = (tuple(sources), tuple(targets))
        try:
            if _apply_card(state, card) != desired:
                continue
        except ValueError:
            continue
        return {
            "from_triangles": [_json_triangle(triangle) for triangle in sources],
            "to_triangles": [_json_triangle(triangle) for triangle in targets],
        }
    raise RuntimeError("could not construct an unambiguous piecewise-affine card")


def _valid_option(token) -> bool:
    return token[0] != token[1] and all(0 <= r < SIZE and 0 <= c < SIZE for r, c in token)


def _mechanism_traps(state, card):
    sources, targets = card
    traps = []
    translated = []
    crossed = []
    for point in state:
        hits = [i for i, triangle in enumerate(sources) if _strictly_inside(point, triangle)]
        if len(hits) != 1:
            return []
        group = hits[0]
        source, target = sources[group], targets[group]
        translated.append((target[0][0] + point[0] - source[0][0],
                           target[0][1] + point[1] - source[0][1]))
        try:
            crossed.append(_map_point(point, source, targets[1 - group]))
        except ValueError:
            crossed.append((-1, -1))
    for token in (tuple(translated), tuple(crossed)):
        if _valid_option(token):
            traps.append(token)
    return traps


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(1200):
        start, middle, final = rng.sample(ANCHOR_PAIRS, 3)
        if any(start[i] == middle[i] or middle[i] == final[i] or start[i] == final[i]
               for i in range(2)):
            continue
        first_assignment = (0, 1) if rng.randrange(2) == 0 else (1, 0)
        second_assignment = tuple(reversed(first_assignment))
        try:
            raw_cards = [
                _make_card(rng, start, middle, first_assignment),
                _make_card(rng, middle, final, second_assignment),
            ]
            cards = tuple(_card(raw) for raw in raw_cards)
            if _evolve(start, cards) != final:
                continue
        except (RuntimeError, ValueError):
            continue

        distractors = []
        preferred = [middle, (final[1], final[0])]
        preferred.extend(_mechanism_traps(middle, cards[1]))
        fallback = list(ANCHOR_PAIRS) + [(pair[1], pair[0]) for pair in ANCHOR_PAIRS]
        rng.shuffle(fallback)
        for token in preferred + fallback:
            if token != final and token not in distractors and _valid_option(token):
                distractors.append(token)
            if len(distractors) == 4:
                break
        if len(distractors) < 4:
            continue
        rng.shuffle(distractors)
        options = [_json_token(token) for token in distractors]
        options.insert(int(seed) % 5, _json_token(final))
        scene = {
            "start_points": _json_token(start),
            "cards": raw_cards,
            "options": options,
        }
        _parse_scene(scene)
        if is_quarantined(scene):
            continue
        return scene
    raise RuntimeError("could not sample a nondegenerate barycentric relay")


def analytic_gold(scene: dict) -> str:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, cards)
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("final state must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, cards)
    distances = [sum(abs(a[0] - b[0]) + abs(a[1] - b[1])
                     for a, b in zip(target, option))
                 for option in options if option != target]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    try:
        start, cards, options = _parse_scene(scene)
        target = _evolve(start, cards)
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError, IndexError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _parse_scene(scene)
    return []


def _center(point, x: int, y: int, cell: int, gap: int) -> tuple[int, int]:
    row, col = point
    return x + col * (cell + gap) + cell // 2, y + row * (cell + gap) + cell // 2


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, gap: int) -> None:
    for row in range(SIZE):
        for col in range(SIZE):
            left, top = x + col * (cell + gap), y + row * (cell + gap)
            draw.rectangle((left, top, left + cell, top + cell), fill=PAPER, outline=BORDER, width=1)


def _dot(draw, point, color, x, y, cell, gap, radius) -> None:
    cx, cy = _center(point, x, y, cell, gap)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius),
                 fill=color, outline="white", width=1)


def _draw_token(draw, token, x, y, cell, gap) -> None:
    _grid(draw, x, y, cell, gap)
    radius = max(4, cell // 3)
    _dot(draw, token[0], INK, x, y, cell, gap, radius)
    _dot(draw, token[1], RED, x, y, cell, gap, radius)


def _draw_triangles(draw, triangles, x, y) -> None:
    _grid(draw, x, y, MAIN_CELL, MAIN_GAP)
    for group, triangle in enumerate(triangles):
        polygon = [_center(point, x, y, MAIN_CELL, MAIN_GAP) for point in triangle]
        draw.line((*polygon, polygon[0]), fill=FAMILY_EDGES[group], width=4, joint="curve")
        for corner, point in enumerate(triangle):
            _dot(draw, point, VERTEX_COLORS[group * 3 + corner],
                 x, y, MAIN_CELL, MAIN_GAP, 6)


def render(scene: dict) -> Image.Image:
    start, cards, options = _parse_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 24), "CONTAINING TRIANGLE  •  SAME-COLOR CORNERS KEEP THE SAME WEIGHTS",
              fill=INK, font=_font(21), anchor="mm")
    draw.rounded_rectangle((10, 60, 202, 430), radius=12, fill="#f8fafb", outline=PALE, width=3)
    draw.text((106, 89), "START", fill=INK, font=_font(20), anchor="mm")
    draw.text((106, 116), "NAVY  •  RED", fill=BORDER, font=_font(12), anchor="mm")
    _draw_token(draw, start, START_X, START_Y, MAIN_CELL, MAIN_GAP)

    for number, (x, card) in enumerate(zip(CARD_XS, cards), 1):
        sources, targets = card
        draw.rounded_rectangle((x, 60, x + 575, 430), radius=12,
                               fill="#f8fafb", outline=PALE, width=3)
        draw.text((x + 288, 88), f"CARD {number}", fill=INK, font=_font(20), anchor="mm")
        draw.text((x + 110, 116), "FROM", fill=BORDER, font=_font(14), anchor="mm")
        draw.text((x + 465, 116), "TO", fill=BORDER, font=_font(14), anchor="mm")
        draw.text((x + 288, 238), "→", fill=INK, font=_font(28), anchor="mm")
        _draw_triangles(draw, sources, x + CARD_FROM_DX, MAIN_Y)
        _draw_triangles(draw, targets, x + CARD_TO_DX, MAIN_Y)
        draw.text((x + 288, 389), "USE THE ONE FROM TRIANGLE CONTAINING EACH BEAD",
                  fill=BORDER, font=_font(11), anchor="mm")

    draw.line((24, 452, WIDTH - 24, 452), fill=PALE, width=2)
    draw.text((WIDTH // 2, 481), "FINAL BEADS", fill=BORDER, font=_font(18), anchor="mm")
    for label, token, x in zip("ABCDE", options, OPTION_XS):
        cx, cy = x + 70, 524
        draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=INK)
        draw.text((cx, cy), label, fill="white", font=_font(21), anchor="mm")
        _draw_token(draw, token, x, OPTION_Y, OPTION_CELL, OPTION_GAP)
    return image
