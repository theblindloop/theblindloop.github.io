"""Independent PNG-only inverse for the piecewise barycentric relay."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1400, 740)
SIZE = 11
START_X, START_Y = 18, 145
CARD_XS = (215, 805)
CARD_FROM_DX, CARD_TO_DX = 18, 372
MAIN_Y, MAIN_CELL, MAIN_GAP = 145, 15, 2
OPTION_XS = (48, 312, 576, 840, 1104)
OPTION_Y, OPTION_CELL, OPTION_GAP = 565, 11, 2

PALETTE = {
    "tail": np.array((32, 54, 74), dtype=np.float32),
    "head": np.array((198, 40, 40), dtype=np.float32),
    "v0": np.array((216, 67, 21), dtype=np.float32),
    "v1": np.array((249, 168, 37), dtype=np.float32),
    "v2": np.array((142, 36, 170), dtype=np.float32),
    "v3": np.array((0, 131, 143), dtype=np.float32),
    "v4": np.array((21, 101, 192), dtype=np.float32),
    "v5": np.array((46, 125, 50), dtype=np.float32),
}


class OracleError(ValueError):
    pass


def _cross(a, b) -> int:
    return a[0] * b[1] - a[1] * b[0]


def _det(triangle) -> int:
    a, b, c = triangle
    return _cross((b[0] - a[0], b[1] - a[1]), (c[0] - a[0], c[1] - a[1]))


def _weights(point, triangle):
    a, b, c = triangle
    det = _det(triangle)
    n0 = _cross((b[0] - point[0], b[1] - point[1]),
                (c[0] - point[0], c[1] - point[1]))
    n1 = _cross((c[0] - point[0], c[1] - point[1]),
                (a[0] - point[0], a[1] - point[1]))
    n2 = _cross((a[0] - point[0], a[1] - point[1]),
                (b[0] - point[0], b[1] - point[1]))
    return n0, n1, n2, det


def _contains(point, triangle) -> bool:
    n0, n1, n2, det = _weights(point, triangle)
    return det != 0 and all(n * det > 0 for n in (n0, n1, n2))


def _map(point, source, target):
    n0, n1, n2, det = _weights(point, source)
    if det == 0 or not all(n * det > 0 for n in (n0, n1, n2)):
        raise OracleError("bead is not strictly inside its source triangle")
    row_num = n0 * target[0][0] + n1 * target[1][0] + n2 * target[2][0]
    col_num = n0 * target[0][1] + n1 * target[1][1] + n2 * target[2][1]
    if row_num % det or col_num % det:
        raise OracleError("mapped bead falls between grid cells")
    return row_num // det, col_num // det


def _classify(array: np.ndarray, cx: int, cy: int, allowed: tuple[str, ...]):
    patch = array[cy - 2:cy + 3, cx - 2:cx + 3].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ranked = sorted((float(np.linalg.norm(mean - PALETTE[name])), name) for name in allowed)
    if ranked[0][0] < 12.0 and (len(ranked) == 1 or ranked[1][0] - ranked[0][0] > 25.0):
        return ranked[0][1]
    return None


def _scan(array, x, y, cell, gap, allowed):
    found = {}
    for row in range(SIZE):
        for col in range(SIZE):
            cx = x + col * (cell + gap) + cell // 2
            cy = y + row * (cell + gap) + cell // 2
            name = _classify(array, cx, cy, allowed)
            if name is not None:
                if name in found:
                    raise OracleError(f"duplicate {name} marker")
                found[name] = (row, col)
    if set(found) != set(allowed):
        raise OracleError("a marker panel is incomplete")
    return found


def _token(array, x, y, cell, gap):
    found = _scan(array, x, y, cell, gap, ("tail", "head"))
    token = found["tail"], found["head"]
    if token[0] == token[1]:
        raise OracleError("beads collide")
    return token


def _triangles(array, x, y):
    names = tuple(f"v{i}" for i in range(6))
    found = _scan(array, x, y, MAIN_CELL, MAIN_GAP, names)
    triangles = tuple(tuple(found[f"v{group * 3 + corner}"] for corner in range(3))
                      for group in range(2))
    if any(abs(_det(triangle)) < 4 for triangle in triangles):
        raise OracleError("a recovered triangle is degenerate")
    if len(set(point for triangle in triangles for point in triangle)) != 6:
        raise OracleError("triangle corners collide")
    return triangles


def _apply(state, sources, targets):
    output = []
    for point in state:
        hits = [index for index, triangle in enumerate(sources) if _contains(point, triangle)]
        if len(hits) != 1:
            raise OracleError("a bead is not inside exactly one FROM triangle")
        mapped = _map(point, sources[hits[0]], targets[hits[0]])
        if not (0 <= mapped[0] < SIZE and 0 <= mapped[1] < SIZE):
            raise OracleError("mapped bead leaves the board")
        output.append(mapped)
    if output[0] == output[1]:
        raise OracleError("the map merges the beads")
    return tuple(output)


def _validate_option_labels(array):
    ink = PALETTE["tail"]
    for x in OPTION_XS:
        cx, cy = x + 70, 524
        patch = array[cy - 18:cy + 19, cx - 18:cx + 19].astype(np.float32)
        if int((np.linalg.norm(patch - ink, axis=2) < 12.0).sum()) < 700:
            raise OracleError("an option label disk is missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_option_labels(array)
    state = _token(array, START_X, START_Y, MAIN_CELL, MAIN_GAP)
    for x in CARD_XS:
        sources = _triangles(array, x + CARD_FROM_DX, MAIN_Y)
        targets = _triangles(array, x + CARD_TO_DX, MAIN_Y)
        state = _apply(state, sources, targets)
    options = tuple(_token(array, x, OPTION_Y, OPTION_CELL, OPTION_GAP) for x in OPTION_XS)
    if len(set(options)) != 5:
        raise OracleError("options are not distinct")
    matches = [index for index, option in enumerate(options) if option == state]
    if len(matches) != 1:
        raise OracleError("final beads do not match exactly one option")
    return "ABCDE"[matches[0]]
