from __future__ import annotations

from PIL import Image
import numpy as np

TOKEN_RGB = {
    "violet": (126, 79, 171),
    "green": (54, 150, 92),
    "cyan": (42, 153, 176),
    "orange": (230, 126, 45),
    "pink": (213, 83, 131),
    "brown": (143, 94, 58),
}
BORDER_RGB = {"red": (205, 55, 55), "blue": (48, 103, 190), "gold": (213, 158, 35)}
TOKEN_ORDER = tuple(TOKEN_RGB)


def _components(mask: np.ndarray):
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        count = 0
        minx = maxx = x
        miny = maxy = y
        sx = sy = 0
        while stack:
            yy, xx = stack.pop()
            count += 1
            sx += xx
            sy += yy
            minx, maxx = min(minx, xx), max(maxx, xx)
            miny, maxy = min(miny, yy), max(maxy, yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append({"area": count, "bbox": (minx, miny, maxx, maxy), "cx": sx / count, "cy": sy / count})
    return out


def _canon(pair):
    return tuple(sorted(pair, key=TOKEN_ORDER.index))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    cards = []
    for label, rgb in BORDER_RGB.items():
        mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
        large = [c for c in _components(mask) if c["bbox"][2] - c["bbox"][0] > 180 and c["bbox"][3] - c["bbox"][1] > 180]
        if len(large) != 1:
            raise ValueError("could not recover the three option frames")
        cards.append((label, large[0]["bbox"]))

    tokens = []
    for color, rgb in TOKEN_RGB.items():
        mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
        for comp in _components(mask):
            if comp["area"] >= 700:
                tokens.append((color, comp))

    def inside(comp, box):
        x0, y0, x1, y1 = box
        return x0 < comp["cx"] < x1 and y0 < comp["cy"] < y1

    top = [(color, comp) for color, comp in tokens if not any(inside(comp, box) for _, box in cards)]
    if len(top) != 6:
        raise ValueError("source strip does not contain six recoverable tiles")
    top_colors = [color for color, _ in sorted(top, key=lambda item: item[1]["cx"])]
    wanted = {_canon((top_colors[i], top_colors[5 - i])) for i in range(3)}

    matches = []
    for label, box in cards:
        local = [(color, comp) for color, comp in tokens if inside(comp, box)]
        if len(local) != 6:
            raise ValueError("an option does not contain six recoverable tiles")
        ordered = sorted(local, key=lambda item: (item[1]["cy"], item[1]["cx"]))
        pairs = {_canon((ordered[i][0], ordered[i + 1][0])) for i in range(0, 6, 2)}
        if pairs == wanted:
            matches.append(label)
    if len(matches) != 1:
        raise ValueError("image does not yield one unique matching option")
    return matches[0]
