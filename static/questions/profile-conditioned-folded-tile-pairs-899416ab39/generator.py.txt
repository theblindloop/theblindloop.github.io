from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

TOKEN_RGB = {
    "violet": (126, 79, 171),
    "green": (54, 150, 92),
    "cyan": (42, 153, 176),
    "orange": (230, 126, 45),
    "pink": (213, 83, 131),
    "brown": (143, 94, 58),
}
BORDER_RGB = {"red": (205, 55, 55), "blue": (48, 103, 190), "gold": (213, 158, 35)}
TOKEN_ORDER = tuple(TOKEN_RGB)


def _canon_pair(pair):
    return tuple(sorted(pair, key=TOKEN_ORDER.index))


def _canon_option(option):
    return tuple(sorted((_canon_pair(p) for p in option)))


def _target(top):
    return [_canon_pair((top[i], top[5 - i])) for i in range(3)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    top = list(TOKEN_ORDER)
    rng.shuffle(top)
    target = _target(top)
    # Two wrong perfect matchings: one shares one target pair; one shares none.
    wrong1 = [target[0], _canon_pair((top[1], top[3])), _canon_pair((top[2], top[4]))]
    wrong2 = [_canon_pair((top[0], top[4])), _canon_pair((top[1], top[3])), _canon_pair((top[2], top[5]))]
    desired = ("red", "blue", "gold")[seed % 3]
    borders = list(BORDER_RGB)
    rng.shuffle(borders)
    correct_index = borders.index(desired)
    wrongs = [wrong1, wrong2]
    rng.shuffle(wrongs)
    options = []
    wi = 0
    for i in range(3):
        option = list(target) if i == correct_index else list(wrongs[wi])
        wi += i != correct_index
        rng.shuffle(option)
        option = [list(pair if rng.randrange(2) else pair[::-1]) for pair in option]
        options.append(option)
    return {"top_colors": top, "options": options, "border_order": borders}


def _scores(scene: dict) -> list[int]:
    wanted = set(_canon_option(_target(scene["top_colors"])))
    return [len(wanted & set(_canon_option(option))) for option in scene["options"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return scene["border_order"][max(range(3), key=scores.__getitem__)]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    valid = len(scene["top_colors"]) == 6 and len(set(scene["top_colors"])) == 6
    return (not valid) or scores.count(max(scores)) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["options"] = [[list(reversed(pair)) for pair in option] for option in twin["options"]]
    return [("reverse_unordered_pair_storage", twin)]


def _font(size: int, bold: bool = False):
    path = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    return ImageFont.truetype(path, size)


def render(scene: dict) -> Image.Image:
    # Access all declared label inputs explicitly; pairs are canonically drawn.
    top = scene["top_colors"]
    options = scene["options"]
    borders = scene["border_order"]
    image = Image.new("RGB", (900, 620), (250, 249, 245))
    d = ImageDraw.Draw(image)
    d.text((450, 24), "FOLD THE STRIP", fill=(28, 31, 38), font=_font(28, True), anchor="ma")
    d.text((450, 62), "Pair 1–6, 2–5, and 3–4", fill=(55, 59, 66), font=_font(21), anchor="ma")
    x0, y0, size = 246, 118, 68
    for i, color in enumerate(top):
        x = x0 + i * size
        d.rectangle((x, y0, x + size, y0 + size), fill=TOKEN_RGB[color], outline=(25, 28, 32), width=4)
        d.text((x + size / 2, y0 + size / 2), str(i + 1), fill=(255, 255, 255), font=_font(23, True), anchor="mm", stroke_width=2, stroke_fill=(35, 35, 35))
    fold_x = x0 + 3 * size
    for yy in range(103, 211, 14):
        d.line((fold_x, yy, fold_x, min(yy + 7, 211)), fill=(25, 28, 32), width=3)
    d.polygon([(fold_x, 218), (fold_x - 10, 202), (fold_x + 10, 202)], fill=(25, 28, 32))
    d.text((fold_x, 224), "FOLD", fill=(25, 28, 32), font=_font(16, True), anchor="ma")
    d.text((450, 278), "Which option has exactly these three unordered pairs?", fill=(35, 38, 45), font=_font(20, True), anchor="ma")
    card_x = [38, 323, 608]
    for idx, (x, border, raw_option) in enumerate(zip(card_x, borders, options)):
        d.rounded_rectangle((x, 320, x + 254, 578), radius=18, fill=(255, 255, 255), outline=BORDER_RGB[border], width=9)
        d.text((x + 127, 345), border.upper(), fill=BORDER_RGB[border], font=_font(19, True), anchor="mm")
        option = sorted((_canon_pair(pair) for pair in raw_option))
        for row, pair in enumerate(option):
            cy = 404 + row * 68
            d.rounded_rectangle((x + 56, cy - 25, x + 198, cy + 25), radius=9, fill=(238, 238, 234), outline=(45, 48, 52), width=3)
            for col, color in enumerate(pair):
                xx = x + 76 + col * 74
                d.rectangle((xx, cy - 19, xx + 38, cy + 19), fill=TOKEN_RGB[color], outline=(30, 33, 37), width=2)
            d.text((x + 127, cy), "+", fill=(34, 37, 42), font=_font(20, True), anchor="mm")
    return image
