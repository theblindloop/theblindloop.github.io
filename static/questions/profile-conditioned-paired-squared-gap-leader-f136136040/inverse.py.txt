"""Independent final-PNG inverse for the paired squared-gap decision."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleError(ValueError):
    """Raised when the required paired-marker structure is not measurable."""


def _runs(values: np.ndarray) -> list[np.ndarray]:
    unique = np.unique(values.astype(np.int32))
    if len(unique) == 0:
        return []
    splits = np.where(np.diff(unique) > 1)[0] + 1
    return [part for part in np.split(unique, splits) if len(part)]


def _marker_centers(mask: np.ndarray) -> dict[tuple[int, int, int], tuple[float, float]]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 900:
        raise OracleError("too little colored marker ink")
    x_runs = _runs(xs)
    if len(x_runs) != 10:
        raise OracleError(f"expected ten checkpoint columns, found {len(x_runs)}")
    x_centers = np.asarray([run.mean() for run in x_runs], dtype=float)
    x_gaps = np.diff(x_centers)
    split = int(np.argmax(x_gaps)) + 1
    if split != 5 or x_gaps[split - 1] < 1.45 * max(np.delete(x_gaps, split - 1)):
        raise OracleError("checkpoint columns do not form two panel columns")

    result: dict[tuple[int, int, int], tuple[float, float]] = {}
    for x_index, x_run in enumerate(x_runs):
        local = np.isin(xs, x_run)
        local_x, local_y = xs[local], ys[local]
        y_runs = _runs(local_y)
        if len(y_runs) != 2:
            raise OracleError("each checkpoint column must contain one marker in each panel row")
        col = 0 if x_index < 5 else 1
        checkpoint = x_index % 5
        for row, y_run in enumerate(y_runs):
            component = np.isin(local_y, y_run)
            if int(component.sum()) < 70:
                raise OracleError("marker component is too small")
            result[(row, col, checkpoint)] = (
                float(local_x[component].mean()),
                float(local_y[component].mean()),
            )
    if len(result) != 20:
        raise OracleError("incomplete colored marker map")
    return result


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    blue_mask = (blue > 115) & (blue - red > 85) & (blue - green > 42)
    orange_mask = (red > 175) & (red - blue > 140) & (red - green > 72)
    blue_centers = _marker_centers(blue_mask)
    orange_centers = _marker_centers(orange_mask)

    scores: list[float] = []
    for panel in range(4):
        row, col = divmod(panel, 2)
        score = 0.0
        for checkpoint in range(5):
            bx, by = blue_centers[(row, col, checkpoint)]
            ox, oy = orange_centers[(row, col, checkpoint)]
            if abs(bx - ox) > 1.0:
                raise OracleError("paired markers are not vertically aligned")
            gap = abs(by - oy)
            if gap < 10.0:
                raise OracleError("paired markers are not visibly separated")
            score += gap * gap
        scores.append(score)

    order = np.argsort(np.asarray(scores))[::-1]
    if scores[int(order[0])] - scores[int(order[1])] < 8.0:
        raise OracleError("raster squared-sum decision is ambiguous")
    return "ABCD"[int(order[0])]
