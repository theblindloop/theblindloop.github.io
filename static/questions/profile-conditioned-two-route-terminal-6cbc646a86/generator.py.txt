"""Deterministic renderer and analytic graph decision for two_route_terminal."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 840
HEIGHT = 360
COLORS = ("red", "blue", "orange")
RGB = {
    "red": (224, 58, 62),
    "blue": (55, 116, 224),
    "orange": (238, 143, 47),
    "green": (36, 166, 91),
    "node": (207, 219, 231),
    "ink": (25, 28, 33),
    "border": (190, 198, 207),
    "background": (255, 255, 255),
}


def _reachable(nodes: list[dict], edges: list[list[str]], removed: int | None = None) -> bool:
    adjacency = {node["id"]: set() for node in nodes}
    for index, (left, right) in enumerate(edges):
        if index == removed:
            continue
        adjacency[left].add(right)
        adjacency[right].add(left)
    start = next(node["id"] for node in nodes if node["role"] == "start")
    terminal = next(node["id"] for node in nodes if node["role"] == "terminal")
    seen = {start}
    queue = deque([start])
    while queue:
        current = queue.popleft()
        for neighbor in adjacency[current] - seen:
            seen.add(neighbor)
            queue.append(neighbor)
    return terminal in seen


def _two_edge_disjoint(nodes: list[dict], edges: list[list[str]]) -> bool:
    return _reachable(nodes, edges) and all(
        _reachable(nodes, edges, removed=index) for index in range(len(edges))
    )


def _base_panel(kind: str, color: str, slot: int, mirror_x: bool, mirror_y: bool) -> dict:
    if kind == "redundant":
        coords = {
            "s": (24, 180), "a": (68, 124), "b": (68, 236),
            "x": (124, 180), "c": (180, 124), "d": (180, 236),
            "t": (236, 180), "y": (124, 72), "z": (178, 72),
        }
        edges = [
            ["s", "a"], ["s", "b"], ["a", "x"], ["b", "x"],
            ["x", "c"], ["x", "d"], ["c", "t"], ["d", "t"],
            ["x", "y"], ["y", "z"],
        ]
    else:
        coords = {
            "s": (24, 180), "a": (68, 124), "b": (68, 236),
            "x": (106, 180), "y": (154, 180), "c": (192, 124),
            "d": (192, 236), "t": (236, 180),
            "z": ((106 if kind == "bottleneck_left" else 154), 72),
        }
        edges = [
            ["s", "a"], ["s", "b"], ["a", "x"], ["b", "x"],
            ["x", "y"], ["y", "c"], ["y", "d"], ["c", "t"],
            ["d", "t"], [("x" if kind == "bottleneck_left" else "y"), "z"],
        ]
    x_offset = 10 + slot * 275
    nodes = []
    for node_id, (x, y) in coords.items():
        if mirror_x:
            x = 260 - x
        if mirror_y:
            y = 360 - y
        role = "start" if node_id == "s" else "terminal" if node_id == "t" else "node"
        fill = "green" if role == "start" else color if role == "terminal" else "node"
        nodes.append({"id": node_id, "x": x_offset + x, "y": y, "role": role, "fill": fill})
    return {"slot": slot, "kind": kind, "terminal_color": color, "nodes": nodes, "edges": edges}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_color = COLORS[int(seed) % 3]
    target_slot = (int(seed) // 3) % 3
    remaining = [color for color in COLORS if color != answer_color]
    rng.shuffle(remaining)
    slot_colors: list[str] = []
    for slot in range(3):
        slot_colors.append(answer_color if slot == target_slot else remaining.pop())
    panels = []
    loser_kinds = iter(("bottleneck_left", "bottleneck_right"))
    for slot in range(3):
        kind = "redundant" if slot == target_slot else next(loser_kinds)
        panels.append(
            _base_panel(kind, slot_colors[slot], slot, bool(rng.getrandbits(1)), bool(rng.getrandbits(1)))
        )
    return {"width": WIDTH, "height": HEIGHT, "panels": panels}


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), RGB["background"])
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        slot = int(panel["slot"])
        x0 = (10 + slot * 275) * scale
        draw.rounded_rectangle(
            (x0 + 2 * scale, 16 * scale, x0 + 264 * scale, 344 * scale),
            radius=12 * scale,
            outline=RGB["border"],
            width=2 * scale,
        )
        lookup = {node["id"]: node for node in panel["nodes"]}
        for left, right in panel["edges"]:
            a, b = lookup[left], lookup[right]
            draw.line(
                (a["x"] * scale, a["y"] * scale, b["x"] * scale, b["y"] * scale),
                fill=RGB["ink"],
                width=7 * scale,
            )
        for node in panel["nodes"]:
            radius = 13
            box = (
                (node["x"] - radius) * scale,
                (node["y"] - radius) * scale,
                (node["x"] + radius) * scale,
                (node["y"] + radius) * scale,
            )
            draw.ellipse(box, fill=RGB[node["fill"]], outline=RGB["ink"], width=3 * scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    winners = []
    for panel in scene["panels"]:
        if _two_edge_disjoint(panel["nodes"], panel["edges"]):
            winners.append(panel["terminal_color"])
    if len(winners) != 1 or winners[0] not in COLORS:
        raise ValueError(f"expected one link-redundant panel, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(
        (1.0 if _two_edge_disjoint(panel["nodes"], panel["edges"]) else 0.0)
        for panel in scene["panels"]
    )
    return float(scores[-1] - scores[-2])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"].reverse()
    return [("reverse_edge_storage", twin)]
