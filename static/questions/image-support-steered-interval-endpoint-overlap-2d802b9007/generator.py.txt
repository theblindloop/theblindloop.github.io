"""Deterministic scene sampler and renderer for compact interval overlap."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


def sample_scene(seed: int) -> dict:
    """Sample balanced, grid-aware scenes without encoding gold as a field."""
    rng = random.Random(int(seed))
    width = 512
    height = 512
    boundary_x = rng.choice((192, 256, 320))
    top_y = rng.choice((160, 224, 288))
    bottom_y = top_y + 64
    offset = rng.choice((22, 24, 26))
    overlaps = int(seed) % 2 == 0
    if overlaps:
        top_inner_x = boundary_x + offset
        bottom_inner_x = boundary_x - offset
    else:
        top_inner_x = boundary_x - offset
        bottom_inner_x = boundary_x + offset
    return {
        "width": width,
        "height": height,
        "top_y": top_y,
        "bottom_y": bottom_y,
        "top_inner_x": top_inner_x,
        "bottom_inner_x": bottom_inner_x,
        "rail_length": rng.choice((76, 84, 92)),
        "rail_width": rng.choice((5, 7)),
        "cap_size": rng.choice((9, 11)),
        "background": [248, 249, 251],
        "top_rail": [240, 155, 137],
        "top_cap": [194, 68, 52],
        "bottom_rail": [118, 177, 221],
        "bottom_cap": [28, 93, 153],
    }


def _rect(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], fill: tuple[int, int, int]):
    draw.rectangle(box, fill=fill)


def render(scene: dict) -> Image.Image:
    """Draw two staggered intervals and their emphasized opposing endpoints."""
    width = int(scene["width"])
    height = int(scene["height"])
    top_y = int(scene["top_y"])
    bottom_y = int(scene["bottom_y"])
    top_x = int(scene["top_inner_x"])
    bottom_x = int(scene["bottom_inner_x"])
    length = int(scene["rail_length"])
    rail_width = int(scene["rail_width"])
    cap_size = int(scene["cap_size"])
    background = tuple(scene["background"])
    top_rail = tuple(scene["top_rail"])
    top_cap = tuple(scene["top_cap"])
    bottom_rail = tuple(scene["bottom_rail"])
    bottom_cap = tuple(scene["bottom_cap"])

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    rh = rail_width // 2
    ch = cap_size // 2

    # Top interval extends leftward to its emphasized right endpoint.
    _rect(draw, (top_x - length, top_y - rh, top_x, top_y + rh), top_rail)
    _rect(draw, (top_x - ch, top_y - ch, top_x + ch, top_y + ch), top_cap)
    # Bottom interval begins at its emphasized left endpoint and extends rightward.
    _rect(draw, (bottom_x, bottom_y - rh, bottom_x + length, bottom_y + rh), bottom_rail)
    _rect(draw, (bottom_x - ch, bottom_y - ch, bottom_x + ch, bottom_y + ch), bottom_cap)

    # Small pale terminal ticks make both finite extents explicit without entering
    # the inverse arm's dark-cap masks.
    outer = (176, 181, 189)
    _rect(draw, (top_x - length - 1, top_y - 8, top_x - length + 1, top_y + 8), outer)
    _rect(draw, (bottom_x + length - 1, bottom_y - 8, bottom_x + length + 1, bottom_y + 8), outer)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if float(scene["top_inner_x"]) >= float(scene["bottom_inner_x"]) else "no"


def margin(scene: dict) -> float:
    return float(scene["top_inner_x"]) - float(scene["bottom_inner_x"])


def is_quarantined(scene: dict) -> bool:
    return abs(float(scene["top_inner_x"]) - float(scene["bottom_inner_x"])) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
