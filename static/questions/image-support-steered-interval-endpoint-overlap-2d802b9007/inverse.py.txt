"""Independent PNG-only inverse arm for compact interval overlap.

This module intentionally imports no candidate implementation modules and sees
only raster pixels. The two dark endpoint colors are measured independently.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


_CORAL = np.array([194, 68, 52], dtype=np.int16)
_BLUE = np.array([28, 93, 153], dtype=np.int16)


def _centroid_x(array: np.ndarray, target: np.ndarray, name: str) -> float:
    delta = array.astype(np.int32) - target.astype(np.int32)
    mask = np.sum(delta * delta, axis=2) <= 18 * 18
    ys, xs = np.nonzero(mask)
    if len(xs) < 55:
        raise ValueError(f"missing or damaged {name} endpoint cap")
    span_x = int(xs.max() - xs.min() + 1)
    span_y = int(ys.max() - ys.min() + 1)
    if span_x < 7 or span_y < 7 or span_x > 13 or span_y > 13:
        raise ValueError(f"invalid {name} endpoint geometry")
    if float(mask[ys.min() : ys.max() + 1, xs.min() : xs.max() + 1].mean()) < 0.78:
        raise ValueError(f"fragmented {name} endpoint cap")
    return float(xs.mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    top_right = _centroid_x(array, _CORAL, "coral")
    bottom_left = _centroid_x(array, _BLUE, "blue")
    separation = top_right - bottom_left
    if abs(separation) < 8.0:
        raise ValueError("endpoint ordering is raster-ambiguous")
    return "yes" if separation >= 0.0 else "no"
