"""Deterministic renderer and analytic arm for the butterfly-wing world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
AA = 3

_PALETTES = (
    ((34, 92, 168), (248, 245, 236)),
    ((126, 52, 154), (246, 249, 240)),
    ((20, 126, 112), (250, 244, 248)),
    ((184, 62, 74), (243, 249, 252)),
    ((165, 96, 24), (247, 246, 252)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wing_pairs = 1 + (int(seed) % 4)
    foreground, background = _PALETTES[rng.randrange(len(_PALETTES))]
    spacing = 300.0 / wing_pairs
    heights = []
    widths = []
    offsets = []
    for _ in range(wing_pairs):
        height_fraction = rng.uniform(0.84, 0.90) if wing_pairs == 1 else rng.uniform(0.66, 0.74)
        heights.append(round(spacing * height_fraction, 3))
        widths.append(round(rng.uniform(204.0, 220.0), 3))
        offsets.append([round(rng.uniform(-2.5, 2.5), 3), round(rng.uniform(-2.5, 2.5), 3)])
    return {
        "wing_pairs": wing_pairs,
        "center_x": round(rng.uniform(246.0, 266.0), 3),
        "center_y": round(rng.uniform(253.0, 267.0), 3),
        "scale": round(rng.uniform(0.91, 1.04), 4),
        "body_color": list(foreground),
        "background_color": list(background),
        "lobe_widths": widths,
        "lobe_heights": heights,
        "side_offsets": offsets,
    }


def _S(value: float) -> int:
    return int(round(value * AA))


def render(scene: dict) -> Image.Image:
    # Direct accesses are intentional: every declared latent field visibly renders.
    pairs = int(scene["wing_pairs"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    color = tuple(int(v) for v in scene["body_color"])
    background = tuple(int(v) for v in scene["background_color"])
    widths = [float(v) for v in scene["lobe_widths"]]
    heights = [float(v) for v in scene["lobe_heights"]]
    offsets = [[float(x) for x in pair] for pair in scene["side_offsets"]]
    if not (len(widths) == len(heights) == len(offsets) == pairs):
        raise ValueError("wing geometry must have one entry per pair")

    image = Image.new("RGB", (WIDTH * AA, HEIGHT * AA), background)
    draw = ImageDraw.Draw(image)
    body_half = 27.0 * scale
    spacing = 300.0 * scale / pairs

    # Broad filled lobes are drawn first and overlap a thick central body. The
    # union is a single connected, high-fill silhouette with no interior holes.
    for index in range(pairs):
        base_y = cy + (index - (pairs - 1) / 2.0) * spacing
        w = widths[index] * scale
        h = heights[index] * scale
        for side, y_delta in ((-1, offsets[index][0]), (1, offsets[index][1])):
            lobe_cx = cx + side * (body_half + 0.38 * w)
            lobe_cy = base_y + y_delta * scale
            box = (
                _S(lobe_cx - w / 2),
                _S(lobe_cy - h / 2),
                _S(lobe_cx + w / 2),
                _S(lobe_cy + h / 2),
            )
            draw.ellipse(box, fill=color)

    body_top = cy - 153.0 * scale
    body_bottom = cy + 153.0 * scale
    draw.rounded_rectangle(
        (_S(cx - body_half), _S(body_top), _S(cx + body_half), _S(body_bottom)),
        radius=_S(body_half),
        fill=color,
    )
    head_y = cy - 164.0 * scale
    head_r = 34.0 * scale
    draw.ellipse(
        (_S(cx - head_r), _S(head_y - head_r), _S(cx + head_r), _S(head_y + head_r)),
        fill=color,
    )
    # Thin antennae are explicit non-wings in the wording. Their small area
    # makes the independent component measurement exclude them mechanically.
    antenna_width = max(3, _S(4.2 * scale))
    left_tip = (cx - 56.0 * scale, head_y - 53.0 * scale)
    right_tip = (cx + 56.0 * scale, head_y - 53.0 * scale)
    root_left = (cx - 13.0 * scale, head_y - 20.0 * scale)
    root_right = (cx + 13.0 * scale, head_y - 20.0 * scale)
    draw.line([(_S(root_left[0]), _S(root_left[1])), (_S(left_tip[0]), _S(left_tip[1]))], fill=color, width=antenna_width)
    draw.line([(_S(root_right[0]), _S(root_right[1])), (_S(right_tip[0]), _S(right_tip[1]))], fill=color, width=antenna_width)
    tip_r = 6.0 * scale
    for tip in (left_tip, right_tip):
        draw.ellipse((_S(tip[0] - tip_r), _S(tip[1] - tip_r), _S(tip[0] + tip_r), _S(tip[1] + tip_r)), fill=color)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(2 * int(scene["wing_pairs"]))


def margin(scene: dict) -> float:
    pairs = int(scene["wing_pairs"])
    scale = float(scene["scale"])
    heights = [float(v) * scale for v in scene["lobe_heights"]]
    offsets = [[float(x) * scale for x in pair] for pair in scene["side_offsets"]]
    spacing = 300.0 * scale / pairs
    gaps = []
    for side_index in (0, 1):
        for index in range(pairs - 1):
            center_gap = spacing + offsets[index + 1][side_index] - offsets[index][side_index]
            gaps.append(center_gap - (heights[index] + heights[index + 1]) / 2.0)
    if not gaps:
        gaps.append(42.0 * scale)
    neck_overlap = min(float(v) * scale * 0.12 for v in scene["lobe_widths"])
    return round(min(gaps + [neck_overlap]), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
