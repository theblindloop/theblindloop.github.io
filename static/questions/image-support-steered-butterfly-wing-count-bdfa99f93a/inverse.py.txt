"""Independent image-only inverse arm for visible butterfly wings."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MIN_FOREGROUND = 2500
_MIN_WING_AREA = 650
_MIN_WING_WIDTH = 24
_MIN_WING_HEIGHT = 16


def _mask_from_image(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    corners = np.concatenate((arr[:12, :12].reshape(-1, 3), arr[:12, -12:].reshape(-1, 3), arr[-12:, :12].reshape(-1, 3), arr[-12:, -12:].reshape(-1, 3)))
    background = np.median(corners, axis=0)
    distance = np.sqrt(np.sum((arr - background) ** 2, axis=2))
    mask = distance > 58.0
    if int(mask.sum()) < _MIN_FOREGROUND:
        raise ValueError("foreground silhouette is absent")
    return mask


def _components(mask: np.ndarray) -> list[dict]:
    work = mask.copy()
    height, width = work.shape
    found = []
    while True:
        points = np.argwhere(work)
        if not len(points):
            break
        y0, x0 = (int(v) for v in points[0])
        work[y0, x0] = False
        queue = deque([(y0, x0)])
        area = 0
        min_x = max_x = x0
        min_y = max_y = y0
        touches = False
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x = min(min_x, x)
            max_x = max(max_x, x)
            min_y = min(min_y, y)
            max_y = max(max_y, y)
            touches = touches or x == 0 or y == 0 or x == width - 1 or y == height - 1
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and work[yy, xx]:
                    work[yy, xx] = False
                    queue.append((yy, xx))
        found.append({"area": area, "x0": min_x, "x1": max_x, "y0": min_y, "y1": max_y, "touches": touches})
    return found


def analyze_image(image: Image.Image) -> dict:
    mask = _mask_from_image(image)
    foreground_components = _components(mask)
    large_foreground = [item for item in foreground_components if item["area"] >= 30]
    if len(large_foreground) != 1:
        raise ValueError("silhouette is not one connected visible component")

    ys, xs = np.nonzero(mask)
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    if min(x0, y0, mask.shape[1] - 1 - x1, mask.shape[0] - 1 - y1) < 8:
        raise ValueError("silhouette lacks canvas clearance")
    crop = mask[y0 : y1 + 1, x0 : x1 + 1]
    # A genuine filled silhouette has no enclosed background island. This also
    # makes destructive interior evidence erasures cause abstention.
    holes = [item for item in _components(~crop) if not item["touches"] and item["area"] >= 20]
    if holes:
        raise ValueError("filled silhouette contains an interior erasure")

    cx = int(round((x0 + x1) / 2.0))
    silhouette_height = y1 - y0 + 1
    # Remove the full central body (about 13% of silhouette width / 8% of
    # height), not merely its center. What remains are independent broad lobes.
    cut_half = max(24, int(round(0.082 * silhouette_height)))
    left = mask.copy()
    left[:, cx - cut_half :] = False
    right = mask.copy()
    right[:, : cx + cut_half + 1] = False

    def wings(side_mask: np.ndarray) -> list[dict]:
        return [
            item
            for item in _components(side_mask)
            if item["area"] >= _MIN_WING_AREA
            and item["x1"] - item["x0"] + 1 >= _MIN_WING_WIDTH
            and item["y1"] - item["y0"] + 1 >= _MIN_WING_HEIGHT
        ]

    left_wings = wings(left)
    right_wings = wings(right)
    if len(left_wings) != len(right_wings):
        raise ValueError("left and right broad-wing counts disagree")
    if not 1 <= len(left_wings) <= 4:
        raise ValueError("visible broad-wing count is outside the declared answer set")
    # The renderer's visible wings are broad, solid oval lobes. Validate that
    # area-like evidence directly: a large rectangular erasure at an outer wing
    # edge must not silently retain the same count. This is independent of any
    # latent geometry and uses only each recovered raster component's area and
    # box. Pairing by vertical centroid also checks that both distant sides carry
    # the same number of comparably intact broad wings.
    for item in left_wings + right_wings:
        box_area = (item["x1"] - item["x0"] + 1) * (item["y1"] - item["y0"] + 1)
        fill = item["area"] / box_area
        if not 0.79 <= fill <= 0.89:
            raise ValueError("a broad wing lobe is not a solid rounded region")
    left_wings.sort(key=lambda item: (item["y0"] + item["y1"]) / 2.0)
    right_wings.sort(key=lambda item: (item["y0"] + item["y1"]) / 2.0)
    pair_balance = []
    for left_item, right_item in zip(left_wings, right_wings, strict=True):
        pair_balance.append(abs(left_item["area"] - right_item["area"]) / max(left_item["area"], right_item["area"]))
    if max(pair_balance) > 0.075:
        raise ValueError("paired broad wings have materially unequal visible area")
    decision = str(len(left_wings) + len(right_wings))
    area_headroom = min(item["area"] - _MIN_WING_AREA for item in left_wings + right_wings)
    balance_headroom = (0.075 - max(pair_balance)) * min(item["area"] for item in left_wings + right_wings)
    return {
        "decision": decision,
        "runner_up_margin": round(float(min(area_headroom, balance_headroom)), 3),
        "left_wings": len(left_wings),
        "right_wings": len(right_wings),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
