"""Mirror-bounce panel world: which panel's reflected beam reaches the red disk.

Every panel shows one straight beam that starts at an orange dot, travels to a
blue mirror bar and stops there, plus one red disk. The continuation past the
mirror is never drawn: the reader must reflect the beam about the mirror with
equal angles and follow the reflected straight line. In exactly one panel that
reflected line passes through the red disk; in the others it clears the disk.

Scene-constant furniture (mirror length, beam length, disk radius) makes ink,
size and count identical across the five panels, so only the reflection
geometry can pick the winner.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "mirror_bounce_panel-renderer-0.1.0"

PANEL = 200
GUTTER = 12
MARGIN = 16
LABEL_BAND = 26
PANELS = 5
CANVAS_W = 2 * MARGIN + PANELS * PANEL + (PANELS - 1) * GUTTER
CANVAS_H = MARGIN + PANEL + LABEL_BAND

FRAME_RGB = (150, 150, 150)
LABEL_RGB = (196, 196, 196)
BACKGROUND = (255, 255, 255)
FRAME_W = 3
BEAM_W = 3
MIRROR_W = 5
SOURCE_R = 6

WALL = 12.0
QUARANTINE_MARGIN = 6.0
HIT_END_CLEARANCE = 22.0
WIN_CORE_CLEARANCE = 9.0
LOSE_CLEARANCE = 10.0
STRAIGHT_CLEARANCE = 12.0
FURNITURE_CLEARANCE = 7.0

PALETTE = (
    {"beam": (20, 140, 60), "mirror": (30, 70, 200), "target": (200, 25, 25),
     "source": (235, 150, 15)},
    {"beam": (15, 150, 75), "mirror": (20, 60, 210), "target": (215, 45, 35),
     "source": (240, 140, 25)},
    {"beam": (30, 130, 50), "mirror": (45, 85, 195), "target": (185, 20, 45),
     "source": (230, 160, 10)},
    {"beam": (10, 145, 85), "mirror": (25, 75, 215), "target": (205, 35, 20),
     "source": (225, 145, 30)},
)


def _unit(vx: float, vy: float) -> tuple[float, float]:
    norm = math.hypot(vx, vy)
    return (vx / norm, vy / norm)


def _ray_clearance(px: float, py: float, ox: float, oy: float, dx: float, dy: float,
                   radius: float) -> float:
    """Signed clearance of a disk from a ray: negative when the ray strikes it."""
    t = max(0.0, (px - ox) * dx + (py - oy) * dy)
    return math.hypot(px - (ox + t * dx), py - (oy + t * dy)) - radius


def _segment_distance(px: float, py: float, ax: float, ay: float, bx: float,
                      by: float) -> float:
    vx, vy = bx - ax, by - ay
    length2 = vx * vx + vy * vy
    t = 0.0 if length2 == 0.0 else ((px - ax) * vx + (py - ay) * vy) / length2
    t = min(1.0, max(0.0, t))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def _panel_geometry(scene: dict, index: int) -> dict:
    """Reduce one stored panel record to the geometry the decision needs."""
    panel = scene["panels"][index]
    ax, ay = panel["mirror"]["ax"], panel["mirror"]["ay"]
    bx, by = panel["mirror"]["bx"], panel["mirror"]["by"]
    mx, my = _unit(bx - ax, by - ay)
    nx, ny = -my, mx
    sx, sy = panel["source"]["cx"], panel["source"]["cy"]
    hx, hy = panel["hit"]["cx"], panel["hit"]["cy"]
    dx, dy = _unit(hx - sx, hy - sy)
    dot = dx * nx + dy * ny
    rx, ry = dx - 2.0 * dot * nx, dy - 2.0 * dot * ny
    return {"sx": sx, "sy": sy, "hx": hx, "hy": hy, "dx": dx, "dy": dy,
            "rx": rx, "ry": ry, "mx": mx, "my": my, "nx": nx, "ny": ny}


def _clearances(scene: dict) -> list[float]:
    out = []
    for index in range(PANELS):
        geo = _panel_geometry(scene, index)
        target = scene["panels"][index]["target"]
        out.append(
            _ray_clearance(target["cx"], target["cy"], geo["hx"], geo["hy"],
                           geo["rx"], geo["ry"], scene["target_radius"])
        )
    return out


def analytic_gold(scene: dict) -> str:
    clearances = _clearances(scene)
    hits = [i for i, value in enumerate(clearances) if value < 0.0]
    if len(hits) != 1:
        raise ValueError(f"scene does not have exactly one struck disk: {hits}")
    return f"panel-{hits[0] + 1}"


def margin(scene: dict) -> float:
    clearances = _clearances(scene)
    hits = [i for i, value in enumerate(clearances) if value < 0.0]
    if len(hits) != 1:
        return 0.0
    winner = hits[0]
    losers = [value for i, value in enumerate(clearances) if i != winner]
    return float(min(-clearances[winner], min(losers)))


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical, answer-preserving rewrites of the stored scene."""
    swapped = {
        **scene,
        "panels": [
            {
                **panel,
                "mirror": {
                    "ax": panel["mirror"]["bx"], "ay": panel["mirror"]["by"],
                    "bx": panel["mirror"]["ax"], "by": panel["mirror"]["ay"],
                },
            }
            for panel in scene["panels"]
        ],
    }
    shifted = {**scene, "palette": scene["palette"] + len(PALETTE)}
    return [("mirror_endpoint_swap", swapped), ("palette_period", shifted)]


def _sample_panel(rng: random.Random, half: float, beam: float, radius: float,
                  winner: bool) -> dict | None:
    theta = rng.uniform(0.0, math.pi)
    mx, my = math.cos(theta), math.sin(theta)
    nx, ny = -my, mx
    span_x = half * abs(mx)
    span_y = half * abs(my)
    if WALL + span_x > PANEL - WALL - span_x or WALL + span_y > PANEL - WALL - span_y:
        return None
    cx = rng.uniform(WALL + span_x, PANEL - WALL - span_x)
    cy = rng.uniform(WALL + span_y, PANEL - WALL - span_y)
    ax, ay = cx - half * mx, cy - half * my
    bx, by = cx + half * mx, cy + half * my
    slide = rng.uniform(-(half - HIT_END_CLEARANCE), half - HIT_END_CLEARANCE)
    hx, hy = cx + slide * mx, cy + slide * my

    side = rng.choice((-1.0, 1.0))
    along = rng.choice((-1.0, 1.0))
    phi = rng.uniform(0.28, 0.95)
    ux = side * math.cos(phi) * nx + along * math.sin(phi) * mx
    uy = side * math.cos(phi) * ny + along * math.sin(phi) * my
    sx, sy = hx + beam * ux, hy + beam * uy
    if not (WALL + 2 <= sx <= PANEL - WALL - 2 and WALL + 2 <= sy <= PANEL - WALL - 2):
        return None
    dx, dy = -ux, -uy
    dot = dx * nx + dy * ny
    rx, ry = dx - 2.0 * dot * nx, dy - 2.0 * dot * ny

    for _ in range(60):
        if winner:
            travel = rng.uniform(42.0, 105.0)
            offset = rng.uniform(-(radius - WIN_CORE_CLEARANCE), radius - WIN_CORE_CLEARANCE)
            tx = hx + travel * rx - offset * ry
            ty = hy + travel * ry + offset * rx
        else:
            tx = rng.uniform(radius + 10.0, PANEL - radius - 10.0)
            ty = rng.uniform(radius + 10.0, PANEL - radius - 10.0)
        if not (radius + 10.0 <= tx <= PANEL - radius - 10.0):
            continue
        if not (radius + 10.0 <= ty <= PANEL - radius - 10.0):
            continue
        reflected = _ray_clearance(tx, ty, hx, hy, rx, ry, radius)
        straight = _ray_clearance(tx, ty, hx, hy, dx, dy, radius)
        if winner and reflected > -WIN_CORE_CLEARANCE:
            continue
        if not winner and reflected < LOSE_CLEARANCE:
            continue
        if straight < STRAIGHT_CLEARANCE:
            continue
        if _segment_distance(tx, ty, ax, ay, bx, by) < radius + FURNITURE_CLEARANCE:
            continue
        if _segment_distance(tx, ty, sx, sy, hx, hy) < radius + FURNITURE_CLEARANCE:
            continue
        if math.hypot(tx - sx, ty - sy) < radius + SOURCE_R + 10.0:
            continue
        return {
            "mirror": {"ax": ax, "ay": ay, "bx": bx, "by": by},
            "hit": {"cx": hx, "cy": hy},
            "source": {"cx": sx, "cy": sy},
            "target": {"cx": tx, "cy": ty},
        }
    return None


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(4000):
        half = rng.uniform(52.0, 70.0)
        beam = rng.uniform(58.0, 88.0)
        radius = rng.uniform(11.0, 14.0)
        winner = rng.randrange(PANELS)
        panels: list[dict] = []
        for index in range(PANELS):
            panel = None
            for _ in range(80):
                panel = _sample_panel(rng, half, beam, radius, index == winner)
                if panel is not None:
                    break
            if panel is None:
                break
            panels.append(panel)
        if len(panels) != PANELS:
            continue
        scene = {
            "canvas": [CANVAS_W, CANVAS_H],
            "palette": rng.randrange(len(PALETTE)),
            "mirror_half_length": half,
            "beam_length": beam,
            "target_radius": radius,
            "panels": panels,
        }
        clearances = _clearances(scene)
        hits = [i for i, value in enumerate(clearances) if value < 0.0]
        if hits != [winner]:
            continue
        if margin(scene) < QUARANTINE_MARGIN + 2.0:
            continue
        if not _cheap_cues_are_broken(scene, winner):
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _cheap_cues_are_broken(scene: dict, winner: int) -> bool:
    """No panel-local proximity statistic may single out the winning panel."""
    to_source, to_hit, straight = [], [], []
    for index in range(PANELS):
        geo = _panel_geometry(scene, index)
        target = scene["panels"][index]["target"]
        to_source.append(math.hypot(target["cx"] - geo["sx"], target["cy"] - geo["sy"]))
        to_hit.append(math.hypot(target["cx"] - geo["hx"], target["cy"] - geo["hy"]))
        straight.append(
            _ray_clearance(target["cx"], target["cy"], geo["hx"], geo["hy"],
                           geo["dx"], geo["dy"], scene["target_radius"])
        )
    for values in (to_source, to_hit, straight):
        if not any(value < values[winner] for i, value in enumerate(values) if i != winner):
            return False
    return True


def _panel_origin(index: int) -> tuple[int, int]:
    return MARGIN + index * (PANEL + GUTTER), MARGIN


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), BACKGROUND)
    draw = ImageDraw.Draw(image)
    colours = PALETTE[scene["palette"] % len(PALETTE)]
    radius = scene["target_radius"]
    for index in range(len(scene["panels"])):
        panel = scene["panels"][index]
        ox, oy = _panel_origin(index)
        draw.rectangle([ox, oy, ox + PANEL, oy + PANEL], outline=FRAME_RGB, width=FRAME_W)
        target = panel["target"]
        draw.ellipse(
            [ox + target["cx"] - radius, oy + target["cy"] - radius,
             ox + target["cx"] + radius, oy + target["cy"] + radius],
            fill=colours["target"],
        )
        source, hit = panel["source"], panel["hit"]
        draw.line(
            [ox + source["cx"], oy + source["cy"], ox + hit["cx"], oy + hit["cy"]],
            fill=colours["beam"], width=BEAM_W,
        )
        mirror = panel["mirror"]
        draw.line(
            [ox + mirror["ax"], oy + mirror["ay"], ox + mirror["bx"], oy + mirror["by"]],
            fill=colours["mirror"], width=MIRROR_W,
        )
        draw.ellipse(
            [ox + source["cx"] - SOURCE_R, oy + source["cy"] - SOURCE_R,
             ox + source["cx"] + SOURCE_R, oy + source["cy"] + SOURCE_R],
            fill=colours["source"],
        )
        draw.text((ox + PANEL // 2 - 2, oy + PANEL + 8), str(index + 1), fill=LABEL_RGB)
    return image
