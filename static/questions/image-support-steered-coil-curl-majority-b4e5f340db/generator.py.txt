"""Renderer for coil_curl_majority.

One coloured hose is drawn as a single connected orthogonal route on a fixed
square peg lattice.  The walk starts at the end capped by a solid block and the
decision is whether the traced walk contains more left turns or more right
turns.  Grey decoy routes are declared decoration.

Analytic gold reads exactly two scene fields: ``path`` and ``start_at_head``.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
PITCH = 56
ORIGIN_X = 40
ORIGIN_Y = 40
GRID = 8                      # lattice points per axis (indices 0..7)
STROKE_W = 7                  # hose width in canvas px
BLOCK = 21                    # side of the solid cap block at the walk start
BACKGROUND = (250, 250, 248)
DECOY_COLOR = (145, 145, 145)
CORD_COLORS = (
    (200, 30, 40),
    (215, 105, 20),
    (30, 140, 60),
    (35, 80, 190),
    (120, 45, 160),
    (10, 130, 140),
)
QUARANTINE_MARGIN = 4.0
TALLY_UNIT = 5.0              # canvas px credited per unit of tally gap

_LEFT = "left"
_RIGHT = "right"


# --------------------------------------------------------------------------
# lattice / route helpers (no scene labels are produced here)
# --------------------------------------------------------------------------
def _px(point):
    i, j = point
    return (ORIGIN_X + PITCH * i, ORIGIN_Y + PITCH * j)


def _compress(path):
    """Collapse a unit-step lattice walk into maximal straight segments."""
    pts = [tuple(p) for p in path]
    out = [pts[0]]
    for k in range(1, len(pts) - 1):
        a, b, c = pts[k - 1], pts[k], pts[k + 1]
        if (b[0] - a[0], b[1] - a[1]) != (c[0] - b[0], c[1] - b[1]):
            out.append(b)
    out.append(pts[-1])
    return out


def _dirs(corners):
    ds = []
    for a, b in zip(corners, corners[1:]):
        dx, dy = b[0] - a[0], b[1] - a[1]
        n = max(abs(dx), abs(dy))
        ds.append((dx // n, dy // n))
    return ds


def _turn_side(prev_d, next_d):
    """Left as seen in the picture for a walker travelling ``prev_d``."""
    return _LEFT if next_d == (prev_d[1], -prev_d[0]) else _RIGHT


def _turns(path, start_at_head):
    """Ordered turn sides plus the lattice corner where each turn happens."""
    pts = [tuple(p) for p in path]
    if not start_at_head:
        pts = pts[::-1]
    corners = _compress(pts)
    ds = _dirs(corners)
    sides = []
    for k in range(1, len(ds)):
        sides.append((_turn_side(ds[k - 1], ds[k]), corners[k]))
    return sides


def _tally(sides):
    left = sum(1 for s, _ in sides if s == _LEFT)
    return left, len(sides) - left


def _majority(sides):
    left, right = _tally(sides)
    if left == right:
        return None
    return _LEFT if left > right else _RIGHT


# --------------------------------------------------------------------------
# required interface
# --------------------------------------------------------------------------
def analytic_gold(scene: dict) -> str:
    """Decision from the latent scene: reads only ``path`` and ``start_at_head``."""
    sides = _turns(scene["path"], scene["start_at_head"])
    left, right = _tally(sides)
    return _LEFT if left >= right else _RIGHT


def _seg_rects(path):
    corners = _compress([tuple(p) for p in path])
    h = STROKE_W / 2.0
    rects = []
    for a, b in zip(corners, corners[1:]):
        ax, ay = _px(a)
        bx, by = _px(b)
        rects.append((min(ax, bx) - h, min(ay, by) - h, max(ax, bx) + h, max(ay, by) + h))
    return rects


def _rect_gap(p, q):
    dx = max(0.0, max(p[0], q[0]) - min(p[2], q[2]))
    dy = max(0.0, max(p[1], q[1]) - min(p[3], q[3]))
    if dx <= 0.0 and dy <= 0.0:
        return 0.0
    return math.hypot(dx, dy)


def margin(scene: dict) -> float:
    """Composition minimum over the tally gap and every visual clearance."""
    sides = _turns(scene["path"], scene["start_at_head"])
    left, right = _tally(sides)
    parts = [TALLY_UNIT * abs(left - right)]

    rects = _seg_rects(scene["path"])
    for a in range(len(rects)):
        for b in range(a + 2, len(rects)):
            parts.append(_rect_gap(rects[a], rects[b]))
    for decoy in scene["decoys"]:
        for dr in _seg_rects(decoy["path"]):
            for cr in rects:
                parts.append(_rect_gap(cr, dr))
    parts.append(float(BLOCK - STROKE_W))
    return round(min(parts), 4)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    """Pixel-identical re-descriptions of the same physical hose."""
    out = []

    reversed_scene = dict(scene)
    reversed_scene["path"] = [list(p) for p in reversed(scene["path"])]
    reversed_scene["start_at_head"] = not scene["start_at_head"]
    out.append(("reverse_route_storage", reversed_scene))

    if len(scene["decoys"]) > 1:
        rotated = dict(scene)
        rotated["decoys"] = scene["decoys"][1:] + scene["decoys"][:1]
        out.append(("permute_decoy_order", rotated))
    return out


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for decoy in sorted(scene["decoys"], key=lambda d: tuple(map(tuple, d["path"]))):
        _draw_route(draw, decoy["path"], decoy["start_at_head"], DECOY_COLOR)
    _draw_route(draw, scene["path"], scene["start_at_head"], tuple(scene["color"]))
    return img


def _draw_route(draw, path, start_at_head, color):
    for rect in sorted(_seg_rects(path)):
        x0, y0, x1, y1 = rect
        draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=color)
    pts = [tuple(p) for p in path]
    cap = pts[0] if start_at_head else pts[-1]
    cx, cy = _px(cap)
    h = BLOCK / 2.0
    draw.rectangle([cx - h, cy - h, cx + h - 1, cy + h - 1], fill=color)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _walk(rng, steps):
    """Random vertex-simple unit-step walk on the lattice."""
    for _ in range(60):
        cur = (rng.randrange(GRID), rng.randrange(GRID))
        seen = {cur}
        pts = [cur]
        ok = True
        for _ in range(steps):
            options = []
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nxt = (cur[0] + dx, cur[1] + dy)
                if 0 <= nxt[0] < GRID and 0 <= nxt[1] < GRID and nxt not in seen:
                    options.append(nxt)
            if not options:
                ok = False
                break
            cur = rng.choice(options)
            seen.add(cur)
            pts.append(cur)
        if ok:
            return pts
    return None


def _mirror(path):
    return [(GRID - 1 - p[0], p[1]) for p in path]


def _half_disagreements(sides, gold):
    mid = GRID // 2
    halves = (
        [s for s, c in sides if c[0] < mid],
        [s for s, c in sides if c[0] >= mid],
        [s for s, c in sides if c[1] < mid],
        [s for s, c in sides if c[1] >= mid],
    )
    bad = 0
    for part in halves:
        left = part.count(_LEFT)
        right = len(part) - left
        local = None if left == right else (_LEFT if left > right else _RIGHT)
        if local != gold:
            bad += 1
    return bad


def _midline_crossings(path):
    """Unit steps of the route that cross the canvas midlines, per axis."""
    mid = GRID // 2
    across_x = across_y = 0
    for a, b in zip(path, path[1:]):
        if (a[0] < mid) != (b[0] < mid):
            across_x += 1
        if (a[1] < mid) != (b[1] < mid):
            across_y += 1
    return across_x, across_y


def _corner_span(sides):
    if not sides:
        return (0, 0)
    xs = [c[0] for _, c in sides]
    ys = [c[1] for _, c in sides]
    return (max(xs) - min(xs), max(ys) - min(ys))


def _candidate_route(rng):
    pts = _walk(rng, rng.randint(13, 18))
    if pts is None:
        return None
    start_at_head = bool(rng.getrandbits(1))
    sides = _turns(pts, start_at_head)
    if not 6 <= len(sides) <= 11:
        return None
    left, right = _tally(sides)
    gap = abs(left - right)
    if not 1 <= gap <= 3:
        return None
    span = _corner_span(sides)
    if min(span) < 4 or max(span) < 5:
        return None
    # the route must re-cross both canvas midlines, so that erasing any half of
    # the picture shatters it into several fragments instead of a shorter route
    crossings = _midline_crossings(pts)
    if min(crossings) < 2:
        return None
    gold = _LEFT if left > right else _RIGHT
    if _half_disagreements(sides, gold) < 2:
        return None
    first = sides[0][0]
    head = sides[: (len(sides) + 1) // 2]
    prefix = _majority(head)
    if first == gold and prefix == gold:
        variant = "control"
    elif first != gold and prefix != gold:
        variant = "counterfactual"
    else:
        return None
    return {
        "path": pts,
        "start_at_head": start_at_head,
        "gold": gold,
        "gap": gap,
        "variant": variant,
        "prior": first,
        "turns": len(sides),
    }


def _pad(points):
    out = set()
    for p in points:
        for dx in (-1, 0, 1):
            for dy in (-1, 0, 1):
                out.add((p[0] + dx, p[1] + dy))
    return out


def _decoys(rng, blocked):
    out = []
    for _ in range(rng.randint(2, 3)):
        for _ in range(60):
            pts = _walk(rng, rng.randint(5, 10))
            if pts is None or not blocked.isdisjoint(pts):
                continue
            out.append({"path": pts, "start_at_head": bool(rng.getrandbits(1))})
            blocked |= _pad(pts)
            break
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random((seed * 6364136223846793005 + 1442695040888963407)
                        % (2 ** 61))
    target_gold = _LEFT if seed % 2 == 0 else _RIGHT
    target_variant = "control" if (seed // 2) % 2 == 0 else "counterfactual"
    target_gap = ((seed // 4) % 2) + 1

    route = None
    for _ in range(60000):
        cand = _candidate_route(rng)
        if cand is None:
            continue
        if cand["variant"] != target_variant or cand["gap"] != target_gap:
            continue
        if cand["gold"] != target_gold:
            cand["path"] = _mirror(cand["path"])
            sides = _turns(cand["path"], cand["start_at_head"])
            cand["gold"] = _majority(sides)
            cand["prior"] = sides[0][0]
        if cand["gold"] != target_gold:
            continue
        route = cand
        break
    if route is None:                                    # pragma: no cover
        raise RuntimeError("route sampling failed for seed %d" % seed)

    decoys = _decoys(rng, _pad(route["path"]))

    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "path": [list(p) for p in route["path"]],
        "start_at_head": route["start_at_head"],
        "color": list(CORD_COLORS[rng.randrange(len(CORD_COLORS))]),
        "decoys": [{"path": [list(p) for p in d["path"]],
                    "start_at_head": d["start_at_head"]} for d in decoys],
        "variant": route["variant"],
        "canonical_prior_answer": route["prior"],
        "turn_count": route["turns"],
        "seed": int(seed),
    }
    return scene
