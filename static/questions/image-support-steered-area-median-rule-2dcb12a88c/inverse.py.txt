"""Independent pixel-only inverse arm for the area-median rule world.

This module receives a rasterised RGB image and nothing else. It never imports
the renderer, prompts, generator, verifier or any analytic-gold code, and it is
never handed a latent scene. It knows only the world's *public* palette, which
the question contract states in words (a grey band, four coloured vertical
rules, dark decoration on white paper).

Inverse procedure
-----------------
1. Segment the shaded band by nearest-palette-colour assignment.
2. Reduce the band mask to a per-column shaded-pixel profile (`pixel_count`).
3. Accumulate that profile and locate the column where the running total first
   reaches half of the whole shaded area - the area-median column. This step is
   what forces the measurement to depend on widely separated parts of the band:
   the running total at the middle is a function of every earlier column, and
   the total is a function of every column.
4. Segment each coloured rule as a tall, single-colour column stack and take its
   centre column.
5. Report the colour of the first rule at or right of the area-median column.

The renderer emits no antialiased pixels, so step 2 recovers exactly the same
integer column areas the analytic arm evaluates; agreement is exact, not
tolerance-based. Materially corrupted rasters (no band, or no rule at or right
of the median) raise or report ``none`` instead of guessing.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "area_median_rule-oracle-0.1.0"

# Public palette declared by the world contract (no scene access).
_BACKGROUND = (255, 255, 255)
_BAND = (154, 154, 154)
_DECOR = (25, 25, 25)
_RULES = {
    "red": (220, 50, 47),
    "green": (0, 140, 60),
    "blue": (30, 80, 220),
    "orange": (245, 150, 20),
}
_MATCH_TOLERANCE = 40.0
_RULE_HEIGHT_FRACTION = 0.5


class OracleMeasurementError(ValueError):
    """The raster carries no measurable shaded region."""


def _class_masks(array: np.ndarray) -> dict[str, np.ndarray]:
    """Nearest-colour labelling over the public palette."""
    names = ["background", "band", "decor", *sorted(_RULES)]
    palette = [_BACKGROUND, _BAND, _DECOR, *[_RULES[name] for name in sorted(_RULES)]]
    stack = np.stack(
        [
            np.linalg.norm(array - np.asarray(color, dtype=np.float32), axis=-1)
            for color in palette
        ],
        axis=-1,
    )
    labels = stack.argmin(axis=-1)
    nearest = np.take_along_axis(stack, labels[..., None], axis=-1)[..., 0]
    labels = np.where(nearest <= _MATCH_TOLERANCE, labels, 0)
    return {name: labels == index for index, name in enumerate(names)}


def column_profile(image: Image.Image) -> np.ndarray:
    """Shaded pixels per image column, measured from the raster alone."""
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    return _class_masks(array)["band"].sum(axis=0).astype(np.int64)


def area_median_column(image: Image.Image) -> int:
    """Column where the cumulative shaded area first reaches half the total."""
    profile = column_profile(image)
    total = int(profile.sum())
    if total <= 0:
        raise OracleMeasurementError("no shaded band pixels in the raster")
    return int(np.searchsorted(np.cumsum(profile), total / 2.0, side="left"))


def rule_centers(image: Image.Image) -> list[tuple[int, str]]:
    """(centre column, colour) of every tall single-colour rule in the raster."""
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    masks = _class_masks(array)
    height = array.shape[0]
    found: list[tuple[int, str]] = []
    for name in sorted(_RULES):
        counts = masks[name].sum(axis=0)
        tall = counts >= _RULE_HEIGHT_FRACTION * height
        run: list[int] = []
        for column in range(tall.shape[0] + 1):
            if column < tall.shape[0] and tall[column]:
                run.append(column)
                continue
            if run:
                found.append((int(round(sum(run) / len(run))), name))
                run = []
    found.sort()
    return found


def decision_from_image(image: Image.Image) -> str:
    """Colour of the first rule carrying at least half the shaded area at left."""
    median = area_median_column(image)
    for center, name in rule_centers(image):
        if center >= median:
            return name
    return "none"
