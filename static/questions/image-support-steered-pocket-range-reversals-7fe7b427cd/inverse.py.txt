"""Independent PNG-only inverse for the pocket range-reversal board."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_ANSWERS = ("A", "B", "C", "D")
_STRIP_COLORS = np.asarray(
    ((246, 177, 53), (238, 164, 72), (238, 184, 61)), dtype=np.int16
)
_TOKEN_COLORS = np.asarray(
    ((20, 178, 195), (38, 174, 143), (48, 156, 210)), dtype=np.int16
)


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y0, x0 = int(raw_y), int(raw_x)
        if visited[y0, x0]:
            continue
        queue: deque[tuple[int, int]] = deque([(y0, x0)])
        visited[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not visited[ny, nx]
                ):
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _nearest_color_mask(array: np.ndarray, colors: np.ndarray, tolerance: float = 8.0) -> np.ndarray:
    values = array.astype(np.int16)
    distances = np.linalg.norm(values[:, :, None, :] - colors[None, None, :, :], axis=3)
    return distances.min(axis=2) <= tolerance


def _card_bounds(array: np.ndarray) -> tuple[int, int, int, int]:
    # Border-median background recovery works after translations and rejects a
    # card with any materially erased area rather than imputing missing commands.
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    foreground = np.linalg.norm(array.astype(np.float32) - background, axis=2) > 24.0
    parts = _components(foreground, minimum=40)
    if not parts:
        raise ValueError("compact reversal board is missing")
    points = max(parts, key=len)
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    width, height = int(x1 - x0 + 1), int(y1 - y0 + 1)
    fill = len(points) / float(width * height)
    if not (118 <= width <= 146 and 118 <= height <= 146):
        raise ValueError("board extent is missing or damaged")
    if abs(width - height) > 2 or fill < 0.985:
        raise ValueError("board is not one intact filled square")
    return int(x0), int(y0), int(x1), int(y1)


def _lane_centers(x0: int, size: int) -> np.ndarray:
    return np.asarray([x0 + round((index + 0.5) * size / 4.0) for index in range(4)])


def _read_start(array: np.ndarray, bounds: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = bounds
    size = x1 - x0 + 1
    mask = _nearest_color_mask(array, _TOKEN_COLORS)
    parts = _components(mask, minimum=40)
    if len(parts) != 1:
        raise ValueError(f"expected one cyan start disk, found {len(parts)}")
    points = parts[0]
    cy, cx = points.mean(axis=0)
    expected_y = y0 + 0.16 * size
    if abs(cy - expected_y) > 0.09 * size:
        raise ValueError("start disk is outside the declared header row")
    lanes = _lane_centers(x0, size)
    distances = np.abs(lanes - cx)
    order = np.argsort(distances)
    if distances[order[0]] > size * 0.08 or distances[order[1]] - distances[order[0]] < size * 0.10:
        raise ValueError("start disk has no unique column")
    return int(order[0])


def _read_intervals(
    array: np.ndarray, bounds: tuple[int, int, int, int]
) -> list[tuple[int, int]]:
    x0, y0, x1, y1 = bounds
    size = x1 - x0 + 1
    mask = _nearest_color_mask(array, _STRIP_COLORS)
    parts = _components(mask, minimum=80)
    if len(parts) != 4:
        raise ValueError(f"expected four intact gold strips, found {len(parts)}")
    parts.sort(key=lambda points: float(points[:, 0].mean()))
    expected_rows = [y0 + fraction * size for fraction in (0.34, 0.49, 0.64, 0.79)]
    lanes = _lane_centers(x0, size)
    intervals: list[tuple[int, int]] = []
    for index, points in enumerate(parts):
        py0, px0 = points.min(axis=0)
        py1, px1 = points.max(axis=0)
        row_center = (float(py0) + float(py1)) / 2.0
        if abs(row_center - expected_rows[index]) > size * 0.055:
            raise ValueError("gold strip order or row placement is ambiguous")
        left_dist = np.abs(lanes - float(px0))
        right_dist = np.abs(lanes - float(px1))
        left = int(np.argmin(left_dist))
        right = int(np.argmin(right_dist))
        if left_dist[left] > size * 0.08 or right_dist[right] > size * 0.08:
            raise ValueError("a strip end cap has no unique slot")
        if not (0 <= left < right < 4):
            raise ValueError("a gold strip does not span a valid inclusive range")
        intervals.append((left, right))
    return intervals


def _apply(column: int, interval: tuple[int, int]) -> int:
    left, right = interval
    return left + right - column if left <= column <= right else column


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise ValueError("raster is too small")
    bounds = _card_bounds(array)
    column = _read_start(array, bounds)
    for interval in _read_intervals(array, bounds):
        column = _apply(column, interval)
    return _ANSWERS[column]
