"""Deterministic renderer and analytic arm for a compact range-reversal board."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 800
ANSWERS = ("A", "B", "C", "D")
ANCHORS = (250, 350, 450, 550)
CARD_SIZES = (126, 132, 138)

_BACKGROUND = (255, 255, 255)
_INK = (30, 38, 50)
_DISTRACTOR = (222, 226, 232)
_PALETTES = (
    ((184, 215, 236), (246, 177, 53), (20, 178, 195)),
    ((219, 198, 235), (238, 164, 72), (38, 174, 143)),
    ((184, 225, 204), (238, 184, 61), (48, 156, 210)),
)


def _apply_interval(column: int, interval: list[int] | tuple[int, int]) -> int:
    left, right = int(interval[0]), int(interval[1])
    return left + right - column if left <= column <= right else column


def _replay(column: int, intervals: list[list[int]] | tuple[tuple[int, int], ...]) -> int:
    for interval in intervals:
        column = _apply_interval(column, interval)
    return column


def _sample_intervals(group: int) -> list[list[int]]:
    """Use fixed total strip ink while sampling an order-live derangement."""
    rng = random.Random(0x5A17_2026 ^ (int(group) * 1_000_003))
    lengths = [2, 2, 3, 4]
    for _ in range(500):
        trial = []
        for length in lengths:
            left = rng.randrange(5 - length)
            trial.append([left, left + length - 1])
        rng.shuffle(trial)
        forward = tuple(_replay(value, trial) for value in range(4))
        reverse = tuple(_replay(value, list(reversed(trial))) for value in range(4))
        if all(forward[value] != value for value in range(4)) and forward != reverse:
            return trial
    raise RuntimeError("could not sample an order-live reversal program")


def sample_scene(seed: int) -> dict:
    """Balance each final column against identical programs and visual styles."""
    seed = int(seed)
    group, desired_final = divmod(seed, 4)
    intervals = _sample_intervals(group)
    start = desired_final
    for interval in reversed(intervals):
        start = _apply_interval(start, interval)
    return {
        "start_col": start,
        "intervals": intervals,
        "anchor_x": ANCHORS[group % len(ANCHORS)],
        "anchor_y": ANCHORS[(group // len(ANCHORS)) % len(ANCHORS)],
        "card_size": CARD_SIZES[(group // 3) % len(CARD_SIZES)],
        "palette_variant": (group // 2) % len(_PALETTES),
        "distractor_variant": group % 4,
    }


def _lane_centers(x0: int, size: int) -> list[int]:
    return [x0 + round((index + 0.5) * size / 4.0) for index in range(4)]


def _row_centers(y0: int, size: int) -> list[int]:
    return [y0 + round(fraction * size) for fraction in (0.34, 0.49, 0.64, 0.79)]


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _draw_distractors(draw: ImageDraw.ImageDraw, variant: int) -> None:
    layouts = (
        ((54, 64), (742, 724)),
        ((742, 64), (54, 724)),
        ((54, 64), (742, 64), (54, 724)),
        ((742, 724), (742, 64), (54, 724)),
    )
    for cx, cy in layouts[int(variant) % len(layouts)]:
        draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), outline=_DISTRACTOR, width=3)
        draw.line((cx - 4, cy, cx + 4, cy), fill=_DISTRACTOR, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), _BACKGROUND)
    draw = ImageDraw.Draw(image)
    size = int(scene["card_size"])
    cx, cy = int(scene["anchor_x"]), int(scene["anchor_y"])
    x0, y0 = cx - size // 2, cy - size // 2
    x1, y1 = x0 + size - 1, y0 + size - 1
    base, strip, token = _PALETTES[int(scene["palette_variant"])]
    _draw_distractors(draw, int(scene["distractor_variant"]))

    # Every pixel in the compact card is non-background. This makes damage to
    # the declared local machine observable instead of silently guessing.
    draw.rectangle((x0, y0, x1, y1), fill=base, outline=_INK, width=3)
    lanes = _lane_centers(x0, size)
    rows = _row_centers(y0, size)
    guide_top = y0 + round(0.24 * size)
    guide_bottom = y0 + round(0.86 * size)
    for x in lanes:
        draw.line((x, guide_top, x, guide_bottom), fill=_INK, width=1)

    bar_half_height = max(4, round(size * 0.032))
    cap_width = 3
    overhang = max(5, round(size * 0.042))
    for row_y, interval in zip(rows, scene["intervals"], strict=True):
        left, right = int(interval[0]), int(interval[1])
        xa, xb = lanes[left] - overhang, lanes[right] + overhang
        draw.rectangle(
            (xa, row_y - bar_half_height, xb, row_y + bar_half_height), fill=strip
        )
        draw.rectangle(
            (xa, row_y - bar_half_height, xa + cap_width - 1, row_y + bar_half_height),
            fill=_INK,
        )
        draw.rectangle(
            (xb - cap_width + 1, row_y - bar_half_height, xb, row_y + bar_half_height),
            fill=_INK,
        )

    marker_y = y0 + round(0.16 * size)
    marker_radius = max(6, round(size * 0.047))
    marker_x = lanes[int(scene["start_col"])]
    draw.ellipse(
        (
            marker_x - marker_radius,
            marker_y - marker_radius,
            marker_x + marker_radius,
            marker_y + marker_radius,
        ),
        fill=token,
        outline=_INK,
        width=2,
    )

    label_font = _font(max(9, round(size * 0.075)))
    label_y = y0 + round(0.88 * size)
    for label, x in zip(ANSWERS, lanes, strict=True):
        box = draw.textbbox((0, 0), label, font=label_font)
        width = box[2] - box[0]
        draw.text((x - width / 2, label_y), label, fill=_INK, font=label_font)
    return image


def analytic_gold(scene: dict) -> str:
    terminal = _replay(int(scene["start_col"]), scene["intervals"])
    return ANSWERS[terminal]


def margin(scene: dict) -> float:
    """Pixel runner-up gap for slot and ordered-row classification."""
    size = float(scene["card_size"])
    lane_gap = size / 4.0
    row_gap = 0.15 * size
    endpoint_clearance = lane_gap / 2.0 - max(5.0, round(size * 0.042))
    row_clearance = row_gap / 2.0 - max(4.0, round(size * 0.032))
    return round(min(endpoint_clearance, row_clearance), 6)


def is_quarantined(scene: dict) -> bool:
    palette = int(scene["palette_variant"])
    return (
        int(scene["card_size"]) < 118
        or palette not in range(len(_PALETTES))
        or margin(scene) < 4.0
        or not (0 <= int(scene["start_col"]) < 4)
        or len(scene["intervals"]) != 4
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
