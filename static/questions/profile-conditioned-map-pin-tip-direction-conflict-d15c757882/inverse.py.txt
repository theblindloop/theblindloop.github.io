"""Independent PNG-only inverse for the map-pin direction decision."""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image


@dataclass(frozen=True)
class PixelMeasurement:
    decision: str
    margin: float
    head_y: float
    top_extent: float
    bottom_extent: float
    blue_pixels: int


def _blue_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = arr[:, :, 0]
    green = arr[:, :, 1]
    blue = arr[:, :, 2]
    return (blue - red >= 38) & (blue - green >= 28) & (blue <= 235)


def measure_from_image(image: Image.Image) -> PixelMeasurement:
    """Recover the rounded-head center and compare its two vertical extents."""
    mask = _blue_mask(image)
    ys, _xs = np.nonzero(mask)
    if len(ys) < 1800:
        raise ValueError("abstain: no large blue map-pin component")

    height = mask.shape[0]
    row_counts = np.bincount(ys, minlength=height).astype(np.float64)
    peak = float(row_counts.max())
    if peak < 45:
        raise ValueError("abstain: blue silhouette is too narrow")
    head_rows = np.flatnonzero(row_counts >= 0.86 * peak)
    if len(head_rows) < 4:
        raise ValueError("abstain: rounded head cannot be localized")
    weights = row_counts[head_rows]
    head_y = float(np.average(head_rows, weights=weights))
    y_min = float(ys.min())
    y_max = float(ys.max())
    top_extent = head_y - y_min
    bottom_extent = y_max - head_y
    gap = abs(bottom_extent - top_extent)
    if gap < 10.0:
        raise ValueError(f"abstain: runner-up extent gap {gap:.2f}px is ambiguous")
    decision = "down" if bottom_extent > top_extent else "up"
    return PixelMeasurement(
        decision=decision,
        margin=gap,
        head_y=head_y,
        top_extent=top_extent,
        bottom_extent=bottom_extent,
        blue_pixels=int(mask.sum()),
    )


def decision_from_image(image: Image.Image) -> str:
    return measure_from_image(image).decision
