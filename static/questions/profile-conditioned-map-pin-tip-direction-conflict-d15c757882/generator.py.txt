"""Deterministic latent sampler and raster renderer for the map-pin world."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 320
SUPERSAMPLE = 3
HEAD_RADIUS = 48.0
HOLE_RADIUS = 17.0
TAIL_LENGTHS = (68.0, 78.0, 92.0, 108.0)
BLUE_PALETTE = (
    (34, 88, 178),
    (25, 108, 188),
    (50, 82, 166),
    (30, 122, 190),
)
BACKGROUNDS = (
    (250, 249, 246),
    (247, 250, 252),
    (252, 248, 244),
    (248, 252, 247),
)
DISTRACTOR_COLORS = (
    (224, 92, 66),
    (71, 166, 91),
    (155, 91, 185),
    (230, 154, 47),
)


def _style_rng(group: int) -> random.Random:
    return random.Random((group * 1_000_003) ^ 0x5A17C9)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one scene; consecutive even/odd seeds form a matched down/up pair."""
    seed = int(seed)
    group = seed // 2
    rng = _style_rng(group)
    direction = "down" if seed % 2 == 0 else "up"
    scale = round(rng.uniform(0.90, 1.08), 6)
    center = [round(rng.uniform(154.0, 166.0), 4), round(rng.uniform(154.0, 166.0), 4)]
    tail_length = TAIL_LENGTHS[group % len(TAIL_LENGTHS)]
    symbol_color = list(BLUE_PALETTE[group % len(BLUE_PALETTE)])
    background_color = list(BACKGROUNDS[(group // len(BLUE_PALETTE)) % len(BACKGROUNDS)])

    corner_centers = [(42, 45), (278, 48), (45, 274), (276, 276)]
    rng.shuffle(corner_centers)
    distractors = []
    for index, (x, y) in enumerate(corner_centers[:3]):
        distractors.append(
            {
                "x": float(x + rng.randint(-5, 5)),
                "y": float(y + rng.randint(-5, 5)),
                "radius": float(rng.randint(5, 8)),
                "color": list(DISTRACTOR_COLORS[(group + index) % len(DISTRACTOR_COLORS)]),
                "shape": "diamond" if (group + index) % 2 else "circle",
            }
        )

    return {
        "tip_direction": direction,
        "center": center,
        "scale": scale,
        "head_radius": HEAD_RADIUS,
        "tail_length": tail_length,
        "hole_radius": HOLE_RADIUS,
        "symbol_color": symbol_color,
        "background_color": background_color,
        "canvas_size": [CANVAS, CANVAS],
        "distractors": distractors,
    }


def _scaled_points(points: list[tuple[float, float]], ss: int) -> list[tuple[int, int]]:
    return [(round(x * ss), round(y * ss)) for x, y in points]


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the declared scene to one deterministic RGB raster."""
    width, height = (int(v) for v in scene["canvas_size"])
    background = tuple(int(v) for v in scene["background_color"])
    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    ss = SUPERSAMPLE

    for marker in scene["distractors"]:
        x = float(marker["x"])
        y = float(marker["y"])
        radius = float(marker["radius"])
        color = tuple(int(v) for v in marker["color"])
        if marker["shape"] == "circle":
            draw.ellipse(
                [
                    round((x - radius) * ss),
                    round((y - radius) * ss),
                    round((x + radius) * ss),
                    round((y + radius) * ss),
                ],
                fill=color,
            )
        else:
            draw.polygon(
                _scaled_points(
                    [(x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)],
                    ss,
                ),
                fill=color,
            )

    cx, cy = (float(v) for v in scene["center"])
    scale = float(scene["scale"])
    radius = float(scene["head_radius"]) * scale
    tail = float(scene["tail_length"]) * scale
    hole = float(scene["hole_radius"]) * scale
    sign = 1.0 if scene["tip_direction"] == "down" else -1.0
    blue = tuple(int(v) for v in scene["symbol_color"])

    base_y = cy + sign * radius * 0.12
    base_half = radius * 0.70
    tip_y = cy + sign * tail
    draw.polygon(
        _scaled_points([(cx - base_half, base_y), (cx + base_half, base_y), (cx, tip_y)], ss),
        fill=blue,
    )
    draw.ellipse(
        [
            round((cx - radius) * ss),
            round((cy - radius) * ss),
            round((cx + radius) * ss),
            round((cy + radius) * ss),
        ],
        fill=blue,
    )
    draw.ellipse(
        [
            round((cx - hole) * ss),
            round((cy - hole) * ss),
            round((cx + hole) * ss),
            round((cy + hole) * ss),
        ],
        fill=background,
    )

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    direction = str(scene["tip_direction"])
    if direction not in {"up", "down"}:
        raise ValueError(f"invalid tip direction {direction!r}")
    return direction


def margin(scene: dict[str, Any]) -> float:
    return (float(scene["tail_length"]) - float(scene["head_radius"])) * float(scene["scale"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 14.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_disjoint_distractor_order", twin)]
