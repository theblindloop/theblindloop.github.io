"""Independent PNG-only inverse arm for the chimeric-code-binding world.

Recovers, from the raster alone: the colour->shape bijection encoded by the
chimeric key at the top, each left slot's colour dot, each right slot's shape
glyph, and each slot's marker (solid vs hollow square); then binds each left
colour to the right slot whose shape the key pairs with it, compares the bound
markers, and counts the agreeing partners.  Uses only chromatic masks, pixel
registration, and glyph corner/fill statistics - no renderer geometry, scene,
prompt, or gold code, and no shared constants beyond the public colour table.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RGB = {
    "cyan": (0, 150, 170),
    "magenta": (200, 40, 120),
    "amber": (232, 158, 32),
    "lime": (40, 170, 90),
}
WORDS = ["zero", "one", "two", "three", "four"]
TOL = 45
MID_X = 430
MARK_OFFY = 40
LG_YS = [90, 190, 290, 390]
LX_COL_X = 170
LX_SHP_X = 350
L_COL_X = 380
R_SHP_X = 820
SLOT_YS = [560, 750, 940, 1130]
ATT_OFFY = -75


def _mask(arr: np.ndarray, color: tuple, x0: int, x1: int, y0: int, y1: int) -> np.ndarray:
    r, g, b = color
    view = arr[y0:y1, x0:x1]
    return (
        (np.abs(view[:, :, 0] - r) <= TOL)
        & (np.abs(view[:, :, 1] - g) <= TOL)
        & (np.abs(view[:, :, 2] - b) <= TOL)
    )


def _dominant_colour(arr: np.ndarray, cx: int, cy: int, r: int) -> str:
    x0, x1, y0, y1 = cx - r - 6, cx + r + 6, cy - r - 6, cy + r + 6
    best, best_n = None, -1
    for colour, rgb in RGB.items():
        n = int(_mask(arr, rgb, x0, x1, y0, y1).sum())
        if n > best_n:
            best, best_n = colour, n
    if best is None:
        raise ValueError(f"no colour dot at ({cx},{cy})")
    return best


def _classify_shape(arr: np.ndarray, cx: int, cy: int, half: int) -> str:
    x0, x1, y0, y1 = cx - half - 4, cx + half + 4, cy - half - 4, cy + half + 4
    box = arr[y0:y1, x0:x1]
    filled = (box[:, :, 0] < 160) & (box[:, :, 1] < 160) & (box[:, :, 2] < 160)
    h, w = filled.shape
    frac = float(filled.sum()) / float(h * w)
    corner = 5
    c00 = bool(filled[0:corner, 0:corner].any())
    c01 = bool(filled[0:corner, w - corner:w].any())
    c10 = bool(filled[h - corner:h, 0:corner].any())
    c11 = bool(filled[h - corner:h, w - corner:w].any())
    if c00 and c01 and c10 and c11:
        return "square"
    if c10 and c11 and not c00 and not c01:
        return "triangle"
    if frac > 0.5:
        return "disc"
    return "diamond"


def _marker_kind(arr: np.ndarray, cx: int, cy: int) -> str:
    y = int(round(cy))
    x = int(round(cx))
    mean = int(arr[y, x].mean())
    return "solid" if mean < 160 else "hollow"


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    legend_map: dict[str, str] = {}
    for i, y in enumerate(LG_YS):
        colour = _dominant_colour(arr, LX_COL_X, y, 28)
        shape = _classify_shape(arr, LX_SHP_X, y, 24)
        legend_map[colour] = shape
    if len(legend_map) != 4 or len(set(legend_map.values())) != 4:
        raise ValueError(f"invalid chimeric key {legend_map}")
    left_colour = [_dominant_colour(arr, L_COL_X, y + ATT_OFFY, 30) for y in SLOT_YS]
    left_marker = [_marker_kind(arr, L_COL_X, y + MARK_OFFY) for y in SLOT_YS]
    right_shape = [_classify_shape(arr, R_SHP_X, y + ATT_OFFY, 28) for y in SLOT_YS]
    right_marker = [_marker_kind(arr, R_SHP_X, y + MARK_OFFY) for y in SLOT_YS]
    right_index = {shape: j for j, shape in enumerate(right_shape)}
    if len(right_index) != 4:
        raise ValueError("duplicate right shapes")
    count = 0
    for i, colour in enumerate(left_colour):
        s = legend_map[colour]
        j = right_index[s]
        if left_marker[i] == right_marker[j]:
            count += 1
    return WORDS[count]
