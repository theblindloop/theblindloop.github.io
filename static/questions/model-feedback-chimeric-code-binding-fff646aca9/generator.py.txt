"""Deterministic renderer and analytic arm for the chimeric-code-binding world.

At the top a CHIMERIC KEY shows four rows; each row pairs a coloured dot (left)
with a shape glyph (right), so the key defines a colour <-> shape bijection.
Below sit two columns of four square slots, left and right.  Every LEFT slot
carries one coloured dot; every RIGHT slot carries one shape glyph.  Every slot
also carries a marker below the attribute: a SOLID dark square or a HOLLOW
square.

A left slot and a right slot are PARTNERS when the right slot's shape is the one
the key pairs with the left slot's colour (a cross-attribute correspondence read
through the chimeric key).  A pair is TRUE when the two bound slots carry the
same marker kind.

The essential decision program is a correspondence binding: read the key's
colour->shape mapping, bind each left colour to the right slot carrying the keyed
shape (ignoring vertical position), verify each bound pair against the marker
attribute, and count how many pairs agree.  The position-aligned reading
(left row i with right row i) is a decoy whenever the keyed binding crosses rows.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1100
HEIGHT = 1260
COLOURS = ["cyan", "magenta", "amber", "lime"]
SHAPES = ["disc", "square", "triangle", "diamond"]
KINDS = ["solid", "hollow"]
WORDS = ["zero", "one", "two", "three", "four"]
RGB = {
    "cyan": (0, 150, 170),
    "magenta": (200, 40, 120),
    "amber": (232, 158, 32),
    "lime": (40, 170, 90),
}
DARK = (45, 45, 45)
BACKGROUND = (250, 250, 250)
SLOT_LINE = (200, 200, 200)
SLOT_SIZE = 200
DOT_RADIUS = 30
GLYPH_SIZE = 56
MARKER_SIZE = 56
LG_YS = [90, 190, 290, 390]
LX_COL_X = 170
LX_SHP_X = 350
L_COL_X = 380
R_SHP_X = 820
SLOT_YS = [560, 750, 940, 1130]
ATT_OFFY = -75
MARK_OFFY = 40


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    legend_cols = list(COLOURS)
    rng.shuffle(legend_cols)
    legend_shapes = list(SHAPES)
    rng.shuffle(legend_shapes)
    left_colours = list(COLOURS)
    rng.shuffle(left_colours)
    right_shapes = list(SHAPES)
    rng.shuffle(right_shapes)
    for _ in range(40):
        keyed = [legend_shapes[legend_cols.index(c)] for c in left_colours]
        if keyed != right_shapes:
            break
        rng.shuffle(right_shapes)
    legend_map = {legend_cols[i]: legend_shapes[i] for i in range(4)}
    target = rng.randint(0, 4)
    partner = {left_colours[i]: right_shapes.index(legend_map[left_colours[i]]) for i in range(4)}
    pairs_match = [True, True, True, True]
    choose_false = rng.sample([0, 1, 2, 3], 4 - target)
    for i in choose_false:
        pairs_match[i] = False
    markers_left = [None] * 4
    markers_right = [None] * 4
    for i, colour in enumerate(left_colours):
        j = partner[colour]
        kind = rng.choice(KINDS)
        markers_left[i] = kind
        markers_right[j] = kind if pairs_match[i] else (
            "hollow" if kind == "solid" else "solid"
        )
    return {
        "canvas_size": [WIDTH, HEIGHT],
        "legend_cols": legend_cols,
        "legend_shapes": legend_shapes,
        "left_colours": left_colours,
        "right_shapes": right_shapes,
        "markers_left": markers_left,
        "markers_right": markers_right,
        "slot_size": SLOT_SIZE,
    }


def _match_count(scene: dict) -> int:
    legend_map = {scene["legend_cols"][i]: scene["legend_shapes"][i] for i in range(4)}
    right_index = {shape: j for j, shape in enumerate(scene["right_shapes"])}
    count = 0
    for i, colour in enumerate(scene["left_colours"]):
        s = legend_map[colour]
        j = right_index[s]
        if scene["markers_left"][i] == scene["markers_right"][j]:
            count += 1
    return count


def _draw_disc(draw: ImageDraw.ImageDraw, cx: int, cy: int, r: int, fill) -> None:
    draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=fill, outline=fill)


def _draw_shape(draw: ImageDraw.ImageDraw, cx: int, cy: int, kind: str, size: int) -> None:
    h = size // 2
    if kind == "disc":
        draw.ellipse((cx - h, cy - h, cx + h, cy + h), fill=DARK, outline=DARK)
    elif kind == "square":
        draw.rectangle((cx - h, cy - h, cx + h, cy + h), fill=DARK, outline=DARK)
    elif kind == "triangle":
        draw.polygon(
            [(cx, cy - h), (cx - h, cy + h), (cx + h, cy + h)], fill=DARK, outline=DARK
        )
    elif kind == "diamond":
        draw.polygon(
            [(cx, cy - h), (cx - h, cy), (cx, cy + h), (cx + h, cy)], fill=DARK, outline=DARK
        )
    else:
        raise ValueError(kind)


def _draw_marker(draw: ImageDraw.ImageDraw, cx: int, cy: int, kind: str, size: int) -> None:
    h = size // 2
    mx0, my0, mx1, my1 = cx - h, cy - h, cx + h, cy + h
    draw.rectangle((mx0, my0, mx1, my1), fill=BACKGROUND, outline=DARK, width=4)
    if kind == "solid":
        draw.rectangle((mx0 + 3, my0 + 3, mx1 - 3, my1 - 3), fill=DARK, outline=DARK)
    else:
        draw.rectangle((mx0 + 10, my0 + 10, mx1 - 10, my1 - 10), fill=DARK, outline=DARK)
        draw.rectangle(
            (mx0 + 14, my0 + 14, mx1 - 14, my1 - 14), fill=BACKGROUND, outline=BACKGROUND
        )


def _draw_slot(
    draw: ImageDraw.ImageDraw,
    cx: int,
    cy: int,
    left: bool,
    colour: str | None,
    shape: str | None,
    marker: str,
) -> None:
    x0, y0, x1, y1 = cx - SLOT_SIZE // 2, cy - SLOT_SIZE // 2, cx + SLOT_SIZE // 2, cy + SLOT_SIZE // 2
    draw.rectangle((x0, y0, x1, y1), outline=SLOT_LINE, width=4)
    ax, ay = cx, cy + ATT_OFFY
    if left:
        _draw_disc(draw, ax, ay, DOT_RADIUS, RGB[colour])
    else:
        _draw_shape(draw, ax, ay, shape, GLYPH_SIZE)
    _draw_marker(draw, cx, cy + MARK_OFFY, marker, MARKER_SIZE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for i, y in enumerate(LG_YS):
        _draw_disc(draw, LX_COL_X, y, 28, RGB[scene["legend_cols"][i]])
        _draw_shape(draw, LX_SHP_X, y, scene["legend_shapes"][i], 48)
    for i, y in enumerate(SLOT_YS):
        _draw_slot(draw, L_COL_X, y, True, scene["left_colours"][i], None, scene["markers_left"][i])
        _draw_slot(draw, R_SHP_X, y, False, None, scene["right_shapes"][i], scene["markers_right"][i])
    return image


def analytic_gold(scene: dict) -> str:
    return WORDS[_match_count(scene)]


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if sorted(scene["legend_cols"]) != sorted(COLOURS):
            return True
        if sorted(scene["legend_shapes"]) != sorted(SHAPES):
            return True
        if len(set(scene["legend_cols"])) != 4 or len(set(scene["legend_shapes"])) != 4:
            return True
        if sorted(scene["left_colours"]) != sorted(COLOURS):
            return True
        if sorted(scene["right_shapes"]) != sorted(SHAPES):
            return True
        if len(scene["markers_left"]) != 4 or len(scene["markers_right"]) != 4:
            return True
        if any(k not in KINDS for k in scene["markers_left"] + scene["markers_right"]):
            return True
        legend_map = {scene["legend_cols"][i]: scene["legend_shapes"][i] for i in range(4)}
        keyed = [legend_map[c] for c in scene["left_colours"]]
        if keyed == scene["right_shapes"]:
            return True
        return False
    except (KeyError, TypeError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list:
    return []
