"""State rendezvous: first time-coincidence of two moving identities.

The image shows FOUR ordered panels read left to right (labeled P1..P4). Each
panel draws a horizontal track split into SIX slots (columns) by thin divider
lines. Two uniquely-colored tokens -- red and green -- each occupy exactly one
slot in every panel, and each token keeps its color, so identity is carried by
color as it moves between slots across the panels.

The decision asks: at which panel (P2, P3, or P4) do the red and green tokens
FIRST occupy the same slot (the same column)? The two tokens always start (in
P1) in different slots, and the scene is constructed so their first
coincidence happens at exactly one of panels 2, 3, or 4. Recovering the answer
requires tracking both colored identities across the ordered panels and finding
the earliest panel in which their slot indices agree -- a temporal, relational
state-tracking decision unavailable from any single frame.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

NPANELS = 4
NSLOT = 6

PANEL_W = 210
PANEL_H = 130
PANEL_GAP = 40
MARGIN_LEFT = 30
MARGIN_TOP = 40
LBL_H = 18
TRACK_INSET = 12
TOKEN_R = 10
STACK_DY = 13

WIDTH = MARGIN_LEFT + NPANELS * PANEL_W + (NPANELS - 1) * PANEL_GAP + MARGIN_LEFT
HEIGHT = MARGIN_TOP + PANEL_H + LBL_H + 20

BG = (255, 255, 255)
BORDER = (120, 120, 120)
RED = (210, 40, 40)
GREEN = (40, 170, 40)

SLOT_W = (PANEL_W - 2 * TRACK_INSET) / NSLOT


def _panel_x0(p: int) -> int:
    return MARGIN_LEFT + p * (PANEL_W + PANEL_GAP)


def _slot_center_x(p: int, i: int) -> int:
    return int(_panel_x0(p) + TRACK_INSET + (i + 0.5) * SLOT_W)


def _track_top() -> int:
    return MARGIN_TOP


def _track_bottom() -> int:
    return MARGIN_TOP + PANEL_H


def _token_y(p: int) -> int:
    return _track_top() + PANEL_H // 2


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    t = rng.randint(1, 3)
    while True:
        a0 = rng.randrange(NSLOT)
        b0 = rng.randrange(NSLOT)
        if a0 != b0:
            break
    A = [0] * NPANELS
    B = [0] * NPANELS
    A[0], B[0] = a0, b0
    for k in range(1, NPANELS):
        if k < t:
            while True:
                a = rng.randrange(NSLOT)
                b = rng.randrange(NSLOT)
                if a != b:
                    break
            A[k], B[k] = a, b
        elif k == t:
            c = rng.randrange(NSLOT)
            A[k], B[k] = c, c
        else:
            A[k] = rng.randrange(NSLOT)
            B[k] = rng.randrange(NSLOT)
    return {"columnsA": A, "columnsB": B}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    A = scene["columnsA"]
    B = scene["columnsB"]

    for p in range(NPANELS):
        x0 = _panel_x0(p)
        top = _track_top()
        bottom = _track_bottom()
        for i in range(NSLOT + 1):
            dx = int(x0 + TRACK_INSET + i * SLOT_W)
            draw.line([(dx, top), (dx, bottom)], fill=BORDER, width=2)
        draw.line([(x0 + TRACK_INSET, top), (x0 + PANEL_W - TRACK_INSET, top)],
                  fill=BORDER, width=2)
        draw.line([(x0 + TRACK_INSET, bottom), (x0 + PANEL_W - TRACK_INSET, bottom)],
                  fill=BORDER, width=2)
        draw.text((x0 + 6, bottom + 4), f"P{p + 1}", fill=BORDER)

    cy = _token_y(0)
    for p in range(NPANELS):
        ca = _slot_center_x(p, A[p])
        cb = _slot_center_x(p, B[p])
        if ca == cb:
            draw.ellipse([ca - TOKEN_R, cy - STACK_DY - TOKEN_R,
                          ca + TOKEN_R, cy - STACK_DY + TOKEN_R], fill=RED)
            draw.ellipse([cb - TOKEN_R, cy + STACK_DY - TOKEN_R,
                          cb + TOKEN_R, cy + STACK_DY + TOKEN_R], fill=GREEN)
        else:
            draw.ellipse([ca - TOKEN_R, cy - TOKEN_R,
                          ca + TOKEN_R, cy + TOKEN_R], fill=RED)
            draw.ellipse([cb - TOKEN_R, cy - TOKEN_R,
                          cb + TOKEN_R, cy + TOKEN_R], fill=GREEN)
    return img


def analytic_gold(scene: dict) -> str:
    A = scene["columnsA"]
    B = scene["columnsB"]
    for k in range(NPANELS):
        if A[k] == B[k]:
            return f"panel {k + 1}"
    return "none"


def margin(scene: dict) -> float:
    return 10.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists: ``columnsA`` and
    ``columnsB`` directly fix the column (and hence the x-pixel position) of
    every red and green token drawn in every panel, so any change to either
    vector changes the raster. There is no hidden field and no
    reparametrization that reproduces identical pixels with a different gold."""
    return []
