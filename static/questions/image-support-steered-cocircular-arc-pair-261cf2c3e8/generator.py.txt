"""Renderer for the cocircular-arc-pair world.

Four thin dark arcs sit far apart on a light page, one near each of the four
page edges. Every arc has the same radius of curvature, the same angular span,
the same length, the same thickness and the same flat ink colour, so each arc is
a fragment of some circle of one common size and only the position of that
circle's centre distinguishes the fragments. Exactly two of the four arcs are
fragments of one and the same circle, i.e. their circles share a centre. The
decision is which two page edges those two arcs lie near.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "cocircular_arc_pair-0.1.0"

WIDTH = 780
HEIGHT = 780
RADIUS = 332.0
SPAN_DEG = 60.0
STROKE = 3
CENTRE_SPREAD = 42.0
QUARANTINE_THRESHOLD = 28.0
CLEAR_BAND_MIN = 58.0
BLURRED_BAND_MAX = 10.0

SLOT_NAMES = ("top", "bottom", "left", "right")
SLOT_DIRECTIONS = {
    "top": (0.0, -1.0),
    "bottom": (0.0, 1.0),
    "left": (-1.0, 0.0),
    "right": (1.0, 0.0),
}
PAIRS = (
    ("top", "bottom"),
    ("left", "right"),
    ("top", "left"),
    ("top", "right"),
    ("bottom", "left"),
    ("bottom", "right"),
)

BACKGROUNDS = ((255, 255, 255), (252, 251, 246), (247, 250, 253))
INK_COLORS = ((22, 24, 31), (17, 36, 60), (44, 19, 23), (19, 44, 29))
CLUTTER_COLORS = ((203, 203, 208), (208, 204, 196), (198, 206, 210))

ARC_CLEARANCE = 26.0
CLUTTER_CLEARANCE = 26.0
EDGE_MARGIN = 8.0


def _arc_points(arc: dict, radius: float, count: int = 41) -> list[tuple[float, float]]:
    half = math.radians(SPAN_DEG) / 2.0
    theta = arc["theta"]
    return [
        (
            arc["cx"] + radius * math.cos(theta - half + 2.0 * half * i / (count - 1)),
            arc["cy"] + radius * math.sin(theta - half + 2.0 * half * i / (count - 1)),
        )
        for i in range(count)
    ]


def _slot_of_point(px: float, py: float, width: int, height: int) -> str:
    dx = px - width / 2.0
    dy = py - height / 2.0
    if abs(dy) >= abs(dx):
        return "top" if dy < 0 else "bottom"
    return "left" if dx < 0 else "right"


def _pair_label(first: str, second: str) -> str:
    order = {name: index for index, name in enumerate(SLOT_NAMES)}
    low, high = sorted((first, second), key=lambda name: order[name])
    return f"{low} and {high}"


def _try_sample(rng: random.Random, pair: tuple[str, str], quarantine: bool) -> dict | None:

    def _in_spread() -> tuple[float, float]:
        radial = CENTRE_SPREAD * math.sqrt(rng.random())
        angle = rng.uniform(0.0, 2.0 * math.pi)
        return (
            WIDTH / 2.0 + radial * math.cos(angle),
            HEIGHT / 2.0 + radial * math.sin(angle),
        )

    # All three distinct circle centres are drawn from one common distribution
    # about the page middle, so the shared centre is not marked out by where it
    # sits; only the coincidence itself distinguishes it.
    shared = _in_spread()
    others = [name for name in SLOT_NAMES if name not in pair]
    distinct = [_in_spread(), _in_spread()]
    if quarantine:
        near = rng.uniform(4.0, BLURRED_BAND_MAX - 2.0)
        angle = rng.uniform(0.0, 2.0 * math.pi)
        distinct[rng.randrange(2)] = (
            shared[0] + near * math.cos(angle),
            shared[1] + near * math.sin(angle),
        )
    else:
        gaps = [
            math.dist(shared, distinct[0]),
            math.dist(shared, distinct[1]),
            math.dist(distinct[0], distinct[1]),
        ]
        if min(gaps) < CLEAR_BAND_MIN:
            return None
    rng.shuffle(distinct)

    centres: dict[str, tuple[float, float]] = {pair[0]: shared, pair[1]: shared}
    for name, centre in zip(others, distinct):
        centres[name] = centre

    arcs = []
    for name in SLOT_NAMES:
        ux, uy = SLOT_DIRECTIONS[name]
        theta = math.atan2(uy, ux) + math.radians(rng.uniform(-11.0, 11.0))
        arcs.append(
            {
                "cx": round(centres[name][0], 4),
                "cy": round(centres[name][1], 4),
                "theta": round(theta, 6),
            }
        )

    points = [_arc_points(arc, RADIUS) for arc in arcs]
    for index, name in enumerate(SLOT_NAMES):
        mid = points[index][len(points[index]) // 2]
        if _slot_of_point(mid[0], mid[1], WIDTH, HEIGHT) != name:
            return None
        bar_x = sum(p[0] for p in points[index]) / len(points[index])
        bar_y = sum(p[1] for p in points[index]) / len(points[index])
        if _slot_of_point(bar_x, bar_y, WIDTH, HEIGHT) != name:
            return None
        if abs(abs(bar_y - HEIGHT / 2.0) - abs(bar_x - WIDTH / 2.0)) < 35.0:
            return None
        for px, py in points[index]:
            if not (
                EDGE_MARGIN + STROKE < px < WIDTH - EDGE_MARGIN - STROKE
                and EDGE_MARGIN + STROKE < py < HEIGHT - EDGE_MARGIN - STROKE
            ):
                return None
    for a in range(4):
        for b in range(a + 1, 4):
            for pa in points[a]:
                for pb in points[b]:
                    if math.dist(pa, pb) < ARC_CLEARANCE:
                        return None

    flat = [point for group in points for point in group]
    clutter = []
    attempts = 0
    wanted = rng.randint(5, 9)
    while len(clutter) < wanted and attempts < 400:
        attempts += 1
        blob = {
            "x": round(rng.uniform(40.0, WIDTH - 40.0), 3),
            "y": round(rng.uniform(40.0, HEIGHT - 40.0), 3),
            "r": round(rng.uniform(7.0, 16.0), 3),
        }
        if any(math.dist((blob["x"], blob["y"]), p) < CLUTTER_CLEARANCE + blob["r"] for p in flat):
            continue
        if any(
            math.dist((blob["x"], blob["y"]), (other["x"], other["y"]))
            < blob["r"] + other["r"] + 14.0
            for other in clutter
        ):
            continue
        clutter.append(blob)

    scene = {
        "arcs": arcs,
        "radius": RADIUS,
        "span_deg": SPAN_DEG,
        "stroke": STROKE,
        "width": WIDTH,
        "height": HEIGHT,
        "bg": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "ink": list(INK_COLORS[rng.randrange(len(INK_COLORS))]),
        "clutter_color": list(CLUTTER_COLORS[rng.randrange(len(CLUTTER_COLORS))]),
        "clutter": clutter,
    }
    gap = margin(scene)
    if quarantine:
        if gap > BLURRED_BAND_MAX:
            return None
    elif gap < CLEAR_BAND_MIN:
        return None
    return scene


def sample_scene(seed: int) -> dict:
    """Sample a scene deterministically from `seed`.

    The answer class is counterbalanced by a seven-cycle over the six unordered
    edge pairs plus one near-boundary (quarantined) slot, so consecutive scene
    seeds sweep every class in turn and rejection sampling cannot bias which
    pair of arcs is co-circular.
    """
    slot = int(seed) % 7
    quarantine = slot == 6
    pair = PAIRS[slot % len(PAIRS)]
    for attempt in range(4000):
        rng = random.Random((int(seed) * 7919 + attempt * 104729) & 0x7FFFFFFF)
        scene = _try_sample(rng, pair, quarantine)
        if scene is not None:
            return scene
    raise RuntimeError(f"no admissible scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Rasterise the scene; every arc is one flat ink colour."""
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    clutter_color = tuple(scene["clutter_color"])
    for blob in scene["clutter"]:
        draw.ellipse(
            [
                blob["x"] - blob["r"],
                blob["y"] - blob["r"],
                blob["x"] + blob["r"],
                blob["y"] + blob["r"],
            ],
            fill=clutter_color,
        )
    ink = tuple(scene["ink"])
    radius = float(scene["radius"])
    half = float(scene["span_deg"]) / 2.0
    for arc in scene["arcs"]:
        box = [
            arc["cx"] - radius,
            arc["cy"] - radius,
            arc["cx"] + radius,
            arc["cy"] + radius,
        ]
        start = math.degrees(arc["theta"]) - half
        draw.arc(box, start=start, end=start + 2.0 * half, fill=ink, width=int(scene["stroke"]))
    return image


def _centres_and_slots(scene: dict) -> tuple[list[tuple[float, float]], list[str]]:
    radius = float(scene["radius"])
    centres = []
    slots = []
    for arc in scene["arcs"]:
        centres.append((float(arc["cx"]), float(arc["cy"])))
        mx = arc["cx"] + radius * math.cos(arc["theta"])
        my = arc["cy"] + radius * math.sin(arc["theta"])
        slots.append(_slot_of_point(mx, my, int(scene["width"]), int(scene["height"])))
    return centres, slots


def _ranked_pairs(scene: dict) -> list[tuple[float, int, int]]:
    centres, _ = _centres_and_slots(scene)
    ranked = [
        (math.dist(centres[a], centres[b]), a, b)
        for a in range(len(centres))
        for b in range(a + 1, len(centres))
    ]
    ranked.sort()
    return ranked


def margin(scene: dict) -> float:
    """Gap between the best and the runner-up co-circular pair, in pixels."""
    ranked = _ranked_pairs(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def analytic_gold(scene: dict) -> str:
    """The pair of page edges whose two arcs share one circle."""
    if is_quarantined(scene):
        return "unclear"
    _, slots = _centres_and_slots(scene)
    if len(set(slots)) != len(slots):
        return "unclear"
    ranked = _ranked_pairs(scene)
    _, a, b = ranked[0]
    return _pair_label(slots[a], slots[b])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the same raster."""
    transforms: list[tuple[str, dict]] = []
    arcs = list(scene["arcs"])

    rotated = dict(scene)
    rotated["arcs"] = arcs[1:] + arcs[:1]
    transforms.append(("cyclic_arc_permutation", rotated))

    reversed_arcs = dict(scene)
    reversed_arcs["arcs"] = list(reversed(arcs))
    transforms.append(("reversed_arc_order", reversed_arcs))

    reordered = dict(scene)
    reordered["clutter"] = list(reversed(scene["clutter"]))
    transforms.append(("reversed_clutter_order", reordered))
    return transforms
