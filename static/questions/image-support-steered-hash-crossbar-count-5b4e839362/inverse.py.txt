"""Independent image-only inverse for connected-glyph crossbar count."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        area = 0
        x0 = x1 = x
        y0 = y1 = y
        while queue:
            cy, cx = queue.popleft()
            area += 1
            x0, x1 = min(x0, cx), max(x1, cx)
            y0, y1 = min(y0, cy), max(y1, cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((area, x0, y0, x1, y1))
    return output


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    output = []
    start = previous = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value != previous + 1:
            output.append((start, previous))
            start = value
        previous = value
    output.append((start, previous))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[0] < 240 or array.shape[1] < 280:
        raise ValueError("raster too small")
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low
    # The glyph is the only strongly chromatic component; neutral marks cannot enter this mask.
    colored = (chroma > 58) & (low < 190) & (high < 225)
    components = [item for item in _components(colored) if item[0] >= 900]
    if len(components) != 1:
        raise ValueError(f"expected one substantial colored glyph, found {len(components)}")
    area, x0, y0, x1, y1 = components[0]
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (68 <= width <= 92 and 66 <= height <= 90 and 0.65 <= width / height <= 1.4):
        raise ValueError("colored glyph has invalid extent")
    if min(x0, y0, array.shape[1] - 1 - x1, array.shape[0] - 1 - y1) < 40:
        raise ValueError("glyph approaches raster boundary")

    local = colored[y0:y1 + 1, x0:x1 + 1]
    row_ink = local.sum(axis=1)
    # Full crossbars occupy most of the glyph width. Stem-only rows occupy under half.
    full_rows = row_ink >= max(42, int(round(width * 0.78)))
    runs = [(a, b) for a, b in _runs(full_rows) if b - a + 1 >= 5]
    count = len(runs)
    if count not in {1, 2, 3}:
        raise ValueError(f"ambiguous crossbar run count: {count}")
    if any((b - a + 1) > 23 for a, b in runs):
        raise ValueError("crossbar is implausibly thick")
    if len(runs) > 1 and min(right[0] - left[1] - 1 for left, right in zip(runs, runs[1:])) < 9:
        raise ValueError("crossbars are not visibly separated")
    # Two persistent vertical columns bind every bar into one hash-style glyph.
    col_ink = local.sum(axis=0)
    stem_runs = [(a, b) for a, b in _runs(col_ink >= int(round(height * 0.78))) if b - a + 1 >= 7]
    if len(stem_runs) != 2:
        raise ValueError("glyph does not contain two persistent upright stems")
    if area < 1200:
        raise ValueError("glyph support is too sparse")
    return "yes" if count == 2 else "no"
