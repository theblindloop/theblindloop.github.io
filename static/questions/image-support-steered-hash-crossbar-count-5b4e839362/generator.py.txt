"""Deterministic Pillow world for a compact connected hash-style glyph."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 384, 320
SUPERSAMPLE = 4
BACKGROUNDS = ((247, 245, 239), (241, 246, 248), (248, 243, 247))
PALETTES = ((24, 105, 178), (176, 52, 83), (24, 139, 104), (113, 65, 173))
DISTRACTOR = (203, 205, 205)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x48415348)
    # Exactly-two controls and noncanonical counterfactuals are balanced 1:1.
    if int(seed) % 2 == 0:
        count = 2
    else:
        count = 1 if (int(seed) // 2) % 2 == 0 else 3
    glyph_width = rng.randint(72, 88)
    glyph_height = rng.randint(70, 86)
    # Keep crossbar ink approximately constant across answer classes.
    thickness_base = {1: 19, 2: 10, 3: 7}[count]
    bar_thickness = thickness_base + rng.choice((-1, 0, 1))
    margin_x = 58
    margin_y = 54
    center_x = rng.randint(margin_x + glyph_width // 2, WIDTH - margin_x - glyph_width // 2)
    center_y = rng.randint(margin_y + glyph_height // 2, HEIGHT - margin_y - glyph_height // 2)
    distractors = []
    for _ in range(5):
        # Pale neutral marks are visually real but never diagnostic.
        x = rng.randint(24, WIDTH - 24)
        y = rng.randint(22, HEIGHT - 22)
        r = rng.randint(2, 4)
        distractors.append([x, y, r])
    return {
        "crossbar_count": count,
        "center_x": center_x,
        "center_y": center_y,
        "glyph_width": glyph_width,
        "glyph_height": glyph_height,
        "stem_width": rng.randint(9, 13),
        "bar_thickness": bar_thickness,
        "palette_index": rng.randrange(len(PALETTES)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "distractors": distractors,
    }


def _geometry(scene: dict) -> tuple[tuple[float, float, float, float], list[tuple[float, float, float, float]], list[tuple[float, float, float, float]]]:
    count = int(scene["crossbar_count"])
    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    width, height = float(scene["glyph_width"]), float(scene["glyph_height"])
    stem_width = float(scene["stem_width"])
    bar_thickness = float(scene["bar_thickness"])
    left, top = cx - width / 2.0, cy - height / 2.0
    right, bottom = cx + width / 2.0, cy + height / 2.0
    stem_offset = width * 0.23
    stems = [
        (cx - stem_offset - stem_width / 2.0, top, cx - stem_offset + stem_width / 2.0, bottom),
        (cx + stem_offset - stem_width / 2.0, top, cx + stem_offset + stem_width / 2.0, bottom),
    ]
    if count == 1:
        centers = [cy]
    elif count == 2:
        centers = [cy - height * 0.19, cy + height * 0.19]
    else:
        centers = [cy - height * 0.27, cy, cy + height * 0.27]
    bars = [(left, y - bar_thickness / 2.0, right, y + bar_thickness / 2.0) for y in centers]
    return (left, top, right, bottom), stems, bars


def render(scene: dict) -> Image.Image:
    scale = SUPERSAMPLE
    background = BACKGROUNDS[int(scene["background_index"]) % len(BACKGROUNDS)]
    color = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(round(value * scale) for value in values)

    # Same-color distractors make their latent ordering a real pixel-identical alias.
    for x, y, radius in scene["distractors"]:
        draw.ellipse(box((x - radius, y - radius, x + radius, y + radius)), fill=DISTRACTOR)

    _, stems, bars = _geometry(scene)
    for rectangle in stems:
        draw.rounded_rectangle(box(rectangle), radius=2 * scale, fill=color)
    for rectangle in bars:
        draw.rounded_rectangle(box(rectangle), radius=2 * scale, fill=color)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if int(scene["crossbar_count"]) == 2 else "no"


def margin(scene: dict) -> float:
    bbox, _, bars = _geometry(scene)
    top, bottom = bbox[1], bbox[3]
    intervals = sorted((bar[1], bar[3]) for bar in bars)
    clearances = [intervals[0][0] - top, bottom - intervals[-1][1]]
    clearances += [right[0] - left[1] for left, right in zip(intervals, intervals[1:])]
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    bbox, stems, bars = _geometry(scene)
    count = int(scene["crossbar_count"])
    return (
        count not in {1, 2, 3}
        or margin(scene) < 10.0
        or float(scene["bar_thickness"]) < 6.0
        or float(scene["stem_width"]) < 9.0
        or bbox[0] < 45.0 or bbox[1] < 42.0
        or bbox[2] > WIDTH - 45.0 or bbox[3] > HEIGHT - 42.0
        or len(stems) != 2 or len(bars) != count
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractors"] = list(reversed([list(item) for item in scene["distractors"]]))
    return [("reverse_identical_distractor_order", twin)]
