"""Independent raster-only inverse for the packet-interleave question world."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 9
CYAN = np.array((0, 154, 174), dtype=np.int16)
GOLD = np.array((238, 174, 25), dtype=np.int16)
DIRECTIONS = ("up", "right", "down", "left")


def _rotate(points: set[tuple[int, int]]) -> set[tuple[int, int]]:
    return {(c, N - 1 - r) for r, c in points}


def _templates() -> dict[str, np.ndarray]:
    up = {
        (1, 4),
        (2, 3), (2, 4), (2, 5),
        (3, 2), (3, 3), (3, 4), (3, 5), (3, 6),
        (4, 4), (5, 4), (6, 4), (7, 4), (8, 4),
    }
    sets = [up]
    for _ in range(3):
        sets.append(_rotate(sets[-1]))
    return {
        name: np.array([[int((r, c) in points) for c in range(N)] for r in range(N)])
        for name, points in zip(DIRECTIONS, sets)
    }


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: int = 5) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - target), axis=2) <= tolerance


def _recover_packet_tiles(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"))
    cyan = _color_mask(arr, CYAN)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 1000:
        raise ValueError("cyan grid fiducial is missing")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    span_x, span_y = x1 - x0, y1 - y0
    if abs(span_x - span_y) > 8 or not 300 <= span_x <= 380:
        raise ValueError("cyan grid bounds are malformed")
    cell = int(round(span_x / N))
    if abs(span_x - N * cell) > 3 or abs(span_y - N * cell) > 3:
        raise ValueError("cyan grid does not resolve into nine cells")

    gold = _color_mask(arr, GOLD)
    gy, gx = np.nonzero(gold)
    if len(gx) < span_y * 5:
        raise ValueError("gold packet divider is missing")
    divider_candidates = []
    for xx in range(max(0, x0), min(arr.shape[1], x1 + 1)):
        count = int(gold[max(0, y0 - 18): min(arr.shape[0], y1 + 19), xx].sum())
        if count > span_y // 2:
            divider_candidates.append(xx)
    if not divider_candidates:
        raise ValueError("full-height gold packet divider is missing")
    divider_x = float(np.mean(divider_candidates))
    split = int(round((divider_x - x0) / cell))
    if split != 5:
        raise ValueError(f"gold divider exposes an invalid {split}|{N - split} packet split")
    horizontal_candidates = []
    for yy in range(max(0, y0), min(arr.shape[0], y1 + 1)):
        count = int(gold[yy, max(0, x0 - 18): min(arr.shape[1], x1 + 19)].sum())
        if count > span_x // 2:
            horizontal_candidates.append(yy)
    if not horizontal_candidates:
        raise ValueError("full-width gold packet divider is missing")
    divider_y = float(np.mean(horizontal_candidates))
    row_split = int(round((divider_y - y0) / cell))
    if row_split != 5:
        raise ValueError(f"gold divider exposes an invalid {row_split}|{N - row_split} row split")

    tiles = np.zeros((N, N), dtype=np.uint8)
    for r in range(N):
        for c in range(N):
            cx = x0 + (c + 0.5) * cell
            cy = y0 + (r + 0.5) * cell
            radius = max(3, cell // 5)
            patch = arr[
                int(cy - radius): int(cy + radius + 1),
                int(cx - radius): int(cx + radius + 1),
            ].astype(np.float32)
            if patch.size == 0:
                raise ValueError("tile center falls outside the raster")
            luminance = 0.2126 * patch[:, :, 0] + 0.7152 * patch[:, :, 1] + 0.0722 * patch[:, :, 2]
            tiles[r, c] = int(float(np.median(luminance)) < 185.0)
    return tiles


def measurement_from_image(image: Image.Image) -> dict:
    packets = _recover_packet_tiles(image)
    columns_rebuilt = np.empty_like(packets)
    columns_rebuilt[:, 0::2] = packets[:, :5]
    columns_rebuilt[:, 1::2] = packets[:, 5:]
    rebuilt = np.empty_like(columns_rebuilt)
    rebuilt[0::2, :] = columns_rebuilt[:5, :]
    rebuilt[1::2, :] = columns_rebuilt[5:, :]
    scored = sorted(
        (int(np.count_nonzero(rebuilt != template)), name)
        for name, template in _templates().items()
    )
    if scored[0][0] == scored[1][0] or scored[1][0] - scored[0][0] < 6:
        raise ValueError("arrow classification is ambiguous")
    return {
        "decision": scored[0][1],
        "best_mismatch": scored[0][0],
        "runner_up_mismatch": scored[1][0],
        "margin": scored[1][0] - scored[0][0],
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measurement_from_image(image)["decision"])
