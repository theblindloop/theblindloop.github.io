"""Deterministic scene sampler and raster renderer for packet interleaving."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

N = 9
DIRECTIONS = ("up", "right", "down", "left")
CYAN = (0, 154, 174)
GOLD = (238, 174, 25)
LIGHT = (241, 245, 246)
INKS = (
    (30, 47, 70),
    (73, 45, 94),
    (30, 82, 70),
    (104, 42, 55),
)


def _rotate(points: Iterable[tuple[int, int]]) -> set[tuple[int, int]]:
    return {(c, N - 1 - r) for r, c in points}


def _template_sets() -> dict[str, set[tuple[int, int]]]:
    up = {
        (1, 4),
        (2, 3), (2, 4), (2, 5),
        (3, 2), (3, 3), (3, 4), (3, 5), (3, 6),
        (4, 4), (5, 4), (6, 4), (7, 4), (8, 4),
    }
    right = _rotate(up)
    down = _rotate(right)
    left = _rotate(down)
    return dict(zip(DIRECTIONS, (up, right, down, left)))


def arrow_templates() -> dict[str, list[list[int]]]:
    return {
        name: [[int((r, c) in points) for c in range(N)] for r in range(N)]
        for name, points in _template_sets().items()
    }


def _packetize(rows: list[list[int]]) -> list[list[int]]:
    column_packetized = [
        [row[c] for c in range(0, N, 2)] + [row[c] for c in range(1, N, 2)]
        for row in rows
    ]
    return column_packetized[0::2] + column_packetized[1::2]


def _interleave(scene: dict) -> list[list[int]]:
    packet_rows = scene["tiles"]
    column_rebuilt = []
    for row in packet_rows:
        left, right = row[:5], row[5:]
        out = []
        for index, value in enumerate(left):
            out.append(int(value))
            if index < len(right):
                out.append(int(right[index]))
        column_rebuilt.append(out)
    top, bottom = column_rebuilt[:5], column_rebuilt[5:]
    rebuilt = []
    for index, row in enumerate(top):
        rebuilt.append(row)
        if index < len(bottom):
            rebuilt.append(bottom[index])
    return rebuilt


def sample_scene(seed: int) -> dict:
    """Balance direction exactly across every four consecutive integer seeds."""
    direction = DIRECTIONS[seed % len(DIRECTIONS)]
    output = arrow_templates()[direction]
    style_band = seed // 4
    cell = (36, 38, 40)[style_band % 3]
    dx = (-18, 0, 18, 7)[style_band % 4]
    dy = (0, 12, -8, 5, -3)[style_band % 5]
    rng = random.Random(seed ^ 0x5EED17)
    # The jitter is independent of direction within each four-seed balance block.
    x = (640 - N * cell) // 2 + dx
    y = 74 + dy
    background = (250 - (style_band % 2) * 3,) * 3
    return {
        "tiles": _packetize(output),
        "x": x,
        "y": y,
        "cell": cell,
        "ink": list(INKS[style_band % len(INKS)]),
        "background": list(background),
    }


def render(scene: dict) -> Image.Image:
    tiles = scene["tiles"]
    x = int(scene["x"])
    y = int(scene["y"])
    cell = int(scene["cell"])
    ink = tuple(scene["ink"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (640, 570), background)
    draw = ImageDraw.Draw(image)

    # A small, answer-neutral shadow keeps the panel legible without encoding state.
    draw.rounded_rectangle(
        (x - 10, y - 10, x + N * cell + 12, y + N * cell + 12),
        radius=9,
        fill=(224, 229, 232),
    )
    for r in range(N):
        for c in range(N):
            x0, y0 = x + c * cell, y + r * cell
            fill = ink if int(tiles[r][c]) else LIGHT
            draw.rectangle((x0 + 3, y0 + 3, x0 + cell - 3, y0 + cell - 3), fill=fill)

    for i in range(N + 1):
        draw.line((x, y + i * cell, x + N * cell, y + i * cell), fill=CYAN, width=2)
        draw.line((x + i * cell, y, x + i * cell, y + N * cell), fill=CYAN, width=2)
    draw.rectangle((x, y, x + N * cell, y + N * cell), outline=CYAN, width=5)

    # These are the required, image-located packet fiducials. They replace the
    # fifth interior cyan boundaries but never cover a tile's sampled center.
    divider_x = x + 5 * cell
    divider_y = y + 5 * cell
    draw.rounded_rectangle(
        (divider_x - 4, y - 15, divider_x + 4, y + N * cell + 15),
        radius=4,
        fill=GOLD,
    )
    draw.rounded_rectangle(
        (x - 15, divider_y - 4, x + N * cell + 15, divider_y + 4),
        radius=4,
        fill=GOLD,
    )
    # Packet-size beads reinforce the visible 5|4 grouping without textual payload.
    bead_y = y - 29
    for c in range(5):
        cx = x + c * cell + cell // 2
        draw.ellipse((cx - 4, bead_y - 4, cx + 4, bead_y + 4), fill=GOLD)
    for c in range(5, N):
        cx = x + c * cell + cell // 2
        draw.ellipse((cx - 3, bead_y - 3, cx + 3, bead_y + 3), outline=(112, 132, 143), width=2)
    bead_x = x - 29
    for r in range(5):
        cy = y + r * cell + cell // 2
        draw.ellipse((bead_x - 4, cy - 4, bead_x + 4, cy + 4), fill=GOLD)
    for r in range(5, N):
        cy = y + r * cell + cell // 2
        draw.ellipse((bead_x - 3, cy - 3, bead_x + 3, cy + 3), outline=(112, 132, 143), width=2)
    return image


def analytic_gold(scene: dict) -> str:
    rebuilt = _interleave(scene)
    matches = [name for name, template in arrow_templates().items() if rebuilt == template]
    if len(matches) != 1:
        raise ValueError(f"rebuilt field has {len(matches)} exact arrow matches")
    return matches[0]


def margin(scene: dict) -> float:
    rebuilt = _interleave(scene)
    scores = []
    for name, template in arrow_templates().items():
        mismatch = sum(rebuilt[r][c] != template[r][c] for r in range(N) for c in range(N))
        scores.append((mismatch, name))
    scores.sort()
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
