"""Renderer, analytic gold and margin for barrier_track_replay.

A token sits in one cell of a drawn linear track of nine cells. Two dark red
walls are drawn inside two of the eight gaps between adjacent cells. Below the
track a row of cream cards is drawn; each card carries a green arrow (pointing
left or right) and a number of purple pips. The cards are applied in drawn
left-to-right order. A card moves the token one cell at a time in its arrow
direction, once per pip, but the token stops early if the next step would cross
a drawn wall or leave the track. The track after the cards is never drawn.

Nothing in this module is visible to oracle.py.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "barrier_track_replay-renderer-0.1.0"

CANVAS = (1000, 470)
N_CELLS = 9
CELL = 72
GAP = 20
PITCH = CELL + GAP
TRACK_X0 = (CANVAS[0] - (N_CELLS * CELL + (N_CELLS - 1) * GAP)) // 2
TRACK_Y0 = 84
WALL_W = 10
TOKEN_R = 20

N_CARDS = 5
CARD_W = 140
CARD_H = 156
CARD_GAP = 22
CARDS_X0 = (CANVAS[0] - (N_CARDS * CARD_W + (N_CARDS - 1) * CARD_GAP)) // 2
CARDS_Y0 = 262
PIP_R = 7
PIP_PITCH = 26

WHITE = (255, 255, 255)
CELL_FILL = (226, 230, 236)
CELL_EDGE = (60, 66, 76)
WALL = (176, 32, 48)
TOKEN = (24, 52, 140)
CARD = (250, 244, 224)
CARD_EDGE = (120, 112, 90)
ARROW = (26, 120, 72)
PIP = (128, 48, 160)
GUIDE = (170, 170, 170)

QUARANTINE_PX = 6.0


# ---------------------------------------------------------------- geometry --
def cell_center(index: int) -> tuple[float, float]:
    return (TRACK_X0 + index * PITCH + CELL / 2.0, TRACK_Y0 + CELL / 2.0)


def gap_center_x(gap: int) -> float:
    """Centre of the drawn gap between cell ``gap`` and cell ``gap + 1``."""
    return TRACK_X0 + gap * PITCH + CELL + GAP / 2.0


def card_box(slot: int) -> tuple[float, float, float, float]:
    x0 = CARDS_X0 + slot * (CARD_W + CARD_GAP)
    return (x0, CARDS_Y0, x0 + CARD_W, CARDS_Y0 + CARD_H)


# ------------------------------------------------------- drawn read-back ----
def _token_cell(scene: dict) -> int:
    """The cell whose centre is nearest the drawn token centre."""
    tx = cell_center(scene["token"]["cell"])[0] + scene["token"]["dx"]
    return min(range(N_CELLS), key=lambda i: abs(cell_center(i)[0] - tx))


def _wall_gaps(scene: dict) -> set[int]:
    """The gaps the drawn walls fall in, read back from drawn x only."""
    gaps = set()
    for wall in scene["walls"]:
        wx = gap_center_x(wall["gap"]) + wall["off"]
        gaps.add(min(range(N_CELLS - 1), key=lambda g: abs(gap_center_x(g) - wx)))
    return gaps


def _ordered_cards(scene: dict) -> list[dict]:
    """Cards in drawn left-to-right order, never in list order."""
    return sorted(scene["cards"], key=lambda c: card_box(c["slot"])[0] + c["dx"])


def _replay(scene: dict) -> tuple[int, int]:
    pos = _token_cell(scene)
    walls = _wall_gaps(scene)
    blocked = 0
    for card in _ordered_cards(scene):
        step = card["dir"]
        stopped = False
        for _ in range(card["steps"]):
            nxt = pos + step
            if nxt < 0 or nxt >= N_CELLS or min(pos, nxt) in walls:
                stopped = True
                break
            pos = nxt
        if stopped:
            blocked += 1
    return pos, blocked


def _naive_final(scene: dict) -> int:
    """The wall-blind shortcut: start plus the signed sum of the pips."""
    pos = _token_cell(scene)
    for card in _ordered_cards(scene):
        pos += card["dir"] * card["steps"]
    return pos


def analytic_gold(scene: dict) -> str:
    pos, blocked = _replay(scene)
    return f"{pos + 1}|{blocked}"


def margin(scene: dict) -> float:
    """Smallest drawn clearance that any read-back of the image depends on."""
    parts = []

    tx = cell_center(scene["token"]["cell"])[0] + scene["token"]["dx"]
    cx = cell_center(_token_cell(scene))[0]
    parts.append(PITCH / 2.0 - abs(tx - cx))
    parts.append(CELL / 2.0 - TOKEN_R - abs(scene["token"]["dy"]))

    # which gap a wall stands in is decided by the drawn bisector between the
    # two neighbouring gap centres; the wall is drawn clear of both cells
    for wall in scene["walls"]:
        wx = gap_center_x(wall["gap"]) + wall["off"]
        near = min(range(N_CELLS - 1), key=lambda g: abs(gap_center_x(g) - wx))
        parts.append(PITCH / 2.0 - abs(wx - gap_center_x(near)))

    boxes = sorted(
        (card_box(c["slot"])[0] + c["dx"], card_box(c["slot"])[2] + c["dx"])
        for c in scene["cards"]
    )
    for left, right in zip(boxes, boxes[1:]):
        parts.append(right[0] - left[1])

    for card in scene["cards"]:
        if card["steps"] > 1:
            parts.append(PIP_PITCH - 2 * PIP_R)

    return round(float(min(parts)), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


# ------------------------------------------------------------- sampling -----
def _draw_scene(rng: random.Random, tight: bool, near: bool = False) -> dict:
    walls = rng.sample(range(N_CELLS - 1), 2)
    scene = {
        "canvas": list(CANVAS),
        "n_cells": N_CELLS,
        "tight": tight,
        "token": {
            "cell": rng.randrange(N_CELLS),
            "dx": 0.0,
            "dy": float(rng.randint(-6, 6)),
        },
        "walls": [
            {"gap": g, "off": float(rng.randint(-3, 3))} for g in sorted(walls)
        ],
        "cards": [
            {
                "slot": s,
                "dir": rng.choice((-1, 1)),
                "steps": rng.randint(1, 4),
                "dx": float(rng.randint(-4, 4)),
                "dy": float(rng.randint(-4, 4)),
            }
            for s in range(N_CARDS)
        ],
    }
    if tight:
        side = rng.choice((-1, 1))
        scene["token"]["dx"] = float(side * (PITCH / 2.0 - rng.uniform(1.0, 5.0)))
        scene["token"]["cell"] = min(
            max(scene["token"]["cell"], 1), N_CELLS - 2
        )
    elif near:
        side = rng.choice((-1, 1))
        scene["token"]["dx"] = float(side * (PITCH / 2.0 - rng.uniform(6.5, 9.5)))
        scene["token"]["cell"] = min(max(scene["token"]["cell"], 1), N_CELLS - 2)
        scene["token"]["dy"] = float(rng.randint(-4, 4))
    else:
        scene["token"]["dx"] = float(rng.randint(-14, 14))
    return scene


def _acceptable(scene: dict) -> bool:
    """Shortcut blocks. Tight scenes only have to be genuinely near-boundary."""
    if scene["tight"]:
        return 0.5 <= margin(scene) < QUARANTINE_PX
    if margin(scene) < QUARANTINE_PX:
        return False
    pos, blocked = _replay(scene)
    if blocked == 0:
        return False  # a wall or an end must actually bite
    stripped = dict(scene)
    stripped["walls"] = []
    if _replay(stripped) == (pos, blocked):
        return False  # at least one drawn wall must change the replay
    if _naive_final(scene) == pos:
        return False  # the wall-blind pip sum must never give the answer
    if pos == _token_cell(scene):
        return False  # the drawn token position is never the answer
    last = _ordered_cards(scene)[-1]
    end = 0 if last["dir"] < 0 else N_CELLS - 1
    if pos == end:
        return False  # nor is the track end the last card points at
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    tight = (seed % 10) == 7
    near = (seed % 10) == 3
    for _ in range(4000):
        scene = _draw_scene(rng, tight, near)
        if _acceptable(scene):
            return scene
    raise RuntimeError(f"no acceptable scene for seed {seed}")


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-orderings of the unordered bags: pixel-identical, answer-preserving."""
    out: list[tuple[str, dict]] = []

    rev_cards = dict(scene)
    rev_cards["cards"] = list(reversed(scene["cards"]))
    out.append(("cards_reversed", rev_cards))

    rot_cards = dict(scene)
    rot_cards["cards"] = scene["cards"][2:] + scene["cards"][:2]
    out.append(("cards_rotated", rot_cards))

    rev_walls = dict(scene)
    rev_walls["walls"] = list(reversed(scene["walls"]))
    out.append(("walls_reversed", rev_walls))

    return out


# ------------------------------------------------------------- rendering ----
def _arrow(draw: ImageDraw.ImageDraw, cx: float, cy: float, direction: int) -> None:
    shaft_half, shaft_h = 30.0, 10.0
    head_w, head_h = 26.0, 34.0
    if direction > 0:
        draw.rectangle(
            [cx - shaft_half, cy - shaft_h / 2, cx + shaft_half - head_w, cy + shaft_h / 2],
            fill=ARROW,
        )
        draw.polygon(
            [
                (cx + shaft_half, cy),
                (cx + shaft_half - head_w, cy - head_h / 2),
                (cx + shaft_half - head_w, cy + head_h / 2),
            ],
            fill=ARROW,
        )
    else:
        draw.rectangle(
            [cx - shaft_half + head_w, cy - shaft_h / 2, cx + shaft_half, cy + shaft_h / 2],
            fill=ARROW,
        )
        draw.polygon(
            [
                (cx - shaft_half, cy),
                (cx - shaft_half + head_w, cy - head_h / 2),
                (cx - shaft_half + head_w, cy + head_h / 2),
            ],
            fill=ARROW,
        )


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), WHITE)
    draw = ImageDraw.Draw(img)

    for i in range(N_CELLS):
        x0 = TRACK_X0 + i * PITCH
        draw.rectangle([x0, TRACK_Y0, x0 + CELL, TRACK_Y0 + CELL], fill=CELL_FILL)
        draw.rectangle(
            [x0, TRACK_Y0, x0 + CELL, TRACK_Y0 + CELL], outline=CELL_EDGE, width=3
        )

    for wall in sorted(scene["walls"], key=lambda w: w["gap"]):
        wx = gap_center_x(wall["gap"]) + wall["off"]
        draw.rectangle(
            [wx - WALL_W / 2, TRACK_Y0 - 16, wx + WALL_W / 2, TRACK_Y0 + CELL + 16],
            fill=WALL,
        )

    tx = cell_center(scene["token"]["cell"])[0] + scene["token"]["dx"]
    ty = cell_center(scene["token"]["cell"])[1] + scene["token"]["dy"]
    draw.ellipse([tx - TOKEN_R, ty - TOKEN_R, tx + TOKEN_R, ty + TOKEN_R], fill=TOKEN)

    for card in sorted(scene["cards"], key=lambda c: c["slot"]):
        x0, y0, x1, y1 = card_box(card["slot"])
        x0 += card["dx"]
        x1 += card["dx"]
        y0 += card["dy"]
        y1 += card["dy"]
        draw.rectangle([x0, y0, x1, y1], fill=CARD)
        draw.rectangle([x0, y0, x1, y1], outline=CARD_EDGE, width=3)
        _arrow(draw, (x0 + x1) / 2.0, y0 + 56, card["dir"])
        n = card["steps"]
        px = (x0 + x1) / 2.0 - (n - 1) * PIP_PITCH / 2.0
        py = y1 - 42
        for k in range(n):
            cx = px + k * PIP_PITCH
            draw.ellipse([cx - PIP_R, py - PIP_R, cx + PIP_R, py + PIP_R], fill=PIP)

    guide_y = CARDS_Y0 + CARD_H + 34
    draw.rectangle([CARDS_X0, guide_y - 3, CARDS_X0 + 700, guide_y + 3], fill=GUIDE)
    draw.polygon(
        [
            (CARDS_X0 + 730, guide_y),
            (CARDS_X0 + 700, guide_y - 14),
            (CARDS_X0 + 700, guide_y + 14),
        ],
        fill=GUIDE,
    )
    return img
