"""Renderer for jar_fill_conservation.

Two open jars stand on a common table line. Each jar interior is ruled into a
lattice of identical square cells (the SAME cell size in both jars). Each jar is
filled from the bottom with a whole number of cell rows of one coloured liquid.

The answer is the pair (which jar holds the greater filled cell area, which jar's
liquid surface stands higher). Scenes are sampled so that width order and level
order always disagree, hence "the higher surface holds more" is true in exactly
half of the scenes: that half is the canonical variant of the taller-means-more
prior, the other half is the counterfactual variant.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "jar_fill_conservation-renderer-0.1.0"

WIDTH = 760
HEIGHT = 500
CELL = 34
JAR_ROWS = 8
BASE_Y = 430
CENTERS = (205, 555)

LIQUIDS = (
    (203, 60, 62),
    (46, 110, 196),
    (34, 132, 92),
    (156, 66, 172),
    (206, 118, 24),
)

WALL = (96, 96, 100)
LATTICE = (203, 203, 208)
FRAME = (168, 168, 172)
TABLE = (120, 120, 124)


def _combos() -> list[tuple[int, int, int, int]]:
    """(narrow_w, narrow_h, wide_w, wide_h) with narrow_w < wide_w, narrow_h > wide_h."""
    out = []
    for nw in range(2, 6):
        for ww in range(nw + 1, 6):
            for nh in range(1, JAR_ROWS + 1):
                for wh in range(1, nh):
                    if nw * nh != ww * wh:
                        out.append((nw, nh, ww, wh))
    return sorted(out)


COMBOS = _combos()


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    canonical = rng.random() < 0.5
    narrow_left = rng.random() < 0.5
    near = rng.random() < 0.22

    pool = []
    for nw, nh, ww, wh in COMBOS:
        narrow_wins = nw * nh > ww * wh
        if narrow_wins != canonical:
            continue
        m = abs(nw * nh - ww * wh)
        if near and m >= 2:
            continue
        if not near and m < 2:
            continue
        pool.append((nw, nh, ww, wh))
    if not pool:
        pool = [c for c in COMBOS if (c[0] * c[1] > c[2] * c[3]) == canonical]
    nw, nh, ww, wh = pool[rng.randrange(len(pool))]

    if narrow_left:
        left_w, left_h, right_w, right_h = nw, nh, ww, wh
    else:
        left_w, left_h, right_w, right_h = ww, wh, nw, nh

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "cell": CELL,
        "jar_rows": JAR_ROWS,
        "left_w": left_w,
        "left_h": left_h,
        "right_w": right_w,
        "right_h": right_h,
        "shift_x": rng.choice([-18, -9, 0, 9, 18]),
        "liquid_rgb": list(LIQUIDS[rng.randrange(len(LIQUIDS))]),
        "tick_offsets": sorted(rng.sample(range(6, 30), 3)),
    }


def _jar_box(scene: dict, w: int, cx: int) -> tuple[int, int, int, int]:
    cell = scene["cell"]
    x0 = cx - (w * cell) // 2
    y1 = BASE_Y
    y0 = y1 - scene["jar_rows"] * cell
    return x0, y0, x0 + w * cell, y1


def _darker(rgb: tuple[int, int, int]) -> tuple[int, int, int]:
    return tuple(max(0, int(c * 0.62)) for c in rgb)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), (255, 255, 255))
    d = ImageDraw.Draw(img)
    cell = scene["cell"]

    d.rectangle([14, 14, scene["width"] - 15, scene["height"] - 15], outline=FRAME, width=2)
    d.line([40, BASE_Y + 6, scene["width"] - 41, BASE_Y + 6], fill=TABLE, width=4)
    for off in scene["tick_offsets"]:
        x = 40 + off * 12
        d.line([x, BASE_Y + 10, x, BASE_Y + 24], fill=TABLE, width=2)
        d.line([scene["width"] - 41 - off * 12, BASE_Y + 10,
                scene["width"] - 41 - off * 12, BASE_Y + 24], fill=TABLE, width=2)

    liquid = tuple(scene["liquid_rgb"])
    seam = _darker(liquid)
    jars = (
        (scene["left_w"], scene["left_h"], CENTERS[0] + scene["shift_x"]),
        (scene["right_w"], scene["right_h"], CENTERS[1] + scene["shift_x"]),
    )
    for w, h, cx in jars:
        x0, y0, x1, y1 = _jar_box(scene, w, cx)
        for k in range(scene["jar_rows"] + 1):
            d.line([x0, y1 - k * cell, x1, y1 - k * cell], fill=LATTICE, width=1)
        for k in range(w + 1):
            d.line([x0 + k * cell, y0, x0 + k * cell, y1], fill=LATTICE, width=1)

        top = y1 - h * cell
        d.rectangle([x0, top, x1 - 1, y1 - 1], fill=liquid)
        for k in range(1, h):
            d.line([x0, y1 - k * cell, x1 - 1, y1 - k * cell], fill=seam, width=1)
        for k in range(1, w):
            d.line([x0 + k * cell, top, x0 + k * cell, y1 - 1], fill=seam, width=1)

        d.line([x0 - 3, y0 - 8, x0 - 3, y1 + 3], fill=WALL, width=4)
        d.line([x1 + 2, y0 - 8, x1 + 2, y1 + 3], fill=WALL, width=4)
        d.line([x0 - 5, y1 + 3, x1 + 4, y1 + 3], fill=WALL, width=4)
    return img


def _areas(scene: dict) -> tuple[int, int]:
    return scene["left_w"] * scene["left_h"], scene["right_w"] * scene["right_h"]


def analytic_gold(scene: dict) -> str:
    al, ar = _areas(scene)
    more = "left" if al > ar else "right"
    higher = "left" if scene["left_h"] > scene["right_h"] else "right"
    return f"more_{more}__higher_{higher}"


def margin(scene: dict) -> float:
    al, ar = _areas(scene)
    return float(abs(al - ar))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every field is rendered and gold is injective in pixels."""
    return []


# ---- perturbations used by verify.py (not latent aliases) -------------------

def repalette(scene: dict, k: int) -> dict:
    alt = dict(scene)
    cur = tuple(scene["liquid_rgb"])
    choices = [c for c in LIQUIDS if c != cur]
    alt["liquid_rgb"] = list(choices[k % len(choices)])
    return alt


def translate(scene: dict, dx: int) -> dict:
    alt = dict(scene)
    alt["shift_x"] = dx
    return alt


def permute_ticks(scene: dict, seed: int) -> dict:
    alt = dict(scene)
    rng = random.Random(seed ^ 0x5EED)
    alt["tick_offsets"] = sorted(rng.sample(range(6, 30), 3))
    return alt


def swap_sides(scene: dict) -> dict:
    alt = dict(scene)
    alt["left_w"], alt["right_w"] = scene["right_w"], scene["left_w"]
    alt["left_h"], alt["right_h"] = scene["right_h"], scene["left_h"]
    return alt


def expected_swapped_gold(decision: str) -> str:
    flip = {"left": "right", "right": "left"}
    more, higher = decision.split("__")
    return f"more_{flip[more[5:]]}__higher_{flip[higher[7:]]}"
