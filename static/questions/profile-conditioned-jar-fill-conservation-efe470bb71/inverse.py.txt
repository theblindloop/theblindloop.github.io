"""Pixel-only oracle for jar_fill_conservation.

Receives the final PNG and nothing else. It segments the two chromatic liquid
regions, compares their pixel areas and their surface heights, and reports the
composite decision. It imports no renderer, prompt, generator, verifier or gold
code and never sees latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "jar_fill_conservation-oracle-0.1.0"
ABSTAIN = "abstain"

CHROMA_MIN = 30
MIN_PIXELS = 400
AREA_REL_TOL = 0.01
LEVEL_MIN_PX = 10


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    out = []
    start = None
    for i, v in enumerate(flags):
        if v and start is None:
            start = i
        elif not v and start is not None:
            out.append((start, i))
            start = None
    if start is not None:
        out.append((start, len(flags)))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    mask = chroma >= CHROMA_MIN
    if int(mask.sum()) < 2 * MIN_PIXELS:
        return ABSTAIN

    cols = mask.any(axis=0)
    runs = [r for r in _runs(cols) if int(mask[:, r[0]:r[1]].sum()) >= MIN_PIXELS]
    if len(runs) != 2:
        return ABSTAIN

    stats = []
    for x0, x1 in runs:
        sub = mask[:, x0:x1]
        rows = np.nonzero(sub.any(axis=1))[0]
        stats.append((int(sub.sum()), int(rows[0])))
    (area_l, top_l), (area_r, top_r) = stats

    scale = max(area_l, area_r)
    if abs(area_l - area_r) <= AREA_REL_TOL * scale:
        return ABSTAIN
    if abs(top_l - top_r) < LEVEL_MIN_PX:
        return ABSTAIN

    more = "left" if area_l > area_r else "right"
    higher = "left" if top_l < top_r else "right"
    return f"more_{more}__higher_{higher}"
