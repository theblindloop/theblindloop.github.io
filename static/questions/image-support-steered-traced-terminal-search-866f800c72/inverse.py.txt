"""Independent pixel-only inverse for traced_terminal_search.

The inverse discovers foreground from the image border color, flood-fills the
component containing the gold marker, and independently scores every declared
terminal-color region by the fraction connected to that component.  It knows
neither scene coordinates nor renderer state.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GOLD = np.asarray((232, 174, 36), dtype=np.float32)
TERMINALS = {
    "red": np.asarray((207, 55, 67), dtype=np.float32),
    "blue": np.asarray((47, 105, 188), dtype=np.float32),
    "green": np.asarray((42, 145, 91), dtype=np.float32),
    "purple": np.asarray((139, 76, 181), dtype=np.float32),
}
COLOR_TOLERANCE = 42.0
FOREGROUND_DELTA = 34.0
MIN_START_PIXELS = 180
MIN_CONNECTED_TERMINAL_PIXELS = 180


def _border_background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    return np.median(border, axis=0)


def _flood(mask: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    reached = np.zeros_like(mask, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for y, x in seeds.tolist():
        if mask[y, x] and not reached[y, x]:
            reached[y, x] = True
            queue.append((int(y), int(x)))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                queue.append((ny, nx))
    return reached


def analyze_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")
    background = _border_background(array)
    foreground = np.linalg.norm(array - background[None, None, :], axis=2) > FOREGROUND_DELTA
    gold_mask = np.linalg.norm(array - GOLD[None, None, :], axis=2) < COLOR_TOLERANCE
    gold_seeds = np.argwhere(gold_mask & foreground)
    if len(gold_seeds) < MIN_START_PIXELS:
        raise ValueError("the gold START marker is missing or too damaged")
    reached = _flood(foreground, gold_seeds)

    scores: dict[str, float] = {}
    connected_pixels: dict[str, int] = {}
    visible_pixels: dict[str, int] = {}
    for name, prototype in TERMINALS.items():
        candidate = np.linalg.norm(array - prototype[None, None, :], axis=2) < COLOR_TOLERANCE
        visible = int(candidate.sum())
        connected = int((candidate & reached).sum())
        visible_pixels[name] = visible
        connected_pixels[name] = connected
        scores[name] = float(connected / visible) if visible else 0.0

    ordered = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    best_name, best_score = ordered[0]
    runner_up = ordered[1][1]
    if connected_pixels[best_name] < MIN_CONNECTED_TERMINAL_PIXELS:
        raise ValueError("no colored terminal remains connected to START")
    if best_score < 0.92 or best_score - runner_up < 0.80:
        raise ValueError("terminal connectivity is not uniquely recoverable")
    return {
        "decision": best_name,
        "margin": float(best_score - runner_up),
        "scores": scores,
        "connected_pixels": connected_pixels,
        "visible_pixels": visible_pixels,
    }


def decision_from_image(image: Image.Image) -> str:
    return str(analyze_image(image)["decision"])
