"""Analytic sampler and raster renderer for traced_terminal_search.

The latent answer is not stored.  The gold arm recomputes which terminal color
occupies the cable bearing the visibly rendered gold start socket.  Four cable
lanes are equally styled, have the same number of bends, and are separated by a
large visible background clearance.
"""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 247, 242)
INK = (31, 39, 48)
SOCKET = (139, 145, 151)
GOLD = (232, 174, 36)
TERMINAL_RGB = {
    "red": (207, 55, 67),
    "blue": (47, 105, 188),
    "green": (42, 145, 91),
    "purple": (139, 76, 181),
}
COLORS = ("red", "blue", "green", "purple")
LINE_WIDTH = 10
MARKER_RADIUS = 16
X_POINTS = (148, 184, 220, 256, 292, 328, 370)
LANE_CENTERS = (118, 210, 302, 394)


def sample_scene(seed: int) -> dict:
    """Return a deterministic, answer/lane-counterbalanced scene."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x5EEDB17)
    start_path = (seed // 4) % 4
    correct_color = COLORS[seed % 4]
    remaining = [color for color in COLORS if color != correct_color]
    rng.shuffle(remaining)
    terminal_colors: list[str] = []
    for lane in range(4):
        terminal_colors.append(correct_color if lane == start_path else remaining.pop())

    dx = (-4, 0, 4)[rng.randrange(3)]
    dy = (-3, 0, 3)[rng.randrange(3)]
    paths: list[list[list[int]]] = []
    offsets = (-12, -7, 0, 7, 12)
    for lane, center in enumerate(LANE_CENTERS):
        interior = [rng.choice(offsets) for _ in X_POINTS[1:-1]]
        # Force several visible bends without coupling their direction to gold.
        if len(set(interior)) < 2:
            interior[1] = -interior[0] if interior[0] else 12
        ys = [center + dy, *[center + dy + value for value in interior], center + dy]
        paths.append([[x + dx, y] for x, y in zip(X_POINTS, ys, strict=True)])

    return {
        "paths": paths,
        "start_path": start_path,
        "terminal_colors": terminal_colors,
        "width": CANVAS,
        "height": CANVAS,
        "background": list(BACKGROUND),
        "ink": list(INK),
        "socket_color": list(SOCKET),
        "gold_color": list(GOLD),
        "terminal_palette": {key: list(value) for key, value in TERMINAL_RGB.items()},
        "line_width": LINE_WIDTH,
        "marker_radius": MARKER_RADIUS,
    }


def _rgb(value: object) -> tuple[int, int, int]:
    if not isinstance(value, (list, tuple)) or len(value) != 3:
        raise ValueError("RGB value must have three channels")
    return tuple(int(channel) for channel in value)


def _validate_paths(paths: object) -> list[list[tuple[int, int]]]:
    if not isinstance(paths, list) or len(paths) != 4:
        raise ValueError("exactly four cable paths are required")
    result: list[list[tuple[int, int]]] = []
    for path in paths:
        if not isinstance(path, list) or len(path) < 4:
            raise ValueError("each cable needs at least four visible vertices")
        points: list[tuple[int, int]] = []
        for point in path:
            if not isinstance(point, (list, tuple)) or len(point) != 2:
                raise ValueError("cable points must be [x, y]")
            points.append((int(point[0]), int(point[1])))
        result.append(points)
    return result


def render(scene: dict) -> Image.Image:
    """Draw the final image with deterministic two-times antialiasing."""
    paths = _validate_paths(scene["paths"])
    start_path = int(scene["start_path"])
    terminal_colors = [str(value) for value in scene["terminal_colors"]]
    width, height = int(scene["width"]), int(scene["height"])
    background = _rgb(scene["background"])
    ink = _rgb(scene["ink"])
    socket = _rgb(scene["socket_color"])
    gold = _rgb(scene["gold_color"])
    palette = {str(key): _rgb(value) for key, value in scene["terminal_palette"].items()}
    line_width = int(scene["line_width"])
    radius = int(scene["marker_radius"])
    if not 0 <= start_path < 4 or sorted(terminal_colors) != sorted(COLORS):
        raise ValueError("invalid start path or terminal assignment")

    ss = 2
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    # A quiet field boundary groups the four candidates without touching them.
    draw.rounded_rectangle(
        (112 * ss, 78 * ss, 406 * ss, 434 * ss),
        radius=15 * ss,
        fill=(241, 240, 235),
        outline=(211, 210, 204),
        width=2 * ss,
    )
    for points in paths:
        scaled = [(x * ss, y * ss) for x, y in points]
        draw.line(scaled, fill=ink, width=line_width * ss, joint="curve")

    for index, points in enumerate(paths):
        left = min(points, key=lambda point: point[0])
        right = max(points, key=lambda point: point[0])
        left_color = gold if index == start_path else socket
        for (cx, cy), color in ((left, left_color), (right, palette[terminal_colors[index]])):
            draw.ellipse(
                ((cx - radius) * ss, (cy - radius) * ss,
                 (cx + radius) * ss, (cy + radius) * ss),
                fill=color,
                outline=ink,
                width=2 * ss,
            )
        if index == start_path:
            # A small dark S makes the designated start explicit without adding
            # another gold component for the inverse arm to disambiguate.
            cx, cy = left
            glyph = ("111", "100", "111", "001", "111")
            scale = 2 * ss
            ox, oy = (cx * ss - 3 * scale // 2, cy * ss - 5 * scale // 2)
            for row, bits in enumerate(glyph):
                for col, bit in enumerate(bits):
                    if bit == "1":
                        draw.rectangle(
                            (ox + col * scale, oy + row * scale,
                             ox + (col + 1) * scale - 1, oy + (row + 1) * scale - 1),
                            fill=ink,
                        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    start_path = int(scene["start_path"])
    terminal_colors = [str(value) for value in scene["terminal_colors"]]
    if not 0 <= start_path < len(terminal_colors):
        raise ValueError("start path is outside the terminal assignment")
    return terminal_colors[start_path]


def margin(scene: dict) -> float:
    """Runner-up gap for the declared binary connectivity score (1 versus 0)."""
    # Read the two label inputs so the value is a consequence of a valid visible
    # assignment, while retaining the natural normalized connectivity margin.
    start_path = int(scene["start_path"])
    terminal_colors = [str(value) for value in scene["terminal_colors"]]
    if not 0 <= start_path < 4 or sorted(terminal_colors) != sorted(COLORS):
        return 0.0
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        paths = _validate_paths(scene["paths"])
        start_path = int(scene["start_path"])
        colors = [str(value) for value in scene["terminal_colors"]]
        line_width = int(scene["line_width"])
    except (KeyError, TypeError, ValueError):
        return True
    if not 0 <= start_path < 4 or sorted(colors) != sorted(COLORS) or line_width < 7:
        return True
    # The sampler uses vertically separated lanes.  A 36 px vertex clearance is
    # far above the antialiased line width and keeps accidental joins observable.
    for left in range(4):
        for right in range(left + 1, 4):
            if min(abs(a[1] - b[1]) for a, b in zip(paths[left], paths[right], strict=True)) < 36:
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Permute storage slots while preserving every rendered object and label."""
    permutation = (1, 0, 3, 2)  # new slot -> old slot
    twin = copy.deepcopy(scene)
    twin["paths"] = [copy.deepcopy(scene["paths"][old]) for old in permutation]
    twin["terminal_colors"] = [scene["terminal_colors"][old] for old in permutation]
    twin["start_path"] = permutation.index(int(scene["start_path"]))
    return [("permute_cable_storage", twin)]
