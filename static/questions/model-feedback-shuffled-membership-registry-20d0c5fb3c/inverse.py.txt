"""Independent image-only inverse for Shuffled Membership Registry."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
DECISIONS = tuple("ABCDEF")
INK = np.array((35, 39, 48), dtype=np.int16)
START = np.array((200, 38, 145), dtype=np.int16)
TAG_COLORS = tuple(np.array(c, dtype=np.int16) for c in ((216, 58, 64), (38, 105, 205), (25, 151, 104), (232, 139, 32), (130, 75, 190), (18, 158, 174)))
CELL = 62
GRID_Y = 228
LEFT_X = 125
RIGHT_X = 755


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _read_board(arr: np.ndarray, x0: int) -> tuple[list[list[int]], list[int]] | None:
    matrix: list[list[int]] = []
    tags: list[int] = []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        tag_crop = arr[cy - 24:cy + 25, x0 - 74:x0 - 25]
        scores = [int(_mask(tag_crop, color).sum()) for color in TAG_COLORS]
        winners = [i for i, score in enumerate(scores) if score > 250]
        if len(winners) != 1:
            return None
        tags.append(winners[0])
        values = []
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            center = arr[cy - 10:cy + 11, cx - 10:cx + 11]
            values.append(1 if int(_mask(center, INK).sum()) > 300 else 0)
        matrix.append(values)
    if sorted(tags) != list(range(N)):
        return None
    if any(sum(row) != 3 for row in matrix):
        return None
    return matrix, tags


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {tag: row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (1280, 760):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        left = _read_board(arr, LEFT_X)
        right = _read_board(arr, RIGHT_X)
        if left is None or right is None:
            return "abstain"
        starts = []
        for col in range(N):
            cx = LEFT_X + col * CELL + CELL // 2
            crop = arr[158:206, cx - 28:cx + 29]
            starts.append(int(_mask(crop, START).sum()) > 900)
        if sum(starts) != 1:
            return "abstain"
        left_sigs = _signatures(*left)
        right_sigs = _signatures(*right)
        if len(set(left_sigs)) != N or set(left_sigs) != set(right_sigs):
            return "abstain"
        target = left_sigs[starts.index(True)]
        matches = [i for i, signature in enumerate(right_sigs) if signature == target]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except Exception:
        return "abstain"
