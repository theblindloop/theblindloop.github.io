"""Deterministic shuffled membership boards for correspondence binding."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1280, 760
N = 6
DECISIONS = tuple("ABCDEF")
INK = (35, 39, 48)
PAPER = (246, 247, 244)
WHITE = (255, 255, 255)
GRID = (116, 123, 132)
START = (200, 38, 145)
TAG_COLORS = (
    (216, 58, 64),
    (38, 105, 205),
    (25, 151, 104),
    (232, 139, 32),
    (130, 75, 190),
    (18, 158, 174),
)
CELL = 62
GRID_Y = 228
LEFT_X = 125
RIGHT_X = 755
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

# Six distinct weight-three signatures. The circulant construction also gives
# every witness row exactly three filled cells, removing row/column count cues.
_CANONICAL = tuple(
    tuple(1 if row in {col, (col + 1) % N, (col + 3) % N} else 0 for col in range(N))
    for row in range(N)
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _valid_matrix(matrix: object) -> bool:
    return (
        isinstance(matrix, list)
        and len(matrix) == N
        and all(isinstance(row, list) and len(row) == N and all(v in (0, 1) for v in row) for row in matrix)
    )


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {int(tag): row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    left_entities = list(range(N))
    rng.shuffle(left_entities)
    left_tags = list(range(N))
    right_tags = list(range(N))
    rng.shuffle(left_tags)
    rng.shuffle(right_tags)

    start_index = rng.randrange(N)
    target = int(seed) % N
    mapping: list[int | None] = [None] * N
    mapping[start_index] = target
    remaining = [i for i in range(N) if i != target]
    rng.shuffle(remaining)
    for col, right_col in zip((i for i in range(N) if i != start_index), remaining, strict=True):
        mapping[col] = right_col

    left = [[_CANONICAL[tag][entity] for entity in left_entities] for tag in left_tags]
    right = [[0 for _ in range(N)] for _ in range(N)]
    for right_row, tag in enumerate(right_tags):
        for left_col, entity in enumerate(left_entities):
            right[right_row][int(mapping[left_col])] = _CANONICAL[tag][entity]

    scene = {
        "left_matrix": left,
        "right_matrix": right,
        "left_row_tags": left_tags,
        "right_row_tags": right_tags,
        "start_index": start_index,
    }
    if is_quarantined(scene):
        raise RuntimeError("constructed registry failed invariants")
    return scene


def analytic_gold(scene: dict) -> str:
    left = _signatures(scene["left_matrix"], scene["left_row_tags"])
    right = _signatures(scene["right_matrix"], scene["right_row_tags"])
    target_signature = left[int(scene["start_index"])]
    matches = [i for i, signature in enumerate(right) if signature == target_signature]
    if len(matches) != 1:
        raise ValueError("START does not have one counterpart")
    return DECISIONS[matches[0]]


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    left = _signatures(scene["left_matrix"], scene["left_row_tags"])
    right = _signatures(scene["right_matrix"], scene["right_row_tags"])
    target = left[int(scene["start_index"])]
    return float(min(sum(a != b for a, b in zip(target, sig, strict=True)) for sig in right if sig != target))


def is_quarantined(scene: dict) -> bool:
    try:
        left, right = scene["left_matrix"], scene["right_matrix"]
        lt, rt = scene["left_row_tags"], scene["right_row_tags"]
        if not _valid_matrix(left) or not _valid_matrix(right):
            return True
        if sorted(lt) != list(range(N)) or sorted(rt) != list(range(N)):
            return True
        if int(scene["start_index"]) not in range(N):
            return True
        ls, rs = _signatures(left, lt), _signatures(right, rt)
        if len(set(ls)) != N or len(set(rs)) != N or set(ls) != set(rs):
            return True
        if any(sum(sig) != 3 for sig in ls + rs):
            return True
        if any(sum(row) != 3 for row in left + right):
            return True
        return rs.count(ls[int(scene["start_index"])]) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int, fill: tuple[int, int, int]) -> None:
    font = _font(size)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - 2), text, font=font, fill=fill)


def _draw_tag(draw: ImageDraw.ImageDraw, cx: int, cy: int, tag: int) -> None:
    color = TAG_COLORS[tag]
    if tag == 0:
        draw.polygon(((cx, cy - 20), (cx - 21, cy + 18), (cx + 21, cy + 18)), fill=color, outline=INK)
    elif tag == 1:
        draw.rectangle((cx - 18, cy - 18, cx + 18, cy + 18), fill=color, outline=INK, width=2)
    elif tag == 2:
        draw.ellipse((cx - 19, cy - 19, cx + 19, cy + 19), fill=color, outline=INK, width=2)
    elif tag == 3:
        draw.polygon(((cx, cy - 21), (cx - 21, cy), (cx, cy + 21), (cx + 21, cy)), fill=color, outline=INK)
    elif tag == 4:
        draw.polygon(((cx - 19, cy - 10), (cx, cy - 21), (cx + 19, cy - 10), (cx + 19, cy + 10), (cx, cy + 21), (cx - 19, cy + 10)), fill=color, outline=INK)
    else:
        draw.rectangle((cx - 7, cy - 21, cx + 7, cy + 21), fill=color, outline=INK, width=2)
        draw.rectangle((cx - 21, cy - 7, cx + 21, cy + 7), fill=color, outline=INK, width=2)


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, matrix: list[list[int]], tags: list[int], labels: tuple[str, ...], caption: str, start: int | None) -> None:
    _centered_text(draw, (x0 + N * CELL // 2, 126), caption, 29, INK)
    for col, label in enumerate(labels):
        cx = x0 + col * CELL + CELL // 2
        if col == start:
            draw.rounded_rectangle((cx - 27, 157, cx + 27, 207), radius=12, fill=START, outline=START)
            _centered_text(draw, (cx, 182), label, 24, WHITE)
            _centered_text(draw, (cx, 145), "START", 15, START)
        else:
            draw.ellipse((cx - 22, 160, cx + 22, 204), fill=WHITE, outline=INK, width=3)
            _centered_text(draw, (cx, 182), label, 23, INK)
    for k in range(N + 1):
        draw.line((x0, GRID_Y + k * CELL, x0 + N * CELL, GRID_Y + k * CELL), fill=GRID, width=2)
        draw.line((x0 + k * CELL, GRID_Y, x0 + k * CELL, GRID_Y + N * CELL), fill=GRID, width=2)
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        _draw_tag(draw, x0 - 49, cy, int(tags[row]))
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            draw.ellipse((cx - 17, cy - 17, cx + 17, cy + 17), fill=INK if matrix[row][col] else WHITE, outline=INK, width=4)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 42), "SAME MEMBERS, SHUFFLED REGISTRY", 35, INK)
    _draw_board(draw, LEFT_X, scene["left_matrix"], scene["left_row_tags"], tuple(str(i + 1) for i in range(N)), "NUMBERED BOARD", int(scene["start_index"]))
    _draw_board(draw, RIGHT_X, scene["right_matrix"], scene["right_row_tags"], DECISIONS, "LETTERED BOARD", None)
    draw.ellipse((190, 652, 224, 686), fill=INK, outline=INK, width=3)
    draw.text((236, 651), "filled = member", font=_font(21), fill=INK)
    draw.ellipse((481, 652, 515, 686), fill=WHITE, outline=INK, width=4)
    draw.text((527, 651), "hollow = not member", font=_font(21), fill=INK)
    draw.text((828, 651), "Match row symbols, not row positions.", font=_font(20), fill=INK)
    return image
