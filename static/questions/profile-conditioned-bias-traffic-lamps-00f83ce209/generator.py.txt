"""Matched traffic-signal renderer with balanced canonical and conflicting counts."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SCALE = 3
COUNTS = (2, 3, 4, 5)
PALETTE = (
    (220, 48, 48),
    (244, 155, 35),
    (49, 181, 91),
    (30, 173, 193),
    (68, 105, 210),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed ^ 0x71AFF1C)
    return {
        "lamp_count": COUNTS[seed % 4],
        "center_dx": rng.randint(-12, 12),
        "center_dy": rng.randint(-4, 4),
        "target_ink": rng.randint(8500, 9800),
        "palette_shift": rng.randrange(5),
        "pole_side": "left" if rng.randrange(2) == 0 else "right",
        "housing_gray": "#343b43" if rng.randrange(2) else [52, 59, 67],
    }


def _geometry(scene: dict) -> tuple[float, list[float]]:
    count = int(scene["lamp_count"])
    radius = math.sqrt(float(scene["target_ink"]) / (math.pi * count))
    center_y = SIZE / 2 + float(scene["center_dy"])
    span = 256.0
    if count == 1:
        centers = [center_y]
    else:
        centers = [center_y - span / 2 + i * span / (count - 1) for i in range(count)]
    return radius, centers


def render(scene: dict) -> Image.Image:
    count = int(scene["lamp_count"])
    if count not in COUNTS:
        raise ValueError("lamp_count must be 2, 3, 4, or 5")
    dx = float(scene["center_dx"])
    dy = float(scene["center_dy"])
    cx = (SIZE / 2 + dx) * SCALE
    housing_gray = scene["housing_gray"]
    if isinstance(housing_gray, list):
        housing_gray = tuple(housing_gray)

    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), (241, 239, 230))
    draw = ImageDraw.Draw(image)
    dark = housing_gray
    outline = (28, 33, 38)
    metal = (111, 120, 124)

    # The pole and bracket vary independently of the answer and remain neutral.
    pole_x = (65 if scene["pole_side"] == "left" else 319) * SCALE
    draw.rounded_rectangle(
        (pole_x - 8 * SCALE, 8 * SCALE, pole_x + 8 * SCALE, 384 * SCALE),
        radius=5 * SCALE,
        fill=metal,
        outline=(70, 78, 82),
        width=2 * SCALE,
    )
    bracket_y = (192 + dy) * SCALE
    x0 = min(pole_x, cx)
    x1 = max(pole_x, cx)
    draw.rounded_rectangle(
        (x0, bracket_y - 8 * SCALE, x1, bracket_y + 8 * SCALE),
        radius=5 * SCALE,
        fill=metal,
        outline=(70, 78, 82),
        width=2 * SCALE,
    )

    left = cx - 69 * SCALE
    right = cx + 69 * SCALE
    top = (14 + dy) * SCALE
    bottom = (370 + dy) * SCALE
    draw.rounded_rectangle(
        (left, top, right, bottom),
        radius=26 * SCALE,
        fill=dark,
        outline=outline,
        width=5 * SCALE,
    )

    radius, centers_y = _geometry(scene)
    shift = int(scene["palette_shift"])
    for index, y in enumerate(centers_y):
        px = cx
        py = y * SCALE
        outer_r = (radius + 6.5) * SCALE
        inner_r = radius * SCALE
        draw.ellipse((px - outer_r, py - outer_r, px + outer_r, py + outer_r), fill=(18, 22, 26))
        color = PALETTE[(index + shift) % len(PALETTE)]
        draw.ellipse((px - inner_r, py - inner_r, px + inner_r, py + inner_r), fill=color)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["lamp_count"]))


def margin(scene: dict) -> float:
    radius, centers_y = _geometry(scene)
    gaps = [b - a - 2.0 * radius for a, b in zip(centers_y, centers_y[1:])]
    return round(min(gaps), 4)


def is_quarantined(scene: dict) -> bool:
    radius, _ = _geometry(scene)
    return margin(scene) < 8.0 or radius < 20.0 or radius > 43.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    value = scene["housing_gray"]
    twin["housing_gray"] = [52, 59, 67] if isinstance(value, str) else "#343b43"
    return [("equivalent_housing_gray_encoding", twin)]
