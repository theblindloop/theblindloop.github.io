"""Independent PNG-only inverse arm for counting saturated lamp components."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _saturated_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    return (high >= 105) & ((high - low) >= 62)


def _component_areas(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    areas: list[int] = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        area = 0
        while queue:
            cy, cx = queue.popleft()
            area += 1
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        areas.append(area)
    return areas


def decision_from_image(image: Image.Image) -> str:
    if image.width < 192 or image.height < 192:
        raise ValueError("image resolution is too low")
    areas = _component_areas(_saturated_mask(image))
    lamps = [area for area in areas if area >= 900]
    speckle = sum(area for area in areas if 12 <= area < 900)
    if speckle > 80:
        raise ValueError("ambiguous fragmented colored evidence")
    if len(lamps) not in (2, 3, 4, 5):
        raise ValueError("visible colored lamp count is absent or out of contract")
    if min(lamps) < 1100 or max(lamps) / min(lamps) > 1.35:
        raise ValueError("colored regions are not consistent separate lamps")
    return str(len(lamps))
