"""Independent pixel-only inverse for the cursor-addressed write program.

The inverse segments saturated symbol ink, discovers upper and lower symbol
groups from relative image halves, classifies filled/ring/triangle shapes by
measured fill ratio, counts pip runs, identifies the sole small cursor marker,
and replays the decoded indirect writes. It receives no scene information.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class PixelParseError(ValueError):
    pass


def _runs(active: np.ndarray, *, minimum: int) -> list[tuple[int, int]]:
    padded = np.pad(active.astype(np.int8), (1, 1))
    edges = np.diff(padded)
    starts = np.flatnonzero(edges == 1)
    ends = np.flatnonzero(edges == -1)
    return [
        (int(start), int(end))
        for start, end in zip(starts, ends, strict=True)
        if end - start >= minimum
    ]


def _symbol_spans(mask: np.ndarray, expected: int) -> list[tuple[int, int]]:
    spans = _runs(mask.sum(axis=0) >= 4, minimum=24)
    if len(spans) != expected:
        raise PixelParseError(f"expected {expected} separated symbols, found {len(spans)}")
    gaps = [spans[index + 1][0] - spans[index][1] for index in range(len(spans) - 1)]
    if gaps and min(gaps) < 12:
        raise PixelParseError("symbol order lacks visible separation")
    return spans


def _parts(mask: np.ndarray, span: tuple[int, int]) -> tuple[float, int, bool, float]:
    """Measure one symbol column without using an absolute raster coordinate."""

    x0, x1 = span
    crop = mask[:, x0:x1]
    row_spans = _runs(crop.sum(axis=1) >= 3, minimum=5)
    if not row_spans:
        raise PixelParseError("symbol has no measurable ink")
    tall = [item for item in row_spans if item[1] - item[0] >= 30]
    if len(tall) != 1:
        raise PixelParseError("symbol does not contain one unambiguous main icon")
    main_y0, main_y1 = tall[0]
    main = crop[main_y0:main_y1]
    occupied_x = np.flatnonzero(main.any(axis=0))
    occupied_y = np.flatnonzero(main.any(axis=1))
    if len(occupied_x) < 24 or len(occupied_y) < 30:
        raise PixelParseError("main icon is too small")
    tight = main[
        occupied_y[0] : occupied_y[-1] + 1,
        occupied_x[0] : occupied_x[-1] + 1,
    ]
    fill_ratio = float(tight.mean())

    lower = [item for item in row_spans if item[0] > main_y1 + 4]
    pip_bands = [item for item in lower if 5 <= item[1] - item[0] <= 13]
    cursor_bands = [item for item in lower if 15 <= item[1] - item[0] <= 25]
    if len(pip_bands) > 1 or len(cursor_bands) > 1:
        raise PixelParseError("multiple rival pip or cursor bands")

    pips = 0
    leads: list[float] = []
    if pip_bands:
        py0, py1 = pip_bands[0]
        pip_spans = _runs(crop[py0:py1].sum(axis=0) >= 3, minimum=5)
        if not 1 <= len(pip_spans) <= 4:
            raise PixelParseError("pip identity is outside 1-4")
        widths = [right - left for left, right in pip_spans]
        gaps = [pip_spans[i + 1][0] - pip_spans[i][1] for i in range(len(pip_spans) - 1)]
        if min(widths) < 6 or (gaps and min(gaps) < 4):
            raise PixelParseError("pip identity is merged or undersized")
        pips = len(pip_spans)
        leads.extend(float(value) for value in widths + gaps)

    has_cursor = bool(cursor_bands)
    if has_cursor:
        cy0, cy1 = cursor_bands[0]
        cursor_crop = crop[cy0:cy1]
        cx = np.flatnonzero(cursor_crop.any(axis=0))
        if len(cx) < 20:
            raise PixelParseError("cursor marker is undersized")
        ratio = float(cursor_crop[:, cx[0] : cx[-1] + 1].mean())
        if not 0.40 <= ratio <= 0.62:
            raise PixelParseError("cursor marker is not a clear triangle")
        leads.append(100.0 * min(ratio - 0.40, 0.62 - ratio))
    return fill_ratio, pips, has_cursor, min(leads) if leads else 99.0


def _decode(image: Image.Image) -> tuple[list[bool], int, list[str], float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[0] < 300 or rgb.shape[1] < 700:
        raise PixelParseError("unexpected raster dimensions")
    high = rgb.max(axis=2)
    low = rgb.min(axis=2)
    saturated = (high - low >= 65) & (high >= 95)

    split = saturated.shape[0] // 2
    upper = saturated[:split]
    lower = saturated[split:]
    cell_spans = _symbol_spans(upper, 4)
    card_spans = _symbol_spans(lower, 8)

    initial_by_label: dict[int, bool] = {}
    cursor_labels: list[int] = []
    leads: list[float] = []
    for span in cell_spans:
        ratio, pips, has_cursor, detail_lead = _parts(upper, span)
        if pips in initial_by_label or not 1 <= pips <= 4:
            raise PixelParseError("cell pip identities are missing or repeated")
        if ratio <= 0.40:
            state = False
            leads.append(100.0 * (0.40 - ratio))
        elif ratio >= 0.68:
            state = True
            leads.append(100.0 * (ratio - 0.68))
        else:
            raise PixelParseError("cell is ambiguous between ring and filled disk")
        initial_by_label[pips] = state
        if has_cursor:
            cursor_labels.append(pips)
        leads.append(detail_lead)
    if set(initial_by_label) != {1, 2, 3, 4}:
        raise PixelParseError("cell identities do not cover 1-4")
    if len(cursor_labels) != 1:
        raise PixelParseError("expected exactly one cursor marker")
    initial = [initial_by_label[index] for index in range(1, 5)]
    initial_cursor = cursor_labels[0] - 1

    operations: list[str] = []
    for span in card_spans:
        ratio, pips, has_cursor, detail_lead = _parts(lower, span)
        if pips or has_cursor:
            raise PixelParseError("operation card has an undeclared label or cursor")
        if ratio <= 0.40:
            operation = "set_off"
            lead = 100.0 * (0.40 - ratio)
        elif 0.42 <= ratio <= 0.60:
            operation = "step"
            lead = 100.0 * min(ratio - 0.42, 0.60 - ratio)
        elif 0.68 <= ratio <= 0.87:
            operation = "set_on"
            lead = 100.0 * min(ratio - 0.68, 0.87 - ratio)
        else:
            raise PixelParseError("card icon is ambiguous among STEP, SET ON, and SET OFF")
        operations.append(operation)
        leads.extend((lead, detail_lead))

    expected = sorted(["step"] * 4 + ["set_on"] * 2 + ["set_off"] * 2)
    if sorted(operations) != expected:
        raise PixelParseError("decoded program lacks the declared balanced vocabulary")
    return initial, initial_cursor, operations, min(leads)


def decision_from_image(image: Image.Image) -> str:
    initial, cursor, operations, _runner_up_margin = _decode(image)
    cells = list(initial)
    pointer = int(cursor)
    for operation in operations:
        if operation == "step":
            pointer = (pointer + 1) % 4
        elif operation == "set_on":
            cells[pointer] = True
        elif operation == "set_off":
            cells[pointer] = False
        else:  # pragma: no cover - guarded by pixel parsing
            raise PixelParseError(f"unrecognized operation {operation!r}")
    return str(sum(cells))
