"""Renderer and analytic arm for the cursor-addressed write world."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1160
HEIGHT = 560
CELL_COUNT = 4
CARD_COUNT = 8

BACKGROUND = (249, 249, 246)
PANEL = (255, 255, 255)
INK = (30, 36, 43)
MUTED = (91, 100, 109)
BORDER = (150, 159, 167)
DECORATION = (179, 184, 188)

# Semantic roles are carried by fill/ring/triangle shape, not hue. Rotating
# these palettes therefore changes substantial pixels without changing gold.
PALETTES = (
    ((12, 145, 166), (230, 119, 30), (125, 72, 184)),
    ((125, 72, 184), (18, 126, 91), (211, 78, 91)),
    ((230, 119, 30), (55, 112, 190), (18, 126, 91)),
    ((211, 78, 91), (125, 72, 184), (12, 145, 166)),
)

_OPERATION_MULTISET = ["step"] * 4 + ["set_on"] * 2 + ["set_off"] * 2
_INITIAL_COUNT_OPTIONS = {
    0: (1, 2),
    1: (0, 2, 3),
    2: (0, 1, 3, 4),
    3: (1, 2, 4),
    4: (2, 3),
}


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _replay(
    initial: Iterable[bool], initial_cursor: int, operations: Iterable[str]
) -> tuple[list[bool], int]:
    cells = [bool(value) for value in initial]
    cursor = int(initial_cursor)
    if len(cells) != CELL_COUNT or not 0 <= cursor < CELL_COUNT:
        raise ValueError("invalid initial memory or cursor")
    for operation in operations:
        if operation == "step":
            cursor = (cursor + 1) % CELL_COUNT
        elif operation == "set_on":
            cells[cursor] = True
        elif operation == "set_off":
            cells[cursor] = False
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return cells, cursor


def _trace(initial: list[bool], cursor: int, operations: list[str]) -> dict[str, Any]:
    cells = list(initial)
    pointer = int(cursor)
    writes: list[tuple[int, bool, bool]] = []
    changed = {"set_on": 0, "set_off": 0}
    for operation in operations:
        if operation == "step":
            pointer = (pointer + 1) % CELL_COUNT
        else:
            value = operation == "set_on"
            writes.append((pointer, cells[pointer], value))
            changed[operation] += int(cells[pointer] != value)
            cells[pointer] = value
    return {"final": cells, "cursor": pointer, "writes": writes, "changed": changed}


def _fixed_address_result(initial: list[bool], cursor: int, operations: list[str]) -> list[bool]:
    cells = list(initial)
    for operation in operations:
        if operation == "set_on":
            cells[cursor] = True
        elif operation == "set_off":
            cells[cursor] = False
    return cells


def _balanced_scene_core(
    rng: random.Random, desired_count: int, initial_count: int
) -> tuple[list[bool], int, list[str]]:
    """Find a compact program whose indirect addressing is causally necessary."""

    for _ in range(50_000):
        initial = [True] * initial_count + [False] * (CELL_COUNT - initial_count)
        rng.shuffle(initial)
        cursor = rng.randrange(CELL_COUNT)
        operations = list(_OPERATION_MULTISET)
        rng.shuffle(operations)
        trace = _trace(initial, cursor, operations)
        if sum(trace["final"]) != desired_count:
            continue
        writes = trace["writes"]
        addressed = {entry[0] for entry in writes}
        if len(addressed) < 2:
            continue
        # At least one address is overwritten with opposite literal values.
        if not any(
            writes[i][0] == writes[j][0] and writes[i][2] != writes[j][2]
            for i in range(len(writes))
            for j in range(i + 1, len(writes))
        ):
            continue
        if sum(_replay(initial, cursor, reversed(operations))[0]) == desired_count:
            continue
        if sum(_fixed_address_result(initial, cursor, operations)) == desired_count:
            continue
        return initial, cursor, operations
    raise RuntimeError(f"could not construct indirect-write program for answer {desired_count}")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample deterministically; consecutive seeds balance all five answers."""

    seed = int(seed)
    rng = random.Random(seed)
    desired_count = seed % (CELL_COUNT + 1)
    options = _INITIAL_COUNT_OPTIONS[desired_count]
    initial_count = options[(seed // (CELL_COUNT + 1)) % len(options)]
    initial, cursor, operations = _balanced_scene_core(rng, desired_count, initial_count)
    display_order = list(range(CELL_COUNT))
    rng.shuffle(display_order)
    decorations = list(range(CARD_COUNT))
    rng.shuffle(decorations)
    palette = [list(color) for color in PALETTES[rng.randrange(len(PALETTES))]]
    shift = rng.randrange(3)
    palette = palette[shift:] + palette[:shift]
    return {
        "initial_cells": initial,
        "initial_cursor": cursor,
        "operations": operations,
        "display_order": display_order,
        "palette": palette,
        "offset_x": rng.randint(-5, 5),
        "offset_y": rng.randint(-4, 4),
        "card_gap": rng.randint(-1, 1),
        "icon_radius": rng.randint(23, 26),
        "ring_width": rng.randint(4, 5),
        "decorations": decorations,
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    cells, _cursor = _replay(
        scene["initial_cells"], scene["initial_cursor"], scene["operations"]
    )
    return str(sum(cells))


def margin(scene: dict[str, Any]) -> float:
    """Conservative runner-up lead for the rendered symbol vocabulary."""

    radius = float(scene["icon_radius"])
    ring_width = float(scene["ring_width"])
    palette = scene["palette"]
    saturation = min(max(color) - min(color) - 65.0 for color in palette)
    disk_ratio = math.pi / 4.0
    ring_ratio = disk_ratio * (1.0 - ((radius - ring_width) / radius) ** 2)
    ring_lead = 100.0 * (0.40 - ring_ratio)
    triangle_lead = 100.0 * min(0.50 - 0.42, 0.60 - 0.50)
    disk_lead = 100.0 * min(disk_ratio - 0.68, 0.87 - disk_ratio)
    pip_clearance = 8.0
    cursor_clearance = 9.0
    separation = 14.0 + float(scene["card_gap"])
    return round(
        min(
            saturation,
            ring_lead,
            triangle_lead,
            disk_lead,
            pip_clearance,
            cursor_clearance,
            separation,
        ),
        6,
    )


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        initial = scene["initial_cells"]
        cursor = scene["initial_cursor"]
        operations = scene["operations"]
        order = scene["display_order"]
        decorations = scene["decorations"]
        palette = scene["palette"]
        valid = (
            len(initial) == CELL_COUNT
            and all(isinstance(value, bool) for value in initial)
            and isinstance(cursor, int)
            and 0 <= cursor < CELL_COUNT
            and len(operations) == CARD_COUNT
            and sorted(operations) == sorted(_OPERATION_MULTISET)
            and sorted(order) == list(range(CELL_COUNT))
            and sorted(decorations) == list(range(CARD_COUNT))
            and len(palette) == 3
            and all(len(color) == 3 for color in palette)
            and -5 <= int(scene["offset_x"]) <= 5
            and -4 <= int(scene["offset_y"]) <= 4
            and -1 <= int(scene["card_gap"]) <= 1
            and 23 <= int(scene["icon_radius"]) <= 26
            and 4 <= int(scene["ring_width"]) <= 5
        )
        return not valid or margin(scene) < 6.5
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
        text,
        font=font,
        fill=fill,
    )


def _draw_pips(
    draw: ImageDraw.ImageDraw, center_x: float, center_y: float, count: int, color
) -> None:
    spacing = 13
    radius = 4
    for index in range(count):
        x = center_x + (index - (count - 1) / 2) * spacing
        draw.ellipse((x - radius, center_y - radius, x + radius, center_y + radius), fill=color)


def _draw_ring(draw: ImageDraw.ImageDraw, center_x: float, center_y: float, radius: int, color, width: int) -> None:
    draw.ellipse(
        (center_x - radius, center_y - radius, center_x + radius, center_y + radius),
        outline=color,
        width=width,
    )


def _draw_decoration(draw: ImageDraw.ImageDraw, center_x: float, y: float, value: int) -> None:
    for bit in range(3):
        x = center_x + (bit - 1) * 9
        box = (x - 2, y - 2, x + 2, y + 2)
        if value & (1 << bit):
            draw.ellipse(box, fill=DECORATION)
        else:
            draw.ellipse(box, outline=DECORATION, width=1)


def render(scene: dict[str, Any]) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render malformed or quarantined scene")
    initial = list(scene["initial_cells"])
    initial_cursor = int(scene["initial_cursor"])
    operations = list(scene["operations"])
    display_order = list(scene["display_order"])
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    offset_x = int(scene["offset_x"])
    offset_y = int(scene["offset_y"])
    card_gap = int(scene["card_gap"])
    icon_radius = int(scene["icon_radius"])
    ring_width = int(scene["ring_width"])
    decorations = list(scene["decorations"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(25, bold=True)
    label_font = _font(14, bold=True)
    small_font = _font(12, bold=True)

    _centered(draw, (WIDTH / 2, 27), "CURSOR-CONTROLLED CELL WRITES", title_font, INK)
    _centered(draw, (WIDTH / 2, 57), "Track both the cursor and the four cell states", _font(14), MUTED)

    top = 76 + offset_y
    bottom = 244 + offset_y
    draw.rounded_rectangle(
        (31 + offset_x, top, 1129 + offset_x, bottom),
        radius=16,
        fill=PANEL,
        outline=BORDER,
        width=3,
    )
    draw.text((49 + offset_x, top + 13), "START", font=label_font, fill=INK)
    state_color = palette[0]
    cursor_color = palette[1]
    for slot, logical_index in enumerate(display_order):
        x = 210 + slot * 248 + offset_x
        y = 135 + offset_y
        if initial[logical_index]:
            draw.ellipse(
                (x - icon_radius, y - icon_radius, x + icon_radius, y + icon_radius),
                fill=state_color,
            )
        else:
            _draw_ring(draw, x, y, icon_radius, state_color, ring_width)
        _draw_pips(draw, x, y + 50, logical_index + 1, state_color)
        if logical_index == initial_cursor:
            cy = y + 85
            draw.polygon(((x, cy - 11), (x - 13, cy + 9), (x + 13, cy + 9)), fill=cursor_color)

    draw.text((31, 254), "READ CARDS LEFT TO RIGHT", font=label_font, fill=INK)
    card_top = 279 + offset_y
    card_bottom = 489 + offset_y
    card_width = 126
    step = 141 + card_gap
    for index, (operation, decoration) in enumerate(zip(operations, decorations)):
        left = 18 + offset_x + index * step
        right = left + card_width
        center_x = (left + right) / 2
        draw.rounded_rectangle(
            (left, card_top, right, card_bottom),
            radius=12,
            fill=PANEL,
            outline=BORDER,
            width=3,
        )
        _centered(draw, (center_x, card_top + 18), str(index + 1), small_font, MUTED)
        center_y = card_top + 78
        if operation == "step":
            color = palette[1]
            draw.polygon(
                (
                    (center_x + icon_radius, center_y),
                    (center_x - icon_radius, center_y - icon_radius),
                    (center_x - icon_radius, center_y + icon_radius),
                ),
                fill=color,
            )
            label = "STEP"
        elif operation == "set_on":
            color = palette[2]
            draw.ellipse(
                (
                    center_x - icon_radius,
                    center_y - icon_radius,
                    center_x + icon_radius,
                    center_y + icon_radius,
                ),
                fill=color,
            )
            label = "SET ON"
        elif operation == "set_off":
            color = palette[2]
            _draw_ring(draw, center_x, center_y, icon_radius, color, ring_width)
            label = "SET OFF"
        else:
            raise ValueError(f"unknown operation {operation!r}")
        _centered(draw, (center_x, card_top + 151), label, _font(11, bold=True), INK)
        _draw_decoration(draw, center_x, card_top + 187, int(decoration))

    _centered(
        draw,
        (WIDTH / 2, 528),
        "filled = ON   ring = OFF     triangle = STEP     SET writes at the cursor",
        _font(13, bold=True),
        MUTED,
    )
    return image
