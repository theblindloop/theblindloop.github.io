"""Deterministic renderer for an ordered two-input quantitative mixer network."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1200
HEIGHT = 800
MACHINES = ("blue", "orange")
CHANNELS = ("teal", "violet")
SIDES = ("left", "right")
MACHINE_COLORS = {"blue": (43, 104, 189), "orange": (218, 124, 48)}
CHANNEL_COLORS = {"teal": (20, 150, 145), "violet": (132, 82, 190)}
BACKGROUND = (235, 240, 243)
PAPER = (255, 255, 252)
INK = (34, 41, 50)
MUTED = (91, 101, 112)
DISPLAY_BG = (34, 43, 53)
SEGMENT_ON = (250, 239, 184)
QUERY_COLOR = (39, 137, 124)

CAL_Y = (220, 330)
CAL_POSITIONS = {
    "blue": (100, 205, 515, 350),
    "orange": (690, 795, 1105, 940),
}
QUERY_Y = (550, 680)
BUILD_NUMBER_X = (120, 225)
BUILD_MACHINE_X = 370
BUILD_CHANNEL_X = 520
FINAL_TOKEN_X = (720, 795)
FINAL_MACHINE_X = 955

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

SEGMENTS = {
    0: "abcedf", 1: "bc", 2: "abged", 3: "abgcd", 4: "fgbc",
    5: "afgcd", 6: "afgecd", 7: "abc", 8: "abcdefg", 9: "abfgcd",
}


def _int(value: object, label: str, lo: int = 0, hi: int = 99) -> int:
    if isinstance(value, bool) or not isinstance(value, int) or not lo <= value <= hi:
        raise ValueError(f"{label} must be an integer in [{lo}, {hi}]")
    return int(value)


def _examples(value: object, label: str) -> tuple[tuple[int, int, int], tuple[int, int, int]]:
    if not isinstance(value, (list, tuple)) or len(value) != 2:
        raise ValueError(f"{label} must contain two calibration triples")
    rows = []
    for index, row in enumerate(value):
        if not isinstance(row, (list, tuple)) or len(row) != 3:
            raise ValueError(f"{label} example {index + 1} must be [left, right, output]")
        rows.append(tuple(_int(v, f"{label} example value", 1, 99) for v in row))
    return rows[0], rows[1]


def _infer_rule(examples: tuple[tuple[int, int, int], tuple[int, int, int]]) -> tuple[int, int]:
    (l1, r1, out1), (l2, r2, out2) = examples
    det = l1 * r2 - l2 * r1
    if det == 0:
        raise ValueError("calibrations do not uniquely determine both coefficients")
    a_num = out1 * r2 - out2 * r1
    b_num = l1 * out2 - l2 * out1
    if a_num % det or b_num % det:
        raise ValueError("calibrations do not determine whole-number coefficients")
    a, b = a_num // det, b_num // det
    if not 1 <= a <= 3 or not 1 <= b <= 3 or a == b:
        raise ValueError("mixer coefficients are outside the declared range")
    if any(a * left + b * right != out for left, right, out in examples):
        raise ValueError("calibrations are inconsistent")
    return a, b


def _rules(scene: dict) -> dict[str, tuple[int, int]]:
    machines = scene.get("machines")
    if not isinstance(machines, dict) or set(machines) != set(MACHINES):
        raise ValueError("machines must contain blue and orange")
    result = {}
    for name in MACHINES:
        spec = machines[name]
        if not isinstance(spec, dict) or set(spec) != {"examples"}:
            raise ValueError(f"{name} machine must contain only examples")
        result[name] = _infer_rule(_examples(spec["examples"], name))
    if result["blue"] == result["orange"]:
        raise ValueError("the two visible mixers must have different rules")
    return result


def _build(scene: dict) -> tuple[dict, dict]:
    build = scene.get("build")
    if not isinstance(build, list) or len(build) != 2:
        raise ValueError("build must contain two channel rows")
    rows = []
    for expected_channel, row in zip(CHANNELS, build, strict=True):
        if not isinstance(row, dict) or set(row) != {"channel", "machine", "inputs"}:
            raise ValueError("each build row needs channel, machine, and inputs")
        if row["channel"] != expected_channel or row["machine"] not in MACHINES:
            raise ValueError("build channel order or machine is invalid")
        inputs = row["inputs"]
        if not isinstance(inputs, (list, tuple)) or len(inputs) != 2:
            raise ValueError("build inputs must be an ordered pair")
        rows.append({
            "channel": expected_channel,
            "machine": str(row["machine"]),
            "inputs": (_int(inputs[0], "build left", 1, 9), _int(inputs[1], "build right", 1, 9)),
        })
    if {row["machine"] for row in rows} != set(MACHINES):
        raise ValueError("build must use each mixer exactly once")
    return rows[0], rows[1]


def _finals(scene: dict) -> tuple[dict, dict]:
    finals = scene.get("finals")
    if not isinstance(finals, dict) or set(finals) != set(SIDES):
        raise ValueError("finals must contain left and right")
    rows = []
    for side in SIDES:
        row = finals[side]
        if not isinstance(row, dict) or set(row) != {"machine", "inputs"}:
            raise ValueError("each final needs a machine and ordered channel inputs")
        inputs = row["inputs"]
        if row["machine"] not in MACHINES or not isinstance(inputs, (list, tuple)) or len(inputs) != 2:
            raise ValueError("invalid final specification")
        if set(inputs) != set(CHANNELS):
            raise ValueError("each final must use both channels")
        rows.append({"machine": str(row["machine"]), "inputs": (str(inputs[0]), str(inputs[1]))})
    if {row["machine"] for row in rows} != set(MACHINES):
        raise ValueError("finals must use each mixer exactly once")
    if rows[1]["inputs"] != tuple(reversed(rows[0]["inputs"])):
        raise ValueError("final input orders must be reversed")
    return rows[0], rows[1]


def _evaluate(scene: dict) -> tuple[dict[str, int], dict[str, int]]:
    rules = _rules(scene)
    build = _build(scene)
    finals = _finals(scene)
    channels: dict[str, int] = {}
    for row in build:
        a, b = rules[row["machine"]]
        left, right = row["inputs"]
        channels[row["channel"]] = a * left + b * right
    if any(not 1 <= value <= 49 for value in channels.values()):
        raise ValueError("an intermediate channel is outside the supported range")
    outputs: dict[str, int] = {}
    for side, row in zip(SIDES, finals, strict=True):
        a, b = rules[row["machine"]]
        outputs[side] = a * channels[row["inputs"][0]] + b * channels[row["inputs"][1]]
    if any(not 1 <= value <= 99 for value in outputs.values()):
        raise ValueError("a final output is outside the supported range")
    return channels, outputs


def _gap(scene: dict) -> int:
    return _evaluate(scene)[1]["left"] - _evaluate(scene)[1]["right"]


def _calibrations(rng: random.Random, rule: tuple[int, int]) -> list[list[int]]:
    a, b = rule
    for _ in range(100):
        l1, r1, l2, r2 = (rng.randint(1, 8) for _ in range(4))
        if l1 * r2 == l2 * r1:
            continue
        rows = [[l1, r1, a * l1 + b * r1], [l2, r2, a * l2 + b * r2]]
        if rng.randrange(2):
            rows.reverse()
        return rows
    raise RuntimeError("failed to sample invertible calibration rows")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 15485863 + 32452843)
    target = {0: 1, 1: -1, 2: 0}[int(seed) % 3]
    chosen = None
    for _ in range(12000):
        rules = {name: (rng.randint(1, 3), rng.randint(1, 3)) for name in MACHINES}
        if rules["blue"] == rules["orange"] or any(a == b for a, b in rules.values()):
            continue
        build_machines = list(MACHINES)
        rng.shuffle(build_machines)
        build = []
        channel_values = {}
        for channel, machine in zip(CHANNELS, build_machines, strict=True):
            inputs = [rng.randint(1, 9), rng.randint(1, 9)]
            a, b = rules[machine]
            channel_values[channel] = a * inputs[0] + b * inputs[1]
            build.append({"channel": channel, "machine": machine, "inputs": inputs})
        left_machine = MACHINES[rng.randrange(2)]
        right_machine = MACHINES[1 - MACHINES.index(left_machine)]
        left_inputs = list(CHANNELS)
        rng.shuffle(left_inputs)
        right_inputs = list(reversed(left_inputs))
        finals = {
            "left": {"machine": left_machine, "inputs": left_inputs},
            "right": {"machine": right_machine, "inputs": right_inputs},
        }
        def run(row: dict) -> int:
            a, b = rules[row["machine"]]
            return a * channel_values[row["inputs"][0]] + b * channel_values[row["inputs"][1]]
        left_out, right_out = run(finals["left"]), run(finals["right"])
        if max(channel_values.values()) <= 49 and max(left_out, right_out) <= 99 and left_out - right_out == target:
            chosen = rules, build, finals
            break
    if chosen is None:
        raise RuntimeError("no mixer network found for requested class")
    rules, build, finals = chosen
    scene = {
        "machines": {name: {"examples": _calibrations(rng, rules[name])} for name in MACHINES},
        "build": build,
        "finals": finals,
    }
    if _gap(scene) != target:
        raise RuntimeError("sampler class construction failed")
    return scene


def analytic_gold(scene: dict) -> str:
    gap = _gap(scene)
    return "left" if gap > 0 else "right" if gap < 0 else "same"


def margin(scene: dict) -> float:
    return float(max(1, abs(_gap(scene))))


def is_quarantined(scene: dict) -> bool:
    try:
        return abs(_gap(scene)) > 1
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(_BOLD_PATH if bold else _FONT_PATH, size=size)


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_arrow(draw: ImageDraw.ImageDraw, x0: int, x1: int, y: int) -> None:
    draw.line((x0, y, x1 - 10, y), fill=INK, width=4)
    draw.polygon(((x1 - 12, y - 8), (x1, y), (x1 - 12, y + 8)), fill=INK)


def _draw_digit(draw: ImageDraw.ImageDraw, cx: int, cy: int, digit: int) -> None:
    centers = {
        "a": (cx, cy - 18, False), "g": (cx, cy, False), "d": (cx, cy + 18, False),
        "f": (cx - 12, cy - 9, True), "b": (cx + 12, cy - 9, True),
        "e": (cx - 12, cy + 9, True), "c": (cx + 12, cy + 9, True),
    }
    for name, (x, y, vertical) in centers.items():
        if name not in SEGMENTS[digit]:
            continue
        box = (x - 3, y - 8, x + 3, y + 8) if vertical else (x - 10, y - 3, x + 10, y + 3)
        draw.rounded_rectangle(box, radius=2, fill=SEGMENT_ON)


def _draw_number(draw: ImageDraw.ImageDraw, cx: int, cy: int, value: int) -> None:
    value = _int(value, "display value")
    draw.rounded_rectangle((cx - 39, cy - 31, cx + 39, cy + 31), radius=8, fill=DISPLAY_BG, outline=(112, 121, 128), width=2)
    tens, ones = divmod(value, 10)
    _draw_digit(draw, cx - 18, cy, tens)
    _draw_digit(draw, cx + 18, cy, ones)


def _draw_machine(draw: ImageDraw.ImageDraw, cx: int, cy: int, name: str) -> None:
    draw.rounded_rectangle((cx - 65, cy - 31, cx + 65, cy + 31), radius=12, fill=MACHINE_COLORS[name], outline=INK, width=3)
    _centered(draw, (cx, cy - 8), name.upper(), _font(16, bold=True), (255, 255, 255))
    _centered(draw, (cx, cy + 13), "MIX", _font(13, bold=True), (255, 255, 255))


def _draw_token(draw: ImageDraw.ImageDraw, cx: int, cy: int, channel: str) -> None:
    draw.ellipse((cx - 30, cy - 30, cx + 30, cy + 30), fill=CHANNEL_COLORS[channel], outline=INK, width=3)
    _centered(draw, (cx, cy), "U" if channel == "teal" else "V", _font(25, bold=True), (255, 255, 255))


def render(scene: dict) -> Image.Image:
    _rules(scene)
    build = _build(scene)
    finals = _finals(scene)
    _evaluate(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 16, WIDTH - 18, HEIGHT - 16), radius=20, fill=PAPER, outline=(190, 199, 203), width=3)
    _centered(draw, (WIDTH / 2, 42), "LEARN THE TWO-INPUT MIXERS", _font(28, bold=True), INK)
    _centered(draw, (WIDTH / 2, 73), "output = a × LEFT + b × RIGHT", _font(19, bold=True), MUTED)

    for name, bounds in (("blue", (35, 100, 575, 390)), ("orange", (625, 100, 1165, 390))):
        x0, y0, x1, y1 = bounds
        draw.rounded_rectangle(bounds, radius=15, fill=(249, 250, 248), outline=MACHINE_COLORS[name], width=4)
        draw.rounded_rectangle((x0 + 4, y0 + 4, x1 - 4, y0 + 55), radius=10, fill=MACHINE_COLORS[name])
        _centered(draw, ((x0 + x1) / 2, y0 + 29), f"{name.upper()} EXAMPLES", _font(21, bold=True), (255, 255, 255))
        left_x, right_x, out_x, machine_x = CAL_POSITIONS[name]
        _centered(draw, (left_x, 176), "LEFT", _font(13, bold=True), MUTED)
        _centered(draw, (right_x, 176), "RIGHT", _font(13, bold=True), MUTED)
        _centered(draw, (out_x, 176), "OUTPUT", _font(13, bold=True), MUTED)
        for (left, right, out), y in zip(_examples(scene["machines"][name]["examples"], name), CAL_Y, strict=True):
            _draw_number(draw, left_x, y, left)
            _draw_number(draw, right_x, y, right)
            _draw_arrow(draw, right_x + 46, machine_x - 70, y)
            _draw_machine(draw, machine_x, y, name)
            _draw_arrow(draw, machine_x + 70, out_x - 46, y)
            _draw_number(draw, out_x, y, out)

    draw.rounded_rectangle((35, 420, 1165, 775), radius=17, fill=(239, 249, 247), outline=QUERY_COLOR, width=4)
    draw.line((590, 438, 590, 755), fill=(171, 190, 186), width=3)
    _centered(draw, (300, 452), "BUILD U AND V", _font(22, bold=True), QUERY_COLOR)
    _centered(draw, (875, 452), "COMPARE FINALS", _font(22, bold=True), QUERY_COLOR)

    for row, y in zip(build, QUERY_Y, strict=True):
        _centered(draw, (BUILD_NUMBER_X[0], 493 if y == QUERY_Y[0] else 623), "LEFT", _font(12, bold=True), MUTED)
        _centered(draw, (BUILD_NUMBER_X[1], 493 if y == QUERY_Y[0] else 623), "RIGHT", _font(12, bold=True), MUTED)
        _draw_number(draw, BUILD_NUMBER_X[0], y, row["inputs"][0])
        _draw_number(draw, BUILD_NUMBER_X[1], y, row["inputs"][1])
        _draw_arrow(draw, 271, 300, y)
        _draw_machine(draw, BUILD_MACHINE_X, y, row["machine"])
        _draw_arrow(draw, 440, 483, y)
        _draw_token(draw, BUILD_CHANNEL_X, y, row["channel"])

    for side, row, y in zip(SIDES, finals, QUERY_Y, strict=True):
        _centered(draw, (630, y), side.upper(), _font(15, bold=True), MUTED)
        for x, channel, port in zip(FINAL_TOKEN_X, row["inputs"], ("L", "R"), strict=True):
            _centered(draw, (x, y - 43), port, _font(13, bold=True), MUTED)
            _draw_token(draw, x, y, channel)
        _draw_arrow(draw, 832, 884, y)
        _draw_machine(draw, FINAL_MACHINE_X, y, row["machine"])
        _draw_arrow(draw, 1025, 1070, y)
        _centered(draw, (1110, y), "?", _font(48, bold=True), INK)
    return image
