"""Deterministic renderer and analytic arm for dihedral_composed_match.

One fully-asymmetric 5x5 glyph is drawn upright inside a magenta-bordered frame.
Four smaller frames below each hold another rendering of the SAME 5x5 glyph, but
carrying one of four operations applied to the reference: identity, a quarter
turn clockwise, a half turn, or a composed operation (a quarter turn clockwise
followed by a top-bottom reflection).  Each frame is tagged with one of four
saturated colours above it.

The counterfactual is a composed geometric transformation: rotate the reference a
quarter turn clockwise, THEN reflect the result across the horizontal midline
(top edge swaps with bottom edge).  The question asks which tagged tile equals
the reference after exactly that composed operation.  Because the source glyph is
fully asymmetric (no nontrivial rotation or reflection symmetry), all four tile
patterns are distinct and share the same occupancy, outline, connectivity and
bounding extent, so the answer can only be recovered by actually composing the
two direction-specified operations and matching the result, never by outline,
identity, or a single pure-rotation shortcut.
"""

from __future__ import annotations

import random

import numpy as np

from PIL import Image, ImageDraw

CANVAS = 720
GRID = 5
N_TILES = 4

OP_IDENTITY = 0
OP_ROT90 = 1
OP_ROT180 = 2
OP_COMPOSE = 3

BG = (250, 250, 250)
INK = (42, 42, 42)
CELL_LIGHT = (238, 238, 238)
WHITE_LINE = (255, 255, 255)

REF_COLOR = (200, 20, 200)
TILE_COLOR = (110, 110, 110)
BORDER_W = 6

REF_ORIGIN = (230, 40)
REF_SIZE = 282

TILE_X = [24, 204, 384, 564]
TILE_Y = 430
TILE_SIZE = 152

TAG_NAMES = ["crimson", "teal", "amber", "violet"]
TAG_RGB = {
    "crimson": (200, 28, 40),
    "teal": (0, 150, 150),
    "amber": (235, 165, 25),
    "violet": (135, 60, 185),
}
TAG_HALF = 12


def rot_cw(arr: np.ndarray) -> np.ndarray:
    return np.rot90(arr, k=-1)


def flip_ud(arr: np.ndarray) -> np.ndarray:
    return np.flipud(arr)


def composed(arr: np.ndarray) -> np.ndarray:
    return flip_ud(rot_cw(arr))


def apply_op(grid: list, op: int) -> list:
    arr = np.asarray(grid, dtype=np.int8).reshape(GRID, GRID)
    if op == OP_IDENTITY:
        result = arr
    elif op == OP_ROT90:
        result = rot_cw(arr)
    elif op == OP_ROT180:
        result = rot_cw(rot_cw(arr))
    elif op == OP_COMPOSE:
        result = composed(arr)
    else:
        raise ValueError(f"unknown op {op!r}")
    return result.tolist()


def _connected(grid: list) -> bool:
    arr = np.asarray(grid, dtype=np.int8)
    filled = list(zip(*np.nonzero(arr)))
    if not filled:
        return False
    pixels = set(filled)
    start = pixels.pop()
    stack = [start]
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if (ny, nx) in pixels:
                pixels.remove((ny, nx))
                stack.append((ny, nx))
    return not pixels


def _dihedral_orbit_size(grid: list) -> int:
    arr = np.asarray(grid, dtype=np.int8).reshape(GRID, GRID)
    seen = set()
    for k in range(4):
        base = np.rot90(arr, k=k)
        seen.add(bytes(base.reshape(-1).tolist()))
        seen.add(bytes(flip_ud(base).reshape(-1).tolist()))
    return len(seen)


def _valid_glyph(grid: list) -> bool:
    if len(grid) != GRID or any(len(row) != GRID for row in grid):
        return False
    flat = [int(v) for row in grid for v in row]
    if any(v not in (0, 1) for v in flat):
        return False
    if not (7 <= sum(flat) <= 15):
        return False
    if not _connected(grid):
        return False
    if _dihedral_orbit_size(grid) != 8:
        return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        cells = [[rng.randint(0, 1) for _ in range(GRID)] for _ in range(GRID)]
        if _valid_glyph(cells):
            break
    ops = [OP_IDENTITY, OP_ROT90, OP_ROT180, OP_COMPOSE]
    rng.shuffle(ops)
    return {
        "canvas_size": CANVAS,
        "glyph": cells,
        "ops": ops,
        "grid": GRID,
        "tag_names": list(TAG_NAMES),
    }


def _draw_glyph(draw, origin, cell, grid) -> None:
    x0, y0 = origin
    for r in range(GRID):
        for k in range(GRID):
            left = x0 + BORDER_W + k * cell
            top = y0 + BORDER_W + r * cell
            fill = INK if int(grid[r][k]) == 1 else CELL_LIGHT
            draw.rectangle(
                (left + 1, top + 1, left + cell - 1, top + cell - 1),
                fill=fill,
                outline=WHITE_LINE,
            )


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    glyph = [[int(v) for v in row] for row in scene["glyph"]]
    ops = [int(v) for v in scene["ops"]]
    image = Image.new("RGB", (size, size), BG)
    draw = ImageDraw.Draw(image)

    ref_cell = (REF_SIZE - 2 * BORDER_W) // GRID
    draw.rectangle(
        (REF_ORIGIN[0], REF_ORIGIN[1], REF_ORIGIN[0] + REF_SIZE, REF_ORIGIN[1] + REF_SIZE),
        fill=None,
        outline=REF_COLOR,
        width=BORDER_W,
    )
    _draw_glyph(draw, REF_ORIGIN, ref_cell, glyph)

    tile_cell = (TILE_SIZE - 2 * BORDER_W) // GRID
    for slot in range(N_TILES):
        x0 = TILE_X[slot]
        y0 = TILE_Y
        draw.rectangle(
            (x0, y0, x0 + TILE_SIZE, y0 + TILE_SIZE),
            fill=None,
            outline=TILE_COLOR,
            width=BORDER_W,
        )
        grid_op = apply_op(glyph, ops[slot])
        _draw_glyph(draw, (x0, y0), tile_cell, grid_op)
        name = scene["tag_names"][slot]
        cx = x0 + TILE_SIZE // 2
        tag_cy = TILE_Y - 3 * TAG_HALF
        draw.rectangle(
            (cx - TAG_HALF, tag_cy - TAG_HALF, cx + TAG_HALF, tag_cy + TAG_HALF),
            fill=TAG_RGB[name],
        )
    return image


def _match_slot(scene: dict) -> int:
    ops = [int(v) for v in scene["ops"]]
    return ops.index(OP_COMPOSE)


def analytic_gold(scene: dict) -> str:
    slot = _match_slot(scene)
    return scene["tag_names"][slot]


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    glyph = scene["glyph"]
    ops = [int(v) for v in scene["ops"]]
    if len(ops) != N_TILES or sorted(ops) != [OP_IDENTITY, OP_ROT90, OP_ROT180, OP_COMPOSE]:
        return True
    if not _valid_glyph(glyph):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
