"""Independent pixel-only inverse for Four-Neighbor Press Correspondence."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 5
FRAME_COLORS = {
    "source": (73, 80, 88),
    "amber": (222, 151, 34),
    "cyan": (30, 166, 186),
    "violet": (132, 93, 190),
}
DARK = np.asarray((39, 43, 48), dtype=np.int16)
LIGHT = np.asarray((244, 239, 225), dtype=np.int16)
PANEL_FILL = np.asarray((211, 209, 200), dtype=np.int16)


class OracleAbstention(ValueError):
    """The raster no longer exposes four intact, unambiguous grids."""


def _near(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> np.ndarray:
    target = np.asarray(color, dtype=np.int16)
    return np.max(np.abs(array.astype(np.int16) - target), axis=2) <= tolerance


def _frame_box(array: np.ndarray, color: tuple[int, int, int]) -> tuple[int, int, int, int]:
    mask = _near(array, color)
    ys, xs = np.nonzero(mask)
    if len(xs) < 4000:
        raise OracleAbstention("a required frame is missing or heavily damaged")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if width < 200 or height < 200 or not (0.92 <= width / height <= 1.08):
        raise OracleAbstention("a required frame is not a large square")
    band = max(3, round(min(width, height) * 0.027))
    perimeter = np.concatenate(
        (
            mask[y0 : y0 + band, x0 : x1 + 1].ravel(),
            mask[y1 - band + 1 : y1 + 1, x0 : x1 + 1].ravel(),
            mask[y0 : y1 + 1, x0 : x0 + band].ravel(),
            mask[y0 : y1 + 1, x1 - band + 1 : x1 + 1].ravel(),
        )
    )
    if float(perimeter.mean()) < 0.90:
        raise OracleAbstention("a frame perimeter is incomplete")
    return x0, y0, x1, y1


def _read_grid(array: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    width, height = x1 - x0 + 1, y1 - y0 + 1
    grid_left = x0 + round(width * 34 / 374)
    grid_top = y0 + round(height * 34 / 374)
    pitch_x = width * 61 / 374
    pitch_y = height * 61 / 374
    output: list[list[int]] = []
    for row in range(GRID_N):
        line = []
        for col in range(GRID_N):
            cx = grid_left + (col + 0.5) * pitch_x
            cy = grid_top + (row + 0.5) * pitch_y
            half_x = max(4, int(pitch_x * 0.30))
            half_y = max(4, int(pitch_y * 0.30))
            xa, xb = max(0, int(cx) - half_x), min(array.shape[1], int(cx) + half_x)
            ya, yb = max(0, int(cy) - half_y), min(array.shape[0], int(cy) + half_y)
            patch = array[ya:yb, xa:xb].astype(np.int16)
            if patch.size == 0:
                raise OracleAbstention("a grid cell falls outside the image")
            dark = np.mean(np.max(np.abs(patch - DARK), axis=2) <= 5)
            light = np.mean(np.max(np.abs(patch - LIGHT), axis=2) <= 5)
            if max(float(dark), float(light)) < 0.72 or abs(float(dark) - float(light)) < 0.35:
                raise OracleAbstention("a grid cell is erased or visually ambiguous")
            line.append(int(dark > light))
        output.append(line)
    return output


def _press(source: list[list[int]]) -> list[list[int]]:
    output = []
    for row in range(GRID_N):
        line = []
        for col in range(GRID_N):
            count = 0
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                rr, cc = row + dr, col + dc
                if 0 <= rr < GRID_N and 0 <= cc < GRID_N:
                    count += source[rr][cc]
            line.append(int(count == 2))
        output.append(line)
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 256:
        raise OracleAbstention("canvas is too small")
    boxes = {name: _frame_box(array, rgb) for name, rgb in FRAME_COLORS.items()}
    if len(set(boxes.values())) != 4:
        raise OracleAbstention("frames are not distinct")
    source = _read_grid(array, boxes["source"])
    target = _press(source)
    matches = []
    for color in ("amber", "cyan", "violet"):
        candidate = _read_grid(array, boxes[color])
        if candidate == target:
            matches.append(color)
    if len(matches) != 1:
        raise OracleAbstention("the raster does not contain one matching candidate")
    return matches[0]
