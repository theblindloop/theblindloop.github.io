"""Renderer and analytic arm for Four-Neighbor Press Correspondence."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

CANVAS = 800
GRID_N = 5
PANEL = 374
PANEL_POSITIONS = ((20, 20), (406, 20), (20, 406), (406, 406))
FRAME_WIDTH = 10
GRID_OFFSET = 34
CELL_PITCH = 61
CELL_INSET = 5

BACKGROUND = (249, 247, 241)
PANEL_FILL = (211, 209, 200)
LIGHT = (244, 239, 225)
DARK = (39, 43, 48)
SOURCE_FRAME = (73, 80, 88)
COLORS = {
    "amber": (222, 151, 34),
    "cyan": (30, 166, 186),
    "violet": (132, 93, 190),
}


def _press(source: list[list[int]]) -> list[list[int]]:
    """Dark iff exactly two orthogonal source neighbors are dark."""
    output: list[list[int]] = []
    for row in range(GRID_N):
        line = []
        for col in range(GRID_N):
            count = 0
            for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                rr, cc = row + dr, col + dc
                if 0 <= rr < GRID_N and 0 <= cc < GRID_N:
                    count += int(source[rr][cc])
            line.append(int(count == 2))
        output.append(line)
    return output


def _flat(grid: list[list[int]]) -> list[int]:
    return [int(value) for row in grid for value in row]


def _grid(values: list[int]) -> list[list[int]]:
    return [values[index : index + GRID_N] for index in range(0, GRID_N * GRID_N, GRID_N)]


def _distance(left: list[list[int]], right: list[list[int]]) -> int:
    return sum(a != b for a, b in zip(_flat(left), _flat(right), strict=True))


def _matching_index(scene: dict) -> int:
    target = _press(scene["source"])
    matches = [
        index for index, candidate in enumerate(scene["candidates"]) if candidate == target
    ]
    if len(matches) != 1:
        raise ValueError("scene does not have one unique transformed candidate")
    if len(scene["tag_colors"]) != 3 or len(set(scene["tag_colors"])) != 3:
        raise ValueError("candidate frame colors must be unique")
    return matches[0]


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) << 17) ^ 0x4E5052455353)
    for _ in range(10000):
        source = [
            [int(rng.random() < 0.5) for _ in range(GRID_N)] for _ in range(GRID_N)
        ]
        target = _press(source)
        source_ink = sum(_flat(source))
        target_ink = sum(_flat(target))
        if 9 <= source_ink <= 16 and 8 <= target_ink <= 17:
            break
    else:  # pragma: no cover - deterministic sampler has ample valid mass
        raise RuntimeError("could not sample a balanced source")

    dark = [index for index, value in enumerate(_flat(target)) if value]
    light = [index for index, value in enumerate(_flat(target)) if not value]
    rng.shuffle(dark)
    rng.shuffle(light)
    distractors = []
    for offset in range(2):
        values = _flat(target)
        values[dark[offset]] = 0
        values[light[offset]] = 1
        distractors.append(_grid(values))

    target_position = (int(seed) // 3) % 3
    candidates: list[list[list[int]]] = [[], [], []]
    candidates[target_position] = target
    open_positions = [index for index in range(3) if index != target_position]
    candidates[open_positions[0]] = distractors[0]
    candidates[open_positions[1]] = distractors[1]

    answer_color = tuple(COLORS)[int(seed) % 3]
    other_colors = [name for name in COLORS if name != answer_color]
    rng.shuffle(other_colors)
    tag_colors = ["", "", ""]
    tag_colors[target_position] = answer_color
    for position, color in zip(open_positions, other_colors, strict=True):
        tag_colors[position] = color

    return {
        "source": source,
        "candidates": candidates,
        "tag_colors": tag_colors,
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    position: tuple[int, int],
    frame: tuple[int, int, int],
    grid: list[list[int]],
) -> None:
    x, y = position
    draw.rectangle((x, y, x + PANEL - 1, y + PANEL - 1), fill=frame)
    draw.rectangle(
        (
            x + FRAME_WIDTH,
            y + FRAME_WIDTH,
            x + PANEL - FRAME_WIDTH - 1,
            y + PANEL - FRAME_WIDTH - 1,
        ),
        fill=PANEL_FILL,
    )
    for row in range(GRID_N):
        for col in range(GRID_N):
            x0 = x + GRID_OFFSET + col * CELL_PITCH + CELL_INSET
            y0 = y + GRID_OFFSET + row * CELL_PITCH + CELL_INSET
            x1 = x + GRID_OFFSET + (col + 1) * CELL_PITCH - CELL_INSET - 1
            y1 = y + GRID_OFFSET + (row + 1) * CELL_PITCH - CELL_INSET - 1
            draw.rectangle((x0, y0, x1, y1), fill=DARK if grid[row][col] else LIGHT)


def render(scene: dict) -> Image.Image:
    source = scene["source"]
    candidates = scene["candidates"]
    tag_colors = scene["tag_colors"]
    if len(candidates) != 3 or len(tag_colors) != 3:
        raise ValueError("render requires three candidate grids and frame colors")
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, PANEL_POSITIONS[0], SOURCE_FRAME, source)
    for position, grid, color in zip(
        PANEL_POSITIONS[1:], candidates, tag_colors, strict=True
    ):
        _draw_panel(draw, position, COLORS[color], grid)
    return image


def analytic_gold(scene: dict) -> str:
    index = _matching_index(scene)
    return scene["tag_colors"][index]


def margin(scene: dict) -> float:
    target = _press(scene["source"])
    distances = sorted(_distance(target, candidate) for candidate in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        index = _matching_index(scene)
        del index
        target = _press(scene["source"])
        ink = sum(_flat(target))
        return margin(scene) < 2.0 or not (8 <= ink <= 17)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    del scene
    return []
