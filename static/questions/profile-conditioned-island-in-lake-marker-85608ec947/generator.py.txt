from __future__ import annotations

import random
from collections import deque
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 560, 320
GRID_W, GRID_H, CELL = 22, 11, 24
OX, OY = 16, 28
COLORS = {
    "red": (220, 45, 55),
    "green": (20, 150, 85),
    "blue": (45, 100, 220),
}


def _component_labels(mask: list[list[bool]]) -> tuple[list[list[int]], list[list[tuple[int, int]]]]:
    h, w = len(mask), len(mask[0])
    labels = [[-1] * w for _ in range(h)]
    comps: list[list[tuple[int, int]]] = []
    for y in range(h):
        for x in range(w):
            if not mask[y][x] or labels[y][x] >= 0:
                continue
            label = len(comps)
            labels[y][x] = label
            q = deque([(x, y)])
            comp = []
            while q:
                px, py = q.popleft()
                comp.append((px, py))
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < w and 0 <= ny < h and mask[ny][nx] and labels[ny][nx] < 0:
                        labels[ny][nx] = label
                        q.append((nx, ny))
            comps.append(comp)
    return labels, comps


def _motif(kind: str, gap_side: int) -> tuple[set[tuple[int, int]], tuple[int, int]]:
    land = set()
    for x in range(6):
        land.add((x, 1)); land.add((x, 8))
    for y in range(2, 8):
        land.add((0, y)); land.add((5, y))
    island = {(2, 4), (3, 4), (2, 5), (3, 5)}
    land |= island
    if kind == "nested":
        land |= {(2, 0), (3, 0)}
    elif kind == "causeway":
        land |= {(2, 2), (2, 3)}
    elif kind == "outlet":
        gaps = [(2, 1), (3, 8)]
        land.remove(gaps[gap_side])
        bumps = [
            {(1, 9), (2, 9), (3, 9)},
            {(2, 0), (3, 0), (4, 0)},
        ]
        land |= bumps[gap_side]
    else:
        raise ValueError(kind)
    assert len(land) == 30
    return land, (3, 5)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    answer = colors[int(seed) % 3]
    other = [c for c in colors if c != answer]
    rng.shuffle(other)
    role_color = {"nested": answer, "causeway": other[0], "outlet": other[1]}
    roles = ["nested", "causeway", "outlet"]
    rng.shuffle(roles)
    cells = [[0 for _ in range(GRID_W)] for _ in range(GRID_H)]
    markers = []
    for panel, role in enumerate(roles):
        base_x = panel * 8
        land, marker = _motif(role, rng.randrange(2))
        # Independent reflections vary the global appearance while retaining thick topology.
        flip_x, flip_y = rng.choice([False, True]), rng.choice([False, True])
        transformed = set()
        for x, y in land:
            tx, ty = (5 - x if flip_x else x), (9 - y if flip_y else y)
            transformed.add((tx, ty))
        mx, my = marker
        mx, my = (5 - mx if flip_x else mx), (9 - my if flip_y else my)
        for x, y in transformed:
            cells[y][base_x + x] = 1
        markers.append({"color": role_color[role], "x": base_x + mx, "y": my})
    return {"cells": cells, "markers": markers}


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    markers = scene["markers"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    for y, row in enumerate(cells):
        for x, value in enumerate(row):
            if value:
                x0, y0 = OX + x * CELL, OY + y * CELL
                draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=(22, 25, 29))
    for marker in markers:
        cx = OX + int(marker["x"]) * CELL + CELL // 2
        cy = OY + int(marker["y"]) * CELL + CELL // 2
        draw.ellipse((cx - 7, cy - 7, cx + 7, cy + 7), fill=COLORS[marker["color"]])
    return image


def _decision(scene: dict) -> str:
    cells = scene["cells"]
    markers = scene["markers"]
    land = [[bool(v) for v in row] for row in cells]
    land_labels, _ = _component_labels(land)
    water = [[not v for v in row] for row in land]
    water_labels, water_comps = _component_labels(water)
    exterior = {
        i for i, comp in enumerate(water_comps)
        if any(x in (0, GRID_W - 1) or y in (0, GRID_H - 1) for x, y in comp)
    }
    winners = []
    for marker in markers:
        x, y = int(marker["x"]), int(marker["y"])
        component = land_labels[y][x]
        adjacent_water = set()
        for py in range(GRID_H):
            for px in range(GRID_W):
                if land_labels[py][px] != component:
                    continue
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < GRID_W and 0 <= ny < GRID_H and water[ny][nx]:
                        adjacent_water.add(water_labels[ny][nx])
        if adjacent_water and adjacent_water.isdisjoint(exterior):
            winners.append(marker["color"])
    if len(winners) != 1:
        raise ValueError(f"expected one nested marked component, got {winners}")
    return winners[0]


def analytic_gold(scene: dict) -> str:
    return _decision(scene)


def margin(scene: dict) -> float:
    # All passages and separations are full 24-pixel cells; subtract marker outline width.
    _ = scene["cells"], scene["markers"]
    return 20.0


def is_quarantined(scene: dict) -> bool:
    try:
        _decision(scene)
        return False
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene["cells"], scene["markers"]
    return []
