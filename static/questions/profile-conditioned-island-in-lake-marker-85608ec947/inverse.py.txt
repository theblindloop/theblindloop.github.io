from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 45, 55)),
    "green": np.array((20, 150, 85)),
    "blue": np.array((45, 100, 220)),
}


def _components(mask: np.ndarray) -> tuple[np.ndarray, list[list[tuple[int, int]]]]:
    h, w = mask.shape
    labels = np.full((h, w), -1, dtype=np.int32)
    comps = []
    for y in range(h):
        for x in range(w):
            if not mask[y, x] or labels[y, x] >= 0:
                continue
            label = len(comps)
            labels[y, x] = label
            q = deque([(x, y)])
            comp = []
            while q:
                px, py = q.popleft(); comp.append((px, py))
                for nx, ny in ((px-1,py),(px+1,py),(px,py-1),(px,py+1)):
                    if 0 <= nx < w and 0 <= ny < h and mask[ny, nx] and labels[ny, nx] < 0:
                        labels[ny, nx] = label; q.append((nx, ny))
            comps.append(comp)
    return labels, comps


def decision_from_image(image: Image.Image) -> str:
    rgb_image = image.convert("RGB")
    # Topology is deliberately 24 px thick. A scene-free 4x reduction suppresses
    # isolated raster noise while leaving every passage at least six pixels wide.
    reduced = rgb_image.resize((rgb_image.width // 4, rgb_image.height // 4), Image.Resampling.NEAREST)
    arr = np.asarray(reduced, dtype=np.int16)
    white = np.all(arr >= 245, axis=2)
    land = ~white
    land_labels, land_comps = _components(land)
    water_labels, water_comps = _components(white)
    if len(land_comps) < 4 or not water_comps:
        raise ValueError("missing land/water topology")
    h, w = white.shape
    exterior = {
        i for i, comp in enumerate(water_comps)
        if any(x in (0, w-1) or y in (0, h-1) for x, y in comp)
    }
    winners = []
    for name, rgb in PALETTE.items():
        distance = np.max(np.abs(arr - rgb), axis=2)
        ys, xs = np.nonzero(distance <= 3)
        if len(xs) < 4:
            raise ValueError(f"missing {name} marker")
        cx, cy = int(np.median(xs)), int(np.median(ys))
        component = int(land_labels[cy, cx])
        if component < 0:
            raise ValueError("marker not on land")
        adjacent = set()
        for px, py in land_comps[component]:
            for nx, ny in ((px-1,py),(px+1,py),(px,py-1),(px,py+1)):
                if 0 <= nx < w and 0 <= ny < h and white[ny, nx]:
                    adjacent.add(int(water_labels[ny, nx]))
        if adjacent and adjacent.isdisjoint(exterior):
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"ambiguous image predicate: {winners}")
    return winners[0]
