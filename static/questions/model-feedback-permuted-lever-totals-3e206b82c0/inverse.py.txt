"""Independent image-only inverse for the fixed lever diagram."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1000
HEIGHT = 620
CENTER_X = 500
BEAM_Y = 190
SPACING = 78
STACK_TOP = 286
BLOCK_H = 24
BLOCK_GAP = 6
DISTANCES = (1, 2, 3, 4, 5)

_BLUE = np.array((58, 99, 166), dtype=np.int16)
_DARK = np.array((35, 42, 52), dtype=np.int16)
_WHITE = np.array((255, 255, 252), dtype=np.int16)


class OracleError(ValueError):
    pass


def _fraction_near(array: np.ndarray, box: tuple[int, int, int, int], color: np.ndarray, tolerance: float) -> float:
    x0, y0, x1, y1 = box
    patch = array[y0:y1, x0:x1].astype(np.int32)
    if patch.size == 0:
        return 0.0
    distance = np.sqrt(np.sum((patch - color.astype(np.int32)) ** 2, axis=2))
    return float(np.mean(distance <= tolerance))


def _stack_counts(array: np.ndarray, side: str) -> list[int]:
    counts: list[int] = []
    for distance in DISTANCES:
        x = CENTER_X + (distance * SPACING if side == "right" else -distance * SPACING)
        occupied = []
        for level in range(5):
            y = STACK_TOP + level * (BLOCK_H + BLOCK_GAP) + BLOCK_H // 2
            fraction = _fraction_near(array, (x - 8, y - 6, x + 9, y + 7), _BLUE, 18.0)
            if fraction >= 0.88:
                occupied.append(True)
            elif fraction <= 0.05:
                occupied.append(False)
            else:
                raise OracleError(f"ambiguous {side} block at distance {distance}, level {level + 1}")
        count = sum(occupied)
        if occupied != [True] * count + [False] * (5 - count) or not 1 <= count <= 5:
            raise OracleError(f"malformed {side} stack at distance {distance}")
        counts.append(count)
    if sorted(counts) != [1, 2, 3, 4, 5]:
        raise OracleError(f"{side} loads are not one each of 1 through 5")
    return counts


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise OracleError(f"unexpected image size {rgb.size}")
    array = np.asarray(rgb)

    if _fraction_near(array, (82, BEAM_Y - 3, 918, BEAM_Y + 4), _DARK, 22.0) < 0.90:
        raise OracleError("beam is missing or broken")
    if _fraction_near(array, (35, 32, 65, 48), _WHITE, 18.0) < 0.90:
        raise OracleError("background frame is malformed")
    for distance in DISTANCES:
        for sign in (-1, 1):
            x = CENTER_X + sign * distance * SPACING
            if _fraction_near(array, (x - 2, BEAM_Y - 17, x + 3, BEAM_Y + 18), _DARK, 22.0) < 0.65:
                raise OracleError("a distance tick is missing")

    left_counts = _stack_counts(array, "left")
    right_counts = _stack_counts(array, "right")
    left_total = sum(distance * count for distance, count in zip(DISTANCES, left_counts, strict=True))
    right_total = sum(distance * count for distance, count in zip(DISTANCES, right_counts, strict=True))
    if left_total > right_total:
        return "left"
    if right_total > left_total:
        return "right"
    return "same"
