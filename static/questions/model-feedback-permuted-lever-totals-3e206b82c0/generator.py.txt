"""Deterministic renderer for paired permuted lever totals."""

from __future__ import annotations

import itertools
import random
from functools import lru_cache

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1000
HEIGHT = 620
CENTER_X = 500
BEAM_Y = 190
SPACING = 78
STACK_TOP = 286
BLOCK_W = 38
BLOCK_H = 24
BLOCK_GAP = 6
DISTANCES = (1, 2, 3, 4, 5)
LOADS = (1, 2, 3, 4, 5)

BACKGROUND = (250, 249, 245)
INK = (35, 42, 52)
MUTED = (103, 112, 122)
BLOCK_FILL = (58, 99, 166)
PIVOT_FILL = (238, 174, 64)

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
_PERMUTATIONS = tuple(itertools.permutations(LOADS))


def _score(values: tuple[int, ...] | list[int]) -> int:
    return sum(distance * int(count) for distance, count in zip(DISTANCES, values, strict=True))


@lru_cache(maxsize=1)
def _pair_pools() -> dict[str, tuple[tuple[tuple[int, ...], tuple[int, ...]], ...]]:
    left = []
    same = []
    for first in _PERMUTATIONS:
        first_score = _score(first)
        for second in _PERMUTATIONS:
            gap = first_score - _score(second)
            if gap == 1:
                left.append((first, second))
            elif gap == 0 and first != second:
                same.append((first, second))
    right = [(second, first) for first, second in left]
    return {"left": tuple(left), "right": tuple(right), "same": tuple(same)}


def sample_scene(seed: int) -> dict:
    """Choose a balanced label and a deterministic close/equal permutation pair."""
    decision = ("left", "right", "same")[int(seed) % 3]
    pool = _pair_pools()[decision]
    rng = random.Random(int(seed) * 104729 + 1709)
    left, right = pool[rng.randrange(len(pool))]
    return {"left_counts": list(left), "right_counts": list(right)}


def _validated_counts(values: object, key: str) -> list[int]:
    if not isinstance(values, (list, tuple)) or len(values) != 5:
        raise ValueError(f"{key} must contain five counts")
    if any(isinstance(value, bool) or not isinstance(value, int) for value in values):
        raise ValueError(f"{key} counts must be integers")
    result = [int(value) for value in values]
    if sorted(result) != list(LOADS):
        raise ValueError(f"{key} must be a permutation of 1..5")
    return result


def _totals(scene: dict) -> tuple[int, int]:
    left = _validated_counts(scene["left_counts"], "left_counts")
    right = _validated_counts(scene["right_counts"], "right_counts")
    return _score(left), _score(right)


def analytic_gold(scene: dict) -> str:
    left_total, right_total = _totals(scene)
    if left_total > right_total:
        return "left"
    if right_total > left_total:
        return "right"
    return "same"


def margin(scene: dict) -> float:
    """Discrete score clearance; equality is one full block-distance step from a sign."""
    left_total, right_total = _totals(scene)
    return float(max(1, abs(left_total - right_total)))


def is_quarantined(scene: dict) -> bool:
    try:
        left_total, right_total = _totals(scene)
    except (KeyError, TypeError, ValueError):
        return True
    return abs(left_total - right_total) > 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(_BOLD_PATH if bold else _FONT_PATH, size=size)


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _x(side: str, distance: int) -> int:
    return CENTER_X + (distance * SPACING if side == "right" else -distance * SPACING)


def render(scene: dict) -> Image.Image:
    left_counts = _validated_counts(scene["left_counts"], "left_counts")
    right_counts = _validated_counts(scene["right_counts"], "right_counts")
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((24, 22, WIDTH - 24, HEIGHT - 22), radius=18, outline=(210, 207, 198), width=3, fill=(255, 255, 252))
    _centered(draw, (CENTER_X, 57), "BLOCK COUNT × DISTANCE", _font(31, bold=True), INK)
    _centered(draw, (CENTER_X, 91), "ADD THE FIVE PRODUCTS ON EACH SIDE", _font(18, bold=True), MUTED)
    _centered(draw, (194, 104), "LEFT", _font(23, bold=True), INK)
    _centered(draw, (806, 104), "RIGHT", _font(23, bold=True), INK)

    draw.rounded_rectangle((75, BEAM_Y - 7, 925, BEAM_Y + 7), radius=5, fill=INK)
    for side, counts in (("left", left_counts), ("right", right_counts)):
        for distance, count in zip(DISTANCES, counts, strict=True):
            x = _x(side, distance)
            draw.line((x, BEAM_Y - 20, x, BEAM_Y + 20), fill=INK, width=5)
            _centered(draw, (x, 146), str(distance), _font(25, bold=True), INK)
            draw.line((x, BEAM_Y + 8, x, STACK_TOP - 12), fill=MUTED, width=4)
            draw.line((x - 14, STACK_TOP - 12, x + 14, STACK_TOP - 12), fill=MUTED, width=4)
            for level in range(count):
                top = STACK_TOP + level * (BLOCK_H + BLOCK_GAP)
                draw.rounded_rectangle(
                    (x - BLOCK_W // 2, top, x + BLOCK_W // 2, top + BLOCK_H),
                    radius=3,
                    fill=BLOCK_FILL,
                    outline=INK,
                    width=2,
                )

    draw.polygon(
        ((CENTER_X, BEAM_Y + 8), (CENTER_X - 48, 272), (CENTER_X + 48, 272)),
        fill=PIVOT_FILL,
        outline=INK,
    )
    draw.ellipse((CENTER_X - 10, BEAM_Y - 10, CENTER_X + 10, BEAM_Y + 10), fill=PIVOT_FILL, outline=INK, width=3)
    _centered(draw, (CENTER_X, 302), "PIVOT", _font(18, bold=True), INK)

    draw.rounded_rectangle((300, 488, 700, 558), radius=10, fill=(239, 244, 251), outline=(155, 169, 190), width=2)
    _centered(draw, (CENTER_X, 510), "one blue block = 1", _font(21, bold=True), INK)
    _centered(draw, (CENTER_X, 540), "side total = Σ(blocks × distance)", _font(20), INK)
    return image
