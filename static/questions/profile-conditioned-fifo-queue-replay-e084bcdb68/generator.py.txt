"""Deterministic visible FIFO queue world."""
from __future__ import annotations

import random
from copy import deepcopy
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 680
SCALE = 3
PALETTES = (
    {"queue": (45, 158, 181), "enqueue": (49, 151, 104), "dequeue": (232, 132, 45)},
    {"queue": (41, 145, 174), "enqueue": (37, 139, 114), "dequeue": (221, 113, 54)},
)
INK = (30, 36, 45)
PAPER = (250, 248, 242)


def _replay(initial_queue: list[int], operations: list[dict[str, Any]]) -> list[int]:
    queue = [int(value) for value in initial_queue]
    for operation in operations:
        if operation["kind"] == "enqueue":
            queue.append(int(operation["label"]))
        elif operation["kind"] == "dequeue":
            if not queue:
                raise ValueError("dequeue from an empty queue")
            queue.pop(0)
        else:
            raise ValueError("unknown operation")
    if not queue:
        raise ValueError("final queue is empty")
    return queue


def sample_scene(seed: int) -> dict:
    """Sample a valid scene; consecutive seeds counterbalance A/B/C/D exactly."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x51F0A9)
    initial_len = rng.choice((2, 3, 4))
    operation_count = 8
    for _ in range(500):
        kinds = [rng.choice(("enqueue", "dequeue")) for _ in range(operation_count)]
        if kinds.count("enqueue") < 2 or kinds.count("dequeue") < 2:
            continue
        level = initial_len
        valid = True
        for kind in kinds:
            if kind == "enqueue":
                level += 1
            elif level > 1:
                level -= 1
            else:
                valid = False
                break
        if valid and 1 <= level <= 5:
            break
    else:
        raise RuntimeError("could not sample a valid queue program")

    initial = [rng.randint(1, 4) for _ in range(initial_len)]
    operations: list[dict[str, Any]] = []
    for kind in kinds:
        if kind == "enqueue":
            operations.append({"kind": kind, "label": rng.randint(1, 4)})
        else:
            operations.append({"kind": kind})

    # A global cyclic relabeling makes the answer seed%4 while preserving all
    # operation structure and relative label statistics.
    current = _replay(initial, operations)[0]
    target = seed % 4 + 1
    shift = (target - current) % 4
    initial = [((value - 1 + shift) % 4) + 1 for value in initial]
    for operation in operations:
        if operation["kind"] == "enqueue":
            operation["label"] = ((operation["label"] - 1 + shift) % 4) + 1

    return {
        "initial_queue": initial,
        "operations": operations,
        "palette_variant": (seed // 4) % len(PALETTES),
        "layout_variant": (seed // 8) % 2,
    }


def analytic_gold(scene: dict) -> str:
    front = _replay(scene["initial_queue"], scene["operations"])[0]
    return "ABCD"[front - 1]


def margin(scene: dict) -> float:
    """Minimum colored-card clearance in output pixels (parser runner-up margin)."""
    return 18.0 if int(scene["layout_variant"]) == 0 else 26.0


def is_quarantined(scene: dict) -> bool:
    try:
        _replay(scene["initial_queue"], scene["operations"])
    except (KeyError, TypeError, ValueError):
        return True
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every latent field is rendered; there is no pixel-identical latent alias.
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _pips(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], count: int) -> None:
    x0, y0, x1, y1 = box
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    dx, dy, radius = 19 * SCALE, 17 * SCALE, 7 * SCALE
    patterns = {
        1: [(0, 0)],
        2: [(-dx, 0), (dx, 0)],
        3: [(-dx, -dy), (0, 0), (dx, dy)],
        4: [(-dx, -dy), (dx, -dy), (-dx, dy), (dx, dy)],
    }
    for ox, oy in patterns[int(count)]:
        draw.ellipse((cx + ox - radius, cy + oy - radius, cx + ox + radius, cy + oy + radius), fill=INK)


def _rounded_card(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    fill: tuple[int, int, int],
    *,
    radius: int = 12,
) -> None:
    draw.rounded_rectangle(box, radius=radius * SCALE, fill=fill, outline=INK, width=3 * SCALE)


def _center_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    initial = [int(value) for value in scene["initial_queue"]]
    operations = list(scene["operations"])
    palette = PALETTES[int(scene["palette_variant"]) % len(PALETTES)]
    layout_variant = int(scene["layout_variant"])
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), PAPER)
    draw = ImageDraw.Draw(canvas)
    title = _font(32 * SCALE, bold=True)
    body = _font(21 * SCALE)
    small = _font(18 * SCALE, bold=True)
    _center_text(draw, (WIDTH * SCALE // 2, 38 * SCALE), "FIRST-IN QUEUE", title, INK)
    _center_text(draw, (WIDTH * SCALE // 2, 82 * SCALE), "1 pip = A     2 pips = B     3 pips = C     4 pips = D", body, INK)

    # Initial queue.
    draw.text((84 * SCALE, 131 * SCALE), "STARTING QUEUE", font=small, fill=INK)
    card_w, card_h = 116 * SCALE, 112 * SCALE
    gap = (26 if layout_variant else 18) * SCALE
    total = len(initial) * card_w + (len(initial) - 1) * gap
    start_x = (WIDTH * SCALE - total) // 2
    top_y = 170 * SCALE
    for index, label in enumerate(initial):
        x0 = start_x + index * (card_w + gap)
        box = (x0, top_y, x0 + card_w, top_y + card_h)
        _rounded_card(draw, box, palette["queue"])
        _pips(draw, (x0 + 18 * SCALE, top_y + 18 * SCALE, x0 + card_w - 18 * SCALE, top_y + card_h - 18 * SCALE), label)
    arrow_y = (top_y + card_h // 2)
    draw.line((48 * SCALE, arrow_y, (start_x - 18 * SCALE), arrow_y), fill=INK, width=4 * SCALE)
    draw.polygon(((start_x - 18 * SCALE, arrow_y), (start_x - 35 * SCALE, arrow_y - 10 * SCALE), (start_x - 35 * SCALE, arrow_y + 10 * SCALE)), fill=INK)
    draw.text((44 * SCALE, (arrow_y + 18 * SCALE)), "FRONT", font=small, fill=INK)
    draw.text(((start_x + total + 18 * SCALE), (arrow_y - 12 * SCALE)), "BACK", font=small, fill=INK)

    # Ordered operation cards.
    draw.text((84 * SCALE, 339 * SCALE), "READ LEFT TO RIGHT", font=small, fill=INK)
    op_w, op_h = 112 * SCALE, 176 * SCALE
    op_gap = (26 if layout_variant else 18) * SCALE
    total_ops = len(operations) * op_w + (len(operations) - 1) * op_gap
    op_x = (WIDTH * SCALE - total_ops) // 2
    op_y = 380 * SCALE
    symbol_font = _font(46 * SCALE, bold=True)
    number_font = _font(17 * SCALE, bold=True)
    for index, operation in enumerate(operations):
        x0 = op_x + index * (op_w + op_gap)
        kind = operation["kind"]
        fill = palette["enqueue"] if kind == "enqueue" else palette["dequeue"]
        box = (x0, op_y, x0 + op_w, op_y + op_h)
        _rounded_card(draw, box, fill)
        _center_text(draw, (x0 + op_w // 2, op_y + 43 * SCALE), "+" if kind == "enqueue" else "−", symbol_font, (255, 255, 255))
        if kind == "enqueue":
            _pips(draw, (x0 + 20 * SCALE, op_y + 86 * SCALE, x0 + op_w - 20 * SCALE, op_y + op_h - 18 * SCALE), int(operation["label"]))
        else:
            _center_text(draw, (x0 + op_w // 2, op_y + 125 * SCALE), "FRONT", number_font, INK)
        _center_text(draw, (x0 + op_w // 2, op_y + op_h + 24 * SCALE), str(index + 1), number_font, INK)

    _center_text(draw, (WIDTH * SCALE // 2, 632 * SCALE), "Green + : add to BACK        Orange − : remove FRONT", body, INK)
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
