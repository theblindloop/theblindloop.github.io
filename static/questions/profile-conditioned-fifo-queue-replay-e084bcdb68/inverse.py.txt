"""Independent image-only inverse for the FIFO card grammar."""
from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "queue": ((45, 158, 181), (41, 145, 174)),
    "enqueue": ((49, 151, 104), (37, 139, 114)),
    "dequeue": ((232, 132, 45), (221, 113, 54)),
}


def _mask_for(arr: np.ndarray, name: str) -> np.ndarray:
    pixels = arr.astype(np.int32)
    distances = [np.sum((pixels - np.asarray(rgb, dtype=np.int32)) ** 2, axis=2) for rgb in _COLORS[name]]
    return np.minimum.reduce(distances) < 38 * 38


def _runs(values: np.ndarray, minimum: int = 12) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = previous = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value > previous + 1:
            if previous - start + 1 >= minimum:
                groups.append((start, previous + 1))
            start = value
        previous = value
    if previous - start + 1 >= minimum:
        groups.append((start, previous + 1))
    return groups


def _boxes(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    boxes = []
    for x0, x1 in _runs(mask.sum(axis=0) > 20, 40):
        ys = np.flatnonzero(mask[:, x0:x1].sum(axis=1) > 8)
        if len(ys):
            boxes.append((x0, int(ys[0]), x1, int(ys[-1]) + 1))
    return boxes


def _pip_count(arr: np.ndarray, box: tuple[int, int, int, int], lower_only: bool) -> int:
    x0, y0, x1, y1 = box
    padx = max(8, int((x1 - x0) * 0.15))
    top = y0 + (int((y1 - y0) * 0.46) if lower_only else max(8, int((y1 - y0) * 0.12)))
    bottom = y1 - max(8, int((y1 - y0) * 0.10))
    crop = arr[top:bottom, x0 + padx:x1 - padx]
    black = np.all(crop < 72, axis=2)
    seen = np.zeros(black.shape, dtype=bool)
    count = 0
    height, width = black.shape
    for yy, xx in zip(*np.nonzero(black)):
        if seen[yy, xx]:
            continue
        stack = [(int(yy), int(xx))]
        seen[yy, xx] = True
        area = 0
        minx = maxx = int(xx)
        miny = maxy = int(yy)
        while stack:
            cy, cx = stack.pop()
            area += 1
            minx, maxx = min(minx, cx), max(maxx, cx)
            miny, maxy = min(miny, cy), max(maxy, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and black[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        bw, bh = maxx - minx + 1, maxy - miny + 1
        if 75 <= area <= 350 and 8 <= bw <= 24 and 8 <= bh <= 24:
            count += 1
    if count not in (1, 2, 3, 4):
        raise ValueError(f"could not decode pip label (found {count})")
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    queue_boxes = _boxes(_mask_for(arr, "queue"))
    enqueue_boxes = _boxes(_mask_for(arr, "enqueue"))
    dequeue_boxes = _boxes(_mask_for(arr, "dequeue"))
    if not 2 <= len(queue_boxes) <= 4:
        raise ValueError(f"expected 2-4 starting tiles, found {len(queue_boxes)}")
    operations = [(box[0], "enqueue", box) for box in enqueue_boxes]
    operations += [(box[0], "dequeue", box) for box in dequeue_boxes]
    operations.sort(key=lambda item: item[0])
    if len(operations) != 8:
        raise ValueError(f"expected 8 operation cards, found {len(operations)}")
    q = deque(_pip_count(arr, box, False) for box in sorted(queue_boxes))
    for _, kind, box in operations:
        if kind == "enqueue":
            q.append(_pip_count(arr, box, True))
        else:
            if not q:
                raise ValueError("visible program dequeues an empty queue")
            q.popleft()
    if not q:
        raise ValueError("visible program leaves no front tile")
    return "ABCD"[q[0] - 1]
