"""Deterministic renderer for the paired-link parity world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 248, 242)
NAVY = (24, 46, 70)
CYAN = (0, 166, 196)
ORANGE = (235, 116, 42)
PALE = (211, 215, 214)
MARKER_RADIUS = 7
STROKE = 6


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; successive integer seeds alternate answer class."""
    rng = random.Random(int(seed))
    parity = int(seed) & 1
    if parity:
        match_bits = [0, 1] if ((int(seed) // 2) & 1) else [1, 0]
    else:
        match_bits = [0, 0] if ((int(seed) // 2) & 1) else [1, 1]

    # Lattice translations keep the two small evidence islands distinct under
    # the public overlapping 8x8 ablation probe without fixing their location.
    offset_x = rng.choice([160, 224, 288, 352])
    offset_y = rng.choice([160, 224, 288, 352])
    vertical = bool(rng.getrandbits(1))
    component_order = [0, 1]
    rng.shuffle(component_order)
    paths = []
    for slot, bit_index in enumerate(component_order):
        separation = 112
        axis = -separation // 2 if slot == 0 else separation // 2
        jitter = 0
        bend = 6 if slot == 0 else -6
        if vertical:
            points = [[-16, axis + jitter], [0, axis + jitter + bend], [16, axis + jitter]]
        else:
            points = [[axis + jitter, -16], [axis + jitter + bend, 0], [axis + jitter, 16]]
        left_color = rng.choice(["cyan", "orange"])
        same = bool(match_bits[bit_index])
        right_color = left_color if same else ("orange" if left_color == "cyan" else "cyan")
        if rng.getrandbits(1):
            points.reverse()
            left_color, right_color = right_color, left_color
        paths.append(
            {
                "points": points,
                "color_a": left_color,
                "color_b": right_color,
            }
        )

    decorations = []
    for _ in range(8):
        # Remote, light, and deliberately outside the local evidence inset.
        side = rng.randrange(4)
        if side == 0:
            x, y = rng.randint(35, 475), rng.randint(28, 70)
        elif side == 1:
            x, y = rng.randint(35, 475), rng.randint(442, 484)
        elif side == 2:
            x, y = rng.randint(28, 70), rng.randint(90, 422)
        else:
            x, y = rng.randint(442, 484), rng.randint(90, 422)
        decorations.append([x, y, rng.choice([4, 6, 8])])

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "paths": paths,
        "decorations": decorations,
    }


def _absolute_points(scene: dict, path: dict) -> list[tuple[int, int]]:
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    return [(ox + int(x), oy + int(y)) for x, y in path["points"]]


def _rgb(name: str) -> tuple[int, int, int]:
    if name == "cyan":
        return CYAN
    if name == "orange":
        return ORANGE
    raise ValueError(f"unknown endpoint color {name!r}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=PALE, width=2)
    for path in scene["paths"]:
        points = _absolute_points(scene, path)
        draw.line(points, fill=NAVY, width=STROKE, joint="curve")
        for point, key in ((points[0], "color_a"), (points[-1], "color_b")):
            x, y = point
            draw.ellipse(
                (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
                fill=_rgb(path[key]),
                outline=BACKGROUND,
                width=2,
            )
    return image


def _same_color_bits(scene: dict) -> list[bool]:
    return [path["color_a"] == path["color_b"] for path in scene["paths"]]


def analytic_gold(scene: dict) -> str:
    return "odd" if sum(_same_color_bits(scene)) % 2 else "even"


def _segment_distance(a: tuple[int, int], b: tuple[int, int], c: tuple[int, int], d: tuple[int, int]) -> float:
    # Conservative endpoint-to-polyline separation is sufficient for quarantine.
    return min(math.dist(p, q) for p in (a, b) for q in (c, d))


def margin(scene: dict) -> float:
    point_sets = [_absolute_points(scene, path) for path in scene["paths"]]
    separation = _segment_distance(point_sets[0][0], point_sets[0][-1], point_sets[1][0], point_sets[1][-1])
    return float(separation - 2 * MARKER_RADIUS - STROKE)


def is_quarantined(scene: dict) -> bool:
    return len(scene["paths"]) != 2 or margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
