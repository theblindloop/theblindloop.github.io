"""Independent image-only inverse arm for paired link parity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((24, 46, 70), dtype=np.int16)
CYAN = np.array((0, 166, 196), dtype=np.int16)
ORANGE = np.array((235, 116, 42), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centroid(points: np.ndarray) -> np.ndarray:
    return points.mean(axis=0)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    navy_parts = _components(_near(array, NAVY), minimum=35)
    if len(navy_parts) != 2:
        raise ValueError(f"expected exactly two complete navy link components, found {len(navy_parts)}")

    markers: list[tuple[str, np.ndarray]] = []
    for label, color in (("cyan", CYAN), ("orange", ORANGE)):
        for part in _components(_near(array, color), minimum=55):
            markers.append((label, part))
    if len(markers) != 4:
        raise ValueError(f"expected four endpoint markers, found {len(markers)}")

    assignments: list[list[str]] = [[], []]
    for label, marker in markers:
        center = _centroid(marker)
        distances = []
        for link in navy_parts:
            # Nearest sampled ink pixel binds the endpoint marker to its trace.
            step = max(1, len(link) // 350)
            distances.append(float(np.linalg.norm(link[::step] - center, axis=1).min()))
        order = np.argsort(distances)
        if distances[int(order[1])] - distances[int(order[0])] < 20.0:
            raise ValueError("endpoint-to-link assignment is ambiguous")
        assignments[int(order[0])].append(label)

    if sorted(len(values) for values in assignments) != [2, 2]:
        raise ValueError("each trace must own exactly two endpoint markers")
    equalities = [values[0] == values[1] for values in assignments]
    return "odd" if sum(equalities) % 2 else "even"
