"""Independent image-only inverse arm for the fragmented-ring world.

The oracle receives one raster. It locates the two colored candidate dots,
extracts dark fragment pixels, and independently fits three radial-distance
clusters around each dot. The lower residual selects the shared center.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class UnreadableImage(ValueError):
    pass


def _color_center(array: np.ndarray, color: tuple[int, int, int], name: str) -> tuple[float, float]:
    # int32 is intentional: squared 8-bit channel deltas overflow int16.
    delta = array.astype(np.int32) - np.asarray(color, dtype=np.int32)
    mask = np.sum(delta * delta, axis=2) < 48 * 48
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise UnreadableImage(f"missing {name} candidate dot")
    return float(xs.mean()), float(ys.mean())


def _three_band_residual(distances: np.ndarray) -> float:
    if len(distances) < 250:
        raise UnreadableImage("too little curved-fragment ink")
    centers = np.quantile(distances, [0.18, 0.50, 0.82]).astype(np.float64)
    for _ in range(40):
        labels = np.abs(distances[:, None] - centers[None, :]).argmin(axis=1)
        # A wrong candidate can lose a tail cluster during Lloyd updates. Keep
        # that center fixed: it remains a valid, deliberately poor runner-up fit.
        updated = np.array(
            [distances[labels == k].mean() if np.any(labels == k) else centers[k] for k in range(3)]
        )
        if np.max(np.abs(updated - centers)) < 1e-7:
            centers = updated
            break
        centers = updated
    labels = np.abs(distances[:, None] - centers[None, :]).argmin(axis=1)
    residual = distances - centers[labels]
    return float(np.sqrt(np.mean(residual * residual)))


def _score(ys: np.ndarray, xs: np.ndarray, center: tuple[float, float]) -> float:
    cx, cy = center
    distances = np.hypot(xs.astype(np.float64) - cx, ys.astype(np.float64) - cy)
    return _three_band_residual(distances)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 64 or array.shape[1] < 64:
        raise UnreadableImage("invalid raster")
    cyan = _color_center(array, (0, 169, 207), "cyan")
    magenta = _color_center(array, (218, 52, 148), "magenta")
    # Dark neutral pixels are arc ink. Saturated candidate colors are excluded.
    channel_spread = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    dark = (array.max(axis=2) < 105) & (channel_spread < 45)
    ys, xs = np.nonzero(dark)
    if len(xs) < 250:
        raise UnreadableImage("missing black curved fragments")
    cyan_score = _score(ys, xs, cyan)
    magenta_score = _score(ys, xs, magenta)
    gap = abs(cyan_score - magenta_score)
    if not np.isfinite(gap) or gap < 3.0:
        raise UnreadableImage("ambiguous shared center")
    return "cyan" if cyan_score < magenta_score else "magenta"
