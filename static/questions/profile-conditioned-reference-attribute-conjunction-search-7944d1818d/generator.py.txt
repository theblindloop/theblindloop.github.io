"""Renderer and analytic contract for the reference-attribute search world."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 900
HEIGHT = 720
BACKGROUND = (250, 250, 248)
INK = (28, 47, 70)
PANEL_LINE = (184, 190, 198)
SHAPES = ("circle", "square", "triangle")
PANEL_LABELS = ("A", "B", "C", "D")
STRIPE_TOLERANCE_DEGREES = 14.0
NEAR_BOUNDARY_BAND_DEGREES = 2.0
PANEL_BOXES = (
    (35, 220, 425, 450),
    (475, 220, 865, 450),
    (35, 470, 425, 700),
    (475, 470, 865, 700),
)
SLOT_OFFSETS = ((105, 85), (285, 85), (105, 175), (285, 175))
REFERENCE_CENTER = (140, 100)


def _slot_center(panel: int, slot: int) -> tuple[float, float]:
    x0, y0, _, _ = PANEL_BOXES[panel]
    dx, dy = SLOT_OFFSETS[slot]
    return (x0 + dx, y0 + dy)


def _angle_delta(first: float, second: float) -> float:
    """Smallest difference between undirected line angles in degrees."""

    return abs((float(first) - float(second) + 90.0) % 180.0 - 90.0)


def _shape_points(shape: str, center: tuple[float, float], scale: float) -> list[tuple[float, float]]:
    cx, cy = center
    half = 38.0 * scale
    if shape == "triangle":
        # Centroid-centered equilateral-like outline; its inradius leaves the
        # short stripe safely detached from the outer contour at every angle.
        height = 1.65 * half
        return [
            (cx, cy - 2.0 * height / 3.0),
            (cx - half, cy + height / 3.0),
            (cx + half, cy + height / 3.0),
        ]
    return []


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[float, float], scale: float, ss: int) -> None:
    cx, cy = center
    half = 38.0 * scale
    width = max(2, int(round(6.5 * scale * ss)))
    if shape == "circle":
        box = tuple(int(round(v * ss)) for v in (cx - half, cy - half, cx + half, cy + half))
        draw.ellipse(box, outline=INK, width=width)
    elif shape == "square":
        box = tuple(int(round(v * ss)) for v in (cx - half, cy - half, cx + half, cy + half))
        draw.rectangle(box, outline=INK, width=width)
    elif shape == "triangle":
        points = [(int(round(x * ss)), int(round(y * ss))) for x, y in _shape_points(shape, center, scale)]
        draw.line(points + [points[0]], fill=INK, width=width, joint="curve")
    else:
        raise ValueError(f"unknown shape {shape!r}")


def _draw_object(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    shape: str,
    stripe_angle: float,
    scale: float,
    ss: int,
) -> None:
    _draw_shape(draw, shape, center, scale, ss)
    cx, cy = center
    half_len = 14.0 * scale
    angle = math.radians(float(stripe_angle))
    dx, dy = half_len * math.cos(angle), half_len * math.sin(angle)
    width = max(2, int(round(6.0 * scale * ss)))
    draw.line(
        [(int(round((cx - dx) * ss)), int(round((cy - dy) * ss))),
         (int(round((cx + dx) * ss)), int(round((cy + dy) * ss)))],
        fill=INK,
        width=width,
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; the seed also counterbalances panel identity."""

    rng = random.Random(int(seed))
    reference_shape = SHAPES[rng.randrange(len(SHAPES))]
    other_shapes = [shape for shape in SHAPES if shape != reference_shape]
    reference_angle = rng.uniform(0.0, 180.0)
    object_scale = rng.uniform(0.93, 1.07)
    target_panel = int(seed) % 4
    near_boundary = (int(seed) % 7 == 0) or (int(seed) % 11 == 0)
    # Near scenes put the shape-only decoy just outside the declared stripe
    # tolerance. It remains a non-match while making the visible boundary
    # genuinely close and therefore quarantinable.
    base_offsets = [0.0, 15.5 if near_boundary else 48.0, -48.0, 82.0]
    objects: list[dict] = []

    for panel in range(4):
        shape_slots = [reference_shape, other_shapes[0], other_shapes[1], other_shapes[0]]
        offsets = list(base_offsets)
        rng.shuffle(shape_slots)
        rng.shuffle(offsets)
        reference_slot = shape_slots.index(reference_shape)
        zero_slot = offsets.index(0.0)
        if panel == target_panel:
            if reference_slot != zero_slot:
                offsets[reference_slot], offsets[zero_slot] = offsets[zero_slot], offsets[reference_slot]
        elif reference_slot == zero_slot:
            swap_slot = (reference_slot + 1) % 4
            offsets[reference_slot], offsets[swap_slot] = offsets[swap_slot], offsets[reference_slot]

        for slot, (shape, offset) in enumerate(zip(shape_slots, offsets)):
            # Compact positional records keep the analytic scene access audit
            # at the public top-level field ``objects``; the four slots are
            # [panel, slot, shape, stripe_angle].
            objects.append([panel, slot, shape, round(reference_angle + offset, 6)])

    rng.shuffle(objects)
    return {
        "reference_shape": reference_shape,
        "reference_angle": round(reference_angle, 6),
        "objects": objects,
        "object_scale": round(object_scale, 6),
        "near_boundary": near_boundary,
    }


def render(scene: dict) -> Image.Image:
    """Render one fixed-layout page; all label-bearing fields are visible."""

    ss = 4
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x0, y0, x1, y1 in PANEL_BOXES:
        draw.rectangle(tuple(int(round(v * ss)) for v in (x0, y0, x1, y1)), outline=PANEL_LINE, width=2 * ss)
    draw.rectangle((25 * ss, 20 * ss, 255 * ss, 180 * ss), outline=PANEL_LINE, width=2 * ss)

    scale = float(scene["object_scale"])
    _draw_object(draw, REFERENCE_CENTER, scene["reference_shape"], float(scene["reference_angle"]), scale, ss)
    for obj in scene["objects"]:
        center = _slot_center(int(obj[0]), int(obj[1]))
        _draw_object(draw, center, obj[2], float(obj[3]), scale, ss)

    image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    labels = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for label, (x0, y0, _, _) in zip(PANEL_LABELS, PANEL_BOXES):
        labels.text((x0 + 13, y0 + 8), label, fill=INK, font=font)
    return image


def _object_match_score(obj: dict, reference_shape: str, reference_angle: float) -> int:
    shape_match = int(obj[2] == reference_shape)
    stripe_match = int(_angle_delta(float(obj[3]), reference_angle) <= STRIPE_TOLERANCE_DEGREES)
    return shape_match + stripe_match


def _scores(scene: dict) -> list[tuple[int, int, int]]:
    """Return (score, panel, slot) for every object, in scene-list order."""

    reference_shape = scene["reference_shape"]
    reference_angle = float(scene["reference_angle"])
    objects = scene["objects"]
    return [
        (_object_match_score(obj, reference_shape, reference_angle), int(obj[0]), int(obj[1]))
        for obj in objects
    ]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    if not scores:
        raise ValueError("scene has no candidate objects")
    top = max(score for score, _, _ in scores)
    winners = [item for item in scores if item[0] == top]
    if top != 2 or len(winners) != 1:
        raise ValueError("scene does not contain a unique two-attribute winner")
    return PANEL_LABELS[winners[0][1]]


def _boundary_clearance(scene: dict) -> float:
    reference_angle = float(scene["reference_angle"])
    return min(
        abs(_angle_delta(float(obj[3]), reference_angle) - STRIPE_TOLERANCE_DEGREES)
        for obj in scene["objects"]
    )


def margin(scene: dict) -> float:
    """Runner-up gap, capped by the closest visible stripe-boundary clearance."""

    scores = sorted((score for score, _, _ in _scores(scene)), reverse=True)
    if len(scores) < 2 or scores[0] != 2:
        return 0.0
    runner_up = scores[1]
    top_gap = (scores[0] - runner_up) / 2.0
    boundary_gap = _boundary_clearance(scene) / STRIPE_TOLERANCE_DEGREES
    return round(min(top_gap, boundary_gap), 6)


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    if not scores:
        return True
    top = max(score for score, _, _ in scores)
    if top != 2 or sum(score == top for score, _, _ in scores) != 1:
        return True
    return margin(scene) < 0.2 or _boundary_clearance(scene) < NEAR_BOUNDARY_BAND_DEGREES


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden alias is claimed; every label-bearing field is rendered."""

    return []
