"""Deterministic cross-controlled correspondence boards."""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1460, 800
N = 6
DECISIONS = tuple("ABCDEF")
INK = (35, 39, 48)
PAPER = (246, 247, 244)
WHITE = (255, 255, 255)
GRID = (116, 123, 132)
START = (198, 38, 139)
FILLED_HEAD = (48, 92, 150)
HOLLOW_HEAD = (141, 75, 145)
SOURCE_COLORS = ((220, 55, 62), (35, 105, 210), (20, 153, 100), (235, 137, 25), (126, 70, 190), (10, 157, 177))
TARGET_COLORS = ((119, 63, 45), (224, 91, 168), (88, 126, 35), (34, 65, 139), (191, 103, 26), (72, 72, 78))
CELL = 58
GRID_Y = 284
SOURCE_X = 76
TARGET_X = 1036
ROUTE_SOURCE_X = 505
CONTROL_SOURCE_X = 585
FILLED_ICON_X = 650
FILLED_TARGET_X = 721
HOLLOW_ICON_X = 817
HOLLOW_TARGET_X = 896
ROUTE_Y = 302
ROUTE_STEP = 55
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

_CANONICAL = tuple(tuple(1 if row in {col, (col + 1) % N, (col + 3) % N} else 0 for col in range(N)) for row in range(N))


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _valid_matrix(matrix: object) -> bool:
    return isinstance(matrix, list) and len(matrix) == N and all(isinstance(row, list) and len(row) == N and all(value in (0, 1) for value in row) for row in matrix)


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {int(tag): row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def _route(signature: tuple[int, ...], controls: list[int], filled: list[int], hollow: list[int]) -> tuple[int, ...]:
    routed: list[int | None] = [None] * N
    for source_tag, value in enumerate(signature):
        control_value = signature[int(controls[source_tag])]
        destination = int(filled[source_tag] if control_value else hollow[source_tag])
        if destination not in range(N) or routed[destination] is not None:
            raise ValueError("condition-selected destinations are not a permutation")
        routed[destination] = int(value)
    if any(value is None for value in routed):
        raise ValueError("incomplete routing")
    return tuple(int(value) for value in routed)


def _routing_maps(signature: tuple[int, ...], controls: list[int], rng: random.Random) -> tuple[list[int], list[int]]:
    selected = list(range(N))
    rng.shuffle(selected)
    conditions = [signature[controls[index]] for index in range(N)]
    ones = [index for index, value in enumerate(conditions) if value]
    zeros = [index for index, value in enumerate(conditions) if not value]
    if len(ones) != 3 or len(zeros) != 3:
        raise ValueError("routing construction expects balanced signature")
    filled = selected.copy()
    hollow = selected.copy()
    for group, mapping in ((zeros, filled), (ones, hollow)):
        shift = 1 + rng.randrange(len(group) - 1)
        for offset, source_tag in enumerate(group):
            mapping[source_tag] = selected[group[(offset + shift) % len(group)]]
    return filled, hollow


def _target_signatures(gold: tuple[int, ...], rng: random.Random) -> list[tuple[int, ...]]:
    pool = [bits for bits in itertools.product((0, 1), repeat=N) if sum(bits) == 3 and bits != gold]
    rng.shuffle(pool)
    needed = tuple(3 - value for value in gold)
    for others in itertools.combinations(pool, N - 1):
        if all(sum(bits[row] for bits in others) == needed[row] for row in range(N)):
            return [gold, *others]
    raise RuntimeError("could not construct balanced target registry")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    entity_order = list(range(N))
    source_tags = list(range(N))
    target_tags = list(range(N))
    route_order = list(range(N))
    rng.shuffle(entity_order)
    rng.shuffle(source_tags)
    rng.shuffle(target_tags)
    rng.shuffle(route_order)
    start_index = rng.randrange(N)
    source = [[_CANONICAL[tag][entity] for entity in entity_order] for tag in source_tags]
    start_signature = _signatures(source, source_tags)[start_index]
    controls = list(range(N))
    while any(index == value for index, value in enumerate(controls)):
        rng.shuffle(controls)
    filled, hollow = _routing_maps(start_signature, controls, rng)
    routed = _route(start_signature, controls, filled, hollow)
    target_signatures = _target_signatures(routed, rng)
    gold_index = int(seed) % N
    target_signatures[0], target_signatures[gold_index] = target_signatures[gold_index], target_signatures[0]
    target = [[target_signatures[col][tag] for col in range(N)] for tag in target_tags]
    scene = {"source_matrix": source, "target_matrix": target, "source_row_tags": source_tags, "target_row_tags": target_tags, "route_order": route_order, "control_mapping": controls, "filled_mapping": filled, "hollow_mapping": hollow, "start_index": start_index}
    if is_quarantined(scene):
        raise RuntimeError("constructed paired-control registry failed invariants")
    return scene


def analytic_gold(scene: dict) -> str:
    source = _signatures(scene["source_matrix"], scene["source_row_tags"])
    target = _signatures(scene["target_matrix"], scene["target_row_tags"])
    routed = _route(source[int(scene["start_index"])], scene["control_mapping"], scene["filled_mapping"], scene["hollow_mapping"])
    matches = [index for index, signature in enumerate(target) if signature == routed]
    if len(matches) != 1:
        raise ValueError("routed START does not have one target")
    return DECISIONS[matches[0]]


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    source = _signatures(scene["source_matrix"], scene["source_row_tags"])
    target = _signatures(scene["target_matrix"], scene["target_row_tags"])
    routed = _route(source[int(scene["start_index"])], scene["control_mapping"], scene["filled_mapping"], scene["hollow_mapping"])
    return float(min(sum(a != b for a, b in zip(routed, other, strict=True)) for other in target if other != routed))


def is_quarantined(scene: dict) -> bool:
    try:
        source, target = scene["source_matrix"], scene["target_matrix"]
        source_tags, target_tags = scene["source_row_tags"], scene["target_row_tags"]
        route_order = scene["route_order"]
        controls = [int(value) for value in scene["control_mapping"]]
        filled = [int(value) for value in scene["filled_mapping"]]
        hollow = [int(value) for value in scene["hollow_mapping"]]
        if not _valid_matrix(source) or not _valid_matrix(target):
            return True
        if any(sorted(values) != list(range(N)) for values in (source_tags, target_tags, route_order, controls, filled, hollow)):
            return True
        if any(index == controls[index] for index in range(N)):
            return True
        if any(filled[index] == hollow[index] for index in range(N)) or int(scene["start_index"]) not in range(N):
            return True
        source_sigs = _signatures(source, source_tags)
        target_sigs = _signatures(target, target_tags)
        if len(set(source_sigs)) != N or len(set(target_sigs)) != N:
            return True
        if any(sum(signature) != 3 for signature in source_sigs + target_sigs):
            return True
        if any(sum(row) != 3 for row in source + target):
            return True
        routed = _route(source_sigs[int(scene["start_index"])], controls, filled, hollow)
        return target_sigs.count(routed) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int, fill: tuple[int, int, int]) -> None:
    font = _font(size)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - 2), text, font=font, fill=fill)


def _draw_badge(draw: ImageDraw.ImageDraw, cx: int, cy: int, tag: int, side: str) -> None:
    color = (SOURCE_COLORS if side == "source" else TARGET_COLORS)[tag]
    if tag == 0:
        draw.polygon(((cx, cy - 18), (cx - 20, cy + 17), (cx + 20, cy + 17)), fill=color, outline=INK)
    elif tag == 1:
        draw.rectangle((cx - 17, cy - 17, cx + 17, cy + 17), fill=color, outline=INK, width=2)
    elif tag == 2:
        draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=color, outline=INK, width=2)
    elif tag == 3:
        draw.polygon(((cx, cy - 19), (cx - 19, cy), (cx, cy + 19), (cx + 19, cy)), fill=color, outline=INK)
    elif tag == 4:
        draw.polygon(((cx - 17, cy - 9), (cx, cy - 19), (cx + 17, cy - 9), (cx + 17, cy + 9), (cx, cy + 19), (cx - 17, cy + 9)), fill=color, outline=INK)
    else:
        draw.rectangle((cx - 6, cy - 19, cx + 6, cy + 19), fill=color, outline=INK, width=2)
        draw.rectangle((cx - 19, cy - 6, cx + 19, cy + 6), fill=color, outline=INK, width=2)


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, matrix: list[list[int]], tags: list[int], labels: tuple[str, ...], caption: str, start: int | None, side: str) -> None:
    _centered_text(draw, (x0 + N * CELL // 2, 142), caption, 28, INK)
    for col, label in enumerate(labels):
        cx = x0 + col * CELL + CELL // 2
        if col == start:
            draw.rounded_rectangle((cx - 25, 190, cx + 25, 236), radius=11, fill=START)
            _centered_text(draw, (cx, 213), label, 22, WHITE)
            _centered_text(draw, (cx, 176), "START", 14, START)
        else:
            draw.ellipse((cx - 20, 193, cx + 20, 233), fill=WHITE, outline=INK, width=3)
            _centered_text(draw, (cx, 213), label, 21, INK)
    for index in range(N + 1):
        draw.line((x0, GRID_Y + index * CELL, x0 + N * CELL, GRID_Y + index * CELL), fill=GRID, width=2)
        draw.line((x0 + index * CELL, GRID_Y, x0 + index * CELL, GRID_Y + N * CELL), fill=GRID, width=2)
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        _draw_badge(draw, x0 - 45, cy, int(tags[row]), side)
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=INK if matrix[row][col] else WHITE, outline=INK, width=4)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 43), "PAIRED-CONTROL BADGE ROUTING", 34, INK)
    _draw_board(draw, SOURCE_X, scene["source_matrix"], scene["source_row_tags"], tuple(str(i + 1) for i in range(N)), "SOURCE", int(scene["start_index"]), "source")
    _draw_board(draw, TARGET_X, scene["target_matrix"], scene["target_row_tags"], DECISIONS, "TARGET", None, "target")
    _centered_text(draw, (716, 118), "ROUTING TABLE", 27, INK)
    draw.rounded_rectangle((471, 154, 967, 653), radius=16, fill=WHITE, outline=GRID, width=3)
    _centered_text(draw, (ROUTE_SOURCE_X, 196), "CARRY", 15, INK)
    _centered_text(draw, (CONTROL_SOURCE_X, 196), "CHECK", 15, INK)
    _centered_text(draw, (FILLED_TARGET_X, 183), "IF CHECK", 13, FILLED_HEAD)
    _centered_text(draw, (FILLED_TARGET_X, 205), "FILLED", 16, FILLED_HEAD)
    _centered_text(draw, (HOLLOW_TARGET_X, 183), "IF CHECK", 13, HOLLOW_HEAD)
    _centered_text(draw, (HOLLOW_TARGET_X, 205), "HOLLOW", 16, HOLLOW_HEAD)
    draw.line((620, 218, 620, 635), fill=GRID, width=2)
    draw.line((782, 218, 782, 635), fill=GRID, width=2)
    for route_row, source_tag in enumerate(scene["route_order"]):
        cy = ROUTE_Y + route_row * ROUTE_STEP
        _draw_badge(draw, ROUTE_SOURCE_X, cy, int(source_tag), "source")
        _draw_badge(draw, CONTROL_SOURCE_X, cy, int(scene["control_mapping"][source_tag]), "source")
        draw.ellipse((FILLED_ICON_X - 12, cy - 12, FILLED_ICON_X + 12, cy + 12), fill=INK, outline=INK)
        draw.line((660, cy, 688, cy), fill=FILLED_HEAD, width=4)
        draw.polygon(((688, cy), (676, cy - 8), (676, cy + 8)), fill=FILLED_HEAD)
        _draw_badge(draw, FILLED_TARGET_X, cy, int(scene["filled_mapping"][source_tag]), "target")
        draw.ellipse((HOLLOW_ICON_X - 12, cy - 12, HOLLOW_ICON_X + 12, cy + 12), fill=WHITE, outline=INK, width=4)
        draw.line((835, cy, 863, cy), fill=HOLLOW_HEAD, width=4)
        draw.polygon(((863, cy), (851, cy - 8), (851, cy + 8)), fill=HOLLOW_HEAD)
        _draw_badge(draw, HOLLOW_TARGET_X, cy, int(scene["hollow_mapping"][source_tag]), "target")
    _centered_text(draw, (WIDTH // 2, 704), "CHECK chooses the route; CARRY keeps its own START state.", 20, INK)
    _centered_text(draw, (WIDTH // 2, 747), "Route all six CARRY states, then match the TARGET column.", 20, INK)
    return image
