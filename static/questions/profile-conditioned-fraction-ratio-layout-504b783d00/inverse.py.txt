"""Independent pixel-only inverse for fraction_ratio_layout.

This module intentionally repeats the public geometry constants and performs
only raster operations.  It never imports renderer, prompts, generate, verify,
or any analytic-gold code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 960
HEIGHT = 560
PANEL_X_RANGES = ((50, 450), (510, 910))


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    """Return (centroid_x, centroid_y, area) for saturated components."""

    ys, xs = np.nonzero(mask)
    visited = np.zeros(mask.shape, dtype=bool)
    components: list[tuple[float, float, int]] = []
    height, width = mask.shape
    for start_y, start_x in zip(ys.tolist(), xs.tolist()):
        if visited[start_y, start_x]:
            continue
        queue: deque[tuple[int, int]] = deque([(start_y, start_x)])
        visited[start_y, start_x] = True
        area = 0
        total_x = 0
        total_y = 0
        while queue:
            y, x = queue.popleft()
            area += 1
            total_x += x
            total_y += y
            for next_y, next_x in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= next_y < height
                    and 0 <= next_x < width
                    and mask[next_y, next_x]
                    and not visited[next_y, next_x]
                ):
                    visited[next_y, next_x] = True
                    queue.append((next_y, next_x))
        if area >= 25:
            components.append((total_x / area, total_y / area, area))
    return components


def _row_counts(arr: np.ndarray, x0: int, x1: int) -> tuple[int, int]:
    crop = arr[80:480, x0:x1]
    channel_range = crop.max(axis=2).astype(np.int16) - crop.min(axis=2).astype(np.int16)
    # The tokens are saturated; gray outlines, black rules, labels, and white
    # background are deliberately excluded without using any scene colors.
    mask = (channel_range >= 52) & (crop.max(axis=2) <= 250)
    components = _components(mask)
    if len(components) < 4:
        raise ValueError("too few colored term components")
    # Components are measured in the crop whose y=0 is the global y=80.
    # The upper row is around crop y=120 and the lower row around crop y=270.
    top = sum(1 for _, y, _ in components if y < 205)
    bottom = sum(1 for _, y, _ in components if y > 230)
    if top < 1 or bottom < 1 or top + bottom != len(components):
        raise ValueError("colored terms do not form two clean rows")
    return top, bottom


def decision_from_image(image: Image.Image) -> str:
    """Recover the larger fraction from final RGB pixels alone."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected image size {image.size}")
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    top_a, bottom_a = _row_counts(arr, *PANEL_X_RANGES[0])
    top_b, bottom_b = _row_counts(arr, *PANEL_X_RANGES[1])
    left = top_a * bottom_b
    right = top_b * bottom_a
    if left == right:
        return "tie"
    return "A" if left > right else "B"
