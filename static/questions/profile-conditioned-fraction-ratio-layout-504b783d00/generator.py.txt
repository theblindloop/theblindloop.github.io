"""Deterministic raster renderer and analytic contract for fraction_ratio_layout."""

from __future__ import annotations

import math
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "fraction_ratio_layout-0.1.0"
WIDTH = 960
HEIGHT = 560
SCALE = 3
PANEL_BOXES = {"A": (40, 70, 460, 490), "B": (500, 70, 920, 490)}
BACKGROUND = (255, 255, 255)
PALETTE = (
    (30, 126, 214),
    (213, 67, 74),
    (27, 153, 112),
    (150, 76, 190),
    (220, 126, 25),
    (22, 151, 164),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    """Use a fixed installed font, with a deterministic Pillow fallback."""

    for candidate in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/dejavu/DejaVuSans.ttf",
    ):
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size * SCALE)
    return ImageFont.load_default()


def _xy(value: float) -> int:
    return int(round(value * SCALE))


def _box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(_xy(value) for value in values)  # type: ignore[return-value]


def _draw_term(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    size: float,
    color: tuple[int, int, int],
    style: int,
) -> None:
    half = size / 2.0
    if style == 0:
        draw.rounded_rectangle(
            _box((cx - half, cy - half, cx + half, cy + half)),
            radius=max(1, _xy(size * 0.18)),
            fill=color,
        )
    elif style == 1:
        draw.ellipse(_box((cx - half, cy - half, cx + half, cy + half)), fill=color)
    else:
        draw.polygon(
            [
                (_xy(cx), _xy(cy - half)),
                (_xy(cx + half), _xy(cy)),
                (_xy(cx), _xy(cy + half)),
                (_xy(cx - half), _xy(cy)),
            ],
            fill=color,
        )


def _draw_fraction(
    draw: ImageDraw.ImageDraw,
    panel: tuple[int, int, int, int],
    label: str,
    numerator: int,
    denominator: int,
    color: tuple[int, int, int],
    token_scale: float,
    style: int,
    bar_thickness: int,
    x_shift: int,
    y_shift: int,
) -> None:
    x0, y0, x1, y1 = panel
    draw.rounded_rectangle(
        _box((x0, y0, x1, y1)),
        radius=_xy(18),
        fill=(255, 255, 255),
        outline=(213, 219, 228),
        width=_xy(2),
    )
    draw.text(
        (_xy(x0 + 25), _xy(y0 + 18)),
        label,
        font=_font(31),
        fill=(39, 48, 62),
    )

    center_x = (x0 + x1) / 2.0 + x_shift
    center_y = 276.0 + y_shift
    token_size = 11.0 * token_scale
    step = 17.0 * token_scale
    max_count = max(numerator, denominator)
    rule_half = (max_count - 1) * step / 2.0 + token_size / 2.0 + 13.0

    draw.line(
        (_xy(center_x - rule_half), _xy(center_y), _xy(center_x + rule_half), _xy(center_y)),
        fill=(49, 58, 72),
        width=_xy(bar_thickness),
    )

    for count, row_y in ((numerator, center_y - 73.0), (denominator, center_y + 73.0)):
        first_x = center_x - (count - 1) * step / 2.0
        for index in range(count):
            _draw_term(
                draw,
                first_x + index * step,
                row_y,
                token_size,
                color,
                style,
            )

    draw.text(
        (_xy(x0 + 25), _xy(y1 - 37)),
        "terms above / terms below",
        font=_font(12),
        fill=(125, 136, 151),
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair of visible fraction layouts."""

    rng = random.Random(int(seed))
    total = rng.choice((18, 19, 20, 21, 22, 23, 24))
    lower = 5
    upper = total - 5
    near = int(seed) % 7 == 0
    if near:
        first = total // 2 - 1
        second = total // 2
    else:
        first = rng.randint(lower, upper)
        options = [value for value in range(lower, upper + 1) if abs(value - first) >= 3]
        second = rng.choice(options)
    pairs = [(first, total - first), (second, total - second)]
    if rng.random() < 0.5:
        pairs.reverse()
    color_a = rng.choice(PALETTE)
    color_b = rng.choice(tuple(color for color in PALETTE if color != color_a))
    return {
        "numerator_a": int(pairs[0][0]),
        "denominator_a": int(pairs[0][1]),
        "numerator_b": int(pairs[1][0]),
        "denominator_b": int(pairs[1][1]),
        "color_a": list(color_a),
        "color_b": list(color_b),
        "token_scale": round(rng.uniform(0.86, 1.04), 4),
        "token_style": rng.randrange(3),
        "bar_thickness": rng.choice((4, 5, 6)),
        "x_shift_a": rng.choice((-12, -8, -4, 0, 5, 9, 12)),
        "x_shift_b": rng.choice((-12, -8, -4, 0, 5, 9, 12)),
        "y_shift_a": rng.choice((-14, -8, 0, 8, 14)),
        "y_shift_b": rng.choice((-14, -8, 0, 8, 14)),
    }


def render(scene: dict) -> Image.Image:
    """Render the public pixels from the scene dictionary."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_fraction(
        draw,
        PANEL_BOXES["A"],
        "A",
        int(scene["numerator_a"]),
        int(scene["denominator_a"]),
        tuple(int(value) for value in scene["color_a"]),
        float(scene["token_scale"]),
        int(scene["token_style"]),
        int(scene["bar_thickness"]),
        int(scene["x_shift_a"]),
        int(scene["y_shift_a"]),
    )
    _draw_fraction(
        draw,
        PANEL_BOXES["B"],
        "B",
        int(scene["numerator_b"]),
        int(scene["denominator_b"]),
        tuple(int(value) for value in scene["color_b"]),
        float(scene["token_scale"]),
        int(scene["token_style"]),
        int(scene["bar_thickness"]),
        int(scene["x_shift_b"]),
        int(scene["y_shift_b"]),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Compare the two visible fractions by exact cross multiplication."""

    numerator_a = int(scene["numerator_a"])
    denominator_a = int(scene["denominator_a"])
    numerator_b = int(scene["numerator_b"])
    denominator_b = int(scene["denominator_b"])
    left = numerator_a * denominator_b
    right = numerator_b * denominator_a
    if left == right:
        return "tie"
    return "A" if left > right else "B"


def margin(scene: dict) -> float:
    """Absolute difference between the two displayed ratios."""

    ratio_a = int(scene["numerator_a"]) / int(scene["denominator_a"])
    ratio_b = int(scene["numerator_b"]) / int(scene["denominator_b"])
    return abs(ratio_a - ratio_b)


def is_quarantined(scene: dict) -> bool:
    """Reject ties and near ties that are not safely separable at this scale."""

    return analytic_gold(scene) == "tie" or margin(scene) < 0.30


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden label aliases: every label input is painted as visible terms."""

    return []
