"""Scene sampling, rasterisation and analytic gold for undershoot_gap_rank.

Five coloured bars lie in five horizontal lanes. Each lane carries its OWN black
target tick. A bar either reaches past its own tick (an overshoot) or stops short
of it (an undershoot). The decision is the colour of the bar with the LARGEST
undershoot, i.e. the largest positive signed residual target-minus-end. Cheap
readings -- the shortest bar, the longest bar, the bar whose end is furthest
left, the lane whose tick stands furthest right, and the largest unsigned
bar-to-tick distance -- are all held away from the answer by construction, and
in every scene the largest unsigned distance belongs to an overshooting lane.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

W = 720
H = 460
XL = 46.0
XR = 690.0
LANE_TOP = 34
LANE_STEP = 82
LANE_H = 50
BAR_H = 26
TICK_W = 5

BG = (255, 255, 255)
GRID = (233, 233, 233)
LANE_RULE = (176, 176, 176)
TICK = (0, 0, 0)

PALETTE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
COLORS = tuple(sorted(PALETTE))

MIN_CLEAR = 14.0
MIN_MARGIN = 12.0
QUARANTINE_MARGIN = 20.0


def _residuals(scene: dict) -> list[tuple[float, str]]:
    """(target minus bar end, colour) pairs; positive means the bar stops short."""

    return [
        (float(lane["target"]) - float(lane["end"]), str(lane["color"]))
        for lane in scene["lanes"]
    ]


def _ranked_shortfalls(scene: dict) -> list[tuple[float, str]]:
    """Undershooting lanes ordered from the largest shortfall downwards."""

    return sorted(
        ((value, color) for value, color in _residuals(scene) if value > 0.0),
        key=lambda item: (-item[0], item[1]),
    )


def _cues_clear(scene: dict) -> bool:
    """Reject any scene in which a surface reading names the answer."""

    lanes = scene["lanes"]
    ranked = _ranked_shortfalls(scene)
    if len(ranked) < 2:
        return False
    if ranked[0][0] - ranked[1][0] < MIN_MARGIN:
        return False
    residuals = _residuals(scene)
    if min(abs(value) for value, _ in residuals) < MIN_CLEAR:
        return False
    if len(ranked) == len(lanes):
        return False
    winner = ranked[0][1]
    biggest_unsigned = max(residuals, key=lambda item: (abs(item[0]), item[1]))
    if biggest_unsigned[1] == winner or biggest_unsigned[0] > 0.0:
        return False
    shortest = min(lanes, key=lambda lane: (lane["end"] - lane["start"], lane["color"]))
    longest = max(lanes, key=lambda lane: (lane["end"] - lane["start"], lane["color"]))
    leftmost_end = min(lanes, key=lambda lane: (lane["end"], lane["color"]))
    rightmost_tick = max(lanes, key=lambda lane: (lane["target"], lane["color"]))
    leftmost_start = min(lanes, key=lambda lane: (lane["start"], lane["color"]))
    blocked = {
        shortest["color"],
        longest["color"],
        leftmost_end["color"],
        rightmost_tick["color"],
        leftmost_start["color"],
    }
    return winner not in blocked


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene for one integer seed."""

    rng = random.Random(seed)
    forced = COLORS[seed % len(COLORS)]
    for _ in range(24000):
        win_short = rng.uniform(46.0, 140.0)
        gap = MIN_MARGIN + 52.0 * rng.random() ** 0.7
        second_short = win_short - gap
        if second_short < MIN_CLEAR + 6.0:
            continue
        third_short = second_short - rng.uniform(14.0, 40.0)
        if third_short < MIN_CLEAR + 4.0:
            continue
        big_over = win_short + rng.uniform(22.0, 90.0)
        small_over = rng.uniform(MIN_CLEAR + 6.0, MIN_CLEAR + 60.0)
        plan = [
            (win_short, "winner"),
            (second_short, "runner"),
            (third_short, "third"),
            (-big_over, "over_big"),
            (-small_over, "over_small"),
        ]

        lanes = []
        ok = True
        for residual, role in plan:
            for _ in range(80):
                end = rng.uniform(210.0, 640.0)
                target = end + residual
                if not XL + 16.0 <= target <= XR - 8.0:
                    continue
                start = rng.uniform(XL + 6.0, end - 70.0)
                if end - start < 70.0:
                    continue
                lanes.append(
                    {
                        "color": "",
                        "slot": 0,
                        "start": round(start, 3),
                        "end": round(end, 3),
                        "target": round(target, 3),
                        "role": role,
                    }
                )
                break
            else:
                ok = False
                break
        if not ok or len(lanes) != 5:
            continue

        colors = [color for color in COLORS if color != forced]
        rng.shuffle(colors)
        colors = [forced] + colors
        slots = list(range(5))
        rng.shuffle(slots)
        for index, lane in enumerate(lanes):
            lane["color"] = colors[index]
            lane["slot"] = slots[index]
            del lane["role"]
        rng.shuffle(lanes)
        decor = sorted(round(rng.uniform(XL + 40.0, XR - 40.0), 2) for _ in range(4))
        scene = {"lanes": lanes, "decor": decor}
        if _cues_clear(scene) and analytic_gold(scene) == forced:
            return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _slot_of(lane: dict) -> int:
    return int(lane["slot"])


def render(scene: dict) -> Image.Image:
    """Rasterise a latent scene; the lane list is drawn as an unordered set."""

    image = Image.new("RGB", (W, H), BG)
    draw = ImageDraw.Draw(image)
    for value in sorted(float(item) for item in scene["decor"]):
        x = int(round(value))
        draw.line([(x, 24), (x, H - 22)], fill=GRID, width=2)
    for lane in sorted(scene["lanes"], key=_slot_of):
        top = LANE_TOP + _slot_of(lane) * LANE_STEP
        draw.line([(int(XL), top + LANE_H + 8), (int(XR), top + LANE_H + 8)], fill=GRID, width=2)
        draw.line([(int(XL) - 6, top), (int(XL) - 6, top + LANE_H)], fill=LANE_RULE, width=3)
        bar_top = top + (LANE_H - BAR_H) // 2
        x0 = int(round(float(lane["start"])))
        x1 = int(round(float(lane["end"])))
        draw.rectangle([x0, bar_top, x1, bar_top + BAR_H], fill=PALETTE[str(lane["color"])])
        tx = int(round(float(lane["target"])))
        draw.rectangle([tx - TICK_W // 2, top, tx + TICK_W // 2, top + LANE_H], fill=TICK)
    return image


def analytic_gold(scene: dict) -> str:
    """Colour of the bar that stops short of its own tick by the largest distance."""

    return _ranked_shortfalls(scene)[0][1]


def margin(scene: dict) -> float:
    """Pixel gap between the largest shortfall and the second largest shortfall."""

    ranked = _ranked_shortfalls(scene)
    return float(round(ranked[0][0] - ranked[1][0], 6))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes excluded from headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabelings that must preserve the gold."""

    lanes = list(scene["lanes"])
    decor = list(scene["decor"])
    return [
        ("reversed_lane_list", {"lanes": list(reversed(lanes)), "decor": list(decor)}),
        ("rotated_lane_list", {"lanes": lanes[2:] + lanes[:2], "decor": list(decor)}),
        ("reversed_decor_list", {"lanes": list(lanes), "decor": list(reversed(decor))}),
    ]
