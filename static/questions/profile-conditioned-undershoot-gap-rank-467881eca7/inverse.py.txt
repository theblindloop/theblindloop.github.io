"""Pixel-only decision arm for undershoot_gap_rank.

Receives the final PNG and nothing else: no latent scene, no rendering module,
no analytic gold. It segments the five coloured bars by hue, locates the black
tick that shares each bar's band of rows, measures the signed distance from the
bar's right end to that tick, keeps only the lanes whose bar stops short of the
tick, and reports the colour with the largest such shortfall.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ABSTAIN = "abstain"

_REFERENCE = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "purple": (148, 60, 180),
}
_TOLERANCE = 40
_SATURATION = 45
_DARK = 70
_EXPECTED = 5
_MIN_BAR_PIXELS = 900
_MIN_BAR_HEIGHT = 18
_MAX_BAR_HEIGHT = 40
_MIN_TICK_PIXELS = 60
_MIN_CLEAR = 6.0
_MIN_LEAD = 4.0


def _columns_of(mask: np.ndarray) -> list[tuple[int, int]]:
    """Maximal runs of populated columns in a boolean mask."""

    occupied = np.flatnonzero(mask.any(axis=0))
    runs: list[tuple[int, int]] = []
    for column in occupied:
        if runs and column == runs[-1][1] + 1:
            runs[-1] = (runs[-1][0], int(column))
        else:
            runs.append((int(column), int(column)))
    return runs


def decision_from_image(image: Image.Image) -> str:
    """Colour with the largest bar-to-tick shortfall, read from the raster alone."""

    data = np.asarray(image.convert("RGB")).astype(np.int16)
    saturation = data.max(axis=2) - data.min(axis=2)
    coloured = saturation > _SATURATION
    dark = (data.max(axis=2) <= _DARK) & (saturation <= _SATURATION)

    shortfalls: dict[str, float] = {}
    seen = 0
    for name, reference in _REFERENCE.items():
        distance = np.abs(data - np.array(reference, dtype=np.int16)).max(axis=2)
        mask = coloured & (distance <= _TOLERANCE)
        if int(mask.sum()) < _MIN_BAR_PIXELS:
            continue
        rows = np.flatnonzero(mask.any(axis=1))
        cols = np.flatnonzero(mask.any(axis=0))
        height = int(rows[-1] - rows[0] + 1)
        if not _MIN_BAR_HEIGHT <= height <= _MAX_BAR_HEIGHT:
            return ABSTAIN
        runs_bar = _columns_of(mask)
        # A tick that sits inside its own bar splits that bar into two runs.
        if len(runs_bar) > 2:
            return ABSTAIN
        if len(runs_bar) == 2 and runs_bar[1][0] - runs_bar[0][1] > 10:
            return ABSTAIN
        seen += 1
        band = dark[max(int(rows[0]) - 2, 0) : int(rows[-1]) + 3, :]
        if int(band.sum()) < _MIN_TICK_PIXELS:
            return ABSTAIN
        runs = _columns_of(band)
        if len(runs) != 1:
            return ABSTAIN
        tick = (runs[0][0] + runs[0][1]) / 2.0
        end = float(cols[-1])
        residual = tick - end
        if abs(residual) < _MIN_LEAD:
            return ABSTAIN
        if residual > 0.0:
            shortfalls[name] = residual
    if seen != _EXPECTED or len(shortfalls) < 2:
        return ABSTAIN
    ranked = sorted(shortfalls.items(), key=lambda item: (-item[1], item[0]))
    if ranked[0][1] - ranked[1][1] < _MIN_CLEAR:
        return ABSTAIN
    return ranked[0][0]
