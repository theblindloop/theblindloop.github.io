"""Bend-comparison tracing world: which route has fewer sharp U-turn bends.

Two independent serpentine routes are drawn, one blue and one green. Each route
starts at a small orange ring and ends at a dark terminal dot. A route is a
horizontal serpentine: it runs straight right, folds back (a sharp U-turn bend),
runs left, folds again, and so on. The number of bends of a route equals the
number of its direction-reversal folds. The question asks which route, blue or
green, has fewer bends along its length. Because a route doubles back on itself,
the bend count is a pure tracing/topological fact of the drawn stroke and cannot
be recovered from bounding-box extent or endpoint reach.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "path_bend_comparison-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_GREEN = (40, 170, 70)
COLOR_ORANGE = (245, 150, 20)
COLOR_DARK = (40, 40, 40)

STROKE = 16          # route stroke thickness (px)
PORT_R = 5           # orange ring radius
TERM_R = 5           # terminal dot radius
X_LEFT = 70.0
X_RIGHT = 410.0
LANE_STEP = 30.0     # vertical step between serpentine lanes
BEND_POOL = [2, 3, 4]

ROUTE_COLORS = ["blue", "green"]
FILL = {"blue": COLOR_BLUE, "green": COLOR_GREEN}


def _serpentine(x0: float, x1: float, y0: float, n_bends: int, dy: float) -> list[tuple[float, float]]:
    """A horizontal serpentine polyline with exactly n_bends sharp folds.

    The path starts at (x0, y0), runs to x1, folds back to x0, folds to x1, and
    so on, stepping vertically by dy after each fold. Each fold is one bend.
    """
    pts: list[tuple[float, float]] = [(x0, y0)]
    y = y0
    x = x0
    for i in range(n_bends):
        x = x1 if i % 2 == 0 else x0
        pts.append((x, y))
        y = y + dy
        pts.append((x, y))
    x = x1 if n_bends % 2 == 0 else x0
    pts.append((x, y))
    return pts


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _sample_band(rng: random.Random, band: str) -> dict:
    w, h = CANVAS
    y0 = 70.0 if band == "blue" else 250.0
    n_bends = rng.choice(BEND_POOL)
    poly = _serpentine(X_LEFT, X_RIGHT, y0, n_bends, LANE_STEP)
    return {
        "color": band,
        "folds": n_bends,
        "polyline": [[float(x), float(y)] for x, y in poly],
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    blue = _sample_band(rng, "blue")
    green = _sample_band(rng, "green")
    # ensure the two route bend counts differ (regenerating the whole route so
    # the stored fold count and the drawn polyline always agree)
    while green["folds"] == blue["folds"]:
        green = _sample_band(rng, "green")
    answer = "blue" if blue["folds"] < green["folds"] else "green"
    scene = {
        "routes": {"blue": blue, "green": green},
        "answer": answer,
    }
    return scene


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    for name in ("blue", "green"):
        route = scene["routes"][name]
        pts = _pts(route["polyline"])
        _draw_polyline(draw, pts, FILL[route["color"]], STROKE)
    for name in ("blue", "green"):
        route = scene["routes"][name]
        pts = _pts(route["polyline"])
        start = pts[0]
        end = pts[-1]
        draw.ellipse([start[0] - PORT_R, start[1] - PORT_R, start[0] + PORT_R, start[1] + PORT_R], fill=COLOR_ORANGE)
        draw.ellipse([end[0] - TERM_R, end[1] - TERM_R, end[0] + TERM_R, end[1] + TERM_R], fill=COLOR_DARK)
    return img


def _count_folds(poly: list) -> int:
    """Number of sharp direction-reversal folds in the latent polyline."""
    pts = _pts(poly)
    if len(pts) < 2:
        return 0
    signs = []
    prev = None
    for a, b in zip(pts, pts[1:]):
        dx = b[0] - a[0]
        if abs(dx) < 1e-9:
            continue
        s = 1 if dx > 0 else -1
        signs.append(s)
    folds = 0
    for i in range(1, len(signs)):
        if signs[i] != signs[i - 1]:
            folds += 1
    return folds


def analytic_gold(scene: dict) -> str:
    """Colour of the route with fewer bends."""
    blue = _count_folds(scene["routes"]["blue"]["polyline"])
    green = _count_folds(scene["routes"]["green"]["polyline"])
    return "blue" if blue < green else "green"


def margin(scene: dict) -> float:
    """Absolute difference in bend counts between the two routes (px-equivalent)."""
    blue = _count_folds(scene["routes"]["blue"]["polyline"])
    green = _count_folds(scene["routes"]["green"]["polyline"])
    return float(abs(blue - green))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    blue = int(scene["routes"]["blue"]["folds"])
    green = int(scene["routes"]["green"]["folds"])
    if blue == green:
        return True
    for name in ("blue", "green"):
        route = scene["routes"][name]
        pts = _pts(route["polyline"])
        for x, y in pts:
            if not (10 <= x <= w - 10 and 10 <= y <= h - 10):
                return True
    # blue and green bands must stay clearly separated so colour segmentation
    # never merges two lanes across routes.
    blue_pts = _pts(scene["routes"]["blue"]["polyline"])
    green_pts = _pts(scene["routes"]["green"]["polyline"])
    b_max_y = max(y for _, y in blue_pts)
    g_min_y = min(y for _, y in green_pts)
    if g_min_y - b_max_y < STROKE * 2:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
