"""Pixel-only inverse arm for the path-bend-comparison tracing world.

Reads only the final RGB PNG. Segments the blue and green route strokes by
nearest-colour labelling. For each route colour it reconstructs the ordered
centreline of the single connected stroke by a geodesic breadth-first distance
field and a monotone descent, then counts the sharp direction-reversal folds
(the horizontal x-extrema) along that order. The answer is the colour of the
route with the smaller fold count. This requires tracing each serpentine stroke:
the number of folds is a topological fact of the drawn line and is not the
bounding-box extent or the end-to-end distance.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "white": (255, 255, 255),
    "blue": (40, 90, 210),
    "green": (40, 170, 70),
    "orange": (245, 150, 20),
    "dark": (40, 40, 40),
}
_NAMES = ["white", "blue", "green", "orange", "dark"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _bfs_dist(mask: np.ndarray, seed) -> np.ndarray:
    h, w = mask.shape
    dist = np.full((h, w), -1, dtype=np.int64)
    dq = deque()
    sy, sx = seed
    dist[sy, sx] = 0
    dq.append((sy, sx))
    while dq:
        y, x = dq.popleft()
        nd = dist[y, x] + 1
        for ny in range(y - 1, y + 2):
            for nx in range(x - 1, x + 2):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and dist[ny, nx] == -1:
                    dist[ny, nx] = nd
                    dq.append((ny, nx))
    return dist


def _trace_order(mask: np.ndarray) -> list[tuple[int, int]]:
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        raise ValueError("route colour not found")
    start = (int(ys[0]), int(xs[0]))
    dist = _bfs_dist(mask, start)
    fy, fx = np.unravel_index(np.argmax(dist), dist.shape)
    far = (int(fy), int(fx))
    order = [far]
    visited = {far}
    cur = far
    h, w = mask.shape
    while True:
        cy, cx = cur
        best = None
        best_d = dist[cy, cx]
        for ny in range(cy - 1, cy + 2):
            for nx in range(cx - 1, cx + 2):
                if (ny, nx) == (cy, cx):
                    continue
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and (ny, nx) not in visited:
                    d = dist[ny, nx]
                    if d != -1 and d < best_d:
                        best_d = d
                        best = (ny, nx)
        if best is None:
            break
        visited.add(best)
        order.append(best)
        cur = best
    return order


def _count_bends(mask: np.ndarray, decimate_px: int = 8) -> int:
    if int(mask.sum()) < 4:
        raise ValueError("route has too little ink")
    order = _trace_order(mask)
    xs = [int(x) for _, x in order]
    ys = [int(y) for y, _ in order]
    if len(xs) < 2:
        return 0
    pts = [(ys[0], xs[0])]
    acc = 0
    for i in range(1, len(order)):
        dy = ys[i] - ys[i - 1]
        dx = xs[i] - xs[i - 1]
        acc += abs(dx) + abs(dy)
        if acc >= decimate_px:
            pts.append((ys[i], xs[i]))
            acc = 0
    if pts[-1] != (ys[-1], xs[-1]):
        pts.append((ys[-1], xs[-1]))
    if len(pts) < 3:
        return 0
    signs = []
    for a, b in zip(pts, pts[1:]):
        dx = b[1] - a[1]
        if abs(dx) < 2:
            continue
        signs.append(1 if dx > 0 else -1)
    folds = 0
    for i in range(1, len(signs)):
        if signs[i] != signs[i - 1]:
            folds += 1
    return folds


def _route_mask(labels, name, idx) -> np.ndarray:
    return labels == idx[name]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue_mask = _route_mask(labels, "blue", idx)
    green_mask = _route_mask(labels, "green", idx)
    blue_bends = _count_bends(blue_mask)
    green_bends = _count_bends(green_mask)
    if blue_bends == green_bends:
        raise ValueError("ambiguous equal bend counts")
    return "blue" if blue_bends < green_bends else "green"
