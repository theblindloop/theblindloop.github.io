"""Pixel-only decision arm for window_overlap_total.

Receives one rasterized PNG and nothing else: no scene, no renderer import, no
analytic gold, no metadata. It recovers the two shaded window column ranges
from picture rows that carry no bars, segments the four coloured bars by exact
colour, checks each one is an intact solid rectangle, counts how many of its
columns fall inside the windows, and names the colour with the greatest total.
It abstains whenever the expected structure is absent or the top-two total gap
is too small to be read reliably.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "window_overlap_total-oracle-1.0.0"

ABSTAIN = "abstain"

_MARK_RGB = {
    "red": (222, 32, 32),
    "green": (24, 148, 64),
    "blue": (36, 72, 214),
    "orange": (240, 152, 24),
}
_WINDOW_RGB = (226, 226, 226)

_MIN_WINDOW_PX = 40
_MIN_BAR_COLUMNS = 40
_MIN_BAR_PIXELS = 300
_MAX_BAR_THICKNESS = 30
_MIN_TOTAL_PX = 1
_MIN_GAP_PX = 1.0


def _runs(flags: np.ndarray) -> list:
    """Inclusive column runs of a boolean row."""
    out = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def decision_from_image(image: Image.Image) -> str:
    data = np.asarray(image.convert("RGB"), dtype=np.int16)
    if data.ndim != 3 or data.shape[2] != 3:
        return ABSTAIN
    height, width = data.shape[0], data.shape[1]
    if height < 60 or width < 120:
        return ABSTAIN

    window = np.array(_WINDOW_RGB, dtype=np.int16)
    bands = None
    for row in (8, height - 9):
        found = [span for span in _runs((data[row] == window).all(axis=1)) if span[1] - span[0] + 1 >= _MIN_WINDOW_PX]
        if len(found) != 2:
            return ABSTAIN
        if bands is None:
            bands = found
        elif found != bands:
            # The two readings disagree, so the windows are not clean columns.
            return ABSTAIN

    inside = np.zeros(width, dtype=bool)
    for span in bands:
        inside[span[0] : span[1] + 1] = True

    readings = []
    for name, target in sorted(_MARK_RGB.items()):
        mask = (data == np.array(target, dtype=np.int16)).all(axis=2)
        count = int(mask.sum())
        if count < _MIN_BAR_PIXELS:
            return ABSTAIN
        rows = np.flatnonzero(mask.any(axis=1))
        cols = np.flatnonzero(mask.any(axis=0))
        y0, y1 = int(rows[0]), int(rows[-1])
        x0, x1 = int(cols[0]), int(cols[-1])
        box_w, box_h = x1 - x0 + 1, y1 - y0 + 1
        if box_w < _MIN_BAR_COLUMNS or not 2 <= box_h <= _MAX_BAR_THICKNESS:
            return ABSTAIN
        if count != box_w * box_h:
            # A solid bar fills its bounding box exactly; a split, clipped or
            # partly erased one does not, and its extent is not trustworthy.
            return ABSTAIN
        columns = mask.any(axis=0)
        total = int(np.count_nonzero(columns & inside))
        if total < _MIN_TOTAL_PX:
            return ABSTAIN
        readings.append((total, name))

    readings.sort(key=lambda item: (-item[0], item[1]))
    if readings[0][0] - readings[1][0] < _MIN_GAP_PX:
        return ABSTAIN
    return readings[0][1]
