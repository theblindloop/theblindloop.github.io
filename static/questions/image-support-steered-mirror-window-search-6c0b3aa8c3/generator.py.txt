"""Deterministic Pillow world for a reflection-consistency visual search."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

ANSWERS = ("top", "upper-middle", "lower-middle", "bottom")
SIZE = 640
BLUE = (36, 99, 168)
DARK_BLUE = (18, 58, 106)
WHITE = (250, 250, 246)


def _pair_mismatches(pattern: list[int]) -> int:
    return sum(pattern[i] != pattern[-1 - i] for i in range(6))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    patterns: list[list[int]] = []
    pair_choices = list(range(6))
    rng.shuffle(pair_choices)
    selected = set(pair_choices[:3])
    symmetric = [int(i in selected or (11 - i) in selected) for i in range(12)]
    for row in range(4):
        if row == winner:
            patterns.append(symmetric)
            continue
        while True:
            bits = [1] * 6 + [0] * 6
            rng.shuffle(bits)
            if _pair_mismatches(bits) >= 2:
                patterns.append(bits)
                break
    return {"patterns": patterns}


def render(scene: dict) -> Image.Image:
    patterns = scene["patterns"]
    image = Image.new("RGB", (SIZE, SIZE), WHITE)
    draw = ImageDraw.Draw(image)
    draw.rectangle((8, 8, 631, 631), fill=BLUE, outline=DARK_BLUE, width=5)
    left, top, right, bottom = 48, 48, 592, 592
    row_h = (bottom - top) / 4
    col_w = (right - left) / 12
    for row in range(1, 4):
        y = round(top + row * row_h)
        draw.line((20, y, 619, y), fill=DARK_BLUE, width=5)
    for col in range(1, 12):
        x = round(left + col * col_w)
        draw.line((x, 25, x, 615), fill=(31, 86, 148), width=2)
    half = 14
    for row, pattern in enumerate(patterns):
        cy = round(top + (row + 0.5) * row_h)
        for col, bit in enumerate(pattern):
            if bit:
                cx = round(left + (col + 0.5) * col_w)
                draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=WHITE)
    return image


def analytic_gold(scene: dict) -> str:
    patterns = scene["patterns"]
    scores = [_pair_mismatches(list(pattern)) for pattern in patterns]
    winners = [i for i, value in enumerate(scores) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have exactly one symmetric strip")
    return ANSWERS[winners[0]]


def margin(scene: dict) -> float:
    patterns = scene["patterns"]
    scores = sorted(_pair_mismatches(list(pattern)) for pattern in patterns)
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        patterns = scene["patterns"]
        scores = sorted(_pair_mismatches(list(pattern)) for pattern in patterns)
        return not (len(patterns) == 4 and scores[0] == 0 and scores[1] >= 2)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
