"""Independent pixel-only inverse for the public 4-by-12 tile layout."""

from __future__ import annotations

import numpy as np
from PIL import Image

ANSWERS = ("top", "upper-middle", "lower-middle", "bottom")


def _components(mask: np.ndarray) -> int:
    # Component enumeration on a conservative 4x reduction; this independently
    # certifies that the broad blue evidence field is one connected area.
    h = mask.shape[0] // 4
    w = mask.shape[1] // 4
    coarse = mask[: h * 4, : w * 4].reshape(h, 4, w, 4).any(axis=(1, 3))
    seen = np.zeros_like(coarse, dtype=bool)
    count = 0
    for sy, sx in np.argwhere(coarse):
        if seen[sy, sx]:
            continue
        count += 1
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and coarse[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
    return count


def _pair_mismatches(bits: list[int]) -> int:
    return sum(bits[i] != bits[-1 - i] for i in range(6))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    blue = (b > 70) & (b > r + 35) & (b > g + 20)
    ys, xs = np.nonzero(blue)
    if len(xs) < 250000:
        raise ValueError("connected blue tile is missing too much area")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if x1 - x0 < 590 or y1 - y0 < 590:
        raise ValueError("blue tile does not span the image")
    if _components(blue) != 1:
        raise ValueError("blue evidence field is not one connected component")

    # Integrity is measured everywhere on the inferred tile, not at stored
    # renderer coordinates. This makes destructive erasure of any distant part
    # observable while tolerating the small declared window apertures.
    for gy in range(10):
        ya = y0 + (y1 - y0 + 1) * gy // 10
        yb = y0 + (y1 - y0 + 1) * (gy + 1) // 10
        for gx in range(10):
            xa = x0 + (x1 - x0 + 1) * gx // 10
            xb = x0 + (x1 - x0 + 1) * (gx + 1) // 10
            if float(blue[ya:yb, xa:xb].mean()) < 0.53:
                raise ValueError("blue tile has a material erased region")

    tile_w, tile_h = x1 - x0 + 1, y1 - y0 + 1
    left = x0 + tile_w * 40 / 624
    right = x0 + tile_w * 584 / 624
    top = y0 + tile_h * 40 / 624
    bottom = y0 + tile_h * 584 / 624
    rows: list[list[int]] = []
    for row in range(4):
        bits = []
        cy = top + (row + 0.5) * (bottom - top) / 4
        for col in range(12):
            cx = left + (col + 0.5) * (right - left) / 12
            patch = blue[round(cy)-7:round(cy)+8, round(cx)-7:round(cx)+8]
            if patch.size != 225:
                raise ValueError("candidate cell falls outside image")
            bits.append(int(float(patch.mean()) < 0.15))
        if sum(bits) != 6:
            raise ValueError("each candidate strip must contain exactly six windows")
        rows.append(bits)
    scores = [_pair_mismatches(bits) for bits in rows]
    winners = [i for i, score in enumerate(scores) if score == 0]
    runners = sorted(scores)
    if len(winners) != 1 or runners[1] < 2:
        raise ValueError("no unique mirror-symmetric strip with sufficient margin")
    return ANSWERS[winners[0]]
