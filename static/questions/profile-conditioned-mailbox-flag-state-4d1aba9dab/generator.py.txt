"""Deterministic mailbox signal-flag raster world.

The mailbox is a familiar semantic host, but the answer is carried only by the
small saturated signal arm. Consecutive seeds form matched pairs: nuisance
style is held fixed while the arm flips between raised and lowered.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "mailbox_flag_state-renderer-0.1.0"
WIDTH = 512
HEIGHT = 512
SCALE = 3

FLAG_STATES = ("raised", "lowered")
FLAG_PALETTE = {
    "red": (215, 49, 58),
    "blue": (38, 105, 204),
    "green": (32, 153, 87),
    "orange": (226, 117, 31),
    "purple": (145, 65, 183),
    "teal": (18, 151, 157),
}
BODY_PALETTE = (
    (226, 235, 239),
    (242, 226, 197),
    (226, 238, 216),
    (238, 226, 239),
    (232, 232, 224),
)
BACKGROUND_PALETTE = (
    (250, 248, 243),
    (242, 247, 249),
    (248, 243, 247),
)
OUTLINE = (46, 56, 66)
DETAIL = (113, 125, 132)
POST = (164, 173, 177)


def _scaled_box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in values)  # type: ignore[return-value]


def _scaled_point(value: tuple[float, float]) -> tuple[int, int]:
    return int(round(value[0] * SCALE)), int(round(value[1] * SCALE))


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, JSON-friendly scene from an integer seed."""

    numeric_seed = int(seed)
    pair_index = numeric_seed // 2
    style_rng = random.Random(pair_index * 104729 + 271828)
    flag_state = FLAG_STATES[numeric_seed % 2]
    body_width = style_rng.randint(194, 244)
    body_height = style_rng.randint(108, 136)
    body_x = style_rng.randint(108, WIDTH - body_width - 108)
    body_y = style_rng.randint(242, 274)
    near_boundary = pair_index % 13 == 0
    flag_length = style_rng.randint(31, 44) if near_boundary else style_rng.randint(84, 122)
    flag_thickness = style_rng.randint(10, 14)
    pennant_width = style_rng.randint(28, 36)
    pennant_height = style_rng.randint(22, 30)
    return {
        "flag_state": flag_state,
        "body_x": body_x,
        "body_y": body_y,
        "body_width": body_width,
        "body_height": body_height,
        "flag_length": flag_length,
        "flag_thickness": flag_thickness,
        "pennant_width": pennant_width,
        "pennant_height": pennant_height,
        "outline_width": style_rng.choice((4, 5, 6)),
        "body_color": list(style_rng.choice(BODY_PALETTE)),
        "flag_color_name": style_rng.choice(tuple(FLAG_PALETTE)),
        "background": list(style_rng.choice(BACKGROUND_PALETTE)),
        "width": WIDTH,
        "height": HEIGHT,
        "supersample": SCALE,
    }


def render(scene: dict) -> Image.Image:
    """Render the final RGB PNG raster from visible scene fields."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = int(scene["supersample"])
    image = Image.new("RGB", (width * scale, height * scale), _rgb(scene["background"]))
    draw = ImageDraw.Draw(image)

    x = int(scene["body_x"])
    y = int(scene["body_y"])
    body_width = int(scene["body_width"])
    body_height = int(scene["body_height"])
    outline_width = int(scene["outline_width"])
    body = (x, y, x + body_width, y + body_height)

    post_x = x + body_width // 2 - 14
    draw.rectangle(
        _scaled_box((post_x, y + body_height - 4, post_x + 28, HEIGHT - 40)),
        fill=POST,
    )
    draw.rectangle(
        _scaled_box((post_x - 30, HEIGHT - 43, post_x + 58, HEIGHT - 34)),
        fill=OUTLINE,
    )
    draw.rounded_rectangle(
        _scaled_box(tuple(float(item) for item in body)),
        radius=31 * scale,
        fill=_rgb(scene["body_color"]),
        outline=OUTLINE,
        width=max(1, outline_width * scale),
    )
    draw.line(
        [_scaled_point((x + 28, y + body_height - 25)), _scaled_point((x + body_width - 27, y + body_height - 25))],
        fill=DETAIL,
        width=2 * scale,
    )
    draw.ellipse(
        _scaled_box((x + 25, y + 23, x + 48, y + 46)),
        fill=(250, 250, 246),
        outline=DETAIL,
        width=2 * scale,
    )

    flag_color = FLAG_PALETTE[str(scene["flag_color_name"])]
    pivot_x = x + body_width - 35
    pivot_y = y + 43
    flag_length = int(scene["flag_length"])
    flag_thickness = int(scene["flag_thickness"])
    pennant_width = int(scene["pennant_width"])
    pennant_height = int(scene["pennant_height"])
    state = str(scene["flag_state"])

    if state == "raised":
        end_x, end_y = pivot_x, pivot_y - flag_length
        draw.line(
            [_scaled_point((pivot_x, pivot_y)), _scaled_point((end_x, end_y))],
            fill=flag_color,
            width=flag_thickness * scale,
        )
        draw.rectangle(
            _scaled_box(
                (
                    end_x,
                    end_y - pennant_height / 2,
                    end_x + pennant_width,
                    end_y + pennant_height / 2,
                )
            ),
            fill=flag_color,
        )
    elif state == "lowered":
        end_x, end_y = pivot_x - flag_length, pivot_y
        draw.line(
            [_scaled_point((pivot_x, pivot_y)), _scaled_point((end_x, end_y))],
            fill=flag_color,
            width=flag_thickness * scale,
        )
        draw.rectangle(
            _scaled_box(
                (
                    end_x - pennant_width / 2,
                    end_y - pennant_height,
                    end_x + pennant_width / 2,
                    end_y,
                )
            ),
            fill=flag_color,
        )
    else:
        raise ValueError(f"unsupported flag state {state!r}")

    draw.ellipse(
        _scaled_box(
            (
                pivot_x - flag_thickness * 0.9,
                pivot_y - flag_thickness * 0.9,
                pivot_x + flag_thickness * 0.9,
                pivot_y + flag_thickness * 0.9,
            )
        ),
        fill=flag_color,
    )

    image = image.resize((width, height), Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    label_draw.text((x + 22, y + body_height + 17), "MAILBOX", fill=OUTLINE, font=ImageFont.load_default())
    return image


def analytic_gold(scene: dict) -> str:
    """The gold is the visibly rendered signal-arm state."""

    return str(scene["flag_state"])


def margin(scene: dict) -> float:
    """Analytic one-sided clearance above the short-arm quarantine floor."""

    return round(float(int(scene["flag_length"]) - 58), 6)


def is_quarantined(scene: dict) -> bool:
    """Quarantine short arms whose raster orientation is intentionally tight."""

    return int(scene["flag_length"]) < 58 or int(scene["flag_thickness"]) < 8


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are claimed; all label inputs are visibly rendered."""

    return []
