"""Independent image-only inverse for the mailbox signal flag.

Only the final RGB raster is accepted. The inverse uses saturation to isolate
the small colored flag from the pale mailbox and measures its pixel extents;
it does not import or inspect renderer state, prompts, metadata, or gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mailbox_flag_state-oracle-0.1.0"
MIN_FLAG_PIXELS = 120
SATURATION_THRESHOLD = 60


def _flag_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    bright = array.max(axis=2) >= 80
    return (spread >= SATURATION_THRESHOLD) & bright


def _extents(image: Image.Image) -> tuple[int, int]:
    mask = _flag_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < MIN_FLAG_PIXELS:
        raise ValueError("no sufficiently large saturated signal flag")
    return int(xs.max() - xs.min()), int(ys.max() - ys.min())


def decision_from_image(image: Image.Image) -> str:
    """Return the state inferred solely from the final PNG pixels."""

    width, height = _extents(image)
    if width == height:
        raise ValueError("signal flag has an ambiguous square extent")
    return "raised" if height > width else "lowered"


def decision_margin(image: Image.Image) -> float:
    """Return the winner-versus-runner-up extent margin in raster pixels."""

    width, height = _extents(image)
    return round(float(abs(height - width)), 6)
