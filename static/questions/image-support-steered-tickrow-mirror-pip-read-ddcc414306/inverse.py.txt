"""Independent pixel-only inverse arm for the tick-row mirror world.

Receives one rendered PNG and the public question contract only. It never
imports the renderer, generator, prompts, verifier or analytic gold, and it
never receives latent scene data. It abstains by raising ``ValueError``
whenever the contract's structure is not recoverable from the pixels.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WHITE_MIN = 248
DARK_MAX = 90
COUNT_WORDS = ("one", "two", "three", "four")

MIN_CARD_AREA = 1800
TICK_W_RANGE = (2, 5)
TICK_H_RANGE = (11, 22)
PIP_RANGE = (5, 12)
GAP_TOLERANCE = 3.0


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Four-connected components of a boolean mask, as coordinate arrays."""

    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    out: list[np.ndarray] = []
    current = 0
    for start in np.argwhere(mask):
        sy, sx = int(start[0]), int(start[1])
        if labels[sy, sx]:
            continue
        current += 1
        stack = [(sy, sx)]
        labels[sy, sx] = current
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = current
                    stack.append((ny, nx))
        out.append(np.asarray(points, dtype=np.int32))
    return out


def _bbox(points: np.ndarray) -> tuple[int, int, int, int]:
    return (
        int(points[:, 0].min()),
        int(points[:, 1].min()),
        int(points[:, 0].max()),
        int(points[:, 1].max()),
    )


def _read_card(dark: np.ndarray, box: tuple[int, int, int, int]) -> tuple[list[float], int]:
    y0, x0, y1, x1 = box
    window = dark[y0 : y1 + 1, x0 : x1 + 1]
    ticks: list[float] = []
    pips = 0
    for points in _components(window):
        by0, bx0, by1, bx1 = _bbox(points)
        height = by1 - by0 + 1
        width = bx1 - bx0 + 1
        if TICK_W_RANGE[0] <= width <= TICK_W_RANGE[1] and TICK_H_RANGE[0] <= height <= TICK_H_RANGE[1]:
            ticks.append((bx0 + bx1) / 2.0)
        elif PIP_RANGE[0] <= width <= PIP_RANGE[1] and PIP_RANGE[0] <= height <= PIP_RANGE[1]:
            pips += 1
        else:
            raise ValueError("card carries a mark that is neither a contract tick nor a pip")
    if len(ticks) != 3:
        raise ValueError("card does not carry exactly three ticks")
    ticks.sort()
    return ticks, pips


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    white = np.all(array >= WHITE_MIN, axis=2)
    cards = [points for points in _components(white) if len(points) >= MIN_CARD_AREA]
    if len(cards) != 3:
        raise ValueError("the board does not show exactly three white cards")

    dark = np.all(array <= DARK_MAX, axis=2)
    readings = []
    for points in cards:
        y0, x0, y1, x1 = _bbox(points)
        if (y1 - y0 + 1) * (x1 - x0 + 1) > 1.40 * len(points):
            raise ValueError("a white card is not a solid rectangle")
        ticks, pips = _read_card(dark, (y0, x0, y1, x1))
        readings.append({"area": len(points), "ticks": ticks, "pips": pips})

    readings.sort(key=lambda item: item["area"], reverse=True)
    seal, key_a, key_b = readings
    if seal["area"] < 1.3 * key_a["area"]:
        raise ValueError("no single card is clearly the larger seal card")
    if seal["pips"] != 0:
        raise ValueError("the seal card carries pips")
    for key in (key_a, key_b):
        if not 1 <= key["pips"] <= 4:
            raise ValueError("a key card does not carry between one and four pips")

    a = seal["ticks"][1] - seal["ticks"][0]
    b = seal["ticks"][2] - seal["ticks"][1]
    if abs(a - b) < 1.0:
        raise ValueError("the seal rhythm is symmetric, so mirroring is undefined")

    mirrored = []
    slid = []
    for key in (key_a, key_b):
        p = key["ticks"][1] - key["ticks"][0]
        q = key["ticks"][2] - key["ticks"][1]
        if abs(p - b) <= GAP_TOLERANCE and abs(q - a) <= GAP_TOLERANCE:
            mirrored.append(key)
        elif abs(p - a) <= GAP_TOLERANCE and abs(q - b) <= GAP_TOLERANCE:
            slid.append(key)
        else:
            raise ValueError("a key rhythm is neither the seal slid nor the seal mirrored")
    if len(mirrored) != 1 or len(slid) != 1:
        raise ValueError("the mirrored key is not unique")

    return COUNT_WORDS[mirrored[0]["pips"] - 1]
