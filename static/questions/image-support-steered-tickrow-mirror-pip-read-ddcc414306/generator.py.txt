"""Tick-row mirror correspondence world.

Three white cards lie close together on a cluttered board: one large SEAL card
and two smaller KEY cards. Every card carries a row of three black ticks on a
shared seven-slot lattice. One key repeats the seal's tick rhythm slid sideways;
the other repeats it mirrored left-to-right. The answer is the number of pips
carried by the mirrored key.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

IMAGE_SIZE = 1280
LATTICE = 8
CELL = IMAGE_SIZE // LATTICE  # 160

BACKGROUND = (231, 223, 205)
CARD_FILL = (255, 255, 255)
CARD_EDGE = (18, 18, 20)
MARK = (18, 18, 20)

SEAL_OUTER = 68
KEY_OUTER = 56
BORDER = 2

SLOTS = 7
SLOT_PITCH = 6
TICK_W = 3
TICK_H = 16
PIP = 8
PIP_PITCH = 12

COUNT_WORDS = ("one", "two", "three", "four")

CLUTTER_COLORS = (
    (176, 132, 96),
    (120, 148, 128),
    (150, 118, 150),
    (96, 124, 160),
    (186, 154, 88),
    (140, 104, 104),
    (108, 140, 152),
    (162, 160, 118),
)

ORIENTATIONS = ("down", "up", "right", "left")


def _triangle_cells(orientation: str, row: int, col: int) -> list[tuple[int, int]]:
    """Three lattice cells whose pairwise separations are 2, sqrt2 and sqrt2."""

    if orientation == "down":
        return [(row, col), (row, col + 2), (row + 1, col + 1)]
    if orientation == "up":
        return [(row + 1, col), (row + 1, col + 2), (row, col + 1)]
    if orientation == "right":
        return [(row, col), (row + 2, col), (row + 1, col + 1)]
    return [(row, col + 1), (row + 2, col + 1), (row + 1, col)]


def _cell_center(cell: tuple[int, int]) -> tuple[int, int]:
    return CELL * cell[1] + CELL // 2, CELL * cell[0] + CELL // 2


def _card_box(cell: tuple[int, int], outer: int) -> tuple[int, int, int, int]:
    cx, cy = _cell_center(cell)
    half = outer // 2
    return cx - half, cy - half, cx + half, cy + half


def _gap_pairs(total: int) -> list[tuple[int, int]]:
    return [(a, total - a) for a in range(1, total) if a != total - a]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(0x51DE * (seed + 1) + 977)

    # Cluster placement: randomize absolute position over the whole board.
    orientation = ORIENTATIONS[rng.randrange(4)]
    if orientation in ("down", "up"):
        row = rng.randrange(0, LATTICE - 1)
        col = rng.randrange(0, LATTICE - 2)
    else:
        row = rng.randrange(0, LATTICE - 2)
        col = rng.randrange(0, LATTICE - 1)
    cells = _triangle_cells(orientation, row, col)
    rng.shuffle(cells)
    seal_cell, key_cell_a, key_cell_b = cells

    # Tick rhythm. total in {3,4,5} keeps at least two sliding offsets available.
    total = rng.choice([3, 4, 4, 5, 5])
    gaps = rng.choice(_gap_pairs(total))
    offsets = list(range(0, SLOTS - total))
    seal_s0 = rng.choice(offsets)
    slide_choices = [value for value in offsets if value != seal_s0]
    slide_s0 = rng.choice(slide_choices)
    mirror_s0 = rng.choice(offsets)

    # Pip attributes: the decision class is cycled, the foil count is distinct.
    answer_pips = 1 + (seed % 4)
    foil_pips = rng.choice([n for n in (1, 2, 3, 4) if n != answer_pips])

    mirror_is_first = rng.random() < 0.5
    mirror_cell = key_cell_a if mirror_is_first else key_cell_b
    slide_cell = key_cell_b if mirror_is_first else key_cell_a

    cards = [
        {
            "role": "seal",
            "cell": list(seal_cell),
            "s0": seal_s0,
            "gaps": [gaps[0], gaps[1]],
            "pips": 0,
        },
        {
            "role": "key",
            "cell": list(mirror_cell),
            "s0": mirror_s0,
            "gaps": [gaps[1], gaps[0]],
            "pips": answer_pips,
        },
        {
            "role": "key",
            "cell": list(slide_cell),
            "s0": slide_s0,
            "gaps": [gaps[0], gaps[1]],
            "pips": foil_pips,
        },
    ]
    rng.shuffle(cards)

    keepout = []
    for card in cards:
        outer = SEAL_OUTER if card["role"] == "seal" else KEY_OUTER
        x0, y0, x1, y1 = _card_box(tuple(card["cell"]), outer)
        keepout.append((x0 - 12, y0 - 12, x1 + 12, y1 + 12))

    clutter = _sample_clutter(rng, keepout)

    return {
        "image_size": IMAGE_SIZE,
        "orientation": orientation,
        "cards": cards,
        "clutter": clutter,
    }


def _sample_clutter(rng: random.Random, keepout: list[tuple[int, int, int, int]]) -> list[dict]:
    placed: list[dict] = []
    boxes: list[tuple[int, int, int, int]] = []
    attempts = 0
    target = 58
    while len(placed) < target and attempts < 4000:
        attempts += 1
        kind = rng.choice(["dot", "square", "bar", "ring"])
        color = CLUTTER_COLORS[rng.randrange(len(CLUTTER_COLORS))]
        if kind == "dot":
            radius = rng.randint(3, 7)
            width = height = 2 * radius
            params = {"radius": radius}
        elif kind == "square":
            width = height = rng.randint(7, 15)
            params = {}
        elif kind == "bar":
            length = rng.randint(12, 26)
            thick = rng.randint(3, 5)
            if rng.random() < 0.5:
                width, height = length, thick
            else:
                width, height = thick, length
            params = {}
        else:
            width = height = rng.randint(10, 18)
            params = {"stroke": 3}
        x0 = rng.randint(6, IMAGE_SIZE - width - 7)
        y0 = rng.randint(6, IMAGE_SIZE - height - 7)
        box = (x0, y0, x0 + width, y0 + height)
        pad = (box[0] - 5, box[1] - 5, box[2] + 5, box[3] + 5)
        if any(_overlap(pad, other) for other in keepout):
            continue
        if any(_overlap(pad, other) for other in boxes):
            continue
        boxes.append(box)
        placed.append({"kind": kind, "box": list(box), "color": list(color), **params})
    return placed


def _overlap(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> bool:
    return not (a[2] < b[0] or b[2] < a[0] or a[3] < b[1] or b[3] < a[1])


def _tick_slots(card: dict) -> list[int]:
    s0 = int(card["s0"])
    g1, g2 = int(card["gaps"][0]), int(card["gaps"][1])
    return [s0, s0 + g1, s0 + g1 + g2]


def _draw_card(draw: ImageDraw.ImageDraw, card: dict) -> None:
    is_seal = card["role"] == "seal"
    outer = SEAL_OUTER if is_seal else KEY_OUTER
    x0, y0, x1, y1 = _card_box(tuple(card["cell"]), outer)
    draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=CARD_EDGE)
    ix0, iy0 = x0 + BORDER, y0 + BORDER
    inner = outer - 2 * BORDER
    draw.rectangle([ix0, iy0, ix0 + inner - 1, iy0 + inner - 1], fill=CARD_FILL)

    row_span = SLOT_PITCH * (SLOTS - 1) + TICK_W
    first_center = ix0 + (inner - row_span) // 2 + TICK_W // 2
    tick_top = iy0 + (10 if is_seal else 8)
    for slot in _tick_slots(card):
        cx = first_center + SLOT_PITCH * slot
        draw.rectangle([cx - 1, tick_top, cx + 1, tick_top + TICK_H - 1], fill=MARK)

    pips = int(card["pips"])
    if pips:
        width = PIP + PIP_PITCH * (pips - 1)
        px = ix0 + (inner - width) // 2
        py = iy0 + inner - PIP - 8
        for index in range(pips):
            left = px + PIP_PITCH * index
            draw.rectangle([left, py, left + PIP - 1, py + PIP - 1], fill=MARK)


def render(scene: dict) -> Image.Image:
    size = int(scene["image_size"])
    image = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for item in scene["clutter"]:
        x0, y0, x1, y1 = item["box"]
        color = tuple(item["color"])
        if item["kind"] == "dot":
            draw.ellipse([x0, y0, x1 - 1, y1 - 1], fill=color)
        elif item["kind"] == "ring":
            draw.ellipse([x0, y0, x1 - 1, y1 - 1], outline=color, width=int(item["stroke"]))
        else:
            draw.rectangle([x0, y0, x1 - 1, y1 - 1], fill=color)
    for card in scene["cards"]:
        _draw_card(draw, card)
    return image


def _roles(scene: dict) -> tuple[dict, dict, dict]:
    seal = next(card for card in scene["cards"] if card["role"] == "seal")
    keys = [card for card in scene["cards"] if card["role"] == "key"]
    a, b = int(seal["gaps"][0]), int(seal["gaps"][1])
    mirrored = [k for k in keys if [int(k["gaps"][0]), int(k["gaps"][1])] == [b, a]]
    slid = [k for k in keys if [int(k["gaps"][0]), int(k["gaps"][1])] == [a, b]]
    if len(keys) != 2 or len(mirrored) != 1 or len(slid) != 1:
        raise ValueError("scene does not carry one mirrored and one slid key")
    return seal, mirrored[0], slid[0]


def analytic_gold(scene: dict) -> str:
    _, mirrored, _ = _roles(scene)
    return COUNT_WORDS[int(mirrored["pips"]) - 1]


def margin(scene: dict) -> float:
    seal, _, _ = _roles(scene)
    a, b = int(seal["gaps"][0]), int(seal["gaps"][1])
    return float(SLOT_PITCH * abs(a - b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabelings that must preserve the analytic gold."""

    import copy

    out: list[tuple[str, dict]] = []

    swapped = copy.deepcopy(scene)
    cards = swapped["cards"]
    swapped["cards"] = [cards[2], cards[1], cards[0]]
    out.append(("reverse_card_records", swapped))

    rotated = copy.deepcopy(scene)
    cards = rotated["cards"]
    rotated["cards"] = [cards[1], cards[2], cards[0]]
    out.append(("rotate_card_records", rotated))

    reversed_clutter = copy.deepcopy(scene)
    reversed_clutter["clutter"] = list(reversed(reversed_clutter["clutter"]))
    out.append(("reverse_clutter_draw_order", reversed_clutter))

    return out
