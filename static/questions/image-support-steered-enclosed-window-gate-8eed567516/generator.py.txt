from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 256
TEAL = (18, 154, 145)
WHITE = (250, 250, 247)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    closed = bool(seed % 2)
    outer_w = rng.randrange(190, 211, 2)
    outer_h = rng.randrange(190, 211, 2)
    wall = rng.randrange(66, 73)
    gate_width = rng.randrange(12, 17)
    gate_depth = rng.randrange(12, 17)
    gate_side = rng.choice(["top", "right", "bottom", "left"])
    if gate_side == "top":
        x, y = 128 - outer_w // 2, 64 - wall // 2
    elif gate_side == "bottom":
        x, y = 128 - outer_w // 2, 192 - outer_h + wall // 2
    elif gate_side == "left":
        x, y = 64 - wall // 2, 128 - outer_h // 2
    else:
        x, y = 192 - outer_w + wall // 2, 128 - outer_h // 2
    return {
        "closed": closed,
        "x": x,
        "y": y,
        "outer_w": outer_w,
        "outer_h": outer_h,
        "wall": wall,
        "gate_side": gate_side,
        "gate_width": gate_width,
        "gate_depth": gate_depth,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), WHITE)
    draw = ImageDraw.Draw(image)
    x, y = int(scene["x"]), int(scene["y"])
    w, h = int(scene["outer_w"]), int(scene["outer_h"])
    wall = int(scene["wall"])
    draw.rectangle((x, y, x + w - 1, y + h - 1), fill=TEAL)
    side = scene["gate_side"]
    depth = int(scene["gate_depth"])
    draw.rectangle((x + wall, y + wall, x + w - wall - 1, y + h - wall - 1), fill=WHITE)
    g = int(scene["gate_width"])
    if side in ("top", "bottom"):
        gx0 = x + (w - g) // 2
        gx1 = gx0 + g - 1
        if side == "top":
            if not bool(scene["closed"]):
                draw.rectangle((gx0, y, gx1, y + wall), fill=WHITE)
            else:
                p0 = y + (wall - depth) // 2
                p1 = p0 + depth - 1
                draw.rectangle((gx0, y, gx1, p0 - 1), fill=WHITE)
                draw.rectangle((gx0, p1 + 1, gx1, y + wall), fill=WHITE)
        else:
            if not bool(scene["closed"]):
                draw.rectangle((gx0, y + h - wall - 1, gx1, y + h - 1), fill=WHITE)
            else:
                p0 = y + h - wall + (wall - depth) // 2
                p1 = p0 + depth - 1
                draw.rectangle((gx0, y + h - wall - 1, gx1, p0 - 1), fill=WHITE)
                draw.rectangle((gx0, p1 + 1, gx1, y + h - 1), fill=WHITE)
    else:
        gy0 = y + (h - g) // 2
        gy1 = gy0 + g - 1
        if side == "left":
            if not bool(scene["closed"]):
                draw.rectangle((x, gy0, x + wall, gy1), fill=WHITE)
            else:
                p0 = x + (wall - depth) // 2
                p1 = p0 + depth - 1
                draw.rectangle((x, gy0, p0 - 1, gy1), fill=WHITE)
                draw.rectangle((p1 + 1, gy0, x + wall, gy1), fill=WHITE)
        else:
            if not bool(scene["closed"]):
                draw.rectangle((x + w - wall - 1, gy0, x + w - 1, gy1), fill=WHITE)
            else:
                p0 = x + w - wall + (wall - depth) // 2
                p1 = p0 + depth - 1
                draw.rectangle((x + w - wall - 1, gy0, p0 - 1, gy1), fill=WHITE)
                draw.rectangle((p1 + 1, gy0, x + w - 1, gy1), fill=WHITE)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["closed"]) else "no"


def margin(scene: dict) -> float:
    return float(min(int(scene["gate_depth"]), int(scene["gate_width"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
