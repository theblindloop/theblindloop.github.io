from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_TEAL = np.array([18, 154, 145], dtype=np.int16)


def _flood_exterior(background: np.ndarray) -> np.ndarray:
    h, w = background.shape
    seen = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    for x in range(w):
        if background[0, x]: queue.append((0, x))
        if background[h - 1, x]: queue.append((h - 1, x))
    for y in range(h):
        if background[y, 0]: queue.append((y, 0))
        if background[y, w - 1]: queue.append((y, w - 1))
    while queue:
        y, x = queue.popleft()
        if seen[y, x] or not background[y, x]:
            continue
        seen[y, x] = True
        if y: queue.append((y - 1, x))
        if y + 1 < h: queue.append((y + 1, x))
        if x: queue.append((y, x - 1))
        if x + 1 < w: queue.append((y, x + 1))
    return seen


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dist = np.sqrt(np.sum((arr - _TEAL) ** 2, axis=2))
    teal = dist < 35.0
    if int(teal.sum()) < 1200:
        raise ValueError("teal sample is absent or materially corrupted")
    ys, xs = np.nonzero(teal)
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    if y0 < 2 or x0 < 2 or y1 > teal.shape[0] - 2 or x1 > teal.shape[1] - 2:
        raise ValueError("sample touches image border")
    exterior = _flood_exterior(~teal)
    enclosed = (~teal) & (~exterior)
    return "yes" if int(enclosed.sum()) >= 500 else "no"
