"""Independent PNG-only inverse for the half-turn center decision."""

from __future__ import annotations

import math
from functools import lru_cache

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            stack = [(x, y)]
            seen[y, x] = True
            group: list[tuple[int, int]] = []
            while stack:
                px, py = stack.pop()
                group.append((px, py))
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((nx, ny))
            groups.append(group)
    return groups


def _center(group: list[tuple[int, int]]) -> tuple[float, float]:
    return (
        sum(x for x, _ in group) / len(group),
        sum(y for _, y in group) / len(group),
    )


def _match_error(points: list[tuple[float, float]], center: tuple[float, float]) -> float:
    cx, cy = center
    turned = [(2.0 * cx - x, 2.0 * cy - y) for x, y in points]
    matrix = [
        [math.hypot(tx - px, ty - py) for px, py in points]
        for tx, ty in turned
    ]
    size = len(points)

    @lru_cache(maxsize=None)
    def visit(row: int, occupied: int) -> float:
        if row == size:
            return 0.0
        best = float("inf")
        for column in range(size):
            bit = 1 << column
            if not occupied & bit:
                best = min(best, matrix[row][column] + visit(row + 1, occupied | bit))
        return best

    return visit(0, 0) / size


def _single_center(mask: np.ndarray, label: str) -> tuple[float, float]:
    groups = [group for group in _components(mask) if len(group) >= 80]
    if len(groups) != 1:
        raise ValueError(f"expected one {label} ring, found {len(groups)}")
    if not 170 <= len(groups[0]) <= 380:
        raise ValueError(f"{label} marker has invalid ink area")
    return _center(groups[0])


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red_channel = pixels[:, :, 0].astype(int)
    green_channel = pixels[:, :, 1].astype(int)
    blue_channel = pixels[:, :, 2].astype(int)

    dark = (red_channel < 75) & (green_channel < 75) & (blue_channel < 75)
    dot_groups = [group for group in _components(dark) if 120 <= len(group) <= 320]
    if len(dot_groups) != 12:
        raise ValueError(f"expected twelve black dots, found {len(dot_groups)}")
    points = [_center(group) for group in dot_groups]

    red_mask = (red_channel > 150) & (red_channel > green_channel + 70) & (red_channel > blue_channel + 55)
    blue_mask = (blue_channel > 140) & (blue_channel > red_channel + 55) & (blue_channel > green_channel + 20)
    centers = {
        "red": _single_center(red_mask, "red"),
        "blue": _single_center(blue_mask, "blue"),
    }
    scores = {name: _match_error(points, center) for name, center in centers.items()}
    ordered = sorted(scores, key=scores.get)
    if scores[ordered[1]] - scores[ordered[0]] < 8.0:
        raise ValueError("ambiguous half-turn center")
    return ordered[0]
