"""Deterministic scene sampler and renderer for global half-turn symmetry."""

from __future__ import annotations

import math
import random
from functools import lru_cache

from PIL import Image, ImageDraw


WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 249, 246)
DOT_COLOR = (25, 29, 35)
RED = (218, 52, 62)
BLUE = (43, 103, 211)
DOT_RADIUS = 8
RING_RADIUS = 13
RING_WIDTH = 5


def _point_list(scene: dict) -> list[tuple[float, float]]:
    return [(float(x), float(y)) for x, y in scene["points"]]


def _candidate(scene: dict, name: str) -> tuple[float, float]:
    x, y = scene[name]
    return float(x), float(y)


def _assignment_score(points: list[tuple[float, float]], center: tuple[float, float]) -> float:
    """Minimum mean distance from the half-turned set to the visible set."""
    cx, cy = center
    transformed = [(2.0 * cx - x, 2.0 * cy - y) for x, y in points]
    costs = [
        [math.hypot(tx - x, ty - y) for x, y in points]
        for tx, ty in transformed
    ]
    n = len(points)

    @lru_cache(maxsize=None)
    def solve(row: int, used: int) -> float:
        if row == n:
            return 0.0
        return min(
            costs[row][column] + solve(row + 1, used | (1 << column))
            for column in range(n)
            if not used & (1 << column)
        )

    return solve(0, 0) / n


def _scores(scene: dict) -> dict[str, float]:
    points = _point_list(scene)
    return {
        "red": _assignment_score(points, _candidate(scene, "red")),
        "blue": _assignment_score(points, _candidate(scene, "blue")),
    }


def sample_scene(seed: int) -> dict:
    """Create six separated antipodal pairs around one counterbalanced ring."""
    rng = random.Random(int(seed) ^ 0x71A5C3D9)
    target = "red" if int(seed) % 2 else "blue"
    cx = 320 + rng.randint(-22, 22)
    cy = 218 + rng.randint(-16, 16)
    direction = rng.uniform(0.0, 2.0 * math.pi)
    offset = 34 + 6 * ((int(seed) // 2) % 4)
    dx = int(round(offset * math.cos(direction)))
    dy = int(round(offset * math.sin(direction)))
    if math.hypot(dx, dy) < 31:
        raise RuntimeError("candidate rings unexpectedly overlap")
    true_center = [cx, cy]
    decoy_center = [cx + dx, cy + dy]

    for _attempt in range(4000):
        vectors: list[tuple[int, int]] = []
        points: list[list[int]] = []
        while len(vectors) < 6:
            radius = rng.uniform(72.0, 174.0)
            angle = rng.uniform(0.0, math.pi)
            vx = int(round(radius * math.cos(angle)))
            vy = int(round(radius * math.sin(angle)))
            if abs(vx) < 18 and abs(vy) < 18:
                continue
            proposed = [[cx + vx, cy + vy], [cx - vx, cy - vy]]
            if any(not (28 <= x <= WIDTH - 28 and 28 <= y <= 402) for x, y in proposed):
                continue
            if any(
                math.hypot(x - qx, y - qy) < 27.0
                for x, y in proposed
                for qx, qy in points
            ):
                continue
            if any(
                math.hypot(x - mx, y - my) < 34.0
                for x, y in proposed
                for mx, my in (true_center, decoy_center)
            ):
                continue
            vectors.append((vx, vy))
            points.extend(proposed)

        rng.shuffle(points)
        scene = {
            "points": points,
            "red": true_center if target == "red" else decoy_center,
            "blue": true_center if target == "blue" else decoy_center,
        }
        scores = _scores(scene)
        if scores[target] < 1e-9 and scores["blue" if target == "red" else "red"] > 18.0:
            return scene
    raise RuntimeError(f"could not sample a robust half-turn scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y in _point_list(scene):
        draw.ellipse(
            (x - DOT_RADIUS, y - DOT_RADIUS, x + DOT_RADIUS, y + DOT_RADIUS),
            fill=DOT_COLOR,
        )
    # Keep scene-field accesses explicit for the rendered-field audit.
    ring_specs = ((scene["red"], RED), (scene["blue"], BLUE))
    for center, color in ring_specs:
        x, y = float(center[0]), float(center[1])
        draw.ellipse(
            (x - RING_RADIUS, y - RING_RADIUS, x + RING_RADIUS, y + RING_RADIUS),
            outline=color,
            width=RING_WIDTH,
        )
    return image


def analytic_gold(scene: dict) -> str:
    # Spell out every label input so the rendered-field audit can observe that
    # gold depends on the two visible rings as well as the unlabeled dot set.
    points = [(float(x), float(y)) for x, y in scene["points"]]
    red = (float(scene["red"][0]), float(scene["red"][1]))
    blue = (float(scene["blue"][0]), float(scene["blue"][1]))
    scores = {
        "red": _assignment_score(points, red),
        "blue": _assignment_score(points, blue),
    }
    return min(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values())
    return values[1] - values[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "points": list(reversed(scene["points"])),
        "red": list(scene["red"]),
        "blue": list(scene["blue"]),
    }
    return [("reverse_unlabeled_point_order", twin)]
