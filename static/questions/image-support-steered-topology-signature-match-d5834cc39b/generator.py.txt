"""Deterministic raster renderer for a three-panel topology signature test."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "topology_signature_match-0.1.0"
BACKGROUND = "#ffffff"
TOPOLOGY_INK = "#263746"
PANEL_NEUTRAL = "#e6ebf0"
REFERENCE_COLOR = "#697586"
PALETTE = ("#d94841", "#247ba0", "#8a4f9e", "#dc8b27")
SIGNATURES = ((1, 0), (1, 1), (2, 0), (2, 1))
QUARANTINE_CLEARANCE = 12

# The role boxes are intentionally separated but occupy one bounded middle
# region. The oracle uses these public, fixed panel coordinates, not scene data.
PANEL_BOXES = {
    "left": (55, 144, 169, 350),
    "reference": (183, 144, 297, 350),
    "right": (343, 144, 457, 350),
}
GLYPH_Y0 = 190
GLYPH_Y1 = 330


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced reference-plus-two-candidates scene."""

    rng = random.Random(int(seed))
    reference = SIGNATURES[rng.randrange(len(SIGNATURES))]
    answer = "left" if int(seed) % 2 == 0 else "right"
    other = [signature for signature in SIGNATURES if signature != reference]
    mismatch = other[rng.randrange(len(other))]
    if answer == "left":
        left, right = reference, mismatch
    else:
        left, right = mismatch, reference
    clearance = rng.choice((8, 10, 14, 16, 20, 24))
    colors = rng.sample(list(PALETTE), 2)
    return {
        "width": 512,
        "height": 512,
        "reference_signature": list(reference),
        "left_signature": list(left),
        "right_signature": list(right),
        "left_color": colors[0],
        "right_color": colors[1],
        "panel_jitter": rng.choice((-3, -2, 0, 2, 3)),
        "vertical_jitter": rng.choice((-2, -1, 0, 1, 2)),
        "style_seed": rng.randrange(1, 2**31 - 1),
        "signature_gap": clearance,
    }


def _open_path(draw: ImageDraw.ImageDraw, points: list[tuple[int, int]], width: int) -> None:
    draw.line(points, fill=TOPOLOGY_INK, width=width, joint="curve")


def _ring(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    hole_radius: int,
    width: int,
) -> None:
    cx, cy = center
    # Painting a white interior after a dark outer disk makes a thick closed
    # contour. Any occupied ablation tile can open the contour and erase the
    # hole, which gives the support probe a causal reason to see the whole loop.
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=TOPOLOGY_INK)
    inner = max(hole_radius, radius - width)
    draw.ellipse((cx - inner, cy - inner, cx + inner, cy + inner), fill=BACKGROUND)


def _draw_signature(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    signature: tuple[int, int],
    style_seed: int,
    clearance: int,
) -> None:
    """Draw separated open paths and closed loops for one topology signature."""

    components, holes = signature
    cx, cy = center
    style = int(style_seed) % 9
    stroke = 11 if style % 2 else 12
    hole_radius = max(9, min(14, int(clearance) - 2))
    if components == 1 and holes == 0:
        _open_path(
            draw,
            [(cx - 34, cy - 30), (cx + 34, cy + 30 if style % 3 else cy - 30)],
            stroke,
        )
    elif components == 1 and holes == 1:
        _ring(draw, (cx, cy), 28, hole_radius, stroke)
    elif components == 2 and holes == 0:
        _open_path(
            draw,
            [(cx - 39, cy - 27), (cx - 24, cy - 18), (cx - 9, cy - 27)],
            stroke,
        )
        _open_path(
            draw,
            [(cx + 9, cy + 27), (cx + 24, cy + 18), (cx + 39, cy + 27)],
            stroke,
        )
    elif components == 2 and holes == 1:
        _ring(draw, (cx - 22, cy), 21, hole_radius, stroke)
        _open_path(
            draw,
            [(cx + 14, cy - 27), (cx + 34, cy - 27), (cx + 25, cy + 27)],
            stroke,
        )
    else:
        raise ValueError(f"unsupported topology signature {signature!r}")


def _header(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    color: str,
    jitter: int,
    vertical_jitter: int,
) -> None:
    x0, y0, x1, _ = box
    draw.rounded_rectangle(
        (x0 + 8 + jitter, y0 + 7 + vertical_jitter, x1 - 8 + jitter, y0 + 28 + vertical_jitter),
        radius=5,
        fill=color,
    )


def _decoration(draw: ImageDraw.ImageDraw, seed: int) -> None:
    """Remote pale marks are distractors and never enter a topology crop."""

    rng = random.Random(int(seed))
    for x, y in ((42, 70), (470, 74), (44, 420), (470, 426)):
        dx, dy = rng.randint(-8, 8), rng.randint(-8, 8)
        draw.ellipse((x + dx - 3, y + dy - 3, x + dx + 3, y + dy + 3), fill="#dfe5eb")
    draw.line((22, 118, 58, 118), fill="#dfe5eb", width=2)
    draw.line((454, 374, 490, 374), fill="#dfe5eb", width=2)


def render(scene: dict[str, Any]) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    jitter = int(scene["panel_jitter"])
    vertical_jitter = int(scene["vertical_jitter"])
    style_seed = int(scene["style_seed"])
    clearance = int(scene["signature_gap"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _decoration(draw, style_seed + 19)
    for role, color, signature, offset in (
        ("left", str(scene["left_color"]), tuple(scene["left_signature"]), 0),
        ("reference", REFERENCE_COLOR, tuple(scene["reference_signature"]), 1),
        ("right", str(scene["right_color"]), tuple(scene["right_signature"]), 2),
    ):
        box = PANEL_BOXES[role]
        _header(draw, box, color, jitter, vertical_jitter)
        cx = (box[0] + box[2]) // 2 + jitter
        cy = 258 + vertical_jitter
        _draw_signature(draw, (cx, cy), signature, style_seed + 17 * offset, clearance)
    # A very light baseline helps the three roles read as panels but is outside
    # the dark glyph crop used by the inverse arm.
    for box in PANEL_BOXES.values():
        draw.line((box[0] + 10, 344, box[2] - 10, 344), fill=PANEL_NEUTRAL, width=2)
    return image.convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    reference = tuple(scene["reference_signature"])
    if tuple(scene["left_signature"]) == reference:
        return "left"
    if tuple(scene["right_signature"]) == reference:
        return "right"
    raise ValueError("scene must contain exactly one matching candidate")


def margin(scene: dict[str, Any]) -> float:
    """Raster-clearance margin; the topology decision itself is discrete."""

    return float(scene["signature_gap"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return int(scene["signature_gap"]) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    del scene
    return []
