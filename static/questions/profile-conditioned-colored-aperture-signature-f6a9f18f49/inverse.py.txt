"""Independent pixel-only inverse for the colored aperture-signature world.

This module intentionally duplicates the public palette rather than importing
renderer state.  It receives only an RGB PNG, extracts colored components, and
classifies each component's central aperture before comparing color signatures.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


ORACLE_VERSION = "colored-aperture-signature-oracle-0.1.0"
BACKGROUND = np.array([248, 248, 244], dtype=np.float32)
PALETTE: dict[str, tuple[int, int, int]] = {
    "crimson": (211, 57, 73),
    "amber": (229, 153, 44),
    "teal": (42, 164, 154),
    "cobalt": (63, 103, 205),
    "violet": (145, 87, 190),
}
COLORS = tuple(PALETTE)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask), strict=True):
        if visited[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 300:
            component = np.zeros_like(mask, dtype=bool)
            yy, xx = zip(*points, strict=True)
            component[np.array(yy), np.array(xx)] = True
            components.append(component)
    return components


def _color_components(image: Image.Image) -> dict[str, list[np.ndarray]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    found: dict[str, list[np.ndarray]] = {}
    for color, rgb in PALETTE.items():
        distance = np.linalg.norm(arr - np.asarray(rgb, dtype=np.float32), axis=-1)
        found[color] = _components(distance <= 72.0)
    return found


def _component_feature(component: np.ndarray) -> tuple[float, float, bool]:
    ys, xs = np.nonzero(component)
    if len(ys) < 300:
        raise ValueError("colored component is too small")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    cx = (x0 + x1) // 2
    cy = (y0 + y1) // 2
    ylo, yhi = max(0, cy - 3), min(component.shape[0], cy + 4)
    xlo, xhi = max(0, cx - 3), min(component.shape[1], cx + 4)
    central_colored_fraction = float(component[ylo:yhi, xlo:xhi].mean())
    has_aperture = central_colored_fraction < 0.30
    return float(xs.mean()), float(ys.mean()), has_aperture


def _signatures(image: Image.Image) -> list[dict[str, bool]]:
    components = _color_components(image)
    signatures = [dict() for _ in range(3)]
    for color in COLORS:
        if len(components[color]) != 3:
            raise ValueError(f"expected one {color} component in each panel")
        for component in components[color]:
            cx, _cy, has_aperture = _component_feature(component)
            panel = 0 if cx < 320 else 1 if cx < 640 else 2
            if color in signatures[panel]:
                raise ValueError("duplicate color in panel")
            signatures[panel][color] = has_aperture
    if any(set(signature) != set(COLORS) for signature in signatures):
        raise ValueError("incomplete color signature")
    return signatures


def decision_from_image(image: Image.Image) -> str:
    """Return A or B after solving the color-labeled aperture correspondence."""

    reference, candidate_a, candidate_b = _signatures(image)
    score_a = sum(reference[color] != candidate_a[color] for color in COLORS)
    score_b = sum(reference[color] != candidate_b[color] for color in COLORS)
    if score_a == score_b:
        raise ValueError("image has an ambiguous aperture-signature decision")
    return "A" if score_a < score_b else "B"
