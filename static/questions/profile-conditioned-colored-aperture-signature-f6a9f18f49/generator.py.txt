"""Renderer and analytic label for the colored aperture-signature world.

The scene contains three panels.  Each panel has one isolated token for each
palette color.  A token is either a solid disk or a thick colored annulus with
an enclosed background aperture.  The question compares the color-to-aperture
mapping, not the token positions.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored-aperture-signature-0.1.0"
WIDTH = 960
HEIGHT = 500
BACKGROUND = (248, 248, 244)
PANEL_FILL = (255, 255, 252)
PANEL_BORDER = (182, 185, 187)
LABEL_INK = (47, 52, 58)
TOKEN_OUTLINE = (39, 44, 51)
COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "crimson": (211, 57, 73),
    "amber": (229, 153, 44),
    "teal": (42, 164, 154),
    "cobalt": (63, 103, 205),
    "violet": (145, 87, 190),
}
COLORS = tuple(COLOR_RGB)
PANEL_LEFTS = (20, 340, 660)
PANEL_TOP = 84
PANEL_WIDTH = 280
PANEL_BOTTOM = 452
SLOT_OFFSETS = (40, 90, 140, 190, 240)
HOLE_RATIO = 0.43


def _signature(panel: list[dict]) -> dict[str, bool]:
    return {str(token["color"]): bool(token["hole"]) for token in panel}


def _scores(scene: dict) -> tuple[int, int]:
    reference = _signature(scene["reference"])
    return tuple(
        sum(reference.get(color) != _signature(panel).get(color) for color in COLORS)
        for panel in (scene["candidate_a"], scene["candidate_b"])
    )  # type: ignore[return-value]


def sample_scene(seed: int) -> dict:
    """Return a deterministic, balanced scene for one integer seed."""

    rng = random.Random(int(seed))
    colors = list(COLORS)
    rng.shuffle(colors)
    hole_slots = sorted(rng.sample(range(5), 2))
    hole_pattern = [index in hole_slots for index in range(5)]

    ref_colors = list(colors)
    rng.shuffle(ref_colors)
    ref_radius = 24
    candidate_radius = 24

    def positions(panel_left: int, radius: int, y: int) -> list[tuple[int, int]]:
        return [(panel_left + offset, y) for offset in SLOT_OFFSETS]

    ref_positions = positions(PANEL_LEFTS[0], ref_radius, 274)
    candidate_positions = positions(PANEL_LEFTS[1], candidate_radius, 274)
    candidate_positions_b = positions(PANEL_LEFTS[2], candidate_radius, 274)

    ref_hole_colors = [ref_colors[index] for index in hole_slots]
    ref_solid_colors = [ref_colors[index] for index in range(5) if index not in hole_slots]
    rng.shuffle(ref_hole_colors)
    rng.shuffle(ref_solid_colors)
    matching_colors: list[str] = []
    hole_iter = iter(ref_hole_colors)
    solid_iter = iter(ref_solid_colors)
    for has_hole in hole_pattern:
        matching_colors.append(next(hole_iter) if has_hole else next(solid_iter))

    decoy_colors = list(matching_colors)
    solid_slot = next(index for index, value in enumerate(hole_pattern) if not value)
    ring_slot = next(index for index, value in enumerate(hole_pattern) if value)
    decoy_colors[ring_slot], decoy_colors[solid_slot] = (
        decoy_colors[solid_slot],
        decoy_colors[ring_slot],
    )

    match_is_a = int(seed) % 2 == 0
    colors_a, colors_b = (
        (matching_colors, decoy_colors) if match_is_a else (decoy_colors, matching_colors)
    )

    def make_panel(
        panel_colors: Iterable[str],
        panel_positions: list[tuple[int, int]],
        radius: int,
    ) -> list[dict]:
        return [
            {
                "color": color,
                "hole": bool(hole_pattern[index]),
                "x": int(x),
                "y": int(y),
                "radius": int(radius),
            }
            for index, (color, (x, y)) in enumerate(zip(panel_colors, panel_positions, strict=True))
        ]

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "reference": make_panel(ref_colors, ref_positions, ref_radius),
        "candidate_a": make_panel(colors_a, candidate_positions, candidate_radius),
        "candidate_b": make_panel(colors_b, candidate_positions_b, candidate_radius),
    }


def render(scene: dict) -> Image.Image:
    """Rasterize a scene with supersampled vector-like Pillow drawing."""

    ss = 3
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    try:
        font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 16 * ss)
        small_font = ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", 13 * ss)
    except OSError:
        font = ImageFont.load_default()
        small_font = font

    def sc(value: float) -> int:
        return int(round(value * ss))

    panels = (scene["reference"], scene["candidate_a"], scene["candidate_b"])
    labels = ("REFERENCE", "A", "B")
    for index, (panel, label) in enumerate(zip(panels, labels, strict=True)):
        left = PANEL_LEFTS[index]
        right = left + PANEL_WIDTH
        draw.rounded_rectangle(
            (sc(left), sc(PANEL_TOP), sc(right), sc(PANEL_BOTTOM)),
            radius=sc(14),
            fill=PANEL_FILL,
            outline=PANEL_BORDER,
            width=sc(2),
        )
        label_box = draw.textbbox((0, 0), label, font=font)
        label_width = label_box[2] - label_box[0]
        draw.text(
            (sc(left + PANEL_WIDTH / 2) - label_width // 2, sc(107)),
            label,
            fill=LABEL_INK,
            font=font,
        )
        subtitle = "solid / aperture"
        subtitle_box = draw.textbbox((0, 0), subtitle, font=small_font)
        subtitle_width = subtitle_box[2] - subtitle_box[0]
        draw.text(
            (sc(left + PANEL_WIDTH / 2) - subtitle_width // 2, sc(134)),
            subtitle,
            fill=(112, 116, 120),
            font=small_font,
        )
        for token in panel:
            color = COLOR_RGB[str(token["color"])]
            x = float(token["x"])
            y = float(token["y"])
            radius = float(token["radius"])
            outer = (sc(x - radius), sc(y - radius), sc(x + radius), sc(y + radius))
            draw.ellipse(outer, fill=color, outline=TOKEN_OUTLINE, width=sc(2))
            if bool(token["hole"]):
                inner_radius = radius * HOLE_RATIO
                inner = (
                    sc(x - inner_radius),
                    sc(y - inner_radius),
                    sc(x + inner_radius),
                    sc(y + inner_radius),
                )
                draw.ellipse(
                    inner,
                    fill=background,
                    outline=TOKEN_OUTLINE,
                    width=sc(2),
                )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate with the reference color-to-hole map."""

    score_a, score_b = _scores(scene)
    if score_a == score_b:
        raise ValueError("candidate aperture signatures are tied")
    return "A" if score_a < score_b else "B"


def margin(scene: dict) -> float:
    """Return the runner-up minus winner signature mismatch count."""

    score_a, score_b = _scores(scene)
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    """Reject malformed or non-unique finite-resolution signature cases."""

    panels = (scene.get("reference"), scene.get("candidate_a"), scene.get("candidate_b"))
    if any(not isinstance(panel, list) or len(panel) != len(COLORS) for panel in panels):
        return True
    for panel in panels:
        colors = [token.get("color") for token in panel]
        if set(colors) != set(COLORS) or len(colors) != len(set(colors)):
            return True
        if sum(bool(token.get("hole")) for token in panel) != 2:
            return True
        if any(float(token.get("radius", 0)) < 18 for token in panel):
            return True
    score_a, score_b = _scores(scene)
    return score_a == score_b or min(score_a, score_b) != 0 or abs(score_a - score_b) < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This minimal scene has no unrendered aliases or render-invariant labels."""

    return []
