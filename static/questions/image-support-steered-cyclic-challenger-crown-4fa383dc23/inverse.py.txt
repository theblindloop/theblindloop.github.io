"""Independent inverse arm using only pixels from the final RGB image."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CARD_FILL = np.array((226, 235, 250), dtype=np.int16)
PIP = np.array((125, 76, 173), dtype=np.int16)
GLYPH = np.array((36, 72, 120), dtype=np.int16)
RING = np.array((34, 139, 94), dtype=np.int16)

BEATS = {"circle": "triangle", "triangle": "square", "square": "circle"}


class OracleAbstention(ValueError):
    """Raised when the raster does not encode one uniquely replayable program."""


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[dict] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) < minimum:
            continue
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        output.append(
            {
                "area": len(points),
                "bbox": (min(xs), min(ys), max(xs), max(ys)),
                "centroid": (sum(xs) / len(xs), sum(ys) / len(ys)),
            }
        )
    return output


def _inside(point: tuple[float, float], bbox: tuple[int, int, int, int], pad: int = 0) -> bool:
    x, y = point
    x0, y0, x1, y1 = bbox
    return x0 - pad <= x <= x1 + pad and y0 - pad <= y <= y1 + pad


def _shape_kind(component: dict) -> str:
    x0, y0, x1, y1 = component["bbox"]
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if not (34 <= width <= 48 and 34 <= height <= 48):
        raise OracleAbstention(f"emblem extent is ambiguous: {width}x{height}")
    fill = component["area"] / float(width * height)
    if fill < 0.43:
        raise OracleAbstention(f"emblem is too incomplete: fill={fill:.3f}")
    if fill < 0.66:
        return "triangle"
    if fill < 0.90:
        return "circle"
    return "square"


def _advance(champion: str, challenger: str) -> str:
    return challenger if BEATS[challenger] == champion else champion


def decision_from_image(image: Image.Image) -> str:
    """Parse region membership, visible ordinal pips, silhouettes, then replay."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleAbstention("expected an RGB raster")

    card_mask = _near(array, CARD_FILL)
    cards = []
    for component in _components(card_mask, minimum=2000):
        x0, y0, x1, y1 = component["bbox"]
        width, height = x1 - x0 + 1, y1 - y0 + 1
        fill = component["area"] / float(width * height)
        if width >= 75 and height >= 65 and fill >= 0.52:
            cards.append(component)
    if len(cards) != 4:
        raise OracleAbstention(f"expected four intact detached cards, found {len(cards)}")

    pips = _components(_near(array, PIP), minimum=55)
    glyphs = _components(_near(array, GLYPH), minimum=250)
    if len(pips) != 10:
        raise OracleAbstention(f"expected ten ordinal pips, found {len(pips)}")
    if len(glyphs) != 5:
        raise OracleAbstention(f"expected five emblem silhouettes, found {len(glyphs)}")

    ordered: dict[int, str] = {}
    assigned_glyphs: set[int] = set()
    for card in cards:
        bbox = card["bbox"]
        pip_count = sum(_inside(component["centroid"], bbox, pad=5) for component in pips)
        inside_glyphs = [
            index
            for index, component in enumerate(glyphs)
            if _inside(component["centroid"], bbox, pad=5)
        ]
        if pip_count not in (1, 2, 3, 4) or len(inside_glyphs) != 1:
            raise OracleAbstention("card rank or challenger binding is ambiguous")
        if pip_count in ordered:
            raise OracleAbstention("card pip ranks are not unique")
        glyph_index = inside_glyphs[0]
        assigned_glyphs.add(glyph_index)
        ordered[pip_count] = _shape_kind(glyphs[glyph_index])
    if set(ordered) != {1, 2, 3, 4} or len(assigned_glyphs) != 4:
        raise OracleAbstention("the four ordered challengers are not uniquely bound")

    rings = _components(_near(array, RING), minimum=500)
    if len(rings) != 1 or rings[0]["area"] < 1500:
        raise OracleAbstention("starting ring is missing or incomplete")
    start_candidates = [
        index
        for index, component in enumerate(glyphs)
        if index not in assigned_glyphs and _inside(component["centroid"], rings[0]["bbox"])
    ]
    if len(start_candidates) != 1:
        raise OracleAbstention("ring does not bind exactly one starting emblem")
    champion = _shape_kind(glyphs[start_candidates[0]])
    for rank in (1, 2, 3, 4):
        champion = _advance(champion, ordered[rank])
    return champion
