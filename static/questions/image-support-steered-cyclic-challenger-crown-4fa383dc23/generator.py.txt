"""Deterministic renderer and analytic arm for the challenger-crown world."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 480
BACKGROUND = (250, 248, 241)
CARD_FILL = (226, 235, 250)
BORDER = (55, 65, 75)
PIP = (125, 76, 173)
GLYPH = (36, 72, 120)
RING = (34, 139, 94)
DISTRACTOR = (224, 132, 58)

SHAPES = ("circle", "triangle", "square")
# Each key defeats its value.
BEATS = {"circle": "triangle", "triangle": "square", "square": "circle"}
CARD_SIZE = (115, 104)
CARD_BASES = ((24, 24), (501, 24), (24, 352), (501, 352))


def _advance(champion: str, challenger: str) -> str:
    if BEATS[challenger] == champion:
        return challenger
    return champion


def _replay(start: str, challengers: list[str] | tuple[str, ...]) -> str:
    champion = start
    for challenger in challengers:
        champion = _advance(champion, challenger)
    return champion


def _transition_counts(start: str, challengers: tuple[str, ...]) -> tuple[int, int]:
    champion = start
    wins = 0
    holds = 0
    for challenger in challengers:
        updated = _advance(champion, challenger)
        if updated == champion:
            holds += 1
        else:
            wins += 1
        champion = updated
    return wins, holds


def _eligible_programs(target: str) -> list[tuple[str, tuple[str, ...]]]:
    programs: list[tuple[str, tuple[str, ...]]] = []
    for start in SHAPES:
        for challengers in itertools.product(SHAPES, repeat=4):
            wins, holds = _transition_counts(start, challengers)
            if (
                _replay(start, challengers) == target
                and start != target
                and challengers[-1] != target
                and set(challengers) == set(SHAPES)
                and wins >= 2
                and holds >= 1
            ):
                programs.append((start, challengers))
    if not programs:
        raise AssertionError(f"no balanced program for {target}")
    return programs


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely replayable scene from one integer seed."""
    rng = random.Random(int(seed))
    target = SHAPES[int(seed) % len(SHAPES)]
    programs = _eligible_programs(target)
    start, challengers = programs[rng.randrange(len(programs))]

    card_slots = list(range(4))
    rng.shuffle(card_slots)
    card_jitter = [[rng.randint(-6, 6), rng.randint(-6, 6)] for _ in range(4)]
    start_offset = [rng.randint(-9, 9), rng.randint(-9, 9)]

    distractor_pool = [
        (177, 178),
        (463, 178),
        (177, 300),
        (463, 300),
        (264, 116),
        (376, 362),
    ]
    rng.shuffle(distractor_pool)
    distractors = [list(point) for point in distractor_pool[: 3 + (int(seed) % 2)]]

    return {
        "start": start,
        "challengers": list(challengers),
        "card_slots": card_slots,
        "card_jitter": card_jitter,
        "start_offset": start_offset,
        "distractors": distractors,
    }


def _draw_shape(draw: ImageDraw.ImageDraw, kind: str, center: tuple[int, int], size: int) -> None:
    cx, cy = center
    half = size // 2
    if kind == "circle":
        draw.ellipse((cx - half, cy - half, cx + half, cy + half), fill=GLYPH)
    elif kind == "square":
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=GLYPH)
    elif kind == "triangle":
        draw.polygon(
            ((cx, cy - half), (cx - half, cy + half), (cx + half, cy + half)),
            fill=GLYPH,
        )
    else:
        raise ValueError(f"unknown emblem {kind!r}")


def _draw_diamond(draw: ImageDraw.ImageDraw, center: tuple[int, int]) -> None:
    x, y = center
    draw.polygon(((x, y - 7), (x + 7, y), (x, y + 7), (x - 7, y)), fill=DISTRACTOR)


def render(scene: dict) -> Image.Image:
    """Render from declared fields only; all order is encoded by visible pips."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    challengers = scene["challengers"]
    slots = scene["card_slots"]
    jitters = scene["card_jitter"]
    for order, challenger in enumerate(challengers, start=1):
        slot = int(slots[order - 1])
        base_x, base_y = CARD_BASES[slot]
        dx, dy = (int(value) for value in jitters[order - 1])
        x, y = base_x + dx, base_y + dy
        width, height = CARD_SIZE
        draw.rectangle((x, y, x + width, y + height), fill=CARD_FILL, outline=BORDER, width=4)

        spacing = 15
        first_x = x + width // 2 - spacing * (order - 1) // 2
        for index in range(order):
            px = first_x + index * spacing
            py = y + 18
            draw.ellipse((px - 5, py - 5, px + 5, py + 5), fill=PIP)
        _draw_shape(draw, challenger, (x + width // 2, y + 70), 42)

    for point in scene["distractors"]:
        _draw_diamond(draw, (int(point[0]), int(point[1])))

    offset_x, offset_y = (int(value) for value in scene["start_offset"])
    center = (WIDTH // 2 + offset_x, HEIGHT // 2 + offset_y)
    radius = 49
    draw.ellipse(
        (center[0] - radius, center[1] - radius, center[0] + radius, center[1] + radius),
        outline=RING,
        width=8,
    )
    _draw_shape(draw, scene["start"], center, 44)
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(scene["start"], scene["challengers"])


def margin(scene: dict) -> float:
    """Discrete runner-up margin: minimum one-symbol edits to change the result."""
    original = analytic_gold(scene)
    values = [scene["start"], *scene["challengers"]]
    for edit_count in range(1, len(values) + 1):
        for positions in itertools.combinations(range(len(values)), edit_count):
            alternatives = [tuple(shape for shape in SHAPES if shape != values[pos]) for pos in positions]
            for replacements in itertools.product(*alternatives):
                changed = list(values)
                for position, replacement in zip(positions, replacements, strict=True):
                    changed[position] = replacement
                if _replay(changed[0], changed[1:]) != original:
                    return float(edit_count)
    return float(len(values))


def is_quarantined(scene: dict) -> bool:
    return (
        scene["start"] not in SHAPES
        or len(scene["challengers"]) != 4
        or any(item not in SHAPES for item in scene["challengers"])
        or sorted(scene["card_slots"]) != [0, 1, 2, 3]
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["distractors"] = list(reversed(scene["distractors"]))
    return [("reverse_unordered_distractor_list", twin)]
