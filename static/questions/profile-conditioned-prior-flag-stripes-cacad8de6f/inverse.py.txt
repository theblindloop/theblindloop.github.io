"""Pixel-only inverse arm: count the striped-flag bands from the raster alone.

This module receives only the rendered PNG and never imports renderer, prompts,
generate, verify, or analytic-gold code, and it accepts no latent scene data.
The flag field is painted as ``n`` alternating red/white horizontal stripes that
fill the canvas, so the number of stripes equals the number of contiguous
color runs encountered down a vertical column.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def decision_from_image(image: Image.Image) -> str:
    """Return the counted number of stripes as a string."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    height, width, _ = arr.shape
    if height < 2:
        return "0"
    column = arr[:, width // 2, :]
    # A run ends whenever the color changes between adjacent rows. Each stripe
    # is a solid, uniform band, so the number of color changes plus one equals
    # the number of stripes.
    changes = int(np.count_nonzero(np.any(column[1:] != column[:-1], axis=1)))
    return str(changes + 1)
