"""Prior-conflict striped-flag world.

A flag field is painted with ``n`` alternating red/white horizontal stripes
that fill the whole canvas. A familiar canonical prior asserts that a US-style
striped flag has THIRTEEN stripes; the counterfactual renders an explicit
non-canonical stripe count and the decision must be read from the visible color
runs alone, never from the prior. The pixel oracle counts the horizontal color
bands present in the raster.

Latent alias mode is ``not_applicable``: the stripe count directly sets the
number of visible color bands, and the top-stripe color and stripe heights are
randomized while label-preserving, so two scenes with the same raster must have
the same band count; the rendering is injective in the label field and there is
no render-invariant alias.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = 512
MIN_BAND = 8.0

# Supported off-quarantine stripe counts. The canonical prior is 13; 11 and 15
# are counterfactuals that contradict it. All are odd counts that alternate
# cleanly into readable red/white bands at full-canvas height.
SUPPORTED_N = (13, 11, 15)

RED = (180, 30, 30)
WHITE = (245, 245, 245)
_STRIPE_COLORS = (RED, WHITE)


def sample_scene(seed: int) -> dict:
    """Deterministic scene dict from an integer seed."""
    rng = random.Random(seed)
    n = SUPPORTED_N[rng.randrange(len(SUPPORTED_N))]
    # Top stripe color is randomized independently of n so neither stripe color
    # nor the red/white proportion is a monotonic cue for the count
    # (label-preserving cheap-cue counterbalance).
    top_color = rng.randrange(2)
    return {
        "n": n,
        "top_stripe": top_color,
        "canvas": CANVAS,
    }


def _band_bounds(n: int) -> list[tuple[int, int]]:
    """Return integer [top, bottom) row bounds for each of n equal bands."""
    height = CANVAS
    bounds = []
    for i in range(n):
        top = int(height * i / n)
        bottom = int(height * (i + 1) / n)
        bounds.append((top, bottom))
    return bounds


def render(scene: dict) -> Image.Image:
    """Render the striped flag field to an RGB image."""
    n = int(scene["n"])
    top_color = int(scene["top_stripe"])
    img = Image.new("RGB", (int(scene["canvas"]), int(scene["canvas"])), WHITE)
    draw = ImageDraw.Draw(img)
    for i, (top, bottom) in enumerate(_band_bounds(n)):
        color = _STRIPE_COLORS[(top_color + i) % 2]
        draw.rectangle([0, top, CANVAS - 1, bottom - 1], fill=color)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision = the visible number of stripes, as a string."""
    return str(scene["n"])


def margin(scene: dict) -> float:
    """Verifiability margin = the pixel height of the thinnest band."""
    return CANVAS / float(scene["n"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine stripe counts whose bands are ambiguous at pixel scale."""
    if int(scene["n"]) not in SUPPORTED_N:
        return True
    return margin(scene) < MIN_BAND


def latent_symmetries(scene: dict):
    """not_applicable: empty, the label is injectively reflected in the raster."""
    return []
