"""Independent image-only inverse for composed frame transfer."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (1000, 700)
N = 5
KEY_CELL, KEY_Y = 28, 66
KEY_X = (42, 238, 548, 744)
STAMP_X, STAMP_Y, STAMP_CELL = 420, 260, 32
CANDIDATE_X = (54, 425, 796)
CANDIDATE_Y, CANDIDATE_CELL = 490, 30
INK = (29, 34, 40)
LANDMARK_RGB = {"red": (214, 65, 70), "blue": (35, 113, 190), "green": (39, 151, 94)}
TAG_RGB = {"amber": (226, 151, 28), "cyan": (20, 157, 181), "violet": (137, 78, 191)}
ALLOWED = {(255, 255, 255), (247, 249, 251), (132, 141, 150), INK, (71, 79, 91), *LANDMARK_RGB.values(), *TAG_RGB.values()}
TRANSFORMS = ("identity", "rot90", "rot180", "rot270", "flip_lr", "flip_ud", "transpose", "anti")


class OracleError(ValueError):
    pass


def _transform(point: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = point
    if name == "identity": return r, c
    if name == "rot90": return c, N - 1 - r
    if name == "rot180": return N - 1 - r, N - 1 - c
    if name == "rot270": return N - 1 - c, r
    if name == "flip_lr": return r, N - 1 - c
    if name == "flip_ud": return N - 1 - r, c
    if name == "transpose": return c, r
    if name == "anti": return N - 1 - c, N - 1 - r
    raise OracleError("unknown transform")


def _apply(points: set[tuple[int, int]], name: str) -> set[tuple[int, int]]:
    return {_transform(point, name) for point in points}


def _read_landmarks(arr: np.ndarray, x: int) -> dict[str, tuple[int, int]]:
    found: dict[str, tuple[int, int]] = {}
    for r in range(N):
        for c in range(N):
            cx, cy = x + c * KEY_CELL + KEY_CELL // 2, KEY_Y + r * KEY_CELL + KEY_CELL // 2
            pixel = tuple(int(v) for v in arr[cy, cx])
            names = [name for name, rgb in LANDMARK_RGB.items() if pixel == rgb]
            if names:
                if names[0] in found:
                    raise OracleError("duplicate landmark color")
                found[names[0]] = (r, c)
            elif pixel != (247, 249, 251):
                raise OracleError("damaged landmark cell center")
    if set(found) != set(LANDMARK_RGB):
        raise OracleError("missing colored landmark")
    return found


def _read_marks(arr: np.ndarray, x: int, y: int, cell: int) -> set[tuple[int, int]]:
    found = set()
    for r in range(N):
        for c in range(N):
            cx, cy = x + c * cell + cell // 2, y + r * cell + cell // 2
            patch = arr[cy - 4:cy + 5, cx - 4:cx + 5]
            count = int(np.all(patch == INK, axis=2).sum())
            if count >= 60:
                found.add((r, c))
            elif count != 0:
                raise OracleError("damaged black cell mark")
    return found


def _infer(source: dict[str, tuple[int, int]], target: dict[str, tuple[int, int]]) -> str:
    matches = [name for name in TRANSFORMS if all(_transform(source[color], name) == target[color] for color in LANDMARK_RGB)]
    if len(matches) != 1:
        raise OracleError("key does not identify one transformation")
    return matches[0]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    if rgb.size != CANVAS:
        raise OracleError("wrong canvas size")
    colors = rgb.getcolors(maxcolors=64)
    if colors is None or not {tuple(color) for _, color in colors} <= ALLOWED:
        raise OracleError("unknown raster colors")
    arr = np.asarray(rgb)
    keys = [_read_landmarks(arr, x) for x in KEY_X]
    first = _infer(keys[0], keys[1])
    second = _infer(keys[2], keys[3])
    stamp = _read_marks(arr, STAMP_X, STAMP_Y, STAMP_CELL)
    if len(stamp) != 5:
        raise OracleError("stamp must contain five marks")
    target = _apply(_apply(stamp, first), second)

    matches = []
    seen_colors = set()
    for x in CANDIDATE_X:
        border = tuple(int(v) for v in arr[CANDIDATE_Y + 2, x + 2])
        names = [name for name, value in TAG_RGB.items() if border == value]
        if len(names) != 1 or names[0] in seen_colors:
            raise OracleError("candidate border colors invalid")
        seen_colors.add(names[0])
        pattern = _read_marks(arr, x, CANDIDATE_Y, CANDIDATE_CELL)
        if len(pattern) != len(stamp):
            raise OracleError("candidate cardinality differs from stamp")
        if pattern == target:
            matches.append(names[0])
    if seen_colors != set(TAG_RGB) or len(matches) != 1:
        raise OracleError("no unique matching candidate")
    return matches[0]
