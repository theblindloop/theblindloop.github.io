"""Deterministic matrices with exactly one solution-set backbone row."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 960
HEIGHT = 700
N = 8
GRID_X = 330
GRID_Y = 132
CELL = 62
ROW_BADGE_X = 269

BG = (244, 241, 233)
PAPER = (255, 255, 253)
INK = (35, 43, 54)
GRID = (166, 176, 186)
DOT = (33, 105, 164)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _permutation_with_value_at(rng: random.Random, value: int, position: int) -> list[int]:
    rest = [item for item in range(N) if item != value]
    rng.shuffle(rest)
    rest.insert(position, value)
    return rest


def _perfect_matchings(allowed: list[list[bool]]) -> list[tuple[int, ...]]:
    return [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(allowed[row][assignment[row]] for row in range(N))
    ]


def _invariant_row(allowed: list[list[bool]]) -> int:
    matches = _perfect_matchings(allowed)
    if len(matches) < 2:
        raise ValueError("visible matrix must have several complete assignments")
    invariant_rows = [
        row for row in range(N)
        if len({assignment[row] for assignment in matches}) == 1
    ]
    if len(invariant_rows) != 1:
        raise ValueError("matrix must have exactly one invariant row")
    return invariant_rows[0]


def sample_scene(seed: int) -> dict:
    """Construct two ambiguous matching blocks separated by one backbone edge.

    Base row zero can locally use column zero or every column in a tight Hall
    block. The tight block has exactly as many neighbors as rows, so stealing
    any of those columns makes completion impossible. Its internal assignment
    and the outer block's assignment both remain genuinely ambiguous.
    """

    rng = random.Random(seed ^ 0xA6B19B4C)
    tight_size = 3 + (seed & 1)
    outer_start = tight_size + 1
    base = [[False] * N for _ in range(N)]

    # The backbone row's locally plausible choices: its invariant edge plus the
    # tight block. None of the tight choices can occur in a full assignment.
    for column in range(tight_size + 1):
        base[0][column] = True

    # A dense ambiguous Hall-tight block on rows/columns 1..tight_size.
    # For size three it is complete; for size four it omits the diagonal.
    for row in range(1, tight_size + 1):
        for column in range(1, tight_size + 1):
            if tight_size == 3 or row != column:
                base[row][column] = True

    # A second ambiguous block on the remaining rows and columns. Each row can
    # use every outer column except its aligned one.
    outer_columns = list(range(outer_start, N))
    for row in outer_columns:
        for column in outer_columns:
            if row != column:
                base[row][column] = True

    # These conspicuous local options are globally unusable because column 0
    # and the Hall-tight columns are exhausted in every complete assignment.
    base[outer_start][0] = True
    for row in outer_columns:
        base[row][rng.randrange(0, tight_size + 1)] = True

    backbone_visual_row = (seed * 3 + 2) % N
    backbone_visual_col = (seed * 5 + 1) % N
    row_order = _permutation_with_value_at(rng, 0, backbone_visual_row)
    col_order = _permutation_with_value_at(rng, 0, backbone_visual_col)
    allowed = [
        [bool(base[row_order[row]][col_order[column]]) for column in range(N)]
        for row in range(N)
    ]
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "allowed": allowed,
    }
    _invariant_row(allowed)
    return scene


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    allowed = scene["allowed"]
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((58, 24, width - 58, height - 25), radius=24, fill=PAPER, outline=INK, width=3)
    draw.text((width // 2, 53), "WHICH ROW NEVER CHANGES?", font=_font(27, True), fill=INK, anchor="mm")
    draw.text((width // 2, 89), "DOT = ALLOWED   •   USE EACH COLUMN ONCE", font=_font(18, True), fill=(71, 82, 94), anchor="mm")

    for column in range(N):
        cx = GRID_X + column * CELL + CELL // 2
        draw.ellipse((cx - 21, GRID_Y - 44, cx + 21, GRID_Y - 2), fill=(245, 196, 80), outline=INK, width=2)
        draw.text((cx, GRID_Y - 23), chr(ord("A") + column), font=_font(23, True), fill=INK, anchor="mm")

    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        draw.ellipse((ROW_BADGE_X - 21, cy - 21, ROW_BADGE_X + 21, cy + 21), fill=(239, 242, 245), outline=INK, width=2)
        draw.text((ROW_BADGE_X, cy), str(row + 1), font=_font(21, True), fill=INK, anchor="mm")
        for column in range(N):
            x0 = GRID_X + column * CELL
            y0 = GRID_Y + row * CELL
            draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), outline=GRID, width=2)
            if bool(allowed[row][column]):
                cx = x0 + CELL // 2
                draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=DOT, outline=INK, width=2)
                draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=(226, 240, 249))

    draw.text((width // 2, 657), "Several full assignments exist. One row keeps its partner.", font=_font(17), fill=(71, 82, 94), anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    return f"row-{_invariant_row(scene['allowed']) + 1}"


def margin(scene: dict) -> float:
    """Count locally allowed alternatives excluded for the invariant row."""

    invariant_row = _invariant_row(scene["allowed"])
    return float(sum(bool(value) for value in scene["allowed"][invariant_row]) - 1)


def is_quarantined(scene: dict) -> bool:
    try:
        allowed = scene["allowed"]
        matches = _perfect_matchings(allowed)
        partner_counts = [len({assignment[row] for assignment in matches}) for row in range(N)]
        invariant_rows = [row for row, count in enumerate(partner_counts) if count == 1]
        row_degrees = [sum(bool(value) for value in row) for row in allowed]
        column_degrees = [sum(bool(allowed[row][column]) for row in range(N)) for column in range(N)]
        return not (
            len(matches) >= 2
            and len(invariant_rows) == 1
            and sum(bool(value) for value in allowed[invariant_rows[0]]) >= 4
            and min(partner_counts[row] for row in range(N) if row != invariant_rows[0]) >= 2
            and min(row_degrees) >= 2
            and min(column_degrees) >= 2
        )
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
