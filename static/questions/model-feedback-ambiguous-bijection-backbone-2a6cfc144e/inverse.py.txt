"""Independent fixed-layout PNG decoder and solution-set intersection solver."""

from __future__ import annotations

import itertools

from PIL import Image

N = 8
GRID_X = 330
GRID_Y = 132
CELL = 62


def _is_blue(pixel: tuple[int, int, int]) -> bool:
    red, green, blue = pixel
    return blue >= 120 and blue >= red + 35 and blue >= green + 20


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (960, 700):
        raise ValueError("unexpected canvas size")
    pixels = raster.load()

    allowed: list[list[bool]] = []
    for row in range(N):
        decoded_row = []
        cy = GRID_Y + row * CELL + CELL // 2
        for column in range(N):
            cx = GRID_X + column * CELL + CELL // 2
            blue = sum(
                _is_blue(pixels[x, y])
                for y in range(cy - 12, cy + 13)
                for x in range(cx - 12, cx + 13)
            )
            decoded_row.append(blue > 260)
        allowed.append(decoded_row)

    assignments = [
        assignment
        for assignment in itertools.permutations(range(N))
        if all(allowed[row][assignment[row]] for row in range(N))
    ]
    if len(assignments) < 2:
        raise ValueError("visible matrix does not have several complete assignments")
    invariant_rows = [
        row for row in range(N)
        if len({assignment[row] for assignment in assignments}) == 1
    ]
    if len(invariant_rows) != 1:
        raise ValueError("visible assignments do not have exactly one invariant row")
    return f"row-{invariant_rows[0] + 1}"
