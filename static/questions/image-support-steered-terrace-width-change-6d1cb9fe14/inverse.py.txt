"""Independent PNG-only inverse arm."""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image


def _colorfulness(rgb: tuple[int, int, int]) -> int:
    return max(rgb) - min(rgb)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    colors = Counter(map(tuple, array.reshape(-1, 3).tolist()))
    choices = [(count, color) for color, count in colors.items() if count > 1200 and _colorfulness(color) > 45]
    if not choices:
        return "abstain"
    _, color = max(choices)
    mask = np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 1200:
        return "abstain"
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    local = mask[y0:y1+1, x0:x1+1]
    row_widths = local.sum(axis=1)
    if len(row_widths) < 45:
        return "abstain"
    # Each tier is a long constant-width plateau. Ignore transition-corrupted or
    # ablated rows by selecting the three most frequent substantial row widths.
    frequencies = Counter(int(v) for v in row_widths if v >= 35)
    plateaus = sorted(((count, width) for width, count in frequencies.items() if count >= 10), reverse=True)
    if len(plateaus) != 3:
        return "abstain"
    valid_widths = {width for _, width in plateaus}
    runs = []
    last = None
    for width in map(int, row_widths):
        value = width if width in valid_widths else None
        if value != last:
            if value is not None:
                runs.append(value)
            last = value
    if len(runs) != 3 or len(set(runs)) != 3:
        return "abstain"
    a, b = abs(runs[0]-runs[1]), abs(runs[1]-runs[2])
    if abs(a-b) < 8:
        return "abstain"
    return "top-to-middle" if a > b else "middle-to-bottom"
