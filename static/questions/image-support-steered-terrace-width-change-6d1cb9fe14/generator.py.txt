"""Deterministic latent sampler, renderer, and analytic arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw


PALETTES = ((18, 136, 156), (35, 146, 108), (51, 111, 181), (132, 78, 181))


def _decision(widths: list[int]) -> str:
    a = abs(int(widths[0]) - int(widths[1]))
    b = abs(int(widths[1]) - int(widths[2]))
    return "top-to-middle" if a > b else "middle-to-bottom"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = "top-to-middle" if int(seed) % 2 == 0 else "middle-to-bottom"
    for _ in range(500):
        widths = [2 * rng.randint(25, 49) + 1 for _ in range(3)]
        if len(set(widths)) < 3:
            continue
        delta_a = abs(widths[0] - widths[1])
        delta_b = abs(widths[1] - widths[2])
        if abs(delta_a - delta_b) < 12:
            continue
        if _decision(widths) == wanted:
            break
    else:
        raise RuntimeError("could not sample separated tier widths")
    tier_height = rng.randint(25, 33)
    center_x = rng.choice((192, 256, 320)) + rng.randint(-7, 7)
    top_y = rng.choice((180, 224, 268)) + rng.randint(-7, 7)
    distractors = []
    for quadrant in rng.sample(range(4), 3):
        xlo, xhi = ((34, 112) if quadrant % 2 == 0 else (400, 478))
        ylo, yhi = ((36, 114) if quadrant < 2 else (398, 476))
        distractors.append({
            "x": rng.randint(xlo, xhi), "y": rng.randint(ylo, yhi),
            "radius": rng.randint(5, 11), "shade": rng.randint(196, 225),
        })
    return {
        "canvas": 512, "center_x": center_x, "top_y": top_y,
        "tier_height": tier_height, "widths": widths,
        "specimen_rgb": list(PALETTES[rng.randrange(len(PALETTES))]),
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas"])
    image = Image.new("RGB", (canvas, canvas), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    for item in scene["distractors"]:
        x, y, r, shade = (int(item[k]) for k in ("x", "y", "radius", "shade"))
        draw.ellipse((x-r, y-r, x+r, y+r), fill=(shade, shade, shade))
    cx, top, height = int(scene["center_x"]), int(scene["top_y"]), int(scene["tier_height"])
    color = tuple(int(v) for v in scene["specimen_rgb"])
    for index, width in enumerate(scene["widths"]):
        half = int(width) // 2
        y0 = top + index * height
        y1 = top + (index + 1) * height - 1
        draw.rectangle((cx-half, y0, cx+half, y1), fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    return _decision(scene["widths"])


def margin(scene: dict) -> float:
    w = [int(value) for value in scene["widths"]]
    return float(abs(abs(w[0]-w[1]) - abs(w[1]-w[2])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_nonoverlapping_distractor_order", twin)]
