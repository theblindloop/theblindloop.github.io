from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

PALETTE = {
    "red": (216, 69, 64),
    "blue": (50, 113, 210),
    "yellow": (226, 173, 35),
}
BACKGROUND = (250, 248, 242)
INK = (32, 36, 42)
TEAL = (16, 151, 153)
SCREEN = (213, 217, 223)
QUARANTINE_BELOW = 0.40


def _random_permutation(rng: random.Random) -> list[int]:
    values = [0, 1, 2]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    spacing = rng.choice([82, 86, 90])
    center = 180 + rng.randint(-6, 6)
    lanes = [center - spacing, center, center + spacing]
    screens = [218 + rng.randint(-5, 5), 400 + rng.randint(-5, 5)]
    permutations = [_random_permutation(rng), _random_permutation(rng)]
    start_lane = rng.randrange(3)
    terminal_colors = ["red", "blue", "yellow"]
    rng.shuffle(terminal_colors)
    draw_order = [0, 1, 2]
    rng.shuffle(draw_order)
    return {
        "lanes": lanes,
        "screens": screens,
        "permutations": permutations,
        "start_lane": start_lane,
        "terminal_colors": terminal_colors,
        "canvas": {"width": 600, "height": 360},
        "style": {
            "background": list(BACKGROUND),
            "ink": list(INK),
            "teal": list(TEAL),
            "screen": list(SCREEN),
            "line_width": 7,
            "screen_half_width": 30,
            "transition_half_width": 52,
        },
        "draw_order": draw_order,
    }


def _compose(permutations: list[list[int]], start: int) -> int:
    lane = start
    for permutation in permutations:
        lane = permutation[lane]
    return lane


def analytic_gold(scene: dict) -> str:
    terminal_lane = _compose(scene["permutations"], int(scene["start_lane"]))
    return str(scene["terminal_colors"][terminal_lane])


def _screen_assignment_gap(lanes: list[float], permutation: list[int], half: float) -> float:
    signatures = []
    for source, target in enumerate(permutation):
        midpoint = (lanes[source] + lanes[target]) / 2.0
        slope = (lanes[target] - lanes[source]) / (2.0 * half)
        signatures.append((midpoint, slope))
    costs = []
    for assignment in itertools.permutations(range(3)):
        cost = 0.0
        for source, target_signature in enumerate(assignment):
            a = signatures[source]
            b = signatures[target_signature]
            cost += abs(a[0] - b[0]) / 24.0 + 3.0 * abs(a[1] - b[1])
        costs.append(cost)
    costs.sort()
    return costs[1] - costs[0]


def margin(scene: dict) -> float:
    half = float(scene["style"]["transition_half_width"])
    return round(
        min(
            _screen_assignment_gap(scene["lanes"], permutation, half)
            for permutation in scene["permutations"]
        ),
        6,
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_BELOW


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {**scene, "draw_order": list(reversed(scene["draw_order"]))}
    return [("reverse_same_ink_path_painter_order", twin)]


def _scaled_points(points: list[tuple[float, float]], scale: int) -> list[tuple[int, int]]:
    return [(round(x * scale), round(y * scale)) for x, y in points]


def render(scene: dict) -> Image.Image:
    width = int(scene["canvas"]["width"])
    height = int(scene["canvas"]["height"])
    style = scene["style"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), tuple(style["background"]))
    draw = ImageDraw.Draw(image)
    lanes = [float(y) for y in scene["lanes"]]
    c1, c2 = [float(x) for x in scene["screens"]]
    transition_half = float(style["transition_half_width"])
    p1, p2 = scene["permutations"]
    for source in scene["draw_order"]:
        middle = int(p1[source])
        terminal = int(p2[middle])
        points = [
            (46, lanes[source]),
            (c1 - transition_half, lanes[source]),
            (c1 + transition_half, lanes[middle]),
            (c2 - transition_half, lanes[middle]),
            (c2 + transition_half, lanes[terminal]),
            (554, lanes[terminal]),
        ]
        draw.line(
            _scaled_points(points, scale),
            fill=tuple(style["ink"]),
            width=int(style["line_width"]) * scale,
            joint="curve",
        )

    screen_half = int(style["screen_half_width"])
    for center_x in scene["screens"]:
        box = [
            round((center_x - screen_half) * scale),
            38 * scale,
            round((center_x + screen_half) * scale),
            (height - 38) * scale,
        ]
        draw.rounded_rectangle(box, radius=7 * scale, fill=tuple(style["screen"]))

    marker_y = lanes[int(scene["start_lane"])]
    draw.ellipse(
        _scaled_points([(22, marker_y - 12), (46, marker_y + 12)], scale),
        fill=tuple(style["background"]),
        outline=tuple(style["teal"]),
        width=6 * scale,
    )
    for lane in scene["draw_order"]:
        y = lanes[lane]
        color = PALETTE[str(scene["terminal_colors"][lane])]
        draw.ellipse(
            _scaled_points([(554, y - 14), (582, y + 14)], scale),
            fill=color,
            outline=tuple(style["ink"]),
            width=3 * scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)
