"""Independent image-only inverse for the Euler-characteristic world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PANEL_BOXES = ((24, 52, 304, 376), (336, 52, 616, 376))


def _panel_mask(image: Image.Image, box: tuple[int, int, int, int]) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    x0, y0, x1, y1 = box
    crop = rgb[y0 + 45 : y1 - 8, x0 + 8 : x1 - 8]
    spread = crop.max(axis=2) - crop.min(axis=2)
    # Frames, labels, and the warm background are achromatic; colored ink is not.
    return (spread > 48) & (crop.max(axis=2) > 95)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, bool]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=np.bool_)
    found: list[tuple[int, int, int, bool]] = []
    for row in range(height):
        for col in range(width):
            if not mask[row, col] or seen[row, col]:
                continue
            queue = deque([(row, col)])
            seen[row, col] = True
            area = 0
            sum_row = 0
            sum_col = 0
            touches_edge = False
            while queue:
                current_row, current_col = queue.popleft()
                area += 1
                sum_row += current_row
                sum_col += current_col
                if (
                    current_row == 0
                    or current_col == 0
                    or current_row == height - 1
                    or current_col == width - 1
                ):
                    touches_edge = True
                for next_row, next_col in (
                    (current_row - 1, current_col),
                    (current_row + 1, current_col),
                    (current_row, current_col - 1),
                    (current_row, current_col + 1),
                ):
                    if (
                        0 <= next_row < height
                        and 0 <= next_col < width
                        and mask[next_row, next_col]
                        and not seen[next_row, next_col]
                    ):
                        seen[next_row, next_col] = True
                        queue.append((next_row, next_col))
            found.append((area, sum_row, sum_col, touches_edge))
    return found


def _euler_from_mask(mask: np.ndarray) -> int:
    colored = [item for item in _components(mask) if item[0] >= 120]
    if not 2 <= len(colored) <= 4:
        raise ValueError(f"expected 2 to 4 colored components, found {len(colored)}")
    background_components = _components(~mask)
    holes = sum(1 for area, _, _, touches_edge in background_components if area >= 80 and not touches_edge)
    return len(colored) - holes


def decision_from_image(image: Image.Image) -> str:
    """Return A, B, or tie using only the final RGB pixels."""

    if image.width < 617 or image.height < 377:
        raise ValueError("image is smaller than the declared panel layout")
    characteristic_a = _euler_from_mask(_panel_mask(image, PANEL_BOXES[0]))
    characteristic_b = _euler_from_mask(_panel_mask(image, PANEL_BOXES[1]))
    if characteristic_a > characteristic_b:
        return "A"
    if characteristic_b > characteristic_a:
        return "B"
    return "tie"
