"""Deterministic raster world for a paired Euler-characteristic comparison."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "euler-characteristic-0.1.0"
WIDTH, HEIGHT = 640, 400
PANEL_BOXES = ((24, 52, 304, 376), (336, 52, 616, 376))
BACKGROUND = (249, 249, 247)
PANEL_BACKGROUND = (253, 253, 251)
FRAME = (86, 89, 96)
TEXT = (28, 31, 36)
PALETTE = (
    (35, 112, 194),
    (207, 72, 59),
    (38, 151, 100),
    (145, 75, 180),
    (224, 132, 28),
)


def _panel_slots(panel_index: int) -> list[tuple[float, float]]:
    x0 = PANEL_BOXES[panel_index][0]
    return [
        (x0 + 72.0, 126.0),
        (x0 + 206.0, 126.0),
        (x0 + 72.0, 264.0),
        (x0 + 206.0, 264.0),
    ]


def _valid_component_count(chi: int) -> list[int]:
    return [c for c in range(2, 5) if 0 <= c - chi <= c and c - chi <= 4]


def _make_panel(rng: random.Random, chi: int, panel_index: int) -> tuple[list[dict], list[dict]]:
    choices = _valid_component_count(chi)
    if not choices:
        raise ValueError(f"unsupported characteristic {chi}")
    component_count = rng.choice(choices)
    hole_count = component_count - chi
    slots = _panel_slots(panel_index)
    rng.shuffle(slots)
    outer: list[dict] = []
    for cx, cy in slots[:component_count]:
        outer.append(
            {
                "x": round(cx + rng.uniform(-4.0, 4.0), 3),
                "y": round(cy + rng.uniform(-4.0, 4.0), 3),
                "rx": round(rng.uniform(28.0, 35.0), 3),
                "ry": round(rng.uniform(27.0, 35.0), 3),
                "phase": round(rng.uniform(0.0, math.tau), 5),
            }
        )
    hole_indices = sorted(rng.sample(range(component_count), hole_count))
    holes: list[dict] = []
    for index in hole_indices:
        component = outer[index]
        radius = min(component["rx"], component["ry"]) * rng.uniform(0.28, 0.38)
        holes.append(
            {
                "x": component["x"],
                "y": component["y"],
                "rx": round(radius, 3),
                "ry": round(radius * rng.uniform(0.88, 1.08), 3),
            }
        )
    return outer, holes


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed is the only source of randomness."""

    rng = random.Random(int(seed))
    mode = int(seed) % 9
    if mode == 0:
        chi_a, chi_b = 2, 2
    elif mode == 1:
        chi_a, chi_b = (2, 1) if int(seed) % 2 else (1, 2)
    else:
        delta = rng.choice((2, 3, 4))
        if int(seed) % 2:
            chi_a = rng.randint(delta, 4)
            chi_b = chi_a - delta
        else:
            chi_b = rng.randint(delta, 4)
            chi_a = chi_b - delta
    outer_a, holes_a = _make_panel(rng, chi_a, 0)
    outer_b, holes_b = _make_panel(rng, chi_b, 1)
    color_a = rng.choice(PALETTE)
    remaining = [color for color in PALETTE if color != color_a]
    color_b = rng.choice(remaining)
    return {
        "outer_a": outer_a,
        "holes_a": holes_a,
        "outer_b": outer_b,
        "holes_b": holes_b,
        "color_a": list(color_a),
        "color_b": list(color_b),
        "canvas_width": WIDTH,
        "canvas_height": HEIGHT,
    }


def _blob_points(component: dict, scale: int) -> list[tuple[float, float]]:
    points = []
    for step in range(40):
        angle = math.tau * step / 40.0
        wobble = 1.0 + 0.065 * math.sin(3.0 * angle + component["phase"])
        points.append(
            (
                (component["x"] + component["rx"] * wobble * math.cos(angle)) * scale,
                (component["y"] + component["ry"] * wobble * math.sin(angle)) * scale,
            )
        )
    return points


def _ellipse_box(item: dict, scale: int) -> tuple[float, float, float, float]:
    return (
        (item["x"] - item["rx"]) * scale,
        (item["y"] - item["ry"]) * scale,
        (item["x"] + item["rx"]) * scale,
        (item["y"] + item["ry"]) * scale,
    )


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    outer: Iterable[dict],
    holes: Iterable[dict],
    color: tuple[int, int, int],
    scale: int,
) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle((x0 * scale, y0 * scale, x1 * scale, y1 * scale), fill=PANEL_BACKGROUND)
    draw.rectangle(
        (x0 * scale, y0 * scale, x1 * scale, y1 * scale),
        outline=FRAME,
        width=2 * scale,
    )
    for component in outer:
        draw.polygon(_blob_points(component, scale), fill=color)
    for hole in holes:
        draw.ellipse(_ellipse_box(hole, scale), fill=PANEL_BACKGROUND)


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new(
        "RGB",
        (int(scene["canvas_width"]) * scale, int(scene["canvas_height"]) * scale),
        BACKGROUND,
    )
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, PANEL_BOXES[0], scene["outer_a"], scene["holes_a"], tuple(scene["color_a"]), scale)
    _draw_panel(draw, PANEL_BOXES[1], scene["outer_b"], scene["holes_b"], tuple(scene["color_b"]), scale)
    image = image.resize((int(scene["canvas_width"]), int(scene["canvas_height"])), Image.Resampling.LANCZOS)
    label_draw = ImageDraw.Draw(image)
    label_draw.text((PANEL_BOXES[0][0] + 12, 25), "A", fill=TEXT)
    label_draw.text((PANEL_BOXES[1][0] + 12, 25), "B", fill=TEXT)
    return image


def analytic_gold(scene: dict) -> str:
    chi_a = len(scene["outer_a"]) - len(scene["holes_a"])
    chi_b = len(scene["outer_b"]) - len(scene["holes_b"])
    if chi_a > chi_b:
        return "A"
    if chi_b > chi_a:
        return "B"
    return "tie"


def margin(scene: dict) -> float:
    chi_a = len(scene["outer_a"]) - len(scene["holes_a"])
    chi_b = len(scene["outer_b"]) - len(scene["holes_b"])
    return float(abs(chi_a - chi_b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
