"""Scene-free pixel inverse arm for key_gate_transit.

Receives only a rasterised PNG.  It segments the saturated colours, groups them
into connected blobs, classifies each blob as a gate (a tall post crossing the
corridor) or a pickup (a small round token above the corridor) from its own
drawn extent, orders the blobs by drawn abscissa, and replays the courier's key
bag from left to right.  It imports no renderer, prompt, generator, verifier or
analytic-gold code and never sees a latent scene.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

ORACLE_VERSION = "key_gate_transit-oracle-0.1.0"

_REFERENCE: dict[str, tuple[int, int, int]] = {
    "blue": (40, 80, 200),
    "green": (30, 150, 60),
    "orange": (235, 130, 20),
    "purple": (140, 50, 175),
    "yellow": (240, 205, 30),
}

_MATCH_TOL = 60.0      # RGB distance a pixel may sit from its reference colour
_MIN_AREA = 120        # ignore stray antialiased specks
_TALL = 60             # a blob at least this tall is a gate, not a token


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Label 4-connected True runs; return (area, x0, y0, x1, y1) per blob."""
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    blobs: list[tuple[int, int, int, int, int]] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        area = 0
        x0 = x1 = sx
        y0 = y1 = sy
        while stack:
            cy, cx = stack.pop()
            area += 1
            if cx < x0:
                x0 = cx
            if cx > x1:
                x1 = cx
            if cy < y0:
                y0 = cy
            if cy > y1:
                y1 = cy
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        blobs.append((area, x0, y0, x1, y1))
    return blobs


def _stations(image: Image.Image) -> list[tuple[float, str, str]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.float64)
    found: list[tuple[float, str, str]] = []
    for name, ref in _REFERENCE.items():
        dist = np.sqrt(((arr - np.array(ref, dtype=np.float64)) ** 2).sum(axis=2))
        mask = dist <= _MATCH_TOL
        if not mask.any():
            continue
        for area, x0, y0, x1, y1 in _components(mask):
            if area < _MIN_AREA:
                continue
            height = y1 - y0 + 1
            kind = "gate" if height >= _TALL else "pickup"
            found.append(((x0 + x1) / 2.0, name, kind))
    found.sort(key=lambda s: s[0])
    return found


def decision_from_image(image: Image.Image) -> str:
    bag: Counter[str] = Counter()
    opened = 0
    for _, colour, kind in _stations(image):
        if kind == "pickup":
            bag[colour] += 1
        elif bag[colour] > 0:
            bag[colour] -= 1
            opened += 1
        else:
            return f"{colour}|{opened}"
    return f"none|{opened}"
