"""Independent pixel-only inverse for the ordinal dot echo image."""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass

import numpy as np
from PIL import Image

INK = np.array([27, 54, 82], dtype=np.int32)
TAG_COLORS = {
    "red": np.array([214, 55, 63], dtype=np.int32),
    "green": np.array([38, 153, 91], dtype=np.int32),
    "blue": np.array([49, 102, 190], dtype=np.int32),
}


@dataclass(frozen=True)
class Component:
    area: int
    cx: float
    cy: float


def _components(mask: np.ndarray, minimum: int = 80) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            output.append(Component(len(xs), float(sum(xs) / len(xs)), float(sum(ys) / len(ys))))
    return output


def _mask_near(array: np.ndarray, color: np.ndarray, tolerance: float = 10.0) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _ordinal(group: list[Component]) -> tuple[int, ...]:
    if len(group) != 5:
        raise ValueError("each dot group must contain five components")
    ordered = sorted(group, key=lambda item: item.cx)
    areas = [item.area for item in ordered]
    sorted_areas = sorted(areas)
    if min(b - a for a, b in zip(sorted_areas, sorted_areas[1:])) < 80:
        raise ValueError("dot-area ranking is ambiguous")
    rank = {area: index for index, area in enumerate(sorted_areas)}
    return tuple(rank[area] for area in areas)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dots = _components(_mask_near(array, INK), minimum=120)
    if len(dots) != 20:
        raise ValueError(f"expected 20 navy dots, found {len(dots)}")

    by_x = sorted(dots, key=lambda item: item.cx)
    gaps = [by_x[i + 1].cx - by_x[i].cx for i in range(len(by_x) - 1)]
    split = int(np.argmax(gaps)) + 1
    left, right = by_x[:split], by_x[split:]
    if len(left) != 5 or len(right) != 15 or gaps[split - 1] < 120:
        raise ValueError("could not infer the detached reference and candidate fields")
    reference = _ordinal(left)

    by_y = sorted(right, key=lambda item: item.cy)
    rows = [by_y[start : start + 5] for start in (0, 5, 10)]
    if any(max(item.cy for item in row) - min(item.cy for item in row) > 8 for row in rows):
        raise ValueError("candidate dots do not form three clear rows")
    patterns = [_ordinal(row) for row in rows]
    matches = [index for index, pattern in enumerate(patterns) if pattern == reference]
    if len(matches) != 1:
        raise ValueError("image does not expose one unique ordinal match")

    tags: list[tuple[str, Component]] = []
    for name, color in TAG_COLORS.items():
        comps = _components(_mask_near(array, color), minimum=200)
        if len(comps) != 1:
            raise ValueError(f"expected one {name} tag")
        tags.append((name, comps[0]))
    tags.sort(key=lambda item: item[1].cy)
    if any(abs(tags[i][1].cy - sum(p.cy for p in rows[i]) / 5.0) > 35 for i in range(3)):
        raise ValueError("colored tags are not bound to the candidate rows")
    return tags[matches[0]][0]
