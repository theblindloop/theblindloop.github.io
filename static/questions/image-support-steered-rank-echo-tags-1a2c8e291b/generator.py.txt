"""Deterministic scene sampler and raster renderer for ordinal dot echoes."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 512
BACKGROUND = [250, 248, 242]
INK = [27, 54, 82]
PANEL = (232, 230, 223)
GOLD = (224, 168, 48)
TAG_RGB = {"red": (214, 55, 63), "green": (38, 153, 91), "blue": (49, 102, 190)}


def _different_order(rng: random.Random, forbidden: set[tuple[int, ...]]) -> list[int]:
    while True:
        order = list(range(5))
        rng.shuffle(order)
        key = tuple(order)
        if key not in forbidden and sum(a != b for a, b in zip(order, next(iter(forbidden)))) >= 3:
            return order


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    reference = list(range(5))
    rng.shuffle(reference)
    match_row = int(seed) % 3
    candidates: list[list[int]] = []
    used = {tuple(reference)}
    for row in range(3):
        if row == match_row:
            candidates.append(list(reference))
        else:
            order = _different_order(rng, used)
            used.add(tuple(order))
            candidates.append(order)
    tags = ["red", "green", "blue"]
    rng.shuffle(tags)
    # Rotate the shuffled assignment by the seed so every long consecutive sweep
    # counterbalances answer color as well as matching row.
    shift = int(seed) % 3
    tags = tags[shift:] + tags[:shift]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "ink": list(INK),
        "reference_order": reference,
        "candidate_orders": candidates,
        "tag_colors": tags,
        "reference_base": rng.randint(7, 10),
        "reference_step": 3 + (int(seed) % 2),
        "candidate_bases": [rng.randint(7, 10) for _ in range(3)],
        "candidate_steps": [3 + ((int(seed) + row + 1) % 2) for row in range(3)],
    }


def _radius(base: int, step: int, rank: int) -> int:
    return int(base + step * rank)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    ink = tuple(scene["ink"])

    # Pale cards clarify grouping but carry no answer styling.
    draw.rounded_rectangle((30, 202, 300, 310), radius=18, fill=PANEL)
    for cy in (86, 256, 426):
        draw.rounded_rectangle((486, cy - 50, 752, cy + 50), radius=18, fill=PANEL)

    # Gold diamond is a visible reference marker, distinct from all answer tags.
    draw.polygon([(48, 256), (58, 246), (68, 256), (58, 266)], fill=GOLD)
    for x, rank in zip((70, 120, 170, 220, 270), scene["reference_order"], strict=True):
        radius = _radius(scene["reference_base"], scene["reference_step"], rank)
        draw.ellipse((x - radius, 256 - radius, x + radius, 256 + radius), fill=ink)

    for row, cy in enumerate((86, 256, 426)):
        for x, rank in zip((500, 550, 600, 650, 700), scene["candidate_orders"][row], strict=True):
            radius = _radius(scene["candidate_bases"][row], scene["candidate_steps"][row], rank)
            draw.ellipse((x - radius, cy - radius, x + radius, cy + radius), fill=ink)
        color = TAG_RGB[scene["tag_colors"][row]]
        draw.rounded_rectangle((730, cy - 14, 754, cy + 14), radius=5, fill=color)
    return image


def analytic_gold(scene: dict) -> str:
    reference = tuple(scene["reference_order"])
    matches = [i for i, order in enumerate(scene["candidate_orders"]) if tuple(order) == reference]
    if len(matches) != 1:
        raise ValueError("scene must contain exactly one matching ordinal row")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    return float(min([scene["reference_step"], *scene["candidate_steps"]]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
