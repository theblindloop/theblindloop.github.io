"""Pixel-only inverse arm: tally half-covered ruling cells from the raster.

Receives the final PNG and nothing else.  It recovers the ruling lattice from
the light grey ruling ink, segments the solid navy region, and counts cells
whose interior is more than half navy.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "halfcover_cell_tally-oracle-0.1.0"

_RULE_REF = np.array([206, 206, 206], dtype=np.int16)
_NAVY_REF = np.array([26, 43, 99], dtype=np.int16)
_RULE_TOL = 14
_NAVY_TOL = 34
ABSTAIN = "abstain"


def _color_mask(arr: np.ndarray, ref: np.ndarray, tol: int) -> np.ndarray:
    return (np.abs(arr - ref[None, None, :]) <= tol).all(axis=2)


def _line_positions(counts: np.ndarray) -> list[int]:
    peak = int(counts.max())
    if peak < 16:
        return []
    hot = np.nonzero(counts >= max(6, int(0.35 * peak)))[0]
    if hot.size == 0:
        return []
    groups: list[list[int]] = [[int(hot[0])]]
    for value in hot[1:]:
        if int(value) - groups[-1][-1] <= 1:
            groups[-1].append(int(value))
        else:
            groups.append([int(value)])
    return [int(round(sum(g) / len(g))) for g in groups]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    rule = _color_mask(arr, _RULE_REF, _RULE_TOL)
    navy = _color_mask(arr, _NAVY_REF, _NAVY_TOL)
    if not navy.any() or not rule.any():
        return ABSTAIN

    xs = _line_positions(rule.sum(axis=0))
    ys = _line_positions(rule.sum(axis=1))
    if len(xs) < 4 or len(ys) < 4:
        return ABSTAIN

    gaps = np.diff(np.array(xs + ys, dtype=float))
    spacings = np.concatenate([np.diff(np.array(xs)), np.diff(np.array(ys))]).astype(float)
    if spacings.size == 0 or spacings.min() < 8:
        return ABSTAIN
    if spacings.max() - spacings.min() > 2:
        return ABSTAIN
    del gaps

    tally = 0
    for j in range(len(ys) - 1):
        y0, y1 = ys[j] + 1, ys[j + 1]
        for i in range(len(xs) - 1):
            x0, x1 = xs[i] + 1, xs[i + 1]
            area = (x1 - x0) * (y1 - y0)
            if area <= 0:
                return ABSTAIN
            if 2 * int(navy[y0:y1, x0:x1].sum()) > area:
                tally += 1
    return str(tally)
