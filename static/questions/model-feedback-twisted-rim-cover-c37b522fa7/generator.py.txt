"""Deterministic analytic and visual world for Twisted Rim Cover."""
from __future__ import annotations
from collections import Counter, deque
import copy, random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, N, CELL = 1200, 800, 7, 42
BACKGROUND=(245,242,235); CARD=(255,255,252); INK=(42,47,57); MUTED=(84,91,104)
GRID=(203,209,216); CARD_EDGE=(91,100,114)
DECISIONS=("A","B","C","D"); TERMINALS={"A":(0,0),"B":(0,6),"C":(6,0),"D":(6,6)}; START=(3,3)
PORTAL_CELLS={"P":((1,2),(5,4)),"Q":((1,4),(5,2))}
DIRECTIONS=((-1,0),(0,1),(1,0),(0,-1)); RIM_NAMES=("red","gold","green","blue")
RIM_COLORS={"red":(224,68,75),"gold":(244,177,43),"green":(50,166,101),"blue":(54,122,214)}
PORTAL_COLORS={"P":(25,151,171),"Q":(146,79,190)}
MARK_COLORS={"S":(215,70,65),"A":(247,174,47),"B":(45,119,204),"C":(76,166,88),"D":(150,82,191)}
LAYER_COLORS={0:(251,251,247),1:(31,35,43)}; MODE_COLORS={False:(72,164,103),True:(231,116,66)}
MODE_KEY_X={"P":399,"Q":619}; PANEL_BOXES=((20,181,386,755),(417,181,783,755),(814,181,1180,755))
GRID_OFFSET=(36,137); FONT_PATH="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _font(size): return ImageFont.truetype(FONT_PATH,size)
def _canon(edge): return tuple(sorted((tuple(edge[0]),tuple(edge[1]))))
def _all_edges():
    blocked={cell for cells in PORTAL_CELLS.values() for cell in cells}; out=[]
    for r in range(N):
        for c in range(N):
            if (r,c) in blocked: continue
            for dr,dc in ((0,1),(1,0)):
                nxt=(r+dr,c+dc)
                if 0<=nxt[0]<N and 0<=nxt[1]<N and nxt not in blocked: out.append(((r,c),nxt))
    return out
ALL_EDGES=_all_edges()
def _odd_walls(plans):
    counts=Counter(_canon(edge) for plan in plans for edge in plan)
    return {edge for edge,count in counts.items() if count%2}
def _portal_map(portals):
    links=[]
    for symbol,entries in portals.items():
        sides=[]
        for entry in entries:
            cell=tuple(entry["cell"]); by_tag={}
            for index,tag in enumerate(entry["rims"]):
                dr,dc=DIRECTIONS[index]; by_tag[tag]=(cell[0]+dr,cell[1]+dc)
            sides.append(by_tag)
        if len(sides)==2:
            links.extend((symbol,sides[0][tag],sides[1][tag]) for tag in RIM_NAMES)
    return links
def _reachable(walls,portals=None,seam_modes=None,collapse=False):
    blocked={cell for cells in PORTAL_CELLS.values() for cell in cells}
    cells=[(r,c) for r in range(N) for c in range(N) if (r,c) not in blocked]
    if collapse:
        graph={cell:set() for cell in cells}
        for edge in ALL_EDGES:
            a,b=_canon(edge)
            if (a,b) not in walls: graph[a].add(b); graph[b].add(a)
        if portals:
            for _,a,b in _portal_map(portals): graph[a].add(b); graph[b].add(a)
        start=START
    else:
        graph={(cell,layer):set() for cell in cells for layer in (0,1)}
        for edge in ALL_EDGES:
            a,b=_canon(edge)
            if (a,b) not in walls:
                for layer in (0,1): graph[(a,layer)].add((b,layer)); graph[(b,layer)].add((a,layer))
        if portals:
            for symbol,a,b in _portal_map(portals):
                flip=int(bool(seam_modes[symbol]))
                for layer in (0,1): graph[(a,layer)].add((b,layer^flip)); graph[(b,layer^flip)].add((a,layer))
        start=(START,0)
    seen={start}; queue=deque([start])
    while queue:
        node=queue.popleft()
        for nxt in graph[node]:
            if nxt not in seen: seen.add(nxt); queue.append(nxt)
    return seen
def _labeled_hits(seen,layers):
    return [label for label,cell in TERMINALS.items() if (cell,int(layers[label])) in seen]
def _partition(rng):
    blocked={cell for cells in PORTAL_CELLS.values() for cell in cells}
    owner={START:"S",**{cell:label for label,cell in TERMINALS.items()},(0,3):"X",(6,3):"Y",(3,0):"Z",(3,6):"W"}
    while len(owner)<N*N-len(blocked):
        frontier=[]
        for cell,label in owner.items():
            for dr,dc in DIRECTIONS:
                nxt=(cell[0]+dr,cell[1]+dc)
                if 0<=nxt[0]<N and 0<=nxt[1]<N and nxt not in blocked and nxt not in owner: frontier.append((nxt,label))
        if not frontier: raise RuntimeError("partition stalled")
        nxt,label=rng.choice(frontier); owner[nxt]=label
    return owner
def sample_scene(seed):
    rng=random.Random(int(seed)); target=DECISIONS[int(seed)%4]
    for _ in range(10000):
        owner=_partition(rng); final_walls=[_canon(e) for e in ALL_EDGES if owner[e[0]]!=owner[e[1]]]
        if not 24<=len(final_walls)<=44: continue
        portals={}
        for symbol,cells in PORTAL_CELLS.items():
            entries=[]
            for cell in cells:
                rims=list(RIM_NAMES); rng.shuffle(rims); entries.append({"cell":list(cell),"rims":rims})
            portals[symbol]=entries
        symbols=list(PORTAL_CELLS); rng.shuffle(symbols); modes={symbols[0]:False,symbols[1]:True}; walls=set(final_walls)
        if any(cell in _reachable(walls,collapse=True) for cell in TERMINALS.values()): continue
        seen=_reachable(walls,portals,modes)
        states={label:[layer for layer in (0,1) if (cell,layer) in seen] for label,cell in TERMINALS.items()}
        physical=[label for label,x in states.items() if x]
        if target not in physical or len(physical)<3 or any(len(states[x])>1 for x in physical): continue
        layers={}
        for label in DECISIONS:
            layers[label]=states[label][0] if label==target else (1-states[label][0] if states[label] else rng.randrange(2))
        wrong=_labeled_hits(_reachable(walls,portals,{"P":False,"Q":False}),layers)
        if target in wrong or not wrong: continue
        nonwalls=[_canon(e) for e in ALL_EDGES if e not in walls]; rng.shuffle(final_walls); rng.shuffle(nonwalls)
        plans=[[],[],[]]
        for i,e in enumerate(final_walls): plans[i%3].append(e)
        pairs=[(0,1),(0,2),(1,2)]; rng.shuffle(pairs)
        for i,e in enumerate(nonwalls[:18]):
            for panel in pairs[i%3]: plans[panel].append(e)
        rng.shuffle(plans); encoded=[]
        for plan in plans:
            rng.shuffle(plan); encoded.append([[list(e[1]),list(e[0])] if rng.getrandbits(1) else [list(e[0]),list(e[1])] for e in plan])
        scene={"plans":encoded,"portals":portals,"seam_modes":modes,"terminal_layers":layers}
        if not is_quarantined(scene): return scene
    raise RuntimeError(f"could not sample accepted scene for seed {seed}")
def analytic_gold(scene):
    hits=_labeled_hits(_reachable(_odd_walls(scene["plans"]),scene["portals"],scene["seam_modes"]),scene["terminal_layers"])
    return hits[0] if len(hits)==1 else "abstain"
def margin(scene): return 1.0 if analytic_gold(scene) in DECISIONS else 0.0
def is_quarantined(scene):
    try:
        if len(scene["plans"])!=3 or set(scene["portals"])!=set(PORTAL_CELLS): return True
        if set(scene["seam_modes"])!=set(PORTAL_CELLS) or sorted(scene["seam_modes"].values())!=[False,True]: return True
        if set(scene["terminal_layers"])!=set(DECISIONS) or any(v not in (0,1) for v in scene["terminal_layers"].values()): return True
        occurrences=[_canon(e) for plan in scene["plans"] for e in plan]; counts=Counter(occurrences)
        if any(e not in ALL_EDGES for e in occurrences) or any(v not in (1,2) for v in counts.values()) or sum(v==2 for v in counts.values())<14: return True
        for symbol,entries in scene["portals"].items():
            if len(entries)!=2 or tuple(tuple(e["cell"]) for e in entries)!=PORTAL_CELLS[symbol]: return True
            if any(sorted(e["rims"])!=sorted(RIM_NAMES) for e in entries): return True
        walls=_odd_walls(scene["plans"])
        if any(cell in _reachable(walls,collapse=True) for cell in TERMINALS.values()): return True
        seen=_reachable(walls,scene["portals"],scene["seam_modes"]); hits=_labeled_hits(seen,scene["terminal_layers"])
        physical=[label for label,cell in TERMINALS.items() if any((cell,k) in seen for k in (0,1))]
        wrong=_labeled_hits(_reachable(walls,scene["portals"],{"P":False,"Q":False}),scene["terminal_layers"])
        return len(hits)!=1 or len(physical)<3 or hits[0] in wrong or not wrong
    except Exception: return True
def latent_symmetries(scene):
    x=copy.deepcopy(scene); x["plans"]=[[[e[1],e[0]] for e in plan] for plan in x["plans"]]
    return [("reverse_all_stored_wall_endpoints",x)]
def _wall_line(edge,gx,gy):
    (r1,c1),(r2,c2)=_canon(edge)
    if r1==r2:
        x=gx+max(c1,c2)*CELL; return (x,gy+r1*CELL,x,gy+(r1+1)*CELL)
    y=gy+max(r1,r2)*CELL; return (gx+c1*CELL,y,gx+(c1+1)*CELL,y)
def _draw_portal(draw,cx,cy,symbol,rims):
    draw.rounded_rectangle((cx-18,cy-18,cx+18,cy+18),radius=5,fill=(236,238,241),outline=INK,width=2)
    boxes=((cx-8,cy-18,cx+8,cy-10),(cx+10,cy-8,cx+18,cy+8),(cx-8,cy+10,cx+8,cy+18),(cx-18,cy-8,cx-10,cy+8))
    for tag,box in zip(rims,boxes): draw.rectangle(box,fill=RIM_COLORS[tag],outline=INK,width=1)
    draw.ellipse((cx-8,cy-8,cx+8,cy+8),fill=PORTAL_COLORS[symbol],outline=INK,width=1); draw.text((cx,cy),symbol,fill=(255,255,255),font=_font(11),anchor="mm")
def _draw_terminal(draw,cx,cy,label,layer):
    draw.ellipse((cx-16,cy-16,cx+16,cy+16),fill=MARK_COLORS[label],outline=INK,width=2)
    fill=(255,255,255) if label in {"S","B","D"} else INK; draw.text((cx,cy-5),label,fill=fill,font=_font(14),anchor="mm")
    draw.rectangle((cx-10,cy+5,cx+10,cy+15),fill=LAYER_COLORS[layer],outline=INK,width=1)
    draw.text((cx,cy+10),"I" if layer==0 else "II",fill=INK if layer==0 else (255,255,255),font=_font(8),anchor="mm")
def render(scene):
    image=Image.new("RGB",(WIDTH,HEIGHT),BACKGROUND); draw=ImageDraw.Draw(image)
    draw.text((WIDTH//2,24),"TWISTED RIM COVER",fill=INK,font=_font(29),anchor="mm")
    draw.text((WIDTH//2,57),"once = WALL   •   twice = OPEN   •   same letter + rim color = JOIN",fill=MUTED,font=_font(16),anchor="mm")
    for symbol in ("P","Q"):
        x=MODE_KEY_X[symbol]; mode=bool(scene["seam_modes"][symbol])
        draw.rounded_rectangle((x,84,x+182,151),radius=14,fill=CARD,outline=CARD_EDGE,width=3)
        draw.rectangle((x+12,100,x+48,136),fill=MODE_COLORS[mode],outline=INK,width=2); draw.text((x+30,118),symbol,fill=(255,255,255),font=_font(18),anchor="mm")
        draw.text((x+112,108),"SWAP I ↔ II" if mode else "KEEP I / II",fill=INK,font=_font(15),anchor="mm"); draw.text((x+112,132),"NO TURN",fill=MUTED,font=_font(12),anchor="mm")
    for panel_index,box in enumerate(PANEL_BOXES):
        x0,y0,x1,y1=box; draw.rounded_rectangle(box,radius=18,fill=CARD,outline=CARD_EDGE,width=4)
        draw.text(((x0+x1)//2,y0+35),f"PLAN {panel_index+1}",fill=INK,font=_font(22),anchor="mm")
        gx,gy=x0+GRID_OFFSET[0],y0+GRID_OFFSET[1]
        for k in range(N+1):
            draw.line((gx+k*CELL,gy,gx+k*CELL,gy+N*CELL),fill=GRID,width=2); draw.line((gx,gy+k*CELL,gx+N*CELL,gy+k*CELL),fill=GRID,width=2)
        draw.rectangle((gx,gy,gx+N*CELL,gy+N*CELL),outline=CARD_EDGE,width=4)
        for e in scene["plans"][panel_index]: draw.line(_wall_line(e,gx,gy),fill=INK,width=10)
        for symbol,entries in scene["portals"].items():
            for entry in entries:
                r,c=entry["cell"]; _draw_portal(draw,gx+c*CELL+CELL//2,gy+r*CELL+CELL//2,symbol,entry["rims"])
        marks={"S":(START,0),**{label:(cell,scene["terminal_layers"][label]) for label,cell in TERMINALS.items()}}
        for label,(cell,layer) in marks.items():
            r,c=cell; _draw_terminal(draw,gx+c*CELL+CELL//2,gy+r*CELL+CELL//2,label,int(layer))
        draw.text(((x0+x1)//2,y1-22),"S starts on layer I",fill=MUTED,font=_font(14),anchor="mm")
    return image
