"""Distance-to-reference search world.

The image is split into four quadrants (top-left, top-right, bottom-left,
bottom-right) by two light-grey divider lines that cross at the image centre. A
small mid-grey diamond reference marker sits exactly at the centre. Each
quadrant holds one dark filled disk (a dot). The dot radii from the centre
differ per quadrant. Exactly one dot -- the one drawn closest to the central
reference marker -- is the target; the other three dots are drawn clearly
farther from the marker and at mutually distinct distances. The decision is the
quadrant whose dot is nearest (Euclidean) to the central reference marker.

All rendering is deterministic and the nearest dot is directly recoverable from
the raster by locating the marker and measuring dot distances.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SIZE = 480
CX = SIZE // 2
CY = SIZE // 2

BG = (255, 255, 255)
SEG = (20, 20, 20)
REF = (150, 150, 150)
GRAY = (235, 235, 235)

SECTORS = ("TL", "TR", "BL", "BR")
# Diagonal direction (sign of x, sign of y) away from the centre for each quadrant.
DIRS = {
    "TL": (-1, -1),
    "TR": (1, -1),
    "BL": (-1, 1),
    "BR": (1, 1),
}

DOT_RADIUS = 9
REF_HALF_DIAG = 14
WINNER_RADII = (78, 80, 82)
DISTRACTOR_RADII = (135, 155, 175, 200, 225)
QUARANTINE_THRESH = 40.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    winner_r = int(WINNER_RADII[int(rng.integers(0, len(WINNER_RADII)))])
    distractor_r = [
        int(value)
        for value in rng.choice(DISTRACTOR_RADII, size=3, replace=False)
    ]
    radii: dict[str, int] = {winner_sector: winner_r}
    for sector, r in zip([s for s in SECTORS if s != winner_sector], distractor_r):
        radii[sector] = r

    dots = []
    for sector in SECTORS:
        dx, dy = DIRS[sector]
        cx = CX + dx * radii[sector]
        cy = CY + dy * radii[sector]
        dots.append({"cx": int(cx), "cy": int(cy), "sector": sector, "radius": int(radii[sector])})
    return {"dots": dots, "winner_sector": winner_sector}


def _stamp_dot(arr: np.ndarray, cx: int, cy: int) -> None:
    ys, xs = np.ogrid[:SIZE, :SIZE]
    mask = (xs - cx) ** 2 + (ys - cy) ** 2 <= DOT_RADIUS ** 2
    arr[mask] = SEG


def _stamp_marker(arr: np.ndarray) -> None:
    ys, xs = np.ogrid[:SIZE, :SIZE]
    mask = np.abs(xs - CX) + np.abs(ys - CY) <= REF_HALF_DIAG
    arr[mask] = REF


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG
    arr[:, CX - 1 : CX + 1] = GRAY
    arr[CY - 1 : CY + 1, :] = GRAY
    _stamp_marker(arr)
    for dot in scene["dots"]:
        _stamp_dot(arr, dot["cx"], dot["cy"])
    return Image.fromarray(arr, "RGB")


def _decision(scene: dict) -> tuple[str, float]:
    distances = {}
    for dot in scene["dots"]:
        d = float(np.hypot(dot["cx"] - CX, dot["cy"] - CY))
        distances[dot["sector"]] = d
    ordered = sorted(distances.items(), key=lambda kv: kv[1])
    winner = ordered[0][0]
    margin = float(ordered[1][1] - ordered[0][1])
    return winner, margin


def analytic_gold(scene: dict) -> str:
    winner, _ = _decision(scene)
    return winner


def margin(scene: dict) -> float:
    _, margin_value = _decision(scene)
    return margin_value


def is_quarantined(scene: dict) -> bool:
    _, margin_value = _decision(scene)
    return bool(margin_value < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
