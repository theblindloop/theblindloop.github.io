"""Independent pixel-only inverse arm for proximity_quadrant_search.

Receives only a PIL image. It locates the mid-grey reference marker at the
image centre by thresholding the mid-grey ink, locates each dark dot by
connected-component analysis of the near-black ink, measures the Euclidean
distance from the marker centroid to every dot centroid, and selects the
quadrant whose dot is nearest to the marker.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np


def _dark_dots(arr: np.ndarray) -> list[tuple[float, float]]:
    r = arr[:, :, 0]
    g = arr[:, :, 1]
    b = arr[:, :, 2]
    mask = (r < 70) & (g < 70) & (b < 70)
    ys, xs = np.nonzero(mask)
    if xs.size == 0:
        return []
    labels, count = _label(mask)
    dots = []
    for label in range(1, count + 1):
        sel = labels == label
        if np.count_nonzero(sel) < 20:
            continue
        py, px = np.nonzero(sel)
        dots.append((float(px.mean()), float(py.mean())))
    return dots


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    rows, cols = mask.shape
    labels = np.zeros_like(mask, dtype=np.int32)
    current = 0
    union = list(range(rows * cols + 1))

    def find(a: int) -> int:
        while union[a] != a:
            union[a] = union[union[a]]
            a = union[a]
        return a

    def union_ab(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            union[max(ra, rb)] = min(ra, rb)

    mapping: list[int] = []
    for y in range(rows):
        for x in range(cols):
            if not mask[y, x]:
                labels[y, x] = 0
                continue
            left = labels[y, x - 1] if x > 0 else 0
            up = labels[y - 1, x] if y > 0 else 0
            if left == 0 and up == 0:
                current += 1
                labels[y, x] = current
            elif left == 0:
                labels[y, x] = up
            elif up == 0:
                labels[y, x] = left
            elif left == up:
                labels[y, x] = left
            else:
                labels[y, x] = min(left, up)
                union_ab(left, up)
    final = {}
    for y in range(rows):
        for x in range(cols):
            label = labels[y, x]
            if label == 0:
                continue
            root = find(label)
            if root not in final:
                final[root] = len(final) + 1
            labels[y, x] = final[root]
    return labels, len(final)


def _marker_center(arr: np.ndarray) -> tuple[float, float] | None:
    r = arr[:, :, 0].astype(int)
    g = arr[:, :, 1].astype(int)
    b = arr[:, :, 2].astype(int)
    mid = (110 <= r) & (r <= 190) & (np.abs(r - g) < 25) & (np.abs(g - b) < 25)
    ys, xs = np.nonzero(mid)
    if xs.size == 0:
        return None
    return (float(xs.mean()), float(ys.mean()))


def _sector_of(cx: float, cy: float, x_split: float, y_split: float) -> str:
    if cx < x_split and cy < y_split:
        return "TL"
    if cx >= x_split and cy < y_split:
        return "TR"
    if cx < x_split and cy >= y_split:
        return "BL"
    return "BR"


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"))
    x_split = w / 2.0
    y_split = h / 2.0

    marker = _marker_center(arr)
    dots = _dark_dots(arr)
    if marker is None or not dots:
        return "TL"

    best_sector = "TL"
    best_distance = float("inf")
    for cx, cy in dots:
        distance = np.hypot(cx - marker[0], cy - marker[1])
        if distance < best_distance:
            best_distance = distance
            best_sector = _sector_of(cx, cy, x_split, y_split)
    return best_sector
