"""Renderer and analytic arm for paired fractional checkpoint progress."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = (640, 640)
PALETTES = (
    {
        "background": [250, 249, 246],
        "route": [28, 55, 88],
        "start": [35, 165, 95],
        "finish": [220, 65, 78],
        "checkpoint": [245, 188, 42],
        "tether": [184, 190, 198],
        "divider": [222, 224, 228],
    },
    {
        "background": [252, 250, 247],
        "route": [52, 45, 102],
        "start": [42, 154, 72],
        "finish": [204, 55, 88],
        "checkpoint": [246, 174, 34],
        "tether": [178, 186, 196],
        "divider": [224, 224, 228],
    },
)
QUARANTINE_GAP = 0.12


def _route(rng: random.Random, base_y: int) -> list[list[float]]:
    levels = [base_y - 42, base_y, base_y + 42]
    xs = [170 + rng.randint(-10, 10), 265 + rng.randint(-10, 10),
          360 + rng.randint(-10, 10), 455 + rng.randint(-10, 10)]
    y = rng.choice(levels)
    points: list[list[float]] = [[82.0, float(y)]]
    for x in xs:
        points.append([float(x), float(y)])
        choices = [value for value in levels if value != y]
        y = rng.choice(choices)
        points.append([float(x), float(y)])
    points.append([558.0, float(y)])
    if rng.random() < 0.5:
        points.reverse()
    return points


def _length(route: list[list[float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(route, route[1:]))


def _point_at(route: list[list[float]], fraction: float) -> list[float]:
    target = fraction * _length(route)
    walked = 0.0
    for a, b in zip(route, route[1:]):
        segment = math.dist(a, b)
        if walked + segment >= target:
            t = (target - walked) / segment
            return [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])]
        walked += segment
    return list(route[-1])


def _progress(route: list[list[float]], checkpoint: list[float]) -> float:
    total = _length(route)
    walked = 0.0
    best = (float("inf"), 0.0)
    px, py = checkpoint
    for a, b in zip(route, route[1:]):
        vx, vy = b[0] - a[0], b[1] - a[1]
        squared = vx * vx + vy * vy
        t = max(0.0, min(1.0, ((px - a[0]) * vx + (py - a[1]) * vy) / squared))
        qx, qy = a[0] + t * vx, a[1] + t * vy
        distance = math.hypot(px - qx, py - qy)
        prefix = walked + t * math.sqrt(squared)
        if distance < best[0]:
            best = (distance, prefix)
        walked += math.sqrt(squared)
    if best[0] > 1e-6:
        raise ValueError("checkpoint is not on its route")
    return best[1] / total


def _progresses(scene: dict) -> tuple[float, float]:
    routes = scene["routes"]
    checkpoints = scene["checkpoint_points"]
    return (_progress(routes[0], checkpoints[0]), _progress(routes[1], checkpoints[1]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    routes = [_route(rng, 130), _route(rng, 510)]
    if seed % 11 == 0:
        low = rng.uniform(0.43, 0.47)
        high = low + rng.uniform(0.065, 0.095)
    else:
        low = rng.uniform(0.24, 0.39)
        high = rng.uniform(0.64, 0.78)
    fractions = [high, low] if seed % 2 == 0 else [low, high]
    checkpoints = [_point_at(route, fraction) for route, fraction in zip(routes, fractions)]
    return {
        "routes": routes,
        "checkpoint_points": checkpoints,
        "palette": copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
        "canvas": list(CANVAS),
        "style": {
            "route_width": 13,
            "endpoint_radius": 13,
            "checkpoint_radius": 11,
            "tether_width": 3,
            "checkpoint_sides": [rng.choice([-1, 1]), rng.choice([-1, 1])],
        },
    }


def analytic_gold(scene: dict) -> str:
    upper, lower = _progresses(scene)
    return "upper" if upper > lower else "lower"


def margin(scene: dict) -> float:
    upper, lower = _progresses(scene)
    return abs(upper - lower)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def _distance_to_route(route: list[list[float]], point: tuple[float, float]) -> float:
    px, py = point
    best = float("inf")
    for a, b in zip(route, route[1:]):
        vx, vy = b[0] - a[0], b[1] - a[1]
        squared = vx * vx + vy * vy
        t = max(0.0, min(1.0, ((px - a[0]) * vx + (py - a[1]) * vy) / squared))
        qx, qy = a[0] + t * vx, a[1] + t * vy
        best = min(best, math.hypot(px - qx, py - qy))
    return best


def _checkpoint_marker(route: list[list[float]], point: list[float], side: int) -> tuple[float, float]:
    px, py = point
    best = (float("inf"), 1.0, 0.0)
    for a, b in zip(route, route[1:]):
        vx, vy = b[0] - a[0], b[1] - a[1]
        squared = vx * vx + vy * vy
        t = max(0.0, min(1.0, ((px - a[0]) * vx + (py - a[1]) * vy) / squared))
        qx, qy = a[0] + t * vx, a[1] + t * vy
        distance = math.hypot(px - qx, py - qy)
        if distance < best[0]:
            length = math.sqrt(squared)
            best = (distance, -vy / length, vx / length)
    _, nx, ny = best
    candidates = []
    for sign in (side, -side):
        offset = 31.0 * sign
        marker = (px + offset * nx, py + offset * ny)
        in_bounds = 18 <= marker[0] <= 622 and 24 <= marker[1] <= 616
        candidates.append((_distance_to_route(route, marker) if in_bounds else -1.0, marker))
    return max(candidates, key=lambda item: item[0])[1]


def render(scene: dict) -> Image.Image:
    width, height = map(int, scene["canvas"])
    palette = {key: tuple(value) for key, value in scene["palette"].items()}
    style = scene["style"]
    image = Image.new("RGB", (width, height), palette["background"])
    draw = ImageDraw.Draw(image)
    draw.line([(36, 320), (604, 320)], fill=palette["divider"], width=2)
    marker_centers = []
    for panel, (route, checkpoint, side) in enumerate(
        zip(scene["routes"], scene["checkpoint_points"], style["checkpoint_sides"])
    ):
        marker = _checkpoint_marker(route, checkpoint, int(side))
        marker_centers.append(marker)
        draw.line([tuple(checkpoint), marker], fill=palette["tether"], width=int(style["tether_width"]))
    for route in scene["routes"]:
        draw.line([tuple(point) for point in route], fill=palette["route"],
                  width=int(style["route_width"]), joint="curve")
    endpoint_radius = int(style["endpoint_radius"])
    checkpoint_radius = int(style["checkpoint_radius"])
    for route, marker in zip(scene["routes"], marker_centers):
        for point, color in ((route[0], palette["start"]), (route[-1], palette["finish"])):
            x, y = point
            draw.ellipse((x-endpoint_radius, y-endpoint_radius, x+endpoint_radius, y+endpoint_radius), fill=color)
        x, y = marker
        draw.ellipse((x-checkpoint_radius, y-checkpoint_radius, x+checkpoint_radius, y+checkpoint_radius), fill=palette["checkpoint"])
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    transformed = []
    for route in twin["routes"]:
        a, b = route[0], route[1]
        midpoint = [(a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0]
        transformed.append([route[0], midpoint, *route[1:]])
    twin["routes"] = transformed
    return [("collinear_route_subdivision", twin)]
