"""Independent pixel-only inverse arm for paired fractional route progress."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy-1, cx), (cy+1, cx), (cy, cx-1), (cy, cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _centers(mask: np.ndarray, name: str) -> list[tuple[float, float]]:
    components = _components(mask, 40)
    if len(components) != 2:
        raise OracleAbstention(f"expected two {name} markers, found {len(components)}")
    areas = [len(points) for points in components]
    if any(area < 300 or area > 750 for area in areas):
        raise OracleAbstention(f"damaged {name} marker")
    centers = [(float(points[:, 1].mean()), float(points[:, 0].mean())) for points in components]
    return sorted(centers, key=lambda point: point[1])


def _nearest(points: np.ndarray, center: tuple[float, float]) -> tuple[int, int]:
    x, y = center
    index = int(np.argmin((points[:, 1] - x) ** 2 + (points[:, 0] - y) ** 2))
    return int(points[index, 0]), int(points[index, 1])


def _distance(mask: np.ndarray, source: tuple[int, int], target: tuple[int, int]) -> int:
    height, width = mask.shape
    distances = np.full(mask.shape, -1, dtype=np.int32)
    distances[source] = 0
    queue = deque([source])
    while queue:
        y, x = queue.popleft()
        if (y, x) == target:
            return int(distances[y, x])
        next_distance = int(distances[y, x]) + 1
        for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = next_distance
                queue.append((ny, nx))
    raise OracleAbstention("marker projections are disconnected")


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    route_mask = (red < 90) & (green < 110) & (blue < 145)
    start_mask = (green > 115) & (green > red * 1.35) & (green > blue * 1.15)
    finish_mask = (red > 170) & (red > green * 1.8) & (red > blue * 1.45)
    checkpoint_mask = (red > 215) & (green > 135) & (green < 215) & (blue < 100)

    route_components = _components(route_mask, 500)
    if len(route_components) != 2:
        raise OracleAbstention(f"expected two route islands, found {len(route_components)}")
    route_components.sort(key=lambda points: float(points[:, 0].mean()))
    if any(len(points) < 5000 for points in route_components):
        raise OracleAbstention("route ink is incomplete")
    if any(int(points[:, 1].max() - points[:, 1].min()) < 430 for points in route_components):
        raise OracleAbstention("route does not span its panel")

    starts = _centers(start_mask, "start")
    finishes = _centers(finish_mask, "finish")
    checkpoints = _centers(checkpoint_mask, "checkpoint")
    progresses = []
    for points, start, finish, checkpoint in zip(route_components, starts, finishes, checkpoints):
        component_mask = np.zeros(route_mask.shape, dtype=bool)
        component_mask[points[:, 0], points[:, 1]] = True
        start_pixel = _nearest(points, start)
        finish_pixel = _nearest(points, finish)
        checkpoint_pixel = _nearest(points, checkpoint)
        prefix = _distance(component_mask, start_pixel, checkpoint_pixel)
        suffix = _distance(component_mask, checkpoint_pixel, finish_pixel)
        if prefix < 80 or suffix < 80:
            raise OracleAbstention("checkpoint is too close to an endpoint")
        progresses.append(prefix / (prefix + suffix))
    gap = abs(progresses[0] - progresses[1])
    if gap < 0.035:
        raise OracleAbstention("pixel progress comparison is ambiguous")
    return "upper" if progresses[0] > progresses[1] else "lower"
