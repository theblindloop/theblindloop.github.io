"""Deterministic renderer and analytic arm for fewer-tick routing."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 720
HEIGHT = 560
BASE_STYLE = {
    "background": [248, 247, 242],
    "ink": [34, 37, 42],
    "start": [35, 171, 96],
    "junction": [137, 86, 205],
    "tick": [238, 190, 38],
    "terminals": {
        "red": [220, 65, 75],
        "blue": [50, 105, 220],
        "orange": [238, 145, 35],
        "cyan": [25, 170, 190],
    },
}
STYLE_OFFSETS = (
    (0, 0, 0),
    (3, -2, 2),
    (-3, 2, -1),
)
TERMINALS = ("T0", "T1", "T2", "T3")
EDGES = (
    ("S", "J0"),
    ("J0", "J1"),
    ("J0", "J2"),
    ("J1", "T0"),
    ("J1", "T1"),
    ("J2", "T2"),
    ("J2", "T3"),
)
QUARANTINE_TICKS = 1.0


def _edge_key(left: str, right: str) -> str:
    return ":".join(sorted((left, right)))


def _shade(rgb: list[int], delta: tuple[int, int, int]) -> list[int]:
    return [max(0, min(255, value + delta[index])) for index, value in enumerate(rgb)]


def _style(variant: int) -> dict:
    delta = STYLE_OFFSETS[variant % len(STYLE_OFFSETS)]
    style = {
        key: _shade(value, delta)
        for key, value in BASE_STYLE.items()
        if key != "terminals"
    }
    style["terminals"] = {
        label: _shade(rgb, delta) for label, rgb in BASE_STYLE["terminals"].items()
    }
    return style


def _tick_pattern(target: int, phase: int) -> dict[str, int]:
    """Return a constant-total annotation whose local minima lead to target.

    Across the three phases, the selected full route has the smallest, middle,
    and third-smallest global tick total. Thus summing a complete route cannot
    replace the two successive local comparisons.
    """

    # selected-root, other-root, selected-leaf, sibling-leaf,
    # unvisited-subtree leaf A, unvisited-subtree leaf B. Each row totals 12.
    patterns = (
        (0, 3, 0, 2, 3, 4),
        (1, 2, 2, 4, 0, 3),
        (0, 1, 4, 5, 0, 2),
    )
    root_selected, root_other, leaf_selected, leaf_other, spare_a, spare_b = patterns[
        phase % len(patterns)
    ]
    selected_child = "J1" if target < 2 else "J2"
    other_child = "J2" if selected_child == "J1" else "J1"
    selected_pair = ("T0", "T1") if selected_child == "J1" else ("T2", "T3")
    other_pair = ("T2", "T3") if selected_child == "J1" else ("T0", "T1")
    selected_terminal = TERMINALS[target]
    sibling_terminal = next(name for name in selected_pair if name != selected_terminal)
    return {
        _edge_key("S", "J0"): 2,
        _edge_key("J0", selected_child): root_selected,
        _edge_key("J0", other_child): root_other,
        _edge_key(selected_child, selected_terminal): leaf_selected,
        _edge_key(selected_child, sibling_terminal): leaf_other,
        _edge_key(other_child, other_pair[0]): spare_a,
        _edge_key(other_child, other_pair[1]): spare_b,
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced visible tree without storing an answer field."""

    seed = int(seed)
    rng = random.Random(seed)
    target = seed % 4
    desired_color = (seed // 4) % 4
    phase = (seed // 4) % 3

    dx = rng.randint(-8, 8)
    dy = rng.randint(-7, 7)
    raw = {
        "S": (62.0, 280.0),
        "J0": (212.0, 280.0),
        "J1": (397.0, 145.0),
        "J2": (397.0, 415.0),
        "T0": (654.0, 65.0),
        "T1": (654.0, 225.0),
        "T2": (654.0, 335.0),
        "T3": (654.0, 495.0),
    }
    nodes: dict[str, list[float]] = {}
    for name, (x, y) in raw.items():
        jx = rng.randint(-3, 3) if name.startswith("J") else rng.randint(-1, 1)
        jy = rng.randint(-3, 3) if name.startswith("J") else rng.randint(-2, 2)
        nodes[name] = [round(x + dx + jx, 3), round(y + dy + jy, 3)]

    colors = list(BASE_STYLE["terminals"])
    answer_color = colors[desired_color]
    remaining = [color for color in colors if color != answer_color]
    rng.shuffle(remaining)
    terminal_colors: dict[str, str] = {}
    for index, terminal in enumerate(TERMINALS):
        terminal_colors[terminal] = answer_color if index == target else remaining.pop()

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "style": _style(seed % len(STYLE_OFFSETS)),
        "nodes": nodes,
        "edges": [list(edge) for edge in EDGES],
        "tick_counts": _tick_pattern(target, phase),
        "terminal_colors": terminal_colors,
    }


def _adjacency(scene: dict) -> dict[str, list[str]]:
    adjacency: dict[str, list[str]] = {}
    for left, right in scene["edges"]:
        adjacency.setdefault(left, []).append(right)
        adjacency.setdefault(right, []).append(left)
    return adjacency


def _route(scene: dict) -> tuple[str, list[float]]:
    adjacency = _adjacency(scene)
    ticks = scene["tick_counts"]
    terminals = set(scene["terminal_colors"])
    previous: str | None = None
    current = "S"
    gaps: list[float] = []
    for _ in range(len(adjacency) + 2):
        if current in terminals:
            return current, gaps
        options = [name for name in adjacency[current] if name != previous]
        if previous is None:
            if len(options) != 1:
                raise ValueError("START must have exactly one link")
            following = options[0]
        else:
            if len(options) != 2:
                raise ValueError("each visited junction must have two forward links")
            scored = sorted((int(ticks[_edge_key(current, name)]), name) for name in options)
            gaps.append(float(scored[1][0] - scored[0][0]))
            if scored[0][0] == scored[1][0]:
                raise ValueError("tied local tick counts")
            following = scored[0][1]
        previous, current = current, following
    raise ValueError("routing did not terminate")


def analytic_gold(scene: dict) -> str:
    terminal, _ = _route(scene)
    return scene["terminal_colors"][terminal]


def margin(scene: dict) -> float:
    _, gaps = _route(scene)
    return min(gaps)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_TICKS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def terminal_color_permutation(scene: dict, mapping: dict[str, str]) -> dict:
    """Public test helper for declared label-transforming color permutations."""

    twin = copy.deepcopy(scene)
    twin["terminal_colors"] = {
        terminal: mapping[color] for terminal, color in scene["terminal_colors"].items()
    }
    return twin


def render(scene: dict) -> Image.Image:
    """Render the annotated network at 2x and downsample for clean pixels."""

    scale = 2
    width, height = int(scene["width"]), int(scene["height"])
    style = scene["style"]
    image = Image.new("RGB", (width * scale, height * scale), tuple(style["background"]))
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]

    def point(name: str) -> tuple[int, int]:
        return (round(nodes[name][0] * scale), round(nodes[name][1] * scale))

    for left, right in scene["edges"]:
        draw.line([point(left), point(right)], fill=tuple(style["ink"]), width=8 * scale)

    for left, right in scene["edges"]:
        count = int(scene["tick_counts"][_edge_key(left, right)])
        if count == 0:
            continue
        x0, y0 = nodes[left]
        x1, y1 = nodes[right]
        vx, vy = x1 - x0, y1 - y0
        length = max(1.0, (vx * vx + vy * vy) ** 0.5)
        px, py = -vy / length, vx / length
        if count == 1:
            positions = [0.5]
        else:
            positions = [0.27 + 0.46 * index / (count - 1) for index in range(count)]
        for t in positions:
            cx, cy = x0 + t * vx, y0 + t * vy
            half = 11.0
            line = [
                ((cx - half * px) * scale, (cy - half * py) * scale),
                ((cx + half * px) * scale, (cy + half * py) * scale),
            ]
            draw.line(line, fill=tuple(style["ink"]), width=10 * scale)
            draw.line(line, fill=tuple(style["tick"]), width=6 * scale)

    for name, (x, y) in nodes.items():
        if name == "S":
            fill, outer, inner = tuple(style["start"]), 20, 16
        elif name in scene["terminal_colors"]:
            label = scene["terminal_colors"][name]
            fill, outer, inner = tuple(style["terminals"][label]), 20, 16
        else:
            fill, outer, inner = tuple(style["junction"]), 18, 14
        outer_box = [
            (x - outer) * scale,
            (y - outer) * scale,
            (x + outer) * scale,
            (y + outer) * scale,
        ]
        inner_box = [
            (x - inner) * scale,
            (y - inner) * scale,
            (x + inner) * scale,
            (y + inner) * scale,
        ]
        draw.ellipse(outer_box, fill=tuple(style["ink"]))
        draw.ellipse(inner_box, fill=fill)

    return image.resize((width, height), Image.Resampling.LANCZOS)
