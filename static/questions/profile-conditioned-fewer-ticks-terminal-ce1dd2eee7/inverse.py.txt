"""Independent pixel-only inverse for fewer-tick routing."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

INK = np.array((34, 37, 42), dtype=np.int16)
YELLOW = np.array((238, 190, 38), dtype=np.int16)
MARKERS = {
    "start": (35, 171, 96),
    "junction": (137, 86, 205),
    "red": (220, 65, 75),
    "blue": (50, 105, 220),
    "orange": (238, 145, 35),
    "cyan": (25, 170, 190),
}
TERMINAL_COLORS = ("red", "blue", "orange", "cyan")
COLOR_TOLERANCE = 30.0
INK_TOLERANCE = 64.0
TICK_TOLERANCE = 30.0
MIN_MARKER_PIXELS = 100
MIN_TICK_PIXELS = 45
NODE_CLEARANCE = 24.0
INTERMEDIATE_CLEARANCE = 25.0
MIN_EDGE_INK_FRACTION = 0.76
MAX_TICK_EDGE_DISTANCE = 8.0


def _components(mask: np.ndarray, minimum: int) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    results: list[tuple[float, float, int]] = []
    seeds_y, seeds_x = np.nonzero(mask)
    for sy, sx in zip(seeds_y.tolist(), seeds_x.tolist()):
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x
            sum_y += y
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if count >= minimum:
            results.append((sum_x / count, sum_y / count, count))
    return results


def _distance_mask(arr: np.ndarray, rgb: np.ndarray | tuple[int, int, int]) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sqrt(np.sum(delta * delta, axis=2))


def _marker_centers(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    centers: dict[str, tuple[float, float]] = {}
    for label, rgb in MARKERS.items():
        components = _components(_distance_mask(arr, rgb) < COLOR_TOLERANCE, MIN_MARKER_PIXELS)
        expected = 3 if label == "junction" else 1
        if len(components) != expected:
            raise ValueError(f"expected {expected} {label} marker(s), found {len(components)}")
        if label == "junction":
            for index, (x, y, _) in enumerate(sorted(components)):
                centers[f"junction_{index}"] = (x, y)
        else:
            x, y, _ = components[0]
            centers[label] = (x, y)
    return centers


def _point_segment_distance(
    point: tuple[float, float], left: tuple[float, float], right: tuple[float, float]
) -> tuple[float, float]:
    vx, vy = right[0] - left[0], right[1] - left[1]
    wx, wy = point[0] - left[0], point[1] - left[1]
    denominator = vx * vx + vy * vy
    if denominator <= 0:
        return float("inf"), 0.0
    t = (wx * vx + wy * vy) / denominator
    px, py = left[0] + t * vx, left[1] + t * vy
    return math.hypot(point[0] - px, point[1] - py), t


def _has_edge(
    left_name: str,
    right_name: str,
    centers: dict[str, tuple[float, float]],
    black: np.ndarray,
) -> bool:
    left, right = centers[left_name], centers[right_name]
    length = math.dist(left, right)
    if length < 2 * NODE_CLEARANCE + 8:
        return False
    for name, point in centers.items():
        if name in {left_name, right_name}:
            continue
        distance, t = _point_segment_distance(point, left, right)
        if 0.06 < t < 0.94 and distance < INTERMEDIATE_CLEARANCE:
            return False

    count = max(40, int(length * 1.35))
    lower = NODE_CLEARANCE / length
    ts = np.linspace(lower, 1.0 - lower, count)
    xs = np.rint(left[0] + ts * (right[0] - left[0])).astype(int)
    ys = np.rint(left[1] + ts * (right[1] - left[1])).astype(int)
    xs = np.clip(xs, 0, black.shape[1] - 1)
    ys = np.clip(ys, 0, black.shape[0] - 1)
    return float(black[ys, xs].mean()) >= MIN_EDGE_INK_FRACTION


def _graph(
    centers: dict[str, tuple[float, float]], arr: np.ndarray
) -> tuple[dict[str, list[str]], list[tuple[str, str]]]:
    black = _distance_mask(arr, INK) < INK_TOLERANCE
    names = list(centers)
    adjacency = {name: [] for name in names}
    edges: list[tuple[str, str]] = []
    for index, left in enumerate(names):
        for right in names[index + 1 :]:
            if _has_edge(left, right, centers, black):
                adjacency[left].append(right)
                adjacency[right].append(left)
                edges.append((left, right))
    if len(adjacency["start"]) != 1:
        raise ValueError(f"START degree is {len(adjacency['start'])}, expected 1")
    for color in TERMINAL_COLORS:
        if len(adjacency[color]) != 1:
            raise ValueError(f"{color} terminal degree is {len(adjacency[color])}, expected 1")
    junctions = [name for name in names if name.startswith("junction_")]
    if any(len(adjacency[name]) != 3 for name in junctions):
        raise ValueError("every purple junction must have degree 3")
    if len(edges) != 7:
        raise ValueError(f"expected seven visible links, found {len(edges)}")
    return adjacency, edges


def _tick_centers(arr: np.ndarray) -> list[tuple[float, float]]:
    components = _components(_distance_mask(arr, YELLOW) < TICK_TOLERANCE, MIN_TICK_PIXELS)
    return [(x, y) for x, y, _ in components]


def _edge_tick_counts(
    edges: list[tuple[str, str]],
    centers: dict[str, tuple[float, float]],
    ticks: list[tuple[float, float]],
) -> dict[frozenset[str], int]:
    counts = {frozenset(edge): 0 for edge in edges}
    for tick in ticks:
        scored = []
        for edge in edges:
            distance, t = _point_segment_distance(tick, centers[edge[0]], centers[edge[1]])
            if 0.12 < t < 0.88:
                scored.append((distance, edge))
        if not scored:
            raise ValueError("yellow tick is not on a link interior")
        scored.sort(key=lambda item: item[0])
        if scored[0][0] > MAX_TICK_EDGE_DISTANCE:
            raise ValueError("yellow tick is too far from every link")
        if len(scored) > 1 and scored[1][0] - scored[0][0] < 10.0:
            raise ValueError("yellow tick ownership is ambiguous")
        counts[frozenset(scored[0][1])] += 1
    return counts


def decision_from_image(image: Image.Image) -> str:
    """Return the reached terminal color using only RGB pixels."""

    arr = np.asarray(image.convert("RGB"))
    centers = _marker_centers(arr)
    adjacency, edges = _graph(centers, arr)
    counts = _edge_tick_counts(edges, centers, _tick_centers(arr))
    previous: str | None = None
    current = "start"
    terminals = set(TERMINAL_COLORS)
    for _ in range(len(centers) + 2):
        if current in terminals:
            return current
        options = [name for name in adjacency[current] if name != previous]
        if previous is None:
            following = options[0]
        else:
            if len(options) != 2:
                raise ValueError(f"visited junction has {len(options)} forward choices")
            scored = sorted((counts[frozenset((current, name))], name) for name in options)
            if scored[0][0] == scored[1][0]:
                raise ValueError("pixel tick-count choice is tied")
            following = scored[0][1]
        previous, current = current, following
    raise ValueError("pixel routing did not reach a terminal")
