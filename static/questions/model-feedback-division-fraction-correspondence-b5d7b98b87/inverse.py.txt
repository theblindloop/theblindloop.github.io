"""Independent pixel-only inverse arm for the division-fraction world.

Recovers the decision (which candidate reproduces the reference division?)
from the final rasterized PNG alone. It detects the engineered two-tone bars
by their exact navy and teal colours, measures, within the reference panel and
each candidate panel, the fraction of the bar's own length that lies left of
the navy/teal boundary, and returns the label of the candidate whose measured
fraction matches the reference's within a tolerance far smaller than the
guaranteed distractor gap.

The three panels occupy fixed rows (renderer geometry), so the arm reads
``decision_from_image`` from the image pixels alone and never receives the
latent scene. No renderer / prompts / generate / verify / analytic-gold code is
imported.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

NAVY = (45, 65, 150)
TEAL = (0, 150, 136)
BAR_HEIGHT = 38
PANEL_TOPS = {"reference": 150, "A": 320, "B": 460}
MATCH_TOL = 0.08


def _fraction_in_band(arr: np.ndarray, top: int) -> tuple[float, bool]:
    """Fraction of the bar's length left of the navy/teal boundary within a row band."""
    band = arr[top : top + BAR_HEIGHT]
    navy = (band == NAVY).all(axis=-1)
    teal = (band == TEAL).all(axis=-1)
    ink = navy | teal
    cols = np.any(ink, axis=0)
    xs = np.nonzero(cols)[0]
    if xs.size == 0:
        return 0.0, False
    min_x = int(xs[0])
    max_x = int(xs[-1])
    width = max_x - min_x + 1
    if width <= 0:
        return 0.0, False
    # Boundary: first column (left to right) that is teal.
    boundary = int(min_x) + int(np.argmax(np.any(teal[:, min_x : max_x + 1], axis=0)))
    frac = (boundary - min_x + 0.5) / width
    return float(frac), True


def decision_from_image(image: Image.Image) -> str:
    """Return A or B: which candidate reproduces the reference left/right division."""
    arr = np.asarray(image.convert("RGB"))
    f_ref, ok_ref = _fraction_in_band(arr, PANEL_TOPS["reference"])
    f_a, ok_a = _fraction_in_band(arr, PANEL_TOPS["A"])
    f_b, ok_b = _fraction_in_band(arr, PANEL_TOPS["B"])
    if not (ok_ref and ok_a and ok_b):
        return "A"
    da = abs(f_a - f_ref)
    db = abs(f_b - f_ref)
    if da <= db:
        return "A"
    return "B"
