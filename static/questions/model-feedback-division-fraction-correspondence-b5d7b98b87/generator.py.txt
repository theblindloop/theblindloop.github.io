"""Division-fraction correspondence world.

Three horizontal two-tone bars are drawn on a white canvas. The reference bar
(at the top) is split by a vertical colour boundary into a navy left part and a
teal right part, so that the navy part occupies a fraction ``f_ref`` of the
bar's own length. Two candidate bars (``A`` and ``B``) are drawn below at
different total widths and positions. Exactly one candidate is split at the
*same fraction* ``f_ref`` of its own length; the other is split at a clearly
different fraction. The decision is which candidate — A or B — reproduces the
reference's left/right division, i.e. a bound comparison of two normalized
quantitative proportions, not of absolute boundary positions or total lengths.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
decision from the final PNG alone by measuring, within each panel, the fraction
of the bar's length that lies left of the navy/teal boundary, then selecting the
candidate whose measured fraction matches the reference's.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 560
BACKGROUND = (255, 255, 255)
NAVY = (45, 65, 150)
TEAL = (0, 150, 136)
LABEL = (35, 35, 35)

BAR_HEIGHT = 38
# Panel top rows: reference / candidate A / candidate B.
PANEL_TOPS = {"reference": 150, "A": 320, "B": 460}

REF_W_LO, REF_W_HI = 220, 300
CAND_W_LO, CAND_W_HI = 170, 300

FRAC_LO, FRAC_HI = 0.10, 0.90
FRAC_SAFE_LO, FRAC_SAFE_HI = 0.20, 0.80
DELTA_LO, DELTA_HI = 0.16, 0.28
QUAR_MIN_MARGIN = 0.15

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)

    ref_w = float(rng.integers(REF_W_LO, REF_W_HI + 1))
    a_w = float(rng.integers(CAND_W_LO, CAND_W_HI + 1))
    b_w = float(rng.integers(CAND_W_LO, CAND_W_HI + 1))
    ref_cx = float(rng.uniform(0.35, 0.65) * WIDTH)
    a_cx = float(rng.uniform(0.30, 0.70) * WIDTH)
    b_cx = float(rng.uniform(0.30, 0.70) * WIDTH)

    f_ref = float(rng.uniform(FRAC_LO, FRAC_HI))
    direction = float(rng.choice([-1.0, 1.0]))
    delta = float(rng.uniform(DELTA_LO, DELTA_HI))
    f_dist = f_ref + direction * delta
    if not (FRAC_LO - 0.05 <= f_dist <= FRAC_HI + 0.05):
        f_dist = f_ref - direction * delta

    correct_index = int(rng.integers(0, 2))
    if correct_index == 0:
        frac_a = f_ref
        frac_b = f_dist
    else:
        frac_a = f_dist
        frac_b = f_ref

    return {
        "reference_frac": float(round(f_ref, 4)),
        "frac_a": float(round(frac_a, 4)),
        "frac_b": float(round(frac_b, 4)),
        "ref_width": float(ref_w),
        "ref_cx": float(ref_cx),
        "a_width": float(a_w),
        "a_cx": float(a_cx),
        "b_width": float(b_w),
        "b_cx": float(b_cx),
        "delta": float(round(delta, 4)),
        "correct_index": correct_index,
    }


def _bar_box(center_x: float, width: float, top: int, height: int) -> tuple[float, float, int, int]:
    x0 = center_x - width / 2.0
    x1 = center_x + width / 2.0
    return x0, x1, top, top + height


def _draw_bar(draw: ImageDraw.ImageDraw, cx: float, width: float, top: int, frac: float) -> None:
    x0, x1, top, bottom = _bar_box(cx, width, top, BAR_HEIGHT)
    boundary = int(round(x0 + frac * width))
    draw.rectangle([x0, top, boundary, bottom], fill=NAVY)
    draw.rectangle([boundary + 1, top, x1, bottom], fill=TEAL)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype(_FONT_PATH, 26)
    except OSError:
        font = ImageFont.load_default()

    f_ref = scene["reference_frac"]
    draw.text((12, 116), "REFERENCE", font=font, fill=LABEL)
    _draw_bar(draw, scene["ref_cx"], scene["ref_width"], PANEL_TOPS["reference"], f_ref)
    draw.text((12, PANEL_TOPS["A"] + 6), "A", font=font, fill=LABEL)
    _draw_bar(draw, scene["a_cx"], scene["a_width"], PANEL_TOPS["A"], scene["frac_a"])
    draw.text((12, PANEL_TOPS["B"] + 6), "B", font=font, fill=LABEL)
    _draw_bar(draw, scene["b_cx"], scene["b_width"], PANEL_TOPS["B"], scene["frac_b"])
    return img


def is_correct_a(scene: dict) -> bool:
    """True iff candidate A is the one split at the reference fraction."""
    return (
        scene["frac_a"] == scene["reference_frac"]
        and scene["frac_b"] != scene["reference_frac"]
    )


def analytic_gold(scene: dict) -> str:
    """Structural gold: which candidate reproduces the reference division."""
    return "A" if is_correct_a(scene) else "B"


def margin(scene: dict) -> float:
    """Commitment magnitude: absolute gap between the two candidate fractions."""
    return float(abs(scene["frac_a"] - scene["frac_b"]))


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where the division would be illegible or ambiguous."""
    f_ref = scene["reference_frac"]
    if not (FRAC_SAFE_LO <= f_ref <= FRAC_SAFE_HI):
        return True
    if margin(scene) < QUAR_MIN_MARGIN:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
