"""Independent image-only inverse for the two-switch braid.

The arm segments public colors, locates the two disconnected switch windows,
infers each local two-port mapping from navy pixels, constructs the serial
two-state graph, and traces the blue-selected lane to a colored terminal.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _mask(array: np.ndarray, name: str) -> np.ndarray:
    """Broad channel rules cover both public palette variants and antialiasing."""
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    if name == "navy":
        return (red < 85) & (green < 100) & (blue < 125) & ((blue - red) > 20)
    if name == "violet":
        return (red >= 90) & (red < 185) & (green < 130) & (blue >= 125) & ((blue - red) < 85)
    if name == "blue":
        return (red < 100) & (green >= 75) & (blue > 145) & ((blue - red) >= 85)
    if name == "red":
        return (red > 165) & (green < 115) & (blue < 125) & ((red - green) > 90)
    if name == "gold":
        return (red > 175) & (green >= 105) & (green < 205) & (blue < 100) & ((red - blue) > 120)
    raise ValueError(f"unknown public color class {name!r}")


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _one_centroid(mask: np.ndarray, name: str, minimum: int = 25) -> tuple[float, float]:
    groups = _components(mask, minimum)
    if len(groups) != 1:
        raise ValueError(f"expected one {name} marker, found {len(groups)}")
    y, x = groups[0].mean(axis=0)
    return float(x), float(y)


def _switch_state(navy: np.ndarray, cx: float, cy: float) -> int:
    yy, xx = np.indices(navy.shape)
    window = navy & (np.abs(xx - cx) <= 18) & (np.abs(yy - cy) <= 18)
    if int(window.sum()) < 120:
        raise ValueError("switch has insufficient visible route ink")
    core = window & (np.abs(xx - cx) <= 5) & (np.abs(yy - cy) <= 5)
    # A crossed map has its explicitly resolved overpass through the window core;
    # a same-lane map leaves the core visibly white between its two strands.
    return 1 if int(core.sum()) >= 12 else 0


def decision_from_image(image: Image.Image) -> str:
    full = np.asarray(image.convert("RGB"), dtype=np.uint8)
    spread = full.max(axis=2).astype(np.int16) - full.min(axis=2).astype(np.int16)
    relevant = (spread > 28) & (full.min(axis=2) < 210)
    points = np.argwhere(relevant)
    if not len(points):
        raise ValueError("no colored braid evidence found")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    pad = 4
    array = full[max(0, y0 - pad) : min(full.shape[0], y1 + pad + 1), max(0, x0 - pad) : min(full.shape[1], x1 + pad + 1)]
    violet_groups = _components(_mask(array, "violet"), 90)
    if len(violet_groups) != 2:
        raise ValueError(f"expected two switch windows, found {len(violet_groups)}")
    centers = []
    for group in violet_groups:
        y, x = group.mean(axis=0)
        centers.append((float(x), float(y)))
    centers.sort()
    (left_x, left_y), (right_x, right_y) = centers
    if right_x - left_x < 120 or abs(left_y - right_y) > 5:
        raise ValueError("switch windows do not form a clear serial pair")

    blue_x, blue_y = _one_centroid(_mask(array, "blue"), "blue start")
    red_x, red_y = _one_centroid(_mask(array, "red"), "red terminal")
    gold_x, gold_y = _one_centroid(_mask(array, "gold"), "gold terminal")
    if not (blue_x < left_x and red_x > right_x and gold_x > right_x):
        raise ValueError("endpoint markers are not attached to the expected window edges")

    navy = _mask(array, "navy")
    mappings = (_switch_state(navy, left_x, left_y), _switch_state(navy, right_x, right_y))
    start_lane = 0 if blue_y < left_y else 1
    red_lane = 0 if red_y < right_y else 1
    gold_lane = 0 if gold_y < right_y else 1
    if red_lane == gold_lane:
        raise ValueError("terminal markers do not occupy separate lanes")

    # Explicit graph trace through two serial permutation edges.
    lane = start_lane
    for swaps_lanes in mappings:
        lane ^= swaps_lanes
    return "red" if lane == red_lane else "gold"
