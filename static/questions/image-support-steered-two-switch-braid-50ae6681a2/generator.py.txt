"""Deterministic analytic world for a compact serial two-switch braid."""

from __future__ import annotations

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
SCALE = 4

PALE = (218, 224, 230)
PALE_2 = (232, 235, 239)
WHITE = (255, 255, 255)

PALETTES = (
    {
        "navy": (28, 48, 72),
        "violet": (128, 82, 171),
        "blue": (34, 112, 195),
        "red": (207, 55, 67),
        "gold": (226, 158, 28),
    },
    {
        "navy": (35, 55, 78),
        "violet": (145, 91, 181),
        "blue": (43, 125, 205),
        "red": (218, 66, 75),
        "gold": (235, 169, 39),
    },
)


def sample_scene(seed: int) -> dict:
    """Cycle answer-bearing bits so every consecutive 16 scenes is balanced."""
    seed = int(seed)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "state_a": seed & 1,
        "state_b": (seed >> 1) & 1,
        "start_lane": (seed >> 2) & 1,
        "red_lane": (seed >> 3) & 1,
        "x_slot": (seed >> 4) & 1,
        "y_slot": (seed >> 5) & 1,
        "stroke_width": 4 + ((seed >> 6) & 1),
        "lane_offset": 10 + ((seed >> 7) & 1),
        "palette_variant": (seed >> 8) & 1,
        "decoy_pattern": (seed * 7 + 3) % 4,
    }


def _centers(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    left_x = 250 + 100 * int(scene["x_slot"])
    right_x = left_x + 200
    center_y = 350 + 100 * int(scene["y_slot"])
    return (left_x, center_y), (right_x, center_y)


def _line(draw: ImageDraw.ImageDraw, points, *, fill, width: int) -> None:
    draw.line([(int(x * SCALE), int(y * SCALE)) for x, y in points], fill=fill, width=width * SCALE, joint="curve")


def _ellipse(draw: ImageDraw.ImageDraw, box, *, fill, outline=None, width=1) -> None:
    draw.ellipse(tuple(int(v * SCALE) for v in box), fill=fill, outline=outline, width=width * SCALE)


def _rectangle(draw: ImageDraw.ImageDraw, box, *, outline, width: int) -> None:
    draw.rounded_rectangle(tuple(int(v * SCALE) for v in box), radius=4 * SCALE, outline=outline, width=width * SCALE)


def _draw_decoration(draw: ImageDraw.ImageDraw, pattern: int) -> None:
    # Remote, low-contrast route-like marks; they never enter analytic gold.
    options = (
        [((70, 120), (185, 120), (185, 205)), ((610, 635), (725, 635), (725, 550))],
        [((85, 640), (170, 555), (205, 590)), ((610, 135), (700, 210), (735, 165))],
        [((70, 210), (145, 165), (205, 215)), ((600, 570), (680, 520), (735, 575))],
        [((90, 575), (180, 655), (215, 610)), ((595, 220), (675, 145), (730, 205))],
    )
    for index, points in enumerate(options[int(pattern)]):
        _line(draw, points, fill=PALE if index == 0 else PALE_2, width=5)
        for x, y in (points[0], points[-1]):
            _ellipse(draw, (x - 5, y - 5, x + 5, y + 5), fill=PALE)


def _draw_switch(draw: ImageDraw.ImageDraw, center, state: int, palette, stroke: int, offset: int) -> None:
    cx, cy = center
    x0, x1 = cx - 17, cx + 17
    if int(state) == 0:
        _line(draw, ((x0, cy - offset), (cx, cy - offset - 4), (x1, cy - offset)), fill=palette["navy"], width=stroke)
        _line(draw, ((x0, cy + offset), (cx, cy + offset + 4), (x1, cy + offset)), fill=palette["navy"], width=stroke)
    else:
        _line(draw, ((x0, cy - offset), (x1, cy + offset)), fill=palette["navy"], width=stroke)
        _line(draw, ((x0, cy + offset), (x1, cy - offset)), fill=palette["navy"], width=stroke)
        # Explicit over/under cue: one diagonal is continuous over a white break.
        _ellipse(draw, (cx - 5, cy - 5, cx + 5, cy + 5), fill=WHITE)
        _line(draw, ((cx - 7, cy - 4), (cx + 7, cy + 4)), fill=palette["navy"], width=stroke)


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    state_a = int(scene["state_a"])
    state_b = int(scene["state_b"])
    start_lane = int(scene["start_lane"])
    red_lane = int(scene["red_lane"])
    stroke = int(scene["stroke_width"])
    offset = int(scene["lane_offset"])
    palette = PALETTES[int(scene["palette_variant"])]
    decoy_pattern = int(scene["decoy_pattern"])
    left, right = _centers(scene)

    image = Image.new("RGB", (width * SCALE, height * SCALE), WHITE)
    draw = ImageDraw.Draw(image)
    _draw_decoration(draw, decoy_pattern)

    # The long parallel rails are structurally redundant between the two local maps.
    for lane in (0, 1):
        y = left[1] + (-offset if lane == 0 else offset)
        _line(draw, ((left[0] + 17, y), (right[0] - 17, y)), fill=palette["navy"], width=stroke)
    _draw_switch(draw, left, state_a, palette, stroke, offset)
    _draw_switch(draw, right, state_b, palette, stroke, offset)

    # Window outlines are visible delimiters, disconnected from one another.
    for cx, cy in (left, right):
        _rectangle(draw, (cx - 22, cy - 22, cx + 22, cy + 22), outline=palette["violet"], width=3)

    start_y = left[1] + (-offset if start_lane == 0 else offset)
    _ellipse(draw, (left[0] - 22, start_y - 6, left[0] - 10, start_y + 6), fill=palette["blue"], outline=WHITE, width=1)
    for lane in (0, 1):
        y = right[1] + (-offset if lane == 0 else offset)
        color = palette["red"] if lane == red_lane else palette["gold"]
        _ellipse(draw, (right[0] + 10, y - 6, right[0] + 22, y + 6), fill=color, outline=WHITE, width=1)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    lane = int(scene["start_lane"]) ^ int(scene["state_a"]) ^ int(scene["state_b"])
    return "red" if lane == int(scene["red_lane"]) else "gold"


def margin(scene: dict) -> float:
    return float(int(scene["lane_offset"]) - int(scene["stroke_width"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
