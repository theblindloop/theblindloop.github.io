"""Renderer and analytic specification for a color-keyed bottleneck board.

Each of three panels contains one bar for each of four repeated colors.  The
panels share one multiset of integer heights, but the color assignment changes.
For each candidate, the score is the smallest symmetric coverage ratio across
the four color matches: min(candidate, reference) / max(candidate, reference).
"""

from __future__ import annotations

import itertools
import random
from fractions import Fraction
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "ratio_bottleneck_compare-renderer-0.1.0"

WIDTH = 1320
HEIGHT = 860
PANEL_X = (35, 465, 895)
PANEL_WIDTH = 390
PANEL_TOP = 145
PANEL_BOTTOM = 735
BASELINE = 655
LEVEL_PITCH = 54
BAR_WIDTH = 48
BAR_OFFSETS = (88, 158, 228, 298)
LEVEL_CHOICES = tuple(range(1, 7))
COLOR_NAMES = ("red", "blue", "green", "violet")
COLOR_SHORT = {"red": "R", "blue": "B", "green": "G", "violet": "V"}
BAR_COLORS = {
    "red": (210, 70, 67),
    "blue": (52, 101, 184),
    "green": (47, 145, 91),
    "violet": (145, 78, 174),
}

BACKGROUND = (248, 249, 246)
PANEL_FILL = (255, 255, 255)
INK = (38, 49, 58)
SUBTLE = (121, 137, 145)
GUIDE = (218, 226, 227)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2),
        text,
        font=font,
        fill=fill,
    )


def _validate_scene(scene: dict) -> tuple[list[int], list[int], list[int]]:
    reference = [int(value) for value in scene["reference_values"]]
    candidate_a = [int(value) for value in scene["candidate_a_values"]]
    candidate_b = [int(value) for value in scene["candidate_b_values"]]
    values = (reference, candidate_a, candidate_b)
    if any(len(item) != len(COLOR_NAMES) for item in values):
        raise ValueError("the board needs four values in each panel")
    if any(value not in LEVEL_CHOICES for item in values for value in item):
        raise ValueError("bar heights must be integer ruler levels from 1 through 6")
    if any(len(set(item)) != len(COLOR_NAMES) for item in values):
        raise ValueError("each panel needs four distinct visible levels")
    if any(sorted(item) != sorted(reference) for item in (candidate_a, candidate_b)):
        raise ValueError("all panels must use one shared height multiset")
    return reference, candidate_a, candidate_b


def _coverage(reference: Iterable[int], candidate: Iterable[int]) -> Fraction:
    ratios = [
        Fraction(min(int(value), int(target)), max(int(value), int(target)))
        for value, target in zip(candidate, reference, strict=True)
    ]
    return min(ratios)


def _l1(reference: Iterable[int], candidate: Iterable[int]) -> int:
    return sum(abs(int(value) - int(target)) for value, target in zip(candidate, reference, strict=True))


def _pairs_for_target(
    reference: tuple[int, ...], target: str, min_gap: Fraction = Fraction(1, 24)
) -> list[tuple[tuple[int, ...], tuple[int, ...]]]:
    permutations = sorted(set(itertools.permutations(reference)))
    adversarial: list[tuple[tuple[int, ...], tuple[int, ...]]] = []
    ordinary: list[tuple[tuple[int, ...], tuple[int, ...]]] = []
    for candidate_a in permutations:
        score_a = _coverage(reference, candidate_a)
        for candidate_b in permutations:
            score_b = _coverage(reference, candidate_b)
            delta = score_a - score_b
            if target == "A" and delta >= min_gap:
                if _l1(reference, candidate_a) < _l1(reference, candidate_b):
                    adversarial.append((candidate_a, candidate_b))
                ordinary.append((candidate_a, candidate_b))
            elif target == "B" and delta <= -min_gap:
                if _l1(reference, candidate_a) > _l1(reference, candidate_b):
                    adversarial.append((candidate_a, candidate_b))
                ordinary.append((candidate_a, candidate_b))
            elif target == "tie" and delta == 0 and candidate_a != candidate_b:
                ordinary.append((candidate_a, candidate_b))
    return adversarial or ordinary


def sample_scene(seed: int) -> dict:
    """Sample a deterministic shared-level board with both signs and ties."""

    rng = random.Random(int(seed))
    levels = tuple(sorted(rng.sample(list(LEVEL_CHOICES), len(COLOR_NAMES))))
    reference = tuple(rng.choice(sorted(set(itertools.permutations(levels)))))
    if int(seed) % 23 == 0:
        target = "tie"
    else:
        target = "A" if int(seed) % 2 == 0 else "B"
    pairs = _pairs_for_target(reference, target)
    if not pairs:
        raise RuntimeError(f"no {target} bottleneck pair exists for levels {levels}")
    candidate_a, candidate_b = rng.choice(pairs)
    return {
        "reference_values": list(reference),
        "candidate_a_values": list(candidate_a),
        "candidate_b_values": list(candidate_b),
    }


def render(scene: dict) -> Image.Image:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    body_font = _font(18)
    panel_font = _font(21)
    small_font = _font(15)
    number_font = _font(17)

    draw.text((42, 23), "COLOR COVERAGE BOARD", font=title_font, fill=INK)
    draw.text((42, 61), "The same four colors repeat; use the shared ruler to match their heights.", font=body_font, fill=INK)
    draw.text((42, 91), "Coverage = smaller height / larger height. A candidate's score is its smallest coverage.", font=small_font, fill=SUBTLE)

    panel_values = (reference, candidate_a, candidate_b)
    panel_titles = ("REFERENCE", "CANDIDATE A", "CANDIDATE B")
    for panel_index, (x0, values, title) in enumerate(zip(PANEL_X, panel_values, panel_titles, strict=True)):
        _centered_text(draw, (x0 + PANEL_WIDTH / 2, 125), title, panel_font, INK)
        draw.rectangle((x0, PANEL_TOP, x0 + PANEL_WIDTH, PANEL_BOTTOM), fill=PANEL_FILL, outline=INK, width=2)
        ruler_x = x0 + 48
        for level in range(0, 7):
            y = BASELINE - level * LEVEL_PITCH
            draw.line((ruler_x, y, x0 + PANEL_WIDTH - 15, y), fill=GUIDE, width=1)
            draw.line((ruler_x - 7, y, ruler_x + 7, y), fill=SUBTLE, width=2)
            _centered_text(draw, (x0 + 23, y), str(level), number_font, INK)
        draw.line((ruler_x, BASELINE - 6 * LEVEL_PITCH, ruler_x, BASELINE), fill=SUBTLE, width=2)
        for color_name, value, offset in zip(COLOR_NAMES, values, BAR_OFFSETS, strict=True):
            left = x0 + offset
            top = BASELINE - int(value) * LEVEL_PITCH
            draw.rectangle((left, top, left + BAR_WIDTH, BASELINE), fill=BAR_COLORS[color_name])
            _centered_text(draw, (left + BAR_WIDTH / 2, BASELINE + 28), COLOR_SHORT[color_name], small_font, INK)
        if panel_index == 0:
            draw.text((x0 + 14, PANEL_BOTTOM - 28), "height levels", font=small_font, fill=SUBTLE)

    draw.text((42, 782), "For each color, match its bar to REFERENCE, form its coverage ratio, and keep the smallest one.", font=body_font, fill=INK)
    draw.text((42, 815), "The answer is the candidate with the larger worst-color ratio; equal scores are tie.", font=small_font, fill=SUBTLE)
    return image


def analytic_gold(scene: dict) -> str:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    score_a = _coverage(reference, candidate_a)
    score_b = _coverage(reference, candidate_b)
    if score_a > score_b:
        return "A"
    if score_b > score_a:
        return "B"
    return "tie"


def margin(scene: dict) -> float:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    return float(_coverage(reference, candidate_a) - _coverage(reference, candidate_b))


def is_quarantined(scene: dict) -> bool:
    reference, candidate_a, candidate_b = _validate_scene(scene)
    return _coverage(reference, candidate_a) == _coverage(reference, candidate_b)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every scene field changes a visible bar top, so there is no latent alias
    # transform to exercise for this direct raster encoding.
    return []
