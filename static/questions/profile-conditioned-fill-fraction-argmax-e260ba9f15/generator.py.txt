"""Filled-fraction argmax selection world.

Which colored vertical bar is filled to the greatest fraction of its own total
height? Every colored bar is drawn as a light-gray full-height skeleton (its
visible total height) with a solid colored fill rising from a common black
baseline. The decision is which color's *within-bar ratio* (colored fill height
divided by that bar's own total skeleton height) is largest.

Latent scene z carries a list ``bars``; three bars each record a color label,
an x-slot, a ``total`` height, and a ``fill`` height. The measured quantity is a
relative, per-mark proportion (fill/total); no single absolute size, position,
or fill amount alone determines the answer because totals and fills are varied
independently.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "fill_fraction_argmax-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
SKELETON = (200, 200, 200)
BASELINE = (35, 35, 35)
BACKGROUND = (250, 250, 250)

WIDTH = 520
HEIGHT = 380
BAR_WIDTH = 46
BASELINE_ROW = 340
SLOTS = [150, 260, 370]

MIN_TOTAL = 90.0
MAX_TOTAL = 200.0
MIN_FILL = 25.0

# Minimum top-two ratio gap for a decisively selectable scene.
MIN_RATIO_MARGIN = 0.12
# Below this top-two ratio gap the scene is a finite-resolution boundary case.
QUARANTINE_RATIO = 0.07


def _ratio(b: dict) -> float:
    return b["fill"] / b["total"]


def sample_scene(seed: int) -> dict:
    """Deterministic scene with a guaranteed, counterbalanced max-ratio winner."""
    rng = random.Random(seed * 1049 + 11)
    winner = PALETTE[seed % 3]
    slots = list(SLOTS)
    rng.shuffle(slots)
    while True:
        totals = {c: rng.uniform(MIN_TOTAL, MAX_TOTAL) for c in PALETTE}
        fills = {c: rng.uniform(MIN_FILL, totals[c] - 45.0) for c in PALETTE}
        max_other = max(fills[c] / totals[c] for c in PALETTE if c != winner)
        target = (max_other + MIN_RATIO_MARGIN) * totals[winner]
        if target < MIN_FILL + 5.0 or target > totals[winner] - 8.0:
            continue
        fills[winner] = target
        ratios = sorted((fills[c] / totals[c] for c in PALETTE), reverse=True)
        if ratios[0] - ratios[1] >= MIN_RATIO_MARGIN:
            break
    bars = []
    for color, slot in zip(PALETTE, slots):
        bars.append(
            {
                "color": color,
                "x0": float(slot),
                "x1": float(slot + BAR_WIDTH),
                "total": float(totals[color]),
                "fill": float(fills[color]),
            }
        )
    return {
        "bars": bars,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three colored bars with gray skeletons, black baseline."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    draw.rectangle([0, BASELINE_ROW - 1, WIDTH, BASELINE_ROW + 1], fill=BASELINE)
    for bar in scene["bars"]:
        x0, x1 = int(bar["x0"]), int(bar["x1"])
        top = int(BASELINE_ROW - bar["total"])
        draw.rectangle([x0, top, x1, BASELINE_ROW], fill=SKELETON)
        ftop = int(BASELINE_ROW - bar["fill"])
        draw.rectangle([x0, ftop, x1, BASELINE_ROW], fill=COLOR_RGB[bar["color"]])
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the bar with the greatest filled-fraction (fill/total)."""
    bars = scene["bars"]
    best = max(bars, key=_ratio)
    return best["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two filled-fraction gap (ratio units)."""
    ratios = sorted(_ratio(b) for b in scene["bars"])
    return float(ratios[-1] - ratios[-2])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two filled-fraction gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_RATIO


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each bar's color label, its x-slot, its total height,
    and its fill height) is directly rendered as visible ink: color and the two
    heights are read straight off the bar and its gray skeleton, and the x-slot
    is the bar's drawn position. Swapping two labels changes which color is
    correct, and any global reflection/rotation turns the raster into a
    distinct image, so no latent transform is both pixel-identical and gold-
    preserving.
    """
    return []
