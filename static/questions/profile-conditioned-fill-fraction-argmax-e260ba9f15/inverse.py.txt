"""Independent pixel-only inverse arm for the fill_fraction_argmax world.

Receives ONLY the rasterized PNG (plus the world's fixed public palette, which
is the naming convention) and re-measures, for every colored vertical bar, its
own fill fraction: the height of the solid colored fill above the black
baseline divided by the height of the light-gray full-height skeleton above the
same baseline. It shares no geometry with renderer.render() and never imports
the renderer, prompts, generator, verifier, or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "fill_fraction_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
SKELETON = (200, 200, 200)
BASELINE = (35, 35, 35)
BACKGROUND = (250, 250, 250)


def _nearest(arr: np.ndarray, refs: np.ndarray) -> np.ndarray:
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _find_baseline(arr: np.ndarray) -> int:
    """Row index of the black baseline (the row whose span is widest dark band)."""
    dark = (np.abs(arr.astype(np.float32) - np.array(BASELINE)).sum(axis=-1) < 60)
    counts = dark.sum(axis=1)
    band = int(np.argmax(counts))
    return band


def decision_from_image(image: Image.Image) -> str:
    """Color of the bar whose own fill fraction (fill/total) is largest."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(SKELETON, dtype=np.float32)]
        + [np.array(BASELINE, dtype=np.float32)]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    labels = _nearest(arr, refs)

    baseline = _find_baseline(arr)

    best_color = PALETTE[0]
    best_ratio = -1.0
    for idx, color in enumerate(PALETTE):
        color_mask = labels == idx
        rows, cols = np.nonzero(color_mask)
        if rows.size == 0:
            continue
        fill_top = int(rows.min())
        cx0, cx1 = int(cols.min()), int(cols.max())
        fill_height = max(1, baseline - fill_top)
        # Light-gray skeleton within this bar's column band, above the fill.
        band = labels[:, cx0 : cx1 + 1]
        skeleton_idx = len(PALETTE)
        skeleton_ys = np.nonzero(band == skeleton_idx)[0]  # y indices within band
        if skeleton_ys.size == 0:
            continue
        total_top = int(skeleton_ys.min())
        total_height = max(1, baseline - total_top)
        ratio = fill_height / total_height
        if ratio > best_ratio:
            best_ratio = ratio
            best_color = color
    return best_color
