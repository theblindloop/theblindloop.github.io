"""Deterministic snowman world with visible structural-section count."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
QUESTION_COUNTS = (2, 3, 4, 5)
PALETTES = (
    {"sky": "#8fc9e8", "outline": "#17324d", "accent": "#d9473f", "ground": "#d9edf5", "arm": "#70462d"},
    {"sky": "#a6c4e6", "outline": "#28364f", "accent": "#de7b2d", "ground": "#e1edf7", "arm": "#76513a"},
    {"sky": "#82c7c4", "outline": "#203a48", "accent": "#b7476f", "ground": "#d8efec", "arm": "#6d4934"},
)
SNOW = "#fbfcf8"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    count = QUESTION_COUNTS[int(seed) % len(QUESTION_COUNTS)]
    flakes: list[list[int]] = []
    for _ in range(10):
        side = rng.choice((0, 1))
        x = rng.randint(28, 128) if side == 0 else rng.randint(384, 484)
        y = rng.randint(38, 385)
        flakes.append([x, y])
    return {
        "section_count": count,
        "palette_id": rng.randrange(len(PALETTES)),
        "x_shift": rng.randint(-18, 18),
        "body_scale": round(rng.uniform(1.0, 1.04), 4),
        "hat_tilt": rng.choice((-1, 0, 1)),
        "arm_pose": rng.choice((0, 1, 2)),
        "flake_points": flakes,
    }


def _palette(scene: dict) -> dict:
    return PALETTES[int(scene["palette_id"]) % len(PALETTES)]


def _layout(scene: dict) -> list[tuple[float, float, float]]:
    n = int(scene["section_count"])
    if n not in QUESTION_COUNTS:
        raise ValueError("section_count must be 2, 3, 4, or 5")
    scale = float(scene["body_scale"])
    weights = [1.18 - 0.36 * i / (n - 1) for i in range(n)]
    norm = math.sqrt(sum(value * value for value in weights))
    # More sections incur proportionally more outline and button cut-outs. These
    # calibrated aggregate squared-radius targets keep total near-white body ink
    # approximately matched across answer identities.
    area_target = {2: 6100.0, 3: 6280.0, 4: 6740.0, 5: 6910.0}[n]
    radii = [value * math.sqrt(area_target) / norm * scale for value in weights]
    gap = 10.0
    x0 = 256.0 + float(scene["x_shift"])
    y = 454.0 - radii[0]
    circles = [(x0, y, radii[0])]
    for i in range(1, n):
        y -= radii[i - 1] + gap + radii[i]
        circles.append((x0, y, radii[i]))
    return circles


def _S(value: float) -> int:
    return int(round(value * 3))


def _ellipse(draw: ImageDraw.ImageDraw, box, *, fill, outline=None, width=1) -> None:
    draw.ellipse(tuple(_S(v) for v in box), fill=fill, outline=outline, width=_S(width))


def _line(draw: ImageDraw.ImageDraw, points, *, fill, width) -> None:
    draw.line([(_S(x), _S(y)) for x, y in points], fill=fill, width=_S(width), joint="curve")


def _polygon(draw: ImageDraw.ImageDraw, points, *, fill, outline=None, width=1) -> None:
    pts = [(_S(x), _S(y)) for x, y in points]
    draw.polygon(pts, fill=fill)
    if outline:
        draw.line(pts + [pts[0]], fill=outline, width=_S(width), joint="curve")


def render(scene: dict) -> Image.Image:
    palette = _palette(scene)
    circles = _layout(scene)
    tilt = int(scene["hat_tilt"])
    pose = int(scene["arm_pose"])
    flakes = list(scene["flake_points"])
    canvas = Image.new("RGB", (WIDTH * 3, HEIGHT * 3), palette["sky"])
    draw = ImageDraw.Draw(canvas)

    # Fixed frame and ground keep matched global composition across count classes.
    _line(draw, [(22, 478), (490, 478)], fill=palette["outline"], width=3)
    draw.rectangle((_S(0), _S(481), _S(512), _S(512)), fill=palette["ground"])
    for x, y in flakes:
        _line(draw, [(x - 3, y), (x + 3, y)], fill=SNOW, width=2)
        _line(draw, [(x, y - 3), (x, y + 3)], fill=SNOW, width=2)

    # Arms are drawn behind the snow structure. Their pose varies independently.
    mid = circles[min(1, len(circles) - 1)]
    cx, cy, r = mid
    arm_dy = (-12, 0, 12)[pose % 3]
    for sign in (-1, 1):
        start = (cx + sign * (r - 4), cy - 2)
        elbow = (cx + sign * (r + 35), cy - 20 + arm_dy)
        tip = (cx + sign * (r + 67), cy - 39 - arm_dy)
        _line(draw, [start, elbow, tip], fill=palette["arm"], width=7)
        _line(draw, [tip, (tip[0] + sign * 13, tip[1] - 9)], fill=palette["arm"], width=5)

    # Large outlined, explicitly separated white regions are the counted evidence.
    for cx, cy, r in circles:
        _ellipse(draw, (cx - r, cy - r, cx + r, cy + r), fill=SNOW, outline=palette["outline"], width=4)

    # Buttons occupy interior pixels but cannot create or merge a white component.
    for cx, cy, r in circles[:-1]:
        if r >= 34:
            _ellipse(draw, (cx - 4, cy - 6, cx + 4, cy + 2), fill=palette["outline"])
            _ellipse(draw, (cx - 4, cy + 10, cx + 4, cy + 18), fill=palette["outline"])

    # Face and a tilted hat make the familiar-object prior visually explicit.
    hx, hy, hr = circles[-1]
    _ellipse(draw, (hx - hr * .34 - 4, hy - hr * .22 - 4, hx - hr * .34 + 4, hy - hr * .22 + 4), fill=palette["outline"])
    _ellipse(draw, (hx + hr * .34 - 4, hy - hr * .22 - 4, hx + hr * .34 + 4, hy - hr * .22 + 4), fill=palette["outline"])
    _polygon(draw, [(hx, hy), (hx + 20, hy + 5), (hx, hy + 8)], fill=palette["accent"], outline=palette["outline"], width=1)
    _line(draw, [(hx - 12, hy + 18), (hx, hy + 22), (hx + 12, hy + 18)], fill=palette["outline"], width=2)
    hat_y = hy - hr - 4
    dx = 5 * tilt
    _polygon(draw, [(hx - 42 + dx, hat_y), (hx + 42 + dx, hat_y), (hx + 36 + dx, hat_y + 10), (hx - 38 + dx, hat_y + 10)], fill=palette["outline"])
    _polygon(draw, [(hx - 27 + dx, hat_y - 22), (hx + 27 + dx, hat_y - 22 + 4 * tilt), (hx + 31 + dx, hat_y), (hx - 30 + dx, hat_y)], fill=palette["outline"])
    draw.rectangle((_S(hx - 28 + dx), _S(hat_y - 9), _S(hx + 29 + dx), _S(hat_y)), fill=palette["accent"])

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["section_count"]))


def margin(scene: dict) -> float:
    circles = _layout(scene)
    min_radius = min(circle[2] for circle in circles)
    gaps = [circles[i][1] - circles[i][2] - (circles[i + 1][1] + circles[i + 1][2]) for i in range(len(circles) - 1)]
    return float(min([min_radius - 26.0, *gaps]))


def is_quarantined(scene: dict) -> bool:
    circles = _layout(scene)
    gaps = [circles[i][1] - circles[i][2] - (circles[i + 1][1] + circles[i + 1][2]) for i in range(len(circles) - 1)]
    return min(gaps) < 9.0 or min(circle[2] for circle in circles) < 27.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
