"""Pixel-only inverse arm for the visible snow-section count."""

from __future__ import annotations

from functools import lru_cache

import numpy as np
from PIL import Image


@lru_cache(maxsize=256)
def _decision_from_rgb(size: tuple[int, int], pixels: bytes) -> str:
    arr = np.frombuffer(pixels, dtype=np.uint8).reshape(size[1], size[0], 3)
    # White snow is near-neutral and bright; all skies/ground/accessories are
    # deliberately well outside this public visual class. Antialiased rims do not
    # affect the solid interiors.
    white = (arr.min(axis=2) >= 235) & ((arr.max(axis=2) - arr.min(axis=2)) <= 14)
    # The public contract locates the vertical stack between ground and hat. The
    # renderer keeps that structural stack in the broad central strip while all
    # small decorative flakes remain in the side margins.
    row_ink = white[:, 150:362].sum(axis=1)
    active = row_ink >= 24
    bands: list[tuple[int, int]] = []
    start = None
    for y, value in enumerate(active.tolist() + [False]):
        if value and start is None:
            start = y
        elif not value and start is not None:
            if y - start >= 20:
                bands.append((start, y))
            start = None
    if len(bands) not in (2, 3, 4, 5):
        raise ValueError(f"ambiguous large-white-section count: {len(bands)}")
    # Each accepted band must show a broad circular interior near its midline.
    if any(int(row_ink[(a + b) // 2]) < 28 for a, b in bands):
        raise ValueError("a candidate section lacks a broad white interior")
    return str(len(bands))


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    return _decision_from_rgb(rgb.size, rgb.tobytes())
