"""Render two compact binary matrices whose GF(2) product is a glyph."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

N = 5
CANVAS = 512
FRAME = 3
PANEL_OFFSET = 64
BACKGROUND = (250, 249, 246)
TEAL = (17, 151, 162)
CORAL = (224, 91, 71)
PALE = (239, 241, 235)
GRID = (188, 191, 185)
INK = (31, 36, 49)


def _template(name: str) -> list[list[int]]:
    if name == "X":
        return [[int(r == c or r + c == N - 1) for c in range(N)] for r in range(N)]
    if name == "plus":
        return [[int(r == 2 or c == 2) for c in range(N)] for r in range(N)]
    if name == "T":
        return [[int(r == 0 or c == 2) for c in range(N)] for r in range(N)]
    if name == "L":
        return [[int(c == 0 or r == N - 1) for c in range(N)] for r in range(N)]
    raise ValueError(name)


TEMPLATE_NAMES = ("X", "plus", "T", "L")
TEMPLATES = {name: _template(name) for name in TEMPLATE_NAMES}


def _inverse_mod2(matrix: list[list[int]]) -> list[list[int]] | None:
    aug = [row[:] + [int(r == c) for c in range(N)] for r, row in enumerate(matrix)]
    for col in range(N):
        pivot = next((r for r in range(col, N) if aug[r][col]), None)
        if pivot is None:
            return None
        aug[col], aug[pivot] = aug[pivot], aug[col]
        for r in range(N):
            if r != col and aug[r][col]:
                aug[r] = [a ^ b for a, b in zip(aug[r], aug[col])]
    return [row[N:] for row in aug]


def _multiply_mod2(a: list[list[int]], b: list[list[int]]) -> list[list[int]]:
    return [
        [sum(a[r][k] * b[k][c] for k in range(N)) % 2 for c in range(N)]
        for r in range(N)
    ]


def _factor_for(target: list[list[int]], rng: random.Random) -> tuple[list[list[int]], list[list[int]]]:
    # Fix both panel ink counts at 12 for every class, eliminating count and total-ink cues.
    for _ in range(200_000):
        positions = rng.sample(range(N * N), 12)
        a = [[int(r * N + c in positions) for c in range(N)] for r in range(N)]
        inv = _inverse_mod2(a)
        if inv is None:
            continue
        b = _multiply_mod2(inv, target)
        if sum(map(sum, b)) == 12:
            return a, b
    raise RuntimeError("could not sample balanced invertible factor")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    name = TEMPLATE_NAMES[int(seed) % len(TEMPLATE_NAMES)]
    a, b = _factor_for(TEMPLATES[name], rng)
    cell = 5
    panel = 2 * FRAME + N * cell
    # The public 8x8 probe expands each 64px tile by a 16px halo. A 31px panel
    # fits in the exclusive core [cell+16, cell+48). Put the pair in diagonally
    # adjacent cores: the impact cells are 4-neighbor-disconnected but jointly focal.
    block_x = rng.randint(0, 5)
    block_y = rng.randint(0, 5)
    origin_x = block_x * 64 + 16
    origin_y = block_y * 64 + 16
    return {
        "matrix_a": a,
        "matrix_b": b,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "swapped": bool(rng.randrange(2)),
        "cell_size": cell,
    }


def _validate_matrix(value: Iterable[Iterable[int]]) -> list[list[int]]:
    matrix = [[int(v) for v in row] for row in value]
    if len(matrix) != N or any(len(row) != N for row in matrix):
        raise ValueError("matrix must be 5 by 5")
    if any(v not in (0, 1) for row in matrix for v in row):
        raise ValueError("matrix cells must be binary")
    return matrix


def _draw_panel(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int,
                matrix: list[list[int]], frame_color: tuple[int, int, int]) -> None:
    size = 2 * FRAME + N * cell
    draw.rectangle((x, y, x + size - 1, y + size - 1), fill=frame_color)
    inner = (x + FRAME, y + FRAME, x + size - FRAME - 1, y + size - FRAME - 1)
    draw.rectangle(inner, fill=PALE)
    for i in range(1, N):
        xx = x + FRAME + i * cell
        yy = y + FRAME + i * cell
        draw.line((xx, y + FRAME, xx, y + FRAME + N * cell - 1), fill=GRID, width=1)
        draw.line((x + FRAME, yy, x + FRAME + N * cell - 1, yy), fill=GRID, width=1)
    for r in range(N):
        for c in range(N):
            if matrix[r][c]:
                x0 = x + FRAME + c * cell + 1
                y0 = y + FRAME + r * cell + 1
                draw.rectangle((x0, y0, x0 + cell - 3, y0 + cell - 3), fill=INK)


def render(scene: dict) -> Image.Image:
    a = _validate_matrix(scene["matrix_a"])
    b = _validate_matrix(scene["matrix_b"])
    x = int(scene["origin_x"])
    y = int(scene["origin_y"])
    swapped = bool(scene["swapped"])
    cell = int(scene["cell_size"])
    if cell != 5:
        raise ValueError("unsupported cell size")
    panel = 2 * FRAME + N * cell
    first = (x, y)
    second = (x + PANEL_OFFSET, y + PANEL_OFFSET)
    (ax, ay), (bx, by) = (second, first) if swapped else (first, second)
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_panel(draw, ax, ay, cell, a, TEAL)
    _draw_panel(draw, bx, by, cell, b, CORAL)
    return image


def _distances(product: list[list[int]]) -> dict[str, int]:
    return {
        name: sum(product[r][c] != grid[r][c] for r in range(N) for c in range(N))
        for name, grid in TEMPLATES.items()
    }


def analytic_gold(scene: dict) -> str:
    product = _multiply_mod2(scene["matrix_a"], scene["matrix_b"])
    ranked = sorted(_distances(product).items(), key=lambda item: (item[1], item[0]))
    return ranked[0][0]


def margin(scene: dict) -> float:
    product = _multiply_mod2(scene["matrix_a"], scene["matrix_b"])
    ranked = sorted(_distances(product).values())
    return float(ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    product = _multiply_mod2(scene["matrix_a"], scene["matrix_b"])
    ranked = sorted(_distances(product).values())
    return ranked[0] == ranked[1]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
