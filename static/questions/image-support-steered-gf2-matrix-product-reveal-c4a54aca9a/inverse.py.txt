"""Independent image-only inverse for the odd-overlap matrix product."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 5
FRAME = 3
TEAL = np.array((17, 151, 162), dtype=np.uint8)
CORAL = np.array((224, 91, 71), dtype=np.uint8)
INK = np.array((31, 36, 49), dtype=np.uint8)


def _template(name: str) -> np.ndarray:
    if name == "X":
        return np.array([[r == c or r + c == 4 for c in range(5)] for r in range(5)])
    if name == "plus":
        return np.array([[r == 2 or c == 2 for c in range(5)] for r in range(5)])
    if name == "T":
        return np.array([[r == 0 or c == 2 for c in range(5)] for r in range(5)])
    if name == "L":
        return np.array([[c == 0 or r == 4 for c in range(5)] for r in range(5)])
    raise ValueError(name)


TEMPLATES = {name: _template(name) for name in ("X", "plus", "T", "L")}


def _read_panel(arr: np.ndarray, color: np.ndarray) -> np.ndarray:
    mask = np.all(arr == color, axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 200:
        raise ValueError("missing or damaged colored frame")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if x1 - x0 != y1 - y0:
        raise ValueError("frame is not square")
    size = x1 - x0 + 1
    if (size - 2 * FRAME) % N:
        raise ValueError("frame does not contain a 5 by 5 grid")
    cell = (size - 2 * FRAME) // N
    if cell != 5:
        raise ValueError("unsupported grid scale")
    out = np.zeros((N, N), dtype=np.uint8)
    for r in range(N):
        for c in range(N):
            cx = x0 + FRAME + c * cell + cell // 2
            cy = y0 + FRAME + r * cell + cell // 2
            patch = arr[cy - 1:cy + 2, cx - 1:cx + 2]
            ink_votes = int(np.all(patch == INK, axis=2).sum())
            if ink_votes >= 5:
                out[r, c] = 1
            elif ink_votes > 0:
                raise ValueError("ambiguous damaged cell")
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    a = _read_panel(arr, TEAL)
    b = _read_panel(arr, CORAL)
    product = (a @ b) % 2
    ranked = sorted(
        ((int(np.count_nonzero(product != grid)), name) for name, grid in TEMPLATES.items()),
        key=lambda item: (item[0], item[1]),
    )
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("ambiguous product glyph")
    return ranked[0][1]
