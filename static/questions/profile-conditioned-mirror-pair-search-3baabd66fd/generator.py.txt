"""Deterministic scenes and Pillow raster for mirror-correspondence search."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")

# (ink, red axis, background, card).  The independent oracle uses broad colour
# families rather than importing or copying these exact tuples.
THEMES = (
    ((30, 62, 93), (205, 55, 62), (248, 247, 243), (237, 241, 243)),
    ((63, 50, 104), (190, 63, 48), (249, 247, 244), (241, 238, 244)),
    ((22, 92, 77), (194, 52, 94), (248, 248, 244), (237, 243, 240)),
    ((105, 48, 78), (184, 68, 42), (249, 247, 242), (244, 238, 240)),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _validate_pattern(pattern: list[list[int]] | tuple[tuple[int, int], ...]) -> None:
    if len(pattern) != 5:
        raise ValueError("each panel must contain five printed rows")
    for row in pattern:
        if len(row) != 2 or any(isinstance(value, bool) or not isinstance(value, int) for value in row):
            raise ValueError("each row must declare integer left/right distance columns")
        if any(value not in (0, 1, 2) for value in row):
            raise ValueError("distance columns must be 0, 1, or 2")


def _pair_score(pattern: list[list[int]] | tuple[tuple[int, int], ...]) -> int:
    """Number of squares with an equal-distance partner on the opposite side."""
    _validate_pattern(pattern)
    return 2 * sum(int(left == right) for left, right in pattern)


def _pattern(left: tuple[int, ...], right: tuple[int, ...]) -> list[list[int]]:
    return [[int(a), int(b)] for a, b in zip(left, right, strict=True)]


def sample_scene(seed: int) -> dict:
    """Sample a fully cue-balanced scene; consecutive seeds cycle A/B/C/D."""
    value = int(seed)
    rng = random.Random(value ^ 0x4D495252)
    target_index = value % 4

    left_values = [0, 0, 1, 1, 2]
    rng.shuffle(left_values)
    left = tuple(left_values)
    alternatives = sorted(set(itertools.permutations(left)))
    by_mismatch: dict[int, list[tuple[int, ...]]] = {}
    for alternative in alternatives:
        mismatches = sum(a != b for a, b in zip(left, alternative, strict=True))
        if mismatches >= 2:
            by_mismatch.setdefault(mismatches, []).append(alternative)

    near = rng.choice(by_mismatch[2])
    remaining = [item for item in alternatives if item != left and item != near]
    medium_pool = [
        item for item in remaining
        if sum(a != b for a, b in zip(left, item, strict=True)) == 4
    ]
    medium = rng.choice(medium_pool or remaining)
    remaining = [item for item in remaining if item != medium]
    far_mismatch = max(
        sum(a != b for a, b in zip(left, item, strict=True)) for item in remaining
    )
    far_pool = [
        item for item in remaining
        if sum(a != b for a, b in zip(left, item, strict=True)) == far_mismatch
    ]
    far = rng.choice(far_pool)
    distractors = [_pattern(left, item) for item in (near, medium, far)]
    rng.shuffle(distractors)
    target = _pattern(left, left)
    patterns: list[list[list[int]]] = []
    for index in range(4):
        patterns.append(target if index == target_index else distractors.pop())

    ink, axis, background, card = THEMES[(value // 4) % len(THEMES)]
    scene = {
        "patterns": patterns,
        "ink_rgb": list(ink),
        "axis_rgb": list(axis),
        "background_rgb": list(background),
        "card_rgb": list(card),
        "cell_size": 20 + 2 * ((value // 16) % 3),
        "x_shift": rng.randint(-7, 7),
        "y_shift": rng.randint(-3, 3),
    }
    if margin(scene) < 4:
        raise RuntimeError("sampled distractor violates the four-square margin")
    return scene


def _draw_panel_pattern(
    draw: ImageDraw.ImageDraw,
    pattern: list[list[int]],
    center: tuple[int, int],
    cell_size: int,
    ink: tuple[int, int, int],
    axis: tuple[int, int, int],
    scale: int,
) -> None:
    _validate_pattern(pattern)
    cx, cy = center
    pitch = cell_size + 14
    row_offsets = (-2, -1, 0, 1, 2)
    guide = (190, 193, 193)
    for row_offset in row_offsets:
        y = cy + row_offset * pitch
        draw.line(
            ((cx - 140) * scale, y * scale, (cx + 140) * scale, y * scale),
            fill=guide,
            width=1 * scale,
        )
    draw.line(
        (cx * scale, (cy - 112) * scale, cx * scale, (cy + 112) * scale),
        fill=axis,
        width=5 * scale,
    )
    half = cell_size / 2.0
    for row_index, (left_column, right_column) in enumerate(pattern):
        y = cy + row_offsets[row_index] * pitch
        for sign, column in ((-1, left_column), (1, right_column)):
            x = cx + sign * (column + 1) * pitch
            box = (
                int(round((x - half) * scale)),
                int(round((y - half) * scale)),
                int(round((x + half) * scale)),
                int(round((y + half) * scale)),
            )
            draw.rounded_rectangle(box, radius=3 * scale, fill=ink)


def render(scene: dict) -> Image.Image:
    """Render four labeled panels, each with a visible vertical reflection axis."""
    scale = 3
    width, height = 920, 720
    patterns = scene["patterns"]
    ink = tuple(scene["ink_rgb"])
    axis = tuple(scene["axis_rgb"])
    background = tuple(scene["background_rgb"])
    card = tuple(scene["card_rgb"])
    cell_size = int(scene["cell_size"])
    x_shift = int(scene["x_shift"])
    y_shift = int(scene["y_shift"])
    if len(patterns) != 4:
        raise ValueError("scene must contain four panel patterns")

    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    neutral = (48, 50, 52)
    rule = (139, 142, 142)

    draw.text(
        (width * scale // 2, 27 * scale),
        "FIND THE PERFECT MIRROR PAIRING",
        fill=neutral,
        font=_font(25 * scale),
        anchor="mm",
    )
    draw.text(
        (width * scale // 2, 58 * scale),
        "MATCH EACH ROW AT THE SAME DISTANCE ACROSS THE RED LINE",
        fill=neutral,
        font=_font(14 * scale),
        anchor="mm",
    )
    draw.line((38 * scale, 78 * scale, (width - 38) * scale, 78 * scale), fill=rule, width=2 * scale)

    boxes = (
        (25, 92, 450, 382),
        (470, 92, 895, 382),
        (25, 405, 450, 695),
        (470, 405, 895, 695),
    )
    centers = ((237, 242), (682, 242), (237, 555), (682, 555))
    for index, (pattern, box, center) in enumerate(zip(patterns, boxes, centers, strict=True)):
        shifted_box = tuple(
            value + (x_shift if position % 2 == 0 else y_shift)
            for position, value in enumerate(box)
        )
        draw.rounded_rectangle(
            tuple(value * scale for value in shifted_box),
            radius=14 * scale,
            fill=card,
            outline=rule,
            width=2 * scale,
        )
        label_x = box[0] + 29 + x_shift
        label_y = box[1] + 27 + y_shift
        draw.ellipse(
            ((label_x - 18) * scale, (label_y - 18) * scale,
             (label_x + 18) * scale, (label_y + 18) * scale),
            fill=(255, 255, 255),
            outline=neutral,
            width=2 * scale,
        )
        draw.text(
            (label_x * scale, label_y * scale),
            LABELS[index],
            fill=neutral,
            font=_font(20 * scale),
            anchor="mm",
        )
        shifted_center = (center[0] + x_shift, center[1] + y_shift)
        _draw_panel_pattern(draw, pattern, shifted_center, cell_size, ink, axis, scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = [_pair_score(pattern) for pattern in scene["patterns"]]
    winners = [index for index, score in enumerate(scores) if score == 10]
    if len(winners) != 1:
        raise ValueError("scene lacks one perfectly mirror-paired panel")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted((_pair_score(pattern) for pattern in scene["patterns"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return int(scene["cell_size"]) < 18 or margin(scene) < 4.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
