"""Deterministic renderer and analytic label for cyclic landmark correspondence."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "cyclic_landmark_correspondence-0.1.0"
CANVAS_SIZE = (1120, 560)
SUPERSAMPLE = 4
MARKER_RADIUS = 18.0
N_MARKERS = 5

PANEL_BOXES = {
    "reference": (24, 58, 344, 502),
    "A": (400, 58, 720, 502),
    "B": (776, 58, 1096, 502),
}
PANEL_CENTERS = {
    "reference": (184.0, 292.0),
    "A": (560.0, 292.0),
    "B": (936.0, 292.0),
}

PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
}
COLORS = tuple(PALETTE)
RADIUS_CHOICES = (88.0, 102.0, 116.0)


def _ring_parts(order: Iterable[str], phase: float, radius: float) -> list[dict]:
    """Make a visible regular ring; positive screen-space angles run clockwise."""

    parts = []
    for index, color in enumerate(order):
        angle = phase + (2.0 * math.pi * index / N_MARKERS)
        parts.append(
            {
                "x": round(radius * math.cos(angle), 4),
                "y": round(radius * math.sin(angle), 4),
                "color": str(color),
            }
        )
    return parts


def _sequence(parts: list[dict]) -> tuple[str, ...]:
    cx = sum(float(part["x"]) for part in parts) / len(parts)
    cy = sum(float(part["y"]) for part in parts) / len(parts)
    ordered = sorted(
        parts,
        key=lambda part: math.atan2(float(part["y"]) - cy, float(part["x"]) - cx),
    )
    return tuple(str(part["color"]) for part in ordered)


def _canonical(sequence: tuple[str, ...]) -> tuple[str, ...]:
    return min(sequence[index:] + sequence[:index] for index in range(len(sequence)))


def _cycle_edges(sequence: tuple[str, ...]) -> frozenset[tuple[str, str]]:
    return frozenset(
        (sequence[index], sequence[(index + 1) % len(sequence)])
        for index in range(len(sequence))
    )


def _cycle_score(reference: tuple[str, ...], candidate: tuple[str, ...]) -> int:
    return len(_cycle_edges(reference) & _cycle_edges(candidate))


def _valid_ring(parts: object) -> bool:
    if not isinstance(parts, list) or len(parts) != N_MARKERS:
        return False
    colors: list[str] = []
    coords: list[tuple[float, float]] = []
    for part in parts:
        if not isinstance(part, dict) or set(part) != {"x", "y", "color"}:
            return False
        try:
            x = float(part["x"])
            y = float(part["y"])
            color = str(part["color"])
        except (TypeError, ValueError):
            return False
        if not math.isfinite(x) or not math.isfinite(y) or color not in PALETTE:
            return False
        coords.append((x, y))
        colors.append(color)
    if len(set(colors)) != N_MARKERS:
        return False
    if len(set(coords)) != N_MARKERS:
        return False
    radii = [math.hypot(x, y) for x, y in coords]
    return min(radii) >= 70.0 and max(radii) <= 130.0 and max(radii) - min(radii) <= 1.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene whose correct candidate is A on even seeds."""

    rng = random.Random(int(seed))
    reference_order = list(COLORS)
    rng.shuffle(reference_order)
    radius = rng.choice(RADIUS_CHOICES)
    reference = _ring_parts(reference_order, rng.uniform(-math.pi, math.pi), radius)

    wrong_order = list(reference_order)
    first, second = rng.sample(range(N_MARKERS), 2)
    wrong_order[first], wrong_order[second] = wrong_order[second], wrong_order[first]
    correct = _ring_parts(reference_order, rng.uniform(-math.pi, math.pi), radius)
    wrong = _ring_parts(wrong_order, rng.uniform(-math.pi, math.pi), radius)

    correct_is_a = int(seed) % 2 == 0
    candidate_a, candidate_b = (correct, wrong) if correct_is_a else (wrong, correct)
    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
    }


def _draw_ring(draw: ImageDraw.ImageDraw, parts: list[dict], center: tuple[float, float]) -> None:
    cx, cy = center
    radius = sum(math.hypot(float(part["x"]), float(part["y"])) for part in parts) / len(parts)
    scale = SUPERSAMPLE
    ring_box = (
        (cx - radius) * scale,
        (cy - radius) * scale,
        (cx + radius) * scale,
        (cy + radius) * scale,
    )
    draw.ellipse(ring_box, outline=(105, 110, 121), width=3 * scale)
    for part in parts:
        px = cx + float(part["x"])
        py = cy + float(part["y"])
        marker_box = (
            (px - MARKER_RADIUS) * scale,
            (py - MARKER_RADIUS) * scale,
            (px + MARKER_RADIUS) * scale,
            (py + MARKER_RADIUS) * scale,
        )
        draw.ellipse(marker_box, fill=PALETTE[str(part["color"])], outline=(32, 35, 43), width=2 * scale)


def render(scene: dict) -> Image.Image:
    """Render only the visible ring landmark lists and fixed panel framing."""

    width, height = CANVAS_SIZE
    work = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), (255, 255, 255))
    draw = ImageDraw.Draw(work)
    for box in PANEL_BOXES.values():
        scaled = tuple(int(value * SUPERSAMPLE) for value in box)
        draw.rounded_rectangle(scaled, radius=12 * SUPERSAMPLE, fill=(249, 250, 252), outline=(75, 80, 92), width=2 * SUPERSAMPLE)
    _draw_ring(draw, scene["reference"], PANEL_CENTERS["reference"])
    _draw_ring(draw, scene["candidate_a"], PANEL_CENTERS["A"])
    _draw_ring(draw, scene["candidate_b"], PANEL_CENTERS["B"])
    image = work.resize(CANVAS_SIZE, Image.Resampling.LANCZOS)

    # Draw labels after reduction so the public contract remains legible while
    # the color masks retain large, exact-color cores for the independent arm.
    final_draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    labels = {"reference": "REFERENCE", "A": "A", "B": "B"}
    for name, box in PANEL_BOXES.items():
        final_draw.text((box[0] + 14, box[1] + 13), labels[name], fill=(25, 28, 35), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate with the same directed cycle up to rotation."""

    reference = _canonical(_sequence(scene["reference"]))
    candidate_a = _canonical(_sequence(scene["candidate_a"]))
    candidate_b = _canonical(_sequence(scene["candidate_b"]))
    matches = (candidate_a == reference, candidate_b == reference)
    if sum(matches) != 1:
        raise ValueError("scene has no unique cyclic-order correspondence")
    return "A" if matches[0] else "B"


def margin(scene: dict) -> float:
    """Return the directed cycle score gap between the two candidates."""

    reference = _sequence(scene["reference"])
    score_a = _cycle_score(reference, _sequence(scene["candidate_a"]))
    score_b = _cycle_score(reference, _sequence(scene["candidate_b"]))
    return float(abs(score_a - score_b))


def is_quarantined(scene: dict) -> bool:
    """Quarantine malformed or tied finite-resolution assignments."""

    try:
        if not all(_valid_ring(scene[key]) for key in ("reference", "candidate_a", "candidate_b")):
            return True
        reference = _canonical(_sequence(scene["reference"]))
        a = _canonical(_sequence(scene["candidate_a"]))
        b = _canonical(_sequence(scene["candidate_b"]))
        scores = (
            _cycle_score(reference, a),
            _cycle_score(reference, b),
        )
        matches = (a == reference, b == reference)
        return sum(matches) != 1 or scores[0] == scores[1]
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden list-order alias is declared; every label input is rendered."""

    return []
