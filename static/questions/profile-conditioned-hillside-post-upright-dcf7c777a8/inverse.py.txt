"""Independent pixel-only inverse arm.

Receives the final PNG and nothing else: no latent parameters, no renderer, no
prompt code, no analytic gold, no filenames and no metadata.  It segments the
two chromatic inks, measures the long axis of the smaller one (the post) and
the top boundary of the larger one (the ground), and reports which of the two
reference directions painted on the page -- the page's vertical side edges, or
the normal of the ground boundary -- the post agrees with.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "hillside_post_upright-oracle-0.1.0"
ABSTAIN = "abstain"

_CHROMA_MIN = 40        # separates ink from white paper and grey decoration
_MIN_CLASS_PX = 600     # a colour class smaller than this is not a painted body
_MIN_POST_PX = 1200
_MIN_GROUND_PX = 8000
_MIN_ELONGATION = 4.0   # the post must read as a long straight bar
_MIN_SEPARATION = 6.0   # degrees between the two reference directions
_MAX_MISFIT = 8.0       # degrees the bar may sit from its nearest reference
_REL_DECISION = 0.35    # required share of the separation between the two fits


def _ang_dist(a: float, b: float) -> float:
    d = abs(a - b) % 180.0
    return min(d, 180.0 - d)


def _classes(arr: np.ndarray) -> list[tuple[int, np.ndarray]]:
    chroma = arr.max(axis=-1).astype(np.int16) - arr.min(axis=-1).astype(np.int16)
    ink = chroma >= _CHROMA_MIN
    if not ink.any():
        return []
    flat = arr.reshape(-1, 3).astype(np.int64)
    key = (flat[:, 0] << 16) | (flat[:, 1] << 8) | flat[:, 2]
    key = np.where(ink.reshape(-1), key, -1)
    vals, counts = np.unique(key[key >= 0], return_counts=True)
    out = []
    for v, c in zip(vals, counts):
        if c >= _MIN_CLASS_PX:
            out.append((int(c), (key.reshape(arr.shape[:2]) == v)))
    out.sort(key=lambda t: -t[0])
    return out


def _axis_deg(mask: np.ndarray) -> tuple[float, float]:
    """Principal-axis direction (deg from +x, mod 180) and elongation ratio."""
    ys, xs = np.nonzero(mask)
    pts = np.stack([xs.astype(np.float64), (-ys).astype(np.float64)], axis=1)
    pts -= pts.mean(axis=0)
    cov = (pts.T @ pts) / max(1, pts.shape[0] - 1)
    evals, evecs = np.linalg.eigh(cov)
    major = evecs[:, int(np.argmax(evals))]
    lo, hi = float(min(evals)), float(max(evals))
    elong = float(np.sqrt(hi / lo)) if lo > 1e-9 else float("inf")
    return float(np.degrees(np.arctan2(major[1], major[0])) % 180.0), elong


def _boundary_deg(arr: np.ndarray, ground: np.ndarray) -> float | None:
    """Direction of the free boundary between the ground body and bare paper.

    Orientation-agnostic: it uses the ground pixels that touch white paper,
    wherever on the page that interface happens to lie.
    """
    paper = (arr >= 245).all(axis=-1)
    near = np.zeros_like(paper)
    near[1:, :] |= paper[:-1, :]
    near[:-1, :] |= paper[1:, :]
    near[:, 1:] |= paper[:, :-1]
    near[:, :-1] |= paper[:, 1:]
    edge = ground & near
    if int(edge.sum()) < 150:
        return None
    deg, elong = _axis_deg(edge)
    if elong < 8.0:
        return None
    return deg


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3:
        return ABSTAIN
    classes = _classes(arr)
    if len(classes) < 2:
        return ABSTAIN
    ground_px, ground = classes[0]
    post_px, post = classes[1]
    if ground_px < _MIN_GROUND_PX or post_px < _MIN_POST_PX:
        return ABSTAIN

    post_deg, elong = _axis_deg(post)
    if elong < _MIN_ELONGATION:
        return ABSTAIN
    edge_deg = _boundary_deg(arr, ground)
    if edge_deg is None:
        return ABSTAIN

    upright_deg = 90.0
    square_deg = (edge_deg + 90.0) % 180.0
    separation = _ang_dist(upright_deg, square_deg)
    if separation < _MIN_SEPARATION:
        return ABSTAIN

    d_up = _ang_dist(post_deg, upright_deg)
    d_sq = _ang_dist(post_deg, square_deg)
    if min(d_up, d_sq) > _MAX_MISFIT:
        return ABSTAIN
    if abs(d_up - d_sq) < _REL_DECISION * separation:
        return ABSTAIN
    return "upright" if d_up < d_sq else "square"
