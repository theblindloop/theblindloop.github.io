"""Scene sampler, renderer and analytic gold for hillside_post_upright.

A straight post is planted on a drawn hillside.  The post is always painted
exactly along one of two reference directions that are themselves drawn on the
page: the page's own side edges (``upright``) or the normal of the sloping
ground line (``square``).  The analytic gold is which of those two drawn
reference frames the painted post agrees with, and the margin is the visible
angular separation of the two frames, i.e. the slope angle of the ground.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "hillside_post_upright-renderer-0.1.0"

WIDTH, HEIGHT = 640, 520
Y_MID = 330                      # ground line height at the horizontal centre
FRAME_INSET = 14
PAPER = (255, 255, 255)
FRAME = (168, 168, 168)
STONE = (150, 150, 150)

MODES = ("upright", "square")
CANONICAL_MODE = "upright"

# Slope magnitudes in degrees.  The list is fixed so that every deterministic
# sweep contains both quarantined (< 12 deg) and off-quarantine scenes.
THETAS = (8, 9, 10, 11, 13, 14, 16, 18, 20, 22, 24, 26)
QUARANTINE_DEG = 12.0

PX_MIN, PX_MAX = 130, 510
POST_LEN_MIN, POST_LEN_MAX = 130, 235
POST_THICK_MIN, POST_THICK_MAX = 14, 22
BURY = 14                        # painted length of the post below the ground line

GROUND_COLORS = (
    (86, 132, 62), (58, 118, 92), (132, 106, 48),
    (104, 128, 54), (70, 122, 130), (140, 96, 60),
)
POST_COLORS = (
    (176, 58, 62), (62, 66, 176), (150, 62, 156),
    (196, 108, 34), (44, 108, 168), (168, 52, 116),
)


def _chroma(c: tuple[int, int, int]) -> int:
    return max(c) - min(c)


def _dist(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return math.sqrt(sum((int(x) - int(y)) ** 2 for x, y in zip(a, b)))


def sample_scene(seed: int) -> dict:
    """Sample one scene.  Mode, slope, colours and geometry are independent."""
    rng = random.Random(seed)
    # Exactly equifrequent modes over any run of consecutive scene seeds.
    mode = MODES[seed % 2]
    theta_mag = THETAS[(seed // 2) % len(THETAS)]
    sign = 1 if rng.random() < 0.5 else -1
    slope_deg = float(sign * theta_mag)

    px = rng.randint(PX_MIN, PX_MAX)
    base_y = Y_MID + math.tan(math.radians(slope_deg)) * (px - WIDTH / 2.0)
    head_room = base_y - 46.0
    post_len = rng.randint(POST_LEN_MIN, min(POST_LEN_MAX, int(head_room)))
    post_thick = rng.randint(POST_THICK_MIN, POST_THICK_MAX)

    ground = GROUND_COLORS[rng.randrange(len(GROUND_COLORS))]
    post = POST_COLORS[rng.randrange(len(POST_COLORS))]
    while _dist(ground, post) < 110 or _chroma(post) < 45 or _chroma(ground) < 45:
        post = POST_COLORS[rng.randrange(len(POST_COLORS))]

    stones = [
        (rng.randint(40, WIDTH - 40), rng.randint(40, 130), rng.randint(7, 13))
        for _ in range(rng.randint(3, 5))
    ]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "mode": mode,
        "slope_deg": slope_deg,
        "ground_mid_y": float(Y_MID),
        "post_x": int(px),
        "post_len": int(post_len),
        "post_thick": int(post_thick),
        "ground_color": list(ground),
        "post_color": list(post),
        "stones": [list(s) for s in stones],
    }


def ground_y(scene: dict, x: float) -> float:
    t = math.tan(math.radians(float(scene["slope_deg"])))
    return float(scene["ground_mid_y"]) + t * (x - float(scene["width"]) / 2.0)


def post_direction_deg(scene: dict) -> float:
    """Direction of the painted post, in degrees from the +x axis, mod 180."""
    if scene["mode"] == "upright":
        return 90.0
    # The ground line runs at -slope_deg in a y-up frame, so its normal, the
    # direction a post square to the hillside is painted along, is 90 - slope.
    return (90.0 - float(scene["slope_deg"])) % 180.0


def _post_polygon(scene: dict) -> list[tuple[float, float]]:
    ang = math.radians(post_direction_deg(scene))
    ux, uy = math.cos(ang), -math.sin(ang)          # unit vector pointing up-page
    hx, hy = -uy, ux                                # unit vector across the post
    bx = float(scene["post_x"])
    by = ground_y(scene, bx)
    half = float(scene["post_thick"]) / 2.0
    top = (bx + ux * float(scene["post_len"]), by + uy * float(scene["post_len"]))
    bot = (bx - ux * BURY, by - uy * BURY)
    return [
        (top[0] + hx * half, top[1] + hy * half),
        (top[0] - hx * half, top[1] - hy * half),
        (bot[0] - hx * half, bot[1] - hy * half),
        (bot[0] + hx * half, bot[1] + hy * half),
    ]


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (w, h), PAPER)
    draw = ImageDraw.Draw(image)

    ground = tuple(int(v) for v in scene["ground_color"])
    draw.polygon(
        [(0, ground_y(scene, 0)), (w, ground_y(scene, w)), (w, h), (0, h)],
        fill=ground,
    )
    # Achromatic stones sit well inside the ground body, never on its top edge.
    for sx, depth, r in scene["stones"]:
        sy = ground_y(scene, sx) + depth
        draw.ellipse([sx - r, sy - r, sx + r, sy + r], fill=STONE)

    draw.polygon(_post_polygon(scene), fill=tuple(int(v) for v in scene["post_color"]))

    draw.rectangle(
        [FRAME_INSET, FRAME_INSET, w - 1 - FRAME_INSET, h - 1 - FRAME_INSET],
        outline=FRAME, width=3,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["mode"])


def margin(scene: dict) -> float:
    """Visible angular separation of the two drawn reference directions."""
    return float(abs(float(scene["slope_deg"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_DEG


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared: every field is painted into the raster."""
    return []


# ---------------------------------------------------------------- nuisances --

def translate(scene: dict, dx: int, dy: int) -> dict:
    alt = dict(scene)
    alt["post_x"] = int(min(PX_MAX, max(PX_MIN, int(scene["post_x"]) + dx)))
    alt["ground_mid_y"] = float(scene["ground_mid_y"]) + dy
    alt["stones"] = [[int(min(scene["width"] - 40, max(40, s[0] + dx))), s[1], s[2]]
                     for s in scene["stones"]]
    base_y = ground_y(alt, float(alt["post_x"]))
    alt["post_len"] = int(min(int(scene["post_len"]), max(POST_LEN_MIN, int(base_y - 46))))
    return alt


def recolor(scene: dict) -> dict:
    alt = dict(scene)
    gi = (GROUND_COLORS.index(tuple(scene["ground_color"])) + 2) % len(GROUND_COLORS)
    pi = (POST_COLORS.index(tuple(scene["post_color"])) + 3) % len(POST_COLORS)
    alt["ground_color"] = list(GROUND_COLORS[gi])
    alt["post_color"] = list(POST_COLORS[pi])
    if _dist(GROUND_COLORS[gi], POST_COLORS[pi]) < 110:
        alt["post_color"] = list(POST_COLORS[(pi + 1) % len(POST_COLORS)])
    return alt


def permute_decor(scene: dict, seed: int) -> dict:
    rng = random.Random(seed)
    alt = dict(scene)
    alt["stones"] = [
        [int(min(scene["width"] - 40, max(40, s[0] + rng.randint(-30, 30)))),
         int(min(130, max(40, s[1] + rng.randint(-20, 20)))), s[2]]
        for s in scene["stones"]
    ]
    return alt


def rescale(scene: dict) -> dict:
    alt = dict(scene)
    alt["post_len"] = int(max(POST_LEN_MIN, round(int(scene["post_len"]) * 0.82)))
    alt["post_thick"] = int(max(POST_THICK_MIN, round(int(scene["post_thick"]) * 0.85)))
    return alt


# ------------------------------------------------- declared label transform --

def set_mode(scene: dict, mode: str) -> dict:
    if mode not in MODES:
        raise ValueError(f"unknown mode: {mode}")
    alt = dict(scene)
    alt["mode"] = mode
    return alt


def flatten_slope(scene: dict) -> dict:
    """Collapse the margin: the two drawn reference directions coincide."""
    alt = dict(scene)
    alt["slope_deg"] = 0.0
    return alt


def ink_pixels(image: Image.Image) -> int:
    px = image.convert("RGB").getdata()
    return sum(1 for c in px if max(c) - min(c) >= 40)
