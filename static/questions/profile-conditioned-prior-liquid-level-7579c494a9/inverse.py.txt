"""Independent pixel-only inverse arm for the liquid-surface levelness world.

Receives ONLY the final PNG and returns whether the liquid surface is LEVEL
("yes") or TILTED ("no"). It shares no geometry with renderer.render or
analytic_gold; it detects the liquid region and the orientation of its top
(surface) edge directly from the raster.

Method: nearest-palette segmentation isolates the liquid colour. The surface is
the TOP edge of the liquid region: for every image column the uppermost liquid
pixel is the surface height at that column (the glass interior above the surface
is empty, so the topmost liquid pixel traces the liquid/air boundary). The
extreme columns where the surface meets the glass walls are trimmed, and a
least-squares line through the remaining column-top heights gives the surface
orientation. The surface is "level" exactly when its absolute tilt is at or
below the decision threshold.

Import discipline: this module imports only PIL and numpy. It never imports
renderer, prompts, generate, verify, or any analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_liquid_level-oracle-0.1.0"

_LIQUID = np.array((38, 118, 222), dtype=np.float32)
_DECISION_DEG = 9.0
_MIN_COLS = 10
_MIN_LIQUID_PX = 60


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    blue = np.linalg.norm(arr - _LIQUID, axis=-1) < 90.0
    rows, cols = np.nonzero(blue)
    if len(rows) < _MIN_LIQUID_PX:
        return "no"

    top: dict[int, int] = {}
    for y, x in zip(rows.tolist(), cols.tolist()):
        if x not in top or y < top[x]:
            top[x] = y
    xs = sorted(top)
    trim = len(xs) // 8
    section = xs[trim:len(xs) - trim] if len(xs) > 2 * trim else xs
    if len(section) < _MIN_COLS:
        return "no"
    X = np.array(section, dtype=np.float64)
    Y = np.array([float(top[x]) for x in section], dtype=np.float64)
    slope, intercept = np.linalg.lstsq(
        np.vstack([X, np.ones_like(X)]).T, Y, rcond=None
    )[0]
    tilt = abs(float(np.degrees(np.arctan(slope))))
    return "yes" if tilt <= _DECISION_DEG else "no"


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    blue = np.linalg.norm(arr - _LIQUID, axis=-1) < 90.0
    rows, cols = np.nonzero(blue)
    top: dict[int, int] = {}
    for y, x in zip(rows.tolist(), cols.tolist()):
        if x not in top or y < top[x]:
            top[x] = y
    xs = sorted(top)
    X = np.array(xs, dtype=np.float64)
    Y = np.array([float(top[x]) for x in xs], dtype=np.float64)
    slope = np.linalg.lstsq(np.vstack([X, np.ones_like(X)]).T, Y, rcond=None)[0][0]
    return {
        "liquid_px": int(blue.sum()),
        "surface_columns": len(xs),
        "surface_tilt_deg": round(float(np.degrees(np.arctan(slope))), 2),
        "size": image.size,
    }
