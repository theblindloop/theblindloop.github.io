"""Liquid surface levelness world under a contradictory gravity prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (whether the
liquid surface is LEVEL / horizontal) when a familiar canonical prior ("a liquid
at rest always has a horizontal, level surface") is contradicted by the actual
image, which in the counterfactual scenes shows the surface of the liquid tilted
with the glass even though the glass itself is upright.

Latent z = {level, slant_deg, fill, width, height}. ``level`` is a boolean: True
means the liquid surface is drawn HORIZONTAL (matches the gravity prior, the
control variant); False means the surface is drawn at the tilt ``slant_deg``
(counterfactual variant that contradicts the prior). ``fill`` sets how full the
glass is; it controls the vertical position of the surface so the surface line
always crosses the full glass width with a long, unambiguous chord. The closed
form gold is

    decision = "yes" if level else "no"

a scene parameter, not a pixel measurement.

The glass is drawn as a plain upright rectangle (a light-gray interior outlined
in dark with a rim and base) so the image unambiguously reads as a transparent
glass of liquid. The liquid fills the region of the glass interior whose (x, y)
are on the liquid side of the surface line. When ``level`` the surface line is
horizontal; otherwise it has gradient tan(slant_deg) from the left glass wall.

The continuous verifiability margin is the angular gap between the actual
surface tilt and the decision threshold. When a (counterfactual) slanted surface
is too close to horizontal the scene is quarantined because the surface angle
becomes pixel-ambiguous.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_liquid_level-0.1.0"

BG = (255, 255, 255)
INTERIOR = (232, 236, 240)   # empty (air) part of the glass
LIQUID = (38, 118, 222)      # liquid colour
OUTLINE = (55, 58, 64)       # glass rim / walls / base

CANVAS = (512, 512)

# Upright glass rectangle (image coordinates).
GX0, GX1 = 176, 336
GY0, GY1 = 140, 420

DECISION_DEG = 9.0     # |surface tilt| above which the surface is "not level"
QUAR_MARGIN_DEG = 4.0
FILL_MIN, FILL_MAX = 0.40, 0.75


def water_y(scene: dict) -> float:
    """Image y of the liquid surface is the TOP edge of the liquid region."""
    return GY0 + (GY1 - GY0) * (1.0 - scene["fill"])


def margin(scene: dict) -> float:
    """Angular gap (deg) between the surface tilt and the decision threshold."""
    tilt = abs(float(scene["slant_deg"]))
    return abs(tilt - DECISION_DEG)


def is_quarantined(scene: dict) -> bool:
    fill = float(scene["fill"])
    if not (FILL_MIN <= fill <= FILL_MAX):
        return True
    return margin(scene) < QUAR_MARGIN_DEG


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        level = rng.random() < 0.5
        fill = rng.uniform(0.45, 0.7)
        if level:
            slant = 0.0
        else:
            slant = rng.uniform(15.0, 28.0)
            if rng.random() < 0.5:
                slant = -slant
        scene = {
            "level": level,
            "slant_deg": slant,
            "fill": fill,
            "width": CANVAS[0],
            "height": CANVAS[1],
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = scene["width"], scene["height"]
    img = np.full((h, w, 3), np.array(BG), dtype=np.uint8)

    img[GY0:GY1 + 1, GX0:GX1 + 1] = np.array(INTERIOR)

    yy, xx = np.mgrid[0:h, 0:w].astype(np.float64)
    inside_glass = (xx >= GX0) & (xx <= GX1) & (yy >= GY0) & (yy <= GY1)
    wy = water_y(scene)
    if scene["level"]:
        liquid = inside_glass & (yy >= wy)
    else:
        th = np.deg2rad(scene["slant_deg"])
        liquid = inside_glass & (yy >= wy + np.tan(th) * (xx - GX0))
    img[liquid] = np.array(LIQUID)

    im = Image.fromarray(img, "RGB")
    draw = ImageDraw.Draw(im)
    draw.rectangle([GX0, GY0, GX1, GY1], outline=OUTLINE, width=4)
    draw.line([GX0, GY0, GX1, GY0], fill=OUTLINE, width=6)          # rim
    draw.line([GX0 - 8, GY1, GX1 + 8, GY1], fill=OUTLINE, width=5)  # base
    return im


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["level"] else "no"


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no render-preserving latent alias exists.

    The liquid region's pixels are determined by ``level`` (surface horizontal)
    and, in the counterfactual, by ``slant_deg`` and ``fill``. Changing the label
    input ``level`` swaps the surface between horizontal and tilted, changing the
    raster; the surface tilt and fill also change the raster. The raster is
    therefore injective in {level, slant_deg, fill}. A declared transform that
    preserved the exact raster would have to keep the surface line identical,
    which fixes every scene variable. We declare no latent aliases and return an
    empty list.
    """
    return []
