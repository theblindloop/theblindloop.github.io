"""Image-only inverse arm: locate the dark frame and count right-side RGB runs."""

from __future__ import annotations

from collections import Counter
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _dark(pixel: tuple[int, int, int]) -> bool:
    return max(pixel) < 70


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    dark_x = []
    dark_y = []
    for yy in range(height):
        for xx in range(width):
            if _dark(rgb.getpixel((xx, yy))):
                dark_x.append(xx)
                dark_y.append(yy)
    if len(dark_x) < 500:
        raise OracleAbstention("no substantial dark rectangular frame")
    left, right = min(dark_x), max(dark_x)
    top, bottom = min(dark_y), max(dark_y)
    if right - left < 200 or bottom - top < 100:
        raise OracleAbstention("frame is too small")
    # Measure several columns in the rightmost quarter; the canton never reaches here.
    sample_xs = [left + int((right - left) * fraction) for fraction in (0.72, 0.78, 0.84, 0.90)]
    decisions = []
    for xx in sample_xs:
        values = [rgb.getpixel((xx, yy)) for yy in range(top + 5, bottom - 4)]
        common = [color for color, _ in Counter(values).most_common(4) if not _dark(color)]
        if len(common) < 2:
            continue
        c0, c1 = common[:2]
        labels = []
        for pixel in values:
            d0 = sum((pixel[k] - c0[k]) ** 2 for k in range(3))
            d1 = sum((pixel[k] - c1[k]) ** 2 for k in range(3))
            labels.append(0 if d0 <= d1 else 1)
        runs = 1 + sum(labels[i] != labels[i - 1] for i in range(1, len(labels)))
        min_run = min(
            b - a
            for a, b in _run_intervals(labels)
        )
        if min_run >= 7:
            decisions.append(runs)
    if not decisions or len(set(decisions)) != 1:
        raise OracleAbstention("right-edge scans disagree or are too thin")
    count = decisions[0]
    if count < 3 or count > 25:
        raise OracleAbstention("implausible stripe count")
    return str(count)


def _run_intervals(labels: list[int]) -> list[tuple[int, int]]:
    starts = [0]
    for i in range(1, len(labels)):
        if labels[i] != labels[i - 1]:
            starts.append(i)
    return list(zip(starts, starts[1:] + [len(labels)]))
