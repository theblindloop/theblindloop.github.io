"""Deterministic flag-stripe world. Analytic labels come only from stripe_count."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 420
COUNTS = (9, 11, 13, 15)
PALETTES = (
    ((183, 35, 48), (244, 239, 226), (30, 52, 94)),
    ((164, 42, 55), (239, 242, 232), (34, 61, 105)),
    ((194, 48, 58), (246, 238, 218), (29, 57, 112)),
    ((151, 31, 49), (235, 241, 235), (40, 56, 98)),
)
BACKGROUNDS = ((218, 225, 231), (231, 224, 211), (218, 230, 220), (226, 220, 232))
FRAME = (24, 27, 31)
STAR = (250, 249, 241)


def sample_scene(seed: int) -> dict:
    # Each block of four adjacent seeds is a matched quartet: identical nuisance
    # styling and one instance of every answer class, including the 13 control.
    rng = random.Random(int(seed) // len(COUNTS))
    count = COUNTS[int(seed) % len(COUNTS)]
    stripe_a, stripe_b, canton = PALETTES[rng.randrange(len(PALETTES))]
    flag_width = rng.randrange(420, 501, 8)
    flag_height = rng.randrange(220, 281, 4)
    flag_x = rng.randrange(42, WIDTH - flag_width - 41)
    flag_y = rng.randrange(52, HEIGHT - flag_height - 51)
    return {
        "stripe_count": count,
        "flag_x": flag_x,
        "flag_y": flag_y,
        "flag_width": flag_width,
        "flag_height": flag_height,
        "stripe_a": list(stripe_a),
        "stripe_b": list(stripe_b),
        "canton_color": list(canton),
        "background": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "canton_fraction": rng.choice((0.36, 0.40, 0.44)),
        "star_phase": rng.randrange(3),
    }


def _star_points(cx: float, cy: float, outer: float) -> list[tuple[float, float]]:
    pts = []
    for i in range(10):
        radius = outer if i % 2 == 0 else outer * 0.42
        angle = -math.pi / 2 + i * math.pi / 5
        pts.append((cx + radius * math.cos(angle), cy + radius * math.sin(angle)))
    return pts


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    x = int(scene["flag_x"])
    y = int(scene["flag_y"])
    w = int(scene["flag_width"])
    h = int(scene["flag_height"])
    n = int(scene["stripe_count"])
    a = tuple(scene["stripe_a"])
    b = tuple(scene["stripe_b"])
    # Integer partition covers every interior row exactly, with no hidden bands at the right edge.
    for i in range(n):
        y0 = y + 5 + (i * (h - 10)) // n
        y1 = y + 5 + ((i + 1) * (h - 10)) // n - 1
        draw.rectangle((x + 5, y0, x + w - 6, y1), fill=a if i % 2 == 0 else b)
    canton_w = int((w - 10) * float(scene["canton_fraction"]))
    canton_h = 5 + (7 * (h - 10)) // 13
    draw.rectangle((x + 5, y + 5, x + 4 + canton_w, y + canton_h), fill=tuple(scene["canton_color"]))
    phase = int(scene["star_phase"])
    rows, cols = 4, 6
    sx0, sy0 = x + 17, y + 16
    sx_step = max(16, (canton_w - 27) // cols)
    sy_step = max(16, (canton_h - 24) // rows)
    radius = max(3, min(6, sy_step // 4))
    for row in range(rows):
        for col in range(cols):
            if (row + col + phase) % 3 != 2:
                draw.polygon(_star_points(sx0 + col * sx_step, sy0 + row * sy_step, radius), fill=STAR)
    draw.rectangle((x, y, x + w - 1, y + h - 1), outline=FRAME, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["stripe_count"]))


def margin(scene: dict) -> float:
    return float((int(scene["flag_height"]) - 10) // int(scene["stripe_count"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
