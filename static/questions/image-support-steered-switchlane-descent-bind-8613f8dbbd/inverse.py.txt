"""Independent pixel-only inverse arm for the switch-lane descent world.

It receives the final PNG and the public question contract only: the six answer
colours, the slate ink of the ladder, the black pointer, and the rule that the
descent starts under the pointer, always moves downwards, and crosses every
switch it meets.  It never imports the renderer, generator, prompt module or
verifier, and it never sees latent scene data.  It abstains by raising.

Pipeline: segment the flat inks; find the black start pointer and the rail it
touches; then walk the ladder locally, descending pixel by pixel and crossing
every switch bar that meets the current rail, so the reading depends on the
whole traversed route being unbroken; only when the descent runs out of rail is
the colour of the bead at that foot read off.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (206, 42, 42),
    "orange": (232, 124, 28),
    "green": (46, 150, 62),
    "teal": (26, 158, 168),
    "blue": (48, 86, 208),
    "purple": (140, 58, 180),
}
RAIL_INK = (92, 100, 112)
POINTER_INK = (18, 18, 18)

HALF_INK = 2                # the ladder is drawn five pixels thick
MIN_RAIL_SPACING = 40       # a switch bar always reaches a genuinely distinct rail
MIN_POINTER_PIXELS = 120
BEAD_MIN_PIXELS = 400
MAX_STEPS = 4000


class Abstain(RuntimeError):
    """Raised when the picture does not support a confident reading."""


def _mask(array: np.ndarray, rgb) -> np.ndarray:
    return (
        (array[:, :, 0] == rgb[0])
        & (array[:, :, 1] == rgb[1])
        & (array[:, :, 2] == rgb[2])
    )


def _first_ink_below(ink: np.ndarray, column: int, start: int) -> int:
    """First row at or below ``start`` where the ladder is present at ``column``."""
    rows = np.flatnonzero(ink[start:, column])
    if not rows.size:
        raise Abstain("no rail under the pointer")
    return int(rows[0]) + start


def _pointer_column(array: np.ndarray, pointer: np.ndarray, ink: np.ndarray) -> tuple[int, int]:
    coords = np.argwhere(pointer)
    if len(coords) < MIN_POINTER_PIXELS:
        raise Abstain("start pointer not found")
    column = int(round(float(coords[:, 1].mean())))
    if not (0 <= column < ink.shape[1]):
        raise Abstain("start pointer off the board")
    top = _first_ink_below(ink, column, int(coords[:, 0].min()))
    if top - int(coords[:, 0].max()) > 2:
        raise Abstain("start pointer detached from its rail")
    return column, top


def _cross(ink: np.ndarray, column: int, row: int, step: int) -> int:
    """Follow a switch bar sideways and return the centre column it lands on."""
    x = column + step * (HALF_INK + 2)
    width = ink.shape[1]
    while 0 <= x < width and ink[row, x]:
        x += step
    landing = x - step * (HALF_INK + 1)
    if not (0 <= landing < width) or not ink[row, landing]:
        raise Abstain("switch bar does not land on a rail")
    if abs(landing - column) < MIN_RAIL_SPACING:
        raise Abstain("switch bar too short to join two rails")
    return landing


def _band_bottom(ink: np.ndarray, column: int, row: int, step: int) -> int:
    """Last row of the switch bar leaving ``column`` towards ``step``."""
    probe = column + step * (HALF_INK + 2)
    y = row
    while y + 1 < ink.shape[0] and ink[y + 1, probe]:
        y += 1
    return y


def _trace(ink: np.ndarray, column: int, row: int) -> tuple[int, int]:
    """Forced monotone descent: take every switch bar that meets the rail."""
    height = ink.shape[0]
    for _ in range(MAX_STEPS):
        left = column - HALF_INK - 2 >= 0 and ink[row, column - HALF_INK - 2]
        right = column + HALF_INK + 2 < ink.shape[1] and ink[row, column + HALF_INK + 2]
        if left and right:
            raise Abstain("two switch bars meet the rail at one height")
        if left or right:
            step = -1 if left else 1
            landing = _cross(ink, column, row, step)
            row = _band_bottom(ink, column, row, step) + 1
            column = landing
            if row >= height or not ink[row, column]:
                raise Abstain("rail ends inside a switch band")
            continue
        if row + 1 >= height or not ink[row + 1, column]:
            return column, row
        row += 1
    raise Abstain("descent did not terminate")


def _bead_colour(array: np.ndarray, centre: int, bottom: int) -> str:
    y0, y1 = bottom + 1, min(bottom + 44, array.shape[0] - 1)
    x0, x1 = max(centre - 20, 0), min(centre + 20, array.shape[1] - 1)
    patch = array[y0:y1 + 1, x0:x1 + 1]
    best, best_count = None, 0
    for name, rgb in PALETTE.items():
        count = int(_mask(patch, rgb).sum())
        if count > best_count:
            best, best_count = name, count
    if best is None or best_count < BEAD_MIN_PIXELS:
        raise Abstain("bead colour at the foot of the rail is not readable")
    return best


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    ink = _mask(array, RAIL_INK)
    pointer = _mask(array, POINTER_INK)
    column, top = _pointer_column(array, pointer, ink)
    column, bottom = _trace(ink, column, top)
    return _bead_colour(array, column, bottom)
