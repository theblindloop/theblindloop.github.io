"""Deterministic renderer and analytic arm for Grain Difference Blur Reveal."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 800
CELL = 100
CARD = 38
INSET = 5
PAYLOAD = 28
MACRO = 4
CLASSES = ("H", "U", "Pi")
QUARANTINE_GAP = 2.0


def _template(name: str) -> tuple[int, ...]:
    bits = [0] * 49
    if name == "H":
        for row in range(7):
            bits[row * 7] = bits[row * 7 + 6] = 1
        for col in range(7):
            bits[3 * 7 + col] = 1
    elif name == "U":
        for row in range(6):
            bits[row * 7] = bits[row * 7 + 6] = 1
        for col in range(7):
            bits[6 * 7 + col] = 1
    elif name == "Pi":
        for col in range(7):
            bits[col] = 1
        for row in range(1, 7):
            bits[row * 7] = bits[row * 7 + 6] = 1
    else:
        raise ValueError(f"unknown template {name!r}")
    return tuple(bits)


TEMPLATES = {name: _template(name) for name in CLASSES}


def _distances(bits: Iterable[int]) -> list[tuple[int, str]]:
    values = tuple(int(value) for value in bits)
    if len(values) != 49 or any(value not in (0, 1) for value in values):
        raise ValueError("reveal_bits must contain 49 binary values")
    return sorted(
        (sum(left != right for left, right in zip(values, template)), name)
        for name, template in TEMPLATES.items()
    )


def _classification(bits: Iterable[int]) -> tuple[str, float]:
    distances = _distances(bits)
    gap = float(distances[1][0] - distances[0][0])
    if gap == 0:
        return "ambiguous", 0.0
    return distances[0][1], gap


def sample_scene(seed: int) -> dict:
    """Balance base identity by seed and deterministically include boundary scenes."""
    rng = random.Random((int(seed) ^ 0x6D2B79F5) & 0xFFFFFFFF)
    base_index = int(seed) % len(CLASSES)
    base = CLASSES[base_index]
    bits = list(TEMPLATES[base])
    toward = TEMPLATES[CLASSES[(base_index + 1) % len(CLASSES)]]
    differing = [index for index, pair in enumerate(zip(bits, toward)) if pair[0] != pair[1]]
    rng.shuffle(differing)
    phase = (int(seed) // len(CLASSES)) % 8
    # Ten cells separate each template pair. Five changes make an exact tie;
    # four make a visible two-cell runner-up gap.
    changes = 5 if phase == 0 else 4 if phase == 1 else 1 if phase == 2 else 0
    for index in differing[:changes]:
        bits[index] = toward[index]
    return {
        "reveal_bits": bits,
        "grain_seed": rng.randrange(1, 2**31),
        "grid_position": [rng.randrange(1, 5), rng.randrange(1, 7)],
        "palette_index": rng.randrange(4),
    }


def analytic_gold(scene: dict) -> str:
    return _classification(scene["reveal_bits"])[0]


def margin(scene: dict) -> float:
    return _classification(scene["reveal_bits"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_grains(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    bits: tuple[int, ...],
    grain_seed: int,
    right_card: bool,
) -> None:
    rng = random.Random(grain_seed)
    masks: list[set[int]] = []
    for _ in range(49):
        masks.append(set(rng.sample(range(4), 2)))
    ox, oy = origin
    light = (246, 243, 234)
    dark = (32, 37, 47)
    draw.rectangle((ox, oy, ox + PAYLOAD - 1, oy + PAYLOAD - 1), fill=light)
    for index, bit in enumerate(bits):
        row, col = divmod(index, 7)
        chosen = masks[index]
        if right_card and bit:
            chosen = set(range(4)) - chosen
        for slot in chosen:
            sy, sx = divmod(slot, 2)
            x = ox + col * MACRO + sx * 2
            y = oy + row * MACRO + sy * 2
            draw.rectangle((x, y, x + 1, y + 1), fill=dark)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    frame: tuple[int, int, int],
    bits: tuple[int, ...],
    grain_seed: int,
    right_card: bool,
) -> None:
    x0, y0 = center[0] - CARD // 2, center[1] - CARD // 2
    draw.rectangle((x0, y0, x0 + CARD - 1, y0 + CARD - 1), fill=frame)
    _draw_grains(draw, (x0 + INSET, y0 + INSET), bits, grain_seed, right_card)
    # A background-colored triangular cutout visibly fixes the upright orientation.
    cx = x0 + CARD // 2
    draw.polygon(((cx - 4, y0), (cx + 4, y0), (cx, y0 + 4)), fill=(255, 255, 255))


def render(scene: dict) -> Image.Image:
    bits = tuple(int(value) for value in scene["reveal_bits"])
    grain_seed = int(scene["grain_seed"])
    col, row = (int(value) for value in scene["grid_position"])
    palette_index = int(scene["palette_index"])
    if not (1 <= col <= 4 and 1 <= row <= 6):
        raise ValueError("grid_position outside supported focal lattice")
    palettes = (
        ((0, 157, 196), (232, 115, 24)),
        ((143, 80, 210), (35, 171, 103)),
        ((39, 103, 210), (220, 62, 92)),
        ((201, 55, 160), (0, 151, 145)),
    )
    frames = palettes[palette_index % len(palettes)]
    image = Image.new("RGB", (CANVAS, CANVAS), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    # Remote grayscale marks vary with the grain assignment but are nondiagnostic.
    nuisance = random.Random(grain_seed ^ 0xA53C9E71)
    for _ in range(14):
        x, y = nuisance.randrange(35, CANVAS - 35), nuisance.randrange(35, CANVAS - 35)
        radius = nuisance.randrange(5, 15)
        shade = nuisance.randrange(205, 236)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=(shade,) * 3, width=2)

    left = (col * CELL + CELL // 2, row * CELL + CELL // 2)
    right = ((col + 2) * CELL + CELL // 2, row * CELL + CELL // 2)
    draw.rounded_rectangle(
        (left[0] - 37, left[1] - 35, right[0] + 37, right[1] + 35),
        radius=14,
        outline=(222, 225, 228),
        width=2,
    )
    _draw_card(draw, left, frames[0], bits, grain_seed, False)
    _draw_card(draw, right, frames[1], bits, grain_seed, True)
    return image
