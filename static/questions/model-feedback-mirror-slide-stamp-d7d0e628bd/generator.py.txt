"""Deterministic geometric-transformation renderer and analytic arm.

The visible computation is deliberately local to each tile: reflect a complete
asymmetric polygon in the tile's dashed line, add the directed vector shown by
the arrow, and compare the result with the pale outlined polygon.  The four
tiles use identical token topology but independent geometry, so slot, palette,
and total ink do not identify the answer.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 1020
HEIGHT = 740
PANEL_W = 480
PANEL_H = 340
MARGIN_X = 20
MARGIN_Y = 20
GAP_X = 20
GAP_Y = 20
PANEL_LABELS = ("A", "B", "C", "D")

BACKGROUND = (247, 249, 250)
PANEL_FILL = (255, 255, 255)
FRAME = (78, 98, 111)
FRAME_LIGHT = (213, 221, 226)
LABEL = (50, 65, 78)
SOURCE = (30, 66, 105)
TARGET_FILL = (255, 244, 216)
TARGET_EDGE = (210, 94, 53)
AXIS = (105, 78, 151)
ARROW_SHAFT = (28, 152, 143)
ARROW_TAIL = (116, 210, 188)
ARROW_HEAD = (11, 99, 101)
TAIL_CORE = (180, 235, 220)
HEAD_CORE = (4, 66, 67)

QUARANTINE_MARGIN = 14.0

# An irregular, concave token.  Its asymmetry makes orientation and reflection
# visible; it is large enough for a mask-only inverse to recover reliably.
BASE_SHAPE = (
    (-44, -28),
    (-10, -28),
    (8, -46),
    (43, -18),
    (27, -2),
    (49, 24),
    (9, 20),
    (-18, 49),
    (-26, 13),
    (-49, 24),
    (-31, -4),
)


def panel_origin(slot: int) -> tuple[int, int]:
    return (
        MARGIN_X + (int(slot) % 2) * (PANEL_W + GAP_X),
        MARGIN_Y + (int(slot) // 2) * (PANEL_H + GAP_Y),
    )


def _rotate_point(point: tuple[float, float], center: tuple[float, float], angle: float) -> tuple[float, float]:
    radians = math.radians(angle)
    c, s = math.cos(radians), math.sin(radians)
    dx, dy = point[0] - center[0], point[1] - center[1]
    return (center[0] + c * dx - s * dy, center[1] + s * dx + c * dy)


def _rounded(points: Iterable[tuple[float, float]]) -> list[list[int]]:
    return [[int(round(x)), int(round(y))] for x, y in points]


def _centroid(points: Iterable[Iterable[float]]) -> tuple[float, float]:
    values = [(float(p[0]), float(p[1])) for p in points]
    return (
        sum(p[0] for p in values) / len(values),
        sum(p[1] for p in values) / len(values),
    )


def _source_polygon(center: tuple[float, float], angle: float) -> list[list[int]]:
    return _rounded(
        _rotate_point((center[0] + dx, center[1] + dy), center, angle)
        for dx, dy in BASE_SHAPE
    )


def _reflect(point: Iterable[float], axis_point: Iterable[float], axis_direction: Iterable[float]) -> tuple[float, float]:
    px, py = float(point[0]), float(point[1])
    ax, ay = float(axis_point[0]), float(axis_point[1])
    dx, dy = float(axis_direction[0]), float(axis_direction[1])
    norm = math.hypot(dx, dy) or 1.0
    dx, dy = dx / norm, dy / norm
    vx, vy = px - ax, py - ay
    projection = vx * dx + vy * dy
    foot = (ax + projection * dx, ay + projection * dy)
    return (2.0 * foot[0] - px, 2.0 * foot[1] - py)


def _transformed_polygon(*, panel: dict) -> list[list[int]]:
    tail = panel["move_tail"]
    head = panel["move_head"]
    tx = float(head[0]) - float(tail[0])
    ty = float(head[1]) - float(tail[1])
    points = []
    for point in panel["source"]:
        rx, ry = _reflect(point, panel["axis_point"], panel["axis_direction"])
        points.append((rx + tx, ry + ty))
    return _rounded(points)


def _mean_vertex_error(left: list[list[int]], right: list[list[int]]) -> float:
    return sum(math.hypot(a[0] - b[0], a[1] - b[1]) for a, b in zip(left, right)) / len(left)


def _inside(points: Iterable[Iterable[int]]) -> bool:
    # The top strip is reserved for the directed arrow, keeping its endpoint
    # cores separate from both polygons in the raster.
    return all(28 <= int(p[0]) <= 452 and 120 <= int(p[1]) <= 318 for p in points)


def _scores(scene: dict) -> list[float]:
    values = []
    for panel in sorted(scene["panels"], key=lambda item: int(item["slot"])):
        values.append(_mean_vertex_error(_transformed_polygon(panel=panel), panel["target"]))
    return values


def _make_panel(rng: random.Random, slot: int, is_winner: bool) -> dict:
    """Rejection-sample one local tile with a clear target mismatch."""

    for _ in range(2000):
        center = (rng.randint(103, 143), rng.randint(151, 201))
        source = _source_polygon(center, rng.uniform(-26.0, 26.0))

        axis_angle = rng.uniform(67.0, 113.0)
        axis_direction = [round(math.cos(math.radians(axis_angle)), 8), round(math.sin(math.radians(axis_angle)), 8)]
        axis_point = [rng.randint(218, 244), rng.randint(164, 205)]

        tail = [rng.randint(286, 354), rng.randint(68, 83)]
        move_angle = rng.uniform(-165.0, 165.0)
        move_length = rng.uniform(25.0, 48.0)
        head = [
            int(round(tail[0] + move_length * math.cos(math.radians(move_angle)))),
            int(round(tail[1] + move_length * math.sin(math.radians(move_angle)))),
        ]
        if not (250 <= head[0] <= 390 and 48 <= head[1] <= 110):
            continue

        provisional = {
            "slot": int(slot),
            "source": source,
            "axis_point": axis_point,
            "axis_direction": axis_direction,
            "move_tail": tail,
            "move_head": head,
        }
        expected = _transformed_polygon(panel=provisional)
        if not _inside(source) or not _inside(expected):
            continue
        if math.hypot(_centroid(source)[0] - _centroid(expected)[0], _centroid(source)[1] - _centroid(expected)[1]) < 105:
            continue

        if is_winner:
            target = expected
        else:
            base_center = _centroid(expected)
            rotation = rng.choice((-1.0, 1.0)) * rng.uniform(9.0, 24.0)
            offset_length = rng.uniform(23.0, 39.0)
            offset_angle = rng.uniform(-180.0, 180.0)
            offset = (
                offset_length * math.cos(math.radians(offset_angle)),
                offset_length * math.sin(math.radians(offset_angle)),
            )
            target = _rounded(
                (
                    _rotate_point((float(p[0]), float(p[1])), base_center, rotation)[0] + offset[0],
                    _rotate_point((float(p[0]), float(p[1])), base_center, rotation)[1] + offset[1],
                )
                for p in expected
            )
            if _mean_vertex_error(expected, target) < 18.0:
                continue
        if not _inside(target):
            continue
        provisional["target"] = target
        return provisional
    raise RuntimeError(f"could not sample tile {slot}")


def sample_scene(seed: int) -> dict:
    """Sample four independently cued tiles with one exact composed match."""

    rng = random.Random(int(seed))
    winner = int(seed) % 4
    for _ in range(200):
        panels = [_make_panel(rng, slot, slot == winner) for slot in range(4)]
        scene = {"panels": panels}
        scores = _scores(scene)
        if scores[winner] == 0.0 and min(scores[index] for index in range(4) if index != winner) >= 18.0:
            return scene
    raise RuntimeError("could not sample a clear four-tile scene")


def _clip_line(point: tuple[float, float], direction: tuple[float, float], bounds: tuple[float, float, float, float]) -> tuple[float, float] | None:
    """Return the parameter interval for a line clipped to a rectangle."""

    low, high = -1.0e9, 1.0e9
    xmin, ymin, xmax, ymax = bounds
    for p, d, lo, hi in ((point[0], direction[0], xmin, xmax), (point[1], direction[1], ymin, ymax)):
        if abs(d) < 1.0e-12:
            if p < lo or p > hi:
                return None
            continue
        a, b = (lo - p) / d, (hi - p) / d
        if a > b:
            a, b = b, a
        low, high = max(low, a), min(high, b)
    return (low, high) if low <= high else None


def _global(origin: tuple[int, int], point: Iterable[float]) -> tuple[float, float]:
    return origin[0] + float(point[0]), origin[1] + float(point[1])


def _arrowhead(draw: ImageDraw.ImageDraw, tip: tuple[float, float], tail: tuple[float, float]) -> None:
    dx, dy = tip[0] - tail[0], tip[1] - tail[1]
    norm = math.hypot(dx, dy) or 1.0
    ux, uy = dx / norm, dy / norm
    px, py = -uy, ux
    back = (tip[0] - 13.0 * ux, tip[1] - 13.0 * uy)
    left = (back[0] + 7.0 * px, back[1] + 7.0 * py)
    right = (back[0] - 7.0 * px, back[1] - 7.0 * py)
    draw.polygon((tip, left, right), fill=ARROW_HEAD)


def _draw_axis(draw: ImageDraw.ImageDraw, origin: tuple[int, int], panel: dict) -> None:
    point = (float(panel["axis_point"][0]), float(panel["axis_point"][1]))
    direction = (float(panel["axis_direction"][0]), float(panel["axis_direction"][1]))
    clipped = _clip_line(point, direction, (18.0, 52.0, 462.0, 326.0))
    if clipped is None:
        return
    start, end = clipped
    cursor = start
    while cursor < end:
        a = min(cursor + 11.0, end)
        p0 = _global(origin, (point[0] + cursor * direction[0], point[1] + cursor * direction[1]))
        p1 = _global(origin, (point[0] + a * direction[0], point[1] + a * direction[1]))
        draw.line((p0, p1), fill=AXIS, width=3)
        cursor += 22.0


def _draw_arrow(draw: ImageDraw.ImageDraw, origin: tuple[int, int], panel: dict) -> None:
    tail = _global(origin, panel["move_tail"])
    head = _global(origin, panel["move_head"])
    draw.line((tail, head), fill=ARROW_SHAFT, width=5)
    _arrowhead(draw, head, tail)
    draw.ellipse((tail[0] - 7, tail[1] - 7, tail[0] + 7, tail[1] + 7), fill=ARROW_TAIL)
    draw.ellipse((head[0] - 7, head[1] - 7, head[0] + 7, head[1] + 7), fill=ARROW_HEAD)
    draw.ellipse((tail[0] - 4, tail[1] - 4, tail[0] + 4, tail[1] + 4), fill=TAIL_CORE)
    draw.ellipse((head[0] - 4, head[1] - 4, head[0] + 4, head[1] + 4), fill=HEAD_CORE)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for panel in sorted(scene["panels"], key=lambda item: int(item["slot"])):
        slot = int(panel["slot"])
        origin = panel_origin(slot)
        x0, y0 = origin
        x1, y1 = x0 + PANEL_W, y0 + PANEL_H
        draw.rounded_rectangle((x0, y0, x1, y1), radius=12, fill=PANEL_FILL, outline=FRAME, width=3)
        draw.text((x0 + 16, y0 + 12), f"tile-{PANEL_LABELS[slot]}", fill=LABEL)
        _draw_axis(draw, origin, panel)
        _draw_arrow(draw, origin, panel)

        target = [_global(origin, point) for point in panel["target"]]
        source = [_global(origin, point) for point in panel["source"]]
        draw.polygon(target, fill=TARGET_FILL, outline=TARGET_EDGE)
        draw.line(target + [target[0]], fill=TARGET_EDGE, width=3, joint="curve")
        draw.polygon(source, fill=SOURCE, outline=SOURCE)
        draw.line(source + [source[0]], fill=SOURCE, width=2, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    panels = sorted(scene["panels"], key=lambda item: int(item["slot"]))
    scores = [_mean_vertex_error(_transformed_polygon(panel=panel), panel["target"]) for panel in panels]
    winner = min(range(len(scores)), key=lambda index: scores[index])
    return f"tile-{PANEL_LABELS[winner]}"


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    best = min(scores)
    return (
        len({round(value, 9) for value in scores if value <= best + 1e-9}) != 1
        or margin(scene) < QUARANTINE_MARGIN
        or any(not _inside(panel["source"]) or not _inside(panel["target"]) for panel in scene["panels"])
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panels"] = list(reversed(twin["panels"]))
    return [("panel_record_order", twin)]
