"""Independent pixel-only inverse for the eccentric washer."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 4-connected components as arrays of (y, x) coordinates."""
    seen = np.zeros(mask.shape, dtype=bool)
    result: list[np.ndarray] = []
    height, width = mask.shape
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(np.asarray(points, dtype=np.int32))
    return result


def _enclosed_component(local_ink: np.ndarray) -> np.ndarray:
    empty = ~local_ink
    outside = np.zeros_like(empty)
    queue: deque[tuple[int, int]] = deque()
    height, width = empty.shape
    for x in range(width):
        for y in (0, height - 1):
            if empty[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(height):
        for x in (0, width - 1):
            if empty[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and empty[ny, nx] and not outside[ny, nx]:
                outside[ny, nx] = True
                queue.append((ny, nx))
    holes = _components(empty & ~outside)
    substantial = [component for component in holes if len(component) >= 250]
    if len(substantial) != 1:
        raise ValueError("one complete enclosed opening was not found")
    return substantial[0]


def _bbox(points: np.ndarray) -> tuple[int, int, int, int]:
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    return int(x0), int(y0), int(x1), int(y1)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB raster")
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    chromatic = (high - low) >= 55
    components = [component for component in _components(chromatic) if len(component) >= 30]
    if len(components) != 1 or len(components[0]) < 1500:
        raise ValueError("one complete colored washer was not found")
    washer = components[0]
    x0, y0, x1, y1 = _bbox(washer)
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (57 <= width <= 61 and 57 <= height <= 61 and abs(width - height) <= 1):
        raise ValueError("outer circular boundary is unresolved")

    local = chromatic[y0 : y1 + 1, x0 : x1 + 1]
    hole = _enclosed_component(local)
    hx0, hy0, hx1, hy1 = _bbox(hole)
    hole_width, hole_height = hx1 - hx0 + 1, hy1 - hy0 + 1
    if not (23 <= hole_width <= 27 and 23 <= hole_height <= 27 and abs(hole_width - hole_height) <= 1):
        raise ValueError("inner circular boundary is unresolved")

    outer_radius = (width + height - 2) / 4.0
    inner_radius = (hole_width + hole_height - 2) / 4.0
    reconstructed_outer_area = float(local.sum() + len(hole))
    expected_outer_area = np.pi * outer_radius * outer_radius
    expected_inner_area = np.pi * inner_radius * inner_radius
    if abs(reconstructed_outer_area - expected_outer_area) > 0.08 * expected_outer_area:
        raise ValueError("outer boundary is incomplete")
    if abs(float(len(hole)) - expected_inner_area) > 0.10 * expected_inner_area:
        raise ValueError("opening boundary is incomplete")

    outer_cx = (x0 + x1) / 2.0
    outer_cy = (y0 + y1) / 2.0
    inner_cx = x0 + (hx0 + hx1) / 2.0
    inner_cy = y0 + (hy0 + hy1) / 2.0
    horizontal = 2.0 * abs(inner_cx - outer_cx)
    vertical = 2.0 * abs(inner_cy - outer_cy)
    signed_gap = horizontal - vertical
    if abs(signed_gap) < 3.0:
        raise ValueError("wall-thickness imbalance boundary is ambiguous")
    return "horizontal" if signed_gap > 0.0 else "vertical"
