"""Deterministic sampler and renderer for an eccentric filled washer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 248, 246)
NEUTRAL_INK = (181, 184, 184)
PALETTES = (
    (20, 151, 157),
    (164, 82, 183),
    (224, 112, 42),
    (52, 155, 92),
)
OFFSET_PAIRS = ((8, 2), (7, 3), (9, 4), (8, 3))


def _nuisance_state(seed: int) -> tuple[int, int, int, int, int, int, int, list[dict[str, int]]]:
    """Adjacent seeds share all answer-irrelevant state and offset magnitudes."""
    rng = random.Random((int(seed) & ~1) ^ 0x73A91D)
    center_x = rng.randrange(1, 7) * 64 + 32
    center_y = rng.randrange(1, 7) * 64 + 32
    outer_radius = rng.choice((28, 29, 30))
    inner_radius = rng.choice((11, 12, 13))
    palette = rng.randrange(len(PALETTES))
    major, minor = rng.choice(OFFSET_PAIRS)
    sign_x = rng.choice((-1, 1))
    sign_y = rng.choice((-1, 1))

    distractors: list[dict[str, int]] = []
    attempts = 0
    while len(distractors) < 8 and attempts < 400:
        attempts += 1
        x = rng.randrange(20, WIDTH - 20)
        y = rng.randrange(20, HEIGHT - 20)
        if abs(x - center_x) < 74 and abs(y - center_y) < 74:
            continue
        if any((x - item["x"]) ** 2 + (y - item["y"]) ** 2 < 34**2 for item in distractors):
            continue
        distractors.append({"x": x, "y": y, "kind": rng.randrange(3), "size": rng.randrange(5, 10)})
    if len(distractors) != 8:
        raise RuntimeError("could not place neutral distractors")
    return center_x, center_y, outer_radius, inner_radius, palette, major, minor, sign_x, sign_y, distractors


def sample_scene(seed: int) -> dict[str, Any]:
    (
        center_x,
        center_y,
        outer_radius,
        inner_radius,
        palette,
        major,
        minor,
        sign_x,
        sign_y,
        distractors,
    ) = _nuisance_state(seed)
    if int(seed) & 1:
        offset_x, offset_y = sign_x * minor, sign_y * major
    else:
        offset_x, offset_y = sign_x * major, sign_y * minor
    return {
        "center_x": center_x,
        "center_y": center_y,
        "outer_radius": outer_radius,
        "inner_radius": inner_radius,
        "offset_x": offset_x,
        "offset_y": offset_y,
        "palette": palette,
        "distractors": distractors,
    }


def _draw_distractor(draw: ImageDraw.ImageDraw, item: dict[str, int]) -> None:
    x, y, size, kind = item["x"], item["y"], item["size"], item["kind"]
    if kind == 0:
        draw.ellipse((x - size, y - size, x + size, y + size), outline=NEUTRAL_INK, width=2)
    elif kind == 1:
        draw.line((x - size, y, x + size, y), fill=NEUTRAL_INK, width=2)
        draw.line((x, y - size, x, y + size), fill=NEUTRAL_INK, width=2)
    else:
        draw.rectangle((x - size, y - size, x + size, y + size), outline=NEUTRAL_INK, width=2)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for item in scene["distractors"]:
        _draw_distractor(draw, item)

    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    outer = int(scene["outer_radius"])
    inner = int(scene["inner_radius"])
    ix = cx + int(scene["offset_x"])
    iy = cy + int(scene["offset_y"])
    color = PALETTES[int(scene["palette"]) % len(PALETTES)]
    draw.ellipse((cx - outer, cy - outer, cx + outer, cy + outer), fill=color)
    draw.ellipse((ix - inner, iy - inner, ix + inner, iy + inner), fill=BACKGROUND)
    return image


def _signed_axis_gap(scene: dict[str, Any]) -> int:
    horizontal = 2 * abs(int(scene["offset_x"]))
    vertical = 2 * abs(int(scene["offset_y"]))
    return horizontal - vertical


def analytic_gold(scene: dict[str, Any]) -> str:
    return "horizontal" if _signed_axis_gap(scene) > 0 else "vertical"


def margin(scene: dict[str, Any]) -> float:
    return float(abs(_signed_axis_gap(scene)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
