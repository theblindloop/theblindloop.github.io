"""Independent pixel-only inverse for blue circular planet bodies.

This module intentionally has no renderer, prompt, generator, verifier, or
analytic-gold imports. It estimates the target color from RGB relationships,
labels connected components, and accepts only compact near-circular bodies.
"""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image


MIN_COMPONENT_AREA = 150
MIN_DIAMETER = 10
MIN_CIRCULARITY = 0.48


def _blue_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = pixels[:, :, 0]
    green = pixels[:, :, 1]
    blue = pixels[:, :, 2]
    return (
        (blue >= 120)
        & ((blue - red) >= 30)
        & ((blue - green) >= 18)
        & (blue >= red + 0.35 * np.maximum(red, 1))
    )


def _components(mask: np.ndarray) -> list[np.ndarray]:
    points = {tuple(point) for point in np.argwhere(mask)}
    components: list[np.ndarray] = []
    while points:
        first = points.pop()
        queue: deque[tuple[int, int]] = deque([first])
        found = [first]
        while queue:
            row, col = queue.popleft()
            for dr in (-1, 0, 1):
                for dc in (-1, 0, 1):
                    if dr == 0 and dc == 0:
                        continue
                    neighbor = (row + dr, col + dc)
                    if neighbor in points:
                        points.remove(neighbor)
                        queue.append(neighbor)
                        found.append(neighbor)
        components.append(np.asarray(found, dtype=np.float64))
    return components


def _component_stats(component: np.ndarray) -> dict[str, float]:
    rows = component[:, 0]
    cols = component[:, 1]
    height = float(rows.max() - rows.min() + 1.0)
    width = float(cols.max() - cols.min() + 1.0)
    area = float(len(component))
    center = component.mean(axis=0)
    distances = np.hypot(rows - center[0], cols - center[1])
    mean_radius = float(distances.mean())
    residual = float(np.mean(np.abs(distances - mean_radius))) if len(distances) else math.inf
    circularity = 1.0 - residual / max(mean_radius, 1.0)
    return {
        "area": area,
        "height": height,
        "width": width,
        "aspect": min(width, height) / max(width, height),
        "fill": area / max(width * height, 1.0),
        "circularity": circularity,
    }


def _stats(image: Image.Image) -> list[dict[str, float]]:
    return [_component_stats(component) for component in _components(_blue_mask(image))]


def _accepted(stat: dict[str, float]) -> bool:
    return (
        stat["area"] >= MIN_COMPONENT_AREA
        and stat["width"] >= MIN_DIAMETER
        and stat["height"] >= MIN_DIAMETER
        and stat["aspect"] >= 0.72
        and stat["fill"] >= 0.42
        and stat["circularity"] >= MIN_CIRCULARITY
    )


def decision_from_image(image: Image.Image) -> str:
    """Count blue, compact, circular connected components in the raster alone."""

    return str(sum(_accepted(stat) for stat in _stats(image)))


def decision_margin(image: Image.Image) -> float:
    """Return the smallest component-to-rejection margin in pixel-derived units."""

    stats = _stats(image)
    accepted = [stat for stat in stats if _accepted(stat)]
    if not accepted:
        return 0.0
    margins: list[float] = []
    for stat in accepted:
        margins.extend(
            (
                stat["area"] - MIN_COMPONENT_AREA,
                (stat["width"] - MIN_DIAMETER) * 10.0,
                (stat["height"] - MIN_DIAMETER) * 10.0,
                (stat["aspect"] - 0.72) * 100.0,
                (stat["fill"] - 0.42) * 100.0,
                (stat["circularity"] - MIN_CIRCULARITY) * 100.0,
            )
        )
    return round(max(0.0, min(margins)), 6)
