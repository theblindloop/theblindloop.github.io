"""Deterministic renderer for the visible blue-planet count world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 720
HEIGHT = 480
TOTAL_PLANETS = 8
SUPERSAMPLE = 4

BLUE_PALETTE = (
    (39, 108, 202),
    (44, 126, 216),
    (31, 91, 181),
)
DISTRACTOR_PALETTES = (
    ((214, 82, 69), (231, 151, 49), (61, 151, 103), (218, 105, 76)),
    ((196, 72, 80), (220, 139, 42), (66, 142, 126), (205, 103, 59)),
    ((225, 99, 52), (197, 157, 53), (53, 145, 117), (204, 81, 91)),
)
BACKGROUNDS = ((246, 242, 232), (239, 244, 239), (247, 239, 228))
PANEL_FILLS = ((253, 250, 242), (245, 249, 246), (250, 246, 238))
PANEL_OUTLINES = ((80, 91, 98), (91, 86, 78), (70, 96, 91))
TITLE_COLORS = ((38, 46, 53), (48, 45, 40), (35, 61, 58))
ORBIT_COLORS = ((150, 158, 160), (161, 151, 139), (143, 161, 153))
SUN_COLORS = ((236, 170, 49), (230, 132, 47), (238, 188, 62))


def _font(size: int, scale: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(
            "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * scale
        )
    except OSError:
        return ImageFont.load_default()


def _box(cx: float, cy: float, rx: float, ry: float, scale: int) -> tuple[int, int, int, int]:
    return (
        int(round((cx - rx) * scale)),
        int(round((cy - ry) * scale)),
        int(round((cx + rx) * scale)),
        int(round((cy + ry) * scale)),
    )


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)


def _pairwise_clearance(planets: list[dict[str, Any]]) -> float:
    clearances: list[float] = []
    for index, left in enumerate(planets):
        for right in planets[index + 1 :]:
            distance = math.hypot(float(left["x"]) - float(right["x"]), float(left["y"]) - float(right["y"]))
            clearances.append(distance - float(left["radius"]) - float(right["radius"]))
    return min(clearances) if clearances else 0.0


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a JSON-serializable scene using only the supplied integer seed."""

    rng = random.Random(int(seed))
    near_boundary = int(seed) % 11 == 0
    background = rng.choice(BACKGROUNDS)
    panel_fill = rng.choice(PANEL_FILLS)
    panel_outline = rng.choice(PANEL_OUTLINES)
    title_color = rng.choice(TITLE_COLORS)
    orbit_color = rng.choice(ORBIT_COLORS)
    sun_color = rng.choice(SUN_COLORS)
    target_blue = rng.choice(BLUE_PALETTE)
    distractor_palette = rng.choice(DISTRACTOR_PALETTES)

    center_x = rng.uniform(348.0, 372.0)
    center_y = rng.uniform(264.0, 284.0)
    orbit_rx = rng.uniform(220.0, 244.0)
    orbit_ry = rng.uniform(128.0, 148.0)
    rotation = rng.uniform(0.0, math.tau)
    body_radius_range = (12.0, 15.0) if near_boundary else (23.0, 28.0)
    blue_count = rng.choice((2, 3, 4, 5, 6, 7, 8))
    blue_indices = sorted(rng.sample(range(TOTAL_PLANETS), blue_count))

    planets: list[dict[str, Any]] = []
    for index in range(TOTAL_PLANETS):
        angle = rotation + math.tau * index / TOTAL_PLANETS
        radius = rng.uniform(*body_radius_range)
        x = center_x + orbit_rx * math.cos(angle)
        y = center_y + orbit_ry * math.sin(angle)
        planets.append(
            {
                "x": round(x, 4),
                "y": round(y, 4),
                "radius": round(radius, 4),
                "ringed": bool(rng.randrange(3) == 0),
                "distractor_color": list(rng.choice(distractor_palette)),
            }
        )

    stars = []
    for _ in range(14):
        stars.append(
            {
                "x": round(rng.uniform(75.0, 645.0), 3),
                "y": round(rng.uniform(115.0, 430.0), 3),
                "radius": round(rng.uniform(0.8, 1.8), 3),
            }
        )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(background),
        "panel_fill": list(panel_fill),
        "panel_outline": list(panel_outline),
        "title_color": list(title_color),
        "orbit_color": list(orbit_color),
        "center_x": round(center_x, 4),
        "center_y": round(center_y, 4),
        "orbit_rx": round(orbit_rx, 4),
        "orbit_ry": round(orbit_ry, 4),
        "planets": planets,
        "blue_indices": blue_indices,
        "target_blue": list(target_blue),
        "sun_color": list(sun_color),
        "sun_radius": 30.0 if not near_boundary else 28.0,
        "stars": stars,
        "near_boundary": near_boundary,
        "antialias": SUPERSAMPLE,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render a scene to an RGB PNG-equivalent image."""

    scale = int(scene["antialias"])
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), _rgb(scene["background"]))
    draw = ImageDraw.Draw(image)

    panel = (36, 78, width - 36, height - 30)
    draw.rounded_rectangle(
        tuple(int(value * scale) for value in panel),
        radius=18 * scale,
        fill=_rgb(scene["panel_fill"]),
        outline=_rgb(scene["panel_outline"]),
        width=2 * scale,
    )

    for star in scene["stars"]:
        draw.ellipse(
            _box(float(star["x"]), float(star["y"]), float(star["radius"]), float(star["radius"]), scale),
            fill=_rgb(scene["orbit_color"]),
        )

    center_x = float(scene["center_x"])
    center_y = float(scene["center_y"])
    orbit_color = _rgb(scene["orbit_color"])
    draw.ellipse(
        _box(center_x, center_y, float(scene["orbit_rx"]), float(scene["orbit_ry"]), scale),
        outline=orbit_color,
        width=2 * scale,
    )
    draw.ellipse(
        _box(
            center_x,
            center_y,
            float(scene["orbit_rx"]) * 0.63,
            float(scene["orbit_ry"]) * 0.63,
            scale,
        ),
        outline=orbit_color,
        width=scale,
    )

    sun_radius = float(scene["sun_radius"])
    draw.ellipse(
        _box(center_x, center_y, sun_radius, sun_radius, scale),
        fill=_rgb(scene["sun_color"]),
        outline=_rgb(scene["title_color"]),
        width=2 * scale,
    )

    blue_indices = set(int(index) for index in scene["blue_indices"])
    for index, planet in enumerate(scene["planets"]):
        x = float(planet["x"])
        y = float(planet["y"])
        radius = float(planet["radius"])
        if bool(planet["ringed"]):
            draw.ellipse(
                _box(x, y, radius * 1.55, radius * 0.48, scale),
                outline=orbit_color,
                width=scale,
            )
        fill = scene["target_blue"] if index in blue_indices else planet["distractor_color"]
        draw.ellipse(
            _box(x, y, radius, radius, scale),
            fill=_rgb(fill),
            outline=_rgb(scene["title_color"]),
            width=2 * scale,
        )

    title_font = _font(20, scale)
    draw.text(
        (58 * scale, 20 * scale),
        "SOLAR SURVEY",
        fill=_rgb(scene["title_color"]),
        font=title_font,
    )
    small_font = _font(10, scale)
    draw.text(
        (58 * scale, 53 * scale),
        "VISIBLE ORBITAL BODIES",
        fill=_rgb(scene["orbit_color"]),
        font=small_font,
    )

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image.convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the declared count; the only label input is the visible assignment."""

    return str(len(scene["blue_indices"]))


def margin(scene: dict[str, Any]) -> float:
    """One-sided pixel-scale margin for body size and separation."""

    radii = [float(planet["radius"]) for planet in scene["planets"]]
    radius_margin = min(radii) - 16.0
    separation_margin = _pairwise_clearance(scene["planets"]) - 18.0
    return round(min(radius_margin, separation_margin), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared for this injective scene representation."""

    return []
