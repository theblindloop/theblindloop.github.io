"""Independent pixel-only inverse for the colored self-crossing trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when destructive or ambiguous pixels do not expose a safe answer."""


def _components(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y0, x0 in np.argwhere(mask):
        y0 = int(y0)
        x0 = int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        size = 0
        while queue:
            y, x = queue.popleft()
            size += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _longest_run(row: np.ndarray) -> tuple[int, int, int]:
    padded = np.pad(row.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    if not len(starts):
        return 0, 0, 0
    lengths = ends - starts
    index = int(np.argmax(lengths))
    return int(lengths[index]), int(starts[index]), int(ends[index] - 1)


def _runs(profile: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(profile.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    return [(int(a), int(b - 1)) for a, b in zip(starts, ends, strict=True)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    maximum = array.max(axis=2)
    minimum = array.min(axis=2)
    mask = ((maximum - minimum) >= 70) & (maximum >= 90) & (minimum <= 150)

    sizes = _components(mask)
    if not sizes or sizes[0] < 1800:
        raise OracleAbstention("no substantial saturated practice trace")
    if len(sizes) > 1 and sizes[1] > 20:
        raise OracleAbstention("the colored trace is visibly disconnected")

    ys, xs = np.nonzero(mask)
    trace_width = int(xs.max() - xs.min() + 1)
    trace_height = int(ys.max() - ys.min() + 1)
    if not 250 <= trace_width <= 360 or not 65 <= trace_height <= 105:
        raise OracleAbstention("trace extent is outside the raster-safe envelope")

    row_runs = [_longest_run(mask[y]) for y in range(mask.shape[0])]
    longest = max(item[0] for item in row_runs)
    if longest < int(0.86 * trace_width):
        raise OracleAbstention("the central run is interrupted")
    candidate_rows = [
        y for y, item in enumerate(row_runs) if item[0] >= max(250, int(0.97 * longest))
    ]
    if not candidate_rows:
        raise OracleAbstention("no unambiguous central run")
    middle = int(round(float(np.median(candidate_rows))))
    _, run_left, run_right = row_runs[middle]

    probe = max(14, int(round(0.20 * trace_height)))
    band = 2
    if middle - probe - band < 0 or middle + probe + band >= mask.shape[0]:
        raise OracleAbstention("insufficient vertical probing clearance")
    above = mask[middle - probe - band : middle - probe + band + 1].any(axis=0)
    below = mask[middle + probe - band : middle + probe + band + 1].any(axis=0)
    through = above & below

    inset = max(22, int(round(0.075 * (run_right - run_left + 1))))
    through[: run_left + inset] = False
    through[run_right - inset + 1 :] = False
    junction_runs = _runs(through)
    widths = [right - left + 1 for left, right in junction_runs]
    if any(width < 5 or width > 18 for width in widths):
        raise OracleAbstention("ambiguous junction width")
    centers = [(left + right) / 2.0 for left, right in junction_runs]
    if any(b - a < 38 for a, b in zip(centers, centers[1:])):
        raise OracleAbstention("junctions are not visibly separated")
    count = len(junction_runs)
    if count not in (1, 2, 3, 4):
        raise OracleAbstention("crossing count is outside the declared answer set")
    return str(count)
