"""Deterministic renderer for a one-trace self-crossing count card."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
TRACE_COLORS = (
    (24, 105, 180),
    (0, 135, 105),
    (126, 65, 170),
    (190, 70, 75),
)
SCALES = (0.92, 0.98, 1.04, 1.08)
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")


def sample_scene(seed: int) -> dict:
    """Sample balanced labels and nuisance attributes from an integer seed."""

    rng = random.Random(int(seed) ^ 0x5E1F_C805)
    return {
        "crossing_count": 1 + (int(seed) % 4),
        "trace_color": list(TRACE_COLORS[(int(seed) // 4) % len(TRACE_COLORS)]),
        "dx": rng.randint(-18, 18),
        "dy": rng.randint(-2, 2),
        "scale": SCALES[rng.randrange(len(SCALES))],
        "x_bias": (-1, 0, 1)[rng.randrange(3)],
    }


def _geometry(scene: dict) -> tuple[float, float, float, float, float, int]:
    scale = float(scene["scale"])
    cx = 320.0 + int(scene["dx"])
    cy = 238.0 + int(scene["dy"])
    width = 304.0 * scale
    height = 70.0 * scale
    return (
        cx - width / 2.0,
        cx + width / 2.0,
        cy - height * 0.47,
        cy + height * 0.53,
        cy,
        max(9, int(round(11 * scale))),
    )


def _route_points(scene: dict) -> list[tuple[int, int]]:
    left, right, top, bottom, middle, _ = _geometry(scene)
    count = int(scene["crossing_count"])
    bias = int(scene["x_bias"])
    usable_left = left + 0.19 * (right - left)
    usable_right = right - 0.19 * (right - left)
    center_shift = bias * 0.026 * (right - left)
    if count == 1:
        crossings = [(usable_left + usable_right) / 2.0 + center_shift]
    else:
        step = (usable_right - usable_left) / (count - 1)
        crossings = [usable_left + index * step + center_shift for index in range(count)]

    points: list[tuple[float, float]] = [(left, middle), (right, middle), (right, top)]
    side = top
    for x in reversed(crossings):
        points.append((x, side))
        side = bottom if side == top else top
        points.append((x, side))
    points.extend(((left, side), (left, middle)))
    return [(int(round(x)), int(round(y))) for x, y in points]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 246, 241))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        (54, 38, WIDTH - 54, HEIGHT - 38),
        radius=24,
        fill=(255, 255, 252),
        outline=(94, 98, 104),
        width=3,
    )
    title_font = ImageFont.truetype(str(FONT_PATH), 25)
    small_font = ImageFont.truetype(str(FONT_PATH), 15)
    title = "FIGURE-EIGHT PRACTICE TRACE"
    title_box = draw.textbbox((0, 0), title, font=title_font)
    draw.text(
        ((WIDTH - (title_box[2] - title_box[0])) / 2, 69),
        title,
        fill=(55, 59, 66),
        font=title_font,
    )
    caption = "COUNT ONLY FOUR-ARM SELF-CROSSINGS"
    caption_box = draw.textbbox((0, 0), caption, font=small_font)
    draw.text(
        ((WIDTH - (caption_box[2] - caption_box[0])) / 2, 400),
        caption,
        fill=(102, 106, 112),
        font=small_font,
    )

    color = tuple(int(value) for value in scene["trace_color"])
    _, _, _, _, _, stroke = _geometry(scene)
    draw.line(_route_points(scene), fill=color, width=stroke, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["crossing_count"]))


def margin(scene: dict) -> float:
    left, right, _, _, _, stroke = _geometry(scene)
    count = int(scene["crossing_count"])
    usable = 0.62 * (right - left)
    separation = usable if count == 1 else usable / (count - 1)
    border_clearance = 0.19 * (right - left)
    return float(min(separation, border_clearance) - 3.2 * stroke)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
