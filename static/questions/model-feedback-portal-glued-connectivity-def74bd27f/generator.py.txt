"""Deterministic two-board portal-connectivity sampler and renderer."""

from __future__ import annotations

from collections import deque
import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 640
BACKGROUND = (250, 248, 244)
INK = (43, 48, 56)
JOINT = (218, 230, 241)
PANEL = (177, 187, 198)
LABEL = (78, 87, 98)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
PORTAL_COLORS = {
    "triangle": (45, 145, 91),
    "square": (47, 108, 196),
    "diamond": (218, 76, 67),
    "hexagon": (145, 124, 31),
}
COLOR_NAMES = tuple(COLORS)
PORTAL_NAMES = tuple(PORTAL_COLORS)


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _parts(node_ids: list[str], edges: list[list[str]], portals: dict[str, list[str]]) -> list[set[str]]:
    adjacency = {node_id: [] for node_id in node_ids}
    for a, b in edges + list(portals.values()):
        adjacency[a].append(b)
        adjacency[b].append(a)
    remaining, found = set(node_ids), []
    while remaining:
        start = min(remaining)
        reached, queue = {start}, deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here]:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        found.append(reached)
        remaining -= reached
    return found


def _portal_distance(start: str, goal: str, edges: list[list[str]], portals: dict[str, list[str]]) -> int:
    adjacency: dict[str, list[tuple[str, int]]] = {}
    for a, b in edges:
        adjacency.setdefault(a, []).append((b, 0))
        adjacency.setdefault(b, []).append((a, 0))
    for a, b in portals.values():
        adjacency.setdefault(a, []).append((b, 1))
        adjacency.setdefault(b, []).append((a, 1))
    best, queue = {(start, 0)}, deque([(start, 0)])
    while queue:
        here, used = queue.popleft()
        if here == goal:
            return used
        for nxt, cost in adjacency.get(here, []):
            state = (nxt, used + cost)
            if state not in best and used + cost <= 8:
                best.add(state)
                queue.append(state)
    return 99


def sample_scene(seed: int) -> dict:
    """Build a quotient graph whose answer path alternates boards four times."""

    rng = random.Random(int(seed))
    xs = {"A": [90, 205, 320], "B": [440, 555, 670]}
    ys = [155, 280, 405, 530]
    nodes = [
        {"id": f"{board}{row}{col}", "board": board, "x": xs[board][col], "y": ys[row]}
        for board in ("A", "B") for row in range(4) for col in range(3)
    ]
    a_rows = list(range(4))
    rng.shuffle(a_rows)
    l0, l1, l2, l3 = a_rows
    if rng.randrange(2):
        r0_rows, leftovers = [0, 1], [2, 3]
    else:
        r0_rows, leftovers = [2, 3], [0, 1]
    rng.shuffle(leftovers)
    r1, r2 = leftovers

    edges: list[list[str]] = []
    for row in range(4):
        edges.extend([_edge(f"A{row}0", f"A{row}1"), _edge(f"A{row}1", f"A{row}2")])
        edges.extend([_edge(f"B{row}0", f"B{row}1"), _edge(f"B{row}1", f"B{row}2")])
    for col in range(3):
        edges.append(_edge(f"B{r0_rows[0]}{col}", f"B{r0_rows[1]}{col}"))

    def take(row_nodes: list[str], count: int) -> list[str]:
        choices = list(row_nodes)
        rng.shuffle(choices)
        return choices[:count]

    a0 = take([f"A{l0}{c}" for c in range(3)], 2)
    a1 = take([f"A{l1}{c}" for c in range(3)], 2)
    a2 = take([f"A{l2}{c}" for c in range(3)], 2)
    a3 = take([f"A{l3}{c}" for c in range(3)], 1)
    b0 = take([f"B{r}{c}" for r in r0_rows for c in range(3)], 3)
    b1 = take([f"B{r1}{c}" for c in range(3)], 3)
    b2 = take([f"B{r2}{c}" for c in range(3)], 1)

    roles = {
        "p0a": a0[0], "ansa": a0[1],
        "p1a": a1[0], "p2a": a1[1],
        "p3a": a2[0], "ansb": a2[1],
        "iso1": a3[0],
        "p0b": b0[0], "p1b": b0[1], "chain1": b0[2],
        "p2b": b1[0], "p3b": b1[1], "chain2": b1[2],
        "iso2": b2[0],
    }
    portal_names = list(PORTAL_NAMES)
    rng.shuffle(portal_names)
    portals = {
        portal_names[0]: sorted([roles["p0a"], roles["p0b"]]),
        portal_names[1]: sorted([roles["p1a"], roles["p1b"]]),
        portal_names[2]: sorted([roles["p2a"], roles["p2b"]]),
        portal_names[3]: sorted([roles["p3a"], roles["p3b"]]),
    }
    answer = COLOR_NAMES[int(seed) % 3]
    distractors = [color for color in COLOR_NAMES if color != answer]
    rng.shuffle(distractors)
    terminal_pairs = {
        answer: sorted([roles["ansa"], roles["ansb"]]),
        distractors[0]: sorted([roles["chain1"], roles["iso1"]]),
        distractors[1]: sorted([roles["chain2"], roles["iso2"]]),
    }
    if rng.randrange(2):
        for node in nodes:
            node["x"] = 760 - int(node["x"])
            node["board"] = "A" if node["board"] == "B" else "B"
    if rng.randrange(2):
        for node in nodes:
            node["y"] = 685 - int(node["y"])
    rng.shuffle(nodes)
    rng.shuffle(edges)
    return {"boards": ["A", "B"], "nodes": nodes, "edges": edges,
            "portals": portals, "terminal_pairs": terminal_pairs}


def _answer(scene: dict) -> str:
    node_ids = [str(node["id"]) for node in scene["nodes"]]
    parts = _parts(node_ids, [list(map(str, e)) for e in scene["edges"]],
                   {str(k): list(map(str, v)) for k, v in scene["portals"].items()})
    answers = [
        str(color) for color, pair in scene["terminal_pairs"].items()
        if any(pair[0] in part and pair[1] in part for part in parts)
    ]
    if len(answers) != 1:
        raise ValueError(f"expected one connected pair, got {answers}")
    return answers[0]


def analytic_gold(scene: dict) -> str:
    answer = _answer(scene)
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    return answer


def margin(scene: dict) -> float:
    _answer(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        node_ids = [str(node["id"]) for node in scene["nodes"]]
        edges = [list(map(str, edge)) for edge in scene["edges"]]
        portals = {str(k): list(map(str, v)) for k, v in scene["portals"].items()}
        terminals = {str(k): list(map(str, v)) for k, v in scene["terminal_pairs"].items()}
        if len(node_ids) != 24 or len(set(node_ids)) != 24 or len(edges) != 19:
            return True
        if set(portals) != set(PORTAL_NAMES) or any(len(v) != 2 for v in portals.values()):
            return True
        if set(terminals) != set(COLOR_NAMES) or any(len(v) != 2 for v in terminals.values()):
            return True
        marked = [n for pair in portals.values() for n in pair] + [n for pair in terminals.values() for n in pair]
        if len(marked) != 14 or len(set(marked)) != 14 or not set(marked) <= set(node_ids):
            return True
        answer = _answer(scene)
        if _portal_distance(terminals[answer][0], terminals[answer][1], edges, portals) != 4:
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _polygon(cx: float, cy: float, radius: float, sides: int, rotation: float) -> list[tuple[float, float]]:
    return [(cx + radius * math.cos(rotation + 2 * math.pi * i / sides),
             cy + radius * math.sin(rotation + 2 * math.pi * i / sides)) for i in range(sides)]


def _draw_portal(draw: ImageDraw.ImageDraw, shape: str, x: int, y: int, scale: int) -> None:
    cx, cy, r = x * scale, y * scale, 19 * scale
    if shape == "square":
        points = [(cx-r, cy-r), (cx+r, cy-r), (cx+r, cy+r), (cx-r, cy+r)]
    elif shape == "diamond":
        points = [(cx, cy-r-2*scale), (cx+r+2*scale, cy), (cx, cy+r+2*scale), (cx-r-2*scale, cy)]
    elif shape == "triangle":
        points = _polygon(cx, cy+2*scale, 22*scale, 3, -math.pi/2)
    else:
        points = _polygon(cx, cy, 21*scale, 6, 0)
    draw.polygon(points, fill=PORTAL_COLORS[shape], outline=INK, width=3*scale)


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH*scale, HEIGHT*scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((35*scale, 85*scale, 365*scale, 595*scale), radius=18*scale,
                           outline=PANEL, width=3*scale)
    draw.rounded_rectangle((395*scale, 85*scale, 725*scale, 595*scale), radius=18*scale,
                           outline=PANEL, width=3*scale)
    draw.text((200*scale, 54*scale), "BOARD A", fill=LABEL, anchor="mm", font=_font(23*scale))
    draw.text((560*scale, 54*scale), "BOARD B", fill=LABEL, anchor="mm", font=_font(23*scale))
    node_map = {str(node["id"]): node for node in scene["nodes"]}
    for a_id, b_id in sorted((_edge(*edge) for edge in scene["edges"])):
        a, b = node_map[a_id], node_map[b_id]
        draw.line([(a["x"]*scale, a["y"]*scale), (b["x"]*scale, b["y"]*scale)],
                  fill=INK, width=7*scale)
    terminal_of = {node: color for color, pair in scene["terminal_pairs"].items() for node in pair}
    portal_of = {node: shape for shape, pair in scene["portals"].items() for node in pair}
    for node_id in sorted(node_map):
        x, y = int(node_map[node_id]["x"]), int(node_map[node_id]["y"])
        if node_id in portal_of:
            _draw_portal(draw, portal_of[node_id], x, y, scale)
        else:
            radius = 18 if node_id in terminal_of else 12
            color = COLORS.get(terminal_of.get(node_id, ""), JOINT)
            draw.ellipse(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale),
                         fill=color, outline=INK, width=3*scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["nodes"] = list(reversed(twin["nodes"]))
    twin["edges"] = list(reversed(twin["edges"]))
    twin["portals"] = {k: list(reversed(v)) for k, v in reversed(list(twin["portals"].items()))}
    twin["terminal_pairs"] = {k: list(reversed(v)) for k, v in reversed(list(twin["terminal_pairs"].items()))}
    return [("reverse_record_orders", twin)]
