"""Independent image-only inverse for two-board portal connectivity."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "ink": (43, 48, 56), "joint": (218, 230, 241),
    "amber": (232, 156, 37), "cyan": (31, 163, 190), "violet": (132, 79, 188),
    "triangle": (45, 145, 91), "square": (47, 108, 196),
    "diamond": (218, 76, 67), "hexagon": (145, 124, 31),
}
COLOR_NAMES = ("amber", "cyan", "violet")
PORTAL_NAMES = ("triangle", "square", "diamond", "hexagon")


class OracleError(ValueError):
    pass


def _mask(arr: np.ndarray, color: tuple[int, int, int], tolerance: int = 15) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - np.asarray(color, dtype=np.int16)), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    remaining = set(map(tuple, np.argwhere(mask)))
    found = []
    while remaining:
        start = remaining.pop()
        stack, points = [start], [start]
        while stack:
            y, x = stack.pop()
            for nxt in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= nxt[0] < height and 0 <= nxt[1] < width and nxt in remaining:
                    remaining.remove(nxt)
                    stack.append(nxt)
                    points.append(nxt)
        if len(points) >= minimum:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _recover_nodes(arr: np.ndarray) -> tuple[list[dict], dict[str, list[int]]]:
    nodes, portal_indices = [], {}
    for color in COLOR_NAMES:
        parts = _components(_mask(arr, PALETTE[color]), 180)
        if len(parts) != 2:
            raise OracleError(f"expected two {color} terminals, found {len(parts)}")
        for part in parts:
            y, x = part.mean(axis=0)
            nodes.append({"point": (float(x), float(y)), "color": color, "portal": None})
    for shape in PORTAL_NAMES:
        parts = _components(_mask(arr, PALETTE[shape]), 150)
        if len(parts) != 2:
            raise OracleError(f"expected two {shape} portals, found {len(parts)}")
        portal_indices[shape] = []
        for part in parts:
            y, x = part.mean(axis=0)
            portal_indices[shape].append(len(nodes))
            nodes.append({"point": (float(x), float(y)), "color": None, "portal": shape})
    joints = [
        part for part in _components(_mask(arr, PALETTE["joint"]), 80)
        if min(abs(float(part[:, 1].mean()) - x) for x in (90, 205, 320, 440, 555, 670)) < 7
        and min(abs(float(part[:, 0].mean()) - y) for y in (155, 280, 405, 530)) < 7
    ]
    if len(joints) != 10:
        raise OracleError(f"expected ten pale junctions, found {len(joints)}")
    for part in joints:
        y, x = part.mean(axis=0)
        nodes.append({"point": (float(x), float(y)), "color": None, "portal": None})
    return nodes, portal_indices


def _coverage(mask: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> float:
    hits = 0
    for t in np.linspace(0.20, 0.80, 64):
        x, y = a[0] + t*(b[0]-a[0]), a[1] + t*(b[1]-a[1])
        ix, iy = int(round(x)), int(round(y))
        hits += bool(mask[max(0, iy-3):iy+4, max(0, ix-3):ix+4].any())
    return hits / 64


def _recover_edges(arr: np.ndarray, nodes: list[dict]) -> list[tuple[int, int]]:
    ink = _mask(arr, PALETTE["ink"], 24)
    edges = []
    for i in range(len(nodes)):
        for j in range(i+1, len(nodes)):
            a, b = nodes[i]["point"], nodes[j]["point"]
            same_board = (a[0] < 380) == (b[0] < 380)
            dx, dy = abs(a[0]-b[0]), abs(a[1]-b[1])
            plausible = (105 < dx < 125 and dy < 6) or (dx < 6 and 115 < dy < 135)
            if same_board and plausible and _coverage(ink, a, b) >= 0.80:
                edges.append((i, j))
    if len(edges) != 19:
        raise OracleError(f"expected nineteen board links, recovered {len(edges)}")
    return edges


def _parts(count: int, edges: list[tuple[int, int]]) -> list[set[int]]:
    adjacency = {i: [] for i in range(count)}
    for a, b in edges:
        adjacency[a].append(b)
        adjacency[b].append(a)
    remaining, found = set(range(count)), []
    while remaining:
        start = min(remaining)
        reached, queue = {start}, deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here]:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        found.append(reached)
        remaining -= reached
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[:2] != (640, 760):
        raise OracleError(f"unexpected image size {image.size}")
    nodes, portals = _recover_nodes(arr)
    edges = _recover_edges(arr, nodes)
    for shape, pair in portals.items():
        if len(pair) != 2 or (nodes[pair[0]]["point"][0] < 380) == (nodes[pair[1]]["point"][0] < 380):
            raise OracleError(f"{shape} portals do not span the boards")
        edges.append(tuple(sorted(pair)))
    parts = _parts(len(nodes), edges)
    answers = []
    for color in COLOR_NAMES:
        pair = [i for i, node in enumerate(nodes) if node["color"] == color]
        if any(pair[0] in part and pair[1] in part for part in parts):
            answers.append(color)
    if len(answers) != 1:
        raise OracleError(f"expected one connected color pair, found {answers}")
    return answers[0]
