"""Renderer and analytic arm for the connected route-gap difference world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "route-gap-delta-0.1.0"
WIDTH = 640
HEIGHT = 480
SCALE = 4
N_GAPS = 32
N_STATIONS = N_GAPS + 1
ROUTE_SPAN = 400.0
SHORT_GAP = 9.0
LONG_GAP = 18.0
BACKGROUND = (248, 249, 245)

# The two outputs have equal ink and equal numbers of rows/columns with useful
# changes.  The answer is carried by order after the declared difference, not
# by total ink or by a single local mark.
CREST = tuple(int(bit) for bit in "00111100011001101100001111111111")
VAULT = tuple(int(bit) for bit in "11111111110000110110011000111100")
TEMPLATES = {"A": CREST, "B": VAULT}
DIFF_POSITIONS = tuple(index for index, pair in enumerate(zip(CREST, VAULT)) if pair[0] != pair[1])

PALETTES = (
    ((24, 49, 87), (211, 57, 63), (43, 147, 91), (196, 126, 31), (232, 240, 239)),
    ((29, 53, 92), (207, 63, 61), (38, 141, 95), (190, 120, 34), (236, 242, 239)),
    ((22, 45, 82), (219, 68, 59), (48, 151, 93), (202, 130, 28), (230, 239, 240)),
)


def _difference(bits: tuple[int, ...] | list[int]) -> tuple[int, ...]:
    """Keep the first bit, then write a one for each binary transition."""

    if len(bits) != N_GAPS or any(int(value) not in (0, 1) for value in bits):
        raise ValueError("gap sequence must contain 32 binary samples")
    values = tuple(int(value) for value in bits)
    return tuple(
        values[index] if index == 0 else values[index] ^ values[index - 1]
        for index in range(N_GAPS)
    )


def _inverse_difference(output: tuple[int, ...] | list[int]) -> tuple[int, ...]:
    if len(output) != N_GAPS or any(int(value) not in (0, 1) for value in output):
        raise ValueError("output sequence must contain 32 binary samples")
    running = 0
    result: list[int] = []
    for index, value in enumerate(output):
        bit = int(value)
        running = bit if index == 0 else running ^ bit
        result.append(running)
    return tuple(result)


def _hamming(left: tuple[int, ...], right: tuple[int, ...]) -> int:
    return sum(first != second for first, second in zip(left, right))


def _classify(output: tuple[int, ...]) -> str:
    distance_a = _hamming(output, CREST)
    distance_b = _hamming(output, VAULT)
    return "A" if distance_a <= distance_b else "B"


def _near_boundary_output(label: str, rng: random.Random) -> list[int]:
    output = list(TEMPLATES[label])
    # Seven flips among the sixteen template disagreements leave a visible,
    # explicitly quarantined two-pixel decision gap.
    for index in rng.sample(DIFF_POSITIONS, 7):
        output[index] = 1 - output[index]
    return output


def sample_scene(seed: int) -> dict:
    """Sample balanced gap labels with independent visible route nuisances."""

    rng = random.Random(int(seed))
    label = "A" if int(seed) % 2 == 0 else "B"
    output = list(TEMPLATES[label])
    if int(seed) % 13 == 0:
        output = _near_boundary_output(label, rng)
    gap_bits = _inverse_difference(output)
    return {
        "gap_bits": list(gap_bits),
        "route_style": int(rng.randrange(4)),
        "x0": int(rng.choice((92, 104, 116, 128))),
        "base_y": int(rng.choice((205, 225, 245, 265))),
        "amplitude": int(rng.choice((24, 30, 36, 42))),
        "phase": float(rng.choice((0.0, 0.25, 0.5, 0.75))),
        "reverse": bool(rng.randrange(2)),
        "stroke": int(rng.choice((7, 8, 9))),
        "palette_variant": int(rng.randrange(len(PALETTES))),
        "window_variant": int(rng.randrange(3)),
    }


def _curve_y(scene: dict, t: float) -> float:
    style = int(scene["route_style"]) % 4
    base = float(scene["base_y"])
    amplitude = float(scene["amplitude"])
    phase = float(scene["phase"])
    if style == 0:
        offset = amplitude * math.sin(math.pi * t + phase)
    elif style == 1:
        offset = -amplitude * math.sin(math.pi * t + phase)
    elif style == 2:
        offset = amplitude * 0.72 * math.sin(2.0 * math.pi * t + phase)
    else:
        offset = amplitude * 0.56 * math.sin(math.pi * t + phase) + amplitude * 0.34 * math.sin(
            2.0 * math.pi * t + phase
        )
    return base + offset


def _station_points(scene: dict) -> list[tuple[float, float]]:
    bits = tuple(int(value) for value in scene["gap_bits"])
    if len(bits) != N_GAPS or any(value not in (0, 1) for value in bits):
        raise ValueError("gap_bits must contain 32 binary samples")
    # The route's left-to-right spacing is the reverse of traversal order when
    # the red cap is drawn on the right.
    left_to_right = bits if not bool(scene["reverse"]) else tuple(reversed(bits))
    raw = [SHORT_GAP if bit == 0 else LONG_GAP for bit in left_to_right]
    total = float(sum(raw))
    x0 = float(scene["x0"])
    points: list[tuple[float, float]] = [(x0, _curve_y(scene, 0.0))]
    distance = 0.0
    for gap in raw:
        distance += float(gap)
        t = distance / total
        points.append((x0 + ROUTE_SPAN * t, _curve_y(scene, t)))
    if len(points) != N_STATIONS:
        raise AssertionError("station count changed")
    return points


def render(scene: dict) -> Image.Image:
    """Render the route, direction caps, and visible station rings."""

    navy, red, green, amber, window = PALETTES[int(scene["palette_variant"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    x0 = float(scene["x0"])
    base_y = float(scene["base_y"])
    margin = 24.0 + 5.0 * (int(scene["window_variant"]) % 3)
    panel = (
        int(round((x0 - margin) * SCALE)),
        int(round((base_y - 112.0) * SCALE)),
        int(round((x0 + ROUTE_SPAN + margin) * SCALE)),
        int(round((base_y + 112.0) * SCALE)),
    )
    draw.rounded_rectangle(
        panel,
        radius=18 * SCALE,
        fill=window,
        outline=(213, 224, 221),
        width=2 * SCALE,
    )

    points = _station_points(scene)
    scaled = [(int(round(x * SCALE)), int(round(y * SCALE))) for x, y in points]
    route_width = int(round(float(scene["stroke"]) * SCALE))
    draw.line(scaled, fill=navy, width=route_width, joint="curve")
    radius = max(2, route_width // 2)
    for px, py in scaled:
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=navy)

    # Hollow rings keep the navy route continuous while exposing all interior
    # station centers as independent visible fiducials.
    ring_radius = 3.0 * SCALE
    ring_width = 1.6 * SCALE
    for px, py in scaled[1:-1]:
        draw.ellipse(
            (
                int(round(px - ring_radius)),
                int(round(py - ring_radius)),
                int(round(px + ring_radius)),
                int(round(py + ring_radius)),
            ),
            outline=amber,
            width=max(1, int(round(ring_width))),
        )

    # Direction caps are intentionally modest so the first and last station
    # gaps remain visible.
    cap_radius = int(round(5.5 * SCALE))
    start = scaled[0] if not bool(scene["reverse"]) else scaled[-1]
    end = scaled[-1] if not bool(scene["reverse"]) else scaled[0]
    draw.ellipse(
        (start[0] - cap_radius, start[1] - cap_radius, start[0] + cap_radius, start[1] + cap_radius),
        fill=red,
    )
    draw.ellipse(
        (end[0] - cap_radius, end[1] - cap_radius, end[0] + cap_radius, end[1] + cap_radius),
        fill=green,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    bits = tuple(int(value) for value in scene["gap_bits"])
    return _classify(_difference(bits))


def margin(scene: dict) -> float:
    bits = tuple(int(value) for value in scene["gap_bits"])
    output = _difference(bits)
    return float(abs(_hamming(output, CREST) - _hamming(output, VAULT)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No unrendered aliases are claimed; all label-bearing fields are drawn.
    return []
