"""Independent pixel-only inverse arm for the dashed-contour shape world.

Receives only the final PNG. It takes the convex hull envelope of all the dark
broken-stroke ink, then classifies the reconstructed outline silhouette over
{circle, square, diamond, triangle} from the bounding-box fill ratio and
vertical centroid offset.

This module imports no renderer / prompts / generate / verify / analytic-gold
code and never receives the latent scene.
"""

from __future__ import annotations
import numpy as np
from PIL import Image

_SQUARE_MIN = 0.85
_CIRCLE_MIN = 0.68
_OFFSET_MIN = 0.08


def _convex_hull(points: np.ndarray) -> np.ndarray:
    pts = np.unique(points, axis=0)
    if len(pts) <= 1:
        return pts
    pts = pts[np.lexsort((pts[:, 1], pts[:, 0]))]

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(tuple(p))
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(tuple(p))
    return np.array(lower[:-1] + upper[:-1])


def _features(arr: np.ndarray) -> tuple[float, float]:
    ys, xs = np.where(arr < 120)
    if xs.size < 3:
        return 0.0, 0.0
    hull = _convex_hull(np.stack([xs.astype(float), ys.astype(float)], axis=1))
    if len(hull) < 3:
        return 0.0, 0.0
    xs0 = hull[:, 0]
    ys0 = hull[:, 1]
    cross_term = xs0 * np.roll(ys0, -1) - ys0 * np.roll(xs0, -1)
    area = 0.5 * abs(np.sum(cross_term))
    cy = float(np.sum((ys0 + np.roll(ys0, -1)) * cross_term) / (6.0 * area))
    bw = float(xs.max() - xs.min() + 1)
    bh = float(ys.max() - ys.min() + 1)
    fill = float(area) / (bw * bh)
    bcenter_y = (float(ys.min()) + float(ys.max())) / 2.0
    offset = abs(cy - bcenter_y) / bh
    return fill, offset


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))[..., 0]
    fill, offset = _features(arr)
    if fill >= _SQUARE_MIN:
        return "square"
    if fill >= _CIRCLE_MIN:
        return "circle"
    return "triangle" if offset > _OFFSET_MIN else "diamond"
