"""Dashed-contour shape world: a closed outline revealed by broken strokes.

A white canvas holds a set of short, separated dark bars laid tip-to-tail
along the perimeter of a single hidden shape. No single bar alone shows the
figure; the whole shape emerges only from the global contour-grouping /
perceptual-continuation Gestalt that links each bar to its next collinear
neighbour across the small gaps. The decision is which of four shapes
{circle, square, diamond, triangle} the continued outline traces.

The renderer owns scene sampling + rasterization + the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) reconstructs the
closed outline envelope from the final PNG alone by taking the convex hull of
the whole broken-stroke ink and classifies its silhouette from the bounding-box
fill ratio and vertical centroid offset.
"""

from __future__ import annotations
import math
import numpy as np
from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
CENTER = 192.0
R = 110.0
BAR_LEN = 30.0
STROKE = 5
SHAPES = ["circle", "square", "diamond", "triangle"]

_SQUARE_MIN = 0.85
_CIRCLE_MIN = 0.68
_OFFSET_MIN = 0.08
_MARGIN_QUARANTINE = 0.05

_SQUARE_R = 1.0
_CIRCLE_R = np.pi / 4.0
_HALF_R = 0.5
_OFFSET_TRI = 1.0 / 6.0
_OFFSET_DIA = 0.0


def _segments(shape: str) -> list[tuple[tuple[float, float], tuple[float, float], float]]:
    c = CENTER
    if shape == "square":
        verts = [(-R, -R), (R, -R), (R, R), (-R, R)]
    elif shape == "diamond":
        verts = [(0.0, -R), (R, 0.0), (0.0, R), (-R, 0.0)]
    elif shape == "triangle":
        verts = [(0.0, -R), (-R, R), (R, R)]
    else:
        raise ValueError(f"unknown shape {shape!r}")
    out = []
    n = len(verts)
    for i in range(n):
        (x0, y0) = verts[i]
        (x1, y1) = verts[(i + 1) % n]
        out.append(((c + x0, c + y0), (c + x1, c + y1), math.atan2(y1 - y0, x1 - x0)))
    return out


def _perimeter_length(shape: str) -> float:
    if shape == "circle":
        return 2.0 * math.pi * R
    return sum(math.hypot(p1[0] - p0[0], p1[1] - p0[1]) for (p0, p1, _a) in _segments(shape))


def _perimeter_points(shape: str, n: int, phase: float) -> list[tuple[float, float, float]]:
    if shape == "circle":
        pts = []
        for i in range(n):
            th = phase + 2.0 * math.pi * i / n
            pts.append((CENTER + R * math.cos(th), CENTER + R * math.sin(th), th + math.pi / 2.0))
        return pts
    segs = _segments(shape)
    seg_len = [math.hypot(p1[0] - p0[0], p1[1] - p0[1]) for (p0, p1, _a) in segs]
    total = sum(seg_len)
    pts = []
    for i in range(n):
        s = (phase + total * i / n) % total
        acc = 0.0
        for (p0, p1, ang), L in zip(segs, seg_len):
            if s <= acc + L or L == 0.0:
                t = (s - acc) / L
                pts.append((p0[0] + t * (p1[0] - p0[0]), p0[1] + t * (p1[1] - p0[1]), ang))
                break
            acc += L
    return pts


def _bars(shape: str, n: int, phase: float) -> list[tuple[float, float, float, float]]:
    out = []
    for (x, y, ang) in _perimeter_points(shape, n, phase):
        dx = math.cos(ang) * BAR_LEN / 2.0
        dy = math.sin(ang) * BAR_LEN / 2.0
        out.append((x - dx, y - dy, x + dx, y + dy))
    return out


def sample_scene(seed: int) -> dict:
    shape = SHAPES[seed % len(SHAPES)]
    rng = np.random.default_rng(seed * 2654435761 % (2**32))
    n = int(rng.integers(18, 25))
    if shape == "circle":
        phase = float(rng.uniform(0, 2.0 * math.pi))
    else:
        phase = float(rng.uniform(0, _perimeter_length(shape)))
    bars = _bars(shape, n, phase)
    return {"shape": shape,
            "bars": [[float(a), float(b), float(c), float(d)] for (a, b, c, d) in bars]}


def _draw_scene(scene: dict) -> Image.Image:
    shape = scene["shape"]
    if shape not in SHAPES:
        raise ValueError(f"unknown shape {shape!r}")
    img = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(img)
    for (x1, y1, x2, y2) in scene["bars"]:
        draw.line([(x1, y1), (x2, y2)], fill=(0, 0, 0), width=STROKE)
    return img


def render(scene: dict) -> Image.Image:
    return _draw_scene(scene)


def analytic_gold(scene: dict) -> str:
    return scene["shape"]


def _convex_hull(points: np.ndarray) -> np.ndarray:
    pts = np.unique(points, axis=0)
    if len(pts) <= 1:
        return pts
    pts = pts[np.lexsort((pts[:, 1], pts[:, 0]))]

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(tuple(p))
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(tuple(p))
    return np.array(lower[:-1] + upper[:-1])


def _hull_features(arr: np.ndarray) -> tuple[float, float]:
    ys, xs = np.where(arr < 120)
    if xs.size < 3:
        return 0.0, 0.0
    hull = _convex_hull(np.stack([xs.astype(float), ys.astype(float)], axis=1))
    if len(hull) < 3:
        return 0.0, 0.0
    n = len(hull)
    xs0 = hull[:, 0]
    ys0 = hull[:, 1]
    cross_term = xs0 * np.roll(ys0, -1) - ys0 * np.roll(xs0, -1)
    area = 0.5 * abs(np.sum(cross_term))
    cx = float(np.sum((xs0 + np.roll(xs0, -1)) * cross_term) / (6.0 * area))
    cy = float(np.sum((ys0 + np.roll(ys0, -1)) * cross_term) / (6.0 * area))
    bw = float(xs.max() - xs.min() + 1)
    bh = float(ys.max() - ys.min() + 1)
    fill = float(area) / (bw * bh)
    bcenter_y = (float(ys.min()) + float(ys.max())) / 2.0
    offset = abs(cy - bcenter_y) / bh
    return fill, offset


def classify_shape(fill: float, offset: float) -> str:
    if fill >= _SQUARE_MIN:
        return "square"
    if fill >= _CIRCLE_MIN:
        return "circle"
    return "triangle" if offset > _OFFSET_MIN else "diamond"


def _shape_scores(fill: float, offset: float) -> list[float]:
    s = {
        "square": 1.0 / (1.0 + 3.0 * abs(fill - _SQUARE_R)),
        "circle": 1.0 / (1.0 + 3.0 * abs(fill - _CIRCLE_R)),
        "diamond": 1.0 / (1.0 + 2.0 * abs(fill - _HALF_R) + 6.0 * abs(offset - _OFFSET_DIA)),
        "triangle": 1.0 / (1.0 + 2.0 * abs(fill - _HALF_R) + 6.0 * abs(offset - _OFFSET_TRI)),
    }
    return sorted(s.values(), reverse=True)


def margin(scene: dict) -> float:
    fill, offset = _hull_features(np.asarray(_draw_scene(scene).convert("RGB"))[..., 0])
    scores = _shape_scores(fill, offset)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < _MARGIN_QUARANTINE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
