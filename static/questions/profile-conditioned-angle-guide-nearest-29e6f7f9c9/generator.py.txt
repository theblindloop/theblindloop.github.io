"""Guide-relative needle-angle measurement world.

Three colored needles radiate from a common central hub on a light panel, each
pointing in a different direction (distinct orientation angle measured from the
horizontal). A neutral dark-gray reference needle also points outward from the
same hub. The decision is a pure measurement: recover, from the rendered pixels,
the orientation angle of every colored needle and the orientation angle of the
gray reference needle, then select the color of the needle whose direction is
CLOSEST to the reference direction (smallest absolute angular difference).

Which color owns the smallest angular difference to the reference is
counterbalanced by shuffling the color labels among the three needle angles and
by randomising the reference direction and the needle angles, so no cheap
positional, colour, or order cue predicts the answer; only the recovered scalar
angular difference does.

Segmenting and orientation-fitting each color separately is reliable because the
needles radiate at well-separated angles and never overlap in the raster.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "angle_guide_nearest-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (78, 78, 90)          # neutral reference needle, distinct from palette + paper
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
CENTER = (240, 175)
NEEDLE_WIDTH = 7                 # stroke width of every colored needle (identical)
REF_WIDTH = 8                    # stroke width of the gray reference needle
HUB_R = 9                        # empty central hub radius (needles stop short of center)
R_INNER = 40.0                   # needle radial start (must clear the hub)
R_OUTER = 150.0                  # needle radial end
ANGLE_MIN = 15                   # minimum needle angle from horizontal (degrees)
ANGLE_MAX = 75                   # maximum needle angle from horizontal (degrees)
MIN_GAP = 16                     # minimum angular separation between any two needles
QUARANTINE_DEG = 4.0             # below this closest-vs-runner angular gap, quarantined
MIN_MARGIN = 5.0                 # emitted scenes must have this minimum angular margin


def sample_scene(seed: int) -> dict:
    """Three well-separated needle angles + a random reference angle + shuffled colors."""
    rng = random.Random(seed * 1049 + 17)
    while True:
        angles = sorted(rng.sample(range(ANGLE_MIN, ANGLE_MAX + 1), len(PALETTE)))
        if all(angles[i + 1] - angles[i] >= MIN_GAP for i in range(len(angles) - 1)):
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    ref = rng.randint(ANGLE_MIN, ANGLE_MAX)
    needles = [{"color": color, "angle": float(angle)} for color, angle in zip(colors, angles)]
    return {
        "needles": needles,
        "reference": {"angle": float(ref)},
        "center": list(CENTER),
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def _endpoints(cx: float, cy: float, angle: float, r1: float, r2: float) -> tuple:
    rad = math.radians(angle)
    return (
        cx + r1 * math.cos(rad),
        cy - r1 * math.sin(rad),
        cx + r2 * math.cos(rad),
        cy - r2 * math.sin(rad),
    )


def render(scene: dict) -> Image.Image:
    """Draw the gray reference needle first, then the colored needles, then the hub."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    cx, cy = scene["center"]
    rx1, ry1, rx2, ry2 = _endpoints(cx, cy, scene["reference"]["angle"], R_INNER, R_OUTER)
    draw.line([rx1, ry1, rx2, ry2], fill=GRAY_REF, width=REF_WIDTH)
    for needle in scene["needles"]:
        x1, y1, x2, y2 = _endpoints(cx, cy, needle["angle"], R_INNER, R_OUTER)
        draw.line([x1, y1, x2, y2], fill=COLOR_RGB[needle["color"]], width=NEEDLE_WIDTH)
    draw.ellipse([cx - HUB_R, cy - HUB_R, cx + HUB_R, cy + HUB_R], fill=BACKGROUND)
    return img


def _winner(scene: dict) -> str:
    ref = scene["reference"]["angle"]
    best = min(scene["needles"], key=lambda n: abs(n["angle"] - ref))
    return best["color"]


def analytic_gold(scene: dict) -> str:
    return _winner(scene)


def margin(scene: dict) -> float:
    """Closest-vs-runner angular gap (degrees) of every needle to the reference."""
    ref = scene["reference"]["angle"]
    diffs = sorted(abs(needle["angle"] - ref) for needle in scene["needles"])
    return float(diffs[1] - diffs[0])


def is_quarantined(scene: dict) -> bool:
    """Near-boundary when the closest-vs-runner angular gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_DEG


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is directly rendered as visible ink."""
    return []
