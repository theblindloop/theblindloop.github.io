"""Independent pixel-only inverse arm for the guide-relative needle world.

This module receives ONLY the rasterized PNG and independently re-measures the
direction of the gray reference needle and the direction of every colored needle,
then returns the color of the needle whose orientation is closest to the
reference. It shares no geometry or grid with renderer.render() and never imports
the renderer, prompts, generator, verifier, or any analytic-gold code.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "angle_guide_nearest-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (78, 78, 90)
BACKGROUND = (250, 250, 250)


def _labels(arr: np.ndarray) -> np.ndarray:
    """Label each pixel to its nearest of {red, blue, green, gray-reference, paper}."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(GRAY_REF, dtype=np.float32), np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _orientation_angle(mask: np.ndarray) -> float:
    """Recover the dominant direction (degrees from horizontal, 0..90) of an ink mask.

    The principal axis of inertia of the mask gives the needle's direction. Because
    every needle is a straight thick ray whose pixels cluster along one line, the
    principal eigenvector of the pixel covariance is a stable, raster-noise-robust
    estimate of its orientation.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 40:
        return float("nan")
    x = xs.astype(np.float64)
    y = ys.astype(np.float64)
    x = x - x.mean()
    y = y - y.mean()
    vals, evec = np.linalg.eigh(np.cov(np.stack([x, y])))
    v = evec[:, int(np.argmax(vals))]
    ang = math.degrees(math.atan2(-v[1], v[0]))  # screen y axis points down
    ang %= 180.0
    if ang > 90.0:
        ang = 180.0 - ang
    return ang


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the needle whose measured direction is closest to the gray one."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    needle_angles: dict[str, float] = {}
    for idx, color in enumerate(PALETTE):
        needle_angles[color] = _orientation_angle(labels == idx)
    ref_angle = _orientation_angle(labels == len(PALETTE))  # gray-reference label
    if not math.isnan(ref_angle):
        valid = {c: a for c, a in needle_angles.items() if not math.isnan(a)}
        if valid:
            return min(valid, key=lambda c: abs(valid[c] - ref_angle))
    return PALETTE[0]
