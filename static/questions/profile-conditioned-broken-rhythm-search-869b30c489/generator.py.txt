"""Deterministic latent sampler, renderer, and analytic arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
LABELS = ("A", "B", "C", "D")
QUARANTINE_MARGIN = 6.0
PALETTES = ((31, 91, 145), (129, 56, 154), (29, 125, 112), (184, 67, 75))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = int(seed) % 4
    defect = rng.randint(10, 15)
    defect_slot = rng.randint(1, 4)
    offsets = []
    for panel in range(4):
        panel_offsets = [0, 36, 72, 108, 144, 180, 216]
        if panel == target:
            panel_offsets[defect_slot] -= defect
        offsets.append(panel_offsets)
    return {
        "panel_offsets": offsets,
        "dash_color": list(PALETTES[rng.randrange(len(PALETTES))]),
        "shift_x": rng.randint(-7, 7),
        "shift_y": rng.randint(-6, 6),
        "dash_width": rng.choice((8, 10, 12)),
        "dash_height": rng.choice((42, 48, 54)),
    }


def _scores(scene: dict) -> list[float]:
    scores = []
    for offsets in scene["panel_offsets"]:
        gaps = [float(b - a) for a, b in zip(offsets, offsets[1:])]
        median = sorted(gaps)[len(gaps) // 2]
        scores.append(max(abs(gap - median) for gap in gaps))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] <= 0 or scores[order[0]] == scores[order[1]]:
        raise ValueError("scene lacks a unique broken rhythm")
    return LABELS[order[0]]


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene), reverse=True)
    return float(min(ordered[0], ordered[0] - ordered[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 247, 243))
    draw = ImageDraw.Draw(image)
    sx, sy = int(scene["shift_x"]), int(scene["shift_y"])
    centers = ((170, 133), (470, 133), (170, 353), (470, 353))
    color = tuple(int(v) for v in scene["dash_color"])
    dash_w, dash_h = int(scene["dash_width"]), int(scene["dash_height"])
    font = ImageFont.load_default()
    for index, ((cx, cy), offsets) in enumerate(zip(centers, scene["panel_offsets"])):
        cx, cy = cx + sx, cy + sy
        draw.rounded_rectangle((cx - 135, cy - 92, cx + 135, cy + 92), radius=13,
                               fill=(255, 255, 255), outline=(158, 163, 168), width=3)
        draw.text((cx - 125, cy - 80), LABELS[index], fill=(54, 57, 62), font=font)
        left = cx - 108
        for offset in offsets:
            x = left + int(offset)
            draw.rectangle((x - dash_w // 2, cy - dash_h // 2,
                            x + (dash_w - 1) // 2, cy + (dash_h - 1) // 2), fill=color)
    return image


def translated(scene: dict, dx: int, dy: int) -> dict:
    twin = deepcopy(scene)
    twin["shift_x"] += int(dx)
    twin["shift_y"] += int(dy)
    return twin
