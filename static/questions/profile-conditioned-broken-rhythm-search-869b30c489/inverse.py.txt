"""Independent pixel-only inverse arm."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    remaining = {tuple(p) for p in np.argwhere(mask)}
    found = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        points = [start]
        while queue:
            y, x = queue.popleft()
            for q in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= q[0] < height and 0 <= q[1] < width and q in remaining:
                    remaining.remove(q)
                    queue.append(q)
                    points.append(q)
        if len(points) >= 250:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    flat = array.reshape(-1, 3)
    spread = flat.max(axis=1) - flat.min(axis=1)
    vivid = flat[spread > 55]
    if len(vivid) < 4000:
        raise ValueError("insufficient dash ink")
    colors, counts = np.unique(vivid, axis=0, return_counts=True)
    color = colors[int(np.argmax(counts))]
    if int(counts.max()) < 4000:
        raise ValueError("no dominant dash color")
    components = _components(np.all(array == color, axis=2))
    if len(components) != 28:
        raise ValueError("expected twenty-eight visible dashes")
    xs = sorted(x for x, _, _ in components)
    ys = sorted(y for _, y, _ in components)
    x_cut = (xs[13] + xs[14]) / 2.0
    y_cut = (ys[13] + ys[14]) / 2.0
    panels = [[] for _ in range(4)]
    for x, y, area in components:
        panel = (2 if y > y_cut else 0) + (1 if x > x_cut else 0)
        panels[panel].append((x, area))
    if any(len(panel) != 7 for panel in panels):
        raise ValueError("could not assign seven dashes per panel")
    scores = []
    for panel in panels:
        centers = sorted(x for x, _ in panel)
        gaps = [b - a for a, b in zip(centers, centers[1:])]
        median = sorted(gaps)[len(gaps) // 2]
        scores.append(max(abs(gap - median) for gap in gaps))
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] < 6.0 or scores[order[0]] - scores[order[1]] < 6.0:
        raise ValueError("ambiguous spacing rhythm; abstaining")
    return LABELS[order[0]]
