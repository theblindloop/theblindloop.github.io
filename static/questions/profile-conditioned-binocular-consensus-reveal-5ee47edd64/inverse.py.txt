"""Independent pixel-only inverse arm for binocular_consensus_reveal.

Receives ONLY the final PNG. It reads the two 32-by-32 noise grids (left panel P,
right panel Q) from fixed declared geometry, executes the declared cipher by
comparing them cell by cell (a cell agrees when P and Q carry the same shade),
and classifies the surviving agreement mask against the two declared reference
symbols (A = filled circle, B = filled square). It reads no scene data and no
analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "binocular_consensus_reveal-oracle-0.1.0"

NCELL = 32
CELL_PX = 8
PANEL = NCELL * CELL_PX
Q_X0 = 280

_A: np.ndarray | None = None
_B: np.ndarray | None = None


def _glyph_mask(glyph: str) -> np.ndarray:
    m = np.zeros((NCELL, NCELL), dtype=bool)
    i, j = np.indices((NCELL, NCELL))
    cy = cx = (NCELL - 1) / 2.0
    if glyph == "A":
        m[(i - cy) ** 2 + (j - cx) ** 2 <= 64.0] = True
    else:
        m[(np.abs(i - cy) <= 7.0) & (np.abs(j - cx) <= 7.0)] = True
    return m


def _ref(glyph: str) -> np.ndarray:
    global _A, _B
    if glyph == "A":
        if _A is None:
            _A = _glyph_mask("A")
        return _A
    if _B is None:
        _B = _glyph_mask("B")
    return _B


def _bits(luma: np.ndarray, c0: int) -> np.ndarray:
    block = luma[0:PANEL, c0:c0 + PANEL].reshape(NCELL, CELL_PX, NCELL, CELL_PX)
    return block.mean(axis=(1, 3)) < 128


def reconstruct(image: Image.Image) -> np.ndarray:
    luma = np.asarray(image.convert("L"), dtype=np.uint8)
    p = _bits(luma, 0)
    q = _bits(luma, Q_X0)
    return p == q


def decision_from_image(image: Image.Image) -> str:
    rec = reconstruct(image)
    diffA = int(np.count_nonzero(rec != _ref("A")))
    diffB = int(np.count_nonzero(rec != _ref("B")))
    if diffA == diffB:
        return "ABSTAIN"
    return "A" if diffA < diffB else "B"


def reveal_margin(image: Image.Image) -> float:
    rec = reconstruct(image)
    diffA = float(np.count_nonzero(rec != _ref("A")))
    diffB = float(np.count_nonzero(rec != _ref("B")))
    return abs(diffB - diffA)
