"""binocular_consensus_reveal: two-panel consensus (equality) reveal world.

Declared transform: compare the two noise panels cell by cell and read the cells
where the two panels AGREE (both dark or both light) - the consensus region.

The image shows, left and right, two 32-by-32 grids of dark and light cells, P
and Q, that each look like pure random noise with no symbol visible in either
panel alone. A hidden symbol was encoded between the two panels: wherever the
symbol is drawn, the two panels carry the SAME random shade (both dark or both
light), and everywhere else the two panels carry OPPOSITE shades (one dark, one
light). Undoing the declared cipher means comparing P and Q cell by cell; the
matching (agreed) shade traces the surviving abstract symbol: a filled circle
(A) or a filled square (B).

Neither panel alone, nor any color, position, or per-panel region, reveals the
symbol; only computing the cell-wise equality between the two panels exposes it.

Latent scene z = {glyph, noise_p}. label = glyph. noise_p is the per-scene random
base panel; panel Q is derived as the same shade as P inside the glyph and the
opposite shade outside it. Only the glyph is the label.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "binocular_consensus_reveal-0.1.0"

NCELL = 32
CELL_PX = 8
PANEL = NCELL * CELL_PX   # 256
Q_X0 = 280                 # right panel origin
W = Q_X0 + PANEL          # 536
H = PANEL                 # 256

NULL_TOL = 0.0             # quarantine threshold (cells) on the A/B reconstruction gap


def glyph_mask(glyph: str) -> np.ndarray:
    """(NCELL, NCELL) bool: True = symbol ink."""
    m = np.zeros((NCELL, NCELL), dtype=bool)
    i, j = np.indices((NCELL, NCELL))
    cy = cx = (NCELL - 1) / 2.0
    if glyph == "A":
        r = 8.0
        m[(i - cy) ** 2 + (j - cx) ** 2 <= r * r] = True
    else:
        half = 7.0
        m[(np.abs(i - cy) <= half) & (np.abs(j - cx) <= half)] = True
    return m


def _noise_p(seed: int) -> np.ndarray:
    rng = np.random.default_rng((int(seed) & 0xFFFFFFFF) ^ 0x3C3C3C3C)
    return rng.integers(0, 2, size=(NCELL, NCELL), dtype=np.uint8) > 0


def build_scene(glyph: str, noise_p: np.ndarray) -> dict:
    return {"glyph": glyph, "noise_p": np.asarray(noise_p, dtype=bool)}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed) & 0xFFFFFFFF)
    glyph = "A" if rng.integers(0, 2) else "B"
    noise_p = rng.integers(0, 2, size=(NCELL, NCELL), dtype=np.uint8) > 0
    return build_scene(glyph=glyph, noise_p=noise_p)


def _p_bits(scene: dict) -> np.ndarray:
    return np.asarray(scene["noise_p"], dtype=bool)


def _q_bits(scene: dict) -> np.ndarray:
    p = _p_bits(scene)
    return np.where(glyph_mask(scene["glyph"]), p, ~p)


def _render_panel(canvas: np.ndarray, bits: np.ndarray, r0: int, c0: int) -> None:
    for r in range(NCELL):
        for c in range(NCELL):
            color = 0 if bits[r, c] else 255
            canvas[r0 + r * CELL_PX:r0 + (r + 1) * CELL_PX,
                   c0 + c * CELL_PX:c0 + (c + 1) * CELL_PX] = color


def render(scene: dict) -> Image.Image:
    arr = np.full((H, W, 3), 255, dtype=np.uint8)
    _render_panel(arr, _p_bits(scene), 0, 0)
    _render_panel(arr, _q_bits(scene), 0, Q_X0)
    return Image.fromarray(arr, "RGB")


def _extract(luma: np.ndarray, r0: int, c0: int) -> np.ndarray:
    block = luma[r0:r0 + PANEL, c0:c0 + PANEL].reshape(
        NCELL, CELL_PX, NCELL, CELL_PX
    )
    return block.mean(axis=(1, 3)) < 128


def reconstruct(scene: dict) -> np.ndarray:
    return _p_bits(scene) == _q_bits(scene)


def margin(scene: dict) -> float:
    rec = reconstruct(scene)
    ga = glyph_mask("A")
    gb = glyph_mask("B")
    diffA = float(np.count_nonzero(rec != ga))
    diffB = float(np.count_nonzero(rec != gb))
    return round(diffB - diffA, 6)


def analytic_gold(scene: dict) -> str:
    return scene["glyph"]


def is_quarantined(scene: dict) -> bool:
    return abs(margin(scene)) < NULL_TOL


def latent_symmetries(scene: dict):
    """not_applicable: every latent field directly changes the rendered pixels
    (glyph decides which cells agree, noise_p sets panel P shades and hence the
    derived panel Q shades), so no two distinct latent vectors reproduce the same
    raster with different labels. Declared empty."""
    return iter(())
