"""Deterministic latent sampler, raster renderer, and analytic route answer."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = "#ffffff"
UNDERLAY = "#475569"
CENTERLINE = "#111827"
JUNCTION = "#9ca3af"
GREEN = "#16a34a"
COLORS = {"orange": "#f97316", "purple": "#8b5cf6", "blue": "#2563eb"}

_NODES = {
    "r": (50, 300),
    "j0": (180, 300),
    "j1": (180, 150),
    "t0": (180, 450),
    "t1": (50, 150),
    "t2": (450, 150),
}
_EDGE_PAIRS = [("r", "j0"), ("j0", "j1"), ("j0", "t0"), ("j1", "t1"), ("j1", "t2")]


def _transform(point: tuple[int, int], turns: int) -> list[int]:
    x, y = point[0] - 256, point[1] - 256
    for _ in range(turns % 4):
        x, y = -y, x
    return [x + 256, y + 256]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    turns = int(seed) % 4
    nodes = {name: _transform(point, turns) for name, point in _NODES.items()}
    edges = [
        {"a": a, "b": b, "points": [list(nodes[a]), list(nodes[b])]}
        for a, b in _EDGE_PAIRS
    ]
    winner = ("orange", "purple", "blue")[int(seed) % 3]
    others = [color for color in COLORS if color != winner]
    rng.shuffle(others)
    terminal_colors = {"t0": others[0], "t1": others[1], "t2": winner}
    return {
        "canvas": [CANVAS, CANVAS],
        "nodes": nodes,
        "edges": edges,
        "root_node": "r",
        "junction_nodes": ["j0", "j1"],
        "terminal_nodes": ["t0", "t1", "t2"],
        "terminal_colors": terminal_colors,
    }


def _adjacency(scene: dict) -> dict[str, list[str]]:
    nodes = scene["nodes"]
    adjacency = {name: [] for name in nodes}
    for edge in scene["edges"]:
        a, b = edge["a"], edge["b"]
        if list(edge["points"][0]) != list(nodes[a]) or list(edge["points"][-1]) != list(nodes[b]):
            raise ValueError("edge endpoints disagree with nodes")
        adjacency[a].append(b)
        adjacency[b].append(a)
    return adjacency


def _cross(nodes: dict, previous: str, current: str, nxt: str) -> float:
    px, py = nodes[previous]
    cx, cy = nodes[current]
    nx, ny = nodes[nxt]
    incoming = (cx - px, cy - py)
    outgoing = (nx - cx, ny - cy)
    return float(incoming[0] * outgoing[1] - incoming[1] * outgoing[0])


def _selected_terminal(scene: dict) -> str:
    nodes = scene["nodes"]
    adjacency = _adjacency(scene)
    previous = scene["root_node"]
    junctions = list(scene["junction_nodes"])
    terminals = set(scene["terminal_nodes"])
    current = junctions[0]
    if current not in adjacency[previous]:
        raise ValueError("root is not attached to first junction")
    for index, wanted_sign in enumerate((-1, 1)):
        if current != junctions[index]:
            raise ValueError("declared junction order is inconsistent")
        choices = [node for node in adjacency[current] if node != previous]
        signed = [(node, _cross(nodes, previous, current, node)) for node in choices]
        matches = [node for node, value in signed if value * wanted_sign > 0]
        if len(matches) != 1:
            raise ValueError("junction does not expose one left and one right branch")
        previous, current = current, matches[0]
    if current not in terminals:
        raise ValueError("turn sequence did not reach a terminal")
    return current


def analytic_gold(scene: dict) -> str:
    terminal = _selected_terminal(scene)
    return scene["terminal_colors"][terminal]


def margin(scene: dict) -> float:
    nodes = scene["nodes"]
    adjacency = _adjacency(scene)
    previous = scene["root_node"]
    margins = []
    for current, wanted_sign in zip(scene["junction_nodes"], (-1, 1)):
        choices = [node for node in adjacency[current] if node != previous]
        angles = []
        for node in choices:
            px, py = nodes[previous]
            cx, cy = nodes[current]
            nx, ny = nodes[node]
            a = (cx - px, cy - py)
            b = (nx - cx, ny - cy)
            cosine = (a[0] * b[0] + a[1] * b[1]) / (math.hypot(*a) * math.hypot(*b))
            angles.append(abs(90.0 - math.degrees(math.acos(max(-1.0, min(1.0, cosine))))))
        margins.append(90.0 - max(angles))
        signed = [(node, _cross(nodes, previous, current, node)) for node in choices]
        previous = current
        current = next(node for node, value in signed if value * wanted_sign > 0)
    return float(min(margins))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edge_lengths = [
        math.hypot(nodes[edge["a"]][0] - nodes[edge["b"]][0], nodes[edge["a"]][1] - nodes[edge["b"]][1])
        for edge in scene["edges"]
    ]
    return margin(scene) < 45.0 or min(edge_lengths) < 60.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    nodes = scene["nodes"]
    edges = scene["edges"]
    root = scene["root_node"]
    junctions = scene["junction_nodes"]
    terminals = scene["terminal_nodes"]
    terminal_colors = scene["terminal_colors"]
    image = Image.new("RGB", (int(width), int(height)), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for edge in edges:
        draw.line([tuple(point) for point in edge["points"]], fill=UNDERLAY, width=11)
    for junction in junctions:
        x, y = nodes[junction]
        draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=JUNCTION)
    for edge in edges:
        draw.line([tuple(point) for point in edge["points"]], fill=CENTERLINE, width=2)
    rx, ry = nodes[root]
    draw.ellipse((rx - 15, ry - 15, rx + 15, ry + 15), fill=BACKGROUND, outline=GREEN, width=6)
    for terminal in terminals:
        x, y = nodes[terminal]
        radius = 14
        draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=COLORS[terminal_colors[terminal]], outline=BACKGROUND, width=2)
    return image
