# These are deliberately equal-scale, hard-edged glyphs. They are the public
# identity interface recovered by the pixel arm, not hidden metadata.
LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)

# The three code marks have distinct silhouettes and are repeated next to the
# visible LOW/MID/HIGH names in the key. Their semantic index is stable even
# when the key's physical order and palette are permuted.
STATE_CODE_PATTERNS = (
    ("01110", "10001", "10001", "10001", "10001", "10001", "01110"),
    ("11111", "00100", "00100", "00100", "00100", "00100", "00100"),
    ("11110", "00001", "00010", "00100", "01000", "10000", "11111"),
)
STATE_CODE_SCALE = 2


def _pattern_points(pattern: tuple[str, ...], scale: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * scale + dx, y * scale + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(scale)
        for dy in range(scale)
    )


LABEL_POINTS = tuple(_pattern_points(pattern, LABEL_SCALE) for pattern in LABEL_PATTERNS)
STATE_CODE_POINTS = tuple(
    _pattern_points(pattern, STATE_CODE_SCALE) for pattern in STATE_CODE_PATTERNS
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    ) if bold else (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
    )
    for name in names:
        if Path(name).is_file():
            return ImageFont.truetype(name, size)
    return ImageFont.load_default()


def _draw_points(
    draw: ImageDraw.ImageDraw,
    origin: tuple[int, int],
    points: frozenset[tuple[int, int]],
    fill: tuple[int, int, int],
) -> None:
    ox, oy = origin
    for x, y in points:
        draw.point((ox + x, oy + y), fill=fill)


def _scores_from_states(states: list[list[int]]) -> list[int]:
    if not isinstance(states, list) or len(states) != PANEL_COUNT:
        raise ValueError("states must contain seven ordered panels")
    checked: list[list[int]] = []
    for panel in states:
        if not isinstance(panel, list) or len(panel) != IDENTITY_COUNT:
            raise ValueError("each state panel must contain four identities")
        values = [int(value) for value in panel]
        if any(value not in range(STATE_COUNT) for value in values):
            raise ValueError("states must use LOW=0, MID=1, HIGH=2")
        checked.append(values)
    return [
        sum(abs(checked[panel + 1][label] - checked[panel][label]) for panel in range(PANEL_COUNT - 1))
        for label in range(IDENTITY_COUNT)
    ]


def _score_gap(scores: list[int]) -> int:
    ordered = sorted(scores, reverse=True)
    return ordered[0] - ordered[1]


def _render_scene_fields(scene: dict) -> tuple[list[list[int]], list[list[int]], list[tuple[int, int, int]], list[int], list[tuple[int, int, int]], int]:
    states = scene["states"]
    positions = scene["positions"]
    _scores_from_states(states)
    if not isinstance(positions, list) or len(positions) != PANEL_COUNT:
        raise ValueError("positions must contain seven ordered panels")
    checked_positions: list[list[int]] = []
    for panel in positions:
        if not isinstance(panel, list) or len(panel) != IDENTITY_COUNT:
            raise ValueError("each position panel must contain four identities")
        values = [int(value) for value in panel]
        if sorted(values) != list(range(SLOT_COUNT)):
            raise ValueError("each position panel must be a slot permutation")
        checked_positions.append(values)
    palette = [tuple(int(channel) for channel in color) for color in scene["palette"]]
    if len(palette) != STATE_COUNT or any(len(color) != 3 for color in palette):
        raise ValueError("palette must contain three RGB colors")
    key_order = [int(value) for value in scene["key_order"]]
    if sorted(key_order) != list(range(STATE_COUNT)):
        raise ValueError("key_order must be a permutation of the three states")
    tints = [tuple(int(channel) for channel in color) for color in scene["panel_tints"]]
    if len(tints) != PANEL_COUNT or any(len(color) != 3 for color in tints):
        raise ValueError("panel_tints must contain seven RGB colors")
    box_size = int(scene["state_box_size"])
    if box_size not in (24, 26, 28):
        raise ValueError("state_box_size is outside the supported public range")
    return states, checked_positions, palette, key_order, tints, box_size


def render(scene: dict) -> Image.Image:
    states, positions, palette, key_order, tints, box_size = _render_scene_fields(scene)
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((28, 25), "STATE TRAVEL", font=_font(34, True), fill=INK)
    draw.text(
        (30, 72),
        "Follow each label through P1–P7. The key gives the ordered state scale.",
        font=_font(20),
        fill=MUTED,
    )
    draw.text((30, 120), "KEY", font=_font(18, True), fill=INK)
    for key_slot, state in enumerate(key_order):
        x = KEY_XS[key_slot]
        y = KEY_Y
        draw.rectangle(
            (x, y, x + box_size - 1, y + box_size - 1),
            fill=palette[state],
            outline=STATE_OUTLINE,
            width=2,
        )
        _draw_points(
            draw,
            (x + STATE_CODE_X_OFFSET, y + STATE_CODE_Y_OFFSET),
            STATE_CODE_POINTS[state],
            INK,
        )
        draw.text((x + 74, y + 3), STATE_NAMES[state], font=_font(19, True), fill=INK)

    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        draw.rounded_rectangle(
            (panel_x, PANEL_TOP, panel_x + PANEL_WIDTH - 1, PANEL_TOP + PANEL_HEIGHT - 1),
            radius=12,
            fill=tints[panel],
            outline=OUTLINE,
            width=2,
        )
        draw.text(
            (panel_x + 14, PANEL_TOP + 15),
            f"P{panel + 1}",
            font=_font(25, True),
            fill=INK,
        )
        draw.text(
            (panel_x + 82, PANEL_TOP + 20),
            "same four labels",
            font=_font(14),
            fill=MUTED,
        )
        for label in range(IDENTITY_COUNT):
            slot = positions[panel][label]
            slot_x, slot_y = SLOT_ORIGINS[slot]
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            draw.rounded_rectangle(
                (card_x, card_y, card_x + CARD_WIDTH - 1, card_y + CARD_HEIGHT - 1),
                radius=10,
                fill=CARD,
                outline=OUTLINE,
                width=2,
            )
            _draw_points(
                draw,
                (card_x + LABEL_X_OFFSET, card_y + LABEL_Y_OFFSET),
                LABEL_POINTS[label],
                INK,
            )
            state = int(states[panel][label])
            draw.rectangle(
                (
                    card_x + STATE_X_OFFSET,
                    card_y + STATE_Y_OFFSET,
                    card_x + STATE_X_OFFSET + box_size - 1,
                    card_y + STATE_Y_OFFSET + box_size - 1,
                ),
                fill=palette[state],
                outline=STATE_OUTLINE,
                width=2,
            )
        draw.text(
            (panel_x + 16, PANEL_TOP + 505),
            "identity slots are shuffled",
            font=_font(14),
            fill=MUTED,
        )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores_from_states(scene["states"])
    ordered = sorted(scores, reverse=True)
    winners = [label for label, score in enumerate(scores) if score == ordered[0]]
    if len(winners) != 1 or ordered[0] - ordered[1] < 2:
        raise ValueError("scene has no unique path-length winner")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    return float(_score_gap(_scores_from_states(scene["states"])))


def is_quarantined(scene: dict) -> bool:
    scores = _scores_from_states(scene["states"])
    ordered = sorted(scores, reverse=True)
    return len([score for score in scores if score == ordered[0]]) != 1 or ordered[0] - ordered[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _path_score(values: list[int]) -> int:
    return sum(abs(values[index + 1] - values[index]) for index in range(len(values) - 1))


def _random_trajectory(rng: random.Random) -> list[int]:
    return [rng.randrange(STATE_COUNT) for _ in range(PANEL_COUNT)]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with a unique weighted-travel winner."""
    seed = int(seed)
    rng = random.Random(seed)
    winner = seed % IDENTITY_COUNT
    chosen: list[list[int]] | None = None
    for _ in range(20000):
        target = _random_trajectory(rng)
        target_score = _path_score(target)
        if target_score < 8 or len(set(target)) < 3:
            continue
        states: list[list[int] | None] = [None] * IDENTITY_COUNT
        states[winner] = target
        possible = True
        for label in range(IDENTITY_COUNT):
            if label == winner:
                continue
            trajectory: list[int] | None = None
            for _ in range(200):
                trial = _random_trajectory(rng)
                if len(set(trial)) >= 2 and _path_score(trial) <= target_score - 2:
                    trajectory = trial
                    break
            if trajectory is None:
                possible = False
                break
            states[label] = trajectory
        if possible:
            trajectories = [list(values) for values in states if values is not None]
            chosen = [
                [trajectories[label][panel] for label in range(IDENTITY_COUNT)]
                for panel in range(PANEL_COUNT)
            ]
            break
    if chosen is None:
        raise RuntimeError("could not sample a separated state path scene")
    positions = [rng.sample(list(range(SLOT_COUNT)), SLOT_COUNT) for _ in range(PANEL_COUNT)]
    palette = [list(color) for color in PALETTES[rng.randrange(len(PALETTES))]]
    key_order = list(range(STATE_COUNT))
    rng.shuffle(key_order)
    panel_tints = [list(PANEL_TINTS[rng.randrange(len(PANEL_TINTS))]) for _ in range(PANEL_COUNT)]
    box_size = (24, 26, 28)[rng.randrange(3)]
    scene = {
        "states": chosen,
        "positions": positions,
        "palette": palette,
        "key_order": key_order,
        "panel_tints": panel_tints,
        "state_box_size": box_size,
    }
    if analytic_gold(scene) != LABELS[winner]:
        raise AssertionError("sampler lost its requested balanced winner")
    return scene
