"""Independent pixel-only inverse for the weighted state-path world."""

from __future__ import annotations

from PIL import Image


PANEL_COUNT = 7
IDENTITY_COUNT = 4
STATE_COUNT = 3
LABELS = ("A", "B", "C", "D")
CANVAS = (1940, 820)
INK = (31, 48, 61)
PANEL_X0 = 24
PANEL_TOP = 222
PANEL_WIDTH = 260
PANEL_GAP = 8
SLOT_ORIGINS = ((14, 88), (134, 88), (14, 332), (134, 332))
LABEL_X_OFFSET = 14
LABEL_Y_OFFSET = 44
LABEL_SCALE = 3
STATE_X_OFFSET = 76
STATE_Y_OFFSET = 43
KEY_XS = (470, 720, 970)
KEY_Y = 112
STATE_CODE_X_OFFSET = 50
STATE_CODE_Y_OFFSET = 5
SWATCH_SAMPLE_OFFSET = 13

LABEL_PATTERNS = (
    ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
)
STATE_CODE_PATTERNS = (
    ("01110", "10001", "10001", "10001", "10001", "10001", "01110"),
    ("11111", "00100", "00100", "00100", "00100", "00100", "00100"),
    ("11110", "00001", "00010", "00100", "01000", "10000", "11111"),
)


def _pattern_points(pattern: tuple[str, ...], scale: int) -> frozenset[tuple[int, int]]:
    return frozenset(
        (x * scale + dx, y * scale + dy)
        for y, row in enumerate(pattern)
        for x, value in enumerate(row)
        if value == "1"
        for dx in range(scale)
        for dy in range(scale)
    )


LABEL_POINTS = tuple(_pattern_points(pattern, LABEL_SCALE) for pattern in LABEL_PATTERNS)
STATE_CODE_POINTS = tuple(_pattern_points(pattern, 2) for pattern in STATE_CODE_PATTERNS)


def _read_label(frame: Image.Image, origin: tuple[int, int]) -> int:
    ox, oy = origin
    observed = frozenset(
        (x, y)
        for x in range(15)
        for y in range(21)
        if frame.getpixel((ox + x, oy + y)) == INK
    )
    matches = [index for index, points in enumerate(LABEL_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("identity glyph is missing, altered, or ambiguous")
    return matches[0]


def _read_state_code(frame: Image.Image, origin: tuple[int, int]) -> int:
    ox, oy = origin
    observed = frozenset(
        (x, y)
        for x in range(10)
        for y in range(14)
        if frame.getpixel((ox + x, oy + y)) == INK
    )
    matches = [state for state, points in enumerate(STATE_CODE_POINTS) if observed == points]
    if len(matches) != 1:
        raise ValueError("state key code is missing or ambiguous")
    return matches[0]


def _decode_key(frame: Image.Image) -> dict[tuple[int, int, int], int]:
    mapping: dict[tuple[int, int, int], int] = {}
    for key_x in KEY_XS:
        color = frame.getpixel((key_x + SWATCH_SAMPLE_OFFSET, KEY_Y + SWATCH_SAMPLE_OFFSET))
        if color in mapping:
            raise ValueError("state key repeats one palette color")
        state = _read_state_code(frame, (key_x + STATE_CODE_X_OFFSET, KEY_Y + STATE_CODE_Y_OFFSET))
        mapping[color] = state
    if len(mapping) != STATE_COUNT or set(mapping.values()) != set(range(STATE_COUNT)):
        raise ValueError("state key does not expose three distinct states")
    return mapping


def decision_from_image(image: Image.Image) -> str:
    """Decode only the final PNG and return the unique weighted-travel winner."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    state_key = _decode_key(frame)
    tables: list[dict[int, int]] = []
    for panel in range(PANEL_COUNT):
        panel_x = PANEL_X0 + panel * (PANEL_WIDTH + PANEL_GAP)
        table: dict[int, int] = {}
        for slot_x, slot_y in SLOT_ORIGINS:
            card_x = panel_x + slot_x
            card_y = PANEL_TOP + slot_y
            label = _read_label(frame, (card_x + LABEL_X_OFFSET, card_y + LABEL_Y_OFFSET))
            if label in table:
                raise ValueError("a panel repeats one identity")
            color = frame.getpixel(
                (card_x + STATE_X_OFFSET + SWATCH_SAMPLE_OFFSET, card_y + STATE_Y_OFFSET + SWATCH_SAMPLE_OFFSET)
            )
            if color not in state_key:
                raise ValueError("tile state swatch is missing or not declared in the key")
            table[label] = state_key[color]
        if set(table) != set(range(IDENTITY_COUNT)):
            raise ValueError("a panel does not contain all four identities")
        tables.append(table)
    scores = [
        sum(abs(tables[panel + 1][label] - tables[panel][label]) for panel in range(PANEL_COUNT - 1))
        for label in range(IDENTITY_COUNT)
    ]
    ordered = sorted(scores, reverse=True)
    winners = [label for label, score in enumerate(scores) if score == ordered[0]]
    if len(winners) != 1 or ordered[0] - ordered[1] < 2:
        raise ValueError("image does not expose one unique weighted-travel winner")
    return LABELS[winners[0]]
