"""Independent image-only inverse arm for directed route matching."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_BORDER = np.array((174, 184, 194), dtype=np.int16)
_COLORS = (
    np.array((43, 58, 73), dtype=np.int16),
    np.array((14, 143, 133), dtype=np.int16),
    np.array((230, 111, 81), dtype=np.int16),
)
_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            yy, xx = queue.popleft()
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append((min(xs), min(ys), max(xs) + 1, max(ys) + 1, len(xs)))
    return result


def _find_cards(array: np.ndarray) -> list[tuple[int, int, int, int]]:
    border = np.max(np.abs(array.astype(np.int16) - _BORDER), axis=2) <= 2
    boxes = []
    for x0, y0, x1, y1, area in _components(border):
        if 170 <= x1 - x0 <= 190 and 200 <= y1 - y0 <= 220 and area > 1000:
            boxes.append((x0, y0, x1, y1))
    if len(boxes) != 5:
        raise ValueError(f"expected five visible cards, found {len(boxes)}")
    boxes.sort(key=lambda box: (box[1], box[0]))
    return boxes


def _object_channels(array: np.ndarray, box) -> np.ndarray:
    x0, y0, x1, y1 = box
    crop = array[y0 + 28 : y1 - 5, x0 + 5 : x1 - 5].astype(np.int16)
    channels = np.stack([np.max(np.abs(crop - color), axis=2) <= 2 for color in _COLORS])
    union = channels.any(axis=0)
    ys, xs = np.nonzero(union)
    if len(xs) < 300 or not channels[1].any() or not channels[2].any():
        raise ValueError("route or endpoint evidence is missing")
    return channels[:, ys.min() : ys.max() + 1, xs.min() : xs.max() + 1]


def _aligned_score(left: np.ndarray, right: np.ndarray) -> float:
    best = 0.0
    for turns in range(4):
        rotated = np.rot90(left, turns, axes=(1, 2))
        height = max(rotated.shape[1], right.shape[1]) + 8
        width = max(rotated.shape[2], right.shape[2]) + 8
        a = np.zeros((3, height, width), dtype=bool)
        b = np.zeros_like(a)
        ay = (height - rotated.shape[1]) // 2
        ax = (width - rotated.shape[2]) // 2
        by = (height - right.shape[1]) // 2
        bx = (width - right.shape[2]) // 2
        a[:, ay : ay + rotated.shape[1], ax : ax + rotated.shape[2]] = rotated
        for dy in range(-2, 3):
            for dx in range(-2, 3):
                b.fill(False)
                y = by + dy
                x = bx + dx
                if y < 0 or x < 0 or y + right.shape[1] > height or x + right.shape[2] > width:
                    continue
                b[:, y : y + right.shape[1], x : x + right.shape[2]] = right
                intersection = np.logical_and(a, b).sum()
                union = np.logical_or(a, b).sum()
                best = max(best, float(intersection / union))
    return best


def _decision_and_margin(image: Image.Image) -> tuple[str, float, list[float]]:
    array = np.asarray(image.convert("RGB"))
    cards = _find_cards(array)
    reference = _object_channels(array, cards[0])
    candidates = sorted(cards[1:], key=lambda box: box[0])
    scores = [_aligned_score(reference, _object_channels(array, box)) for box in candidates]
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    gap = scores[order[0]] - scores[order[1]]
    if scores[order[0]] < 0.80 or gap < 0.05:
        raise ValueError(f"ambiguous route match: scores={scores}, gap={gap:.4f}")
    return _LABELS[order[0]], gap, scores


def decision_from_image(image: Image.Image) -> str:
    return _decision_and_margin(image)[0]
