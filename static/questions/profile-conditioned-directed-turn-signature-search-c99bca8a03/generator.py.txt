"""Deterministic scene sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 620
CARD_W, CARD_H = 180, 210
STEP = 34
LABELS = ("A", "B", "C", "D")
ROUTE = (43, 58, 73)
START = (14, 143, 133)
END = (230, 111, 81)
BORDER = (174, 184, 194)
TEXT = (48, 58, 68)
PAPER = (250, 249, 246)
CARD = (255, 255, 255)


def _vertices(word: str) -> list[tuple[int, int]]:
    directions = ((1, 0), (0, 1), (-1, 0), (0, -1))
    direction = 0
    points = [(0, 0)]
    for index in range(5):
        x, y = points[-1]
        dx, dy = directions[direction]
        points.append((x + STEP * dx, y + STEP * dy))
        if index < 4:
            direction = (direction + (3 if word[index] == "L" else 1)) % 4
    return points


def _segments_intersect(a, b, c, d) -> bool:
    if a[0] == b[0] and c[1] == d[1]:
        return min(c[0], d[0]) <= a[0] <= max(c[0], d[0]) and min(a[1], b[1]) <= c[1] <= max(a[1], b[1])
    if a[1] == b[1] and c[0] == d[0]:
        return min(a[0], b[0]) <= c[0] <= max(a[0], b[0]) and min(c[1], d[1]) <= a[1] <= max(c[1], d[1])
    if a[0] == b[0] == c[0] == d[0]:
        return max(min(a[1], b[1]), min(c[1], d[1])) <= min(max(a[1], b[1]), max(c[1], d[1]))
    if a[1] == b[1] == c[1] == d[1]:
        return max(min(a[0], b[0]), min(c[0], d[0])) <= min(max(a[0], b[0]), max(c[0], d[0]))
    return False


def _simple(word: str) -> bool:
    points = _vertices(word)
    return not any(
        _segments_intersect(points[i], points[i + 1], points[j], points[j + 1])
        for i in range(5)
        for j in range(i + 2, 5)
    )


VALID_WORDS = tuple(
    word for word in map("".join, itertools.product("LR", repeat=4)) if _simple(word)
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = rng.choice(VALID_WORDS)
    distractors = list(rng.sample([w for w in VALID_WORDS if w != reference], 3))
    target_label = LABELS[int(seed) % len(LABELS)]
    rng.shuffle(distractors)
    candidate_turns = {}
    cursor = 0
    for label in LABELS:
        if label == target_label:
            candidate_turns[label] = reference
        else:
            candidate_turns[label] = distractors[cursor]
            cursor += 1
    route_rotations = {"REFERENCE": rng.randrange(4)}
    route_rotations.update({label: rng.randrange(4) for label in LABELS})
    return {
        "reference_turns": reference,
        "candidate_turns": candidate_turns,
        "route_rotations": route_rotations,
    }


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_turns"]
    winners = [label for label, word in scene["candidate_turns"].items() if word == reference]
    if len(winners) != 1:
        raise ValueError(f"expected one matching turn word, got {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    reference = scene["reference_turns"]
    candidates = scene["candidate_turns"]
    winner = analytic_gold(scene)
    distances = [
        sum(a != b for a, b in zip(reference, word, strict=True))
        for label, word in candidates.items()
        if label != winner
    ]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    words: Iterable[str] = [scene["reference_turns"], *scene["candidate_turns"].values()]
    return margin(scene) < 1.0 or any(word not in VALID_WORDS for word in words)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _rotate(points: list[tuple[int, int]], turns: int) -> list[tuple[int, int]]:
    result = list(points)
    for _ in range(int(turns) % 4):
        result = [(-y, x) for x, y in result]
    return result


def _triangle(center: tuple[int, int], direction: tuple[int, int], radius: int = 11):
    cx, cy = center
    dx, dy = direction
    px, py = -dy, dx
    return [
        (cx + radius * dx, cy + radius * dy),
        (cx - 8 * dx + 8 * px, cy - 8 * dy + 8 * py),
        (cx - 8 * dx - 8 * px, cy - 8 * dy - 8 * py),
    ]


def _draw_card(draw: ImageDraw.ImageDraw, box, heading: str, word: str, rotation: int) -> None:
    x, y, w, h = box
    draw.rounded_rectangle((x, y, x + w, y + h), radius=10, fill=CARD, outline=BORDER, width=3)
    font = ImageFont.load_default()
    draw.text((x + 10, y + 9), heading, fill=TEXT, font=font)
    points = _rotate(_vertices(word), rotation)
    min_x, max_x = min(p[0] for p in points), max(p[0] for p in points)
    min_y, max_y = min(p[1] for p in points), max(p[1] for p in points)
    area_left, area_top = x + 16, y + 35
    area_w, area_h = w - 32, h - 49
    ox = area_left + (area_w - (max_x - min_x)) // 2 - min_x
    oy = area_top + (area_h - (max_y - min_y)) // 2 - min_y
    placed = [(px + ox, py + oy) for px, py in points]
    draw.line(placed, fill=ROUTE, width=8, joint="curve")
    dx = placed[1][0] - placed[0][0]
    dy = placed[1][1] - placed[0][1]
    direction = (0 if dx == 0 else (1 if dx > 0 else -1), 0 if dy == 0 else (1 if dy > 0 else -1))
    draw.polygon(_triangle(placed[0], direction), fill=START)
    ex, ey = placed[-1]
    draw.rectangle((ex - 9, ey - 9, ex + 9, ey + 9), fill=END)


def render(scene: dict) -> Image.Image:
    reference = scene["reference_turns"]
    candidates = scene["candidate_turns"]
    rotations = scene["route_rotations"]
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = ImageFont.load_default()
    draw.text((28, 18), "FOLLOW TRIANGLE  ->  SQUARE   |   MATCH THE LEFT/RIGHT TURN SEQUENCE", fill=TEXT, font=title_font)
    _draw_card(draw, (410, 55, CARD_W, CARD_H), "REFERENCE", reference, rotations["REFERENCE"])
    draw.line((500, 280, 500, 316), fill=BORDER, width=2)
    for index, label in enumerate(LABELS):
        _draw_card(draw, (50 + index * 240, 340, CARD_W, CARD_H), label, candidates[label], rotations[label])
    return image
