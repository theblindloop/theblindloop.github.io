"""Deterministic renderer for an unlabeled two-family dot constellation."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

W = H = 360
SCALE = 4
PALETTES = [
    ("#f7f4ed", "#172b3a"),
    ("#f3f7f5", "#35233f"),
    ("#f8f2f3", "#173d38"),
    ("#f1f4f8", "#45291f"),
]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    structure = "rails" if int(seed) % 2 == 0 else "rings"
    background, ink = PALETTES[rng.randrange(len(PALETTES))]
    cx = rng.uniform(174.0, 186.0)
    cy = rng.uniform(174.0, 186.0)
    rotation = rng.uniform(-math.pi, math.pi)
    points: list[list[float]] = []

    if structure == "rails":
        half_gap = rng.uniform(37.0, 48.0)
        extent = rng.uniform(112.0, 126.0)
        ux, uy = math.cos(rotation), math.sin(rotation)
        vx, vy = -uy, ux
        # Matched longitudinal ranks make both rails equally sampled, while
        # independent jitter prevents a rigid fixed template.
        for side in (-1.0, 1.0):
            for index in range(12):
                along = -extent + 2.0 * extent * index / 11.0
                along += rng.uniform(-2.4, 2.4)
                across = side * half_gap + rng.uniform(-1.7, 1.7)
                points.append([cx + along * ux + across * vx, cy + along * uy + across * vy])
    else:
        inner = rng.uniform(58.0, 70.0)
        outer = rng.uniform(111.0, 126.0)
        relative_phase = rng.uniform(0.12, 0.42)
        for ring_index, radius in enumerate((inner, outer)):
            phase = rotation + ring_index * relative_phase
            for index in range(12):
                angle = phase + index * 2.0 * math.pi / 12.0 + rng.uniform(-0.018, 0.018)
                noisy_radius = radius + rng.uniform(-1.7, 1.7)
                points.append([cx + noisy_radius * math.cos(angle), cy + noisy_radius * math.sin(angle)])

    rng.shuffle(points)
    return {
        "structure": structure,
        "width": W,
        "height": H,
        "background": background,
        "ink": ink,
        "dot_radius": rng.uniform(5.2, 6.1),
        "points": points,
    }


def render(scene: dict) -> Image.Image:
    # Reading the label field here makes explicit that it is the declared
    # rendered model family; the pixels themselves remain entirely specified by
    # its realized point set.
    structure = scene["structure"]
    if structure not in ("rails", "rings"):
        raise ValueError("unknown constellation structure")
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), scene["background"])
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"]) * SCALE
    for x, y in scene["points"]:
        sx, sy = float(x) * SCALE, float(y) * SCALE
        draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=scene["ink"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["structure"])


def _two_cluster_sse(values: np.ndarray) -> float:
    centers = np.asarray([np.quantile(values, 0.25), np.quantile(values, 0.75)], dtype=float)
    for _ in range(20):
        labels = np.argmin(np.abs(values[:, None] - centers[None, :]), axis=1)
        updated = np.asarray([
            values[labels == k].mean() if np.any(labels == k) else centers[k]
            for k in range(2)
        ])
        if np.max(np.abs(updated - centers)) < 1e-10:
            break
        centers = updated
    return float(np.sum((values - centers[labels]) ** 2))


def _model_residuals(points: list[list[float]]) -> tuple[float, float]:
    xy = np.asarray(points, dtype=float)
    centered = xy - xy.mean(axis=0)
    covariance = centered.T @ centered / len(xy)
    _, vectors = np.linalg.eigh(covariance)
    normal = vectors[:, 0]
    rail_sse = _two_cluster_sse(centered @ normal)
    radii = np.linalg.norm(centered, axis=1)
    ring_sse = _two_cluster_sse(radii)
    scale = max(float(np.sum(centered * centered)), 1.0)
    return rail_sse / scale, ring_sse / scale


def margin(scene: dict) -> float:
    rail, ring = _model_residuals(scene["points"])
    return abs(rail - ring)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.012


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_unordered_point_serialization", twin)]
