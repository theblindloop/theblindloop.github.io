"""Independent pixel-only inverse for the two-family constellation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 45:
            output.append(np.asarray(pixels, dtype=float))
    return output


def _dot_centers(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate((arr[:3].reshape(-1, 3), arr[-3:].reshape(-1, 3), arr[:, :3].reshape(-1, 3), arr[:, -3:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    mask = np.linalg.norm(arr - background, axis=2) > 50.0
    components = _components(mask)
    if len(components) != 24:
        raise ValueError(f"expected 24 separated dot components, found {len(components)}")
    centers = []
    for component in components:
        if not 65 <= len(component) <= 150:
            raise ValueError("component is not dot-like")
        # Array coordinates are y,x; return Cartesian x,y centers.
        centers.append(component[:, ::-1].mean(axis=0) + 0.5)
    return np.asarray(centers, dtype=float)


def _split_error(values: np.ndarray) -> float:
    # Independent exhaustive ordered split: for well-separated two-family data,
    # the minimum within-group SSE is the globally optimal 1-D two-means fit.
    ordered = np.sort(values)
    best = float("inf")
    for cut in range(5, len(ordered) - 4):
        left, right = ordered[:cut], ordered[cut:]
        error = float(np.sum((left - left.mean()) ** 2) + np.sum((right - right.mean()) ** 2))
        best = min(best, error)
    return best


def decision_from_image(image: Image.Image) -> str:
    xy = _dot_centers(image)
    centered = xy - xy.mean(axis=0)
    covariance = np.cov(centered, rowvar=False, bias=True)
    values, vectors = np.linalg.eigh(covariance)
    if values[1] <= 1.0:
        raise ValueError("constellation has insufficient spread")
    rail_error = _split_error(centered @ vectors[:, 0])
    ring_error = _split_error(np.linalg.norm(centered, axis=1))
    total = float(np.sum(centered * centered))
    rail_score, ring_score = rail_error / total, ring_error / total
    if abs(rail_score - ring_score) < 0.010:
        raise ValueError("ambiguous global model evidence")
    return "rails" if rail_score < ring_score else "rings"
