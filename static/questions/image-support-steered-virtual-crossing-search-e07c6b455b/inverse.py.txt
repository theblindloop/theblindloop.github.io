"""Independent pixel-only inverse for virtual-crossing search.

The implementation knows the public palette and visual grammar, but receives no
latent scene and imports no candidate module. It segments the black dashes and
each colored dot, fits both dash centerlines by PCA, and selects the dot with the
smallest worst point-to-line residual.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (210, 47, 58),
    "gold": (220, 145, 24),
    "violet": (126, 68, 184),
    "green": (32, 145, 83),
}
BLACK = (54, 59, 64)
MASK_RADIUS = 72.0
MIN_COMPONENT = 22


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= MIN_COMPONENT:
            output.append(np.asarray(pixels, dtype=np.float64))
    return output


def _shape(component: np.ndarray) -> tuple[np.ndarray, float, np.ndarray]:
    # Convert (row, col) to Cartesian (x, y) pixel centers.
    points = component[:, ::-1] + 0.5
    center = points.mean(axis=0)
    centered = points - center
    values, vectors = np.linalg.eigh(centered.T @ centered / len(points))
    aspect = math.sqrt((float(values[-1]) + 0.2) / (float(values[0]) + 0.2))
    direction = vectors[:, -1]
    return center, aspect, direction


def _dot_center(components: list[np.ndarray]) -> np.ndarray | None:
    shapes = [_shape(component) for component in components]
    dots = [item for item in shapes if item[1] < 1.7]
    if len(dots) != 1:
        return None
    return dots[0][0]


def _candidate_score(dot: np.ndarray, bars: list[tuple[np.ndarray, float, np.ndarray]]) -> float:
    residuals = []
    for center, _, direction in bars:
        delta = dot - center
        residuals.append(abs(float(delta[0] * direction[1] - delta[1] * direction[0])))
    return max(residuals)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    black_delta = array - np.asarray(BLACK, dtype=np.float32)
    black_mask = np.sqrt(np.sum(black_delta * black_delta, axis=2)) < MASK_RADIUS
    bars = [item for item in (_shape(component) for component in _components(black_mask)) if item[1] > 2.15]
    if len(bars) != 2:
        raise ValueError("both separated black dashes must be visible")
    scored: list[tuple[float, str]] = []
    for name, rgb in PALETTE.items():
        delta = array - np.asarray(rgb, dtype=np.float32)
        mask = np.sqrt(np.sum(delta * delta, axis=2)) < MASK_RADIUS
        dot = _dot_center(_components(mask))
        if dot is not None:
            scored.append((_candidate_score(dot, bars), name))
    if not scored:
        raise ValueError("no colored candidate dot is visible")
    scored.sort()
    if len(scored) > 1 and scored[1][0] - scored[0][0] < 1.25:
        raise ValueError("ambiguous virtual-crossing winner")
    return scored[0][1]
