"""Deterministic latent sampler and raster renderer for virtual-crossing search."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLORS = {
    "red": (210, 47, 58),
    "gold": (220, 145, 24),
    "violet": (126, 68, 184),
    "green": (32, 145, 83),
}
COLOR_NAMES = tuple(COLORS)
CANVAS = 512
AA = 3
QUARANTINE_MARGIN = 4.0


def _point_line_distance(point: list[float], segment: list[list[float]]) -> float:
    px, py = point
    (x1, y1), (x2, y2) = segment
    dx, dy = x2 - x1, y2 - y1
    return abs(dx * (y1 - py) - (x1 - px) * dy) / math.hypot(dx, dy)


def _group_score(group: dict, bars: list[list[list[float]]]) -> float:
    return max(_point_line_distance(group["dot"], bar) for bar in bars)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = COLOR_NAMES[int(seed) % len(COLOR_NAMES)]
    scale = rng.choice((0.94, 1.0, 1.06))
    # These are grid-cell interiors rather than a fixed answer coordinate.  The
    # choice randomizes the entire local world while leaving clear background
    # gaps between its three causal glyph regions at finite resolution.
    cx = rng.choice((224, 288))
    cy = rng.choice((224, 288))
    theta1 = rng.randrange(4) * math.pi / 2.0
    theta2 = theta1 + math.pi / 2.0
    bars = []
    for theta in (theta1, theta2):
        ux, uy = math.cos(theta), math.sin(theta)
        radial = 128.0
        half = 12.0 * scale
        bx, by = cx + radial * ux, cy + radial * uy
        bars.append(
            [
                [round(bx - half * ux, 4), round(by - half * uy, 4)],
                [round(bx + half * ux, 4), round(by + half * uy, 4)],
            ]
        )
    groups = []
    distractor_angles = [theta1 + value for value in (0.35, 2.05, 3.7)]
    rng.shuffle(distractor_angles)
    distractor_index = 0
    for name in COLOR_NAMES:
        if name == winner:
            dot = [cx, cy]
        else:
            phi = distractor_angles[distractor_index] + rng.uniform(-0.12, 0.12)
            distractor_index += 1
            magnitude = rng.uniform(18.0, 27.0) * scale
            dot = [cx + magnitude * math.cos(phi), cy + magnitude * math.sin(phi)]
        groups.append(
            {
                "color": name,
                "dot": [round(dot[0], 4), round(dot[1], 4)],
            }
        )
    return {
        "canvas": [CANVAS, CANVAS],
        "center": [cx, cy],
        "scale": scale,
        "bars": bars,
        "groups": groups,
    }


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width * AA, height * AA), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    # A faint local backdrop makes the inset visually coherent but carries no
    # answer information and is deliberately ignored by the inverse arm.
    cx, cy = scene["center"]
    size = 160
    draw.rounded_rectangle(
        ((cx - size) * AA, (cy - size) * AA, (cx + size) * AA, (cy + size) * AA),
        radius=14 * AA,
        fill=(242, 241, 236),
        outline=(218, 216, 208),
        width=2 * AA,
    )
    for bar in scene["bars"]:
        points = [(round(p[0] * AA), round(p[1] * AA)) for p in bar]
        draw.line(points, fill=(54, 59, 64), width=8 * AA)
    for group in scene["groups"]:
        color = COLORS[group["color"]]
        x, y = group["dot"]
        radius = 5.5 * float(scene["scale"])
        draw.ellipse(
            ((x - radius) * AA, (y - radius) * AA, (x + radius) * AA, (y + radius) * AA),
            fill=color,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    # Keep the label-bearing top-level fields explicit.  The geometry helper is
    # deliberately called directly here so the rendered-field audit does not
    # mistake a nested group key for a hidden top-level scene variable.
    bars = scene["bars"]
    scores = sorted(
        (
            max(_point_line_distance(group["dot"], bar) for bar in bars),
            group["color"],
        )
        for group in scene["groups"]
    )
    if len(scores) != 4 or scores[1][0] - scores[0][0] <= 1e-9:
        raise ValueError("scene has no unique virtual-crossing winner")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = sorted(_group_score(group, scene["bars"]) for group in scene["groups"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["groups"] = list(reversed(twin["groups"]))
    return [("reverse_nonoverlapping_group_records", twin)]
