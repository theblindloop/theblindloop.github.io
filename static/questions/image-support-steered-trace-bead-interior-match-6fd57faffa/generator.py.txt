"""Trace-bead interior attribute-correspondence world.

One short, clearly thin, connected trace (a graphite polyline) sits inside a
small local inset whose absolute origin is randomized across scenes. Threaded
onto the trace at two INTERIOR waypoints (not at its ends) are two small coloured
beads. The attribute-correspondence decision: match the bound attribute colour
held by the first interior bead to the bound attribute colour held by the second
interior bead of the SAME connected trace, and report whether they are identical.

The decisive pixel support is the connected thin trace itself: to answer, the
observer must follow the trace interior between the two beads; a crop that holds
only one bead, or that severs the connecting trace, cannot establish that the two
bound attributes belong to the same single route. Position is painted directly
into the raster so the whole decisive path stays inside one focal neighborhood.

Latent state z = {bead_a_color, bead_b_color, ox, oy}. Only the two bead colour
fields influence analytic gold. The parametrisation is injective over its label
inputs (no unused hidden field, no nontrivial pixel-identical symmetry), declared
``not_applicable``.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "trace_bead_interior_match-0.1.0"

CANVAS = 512
CELL = 64
TRACE_WIDTH = 9
BEAD_RADIUS = 10

WHITE = (255, 255, 255)
GRAPHITE = (72, 72, 72)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

# Coarse cell-aligned origins keep the decisive path well inside the canvas and
# localized; ORIGIN_X/Y choices keep the trace within a small inset region.
ORIGIN_X = [3 * CELL, 4 * CELL]
ORIGIN_Y = [3 * CELL, 4 * CELL]

# Waypoints (offsets from origin). W0 is a short leading stub, W1 and W3 are the
# two INTERIOR beads, W4 a short trailing stub. The connector W1-W2-W3 is the
# decisive thin interior.
WAYPOINTS = [(24, 20), (64, 20), (104, 46), (144, 20), (184, 20)]
BEAD_A_INDEX = 1
BEAD_B_INDEX = 3


def _trace_points(scene: dict) -> list[tuple[int, int]]:
    ox, oy = scene["ox"], scene["oy"]
    return [(ox + dx, oy + dy) for (dx, dy) in WAYPOINTS]


def _bead_centres(scene: dict) -> tuple[tuple[int, int], tuple[int, int]]:
    pts = _trace_points(scene)
    return pts[BEAD_A_INDEX], pts[BEAD_B_INDEX]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ox = rng.choice(ORIGIN_X)
    oy = rng.choice(ORIGIN_Y) + rng.choice([0, 0])
    # Deterministically alternate the same/different outcome across the scene
    # stream so every even-length dataset carries both classes off-quarantine.
    same = (seed % 2) == 0
    bead_a_color = rng.choice(PALETTE_NAMES)
    if same:
        bead_b_color = bead_a_color
    else:
        bead_b_color = rng.choice([name for name in PALETTE_NAMES if name != bead_a_color])
    return {
        "bead_a_color": bead_a_color,
        "bead_b_color": bead_b_color,
        "ox": ox,
        "oy": oy,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    pts = _trace_points(scene)
    # Draw the whole connected trace as a thin continuous route.
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=GRAPHITE, width=TRACE_WIDTH)
    for idx, color_name in (
        (BEAD_A_INDEX, scene["bead_a_color"]),
        (BEAD_B_INDEX, scene["bead_b_color"]),
    ):
        cx, cy = pts[idx]
        draw.ellipse(
            [cx - BEAD_RADIUS, cy - BEAD_RADIUS, cx + BEAD_RADIUS, cy + BEAD_RADIUS],
            fill=PALETTE[color_name],
        )
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["bead_a_color"] == scene["bead_b_color"] else "no"


def margin(scene: dict) -> float:
    a = np.array(PALETTE[scene["bead_a_color"]], dtype=float)
    b = np.array(PALETTE[scene["bead_b_color"]], dtype=float)
    return float(np.linalg.norm(a - b))


def is_quarantined(scene: dict) -> bool:
    # Palette colours are pairwise well-separated and beads are solid-filled, so
    # colour identity is a crisp finite-resolution event; equal-colour scenes are
    # a legitimate class, not an ambiguous boundary.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {
        "bead_a_color": scene["bead_a_color"],
        "bead_b_color": scene["bead_b_color"],
        "ox": scene["ox"],
        "oy": scene["oy"],
    }
