"""Independent pixel-only oracle for the trace-bead interior world.

Receives only the final PNG. It never sees the latent scene, renderer geometry
API, or analytic gold. It re-measures the raster independently:

1. Segment the white background from the foreground ink (vectorised nearest-
   marker classification against the white background, the graphite trace, and
   the bead palette).
2. There must be exactly two small bead discs (palette-coloured components); each
   must carry a single dominant palette colour.
3. The two bead discs must lie in the SAME connected foreground component: the
   thin graphite trace must connect them through its interior. If the connecting
   trace is severed or a bead is removed, the oracle abstains by raising.

Only then does it answer whether the two bound attributes (bead colours) are
identical. The palette letters and segmentation constants are re-derived
independently from the raster; no renderer/gold code is imported.
"""

from __future__ import annotations

from collections import Counter

import numpy as np
from PIL import Image

ORACLE_VERSION = "trace_bead_interior_match-oracle-0.1.0"

CANVAS = 512
WHITE = (255, 255, 255)
GRAPHITE = (72, 72, 72)

PALETTE: dict[str, tuple[int, int, int]] = {
    "coral": (232, 93, 77),
    "teal": (31, 158, 141),
    "violet": (106, 81, 176),
}
PALETTE_NAMES = tuple(PALETTE)

# Ordinal markers: 0 white, 1 graphite, 2.. bead palette.
_MARKERS = (WHITE, GRAPHITE) + tuple(PALETTE.values())
_BEAD_OFFSET = 2
_BG_TOL = 24.0


class TraceBeadCompromisedError(ValueError):
    """The decisive trace interior or a bead is missing/damaged."""


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    ys, xs = np.nonzero(mask)
    pts = set(zip(ys.tolist(), xs.tolist()))
    components = []
    while pts:
        seed = pts.pop()
        cells = [seed]
        stack = [seed]
        while stack:
            y, x = stack.pop()
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (yy, xx) in pts:
                    pts.remove((yy, xx))
                    stack.append((yy, xx))
                    cells.append((yy, xx))
        components.append(cells)
    return components


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (CANVAS, CANVAS):
        raise TraceBeadCompromisedError("unexpected image size")
    arr = np.asarray(image, dtype=np.uint8)
    pix = arr.reshape(-1, 3).astype(np.float32)
    marker = np.asarray(_MARKERS, dtype=np.float32)
    dists = np.linalg.norm(pix[:, None, :] - marker[None, :, :], axis=2)
    nearest = np.argmin(dists, axis=1)
    nrows, ncols = arr.shape[:2]
    labels = nearest.reshape(nrows, ncols)

    foreground = labels != 0
    bead = labels >= _BEAD_OFFSET

    # The foreground (trace + beads) must be a single connected route so the two
    # beads demonstrably belong to the same connected trace.
    fg_components = _components(foreground)
    if len(fg_components) != 1:
        raise TraceBeadCompromisedError(
            f"trace+beads must form one connected component, found {len(fg_components)}"
        )

    # Locate the two bead discs (palette-coloured components separate from the
    # grey trace filler between them).
    bead_components = _components(bead)
    if len(bead_components) != 2:
        raise TraceBeadCompromisedError(
            f"expected exactly two bead discs, found {len(bead_components)}"
        )

    bead_colours = []
    for cells in bead_components:
        votes: Counter = Counter()
        for y, x in cells:
            votes[PALETTE_NAMES[int(labels[y, x]) - _BEAD_OFFSET]] += 1
        if not votes:
            raise TraceBeadCompromisedError("a bead carries no dominant colour")
        colour, count = votes.most_common(1)[0]
        if count < 0.55 * len(cells):
            raise TraceBeadCompromisedError("a bead colour is split across markers")
        bead_colours.append(colour)

    return "yes" if bead_colours[0] == bead_colours[1] else "no"
