"""Deterministic renderer and analytic arm for Parity Wall Escape."""

from __future__ import annotations

from collections import Counter, deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 800
N = 6
CELL = 48
BACKGROUND = (245, 242, 235)
CARD = (255, 255, 252)
INK = (42, 47, 57)
MUTED = (84, 91, 104)
GRID = (203, 209, 216)
CARD_EDGE = (91, 100, 114)
DECISIONS = ("A", "B", "C", "D")
TERMINALS = {"A": (0, 0), "B": (0, 5), "C": (5, 0), "D": (5, 5)}
START = (2, 2)
MARK_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (80, 170, 92), "D": (153, 88, 205),
}
PANEL_BOXES = ((20, 190, 386, 750), (417, 190, 783, 750), (814, 190, 1180, 750))
GRID_OFFSET = (39, 132)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _canon(edge) -> tuple[tuple[int, int], tuple[int, int]]:
    a = tuple(edge[0])
    b = tuple(edge[1])
    return tuple(sorted((a, b)))


def _all_edges() -> list[tuple[tuple[int, int], tuple[int, int]]]:
    edges = []
    for r in range(N):
        for c in range(N):
            if c + 1 < N:
                edges.append(((r, c), (r, c + 1)))
            if r + 1 < N:
                edges.append(((r, c), (r + 1, c)))
    return edges


ALL_EDGES = _all_edges()


def _odd_walls(plans) -> set[tuple[tuple[int, int], tuple[int, int]]]:
    counts = Counter(_canon(edge) for plan in plans for edge in plan)
    return {edge for edge, count in counts.items() if count % 2}


def _reachable(walls) -> set[tuple[int, int]]:
    walls = set(walls)
    seen = {START}
    queue = deque([START])
    while queue:
        r, c = queue.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt = (r + dr, c + dc)
            if not (0 <= nxt[0] < N and 0 <= nxt[1] < N):
                continue
            if _canon(((r, c), nxt)) in walls or nxt in seen:
                continue
            seen.add(nxt)
            queue.append(nxt)
    return seen


def _short_path(rng: random.Random, start, goal) -> list[tuple[int, int]]:
    cur = start
    out = [cur]
    while cur != goal:
        r, c = cur
        moves = []
        if r != goal[0]:
            moves.append((r + (1 if goal[0] > r else -1), c))
        if c != goal[1]:
            moves.append((r, c + (1 if goal[1] > c else -1)))
        cur = rng.choice(moves)
        out.append(cur)
    return out


def _partition(rng: random.Random, target: str) -> dict[tuple[int, int], str]:
    """Grow four connected regions; START is prejoined to the target terminal."""
    for _ in range(200):
        owner = {cell: target for cell in _short_path(rng, START, TERMINALS[target])}
        for label, cell in TERMINALS.items():
            owner[cell] = label
        while len(owner) < N * N:
            frontier = []
            for cell, label in owner.items():
                r, c = cell
                for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                    nxt = (r + dr, c + dc)
                    if 0 <= nxt[0] < N and 0 <= nxt[1] < N and nxt not in owner:
                        frontier.append((cell, nxt, label))
            _, nxt, label = rng.choice(frontier)
            owner[nxt] = label
        sizes = Counter(owner.values())
        walls = {edge for edge in ALL_EDGES if owner[edge[0]] != owner[edge[1]]}
        if min(sizes.values()) >= 3 and 13 <= len(walls) <= 28:
            return owner
    raise RuntimeError("could not grow a balanced partition")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = DECISIONS[int(seed) % len(DECISIONS)]
    owner = _partition(rng, target)
    final_walls = [edge for edge in ALL_EDGES if owner[edge[0]] != owner[edge[1]]]
    final_set = set(final_walls)
    nonwalls = [edge for edge in ALL_EDGES if edge not in final_set]
    rng.shuffle(final_walls)
    rng.shuffle(nonwalls)
    decoys = nonwalls[:18]

    plans = [[], [], []]
    for i, edge in enumerate(final_walls):
        plans[i % 3].append(edge)
    pairs = [(0, 1), (1, 2), (0, 2)]
    rng.shuffle(pairs)
    for i, edge in enumerate(decoys):
        for plan_index in pairs[i % 3]:
            plans[plan_index].append(edge)
    rng.shuffle(plans)
    encoded = []
    for plan in plans:
        rng.shuffle(plan)
        encoded.append([
            [list(edge[1]), list(edge[0])] if rng.getrandbits(1)
            else [list(edge[0]), list(edge[1])]
            for edge in plan
        ])
    scene = {"plans": encoded}
    assert not is_quarantined(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    seen = _reachable(_odd_walls(scene["plans"]))
    matches = [label for label, cell in TERMINALS.items() if cell in seen]
    return matches[0] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        plans = scene["plans"]
        if len(plans) != 3:
            return True
        occurrences = [_canon(edge) for plan in plans for edge in plan]
        if any(edge not in ALL_EDGES for edge in occurrences):
            return True
        counts = Counter(occurrences)
        if not all(count in (1, 2) for count in counts.values()):
            return True
        if sum(count == 2 for count in counts.values()) < 15:
            return True
        if not all(14 <= len(plan) <= 25 for plan in plans):
            return True
        return analytic_gold(scene) not in DECISIONS
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = copy.deepcopy(scene)
    transformed["plans"] = [
        [[edge[1], edge[0]] for edge in plan] for plan in transformed["plans"]
    ]
    return [("reverse_all_stored_wall_endpoints", transformed)]


def _wall_line(edge, gx: int, gy: int) -> tuple[int, int, int, int]:
    (r1, c1), (r2, c2) = _canon(edge)
    if r1 == r2:
        x = gx + max(c1, c2) * CELL
        return (x, gy + r1 * CELL, x, gy + (r1 + 1) * CELL)
    y = gy + max(r1, r2) * CELL
    return (gx + c1 * CELL, y, gx + (c1 + 1) * CELL, y)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 28), "PARITY WALL ESCAPE", fill=INK, font=_font(31), anchor="mm")
    draw.text((WIDTH // 2, 67), "same grid position = same wall", fill=MUTED,
              font=_font(19), anchor="mm")
    draw.rounded_rectangle((205, 92, 995, 151), radius=18, fill=(235, 238, 242),
                           outline=CARD_EDGE, width=3)
    draw.text((WIDTH // 2, 121), "shown once = WALL     shown twice = OPEN",
              fill=INK, font=_font(22), anchor="mm")

    for p, box in enumerate(PANEL_BOXES):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=CARD, outline=CARD_EDGE, width=4)
        draw.text(((x0 + x1) // 2, y0 + 37), f"PLAN {p + 1}",
                  fill=INK, font=_font(23), anchor="mm")
        gx, gy = x0 + GRID_OFFSET[0], y0 + GRID_OFFSET[1]
        for k in range(N + 1):
            draw.line((gx + k * CELL, gy, gx + k * CELL, gy + N * CELL), fill=GRID, width=2)
            draw.line((gx, gy + k * CELL, gx + N * CELL, gy + k * CELL), fill=GRID, width=2)
        draw.rectangle((gx, gy, gx + N * CELL, gy + N * CELL), outline=CARD_EDGE, width=4)
        for edge in scene["plans"][p]:
            draw.line(_wall_line(edge, gx, gy), fill=INK, width=10)
        markers = {"S": START, **TERMINALS}
        for label, (r, c) in markers.items():
            cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
            draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16),
                         fill=MARK_COLORS[label], outline=INK, width=2)
            fill = (255, 255, 255) if label in {"S", "B", "D"} else (24, 27, 33)
            draw.text((cx, cy + 1), label, fill=fill, font=_font(17), anchor="mm")
        draw.text(((x0 + x1) // 2, y1 - 24), "S = START", fill=MUTED,
                  font=_font(16), anchor="mm")
    return image
