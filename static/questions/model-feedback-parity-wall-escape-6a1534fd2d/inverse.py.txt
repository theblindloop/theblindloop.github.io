"""Independent PNG-only inverse for Parity Wall Escape."""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 1200, 800
N, CELL = 6, 48
INK = np.array((42, 47, 57), dtype=np.int16)
GRID = np.array((203, 209, 216), dtype=np.int16)
PANEL_ORIGINS = ((20, 190), (417, 190), (814, 190))
GRID_OFFSET = (39, 132)
START = (2, 2)
TERMINALS = {"A": (0, 0), "B": (0, 5), "C": (5, 0), "D": (5, 5)}
MARK_COLORS = {
    "S": (218, 65, 65), "A": (247, 174, 47), "B": (60, 132, 222),
    "C": (80, 170, 92), "D": (153, 88, 205),
}


def _all_edges():
    out = []
    for r in range(N):
        for c in range(N):
            if c + 1 < N:
                out.append(((r, c), (r, c + 1)))
            if r + 1 < N:
                out.append(((r, c), (r + 1, c)))
    return out


ALL_EDGES = _all_edges()


def _marker_ok(arr: np.ndarray, gx: int, gy: int, label: str, cell) -> bool:
    r, c = cell
    cx, cy = gx + c * CELL + CELL // 2, gy + r * CELL + CELL // 2
    patch = arr[cy - 12:cy + 13, cx - 12:cx + 13].astype(np.int16)
    ref = np.array(MARK_COLORS[label], dtype=np.int16)
    return int((np.abs(patch - ref).max(axis=2) <= 3).sum()) >= 260


def _segment_patch(arr: np.ndarray, gx: int, gy: int, edge):
    (r1, c1), (r2, c2) = edge
    if r1 == r2:
        x = gx + max(c1, c2) * CELL
        cy = gy + r1 * CELL + CELL // 2
        return arr[cy - 13:cy + 14, x - 4:x + 5]
    y = gy + max(r1, r2) * CELL
    cx = gx + c1 * CELL + CELL // 2
    return arr[y - 4:y + 5, cx - 13:cx + 14]


def _read_plan(arr: np.ndarray, x0: int, y0: int):
    gx, gy = x0 + GRID_OFFSET[0], y0 + GRID_OFFSET[1]
    if not _marker_ok(arr, gx, gy, "S", START):
        return None
    if any(not _marker_ok(arr, gx, gy, label, cell) for label, cell in TERMINALS.items()):
        return None
    walls = []
    for edge in ALL_EDGES:
        patch = _segment_patch(arr, gx, gy, edge).astype(np.int16)
        ink_fraction = float((np.abs(patch - INK).max(axis=2) <= 3).mean())
        grid_fraction = float((np.abs(patch - GRID).max(axis=2) <= 3).mean())
        if ink_fraction >= 0.72:
            walls.append(edge)
        elif grid_fraction < 0.13:
            return None
    return walls


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        return "abstain"
    arr = np.asarray(image.convert("RGB"))
    plans = []
    for origin in PANEL_ORIGINS:
        plan = _read_plan(arr, *origin)
        if plan is None:
            return "abstain"
        plans.append(plan)
    counts = Counter(edge for plan in plans for edge in plan)
    if any(count not in (1, 2) for count in counts.values()):
        return "abstain"
    walls = {edge for edge, count in counts.items() if count % 2}
    seen = {START}
    queue = deque([START])
    while queue:
        r, c = queue.popleft()
        for dr, dc in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            nxt = (r + dr, c + dc)
            if not (0 <= nxt[0] < N and 0 <= nxt[1] < N):
                continue
            edge = tuple(sorted(((r, c), nxt)))
            if edge in walls or nxt in seen:
                continue
            seen.add(nxt)
            queue.append(nxt)
    matches = [label for label, cell in TERMINALS.items() if cell in seen]
    return matches[0] if len(matches) == 1 else "abstain"
