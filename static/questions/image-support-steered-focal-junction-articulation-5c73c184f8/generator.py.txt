"""Deterministic raster world for a focal vertex-cut topology question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "focal_junction_articulation-0.1.0"
BACKGROUND = (255, 255, 255)
NETWORK_INK = (39, 49, 59)
REMOTE_INK = (224, 230, 236)
PALETTE = ("#e4572e", "#2185d0", "#18a999", "#b23a8f")
CANVAS = (512, 512)
SUPERSAMPLE = 4


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced topology and clearance cells with a compact random placement."""

    rng = random.Random(int(seed))
    bucket = int(seed) % 8
    topology = "articulation" if bucket in {0, 1, 4, 5} else "redundant"
    low_clearance = bucket in {0, 3, 6}
    cell_x = rng.randint(1, 6)
    cell_y = rng.randint(1, 6)
    # The complete gadget stays inside one 64 px high-resolution cell, while
    # the cell choice still randomizes its absolute position on the canvas.
    origin_x = (cell_x + 0.5) * 64.0 + rng.uniform(-1.5, 1.5)
    origin_y = (cell_y + 0.5) * 64.0 + rng.uniform(-1.5, 1.5)
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        "origin_x": round(origin_x, 3),
        "origin_y": round(origin_y, 3),
        "topology": topology,
        "node_color": rng.choice(PALETTE),
        "orientation": rng.randrange(4),
        "route_side": rng.choice((-1, 1)),
        "terminal_span": 14,
        "route_offset": 12 if low_clearance else 17,
        "stroke_width": 5 if low_clearance else 6,
        "node_radius": 6,
        "terminal_radius": rng.choice((4, 5)),
        "decor_seed": rng.randrange(1, 2**31 - 1),
    }


def _rotate(point: tuple[float, float], orientation: int) -> tuple[float, float]:
    x, y = point
    turn = int(orientation) % 4
    if turn == 0:
        return x, y
    if turn == 1:
        return -y, x
    if turn == 2:
        return -x, -y
    return y, -x


def _point(scene: dict[str, Any], local: tuple[float, float], scale: int) -> tuple[int, int]:
    rx, ry = _rotate(local, int(scene["orientation"]))
    return (
        int(round((float(scene["origin_x"]) + rx) * scale)),
        int(round((float(scene["origin_y"]) + ry) * scale)),
    )


def _line(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    points: list[tuple[float, float]],
    fill: tuple[int, int, int],
    width: int,
    scale: int,
) -> None:
    draw.line(
        [_point(scene, point, scale) for point in points],
        fill=fill,
        width=max(1, int(round(width * scale))),
        joint="curve",
    )


def _disk(
    draw: ImageDraw.ImageDraw,
    scene: dict[str, Any],
    center: tuple[float, float],
    radius: float,
    fill: Any,
    scale: int,
) -> None:
    # A circle is invariant to the graph rotation; its center still follows it.
    cx, cy = _point(scene, center, scale)
    rr = int(round(float(radius) * scale))
    draw.ellipse((cx - rr, cy - rr, cx + rr, cy + rr), fill=fill)


def _draw_remote_marks(draw: ImageDraw.ImageDraw, scene: dict[str, Any], scale: int) -> None:
    """Draw pale, remote decoration that carries no topology information."""

    rng = random.Random(int(scene["decor_seed"]))
    for base_x, base_y in ((34, 34), (478, 34), (34, 478), (478, 478)):
        x = base_x + rng.randint(-7, 7)
        y = base_y + rng.randint(-7, 7)
        rr = 3 * scale
        draw.ellipse(
            (x * scale - rr, y * scale - rr, x * scale + rr, y * scale + rr),
            fill=REMOTE_INK,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the public scene into the final RGB PNG."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, scene, scale)

    span = float(scene["terminal_span"])
    route_offset = float(scene["route_offset"]) * float(scene["route_side"])
    stroke = int(scene["stroke_width"])
    node_radius = float(scene["node_radius"])
    terminal_radius = float(scene["terminal_radius"])
    left = (-span, 0.0)
    right = (span, 0.0)

    if str(scene["topology"]) == "redundant":
        _line(
            draw,
            scene,
            [left, (left[0], route_offset), (right[0], route_offset), right],
            NETWORK_INK,
            stroke,
            scale,
        )

    # The direct route is shared by both topologies. The colored disk is painted
    # last, so it is the visible vertex whose removal creates the counterfactual.
    _line(draw, scene, [left, right], NETWORK_INK, stroke, scale)
    _disk(draw, scene, left, terminal_radius, NETWORK_INK, scale)
    _disk(draw, scene, right, terminal_radius, NETWORK_INK, scale)
    _disk(draw, scene, (0.0, 0.0), node_radius, str(scene["node_color"]), scale)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the vertex-cut decision from the single rendered label input."""

    return "disconnects" if str(scene["topology"]) == "articulation" else "stays_connected"


def _clearance(scene: dict[str, Any]) -> float:
    span_gap = float(scene["terminal_span"]) - float(scene["node_radius"])
    bypass_gap = abs(float(scene["route_offset"])) - float(scene["stroke_width"])
    return min(span_gap, bypass_gap)


def margin(scene: dict[str, Any]) -> float:
    value = _clearance(scene)
    return value if analytic_gold(scene) == "disconnects" else -value


def is_quarantined(scene: dict[str, Any]) -> bool:
    return _clearance(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """This renderer declares no intentionally pixel-identical latent aliases."""

    del scene
    return []
