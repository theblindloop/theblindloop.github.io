"""Independent pixel-only oracle for the focal junction topology."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    points = np.argwhere(mask)
    seen = np.zeros(mask.shape, dtype=bool)
    height, width = mask.shape
    found: list[np.ndarray] = []
    for y, x in points:
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        values: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            values.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(values) >= minimum:
            found.append(np.asarray(values, dtype=np.int32))
    return found


def _dilate(mask: np.ndarray, radius: int) -> np.ndarray:
    if radius <= 0:
        return mask.copy()
    height, width = mask.shape
    padded = np.pad(mask, radius, mode="constant")
    output = np.zeros_like(mask)
    for dy in range(2 * radius + 1):
        for dx in range(2 * radius + 1):
            output |= padded[dy : dy + height, dx : dx + width]
    return output


def _saturated_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    high = values.max(axis=2)
    low = values.min(axis=2)
    return (high - low >= 55) & (high >= 100) & (low <= 245)


def decision_from_image(image: Image.Image) -> str:
    """Recover the topology after removing the colored disk from PNG pixels."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    saturated = _saturated_mask(array)
    colored = _components(saturated, minimum=24)
    if len(colored) != 1:
        raise ValueError(f"expected one colored junction, found {len(colored)}")

    marker = colored[0]
    center_y = int(round(float(marker[:, 0].mean())))
    center_x = int(round(float(marker[:, 1].mean())))
    half = 36
    height, width = array.shape[:2]
    y0 = max(0, center_y - half)
    y1 = min(height, center_y + half + 1)
    x0 = max(0, center_x - half)
    x1 = min(width, center_x + half + 1)
    crop = array[y0:y1, x0:x1]
    crop_color = saturated[y0:y1, x0:x1]

    # The dark network is recovered from intensity alone. A generous removal
    # disk models erasing the whole colored vertex, including antialiasing.
    values = crop.astype(np.float32)
    foreground = np.linalg.norm(values - 255.0, axis=2) > 80.0
    remaining = foreground & ~_dilate(crop_color, radius=5)
    network_components = _components(remaining, minimum=8)
    if len(network_components) < 1:
        raise ValueError("network disappeared after removing the colored junction")
    if len(network_components) >= 2:
        return "disconnects"
    return "stays_connected"
