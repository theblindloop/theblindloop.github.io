"""Renderer and analytic gold for the synchronous ring rewrite world.

Seven colored disks sit on a ring.  A key of six framed tiles maps an unordered
pair of colors to a result color.  Every disk is replaced at the same instant by
the color the key gives for the unordered pair of its two ring neighbours, using
the colors as they stood before the round.  After five such rounds the color of
the disk inside the black annulus is the decision.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

COLORS = ("amber", "crimson", "teal")
INK = {
    "amber": (238, 170, 20),
    "crimson": (200, 32, 64),
    "teal": (0, 148, 152),
}
WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GRAY = (170, 170, 170)

CANVAS = (1080, 1000)
N_DISKS = 7
ROUNDS = 5

TILE_W = 330
TILE_H = 124
TILE_GAP_X = 21
TILE_GAP_Y = 26
TILE_X0 = 24
TILE_Y0 = 40
FRAME = 6
DOT_R = 26
DOT1_DX = 62
DOT2_DX = 130
DOT_DY = 62
ARROW_X0 = 176
ARROW_X1 = 216
ARROW_DY = 24
SQ_X0 = 240
SQ_SIDE = 64
SQ_DY = 30

RING_CX = 540
RING_CY = 660
RING_R = 218
DISK_R = 54
ANN_IN = 68
ANN_OUT = 82

PAIR_KEYS = ("0-0", "0-1", "0-2", "1-1", "1-2", "2-2")


def _pair_key(a, b):
    lo = a if a <= b else b
    hi = b if a <= b else a
    return "%d-%d" % (lo, hi)


def _evolve(colors, rule, rounds):
    n = len(colors)
    cur = list(colors)
    for _ in range(rounds):
        cur = [rule[_pair_key(cur[(i - 1) % n], cur[(i + 1) % n])] for i in range(n)]
    return cur


def _inplace_sweep(colors, rule, rounds, forward):
    """Wrong program: rewrite the ring one disk at a time, in place."""
    n = len(colors)
    cur = list(colors)
    order = range(n) if forward else range(n - 1, -1, -1)
    for _ in range(rounds):
        for i in order:
            cur[i] = rule[_pair_key(cur[(i - 1) % n], cur[(i + 1) % n])]
    return cur


def _narrow_cone(colors, rule, rounds, marked):
    """Wrong program: only the marked disk and its neighbours are ever updated."""
    n = len(colors)
    cur = list(colors)
    live = ((marked - 1) % n, marked, (marked + 1) % n)
    for _ in range(rounds):
        nxt = list(cur)
        for i in live:
            nxt[i] = rule[_pair_key(cur[(i - 1) % n], cur[(i + 1) % n])]
        cur = nxt
    return cur


def _own_and_neighbour(colors, rule, rounds):
    """Wrong program: pair a disk's own color with one neighbour's color."""
    n = len(colors)
    cur = list(colors)
    for _ in range(rounds):
        cur = [rule[_pair_key(cur[i], cur[(i - 1) % n])] for i in range(n)]
    return cur


def _tile_origin(index, jitter):
    col = index % 3
    row = index // 3
    dx, dy = jitter["tiles"][index]
    x = TILE_X0 + col * (TILE_W + TILE_GAP_X) + dx
    y = TILE_Y0 + row * (TILE_H + TILE_GAP_Y) + dy
    return x, y


def _disk_center(index, jitter):
    cx = RING_CX + jitter["ring"][0]
    cy = RING_CY + jitter["ring"][1]
    r = RING_R + jitter["radius"]
    ang = jitter["phase"] + index * (2.0 * math.pi / N_DISKS) + jitter["angles"][index]
    return cx + r * math.cos(ang), cy + r * math.sin(ang)


def sample_scene(seed):
    rng = random.Random(seed)
    for _ in range(20000):
        rule = {key: rng.randrange(3) for key in PAIR_KEYS}
        if len(set(rule.values())) < 3:
            continue
        colors = [rng.randrange(3) for _ in range(N_DISKS)]
        if len(set(colors)) < 3:
            continue
        marked = rng.randrange(N_DISKS)

        gens = [list(colors)]
        for _r in range(ROUNDS + 1):
            gens.append(_evolve(gens[-1], rule, 1))
        gold = gens[ROUNDS][marked]

        if gold == colors[marked]:
            continue
        if gold == gens[ROUNDS - 1][marked] or gold == gens[ROUNDS + 1][marked]:
            continue
        if gold == _inplace_sweep(colors, rule, ROUNDS, True)[marked]:
            continue
        if gold == _inplace_sweep(colors, rule, ROUNDS, False)[marked]:
            continue
        if gold == _narrow_cone(colors, rule, ROUNDS, marked)[marked]:
            continue
        if gold == _own_and_neighbour(colors, rule, ROUNDS)[marked]:
            continue
        # the marked disk must genuinely move through the run
        track = [g[marked] for g in gens[: ROUNDS + 1]]
        if sum(1 for a, b in zip(track, track[1:]) if a != b) < 2:
            continue
        # most drawn disks must be load bearing
        decisive = 0
        for i in range(N_DISKS):
            alt = list(colors)
            moved = False
            for c in range(3):
                if c == colors[i]:
                    continue
                alt[i] = c
                if _evolve(alt, rule, ROUNDS)[marked] != gold:
                    moved = True
                    break
            decisive += 1 if moved else 0
        if decisive < 5:
            continue
        # most of the drawn key must be load bearing
        used = set()
        cur = list(colors)
        for _r in range(ROUNDS):
            for i in range(N_DISKS):
                used.add(_pair_key(cur[(i - 1) % N_DISKS], cur[(i + 1) % N_DISKS]))
            cur = _evolve(cur, rule, 1)
        if len(used) < 4:
            continue
        hot = 0
        for key in sorted(used):
            for c in range(3):
                if c == rule[key]:
                    continue
                alt_rule = dict(rule)
                alt_rule[key] = c
                if _evolve(colors, alt_rule, ROUNDS)[marked] != gold:
                    hot += 1
                    break
        if hot < 3:
            continue

        order = list(PAIR_KEYS)
        rng.shuffle(order)
        jitter = {
            "ring": [rng.randint(-8, 8), rng.randint(-8, 8)],
            "radius": rng.randint(-6, 6),
            "phase": rng.uniform(0.0, 2.0 * math.pi),
            "angles": [rng.uniform(-0.017, 0.017) for _ in range(N_DISKS)],
            "tiles": [[rng.randint(-3, 3), rng.randint(-3, 3)] for _ in range(6)],
        }
        return {
            "colors": colors,
            "rule": rule,
            "marked": marked,
            "tile_order": order,
            "jitter": jitter,
        }
    raise RuntimeError("sampler exhausted for seed %r" % (seed,))


def analytic_gold(scene):
    colors = list(scene["colors"])
    rule = scene["rule"]
    marked = int(scene["marked"])
    return COLORS[_evolve(colors, rule, ROUNDS)[marked]]


def margin(scene):
    jitter = scene["jitter"]
    clearances = [
        float(ANN_IN - DISK_R),
        float(DOT2_DX - DOT1_DX - 2 * DOT_R),
        float(ARROW_X0 - (DOT2_DX + DOT_R)),
        float(SQ_X0 - ARROW_X1),
        float(TILE_W - FRAME - (SQ_X0 + SQ_SIDE)),
        float(DOT1_DX - DOT_R - FRAME),
        float(TILE_H - FRAME - (SQ_DY + SQ_SIDE)),
    ]
    tiles = [_tile_origin(i, jitter) for i in range(6)]
    for i in range(6):
        for j in range(i + 1, 6):
            xi, yi = tiles[i]
            xj, yj = tiles[j]
            gap_x = max(xi, xj) - min(xi, xj) - TILE_W
            gap_y = max(yi, yj) - min(yi, yj) - TILE_H
            clearances.append(float(max(gap_x, gap_y)))
    for x, y in tiles:
        clearances.append(float(x))
        clearances.append(float(y))
        clearances.append(float(CANVAS[0] - (x + TILE_W)))
    centers = [_disk_center(i, jitter) for i in range(N_DISKS)]
    for i in range(N_DISKS):
        xi, yi = centers[i]
        clearances.append(float(xi - ANN_OUT))
        clearances.append(float(CANVAS[0] - xi - ANN_OUT))
        clearances.append(float(CANVAS[1] - yi - ANN_OUT))
        for j in range(i + 1, N_DISKS):
            xj, yj = centers[j]
            dist = math.hypot(xi - xj, yi - yj)
            clearances.append(float(dist - 2 * DISK_R))
            clearances.append(float(dist - ANN_OUT - DISK_R))
    lowest_tile = max(y for _x, y in tiles) + TILE_H
    top_ring = min(y for _x, y in centers) - ANN_OUT
    clearances.append(float(top_ring - lowest_tile))
    return round(min(clearances), 6)


def is_quarantined(scene):
    return margin(scene) < 11.0


def latent_symmetries(scene):
    return []


def _draw_tile(draw, x, y, pair, result):
    draw.rectangle([x, y, x + TILE_W - 1, y + TILE_H - 1], outline=GRAY, width=FRAME)
    for dx, color in ((DOT1_DX, pair[0]), (DOT2_DX, pair[1])):
        cx = x + dx
        cy = y + DOT_DY
        draw.ellipse(
            [cx - DOT_R, cy - DOT_R, cx + DOT_R, cy + DOT_R], fill=INK[COLORS[color]]
        )
    draw.polygon(
        [
            (x + ARROW_X0, y + DOT_DY - ARROW_DY),
            (x + ARROW_X1, y + DOT_DY),
            (x + ARROW_X0, y + DOT_DY + ARROW_DY),
        ],
        fill=BLACK,
    )
    draw.rectangle(
        [x + SQ_X0, y + SQ_DY, x + SQ_X0 + SQ_SIDE - 1, y + SQ_DY + SQ_SIDE - 1],
        fill=INK[COLORS[result]],
    )


def render(scene):
    colors = list(scene["colors"])
    rule = scene["rule"]
    marked = int(scene["marked"])
    order = list(scene["tile_order"])
    jitter = scene["jitter"]

    image = Image.new("RGB", CANVAS, WHITE)
    draw = ImageDraw.Draw(image)

    for slot, key in enumerate(order):
        left, right = key.split("-")
        x, y = _tile_origin(slot, jitter)
        _draw_tile(draw, x, y, (int(left), int(right)), rule[key])

    for index in range(N_DISKS):
        cx, cy = _disk_center(index, jitter)
        if index == marked:
            draw.ellipse(
                [cx - ANN_OUT, cy - ANN_OUT, cx + ANN_OUT, cy + ANN_OUT], fill=BLACK
            )
            draw.ellipse(
                [cx - ANN_IN, cy - ANN_IN, cx + ANN_IN, cy + ANN_IN], fill=WHITE
            )
        draw.ellipse(
            [cx - DISK_R, cy - DISK_R, cx + DISK_R, cy + DISK_R],
            fill=INK[COLORS[colors[index]]],
        )
    return image
