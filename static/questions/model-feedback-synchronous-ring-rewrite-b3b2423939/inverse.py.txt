"""Pixel-only oracle for the synchronous ring rewrite world.

The oracle receives a rasterized PNG and nothing else.  It segments the exact
inks, recovers the six key tiles as an unordered-pair rule, recovers the seven
ring disks and their cyclic neighbour relation, locates the black annulus, and
then runs five synchronous rounds itself.
"""

from __future__ import annotations

import math

import numpy as np

COLOR_NAMES = ("amber", "crimson", "teal")
COLOR_RGB = ((238, 170, 20), (200, 32, 64), (0, 148, 152))
GRAY_RGB = (170, 170, 170)
BLACK_RGB = (0, 0, 0)
ROUNDS = 5


def _mask(arr, rgb):
    return (
        (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])
    )


def _components(mask):
    height = mask.shape[0]
    runs = []
    parent = []

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    prev_runs = []
    for y in range(height):
        row = mask[y]
        idx = np.flatnonzero(row)
        cur_runs = []
        if idx.size:
            breaks = np.flatnonzero(np.diff(idx) > 1)
            starts = np.concatenate(([0], breaks + 1))
            ends = np.concatenate((breaks, [idx.size - 1]))
            for s, e in zip(starts, ends):
                x0 = int(idx[s])
                x1 = int(idx[e])
                rid = len(runs)
                runs.append((y, x0, x1))
                parent.append(rid)
                cur_runs.append((x0, x1, rid))
        for x0, x1, rid in cur_runs:
            for px0, px1, prid in prev_runs:
                if px0 <= x1 and x0 <= px1:
                    union(prid, rid)
        prev_runs = cur_runs

    groups = {}
    for rid, (y, x0, x1) in enumerate(runs):
        root = find(rid)
        entry = groups.get(root)
        width = x1 - x0 + 1
        if entry is None:
            groups[root] = [x0, x1, y, y, width, float(x0 + x1) * width / 2.0,
                            float(y) * width]
        else:
            entry[0] = min(entry[0], x0)
            entry[1] = max(entry[1], x1)
            entry[2] = min(entry[2], y)
            entry[3] = max(entry[3], y)
            entry[4] += width
            entry[5] += float(x0 + x1) * width / 2.0
            entry[6] += float(y) * width
    out = []
    for entry in groups.values():
        area = entry[4]
        out.append(
            {
                "x0": entry[0],
                "x1": entry[1],
                "y0": entry[2],
                "y1": entry[3],
                "area": area,
                "cx": entry[5] / area,
                "cy": entry[6] / area,
            }
        )
    return out


def _inside(box, cx, cy):
    return box["x0"] <= cx <= box["x1"] and box["y0"] <= cy <= box["y1"]


def _color_index(arr, comp):
    ys = int(round(comp["cy"]))
    xs = int(round(comp["cx"]))
    pixel = tuple(int(v) for v in arr[ys, xs])
    if pixel in COLOR_RGB:
        return COLOR_RGB.index(pixel)
    # fall back to the most frequent declared hue inside the bounding box
    region = arr[comp["y0"]: comp["y1"] + 1, comp["x0"]: comp["x1"] + 1]
    best = None
    best_count = -1
    for index, rgb in enumerate(COLOR_RGB):
        count = int(_mask(region, rgb).sum())
        if count > best_count:
            best_count = count
            best = index
    return best


def _pair_key(a, b):
    return "%d-%d" % (min(a, b), max(a, b))


def decision_from_image(image):
    arr = np.asarray(image.convert("RGB"))
    tiles = [c for c in _components(_mask(arr, GRAY_RGB)) if c["area"] > 500]
    if len(tiles) != 6:
        raise ValueError("expected six key tiles, found %d" % len(tiles))

    hue_mask = np.zeros(arr.shape[:2], dtype=bool)
    for rgb in COLOR_RGB:
        hue_mask |= _mask(arr, rgb)
    blobs = [c for c in _components(hue_mask) if c["area"] > 300]

    rule = {}
    ring = []
    for tile in tiles:
        members = [b for b in blobs if _inside(tile, b["cx"], b["cy"])]
        if len(members) != 3:
            raise ValueError("tile does not hold three marks")
        members.sort(key=lambda b: b["cx"])
        left = _color_index(arr, members[0])
        right = _color_index(arr, members[1])
        result = _color_index(arr, members[2])
        rule[_pair_key(left, right)] = result
    for blob in blobs:
        if not any(_inside(tile, blob["cx"], blob["cy"]) for tile in tiles):
            ring.append(blob)
    if len(ring) != 7:
        raise ValueError("expected seven ring disks, found %d" % len(ring))
    if len(rule) != 6:
        raise ValueError("key does not define six distinct pairs")

    cx = sum(b["cx"] for b in ring) / 7.0
    cy = sum(b["cy"] for b in ring) / 7.0
    ring.sort(key=lambda b: math.atan2(b["cy"] - cy, b["cx"] - cx))
    colors = [_color_index(arr, b) for b in ring]

    annuli = [
        c
        for c in _components(_mask(arr, BLACK_RGB))
        if c["area"] > 400 and not any(_inside(t, c["cx"], c["cy"]) for t in tiles)
    ]
    if len(annuli) != 1:
        raise ValueError("expected exactly one annulus, found %d" % len(annuli))
    ring_mark = annuli[0]
    marked = min(
        range(7),
        key=lambda i: math.hypot(ring[i]["cx"] - ring_mark["cx"], ring[i]["cy"] - ring_mark["cy"]),
    )

    cur = list(colors)
    for _ in range(ROUNDS):
        cur = [rule[_pair_key(cur[(i - 1) % 7], cur[(i + 1) % 7])] for i in range(7)]
    return COLOR_NAMES[cur[marked]]
