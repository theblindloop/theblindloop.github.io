"""Deterministic renderer and analytic arm for a route-multiplicity network."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 600, 420
BACKGROUND = (248, 247, 243)
DARK = (32, 38, 48)
JUNCTION = (148, 158, 170)
GREEN = (30, 166, 102)
TERMINAL_RGB = {
    "orange": (235, 126, 36),
    "purple": (132, 78, 186),
    "blue": (48, 126, 205),
}
LANE_Y = (85, 210, 335)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    lane_types = [1, 2, 4]
    rng.shuffle(lane_types)

    target_color = ("orange", "purple", "blue")[int(seed) % 3]
    remaining = [name for name in TERMINAL_RGB if name != target_color]
    rng.shuffle(remaining)
    terminal_colors = [None, None, None]
    target_lane = lane_types.index(2)
    terminal_colors[target_lane] = target_color
    other_lanes = [i for i in range(3) if i != target_lane]
    for lane, color in zip(other_lanes, remaining):
        terminal_colors[lane] = color
    return {"lane_types": lane_types, "terminal_colors": terminal_colors}


def _arm(topology: int, y: int, hub: tuple[int, int]) -> tuple[list[tuple[int, int]], list[tuple[tuple[int, int], tuple[int, int]]]]:
    root = (170, y)
    terminal = (548, y)
    nodes: list[tuple[int, int]] = [root]
    edges: list[tuple[tuple[int, int], tuple[int, int]]] = [(hub, root)]
    if topology == 1:
        a, b = (300, y), (495, y)
        nodes.extend((a, b))
        edges.extend(((root, a), (a, b), (b, terminal)))
    elif topology == 2:
        split, upper, lower, join, end = (235, y), (315, y - 34), (315, y + 34), (405, y), (495, y)
        nodes.extend((split, upper, lower, join, end))
        edges.extend(
            ((root, split), (split, upper), (upper, join), (split, lower),
             (lower, join), (join, end), (end, terminal))
        )
    elif topology == 4:
        split1, up1, down1, join1 = (205, y), (255, y - 27), (255, y + 27), (305, y)
        split2, up2, down2, join2, end = (345, y), (390, y - 27), (390, y + 27), (440, y), (495, y)
        nodes.extend((split1, up1, down1, join1, split2, up2, down2, join2, end))
        edges.extend(
            ((root, split1), (split1, up1), (up1, join1), (split1, down1),
             (down1, join1), (join1, split2), (split2, up2), (up2, join2),
             (split2, down2), (down2, join2), (join2, end), (end, terminal))
        )
    else:
        raise ValueError(f"unsupported topology {topology}")
    return nodes, edges


def _geometry(lane_types: Iterable[int]):
    start, hub = (48, 210), (112, 210)
    internal = [hub]
    edges = [(start, hub)]
    terminals = []
    for topology, y in zip(lane_types, LANE_Y):
        arm_nodes, arm_edges = _arm(int(topology), y, hub)
        internal.extend(arm_nodes)
        edges.extend(arm_edges)
        terminals.append((548, y))
    return start, internal, terminals, edges


def render(scene: dict) -> Image.Image:
    lane_types = scene["lane_types"]
    terminal_colors = scene["terminal_colors"]
    if sorted(lane_types) != [1, 2, 4] or sorted(terminal_colors) != sorted(TERMINAL_RGB):
        raise ValueError("malformed scene")
    start, internal, terminals, edges = _geometry(lane_types)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        draw.line((left, right), fill=DARK, width=8)
    for x, y in internal:
        draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=JUNCTION, outline=DARK, width=2)
    sx, sy = start
    draw.ellipse((sx - 15, sy - 15, sx + 15, sy + 15), fill=GREEN, outline=DARK, width=2)
    draw.ellipse((sx - 7, sy - 7, sx + 7, sy + 7), fill=BACKGROUND)
    for (x, y), color in zip(terminals, terminal_colors):
        draw.rectangle((x - 14, y - 14, x + 14, y + 14), fill=TERMINAL_RGB[color], outline=DARK, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    lane_types = scene["lane_types"]
    terminal_colors = scene["terminal_colors"]
    if lane_types.count(2) != 1:
        raise ValueError("scene must have one two-route arm")
    return str(terminal_colors[lane_types.index(2)])


def margin(scene: dict) -> float:
    lane_types = scene["lane_types"]
    # Score is negative distance from the requested route count of two.
    distances = sorted(abs(int(value) - 2) for value in lane_types)
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    lane_types = scene["lane_types"]
    terminal_colors = scene["terminal_colors"]
    return sorted(lane_types) != [1, 2, 4] or sorted(terminal_colors) != sorted(TERMINAL_RGB)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
