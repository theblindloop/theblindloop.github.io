"""Independent image-only inverse arm for route multiplicity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

DARK = (32, 38, 48)
JUNCTION = (148, 158, 170)
GREEN = (30, 166, 102)
TERMINALS = {
    "orange": (235, 126, 36),
    "purple": (132, 78, 186),
    "blue": (48, 126, 205),
}


def _components(mask: np.ndarray, minimum: int = 8) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    unseen = {(int(x), int(y)) for y, x in np.argwhere(mask)}
    groups: list[list[tuple[int, int]]] = []
    while unseen:
        first = unseen.pop()
        queue = [first]
        group = [first]
        while queue:
            x, y = queue.pop()
            for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if point in unseen:
                    unseen.remove(point)
                    queue.append(point)
                    group.append(point)
        if len(group) >= minimum:
            groups.append(group)
    return groups


def _centers(arr: np.ndarray, rgb: tuple[int, int, int], minimum: int = 8) -> list[tuple[float, float]]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    result = []
    for group in _components(mask, minimum):
        result.append((sum(x for x, _ in group) / len(group), sum(y for _, y in group) / len(group)))
    return result


def _point_segment_distance(px: float, py: float, ax: float, ay: float, bx: float, by: float) -> float:
    dx, dy = bx - ax, by - ay
    denom = dx * dx + dy * dy
    if denom == 0:
        return ((px - ax) ** 2 + (py - ay) ** 2) ** 0.5
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denom))
    qx, qy = ax + t * dx, ay + t * dy
    return ((px - qx) ** 2 + (py - qy) ** 2) ** 0.5


def _has_intermediate(a: tuple[float, float], b: tuple[float, float], vertices: list[tuple[float, float]]) -> bool:
    ax, ay = a
    bx, by = b
    length = ((bx - ax) ** 2 + (by - ay) ** 2) ** 0.5
    for point in vertices:
        if point == a or point == b:
            continue
        px, py = point
        if _point_segment_distance(px, py, ax, ay, bx, by) < 7:
            if ((px - ax) ** 2 + (py - ay) ** 2) ** 0.5 < length - 7 and ((px - bx) ** 2 + (py - by) ** 2) ** 0.5 < length - 7:
                return True
    return False


def _edge_present(dark: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    ax, ay = a
    bx, by = b
    length = ((bx - ax) ** 2 + (by - ay) ** 2) ** 0.5
    if length < 20:
        return False
    samples = max(8, int(length))
    hits = 0
    tested = 0
    end_fraction = min(0.24, 12.0 / length)
    height, width = dark.shape
    for i in range(samples + 1):
        t = i / samples
        if t <= end_fraction or t >= 1.0 - end_fraction:
            continue
        x = int(round(ax + t * (bx - ax)))
        y = int(round(ay + t * (by - ay)))
        x0, x1 = max(0, x - 2), min(width, x + 3)
        y0, y1 = max(0, y - 2), min(height, y + 3)
        tested += 1
        hits += bool(dark[y0:y1, x0:x1].any())
    return tested > 0 and hits / tested > 0.90


def _simple_path_count(graph: list[set[int]], start: int, target: int, cap: int = 20) -> int:
    count = 0
    stack: list[tuple[int, frozenset[int]]] = [(start, frozenset((start,)))]
    while stack:
        node, visited = stack.pop()
        if node == target:
            count += 1
            if count >= cap:
                return count
            continue
        for nxt in graph[node]:
            if nxt not in visited:
                stack.append((nxt, visited | {nxt}))
    return count


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    starts = _centers(arr, GREEN, minimum=40)
    if len(starts) != 1:
        raise ValueError(f"expected one green start, found {len(starts)}")
    junctions = _centers(arr, JUNCTION, minimum=30)
    if len(junctions) < 8:
        raise ValueError("too few visible junction markers")
    terminals: list[tuple[str, tuple[float, float]]] = []
    for name, rgb in TERMINALS.items():
        found = _centers(arr, rgb, minimum=80)
        if len(found) != 1:
            raise ValueError(f"expected one {name} terminal, found {len(found)}")
        terminals.append((name, found[0]))

    vertices = [starts[0], *junctions, *(point for _, point in terminals)]
    dark = np.all(arr == np.asarray(DARK, dtype=np.uint8), axis=2)
    graph = [set() for _ in vertices]
    for i, left in enumerate(vertices):
        for j in range(i + 1, len(vertices)):
            right = vertices[j]
            if _has_intermediate(left, right, vertices):
                continue
            if _edge_present(dark, left, right):
                graph[i].add(j)
                graph[j].add(i)
    if not graph[0]:
        raise ValueError("start has no recovered edge")

    winners = []
    terminal_offset = 1 + len(junctions)
    for index, (name, _) in enumerate(terminals):
        count = _simple_path_count(graph, 0, terminal_offset + index)
        if count == 2:
            winners.append(name)
    if len(winners) != 1:
        raise ValueError(f"expected one two-route terminal, found {winners}")
    return winners[0]
