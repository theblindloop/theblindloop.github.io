"""Independent PNG-only inverse arm for Rollback Bead Trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PROTOTYPES = {
    "F": ((39, 177, 218), (30, 158, 204)),
    "S": ((238, 174, 48), (246, 188, 66)),
    "R": ((229, 91, 100), (213, 76, 91)),
    "START": ((119, 92, 214),),
}


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        # The dark outline leaves about 73--75 solid-color pixels per clean bead.
        # Reject a bead after more than a thin antialias fringe has been erased.
        if len(points) >= 65:
            array = np.asarray(points, dtype=np.float64)
            found.append((len(points), float(array[:, 1].mean()), float(array[:, 0].mean())))
    return found


def _mask(array: np.ndarray, prototypes: tuple[tuple[int, int, int], ...]) -> np.ndarray:
    rgb = array.astype(np.int16)
    distances = [np.max(np.abs(rgb - np.asarray(color, dtype=np.int16)), axis=2) for color in prototypes]
    return np.min(np.stack(distances), axis=0) <= 28


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    tokens: list[tuple[str, float, float]] = []
    for operation in ("F", "S", "R"):
        components = _components(_mask(array, PROTOTYPES[operation]))
        tokens.extend((operation, x, y) for _, x, y in components)
    starts = _components(_mask(array, PROTOTYPES["START"]))
    if len(starts) != 1 or len(tokens) != 6:
        raise ValueError("bead trace is incomplete or ambiguous")
    counts = {key: sum(token[0] == key for token in tokens) for key in ("F", "S", "R")}
    if counts != {"F": 3, "S": 1, "R": 2}:
        raise ValueError("operation multiset is invalid")

    _, start_x, start_y = starts[0]
    points = np.asarray([(x, y) for _, x, y in tokens], dtype=np.float64)
    direction = points.mean(axis=0) - np.asarray([start_x, start_y])
    norm = float(np.linalg.norm(direction))
    if norm < 25.0:
        raise ValueError("trace direction is unresolved")
    direction /= norm
    projections = (points - np.asarray([start_x, start_y])) @ direction
    order = np.argsort(projections)
    sorted_projections = projections[order]
    if np.any(np.diff(sorted_projections) < 9.0):
        raise ValueError("bead order is ambiguous")
    sequence = [tokens[int(index)][0] for index in order]

    lamp = False
    checkpoint = False
    for operation in sequence:
        if operation == "F":
            lamp = not lamp
        elif operation == "S":
            checkpoint = lamp
        else:
            lamp = checkpoint
    return "yes" if lamp else "no"
