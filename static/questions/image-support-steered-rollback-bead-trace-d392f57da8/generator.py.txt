"""Deterministic renderer and analytic replay for Rollback Bead Trace."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
OP_COUNTS = {"F": 3, "S": 1, "R": 2}
PALETTES = (
    {"F": (39, 177, 218), "S": (238, 174, 48), "R": (229, 91, 100)},
    {"F": (30, 158, 204), "S": (246, 188, 66), "R": (213, 76, 91)},
)
BACKGROUND = (248, 244, 235)
WIRE = (42, 51, 61)
START = (119, 92, 214)


def _replay(operations: list[str] | tuple[str, ...]) -> bool:
    lamp = False
    checkpoint = False
    for operation in operations:
        if operation == "F":
            lamp = not lamp
        elif operation == "S":
            checkpoint = lamp
        elif operation == "R":
            lamp = checkpoint
        else:
            raise ValueError(f"unknown operation {operation!r}")
    return lamp


def _sequence_pools() -> dict[bool, tuple[tuple[str, ...], ...]]:
    sequences = sorted(set(itertools.permutations("FFFSRR")))
    return {
        state: tuple(sequence for sequence in sequences if _replay(sequence) is state)
        for state in (False, True)
    }


SEQUENCE_POOLS = _sequence_pools()
assert len(SEQUENCE_POOLS[False]) == len(SEQUENCE_POOLS[True]) == 30


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = bool(int(seed) % 2)
    operations = list(rng.choice(SEQUENCE_POOLS[target]))
    orientation = rng.choice(["east", "west", "south", "north"])
    # Cell-centered anchors keep the short support away from arbitrary 8x8 grid lines.
    anchor_x = rng.choice([160, 224, 288, 352]) + rng.randint(-3, 3)
    anchor_y = rng.choice([160, 224, 288, 352]) + rng.randint(-3, 3)
    return {
        "operations": operations,
        "orientation": orientation,
        "anchor_x": anchor_x,
        "anchor_y": anchor_y,
        "canvas_size": CANVAS,
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def _basis(scene: dict) -> tuple[float, float]:
    return {
        "east": (1.0, 0.0),
        "west": (-1.0, 0.0),
        "south": (0.0, 1.0),
        "north": (0.0, -1.0),
    }[scene["orientation"]]


def _point(scene: dict, along: float, across: float = 0.0) -> tuple[float, float]:
    ux, uy = _basis(scene)
    vx, vy = -uy, ux
    return (
        float(scene["anchor_x"]) + along * ux + across * vx,
        float(scene["anchor_y"]) + along * uy + across * vy,
    )


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    if size != CANVAS:
        raise ValueError("unsupported canvas")
    operations = list(scene["operations"])
    if len(operations) != 6:
        raise ValueError("six operations required")
    palette = PALETTES[int(scene["palette_variant"])]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def p(point: tuple[float, float]) -> tuple[int, int]:
        return (round(point[0] * scale), round(point[1] * scale))

    start = _point(scene, -55)
    end = _point(scene, 55)
    draw.line([p(start), p(end)], fill=WIRE, width=5 * scale)

    # Round purple START terminus.
    sx, sy = p(start)
    sr = 8 * scale
    draw.ellipse((sx - sr, sy - sr, sx + sr, sy + sr), fill=START, outline=WIRE, width=2 * scale)

    for operation, along in zip(operations, (-40, -24, -8, 8, 24, 40), strict=True):
        cx, cy = p(_point(scene, along))
        radius = 7 * scale
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=palette[operation],
            outline=WIRE,
            width=2 * scale,
        )

    # Filled arrow at the reading end.
    tip = _point(scene, 61)
    base_a = _point(scene, 48, 8)
    base_b = _point(scene, 48, -8)
    draw.polygon([p(tip), p(base_a), p(base_b)], fill=WIRE)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if _replay(scene["operations"]) else "no"


def margin(scene: dict) -> float:
    # A valid replay has a one-bit separation from the only runner-up decision.
    _replay(scene["operations"])
    return 1.0


def is_quarantined(scene: dict) -> bool:
    operations = list(scene.get("operations", []))
    return len(operations) != 6 or any(operations.count(key) != value for key, value in OP_COUNTS.items())


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
