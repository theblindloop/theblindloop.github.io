"""Bracket-cord partner binding world.

One thin slate cord runs as a staircase from one corner of the board to the
opposite corner. Twelve small coloured squares (beads) sit on the cord. A bead
is either SOLID (an opening bead) or HOLLOW, i.e. drawn with a cream window cut
out of its middle (a closing bead). Read along the cord from the black round
cap, the solid/hollow beads form a balanced, properly nested bracket word: each
hollow bead closes the most recently opened solid bead that is still open.

Exactly one solid bead is surrounded by a magenta square frame. The answer is
the colour of the hollow bead that closes it. That partner is always at least
five beads further along the cord and lies far away from the framed bead, so the
answer can only be obtained by walking the whole intervening stretch of the one
cord and keeping a nesting count. The cord deliberately contains one leftward
run, so the reading order along the cord is not the order of the beads by
coordinate; scenes in which a coordinate sort would give the right answer are
rejected at sampling time.

Geometry is laid out so that non-adjacent stretches of the cord stay far apart,
no bead sits on a corner of the staircase, and the only near-boundary quantity
is the width of the cream window that distinguishes a hollow bead from a solid
one.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "bracket_cord_partner_bind-0.1.0"

CANVAS_W = 880
CANVAS_H = 880
MARGIN = 78

BACKGROUND = (250, 246, 236)
CORD = (92, 100, 112)
CORD_WIDTH = 7
CAP = (18, 18, 18)
CAP_RADIUS = 13
FRAME = (214, 16, 140)
FRAME_SIDE = 60
FRAME_STROKE = 5

BEAD_SIDE = 34
N_BEADS = 12
QUARANTINE_MARGIN = 8.0

COLOR_NAMES = ("red", "orange", "green", "teal", "blue", "purple")
COLORS = {
    "red": (206, 42, 42),
    "orange": (232, 124, 28),
    "green": (46, 150, 62),
    "teal": (26, 158, 168),
    "blue": (48, 86, 208),
    "purple": (140, 58, 180),
}

_STEPS = 7  # seven horizontal and seven vertical runs along the staircase


# ---------------------------------------------------------------- geometry


def _canonical_path(rng: random.Random) -> list[list[int]]:
    """A staircase from (MARGIN, MARGIN) to the opposite corner.

    x-cuts advance monotonically except for one deliberate leftward run; the
    y-levels always advance, which keeps the polyline a simple (non
    self-intersecting) curve while still defeating coordinate sorting.
    """
    span_x = CANVAS_W - 2 * MARGIN
    span_y = CANVAS_H - 2 * MARGIN
    xs = [MARGIN + round(i * span_x / _STEPS) for i in range(_STEPS + 1)]
    ys = [MARGIN + round(i * span_y / _STEPS) for i in range(_STEPS + 1)]
    for i in range(1, _STEPS):
        xs[i] += rng.randint(-15, 15)
        ys[i] += rng.randint(-15, 15)

    back = rng.choice((2, 3, 4))
    xs[back] += 150
    xs[back + 1] -= 60

    points = [[xs[0], ys[0]]]
    for i in range(_STEPS):
        points.append([xs[i + 1], ys[i]])
        points.append([xs[i + 1], ys[i + 1]])
    return points


def _segments(path: list[list[int]]) -> list[tuple[list[int], list[int], float]]:
    out = []
    for a, b in zip(path, path[1:]):
        length = math.hypot(b[0] - a[0], b[1] - a[1])
        if length > 0.0:
            out.append((a, b, length))
    return out


def _total_length(path: list[list[int]]) -> float:
    return sum(seg[2] for seg in _segments(path))


def _point_at(path: list[list[int]], t: float) -> tuple[int, int]:
    walked = 0.0
    for a, b, length in _segments(path):
        if t <= walked + length:
            frac = (t - walked) / length
            return (round(a[0] + frac * (b[0] - a[0])), round(a[1] + frac * (b[1] - a[1])))
        walked += length
    end = path[-1]
    return (end[0], end[1])


def _vertex_arcs(path: list[list[int]]) -> list[float]:
    arcs = [0.0]
    for _, _, length in _segments(path):
        arcs.append(arcs[-1] + length)
    return arcs


def arc_position(path: list[list[int]], point: tuple[int, int]) -> float:
    """Arc length of the cord point nearest to ``point``.

    Bead centres are placed exactly on the polyline, so this recovers their
    position along the cord without using any stored ordering.
    """
    best = (float("inf"), 0.0)
    walked = 0.0
    for a, b, length in _segments(path):
        dx, dy = b[0] - a[0], b[1] - a[1]
        frac = ((point[0] - a[0]) * dx + (point[1] - a[1]) * dy) / (length * length)
        frac = min(1.0, max(0.0, frac))
        px, py = a[0] + frac * dx, a[1] + frac * dy
        distance = math.hypot(point[0] - px, point[1] - py)
        if distance < best[0]:
            best = (distance, walked + frac * length)
        walked += length
    return best[1]


def _bead_arcs(rng: random.Random, path: list[list[int]]) -> list[float] | None:
    """Evenly spaced bead positions nudged off the staircase corners."""
    total = _total_length(path)
    vertices = _vertex_arcs(path)[1:-1]
    chosen: list[float] = []
    for index in range(N_BEADS):
        base = (index + 1) * total / (N_BEADS + 1) + rng.randint(-8, 8)
        best = None
        for delta in range(0, 46, 2):
            for candidate in (base - delta, base + delta):
                if candidate < 46.0 or candidate > total - 46.0:
                    continue
                if all(abs(candidate - v) >= 30.0 for v in vertices):
                    best = candidate
                    break
            if best is not None:
                break
        if best is None:
            return None
        chosen.append(best)
    chosen.sort()
    if any(b - a < 78.0 for a, b in zip(chosen, chosen[1:])):
        return None
    return chosen


def _run_clearance(path: list[list[int]]) -> float:
    """Smallest distance between two non-adjacent runs of the cord."""
    segs = _segments(path)
    best = float("inf")
    for i, (a1, b1, _) in enumerate(segs):
        for j, (a2, b2, _) in enumerate(segs):
            if j <= i + 1:
                continue
            for p in (a1, b1):
                for q in (a2, b2):
                    best = min(best, math.hypot(p[0] - q[0], p[1] - q[1]))
            # axis-aligned runs: also compare the perpendicular offsets
            if a1[0] == b1[0] and a2[0] == b2[0]:
                lo1, hi1 = sorted((a1[1], b1[1]))
                lo2, hi2 = sorted((a2[1], b2[1]))
                if hi1 >= lo2 and hi2 >= lo1:
                    best = min(best, abs(a1[0] - a2[0]))
            if a1[1] == b1[1] and a2[1] == b2[1]:
                lo1, hi1 = sorted((a1[0], b1[0]))
                lo2, hi2 = sorted((a2[0], b2[0]))
                if hi1 >= lo2 and hi2 >= lo1:
                    best = min(best, abs(a1[1] - a2[1]))
    return best


# ---------------------------------------------------------------- brackets


def _dyck_word(rng: random.Random) -> list[bool]:
    """A balanced, properly nested open/close word of length ``N_BEADS``."""
    for _ in range(200):
        word = [True] * (N_BEADS // 2) + [False] * (N_BEADS // 2)
        rng.shuffle(word)
        depth = 0
        ok = True
        for item in word:
            depth += 1 if item else -1
            if depth < 0:
                ok = False
                break
        if ok and depth == 0:
            return word
    raise RuntimeError("could not sample a balanced bracket word")


def match_partners(word: list[bool]) -> dict[int, int] | None:
    """Map each opening index to the index that closes it, or ``None``."""
    stack: list[int] = []
    pairs: dict[int, int] = {}
    for index, is_open in enumerate(word):
        if is_open:
            stack.append(index)
        else:
            if not stack:
                return None
            pairs[stack.pop()] = index
    return None if stack else pairs


# ---------------------------------------------------------------- sampling


def _reading_order(scene: dict) -> list[int]:
    """Bead indices in cord order, starting at the capped end."""
    path = [list(p) for p in scene["path"]]
    order = sorted(
        range(len(scene["beads"])),
        key=lambda i: arc_position(path, (scene["beads"][i]["x"], scene["beads"][i]["y"])),
    )
    return order if scene["cap_at_start"] else list(reversed(order))


def _answer_for_order(scene: dict, order: list[int]) -> str | None:
    beads = scene["beads"]
    word = [bool(beads[i]["open"]) for i in order]
    pairs = match_partners(word)
    if pairs is None:
        return None
    marked_slot = order.index(int(scene["marked"]))
    if marked_slot not in pairs:
        return None
    return str(beads[order[pairs[marked_slot]]]["color"])


def _coordinate_order(scene: dict) -> list[int]:
    """The cheap shortcut order: sort by row band, then by column."""
    beads = scene["beads"]
    return sorted(range(len(beads)), key=lambda i: (beads[i]["y"], beads[i]["x"]))


def _radial_order(scene: dict) -> list[int]:
    """The cheap shortcut order: sort by straight-line distance from the cap."""
    path = scene["path"]
    cap = path[0] if scene["cap_at_start"] else path[-1]
    beads = scene["beads"]
    return sorted(
        range(len(beads)),
        key=lambda i: math.hypot(beads[i]["x"] - cap[0], beads[i]["y"] - cap[1]),
    )


def _hole_size(seed: int) -> int:
    if seed % 5 == 3:
        return 5 + 2 * (seed % 2)
    return 11 + 2 * (seed % 5)


def sample_scene(seed: int) -> dict:
    """Deterministically sample one balanced scene."""
    rng = random.Random(seed * 7919 + 13)
    target = COLOR_NAMES[seed % len(COLOR_NAMES)]
    hole = _hole_size(seed)
    flip_x = bool(rng.getrandbits(1))
    flip_y = bool(rng.getrandbits(1))
    diagonal = math.hypot(CANVAS_W, CANVAS_H)

    for _ in range(400):
        path = _canonical_path(rng)
        if flip_x:
            path = [[CANVAS_W - x, y] for x, y in path]
        if flip_y:
            path = [[x, CANVAS_H - y] for x, y in path]
        if _run_clearance(path) < 60.0:
            continue
        arcs = _bead_arcs(rng, path)
        if arcs is None:
            continue
        points = [_point_at(path, t) for t in arcs]
        cap_at_start = bool(rng.getrandbits(1))
        order = list(range(N_BEADS)) if cap_at_start else list(reversed(range(N_BEADS)))

        word = _dyck_word(rng)
        pairs = match_partners(word)
        assert pairs is not None
        eligible = [
            (o, c)
            for o, c in sorted(pairs.items())
            if c - o >= 5
            and c != N_BEADS - 1
            and math.hypot(
                points[order[o]][0] - points[order[c]][0],
                points[order[o]][1] - points[order[c]][1],
            )
            >= 0.32 * diagonal
        ]
        if not eligible:
            continue
        marked_slot, partner_slot = eligible[rng.randrange(len(eligible))]

        closes = [s for s, is_open in enumerate(word) if not is_open]
        traps = {marked_slot}
        after = [s for s in closes if s > marked_slot]
        if after:
            traps.add(after[0])
        traps.add(closes[-1])
        nearest_close = min(
            closes,
            key=lambda s: math.hypot(
                points[order[s]][0] - points[order[marked_slot]][0],
                points[order[s]][1] - points[order[marked_slot]][1],
            ),
        )
        traps.add(nearest_close)
        traps.discard(partner_slot)

        others = [s for s in range(N_BEADS) if s != partner_slot and s not in traps]
        if not others:
            continue
        slot_colors = [COLOR_NAMES[rng.randrange(len(COLOR_NAMES))] for _ in range(N_BEADS)]
        slot_colors[partner_slot] = target
        slot_colors[others[rng.randrange(len(others))]] = target
        for slot in traps:
            while slot_colors[slot] == target:
                slot_colors[slot] = COLOR_NAMES[rng.randrange(len(COLOR_NAMES))]
        if len(set(slot_colors)) < 4:
            continue

        beads = [None] * N_BEADS
        for slot, bead_index in enumerate(order):
            beads[bead_index] = {
                "x": int(points[bead_index][0]),
                "y": int(points[bead_index][1]),
                "color": slot_colors[slot],
                "open": bool(word[slot]),
            }
        scene = {
            "path": [[int(x), int(y)] for x, y in path],
            "beads": beads,
            "marked": int(order[marked_slot]),
            "cap_at_start": cap_at_start,
            "hole": int(hole),
        }
        if _answer_for_order(scene, _reading_order(scene)) != target:
            continue
        if _answer_for_order(scene, _coordinate_order(scene)) == target:
            continue
        if _answer_for_order(scene, _radial_order(scene)) == target:
            continue
        return scene
    raise RuntimeError(f"could not sample a balanced scene for seed {seed}")


# ---------------------------------------------------------------- rendering


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BACKGROUND)
    draw = ImageDraw.Draw(image)
    path = [(int(x), int(y)) for x, y in scene["path"]]
    half = CORD_WIDTH // 2

    for a, b in zip(path, path[1:]):
        draw.line([a, b], fill=CORD, width=CORD_WIDTH)
    for x, y in path:
        draw.rectangle([x - half, y - half, x + half, y + half], fill=CORD)

    cap = path[0] if scene["cap_at_start"] else path[-1]
    draw.ellipse(
        [cap[0] - CAP_RADIUS, cap[1] - CAP_RADIUS, cap[0] + CAP_RADIUS, cap[1] + CAP_RADIUS],
        fill=CAP,
    )

    side = BEAD_SIDE // 2
    hole = int(scene["hole"]) // 2
    for bead in scene["beads"]:
        x, y = int(bead["x"]), int(bead["y"])
        draw.rectangle([x - side, y - side, x + side, y + side], fill=COLORS[bead["color"]])
        if not bead["open"]:
            draw.rectangle([x - hole, y - hole, x + hole, y + hole], fill=BACKGROUND)

    marked = scene["beads"][int(scene["marked"])]
    mx, my = int(marked["x"]), int(marked["y"])
    outer = FRAME_SIDE // 2
    for step in range(FRAME_STROKE):
        radius = outer - step
        draw.rectangle([mx - radius, my - radius, mx + radius, my + radius], outline=FRAME)
    return image


# ---------------------------------------------------------------- gold


def decisive_pair(scene: dict) -> tuple[int, int]:
    """Bead indices of the framed bead and of the bead that closes it."""
    order = _reading_order(scene)
    word = [bool(scene["beads"][i]["open"]) for i in order]
    pairs = match_partners(word)
    if pairs is None:
        raise ValueError("scene bracket word is not balanced")
    marked_slot = order.index(int(scene["marked"]))
    if marked_slot not in pairs:
        raise ValueError("framed bead does not open a group")
    return int(scene["marked"]), int(order[pairs[marked_slot]])


def analytic_gold(scene: dict) -> str:
    _, partner = decisive_pair(scene)
    return str(scene["beads"][partner]["color"])


def margin(scene: dict) -> float:
    """Smallest visible clearance in pixels; the bead window dominates."""
    beads = scene["beads"]
    values = [float(int(scene["hole"]))]
    for i, a in enumerate(beads):
        for b in beads[i + 1 :]:
            gap = math.hypot(a["x"] - b["x"], a["y"] - b["y"]) - BEAD_SIDE
            values.append(float(gap))
    values.append(_run_clearance([list(p) for p in scene["path"]]) - CORD_WIDTH)
    marked = beads[int(scene["marked"])]
    for i, bead in enumerate(beads):
        if i == int(scene["marked"]):
            continue
        gap = (
            math.hypot(marked["x"] - bead["x"], marked["y"] - bead["y"])
            - FRAME_SIDE / 2.0
            - BEAD_SIDE / 2.0
        )
        values.append(float(gap))
    for x, y in scene["path"]:
        values.append(float(min(x, y, CANVAS_W - x, CANVAS_H - y) - CORD_WIDTH))
    return round(min(values), 6)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cord vertex subdivision is a pure alias.

    The raster is drawn from the polyline, and the analytic answer is recomputed
    from arc length along that polyline. Inserting collinear vertices therefore
    re-draws a byte-identical picture and leaves the answer unchanged.
    """
    path = [list(p) for p in scene["path"]]
    a, b = path[0], path[1]
    single = [a, [(a[0] + b[0]) // 2, (a[1] + b[1]) // 2], *path[1:]]
    quarter = [
        a,
        [(3 * a[0] + b[0]) // 4, (3 * a[1] + b[1]) // 4],
        [(a[0] + b[0]) // 2, (a[1] + b[1]) // 2],
        [(a[0] + 3 * b[0]) // 4, (a[1] + 3 * b[1]) // 4],
        *path[1:],
    ]
    return [
        ("cord_vertex_subdivision", {**scene, "path": single}),
        ("cord_vertex_quarter_subdivision", {**scene, "path": quarter}),
    ]
