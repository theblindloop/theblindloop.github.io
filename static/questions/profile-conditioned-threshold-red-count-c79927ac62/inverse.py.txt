"""Independent pixel-only inverse arm for the threshold-subset count world.

This module receives ONLY the rasterized PNG. It re-estimates the gray threshold
line from its own pixels (centroid + principal axis of the line mask, with the
direction fixed to point left-to-right) and then counts how many red dots lie
strictly on the upper side of that recovered line. It imports only numpy and
PIL, shares no code or coordinates with the renderer, and never reads latent
scene data, analytic gold, or any renderer/generator/verifier module.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "threshold_red_count-oracle-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY = (95, 95, 100)
BACKGROUND = (250, 250, 250)

DOT_PIXEL_MIN = 20


def _labels(arr: np.ndarray) -> np.ndarray:
    refs = np.stack(
        [
            np.array(COLOR_RGB["red"], dtype=np.float32),
            np.array(COLOR_RGB["blue"], dtype=np.float32),
            np.array(COLOR_RGB["green"], dtype=np.float32),
            np.array(GRAY, dtype=np.float32),
            np.array(BACKGROUND, dtype=np.float32),
        ],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _line_direction_and_center(gray_mask: np.ndarray):
    ys, xs = np.nonzero(gray_mask)
    if len(ys) < 50:
        return None
    x = xs.astype(np.float64)
    y = ys.astype(np.float64)
    cx = x.mean()
    cy = y.mean()
    xc = x - cx
    yc = y - cy
    cov = np.cov(np.stack([xc, yc]))
    vals, evec = np.linalg.eigh(cov)
    v = evec[:, int(np.argmax(vals))]
    ux, uy = float(v[0]), float(v[1])
    norm = math.hypot(ux, uy)
    ux /= norm
    uy /= norm
    if ux < 0:
        ux, uy = -ux, -uy
    return cx, cy, ux, uy


def _red_centroids(red_mask: np.ndarray):
    ys, xs = np.nonzero(red_mask)
    if len(ys) == 0:
        return []
    points = list(set(zip(xs.tolist(), ys.tolist())))
    seen = set()
    centroids = []
    for start in points:
        if start in seen:
            continue
        comp = [start]
        seen.add(start)
        idx = 0
        sx = sy = 0
        while idx < len(comp):
            x, y = comp[idx]
            idx += 1
            sx += x
            sy += y
            for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nb = (x + dx, y + dy)
                if nb in seen or nb not in points:
                    continue
                seen.add(nb)
                comp.append(nb)
        if len(comp) >= DOT_PIXEL_MIN:
            centroids.append((sx / len(comp), sy / len(comp)))
    return centroids


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    gray_idx = 3
    line = _line_direction_and_center(labels == gray_idx)
    if line is None:
        return "0"
    cx, cy, ux, uy = line
    red_centroids = _red_centroids(labels == 0)
    count = 0
    for x, y in red_centroids:
        d = ux * (y - cy) - uy * (x - cx)
        if d < 0:
            count += 1
    return str(count)
