"""Threshold-subset count measurement world.

A gray straight threshold line crosses a light panel at a randomised small tilt.
Red, blue and green dots are placed on both sides of the line. The decision is a
measurement recovered purely from the raster: estimate the threshold line's
position and running direction from its gray pixels, then count exactly how many
red dots lie strictly on the specified (upper) side of that recovered line.

The analysed quantity is a subset count conditioned on a RECOVERED sloped
threshold, not a total disk tally and not a single-dot side predicate. Which
side is "above" is fixed by a left-to-right convention on the line's direction,
so the pixel oracle and the analytic gold always agree. Every scene carries a
clear perpendicular clearance between the line and each dot so the side
assignment is never ambiguous at finite resolution.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "threshold_red_count-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY = (95, 95, 100)
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
CENTER = (240, 175)
LINE_WIDTH = 4
DOT_R = 7
LINE_TILT_MAX = 20          # |tilt| in degrees; the line stays near-horizontal
RED_COUNT = 6               # total red dots; the answer is how many lie above
BLUE_COUNT = 4              # distractor dots (their placement is irrelevant)
GREEN_COUNT = 3             # distractor dots
EDGE_MARGIN = 24
DOT_SEP = 22                # min centre distance between any two dots
CLEARANCE_MIN = 5.0         # quarantine: a red dot closer than this is ambiguous
MIN_MARGIN = 7.0            # emitted scenes keep every red dot at least this far


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 733 + 41)
    cx, cy = CENTER
    while True:
        tilt = rng.uniform(-LINE_TILT_MAX, LINE_TILT_MAX)
        line = {"angle_deg": float(tilt), "center": [cx, cy]}
        angle = math.radians(tilt)
        ux = math.cos(angle)          # angle restricted so ux > 0 (left-to-right)
        uy = math.sin(angle)

        def dot_positions(count: int):
            points = []
            attempts = 0
            while len(points) < count and attempts < count * 600:
                attempts += 1
                x = rng.uniform(EDGE_MARGIN, WIDTH - EDGE_MARGIN)
                y = rng.uniform(EDGE_MARGIN, HEIGHT - EDGE_MARGIN)
                d = ux * (y - cy) - uy * (x - cx)
                if abs(d) < CLEARANCE_MIN:
                    continue
                if any((x - px) ** 2 + (y - py) ** 2 < DOT_SEP ** 2 for px, py in points):
                    continue
                points.append([round(x, 3), round(y, 3)])
            return points

        red = dot_positions(RED_COUNT)
        if len(red) < RED_COUNT:
            continue
        blue = dot_positions(BLUE_COUNT)
        green = dot_positions(GREEN_COUNT)
        if len(blue) < BLUE_COUNT or len(green) < GREEN_COUNT:
            continue
        tiles = [
            (ux * (y - cy) - uy * (x - cx)) for x, y in red
        ]
        if max(abs(t) for t in tiles) < MIN_MARGIN:
            continue
        scene = {
            "line": line,
            "red_dots": red,
            "blue_dots": blue,
            "green_dots": green,
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        if _count_red_above(scene) is None:
            continue
        return scene


def _line_params(scene: dict):
    tilt = scene["line"]["angle_deg"]
    cx, cy = scene["line"]["center"]
    angle = math.radians(tilt)
    ux = math.cos(angle)
    uy = math.sin(angle)
    if ux < 0:
        ux, uy = -ux, -uy
    return cx, cy, ux, uy


def _count_red_above(scene: dict):
    cx, cy, ux, uy = _line_params(scene)
    n = 0
    clearances = []
    for x, y in scene["red_dots"]:
        d = (ux * (y - cy) - uy * (x - cx)) / math.hypot(ux, uy)
        clearances.append(abs(d))
        if d < 0:
            n += 1
    return n, min(clearances)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    cx, cy = scene["line"]["center"]
    angle = math.radians(scene["line"]["angle_deg"])
    half = 400.0
    x1 = cx - half * math.cos(angle)
    y1 = cy - half * math.sin(angle)
    x2 = cx + half * math.cos(angle)
    y2 = cy + half * math.sin(angle)
    draw.line([x1, y1, x2, y2], fill=GRAY, width=LINE_WIDTH)
    for color, dots in (("red", scene["red_dots"]), ("blue", scene["blue_dots"]), ("green", scene["green_dots"])):
        for x, y in dots:
            draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=COLOR_RGB[color])
    return img


def analytic_gold(scene: dict) -> str:
    n, _ = _count_red_above(scene)
    return str(n)


def margin(scene: dict) -> float:
    _, clearance = _count_red_above(scene)
    return float(clearance)


def is_quarantined(scene: dict) -> bool:
    n, clearance = _count_red_above(scene)
    if n is None:
        return True
    return clearance < CLEARANCE_MIN


def latent_symmetries(scene: dict) -> list:
    return []
