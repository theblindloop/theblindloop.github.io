"""Deterministic renderer and analytic arm for center-of-area pin search."""

from __future__ import annotations

import math
import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 246, 240)
CYAN = (42, 174, 190)
WHITE = (255, 255, 255)
PIN_COLORS = {
    "red": (211, 54, 72),
    "gold": (226, 157, 28),
    "violet": (132, 76, 180),
    "green": (45, 145, 91),
}
PIN_ORDER = tuple(PIN_COLORS)
OUTER_RADIUS = 13
INNER_RADIUS = 9


def _make_rows(rng: random.Random) -> list[str]:
    """Make a thick, asymmetric, connected rounded-square polyomino."""
    grid = np.zeros((17, 17), dtype=np.uint8)
    grid[2:15, 1:16] = 1
    grid[1:16, 2:15] = 1
    # Asymmetric edge steps move the centroid without creating thin tendrils.
    side = rng.randrange(4)
    shift = rng.randrange(-4, 5)
    if side == 0:
        grid[0:2, 6 + shift // 2 : 11 + shift // 2] = 1
        grid[13:17, 1:4] = 0
    elif side == 1:
        grid[6 + shift // 2 : 11 + shift // 2, 15:17] = 1
        grid[1:4, 0:4] = 0
    elif side == 2:
        grid[15:17, 6 + shift // 2 : 11 + shift // 2] = 1
        grid[0:4, 13:17] = 0
    else:
        grid[6 + shift // 2 : 11 + shift // 2, 0:2] = 1
        grid[13:17, 13:17] = 0
    # A second broad two-cell-deep step gives non-reflection-equivalent scenes.
    y = rng.randrange(4, 10)
    if side % 2 == 0:
        grid[y : y + 4, 0:2] = 1
    else:
        grid[0:2, y : y + 4] = 1
    return ["".join("1" if value else "0" for value in row) for row in grid]


def _plate_pixels(scene: dict) -> np.ndarray:
    spec = scene["plate_mask"]
    x0, y0, cell = int(spec["x"]), int(spec["y"]), int(spec["cell"])
    mask = np.zeros((HEIGHT, WIDTH), dtype=bool)
    for gy, row in enumerate(spec["rows"]):
        for gx, value in enumerate(row):
            if value == "1":
                mask[y0 + gy * cell : y0 + (gy + 1) * cell,
                     x0 + gx * cell : x0 + (gx + 1) * cell] = True
    return mask


def _visible_cyan(scene: dict) -> np.ndarray:
    mask = _plate_pixels(scene)
    yy, xx = np.ogrid[:HEIGHT, :WIDTH]
    for pin in scene["pins"]:
        px, py = int(pin["x"]), int(pin["y"])
        mask[(xx - px) ** 2 + (yy - py) ** 2 <= OUTER_RADIUS ** 2] = False
    return mask


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("plate has no visible cyan area")
    return float(xs.mean()), float(ys.mean())


def _ranking(scene: dict) -> list[tuple[float, str]]:
    cx, cy = _centroid(_visible_cyan(scene))
    return sorted(
        (math.hypot(float(pin["x"]) - cx, float(pin["y"]) - cy), str(pin["color"]))
        for pin in scene["pins"]
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = PIN_ORDER[int(seed) % len(PIN_ORDER)]
    for _ in range(200):
        cell = 14
        plate = {
            "x": 137 + rng.randrange(-18, 19),
            "y": 137 + rng.randrange(-18, 19),
            "cell": cell,
            "rows": _make_rows(rng),
        }
        provisional = {"plate_mask": plate, "pins": []}
        cx, cy = _centroid(_plate_pixels(provisional))
        rotation = rng.uniform(0.0, 2.0 * math.pi)
        radii = [rng.uniform(19.0, 25.0), rng.uniform(42.0, 48.0),
                 rng.uniform(55.0, 63.0), rng.uniform(68.0, 76.0)]
        remaining = [color for color in PIN_ORDER if color != desired]
        rng.shuffle(remaining)
        assignments = [desired, *remaining]
        pins = []
        for index, (radius, color) in enumerate(zip(radii, assignments, strict=True)):
            angle = rotation + index * (2.0 * math.pi / 4.0) + rng.uniform(-0.18, 0.18)
            pins.append({
                "color": color,
                "x": int(round(cx + radius * math.cos(angle))),
                "y": int(round(cy + radius * math.sin(angle))),
            })
        rng.shuffle(pins)
        scene = {"plate_mask": plate, "pins": pins, "note": f"draw-{seed}-a"}
        ranked = _ranking(scene)
        if ranked[0][1] == desired and ranked[1][0] - ranked[0][0] >= 14.0:
            return scene
    raise RuntimeError("could not sample a clear centroid-pin scene")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    pixels = np.asarray(image).copy()
    pixels[_visible_cyan(scene)] = CYAN
    image = Image.fromarray(pixels, mode="RGB")
    draw = ImageDraw.Draw(image)
    for pin in scene["pins"]:
        x, y = int(pin["x"]), int(pin["y"])
        draw.ellipse((x - OUTER_RADIUS, y - OUTER_RADIUS,
                      x + OUTER_RADIUS, y + OUTER_RADIUS), fill=WHITE)
        draw.ellipse((x - INNER_RADIUS, y - INNER_RADIUS,
                      x + INNER_RADIUS, y + INNER_RADIUS),
                     fill=PIN_COLORS[str(pin["color"])])
    return image


def analytic_gold(scene: dict) -> str:
    return _ranking(scene)[0][1]


def margin(scene: dict) -> float:
    ranked = _ranking(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["note"] = "render-invariant-audit-note"
    return [("redundant_note_alias", twin)]
