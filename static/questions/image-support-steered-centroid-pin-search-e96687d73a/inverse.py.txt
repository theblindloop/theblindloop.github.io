"""Independent image-only inverse arm for center-of-area pin search."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

CYAN = np.array((42, 174, 190), dtype=np.int32)
PIN_COLORS = {
    "red": np.array((211, 54, 72), dtype=np.int32),
    "gold": np.array((226, 157, 28), dtype=np.int32),
    "violet": np.array((132, 76, 180), dtype=np.int32),
    "green": np.array((45, 145, 91), dtype=np.int32),
}


def _mask_near(array: np.ndarray, color: np.ndarray, tolerance: float = 10.0) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sqrt(np.sum(delta * delta, axis=2)) <= tolerance


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        component: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            component.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(component) > len(best):
            best = component
    output = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        output[np.asarray(ys), np.asarray(xs)] = True
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan = _largest_component(_mask_near(array, CYAN))
    ys, xs = np.nonzero(cyan)
    # A center-of-area estimate is not meaningful after a large crop removes
    # roughly a tenth of this uniformly filled plate.  The clean renderer
    # yields 41,396 cyan pixels with wide headroom over this integrity floor.
    if len(xs) < 37000:
        raise ValueError("connected cyan plate is missing or too damaged")
    cx, cy = float(xs.mean()), float(ys.mean())
    ranking = []
    for name, color in PIN_COLORS.items():
        mask = _mask_near(array, color)
        py, px = np.nonzero(mask)
        if len(px) < 180:
            raise ValueError(f"{name} pin is missing or too damaged")
        if len(px) > 400:
            raise ValueError(f"{name} pin has implausible area")
        ranking.append((math.hypot(float(px.mean()) - cx, float(py.mean()) - cy), name))
    ranking.sort()
    if ranking[1][0] - ranking[0][0] < 5.0:
        raise ValueError("nearest-pin decision is ambiguous")
    return ranking[0][1]
