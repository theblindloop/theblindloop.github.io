"""Horizontal interval-overlap argmax world.

Which colored horizontal bar shares the longest horizontal overlap with a gray
reference bar, measured as the length of the shared x-span.

Latent scene z carries a list ``rods`` of horizontal filled bars. The first
three carry distinct color labels (red, blue, green) and the fourth is the gray
``reference`` bar. Every bar spans its own isolated horizontal band (distinct
y row) so no two bars' ink touches, and the whole decision is read from the
length of the shared x-interval between each colored rod and the reference.

Mechanism: measure a recovered scalar quantity (shared horizontal interval
length) and take its argmax across a colour-labelled set. The reference bar is a
second visible mark, so the measurement is a relative one between every colored
mark and the neutral reference.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "interval_overlap_argmax-0.1.0"

# Fixed palette shared with oracle.py (plain list so both agree on the named
# answer set without importing each other).
PALETTE = ["red", "blue", "green"]
REFERENCE_COLOR = "gray"
ALL_ROD_COLORS = ["red", "blue", "green", "gray"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
    "gray": (120, 120, 120),
}
BACKGROUND = (250, 250, 250)

WIDTH = 520
HEIGHT = 380
BAR_HEIGHT = 12  # rod thickness in pixels

# Each colored rod is assigned one distinct row band; the reference rod is drawn
# in its own band. Row position never carries the answer.
ROD_ROWS = [160, 220, 280]
REFERENCE_ROW = 100

X_MIN = 30
X_MAX = 330
MIN_ROD_WIDTH = 120.0
MAX_ROD_WIDTH = 230.0

# Minimum top-two overlap-length gap for a decisively selectable scene (px).
MIN_MARGIN_PX = 10.0
# Below this top-two overlap gap the scene is quarantined as near-boundary yet
# still decidable (well above pixel-fit noise).
QUARANTINE_PX = 14.0


def _overlap_len(x0a: float, x1a: float, x0b: float, x1b: float) -> float:
    lo = max(x0a, x0b)
    hi = min(x1a, x1b)
    return max(0.0, hi - lo)


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced extremum."""
    rng = random.Random(seed * 1049 + 11)
    # Reference interval chosen first.
    ref_x0 = rng.uniform(X_MIN, 90.0)
    ref_x1 = ref_x0 + rng.uniform(MIN_ROD_WIDTH, MAX_ROD_WIDTH)
    while True:
        rods = []
        for row in ROD_ROWS:
            x0 = rng.uniform(X_MIN, X_MAX - MIN_ROD_WIDTH)
            width = rng.uniform(MIN_ROD_WIDTH, MAX_ROD_WIDTH)
            rods.append({"x0": x0, "x1": x0 + width, "row": row})
        overlaps = [_overlap_len(r["x0"], r["x1"], ref_x0, ref_x1) for r in rods]
        ordered = sorted(overlaps, reverse=True)
        if ordered[0] - ordered[1] >= MIN_MARGIN_PX:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    rods = [
        {"color": colors[idx], "x0": r["x0"], "x1": r["x1"], "row": r["row"]}
        for idx, r in enumerate(rods)
    ]
    reference = {"color": REFERENCE_COLOR, "x0": ref_x0, "x1": ref_x1, "row": REFERENCE_ROW}
    return {"rods": rods + [reference], "width": WIDTH, "height": HEIGHT, "background": "light"}


def render(scene: dict) -> Image.Image:
    """Render I = R(z): four horizontal bars, one gray reference and three colored."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for rod in scene["rods"]:
        x0, x1 = rod["x0"], rod["x1"]
        y0 = rod["row"] - BAR_HEIGHT / 2.0
        draw.rectangle([x0, y0, x1, y0 + BAR_HEIGHT], fill=COLOR_RGB[rod["color"]])
    return img


def _best_overlap(rods: list[dict]) -> tuple[dict, float, float]:
    """Return (max-overlap colored rod, its overlap len, top-two margin)."""
    reference = next(rod for rod in rods if rod["color"] == REFERENCE_COLOR)
    keyed = []
    for rod in rods:
        if rod["color"] == REFERENCE_COLOR:
            continue
        ov = _overlap_len(rod["x0"], rod["x1"], reference["x0"], reference["x1"])
        keyed.append((rod, ov))
    keyed.sort(key=lambda item: item[1], reverse=True)
    top, second = keyed[0], keyed[1]
    return top[0], top[1], top[1] - second[1]


def analytic_gold(scene: dict) -> str:
    """Color of the rod with the longest shared horizontal overlap with the reference."""
    top, _ov, _margin = _best_overlap(scene["rods"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two overlap-length gap in pixels."""
    _top, _ov, margin_px = _best_overlap(scene["rods"])
    return float(margin_px)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two overlap gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each rod's color label, its x-span, and its row) is
    directly rendered as visible ink, so there is no hidden scene field that
    could change the gold without changing the raster. Swapping two rods' color
    labels changes which color is correct, and any global flip/rotation turns
    the raster into a distinct image.
    """
    return []
