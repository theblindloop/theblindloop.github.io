"""Independent pixel-only inverse arm for the interval_overlap_argmax world.

Receives ONLY the rasterized PNG (plus the world's fixed public palette, which
is the naming convention) and re-measures, for every colored horizontal bar, the
length of its shared x-span with the gray reference bar. It shares no geometry
with renderer.render() and never imports the renderer, prompts, generator,
verifier, or any analytical gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "interval_overlap_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
    "gray": (120, 120, 120),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry or the background.

    argmin leaves no unclassified antialias band; blends split at ~50% and are
    dominated by the background, keeping each rod mask clean.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(COLOR_RGB["gray"], dtype=np.float32)]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _columns(mask: np.ndarray) -> set[int]:
    """Set of x columns where any pixel of the given rod color appears."""
    cols = np.nonzero(mask.any(axis=0))[0]
    return set(int(c) for c in cols)


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the rod sharing the longest horizontal overlap with gray."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    gray_idx = len(PALETTE)  # index of gray in the refs stack
    gray_cols = _columns(labels == gray_idx)
    if not gray_cols:
        return PALETTE[0]
    best_color = PALETTE[0]
    best_overlap = -1.0
    for idx, color in enumerate(PALETTE):
        color_cols = _columns(labels == idx)
        overlap = len(color_cols & gray_cols)
        if overlap > best_overlap:
            best_overlap = overlap
            best_color = color
    return best_color
