"""Scene-free pixel arm for rotor_switch_relay.

``decision_from_image`` receives one RGB image and nothing else. It imports only
numpy and PIL, pulls in none of the world's other modules, and never sees a
latent scene or any analytic value. It re-finds the switch blocks, the tongue
side of each block, the number of released balls and the left-to-right order of
the coloured bins from the raster alone, then replays the releases, throwing
every tongue it uses.

Raises ValueError when the drawn apparatus is missing or malformed, so an erased
or corrupted board refuses instead of guessing.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "rotor_switch_relay-oracle-0.1.0"

_PIN = (90, 95, 105)
_BALL = (20, 40, 90)
_BINS = {
    "red": (215, 45, 45),
    "green": (35, 150, 70),
    "yellow": (240, 195, 25),
    "orange": (240, 125, 20),
    "purple": (145, 50, 185),
}
_ROWS = 4
_MIN_BLOB = 200
_MIN_BIN = 1500
_ROW_GAP = 40.0


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])


def _components(mask: np.ndarray, min_size: int) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[dict] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        q = deque([(sy, sx)])
        seen[sy, sx] = True
        px: list[int] = []
        py: list[int] = []
        while q:
            y, x = q.popleft()
            px.append(x)
            py.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(px) >= min_size:
            ax = np.asarray(px)
            out.append(
                {
                    "cx": float(ax.mean()),
                    "cy": float(np.mean(py)),
                    "x0": int(ax.min()),
                    "x1": int(ax.max()),
                    "xs": ax,
                    "size": len(px),
                }
            )
    return out


def _tongue_points_right(blob: dict) -> bool:
    """The tongue widens the box on its own side, so the block's bulk falls on
    the far side of the box centre."""
    mid = (blob["x0"] + blob["x1"]) / 2.0
    left = int((blob["xs"] < mid).sum())
    right = int((blob["xs"] > mid).sum())
    if left == right:
        raise ValueError("switch tongue side is not resolvable")
    return left > right


def _board(blobs: list[dict]) -> list[list[bool]]:
    blobs = sorted(blobs, key=lambda b: b["cy"])
    rows: list[list[dict]] = []
    for b in blobs:
        if rows and abs(b["cy"] - rows[-1][-1]["cy"]) <= _ROW_GAP:
            rows[-1].append(b)
        else:
            rows.append([b])
    if len(rows) != _ROWS or any(len(r) != i + 1 for i, r in enumerate(rows)):
        raise ValueError(f"switch rows are {[len(r) for r in rows]}, expected 1,2,3,4")
    return [[_tongue_points_right(b) for b in sorted(r, key=lambda b: b["cx"])] for r in rows]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    switches = _components(_mask(arr, _PIN), _MIN_BLOB)
    if len(switches) != _ROWS * (_ROWS + 1) // 2:
        raise ValueError(f"found {len(switches)} switch blocks, expected 10")
    state = _board(switches)

    n_balls = len(_components(_mask(arr, _BALL), _MIN_BLOB))
    if not 1 <= n_balls <= 12:
        raise ValueError(f"found {n_balls} released balls")

    bins: list[tuple[float, str]] = []
    for name, rgb in _BINS.items():
        comps = _components(_mask(arr, rgb), _MIN_BIN)
        if len(comps) != 1:
            raise ValueError(f"bin colour {name} appears {len(comps)} times")
        bins.append((comps[0]["cx"], name))
    bins.sort()
    order = [name for _, name in bins]

    slots: list[int] = []
    for _ in range(n_balls):
        pos = 0
        for r in range(_ROWS):
            right = state[r][pos]
            state[r][pos] = not right
            if right:
                pos += 1
        slots.append(pos)
    return f"{order[slots[-1]]}|{order[slots[0]]}"
