"""Scene sampling, rendering and analytic gold for rotor_switch_relay.

A small pin board carries ten switch blocks arranged in four rows of one, two,
three and four. Every switch has a short tongue sticking out of its left or its
right side. Balls are released one at a time from the single top switch; a ball
entering a switch leaves towards the side the tongue currently points, and the
switch then immediately throws its tongue to the other side. The drawn switches
are therefore not a fixed program: each ball rewrites the board for the balls
that follow it, so the same drawn switch acts differently on different balls.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rotor_switch_relay-renderer-0.1.0"

CANVAS = (860, 620)
ROWS = 4
SLOTS = ROWS + 1
PITCH = 118.0            # centre-to-centre horizontal pitch of switches and bins
ROW_H = 92.0             # centre-to-centre vertical pitch of switch rows
TOP_Y = 214.0
CX = 430.0
BODY = 15                # switch block half side
NUB_H = 6                # switch tongue half height
TRAY_Y = 74
TRAY_X0 = 132
TRAY_DX = 70
BALL_R = 16
BIN_TOP = 552
BIN_BOT = 602
BIN_HALF_W = 50

BG = (255, 255, 255)
PIN = (90, 95, 105)
BALL = (20, 40, 90)
GUIDE = (231, 231, 231)
TRAY_BAR = (208, 208, 208)

BIN_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (215, 45, 45),
    "green": (35, 150, 70),
    "yellow": (240, 195, 25),
    "orange": (240, 125, 20),
    "purple": (145, 50, 185),
}
COLOR_NAMES = ("red", "green", "yellow", "orange", "purple")


def slot_x(index: int, row: int) -> float:
    """Drawn x of switch/exit slot `index` of a row holding `row + 1` items."""
    return CX + (index - row / 2.0) * PITCH


def row_y(row: int) -> float:
    return TOP_Y + row * ROW_H


# --------------------------------------------------------------------------
# reading the latent scene back from drawn geometry (never from list order)
# --------------------------------------------------------------------------


def _read_board(scene: dict) -> list[list[dict]]:
    rows: list[list[dict]] = [[] for _ in range(ROWS)]
    for sw in scene["switches"]:
        r = min(range(ROWS), key=lambda i: abs(row_y(i) - sw["cy"]))
        rows[r].append(sw)
    for r in range(ROWS):
        if len(rows[r]) != r + 1:
            raise ValueError(f"row {r} recovered {len(rows[r])} switches")
        rows[r].sort(key=lambda s: s["cx"])
    return rows


def _read_bins(scene: dict) -> list[str]:
    return [b["color"] for b in sorted(scene["bins"], key=lambda b: b["cx"])]


def _replay(rows: list[list[dict]], n_balls: int, flip: bool = True) -> list[int]:
    """Slot reached by each ball in release order."""
    state = [[sw["dir"] for sw in row] for row in rows]
    out: list[int] = []
    for _ in range(n_balls):
        pos = 0
        for r in range(ROWS):
            d = state[r][pos]
            if flip:
                state[r][pos] = -d
            if d > 0:
                pos += 1
        out.append(pos)
    return out


def analytic_gold(scene: dict) -> str:
    rows = _read_board(scene)
    bins = _read_bins(scene)
    n = len(scene["balls"])
    slots = _replay(rows, n)
    return f"{bins[slots[-1]]}|{bins[slots[0]]}"


def margin(scene: dict) -> float:
    rows = _read_board(scene)
    terms = [float(sw["nub"]) for row in rows for sw in row]
    for row in rows:
        for a, b in zip(row, row[1:]):
            terms.append(abs(b["cx"] - a["cx"]) - 2.0 * BODY - a["nub"] - b["nub"])
    balls = sorted(b["cx"] for b in scene["balls"])
    for a, b in zip(balls, balls[1:]):
        terms.append(b - a - 2.0 * BALL_R)
    return round(min(terms), 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    out: list[tuple[str, dict]] = []
    sw = list(scene["switches"])
    out.append(("switch_bag_reversed", {**scene, "switches": list(reversed(sw))}))
    out.append(("switch_bag_rotated", {**scene, "switches": sw[3:] + sw[:3]}))
    out.append(("switch_bag_sorted_by_x", {**scene, "switches": sorted(sw, key=lambda s: s["cx"])}))
    out.append(("bin_bag_reversed", {**scene, "bins": list(reversed(scene["bins"]))}))
    out.append(("ball_bag_reversed", {**scene, "balls": list(reversed(scene["balls"]))}))
    return out


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------


def _build(rng: random.Random, mode: str) -> dict:
    nubs = [rng.choice([12, 14, 16, 18]) for _ in range(ROWS * (ROWS + 1) // 2)]
    if mode == "tight":
        nubs[rng.randrange(len(nubs))] = rng.choice([3, 4, 5])
    elif mode == "near":
        nubs[rng.randrange(len(nubs))] = rng.choice([7, 8, 9, 10])
    switches = []
    i = 0
    for r in range(ROWS):
        for k in range(r + 1):
            switches.append(
                {
                    "cx": slot_x(k, r),
                    "cy": row_y(r),
                    "dir": rng.choice([-1, 1]),
                    "nub": nubs[i],
                }
            )
            i += 1
    rng.shuffle(switches)
    n_balls = rng.choice([3, 4, 5, 6])
    balls = [{"cx": float(TRAY_X0 + j * TRAY_DX), "cy": float(TRAY_Y)} for j in range(n_balls)]
    order = list(COLOR_NAMES)
    rng.shuffle(order)
    bins = [{"cx": slot_x(i, ROWS), "color": order[i]} for i in range(SLOTS)]
    rng.shuffle(bins)
    return {
        "canvas": list(CANVAS),
        "rows": ROWS,
        "mode": mode,
        "switches": switches,
        "balls": balls,
        "bins": bins,
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % SLOTS
    mode = "tight" if seed % 10 == 1 else ("near" if seed % 10 in (4, 6) else "wide")
    for _ in range(6000):
        scene = _build(rng, mode)
        rows = _read_board(scene)
        n = len(scene["balls"])
        slots = _replay(rows, n)
        if slots[-1] != target:
            continue
        if slots[-1] == slots[0]:
            continue                      # reading only the drawn switches must fail
        if _replay(rows, n, flip=False)[-1] == slots[-1]:
            continue                      # the rewriting must be load bearing
        return scene
    raise RuntimeError(f"no scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    d = ImageDraw.Draw(img)

    for i in range(SLOTS):
        x = slot_x(i, ROWS)
        d.rectangle([x - 1, row_y(ROWS - 1) + 30, x + 1, BIN_TOP - 4], fill=GUIDE)
    d.rectangle([TRAY_X0 - 40, TRAY_Y + BALL_R + 8, TRAY_X0 + 5 * TRAY_DX + 40, TRAY_Y + BALL_R + 12], fill=TRAY_BAR)

    for b in scene["bins"]:
        x = b["cx"]
        d.rectangle([x - BIN_HALF_W, BIN_TOP, x + BIN_HALF_W, BIN_BOT], fill=BIN_COLORS[b["color"]])

    for sw in scene["switches"]:
        x, y, L = sw["cx"], sw["cy"], sw["nub"]
        d.rectangle([x - BODY, y - BODY, x + BODY, y + BODY], fill=PIN)
        if sw["dir"] > 0:
            d.rectangle([x + BODY, y - NUB_H, x + BODY + L, y + NUB_H], fill=PIN)
        else:
            d.rectangle([x - BODY - L, y - NUB_H, x - BODY, y + NUB_H], fill=PIN)

    for b in scene["balls"]:
        x, y = b["cx"], b["cy"]
        d.ellipse([x - BALL_R, y - BALL_R, x + BALL_R, y + BALL_R], fill=BALL)
    return img
