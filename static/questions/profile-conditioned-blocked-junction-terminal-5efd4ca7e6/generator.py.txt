"""Deterministic renderer and analytic world for a blocked-junction question."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "blocked_junction_terminal-renderer-0.1.0"
WIDTH = 720
HEIGHT = 640
BACKGROUND = (248, 248, 244)
INK = (30, 44, 67)
GREEN = (38, 160, 91)
RED = (211, 66, 61)
BLUE = (62, 112, 202)
MARKER = (151, 70, 169)


def _point(x: float, y: float, tx: float, ty: float) -> list[float]:
    return [round(360.0 + tx + x, 4), round(320.0 + ty + y, 4)]


def _polyline(points: Iterable[Iterable[float]]) -> list[list[float]]:
    return [[float(x), float(y)] for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, non-crossing two-branch tree."""

    seed = int(seed)
    rng = random.Random(seed)
    tx = rng.uniform(-34.0, 34.0)
    ty = rng.uniform(-22.0, 22.0)
    fork_x = rng.uniform(-88.0, -72.0)
    gate_x = rng.uniform(-18.0, 12.0)
    top_gate_y = -rng.uniform(102.0, 121.0)
    bottom_gate_y = rng.uniform(102.0, 121.0)
    top_terminal_x = rng.uniform(214.0, 246.0)
    bottom_terminal_x = rng.uniform(214.0, 246.0)
    top_terminal_y = top_gate_y - rng.uniform(14.0, 32.0)
    bottom_terminal_y = bottom_gate_y + rng.uniform(14.0, 32.0)
    top_spur_x = gate_x - rng.uniform(55.0, 74.0)
    bottom_spur_x = gate_x - rng.uniform(55.0, 74.0)
    top_spur_y = top_gate_y - rng.uniform(45.0, 64.0)
    bottom_spur_y = bottom_gate_y + rng.uniform(45.0, 64.0)
    stroke = rng.uniform(8.0, 11.0)
    marker_radius = rng.uniform(21.0, 25.0)
    endpoint_radius = rng.uniform(15.0, 18.0)

    def t(x: float, y: float) -> list[float]:
        return _point(x, y, tx, ty)

    source = t(-270.0, 0.0)
    fork = t(fork_x, 0.0)
    top_gate = t(gate_x, top_gate_y)
    bottom_gate = t(gate_x, bottom_gate_y)
    top_terminal = t(top_terminal_x, top_terminal_y)
    bottom_terminal = t(bottom_terminal_x, bottom_terminal_y)
    top_spur = t(top_spur_x, top_spur_y)
    bottom_spur = t(bottom_spur_x, bottom_spur_y)

    # Each branch has a marked degree-three junction: two route arms and one
    # short dead-end spur. The square/circle assignment and terminal colors are
    # independent deterministic permutations of the same visible geometry.
    edges = [
        _polyline([source, fork]),
        _polyline([fork, t(fork_x - 42.0, top_gate_y * 0.48), top_gate]),
        _polyline([top_gate, t(gate_x + 86.0, top_gate_y * 0.82), top_terminal]),
        _polyline([top_gate, top_spur]),
        _polyline([fork, t(fork_x - 42.0, bottom_gate_y * 0.48), bottom_gate]),
        _polyline([bottom_gate, t(gate_x + 86.0, bottom_gate_y * 0.82), bottom_terminal]),
        _polyline([bottom_gate, bottom_spur]),
    ]

    square_top = (seed // 2) % 2 == 0
    marker_shapes = ["square", "circle"] if square_top else ["circle", "square"]
    color_flip = seed % 2
    colors = [RED, BLUE] if color_flip == 0 else [BLUE, RED]
    terminals = [
        {"position": top_terminal, "color": "red" if colors[0] == RED else "blue"},
        {"position": bottom_terminal, "color": "red" if colors[1] == RED else "blue"},
    ]

    return {
        "canvas": [WIDTH, HEIGHT],
        "background": list(BACKGROUND),
        "palette": {
            "ink": list(INK),
            "green": list(GREEN),
            "red": list(RED),
            "blue": list(BLUE),
            "marker": list(MARKER),
        },
        "source": source,
        "fork": fork,
        "junctions": [{"position": top_gate}, {"position": bottom_gate}],
        "terminals": terminals,
        "edges": edges,
        "marker_shapes": marker_shapes,
        "stroke": round(stroke, 4),
        "marker_radius": round(marker_radius, 4),
        "endpoint_radius": round(endpoint_radius, 4),
        "spur_length": round(abs(top_spur_x - gate_x), 4),
        "translation": [round(tx, 4), round(ty, 4)],
        "branch_flip": int(square_top),
        "color_flip": color_flip,
    }


def _round_line(draw: ImageDraw.ImageDraw, points: list[list[float]], color, width: int) -> None:
    xy = [(int(round(x)), int(round(y))) for x, y in points]
    draw.line(xy, fill=color, width=width, joint="curve")
    radius = width / 2.0
    for x, y in xy:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def _marker(draw: ImageDraw.ImageDraw, center: list[float], shape: str, color, radius: float, width: int) -> None:
    x, y = center
    box = (x - radius, y - radius, x + radius, y + radius)
    if shape == "circle":
        draw.ellipse(box, outline=color, width=width)
    elif shape == "square":
        draw.rectangle(box, outline=color, width=width)
    else:
        raise ValueError(f"unknown marker shape {shape!r}")


def render(scene: dict) -> Image.Image:
    width, height = (int(scene["canvas"][0]), int(scene["canvas"][1]))
    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    def scaled(points: list[list[float]]) -> list[list[float]]:
        return [[x * ss, y * ss] for x, y in points]

    ink = tuple(scene["palette"]["ink"])
    line_width = max(3, int(round(float(scene["stroke"]) * ss)))
    for edge in scene["edges"]:
        _round_line(draw, scaled(edge), ink, line_width)

    endpoint_radius = float(scene["endpoint_radius"]) * ss
    sx, sy = (float(v) * ss for v in scene["source"])
    draw.ellipse((sx - endpoint_radius, sy - endpoint_radius, sx + endpoint_radius, sy + endpoint_radius), fill=tuple(scene["palette"]["green"]))
    for terminal in scene["terminals"]:
        x, y = (float(v) * ss for v in terminal["position"])
        color = tuple(scene["palette"][terminal["color"]])
        draw.ellipse((x - endpoint_radius, y - endpoint_radius, x + endpoint_radius, y + endpoint_radius), fill=color)

    marker_width = max(3, int(round(4.0 * ss)))
    marker_radius = float(scene["marker_radius"]) * ss
    marker_color = tuple(scene["palette"]["marker"])
    for junction, shape in zip(scene["junctions"], scene["marker_shapes"]):
        center = [float(v) * ss for v in junction["position"]]
        _marker(draw, center, shape, marker_color, marker_radius, marker_width)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Compute the surviving branch from visible label-bearing fields."""

    shapes = scene["marker_shapes"]
    terminals = scene["terminals"]
    square_index = shapes.index("square")
    return str(terminals[1 - square_index]["color"])


def margin(scene: dict) -> float:
    """Return the smallest useful pixel clearance for the counterfactual cut."""

    junctions = scene["junctions"]
    top_y = float(junctions[0]["position"][1])
    bottom_y = float(junctions[1]["position"][1])
    radius = float(scene["marker_radius"])
    fork_y = float(scene["fork"][1])
    return float(min(abs(top_y - bottom_y) - 2.0 * radius, abs(top_y - fork_y) - radius, abs(bottom_y - fork_y) - radius))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no unused latent aliases; every gold-bearing field is drawn."""

    return []
