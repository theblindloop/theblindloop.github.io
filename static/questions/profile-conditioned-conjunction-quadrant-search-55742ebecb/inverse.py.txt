"""Independent pixel-only inverse arm for conjunction_quadrant_search.

Receives only the final RGB image. It recovers the frame shape (circle vs
square) and central-bar orientation (horizontal vs vertical) of the reference
glyph in the top panel, then measures the same two attributes of the glyph in
each of the four quadrants and selects the single quadrant whose glyph matches
the reference on BOTH attributes. Imports nothing from the renderer, prompts,
generator, or verifier.
"""

from __future__ import annotations

import numpy as np

SIZE = 480
TOP_H = 122
MAIN_TOP = 126
MAIN_BOT = SIZE
MID_Y = (MAIN_TOP + MAIN_BOT) // 2

REF_CENTER = (240, 61)
REF_R = 28.0
QUAD_R = 46.0

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, (MAIN_TOP + MID_Y) // 2),
    "TR": (3 * SIZE // 4, (MAIN_TOP + MID_Y) // 2),
    "BL": (SIZE // 4, (MID_Y + MAIN_BOT) // 2),
    "BR": (3 * SIZE // 4, (MID_Y + MAIN_BOT) // 2),
}

BW_THRESH = 150
CORNER_FRAC = 0.12
CORNER_OCC_CHOOSE = 2
BAR_RADIUS_FRAC = 0.42
BAR_ASPECT = 1.35


def _classify(mask: np.ndarray, cx: int, cy: int, r: float) -> tuple[str, str]:
    ys, xs = np.nonzero(mask)

    ll = (xs >= cx - r) & (xs <= cx + r) & (ys >= cy - r) & (ys <= cy + r)
    if ll.sum() == 0:
        return "circle", "h"
    bx0, bx1 = int(xs[ll].min()), int(xs[ll].max())
    by0, by1 = int(ys[ll].min()), int(ys[ll].max())

    corner_w = max(2, int(CORNER_FRAC * (bx1 - bx0)))
    corner_h = max(2, int(CORNER_FRAC * (by1 - by0)))
    occupied = 0
    for cxs, cys in ((bx0, by0), (bx1, by0), (bx0, by1), (bx1, by1)):
        win = (
            (xs >= cxs - corner_w)
            & (xs <= cxs + corner_w)
            & (ys >= cys - corner_h)
            & (ys <= cys + corner_h)
        )
        if win.sum() > 0:
            occupied += 1
    shape = "square" if occupied >= CORNER_OCC_CHOOSE else "circle"

    rad = BAR_RADIUS_FRAC * r
    central = (xs - cx) ** 2 + (ys - cy) ** 2 <= rad * rad
    if central.sum() == 0:
        return shape, "h"
    bxs = xs[central]
    bys = ys[central]
    w = int(bxs.max()) - int(bxs.min()) + 1
    h = int(bys.max()) - int(bys.min()) + 1
    if w >= h * BAR_ASPECT:
        bar = "h"
    elif h >= w * BAR_ASPECT:
        bar = "v"
    else:
        bar = "h"
    return shape, bar


_CELL_X = {"TL": (0, 240), "TR": (240, SIZE), "BL": (0, 240), "BR": (240, SIZE)}
_CELL_Y = {"TL": (MAIN_TOP, MID_Y), "TR": (MAIN_TOP, MID_Y), "BL": (MID_Y, MAIN_BOT), "BR": (MID_Y, MAIN_BOT)}


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < BW_THRESH

    ref_mask = np.zeros_like(dark)
    ref_mask[:TOP_H, :] = dark[:TOP_H, :]
    ref_shape, ref_bar = _classify(ref_mask, REF_CENTER[0], REF_CENTER[1], REF_R)

    for sector in SECTORS:
        x0, x1 = _CELL_X[sector]
        y0, y1 = _CELL_Y[sector]
        mask = np.zeros_like(dark)
        mask[y0:y1, x0:x1] = dark[y0:y1, x0:x1]
        cx, cy = SECTOR_CENTERS[sector]
        quad = _classify(mask, cx, cy, QUAD_R)
        if quad[0] == ref_shape and quad[1] == ref_bar:
            return sector
    return "TL"
