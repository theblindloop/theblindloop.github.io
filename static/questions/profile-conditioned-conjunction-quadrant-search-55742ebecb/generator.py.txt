"""Two-feature conjunction visual-search world.

A reference glyph appears in the top panel: an outline frame (a circle or a
square) that contains one central dark bar (horizontal or vertical). Below, the
image is split into a 2x2 grid of quadrants, each holding one larger glyph with
the same two structural attributes: a frame shape (circle or square) and a
central bar orientation (horizontal or vertical).

The four quadrants carry the four combinations of {circle,square}x{H,V}
bijectively. Exactly one quadrant matches the reference on BOTH attributes; the
other three each differ on at least one attribute. The decision is the quadrant
whose glyph matches the reference shape AND bar orientation. All rendering is
deterministic and every attribute is visible in the raster.
"""

from __future__ import annotations

from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
POLY = (20, 20, 20)
DIV = (205, 205, 205)

TOP_H = 122
MAIN_TOP = 126
MAIN_BOT = SIZE
MID_Y = (MAIN_TOP + MAIN_BOT) // 2  # 303

REF_CENTER = (240, 61)
REF_R = 28.0
REF_T = 6
REF_BAR_FRAC = 0.68
REF_BAR_THICK = 0.40

QUAD_R = 46.0
QUAD_T = 9
QUAD_BAR_FRAC = 0.68
QUAD_BAR_THICK = 0.42

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, (MAIN_TOP + MID_Y) // 2),      # (120, 214)
    "TR": (3 * SIZE // 4, (MAIN_TOP + MID_Y) // 2),  # (360, 214)
    "BL": (SIZE // 4, (MID_Y + MAIN_BOT) // 2),      # (120, 391)
    "BR": (3 * SIZE // 4, (MID_Y + MAIN_BOT) // 2),  # (360, 391)
}

SHAPES = ("circle", "square")
BARS = ("h", "v")

_QUARANTINE_THRESH = 0.5


def _glyph(
    draw: ImageDraw.ImageDraw,
    cx: float,
    cy: float,
    r: float,
    t: int,
    shape: str,
    bar: str,
    bar_frac: float,
    bar_thick: float,
) -> None:
    x0, y0, x1, y1 = cx - r, cy - r, cx + r, cy + r
    if shape == "circle":
        draw.ellipse([x0, y0, x1, y1], fill=None, outline=POLY, width=t)
    else:
        draw.rectangle([x0, y0, x1, y1], fill=None, outline=POLY, width=t)
    half = bar_frac * r
    thick = bar_thick * r
    if bar == "h":
        draw.rectangle(
            [cx - half, cy - thick / 2, cx + half, cy + thick / 2], fill=POLY
        )
    else:
        draw.rectangle(
            [cx - thick / 2, cy - half, cx + thick / 2, cy + half], fill=POLY
        )


def sample_scene(seed: int) -> dict:
    import numpy as np

    rng = np.random.default_rng(seed)
    ref_shape = SHAPES[int(rng.integers(0, 2))]
    ref_bar = BARS[int(rng.integers(0, 2))]
    combos = [(s, b) for s in SHAPES for b in BARS]
    rng.shuffle(combos)
    quadrants = {}
    for idx, sector in enumerate(SECTORS):
        quadrants[sector] = {"shape": combos[idx][0], "bar": combos[idx][1]}
    return {
        "reference": {"shape": ref_shape, "bar": ref_bar},
        "quadrants": quadrants,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    draw.line([0, TOP_H, SIZE, TOP_H], fill=DIV, width=2)
    draw.line([SIZE // 2, MAIN_TOP, SIZE // 2, MAIN_BOT], fill=DIV, width=2)
    draw.line([0, MID_Y, SIZE, MID_Y], fill=DIV, width=2)

    ref = scene["reference"]
    _glyph(
        draw,
        REF_CENTER[0],
        REF_CENTER[1],
        REF_R,
        REF_T,
        ref["shape"],
        ref["bar"],
        REF_BAR_FRAC,
        REF_BAR_THICK,
    )

    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        quad = scene["quadrants"][sector]
        _glyph(
            draw,
            cx,
            cy,
            QUAD_R,
            QUAD_T,
            quad["shape"],
            quad["bar"],
            QUAD_BAR_FRAC,
            QUAD_BAR_THICK,
        )

    return img


def _matching_quadrant(scene: dict) -> str:
    ref = scene["reference"]
    for sector in SECTORS:
        quad = scene["quadrants"][sector]
        if quad["shape"] == ref["shape"] and quad["bar"] == ref["bar"]:
            return sector
    raise RuntimeError("no quadrant matches the reference")


def analytic_gold(scene: dict) -> str:
    return _matching_quadrant(scene)


def margin(scene: dict) -> float:
    ref = scene["reference"]
    matches = {}
    for sector in SECTORS:
        quad = scene["quadrants"][sector]
        matches[sector] = int(quad["shape"] == ref["shape"]) + int(
            quad["bar"] == ref["bar"]
        )
    winner = max(matches, key=lambda s: (matches[s], s))
    runner_up = max((m for s, m in matches.items() if s != winner), default=0)
    return float(matches[winner] - runner_up)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < _QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
