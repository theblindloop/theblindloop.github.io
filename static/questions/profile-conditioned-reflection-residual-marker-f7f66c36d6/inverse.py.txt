"""Independent pixel-only inverse arm."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            found.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red_mask = (array[:, :, 0] > 170) & (array[:, :, 1] < 100) & (array[:, :, 2] < 120)
    blue_mask = (array[:, :, 0] < 90) & (array[:, :, 1] < 150) & (array[:, :, 2] > 150)
    black_mask = (array.max(axis=2) < 80)
    gray_mask = (
        (array[:, :, 0] >= 105)
        & (array[:, :, 0] <= 185)
        & (np.abs(array[:, :, 0] - array[:, :, 1]) < 18)
        & (np.abs(array[:, :, 1] - array[:, :, 2]) < 18)
    )

    reds = _components(red_mask)
    blues = _components(blue_mask)
    blacks = _components(black_mask)
    grays = _components(gray_mask, minimum=200)
    if len(reds) != 1 or len(blues) != 1 or not blacks or len(grays) != 1:
        raise ValueError("image does not contain the required axis and dot colors")
    axis_x = grays[0][0]
    points = [(x, y, "red") for x, y, _ in reds]
    points += [(x, y, "blue") for x, y, _ in blues]
    points += [(x, y, "black") for x, y, _ in blacks]

    scores: dict[str, float] = {}
    for x, y, color in points[:2]:
        target_x = 2.0 * axis_x - x
        scores[color] = min(
            math.hypot(px - target_x, py - y)
            for px, py, _ in points
            if not (px == x and py == y)
        )
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 12.0:
        raise ValueError("ambiguous reflected residual")
    return ordered[0][0]
