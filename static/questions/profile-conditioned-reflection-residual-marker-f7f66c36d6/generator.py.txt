"""Deterministic renderer for a nearly mirror-paired dot constellation."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 512, 384
BACKGROUND = (248, 246, 240)
INK = (25, 29, 35)
AXIS = (145, 150, 158)
RED = (220, 55, 65)
BLUE = (42, 105, 210)
RADIUS = 9


def _residual_scores(scene: dict) -> dict[str, float]:
    axis_x = float(scene["axis_x"])
    dots = scene["dots"]
    points = [(float(dot[0]), float(dot[1]), str(dot[2])) for dot in dots]
    scores: dict[str, float] = {}
    for x, y, color in points:
        if color not in {"red", "blue"}:
            continue
        target = (2.0 * axis_x - x, y)
        distances = [
            math.hypot(px - target[0], py - target[1])
            for px, py, _ in points
            if not (px == x and py == y)
        ]
        scores[color] = min(distances)
    return scores


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = "red" if int(seed) % 2 == 0 else "blue"
    matched_color = "blue" if answer == "red" else "red"
    axis_x = rng.randint(238, 274)
    pair_count = 4 + rng.randrange(2)

    # Rejection sampling produces irregular paired constellations without a
    # reusable outline or fixed angular template.
    left: list[tuple[int, int]] = []
    attempts = 0
    while len(left) < pair_count and attempts < 2000:
        attempts += 1
        candidate = (axis_x - rng.randint(48, 148), rng.randint(48, HEIGHT - 49))
        if all(math.hypot(candidate[0] - x, candidate[1] - y) >= 42 for x, y in left):
            left.append(candidate)
    if len(left) != pair_count:
        raise RuntimeError("could not place separated mirror pairs")

    colored_pair = rng.randrange(pair_count)
    colored_side = rng.randrange(2)
    dots: list[list[object]] = []
    for index, (x, y) in enumerate(left):
        mirror_x = 2 * axis_x - x
        left_color = matched_color if index == colored_pair and colored_side == 0 else "black"
        right_color = matched_color if index == colored_pair and colored_side == 1 else "black"
        dots.append([x, y, left_color])
        dots.append([mirror_x, y, right_color])

    residual_side = rng.randrange(2)
    residual = None
    for _ in range(2000):
        dx = rng.randint(52, 145)
        rx = axis_x - dx if residual_side == 0 else axis_x + dx
        ry = rng.randint(48, HEIGHT - 49)
        target = (2 * axis_x - rx, ry)
        if min(math.hypot(rx - x, ry - y) for x, y, _ in dots) < 38:
            continue
        if min(math.hypot(target[0] - x, target[1] - y) for x, y, _ in dots) < 34:
            continue
        residual = [rx, ry, answer]
        break
    if residual is None:
        raise RuntimeError("could not place an unambiguous residual")
    dots.append(residual)
    rng.shuffle(dots)
    return {"axis_x": axis_x, "dots": dots}


def render(scene: dict) -> Image.Image:
    axis_x = int(scene["axis_x"])
    dots = scene["dots"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.line((axis_x, 25, axis_x, HEIGHT - 26), fill=AXIS, width=5)
    palette = {"black": INK, "red": RED, "blue": BLUE}
    for x, y, color in dots:
        x, y = int(x), int(y)
        draw.ellipse(
            (x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS),
            fill=palette[str(color)],
            outline=BACKGROUND,
            width=1,
        )
    return image


def analytic_gold(scene: dict) -> str:
    scores = _residual_scores(scene)
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    scores = _residual_scores(scene)
    values = sorted(scores.values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
