"""Pixel-only inverse arm for the broken-ring completion question.

Receives a rendered PNG and the public question contract only: dark arcs are
fragments of one circle, four saturated dots are named red, blue, green and
orange, and exactly one dot lies on the completed circle.  The arm rebuilds the
global contour by fitting a circle to every arc pixel at once, then reports the
dot whose distance to that circle's centre is closest to the fitted radius.

It imports no renderer, prompt, generator, verifier or gold code and never sees
latent scene data.
"""

from __future__ import annotations

import colorsys
import math

from PIL import Image

# Hue windows for the four contract colour names, in degrees.
HUE_WINDOWS: tuple[tuple[str, float, float], ...] = (
    ("red", 340.0, 20.0),
    ("orange", 20.0, 75.0),
    ("green", 90.0, 185.0),
    ("blue", 190.0, 265.0),
)


def _classify(pixels, width: int, height: int):
    """Split the raster into dark arc pixels and saturated marker pixels."""
    arc_points: list[tuple[float, float]] = []
    marker_points: dict[str, list[tuple[float, float]]] = {}
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y]
            high = max(r, g, b)
            low = min(r, g, b)
            value = high / 255.0
            sat = 0.0 if high == 0 else (high - low) / high
            if value < 0.45 and sat < 0.45:
                arc_points.append((x + 0.5, y + 0.5))
            elif sat > 0.45 and value > 0.5:
                hue = colorsys.rgb_to_hsv(r / 255.0, g / 255.0, b / 255.0)[0] * 360.0
                for name, lo, hi in HUE_WINDOWS:
                    inside = (hue >= lo or hue < hi) if lo > hi else (lo <= hue < hi)
                    if inside:
                        marker_points.setdefault(name, []).append((x + 0.5, y + 0.5))
                        break
    return arc_points, marker_points


def _fit_circle(points: list[tuple[float, float]]) -> tuple[float, float, float]:
    """Algebraic least-squares circle fit (Kasa) over the arc band."""
    n = float(len(points))
    mx = sum(p[0] for p in points) / n
    my = sum(p[1] for p in points) / n
    sxx = syy = sxy = sxz = syz = 0.0
    for px, py in points:
        u, v = px - mx, py - my
        z = u * u + v * v
        sxx += u * u
        syy += v * v
        sxy += u * v
        sxz += u * z
        syz += v * z
    det = 2.0 * (sxx * syy - sxy * sxy)
    if abs(det) < 1e-9:
        raise ValueError("degenerate arc geometry")
    ux = (syy * sxz - sxy * syz) / det
    uy = (sxx * syz - sxy * sxz) / det
    cx, cy = mx + ux, my + uy
    radius = sum(math.hypot(px - cx, py - cy) for px, py in points) / n
    return cx, cy, radius


def measure(image: Image.Image) -> dict:
    """Diagnostics: fitted circle, per-colour radial deviation and the margin."""
    rgb = image.convert("RGB")
    width, height = rgb.size
    arc_points, marker_points = _classify(rgb.load(), width, height)
    if len(arc_points) < 200:
        raise ValueError("no arc evidence found")
    if len(marker_points) != 4:
        raise ValueError(f"expected four coloured dots, found {sorted(marker_points)}")

    cx, cy, radius = _fit_circle(arc_points)
    deviations: dict[str, float] = {}
    for name, points in marker_points.items():
        if len(points) < 30:
            raise ValueError(f"dot {name} is too small to localise")
        px = sum(p[0] for p in points) / len(points)
        py = sum(p[1] for p in points) / len(points)
        deviations[name] = abs(math.hypot(px - cx, py - cy) - radius)

    ranked = sorted(deviations.items(), key=lambda item: item[1])
    return {
        "center": (cx, cy),
        "radius": radius,
        "deviations": deviations,
        "decision": ranked[0][0],
        "margin": ranked[1][1] - ranked[0][1],
    }


def decision_from_image(image: Image.Image) -> str:
    """Return the colour name of the dot lying on the completed circle."""
    return str(measure(image)["decision"])
