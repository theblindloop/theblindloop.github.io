"""Independent image-only inverse arm for the two visible iris rings."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public visual palette is duplicated deliberately; no renderer state is imported.
_PALETTE = {
    "blue": (28, 104, 205),
    "green": (31, 145, 82),
    "hazel": (163, 91, 30),
    "violet": (126, 62, 180),
}
_COLOR_THRESHOLD = 45.0
_MIN_AREA = 520
_MAX_AREA = 1250


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    work = mask.copy()
    found: list[tuple[int, float, float]] = []
    height, width = work.shape
    for sy, sx in np.argwhere(work):
        sy, sx = int(sy), int(sx)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(sy, sx)])
        ys: list[int] = []
        xs: list[int] = []
        while queue:
            y, x = queue.popleft()
            ys.append(y)
            xs.append(x)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        area = len(xs)
        if area >= _MIN_AREA:
            found.append((area, float(np.mean(xs)), float(np.mean(ys))))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    recovered: list[tuple[float, str, int, float]] = []
    for name, rgb in _PALETTE.items():
        distance = np.linalg.norm(arr - np.asarray(rgb, dtype=np.float32), axis=2)
        for area, cx, cy in _components(distance < _COLOR_THRESHOLD):
            if not _MIN_AREA <= area <= _MAX_AREA:
                raise ValueError("chromatic component has an ambiguous size")
            recovered.append((cx, name, area, cy))
    if len(recovered) != 2:
        raise ValueError(f"expected exactly two visible iris components, found {len(recovered)}")
    recovered.sort()
    left, right = recovered
    if not 90.0 <= right[0] - left[0] <= 165.0:
        raise ValueError("iris separation is outside the declared compact face layout")
    if abs(right[3] - left[3]) > 8.0:
        raise ValueError("iris components are not visibly aligned")
    if min(left[2], right[2]) / max(left[2], right[2]) < 0.80:
        raise ValueError("iris component sizes are too imbalanced")
    return "yes" if left[1] == right[1] else "no"
