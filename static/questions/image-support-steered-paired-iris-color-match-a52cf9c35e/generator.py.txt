"""Deterministic Pillow world for a visible two-iris color comparison."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SCALE = 4
BACKGROUND = (248, 247, 242)
IRIS_COLORS = {
    "blue": (28, 104, 205),
    "green": (31, 145, 82),
    "hazel": (163, 91, 30),
    "violet": (126, 62, 180),
}
SKIN_TONES = {
    "light": (238, 201, 170),
    "medium": (198, 143, 104),
    "deep": (119, 76, 58),
}


def sample_scene(seed: int) -> dict:
    """Pair adjacent seeds by shared group; parity selects control/counterfactual."""
    group = int(seed) // 2
    same = int(seed) % 2 == 0
    rng = random.Random(group ^ 0x51A7)
    names = tuple(IRIS_COLORS)
    left_index = group % len(names)
    offset = 1 + (group // len(names)) % (len(names) - 1)
    return {
        "left_cell": rng.choice((1, 2, 3, 4)),
        "row_cell": rng.choice((2, 3, 4, 5)),
        "iris_radius": rng.choice((17, 18, 19)),
        "left_color": names[left_index],
        "right_color": names[left_index] if same else names[(left_index + offset) % len(names)],
        "skin_tone": rng.choice(tuple(SKIN_TONES)),
        "mouth_curve": rng.choice((-1, 0, 1)),
    }


def _ellipse(draw: ImageDraw.ImageDraw, box: tuple[float, float, float, float], **kwargs) -> None:
    draw.ellipse(tuple(round(v * SCALE) for v in box), **kwargs)


def _line(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], **kwargs) -> None:
    draw.line([(round(x * SCALE), round(y * SCALE)) for x, y in points], **kwargs)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    lx = int(scene["left_cell"]) * 64 + 32
    rx = lx + 128
    cy = int(scene["row_cell"]) * 64 + 32
    radius = int(scene["iris_radius"])
    skin = SKIN_TONES[str(scene["skin_tone"])]
    ink = (47, 43, 42)
    white = (252, 252, 249)

    # The neutral host establishes a familiar face but is nondiagnostic.
    _ellipse(draw, (lx - 52, cy - 76, rx + 52, cy + 76), fill=skin, outline=ink, width=4 * SCALE)
    for ex in (lx, rx):
        _ellipse(draw, (ex - 38, cy - 28, ex + 38, cy + 28), fill=white, outline=ink, width=4 * SCALE)
    _line(draw, [(lx + 64, cy - 4), (lx + 58, cy + 20), (lx + 70, cy + 20)], fill=ink, width=3 * SCALE, joint="curve")

    mouth = int(scene["mouth_curve"])
    my = cy + 45
    if mouth == 0:
        _line(draw, [(lx + 43, my), (lx + 85, my)], fill=ink, width=4 * SCALE)
    else:
        pts = []
        for step in range(25):
            x = lx + 40 + step * 2
            u = (step - 12) / 12
            y = my + mouth * (7 * (u * u - 1))
            pts.append((x, y))
        _line(draw, pts, fill=ink, width=4 * SCALE)

    for ex, name in ((lx, str(scene["left_color"])), (rx, str(scene["right_color"]))):
        _ellipse(draw, (ex - radius, cy - radius, ex + radius, cy + radius), fill=IRIS_COLORS[name])
        pupil = max(5, radius // 3)
        _ellipse(draw, (ex - pupil, cy - pupil, ex + pupil, cy + pupil), fill=(20, 19, 21))

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["left_color"] == scene["right_color"] else "no"


def margin(scene: dict) -> float:
    # Minimum palette separation and component-size headroom, both pixel-observable.
    left = IRIS_COLORS[str(scene["left_color"])]
    right = IRIS_COLORS[str(scene["right_color"])]
    used = (left, right)
    palette_gap = min(
        math.dist(color, other)
        for color in used
        for other in IRIS_COLORS.values()
        if color != other
    )
    return round(min(float(scene["iris_radius"]) - 12.0, palette_gap / 20.0), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
