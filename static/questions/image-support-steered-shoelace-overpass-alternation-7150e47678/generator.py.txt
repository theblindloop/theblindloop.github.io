"""Deterministic renderer for a visibly braided, single U-joined shoelace."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANONICAL = ["/", "\\", "/", "\\", "/"]
PALETTES = [(35, 105, 215), (25, 125, 205), (62, 92, 210), (20, 112, 190)]


def sample_scene(seed: int) -> dict:
    # Adjacent seeds are a matched control/counterfactual pair: geometry, palette,
    # and total ink are identical, while one overpass orientation changes.
    paired = random.Random((int(seed) // 2) ^ 0x51A7C0DE)
    scene = {
        "width": 512,
        "height": 512,
        "background": [248, 247, 242],
        "lace_color": list(PALETTES[paired.randrange(len(PALETTES))]),
        "center_x": 256 + paired.randint(-10, 10),
        "top_y": 150 + paired.randint(-5, 5),
        "spacing": 43 + paired.randint(-1, 1),
        "amplitude": 53 + paired.randint(-3, 3),
        "stroke": 13 + 2 * paired.randrange(2),
        "over_pattern": list(CANONICAL),
    }
    if int(seed) % 2:
        flip = paired.randrange(5)
        scene["over_pattern"][flip] = "\\" if scene["over_pattern"][flip] == "/" else "/"
    return scene


def _strand_x(scene: dict, strand: int, y: float) -> float:
    phase = math.pi * (y - float(scene["top_y"])) / float(scene["spacing"])
    sign = 1.0 if strand == 0 else -1.0
    return float(scene["center_x"]) + sign * float(scene["amplitude"]) * math.sin(phase)


def _points(scene: dict, strand: int, y0: float, y1: float, scale: int) -> list[tuple[int, int]]:
    points = []
    steps = max(2, int((y1 - y0) * 2))
    for index in range(steps + 1):
        y = y0 + (y1 - y0) * index / steps
        points.append((round(_strand_x(scene, strand, y) * scale), round(y * scale)))
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    blue = tuple(int(v) for v in scene["lace_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    top = float(scene["top_y"])
    spacing = float(scene["spacing"])
    stroke = int(scene["stroke"]) * scale
    y0, y1 = top - 0.58 * spacing, top + 4.72 * spacing

    for strand in (0, 1):
        draw.line(_points(scene, strand, y0, y1, scale), fill=blue, width=stroke, joint="curve")

    # A visible U join makes the two upward ends one physical shoelace.
    bottom_y = top + 4.72 * spacing
    left = min(_strand_x(scene, 0, bottom_y), _strand_x(scene, 1, bottom_y))
    right = max(_strand_x(scene, 0, bottom_y), _strand_x(scene, 1, bottom_y))
    box = [
        round(left * scale),
        round((bottom_y - 0.34 * spacing) * scale),
        round(right * scale),
        round((bottom_y + 0.34 * spacing) * scale),
    ]
    draw.arc(box, start=0, end=180, fill=blue, width=stroke)

    # At each crossing, erase both strands locally, then redraw only the strand
    # named as the top diagonal. The broad interruption is the visible boundary.
    gap_radius = (int(scene["stroke"]) * 1.45 + 5.0) * scale
    redraw_half = 0.48 * spacing
    for index, orientation in enumerate(scene["over_pattern"]):
        cy = top + index * spacing
        cx = float(scene["center_x"])
        draw.ellipse(
            [cx * scale - gap_radius, cy * scale - gap_radius,
             cx * scale + gap_radius, cy * scale + gap_radius],
            fill=background,
        )
        # Strand 0 has dx/dy sign (-1)^index; positive is a screen "\\".
        strand0_orientation = "\\" if index % 2 == 0 else "/"
        chosen = 0 if orientation == strand0_orientation else 1
        draw.line(
            _points(scene, chosen, cy - redraw_half, cy + redraw_half, scale),
            fill=blue,
            width=stroke,
            joint="curve",
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if list(scene["over_pattern"]) == CANONICAL else "no"


def margin(scene: dict) -> float:
    return float(scene["stroke"]) * 1.45 + 5.0


def is_quarantined(scene: dict) -> bool:
    return bool(float(scene["spacing"]) < 38 or float(scene["stroke"]) < 11)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
