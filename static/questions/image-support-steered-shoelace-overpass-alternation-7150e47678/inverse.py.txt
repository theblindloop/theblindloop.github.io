"""Independent PNG-only oracle for ordered shoelace overpasses."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _blue_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    return (blue > red + 65) & (blue > green + 38) & (blue > 145)


def _component_sizes(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny, nx in ((cy-1, cx), (cy+1, cx), (cy, cx-1), (cy, cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _clusters(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    groups = []
    start = previous = int(indices[0])
    for raw in indices[1:]:
        value = int(raw)
        if value > previous + 1:
            groups.append((start, previous))
            start = value
        previous = value
    groups.append((start, previous))
    return groups


def decision_from_image(image: Image.Image) -> str:
    mask = _blue_mask(image)
    points = np.argwhere(mask)
    if len(points) < 5000:
        raise ValueError("insufficient visible lace ink")
    y_min, x_min = points.min(axis=0)
    y_max, x_max = points.max(axis=0)
    box_h, box_w = int(y_max-y_min+1), int(x_max-x_min+1)
    if not (185 <= box_h <= 285 and 90 <= box_w <= 155):
        raise ValueError("lace extent is incomplete or malformed")

    # Destructive erasures of an interior route segment create substantial extra
    # fragments. Tiny antialias islands are ignored.
    substantial = [size for size in _component_sizes(mask) if size >= 80]
    if len(substantial) > 8:
        raise ValueError("lace continuity is too fragmented")

    center_x = int(round(np.median(points[:, 1])))
    half_strip = max(5, box_w // 18)
    central = mask[:, max(0, center_x-half_strip):center_x+half_strip+1].sum(axis=1)
    active = central >= max(4, half_strip // 2)
    groups = [g for g in _clusters(active) if g[1]-g[0]+1 >= 4]
    centers = [int(round((a+b)/2)) for a, b in groups]

    # Select the evenly spaced run of five crossing clusters; the bottom U may
    # contribute an additional later cluster.
    chosen = None
    for start in range(max(1, len(centers)-4)):
        run = centers[start:start+5]
        if len(run) < 5:
            continue
        gaps = np.diff(run)
        if 37 <= float(np.median(gaps)) <= 49 and float(gaps.max()-gaps.min()) <= 7:
            chosen = run
            break
    if chosen is None:
        raise ValueError(f"could not recover five intact crossings: {centers}")

    pattern: list[str] = []
    for cy in chosen:
        radius = max(15, int(round(np.median(np.diff(chosen)))) // 2)
        y0, y1 = max(0, cy-radius), min(mask.shape[0], cy+radius+1)
        x0, x1 = max(0, center_x-radius*3), min(mask.shape[1], center_x+radius*3+1)
        yy, xx = np.mgrid[y0:y1, x0:x1]
        dy, dx = yy-cy, xx-center_x
        local = mask[y0:y1, x0:x1]
        radial_y = np.abs(dy)
        # Only the erased crossing core is diagnostic; outer portions contain
        # both strands. Quadrant sign distinguishes / from \\ without coordinates.
        core = local & (radial_y >= 5) & (radial_y <= 18) & (np.abs(dx) <= radius*2)
        slash = int(np.count_nonzero(core & (dx*dy < 0)))
        backslash = int(np.count_nonzero(core & (dx*dy > 0)))
        total = slash + backslash
        if total < 65 or abs(slash-backslash) < 45:
            raise ValueError("ambiguous crossing continuity")
        pattern.append("/" if slash > backslash else "\\")
    return "yes" if pattern == ["/", "\\", "/", "\\", "/"] else "no"
