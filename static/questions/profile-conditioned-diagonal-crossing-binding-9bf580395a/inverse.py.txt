"""Independent pixel-only inverse for the diagonal-crossing question."""

from __future__ import annotations

import numpy as np
from PIL import Image


PANEL_RECTS = ((38, 112, 454, 500), (506, 112, 922, 500))
PLOT_RECTS = ((92, 166, 430, 468), (560, 166, 898, 468))
LABELS = ("A", "B", "C")
TAG_COLORS = {"P": np.array((35, 89, 176), dtype=np.float32), "Q": np.array((187, 78, 48), dtype=np.float32)}
LEGEND_SWATCH_RECTS = ((270, 48, 294, 72), (420, 48, 444, 72), (570, 48, 594, 72))


def _badge_scores(pixels: np.ndarray, panel: tuple[int, int, int, int]) -> dict[str, float]:
    x0, y0, _, _ = panel
    badge = pixels[y0 + 14 : y0 + 44, x0 + 14 : x0 + 50].astype(np.float32)
    return {
        name: float(np.mean(np.linalg.norm(badge - color, axis=2)))
        for name, color in TAG_COLORS.items()
    }


def _panel_plots(pixels: np.ndarray) -> dict[str, tuple[int, int, int, int]]:
    result: dict[str, tuple[int, int, int, int]] = {}
    for panel, plot in zip(PANEL_RECTS, PLOT_RECTS):
        scores = _badge_scores(pixels, panel)
        name = min(scores, key=scores.get)
        if name in result:
            raise ValueError("panel labels are not uniquely recoverable")
        result[name] = plot
    if set(result) != {"P", "Q"}:
        raise ValueError("both visible panel tags are required")
    return result


def _legend_colors(pixels: np.ndarray) -> list[np.ndarray]:
    colors = []
    for x0, y0, x1, y1 in LEGEND_SWATCH_RECTS:
        patch = pixels[y0 + 5 : y1 - 5, x0 + 5 : x1 - 5].reshape(-1, 3)
        colors.append(np.median(patch.astype(np.float32), axis=0))
    return colors


def _marker_centroid(
    pixels: np.ndarray,
    plot: tuple[int, int, int, int],
    color: np.ndarray,
) -> tuple[float, float]:
    x0, y0, x1, y1 = plot
    crop = pixels[y0 : y1 + 1, x0 : x1 + 1].astype(np.float32)
    distance = np.linalg.norm(crop - color, axis=2)
    ys, xs = np.nonzero(distance <= 42.0)
    if len(xs) < 80:
        raise ValueError("colored marker is not recoverable")
    return float(x0 + np.mean(xs)), float(y0 + np.mean(ys))


def _signed_side(point: tuple[float, float], plot: tuple[int, int, int, int]) -> float:
    x0, y0, x1, y1 = plot
    px, py = point
    normalized_x = (px - x0) / (x1 - x0)
    normalized_y = 1.0 - (py - y0) / (y1 - y0)
    return normalized_y - normalized_x


def decision_from_image(image: Image.Image) -> str:
    """Recover the unique label using only the final RGB raster."""

    pixels = np.asarray(image.convert("RGB"))
    plots = _panel_plots(pixels)
    colors = _legend_colors(pixels)
    reversals = []
    for label, color in zip(LABELS, colors):
        p_side = _signed_side(_marker_centroid(pixels, plots["P"], color), plots["P"])
        q_side = _signed_side(_marker_centroid(pixels, plots["Q"], color), plots["Q"])
        if p_side < -0.045 and q_side > 0.045:
            reversals.append(label)
    if len(reversals) != 1:
        raise ValueError("image does not expose one clear below-to-above reversal")
    return reversals[0]
