"""Deterministic paired-panel renderer and analytic decision."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont


CANVAS_SIZE = (960, 560)
PANEL_RECTS = ((38, 112, 454, 500), (506, 112, 922, 500))
PLOT_RECTS = ((92, 166, 430, 468), (560, 166, 898, 468))
LABELS = ("A", "B", "C")
TAG_COLORS = {"P": (35, 89, 176), "Q": (187, 78, 48)}
PALETTE = (
    (211, 58, 66),
    (41, 107, 204),
    (37, 155, 110),
    (146, 82, 197),
    (224, 132, 33),
    (205, 66, 143),
)
LEGEND_SWATCH_RECTS = ((270, 48, 294, 72), (420, 48, 444, 72), (570, 48, 594, 72))
QUARANTINE_CLEARANCE = 0.050
ORACLE_CLEARANCE = 0.045


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _sample_x(rng: random.Random) -> list[float]:
    values = [base + rng.uniform(-0.022, 0.022) for base in (0.27, 0.50, 0.73)]
    rng.shuffle(values)
    return [round(value, 6) for value in values]


def _side_value(x: float, above: bool, rng: random.Random, near: bool) -> float:
    low, high = (0.058, 0.086) if near else (0.130, 0.190)
    clearance = rng.uniform(low, high)
    value = x + clearance if above else x - clearance
    return round(min(0.91, max(0.09, value)), 6)


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene with exactly one directed status reversal."""

    rng = random.Random(int(seed))
    winner = int(seed) % 3
    near = int(seed) % 13 in (0, 1)
    p_x = _sample_x(rng)
    q_x = _sample_x(rng)
    p_y: list[float] = []
    q_y: list[float] = []
    for index in range(3):
        if index == winner:
            p_y.append(_side_value(p_x[index], False, rng, near))
            q_y.append(_side_value(q_x[index], True, rng, near))
            continue
        mode = rng.randrange(3)
        if mode == 0:
            p_y.append(_side_value(p_x[index], False, rng, near))
            q_y.append(_side_value(q_x[index], False, rng, near))
        elif mode == 1:
            p_y.append(_side_value(p_x[index], True, rng, near))
            q_y.append(_side_value(q_x[index], True, rng, near))
        else:
            p_y.append(_side_value(p_x[index], True, rng, near))
            q_y.append(_side_value(q_x[index], False, rng, near))
    colors = [list(color) for color in rng.sample(PALETTE, 3)]
    panel_order = ["P", "Q"]
    rng.shuffle(panel_order)
    return {
        "p_x": p_x,
        "p_y": p_y,
        "q_x": q_x,
        "q_y": q_y,
        "colors": colors,
        "marker_radius": rng.randint(10, 13),
        "panel_order": panel_order,
    }


def _reversal_indices(scene: dict) -> list[int]:
    p_x = scene["p_x"]
    p_y = scene["p_y"]
    q_x = scene["q_x"]
    q_y = scene["q_y"]
    return [
        index
        for index in range(3)
        if p_y[index] < p_x[index] and q_y[index] > q_x[index]
    ]


def analytic_gold(scene: dict) -> str:
    indices = _reversal_indices(scene)
    if len(indices) != 1:
        raise ValueError("scene is ambiguous: expected one below-to-above reversal")
    return LABELS[indices[0]]


def _transition_scores(scene: dict) -> list[float]:
    scores = []
    for p_x, p_y, q_x, q_y in zip(
        scene["p_x"], scene["p_y"], scene["q_x"], scene["q_y"]
    ):
        scores.append(min(p_x - p_y, q_y - q_x))
    return scores


def margin(scene: dict) -> float:
    scores = sorted(_transition_scores(scene), reverse=True)
    clearance = min(
        abs(x - y)
        for x, y in zip(scene["p_x"], scene["p_y"])
    )
    clearance = min(
        clearance,
        min(abs(x - y) for x, y in zip(scene["q_x"], scene["q_y"])),
    )
    return float(min(scores[0] - scores[1], clearance))


def is_quarantined(scene: dict) -> bool:
    return len(_reversal_indices(scene)) != 1 or margin(scene) < QUARANTINE_CLEARANCE


def _point_xy(plot: tuple[int, int, int, int], x_value: float, y_value: float) -> tuple[int, int]:
    x0, y0, x1, y1 = plot
    return round(x0 + x_value * (x1 - x0)), round(y1 - y_value * (y1 - y0))


def _draw_diagonal(draw: ImageDraw.ImageDraw, plot: tuple[int, int, int, int]) -> None:
    x0, y0, x1, y1 = plot
    for index in range(0, 32, 2):
        start = index / 32.0
        end = min(1.0, (index + 1) / 32.0)
        draw.line(
            (
                x0 + (x1 - x0) * start,
                y1 - (y1 - y0) * start,
                x0 + (x1 - x0) * end,
                y1 - (y1 - y0) * end,
            ),
            fill=(45, 52, 66),
            width=3,
        )


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel: tuple[int, int, int, int],
    plot: tuple[int, int, int, int],
    name: str,
    xs: list[float],
    ys: list[float],
    colors: list[list[int]],
    radius: int,
    font: ImageFont.ImageFont,
) -> None:
    x0, y0, x1, y1 = panel
    draw.rounded_rectangle(panel, radius=12, fill=(246, 248, 252), outline=(178, 188, 202), width=2)
    draw.rounded_rectangle((x0 + 14, y0 + 13, x0 + 50, y0 + 45), radius=6, fill=TAG_COLORS[name])
    draw.text((x0 + 27, y0 + 21), name, fill=(255, 255, 255), anchor="mm", font=font)
    draw.text((x0 + 63, y0 + 21), f"Panel {name}", fill=(27, 35, 48), font=font)
    px0, py0, px1, py1 = plot
    draw.rectangle(plot, fill=(255, 255, 255), outline=(83, 94, 110), width=2)
    for fraction in (0.25, 0.50, 0.75):
        gx = round(px0 + fraction * (px1 - px0))
        gy = round(py1 - fraction * (py1 - py0))
        draw.line((gx, py0, gx, py1), fill=(230, 234, 240), width=1)
        draw.line((px0, gy, px1, gy), fill=(230, 234, 240), width=1)
    _draw_diagonal(draw, plot)
    draw.line((px0, py1, px1, py1), fill=(70, 80, 96), width=2)
    draw.line((px0, py0, px0, py1), fill=(70, 80, 96), width=2)
    draw.text((px0 - 3, py1 + 9), "0", fill=(70, 80, 96), font=font)
    draw.text((px1 - 5, py1 + 9), "1", fill=(70, 80, 96), font=font)
    draw.text((px0 - 13, py0 - 2), "1", fill=(70, 80, 96), font=font)
    draw.text((px0 - 13, py1 - 7), "0", fill=(70, 80, 96), font=font)
    for x_value, y_value, color in zip(xs, ys, colors):
        cx, cy = _point_xy(plot, x_value, y_value)
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=tuple(color))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, (250, 251, 253))
    draw = ImageDraw.Draw(image)
    font = _font()
    draw.text((38, 20), "paired measurements against a common diagonal", fill=(27, 35, 48), font=font)
    draw.text((38, 38), "Track each colored sample between panels P and Q", fill=(82, 91, 106), font=font)
    for label, swatch in zip(LABELS, LEGEND_SWATCH_RECTS):
        draw.rectangle(swatch, fill=tuple(scene["colors"][LABELS.index(label)]))
        draw.text((swatch[2] + 9, swatch[1] + 5), label, fill=(42, 49, 62), font=font)
    draw.text((705, 56), "dashed line: y = x", fill=(67, 76, 90), font=font)
    values = {
        "P": (scene["p_x"], scene["p_y"]),
        "Q": (scene["q_x"], scene["q_y"]),
    }
    for panel, plot, name in zip(PANEL_RECTS, PLOT_RECTS, scene["panel_order"]):
        xs, ys = values[name]
        _draw_panel(draw, panel, plot, name, xs, ys, scene["colors"], scene["marker_radius"], font)
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
