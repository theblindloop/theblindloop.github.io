"""Deterministic renderer and analytic answer for the state-pattern world.

The image contains five visibly numbered panels, four identity-coded tiles per
panel, a state key, an identity key, and a five-position TARGET code.  The
analytic answer only reads the two semantic fields ``states`` and
``target_pattern``; all layout and palette fields are rendered nuisance
parameters.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1200
HEIGHT = 720
PANEL_CENTERS = (125, 365, 605, 845, 1085)
PANEL_LEFTS = (20, 260, 500, 740, 980)
PANEL_WIDTH = 210
PANEL_TOP = 185
PANEL_BOTTOM = 575
ROW_Y = (250, 335, 420, 505)
TARGET_Y = 110
KEY_Y = 647
LABEL_KEY_X = (75, 150, 225, 300)
STATE_KEY_X = (500, 580, 660)
LABELS = ("A", "B", "C", "D")

# These sets are deliberately disjoint.  Their order is permuted per scene and
# the visible key tells the inverse arm which color denotes each semantic code.
LABEL_COLORS = (
    (214, 70, 70),
    (52, 112, 218),
    (55, 166, 100),
    (155, 78, 188),
)
STATE_COLORS = (
    (242, 166, 48),
    (30, 165, 177),
    (220, 67, 145),
)
PANEL_TINTS = (
    (247, 250, 255),
    (250, 248, 244),
    (247, 252, 248),
    (250, 247, 252),
    (248, 250, 250),
)
STATE_SHAPE_NAMES = ("circle", "square", "triangle")
TARGET_LIBRARY = (
    (0, 1, 1, 2, 0),
    (1, 2, 0, 0, 1),
    (2, 0, 1, 1, 2),
    (0, 2, 2, 1, 0),
    (2, 1, 1, 0, 2),
    (1, 0, 0, 2, 1),
)


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _rgb(value: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _text_center(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, fill) -> None:
    font = _font()
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) // 2, xy[1] - (box[3] - box[1]) // 2 - 1),
        text,
        font=font,
        fill=fill,
    )


def _draw_state(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    shape: str,
    color: tuple[int, int, int],
) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    elif shape == "square":
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 5),
            fill=color,
        )
    elif shape == "triangle":
        draw.polygon(
            ((cx, cy - radius), (cx - radius, cy + radius), (cx + radius, cy + radius)),
            fill=color,
        )
    else:
        raise ValueError(f"unknown state shape {shape!r}")


def _scores(states: list[list[int]], target: list[int]) -> list[int]:
    if len(states) != 4 or len(target) != 5:
        raise ValueError("expected four trajectories and a five-position target")
    scores = []
    for trajectory in states:
        if len(trajectory) != 5 or any(value not in (0, 1, 2) for value in trajectory):
            raise ValueError("invalid three-state trajectory")
        scores.append(sum(int(a == b) for a, b in zip(trajectory, target)))
    return scores


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely answerable scene from one integer seed."""

    rng = random.Random(int(seed))
    target = list(rng.choice(TARGET_LIBRARY))
    answer_index = (int(seed) * 7 + 1) % 4
    trajectories: list[list[int] | None] = [None, None, None, None]
    trajectories[answer_index] = list(target)

    near_boundary = int(seed) % 11 == 0
    used: set[tuple[int, ...]] = {tuple(target)}
    if near_boundary:
        # One distractor differs in exactly one position, making the visible
        # runner-up gap one and exercising explicit quarantine behavior.
        near = list(target)
        near[0] = (near[0] + 1) % 3
        free = next(index for index in range(4) if trajectories[index] is None)
        trajectories[free] = near
        used.add(tuple(near))

    for index in range(4):
        if trajectories[index] is not None:
            continue
        for _ in range(10000):
            candidate = [rng.randrange(3) for _ in range(5)]
            matches = sum(a == b for a, b in zip(candidate, target))
            limit = 3
            if tuple(candidate) not in used and matches <= limit:
                trajectories[index] = candidate
                used.add(tuple(candidate))
                break
        if trajectories[index] is None:
            raise RuntimeError("could not construct a unique trajectory set")

    row_orders = [rng.sample(range(4), 4) for _ in range(5)]
    label_palette = [list(color) for color in LABEL_COLORS]
    state_palette = [list(color) for color in STATE_COLORS]
    rng.shuffle(label_palette)
    rng.shuffle(state_palette)
    panel_tints = [list(PANEL_TINTS[index]) for index in rng.sample(range(5), 5)]
    state_shapes = list(STATE_SHAPE_NAMES)
    rng.shuffle(state_shapes)

    return {
        "states": [list(trajectory) for trajectory in trajectories],
        "target_pattern": target,
        "row_orders": row_orders,
        "label_palette": label_palette,
        "state_palette": state_palette,
        "panel_tints": panel_tints,
        "state_shapes": state_shapes,
        "marker_radius": rng.choice((15, 16, 17, 18)),
        "tile_outline_width": rng.choice((2, 3)),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    label_palette = [_rgb(color) for color in scene["label_palette"]]
    state_palette = [_rgb(color) for color in scene["state_palette"]]
    panel_tints = [_rgb(color) for color in scene["panel_tints"]]
    states = scene["states"]
    target = scene["target_pattern"]
    row_orders = scene["row_orders"]
    state_shapes = scene["state_shapes"]
    radius = int(scene["marker_radius"])
    outline_width = int(scene["tile_outline_width"])
    dark = (35, 43, 55)
    muted = (103, 113, 127)

    draw.rectangle((0, 0, WIDTH - 1, 45), fill=(30, 42, 62))
    draw.text((25, 15), "TRACK THE CODE", font=_font(), fill=(255, 255, 255))
    draw.text((25, 67), "TARGET", font=_font(), fill=dark)
    draw.text((25, 91), "five states", font=_font(), fill=muted)

    for panel_index, center_x in enumerate(PANEL_CENTERS):
        draw.rounded_rectangle(
            (PANEL_LEFTS[panel_index], PANEL_TOP, PANEL_LEFTS[panel_index] + PANEL_WIDTH, PANEL_BOTTOM),
            radius=12,
            fill=panel_tints[panel_index],
            outline=(188, 196, 208),
            width=2,
        )
        draw.rounded_rectangle(
            (center_x - 31, TARGET_Y - 31, center_x + 31, TARGET_Y + 31),
            radius=10,
            fill=(255, 255, 255),
            outline=(188, 196, 208),
            width=2,
        )
        _draw_state(
            draw,
            (center_x, TARGET_Y),
            radius,
            state_shapes[target[panel_index]],
            state_palette[target[panel_index]],
        )
        _text_center(draw, (PANEL_LEFTS[panel_index] + 25, 164), f"P{panel_index + 1}", dark)
        if panel_index < len(PANEL_CENTERS) - 1:
            draw.line(
                (center_x + 108, 176, PANEL_CENTERS[panel_index + 1] - 108, 176),
                fill=(150, 160, 174),
                width=2,
            )
            draw.polygon(
                (
                    (PANEL_CENTERS[panel_index + 1] - 112, 172),
                    (PANEL_CENTERS[panel_index + 1] - 104, 176),
                    (PANEL_CENTERS[panel_index + 1] - 112, 180),
                ),
                fill=(150, 160, 174),
            )

        tile_left = PANEL_LEFTS[panel_index] + 15
        tile_right = PANEL_LEFTS[panel_index] + PANEL_WIDTH - 15
        for row, identity in enumerate(row_orders[panel_index]):
            y = ROW_Y[row]
            draw.rounded_rectangle(
                (tile_left, y - 31, tile_right, y + 31),
                radius=9,
                fill=(255, 255, 255),
                outline=(133, 145, 162),
                width=outline_width,
            )
            draw.rectangle(
                (tile_left + 8, y - 17, tile_left + 38, y + 17),
                fill=label_palette[identity],
            )
            _text_center(draw, (tile_left + 23, y), LABELS[identity], (255, 255, 255))
            _draw_state(
                draw,
                (tile_left + 135, y),
                radius,
                state_shapes[states[identity][panel_index]],
                state_palette[states[identity][panel_index]],
            )

    draw.text((30, 606), "IDENTITY KEY", font=_font(), fill=dark)
    for identity, x in enumerate(LABEL_KEY_X):
        draw.rectangle((x - 15, KEY_Y - 14, x + 15, KEY_Y + 14), fill=label_palette[identity])
        _text_center(draw, (x, KEY_Y + 25), LABELS[identity], dark)

    draw.text((430, 606), "STATE KEY", font=_font(), fill=dark)
    for state, x in enumerate(STATE_KEY_X):
        _draw_state(draw, (x, KEY_Y), 13, state_shapes[state], state_palette[state])
        _text_center(draw, (x, KEY_Y + 25), str(state), dark)

    draw.text((790, 632), "Match all five positions", font=_font(), fill=muted)
    draw.text((790, 651), "A  B  C  or  D", font=_font(), fill=dark)
    return image


def analytic_gold(scene: dict) -> str:
    states = scene["states"]
    target = scene["target_pattern"]
    scores = _scores(states, target)
    best = max(scores)
    if scores.count(best) != 1 or best != 5:
        raise ValueError(f"scene does not contain one exact target trajectory: {scores}")
    return LABELS[scores.index(best)]


def margin(scene: dict) -> float:
    states = scene["states"]
    target = scene["target_pattern"]
    scores = sorted(_scores(states, target), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    states = scene["states"]
    target = scene["target_pattern"]
    scores = _scores(states, target)
    best = max(scores)
    return scores.count(best) != 1 or best != 5 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No render-invariant alias is asserted: the identity and state fields that
    # determine gold are both rendered in keyed visible locations.
    return []
