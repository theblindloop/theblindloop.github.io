"""Renderer and analytic gold for the stairpush_corridor_replay world.

One thin staircase corridor of seventeen square cells is drawn as a single
connected serpentine chain. Every cell carries a floor pip that is either a
solid black disc (step away from the black head cap) or a hollow black ring
(step back towards the head cap). One red disc is the shover and three brown
blocks are crates. The shover repeatedly reads the pip of the cell it stands
on and steps one cell that way, shoving a crate that stands in the way one
further cell along; the replay stops as soon as the step cannot be taken.

The answer is the pale tint of the cell where the shover comes to rest.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "stairpush_corridor_replay-renderer-0.1.0"

CANVAS = (640, 480)
CELL = 44
STEP_CAP = 40

# Exact aliasing-free palette; every value below is unique in the raster.
BACKGROUND = (255, 255, 255)
BORDER = (90, 90, 90)
PIP = (0, 0, 0)
CRATE = (110, 70, 35)
SHOVER = (200, 30, 30)
DECOR = (210, 210, 210)

ZONE_NAMES = ("red", "green", "blue", "yellow")
ZONE_TINTS = {
    "red": (247, 205, 205),
    "green": (205, 240, 210),
    "blue": (205, 220, 247),
    "yellow": (250, 240, 195),
}

# Grid coordinates (column, row) of the seventeen corridor cells, in path order
# from the head cap end to the tail wall end.
PATH_CELLS = (
    [(c, 0) for c in range(9)]
    + [(8, 1)]
    + [(c, 2) for c in range(8, -1, -1)]
)
GRID_COLS = 9
GRID_ROWS = 3
N_CELLS = len(PATH_CELLS)
N_CRATES = 3


def _replay(pips: list[int], start: int, crates: frozenset[int]) -> tuple[int, int]:
    """Return (resting cell index, steps taken); -1 steps means non-terminating."""
    occupied = set(crates)
    pos = start
    for taken in range(STEP_CAP + 1):
        step = 1 if pips[pos] else -1
        nxt = pos + step
        if nxt < 0 or nxt >= N_CELLS:
            return pos, taken
        if nxt in occupied:
            beyond = nxt + step
            if beyond < 0 or beyond >= N_CELLS or beyond in occupied:
                return pos, taken
            occupied.discard(nxt)
            occupied.add(beyond)
        pos = nxt
    return pos, -1


def _zone_of(zones: list[int], index: int) -> str:
    return ZONE_NAMES[zones[index]]


def _sample_zones(rng: random.Random) -> list[int]:
    while True:
        cuts = sorted(rng.sample(range(1, N_CELLS), 3))
        lengths = [cuts[0], cuts[1] - cuts[0], cuts[2] - cuts[1], N_CELLS - cuts[2]]
        if min(lengths) >= 4:
            break
    order = list(range(4))
    rng.shuffle(order)
    zones: list[int] = []
    for band, length in enumerate(lengths):
        zones.extend([order[band]] * length)
    return zones


def _sample_decor(rng: random.Random, box: tuple[int, int, int, int], clear: float) -> list[dict]:
    x0, y0, x1, y1 = box
    keep_out = (x0 - clear, y0 - clear, x1 + clear, y1 + clear)
    decor: list[dict] = []
    attempts = 0
    while len(decor) < rng.randint(4, 7) and attempts < 400:
        attempts += 1
        w = rng.randint(24, 64)
        h = rng.randint(24, 64)
        px = rng.randint(6, CANVAS[0] - w - 6)
        py = rng.randint(6, CANVAS[1] - h - 6)
        if not (px + w < keep_out[0] or px > keep_out[2] or py + h < keep_out[1] or py > keep_out[3]):
            continue
        decor.append({"kind": rng.choice(["ring", "slash", "bar"]), "box": [px, py, px + w, py + h]})
    return decor


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene, balanced on the answering tint."""
    rng = random.Random(seed * 7919 + 13)
    target = ZONE_NAMES[seed % 4]
    for _ in range(6000):
        pips = [rng.randint(0, 1) for _ in range(N_CELLS)]
        crates = frozenset(rng.sample(range(1, N_CELLS - 1), N_CRATES))
        free = [i for i in range(N_CELLS) if i not in crates]
        start = rng.choice(free)
        rest, steps = _replay(pips, start, crates)
        if steps < 5:
            continue
        zones = _sample_zones(rng)
        answer = _zone_of(zones, rest)
        if answer != target:
            continue
        if _zone_of(zones, start) == answer:
            continue
        sensitive = 0
        for i in range(N_CELLS):
            flipped = list(pips)
            flipped[i] ^= 1
            alt_rest, alt_steps = _replay(flipped, start, crates)
            if alt_steps < 0 or _zone_of(zones, alt_rest) != answer:
                sensitive += 1
        if sensitive < 3:
            continue

        x0 = rng.randint(46, CANVAS[0] - GRID_COLS * CELL - 46)
        y0 = rng.randint(44, CANVAS[1] - GRID_ROWS * CELL - 44)
        clear = round(rng.choice([2.4, 3.6, 5.0, 7.5, 10.0, 13.0]), 2)
        box = (x0, y0, x0 + GRID_COLS * CELL, y0 + GRID_ROWS * CELL)
        return {
            "pips": pips,
            "start": start,
            "crates": sorted(crates),
            "zones": zones,
            "x0": x0,
            "y0": y0,
            "decor_clear": clear,
            "decor": _sample_decor(rng, box, clear),
            "rest": rest,
            "steps": steps,
            "sensitive": sensitive,
        }
    raise RuntimeError(f"no admissible scene for seed {seed}")


def _cell_box(scene: dict, index: int) -> tuple[int, int, int, int]:
    col, row = PATH_CELLS[index]
    x = scene["x0"] + col * CELL
    y = scene["y0"] + row * CELL
    return x, y, x + CELL, y + CELL


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)

    for item in scene["decor"]:
        bx0, by0, bx1, by1 = item["box"]
        if item["kind"] == "ring":
            draw.ellipse([bx0, by0, bx1, by1], outline=DECOR, width=3)
        elif item["kind"] == "slash":
            draw.line([bx0, by1, bx1, by0], fill=DECOR, width=3)
        else:
            draw.rectangle([bx0, by0, bx1, (by0 + by1) // 2], outline=DECOR, width=3)

    crates = set(scene["crates"])
    for index in range(N_CELLS):
        x, y, x1, y1 = _cell_box(scene, index)
        tint = ZONE_TINTS[ZONE_NAMES[scene["zones"][index]]]
        draw.rectangle([x, y, x1, y1], fill=tint, outline=BORDER, width=2)
        cx = x + CELL // 2
        py = y + 12
        if scene["pips"][index]:
            draw.ellipse([cx - 5, py - 5, cx + 5, py + 5], fill=PIP)
        else:
            draw.ellipse([cx - 5, py - 5, cx + 5, py + 5], fill=BACKGROUND, outline=PIP, width=2)
        gy = y + CELL - 13
        if index in crates:
            draw.rectangle([cx - 7, gy - 7, cx + 7, gy + 7], fill=CRATE)
        if index == scene["start"]:
            draw.ellipse([cx - 8, gy - 8, cx + 8, gy + 8], fill=SHOVER)

    hx, hy, _, hy1 = _cell_box(scene, 0)
    draw.rectangle([hx - 11, hy, hx - 3, hy1], fill=PIP)
    _, ty, tx1, ty1 = _cell_box(scene, N_CELLS - 1)
    draw.rectangle([tx1 + 3, ty, tx1 + 11, ty1], fill=BORDER)
    return image


def analytic_gold(scene: dict) -> str:
    rest, steps = _replay(scene["pips"], scene["start"], frozenset(scene["crates"]))
    if steps < 0:
        raise RuntimeError("non-terminating replay reached analytic gold")
    return ZONE_NAMES[scene["zones"][rest]]


def margin(scene: dict) -> float:
    """Composite legibility clearance in pixels; smallest resolvable separation."""
    pip_to_piece = float((CELL - 13 - 8) - (12 + 5))
    return float(min(scene["decor_clear"], pip_to_piece, 6.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is painted into the raster."""
    return []
