"""Renderer and analytic arm for the far-field blur-ledger world.

Four disconnected corner windows contain randomized two-tone micro-raster
carriers.  A declared 3-by-3 low-pass average turns each carrier into a 4-by-4
tile; the four tiles are then bound by their visible compass positions.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw


WIDTH = 768
HEIGHT = 512
WINDOW_SIDE = 136
INNER_OFFSET = 18
GRID_SIDE = 12
CELL = 8
TILE_SIDE = 4
SYMBOL_SIDE = 8


def _rows(rows: tuple[str, ...]) -> tuple[tuple[int, ...], ...]:
    if len(rows) != SYMBOL_SIDE or any(len(row) != SYMBOL_SIDE for row in rows):
        raise AssertionError("symbol templates must be 8 by 8")
    return tuple(tuple(int(char == "#") for char in row) for row in rows)


ORBIT = _rows(
    (
        ".##..##.",
        "#..##..#",
        "#..##..#",
        ".##..##.",
        "####..##",
        "....##..",
        "####..##",
        "....##..",
    )
)
LATTICE = _rows(
    (
        "####.#.#",
        "....#.#.",
        "####.#.#",
        "....#.#.",
        "##..####",
        "..##....",
        "##..####",
        "..##....",
    )
)
TEMPLATES = {"orbit": ORBIT, "lattice": LATTICE}
SYMBOLS = tuple(TEMPLATES)


BACKGROUNDS = (
    (246, 244, 236),
    (240, 246, 242),
    (245, 241, 248),
)
PALETTES = (
    {
        "frame": (31, 55, 83),
        "light": (222, 224, 207),
        "dark": (48, 64, 104),
        "anchor": (224, 76, 43),
        "distractor": (75, 139, 132),
    },
    {
        "frame": (57, 43, 80),
        "light": (230, 218, 199),
        "dark": (67, 58, 112),
        "anchor": (226, 91, 47),
        "distractor": (69, 126, 170),
    },
    {
        "frame": (35, 76, 70),
        "light": (218, 228, 208),
        "dark": (43, 81, 91),
        "anchor": (221, 82, 44),
        "distractor": (126, 86, 155),
    },
    {
        "frame": (72, 52, 40),
        "light": (235, 220, 190),
        "dark": (78, 54, 68),
        "anchor": (224, 78, 42),
        "distractor": (52, 122, 144),
    },
)


def _hamming(left: tuple[tuple[int, ...], ...], right: tuple[tuple[int, ...], ...]) -> int:
    return sum(
        value_left != value_right
        for row_left, row_right in zip(left, right, strict=True)
        for value_left, value_right in zip(row_left, row_right, strict=True)
    )


def _tile(template: tuple[tuple[int, ...], ...], row: int, column: int) -> tuple[tuple[int, ...], ...]:
    return tuple(
        tuple(template[row * TILE_SIDE + local_row][column * TILE_SIDE + local_column] for local_column in range(TILE_SIDE))
        for local_row in range(TILE_SIDE)
    )


def _cell_order(scene: dict, window_index: int, block_index: int) -> list[int]:
    rng = random.Random(
        int(scene["carrier_seed"]) + 1009 * window_index + 9176 * block_index
    )
    order = list(range(9))
    rng.shuffle(order)
    return order


def _window_base(window: dict) -> tuple[int, int]:
    return int(round(float(window["x"]))), int(round(float(window["y"])))


def sample_scene(seed: int) -> dict:
    """Sample paired labels with shared nuisance geometry and carriers."""

    value = int(seed)
    symbol = SYMBOLS[value & 1]
    rng = random.Random((value // 2) * 1000003 + 739391)
    bases = ((34.0, 28.0), (598.0, 28.0), (32.0, 342.0), (596.0, 342.0))
    windows: list[dict] = []
    for index, (base_x, base_y) in enumerate(bases):
        windows.append(
            {
                "row": index // 2,
                "column": index % 2,
                "x": base_x + rng.uniform(-9.0, 9.0),
                "y": base_y + rng.uniform(-9.0, 9.0),
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "windows": windows,
        "palette": rng.randrange(len(PALETTES)),
        "background": rng.randrange(len(BACKGROUNDS)),
        "symbol": symbol,
        "carrier_seed": rng.randrange(1, 2**31 - 1),
        "marker_style": rng.randrange(2),
        "distractor_seed": rng.randrange(1, 2**31 - 1),
        "distractor_count": rng.randrange(2, 5),
    }


def _draw_distractors(draw: ImageDraw.ImageDraw, scene: dict, palette: dict) -> None:
    rng = random.Random(int(scene["distractor_seed"]))
    windows = scene["windows"]
    placed = 0
    attempts = 0
    while placed < int(scene["distractor_count"]) and attempts < 300:
        attempts += 1
        x = rng.randrange(18, WIDTH - 18)
        y = rng.randrange(18, HEIGHT - 18)
        if any(
            float(window["x"]) - 20 <= x <= float(window["x"]) + WINDOW_SIDE + 20
            and float(window["y"]) - 20 <= y <= float(window["y"]) + WINDOW_SIDE + 20
            for window in windows
        ):
            continue
        radius = 3 if placed % 2 else 4
        if placed % 2:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette["distractor"])
        else:
            draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=palette["distractor"])
        placed += 1


def render(scene: dict) -> Image.Image:
    """Render four disconnected anchored windows and their cell carriers."""

    if (int(scene["width"]), int(scene["height"])) != (WIDTH, HEIGHT):
        raise ValueError("unsupported canvas size")
    symbol = str(scene["symbol"])
    if symbol not in TEMPLATES:
        raise ValueError(f"unknown symbol {symbol!r}")
    windows = list(scene["windows"])
    if len(windows) != 4:
        raise ValueError("four windows are required")
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    background = BACKGROUNDS[int(scene["background"]) % len(BACKGROUNDS)]
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, scene, palette)
    template = TEMPLATES[symbol]
    for index, window in enumerate(windows):
        x0, y0 = _window_base(window)
        if not (20 <= x0 <= WIDTH - WINDOW_SIDE - 20 and 16 <= y0 <= HEIGHT - WINDOW_SIDE - 16):
            raise ValueError("window outside canvas margin")
        draw.rectangle((x0, y0, x0 + WINDOW_SIDE - 1, y0 + WINDOW_SIDE - 1), fill=palette["frame"])
        inner_x = x0 + INNER_OFFSET
        inner_y = y0 + INNER_OFFSET
        draw.rectangle(
            (inner_x, inner_y, inner_x + GRID_SIDE * CELL - 1, inner_y + GRID_SIDE * CELL - 1),
            fill=palette["light"],
        )
        tile = _tile(template, int(window["row"]), int(window["column"]))
        for block_row in range(TILE_SIDE):
            for block_column in range(TILE_SIDE):
                bit = tile[block_row][block_column]
                dark_count = 6 if bit else 3
                dark_indices = set(_cell_order(scene, index, block_row * TILE_SIDE + block_column)[:dark_count])
                for local in range(9):
                    row = block_row * 3 + local // 3
                    column = block_column * 3 + local % 3
                    color = palette["dark"] if local in dark_indices else palette["light"]
                    left = inner_x + column * CELL
                    top = inner_y + row * CELL
                    draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=color)
        marker_box = (x0 + 7, y0 + 7, x0 + 18, y0 + 18)
        if int(scene["marker_style"]) % 2:
            draw.ellipse(marker_box, fill=palette["anchor"])
        else:
            draw.rectangle((x0 + 8, y0 + 8, x0 + 17, y0 + 17), fill=palette["anchor"])
    return image


def analytic_gold(scene: dict) -> str:
    symbol = str(scene["symbol"])
    if symbol not in TEMPLATES:
        raise ValueError(f"unknown symbol {symbol!r}")
    return symbol


def margin(scene: dict) -> float:
    symbol = str(scene["symbol"])
    other = "lattice" if symbol == "orbit" else "orbit"
    return float(_hamming(TEMPLATES[symbol], TEMPLATES[other]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
