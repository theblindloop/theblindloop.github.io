"""Independent pixel-only inverse for the far-field blur ledger."""

from __future__ import annotations

import numpy as np
from PIL import Image


WIDTH = 768
HEIGHT = 512
WINDOW_SIDE = 136
INNER_OFFSET = 18
GRID_SIDE = 12
CELL = 8
TILE_SIDE = 4
SYMBOL_SIDE = 8
TONE_TOLERANCE = 24.0


def _rows(rows: tuple[str, ...]) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(int(char == "#") for char in row) for row in rows)


ORBIT = _rows(
    (
        ".##..##.",
        "#..##..#",
        "#..##..#",
        ".##..##.",
        "####..##",
        "....##..",
        "####..##",
        "....##..",
    )
)
LATTICE = _rows(
    (
        "####.#.#",
        "....#.#.",
        "####.#.#",
        "....#.#.",
        "##..####",
        "..##....",
        "##..####",
        "..##....",
    )
)
TEMPLATES = {"orbit": ORBIT, "lattice": LATTICE}


def _components(mask: np.ndarray, minimum: int = 30) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        visited[y, x] = True
        stack = [(y, x)]
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
        if len(points) >= minimum:
            result.append(np.asarray(points, dtype=np.float64))
    return result


def _anchor_mask(array: np.ndarray) -> np.ndarray:
    red, green, blue = [array[:, :, index].astype(np.int16) for index in range(3)]
    return (red > 150) & (red - green > 58) & (red - blue > 52) & (green < 160)


def _origins(array: np.ndarray) -> list[tuple[int, int]]:
    components = _components(_anchor_mask(array))
    if len(components) != 4:
        raise ValueError("expected four visible anchor fiducials")
    centers = np.asarray(
        [(float(component[:, 1].mean()), float(component[:, 0].mean())) for component in components],
        dtype=np.float64,
    )
    x_mid = float(np.median(centers[:, 0]))
    y_mid = float(np.median(centers[:, 1]))
    groups: dict[tuple[int, int], tuple[int, int]] = {}
    for x, y in centers:
        key = (int(y >= y_mid), int(x >= x_mid))
        if key in groups:
            raise ValueError("anchors do not define four compass quadrants")
        x0 = int(round(float(x) - 12.5))
        y0 = int(round(float(y) - 12.5))
        groups[key] = (x0, y0)
    if set(groups) != {(0, 0), (0, 1), (1, 0), (1, 1)}:
        raise ValueError("missing compass quadrant anchor")
    for x0, y0 in groups.values():
        if not (0 <= x0 < WIDTH and 0 <= y0 < HEIGHT and x0 + WINDOW_SIDE <= WIDTH and y0 + WINDOW_SIDE <= HEIGHT):
            raise ValueError("anchor implies an invalid window")
    return [groups[(row, column)] for row in (0, 1) for column in (0, 1)]


def _cell_samples(array: np.ndarray, x0: int, y0: int) -> np.ndarray:
    inner_x = x0 + INNER_OFFSET
    inner_y = y0 + INNER_OFFSET
    samples: list[np.ndarray] = []
    for row in range(GRID_SIDE):
        for column in range(GRID_SIDE):
            left = inner_x + column * CELL + 2
            top = inner_y + row * CELL + 2
            patch = array[top : top + CELL - 4, left : left + CELL - 4]
            if patch.shape[:2] != (CELL - 4, CELL - 4):
                raise ValueError("micro-raster is clipped")
            samples.append(np.median(patch.reshape(-1, 3), axis=0))
    return np.asarray(samples, dtype=np.float32)


def _tones(samples: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    quantized = (samples // 8).astype(np.int16)
    values, counts = np.unique(quantized, axis=0, return_counts=True)
    if len(values) < 2:
        raise ValueError("two cell tones are not visible")
    order = np.argsort(-counts)
    if int(counts[order[1]]) < 45:
        raise ValueError("one cell tone has insufficient coverage")
    centers = []
    for index in order[:2]:
        centers.append(samples[np.all(quantized == values[index], axis=1)].mean(axis=0))
    if float(np.linalg.norm(centers[0] - centers[1])) < 40.0:
        raise ValueError("cell tones are not separated")
    return np.asarray(centers[0], dtype=np.float32), np.asarray(centers[1], dtype=np.float32)


def _blur_tile(array: np.ndarray, origin: tuple[int, int]) -> tuple[tuple[int, ...], ...]:
    x0, y0 = origin
    samples = _cell_samples(array, x0, y0)
    tone_a, tone_b = _tones(samples)
    distances = np.minimum(
        np.linalg.norm(samples - tone_a[None, :], axis=1),
        np.linalg.norm(samples - tone_b[None, :], axis=1),
    )
    if float(np.max(distances)) > TONE_TOLERANCE:
        raise ValueError("a micro-raster cell is incomplete or ablated")
    luminance = samples @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32)
    low = float(min(tone_a @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32), tone_b @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32)))
    high = float(max(tone_a @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32), tone_b @ np.asarray((0.299, 0.587, 0.114), dtype=np.float32)))
    threshold = (low + high) / 2.0
    tile: list[tuple[int, ...]] = []
    for block_row in range(TILE_SIDE):
        output_row: list[int] = []
        for block_column in range(TILE_SIDE):
            start = (block_row * 3) * GRID_SIDE + block_column * 3
            block_indices = [start + local_row * GRID_SIDE + local_column for local_row in range(3) for local_column in range(3)]
            mean_luminance = float(np.mean(luminance[block_indices]))
            if abs(mean_luminance - threshold) < (high - low) * 0.08:
                raise ValueError("blurred block is near the decision boundary")
            output_row.append(int(mean_luminance < threshold))
        tile.append(tuple(output_row))
    return tuple(tile)


def _assemble(tiles: list[tuple[tuple[int, ...], ...]]) -> tuple[tuple[int, ...], ...]:
    rows: list[tuple[int, ...]] = []
    for panel_row in (0, 1):
        for local_row in range(TILE_SIDE):
            rows.append(tiles[panel_row * 2][local_row] + tiles[panel_row * 2 + 1][local_row])
    return tuple(rows)


def _score(observed: tuple[tuple[int, ...], ...], expected: tuple[tuple[int, ...], ...]) -> int:
    return sum(
        value != target
        for observed_row, expected_row in zip(observed, expected, strict=True)
        for value, target in zip(observed_row, expected_row, strict=True)
    )


def decision_from_image(image: Image.Image) -> str:
    """Recover ``orbit`` or ``lattice`` using only the final RGB pixels."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError("unexpected raster dimensions")
    origins = _origins(array)
    tiles = [_blur_tile(array, origin) for origin in origins]
    observed = _assemble(tiles)
    scores = {name: _score(observed, template) for name, template in TEMPLATES.items()}
    ranked = sorted(scores.items(), key=lambda item: item[1])
    if ranked[1][1] - ranked[0][1] < 8:
        raise ValueError("blurred symbol margin is ambiguous")
    return ranked[0][0]
