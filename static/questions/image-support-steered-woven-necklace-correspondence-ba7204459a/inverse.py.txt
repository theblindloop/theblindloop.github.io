"""Independent pixel-only inverse arm for the contiguous cyclic-order quilt."""

from __future__ import annotations

import numpy as np
from PIL import Image

_WHITE = np.array((255, 255, 255), dtype=np.uint8)
_TOKEN_COLORS = {
    (205, 70, 78),
    (43, 119, 190),
    (55, 151, 91),
    (225, 132, 38),
}
_TAG_COLORS = {
    (244, 181, 35): "amber",
    (30, 174, 190): "cyan",
    (145, 84, 185): "violet",
}
_ROWS = 4
_COLS = 13
_SYMBOLS = 12


def _distance_to_orbit(reference: tuple, candidate: tuple) -> int:
    forms = []
    for direction in (reference, tuple(reversed(reference))):
        for offset in range(len(reference)):
            forms.append(direction[offset:] + direction[:offset])
    return min(
        sum(left != right for left, right in zip(form, candidate, strict=True))
        for form in forms
    )


def _center_color(crop: np.ndarray, row: int, column: int) -> tuple[int, int, int]:
    height, width = crop.shape[:2]
    y = min(height - 1, int((row + 0.5) * height / _ROWS))
    x = min(width - 1, int((column + 0.5) * width / _COLS))
    return tuple(int(value) for value in crop[y, x])


def decision_from_image(image: Image.Image) -> str:
    """Recover the tag only from a final RGB raster; raise on damaged evidence."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    foreground = np.any(array != _WHITE, axis=2)
    ys, xs = np.nonzero(foreground)
    if not len(xs):
        raise ValueError("no quilt foreground")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    crop = array[y0:y1, x0:x1]
    height, width = crop.shape[:2]
    if width < 260 or height < 160:
        raise ValueError("quilt is too small")
    if width % _COLS or height % _ROWS:
        raise ValueError("foreground does not form the declared 4 by 13 quilt")
    if abs(width / height - 1.625) > 0.08:
        raise ValueError("quilt aspect is malformed")
    if np.any(np.all(crop == _WHITE, axis=2)):
        raise ValueError("the connected quilt surface has been erased")

    reference = tuple(_center_color(crop, 0, column) for column in range(1, _COLS))
    rows = [
        tuple(_center_color(crop, row, column) for column in range(1, _COLS))
        for row in range(1, _ROWS)
    ]
    tags = [_center_color(crop, row, 0) for row in range(1, _ROWS)]
    if any(color not in _TOKEN_COLORS for color in reference):
        raise ValueError("reference contains an unknown token color")
    if any(color not in _TOKEN_COLORS for row in rows for color in row):
        raise ValueError("candidate contains an unknown token color")
    if len(set(tags)) != 3 or any(color not in _TAG_COLORS for color in tags):
        raise ValueError("answer tags are missing or repeated")
    histogram = sorted(reference)
    if any(sorted(row) != histogram for row in rows):
        raise ValueError("row token histograms differ")

    distances = [_distance_to_orbit(reference, row) for row in rows]
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("cyclic correspondence is not unique")
    if sorted(distances)[1] < 2:
        raise ValueError("runner-up is inside the quarantine boundary")
    return _TAG_COLORS[tags[matches[0]]]
