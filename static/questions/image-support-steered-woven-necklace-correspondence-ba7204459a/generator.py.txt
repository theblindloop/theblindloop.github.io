"""Deterministic latent sampler, renderer, and analytic arm for the quilt world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_WIDTH = 768
CANVAS_HEIGHT = 512
ROWS = 4
SYMBOLS = 12
COLS = SYMBOLS + 1
CELL_WIDTH = 52
CELL_HEIGHT = 104
QUILT_WIDTH = COLS * CELL_WIDTH
QUILT_HEIGHT = ROWS * CELL_HEIGHT

BACKGROUND = (255, 255, 255)
BASE = (239, 228, 194)
GRID = (53, 47, 43)
REFERENCE_TAG = (112, 113, 108)

TOKEN_RGB = {
    "coral": (205, 70, 78),
    "azure": (43, 119, 190),
    "leaf": (55, 151, 91),
    "tangerine": (225, 132, 38),
}
TAG_RGB = {
    "amber": (244, 181, 35),
    "cyan": (30, 174, 190),
    "violet": (145, 84, 185),
}
TAG_NAMES = tuple(TAG_RGB)
TOKEN_NAMES = tuple(TOKEN_RGB)


def _rotations(values: list[int] | tuple[int, ...]) -> Iterable[tuple[int, ...]]:
    word = tuple(values)
    for direction in (word, tuple(reversed(word))):
        for offset in range(len(word)):
            yield direction[offset:] + direction[:offset]


def _distance_to_orbit(reference: list[int], candidate: list[int]) -> int:
    return min(
        sum(left != right for left, right in zip(form, candidate, strict=True))
        for form in _rotations(reference)
    )


def _transform(reference: list[int], offset: int, reverse: bool) -> list[int]:
    word = list(reversed(reference)) if reverse else list(reference)
    return word[offset:] + word[:offset]


def _make_distractor(
    rng: random.Random,
    correct: list[int],
    reference: list[int],
    minimum_distance: int,
) -> list[int]:
    for _ in range(2000):
        candidate = list(correct)
        swaps = 1 if minimum_distance <= 2 else 2
        for _ in range(swaps):
            options = [
                (i, j)
                for i in range(SYMBOLS)
                for j in range(i + 1, SYMBOLS)
                if candidate[i] != candidate[j]
            ]
            i, j = rng.choice(options)
            candidate[i], candidate[j] = candidate[j], candidate[i]
        if _distance_to_orbit(reference, candidate) >= minimum_distance:
            return candidate
    raise RuntimeError("could not construct a separated distractor")


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed residues counterbalance answer and row."""

    rng = random.Random(int(seed))
    for _ in range(1000):
        reference = [symbol for symbol in range(4) for _ in range(3)]
        rng.shuffle(reference)
        if len(set(_rotations(reference))) == 2 * SYMBOLS:
            break
    else:
        raise RuntimeError("failed to sample an asymmetric reference cycle")

    offset = rng.randrange(SYMBOLS)
    reverse = bool(rng.randrange(2))
    correct = _transform(reference, offset, reverse)
    distractor_near = _make_distractor(rng, correct, reference, 2)
    distractor_far = _make_distractor(rng, correct, reference, 4)

    correct_row = (int(seed) // 3) % 3
    correct_tag = TAG_NAMES[int(seed) % 3]
    remaining_tags = [tag for tag in TAG_NAMES if tag != correct_tag]
    rng.shuffle(remaining_tags)
    candidate_tags = list(remaining_tags)
    candidate_tags.insert(correct_row, correct_tag)

    distractors = [distractor_near, distractor_far]
    rng.shuffle(distractors)
    candidate_sequences = list(distractors)
    candidate_sequences.insert(correct_row, correct)

    token_palette = list(TOKEN_NAMES)
    rng.shuffle(token_palette)
    return {
        "canvas_width": CANVAS_WIDTH,
        "canvas_height": CANVAS_HEIGHT,
        "left": 36 + rng.randrange(17),
        "top": 40 + rng.randrange(17),
        "reference": reference,
        "candidate_sequences": candidate_sequences,
        "candidate_tags": candidate_tags,
        "token_palette": token_palette,
    }


def _draw_patch(
    draw: ImageDraw.ImageDraw,
    left: int,
    top: int,
    color: tuple[int, int, int],
    *,
    tag: bool = False,
) -> None:
    inset_x = 7 if tag else 8
    inset_y = 12 if tag else 15
    radius = 12 if tag else 9
    draw.rounded_rectangle(
        (
            left + inset_x,
            top + inset_y,
            left + CELL_WIDTH - inset_x - 1,
            top + CELL_HEIGHT - inset_y - 1,
        ),
        radius=radius,
        fill=color,
    )


def render(scene: dict) -> Image.Image:
    """Render the full nonwhite quilt as one connected area-like component."""

    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    left = int(scene["left"])
    top = int(scene["top"])
    reference = [int(value) for value in scene["reference"]]
    candidate_sequences = [
        [int(value) for value in row] for row in scene["candidate_sequences"]
    ]
    candidate_tags = [str(value) for value in scene["candidate_tags"]]
    token_palette = [str(value) for value in scene["token_palette"]]

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle(
        (left, top, left + QUILT_WIDTH - 1, top + QUILT_HEIGHT - 1),
        fill=BASE,
    )

    _draw_patch(draw, left, top, REFERENCE_TAG, tag=True)
    all_rows = [reference, *candidate_sequences]
    for row_index, sequence in enumerate(all_rows):
        row_top = top + row_index * CELL_HEIGHT
        if row_index:
            _draw_patch(
                draw,
                left,
                row_top,
                TAG_RGB[candidate_tags[row_index - 1]],
                tag=True,
            )
        for column, symbol in enumerate(sequence, start=1):
            _draw_patch(
                draw,
                left + column * CELL_WIDTH,
                row_top,
                TOKEN_RGB[token_palette[symbol]],
            )

    for column in range(COLS + 1):
        x = left + column * CELL_WIDTH
        if column == 0:
            x += 1
        elif column == COLS:
            x -= 2
        draw.line((x, top, x, top + QUILT_HEIGHT - 1), fill=GRID, width=3)
    for row in range(ROWS + 1):
        y = top + row * CELL_HEIGHT
        if row == 0:
            y += 1
        elif row == ROWS:
            y -= 2
        draw.line((left, y, left + QUILT_WIDTH - 1, y), fill=GRID, width=3)
    return image


def _candidate_distances(scene: dict) -> list[int]:
    reference = [int(value) for value in scene["reference"]]
    candidates = [
        [int(value) for value in row] for row in scene["candidate_sequences"]
    ]
    return [_distance_to_orbit(reference, candidate) for candidate in candidates]


def analytic_gold(scene: dict) -> str:
    distances = _candidate_distances(scene)
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one matching band")
    tags = [str(value) for value in scene["candidate_tags"]]
    return tags[matches[0]]


def margin(scene: dict) -> float:
    distances = sorted(_candidate_distances(scene))
    if not distances:
        return 0.0
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    distances = _candidate_distances(scene)
    return distances.count(0) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
