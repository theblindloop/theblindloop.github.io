"""Independent PNG-only inverse arm for the key-calibrated averaging task."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 9


def _blank() -> list[list[int]]:
    return [[0 for _ in range(GRID_N)] for _ in range(GRID_N)]


def _template(points: list[tuple[int, int]]) -> tuple[tuple[int, ...], ...]:
    values = _blank()
    for x, y in points:
        values[y][x] = 1
    return tuple(tuple(row) for row in values)


# These are the four public answer shapes, independently encoded here rather
# than imported from the forward renderer.
_SHAPES = {
    "tee": _template([(x, 1) for x in range(2, 7)] + [(4, y) for y in range(2, 8)]),
    "corner": _template([(2, y) for y in range(1, 8)] + [(x, 7) for x in range(3, 7)]),
    "chevron": _template(
        [
            (1, 1), (7, 1), (1, 2), (7, 2), (2, 3), (6, 3),
            (2, 4), (6, 4), (3, 5), (5, 5), (4, 6),
        ]
    ),
    "fork": _template(
        [(1, 1), (7, 1), (2, 2), (6, 2), (3, 3), (5, 3)]
        + [(4, y) for y in range(4, 9)]
    ),
}


def _frame_geometry(array: np.ndarray) -> tuple[int, int, int]:
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    cyan = (red < 70) & (green > 125) & (green < 195) & (blue > 145)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 100:
        raise ValueError("cyan frame missing")
    xmin, xmax = int(xs.min()), int(xs.max())
    ymin, ymax = int(ys.min()), int(ys.max())
    mid_y = (ymin + ymax) // 2
    mid_x = (xmin + xmax) // 2
    horizontal = cyan[mid_y, xmin : xmax + 1]
    vertical = cyan[ymin : ymax + 1, mid_x]
    left = int(np.argmax(~horizontal)) if np.any(~horizontal) else 0
    top = int(np.argmax(~vertical)) if np.any(~vertical) else 0
    right = int(np.argmax(~horizontal[::-1])) if np.any(~horizontal) else 0
    bottom = int(np.argmax(~vertical[::-1])) if np.any(~vertical) else 0
    if min(left, right, top, bottom) < 2:
        raise ValueError("cyan frame is incomplete")
    inner_w = xmax - xmin + 1 - left - right
    inner_h = ymax - ymin + 1 - top - bottom
    if inner_w != inner_h or inner_w % GRID_N:
        raise ValueError("cyan frame does not bound a square 9-by-9 field")
    cell = inner_w // GRID_N
    if not 36 <= cell <= 54:
        raise ValueError("implausible averaging-window size")
    return xmin + left, ymin + top, cell


def _check_gold_key(array: np.ndarray, cell: int) -> None:
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    gold = (red > 180) & (green > 115) & (green < 200) & (blue < 90)
    ys, xs = np.nonzero(gold)
    if len(xs) < 30:
        raise ValueError("gold averaging key missing")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if abs(width - cell) > 2 or abs(height - cell) > 2:
        raise ValueError("gold key does not match one grid cell")


def measure(image: Image.Image) -> dict:
    # int32 avoids overflow in the integer luminance coefficients below.
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    x0, y0, cell = _frame_geometry(array)
    _check_gold_key(array, cell)
    luminance = (
        299 * array[..., 0] + 587 * array[..., 1] + 114 * array[..., 2]
    ) // 1000
    counts = np.zeros((GRID_N, GRID_N), dtype=np.int32)
    for row in range(GRID_N):
        for col in range(GRID_N):
            patch = luminance[
                y0 + row * cell + 2 : y0 + (row + 1) * cell - 2,
                x0 + col * cell + 2 : x0 + (col + 1) * cell - 2,
            ]
            counts[row, col] = int((patch < 100).sum())

    ordered = np.sort(counts.ravel())
    gaps = np.diff(ordered)
    split_at = int(np.argmax(gaps))
    density_gap = int(gaps[split_at])
    if density_gap < 10 or split_at < 20 or split_at > 76:
        raise ValueError("cell-density bands are erased or ambiguous")
    threshold = (int(ordered[split_at]) + int(ordered[split_at + 1])) / 2.0
    cells = counts > threshold

    ranked: list[tuple[int, str]] = []
    for name, shape in _SHAPES.items():
        target = np.asarray(shape, dtype=bool)
        ranked.append((int(np.count_nonzero(cells != target)), name))
    ranked.sort()
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("glyph classification tie")
    return {
        "decision": ranked[0][1],
        "template_margin": ranked[1][0] - ranked[0][0],
        "density_gap_pixels": density_gap,
        "cell_px": cell,
        "high_cell_count": int(cells.sum()),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure(image)["decision"])
