"""Deterministic renderer and analytic arm for a local-average reveal."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = (720, 560)
GRID_N = 9
MICRO_N = 6
MICRO_COUNT = MICRO_N * MICRO_N
FRAME = (24, 162, 185)
GOLD = (222, 158, 35)
BACKGROUND = (248, 247, 242)
CELL_BACKGROUND = (239, 242, 239)
GRID_LINE = (198, 207, 207)
DARK_PALETTES = (
    (25, 39, 66),
    (60, 31, 69),
    (20, 61, 55),
)


def _blank() -> list[list[int]]:
    return [[0 for _ in range(GRID_N)] for _ in range(GRID_N)]


def _make_template(points: Iterable[tuple[int, int]]) -> tuple[tuple[int, ...], ...]:
    cells = _blank()
    for x, y in points:
        cells[y][x] = 1
    return tuple(tuple(row) for row in cells)


# All four public glyphs contain exactly eleven high-density cells. This makes
# total ink invariant to answer identity when nuisance parameters are matched.
TEMPLATES: dict[str, tuple[tuple[int, ...], ...]] = {
    "tee": _make_template(
        [(x, 1) for x in range(2, 7)] + [(4, y) for y in range(2, 8)]
    ),
    "corner": _make_template(
        [(2, y) for y in range(1, 8)] + [(x, 7) for x in range(3, 7)]
    ),
    "chevron": _make_template(
        [
            (1, 1), (7, 1),
            (1, 2), (7, 2),
            (2, 3), (6, 3),
            (2, 4), (6, 4),
            (3, 5), (5, 5),
            (4, 6),
        ]
    ),
    "fork": _make_template(
        [
            (1, 1), (7, 1),
            (2, 2), (6, 2),
            (3, 3), (5, 3),
        ] + [(4, y) for y in range(4, 9)]
    ),
}
LABELS = tuple(TEMPLATES)


def _distances(cells: list[list[int]] | tuple[tuple[int, ...], ...]) -> list[tuple[int, str]]:
    return sorted(
        (
            sum(
                int(bool(cells[y][x]) != bool(template[y][x]))
                for y in range(GRID_N)
                for x in range(GRID_N)
            ),
            name,
        )
        for name, template in TEMPLATES.items()
    )


def _template_decision(cells: list[list[int]] | tuple[tuple[int, ...], ...]) -> tuple[str, int]:
    ranked = _distances(cells)
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("glyph template tie")
    return ranked[0][1], ranked[1][0] - ranked[0][0]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; each adjacent quartet shares all nuisances."""
    label_index = int(seed) % len(LABELS)
    group = int(seed) // len(LABELS)
    nuisance = random.Random(group ^ 0x5A17C3)
    contrast_gap = (2, 4, 8, 12)[group % 4]
    low_count = (MICRO_COUNT - contrast_gap) // 2
    high_count = low_count + contrast_gap
    cell_px = (42, 45, 48)[(group // 3) % 3]
    return {
        "cells": [list(row) for row in TEMPLATES[LABELS[label_index]]],
        "texture_seed": nuisance.randrange(0, 2**31),
        "palette_index": group % len(DARK_PALETTES),
        "cell_px": cell_px,
        "origin_x": 48 + 4 * nuisance.randrange(0, 7),
        "origin_y": 46 + 4 * nuisance.randrange(0, 7),
        "low_count": low_count,
        "high_count": high_count,
    }


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    texture_seed = int(scene["texture_seed"])
    palette_index = int(scene["palette_index"])
    cell = int(scene["cell_px"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    low_count = int(scene["low_count"])
    high_count = int(scene["high_count"])
    if len(cells) != GRID_N or any(len(row) != GRID_N for row in cells):
        raise ValueError("cells must be a 9-by-9 matrix")
    if not (0 <= low_count < high_count <= MICRO_COUNT):
        raise ValueError("invalid microtexture density levels")
    if palette_index not in range(len(DARK_PALETTES)):
        raise ValueError("invalid palette_index")

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    span = GRID_N * cell
    border = 5
    draw.rectangle(
        (x0 - border, y0 - border, x0 + span + border - 1, y0 + span + border - 1),
        fill=FRAME,
    )
    draw.rectangle((x0, y0, x0 + span - 1, y0 + span - 1), fill=CELL_BACKGROUND)

    pitch = max(4, (cell - 8) // MICRO_N)
    mark = pitch - 1
    cluster = (MICRO_N - 1) * pitch + mark
    inset = (cell - cluster) // 2
    dark = DARK_PALETTES[palette_index]
    for row in range(GRID_N):
        for col in range(GRID_N):
            order = list(range(MICRO_COUNT))
            local_seed = texture_seed ^ (row * 1009 + col * 9176 + 0x6D2B79F5)
            random.Random(local_seed).shuffle(order)
            count = high_count if bool(cells[row][col]) else low_count
            left = x0 + col * cell + inset
            top = y0 + row * cell + inset
            for index in order[:count]:
                mx, my = index % MICRO_N, index // MICRO_N
                px = left + mx * pitch
                py = top + my * pitch
                draw.rectangle((px, py, px + mark - 1, py + mark - 1), fill=dark)

    for k in range(1, GRID_N):
        gx = x0 + k * cell
        gy = y0 + k * cell
        draw.line((gx, y0, gx, y0 + span - 1), fill=GRID_LINE, width=1)
        draw.line((x0, gy, x0 + span - 1, gy), fill=GRID_LINE, width=1)

    key_x = x0 + span + 66
    key_y = y0 + 2 * cell
    draw.rectangle((key_x, key_y, key_x + cell - 1, key_y + cell - 1), outline=GOLD, width=4)
    return image


def analytic_gold(scene: dict) -> str:
    return _template_decision(scene["cells"])[0]


def margin(scene: dict) -> float:
    _, template_gap = _template_decision(scene["cells"])
    density_gap = int(scene["high_count"]) - int(scene["low_count"])
    return float(min(template_gap, density_gap) * template_gap / max(template_gap, 1))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
