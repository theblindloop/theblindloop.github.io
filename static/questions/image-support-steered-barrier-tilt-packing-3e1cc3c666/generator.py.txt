"""Deterministic renderer and analytic arm for Barrier Tilt Packing."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SIZE = 5
CELL = 36
HEADER = 38
FOOTER = 24
PANEL_WIDTH = SIZE * CELL
PANEL_HEIGHT = HEADER + SIZE * CELL + FOOTER
GRID_PAD = 4
LETTERS = "ABCDE"
DIRECTIONS = ("left", "up", "right", "down")

PANEL = (43, 47, 55)
EMPTY = (248, 246, 238)
WALL = (184, 82, 76)
WALL_STRIPE = (246, 190, 118)
COMMAND_BG = (235, 220, 174)
COMMAND_INK = (91, 69, 132)
COMMAND_TIP = (216, 74, 63)
TRACK = (242, 157, 38)
SHADOW = (207, 203, 194)
BEAD_PALETTES = (
    (39, 119, 145),
    (54, 130, 94),
    (117, 83, 153),
    (183, 103, 43),
)
BACKGROUNDS = ((241, 238, 229), (235, 240, 237), (239, 235, 242))


def _tilt(grid: list[list[int]], direction: str) -> list[list[int]]:
    """Stably pack beads (1/2) inside wall-separated line segments."""

    output = [row[:] for row in grid]
    horizontal = direction in ("left", "right")
    toward_start = direction in ("left", "up")
    for fixed in range(SIZE):
        line = [grid[fixed][moving] if horizontal else grid[moving][fixed] for moving in range(SIZE)]
        walls = [index for index, value in enumerate(line) if value == 3]
        bounds = [-1, *walls, SIZE]
        packed = line[:]
        for start, stop in zip(bounds, bounds[1:]):
            cells = list(range(start + 1, stop))
            beads = [line[index] for index in cells if line[index] in (1, 2)]
            for index in cells:
                packed[index] = 0
            destinations = cells[: len(beads)] if toward_start else cells[len(cells) - len(beads) :]
            for index, bead in zip(destinations, beads):
                packed[index] = bead
        for moving, value in enumerate(packed):
            if horizontal:
                output[fixed][moving] = value
            else:
                output[moving][fixed] = value
    return output


def _replay(scene: dict) -> list[list[int]]:
    grid = [[int(value) for value in row] for row in scene["occupants"]]
    for direction in scene["commands"]:
        grid = _tilt(grid, str(direction))
    return grid


def _target_column(grid: list[list[int]]) -> int:
    positions = [(row, col) for row in range(SIZE) for col in range(SIZE) if grid[row][col] == 2]
    if len(positions) != 1:
        raise ValueError("grid must contain exactly one tracked bead")
    return positions[0][1]


def analytic_gold(scene: dict) -> str:
    return LETTERS[_target_column(_replay(scene))]


def _command_margin(scene: dict) -> int:
    baseline = analytic_gold(scene)
    commands = list(scene["commands"])
    for changed in range(1, len(commands) + 1):
        for indexes in itertools.combinations(range(len(commands)), changed):
            choices = [[item for item in DIRECTIONS if item != commands[index]] for index in indexes]
            for replacements in itertools.product(*choices):
                twin = dict(scene)
                alternate = commands[:]
                for index, replacement in zip(indexes, replacements):
                    alternate[index] = replacement
                twin["commands"] = alternate
                if analytic_gold(twin) != baseline:
                    return changed
    return len(commands) + 1


def _order_is_live(scene: dict) -> bool:
    baseline = analytic_gold(scene)
    commands = list(scene["commands"])
    for index in range(len(commands) - 1):
        twin = dict(scene)
        swapped = commands[:]
        swapped[index], swapped[index + 1] = swapped[index + 1], swapped[index]
        twin["commands"] = swapped
        if analytic_gold(twin) != baseline:
            return True
    return False


def sample_scene(seed: int) -> dict:
    """Sample fixed-count scenes with seed-mod-five answer balance."""

    seed = int(seed)
    desired = LETTERS[seed % len(LETTERS)]
    rng = random.Random((seed ^ 0xB4A2_19D7) & ((1 << 64) - 1))
    group = seed // len(LETTERS)
    for _ in range(2000):
        locations = list(range(SIZE * SIZE))
        rng.shuffle(locations)
        walls = set(locations[:4])
        bead_cells = locations[4:12]
        target_cell = bead_cells[rng.randrange(len(bead_cells))]
        occupants = [[0] * SIZE for _ in range(SIZE)]
        for location in walls:
            occupants[location // SIZE][location % SIZE] = 3
        for location in bead_cells:
            occupants[location // SIZE][location % SIZE] = 2 if location == target_cell else 1
        commands = list(DIRECTIONS)
        rng.shuffle(commands)
        scene = {
            "occupants": occupants,
            "commands": commands,
            "panel_x": 166 + ((group // 7) % 5) - 2,
            "panel_y": 135 + ((group // 35) % 3) - 1,
            "palette_variant": (group // 3) % len(BEAD_PALETTES),
            "background_variant": group % len(BACKGROUNDS),
        }
        if analytic_gold(scene) != desired:
            continue
        if _target_column(occupants) == _target_column(_replay(scene)):
            continue
        if _command_margin(scene) != 1 or not _order_is_live(scene):
            continue
        return scene
    raise RuntimeError(f"could not sample a balanced live packing scene for seed {seed}")


def margin(scene: dict) -> float:
    """Minimum visible command substitutions needed to change the final column."""

    return float(_command_margin(scene))


def is_quarantined(scene: dict) -> bool:
    try:
        grid = scene["occupants"]
        flat = [int(value) for row in grid for value in row]
        valid_grid = len(grid) == SIZE and all(len(row) == SIZE for row in grid)
        valid_counts = flat.count(0) == 13 and flat.count(1) == 7 and flat.count(2) == 1 and flat.count(3) == 4
        valid_commands = len(scene["commands"]) == 4 and set(scene["commands"]) == set(DIRECTIONS)
        panel_x, panel_y = int(scene["panel_x"]), int(scene["panel_y"])
        visible = 8 <= panel_x and panel_x + PANEL_WIDTH + 8 < WIDTH and 8 <= panel_y and panel_y + PANEL_HEIGHT + 8 < HEIGHT
        return not (valid_grid and valid_counts and valid_commands and visible and margin(scene) == 1.0)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, *, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _arrow_points(cx: int, cy: int, direction: str) -> list[tuple[int, int]]:
    right = [(-10, -3), (1, -3), (1, -8), (10, 0), (1, 8), (1, 3), (-10, 3)]
    if direction == "right":
        transform = lambda x, y: (x, y)
    elif direction == "left":
        transform = lambda x, y: (-x, y)
    elif direction == "down":
        transform = lambda x, y: (-y, x)
    elif direction == "up":
        transform = lambda x, y: (y, -x)
    else:
        raise ValueError(f"invalid direction {direction!r}")
    return [(cx + dx, cy + dy) for dx, dy in (transform(x, y) for x, y in right)]


def _tip(cx: int, cy: int, direction: str) -> tuple[int, int]:
    return {
        "left": (cx - 10, cy),
        "right": (cx + 10, cy),
        "up": (cx, cy - 10),
        "down": (cx, cy + 10),
    }[direction]


def render(scene: dict) -> Image.Image:
    occupants = scene["occupants"]
    commands = scene["commands"]
    x0, y0 = int(scene["panel_x"]), int(scene["panel_y"])
    bead = BEAD_PALETTES[int(scene["palette_variant"])]
    background = BACKGROUNDS[int(scene["background_variant"])]

    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0 + 6, y0 + 7, x0 + PANEL_WIDTH + 5, y0 + PANEL_HEIGHT + 6), fill=SHADOW)
    draw.rectangle((x0, y0, x0 + PANEL_WIDTH - 1, y0 + PANEL_HEIGHT - 1), fill=PANEL)

    for index, direction in enumerate(commands):
        left = x0 + 10 + index * 42
        top = y0 + 4
        draw.rectangle((left, top, left + 33, top + 29), fill=COMMAND_BG)
        cx, cy = left + 16, top + 14
        draw.polygon(_arrow_points(cx, cy, str(direction)), fill=COMMAND_INK)
        tx, ty = _tip(cx, cy, str(direction))
        draw.ellipse((tx - 2, ty - 2, tx + 2, ty + 2), fill=COMMAND_TIP)

    board_y = y0 + HEADER
    for row in range(SIZE):
        for col in range(SIZE):
            left = x0 + col * CELL + GRID_PAD
            top = board_y + row * CELL + GRID_PAD
            right = x0 + (col + 1) * CELL - GRID_PAD - 1
            bottom = board_y + (row + 1) * CELL - GRID_PAD - 1
            draw.rectangle((left, top, right, bottom), fill=EMPTY)
            value = int(occupants[row][col])
            cx = x0 + col * CELL + CELL // 2
            cy = board_y + row * CELL + CELL // 2
            if value == 3:
                draw.rounded_rectangle((cx - 11, cy - 11, cx + 11, cy + 11), radius=3, fill=WALL)
                for offset in (-8, 0, 8):
                    draw.line((cx - 10, cy + offset, cx + offset, cy - 10), fill=WALL_STRIPE, width=3)
                    draw.line((cx - offset, cy + 10, cx + 10, cy - offset), fill=WALL_STRIPE, width=3)
            elif value in (1, 2):
                if value == 2:
                    draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), fill=TRACK)
                    radius = 8
                else:
                    radius = 10
                draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=bead)

    footer_y = board_y + SIZE * CELL
    font = _font(14, bold=True)
    for col, label in enumerate(LETTERS):
        cx = x0 + col * CELL + CELL // 2
        draw.text((cx, footer_y + FOOTER // 2), label, font=font, fill=EMPTY, anchor="mm")
    return image
