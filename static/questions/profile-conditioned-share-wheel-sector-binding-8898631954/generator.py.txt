"""Share-of-total correspondence world: colored bars versus a glyph-tagged wheel.

A scene holds four upright bars with integer pixel heights and four wheel
sectors with integer degree spans that sum to a full turn. The rendered wheel is
cut by dark spokes at the cumulative span angles, and a black glyph tag sits just
outside the middle of each sector.

The intended reading is a change of representation under normalization: a bar's
share of the total height of all four bars equals its sector's share of the full
turn. Absolute lengths carry no information because no unit is drawn, so the only
usable invariant is the share. The decision is the glyph tagging the sector that
matches the red bar.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "share_wheel_sector_binding-renderer-0.1.0"

CANVAS = (520, 620)
BACKGROUND = (250, 250, 250)
INK = (0, 0, 0)
WHEEL_FILL = (205, 205, 205)
SPOKE = (70, 70, 70)

BAR_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (214, 45, 40),
    "blue": (40, 80, 200),
    "green": (30, 150, 70),
    "orange": (235, 140, 25),
}
COLOR_ORDER = ("red", "blue", "green", "orange")
GLYPHS = ("square", "circle", "triangle", "diamond")

TARGET_COLOR = "red"
QUARANTINE_MARGIN_DEG = 8.0

BASELINE_Y = 250
BAR_SLOT_X = (95, 185, 275, 365)
BAR_HALF_WIDTH = 23
DIVIDER_Y = 275
WHEEL_CENTER = (260, 440)
WHEEL_RADIUS = 100
GLYPH_RING = 118
SPOKE_WIDTH = 5
OUTLINE_WIDTH = 4

# Minimum pairwise degree gap between sector spans, by difficulty band. The last
# band is deliberately inside the quarantine region.
BANDS = (
    ("wide", 25, 60),
    ("medium", 16, 24),
    ("near", 11, 15),
    ("wide", 25, 60),
    ("medium", 16, 24),
    ("tight", 4, 7),
)


def _spans(rng: random.Random, low_gap: int, high_gap: int) -> list[int] | None:
    """Four integer spans summing to 360 with a pairwise gap inside the band."""
    for _ in range(400):
        cuts = sorted(rng.sample(range(20, 341), 3))
        spans = [cuts[0], cuts[1] - cuts[0], cuts[2] - cuts[1], 360 - cuts[2]]
        if min(spans) < 45 or max(spans) > 140:
            continue
        ordered = sorted(spans)
        gaps = [ordered[i + 1] - ordered[i] for i in range(3)]
        if low_gap <= min(gaps) <= high_gap:
            rng.shuffle(spans)
            return spans
    return None


def _ranked(scene: dict) -> list[tuple[float, int]]:
    """Absolute share mismatch, in degrees, between the red bar and each sector.

    Reads only the bar heights and colors and the sector spans, so the answer is
    a consequence of visible extents alone.
    """
    bars = scene["bars"]
    sectors = scene["sectors"]
    total = float(sum(bar["height"] for bar in bars))
    target = [bar for bar in bars if bar["color"] == TARGET_COLOR][0]
    wanted = 360.0 * float(target["height"]) / total
    scored = [
        (abs(wanted - float(sector["span"])), index)
        for index, sector in enumerate(sectors)
    ]
    scored.sort(key=lambda item: (item[0], item[1]))
    return scored


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene from an integer seed."""
    rng = random.Random(seed * 7919 + 13)
    band_name, low_gap, high_gap = BANDS[seed % len(BANDS)]
    forced_glyph = GLYPHS[seed % len(GLYPHS)]

    for _ in range(600):
        spans = _spans(rng, low_gap, high_gap)
        if spans is None:
            continue
        scale = rng.uniform(0.86, 1.02)
        heights = [int(round(span * scale)) for span in spans]
        if len(set(heights)) < 4:
            continue

        # Colors are attached to spans, and slots are shuffled separately, so the
        # answer is uncorrelated with the red bar's position, rank, or size.
        color_perm = list(COLOR_ORDER)
        rng.shuffle(color_perm)
        slots = list(BAR_SLOT_X)
        rng.shuffle(slots)
        bars = [
            {"color": color_perm[i], "x": slots[i], "height": heights[i]}
            for i in range(4)
        ]
        target_index = color_perm.index(TARGET_COLOR)

        # Sector order is an independent permutation of the same span multiset.
        sector_perm = list(range(4))
        rng.shuffle(sector_perm)
        others = [glyph for glyph in GLYPHS if glyph != forced_glyph]
        rng.shuffle(others)
        glyph_stream = iter(others)
        sectors = []
        for span_index in sector_perm:
            glyph = forced_glyph if span_index == target_index else next(glyph_stream)
            sectors.append({"span": spans[span_index], "glyph": glyph})

        scene = {
            "bars": bars,
            "sectors": sectors,
            "start_angle": rng.randrange(0, 360),
            "wheel": {
                "cx": WHEEL_CENTER[0],
                "cy": WHEEL_CENTER[1],
                "r": WHEEL_RADIUS,
                "ring": GLYPH_RING,
            },
            "layout": {
                "band": band_name,
                "baseline_y": BASELINE_Y,
                "divider_y": DIVIDER_Y,
                "half_width": BAR_HALF_WIDTH,
            },
        }

        scored = _ranked(scene)
        best_span = sectors[scored[0][1]]["span"]
        gap = scored[1][0] - scored[0][0]
        # The intended sector must win outright, the pixel arm must be able to
        # resolve the gap, and the declared band must be honored.
        if best_span != spans[target_index]:
            continue
        # The band directly controls the runner-up margin, so every seed block
        # covers wide, medium, near-boundary and quarantined cases.
        if not (float(low_gap) <= gap <= float(high_gap)):
            continue
        if gap < 2.0:
            continue
        if band_name == "tight" and gap >= QUARANTINE_MARGIN_DEG:
            continue
        if band_name != "tight" and gap < QUARANTINE_MARGIN_DEG:
            continue
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _glyph(draw: ImageDraw.ImageDraw, glyph: str, cx: float, cy: float) -> None:
    x, y = cx, cy
    if glyph == "square":
        draw.rectangle([x - 8, y - 8, x + 8, y + 8], fill=INK)
    elif glyph == "circle":
        draw.ellipse([x - 9, y - 9, x + 9, y + 9], fill=INK)
    elif glyph == "triangle":
        draw.polygon([(x, y - 10), (x - 11, y + 9), (x + 11, y + 9)], fill=INK)
    elif glyph == "diamond":
        draw.polygon([(x, y - 11), (x + 11, y), (x, y + 11), (x - 11, y)], fill=INK)
    else:
        raise ValueError(f"unknown glyph {glyph!r}")


def render(scene: dict) -> Image.Image:
    """Draw the bar panel above and the glyph-tagged sector wheel below."""
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)

    layout = scene["layout"]
    baseline = int(layout["baseline_y"])
    half = int(layout["half_width"])
    for bar in scene["bars"]:
        left = int(bar["x"]) - half
        top = baseline - int(bar["height"])
        draw.rectangle(
            [left, top, left + 2 * half, baseline - 1], fill=BAR_COLORS[bar["color"]]
        )
    draw.rectangle([50, baseline, 470, baseline + 3], fill=INK)
    draw.rectangle([0, int(layout["divider_y"]), CANVAS[0] - 1, int(layout["divider_y"]) + 2], fill=INK)

    wheel = scene["wheel"]
    cx, cy = float(wheel["cx"]), float(wheel["cy"])
    radius = float(wheel["r"])
    ring = float(wheel["ring"])
    draw.ellipse([cx - radius, cy - radius, cx + radius, cy + radius], fill=WHEEL_FILL)
    draw.ellipse(
        [cx - radius, cy - radius, cx + radius, cy + radius],
        outline=SPOKE,
        width=OUTLINE_WIDTH,
    )

    start = float(scene["start_angle"])
    cumulative = 0.0
    for sector in scene["sectors"]:
        span = float(sector["span"])
        edge = (start + cumulative) % 360.0
        mid = (start + cumulative + span / 2.0) % 360.0
        ex = cx + radius * math.cos(math.radians(edge))
        ey = cy - radius * math.sin(math.radians(edge))
        draw.line([cx, cy, ex, ey], fill=SPOKE, width=SPOKE_WIDTH)
        gx = cx + ring * math.cos(math.radians(mid))
        gy = cy - ring * math.sin(math.radians(mid))
        _glyph(draw, sector["glyph"], gx, gy)
        cumulative += span
    return image


def analytic_gold(scene: dict) -> str:
    """Glyph tagging the sector whose turn share matches the red bar's share."""
    scored = _ranked(scene)
    return str(scene["sectors"][scored[0][1]]["glyph"])


def margin(scene: dict) -> float:
    """Degrees between the winning sector span and the runner-up span."""
    scored = _ranked(scene)
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene: dict) -> bool:
    """Finite-resolution boundary cases: the runner-up gap is under 8 degrees."""
    return bool(margin(scene) < QUARANTINE_MARGIN_DEG)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Render-invariant relabelings of the bookkeeping order in the scene."""
    twins: list[tuple[str, dict]] = []

    rotated = dict(scene)
    head = scene["sectors"][0]
    rotated["sectors"] = list(scene["sectors"][1:]) + [head]
    rotated["start_angle"] = (float(scene["start_angle"]) + float(head["span"])) % 360.0
    twins.append(("sector_list_origin_shift", rotated))

    reversed_bars = dict(scene)
    reversed_bars["bars"] = list(reversed(scene["bars"]))
    twins.append(("bar_list_order_reversed", reversed_bars))

    rolled_bars = dict(scene)
    rolled_bars["bars"] = list(scene["bars"][2:]) + list(scene["bars"][:2])
    twins.append(("bar_list_order_rolled", rolled_bars))
    return twins
