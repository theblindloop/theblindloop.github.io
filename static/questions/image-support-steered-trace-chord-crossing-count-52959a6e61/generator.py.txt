"""Deterministic latent sampler and Pillow renderer for the chord-crossing world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
PALETTE = ("#12A4A6", "#7A5AF8", "#D56A35", "#2F80ED")
CORE = "#162F3D"
DISTRACTOR = "#D5DBE1"


def _signed_distance(point: Iterable[float], start: Iterable[float], end: Iterable[float]) -> float:
    px, py = point
    ax, ay = start
    bx, by = end
    dx, dy = bx - ax, by - ay
    return ((px - ax) * dy - (py - ay) * dx) / math.hypot(dx, dy)


def _crossing_positions(points: list[list[int]]) -> list[float]:
    start, end = points[0], points[-1]
    signed = [_signed_distance(point, start, end) for point in points]
    crossings: list[float] = []
    for index in range(1, len(points) - 2):
        left, right = signed[index], signed[index + 1]
        if left * right < 0:
            fraction = abs(left) / (abs(left) + abs(right))
            crossings.append(points[index][0] + fraction * (points[index + 1][0] - points[index][0]))
    return crossings


def _decision(points: list[list[int]]) -> str:
    return str(len(_crossing_positions(points)))


def _margin_components(points: list[list[int]]) -> tuple[float, float]:
    start, end = points[0], points[-1]
    clearances = [abs(_signed_distance(point, start, end)) for point in points[1:-1]]
    crossings = _crossing_positions(points)
    spacing = min((b - a for a, b in zip(crossings, crossings[1:])), default=999.0)
    return min(clearances), spacing


def _sign_pattern(rng: random.Random, crossing_count: int, size: int = 7) -> list[int]:
    # Positive integer run lengths summing to size create exactly k sign changes.
    cuts = sorted(rng.sample(range(1, size), crossing_count))
    bounds = [0, *cuts, size]
    first = rng.choice((-1, 1))
    signs: list[int] = []
    for run, (left, right) in enumerate(zip(bounds, bounds[1:])):
        signs.extend([first * (-1 if run % 2 else 1)] * (right - left))
    return signs


def _distractors(rng: random.Random, points: list[list[int]]) -> list[list[int]]:
    x0, x1 = points[0][0] - 35, points[-1][0] + 35
    y0 = min(p[1] for p in points) - 35
    y1 = max(p[1] for p in points) + 35
    values: list[list[int]] = []
    for _ in range(500):
        if len(values) == 5:
            break
        x, y = rng.randint(28, WIDTH - 28), rng.randint(28, HEIGHT - 28)
        radius = rng.choice((3, 4, 5))
        if x0 <= x <= x1 and y0 <= y <= y1:
            continue
        if any((x - ox) ** 2 + (y - oy) ** 2 < 35 ** 2 for ox, oy, _ in values):
            continue
        values.append([x, y, radius])
    if len(values) != 5:
        raise RuntimeError("could not place neutral distractors")
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = 1 + (int(seed) % 3)
    pair_column = rng.randint(1, 5)
    row = rng.randint(1, 6)
    x_start = pair_column * 80 + 24
    x_end = (pair_column + 2) * 80 - 24
    baseline = row * 80 + 40
    endpoint_left = rng.randint(-3, 3)
    endpoint_right = rng.randint(-3, 3)
    xs = [round(x_start + i * (x_end - x_start) / 8) for i in range(9)]
    signs = _sign_pattern(rng, target, size=7)
    deviations = [sign * rng.randint(8, 14) for sign in signs]
    points: list[list[int]] = [[xs[0], baseline + endpoint_left]]
    for i, deviation in enumerate(deviations, start=1):
        chord_y = endpoint_left + (endpoint_right - endpoint_left) * i / 8
        # Positive mathematical signed distance corresponds to a negative page-y offset.
        points.append([xs[i], round(baseline + chord_y - deviation)])
    points.append([xs[-1], baseline + endpoint_right])
    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "#FFFFFF",
        "points": points,
        "trace_color": PALETTE[rng.randrange(len(PALETTE))],
        "stroke_width": rng.choice((7, 9)),
        "distractors": _distractors(rng, points),
    }
    if _decision(points) != str(target) or is_quarantined(scene):
        raise AssertionError("constructed scene violated its target class or clearance")
    return scene


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), scene["background"])
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["distractors"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=DISTRACTOR)
    points = [tuple(point) for point in scene["points"]]
    width = int(scene["stroke_width"])
    draw.line(points, fill=scene["trace_color"], width=width, joint="curve")
    draw.line(points, fill=CORE, width=3, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    return _decision(scene["points"])


def margin(scene: dict) -> float:
    clearance, spacing = _margin_components(scene["points"])
    return float(min(clearance, spacing / 2.0))


def is_quarantined(scene: dict) -> bool:
    clearance, spacing = _margin_components(scene["points"])
    return clearance < 6.0 or spacing < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_unordered_distractors", twin)]
