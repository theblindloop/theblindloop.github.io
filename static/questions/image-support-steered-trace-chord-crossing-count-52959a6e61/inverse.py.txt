"""Independent pixel-only inverse for the mini-trace crossing count."""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _centerline(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # The trace's dark neutral core is the only low-luminance component; outer
    # palette and pale distractors are deliberately irrelevant to this mask.
    mask = (array.max(axis=2) < 90) & (array.min(axis=2) < 45)
    ys, xs = np.nonzero(mask)
    if len(xs) < 220:
        raise OracleAbstention("insufficient connected trace core")
    unique_x = np.unique(xs)
    if unique_x[-1] - unique_x[0] < 95:
        raise OracleAbstention("trace core is truncated")
    if len(unique_x) != unique_x[-1] - unique_x[0] + 1:
        raise OracleAbstention("trace core has a horizontal gap")
    centers = np.array([np.median(ys[xs == x]) for x in unique_x], dtype=np.float64)
    # A short symmetric filter suppresses raster stair-step noise without using
    # any renderer coordinates or latent values.
    padded = np.pad(centers, (2, 2), mode="edge")
    smooth = np.convolve(padded, np.ones(5) / 5.0, mode="valid")
    return unique_x.astype(np.float64), smooth


def decision_from_image(image: Image.Image) -> str:
    xs, ys = _centerline(image)
    chord = ys[0] + (ys[-1] - ys[0]) * (xs - xs[0]) / (xs[-1] - xs[0])
    residual = ys - chord
    # Exclude caps, compress near-zero raster samples, and count robust sign runs.
    usable = residual[5:-5]
    strong = usable[np.abs(usable) >= 2.5]
    if len(strong) < 30:
        raise OracleAbstention("no stable lobes away from endpoint chord")
    signs = np.sign(strong).astype(int)
    runs = [int(signs[0])]
    for sign in signs[1:]:
        if int(sign) != runs[-1]:
            runs.append(int(sign))
    crossings = len(runs) - 1
    if crossings not in (1, 2, 3):
        raise OracleAbstention(f"unsupported crossing count {crossings}")
    return str(crossings)
