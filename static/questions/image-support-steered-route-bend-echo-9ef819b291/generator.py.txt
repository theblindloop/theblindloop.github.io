"""Deterministic renderer for a global bend-order comparison on one route."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw


INKS = ["#176f9f", "#694aa8", "#a23f68", "#167b68", "#9a5b18"]
BACKGROUNDS = ["#f8f6ef", "#f3f7f5", "#f7f4f8"]
MARKERS = ["#45464d", "#55483e", "#394c4b"]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Four visibly distinct magnitudes, then a free permutation. Both labels use
    # exactly the same multiset, positions, endpoints, and total curve length.
    amplitudes = [rng.uniform(25.0 + 13.0 * i, 29.0 + 13.0 * i) for i in range(4)]
    rng.shuffle(amplitudes)
    left_signs = [rng.choice([-1, 1]) for _ in range(4)]
    right_signs = [rng.choice([-1, 1]) for _ in range(4)]
    return {
        "width": 512,
        "height": 384,
        "mid_x": rng.uniform(249.0, 263.0),
        "baseline": rng.uniform(181.0, 203.0),
        "half_span": rng.uniform(218.0, 224.0),
        "amplitudes": [round(value, 6) for value in amplitudes],
        "left_signs": left_signs,
        "right_signs": right_signs,
        "relation": "same" if int(seed) % 2 == 0 else "reverse",
        "tilt": rng.uniform(-13.0, 13.0),
        "stroke": rng.uniform(7.0, 9.0),
        "background": rng.choice(BACKGROUNDS),
        "ink": rng.choice(INKS),
        "marker": rng.choice(MARKERS),
    }


def _rgb(value: str) -> tuple[int, int, int]:
    value = value.lstrip("#")
    return tuple(int(value[i:i + 2], 16) for i in (0, 2, 4))


def _route_points(scene: dict, supersample: int) -> list[tuple[float, float]]:
    mid = float(scene["mid_x"])
    span = float(scene["half_span"])
    baseline = float(scene["baseline"])
    tilt = float(scene["tilt"])
    amplitudes = [float(value) for value in scene["amplitudes"]]
    left_signs = [int(value) for value in scene["left_signs"]]
    right_signs = [int(value) for value in scene["right_signs"]]
    relation = str(scene["relation"])
    if relation not in {"same", "reverse"}:
        raise ValueError("relation must be same or reverse")
    right_amplitudes = amplitudes if relation == "same" else amplitudes[::-1]
    all_amplitudes = amplitudes + right_amplitudes
    all_signs = left_signs + right_signs
    points = []
    samples_per_bend = 32
    for bend in range(8):
        for step in range(samples_per_bend):
            u = (bend + step / samples_per_bend) / 8.0
            local = step / samples_per_bend
            x = mid - span + 2.0 * span * u
            base_y = baseline + tilt * ((x - mid) / span)
            y = base_y + all_signs[bend] * all_amplitudes[bend] * math.sin(math.pi * local)
            points.append((x * supersample, y * supersample))
    x = mid + span
    points.append((x * supersample, (baseline + tilt) * supersample))
    return points


def render(scene: dict) -> Image.Image:
    # Every declared field is read here for the rendered-field audit.
    width, height = int(scene["width"]), int(scene["height"])
    mid, baseline = float(scene["mid_x"]), float(scene["baseline"])
    _ = (float(scene["half_span"]), list(scene["amplitudes"]), list(scene["left_signs"]),
         list(scene["right_signs"]), str(scene["relation"]), float(scene["tilt"]))
    stroke = float(scene["stroke"])
    background, ink, marker = scene["background"], scene["ink"], scene["marker"]
    ss = 3
    image = Image.new("RGB", (width * ss, height * ss), _rgb(background))
    draw = ImageDraw.Draw(image)
    points = _route_points(scene, ss)
    # Paint the neutral midpoint ring first, then carry the saturated route
    # continuously across it so the marker cannot sever the decisive trace.
    ring_r = 12.5 * ss
    ring_w = max(3, round(2.2 * ss))
    draw.ellipse((mid * ss - ring_r, baseline * ss - ring_r,
                  mid * ss + ring_r, baseline * ss + ring_r),
                 outline=_rgb(marker), width=ring_w)
    draw.line(points, fill=_rgb(ink), width=round(stroke * ss), joint="curve")
    radius = stroke * ss / 2.0
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=_rgb(ink))
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["relation"])


def margin(scene: dict) -> float:
    values = [float(value) for value in scene["amplitudes"]]
    return round(math.sqrt(sum((a - b) ** 2 for a, b in zip(values, values[::-1])) / 4.0), 9)


def is_quarantined(scene: dict) -> bool:
    values = sorted(float(value) for value in scene["amplitudes"])
    separation = min(b - a for a, b in zip(values, values[1:]))
    return separation < 8.0 or margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
