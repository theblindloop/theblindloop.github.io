"""Independent pixel-only inverse for the paired histogram question."""

from __future__ import annotations

from PIL import Image


WIDTH = 1200
HEIGHT = 700
BINS = 7
COUNT_SCALE = 18
BASELINE = 520
PLOT_TOP = 235
BAR_WIDTH = 38
BAR_STEP = 50
BAR_X0 = (155, 725)
QUARANTINE_GAP = 0.10


def _is_colored(pixel: tuple[int, int, int]) -> bool:
    """Recognize saturated bar ink while rejecting the grayscale chart furniture."""

    return max(pixel) - min(pixel) >= 40 and max(pixel) >= 75


def _longest_run(values: list[int]) -> int:
    best = 0
    current = 0
    previous = None
    for value in values:
        if previous is not None and value == previous + 1:
            current += 1
        else:
            current = 1
        best = max(best, current)
        previous = value
    return best


def _decode_panel(image: Image.Image, x0: int) -> list[int]:
    counts: list[int] = []
    for index in range(BINS):
        center_x = x0 + index * BAR_STEP + BAR_WIDTH // 2
        colored_rows = [
            y
            for y in range(PLOT_TOP, BASELINE)
            if _is_colored(image.getpixel((center_x, y)))
        ]
        height = _longest_run(colored_rows)
        if height < COUNT_SCALE * 2:
            raise ValueError("a histogram bar is missing or too short")
        count = round(height / COUNT_SCALE)
        if count < 1 or abs(height - count * COUNT_SCALE) > 2:
            raise ValueError("bar height is not stable on the declared count scale")
        counts.append(int(count))
    return counts


def _stat(counts: list[int]) -> float:
    total = sum(counts)
    if len(counts) != BINS or total <= 0:
        raise ValueError("decoded histogram is invalid")
    weighted_mean = sum((index + 1) * value for index, value in enumerate(counts)) / total
    cumulative = 0
    for index, value in enumerate(counts, start=1):
        cumulative += value
        if cumulative * 2 >= total:
            return weighted_mean - float(index)
    raise ValueError("decoded histogram has no median bin")


def decision_from_image(image: Image.Image) -> str:
    """Recover A, B, or tie from the final raster alone."""

    if image.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected image size {image.size}")
    counts_a = _decode_panel(image, BAR_X0[0])
    counts_b = _decode_panel(image, BAR_X0[1])
    score_a = _stat(counts_a)
    score_b = _stat(counts_b)
    gap = score_a - score_b
    if abs(gap) < 1e-12:
        return "tie"
    return "A" if gap > 0 else "B"
