"""Deterministic renderer and analytic side for paired histogram asymmetry."""

from __future__ import annotations

import random
from typing import Sequence

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "histogram_skew_compare-renderer-0.1.0"
WIDTH = 1200
HEIGHT = 700
BINS = 7
COUNT_SCALE = 18
MAX_COUNT = 15
BASELINE = 520
PLOT_TOP = 235
BAR_WIDTH = 38
BAR_STEP = 50
BAR_X0 = {"A": 155, "B": 725}
AXIS_X = {"A": 120, "B": 690}
PLOT_RIGHT = {"A": 560, "B": 1130}
PANEL_BOXES = {"A": (45, 55, 580, 620), "B": (620, 55, 1155, 620)}
QUARANTINE_GAP = 0.10

BAR_COLORS = (
    (37, 105, 173),
    (205, 91, 64),
    (55, 145, 104),
    (132, 83, 173),
    (205, 135, 45),
    (33, 149, 151),
)
_LABEL_FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
_TEXT_FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(path: str, size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _stat(counts: Sequence[int]) -> float:
    """Weighted mean bin number minus first-cumulative-half median bin."""

    values = [int(value) for value in counts]
    total = sum(values)
    if len(values) != BINS or total <= 0:
        raise ValueError("a histogram must contain seven positive counts")
    weighted_mean = sum((index + 1) * value for index, value in enumerate(values)) / total
    cumulative = 0
    median_bin = BINS
    for index, value in enumerate(values, start=1):
        cumulative += value
        if cumulative * 2 >= total:
            median_bin = index
            break
    return weighted_mean - float(median_bin)


def _sample_pair(rng: random.Random, near: bool) -> tuple[list[int], list[int]]:
    """Find two permutations of one count multiset with a controlled gap."""

    for _ in range(120):
        base = sorted(rng.sample(range(3, MAX_COUNT + 1), BINS))
        candidates: list[tuple[float, list[int]]] = []
        for _ in range(260):
            values = list(base)
            rng.shuffle(values)
            candidates.append((_stat(values), values))
        candidates.sort(key=lambda item: item[0])
        if near:
            for low_index, (low_score, low_values) in enumerate(candidates):
                for high_score, high_values in candidates[low_index + 1 : low_index + 90]:
                    gap = high_score - low_score
                    if 0.012 <= gap < QUARANTINE_GAP:
                        return list(low_values), list(high_values)
        else:
            low_score, low_values = candidates[0]
            high_score, high_values = candidates[-1]
            if high_score - low_score >= 0.32:
                return list(low_values), list(high_values)
    raise RuntimeError("could not sample a controlled histogram asymmetry pair")


def sample_scene(seed: int) -> dict:
    """Sample matched count multisets, balanced winning side, and nuisance colors."""

    rng = random.Random(int(seed))
    near = int(seed) % 13 == 0
    low, high = _sample_pair(rng, near=near)
    if rng.randrange(2) == 0:
        counts_a, counts_b = high, low
    else:
        counts_a, counts_b = low, high
    color_a = list(rng.choice(BAR_COLORS))
    color_b = list(rng.choice(tuple(color for color in BAR_COLORS if color != tuple(color_a))))
    return {
        "counts_a": [int(value) for value in counts_a],
        "counts_b": [int(value) for value in counts_b],
        "color_a": color_a,
        "color_b": color_b,
    }


def _centered_text(
    draw: ImageDraw.ImageDraw,
    center_x: float,
    y: int,
    text: str,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((round(center_x - (box[2] - box[0]) / 2), y), text, fill=fill, font=font)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel: str,
    counts: Sequence[int],
    color: Sequence[int],
    label_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    text_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
) -> None:
    px0, py0, px1, py1 = PANEL_BOXES[panel]
    axis_x = AXIS_X[panel]
    bar_x0 = BAR_X0[panel]
    draw.rounded_rectangle(
        (px0, py0, px1, py1),
        radius=18,
        fill=(249, 251, 254),
        outline=(190, 199, 211),
        width=2,
    )
    _centered_text(draw, (px0 + px1) / 2, 78, f"Panel {panel}", label_font, (35, 47, 62))
    _centered_text(
        draw,
        (px0 + px1) / 2,
        116,
        "seven-bin count distribution",
        text_font,
        (83, 96, 111),
    )
    draw.line((axis_x, PLOT_TOP, axis_x, BASELINE), fill=(64, 75, 88), width=3)
    draw.line((axis_x, BASELINE, PLOT_RIGHT[panel], BASELINE), fill=(64, 75, 88), width=3)
    for tick in (0, 5, 10, 15):
        y = BASELINE - tick * COUNT_SCALE
        draw.line((axis_x - 7, y, axis_x, y), fill=(64, 75, 88), width=2)
        label_box = draw.textbbox((0, 0), str(tick), font=text_font)
        draw.text(
            (axis_x - 15 - (label_box[2] - label_box[0]), y - 8),
            str(tick),
            fill=(83, 96, 111),
            font=text_font,
        )
    draw.text((axis_x - 7, PLOT_TOP - 28), "count", fill=(83, 96, 111), font=text_font)
    for index, count in enumerate(counts):
        left = bar_x0 + index * BAR_STEP
        right = left + BAR_WIDTH
        height = int(count) * COUNT_SCALE
        draw.rectangle(
            (left, BASELINE - height, right, BASELINE - 1),
            fill=tuple(int(value) for value in color),
        )
        _centered_text(draw, (left + right) / 2, BASELINE + 16, str(index + 1), text_font, (83, 96, 111))
    _centered_text(
        draw,
        (px0 + px1) / 2,
        575,
        "bar height = count; bin centers are 1 through 7",
        text_font,
        (83, 96, 111),
    )


def render(scene: dict) -> Image.Image:
    """Render the visible paired histograms as a crisp RGB raster."""

    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    label_font = _font(_LABEL_FONT, 26)
    text_font = _font(_TEXT_FONT, 15)
    _draw_panel(draw, "A", scene["counts_a"], scene["color_a"], label_font, text_font)
    _draw_panel(draw, "B", scene["counts_b"], scene["color_b"], label_font, text_font)
    return image


def analytic_gold(scene: dict) -> str:
    """Return the panel with the larger mean-minus-median statistic."""

    score_a = _stat(scene["counts_a"])
    score_b = _stat(scene["counts_b"])
    if abs(score_a - score_b) < 1e-12:
        return "tie"
    return "A" if score_a > score_b else "B"


def margin(scene: dict) -> float:
    """Signed statistic difference: panel A minus panel B."""

    return _stat(scene["counts_a"]) - _stat(scene["counts_b"])


def is_quarantined(scene: dict) -> bool:
    """Exclude exact and finite-resolution near-equality cases."""

    return abs(margin(scene)) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """This injective parametrization declares no render-invariant aliases."""

    return []
