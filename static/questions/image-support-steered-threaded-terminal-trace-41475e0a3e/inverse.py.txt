"""Independent pixel-only inverse arm for threaded_terminal_trace.

This module deliberately knows only public palette classes and pixel topology.
It does not import or inspect renderer state.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "orange": np.array([235, 104, 35], dtype=np.int16),
    "blue": np.array([35, 105, 210], dtype=np.int16),
    "green": np.array([38, 157, 91], dtype=np.int16),
}


def _near(array: np.ndarray, color: np.ndarray, radius: float = 52.0) -> np.ndarray:
    # int32 is intentional: squared RGB deltas overflow int16.
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int64) <= radius * radius


def _dilate(mask: np.ndarray, steps: int) -> np.ndarray:
    out = mask.copy()
    for _ in range(steps):
        grown = out.copy()
        grown[1:] |= out[:-1]
        grown[:-1] |= out[1:]
        grown[:, 1:] |= out[:, :-1]
        grown[:, :-1] |= out[:, 1:]
        grown[1:, 1:] |= out[:-1, :-1]
        grown[:-1, :-1] |= out[1:, 1:]
        grown[1:, :-1] |= out[:-1, 1:]
        grown[:-1, 1:] |= out[1:, :-1]
        out = grown
    return out


def _component(mask: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    reached = np.zeros_like(mask)
    queue: deque[tuple[int, int]] = deque()
    for y, x in np.argwhere(mask & seeds):
        reached[y, x] = True
        queue.append((int(y), int(x)))
    if not queue:
        raise ValueError("no black filament is incident to the orange marker")
    while queue:
        y, x = queue.popleft()
        for ny in range(max(0, y - 1), min(height, y + 2)):
            for nx in range(max(0, x - 1), min(width, x + 2)):
                if mask[ny, nx] and not reached[ny, nx]:
                    reached[ny, nx] = True
                    queue.append((ny, nx))
    return reached


def decision_from_image(image: Image.Image) -> str:
    """Return the terminal color incident to the orange-connected black trace."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    orange = _near(array, PALETTE["orange"])
    blue = _near(array, PALETTE["blue"])
    green = _near(array, PALETTE["green"])
    for name, mask in (("orange", orange), ("blue", blue), ("green", green)):
        if int(mask.sum()) < 80:
            raise ValueError(f"missing or damaged {name} marker")
    black = np.max(array, axis=2) < 82
    route = _component(black, _dilate(orange, 4))
    touches = []
    for name, marker in (("blue", blue), ("green", green)):
        contact = route & _dilate(marker, 4)
        if int(contact.sum()) >= 2:
            touches.append(name)
    if len(touches) != 1:
        raise ValueError(f"orange component reaches {touches}, expected exactly one terminal")
    return touches[0]
