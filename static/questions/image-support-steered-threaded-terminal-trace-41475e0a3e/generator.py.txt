"""Deterministic scene sampler and renderer for threaded_terminal_trace."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

ORANGE = (235, 104, 35)
BLUE = (35, 105, 210)
GREEN = (38, 157, 91)
GRAY = (145, 145, 150)
INK = (24, 27, 31)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, movable, compact tangle from one integer seed."""
    rng = random.Random(int(seed))
    target_terminal = "blue" if int(seed) % 2 == 0 else "green"
    terminal_order = ["blue", "green"] if (int(seed) // 2) % 2 == 0 else ["green", "blue"]
    origin_x = rng.choice([86, 150, 214])
    origin_y = rng.choice([84, 148, 212, 276, 340])
    terminal_y = {terminal_order[0]: 0, terminal_order[1]: 24}
    end_y = terminal_y[target_terminal]
    other_y = 24 - end_y
    jitter = rng.choice([-2, -1, 0, 1, 2])
    target_points = [
        [0, other_y], [20, other_y], [40, other_y + jitter],
        [60, 12], [81, end_y - jitter], [101, end_y], [120, end_y],
    ]
    distractor_points = [
        [0, end_y], [19, end_y], [39, end_y - jitter],
        [60, 12], [82, other_y + jitter], [102, other_y], [120, other_y],
    ]
    return {
        "width": 512,
        "height": 512,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "target_terminal": target_terminal,
        "terminal_order": terminal_order,
        "target_points": target_points,
        "distractor_points": distractor_points,
        "background": [248, 248, 245],
    }


def _points(scene: dict, key: str, scale: int) -> list[tuple[int, int]]:
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    return [((ox + int(x)) * scale, (oy + int(y)) * scale) for x, y in scene[key]]


def _disk(draw: ImageDraw.ImageDraw, xy: tuple[int, int], radius: int, color: tuple[int, int, int]) -> None:
    x, y = xy
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    """Render two crossing strands; the orange strand is visibly overpassing."""
    scale = 4
    target_terminal = str(scene["target_terminal"])
    if target_terminal not in {"blue", "green"}:
        raise ValueError("target_terminal must name a visible colored terminal")
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    target = _points(scene, "target_points", scale)
    distractor = _points(scene, "distractor_points", scale)

    # Under-strand first. The wide background halo on the target is the explicit
    # bridge convention: it prevents an accidental junction at the crossing.
    draw.line(distractor, fill=INK, width=6 * scale, joint="curve")
    draw.line(target, fill=background, width=16 * scale, joint="curve")
    draw.line(target, fill=INK, width=6 * scale, joint="curve")

    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    end_positions = {
        scene["terminal_order"][0]: (ox + 120, oy),
        scene["terminal_order"][1]: (ox + 120, oy + 24),
    }
    start_y = int(scene["target_points"][0][1])
    distractor_y = int(scene["distractor_points"][0][1])
    _disk(draw, (ox * scale, (oy + start_y) * scale), 9 * scale, ORANGE)
    _disk(draw, (ox * scale, (oy + distractor_y) * scale), 8 * scale, GRAY)
    for name, color in (("blue", BLUE), ("green", GREEN)):
        x, y = end_positions[name]
        _disk(draw, (x * scale, y * scale), 9 * scale, color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["target_terminal"])


def margin(scene: dict) -> float:
    # The nearest wrong-terminal centre is 24 px from the correct centre and
    # both marker radii are 9 px: 6 px of visible separation remains.
    return 6.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
