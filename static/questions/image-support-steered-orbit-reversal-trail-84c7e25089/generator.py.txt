"""Latent sampler, Pillow renderer, and analytic arm."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
REVERSAL_CUTOFF = 0.65

TRAIL_COLORS = ("#17324d", "#3b2859", "#194a3a", "#5a2634")
BACKGROUNDS = ("#fbfaf4", "#f6f8fb", "#fffaf1", "#f7fbf7")


def _centerline(rng: random.Random, reversing: bool) -> list[list[float]]:
    count = 181
    inner = rng.uniform(76.0, 86.0)
    radial_gain = rng.uniform(45.0, 55.0)
    rotation = rng.uniform(-math.pi, math.pi)
    handedness = rng.choice((-1.0, 1.0))
    phase = rng.uniform(-math.pi, math.pi)
    points: list[list[float]] = []
    if reversing:
        turn = rng.uniform(0.45, 0.55)
        forward = rng.uniform(2.75, 3.25)
        backward = rng.uniform(1.35, 2.95)
        start_angle = -forward / 2.0
    else:
        sweep = rng.uniform(3.75, 4.25)
        start_angle = -sweep / 2.0
    for index in range(count):
        t = index / (count - 1)
        radius = inner + radial_gain * t + 2.5 * math.sin(2.0 * math.pi * t + phase)
        if reversing:
            if t <= turn:
                local = t / turn
                angle = start_angle + forward * local
            else:
                local = (t - turn) / (1.0 - turn)
                angle = start_angle + forward - backward * local
        else:
            angle = start_angle + sweep * t + 0.055 * math.sin(2.0 * math.pi * t + phase)
        theta = rotation + handedness * angle
        x = WIDTH / 2.0 + radius * math.cos(theta)
        y = HEIGHT / 2.0 + radius * math.sin(theta)
        points.append([round(x, 4), round(y, 4)])
    return points


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reversing = bool(int(seed) & 1)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "points": _centerline(rng, reversing),
        "stroke_width": rng.choice((10, 11, 12, 13)),
        "trail_color": rng.choice(TRAIL_COLORS),
        "background_color": rng.choice(BACKGROUNDS),
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    points = [(float(x) * SUPERSAMPLE, float(y) * SUPERSAMPLE) for x, y in scene["points"]]
    stroke_width = int(scene["stroke_width"])
    trail_color = str(scene["trail_color"])
    background_color = str(scene["background_color"])
    image = Image.new("RGB", (width * SUPERSAMPLE, height * SUPERSAMPLE), background_color)
    draw = ImageDraw.Draw(image)
    scaled_width = stroke_width * SUPERSAMPLE
    draw.line(points, fill=trail_color, width=scaled_width, joint="curve")
    radius = scaled_width / 2.0
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=trail_color)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _minor_angular_travel(scene: dict) -> float:
    width = float(scene["width"])
    height = float(scene["height"])
    points = scene["points"]
    angles = [math.atan2(float(y) - height / 2.0, float(x) - width / 2.0) for x, y in points]
    unwrapped = [angles[0]]
    for angle in angles[1:]:
        delta = (angle - unwrapped[-1] + math.pi) % (2.0 * math.pi) - math.pi
        unwrapped.append(unwrapped[-1] + delta)
    deltas = [b - a for a, b in zip(unwrapped, unwrapped[1:])]
    positive = sum(max(delta, 0.0) for delta in deltas)
    negative = sum(max(-delta, 0.0) for delta in deltas)
    return min(positive, negative)


def analytic_gold(scene: dict) -> str:
    return "reverses" if _minor_angular_travel(scene) > REVERSAL_CUTOFF else "one direction"


def margin(scene: dict) -> float:
    return round(abs(_minor_angular_travel(scene) - REVERSAL_CUTOFF), 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.25


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
