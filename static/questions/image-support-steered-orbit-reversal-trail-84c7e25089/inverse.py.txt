"""Independent pixel-only inverse for the orbit-reversal trail."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

REVERSAL_CUTOFF = 0.65


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    border = np.concatenate(
        (array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3),
         array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array - background, axis=2)
    return distance > 52.0


def _component_check(mask: np.ndarray) -> None:
    height, width = mask.shape
    remaining = mask.copy()
    sizes: list[int] = []
    for y, x in np.argwhere(remaining):
        y = int(y)
        x = int(x)
        if not remaining[y, x]:
            continue
        remaining[y, x] = False
        queue = deque([(y, x)])
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if remaining[ny, nx]:
                        remaining[ny, nx] = False
                        queue.append((ny, nx))
        sizes.append(size)
    if not sizes or max(sizes) < 1200:
        raise ValueError("trail is absent or too short")
    total = sum(sizes)
    if max(sizes) / total < 0.985:
        raise ValueError("trail is not one unbroken component")


def _ordered_angles(mask: np.ndarray) -> np.ndarray:
    ys, xs = np.nonzero(mask)
    height, width = mask.shape
    dx = xs.astype(np.float64) + 0.5 - width / 2.0
    dy = ys.astype(np.float64) + 0.5 - height / 2.0
    radii = np.hypot(dx, dy)
    angles = np.arctan2(dy, dx)
    lo, hi = np.percentile(radii, (2.0, 98.0))
    if hi - lo < 40.0:
        raise ValueError("trail radial span is too small")
    bin_width = 3.0
    bins = np.floor((radii - lo) / bin_width).astype(int)
    centers: list[float] = []
    for index in range(int((hi - lo) // bin_width) + 1):
        chosen = angles[bins == index]
        if len(chosen) < 14:
            continue
        vector = np.exp(1j * chosen).mean()
        if abs(vector) < 0.72:
            raise ValueError("trail ordering is ambiguous")
        centers.append(float(np.angle(vector)))
    if len(centers) < 14:
        raise ValueError("too few radial samples to trace trail")
    values = np.unwrap(np.asarray(centers, dtype=np.float64))
    if len(values) >= 5:
        values = np.convolve(values, np.ones(3) / 3.0, mode="valid")
    return values


def decision_from_image(image: Image.Image) -> str:
    mask = _foreground(image)
    _component_check(mask)
    angles = _ordered_angles(mask)
    deltas = np.diff(angles)
    positive = float(np.maximum(deltas, 0.0).sum())
    negative = float(np.maximum(-deltas, 0.0).sum())
    minority = min(positive, negative)
    return "reverses" if minority > REVERSAL_CUTOFF else "one direction"
