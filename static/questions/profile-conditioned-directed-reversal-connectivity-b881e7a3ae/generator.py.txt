"""Deterministic directed-network scene generator and renderer."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


NODES = 7
UNDIRECTED = [(i, (i + 1) % 6) for i in range(6)] + [(i, 6) for i in range(6)]
LABELS = ("red", "green", "blue")
PALETTES = (
    {"red": [218, 67, 65], "green": [31, 153, 91], "blue": [48, 105, 210]},
    {"red": [196, 55, 70], "green": [35, 142, 82], "blue": [50, 94, 190]},
    {"red": [230, 76, 60], "green": [27, 164, 98], "blue": [40, 116, 220]},
)


def _reachable_count(n: int, edges: Iterable[tuple[int, int]]) -> int:
    adjacency = [[] for _ in range(n)]
    for u, v in edges:
        adjacency[u].append(v)
    total = 0
    for start in range(n):
        seen = {start}
        stack = [start]
        while stack:
            for nxt in adjacency[stack.pop()]:
                if nxt not in seen:
                    seen.add(nxt)
                    stack.append(nxt)
        total += len(seen) - 1
    return total


def _reversed(edges: list[tuple[int, int]], index: int) -> list[tuple[int, int]]:
    out = list(edges)
    u, v = out[index]
    out[index] = (v, u)
    return out


def _choose_graph(rng: random.Random) -> tuple[list[tuple[int, int]], int, list[int]]:
    """Find a wheel orientation with one useful and two plausible decoy reversals."""
    full = NODES * (NODES - 1)
    for _ in range(20000):
        edges = [(a, b) if rng.randrange(2) else (b, a) for a, b in UNDIRECTED]
        base = _reachable_count(NODES, edges)
        if base == full or base < 24:
            continue
        scores = [_reachable_count(NODES, _reversed(edges, i)) for i in range(len(edges))]
        good = [i for i, score in enumerate(scores) if score == full]
        bad = [i for i, score in enumerate(scores) if score < full - 1]
        if len(good) != 1 or len(bad) < 2:
            continue
        correct = good[0]
        # Equal-length wheel edges, with decoys from distinct parts of the drawing.
        rng.shuffle(bad)
        decoys = [i for i in bad if i != correct][:2]
        if len(decoys) == 2:
            return edges, correct, [correct, *decoys]
    raise RuntimeError("could not sample a suitable directed wheel")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    edges, correct, candidates = _choose_graph(rng)
    rng.shuffle(candidates)
    answer_label = LABELS[int(seed) % 3]
    remaining_labels = [x for x in LABELS if x != answer_label]
    rng.shuffle(remaining_labels)
    labels_by_edge = {correct: answer_label}
    labels_by_edge.update({edge: label for edge, label in zip((i for i in candidates if i != correct), remaining_labels)})

    angle = rng.uniform(0, 2 * math.pi)
    reflect = -1 if rng.randrange(2) else 1
    radius = rng.uniform(172.0, 196.0)
    cx = 320.0 + rng.uniform(-18.0, 18.0)
    cy = 320.0 + rng.uniform(-18.0, 18.0)
    nodes = []
    for i in range(6):
        theta = angle + reflect * i * math.pi / 3
        nodes.append([round(cx + radius * math.cos(theta), 3), round(cy + radius * math.sin(theta), 3)])
    nodes.append([round(cx, 3), round(cy, 3)])
    palette = PALETTES[rng.randrange(len(PALETTES))]
    return {
        "nodes": nodes,
        "edges": [[u, v] for u, v in edges],
        "candidate_colors": [
            {"edge": i, "label": labels_by_edge[i], "rgb": palette[labels_by_edge[i]]}
            for i in sorted(labels_by_edge)
        ],
        "canvas": [640, 640],
        "station_radius": 18,
        "line_width": 7,
    }


def _scores(scene: dict) -> dict[str, int]:
    n = len(scene["nodes"])
    edges = [tuple(edge) for edge in scene["edges"]]
    return {
        item["label"]: _reachable_count(n, _reversed(edges, int(item["edge"])))
        for item in scene["candidate_colors"]
    }


def analytic_gold(scene: dict) -> str:
    # Access nodes, edges, and candidate_colors explicitly: these are all label inputs.
    scores = _scores(scene)
    full = len(scene["nodes"]) * (len(scene["nodes"]) - 1)
    winners = [label for label, score in scores.items() if score == full]
    if len(winners) != 1:
        raise ValueError(f"scene lacks a unique strong-connectivity reversal: {scores}")
    return winners[0]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene).values(), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < 2:
        return True
    nodes = [tuple(p) for p in scene["nodes"]]
    if min(math.dist(a, b) for i, a in enumerate(nodes) for b in nodes[i + 1 :]) < 130:
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow(draw: ImageDraw.ImageDraw, p0, p1, color, width, node_radius, scale):
    x0, y0 = p0[0] * scale, p0[1] * scale
    x1, y1 = p1[0] * scale, p1[1] * scale
    dx, dy = x1 - x0, y1 - y0
    dist = math.hypot(dx, dy)
    ux, uy = dx / dist, dy / dist
    px, py = -uy, ux
    start_gap = (node_radius + 3) * scale
    end_gap = (node_radius + 3) * scale
    end = (x1 - ux * end_gap, y1 - uy * end_gap)
    start = (x0 + ux * start_gap, y0 + uy * start_gap)
    # Put the direction marker on the isolated interior of the shaft. This avoids
    # arrowhead collisions where several spokes meet one station.
    tip = (x0 + dx * 0.70, y0 + dy * 0.70)
    base = (tip[0] - ux * 22 * scale, tip[1] - uy * 22 * scale)
    draw.line([start, end], fill=color, width=width * scale)
    half = 15 * scale
    draw.polygon([tip, (base[0] + px * half, base[1] + py * half), (base[0] - px * half, base[1] - py * half)], fill=color)


def render(scene: dict) -> Image.Image:
    width, height = (int(x) for x in scene["canvas"])
    station_radius = int(scene["station_radius"])
    line_width = int(scene["line_width"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    nodes = [tuple(p) for p in scene["nodes"]]
    edges = [tuple(e) for e in scene["edges"]]
    candidate = {int(x["edge"]): tuple(x["rgb"]) for x in scene["candidate_colors"]}
    for i, (u, v) in enumerate(edges):
        color = candidate.get(i, (82, 86, 96))
        _arrow(draw, nodes[u], nodes[v], color, line_width, station_radius, scale)
    for x, y in nodes:
        box = [(x - station_radius) * scale, (y - station_radius) * scale,
               (x + station_radius) * scale, (y + station_radius) * scale]
        draw.ellipse(box, fill=(38, 188, 200), outline=(15, 82, 92), width=3 * scale)
        inner = 5 * scale
        draw.ellipse([x * scale - inner, y * scale - inner, x * scale + inner, y * scale + inner], fill=(244, 252, 252))
    return image.resize((width, height), Image.Resampling.LANCZOS)
