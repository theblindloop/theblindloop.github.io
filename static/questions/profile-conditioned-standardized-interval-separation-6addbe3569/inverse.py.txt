"""Independent pixel-only inverse for paired capped intervals."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

# Public label palette, deliberately duplicated rather than imported.
_PALETTE = {
    "blue": (24, 112, 181),
    "orange": (230, 126, 34),
    "green": (36, 145, 92),
}


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask), strict=True):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs = []
        ys = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= 120:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _measure_scores(image: Image.Image) -> dict[str, float]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    scores = {}
    for name, rgb in _PALETTE.items():
        delta = pixels - np.asarray(rgb, dtype=np.int32)
        mask = np.sum(delta * delta, axis=2) < 42 * 42
        components = _components(mask)
        if len(components) != 2:
            raise ValueError(f"expected two visible {name} intervals, found {len(components)}")
        boxes = sorted(components, key=lambda box: (box[0] + box[2]) / 2.0)
        centers = [(box[0] + box[2]) / 2.0 for box in boxes]
        widths = [float(box[2] - box[0]) for box in boxes]
        if min(widths) < 35.0:
            raise ValueError(f"{name} interval is too short to measure")
        scores[name] = abs(centers[1] - centers[0]) / ((widths[0] + widths[1]) / 2.0)
    return scores


def decision_from_image(image: Image.Image) -> str:
    scores = _measure_scores(image)
    ordered = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ordered[0][1] - ordered[1][1] < 0.12:
        raise ValueError("ambiguous raster: top standardized separations are too close")
    return ordered[0][0]
