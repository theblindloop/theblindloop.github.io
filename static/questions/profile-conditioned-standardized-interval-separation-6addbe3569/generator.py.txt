"""Latent sampler and renderer for standardized paired-interval separation."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

PALETTE = {
    "blue": (24, 112, 181),
    "orange": (230, 126, 34),
    "green": (36, 145, 92),
}
COLORS = tuple(PALETTE)
QUARANTINE_GAP = 0.20


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _scores(scene: dict[str, Any]) -> dict[str, float]:
    scores = {}
    for row in scene["rows"]:
        intervals = row["intervals"]
        centers = [(float(item[0]) + float(item[1])) / 2.0 for item in intervals]
        widths = [float(item[1]) - float(item[0]) for item in intervals]
        scores[row["color"]] = abs(centers[1] - centers[0]) / ((widths[0] + widths[1]) / 2.0)
    return scores


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = list(COLORS)
    rng.shuffle(colors)
    winner = COLORS[int(seed) % len(COLORS)]

    # Large fixed bands make the winner stable to rasterization while jittering
    # all endpoints, widths, row order, and total ink independently.
    low = 0.82 + rng.uniform(-0.08, 0.08)
    middle = 1.52 + rng.uniform(-0.09, 0.09)
    high = 2.48 + rng.uniform(-0.10, 0.10)
    remaining = [color for color in COLORS if color != winner]
    rng.shuffle(remaining)
    ratio_for = {winner: high, remaining[0]: middle, remaining[1]: low}

    rows = []
    y_slots = [132, 222, 312]
    for slot, color in zip(y_slots, colors, strict=True):
        w1 = rng.randint(72, 116)
        w2 = rng.randint(72, 116)
        center_gap = ratio_for[color] * ((w1 + w2) / 2.0)
        pair_midpoint = rng.uniform(350.0, 382.0)
        c1 = pair_midpoint - center_gap / 2.0
        c2 = pair_midpoint + center_gap / 2.0
        intervals = [
            [round(c1 - w1 / 2.0, 3), round(c1 + w1 / 2.0, 3)],
            [round(c2 - w2 / 2.0, 3), round(c2 + w2 / 2.0, 3)],
        ]
        if rng.random() < 0.5:
            intervals.reverse()
        rows.append({"color": color, "y": slot, "intervals": intervals})

    return {"width": 720, "height": 420, "rows": rows}


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    rows = scene["rows"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), (250, 251, 253))
    draw = ImageDraw.Draw(image)

    def text(x: float, y: float, value: str, fill: tuple[int, int, int], size: int) -> None:
        draw.text((int(x * scale), int(y * scale)), value, fill=fill, font=_font(size * scale))

    text(26, 18, "PAIRED INTERVALS", (25, 32, 42), 20)
    text(26, 48, "center gap / average full width", (88, 96, 108), 14)
    draw.line((88 * scale, 88 * scale, 674 * scale, 88 * scale), fill=(210, 214, 220), width=2 * scale)

    for row in rows:
        y = int(row["y"])
        color_name = str(row["color"])
        color = PALETTE[color_name]
        text(26, y - 10, color_name.upper(), (40, 45, 52), 14)
        draw.rounded_rectangle(
            (88 * scale, (y - 35) * scale, 674 * scale, (y + 35) * scale),
            radius=8 * scale,
            outline=(222, 225, 230),
            width=1 * scale,
        )
        ordered = sorted(row["intervals"], key=lambda item: (item[0] + item[1]) / 2.0)
        for dy, interval in zip((-14, 14), ordered, strict=True):
            x1, x2 = (float(interval[0]), float(interval[1]))
            yy = y + dy
            draw.line((int(x1 * scale), yy * scale, int(x2 * scale), yy * scale), fill=color, width=6 * scale)
            draw.line((int(x1 * scale), (yy - 10) * scale, int(x1 * scale), (yy + 10) * scale), fill=color, width=6 * scale)
            draw.line((int(x2 * scale), (yy - 10) * scale, int(x2 * scale), (yy + 10) * scale), fill=color, width=6 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return max(scores, key=scores.get)


def margin(scene: dict) -> float:
    values = sorted(_scores(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for row in twin["rows"]:
        row["intervals"] = list(reversed(row["intervals"]))
    return [("reverse_unordered_interval_storage", twin)]
