"""Independent pixel-only inverse arm for the trace-crossing comparison world.

This module receives ONLY the rasterized PNG and decides which colour's wavy
trace crosses the gray horizontal line more times. It shares no geometry with
renderer.render() and never imports the renderer, prompts, generator, verifier,
or any analytic gold code.

Pipeline: (1) label each pixel to its nearest fixed colour (background, gray
line, red trace, blue trace); (2) recover the gray line's row; (3) for each
colour, recover its per-column trace row and count crossings (sign changes of
trace-row minus line-row read left to right); (4) select the colour with the
larger crossing count.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "trace_cross_pair-oracle-0.1.0"

RED_RGB = (200, 20, 20)
BLUE_RGB = (20, 20, 200)
LINE_RGB = (128, 128, 128)
BACKGROUND = (250, 250, 250)
_COLOURS = [BACKGROUND, LINE_RGB, RED_RGB, BLUE_RGB]


def _label(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest fixed colour (0 bg,1 line,2 red,3 blue)."""
    refs = np.stack([np.array(c, dtype=np.float32) for c in _COLOURS], axis=0)
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _line_row(labels: np.ndarray) -> float:
    """Median row of the gray (line=1) ink."""
    rows = np.nonzero((labels == 1).any(axis=1))[0]
    if len(rows) == 0:
        return float(np.float32(labels.shape[0]) / 2.0)
    return float(np.median(rows))


def _trace_rows(labels: np.ndarray, colour: int) -> list[float]:
    """Per-column trace row for the given labelled colour (2=red,3=blue)."""
    rows = []
    mask = labels == colour
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return rows
    per_col: dict[int, list[float]] = {}
    for x, y in zip(xs.tolist(), ys.tolist()):
        per_col.setdefault(x, []).append(float(y))
    for x in range(labels.shape[1]):
        if x in per_col:
            rows.append(float(np.mean(per_col[x])))
        else:
            rows.append(float("nan"))
    return rows


def _crossings(trace_rows: list[float], line_row: float) -> int:
    """Count sign changes of (trace - line) over columns where the trace exists."""
    signs: list[int] = []
    for value in trace_rows:
        if np.isnan(value):
            continue
        diff = value - line_row
        if abs(diff) < 0.5:  # column sits essentially on the line; skip
            continue
        signs.append(1 if diff > 0 else -1)
    count = 0
    for i in range(1, len(signs)):
        if signs[i] != signs[i - 1]:
            count += 1
    return count


def decision_from_image(image: Image.Image) -> str:
    """Return the colour whose trace crosses the gray line more times."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    line_row = _line_row(labels)
    red = _crossings(_trace_rows(labels, 2), line_row)
    blue = _crossings(_trace_rows(labels, 3), line_row)
    if red > blue:
        return "red"
    if blue > red:
        return "blue"
    return "red" if red >= blue else "blue"
