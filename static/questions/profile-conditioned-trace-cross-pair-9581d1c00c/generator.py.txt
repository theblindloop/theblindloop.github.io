"""Trace-crossing comparison world: which colored wavy trace crosses the line more.

A single gray horizontal reference line crosses the panel. Two colored wavy
traces (one red, one blue) each undulate across the full width, alternately
rising above and dipping below the line. The question asks which color's trace
crosses the gray line more times. This is a pure *measurement* problem: the
pixel arm recovers the row of the gray line and each trace's vertical profile,
counts each trace's crossings (sign changes of trace-row minus line-row read
left to right), and selects the color with the larger count.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "trace_cross_pair-0.1.0"

# Fixed colour book. The independent oracle re-derives these same RGB values
# from its own constants; it shares no geometry with render.render().
RED_RGB = (200, 20, 20)
BLUE_RGB = (20, 20, 200)
LINE_RGB = (128, 128, 128)
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
LINE_ROW = 170.0
STROKE_WIDTH = 2
TARGET_CYCLES = (1, 2, 3)

# A trace must push at least this far past the line on its shallow side so each
# crossing is cleanly above pixel noise in the per-column sign reading.
MIN_MARGIN_PX = 12.0
# Below this excursion depth a scene is quarantined as a finite-resolution
# boundary case (a grazing crossing), but it is still emitted and certified.
QUARANTINE_PX = 20.0
# Limits for the trace-dive depth (a_c = excursion depth on the shallow side).
_DIVE_MIN = 14.0
_DIVE_MAX = 55.0
# Vertical offset of each trace's base from the line.
_BASE_OFFSET_MIN = 20.0
_BASE_OFFSET_MAX = 42.0


def _sample_trace(rng: random.Random, side: str, line_row: float) -> dict:
    """Sample one waveform of the given side that crosses ``line_row`` cleanly.

    cycles is the number of full cosine cycles in the panel -> 2*cycles
    crossings of ``line_row``. dive is the excursion depth on the shallow side,
    i.e. how far the trace pushes past the line, which is the verifiability
    margin. base_offset places the trace's centre on one side of the line so
    the two coloured traces remain vertically separated.
    """
    cycles = rng.choice(TARGET_CYCLES)
    dive = rng.uniform(_DIVE_MIN, _DIVE_MAX)
    base_offset = rng.uniform(_BASE_OFFSET_MIN, _BASE_OFFSET_MAX)
    phase = rng.uniform(0.0, 2.0 * math.pi)
    sign = 1.0 if side == "red" else -1.0
    return {
        "cycles": int(cycles),
        "dive": float(dive),
        "base_offset": float(base_offset),
        "phase": float(phase),
        "sign": sign,
        "amp": float(dive + base_offset),
    }


def _waveform(trace: dict, line_row: float, x: float) -> float:
    """Vertical row of the trace at horizontal position x (image coords)."""
    base = line_row + trace["sign"] * trace["base_offset"]
    letter = 2.0 * math.pi * trace["cycles"] * x / 240.0 + trace["phase"]
    # A cosine of amplitude trip["amp"] centred at `base`; the swing toward the
    # line is amp - base_offset = dive past the line, giving 2*cycles crossings.
    return base + trace["amp"] * math.cos(letter)


def _crossings(trace: dict, line_row: float) -> int:
    """Analytic count of crossings of the line by this trace."""
    return 2 * int(trace["cycles"])


def _excursion_depth(trace: dict) -> float:
    """Shallow-side clearance of this trace past the line (verifiability)."""
    return float(trace["dive"])


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a clear, non-tied crossing count.

    The two traces are given independent cycle counts so their crossing counts
    differ; a tie is resampled. Returns a plain JSON-serialisable dict with
    ``line_row`` and ``traces`` (one waveform per colour).
    """
    rng = random.Random(seed * 1013 + 11)
    while True:
        red = _sample_trace(rng, "red", LINE_ROW)
        blue = _sample_trace(rng, "blue", LINE_ROW)
        _red = _crossings(red, LINE_ROW)
        _blue = _crossings(blue, LINE_ROW)
        if _red == _blue:
            continue
        if margin({"line_row": LINE_ROW, "traces": {"red": red, "blue": blue}}) < MIN_MARGIN_PX:
            continue
        scene = {
            "line_row": LINE_ROW,
            "traces": {"red": red, "blue": blue},
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        return scene


def _draw_trace(draw: ImageDraw.ImageDraw, scene: dict, colour: tuple) -> None:
    line_row = scene["line_row"]
    red_trace = scene["traces"]["red"]
    blue_trace = scene["traces"]["blue"]
    trace = red_trace if colour == RED_RGB else blue_trace
    points = [_waveform(trace, line_row, x) for x in range(WIDTH)]
    prev = (0.0, points[0])
    for x in range(1, WIDTH):
        nxt = (float(x), points[x])
        draw.line([prev, nxt], fill=colour, width=STROKE_WIDTH)
        prev = nxt


def render(scene: dict) -> Image.Image:
    """Render I = R(z): a gray line plus the two colored wavy traces."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    line_row = scene["line_row"]
    traces = scene["traces"]
    draw.line([(0, line_row), (scene["width"], line_row)], fill=LINE_RGB, width=2)
    _draw_trace(draw, scene, RED_RGB)
    _draw_trace(draw, scene, BLUE_RGB)
    return img


def analytic_gold(scene: dict) -> str:
    """Colour whose trace crosses the line more times."""
    red_n = 2 * int(scene["traces"]["red"]["cycles"])
    blue_n = 2 * int(scene["traces"]["blue"]["cycles"])
    if red_n > blue_n:
        return "red"
    return "blue"


def margin(scene: dict) -> float:
    """Smallest shallow-side excursion depth over both traces."""
    return float(
        min(_excursion_depth(scene["traces"]["red"]), _excursion_depth(scene["traces"]["blue"]))
    )


def is_quarantined(scene: dict) -> bool:
    """Near-grazing: some trace barely pushes past the line."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is drawn verbatim as visible ink.

    The line row and each trace's cycles/amplitude/phase all become on-screen
    strokes, so there is no hidden scene field that could change the crossing
    count without changing the raster. No nontrivial pixel-identical latent
    transformation exists.
    """
    return []
