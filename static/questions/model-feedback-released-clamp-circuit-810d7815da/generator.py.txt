"""Sampler and renderer for a visibly clamped Boolean circuit."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

SIZE = (768, 640)
BACKGROUND = (247, 245, 238)
INK = (35, 42, 52)
CARD = (255, 255, 252)
BORDER = (105, 113, 124)
ONE = (30, 55, 78)
ZERO = (229, 232, 226)
XOR = (126, 73, 163)
AND = (28, 126, 125)
AMBER = (238, 163, 38)
INPUT_X = (12, 203, 394, 585)
GATE_X = (8, 264, 520)
GATE_Y = (166, 302)
OUTPUT_X = (5, 198, 391, 584)
OUTPUT_Y = 466
DECISIONS = ("output-A", "output-B", "output-C", "output-D")

# Fixed visible source binding. Operations and values vary per scene.
GATE_PARENTS = (
    ("I1", "I2"), ("I2", "I3"), ("I3", "I4"),
    ("G1", "G2"), ("G2", "G3"), ("G1", "G3"),
)
OUTPUT_PARENTS = (("G4", "G5"), ("G4", "G6"), ("G5", "G6"), ("G4", "G3"))


def _font(size: int) -> ImageFont.ImageFont:
    path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path), size=size) if path.exists() else ImageFont.load_default()


def _apply(op: str, a: int, b: int) -> int:
    if op == "XOR":
        return int(bool(a) != bool(b))
    if op == "AND":
        return int(bool(a) and bool(b))
    raise ValueError(op)


def evaluate(inputs: list[int], operations: list[str], clamp_gate: int | None = None,
             forced_value: int | None = None) -> tuple[list[int], list[int]]:
    if len(inputs) != 4 or len(operations) != 10:
        raise ValueError("four inputs and ten operations required")
    values = {f"I{i + 1}": int(value) for i, value in enumerate(inputs)}
    gates: list[int] = []
    for index, (left, right) in enumerate(GATE_PARENTS):
        value = _apply(operations[index], values[left], values[right])
        if clamp_gate == index:
            if forced_value not in (0, 1):
                raise ValueError("clamp requires a binary forced value")
            value = int(forced_value)
        gates.append(value)
        values[f"G{index + 1}"] = value
    outputs = [
        _apply(operations[6 + index], values[left], values[right])
        for index, (left, right) in enumerate(OUTPUT_PARENTS)
    ]
    return gates, outputs


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x4C414D50)
    target = int(seed) % 4
    for _ in range(20000):
        inputs = [rng.randrange(2) for _ in range(4)]
        operations = [rng.choice(("XOR", "AND")) for _ in range(10)]
        clamp_gate = rng.randrange(3)
        natural_gates, natural_outputs = evaluate(inputs, operations)
        forced_value = 1 - natural_gates[clamp_gate]
        current_gates, current_outputs = evaluate(inputs, operations, clamp_gate, forced_value)
        changed = [i for i, (a, b) in enumerate(zip(natural_outputs, current_outputs)) if a != b]
        if changed == [target] and sum(current_gates) not in (0, 6):
            return {
                "inputs": inputs,
                "operations": operations,
                "clamp_gate": clamp_gate,
                "forced_value": forced_value,
                "current_gates": current_gates,
                "current_outputs": current_outputs,
            }
    raise RuntimeError("could not sample a balanced counterfactual scene")


def _validate(scene: dict) -> tuple[list[int], list[str], int, int, list[int], list[int]]:
    inputs = [int(v) for v in scene["inputs"]]
    operations = [str(v) for v in scene["operations"]]
    clamp_gate = int(scene["clamp_gate"])
    forced_value = int(scene["forced_value"])
    current_gates = [int(v) for v in scene["current_gates"]]
    current_outputs = [int(v) for v in scene["current_outputs"]]
    if len(inputs) != 4 or len(operations) != 10 or len(current_gates) != 6 or len(current_outputs) != 4:
        raise ValueError("wrong circuit size")
    if any(v not in (0, 1) for v in inputs + current_gates + current_outputs):
        raise ValueError("all lamps must be binary")
    if any(op not in ("XOR", "AND") for op in operations) or not 0 <= clamp_gate < 3:
        raise ValueError("invalid circuit")
    expected_gates, expected_outputs = evaluate(inputs, operations, clamp_gate, forced_value)
    if current_gates != expected_gates or current_outputs != expected_outputs:
        raise ValueError("displayed current state is inconsistent")
    return inputs, operations, clamp_gate, forced_value, current_gates, current_outputs


def analytic_gold(scene: dict) -> str:
    try:
        inputs, operations, _clamp_gate, _forced, _gates, current = _validate(scene)
        _natural_gates, natural = evaluate(inputs, operations)
        changed = [index for index, (a, b) in enumerate(zip(natural, current)) if a != b]
        return DECISIONS[changed[0]] if len(changed) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str,
              font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
              text, font=font, fill=fill)


def _lamp(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int) -> None:
    x, y = center
    draw.ellipse((x - 20, y - 20, x + 20, y + 20), fill=ONE if value else ZERO, outline=INK, width=3)
    _centered(draw, center, str(value), _font(22), (255, 255, 255) if value else INK)


def _operation(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], op: str) -> None:
    draw.rounded_rectangle(box, radius=10, fill=XOR if op == "XOR" else AND)
    _centered(draw, ((box[0] + box[2]) // 2, (box[1] + box[3]) // 2), op, _font(17), (255, 255, 255))


def render(scene: dict) -> Image.Image:
    inputs, operations, clamp_gate, _forced, gates, outputs = _validate(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered(draw, (384, 25), "CURRENT CIRCUIT — AMBER CLAMP ACTIVE", _font(23), INK)
    _centered(draw, (384, 51), "1 = dark    0 = light", _font(16), BORDER)

    for index, (x, value) in enumerate(zip(INPUT_X, inputs)):
        draw.rounded_rectangle((x, 66, x + 171, 137), radius=12, fill=CARD, outline=BORDER, width=3)
        draw.text((x + 16, 86), f"INPUT I{index + 1}", font=_font(19), fill=INK)
        _lamp(draw, (x + 142, 102), value)

    for index, ((left, right), value) in enumerate(zip(GATE_PARENTS, gates)):
        col, row = index % 3, index // 3
        x, y = GATE_X[col], GATE_Y[row]
        outline = AMBER if index == clamp_gate else BORDER
        draw.rectangle((x, y, x + 240, y + 115), fill=outline)
        draw.rectangle((x + 7, y + 7, x + 233, y + 108), fill=CARD)
        draw.text((x + 14, y + 13), f"G{index + 1}", font=_font(20), fill=INK)
        if index == clamp_gate:
            draw.text((x + 139, y + 15), "CLAMP", font=_font(16), fill=(174, 100, 0))
        draw.text((x + 20, y + 68), left, font=_font(18), fill=INK)
        _operation(draw, (x + 62, y + 53, x + 137, y + 94), operations[index])
        draw.text((x + 145, y + 68), right, font=_font(18), fill=INK)
        draw.text((x + 178, y + 68), "=", font=_font(18), fill=INK)
        _lamp(draw, (x + 215, y + 74), value)

    draw.line((18, 438, 750, 438), fill=BORDER, width=3)
    _centered(draw, (384, 450), "OUTPUTS", _font(18), INK)
    for index, ((left, right), value) in enumerate(zip(OUTPUT_PARENTS, outputs)):
        x, y = OUTPUT_X[index], OUTPUT_Y
        draw.rounded_rectangle((x, y, x + 179, y + 143), radius=12, fill=CARD, outline=BORDER, width=3)
        draw.text((x + 12, y + 12), f"OUTPUT {chr(65 + index)}", font=_font(18), fill=INK)
        draw.text((x + 11, y + 70), left, font=_font(16), fill=INK)
        _operation(draw, (x + 48, y + 55, x + 115, y + 94), operations[6 + index])
        draw.text((x + 121, y + 70), right, font=_font(16), fill=INK)
        _lamp(draw, (x + 90, y + 118), value)
    return image
