"""Independent fixed-layout pixel inverse for the released-clamp circuit."""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = (247, 245, 238)
ONE = (30, 55, 78)
ZERO = (229, 232, 226)
XOR = (126, 73, 163)
AND = (28, 126, 125)
AMBER = (238, 163, 38)
BORDER = (105, 113, 124)
INPUT_X = (12, 203, 394, 585)
GATE_X = (8, 264, 520)
GATE_Y = (166, 302)
OUTPUT_X = (5, 198, 391, 584)
OUTPUT_Y = 466
DECISIONS = ("output-A", "output-B", "output-C", "output-D")
GATE_PARENTS = (
    ("I1", "I2"), ("I2", "I3"), ("I3", "I4"),
    ("G1", "G2"), ("G2", "G3"), ("G1", "G3"),
)
OUTPUT_PARENTS = (("G4", "G5"), ("G4", "G6"), ("G5", "G6"), ("G4", "G3"))


def _near(pixel: np.ndarray, color: tuple[int, int, int], tolerance: int = 4) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _lamp(arr: np.ndarray, x: int, y: int) -> int | None:
    pixel = arr[y, x + 11]
    if _near(pixel, ONE):
        return 1
    if _near(pixel, ZERO):
        return 0
    return None


def _apply(op: str, a: int, b: int) -> int:
    return int(bool(a) != bool(b)) if op == "XOR" else int(bool(a) and bool(b))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (640, 768, 3) or not _near(arr[630, 760], BACKGROUND):
        return "abstain"
    inputs = [_lamp(arr, x + 142, 102) for x in INPUT_X]
    if any(value is None for value in inputs):
        return "abstain"

    operations: list[str] = []
    clamp_hits: list[int] = []
    displayed_gates: list[int] = []
    for index in range(6):
        x, y = GATE_X[index % 3], GATE_Y[index // 3]
        op_pixel = arr[y + 74, x + 67]
        op = "XOR" if _near(op_pixel, XOR) else "AND" if _near(op_pixel, AND) else None
        value = _lamp(arr, x + 215, y + 74)
        border = arr[y + 2, x + 2]
        if op is None or value is None:
            return "abstain"
        operations.append(op)
        displayed_gates.append(value)
        if _near(border, AMBER):
            clamp_hits.append(index)
        elif not _near(border, BORDER):
            return "abstain"
    if len(clamp_hits) != 1 or clamp_hits[0] >= 3:
        return "abstain"

    displayed_outputs: list[int] = []
    for x in OUTPUT_X:
        op_pixel = arr[OUTPUT_Y + 74, x + 53]
        if _near(op_pixel, XOR):
            operations.append("XOR")
        elif _near(op_pixel, AND):
            operations.append("AND")
        else:
            return "abstain"
        value = _lamp(arr, x + 90, OUTPUT_Y + 118)
        if value is None:
            return "abstain"
        displayed_outputs.append(value)

    values = {f"I{i + 1}": int(value) for i, value in enumerate(inputs)}
    natural_gates: list[int] = []
    for index, (left, right) in enumerate(GATE_PARENTS):
        value = _apply(operations[index], values[left], values[right])
        natural_gates.append(value)
        values[f"G{index + 1}"] = value

    current_values = {f"I{i + 1}": int(value) for i, value in enumerate(inputs)}
    for index, (left, right) in enumerate(GATE_PARENTS):
        value = _apply(operations[index], current_values[left], current_values[right])
        if index == clamp_hits[0]:
            value = displayed_gates[index]
        if value != displayed_gates[index]:
            return "abstain"
        current_values[f"G{index + 1}"] = value
    if displayed_gates[clamp_hits[0]] == natural_gates[clamp_hits[0]]:
        return "abstain"

    natural_outputs = [
        _apply(operations[6 + index], values[left], values[right])
        for index, (left, right) in enumerate(OUTPUT_PARENTS)
    ]
    expected_current = [
        _apply(operations[6 + index], current_values[left], current_values[right])
        for index, (left, right) in enumerate(OUTPUT_PARENTS)
    ]
    if expected_current != displayed_outputs:
        return "abstain"
    changed = [i for i, (a, b) in enumerate(zip(natural_outputs, displayed_outputs)) if a != b]
    return DECISIONS[changed[0]] if len(changed) == 1 else "abstain"
