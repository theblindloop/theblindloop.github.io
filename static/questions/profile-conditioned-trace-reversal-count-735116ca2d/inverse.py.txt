"""Independent image-only inverse for the trace reversal question."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _saturated_trace_mask(image: Image.Image) -> np.ndarray:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = pixels.max(axis=2) - pixels.min(axis=2)
    # Axes/grid/text are achromatic; the traces are the only saturated marks.
    mask = (spread >= 45) & (pixels.mean(axis=2) <= 245)
    mask[:80, :] = False
    mask[375:, :] = False
    return mask


def _groups(active: np.ndarray) -> list[tuple[int, int]]:
    groups: list[tuple[int, int]] = []
    start: int | None = None
    for index, value in enumerate(active.tolist() + [False]):
        if value and start is None:
            start = index
        elif not value and start is not None:
            groups.append((start, index - 1))
            start = None
    return groups


def _marker_centers(mask: np.ndarray, left: int, right: int) -> list[int]:
    roi = mask[:, left:right]
    column_ink = roi.sum(axis=0)
    # A marker is taller than a five-pixel trace stroke.  This identifies the
    # eight marker columns without relying on renderer constants or scene data.
    groups = _groups(column_ink >= 10)
    if len(groups) == 8:
        return [left + round((start + end) / 2) for start, end in groups]
    occupied = np.flatnonzero(roi.any(axis=0))
    if occupied.size < 2:
        raise ValueError("no saturated trace found")
    x0 = left + int(occupied.min()) + 8
    x1 = left + int(occupied.max()) - 8
    return [round(x0 + (x1 - x0) * i / 7) for i in range(8)]


def _trace_reversals(mask: np.ndarray, left: int, right: int) -> int:
    centers = _marker_centers(mask, left, right)
    y_centers: list[float] = []
    for center in centers:
        lo = max(left, center - 3)
        hi = min(right, center + 4)
        ys = np.flatnonzero(mask[:, lo:hi])
        if ys.size == 0:
            raise ValueError("marker column has no trace pixels")
        y_centers.append(float(np.median(ys)))
    slopes: list[float] = []
    for index in range(7):
        # A two-point least-squares line fit is deliberately performed on the
        # recovered marker centers rather than reading any latent coordinates.
        slopes.append(float(np.polyfit((0.0, 1.0), (y_centers[index], y_centers[index + 1]), 1)[0]))
    signs = [1 if slope > 4 else -1 if slope < -4 else 0 for slope in slopes]
    if 0 in signs:
        raise ValueError("raster slope is too close to horizontal")
    return sum(1 for before, after in zip(signs, signs[1:]) if before != after)


def decision_from_image(image: Image.Image) -> str:
    """Recover A/B using only the final RGB raster."""

    mask = _saturated_trace_mask(image)
    left_count = _trace_reversals(mask, 0, image.width // 2)
    right_count = _trace_reversals(mask, image.width // 2, image.width)
    if left_count == right_count:
        raise ValueError("tie is quarantined")
    return "A" if left_count > right_count else "B"
