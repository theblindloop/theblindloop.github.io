"""Deterministic two-panel trace renderer and analytic reference."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS_SIZE = (900, 460)
PANEL_SPLIT = 450
PLOT_X = ((82, 388), (512, 818))
PLOT_Y = (92, 358)
MARKER_COUNT = 8
Y_CENTER = 225
Y_SCALE = 1.10
LINE_WIDTH = 5
MARKER_RADIUS = 8

PALETTE = (
    (210, 75, 75),
    (54, 112, 202),
    (35, 145, 112),
    (137, 86, 196),
    (218, 126, 39),
    (190, 65, 143),
)

PATTERNS: dict[int, tuple[int, ...]] = {
    count: tuple(
        (1 if index % 2 == 0 else -1) if index <= count else (1 if count % 2 == 0 else -1)
        for index in range(MARKER_COUNT - 1)
    )
    for count in range(MARKER_COUNT - 1)
}


def _turn_count(values: Iterable[int | float]) -> int:
    values = list(values)
    signs: list[int] = []
    for left, right in zip(values, values[1:]):
        delta = right - left
        signs.append(1 if delta > 0 else -1 if delta < 0 else 0)
    return sum(
        1
        for left, right in zip(signs, signs[1:])
        if left != 0 and right != 0 and left != right
    )


def _make_trace(
    rng: random.Random,
    reversal_count: int,
    steps: list[int],
) -> list[int]:
    signs = list(PATTERNS[reversal_count])
    if rng.random() < 0.5:
        signs = [-value for value in signs]
    value = rng.randint(-8, 8)
    points = [value]
    for sign, step in zip(signs, steps):
        value += sign * step
        points.append(value)
    return points


def sample_scene(seed: int) -> dict:
    """Sample a balanced pair of eight-marker traces from one integer seed."""

    rng = random.Random(int(seed))
    low = rng.randint(0, 4)
    high = min(6, low + rng.randint(1, 2))
    # Consecutive stress seeds give a deterministic A/B counterbalance while
    # color and geometry remain independently randomized.
    winner = "A" if int(seed) % 2 == 0 else "B"
    steps = [rng.randint(10, 14) for _ in range(MARKER_COUNT - 1)]
    low_trace = _make_trace(rng, low, steps)
    high_trace = _make_trace(rng, high, steps)
    if winner == "A":
        a_values, b_values = high_trace, low_trace
    else:
        a_values, b_values = low_trace, high_trace
    a_color, b_color = rng.sample(PALETTE, 2)
    return {
        "a_values": a_values,
        "b_values": b_values,
        "a_color": list(a_color),
        "b_color": list(b_color),
    }


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _marker_xs(x0: int, x1: int) -> list[int]:
    return [round(x0 + (x1 - x0) * index / (MARKER_COUNT - 1)) for index in range(MARKER_COUNT)]


def _draw_trace(
    draw: ImageDraw.ImageDraw,
    values: list[int],
    color: tuple[int, int, int] | list[int],
    x0: int,
    x1: int,
) -> None:
    points = [(x, round(Y_CENTER - value * Y_SCALE)) for x, value in zip(_marker_xs(x0, x1), values)]
    draw.line(points, fill=tuple(color), width=LINE_WIDTH, joint="curve")
    for x, y in points:
        draw.ellipse(
            (x - MARKER_RADIUS, y - MARKER_RADIUS, x + MARKER_RADIUS, y + MARKER_RADIUS),
            fill=tuple(color),
        )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS_SIZE, (250, 251, 253))
    draw = ImageDraw.Draw(image)
    font = _font()
    draw.text((28, 20), "direction changes in ordered traces", fill=(32, 38, 48), font=font)
    for panel_index, (panel_left, panel_right) in enumerate(((45, 425), (475, 855))):
        x0, x1 = PLOT_X[panel_index]
        draw.rounded_rectangle(
            (panel_left, 60, panel_right, 392),
            radius=8,
            fill=(247, 248, 251),
            outline=(192, 198, 208),
            width=2,
        )
        draw.text((panel_left + 14, 69), "A" if panel_index == 0 else "B", fill=(24, 30, 40), font=font)
        for grid_y in (125, 175, 225, 275, 325):
            draw.line((x0, grid_y, x1, grid_y), fill=(224, 228, 234), width=1)
        for grid_x in _marker_xs(x0, x1):
            draw.line((grid_x, PLOT_Y[0], grid_x, PLOT_Y[1]), fill=(238, 240, 244), width=1)
        draw.line((x0, PLOT_Y[1], x1, PLOT_Y[1]), fill=(84, 91, 103), width=2)
        draw.line((x0, PLOT_Y[0], x0, PLOT_Y[1]), fill=(84, 91, 103), width=2)
        draw.text((x1 - 12, PLOT_Y[1] + 8), "x", fill=(84, 91, 103), font=font)
    _draw_trace(draw, scene["a_values"], scene["a_color"], *PLOT_X[0])
    _draw_trace(draw, scene["b_values"], scene["b_color"], *PLOT_X[1])
    return image


def analytic_gold(scene: dict) -> str:
    a_count = _turn_count(scene["a_values"])
    b_count = _turn_count(scene["b_values"])
    return "A" if a_count >= b_count else "B"


def margin(scene: dict) -> float:
    return float(abs(_turn_count(scene["a_values"]) - _turn_count(scene["b_values"])))


def is_quarantined(scene: dict) -> bool:
    traces = (scene["a_values"], scene["b_values"])
    has_flat_segment = any(
        right == left
        for values in traces
        for left, right in zip(values, values[1:])
    )
    min_step = min(
        abs(right - left)
        for values in traces
        for left, right in zip(values, values[1:])
    )
    return has_flat_segment or min_step < 4 or margin(scene) == 0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
