"""Analytic scene sampler and raster renderer for Shared Loop Pair.

The visible object is a small cactus graph: two four-edge cycles share exactly
one articulation station.  The answer is not stored in the scene.  It is
recomputed from the undirected edge list and the two marked vertex indices.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 640
INK_WIDTH = 5
OUTER_RADIUS = 8
INNER_RADIUS = 5

PALETTES = (
    {
        "ink": [29, 48, 70],
        "neutral": [177, 211, 226],
        "green": [36, 161, 104],
        "orange": [238, 132, 52],
        "decor": [233, 236, 239],
        "background": [255, 255, 255],
    },
    {
        "ink": [55, 43, 76],
        "neutral": [172, 210, 204],
        "green": [18, 143, 118],
        "orange": [226, 108, 61],
        "decor": [235, 233, 238],
        "background": [255, 255, 255],
    },
)

# Logical graph: vertex zero is the only articulation.  Each tuple is an
# undirected edge; its stored direction has no meaning.
BASE_EDGES = (
    (0, 2), (2, 1), (1, 3), (3, 0),
    (0, 5), (5, 4), (4, 6), (6, 0),
)
LEFT = (1, 2, 3)
RIGHT = (4, 5, 6)


def _all_simple_paths(edges: list[list[int]], start: int, goal: int) -> list[list[int]]:
    adjacency: dict[int, set[int]] = {}
    for a, b in edges:
        adjacency.setdefault(int(a), set()).add(int(b))
        adjacency.setdefault(int(b), set()).add(int(a))
    paths: list[list[int]] = []

    def visit(node: int, path: list[int]) -> None:
        if node == goal:
            paths.append(path[:])
            return
        for nxt in sorted(adjacency.get(node, ())):
            if nxt not in path:
                visit(nxt, path + [nxt])

    visit(start, [start])
    return paths


def _share_cycle(edges: list[list[int]], first: int, second: int) -> bool:
    """True exactly when a simple cycle contains both vertices.

    Two vertices of an undirected graph share a simple cycle iff there are two
    paths between them whose internal vertex sets are disjoint.  The graphs are
    tiny, so exhaustive path enumeration is clearer than a hidden library call.
    """

    paths = _all_simple_paths(edges, int(first), int(second))
    for i, left in enumerate(paths):
        left_inside = set(left[1:-1])
        left_edges = {frozenset(edge) for edge in zip(left, left[1:])}
        for right in paths[i + 1 :]:
            right_inside = set(right[1:-1])
            right_edges = {frozenset(edge) for edge in zip(right, right[1:])}
            if left_inside.isdisjoint(right_inside) and left_edges.isdisjoint(right_edges):
                return True
    return False


def _decorations(rng: random.Random, cx: int, cy: int) -> list[list[int]]:
    """Generate remote, low-contrast dashes that cannot influence the answer."""

    values: list[list[int]] = []
    attempts = 0
    while len(values) < 10 and attempts < 1000:
        attempts += 1
        x = rng.randint(34, WIDTH - 55)
        y = rng.randint(34, HEIGHT - 35)
        if abs(x - cx) < 120 and abs(y - cy) < 70:
            continue
        length = rng.randint(10, 20)
        vertical = bool(rng.getrandbits(1))
        values.append([x, y, x if vertical else x + length, y + length if vertical else y])
    return values


def sample_scene(seed: int) -> dict:
    """Sample one balanced, translated, reflection-varied scene."""

    rng = random.Random(int(seed))
    # These centers sit in the exclusive core of an 8x8 overlapping-ablation
    # row.  The graph therefore moves substantially while remaining one-row
    # tall for the requested focal/pathlike support behavior.
    cx = rng.choice((200, 280, 360, 440))
    cy = rng.choice((120, 200, 280, 360, 440, 520))
    xflip = -1 if rng.getrandbits(1) else 1
    yflip = -1 if rng.getrandbits(1) else 1
    inner = rng.choice((29, 31, 33))
    tip = rng.choice((62, 64, 66))
    rise = rng.choice((10, 11, 12))

    local = [
        (0, 0),
        (-tip, 0), (-inner, -rise), (-inner, rise),
        (tip, 0), (inner, -rise), (inner, rise),
    ]
    nodes = [[cx + xflip * x, cy + yflip * y] for x, y in local]

    want_yes = (int(seed) % 2 == 0)
    if want_yes:
        block = LEFT if rng.getrandbits(1) else RIGHT
        pair = rng.sample(block, 2)
    else:
        pair = [rng.choice(LEFT), rng.choice(RIGHT)]
    if rng.getrandbits(1):
        pair.reverse()

    palette = copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))])
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "nodes": nodes,
        "edges": [list(edge) for edge in BASE_EDGES],
        "green": int(pair[0]),
        "orange": int(pair[1]),
        "decoration_segments": _decorations(rng, cx, cy),
        "palette": palette,
    }


def render(scene: dict) -> Image.Image:
    """Render the final RGB image with deterministic supersampling."""

    width = int(scene["width"])
    height = int(scene["height"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    green = int(scene["green"])
    orange = int(scene["orange"])
    decorations = scene["decoration_segments"]
    palette = scene["palette"]
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)

    for x0, y0, x1, y1 in decorations:
        draw.line(
            (x0 * scale, y0 * scale, x1 * scale, y1 * scale),
            fill=tuple(palette["decor"]),
            width=2 * scale,
        )

    for a, b in edges:
        ax, ay = nodes[int(a)]
        bx, by = nodes[int(b)]
        draw.line(
            (ax * scale, ay * scale, bx * scale, by * scale),
            fill=tuple(palette["ink"]),
            width=INK_WIDTH * scale,
        )

    for index, (x, y) in enumerate(nodes):
        outer = OUTER_RADIUS * scale
        inner_radius = INNER_RADIUS * scale
        center = (x * scale, y * scale)
        draw.ellipse(
            (center[0] - outer, center[1] - outer, center[0] + outer, center[1] + outer),
            fill=tuple(palette["ink"]),
        )
        fill = palette["green"] if index == green else palette["orange"] if index == orange else palette["neutral"]
        draw.ellipse(
            (
                center[0] - inner_radius,
                center[1] - inner_radius,
                center[0] + inner_radius,
                center[1] + inner_radius,
            ),
            fill=tuple(fill),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if _share_cycle(scene["edges"], int(scene["green"]), int(scene["orange"])) else "no"


def margin(scene: dict) -> float:
    """A conservative visible clearance to the discrete raster boundary.

    The topological runner-up would require deleting a complete five-pixel link
    or adding a new one across at least the nearest nonedge clearance.  We
    report the smaller certified stroke-scale clearance in pixels.
    """

    nodes = scene["nodes"]
    edges = {frozenset((int(a), int(b))) for a, b in scene["edges"]}
    nearest_nonedge = math.inf
    for i, (ax, ay) in enumerate(nodes):
        for j, (bx, by) in enumerate(nodes[i + 1 :], start=i + 1):
            if frozenset((i, j)) not in edges:
                nearest_nonedge = min(nearest_nonedge, math.hypot(bx - ax, by - ay))
    return float(min(INK_WIDTH, max(0.0, nearest_nonedge - 2 * OUTER_RADIUS)))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    if len(nodes) != 7 or len(scene["edges"]) != 8:
        return True
    if scene["green"] == scene["orange"]:
        return True
    minimum = min(
        math.hypot(nodes[i][0] - nodes[j][0], nodes[i][1] - nodes[j][1])
        for i in range(len(nodes))
        for j in range(i + 1, len(nodes))
    )
    return minimum < 18.0 or margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = twin["edges"][1:] + twin["edges"][:1]
    return [("cyclically_reorder_undirected_edges", twin)]
