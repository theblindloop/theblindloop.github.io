"""Independent PNG-only inverse arm for crossing-free redraw."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image

CYAN = np.array((22, 184, 204), dtype=np.int16)
LINK = np.array((82, 88, 98), dtype=np.int16)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        group = []
        while queue:
            yy, xx = queue.popleft()
            group.append((yy, xx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = yy + dy, xx + dx
                    if (dy or dx) and 0 <= ny < height and 0 <= nx < width \
                            and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(group) > 100:
            groups.append(group)
    return groups


def _recover_graph(image: Image.Image) -> tuple[list[tuple[float, float]], set[frozenset[int]]]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan_mask = np.max(np.abs(array - CYAN), axis=2) <= 5
    groups = _components(cyan_mask)
    if len(groups) != 6:
        raise ValueError(f"expected six cyan stations, found {len(groups)}")
    centers = []
    for group in groups:
        ys = np.array([p[0] for p in group])
        xs = np.array([p[1] for p in group])
        centers.append((float(xs.mean()), float(ys.mean())))
    centers.sort(key=lambda point: (point[1], point[0]))

    link_mask = np.max(np.abs(array - LINK), axis=2) <= 5
    height, width = link_mask.shape
    edges: set[frozenset[int]] = set()
    for a, b in itertools.combinations(range(6), 2):
        x1, y1 = centers[a]
        x2, y2 = centers[b]
        hits = 0
        total = 0
        for t in np.linspace(0.11, 0.89, 120):
            x = int(round(x1 + t * (x2 - x1)))
            y = int(round(y1 + t * (y2 - y1)))
            y0, y1s = max(0, y - 2), min(height, y + 3)
            x0, x1s = max(0, x - 2), min(width, x + 3)
            hits += bool(link_mask[y0:y1s, x0:x1s].any())
            total += 1
        if hits / total > 0.78:
            edges.add(frozenset((a, b)))
    if len(edges) != 9 or any(sum(v in edge for edge in edges) != 3 for v in range(6)):
        raise ValueError("could not recover a six-station cubic graph")
    return centers, edges


def _contains_k33(edges: set[frozenset[int]]) -> bool:
    for left in itertools.combinations(range(6), 3):
        right = tuple(v for v in range(6) if v not in left)
        if all(frozenset((a, b)) in edges for a in left for b in right):
            return True
    return False


def decision_from_image(image: Image.Image) -> str:
    _, edges = _recover_graph(image)
    return "no" if _contains_k33(edges) else "yes"
