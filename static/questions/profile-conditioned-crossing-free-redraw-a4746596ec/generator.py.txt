"""Deterministic renderer and analytic arm for crossing-free redraw."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 640
BG = (248, 247, 242)
LINK = (82, 88, 98)
CYAN = (22, 184, 204)
OUTLINE = (20, 42, 48)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    planar = int(seed) % 2 == 0
    if planar:
        edges = [(0, 1), (1, 2), (0, 2), (3, 4), (4, 5), (3, 5),
                 (0, 3), (1, 4), (2, 5)]
    else:
        edges = [(a, b) for a in range(3) for b in range(3, 6)]

    slots = list(range(6))
    rng.shuffle(slots)
    rotation = math.radians(rng.randrange(0, 360))
    cx = 320 + rng.randrange(-18, 19)
    cy = 320 + rng.randrange(-18, 19)
    radius = rng.randrange(205, 226)
    positions = []
    for vertex in range(6):
        angle = rotation + slots[vertex] * math.tau / 6
        positions.append([round(cx + radius * math.cos(angle)),
                          round(cy + radius * math.sin(angle))])
    rng.shuffle(edges)
    return {"nodes": positions, "edges": [list(edge) for edge in edges]}


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    image = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((24, 24, SIZE - 25, SIZE - 25), radius=18,
                           outline=(190, 194, 196), width=3)
    for a, b in edges:
        draw.line((tuple(nodes[a]), tuple(nodes[b])), fill=LINK, width=7)
    for x, y in nodes:
        draw.ellipse((x - 17, y - 17, x + 17, y + 17), fill=CYAN,
                     outline=OUTLINE, width=3)
    return image


def _contains_k33(node_count: int, edges: list[list[int]]) -> bool:
    adjacency = {frozenset(edge) for edge in edges}
    vertices = range(node_count)
    for left in itertools.combinations(vertices, 3):
        right = tuple(v for v in vertices if v not in left)
        if len(right) == 3 and all(frozenset((a, b)) in adjacency
                                   for a in left for b in right):
            return True
    return False


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    return "no" if _contains_k33(len(nodes), edges) else "yes"


def margin(scene: dict) -> float:
    """Discrete one-edge boundary from the opposite graph property."""
    return 1.0


def _point_segment_distance(px: float, py: float, ax: float, ay: float,
                            bx: float, by: float) -> float:
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / (dx*dx + dy*dy)))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    edges = scene["edges"]
    if min(math.dist(a, b) for a, b in itertools.combinations(nodes, 2)) < 80:
        return True
    for a, b in edges:
        ax, ay = nodes[a]
        bx, by = nodes[b]
        for index, (px, py) in enumerate(nodes):
            if index not in (a, b) and _point_segment_distance(px, py, ax, ay, bx, by) < 38:
                return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["nodes"] = list(reversed(twin["nodes"]))
    n = len(twin["nodes"])
    twin["edges"] = [[n - 1 - a, n - 1 - b] for a, b in reversed(scene["edges"])]
    return [("reverse_vertex_and_edge_serialization", twin)]
