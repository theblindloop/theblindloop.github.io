"""Deterministic renderer and analytic graph arm for cycle multiplicity."""

from __future__ import annotations

import random
from collections import defaultdict
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 640
BACKGROUND = (250, 250, 247)
GRAY = (105, 112, 124)
BLACK = (24, 28, 34)
PALETTE = {
    "red": (218, 55, 67),
    "green": (24, 151, 96),
    "blue": (47, 101, 211),
}

_BASE_NODES = {
    "a": (90, 320), "b": (270, 320),
    "u1": (145, 205), "u2": (225, 220),
    "m1": (150, 315), "m2": (225, 285),
    "l1": (145, 430), "l2": (225, 415),
    "x1": (325, 265), "x2": (385, 330),
    "q0": (445, 280), "q1": (510, 200), "r": (580, 245),
    "q2": (565, 350), "q3": (475, 385),
}

_BASE_EDGES = [
    ("a", "u1"), ("u1", "u2"), ("u2", "b"),
    ("a", "m1"), ("m1", "m2"), ("m2", "b"),
    ("a", "l1"), ("l1", "l2"), ("l2", "b"),
    ("b", "x1"), ("x1", "x2"), ("x2", "q0"),
    ("q0", "q1"), ("q1", "r"), ("r", "q2"),
    ("q2", "q3"), ("q3", "q0"),
]

_ROLE_EDGES = {2: ("u1", "u2"), 1: ("q1", "r"), 0: ("x1", "x2")}


def _transform(point: tuple[int, int], quarter_turns: int, mirror: bool,
               dx: int, dy: int) -> tuple[int, int]:
    x, y = point
    x -= SIZE // 2
    y -= SIZE // 2
    if mirror:
        x = -x
    for _ in range(quarter_turns):
        x, y = -y, x
    return (x + SIZE // 2 + dx, y + SIZE // 2 + dy)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer = ("red", "green", "blue")[int(seed) % 3]
    others = [color for color in PALETTE if color != answer]
    rng.shuffle(others)
    role_colors = {2: answer, 1: others[0], 0: others[1]}
    turns = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    dx, dy = rng.randint(-8, 8), rng.randint(-8, 8)
    nodes = {
        name: list(_transform(point, turns, mirror, dx, dy))
        for name, point in _BASE_NODES.items()
    }
    colored_edges = [
        {"edge": list(_ROLE_EDGES[role]), "color": role_colors[role]}
        for role in (0, 1, 2)
    ]
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in _BASE_EDGES],
        "colored_edges": colored_edges,
    }


def _edge_key(edge) -> tuple[str, str]:
    a, b = edge
    return tuple(sorted((str(a), str(b))))


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    colored_edges = scene["colored_edges"]
    colors = {_edge_key(item["edge"]): PALETTE[item["color"]] for item in colored_edges}
    image = Image.new("RGB", (SIZE, SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for edge in edges:
        a, b = edge
        draw.line([tuple(nodes[a]), tuple(nodes[b])], fill=colors.get(_edge_key(edge), GRAY),
                  width=11)
    for x, y in nodes.values():
        draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=BLACK)
    return image


def _simple_path_count(edges, start: str, end: str, removed) -> int:
    removed_key = _edge_key(removed)
    graph: dict[str, list[str]] = defaultdict(list)
    for edge in edges:
        if _edge_key(edge) == removed_key:
            continue
        a, b = map(str, edge)
        graph[a].append(b)
        graph[b].append(a)

    def visit(node: str, seen: set[str]) -> int:
        if node == end:
            return 1
        return sum(visit(nxt, seen | {nxt}) for nxt in graph[node] if nxt not in seen)

    return visit(start, {start})


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    colored_edges = scene["colored_edges"]
    matches = []
    for item in colored_edges:
        a, b = map(str, item["edge"])
        if _simple_path_count(edges, a, b, item["edge"]) == 2:
            matches.append(str(item["color"]))
    if len(matches) != 1:
        raise ValueError(f"expected one two-loop link, found {matches}")
    return matches[0]


def margin(scene: dict) -> float:
    edges = scene["edges"]
    colored_edges = scene["colored_edges"]
    counts = []
    for item in colored_edges:
        a, b = map(str, item["edge"])
        counts.append(_simple_path_count(edges, a, b, item["edge"]))
    target_gap = min(abs(value - 2) for value in counts if value != 2)
    return float(target_gap)


def is_quarantined(scene: dict) -> bool:
    # The construction uses wide links, large junctions, and no non-junction crossings.
    _ = scene["nodes"]
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = deepcopy(scene)
    return []
