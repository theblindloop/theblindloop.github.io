"""Jog-indexed rail walk: a distant staircase counts, a ringed rail orders.

Two widely separated structures sit in diagonally opposite corners of the
canvas.  One corner holds a teal staircase wire whose number of downward drops
is an index N in {1,2,3}.  The opposite corner holds a dark serpentine rail with
three coloured bands painted into it, plus a hollow green marker parked just
beyond one of the rail's two loose ends.  The answer is the colour of the N-th
band met when walking the rail away from the ringed end.

Every drawn element is an axis-aligned rectangle (the marker is a rectangle
outline), so the raster is crisp and byte-deterministic.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 700
HEIGHT = 560

BG = (250, 250, 248)
TEAL = (18, 142, 150)
RAIL = (44, 48, 58)
ORANGE = (232, 124, 28)
PURPLE = (124, 56, 182)
YELLOW = (232, 214, 46)
GREEN = (32, 158, 72)
GRAY = (150, 152, 158)

COLORS = ("orange", "purple", "yellow")
_RGB = {"orange": ORANGE, "purple": PURPLE, "yellow": YELLOW}

BOX_W = 290
BOX_H = 230
CORNERS = {
    "TL": (24, 24),
    "TR": (WIDTH - 24 - BOX_W, 24),
    "BL": (24, HEIGHT - 24 - BOX_H),
    "BR": (WIDTH - 24 - BOX_W, HEIGHT - 24 - BOX_H),
}
OPPOSITE = {"TL": "BR", "BR": "TL", "TR": "BL", "BL": "TR"}

HALF = 4              # half width of every stroke (stroke width 9)
PIECE_HALF = 13       # half length of a coloured band (band length 27)
RING_OFF = 40         # marker centre offset beyond the ringed rail end
RING_HALF = 13
RING_W = 4
DECOY_HALF = 8
DECOY_CLEAR = 24

QUARANTINE_MARGIN = 12.0


def _bbox(rects: list[tuple[int, int, int, int]]) -> tuple[int, int, int, int]:
    return (
        min(r[0] for r in rects),
        min(r[1] for r in rects),
        max(r[2] for r in rects),
        max(r[3] for r in rects),
    )


def _gap(a: tuple[int, int, int, int], b: tuple[int, int, int, int]) -> float:
    gx = max(b[0] - a[2] - 1, a[0] - b[2] - 1)
    gy = max(b[1] - a[3] - 1, a[1] - b[3] - 1)
    return float(max(gx, gy))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    wire_corner = rng.choice(["TL", "TR", "BL", "BR"])
    rail_corner = OPPOSITE[wire_corner]
    tight = rng.random() < 0.3

    # ---- teal staircase: N downward drops, N+1 horizontal runs ----------
    drops_n = rng.choice([1, 2, 3])
    lo, hi = (34, 38) if tight else (40, 56)
    max_run = min(hi, (BOX_W - 56) // (drops_n + 1))
    max_drop = min(hi, (BOX_H - 60) // drops_n)
    runs = [rng.randint(lo, max(lo, max_run)) for _ in range(drops_n + 1)]
    drops = [rng.randint(lo, max(lo, max_drop)) for _ in range(drops_n)]

    wx, wy = CORNERS[wire_corner]
    sx = wx + 20 + rng.randint(0, max(0, BOX_W - 40 - sum(runs)))
    sy = wy + 20 + rng.randint(0, max(0, BOX_H - 40 - sum(drops)))
    points = [[sx, sy]]
    x, y = sx, sy
    for index in range(drops_n + 1):
        x += runs[index]
        points.append([x, y])
        if index < drops_n:
            y += drops[index]
            points.append([x, y])

    # ---- dark serpentine rail ------------------------------------------
    bx, by = CORNERS[rail_corner]
    rail_a = bx + 56
    rail_b = bx + BOX_W - 56
    row0 = by + rng.randint(32, 42)
    row2 = by + BOX_H - rng.randint(32, 42)
    row1 = (row0 + row2) // 2 + rng.randint(-8, 8)
    rows = [row0, row1, row2]

    # ---- three coloured bands spread over the three rail rows ----------
    counts = rng.choice(
        [(1, 1, 1), (1, 1, 1), (2, 1, 0), (0, 1, 2), (2, 0, 1), (1, 2, 0), (1, 0, 2), (0, 2, 1)]
    )
    sep = 40 if tight else 46
    low_x = rail_a + 36
    high_x = rail_b - 36
    pieces: list[dict] = []
    for row_index, count in enumerate(counts):
        if count == 1:
            pieces.append({"row": row_index, "x": rng.randint(low_x, high_x)})
        elif count == 2:
            first = rng.randint(low_x, high_x - sep)
            second = rng.randint(first + sep, high_x)
            pieces.append({"row": row_index, "x": first})
            pieces.append({"row": row_index, "x": second})
    palette = list(COLORS)
    rng.shuffle(palette)
    for piece, colour in zip(pieces, palette):
        piece["color"] = colour

    start_end = rng.choice(["head", "tail"])
    if start_end == "head":
        ring = [rail_a - RING_OFF, row0]
    else:
        ring = [rail_b + RING_OFF, row2]

    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "wire_corner": wire_corner,
        "rail_corner": rail_corner,
        "wire_points": points,
        "rail_a": rail_a,
        "rail_b": rail_b,
        "rail_rows": rows,
        "pieces": pieces,
        "ring": ring,
        "decoys": [],
    }

    # ---- gray decoys in the two unused corners and the central band ----
    zones = [CORNERS[key] for key in CORNERS if key not in (wire_corner, rail_corner)]
    zones = [(zx + 20, zy + 20, BOX_W - 40, BOX_H - 40) for zx, zy in zones]
    zones.append((240, 210, 220, 140))
    fixed = [item["rect"] for item in _elements(scene)]
    decoys: list[list[int]] = []
    for _ in range(rng.choice([2, 3, 4, 5])):
        zx, zy, zw, zh = zones[rng.randrange(len(zones))]
        px = zx + rng.randint(0, zw)
        py = zy + rng.randint(0, zh)
        box = (px - DECOY_HALF, py - DECOY_HALF, px + DECOY_HALF, py + DECOY_HALF)
        if box[0] < 8 or box[1] < 8 or box[2] > WIDTH - 9 or box[3] > HEIGHT - 9:
            continue
        others = fixed + [
            (dx - DECOY_HALF, dy - DECOY_HALF, dx + DECOY_HALF, dy + DECOY_HALF)
            for dx, dy in decoys
        ]
        if all(_gap(box, other) >= DECOY_CLEAR for other in others):
            decoys.append([px, py])
    scene["decoys"] = decoys
    return scene


def _elements(scene: dict) -> list[dict]:
    """Every drawn rectangle with its role, in painting order."""

    out: list[dict] = []
    for px, py in scene["decoys"]:
        out.append(
            {
                "role": "decoy",
                "color": GRAY,
                "rect": (px - DECOY_HALF, py - DECOY_HALF, px + DECOY_HALF, py + DECOY_HALF),
            }
        )

    points = scene["wire_points"]
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        if y0 == y1:
            rect = (min(x0, x1) - HALF, y0 - HALF, max(x0, x1) + HALF, y0 + HALF)
        else:
            rect = (x0 - HALF, min(y0, y1) - HALF, x0 + HALF, max(y0, y1) + HALF)
        out.append({"role": "wire", "color": TEAL, "rect": rect})

    rail_a = scene["rail_a"]
    rail_b = scene["rail_b"]
    rows = scene["rail_rows"]
    for row in rows:
        out.append(
            {"role": "rail", "color": RAIL, "rect": (rail_a - HALF, row - HALF, rail_b + HALF, row + HALF)}
        )
    for index, link_x in enumerate((rail_b, rail_a)):
        top, bottom = rows[index], rows[index + 1]
        out.append(
            {
                "role": "rail",
                "color": RAIL,
                "rect": (link_x - HALF, top - HALF, link_x + HALF, bottom + HALF),
            }
        )

    for piece in scene["pieces"]:
        row = rows[piece["row"]]
        out.append(
            {
                "role": f"band_{piece['color']}",
                "color": _RGB[piece["color"]],
                "rect": (piece["x"] - PIECE_HALF, row - HALF, piece["x"] + PIECE_HALF, row + HALF),
            }
        )

    cx, cy = scene["ring"]
    out.append(
        {
            "role": "ring",
            "color": GREEN,
            "rect": (cx - RING_HALF, cy - RING_HALF, cx + RING_HALF, cy + RING_HALF),
            "outline": True,
        }
    )
    return out


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BG)
    draw = ImageDraw.Draw(image)
    for item in _elements(scene):
        if item.get("outline"):
            draw.rectangle(item["rect"], outline=item["color"], width=RING_W)
        else:
            draw.rectangle(item["rect"], fill=item["color"])
    return image


def drop_count(scene: dict) -> int:
    """Number of downward drops of the staircase, read off the drawn polyline."""

    points = scene["wire_points"]
    return sum(1 for (x0, _), (x1, _) in zip(points, points[1:]) if x0 == x1)


def step_lengths(scene: dict) -> list[int]:
    points = scene["wire_points"]
    return [
        abs(x1 - x0) + abs(y1 - y0) for (x0, y0), (x1, y1) in zip(points, points[1:])
    ]


def rail_ends(scene: dict) -> list[tuple[int, int]]:
    rows = scene["rail_rows"]
    return [(scene["rail_a"], rows[0]), (scene["rail_b"], rows[2])]


def marked_end(scene: dict) -> int:
    """Index of the rail end the green marker is parked beyond."""

    cx, cy = scene["ring"]
    distances = [((cx - ex) ** 2 + (cy - ey) ** 2) ** 0.5 for ex, ey in rail_ends(scene)]
    return 0 if distances[0] <= distances[1] else 1


def _arc_lengths(scene: dict) -> list[float]:
    """Distance of each coloured band centre from the ringed end, along the rail."""

    rail_a = scene["rail_a"]
    rail_b = scene["rail_b"]
    row0, row1, row2 = scene["rail_rows"]
    span = rail_b - rail_a
    total = 3 * span + (row2 - row0)
    out = []
    for piece in scene["pieces"]:
        row = piece["row"]
        if row == 0:
            value = piece["x"] - rail_a
        elif row == 1:
            value = span + (row1 - row0) + (rail_b - piece["x"])
        else:
            value = 2 * span + (row2 - row0) + (piece["x"] - rail_a)
        if marked_end(scene) == 1:
            value = total - value
        out.append(float(value))
    return out


def _walk_order(scene: dict) -> list[dict]:
    keyed = sorted(zip(_arc_lengths(scene), range(len(scene["pieces"]))))
    return [scene["pieces"][index] for _, index in keyed]


def analytic_gold(scene: dict) -> str:
    return _walk_order(scene)[drop_count(scene) - 1]["color"]


def margin(scene: dict) -> float:
    items = _elements(scene)
    groups: dict[str, list[tuple[int, int, int, int]]] = {}
    for index, item in enumerate(items):
        key = item["role"] if item["role"] in ("wire", "rail", "ring") else f"{item['role']}#{index}"
        if item["role"].startswith("band"):
            key = "rail"          # bands are painted into the rail itself
        groups.setdefault(key, []).append(item["rect"])
    boxes = {key: _bbox(rects) for key, rects in groups.items()}
    keys = sorted(boxes)
    clearance = min(
        _gap(boxes[keys[i]], boxes[keys[j]])
        for i in range(len(keys))
        for j in range(i + 1, len(keys))
    )

    staircase = min(step_lengths(scene)) - 22.0

    lengths = sorted(_arc_lengths(scene))
    band_gap = min(
        (lengths[i + 1] - lengths[i]) - 2.0 * PIECE_HALF for i in range(len(lengths) - 1)
    )

    ends = rail_ends(scene)
    cx, cy = scene["ring"]
    distances = sorted(((cx - ex) ** 2 + (cy - ey) ** 2) ** 0.5 for ex, ey in ends)
    ring_edge = (distances[1] - distances[0]) / 2.0

    return round(float(min(clearance, staircase, band_gap, ring_edge)), 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Latent rewrites that are pixel-identical and answer-preserving."""

    out: list[tuple[str, dict]] = []
    flipped = dict(scene)
    flipped["decoys"] = list(reversed(scene["decoys"]))
    out.append(("decoy_order_reversed", flipped))

    rotated = dict(scene)
    pieces = list(scene["pieces"])
    if pieces:
        pieces = pieces[1:] + pieces[:1]
    rotated["pieces"] = pieces
    out.append(("band_list_rotated", rotated))
    return out
