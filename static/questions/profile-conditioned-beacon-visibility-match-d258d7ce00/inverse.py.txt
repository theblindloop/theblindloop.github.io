from __future__ import annotations

import itertools
import numpy as np
from PIL import Image

BEACONS = [(25,170,205), (210,55,145), (45,165,75), (235,125,30)]
BORDERS = {"red": (205,45,45), "blue": (45,85,205), "gold": (210,155,25)}
NEUTRAL = (110,110,118)
WALL = (55,55,62)


def _bbox(mask):
    ys, xs = np.nonzero(mask)
    if len(xs) == 0: raise ValueError("required visible component is absent")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _centroid(mask):
    ys, xs = np.nonzero(mask)
    if len(xs) < 20: raise ValueError("beacon component is absent")
    return float(xs.mean()), float(ys.mean())


def _segment_hits_rect(a, b, rect):
    x0, y0 = a; x1, y1 = b
    left, top, right, bottom = rect
    dx, dy = x1-x0, y1-y0
    lo, hi = 0.0, 1.0
    for p, q in ((-dx,x0-left), (dx,right-x0), (-dy,y0-top), (dy,bottom-y0)):
        if p == 0:
            if q < 0: return False
        else:
            t = q/p
            if p < 0: lo = max(lo,t)
            else: hi = min(hi,t)
            if lo > hi: return False
    return True


def _signature(arr, box):
    left, top, right, bottom = box
    crop = arr[top+6:bottom-5, left+6:right-5]
    points = []
    for rgb in BEACONS:
        points.append(_centroid(np.all(crop == np.array(rgb, dtype=np.uint8), axis=2)))
    wall_mask = np.all(crop == np.array(WALL, dtype=np.uint8), axis=2)
    # Connected components by simple flood fill; walls never touch.
    seen = np.zeros(wall_mask.shape, dtype=bool); walls = []
    h, w = wall_mask.shape
    for y, x in zip(*np.nonzero(wall_mask)):
        if seen[y,x]: continue
        stack=[(y,x)]; seen[y,x]=True; xs=[]; ys=[]
        while stack:
            cy,cx=stack.pop(); xs.append(cx); ys.append(cy)
            for ny,nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0<=ny<h and 0<=nx<w and wall_mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx]=True; stack.append((ny,nx))
        if len(xs)>100: walls.append((min(xs),min(ys),max(xs),max(ys)))
    if len(walls) != 2: raise ValueError("expected two separate opaque walls")
    return tuple((i,j) for i,j in itertools.combinations(range(4),2)
                 if not any(_segment_hits_rect(points[i],points[j],wall) for wall in walls))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    ref_box = _bbox(np.all(arr == np.array(NEUTRAL,dtype=np.uint8), axis=2))
    reference = _signature(arr, ref_box)
    matches=[]
    for name, rgb in BORDERS.items():
        box = _bbox(np.all(arr == np.array(rgb,dtype=np.uint8), axis=2))
        if _signature(arr, box) == reference: matches.append(name)
    if len(matches) != 1: raise ValueError("pixel visibility match is not unique")
    return matches[0]
