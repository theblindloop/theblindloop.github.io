from __future__ import annotations

import copy
import itertools
import math
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 840, 640
CARD_W, CARD_H = 240, 190
BEACON_COLORS = ["cyan", "magenta", "green", "orange"]
RGB = {
    "cyan": (25, 170, 205), "magenta": (210, 55, 145),
    "green": (45, 165, 75), "orange": (235, 125, 30),
    "red": (205, 45, 45), "blue": (45, 85, 205), "gold": (210, 155, 25),
}
WALL = (55, 55, 62)
NEUTRAL = (110, 110, 118)

POINTS = [[45, 35], [185, 35], [50, 145], [185, 145]]
# Every layout contains one 18x70 vertical and one 70x18 horizontal wall.
# They have distinct induced visibility graphs and at least seven pixels of
# center-line clearance from every blocking/nonblocking transition.
LAYOUTS = [
    [[142, 93, 160, 163], [26, 103, 96, 121]],
    [[141, 83, 159, 153], [23, 80, 93, 98]],
    [[153, 20, 171, 90], [101, 95, 171, 113]],
    [[112, 47, 130, 117], [131, 93, 201, 111]],
    [[126, 94, 144, 164], [145, 110, 215, 128]],
    [[70, 19, 88, 89], [146, 80, 216, 98]],
    [[75, 87, 93, 157], [60, 60, 130, 78]],
    [[70, 94, 88, 164], [125, 61, 195, 79]],
    [[111, 45, 129, 115], [97, 116, 167, 134]],
    [[119, 26, 137, 96], [36, 70, 106, 88]],
    [[100, 24, 118, 94], [125, 69, 195, 87]],
    [[116, 43, 134, 113], [15, 81, 85, 99]],
]


def _segment_hits_rect(a, b, rect):
    x0, y0 = a; x1, y1 = b
    left, top, right, bottom = rect
    dx, dy = x1 - x0, y1 - y0
    lo, hi = 0.0, 1.0
    for p, q in ((-dx, x0-left), (dx, right-x0), (-dy, y0-top), (dy, bottom-y0)):
        if p == 0:
            if q < 0: return False
        else:
            t = q / p
            if p < 0: lo = max(lo, t)
            else: hi = min(hi, t)
            if lo > hi: return False
    return True


def _signature_from_arrays(points, walls):
    return tuple((i, j) for i, j in itertools.combinations(range(4), 2)
                 if not any(_segment_hits_rect(points[i], points[j], wall) for wall in walls))


def _flip_card(card, flip_x, flip_y):
    points = [[CARD_W-x if flip_x else x, CARD_H-y if flip_y else y]
              for x, y in card["points"]]
    walls = []
    for left, top, right, bottom in card["walls"]:
        if flip_x: left, right = CARD_W-right, CARD_W-left
        if flip_y: top, bottom = CARD_H-bottom, CARD_H-top
        walls.append([left, top, right, bottom])
    return {"points": points, "walls": walls}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    ref_i = rng.randrange(len(LAYOUTS))
    others = [i for i in range(len(LAYOUTS)) if i != ref_i]
    rng.shuffle(others)
    chosen = [ref_i, others[0], others[1]]
    target_pos = seed % 3
    option_layouts = [None, None, None]
    option_layouts[target_pos] = ref_i
    distractor_positions = [i for i in range(3) if i != target_pos]
    option_layouts[distractor_positions[0]] = chosen[1]
    option_layouts[distractor_positions[1]] = chosen[2]
    ref = {"points": copy.deepcopy(POINTS), "walls": copy.deepcopy(LAYOUTS[ref_i])}
    cards = [ref]
    for index in option_layouts:
        raw = {"points": copy.deepcopy(POINTS), "walls": copy.deepcopy(LAYOUTS[index])}
        cards.append(_flip_card(raw, bool(rng.randrange(2)), bool(rng.randrange(2))))
    base = ["red", "blue", "gold"]
    shift = (seed // 3) % 3
    borders = base[shift:] + base[:shift]
    return {"cards": cards, "borders": borders}


def analytic_gold(scene: dict) -> str:
    cards = scene["cards"]
    borders = scene["borders"]
    reference = _signature_from_arrays(cards[0]["points"], cards[0]["walls"])
    matches = [i for i in range(3)
               if _signature_from_arrays(cards[i+1]["points"], cards[i+1]["walls"]) == reference]
    if len(matches) != 1: raise ValueError("visibility correspondence is not unique")
    return borders[matches[0]]


def _point_rect_signed_distance(point, rect):
    x, y = point; left, top, right, bottom = rect
    dx, dy = max(left-x, 0, x-right), max(top-y, 0, y-bottom)
    if dx or dy: return math.hypot(dx, dy)
    return -min(x-left, right-x, y-top, bottom-y)


def _card_clearance(card):
    values = []
    for i, j in itertools.combinations(range(4), 2):
        wall_minima = []
        a, b = card["points"][i], card["points"][j]
        for wall in card["walls"]:
            samples = [_point_rect_signed_distance(
                (a[0] + (b[0]-a[0])*k/200, a[1] + (b[1]-a[1])*k/200), wall)
                for k in range(201)]
            wall_minima.append(min(samples))
        if any(v <= 0 for v in wall_minima):
            values.append(max(-v for v in wall_minima if v <= 0))
        else:
            values.append(min(wall_minima))
    return min(values)


def margin(scene: dict) -> float:
    cards = scene["cards"]
    borders = scene["borders"]
    # Borders are intentionally accessed because they participate in the final
    # selection label, while clearance is the geometric runner-up safety margin.
    if len(borders) != 3: return 0.0
    return round(min(_card_clearance(card) for card in cards), 6)


def is_quarantined(scene: dict) -> bool:
    cards = scene["cards"]
    borders = scene["borders"]
    if len(set(borders)) != 3: return True
    ref = _signature_from_arrays(cards[0]["points"], cards[0]["walls"])
    count = sum(_signature_from_arrays(c["points"], c["walls"]) == ref for c in cards[1:])
    return count != 1 or min(_card_clearance(c) for c in cards) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for card in twin["cards"]:
        card["walls"].reverse()
    return [("reverse_wall_storage_order", twin)]


def _draw_card(draw, origin, card, border_rgb, label):
    ox, oy = origin
    draw.rectangle([ox, oy, ox+CARD_W, oy+CARD_H], fill=(255,255,255), outline=border_rgb, width=5)
    for left, top, right, bottom in card["walls"]:
        # Treat latent rectangles as half-open so neighboring catalog walls do
        # not become a single 4-connected component at raster resolution.
        draw.rectangle([ox+left, oy+top, ox+right-1, oy+bottom-1], fill=WALL)
    for color, (x, y) in zip(BEACON_COLORS, card["points"]):
        draw.ellipse([ox+x-10, oy+y-10, ox+x+10, oy+y+10], fill=RGB[color], outline=(25,25,28), width=2)
    draw.text((ox+10, oy+CARD_H-18), label, fill=(25,25,28), font=ImageFont.load_default())


def render(scene: dict) -> Image.Image:
    cards = scene["cards"]
    borders = scene["borders"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (246, 244, 238))
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH//2, 18), "MATCH THE COMPLETE BEACON VISIBILITY GRAPH", anchor="ma", fill=(25,25,28), font=ImageFont.load_default())
    _draw_card(draw, (300, 48), cards[0], NEUTRAL, "REFERENCE")
    origins = [(30, 380), (300, 380), (570, 380)]
    for i, origin in enumerate(origins):
        _draw_card(draw, origin, cards[i+1], RGB[borders[i]], "OPTION")
    return image
