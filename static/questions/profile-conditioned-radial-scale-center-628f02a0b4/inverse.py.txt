"""Independent PNG-only inverse arm for the radial scale-gradient decision."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

BLACK = np.array((30, 34, 40), dtype=np.int16)
RED = np.array((218, 55, 63), dtype=np.int16)
BLUE = np.array((42, 103, 221), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        stack = [(int(y0), int(x0))]
        seen[y0, x0] = True
        xs, ys = [], []
        while stack:
            y, x = stack.pop()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y-1, x), (y+1, x), (y, x-1), (y, x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 20:
            result.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return result


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _ranks(values: list[float]) -> list[float]:
    order = sorted(range(len(values)), key=lambda i: values[i])
    result = [0.0] * len(values)
    start = 0
    while start < len(order):
        stop = start + 1
        while stop < len(order) and values[order[stop]] == values[order[start]]:
            stop += 1
        average = (start + stop - 1) / 2.0
        for index in order[start:stop]:
            result[index] = average
        start = stop
    return result


def _corr(a: list[float], b: list[float]) -> float:
    ma, mb = sum(a) / len(a), sum(b) / len(b)
    da, db = [x-ma for x in a], [x-mb for x in b]
    den = math.sqrt(sum(x*x for x in da) * sum(x*x for x in db))
    return 0.0 if den == 0 else sum(x*y for x, y in zip(da, db)) / den


def _marker(arr: np.ndarray, color: np.ndarray) -> tuple[float, float]:
    comps = _components(_mask(arr, color))
    if len(comps) != 1:
        raise ValueError(f"expected one colored marker, found {len(comps)}")
    return comps[0][0], comps[0][1]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    dots = _components(_mask(arr, BLACK))
    if len(dots) != 16:
        raise ValueError(f"expected 16 black disks, found {len(dots)}")
    centers = [(x, y) for x, y, _ in dots]
    size_ranks = _ranks([float(area) for _, _, area in dots])
    scores = {}
    for name, color in (("red", RED), ("blue", BLUE)):
        mx, my = _marker(arr, color)
        distances = [math.hypot(x-mx, y-my) for x, y in centers]
        scores[name] = _corr(_ranks(distances), size_ranks)
    if abs(scores["red"] - scores["blue"]) < 0.18:
        raise ValueError("ambiguous radial scale gradient")
    return max(("red", "blue"), key=lambda name: scores[name])
