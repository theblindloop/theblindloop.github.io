"""Renderer and analytic arm for a distributed radial size-gradient field."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 246, 241)
BLACK = (30, 34, 40)
RED = (218, 55, 63)
BLUE = (42, 103, 221)
QUARANTINE_MARGIN = 0.18


def _ranks(values: Iterable[float]) -> list[float]:
    values = list(values)
    order = sorted(range(len(values)), key=lambda i: values[i])
    ranks = [0.0] * len(values)
    start = 0
    while start < len(order):
        stop = start + 1
        while stop < len(order) and values[order[stop]] == values[order[start]]:
            stop += 1
        average = (start + stop - 1) / 2.0
        for index in order[start:stop]:
            ranks[index] = average
        start = stop
    return ranks


def _pearson(a: list[float], b: list[float]) -> float:
    ma = sum(a) / len(a)
    mb = sum(b) / len(b)
    da = [x - ma for x in a]
    db = [x - mb for x in b]
    den = math.sqrt(sum(x * x for x in da) * sum(x * x for x in db))
    return 0.0 if den == 0 else sum(x * y for x, y in zip(da, db)) / den


def _score(dots: list[dict], marker: list[int]) -> float:
    distances = [math.hypot(d["x"] - marker[0], d["y"] - marker[1]) for d in dots]
    radii = [float(d["r"]) for d in dots]
    return _pearson(_ranks(distances), _ranks(radii))


def _scores(scene: dict) -> dict[str, float]:
    return {
        color: _score(scene["dots"], scene["markers"][color])
        for color in ("red", "blue")
    }


def sample_scene(seed: int) -> dict:
    """Build one balanced scene; no answer field is retained in the latent."""
    rng = random.Random(int(seed))
    target_color = "red" if int(seed) % 2 == 0 else "blue"
    for _ in range(200):
        # Physical anchor assignment is randomized independently of answer color.
        y_left = rng.choice((92, 156, 356, 420))
        y_right = rng.choice((92, 156, 356, 420))
        physical = [[48, y_left], [464, y_right]]
        rng.shuffle(physical)
        markers = {"red": physical[0], "blue": physical[1]}

        centers = []
        for gy in range(4):
            for gx in range(4):
                centers.append(
                    [
                        104 + gx * 101 + rng.randint(-17, 17),
                        82 + gy * 116 + rng.randint(-17, 17),
                    ]
                )
        target = markers[target_color]
        order = sorted(
            range(len(centers)),
            key=lambda i: (math.hypot(centers[i][0] - target[0], centers[i][1] - target[1]), i),
        )
        radii = [0] * len(centers)
        for rank, index in enumerate(order):
            radii[index] = 7 + rank
        dots = [
            {"x": int(center[0]), "y": int(center[1]), "r": int(radius)}
            for center, radius in zip(centers, radii)
        ]
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "dots": dots,
            "markers": markers,
            "scale": 1,
        }
        scores = _scores(scene)
        if abs(scores["red"] - scores["blue"]) >= 0.30:
            return scene
    raise RuntimeError("could not sample a separated radial-gradient scene")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    scale = int(scene["scale"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for dot in scene["dots"]:
        x, y, r = (int(dot[k]) * scale for k in ("x", "y", "r"))
        draw.ellipse((x - r, y - r, x + r, y + r), fill=BLACK)
    for color, rgb in (("red", RED), ("blue", BLUE)):
        x, y = (int(v) * scale for v in scene["markers"][color])
        r = 11 * scale
        # Equal-size colored diamonds are visually distinct from the black disks.
        draw.polygon([(x, y-r), (x+r, y), (x, y+r), (x-r, y)], fill=rgb)
        draw.ellipse((x-3*scale, y-3*scale, x+3*scale, y+3*scale), fill=BACKGROUND)
    return image


def analytic_gold(scene: dict) -> str:
    scores = {
        color: _score(scene["dots"], scene["markers"][color])
        for color in ("red", "blue")
    }
    return max(("red", "blue"), key=lambda color: scores[color])


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return abs(scores["red"] - scores["blue"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "width": scene["width"],
        "height": scene["height"],
        "dots": list(reversed(scene["dots"])),
        "markers": {"red": list(scene["markers"]["red"]), "blue": list(scene["markers"]["blue"])},
        "scale": scene["scale"],
    }
    return [("reverse_dot_record_order", twin)]
