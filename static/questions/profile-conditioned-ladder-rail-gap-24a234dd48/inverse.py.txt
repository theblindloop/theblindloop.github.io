"""Independent pixel-only inverse arm for the ladder rail-gap decision.

Only the final RGB raster is accepted.  Saturated rail pixels are isolated from
the neutral plaque and rungs, then their clear separation is measured near the
two visible rail endpoint bands.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _band_gap(mask: np.ndarray, y0: int, y1: int, center: float) -> float:
    gaps: list[float] = []
    for y in range(max(0, y0), min(mask.shape[0], y1)):
        xs = np.flatnonzero(mask[y])
        left = xs[xs < center]
        right = xs[xs > center]
        if left.size >= 5 and right.size >= 5:
            gaps.append(float(right.min() - left.max() - 1))
    if len(gaps) < 4:
        raise ValueError("rail evidence is incomplete at an endpoint")
    return float(np.median(gaps))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = rgb.max(axis=2) - rgb.min(axis=2)
    # Every permitted rail palette is strongly chromatic; all other artwork is gray.
    mask = (chroma >= 55) & (rgb.max(axis=2) <= 235)
    ys, xs = np.nonzero(mask)
    if xs.size < 4500:
        raise ValueError("insufficient colored rail evidence")
    y_min = int(np.percentile(ys, 0.2))
    y_max = int(np.percentile(ys, 99.8))
    if y_max - y_min < 270:
        raise ValueError("colored rails do not span both endpoint regions")
    center = float(np.median(xs))
    band = max(10, int(round((y_max - y_min) * 0.06)))
    top_gap = _band_gap(mask, y_min + 2, y_min + band, center)
    bottom_gap = _band_gap(mask, y_max - band, y_max - 2, center)
    difference = top_gap - bottom_gap
    if difference < -8.0:
        return "narrower"
    if difference > 8.0:
        return "wider"
    return "equal"
