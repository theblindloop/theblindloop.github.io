"""Deterministic Pillow renderer for a prior-conflicting ladder.

Gold is computed only from ``gap_relation``.  Rendering exposes that label as
the difference between the two colored rails' clear endpoint gaps.  Neutral
rungs and chrome never carry the answer.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
SCALE = 3
RELATIONS = ("narrower", "equal", "wider")
PALETTE = (
    (24, 112, 196),
    (198, 58, 54),
    (25, 145, 92),
    (151, 75, 190),
)
DELTAS = (48, 60)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0x51A77E)
    relation = RELATIONS[int(seed) % len(RELATIONS)]
    color = PALETTE[(int(seed) // len(RELATIONS)) % len(PALETTE)]
    rung_count = 5 + ((int(seed) // 12) % 2)
    order = list(range(rung_count))
    rng.shuffle(order)
    return {
        "gap_relation": relation,
        "center_x": rng.randint(244, 268),
        "top_y": rng.randint(108, 124),
        "bottom_y": rng.randint(410, 430),
        "bottom_gap": rng.choice((118, 132, 146)),
        "rail_width": rng.choice((18, 20, 22)),
        "rail_color": list(color),
        "rung_count": rung_count,
        "rung_order": order,
    }


def _geometry(scene: dict[str, Any]) -> dict[str, Any]:
    relation = scene["gap_relation"]
    bottom_gap = int(scene["bottom_gap"])
    delta = DELTAS[(bottom_gap // 2) % 2]
    top_gap = bottom_gap + {"narrower": -delta, "equal": 0, "wider": delta}[relation]
    cx = int(scene["center_x"])
    width = int(scene["rail_width"])
    top_y = int(scene["top_y"])
    bottom_y = int(scene["bottom_y"])
    return {
        "top_gap": top_gap,
        "bottom_gap": bottom_gap,
        "left_top_inner": cx - top_gap / 2,
        "right_top_inner": cx + top_gap / 2,
        "left_bottom_inner": cx - bottom_gap / 2,
        "right_bottom_inner": cx + bottom_gap / 2,
        "width": width,
        "top_y": top_y,
        "bottom_y": bottom_y,
    }


def _font(size: int) -> ImageFont.ImageFont:
    for path in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf",
    ):
        try:
            return ImageFont.truetype(path, size)
        except OSError:
            pass
    return ImageFont.load_default()


def render(scene: dict[str, Any]) -> Image.Image:
    # Explicit reads make the public rendered-field audit precise.
    color = tuple(int(v) for v in scene["rail_color"])
    rung_count = int(scene["rung_count"])
    rung_order = [int(v) for v in scene["rung_order"]]
    g = _geometry(scene)
    s = SCALE
    image = Image.new("RGB", (WIDTH * s, HEIGHT * s), (247, 247, 244))
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(round(v * s) for v in values)  # type: ignore[return-value]

    # Neutral player-card-like plaque establishes a coherent visual world but
    # contains no saturated pixels used by the inverse arm.
    draw.rounded_rectangle(box((54, 40, 458, 472)), radius=24 * s,
                           fill=(239, 240, 238), outline=(70, 73, 76), width=3 * s)
    draw.rounded_rectangle(box((74, 58, 438, 94)), radius=10 * s,
                           fill=(218, 220, 218), outline=(105, 108, 110), width=2 * s)
    label = "LADDER"
    font = _font(17 * s)
    bb = draw.textbbox((0, 0), label, font=font)
    draw.text(((WIDTH * s - (bb[2] - bb[0])) / 2, 66 * s), label,
              fill=(48, 50, 52), font=font)

    top_y = g["top_y"]
    bottom_y = g["bottom_y"]
    ys = [top_y + (i + 1) * (bottom_y - top_y) / (rung_count + 1)
          for i in range(rung_count)]
    left_at = lambda y: g["left_top_inner"] + (g["left_bottom_inner"] - g["left_top_inner"]) * ((y - top_y) / (bottom_y - top_y))
    right_at = lambda y: g["right_top_inner"] + (g["right_bottom_inner"] - g["right_top_inner"]) * ((y - top_y) / (bottom_y - top_y))

    # Reordering these disjoint rung bands is a real pixel-identical alias.
    for index in rung_order:
        y = ys[index]
        draw.rounded_rectangle(
            box((left_at(y) - 1, y - 6, right_at(y) + 1, y + 6)),
            radius=5 * s,
            fill=(91, 94, 96),
            outline=(54, 57, 59),
            width=2 * s,
        )

    w = g["width"]
    left_poly = [
        (g["left_top_inner"] - w, top_y),
        (g["left_top_inner"], top_y),
        (g["left_bottom_inner"], bottom_y),
        (g["left_bottom_inner"] - w, bottom_y),
    ]
    right_poly = [
        (g["right_top_inner"], top_y),
        (g["right_top_inner"] + w, top_y),
        (g["right_bottom_inner"] + w, bottom_y),
        (g["right_bottom_inner"], bottom_y),
    ]
    draw.polygon([(round(x * s), round(y * s)) for x, y in left_poly], fill=color)
    draw.polygon([(round(x * s), round(y * s)) for x, y in right_poly], fill=color)

    # Small neutral feet make the object unmistakably ladder-like.
    for x in (g["left_bottom_inner"] - w / 2, g["right_bottom_inner"] + w / 2):
        draw.rounded_rectangle(box((x - 16, bottom_y - 1, x + 16, bottom_y + 10)),
                               radius=4 * s, fill=(74, 77, 79))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    relation = str(scene["gap_relation"])
    if relation not in RELATIONS:
        raise ValueError(f"invalid relation: {relation}")
    return relation


def margin(scene: dict[str, Any]) -> float:
    g = _geometry(scene)
    difference = abs(float(g["top_gap"] - g["bottom_gap"]))
    # Exact parallel controls are 48 px from the nearest sampled alternative.
    return 48.0 if difference == 0 else difference


def is_quarantined(scene: dict[str, Any]) -> bool:
    g = _geometry(scene)
    difference = abs(float(g["top_gap"] - g["bottom_gap"]))
    return 0.0 < difference < 12.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["rung_order"] = list(reversed([int(v) for v in scene["rung_order"]]))
    return [("reverse_disjoint_rung_painter_order", twin)]
