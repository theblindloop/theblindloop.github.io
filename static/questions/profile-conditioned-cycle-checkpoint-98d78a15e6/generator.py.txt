"""Renderer and analytic arm for the cycle-checkpoint world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = (640, 520)
SCALE = 4
NODE_RADIUS = 14
COLORS = ("orange", "purple", "blue")
RGB = {
    "orange": (232, 112, 35),
    "purple": (139, 92, 246),
    "blue": (37, 99, 235),
}
INK = (39, 49, 66)
NEUTRAL = (203, 213, 225)
OUTLINE = (71, 85, 105)


def _two_core(edges: list[list[int]], n: int) -> set[int]:
    adj = [set() for _ in range(n)]
    for a, b in edges:
        adj[a].add(b)
        adj[b].add(a)
    alive = set(range(n))
    queue = [i for i in alive if len(adj[i]) < 2]
    while queue:
        u = queue.pop()
        if u not in alive:
            continue
        alive.remove(u)
        for v in list(adj[u]):
            adj[v].discard(u)
            if v in alive and len(adj[v]) < 2:
                queue.append(v)
    return alive


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # Nodes 0..3 form the sole cycle. Nodes 4 and 6 are degree-two tail
    # checkpoints, matching the degree of cycle checkpoint 0.
    base = [
        (-82, -67), (82, -67), (82, 67), (-82, 67),
        (166, -111), (224, -145), (-166, 111), (-224, 145),
    ]
    jittered = [(x + rng.randint(-5, 5), y + rng.randint(-5, 5)) for x, y in base]
    quarter_turns = rng.randrange(4)
    reflect = bool(rng.randrange(2))
    dx, dy = rng.randint(-8, 8), rng.randint(-8, 8)
    points = []
    for x, y in jittered:
        if reflect:
            x = -x
        for _ in range(quarter_turns):
            x, y = -y, x
        points.append([int(320 + dx + x), int(260 + dy + y)])

    edges = [[0, 1], [1, 2], [2, 3], [3, 0], [1, 4], [4, 5], [3, 6], [6, 7]]
    rng.shuffle(edges)
    answer_color = COLORS[int(seed) % len(COLORS)]
    other_colors = [c for c in COLORS if c != answer_color]
    rng.shuffle(other_colors)
    checkpoint_colors = [answer_color, *other_colors]
    return {
        "canvas": list(CANVAS),
        "nodes": points,
        "edges": edges,
        "checkpoint_nodes": [0, 4, 6],
        "checkpoint_colors": checkpoint_colors,
    }


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width * SCALE, height * SCALE), "white")
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for a, b in scene["edges"]:
        p = tuple(v * SCALE for v in nodes[a])
        q = tuple(v * SCALE for v in nodes[b])
        draw.line([p, q], fill=INK, width=8 * SCALE)

    color_for_node = dict(zip(scene["checkpoint_nodes"], scene["checkpoint_colors"], strict=True))
    r = NODE_RADIUS * SCALE
    for i, (x, y) in enumerate(nodes):
        cx, cy = x * SCALE, y * SCALE
        box = (cx - r, cy - r, cx + r, cy + r)
        fill = RGB[color_for_node[i]] if i in color_for_node else NEUTRAL
        draw.ellipse(box, fill=fill, outline=OUTLINE, width=3 * SCALE)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    core = _two_core(scene["edges"], max(max(edge) for edge in scene["edges"]) + 1)
    winners = [
        color
        for node, color in zip(scene["checkpoint_nodes"], scene["checkpoint_colors"], strict=True)
        if node in core
    ]
    if len(winners) != 1:
        raise ValueError(f"expected one colored checkpoint on cycle, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    """Runner-up gap in the binary cycle-membership score (1 versus 0)."""
    core = _two_core(scene["edges"], len(scene["nodes"]))
    scores = sorted((1.0 if n in core else 0.0 for n in scene["checkpoint_nodes"]), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    nodes = scene["nodes"]
    if margin(scene) < 1.0:
        return True
    min_edge = min(math.dist(nodes[a], nodes[b]) for a, b in scene["edges"])
    min_nodes = min(math.dist(nodes[a], nodes[b]) for a in range(len(nodes)) for b in range(a + 1, len(nodes)))
    return min_edge < 45 or min_nodes < 38


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["edges"] = list(reversed(twin["edges"]))
    return [("reverse_edge_storage_order", twin)]
