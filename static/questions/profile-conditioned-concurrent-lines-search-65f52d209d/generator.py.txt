"""Deterministic raster world for three-line concurrency search."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1000
HEIGHT = 350
LABELS = ("A", "B", "C", "D")
BASE_COLORS = ((202, 52, 96), (20, 145, 157), (226, 143, 24))


def _rotate(vector: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = vector
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return x, y


def _line_intersection(a: list[list[int]], b: list[list[int]]) -> tuple[float, float]:
    x1, y1 = a[0]
    x2, y2 = a[1]
    x3, y3 = b[0]
    x4, y4 = b[1]
    den = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
    if abs(den) < 1e-9:
        return 1e6, 1e6
    cross1 = x1 * y2 - y1 * x2
    cross2 = x3 * y4 - y3 * x4
    px = (cross1 * (x3 - x4) - (x1 - x2) * cross2) / den
    py = (cross1 * (y3 - y4) - (y1 - y2) * cross2) / den
    return px, py


def _panel_score(lines: list[list[list[int]]]) -> float:
    points = [
        _line_intersection(lines[0], lines[1]),
        _line_intersection(lines[0], lines[2]),
        _line_intersection(lines[1], lines[2]),
    ]
    cx = sum(point[0] for point in points) / 3.0
    cy = sum(point[1] for point in points) / 3.0
    return math.sqrt(sum((x - cx) ** 2 + (y - cy) ** 2 for x, y in points) / 3.0)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene without storing an answer-only variable."""

    rng = random.Random(int(seed))
    winner = int(seed) % 4
    quarter_turns = rng.randrange(4)
    base_vectors = [(70, 25), (18, 72), (-61, 42)]
    vectors = [_rotate(vector, quarter_turns) for vector in base_vectors]
    offset_scale = (0.5, 0.75, 1.0)[(int(seed) // 4) % 3]
    base_offsets = [(-24, 24, -24), (24, -16, 24), (0, 24, -24)]
    distractor_offsets = [
        tuple(round(value * offset_scale) for value in pattern) for pattern in base_offsets
    ]
    rng.shuffle(distractor_offsets)
    panels = []
    distractor_index = 0
    for panel_index in range(4):
        cx = 125 + panel_index * 250
        cy = 192
        if panel_index == winner:
            offsets = (0, 0, 0)
        else:
            offsets = distractor_offsets[distractor_index]
            distractor_index += 1
        lines = []
        for (dx, dy), offset in zip(vectors, offsets):
            length = math.hypot(dx, dy)
            shift_x = round(-dy * offset / length)
            shift_y = round(dx * offset / length)
            mx = cx + shift_x
            my = cy + shift_y
            lines.append([[mx - dx, my - dy], [mx + dx, my + dy]])
        panels.append({"lines": lines})
    palette = list(BASE_COLORS)
    rng.shuffle(palette)
    return {"panels": panels, "palette": [list(color) for color in palette]}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    palette = [tuple(color) for color in scene["palette"]]
    draw.text((WIDTH // 2, 20), "THREE-LINE CONCURRENCY", fill=(47, 49, 54), font=font, anchor="mm")
    for panel_index, panel in enumerate(scene["panels"]):
        left = panel_index * 250 + 15
        right = panel_index * 250 + 235
        draw.rounded_rectangle((left, 62, right, 320), radius=8, outline=(172, 174, 178), width=2)
        draw.text(((left + right) // 2, 47), LABELS[panel_index], fill=(35, 37, 42), font=font, anchor="mm")
        for color, endpoints in zip(palette, panel["lines"]):
            draw.line((tuple(endpoints[0]), tuple(endpoints[1])), fill=color, width=7)
    return image


def analytic_gold(scene: dict) -> str:
    scores = []
    for panel in scene["panels"]:
        lines = panel["lines"]
        scores.append(_panel_score(lines))
    return LABELS[min(range(4), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    scores = sorted(_panel_score(panel["lines"]) for panel in scene["panels"])
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
