"""Independent pixel-only oracle for the concurrency raster."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


LABELS = ("A", "B", "C", "D")
LINE_COLORS = ((202, 52, 96), (20, 145, 157), (226, 143, 24))


def _fit_line(points: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    center = points.mean(axis=0)
    centered = points - center
    _, _, vectors = np.linalg.svd(centered, full_matrices=False)
    direction = vectors[0]
    return center, direction


def _intersection(first: tuple[np.ndarray, np.ndarray], second: tuple[np.ndarray, np.ndarray]) -> np.ndarray:
    center_a, direction_a = first
    center_b, direction_b = second
    matrix = np.column_stack((direction_a, -direction_b))
    if abs(float(np.linalg.det(matrix))) < 0.08:
        raise ValueError("line fits are too nearly parallel")
    factors = np.linalg.solve(matrix, center_b - center_a)
    return center_a + factors[0] * direction_a


def _score_panel(array: np.ndarray, left: int, right: int) -> float:
    fits = []
    crop = array[:, left:right, :]
    for color in LINE_COLORS:
        target = np.asarray(color, dtype=np.int16)
        delta = np.abs(crop.astype(np.int16) - target)
        ys, xs = np.nonzero(np.max(delta, axis=2) <= 3)
        if len(xs) < 80:
            raise ValueError("not enough colored pixels to fit a line")
        points = np.column_stack((xs.astype(float) + left, ys.astype(float)))
        fits.append(_fit_line(points))
    intersections = np.vstack(
        (_intersection(fits[0], fits[1]), _intersection(fits[0], fits[2]), _intersection(fits[1], fits[2]))
    )
    center = intersections.mean(axis=0)
    return float(math.sqrt(np.mean(np.sum((intersections - center) ** 2, axis=1))))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    width = array.shape[1]
    edges = [round(index * width / 4) for index in range(5)]
    scores = [_score_panel(array, edges[index], edges[index + 1]) for index in range(4)]
    order = sorted(range(4), key=lambda index: scores[index])
    if scores[order[1]] - scores[order[0]] < 5.0:
        raise ValueError("ambiguous concurrency winner")
    return LABELS[order[0]]
