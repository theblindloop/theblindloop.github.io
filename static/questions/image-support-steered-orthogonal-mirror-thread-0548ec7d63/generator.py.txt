"""Deterministic scene sampler and renderer for orthogonal_mirror_thread."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
QUESTION = (
    "The blue thread is one closed curve. Considering the whole curve, is its "
    "mirror axis rising or falling from left to right? Answer rising or falling."
)


def _profile_gap(values: list[int]) -> float:
    pairs = [abs(values[i] - values[-1 - i]) for i in range(1, len(values) // 2)]
    return float(sum(pairs) / len(pairs))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    axis = "rising" if int(seed) % 2 == 0 else "falling"
    while True:
        profile = [0]
        profile.extend(rng.randint(28, 66) for _ in range(11))
        profile.append(0)
        profile[rng.randrange(2, 11)] = 72
        if _profile_gap(profile) >= 12.0:
            break
    palettes = [
        ([30, 103, 181], [247, 245, 237]),
        ([28, 121, 166], [250, 247, 239]),
        ([46, 93, 173], [246, 246, 240]),
    ]
    ink, background = rng.choice(palettes)
    return {
        "axis": axis,
        "center_x": rng.randint(184, 200),
        "center_y": rng.randint(184, 200),
        "radial_profile": profile,
        "rotation_half_turn": bool(rng.getrandbits(1)),
        "stroke_width": rng.randint(7, 10),
        "ink_rgb": ink,
        "background_rgb": background,
    }


def _chaikin(points: list[tuple[float, float]], rounds: int = 3) -> list[tuple[float, float]]:
    current = points
    for _ in range(rounds):
        refined: list[tuple[float, float]] = []
        for index, first in enumerate(current):
            second = current[(index + 1) % len(current)]
            refined.append((0.75 * first[0] + 0.25 * second[0], 0.75 * first[1] + 0.25 * second[1]))
            refined.append((0.25 * first[0] + 0.75 * second[0], 0.25 * first[1] + 0.75 * second[1]))
        current = refined
    return current


def _path_points(scene: dict) -> list[tuple[float, float]]:
    profile = [float(value) for value in scene["radial_profile"]]
    ys = [-166.0 + 332.0 * index / (len(profile) - 1) for index in range(len(profile))]
    right = list(zip(profile, ys))
    left = [(-profile[index], ys[index]) for index in range(len(profile) - 2, 0, -1)]
    points = _chaikin(right + left)
    output = []
    for x, y in points:
        if scene["rotation_half_turn"]:
            x, y = -x, -y
        angle = math.radians(45.0 if scene["axis"] == "rising" else -45.0)
        x, y = x * math.cos(angle) - y * math.sin(angle), x * math.sin(angle) + y * math.cos(angle)
        output.append((x + float(scene["center_x"]), y + float(scene["center_y"])))
    return output


def render(scene: dict) -> Image.Image:
    scale = 4
    background = tuple(int(value) for value in scene["background_rgb"])
    ink = tuple(int(value) for value in scene["ink_rgb"])
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)
    points = _path_points(scene)
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    scaled.append(scaled[0])
    draw.line(
        scaled,
        fill=ink,
        width=int(scene["stroke_width"]) * scale,
        joint="curve",
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["axis"])


def margin(scene: dict) -> float:
    return _profile_gap([int(value) for value in scene["radial_profile"]])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
