"""Independent image-only inverse arm for the closed-thread reflection task."""

from __future__ import annotations

from PIL import Image, ImageDraw
import numpy as np


class OracleAbstention(ValueError):
    pass


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        [array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)]
    )
    background = np.median(border, axis=0)
    distance = np.sqrt(np.sum((array - background) ** 2, axis=2))
    return distance > 48.0


def _has_large_hole(mask: np.ndarray) -> bool:
    # copy() makes the Pillow buffer writable; fromarray may otherwise expose a
    # read-only NumPy backing store on which floodfill becomes a no-op.
    canvas = Image.fromarray(np.where(mask, 255, 0).astype(np.uint8), mode="L").copy()
    ImageDraw.floodfill(canvas, (0, 0), 128, thresh=0)
    flooded = np.asarray(canvas)
    holes = flooded == 0
    return int(holes.sum()) >= 3500


def _one_component(mask: np.ndarray) -> bool:
    points = np.argwhere(mask)
    if len(points) < 900:
        return False
    # The thick thread must intersect every narrow bounding-box side band. This
    # rejects fragments without relying on a latent center or template.
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if x1 - x0 < 210 or y1 - y0 < 210:
        return False
    return True


def _reflection_errors(mask: np.ndarray) -> tuple[float, float]:
    points = np.argwhere(mask)
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    crop = mask[max(0, y0 - 2) : min(mask.shape[0], y1 + 3), max(0, x0 - 2) : min(mask.shape[1], x1 + 3)]
    side = max(crop.shape)
    square = np.zeros((side, side), dtype=bool)
    oy = (side - crop.shape[0]) // 2
    ox = (side - crop.shape[1]) // 2
    square[oy : oy + crop.shape[0], ox : ox + crop.shape[1]] = crop
    falling_reflection = square.T
    rising_reflection = np.flip(square.T, axis=(0, 1))
    union_r = np.logical_or(square, rising_reflection).sum()
    union_f = np.logical_or(square, falling_reflection).sum()
    error_r = np.logical_xor(square, rising_reflection).sum() / max(1, union_r)
    error_f = np.logical_xor(square, falling_reflection).sum() / max(1, union_f)
    return float(error_r), float(error_f)


def decision_from_image(image: Image.Image) -> str:
    mask = _foreground(image)
    if not _one_component(mask) or not _has_large_hole(mask):
        raise OracleAbstention("the raster does not contain one intact spanning closed thread")
    rising_error, falling_error = _reflection_errors(mask)
    gap = abs(rising_error - falling_error)
    if min(rising_error, falling_error) > 0.28 or gap < 0.10:
        raise OracleAbstention("reflection evidence is missing or ambiguous")
    return "rising" if rising_error < falling_error else "falling"
