"""Deterministic latent sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import itertools
import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 700
BEAD_NAMES = ("cyan", "orange", "violet", "green")
BEAD_RGB = {
    "cyan": (25, 156, 196),
    "orange": (238, 126, 34),
    "violet": (139, 83, 184),
    "green": (51, 160, 92),
}
BORDER_RGB = {
    "red": (210, 58, 58),
    "blue": (48, 102, 190),
    "gold": (205, 154, 28),
    "reference": (36, 132, 128),
}
LENGTHS = (42, 58, 74, 90)
PANEL_BOXES = ((325, 70, 575, 310), (35, 400, 285, 640), (325, 400, 575, 640), (615, 400, 865, 640))


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _different_permutations(reference: tuple[int, ...], rng: random.Random) -> list[list[int]]:
    pool = []
    for item in itertools.permutations(range(4)):
        if item == reference:
            continue
        matches = sum(a == b for a, b in zip(reference, item))
        if matches <= 2:
            pool.append(item)
    rng.shuffle(pool)
    return [list(pool[0]), list(pool[1])]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(range(4))
    rng.shuffle(reference)
    distractors = _different_permutations(tuple(reference), rng)

    winner_position = (int(seed) // 3) % 3
    options: list[list[int] | None] = [None, None, None]
    options[winner_position] = list(reference)
    di = 0
    for pos in range(3):
        if options[pos] is None:
            options[pos] = distractors[di]
            di += 1

    desired_decision = ("red", "blue", "gold")[int(seed) % 3]
    remaining = [x for x in ("red", "blue", "gold") if x != desired_decision]
    rng.shuffle(remaining)
    border_order: list[str | None] = [None, None, None]
    border_order[winner_position] = desired_decision
    ri = 0
    for pos in range(3):
        if border_order[pos] is None:
            border_order[pos] = remaining[ri]
            ri += 1

    angle_orders = []
    rotations = []
    for _ in range(4):
        order = list(range(4))
        rng.shuffle(order)
        angle_orders.append(order)
        rotations.append(rng.randrange(4))
    return {
        "rankings": [reference] + [list(x) for x in options],
        "angle_orders": angle_orders,
        "rotations": rotations,
        "border_order": [str(x) for x in border_order],
    }


def _matching_index(scene: dict) -> int:
    rankings = scene["rankings"]
    reference = tuple(rankings[0])
    matches = [i for i, value in enumerate(rankings[1:]) if tuple(value) == reference]
    if len(matches) != 1:
        raise ValueError("scene must have one exact rank match")
    return matches[0]


def analytic_gold(scene: dict) -> str:
    rankings = scene["rankings"]
    border_order = scene["border_order"]
    reference = tuple(rankings[0])
    matches = [i for i, value in enumerate(rankings[1:]) if tuple(value) == reference]
    if len(matches) != 1:
        raise ValueError("scene must have one exact rank match")
    return str(border_order[matches[0]])


def margin(scene: dict) -> float:
    rankings = scene["rankings"]
    reference = tuple(rankings[0])
    scores = sorted(
        (sum(a == b for a, b in zip(reference, option)) for option in rankings[1:]),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    rankings = scene["rankings"]
    reference = tuple(rankings[0])
    exact = sum(tuple(option) == reference for option in rankings[1:])
    return exact != 1 or margin(scene) < 2.0 or min(b - a for a, b in zip(LENGTHS, LENGTHS[1:])) < 16


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["rotations"] = [int(value) + 4 for value in scene["rotations"]]
    return [("rotation_plus_full_turn", twin)]


def _draw_centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _draw_panel(draw: ImageDraw.ImageDraw, box, border_name: str, ranking, angle_order, rotation: int, label: str):
    x0, y0, x1, y1 = box
    border = BORDER_RGB[border_name]
    draw.rounded_rectangle(box, radius=15, fill=(253, 253, 250), outline=border, width=7)
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2 + 8
    rank_by_color = {color_index: rank for rank, color_index in enumerate(ranking)}
    angle_by_color = {color_index: slot for slot, color_index in enumerate(angle_order)}
    endpoints = {}
    for color_index in range(4):
        rank = rank_by_color[color_index]
        slot = angle_by_color[color_index]
        degrees = 18 + 90 * ((slot + int(rotation)) % 4)
        radians = math.radians(degrees)
        length = LENGTHS[rank]
        ex = int(round(cx + length * math.cos(radians)))
        ey = int(round(cy + length * math.sin(radians)))
        endpoints[color_index] = (ex, ey)
        draw.line((cx, cy, ex, ey), fill=(155, 164, 171), width=6)
    draw.ellipse((cx - 13, cy - 13, cx + 13, cy + 13), fill=(38, 50, 56), outline=(18, 25, 28), width=2)
    for color_index, (ex, ey) in endpoints.items():
        name = BEAD_NAMES[color_index]
        draw.ellipse((ex - 12, ey - 12, ex + 12, ey + 12), fill=BEAD_RGB[name], outline=(255, 255, 255), width=2)
    _draw_centered(draw, ((x0 + x1) // 2, y0 + 23), label, _font(18, True), (35, 39, 42))


def render(scene: dict) -> Image.Image:
    rankings = scene["rankings"]
    angle_orders = scene["angle_orders"]
    rotations = scene["rotations"]
    border_order = scene["border_order"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (244, 242, 235))
    draw = ImageDraw.Draw(image)
    _draw_centered(draw, (WIDTH // 2, 25), "SPOKE-LENGTH RANK MATCH", _font(24, True), (28, 33, 36))
    _draw_centered(draw, (WIDTH // 2, 51), "Match the color order from shortest spoke to longest", _font(16), (65, 70, 74))
    _draw_panel(draw, PANEL_BOXES[0], "reference", rankings[0], angle_orders[0], rotations[0] % 4, "REFERENCE")
    for i in range(3):
        _draw_panel(draw, PANEL_BOXES[i + 1], border_order[i], rankings[i + 1], angle_orders[i + 1], rotations[i + 1] % 4, f"OPTION {i + 1}")
    _draw_centered(draw, (WIDTH // 2, 675), "Answer with the matching option's border color: red, blue, or gold", _font(16, True), (42, 46, 48))
    return image
