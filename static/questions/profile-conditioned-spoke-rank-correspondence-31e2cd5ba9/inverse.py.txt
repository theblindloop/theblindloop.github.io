"""Independent inverse arm: infer the answer from raster pixels alone."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BEADS = (
    ("cyan", (25, 156, 196)),
    ("orange", (238, 126, 34)),
    ("violet", (139, 83, 184)),
    ("green", (51, 160, 92)),
)
BORDERS = (
    ("reference", (36, 132, 128)),
    ("red", (210, 58, 58)),
    ("blue", (48, 102, 190)),
    ("gold", (205, 154, 28)),
)
HUB = (38, 50, 56)


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 18.0) -> np.ndarray:
    delta = arr.astype(np.int32) - np.array(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 20):
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            py, px = queue.popleft()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            data = np.asarray(points)
            output.append((len(points), int(data[:, 1].min()), int(data[:, 0].min()), int(data[:, 1].max()), int(data[:, 0].max())))
    return output


def _panel_boxes(arr: np.ndarray) -> dict[str, tuple[int, int, int, int]]:
    found = {}
    for name, rgb in BORDERS:
        large = [c for c in _components(_mask(arr, rgb), minimum=500) if c[3] - c[1] > 180 and c[4] - c[2] > 180]
        if len(large) != 1:
            raise ValueError(f"could not isolate {name} panel border")
        _, x0, y0, x1, y1 = large[0]
        found[name] = (x0, y0, x1, y1)
    return found


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 30:
        raise ValueError("missing visual marker")
    return float(xs.mean()), float(ys.mean())


def _hub_centroid(mask: np.ndarray) -> tuple[float, float]:
    compact = [item for item in _components(mask, minimum=120) if item[3] - item[1] < 45 and item[4] - item[2] < 45]
    if len(compact) != 1:
        raise ValueError("could not isolate central hub")
    _, x0, y0, x1, y1 = compact[0]
    isolated = np.zeros_like(mask)
    isolated[y0:y1 + 1, x0:x1 + 1] = mask[y0:y1 + 1, x0:x1 + 1]
    return _centroid(isolated)


def _ranking(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[str, ...]:
    x0, y0, x1, y1 = box
    pad = 12
    interior = np.zeros(arr.shape[:2], dtype=bool)
    interior[y0 + pad:y1 - pad + 1, x0 + pad:x1 - pad + 1] = True
    hub = _hub_centroid(_mask(arr, HUB) & interior)
    distances = []
    for name, rgb in BEADS:
        bead = _centroid(_mask(arr, rgb) & interior)
        distance = ((bead[0] - hub[0]) ** 2 + (bead[1] - hub[1]) ** 2) ** 0.5
        distances.append((distance, name))
    distances.sort()
    gaps = [distances[i + 1][0] - distances[i][0] for i in range(3)]
    if min(gaps) < 10.0:
        raise ValueError("ambiguous spoke-length rank")
    return tuple(name for _, name in distances)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    boxes = _panel_boxes(arr)
    reference = _ranking(arr, boxes["reference"])
    matches = [name for name in ("red", "blue", "gold") if _ranking(arr, boxes[name]) == reference]
    if len(matches) != 1:
        raise ValueError("raster does not contain one unique matching option")
    return matches[0]
