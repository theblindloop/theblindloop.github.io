"""Independent image-only inverse arm for Catchment Area Fork.

Only fixed public palette/structure rules are used.  No renderer, scene,
manifest, filename, or analytic state is imported or accepted.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = np.array((247, 248, 244), dtype=np.uint8)
NAVY = np.array((25, 55, 79), dtype=np.uint8)
BLUE = np.array((35, 115, 190), dtype=np.uint8)
PURPLE = np.array((125, 76, 170), dtype=np.uint8)
RED = np.array((205, 55, 65), dtype=np.uint8)
GOLD = np.array((225, 165, 35), dtype=np.uint8)


class OracleAbstention(ValueError):
    """The raster is not a complete, unambiguous member of the public world."""


def _mask(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(array == color, axis=2)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = deque([(sy, sx)])
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int16))
    return sorted(output, key=len, reverse=True)


def _single_marker(array: np.ndarray, color: np.ndarray, name: str) -> tuple[float, float]:
    components = _components(_mask(array, color))
    if len(components) != 1 or not 270 <= len(components[0]) <= 360:
        raise OracleAbstention(f"{name} marker is missing, clipped, or fragmented")
    yx = components[0].mean(axis=0)
    return float(yx[1]), float(yx[0])


def _nearest_component(components: list[np.ndarray], center: tuple[float, float]) -> int:
    cx, cy = center
    best = None
    for index, points in enumerate(components):
        distance2 = np.min((points[:, 1] - cx) ** 2 + (points[:, 0] - cy) ** 2)
        item = (float(distance2), index)
        if best is None or item < best:
            best = item
    if best is None or best[0] > 28.0**2:
        raise OracleAbstention("a terminal is not attached to a downstream navy branch")
    return int(best[1])


def decision_from_image(image: Image.Image) -> str:
    """Cut at the visible fork, count the two terminal-linked regions, select larger."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise OracleAbstention("unexpected canvas size")
    if not np.all(array[0, 0] == BACKGROUND):
        raise OracleAbstention("unexpected background")

    blue = _single_marker(array, BLUE, "blue")
    purple = _single_marker(array, PURPLE, "purple")
    red = _single_marker(array, RED, "red")
    gold = _single_marker(array, GOLD, "gold")
    navy = _mask(array, NAVY)
    navy_count = int(navy.sum())
    # The two basin widths exchange equal area.  This tight whole-region check
    # makes any substantial evidence erasure an abstention rather than a guess.
    if not 8670 <= navy_count <= 8785:
        raise OracleAbstention(f"navy catchment has invalid area {navy_count}")

    px, py = purple
    yy, xx = np.ogrid[:512, :512]
    cut = navy.copy()
    cut[(xx - px) ** 2 + (yy - py) ** 2 <= 24.0**2] = False
    components = [component for component in _components(cut) if len(component) >= 120]
    if len(components) != 3:
        raise OracleAbstention(f"fork cut produced {len(components)} substantial regions")

    red_index = _nearest_component(components, red)
    gold_index = _nearest_component(components, gold)
    blue_index = _nearest_component(components, blue)
    if len({red_index, gold_index, blue_index}) != 3:
        raise OracleAbstention("start and terminals do not occupy three distinct fork arms")
    red_area = len(components[red_index])
    gold_area = len(components[gold_index])
    if min(red_area, gold_area) < 1900:
        raise OracleAbstention("a downstream branch is incomplete")
    if abs(red_area - gold_area) < 250:
        raise OracleAbstention("downstream area comparison is within the raster boundary band")
    return "red" if red_area > gold_area else "gold"
