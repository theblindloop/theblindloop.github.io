"""Deterministic renderer and analytic arm for Catchment Area Fork.

The analytic arm uses only which latent side has the wider filled basin and
which terminal color is attached to that side.  The independent oracle must
recover the same relation by cutting the raster at the visible purple fork and
counting downstream navy pixels.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 248, 244)
NAVY = (25, 55, 79)
BLUE = (35, 115, 190)
PURPLE = (125, 76, 170)
RED = (205, 55, 65)
GOLD = (225, 165, 35)

BASE_FORK = (256, 294)
BASE_START = (256, 360)
BASE_TERMINALS = {"left": (176, 180), "right": (336, 180)}
BASE_PATHS = {
    "left": [(256, 294), (228, 276), (205, 244), (184, 212), (176, 180)],
    "right": [(256, 294), (284, 276), (307, 244), (328, 212), (336, 180)],
}
BASE_BASINS = {"left": (205, 244), "right": (307, 244)}


def sample_scene(seed: int) -> dict:
    """Return one balanced, deterministic JSON-compatible scene."""
    rng = random.Random(int(seed) ^ 0xCA7C_4E17)
    return {
        "wide_side": "left" if (int(seed) // 2) % 2 else "right",
        "red_side": "left" if int(seed) % 2 else "right",
        "area_gap": (5, 8, 11)[(int(seed) // 3) % 3],
        "rotation_quadrants": (int(seed) // 4) % 4,
        "mirror": bool((int(seed) // 16) % 2),
        "dx": (-8, 0, 8)[(int(seed) // 7) % 3],
        "dy": (-8, 0, 8)[(int(seed) // 11) % 3],
        "stroke": 22,
        "basin_height": 50,
        "draw_order": ["left", "right"] if rng.random() < 0.5 else ["right", "left"],
    }


def _point(scene: dict, point: tuple[int, int]) -> tuple[int, int]:
    x, y = point
    if scene["mirror"]:
        x = 2 * 256 - x
    for _ in range(int(scene["rotation_quadrants"]) % 4):
        x, y = 512 - y, x
    return (int(x + scene["dx"]), int(y + scene["dy"]))


def _points(scene: dict, points: Iterable[tuple[int, int]]) -> list[tuple[int, int]]:
    return [_point(scene, point) for point in points]


def _basin_polygon(scene: dict, side: str) -> list[tuple[int, int]]:
    gap = int(scene["area_gap"])
    width = 48 + gap if side == scene["wide_side"] else 48 - gap
    height = int(scene["basin_height"])
    cx, cy = BASE_BASINS[side]
    x0, x1 = cx - width // 2, cx + (width - width // 2)
    y0, y1 = cy - height // 2, cy + (height - height // 2)
    return _points(scene, [(x0, y0), (x1, y0), (x1, y1), (x0, y1)])


def render(scene: dict) -> Image.Image:
    """Render an exact-palette RGB image; no latent answer is embedded elsewhere."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    stroke = int(scene["stroke"])

    draw.line(
        _points(scene, [BASE_START, (256, 330), BASE_FORK]),
        fill=NAVY,
        width=stroke,
        joint="curve",
    )
    for side in scene["draw_order"]:
        draw.line(_points(scene, BASE_PATHS[side]), fill=NAVY, width=stroke, joint="curve")
        draw.polygon(_basin_polygon(scene, side), fill=NAVY)

    terminals = {side: _point(scene, point) for side, point in BASE_TERMINALS.items()}
    start = _point(scene, BASE_START)
    fork = _point(scene, BASE_FORK)

    def ring(center: tuple[int, int], color: tuple[int, int, int]) -> None:
        x, y = center
        draw.ellipse((x - 14, y - 14, x + 14, y + 14), fill=BACKGROUND)
        draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)

    ring(start, BLUE)
    for side, center in terminals.items():
        ring(center, RED if side == scene["red_side"] else GOLD)
    x, y = fork
    draw.ellipse((x - 10, y - 10, x + 10, y + 10), fill=PURPLE)
    return image


def analytic_gold(scene: dict) -> str:
    """Return the color attached to the larger downstream catchment."""
    return "red" if scene["wide_side"] == scene["red_side"] else "gold"


def margin(scene: dict) -> float:
    """Nominal rectangular area gap; equal path ink cancels by mirror symmetry."""
    return float(2 * int(scene["area_gap"]) * int(scene["basin_height"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 360.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Same-color branch draw order is a genuine render-invariant latent alias."""
    twin = dict(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_same_color_branch_draw_order", twin)]
