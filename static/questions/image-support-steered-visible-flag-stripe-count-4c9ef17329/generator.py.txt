"""Analytic sampler and renderer for a prior-conflict flag stripe count."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (255, 255, 255)
COUNTS = (9, 11, 13, 15, 17)
PALETTES = (
    ((178, 42, 54), (244, 221, 172), (38, 70, 122)),
    ((30, 116, 105), (242, 207, 128), (83, 55, 126)),
    ((193, 79, 45), (225, 224, 188), (31, 88, 128)),
)
MIN_BAND_HEIGHT = 6.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    stripe_count = COUNTS[int(seed) % len(COUNTS)]
    width = rng.randrange(238, 263)
    height = rng.randrange(164, 181)
    x = rng.randrange(176, WIDTH - width - 176 + 1)
    y = rng.randrange(138, HEIGHT - height - 138 + 1)
    palette = PALETTES[rng.randrange(len(PALETTES))]
    if rng.randrange(2):
        dark_color, light_color = palette[1], palette[0]
    else:
        dark_color, light_color = palette[0], palette[1]
    return {
        "stripe_count": stripe_count,
        "x": x,
        "y": y,
        "width": width,
        "height": height,
        "dark_color": list(dark_color),
        "light_color": list(light_color),
        "canton_color": list(palette[2]),
        "canton_side": "left" if rng.randrange(2) == 0 else "right",
        "star_phase": rng.randrange(8),
    }


def _band_bounds(scene: dict, index: int) -> tuple[int, int]:
    top = int(scene["y"]) + round(index * int(scene["height"]) / int(scene["stripe_count"]))
    bottom = int(scene["y"]) + round((index + 1) * int(scene["height"]) / int(scene["stripe_count"]))
    return top, bottom


def render(scene: dict) -> Image.Image:
    # Every declared latent field is deliberately accessed here.
    stripe_count = int(scene["stripe_count"])
    x, y = int(scene["x"]), int(scene["y"])
    width, height = int(scene["width"]), int(scene["height"])
    dark = tuple(int(v) for v in scene["dark_color"])
    light = tuple(int(v) for v in scene["light_color"])
    canton = tuple(int(v) for v in scene["canton_color"])
    side = str(scene["canton_side"])
    phase = int(scene["star_phase"]) % 2

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for index in range(stripe_count):
        top, bottom = _band_bounds(scene, index)
        draw.rectangle((x, top, x + width - 1, bottom - 1), fill=dark if index % 2 == 0 else light)

    canton_width = round(width * 0.42)
    canton_rows = (stripe_count + 1) // 2
    _, canton_bottom = _band_bounds(scene, canton_rows - 1)
    if side == "left":
        cx0, cx1 = x, x + canton_width - 1
    else:
        cx0, cx1 = x + width - canton_width, x + width - 1
    draw.rectangle((cx0, y, cx1, canton_bottom - 1), fill=canton)

    # Small neutral stars create flag identity but never encode the answer.
    star = (248, 245, 218)
    columns, rows = 5, 4
    for row in range(rows):
        for column in range(columns):
            offset = (row + phase) % 2
            sx = cx0 + round((column + 1) * (canton_width - 8) / (columns + 1)) + offset
            sy = y + round((row + 1) * (canton_bottom - y - 8) / (rows + 1))
            draw.ellipse((sx - 2, sy - 2, sx + 2, sy + 2), fill=star)
    return image


def analytic_gold(scene: dict) -> str:
    return str(int(scene["stripe_count"]))


def margin(scene: dict) -> float:
    return float(int(scene["height"]) / int(scene["stripe_count"]) - MIN_BAND_HEIGHT)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["star_phase"] = int(scene["star_phase"]) + 2
    return [("star_phase_plus_two", twin)]
