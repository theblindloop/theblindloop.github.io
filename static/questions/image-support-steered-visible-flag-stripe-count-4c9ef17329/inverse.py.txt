"""Independent image-only inverse for the visible flag stripe count.

The inverse knows only the public visual contract: a single filled rectangular
flag on a white field, horizontal alternating bands, and a top canton that may
shorten bands on either side. It neither imports nor receives renderer state.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _runs(colors: np.ndarray) -> int:
    count = 1
    previous = colors[0].astype(np.float32)
    for color in colors[1:]:
        current = color.astype(np.float32)
        if float(np.linalg.norm(current - previous)) > 45.0:
            count += 1
            previous = current
        else:
            previous = 0.75 * previous + 0.25 * current
    return count


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 200 or array.shape[1] < 200:
        raise ValueError("canvas is too small")
    delta = np.linalg.norm(array.astype(np.float32) - 255.0, axis=2)
    foreground = delta > 28.0
    ys, xs = np.nonzero(foreground)
    if len(xs) < 10000:
        raise ValueError("flag region is missing")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (220 <= width <= 285 and 150 <= height <= 195):
        raise ValueError("foreground is not the expected regional flag rectangle")
    region = foreground[y0 : y1 + 1, x0 : x1 + 1]
    row_fill = region.mean(axis=1)
    column_fill = region.mean(axis=0)
    if float(row_fill.min()) < 0.82 or float(column_fill.min()) < 0.72:
        raise ValueError("flag interior is incomplete")

    strip_width = max(10, round(width * 0.13))
    left = array[y0 : y1 + 1, x0 : x0 + strip_width]
    right = array[y0 : y1 + 1, x1 - strip_width + 1 : x1 + 1]
    left_colors = np.median(left, axis=1).astype(np.uint8)
    right_colors = np.median(right, axis=1).astype(np.uint8)
    left_runs, right_runs = _runs(left_colors), _runs(right_colors)
    count = max(left_runs, right_runs)
    if count not in {9, 11, 13, 15, 17}:
        raise ValueError("stripe count is outside the public answer set")
    if abs(left_runs - right_runs) > count - 3:
        raise ValueError("two flag sides give an ambiguous count")
    return str(count)
