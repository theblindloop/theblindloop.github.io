"""Route-membership world: which colored dot lies on the drawn route.

A single winding route is drawn in a route colour (orange) that is distinct
from the two candidate dot colours (red, blue). Exactly one colored dot sits
ON the route stroke (its ink overlaps the stroke), and the other colored dot
is placed well off the route with enforced clearance from all route ink. The
question identifies no route by colour; instead the model must decide which of
the two colored dots is a member of (drawn on top of / touching) the single
route stroke. Because route colour is disjoint from the candidate dot colours,
the answer is not recoverable by colour-name matching, straight-line proximity
alone (the off dot is deliberately farther), or counting; it requires an
on-stroke membership / ink-adjacency decision.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "dot_on_colored_route-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_ROUTE = (235, 140, 30)
COLOR_RED = (225, 40, 40)
COLOR_BLUE = (45, 80, 220)

STROKE = 16
DOT_R = 10
MARGIN_PAD = 46
CLEAR_MARGIN = 34.0  # off-dot center must keep >= this distance from route ink

DOT_COLORS = ["red", "blue"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _seg_dist(p, a, b):
    ax, ay = a
    bx, by = b
    px, py = p
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return math.dist(p, a)
    t = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _dist_to_path(p, pts):
    best = float("inf")
    for a, b in zip(pts, pts[1:]):
        best = min(best, _seg_dist(p, a, b))
    return best


def _serpentine(rng: random.Random) -> list:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    n_lanes = rng.randint(3, 5)
    step_h = (CANVAS[1] - 2 * MARGIN_PAD) / (n_lanes - 1) if n_lanes > 1 else 0.0
    x_start = rng.uniform(lo + 10, lo + 40)
    x_end = rng.uniform(hi - 55, hi - 10)
    path = []
    for lane in range(n_lanes):
        y = MARGIN_PAD + lane * step_h
        if lane % 2 == 0:
            path.append((x_start, y))
            path.append((x_end, y))
        else:
            path.append((x_end, y))
            path.append((x_start, y))
    return _pts(path)


def _sample_scene(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(30000):
        pts = _serpentine(rng)
        total = _path_length(pts)
        # on-dot placed at an interior arc-length of the route (away from ends)
        frac = rng.uniform(0.30, 0.70)
        on_pt = _point_at(pts, frac * total)
        # assign on-dot and off-dot colours: red and blue, random which is on
        order = list(DOT_COLORS)
        rng.shuffle(order)
        on_color, off_color = order[0], order[1]

        # choose off-dot location far from all route ink and from the on-dot
        off_pt = _find_off_point(rng, pts, on_pt)
        if off_pt is None:
            continue

        on_center = (round(on_pt[0], 3), round(on_pt[1], 3))
        off_center = off_pt
        # validity
        if not (DOT_R + 8 <= on_center[0] <= w - DOT_R - 8
                and DOT_R + 8 <= on_center[1] <= h - DOT_R - 8):
            continue
        if not (DOT_R + 8 <= off_center[0] <= w - DOT_R - 8
                and DOT_R + 8 <= off_center[1] <= h - DOT_R - 8):
            continue
        # the on-dot must genuinely overlap the route => its center is on the stroke
        if _dist_to_path(on_center, pts) > STROKE / 2.0:
            continue

        dots = [
            {"color": on_color, "center": on_center, "on_route": True},
            {"color": off_color, "center": [off_center[0], off_center[1]], "on_route": False},
        ]
        scene = {
            "route": pts,
            "dots": dots,
        }
        return scene
    raise RuntimeError("could not satisfy on-route constraints")


def _find_off_point(rng, pts, on_pt):
    w, h = CANVAS
    for _ in range(2000):
        x = rng.uniform(MARGIN_PAD + 16, w - MARGIN_PAD - 16)
        y = rng.uniform(MARGIN_PAD + 16, h - MARGIN_PAD - 16)
        if _dist_to_path((x, y), pts) < CLEAR_MARGIN:
            continue
        if math.dist((x, y), on_pt) < CLEAR_MARGIN:
            continue
        return (round(x, 3), round(y, 3))
    return None


def sample_scene(seed: int) -> dict:
    return _sample_scene(random.Random(seed))


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def _dot_rgb(color: str):
    return {"red": COLOR_RED, "blue": COLOR_BLUE}[color]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    _draw_polyline(draw, _pts(scene["route"]), COLOR_ROUTE, STROKE)
    for dot in scene["dots"]:
        cx, cy = float(dot["center"][0]), float(dot["center"][1])
        r = DOT_R
        draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=_dot_rgb(dot["color"]))
    return img


def _on_dot(scene: dict):
    return next(d for d in scene["dots"] if d["on_route"])


def analytic_gold(scene: dict) -> str:
    """Colour of the dot that lies ON the route."""
    return _on_dot(scene)["color"]


def margin(scene: dict) -> float:
    """Clearance (px) of the OFF dot from all route ink — the one-sided decision margin."""
    off = next(d for d in scene["dots"] if not d["on_route"])
    pts = _pts(scene["route"])
    return float(_dist_to_path((off["center"][0], off["center"][1]), pts))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < CLEAR_MARGIN:
        return True
    for dot in scene["dots"]:
        cx, cy = dot["center"]
        if not (DOT_R + 8 <= cx <= w - DOT_R - 8 and DOT_R + 8 <= cy <= h - DOT_R - 8):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
