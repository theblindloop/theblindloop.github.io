"""Pixel-only inverse arm for the dot_on_colored_route world.

Reads only the final RGB PNG. Segments the pixels into the known palette
(orange route, red and blue dots on white). The route is identified by its
orange colour, disjoint from the candidate dot colours. The answer is the
colour of the dot whose ink is on/touching the orange route stroke (an
8-adjacency / membership test), not the dot that is merely closer in a
straight line.

Imports only PIL/numpy; it never reads renderer/prompt/generator/gold code or
latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "orange": (235, 140, 30),
    "red": (225, 40, 40),
    "blue": (45, 80, 220),
    "white": (255, 255, 255),
}
_NAMES = ["orange", "red", "blue", "white"]

# Dilation radius used for the "on/touching the route" membership test.
_ADJ = 6


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _mask_within(mask: np.ndarray, radius: int) -> np.ndarray:
    """Return the mask of pixels within Chebyshev radius of any True pixel."""
    if not mask.any():
        return mask
    out = mask.copy()
    for dy in range(-radius, radius + 1):
        for dx in range(-radius, radius + 1):
            rolled = np.roll(mask, (dy, dx), axis=(0, 1))
            out |= rolled
    return out


def _touches(route_mask: np.ndarray, dot_mask: np.ndarray) -> bool:
    if not dot_mask.any():
        return False
    dilated = _mask_within(route_mask, _ADJ)
    return bool((dilated & dot_mask).any())


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    route = labels == idx["orange"]
    if not route.any():
        raise ValueError("route not found")
    found = [c for c in ("red", "blue") if _touches(route, labels == idx[c])]
    if len(found) != 1:
        raise ValueError(f"expected exactly one dot on the route, got {found}")
    return found[0]
