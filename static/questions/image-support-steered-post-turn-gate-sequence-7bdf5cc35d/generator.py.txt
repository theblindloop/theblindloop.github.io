"""Deterministic renderer and analytic arm for post-turn gate order."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 600
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE_RGB = (27, 53, 76)
GREEN_RGB = (43, 174, 103)
RED_RGB = (218, 70, 66)
AMBER_RGB = (235, 151, 35)
VIOLET_RGB = (143, 84, 194)
QUARANTINE_MARGIN = 48.0


def _distance(first: Iterable[float], second: Iterable[float]) -> float:
    ax, ay = (float(value) for value in first)
    bx, by = (float(value) for value in second)
    return math.hypot(ax - bx, ay - by)


def _points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in scene["route_points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(cumulative[-1] + _distance(first, second))
    return cumulative, cumulative[-1]


def _point_at_fraction(
    points: list[tuple[float, float]], fraction: float
) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span == 0.0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0)
    return points[-1]


def _oriented(point: tuple[float, float], orientation: int) -> tuple[float, float]:
    x, y = point
    if orientation == 0:
        return x, y
    if orientation == 1:
        return -y, x
    if orientation == 2:
        return -x, -y
    return y, -x


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one compact, non-self-intersecting two-segment route.

    The base route turns once, exactly at 90 degrees. Both gates are on the
    second leg. Every other visible nuisance is independent of the gate-label
    assignment, and every sixth deterministic scene is close to the visible
    equal-order boundary but still separated enough for pixel recovery.
    """

    rng = random.Random(int(seed))
    orientation = rng.randrange(4)
    base = [(0.0, 0.0), (40.0, 0.0), (40.0, 120.0)]
    rotated = [_oriented(point, orientation) for point in base]
    min_x = min(point[0] for point in rotated)
    min_y = min(point[1] for point in rotated)
    max_x = max(point[0] for point in rotated)
    max_y = max(point[1] for point in rotated)
    span_x = max_x - min_x
    span_y = max_y - min_y
    tx = rng.uniform(68.0 - min_x, WIDTH - 68.0 - max_x)
    ty = rng.uniform(58.0 - min_y, HEIGHT - 58.0 - max_y)
    # Keep the short transverse leg inside one ablation cell while choosing
    # that cell randomly. This preserves absolute-position nuisance variance
    # without allowing a harmless grid boundary to turn one thin trace into a
    # falsely regional support footprint.
    if span_x <= span_y:
        cell_width = WIDTH / 8.0
        cell = rng.randrange(8)
        tx = rng.uniform(
            cell * cell_width + 18.0 - min_x,
            (cell + 1) * cell_width - 18.0 - max_x,
        )
    else:
        cell_height = HEIGHT / 8.0
        cell = rng.randrange(8)
        ty = rng.uniform(
            cell * cell_height + 18.0 - min_y,
            (cell + 1) * cell_height - 18.0 - max_y,
        )
    route_points = [
        [round(x + tx, 3), round(y + ty, 3)] for x, y in rotated
    ]

    near_boundary = int(seed) % 6 == 0
    if near_boundary:
        t1 = rng.uniform(0.35, 0.39)
        t2 = rng.uniform(0.67, 0.71)
    else:
        t1 = rng.uniform(0.24, 0.30)
        t2 = rng.uniform(0.72, 0.80)
    total = 40.0 + 120.0
    fraction1 = (40.0 + 120.0 * t1) / total
    fraction2 = (40.0 + 120.0 * t2) / total
    first_label = "amber" if int(seed) % 2 == 0 else "violet"
    second_label = "violet" if first_label == "amber" else "amber"
    gates = [
        {"label": first_label, "fraction": round(fraction1, 6)},
        {"label": second_label, "fraction": round(fraction2, 6)},
    ]
    return {
        "route_points": route_points,
        "gates": gates,
        "route_width": round(rng.uniform(8.0, 9.0), 3),
        "gate_radius": round(rng.uniform(9.5, 10.5), 3),
        "orientation": orientation,
        "translation": [round(tx, 3), round(ty, 3)],
        "scene_variant": "near_boundary" if near_boundary else "clear",
    }


def _scaled_points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _draw_round_route(
    draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], width: int
) -> None:
    scaled = _scaled_points(points)
    radius = max(1, width // 2)
    draw.line(scaled, fill=ROUTE_RGB, width=width, joint="curve")
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ROUTE_RGB)


def _draw_port(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    r = round((radius + 1.5) * SCALE)
    draw.ellipse((x - r, y - r, x + r, y + r), fill=color)


def _draw_gate(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    outer = round((radius + 3.0) * SCALE)
    width = max(2, round(4.0 * SCALE))
    draw.ellipse(
        (x - outer, y - outer, x + outer, y + outer),
        outline=color,
        width=width,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize the route and visible ports/gates; no latent data is embedded."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    width = max(1, round(float(scene["route_width"]) * SCALE))
    gate_radius = float(scene["gate_radius"])
    _draw_round_route(draw, points, width)
    colors = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
    for gate in scene["gates"]:
        center = _point_at_fraction(points, float(gate["fraction"]))
        _draw_gate(draw, center, colors[str(gate["label"])], gate_radius)
    _draw_port(draw, points[0], GREEN_RGB, gate_radius)
    _draw_port(draw, points[-1], RED_RGB, gate_radius)
    # A final centerline pass keeps the route connected through every ring and
    # leaves the marker identity visible as an annulus around the thin trace.
    _draw_round_route(draw, points, width)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    gates = list(scene["gates"])
    if len(gates) != 2:
        raise ValueError("scene needs exactly two gates")
    labels = {str(gate["label"]) for gate in gates}
    if labels != {"amber", "violet"}:
        raise ValueError("gates must be amber and violet")
    ordered = sorted(gates, key=lambda gate: float(gate["fraction"]))
    if not float(ordered[0]["fraction"]) < float(ordered[1]["fraction"]):
        raise ValueError("gates must have distinct route order")
    return str(ordered[0]["label"])


def margin(scene: dict[str, Any]) -> float:
    points = _points(scene)
    _, total = _route_lengths(points)
    fractions = sorted(float(gate["fraction"]) for gate in scene["gates"])
    return float((fractions[1] - fractions[0]) * total)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # There are no hidden label-preserving aliases: all gate labels and
    # positions are rendered as visible colored rings.
    return []
