"""Independent pixel-only inverse arm for the disk-spacing correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray reference object: a pair of two gray
    disks separated horizontally by a center-to-center distance from the
    canonical set {70,110,150,190,230};
  * a horizontal dark divider separates it from a column of three colored
    candidate objects, one red, one green, one blue, each a pair of two
    same-colored disks separated by a distinct canonical center-to-center
    distance;
  * the task: decide which object color has exactly the same pair spacing as the
    gray reference.

The oracle, from pixels alone: finds the divider, segments disks by palette
color, takes the connected components of each color, groups the exactly two
disks that form each object, measures each object's center-to-center spacing as
the distance between its two disk centroids, and returns the color of the
candidate object whose spacing equals the reference's.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "disk_spacing_correspondence-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
_CANONICAL = [70, 110, 150, 190, 230]
_COLOR_DIST = 60.0
_N_DARK = 90.0
_MIN_AREA = 60
_MAX_DIAM = 80
_TOL = 12.0


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _components(mask: np.ndarray) -> list[dict]:
    """Return centroid/area of each 4-connected component of a binary mask.

    Iterates only the sparse ink pixels, so it stays fast for disk-size objects.
    """
    h, w = mask.shape
    coords = np.argwhere(mask)
    if coords.size == 0:
        return []
    comps = []
    visited = set()
    for r, c in coords:
        key = (int(r), int(c))
        if key in visited:
            continue
        stack = [key]
        visited.add(key)
        cells = []
        while stack:
            cr, cc = stack.pop()
            cells.append((cr, cc))
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = cr + dr, cc + dc
                if 0 <= nr < h and 0 <= nc < w and mask[nr, nc] and (nr, nc) not in visited:
                    visited.add((nr, nc))
                    stack.append((nr, nc))
        cells.sort()
        rows = [cell[0] for cell in cells]
        cols = [cell[1] for cell in cells]
        area = len(cells)
        height = max(rows) - min(rows) + 1
        width = max(cols) - min(cols) + 1
        if _MIN_AREA <= area and height <= _MAX_DIAM and width <= _MAX_DIAM:
            comps.append(
                {
                    "x": float(np.mean(cols)),
                    "y": float(np.mean(rows)),
                    "area": area,
                }
            )
    return comps


def _pair_spacing(mask: np.ndarray, divider_mid: float, above: bool) -> float:
    """Find the two same-color disks forming one object and measure their
    inter-disk center distance. ``above`` selects the reference band (above the
    divider) vs the candidate bands (below)."""
    comps = _components(mask)
    comps = [c for c in comps if (c["y"] < divider_mid if above else c["y"] >= divider_mid)]
    if len(comps) != 2:
        raise ValueError(f"expected exactly 2 disks, found {len(comps)}")
    d = math.hypot(comps[0]["x"] - comps[1]["x"], comps[0]["y"] - comps[1]["y"])
    return min(_CANONICAL, key=lambda value: abs(value - d))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    dark = arr.max(axis=2) < _N_DARK
    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    gray_mask = _classify(arr, GRAY)
    ref_spacing = _pair_spacing(gray_mask, divider_mid, above=True)

    color_masks = {name: _classify(arr, rgb) for name, rgb in PALETTE.items()}
    for name, mask in color_masks.items():
        spacing = _pair_spacing(mask, divider_mid, above=False)
        if abs(spacing - ref_spacing) <= _TOL:
            return name
    raise ValueError(f"no candidate pair matched reference spacing {ref_spacing}")


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
