"""Disk-spacing correspondence renderer (paper-semantic correspondence family).

A cross-panel correspondence task whose binding key is the *center-to-center
distance* (spread) between the two same-colored disks of a two-disk object.

A neutral gray reference object at the top is a single pair of two gray disks
separated horizontally by a specific center distance ``ref_dist`` drawn from a
finite canonical set. Below a divider sit three colored candidate objects A, B,
C (red / green / blue), each a pair of two same-colored disks separated by a
candidate center distance, pairwise distinct, and exactly one of them sharing
the reference's spacing. The question asks for the color of the candidate whose
pair spacing equals the reference's.

The pixel-only inverse arm must, for every panel, segment the disks, recover
each object's pair (two same-colored disks) and its center-to-center spacing,
establish the cross-panel correspondence by spacing equality, and only then read
out the bound color of the single matched object.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_dist   : center-to-center distance (px) of the gray reference pair,
                 drawn from the canonical set CANONICAL_DISTS.
    cand_dists : length-3 list of candidate pair spacings; each is a distinct
                 canonical spacing and exactly one equals ``ref_dist``.
    cand_colors: the fixed 3-color palette bound one-to-one to the candidates.
                 Each candidate pair is drawn in its bound color.
    radius     : fixed disk radius (px) shared by every disk.
    ref_y, cand_ys, divider_y, width, height: fixed geometry and layout.

The analytic gold is the color of the candidate whose pair spacing equals the
reference's. Because the reference spacing, each candidate spacing, and the
bound colors are all rendered verbatim, the gold is uniquely determined by the
raster.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "disk_spacing_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)
GRAY = (150, 150, 150)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

# Canonical center-to-center spacings (px). Adjacent values are separated by
# 40px, far above the finite-resolution oracle tolerance.
CANONICAL_DISTS = [70, 110, 150, 190, 230]

WIDTH = 900
HEIGHT = 780
DIVIDER_Y = 190
DIVIDER_THICK = 3
RADIUS = 18          # fixed disk radius shared by every disk
REF_Y = 110
CAND_YS = [330, 470, 610]
REF_CX = 450
CAND_CX = 450

_QUARANTINE_FLOOR = 25.0  # px


def _draw_pair(draw, cx: int, cy: int, dist: float, color) -> None:
    half = dist / 2.0
    r = RADIUS
    draw.ellipse(
        [cx - half - r, cy - r, cx - half + r, cy + r], fill=color
    )
    draw.ellipse([cx + half - r, cy - r, cx + half + r, cy + r], fill=color)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_dist = rng.choice(CANONICAL_DISTS)
    pool = [d for d in CANONICAL_DISTS if d != ref_dist]
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_dists = [ref_dist] + distractors
    rng.shuffle(cand_dists)

    return {
        "ref_dist": ref_dist,
        "cand_dists": cand_dists,
        "cand_colors": list(PALETTE.keys()),
        "radius": RADIUS,
        "ref_y": REF_Y,
        "cand_ys": list(CAND_YS),
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_pair(draw, REF_CX, int(scene["ref_y"]), float(scene["ref_dist"]), GRAY)
    ys = scene["cand_ys"]
    for i, y in enumerate(ys):
        color = scene["cand_colors"][i]
        _draw_pair(draw, CAND_CX, int(y), float(scene["cand_dists"][i]), PALETTE[color])
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the candidate pair whose spacing
    equals the reference spacing."""
    idx = scene["cand_dists"].index(scene["ref_dist"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute spacing difference (px) between the
    reference pair and any distractor candidate pair."""
    ref = scene["ref_dist"]
    diffs = [abs(ref - d) for d in scene["cand_dists"] if d != ref]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene where a distractor pair comes within
    _QUARANTINE_FLOOR px of the reference spacing (never produced because
    canonical spacings are separated by 40px)."""
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing scene field (reference spacing,
    each candidate spacing, and the bound colors) is rendered verbatim, so the
    raster uniquely determines the gold; no pixel-identical transform exists."""
    return []
