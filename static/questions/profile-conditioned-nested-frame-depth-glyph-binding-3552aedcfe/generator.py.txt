"""Nested-frame containment chain bound to a linear glyph row.

Scene: four coloured rectangular frames drawn strictly nested inside one another
(a containment chain, not a concentric ring set: every frame sits at its own
offset). A separate bottom row shows four grey tiles, each carrying one black
glyph. The declared rule is that the tiles list the frames from the innermost
frame (leftmost tile) outwards to the outermost frame (rightmost tile).

The decision is the glyph bound to the RED frame, i.e. the glyph of the tile
whose index equals the number of frames strictly inside the red frame.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 560
HEIGHT = 420
BACKGROUND = (250, 250, 247)

FRAME_WIDTH = 4
PANEL = (30, 44, 530, 318)

TILE_Y0 = 340
TILE_Y1 = 404
TILE_W = 84
TILE_X0 = 44
TILE_STEP = 128
TILE_OUTLINE = (150, 150, 160)
TILE_FILL = (233, 233, 236)
TEXT_COLOR = (70, 70, 110)
GLYPH_COLOR = (0, 0, 0)
GLYPH_BOX = 46

COLORS = {
    "red": (214, 42, 42),
    "blue": (38, 84, 214),
    "green": (26, 140, 60),
    "purple": (140, 52, 180),
}
COLOR_NAMES = ("red", "blue", "green", "purple")
GLYPHS = ("disk", "square", "triangle", "ring")

QUARANTINE_MARGIN = 5.0


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene; every stored field is drawn in the raster."""
    rng = random.Random(seed * 7919 + 13)

    px0, py0, px1, py1 = PANEL
    outer_w = rng.randint(330, px1 - px0)
    outer_h = rng.randint(212, py1 - py0)
    ox0 = rng.randint(px0, px1 - outer_w)
    oy0 = rng.randint(py0, py1 - outer_h)
    rects = [[ox0, oy0, ox0 + outer_w, oy0 + outer_h]]

    tight_pair = rng.randrange(3) if rng.random() < 0.22 else -1
    tight_side = rng.randrange(4)
    for step in range(3):
        prev = rects[-1]
        insets = [rng.randint(9, 26) for _ in range(4)]
        if step == tight_pair:
            insets[tight_side] = rng.randint(5, 8)
        rects.append(
            [
                prev[0] + insets[0],
                prev[1] + insets[1],
                prev[2] - insets[2],
                prev[3] - insets[3],
            ]
        )

    order = list(COLOR_NAMES)
    rng.shuffle(order)
    frames = [
        {"color": order[depth], "rect": [int(v) for v in rects[depth]]}
        for depth in range(4)
    ]

    tiles = list(GLYPHS)
    rng.shuffle(tiles)
    return {"frames": frames, "tiles": tiles}


def _rects(scene: dict) -> list[list[int]]:
    return [frame["rect"] for frame in scene["frames"]]


def _clearances(scene: dict) -> list[float]:
    """Visible white gaps between each consecutive pair of nested frames."""
    gaps = []
    rects = _rects(scene)
    for outer, inner in zip(rects, rects[1:]):
        gaps.extend(
            [
                inner[0] - outer[0] - FRAME_WIDTH,
                inner[1] - outer[1] - FRAME_WIDTH,
                outer[2] - inner[2] - FRAME_WIDTH,
                outer[3] - inner[3] - FRAME_WIDTH,
            ]
        )
    return [float(gap) for gap in gaps]


def _red_depth_from_inside(scene: dict) -> int:
    """Number of frames strictly inside the red frame (0 = innermost)."""
    frames = scene["frames"]
    for depth, frame in enumerate(frames):
        if frame["color"] == "red":
            return len(frames) - 1 - depth
    raise ValueError("scene has no red frame")


def analytic_gold(scene: dict) -> str:
    return scene["tiles"][_red_depth_from_inside(scene)]


def margin(scene: dict) -> float:
    """Smallest visible separation in the containment chain, in pixels."""
    return float(min(_clearances(scene)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases: every scene field is rendered and label-bearing."""
    return []


def _draw_glyph(draw: ImageDraw.ImageDraw, glyph: str, cx: int, cy: int) -> None:
    half = GLYPH_BOX // 2
    box = [cx - half, cy - half, cx + half, cy + half]
    if glyph == "square":
        draw.rectangle(box, fill=GLYPH_COLOR)
    elif glyph == "disk":
        draw.ellipse(box, fill=GLYPH_COLOR)
    elif glyph == "ring":
        draw.ellipse(box, outline=GLYPH_COLOR, width=8)
    elif glyph == "triangle":
        draw.polygon(
            [(cx, cy - half), (cx - half, cy + half), (cx + half, cy + half)],
            fill=GLYPH_COLOR,
        )
    else:
        raise ValueError(f"unknown glyph {glyph!r}")


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((30, 20), "FRAMES", fill=TEXT_COLOR)
    for frame in scene["frames"]:
        draw.rectangle(
            frame["rect"], outline=COLORS[frame["color"]], width=FRAME_WIDTH
        )

    draw.text((30, 326), "TILES: innermost frame at left", fill=TEXT_COLOR)
    for index, glyph in enumerate(scene["tiles"]):
        x0 = TILE_X0 + index * TILE_STEP
        draw.rectangle(
            [x0, TILE_Y0, x0 + TILE_W, TILE_Y1],
            fill=TILE_FILL,
            outline=TILE_OUTLINE,
            width=3,
        )
        _draw_glyph(draw, glyph, x0 + TILE_W // 2, (TILE_Y0 + TILE_Y1) // 2)
    return image
