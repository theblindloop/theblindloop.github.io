"""Pixel-only inverse arm.

Receives one rendered PNG and nothing else. It segments the four saturated
frame colours, checks that their extents form a strict containment chain, ranks
the red frame by how many frames lie inside it, then reads the glyph on the
tile at that index in the bottom row. It imports no renderer, prompt,
generator, verifier, or analytic-gold code and never sees a latent scene.
"""

from __future__ import annotations

from PIL import Image

ABSTAIN = "abstain"

_SAT_MIN = 60
_BLACK_MAX = 60
_MIN_COLOR_PIXELS = 300
_MIN_GLYPH_PIXELS = 200


def _hue_name(r: int, g: int, b: int) -> str | None:
    """Coarse hue bucket for a saturated pixel colour."""
    high = max(r, g, b)
    low = min(r, g, b)
    if high - low < _SAT_MIN:
        return None
    if r == high:
        if b > g + 40:
            return "purple" if b - r > -90 and b > 120 else "red"
        return "red"
    if g == high:
        return "green"
    if b == high:
        return "purple" if r > g + 40 else "blue"
    return None


def _color_extents(image: Image.Image) -> dict[str, tuple[int, int, int, int, int]]:
    """Bounding box plus pixel count for every hue bucket present."""
    width, height = image.size
    pixels = image.load()
    boxes: dict[str, list[int]] = {}
    for y in range(height):
        for x in range(width):
            name = _hue_name(*pixels[x, y][:3])
            if name is None:
                continue
            box = boxes.get(name)
            if box is None:
                boxes[name] = [x, y, x, y, 1]
            else:
                box[0] = min(box[0], x)
                box[1] = min(box[1], y)
                box[2] = max(box[2], x)
                box[3] = max(box[3], y)
                box[4] += 1
    return {
        name: (box[0], box[1], box[2], box[3], box[4])
        for name, box in boxes.items()
        if box[4] >= _MIN_COLOR_PIXELS
    }


def _strict_chain(boxes: list[tuple[int, int, int, int, int]]) -> bool:
    for outer, inner in zip(boxes, boxes[1:]):
        if not (
            inner[0] > outer[0]
            and inner[1] > outer[1]
            and inner[2] < outer[2]
            and inner[3] < outer[3]
        ):
            return False
    return True


def _black_components(image: Image.Image) -> list[tuple[int, int, int, int, int]]:
    """Four-connected components of near-black ink, as bbox plus area."""
    width, height = image.size
    pixels = image.load()
    mask = bytearray(width * height)
    for y in range(height):
        row = y * width
        for x in range(width):
            r, g, b = pixels[x, y][:3]
            if r < _BLACK_MAX and g < _BLACK_MAX and b < _BLACK_MAX:
                mask[row + x] = 1

    components = []
    for start in range(width * height):
        if not mask[start]:
            continue
        stack = [start]
        mask[start] = 0
        x0 = x1 = start % width
        y0 = y1 = start // width
        area = 0
        while stack:
            index = stack.pop()
            x = index % width
            y = index // width
            area += 1
            if x < x0:
                x0 = x
            if x > x1:
                x1 = x
            if y < y0:
                y0 = y
            if y > y1:
                y1 = y
            if x > 0 and mask[index - 1]:
                mask[index - 1] = 0
                stack.append(index - 1)
            if x + 1 < width and mask[index + 1]:
                mask[index + 1] = 0
                stack.append(index + 1)
            if y > 0 and mask[index - width]:
                mask[index - width] = 0
                stack.append(index - width)
            if y + 1 < height and mask[index + width]:
                mask[index + width] = 0
                stack.append(index + width)
        if area >= _MIN_GLYPH_PIXELS:
            components.append((x0, y0, x1, y1, area))
    components.sort(key=lambda c: c[0])
    return components


def _classify(image: Image.Image, comp: tuple[int, int, int, int, int]) -> str:
    x0, y0, x1, y1, area = comp
    box_w = x1 - x0 + 1
    box_h = y1 - y0 + 1
    if box_w < 20 or box_h < 20 or abs(box_w - box_h) > 8:
        return ABSTAIN
    fill = area / float(box_w * box_h)
    r, g, b = image.getpixel(((x0 + x1) // 2, (y0 + y1) // 2))[:3]
    center_ink = r < _BLACK_MAX and g < _BLACK_MAX and b < _BLACK_MAX
    if not center_ink:
        return "ring" if 0.25 <= fill <= 0.62 else ABSTAIN
    if fill >= 0.88:
        return "square"
    if fill >= 0.66:
        return "disk"
    if fill >= 0.38:
        return "triangle"
    return ABSTAIN


def decision_from_image(image: Image.Image) -> str:
    """Return the glyph bound to the red frame, or ``abstain``."""
    rgb = image.convert("RGB")
    extents = _color_extents(rgb)
    if len(extents) != 4 or "red" not in extents:
        return ABSTAIN

    ordered = sorted(
        extents.items(),
        key=lambda item: (item[1][2] - item[1][0]) * (item[1][3] - item[1][1]),
        reverse=True,
    )
    if not _strict_chain([box for _, box in ordered]):
        return ABSTAIN

    inside_red = [name for name, _ in ordered].index("red")
    rank_from_inner = len(ordered) - 1 - inside_red

    components = _black_components(rgb)
    if len(components) != 4:
        return ABSTAIN
    tops = [c[1] for c in components]
    if max(tops) - min(tops) > 12:
        return ABSTAIN
    return _classify(rgb, components[rank_from_inner])
