"""Renderer and analytic arm for the chromatic-delta route question."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "chromatic-delta-route-0.1.0"
WIDTH = 640
HEIGHT = 480
N_SECTIONS = 33
N_TRANSITIONS = N_SECTIONS - 1
ROUTE_SPAN = 360.0
BACKGROUND = (247, 248, 244)

# Both transition templates contain sixteen changes and their cumulative
# section carriers contain the same number of each color.  Thus the absolute
# color inventory, initial color, and total route ink do not identify the label.
LOOM = tuple(int(bit) for bit in "10011001011001100110011010011001")
KNOT = tuple(int(bit) for bit in "00011100000110001111101100111100")
TEMPLATES = {"A": LOOM, "B": KNOT}
DIFF_POSITIONS = tuple(index for index, (a, b) in enumerate(zip(LOOM, KNOT)) if a != b)

# route-color-0, route-color-1, red start, green end, page background,
# and the quiet route window.  The oracle uses chromatic relations, not a
# renderer import or a scene field, so these palette variants are nuisance.
PALETTES = (
    ((31, 68, 135), (194, 126, 38), (214, 55, 60), (44, 148, 88), (247, 248, 244), (232, 236, 231)),
    ((35, 73, 142), (188, 119, 34), (207, 58, 63), (39, 143, 83), (246, 248, 245), (234, 237, 233)),
    ((27, 62, 124), (199, 132, 45), (218, 61, 58), (49, 151, 91), (248, 247, 243), (231, 235, 230)),
)


def _hamming(code: tuple[int, ...], template: tuple[int, ...]) -> int:
    return sum(left != right for left, right in zip(code, template))


def _classify(code: tuple[int, ...]) -> str:
    if len(code) != N_TRANSITIONS:
        raise ValueError("transition code must contain 32 bits")
    distance_a = _hamming(code, LOOM)
    distance_b = _hamming(code, KNOT)
    return "A" if distance_a <= distance_b else "B"


def sample_scene(seed: int) -> dict:
    """Return a balanced scene with visible carrier and route nuisance."""

    rng = random.Random(int(seed))
    label = "A" if int(seed) % 2 == 0 else "B"
    transitions = list(TEMPLATES[label])
    # The main population is comfortably away from the template boundary.
    # A deterministic minority is deliberately near or on that boundary.
    if int(seed) % 11 == 0:
        for index in rng.sample(DIFF_POSITIONS, len(DIFF_POSITIONS) // 2):
            transitions[index] = 1 - transitions[index]
    elif int(seed) % 17 == 0:
        for index in rng.sample(DIFF_POSITIONS, min(6, len(DIFF_POSITIONS))):
            transitions[index] = 1 - transitions[index]
    return {
        "transitions": transitions,
        "initial_color": int(rng.randrange(2)),
        "x0": int(rng.choice((122, 132, 142, 152))),
        "y0": int(rng.choice((175, 205, 235, 265))),
        "route_shape": int(rng.randrange(4)),
        "stroke": int(rng.choice((8, 9, 10))),
        "palette_variant": int(rng.randrange(len(PALETTES))),
        "reverse": bool(rng.randrange(2)),
        "distractor_seed": int(rng.randrange(1_000_000_000)),
    }


def _section_bits(scene: dict) -> tuple[int, ...]:
    transitions = tuple(int(value) for value in scene["transitions"])
    if len(transitions) != N_TRANSITIONS or any(value not in (0, 1) for value in transitions):
        raise ValueError("scene transition code must contain 32 binary values")
    bits = [int(scene["initial_color"]) & 1]
    for transition in transitions:
        bits.append(bits[-1] ^ transition)
    return tuple(bits)


def _route_points(scene: dict) -> list[tuple[float, float]]:
    shape = int(scene["route_shape"]) % 4
    y0 = float(scene["y0"])
    points: list[tuple[float, float]] = []
    for index in range(N_SECTIONS + 1):
        t = index / float(N_SECTIONS)
        if shape == 0:
            offset = 38.0 * math.sin(math.pi * t)
        elif shape == 1:
            offset = -38.0 * math.sin(math.pi * t)
        elif shape == 2:
            offset = 29.0 * math.sin(2.0 * math.pi * t + 0.35)
        else:
            offset = 24.0 * math.sin(math.pi * t) + 13.0 * math.sin(2.0 * math.pi * t + 0.4)
        x = float(scene["x0"]) + ROUTE_SPAN * t
        points.append((x, y0 + offset))
    if bool(scene["reverse"]):
        points.reverse()
    return points


def _palette(scene: dict) -> tuple[tuple[int, ...], ...]:
    return PALETTES[int(scene["palette_variant"]) % len(PALETTES)]


def _draw_distractors(draw: ImageDraw.ImageDraw, scene: dict, scale: int) -> None:
    rng = random.Random(int(scene["distractor_seed"]))
    quiet = (206, 211, 207)
    for _ in range(5):
        x = rng.choice((32, 54, 586, 608)) + rng.randrange(-8, 9)
        y = rng.choice((48, 72, 407, 432)) + rng.randrange(-7, 8)
        radius = rng.choice((2, 3, 4)) * scale
        cx, cy = int(x * scale), int(y * scale)
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=quiet)


def render(scene: dict) -> Image.Image:
    """Render one antialiased RGB image from the visible route scene."""

    scale = 4
    route0, route1, red, green, background, window = _palette(scene)
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), background)
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, scene, scale)

    x0 = float(scene["x0"])
    x1 = x0 + ROUTE_SPAN
    y0 = float(scene["y0"])
    window_box = (
        int(round((x0 - 24.0) * scale)),
        int(round((y0 - 82.0) * scale)),
        int(round((x1 + 24.0) * scale)),
        int(round((y0 + 82.0) * scale)),
    )
    draw.rounded_rectangle(window_box, radius=15 * scale, fill=window)

    points = _route_points(scene)
    bits = _section_bits(scene)
    scaled = [(int(round(x * scale)), int(round(y * scale))) for x, y in points]
    width = max(1, int(scene["stroke"]) * scale)
    radius = width / 2.0
    colors = (route0, route1)
    for index in range(N_SECTIONS):
        p0, p1 = scaled[index], scaled[index + 1]
        color = colors[bits[index]]
        draw.line((p0, p1), fill=color, width=width)
        # The overlapping endpoint disks make the union of the two section
        # colors one connected thin component without changing midpoint samples.
        for px, py in (p0, p1):
            draw.ellipse(
                (int(round(px - radius)), int(round(py - radius)),
                 int(round(px + radius)), int(round(py + radius))),
                fill=color,
            )

    cap_radius = 4.0 * scale
    start = scaled[0]
    end = scaled[-1]
    draw.ellipse(
        (int(round(start[0] - cap_radius)), int(round(start[1] - cap_radius)),
         int(round(start[0] + cap_radius)), int(round(start[1] + cap_radius))),
        fill=red,
    )
    draw.ellipse(
        (int(round(end[0] - cap_radius)), int(round(end[1] - cap_radius)),
         int(round(end[0] + cap_radius)), int(round(end[1] + cap_radius))),
        fill=green,
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Analytic label; the transition code is its only scene input."""

    code = tuple(int(value) for value in scene["transitions"])
    return _classify(code)


def margin(scene: dict) -> float:
    code = tuple(int(value) for value in scene["transitions"])
    if len(code) != N_TRANSITIONS:
        raise ValueError("transition code must contain 32 bits")
    return float(abs(_hamming(code, LOOM) - _hamming(code, KNOT)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No render-invariant latent alias is claimed.  The transition carrier and
    # every nuisance field are represented in the rendered pixels.
    return []
