"""Independent raster-only inverse arm for three-circle common intersection."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "amber": np.array((225, 145, 24), dtype=float),
    "magenta": np.array((202, 52, 112), dtype=float),
    "teal": np.array((20, 151, 151), dtype=float),
}
LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray, minimum: int = 100) -> list[tuple[int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    boxes = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            yy, xx = stack.pop()
            xs.append(xx); ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            boxes.append((min(xs), min(ys), max(xs), max(ys)))
    return boxes


def _fit_circle(points: np.ndarray) -> tuple[float, float, float]:
    # Algebraic least squares: x^2+y^2 = 2cx*x + 2cy*y + q.
    if len(points) < 100:
        raise ValueError("too few colored outline pixels")
    y = points[:, 0].astype(float)
    x = points[:, 1].astype(float)
    a = np.column_stack((2.0 * x, 2.0 * y, np.ones_like(x)))
    cx, cy, q = np.linalg.lstsq(a, x * x + y * y, rcond=None)[0]
    radius = math.sqrt(max(0.0, q + cx * cx + cy * cy))
    return float(cx), float(cy), radius


def _mec_radius(points: list[tuple[float, float]]) -> float:
    def d(p, q): return math.hypot(p[0] - q[0], p[1] - q[1])
    a, b, c = d(points[1], points[2]), d(points[0], points[2]), d(points[0], points[1])
    sides = sorted((a, b, c))
    if sides[2] ** 2 >= sides[0] ** 2 + sides[1] ** 2:
        return sides[2] / 2.0
    area2 = abs((points[1][0] - points[0][0]) * (points[2][1] - points[0][1])
                - (points[1][1] - points[0][1]) * (points[2][0] - points[0][0]))
    if area2 < 1e-6:
        return sides[2] / 2.0
    return a * b * c / (2.0 * area2)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=float)
    # Enumerate the four visible pale panel interiors; no latent coordinates are used.
    panel_color = np.array((247, 249, 252), dtype=float)
    panel_mask = np.linalg.norm(arr - panel_color, axis=2) < 8.0
    boxes = sorted(_components(panel_mask, minimum=5000), key=lambda b: b[0])
    if len(boxes) != 4:
        raise ValueError(f"expected four visible panels, found {len(boxes)}")
    scores = []
    for x0, y0, x1, y1 in boxes:
        crop = arr[y0:y1 + 1, x0:x1 + 1]
        fits = []
        for color in PALETTE.values():
            mask = np.linalg.norm(crop - color, axis=2) < 58.0
            yy, xx = np.nonzero(mask)
            fits.append(_fit_circle(np.column_stack((yy, xx))))
        centers = [(fit[0], fit[1]) for fit in fits]
        radius = float(np.median([fit[2] for fit in fits]))
        scores.append(radius - _mec_radius(centers))
    order = sorted(range(4), key=lambda i: scores[i], reverse=True)
    if scores[order[0]] <= 2.0 or scores[order[0]] - scores[order[1]] <= 5.0:
        raise ValueError(f"ambiguous common-region scores {scores}")
    return LABELS[order[0]]
