"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 280
PANEL_W, PANEL_H = 184, 228
PANEL_X0, PANEL_GAP, PANEL_Y = 20, 12, 34
RADIUS = 40.0
COLORS = {
    "amber": (225, 145, 24),
    "magenta": (202, 52, 112),
    "teal": (20, 151, 151),
}
LABELS = ("A", "B", "C", "D")


def _dist(a: tuple[float, float], b: tuple[float, float]) -> float:
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _mec_radius(points: Iterable[tuple[float, float]]) -> float:
    """Radius of the minimum enclosing circle of exactly three points."""
    p = list(points)
    a, b, c = _dist(p[1], p[2]), _dist(p[0], p[2]), _dist(p[0], p[1])
    sides = sorted((a, b, c))
    if sides[2] ** 2 >= sides[0] ** 2 + sides[1] ** 2:
        return sides[2] / 2.0
    area2 = abs(
        (p[1][0] - p[0][0]) * (p[2][1] - p[0][1])
        - (p[1][1] - p[0][1]) * (p[2][0] - p[0][0])
    )
    if area2 < 1e-9:
        return sides[2] / 2.0
    return a * b * c / (2.0 * area2)


def _panel_margin(circles: list[dict]) -> float:
    radius = float(circles[0]["r"])
    return radius - _mec_radius([(float(c["x"]), float(c["y"])) for c in circles])


def _triangle(side: float, rotation: float, cx: float, cy: float) -> list[tuple[float, float]]:
    radial = side / math.sqrt(3.0)
    return [
        (
            cx + radial * math.cos(rotation + 2.0 * math.pi * k / 3.0),
            cy + radial * math.sin(rotation + 2.0 * math.pi * k / 3.0),
        )
        for k in range(3)
    ]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    panels = []
    distractor_sides = [75.5, 76.5, 77.5]
    rng.shuffle(distractor_sides)
    di = 0
    for index in range(4):
        if index == winner:
            side = rng.uniform(57.0, 61.0)  # common-region clearance >= 4.78 px
        else:
            side = distractor_sides[di] + rng.uniform(-0.20, 0.20)
            di += 1
        rotation = rng.uniform(0.0, 2.0 * math.pi)
        cx = 92.0 + rng.uniform(-3.0, 3.0)
        cy = 118.0 + rng.uniform(-3.0, 3.0)
        centers = _triangle(side, rotation, cx, cy)
        color_order = list(COLORS)
        rng.shuffle(color_order)
        circles = [
            {"color": color, "x": round(pt[0], 4), "y": round(pt[1], 4), "r": RADIUS}
            for color, pt in zip(color_order, centers)
        ]
        rng.shuffle(circles)
        panels.append(circles)
    return {"circles": panels}


def analytic_gold(scene: dict) -> str:
    scores = [_panel_margin(panel) for panel in scene["circles"]]
    return LABELS[max(range(4), key=lambda i: scores[i])]


def margin(scene: dict) -> float:
    scores = sorted((_panel_margin(panel) for panel in scene["circles"]), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    scores = [_panel_margin(panel) for panel in scene["circles"]]
    ordered = sorted(scores, reverse=True)
    return min(abs(value) for value in scores) < 3.0 or ordered[0] - ordered[1] < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {"circles": []}
    for panel in scene["circles"]:
        circles = [dict(circle) for circle in panel]
        circles[0], circles[1] = circles[1], circles[0]
        twin["circles"].append(circles)
    return [("permute_circle_records", twin)]


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=19 * scale)
    for index, panel in enumerate(scene["circles"]):
        left = PANEL_X0 + index * (PANEL_W + PANEL_GAP)
        box = (left, PANEL_Y, left + PANEL_W, PANEL_Y + PANEL_H)
        draw.rounded_rectangle(tuple(v * scale for v in box), radius=8 * scale,
                               fill=(247, 249, 252), outline=(91, 101, 115), width=2 * scale)
        label = LABELS[index]
        draw.text(((left + 8) * scale, (PANEL_Y + 6) * scale), label,
                  fill=(27, 32, 40), font=font)
        # Sorting makes serialized set order a real pixel-identical latent alias.
        for circle in sorted(panel, key=lambda item: item["color"]):
            x = left + float(circle["x"])
            y = PANEL_Y + float(circle["y"])
            r = float(circle["r"])
            bbox = tuple(v * scale for v in (x - r, y - r, x + r, y + r))
            draw.ellipse(bbox, outline=COLORS[circle["color"]], width=5 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
