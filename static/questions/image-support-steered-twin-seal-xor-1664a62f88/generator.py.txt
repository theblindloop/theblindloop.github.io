"""Twin-seal difference symbol: two disconnected dot-grid seals whose
cell-wise XOR (declared transform) reveals one of four symbols.

The answer depends on both seals: every dot-grid position contributes to the
difference mask, so neither seal alone and no single crop determines it.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

W, H = 400, 240
RADIUS = 6
PITCH = 20

AMBER = (247, 170, 50)
GRAY = (150, 150, 150)
WHITE = (255, 255, 255)

SYMBOLS = {
    "exe": {(0, 0), (0, 2), (1, 1), (2, 0), (2, 2)},
    "plus": {(1, 0), (1, 1), (1, 2), (0, 1), (2, 1)},
    "frame": {
        (0, 0), (0, 1), (0, 2),
        (1, 0), (1, 2),
        (2, 0), (2, 1), (2, 2),
    },
    "block": {(0, 0), (0, 1), (0, 2), (1, 0), (1, 1), (1, 2), (2, 0), (2, 1), (2, 2)},
}

SYMBOL_ORDER = tuple(sorted(SYMBOLS))

LEFT_CENTER = (120, 120)
RIGHT_CENTER = (280, 120)


def _rc():
    for r in range(3):
        for c in range(3):
            yield r, c


def _symbol_bits(symbol: str) -> list[int]:
    return [1 if (r, c) in SYMBOLS[symbol] else 0 for r in range(3) for c in range(3)]


def sample_scene(seed: int) -> dict:
    rng = np.random.RandomState(int(seed))
    symbol = SYMBOL_ORDER[int(rng.randint(0, len(SYMBOL_ORDER)))]
    base = [int(rng.randint(0, 2)) for _ in range(9)]
    right = [int(a ^ b) for a, b in zip(base, _symbol_bits(symbol))]
    return {
        "symbol": symbol,
        "base": base,
        "right": right,
        "left_center": list(LEFT_CENTER),
        "right_center": list(RIGHT_CENTER),
    }


def _draw_seal(draw: ImageDraw.ImageDraw, center, mask) -> None:
    cx, cy = int(center[0]), int(center[1])
    for r in range(3):
        for c in range(3):
            bit = mask[r * 3 + c]
            x = cx + (c - 1) * PITCH
            y = cy + (r - 1) * PITCH
            color = AMBER if bit else GRAY
            draw.ellipse([x - RADIUS, y - RADIUS, x + RADIUS, y + RADIUS], fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (W, H), WHITE)
    draw = ImageDraw.Draw(image)
    right = [
        int(a ^ b) for a, b in zip(scene["base"], _symbol_bits(scene["symbol"]))
    ]
    _draw_seal(draw, scene["left_center"], scene["base"])
    _draw_seal(draw, scene["right_center"], right)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["symbol"]


def _symbol_counts(scene: dict) -> dict[str, int]:
    xor = [
        int(a ^ b)
        for a, b in zip(scene["base"], scene["right"])
    ]
    counts = {}
    for name in SYMBOL_ORDER:
        cells = SYMBOLS[name]
        counts[name] = sum(
            1 for (r, c), bit in zip(_rc(), xor) if bool(bit) == ((r, c) in cells)
        )
    return counts


def margin(scene: dict) -> float:
    counts = _symbol_counts(scene)
    best = max(counts.values())
    runner_up = sorted(counts.values(), reverse=True)[1]
    return float(best - runner_up)


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
