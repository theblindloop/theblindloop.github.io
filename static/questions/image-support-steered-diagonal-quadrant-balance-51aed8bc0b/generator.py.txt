"""Renderer and analytic arm for diagonal quadrant balance."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
INKS = [
    (31, 104, 190),
    (21, 139, 126),
    (111, 78, 181),
    (190, 86, 45),
]
BACKGROUNDS = [(247, 244, 235), (244, 247, 250), (250, 246, 242)]


def _grown_quadrant(count: int, rng: random.Random) -> set[tuple[int, int]]:
    """A connected 5x5 quadrant anchored to both inward axes."""
    occupied = {(u, v) for u in range(5) for v in range(5) if u <= 1 or v <= 1}
    while len(occupied) < count:
        frontier = []
        for u in range(2, 5):
            for v in range(2, 5):
                if (u, v) in occupied:
                    continue
                if any((u + du, v + dv) in occupied for du, dv in ((1, 0), (-1, 0), (0, 1), (0, -1))):
                    frontier.append((u, v))
        occupied.add(rng.choice(frontier))
    return occupied


def _split_total(total: int, rng: random.Random) -> tuple[int, int]:
    low = max(16, total - 25)
    high = min(25, total - 16)
    first = rng.randint(low, high)
    return first, total - first


def _map_quadrant(name: str, cells: Iterable[tuple[int, int]]) -> list[list[int]]:
    mapped = []
    for u, v in cells:
        if name == "ul":
            col, row = 4 - u, 4 - v
        elif name == "ur":
            col, row = 5 + u, 4 - v
        elif name == "ll":
            col, row = 4 - u, 5 + v
        else:
            col, row = 5 + u, 5 + v
        mapped.append([col, row])
    return mapped


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    magnitude = 2 if seed % 8 in (0, 1) else (6, 10, 14)[(seed // 2) % 3]
    signed_gap = magnitude if seed % 2 == 0 else -magnitude
    rising_total = (84 + signed_gap) // 2
    falling_total = 84 - rising_total
    ll_count, ur_count = _split_total(rising_total, rng)
    ul_count, lr_count = _split_total(falling_total, rng)
    counts = {"ul": ul_count, "ur": ur_count, "ll": ll_count, "lr": lr_count}
    cells: list[list[int]] = []
    for name in ("ul", "ur", "ll", "lr"):
        cells.extend(_map_quadrant(name, _grown_quadrant(counts[name], rng)))
    cells.sort(key=lambda item: (item[1], item[0]))
    return {
        "cells": cells,
        "center_x": rng.randrange(228, 285),
        "center_y": rng.randrange(228, 285),
        "tile_size": rng.choice((18, 20, 22)),
        "ink_rgb": list(rng.choice(INKS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
    }


def _quadrant_counts(scene: dict) -> tuple[int, int, int, int]:
    ul = ur = ll = lr = 0
    for col, row in scene["cells"]:
        if row < 5 and col < 5:
            ul += 1
        elif row < 5 and col >= 5:
            ur += 1
        elif row >= 5 and col < 5:
            ll += 1
        else:
            lr += 1
    return ul, ur, ll, lr


def analytic_gold(scene: dict) -> str:
    ul, ur, ll, lr = _quadrant_counts(scene)
    return "rising" if ll + ur > ul + lr else "falling"


def margin(scene: dict) -> float:
    ul, ur, ll, lr = _quadrant_counts(scene)
    return float(abs((ll + ur) - (ul + lr)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    background = tuple(scene["background_rgb"])
    ink = tuple(scene["ink_rgb"])
    cx = int(scene["center_x"])
    cy = int(scene["center_y"])
    size = int(scene["tile_size"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    left = cx - 5 * size
    top = cy - 5 * size
    for col, row in scene["cells"]:
        x0 = left + col * size
        y0 = top + row * size
        outline = tuple(int(value * 0.68) for value in ink)
        draw.rectangle(
            (x0, y0, x0 + size - 1, y0 + size - 1),
            fill=ink,
            outline=outline,
            width=1,
        )

    # The short plus marks the shared quadrant origin without slicing the island.
    halo = tuple(min(255, value + 6) for value in background)
    draw.ellipse((cx - 10, cy - 10, cx + 9, cy + 9), fill=halo)
    draw.line((cx - 7, cy, cx + 6, cy), fill=(25, 28, 33), width=3)
    draw.line((cx, cy - 7, cx, cy + 6), fill=(25, 28, 33), width=3)
    return image
