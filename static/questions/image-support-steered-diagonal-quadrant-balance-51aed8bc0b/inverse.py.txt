"""Independent PNG-only inverse arm for diagonal quadrant balance."""

from __future__ import annotations

import numpy as np
from PIL import Image


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = arr.max(axis=2) < 65
    ys, xs = np.nonzero(black)
    if len(xs) < 12:
        raise ValueError("center cross is absent")
    cx = int(round(float(np.median(xs))))
    cy = int(round(float(np.median(ys))))

    spread = arr.max(axis=2) - arr.min(axis=2)
    ink = (spread > 55) & (arr.max(axis=2) < 230) & ~black
    if int(ink.sum()) < 7000:
        raise ValueError("connected colored island is absent or too small")

    iy, ix = np.nonzero(ink)
    width = int(ix.max() - ix.min() + 1)
    height = int(iy.max() - iy.min() + 1)
    tile_size = int(round((width + height) / 20.0))
    expected_area = 84 * tile_size * tile_size - 308
    if tile_size not in {18, 20, 22} or abs(width - 10 * tile_size) > 1 or abs(height - 10 * tile_size) > 1:
        raise ValueError("the complete square-cell frame cannot be reconstructed")
    if abs(int(ink.sum()) - expected_area) > 0.025 * expected_area:
        raise ValueError("the declared 84-square island is materially incomplete")

    yy, xx = np.indices(ink.shape)
    ul = int((ink & (xx < cx) & (yy < cy)).sum())
    ur = int((ink & (xx >= cx) & (yy < cy)).sum())
    ll = int((ink & (xx < cx) & (yy >= cy)).sum())
    lr = int((ink & (xx >= cx) & (yy >= cy)).sum())
    signed = (ll + ur) - (ul + lr)
    if signed == 0:
        raise ValueError("opposite quadrant areas are tied")
    return "rising" if signed > 0 else "falling"
