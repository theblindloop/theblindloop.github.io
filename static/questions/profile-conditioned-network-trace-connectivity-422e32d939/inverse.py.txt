"""Pixel-only inverse arm for the tracing-connectivity world.

Reads only the final RGB PNG. Segments the blue network strokes by nearest-color
labeling, labels their 4-connected components, then recovers which blue
component each colored marker (and the orange start ring) sits in. The decision
is the color whose blue component also contains the start component -- that is
the terminal reachable from the start by tracing only the blue lines.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "red": (225, 40, 40),
    "green": (40, 170, 70),
    "orange": (245, 150, 20),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "red", "green", "orange", "white"]
# Radius (px) searched around a marker centroid to find its supporting stroke.
_NEAR_R = 7
# Diagonal of a marker used to expand the search box for its centroid band.
_MARKER_R = 9


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack(
        [np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1
    )
    return dists.argmin(axis=-1)


def _label_components(mask: np.ndarray) -> np.ndarray:
    """4-connected labeling (background = -1) of a boolean mask."""
    h, w = mask.shape
    lbl = np.full((h, w), -1, dtype=np.int32)
    cur = 0
    ys, xs = np.nonzero(mask)
    for y, x in zip(ys.tolist(), xs.tolist()):
        if lbl[y, x] != -1:
            continue
        stack = [(y, x)]
        lbl[y, x] = cur
        while stack:
            yy, xx = stack.pop()
            if yy > 0 and mask[yy - 1, xx] and lbl[yy - 1, xx] == -1:
                lbl[yy - 1, xx] = cur
                stack.append((yy - 1, xx))
            if yy < h - 1 and mask[yy + 1, xx] and lbl[yy + 1, xx] == -1:
                lbl[yy + 1, xx] = cur
                stack.append((yy + 1, xx))
            if xx > 0 and mask[yy, xx - 1] and lbl[yy, xx - 1] == -1:
                lbl[yy, xx - 1] = cur
                stack.append((yy, xx - 1))
            if xx < w - 1 and mask[yy, xx + 1] and lbl[yy, xx + 1] == -1:
                lbl[yy, xx + 1] = cur
                stack.append((yy, xx + 1))
        cur += 1
    return lbl


def _marker_component(lbl: np.ndarray, mask: np.ndarray) -> int | None:
    """Blue-component label at the marker: nearest blue pixel near its centroid."""
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return None
    cy = int(round(float(ys.mean())))
    cx = int(round(float(xs.mean())))
    h, w = lbl.shape
    r = _NEAR_R
    best = None
    for dy in range(-r, r + 1):
        for dx in range(-r, r + 1):
            yy, xx = cy + dy, cx + dx
            if 0 <= yy < h and 0 <= xx < w and lbl[yy, xx] != -1:
                return int(lbl[yy, xx])
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["blue"]
    comps = _label_components(blue)

    start_comp = _marker_component(comps, labels == idx["orange"])
    if start_comp is None:
        raise ValueError("start marker component not found")

    for name in ("red", "green"):
        comp = _marker_component(comps, labels == idx[name])
        if comp is not None and comp == start_comp:
            return name
    # Defensive: neither traced to the start component.
    raise ValueError("no terminal shared the start component")
