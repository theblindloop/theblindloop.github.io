"""Tracing world: which colored terminal shares the start ring's connected route.

A winding blue network stroke (the ``main`` polyline) is always a single
connected component. A short, separate blue ``island`` stroke is a second
component placed with a clearly visible white gap. The orange start marker sits
on the main component. Exactly one of the two colored terminals (red / green)
sits on the main component and the other sits on the island. The decision is the
color of the terminal that is reachable from the start by following only the
blue lines -- i.e. the terminal whose blue component also contains the start.
"""

from __future__ import annotations

import math
import random
from typing import Any

import numpy as np
from PIL import Image, ImageDraw

RENDERER_VERSION = "network_trace_connectivity-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20          # blue stroke thickness (px)
START_R = 5         # start marker radius
TERM_R = 4          # terminal dot radius
MARGIN_PAD = 46     # keep geometry away from the image borders

# The island stroke must stay at least this far (px) from the main network, so
# the white gap that separates them is unambiguously visible at pixel scale.
GAP_MIN = 28.0
# Two markers must not crowd each other.
MARKER_SEP = 24.0

SUBSTRATE = "blue"
TERMINAL_COLORS = ["red", "green"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    """Point at arc-distance ``s`` along the polyline (s within total length)."""
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _sample_pts(segments: list[tuple[tuple[float, float], tuple[float, float]]],
                step: float = 2.0) -> list[tuple[float, float]]:
    """Sample points densely along segments (used for analytic separation)."""
    out: list[tuple[float, float]] = []
    for a, b in segments:
        d = math.dist(a, b)
        n = max(1, int(d / step))
        for i in range(n + 1):
            t = i / n
            out.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
    return out


def _polyline_segments(points: list[tuple[float, float]]):
    return list(zip(points, points[1:]))


def _min_clearance(main_pts, island_pts) -> float:
    """Minimum distance between dense samples of the island and main polylines."""
    best = float("inf")
    for p in island_pts:
        for q in main_pts:
            d = math.dist(p, q)
            if d < best:
                best = d
    return best


def _rejection_loop(rng: random.Random):
    """Sample one scene that meets all construction constraints."""
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(20000):
        n_verts = rng.randint(4, 6)
        seg_len = 90.0
        ys = [rng.uniform(lo, hi)]
        xs = [rng.uniform(lo, hi)]
        for _ in range(n_verts):
            xs.append(xs[-1] + rng.uniform(0.45, 1.0) * seg_len)
            ys.append(rng.uniform(lo, hi))
        # Fold the path back so it fits inside the canvas.
        if xs[-1] > w - MARGIN_PAD:
            fold = (xs[-1] - (w - MARGIN_PAD))
            xs = [x - max(0.0, fold * (i / n_verts)) for i, x in enumerate(xs)]
        main = _pts(list(zip(xs, ys)))
        total = _path_length(main)

        # Start near the beginning, connected terminal near the end.
        s_start = rng.uniform(4, 0.6 * total)
        s_conn = rng.uniform(0.7 * total, total - 4)
        start = _point_at(main, s_start)
        conn_pt = _point_at(main, s_conn)

        # Island: a short 1-2 segment blue stroke placed in a free region.
        for _ in range(60):
            base = (rng.uniform(lo, hi), rng.uniform(lo, hi))
            ang = rng.uniform(0.0, 2 * math.pi)
            ilen = rng.uniform(55, 90)
            tip = (base[0] + ilen * math.cos(ang), base[1] + ilen * math.sin(ang))
            clamp = lambda v, a, b: max(a, min(b, v))
            tip = (clamp(tip[0], lo, hi), clamp(tip[1], lo, hi))
            island = [base, tip] if rng.random() < 0.5 else [base, tip, base]
            islands = _polyline_segments(island)
            gap = _min_clearance(
                _sample_pts(_polyline_segments(main)), _sample_pts(_polyline_segments(island))
            )
            if gap >= GAP_MIN + 3.0:
                break
        else:
            continue

        island_pt = _point_at(island, _path_length(island) * rng.uniform(0.3, 0.7))

        connected_idx = rng.choice([0, 1])

        def _good(mk):
            px, py = mk
            return 20 < px < w - 20 and 20 < py < h - 20

        if not (_good(start) and _good(conn_pt) and _good(island_pt)):
            continue
        d_sc = math.dist(start, conn_pt)
        d_si = math.dist(start, island_pt)
        d_ci = math.dist(conn_pt, island_pt)
        if min(d_sc, d_si, d_ci) < MARKER_SEP:
            continue

        # Render the connected terminal on the main network and the other on the
        # island, so the scene's answer-bearing label matches the geometry.
        base_pos = [conn_pt, island_pt]
        pos = base_pos if connected_idx == 0 else [base_pos[1], base_pos[0]]

        scene = {
            "canvas_w": w,
            "canvas_h": h,
            "main": main,
            "island_pts": island,
            "start": start,
            "term_pos": pos,
            "term_colors": list(TERMINAL_COLORS),
            "connected_idx": connected_idx,
            "gap": gap,
        }
        return scene
    raise RuntimeError("could not satisfy isolation constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    w = int(scene["canvas_w"])
    h = int(scene["canvas_h"])
    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    main = _pts(scene["main"])
    island = _pts(scene["island_pts"])
    _draw_polyline(draw, main, COLOR_BLUE, STROKE)
    _draw_polyline(draw, island, COLOR_BLUE, STROKE)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN}
    connected_idx = int(scene["connected_idx"])
    for i, pos in enumerate(scene["term_pos"]):
        c = colors[scene["term_colors"][i]]
        r = TERM_R
        px, py = float(pos[0]), float(pos[1])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=c)
    sx, sy = float(scene["start"][0]), float(scene["start"][1])
    draw.ellipse([sx - START_R, sy - START_R, sx + START_R, sy + START_R], fill=COLOR_ORANGE)
    return img


def analytic_gold(scene: dict) -> str:
    """Color of the terminal whose blue component also holds the start ring."""
    return scene["term_colors"][int(scene["connected_idx"])]


def margin(scene: dict) -> float:
    """Isolation clearance: the white gap between the island and main network."""
    main = _pts(scene["main"])
    island = _pts(scene["island_pts"])
    return _min_clearance(
        _sample_pts(_polyline_segments(main)), _sample_pts(_polyline_segments(island))
    )


def is_quarantined(scene: dict) -> bool:
    if float(scene.get("gap", 0.0)) < GAP_MIN:
        return True
    start = _pts([scene["start"]])[0]
    positions = [_pts([p])[0] for p in scene["term_pos"]]
    all_mk = [start] + positions
    for i in range(len(all_mk)):
        for j in range(i + 1, len(all_mk)):
            if math.dist(all_mk[i], all_mk[j]) < MARKER_SEP:
                return True
    w, h = CANVAS
    for mx, my in all_mk:
        if not (20 <= mx <= w - 20 and 20 <= my <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant transform: the answer-bearing data (which
    colored dot sits on the main component) is directly visible, and swapping
    any colored position changes the pixels as well as the gold. Declared empty."""
    return []
