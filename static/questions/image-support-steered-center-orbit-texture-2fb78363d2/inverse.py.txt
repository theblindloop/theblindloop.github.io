"""Independent pixel-only inverse for the center-orbit texture tile.

The inverse receives only the final RGB raster.  It locates the sole chromatic
tile, validates its connected area, extracts its eight enclosed pale components,
fits an undirected line axis to each, and compares their pooled angular residual
to radial and tangential fields about the recovered tile center.
"""

from __future__ import annotations

from collections import deque
from dataclasses import dataclass
import math

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not expose an unambiguous instance of the public contract."""


@dataclass(frozen=True)
class Measurement:
    decision: str
    mean_radial_angle_deg: float
    runner_up_gap_deg: float
    dash_count: int
    tile_bbox: tuple[int, int, int, int]
    tile_fill: float


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = np.asarray(mask, dtype=bool).copy()
    height, width = work.shape
    components: list[np.ndarray] = []
    for start_y, start_x in np.argwhere(work):
        sy, sx = int(start_y), int(start_x)
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        components.append(np.asarray(points, dtype=np.float64))
    return sorted(components, key=len, reverse=True)


def _tile_mask(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(border.astype(np.float32), axis=0)
    delta = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    chroma = array.max(axis=2).astype(np.int16) - array.min(axis=2).astype(np.int16)
    return (delta >= 58.0) & (chroma >= 48)


def _axis_angle(component: np.ndarray, center_xy: np.ndarray) -> float:
    # Component coordinates arrive as (y, x); PCA is performed in (x, y).
    xy = component[:, ::-1]
    centroid = xy.mean(axis=0)
    centered = xy - centroid
    covariance = centered.T @ centered / len(centered)
    values, vectors = np.linalg.eigh(covariance)
    if values[1] < 4.0 or values[1] / max(values[0], 1e-6) < 3.0:
        raise OracleAbstention("a pale component is not a resolvable dash")
    axis = vectors[:, 1]
    radial = centroid - center_xy
    radius = float(np.linalg.norm(radial))
    if not 25.0 <= radius <= 40.0:
        raise OracleAbstention("a dash is outside the declared orientation ring")
    radial /= radius
    cosine = float(np.clip(abs(np.dot(axis, radial)), 0.0, 1.0))
    return math.degrees(math.acos(cosine))


def measure_image(image: Image.Image) -> Measurement:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or array.shape[:2] != (512, 512):
        raise OracleAbstention("unsupported raster dimensions")

    mask = _tile_mask(array)
    ys, xs = np.nonzero(mask)
    if len(xs) < 8500:
        raise OracleAbstention("too little connected tile area")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    width, height = x1 - x0, y1 - y0
    if not (106 <= width <= 114 and 106 <= height <= 114 and abs(width - height) <= 1):
        raise OracleAbstention("chromatic evidence does not have the declared compact extent")

    local = mask[y0:y1, x0:x1]
    foreground_components = _components(local)
    if len(foreground_components) != 1:
        raise OracleAbstention("the chromatic tile is not one connected component")
    tile_fill = float(local.mean())
    if not 0.885 <= tile_fill <= 0.94:
        raise OracleAbstention("tile area is inconsistent with eight intact pale dashes")
    if not (local[0].all() and local[-1].all() and local[:, 0].all() and local[:, -1].all()):
        raise OracleAbstention("the tile boundary is damaged")

    pale_components = _components(~local)
    dashes = [component for component in pale_components if 70 <= len(component) <= 180]
    if len(dashes) != 8 or len(pale_components) != 8:
        raise OracleAbstention("exactly eight enclosed pale dashes were not recovered")

    center_xy = np.asarray([(width - 1) / 2.0, (height - 1) / 2.0])
    angles = np.asarray([_axis_angle(component, center_xy) for component in dashes])
    if float(np.std(angles)) > 4.5:
        raise OracleAbstention("dash axes do not form one coherent orientation field")
    mean_angle = float(np.mean(angles))
    gap = abs(mean_angle - 45.0)
    if gap <= 5.0:
        raise OracleAbstention("orientation field is too close to the radial/tangential tie")
    decision = "toward" if mean_angle < 45.0 else "around"
    return Measurement(
        decision=decision,
        mean_radial_angle_deg=mean_angle,
        runner_up_gap_deg=gap,
        dash_count=len(dashes),
        tile_bbox=(x0, y0, x1, y1),
        tile_fill=tile_fill,
    )


def decision_from_image(image: Image.Image) -> str:
    return measure_image(image).decision
