"""Deterministic renderer for a compact radial-versus-tangential texture field."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
BACKGROUND = (246, 244, 237)
PALETTE = (
    (24, 112, 148),
    (112, 66, 171),
    (191, 72, 42),
    (35, 132, 93),
    (174, 54, 116),
)
DISTRACTOR_RGB = (151, 149, 144)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, randomized field with a visible 45-degree boundary."""
    value = int(seed)
    rng = random.Random(value ^ 0xC07B_17)
    # Consecutive scene seeds sweep four difficulty levels in each answer class.
    near_or_far = (16, 24, 32, 38)[(value // 2) % 4]
    field_angle = near_or_far if value % 2 == 0 else 90 - near_or_far
    order = list(range(8))
    rng.shuffle(order)
    center_x = rng.choice((160, 352))
    center_y = rng.choice((160, 352))

    # Achromatic remote marks carry no orientation-field information.  Their
    # positions and shapes are sampled without reference to the answer class.
    distractors: list[dict] = []
    while len(distractors) < 7:
        x = rng.randrange(34, CANVAS_SIZE - 34)
        y = rng.randrange(34, CANVAS_SIZE - 34)
        if abs(x - center_x) < 94 and abs(y - center_y) < 94:
            continue
        distractors.append(
            {
                "x": x,
                "y": y,
                "kind": rng.choice(("dot", "square")),
                "size": rng.choice((3, 4, 5)),
            }
        )

    return {
        "canvas_size": CANVAS_SIZE,
        "center_x": center_x,
        "center_y": center_y,
        "tile_size": rng.choice((108, 112)),
        "base_rotation_deg": rng.randrange(0, 45),
        "field_angle_deg": field_angle,
        "dash_order": order,
        "tile_rgb": list(rng.choice(PALETTE)),
        "background_rgb": list(BACKGROUND),
        "distractor_marks": distractors,
    }


def _dash_polygon(scene: dict, index: int) -> list[tuple[int, int]]:
    size = int(scene["tile_size"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radial_deg = float(scene["base_rotation_deg"]) + 45.0 * int(index)
    radial = math.radians(radial_deg)
    px = cx + (0.305 * size) * math.cos(radial)
    py = cy + (0.305 * size) * math.sin(radial)
    axis = math.radians(radial_deg + float(scene["field_angle_deg"]))
    ux, uy = math.cos(axis), math.sin(axis)
    vx, vy = -uy, ux
    half_length = 0.086 * size
    half_width = 0.026 * size
    return [
        (
            round(px + along * half_length * ux + across * half_width * vx),
            round(py + along * half_length * uy + across * half_width * vy),
        )
        for along, across in ((-1, -1), (1, -1), (1, 1), (-1, 1))
    ]


def render(scene: dict) -> Image.Image:
    canvas = int(scene["canvas_size"])
    background = tuple(int(value) for value in scene["background_rgb"])
    image = Image.new("RGB", (canvas, canvas), background)
    draw = ImageDraw.Draw(image)

    for mark in scene["distractor_marks"]:
        x, y, radius = int(mark["x"]), int(mark["y"]), int(mark["size"])
        bounds = (x - radius, y - radius, x + radius, y + radius)
        if str(mark["kind"]) == "dot":
            draw.ellipse(bounds, fill=DISTRACTOR_RGB)
        else:
            draw.rectangle(bounds, fill=DISTRACTOR_RGB)

    size = int(scene["tile_size"])
    x0 = int(scene["center_x"]) - size // 2
    y0 = int(scene["center_y"]) - size // 2
    x1, y1 = x0 + size - 1, y0 + size - 1
    if min(x0, y0) < 24 or max(x1, y1) >= canvas - 24:
        raise ValueError("texture tile is too close to the canvas edge")
    draw.rectangle((x0, y0, x1, y1), fill=tuple(int(value) for value in scene["tile_rgb"]))
    for index in scene["dash_order"]:
        draw.polygon(_dash_polygon(scene, int(index)), fill=background)
    return image


def analytic_gold(scene: dict) -> str:
    return "toward" if float(scene["field_angle_deg"]) < 45.0 else "around"


def margin(scene: dict) -> float:
    """Runner-up angular gap, in degrees, from the visible 45-degree tie."""
    return abs(float(scene["field_angle_deg"]) - 45.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["dash_order"] = list(reversed([int(value) for value in scene["dash_order"]]))
    return [("reverse_commutative_dash_order", twin)]
