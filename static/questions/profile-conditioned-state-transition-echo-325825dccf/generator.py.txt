"""Renderer and analytic side for the state-transition-frequency world.

The raster contains five ordered snapshots of four identity-coded tiles.  Tile
rows are independently shuffled in each panel.  The answer is the directed
state edge that occurs most often when each identity's trajectory is replayed.
"""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
STATE_EDGES = ((0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1))
TRANSITION_NAMES = tuple(
    f"{STATE_NAMES[source]}->{STATE_NAMES[destination]}"
    for source, destination in STATE_EDGES
)

CANVAS = (1865, 800)
BACKGROUND = (247, 249, 252)
INK = (35, 45, 58)
MUTED = (91, 104, 119)
GRID = (185, 195, 207)
CARD = (255, 255, 255)
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (255, 255, 255),
    (247, 251, 254),
    (253, 249, 244),
    (249, 253, 247),
)

PANEL_LEFTS = (28, 395, 762, 1129, 1496)
PANEL_TOP = 220
PANEL_W = 340
PANEL_H = 540
TILE_W = 144
TILE_H = 195
SLOT_OFFSETS = ((18, 86), (178, 86), (18, 323), (178, 323))
IDENTITY_STRIP_W = 34
STATE_MARKER_OFFSET_X = 88
STATE_MARKER_OFFSET_Y = TILE_H // 2

IDENTITY_KEY_X = 28
IDENTITY_KEY_Y = 112
IDENTITY_KEY_STEP = 92
STATE_KEY_X = 450
STATE_KEY_Y = 112
STATE_KEY_STEP = 170
KEY_RADIUS = 15

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)

_VALID_PATHS = tuple(
    (start, *suffix)
    for start in range(len(STATE_NAMES))
    for suffix in itertools.product(range(len(STATE_NAMES)), repeat=4)
    if all(left != right for left, right in zip((start, *suffix), (*suffix,)))
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _identity_color(scene: dict, label: int) -> tuple[int, int, int]:
    return IDENTITY_PALETTE[int(scene["label_palette_order"][int(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_PALETTE[int(scene["state_palette_order"][int(state)])]


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(3, int(radius))
    if int(state) == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif int(state) == 1:
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 4),
            fill=color,
            outline=INK,
            width=2,
        )
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((IDENTITY_KEY_X, IDENTITY_KEY_Y - 32), "IDENTITY KEY", font=small_font, fill=MUTED)
    for label, name in enumerate(LABELS):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        draw.rectangle((left, IDENTITY_KEY_Y, left + 34, IDENTITY_KEY_Y + 32), fill=_identity_color(scene, label), outline=INK, width=1)
        _centered_text(draw, (left + 58, IDENTITY_KEY_Y + 16), name, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((STATE_KEY_X, STATE_KEY_Y - 32), "STATE KEY", font=small_font, fill=MUTED)
    for state, name in enumerate(STATE_NAMES):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 16
        _draw_state_marker(draw, (cx, cy), state, _state_color(scene, state), KEY_RADIUS)
        _centered_text(draw, (cx + 72, cy), name, label_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    left: int,
    top: int,
    label: int,
    state: int,
    label_font,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + TILE_W, top + TILE_H),
        radius=12,
        fill=CARD,
        outline=GRID,
        width=2,
    )
    identity_color = _identity_color(scene, label)
    draw.rectangle((left, top, left + IDENTITY_STRIP_W, top + TILE_H), fill=identity_color)
    _centered_text(
        draw,
        (left + IDENTITY_STRIP_W / 2, top + TILE_H / 2),
        LABELS[label],
        label_font,
        (255, 255, 255),
    )
    _draw_state_marker(
        draw,
        (left + STATE_MARKER_OFFSET_X, top + STATE_MARKER_OFFSET_Y),
        state,
        _state_color(scene, state),
        int(scene["state_marker_radius"]),
    )


def _draw_arrow(draw: ImageDraw.ImageDraw, left: int, right: int) -> None:
    y = PANEL_TOP - 18
    start = (left + PANEL_W + 8, y)
    end = (right - 8, y)
    draw.line((*start, *end), fill=MUTED, width=3)
    x, yy = end
    draw.polygon(((x, yy), (x - 12, yy - 7), (x - 12, yy + 7)), fill=MUTED)


def _transition_counts(panel_states: list[list[int]]) -> dict[tuple[int, int], int]:
    counts = {edge: 0 for edge in STATE_EDGES}
    if len(panel_states) < 2:
        raise ValueError("at least two ordered panels are required")
    width = len(LABELS)
    if any(len(row) != width for row in panel_states):
        raise ValueError("each panel must expose four identity states")
    for left, right in zip(panel_states, panel_states[1:]):
        for before, after in zip(left, right):
            if int(before) == int(after):
                continue
            edge = (int(before), int(after))
            if edge not in counts:
                raise ValueError("state values must be distinct or one of the public states")
            counts[edge] += 1
    return counts


def _winner_and_gap(panel_states: list[list[int]]) -> tuple[tuple[int, int], int]:
    counts = _transition_counts(panel_states)
    ranked = sorted(STATE_EDGES, key=lambda edge: (-counts[edge], STATE_EDGES.index(edge)))
    return ranked[0], counts[ranked[0]] - counts[ranked[1]]


def _make_trajectories(target: tuple[int, int], rng: random.Random) -> list[list[int]]:
    """Sample four valid trajectories until target is a comfortable unique winner."""

    best: tuple[int, int, list[tuple[int, ...]]] | None = None
    for _ in range(12000):
        paths = [tuple(_VALID_PATHS[rng.randrange(len(_VALID_PATHS))]) for _ in LABELS]
        panels = [[paths[label][panel] for label in range(len(LABELS))] for panel in range(5)]
        counts = _transition_counts(panels)
        ranked = sorted(STATE_EDGES, key=lambda edge: (-counts[edge], STATE_EDGES.index(edge)))
        target_count = counts[target]
        runner_count = counts[ranked[1]]
        score = (target_count - runner_count, target_count)
        if best is None or score > best[:2]:
            best = (score[0], score[1], paths)
        if ranked[0] == target and score[0] >= 2:
            return [list(path) for path in paths]
    if best is None:
        raise RuntimeError("could not sample state trajectories")
    return [list(path) for path in best[2]]


def sample_scene(seed: int) -> dict:
    """Sample balanced trajectories and independent visual nuisances."""

    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    target = STATE_EDGES[logic_rng.randrange(len(STATE_EDGES))]
    paths = _make_trajectories(target, logic_rng)
    panel_states = [
        [int(paths[label][panel]) for label in range(len(LABELS))]
        for panel in range(5)
    ]

    label_palette_order = list(range(len(LABELS)))
    state_palette_order = list(range(len(STATE_NAMES)))
    style_rng.shuffle(label_palette_order)
    style_rng.shuffle(state_palette_order)
    slot_orders = []
    for _ in range(5):
        order = list(range(len(LABELS)))
        style_rng.shuffle(order)
        slot_orders.append(order)
    panel_tints = [style_rng.randrange(len(PANEL_TINTS)) for _ in range(5)]

    return {
        "canvas": list(CANVAS),
        "panel_states": panel_states,
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "slot_orders": slot_orders,
        "panel_tints": panel_tints,
        "state_marker_radius": 18 + style_rng.randrange(4),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(28)
    section_font = _font(20)
    label_font = _font(20)
    small_font = _font(15)

    draw.text((28, 22), "STATE TRANSITION ECHO", font=title_font, fill=INK)
    draw.text(
        (28, 62),
        "Bind each identity across the ordered panels, then count directed state changes.",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)
    draw.text((1030, 88), "VISIBLE PANEL ORDER", font=small_font, fill=MUTED)
    draw.text((1030, 112), "P1  →  P2  →  P3  →  P4  →  P5", font=section_font, fill=INK)
    draw.text((1030, 151), "Count only changes between adjacent panels.", font=small_font, fill=MUTED)

    for panel in range(len(PANEL_LEFTS)):
        left = PANEL_LEFTS[panel]
        draw.rounded_rectangle(
            (left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H),
            radius=14,
            fill=PANEL_TINTS[int(scene["panel_tints"][panel])],
            outline=INK,
            width=2,
        )
        _centered_text(draw, (left + PANEL_W / 2, PANEL_TOP + 28), f"P{panel + 1}", section_font, INK)
        draw.line((left + 22, PANEL_TOP + 54, left + PANEL_W - 22, PANEL_TOP + 54), fill=GRID, width=1)
        for slot, (dx, dy) in enumerate(SLOT_OFFSETS):
            label = int(scene["slot_orders"][panel][slot])
            state = int(scene["panel_states"][panel][label])
            _draw_tile(draw, scene, left + dx, PANEL_TOP + dy, label, state, label_font)
    for left, right in zip(PANEL_LEFTS, PANEL_LEFTS[1:]):
        _draw_arrow(draw, left, right)
    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique most frequent directed state transition."""

    winner, gap = _winner_and_gap(scene["panel_states"])
    if gap < 1:
        raise ValueError("transition histogram has no unique winner")
    return TRANSITION_NAMES[STATE_EDGES.index(winner)]


def margin(scene: dict) -> float:
    """Return the top-vs-runner transition count gap."""

    _winner, gap = _winner_and_gap(scene["panel_states"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant answer-bearing alias is declared."""

    return []
