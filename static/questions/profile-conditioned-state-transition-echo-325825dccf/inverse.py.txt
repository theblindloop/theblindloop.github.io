"""Independent pixel-only inverse for the state-transition-frequency world.

Only the public raster contract is duplicated here.  The oracle reads keyed
colors from the image, follows identity stripes across the five fixed panels,
and recomputes the directed transition histogram without importing renderer,
prompts, generation, verification, or analytic-gold code.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
STATE_EDGES = ((0, 1), (0, 2), (1, 0), (1, 2), (2, 0), (2, 1))
TRANSITION_NAMES = tuple(
    f"{STATE_NAMES[source]}->{STATE_NAMES[destination]}"
    for source, destination in STATE_EDGES
)
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
CANVAS = (1865, 800)
PANEL_LEFTS = (28, 395, 762, 1129, 1496)
PANEL_TOP = 220
TILE_H = 195
SLOT_OFFSETS = ((18, 86), (178, 86), (18, 323), (178, 323))
IDENTITY_STRIP_SAMPLE_X = 8
STATE_MARKER_OFFSET_X = 88
STATE_MARKER_OFFSET_Y = TILE_H // 2
IDENTITY_KEY_X = 28
IDENTITY_KEY_Y = 112
IDENTITY_KEY_STEP = 92
STATE_KEY_X = 450
STATE_KEY_Y = 112
STATE_KEY_STEP = 170


def _key_color(frame: Image.Image, xy: tuple[int, int]) -> tuple[int, int, int]:
    color = frame.getpixel(xy)
    if not isinstance(color, tuple) or len(color) != 3:
        raise ValueError("key sample is not RGB")
    return tuple(int(value) for value in color)


def _read_palettes(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    identities: dict[tuple[int, int, int], int] = {}
    for label in range(len(LABELS)):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        color = _key_color(frame, (left + IDENTITY_STRIP_SAMPLE_X, IDENTITY_KEY_Y + 16))
        if color in identities or color not in IDENTITY_PALETTE:
            raise ValueError("identity key is unreadable or reuses a color")
        identities[color] = label

    states: dict[tuple[int, int, int], int] = {}
    for state in range(len(STATE_NAMES)):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        color = _key_color(frame, (cx, STATE_KEY_Y + 16))
        if color in states or color not in STATE_PALETTE:
            raise ValueError("state key is unreadable or reuses a color")
        states[color] = state
    return identities, states


def _snapshot(
    frame: Image.Image,
    identities: dict[tuple[int, int, int], int],
    states: dict[tuple[int, int, int], int],
    panel: int,
) -> list[int]:
    values: list[int | None] = [None] * len(LABELS)
    seen: set[int] = set()
    panel_left = PANEL_LEFTS[panel]
    for dx, dy in SLOT_OFFSETS:
        identity_color = _key_color(
            frame,
            (panel_left + dx + IDENTITY_STRIP_SAMPLE_X, PANEL_TOP + dy + TILE_H // 2),
        )
        if identity_color not in identities:
            raise ValueError("panel tile has no readable identity stripe")
        label = identities[identity_color]
        if label in seen:
            raise ValueError("panel repeats an identity")
        seen.add(label)
        state_color = _key_color(
            frame,
            (panel_left + dx + STATE_MARKER_OFFSET_X, PANEL_TOP + dy + STATE_MARKER_OFFSET_Y),
        )
        if state_color not in states:
            raise ValueError("panel tile has no readable state marker")
        values[label] = states[state_color]
    if seen != set(range(len(LABELS))) or any(value is None for value in values):
        raise ValueError("panel does not expose exactly four identities")
    return [int(value) for value in values]


def decision_from_image(image: Image.Image) -> str:
    """Decode five snapshots and return the unique most frequent state edge."""

    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identities, states = _read_palettes(frame)
    snapshots = [_snapshot(frame, identities, states, panel) for panel in range(len(PANEL_LEFTS))]
    counts = {edge: 0 for edge in STATE_EDGES}
    for left, right in zip(snapshots, snapshots[1:]):
        for before, after in zip(left, right):
            if before != after:
                edge = (before, after)
                if edge not in counts:
                    raise ValueError("decoded state edge is outside the public key")
                counts[edge] += 1
    ranked = sorted(STATE_EDGES, key=lambda edge: (-counts[edge], STATE_EDGES.index(edge)))
    if counts[ranked[0]] == counts[ranked[1]]:
        raise ValueError("pixel evidence has no unique transition winner")
    return TRANSITION_NAMES[STATE_EDGES.index(ranked[0])]
