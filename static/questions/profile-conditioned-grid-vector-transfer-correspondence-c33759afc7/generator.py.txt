"""Deterministic scene sampler, renderer, and analytic arm."""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 760
HEIGHT = 430
GRID_SIZE = 5
CELL_SIZE = 52
PANEL_Y = 105
LEFT_X = 66
RIGHT_X = 434

BACKGROUND = (246, 247, 243)
PANEL_FILL = (255, 255, 255)
INK = (36, 42, 48)
GRID_INK = (116, 124, 132)
CYAN = (20, 180, 190)
YELLOW = (245, 190, 25)
MAGENTA = (210, 55, 170)
ANSWER_RGB = {
    "red": (220, 55, 55),
    "green": (35, 155, 90),
    "blue": (50, 105, 220),
    "orange": (235, 140, 35),
}
ANSWER_COLORS = tuple(ANSWER_RGB)

_VECTORS = tuple(
    (dr, dc)
    for dr in range(-2, 3)
    for dc in range(-2, 3)
    if (dr, dc) != (0, 0) and abs(dr) + abs(dc) >= 2
)


def _valid_starts(vector: tuple[int, int]) -> list[tuple[int, int]]:
    dr, dc = vector
    return [
        (row, col)
        for row in range(GRID_SIZE)
        for col in range(GRID_SIZE)
        if 0 <= row + dr < GRID_SIZE and 0 <= col + dc < GRID_SIZE
    ]


def _add(cell: list[int] | tuple[int, int], vector: tuple[int, int]) -> tuple[int, int]:
    return int(cell[0]) + vector[0], int(cell[1]) + vector[1]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, unique vector-transfer problem from an integer seed."""
    rng = random.Random(int(seed))
    vector = _VECTORS[int(seed) % len(_VECTORS)]
    starts = _valid_starts(vector)
    reference_start = rng.choice(starts)
    target_start = rng.choice(starts)
    if len(starts) > 1 and target_start == reference_start:
        target_start = starts[(starts.index(target_start) + 1 + rng.randrange(len(starts) - 1)) % len(starts)]
    reference_end = _add(reference_start, vector)
    target_destination = _add(target_start, vector)

    answer_color = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    excluded = {target_start, target_destination}
    available = [
        (row, col)
        for row in range(GRID_SIZE)
        for col in range(GRID_SIZE)
        if (row, col) not in excluded
    ]
    rng.shuffle(available)

    # Ensure a visible runner-up while varying its direction and the remaining clutter.
    adjacent = [
        cell
        for cell in available
        if abs(cell[0] - target_destination[0]) + abs(cell[1] - target_destination[1]) == 1
    ]
    first_distractor = rng.choice(adjacent) if adjacent else available[0]
    chosen = [first_distractor]
    chosen.extend(cell for cell in available if cell != first_distractor)
    chosen = chosen[:3]

    markers = [{"color": answer_color, "cell": list(target_destination)}]
    other_colors = [color for color in ANSWER_COLORS if color != answer_color]
    rng.shuffle(other_colors)
    markers.extend(
        {"color": color, "cell": list(cell)}
        for color, cell in zip(other_colors, chosen, strict=True)
    )
    rng.shuffle(markers)

    reference_on_left = bool(int(seed) % 2 == 0)
    reference_origin = [LEFT_X if reference_on_left else RIGHT_X, PANEL_Y]
    target_origin = [RIGHT_X if reference_on_left else LEFT_X, PANEL_Y]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "grid_size": GRID_SIZE,
        "cell_size": CELL_SIZE,
        "reference_origin": reference_origin,
        "target_origin": target_origin,
        "reference_start": list(reference_start),
        "reference_end": list(reference_end),
        "target_start": list(target_start),
        "target_markers": markers,
        "marker_radius": 13 + (int(seed) % 3),
        "reference_on_left": reference_on_left,
    }


def _cell_center(origin: list[int], cell: list[int] | tuple[int, int], cell_size: int) -> tuple[int, int]:
    row, col = int(cell[0]), int(cell[1])
    return (
        int(origin[0]) + col * cell_size + cell_size // 2,
        int(origin[1]) + row * cell_size + cell_size // 2,
    )


def _destination(scene: dict[str, Any]) -> tuple[int, int]:
    dr = int(scene["reference_end"][0]) - int(scene["reference_start"][0])
    dc = int(scene["reference_end"][1]) - int(scene["reference_start"][1])
    return int(scene["target_start"][0]) + dr, int(scene["target_start"][1]) + dc


def analytic_gold(scene: dict[str, Any]) -> str:
    destination = _destination(scene)
    matches = [
        str(marker["color"])
        for marker in scene["target_markers"]
        if tuple(marker["cell"]) == destination
    ]
    if len(matches) != 1:
        raise ValueError(f"destination must have exactly one colored marker, found {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    destination = _destination(scene)
    cell_size = float(scene["cell_size"])
    distances = sorted(
        math.hypot(
            float(marker["cell"][0]) - destination[0],
            float(marker["cell"][1]) - destination[1],
        )
        * cell_size
        for marker in scene["target_markers"]
    )
    if len(distances) < 2:
        return 0.0
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict[str, Any]) -> bool:
    visible_clearance = 2.0 * float(scene["marker_radius"]) + 4.0
    return margin(scene) <= visible_clearance


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["target_markers"] = list(reversed(twin["target_markers"]))
    return [("reverse_target_marker_storage", twin)]


def _draw_grid(draw: ImageDraw.ImageDraw, origin: list[int], grid_size: int, cell_size: int) -> None:
    x0, y0 = int(origin[0]), int(origin[1])
    extent = grid_size * cell_size
    draw.rounded_rectangle(
        [x0 - 16, y0 - 48, x0 + extent + 16, y0 + extent + 22],
        radius=14,
        fill=PANEL_FILL,
        outline=INK,
        width=3,
    )
    for index in range(grid_size + 1):
        x = x0 + index * cell_size
        y = y0 + index * cell_size
        draw.line([x, y0, x, y0 + extent], fill=GRID_INK, width=2)
        draw.line([x0, y, x0 + extent, y], fill=GRID_INK, width=2)


def _draw_centered_label(draw: ImageDraw.ImageDraw, text: str, center_x: int, y: int) -> None:
    font = ImageFont.load_default()
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    draw.text((center_x - width // 2, y), text, fill=INK, font=font)


def _draw_ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse([x - radius, y - radius, x + radius, y + radius], outline=color, width=7)


def _draw_dot(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int]) -> None:
    x, y = center
    draw.ellipse(
        [x - radius, y - radius, x + radius, y + radius],
        fill=color,
        outline=INK,
        width=2,
    )


def _draw_arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    sx, sy = start
    ex, ey = end
    dx, dy = ex - sx, ey - sy
    length = math.hypot(dx, dy)
    if length == 0:
        raise ValueError("reference displacement cannot be zero")
    ux, uy = dx / length, dy / length
    begin = (sx + ux * 25, sy + uy * 25)
    tip = (ex - ux * 21, ey - uy * 21)
    draw.line([begin, tip], fill=INK, width=5)
    px, py = -uy, ux
    base_x, base_y = tip[0] - ux * 13, tip[1] - uy * 13
    draw.polygon(
        [tip, (base_x + px * 7, base_y + py * 7), (base_x - px * 7, base_y - py * 7)],
        fill=INK,
    )


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    grid_size = int(scene["grid_size"])
    cell_size = int(scene["cell_size"])
    reference_origin = scene["reference_origin"]
    target_origin = scene["target_origin"]
    _draw_grid(draw, reference_origin, grid_size, cell_size)
    _draw_grid(draw, target_origin, grid_size, cell_size)

    ref_center = int(reference_origin[0]) + grid_size * cell_size // 2
    target_center = int(target_origin[0]) + grid_size * cell_size // 2
    _draw_centered_label(draw, "REFERENCE MOVE", ref_center, int(reference_origin[1]) - 38)
    _draw_centered_label(draw, "APPLY SAME MOVE", target_center, int(target_origin[1]) - 38)

    reference_start = _cell_center(reference_origin, scene["reference_start"], cell_size)
    reference_end = _cell_center(reference_origin, scene["reference_end"], cell_size)
    _draw_arrow(draw, reference_start, reference_end)
    _draw_ring(draw, reference_start, 22, CYAN)
    _draw_dot(draw, reference_end, int(scene["marker_radius"]), YELLOW)

    target_start = _cell_center(target_origin, scene["target_start"], cell_size)
    _draw_ring(draw, target_start, 22, MAGENTA)
    for marker in scene["target_markers"]:
        center = _cell_center(target_origin, marker["cell"], cell_size)
        _draw_dot(draw, center, int(scene["marker_radius"]), ANSWER_RGB[str(marker["color"])])

    _draw_centered_label(
        draw,
        "Same row change + same column change; no edge wrapping",
        int(scene["width"]) // 2,
        int(scene["height"]) - 32,
    )
    # Accessed deliberately: panel order is a rendered latent even though origins carry it too.
    if not isinstance(scene["reference_on_left"], bool):
        raise ValueError("reference_on_left must be boolean")
    return image
