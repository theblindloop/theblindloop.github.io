"""Independent pixel-only inverse for the vector-transfer world."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

_SPECIAL = {
    "reference_start": (20, 180, 190),
    "reference_end": (245, 190, 25),
    "target_start": (210, 55, 170),
}
_ANSWERS = {
    "red": (220, 55, 55),
    "green": (35, 155, 90),
    "blue": (50, 105, 220),
    "orange": (235, 140, 35),
}


def _centroid(arr: np.ndarray, rgb: tuple[int, int, int], name: str) -> tuple[float, float]:
    mask = np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 120:
        raise ValueError(f"missing or materially erased {name} marker")
    return float(xs.mean()), float(ys.mean())


def decision_from_image(image: Image.Image) -> str:
    """Recover only from pixels: segment landmarks, transport their vector, match a dot."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    reference_start = _centroid(arr, _SPECIAL["reference_start"], "cyan reference-start")
    reference_end = _centroid(arr, _SPECIAL["reference_end"], "yellow reference-end")
    target_start = _centroid(arr, _SPECIAL["target_start"], "magenta target-start")

    destination = (
        target_start[0] + reference_end[0] - reference_start[0],
        target_start[1] + reference_end[1] - reference_start[1],
    )
    distances = []
    for color, rgb in _ANSWERS.items():
        center = _centroid(arr, rgb, color)
        distances.append((math.hypot(center[0] - destination[0], center[1] - destination[1]), color))
    distances.sort()
    winner_distance, winner = distances[0]
    runner_up_distance = distances[1][0]
    if winner_distance > 4.0:
        raise ValueError("no colored dot occupies the transported destination")
    if runner_up_distance - winner_distance <= 30.0:
        raise ValueError("ambiguous destination; abstaining")
    return winner
