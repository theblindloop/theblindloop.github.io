"""Deterministic renderer for flexible-slot orders over shared stock."""
from __future__ import annotations
import itertools, random
from PIL import Image, ImageDraw, ImageFont

SIZE=(960,720); BACKGROUND=(246,244,238); PANEL=(255,254,250); INK=(38,42,48); MUTED=(214,211,203); ACCENT=(178,76,54)
TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91)); STAR_COLOR=(238,185,45); RUN_COLOR=(72,79,89)
RECIPE_COLORS=((208,73,73),(29,151,151),(128,92,184)); DECISIONS=("order-A","order-B","order-C","order-D")
RECIPE_X=(20,335,650); STOCK_X=(28,263,498,733); ORDER_X=(18,254,490,726); SLOT_OFFSETS=(48,98,148); ROW_Y=(504,579,654)
FONT_PATH="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _font(size): return ImageFont.truetype(FONT_PATH,size)
def _reward(plan,recipes): return sum(int(plan[r])*int(recipes[r][1]) for r in range(3))

def _feasible(plan,stock,recipes):
    states={(0,0,0,0)}
    for r,runs in enumerate(plan):
        for choices in recipes[r][0]:
            for _ in range(int(runs)):
                nxt=set()
                for used in states:
                    for color in choices:
                        update=list(used); update[color]+=1
                        if update[color]<=stock[color]: nxt.add(tuple(update))
                states=nxt
                if not states: return False
    return True

def _relaxed_feasible(plan,stock,recipes):
    fixed=[0,0,0,0]; total=0
    for r,runs in enumerate(plan):
        for choices in recipes[r][0]:
            total+=int(runs)
            if len(choices)==1: fixed[choices[0]]+=int(runs)
    return total<=sum(stock) and all(fixed[c]<=stock[c] for c in range(4))

def sample_scene(seed):
    rng=random.Random(int(seed)); pairs=list(itertools.combinations(range(4),2))
    for _ in range(10000):
        recipes=[]
        for _r in range(3):
            shuffled=pairs[:]; rng.shuffle(shuffled)
            recipes.append((((rng.randrange(4),),tuple(shuffled[0]),tuple(shuffled[1])),rng.randint(3,7)))
        recipes=tuple(recipes); stock=tuple(rng.randint(4,9) for _ in range(4))
        plans=[p for p in itertools.product(range(5),repeat=3) if 3<=sum(p)<=9]
        feasible=[p for p in plans if _feasible(p,stock,recipes)]
        hall_bad=[p for p in plans if not _feasible(p,stock,recipes) and _relaxed_feasible(p,stock,recipes)]
        rng.shuffle(feasible); rng.shuffle(hall_bad); chosen=None
        for best in feasible[:80]:
            score=_reward(best,recipes)
            lower=[p for p in feasible if _reward(p,recipes)<=score-2 and p!=best]
            higher_bad=[p for p in hall_bad if _reward(p,recipes)>score]
            if lower and len(higher_bad)>=2:
                chosen=(best,rng.choice(lower),*rng.sample(higher_bad,2)); break
        if chosen is None or len(set(chosen))!=4: continue
        distractors=list(chosen[1:]); rng.shuffle(distractors); distractors.insert(int(seed)%4,chosen[0])
        return {"stock_counts":list(stock),"recipes":[{"slots":[list(s) for s in slots],"stars":stars} for slots,stars in recipes],"order_runs":[list(p) for p in distractors],"layout_variant":rng.randrange(64)}
    raise RuntimeError("unable to sample a valid flexible-stock scene")

def _validated(scene):
    stock=tuple(int(x) for x in scene["stock_counts"])
    recipes=tuple((tuple(tuple(sorted(int(c) for c in slot)) for slot in item["slots"]),int(item["stars"])) for item in scene["recipes"])
    orders=tuple(tuple(int(x) for x in plan) for plan in scene["order_runs"]); variant=int(scene["layout_variant"])
    if len(stock)!=4 or len(recipes)!=3 or len(orders)!=4: raise ValueError("wrong collection size")
    if any(not 4<=x<=9 for x in stock): raise ValueError("stock out of range")
    if any(len(slots)!=3 or sum(len(s)==1 for s in slots)!=1 or any(len(s) not in (1,2) or len(set(s))!=len(s) or any(c not in range(4) for c in s) for s in slots) for slots,_ in recipes): raise ValueError("bad slots")
    if any(not 3<=stars<=7 for _,stars in recipes): raise ValueError("reward out of range")
    if any(len(p)!=3 or any(not 0<=x<=4 for x in p) or not 3<=sum(p)<=9 for p in orders): raise ValueError("order out of range")
    if not 0<=variant<64: raise ValueError("bad layout variant")
    return stock,recipes,orders,variant

def analytic_gold(scene):
    try:
        stock,recipes,orders,_=_validated(scene)
        scored=[(i,_reward(p,recipes)) for i,p in enumerate(orders) if _feasible(p,stock,recipes)]
        if not scored: return "abstain"
        top=max(v for _,v in scored); winners=[i for i,v in scored if v==top]
        return DECISIONS[winners[0]] if len(winners)==1 else "abstain"
    except (KeyError,TypeError,ValueError,IndexError): return "abstain"

def margin(scene):
    stock,recipes,orders,_=_validated(scene)
    scores=sorted((_reward(p,recipes) for p in orders if _feasible(p,stock,recipes)),reverse=True)
    return float(scores[0]-scores[1]) if len(scores)>=2 and scores[0]>scores[1] else 0.0

def is_quarantined(scene):
    try:
        stock,recipes,orders,_=_validated(scene); scored=[_reward(p,recipes) for p in orders if _feasible(p,stock,recipes)]; top=max(scored,default=-1)
        strategic=sum(not _feasible(p,stock,recipes) and _relaxed_feasible(p,stock,recipes) and _reward(p,recipes)>top for p in orders)
        return analytic_gold(scene)=="abstain" or margin(scene)<2 or len(scored)<2 or len(set(orders))!=4 or strategic<2
    except (KeyError,TypeError,ValueError): return True

def latent_symmetries(scene): return []

def _disk(draw,cx,cy,choices,reverse=False,radius=17):
    colors=tuple(TOKEN_COLORS[c] for c in choices)
    if len(colors)==1: draw.ellipse((cx-radius,cy-radius,cx+radius,cy+radius),fill=colors[0],outline=INK,width=2)
    else:
        if reverse: colors=colors[::-1]
        draw.pieslice((cx-radius,cy-radius,cx+radius,cy+radius),90,270,fill=colors[0]); draw.pieslice((cx-radius,cy-radius,cx+radius,cy+radius),270,90,fill=colors[1])
        draw.ellipse((cx-radius,cy-radius,cx+radius,cy+radius),outline=INK,width=2); draw.line((cx,cy-radius+2,cx,cy+radius-2),fill=INK,width=2)

def _component_disks(draw,count,box,color,reverse,radius=8,columns=5):
    x0,y0,x1,y1=box; positions=[(i%columns,i//columns) for i in range(int(count))]
    if reverse: positions.reverse()
    gap=24; used_cols=min(columns,max(1,count)); rows=max(1,(count+columns-1)//columns)
    sx=(x0+x1)//2-(used_cols-1)*gap//2; sy=(y0+y1)//2-(rows-1)*gap//2
    for col,row in positions:
        cx,cy=sx+col*gap,sy+row*gap; draw.ellipse((cx-radius,cy-radius,cx+radius,cy+radius),fill=color,outline=INK,width=1)

def render(scene):
    stock,recipes,orders,variant=_validated(scene); image=Image.new("RGB",SIZE,BACKGROUND); draw=ImageDraw.Draw(image)
    title,label,small=_font(21),_font(16),_font(13)
    draw.text((22,10),"ONE TOKEN PER SLOT · SPLIT SLOT = EITHER COLOR",font=title,fill=INK)
    for ri,((slots,stars),x0) in enumerate(zip(recipes,RECIPE_X)):
        draw.rounded_rectangle((x0,43,x0+290,213),radius=11,fill=PANEL,outline=RECIPE_COLORS[ri],width=4); draw.text((x0+12,53),f"RECIPE {ri+1}",font=small,fill=INK); draw.text((x0+12,82),"SLOTS",font=small,fill=INK)
        for si,(slot,dx) in enumerate(zip(slots,SLOT_OFFSETS)): _disk(draw,x0+dx,133,slot,bool((variant>>(ri*2+si%2))&1))
        draw.line((x0+174,133,x0+204,133),fill=ACCENT,width=5); draw.polygon(((x0+204,133),(x0+193,124),(x0+193,142)),fill=ACCENT)
        _component_disks(draw,stars,(x0+208,96,x0+282,176),STAR_COLOR,bool((variant>>(ri+3))&1),radius=7,columns=3); draw.text((x0+214,181),"STARS",font=small,fill=INK)
    draw.text((22,225),"SHARED START STOCK",font=label,fill=INK)
    for ci,(count,x0) in enumerate(zip(stock,STOCK_X)):
        draw.rounded_rectangle((x0,251,x0+198,381),radius=10,fill=PANEL,outline=INK,width=3); draw.text((x0+12,261),f"COLOR {ci+1}",font=small,fill=INK)
        _component_disks(draw,count,(x0+15,292,x0+183,367),TOKEN_COLORS[ci],bool((variant>>ci)&1),radius=9,columns=5)
    draw.text((22,397),"ORDERS: ONE MARK = ONE RECIPE RUN",font=label,fill=INK)
    for oi,(plan,x0) in enumerate(zip(orders,ORDER_X)):
        draw.rounded_rectangle((x0,426,x0+216,706),radius=11,fill=PANEL,outline=INK,width=3); draw.text((x0+66,438),f"order {chr(65+oi)}",font=small,fill=INK); draw.line((x0+10,470,x0+206,470),fill=MUTED,width=2)
        for ri,(runs,cy) in enumerate(zip(plan,ROW_Y)):
            draw.ellipse((x0+13,cy-14,x0+41,cy+14),fill=RECIPE_COLORS[ri],outline=INK,width=2); text=str(ri+1); tw=draw.textbbox((0,0),text,font=small)[2]; draw.text((x0+27-tw//2,cy-9),text,font=small,fill=PANEL)
            _component_disks(draw,runs,(x0+54,cy-25,x0+202,cy+25),RUN_COLOR,bool((variant+oi+ri)&1),radius=8,columns=5)
            if ri<2: draw.line((x0+10,cy+36,x0+206,cy+36),fill=MUTED,width=1)
    return image
