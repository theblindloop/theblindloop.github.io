"""Independent image-only inverse for flexible-slot stock orders."""
from __future__ import annotations
import numpy as np
from PIL import Image

TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91)); STAR_COLOR=(238,185,45); RUN_COLOR=(72,79,89)
RECIPE_X=(20,335,650); STOCK_X=(28,263,498,733); ORDER_X=(18,254,490,726); SLOT_OFFSETS=(48,98,148); ROW_Y=(504,579,654)

def _components(mask,low,high):
    h,w=mask.shape; remaining=set(int(v) for v in np.flatnonzero(mask)); sizes=[]
    while remaining:
        stack=[remaining.pop()]; size=0
        while stack:
            v=stack.pop(); y,x=divmod(v,w); size+=1
            for n in (v-1,v+1,v-w,v+w):
                if n in remaining:
                    ny,nx=divmod(n,w)
                    if abs(nx-x)+abs(ny-y)==1: remaining.remove(n); stack.append(n)
        if size>=low: sizes.append(size)
    return None if any(size>high for size in sizes) else len(sizes)

def _count_color(arr,box,color,low=100,high=280):
    x0,y0,x1,y1=box; crop=arr[y0:y1,x0:x1]; delta=np.max(np.abs(crop.astype(np.int16)-np.asarray(color,dtype=np.int16)),axis=2)
    return _components(delta<=2,low,high)

def _sample_color(pixel):
    distances=[max(abs(int(pixel[k])-color[k]) for k in range(3)) for color in TOKEN_COLORS]; best=min(range(4),key=distances.__getitem__)
    return best if distances[best]<=2 else None

def _slot(arr,cx,cy):
    left=_sample_color(arr[cy,cx-8]); right=_sample_color(arr[cy,cx+8])
    if left is None or right is None: return None
    return (left,) if left==right else tuple(sorted((left,right)))

def _feasible(plan,stock,recipes):
    states={(0,0,0,0)}
    for r,runs in enumerate(plan):
        for choices in recipes[r][0]:
            for _ in range(runs):
                nxt=set()
                for used in states:
                    for color in choices:
                        update=list(used); update[color]+=1
                        if update[color]<=stock[color]: nxt.add(tuple(update))
                states=nxt
                if not states: return False
    return True

def decision_from_image(image):
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(720,960,3): return "abstain"
    recipes=[]
    for x0 in RECIPE_X:
        slots=tuple(_slot(arr,x0+dx,133) for dx in SLOT_OFFSETS); stars=_count_color(arr,(x0+206,92,x0+286,179),STAR_COLOR,80,210)
        if any(s is None for s in slots) or sum(len(s)==1 for s in slots)!=1 or stars is None or not 3<=stars<=7: return "abstain"
        recipes.append((slots,stars))
    stock=tuple(_count_color(arr,(x0+10,286,x0+188,372),TOKEN_COLORS[i],150,360) for i,x0 in enumerate(STOCK_X))
    if any(v is None or not 4<=v<=9 for v in stock): return "abstain"
    orders=[]
    for x0 in ORDER_X:
        plan=tuple(_count_color(arr,(x0+50,cy-29,x0+206,cy+29),RUN_COLOR,100,280) for cy in ROW_Y)
        if any(v is None or not 0<=v<=4 for v in plan) or not 3<=sum(plan)<=9: return "abstain"
        orders.append(plan)
    scored=[(i,sum(plan[r]*recipes[r][1] for r in range(3))) for i,plan in enumerate(orders) if _feasible(plan,stock,recipes)]
    if not scored: return "abstain"
    top=max(v for _,v in scored); winners=[i for i,v in scored if v==top]
    return f"order-{chr(65+winners[0])}" if len(winners)==1 else "abstain"
