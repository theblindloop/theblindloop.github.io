"""Deterministic renderer and analytic arm for landmark-board correspondence."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

CANVAS = (960, 590)
GRID_N = 4
CELL = 78
BOARD = GRID_N * CELL
BOARD_X = (70, 578)
BOARD_Y = 150

GRID_INK = (48, 63, 83)
TEXT_INK = (24, 31, 43)
GLYPH_INK = (17, 24, 39)
MAGENTA = (202, 38, 145)
COLORS = {
    "red": (222, 61, 69),
    "green": (38, 166, 101),
    "blue": (43, 111, 214),
}
TRANSFORMS = (
    "identity",
    "rot90",
    "rot180",
    "rot270",
    "mirror_lr",
    "mirror_lr_rot90",
    "mirror_lr_rot180",
    "mirror_lr_rot270",
)

# Nine-bit, upright 3x3 landmark glyphs. They are deliberately asymmetric as a
# set and have ample black pixel area for the independent cell decoder.
GLYPH_PATTERNS = (
    0b111010010,  # T
    0b100100111,  # L
    0b101111101,  # H
    0b110011110,  # zig
    0b010111010,  # plus
    0b111101111,  # hollow centre
    0b100010001,  # descending diagonal
    0b001010100,  # ascending diagonal
)


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _apply_transform(pos: Iterable[int], name: str, n: int = GRID_N) -> tuple[int, int]:
    r, c = (int(v) for v in pos)
    if name.startswith("mirror_lr"):
        c = n - 1 - c
        suffix = name.removeprefix("mirror_lr")
        turns = {"": 0, "_rot90": 1, "_rot180": 2, "_rot270": 3}[suffix]
    else:
        turns = {"identity": 0, "rot90": 1, "rot180": 2, "rot270": 3}[name]
    for _ in range(turns):
        r, c = c, n - 1 - r
    return r, c


def _transform_scores(landmarks: list[list[int]], true_name: str, n: int) -> list[int]:
    true_positions = {
        int(pattern): _apply_transform((r, c), true_name, n)
        for pattern, r, c in landmarks
    }
    return [
        sum(
            _apply_transform((r, c), candidate, n) == true_positions[int(pattern)]
            for pattern, r, c in landmarks
        )
        for candidate in TRANSFORMS
    ]


def sample_scene(seed: int) -> dict:
    """Sample one balanced, non-ambiguous scene from an integer seed."""
    rng = random.Random(int(seed))
    transform = TRANSFORMS[(int(seed) // 3) % len(TRANSFORMS)]

    # Resample landmark cells until the correct mapping leads every alternative
    # by at least two exact identity-position matches.
    all_cells = [(r, c) for r in range(GRID_N) for c in range(GRID_N)]
    for _ in range(200):
        chosen = rng.sample(all_cells, 5)
        patterns = rng.sample(GLYPH_PATTERNS, 4)
        landmarks = [[patterns[i], *chosen[i]] for i in range(4)]
        scores = sorted(_transform_scores(landmarks, transform, GRID_N), reverse=True)
        if scores[0] - scores[1] >= 2:
            query = chosen[4]
            break
    else:  # pragma: no cover - combinatorially unreachable with this pool
        raise RuntimeError("could not sample a separated transform assignment")

    mapped_query = _apply_transform(query, transform, GRID_N)
    occupied_target = {
        _apply_transform((r, c), transform, GRID_N) for _, r, c in landmarks
    }
    distractor_cells = rng.sample(
        [p for p in all_cells if p not in occupied_target and p != mapped_query], 2
    )

    answer_color = ("red", "green", "blue")[int(seed) % 3]
    other_colors = [name for name in COLORS if name != answer_color]
    rng.shuffle(other_colors)
    color_cells = {answer_color: list(mapped_query)}
    for name, pos in zip(other_colors, distractor_cells, strict=True):
        color_cells[name] = list(pos)

    return {
        "grid_size": GRID_N,
        "landmarks": landmarks,
        "transform": transform,
        "query_cell": list(query),
        "color_cells": color_cells,
        "reference_on_left": bool(int(seed) % 2 == 0),
        "background": ("#f6f8fc", "#fffaf2")[(int(seed) // 48) % 2],
    }


def _mapped_cell(scene: dict) -> tuple[int, int]:
    return _apply_transform(
        scene["query_cell"], scene["transform"], int(scene.get("grid_size", GRID_N))
    )


def analytic_gold(scene: dict) -> str:
    mapped = _apply_transform(
        scene["query_cell"], scene["transform"], GRID_N
    )
    matches = [
        name for name, pos in scene["color_cells"].items() if tuple(int(v) for v in pos) == mapped
    ]
    if len(matches) != 1:
        raise ValueError("mapped target cell must contain exactly one answer color")
    return matches[0]


def margin(scene: dict) -> float:
    scores = sorted(
        _transform_scores(
            scene["landmarks"], scene["transform"], int(scene.get("grid_size", GRID_N))
        ),
        reverse=True,
    )
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 2.0 or analytic_gold(scene) not in COLORS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["landmarks"] = list(reversed(twin["landmarks"]))
    return [("reverse_disjoint_landmark_draw_order", twin)]


def _center(x0: int, y0: int, pos: Iterable[int]) -> tuple[int, int]:
    r, c = (int(v) for v in pos)
    return x0 + c * CELL + CELL // 2, y0 + r * CELL + CELL // 2


def _draw_glyph(draw: ImageDraw.ImageDraw, cx: int, cy: int, pattern: int) -> None:
    tile = 58
    draw.rounded_rectangle(
        (cx - tile // 2, cy - tile // 2, cx + tile // 2, cy + tile // 2),
        radius=10,
        fill=(236, 231, 211),
        outline=(201, 193, 163),
        width=2,
    )
    block = 9
    pitch = 15
    for k in range(9):
        bit = (pattern >> (8 - k)) & 1
        if not bit:
            continue
        rr, cc = divmod(k, 3)
        x = cx + (cc - 1) * pitch
        y = cy + (rr - 1) * pitch
        draw.rounded_rectangle(
            (x - block // 2, y - block // 2, x + block // 2, y + block // 2),
            radius=2,
            fill=GLYPH_INK,
        )


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, role: str, scene: dict) -> None:
    y0 = BOARD_Y
    draw.rounded_rectangle(
        (x0 - 15, y0 - 58, x0 + BOARD + 15, y0 + BOARD + 24),
        radius=18,
        fill=(255, 255, 255),
        outline=(215, 221, 230),
        width=2,
    )
    header = "REFERENCE" if role == "reference" else "TARGET"
    hb = draw.textbbox((0, 0), header, font=_font(23, bold=True))
    draw.text(
        (x0 + (BOARD - (hb[2] - hb[0])) / 2, y0 - 45),
        header,
        fill=TEXT_INK,
        font=_font(23, bold=True),
    )
    draw.rectangle((x0, y0, x0 + BOARD, y0 + BOARD), fill=(250, 251, 253))
    for i in range(GRID_N + 1):
        p = i * CELL
        draw.line((x0 + p, y0, x0 + p, y0 + BOARD), fill=GRID_INK, width=4)
        draw.line((x0, y0 + p, x0 + BOARD, y0 + p), fill=GRID_INK, width=4)

    transform = scene["transform"]
    for pattern, r, c in scene["landmarks"]:
        pos = (r, c) if role == "reference" else _apply_transform((r, c), transform)
        _draw_glyph(draw, *_center(x0, y0, pos), int(pattern))

    if role == "reference":
        cx, cy = _center(x0, y0, scene["query_cell"])
        draw.ellipse((cx - 25, cy - 25, cx + 25, cy + 25), outline=MAGENTA, width=7)
        draw.ellipse((cx - 5, cy - 5, cx + 5, cy + 5), fill=MAGENTA)
    else:
        for name, pos in scene["color_cells"].items():
            cx, cy = _center(x0, y0, pos)
            draw.ellipse(
                (cx - 23, cy - 23, cx + 23, cy + 23),
                fill=COLORS[name],
                outline=(255, 255, 255),
                width=4,
            )


def render(scene: dict) -> Image.Image:
    if int(scene["grid_size"]) != GRID_N:
        raise ValueError("only a 4x4 grid is supported")
    # Access and validate every gold-bearing field at the public render boundary.
    # The actual drawing remains factored into _draw_board below.
    if scene["transform"] not in TRANSFORMS:
        raise ValueError("unknown board transform")
    if len(scene["query_cell"]) != 2:
        raise ValueError("query_cell must be a row-column pair")
    if set(scene["color_cells"]) != set(COLORS):
        raise ValueError("target must contain one disk of every answer color")
    image = Image.new("RGB", CANVAS, scene["background"])
    draw = ImageDraw.Draw(image)
    title = "MATCH LANDMARKS  •  MAP THE RING  •  READ TARGET COLOR"
    tb = draw.textbbox((0, 0), title, font=_font(25, bold=True))
    draw.text(
        ((CANVAS[0] - (tb[2] - tb[0])) / 2, 27),
        title,
        fill=TEXT_INK,
        font=_font(25, bold=True),
    )
    left_role = "reference" if scene["reference_on_left"] else "target"
    right_role = "target" if scene["reference_on_left"] else "reference"
    _draw_board(draw, BOARD_X[0], left_role, scene)
    _draw_board(draw, BOARD_X[1], right_role, scene)

    note = "Symbols remain upright; cell positions may rotate or mirror."
    nb = draw.textbbox((0, 0), note, font=_font(18))
    draw.text(
        ((CANVAS[0] - (nb[2] - nb[0])) / 2, 528),
        note,
        fill=(75, 85, 101),
        font=_font(18),
    )
    return image
