"""Deterministic Pillow renderer and analytic priority-queue replay arm."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 600
VALUES = (1, 2, 3, 4)
QUESTION = "ADD inserts a job. SERVE removes the smallest. Which job is next?"


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size)


def _normal_op(value: str) -> str:
    if value in {"add", "insert"}:
        return "add"
    if value in {"serve", "take_min"}:
        return "serve"
    raise ValueError(f"unknown operation {value!r}")


def _replay(initial: list[int], operations: list[dict]) -> list[int]:
    waiting = list(initial)
    for item in operations:
        kind = _normal_op(item["op"])
        if kind == "add":
            value = int(item["value"])
            if value not in VALUES:
                raise ValueError("invalid job number")
            waiting.append(value)
        elif waiting:
            waiting.remove(min(waiting))
    return waiting


def sample_scene(seed: int) -> dict:
    """Sample a balanced, nonempty six-step priority replay."""
    rng = random.Random(seed * 130363 + 71)
    target = VALUES[seed % len(VALUES)]
    for _ in range(20000):
        initial = [rng.choice(VALUES) for _ in range(rng.choice((2, 3, 4)))]
        operations: list[dict] = []
        for _step in range(6):
            if rng.random() < 0.55:
                operations.append({"op": "add", "value": rng.choice(VALUES)})
            else:
                operations.append({"op": "serve", "value": None})
        kinds = [item["op"] for item in operations]
        final = _replay(initial, operations)
        if (
            final
            and min(final) == target
            and kinds.count("add") >= 2
            and kinds.count("serve") >= 2
            and len(set(initial + [int(x["value"]) for x in operations if x["op"] == "add"])) >= 3
        ):
            break
    else:
        raise RuntimeError("could not sample a balanced priority scene")

    distractors = [[78 + i * 196 + rng.randint(-7, 7), 303 + rng.randint(-4, 4)] for i in range(5)]
    return {
        "initial": initial,
        "operations": operations,
        "palette_variant": (seed // 4) % 2,
        "shift_x": (-16, 0, 16)[(seed // 8) % 3],
        "distractors": distractors,
    }


def _draw_job(draw: ImageDraw.ImageDraw, value: int, cx: int, cy: int,
              blue: tuple[int, int, int], card_bg: tuple[int, int, int],
              pale: tuple[int, int, int]) -> None:
    if value not in VALUES:
        raise ValueError("job number must be 1 through 4")
    draw.rounded_rectangle((cx - 38, cy - 38, cx + 38, cy + 38), radius=12,
                           fill=card_bg, outline=pale, width=3)
    positions = ((cx - 12, cy - 12), (cx + 12, cy - 12),
                 (cx - 12, cy + 12), (cx + 12, cy + 12))
    for index, (x, y) in enumerate(positions):
        fill = blue if index < value else pale
        draw.ellipse((x - 7, y - 7, x + 7, y + 7), fill=fill)


def render(scene: dict) -> Image.Image:
    initial = [int(x) for x in scene["initial"]]
    operations = scene["operations"]
    variant = int(scene["palette_variant"])
    shift_x = int(scene["shift_x"])
    blue = ((35, 99, 181), (74, 78, 187))[variant]
    green = (45, 151, 78)
    red = (207, 65, 59)
    ink = (31, 41, 53)
    muted = (101, 112, 126)
    card_bg = (249, 250, 252)
    pale = (226, 231, 237)
    image = Image.new("RGB", (WIDTH, HEIGHT), (255, 255, 255))
    draw = ImageDraw.Draw(image)

    draw.text((48, 28), "PRIORITY SCHEDULER REPLAY", fill=ink, font=_font(28, True))
    draw.text((48, 68), QUESTION, fill=muted, font=_font(18))
    draw.text((48, 113), "WAITING JOBS", fill=ink, font=_font(17, True))

    tray_left, tray_right = 190 + shift_x, 770 + shift_x
    draw.rounded_rectangle((tray_left, 140, tray_right, 282), radius=18,
                           fill=card_bg, outline=pale, width=3)
    spacing = 105
    start_x = 480 + shift_x - (len(initial) - 1) * spacing // 2
    for index, value in enumerate(initial):
        _draw_job(draw, value, start_x + index * spacing, 218, blue, card_bg, pale)

    for x, y in scene["distractors"]:
        draw.ellipse((x - 4 + shift_x, y - 4, x + 4 + shift_x, y + 4), fill=pale)

    draw.text((48, 310), "READ CARDS LEFT TO RIGHT", fill=ink, font=_font(18, True))
    for index, item in enumerate(operations):
        cx = 142 + shift_x + index * 135
        x0, x1 = cx - 56, cx + 56
        draw.rounded_rectangle((x0, 348, x1, 529), radius=14,
                               fill=card_bg, outline=pale, width=3)
        kind = _normal_op(item["op"])
        header = green if kind == "add" else red
        draw.rectangle((x0 + 3, 351, x1 - 3, 391), fill=header)
        label = "ADD" if kind == "add" else "SERVE"
        bounds = draw.textbbox((0, 0), label, font=_font(16, True))
        draw.text((cx - (bounds[2] - bounds[0]) // 2, 361), label,
                  fill=(255, 255, 255), font=_font(16, True))
        if kind == "add":
            _draw_job(draw, int(item["value"]), cx, 455, blue, card_bg, pale)
        else:
            draw.line((cx - 26, 454, cx + 26, 454), fill=red, width=9)
            draw.polygon([(cx + 26, 454), (cx + 12, 443), (cx + 12, 465)], fill=red)
        draw.text((cx - 5, 537), str(index + 1), fill=muted, font=_font(15, True))

    draw.text((48, 572), "Green ADD = insert   |   Red SERVE = remove current smallest",
              fill=muted, font=_font(16))
    return image


def analytic_gold(scene: dict) -> str:
    final = _replay(scene["initial"], scene["operations"])
    if not final:
        raise ValueError("empty final scheduler")
    return str(min(final))


def margin(scene: dict) -> float:
    """Runner-up segmentation margin: 8 px between neighboring dot rims."""
    return 8.0


def is_quarantined(scene: dict) -> bool:
    try:
        initial = [int(x) for x in scene["initial"]]
        operations = scene["operations"]
        if not 2 <= len(initial) <= 4 or len(operations) != 6:
            return True
        if any(x not in VALUES for x in initial):
            return True
        for item in operations:
            kind = _normal_op(item["op"])
            if kind == "add" and int(item["value"]) not in VALUES:
                return True
        return not bool(_replay(initial, operations))
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twins: list[tuple[str, dict]] = []
    if len(scene["distractors"]) > 1:
        twin = deepcopy(scene)
        twin["distractors"] = list(reversed(twin["distractors"]))
        twins.append(("distractor_serialization_permutation", twin))
    for index, item in enumerate(scene["operations"]):
        twin = deepcopy(scene)
        twin["operations"][index]["op"] = "insert" if _normal_op(item["op"]) == "add" else "take_min"
        twins.append(("operation_keyword_alias", twin))
        break
    return twins
