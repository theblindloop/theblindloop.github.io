"""Independent image-only inverse for Priority Scheduler Replay."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 20) -> list[dict]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    results: list[dict] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        todo = deque([(y, x)])
        visited[y, x] = True
        points: list[tuple[int, int]] = []
        while todo:
            cy, cx = todo.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    todo.append((ny, nx))
        if len(points) >= minimum:
            ys = [p[0] for p in points]
            xs = [p[1] for p in points]
            results.append({"area": len(points), "x0": min(xs), "x1": max(xs),
                            "y0": min(ys), "y1": max(ys),
                            "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
    return results


def _job_dots(mask: np.ndarray) -> list[dict]:
    dots = []
    for component in _components(mask, minimum=100):
        width = component["x1"] - component["x0"] + 1
        height = component["y1"] - component["y0"] + 1
        if 13 <= width <= 17 and 13 <= height <= 17 and 130 <= component["area"] <= 210:
            dots.append(component)
    return dots


def _initial_values(dots: list[dict], header_top: float) -> list[int]:
    initial = sorted((d for d in dots if d["cy"] < header_top), key=lambda d: d["cx"])
    if not initial:
        raise ValueError("no initial job dots found")
    groups: list[list[dict]] = [[initial[0]]]
    for dot in initial[1:]:
        if dot["cx"] - groups[-1][-1]["cx"] > 48:
            groups.append([dot])
        else:
            groups[-1].append(dot)
    values = [len(group) for group in groups]
    if not 2 <= len(values) <= 4 or any(value not in (1, 2, 3, 4) for value in values):
        raise ValueError(f"invalid initial jobs {values}")
    return values


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    blue_mask = (blue > 135) & (blue - red > 65) & (blue - green > 45)
    green_mask = (green > 115) & (green - red > 45) & (green - blue > 35)
    red_mask = (red > 160) & (red - green > 80) & (red - blue > 75)
    dots = _job_dots(blue_mask)

    headers: list[tuple[str, dict]] = []
    for name, mask in (("add", green_mask), ("serve", red_mask)):
        for component in _components(mask, minimum=1500):
            width = component["x1"] - component["x0"] + 1
            height = component["y1"] - component["y0"] + 1
            if width > 85 and 30 <= height <= 45:
                headers.append((name, component))
    headers.sort(key=lambda pair: pair[1]["cx"])
    if len(headers) != 6:
        raise ValueError(f"expected six ordered operation headers, found {len(headers)}")

    waiting = _initial_values(dots, min(header["y0"] for _, header in headers))
    assigned = sum(waiting)
    for kind, header in headers:
        if kind == "serve":
            if waiting:
                waiting.remove(min(waiting))
            continue
        card_dots = [dot for dot in dots
                     if dot["cy"] > header["y1"] and abs(dot["cx"] - header["cx"]) < 32]
        value = len(card_dots)
        if value not in (1, 2, 3, 4):
            raise ValueError(f"ADD card has invalid dot count {value}")
        assigned += value
        waiting.append(value)

    if assigned != len(dots):
        raise ValueError("unassigned blue evidence")
    if not waiting:
        raise ValueError("scheduler empty after replay")
    return str(min(waiting))
