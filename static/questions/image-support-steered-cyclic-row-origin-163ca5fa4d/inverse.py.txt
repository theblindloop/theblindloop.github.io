"""Independent pixel-only inverse for the cyclic-row badge.

It intentionally duplicates no renderer imports or latent access. Geometry is
estimated from purple raster pixels, then each cell and amber row-origin mark is
sampled from the final RGB image.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

N = 7


def _templates() -> dict[str, np.ndarray]:
    grids: dict[str, np.ndarray] = {}
    grids["X"] = np.asarray([[int(c == r or c == 6 - r) for c in range(7)] for r in range(7)], dtype=np.uint8)
    grids["plus"] = np.asarray([[int(r == 3 or c == 3) for c in range(7)] for r in range(7)], dtype=np.uint8)
    grids["T"] = np.asarray([[int(r == 0 or (c == 3 and r > 0)) for c in range(7)] for r in range(7)], dtype=np.uint8)
    grids["L"] = np.asarray([[int(c == 0 or r == 6) for c in range(7)] for r in range(7)], dtype=np.uint8)
    return grids


TEMPLATES = _templates()


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _badge_bounds(array: np.ndarray) -> tuple[int, int, int, int]:
    red = array[:, :, 0].astype(int)
    green = array[:, :, 1].astype(int)
    blue = array[:, :, 2].astype(int)
    purple = (red > 70) & (blue > 105) & (green < 105) & (red - green > 20) & (blue - green > 30)
    candidates = []
    for x0, y0, x1, y1, area in _components(purple):
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 60 <= width <= 95 and 60 <= height <= 95 and area >= 500:
            candidates.append((area, x0, y0, x1, y1))
    if len(candidates) != 1:
        raise ValueError("exactly one intact compact badge is required")
    _, x0, y0, x1, y1 = candidates[0]
    if abs((x1 - x0) - (y1 - y0)) > 3:
        raise ValueError("badge is not square")
    return x0, y0, x1, y1


def _recover(image: Image.Image) -> tuple[str, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, x1, y1 = _badge_bounds(array)
    width = x1 - x0 + 1
    pad = int(round(width / 16.0))
    cell = (width - 2 * pad) / N
    if not 8.0 <= cell <= 12.0:
        raise ValueError("unstable badge sampling")

    shown = np.zeros((N, N), dtype=np.uint8)
    origins = []
    for r in range(N):
        origin_hits = []
        for c in range(N):
            left = x0 + pad + c * cell
            top = y0 + pad + r * cell
            cx = int(round(left + 0.65 * cell))
            cy = int(round(top + 0.65 * cell))
            pixel = array[cy, cx].astype(int)
            shown[r, c] = int(pixel.mean() < 130)
            patch = array[int(round(top + 1)):int(round(top + 5)), int(round(left + 1)):int(round(left + 5))].astype(int)
            amber = (patch[:, :, 0] > 205) & (patch[:, :, 1] > 110) & (patch[:, :, 1] < 210) & (patch[:, :, 2] < 95)
            if int(amber.sum()) >= 8:
                origin_hits.append(c)
        if len(origin_hits) != 1:
            raise ValueError("each row must expose exactly one amber origin")
        origins.append(origin_hits[0])

    rectified = np.zeros_like(shown)
    for r, origin in enumerate(origins):
        rectified[r] = np.roll(shown[r], -origin)
    scores = sorted((int(np.count_nonzero(rectified != template)), name) for name, template in TEMPLATES.items())
    if len(scores) < 2 or scores[0][0] == scores[1][0]:
        raise ValueError("ambiguous rectified symbol")
    return scores[0][1], float(scores[1][0] - scores[0][0])


def decision_and_margin_from_image(image: Image.Image) -> tuple[str, float]:
    return _recover(image)


def decision_from_image(image: Image.Image) -> str:
    return _recover(image)[0]
