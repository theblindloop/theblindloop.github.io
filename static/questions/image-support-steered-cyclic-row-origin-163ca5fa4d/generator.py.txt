"""Deterministic renderer for a compact cyclic-row rectification question."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 640
N = 7
CELL = 10
PAD = 5
BADGE = N * CELL + 2 * PAD
SYMBOLS = ("X", "plus", "T", "L")
PALETTES = (
    ((111, 72, 154), (25, 49, 64), (244, 174, 54)),
    ((133, 64, 139), (35, 55, 70), (247, 160, 49)),
)


def _templates() -> dict[str, list[list[int]]]:
    grids: dict[str, list[list[int]]] = {}
    x = [[int(c == r or c == N - 1 - r) for c in range(N)] for r in range(N)]
    plus = [[int(r == N // 2 or c == N // 2) for c in range(N)] for r in range(N)]
    tee = [[int(r == 0 or (c == N // 2 and r > 0)) for c in range(N)] for r in range(N)]
    ell = [[int(c == 0 or r == N - 1) for c in range(N)] for r in range(N)]
    grids.update({"X": x, "plus": plus, "T": tee, "L": ell})
    return grids


TEMPLATES = _templates()


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random((int(seed) ^ 0xC7C1C) & 0xFFFFFFFFFFFFFFFF)
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    # A compact cue is translated broadly enough to prevent absolute-position shortcuts.
    badge_x = rng.randrange(85, CANVAS - BADGE - 84)
    badge_y = rng.randrange(85, CANVAS - BADGE - 84)
    shifts = [rng.randrange(N) for _ in range(N)]
    palette = rng.randrange(len(PALETTES))
    decorations = []
    for _ in range(6):
        while True:
            x = rng.randrange(35, CANVAS - 35)
            y = rng.randrange(35, CANVAS - 35)
            if not (badge_x - 45 <= x <= badge_x + BADGE + 45 and badge_y - 45 <= y <= badge_y + BADGE + 45):
                break
        decorations.append([x, y, rng.randrange(3, 8)])
    return {
        "symbol": symbol,
        "row_shifts": shifts,
        "badge_x": badge_x,
        "badge_y": badge_y,
        "palette": palette,
        "decorations": decorations,
    }


def _display_grid(scene: dict[str, Any]) -> tuple[list[list[int]], list[int]]:
    base = TEMPLATES[scene["symbol"]]
    shifts = [int(value) % N for value in scene["row_shifts"]]
    shown = []
    origins = []
    for row, shift in zip(base, shifts, strict=True):
        shown.append(row[-shift:] + row[:-shift] if shift else list(row))
        origins.append(shift)
    return shown, origins


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), (249, 248, 244))
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        shade = (218 + int(radius) % 2 * 8,) * 3
        draw.ellipse((int(x) - int(radius), int(y) - int(radius), int(x) + int(radius), int(y) + int(radius)), outline=shade, width=2)

    border, dark, amber = PALETTES[int(scene["palette"]) % len(PALETTES)]
    bx, by = int(scene["badge_x"]), int(scene["badge_y"])
    draw.rounded_rectangle((bx, by, bx + BADGE - 1, by + BADGE - 1), radius=7, fill=border)
    draw.rectangle((bx + PAD, by + PAD, bx + PAD + N * CELL - 1, by + PAD + N * CELL - 1), fill=(241, 238, 229))
    shown, origins = _display_grid(scene)
    for r in range(N):
        for c in range(N):
            x0 = bx + PAD + c * CELL
            y0 = by + PAD + r * CELL
            fill = dark if shown[r][c] else (241, 238, 229)
            draw.rectangle((x0 + 1, y0 + 1, x0 + CELL - 2, y0 + CELL - 2), fill=fill)
            if c == origins[r]:
                draw.rectangle((x0 + 1, y0 + 1, x0 + 4, y0 + 4), fill=amber)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["symbol"])


def margin(scene: dict[str, Any]) -> float:
    chosen = TEMPLATES[scene["symbol"]]
    distances = []
    for name, template in TEMPLATES.items():
        if name != scene["symbol"]:
            distances.append(sum(a != b for ra, rb in zip(chosen, template, strict=True) for a, b in zip(ra, rb, strict=True)))
    return float(min(distances))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
