"""Code-native renderer and analytic world for a two-stage grid motion."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1120
HEIGHT = 720
BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
INK = (24, 32, 48)
MUTED = (84, 96, 115)
GRID = (194, 202, 214)
AXIS = (72, 91, 120)
GOLD = (232, 171, 24)

GRID_LEFT = 105
GRID_TOP = 165
CELL = 92
BOARD_CENTER = (GRID_LEFT + 2 * CELL + CELL // 2, GRID_TOP + 2 * CELL + CELL // 2)

TURN_BANDS = {
    "cw": (35, 111, 211),
    "ccw": (119, 70, 184),
}
MIRROR_BANDS = {
    "slash": (0, 148, 120),
    "backslash": (211, 86, 43),
}
SECTORS = ("northwest", "northeast", "southwest", "southeast")
START_CELLS = [(x, y) for x in range(-2, 3) for y in range(-2, 3) if x != 0 and y != 0]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, int], str, str]:
    cell = scene["start_cell"]
    if not isinstance(cell, (list, tuple)) or len(cell) != 2:
        raise ValueError("start_cell must contain two coordinates")
    x, y = int(cell[0]), int(cell[1])
    if (x, y) not in START_CELLS:
        raise ValueError("start_cell must be an off-axis cell in the 5 by 5 board")
    turn = str(scene["turn_direction"])
    mirror = str(scene["mirror_axis"])
    if turn not in TURN_BANDS:
        raise ValueError("turn_direction must be cw or ccw")
    if mirror not in MIRROR_BANDS:
        raise ValueError("mirror_axis must be slash or backslash")
    return (x, y), turn, mirror


def _apply_motion(cell: tuple[int, int], turn: str, mirror: str) -> tuple[int, int]:
    x, y = cell
    # Image coordinates use +x to the right and +y downward.  Thus a visual
    # clockwise quarter-turn is (x,y) -> (-y,x).
    if turn == "cw":
        x, y = -y, x
    else:
        x, y = y, -x
    # A slash is the bottom-left to top-right diagonal; a backslash is the
    # top-left to bottom-right diagonal.
    if mirror == "slash":
        x, y = -y, -x
    else:
        x, y = y, x
    return x, y


def _sector(cell: tuple[int, int]) -> str:
    x, y = cell
    if x == 0 or y == 0:
        raise ValueError("cell lies on the central sector boundary")
    if x < 0 and y < 0:
        return "northwest"
    if x > 0 and y < 0:
        return "northeast"
    if x < 0 and y > 0:
        return "southwest"
    return "southeast"


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced off-axis start and two independent motion cards."""
    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 17)
    wanted = SECTORS[(seed * 2_654_435_761 + 101) % len(SECTORS)]
    options = [
        (cell, turn, mirror)
        for cell in START_CELLS
        for turn in TURN_BANDS
        for mirror in MIRROR_BANDS
        if _sector(_apply_motion(cell, turn, mirror)) == wanted
    ]
    cell, turn, mirror = rng.choice(options)
    return {
        "start_cell": [cell[0], cell[1]],
        "turn_direction": turn,
        "mirror_axis": mirror,
    }


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=(207, 214, 224), width=2)


def _draw_turn_icon(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: str
) -> None:
    cx, cy = center
    box = (cx - 58, cy - 58, cx + 58, cy + 58)
    draw.arc(box, start=35, end=315, fill=INK, width=9)
    if direction == "cw":
        draw.polygon([(cx + 45, cy - 52), (cx + 71, cy - 38), (cx + 43, cy - 25)], fill=INK)
    else:
        draw.polygon([(cx - 45, cy + 52), (cx - 71, cy + 38), (cx - 43, cy + 25)], fill=INK)
    draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=INK)


def _draw_mirror_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], axis: str) -> None:
    cx, cy = center
    if axis == "slash":
        draw.line((cx - 58, cy + 58, cx + 58, cy - 58), fill=INK, width=11)
        draw.polygon([(cx - 72, cy + 54), (cx - 51, cy + 72), (cx - 52, cy + 43)], fill=INK)
        draw.polygon([(cx + 72, cy - 54), (cx + 51, cy - 72), (cx + 52, cy - 43)], fill=INK)
    else:
        draw.line((cx - 58, cy - 58, cx + 58, cy + 58), fill=INK, width=11)
        draw.polygon([(cx - 72, cy - 54), (cx - 51, cy - 72), (cx - 52, cy - 43)], fill=INK)
        draw.polygon([(cx + 72, cy + 54), (cx + 51, cy + 72), (cx + 52, cy + 43)], fill=INK)


def _cell_center(cell: tuple[int, int]) -> tuple[int, int]:
    return (BOARD_CENTER[0] + cell[0] * CELL, BOARD_CENTER[1] + cell[1] * CELL)


def render(scene: dict[str, Any]) -> Image.Image:
    cell, turn, mirror = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 24), "PIVOT-FRAME MOTION", font=_font(30, True), fill=INK)
    draw.text(
        (40, 62),
        "Track the gold dot through the two cards in order.",
        font=_font(17),
        fill=MUTED,
    )

    _panel(draw, (35, 95, 650, 680))
    draw.text((58, 112), "5 BY 5 MOTION BOARD", font=_font(19, True), fill=INK)
    draw.ellipse((505, 112, 521, 128), fill=GOLD, outline=INK, width=1)
    draw.text((530, 108), "GOLD START", font=_font(14, True), fill=MUTED)

    grid_right = GRID_LEFT + 5 * CELL
    grid_bottom = GRID_TOP + 5 * CELL
    for index in range(6):
        x = GRID_LEFT + index * CELL
        y = GRID_TOP + index * CELL
        draw.line((x, GRID_TOP, x, grid_bottom), fill=GRID, width=2)
        draw.line((GRID_LEFT, y, grid_right, y), fill=GRID, width=2)
    draw.line((BOARD_CENTER[0], GRID_TOP, BOARD_CENTER[0], grid_bottom), fill=AXIS, width=4)
    draw.line((GRID_LEFT, BOARD_CENTER[1], grid_right, BOARD_CENTER[1]), fill=AXIS, width=4)
    draw.text((61, 154), "N", font=_font(14, True), fill=AXIS)
    draw.text((337, 642), "S", font=_font(14, True), fill=AXIS, anchor="ma")
    draw.text((82, 389), "W", font=_font(14, True), fill=AXIS, anchor="mm")
    draw.text((593, 389), "E", font=_font(14, True), fill=AXIS, anchor="mm")
    draw.text((61, 139), "NW", font=_font(18, True), fill=MUTED)
    draw.text((565, 139), "NE", font=_font(18, True), fill=MUTED)
    draw.text((61, 635), "SW", font=_font(18, True), fill=MUTED)
    draw.text((565, 635), "SE", font=_font(18, True), fill=MUTED)

    sx, sy = _cell_center(cell)
    draw.ellipse((sx - 27, sy - 27, sx + 27, sy + 27), fill=GOLD, outline=INK, width=4)
    draw.ellipse((sx - 9, sy - 13, sx - 2, sy - 6), fill=(255, 239, 164))

    _panel(draw, (695, 95, 1085, 320))
    draw.rectangle((720, 116, 1060, 148), fill=TURN_BANDS[turn])
    draw.text((740, 132), "MOTION 1  ·  QUARTER-TURN", font=_font(16, True), fill=(255, 255, 255), anchor="lm")
    draw.text((740, 174), "Turn around the board center", font=_font(18, True), fill=INK)
    _draw_turn_icon(draw, (815, 247), turn)
    turn_label = "CW" if turn == "cw" else "CCW"
    turn_word = "clockwise" if turn == "cw" else "counterclockwise"
    draw.text((900, 231), turn_label, font=_font(27, True), fill=INK, anchor="lm")
    draw.text((900, 268), turn_word, font=_font(13), fill=MUTED, anchor="lm")

    _panel(draw, (695, 350, 1085, 575))
    draw.rectangle((720, 371, 1060, 403), fill=MIRROR_BANDS[mirror])
    draw.text((740, 387), "MOTION 2  ·  REFLECT", font=_font(16, True), fill=(255, 255, 255), anchor="lm")
    draw.text((740, 423), "Reflect across this diagonal", font=_font(16, True), fill=INK)
    _draw_mirror_icon(draw, (815, 515), mirror)
    axis_label = "/" if mirror == "slash" else "\\"
    draw.text((920, 492), axis_label, font=_font(46, True), fill=INK, anchor="mm")
    draw.text((955, 537), "slash" if mirror == "slash" else "backslash", font=_font(15, True), fill=MUTED, anchor="mm")

    _panel(draw, (695, 610, 1085, 680), radius=14)
    draw.text((720, 633), "ORDER", font=_font(15, True), fill=MUTED, anchor="lm")
    draw.text((805, 633), "1  then  2", font=_font(20, True), fill=INK, anchor="lm")
    draw.text((975, 633), "final sector", font=_font(15), fill=MUTED, anchor="lm")
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    cell, turn, mirror = _validate_scene(scene)
    return _sector(_apply_motion(cell, turn, mirror))


def margin(scene: dict[str, Any]) -> float:
    cell, turn, mirror = _validate_scene(scene)
    x, y = _apply_motion(cell, turn, mirror)
    return float(min(abs(x), abs(y)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    cell, turn, mirror = _validate_scene(scene)
    x, y = _apply_motion(cell, turn, mirror)
    return x == 0 or y == 0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
