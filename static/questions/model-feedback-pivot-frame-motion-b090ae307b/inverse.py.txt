"""Independent image-only inverse for the pivot-frame motion world."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1120
HEIGHT = 720
GOLD = np.array((232, 171, 24), dtype=np.int16)
BOARD_CENTER = (335, 395)
CELL = 92
TURN_BANDS = {
    "cw": np.array((35, 111, 211), dtype=np.int16),
    "ccw": np.array((119, 70, 184), dtype=np.int16),
}
MIRROR_BANDS = {
    "slash": np.array((0, 148, 120), dtype=np.int16),
    "backslash": np.array((211, 86, 43), dtype=np.int16),
}


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise ValueError("the gold start mark is not recoverable")
    return float(xs.mean()), float(ys.mean())


def _card_value(
    arr: np.ndarray,
    bounds: tuple[int, int, int, int],
    colors: dict[str, np.ndarray],
) -> str:
    x0, y0, x1, y1 = bounds
    crop = arr[y0:y1, x0:x1]
    counts = {name: int(_near(crop, color, 8).sum()) for name, color in colors.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    other = max(value for key, value in counts.items() if key != name)
    if count < 2000 or count <= other * 2:
        raise ValueError(f"operation card is unreadable: {counts}")
    return name


def _apply_motion(cell: tuple[int, int], turn: str, mirror: str) -> tuple[int, int]:
    x, y = cell
    if turn == "cw":
        x, y = -y, x
    else:
        x, y = y, -x
    if mirror == "slash":
        x, y = -y, -x
    else:
        x, y = y, x
    return x, y


def _sector(cell: tuple[int, int]) -> str:
    x, y = cell
    if x == 0 or y == 0:
        raise ValueError("transformed mark lies on a sector boundary")
    if x < 0 and y < 0:
        return "northwest"
    if x > 0 and y < 0:
        return "northeast"
    if x < 0 and y > 0:
        return "southwest"
    return "southeast"


def decision_from_image(image: Image.Image) -> str:
    """Recover the final sector from RGB pixels only."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected canvas size {rgb.size}")
    arr = np.asarray(rgb, dtype=np.int16)
    gx, gy = _centroid(_near(arr, GOLD, 12))
    start_x = int(round((gx - BOARD_CENTER[0]) / CELL))
    start_y = int(round((gy - BOARD_CENTER[1]) / CELL))
    if not (-2 <= start_x <= 2 and -2 <= start_y <= 2) or start_x == 0 or start_y == 0:
        raise ValueError(f"gold mark is not on a recoverable off-axis cell: {(gx, gy)}")
    turn = _card_value(arr, (720, 116, 1060, 149), TURN_BANDS)
    mirror = _card_value(arr, (720, 371, 1060, 404), MIRROR_BANDS)
    return _sector(_apply_motion((start_x, start_y), turn, mirror))
