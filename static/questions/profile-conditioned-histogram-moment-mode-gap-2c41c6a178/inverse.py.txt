"""Independent PNG-only inverse for histogram moment-to-mode gaps.

This module intentionally duplicates only the public color vocabulary.  It does
not import renderer, scene data, coordinates, analytic labels, or generation code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_COLOR_RGB = {
    "red": (216, 66, 78),
    "blue": (40, 120, 181),
    "green": (42, 157, 104),
    "orange": (225, 132, 44),
}


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    if values.size == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = previous = int(values[0])
    for raw in values[1:]:
        value = int(raw)
        if value != previous + 1:
            groups.append((start, previous))
            start = value
        previous = value
    groups.append((start, previous))
    return groups


def _measure_histogram(array: np.ndarray, color: str) -> list[int]:
    rgb = np.array(_COLOR_RGB[color], dtype=np.uint8)
    mask = np.all(array == rgb, axis=2)
    x_groups = _runs(np.flatnonzero(mask.sum(axis=0) >= 24))
    if len(x_groups) != 5:
        raise ValueError(f"{color}: expected five separable bars, recovered {len(x_groups)}")
    measured: list[tuple[float, int, int]] = []
    for x0, x1 in x_groups:
        sub = mask[:, x0 : x1 + 1]
        ys, _ = np.nonzero(sub)
        if len(ys) < 900:
            raise ValueError(f"{color}: colored component is too small")
        y0, y1 = int(ys.min()), int(ys.max())
        width, height = x1 - x0 + 1, y1 - y0 + 1
        occupancy = float(sub[y0 : y1 + 1].mean())
        if not 35 <= width <= 41 or height < 24 or occupancy < 0.985:
            raise ValueError(f"{color}: component is not a solid histogram bar")
        measured.append(((x0 + x1) / 2.0, height, y1))
    measured.sort(key=lambda item: item[0])
    widths = [x_groups[index][1] - x_groups[index][0] + 1 for index in range(5)]
    bottoms = [item[2] for item in measured]
    centers = [item[0] for item in measured]
    gaps = [centers[index + 1] - centers[index] for index in range(4)]
    if max(widths) - min(widths) > 1 or max(bottoms) - min(bottoms) > 1:
        raise ValueError(f"{color}: bars do not share width and baseline")
    if max(gaps) - min(gaps) > 1 or min(gaps) < max(widths) + 5:
        raise ValueError(f"{color}: bars are not five equally spaced bins")
    return [item[1] for item in measured]


def _score(heights: list[int]) -> float:
    total = sum(heights)
    if total <= 0:
        raise ValueError("histogram has no colored mass")
    ranked = sorted(heights, reverse=True)
    if ranked[0] - ranked[1] < 6:
        raise ValueError("tallest-bin identity is raster-ambiguous")
    mode = heights.index(max(heights))
    mean = sum(index * value for index, value in enumerate(heights)) / total
    return float(mean - mode)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 300 or array.shape[1] < 500:
        raise ValueError("image is too small for stable histogram measurement")
    histograms = {color: _measure_histogram(array, color) for color in _COLOR_RGB}
    totals = [sum(values) for values in histograms.values()]
    if max(totals) - min(totals) > 4:
        raise ValueError("the four histograms do not have the declared equal total height")
    scores = {color: _score(values) for color, values in histograms.items()}
    ranked = sorted(scores, key=lambda color: scores[color], reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 0.10:
        raise ValueError("moment-to-mode winner is raster-ambiguous")
    return ranked[0]
