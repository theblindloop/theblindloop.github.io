"""Deterministic renderer and analytic arm for histogram moment-to-mode gaps."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 800, 600
PAPER = "#FAFBFD"
PANEL = "#FFFFFF"
INK = "#263746"
GRID = "#D9E0E8"
DECORATION = "#AAB4C3"
COLORS = ("red", "blue", "green", "orange")
COLOR_HEX = {
    "red": "#D8424E",
    "blue": "#2878B5",
    "green": "#2A9D68",
    "orange": "#E1842C",
}
PANEL_SLOTS = ((38, 78), (402, 78), (38, 318), (402, 318))
PANEL_WIDTH, PANEL_HEIGHT = 330, 210
BAR_WIDTH, BAR_GAP = 38, 13
BAR_TOTAL = 400
QUARANTINE_MARGIN = 0.18


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)


def _random_heights(rng: random.Random) -> list[int]:
    """Produce five positive integer heights summing to a fixed visible mass."""
    base = 30
    remainder = BAR_TOTAL - 5 * base
    weights = [rng.uniform(0.18, 2.70) for _ in range(5)]
    weight_sum = sum(weights)
    raw = [remainder * value / weight_sum for value in weights]
    extras = [int(math.floor(value)) for value in raw]
    # Allocate residual pixels by fractional remainder, avoiding positional bias.
    order = sorted(range(5), key=lambda index: raw[index] - extras[index], reverse=True)
    for index in order[: remainder - sum(extras)]:
        extras[index] += 1
    return [base + value for value in extras]


def _score(heights: list[int]) -> float:
    if len(heights) != 5 or any(int(value) <= 0 for value in heights):
        raise ValueError("each histogram requires five positive bar heights")
    total = float(sum(int(value) for value in heights))
    maximum = max(int(value) for value in heights)
    modes = [index for index, value in enumerate(heights) if int(value) == maximum]
    if len(modes) != 1:
        raise ValueError("the tallest bar must be unique")
    mean_position = sum(index * int(value) for index, value in enumerate(heights)) / total
    return float(mean_position - modes[0])


def _records(scene: dict) -> dict[str, dict]:
    records = scene.get("histograms", [])
    if len(records) != 4:
        raise ValueError("exactly four histogram records are required")
    by_color = {str(record["color"]): record for record in records}
    if set(by_color) != set(COLORS):
        raise ValueError("histograms must be bound once to each declared color")
    for record in records:
        heights = [int(value) for value in record["heights"]]
        if len(heights) != 5 or sum(heights) != BAR_TOTAL:
            raise ValueError("each histogram must have five bars totaling 400 pixels")
        _score(heights)
    return by_color


def _scores(scene: dict) -> dict[str, float]:
    return {
        color: _score([int(value) for value in record["heights"]])
        for color, record in _records(scene).items()
    }


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed ^ 0x6D0E17)
    target = COLORS[seed % len(COLORS)]
    for _ in range(20_000):
        records = [
            {"color": color, "heights": _random_heights(rng)}
            for color in COLORS
        ]
        # Tall bars remain within a common panel scale and have a raster-safe mode.
        if any(max(record["heights"]) > 150 for record in records):
            continue
        if any(
            sorted(record["heights"], reverse=True)[0]
            - sorted(record["heights"], reverse=True)[1]
            < 10
            for record in records
        ):
            continue
        scores = {record["color"]: _score(record["heights"]) for record in records}
        ranked = sorted(COLORS, key=lambda color: scores[color], reverse=True)
        if ranked[0] != target or scores[ranked[0]] - scores[ranked[1]] < 0.22:
            continue

        # Target color and target panel occupy every pairing over sixteen seeds.
        target_slot = (seed // len(COLORS)) % len(PANEL_SLOTS)
        others = [color for color in COLORS if color != target]
        rng.shuffle(others)
        panel_order: list[str | None] = [None] * 4
        panel_order[target_slot] = target
        for slot, color in zip((i for i, value in enumerate(panel_order) if value is None), others):
            panel_order[slot] = color
        decorations = [
            {"x": rng.randint(18, WIDTH - 18), "y": rng.randint(58, 66)}
            for _ in range(4)
        ]
        return {
            "histograms": records,
            "panel_order": [str(value) for value in panel_order],
            "offset_x": rng.randint(-5, 5),
            "offset_y": rng.randint(-3, 3),
            "decorations": decorations,
        }
    raise RuntimeError("could not sample a separated moment-to-mode scene")


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return max(COLORS, key=lambda color: scores[color])


def margin(scene: dict) -> float:
    ranked = sorted(_scores(scene).values(), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    records = _records(scene)
    mode_clearance = min(
        sorted((int(value) for value in record["heights"]), reverse=True)[0]
        - sorted((int(value) for value in record["heights"]), reverse=True)[1]
        for record in records.values()
    )
    shortest = min(int(value) for record in records.values() for value in record["heights"])
    return margin(scene) < QUARANTINE_MARGIN or mode_clearance < 6 or shortest < 24


def render(scene: dict) -> Image.Image:
    records = _records(scene)
    order = [str(value) for value in scene["panel_order"]]
    if len(order) != 4 or set(order) != set(COLORS):
        raise ValueError("panel_order must be a permutation of the four colors")
    dx, dy = int(scene.get("offset_x", 0)), int(scene.get("offset_y", 0))
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((38 + dx, 18 + dy), "FIVE-BIN DISTRIBUTIONS", font=_font(23, True), fill=INK)
    draw.text((395 + dx, 24 + dy), "equal width  •  equal total height", font=_font(15), fill=INK)

    for item in scene.get("decorations", []):
        x, y = int(item["x"]) + dx, int(item["y"]) + dy
        draw.ellipse((x - 3, y - 3, x + 3, y + 3), fill=DECORATION)

    for slot, color in enumerate(order):
        sx, sy = PANEL_SLOTS[slot]
        sx, sy = sx + dx, sy + dy
        draw.rounded_rectangle(
            (sx, sy, sx + PANEL_WIDTH, sy + PANEL_HEIGHT),
            radius=9,
            fill=PANEL,
            outline=GRID,
            width=2,
        )
        baseline = sy + 170
        left = sx + 43
        right = left + 5 * BAR_WIDTH + 4 * BAR_GAP
        for height in (50, 100, 150):
            y = baseline - height
            draw.line((left - 9, y, right + 8, y), fill=GRID, width=1)
        draw.line((left - 9, baseline, right + 8, baseline), fill=INK, width=2)
        for index in range(6):
            x = left - BAR_GAP // 2 + index * (BAR_WIDTH + BAR_GAP)
            draw.line((x, baseline, x, baseline + 5), fill=INK, width=1)
        heights = [int(value) for value in records[color]["heights"]]
        for index, height in enumerate(heights):
            x = left + index * (BAR_WIDTH + BAR_GAP)
            draw.rectangle(
                (x, baseline - height, x + BAR_WIDTH - 1, baseline - 1),
                fill=COLOR_HEX[color],
            )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["histograms"] = list(reversed(twin["histograms"]))
    return [("reverse_color_record_storage", twin)]
