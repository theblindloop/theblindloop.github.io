"""Two-panel polyomino correspondence world.

Top panel: four four-cell block shapes, each painted a distinct color.
Bottom panel: the same four shapes in gray, re-ordered along the row and each
independently turned by a multiple of a quarter turn (never mirrored). One
bottom shape is enclosed in a black frame. The gold answer is the color that
the framed shape carries in the top panel.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 640
HEIGHT = 400
BACKGROUND = (255, 255, 255)
DIVIDER_Y = 190
DIVIDER_COLOR = (90, 90, 90)
GRAY = (125, 125, 125)
FRAME = (0, 0, 0)

PITCH = 26
CELL = 22
SLOT_PITCH = 150
SLOT_LEFT = 20
TOP_Y = 40
BOTTOM_Y = 230
JITTER = 6

COLOR_NAMES = ("red", "blue", "green", "orange")
COLOR_RGB = {
    "red": (214, 40, 40),
    "blue": (40, 70, 210),
    "green": (25, 145, 60),
    "orange": (238, 150, 20),
}

SHAPES = {
    "I": ((0, 0), (1, 0), (2, 0), (3, 0)),
    "O": ((0, 0), (1, 0), (0, 1), (1, 1)),
    "T": ((0, 0), (1, 0), (2, 0), (1, 1)),
    "S": ((1, 0), (2, 0), (0, 1), (1, 1)),
    "Z": ((0, 0), (1, 0), (1, 1), (2, 1)),
    "J": ((0, 0), (0, 1), (1, 1), (2, 1)),
    "L": ((2, 0), (0, 1), (1, 1), (2, 1)),
}
SHAPE_KEYS = tuple(sorted(SHAPES))
TWIN_PAIRS = (("S", "Z"), ("J", "L"))


def _normalize(cells) -> tuple:
    xs = min(c[0] for c in cells)
    ys = min(c[1] for c in cells)
    return tuple(sorted((c[0] - xs, c[1] - ys) for c in cells))


def _rotate(cells, quarter: int) -> tuple:
    out = list(cells)
    for _ in range(quarter % 4):
        out = [(y, -x) for (x, y) in out]
    return _normalize(out)


def _canonical(cells) -> tuple:
    return min(_rotate(cells, q) for q in range(4))


def _cell_distance(a, b) -> int:
    """Smallest symmetric-difference size between shape a and any turn of b."""
    best = 8
    sa = set(_normalize(a))
    for q in range(4):
        sb = set(_rotate(b, q))
        best = min(best, len(sa ^ sb))
    return best


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    want_minimum = rng.random() < 0.5
    keys, perm, rots, query_slot = None, None, None, None
    for attempt in range(240):
        keys = rng.sample(list(SHAPE_KEYS), 4)
        perm = list(range(4))
        rng.shuffle(perm)
        rots = [rng.randrange(4) for _ in range(4)]
        query_slot = rng.randrange(4)
        holder = perm[query_slot]
        gap = min(
            _cell_distance(SHAPES[keys[holder]], SHAPES[keys[k]])
            for k in range(4)
            if k != holder
        )
        if (gap <= 2) == want_minimum:
            break
    near_twin = want_minimum

    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    target = COLOR_NAMES[seed % 4]
    holder = perm[query_slot]
    other = colors.index(target)
    colors[other], colors[holder] = colors[holder], colors[other]

    jitter = [
        [rng.randint(-JITTER, JITTER), rng.randint(-JITTER, JITTER)] for _ in range(8)
    ]
    return {
        "shape_keys": keys,
        "colors": colors,
        "perm": perm,
        "rots": rots,
        "query_slot": query_slot,
        "jitter": jitter,
        "near_twin": near_twin,
    }


def _placed_cells(cells, slot: int, row_y: int, jx: int, jy: int):
    cells = _normalize(cells)
    width = (max(c[0] for c in cells) + 1) * PITCH
    height = (max(c[1] for c in cells) + 1) * PITCH
    x0 = SLOT_LEFT + slot * SLOT_PITCH + (SLOT_PITCH - width) // 2 + jx
    y0 = row_y + (4 * PITCH - height) // 2 + jy
    return [(x0 + c[0] * PITCH, y0 + c[1] * PITCH) for c in cells]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle([0, DIVIDER_Y, WIDTH - 1, DIVIDER_Y + 2], fill=DIVIDER_COLOR)

    for slot in range(4):
        key = scene["shape_keys"][slot]
        jx, jy = scene["jitter"][slot]
        rgb = COLOR_RGB[scene["colors"][slot]]
        for px, py in _placed_cells(SHAPES[key], slot, TOP_Y, jx, jy):
            draw.rectangle([px, py, px + CELL - 1, py + CELL - 1], fill=rgb)

    for slot in range(4):
        key = scene["shape_keys"][scene["perm"][slot]]
        cells = _rotate(SHAPES[key], scene["rots"][slot])
        jx, jy = scene["jitter"][4 + slot]
        placed = _placed_cells(cells, slot, BOTTOM_Y, jx, jy)
        for px, py in placed:
            draw.rectangle([px, py, px + CELL - 1, py + CELL - 1], fill=GRAY)
        if slot == scene["query_slot"]:
            x0 = min(p[0] for p in placed) - 11
            y0 = min(p[1] for p in placed) - 11
            x1 = max(p[0] for p in placed) + CELL + 10
            y1 = max(p[1] for p in placed) + CELL + 10
            draw.rectangle([x0, y0, x1, y1], outline=FRAME, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["colors"][scene["perm"][scene["query_slot"]]]


def margin(scene: dict) -> float:
    """Mismatching ink area, in pixels, for the closest wrong top shape."""
    holder = scene["perm"][scene["query_slot"]]
    marked = SHAPES[scene["shape_keys"][holder]]
    others = [
        _cell_distance(marked, SHAPES[scene["shape_keys"][k]])
        for k in range(4)
        if k != holder
    ]
    return float(min(others) * CELL * CELL)


def is_quarantined(scene: dict) -> bool:
    """Below one moved cell of mismatch the runner-up is not separable."""
    return margin(scene) < 1.5 * CELL * CELL


def latent_symmetries(scene: dict) -> list:
    return []
