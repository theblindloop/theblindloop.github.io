"""Deterministic snowman renderer and latent analytic arm."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

CANVAS = 512
SCALE = 3
NOSE_RGB = {
    "orange": (232, 105, 24),
    "blue": (35, 119, 224),
    "purple": (158, 76, 181),
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 1_000_003 + 71)
    color = tuple(NOSE_RGB)[seed % len(NOSE_RGB)]
    face_radius = rng.randint(66, 76)
    # Several absolute positions, each kept inside a fine ablation cell so the
    # compact cue is not artificially split by the measurement grid.
    cx = rng.choice((160, 224, 288, 352)) + rng.randint(-2, 2)
    cy = rng.choice((160, 224)) + rng.randint(-2, 2)
    nose_size = rng.randint(30, 34)
    direction = "right" if rng.randrange(2) == 0 else "left"
    snowflakes = []
    for _ in range(7):
        for _attempt in range(100):
            x, y = rng.randint(28, 484), rng.randint(28, 484)
            if (x - cx) ** 2 + (y - (cy + 80)) ** 2 > (face_radius + 105) ** 2:
                snowflakes.append([x, y, rng.randint(5, 9)])
                break
    return {
        "nose_color": color,
        "face_center": [cx, cy],
        "face_radius": face_radius,
        "nose_size": nose_size,
        "nose_direction": direction,
        "snowflakes": snowflakes,
        "background": [250, 252, 254],
        "width": CANVAS,
        "height": CANVAS,
    }


def _scaled_point(point: tuple[float, float]) -> tuple[int, int]:
    return (round(point[0] * SCALE), round(point[1] * SCALE))


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)

    def ellipse(box, *, fill, outline=None, width=1):
        draw.ellipse(tuple(round(v * SCALE) for v in box), fill=fill, outline=outline, width=width * SCALE)

    def line(points, *, fill, width=1):
        draw.line([_scaled_point(tuple(p)) for p in points], fill=fill, width=width * SCALE)

    cx, cy = map(float, scene["face_center"])
    radius = float(scene["face_radius"])
    body_y = cy + radius + 72
    neutral_fill = (239, 242, 245)
    outline = (62, 67, 73)
    # Neutral context: body, head, twig arms, face, buttons and snowflakes.
    ellipse((cx - radius * 0.88, body_y - radius * 0.70, cx + radius * 0.88, body_y + radius * 1.35), fill=neutral_fill, outline=outline, width=4)
    line(((cx - radius * 0.72, body_y), (cx - radius * 1.40, body_y - 28)), fill=outline, width=5)
    line(((cx + radius * 0.72, body_y), (cx + radius * 1.40, body_y - 24)), fill=outline, width=5)
    ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=neutral_fill, outline=outline, width=4)
    for dx in (-radius * 0.34, radius * 0.34):
        ellipse((cx + dx - 7, cy - 25 - 7, cx + dx + 7, cy - 25 + 7), fill=(42, 46, 50))
    for dx, dy in ((-28, 34), (-14, 44), (0, 47), (14, 44), (28, 34)):
        ellipse((cx + dx - 3.5, cy + dy - 3.5, cx + dx + 3.5, cy + dy + 3.5), fill=(75, 80, 85))
    for offset in (30, 78):
        ellipse((cx - 6, body_y + offset - 6, cx + 6, body_y + offset + 6), fill=(68, 73, 78))

    # The only chromatic component is the connected triangular nose.
    size = float(scene["nose_size"])
    sign = 1 if scene["nose_direction"] == "right" else -1
    base_x = cx - sign * size * 0.36
    nose = [
        _scaled_point((base_x, cy - size * 0.48)),
        _scaled_point((base_x, cy + size * 0.48)),
        _scaled_point((cx + sign * size * 1.12, cy + size * 0.06)),
    ]
    draw.polygon(nose, fill=NOSE_RGB[scene["nose_color"]])

    for x, y, snow_size in scene["snowflakes"]:
        s = float(snow_size)
        line(((x - s, y), (x + s, y)), fill=(176, 184, 192), width=2)
        line(((x, y - s), (x, y + s)), fill=(176, 184, 192), width=2)
        line(((x - s * 0.7, y - s * 0.7), (x + s * 0.7, y + s * 0.7)), fill=(176, 184, 192), width=2)
        line(((x - s * 0.7, y + s * 0.7), (x + s * 0.7, y - s * 0.7)), fill=(176, 184, 192), width=2)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["nose_color"])


def margin(scene: dict) -> float:
    # Approximate triangle area above the independent oracle's area floor.
    return round(0.72 * float(scene["nose_size"]) ** 2 - 120.0, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    if len(scene["snowflakes"]) < 2:
        return []
    twin = copy.deepcopy(scene)
    twin["snowflakes"][0], twin["snowflakes"][1] = twin["snowflakes"][1], twin["snowflakes"][0]
    return [("permute_neutral_snowflake_draw_order", twin)]
