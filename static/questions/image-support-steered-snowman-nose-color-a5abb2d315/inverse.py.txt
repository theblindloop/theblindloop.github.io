"""Independent final-PNG inverse arm for the chromatic nose component."""

from __future__ import annotations

import numpy as np
from PIL import Image

# Public answer palette is duplicated intentionally; no renderer state is read.
_COLORS = {
    "orange": np.array((232, 105, 24), dtype=np.int32),
    "blue": np.array((35, 119, 224), dtype=np.int32),
    "purple": np.array((158, 76, 181), dtype=np.int32),
}
_MIN_AREA = 120
_MAX_RUNNER_AREA = 45


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    counts: dict[str, int] = {}
    masks: dict[str, np.ndarray] = {}
    for name, color in _COLORS.items():
        distance = np.sqrt(np.sum((arr - color) ** 2, axis=2))
        mask = distance < 58.0
        masks[name] = mask
        counts[name] = int(mask.sum())
    ranked = sorted(counts, key=counts.get, reverse=True)
    winner, runner = ranked[0], ranked[1]
    if counts[winner] < _MIN_AREA or counts[runner] > _MAX_RUNNER_AREA:
        raise ValueError("missing or color-ambiguous triangular nose")
    ys, xs = np.nonzero(masks[winner])
    if not len(xs):
        raise ValueError("empty nose mask")
    box_area = (int(xs.max()) - int(xs.min()) + 1) * (int(ys.max()) - int(ys.min()) + 1)
    fill = counts[winner] / box_area
    if box_area > 2500 or not 0.18 <= fill <= 0.85:
        raise ValueError("chromatic component is not one compact triangle")
    return winner
