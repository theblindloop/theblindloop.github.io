"""Renderer and analytic arm for ordered extrema along a route."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 247, 243)
CYAN = (18, 164, 180)
CORAL = (225, 83, 68)
BLUE = (48, 101, 196)
INK = (53, 61, 68)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, non-self-crossing, horizontally backtracking route."""
    rng = random.Random(int(seed))
    # The base route's top extreme precedes its bottom extreme. Alternating
    # reversal makes every consecutive seed pair class-balanced.
    base = [
        (10, 95),
        (110, 62),
        (70, 10),
        (210, 48),
        (260, 110),
        (140, 160),
        (330, 125),
    ]
    points = []
    for index, (x, y) in enumerate(base):
        # Preserve generous extremum gaps while varying every vertex.
        jx = rng.randint(-8, 8)
        jy = rng.randint(-6, 6)
        if index == 2:
            jy = rng.randint(-3, 3)
        elif index == 5:
            jy = rng.randint(-3, 3)
        points.append((x + jx, y + jy))

    # Horizontal reflection is independent of the answer and breaks any
    # fixed start-side shortcut. Translation keeps the route regional.
    if rng.random() < 0.5:
        points = [(340 - x, y) for x, y in points]
    dx = rng.randint(82, 94)
    dy = rng.randint(148, 174)
    points = [(x + dx, y + dy) for x, y in points]
    if int(seed) % 2:
        points.reverse()
    return {"points": [[int(x), int(y)] for x, y in points]}


def _polyline_lengths(points: list[list[int]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for left, right in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(left, right))
    return cumulative, cumulative[-1]


def analytic_gold(scene: dict) -> str:
    points = scene["points"]
    top_index = min(range(len(points)), key=lambda i: points[i][1])
    bottom_index = max(range(len(points)), key=lambda i: points[i][1])
    return "before" if top_index < bottom_index else "after"


def margin(scene: dict) -> float:
    points = scene["points"]
    cumulative, _ = _polyline_lengths(points)
    top_index = min(range(len(points)), key=lambda i: points[i][1])
    bottom_index = max(range(len(points)), key=lambda i: points[i][1])
    return round(abs(cumulative[top_index] - cumulative[bottom_index]), 6)


def is_quarantined(scene: dict) -> bool:
    points = scene["points"]
    ys = sorted(point[1] for point in points)
    unique_top = ys[1] - ys[0]
    unique_bottom = ys[-1] - ys[-2]
    return unique_top < 24 or unique_bottom < 24 or margin(scene) < 100


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    points = [tuple(point) for point in scene["points"]]
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Quiet measurement card gives the regional route a stable visual field.
    draw.rounded_rectangle((54, 60, 458, 452), radius=22, fill=(255, 255, 255), outline=(216, 218, 215), width=2)
    draw.text((72, 76), "ORDER ALONG THE ROUTE", fill=INK)
    draw.line(points, fill=CYAN, width=11, joint="curve")
    for x, y in points[1:-1]:
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=CYAN)

    start = points[0]
    finish = points[-1]
    # Endpoint markers visibly establish traversal direction and cover the
    # tube ends without touching any other segment.
    for (x, y), color in ((start, CORAL), (finish, BLUE)):
        draw.ellipse((x - 12, y - 12, x + 12, y + 12), fill=(255, 255, 255))
        draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)
    return image
