"""Independent pixel-only inverse for ordered extrema along a route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _mask_near(rgb: np.ndarray, target: tuple[int, int, int], tolerance: int = 16) -> np.ndarray:
    delta = np.abs(rgb.astype(np.int16) - np.asarray(target, dtype=np.int16))
    return np.max(delta, axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        groups.append(group)
    return groups


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError("missing endpoint marker")
    return float(xs.mean()), float(ys.mean())


def _distances(mask: np.ndarray, source: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    distances = np.full((height, width), -1, dtype=np.int32)
    sy, sx = source
    distances[sy, sx] = 0
    queue = deque([(sy, sx)])
    while queue:
        y, x = queue.popleft()
        next_distance = int(distances[y, x]) + 1
        for ny in range(max(0, y - 1), min(height, y + 2)):
            for nx in range(max(0, x - 1), min(width, x + 2)):
                if mask[ny, nx] and distances[ny, nx] < 0:
                    distances[ny, nx] = next_distance
                    queue.append((ny, nx))
    return distances


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    route = _mask_near(rgb, (18, 164, 180))
    coral = _mask_near(rgb, (225, 83, 68))
    blue = _mask_near(rgb, (48, 101, 196))
    if int(route.sum()) < 2500:
        return "abstain"
    groups = [group for group in _components(route) if len(group) >= 50]
    if len(groups) != 1 or len(groups[0]) < int(route.sum()) * 0.98:
        return "abstain"
    try:
        start_x, start_y = _centroid(coral)
        finish_x, finish_y = _centroid(blue)
    except ValueError:
        return "abstain"

    ys, xs = np.nonzero(route)
    start_index = int(np.argmin((xs - start_x) ** 2 + (ys - start_y) ** 2))
    finish_gap = float(np.min((xs - finish_x) ** 2 + (ys - finish_y) ** 2))
    if finish_gap > 500:
        return "abstain"
    source = (int(ys[start_index]), int(xs[start_index]))
    distances = _distances(route, source)
    min_y = int(ys.min())
    max_y = int(ys.max())
    top_values = distances[(route) & (np.indices(route.shape)[0] <= min_y + 2)]
    bottom_values = distances[(route) & (np.indices(route.shape)[0] >= max_y - 2)]
    top_values = top_values[top_values >= 0]
    bottom_values = bottom_values[bottom_values >= 0]
    if not len(top_values) or not len(bottom_values):
        return "abstain"
    top_distance = int(top_values.min())
    bottom_distance = int(bottom_values.min())
    if abs(top_distance - bottom_distance) < 25:
        return "abstain"
    return "before" if top_distance < bottom_distance else "after"
