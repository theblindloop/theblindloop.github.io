"""Independent image-only inverse arm for Cyclic Conveyor Replay."""

from __future__ import annotations

import numpy as np
from PIL import Image

MAIN = np.array((20, 100, 102), dtype=np.uint8)
CARD = np.array((102, 82, 160), dtype=np.uint8)
HILITE = np.array((54, 177, 190), dtype=np.uint8)
RING = np.array((212, 48, 116), dtype=np.uint8)
TIP = np.array((226, 67, 54), dtype=np.uint8)
POSITIONS = (
    "top-left", "top-middle", "top-right",
    "middle-left", "center", "middle-right",
    "bottom-left", "bottom-middle", "bottom-right",
)


def _mask(rgb: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.all(rgb == color, axis=2)


def _bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        raise ValueError("required visual evidence is absent")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _x_groups(xs: np.ndarray) -> list[tuple[int, int]]:
    values = sorted(set(int(x) for x in xs))
    if not values:
        return []
    groups: list[list[int]] = [[values[0]]]
    for value in values[1:]:
        if value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return [(group[0], group[-1]) for group in groups]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.ndim != 3 or rgb.shape[0] < 300 or rgb.shape[1] < 700:
        raise ValueError("unexpected raster")

    main_box = _bbox(_mask(rgb, MAIN))
    x0, y0, x1, y1 = main_box
    if x1 - x0 < 180 or y1 - y0 < 180:
        raise ValueError("initial board is malformed")
    ring = _mask(rgb, RING)
    ry, rx = np.nonzero(ring)
    if len(rx) < 100:
        raise ValueError("ring is missing")
    cx, cy = float(rx.mean()), float(ry.mean())
    col = min(2, max(0, int(3 * (cx - x0) / max(1, x1 - x0))))
    row = min(2, max(0, int(3 * (cy - y0) / max(1, y1 - y0))))

    card_mask = _mask(rgb, CARD)
    card_ys, card_xs = np.nonzero(card_mask)
    groups = [(a, b) for a, b in _x_groups(card_xs) if b - a >= 100]
    if len(groups) != 5:
        raise ValueError("card order is ambiguous")

    operations = []
    for left, right in groups:
        local_card = card_mask[:, left:right + 1]
        bx0, by0, bx1, by1 = _bbox(local_card)
        card_top, card_bottom = by0, by1
        crop_hi = _mask(rgb[card_top:card_bottom + 1, left:right + 1], HILITE)
        hx0, hy0, hx1, hy1 = _bbox(crop_hi)
        hwidth, hheight = hx1 - hx0 + 1, hy1 - hy0 + 1
        if hwidth > 2 * hheight:
            axis = "row"
            line = min(2, max(0, int((((hy0 + hy1) / 2) - 57) / 30)))
        elif hheight > 2 * hwidth:
            axis = "col"
            line = min(2, max(0, int((((hx0 + hx1) / 2) - 15) / 30)))
        else:
            raise ValueError("highlighted line is ambiguous")

        crop_tip = _mask(rgb[card_top:card_bottom + 1, left:right + 1], TIP)
        ty, tx = np.nonzero(crop_tip)
        if len(tx) < 20:
            raise ValueError("arrow tip is missing")
        tip_x, tip_y = float(tx.mean()), float(ty.mean())
        local_cx = (right - left) / 2
        # The direction dot lies beyond the arrow tip. Horizontal image y grows
        # downward, so +1 means right or down, matching analytic replay.
        direction = (1 if tip_x > local_cx else -1) if axis == "row" else (1 if tip_y > 215 else -1)
        operations.append((axis, line, direction))

    for axis, line, direction in operations:
        if axis == "row" and row == line:
            col = (col + direction) % 3
        elif axis == "col" and col == line:
            row = (row + direction) % 3
    return POSITIONS[3 * row + col]
