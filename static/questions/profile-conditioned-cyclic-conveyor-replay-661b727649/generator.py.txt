"""Deterministic renderer and analytic arm for Cyclic Conveyor Replay."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1024, 560
BG = ((250, 252, 253), (255, 249, 239))
CARD_BG = ((242, 246, 249), (248, 242, 232))
INK = (35, 45, 58)
MAIN = (20, 100, 102)
CARD = (102, 82, 160)
GRID = (66, 75, 88)
HILITE = (54, 177, 190)
RING = (212, 48, 116)
TIP = (226, 67, 54)


def _inverse_start(target_row: int, target_col: int, operations: list[dict]) -> tuple[int, int]:
    row, col = target_row, target_col
    for op in reversed(operations):
        if op["axis"] == "row" and row == op["line"]:
            col = (col - op["direction"]) % 3
        elif op["axis"] == "col" and col == op["line"]:
            row = (row - op["direction"]) % 3
    return row, col


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0xC0A7E5)
    target = (int(seed) * 7 + 3) % 9
    row, col = divmod(target, 3)
    axes = ["row", "col", "row", "col", "row"]
    rng.shuffle(axes)
    directions = [rng.choice((-1, 1)) for _ in range(5)]
    incident = [True, True, True, False, False]
    rng.shuffle(incident)
    operations: list[dict | None] = [None] * 5
    # Build backward from a balanced target, guaranteeing three relevant and
    # two visible distractor operations.
    for index in reversed(range(5)):
        axis, direction = axes[index], directions[index]
        coordinate = row if axis == "row" else col
        if incident[index]:
            line = coordinate
            if axis == "row":
                col = (col - direction) % 3
            else:
                row = (row - direction) % 3
        else:
            line = rng.choice([value for value in range(3) if value != coordinate])
        operations[index] = {"axis": axis, "line": line, "direction": direction}
    return {
        "start_row": row,
        "start_col": col,
        "operations": operations,
        "palette": (int(seed) // 9) % 2,
    }


def _replay(scene: dict) -> tuple[int, int]:
    row = int(scene["start_row"])
    col = int(scene["start_col"])
    for op in scene["operations"]:
        axis = op["axis"]
        line = int(op["line"])
        direction = int(op["direction"])
        if axis == "row" and row == line:
            col = (col + direction) % 3
        elif axis == "col" and col == line:
            row = (row + direction) % 3
    return row, col


POSITIONS = (
    "top-left", "top-middle", "top-right",
    "middle-left", "center", "middle-right",
    "bottom-left", "bottom-middle", "bottom-right",
)


def analytic_gold(scene: dict) -> str:
    row, col = _replay(scene)
    return POSITIONS[3 * row + col]


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    return 12.0


def is_quarantined(scene: dict) -> bool:
    try:
        if int(scene["start_row"]) not in range(3) or int(scene["start_col"]) not in range(3):
            return True
        if int(scene["palette"]) not in (0, 1) or len(scene["operations"]) != 5:
            return True
        for op in scene["operations"]:
            if op["axis"] not in ("row", "col"):
                return True
            if int(op["line"]) not in range(3) or int(op["direction"]) not in (-1, 1):
                return True
    except (KeyError, TypeError, ValueError):
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, axis: str, direction: int) -> None:
    if axis == "row":
        x0, x1 = cx - 28, cx + 28
        draw.line((x0, cy, x1, cy), fill=INK, width=5)
        tip = x1 if direction == 1 else x0
        base = tip - 10 * direction
        draw.polygon([(tip, cy), (base, cy - 9), (base, cy + 9)], fill=INK)
        dot_x = tip + 8 * direction
        draw.ellipse((dot_x - 4, cy - 4, dot_x + 4, cy + 4), fill=TIP)
    else:
        y0, y1 = cy - 25, cy + 25
        draw.line((cx, y0, cx, y1), fill=INK, width=5)
        tip = y1 if direction == 1 else y0
        base = tip - 10 * direction
        draw.polygon([(cx, tip), (cx - 9, base), (cx + 9, base)], fill=INK)
        dot_y = tip + 8 * direction
        draw.ellipse((cx - 4, dot_y - 4, cx + 4, dot_y + 4), fill=TIP)


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("malformed scene")
    palette = int(scene["palette"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BG[palette])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((40, 32), "CYCLIC CONVEYOR", fill=INK, font=font)
    draw.text((40, 53), "START", fill=MAIN, font=font)
    draw.text((328, 53), "READ CARDS LEFT TO RIGHT", fill=CARD, font=font)

    # Large initial board; its distinctive outline lets the inverse arm infer
    # cell geometry instead of receiving any coordinates or scene fields.
    gx, gy, size = 40, 145, 240
    draw.rectangle((gx, gy, gx + size, gy + size), fill=CARD_BG[palette], outline=MAIN, width=5)
    for k in (1, 2):
        draw.line((gx + 80 * k, gy, gx + 80 * k, gy + size), fill=GRID, width=3)
        draw.line((gx, gy + 80 * k, gx + size, gy + 80 * k), fill=GRID, width=3)
    mr = int(scene["start_row"])
    mc = int(scene["start_col"])
    mx, my = gx + 40 + 80 * mc, gy + 40 + 80 * mr
    draw.ellipse((mx - 22, my - 22, mx + 22, my + 22), outline=RING, width=8)
    draw.ellipse((mx - 7, my - 7, mx + 7, my + 7), fill=RING)

    for index, op in enumerate(scene["operations"]):
        x0, y0, cw, ch = 325 + 137 * index, 112, 120, 326
        draw.rounded_rectangle((x0, y0, x0 + cw, y0 + ch), radius=8, fill=CARD_BG[palette], outline=CARD, width=4)
        draw.text((x0 + 56, y0 + 16), str(index + 1), fill=CARD, font=font)
        mgx, mgy, ms = x0 + 15, y0 + 57, 90
        line = int(op["line"])
        if op["axis"] == "row":
            draw.rectangle((mgx + 2, mgy + 30 * line + 3, mgx + ms - 2, mgy + 30 * (line + 1) - 3), fill=HILITE)
        else:
            draw.rectangle((mgx + 30 * line + 3, mgy + 2, mgx + 30 * (line + 1) - 3, mgy + ms - 2), fill=HILITE)
        draw.rectangle((mgx, mgy, mgx + ms, mgy + ms), outline=GRID, width=3)
        for k in (1, 2):
            draw.line((mgx + 30 * k, mgy, mgx + 30 * k, mgy + ms), fill=GRID, width=2)
            draw.line((mgx, mgy + 30 * k, mgx + ms, mgy + 30 * k), fill=GRID, width=2)
        _arrow(draw, x0 + 60, y0 + 215, op["axis"], int(op["direction"]))
        draw.text((x0 + 34, y0 + 278), "WRAP", fill=INK, font=font)

    draw.text((40, 472), "Each highlighted line shifts one cell; items wrap across its edge.", fill=INK, font=font)
    return image
