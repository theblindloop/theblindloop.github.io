"""Renderer for a primal/dual connectivity intervention."""
from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 512
PALETTES = [(31, 143, 132), (44, 127, 164)]
YELLOW = (246, 190, 40)
WHITE = (250, 250, 247)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "role": "land splits" if seed % 2 == 0 else "courtyard opens",
        "variant": rng.randrange(4),
        "mirror": rng.randrange(2),
        "palette": rng.randrange(2),
    }


def _mx(x: int, mirror: int) -> int:
    return SIZE - 1 - x if mirror else x


def _rect(draw: ImageDraw.ImageDraw, box, fill, mirror: int) -> None:
    x0, y0, x1, y1 = box
    if mirror:
        x0, x1 = _mx(x1, 1), _mx(x0, 1)
    draw.rectangle((x0, y0, x1, y1), fill=fill)


def render(scene: dict) -> Image.Image:
    role = scene["role"]
    variant = int(scene["variant"])
    mirror = int(scene["mirror"])
    teal = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (SIZE, SIZE), WHITE)
    draw = ImageDraw.Draw(image)
    # The same 81x81 patch occupies the center in both dual constructions.
    patch = (216, 216, 296, 296)
    if role == "land splits":
        # Two broad lobes whose only edge connection is the yellow patch.
        _rect(draw, (28, 42, 215, 469), teal, mirror)
        _rect(draw, (297, 42, 483, 469), teal, mirror)
        # Balanced enclosed pockets are irrelevant but prevent a no-hole cue.
        dy = (variant - 1) * 4
        _rect(draw, (72, 105 + dy, 155, 182 + dy), WHITE, mirror)
        _rect(draw, (357, 330 - dy, 440, 407 - dy), WHITE, mirror)
    elif role == "courtyard opens":
        # One broad sheet; white corridor and cavity are separated by the patch.
        _rect(draw, (28, 42, 483, 469), teal, mirror)
        dx = (variant - 1) * 4
        _rect(draw, (216 + dx, 42, 296 + dx, 215), WHITE, mirror)
        _rect(draw, (164, 297, 348, 407), WHITE, mirror)
        # A second robust enclosed pocket balances hole count and distant ink.
        _rect(draw, (67, 325, 145, 402), WHITE, mirror)
    else:
        raise ValueError("unknown role")
    _rect(draw, patch, YELLOW, mirror)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["role"]


def margin(scene: dict) -> float:
    # Minimum decisive neck/gate thickness in pixels.
    _ = (scene["variant"], scene["mirror"], scene["palette"])
    return 81.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["role"], scene["variant"], scene["mirror"], scene["palette"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["palette"] = int(scene["palette"]) + len(PALETTES)
    return [("palette_period_alias", twin)]
