"""Independent pixel-only inverse arm."""
from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray):
    """Run-length connected components; fast even under many grid ablations."""
    h, w = mask.shape
    parent=[]; size=[]; border=[]; previous=[]
    def root(i):
        while parent[i] != i:
            parent[i] = parent[parent[i]]; i = parent[i]
        return i
    def join(a,b):
        a,b=root(a),root(b)
        if a==b: return a
        if size[a] < size[b]: a,b=b,a
        parent[b]=a; size[a]+=size[b]; border[a]=border[a] or border[b]
        return a
    for y in range(h):
        row=mask[y]
        edges=np.flatnonzero(np.diff(np.r_[False,row,False].astype(np.int8)))
        current=[]
        for x0,x1 in edges.reshape(-1,2):
            i=len(parent); parent.append(i); size.append(int(x1-x0))
            border.append(y in (0,h-1) or x0==0 or x1==w)
            current.append((int(x0),int(x1),i))
        p=0
        for x0,x1,i in current:
            while p < len(previous) and previous[p][1] <= x0: p += 1
            q=p
            while q < len(previous) and previous[q][0] < x1:
                join(i,previous[q][2]); q += 1
        previous=current
    return [(size[i],border[i]) for i in range(len(parent)) if root(i)==i]


def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"), dtype=np.int16)
    if a.shape != (512, 512, 3):
        raise ValueError("unexpected canvas")
    # Saturated warm pixels are the patch; saturated cool pixels are land.
    yellow = (a[:,:,0] > 190) & (a[:,:,1] > 125) & (a[:,:,2] < 100)
    teal = (a[:,:,1] > 90) & (a[:,:,1] > a[:,:,0] + 25) & (a[:,:,2] > 90)
    if not 6000 <= int(yellow.sum()) <= 7200:
        raise ValueError("yellow patch missing or damaged")
    union = teal | yellow
    # Legal scenes have nearly equal, high-fill foreground area in both roles.
    # This is an integrity check, not the decision: tolerate small raster noise,
    # but abstain after a substantial remote chunk of the visible sheet is erased.
    foreground_area = int(union.sum())
    if min(abs(foreground_area - 153957), abs(foreground_area - 154377)) > 96:
        raise ValueError("broad land sheet has been substantially erased")
    union_comps = [c for c in _components(union) if c[0] > 100]
    if len(union_comps) != 1 or union_comps[0][0] < 120000:
        raise ValueError("land sheet missing or damaged")
    teal_comps = [c for c in _components(teal) if c[0] > 100]
    white_before = ~union
    before = [c for c in _components(white_before) if c[0] > 100]
    white_after = ~teal
    after = [c for c in _components(white_after) if c[0] > 100]
    enclosed_before = sum(not touches_border for _,touches_border in before)
    enclosed_after = sum(not touches_border for _,touches_border in after)
    if enclosed_before != 2:
        raise ValueError("expected two robust enclosed white regions")
    if len(teal_comps) == 2 and enclosed_after == enclosed_before:
        return "land splits"
    if len(teal_comps) == 1 and enclosed_after == enclosed_before - 1 and any(b for _,b in after):
        return "courtyard opens"
    raise ValueError("image does not exhibit exactly one declared dual effect")
