"""Deterministic latent sampler and raster renderer for a traffic-signal prior conflict."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw


CANONICAL_ORDER = ("red", "amber", "green")
LAMP_ORDERS = tuple(itertools.permutations(CANONICAL_ORDER))
LAMP_RGB = {
    "red": (220, 45, 45),
    "amber": (244, 180, 24),
    "green": (38, 166, 91),
}
_BACKGROUNDS = (
    (242, 247, 250),
    (249, 244, 235),
    (238, 246, 239),
    (245, 241, 249),
)
_HOUSINGS = ((48, 55, 62), (57, 63, 70), (43, 52, 55), (62, 58, 53))
_BORDERS = ((22, 26, 30), (30, 33, 37), (24, 31, 30), (34, 31, 29))
_POLES = ((74, 80, 84), (82, 76, 70), (66, 78, 76), (77, 73, 82))
_SUPERSAMPLE = 4


def sample_scene(seed: int) -> dict:
    """Sample one scene; each consecutive six seeds form a nuisance-matched set."""
    seed = int(seed)
    group = seed // len(LAMP_ORDERS)
    order = list(LAMP_ORDERS[seed % len(LAMP_ORDERS)])
    rng = random.Random(0x5A17C0DE + group * 1_000_003)

    radius = rng.randint(29, 36)
    # Every fourth matched group is a declared near-boundary/quarantine set.
    gap = rng.randint(4, 7) if group % 4 == 0 else rng.randint(17, 30)
    housing_width = rng.randint(max(132, 2 * radius + 52), 168)
    pad_y = rng.randint(17, 23)
    housing_height = 2 * pad_y + 6 * radius + 2 * gap

    return {
        "width": 384,
        "height": 512,
        "background_rgb": list(rng.choice(_BACKGROUNDS)),
        "housing_rgb": list(rng.choice(_HOUSINGS)),
        "border_rgb": list(rng.choice(_BORDERS)),
        "pole_rgb": list(rng.choice(_POLES)),
        "center_x": rng.randint(174, 210),
        "top_y": rng.randint(28, 48),
        "housing_width": housing_width,
        "housing_height": housing_height,
        "lamp_radius": radius,
        "lamp_gap": gap,
        "lamp_order": order,
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)  # type: ignore[return-value]


def render(scene: dict) -> Image.Image:
    """Render a single familiar traffic-signal silhouette from public scene fields."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(scene["background_rgb"])
    housing = tuple(scene["housing_rgb"])
    border = tuple(scene["border_rgb"])
    pole = tuple(scene["pole_rgb"])
    center_x = int(scene["center_x"])
    top_y = int(scene["top_y"])
    housing_width = int(scene["housing_width"])
    housing_height = int(scene["housing_height"])
    radius = int(scene["lamp_radius"])
    gap = int(scene["lamp_gap"])
    order = list(scene["lamp_order"])

    scale = _SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    left = center_x - housing_width / 2
    right = center_x + housing_width / 2
    bottom = top_y + housing_height

    pole_width = max(20, int(housing_width * 0.18))
    pole_bottom = height - 36
    draw.rounded_rectangle(
        _scaled_box((center_x - pole_width / 2, bottom - 2, center_x + pole_width / 2, pole_bottom), scale),
        radius=5 * scale,
        fill=pole,
        outline=border,
        width=3 * scale,
    )
    base_width = int(housing_width * 0.68)
    draw.rounded_rectangle(
        _scaled_box((center_x - base_width / 2, pole_bottom - 9, center_x + base_width / 2, pole_bottom + 12), scale),
        radius=7 * scale,
        fill=pole,
        outline=border,
        width=3 * scale,
    )
    draw.rounded_rectangle(
        _scaled_box((left, top_y, right, bottom), scale),
        radius=18 * scale,
        fill=housing,
        outline=border,
        width=5 * scale,
    )

    pad_y = (housing_height - (6 * radius + 2 * gap)) / 2
    first_y = top_y + pad_y + radius
    for slot, color_name in enumerate(order):
        cy = first_y + slot * (2 * radius + gap)
        # A dark bezel and equal-sized colored interior preserve total ink across orders.
        draw.ellipse(
            _scaled_box((center_x - radius - 7, cy - radius - 7, center_x + radius + 7, cy + radius + 7), scale),
            fill=border,
        )
        draw.ellipse(
            _scaled_box((center_x - radius, cy - radius, center_x + radius, cy + radius), scale),
            fill=LAMP_RGB[color_name],
        )
        # Same-position highlight on every lamp: no answer-specific styling.
        highlight_r = max(4, radius // 6)
        hx = center_x - radius * 0.34
        hy = cy - radius * 0.34
        draw.ellipse(
            _scaled_box((hx - highlight_r, hy - highlight_r, hx + highlight_r, hy + highlight_r), scale),
            fill=(255, 255, 255),
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    order = list(scene["lamp_order"])
    return ("top", "middle", "bottom")[order.index("red")]


def margin(scene: dict) -> float:
    """Runner-up separation: adjacent lamp-centre gap in final-image pixels."""
    return float(2 * int(scene["lamp_radius"]) + int(scene["lamp_gap"]))


def is_quarantined(scene: dict) -> bool:
    return int(scene["lamp_gap"]) < 8 or int(scene["lamp_radius"]) < 16


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All nuisance fields deliberately affect pixels, so no latent alias is declared.
    return []
