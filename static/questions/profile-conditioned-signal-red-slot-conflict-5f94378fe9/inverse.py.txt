"""Independent pixel-only inverse for the red lamp's vertical slot."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _semantic_masks(image: Image.Image) -> dict[str, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (arr[:, :, 0] > 170) & (arr[:, :, 1] < 115) & (arr[:, :, 2] < 115)
    amber = (
        (arr[:, :, 0] > 175)
        & (arr[:, :, 1] > 105)
        & (arr[:, :, 1] < 225)
        & (arr[:, :, 2] < 95)
    )
    green = (
        (arr[:, :, 1] > 120)
        & (arr[:, :, 0] < 120)
        & (arr[:, :, 2] < 140)
        & (arr[:, :, 1] > arr[:, :, 0] + 45)
    )
    return {"red": red, "amber": amber, "green": green}


def _largest_component(mask: np.ndarray) -> np.ndarray:
    """Return (y,x) pixels of the largest four-connected component."""
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        raise ValueError("required lamp color is absent")
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    seen = np.zeros_like(sub, dtype=bool)
    largest: list[tuple[int, int]] = []
    for sy, sx in np.argwhere(sub):
        sy = int(sy)
        sx = int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        component: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            component.append((y + y0, x + x0))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < sub.shape[0] and 0 <= nx < sub.shape[1] and sub[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(component) > len(largest):
            largest = component
    coords = np.asarray(largest, dtype=np.float64)
    if len(coords) < 500:
        raise ValueError("lamp component is too small to identify safely")
    return coords


def decision_from_image(image: Image.Image) -> str:
    """Return top/middle/bottom using only semantic-color components in the PNG."""
    measurements: dict[str, tuple[float, float, int]] = {}
    for name, mask in _semantic_masks(image).items():
        component = _largest_component(mask)
        measurements[name] = (
            float(component[:, 0].mean()),
            float(component[:, 1].mean()),
            len(component),
        )

    centers_y = {name: value[0] for name, value in measurements.items()}
    centers_x = [value[1] for value in measurements.values()]
    areas = [value[2] for value in measurements.values()]
    ranked = sorted(centers_y, key=centers_y.get)
    gaps = [centers_y[ranked[i + 1]] - centers_y[ranked[i]] for i in range(2)]
    if min(gaps) < 12.0:
        raise ValueError("lamp ranks are visually ambiguous")
    if max(centers_x) - min(centers_x) > 12.0:
        raise ValueError("lamp components do not form one vertical signal")
    if max(areas) / min(areas) > 1.35:
        raise ValueError("lamp component sizes are inconsistent")
    return ("top", "middle", "bottom")[ranked.index("red")]
