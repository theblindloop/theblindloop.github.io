"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 800
BG = (248, 246, 239)
NAVY = (28, 47, 68)
CYAN = (46, 190, 196)
CREAM = (244, 222, 164)
GOLD = (239, 178, 55)
TAG_RGB = {
    "red": (215, 72, 73),
    "green": (69, 166, 104),
    "blue": (62, 112, 202),
}
PLAQUE = (250, 250, 550, 550)


def _majority(source: list[list[int]]) -> list[list[int]]:
    return [
        [int(sum(source[y + dy][x + dx] for dy in range(3) for dx in range(3)) >= 5)
         for x in range(3)]
        for y in range(3)
    ]


def _distance(left: list[list[int]], right: list[list[int]]) -> int:
    return sum(a != b for ra, rb in zip(left, right) for a, b in zip(ra, rb))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        source = [[rng.randrange(2) for _ in range(5)] for _ in range(5)]
        total = sum(map(sum, source))
        target = _majority(source)
        out_total = sum(map(sum, target))
        if 8 <= total <= 17 and 2 <= out_total <= 7:
            break

    match_position = int(seed) % 3
    candidates: list[list[list[int]]] = [None, None, None]  # type: ignore[list-item]
    candidates[match_position] = [row[:] for row in target]
    used = {tuple(map(tuple, target))}
    for position in range(3):
        if position == match_position:
            continue
        while True:
            altered = [row[:] for row in target]
            flips = rng.sample(range(9), 2 + ((position + int(seed)) & 1))
            for index in flips:
                altered[index // 3][index % 3] ^= 1
            key = tuple(map(tuple, altered))
            if key not in used:
                used.add(key)
                candidates[position] = altered
                break
    # Cyclic tag assignment counterbalances tag color against fixed position.
    colors = ["red", "green", "blue"]
    shift = (int(seed) // 3) % 3
    tag_colors = colors[shift:] + colors[:shift]
    return {"source": source, "candidates": candidates, "tag_colors": tag_colors}


def analytic_gold(scene: dict) -> str:
    target = _majority(scene["source"])
    distances = [_distance(candidate, target) for candidate in scene["candidates"]]
    winner = min(range(3), key=distances.__getitem__)
    return scene["tag_colors"][winner]


def margin(scene: dict) -> float:
    target = _majority(scene["source"])
    distances = sorted(_distance(candidate, target) for candidate in scene["candidates"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["source"]) != 5 or any(len(row) != 5 for row in scene["source"]):
            return True
        if len(scene["candidates"]) != 3 or len(scene["tag_colors"]) != 3:
            return True
        if sorted(scene["tag_colors"]) != ["blue", "green", "red"]:
            return True
        if any(len(card) != 3 or any(len(row) != 3 for row in card)
               for card in scene["candidates"]):
            return True
        target = _majority(scene["source"])
        distances = sorted(_distance(card, target) for card in scene["candidates"])
        return distances[0] != 0 or distances.count(0) != 1 or distances[1] < 2
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_grid(draw: ImageDraw.ImageDraw, matrix: list[list[int]], x: int, y: int,
               cell: int, gap: int = 4) -> None:
    for row_index, row in enumerate(matrix):
        for col_index, value in enumerate(row):
            left = x + col_index * (cell + gap)
            top = y + row_index * (cell + gap)
            draw.rectangle((left, top, left + cell - 1, top + cell - 1),
                           fill=CYAN if value else CREAM)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    # One thick, connected, medium-span seal carries all decisive evidence.
    draw.rounded_rectangle(PLAQUE, radius=22, fill=NAVY)

    source_x, source_y, source_cell, source_gap = 328, 266, 22, 4
    _draw_grid(draw, scene["source"], source_x, source_y, source_cell, source_gap)
    source_w = 5 * source_cell + 4 * source_gap
    draw.rectangle((source_x - 7, source_y - 7,
                    source_x + source_w + 6, source_y + source_w + 6),
                   outline=GOLD, width=4)

    candidate_y, candidate_cell, candidate_gap = 442, 18, 4
    candidate_xs = (272, 367, 462)
    for index, x in enumerate(candidate_xs):
        _draw_grid(draw, scene["candidates"][index], x, candidate_y,
                   candidate_cell, candidate_gap)
        color = TAG_RGB[scene["tag_colors"][index]]
        center_x = x + (3 * candidate_cell + 2 * candidate_gap) // 2
        draw.ellipse((center_x - 10, 516, center_x + 10, 536), fill=color)
    return image
