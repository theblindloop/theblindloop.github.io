"""Deterministic renderer and analytic label for a single-cycle network.

The visible graph has a bridge from green to the left side of a rectangle, a
blue-or-red leaf attached before that bridge, and the other colored leaf
attached after the rectangle's cycle.  The color assignment is randomized, so
the answer is never a fixed color.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "cycle_route_terminal-0.1.0"
WIDTH = 960
HEIGHT = 720
BASE_POINTS = {
    "g": (180.0, 360.0),
    "p": (320.0, 360.0),
    "c0": (440.0, 360.0),
    "c1": (660.0, 360.0),
    "c2": (660.0, 190.0),
    "c3": (440.0, 190.0),
    "b": (320.0, 550.0),
    "r": (780.0, 360.0),
}
EDGES = [
    ["g", "p"],
    ["p", "c0"],
    ["c0", "c1"],
    ["c1", "c2"],
    ["c2", "c3"],
    ["c3", "c0"],
    ["p", "b"],
    ["c1", "r"],
]
PALETTE = {
    "green": (42, 163, 95),
    "red": (211, 72, 72),
    "blue": (52, 99, 188),
    "junction": (255, 255, 255),
}


def _rotate(point: tuple[float, float], quarter_turns: int) -> tuple[float, float]:
    x, y = point
    cx, cy = WIDTH / 2.0, HEIGHT / 2.0
    dx, dy = x - cx, y - cy
    q = quarter_turns % 4
    if q == 0:
        rx, ry = dx, dy
    elif q == 1:
        rx, ry = -dy, dx
    elif q == 2:
        rx, ry = -dx, -dy
    else:
        rx, ry = dy, -dx
    return cx + rx, cy + ry


def _node_color_assignment(cycle_color: str) -> dict[str, str]:
    return {
        "g": "green",
        "b": "blue" if cycle_color == "red" else "red",
        "r": cycle_color,
        "p": "junction",
        "c0": "junction",
        "c1": "junction",
        "c2": "junction",
        "c3": "junction",
    }


def _distances(scene: dict) -> Iterable[float]:
    nodes = {node["id"]: node for node in scene["nodes"]}
    linked = {frozenset(edge) for edge in scene["edges"]}
    ids = list(nodes)
    for i, left_id in enumerate(ids):
        for right_id in ids[i + 1 :]:
            if frozenset((left_id, right_id)) in linked:
                continue
            left, right = nodes[left_id], nodes[right_id]
            yield math.hypot(left["x"] - right["x"], left["y"] - right["y"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    quarter_turns = rng.randrange(4)
    tx = rng.uniform(-16.0, 16.0)
    ty = rng.uniform(-12.0, 12.0)
    cycle_color = rng.choice(("red", "blue"))
    colors = _node_color_assignment(cycle_color)

    nodes: list[dict] = []
    for node_id, base in BASE_POINTS.items():
        x, y = _rotate(base, quarter_turns)
        x += tx
        y += ty
        if colors[node_id] == "junction":
            radius = 0.0
        else:
            radius = rng.uniform(20.0, 28.0)
        nodes.append(
            {
                "id": node_id,
                "x": round(x, 3),
                "y": round(y, 3),
                "color": colors[node_id],
                "radius": round(radius, 3),
            }
        )

    # These strokes are visible but deliberately pale; they do not belong to
    # the dark graph and test color/connected-component selectivity.
    distractor_templates = [
        [(90.0, 105.0), (260.0, 105.0)],
        [(700.0, 585.0), (870.0, 585.0)],
        [(110.0, 635.0), (250.0, 635.0), (300.0, 610.0)],
    ]
    distractors = []
    for template in distractor_templates:
        transformed = []
        for point in template:
            x, y = _rotate(point, quarter_turns)
            transformed.append([round(x + tx, 3), round(y + ty, 3)])
        distractors.append(transformed)

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
        "route_color": [37, 61, 77],
        "route_width": rng.randint(11, 14),
        "nodes": nodes,
        "edges": [edge[:] for edge in EDGES],
        "distractors": distractors,
    }


def _adjacency(nodes: list[dict], edges: list[list[str]]) -> dict[str, list[str]]:
    adjacency = {node["id"]: [] for node in nodes}
    for left, right in edges:
        adjacency[left].append(right)
        adjacency[right].append(left)
    return adjacency


def _simple_path_count(adjacency: dict[str, list[str]], start: str, target: str) -> int:
    visited: set[str] = set()

    def visit(node: str) -> int:
        if node == target:
            return 1
        visited.add(node)
        total = 0
        for neighbor in adjacency.get(node, []):
            if neighbor in visited:
                continue
            total += visit(neighbor)
            if total >= 2:
                total = 2
                break
        visited.remove(node)
        return total

    return visit(start)


def analytic_gold(scene: dict) -> str:
    nodes = scene["nodes"]
    edges = scene["edges"]
    by_color = {node["color"]: node["id"] for node in nodes if node["color"] in {"green", "red", "blue"}}
    adjacency = _adjacency(nodes, edges)
    start = by_color["green"]
    red_count = _simple_path_count(adjacency, start, by_color["red"])
    blue_count = _simple_path_count(adjacency, start, by_color["blue"])
    if red_count == blue_count:
        return "unknown"
    return "red" if red_count > blue_count else "blue"


def margin(scene: dict) -> float:
    nodes = {node["id"]: node for node in scene["nodes"]}
    radii = [node["radius"] for node in scene["nodes"] if node["radius"] > 0]
    marker_clearance = min(_distances(scene)) - (max(radii) if radii else 0.0)
    # A second term is a structural runner-up gap: the two valid counts are
    # separated by one route, while pixel clearance controls observability.
    adjacency = _adjacency(scene["nodes"], scene["edges"])
    by_color = {node["color"]: node["id"] for node in scene["nodes"] if node["color"] in {"green", "red", "blue"}}
    counts = sorted(
        [_simple_path_count(adjacency, by_color["green"], by_color["red"]),
         _simple_path_count(adjacency, by_color["green"], by_color["blue"])],
        reverse=True,
    )
    return float(round(min(marker_clearance - scene["route_width"], 100.0 * (counts[0] - counts[1])), 3))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0 or analytic_gold(scene) not in {"red", "blue"}


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Every field that can affect the label is explicitly rendered.  There is
    # no unused permutation or ordering alias in this minimal graph encoding.
    return []


def render(scene: dict) -> Image.Image:
    supersample = 2
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * supersample, height * supersample), scene["background"])
    draw = ImageDraw.Draw(image)

    def scaled_path(path: list[list[float]]) -> list[tuple[float, float]]:
        return [(point[0] * supersample, point[1] * supersample) for point in path]

    for path in scene["distractors"]:
        draw.line(
            scaled_path(path),
            fill=(190, 201, 211),
            width=5 * supersample,
            joint="curve",
        )

    by_id = {node["id"]: node for node in scene["nodes"]}
    route_rgb = tuple(int(value) for value in scene["route_color"])
    route_width = max(1, int(round(scene["route_width"] * supersample)))
    for left_id, right_id in scene["edges"]:
        left, right = by_id[left_id], by_id[right_id]
        draw.line(
            [(left["x"] * supersample, left["y"] * supersample),
             (right["x"] * supersample, right["y"] * supersample)],
            fill=route_rgb,
            width=route_width,
            joint="curve",
        )

    for node in scene["nodes"]:
        if node["color"] not in PALETTE or node["color"] == "junction":
            continue
        radius = node["radius"] * supersample
        x, y = node["x"] * supersample, node["y"] * supersample
        fill = PALETTE[node["color"]]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)

    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image
