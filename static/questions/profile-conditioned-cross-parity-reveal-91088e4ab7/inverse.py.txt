"""Independent PNG-only inverse arm for the cross-neighbor parity world."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 12


def _make_templates() -> dict[str, np.ndarray]:
    result = {name: np.zeros((N, N), dtype=np.uint8) for name in ("flag", "hook", "bolt", "cup")}
    for r in range(2, 10):
        result["flag"][r, 2:4] = 1
    result["flag"][2:5, 4:9] = 1
    result["flag"][9, 1:6] = 1
    result["hook"][2:9, 2:4] = 1
    result["hook"][8:10, 3:9] = 1
    result["hook"][7, 7:9] = 1
    for r, c in ((2,8),(2,9),(3,7),(3,8),(4,6),(4,7),(5,5),(5,6),(5,7),(6,4),(6,5),(6,6),(6,7),(7,3),(7,4),(7,5),(8,2),(8,3),(9,1),(9,2),(4,8),(7,6),(8,4),(3,9)):
        result["bolt"][r, c] = 1
    result["cup"][3:8, 2:4] = 1
    result["cup"][3:8, 8:10] = 1
    result["cup"][8:10, 3:9] = 1
    return result


TEMPLATES = _make_templates()


def _source_cells(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (arr[:, :, 0] < 40) & (arr[:, :, 1] > 145) & (arr[:, :, 1] < 205) & (arr[:, :, 2] > 165)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 500:
        raise ValueError("cyan grid frame is missing")
    min_x, max_x, min_y, max_y = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = max_x - min_x + 1, max_y - min_y + 1
    if not (224 <= width <= 232 and 224 <= height <= 232):
        raise ValueError("cyan fiducial has invalid dimensions")
    x0, y0 = min_x + 6, min_y + 6
    cell_w, cell_h = (width - 12) / N, (height - 12) / N
    cells = np.zeros((N, N), dtype=np.uint8)
    for r in range(N):
        for c in range(N):
            cx = int(round(x0 + (c + 0.5) * cell_w))
            cy = int(round(y0 + (r + 0.5) * cell_h))
            patch = arr[cy - 3:cy + 4, cx - 3:cx + 4]
            cells[r, c] = int(float(patch.mean()) < 90.0)
    return cells


def _transform(cells: np.ndarray) -> np.ndarray:
    p = np.pad(cells, 1)
    return p[1:-1,1:-1] ^ p[:-2,1:-1] ^ p[2:,1:-1] ^ p[1:-1,:-2] ^ p[1:-1,2:]


def decision_from_image(image: Image.Image) -> str:
    transformed = _transform(_source_cells(image))
    scores = sorted((int(np.count_nonzero(transformed ^ template)), name) for name, template in TEMPLATES.items())
    if scores[0][0] == scores[1][0]:
        raise ValueError("symbol classification is tied")
    return scores[0][1]
