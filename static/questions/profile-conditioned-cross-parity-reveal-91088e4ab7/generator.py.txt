"""Scene sampler and renderer for the cross-neighbor parity world."""

from __future__ import annotations

import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

N = 12
CELL = 18
GRID_PX = N * CELL
CYAN = (0, 174, 199)
MAGENTA = (220, 45, 135)
YELLOW = (244, 190, 35)


def _coords(*parts: set[tuple[int, int]]) -> np.ndarray:
    out = np.zeros((N, N), dtype=np.uint8)
    for part in parts:
        for r, c in part:
            out[r, c] = 1
    return out


def _templates() -> dict[str, np.ndarray]:
    flag = _coords(
        {(r, c) for r in range(2, 10) for c in (2, 3)},
        {(r, c) for r in range(2, 5) for c in range(4, 9)},
        {(9, c) for c in range(1, 6)},
    )
    hook = _coords(
        {(r, c) for r in range(2, 9) for c in (2, 3)},
        {(r, c) for r in (8, 9) for c in range(3, 9)},
        {(7, c) for c in (7, 8)},
    )
    bolt = _coords(
        {(2, 8), (2, 9), (3, 7), (3, 8), (4, 6), (4, 7),
         (5, 5), (5, 6), (5, 7), (6, 4), (6, 5), (6, 6),
         (6, 7), (7, 3), (7, 4), (7, 5), (8, 2), (8, 3),
         (9, 1), (9, 2), (4, 8), (7, 6), (8, 4), (3, 9)},
    )
    cup = _coords(
        {(r, c) for r in range(3, 8) for c in (2, 3, 8, 9)},
        {(r, c) for r in (8, 9) for c in range(3, 9)},
    )
    return {"flag": flag, "hook": hook, "bolt": bolt, "cup": cup}


TEMPLATES = _templates()
DECISIONS = tuple(TEMPLATES)


def _cross_parity(cells: np.ndarray) -> np.ndarray:
    padded = np.pad(cells.astype(np.uint8), 1)
    return (
        padded[1:-1, 1:-1]
        ^ padded[:-2, 1:-1]
        ^ padded[2:, 1:-1]
        ^ padded[1:-1, :-2]
        ^ padded[1:-1, 2:]
    )


def _inverse_cross(target: np.ndarray) -> np.ndarray:
    """Solve A*x=target over GF(2); the 12x12 white-boundary operator is invertible."""
    size = N * N
    rows: list[int] = []
    rhs: list[int] = []
    for r in range(N):
        for c in range(N):
            bits = 0
            for dr, dc in ((0, 0), (-1, 0), (1, 0), (0, -1), (0, 1)):
                rr, cc = r + dr, c + dc
                if 0 <= rr < N and 0 <= cc < N:
                    bits |= 1 << (rr * N + cc)
            rows.append(bits)
            rhs.append(int(target[r, c]))
    pivot_row = 0
    pivot_for_col: dict[int, int] = {}
    for col in range(size):
        pivot = next((i for i in range(pivot_row, size) if (rows[i] >> col) & 1), None)
        if pivot is None:
            continue
        rows[pivot_row], rows[pivot] = rows[pivot], rows[pivot_row]
        rhs[pivot_row], rhs[pivot] = rhs[pivot], rhs[pivot_row]
        for i in range(size):
            if i != pivot_row and ((rows[i] >> col) & 1):
                rows[i] ^= rows[pivot_row]
                rhs[i] ^= rhs[pivot_row]
        pivot_for_col[col] = pivot_row
        pivot_row += 1
    if pivot_row != size:
        raise RuntimeError("cross-parity matrix unexpectedly singular")
    solution = np.zeros(size, dtype=np.uint8)
    for col, row in pivot_for_col.items():
        solution[col] = rhs[row]
    return solution.reshape(N, N)


def _scores(transformed: np.ndarray) -> list[tuple[int, str]]:
    return sorted((int(np.count_nonzero(transformed ^ template)), name) for name, template in TEMPLATES.items())


def _decoded(cells: list[list[int]]) -> tuple[str, float]:
    transformed = _cross_parity(np.asarray(cells, dtype=np.uint8))
    scores = _scores(transformed)
    return scores[0][1], float(scores[1][0] - scores[0][0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label_index = int(seed) % len(DECISIONS)
    label = DECISIONS[label_index]
    target = TEMPLATES[label].copy()
    other = TEMPLATES[DECISIONS[(label_index + 1 + rng.randrange(3)) % len(DECISIONS)]]
    differing = [(r, c) for r in range(N) for c in range(N) if target[r, c] != other[r, c]]
    rng.shuffle(differing)
    # Easy, medium, and near-boundary cases are deterministic and class-balanced.
    desired = (0, max(1, len(differing) // 5), max(1, len(differing) // 2 - 2))[int(seed) % 3]
    for flips in range(desired, -1, -1):
        candidate = target.copy()
        for r, c in differing[:flips]:
            candidate[r, c] ^= 1
        source = _inverse_cross(candidate)
        decision, gap = _decoded(source.tolist())
        if decision == label and gap > 0:
            break
    palettes = [
        ((231, 72, 82), (78, 110, 220)),
        ((126, 76, 186), (230, 126, 34)),
        ((54, 150, 95), (205, 75, 148)),
    ]
    colors = palettes[(int(seed) // 4) % len(palettes)]
    decorations = [
        {"x": 380, "y": 82, "color": list(colors[0]), "kind": "diamond"},
        {"x": 398, "y": 226, "color": list(colors[1]), "kind": "circle"},
    ]
    if rng.randrange(2):
        decorations.reverse()
    return {
        "cells": source.astype(int).tolist(),
        "grid_x": 72 + 12 * rng.randrange(4),
        "grid_y": 70 + 10 * rng.randrange(3),
        "decorations": decorations,
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, decorations: list[dict]) -> None:
    for item in sorted(decorations, key=lambda value: (value["x"], value["y"], value["kind"])):
        x, y = int(item["x"]), int(item["y"])
        color = tuple(int(v) for v in item["color"])
        if item["kind"] == "diamond":
            draw.polygon([(x, y - 10), (x + 10, y), (x, y + 10), (x - 10, y)], fill=color)
        else:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)


def render(scene: dict) -> Image.Image:
    cells = np.asarray(scene["cells"], dtype=np.uint8)
    x, y = int(scene["grid_x"]), int(scene["grid_y"])
    image = Image.new("RGB", (480, 360), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene["decorations"])
    # Exact six-pixel cyan frame; the inverse arm detects this visible fiducial.
    draw.rectangle((x - 6, y - 6, x + GRID_PX + 5, y + GRID_PX + 5), fill=CYAN)
    draw.rectangle((x, y, x + GRID_PX - 1, y + GRID_PX - 1), fill=(255, 255, 255))
    for r in range(N):
        for c in range(N):
            left, top = x + c * CELL, y + r * CELL
            if cells[r, c]:
                draw.rectangle((left + 2, top + 2, left + CELL - 3, top + CELL - 3), fill=(20, 22, 25))
    for k in range(N + 1):
        draw.line((x + k * CELL, y, x + k * CELL, y + GRID_PX), fill=(204, 207, 207), width=1)
        draw.line((x, y + k * CELL, x + GRID_PX, y + k * CELL), fill=(204, 207, 207), width=1)
    draw.rectangle((x - 3, y - 18, x + 13, y - 7), fill=MAGENTA)
    draw.rectangle((x + GRID_PX - 14, y - 18, x + GRID_PX + 2, y - 7), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return _decoded(scene["cells"])[0]


def margin(scene: dict) -> float:
    return _decoded(scene["cells"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("decoration_serialization_order", twin)]
