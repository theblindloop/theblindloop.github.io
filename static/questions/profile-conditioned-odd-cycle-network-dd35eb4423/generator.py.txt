"""Deterministic renderer and analytic arm for the odd-cycle graph world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
BACKGROUND = [248, 248, 244]
LINK_COLOR = (38, 42, 48)
NODE_COLORS = [
    [42, 111, 190],
    [39, 139, 91],
    [126, 78, 184],
]

# A cube graph: connected, 3-regular, eight vertices, and bipartite.
_BIPARTITE_NODES = [
    (88, 88), (424, 88), (424, 424), (88, 424),
    (192, 192), (320, 192), (320, 320), (192, 320),
]
_BIPARTITE_EDGES = [
    (0, 1), (1, 2), (2, 3), (0, 3),
    (4, 5), (5, 6), (6, 7), (4, 7),
    (0, 4), (1, 5), (2, 6), (3, 7),
]

# Two K4-minus-edge diamonds joined at their tips: also connected, 3-regular,
# and matched in vertex/edge/degree counts, but it visibly contains triangles.
_ODD_NODES = [
    (72, 256), (184, 256), (128, 56), (128, 456),
    (384, 56), (384, 456), (328, 256), (440, 256),
]
_ODD_EDGES = [
    (0, 1), (0, 2), (1, 2), (0, 3), (1, 3),
    (6, 7), (6, 4), (7, 4), (6, 5), (7, 5),
    (2, 4), (3, 5),
]


def _transform_nodes(points: list[tuple[int, int]], seed: int) -> list[list[int]]:
    rng = random.Random(seed ^ 0x5EEDC0DE)
    quarter_turns = (seed // 2) % 4
    reflect = bool((seed // 8) % 2)
    scale = (0.91, 0.95, 0.99)[(seed // 16) % 3]
    tx = rng.randint(-5, 5)
    ty = rng.randint(-5, 5)
    transformed: list[list[int]] = []
    for x, y in points:
        dx = x - 256
        dy = y - 256
        if reflect:
            dx = -dx
        for _ in range(quarter_turns):
            dx, dy = -dy, dx
        jx = rng.randint(-3, 3)
        jy = rng.randint(-3, 3)
        transformed.append([
            int(round(256 + scale * dx + tx + jx)),
            int(round(256 + scale * dy + ty + jy)),
        ])
    return transformed


def sample_scene(seed: int) -> dict:
    """Sample balanced graph classes with class-independent visual nuisance."""
    has_odd_cycle = bool(seed % 2)
    base_nodes = _ODD_NODES if has_odd_cycle else _BIPARTITE_NODES
    base_edges = _ODD_EDGES if has_odd_cycle else _BIPARTITE_EDGES
    edges = [list(edge) for edge in base_edges]
    random.Random(seed ^ 0xA11CE).shuffle(edges)
    return {
        "nodes": _transform_nodes(base_nodes, seed),
        "edges": edges,
        "node_radius": (14, 15, 16)[(seed // 6) % 3],
        "link_width": (6, 8, 10, 12)[(seed // 2) % 4],
        "node_color": NODE_COLORS[(seed // 8) % len(NODE_COLORS)],
        "background": BACKGROUND,
        "canvas_size": CANVAS_SIZE,
    }


def render(scene: dict) -> Image.Image:
    """Render dark straight links first and colored circular junctions second."""
    nodes = scene["nodes"]
    edges = scene["edges"]
    node_radius = int(scene["node_radius"])
    link_width = int(scene["link_width"])
    node_color = tuple(scene["node_color"])
    background = tuple(scene["background"])
    canvas_size = int(scene["canvas_size"])
    ss = 3
    image = Image.new("RGB", (canvas_size * ss, canvas_size * ss), background)
    draw = ImageDraw.Draw(image)
    for a, b in edges:
        p = tuple(int(v * ss) for v in nodes[a])
        q = tuple(int(v * ss) for v in nodes[b])
        draw.line([p, q], fill=LINK_COLOR, width=link_width * ss)
    for x, y in nodes:
        box = [
            (x - node_radius) * ss,
            (y - node_radius) * ss,
            (x + node_radius) * ss,
            (y + node_radius) * ss,
        ]
        draw.ellipse(
            box,
            fill=node_color,
            outline=LINK_COLOR,
            width=2 * ss,
        )
    return image.resize((canvas_size, canvas_size), Image.Resampling.LANCZOS)


def _graph_has_odd_cycle(nodes: list, edges: list) -> bool:
    adjacency = [set() for _ in nodes]
    for raw_a, raw_b in edges:
        a, b = int(raw_a), int(raw_b)
        if a == b or not (0 <= a < len(nodes) and 0 <= b < len(nodes)):
            raise ValueError("invalid graph edge")
        adjacency[a].add(b)
        adjacency[b].add(a)
    colors: list[int | None] = [None] * len(nodes)
    for start in range(len(nodes)):
        if colors[start] is not None:
            continue
        colors[start] = 0
        queue = [start]
        for vertex in queue:
            for neighbor in adjacency[vertex]:
                if colors[neighbor] is None:
                    colors[neighbor] = 1 - int(colors[vertex])
                    queue.append(neighbor)
                elif colors[neighbor] == colors[vertex]:
                    return True
    return False


def analytic_gold(scene: dict) -> str:
    return "yes" if _graph_has_odd_cycle(scene["nodes"], scene["edges"]) else "no"


def margin(scene: dict) -> float:
    """One-sided raster robustness: spare link thickness above 5 pixels."""
    return float(min(int(scene["link_width"]) - 5, int(scene["node_radius"]) - 10))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["edges"] = list(reversed(scene["edges"]))
    return [("reverse_serialized_edge_order", twin)]
