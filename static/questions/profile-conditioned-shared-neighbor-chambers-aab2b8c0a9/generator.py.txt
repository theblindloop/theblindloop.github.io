"""Deterministic scene sampler and renderer.

The latent graph is fully visible: round white chambers are vertices and thick
gray corridor bars are edges.  Colored dots bind two vertices to each answer
label.  Gold is the size of the intersection of each pair's open neighborhoods.
"""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

CANVAS = 512
GRID = (100, 256, 412)
RADIUS = 38
CORRIDOR_WIDTH = 16
COLORS = {
    "red": (220, 55, 65),
    "green": (38, 158, 88),
    "blue": (52, 104, 210),
}


def _base_edges() -> list[tuple[int, int]]:
    result = []
    for row in range(3):
        for col in range(3):
            i = row * 3 + col
            if col < 2:
                result.append((i, i + 1))
            if row < 2:
                result.append((i, i + 3))
    return result


def _adjacency(edges: list[list[int]] | list[tuple[int, int]]) -> list[set[int]]:
    adj = [set() for _ in range(9)]
    for u, v in edges:
        adj[u].add(v)
        adj[v].add(u)
    return adj


def _shared_count(adj: list[set[int]], pair: tuple[int, int]) -> int:
    return len(adj[pair[0]] & adj[pair[1]])


def _transform_point(x: int, y: int, transform: int) -> tuple[int, int]:
    # The eight symmetries of the square. These change pixels and are ordinary
    # sampled variation, not latent aliases.
    if transform >= 4:
        x = CANVAS - x
        transform -= 4
    for _ in range(transform):
        x, y = CANVAS - y, x
    return int(x), int(y)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    base = _base_edges()
    chosen = None
    for _ in range(5000):
        edges = [edge for edge in base if rng.random() < 0.72]
        adj = _adjacency(edges)
        by_count = {0: [], 1: [], 2: []}
        for pair in itertools.combinations(range(9), 2):
            count = _shared_count(adj, pair)
            if count in by_count:
                by_count[count].append(pair)
        for values in by_count.values():
            rng.shuffle(values)
        for p2 in by_count[2]:
            for p1 in by_count[1]:
                for p0 in by_count[0]:
                    if len(set((*p2, *p1, *p0))) == 6:
                        chosen = (edges, p2, p1, p0)
                        break
                if chosen:
                    break
            if chosen:
                break
        if chosen:
            break
    if chosen is None:
        raise RuntimeError("could not sample a balanced chamber graph")

    edges, pair_two, pair_one, pair_zero = chosen
    labels = ["red", "green", "blue"]
    shift = int(seed) % 3
    labels = labels[shift:] + labels[:shift]
    marker_for: dict[int, str] = {}
    for label, pair in zip(labels, (pair_two, pair_one, pair_zero)):
        for index in pair:
            marker_for[index] = label

    transform = rng.randrange(8)
    nodes = []
    for index in range(9):
        x, y = GRID[index % 3], GRID[index // 3]
        x, y = _transform_point(x, y, transform)
        nodes.append({"x": x, "y": y, "marker": marker_for.get(index, "none")})
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in sorted(edges)],
    }


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    scale = 4
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), (248, 247, 242))
    draw = ImageDraw.Draw(image)

    for u, v in edges:
        a, b = nodes[u], nodes[v]
        draw.line(
            (a["x"] * scale, a["y"] * scale, b["x"] * scale, b["y"] * scale),
            fill=(92, 108, 124),
            width=CORRIDOR_WIDTH * scale,
        )

    for node in nodes:
        x, y = node["x"] * scale, node["y"] * scale
        r = RADIUS * scale
        draw.ellipse((x - r, y - r, x + r, y + r), fill=(255, 255, 255),
                     outline=(25, 29, 35), width=6 * scale)
        marker = node["marker"]
        if marker != "none":
            mr = 12 * scale
            draw.ellipse((x - mr, y - mr, x + mr, y + mr), fill=COLORS[marker])

    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def _pair_counts(scene: dict) -> dict[str, int]:
    nodes = scene["nodes"]
    edges = scene["edges"]
    adj = _adjacency(edges)
    pairs: dict[str, list[int]] = {color: [] for color in COLORS}
    for index, node in enumerate(nodes):
        marker = node["marker"]
        if marker in pairs:
            pairs[marker].append(index)
    if any(len(pair) != 2 for pair in pairs.values()):
        return {}
    return {color: len(adj[pair[0]] & adj[pair[1]]) for color, pair in pairs.items()}


def analytic_gold(scene: dict) -> str:
    counts = _pair_counts(scene)
    winners = [color for color, count in counts.items() if count == 2]
    return winners[0] if len(winners) == 1 else "ambiguous"


def margin(scene: dict) -> float:
    counts = _pair_counts(scene)
    winner = analytic_gold(scene)
    if winner == "ambiguous":
        return 0.0
    return float(min(abs(count - 2) for color, count in counts.items() if color != winner))


def is_quarantined(scene: dict) -> bool:
    if analytic_gold(scene) == "ambiguous" or margin(scene) < 1.0:
        return True
    nodes = scene["nodes"]
    if len(nodes) != 9 or len(scene["edges"]) < 5:
        return True
    # All sampled links are long axis-aligned bars with a wide visible span.
    for u, v in scene["edges"]:
        dx = abs(nodes[u]["x"] - nodes[v]["x"])
        dy = abs(nodes[u]["y"] - nodes[v]["y"])
        if min(dx + dy, max(dx, dy)) < 120 or (dx and dy):
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
