"""Independent pixel-only inverse arm for the chamber-corridor diagram."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((220, 55, 65), dtype=np.int32),
    "green": np.array((38, 158, 88), dtype=np.int32),
    "blue": np.array((52, 104, 210), dtype=np.int32),
}
CORRIDOR = np.array((92, 108, 124), dtype=np.int32)


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(int(y0), int(x0))])
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centers_from_rings(arr: np.ndarray) -> list[tuple[float, float]]:
    dark = np.max(arr, axis=2) < 65
    components = _components(dark, 220)
    centers = [(float(c[:, 1].mean()), float(c[:, 0].mean())) for c in components]
    if len(centers) != 9:
        raise ValueError(f"expected 9 visible chamber rings, found {len(centers)}")
    return centers


def _nearest(center: tuple[float, float], centers: list[tuple[float, float]]) -> int:
    distances = [(x - center[0]) ** 2 + (y - center[1]) ** 2 for x, y in centers]
    order = np.argsort(distances)
    if len(order) > 1 and distances[int(order[1])] - distances[int(order[0])] < 400:
        raise ValueError("ambiguous corridor endpoint")
    return int(order[0])


def _recover_edges(arr: np.ndarray, centers: list[tuple[float, float]]) -> list[tuple[int, int]]:
    dist = np.sqrt(np.sum((arr - CORRIDOR) ** 2, axis=2))
    bars = _components(dist < 28.0, 350)
    edges = set()
    for component in bars:
        ys, xs = component[:, 0], component[:, 1]
        if np.ptp(xs) > np.ptp(ys):
            endpoint_a = (float(xs.min()), float(np.median(ys)))
            endpoint_b = (float(xs.max()), float(np.median(ys)))
        else:
            endpoint_a = (float(np.median(xs)), float(ys.min()))
            endpoint_b = (float(np.median(xs)), float(ys.max()))
        u, v = _nearest(endpoint_a, centers), _nearest(endpoint_b, centers)
        if u == v:
            raise ValueError("corridor has only one recoverable endpoint")
        edges.add(tuple(sorted((u, v))))
    if len(edges) < 5 or len(edges) != len(bars):
        raise ValueError("corridor segmentation is incomplete or merged")
    return sorted(edges)


def _recover_pairs(arr: np.ndarray, centers: list[tuple[float, float]]) -> dict[str, list[int]]:
    result = {}
    for color, target in PALETTE.items():
        dist = np.sqrt(np.sum((arr - target) ** 2, axis=2))
        blobs = _components(dist < 45.0, 150)
        if len(blobs) != 2:
            raise ValueError(f"expected two {color} dots, found {len(blobs)}")
        indices = []
        for blob in blobs:
            point = (float(blob[:, 1].mean()), float(blob[:, 0].mean()))
            indices.append(_nearest(point, centers))
        if len(set(indices)) != 2:
            raise ValueError(f"{color} dots do not bind two chambers")
        result[color] = indices
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    centers = _centers_from_rings(arr)
    edges = _recover_edges(arr, centers)
    pairs = _recover_pairs(arr, centers)
    adjacency = [set() for _ in centers]
    for u, v in edges:
        adjacency[u].add(v)
        adjacency[v].add(u)
    counts = {
        color: len(adjacency[pair[0]] & adjacency[pair[1]])
        for color, pair in pairs.items()
    }
    winners = [color for color, count in counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"pixel evidence has {len(winners)} valid answers: {counts}")
    return winners[0]
