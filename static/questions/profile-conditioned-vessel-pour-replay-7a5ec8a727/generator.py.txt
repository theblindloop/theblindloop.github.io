"""Deterministic scene sampler and renderer for capacity-limited pours."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 640
CAPACITY = 4
N_OPERATIONS = 5
IDENTITY_COLORS = ((226, 84, 84), (56, 158, 92), (126, 88, 190))
FILL = (48, 190, 210)
DARK = (35, 43, 54)
LIGHT = (239, 243, 247)
ORANGE = (242, 148, 38)
WHITE = (255, 255, 255)


def _apply(levels: list[int], source: int, destination: int) -> int:
    amount = min(levels[source], CAPACITY - levels[destination])
    levels[source] -= amount
    levels[destination] += amount
    return amount


def _replay(scene: dict) -> list[int]:
    levels = [int(value) for value in scene["initial"]]
    for source, destination in scene["operations"]:
        _apply(levels, int(source), int(destination))
    return levels


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed modulo five selects the final answer class."""
    rng = random.Random(int(seed))
    desired = int(seed) % (CAPACITY + 1)
    query = (int(seed) // (CAPACITY + 1)) % 3
    pairs = [(a, b) for a in range(3) for b in range(3) if a != b]
    for _ in range(20000):
        initial = [rng.randrange(CAPACITY + 1) for _ in range(3)]
        if not 2 <= sum(initial) <= 10:
            continue
        operations = [list(rng.choice(pairs)) for _ in range(N_OPERATIONS)]
        levels = initial.copy()
        effective = 0
        for source, destination in operations:
            effective += _apply(levels, source, destination) > 0
        if levels[query] == desired and effective >= 2:
            return {"initial": initial, "operations": operations, "query": query}
    raise RuntimeError("could not sample requested balanced pour scene")


def analytic_gold(scene: dict) -> str:
    return str(_replay(scene)[int(scene["query"])])


def margin(scene: dict) -> float:
    # Runner-up decoding clearance: minimum blank separator between unit cells.
    return 8.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, fill=DARK) -> None:
    font = ImageFont.load_default()
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) // 2, xy[1] - (box[3] - box[1]) // 2), text, fill=fill, font=font)


def render(scene: dict) -> Image.Image:
    initial = [int(value) for value in scene["initial"]]
    operations = [[int(a), int(b)] for a, b in scene["operations"]]
    query = int(scene["query"])
    if len(initial) != 3 or len(operations) != N_OPERATIONS or not 0 <= query < 3:
        raise ValueError("invalid scene shape")

    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((28, 18, 932, 392), radius=18, fill=LIGHT, outline=DARK, width=3)
    _centered(draw, (480, 38), "START: EACH VESSEL HOLDS 4 UNITS")

    centers = (175, 480, 785)
    for slot, (cx, color, level) in enumerate(zip(centers, IDENTITY_COLORS, initial, strict=True)):
        draw.ellipse((cx - 30, 55, cx + 30, 115), fill=color, outline=DARK, width=3)
        if slot == query:
            draw.ellipse((cx - 40, 45, cx + 40, 125), outline=ORANGE, width=7)
        left, right = cx - 72, cx + 72
        top, bottom = 145, 357
        draw.rounded_rectangle((left, top, right, bottom), radius=10, fill=WHITE, outline=DARK, width=4)
        cell_h, gap = 44, 8
        for unit in range(CAPACITY):
            y1 = bottom - 13 - unit * (cell_h + gap)
            y0 = y1 - cell_h
            fill = FILL if unit < level else WHITE
            draw.rectangle((left + 10, y0, right - 10, y1), fill=fill, outline=(170, 180, 190), width=2)

    _centered(draw, (480, 411), "POUR CARDS: SOURCE  ->  DESTINATION")
    card_y0, card_y1 = 432, 606
    for index, (source, destination) in enumerate(operations):
        x0 = 35 + index * 180
        x1 = x0 + 170
        draw.rounded_rectangle((x0, card_y0, x1, card_y1), radius=12, fill=LIGHT, outline=DARK, width=3)
        _centered(draw, ((x0 + x1) // 2, card_y0 + 17), str(index + 1))
        cy = 524
        draw.rectangle((x0 + 26, cy - 18, x0 + 62, cy + 18), fill=IDENTITY_COLORS[source])
        draw.line((x0 + 72, cy, x0 + 105, cy), fill=DARK, width=5)
        draw.polygon(((x0 + 105, cy), (x0 + 94, cy - 9), (x0 + 94, cy + 9)), fill=DARK)
        draw.rectangle((x0 + 116, cy - 18, x0 + 152, cy + 18), fill=IDENTITY_COLORS[destination])
        _centered(draw, ((x0 + x1) // 2, 579), "POUR")
    return image
