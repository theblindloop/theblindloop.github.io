"""Independent pixel-only inverse for the public vessel/card visual grammar."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CAPACITY = 4
IDENTITY_COLORS = ((226, 84, 84), (56, 158, 92), (126, 88, 190))
FILL = (48, 190, 210)
ORANGE = (242, 148, 38)


def _components(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in zip(*np.nonzero(mask & ~seen), strict=True):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [p[0] for p in points]
        xs = [p[1] for p in points]
        output.append({
            "area": len(points),
            "cx": sum(xs) / len(xs),
            "cy": sum(ys) / len(ys),
            "bbox": (min(xs), min(ys), max(xs), max(ys)),
        })
    return output


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.array(rgb, dtype=np.uint8), axis=2)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 300 or arr.shape[1] < 500:
        raise ValueError("image does not contain the public pour layout")
    h, w = arr.shape[:2]

    headers = []
    card_icons = []
    for identity, color in enumerate(IDENTITY_COLORS):
        for comp in _components(_mask(arr, color)):
            if 2200 <= comp["area"] <= 3000 and comp["cy"] < h * 0.30:
                headers.append((comp["cx"], comp["cy"], identity))
            elif 1100 <= comp["area"] <= 1500 and comp["cy"] > h * 0.60:
                card_icons.append((comp["cx"], comp["cy"], identity))
    if len(headers) != 3 or len(card_icons) != 10:
        raise ValueError("identity markers are missing or ambiguous")
    headers.sort()

    rings = [c for c in _components(_mask(arr, ORANGE)) if c["area"] >= 900]
    if len(rings) != 1:
        raise ValueError("query ring is missing or ambiguous")
    query = min(range(3), key=lambda i: abs(headers[i][0] - rings[0]["cx"]))
    if abs(headers[query][0] - rings[0]["cx"]) > w * 0.05:
        raise ValueError("query ring does not bind a vessel")

    levels = [0, 0, 0]
    fill_components = [c for c in _components(_mask(arr, FILL)) if c["area"] >= 4000]
    for comp in fill_components:
        if not (h * 0.20 < comp["cy"] < h * 0.60):
            continue
        slot = min(range(3), key=lambda i: abs(headers[i][0] - comp["cx"]))
        if abs(headers[slot][0] - comp["cx"]) > w * 0.10:
            raise ValueError("fill cell does not bind a vessel")
        levels[slot] += 1
    if any(level > CAPACITY for level in levels):
        raise ValueError("too many unit cells in a vessel")

    card_icons.sort(key=lambda item: item[0])
    for index in range(0, len(card_icons), 2):
        left, right = card_icons[index], card_icons[index + 1]
        if left[2] == right[2] or right[0] - left[0] > w * 0.13:
            raise ValueError("pour endpoints are ambiguous")
        source, destination = left[2], right[2]
        amount = min(levels[source], CAPACITY - levels[destination])
        levels[source] -= amount
        levels[destination] += amount
    return str(levels[query])
