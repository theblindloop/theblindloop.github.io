"""Independent pixel-only inverse for the envelope stamp corner."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "envelope_stamp_corner-oracle-0.1.0"


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected components without using renderer geometry."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= 180:
            components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _stamp_centroid(image: Image.Image) -> tuple[float, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    channel_span = array.max(axis=2) - array.min(axis=2)
    brightness = array.mean(axis=2)
    # The stamp is the only bright, strongly saturated region. Pale paper,
    # dark seams, and the background are deliberately outside this mask.
    mask = (channel_span >= 72) & (brightness >= 55) & (brightness <= 245)
    components = _components(mask)
    if not components:
        raise ValueError("no measurable postage-stamp component")
    component = max(components, key=len)
    ys, xs = component[:, 0], component[:, 1]
    if len(xs) < 1800:
        raise ValueError("saturated component is too small to be a stamp")
    return float(xs.mean()), float(ys.mean())


def _envelope_center(image: Image.Image) -> tuple[float, float]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    low = array.max(axis=2) < 150
    components = _components(low)
    if not components:
        raise ValueError("envelope outline is not measurable")
    # The outer border is the component with the largest extent, while the
    # stamp's dark cancellation mark remains strictly inside that extent.
    points = np.concatenate(components, axis=0)
    ys, xs = points[:, 0], points[:, 1]
    if int(xs.max() - xs.min()) < 250 or int(ys.max() - ys.min()) < 140:
        raise ValueError("envelope frame is not measurable")
    return (float(xs.min() + xs.max()) / 2.0, float(ys.min() + ys.max()) / 2.0)


def decision_from_image(image: Image.Image) -> str:
    """Recover one of four corners from the final raster alone."""

    sx, sy = _stamp_centroid(image)
    cx, cy = _envelope_center(image)
    horizontal = "left" if sx < cx else "right"
    vertical = "top" if sy < cy else "bottom"
    if abs(sx - cx) < 8.0 or abs(sy - cy) < 8.0:
        raise ValueError("stamp is too close to an envelope centerline")
    return f"{vertical}-{horizontal}"
