"""Deterministic raster world for a postage stamp's visible envelope corner."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "envelope_stamp_corner-renderer-0.1.0"
SUPERSAMPLE = 4

BACKGROUND_PALETTE = (
    (238, 243, 247),
    (247, 241, 234),
    (239, 247, 239),
    (246, 239, 248),
)
ENVELOPE_PALETTE = (
    (252, 252, 248),
    (250, 245, 232),
    (238, 249, 246),
    (247, 239, 251),
)
FLAP_PALETTE = (
    (245, 249, 246),
    (249, 241, 229),
    (234, 245, 244),
    (243, 235, 248),
)
OUTLINE_PALETTE = (
    (42, 54, 66),
    (54, 48, 62),
    (46, 65, 58),
    (67, 52, 43),
)
STAMP_PALETTE = (
    (236, 64, 74),
    (48, 119, 224),
    (34, 171, 111),
    (235, 143, 40),
    (151, 78, 213),
)
CORNERS = ("top-left", "top-right", "bottom-left", "bottom-right")


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _scaled_box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in values)  # type: ignore[return-value]


def _body_bounds(scene: dict[str, Any]) -> tuple[float, float, float, float]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    width = float(scene["body_width"])
    height = float(scene["body_height"])
    return (cx - width / 2.0, cy - height / 2.0, cx + width / 2.0, cy + height / 2.0)


def _stamp_center(scene: dict[str, Any]) -> tuple[float, float]:
    left, top, right, bottom = _body_bounds(scene)
    corner = str(scene["stamp_corner"])
    x = left + float(scene["stamp_inset_x"]) if corner.endswith("left") else right - float(scene["stamp_inset_x"])
    y = top + float(scene["stamp_inset_y"]) if corner.startswith("top") else bottom - float(scene["stamp_inset_y"])
    return x, y


def _stamp_box(scene: dict[str, Any]) -> tuple[float, float, float, float]:
    x, y = _stamp_center(scene)
    half_w = float(scene["stamp_width"]) / 2.0
    half_h = float(scene["stamp_height"]) / 2.0
    return (x - half_w, y - half_h, x + half_w, y + half_h)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, JSON-friendly envelope scene."""

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 73_021)
    corner = CORNERS[seed % len(CORNERS)]
    width, height = 640, 440
    body_width = rng.randint(390, 430)
    body_height = rng.randint(244, 278)
    center_x = round(rng.uniform(300.0, 340.0), 4)
    center_y = round(rng.uniform(211.0, 229.0), 4)
    stamp_width = rng.randint(76, 94)
    stamp_height = rng.randint(56, 72)
    if seed % 11 == 0:
        # Explicit finite-resolution near-boundary band. It remains visibly
        # classifiable, but its centerline clearance is quarantined.
        stamp_inset_x = round(rng.uniform(102.0, 124.0), 4)
        stamp_inset_y = round(rng.uniform(76.0, 91.0), 4)
    else:
        stamp_inset_x = round(rng.uniform(54.0, 72.0), 4)
        stamp_inset_y = round(rng.uniform(38.0, 56.0), 4)
    return {
        "stamp_corner": corner,
        "width": width,
        "height": height,
        "center_x": center_x,
        "center_y": center_y,
        "body_width": body_width,
        "body_height": body_height,
        "stamp_width": stamp_width,
        "stamp_height": stamp_height,
        "stamp_inset_x": stamp_inset_x,
        "stamp_inset_y": stamp_inset_y,
        "outline_width": rng.choice((5, 6, 7)),
        "background_color": list(rng.choice(BACKGROUND_PALETTE)),
        "envelope_color": list(rng.choice(ENVELOPE_PALETTE)),
        "flap_color": list(rng.choice(FLAP_PALETTE)),
        "outline_color": list(rng.choice(OUTLINE_PALETTE)),
        "stamp_color": list(rng.choice(STAMP_PALETTE)),
        "antialias": True,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one front-view envelope with a saturated stamp."""

    width, height = int(scene["width"]), int(scene["height"])
    scale = SUPERSAMPLE if bool(scene["antialias"]) else 1
    background = _rgb(scene["background_color"])
    envelope = _rgb(scene["envelope_color"])
    flap = _rgb(scene["flap_color"])
    outline = _rgb(scene["outline_color"])
    stamp = _rgb(scene["stamp_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    left, top, right, bottom = _body_bounds(scene)
    cx = float(scene["center_x"])
    flap_y = top + float(scene["body_height"]) * 0.54
    line_width = max(1, int(scene["outline_width"])) * scale
    draw.rectangle(
        _scaled_box((left, top, right, bottom), scale),
        fill=envelope,
        outline=outline,
        width=line_width,
    )
    draw.polygon(
        [
            (int(round(left * scale)), int(round(top * scale))),
            (int(round(cx * scale)), int(round(flap_y * scale))),
            (int(round(right * scale)), int(round(top * scale))),
        ],
        fill=flap,
    )
    draw.line(
        [(int(round(left * scale)), int(round(top * scale))), (int(round(cx * scale)), int(round(flap_y * scale))), (int(round(right * scale)), int(round(top * scale)))],
        fill=outline,
        width=line_width,
        joint="curve",
    )
    draw.line(
        [(int(round(left * scale)), int(round(bottom * scale))), (int(round(cx * scale)), int(round(flap_y * scale)))],
        fill=outline,
        width=max(1, scale * 2),
    )
    draw.line(
        [(int(round(right * scale)), int(round(bottom * scale))), (int(round(cx * scale)), int(round(flap_y * scale)))],
        fill=outline,
        width=max(1, scale * 2),
    )
    # Address bars make the object legible without adding text or another
    # color-bearing object that the inverse could mistake for the stamp.
    for offset, length in ((0.0, 118.0), (17.0, 92.0), (34.0, 106.0)):
        y = top + float(scene["body_height"]) * 0.68 + offset
        x0 = left + 46.0
        draw.line(
            [(int(round(x0 * scale)), int(round(y * scale))), (int(round((x0 + length) * scale)), int(round(y * scale)))],
            fill=outline,
            width=max(1, scale * 2),
        )

    stamp_box = _scaled_box(_stamp_box(scene), scale)
    draw.rectangle(stamp_box, fill=stamp, outline=outline, width=line_width)
    # A small cancellation mark reinforces the stamp reading while leaving the
    # saturated component connected for the independent inverse.
    sx0, sy0, sx1, sy1 = stamp_box
    mark_x = (sx0 + sx1) // 2
    mark_y = (sy0 + sy1) // 2
    draw.ellipse(
        (mark_x - 7 * scale, mark_y - 7 * scale, mark_x + 7 * scale, mark_y + 7 * scale),
        outline=outline,
        width=max(1, scale * 2),
    )
    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible corner label encoded by the stamp location."""

    return str(scene["stamp_corner"])


def margin(scene: dict[str, Any]) -> float:
    """Centerline clearance of the stamp centroid in pixels."""

    sx, sy = _stamp_center(scene)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    return round(min(abs(sx - cx), abs(sy - cy)), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 60.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; corner is a visible label input."""

    return []
