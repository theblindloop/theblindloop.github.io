"""Renderer for the hinge-fold difference reveal world.

Two five-by-five cell panels are drawn far apart on the canvas.  One of them
carries a thick hinge bar along its inner edge.  The declared raster operation
folds the hinged panel across that hinge onto the other panel - a left-right
mirror followed by superposition - and marks every cell that is dark in exactly
one of the two stacked layers.  The nine marked cells spell one of five
symbols.  Neither panel contains the symbol and neither panel alone constrains
it, because the mirrored partner grid is the one-time pad of the other.
"""

from __future__ import annotations

import hashlib
import json
from typing import Iterable

import numpy as np
from PIL import Image

RENDERER_VERSION = "hinge_fold_difference_reveal-renderer-0.1.0"

GRID = 5
CELL_PX = 32
BORDER_PX = 3
HINGE_PX = 11
CANVAS = (620, 262)
PANEL_PX = GRID * CELL_PX + 2 * BORDER_PX  # 166

FRAME_RGB = (28, 58, 168)
HINGE_RGB = (232, 122, 26)
CLUTTER_RGB = (58, 148, 122)
BACKGROUND_HUE = (32, -24, 40)  # the background is visibly tinted, never a cell tone

QUARANTINE_CLEARANCE = 34.0
PAD_INK = 12  # dark cells in the plain panel

SYMBOLS = ("plus", "ex", "tee", "ell", "seven")

_TEMPLATE_ART = {
    "plus": ("..#..", "..#..", "#####", "..#..", "..#.."),
    "ex": ("#...#", ".#.#.", "..#..", ".#.#.", "#...#"),
    "tee": ("#####", "..#..", "..#..", "..#..", "..#.."),
    "ell": ("#....", "#....", "#....", "#....", "#####"),
    "seven": ("#####", "...#.", "..#..", ".#...", "#...."),
}


def _template(name: str) -> np.ndarray:
    rows = _TEMPLATE_ART[name]
    return np.array([[1 if ch == "#" else 0 for ch in row] for row in rows], dtype=np.int8)


TEMPLATES = {name: _template(name) for name in SYMBOLS}


def _hamming(a: np.ndarray, b: np.ndarray) -> int:
    return int(np.count_nonzero(a != b))


def _min_template_distance(grid: np.ndarray) -> int:
    return min(_hamming(grid, tpl) for tpl in TEMPLATES.values())


def _rng(seed: int) -> np.random.Generator:
    digest = hashlib.sha256(f"hinge-fold-difference|{int(seed)}".encode()).digest()
    return np.random.default_rng(int.from_bytes(digest[:8], "big"))


def left_grid(scene: dict) -> np.ndarray:
    return np.array(scene["left_grid"], dtype=np.int8)


def right_grid(scene: dict) -> np.ndarray:
    return np.array(scene["right_grid"], dtype=np.int8)


def sample_scene(seed: int) -> dict:
    rng = _rng(seed)
    symbol = SYMBOLS[int(rng.integers(len(SYMBOLS)))]
    target = TEMPLATES[symbol]
    on = np.argwhere(target == 1)
    off = np.argwhere(target == 0)

    pad = None
    for _ in range(600):
        candidate = np.zeros((GRID, GRID), dtype=np.int8)
        take_on = rng.choice(len(on), size=4, replace=False)
        take_off = rng.choice(len(off), size=PAD_INK - 4, replace=False)
        for index in take_on:
            candidate[tuple(on[index])] = 1
        for index in take_off:
            candidate[tuple(off[index])] = 1
        folded = np.bitwise_xor(candidate, target)  # the hinged panel seen after folding
        hinged_raw = folded[:, ::-1]
        naive = np.bitwise_xor(candidate, hinged_raw)  # what skipping the fold would give
        checks = (
            candidate,
            candidate[:, ::-1],
            hinged_raw,
            folded,
            naive,
            naive[:, ::-1],
        )
        if all(_min_template_distance(grid) >= 4 for grid in checks):
            pad = candidate
            break
    if pad is None:  # pragma: no cover - the constraint is easily satisfiable
        raise RuntimeError("could not sample a counterbalanced pad")

    folded = np.bitwise_xor(pad, TEMPLATES[symbol])
    hinged_raw = folded[:, ::-1]

    hinge_side = "left" if bool(rng.integers(2)) else "right"
    if hinge_side == "right":
        left_grid, right_grid = pad, hinged_raw
    else:
        left_grid, right_grid = hinged_raw, pad

    left_origin = [int(rng.integers(18, 36)), int(rng.integers(38, 62))]
    right_origin = [int(rng.integers(418, 436)), int(rng.integers(38, 62))]

    pale_value = int(rng.integers(198, 236))
    gap = int(rng.integers(56, 133))
    ink_value = pale_value - gap
    tint = [int(rng.integers(-7, 8)) for _ in range(3)]
    background_value = int(rng.integers(118, 158))

    clutter: list[list[int]] = []
    for _ in range(int(rng.integers(2, 5))):
        size = int(rng.integers(7, 12))
        x = int(rng.integers(214, 400))
        y = int(rng.integers(8, CANVAS[1] - size - 8))
        clutter.append([x, y, size])

    return {
        "canvas": list(CANVAS),
        "cell_px": CELL_PX,
        "border_px": BORDER_PX,
        "hinge_px": HINGE_PX,
        "left_origin": left_origin,
        "right_origin": right_origin,
        "hinge_side": hinge_side,
        "left_grid": left_grid.astype(int).tolist(),
        "right_grid": right_grid.astype(int).tolist(),
        "symbol": symbol,
        "pale_value": pale_value,
        "ink_value": ink_value,
        "background_value": background_value,
        "tint": tint,
        "clearance": gap / 2.0,
        "clutter": clutter,
    }


def revealed_grid(scene: dict) -> np.ndarray:
    """The five-by-five difference grid produced by the declared fold."""

    left = left_grid(scene)
    right = right_grid(scene)
    if scene["hinge_side"] == "right":
        folded = right[:, ::-1]
        return np.bitwise_xor(left, folded)
    folded = left[:, ::-1]
    return np.bitwise_xor(right, folded)


def analytic_gold(scene: dict) -> str:
    grid = revealed_grid(scene)
    distances = sorted((_hamming(grid, tpl), name) for name, tpl in TEMPLATES.items())
    if distances[0][0] != 0:
        raise ValueError("revealed grid is not a declared symbol")
    return distances[0][1]


def margin(scene: dict) -> float:
    return float((int(scene["pale_value"]) - int(scene["ink_value"])) / 2.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases are declared; every sampled field is rendered."""

    return []


def _tone(value: int, tint: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(np.clip(value + channel, 0, 255)) for channel in tint)  # type: ignore[return-value]


def panel_origin(scene: dict, side: str) -> tuple[int, int]:
    x, y = scene["left_origin"] if side == "left" else scene["right_origin"]
    return int(x), int(y)


def cell_box(scene: dict, side: str, row: int, col: int) -> tuple[int, int, int, int]:
    x, y = panel_origin(scene, side)
    border = int(scene["border_px"])
    cell = int(scene["cell_px"])
    x0 = x + border + col * cell
    y0 = y + border + row * cell
    return x0, y0, x0 + cell, y0 + cell


def hinge_box(scene: dict) -> tuple[int, int, int, int]:
    side = scene["hinge_side"]
    x, y = panel_origin(scene, side)
    width = int(scene["hinge_px"])
    if side == "right":  # hinge lies on the right panel's left (inner) edge
        return x - width - 2, y, x - 2, y + PANEL_PX
    return x + PANEL_PX + 2, y, x + PANEL_PX + 2 + width, y + PANEL_PX


def render(scene: dict) -> Image.Image:
    width, height = (int(v) for v in scene["canvas"])
    tint = scene["tint"]
    pixels = np.zeros((height, width, 3), dtype=np.uint8)
    pixels[:, :] = _tone(
        int(scene["background_value"]),
        [int(tint[channel]) + BACKGROUND_HUE[channel] for channel in range(3)],
    )

    for x, y, size in scene["clutter"]:
        pixels[int(y) : int(y) + int(size), int(x) : int(x) + int(size)] = CLUTTER_RGB

    border = int(scene["border_px"])
    for side in ("left", "right"):
        x, y = panel_origin(scene, side)
        pixels[y : y + PANEL_PX, x : x + PANEL_PX] = FRAME_RGB
        grid = left_grid(scene) if side == "left" else right_grid(scene)
        for row in range(GRID):
            for col in range(GRID):
                x0, y0, x1, y1 = cell_box(scene, side, row, col)
                value = int(scene["ink_value"]) if grid[row, col] else int(scene["pale_value"])
                pixels[y0:y1, x0:x1] = _tone(value, tint)
        assert border > 0

    hx0, hy0, hx1, hy1 = hinge_box(scene)
    pixels[hy0:hy1, hx0:hx1] = HINGE_RGB
    return Image.fromarray(pixels, mode="RGB")


def scene_json(scene: dict) -> str:  # pragma: no cover - convenience helper
    return json.dumps(scene, sort_keys=True)
