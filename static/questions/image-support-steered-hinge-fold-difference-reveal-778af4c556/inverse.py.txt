"""Pixel-only inverse arm for the hinge-fold difference reveal world.

The arm receives a single PIL image and nothing else.  It locates the two panel
frames and the hinge bar as visible fiducials, reads each panel's five-by-five
two-tone cell grid at the rendered tone midpoint, executes the declared fold
(mirror the hinged panel, superimpose, keep cells dark in exactly one layer)
and classifies the resulting grid against the declared symbol alphabet with a
reported Hamming margin.  Ambiguity raises ValueError, which is an abstention.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "hinge_fold_difference_reveal-oracle-0.1.0"

GRID = 5
FRAME_RGB = (28, 58, 168)
HINGE_RGB = (232, 122, 26)

SYMBOL_ART = {
    "plus": ("..#..", "..#..", "#####", "..#..", "..#.."),
    "ex": ("#...#", ".#.#.", "..#..", ".#.#.", "#...#"),
    "tee": ("#####", "..#..", "..#..", "..#..", "..#.."),
    "ell": ("#....", "#....", "#....", "#....", "#####"),
    "seven": ("#####", "...#.", "..#..", ".#...", "#...."),
}
TEMPLATES = {
    name: np.array([[1 if ch == "#" else 0 for ch in row] for row in art], dtype=np.int8)
    for name, art in SYMBOL_ART.items()
}


def _exact_mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.array(rgb, dtype=np.int16), axis=-1)


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    runs: list[tuple[int, int]] = []
    start = None
    for index, flag in enumerate(flags):
        if flag and start is None:
            start = index
        elif not flag and start is not None:
            runs.append((start, index))
            start = None
    if start is not None:
        runs.append((start, len(flags)))
    return runs


def _panel_boxes(frame: np.ndarray) -> list[tuple[int, int, int, int]]:
    columns = frame.any(axis=0)
    column_runs = [run for run in _runs(columns) if run[1] - run[0] >= 40]
    if len(column_runs) != 2:
        raise ValueError(f"expected two panel frames, found {len(column_runs)}")
    boxes = []
    for x0, x1 in column_runs:
        rows = frame[:, x0:x1].any(axis=1)
        row_runs = [run for run in _runs(rows) if run[1] - run[0] >= 40]
        if len(row_runs) != 1:
            raise ValueError("panel frame is not a single vertical block")
        y0, y1 = row_runs[0]
        side = x1 - x0
        if abs(side - (y1 - y0)) > 2 or side < 60:
            raise ValueError("panel frame is not square")
        boxes.append((x0, y0, x1, y1))
    return sorted(boxes, key=lambda box: box[0])


def _border_thickness(frame: np.ndarray, box: tuple[int, int, int, int]) -> int:
    x0, y0, x1, y1 = box
    middle = (y0 + y1) // 2
    strip = frame[middle, x0:x1]
    runs = _runs(strip)
    if len(runs) != 2:
        raise ValueError("panel frame is not a hollow rectangle")
    thickness = runs[0][1] - runs[0][0]
    if thickness < 1 or thickness > 8 or (runs[1][1] - runs[1][0]) != thickness:
        raise ValueError("panel frame thickness is ambiguous")
    return thickness


def _cell_values(arr: np.ndarray, box: tuple[int, int, int, int], thickness: int) -> np.ndarray:
    x0, y0, x1, y1 = box
    inner_x0, inner_y0 = x0 + thickness, y0 + thickness
    span = (x1 - thickness) - inner_x0
    if span != (y1 - thickness) - inner_y0 or span % GRID != 0:
        raise ValueError("panel interior is not a clean five-by-five lattice")
    cell = span // GRID
    inset = max(3, cell // 5)
    values = np.zeros((GRID, GRID), dtype=np.float64)
    for row in range(GRID):
        for col in range(GRID):
            cy0 = inner_y0 + row * cell + inset
            cx0 = inner_x0 + col * cell + inset
            patch = arr[cy0 : cy0 + cell - 2 * inset, cx0 : cx0 + cell - 2 * inset]
            if patch.size == 0:
                raise ValueError("empty cell patch")
            luma = patch @ np.array([0.299, 0.587, 0.114])
            if float(luma.max() - luma.min()) > 2.5:
                raise ValueError("cell interior is not a flat rendered tone")
            channels = patch.reshape(-1, 3).mean(axis=0)
            if float(channels.max() - channels.min()) > 22.0:
                raise ValueError("cell interior is not a near-neutral rendered cell tone")
            values[row, col] = float(luma.mean())
    return values


def _binarize(values: np.ndarray) -> np.ndarray:
    flat = values.reshape(-1)
    low, high = float(flat.min()), float(flat.max())
    if high - low < 24.0:
        raise ValueError("panel does not show two separated tones")
    mid = (low + high) / 2.0
    slack = (high - low) * 0.28
    if np.any((np.abs(flat - low) > slack) & (np.abs(flat - high) > slack)):
        raise ValueError("a cell tone is neither the rendered pale nor the rendered ink tone")
    return (values < mid).astype(np.int8)


def _hinged_side(arr: np.ndarray, boxes: list[tuple[int, int, int, int]]) -> str:
    mask = _exact_mask(arr, HINGE_RGB)
    if int(mask.sum()) < 200:
        raise ValueError("no hinge bar found")
    ys, xs = np.nonzero(mask)
    hx0, hx1 = int(xs.min()), int(xs.max()) + 1
    hy0, hy1 = int(ys.min()), int(ys.max()) + 1
    if int(mask.sum()) != (hx1 - hx0) * (hy1 - hy0):
        raise ValueError("hinge bar is not a solid rectangle")
    if (hy1 - hy0) <= (hx1 - hx0):
        raise ValueError("hinge bar is not an upright spine")
    left, right = boxes
    gap_right = left[0] - hx1  # hinge sits left of the left panel -> not adjacent inside
    del gap_right
    to_left_inner = abs(hx0 - left[2])
    to_right_inner = abs(right[0] - hx1)
    if to_left_inner <= 6 and to_right_inner > 6:
        return "left"
    if to_right_inner <= 6 and to_left_inner > 6:
        return "right"
    raise ValueError("hinge bar is not adjacent to exactly one panel's inner edge")


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    frame = _exact_mask(arr, FRAME_RGB)
    if int(frame.sum()) < 400:
        raise ValueError("no panel frames found")
    boxes = _panel_boxes(frame)
    hinged = _hinged_side(arr, boxes)

    grids: dict[str, np.ndarray] = {}
    for side, box in zip(("left", "right"), boxes):
        thickness = _border_thickness(frame, box)
        grids[side] = _binarize(_cell_values(arr, box, thickness))

    other = "right" if hinged == "left" else "left"
    folded = grids[hinged][:, ::-1]
    revealed = np.bitwise_xor(grids[other], folded)

    scored = sorted(
        (int(np.count_nonzero(revealed != tpl)), name) for name, tpl in TEMPLATES.items()
    )
    best, runner_up = scored[0], scored[1]
    if best[0] != 0 or runner_up[0] - best[0] < 4:
        raise ValueError(
            f"revealed grid matches no declared symbol (best={best}, next={runner_up})"
        )
    return best[1]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Convenience diagnostic: the decision plus its runner-up Hamming margin."""

    decision = decision_from_image(image)
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    frame = _exact_mask(arr, FRAME_RGB)
    boxes = _panel_boxes(frame)
    thickness = _border_thickness(frame, boxes[0])
    values = _cell_values(arr, boxes[0], thickness)
    flat = values.reshape(-1)
    return decision, float((flat.max() - flat.min()) / 2.0)
