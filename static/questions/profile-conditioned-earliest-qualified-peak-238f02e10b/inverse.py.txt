"""Pixel-only inverse arm for earliest_qualified_peak.

Receives one rasterized panel and nothing else: no latent parameters, no
analytic gold, no renderer, prompt, generator or verifier code.  It recovers
the dashed reference row, segments each coloured trace, locates each trace's
highest drawn point, applies the qualification rule and returns the colour of
the leftmost qualifying peak, abstaining when the evidence is not clean.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

TRACE_RGB = {
    "red": (214, 32, 32),
    "green": (18, 146, 44),
    "blue": (28, 62, 214),
    "orange": (240, 142, 8),
}
CHANNEL_TOLERANCE = 34
DARK_LEVEL = 60
MIN_DASH_FRACTION = 0.20
MIN_TRACE_PIXELS = 200
QUALIFY_GUARD = 2.5
MIN_PEAK_GAP_PX = 25
ABSTAIN = "abstain"


def _reference_row(pixels: np.ndarray) -> float | None:
    dark = pixels.max(axis=2) < DARK_LEVEL
    counts = dark.sum(axis=1)
    strongest = int(counts.max()) if counts.size else 0
    if strongest < MIN_DASH_FRACTION * pixels.shape[1]:
        return None
    rows = np.flatnonzero(counts >= 0.5 * strongest)
    if rows.size == 0 or rows.size > 6 or int(rows[-1] - rows[0]) != rows.size - 1:
        return None
    return float(rows.mean())


def _highest_point(pixels: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float] | None:
    target = np.array(rgb, dtype=np.int16)
    mask = (np.abs(pixels - target) <= CHANNEL_TOLERANCE).all(axis=2)
    if int(mask.sum()) < MIN_TRACE_PIXELS:
        return None
    rows = np.flatnonzero(mask.any(axis=1))
    top = int(rows[0])
    band = mask[top : top + 3, :]
    columns = np.flatnonzero(band.any(axis=0))
    if columns.size == 0:
        return None
    return float(top), float(columns.mean())


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    reference = _reference_row(pixels)
    if reference is None:
        return ABSTAIN
    qualified: list[tuple[float, str]] = []
    for name, rgb in sorted(TRACE_RGB.items()):
        found = _highest_point(pixels, rgb)
        if found is None:
            return ABSTAIN
        top, column = found
        if top < reference - QUALIFY_GUARD:
            qualified.append((column, name))
    if len(qualified) < 2:
        return ABSTAIN
    qualified.sort()
    if qualified[1][0] - qualified[0][0] < MIN_PEAK_GAP_PX:
        return ABSTAIN
    return qualified[0][1]
