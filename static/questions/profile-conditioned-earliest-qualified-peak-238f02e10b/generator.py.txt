"""earliest_qualified_peak: scene sampling, rendering and analytic gold.

The panel shows five piecewise-linear traces (four coloured, one grey
distractor) whose vertices sit on nine pale vertical gridlines, plus one black
dashed horizontal line.  A coloured trace QUALIFIES when its highest vertex
lies above the dashed line; the decision is the colour of the qualifying trace
whose highest vertex sits furthest to the left.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 560
PANEL_LEFT, PANEL_TOP, PANEL_RIGHT, PANEL_BOTTOM = 60, 40, 860, 500
N_STATIONS = 9
STATIONS = [90 + 90 * index for index in range(N_STATIONS)]
ROW_MIN, ROW_MAX = 72, 468

COLOR_NAMES = ("red", "green", "blue", "orange")
COLOR_RGB = {
    "red": (214, 32, 32),
    "green": (18, 146, 44),
    "blue": (28, 62, 214),
    "orange": (240, 142, 8),
}
GREY_RGB = (150, 150, 150)
GRID_RGB = (226, 226, 226)
FRAME_RGB = (110, 110, 110)
LINE_RGB = (0, 0, 0)
BACKGROUND = (255, 255, 255)

STROKE = 3
DASH_ON, DASH_OFF = 12, 8
MIN_CLEARANCE = 6          # px between a peak vertex and the dashed line
MIN_PEAK_DROP = 12         # px between a trace's peak and its own runner-up vertex
MIN_SEP_AT_PEAK = 9        # px of vertical room around every peak (anti-occlusion)
PEAK_WINDOW = 26           # px half-width of the protected neighbourhood of a peak
QUARANTINE_MARGIN = 12.0
MIN_CROSSINGS = 2


def _peak_index(rows: list[int]) -> int:
    """Index of the single highest (smallest row) vertex of one trace."""

    best = 0
    for index in range(1, len(rows)):
        if rows[index] < rows[best]:
            best = index
    return best


def _value_at(rows: list[int], xs: list[int], x: float) -> float:
    """Linear interpolation of a trace between its station vertices."""

    if x <= xs[0]:
        return float(rows[0])
    if x >= xs[-1]:
        return float(rows[-1])
    for index in range(1, len(xs)):
        if x <= xs[index]:
            span = xs[index] - xs[index - 1]
            frac = (x - xs[index - 1]) / span
            return rows[index - 1] + frac * (rows[index] - rows[index - 1])
    return float(rows[-1])


def _peak_is_clear(rows: list[int], others: list[list[int]], xs: list[int]) -> bool:
    """True when no other trace runs within MIN_SEP_AT_PEAK of this peak."""

    peak = _peak_index(rows)
    centre = xs[peak]
    samples = [centre - PEAK_WINDOW + step * (PEAK_WINDOW // 4) for step in range(9)]
    for other in others:
        for x in samples:
            if abs(_value_at(rows, xs, x) - _value_at(other, xs, x)) < MIN_SEP_AT_PEAK:
                return False
    return True


def _crossing_count(traces: list[list[int]]) -> int:
    total = 0
    for first in range(len(traces)):
        for second in range(first + 1, len(traces)):
            signs = [
                1 if a < b else -1
                for a, b in zip(traces[first], traces[second])
                if a != b
            ]
            total += sum(1 for a, b in zip(signs, signs[1:]) if a != b)
    return total


def _clearance(rng: random.Random) -> int:
    if rng.random() < 0.30:
        return rng.randint(MIN_CLEARANCE, 14)
    return rng.randint(16, 78)


def _draw_trace(canvas: ImageDraw.ImageDraw, rows, xs, rgb) -> None:
    points = [(x, row) for x, row in zip(xs, rows)]
    canvas.line(points, fill=rgb, width=STROKE, joint="curve")


def sample_scene(seed: int) -> dict:
    """Sample one balanced, non-degenerate scene."""

    rng = random.Random((int(seed) * 100_003) ^ 0x5EED)
    for _ in range(400):
        line_row = rng.randint(196, 356)
        left_disqualified = rng.randint(0, 2)
        qualifying = rng.randint(2, 4 - left_disqualified)
        peak_slots = sorted(rng.sample(range(N_STATIONS), 4))

        flags = [False] * 4
        flags[left_disqualified] = True
        remaining = list(range(left_disqualified + 1, 4))
        rng.shuffle(remaining)
        for rank in remaining[: qualifying - 1]:
            flags[rank] = True

        # Counterbalance the "answer is simply the highest peak" shortcut: the
        # winner is the tallest qualifying trace only about a quarter of the time.
        clearances = [_clearance(rng) for _ in range(4)]
        qualifier_ranks = [rank for rank in range(4) if flags[rank]]
        pool = sorted((clearances[rank] for rank in qualifier_ranks), reverse=True)
        if len(set(pool)) != len(pool):
            continue
        winner_rank = qualifier_ranks[0]
        chosen = pool[0] if rng.random() < 0.25 else rng.choice(pool[1:])
        pool.remove(chosen)
        rng.shuffle(pool)
        clearances[winner_rank] = chosen
        for rank, value in zip(qualifier_ranks[1:], pool):
            clearances[rank] = value

        trace_rows: list[list[int]] = []
        ok = True
        for rank in range(4):
            clearance = clearances[rank]
            peak_row = line_row - clearance if flags[rank] else line_row + clearance
            if not ROW_MIN <= peak_row <= ROW_MAX - MIN_PEAK_DROP:
                ok = False
                break
            low = peak_row + MIN_PEAK_DROP
            high = min(ROW_MAX, peak_row + 190)
            if high - low < 30:
                ok = False
                break
            rows = [rng.randint(low, high) for _ in range(N_STATIONS)]
            rows[peak_slots[rank]] = peak_row
            trace_rows.append(rows)
        if not ok:
            continue

        grey_peak_slot = rng.randrange(N_STATIONS)
        grey_peak_row = rng.randint(ROW_MIN, ROW_MAX - 120)
        if rng.random() < 0.5 and peak_slots[left_disqualified] > 0:
            grey_peak_slot = rng.randrange(peak_slots[left_disqualified])
            grey_peak_row = line_row - rng.randint(12, 70)
            if grey_peak_row < ROW_MIN:
                continue
        grey_rows = [
            rng.randint(grey_peak_row + MIN_PEAK_DROP, min(ROW_MAX, grey_peak_row + 190))
            for _ in range(N_STATIONS)
        ]
        grey_rows[grey_peak_slot] = grey_peak_row

        every = trace_rows + [grey_rows]
        if any(
            not _peak_is_clear(rows, [other for other in every if other is not rows], STATIONS)
            for rows in every
        ):
            continue
        if _crossing_count(trace_rows) < MIN_CROSSINGS:
            continue

        trace_colors = list(COLOR_NAMES)
        rng.shuffle(trace_colors)
        draw_order = trace_colors + ["grey"]
        rng.shuffle(draw_order)
        scene = {
            "stations": list(STATIONS),
            "line_row": line_row,
            "trace_rows": trace_rows,
            "trace_colors": trace_colors,
            "grey_rows": grey_rows,
            "draw_order": draw_order,
        }
        if margin(scene) < float(MIN_CLEARANCE):
            continue
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


def render(scene: dict) -> Image.Image:
    """Rasterize one scene with hard-edged, unantialiased marks."""

    xs = scene["stations"]
    line_row = scene["line_row"]
    trace_rows = scene["trace_rows"]
    trace_colors = scene["trace_colors"]
    grey_rows = scene["grey_rows"]
    draw_order = scene["draw_order"]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    canvas = ImageDraw.Draw(image)
    for x in xs:
        canvas.line([(x, PANEL_TOP + 1), (x, PANEL_BOTTOM - 1)], fill=GRID_RGB, width=1)
    canvas.rectangle(
        [PANEL_LEFT, PANEL_TOP, PANEL_RIGHT, PANEL_BOTTOM], outline=FRAME_RGB, width=1
    )
    x = PANEL_LEFT + 6
    while x + DASH_ON <= PANEL_RIGHT - 6:
        canvas.rectangle([x, line_row, x + DASH_ON - 1, line_row + 1], fill=LINE_RGB)
        x += DASH_ON + DASH_OFF
    for name in draw_order:
        if name == "grey":
            _draw_trace(canvas, grey_rows, xs, GREY_RGB)
        else:
            _draw_trace(canvas, trace_rows[trace_colors.index(name)], xs, COLOR_RGB[name])
    return image


def analytic_gold(scene: dict) -> str:
    """Colour of the qualifying trace whose highest vertex is furthest left."""

    xs = scene["stations"]
    line_row = scene["line_row"]
    trace_rows = scene["trace_rows"]
    trace_colors = scene["trace_colors"]
    best_x = None
    best_color = ""
    for index in range(len(trace_rows)):
        rows = trace_rows[index]
        slot = _peak_index(rows)
        if rows[slot] < line_row:
            x = xs[slot]
            if best_x is None or x < best_x:
                best_x = x
                best_color = trace_colors[index]
    return best_color


def _peak_table(scene: dict) -> list[tuple[int, int, bool]]:
    xs = scene["stations"]
    line_row = scene["line_row"]
    table = []
    for rows in scene["trace_rows"]:
        slot = _peak_index(rows)
        table.append((xs[slot], abs(rows[slot] - line_row), rows[slot] < line_row))
    return table


def margin(scene: dict) -> float:
    """Smallest vertical move, in pixels, that would change the answer.

    Only two events can change the decision: the winner dropping below the
    dashed line, or a disqualified trace whose peak is further left than the
    winner's rising above it.  Traces to the right of the winner cannot.
    """

    table = _peak_table(scene)
    winners = [entry for entry in table if entry[2]]
    winner_x = min(entry[0] for entry in winners)
    relevant = [entry[1] for entry in table if entry[2] and entry[0] == winner_x]
    relevant += [entry[1] for entry in table if not entry[2] and entry[0] < winner_x]
    return float(min(relevant))


def is_quarantined(scene: dict) -> bool:
    """Near-boundary scenes are flagged and excluded from headline claims."""

    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist; see candidate.json latent_alias.justification."""

    return []
