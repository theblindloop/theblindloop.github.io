"""Independent pixel-only inverse for the rotated lattice world."""

from __future__ import annotations

from collections import deque

import numpy as np


def _component_centers(mask: np.ndarray, minimum: int) -> list[tuple[float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    centers: list[tuple[float, float]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            centers.append((float(np.mean(xs)), float(np.mean(ys))))
    return centers


def _cluster_levels(values: np.ndarray, tolerance: float = 11.0) -> np.ndarray:
    ordered = sorted(float(value) for value in values)
    groups: list[list[float]] = []
    for value in ordered:
        if not groups or value - float(np.mean(groups[-1])) > tolerance:
            groups.append([value])
        else:
            groups[-1].append(value)
    return np.array([float(np.mean(group)) for group in groups], dtype=float)


def _missing_intersection(points: np.ndarray) -> tuple[np.ndarray, float]:
    center = points.mean(axis=0)
    covariance = np.cov((points - center).T)
    values, vectors = np.linalg.eigh(covariance)
    axes = vectors[:, np.argsort(values)[::-1]]
    projected = (points - center) @ axes
    first = _cluster_levels(projected[:, 0])
    second = _cluster_levels(projected[:, 1])
    if sorted((len(first), len(second))) != [5, 6]:
        raise ValueError("dot field does not resolve into a 5 by 6 lattice")
    candidates: list[tuple[float, np.ndarray]] = []
    for a in first:
        for b in second:
            position = center + np.array([a, b]) @ axes.T
            nearest = float(np.min(np.linalg.norm(points - position, axis=1)))
            candidates.append((nearest, position))
    candidates.sort(key=lambda item: item[0], reverse=True)
    if candidates[0][0] < 20.0 or candidates[0][0] - candidates[1][0] < 12.0:
        raise ValueError("no unique missing lattice intersection")
    spacings = []
    for levels in (first, second):
        spacings.extend(np.diff(np.sort(levels)).tolist())
    return candidates[0][1], float(np.median(spacings))


def decision_from_image(image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = (rgb[:, :, 0] < 80) & (rgb[:, :, 1] < 80) & (rgb[:, :, 2] < 80)
    red = (rgb[:, :, 0] > 150) & (rgb[:, :, 1] < 120) & (rgb[:, :, 2] < 140)
    blue = (rgb[:, :, 0] < 110) & (rgb[:, :, 1] < 165) & (rgb[:, :, 2] > 150)
    dots = _component_centers(black, minimum=60)
    red_centers = _component_centers(red, minimum=35)
    blue_centers = _component_centers(blue, minimum=35)
    if len(dots) != 29 or len(red_centers) != 1 or len(blue_centers) != 1:
        raise ValueError("required dot and ring evidence is absent")
    missing, spacing = _missing_intersection(np.array(dots, dtype=float))
    red_distance = float(np.linalg.norm(np.array(red_centers[0]) - missing)) / spacing
    blue_distance = float(np.linalg.norm(np.array(blue_centers[0]) - missing)) / spacing
    if abs(red_distance - blue_distance) < 0.20:
        raise ValueError("ambiguous ring completion residuals")
    return "red" if red_distance < blue_distance else "blue"
