"""Deterministic Pillow world for global lattice completion."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
INK = (27, 32, 39)
RED = (218, 55, 72)
BLUE = (40, 111, 218)
FRAME = (216, 221, 227)


def _axes(scene: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    angle = math.radians(float(scene["angle_deg"]))
    return (math.cos(angle), math.sin(angle)), (-math.sin(angle), math.cos(angle))


def _site(scene: dict, row: int, col: int) -> tuple[float, float]:
    u, v = _axes(scene)
    ox, oy = scene["origin"]
    return (
        ox + col * float(scene["spacing_x"]) * u[0] + row * float(scene["spacing_y"]) * v[0],
        oy + col * float(scene["spacing_x"]) * u[1] + row * float(scene["spacing_y"]) * v[1],
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    rows = 5
    cols = 6
    spacing_x = rng.randint(55, 61)
    spacing_y = rng.randint(48, 53)
    angle_deg = rng.choice([-27, -22, -17, 16, 21, 26]) + rng.uniform(-1.2, 1.2)
    angle = math.radians(angle_deg)
    u = (math.cos(angle), math.sin(angle))
    v = (-math.sin(angle), math.cos(angle))
    center = (256 + rng.randint(-12, 12), 256 + rng.randint(-12, 12))
    origin = (
        center[0] - 0.5 * (cols - 1) * spacing_x * u[0] - 0.5 * (rows - 1) * spacing_y * v[0],
        center[1] - 0.5 * (cols - 1) * spacing_x * u[1] - 0.5 * (rows - 1) * spacing_y * v[1],
    )
    missing_rc = [rng.randrange(1, rows - 1), rng.randrange(1, cols - 1)]
    base = {
        "rows": rows,
        "cols": cols,
        "spacing_x": spacing_x,
        "spacing_y": spacing_y,
        "angle_deg": angle_deg,
        "origin": [origin[0], origin[1]],
        "missing_rc": missing_rc,
        "dot_radius": rng.choice([6, 7]),
        "ring_radius": 10,
    }
    target = _site(base, missing_rc[0], missing_rc[1])
    phi = rng.uniform(0.0, 2.0 * math.pi)
    # The distractor remains near the missing cell, so local empty-space size is
    # weak; only agreement with the complete row/column system is decisive.
    offset_length = min(spacing_x, spacing_y) * rng.uniform(0.48, 0.68)
    distractor = (target[0] + offset_length * math.cos(phi), target[1] + offset_length * math.sin(phi))
    if seed % 2:
        base["red_center"] = [target[0], target[1]]
        base["blue_center"] = [distractor[0], distractor[1]]
    else:
        base["red_center"] = [distractor[0], distractor[1]]
        base["blue_center"] = [target[0], target[1]]
    return base


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 247))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((20, 20, WIDTH - 21, HEIGHT - 21), radius=18, outline=FRAME, width=2)
    rows = int(scene["rows"])
    cols = int(scene["cols"])
    missing = tuple(int(value) for value in scene["missing_rc"])
    radius = int(scene["dot_radius"])
    for row in range(rows):
        for col in range(cols):
            if (row, col) == missing:
                continue
            x, y = _site(scene, row, col)
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=INK)
    ring_radius = int(scene["ring_radius"])
    for center, color in ((scene["red_center"], RED), (scene["blue_center"], BLUE)):
        x, y = center
        draw.ellipse(
            (x - ring_radius, y - ring_radius, x + ring_radius, y + ring_radius),
            outline=color,
            width=4,
        )
    return image


def _completion_target(scene: dict) -> tuple[float, float]:
    row, col = scene["missing_rc"]
    angle = math.radians(float(scene["angle_deg"]))
    ux, uy = math.cos(angle), math.sin(angle)
    vx, vy = -uy, ux
    ox, oy = scene["origin"]
    return (
        ox + col * float(scene["spacing_x"]) * ux + row * float(scene["spacing_y"]) * vx,
        oy + col * float(scene["spacing_x"]) * uy + row * float(scene["spacing_y"]) * vy,
    )


def _ring_residuals(scene: dict) -> tuple[float, float]:
    tx, ty = _completion_target(scene)
    rx, ry = scene["red_center"]
    bx, by = scene["blue_center"]
    return math.hypot(rx - tx, ry - ty), math.hypot(bx - tx, by - ty)


def analytic_gold(scene: dict) -> str:
    red_residual, blue_residual = _ring_residuals(scene)
    return "red" if red_residual < blue_residual else "blue"


def margin(scene: dict) -> float:
    red_residual, blue_residual = _ring_residuals(scene)
    return abs(red_residual - blue_residual) / min(float(scene["spacing_x"]), float(scene["spacing_y"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.25


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
