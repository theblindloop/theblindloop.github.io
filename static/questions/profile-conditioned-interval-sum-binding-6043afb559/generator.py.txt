"""interval_sum_binding — three groups of floating colored range bars.

Every group shows the same three colors exactly once, drawn as a *floating*
rectangle: the rectangle's vertical position carries no meaning, only its
length does. Beside a single shared axis a ladder of tick marks fixes the unit
(consecutive ticks are one unit apart). The decision is

    winner = argmax_color  sum over the three groups of interval_length_units

so the reader must bind three separated marks by color, measure each one's
extent rather than its endpoint, and aggregate before comparing. The sampler
counterbalances the three cheap shortcuts that could otherwise answer it:
the color owning the single longest interval, the color reaching the highest
top edge, and the color covering the most painted area (widths vary).

Quantization note (this world's only latent alias): an interval is drawn with
an integer pixel length h = round(length * px_per_unit) and an integer bottom
row, so every length inside one rounding bucket produces a byte-identical
raster. Gold is therefore defined on the quantized effective lengths, and
``latent_symmetries`` returns explicit sub-pixel perturbations that leave both
the raster and the answer alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "interval_sum_binding-1.0.0"

# --- fixed page geometry (integer coordinates only; no antialiasing anywhere) --
WIDTH = 760
HEIGHT = 462
Y_BASE = 400              # the zero tick row, before the global offset
AXIS_X = 70               # left column of the 3px shared vertical axis
TICK_OUTER = 18           # tick spans x in [AXIS_X - TICK_OUTER, AXIS_X - TICK_INNER]
TICK_INNER = 5
GROUP_X = (200, 410, 620)  # group centers, before the global offset
SLOTS = (-62, 0, 62)       # within-group slot offsets
LEGEND_X = (486, 546, 606)
LEGEND_Y = 22
LEGEND_SIDE = 11           # deliberately shorter than any interval
INK = (20, 20, 20)
BACKGROUND = (255, 255, 255)

COLOR_NAMES = ("red", "green", "blue")
COLOR_RGB = {
    "red": (214, 39, 40),
    "green": (44, 160, 44),
    "blue": (31, 119, 180),
}

QUARANTINE_UNITS = 0.50   # a top-two total gap below this is a boundary case
BANDS = {
    "wide": (1.00, 3.60),
    "clear": (0.62, 1.00),
    "ambiguous": (0.03, 0.34),
}
MIN_LEN = 1.00
MAX_LEN = 3.30
WIDTH_CHOICES = (7, 9, 11, 13, 15, 17)


# --- quantization helpers --------------------------------------------------------


def _quantized_lengths(scene: dict) -> list[list[int]]:
    """Pixel lengths actually painted: h = round(length * px_per_unit)."""
    ppu = scene["px_per_unit"]
    return [[int(round(value * ppu)) for value in row] for row in scene["lengths_units"]]


def _bottom_rows(scene: dict) -> list[list[int]]:
    ppu = scene["px_per_unit"]
    return [[int(round(value * ppu)) for value in row] for row in scene["bottoms_units"]]


def _totals_px(scene: dict) -> list[int]:
    return [sum(row) for row in _quantized_lengths(scene)]


def _cues(scene: dict) -> dict:
    """The three cheap shortcuts, computed on what is actually painted."""
    lengths = _quantized_lengths(scene)
    bottoms = _bottom_rows(scene)
    widths = scene["widths_px"]
    return {
        "longest_single": [max(row) for row in lengths],
        "top_edge": [
            max(bottoms[c][g] + lengths[c][g] for g in range(3)) for c in range(3)
        ],
        "painted_area": [
            sum(lengths[c][g] * widths[c][g] for g in range(3)) for c in range(3)
        ],
    }


def _argmax_gap(values: list[int]) -> tuple[int, int]:
    order = sorted(range(len(values)), key=lambda i: values[i], reverse=True)
    return order[0], values[order[0]] - values[order[1]]


# --- scene sampling ---------------------------------------------------------------


def sample_scene(seed: int) -> dict:
    """Deterministic, counterbalanced scene sampler."""
    rng = random.Random(seed)
    ppu = rng.randint(28, 34)
    n_ticks = rng.randint(9, 11)
    ceiling = (n_ticks - 1) - 0.25          # highest usable unit level
    target = rng.randrange(3)
    band = "ambiguous" if rng.random() < 0.18 else ("wide" if rng.random() < 0.6 else "clear")
    low, high = BANDS[band]
    want = {
        "longest_single": rng.random() < 0.5,
        "top_edge": rng.random() < 0.5,
        "painted_area": rng.random() < 0.5,
    }

    required = ["longest_single", "top_edge", "painted_area"]
    scene: dict | None = None
    for attempt in range(9000):
        if attempt == 6000:               # graceful relaxation, still deterministic
            required = ["longest_single", "painted_area"]
        if attempt == 8000:
            required = ["longest_single"]
        draw_lengths = [
            [round(rng.uniform(MIN_LEN, MAX_LEN), 2) for _ in range(3)] for _ in range(3)
        ]
        # place the intended winner's row on the color the sampler picked
        order = sorted(range(3), key=lambda c: sum(draw_lengths[c]), reverse=True)
        rows: list[list[float]] = [[], [], []]
        rows[target] = draw_lengths[order[0]]
        rest = [c for c in range(3) if c != target]
        rng.shuffle(rest)
        rows[rest[0]] = draw_lengths[order[1]]
        rows[rest[1]] = draw_lengths[order[2]]

        widths = [[rng.choice(WIDTH_CHOICES) for _ in range(3)] for _ in range(3)]
        bottoms = [
            [
                round(rng.uniform(0.15, max(0.25, ceiling - rows[c][g])), 2)
                for g in range(3)
            ]
            for c in range(3)
        ]
        probe = {
            "lengths_units": rows,
            "bottoms_units": bottoms,
            "widths_px": widths,
            "px_per_unit": ppu,
        }
        best, gap = _argmax_gap(_totals_px(probe))
        if best != target or not (low <= gap / ppu <= high):
            continue
        cues = _cues(probe)
        if any(
            _argmax_gap(cues[name])[1] <= 0
            or (_argmax_gap(cues[name])[0] == target) != want[name]
            for name in required
        ):
            continue
        scene = {
            "lengths_units": rows,
            "bottoms_units": bottoms,
            "widths_px": widths,
            "slot_order": [rng.sample(range(3), 3) for _ in range(3)],
            "legend_order": rng.sample(range(3), 3),
            "px_per_unit": ppu,
            "n_ticks": n_ticks,
            "offset": [rng.randint(-6, 6), rng.randint(-5, 5)],
        }
        break

    if scene is None:                      # never observed on the public seeds
        raise RuntimeError(f"sampler failed to satisfy the counterbalance for seed {seed}")
    return scene


# --- rendering ---------------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dx, dy = scene["offset"]
    ppu = scene["px_per_unit"]
    base = Y_BASE + dy
    axis = AXIS_X + dx
    top_tick = base - (scene["n_ticks"] - 1) * ppu

    draw.rectangle([axis, top_tick, axis + 2, base], fill=INK)
    draw.rectangle([axis, base, WIDTH - 30 + dx, base + 2], fill=INK)
    for index in range(scene["n_ticks"]):
        row = base - index * ppu
        draw.rectangle([axis - TICK_OUTER, row, axis - TICK_INNER, row + 1], fill=INK)

    lengths = _quantized_lengths(scene)
    bottoms = _bottom_rows(scene)
    for group in range(3):
        center = GROUP_X[group] + dx
        for slot, color_index in enumerate(scene["slot_order"][group]):
            width = scene["widths_px"][color_index][group]
            height = lengths[color_index][group]
            bottom_row = base - bottoms[color_index][group]
            left = center + SLOTS[slot] - width // 2
            draw.rectangle(
                [left, bottom_row - height + 1, left + width - 1, bottom_row],
                fill=COLOR_RGB[COLOR_NAMES[color_index]],
            )
        draw.text((center - 8, base + 8), f"G{group + 1}", fill=INK)

    for position, color_index in enumerate(scene["legend_order"]):
        left = LEGEND_X[position] + dx
        top = LEGEND_Y + dy
        draw.rectangle(
            [left, top, left + LEGEND_SIDE - 1, top + LEGEND_SIDE - 1],
            fill=COLOR_RGB[COLOR_NAMES[color_index]],
        )
    draw.text((LEGEND_X[0] - 62 + dx, LEGEND_Y + dy), "key:", fill=INK)
    return image


# --- gold, margin, quarantine --------------------------------------------------------


def analytic_gold(scene: dict) -> str:
    """Color whose three painted interval lengths sum to the greatest total."""
    best, _ = _argmax_gap(_totals_px(scene))
    return COLOR_NAMES[best]


def margin(scene: dict) -> float:
    """Top-two gap of the color totals, in tick units."""
    _, gap = _argmax_gap(_totals_px(scene))
    return gap / scene["px_per_unit"]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_UNITS


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Sub-pixel perturbations: same raster, same gold, different latent scene.

    ``render`` quantizes each interval to h = round(length * px_per_unit) and each
    bottom offset the same way; any value inside one rounding bucket is invisible.
    These are the world's only aliases and they are exercised, not asserted.
    """
    ppu = scene["px_per_unit"]
    lengths = _quantized_lengths(scene)
    bottoms = _bottom_rows(scene)
    transforms: list[tuple[str, dict]] = []
    for color in range(3):
        for nudge, tag in ((0.25, "up"), (-0.25, "down")):
            twin = dict(scene)
            rows = [list(row) for row in scene["lengths_units"]]
            rows[color][color] = (lengths[color][color] + nudge) / ppu
            twin["lengths_units"] = rows
            transforms.append((f"subpixel_length_{COLOR_NAMES[color]}_{tag}", twin))
        twin = dict(scene)
        rows = [list(row) for row in scene["bottoms_units"]]
        rows[color][0] = (bottoms[color][0] + 0.25) / ppu
        twin["bottoms_units"] = rows
        transforms.append((f"subpixel_bottom_{COLOR_NAMES[color]}", twin))
    return transforms


# --- helpers used by tests, the generator and the diagnostics ---------------------------


def totals_units(scene: dict) -> list[float]:
    ppu = scene["px_per_unit"]
    return [total / ppu for total in _totals_px(scene)]


def totals_px(scene: dict) -> list[int]:
    return _totals_px(scene)


def cue_congruence(scene: dict) -> dict:
    """For each cheap shortcut, whether it happens to point at the answer."""
    gold = analytic_gold(scene)
    return {
        name: COLOR_NAMES[_argmax_gap(values)[0]] == gold
        for name, values in _cues(scene).items()
    }
