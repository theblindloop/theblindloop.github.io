"""Independent pixel-only inverse for marker constellations.

This module deliberately duplicates the public visual constants but does not
import the renderer or any analytic scene code.  It extracts teal connected
components, estimates their centroids, partitions them into the three visible
panels, applies the declared quarter-turn, and evaluates every point
assignment.
"""

from __future__ import annotations

import itertools
import math
from collections import deque

import numpy as np
from PIL import Image

_MARKER = np.asarray((27, 126, 154), dtype=np.float32)
_PANEL_CENTERS = ((220.0, 200.0), (580.0, 200.0), (400.0, 440.0))
_COLOR_RADIUS = 108.0
_MIN_COMPONENT = 180


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= _MIN_COMPONENT:
            found.append(np.asarray(pixels, dtype=np.float64))
    return found


def _marker_centroids(image: Image.Image) -> list[tuple[float, float]]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.linalg.norm(array - _MARKER[None, None, :], axis=2)
    centroids = []
    for pixels in _components(distance < _COLOR_RADIUS):
        # Pixel centers are sufficient; the same bias is present for all disks.
        y_mean, x_mean = pixels.mean(axis=0)
        centroids.append((float(x_mean), float(y_mean)))
    if len(centroids) != 9:
        raise ValueError(f"expected nine visible marker components, recovered {len(centroids)}")
    return centroids


def _group(centroids: list[tuple[float, float]]) -> list[list[tuple[float, float]]]:
    groups = [[] for _ in _PANEL_CENTERS]
    for point in centroids:
        index = min(range(3), key=lambda i: math.hypot(point[0] - _PANEL_CENTERS[i][0], point[1] - _PANEL_CENTERS[i][1]))
        groups[index].append(point)
    if any(len(group) != 3 for group in groups):
        raise ValueError("each panel must contain exactly three marker components")
    return groups


def _local(group: list[tuple[float, float]], center: tuple[float, float]) -> list[tuple[float, float]]:
    return [(x - center[0], y - center[1]) for x, y in group]


def _quarter_turn(points: list[tuple[float, float]]) -> list[tuple[float, float]]:
    return [(-y, x) for x, y in points]


def _assignment_error(reference: list[tuple[float, float]], candidate: list[tuple[float, float]]) -> float:
    scores = []
    for order in itertools.permutations(range(3)):
        scores.append(
            sum(
                math.hypot(reference[i][0] - candidate[order[i]][0], reference[i][1] - candidate[order[i]][1])
                for i in range(3)
            )
            / 3.0
        )
    return min(scores)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only the supplied raster and its public layout rule."""

    reference, candidate_a, candidate_b = _group(_marker_centroids(image))
    ref = _local(reference, _PANEL_CENTERS[0])
    a = _quarter_turn(_local(candidate_a, _PANEL_CENTERS[1]))
    b = _quarter_turn(_local(candidate_b, _PANEL_CENTERS[2]))
    error_a = _assignment_error(ref, a)
    error_b = _assignment_error(ref, b)
    if abs(error_a - error_b) < 1e-9:
        raise ValueError("candidate residuals are indistinguishable")
    return "A" if error_a < error_b else "B"
