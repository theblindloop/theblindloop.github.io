"""Renderer and analytic arm for quarter-turned marker constellations.

The renderer owns only latent coordinate records.  The inverse arm in
``oracle.py`` independently recovers the colored connected components from the
PNG and repeats the geometric assignment.
"""

from __future__ import annotations

import copy
import itertools
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 800
HEIGHT = 600
BACKGROUND = (255, 255, 255)
MARKER = (27, 126, 154)
FRAME = (205, 216, 220)
TEXT = (45, 57, 64)
PANEL_CENTERS = ((220.0, 200.0), (580.0, 200.0), (400.0, 440.0))
MARKER_RADIUS = 15.0
PANEL_HALF_WIDTH = 62.0
PANEL_HALF_HEIGHT = 70.0
QUARANTINE_GAP = 6.0


def _rotate_cw(point: tuple[float, float]) -> tuple[float, float]:
    """Rotate an image-coordinate offset clockwise by a quarter turn."""

    x, y = point
    return (-y, x)


def _rotate_ccw(point: tuple[float, float]) -> tuple[float, float]:
    x, y = point
    return (y, -x)


def _as_points(value: Iterable[Iterable[float]]) -> list[tuple[float, float]]:
    return [(float(item[0]), float(item[1])) for item in value]


def _local_points(points: Iterable[Iterable[float]], center: tuple[float, float]) -> list[tuple[float, float]]:
    cx, cy = center
    return [(x - cx, y - cy) for x, y in _as_points(points)]


def _assignment_error(
    reference: list[tuple[float, float]], candidate: list[tuple[float, float]]
) -> float:
    """Minimum mean Euclidean residual over the six three-point assignments."""

    if len(reference) != 3 or len(candidate) != 3:
        raise ValueError("each constellation must contain exactly three markers")
    errors = []
    for order in itertools.permutations(range(3)):
        total = sum(
            math.hypot(reference[i][0] - candidate[order[i]][0], reference[i][1] - candidate[order[i]][1])
            for i in range(3)
        )
        errors.append(total / 3.0)
    return min(errors)


def _shape_sample(rng: random.Random) -> list[tuple[float, float]]:
    """Draw a broad, separated scalene triangle around the panel center."""

    for _ in range(800):
        raw = [(rng.uniform(-24.0, 24.0), rng.uniform(-24.0, 24.0)) for _ in range(3)]
        mx = sum(x for x, _ in raw) / 3.0
        my = sum(y for _, y in raw) / 3.0
        centered = [(x - mx, y - my) for x, y in raw]
        theta = rng.uniform(-math.pi, math.pi)
        cs, sn = math.cos(theta), math.sin(theta)
        rotated = [(x * cs - y * sn, x * sn + y * cs) for x, y in centered]
        if any(abs(x) > 31.0 or abs(y) > 31.0 for x, y in rotated):
            continue
        gaps = [
            math.hypot(rotated[i][0] - rotated[j][0], rotated[i][1] - rotated[j][1])
            for i, j in ((0, 1), (0, 2), (1, 2))
        ]
        if min(gaps) >= 34.0 and max(gaps) <= 70.0:
            return rotated
    raise RuntimeError("could not sample a separated constellation")


def _perturbation(
    rng: random.Random, exact: list[tuple[float, float]]
) -> list[tuple[float, float]]:
    """Move two records in opposite directions, preserving centroid and ink."""

    for _ in range(500):
        i, j = rng.sample(range(3), 2)
        angle = rng.uniform(-math.pi, math.pi)
        length = rng.uniform(16.0, 22.0)
        dx, dy = length * math.cos(angle), length * math.sin(angle)
        trial = list(exact)
        trial[i] = (trial[i][0] + dx, trial[i][1] + dy)
        trial[j] = (trial[j][0] - dx, trial[j][1] - dy)
        if any(abs(x) > 39.0 or abs(y) > 39.0 for x, y in trial):
            continue
        gaps = [
            math.hypot(trial[a][0] - trial[b][0], trial[a][1] - trial[b][1])
            for a, b in ((0, 1), (0, 2), (1, 2))
        ]
        if min(gaps) >= 32.0 and max(gaps) <= 82.0:
            return trial
    raise RuntimeError("could not sample a separated perturbation")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_local = _shape_sample(rng)
    ref_center = PANEL_CENTERS[0]
    reference = [(ref_center[0] + x, ref_center[1] + y) for x, y in reference_local]
    exact_local = [_rotate_ccw(point) for point in reference_local]
    exact_center_a = PANEL_CENTERS[1]
    exact_a = [(exact_center_a[0] + x, exact_center_a[1] + y) for x, y in exact_local]
    exact_center_b = PANEL_CENTERS[2]
    exact_b = [(exact_center_b[0] + x, exact_center_b[1] + y) for x, y in exact_local]
    perturbed_local = _perturbation(rng, exact_local)
    perturbed_a = [(exact_center_a[0] + x, exact_center_a[1] + y) for x, y in perturbed_local]
    perturbed_b = [(exact_center_b[0] + x, exact_center_b[1] + y) for x, y in perturbed_local]
    # The generation schedule advances by an odd stride, so this gives a
    # deterministic near-perfect A/B balance on every public stress sweep
    # without changing any rendered nuisance outside the constellation itself.
    if int(seed) % 2 == 0:
        cand_a, cand_b = exact_a, perturbed_b
    else:
        cand_a, cand_b = perturbed_a, exact_b
    return {
        "ref_points": [[round(x, 6), round(y, 6)] for x, y in reference],
        "cand_a_points": [[round(x, 6), round(y, 6)] for x, y in cand_a],
        "cand_b_points": [[round(x, 6), round(y, 6)] for x, y in cand_b],
    }


def render(scene: dict) -> Image.Image:
    """Render a central three-panel board with antialiased equal disks."""

    supersample = 4
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), BACKGROUND)
    draw = ImageDraw.Draw(image)
    header_font = ImageFont.load_default(size=13 * supersample)
    label_font = ImageFont.load_default(size=14 * supersample)
    note_font = ImageFont.load_default(size=9 * supersample)

    def sc(value: float) -> int:
        return int(round(value * supersample))

    def line(points, fill, width=1):
        draw.line([(sc(x), sc(y)) for x, y in points], fill=fill, width=width * supersample)

    # A quiet triangular central board keeps the decisive components in one
    # medium-sized region while leaving visible gaps between ablation clusters.
    draw.rounded_rectangle(
        [sc(152), sc(104), sc(648), sc(536)],
        radius=sc(14),
        outline=FRAME,
        width=2 * supersample,
    )
    draw.text((sc(285), sc(52)), "THREE-MARKER CORRESPONDENCE", fill=TEXT, font=header_font)
    labels = ("REFERENCE", "A", "B")
    for index, ((cx, cy), label) in enumerate(zip(PANEL_CENTERS, labels, strict=True)):
        draw.rounded_rectangle(
            [
                sc(cx - PANEL_HALF_WIDTH),
                sc(cy - PANEL_HALF_HEIGHT),
                sc(cx + PANEL_HALF_WIDTH),
                sc(cy + PANEL_HALF_HEIGHT),
            ],
            radius=sc(10),
            outline=FRAME,
            width=2 * supersample,
        )
        bbox = draw.textbbox((0, 0), label, font=label_font)
        label_width = bbox[2] - bbox[0]
        draw.text((sc(cx) - label_width // 2, sc(cy - 91)), label, fill=TEXT, font=label_font)
        if index:
            note = "TURN 90 CW"
            note_box = draw.textbbox((0, 0), note, font=note_font)
            note_width = note_box[2] - note_box[0]
            draw.text((sc(cx) - note_width // 2, sc(cy + 83)), note, fill=TEXT, font=note_font)

    point_lists = (scene["ref_points"], scene["cand_a_points"], scene["cand_b_points"])
    for points in point_lists:
        for x, y in points:
            draw.ellipse(
                [sc(float(x) - MARKER_RADIUS), sc(float(y) - MARKER_RADIUS), sc(float(x) + MARKER_RADIUS), sc(float(y) + MARKER_RADIUS)],
                fill=MARKER,
            )

    # Small visual cues for the declared turn, separate from the marker ink.
    for cx, cy in (PANEL_CENTERS[1], PANEL_CENTERS[2]):
        arrow_y = cy - 52.0
        line([(cx - 20, arrow_y), (cx + 20, arrow_y)], FRAME, width=1)
        line([(cx + 20, arrow_y), (cx + 14, arrow_y - 4)], FRAME, width=1)
        line([(cx + 20, arrow_y), (cx + 14, arrow_y + 4)], FRAME, width=1)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def _candidate_error(
    reference_points: Iterable[Iterable[float]],
    candidate_points: Iterable[Iterable[float]],
    reference_center: tuple[float, float],
    candidate_center: tuple[float, float],
) -> float:
    reference = _local_points(reference_points, reference_center)
    candidate = [_rotate_cw(point) for point in _local_points(candidate_points, candidate_center)]
    return _assignment_error(reference, candidate)


def analytic_gold(scene: dict) -> str:
    # Keep all label-bearing fields explicit: this is both auditable and avoids
    # hiding a gold input behind a dynamic dictionary key.
    error_a = _candidate_error(
        scene["ref_points"], scene["cand_a_points"], PANEL_CENTERS[0], PANEL_CENTERS[1]
    )
    error_b = _candidate_error(
        scene["ref_points"], scene["cand_b_points"], PANEL_CENTERS[0], PANEL_CENTERS[2]
    )
    return "A" if error_a < error_b else "B"


def margin(scene: dict) -> float:
    error_a = _candidate_error(
        scene["ref_points"], scene["cand_a_points"], PANEL_CENTERS[0], PANEL_CENTERS[1]
    )
    error_b = _candidate_error(
        scene["ref_points"], scene["cand_b_points"], PANEL_CENTERS[0], PANEL_CENTERS[2]
    )
    return abs(error_a - error_b)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise order aliases: disk drawing and set assignment ignore list order."""

    twins = []
    for key in ("ref_points", "cand_a_points", "cand_b_points"):
        twin = copy.deepcopy(scene)
        twin[key] = list(reversed(twin[key]))
        twins.append((f"reverse_{key}", twin))
    return twins
