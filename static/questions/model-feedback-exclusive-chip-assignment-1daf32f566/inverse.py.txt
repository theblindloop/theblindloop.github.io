"""Independent image-only inverse for Exclusive Chip Assignment."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

DECISIONS = ("parcel-A", "parcel-B", "parcel-C", "parcel-D")
PALETTE = np.array(((220, 55, 70), (35, 110, 210), (35, 160, 90), (235, 170, 35), (120, 75, 190), (20, 170, 185), (235, 105, 35), (105, 75, 50)), dtype=np.int16)
START = np.array((205, 40, 145), dtype=np.int16)


def _read_set(arr: np.ndarray, x0: int, x1: int, y0: int, y1: int) -> list[int]:
    crop = arr[y0:y1, x0:x1].astype(np.int16)
    found = []
    for i, color in enumerate(PALETTE):
        dist = np.max(np.abs(crop - color), axis=2)
        if int((dist <= 12).sum()) >= 500:
            found.append(i)
    return found


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (900, 650):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        badges, parcels, starts = [], [], []
        for row in range(4):
            y0 = 110 + 128 * row
            badges.append(_read_set(arr, 45, 405, y0, y0 + 100))
            parcels.append(_read_set(arr, 495, 855, y0, y0 + 100))
            border = arr[y0:y0 + 100, 45:405].astype(np.int16)
            starts.append(int((np.max(np.abs(border - START), axis=2) <= 12).sum()) >= 700)
        if sum(starts) != 1:
            return "abstain"
        if sorted(map(len, badges)) != [3, 3, 3, 4] or sorted(map(len, parcels)) != [3, 3, 3, 4]:
            return "abstain"
        if sum(map(len, badges)) != 13 or sum(map(len, parcels)) != 13:
            return "abstain"
        allowed = [[len(set(a) & set(b)) == 1 for b in parcels] for a in badges]
        solutions = [p for p in itertools.permutations(range(4)) if all(allowed[i][p[i]] for i in range(4))]
        start = starts.index(1)
        if len(solutions) != 1 or sum(allowed[start]) < 2:
            return "abstain"
        return DECISIONS[solutions[0][start]]
    except Exception:
        return "abstain"
