"""Deterministic world for a globally constrained correspondence puzzle."""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 650
DECISIONS = ("parcel-A", "parcel-B", "parcel-C", "parcel-D")
PALETTE = (
    (220, 55, 70), (35, 110, 210), (35, 160, 90), (235, 170, 35),
    (120, 75, 190), (20, 170, 185), (235, 105, 35), (105, 75, 50),
)
INK = (35, 39, 48)
CARD = (250, 248, 240)
START = (205, 40, 145)
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

# Compatibility is intersection size exactly one.  This instance has the
# matrix [1111,1000,0100,1010] and the sole matching [3,0,1,2].
_BADGES = ((5, 6, 7), (1, 3, 5, 7), (0, 4, 6), (1, 4, 7))
_PARCELS = ((0, 4, 5), (1, 4, 7), (1, 2, 5), (0, 1, 4, 5))


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _solutions(badges: list[list[int]], parcels: list[list[int]]) -> list[tuple[int, ...]]:
    allowed = [[len(set(a) & set(b)) == 1 for b in parcels] for a in badges]
    return [p for p in itertools.permutations(range(4)) if all(allowed[i][p[i]] for i in range(4))]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    chip_perm = list(range(8))
    rng.shuffle(chip_perm)
    badges = [[chip_perm[x] for x in card] for card in _BADGES]
    parcels = [[chip_perm[x] for x in card] for card in _PARCELS]

    # Vary badge placement while retaining which latent card is START.
    badge_order = list(range(4))
    rng.shuffle(badge_order)
    start_index = badge_order.index(0)
    badges = [badges[i] for i in badge_order]

    # Place START's unique partner at a seed-balanced visible parcel label.
    target_row = int(seed) % 4
    other = [0, 1, 2]
    rng.shuffle(other)
    parcel_order: list[int | None] = [None] * 4
    parcel_order[target_row] = 3
    for row, idx in zip((r for r in range(4) if r != target_row), other):
        parcel_order[row] = idx
    parcels = [parcels[int(i)] for i in parcel_order]
    scene = {"badge_sets": badges, "parcel_sets": parcels, "start_index": start_index}
    if is_quarantined(scene):
        raise RuntimeError("constructed scene failed assignment invariants")
    return scene


def analytic_gold(scene: dict) -> str:
    solutions = _solutions(scene["badge_sets"], scene["parcel_sets"])
    if len(solutions) != 1:
        raise ValueError("scene does not have one complete pairing")
    partner = solutions[0][int(scene["start_index"])]
    return DECISIONS[partner]


def margin(scene: dict) -> float:
    """One assignment separates the valid scene from the discrete ambiguity boundary."""
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        badges = scene["badge_sets"]
        parcels = scene["parcel_sets"]
        start = int(scene["start_index"])
        if len(badges) != 4 or len(parcels) != 4 or start not in range(4):
            return True
        if sorted(map(len, badges)) != [3, 3, 3, 4] or sorted(map(len, parcels)) != [3, 3, 3, 4]:
            return True
        if any(len(set(card)) != len(card) or any(x not in range(8) for x in card) for card in badges + parcels):
            return True
        allowed = [[len(set(a) & set(b)) == 1 for b in parcels] for a in badges]
        if sum(allowed[start]) < 2:
            return True
        return len(_solutions(badges, parcels)) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = {
        "badge_sets": [list(reversed(card)) for card in scene["badge_sets"]],
        "parcel_sets": [list(reversed(card)) for card in scene["parcel_sets"]],
        "start_index": scene["start_index"],
    }
    return [("reverse_storage_order_within_sets", transformed)]


def _draw_card(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], chips: list[int], label: str, start: bool) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=14, fill=CARD, outline=START if start else INK, width=7 if start else 3)
    draw.text((x0 + 16, y0 + 10), label, font=_font(23), fill=START if start else INK)
    ordered = sorted(chips)
    gap = 55
    left = x1 - 25 - gap * (len(ordered) - 1)
    cy = (y0 + y1) // 2 + 15
    for k, chip in enumerate(ordered):
        cx = left + k * gap
        draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=PALETTE[chip], outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (243, 246, 249))
    draw = ImageDraw.Draw(image)
    draw.text((50, 20), "ONE-TO-ONE CHIP MATCH", font=_font(34), fill=INK)
    draw.text((85, 72), "BADGES", font=_font(25), fill=INK)
    draw.text((590, 72), "PARCELS", font=_font(25), fill=INK)
    for row in range(4):
        y0 = 110 + 128 * row
        _draw_card(draw, (45, y0, 405, y0 + 100), scene["badge_sets"][row], "START" if row == scene["start_index"] else f"badge {row + 1}", row == scene["start_index"])
        _draw_card(draw, (495, y0, 855, y0 + 100), scene["parcel_sets"][row], f"parcel {chr(65 + row)}", False)
    return image
