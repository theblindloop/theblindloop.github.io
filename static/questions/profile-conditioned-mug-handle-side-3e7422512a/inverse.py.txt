"""Independent pixel-only inverse for the visible mug-handle-side raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "mug_handle_side-oracle-0.1.0"


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    h, w, _ = array.shape
    corner_blocks = np.concatenate(
        (
            array[:8, :8].reshape(-1, 3),
            array[:8, w - 8 :].reshape(-1, 3),
            array[h - 8 :, :8].reshape(-1, 3),
            array[h - 8 :, w - 8 :].reshape(-1, 3),
        ),
        axis=0,
    )
    background = np.median(corner_blocks, axis=0)
    distance = np.sqrt(np.sum((array.astype(np.float64) - background) ** 2, axis=2))
    return distance > 24.0


def _longest_true_run(values: np.ndarray) -> tuple[int, int]:
    best_start = -1
    best_end = -1
    start = None
    for index, value in enumerate(values.tolist() + [False]):
        if value and start is None:
            start = index
        elif not value and start is not None:
            if best_start < 0 or index - start > best_end - best_start:
                best_start, best_end = start, index - 1
            start = None
    if best_start < 0:
        raise ValueError("central mug body was not recovered")
    return best_start, best_end


def decision_from_image(image: Image.Image) -> str:
    """Recover left/right by comparing silhouette extension beyond the body run."""

    mask = _ink_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < 800:
        raise ValueError("mug silhouette is not measurable")

    counts = np.bincount(xs, minlength=mask.shape[1]).astype(np.float64)
    peak = float(np.max(counts))
    central_columns = counts >= max(80.0, peak * 0.65)
    body_left, body_right = _longest_true_run(central_columns)
    outer_left = int(np.min(xs))
    outer_right = int(np.max(xs))
    left_extension = body_left - outer_left
    right_extension = outer_right - body_right
    if abs(right_extension - left_extension) < 10:
        raise ValueError("mug handle side is ambiguous at raster resolution")
    return "right" if right_extension > left_extension else "left"
