"""Deterministic raster renderer for the visible mug-handle-side world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "mug_handle_side-renderer-0.1.0"
WIDTH = 520
HEIGHT = 380
SUPERSAMPLE = 4

BACKGROUND_PALETTE = (
    (250, 247, 239),
    (239, 248, 250),
    (247, 241, 250),
    (242, 249, 240),
    (250, 242, 236),
)
BODY_PALETTE = (
    (220, 143, 96),
    (99, 166, 157),
    (204, 111, 151),
    (116, 143, 202),
    (224, 185, 78),
    (146, 119, 178),
)
OUTLINE_PALETTE = (
    (43, 48, 56),
    (52, 45, 61),
    (45, 61, 54),
    (63, 48, 42),
)
COFFEE_PALETTE = (
    (91, 57, 43),
    (61, 48, 76),
    (74, 66, 48),
)


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in values)  # type: ignore[return-value]


def _draw_round_rectangle(
    draw: ImageDraw.ImageDraw,
    values: tuple[float, float, float, float],
    radius: float,
    scale: int,
    *,
    fill: tuple[int, int, int],
    outline: tuple[int, int, int],
    width: int,
) -> None:
    draw.rounded_rectangle(
        _box(values, scale),
        radius=max(1, int(round(radius * scale))),
        fill=fill,
        outline=outline,
        width=max(1, int(width)) * scale,
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, JSON-friendly front-view mug scene."""

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 91_337)
    handle_side = "left" if seed % 2 == 0 else "right"
    body_width = rng.randint(164, 208)
    body_height = rng.randint(136, 172)
    body_top = round(rng.uniform(108.0, 132.0), 4)
    body_center_x = round(rng.uniform(228.0, 292.0), 4)

    # A few deterministic near-boundary scenes make the finite-resolution
    # quarantine observable. They remain large enough for the inverse to read.
    if seed % 17 == 0:
        handle_extent = round(rng.uniform(39.0, 47.0), 4)
    else:
        handle_extent = round(rng.uniform(56.0, 84.0), 4)

    return {
        "handle_side": handle_side,
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": body_center_x,
        "body_top": body_top,
        "body_width": body_width,
        "body_height": body_height,
        "handle_extent": handle_extent,
        "handle_top_offset": round(rng.uniform(24.0, 34.0), 4),
        "handle_bottom_offset": round(rng.uniform(25.0, 36.0), 4),
        "body_radius": round(rng.uniform(21.0, 29.0), 4),
        "handle_radius": round(rng.uniform(18.0, 27.0), 4),
        "outline_width": rng.choice((5, 6, 7)),
        "body_color": list(rng.choice(BODY_PALETTE)),
        "background_color": list(rng.choice(BACKGROUND_PALETTE)),
        "outline_color": list(rng.choice(OUTLINE_PALETTE)),
        "coffee_color": list(rng.choice(COFFEE_PALETTE)),
        "rim_inset": round(rng.uniform(17.0, 27.0), 4),
        "antialias": True,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the mug, drawing the handle behind the body so its attachment is visible."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE if bool(scene["antialias"]) else 1
    background = _rgb(scene["background_color"])
    body_color = _rgb(scene["body_color"])
    outline = _rgb(scene["outline_color"])
    coffee = _rgb(scene["coffee_color"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)

    centre_x = float(scene["center_x"])
    top = float(scene["body_top"])
    body_width = float(scene["body_width"])
    body_height = float(scene["body_height"])
    bottom = top + body_height
    left = centre_x - body_width / 2.0
    right = centre_x + body_width / 2.0
    extent = float(scene["handle_extent"])
    top_offset = float(scene["handle_top_offset"])
    bottom_offset = float(scene["handle_bottom_offset"])
    ring_top = top + top_offset
    ring_bottom = bottom - bottom_offset

    if str(scene["handle_side"]) == "right":
        outer = (right - 13.0, ring_top, right + extent, ring_bottom)
        inner = (right + 5.0, ring_top + 23.0, right + extent - 15.0, ring_bottom - 23.0)
    else:
        outer = (left - extent, ring_top, left + 13.0, ring_bottom)
        inner = (left - extent + 15.0, ring_top + 23.0, left - 5.0, ring_bottom - 23.0)

    _draw_round_rectangle(
        draw,
        outer,
        float(scene["handle_radius"]),
        scale,
        fill=body_color,
        outline=outline,
        width=int(scene["outline_width"]),
    )
    _draw_round_rectangle(
        draw,
        inner,
        max(8.0, float(scene["handle_radius"]) - 7.0),
        scale,
        fill=background,
        outline=outline,
        width=int(scene["outline_width"]),
    )

    _draw_round_rectangle(
        draw,
        (left, top, right, bottom),
        float(scene["body_radius"]),
        scale,
        fill=body_color,
        outline=outline,
        width=int(scene["outline_width"]),
    )

    rim_inset = float(scene["rim_inset"])
    draw.ellipse(
        _box((left + rim_inset, top - 13.0, right - rim_inset, top + 36.0), scale),
        fill=coffee,
        outline=outline,
        width=max(1, int(scene["outline_width"])) * scale,
    )

    if scale > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the side encoded by the visible handle attachment."""

    return str(scene["handle_side"])


def margin(scene: dict[str, Any]) -> float:
    """Return the visible protrusion beyond the rounded-body baseline."""

    return round(float(scene["handle_extent"]) - 30.0, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Quarantine shallow handle protrusions."""

    return margin(scene) < 20.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; the side label is visibly rendered."""

    return []
