"""Deterministic scene model and Pillow renderer for Tide Turn Ledger."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 600, 480
HEADINGS = ("up", "right", "down", "left")
CARD = (145, 105, 455, 375)
PANEL_X = (159, 259, 359)
PANEL_Y = 205
CELL_W, CELL_H = 19, 27
BLUE = (25, 125, 154)
CREAM = (247, 242, 228)
CARD_COLOR = (219, 227, 232)
GOLD = (244, 183, 50)
INK = (35, 42, 48)
BG = (255, 255, 255)


def _connected_cells(rng: random.Random, count: int) -> list[list[int]]:
    start = (rng.randrange(1, 3), rng.randrange(1, 3))
    cells = {start}
    while len(cells) < count:
        frontier = []
        for r, c in cells:
            for q in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if 0 <= q[0] < 4 and 0 <= q[1] < 4 and q not in cells:
                    frontier.append(q)
        cells.add(rng.choice(sorted(set(frontier))))
    return [list(p) for p in sorted(cells)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    start_heading = HEADINGS[rng.randrange(4)]
    # Balanced sign histories, decorrelated from starting direction.
    patterns = ((1, 1), (1, -1), (-1, 1), (-1, -1))
    s1, s2 = patterns[(int(seed) // 4) % 4]
    base = 7 + rng.randrange(3)
    d1 = 2 + rng.randrange(2)
    d2 = 2 + rng.randrange(2)
    counts = [base, base + s1 * d1, base + s1 * d1 + s2 * d2]
    if min(counts) < 3 or max(counts) > 14:
        base = 8
        counts = [base, base + s1 * 2, base + s1 * 2 + s2 * 2]
    return {
        "start_heading": start_heading,
        "basins": [_connected_cells(random.Random(rng.randrange(1 << 30)), n) for n in counts],
    }


def _counts(scene: dict) -> list[int]:
    return [len(mask) for mask in scene["basins"]]


def analytic_gold(scene: dict) -> str:
    heading = HEADINGS.index(scene["start_heading"])
    counts = _counts(scene)
    for before, after in zip(counts, counts[1:]):
        if after == before:
            raise ValueError("equal-area transition is ambiguous")
        heading = (heading + (1 if after > before else -1)) % 4
    return HEADINGS[heading]


def margin(scene: dict) -> float:
    counts = _counts(scene)
    return float(min(abs(b - a) for a, b in zip(counts, counts[1:])) * CELL_W * CELL_H)


def is_quarantined(scene: dict) -> bool:
    counts = _counts(scene)
    return any(a == b for a, b in zip(counts, counts[1:]))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, heading: str) -> None:
    vectors = {"up": (0, -1), "right": (1, 0), "down": (0, 1), "left": (-1, 0)}
    dx, dy = vectors[heading]
    ex, ey = cx + dx * 24, cy + dy * 24
    draw.line((cx - dx * 9, cy - dy * 9, ex, ey), fill=INK, width=8)
    px, py = -dy, dx
    draw.polygon([(ex, ey), (ex - dx * 11 + px * 8, ey - dy * 11 + py * 8),
                  (ex - dx * 11 - px * 8, ey - dy * 11 - py * 8)], fill=INK)


def render(scene: dict) -> Image.Image:
    # Access both label inputs directly so the rendered-field audit is explicit.
    start_heading = scene["start_heading"]
    basins = scene["basins"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    d = ImageDraw.Draw(image)
    d.rounded_rectangle(CARD, radius=12, fill=CARD_COLOR, outline=INK, width=5)
    d.text((164, 120), "START", fill=INK, stroke_width=0)
    d.ellipse((270, 118, 330, 178), fill=GOLD, outline=INK, width=3)
    _arrow(d, 300, 148, start_heading)
    d.line((163, 190, 437, 190), fill=INK, width=4)
    for i, (x0, cells) in enumerate(zip(PANEL_X, basins), start=1):
        x1, y1 = x0 + 4 * CELL_W, PANEL_Y + 4 * CELL_H
        d.rectangle((x0 - 4, PANEL_Y - 4, x1 + 4, y1 + 4), fill=INK)
        d.rectangle((x0, PANEL_Y, x1, y1), fill=CREAM)
        flooded = {tuple(p) for p in cells}
        for r in range(4):
            for c in range(4):
                if (r, c) in flooded:
                    d.rectangle((x0 + c * CELL_W, PANEL_Y + r * CELL_H,
                                 x0 + (c + 1) * CELL_W, PANEL_Y + (r + 1) * CELL_H), fill=BLUE)
        for c in range(5):
            xx = x0 + c * CELL_W
            d.line((xx, PANEL_Y, xx, y1), fill=INK, width=2)
        for r in range(5):
            yy = PANEL_Y + r * CELL_H
            d.line((x0, yy, x1, yy), fill=INK, width=2)
        # Order markers are redundant to spatial order but visible to people.
        d.ellipse((x0 + 27, 181, x0 + 49, 203), fill=GOLD, outline=INK, width=2)
        d.text((x0 + 35, 185), str(i), fill=INK)
    d.polygon([(247, 337), (253, 343), (247, 349)], fill=INK)
    d.polygon([(347, 337), (353, 343), (347, 349)], fill=INK)
    return image
