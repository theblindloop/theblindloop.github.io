"""Independent pixel-only inverse for Tide Turn Ledger."""

from __future__ import annotations

import numpy as np

HEADINGS = ("up", "right", "down", "left")
BLUE = np.array((25, 125, 154), dtype=np.int16)
PALE = np.array((219, 227, 232), dtype=np.int16)
GOLD = np.array((244, 183, 50), dtype=np.int16)
INK = np.array((35, 42, 48), dtype=np.int16)


def _mask(arr: np.ndarray, color: np.ndarray, tol: int = 8) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tol


def _components(mask: np.ndarray, minimum: int = 10) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while stack:
            cy, cx = stack.pop(); pts.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; stack.append((ny,nx))
        if len(pts) >= minimum:
            out.append(np.asarray(pts))
    return out


def decision_from_image(image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or min(arr.shape[:2]) < 100:
        raise ValueError("ledger raster is missing")

    # Recover the single connected card by its neutral paper field. Its invariant
    # ink area is a genuine integrity cue: destructive erasure makes replay unsafe.
    pale_parts = _components(_mask(arr, PALE), 1000)
    if len(pale_parts) != 1:
        raise ValueError("chronicle card is fragmented")
    paper = pale_parts[0]
    y0, x0 = paper.min(axis=0); y1, x1 = paper.max(axis=0)
    ph, pw = y1-y0+1, x1-x0+1
    if not (230 <= ph <= 290 and 270 <= pw <= 330 and 42000 <= len(paper) <= 54000):
        raise ValueError("chronicle card is incomplete")

    gold_parts = _components(_mask(arr, GOLD), 40)
    if len(gold_parts) != 4:
        raise ValueError("start or order markers are damaged")
    start = max(gold_parts, key=len)
    if not 1900 <= len(start) <= 2200:
        raise ValueError("start dial is damaged")
    gy0, gx0 = start.min(axis=0); gy1, gx1 = start.max(axis=0)
    cy, cx = start.mean(axis=0)
    crop_ink = _mask(arr, INK)[gy0:gy1+1, gx0:gx1+1]
    iy, ix = np.nonzero(crop_ink)
    if len(ix) < 120:
        raise ValueError("start arrow is unreadable")
    vx = float(np.mean(ix + gx0 - cx)); vy = float(np.mean(iy + gy0 - cy))
    if abs(vx) > abs(vy):
        start_heading = "right" if vx > 0 else "left"
    else:
        start_heading = "down" if vy > 0 else "up"

    blue = _mask(arr, BLUE)
    parts = _components(blue, 100)
    if not parts:
        raise ValueError("no flooded basin cells")
    # Each blue cell is separated by visible grid ink. Group recovered cells into
    # snapshots by their centroid x-order; no latent coordinates are consulted.
    cell_centers = sorted([(float(p[:,1].mean()), float(p[:,0].mean()), len(p)) for p in parts])
    if any(not 300 <= area <= 520 for _, _, area in cell_centers):
        raise ValueError("a flooded cell is damaged")
    gaps = [(cell_centers[i+1][0]-cell_centers[i][0], i) for i in range(len(cell_centers)-1)]
    if len(gaps) < 2:
        raise ValueError("fewer than three snapshots")
    cuts = sorted(i for _, i in sorted(gaps, reverse=True)[:2])
    groups = [cell_centers[:cuts[0]+1], cell_centers[cuts[0]+1:cuts[1]+1], cell_centers[cuts[1]+1:]]
    if any(not 3 <= len(g) <= 14 for g in groups):
        raise ValueError("snapshot cell count is implausible")
    counts = [len(g) for g in groups]
    if counts[0] == counts[1] or counts[1] == counts[2]:
        raise ValueError("equal-area transition")
    heading = HEADINGS.index(start_heading)
    for a, b in zip(counts, counts[1:]):
        heading = (heading + (1 if b > a else -1)) % 4
    return HEADINGS[heading]
