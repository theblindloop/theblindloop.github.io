"""Scene-free pixel inverse for the colored histogram bars."""

from __future__ import annotations

import math

from PIL import Image

COLORS = ("blue", "orange", "green")
RGB = {
    "blue": (45, 114, 210),
    "orange": (230, 126, 34),
    "green": (38, 166, 91),
}


def _bar_heights(image: Image.Image, rgb: tuple[int, int, int]) -> list[int]:
    pixels = image.convert("RGB")
    width, height = pixels.size
    occupied_x = []
    for x in range(width):
        if any(pixels.getpixel((x, y)) == rgb for y in range(height)):
            occupied_x.append(x)
    runs = []
    for x in occupied_x:
        if not runs or x > runs[-1][-1] + 1:
            runs.append([x])
        else:
            runs[-1].append(x)
    if len(runs) != 4:
        raise ValueError(f"expected four separated bars, found {len(runs)}")
    heights = []
    for run in runs:
        ys = [
            y
            for x in run
            for y in range(height)
            if pixels.getpixel((x, y)) == rgb
        ]
        if not ys:
            raise ValueError("empty colored component")
        heights.append(max(ys) - min(ys) + 1)
    return heights


def _entropy(values: list[int]) -> float:
    total = float(sum(values))
    if total <= 0:
        raise ValueError("zero histogram total")
    return -sum((v / total) * math.log2(v / total) for v in values if v > 0)


def decision_from_image(image: Image.Image) -> str:
    scores = {color: _entropy(_bar_heights(image, RGB[color])) for color in COLORS}
    ranked = sorted(scores, key=scores.get, reverse=True)
    if scores[ranked[0]] - scores[ranked[1]] < 0.06:
        raise ValueError("ambiguous entropy margin")
    return ranked[0]
