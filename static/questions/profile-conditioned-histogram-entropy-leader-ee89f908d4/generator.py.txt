"""Deterministic renderer and analytic arm for histogram entropy."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 360
COLORS = ("blue", "orange", "green")
RGB = {
    "blue": (45, 114, 210),
    "orange": (230, 126, 34),
    "green": (38, 166, 91),
}
MIN_MARGIN = 0.08

_HIGH = (
    (28, 24, 26, 22),
    (23, 27, 21, 29),
    (25, 30, 24, 21),
    (26, 20, 29, 25),
)
_MID = (
    (45, 25, 18, 12),
    (42, 30, 18, 10),
    (48, 22, 19, 11),
    (40, 34, 17, 9),
)
_LOW = (
    (68, 14, 10, 8),
    (64, 20, 9, 7),
    (70, 12, 10, 8),
    (62, 22, 10, 6),
)


def _entropy(values) -> float:
    total = float(sum(values))
    return -sum((v / total) * math.log2(v / total) for v in values if v > 0)


def _mapping(scene: dict) -> dict[str, list[int]]:
    return {
        color: [int(v) for v in values]
        for color, values in zip(scene["colors"], scene["heights"])
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = COLORS[int(seed) % len(COLORS)]
    others = [color for color in COLORS if color != winner]
    rng.shuffle(others)
    assigned = {
        winner: list(rng.choice(_HIGH)),
        others[0]: list(rng.choice(_MID)),
        others[1]: list(rng.choice(_LOW)),
    }
    for values in assigned.values():
        rng.shuffle(values)
    colors = list(COLORS)
    rng.shuffle(colors)
    panel_order = list(COLORS)
    rng.shuffle(panel_order)
    return {
        "heights": [assigned[color] for color in colors],
        "colors": colors,
        "panel_order": panel_order,
    }


def render(scene: dict) -> Image.Image:
    values_by_color = _mapping(scene)
    panel_order = list(scene["panel_order"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((22, 18), "FOUR-BIN FREQUENCY HISTOGRAMS", fill=(28, 31, 36), font=font)
    draw.text((22, 36), "All panels have the same total bar height", fill=(75, 78, 84), font=font)

    for panel_index, color in enumerate(panel_order):
        left = 25 + 235 * panel_index
        top, right, baseline = 70, left + 200, 300
        draw.rounded_rectangle((left, top, right, 325), radius=8, fill=(255, 255, 255), outline=(214, 215, 218), width=2)
        caption = color.upper()
        caption_box = draw.textbbox((0, 0), caption, font=font)
        caption_width = caption_box[2] - caption_box[0]
        draw.text((left + (200 - caption_width) // 2, 82), caption, fill=(35, 38, 43), font=font)
        draw.line((left + 14, baseline + 1, right - 14, baseline + 1), fill=(45, 47, 52), width=2)
        for bin_index, value in enumerate(values_by_color[color]):
            x0 = left + 20 + 43 * bin_index
            x1 = x0 + 28
            bar_height = 2 * int(value)
            draw.rectangle((x0, baseline - bar_height + 1, x1, baseline), fill=RGB[color])
            draw.text((x0 + 10, 307), str(bin_index + 1), fill=(80, 82, 87), font=font)
    return image


def analytic_gold(scene: dict) -> str:
    values_by_color = _mapping(scene)
    scores = {color: _entropy(values_by_color[color]) for color in COLORS}
    return max(COLORS, key=lambda color: scores[color])


def margin(scene: dict) -> float:
    values_by_color = _mapping(scene)
    scores = sorted((_entropy(values_by_color[color]) for color in COLORS), reverse=True)
    return scores[0] - scores[1]


def is_quarantined(scene: dict) -> bool:
    values_by_color = _mapping(scene)
    return margin(scene) < MIN_MARGIN or min(min(values) for values in values_by_color.values()) < 6


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["colors"] = list(reversed(twin["colors"]))
    twin["heights"] = list(reversed(twin["heights"]))
    return [("reverse_latent_binding_order", twin)]
