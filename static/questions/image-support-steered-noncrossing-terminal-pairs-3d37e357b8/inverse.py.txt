"""Independent pixel-only inverse for cyclic terminal order."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, float, float]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 80:
            ys = np.fromiter((point[0] for point in points), dtype=np.float64)
            xs = np.fromiter((point[1] for point in points), dtype=np.float64)
            found.append((len(points), float(xs.mean()), float(ys.mean())))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Return whether the two same-color boundary pairs can be joined disjointly."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    cyan_mask = (blue >= 135) & (green >= 120) & (red <= 85) & (blue - red >= 70)
    orange_mask = (red >= 175) & (green >= 55) & (green <= 165) & (blue <= 90)
    cyan = _components(cyan_mask)
    orange = _components(orange_mask)
    if len(cyan) != 2 or len(orange) != 2:
        raise ValueError(f"expected two cyan and two orange terminals, got {len(cyan)} and {len(orange)}")

    terminals = [(x, y, "cyan") for _area, x, y in cyan]
    terminals += [(x, y, "orange") for _area, x, y in orange]
    center_x = sum(item[0] for item in terminals) / 4.0
    center_y = sum(item[1] for item in terminals) / 4.0
    ordered = sorted(
        terminals,
        key=lambda item: math.atan2(item[1] - center_y, item[0] - center_x),
    )
    labels = [item[2] for item in ordered]
    alternating = labels[0] == labels[2] and labels[1] == labels[3]
    return "no" if alternating else "yes"
