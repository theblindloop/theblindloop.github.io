"""Deterministic renderer and analytic arm for paired boundary terminals."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 249, 246)
BLACK = (30, 34, 38)
PALE = (196, 199, 201)
PALE_FILL = (239, 239, 235)
PALETTES = (
    {"cyan": (0, 164, 204), "orange": (238, 117, 20)},
    {"cyan": (15, 177, 196), "orange": (224, 91, 26)},
    {"cyan": (0, 145, 190), "orange": (244, 133, 32)},
)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed parity controls only the answer class."""
    rng = random.Random(int(seed))
    center_x = rng.choice((160, 288, 352))
    center_y = rng.choice((160, 288, 352))
    arena_radius = 91
    terminal_radius = rng.choice((10, 11, 12))
    start_angle = 45 + 90 * rng.randrange(4)
    angle_jitter = [rng.uniform(-1.7, 1.7) for _ in range(4)]

    if int(seed) % 2 == 0:
        colors = ["cyan", "cyan", "orange", "orange"]  # nonalternating: yes
    else:
        colors = ["cyan", "orange", "cyan", "orange"]  # alternating: no
    shift = rng.randrange(4)
    colors = colors[shift:] + colors[:shift]
    if rng.choice((False, True)):
        colors = ["orange" if value == "cyan" else "cyan" for value in colors]
    if rng.choice((False, True)):
        colors = [colors[0], colors[3], colors[2], colors[1]]

    distractors = []
    kinds = ("square", "diamond", "ring")
    for _ in range(5):
        for _attempt in range(200):
            x = rng.randrange(36, WIDTH - 35)
            y = rng.randrange(36, HEIGHT - 35)
            if math.hypot(x - center_x, y - center_y) > 145:
                distractors.append([x, y, rng.choice(kinds), rng.choice((5, 6, 7))])
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "center_x": center_x,
        "center_y": center_y,
        "arena_radius": arena_radius,
        "terminal_radius": terminal_radius,
        "start_angle": start_angle,
        "angle_jitter": angle_jitter,
        "cyclic_colors": colors,
        "palette_variant": rng.randrange(len(PALETTES)),
        "distractors": distractors,
    }


def _terminal_centers(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["arena_radius"])
    start = float(scene["start_angle"])
    jitter = scene["angle_jitter"]
    centers = []
    for index in range(4):
        angle = math.radians(start + 90 * index + float(jitter[index]))
        centers.append((cx + radius * math.cos(angle), cy + radius * math.sin(angle)))
    return centers


def render(scene: dict) -> Image.Image:
    """Render the final RGB image with antialiasing by 4x supersampling."""
    width = int(scene["width"])
    height = int(scene["height"])
    scale = 4
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for x, y, kind, size in scene["distractors"]:
        x, y, size = int(x) * scale, int(y) * scale, int(size) * scale
        if kind == "square":
            draw.rectangle((x - size, y - size, x + size, y + size), outline=PALE, width=2 * scale)
        elif kind == "diamond":
            draw.polygon(((x, y - size), (x + size, y), (x, y + size), (x - size, y)), outline=PALE)
        else:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=PALE, width=2 * scale)

    cx = float(scene["center_x"]) * scale
    cy = float(scene["center_y"]) * scale
    arena_radius = float(scene["arena_radius"]) * scale
    draw.ellipse(
        (cx - arena_radius, cy - arena_radius, cx + arena_radius, cy + arena_radius),
        fill=PALE_FILL,
        outline=BLACK,
        width=6 * scale,
    )

    palette = PALETTES[int(scene["palette_variant"])]
    colors = scene["cyclic_colors"]
    terminal_radius = float(scene["terminal_radius"]) * scale
    for (x, y), color_name in zip(_terminal_centers(scene), colors, strict=True):
        x *= scale
        y *= scale
        draw.ellipse(
            (x - terminal_radius, y - terminal_radius, x + terminal_radius, y + terminal_radius),
            fill=palette[color_name],
            outline=BACKGROUND,
            width=2 * scale,
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    colors = scene["cyclic_colors"]
    alternating = colors[0] == colors[2] and colors[1] == colors[3]
    return "no" if alternating else "yes"


def margin(scene: dict) -> float:
    centers = _terminal_centers(scene)
    radius = float(scene["terminal_radius"])
    gaps = []
    for index, point in enumerate(centers):
        other = centers[(index + 1) % 4]
        gaps.append(math.dist(point, other) - 2.0 * radius)
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
