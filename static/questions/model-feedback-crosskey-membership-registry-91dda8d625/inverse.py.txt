"""Independent image-only inverse for Cross-Key Membership Registry."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
DECISIONS = tuple("ABCDEF")
INK = np.array((35, 39, 48), dtype=np.int16)
START = np.array((198, 38, 139), dtype=np.int16)
LEFT_COLORS = tuple(np.array(color, dtype=np.int16) for color in (
    (220, 55, 62), (35, 105, 210), (20, 153, 100), (235, 137, 25), (126, 70, 190), (10, 157, 177)
))
RIGHT_COLORS = tuple(np.array(color, dtype=np.int16) for color in (
    (119, 63, 45), (224, 91, 168), (88, 126, 35), (34, 65, 139), (191, 103, 26), (72, 72, 78)
))
CELL = 58
GRID_Y = 270
LEFT_X = 88
RIGHT_X = 1024
KEY_LEFT_X = 610
KEY_RIGHT_X = 842
KEY_Y = 284
KEY_STEP = 57


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _tag_in_crop(crop: np.ndarray, colors: tuple[np.ndarray, ...]) -> int | None:
    scores = [int(_mask(crop, color).sum()) for color in colors]
    winners = [index for index, score in enumerate(scores) if score > 170]
    return winners[0] if len(winners) == 1 else None


def _read_board(arr: np.ndarray, x0: int, colors: tuple[np.ndarray, ...]) -> tuple[list[list[int]], list[int]] | None:
    matrix: list[list[int]] = []
    tags: list[int] = []
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        tag = _tag_in_crop(arr[cy - 22:cy + 23, x0 - 69:x0 - 20], colors)
        if tag is None:
            return None
        tags.append(tag)
        values = []
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            center = arr[cy - 9:cy + 10, cx - 9:cx + 10]
            values.append(1 if int(_mask(center, INK).sum()) > 250 else 0)
        matrix.append(values)
    if sorted(tags) != list(range(N)) or any(sum(row) != 3 for row in matrix):
        return None
    return matrix, tags


def _read_key(arr: np.ndarray) -> list[int] | None:
    mapping: list[int | None] = [None] * N
    for row in range(N):
        cy = KEY_Y + row * KEY_STEP
        left = _tag_in_crop(arr[cy - 22:cy + 23, KEY_LEFT_X - 23:KEY_LEFT_X + 24], LEFT_COLORS)
        right = _tag_in_crop(arr[cy - 22:cy + 23, KEY_RIGHT_X - 23:KEY_RIGHT_X + 24], RIGHT_COLORS)
        if left is None or right is None or mapping[left] is not None:
            return None
        mapping[left] = right
    if sorted(value for value in mapping if value is not None) != list(range(N)):
        return None
    return [int(value) for value in mapping]


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {tag: row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (1460, 800):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        left = _read_board(arr, LEFT_X, LEFT_COLORS)
        right = _read_board(arr, RIGHT_X, RIGHT_COLORS)
        mapping = _read_key(arr)
        if left is None or right is None or mapping is None:
            return "abstain"
        starts = []
        for col in range(N):
            cx = LEFT_X + col * CELL + CELL // 2
            starts.append(int(_mask(arr[179:229, cx - 27:cx + 28], START).sum()) > 750)
        if sum(starts) != 1:
            return "abstain"
        left_sigs = _signatures(*left)
        right_matrix, right_tags = right
        right_by_tag = {tag: row for tag, row in zip(right_tags, right_matrix, strict=True)}
        translated_right = [
            tuple(right_by_tag[mapping[left_tag]][col] for left_tag in range(N))
            for col in range(N)
        ]
        if len(set(left_sigs)) != N or set(left_sigs) != set(translated_right):
            return "abstain"
        target = left_sigs[starts.index(True)]
        matches = [index for index, signature in enumerate(translated_right) if signature == target]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except Exception:
        return "abstain"
