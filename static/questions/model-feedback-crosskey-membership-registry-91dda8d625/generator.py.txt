"""Deterministic cross-key membership boards for correspondence binding."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1460, 800
N = 6
DECISIONS = tuple("ABCDEF")
INK = (35, 39, 48)
PAPER = (246, 247, 244)
WHITE = (255, 255, 255)
GRID = (116, 123, 132)
START = (198, 38, 139)
LEFT_TAG_COLORS = (
    (220, 55, 62),
    (35, 105, 210),
    (20, 153, 100),
    (235, 137, 25),
    (126, 70, 190),
    (10, 157, 177),
)
RIGHT_TAG_COLORS = (
    (119, 63, 45),
    (224, 91, 168),
    (88, 126, 35),
    (34, 65, 139),
    (191, 103, 26),
    (72, 72, 78),
)
CELL = 58
GRID_Y = 270
LEFT_X = 88
RIGHT_X = 1024
KEY_LEFT_X = 610
KEY_RIGHT_X = 842
KEY_Y = 284
KEY_STEP = 57
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

# A regular incidence family: each row and column has weight three, signatures
# are distinct, and every pair differs in at least two positions.
_CANONICAL = tuple(
    tuple(1 if row in {col, (col + 1) % N, (col + 3) % N} else 0 for col in range(N))
    for row in range(N)
)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _valid_matrix(matrix: object) -> bool:
    return (
        isinstance(matrix, list)
        and len(matrix) == N
        and all(isinstance(row, list) and len(row) == N and all(v in (0, 1) for v in row) for row in matrix)
    )


def _signatures(matrix: list[list[int]], tags: list[int]) -> list[tuple[int, ...]]:
    by_tag = {int(tag): row for tag, row in zip(tags, matrix, strict=True)}
    return [tuple(by_tag[tag][col] for tag in range(N)) for col in range(N)]


def _translated_right_signatures(scene: dict) -> list[tuple[int, ...]]:
    by_right_tag = {
        int(tag): row
        for tag, row in zip(scene["right_row_tags"], scene["right_matrix"], strict=True)
    }
    mapping = [int(value) for value in scene["tag_mapping"]]
    return [
        tuple(by_right_tag[mapping[left_tag]][col] for left_tag in range(N))
        for col in range(N)
    ]


def _random_derangement(rng: random.Random) -> list[int]:
    while True:
        values = list(range(N))
        rng.shuffle(values)
        if all(index != value for index, value in enumerate(values)):
            return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    entities = list(range(N))
    rng.shuffle(entities)
    left_tags = list(range(N))
    right_tags = list(range(N))
    key_order = list(range(N))
    rng.shuffle(left_tags)
    rng.shuffle(right_tags)
    rng.shuffle(key_order)
    tag_mapping = _random_derangement(rng)

    start_index = rng.randrange(N)
    target = int(seed) % N
    column_map: list[int | None] = [None] * N
    column_map[start_index] = target
    remaining = [index for index in range(N) if index != target]
    rng.shuffle(remaining)
    for left_col, right_col in zip(
        (index for index in range(N) if index != start_index), remaining, strict=True
    ):
        column_map[left_col] = right_col

    left = [[_CANONICAL[tag][entity] for entity in entities] for tag in left_tags]
    inverse_mapping = {right_tag: left_tag for left_tag, right_tag in enumerate(tag_mapping)}
    right = [[0 for _ in range(N)] for _ in range(N)]
    for right_row, right_tag in enumerate(right_tags):
        left_tag = inverse_mapping[right_tag]
        for left_col, entity in enumerate(entities):
            right[right_row][int(column_map[left_col])] = _CANONICAL[left_tag][entity]

    scene = {
        "left_matrix": left,
        "right_matrix": right,
        "left_row_tags": left_tags,
        "right_row_tags": right_tags,
        "key_left_order": key_order,
        "tag_mapping": tag_mapping,
        "start_index": start_index,
    }
    if is_quarantined(scene):
        raise RuntimeError("constructed cross-key registry failed invariants")
    return scene


def analytic_gold(scene: dict) -> str:
    left = _signatures(scene["left_matrix"], scene["left_row_tags"])
    right = _translated_right_signatures(scene)
    target = left[int(scene["start_index"])]
    matches = [index for index, signature in enumerate(right) if signature == target]
    if len(matches) != 1:
        raise ValueError("START does not have one translated counterpart")
    return DECISIONS[matches[0]]


def margin(scene: dict) -> float:
    if is_quarantined(scene):
        return 0.0
    left = _signatures(scene["left_matrix"], scene["left_row_tags"])
    right = _translated_right_signatures(scene)
    target = left[int(scene["start_index"])]
    return float(min(sum(a != b for a, b in zip(target, other, strict=True)) for other in right if other != target))


def is_quarantined(scene: dict) -> bool:
    try:
        left, right = scene["left_matrix"], scene["right_matrix"]
        left_tags, right_tags = scene["left_row_tags"], scene["right_row_tags"]
        key_order = scene["key_left_order"]
        mapping = [int(value) for value in scene["tag_mapping"]]
        if not _valid_matrix(left) or not _valid_matrix(right):
            return True
        if any(sorted(values) != list(range(N)) for values in (left_tags, right_tags, key_order, mapping)):
            return True
        if any(index == mapping[index] for index in range(N)):
            return True
        if int(scene["start_index"]) not in range(N):
            return True
        left_sigs = _signatures(left, left_tags)
        right_sigs = _translated_right_signatures(scene)
        if len(set(left_sigs)) != N or len(set(right_sigs)) != N or set(left_sigs) != set(right_sigs):
            return True
        if any(sum(signature) != 3 for signature in left_sigs + right_sigs):
            return True
        if any(sum(row) != 3 for row in left + right):
            return True
        target = left_sigs[int(scene["start_index"])]
        return right_sigs.count(target) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, size: int, fill: tuple[int, int, int]) -> None:
    font = _font(size)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - 2), text, font=font, fill=fill)


def _draw_badge(draw: ImageDraw.ImageDraw, cx: int, cy: int, tag: int, side: str) -> None:
    colors = LEFT_TAG_COLORS if side == "left" else RIGHT_TAG_COLORS
    color = colors[tag]
    if tag == 0:
        draw.polygon(((cx, cy - 18), (cx - 20, cy + 17), (cx + 20, cy + 17)), fill=color, outline=INK)
    elif tag == 1:
        draw.rectangle((cx - 17, cy - 17, cx + 17, cy + 17), fill=color, outline=INK, width=2)
    elif tag == 2:
        draw.ellipse((cx - 18, cy - 18, cx + 18, cy + 18), fill=color, outline=INK, width=2)
    elif tag == 3:
        draw.polygon(((cx, cy - 19), (cx - 19, cy), (cx, cy + 19), (cx + 19, cy)), fill=color, outline=INK)
    elif tag == 4:
        draw.polygon(((cx - 17, cy - 9), (cx, cy - 19), (cx + 17, cy - 9), (cx + 17, cy + 9), (cx, cy + 19), (cx - 17, cy + 9)), fill=color, outline=INK)
    else:
        draw.rectangle((cx - 6, cy - 19, cx + 6, cy + 19), fill=color, outline=INK, width=2)
        draw.rectangle((cx - 19, cy - 6, cx + 19, cy + 6), fill=color, outline=INK, width=2)


def _draw_board(draw: ImageDraw.ImageDraw, x0: int, matrix: list[list[int]], tags: list[int], labels: tuple[str, ...], caption: str, start: int | None, side: str) -> None:
    _centered_text(draw, (x0 + N * CELL // 2, 138), caption, 28, INK)
    for col, label in enumerate(labels):
        cx = x0 + col * CELL + CELL // 2
        if col == start:
            draw.rounded_rectangle((cx - 25, 181, cx + 25, 227), radius=11, fill=START)
            _centered_text(draw, (cx, 204), label, 22, WHITE)
            _centered_text(draw, (cx, 169), "START", 14, START)
        else:
            draw.ellipse((cx - 20, 184, cx + 20, 224), fill=WHITE, outline=INK, width=3)
            _centered_text(draw, (cx, 204), label, 21, INK)
    for index in range(N + 1):
        draw.line((x0, GRID_Y + index * CELL, x0 + N * CELL, GRID_Y + index * CELL), fill=GRID, width=2)
        draw.line((x0 + index * CELL, GRID_Y, x0 + index * CELL, GRID_Y + N * CELL), fill=GRID, width=2)
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        _draw_badge(draw, x0 - 45, cy, int(tags[row]), side)
        for col in range(N):
            cx = x0 + col * CELL + CELL // 2
            draw.ellipse((cx - 15, cy - 15, cx + 15, cy + 15), fill=INK if matrix[row][col] else WHITE, outline=INK, width=4)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 44), "CROSS-KEY MEMBERSHIP", 35, INK)
    _draw_board(draw, LEFT_X, scene["left_matrix"], scene["left_row_tags"], tuple(str(i + 1) for i in range(N)), "SOURCE", int(scene["start_index"]), "left")
    _draw_board(draw, RIGHT_X, scene["right_matrix"], scene["right_row_tags"], DECISIONS, "TARGET", None, "right")
    _centered_text(draw, ((KEY_LEFT_X + KEY_RIGHT_X) // 2, 138), "BADGE KEY", 28, INK)
    draw.rounded_rectangle((548, 242, 904, 637), radius=16, fill=WHITE, outline=GRID, width=3)
    for key_row, left_tag in enumerate(scene["key_left_order"]):
        cy = KEY_Y + key_row * KEY_STEP
        _draw_badge(draw, KEY_LEFT_X, cy, int(left_tag), "left")
        draw.line((655, cy, 797, cy), fill=INK, width=4)
        draw.polygon(((797, cy), (781, cy - 9), (781, cy + 9)), fill=INK)
        _draw_badge(draw, KEY_RIGHT_X, cy, int(scene["tag_mapping"][left_tag]), "right")
    draw.ellipse((420, 697, 450, 727), fill=INK, outline=INK)
    draw.text((463, 695), "filled = member", font=_font(20), fill=INK)
    draw.ellipse((704, 697, 734, 727), fill=WHITE, outline=INK, width=4)
    draw.text((747, 695), "hollow = not member", font=_font(20), fill=INK)
    _centered_text(draw, (WIDTH // 2, 765), "Translate row badges through the center key.", 20, INK)
    return image
