"""Independent pixel-only inverse arm for the vanishing-point-selection world.

Recovers which color marker sits on the shared convergence point of the colored
line segments from the final PNG alone. It never sees the latent scene, analytic
gold, or renderer code. It segments each colored line by palette, fits each to
its supporting line, computes their shared (least-squares) intersection, then
selects the color marker nearest that point.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

BACKGROUND = (255, 255, 255)
LINE_COLORS = {
    "magenta": (220, 40, 220),
    "teal": (20, 190, 200),
    "amber": (240, 190, 40),
    "lime": (110, 220, 70),
}
LINE_NAMES = ["magenta", "teal", "amber", "lime"]
MARKER_COLORS = {
    "red": (230, 50, 50),
    "green": (50, 190, 90),
    "blue": (50, 90, 230),
    "purple": (170, 60, 200),
}
MARKER_NAMES = ["red", "green", "blue", "purple"]

_TH = 40


def _mask(arr, rgb):
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    return (
        (np.abs(r - rgb[0]) < _TH) & (np.abs(g - rgb[1]) < _TH) & (np.abs(b - rgb[2]) < _TH)
    )


def _fit_line(mask: np.ndarray):
    coords = np.argwhere(mask)
    if len(coords) < 30:
        return None
    ys = coords[:, 0].astype(float)
    xs = coords[:, 1].astype(float)
    px, py = xs.mean(), ys.mean()
    cov = np.cov(np.stack([xs, ys]))
    eigvals, eigvecs = np.linalg.eigh(cov)
    u = eigvecs[:, int(np.argmax(eigvals))]
    norm = math.hypot(u[0], u[1])
    if norm == 0.0:
        return None
    u = u / norm
    return (float(u[0]), float(u[1]), float(px), float(py))


def _convergence_point(lines) -> tuple[float, float]:
    ux = np.array([line[0] for line in lines])
    uy = np.array([line[1] for line in lines])
    px = np.array([line[2] for line in lines])
    py = np.array([line[3] for line in lines])
    nx = -uy
    ny = ux
    b = nx * px + ny * py
    A = np.stack([nx, ny], axis=1)
    try:
        x, y = np.linalg.solve(A.T @ A, A.T @ b)
    except np.linalg.LinAlgError:
        x, y = px.mean(), py.mean()
    return (float(x), float(y))


def _marker_center(mask: np.ndarray):
    coords = np.argwhere(mask)
    if len(coords) < 20:
        return None
    return (float(coords[:, 1].mean()), float(coords[:, 0].mean()))


def decision_from_image(image: Image.Image) -> str:
    """The color name of the marker sitting on the shared convergence point."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)

    lines = []
    for color in LINE_NAMES:
        line = _fit_line(_mask(arr, LINE_COLORS[color]))
        if line is not None:
            lines.append(line)
    if not lines:
        return MARKER_NAMES[0]

    px, py = _convergence_point(lines)

    best_color = None
    best_dist = math.inf
    found = 0
    for color in MARKER_NAMES:
        center = _marker_center(_mask(arr, MARKER_COLORS[color]))
        if center is None:
            continue
        found += 1
        dist = math.hypot(center[0] - px, center[1] - py)
        if dist < best_dist:
            best_dist = dist
            best_color = color

    if best_color is None:
        return MARKER_NAMES[0]
    return best_color
