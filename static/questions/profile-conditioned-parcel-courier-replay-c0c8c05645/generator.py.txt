"""Scene sampler and Pillow renderer for courier/parcel state replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 820, 520
BACKGROUND = (248, 247, 242)
INK = (30, 32, 36)
PALETTES = (
    {"courier": (25, 110, 205), "parcel": (200, 40, 135), "move": (40, 155, 95)},
    {"courier": (20, 150, 145), "parcel": (185, 55, 175), "move": (215, 105, 35)},
    {"courier": (95, 75, 190), "parcel": (220, 65, 75), "move": (25, 140, 190)},
)


def _replay(courier_start: int, parcel_start: int, commands: list[dict[str, Any]]) -> tuple[int, int, bool, dict[str, int]]:
    courier = int(courier_start)
    parcel = int(parcel_start)
    carried = False
    stats = {"pick": 0, "drop": 0, "carried_move": 0, "noop": 0}
    for command in commands:
        op = command["op"]
        if op == "move":
            step = 1 if command["direction"] == "right" else -1
            destination = courier + step
            if 1 <= destination <= 4:
                courier = destination
                if carried:
                    parcel = courier
                    stats["carried_move"] += 1
            else:
                stats["noop"] += 1
        elif op == "pick":
            if not carried and courier == parcel:
                carried = True
                stats["pick"] += 1
            else:
                stats["noop"] += 1
        elif op == "drop":
            if carried:
                parcel = courier
                carried = False
                stats["drop"] += 1
            else:
                stats["noop"] += 1
        else:
            raise ValueError(f"unknown operation {op!r}")
    return courier, parcel, carried, stats


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed modulo four fixes the final parcel room."""
    rng = random.Random(int(seed))
    target = int(seed) % 4 + 1
    palette_variant = rng.randrange(len(PALETTES))
    shift = rng.choice([-10, -5, 0, 5, 10])
    decorations = [rng.randrange(3) for _ in range(6)]
    for _ in range(60000):
        courier_start = rng.randrange(1, 5)
        parcel_start = rng.randrange(1, 5)
        if courier_start == parcel_start or parcel_start == target:
            continue
        commands: list[dict[str, Any]] = []
        for _index in range(6):
            op = rng.choices(("move", "pick", "drop"), weights=(6, 2, 2), k=1)[0]
            if op == "move":
                commands.append({"op": "move", "direction": rng.choice(("left", "right"))})
            else:
                commands.append({"op": op})
        _, final, carried, stats = _replay(courier_start, parcel_start, commands)
        kinds = {command["op"] for command in commands}
        if (
            final == target
            and not carried
            and stats["pick"] >= 1
            and stats["drop"] >= 1
            and stats["carried_move"] >= 1
            and kinds == {"move", "pick", "drop"}
        ):
            return {
                "courier_start": courier_start,
                "parcel_start": parcel_start,
                "commands": commands,
                "palette_variant": palette_variant,
                "shift": shift,
                "decorations": decorations,
            }
    raise RuntimeError(f"could not sample answer stratum {target}")


def _dot_row(draw: ImageDraw.ImageDraw, center_x: int, y: int, count: int, radius: int = 5) -> None:
    spacing = 16
    start = center_x - spacing * (count - 1) / 2
    for index in range(count):
        x = int(round(start + index * spacing))
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=INK)


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: str, color: tuple[int, int, int]) -> None:
    sign = 1 if direction == "right" else -1
    draw.rectangle((cx - 28, cy - 7, cx + 28, cy + 7), fill=color)
    tip = cx + sign * 38
    base = cx + sign * 8
    draw.polygon([(tip, cy), (base, cy - 26), (base, cy + 26)], fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=15)
    palette = PALETTES[int(scene["palette_variant"])]
    shift = int(scene["shift"])

    draw.text((34 + shift, 24), "START: rooms 1 to 4", fill=INK, font=font)
    room_lefts = [45 + shift + 190 * index for index in range(4)]
    room_boxes = []
    for index, left in enumerate(room_lefts, start=1):
        box = (left, 70, left + 135, 210)
        room_boxes.append(box)
        draw.rectangle(box, outline=INK, width=4)
        _dot_row(draw, left + 67, 94, index, radius=5)

    courier_box = room_boxes[int(scene["courier_start"]) - 1]
    courier_x = (courier_box[0] + courier_box[2]) // 2
    draw.ellipse((courier_x - 24, 116, courier_x + 24, 164), outline=palette["courier"], width=8)
    draw.line((courier_x - 13, 140, courier_x + 13, 140), fill=palette["courier"], width=7)
    draw.line((courier_x, 127, courier_x, 153), fill=palette["courier"], width=7)

    parcel_box = room_boxes[int(scene["parcel_start"]) - 1]
    parcel_x = (parcel_box[0] + parcel_box[2]) // 2
    draw.rounded_rectangle((parcel_x - 23, 170, parcel_x + 23, 204), radius=5, fill=palette["parcel"])
    draw.line((parcel_x, 170, parcel_x, 204), fill=BACKGROUND, width=4)

    draw.text((34 + shift, 245), "CARDS: read left to right", fill=INK, font=font)
    for index, command in enumerate(scene["commands"]):
        left = 24 + shift + 130 * index
        top, right, bottom = 280, left + 108, 474
        draw.rectangle((left, top, right, bottom), outline=INK, width=4)
        cx = (left + right) // 2
        if command["op"] == "move":
            _arrow(draw, cx, 365, command["direction"], palette["move"])
        elif command["op"] == "pick":
            draw.ellipse((cx - 25, 337, cx + 25, 387), fill=palette["parcel"])
        else:
            draw.ellipse((cx - 25, 337, cx + 25, 387), outline=palette["parcel"], width=8)
        for mark in range(int(scene["decorations"][index])):
            x = left + 22 + mark * 18
            draw.polygon([(x, 447), (x + 5, 452), (x, 457), (x - 5, 452)], fill=(180, 182, 180))
    return image


def analytic_gold(scene: dict) -> str:
    _, parcel, _, _ = _replay(scene["courier_start"], scene["parcel_start"], scene["commands"])
    return str(parcel)


def margin(scene: dict) -> float:
    _ = (scene["courier_start"], scene["parcel_start"], scene["commands"])
    return 1.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["courier_start"], scene["parcel_start"], scene["commands"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
