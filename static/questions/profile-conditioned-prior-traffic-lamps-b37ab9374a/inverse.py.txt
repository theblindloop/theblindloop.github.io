"""Independent pixel-only inverse arm for the traffic-signal lamp world.

Receives ONLY the final PNG and returns the number of lamps as a string. It
shares no scene constants with the renderer (only the generic facts, visible in
the image, that the lamps are saturated-coloured round disks on a neutral dark
backplate against a plain white background). It never imports renderer, prompts,
generate, verify, or analytic-gold code, and accepts no latent scene data.

Method: the lamps are the only saturated-colour pixels on the canvas -- the
housing is neutral grey and the background is white. The oracle thresholds the
raster on colour saturation to build a lamp mask, then counts the 8-connected
components of that mask. Because every lamp is a separate, non-touching disk,
one connected component corresponds to exactly one lamp, robustly across the
number of lamps, their colours, the housing size and the signal's position.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_traffic_lamps-oracle-0.1.0"

_SATURATION_MIN = 60.0
_BRIGHTNESS_MIN = 90.0


def _lamp_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.float32)
    g = arr[..., 1].astype(np.float32)
    b = arr[..., 2].astype(np.float32)
    mx = np.maximum(np.maximum(r, g), b)
    mn = np.minimum(np.minimum(r, g), b)
    saturation = mx - mn
    return (saturation >= _SATURATION_MIN) & (mx >= _BRIGHTNESS_MIN)


def _count_components(mask: np.ndarray) -> int:
    """Count 8-connected components of a boolean mask (no scipy dependency)."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    label = 0
    for i in range(h):
        row_indices = np.nonzero(mask[i])[0]
        for j in row_indices:
            if visited[i, j]:
                continue
            label += 1
            stack = [(i, j)]
            visited[i, j] = True
            while stack:
                x, y = stack.pop()
                x0 = max(0, x - 1)
                x1 = min(h, x + 2)
                y0 = max(0, y - 1)
                y1 = min(w, y + 2)
                region = mask[x0:x1, y0:y1] & ~visited[x0:x1, y0:y1]
                ys, xs = np.nonzero(region)
                for xx, yy in zip(ys + x0, xs + y0):
                    visited[xx, yy] = True
                    stack.append((xx, yy))
    return label


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    mask = _lamp_mask(arr)
    return str(_count_components(mask))


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"))
    mask = _lamp_mask(arr)
    return {
        "lamp_px": int(mask.sum()),
        "components": _count_components(mask),
        "size": image.size,
    }
