"""Traffic-signal lamp-count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of round lamps) when the familiar canonical prior ("a traffic signal has
exactly three lamps: red, yellow, green") is contradicted by the actual image,
which in the counterfactual scenes shows a traffic signal with a visibly
different number of lamps.

Latent z = {lamps, radius, spacing, cx, top, housing_w, housing_h, colors,
width, height}. ``lamps`` is the number of round lamps actually painted in the
signal housing, ``radius`` the lamp radius in pixels, ``spacing`` the vertical
centre-to-centre distance between adjacent lamps, ``cx`` the horizontal centre
of the housing, ``top`` the vertical centre of the topmost lamp, ``housing_w``
and ``housing_h`` the housing backplate size, ``colors`` the per-lamp paint
colours, and ``width``/``height`` the canvas size. The closed-form gold is

    decision = str(lamps)

a scene parameter, not a pixel measurement.

The renderer paints a dark, rounded housing (a signal backplate) on a plain
white background, then draws ``lamps`` filled, saturated-coloured round lamps in
a vertical column down the centre of the housing. Each lamp is a separate,
non-touching disk so the image unambiguously reads as a traffic signal, and the
lamps are the only saturated-coloured pixels on the canvas (the housing is
neutral dark grey, the background is white). The pixel-only inverse arm
therefore counts the lamps by segmenting the saturated-colour pixels and
counting the resulting connected components, one per lamp.

The continuous verifiability margin is the edge-to-edge gap between adjacent
lamps in pixels: the minimum separation that keeps neighbouring lamps visually
resolvable as distinct disks. When the lamps crowd together the count becomes
pixel-ambiguous, so such scenes are quarantined.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_traffic_lamps-0.1.0"

CANVAS = (432, 432)
BG = (255, 255, 255)          # white paper background
HOUSING = (46, 48, 54)        # neutral dark-grey signal housing (low saturation)
RIM = (22, 23, 27)            # darker housing rim (kept neutral / low saturation)
PALETTE = [
    (222, 34, 60),    # red
    (246, 206, 34),   # yellow
    (52, 178, 96),    # green
    (40, 122, 224),   # blue
    (204, 62, 216),   # magenta
    (36, 202, 198),   # cyan
    (250, 146, 30),   # orange
]
SUPPORTED_LAMPS = (2, 3, 4, 5)
MIN_RADIUS = 11
MIN_EDGE_GAP = 6


def margin(scene: dict) -> float:
    """Edge-to-edge gap (px) between adjacent lamps: the resolvable separation."""
    return float(int(scene["spacing"]) - 2 * int(scene["radius"]))


def is_quarantined(scene: dict) -> bool:
    lamps = int(scene["lamps"])
    radius = int(scene["radius"])
    spacing = int(scene["spacing"])
    w, h = int(scene["width"]), int(scene["height"])
    hw, hh = int(scene["housing_w"]), int(scene["housing_h"])
    cx = int(scene["cx"])
    top = int(scene["top"])
    if lamps not in SUPPORTED_LAMPS:
        return True
    if radius < MIN_RADIUS:
        return True
    if spacing - 2 * radius < MIN_EDGE_GAP:
        return True
    if cx - hw // 2 < 2 or cx + hw // 2 > w - 2:
        return True
    if top - (hh // 2) < 2 or top + (hh - hh // 2) > h - 2:
        return True
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    w, h = CANVAS
    while True:
        lamps = rng.choice(SUPPORTED_LAMPS)
        radius = rng.randint(MIN_RADIUS, 20)
        edge_gap = rng.randint(MIN_EDGE_GAP, 22)
        spacing = 2 * radius + edge_gap
        colors = [list(rng.choice(PALETTE)) for _ in range(lamps)]
        v_margin = rng.randint(10, 26)
        h_margin = rng.randint(16, 30)
        housing_w = 2 * (radius + h_margin)
        housing_h = spacing * (lamps - 1) + 2 * (radius + v_margin)
        cx = rng.randint(housing_w // 2 + 2, w - housing_w // 2 - 2)
        top = rng.randint(radius + v_margin + 2, h - (radius + v_margin) - 2 - (spacing * (lamps - 1)))
        scene = {
            "lamps": lamps,
            "radius": radius,
            "spacing": spacing,
            "cx": cx,
            "top": top,
            "housing_w": housing_w,
            "housing_h": housing_h,
            "colors": colors,
            "width": w,
            "height": h,
        }
        if not is_quarantined(scene):
            return scene


def _round_rect(draw: ImageDraw.ImageDraw, box, radius, fill):
    x0, y0, x1, y1 = box
    r = radius
    draw.rectangle([x0 + r, y0, x1 - r, y1], fill=fill)
    draw.rectangle([x0, y0 + r, x1, y1 - r], fill=fill)
    draw.ellipse([x0, y0, x0 + 2 * r, y0 + 2 * r], fill=fill)
    draw.ellipse([x1 - 2 * r, y0, x1, y0 + 2 * r], fill=fill)
    draw.ellipse([x0, y1 - 2 * r, x0 + 2 * r, y1], fill=fill)
    draw.ellipse([x1 - 2 * r, y1 - 2 * r, x1, y1], fill=fill)


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    lamps = int(scene["lamps"])
    radius = int(scene["radius"])
    spacing = int(scene["spacing"])
    cx = int(scene["cx"])
    top = int(scene["top"])
    hw, hh = int(scene["housing_w"]), int(scene["housing_h"])
    colors = [tuple(int(v) for v in color) for color in scene["colors"]]

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)

    x0 = cx - hw // 2
    y0 = top - hh // 2
    rim = 4
    # darker outer rim, then the neutral housing backplate
    _round_rect(draw, [x0 - rim, y0 - rim, x0 + hw + rim, y0 + hh + rim], 18, RIM)
    _round_rect(draw, [x0, y0, x0 + hw, y0 + hh], 14, HOUSING)

    # vertical column of lamps, each a separate saturated-coloured disk
    for index in range(lamps):
        cy = top + index * spacing
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=colors[index],
        )
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["lamps"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent alias exists: the scene maps injectively.

    The lamp count, lamp radius, spacing, housing extents, position and the
    per-lamp colours are all directly visible in the raster, and changing any
    declared field changes the pixels. Changing the lamp count changes the
    number of painted disks, which is impossible to recover from any other
    field. This is a `not_applicable` latent-alias mode, so no transforms are
    declared.
    """
    return []
