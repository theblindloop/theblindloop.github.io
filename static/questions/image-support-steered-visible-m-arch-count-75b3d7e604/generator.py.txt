"""Deterministic renderer and analytic arm for visible arch counting."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


PALETTES = (
    ((28, 92, 196), (108, 112, 120), (250, 248, 242)),
    ((116, 52, 190), (92, 100, 108), (248, 250, 246)),
    ((0, 126, 132), (104, 104, 112), (252, 248, 244)),
    ((210, 76, 28), (94, 102, 112), (247, 250, 252)),
)


def sample_scene(seed: int) -> dict:
    """Sample matched triples: consecutive seeds share style but vary count."""
    high_count = 1 + (int(seed) % 3)
    style_rng = random.Random(30_000 + int(seed) // 3)
    ink, guide, background = PALETTES[style_rng.randrange(len(PALETTES))]
    slot_rng = random.Random(91_000 + int(seed))
    high_slots = sorted(slot_rng.sample(range(4), high_count))
    return {
        "high_count": high_count,
        "high_slots": high_slots,
        "left": style_rng.randint(102, 116),
        "baseline_y": style_rng.randint(322, 344),
        "total_width": style_rng.randint(286, 300),
        "crest_offset": style_rng.randint(91, 105),
        "guide_gap": style_rng.randint(17, 23),
        "stroke_width": style_rng.randint(9, 12),
        "ink_color": list(ink),
        "guide_color": list(guide),
        "background": list(background),
        "width": 512,
        "height": 512,
    }


def _points(scene: dict, scale: int = 1) -> list[tuple[int, int]]:
    left = float(scene["left"])
    baseline = float(scene["baseline_y"])
    total_width = float(scene["total_width"])
    crest_offset = float(scene["crest_offset"])
    guide_gap = float(scene["guide_gap"])
    high_slots = {int(value) for value in scene["high_slots"]}
    high_count = int(scene["high_count"])
    if high_count != len(high_slots) or not high_slots <= set(range(4)):
        raise ValueError("inconsistent high-arch specification")
    slot_width = total_width / 4.0
    guide_y = baseline - crest_offset + guide_gap
    low_crest_y = guide_y + guide_gap
    points: list[tuple[int, int]] = []
    samples_per_slot = 80
    for slot in range(4):
        crest_y = guide_y - guide_gap if slot in high_slots else low_crest_y
        rise = baseline - crest_y
        for step in range(samples_per_slot + 1):
            u = step / samples_per_slot
            x = left + slot_width * (slot + u)
            y = baseline - rise * math.sin(math.pi * u) ** 1.22
            points.append((round(x * scale), round(y * scale)))
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(int(v) for v in scene["background"])
    ink = tuple(int(v) for v in scene["ink_color"])
    guide = tuple(int(v) for v in scene["guide_color"])
    left = int(scene["left"])
    baseline = int(scene["baseline_y"])
    total_width = int(scene["total_width"])
    crest_offset = int(scene["crest_offset"])
    guide_gap = int(scene["guide_gap"])
    stroke_width = int(scene["stroke_width"])
    # Access and validate the answer-bearing field in the rendering arm.
    if int(scene["high_count"]) != len(scene["high_slots"]):
        raise ValueError("high_count disagrees with high_slots")
    canvas = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(canvas)
    guide_y = (baseline - crest_offset + guide_gap) * scale
    guide_left = (left - 10) * scale
    guide_right = (left + total_width + 10) * scale
    dash, gap = 13 * scale, 8 * scale
    x = guide_left
    while x < guide_right:
        draw.line((x, guide_y, min(x + dash, guide_right), guide_y), fill=guide, width=2 * scale)
        x += dash + gap
    draw.line(_points(scene, scale), fill=ink, width=stroke_width * scale, joint="curve")
    return canvas.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["high_count"]))


def margin(scene: dict) -> float:
    # Both high and low crests are guide_gap pixels from the visible guide.
    return float(scene["guide_gap"]) - float(scene["stroke_width"]) / 2.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    transformed = dict(scene)
    transformed["high_slots"] = list(reversed(scene["high_slots"]))
    return [("reverse_high_slot_list", transformed)]
