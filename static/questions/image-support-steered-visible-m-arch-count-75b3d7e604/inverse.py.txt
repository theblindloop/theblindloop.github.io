"""Independent final-PNG inverse arm; no latent scene information is used."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(y, x)])
        group: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            group.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(values.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1) - 1
    return [(int(a), int(b)) for a, b in zip(starts, ends, strict=True)]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas")
    border = np.concatenate((array[:4].reshape(-1, 3), array[-4:].reshape(-1, 3), array[:, :4].reshape(-1, 3), array[:, -4:].reshape(-1, 3)))
    background = np.median(border, axis=0)
    spread = array.max(axis=2) - array.min(axis=2)
    delta = np.linalg.norm(array - background, axis=2)

    # Saturated ink is the one continuous answer-bearing trace.
    trace = (spread > 62) & (delta > 72)
    groups = [group for group in _components(trace) if len(group) >= 80]
    if len(groups) != 1:
        raise ValueError("trace is missing, broken, or ambiguous")
    trace = np.zeros(trace.shape, dtype=bool)
    ys, xs = zip(*groups[0])
    trace[np.asarray(ys), np.asarray(xs)] = True
    y0, y1, x0, x1 = min(ys), max(ys), min(xs), max(xs)
    if not (270 <= x1 - x0 <= 325 and 65 <= y1 - y0 <= 135 and len(groups[0]) >= 1800):
        raise ValueError("trace geometry is corrupted")

    # The low-saturation dashed row is the public qualification boundary.
    achromatic = (spread < 28) & (delta > 42) & (array.mean(axis=2) < 205)
    row_counts = achromatic.sum(axis=1)
    guide_y = int(np.argmax(row_counts))
    if row_counts[guide_y] < 75:
        raise ValueError("visible dashed guide is missing")

    top = np.full(array.shape[1], array.shape[0], dtype=np.int32)
    for x in range(x0, x1 + 1):
        column = np.flatnonzero(trace[:, x])
        if column.size:
            top[x] = int(column.min())
    qualifying = top[x0:x1 + 1] < guide_y - 3
    runs = [(a, b) for a, b in _runs(qualifying) if b - a + 1 >= 13]
    count = len(runs)
    if count not in (1, 2, 3):
        raise ValueError("arch count is outside the public answer set")
    if any(b - a + 1 < 18 for a, b in runs):
        raise ValueError("guide-crossing arch is too narrow")
    return str(count)
