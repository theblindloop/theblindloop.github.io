"""Key/socket interlock correspondence world.

Four gray keys (top) and four colored sockets (bottom). Each key is a stack of
four left-aligned bars; each socket is the same kind of stack drawn rotated by
180 degrees (bars right-aligned, row order reversed). A socket interlocks with a
key when, laid onto it, every meeting pair of bars has the same combined length,
i.e. the two stacks together form a flush block with no gap and no overlap.

All four key profiles carry the same total bar length, so total ink, mean bar
length and every per-object magnitude cue are constant across keys and across
sockets; only the arrangement of the bars decides the assignment.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 560
BACKGROUND = (255, 255, 255)

SLOT_CENTERS = (100, 280, 460, 640)
FULL = 180  # combined length of an interlocking bar pair, in pixels
HALF = FULL // 2

BAR_HEIGHT = 26
BAR_GAP = 6
ROWS = 4
STACK_HEIGHT = ROWS * BAR_HEIGHT + (ROWS - 1) * BAR_GAP

KEY_TOP = 120
SOCKET_TOP = 330
MARKER_TOP = 78
MARKER_BOTTOM = 110
MARKER_HALF_WIDTH = 22

KEY_COLOR = (135, 135, 135)
MARKER_COLOR = (0, 0, 0)

SOCKET_COLORS = {
    "red": (214, 40, 40),
    "orange": (240, 150, 20),
    "green": (34, 150, 64),
    "blue": (40, 80, 214),
}
COLOR_NAMES = ("red", "orange", "green", "blue")

BAR_MIN = 30
BAR_MAX = 150
BAR_SUM = ROWS * HALF  # 360: every key and every socket carries the same ink

QUARANTINE_MARGIN = 8.0


def _random_profile(rng: random.Random) -> list[int]:
    while True:
        head = [rng.randint(BAR_MIN, BAR_MAX) for _ in range(ROWS - 1)]
        last = BAR_SUM - sum(head)
        if BAR_MIN <= last <= BAR_MAX:
            return head + [last]


def _perturb(profile: list[int], rng: random.Random, delta: int) -> list[int] | None:
    """Return a sum-preserving neighbour at L-infinity distance ``delta``."""
    i, j = rng.sample(range(ROWS), 2)
    out = list(profile)
    out[i] += delta
    out[j] -= delta
    if all(BAR_MIN <= v <= BAR_MAX for v in out):
        return out
    return None


def _spread(a: list[int], b: list[int]) -> int:
    diff = [x - y for x, y in zip(a, b)]
    return max(diff) - min(diff)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(4000):
        profiles = [_random_profile(rng)]
        # With probability ~1/2 plant a close competitor so that the decision
        # margin sweeps the whole range down to the legibility boundary.
        if rng.random() < 0.5:
            delta = rng.choice([2, 3, 4, 5, 6, 8, 10, 14, 20, 28, 40])
            near = _perturb(profiles[0], rng, delta)
            if near is not None:
                profiles.append(near)
        while len(profiles) < ROWS:
            profiles.append(_random_profile(rng))
        if any(profiles[i] == profiles[j] for i in range(4) for j in range(i + 1, 4)):
            continue
        marked_slot = rng.randrange(4)
        # Independent random placement of key profiles and socket colours keeps
        # the answer colour, the answer slot and the marked slot uncorrelated.
        rng.shuffle(profiles)
        key_slots = list(range(4))
        socket_slots = list(range(4))
        rng.shuffle(socket_slots)
        colors = list(COLOR_NAMES)
        rng.shuffle(colors)

        keys = [{"slot": key_slots[i], "bars": profiles[i]} for i in range(4)]
        sockets = []
        for i in range(4):
            bars = [FULL - v for v in reversed(profiles[i])]
            sockets.append(
                {"slot": socket_slots[i], "color": colors[i], "bars": bars}
            )
        scene = {
            "keys": sorted(keys, key=lambda k: k["slot"]),
            "sockets": sorted(sockets, key=lambda s: s["slot"]),
            "marked_slot": marked_slot,
        }
        return scene
    raise RuntimeError("scene sampling failed")


def _stack_y(index: int, top: int) -> tuple[int, int]:
    y0 = top + index * (BAR_HEIGHT + BAR_GAP)
    return y0, y0 + BAR_HEIGHT - 1


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for key in sorted(scene["keys"], key=lambda k: k["slot"]):
        x0 = SLOT_CENTERS[key["slot"]] - 80
        for row, length in enumerate(key["bars"]):
            y0, y1 = _stack_y(row, KEY_TOP)
            draw.rectangle([x0, y0, x0 + length - 1, y1], fill=KEY_COLOR)

    for socket in sorted(scene["sockets"], key=lambda s: s["slot"]):
        x1 = SLOT_CENTERS[socket["slot"]] + 80
        color = SOCKET_COLORS[socket["color"]]
        for row, length in enumerate(socket["bars"]):
            y0, y1 = _stack_y(row, SOCKET_TOP)
            draw.rectangle([x1 - length + 1, y0, x1, y1], fill=color)

    cx = SLOT_CENTERS[scene["marked_slot"]]
    draw.polygon(
        [
            (cx - MARKER_HALF_WIDTH, MARKER_TOP),
            (cx + MARKER_HALF_WIDTH, MARKER_TOP),
            (cx, MARKER_BOTTOM),
        ],
        fill=MARKER_COLOR,
    )
    return image


def _marked_key(scene: dict) -> dict:
    for key in scene["keys"]:
        if key["slot"] == scene["marked_slot"]:
            return key
    raise ValueError("marked slot has no key")


def _fit_spreads(scene: dict) -> list[tuple[int, dict]]:
    """(interlock defect in pixels, socket) for every socket, marked key."""
    bars = _marked_key(scene)["bars"]
    out = []
    for socket in scene["sockets"]:
        laid = list(reversed(socket["bars"]))  # socket is drawn rotated 180 deg
        totals = [a + b for a, b in zip(bars, laid)]
        out.append((max(totals) - min(totals), socket))
    return out


def analytic_gold(scene: dict) -> str:
    ranked = sorted(_fit_spreads(scene), key=lambda item: (item[0], item[1]["slot"]))
    best = ranked[0][1]
    return f"{best['color']}|{best['slot'] + 1}"


def margin(scene: dict) -> float:
    ranked = sorted(_fit_spreads(scene), key=lambda item: (item[0], item[1]["slot"]))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the latent scene."""
    out: list[tuple[str, dict]] = []

    reversed_keys = {
        "keys": list(reversed(scene["keys"])),
        "sockets": scene["sockets"],
        "marked_slot": scene["marked_slot"],
    }
    out.append(("key_list_order_reversed", reversed_keys))

    rotated_sockets = {
        "keys": scene["keys"],
        "sockets": scene["sockets"][1:] + scene["sockets"][:1],
        "marked_slot": scene["marked_slot"],
    }
    out.append(("socket_list_order_rotated", rotated_sockets))

    both = {
        "keys": list(reversed(scene["keys"])),
        "sockets": list(reversed(scene["sockets"])),
        "marked_slot": scene["marked_slot"],
    }
    out.append(("both_list_orders_reversed", both))
    return out
