"""Pixel-only decision oracle.

Receives a rendered RGB image and nothing else: no latent scene, no renderer
import, no gold. It segments the gray key stacks, the saturated socket stacks and
the black pointer straight from pixels, measures every bar length by counting
pixels, then solves the cross-panel interlock assignment and reports the color
and left-to-right position of the socket that fits the marked key.
"""

from __future__ import annotations

import colorsys

import numpy as np
from PIL import Image

_WHITE = 235
_DARK = 70
_GRAY_TOL = 22
_SAT_MIN = 60


def _hue_name(rgb: tuple[int, int, int]) -> str:
    r, g, b = (channel / 255.0 for channel in rgb)
    hue = colorsys.rgb_to_hsv(r, g, b)[0] * 360.0
    if hue < 20 or hue >= 330:
        return "red"
    if hue < 70:
        return "orange"
    if hue < 180:
        return "green"
    return "blue"


def _runs(flags: np.ndarray) -> list[tuple[int, int]]:
    """Inclusive index ranges of contiguous True runs."""
    out = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            out.append((start, index - 1))
            start = None
    if start is not None:
        out.append((start, len(flags) - 1))
    return out


def _stacks(mask: np.ndarray) -> list[tuple[int, int, list[int]]]:
    """(x_left, x_right, bar lengths top-to-bottom) for each horizontal group."""
    stacks = []
    for x0, x1 in _runs(mask.any(axis=0)):
        counts = mask[:, x0 : x1 + 1].sum(axis=1)
        bars = []
        for y0, y1 in _runs(counts > 0):
            band = np.sort(counts[y0 : y1 + 1])
            bars.append(int(band[len(band) // 2]))
        if len(bars) >= 2:
            stacks.append((x0, x1, bars))
    return stacks


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    hi = arr.max(axis=2)
    lo = arr.min(axis=2)
    chroma = hi - lo

    background = (hi >= _WHITE) & (chroma <= _GRAY_TOL)
    pointer = hi <= _DARK
    gray = (~background) & (~pointer) & (chroma <= _GRAY_TOL)
    colored = (~background) & (~pointer) & (chroma >= _SAT_MIN)

    keys = _stacks(gray)
    if not keys:
        raise ValueError("no key stacks found")

    sockets = []
    unique = np.unique(arr[colored].reshape(-1, 3), axis=0) if colored.any() else []
    for row in unique:
        value = (int(row[0]), int(row[1]), int(row[2]))
        mask = colored & np.all(arr == np.array(value, dtype=np.int16), axis=2)
        name = _hue_name(value)
        for x0, x1, bars in _stacks(mask):
            sockets.append({"left": x0, "right": x1, "color": name, "bars": bars})
    if len(sockets) < 2:
        raise ValueError("fewer than two socket stacks found")
    sockets.sort(key=lambda s: s["right"])
    for index, socket in enumerate(sockets):
        socket["position"] = index + 1

    if not pointer.any():
        raise ValueError("no pointer found")
    pointer_x = float(np.flatnonzero(pointer.any(axis=0)).mean())

    inside = [k for k in keys if k[0] <= pointer_x <= k[1]]
    pool = inside if inside else keys
    marked = min(pool, key=lambda k: abs((k[0] + k[1]) / 2.0 - pointer_x))
    key_bars = marked[2]

    scored = []
    for socket in sockets:
        bars = list(reversed(socket["bars"]))  # the socket is drawn rotated 180 deg
        if len(bars) != len(key_bars):
            continue
        totals = [a + b for a, b in zip(key_bars, bars)]
        scored.append((max(totals) - min(totals), socket["position"], socket["color"]))
    if not scored:
        raise ValueError("no comparable socket stack")
    scored.sort()
    defect, position, color = scored[0]
    return f"{color}|{position}"
