"""Independent raster-only inverse for the public membership-code question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

REGION_PALETTE = {
    "violet": (116, 70, 190),
    "cyan": (0, 151, 178),
    "gold": (206, 145, 0),
}
DOT_PALETTE = {
    "red": (220, 48, 48),
    "green": (28, 156, 88),
    "blue": (42, 99, 220),
    "orange": (235, 117, 28),
}
MAGENTA = (218, 45, 145)
MASK_RADIUS = 38.0


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= MASK_RADIUS * MASK_RADIUS


def _circle_from_pixels(mask: np.ndarray, x_start: int, x_stop: int) -> tuple[float, float, float]:
    restricted = mask.copy()
    restricted[:, :x_start] = False
    restricted[:, x_stop:] = False
    ys, xs = np.nonzero(restricted)
    if len(xs) < 300:
        raise ValueError("a colored circle outline is missing")
    x0, x1 = float(xs.min()), float(xs.max())
    y0, y1 = float(ys.min()), float(ys.max())
    cx, cy = (x0 + x1) / 2.0, (y0 + y1) / 2.0
    radius = ((x1 - x0) + (y1 - y0)) / 4.0
    if not 105 <= radius <= 140:
        raise ValueError("colored outline does not fit the public circle design")
    return cx, cy, radius


def _centroid(mask: np.ndarray, x_start: int, x_stop: int, minimum: int) -> tuple[float, float]:
    restricted = mask.copy()
    restricted[:, :x_start] = False
    restricted[:, x_stop:] = False
    ys, xs = np.nonzero(restricted)
    if len(xs) < minimum:
        raise ValueError("required point marker is missing")
    return float(xs.mean()), float(ys.mean())


def _signature(point: tuple[float, float], circles: list[tuple[float, float, float]]) -> tuple[int, ...]:
    x, y = point
    return tuple(int(math.hypot(x - cx, y - cy) < radius) for cx, cy, radius in circles)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    height, width = array.shape[:2]
    if width < 700 or height < 350:
        raise ValueError("image is too small for the public layout")
    split = width // 2
    circle_masks = {name: _mask(array, rgb) for name, rgb in REGION_PALETTE.items()}
    reference_circles = [
        _circle_from_pixels(circle_masks[name], 0, split)
        for name in REGION_PALETTE
    ]
    target_circles = [
        _circle_from_pixels(circle_masks[name], split, width)
        for name in REGION_PALETTE
    ]
    reference_point = _centroid(_mask(array, MAGENTA), 0, split, 60)
    reference_signature = _signature(reference_point, reference_circles)
    matches = []
    for name, rgb in DOT_PALETTE.items():
        point = _centroid(_mask(array, rgb), split, width, 100)
        if _signature(point, target_circles) == reference_signature:
            matches.append(name)
    if len(matches) != 1:
        raise ValueError(f"raster has {len(matches)} membership matches")
    return matches[0]
