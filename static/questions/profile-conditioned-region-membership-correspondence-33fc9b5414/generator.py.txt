"""Deterministic renderer and analytic arm for region-membership correspondence."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 560
BACKGROUND = (248, 249, 252)
PANEL_FILL = (255, 255, 255)
INK = (31, 41, 55)
PANEL_EDGE = (173, 181, 194)
MAGENTA = (218, 45, 145)
REGION_PALETTE = {
    "violet": (116, 70, 190),
    "cyan": (0, 151, 178),
    "gold": (206, 145, 0),
}
DOT_PALETTE = {
    "red": (220, 48, 48),
    "green": (28, 156, 88),
    "blue": (42, 99, 220),
    "orange": (235, 117, 28),
}
REGION_NAMES = tuple(REGION_PALETTE)
DOT_NAMES = tuple(DOT_PALETTE)
PANEL_BOUNDS = ((24, 78, 486, 536), (514, 78, 976, 536))


def _ordered(circles: Iterable[dict]) -> list[dict]:
    return sorted(circles, key=lambda item: REGION_NAMES.index(item["color"]))


def _membership(point: list[int] | tuple[int, int], circles: list[dict]) -> tuple[int, ...]:
    x, y = point
    return tuple(
        int(math.hypot(x - circle["cx"], y - circle["cy"]) < circle["r"])
        for circle in _ordered(circles)
    )


def _clearance(point: list[int] | tuple[int, int], circles: list[dict]) -> float:
    x, y = point
    return min(
        abs(math.hypot(x - circle["cx"], y - circle["cy"]) - circle["r"])
        for circle in circles
    )


def _circle_layout(rng: random.Random, panel_index: int) -> list[dict]:
    x0, y0, x1, y1 = PANEL_BOUNDS[panel_index]
    cx = (x0 + x1) // 2 + rng.randint(-7, 7)
    cy = (y0 + y1) // 2 + rng.randint(-7, 7)
    orbit = rng.choice((76, 80, 84))
    radius = rng.choice((120, 124, 128))
    phase = rng.choice(tuple(math.radians(v) for v in range(0, 360, 30)))
    positions = [
        (
            int(round(cx + orbit * math.cos(phase + 2 * math.pi * i / 3))),
            int(round(cy + orbit * math.sin(phase + 2 * math.pi * i / 3))),
        )
        for i in range(3)
    ]
    colors = list(REGION_NAMES)
    rng.shuffle(colors)
    return [
        {"color": color, "cx": pos[0], "cy": pos[1], "r": radius}
        for color, pos in zip(colors, positions, strict=True)
    ]


def _signature_locations(circles: list[dict], bounds: tuple[int, int, int, int]) -> dict:
    x0, y0, x1, y1 = bounds
    found: dict[tuple[int, ...], list[tuple[float, int, int]]] = {}
    for y in range(y0 + 22, y1 - 21, 5):
        for x in range(x0 + 22, x1 - 21, 5):
            clearance = _clearance((x, y), circles)
            if clearance < 18:
                continue
            signature = _membership((x, y), circles)
            found.setdefault(signature, []).append((clearance, x, y))
    return {
        signature: sorted(values, reverse=True)[:40]
        for signature, values in found.items()
    }


def _pick_location(
    rng: random.Random, locations: dict, signature: tuple[int, ...]
) -> list[int]:
    pool = locations.get(signature, [])
    if not pool:
        raise ValueError(f"layout does not realize membership signature {signature}")
    _, x, y = rng.choice(pool[: min(24, len(pool))])
    return [x, y]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Answer color cycles exactly with the integer seed."""
    rng = random.Random(int(seed))
    for _ in range(100):
        reference_circles = _circle_layout(rng, 0)
        target_circles = _circle_layout(rng, 1)
        ref_locations = _signature_locations(reference_circles, PANEL_BOUNDS[0])
        target_locations = _signature_locations(target_circles, PANEL_BOUNDS[1])
        shared = sorted(set(ref_locations) & set(target_locations))
        if len(shared) == 8:
            break
    else:
        raise RuntimeError("could not construct complete membership layouts")

    reference_signature = shared[int(seed) % len(shared)]
    reference_point = _pick_location(rng, ref_locations, reference_signature)
    desired_runner_distance = 1 + ((int(seed) // 4) % 2)
    alternatives = [
        signature
        for signature in shared
        if signature != reference_signature
    ]
    rng.shuffle(alternatives)
    alternatives.sort(
        key=lambda sig: (
            abs(sum(a != b for a, b in zip(sig, reference_signature, strict=True)) - desired_runner_distance),
            rng.random(),
        )
    )
    chosen = [reference_signature, *alternatives[:3]]
    answer_index = int(seed) % len(DOT_NAMES)
    assigned = [None] * len(DOT_NAMES)
    assigned[answer_index] = reference_signature
    distractors = iter(chosen[1:])
    for index in range(len(assigned)):
        if assigned[index] is None:
            assigned[index] = next(distractors)

    target_points = [
        {
            "color": color,
            "x": _pick_location(rng, target_locations, signature)[0],
            "y": _pick_location(rng, target_locations, signature)[1],
        }
        for color, signature in zip(DOT_NAMES, assigned, strict=True)
    ]
    # Re-pick each point only once; the two coordinates above can come from separate
    # valid locations, so canonicalize them to one valid sampled point.
    for point, signature in zip(target_points, assigned, strict=True):
        point["x"], point["y"] = _pick_location(rng, target_locations, signature)
    rng.shuffle(target_points)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "reference_circles": reference_circles,
        "target_circles": target_circles,
        "reference_point": reference_point,
        "target_points": target_points,
    }


def analytic_gold(scene: dict) -> str:
    reference_signature = _membership(scene["reference_point"], scene["reference_circles"])
    matches = [
        point["color"]
        for point in scene["target_points"]
        if _membership((point["x"], point["y"]), scene["target_circles"])
        == reference_signature
    ]
    if len(matches) != 1:
        raise ValueError(f"expected one membership match, found {matches}")
    return matches[0]


def margin(scene: dict) -> float:
    reference_signature = _membership(scene["reference_point"], scene["reference_circles"])
    distances = sorted(
        sum(a != b for a, b in zip(
            _membership((point["x"], point["y"]), scene["target_circles"]),
            reference_signature,
            strict=True,
        ))
        for point in scene["target_points"]
    )
    return float(distances[1] - distances[0]) if len(distances) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    reference_signature = _membership(scene["reference_point"], scene["reference_circles"])
    target_signatures = [
        _membership((point["x"], point["y"]), scene["target_circles"])
        for point in scene["target_points"]
    ]
    all_points_clear = _clearance(scene["reference_point"], scene["reference_circles"]) > 8
    all_points_clear = all_points_clear and all(
        _clearance((point["x"], point["y"]), scene["target_circles"]) > 8
        for point in scene["target_points"]
    )
    return (
        not all_points_clear
        or len(set(target_signatures)) != len(target_signatures)
        or target_signatures.count(reference_signature) != 1
        or margin(scene) <= 0
    )


def _draw_dot(draw: ImageDraw.ImageDraw, x: int, y: int, fill: tuple[int, int, int]) -> None:
    draw.ellipse((x - 13, y - 13, x + 13, y + 13), fill=fill, outline=INK, width=3)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for bounds in PANEL_BOUNDS:
        draw.rounded_rectangle(bounds, radius=12, fill=PANEL_FILL, outline=PANEL_EDGE, width=2)
    draw.text((42, 92), "REFERENCE", fill=INK, font=font)
    draw.text((532, 92), "TARGET", fill=INK, font=font)
    draw.text((42, 515), "match the same three inside/outside relations", fill=INK, font=font)
    for circle in sorted(scene["reference_circles"], key=lambda item: item["color"]):
        r = circle["r"]
        draw.ellipse(
            (circle["cx"] - r, circle["cy"] - r, circle["cx"] + r, circle["cy"] + r),
            outline=REGION_PALETTE[circle["color"]],
            width=6,
        )
    for circle in sorted(scene["target_circles"], key=lambda item: item["color"]):
        r = circle["r"]
        draw.ellipse(
            (circle["cx"] - r, circle["cy"] - r, circle["cx"] + r, circle["cy"] + r),
            outline=REGION_PALETTE[circle["color"]],
            width=6,
        )
    rx, ry = scene["reference_point"]
    draw.ellipse((rx - 16, ry - 16, rx + 16, ry + 16), outline=MAGENTA, width=6)
    draw.ellipse((rx - 5, ry - 5, rx + 5, ry + 5), fill=INK)
    for point in sorted(scene["target_points"], key=lambda item: item["color"]):
        _draw_dot(draw, point["x"], point["y"], DOT_PALETTE[point["color"]])
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["target_points"] = list(reversed(twin["target_points"]))
    return [("reverse_target_record_order", twin)]
