"""Renderer for `relay_jump_pip_tally`.

Scene world: six separated relay cards are scattered on a coarse lattice across a
wide canvas. Every card carries an address badge (left colour chip), a forwarding
pointer (right colour chip) naming the address of the next card, and a tally of
one to three dark pips. A dark triangular tab floating just above exactly one
card marks where the relay starts. The walk lands on four cards (the start card
plus three colour-addressed jumps) and the decision is the total number of pips
on the cards it lands on.

The decisive evidence is therefore split across four widely separated, mutually
disconnected cards: each landed card contributes its own pip count to the total,
and each forwarding pointer is needed to reach the next distant card.

Only Pillow/NumPy are used; the LaTeX renderer would add no visual value to
flat colour chips and pips, so it is deliberately not used.
"""

from __future__ import annotations

import random
from typing import Dict, List, Tuple

from PIL import Image, ImageDraw

CANVAS = (900, 620)

BG = (255, 255, 255)
FACE = (232, 232, 232)
BORDER = (96, 96, 96)
DARK = (34, 34, 34)
DECOR = (158, 158, 158)

ADDRESS_COLOURS: Dict[str, Tuple[int, int, int]] = {
    "red": (214, 40, 40),
    "green": (34, 150, 68),
    "blue": (40, 82, 214),
    "purple": (138, 48, 176),
    "orange": (240, 152, 24),
    "teal": (0, 156, 156),
}
COLOUR_NAMES: Tuple[str, ...] = tuple(sorted(ADDRESS_COLOURS))

CARD_W, CARD_H = 190, 124
CHIP = 34
CHIP_PAD = 12
PIP = 14
PIP_BOTTOM_PAD = 16
TAB_W, TAB_H = 28, 18

N_CARDS = 6
N_LANDINGS = 4  # start card plus three jumps

SLOT_COLS, SLOT_ROWS = 3, 2
QUARANTINE_MARGIN = 4.0

MIN_TOTAL = N_LANDINGS * 1
MAX_TOTAL = N_LANDINGS * 3


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------
def _slot_boxes() -> List[Tuple[int, int, int, int]]:
    """Lattice cells (x0, y0, x1, y1) that each hold one card."""
    w, h = CANVAS
    mx, my = 24, 24
    cw = (w - 2 * mx) // SLOT_COLS
    ch = (h - 2 * my) // SLOT_ROWS
    boxes = []
    for r in range(SLOT_ROWS):
        for c in range(SLOT_COLS):
            x0 = mx + c * cw
            y0 = my + r * ch
            boxes.append((x0, y0, x0 + cw, y0 + ch))
    return boxes


def _card_rects(rng: random.Random) -> List[Tuple[int, int]]:
    """Top-left corner of each card, one per lattice slot, jittered inside it."""
    out = []
    for (x0, y0, x1, y1) in _slot_boxes():
        jx0 = x0 + 6
        jx1 = max(jx0, x1 - CARD_W - 6)
        jy0 = y0 + TAB_H + 10
        jy1 = max(jy0, y1 - CARD_H - 6)
        out.append((rng.randint(jx0, jx1), rng.randint(jy0, jy1)))
    return out


def _tallies_for_total(rng: random.Random, total: int, n: int) -> List[int]:
    """Random composition of `total` into `n` parts each in 1..3."""
    for _ in range(400):
        parts = [rng.randint(1, 3) for _ in range(n)]
        if sum(parts) == total:
            return parts
    parts = [1] * n
    left = total - n
    idx = list(range(n))
    rng.shuffle(idx)
    for i in idx:
        add = min(2, left)
        parts[i] += add
        left -= add
    return parts


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    for _attempt in range(600):
        # address colours assigned to the six cards
        names = list(COLOUR_NAMES)
        rng.shuffle(names)
        addresses = names[:N_CARDS]

        # relay order: a single 6-cycle, so every pointer resolves
        chain = list(range(N_CARDS))
        rng.shuffle(chain)
        pointer_of = {}
        for i, card in enumerate(chain):
            pointer_of[card] = addresses[chain[(i + 1) % N_CARDS]]

        landed = chain[:N_LANDINGS]
        target_total = rng.randint(MIN_TOTAL, MAX_TOTAL)

        tallies = [0] * N_CARDS
        land_parts = _tallies_for_total(rng, target_total, N_LANDINGS)
        for card, part in zip(landed, land_parts):
            tallies[card] = part
        for card in range(N_CARDS):
            if tallies[card] == 0:
                tallies[card] = rng.randint(1, 3)

        total = sum(tallies[c] for c in landed)

        # placement: card index -> lattice slot (independent of relay order)
        slots = list(range(N_CARDS))
        rng.shuffle(slots)
        corners = _card_rects(rng)
        origin = [None] * N_CARDS
        for card, slot in enumerate(slots):
            origin[card] = corners[slot]

        # cheap-cue counterbalancing: neither the whole-image pip total nor the
        # naive reading-order-first-four total may equal the answer.
        all_total = sum(tallies)
        order_by_position = sorted(range(N_CARDS), key=lambda c: (origin[c][1] // 120, origin[c][0]))
        reading_total = sum(tallies[c] for c in order_by_position[:N_LANDINGS])
        if all_total == total or reading_total == total:
            continue
        # the start card must not simply be the top-left card either
        if order_by_position[0] == chain[0]:
            continue

        pip_gap = rng.choice([3, 4, 5, 6, 8, 10, 12, 14])
        tab_gap = rng.randint(4, 8)
        decor_clear = rng.choice([4, 6, 9, 12, 18, 24])

        scene = {
            "seed": int(seed),
            "addresses": addresses,
            "pointer_of": {str(k): v for k, v in pointer_of.items()},
            "tallies": tallies,
            "origin": [list(o) for o in origin],
            "chain": chain,
            "start_card": chain[0],
            "landed": landed,
            "total": total,
            "pip_gap": pip_gap,
            "tab_gap": tab_gap,
            "decor_clear": decor_clear,
            "decor": _sample_decor(rng, origin, chain[0], tab_gap, decor_clear),
        }
        return scene

    raise RuntimeError("scene sampling failed")


def _card_box(origin: Tuple[int, int]) -> Tuple[int, int, int, int]:
    x, y = origin
    return (x, y, x + CARD_W, y + CARD_H)


def _tab_box(origin: Tuple[int, int], tab_gap: int) -> Tuple[int, int, int, int]:
    x, y = origin
    cx = x + CARD_W // 2
    return (cx - TAB_W // 2, y - tab_gap - TAB_H, cx + TAB_W // 2, y - tab_gap)


def _sample_decor(rng, origin, start_card, tab_gap, decor_clear) -> List[dict]:
    """Inert grey decoration kept clear of every card and of the start tab."""
    keep_out = [_card_box(o) for o in origin]
    keep_out.append(_tab_box(origin[start_card], tab_gap))
    w, h = CANVAS
    out: List[dict] = []
    tries = 0
    want = rng.randint(7, 12)
    while len(out) < want and tries < 4000:
        tries += 1
        size = rng.randint(10, 22)
        x = rng.randint(6, w - size - 6)
        y = rng.randint(6, h - size - 6)
        box = (x, y, x + size, y + size)
        ok = True
        for kb in keep_out:
            if _box_gap(box, kb) < decor_clear:
                ok = False
                break
        if not ok:
            continue
        for prev in out:
            pb = (prev["x"], prev["y"], prev["x"] + prev["size"], prev["y"] + prev["size"])
            if _box_gap(box, pb) < 6:
                ok = False
                break
        if not ok:
            continue
        out.append({"x": x, "y": y, "size": size, "shape": rng.choice(["square", "disk"])})
    return out


def _box_gap(a, b) -> float:
    dx = max(a[0] - b[2], b[0] - a[2], 0)
    dy = max(a[1] - b[3], b[1] - a[3], 0)
    if dx == 0 and dy == 0:
        return 0.0
    return float((dx * dx + dy * dy) ** 0.5)


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    d = ImageDraw.Draw(img)

    for dec in scene["decor"]:
        box = (dec["x"], dec["y"], dec["x"] + dec["size"], dec["y"] + dec["size"])
        if dec["shape"] == "square":
            d.rectangle(box, fill=DECOR)
        else:
            d.ellipse(box, fill=DECOR)

    for card in range(N_CARDS):
        x, y = scene["origin"][card]
        d.rectangle(_card_box((x, y)), fill=FACE, outline=BORDER, width=3)

        badge = ADDRESS_COLOURS[scene["addresses"][card]]
        pointer = ADDRESS_COLOURS[scene["pointer_of"][str(card)]]
        d.rectangle((x + CHIP_PAD, y + CHIP_PAD, x + CHIP_PAD + CHIP, y + CHIP_PAD + CHIP), fill=badge)
        d.rectangle(
            (x + CARD_W - CHIP_PAD - CHIP, y + CHIP_PAD, x + CARD_W - CHIP_PAD, y + CHIP_PAD + CHIP),
            fill=pointer,
        )

        gap = scene["pip_gap"]
        n = scene["tallies"][card]
        py1 = y + CARD_H - PIP_BOTTOM_PAD
        py0 = py1 - PIP
        px = x + CHIP_PAD + 4
        for i in range(n):
            x0 = px + i * (PIP + gap)
            d.rectangle((x0, py0, x0 + PIP, py1), fill=DARK)

    tx0, ty0, tx1, ty1 = _tab_box(scene["origin"][scene["start_card"]], scene["tab_gap"])
    d.polygon([(tx0, ty1), (tx1, ty1), ((tx0 + tx1) // 2, ty0)], fill=DARK)
    return img


# --------------------------------------------------------------------------
# analytic gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------
def _replay(scene: dict) -> int:
    by_address = {name: card for card, name in enumerate(scene["addresses"])}
    cur = scene["start_card"]
    total = scene["tallies"][cur]
    for _ in range(N_LANDINGS - 1):
        cur = by_address[scene["pointer_of"][str(cur)]]
        total += scene["tallies"][cur]
    return total


def analytic_gold(scene: dict) -> str:
    return str(_replay(scene))


def margin(scene: dict) -> float:
    origins = [tuple(o) for o in scene["origin"]]
    boxes = [_card_box(o) for o in origins]
    tab = _tab_box(origins[scene["start_card"]], scene["tab_gap"])

    gaps = []
    for i in range(N_CARDS):
        for j in range(i + 1, N_CARDS):
            gaps.append(_box_gap(boxes[i], boxes[j]))
    for i in range(N_CARDS):
        if i != scene["start_card"]:
            gaps.append(_box_gap(tab, boxes[i]))

    # start-tab attribution clearance: nearest card versus runner-up card
    tcx = (tab[0] + tab[2]) / 2.0
    tcy = (tab[1] + tab[3]) / 2.0
    dists = sorted(
        (_point_box_dist(tcx, tcy, boxes[i]), i) for i in range(N_CARDS)
    )
    attribution = (dists[1][0] - dists[0][0]) / 10.0
    correct_owner = dists[0][1] == scene["start_card"]

    decor_clear = min(
        [_box_gap((dc["x"], dc["y"], dc["x"] + dc["size"], dc["y"] + dc["size"]), b)
         for dc in scene["decor"] for b in boxes + [tab]]
        or [99.0]
    )

    m = min(min(gaps), float(scene["pip_gap"]), attribution, decor_clear, float(scene["tab_gap"]))
    if not correct_owner:
        m = 0.0
    return round(float(m), 3)


def _point_box_dist(px: float, py: float, box) -> float:
    dx = max(box[0] - px, px - box[2], 0.0)
    dy = max(box[1] - py, py - box[3], 0.0)
    return float((dx * dx + dy * dy) ** 0.5)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> List[Tuple[str, dict]]:
    """No latent aliases exist for this world, so no transform is declared.

    Every scene field the analytic gold reads - the address colours, the
    forwarding addresses, the pip tallies and the start card - is drawn
    explicitly as a colour chip, a pip row or the floating start tab. Two latent
    scenes with different gold therefore cannot share a raster, and there is no
    render-invariant relabelling to exercise: recolouring the addresses, moving
    cards between lattice slots or dropping the inert decoration all change the
    pixels, so they are answer-preserving scene edits, not latent aliases.
    """
    return []
