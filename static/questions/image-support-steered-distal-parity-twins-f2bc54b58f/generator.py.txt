"""Distal disconnected parity-twin world.

A single blank white canvas holds two disconnected light-grey pin-badges that
are deliberately FAR APART: one near the top-left corner and one near the
bottom-right corner. Each badge carries a small number of dark navy pins on its
field. The decision is an attribute-correspondence predicate: do the two far-
apart badges carry the SAME parity of pin count (both odd or both even)?

Both disconnected badges are jointly decisive: neither lies within the local
span of the other, and a crop holding only one badge can never establish the
parity comparison. The badge positions are fixed constants (top-left and
bottom-right), so the decisive support is two disconnected components spanning
widely separated parts of the image (distributed + multipart); un-stored
positions cannot act as a cheap label cue across scenes.

Latent state z = {k_left, k_right} = the pin-count of each badge. Only these
two count fields influence analytic gold. The concrete arrangement of pins of a
given badge is a deterministic function of its pin-count (the first k canonical
spots), so rendering is fully determined by the two count fields.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "distal_parity_twins-0.1.0"

CANVAS = 512
BOX = 80
PIN_RADIUS = 6

WHITE = (255, 255, 255)
BOX_FILL = (208, 214, 228)
PIN = (24, 30, 66)

# Fixed distant badge placements: top-left corner and bottom-right corner.
BOX_A_ORIGIN = (48, 48)
BOX_B_ORIGIN = (384, 384)

# Canonical pin spots inside a badge (offsets from the badge top-left).
PIN_OFFSETS = ((24, 24), (56, 24), (24, 56), (56, 56))


def _pin_centres(ox: int, oy: int, count: int) -> list[tuple[int, int]]:
    return [(ox + dx, oy + dy) for dx, dy in PIN_OFFSETS[:count]]


def _badges(scene: dict) -> tuple[dict, dict]:
    left = {
        "origin": BOX_A_ORIGIN,
        "count": scene["k_left"],
        "colour_name": "left_badge",
    }
    right = {
        "origin": BOX_B_ORIGIN,
        "count": scene["k_right"],
        "colour_name": "right_badge",
    }
    return left, right


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return {
        "k_left": rng.randint(1, 4),
        "k_right": rng.randint(1, 4),
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(img)
    for badge in _badges(scene):
        ox, oy = badge["origin"]
        draw.rectangle([ox, oy, ox + BOX - 1, oy + BOX - 1], fill=BOX_FILL)
        for cx, cy in _pin_centres(ox, oy, badge["count"]):
            draw.ellipse(
                [cx - PIN_RADIUS, cy - PIN_RADIUS, cx + PIN_RADIUS, cy + PIN_RADIUS],
                fill=PIN,
            )
    return img


def analytic_gold(scene: dict) -> str:
    return "yes" if (scene["k_left"] % 2) == (scene["k_right"] % 2) else "no"


def margin(scene: dict) -> float:
    return float((scene["k_left"] % 2) == (scene["k_right"] % 2))


def is_quarantined(scene: dict) -> bool:
    # The pin count is a crisp finite-resolution integer and the palette uses
    # well-separated colours, so parity is a clean discrete event. Both classes
    # are legitimate; nothing is quarantined.
    return False


def latent_symmetries(scene: dict) -> list:
    return []


def scene_targets(scene: dict) -> dict:
    return {"k_left": scene["k_left"], "k_right": scene["k_right"]}
