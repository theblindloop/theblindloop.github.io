"""Independent pixel-only oracle for the distal parity-twin world.

Receives only the final PNG. It never sees the latent scene, renderer geometry
API, or analytic gold. It re-measures the raster independently:

1. It segments the light badge-fill colour, which should yield exactly two
   disconnected connected components (the top-left and bottom-right badges).
2. If the number of such components is not exactly two, or any badge's bounding
   box / fill area is damaged (a decisive component was removed, occluded, or
   replaced by background), the oracle abstains by raising. This is what makes
   each distant disconnected badge jointly decisive.
3. For each intact badge it counts the dark navy pin connected components inside
   that badge's box and reads its parity, then answers whether the two parities
   are the same.

The colour thresholds and geometry constants are re-derived independently from
the raster; no renderer/gold code is imported. The implementation is fully
vectorised so destructive probers can exercise it quickly.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "distal_parity_twins-oracle-0.1.0"

CANVAS = 512
BOX = 80
WHITE = (255, 255, 255)
BOX_FILL = (208, 214, 228)
PIN = (24, 30, 66)


class BadgeCompromisedError(ValueError):
    """One or both of the distant decisive badges are missing/damaged."""


def _norm_dist(flat: np.ndarray, colour: tuple[int, int, int]) -> np.ndarray:
    return np.linalg.norm(flat - np.asarray(colour, dtype=np.float32), axis=1)


def _component_labels(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen: set[tuple[int, int]] = set()
    components: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        point = (int(y), int(x))
        if point in seen:
            continue
        stack = [point]
        seen.add(point)
        cells: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            cells.append((cy, cx))
            for neighbor in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and neighbor not in seen:
                    seen.add(neighbor)
                    stack.append(neighbor)
        components.append(cells)
    return components


def _count_pins_in_bbox(mask: np.ndarray, y0: int, y1: int, x0: int, x1: int) -> int:
    seen: set[tuple[int, int]] = set()
    count = 0
    sub = np.argwhere(mask[y0:y1, x0:x1])
    for dy, dx in sub:
        point = (int(y0 + dy), int(x0 + dx))
        if point in seen:
            continue
        count += 1
        stack = [point]
        seen.add(point)
        while stack:
            cy, cx = stack.pop()
            for neighbor in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                ny, nx = neighbor
                if y0 <= ny < y1 and x0 <= nx < x1 and mask[ny, nx] and neighbor not in seen:
                    seen.add(neighbor)
                    stack.append(neighbor)
    return count


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != (CANVAS, CANVAS):
        raise BadgeCompromisedError("unexpected image size")
    arr = np.asarray(image, dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[2] != 3:
        raise BadgeCompromisedError("unexpected channel count")

    height, width = arr.shape[:2]
    flat = arr.reshape(-1, 3).astype(np.float32)
    dist_white = _norm_dist(flat, WHITE)
    dist_box = _norm_dist(flat, BOX_FILL)
    box_mask = ((dist_white > 24.0) & (dist_box <= 45.0)).reshape(height, width)

    components = _component_labels(box_mask)
    if len(components) != 2:
        raise BadgeCompromisedError(
            f"expected exactly two distant badges, found {len(components)}"
        )

    dist_pin = _norm_dist(flat, PIN).reshape(height, width)
    pin_mask = ((dist_white.reshape(height, width) > 24.0) & (dist_pin <= 40.0))

    parities = []
    for cells in components:
        ys = [c[0] for c in cells]
        xs = [c[1] for c in cells]
        h = max(ys) - min(ys) + 1
        w = max(xs) - min(xs) + 1
        if w < BOX - 4 or h < BOX - 4:
            raise BadgeCompromisedError("a distant badge bounding box is damaged")
        if len(cells) < 0.78 * (BOX * BOX):
            raise BadgeCompromisedError("a distant badge fill area is depleted")
        y0, x0 = min(ys), min(xs)
        y1, x1 = max(ys) + 1, max(xs) + 1
        pin_count = _count_pins_in_bbox(pin_mask, y0, y1, x0, x1)
        if not 1 <= pin_count <= 4:
            raise BadgeCompromisedError(f"unexpected pin count {pin_count} in a badge")
        parities.append(pin_count % 2)

    return "yes" if parities[0] == parities[1] else "no"
