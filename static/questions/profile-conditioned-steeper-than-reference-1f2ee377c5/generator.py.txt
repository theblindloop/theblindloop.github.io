"""Slope-threshold count measurement world.

Three colored diagonal segments (red, blue, green) and one neutral gray
reference diagonal segment are drawn on a light panel, each confined to its own
separate vertical column so no two strokes overlap. The decision is a pure
threshold measurement of ORIENTATION: recover the tilt of the gray reference
segment (its angle away from horizontal, i.e. its steepness) and the tilt of
every colored segment from the raster, then count how many colored segments are
STEEPER than the gray reference segment, where steeper means having a greater
angle away from horizontal (in either rising or falling direction).

Steepness is the magnitude of the tilt angle from horizontal. The integer count
is counterbalanced over the independently sampled tilt of each colored segment,
the independently sampled reference tilt, and an arbitrary shuffle of the three
colour labels among the three coloured columns, so neither the steepest segment,
the reference tilt alone, any single colored segment, total ink, segment length,
nor any positional or colour ordering cue discloses the count; only the
recovered orientation threshold comparison of each coloured tilt against the
gray reference tilt does.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "steeper_than_reference-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (96, 96, 106)  # neutral reference segment, distinct from palette + paper
BACKGROUND = (250, 250, 250)

WIDTH = 520
HEIGHT = 340
CY = 170          # vertical centre of every segment
HALF_RUN = 58     # horizontal half-extent of every segment (pixels)
STROKE_W = 12     # segment thickness
COLUMN_X = [100, 220, 340]  # x-centre of each coloured column
REF_X = 460                 # x-centre of the gray reference column

ANG_MIN = 8.0      # coloured tilt bounds (degrees from horizontal)
ANG_MAX = 60.0
REF_ANG_MIN = 14.0  # reference tilt bounds
REF_ANG_MAX = 46.0
QUARANTINE_PX = 3.0          # below this |coloured_deg - ref_deg| tilt gap is ambiguous
MIN_MARGIN = 5.0             # emitted scenes keep every coloured tilt this far from reference


def _segment_angle(cx: float, cy: float, tilt_deg: float, sign: int) -> tuple[float, float, float, float]:
    """Endpoint coordinates for a segment of given tilt magnitude and slope sign."""
    slope = math.tan(math.radians(tilt_deg))
    rise = HALF_RUN * slope
    x0, x1 = cx - HALF_RUN, cx + HALF_RUN
    y0, y1 = cy - sign * rise, cy + sign * rise
    return x0, y0, x1, y1


def sample_scene(seed: int) -> dict:
    """Three colored tilts + a reference tilt; colors shuffled across columns."""
    rng = random.Random(seed * 2143 + 29)
    while True:
        ref_deg = rng.uniform(REF_ANG_MIN, REF_ANG_MAX)
        tilts = [rng.uniform(ANG_MIN, ANG_MAX) for _ in PALETTE]
        if all(abs(t - ref_deg) >= MIN_MARGIN for t in tilts):
            break
    signs = [rng.choice((-1, 1)) for _ in PALETTE]
    ref_sign = rng.choice((-1, 1))
    colors = list(PALETTE)
    rng.shuffle(colors)
    colored = []
    for color, tilt, sign, cx in zip(colors, tilts, signs, COLUMN_X):
        x0, y0, x1, y1 = _segment_angle(cx, CY, tilt, sign)
        colored.append(
            {
                "color": color,
                "x0": x0, "y0": y0, "x1": x1, "y1": y1,
                "tilt_deg": tilt,
            }
        )
    rx0, ry0, rx1, ry1 = _segment_angle(REF_X, CY, ref_deg, ref_sign)
    reference = {
        "x0": rx0, "y0": ry0, "x1": rx1, "y1": ry1,
        "tilt_deg": ref_deg,
    }
    return {
        "colored": colored,
        "reference": reference,
        "width": WIDTH,
        "height": HEIGHT,
    }


def render(scene: dict) -> Image.Image:
    """Draw the gray reference segment and the three colored segments."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    ref = scene["reference"]
    draw.line(
        [ref["x0"], ref["y0"], ref["x1"], ref["y1"]],
        fill=GRAY_REF, width=STROKE_W,
        joint="curve",
    )
    for seg in scene["colored"]:
        draw.line(
            [seg["x0"], seg["y0"], seg["x1"], seg["y1"]],
            fill=COLOR_RGB[seg["color"]], width=STROKE_W,
            joint="curve",
        )
    return img


def _steepness(tilt_deg: float) -> float:
    """Steepness of a segment is the magnitude of its tilt from horizontal."""
    return abs(tilt_deg)


def analytic_gold(scene: dict) -> str:
    """Count colored segments whose tilt magnitude exceeds the reference tilt."""
    ref_deg = _steepness(scene["reference"]["tilt_deg"])
    count = sum(
        1 for seg in scene["colored"] if _steepness(seg["tilt_deg"]) > ref_deg
    )
    return str(count)


def margin(scene: dict) -> float:
    """Smallest |colored_deg - ref_deg| tilt gap; the threshold decision margin."""
    ref_deg = _steepness(scene["reference"]["tilt_deg"])
    return float(min(abs(_steepness(seg["tilt_deg"]) - ref_deg) for seg in scene["colored"]))


def is_quarantined(scene: dict) -> bool:
    """Ambiguous when a colored tilt is within QUARANTINE of the reference tilt."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every latent field is directly rendered as visible ink."""
    return []
