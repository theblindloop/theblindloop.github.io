"""Independent pixel-only inverse arm for the slope-threshold world.

This module receives ONLY the rasterized PNG and independently recovers the tilt
(angle away from horizontal) of each colour's stroke and of the gray reference
stroke by colour-segmented principal-axis fits, then returns the number of
coloured strokes whose measured tilt magnitude exceeds the measured gray
reference tilt magnitude. It shares no geometry or constants with renderer and
never imports the renderer, prompts, generator, verifier, or analytic gold.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "steeper_than_reference-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
GRAY_REF = (96, 96, 106)
BACKGROUND = (250, 250, 250)


def _labels(arr: np.ndarray) -> np.ndarray:
    """Label each pixel to its nearest of {red, blue, green, gray, paper}."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(GRAY_REF, dtype=np.float32), np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _tilt_deg(mask: np.ndarray) -> float:
    """Angle magnitude (degrees from horizontal) of a single-colour stroke mask."""
    ys, xs = np.nonzero(mask)
    if len(ys) < 40:
        return float("nan")
    coords = np.stack([xs, ys], axis=1).astype(np.float64)
    centered = coords - coords.mean(axis=0)
    cov = centered.T @ centered
    vals, vecs = np.linalg.eigh(cov)
    axis = vecs[:, int(np.argmax(vals))]
    dx, dy = float(axis[0]), float(axis[1])
    ang = math.degrees(math.atan2(abs(dy), abs(dx)))
    return ang


def decision_from_image(image: Image.Image) -> str:
    """Return the count of coloured strokes steeper than the gray reference stroke."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    gray_deg = _tilt_deg(labels == len(PALETTE))  # gray-reference label
    if math.isnan(gray_deg):
        return "0"
    count = 0
    for idx in range(len(PALETTE)):
        deg = _tilt_deg(labels == idx)
        if not math.isnan(deg) and deg > gray_deg:
            count += 1
    return str(count)
