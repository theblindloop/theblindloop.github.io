"""Independent pixel-only inverse for the checkerboard column task.

This module intentionally contains no import from the renderer, prompt layer,
generator, verifier, or analytic gold.  It identifies the board's fixed dark
grid network in the final RGB image, isolates its largest connected component,
and counts full-height vertical line runs.  Remote decorations use other
colors and therefore cannot answer the question.
"""

from __future__ import annotations

from collections import deque

from PIL import Image


_GRID_COLOR = (30, 39, 51)
_HUB_COLOR = (24, 145, 104)
_MIN_COMPONENT_PIXELS = 180
_MIN_LINE_HEIGHT = 24
_MIN_HUB_PIXELS = 16


def _grid_mask(image: Image.Image) -> tuple[bytearray, int, int]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    mask = bytearray(width * height)
    for index, pixel in enumerate(rgb.getdata()):
        if pixel == _GRID_COLOR:
            mask[index] = 1
    return mask, width, height


def _largest_component(mask: bytearray, width: int, height: int) -> bytearray:
    """Return a 4-connected component mask for the board grid."""

    seen = bytearray(len(mask))
    best: list[int] = []
    for start, active in enumerate(mask):
        if not active or seen[start]:
            continue
        queue: deque[int] = deque([start])
        seen[start] = 1
        component: list[int] = []
        while queue:
            index = queue.popleft()
            component.append(index)
            y, x = divmod(index, width)
            if x:
                neighbor = index - 1
                if mask[neighbor] and not seen[neighbor]:
                    seen[neighbor] = 1
                    queue.append(neighbor)
            if x + 1 < width:
                neighbor = index + 1
                if mask[neighbor] and not seen[neighbor]:
                    seen[neighbor] = 1
                    queue.append(neighbor)
            if y:
                neighbor = index - width
                if mask[neighbor] and not seen[neighbor]:
                    seen[neighbor] = 1
                    queue.append(neighbor)
            if y + 1 < height:
                neighbor = index + width
                if mask[neighbor] and not seen[neighbor]:
                    seen[neighbor] = 1
                    queue.append(neighbor)
        if len(component) > len(best):
            best = component
    if len(best) < _MIN_COMPONENT_PIXELS:
        raise ValueError("no sufficiently large connected checkerboard grid")
    selected = bytearray(len(mask))
    for index in best:
        selected[index] = 1
    return selected


def _bbox(mask: bytearray, width: int, height: int) -> tuple[int, int, int, int]:
    points = [index for index, active in enumerate(mask) if active]
    if not points:
        raise ValueError("empty connected grid")
    ys = [index // width for index in points]
    xs = [index % width for index in points]
    return min(xs), min(ys), max(xs), max(ys)


def _vertical_line_groups(
    component: bytearray,
    width: int,
    height: int,
    box: tuple[int, int, int, int],
) -> list[tuple[int, int]]:
    left, top, right, bottom = box
    local_height = bottom - top + 1
    threshold = max(_MIN_LINE_HEIGHT, int(local_height * 0.45))
    counts: list[int] = []
    for x in range(left, right + 1):
        counts.append(
            sum(1 for y in range(top, bottom + 1) if component[y * width + x])
        )
    groups: list[tuple[int, int]] = []
    start: int | None = None
    for offset, count in enumerate(counts):
        if count >= threshold and start is None:
            start = offset
        elif count < threshold and start is not None:
            groups.append((left + start, left + offset - 1))
            start = None
    if start is not None:
        groups.append((left + start, right))
    return groups


def decision_from_image(image: Image.Image) -> str:
    """Count the visible framed checkerboard columns from a final PNG only."""

    rgb = image.convert("RGB")
    hub_pixels = sum(1 for pixel in rgb.getdata() if pixel == _HUB_COLOR)
    if hub_pixels < _MIN_HUB_PIXELS:
        raise ValueError("compact checkerboard identity seal is missing")
    mask, width, height = _grid_mask(rgb)
    component = _largest_component(mask, width, height)
    box = _bbox(component, width, height)
    left, top, right, bottom = box
    if right - left < 70 or bottom - top < 70:
        raise ValueError("checkerboard grid is below the observable size margin")
    groups = _vertical_line_groups(component, width, height, box)
    if not 4 <= len(groups) <= 8:
        raise ValueError("ambiguous vertical line count")
    # A valid board exposes full-height line runs.  Refusing a partial run is
    # an explicit finite-resolution abstention rule: erasing any horizontal
    # slice of the compact grid cannot leave a deceptively confident count.
    left, top, right, bottom = box
    required_height = int((bottom - top + 1) * 0.80)
    for group_left, group_right in groups:
        strength = max(
            sum(1 for y in range(top, bottom + 1) if component[y * width + x])
            for x in range(group_left, group_right + 1)
        )
        if strength < required_height:
            raise ValueError("vertical grid line is partially erased")
    return str(len(groups) - 1)
