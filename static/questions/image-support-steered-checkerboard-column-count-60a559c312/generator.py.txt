"""Renderer for the compact checkerboard-column prior-conflict world.

The scene is deliberately small and flat.  One framed alternating-tile board
contains the answer; remote marks are visible clutter but never use the board's
grid-line color and never affect analytic gold.  Rendering is native RGB so
the independent oracle can measure stable raster boundaries without reading
latent scene state.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "checkerboard-column-count-0.1.0"
WIDTH = 512
HEIGHT = 512
GRID_COLOR = (30, 39, 51)
GRID_THICKNESS = 4
HUB_COLOR = (24, 145, 104)
BOARD_SIZES = ((80, 112), (88, 120), (96, 128))
COLUMN_CHOICES = (3, 4, 5, 6, 7)
ROW_CHOICES = (4, 5, 6)

# Palette permutations are label-preserving: the question always asks for
# columns, never for a particular ink color.
BOARD_PALETTES = (
    ((198, 78, 72), (246, 215, 151)),
    ((47, 113, 122), (214, 232, 220)),
    ((119, 79, 151), (238, 215, 171)),
)

# Remote decorations intentionally exclude GRID_COLOR.  They are sampled and
# rendered independently of n_columns, so their location, order, and ink do
# not predict the answer.
REMOTE_COLORS = (
    (139, 158, 169),
    (197, 147, 98),
    (123, 151, 111),
    (170, 132, 174),
    (105, 139, 170),
)


def _rgb(value: Any) -> tuple[int, int, int]:
    values = tuple(int(item) for item in value)
    if len(values) != 3:
        raise ValueError("RGB color must contain three channels")
    return values


def _outside_board(
    x: int,
    y: int,
    width: int,
    height: int,
    board_x: int,
    board_y: int,
    board_width: int,
    board_height: int,
    padding: int = 18,
) -> bool:
    return (
        x + width <= board_x - padding
        or x >= board_x + board_width + padding
        or y + height <= board_y - padding
        or y >= board_y + board_height + padding
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one JSON-serializable scene deterministically from ``seed``."""

    rng = random.Random(int(seed))
    n_columns = rng.choice(COLUMN_CHOICES)
    n_rows = rng.choice(ROW_CHOICES)
    board_width, board_height = rng.choice(BOARD_SIZES)
    board_x = rng.randint(34, WIDTH - board_width - 34)
    board_y = rng.randint(34, HEIGHT - board_height - 34)
    palette_index = rng.randrange(len(BOARD_PALETTES))
    phase = rng.randrange(2)

    remote_marks: list[dict[str, Any]] = []
    # Six small marks create a stable but answer-independent visual world.  A
    # rejection loop keeps them away from the focal board and from one another.
    for index in range(6):
        for _attempt in range(200):
            kind = "ring" if rng.random() < 0.5 else "bar"
            size = rng.randint(16, 30)
            mark_width = size if kind == "ring" else rng.randint(14, 34)
            mark_height = size if kind == "ring" else rng.randint(10, 22)
            x = rng.randint(20, WIDTH - mark_width - 20)
            y = rng.randint(20, HEIGHT - mark_height - 20)
            if not _outside_board(
                x,
                y,
                mark_width,
                mark_height,
                board_x,
                board_y,
                board_width,
                board_height,
            ):
                continue
            if any(
                not (
                    x + mark_width + 12 <= int(other["x"])
                    or int(other["x"]) + int(other["width"]) + 12 <= x
                    or y + mark_height + 12 <= int(other["y"])
                    or int(other["y"]) + int(other["height"]) + 12 <= y
                )
                for other in remote_marks
            ):
                continue
            remote_marks.append(
                {
                    "kind": kind,
                    "x": x,
                    "y": y,
                    "width": mark_width,
                    "height": mark_height,
                    "color": list(rng.choice(REMOTE_COLORS)),
                    "stroke": 3 if kind == "ring" else 0,
                }
            )
            break
        else:
            raise RuntimeError("could not place a remote decorative mark")

    # A per-scene permutation changes draw order without allowing a remote mark
    # to occlude the focal board.
    rng.shuffle(remote_marks)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "n_columns": n_columns,
        "n_rows": n_rows,
        "board_x": board_x,
        "board_y": board_y,
        "board_width": board_width,
        "board_height": board_height,
        "palette_index": palette_index,
        "phase": phase,
        "remote_marks": remote_marks,
    }


def _draw_remote_marks(draw: ImageDraw.ImageDraw, marks: list[dict[str, Any]]) -> None:
    for mark in marks:
        x = int(mark["x"])
        y = int(mark["y"])
        width = int(mark["width"])
        height = int(mark["height"])
        color = _rgb(mark["color"])
        if mark["kind"] == "ring":
            stroke = max(1, int(mark.get("stroke", 3)))
            draw.ellipse((x, y, x + width - 1, y + height - 1), outline=color, width=stroke)
        elif mark["kind"] == "bar":
            draw.rounded_rectangle(
                (x, y, x + width - 1, y + height - 1),
                radius=max(2, min(width, height) // 4),
                fill=color,
            )
        else:
            raise ValueError(f"unknown remote mark kind: {mark['kind']!r}")


def _edges(start: int, span: int, count: int) -> list[int]:
    return [start + round(i * span / count) for i in range(count + 1)]


def _draw_checkerboard(scene: dict[str, Any], draw: ImageDraw.ImageDraw) -> None:
    x0 = int(scene["board_x"])
    y0 = int(scene["board_y"])
    board_width = int(scene["board_width"])
    board_height = int(scene["board_height"])
    n_columns = int(scene["n_columns"])
    n_rows = int(scene["n_rows"])
    phase = int(scene["phase"]) & 1
    primary, secondary = BOARD_PALETTES[int(scene["palette_index"]) % len(BOARD_PALETTES)]

    x_edges = _edges(x0, board_width, n_columns)
    y_edges = _edges(y0, board_height, n_rows)
    right_line = x0 + board_width - GRID_THICKNESS
    bottom_line = y0 + board_height - GRID_THICKNESS

    # Fill the tile interiors first.  The dark grid is drawn afterward as one
    # connected area-like network with a stable exact RGB color.
    for row in range(n_rows):
        for column in range(n_columns):
            left_line = x_edges[column] if column < n_columns else right_line
            next_line = x_edges[column + 1] if column + 1 < n_columns else right_line
            top_line = y_edges[row] if row < n_rows else bottom_line
            next_top_line = y_edges[row + 1] if row + 1 < n_rows else bottom_line
            left = left_line + GRID_THICKNESS
            right = next_line - 1
            top = top_line + GRID_THICKNESS
            bottom = next_top_line - 1
            if right < left or bottom < top:
                raise ValueError("checkerboard cell is below the raster safety margin")
            fill = primary if (row + column + phase) % 2 == 0 else secondary
            draw.rectangle((left, top, right, bottom), fill=fill)

    # A fixed center seal is part of the framed board, not an answer label. It
    # gives the image-only arm a compact, area-like identity anchor amid the
    # remote marks; it is deliberately independent of column count and palette.
    seal_column = n_columns // 2
    seal_row = n_rows // 2
    seal_left = x_edges[seal_column] + GRID_THICKNESS
    seal_right = (x_edges[seal_column + 1] if seal_column + 1 < n_columns else right_line) - 1
    seal_top = y_edges[seal_row] + GRID_THICKNESS
    seal_bottom = (y_edges[seal_row + 1] if seal_row + 1 < n_rows else bottom_line) - 1
    seal_size = min(5, seal_right - seal_left + 1, seal_bottom - seal_top + 1)
    seal_x = (seal_left + seal_right - seal_size + 1) // 2
    seal_y = (seal_top + seal_bottom - seal_size + 1) // 2
    draw.rectangle((seal_x, seal_y, seal_x + seal_size - 1, seal_y + seal_size - 1), fill=HUB_COLOR)

    vertical_starts = [x_edges[i] for i in range(n_columns)] + [right_line]
    horizontal_starts = [y_edges[i] for i in range(n_rows)] + [bottom_line]
    for x in vertical_starts:
        draw.rectangle(
            (x, y0, min(x + GRID_THICKNESS - 1, x0 + board_width - 1), y0 + board_height - 1),
            fill=GRID_COLOR,
        )
    for y in horizontal_starts:
        draw.rectangle(
            (x0, y, x0 + board_width - 1, min(y + GRID_THICKNESS - 1, y0 + board_height - 1)),
            fill=GRID_COLOR,
        )


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only from the serialized scene dictionary."""

    width = int(scene.get("width", WIDTH))
    height = int(scene.get("height", HEIGHT))
    if (width, height) != (WIDTH, HEIGHT):
        raise ValueError("this world uses a fixed 512 by 512 canvas")
    image = Image.new("RGB", (width, height), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, list(scene["remote_marks"]))
    _draw_checkerboard(scene, draw)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible answer label; it is not used by the pixel oracle."""

    value = int(scene["n_columns"])
    if value not in COLUMN_CHOICES:
        raise ValueError(f"column count {value} is outside the public answer set")
    return str(value)


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: the narrowest tile interior in pixels."""

    board_width = float(scene["board_width"])
    board_height = float(scene["board_height"])
    columns = int(scene["n_columns"])
    rows = int(scene["n_rows"])
    return min(board_width / columns, board_height / rows) - 2.0 * GRID_THICKNESS


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude only grids whose visible tile interiors approach pixel scale."""

    return margin(scene) < 5.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """This renderer declares no latent aliases; all label inputs are visible."""

    del scene
    return []
