"""Independent pixel-only inverse for the identity flip tally world.

The implementation discovers state squares as saturated connected components,
groups them into panels by their measured x positions, reads pip counts from
nearby dark components, and compares observed RGB states across adjacent
panels. It receives no scene, coordinates, metadata, or renderer imports.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[dict] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            py, px = queue.popleft()
            points.append((py, px))
            for ny, nx in ((py - 1, px), (py + 1, px), (py, px - 1), (py, px + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        found.append(
            {
                "area": len(points),
                "x0": min(xs),
                "x1": max(xs),
                "y0": min(ys),
                "y1": max(ys),
                "cx": sum(xs) / len(xs),
                "cy": sum(ys) / len(ys),
            }
        )
    return found


def _state_squares(arr: np.ndarray) -> list[dict]:
    spread = arr.max(axis=2).astype(np.int16) - arr.min(axis=2).astype(np.int16)
    saturated = (spread >= 65) & (arr.max(axis=2) >= 115)
    squares = []
    for component in _components(saturated):
        box_w = component["x1"] - component["x0"] + 1
        box_h = component["y1"] - component["y0"] + 1
        if 650 <= component["area"] <= 1100 and 25 <= box_w <= 35 and 25 <= box_h <= 35:
            x = round(component["cx"])
            y = round(component["cy"])
            component["rgb"] = tuple(int(value) for value in arr[y, x])
            squares.append(component)
    if len(squares) != 20:
        raise OracleError(f"expected 20 state squares, recovered {len(squares)}")
    return squares


def _panel_groups(squares: list[dict]) -> list[list[dict]]:
    ordered = sorted(squares, key=lambda item: item["cx"])
    groups: list[list[dict]] = []
    for square in ordered:
        if not groups or square["cx"] - groups[-1][-1]["cx"] > 45:
            groups.append([square])
        else:
            groups[-1].append(square)
    if len(groups) != 5 or any(len(group) != 4 for group in groups):
        raise OracleError(f"panel geometry is ambiguous: {[len(group) for group in groups]}")
    return [sorted(group, key=lambda item: item["cy"]) for group in groups]


def _pip_count(arr: np.ndarray, square: dict) -> int:
    cx = round(square["cx"])
    cy = round(square["cy"])
    height, width = arr.shape[:2]
    x0, x1 = max(0, cx - 114), min(width, cx - 43)
    y0, y1 = max(0, cy - 11), min(height, cy + 12)
    crop = arr[y0:y1, x0:x1]
    dark = (crop[:, :, 0] < 60) & (crop[:, :, 1] < 70) & (crop[:, :, 2] < 90)
    pips = []
    for component in _components(dark):
        box_w = component["x1"] - component["x0"] + 1
        box_h = component["y1"] - component["y0"] + 1
        if 55 <= component["area"] <= 130 and 8 <= box_w <= 13 and 8 <= box_h <= 13:
            pips.append(component)
    if not 1 <= len(pips) <= 4:
        raise OracleError(f"invalid pip count {len(pips)} near state square")
    return len(pips)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    observations: dict[int, list[tuple[int, int, int]]] = {1: [], 2: [], 3: [], 4: []}
    for panel in _panel_groups(_state_squares(arr)):
        seen = set()
        for square in panel:
            identity = _pip_count(arr, square)
            if identity in seen:
                raise OracleError(f"duplicate {identity}-pip identity in one panel")
            seen.add(identity)
            observations[identity].append(square["rgb"])
        if seen != {1, 2, 3, 4}:
            raise OracleError(f"incomplete identity set {sorted(seen)}")

    changes = {
        identity: sum(left != right for left, right in zip(colors, colors[1:]))
        for identity, colors in observations.items()
    }
    winners = [identity for identity, count in changes.items() if count == 3]
    if len(winners) != 1:
        raise OracleError(f"expected one three-change identity, found {winners}")
    return "ABCD"[winners[0] - 1]
