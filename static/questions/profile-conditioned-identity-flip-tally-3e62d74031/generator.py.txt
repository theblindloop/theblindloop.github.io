"""Deterministic latent sampler and Pillow renderer for identity flip tally."""

from __future__ import annotations

import copy
import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1180
HEIGHT = 610
LABELS = ("A", "B", "C", "D")
PANEL_COUNT = 5
TARGET_CHANGES = 3

STATE_PALETTES = (
    ("#176B87", "#E87524"),
    ("#3157A4", "#D64F70"),
    ("#157A6E", "#C95722"),
    ("#6B4FA1", "#D17C16"),
)
BODY_TINTS = ("#F7F3E8", "#EEF4F2", "#F3F0F7", "#F5F1EE")


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    candidates = (
        Path("/usr/share/fonts/truetype/dejavu") / name,
        Path("/usr/share/fonts/truetype/dejavu") / name,
    )
    for path in candidates:
        if path.is_file():
            return ImageFont.truetype(str(path), size=size)
    return ImageFont.load_default(size=size)


def _sequence_with_changes(rng: random.Random, count: int) -> list[int]:
    edges = set(rng.sample(range(PANEL_COUNT - 1), count))
    value = rng.randrange(2)
    values = [value]
    for edge in range(PANEL_COUNT - 1):
        if edge in edges:
            value = 1 - value
        values.append(value)
    return values


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Consecutive seeds cycle the selected identity."""
    rng = random.Random((int(seed) << 17) ^ 0x5A17D3)
    target = LABELS[int(seed) % len(LABELS)]
    distractor_counts = rng.sample([0, 1, 2, 4], 3)
    assigned_counts: dict[str, int] = {target: TARGET_CHANGES}
    for label, count in zip((item for item in LABELS if item != target), distractor_counts):
        assigned_counts[label] = count

    states = {label: _sequence_with_changes(rng, assigned_counts[label]) for label in LABELS}
    positions = []
    for _ in range(PANEL_COUNT):
        order = list(LABELS)
        rng.shuffle(order)
        positions.append(order)

    palette = list(STATE_PALETTES[rng.randrange(len(STATE_PALETTES))])
    if rng.randrange(2):
        palette.reverse()
    return {
        "states": states,
        "positions": positions,
        "state_colors": palette,
        "body_tint": BODY_TINTS[rng.randrange(len(BODY_TINTS))],
    }


def _flip_counts(scene: dict) -> dict[str, int]:
    states = scene["states"]
    return {
        label: sum(left != right for left, right in itertools.pairwise(states[label]))
        for label in LABELS
    }


def analytic_gold(scene: dict) -> str:
    winners = [label for label, count in _flip_counts(scene).items() if count == TARGET_CHANGES]
    if len(winners) != 1:
        raise ValueError(f"expected one identity with {TARGET_CHANGES} changes, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = _flip_counts(scene)
    winner = analytic_gold(scene)
    return float(min(abs(count - TARGET_CHANGES) for label, count in counts.items() if label != winner))


def is_quarantined(scene: dict) -> bool:
    try:
        states = scene["states"]
        positions = scene["positions"]
        colors = scene["state_colors"]
        structurally_valid = (
            set(states) == set(LABELS)
            and all(len(states[label]) == PANEL_COUNT for label in LABELS)
            and all(set(states[label]) <= {0, 1} for label in LABELS)
            and len(positions) == PANEL_COUNT
            and all(sorted(order) == list(LABELS) for order in positions)
            and len(colors) == 2
            and colors[0] != colors[1]
        )
        return not structurally_valid or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Swap the arbitrary binary coding while retaining identical visible pixels."""
    twin = copy.deepcopy(scene)
    twin["states"] = {
        label: [1 - value for value in values]
        for label, values in scene["states"].items()
    }
    twin["state_colors"] = list(reversed(scene["state_colors"]))
    return [("complement_state_codes_and_swap_palette", twin)]


def _pip_centers(left: int, count: int, cy: int) -> list[tuple[int, int]]:
    spacing = 17
    center = left + 50
    start = center - spacing * (count - 1) / 2
    return [(round(start + index * spacing), cy) for index in range(count)]


def render(scene: dict) -> Image.Image:
    """Render the complete question world; every label-relevant field is visible."""
    image = Image.new("RGB", (WIDTH, HEIGHT), "#FAFAF8")
    draw = ImageDraw.Draw(image)
    title_font = _font(27, bold=True)
    subtitle_font = _font(17)
    panel_font = _font(22, bold=True)
    small_font = _font(14, bold=True)

    draw.text(
        (WIDTH // 2, 27),
        "TRACK EACH PIP IDENTITY  •  COUNT COLOR CHANGES",
        font=title_font,
        fill="#1F2937",
        anchor="mm",
    )
    draw.text(
        (WIDTH // 2, 63),
        "A = 1 pip   B = 2 pips   C = 3 pips   D = 4 pips   •   Find exactly 3 changes",
        font=subtitle_font,
        fill="#374151",
        anchor="mm",
    )

    panel_y = 105
    panel_h = 438
    panel_w = 214
    gap = 12
    start_x = 25
    token_w = 166
    token_h = 66
    token_left_offset = 24
    token_top = panel_y + 56
    row_step = 87

    positions = scene["positions"]
    states = scene["states"]
    state_colors = scene["state_colors"]
    body_tint = scene["body_tint"]

    for panel_index in range(PANEL_COUNT):
        panel_x = start_x + panel_index * (panel_w + gap)
        draw.rounded_rectangle(
            (panel_x, panel_y, panel_x + panel_w, panel_y + panel_h),
            radius=15,
            fill="#FFFFFF",
            outline="#8B95A5",
            width=3,
        )
        draw.text(
            (panel_x + panel_w // 2, panel_y + 26),
            str(panel_index + 1),
            font=panel_font,
            fill="#263244",
            anchor="mm",
        )
        if panel_index < PANEL_COUNT - 1:
            arrow_y = panel_y + panel_h // 2
            arrow_x = panel_x + panel_w + 2
            draw.polygon(
                [(arrow_x, arrow_y - 6), (arrow_x + 8, arrow_y), (arrow_x, arrow_y + 6)],
                fill="#5E6877",
            )

        for slot, label in enumerate(positions[panel_index]):
            token_x = panel_x + token_left_offset
            token_y = token_top + slot * row_step
            draw.rounded_rectangle(
                (token_x, token_y, token_x + token_w, token_y + token_h),
                radius=13,
                fill=body_tint,
                outline="#788392",
                width=3,
            )
            pip_y = token_y + token_h // 2
            for px, py in _pip_centers(token_x, LABELS.index(label) + 1, pip_y):
                draw.ellipse((px - 5, py - 5, px + 5, py + 5), fill="#1F2937")
            square_left = token_x + 118
            square_top = token_y + 18
            color = state_colors[states[label][panel_index]]
            draw.rectangle(
                (square_left, square_top, square_left + 30, square_top + 30),
                fill=color,
                outline="#FFFFFF",
                width=2,
            )
            draw.text(
                (token_x + 84, token_y + token_h - 8),
                "ID",
                font=small_font,
                fill="#697386",
                anchor="ms",
            )

    draw.text(
        (WIDTH // 2, 579),
        "Compare adjacent panels only • moving to another row is not a state change",
        font=subtitle_font,
        fill="#374151",
        anchor="mm",
    )
    return image
