"""Independent pixel-only inverse arm for the dot-loop topology world.

Recovers every dark isolated dot as a centroid from the final raster, then
reconstructs the global proximity graph: any two dots whose centres are closer
than a threshold (1.5 x the median nearest-neighbour distance) are connected.
Reading the degree structure of the single resulting component decides the
answer -- ``loop`` (every dot connects to exactly two others, one closed
cycle) versus ``chain`` (exactly two free-end dots of degree one along one
open path).

This module imports only numpy/PIL, never receives the latent scene, and does
not import renderer / prompts / generate / verify / analytic-gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_DARK = 120
_THRESHOLD_FACTOR = 1.5


def _centroids(image: Image.Image) -> list[tuple[float, float]]:
    arr = np.asarray(image.convert("RGB"))
    mask = np.any(arr < _DARK, axis=-1)
    h, w = mask.shape
    lab = np.zeros((h, w), dtype=int)
    current = 1
    for y in range(h):
        for x in range(w):
            if mask[y, x] and lab[y, x] == 0:
                stack = [(y, x)]
                lab[y, x] = current
                while stack:
                    yy, xx = stack.pop()
                    for ny, nx in ((yy + 1, xx), (yy - 1, xx), (yy, xx + 1), (yy, xx - 1)):
                        if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and lab[ny, nx] == 0:
                            lab[ny, nx] = current
                            stack.append((ny, nx))
                current += 1
    centroids = []
    for label in range(1, current):
        ys, xs = np.where(lab == label)
        centroids.append((float(xs.mean()), float(ys.mean())))
    return centroids


def _structure(points, threshold: float, n: int) -> str:
    deg = [0] * n
    adj = [[] for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            if np.sqrt(((points[i] - points[j]) ** 2).sum()) <= threshold:
                deg[i] += 1
                deg[j] += 1
                adj[i].append(j)
                adj[j].append(i)
    seen = set()
    components = 0
    for i in range(n):
        if i in seen:
            continue
        components += 1
        stack = [i]
        seen.add(i)
        while stack:
            u = stack.pop()
            for v in adj[u]:
                if v not in seen:
                    seen.add(v)
                    stack.append(v)
    if components != 1:
        return "chain"
    if all(d == 2 for d in deg):
        return "loop"
    d1 = sum(1 for d in deg if d == 1)
    d2 = sum(1 for d in deg if d == 2)
    if d1 == 2 and d1 + d2 == n:
        return "chain"
    return "loop"


def decision_from_image(image: Image.Image) -> str:
    points = _centroids(image)
    n = len(points)
    if n < 4:
        return "chain"
    pts = np.array(points)
    nn = []
    for i in range(n):
        d = np.sqrt(((pts - pts[i]) ** 2).sum(1))
        d[i] = 1e9
        nn.append(float(d.min()))
    s = float(np.median(nn))
    threshold = _THRESHOLD_FACTOR * s
    return _structure(pts, threshold, n)
