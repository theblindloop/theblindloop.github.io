"""Deterministic latent sampler and renderer for two-band deinterlacing."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("H", "U", "C", "N")
QUESTION = (
    "Each colored panel has four rows. Rebuild one 8-row tile by taking orange row 1, "
    "then teal row 1, then orange row 2, then teal row 2, and so on, without reversing "
    "any row. A few dark square speckles may be present. Which symbol is the result "
    "closest to: H, U, C, or N?"
)

# Equal-ink 5x5 glyphs (13 dark cells each), centered in an 8x8 reconstruction.
_CORE = {
    "H": ("10001", "10001", "11111", "10001", "10001"),
    "U": ("10001", "10001", "10001", "10001", "11111"),
    "C": ("11111", "10000", "10000", "10000", "11111"),
    "N": ("10001", "11001", "10101", "10011", "10001"),
}

_PALETTES = (
    {"orange": (224, 118, 34), "orange_fill": (255, 226, 186), "teal": (22, 145, 151), "teal_fill": (190, 235, 232)},
    {"orange": (205, 92, 26), "orange_fill": (252, 215, 174), "teal": (15, 128, 139), "teal_fill": (177, 225, 226)},
)
_INK = (25, 34, 53)
_BG = (248, 247, 242)


def _template(symbol: str) -> list[list[int]]:
    out = [[0 for _ in range(8)] for _ in range(8)]
    for r, row in enumerate(_CORE[symbol], start=1):
        for c, value in enumerate(row, start=1):
            out[r][c] = int(value)
    return out


def _dist(a: Iterable[Iterable[int]], b: Iterable[Iterable[int]]) -> int:
    return sum(int(x != y) for ar, br in zip(a, b) for x, y in zip(ar, br))


def _scores(bits: list[list[int]]) -> list[tuple[int, str]]:
    return sorted((_dist(bits, _template(symbol)), symbol) for symbol in SYMBOLS)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    base = _template(symbol)
    swap_count = (int(seed) // len(SYMBOLS)) % 3
    bits = [row[:] for row in base]
    # Balanced swaps preserve total ink and are accepted only if the intended
    # class remains unique with an off-quarantine gap.
    for _ in range(200):
        trial = [row[:] for row in base]
        on = [(r, c) for r in range(8) for c in range(8) if trial[r][c]]
        off = [(r, c) for r in range(8) for c in range(8) if not trial[r][c]]
        rng.shuffle(on)
        rng.shuffle(off)
        for r, c in on[:swap_count]:
            trial[r][c] = 0
        for r, c in off[:swap_count]:
            trial[r][c] = 1
        scores = _scores(trial)
        if scores[0][1] == symbol and scores[1][0] - scores[0][0] >= 2:
            bits = trial
            break
    return {
        "bits": bits,
        "orange_left": bool(rng.getrandbits(1)),
        "jitter_x": rng.randint(-2, 2),
        "jitter_y": rng.randint(-2, 2),
        "palette_variant": rng.randrange(len(_PALETTES)),
    }


def _panel(draw: ImageDraw.ImageDraw, x: int, y: int, rows: list[list[int]], role: str, palette: dict) -> None:
    # 88x64 panel: four 14px rows by eight 10px columns inside a 4px frame.
    draw.rectangle((x, y, x + 87, y + 63), fill=palette[role])
    draw.rectangle((x + 4, y + 4, x + 83, y + 59), fill=palette[f"{role}_fill"])
    for r, row in enumerate(rows):
        for c, bit in enumerate(row):
            x0, y0 = x + 4 + 10 * c, y + 4 + 14 * r
            if bit:
                draw.rectangle((x0 + 2, y0 + 2, x0 + 7, y0 + 11), fill=_INK)


def render(scene: dict) -> Image.Image:
    bits = [[int(value) for value in row] for row in scene["bits"]]
    orange_left = bool(scene["orange_left"])
    jitter_x = int(scene["jitter_x"])
    jitter_y = int(scene["jitter_y"])
    palette = _PALETTES[int(scene["palette_variant"]) % len(_PALETTES)]
    image = Image.new("RGB", (WIDTH, HEIGHT), _BG)
    draw = ImageDraw.Draw(image)
    orange_rows = bits[0::2]
    teal_rows = bits[1::2]
    left_x, right_x = 82 + jitter_x, 342 + jitter_x
    y = 224 + jitter_y
    orange_x, teal_x = (left_x, right_x) if orange_left else (right_x, left_x)
    _panel(draw, orange_x, y, orange_rows, "orange", palette)
    _panel(draw, teal_x, y, teal_rows, "teal", palette)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores([[int(value) for value in row] for row in scene["bits"]])
    if scores[0][0] == scores[1][0]:
        raise ValueError("ambiguous glyph")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = _scores([[int(value) for value in row] for row in scene["bits"]])
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
