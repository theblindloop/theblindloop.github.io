"""Independent image-only inverse for the declared deinterlacing transform."""

from __future__ import annotations

import numpy as np
from PIL import Image

_SYMBOLS = ("H", "U", "C", "N")
_CORE = {
    "H": ("10001", "10001", "11111", "10001", "10001"),
    "U": ("10001", "10001", "10001", "10001", "11111"),
    "C": ("11111", "10000", "10000", "10000", "11111"),
    "N": ("10001", "11001", "10101", "10011", "10001"),
}


def _template(symbol: str) -> list[list[int]]:
    out = [[0] * 8 for _ in range(8)]
    for r, row in enumerate(_CORE[symbol], start=1):
        for c, value in enumerate(row, start=1):
            out[r][c] = int(value)
    return out


def _bbox(mask: np.ndarray, role: str) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 900:
        raise ValueError(f"missing {role} fiducial")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    if not (86 <= x1 - x0 + 1 <= 90 and 62 <= y1 - y0 + 1 <= 66):
        raise ValueError(f"damaged {role} frame")
    return x0, y0, x1, y1


def _read_panel(arr: np.ndarray, box: tuple[int, int, int, int], role: str) -> list[list[int]]:
    x0, y0, x1, y1 = box
    rows = []
    for r in range(4):
        row = []
        for c in range(8):
            patch = arr[y0 + 4 + 14 * r : y0 + 4 + 14 * (r + 1), x0 + 4 + 10 * c : x0 + 4 + 10 * (c + 1)]
            if patch.shape != (14, 10, 3):
                raise ValueError("truncated panel")
            # Erasure is distinct from a legitimate pale blank cell.
            white = (patch[:, :, 0] > 242) & (patch[:, :, 1] > 242) & (patch[:, :, 2] > 235)
            if float(white.mean()) > 0.10:
                raise ValueError(f"erased {role} cell")
            dark = (patch[:, :, 0] < 80) & (patch[:, :, 1] < 80) & (patch[:, :, 2] < 100)
            fraction = float(dark.mean())
            if fraction > 0.25:
                row.append(1)
            elif fraction < 0.05:
                row.append(0)
            else:
                raise ValueError(f"ambiguous {role} cell")
        rows.append(row)
    return rows


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    r, g, b = arr[:, :, 0], arr[:, :, 1], arr[:, :, 2]
    orange = (r > 185) & (g > 70) & (g < 170) & (b < 90)
    teal = (r < 70) & (g > 105) & (g < 180) & (b > 105) & (b < 190)
    orange_rows = _read_panel(arr, _bbox(orange, "orange"), "orange")
    teal_rows = _read_panel(arr, _bbox(teal, "teal"), "teal")
    bits = []
    for orange_row, teal_row in zip(orange_rows, teal_rows):
        bits.extend((orange_row, teal_row))
    scores = []
    for symbol in _SYMBOLS:
        template = _template(symbol)
        distance = sum(bits[y][x] != template[y][x] for y in range(8) for x in range(8))
        scores.append((distance, symbol))
    scores.sort()
    if scores[1][0] - scores[0][0] < 2:
        raise ValueError("ambiguous reconstructed glyph")
    return scores[0][1]
