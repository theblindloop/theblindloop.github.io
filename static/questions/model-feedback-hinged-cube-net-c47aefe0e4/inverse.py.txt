"""Independent image-only inversion of the oriented cube net."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": (211, 73, 69),
    "blue": (54, 111, 184),
    "green": (61, 145, 91),
    "amber": (230, 166, 42),
    "violet": (139, 85, 174),
    "cyan": (42, 164, 181),
}
STEP = {"N": (-1, 0), "E": (0, 1), "S": (1, 0), "W": (0, -1)}


def _neg(v):
    return tuple(-x for x in v)


def _turn(frame, side):
    east, south, normal = frame
    if side == "E": return (_neg(normal), south, east)
    if side == "W": return (normal, south, _neg(east))
    if side == "N": return (east, normal, _neg(south))
    if side == "S": return (east, _neg(normal), south)
    raise ValueError(side)


def _front_frame(up):
    return {
        "N": ((1, 0, 0), (0, -1, 0), (0, 0, 1)),
        "E": ((0, 1, 0), (1, 0, 0), (0, 0, 1)),
        "S": ((-1, 0, 0), (0, 1, 0), (0, 0, 1)),
        "W": ((0, -1, 0), (-1, 0, 0), (0, 0, 1)),
    }[up]


def _color_centers(image):
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    centers = {}
    for name, rgb in PALETTE.items():
        dist = np.sqrt(((arr - np.array(rgb, dtype=np.int32)) ** 2).sum(axis=2))
        ys, xs = np.nonzero(dist < 24)
        if len(xs) < 5000:
            raise ValueError(f"missing face color {name}")
        centers[name] = (float(xs.mean()), float(ys.mean()))
    return centers, arr


def decision_from_image(image: Image.Image) -> str:
    centers, arr = _color_centers(image)
    points = list(centers.values())
    # Adjacent face centers are separated by the common smallest axial spacing.
    gaps = []
    for i, (x1, y1) in enumerate(points):
        for x2, y2 in points[i + 1:]:
            if abs(y1 - y2) < 4 and abs(x1 - x2) > 50:
                gaps.append(abs(x1 - x2))
            if abs(x1 - x2) < 4 and abs(y1 - y2) > 50:
                gaps.append(abs(y1 - y2))
    if not gaps:
        raise ValueError("cannot infer face spacing")
    spacing = min(gaps)
    origin_x = min(x for x, _ in points)
    origin_y = min(y for _, y in points)
    at = {}
    by_color = {}
    for color, (x, y) in centers.items():
        pos = (int(round((y - origin_y) / spacing)), int(round((x - origin_x) / spacing)))
        if pos in at:
            raise ValueError("two colors occupy one face")
        at[pos] = color
        by_color[color] = pos

    dark = np.max(arr, axis=2) < 38
    anchor = None
    best = 0
    for color, (cx, cy) in centers.items():
        x0, x1 = int(cx - spacing * .42), int(cx + spacing * .42) + 1
        y0, y1 = int(cy - spacing * .42), int(cy + spacing * .42) + 1
        count = int(dark[max(0,y0):y1, max(0,x0):x1].sum())
        if count > best:
            best, anchor = count, color
    if anchor is None or best < 250:
        raise ValueError("anchor marker missing")
    cx, cy = centers[anchor]
    ys, xs = np.nonzero(dark)
    local = (np.abs(xs - cx) < spacing * .45) & (np.abs(ys - cy) < spacing * .45)
    xs, ys = xs[local], ys[local]
    far = (xs - cx) ** 2 + (ys - cy) ** 2 > (spacing * .20) ** 2
    if int(far.sum()) < 30:
        raise ValueError("arrow missing")
    vx = float((xs[far] - cx).mean())
    vy = float((ys[far] - cy).mean())
    if abs(vx) > abs(vy):
        up = "E" if vx > 0 else "W"
    else:
        up = "S" if vy > 0 else "N"

    start = by_color[anchor]
    frames = {start: _front_frame(up)}
    queue = deque([start])
    while queue:
        pos = queue.popleft()
        for side, (dr, dc) in STEP.items():
            nxt = (pos[0] + dr, pos[1] + dc)
            if nxt not in at:
                continue
            proposed = _turn(frames[pos], side)
            if nxt in frames and frames[nxt] != proposed:
                raise ValueError("inconsistent cube fold")
            if nxt not in frames:
                frames[nxt] = proposed
                queue.append(nxt)
    if len(frames) != 6 or len({frame[2] for frame in frames.values()}) != 6:
        raise ValueError("raster is not a valid cube net")
    target = next(pos for pos, frame in frames.items() if frame[2] == (1, 0, 0))
    return at[target]
