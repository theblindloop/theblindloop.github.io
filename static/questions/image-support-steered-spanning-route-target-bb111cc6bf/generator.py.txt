"""Spanning-route target world renderer.

Latent scene: a single continuous dark route (a connected polyline) begins at a
reference start marker on the left edge and spans the whole canvas width to end
at exactly one colored destination marker near the right edge. Two other colored
destination markers are present, far from the route (one top, one bottom), so
they are clearly disconnected. The question asks which colored destination the
route reaches from the reference marker.

The decisive support is the single connected thin route that spans widely
separated left and right regions of the canvas (distributed), and requires
following the whole elongated trace (pathlike) rather than reading one endpoint:
ablating any interior stretch of the route severs the start from its reachable
destination and changes the pixel oracle's decision.

```text
I = R(z)
reach = color of destination whose center equals route[-1]
```
"""

from __future__ import annotations

import dataclasses
import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "spanning_route_target-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)
ROUTE_COLOR = (25, 25, 25)
START_COLOR = (245, 150, 30)  # orange reference start marker
DEST_COLORS = {
    "red": (214, 48, 49),
    "blue": (41, 82, 214),
    "green": (38, 166, 91),
}
DEST_RADIUS = 13.0
START_RADIUS = 12.0
ROUTE_WIDTH = 6

# Route corridor: a near-horizontal band spanning the full canvas width, so the
# decisive trace reaches both the left and right edges of the image.
X_LEFT = (40.0, 70.0)      # start x (left edge region)
X_RIGHT = (442.0, 472.0)   # destination x (right edge region)
Y_BAND = (200.0, 312.0)    # vertical corridor the route is confined to

# Distractor destination anchor regions, far above and far below the corridor.
TOP_Y = 55.0
BOTTOM_Y = 457.0

MIN_CLEARANCE = 42.0


@dataclasses.dataclass(frozen=True)
class RouteScene:
    start: tuple
    route: list
    destinations: list
    width: int = 512
    height: int = 512


def _dist(p, q):
    return math.dist(p, q)


def _segment_clearance(a, b, center):
    ax, ay = a
    bx, by = b
    px, py = center
    dx, dy = bx - ax, by - ay
    length_sq = dx * dx + dy * dy
    if length_sq == 0:
        return math.dist(a, center)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / length_sq))
    nearest = (ax + t * dx, ay + t * dy)
    return math.dist(nearest, center)


def _route_clearance(route, center):
    if len(route) < 2:
        return math.dist(route[0], center)
    return min(
        _segment_clearance(a, b, center) for a, b in zip(route[:-1], route[1:])
    )


def _sample_route(rng, start, dest_center, unreached_centers):
    """Sample a connected polyline from start to dest_center spanning full width.

    Interior knots advance monotonically in x across the canvas while staying in
    the horizontal Y_BAND corridor, so the trace is a thin elongated component
    that reaches both the left and right edges. Every segment keeps
    MIN_CLEARANCE from every unreached destination center.
    """
    sx, sy = start
    dx, dy = dest_center
    span = dx - sx
    for _ in range(20000):
        n_knots = rng.randint(3, 5)
        xs = sorted(sx + span * rng.uniform(0.08, 0.92) for _ in range(n_knots))
        waypoints = [start]
        prev = start
        ok = True
        for kx in xs:
            ky = rng.uniform(Y_BAND[0] + 6, Y_BAND[1] - 6)
            # keep roughly monotonic y drift bounded
            ky = 0.5 * ky + 0.5 * prev[1]
            ky = min(max(ky, Y_BAND[0] + 4), Y_BAND[1] - 4)
            node = (kx, ky)
            if _dist(prev, node) < 20:
                continue
            waypoints.append(node)
            prev = node
        waypoints.append(dest_center)
        ok = True
        for a, b in zip(waypoints[:-1], waypoints[1:]):
            for center in unreached_centers:
                if _segment_clearance(a, b, center) < MIN_CLEARANCE:
                    ok = False
                    break
            if not ok:
                break
        if ok:
            return waypoints
    raise RuntimeError("could not sample a clear spanning route")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start_x = rng.uniform(*X_LEFT)
    start_y = rng.uniform(Y_BAND[0], Y_BAND[1])
    start = (start_x, start_y)

    dest_x = rng.uniform(*X_RIGHT)
    dest_y = rng.uniform(Y_BAND[0], Y_BAND[1])
    dest_center = (dest_x, dest_y)

    colors = ["red", "blue", "green"]
    rng.shuffle(colors)
    reached_color = colors[0]
    unreached_colors = colors[1:]

    # Distractor destinations far above and below the corridor.
    top_x = rng.uniform(330.0, 470.0)
    bottom_x = rng.uniform(60.0, 220.0)
    unreached_centers = [(top_x, TOP_Y), (bottom_x, BOTTOM_Y)]

    route = _sample_route(rng, start, dest_center, unreached_centers)

    destinations = [{"pos": (dest_center[0], dest_center[1]), "color": reached_color}]
    for center, color in zip(unreached_centers, unreached_colors, strict=True):
        destinations.append({"pos": center, "color": color})
    rng.shuffle(destinations)

    return {
        "width": 512,
        "height": 512,
        "start": start,
        "route": route,
        "destinations": destinations,
    }


def _reached_color(scene) -> str:
    end = tuple(scene["route"][-1])
    for d in scene["destinations"]:
        if tuple(d["pos"]) == end:
            return d["color"]
    raise ValueError("route endpoint does not match any destination")


def render(scene) -> Image.Image:
    w = int(scene["width"])
    h = int(scene["height"])
    ss = 2
    img = Image.new("RGB", (w * ss, h * ss), BG)
    draw = ImageDraw.Draw(img)

    def S(p):
        return (p[0] * ss, p[1] * ss)

    sx, sy = S(scene["start"])
    draw.ellipse(
        [sx - START_RADIUS * ss, sy - START_RADIUS * ss, sx + START_RADIUS * ss, sy + START_RADIUS * ss],
        fill=START_COLOR,
    )
    for d in scene["destinations"]:
        cx, cy = S(d["pos"])
        color = DEST_COLORS[d["color"]]
        draw.ellipse(
            [cx - DEST_RADIUS * ss, cy - DEST_RADIUS * ss, cx + DEST_RADIUS * ss, cy + DEST_RADIUS * ss],
            fill=color,
        )

    pts = [S(p) for p in scene["route"]]
    draw.line(pts, fill=ROUTE_COLOR, width=ROUTE_WIDTH * ss, joint="curve")

    img = img.resize((w, h), Image.LANCZOS)
    return img


def analytic_gold(scene) -> str:
    return _reached_color(scene)


def margin(scene) -> float:
    """Runner-up margin: clearance from the route to the nearest non-answer dest."""
    reached = _reached_color(scene)
    best = float("inf")
    for d in scene["destinations"]:
        if d["color"] == reached:
            continue
        seg_min = _route_clearance(scene["route"], d["pos"])
        best = min(best, seg_min - DEST_RADIUS)
    return float(best if best != float("inf") else 0.0)


def is_quarantined(scene) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene):
    """not_applicable: the latent parametrization is minimal and injective.

    Every label-input field (route, destinations) is directly and visibly
    realised in the raster; there is no latent field hidden from rendering and
    no pixel-identical alternative scene that changes the answer.
    """
    return iter(())
