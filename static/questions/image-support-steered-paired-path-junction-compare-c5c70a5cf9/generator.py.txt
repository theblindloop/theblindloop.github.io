"""Deterministic latent sampler, renderer, and analytic graph arm."""

from __future__ import annotations

import copy
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 420

PALETTES = (
    {
        "dark": (28, 43, 61),
        "neutral": (128, 87, 166),
        "start": (37, 108, 190),
        "finish": (225, 123, 38),
        "left_card": (246, 249, 253),
        "right_card": (253, 249, 244),
    },
    {
        "dark": (35, 49, 58),
        "neutral": (143, 82, 158),
        "start": (28, 116, 180),
        "finish": (232, 113, 35),
        "left_card": (248, 250, 252),
        "right_card": (252, 248, 246),
    },
    {
        "dark": (23, 52, 61),
        "neutral": (112, 91, 174),
        "start": (47, 104, 188),
        "finish": (214, 132, 29),
        "left_card": (245, 250, 250),
        "right_card": (253, 250, 242),
    },
)

BASE_NODES = {
    "j0": (95, 210),
    "j1": (165, 210),
    "j2": (235, 210),
    "a": (55, 95),
    "b": (55, 325),
    "c": (165, 70),
    "d": (275, 95),
    "e": (275, 325),
}

BASE_EDGES = (
    ("j0", "j1"),
    ("j1", "j2"),
    ("j0", "a"),
    ("j0", "b"),
    ("j1", "c"),
    ("j2", "d"),
    ("j2", "e"),
)

PAIRS_BY_SCORE = {
    1: (("a", "b"), ("d", "e")),
    2: (("a", "c"), ("b", "c"), ("c", "d"), ("c", "e")),
    3: (("a", "d"), ("a", "e"), ("b", "d"), ("b", "e")),
}


def _placed_nodes(
    *, offset_x: int, flip_x: bool, flip_y: bool, dx: int, dy: int
) -> dict[str, list[int]]:
    nodes: dict[str, list[int]] = {}
    for name, (x, y) in BASE_NODES.items():
        if flip_x:
            x = 330 - x
        if flip_y:
            y = 400 - y
        nodes[name] = [x + offset_x + dx, y + dy]
    return nodes


def _endpoint_pair(rng: random.Random, score: int) -> tuple[str, str]:
    start, finish = rng.choice(PAIRS_BY_SCORE[score])
    if rng.randrange(2):
        start, finish = finish, start
    return start, finish


def sample_scene(seed: int) -> dict:
    """Sample balanced left/right answers with equal topology and ink per panel."""

    rng = random.Random(int(seed))
    low_score = 1 if (seed // 2) % 2 == 0 else 2
    high_score = low_score + 1
    if seed % 2:
        left_score, right_score = high_score, low_score
    else:
        left_score, right_score = low_score, high_score

    left_start, left_finish = _endpoint_pair(rng, left_score)
    right_start, right_finish = _endpoint_pair(rng, right_score)
    left_nodes = _placed_nodes(
        offset_x=0,
        flip_x=bool(rng.randrange(2)),
        flip_y=bool(rng.randrange(2)),
        dx=rng.randint(-5, 5),
        dy=rng.randint(-4, 4),
    )
    right_nodes = _placed_nodes(
        offset_x=630,
        flip_x=bool(rng.randrange(2)),
        flip_y=bool(rng.randrange(2)),
        dx=rng.randint(-5, 5),
        dy=rng.randint(-4, 4),
    )
    left_edges = [list(edge) for edge in BASE_EDGES]
    right_edges = [list(edge) for edge in BASE_EDGES]
    rng.shuffle(left_edges)
    rng.shuffle(right_edges)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "left_nodes": left_nodes,
        "right_nodes": right_nodes,
        "left_edges": left_edges,
        "right_edges": right_edges,
        "left_start": left_start,
        "left_finish": left_finish,
        "right_start": right_start,
        "right_finish": right_finish,
        "palette_id": rng.randrange(len(PALETTES)),
        "stroke_width": 9,
        "node_radius": 15,
    }


def _draw_tree(
    draw: ImageDraw.ImageDraw,
    nodes: dict[str, list[int]],
    edges: list[list[str]],
    start: str,
    finish: str,
    palette: dict,
    stroke_width: int,
    node_radius: int,
    card_color: tuple[int, int, int],
) -> None:
    for first, second in edges:
        draw.line(
            [tuple(nodes[first]), tuple(nodes[second])],
            fill=palette["dark"],
            width=stroke_width,
        )
    halo = node_radius + 4
    for name, (x, y) in nodes.items():
        draw.ellipse((x - halo, y - halo, x + halo, y + halo), fill=card_color)
        color = palette["neutral"]
        if name == start:
            color = palette["start"]
        elif name == finish:
            color = palette["finish"]
        draw.ellipse(
            (x - node_radius, y - node_radius, x + node_radius, y + node_radius),
            fill=color,
        )
        # A small highlight makes the circular nodes legible without affecting roles.
        draw.ellipse((x - 5, y - 6, x - 1, y - 2), fill=(255, 255, 255))


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    palette = PALETTES[int(scene["palette_id"])]
    image = Image.new("RGB", (width, height), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        (18, 20, 312, height - 20),
        radius=24,
        fill=palette["left_card"],
        outline=(210, 220, 230),
        width=3,
    )
    draw.rounded_rectangle(
        (648, 20, 942, height - 20),
        radius=24,
        fill=palette["right_card"],
        outline=(226, 218, 208),
        width=3,
    )
    _draw_tree(
        draw,
        scene["left_nodes"],
        scene["left_edges"],
        scene["left_start"],
        scene["left_finish"],
        palette,
        int(scene["stroke_width"]),
        int(scene["node_radius"]),
        palette["left_card"],
    )
    _draw_tree(
        draw,
        scene["right_nodes"],
        scene["right_edges"],
        scene["right_start"],
        scene["right_finish"],
        palette,
        int(scene["stroke_width"]),
        int(scene["node_radius"]),
        palette["right_card"],
    )
    return image


def _junctions_on_path(edges: list[list[str]], start: str, finish: str) -> int:
    adjacency: dict[str, set[str]] = {}
    for first, second in edges:
        adjacency.setdefault(first, set()).add(second)
        adjacency.setdefault(second, set()).add(first)
    if start not in adjacency or finish not in adjacency:
        raise ValueError("endpoint is absent from graph")
    parents: dict[str, str | None] = {start: None}
    queue = deque([start])
    while queue:
        current = queue.popleft()
        if current == finish:
            break
        for neighbor in sorted(adjacency[current]):
            if neighbor not in parents:
                parents[neighbor] = current
                queue.append(neighbor)
    if finish not in parents:
        raise ValueError("endpoints are disconnected")
    path = []
    current: str | None = finish
    while current is not None:
        path.append(current)
        current = parents[current]
    return sum(len(adjacency[node]) == 3 for node in path)


def analytic_gold(scene: dict) -> str:
    left_edges = scene["left_edges"]
    right_edges = scene["right_edges"]
    left_start = scene["left_start"]
    left_finish = scene["left_finish"]
    right_start = scene["right_start"]
    right_finish = scene["right_finish"]
    left = _junctions_on_path(left_edges, left_start, left_finish)
    right = _junctions_on_path(right_edges, right_start, right_finish)
    if left == right:
        raise ValueError("equal counts have no left/right decision")
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    left = _junctions_on_path(scene["left_edges"], scene["left_start"], scene["left_finish"])
    right = _junctions_on_path(
        scene["right_edges"], scene["right_start"], scene["right_finish"]
    )
    return float(abs(left - right))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["left_edges"] = list(reversed(twin["left_edges"]))
    twin["right_edges"] = list(reversed(twin["right_edges"]))
    return [("reverse_edge_draw_lists", twin)]
