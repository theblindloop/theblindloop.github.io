"""Scene sampler and Pillow renderer for a coupled gate-and-pawn replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 820, 520
BACKGROUND = (248, 247, 242)
INK = (30, 32, 36)
PALETTES = (
    {"pawn": (190, 35, 120), "gate": (35, 105, 190), "flip": (244, 190, 50), "move": (45, 155, 95)},
    {"pawn": (120, 60, 195), "gate": (20, 150, 140), "flip": (238, 145, 45), "move": (196, 60, 60)},
    {"pawn": (220, 70, 50), "gate": (105, 75, 180), "flip": (225, 190, 25), "move": (30, 135, 190)},
)


def _replay(start_room: int, gate_states: list[bool], commands: list[dict[str, Any]]) -> tuple[int, int, int]:
    room = int(start_room)
    gates = [bool(value) for value in gate_states]
    moved = blocked = 0
    for command in commands:
        if command["op"] == "flip":
            index = int(command["gate"]) - 1
            gates[index] = not gates[index]
        else:
            step = 1 if command["direction"] == "right" else -1
            destination = room + step
            if destination < 1 or destination > 4:
                blocked += 1
                continue
            gate_index = min(room, destination) - 1
            if gates[gate_index]:
                room = destination
                moved += 1
            else:
                blocked += 1
    return room, moved, blocked


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; seed modulo four selects the final-answer stratum."""
    rng = random.Random(int(seed))
    target = int(seed) % 4 + 1
    palette_variant = rng.randrange(len(PALETTES))
    shift = rng.choice([-10, -5, 0, 5, 10])
    decorations = [rng.randrange(3) for _ in range(6)]
    for _ in range(20000):
        start = rng.randrange(1, 5)
        gates = [bool(rng.randrange(2)) for _ in range(3)]
        commands: list[dict[str, Any]] = []
        kinds = ["flip"] * 3 + ["move"] * 3
        rng.shuffle(kinds)
        for kind in kinds:
            if kind == "flip":
                commands.append({"op": "flip", "gate": rng.randrange(1, 4)})
            else:
                commands.append({"op": "move", "direction": rng.choice(["left", "right"])})
        final, moved, blocked = _replay(start, gates, commands)
        flipped = {item["gate"] for item in commands if item["op"] == "flip"}
        if final == target and moved >= 1 and blocked >= 1 and len(flipped) >= 2:
            return {
                "start_room": start,
                "gate_states": gates,
                "commands": commands,
                "palette_variant": palette_variant,
                "shift": shift,
                "decorations": decorations,
            }
    raise RuntimeError(f"could not sample answer stratum {target}")


def _dot_row(draw: ImageDraw.ImageDraw, center_x: int, y: int, count: int, radius: int = 5) -> None:
    spacing = 16
    start = center_x - spacing * (count - 1) / 2
    for index in range(count):
        x = int(round(start + index * spacing))
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=INK)


def _arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, direction: str, color: tuple[int, int, int]) -> None:
    sign = 1 if direction == "right" else -1
    draw.rectangle((cx - 28, cy - 7, cx + 28, cy + 7), fill=color)
    tip = cx + sign * 38
    base = cx + sign * 8
    draw.polygon([(tip, cy), (base, cy - 26), (base, cy + 26)], fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=15)
    palette = PALETTES[int(scene["palette_variant"])]
    shift = int(scene["shift"])

    draw.text((34 + shift, 24), "START: rooms 1 to 4", fill=INK, font=font)
    room_lefts = [45 + shift + 190 * index for index in range(4)]
    room_boxes = []
    for index, left in enumerate(room_lefts, start=1):
        box = (left, 70, left + 135, 205)
        room_boxes.append(box)
        draw.rectangle(box, outline=INK, width=4)
        _dot_row(draw, left + 67, 94, index, radius=5)
    pawn_box = room_boxes[int(scene["start_room"]) - 1]
    pawn_x = (pawn_box[0] + pawn_box[2]) // 2
    draw.ellipse((pawn_x - 24, 139, pawn_x + 24, 187), outline=palette["pawn"], width=8)
    draw.ellipse((pawn_x - 8, 155, pawn_x + 8, 171), fill=palette["pawn"])

    for index, is_open in enumerate(scene["gate_states"]):
        gx = room_boxes[index][2] + 27
        if is_open:
            draw.rectangle((gx - 25, 158, gx - 5, 168), fill=palette["gate"])
            draw.rectangle((gx + 5, 158, gx + 25, 168), fill=palette["gate"])
        else:
            draw.rectangle((gx - 5, 130, gx + 5, 152), fill=palette["gate"])
            draw.rectangle((gx - 5, 174, gx + 5, 196), fill=palette["gate"])

    draw.text((34 + shift, 245), "CARDS: read left to right", fill=INK, font=font)
    for index, command in enumerate(scene["commands"]):
        left = 24 + shift + 130 * index
        top, right, bottom = 280, left + 108, 474
        if command["op"] == "flip":
            draw.rectangle((left + 5, top + 5, right - 5, bottom - 5), fill=palette["flip"])
        draw.rectangle((left, top, right, bottom), outline=INK, width=4)
        cx = (left + right) // 2
        if command["op"] == "flip":
            draw.line((cx - 25, 341, cx + 25, 341), fill=INK, width=4)
            draw.arc((cx - 27, 316, cx + 27, 370), 35, 300, fill=INK, width=4)
            _dot_row(draw, cx, 412, int(command["gate"]), radius=6)
        else:
            _arrow(draw, cx, 372, command["direction"], palette["move"])
        # Pale gray diamonds are answer-independent nuisance marks.
        for mark in range(int(scene["decorations"][index])):
            x = left + 22 + mark * 18
            draw.polygon([(x, 447), (x + 5, 452), (x, 457), (x - 5, 452)], fill=(180, 182, 180))
    return image


def analytic_gold(scene: dict) -> str:
    final, _, _ = _replay(scene["start_room"], scene["gate_states"], scene["commands"])
    return str(final)


def margin(scene: dict) -> float:
    # The nearest runner-up is one discrete room identity away.
    _ = (scene["start_room"], scene["gate_states"], scene["commands"])
    return 1.0


def is_quarantined(scene: dict) -> bool:
    _ = (scene["start_room"], scene["gate_states"], scene["commands"])
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
