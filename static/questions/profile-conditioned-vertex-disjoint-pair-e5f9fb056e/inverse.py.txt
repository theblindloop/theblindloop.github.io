"""Independent pixel-only graph inverse for the vertex-disjoint-pair world."""

from __future__ import annotations

import itertools
from collections import deque

import numpy as np
from PIL import Image

_NAMES = ("background", "edge", "black", "red", "green", "blue")
_RGB = np.array([
    [250, 250, 247], [116, 123, 132], [25, 28, 34],
    [218, 55, 72], [38, 160, 91], [54, 104, 218],
], dtype=np.int32)


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= minimum:
            found.append(np.asarray(pixels, dtype=np.int32))
    return found


def _connected(edges: list[tuple[int, int]], source: int, target: int, removed: set[int]) -> bool:
    adjacency: dict[int, list[int]] = {}
    for a, b in edges:
        if a in removed or b in removed:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    seen = {source}
    queue = deque([source])
    while queue:
        u = queue.popleft()
        if u == target:
            return True
        for v in adjacency.get(u, []):
            if v not in seen:
                seen.add(v)
                queue.append(v)
    return False


def _vertex_connectivity(edges: list[tuple[int, int]], pair: list[int]) -> int:
    vertices = sorted({v for edge in edges for v in edge} - set(pair))
    if not _connected(edges, pair[0], pair[1], set()):
        return 0
    for size in range(1, len(vertices) + 1):
        for subset in itertools.combinations(vertices, size):
            if not _connected(edges, pair[0], pair[1], set(subset)):
                return size
    return len(vertices) + 1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = ((array[:, :, None, :] - _RGB[None, None, :, :]) ** 2).sum(axis=3)
    labels = distances.argmin(axis=2)

    centers: list[tuple[float, float, str]] = []
    for label, name in enumerate(_NAMES[2:], start=2):
        comps = _components(labels == label, 160)
        expected = 10 if name == "black" else 2
        if len(comps) != expected:
            raise ValueError(f"expected {expected} {name} stations, found {len(comps)}")
        for pixels in comps:
            centers.append((float(pixels[:, 1].mean()), float(pixels[:, 0].mean()), name))

    gray_components = _components(labels == 1, 35)
    if len(gray_components) != 18:
        raise ValueError(f"expected 18 separated gray links, found {len(gray_components)}")
    edges: set[tuple[int, int]] = set()
    for pixels in gray_components:
        xy = pixels[:, ::-1].astype(np.float64)
        proximity = []
        for index, (cx, cy, _) in enumerate(centers):
            dist2 = ((xy[:, 0] - cx) ** 2 + (xy[:, 1] - cy) ** 2).min()
            proximity.append((float(dist2), index))
        endpoints = sorted(proximity)[:2]
        if endpoints[1][0] > 420.0:
            raise ValueError("a gray link has no visible station endpoint")
        edge = tuple(sorted((endpoints[0][1], endpoints[1][1])))
        if edge[0] == edge[1] or edge in edges:
            raise ValueError("ambiguous or duplicate recovered link")
        edges.add(edge)
    edge_list = sorted(edges)

    winners = []
    for color in ("red", "green", "blue"):
        pair = [i for i, (_, _, name) in enumerate(centers) if name == color]
        if len(pair) != 2:
            raise ValueError(f"expected two {color} terminals")
        if _vertex_connectivity(edge_list, pair) == 2:
            winners.append(color)
    if len(winners) != 1:
        raise ValueError(f"expected one pair with vertex connectivity two, found {winners}")
    return winners[0]
