"""Analytic sampler and renderer for the vertex-disjoint-pair world."""

from __future__ import annotations

import copy
import itertools
import random
from collections import deque

from PIL import Image, ImageDraw

CANVAS = [600, 480]
PALETTE = {
    "background": [250, 250, 247],
    "edge": [116, 123, 132],
    "black": [25, 28, 34],
    "red": [218, 55, 72],
    "green": [38, 160, 91],
    "blue": [54, 104, 218],
}
COLORS = ("red", "green", "blue")

# Each role is a small two-terminal subgraph with vertex-connectivity 1, 2, or 3.
# Roles are permuted among the three slots. Each module has exactly one attachment
# to the black hub, so the decorative global connection cannot add a terminal route.
_SLOTS = [(145, 110), (455, 110), (300, 385)]
_ATTACHMENTS = [(210, 210), (390, 210), (300, 270)]
_HUB = (300, 205)


def _build_graph(roles_by_slot: tuple[int, int, int]) -> tuple[list[list[int]], list[list[int]], list[list[int]]]:
    nodes: list[list[int]] = []
    edges: list[list[int]] = []
    pair_roles: list[list[int] | None] = [None, None, None]
    attachment_indices: list[int] = []
    for slot, role in enumerate(roles_by_slot):
        cx, cy = _SLOTS[slot]
        left = len(nodes)
        nodes.extend([[cx - 65, cy], [cx + 65, cy]])
        right = left + 1
        internal: list[int] = []
        offsets = (0,) if role == 0 else ((-45, 45) if role == 1 else (-58, 0, 58))
        for dy in offsets:
            internal.append(len(nodes))
            nodes.append([cx, cy + dy])
        for middle in internal:
            edges.extend([[left, middle], [middle, right]])
        pair_roles[role] = [left, right]
        attach = len(nodes)
        nodes.append(list(_ATTACHMENTS[slot]))
        chosen = internal[-1] if slot < 2 else internal[0]
        edges.append([chosen, attach])
        attachment_indices.append(attach)
    hub = len(nodes)
    nodes.append(list(_HUB))
    edges.extend([[attach, hub] for attach in attachment_indices])
    return nodes, edges, [pair for pair in pair_roles if pair is not None]


def _transform_nodes(nodes: list[list[int]], code: int, dx: int, dy: int) -> list[list[int]]:
    out = []
    for x, y in nodes:
        if code & 1:
            x = CANVAS[0] - x
        if code & 2:
            y = CANVAS[1] - y
        out.append([x + dx, y + dy])
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    roles_by_slot = list(itertools.permutations((0, 1, 2)))[(seed // 3) % 6]
    base_nodes, edges, pair_roles = _build_graph(roles_by_slot)
    nodes = [[x + rng.randint(-3, 3), y + rng.randint(-3, 3)] for x, y in base_nodes]
    nodes = _transform_nodes(nodes, (seed // 3) % 4, rng.randint(-3, 3), rng.randint(-3, 3))
    shift = seed % 3
    assigned = [COLORS[(shift + i) % 3] for i in range(3)]
    pairs = {assigned[i]: list(pair_roles[i]) for i in range(3)}
    return {
        "canvas": list(CANVAS),
        "nodes": nodes,
        "edges": copy.deepcopy(edges),
        "pairs": pairs,
        "palette": copy.deepcopy(PALETTE),
        "node_radius": 18,
        "edge_width": 7,
    }


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    palette = scene["palette"]
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    nodes = scene["nodes"]
    for a, b in scene["edges"]:
        draw.line(
            [(nodes[a][0] * scale, nodes[a][1] * scale),
             (nodes[b][0] * scale, nodes[b][1] * scale)],
            fill=tuple(palette["edge"]), width=scene["edge_width"] * scale,
        )
    node_colors = {index: "black" for index in range(len(nodes))}
    for color, endpoints in scene["pairs"].items():
        for index in endpoints:
            node_colors[index] = color
    radius = scene["node_radius"] * scale
    for index, (x, y) in enumerate(nodes):
        cx, cy = x * scale, y * scale
        draw.ellipse(
            [cx - radius, cy - radius, cx + radius, cy + radius],
            fill=tuple(palette[node_colors[index]]),
            outline=tuple(palette["background"]), width=scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _connected(edges: list[tuple[int, int]], source: int, target: int, removed: set[int]) -> bool:
    if source in removed or target in removed:
        return False
    adjacency: dict[int, list[int]] = {}
    for a, b in edges:
        if a in removed or b in removed:
            continue
        adjacency.setdefault(a, []).append(b)
        adjacency.setdefault(b, []).append(a)
    seen = {source}
    queue = deque([source])
    while queue:
        node = queue.popleft()
        if node == target:
            return True
        for nxt in adjacency.get(node, []):
            if nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return False


def _vertex_connectivity(edges: list[tuple[int, int]], pair: list[int]) -> int:
    vertices = sorted({v for edge in edges for v in edge} - set(pair))
    if not _connected(edges, pair[0], pair[1], set()):
        return 0
    for size in range(1, len(vertices) + 1):
        for subset in itertools.combinations(vertices, size):
            if not _connected(edges, pair[0], pair[1], set(subset)):
                return size
    return len(vertices) + 1


def analytic_gold(scene: dict) -> str:
    edges = [tuple(edge) for edge in scene["edges"]]
    winners = [color for color in COLORS if _vertex_connectivity(edges, scene["pairs"][color]) == 2]
    if len(winners) != 1:
        raise ValueError(f"expected one pair with vertex connectivity two, found {winners}")
    return winners[0]


def margin(scene: dict) -> float:
    edges = [tuple(edge) for edge in scene["edges"]]
    distances = sorted(abs(_vertex_connectivity(edges, pair) - 2) for pair in scene["pairs"].values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        edges = [tuple(edge) for edge in scene["edges"]]
        scores = sorted(_vertex_connectivity(edges, pair) for pair in scene["pairs"].values())
        return scores != [1, 2, 3] or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    count = len(scene["nodes"])
    remap = {old: count - 1 - old for old in range(count)}
    twin["nodes"] = list(reversed(scene["nodes"]))
    twin["edges"] = [[remap[a], remap[b]] for a, b in scene["edges"]]
    twin["pairs"] = {color: [remap[a], remap[b]] for color, (a, b) in scene["pairs"].items()}
    return [("reverse_node_identifier_order", twin)]
