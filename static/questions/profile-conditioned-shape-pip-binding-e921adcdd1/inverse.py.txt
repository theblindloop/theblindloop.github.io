"""Pixel-only inverse for shape-to-pip binding correspondence.

This module deliberately contains its own palette, component extraction, shape
classification, and assignment logic. It receives only a raster image.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": (210, 62, 62),
    "blue": (55, 105, 200),
    "gold": (220, 165, 40),
    "reference": (128, 73, 168),
    "shape": (103, 188, 211),
    "pip": (25, 30, 35),
}


def _near(array: np.ndarray, color: tuple[int, int, int], tolerance: int = 8) -> np.ndarray:
    # int32 avoids overflow when three squared RGB deltas are summed.
    delta = array.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(x0), int(y0))])
        seen[y0, x0] = True
        points = []
        while queue:
            x, y = queue.popleft()
            points.append((x, y))
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((nx, ny))
        if len(points) < minimum:
            continue
        xs = [point[0] for point in points]
        ys = [point[1] for point in points]
        output.append(
            {
                "area": len(points),
                "bbox": (min(xs), min(ys), max(xs), max(ys)),
                "center": (sum(xs) / len(xs), sum(ys) / len(ys)),
            }
        )
    return output


def _palette_bbox(array: np.ndarray, name: str) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(_near(array, PALETTE[name]))
    if len(xs) < 100:
        raise ValueError(f"missing {name} card border")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _shape_name(component: dict) -> str:
    x0, y0, x1, y1 = component["bbox"]
    density = component["area"] / ((x1 - x0 + 1) * (y1 - y0 + 1))
    if density > 0.88:
        return "square"
    if density > 0.65:
        return "circle"
    return "triangle"


def _read_card(array: np.ndarray, card_bbox: tuple[int, int, int, int]) -> dict[str, int]:
    bx0, by0, bx1, by1 = card_bbox
    shapes = []
    for component in _components(_near(array, PALETTE["shape"]), minimum=700):
        cx, cy = component["center"]
        if bx0 < cx < bx1 and by0 < cy < by1:
            shapes.append(component)
    if len(shapes) != 3:
        raise ValueError(f"expected three visible shapes, found {len(shapes)}")

    pips = _components(_near(array, PALETTE["pip"]), minimum=30)
    binding = {}
    for component in shapes:
        x0, y0, x1, y1 = component["bbox"]
        count = sum(1 for pip in pips if x0 < pip["center"][0] < x1 and y0 < pip["center"][1] < y1)
        name = _shape_name(component)
        if name in binding or not 1 <= count <= 5:
            raise ValueError("ambiguous shape identity or pip count")
        binding[name] = count
    if set(binding) != {"circle", "triangle", "square"}:
        raise ValueError("shape inventory incomplete")
    return binding


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    reference = _read_card(array, _palette_bbox(array, "reference"))
    scores = {}
    for color in ("red", "blue", "gold"):
        candidate = _read_card(array, _palette_bbox(array, color))
        scores[color] = sum(candidate[shape] == reference[shape] for shape in reference)
    ranking = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranking[0][1] != 3 or ranking[0][1] - ranking[1][1] < 2:
        raise ValueError("no unique full correspondence with sufficient runner-up margin")
    return ranking[0][0]
