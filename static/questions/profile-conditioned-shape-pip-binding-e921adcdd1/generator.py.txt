"""Deterministic renderer for shape-to-pip binding correspondence."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 590
BACKGROUND = (248, 248, 245)
SHAPE_FILL = (103, 188, 211)
PIP = (25, 30, 35)
BORDERS = {
    "red": (210, 62, 62),
    "blue": (55, 105, 200),
    "gold": (220, 165, 40),
    "reference": (128, 73, 168),
}
SHAPES = ("circle", "triangle", "square")
ANSWER_COLORS = ("red", "blue", "gold")
CARD_W, CARD_H = 300, 170
CARD_POSITIONS = ((330, 35), (10, 365), (330, 365), (650, 365))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_counts = rng.sample([1, 2, 3, 4, 5], 3)

    wrong = [
        list(p)
        for p in itertools.permutations(reference_counts)
        if list(p) != reference_counts and sum(a == b for a, b in zip(p, reference_counts)) <= 1
    ]
    rng.shuffle(wrong)

    correct_position = (int(seed) // 3) % 3
    options: list[list[int] | None] = [None, None, None]
    options[correct_position] = list(reference_counts)
    wrong_iter = iter(wrong)
    for index in range(3):
        if options[index] is None:
            options[index] = next(wrong_iter)

    gold_color = ANSWER_COLORS[int(seed) % 3]
    other_colors = [color for color in ANSWER_COLORS if color != gold_color]
    if int(seed) % 2:
        other_colors.reverse()
    border_colors: list[str | None] = [None, None, None]
    border_colors[correct_position] = gold_color
    color_iter = iter(other_colors)
    for index in range(3):
        if border_colors[index] is None:
            border_colors[index] = next(color_iter)

    shape_orders = []
    rotations = []
    for _ in range(4):
        order = list(SHAPES)
        rng.shuffle(order)
        shape_orders.append(order)
        rotations.append(
            {
                "circle": 0,
                "triangle": rng.choice([0, 120, 240]),
                "square": rng.choice([0, 90]),
            }
        )

    return {
        "reference_counts": reference_counts,
        "option_counts": options,
        "shape_orders": shape_orders,
        "border_colors": border_colors,
        "rotations": rotations,
    }


def _polygon(cx: int, cy: int, radius: float, sides: int, angle_deg: float) -> list[tuple[int, int]]:
    return [
        (
            round(cx + radius * math.cos(math.radians(angle_deg - 90 + 360 * i / sides))),
            round(cy + radius * math.sin(math.radians(angle_deg - 90 + 360 * i / sides))),
        )
        for i in range(sides)
    ]


def _pip_offsets(count: int) -> list[tuple[int, int]]:
    patterns = {
        1: [(0, 0)],
        2: [(-8, -8), (8, 8)],
        3: [(-9, -9), (0, 0), (9, 9)],
        4: [(-9, -9), (9, -9), (-9, 9), (9, 9)],
        5: [(-10, -10), (10, -10), (0, 0), (-10, 10), (10, 10)],
    }
    return patterns[int(count)]


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[int, int], count: int, rotation: int) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse((cx - 34, cy - 34, cx + 34, cy + 34), fill=SHAPE_FILL)
    elif shape == "triangle":
        draw.polygon(_polygon(cx, cy + 2, 40, 3, rotation), fill=SHAPE_FILL)
    elif shape == "square":
        draw.polygon(_polygon(cx, cy, 44, 4, 45 + rotation), fill=SHAPE_FILL)
    else:
        raise ValueError(f"unknown shape {shape!r}")
    for dx, dy in _pip_offsets(count):
        draw.ellipse((cx + dx - 4, cy + dy - 4, cx + dx + 4, cy + dy + 4), fill=PIP)


def _draw_card(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    border: str,
    counts: list[int],
    order: list[str],
    rotations: dict[str, int],
) -> None:
    x, y = xy
    color = BORDERS[border]
    draw.rectangle((x, y, x + CARD_W, y + CARD_H), fill=color)
    draw.rectangle((x + 8, y + 8, x + CARD_W - 8, y + CARD_H - 8), fill=BACKGROUND)
    count_by_shape = dict(zip(SHAPES, counts))
    for slot, shape in enumerate(order):
        center = (x + 60 + slot * 90, y + 86)
        _draw_shape(draw, shape, center, count_by_shape[shape], rotations[shape])


def render(scene: dict) -> Image.Image:
    reference_counts = scene["reference_counts"]
    option_counts = scene["option_counts"]
    shape_orders = scene["shape_orders"]
    border_colors = scene["border_colors"]
    rotations = scene["rotations"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_card(draw, CARD_POSITIONS[0], "reference", reference_counts, shape_orders[0], rotations[0])
    for index in range(3):
        _draw_card(
            draw,
            CARD_POSITIONS[index + 1],
            border_colors[index],
            option_counts[index],
            shape_orders[index + 1],
            rotations[index + 1],
        )
    return image


def analytic_gold(scene: dict) -> str:
    reference_counts = scene["reference_counts"]
    option_counts = scene["option_counts"]
    border_colors = scene["border_colors"]
    matches = [index for index, counts in enumerate(option_counts) if counts == reference_counts]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one full binding match")
    return str(border_colors[matches[0]])


def _scores(scene: dict) -> list[int]:
    reference = scene["reference_counts"]
    return [sum(left == right for left, right in zip(reference, counts)) for counts in scene["option_counts"]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return scores.count(max(scores)) != 1 or max(scores) != 3 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
