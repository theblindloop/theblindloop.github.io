WIDTH = 640
HEIGHT = 640
BACKGROUND = (247, 247, 242)

PANEL_PALETTES = (

def _stable_count(seed: int) -> int:
    """Balance the familiar seven-column case against visible counterfactuals."""
    if seed % 2:
        return 7
    counterfactuals = (4, 5, 6, 8, 9, 10)
    return counterfactuals[(seed * 37 + 11) % len(counterfactuals)]


def sample_scene(seed: int) -> dict:
    """Sample one visible route with nuisance factors independent of the count."""
    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 41)
    column_count = _stable_count(seed)
    center_x = rng.uniform(270.0, 370.0)
    center_y = rng.uniform(315.0, 350.0)
    route_span = rng.uniform(278.0, 322.0)
    route_height = rng.uniform(112.0, 138.0)
    diagonal_slope = rng.uniform(0.26, 0.36)
    route_width = rng.uniform(9.0, 12.0)
    # A small, still-readable near-boundary subgroup makes quarantine explicit.
    if (seed * 29 + 7) % 23 == 0:
        route_width = rng.uniform(6.0, 7.0)
    x_left = center_x - route_span / 2.0
    gap = route_span / float(column_count - 1)
    jitter_limit = min(4.0, gap * 0.12)
    x_positions = [
        x_left + index * gap + rng.uniform(-jitter_limit, jitter_limit)
        for index in range(column_count)
    ]
    x_positions[0] = x_left
    x_positions[-1] = center_x + route_span / 2.0
    panel_width = route_span + rng.uniform(76.0, 94.0)
    panel_height = route_height + diagonal_slope * route_span + rng.uniform(112.0, 132.0)
    panel_top = center_y - panel_height / 2.0
    top_y = panel_top + rng.uniform(66.0, 78.0)
    bottom_y = top_y + route_height

    palette_index = (seed * 17 + 3) % len(ROUTE_PALETTES)
    panel_index = (seed * 19 + 1) % len(PANEL_PALETTES)
    rail_index = (seed * 23 + 9) % len(RAIL_PALETTES)
    mark_offset = rng.uniform(0.0, 18.0)
    return {
        "column_count": column_count,
        "x_positions": [round(value, 6) for value in x_positions],
        "top_y": round(top_y, 6),
        "bottom_y": round(bottom_y, 6),
        "diagonal_slope": round(diagonal_slope, 6),
        "center_x": round(center_x, 6),
        "center_y": round(center_y, 6),
        "panel_width": round(panel_width, 6),
        "panel_height": round(panel_height, 6),
        "route_color": list(ROUTE_PALETTES[palette_index]),
        "route_width": round(route_width, 6),
        "panel_color": list(PANEL_PALETTES[panel_index]),
        "rail_color": list(RAIL_PALETTES[rail_index]),
        "row_color": list(ROW_PALETTES[(seed * 31 + 5) % len(ROW_PALETTES)]),
        "label_color": list(LABEL_PALETTES[(seed * 13 + 4) % len(LABEL_PALETTES)]),
        "mark_color": list(MARK_PALETTES[(seed * 11 + 6) % len(MARK_PALETTES)]),
        "mark_offset": round(mark_offset, 6),
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "antialias": True,
    }


def _scale_box(box: tuple[float, float, float, float], scale: int) -> tuple[float, float, float, float]:
    return tuple(value * scale for value in box)  # type: ignore[return-value]


_BITMAP = {
    "W": ("10001", "10001", "10101", "10101", "01010", "01010", "01010"),
    "E": ("11111", "10000", "10000", "11110", "10000", "10000", "11111"),
    "K": ("10001", "10010", "10100", "11000", "10100", "10010", "10001"),
}


def _draw_week_label(draw: ImageDraw.ImageDraw, x: float, y: float, scale: int, color: tuple[int, int, int]) -> None:
    cursor = int(round(x * scale))
    top = int(round(y * scale))
    pixel = max(1, int(round(3.0 * scale)))
    for letter in "WEEK":
        glyph = _BITMAP[letter]
        for row, bits in enumerate(glyph):
            for column, bit in enumerate(bits):
                if bit == "1":
                    draw.rectangle(
                        (cursor + column * pixel, top + row * pixel,
                         cursor + (column + 1) * pixel - 1, top + (row + 1) * pixel - 1),
                        fill=color,
                    )
        cursor += 7 * pixel


def _route_points(scene: dict) -> list[tuple[float, float]]:
    """Return the route after sorting equivalent serialized x-position order."""
    column_count = int(scene["column_count"])
    x_positions = sorted(float(value) for value in scene["x_positions"])
    if len(x_positions) != column_count:
        raise ValueError("column_count must equal the number of x positions")
    top_y = float(scene["top_y"])
    bottom_y = float(scene["bottom_y"])
    slope = float(scene["diagonal_slope"])

    def lane_ends(x_value: float) -> tuple[float, float]:
        shift = slope * (x_value - x_positions[0])
        return top_y + shift, bottom_y + shift

    first_top, _ = lane_ends(x_positions[0])
    points: list[tuple[float, float]] = [(x_positions[0], first_top)]
    for index, x_value in enumerate(x_positions):
        lane_top, lane_bottom = lane_ends(x_value)
        end_y = lane_bottom if index % 2 == 0 else lane_top
        points.append((x_value, end_y))
        if index + 1 < column_count:
            next_top, next_bottom = lane_ends(x_positions[index + 1])
            next_end = next_bottom if index % 2 == 0 else next_top
            points.append((x_positions[index + 1], next_end))
    return points


def render(scene: dict) -> Image.Image:
    """Render one panel, with the decisive evidence as a single saturated path."""
    width = int(scene["width"])
    height = int(scene["height"])
    supersample = 4 if bool(scene["antialias"]) else 1
    scale = supersample
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    panel_width = float(scene["panel_width"])
    panel_height = float(scene["panel_height"])
    panel = (cx - panel_width / 2.0, cy - panel_height / 2.0,
             cx + panel_width / 2.0, cy + panel_height / 2.0)
    draw.rounded_rectangle(
        _scale_box(panel, scale),
        radius=16.0 * scale,
        fill=tuple(scene["panel_color"]),
        outline=tuple(scene["rail_color"]),
        width=max(2, int(round(3.0 * scale))),
    )

    label_x = panel[0] + 24.0
    label_y = panel[1] + 18.0
    _draw_week_label(draw, label_x, label_y, scale, tuple(scene["label_color"]))

    top_y = float(scene["top_y"])
    bottom_y = float(scene["bottom_y"])
    inner_left = min(float(value) for value in scene["x_positions"]) - 27.0
    inner_right = max(float(value) for value in scene["x_positions"]) + 27.0
    row_color = tuple(scene["row_color"])
    slope = float(scene["diagonal_slope"])
    first_x = min(float(value) for value in scene["x_positions"])
    for fraction in (1.0 / 3.0, 2.0 / 3.0):
        left_y = top_y + (bottom_y - top_y) * fraction + slope * (inner_left - first_x)
        right_y = top_y + (bottom_y - top_y) * fraction + slope * (inner_right - first_x)
        draw.line(
            (int(round(inner_left * scale)), int(round(left_y * scale)),
             int(round(inner_right * scale)), int(round(right_y * scale))),
            fill=row_color,
            width=max(1, int(round(2.0 * scale))),
        )

    # Small achromatic marks provide nuisance variation away from the route.
    mark_color = tuple(scene["mark_color"])
    offset = float(scene["mark_offset"])
    for index, (mx, my) in enumerate(((78.0, 132.0), (560.0, 150.0), (95.0, 516.0), (548.0, 500.0))):
        shift = offset if index % 2 == 0 else -offset
        size = 5.0 + (index % 2) * 2.0
        draw.ellipse(_scale_box((mx + shift - size, my - size, mx + shift + size, my + size), scale), fill=mark_color)

    route_color = tuple(scene["route_color"])
    route_width = max(1, int(round(float(scene["route_width"]) * scale)))
    points = [(int(round(x * scale)), int(round(y * scale))) for x, y in _route_points(scene)]
    draw.line(points, fill=route_color, width=route_width)
    radius = route_width / 2.0
    for x, y in points:
        draw.ellipse((int(round(x - radius)), int(round(y - radius)),
                      int(round(x + radius)), int(round(y + radius))), fill=route_color)

    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict) -> int:
    """The only analytic label input is the visible route's lane count."""
    return int(scene["column_count"])


def margin(scene: dict) -> float:
    """Return a one-sided raster visibility margin for the repeated lanes."""
    x_positions = sorted(float(value) for value in scene["x_positions"])
    gaps = [x_positions[index + 1] - x_positions[index] for index in range(len(x_positions) - 1)]
    route_width = float(scene["route_width"])
    lane_height = float(scene["bottom_y"]) - float(scene["top_y"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    panel_width = float(scene["panel_width"])
    panel_height = float(scene["panel_height"])
    panel_clearance = min(
        cx - panel_width / 2.0,
        float(scene["width"]) - cx - panel_width / 2.0,
        cy - panel_height / 2.0,
        float(scene["height"]) - cy - panel_height / 2.0,
    )
    color_contrast = min(
        abs(int(scene["route_color"][2]) - int(scene["panel_color"][2])),
        abs(int(scene["route_color"][1]) - int(scene["panel_color"][1])),
    )
    return float(min(
        route_width - 7.5,
        min(gaps) - route_width - 8.0,
        lane_height - 70.0,
        panel_clearance - 18.0,
        color_contrast - 18.0,
    ))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Cyclic serialization order of x positions is invisible after sorting."""
    positions = list(scene["x_positions"])
    twin = dict(scene)
    twin["x_positions"] = positions[1:] + positions[:1]
    return [("cyclic_x_position_serialization", twin)]
