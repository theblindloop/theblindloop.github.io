"""Independent pixel-only inverse for the connected WEEK trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstain(ValueError):
    """The raster does not expose one clean connected lane trace."""


def _route_mask(image: Image.Image) -> np.ndarray:
    values = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = values[..., 0]
    green = values[..., 1]
    blue = values[..., 2]
    # The route is the only saturated blue/teal mark. Neutral panel, rails,
    # row guides, label, and remote marks fail this sign-blind color test.
    return (blue - red >= 42) & (green - red >= 38) & (blue >= 90)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not dy and not dx:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 80:
            components.append(np.asarray(points, dtype=np.float64))
    return components


def _vertical_run_centers(points: np.ndarray) -> tuple[list[float], float]:
    y_min = float(points[:, 0].min())
    y_max = float(points[:, 0].max())
    x_min = int(points[:, 1].min())
    x_max = int(points[:, 1].max())
    height = y_max - y_min
    if height < 150.0 or x_max - x_min < 100:
        raise OracleAbstain("trace is below the regional visibility floor")
    # A lane is identified by a tall local y-span. This remains valid when
    # the entire serpentine trace drifts diagonally across the panel.
    spans = np.zeros(x_max - x_min + 1, dtype=np.float64)
    for x in range(x_min, x_max + 1):
        column = points[points[:, 1].astype(int) == x, 0]
        if len(column):
            spans[x - x_min] = float(column.max() - column.min())
    active = spans >= max(38.0, 0.22 * height)
    runs: list[tuple[int, int]] = []
    start: int | None = None
    for index, value in enumerate(active):
        if value and start is None:
            start = index
        elif not value and start is not None:
            runs.append((start, index - 1))
            start = None
    if start is not None:
        runs.append((start, len(active) - 1))
    if len(runs) < 4 or len(runs) > 10:
        raise OracleAbstain(f"expected four through ten lane runs, found {len(runs)}")
    centers = [x_min + (left + right) / 2.0 for left, right in runs]
    if len(centers) > 1:
        gaps = np.diff(np.asarray(centers))
        if float(gaps.min()) < 16.0:
            raise OracleAbstain("neighboring lane runs are not separated")
        separation_margin = float(gaps.min() - 16.0)
    else:
        separation_margin = 0.0
    return centers, separation_margin


def decision_from_image(image: Image.Image) -> int:
    """Count long vertical traversals from the PNG without scene access."""
    mask = _route_mask(image)
    components = _components(mask)
    if not components:
        raise OracleAbstain("no saturated connected trace found")
    components.sort(key=len, reverse=True)
    largest = components[0]
    if len(largest) < 500:
        raise OracleAbstain("connected trace has too little visible ink")
    if len(components) > 1 and len(components[1]) > 0.28 * len(largest):
        raise OracleAbstain("more than one substantial saturated trace component")
    centers, _ = _vertical_run_centers(largest)
    return len(centers)
