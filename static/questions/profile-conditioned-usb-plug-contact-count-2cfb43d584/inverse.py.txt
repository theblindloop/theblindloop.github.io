"""Independent pixel-only inverse arm for gold contact-pad counting."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        count = 0
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            yy, xx = queue.popleft()
            count += 1
            xs.append(xx)
            ys.append(yy)
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 250:
            found.append((count, min(xs), min(ys), max(xs), max(ys)))
    return found


def decision_from_image(image: Image.Image) -> str:
    """Return the count using only RGB pixels; raise when evidence is absent."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    # Wide but specific gold threshold, independent of renderer constants/code.
    mask = (r >= 190) & (r <= 250) & (g >= 125) & (g <= 205) & (b <= 90) & ((r - g) >= 35)
    components = _components(mask)
    valid = []
    for area, x0, y0, x1, y1 in components:
        width, height = x1 - x0 + 1, y1 - y0 + 1
        fill = area / float(width * height)
        if width >= 17 and height >= 30 and fill >= 0.72:
            valid.append((area, x0, y0, x1, y1))
    if len(valid) not in (3, 4, 5):
        raise ValueError(f"expected 3, 4, or 5 measurable gold contact pads, found {len(valid)}")
    return str(len(valid))
