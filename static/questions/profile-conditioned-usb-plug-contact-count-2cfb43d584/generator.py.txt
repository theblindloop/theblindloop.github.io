"""Deterministic renderer for a prior-conflict USB contact-count world."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
GOLD = (232, 174, 48)
BACKGROUND = (244, 247, 250)

BODY_COLORS = ((38, 100, 155), (126, 62, 139), (36, 126, 102), (184, 73, 58))
SHELL_COLORS = ((174, 184, 193), (199, 205, 211), (150, 163, 174))
SCALES = (0.88, 1.0, 1.12)
TARGET_AREAS = (8200.0, 10400.0, 12600.0)
PAD_WIDTHS = (29.0, 34.0, 38.0)


def sample_scene(seed: int) -> dict:
    """Sample balanced label and nuisance factors without using global state."""
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    count = 3 + (int(seed) % 3)
    style_index = (int(seed) // 3) % 3
    return {
        "contact_count": count,
        "body_color": list(BODY_COLORS[(int(seed) // 9) % len(BODY_COLORS)]),
        "shell_color": list(SHELL_COLORS[(int(seed) // 27) % len(SHELL_COLORS)]),
        "x_offset": rng.choice((-28, -14, 0, 14, 28)),
        "y_offset": rng.choice((-20, -10, 0, 10, 20)),
        "scale": SCALES[(int(seed) // 81) % len(SCALES)],
        "orientation": "contacts_top" if ((int(seed) // 243) % 2 == 0) else "contacts_bottom",
        "construction_order": "left_to_right" if rng.randrange(2) == 0 else "right_to_left",
        "target_gold_area": TARGET_AREAS[style_index],
        "pad_width": PAD_WIDTHS[(style_index + (int(seed) // 9)) % len(PAD_WIDTHS)],
    }


def _geometry(scene: dict) -> dict:
    count = int(scene["contact_count"])
    scale = float(scene["scale"])
    cx = WIDTH / 2 + float(scene["x_offset"])
    cy = HEIGHT / 2 + float(scene["y_offset"])
    shell_w, shell_h = 324 * scale, 224 * scale
    socket_w, socket_h = 272 * scale, 148 * scale
    body_w, body_h = 376 * scale, 112 * scale

    body_y0 = cy + shell_h / 2 - 2 * scale
    body = (cx - body_w / 2, body_y0, cx + body_w / 2, body_y0 + body_h)
    shell = (cx - shell_w / 2, cy - shell_h / 2, cx + shell_w / 2, cy + shell_h / 2)
    socket = (cx - socket_w / 2, cy - socket_h / 2, cx + socket_w / 2, cy + socket_h / 2)

    pad_w = float(scene["pad_width"]) * scale
    target_area = float(scene["target_gold_area"]) * scale * scale
    pad_h = max(42 * scale, min(92 * scale, target_area / (count * pad_w)))
    gap = (socket_w - count * pad_w) / (count + 1)
    if scene["orientation"] == "contacts_top":
        py0 = socket[1] + 17 * scale
        py1 = py0 + pad_h
    else:
        py1 = socket[3] - 17 * scale
        py0 = py1 - pad_h
    pads = []
    for i in range(count):
        x0 = socket[0] + gap * (i + 1) + pad_w * i
        pads.append((x0, py0, x0 + pad_w, py1))
    return {"body": body, "shell": shell, "socket": socket, "pads": pads, "gap": gap}


def render(scene: dict) -> Image.Image:
    """Render a clean antialiased front view; no text or answer metadata."""
    factor = 3
    g = _geometry(scene)
    image = Image.new("RGB", (WIDTH * factor, HEIGHT * factor), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
        return tuple(round(v * factor) for v in values)

    body = box(g["body"])
    shell = box(g["shell"])
    socket = box(g["socket"])
    radius = round(18 * float(scene["scale"]) * factor)
    draw.rounded_rectangle(body, radius=round(25 * factor), fill=tuple(scene["body_color"]), outline=(26, 40, 52), width=5 * factor)
    draw.rounded_rectangle(shell, radius=radius, fill=tuple(scene["shell_color"]), outline=(54, 64, 73), width=5 * factor)
    draw.rounded_rectangle(socket, radius=round(10 * factor), fill=(28, 33, 38), outline=(84, 94, 102), width=3 * factor)

    # Pads do not overlap, so construction order is a true latent alias.
    for pad in g["pads"]:
        draw.rounded_rectangle(box(pad), radius=round(3 * factor), fill=GOLD, outline=(151, 100, 15), width=2 * factor)

    # Symmetric shell highlights reinforce the plug gestalt without affecting count.
    sx0, sy0, sx1, sy1 = shell
    draw.line((sx0 + 28 * factor, sy0 + 13 * factor, sx1 - 28 * factor, sy0 + 13 * factor), fill=(230, 234, 237), width=3 * factor)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["contact_count"]))


def margin(scene: dict) -> float:
    """Runner-up separation margin: minimum dark gap between adjacent pads."""
    return float(_geometry(scene)["gap"])


def is_quarantined(scene: dict) -> bool:
    g = _geometry(scene)
    min_dim = min(min(p[2] - p[0], p[3] - p[1]) for p in g["pads"])
    return bool(g["gap"] < 12.0 or min_dim < 18.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["construction_order"] = (
        "right_to_left" if scene["construction_order"] == "left_to_right" else "left_to_right"
    )
    return [("reverse_nonoverlapping_draw_order", twin)]
