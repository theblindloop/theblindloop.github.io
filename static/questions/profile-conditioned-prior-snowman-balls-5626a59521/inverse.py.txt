"""Independent pixel-only inverse oracle for the snowman ball-count world.

Receives only the final PNG and re-counts the stacked snowballs directly from
the raster. It imports nothing from the renderer, prompts, generator, verifier,
or analytic-gold code, and accepts no latent scene data.

Method: the snowman is a vertical stack of near-white snowball discs on a light
blue sky. Step 1 thresholds the near-white snow ink and keeps the blue sky out.
Step 2 measures the horizontal width of the snow silhouette row by row. Each
solid snowball contributes one single-peaked "bulge" to that width profile;
where two stacked balls touch, the profile dips to a local minimum (the pinch).
Counting the number of bulges (local maxima of the width profile) therefore
equals the number of stacked snowballs.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _snow_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    # Near-white snowball fill; the light-blue sky has a much lower red channel.
    return (r > 226) & (g > 226) & (b > 210)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    mask = _snow_mask(arr)
    if mask.sum() == 0:
        return "0"
    cols = mask.sum(axis=1).astype(np.float64)
    ys = np.nonzero(cols)[0]
    if len(ys) == 0:
        return "0"
    lo, hi = int(ys.min()), int(ys.max())
    profile = cols[lo : hi + 1]

    # Smooth the row-width profile to remove single-row antialias noise.
    kernel = np.ones(5, dtype=np.float64) / 5.0
    if len(profile) >= 5:
        profile = np.convolve(profile, kernel, mode="same")

    return str(_count_lobes(profile))


def _count_lobes(prof: np.ndarray) -> int:
    """Count the separated arches (one per stacked snowball) in a width profile.

    Each solid snowball yields one arch; consecutive arch peaks are separated by
    a deep valley (the pinch where two stacked balls touch), while the small
    roughness on a single arch's near-flat top only produces shallow dips
    between adjacent maxima. We therefore collect all local maxima and merge any
    adjacent pair whose intervening valley does not drop below half the smaller
    peak (i.e. they belong to the same arch).
    """
    n = len(prof)
    if n < 3:
        return 0
    peaks: list[int] = []
    for i in range(1, n - 1):
        if prof[i] > prof[i - 1] and prof[i] >= prof[i + 1]:
            if not peaks or i - peaks[-1] >= 3:
                peaks.append(i)
    if not peaks:
        return 0
    count = len(peaks)
    for k in range(len(peaks) - 1):
        a = peaks[k]
        b = peaks[k + 1]
        valley = float(prof[a + 1 : b].min())
        joined = valley >= 0.5 * min(float(prof[a]), float(prof[b]))
        if joined:
            count -= 1
    return count
