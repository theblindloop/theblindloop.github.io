"""Prior-conflict snowman ball-count world renderer.

Latent state z = {n_balls, base_r, bottom_y, center_x, shrink_chain,
h_offsets}.

A classic snowman is drawn as a vertical stack of solid near-white snowball
circles on a light-blue sky field. The largest ball sits at the bottom and each
higher ball is a slightly smaller disc resting on the one below (externally
tangent), so every ball is a distinct filled circle and together they form a
single connected vertical silhouette.

The closed-form gold is the number of snowballs drawn:

    ball_count = n_balls   (a scene parameter, not a pixel measurement)

The canonical prior -- that a snowman is made of three stacked snowballs -- is
recorded only in the question narrative and is never used as ground truth. The
visible truth is however many snowballs are actually drawn.

Continuous verifiability margin (px-equivalent, reported as a diameter in px):
the diameter of the smallest snowball in the stack. As long as the smallest
drawn ball stays above a finite-resolution floor it remains a resolvable,
individually countable circular lobe.

Latent-alias mode: not_applicable. The ball count is pinned by n_balls, which
is rendered as `n_balls` stacked near-white snowball lobes; different n_balls
always produce a different (and differently-counted) raster, so there is no
render-invariant latent alias and sample_scene returns no pixel-identical
twins.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS_W = 384
CANVAS_H = 384
CANVAS_BG = (200, 230, 250)     # light blue sky
SNOW_COLOR = (250, 250, 252)    # near-white snowball fill

N_BALLS_RANGE = (2, 4)          # inclusive; canonical prior is 3
BASE_R_LO = 46.0
BASE_R_HI = 62.0
SHRINK_RANGE = (0.70, 0.84)     # per-step radius reduction between stacked balls
MIN_BALL_DIAM = 30.0            # resolution floor below which count is ambiguous
TOP_MARGIN = 34.0               # px clearance for the topmost ball above the canvas top
BOTTOM_Y = 338.0
CENTER_X_LO = 182.0
CENTER_X_HI = 202.0
H_JITTER = 12.0                 # max horizontal offset of a ball from stack axis

RENDERER_VERSION = "prior_snowman_balls-0.1.0"


def _balls(scene: dict[str, Any]) -> list[tuple[float, float, float]]:
    """Return [(cx, cy, r), ...] for each snowball, bottom to top, tangent-stacked."""
    n = int(scene["n_balls"])
    cx0 = float(scene["center_x"])
    base_r = float(scene["base_r"])
    bottom_y = float(scene["bottom_y"])
    shrink = scene["shrink_chain"]
    h_offsets = scene["h_offsets"]
    balls = []
    r = base_r
    cy = bottom_y - r
    for k in range(n):
        offset = float(h_offsets[k]) if k < len(h_offsets) else 0.0
        cx = cx0 + offset
        balls.append((cx, cy, r))
        if k < n - 1:
            next_r = r * float(shrink[k])
            cy = cy - (r + next_r)
            r = next_r
    return balls


def _radii_from(base_r: float, shrink: list[float]) -> list[float]:
    radii = [base_r]
    for ratio in shrink:
        radii.append(radii[-1] * float(ratio))
    return radii


def _valid(n_balls: int, base_r: float, shrink: list[float], bottom_y: float) -> bool:
    radii = _radii_from(base_r, shrink)
    if len(radii) != n_balls:
        return False
    if 2.0 * min(radii) < MIN_BALL_DIAM:
        return False
    total_height = 2.0 * sum(radii)
    top = bottom_y - total_height
    if top < TOP_MARGIN:
        return False
    return True


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_balls = rng.randint(*N_BALLS_RANGE)
    center_x = rng.uniform(CENTER_X_LO, CENTER_X_HI)
    bottom_y = BOTTOM_Y
    shrink: list[float] = []
    base_r = 0.0
    for _attempt in range(300):
        base_r = rng.uniform(BASE_R_LO, BASE_R_HI)
        shrink = []
        for _ in range(n_balls - 1):
            shrink.append(rng.uniform(*SHRINK_RANGE))
        if _valid(n_balls, base_r, shrink, bottom_y):
            break
    h_offsets = [round(rng.uniform(-H_JITTER, H_JITTER), 3) for _ in range(n_balls)]
    return {
        "n_balls": n_balls,
        "base_r": round(base_r, 3),
        "bottom_y": round(bottom_y, 3),
        "center_x": round(center_x, 3),
        "shrink_chain": [round(v, 6) for v in shrink],
        "h_offsets": h_offsets,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    n = int(scene["n_balls"])
    if n < 1:
        raise ValueError("a snowman needs at least one snowball")
    scale = 2
    big_w = CANVAS_W * scale
    big_h = CANVAS_H * scale
    big = Image.new("RGB", (big_w, big_h), CANVAS_BG)
    draw = ImageDraw.Draw(big)
    for (cx, cy, r) in _balls(scene):
        draw.ellipse(
            [
                (cx - r) * scale,
                (cy - r) * scale,
                (cx + r) * scale,
                (cy + r) * scale,
            ],
            fill=SNOW_COLOR,
        )
    return big.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)


def _min_ball_diam(scene: dict[str, Any]) -> float:
    radii = [r for (_cx, _cy, r) in _balls(scene)]
    return 2.0 * min(radii)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_balls"]))


def margin(scene: dict[str, Any]) -> float:
    # Decision-relevant separation is the diameter of the smallest snowball:
    # below the resolution floor that smallest lobe becomes ambiguous.
    return _min_ball_diam(scene)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return _min_ball_diam(scene) < MIN_BALL_DIAM


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_balls; there is no
    # declared pixel-identical alias to exercise.
    return []
