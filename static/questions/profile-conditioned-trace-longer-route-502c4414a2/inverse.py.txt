"""Pixel-only inverse arm for the trace-longer-route world.

Reads only the final RGB PNG. Segments route pixels by nearest-colour labelling
against the declared palette (blue, green, white), then compares the total ink
of the blue route against the total ink of the green route. Because every route
is drawn at a fixed, identical stroke width, total coloured ink is exactly
proportional to the geodesic (on-stroke) length of that route, so the route with
more ink is the longer route.

No renderer, prompt, generator, verifier, latent scene or analytic gold is
imported or read; only PIL/numpy and the public colour contract.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 230),
    "green": (20, 180, 70),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "green", "white"]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    labels = dists.argmin(axis=-1)
    idx = {name: i for i, name in enumerate(_NAMES)}
    count_blue = int(np.count_nonzero(labels == idx["blue"]))
    count_green = int(np.count_nonzero(labels == idx["green"]))
    if count_blue == 0 and count_green == 0:
        raise ValueError("no route pixels found in the image")
    return "blue" if count_blue >= count_green else "green"
