"""Tracing world: which of two winding routes is longer (geodesic length).

A single scene draws two routes, one blue and one green. Each route is a
serpentine (zig-zag) polyline that folds back and forth on itself inside its
own horizontal band, so its total length is a geodesic-on-the-stroke fact:
the straight-line span (bounding box / endpoint distance) badly underestimates
the true length because the route winds. Walking along each stroke, the longer
route is the one with the greater on-stroke length. The two routes are placed
in separate vertical bands so their strokes never touch or overlap, keeping the
colour segmentation of every route pixel unambiguous.

The decision (blue or green) is fully determined by the final pixels: a viewer
reconstructs each route from its stroke, measures its length, and compares.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "trace_longer_route-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 230)
COLOR_GREEN = (20, 180, 70)

STROKE = 14
PAD = 50
BAND_TOP = 80
BAND_BOT = 210
BAND_TOP2 = 270
BAND_BOT2 = 400

MIN_RATIO = 0.12   # min relative length margin between the two routes
ABS_QUARANTINE = 50.0

ANSWER_VALUES = ("blue", "green")


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _serpentine(
    rng: random.Random,
    x_lo: float,
    x_hi: float,
    y_top: float,
    y_bot: float,
    n_lanes: int,
) -> list[tuple[float, float]]:
    up = rng.random() < 0.5
    pts: list[tuple[float, float]] = []
    for lane in range(n_lanes):
        frac = lane / (n_lanes - 1) if n_lanes > 1 else 0.0
        y = y_top + (y_bot - y_top) * frac
        if up:
            pts.append((x_lo, y))
            pts.append((x_hi, y))
        else:
            pts.append((x_hi, y))
            pts.append((x_lo, y))
        up = not up
    return pts


def _rejection_loop(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(20000):
        # Blue route: upper band.
        n_bl = rng.randint(4, 7)
        x_lo_b = rng.uniform(PAD, PAD + 40)
        x_hi_b = rng.uniform(w - PAD - 40, w - PAD)
        blue_pts = _serpentine(rng, x_lo_b, x_hi_b, BAND_TOP, BAND_BOT, n_bl)
        len_b = _path_length(blue_pts)

        # Green route: lower band.
        n_gr = rng.randint(4, 7)
        x_lo_g = rng.uniform(PAD, PAD + 40)
        x_hi_g = rng.uniform(w - PAD - 40, w - PAD)
        green_pts = _serpentine(rng, x_lo_g, x_hi_g, BAND_TOP2, BAND_BOT2, n_gr)
        len_g = _path_length(green_pts)

        m = abs(len_b - len_g)
        if m < MIN_RATIO * max(len_b, len_g):
            continue

        scene = {
            "routes": {
                "blue": [[float(p[0]), float(p[1])] for p in blue_pts],
                "green": [[float(p[0]), float(p[1])] for p in green_pts],
            }
        }
        return scene
    raise RuntimeError("could not satisfy route constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    return _rejection_loop(rng)


def _draw_route(draw: ImageDraw.ImageDraw, pts, color):
    draw.line(pts, fill=color, width=STROKE, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    colors = {"blue": COLOR_BLUE, "green": COLOR_GREEN}
    for name, pts in scene["routes"].items():
        _draw_route(draw, _pts(pts), colors[name])
    return img


def _route_length(scene: dict, name: str) -> float:
    return _path_length(_pts(scene["routes"][name]))


def analytic_gold(scene: dict) -> str:
    """Return the colour of the longer route (greater on-stroke length)."""
    lb = _route_length(scene, "blue")
    lg = _route_length(scene, "green")
    return "blue" if lb >= lg else "green"


def margin(scene: dict) -> float:
    """Absolute length difference between the two routes (px)."""
    lb = _route_length(scene, "blue")
    lg = _route_length(scene, "green")
    return abs(lb - lg)


def is_quarantined(scene: dict) -> bool:
    lb = _route_length(scene, "blue")
    lg = _route_length(scene, "green")
    dif = abs(lb - lg)
    if dif < ABS_QUARANTINE:
        return True
    if dif < MIN_RATIO * max(lb, lg):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant transform: the route colours and the
    geometry of each route are directly visible, and any change to either
    route's colour or length changes both the pixels and the gold. Declared
    empty (not_applicable)."""
    return []
