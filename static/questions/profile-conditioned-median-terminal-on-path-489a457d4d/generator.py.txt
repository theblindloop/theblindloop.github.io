"""Tracing world: which coloured terminal is the median along the winding path.

A single winding blue path (one connected, non-self-crossing serpentine
polyline) is drawn. An orange port ring sits near the start of the path. Three
coloured terminal dots (red, green, magenta) sit at three mutually separated
on-path positions. Walking only along the blue path from the orange ring, the
solution is the colour of the terminal whose on-path (geodesic) distance from
the ring is the MIDDLE (median) of the three. Because the path winds back and
forth in a serpentine, the on-path order of the three terminals need not follow
their straight-line proximity to the ring, so the decision cannot be solved by
Euclidean projection; it requires tracing the drawn stroke and ranking the three
markers by their on-path distance from the shared port.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "median_terminal_on_path-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_BLUE = (40, 90, 210)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (40, 170, 70)
COLOR_MAGENTA = (205, 45, 205)
COLOR_ORANGE = (245, 150, 20)

STROKE = 20          # blue stroke thickness (px)
PORT_R = 5           # orange port radius
MARK_R = 4           # terminal marker radius
MARGIN_PAD = 46      # keep geometry away from image borders
LANE_GAP = 30.0      # min gap between serpentine lanes so markers map to their own stroke
ARC_SEP = 34.0       # min arc distance between consecutive terminals and from the port
END_MARGIN = 40.0    # min distance from path endpoints for the terminals

TERMINAL_COLORS = ["red", "green", "magenta"]


def _pts(points: list) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _point_at(points: list[tuple[float, float]], s: float) -> tuple[float, float]:
    remaining = s
    for a, b in zip(points, points[1:]):
        seg = math.dist(a, b)
        if remaining <= seg or seg == 0:
            t = 0.0 if seg == 0 else remaining / seg
            return (a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t)
        remaining -= seg
    return points[-1]


def _rejection_loop(rng: random.Random, forced_median: str) -> dict:
    w, h = CANVAS
    lo, hi = MARGIN_PAD, w - MARGIN_PAD
    for _ in range(30000):
        lane_vert_gap = LANE_GAP
        n_lanes = rng.randint(4, 7)
        y_lo = lo + 10
        y_hi = hi - 10
        usable_h = y_hi - y_lo
        span_h = (n_lanes - 1) * lane_vert_gap
        span_h = min(span_h, usable_h)
        step_h = span_h / (n_lanes - 1) if n_lanes > 1 else 0.0
        x_start = rng.uniform(lo + 15, lo + 50)
        x_end = rng.uniform(hi - 50, hi - 15)
        path = []
        for lane in range(n_lanes):
            y = y_lo + lane * step_h
            if lane % 2 == 0:
                path.append((x_start, y))
                path.append((x_end, y))
            else:
                path.append((x_end, y))
                path.append((x_start, y))
        path = _pts(path)
        total = _path_length(path)
        if total < END_MARGIN * 6:
            continue

        s_port = END_MARGIN * rng.uniform(1.0, 1.5)

        lo_s = s_port + 4 * ARC_SEP
        hi_s = total - END_MARGIN - 20
        if hi_s - lo_s < 8 * ARC_SEP:
            continue

        # Three ordered on-path positions, mutually separated by >= ARC_SEP.
        p1 = rng.uniform(lo_s, lo_s + 3 * ARC_SEP)
        p3 = rng.uniform(hi_s - 3 * ARC_SEP, hi_s)
        low2, high2 = p1 + ARC_SEP, p3 - ARC_SEP
        if high2 - low2 < ARC_SEP:
            continue
        p2 = rng.uniform(low2, high2)
        positions = {"p1": p1, "p2": p2, "p3": p3}

        # Assign colours to the three positions; the forced median colour is
        # placed at the middle on-path position p2, the other two colours
        # shuffle into the near and far positions so the answer colour is
        # decorrelated from spatial position / lane side.
        others = [c for c in TERMINAL_COLORS if c != forced_median]
        rng.shuffle(others)
        median_color = forced_median
        order = [others[0], median_color, others[1]]
        owner = {positions[k]: order[i] for i, k in enumerate(("p1", "p2", "p3"))}

        marks = []
        for key in ("p1", "p2", "p3"):
            pos = _point_at(path, positions[key])
            col = owner[positions[key]]
            marks.append(
                {"x": float(pos[0]), "y": float(pos[1]), "color": col, "slot": key}
            )

        allgood = True
        port = _point_at(path, s_port)
        for m in marks + [{"x": port[0], "y": port[1]}]:
            if not (20 < m["x"] < w - 20 and 20 < m["y"] < h - 20):
                allgood = False
                break
        if not allgood:
            continue

        scene = {
            "path": path,
            "port": [float(port[0]), float(port[1])],
            "marks": marks,
            "answer": median_color,
        }
        return scene
    raise RuntimeError("could not satisfy path constraints")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    forced_median = TERMINAL_COLORS[seed % 3]
    return _rejection_loop(rng, forced_median)


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, color, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    path = _pts(scene["path"])
    _draw_polyline(draw, path, COLOR_BLUE, STROKE)
    colors = {"red": COLOR_RED, "green": COLOR_GREEN, "magenta": COLOR_MAGENTA}
    for mark in scene["marks"]:
        r = MARK_R
        px, py = float(mark["x"]), float(mark["y"])
        draw.ellipse([px - r, py - r, px + r, py + r], fill=colors[mark["color"]])
    px, py = float(scene["port"][0]), float(scene["port"][1])
    draw.ellipse([px - PORT_R, py - PORT_R, px + PORT_R, py + PORT_R], fill=COLOR_ORANGE)
    return img


def _arc_position(scene: dict, pt: tuple[float, float]) -> float:
    best = None
    best_d = float("inf")
    cum = 0.0
    for a, b in zip(_pts(scene["path"]), _pts(scene["path"])[1:]):
        abx, aby = b[0] - a[0], b[1] - a[1]
        seg2 = abx * abx + aby * aby
        t = 0.0 if seg2 == 0 else ((pt[0] - a[0]) * abx + (pt[1] - a[1]) * aby) / seg2
        t = max(0.0, min(1.0, t))
        proj = (a[0] + abx * t, a[1] + aby * t)
        d = math.dist(pt, proj)
        if d < best_d:
            best_d = d
            best = cum + math.dist(a, proj)
        cum += math.dist(a, b)
    return best


def analytic_gold(scene: dict) -> str:
    """Colour of the terminal whose on-path distance from the ring is the median."""
    dists = []
    for mark in scene["marks"]:
        d = abs(_arc_position(scene, (mark["x"], mark["y"])) - _arc_position(scene, tuple(scene["port"])))
        dists.append((d, mark["color"]))
    dists.sort(key=lambda item: item[0])
    return dists[1][1]


def margin(scene: dict) -> float:
    """Smallest on-path separation between the three ordered terminals (px)."""
    dists = sorted(
        abs(
            _arc_position(scene, (m["x"], m["y"]))
            - _arc_position(scene, tuple(scene["port"]))
        )
        for m in scene["marks"]
    )
    gaps = [
        b - a for a, b in zip([dists[0]] + dists, dists) if b - a > 1e-9
    ]
    return min(gaps) if gaps else 0.0


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < ARC_SEP * 1.5:
        return True
    for m in scene["marks"]:
        if not (20 <= m["x"] <= w - 20 and 20 <= m["y"] <= h - 20):
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
