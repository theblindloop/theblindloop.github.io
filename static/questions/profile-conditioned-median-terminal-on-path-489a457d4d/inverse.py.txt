"""Pixel-only inverse arm for the median-terminal tracing world.

Reads only the final RGB PNG. Segments colours by nearest-palette labelling,
finds the blue path stroke, and computes a geodesic distance field from the
orange ring along the blue path by a breadth-first search over the blue pixels.
Each terminal (red, green, magenta) is mapped to the BFS distance of the blue
pixel nearest its marker; the decision is the colour of the terminal with the
MIDDLE (median) on-path distance from the ring. This requires tracing the
winding stroke and ranking three markers geodesically: straight-line proximity
to the ring does not determine the median because the path folds back on itself.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "blue": (40, 90, 210),
    "red": (225, 40, 40),
    "green": (40, 170, 70),
    "magenta": (205, 45, 205),
    "orange": (245, 150, 20),
    "white": (255, 255, 255),
}
_NAMES = ["blue", "red", "green", "magenta", "orange", "white"]
_TERMINALS = ["red", "green", "magenta"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _nearest_blue_to_marker(labels, blue_mask, name, idx, w, h, r=10):
    marker = labels == idx[name]
    ys, xs = np.nonzero(marker)
    if len(ys) == 0:
        return None
    cy = int(round(float(ys.mean())))
    cx = int(round(float(xs.mean())))
    best = None
    best_d = float("inf")
    for yy, xx in zip(ys.tolist(), xs.tolist()):
        for dy in range(-r, r + 1):
            for dx in range(-r, r + 1):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < h and 0 <= nx < w and blue_mask[ny, nx]:
                    dd = (ny - cy) ** 2 + (nx - cx) ** 2
                    if dd < best_d:
                        best_d = dd
                        best = (ny, nx)
    return best


def _bfs_distances(blue_mask: np.ndarray, seed_pts: list) -> np.ndarray:
    h, w = blue_mask.shape
    dist = np.full((h, w), -1, dtype=np.int64)
    dq = deque()
    for (y, x) in seed_pts:
        if 0 <= y < h and 0 <= x < w and blue_mask[y, x]:
            if dist[y, x] == -1:
                dist[y, x] = 0
                dq.append((y, x))
    while dq:
        y, x = dq.popleft()
        nd = dist[y, x] + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and blue_mask[ny, nx] and dist[ny, nx] == -1:
                dist[ny, nx] = nd
                dq.append((ny, nx))
    return dist


def _port_seed_blue(labels, idx):
    orange = labels == idx["orange"]
    h, w = orange.shape
    seeds = []
    for y in range(h):
        for x in range(w):
            if orange[y, x]:
                for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                    if 0 <= ny < h and 0 <= nx < w and labels[ny, nx] == idx["blue"]:
                        seeds.append((ny, nx))
    return seeds


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    blue = labels == idx["blue"]
    h, w = blue.shape

    seeds = _port_seed_blue(labels, idx)
    if not seeds:
        raise ValueError("port does not touch the blue path")
    dist = _bfs_distances(blue, seeds)
    if not (dist >= 0).any():
        raise ValueError("no blue reachable from port")

    marker_dists = {}
    for name in _TERMINALS:
        near = _nearest_blue_to_marker(labels, blue, name, idx, w, h)
        if near is None:
            raise ValueError(f"{name} marker not found")
        yy, xx = near
        if dist[yy, xx] < 0:
            raise ValueError(f"{name} marker not on reachable path")
        marker_dists[name] = dist[yy, xx]

    ranked = sorted(_TERMINALS, key=lambda name: marker_dists[name])
    return ranked[1]
