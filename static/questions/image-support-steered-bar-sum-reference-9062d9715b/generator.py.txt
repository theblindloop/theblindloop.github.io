"""Deterministic renderer and analytic arm for additive local bar measurement."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (247, 249, 250)
TICK_HALF_WIDTH = 1
TICK_HALF_HEIGHT = 5
QUARANTINE_PX = 4.0

PALETTES = (
    {"coral": (224, 88, 86), "blue": (48, 112, 184), "gold": (211, 158, 35)},
    {"coral": (205, 79, 78), "blue": (40, 126, 178), "gold": (224, 166, 38)},
    {"coral": (235, 102, 91), "blue": (55, 105, 170), "gold": (196, 145, 29)},
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    coral = rng.randint(8, 10)
    blue = rng.randint(8, 10)
    summed = coral + blue
    near = seed % 8 == 0
    delta = rng.randint(1, 2) if near else rng.randint(3, 4)
    positive = seed % 2 == 0
    gold = summed - delta if positive else summed + delta
    labels = ["coral", "blue", "gold"]
    rng.shuffle(labels)
    draw_order = labels[:]
    rng.shuffle(draw_order)
    return {
        "coral_length": coral,
        "blue_length": blue,
        "gold_length": gold,
        "center_x": rng.choice((160, 224, 288, 352)),
        "center_y": rng.choice((160, 224, 288, 352)),
        "row_order": labels,
        "draw_order": draw_order,
        "palette_index": rng.randrange(len(PALETTES)),
        "diag_sign": rng.choice((-1, 1)),
        "x_jitter": {label: rng.randint(-2, 2) for label in labels},
        "distractors": [
            [rng.randint(28, 484), rng.randint(28, 484), rng.randint(2, 4)]
            for _ in range(7)
        ],
    }


def _dot_centers(scene: dict) -> dict[str, tuple[tuple[int, int], tuple[int, int]]]:
    lengths = {
        "coral": int(scene["coral_length"]),
        "blue": int(scene["blue_length"]),
        "gold": int(scene["gold_length"]),
    }
    row_order = list(scene["row_order"])
    centers = {}
    for row, label in enumerate(row_order):
        length = lengths[label]
        offset = row - 1
        cy = int(scene["center_y"]) + offset * 64
        cx = (
            int(scene["center_x"])
            + offset * 64 * int(scene["diag_sign"])
            + int(scene.get("x_jitter", {}).get(label, 0))
        )
        left = cx - length // 2
        centers[label] = ((left, cy), (left + length, cy))
    return centers


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene.get("distractors", []):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=(211, 216, 219))
    centers = _dot_centers(scene)
    palette = PALETTES[int(scene["palette_index"])]
    for label in scene["draw_order"]:
        for cx, cy in centers[label]:
            draw.rectangle(
                (cx - TICK_HALF_WIDTH, cy - TICK_HALF_HEIGHT,
                 cx + TICK_HALF_WIDTH, cy + TICK_HALF_HEIGHT),
                fill=palette[label],
            )
    return image


def analytic_gold(scene: dict) -> str:
    total = int(scene["coral_length"]) + int(scene["blue_length"])
    return "yes" if total > int(scene["gold_length"]) else "no"


def margin(scene: dict) -> float:
    return float(abs(int(scene["coral_length"]) + int(scene["blue_length"]) - int(scene["gold_length"])))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["draw_order"] = list(reversed(scene["draw_order"]))
    return [("reverse_disjoint_draw_order", twin)]
