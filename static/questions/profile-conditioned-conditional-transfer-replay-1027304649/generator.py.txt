"""Deterministic renderer and analytic replay for Conditional Transfer Replay."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 560
SHAPES = ("circle", "triangle", "square", "diamond")
PALETTE = ("blue", "green", "orange", "purple")
RGB = {
    "paper": (250, 251, 253),
    "dark": (29, 35, 45),
    "muted": (101, 111, 126),
    "card": (222, 234, 247),
    "start": (250, 231, 173),
    "arrow": (205, 47, 121),
    "blue": (25, 111, 191),
    "green": (25, 148, 102),
    "orange": (230, 126, 34),
    "purple": (126, 82, 171),
}


def _replay(start: str, routes: list[list[str]]) -> str:
    state = start
    for source, destination in routes:
        if state == source:
            state = destination
    return state


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0xC07D17)
    desired = SHAPES[int(seed) % 4]
    last_effective = (int(seed) // 4) % 4
    start = rng.choice(SHAPES)
    if last_effective == 0 and start == desired:
        start = SHAPES[(SHAPES.index(start) + 1) % 4]

    routes: list[list[str]] = []
    state = start
    for index in range(4):
        if index == last_effective:
            if state == desired:
                alternatives = [shape for shape in SHAPES if shape != desired]
                source = rng.choice(alternatives)
                destination = rng.choice([shape for shape in SHAPES if shape != source])
                routes.append([source, destination])
            else:
                routes.append([state, desired])
                state = desired
        elif index < last_effective and rng.random() < 0.55:
            destinations = [shape for shape in SHAPES if shape != state and shape != desired]
            if destinations:
                destination = rng.choice(destinations)
                routes.append([state, destination])
                state = destination
            else:
                source = rng.choice([shape for shape in SHAPES if shape != state])
                routes.append([source, rng.choice([shape for shape in SHAPES if shape != source])])
        else:
            source = rng.choice([shape for shape in SHAPES if shape != state])
            destination = rng.choice([shape for shape in SHAPES if shape != source])
            routes.append([source, destination])

    if state != desired:
        # The only possible case is a deliberately skipped card at the designated
        # position because the state already equaled the desired answer.
        assert state == desired

    palette_order = list(PALETTE)
    rng.shuffle(palette_order)
    layout_shift = [rng.randint(-6, 6), rng.randint(-5, 5)]
    scene = {
        "start_symbol": start,
        "routes": routes,
        "palette_order": palette_order,
        "layout_shift": layout_shift,
    }
    assert analytic_gold(scene) == desired
    return scene


def _draw_shape(
    draw: ImageDraw.ImageDraw, shape: str, cx: int, cy: int, color: tuple[int, int, int]
) -> None:
    radius = 24
    if shape == "circle":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    elif shape == "triangle":
        draw.polygon(((cx, cy - radius), (cx - 25, cy + radius), (cx + 25, cy + radius)), fill=color)
    elif shape == "square":
        draw.rectangle((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    elif shape == "diamond":
        draw.polygon(((cx, cy - 25), (cx - 25, cy), (cx, cy + 25), (cx + 25, cy)), fill=color)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict[str, Any]) -> Image.Image:
    start = str(scene["start_symbol"])
    routes = [[str(value) for value in pair] for pair in scene["routes"]]
    palette_order = [str(value) for value in scene["palette_order"]]
    shift_x, shift_y = [int(value) for value in scene["layout_shift"]]
    if start not in SHAPES or len(routes) != 4:
        raise ValueError("scene requires one valid start and four routes")
    if any(len(pair) != 2 or pair[0] not in SHAPES or pair[1] not in SHAPES for pair in routes):
        raise ValueError("every route must have two valid symbol endpoints")
    if sorted(palette_order) != sorted(PALETTE):
        raise ValueError("palette_order must permute the four public colors")
    colors = {shape: RGB[name] for shape, name in zip(SHAPES, palette_order, strict=True)}

    image = Image.new("RGB", (WIDTH, HEIGHT), RGB["paper"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((34 + shift_x, 24 + shift_y), "PARCEL ROUTE: apply cards from left to right", fill=RGB["dark"], font=font)
    draw.text((34 + shift_x, 44 + shift_y), "Move only when the left symbol matches the parcel's current station.", fill=RGB["muted"], font=font)

    start_left, start_top = 374 + shift_x, 76 + shift_y
    draw.rectangle((start_left, start_top, start_left + 212, start_top + 156), fill=RGB["start"])
    draw.text((start_left + 86, start_top + 17), "START", fill=RGB["dark"], font=font)
    _draw_shape(draw, start, start_left + 106, start_top + 94, colors[start])

    card_y = 318 + shift_y
    for index, (source, destination) in enumerate(routes):
        left = 51 + shift_x + index * 226
        draw.rectangle((left, card_y, left + 181, card_y + 166), fill=RGB["card"])
        draw.text((left + 84, card_y + 13), str(index + 1), fill=RGB["muted"], font=font)
        _draw_shape(draw, source, left + 44, card_y + 92, colors[source])
        draw.rectangle((left + 76, card_y + 87, left + 104, card_y + 97), fill=RGB["arrow"])
        draw.polygon(
            ((left + 104, card_y + 76), (left + 125, card_y + 92), (left + 104, card_y + 108)),
            fill=RGB["arrow"],
        )
        _draw_shape(draw, destination, left + 154, card_y + 92, colors[destination])

    draw.text((324 + shift_x, 523 + shift_y), "source  ->  destination     (otherwise stay)", fill=RGB["muted"], font=font)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return _replay(str(scene["start_symbol"]), [[str(value) for value in pair] for pair in scene["routes"]])


def margin(scene: dict[str, Any]) -> float:
    _ = scene["routes"]
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        start = str(scene["start_symbol"])
        routes = scene["routes"]
        palette = scene["palette_order"]
        shift = scene["layout_shift"]
        return not (
            start in SHAPES
            and len(routes) == 4
            and all(len(pair) == 2 and pair[0] in SHAPES and pair[1] in SHAPES for pair in routes)
            and sorted(palette) == sorted(PALETTE)
            and len(shift) == 2
        )
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
