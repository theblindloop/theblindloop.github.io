"""Independent image-only inverse for adjacent horizontal transitions.

The implementation intentionally duplicates the public 9x10 raster contract and
arrow references.  It does not import or inspect renderer state, scene data,
manifests, filenames, metadata, or analytic gold.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

ROWS = 9
SOURCE_COLS = 10
RESULT_COLS = 9
QUARANTINE_GAP = 4.0
LABELS = ("up", "right", "down", "left")
FRAME_COLORS = np.array(
    ((26, 174, 190), (51, 142, 218), (0, 160, 150), (0, 150, 190)), dtype=np.int32
)

UP = (
    "....#....",
    "...###...",
    "..#####..",
    "....#....",
    "....#....",
    "....#....",
    "....#....",
    "....#....",
    "....#....",
)


def _rotate(rows: tuple[str, ...]) -> tuple[str, ...]:
    return tuple("".join(rows[ROWS - 1 - r][c] for r in range(ROWS)) for c in range(ROWS))


RIGHT = _rotate(UP)
DOWN = _rotate(RIGHT)
LEFT = _rotate(DOWN)
TEMPLATES = dict(zip(LABELS, (UP, RIGHT, DOWN, LEFT), strict=True))


class OracleError(ValueError):
    pass


def _find_grid(array: np.ndarray) -> tuple[int, int, int]:
    # int32 avoids overflow when squaring an 8-bit color difference.
    rgb = array.astype(np.int32)
    distance2 = ((rgb[:, :, None, :] - FRAME_COLORS[None, None, :, :]) ** 2).sum(axis=3)
    mask = distance2.min(axis=2) <= 12 * 12
    ys, xs = np.nonzero(mask)
    if len(xs) < 250:
        raise OracleError("cyan frame is missing or erased")
    left, right = int(xs.min()), int(xs.max())
    top, bottom = int(ys.min()), int(ys.max())

    # Infer cell size jointly from both frame spans.  The public frame has a
    # 6-pixel moat on each side, but its absolute image location is unknown.
    candidates = []
    for cell in range(22, 38):
        error = abs((right - left + 1) - (SOURCE_COLS * cell + 13))
        error += abs((bottom - top + 1) - (ROWS * cell + 13))
        candidates.append((error, cell))
    error, cell = min(candidates)
    if error > 4:
        raise OracleError("frame proportions do not contain a 9-by-10 field")
    return left + 6, top + 6, cell


def _source_from_pixels(image: Image.Image) -> list[list[int]]:
    array = np.asarray(image.convert("RGB"))
    x0, y0, cell = _find_grid(array)
    samples = []
    for row in range(ROWS):
        sample_row = []
        for col in range(SOURCE_COLS):
            x = x0 + col * cell + cell // 2
            y = y0 + row * cell + cell // 2
            if not (0 <= x < array.shape[1] and 0 <= y < array.shape[0]):
                raise OracleError("a sampled cell falls outside the raster")
            sample_row.append(tuple(int(v) for v in array[y, x]))
        samples.append(sample_row)

    colors = sorted({color for row in samples for color in row})
    if len(colors) != 2:
        raise OracleError(f"expected exactly two tile-center colors, found {len(colors)}")
    separation = sum((colors[0][axis] - colors[1][axis]) ** 2 for axis in range(3)) ** 0.5
    if separation < 45:
        raise OracleError("tile colors have insufficient separation")
    mapping = {colors[0]: 0, colors[1]: 1}
    return [[mapping[color] for color in row] for row in samples]


def _measure(image: Image.Image) -> dict:
    source = _source_from_pixels(image)
    decoded = [[int(row[col] != row[col + 1]) for col in range(RESULT_COLS)] for row in source]
    scores = []
    for label in LABELS:
        template = TEMPLATES[label]
        distance = sum(
            decoded[row][col] != (template[row][col] == "#")
            for row in range(ROWS)
            for col in range(RESULT_COLS)
        )
        scores.append((int(distance), label))
    scores.sort()
    best_distance, decision = scores[0]
    margin = float(scores[1][0] - best_distance)
    if margin <= 0:
        raise OracleError("top arrow templates are tied")
    return {
        "decision": decision,
        "margin": margin,
        "best_distance": best_distance,
        "quarantined": margin <= QUARANTINE_GAP,
        "decoded": decoded,
    }


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    report = _measure(image)
    return str(report["decision"]), float(report["margin"])


def decision_from_image(image: Image.Image) -> str:
    return str(_measure(image)["decision"])
