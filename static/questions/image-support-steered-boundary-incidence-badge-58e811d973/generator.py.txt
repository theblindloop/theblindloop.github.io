"""Deterministic renderer and analytic arm for Two-Region Boundary Badge.

The amber rectangle is a solid tile embedded in a compact teal partition.  It
always touches a white bay above and below.  A thick side corridor either joins
those bays (one incident white component) or is interrupted (two incident
components).  The closed case moves exactly the missing corridor area into a
dead-end bay, keeping teal/white/amber pixel totals independent of the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 384
HEIGHT = 384
BACKGROUND = (250, 249, 245)
DECORATION = (211, 216, 216)
PALETTES = (
    ((18, 126, 133), (235, 148, 32)),
    ((25, 108, 128), (222, 126, 35)),
    ((20, 143, 127), (240, 157, 44)),
)
WHITE = BACKGROUND
MIN_MARGIN = 6


def _nuisance_rng(seed: int, salt: int) -> random.Random:
    return random.Random((int(seed) * 0x9E3779B1 + salt) & 0xFFFFFFFF)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, translated, rotated scene from one integer seed."""

    position_rng = _nuisance_rng(seed, 0xA341316C)
    style_rng = _nuisance_rng(seed, 0xC8013EA4)
    decoration_rng = _nuisance_rng(seed, 0xAD90777D)
    connected_bays = bool(int(seed) & 1)
    badge_size = (88, 96)[style_rng.randrange(2)]
    half = badge_size // 2
    center_x = position_rng.randint(half + 34, WIDTH - half - 34)
    center_y = position_rng.randint(half + 34, HEIGHT - half - 34)
    rotation = 90 * style_rng.randrange(4)
    connector_side = style_rng.randrange(2)
    palette_index = style_rng.randrange(len(PALETTES))
    margin_px = (8, 10, 12)[style_rng.randrange(3)]

    decorations: list[list[int | str]] = []
    target_count = 5 + decoration_rng.randrange(4)
    for _ in range(target_count * 20):
        if len(decorations) >= target_count:
            break
        kind = "disk" if decoration_rng.randrange(2) else "square"
        size = decoration_rng.randint(7, 13)
        x = decoration_rng.randint(18, WIDTH - 18 - size)
        y = decoration_rng.randint(18, HEIGHT - 18 - size)
        # Keep nuisance marks well outside the compact topological badge.
        if abs((x + size / 2) - center_x) < half + 30 and abs(
            (y + size / 2) - center_y
        ) < half + 30:
            continue
        decorations.append([kind, x, y, size])
    if len(decorations) < target_count:
        raise RuntimeError("could not place remote decorations")

    return {
        "connected_bays": connected_bays,
        "center_x": center_x,
        "center_y": center_y,
        "rotation": rotation,
        "connector_side": connector_side,
        "palette_index": palette_index,
        "decorations": decorations,
        "badge_size": badge_size,
        "margin_px": margin_px,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _build_badge(scene: dict) -> Image.Image:
    size = int(scene["badge_size"])
    teal, amber = PALETTES[int(scene["palette_index"])]
    connected = bool(scene["connected_bays"])
    side = int(scene["connector_side"])
    neck = int(scene["margin_px"])

    image = Image.new("RGB", (size, size), teal)
    draw = ImageDraw.Draw(image)
    mid = size // 2
    chamber_left = 18
    chamber_right = size - 19
    upper_top = 12
    upper_bottom = mid - 11
    lower_top = mid + 10
    lower_bottom = size - 13

    # Inclusive Pillow coordinates. The chambers stop immediately above and
    # below the 20-pixel-high amber tile.
    draw.rectangle(
        (chamber_left, upper_top, chamber_right, upper_bottom), fill=WHITE
    )
    draw.rectangle(
        (chamber_left, lower_top, chamber_right, lower_bottom), fill=WHITE
    )

    if side == 0:
        x0, x1 = chamber_left, chamber_left + neck - 1
        compensation_x0 = chamber_right - neck + 1
    else:
        x0, x1 = chamber_right - neck + 1, chamber_right
        compensation_x0 = chamber_left

    corridor_top = upper_bottom + 1
    corridor_bottom = lower_top - 1
    corridor_height = corridor_bottom - corridor_top + 1
    if connected:
        draw.rectangle((x0, corridor_top, x1, corridor_bottom), fill=WHITE)
    else:
        # Two equal stubs do not meet. The missing neck*neck gate area is moved
        # to a top-chamber dead end on the opposite side, exactly balancing ink.
        stub_total = corridor_height - neck
        top_stub = stub_total // 2
        bottom_stub = stub_total - top_stub
        if top_stub:
            draw.rectangle(
                (x0, corridor_top, x1, corridor_top + top_stub - 1), fill=WHITE
            )
        if bottom_stub:
            draw.rectangle(
                (x0, corridor_bottom - bottom_stub + 1, x1, corridor_bottom),
                fill=WHITE,
            )
        draw.rectangle(
            (
                compensation_x0,
                corridor_top,
                compensation_x0 + neck - 1,
                corridor_top + neck - 1,
            ),
            fill=WHITE,
        )

    amber_left = mid - 9
    amber_right = mid + 8
    draw.rectangle(
        (amber_left, corridor_top, amber_right, corridor_bottom), fill=amber
    )

    rotation = int(scene["rotation"])
    if rotation == 90:
        image = image.transpose(Image.Transpose.ROTATE_90)
    elif rotation == 180:
        image = image.transpose(Image.Transpose.ROTATE_180)
    elif rotation == 270:
        image = image.transpose(Image.Transpose.ROTATE_270)
    elif rotation != 0:
        raise ValueError(f"unsupported rotation {rotation}")
    return image


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    canvas = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    for kind, x, y, size in scene["decorations"]:
        box = (int(x), int(y), int(x) + int(size), int(y) + int(size))
        if kind == "disk":
            draw.ellipse(box, fill=DECORATION)
        elif kind == "square":
            draw.rectangle(box, fill=DECORATION)
        else:
            raise ValueError(f"unsupported decoration {kind!r}")

    badge = _build_badge(scene)
    x0 = int(scene["center_x"]) - badge.width // 2
    y0 = int(scene["center_y"]) - badge.height // 2
    canvas.paste(badge, (x0, y0))
    return canvas


def analytic_gold(scene: dict) -> str:
    # Connected bays give one component incident to both tile faces, so the
    # answer to "two different regions?" is no.
    return "no" if bool(scene["connected_bays"]) else "yes"


def margin(scene: dict) -> float:
    """Thickness of the open corridor or the closed separating gate, in pixels."""

    return float(scene["margin_px"])


def is_quarantined(scene: dict) -> bool:
    return int(scene["margin_px"]) < MIN_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # All nuisance values are visibly rendered, so no pixel-identical latent
    # aliases are asserted under the not_applicable declaration.
    return []
