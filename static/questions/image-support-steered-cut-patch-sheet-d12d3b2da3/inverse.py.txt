"""Independent pixel-only inverse for Center in the Cutout.

This module intentionally imports no renderer or scene code.  It segments the
teal sheet, validates that enough clean evidence remains, labels its bounded
complement, computes the sheet's pixel-area centroid, and tests membership.
"""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

_TEALS = np.asarray(((9, 108, 114), (24, 126, 125)), dtype=np.int16)
_EXPECTED_TEAL = 92 * 14 * 14
_COUNT_TOLERANCE = 300
_EXPECTED_DIMS = {(182, 126), (126, 182)}


def _teal_mask(image: Image.Image) -> np.ndarray:
    if image.size != (320, 320):
        raise ValueError(f"unexpected canvas {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    palette = _TEALS.astype(np.int32)
    distances = ((array[:, :, None, :] - palette[None, None, :, :]) ** 2).sum(axis=3)
    return distances.min(axis=2) <= 35 * 35


def _component_sizes(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = [(sy, sx)]
        size = 0
        while queue:
            y, x = queue.pop()
            size += 1
            if y and mask[y - 1, x] and not seen[y - 1, x]:
                seen[y - 1, x] = True
                queue.append((y - 1, x))
            if y + 1 < height and mask[y + 1, x] and not seen[y + 1, x]:
                seen[y + 1, x] = True
                queue.append((y + 1, x))
            if x and mask[y, x - 1] and not seen[y, x - 1]:
                seen[y, x - 1] = True
                queue.append((y, x - 1))
            if x + 1 < width and mask[y, x + 1] and not seen[y, x + 1]:
                seen[y, x + 1] = True
                queue.append((y, x + 1))
        sizes.append(size)
    return sizes


def _bounded_components(free: np.ndarray) -> list[np.ndarray]:
    height, width = free.shape
    seen = np.zeros_like(free, dtype=bool)
    bounded: list[np.ndarray] = []
    for sy, sx in np.argwhere(free):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        seen[sy, sx] = True
        queue = [(sy, sx)]
        points: list[tuple[int, int]] = []
        touches_edge = False
        while queue:
            y, x = queue.pop()
            points.append((y, x))
            touches_edge |= y == 0 or x == 0 or y == height - 1 or x == width - 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and free[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if not touches_edge:
            component = np.zeros_like(free, dtype=bool)
            yy, xx = zip(*points)
            component[np.asarray(yy), np.asarray(xx)] = True
            bounded.append(component)
    return bounded


def decision_from_image(image: Image.Image) -> str:
    teal = _teal_mask(image)
    teal_count = int(teal.sum())
    if abs(teal_count - _EXPECTED_TEAL) > _COUNT_TOLERANCE:
        raise ValueError(f"incomplete teal evidence: {teal_count} pixels")
    ys, xs = np.nonzero(teal)
    if not len(xs):
        raise ValueError("teal sheet is absent")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    dims = (x1 - x0 + 1, y1 - y0 + 1)
    if dims not in _EXPECTED_DIMS:
        raise ValueError(f"unexpected sheet extent {dims}")
    if _component_sizes(teal) != [teal_count]:
        raise ValueError("teal sheet is not one four-connected component")

    # Work in a one-pixel padded crop.  Any non-teal component touching this
    # crop's edge is outside space; the one that does not is the pocket.
    pad = 1
    crop = teal[y0 - pad : y1 + pad + 1, x0 - pad : x1 + pad + 1]
    pockets = _bounded_components(~crop)
    if len(pockets) != 1:
        raise ValueError(f"expected one bounded pocket, found {len(pockets)}")
    pocket = pockets[0]
    pocket_area = int(pocket.sum())
    if abs(pocket_area - 9 * 14 * 14) > 80:
        raise ValueError(f"damaged pocket area {pocket_area}")

    cx, cy = float(xs.mean()), float(ys.mean())
    local_x = int(round(cx)) - (x0 - pad)
    local_y = int(round(cy)) - (y0 - pad)
    if not (0 <= local_y < pocket.shape[0] and 0 <= local_x < pocket.shape[1]):
        raise ValueError("derived centroid left the sheet crop")
    inside = bool(pocket[local_y, local_x])

    py, px = np.nonzero(pocket)
    px0, px1 = float(px.min()), float(px.max())
    py0, py1 = float(py.min()), float(py.max())
    local_cx, local_cy = cx - (x0 - pad), cy - (y0 - pad)
    if inside:
        clearance = min(local_cx - px0, px1 - local_cx, local_cy - py0, py1 - local_cy)
    else:
        dx = max(px0 - local_cx, 0.0, local_cx - px1)
        dy = max(py0 - local_cy, 0.0, local_cy - py1)
        clearance = math.hypot(dx, dy)
    if clearance < 8.0:
        raise ValueError(f"centroid/pocket boundary is ambiguous ({clearance:.2f}px)")
    return "yes" if inside else "no"
