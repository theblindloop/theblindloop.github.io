"""Deterministic latent sampler and raster renderer for Center in the Cutout."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 320
HEIGHT = 320
CELL = 14
BACKGROUND = (250, 249, 245)
PALE = (205, 208, 202)
PALE_EDGE = (187, 191, 184)
PALETTES = (
    ((9, 108, 114), (6, 91, 98)),
    ((24, 126, 125), (15, 103, 103)),
)


def _outer_cells() -> set[tuple[int, int]]:
    # A 9x9 core plus a broad 4x5 wing.  The sheet is compact but its mass
    # centroid is not the center of its 13x9 bounding box.
    cells = {(x, y) for x in range(9) for y in range(9)}
    cells.update((x, y) for x in range(9, 13) for y in range(2, 7))
    return cells


def _base_teal_cells(centroid_in_pocket: bool) -> set[tuple[int, int]]:
    # Both answers have 92 teal cells and one 3x3 pocket.  For yes, the pocket
    # surrounds the area centroid; for no, it is displaced into the wing.
    hole_x = 4 if centroid_in_pocket else 7
    hole = {(x, y) for x in range(hole_x, hole_x + 3) for y in range(3, 6)}
    return _outer_cells() - hole


def _rotate(cells: set[tuple[int, int]], turns: int) -> tuple[set[tuple[int, int]], int, int]:
    width, height = 13, 9
    output = set(cells)
    for _ in range(turns % 4):
        output = {(height - 1 - y, x) for x, y in output}
        width, height = height, width
    return output, width, height


def _teal_cells(scene: dict[str, Any]) -> tuple[set[tuple[int, int]], int, int]:
    return _rotate(
        _base_teal_cells(bool(scene["centroid_in_pocket"])),
        int(scene["rotation"]),
    )


def _make_decorations(rng: random.Random) -> list[dict[str, int | str]]:
    anchors = ((34, 40), (284, 42), (37, 278), (282, 280), (160, 28))
    decorations: list[dict[str, int | str]] = []
    for index, (x, y) in enumerate(anchors):
        decorations.append(
            {
                "x": x + rng.randint(-8, 8),
                "y": y + rng.randint(-8, 8),
                "size": rng.randint(5, 9),
                "kind": "disk" if (rng.randrange(2) + index) % 2 else "square",
            }
        )
    return decorations


def sample_scene(seed: int) -> dict[str, Any]:
    # Adjacent seeds form a counterbalanced pair: answer changes while every
    # nuisance variable remains identical.
    pair = int(seed) // 2
    rng = random.Random(pair ^ 0x5EEDC0DE)
    rotation = rng.randrange(4)
    grid_width, grid_height = ((13, 9) if rotation % 2 == 0 else (9, 13))
    center_x = 160 + rng.randint(-7, 7)
    center_y = 160 + rng.randint(-7, 7)
    return {
        "centroid_in_pocket": bool(int(seed) % 2),
        "origin_x": center_x - (grid_width * CELL) // 2,
        "origin_y": center_y - (grid_height * CELL) // 2,
        "rotation": rotation,
        "palette": rng.randrange(len(PALETTES)),
        "decorations": _make_decorations(rng),
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    for item in scene["decorations"]:
        x, y, size = int(item["x"]), int(item["y"]), int(item["size"])
        box = (x - size, y - size, x + size, y + size)
        if item["kind"] == "disk":
            draw.ellipse(box, fill=PALE, outline=PALE_EDGE, width=1)
        else:
            draw.rectangle(box, fill=PALE, outline=PALE_EDGE, width=1)


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene)
    cells, _, _ = _teal_cells(scene)
    teal, _edge = PALETTES[int(scene["palette"])]
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    for x, y in sorted(cells, key=lambda point: (point[1], point[0])):
        x0, y0 = ox + x * CELL, oy + y * CELL
        draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=teal)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if bool(scene["centroid_in_pocket"]) else "no"


def _centroid_clearance_cells(scene: dict[str, Any]) -> float:
    inside = bool(scene["centroid_in_pocket"])
    cells = _base_teal_cells(inside)
    cx = sum(x + 0.5 for x, _ in cells) / len(cells)
    cy = sum(y + 0.5 for _, y in cells) / len(cells)
    hx0, hx1 = ((4.0, 7.0) if inside else (7.0, 10.0))
    hy0, hy1 = 3.0, 6.0
    if hx0 < cx < hx1 and hy0 < cy < hy1:
        return min(cx - hx0, hx1 - cx, cy - hy0, hy1 - cy)
    dx = max(hx0 - cx, 0.0, cx - hx1)
    dy = max(hy0 - cy, 0.0, cy - hy1)
    return math.hypot(dx, dy)


def margin(scene: dict[str, Any]) -> float:
    """Signed pixel clearance: positive inside the pocket, negative outside."""
    magnitude = _centroid_clearance_cells(scene) * CELL
    return round(magnitude if bool(scene["centroid_in_pocket"]) else -magnitude, 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        cells, grid_width, grid_height = _teal_cells(scene)
        ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
        palette = int(scene["palette"])
    except (KeyError, TypeError, ValueError):
        return True
    return bool(
        len(cells) != 92
        or not 0 <= palette < len(PALETTES)
        or ox < 8
        or oy < 8
        or ox + grid_width * CELL > WIDTH - 8
        or oy + grid_height * CELL > HEIGHT - 8
        or abs(margin(scene)) < 8.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
