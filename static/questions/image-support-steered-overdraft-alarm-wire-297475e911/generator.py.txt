"""Render a short connected event wire for prefix-underflow tracking."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (250, 249, 246)
INK = (34, 42, 52)
CYAN = (24, 174, 194)
CORAL = (232, 91, 79)
GOLD = (243, 190, 38)
STEP = 15
BEAD_RADIUS = 6
EVENTS = 8


def _replay(ops: list[str]) -> str:
    balance = 0
    alarm = False
    for op in ops:
        if op == "deposit":
            balance += 1
        elif balance > 0:
            balance -= 1
        else:
            alarm = True
    return "on" if alarm else "off"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "on" if int(seed) % 2 else "off"
    base = ["deposit"] * 4 + ["withdraw"] * 4
    for _ in range(500):
        ops = base[:]
        rng.shuffle(ops)
        if _replay(ops) == desired:
            break
    else:  # pragma: no cover
        raise RuntimeError("could not sample requested balanced alarm class")
    direction = rng.choice((-1, 1))
    # The complete decisive wire stays in one 64 px row and three adjacent
    # columns of the public 8x8 support probe, while its absolute cell varies.
    column_band = rng.choice((1, 2, 3, 4))
    row_band = rng.choice((1, 2, 3, 4, 5, 6))
    return {
        "ops": ops,
        "x": 64 * column_band + 44,
        "y": 64 * row_band + 32,
        "direction": direction,
    }


def render(scene: dict) -> Image.Image:
    ops = list(scene["ops"])
    x = int(scene["x"])
    y = int(scene["y"])
    direction = int(scene["direction"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    last_x = x + STEP * (len(ops) - 1)

    # A single dark backbone visibly joins the start, all operations, and tip.
    draw.line((x - 17, y, last_x + 17, y), fill=INK, width=5)
    for index, op in enumerate(ops):
        spatial = index if direction == 1 else len(ops) - 1 - index
        cx = x + spatial * STEP
        color = CYAN if op == "deposit" else CORAL
        draw.ellipse(
            (cx - BEAD_RADIUS, y - BEAD_RADIUS, cx + BEAD_RADIUS, y + BEAD_RADIUS),
            fill=color,
            outline=INK,
            width=1,
        )

    if direction == 1:
        start_x = x - 15
        tip_x = last_x + 19
        draw.polygon([(tip_x, y), (tip_x - 10, y - 7), (tip_x - 10, y + 7)], fill=INK)
    else:
        start_x = last_x + 15
        tip_x = x - 19
        draw.polygon([(tip_x, y), (tip_x + 10, y - 7), (tip_x + 10, y + 7)], fill=INK)
    draw.ellipse((start_x - 7, y - 7, start_x + 7, y + 7), fill=GOLD, outline=INK, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return _replay(list(scene["ops"]))


def margin(scene: dict) -> float:
    # A competing parse must erase or recolor at least one bead radius.
    return float(BEAD_RADIUS)


def is_quarantined(scene: dict) -> bool:
    ops = scene.get("ops", [])
    return not (
        isinstance(ops, list)
        and len(ops) == EVENTS
        and ops.count("deposit") == 4
        and ops.count("withdraw") == 4
        and scene.get("direction") in (-1, 1)
        and 70 <= int(scene.get("x", -1)) <= 390
        and 80 <= int(scene.get("y", -1)) <= 432
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
