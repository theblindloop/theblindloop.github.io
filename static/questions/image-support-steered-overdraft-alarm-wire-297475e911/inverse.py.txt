"""Independent image-only inverse for the overdraft alarm wire."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

CYAN = np.array((24, 174, 194), dtype=np.int16)
CORAL = np.array((232, 91, 79), dtype=np.int16)
GOLD = np.array((243, 190, 38), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 7) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[np.ndarray] = []
    height, width = mask.shape
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan = _mask(array, CYAN)
    coral = _mask(array, CORAL)
    event_mask = cyan | coral
    pieces = [part for part in _components(event_mask) if len(part) >= 75]
    if len(pieces) != 8:
        raise ValueError("expected eight intact event beads")

    events = []
    for part in pieces:
        ys, xs = part[:, 0], part[:, 1]
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        if not (11 <= width <= 13 and 11 <= height <= 13 and 90 <= len(part) <= 130):
            raise ValueError("damaged or ambiguous event bead")
        cyan_count = int(cyan[ys, xs].sum())
        coral_count = int(coral[ys, xs].sum())
        if max(cyan_count, coral_count) != len(part) or min(cyan_count, coral_count) != 0:
            raise ValueError("mixed event color")
        events.append((float(np.mean(xs)), float(np.mean(ys)), "deposit" if cyan_count else "withdraw"))

    if sum(op == "deposit" for _, _, op in events) != 4:
        raise ValueError("event totals are not counterbalanced")
    if max(y for _, y, _ in events) - min(y for _, y, _ in events) > 1.0:
        raise ValueError("events do not lie on one thin connected wire")
    events.sort(key=lambda item: item[0])
    gaps = np.diff([x for x, _, _ in events])
    if len(gaps) != 7 or float(gaps.min()) < 13.0 or float(gaps.max()) > 17.0:
        raise ValueError("event order is incomplete or uneven")

    gold_points = np.argwhere(_mask(array, GOLD))
    if not 90 <= len(gold_points) <= 180:
        raise ValueError("round gold start is missing or damaged")
    gold_y = float(np.mean(gold_points[:, 0]))
    gold_x = float(np.mean(gold_points[:, 1]))
    if abs(gold_y - float(np.mean([y for _, y, _ in events]))) > 2.0:
        raise ValueError("start marker is detached from the event wire")
    if gold_x > float(np.mean([x for x, _, _ in events])):
        events.reverse()

    balance = 0
    alarm = False
    for _, _, op in events:
        if op == "deposit":
            balance += 1
        elif balance > 0:
            balance -= 1
        else:
            alarm = True
    return "on" if alarm else "off"
