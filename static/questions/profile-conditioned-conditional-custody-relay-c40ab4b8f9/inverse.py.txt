"""Independent PNG-only inverse for the conditional custody relay."""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = tuple(np.array(color, dtype=np.int16) for color in (
    (84, 178, 226), (231, 112, 157), (104, 190, 128), (173, 126, 218),
))
_TEMPLATES = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}


class PixelParseError(ValueError):
    """The raster does not contain one unambiguous public-contract parse."""


def _near(arr: np.ndarray, color: np.ndarray, tolerance: float = 24.0) -> np.ndarray:
    return np.linalg.norm(arr.astype(np.int16) - color, axis=2) <= tolerance


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if not len(indices):
        return []
    breaks = np.flatnonzero(np.diff(indices) > 1)
    starts = np.r_[indices[0], indices[breaks + 1]]
    ends = np.r_[indices[breaks], indices[-1]]
    return [(int(a), int(b)) for a, b in zip(starts, ends, strict=True)]


def _rect_islands(mask: np.ndarray, minimum_area: int) -> list[tuple[int, int, int, int]]:
    boxes: list[tuple[int, int, int, int]] = []
    for y0, y1 in _runs(mask.any(axis=1)):
        for x0, x1 in _runs(mask[y0:y1 + 1].any(axis=0)):
            island = mask[y0:y1 + 1, x0:x1 + 1]
            area = int(island.sum())
            if area >= minimum_area:
                local_y = np.flatnonzero(island.any(axis=1))
                boxes.append((x0, y0 + int(local_y.min()), x1, y0 + int(local_y.max())))
    return boxes


def _recognize(dark: np.ndarray, box: tuple[int, int, int, int]) -> tuple[str, float]:
    x0, y0, x1, y1 = box
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    hypotheses: list[tuple[int, str, int]] = []
    # The tile supplies its own center. Decode the public 5x7 block alphabet at
    # each allowed cell pitch; no canvas coordinate or scene value is used.
    for cell_size in (2, 3, 4):
        pitch = cell_size + 1
        width, height = 5 * pitch - 1, 7 * pitch - 1
        gx0, gy0 = cx - width / 2, cy - height / 2
        observed = np.zeros((7, 5), dtype=bool)
        for row in range(7):
            for col in range(5):
                px = int(round(gx0 + col * pitch + cell_size / 2))
                py = int(round(gy0 + row * pitch + cell_size / 2))
                observed[row, col] = bool(dark[py, px])
        for label, rows in _TEMPLATES.items():
            expected = np.array([[bit == "1" for bit in row] for row in rows])
            hypotheses.append((int(np.count_nonzero(observed != expected)), label, cell_size))
    hypotheses.sort(key=lambda item: (item[0], item[1], item[2]))
    best_score, best_label, best_scale = hypotheses[0]
    wrong = next(item for item in hypotheses if item[1] != best_label)
    ranking = [(best_label, best_score), (wrong[1], wrong[0])]
    gap = wrong[0] - best_score
    if best_score > 3 or gap < 2:
        raise PixelParseError(f"ambiguous station glyph scores {ranking} at scale {best_scale}")
    return best_label, float(gap)


def _orange(arr: np.ndarray) -> np.ndarray:
    return ((arr[..., 0] > 190) & (arr[..., 1] > 78) &
            (arr[..., 1] < 170) & (arr[..., 2] < 90))


def _decode(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[1] < 500 or arr.shape[0] < 180:
        raise PixelParseError("raster is too small for the custody ledger")
    tile_mask = np.zeros(arr.shape[:2], dtype=bool)
    for color in _PALETTE:
        tile_mask |= _near(arr, color)
    tiles = _rect_islands(tile_mask, minimum_area=900)
    tiles = [box for box in tiles if 35 <= box[2] - box[0] + 1 <= 65 and
             35 <= box[3] - box[1] + 1 <= 65]
    if len(tiles) != 13:
        raise PixelParseError(f"expected thirteen station tiles, found {len(tiles)}")
    tiles.sort(key=lambda box: (box[0] + box[2]) / 2)
    dark = np.all(arr < 92, axis=2)
    parsed = [_recognize(dark, box) for box in tiles]
    labels = [item[0] for item in parsed]
    margins = [item[1] for item in parsed]
    holder = labels[0]
    orange = _orange(arr)
    for index in range(6):
        left_box = tiles[1 + 2 * index]
        right_box = tiles[2 + 2 * index]
        if left_box[2] >= right_box[0]:
            raise PixelParseError("transfer endpoints overlap")
        xa, xb = left_box[2] + 1, right_box[0]
        ya = max(0, min(left_box[1], right_box[1]) - 12)
        yb = min(arr.shape[0], max(left_box[3], right_box[3]) + 13)
        arrow_ink = int(orange[ya:yb, xa:xb].sum())
        if arrow_ink < 70:
            raise PixelParseError(f"transfer card {index + 1} has no intact direction arrow")
        sender, receiver = labels[1 + 2 * index:3 + 2 * index]
        if sender == receiver:
            raise PixelParseError(f"transfer card {index + 1} repeats one endpoint")
        if holder == sender:
            holder = receiver
    return holder, float(min(margins))


def decision_from_image(image: Image.Image) -> str:
    return _decode(image)[0]


def decision_margin_from_image(image: Image.Image) -> float:
    """Cell-level Hamming gap between the winning glyph parse and runner-up."""
    return _decode(image)[1]
