"""Deterministic sampler and raster renderer for conditional parcel custody."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1280, 330
LABELS = ("A", "B", "C", "D")
PALETTE = (
    (84, 178, 226),
    (231, 112, 157),
    (104, 190, 128),
    (173, 126, 218),
)
GLYPHS = {
    "A": ("01110", "10001", "10001", "11111", "10001", "10001", "10001"),
    "B": ("11110", "10001", "10001", "11110", "10001", "10001", "11110"),
    "C": ("01111", "10000", "10000", "10000", "10000", "10000", "01111"),
    "D": ("11110", "10001", "10001", "10001", "10001", "10001", "11110"),
}
STATUS_PATTERNS = (
    "EIEIEI", "IEIEIE", "EEIIEI", "IEEIEI", "EIIEIE", "IIEEIE",
)


def _replay(initial_holder: str, transfers: list[list[str]]) -> str:
    holder = str(initial_holder)
    for sender, receiver in transfers:
        if holder == sender:
            holder = receiver
    return holder


def _make_base_program(rng: random.Random, pattern: str) -> tuple[int, list[tuple[int, int]]]:
    """Build a canonical program ending at station 0 with declared E/I events."""
    initial = rng.choice((1, 2, 3))
    holder = initial
    transfers: list[tuple[int, int]] = []
    effective_left = pattern.count("E")
    for status in pattern:
        if status == "E":
            effective_left -= 1
            sender = holder
            if effective_left == 0:
                receiver = 0
            else:
                receiver = rng.choice([value for value in range(1, 4) if value != holder])
            holder = receiver
        else:
            sender = rng.choice([value for value in range(4) if value != holder])
            receiver = rng.choice([value for value in range(4) if value != sender])
        transfers.append((sender, receiver))
    if holder != 0:
        raise AssertionError("program construction failed")
    return initial, transfers


def sample_scene(seed: int) -> dict:
    """Return quartets balanced across A-D with identical visual nuisance settings."""
    seed = int(seed)
    target_shift = seed % 4
    group = seed // 4
    program_rng = random.Random(0xC0570D1 + group * 7919)
    style_rng = random.Random(0x51A7E + group * 1543)
    pattern = STATUS_PATTERNS[group % len(STATUS_PATTERNS)]
    initial, base_transfers = _make_base_program(program_rng, pattern)

    def shifted(value: int) -> str:
        return LABELS[(value + target_shift) % 4]

    colors = list(PALETTE)
    style_rng.shuffle(colors)
    scene = {
        "initial_holder": shifted(initial),
        "transfers": [[shifted(a), shifted(b)] for a, b in base_transfers],
        "label_colors": [list(color) for color in colors],
        "glyph_scale": style_rng.choice((2, 3, 4)),
        "card_jitter": [style_rng.randint(-4, 4) for _ in range(7)],
    }
    if analytic_gold(scene) != LABELS[target_shift]:
        raise AssertionError("counterbalancing failed")
    return scene


def analytic_gold(scene: dict) -> str:
    return _replay(scene["initial_holder"], scene["transfers"])


def _hamming(left: str, right: str) -> int:
    return sum(a != b for row_a, row_b in zip(GLYPHS[left], GLYPHS[right], strict=True)
               for a, b in zip(row_a, row_b, strict=True))


def margin(scene: dict) -> float:
    """Ideal pixel disagreement gap to the nearest wrong glyph parse."""
    used = [scene["initial_holder"]]
    used.extend(label for pair in scene["transfers"] for label in pair)
    scale = int(scene["glyph_scale"])
    runner_up = min(_hamming(label, other) for label in used for other in LABELS if other != label)
    return float(runner_up * scale * scale)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _scaled_box(box: tuple[float, float, float, float], supersample: int) -> tuple[int, int, int, int]:
    return tuple(round(value * supersample) for value in box)  # type: ignore[return-value]


def _glyph(draw: ImageDraw.ImageDraw, label: str, center: tuple[float, float],
           cell: int, supersample: int) -> None:
    rows = GLYPHS[label]
    pitch = cell + 1
    width, height = 5 * pitch - 1, 7 * pitch - 1
    x0, y0 = center[0] - width / 2, center[1] - height / 2
    for row, bits in enumerate(rows):
        for col, bit in enumerate(bits):
            if bit == "1":
                x = x0 + col * pitch
                y = y0 + row * pitch
                draw.rectangle(_scaled_box((x, y, x + cell, y + cell), supersample), fill=(18, 25, 32))


def _tile(draw: ImageDraw.ImageDraw, label: str, center: tuple[float, float],
          color: tuple[int, int, int], glyph_scale: int, supersample: int) -> None:
    cx, cy = center
    draw.rounded_rectangle(
        _scaled_box((cx - 27, cy - 27, cx + 27, cy + 27), supersample),
        radius=8 * supersample, fill=color, outline=(31, 39, 48), width=3 * supersample,
    )
    _glyph(draw, label, center, glyph_scale, supersample)


def render(scene: dict) -> Image.Image:
    initial = str(scene["initial_holder"])
    transfers = [list(pair) for pair in scene["transfers"]]
    colors = [tuple(int(v) for v in color) for color in scene["label_colors"]]
    glyph_scale = int(scene["glyph_scale"])
    jitter = [int(value) for value in scene["card_jitter"]]
    if initial not in LABELS or len(transfers) != 6:
        raise ValueError("scene must show one start holder and six transfers")
    if any(len(pair) != 2 or pair[0] not in LABELS or pair[1] not in LABELS or pair[0] == pair[1]
           for pair in transfers):
        raise ValueError("invalid transfer card")
    if len(colors) != 4 or len(set(colors)) != 4 or any(len(color) != 3 for color in colors):
        raise ValueError("invalid label palette")
    if glyph_scale not in (2, 3, 4) or len(jitter) != 7 or any(abs(v) > 4 for v in jitter):
        raise ValueError("invalid visible style")

    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), (248, 248, 245))
    draw = ImageDraw.Draw(image)
    dark = (30, 38, 47)
    card_fill = (237, 243, 248)
    arrow = (238, 126, 34)
    draw.text((24 * ss, 14 * ss), "PARCEL CUSTODY RELAY", font=_font(25 * ss, True), fill=dark)
    draw.text((24 * ss, 49 * ss), "Read cards left to right. Transfer only if sender holds the parcel.",
              font=_font(15 * ss), fill=(68, 77, 87))

    label_color = {label: colors[index] for index, label in enumerate(LABELS)}
    top, bottom = 88, 278
    draw.rounded_rectangle(_scaled_box((20, top, 140, bottom), ss), radius=10 * ss,
                           fill=(250, 246, 223), outline=(61, 72, 84), width=3 * ss)
    draw.text((48 * ss, 104 * ss), "START", font=_font(16 * ss, True), fill=dark)
    _tile(draw, initial, (80, 181 + jitter[0]), label_color[initial], glyph_scale, ss)
    draw.rectangle(_scaled_box((54, 240, 106, 257), ss), fill=(255, 255, 255),
                   outline=(61, 72, 84), width=2 * ss)
    draw.line([(60 * ss, 248 * ss), (100 * ss, 248 * ss)], fill=(61, 72, 84), width=2 * ss)

    for index, (sender, receiver) in enumerate(transfers):
        x0 = 154 + index * 184
        x1 = x0 + 174
        draw.rounded_rectangle(_scaled_box((x0, top, x1, bottom), ss), radius=10 * ss,
                               fill=card_fill, outline=(61, 72, 84), width=3 * ss)
        draw.text(((x0 + 12) * ss, 104 * ss), str(index + 1), font=_font(15 * ss, True), fill=dark)
        cy = 181 + jitter[index + 1]
        left_center, right_center = (x0 + 43, cy), (x0 + 131, cy)
        _tile(draw, sender, left_center, label_color[sender], glyph_scale, ss)
        _tile(draw, receiver, right_center, label_color[receiver], glyph_scale, ss)
        draw.line([((x0 + 74) * ss, cy * ss), ((x0 + 100) * ss, cy * ss)],
                  fill=arrow, width=6 * ss)
        draw.polygon([((x0 + 104) * ss, cy * ss), ((x0 + 94) * ss, (cy - 9) * ss),
                      ((x0 + 94) * ss, (cy + 9) * ss)], fill=arrow)
        draw.text(((x0 + 49) * ss, 243 * ss), "SEND", font=_font(11 * ss, True), fill=(78, 87, 98))

    draw.text((24 * ss, 298 * ss), "LEFT  ->  RIGHT     |     SENDER MISMATCH = NO MOVE",
              font=_font(14 * ss, True), fill=(68, 77, 87))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
