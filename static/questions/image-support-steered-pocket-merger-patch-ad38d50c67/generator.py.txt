"""Deterministic latent sampler and renderer for Pocket-Merger Patch."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
LOCAL_SIZE = 96
CLEARANCE = 8

PALE = (222, 226, 230)
PALE_2 = (232, 229, 223)
PALETTES = (
    ((18, 104, 109), (232, 151, 43)),
    ((31, 91, 124), (221, 129, 36)),
    ((25, 119, 94), (239, 164, 54)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    decorations = [
        [18, 18, 34, 28, 0],
        [478, 22, 494, 34, 1],
        [20, 477, 37, 491, 1],
        [474, 474, 494, 487, 0],
    ]
    rng.shuffle(decorations)
    return {
        "merges": seed % 2 == 0,
        "cell_x": rng.randint(28, WIDTH - LOCAL_SIZE - 28),
        "cell_y": rng.randint(28, HEIGHT - LOCAL_SIZE - 28),
        "rotation": rng.randrange(4),
        "palette_variant": rng.randrange(len(PALETTES)),
        "decorations": decorations,
        "clearance": CLEARANCE,
    }


def _local_image(scene: dict) -> Image.Image:
    teal, amber = PALETTES[int(scene["palette_variant"])]
    local = Image.new("RGB", (LOCAL_SIZE, LOCAL_SIZE), teal)
    draw = ImageDraw.Draw(local)
    # Both classes have two equal-area pockets and an identical 32x32 patch.
    # In the merger class the right pocket touches the patch. In the other
    # class a 12-pixel teal wall remains between them.
    draw.rectangle((6, 20, 29, 75), fill="white")
    right_x = 54 if bool(scene["merges"]) else 64
    draw.rectangle((right_x, 20, right_x + 23, 75), fill="white")
    draw.rectangle((30, 36, 53, 59), fill=amber)
    rotation = int(scene["rotation"]) % 4
    for _ in range(rotation):
        local = local.transpose(Image.Transpose.ROTATE_90)
    return local


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), "white")
    draw = ImageDraw.Draw(image)
    for x0, y0, x1, y1, tone in scene["decorations"]:
        draw.rectangle((x0, y0, x1, y1), fill=PALE if tone == 0 else PALE_2)
    local = _local_image(scene)
    image.paste(local, (int(scene["cell_x"]), int(scene["cell_y"])))
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["merges"] else "no"


def margin(scene: dict) -> float:
    return float(scene["clearance"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
