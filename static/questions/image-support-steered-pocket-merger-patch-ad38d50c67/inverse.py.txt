"""Independent image-only inverse arm for the patch-erasure topology."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The raster does not satisfy the observable world boundary."""


def _components(mask: np.ndarray) -> list[tuple[int, bool]]:
    """Return (area, touches_border) for four-connected true components."""
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out: list[tuple[int, bool]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        area = 0
        border = False
        while queue:
            cy, cx = queue.popleft()
            area += 1
            border |= cy == 0 or cx == 0 or cy == h - 1 or cx == w - 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        out.append((area, border))
    return out


def _solid_masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    mx = arr.max(axis=2).astype(np.int16)
    mn = arr.min(axis=2).astype(np.int16)
    saturated = (mx - mn) * 100 >= mx * 30
    colors, counts = np.unique(arr[saturated], axis=0, return_counts=True)
    candidates: list[tuple[int, tuple[int, int, int]]] = []
    for color, count in zip(colors, counts):
        c = color.astype(float) / 255.0
        saturation = (c.max() - c.min()) / max(c.max(), 1e-9)
        if saturation >= 0.30 and count >= 400:
            candidates.append((int(count), tuple(int(v) for v in color)))
    if len(candidates) != 2:
        raise OracleAbstention(f"expected two substantial saturated inks, found {len(candidates)}")
    candidates.sort(reverse=True)
    teal_color = np.array(candidates[0][1], dtype=np.uint8)
    amber_color = np.array(candidates[1][1], dtype=np.uint8)
    teal = np.all(arr == teal_color, axis=2)
    amber = np.all(arr == amber_color, axis=2)
    amber_area = int(amber.sum())
    if not 500 <= amber_area <= 650:
        raise OracleAbstention(f"amber patch area {amber_area} is outside the observable band")
    amber_components = _components(amber)
    if len(amber_components) != 1 or amber_components[0][1]:
        raise OracleAbstention("amber patch must be one interior component")
    return teal, amber


def _bounded_count(solid: np.ndarray) -> int:
    components = _components(~solid)
    return sum(not touches_border and area >= 200 for area, touches_border in components)


def decision_from_image(image: Image.Image) -> str:
    """Erase the smaller saturated patch and compare bounded components."""
    teal, amber = _solid_masks(image)
    solid = teal | amber
    ys, xs = np.nonzero(solid)
    if not len(xs):
        raise OracleAbstention("solid emblem is absent")
    y0, y1 = max(0, int(ys.min()) - 1), min(solid.shape[0], int(ys.max()) + 2)
    x0, x1 = max(0, int(xs.min()) - 1), min(solid.shape[1], int(xs.max()) + 2)
    solid_crop = solid[y0:y1, x0:x1]
    teal_crop = teal[y0:y1, x0:x1]
    before = _bounded_count(solid_crop)
    after = _bounded_count(teal_crop)
    if before != 2:
        raise OracleAbstention(f"expected exactly two initial pockets, found {before}")
    if after not in (1, 2):
        raise OracleAbstention(f"unsupported post-erasure pocket count {after}")
    return "yes" if after == 1 else "no"
