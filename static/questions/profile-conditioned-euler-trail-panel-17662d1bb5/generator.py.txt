"""Deterministic renderer for the one-stroke network question."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 360
PANEL_X = (20, 330, 640)
PANEL_Y = 48
BASE_POSITIONS = ((45, 54), (143, 36), (240, 62), (244, 188), (148, 236), (42, 196))
ALL_EDGES = tuple(itertools.combinations(range(6), 2))


def _connected(edges: Iterable[Iterable[int]]) -> bool:
    adjacency = [set() for _ in range(6)]
    for a, b in edges:
        adjacency[a].add(b)
        adjacency[b].add(a)
    seen = {0}
    pending = [0]
    while pending:
        pending.extend(adjacency[pending.pop()] - seen)
        seen.update(pending)
    return len(seen) == 6


def _odd_count(edges: Iterable[Iterable[int]]) -> int:
    degree = [0] * 6
    for a, b in edges:
        degree[a] += 1
        degree[b] += 1
    return sum(value % 2 for value in degree)


def _sample_edges(rng: random.Random, wanted_odd: int, positions: list[list[int]]) -> list[list[int]]:
    allowed = []
    for a, b in ALL_EDGES:
        px, py = positions[a]
        qx, qy = positions[b]
        length2 = (qx - px) ** 2 + (qy - py) ** 2
        clear = True
        for c, (rx, ry) in enumerate(positions):
            if c in (a, b):
                continue
            along = ((rx - px) * (qx - px) + (ry - py) * (qy - py)) / length2
            if 0.08 < along < 0.92:
                sx, sy = px + along * (qx - px), py + along * (qy - py)
                if ((rx - sx) ** 2 + (ry - sy) ** 2) ** 0.5 < 28:
                    clear = False
                    break
        if clear:
            allowed.append((a, b))
    for _ in range(10000):
        edges = sorted(rng.sample(allowed, 7))
        if _connected(edges) and _odd_count(edges) == wanted_odd:
            return [list(edge) for edge in edges]
    raise RuntimeError("could not sample a graph with the requested parity")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    targets = [2, 4, 6]
    rng.shuffle(targets)
    panels = []
    for target in targets:
        positions = [
            [x + rng.randint(-8, 8), y + rng.randint(-8, 8)] for x, y in BASE_POSITIONS
        ]
        panels.append({"positions": positions, "edges": _sample_edges(rng, target, positions)})
    return {"panels": panels}


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    winners = [index for index, panel in enumerate(panels) if _odd_count(panel["edges"]) in (0, 2)]
    if len(winners) != 1:
        raise ValueError(f"expected one Euler-trail panel, found {winners}")
    return ("left", "center", "right")[winners[0]]


def margin(scene: dict) -> float:
    counts = [_odd_count(panel["edges"]) for panel in scene["panels"]]
    losers = [count for count in counts if count not in (0, 2)]
    return float(min(abs(count - 2) for count in losers)) if losers else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        panels = scene["panels"]
        if len(panels) != 3:
            return True
        counts = []
        for panel in panels:
            positions, edges = panel["positions"], panel["edges"]
            normalized = {tuple(sorted(edge)) for edge in edges}
            if len(positions) != 6 or len(edges) != 7 or len(normalized) != 7:
                return True
            if any(a == b or not (0 <= a < 6 and 0 <= b < 6) for a, b in normalized):
                return True
            if not _connected(normalized):
                return True
            for a, b in itertools.combinations(range(6), 2):
                dx = positions[a][0] - positions[b][0]
                dy = positions[a][1] - positions[b][1]
                if dx * dx + dy * dy < 42 * 42:
                    return True
            counts.append(_odd_count(normalized))
        return sorted(counts) != [2, 4, 6]
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), "white")
    draw = ImageDraw.Draw(image)
    labels = ("LEFT", "CENTER", "RIGHT")

    for index, panel in enumerate(panels):
        ox, oy = PANEL_X[index], PANEL_Y
        box = (ox, 22, ox + 300, 326)
        box = tuple(value * scale for value in box)
        draw.rounded_rectangle(box, radius=12 * scale, fill="#f8fafc", outline="#3269b4", width=3 * scale)
        points = [((ox + x) * scale, (oy + y) * scale) for x, y in panel["positions"]]
        for a, b in panel["edges"]:
            draw.line((points[a], points[b]), fill="#484e54", width=9 * scale)
        radius = 15 * scale
        for x, y in points:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill="#20b9d0", outline="#111820", width=4 * scale)
        font = _font(16 * scale)
        label = labels[index]
        label_box = draw.textbbox((0, 0), label, font=font)
        text_width = label_box[2] - label_box[0]
        draw.text(((ox + 150) * scale - text_width / 2, 302 * scale), label, font=font, fill="#24548f")

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
