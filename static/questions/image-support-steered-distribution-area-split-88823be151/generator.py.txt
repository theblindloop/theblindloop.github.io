"""Deterministic renderer and analytic arm for a filled distribution silhouette."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 384, 320
BACKGROUND = (248, 248, 244)
DIVIDER = (54, 61, 68)
PALETTES = ((22, 151, 154), (30, 137, 170), (27, 158, 126), (42, 133, 166))
QUARANTINE_MARGIN = 1200.0


def sample_scene(seed: int) -> dict:
    """Sample balanced strong and near-boundary silhouettes.

    Modes repeat every six seeds: two strong left, two strong right, and one
    quarantined example of each class. Palette and small translations vary
    independently of the answer.
    """

    rng = random.Random(int(seed))
    mode = int(seed) % 6
    more_on_left = mode in (0, 2, 4)
    strength = 4 if mode in (2, 3) else 20
    x0 = 32 + rng.randint(-2, 2)
    x_coords = [x0 + 20 * index for index in range(17)]

    # Mirror only the neutral carrier, then add the label-bearing thickness.
    # This makes total ink and palette balanced while keeping the integral clear.
    top_left = [43 + rng.randint(-4, 4) for _ in range(9)]
    bottom_left = [277 + rng.randint(-4, 4) for _ in range(9)]
    top_left[0] = 38
    bottom_left[0] = 282
    top_profile = top_left + list(reversed(top_left[:-1]))
    bottom_profile = bottom_left + list(reversed(bottom_left[:-1]))

    indices: Iterable[int] = range(0, 8) if more_on_left else range(9, 17)
    upper_gain = strength // 2
    lower_gain = strength - upper_gain
    for index in indices:
        top_profile[index] -= upper_gain
        bottom_profile[index] += lower_gain

    return {
        "x_coords": x_coords,
        "divider_x": x_coords[8],
        "top_profile": top_profile,
        "bottom_profile": bottom_profile,
        "palette_index": rng.randrange(len(PALETTES)),
    }


def _polygon(scene: dict) -> list[tuple[int, int]]:
    xs = [int(value) for value in scene["x_coords"]]
    top = [int(value) for value in scene["top_profile"]]
    bottom = [int(value) for value in scene["bottom_profile"]]
    # Accessing divider_x here makes the rendered-field audit explicit even
    # though it is also drawn separately below.
    divider = int(scene["divider_x"])
    if len(xs) != 17 or len(top) != 17 or len(bottom) != 17 or divider != xs[8]:
        raise ValueError("scene profiles must have 17 aligned samples")
    if any(b - t < 180 for t, b in zip(top, bottom, strict=True)):
        raise ValueError("silhouette is too thin")
    return list(zip(xs, top, strict=True)) + list(
        reversed(list(zip(xs, bottom, strict=True)))
    )


def _side_areas(scene: dict) -> tuple[float, float]:
    xs = [float(value) for value in scene["x_coords"]]
    divider = float(scene["divider_x"])
    top = [float(value) for value in scene["top_profile"]]
    bottom = [float(value) for value in scene["bottom_profile"]]
    thickness = [b - t for t, b in zip(top, bottom, strict=True)]
    left = right = 0.0
    for index in range(len(xs) - 1):
        x_a, x_b = xs[index], xs[index + 1]
        if not x_a < x_b:
            raise ValueError("x coordinates must increase")
        area = (x_b - x_a) * (thickness[index] + thickness[index + 1]) / 2.0
        if x_b <= divider:
            left += area
        elif x_a >= divider:
            right += area
        else:
            raise ValueError("divider must coincide with a profile sample")
    return left, right


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    polygon = _polygon(scene)
    palette_index = int(scene["palette_index"])
    color = PALETTES[palette_index]
    draw.polygon(polygon, fill=color)

    divider = int(scene["divider_x"])
    # A dark dashed reference is painted over the area. Gaps between dashes
    # preserve the single connected teal component.
    for y in range(10, HEIGHT - 10, 16):
        draw.rectangle((divider - 1, y, divider + 1, min(y + 8, HEIGHT - 11)), fill=DIVIDER)
    return image


def analytic_gold(scene: dict) -> str:
    left, right = _side_areas(scene)
    return "left" if left > right else "right"


def margin(scene: dict) -> float:
    left, right = _side_areas(scene)
    return abs(left - right)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
