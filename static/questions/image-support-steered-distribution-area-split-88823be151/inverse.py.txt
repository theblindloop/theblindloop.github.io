"""Independent pixel-only inverse arm for the distribution silhouette."""

from __future__ import annotations

import numpy as np
from PIL import Image

REGION_COLORS = np.asarray(
    ((22, 151, 154), (30, 137, 170), (27, 158, 126), (42, 133, 166)),
    dtype=np.int16,
)
DIVIDER_COLOR = np.asarray((54, 61, 68), dtype=np.int16)


class PixelOracleError(ValueError):
    """The raster no longer contains a measurable clean silhouette."""


def _mask_for_colors(array: np.ndarray, colors: np.ndarray, tolerance: int = 3) -> np.ndarray:
    delta = np.abs(array.astype(np.int16)[:, :, None, :] - colors[None, None, :, :])
    return np.any(np.max(delta, axis=3) <= tolerance, axis=2)


def _validate_silhouette(mask: np.ndarray, divider_x: int) -> None:
    ys, xs = np.nonzero(mask)
    if len(xs) < 50000:
        raise PixelOracleError("too little colored area")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if x1 - x0 + 1 < 315 or y1 - y0 + 1 < 225:
        raise PixelOracleError("silhouette extent is truncated")

    # Every clean polygon column is one interval. An erased interior tile makes
    # many columns split into separate runs, so the inverse arm abstains instead
    # of silently manufacturing an area measurement from corrupted evidence.
    for x in range(x0, x1 + 1):
        if abs(x - divider_x) <= 2:
            continue
        column = np.flatnonzero(mask[:, x])
        if len(column) == 0:
            raise PixelOracleError("missing silhouette column")
        if int(np.max(np.diff(column), initial=1)) > 1:
            raise PixelOracleError("silhouette has an interior erasure")


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    region = _mask_for_colors(array, REGION_COLORS)
    divider = _mask_for_colors(array, DIVIDER_COLOR[None, :], tolerance=3)
    column_counts = divider.sum(axis=0)
    if int(column_counts.max(initial=0)) < 25:
        raise PixelOracleError("visible dashed divider is absent")
    divider_x = int(np.argmax(column_counts)) + 1  # center of the three-pixel stroke
    _validate_silhouette(region, divider_x)

    left_area = int(region[:, :divider_x].sum())
    right_area = int(region[:, divider_x + 1 :].sum())
    gap = left_area - right_area
    if abs(gap) < 150:
        raise PixelOracleError("rasterized area comparison is ambiguous")
    return "left" if gap > 0 else "right"
