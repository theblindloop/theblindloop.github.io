"""Deterministic Pillow renderer with analytic labels from latent scene fields."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

W, H = 512, 384
RED = (220, 48, 55)
BLUE = (38, 105, 210)
INK = (28, 35, 45)
BODY = (244, 201, 65)
LIGHT = (249, 250, 252)
GRAY = (145, 154, 168)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C9)
    return {
        "positive_side": "left" if seed % 2 else "right",
        "left_wire_color": "red" if (seed // 2) % 2 else "blue",
        "offset_x": rng.randint(-18, 18),
        "offset_y": rng.randint(-10, 10),
        "scale": [0.90, 1.00, 1.10][rng.randrange(3)],
        "route_variant": rng.randrange(3),
        "screw_variant": rng.randrange(3),
    }


def _pt(cx: int, cy: int, scale: float, ox: int, oy: int) -> tuple[int, int]:
    return (round(cx + ox), round(cy + oy))


def render(scene: dict) -> Image.Image:
    # Direct accesses are intentional: the public rendered-field audit can see
    # that every declared latent field contributes to final pixels.
    positive_side = scene["positive_side"]
    left_wire_color = scene["left_wire_color"]
    ox = int(scene["offset_x"])
    oy = int(scene["offset_y"])
    scale = float(scene["scale"])
    route_variant = int(scene["route_variant"])
    screw_variant = int(scene["screw_variant"])

    image = Image.new("RGB", (W, H), LIGHT)
    draw = ImageDraw.Draw(image)
    cx, cy = 256 + ox, 205 + oy
    half_w, half_h = round(145 * scale), round(82 * scale)
    body = (cx - half_w, cy - half_h, cx + half_w, cy + half_h)
    draw.rounded_rectangle(body, radius=round(18 * scale), fill=BODY, outline=INK, width=round(6 * scale))

    terminal_dx = round(72 * scale)
    terminal_y = cy - half_h
    left_x, right_x = cx - terminal_dx, cx + terminal_dx
    wire_width = max(10, round(13 * scale))
    left_rgb = RED if left_wire_color == "red" else BLUE
    right_rgb = BLUE if left_wire_color == "red" else RED

    top_y = round(42 + oy / 2)
    bend = round((route_variant - 1) * 14 * scale)
    for x, rgb, direction in ((left_x, left_rgb, -1), (right_x, right_rgb, 1)):
        mid_y = terminal_y - round(55 * scale)
        points = [(x + direction * bend, top_y), (x + direction * bend, mid_y), (x, mid_y), (x, terminal_y - round(19 * scale))]
        draw.line(points, fill=rgb, width=wire_width, joint="curve")

    pad_r = round(25 * scale)
    for x in (left_x, right_x):
        draw.ellipse((x - pad_r, terminal_y - pad_r, x + pad_r, terminal_y + pad_r), fill=(255, 255, 255), outline=INK, width=round(5 * scale))

    plus_x = left_x if positive_side == "left" else right_x
    minus_x = right_x if positive_side == "left" else left_x
    mark_half = round(13 * scale)
    mark_w = max(5, round(6 * scale))
    draw.line((plus_x - mark_half, terminal_y, plus_x + mark_half, terminal_y), fill=INK, width=mark_w)
    draw.line((plus_x, terminal_y - mark_half, plus_x, terminal_y + mark_half), fill=INK, width=mark_w)
    draw.line((minus_x - mark_half, terminal_y, minus_x + mark_half, terminal_y), fill=INK, width=mark_w)

    # Gray screw heads are harmless distractors; their orientation is balanced
    # and their color is deliberately outside all oracle masks.
    screw_y = cy + round(44 * scale)
    for index, sx in enumerate((cx - round(85 * scale), cx + round(85 * scale))):
        sr = round(9 * scale)
        draw.ellipse((sx - sr, screw_y - sr, sx + sr, screw_y + sr), fill=(225, 228, 233), outline=GRAY, width=3)
        if (screw_variant + index) % 2:
            draw.line((sx - sr + 3, screw_y, sx + sr - 3, screw_y), fill=GRAY, width=3)
        else:
            draw.line((sx, screw_y - sr + 3, sx, screw_y + sr - 3), fill=GRAY, width=3)

    # A neutral centered polarity divider reinforces that the two terminals are
    # peers without revealing the answer.
    draw.line((cx, cy - round(20 * scale), cx, cy + round(18 * scale)), fill=(185, 145, 40), width=3)
    return image


def analytic_gold(scene: dict) -> str:
    positive_side = scene["positive_side"]
    left_wire_color = scene["left_wire_color"]
    if positive_side == "left":
        return left_wire_color
    return "blue" if left_wire_color == "red" else "red"


def margin(scene: dict) -> float:
    # Visible terminal separation is the runner-up binding margin in pixels.
    return float(round(144 * float(scene["scale"]), 6))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 110.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
