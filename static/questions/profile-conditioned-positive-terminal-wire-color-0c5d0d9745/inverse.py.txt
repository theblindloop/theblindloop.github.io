"""Independent, scene-free inverse arm operating only on RGB pixels."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    out = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        out.append({"area": len(xs), "x0": min(xs), "x1": max(xs), "y0": min(ys), "y1": max(ys), "cx": sum(xs) / len(xs), "cy": sum(ys) / len(ys)})
    return out


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (rgb[:, :, 0] > 170) & (rgb[:, :, 1] < 105) & (rgb[:, :, 2] < 115)
    blue = (rgb[:, :, 2] > 155) & (rgb[:, :, 0] < 105) & (rgb[:, :, 1] < 160)
    black = (rgb[:, :, 0] < 70) & (rgb[:, :, 1] < 75) & (rgb[:, :, 2] < 85)

    color_centers = {}
    for name, mask in (("red", red), ("blue", blue)):
        comps = [c for c in _components(mask) if c["area"] > 180]
        if len(comps) != 1:
            raise ValueError(f"expected one {name} wire component, found {len(comps)}")
        color_centers[name] = comps[0]["cx"]

    marks = []
    for comp in _components(black):
        bw = comp["x1"] - comp["x0"] + 1
        bh = comp["y1"] - comp["y0"] + 1
        if 35 <= comp["area"] <= 800 and bw <= 45 and bh <= 45 and 70 < comp["cy"] < 210:
            marks.append((comp, bw, bh))
    pluses = [item for item in marks if 0.72 <= item[1] / item[2] <= 1.38 and item[2] >= 18]
    minuses = [item for item in marks if item[1] / item[2] >= 2.2 and item[2] <= 12]
    if len(pluses) != 1 or len(minuses) != 1:
        raise ValueError(f"ambiguous polarity marks: plus={len(pluses)} minus={len(minuses)}")
    plus_x = pluses[0][0]["cx"]
    distances = {name: abs(cx - plus_x) for name, cx in color_centers.items()}
    ranked = sorted(distances, key=distances.get)
    if distances[ranked[1]] - distances[ranked[0]] < 70:
        raise ValueError("wire-to-plus binding margin is too small")
    return ranked[0]
