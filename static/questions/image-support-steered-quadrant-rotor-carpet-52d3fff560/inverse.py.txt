"""Independent pixel-only inverse arm for the quadrant-rotor carpet.

The parser uses only role geometry: one large dark card, one large grid, four
smaller grids, a colored disk, and four bright 2-by-2 occupancy blocks.  Palette
values, absolute placement, renderer state, and latent scene data are unavailable.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image

CELL_NAMES = (
    "top-left",
    "top",
    "top-right",
    "left",
    "center",
    "right",
    "bottom-left",
    "bottom",
    "bottom-right",
)


@dataclass(frozen=True)
class Component:
    area: int
    x0: int
    y0: int
    x1: int
    y1: int

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1


def _components(mask: np.ndarray) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        seen[y, x] = True
        stack = [(y, x)]
        area = 0
        x0 = x1 = x
        y0 = y1 = y
        while stack:
            cy, cx = stack.pop()
            area += 1
            x0, x1 = min(x0, cx), max(x1, cx)
            y0, y1 = min(y0, cy), max(y1, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        output.append(Component(area, x0, y0, x1, y1))
    return output


def _rotate(cell: int, quadrant: int) -> int:
    row, col = divmod(cell, 3)
    qrow, qcol = divmod(quadrant, 2)
    if not (qrow <= row <= qrow + 1 and qcol <= col <= qcol + 1):
        return cell
    local_row, local_col = row - qrow, col - qcol
    return (qrow + local_col) * 3 + qcol + (1 - local_row)


def _cell_index(cx: float, cy: float, grid: Component) -> int:
    col = int(np.clip(np.floor(3.0 * (cx - grid.x0) / max(1, grid.width - 1)), 0, 2))
    row = int(np.clip(np.floor(3.0 * (cy - grid.y0) / max(1, grid.height - 1)), 0, 2))
    return row * 3 + col


def _decode_command(bright: np.ndarray, grid: Component) -> int:
    occupied: list[tuple[int, int]] = []
    for row in range(3):
        y0 = grid.y0 + row * (grid.height - 1) // 3 + 2
        y1 = grid.y0 + (row + 1) * (grid.height - 1) // 3 - 1
        for col in range(3):
            x0 = grid.x0 + col * (grid.width - 1) // 3 + 2
            x1 = grid.x0 + (col + 1) * (grid.width - 1) // 3 - 1
            if x1 <= x0 or y1 <= y0:
                raise ValueError("command grid has no resolvable cell interior")
            if int(bright[y0 : y1 + 1, x0 : x1 + 1].sum()) >= 12:
                occupied.append((row, col))
    if len(occupied) != 4:
        raise ValueError(f"command has {len(occupied)} bright cells rather than four")
    rows = sorted({row for row, _ in occupied})
    cols = sorted({col for _, col in occupied})
    if len(rows) != 2 or len(cols) != 2 or rows[1] != rows[0] + 1 or cols[1] != cols[0] + 1:
        raise ValueError("bright command cells do not form one contiguous 2-by-2 block")
    return rows[0] * 2 + cols[0]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    if array.shape[:2] != (512, 512):
        raise ValueError(f"expected a 512-by-512 raster, got {array.shape[1]} by {array.shape[0]}")

    # All supported card palettes are dark, but their hue is deliberately ignored.
    dark = np.max(array, axis=2) < 145
    ys, xs = np.nonzero(dark)
    if len(xs) < 25000:
        raise ValueError("the connected instruction card is missing or materially erased")
    bx0, bx1 = int(xs.min()), int(xs.max())
    by0, by1 = int(ys.min()), int(ys.max())
    card_w, card_h = bx1 - bx0 + 1, by1 - by0 + 1
    if not (184 <= card_w <= 192 and 212 <= card_h <= 220):
        raise ValueError(f"instruction card geometry is incomplete: {card_w} by {card_h}")
    dark_fraction = float(dark[by0 : by1 + 1, bx0 : bx1 + 1].mean())
    if not (0.815 <= dark_fraction <= 0.94):
        raise ValueError(f"instruction card interior is incomplete (dark fill {dark_fraction:.3f})")

    # Near-white line networks remain isolated inside the dark connected card.
    light = np.all(array > 205, axis=2)
    crop_components = _components(light[by0 : by1 + 1, bx0 : bx1 + 1])
    components = [
        Component(c.area, c.x0 + bx0, c.y0 + by0, c.x1 + bx0, c.y1 + by0)
        for c in crop_components
    ]
    main_candidates = [c for c in components if 82 <= c.width <= 98 and 82 <= c.height <= 98 and c.area > 700]
    command_candidates = [c for c in components if 32 <= c.width <= 41 and 32 <= c.height <= 41 and c.area > 180]
    if len(main_candidates) != 1:
        raise ValueError(f"expected one large start grid, found {len(main_candidates)}")
    if len(command_candidates) != 4:
        raise ValueError(f"expected four command grids, found {len(command_candidates)}")
    main = main_candidates[0]
    command_grids = sorted(command_candidates, key=lambda c: c.y0)
    if any(command_grids[i + 1].y0 - command_grids[i].y0 < 40 for i in range(3)):
        raise ValueError("command order is not visually resolved")

    maximum = np.max(array, axis=2)
    minimum = np.min(array, axis=2)
    bright_color = (maximum - minimum > 42) & (np.mean(array, axis=2) > 96)

    token_patch = bright_color[main.y0 : main.y1 + 1, main.x0 : main.x1 + 1]
    ty, tx = np.nonzero(token_patch)
    if not (35 <= len(tx) <= 260):
        raise ValueError(f"token disk has unresolved area {len(tx)}")
    token_cell = _cell_index(float(tx.mean() + main.x0), float(ty.mean() + main.y0), main)

    commands = [_decode_command(bright_color, grid) for grid in command_grids]
    cell = token_cell
    for quadrant in commands:
        cell = _rotate(cell, quadrant)
    return CELL_NAMES[cell]
