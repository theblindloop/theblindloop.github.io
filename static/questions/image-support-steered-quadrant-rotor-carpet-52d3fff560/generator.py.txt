"""Deterministic latent sampler and raster renderer for the rotor carpet."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CARD_W = 188
CARD_H = 216
CELL_NAMES = (
    "top-left",
    "top",
    "top-right",
    "left",
    "center",
    "right",
    "bottom-left",
    "bottom",
    "bottom-right",
)
CARD_COLORS = ((30, 46, 70), (65, 38, 76), (27, 68, 61))
TOKEN_COLORS = ((37, 202, 211), (240, 91, 137), (157, 211, 77))
ACCENT_COLORS = ((247, 180, 48), (177, 116, 235), (244, 106, 74))
GRID_COLOR = (240, 244, 249)
BACKGROUND = (255, 255, 255)


def _rotate_cell(cell: int, quadrant: int, clockwise: bool = True) -> int:
    row, col = divmod(int(cell), 3)
    qrow, qcol = divmod(int(quadrant), 2)
    if not (qrow <= row <= qrow + 1 and qcol <= col <= qcol + 1):
        return int(cell)
    local = (row - qrow, col - qcol)
    if clockwise:
        mapped = (local[1], 1 - local[0])
    else:
        mapped = (1 - local[1], local[0])
    return (qrow + mapped[0]) * 3 + qcol + mapped[1]


def _terminal_cell(scene: dict) -> int:
    cell = int(scene["token_cell"])
    for quadrant in scene["commands"]:
        cell = _rotate_cell(cell, int(quadrant), clockwise=True)
    return cell


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51A7C0DE)
    target = int(seed) % 9
    for _ in range(1000):
        commands = [rng.randrange(4) for _ in range(4)]
        start = target
        for quadrant in reversed(commands):
            start = _rotate_cell(start, quadrant, clockwise=False)
        probe = start
        moves = 0
        for quadrant in commands:
            nxt = _rotate_cell(probe, quadrant, clockwise=True)
            moves += int(nxt != probe)
            probe = nxt
        if probe == target and start != target and moves >= 2 and len(set(commands)) >= 2:
            break
    else:
        raise RuntimeError("could not sample a sufficiently compositional rotor program")
    return {
        "card_x": 162 + rng.randint(-4, 4),
        "card_y": 148 + rng.randint(-4, 4),
        "token_cell": start,
        "commands": commands,
        "token_radius": 5 + rng.randrange(3),
        "card_palette": rng.randrange(len(CARD_COLORS)),
        "token_palette": rng.randrange(len(TOKEN_COLORS)),
        "accent_palette": rng.randrange(len(ACCENT_COLORS)),
    }


def render(scene: dict) -> Image.Image:
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), BACKGROUND)
    draw = ImageDraw.Draw(image)
    card_x = int(scene["card_x"])
    card_y = int(scene["card_y"])
    token_cell = int(scene["token_cell"])
    commands = [int(value) for value in scene["commands"]]
    token_radius = int(scene["token_radius"])
    card_color = CARD_COLORS[int(scene["card_palette"]) % len(CARD_COLORS)]
    token_color = TOKEN_COLORS[int(scene["token_palette"]) % len(TOKEN_COLORS)]
    accent_color = ACCENT_COLORS[int(scene["accent_palette"]) % len(ACCENT_COLORS)]

    def box(values: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
        return tuple(int(value * ss) for value in values)

    draw.rectangle(box((card_x, card_y, card_x + CARD_W - 1, card_y + CARD_H - 1)), fill=card_color)

    # Large start grid.
    gx, gy, cell = card_x + 18, card_y + 63, 30
    for k in range(4):
        draw.line(
            [(gx * ss, (gy + k * cell) * ss), ((gx + 3 * cell) * ss, (gy + k * cell) * ss)],
            fill=GRID_COLOR,
            width=3 * ss,
        )
        draw.line(
            [((gx + k * cell) * ss, gy * ss), ((gx + k * cell) * ss, (gy + 3 * cell) * ss)],
            fill=GRID_COLOR,
            width=3 * ss,
        )
    token_row, token_col = divmod(token_cell, 3)
    tx = gx + token_col * cell + cell // 2
    ty = gy + token_row * cell + cell // 2
    draw.ellipse(
        box((tx - token_radius, ty - token_radius, tx + token_radius, ty + token_radius)),
        fill=token_color,
    )

    # Four small command grids; their vertical order is the program order.
    mx, first_y, small, stride = card_x + 133, card_y + 15, 12, 49
    for index, quadrant in enumerate(commands):
        my = first_y + index * stride
        qrow, qcol = divmod(quadrant, 2)
        for row in range(qrow, qrow + 2):
            for col in range(qcol, qcol + 2):
                x0 = mx + col * small + 2
                y0 = my + row * small + 2
                draw.rectangle(
                    box((x0, y0, x0 + small - 4, y0 + small - 4)),
                    fill=accent_color,
                )
        for k in range(4):
            draw.line(
                [(mx * ss, (my + k * small) * ss), ((mx + 3 * small) * ss, (my + k * small) * ss)],
                fill=GRID_COLOR,
                width=2 * ss,
            )
            draw.line(
                [((mx + k * small) * ss, my * ss), ((mx + k * small) * ss, (my + 3 * small) * ss)],
                fill=GRID_COLOR,
                width=2 * ss,
            )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return CELL_NAMES[_terminal_cell(scene)]


def margin(scene: dict) -> float:
    return float(int(scene["token_radius"]) - 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
