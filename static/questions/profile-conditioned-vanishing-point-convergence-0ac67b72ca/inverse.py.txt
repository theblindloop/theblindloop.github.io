"""Independent pixel-only inverse for line concurrence."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (214, 65, 68),
    "blue": (48, 106, 190),
    "green": (45, 151, 91),
}
COLOR_TOLERANCE = 92.0
MIN_COMPONENT_PIXELS = 28


def _connected_components(mask: np.ndarray) -> list[np.ndarray]:
    ys, xs = np.nonzero(mask)
    if len(xs) == 0:
        return []
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    local = mask[y0:y1, x0:x1]
    visited = np.zeros(local.shape, dtype=bool)
    components: list[np.ndarray] = []
    height, width = local.shape
    for yy, xx in zip(*np.nonzero(local)):
        if visited[yy, xx]:
            continue
        visited[yy, xx] = True
        stack = [(int(yy), int(xx))]
        points: list[tuple[float, float]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((float(cx + x0), float(cy + y0)))
            for ny, nx in (
                (cy - 1, cx),
                (cy + 1, cx),
                (cy, cx - 1),
                (cy, cx + 1),
                (cy - 1, cx - 1),
                (cy - 1, cx + 1),
                (cy + 1, cx - 1),
                (cy + 1, cx + 1),
            ):
                if 0 <= ny < height and 0 <= nx < width and local[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= MIN_COMPONENT_PIXELS:
            components.append(np.asarray(points, dtype=np.float64))
    return components


def _field_features(image: Image.Image, color: tuple[int, int, int]) -> tuple[np.ndarray, np.ndarray]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.float64)
    target = np.asarray(color, dtype=np.float64)
    mask = np.sqrt(np.sum((pixels - target) ** 2, axis=2)) <= COLOR_TOLERANCE
    components = _connected_components(mask)
    centers: list[np.ndarray] = []
    directions: list[np.ndarray] = []
    for component in components:
        center = component.mean(axis=0)
        centered = component - center
        covariance = centered.T @ centered / max(1, len(component) - 1)
        values, vectors = np.linalg.eigh(covariance)
        if float(values.max()) < 2.0:
            continue
        direction = vectors[:, int(np.argmax(values))]
        centers.append(center)
        directions.append(direction / max(np.linalg.norm(direction), 1e-12))
    if len(centers) < 12:
        raise ValueError("not enough separated stroke components for a field")
    if len(centers) > 18:
        ranked = sorted(
            zip(components, centers, directions),
            key=lambda item: len(item[0]),
            reverse=True,
        )[:18]
        centers = [item[1] for item in ranked]
        directions = [item[2] for item in ranked]
    return np.asarray(centers), np.asarray(directions)


def _convergence_residual(points: np.ndarray, directions: np.ndarray) -> float:
    normals = np.column_stack((-directions[:, 1], directions[:, 0]))
    rhs = np.sum(normals * points, axis=1)
    matrix = normals.T @ normals
    if abs(float(np.linalg.det(matrix))) < 1e-8:
        raise ValueError("stroke axes do not constrain a focal point")
    center = np.linalg.solve(matrix, normals.T @ rhs)
    residuals = normals @ center - rhs
    return float(math.sqrt(np.mean(residuals * residuals)))


def decision_from_image(image: Image.Image) -> str:
    """Return the color whose segmented stroke axes are most concurrent."""
    scores = {}
    for name, color in PALETTE.items():
        points, directions = _field_features(image, color)
        scores[name] = _convergence_residual(points, directions)
    return min(scores, key=lambda name: (scores[name], name))
