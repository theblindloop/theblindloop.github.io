"""Renderer and analytic gold for a global line-concurrence question."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 960
HEIGHT = 360
PANEL_CENTERS = (160.0, 480.0, 800.0)
PANEL_Y = 180.0
BACKGROUND = (248, 249, 246)
PANEL_EDGE = (210, 216, 211)
COLOR_RGB = {
    "red": (214, 65, 68),
    "blue": (48, 106, 190),
    "green": (45, 151, 91),
}
COLORS = tuple(COLOR_RGB)
MIN_MARGIN = 8.0


def _convergence_residual(points: list[list[float]], angles: list[float]) -> float:
    """Return the RMS distance from all segment axes to their best intersection."""
    if len(points) != len(angles) or len(points) < 3:
        return float("inf")
    a00 = a01 = a11 = b0 = b1 = 0.0
    normals: list[tuple[float, float]] = []
    for point, angle in zip(points, angles):
        nx = -math.sin(float(angle))
        ny = math.cos(float(angle))
        px, py = float(point[0]), float(point[1])
        rhs = nx * px + ny * py
        a00 += nx * nx
        a01 += nx * ny
        a11 += ny * ny
        b0 += nx * rhs
        b1 += ny * rhs
        normals.append((nx, ny))
    determinant = a00 * a11 - a01 * a01
    if determinant < 1e-8:
        return float("inf")
    center_x = (b0 * a11 - b1 * a01) / determinant
    center_y = (a00 * b1 - a01 * b0) / determinant
    squared = []
    for point, (nx, ny) in zip(points, normals):
        residual = nx * (float(point[0]) - center_x) + ny * (float(point[1]) - center_y)
        squared.append(residual * residual)
    return math.sqrt(sum(squared) / len(squared))


def _scores(scene: dict[str, Any]) -> list[tuple[str, float]]:
    return [
        (
            str(field["color"]),
            _convergence_residual(field["points"], field["angles"]),
        )
        for field in scene["fields"]
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample three equal-marginal fields with one concurrent assignment."""
    rng = random.Random(int(seed))
    count = rng.choice((14, 16))
    phase = rng.uniform(0.0, 2.0 * math.pi)
    radius_x = rng.uniform(88.0, 101.0)
    radius_y = rng.uniform(84.0, 99.0)
    focal_x = rng.uniform(-7.0, 7.0)
    focal_y = rng.uniform(-7.0, 7.0)
    point_angles = [
        phase + (2.0 * math.pi * index / count) + rng.uniform(-0.035, 0.035)
        for index in range(count)
    ]
    local_points = [
        [
            round(radius_x * math.cos(angle) + rng.uniform(-2.0, 2.0), 4),
            round(radius_y * math.sin(angle) + rng.uniform(-2.0, 2.0), 4),
        ]
        for angle in point_angles
    ]
    local_focal = [focal_x, focal_y]
    local_angles = [
        math.atan2(local_focal[1] - point[1], local_focal[0] - point[0])
        for point in local_points
    ]
    shifted = local_angles[count // 3 :] + local_angles[: count // 3]
    reversed_assignment = list(reversed(local_angles))
    roles = [local_angles, shifted, reversed_assignment]
    rng.shuffle(roles)

    target_color = COLORS[int(seed) % len(COLORS)]
    other_colors = [color for color in COLORS if color != target_color]
    rng.shuffle(other_colors)
    role_colors: dict[int, str] = {}
    target_role_index = next(index for index, role in enumerate(roles) if role is local_angles)
    for index in range(3):
        if index == target_role_index:
            role_colors[index] = target_color
        else:
            role_colors[index] = other_colors.pop()

    length = round(rng.uniform(29.0, 35.0), 4)
    stroke = round(rng.uniform(5.2, 6.6), 4)
    fields: list[dict[str, Any]] = []
    for slot, assignment in enumerate(roles):
        cx = PANEL_CENTERS[slot]
        cy = PANEL_Y
        points = [[round(cx + point[0], 4), round(cy + point[1], 4)] for point in local_points]
        fields.append(
            {
                "slot": slot,
                "color": role_colors[slot],
                "points": points,
                "angles": [round(float(angle), 7) for angle in assignment],
                "length": length,
                "stroke": stroke,
            }
        )
    rng.shuffle(fields)
    return {"fields": fields}


def render(scene: dict[str, Any], supersample: int = 4) -> Image.Image:
    scale = int(supersample)
    if scale < 1:
        raise ValueError("supersample must be positive")
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for center_x in PANEL_CENTERS:
        draw.rounded_rectangle(
            (
                int(round((center_x - 132.0) * scale)),
                int(round(34.0 * scale)),
                int(round((center_x + 132.0) * scale)),
                int(round(326.0 * scale)),
            ),
            radius=int(round(16.0 * scale)),
            outline=PANEL_EDGE,
            width=max(1, int(round(2.0 * scale))),
        )
    for field in sorted(scene["fields"], key=lambda item: int(item["slot"])):
        color = COLOR_RGB[str(field["color"])]
        half = float(field["length"]) / 2.0
        width = max(1, int(round(float(field["stroke"]) * scale)))
        for point, angle in zip(field["points"], field["angles"]):
            dx = half * math.cos(float(angle))
            dy = half * math.sin(float(angle))
            draw.line(
                (
                    (float(point[0]) - dx) * scale,
                    (float(point[1]) - dy) * scale,
                    (float(point[0]) + dx) * scale,
                    (float(point[1]) + dy) * scale,
                ),
                fill=color,
                width=width,
            )
    if scale != 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    scores = sorted(_scores(scene), key=lambda item: (item[1], item[0]))
    return scores[0][0]


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(score for _, score in _scores(scene))
    return round(scores[1] - scores[0], 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < MIN_MARGIN
    except (ValueError, IndexError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent alias is declared; every field record is label-bearing geometry."""
    return []
