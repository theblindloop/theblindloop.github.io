"""Deterministic single-junction-resilient seam graph sampler and renderer."""

from __future__ import annotations

from collections import deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1020, 650
BACKGROUND = (250, 248, 244)
INK = (42, 47, 55)
JOINT = (218, 230, 241)
PANEL = (164, 176, 189)
LABEL = (67, 76, 88)
SAME_FILL = (207, 232, 219)
FLIP_FILL = (242, 215, 221)
SOCKET = (185, 201, 218)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
COLOR_NAMES = tuple(COLORS)
PATCH_BASES = (25, 365, 705)
GRID_X = (18, 130, 242)
GRID_Y = (160, 280, 400, 520)

# One audited block-cut topology. Per-patch row reflections generate eight
# visibly different, connectivity-preserving seam presentations.
BASE_PATCH_EDGES = (
    (("P0r0c1", "P0r0c2"), ("P0r1c1", "P0r1c2"),
     ("P0r2c0", "P0r2c1"), ("P0r2c1", "P0r2c2"),
     ("P0r3c0", "P0r3c1"), ("P0r0c2", "P0r1c2"),
     ("P0r1c0", "P0r2c0"), ("P0r1c1", "P0r2c1"),
     ("P0r2c0", "P0r3c0"), ("P0r2c1", "P0r3c1")),
    (("P1r0c0", "P1r0c1"), ("P1r1c0", "P1r1c1"),
     ("P1r1c1", "P1r1c2"), ("P1r2c0", "P1r2c1"),
     ("P1r2c1", "P1r2c2"), ("P1r3c0", "P1r3c1"),
     ("P1r3c1", "P1r3c2"), ("P1r1c1", "P1r2c1"),
     ("P1r1c2", "P1r2c2"), ("P1r2c1", "P1r3c1"),
     ("P1r2c2", "P1r3c2")),
    (("P2r0c0", "P2r0c1"), ("P2r3c0", "P2r3c1"),
     ("P2r3c1", "P2r3c2"), ("P2r0c1", "P2r1c1"),
     ("P2r1c1", "P2r2c1"), ("P2r2c1", "P2r3c1")),
)
BASE_PAIRS = (
    ("P0r1c1", "P2r3c0"),  # junction-resilient
    ("P0r2c0", "P1r2c1"),  # edge-resilient but has a cut junction
    ("P0r3c1", "P1r3c2"),  # edge-resilient but has a cut junction
)


def _node(patch: int, row: int, col: int) -> str:
    return f"P{patch}r{row}c{col}"


def _decode(node_id: str) -> tuple[int, int, int]:
    return int(node_id[1]), int(node_id[3]), int(node_id[5])


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _all_nodes() -> list[str]:
    return [_node(p, r, c) for p in range(3) for r in range(4) for c in range(3)]


def _graph(scene: dict) -> tuple[dict[str, list[str]], list[set[str]]]:
    nodes = _all_nodes()
    adjacency = {node: [] for node in nodes}
    for patch in scene["patches"]:
        for a, b in patch["edges"]:
            adjacency[a].append(b)
            adjacency[b].append(a)
    for seam in scene["seams"]:
        left = int(seam["left"])
        mode = str(seam["mode"])
        for row in range(4):
            other = row if mode == "same" else 3 - row
            a, b = _node(left, row, 2), _node(left + 1, other, 0)
            adjacency[a].append(b)
            adjacency[b].append(a)
    remaining, parts = set(nodes), []
    while remaining:
        start = min(remaining)
        reached, queue = {start}, deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here]:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        parts.append(reached)
        remaining -= reached
    return adjacency, parts


def _local_edges(scene: dict) -> list[tuple[str, str]]:
    return [tuple(sorted(edge)) for patch in scene["patches"] for edge in patch["edges"]]


def _connected_after_edge(scene: dict, start: str, goal: str,
                          removed: tuple[str, str] | None = None) -> bool:
    adjacency, _ = _graph(scene)
    if removed is not None:
        a, b = removed
        adjacency[a].remove(b)
        adjacency[b].remove(a)
    reached, queue = {start}, deque([start])
    while queue:
        here = queue.popleft()
        if here == goal:
            return True
        for nxt in adjacency[here]:
            if nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    return goal in reached


def _edge_resilient(scene: dict, pair: list[str] | tuple[str, str]) -> bool:
    a, b = pair
    return _connected_after_edge(scene, a, b) and all(
        _connected_after_edge(scene, a, b, edge) for edge in _local_edges(scene)
    )


def _edge_parts(scene: dict) -> list[set[str]]:
    bridges = {edge for edge in _local_edges(scene)
               if not _connected_after_edge(scene, edge[0], edge[1], edge)}
    adjacency, _ = _graph(scene)
    for a, b in bridges:
        adjacency[a].remove(b)
        adjacency[b].remove(a)
    remaining, parts = set(_all_nodes()), []
    while remaining:
        start = min(remaining)
        reached, queue = {start}, deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here]:
                if nxt not in reached:
                    reached.add(nxt)
                    queue.append(nxt)
        parts.append(reached)
        remaining -= reached
    return parts


def _is_socket(node_id: str) -> bool:
    patch, _, col = _decode(node_id)
    return (patch < 2 and col == 2) or (patch > 0 and col == 0)


def _connected_without_junction(scene: dict, start: str, goal: str,
                                removed: str | None = None) -> bool:
    adjacency, _ = _graph(scene)
    if removed is not None:
        for neighbor in list(adjacency[removed]):
            adjacency[neighbor].remove(removed)
        adjacency[removed] = []
    reached, queue = {start}, deque([start])
    while queue:
        here = queue.popleft()
        if here == goal:
            return True
        for nxt in adjacency[here]:
            if nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    return goal in reached


def _removable_junctions(scene: dict, protected: set[str] | None = None) -> list[str]:
    terminals = ({node for pair in scene.get("terminals", {}).values() for node in pair}
                 if protected is None else protected)
    return [node for node in _all_nodes() if node not in terminals and not _is_socket(node)]


def _junction_failures(scene: dict, pair: list[str] | tuple[str, str],
                       protected: set[str] | None = None) -> list[str]:
    a, b = pair
    return [node for node in _removable_junctions(scene, protected)
            if not _connected_without_junction(scene, a, b, node)]


def _resilient(scene: dict, pair: list[str] | tuple[str, str]) -> bool:
    a, b = pair
    return _connected_without_junction(scene, a, b) and not _junction_failures(scene, pair)


def _choose_terminals(rng: random.Random, scene: dict, answer: str) -> dict[str, list[str]] | None:
    adjacency, parts = _graph(scene)
    active = [n for n in _all_nodes() if adjacency[n]]
    robust_parts = _edge_parts(scene)
    pairs = [(a, b) for i, a in enumerate(active) for b in active[i + 1:]
             if _decode(a)[0] != _decode(b)[0]
             and any(a in part and b in part for part in parts)
             and any(a in part and b in part for part in robust_parts)]
    rng.shuffle(pairs)
    classified = [(pair, _junction_failures(scene, pair, set(pair))) for pair in pairs]
    sturdy = [pair for pair, failures in classified
              if not failures and {_decode(pair[0])[0], _decode(pair[1])[0]} == {0, 2}]
    fragile = [pair for pair, failures in classified if 1 <= len(failures) <= 3]
    for good in sturdy[:80]:
        alternatives = [pair for pair in fragile[:180] if not set(pair) & set(good)]
        for i, first in enumerate(alternatives):
            for second in alternatives[i + 1:]:
                if len(set((*good, *first, *second))) != 6:
                    continue
                wrong = [color for color in COLOR_NAMES if color != answer]
                rng.shuffle(wrong)
                terminals = {
                    answer: sorted(good),
                    wrong[0]: sorted(first),
                    wrong[1]: sorted(second),
                }
                scene["terminals"] = terminals
                failures = {color: _junction_failures(scene, pair)
                            for color, pair in terminals.items()}
                if (not failures[answer]
                        and all(1 <= len(failures[color]) <= 3 for color in wrong)
                        and set(failures[wrong[0]]) != set(failures[wrong[1]])):
                    return terminals
    return None


def sample_scene(seed: int) -> dict:
    """Sample seam graphs with one pair robust to every pale-junction loss."""

    seed = int(seed)
    flips = tuple(bool((seed // 6) >> patch & 1) for patch in range(3))

    def transform(node_id: str) -> str:
        patch, row, col = _decode(node_id)
        return _node(patch, 3-row if flips[patch] else row, col)

    patches = []
    for patch, base_edges in enumerate(BASE_PATCH_EDGES):
        edges = sorted(_edge(transform(a), transform(b)) for a, b in base_edges)
        patches.append({"name": chr(65 + patch), "edges": edges})
    base_modes = ("flip", "same")
    seams = []
    for left, base_mode in enumerate(base_modes):
        parity = (base_mode == "flip") ^ flips[left] ^ flips[left + 1]
        seams.append({"left": left, "right": left + 1,
                      "mode": "flip" if parity else "same"})
    transformed_pairs = [sorted(transform(node) for node in pair) for pair in BASE_PAIRS]
    answer = COLOR_NAMES[seed % 3]
    wrong = [color for color in COLOR_NAMES if color != answer]
    if (seed // 3) % 2:
        wrong.reverse()
    terminals = {answer: transformed_pairs[0], wrong[0]: transformed_pairs[1],
                 wrong[1]: transformed_pairs[2]}
    scene = {"patches": patches, "seams": seams, "terminals": terminals}
    if is_quarantined(scene):
        raise RuntimeError(f"constructed invalid scene for seed {seed}")
    return scene


def _answer(scene: dict) -> str:
    answers = [str(color) for color, pair in scene["terminals"].items()
               if len(pair) == 2 and _resilient(scene, pair)]
    if len(answers) != 1:
        raise ValueError(f"expected one resilient pair, got {answers}")
    return answers[0]


def analytic_gold(scene: dict) -> str:
    answer = _answer(scene)
    if answer not in COLORS:
        raise ValueError("unknown answer color")
    return answer


def margin(scene: dict) -> float:
    _answer(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["patches"]) != 3 or len(scene["seams"]) != 2:
            return True
        if {p["name"] for p in scene["patches"]} != {"A", "B", "C"}:
            return True
        nodes = set(_all_nodes())
        local_edges = []
        for patch_index, patch in enumerate(sorted(scene["patches"], key=lambda p: p["name"])):
            for edge in patch["edges"]:
                if len(edge) != 2 or edge[0] not in nodes or edge[1] not in nodes:
                    return True
                p1, r1, c1 = _decode(edge[0])
                p2, r2, c2 = _decode(edge[1])
                if p1 != patch_index or p2 != patch_index or abs(r1-r2) + abs(c1-c2) != 1:
                    return True
                local_edges.append(tuple(sorted(edge)))
        if len(local_edges) != len(set(local_edges)) or not 12 <= len(local_edges) <= 42:
            return True
        seams = sorted(scene["seams"], key=lambda s: int(s["left"]))
        if [(s["left"], s["right"]) for s in seams] != [(0, 1), (1, 2)]:
            return True
        if any(s["mode"] not in {"same", "flip"} for s in seams):
            return True
        if set(scene["terminals"]) != set(COLOR_NAMES):
            return True
        marked = [n for pair in scene["terminals"].values() for n in pair]
        if any(len(pair) != 2 for pair in scene["terminals"].values()):
            return True
        if len(marked) != 6 or len(set(marked)) != 6 or not set(marked) <= nodes:
            return True
        adjacency, parts = _graph(scene)
        if any(not adjacency[node] for node in marked):
            return True
        answer = _answer(scene)
        a, b = scene["terminals"][answer]
        if {_decode(a)[0], _decode(b)[0]} != {0, 2}:
            return True
        if any(_decode(a)[0] == _decode(b)[0]
               for a, b in scene["terminals"].values()):
            return True
        if not all(_connected_after_edge(scene, *pair) for pair in scene["terminals"].values()):
            return True
        # The parent's edge-disjoint-route strategy must be non-diagnostic here.
        if not all(_edge_resilient(scene, pair) for pair in scene["terminals"].values()):
            return True
        if any(_resilient(scene, pair) != (color == answer)
               for color, pair in scene["terminals"].items()):
            return True
        if any(not 1 <= len(_junction_failures(scene, pair)) <= 3
               for color, pair in scene["terminals"].items() if color != answer):
            return True
        return False
    except (KeyError, TypeError, ValueError):
        return True


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _xy(node_id: str) -> tuple[int, int]:
    patch, row, col = _decode(node_id)
    return PATCH_BASES[patch] + GRID_X[col], GRID_Y[row]


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    patches = {str(patch["name"]): patch for patch in scene["patches"]}
    for patch_index, base in enumerate(PATCH_BASES):
        draw.rounded_rectangle((base*scale, 105*scale, (base+260)*scale, 575*scale),
                               radius=15*scale, outline=PANEL, width=3*scale)
        draw.text(((base+130)*scale, 69*scale), f"PATCH {chr(65+patch_index)}",
                  fill=LABEL, anchor="mm", font=_font(22*scale))
        for a, b in sorted(_edge(*edge) for edge in patches[chr(65+patch_index)]["edges"]):
            draw.line([tuple(v*scale for v in _xy(a)), tuple(v*scale for v in _xy(b))],
                      fill=INK, width=7*scale)

    terminal_of = {node: color for color, pair in scene["terminals"].items() for node in pair}
    for node_id in sorted(_all_nodes()):
        x, y = _xy(node_id)
        patch, _, col = _decode(node_id)
        radius = 18 if node_id in terminal_of else 11
        fill = COLORS[terminal_of[node_id]] if node_id in terminal_of else (SOCKET if (patch < 2 and col == 2) or (patch > 0 and col == 0) else JOINT)
        if fill == SOCKET:
            draw.rounded_rectangle(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale),
                                   radius=4*scale, fill=fill, outline=INK, width=3*scale)
        else:
            draw.ellipse(((x-radius)*scale, (y-radius)*scale, (x+radius)*scale, (y+radius)*scale),
                         fill=fill, outline=INK, width=3*scale)

    for seam in sorted(scene["seams"], key=lambda s: int(s["left"])):
        left = int(seam["left"])
        cx = 325 + 340 * left
        mode = str(seam["mode"])
        fill = SAME_FILL if mode == "same" else FLIP_FILL
        draw.rounded_rectangle(((cx-27)*scale, 112*scale, (cx+27)*scale, 568*scale),
                               radius=12*scale, fill=fill, outline=PANEL, width=2*scale)
        draw.rounded_rectangle(((cx-43)*scale, 82*scale, (cx+43)*scale, 130*scale),
                               radius=9*scale, fill=fill, outline=INK, width=2*scale)
        draw.text((cx*scale, 99*scale), mode.upper(), fill=INK, anchor="mm", font=_font(16*scale))
        draw.text((cx*scale, 119*scale), "↑ ↑" if mode == "same" else "↑ ↓",
                  fill=INK, anchor="mm", font=_font(15*scale))
        for row in range(4):
            other = row if mode == "same" else 3-row
            y1, y2 = GRID_Y[row], GRID_Y[other]
            draw.line(((cx-18)*scale, y1*scale, (cx+18)*scale, y2*scale), fill=LABEL, width=3*scale)
            draw.ellipse(((cx-22)*scale, (y1-4)*scale, (cx-14)*scale, (y1+4)*scale), fill=INK)
            draw.ellipse(((cx+14)*scale, (y2-4)*scale, (cx+22)*scale, (y2+4)*scale), fill=INK)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["patches"] = list(reversed(twin["patches"]))
    for patch in twin["patches"]:
        patch["edges"] = list(reversed(patch["edges"]))
    twin["seams"] = list(reversed(twin["seams"]))
    twin["terminals"] = {k: list(reversed(v)) for k, v in reversed(list(twin["terminals"].items()))}
    return [("reverse_record_orders", twin)]
