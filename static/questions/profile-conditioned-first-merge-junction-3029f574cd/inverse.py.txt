"""Independent PNG-only inverse arm for the first common route junction."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

INK = (46, 52, 59)
GRAY = (151, 158, 166)
GREEN = (32, 164, 94)
RED = (220, 70, 67)
YELLOW = (244, 190, 46)
COLORS = {
    "orange": (238, 135, 35),
    "purple": (135, 82, 190),
    "blue": (51, 105, 210),
}


def _components(mask: np.ndarray, minimum: int = 25) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            result.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return result


def _one(array: np.ndarray, rgb: tuple[int, int, int], label: str) -> tuple[float, float]:
    found = _components(np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2))
    if len(found) != 1:
        raise ValueError(f"expected one {label} marker")
    return found[0][0], found[0][1]


def _ink_link(array: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    x1, y1 = a
    x2, y2 = b
    distance = float(np.hypot(x2 - x1, y2 - y1))
    if distance < 50 or distance > 230:
        return False
    ink = np.all(array == np.asarray(INK, dtype=np.uint8), axis=2)
    hits = 0
    count = 64
    for index in range(count):
        t = 0.16 + 0.68 * index / (count - 1)
        x = int(round(x1 + t * (x2 - x1)))
        y = int(round(y1 + t * (y2 - y1)))
        y0, y1w = max(0, y - 2), min(ink.shape[0], y + 3)
        x0, x1w = max(0, x - 2), min(ink.shape[1], x + 3)
        hits += bool(ink[y0:y1w, x0:x1w].any())
    return hits / count >= 0.92


def _path(graph: list[set[int]], start: int, goal: int) -> list[int]:
    queue = deque([start])
    previous = {start: -1}
    while queue:
        node = queue.popleft()
        if node == goal:
            break
        for neighbor in graph[node]:
            if neighbor not in previous:
                previous[neighbor] = node
                queue.append(neighbor)
    if goal not in previous:
        raise ValueError("root is unreachable from a source")
    result = []
    node = goal
    while node != -1:
        result.append(node)
        node = previous[node]
    return list(reversed(result))


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    nodes: list[tuple[float, float, str | None]] = []
    green = _one(array, GREEN, "green source")
    red = _one(array, RED, "red source")
    root = _one(array, YELLOW, "yellow root")
    nodes.extend([(green[0], green[1], None), (red[0], red[1], None), (root[0], root[1], None)])
    for x, y, _ in _components(np.all(array == np.asarray(GRAY, dtype=np.uint8), axis=2)):
        nodes.append((x, y, None))
    for name, rgb in COLORS.items():
        found = _components(np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2))
        if len(found) != 1:
            raise ValueError(f"expected one {name} candidate")
        nodes.append((found[0][0], found[0][1], name))
    if len(nodes) != 16:
        raise ValueError("expected sixteen explicit tree nodes")

    graph = [set() for _ in nodes]
    for left in range(len(nodes)):
        for right in range(left + 1, len(nodes)):
            if _ink_link(array, nodes[left][:2], nodes[right][:2]):
                graph[left].add(right)
                graph[right].add(left)
    if sum(len(neighbors) for neighbors in graph) != 30 or any(not neighbors for neighbors in graph):
        raise ValueError("visible links do not reconstruct a tree")

    path_green = _path(graph, 0, 2)
    path_red = _path(graph, 1, 2)
    red_set = set(path_red)
    shared = [node for node in path_green if node in red_set]
    if not shared:
        raise ValueError("source routes have no common junction")
    first = shared[0]
    answer = nodes[first][2]
    if answer not in COLORS:
        raise ValueError("first common junction is not a colored candidate")
    return answer
