"""Deterministic renderer and analytic arm for a two-source route merge."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BG = (248, 247, 243)
PAPER = (255, 255, 255)
INK = (46, 52, 59)
GRAY = (151, 158, 166)
GREEN = (32, 164, 94)
RED = (220, 70, 67)
YELLOW = (244, 190, 46)
ORANGE = (238, 135, 35)
PURPLE = (135, 82, 190)
BLUE = (51, 105, 210)
CANDIDATE_RGB = {"orange": ORANGE, "purple": PURPLE, "blue": BLUE}

# A fixed abstract tree. Layout transforms change pixels but not its topology.
EDGES = [
    ("L0", "A"), ("L1", "A"), ("L2", "B"), ("L3", "B"),
    ("L4", "C"), ("L5", "C"), ("L6", "D"), ("L7", "D"),
    ("A", "E"), ("B", "E"), ("C", "F"), ("D", "F"),
    ("E", "G"), ("F", "G"), ("G", "R"),
]
CANDIDATE_NODES = ("A", "E", "G")
BASE_POSITIONS = {
    "L0": (67, 58), "L1": (67, 108), "L2": (67, 158), "L3": (67, 208),
    "L4": (67, 272), "L5": (67, 322), "L6": (67, 372), "L7": (67, 422),
    "A": (197, 83), "B": (197, 183), "C": (197, 297), "D": (197, 397),
    "E": (337, 133), "F": (337, 347), "G": (467, 240), "R": (575, 240),
}


def _positions(scene: dict) -> dict[str, tuple[int, int]]:
    dx, dy = (int(v) for v in scene["translation"])
    flip = bool(scene["vertical_flip"])
    result = {}
    for name, (x, y) in BASE_POSITIONS.items():
        fy = HEIGHT - y if flip else y
        result[name] = (x + dx, fy + dy)
    return result


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    winner = seed % 3
    if winner == 0:
        pair = rng.choice([(0, 1), (1, 0)])
    elif winner == 1:
        pair = (rng.choice((0, 1)), rng.choice((2, 3)))
    else:
        pair = (rng.choice((0, 1, 2, 3)), rng.choice((4, 5, 6, 7)))
    if rng.getrandbits(1):
        pair = (pair[1], pair[0])

    permutations = [
        ["orange", "purple", "blue"], ["orange", "blue", "purple"],
        ["purple", "orange", "blue"], ["purple", "blue", "orange"],
        ["blue", "orange", "purple"], ["blue", "purple", "orange"],
    ]
    candidate_colors = permutations[(seed // 3) % len(permutations)]
    return {
        "source_pair": list(pair),
        "candidate_colors": candidate_colors,
        "vertical_flip": bool(rng.getrandbits(1)),
        "translation": [rng.randint(-6, 6), rng.randint(-4, 4)],
    }


def render(scene: dict) -> Image.Image:
    positions = _positions(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((22, 20, 618, 460), radius=18, fill=PAPER, outline=(215, 218, 220), width=2)

    for left, right in EDGES:
        draw.line((positions[left], positions[right]), fill=INK, width=7)

    sources = {f"L{scene['source_pair'][0]}": GREEN, f"L{scene['source_pair'][1]}": RED}
    candidates = dict(zip(CANDIDATE_NODES, scene["candidate_colors"]))
    for name, (x, y) in positions.items():
        if name == "R":
            draw.polygon([(x, y - 17), (x + 17, y), (x, y + 17), (x - 17, y)], fill=YELLOW, outline=PAPER)
            draw.line([(x, y - 15), (x + 15, y), (x, y + 15), (x - 15, y), (x, y - 15)], fill=INK, width=2)
            continue
        fill = sources.get(name)
        radius = 14 if fill else 11
        if name in candidates:
            fill = CANDIDATE_RGB[candidates[name]]
            radius = 14
        if fill is None:
            fill = GRAY
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill, outline=PAPER, width=3)
        if name in sources:
            draw.ellipse((x - 10, y - 10, x + 10, y + 10), outline=INK, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    first = int(scene["source_pair"][0])
    second = int(scene["source_pair"][1])
    if {first, second} == {0, 1}:
        node = "A"
    elif first < 4 and second < 4:
        node = "E"
    elif (first < 4) != (second < 4):
        node = "G"
    else:
        raise ValueError("source pair does not merge at a colored candidate")
    return dict(zip(CANDIDATE_NODES, scene["candidate_colors"]))[node]


def margin(scene: dict) -> float:
    positions = _positions(scene)
    answer = analytic_gold(scene)
    node = dict(zip(scene["candidate_colors"], CANDIDATE_NODES))[answer]
    parent = {"A": "E", "E": "G", "G": "R"}[node]
    x1, y1 = positions[node]
    x2, y2 = positions[parent]
    return float(round(math.hypot(x2 - x1, y2 - y1), 6))


def is_quarantined(scene: dict) -> bool:
    try:
        pair = [int(v) for v in scene["source_pair"]]
        colors = list(scene["candidate_colors"])
        translation = list(scene["translation"])
        if len(pair) != 2 or len(set(pair)) != 2 or any(v < 0 or v > 7 for v in pair):
            return True
        if sorted(colors) != ["blue", "orange", "purple"]:
            return True
        if len(translation) != 2 or any(abs(int(v)) > 6 for v in translation):
            return True
        analytic_gold(scene)
        return margin(scene) < 95.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
