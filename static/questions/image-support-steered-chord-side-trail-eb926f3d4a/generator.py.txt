"""Deterministic renderer for the endpoint-chord side-occupancy world."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


CANVAS = 512
SUPERSAMPLE = 4
QUARANTINE_CLEARANCE = 6.5

INKS = (
    (31, 69, 104),
    (91, 45, 111),
    (24, 92, 82),
    (112, 54, 45),
)
BACKGROUNDS = (
    (249, 247, 241),
    (246, 249, 250),
    (250, 247, 249),
    (247, 249, 244),
)


def sample_scene(seed: int) -> dict:
    """Pair adjacent seeds so label changes while every nuisance stays fixed."""
    rng = random.Random(seed // 2)
    return {
        "width": CANVAS,
        "height": CANVAS,
        "center_x": rng.choice((220.0, 224.0, 288.0, 292.0)),
        "center_y": rng.choice((160.0, 224.0, 288.0, 352.0)),
        "span": rng.choice((92.0, 96.0, 100.0)),
        "amplitude": rng.choice((6.0, 8.0, 10.0, 11.0)),
        "stroke": rng.choice((5.0, 5.5, 6.0)),
        "ink_rgb": list(rng.choice(INKS)),
        "background_rgb": list(rng.choice(BACKGROUNDS)),
        "mirror": rng.choice((-1, 1)),
        "bend_class": "one side" if seed % 2 == 0 else "both sides",
        "draw_reverse": bool(rng.getrandbits(1)),
    }


def _curve_points(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    span = float(scene["span"])
    amplitude = float(scene["amplitude"])
    mirror = int(scene["mirror"])
    bend_class = scene["bend_class"]
    points = []
    for index in range(161):
        t = index / 160.0
        x = cx + span * (t - 0.5)
        if bend_class == "one side":
            # Range [-A,+A]. Its chord is y=-A, so all interior samples
            # occupy one half-plane. Derivative amplitude is 2*pi*A.
            y_local = 2.0 * amplitude * math.sin(math.pi * t) - amplitude
        elif bend_class == "both sides":
            # Same range and derivative distribution, but opposite lobe signs.
            y_local = amplitude * math.sin(2.0 * math.pi * t)
        else:
            raise ValueError(f"unknown bend_class {bend_class!r}")
        points.append((x, cy + mirror * y_local))
    if bool(scene["draw_reverse"]):
        points.reverse()
    return points


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(v) for v in scene["background_rgb"])
    ink = tuple(int(v) for v in scene["ink_rgb"])
    scale = SUPERSAMPLE
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    points = _curve_points(scene)
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    line_width = max(1, round(float(scene["stroke"]) * scale))
    draw.line(scaled, fill=ink, width=line_width, joint="curve")
    radius = line_width / 2.0
    # Fixed endpoint order keeps the latent draw-order alias raster-identical.
    natural = _curve_points({**scene, "draw_reverse": False})
    for x, y in (natural[0], natural[-1]):
        sx, sy = x * scale, y * scale
        draw.ellipse((sx - radius, sy - radius, sx + radius, sy + radius), fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["bend_class"])


def margin(scene: dict) -> float:
    return float(scene["amplitude"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
