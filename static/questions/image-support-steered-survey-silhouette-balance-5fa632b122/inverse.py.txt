"""Independent pixel-only inverse arm for the survey silhouette.

The oracle receives only a final RGB raster.  It finds the sole saturated
filled region, validates that every occupied column is a continuous run down to
one common baseline, splits the recovered horizontal extent at its midpoint,
and compares colored pixel area.  Equal widths make area proportional to mean
height.  No renderer state, scene data, file metadata, or fixed coordinates are
used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _filled_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    chroma = high - low
    # The public visual contract uses one moderately dark saturated fill on a
    # neutral paper, with neutral axes, type, and grid lines.
    return (chroma >= 48) & (high >= 105) & (low <= 175)


def _measure(image: Image.Image) -> tuple[int, int]:
    mask = _filled_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < image.width * image.height * 0.20:
        raise ValueError("filled survey region is missing or materially erased")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < image.width * 0.68 or height < image.height * 0.42:
        raise ValueError("filled region has implausible extent")
    if width % 2:
        raise ValueError("filled region cannot be split into equal pixel widths")

    sub = mask[:, x0 : x1 + 1]
    present = sub.any(axis=0)
    if not bool(present.all()):
        raise ValueError("filled region has a severed horizontal span")
    tops = np.argmax(sub, axis=0)
    bottoms = sub.shape[0] - 1 - np.argmax(sub[::-1], axis=0)
    counts = sub.sum(axis=0)
    full_runs = bottoms - tops + 1
    # Clean silhouettes have no internal holes.  A tiny allowance covers the
    # two-pixel neutral step rim; block erasures cause much larger deficits.
    if int(np.max(full_runs - counts)) > 3:
        raise ValueError("filled region contains an internal erasure")
    if int(bottoms.max() - bottoms.min()) > 2:
        raise ValueError("common baseline is incomplete")

    midpoint = width // 2
    left_area = int(sub[:, :midpoint].sum())
    right_area = int(sub[:, midpoint:].sum())
    if left_area == right_area:
        raise ValueError("pixel areas tie at finite resolution")
    return left_area, right_area


def decision_from_image(image: Image.Image) -> str:
    left_area, right_area = _measure(image)
    return "left" if left_area > right_area else "right"
