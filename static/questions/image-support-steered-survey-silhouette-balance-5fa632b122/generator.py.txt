"""Deterministic renderer and analytic arm for the survey-silhouette world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 480
N_PLATEAUS = 8
PLATEAU_WIDTH = 64
SILHOUETTE_WIDTH = N_PLATEAUS * PLATEAU_WIDTH
QUARANTINE_PX = 3.0
BACKGROUND = (247, 245, 239)
INK = (37, 43, 49)
GRID = (211, 211, 205)
PALETTE = (
    (20, 132, 158),
    (38, 132, 96),
    (70, 91, 170),
    (190, 88, 53),
    (151, 74, 145),
    (187, 119, 24),
)


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample paired, cue-balanced scenes from one integer seed.

    Seeds 2k and 2k+1 share palette, placement, total colored area, and the
    multiset of plateau heights.  The latter scene swaps the two four-plateau
    halves, so its answer is the exact opposite without introducing a cheap
    nuisance correlate.
    """

    pair_seed = int(seed) // 2
    rng = random.Random(0x5A17_2026 + pair_seed * 1_000_003)
    bases = [rng.randint(308, 326) for _ in range(4)]

    # One pair in six is deliberately close to the raster boundary and is
    # quarantined; the others have a comfortable mean-height gap.
    if pair_seed % 6 == 0:
        target_mean_gap = 2
    else:
        target_mean_gap = (10, 14, 18, 22, 26)[pair_seed % 5]

    # Mixed-sign contributions force integration.  In the canonical scene two
    # left columns are lower than their right partners and two are higher, while
    # their signed sum has the requested positive margin.
    neg_a = rng.randint(3, 8)
    neg_b = rng.randint(4, 9)
    positive_total = 2 * target_mean_gap + neg_a + neg_b
    pos_a = positive_total // 2 + rng.randint(-2, 2)
    pos_b = positive_total - pos_a
    deltas = [-neg_a, -neg_b, pos_a, pos_b]
    rng.shuffle(deltas)

    canonical_left = [base + delta for base, delta in zip(bases, deltas, strict=True)]
    canonical_right = [base - delta for base, delta in zip(bases, deltas, strict=True)]
    heights = (
        canonical_left + canonical_right
        if int(seed) % 2 == 0
        else canonical_right + canonical_left
    )
    return {
        "heights": heights,
        "fill_rgb": list(PALETTE[pair_seed % len(PALETTE)]),
        "x0": 48 + 4 * (pair_seed % 5),
        "baseline_y": 420 + 2 * (pair_seed % 4),
    }


def _half_sums(scene: dict[str, Any]) -> tuple[int, int]:
    heights = [int(value) for value in scene["heights"]]
    if len(heights) != N_PLATEAUS or any(value <= 0 for value in heights):
        raise ValueError("heights must contain eight positive plateau heights")
    return sum(heights[:4]), sum(heights[4:])


def analytic_gold(scene: dict[str, Any]) -> str:
    left, right = _half_sums(scene)
    if left == right:
        raise ValueError("equal integrated heights have no unique answer")
    return "left" if left > right else "right"


def margin(scene: dict[str, Any]) -> float:
    """Runner-up margin: absolute difference of the two mean heights, in px."""

    left, right = _half_sums(scene)
    return abs(left - right) / 4.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The representation is injective over declared fields; ordinary palette
    # and translation invariances change pixels and are tested as metamorphisms.
    return []


def render(scene: dict[str, Any]) -> Image.Image:
    heights = [int(value) for value in scene["heights"]]
    fill = tuple(int(value) for value in scene["fill_rgb"])
    x0 = int(scene["x0"])
    baseline_y = int(scene["baseline_y"])
    if len(heights) != N_PLATEAUS:
        raise ValueError("renderer requires eight plateau heights")
    if not all(0 <= channel <= 255 for channel in fill):
        raise ValueError("fill_rgb is outside RGB range")
    if x0 < 28 or x0 + SILHOUETTE_WIDTH >= WIDTH - 28:
        raise ValueError("silhouette does not fit horizontally")
    if baseline_y >= HEIGHT - 34 or baseline_y - max(heights) < 48:
        raise ValueError("silhouette does not fit vertically")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Neutral plotting guides are deliberately behind the colored evidence.
    for y in range(90, baseline_y, 55):
        draw.line((x0 - 10, y, x0 + SILHOUETTE_WIDTH + 10, y), fill=GRID, width=1)
    draw.line((x0, baseline_y, x0 + SILHOUETTE_WIDTH - 1, baseline_y), fill=INK, width=3)

    # Adjacent rectangles share their baseline and edges, producing one thick,
    # connected area rather than separate bars.
    tops: list[int] = []
    for index, height in enumerate(heights):
        left = x0 + index * PLATEAU_WIDTH
        right = left + PLATEAU_WIDTH - 1
        top = baseline_y - height
        tops.append(top)
        draw.rectangle((left, top, right, baseline_y - 2), fill=fill)

    # A dark stepped rim makes the measured upper boundary explicit while
    # leaving the colored region connected below every step.
    for index, top in enumerate(tops):
        left = x0 + index * PLATEAU_WIDTH
        right = left + PLATEAU_WIDTH - 1
        draw.line((left, top, right, top), fill=INK, width=2)
        if index:
            draw.line((left, tops[index - 1], left, top), fill=INK, width=2)

    # The midpoint is visibly marked outside the colored component; it does not
    # sever the one connected evidence region.
    midpoint = x0 + SILHOUETTE_WIDTH // 2
    draw.line((midpoint, baseline_y + 2, midpoint, baseline_y + 18), fill=INK, width=3)
    draw.polygon(
        ((midpoint - 7, baseline_y + 19), (midpoint + 7, baseline_y + 19), (midpoint, baseline_y + 27)),
        fill=INK,
    )

    title_font = _font(20, bold=True)
    label_font = _font(16, bold=True)
    small_font = _font(13)
    draw.text((x0, 18), "SURVEY SILHOUETTE", font=title_font, fill=INK)
    draw.text((x0 + 72, 49), "LEFT HALF", font=label_font, fill=INK)
    draw.text((midpoint + 72, 49), "RIGHT HALF", font=label_font, fill=INK)
    draw.text((x0, baseline_y + 29), "common baseline", font=small_font, fill=INK)
    draw.text((midpoint + 12, baseline_y + 12), "equal-width midpoint", font=small_font, fill=INK)
    return image
