"""Deterministic renderer and analytic gold for an inversion-parity search."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

CANVAS_SIZE = (960, 420)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (16, 18, 244, 402),
    (252, 18, 480, 402),
    (488, 18, 716, 402),
    (724, 18, 952, 402),
)
BACKGROUND = (252, 252, 249)
PANEL_FILL = (246, 246, 241)
PANEL_EDGE = (188, 188, 179)
LABEL_INK = (112, 112, 105)
DOT_INK = (22, 27, 34)
DOT_RADIUS = 15
X_RANKS = (46, 92, 138, 184)
Y_RANKS = (110, 164, 218, 272)


def _inversion_count(dots: list[dict]) -> int:
    """Count lower-before-higher pairs after sorting dots from left to right."""

    ordered = sorted(dots, key=lambda dot: (int(dot["x"]), int(dot["y"])))
    y_order = sorted(range(len(ordered)), key=lambda index: (int(ordered[index]["y"]), index))
    y_ranks = [0] * len(ordered)
    for rank, index in enumerate(y_order):
        y_ranks[index] = rank
    inversions = 0
    for left in range(len(y_ranks)):
        for right in range(left + 1, len(y_ranks)):
            if y_ranks[left] > y_ranks[right]:
                inversions += 1
    return inversions


def _panel_parity(panel: dict) -> int:
    dots = panel.get("dots")
    if not isinstance(dots, list) or len(dots) != 4:
        raise ValueError("each panel must contain four dots")
    xs = [int(dot["x"]) for dot in dots]
    ys = [int(dot["y"]) for dot in dots]
    if len(set(xs)) != 4 or len(set(ys)) != 4:
        raise ValueError("dot ranks must be distinct")
    if sorted(xs) != list(X_RANKS) or sorted(ys) != list(Y_RANKS):
        raise ValueError("unexpected dot rank coordinates")
    return _inversion_count(dots) % 2


def _parities(scene: dict) -> list[int]:
    panels = scene["panels"]
    if not isinstance(panels, list) or len(panels) != 4:
        raise ValueError("scene must contain four panels")
    return [_panel_parity(panel) for panel in panels]


def sample_scene(seed: int) -> dict:
    """Sample one even permutation and three odd permutations."""

    rng = random.Random(seed * 1009 + 41)
    permutations = list(itertools.permutations(range(4)))

    def parity(permutation: tuple[int, ...]) -> int:
        return sum(
            permutation[left] > permutation[right]
            for left in range(4)
            for right in range(left + 1, 4)
        ) % 2

    evens = [permutation for permutation in permutations if parity(permutation) == 0]
    odds = [permutation for permutation in permutations if parity(permutation) == 1]
    target_slot = seed % 4
    target = evens[rng.randrange(len(evens))]
    distractors = rng.sample(odds, 3)
    chosen = []
    distractor_index = 0
    for panel_index, label in enumerate(LABELS):
        permutation = target if panel_index == target_slot else distractors[distractor_index]
        if panel_index != target_slot:
            distractor_index += 1
        # A common reflection changes the visible ordering but preserves parity
        # for four ranks, adding a benign scene-level nuisance without a hidden
        # label-bearing field.
        x_ranks = list(X_RANKS)
        y_ranks = list(Y_RANKS)
        if rng.randrange(2):
            x_ranks.reverse()
        if rng.randrange(2):
            y_ranks.reverse()
        dots = [
            {"x": x_ranks[x_rank], "y": y_ranks[y_rank]}
            for x_rank, y_rank in enumerate(permutation)
        ]
        chosen.append({"label": label, "dots": dots})
    scene = {"panels": chosen}
    if _parities(scene).count(0) != 1:
        raise AssertionError("sampler failed to create one even permutation")
    return scene


def render(scene: dict) -> Image.Image:
    """Render four equal cards and the four dark dots in each card."""

    image = Image.new("RGB", CANVAS_SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for panel, (x0, y0, x1, y1) in zip(scene["panels"], PANEL_BOXES, strict=True):
        draw.rounded_rectangle(
            (x0, y0, x1 - 1, y1 - 1),
            radius=12,
            fill=PANEL_FILL,
            outline=PANEL_EDGE,
            width=2,
        )
        draw.text((x0 + 12, y0 + 12), str(panel["label"]), font=font, fill=LABEL_INK)
        for dot in panel["dots"]:
            cx = x0 + int(dot["x"])
            cy = y0 + int(dot["y"])
            draw.ellipse(
                (cx - DOT_RADIUS, cy - DOT_RADIUS, cx + DOT_RADIUS, cy + DOT_RADIUS),
                fill=DOT_INK,
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the label of the unique panel with even inversion parity."""

    panels = scene["panels"]
    parities: list[int] = []
    for panel in panels:
        dots = panel["dots"]
        ordered = sorted(dots, key=lambda dot: (int(dot["x"]), int(dot["y"])))
        y_order = sorted(
            range(len(ordered)),
            key=lambda index: (int(ordered[index]["y"]), index),
        )
        y_ranks = [0] * len(ordered)
        for rank, index in enumerate(y_order):
            y_ranks[index] = rank
        inversions = sum(
            y_ranks[left] > y_ranks[right]
            for left in range(4)
            for right in range(left + 1, 4)
        )
        parities.append(inversions % 2)
    if parities.count(0) != 1:
        raise ValueError(f"expected one even panel, received parities {parities}")
    return str(panels[parities.index(0)]["label"])


def margin(scene: dict) -> float:
    """Return the score gap: even panels score one and odd panels score zero."""

    parities = _parities(scene)
    scores = sorted((1 if parity == 0 else 0 for parity in parities), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    """Quarantine tied, malformed, or non-separated rank layouts."""

    try:
        panels = scene["panels"]
        if len(panels) != 4 or _parities(scene).count(0) != 1:
            return True
        for panel in panels:
            dots = panel["dots"]
            if min(
                abs(int(left["x"]) - int(right["x"]))
                for left, right in itertools.combinations(dots, 2)
            ) < 20:
                return True
            if min(
                abs(int(left["y"]) - int(right["y"]))
                for left, right in itertools.combinations(dots, 2)
            ) < 20:
                return True
    except (KeyError, TypeError, ValueError):
        return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No hidden aliases are declared; all label-bearing fields are visible."""

    return []
