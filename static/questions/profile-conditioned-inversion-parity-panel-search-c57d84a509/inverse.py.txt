"""Independent pixel-only inverse for inversion parity."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANVAS_SIZE = (960, 420)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (16, 18, 244, 402),
    (252, 18, 480, 402),
    (488, 18, 716, 402),
    (724, 18, 952, 402),
)


class OracleAmbiguous(ValueError):
    """Raised when the final raster does not expose four rankable dots per panel."""


def _component_pixels(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[list[tuple[int, int]]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            visited[y, x] = True
            queue: deque[tuple[int, int]] = deque([(x, y)])
            pixels: list[tuple[int, int]] = []
            while queue:
                cx, cy = queue.popleft()
                pixels.append((cx, cy))
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if (
                        0 <= nx < width
                        and 0 <= ny < height
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((nx, ny))
            components.append(pixels)
    return components


def _panel_dots(array: np.ndarray, box: tuple[int, int, int, int]) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = box
    crop = array[y0 + 48 : y1 - 24, x0 + 25 : x1 - 20]
    dark = np.max(crop, axis=2) < 80
    components = [pixels for pixels in _component_pixels(dark) if 400 <= len(pixels) <= 1200]
    if len(components) != 4:
        raise OracleAmbiguous(f"expected four dark dot components, found {len(components)}")
    centers = []
    for pixels in components:
        xs = [point[0] for point in pixels]
        ys = [point[1] for point in pixels]
        centers.append((float(sum(xs) / len(xs)), float(sum(ys) / len(ys))))
    x_sorted = sorted(centers)
    if min(x_sorted[index + 1][0] - x_sorted[index][0] for index in range(3)) < 20:
        raise OracleAmbiguous("horizontal dot ranks are too close")
    y_sorted = sorted(centers, key=lambda center: (center[1], center[0]))
    if min(y_sorted[index + 1][1] - y_sorted[index][1] for index in range(3)) < 20:
        raise OracleAmbiguous("vertical dot ranks are too close")
    rank_by_identity = {id(center): rank for rank, center in enumerate(y_sorted)}
    permutation = [rank_by_identity[id(center)] for center in x_sorted]
    inversions = sum(
        permutation[left] > permutation[right]
        for left in range(4)
        for right in range(left + 1, 4)
    )
    return [(float(inversions % 2), float(inversions))]


def decision_from_image(image: Image.Image) -> str:
    """Enumerate all panel dot components and select the sole even permutation."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    parities = [_panel_dots(array, box)[0][0] for box in PANEL_BOXES]
    if parities.count(0.0) != 1:
        raise OracleAmbiguous(f"even-parity panel count is ambiguous: {parities}")
    return LABELS[parities.index(0.0)]
