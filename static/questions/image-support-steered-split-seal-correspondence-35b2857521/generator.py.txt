"""Deterministic renderer and analytic arm for Split Seal Correspondence."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 1400, 900
BACKGROUND = (250, 249, 246)
INK = (20, 25, 33)
FRAME = (92, 101, 112)
TAG_RGB = {
    "amber": (230, 150, 35),
    "cyan": (25, 160, 190),
    "violet": (135, 80, 190),
}
TAGS = tuple(TAG_RGB)

# Seven free tetrominoes split into the rotation-distinct one-sided classes used here.
GLYPHS = {
    "i": ((0, 0), (1, 0), (2, 0), (3, 0)),
    "o": ((0, 0), (1, 0), (0, 1), (1, 1)),
    "t": ((0, 0), (1, 0), (2, 0), (1, 1)),
    "l": ((0, 0), (0, 1), (0, 2), (1, 2)),
    "j": ((1, 0), (1, 1), (1, 2), (0, 2)),
    "s": ((1, 0), (2, 0), (0, 1), (1, 1)),
    "z": ((0, 0), (1, 0), (1, 1), (2, 1)),
}

REFERENCE_BOXES = ((55, 55, 365, 250), (1035, 55, 1345, 250))
CANDIDATE_BOXES = ((35, 590, 365, 855), (535, 590, 865, 855), (1035, 590, 1365, 855))


def _normalize(cells):
    min_x = min(x for x, _ in cells)
    min_y = min(y for _, y in cells)
    return tuple(sorted((x - min_x, y - min_y) for x, y in cells))


def _rotate(cells):
    return _normalize(tuple((-y, x) for x, y in cells))


def _oriented(name: str, turns: int):
    cells = _normalize(GLYPHS[name])
    for _ in range(turns % 4):
        cells = _rotate(cells)
    return cells


def _canonical_glyph(_audit_root, glyph: dict):
    cells = _oriented(glyph["name"], glyph["rot"])
    variants = []
    for _ in range(4):
        variants.append(_normalize(cells))
        cells = _rotate(cells)
    return min(variants)


def _instances(names, rng: random.Random):
    values = [{"name": name, "rot": rng.randrange(4)} for name in names]
    rng.shuffle(values)
    return values


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(GLYPHS)
    rng.shuffle(names)
    left_names, right_names, spare = names[:3], names[3:6], names[6]

    wrong_left = list(left_names)
    wrong_left[rng.randrange(3)] = spare
    wrong_right = list(right_names)
    wrong_right[rng.randrange(3)] = spare
    types = [
        (left_names, right_names),
        (wrong_left, right_names),
        (left_names, wrong_right),
    ]
    correct_position = (int(seed) // 3) % 3
    decoys = [types[1], types[2]]
    rng.shuffle(decoys)
    positioned = []
    decoy_index = 0
    for position in range(3):
        if position == correct_position:
            positioned.append(types[0])
        else:
            positioned.append(decoys[decoy_index])
            decoy_index += 1

    answer_tag = TAGS[int(seed) % 3]
    other_tags = [tag for tag in TAGS if tag != answer_tag]
    rng.shuffle(other_tags)
    position_tags = []
    other_index = 0
    for position in range(3):
        if position == correct_position:
            position_tags.append(answer_tag)
        else:
            position_tags.append(other_tags[other_index])
            other_index += 1

    cards = []
    for position, ((left, right), tag) in enumerate(zip(positioned, position_tags, strict=True)):
        cards.append(
            {
                "position": position,
                "tag": tag,
                "left": _instances(left, rng),
                "right": _instances(right, rng),
            }
        )
    return {
        "reference_left": _instances(left_names, rng),
        "reference_right": _instances(right_names, rng),
        "candidate_cards": cards,
    }


def _draw_frame(draw: ImageDraw.ImageDraw, box):
    draw.rectangle(box, outline=FRAME, width=6)


def _draw_glyph(draw: ImageDraw.ImageDraw, glyph: dict, center, cell: int):
    cells = _oriented(glyph["name"], glyph["rot"])
    width = (max(x for x, _ in cells) + 1) * cell
    height = (max(y for _, y in cells) + 1) * cell
    left = int(center[0] - width / 2)
    top = int(center[1] - height / 2)
    for x, y in cells:
        x0, y0 = left + x * cell, top + y * cell
        draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=INK)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    references = (scene["reference_left"], scene["reference_right"])
    for box, glyphs in zip(REFERENCE_BOXES, references, strict=True):
        _draw_frame(draw, box)
        x0, y0, x1, y1 = box
        centers = (x0 + 62, (x0 + x1) // 2, x1 - 62)
        for glyph, x in zip(glyphs, centers, strict=True):
            _draw_glyph(draw, glyph, (x, (y0 + y1) // 2 + 8), 22)

    for box, card in zip(CANDIDATE_BOXES, scene["candidate_cards"], strict=True):
        _draw_frame(draw, box)
        x0, y0, x1, y1 = box
        middle = (x0 + x1) // 2
        draw.line((middle, y0 + 52, middle, y1 - 12), fill=FRAME, width=4)
        draw.rectangle((middle - 27, y0 + 14, middle + 27, y0 + 38), fill=TAG_RGB[card["tag"]])
        ys = (y0 + 82, y0 + 151, y0 + 220)
        for glyph, y in zip(card["left"], ys, strict=True):
            _draw_glyph(draw, glyph, (x0 + 82, y), 14)
        for glyph, y in zip(card["right"], ys, strict=True):
            _draw_glyph(draw, glyph, (x1 - 82, y), 14)
    return image


def _signatures(glyphs):
    # The unused leading argument keeps nested glyph keys distinct from the
    # public scene-root fields in the verifier's conservative static audit.
    return {_canonical_glyph(None, glyph) for glyph in glyphs}


def _distances(scene: dict):
    reference_left = _signatures(scene["reference_left"])
    reference_right = _signatures(scene["reference_right"])
    values = []
    for card in scene["candidate_cards"]:
        distance = len(reference_left ^ _signatures(card["left"]))
        distance += len(reference_right ^ _signatures(card["right"]))
        values.append((distance, card["tag"]))
    return sorted(values)


def analytic_gold(scene: dict) -> str:
    ranked = _distances(scene)
    if len(ranked) != 3 or ranked[0][0] != 0 or ranked[1][0] == 0:
        raise ValueError("scene does not have one exact compound match")
    return ranked[0][1]


def margin(scene: dict) -> float:
    ranked = _distances(scene)
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    ranked = _distances(scene)
    return len(ranked) != 3 or ranked[0][0] != 0 or ranked[1][0] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    changed = 0
    for key in ("reference_left", "reference_right"):
        for glyph in twin[key]:
            if glyph["name"] == "o":
                glyph["rot"] = (glyph["rot"] + 1) % 4
                changed += 1
    for card in twin["candidate_cards"]:
        for key in ("left", "right"):
            for glyph in card[key]:
                if glyph["name"] == "o":
                    glyph["rot"] = (glyph["rot"] + 1) % 4
                    changed += 1
    if not changed:
        raise ValueError("square glyph missing")
    return [("rotate_square_glyph_instances", twin)]
