"""Independent inverse using only final RGB pixels and the public rule."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack, points = [(y, x)], []
        seen[y, x] = True
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy-1,cx), (cy+1,cx), (cy,cx-1), (cy,cx+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        groups.append(np.asarray(points, dtype=np.int32))
    return groups


def _node_centers(array: np.ndarray) -> list[tuple[float, float]]:
    r, g, b = (array[..., index].astype(np.int16) for index in range(3))
    mask = (r > 125) & (r > g + 65) & (b > g + 25) & (g < 125)
    groups = [group for group in _components(mask) if 4 <= len(group) <= 40]
    if len(groups) != 51:
        raise ValueError(f"expected 51 visible zigzag vertices, found {len(groups)}")
    return [(float(group[:, 1].mean()), float(group[:, 0].mean())) for group in groups]


def _cap(array: np.ndarray, which: str) -> tuple[float, float]:
    r, g, b = (array[..., index].astype(np.int16) for index in range(3))
    if which == "amber":
        mask = (r > 180) & (g > 105) & (g < 205) & (b < 100) & (r-g > 35)
    else:
        mask = (g > 115) & (b > 105) & (r < 105) & (np.abs(g-b) < 70)
    groups = [group for group in _components(mask) if len(group) >= 45]
    if len(groups) != 1:
        raise ValueError(f"missing or ambiguous {which} cap")
    group = groups[0]
    return float(group[:, 1].mean()), float(group[:, 0].mean())


def _validate_route(array: np.ndarray, centers: list[tuple[float, float]]) -> None:
    dark = np.max(array, axis=2) < 118
    groups = [group for group in _components(dark) if len(group) >= 200]
    if len(groups) != 1:
        raise ValueError("route is erased, split, or not a single connected trace")
    route = groups[0]
    if route[:, 1].max() - route[:, 1].min() < 320:
        raise ValueError("route does not span the declared tooth sequence")
    route_mask = np.zeros(dark.shape, dtype=bool)
    route_mask[route[:, 0], route[:, 1]] = True
    for x, y in centers:
        xi, yi = int(round(x)), int(round(y))
        if not route_mask[max(0, yi-7):yi+8, max(0, xi-7):xi+8].any():
            raise ValueError("a marked vertex is detached from the route")


def _flat(template: tuple[str, ...]) -> tuple[int, ...]:
    return tuple(int(value) for row in template for value in row)


def measure_from_image(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers = _node_centers(array)
    _validate_route(array, centers)
    amber, teal = _cap(array, "amber"), _cap(array, "teal")
    if teal[0] - amber[0] < 320:
        raise ValueError("endpoint order is missing or ambiguous")

    centers.sort(key=lambda point: point[0])
    x_gaps = np.diff([point[0] for point in centers])
    if x_gaps.min() < 1.5 or x_gaps.max() > 11.0:
        raise ValueError("vertex traversal order is ambiguous")

    bits: list[int] = []
    advantages: list[float] = []
    for index in range(25):
        first, apex, last = centers[2*index:2*index+3]
        first_length = math.dist(first, apex)
        second_length = math.dist(apex, last)
        advantage = first_length - second_length
        if abs(advantage) < 1.5:
            raise ValueError("a leg pair is too close to equality")
        bits.append(int(advantage > 0.0))
        advantages.append(abs(advantage))

    scores = sorted((sum(a != b for a, b in zip(bits, _flat(template))), name)
                    for name, template in TEMPLATES.items())
    runner_up_margin = scores[1][0] - scores[0][0]
    if scores[0][0] != 0 or runner_up_margin < 4:
        raise ValueError("reconstructed tile has insufficient template margin")
    return {
        "decision": scores[0][1],
        "runner_up_margin": float(runner_up_margin),
        "minimum_leg_advantage": round(min(advantages), 6),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure_from_image(image)["decision"])
