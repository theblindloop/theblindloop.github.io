"""Renderer and analytic arm for Paired-Leg Zigzag Reveal."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 512
SYMBOLS = ("X", "plus", "T", "L")
TEMPLATES = {
    "X": ("10001", "01010", "00100", "01010", "10001"),
    "plus": ("00100", "00100", "11111", "00100", "00100"),
    "T": ("11111", "00100", "00100", "00100", "00100"),
    "L": ("10000", "10000", "10000", "10000", "11111"),
}

PALETTES = (
    {"bg": (250, 248, 242), "ink": (27, 48, 72), "node": (184, 55, 132)},
    {"bg": (244, 249, 250), "ink": (48, 38, 73), "node": (198, 63, 126)},
)
AMBER = (238, 158, 42)
TEAL = (40, 174, 164)


def _bits(symbol: str) -> tuple[int, ...]:
    return tuple(int(value) for row in TEMPLATES[symbol] for value in row)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5E61A9)
    return {
        "symbol": SYMBOLS[int(seed) % 4],
        "x0": rng.randint(102, 110),
        "y0": rng.randint(245, 267),
        "tooth_span": 12,
        "tooth_height": rng.choice((11, 12, 13)),
        "vertical_flip": bool(rng.getrandbits(1)),
        "palette": rng.randrange(2),
        "stroke": rng.choice((4, 5)),
        "decor_seed": rng.randrange(256),
    }


def _route_points(scene: dict) -> list[tuple[float, float]]:
    x0 = float(scene["x0"])
    y0 = float(scene["y0"])
    span = float(scene["tooth_span"])
    height = float(scene["tooth_height"])
    flip = -1.0 if bool(scene["vertical_flip"]) else 1.0
    points = [(x0, y0)]
    for index, bit in enumerate(_bits(str(scene["symbol"]))):
        left = x0 + index * span
        # Reflection of the apex about the tooth midpoint swaps the two leg
        # lengths while preserving span, height, and their total length.
        apex_dx = span - 3.0 if bit else 3.0
        direction = flip * (-1.0 if index % 2 else 1.0)
        points.append((left + apex_dx, y0 + direction * height))
        points.append((left + span, y0))
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), palette["bg"])
    draw = ImageDraw.Draw(image)

    rng = random.Random((int(scene["decor_seed"]) % 256) + 9109)
    for _ in range(12):
        x = rng.randint(30, 482)
        y = rng.choice((rng.randint(28, 82), rng.randint(430, 484)))
        radius = rng.randint(2, 4)
        draw.ellipse(((x-radius)*scale, (y-radius)*scale,
                      (x+radius)*scale, (y+radius)*scale), fill=(218, 218, 215))

    raw_points = _route_points(scene)
    points = [(round(x*scale), round(y*scale)) for x, y in raw_points]
    width = int(scene["stroke"]) * scale
    draw.line(points, fill=palette["ink"], width=width, joint="curve")

    # Every comparison endpoint is explicitly marked; dark outlines touch the
    # under-route, leaving one validated connected component.
    for x, y in points:
        outer, inner = 4 * scale, 2 * scale
        draw.ellipse((x-outer, y-outer, x+outer, y+outer), fill=palette["ink"])
        draw.ellipse((x-inner, y-inner, x+inner, y+inner), fill=palette["node"])

    start, end = points[0], points[-1]
    sx, sy = start[0] - 17*scale, start[1]
    ex, ey = end[0] + 17*scale, end[1]
    draw.line((sx, sy, start[0], start[1]), fill=palette["ink"], width=width)
    draw.polygon(((sx-8*scale, sy+8*scale), (sx-8*scale, sy-8*scale),
                  (sx+7*scale, sy)), fill=AMBER, outline=palette["ink"])
    draw.line((end[0], end[1], ex, ey), fill=palette["ink"], width=width)
    radius = 7*scale
    draw.ellipse((ex-radius, ey-radius, ex+radius, ey+radius),
                 fill=TEAL, outline=palette["ink"], width=scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def _leg_advantage(scene: dict) -> float:
    span = float(scene["tooth_span"])
    height = float(scene["tooth_height"])
    return math.hypot(height, span - 3.0) - math.hypot(height, 3.0)


def margin(scene: dict) -> float:
    return round(_leg_advantage(scene), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decor_seed"] = int(twin["decor_seed"]) + 256
    return [("low_byte_decoration_alias", twin)]
