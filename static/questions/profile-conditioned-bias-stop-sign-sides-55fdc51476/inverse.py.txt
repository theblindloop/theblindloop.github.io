"""Independent pixel-only inverse oracle for the STOP-sign side-count world.

Receives only the final PNG and re-counts the straight sides of the red polygon
directly from the raster. It imports nothing from the renderer, prompts,
generator, verifier, or analytic-gold code, and accepts no latent scene data.

Method: segment the filled red body by colour and estimate its centroid. Every
vertex of a regular convex n-gon is the unique point at the maximum distance
(the circumradius) from the centroid, so the pixels lying within a small
tolerance of that maximum distance form exactly n compact angular clusters, one
per vertex. Counting those clusters gives the number of sides. The colour
constant below is duplicated deliberately rather than imported, so a
renderer/oracle disagreement is exposed by the certification path instead of
becoming a shared-code blind spot.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_RADIUS_TOL = 1.5      # px tolerance below the maximum distance that still counts as "on a vertex"
_GAP_DEG = 8.0         # angular gap (deg) that separates two distinct vertex clusters


def _red_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    return (r > 190) & (g < 90) & (b < 90)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    mask = _red_mask(arr)
    ys, xs = np.nonzero(mask)
    if ys.size == 0:
        return "0"

    cy = float(ys.mean())
    cx = float(xs.mean())
    radius = np.hypot(xs - cx, ys - cy)
    angles = np.degrees(np.arctan2(ys - cy, xs - cx))

    dmax = float(radius.max())
    vertices = np.nonzero(radius >= dmax - _RADIUS_TOL)[0]
    if vertices.size == 0:
        return "0"
    a = np.sort(np.mod(angles[vertices], 360.0))
    n = a.size

    # A circle splits into `clusters` regions by its `clusters` large gaps; the
    # number of angular clusters equals the number of polygon vertices.
    clusters = 0
    for k in range(n):
        nxt = a[(k + 1) % n] + (360.0 if k == n - 1 else 0.0)
        if nxt - a[k] > _GAP_DEG:
            clusters += 1
    return str(clusters)
