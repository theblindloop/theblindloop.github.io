"""Prior-conflict STOP-sign side-count world renderer.

Latent state z = {n_sides, center_x, center_y, rotation_degrees, size}.

A single filled red regular polygon (the stop-sign body) is drawn on a plain
white canvas. The closed-form gold is the scene's side count:

    side_count = n_sides            (a scene parameter, not a pixel measurement)

The canonical prior — that a STOP sign is an eight-sided octagon — is recorded
only in the question narrative; it is never used as ground truth. The visible
truth is whatever number of sides the polygon actually shows.

Continuous verifiability margin (px): the minimum radial prominence of any
vertex over its neighbouring side-midpoint valley,

    m = size * (1 - cos(pi / n_sides))

which is the radial peak-to-valley contrast of the polygon's angular profile.
It stays comfortably above pixel scale across the sampled parameter range and
governs how clearly the corners (and therefore the side count) are resolvable.

Latent-alias mode: not_applicable. The side count is determined by n_sides, which
is rendered directly (a regular n-gon), so there is no latent variable whose
alias could fuzz the gold; sample_scene returns no pixel-identical twins and
different n_sides always produce a different raster.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image

CANVAS_W = 384
CANVAS_H = 384
BACKGROUND = "white"
POLYGON_COLOR = "red"

N_SIDES_RANGE = (4, 10)      # inclusive
SIZE_RANGE = (95.0, 120.0)   # circumradius, px; always well above pixel scale
CENTER_MARGIN = 140.0        # keep polygon fully inside the canvas

RENDERER_VERSION = "prior_stop_sign_sides-0.1.0"


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(seed)
    n_sides = rng.randint(*N_SIDES_RANGE)
    size = rng.uniform(*SIZE_RANGE)
    cx = rng.uniform(CENTER_MARGIN, CANVAS_W - CENTER_MARGIN)
    cy = rng.uniform(CENTER_MARGIN, CANVAS_H - CENTER_MARGIN)
    rotation = rng.uniform(0.0, 360.0)
    return {
        "n_sides": n_sides,
        "center_x": round(cx, 3),
        "center_y": round(cy, 3),
        "rotation_degrees": round(rotation, 3),
        "size": round(size, 3),
    }


def _vertices(scene: dict[str, Any]) -> list[tuple[float, float]]:
    n = int(scene["n_sides"])
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    size = float(scene["size"])
    rot = math.radians(float(scene["rotation_degrees"]))
    return [
        (cx + size * math.cos(rot + 2.0 * math.pi * k / n),
         cy + size * math.sin(rot + 2.0 * math.pi * k / n))
        for k in range(n)
    ]


def render(scene: dict[str, Any]) -> Image.Image:
    n = int(scene["n_sides"])
    if n < 3:
        raise ValueError("a polygon needs at least three sides")
    scale = 2
    big_w = CANVAS_W * scale
    big_h = CANVAS_H * scale
    big = Image.new("RGB", (big_w, big_h), BACKGROUND)
    from PIL import ImageDraw

    draw = ImageDraw.Draw(big)
    verts = [(x * scale, y * scale) for (x, y) in _vertices(scene)]
    draw.polygon(verts, fill=POLYGON_COLOR)
    return big.resize((CANVAS_W, CANVAS_H), Image.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["n_sides"]))


def margin(scene: dict[str, Any]) -> float:
    n = int(scene["n_sides"])
    size = float(scene["size"])
    return size * (1.0 - math.cos(math.pi / n))


def is_quarantined(scene: dict[str, Any]) -> bool:
    # Vertices become pixel-ambiguous only when the radial peak prominence drops
    # below a small finite-resolution floor. Well inside the sampled range the
    # margin is comfortably above this threshold, so quarantine is rare.
    return margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # not_applicable: the gold is pinned by the rendered n_sides; there is no
    # declared pixel-identical alias to exercise.
    return []
