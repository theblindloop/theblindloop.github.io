"""Independent pixel-only inverse arm; this module sees only a PIL raster."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y0, x0 in np.argwhere(mask):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        xs: list[int] = []
        ys: list[int] = []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < h and 0 <= xx < w and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        result.append((len(xs), float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)))
    return result


def _fit_x_from_y(points: list[tuple[float, float]]) -> tuple[float, float]:
    arr = np.asarray(points, dtype=np.float64)
    y = arr[:, 1]
    x = arr[:, 0]
    design = np.column_stack([y, np.ones(len(y))])
    (a, b), *_ = np.linalg.lstsq(design, x, rcond=None)
    return float(a), float(b)


def _measure(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Rail ink is dark in all channels. Saturated candidate colors and pale
    # distractors/background cannot satisfy this scene-independent predicate.
    dark = np.max(arr, axis=2) < 90
    blobs = [c for c in _components(dark) if 70 <= c[0] <= 210]
    if len(blobs) != 12:
        raise ValueError(f"expected 12 large dark rail dots, found {len(blobs)}")
    pts = [(x, y) for _, x, y in blobs]
    pts.sort(key=lambda p: (p[1], p[0]))
    levels = [sorted(pts[i:i + 2]) for i in range(0, 12, 2)]
    if any(abs(pair[0][1] - pair[1][1]) > 4.0 for pair in levels):
        raise ValueError("rail dots do not form six paired levels")
    left = [pair[0] for pair in levels]
    right = [pair[1] for pair in levels]
    a0, b0 = _fit_x_from_y(left)
    a1, b1 = _fit_x_from_y(right)
    if abs(a0 - a1) < 0.08:
        raise ValueError("rail continuation intersection is unstable")
    meet_y = (b1 - b0) / (a0 - a1)
    meet_x = a0 * meet_y + b0

    red_mask = (arr[:, :, 0] > 150) & (arr[:, :, 1] < 125) & (arr[:, :, 2] < 130)
    blue_mask = (arr[:, :, 2] > 145) & (arr[:, :, 0] < 125) & (arr[:, :, 1] < 155)
    candidates = {}
    for name, mask in (("red", red_mask), ("blue", blue_mask)):
        blobs_color = [c for c in _components(mask) if c[0] > 120]
        if len(blobs_color) != 1:
            raise ValueError(f"expected one {name} marker, found {len(blobs_color)}")
        _, x, y = blobs_color[0]
        candidates[name] = (x, y)
    distances = {name: math.dist((meet_x, meet_y), point) for name, point in candidates.items()}
    ordered = sorted(distances.items(), key=lambda item: item[1])
    return ordered[0][0], ordered[1][1] - ordered[0][1]


def decision_from_image(image: Image.Image) -> str:
    decision, gap = _measure(image)
    if gap < 18.0:
        raise ValueError("ambiguous candidate-distance margin")
    return decision
