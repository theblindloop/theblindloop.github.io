"""Deterministic latent scene and Pillow renderer for dotted rail convergence."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
RED = "#dc343b"
BLUE = "#2767d8"


def _fit_x_from_y(points: list[list[float]]) -> tuple[float, float]:
    """Least-squares x = a*y+b, used only by the analytic scene arm."""
    n = float(len(points))
    my = sum(p[1] for p in points) / n
    mx = sum(p[0] for p in points) / n
    denom = sum((p[1] - my) ** 2 for p in points)
    if denom < 1e-9:
        raise ValueError("chain has no vertical extent")
    a = sum((p[1] - my) * (p[0] - mx) for p in points) / denom
    return a, mx - a * my


def _meeting(rail_points: list[list[list[float]]]) -> tuple[float, float]:
    a0, b0 = _fit_x_from_y(rail_points[0])
    a1, b1 = _fit_x_from_y(rail_points[1])
    if abs(a0 - a1) < 1e-6:
        raise ValueError("parallel chains")
    y = (b1 - b0) / (a0 - a1)
    return a0 * y + b0, y


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    meet_above = bool((seed // 2) % 2 == 0)
    target_color = "red" if seed % 2 == 0 else "blue"
    target_slot = (seed // 4) % 2

    meet_x = rng.uniform(176.0, 336.0)
    meet_y = rng.uniform(60.0, 91.0) if meet_above else rng.uniform(421.0, 452.0)
    ys = [165.0 + 49.0 * i for i in range(6)]
    if not meet_above:
        ys = [347.0 - 49.0 * i for i in range(6)]
    far_y = max(ys, key=lambda y: abs(y - meet_y))
    half_width = rng.uniform(118.0, 154.0)
    rails: list[list[list[float]]] = [[], []]
    for side in (-1.0, 1.0):
        end_x = meet_x + side * half_width
        chain = []
        for y in ys:
            t = (y - meet_y) / (far_y - meet_y)
            chain.append([meet_x + (end_x - meet_x) * t, y])
        rails[0 if side < 0 else 1] = chain

    offset = (34.0, 70.0, 106.0)[seed % 3]
    direction = -1.0 if (seed // 7) % 2 == 0 else 1.0
    distract_x = meet_x + direction * offset
    if not 35.0 <= distract_x <= WIDTH - 35.0:
        distract_x = meet_x - direction * offset
    positions = [[meet_x, meet_y], [distract_x, meet_y]]
    if target_slot:
        positions.reverse()
    colors = [target_color, "blue" if target_color == "red" else "red"]
    if target_slot:
        colors.reverse()

    noise = []
    for _ in range(8):
        for _attempt in range(100):
            p = [rng.uniform(30, WIDTH - 30), rng.uniform(125, 387)]
            if all(math.dist(p, q) > 18 for chain in rails for q in chain):
                noise.append(p)
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": rng.choice(["#ffffff", "#fffdf7", "#f8fbff"]),
        "ink": rng.choice(["#171923", "#241a2b", "#10252a"]),
        "dot_radius": rng.choice([6, 7]),
        "rail_points": rails,
        "noise_points": noise,
        "candidate_points": positions,
        "candidate_colors": colors,
    }


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new(
        "RGB", (int(scene["width"]) * scale, int(scene["height"]) * scale), scene["background"]
    )
    draw = ImageDraw.Draw(image)
    radius = float(scene["dot_radius"]) * scale
    for chain in scene["rail_points"]:
        for x, y in chain:
            draw.ellipse(
                ((x * scale - radius, y * scale - radius),
                 (x * scale + radius, y * scale + radius)),
                fill=scene["ink"],
            )
    # Tiny pale-gray squares are explicitly non-rail clutter and cannot pass the
    # dark-component area filter used by the independent inverse arm.
    for x, y in scene["noise_points"]:
        r = 2.5 * scale
        draw.rectangle((x * scale - r, y * scale - r, x * scale + r, y * scale + r), fill="#aeb6c2")
    palette = {"red": RED, "blue": BLUE}
    for (x, y), color in zip(scene["candidate_points"], scene["candidate_colors"]):
        r = 10.0 * scale
        draw.ellipse(
            ((x * scale - r, y * scale - r), (x * scale + r, y * scale + r)),
            fill=palette[color], outline="#ffffff", width=2 * scale,
        )
    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    meeting = _meeting(scene["rail_points"])
    distances = [math.dist(meeting, p) for p in scene["candidate_points"]]
    winner = 0 if distances[0] < distances[1] else 1
    return scene["candidate_colors"][winner]


def margin(scene: dict) -> float:
    meeting = _meeting(scene["rail_points"])
    distances = sorted(math.dist(meeting, p) for p in scene["candidate_points"])
    return distances[1] - distances[0]


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 24.0
    except (ValueError, ZeroDivisionError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["rail_points"] = [list(reversed(chain)) for chain in reversed(twin["rail_points"])]
    twin["noise_points"] = list(reversed(twin["noise_points"]))
    return [("reverse_disjoint_painter_order", twin)]
