"""Latent sampler and raster renderer for the serpentine spur-code world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 768
RUN_LENGTH = 156
ROW_OFFSETS = (-18, -6, 6, 18)
SPUR_DISTANCES = (30, 60, 90, 120)
ANSWER_COLORS = ("red", "green", "blue")
RGB = {
    "red": (214, 60, 60),
    "green": (45, 157, 91),
    "blue": (53, 105, 210),
    "gold": (226, 174, 38),
}
PALETTES = (
    ((28, 34, 40), (246, 247, 242)),
    ((24, 47, 57), (248, 245, 239)),
    ((49, 35, 47), (244, 248, 246)),
)


def _different_code(reference: list[int], flips: Iterable[int]) -> list[int]:
    code = list(reference)
    for index in flips:
        code[index] = 1 - code[index]
    return code


def sample_scene(seed: int) -> dict:
    """Sample a balanced, unambiguous scene from one integer seed."""
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    reference = [rng.randrange(2) for _ in range(4)]
    answer = ANSWER_COLORS[int(seed) % len(ANSWER_COLORS)]
    candidate_colors = list(ANSWER_COLORS)
    rng.shuffle(candidate_colors)
    target_row = candidate_colors.index(answer)

    one_flip = rng.randrange(4)
    two_flips = rng.sample([i for i in range(4) if i != one_flip], 2)
    distractors = [_different_code(reference, [one_flip]), _different_code(reference, two_flips)]
    rng.shuffle(distractors)
    candidate_codes: list[list[int]] = []
    distractor_index = 0
    for row in range(3):
        if row == target_row:
            candidate_codes.append(list(reference))
        else:
            candidate_codes.append(distractors[distractor_index])
            distractor_index += 1

    # Each placement is centered in one 96-pixel grid row. Horizontally the
    # entire cue crosses exactly three adjacent grid columns under v5 ablation.
    x0 = rng.choice((66, 258, 450)) + rng.randint(-3, 3)
    y_center = rng.choice((48, 240, 432, 624))
    ink, background = rng.choice(PALETTES)
    return {
        "reference": reference,
        "candidate_codes": candidate_codes,
        "candidate_colors": candidate_colors,
        "x0": x0,
        "y_center": y_center,
        "ink_color": list(ink),
        "background": list(background),
    }


def _code_distances(scene: dict) -> list[int]:
    reference = scene["reference"]
    return [
        sum(int(a != b) for a, b in zip(reference, code, strict=True))
        for code in scene["candidate_codes"]
    ]


def analytic_gold(scene: dict) -> str:
    distances = _code_distances(scene)
    winner = min(range(3), key=distances.__getitem__)
    return str(scene["candidate_colors"][winner])


def margin(scene: dict) -> float:
    ordered = sorted(_code_distances(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    distances = _code_distances(scene)
    return distances.count(0) != 1 or sorted(distances)[1] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_run_spurs(
    draw: ImageDraw.ImageDraw,
    *,
    start_x: int,
    y: int,
    direction: int,
    code: list[int],
    ink: tuple[int, int, int],
) -> None:
    for distance, side in zip(SPUR_DISTANCES, code, strict=True):
        x = start_x + direction * distance
        # bit 0 = traversal-left, bit 1 = traversal-right. Image y grows down.
        page_sign = -direction if side == 0 else direction
        draw.line((x, y, x, y + page_sign * 7), fill=ink, width=3)


def render(scene: dict) -> Image.Image:
    background = tuple(int(v) for v in scene["background"])
    ink = tuple(int(v) for v in scene["ink_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    x0 = int(scene["x0"])
    x1 = x0 + RUN_LENGTH
    ys = [int(scene["y_center"]) + offset for offset in ROW_OFFSETS]

    for y in ys:
        draw.line((x0, y, x1, y), fill=ink, width=3)
    # Square U-bends make all four runs one connected ordered strand.
    draw.line((x1, ys[0], x1, ys[1]), fill=ink, width=3)
    draw.line((x0, ys[1], x0, ys[2]), fill=ink, width=3)
    draw.line((x1, ys[2], x1, ys[3]), fill=ink, width=3)

    codes = [scene["reference"], *scene["candidate_codes"]]
    starts = [x0, x1, x0, x1]
    directions = [1, -1, 1, -1]
    for y, start_x, direction, code in zip(ys, starts, directions, codes, strict=True):
        _draw_run_spurs(
            draw, start_x=start_x, y=y, direction=direction, code=code, ink=ink
        )

    bead_names = ["gold", *scene["candidate_colors"]]
    for name, x, y in zip(bead_names, starts, ys, strict=True):
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=RGB[name], outline=ink, width=1)
    return image
