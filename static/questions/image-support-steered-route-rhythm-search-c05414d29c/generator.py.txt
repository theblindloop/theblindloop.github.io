"""Deterministic latent scene sampler and Pillow renderer."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 512, 320
BACKGROUND = (250, 249, 245)
BLUE = (32, 104, 184)
INK = (26, 30, 34)
TEXT = (92, 96, 100)
COLORS = {
    "red": (211, 63, 63),
    "gold": (218, 159, 35),
    "violet": (132, 79, 185),
    "green": (45, 145, 91),
}
COLOR_NAMES = tuple(COLORS)
ALL_MOVES = tuple(
    tuple(values)
    for values in itertools.product((-1, 1), repeat=6)
    if sum(values) == 0
)


def _hamming(left: Iterable[int], right: Iterable[int]) -> int:
    return sum(a != b for a, b in zip(left, right))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference = list(ALL_MOVES[rng.randrange(len(ALL_MOVES))])
    alternatives = [list(item) for item in ALL_MOVES if list(item) != reference]
    rng.shuffle(alternatives)

    # Ensure structured near/far distractors while keeping the target unique.
    chosen = []
    for desired in (2, 4, 6):
        pool = [item for item in alternatives if _hamming(item, reference) == desired]
        if not pool:
            pool = [item for item in alternatives if item not in chosen]
        item = pool[rng.randrange(len(pool))]
        chosen.append(item)

    target_index = int(seed) % 4
    candidates = []
    distractor_index = 0
    for index in range(4):
        if index == target_index:
            candidates.append(reference[:])
        else:
            candidates.append(chosen[distractor_index])
            distractor_index += 1

    gate_colors = list(COLOR_NAMES)
    rng.shuffle(gate_colors)
    return {
        "reference_moves": reference,
        "candidate_moves": candidates,
        "gate_colors": gate_colors,
        "start_level": 218 + (int(seed) % 3 - 1) * 2,
    }


def _points(start_x: int, start_y: int, moves: list[int]) -> list[tuple[int, int]]:
    points = [(start_x, start_y)]
    x, y = start_x, start_y
    for move in moves:
        x += 8
        y += int(move) * 7
        points.append((x, y))
    return points


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    reference = [int(v) for v in scene["reference_moves"]]
    candidates = [[int(v) for v in item] for item in scene["candidate_moves"]]
    gate_colors = [str(v) for v in scene["gate_colors"]]
    bases = [int(scene["start_level"]), 188, 158, 128, 98]
    starts = [132, 192, 252, 312, 372]

    # Pale framing is deliberately non-decisive and remains outside oracle colors.
    draw.rounded_rectangle((112, 62, 440, 246), radius=18, fill=(240, 239, 234), outline=(213, 212, 206), width=2)
    draw.text((128, 230), "SAMPLE", fill=TEXT, font=font)
    draw.text((270, 230), "MATCH THE BEND RHYTHM", fill=TEXT, font=font)

    all_patterns = [reference, *candidates]
    all_points = [_points(x, y, moves) for x, y, moves in zip(starts, bases, all_patterns)]

    # One connected blue route: five patterned sections joined diagonally.
    for index, points in enumerate(all_points):
        draw.line(points, fill=BLUE, width=5, joint="curve")
        if index + 1 < len(all_points):
            draw.line((points[-1], all_points[index + 1][0]), fill=BLUE, width=5)

    # Two black sample gates, attached to the route.
    ref_start = all_points[0][0]
    ref_end = all_points[0][-1]
    for x, y in (ref_start, ref_end):
        draw.rectangle((x - 2, y - 12, x + 2, y + 12), fill=INK)

    # Four equal-size colored address gates, each attached at a candidate start.
    for name, points in zip(gate_colors, all_points[1:]):
        x, y = points[0]
        draw.rectangle((x - 3, y - 13, x + 3, y + 13), fill=(232, 231, 226))
        draw.rectangle((x - 2, y - 11, x + 2, y + 11), fill=COLORS[name])

    return image


def _mismatches(scene: dict) -> list[int]:
    reference = scene["reference_moves"]
    return [_hamming(reference, candidate) for candidate in scene["candidate_moves"]]


def analytic_gold(scene: dict) -> str:
    mismatches = _mismatches(scene)
    winners = [index for index, value in enumerate(mismatches) if value == 0]
    if len(winners) != 1:
        raise ValueError("scene has no unique exact route-rhythm match")
    return str(scene["gate_colors"][winners[0]])


def margin(scene: dict) -> float:
    ordered = sorted(_mismatches(scene))
    return float(ordered[1] - ordered[0])


def is_quarantined(scene: dict) -> bool:
    mismatches = _mismatches(scene)
    return mismatches.count(0) != 1 or sorted(mismatches)[1] < 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
