"""Independent image-only inverse arm for the route-rhythm world."""

from __future__ import annotations

import numpy as np
from PIL import Image

BLUE = np.array((32, 104, 184), dtype=np.uint8)
INK = np.array((26, 30, 34), dtype=np.uint8)
COLORS = {
    "red": np.array((211, 63, 63), dtype=np.uint8),
    "gold": np.array((218, 159, 35), dtype=np.uint8),
    "violet": np.array((132, 79, 185), dtype=np.uint8),
    "green": np.array((45, 145, 91), dtype=np.uint8),
}


def _column_groups(mask: np.ndarray, minimum_pixels: int) -> list[tuple[int, int, float]]:
    counts = mask.sum(axis=0)
    columns = np.flatnonzero(counts >= minimum_pixels)
    if not len(columns):
        return []
    groups = []
    start = previous = int(columns[0])
    for value in columns[1:]:
        value = int(value)
        if value > previous + 1:
            region = mask[:, start : previous + 1]
            ys, xs = np.nonzero(region)
            groups.append((start, previous, float(np.median(ys))))
            start = value
        previous = value
    region = mask[:, start : previous + 1]
    ys, xs = np.nonzero(region)
    groups.append((start, previous, float(np.median(ys))))
    return groups


def _marker(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("a colored candidate gate is missing")
    if int(xs.max() - xs.min()) > 7 or int(ys.max() - ys.min()) < 16:
        raise ValueError("a colored candidate gate is malformed")
    return float(np.median(xs)), float(np.median(ys))


def _route_word(blue: np.ndarray, start_x: float, width: float) -> tuple[int, ...]:
    moves = []
    step = width / 6.0
    for index in range(6):
        left = int(round(start_x + index * step + 3))
        right = int(round(start_x + (index + 1) * step - 3))
        if right - left < 2:
            raise ValueError("route section is too narrow")
        xs = []
        ys = []
        for x in range(left, right + 1):
            column_ys = np.flatnonzero(blue[:, x])
            if len(column_ys):
                xs.append(float(x))
                ys.append(float(np.median(column_ys)))
        if len(xs) < 2:
            raise ValueError("route evidence is interrupted")
        slope = float(np.polyfit(np.asarray(xs), np.asarray(ys), 1)[0])
        if abs(slope) < 0.3:
            raise ValueError("route move has no clear direction")
        moves.append(1 if slope > 0 else -1)
    return tuple(moves)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    blue = np.all(array == BLUE, axis=2)
    if int(blue.sum()) < 700:
        raise ValueError("connected blue route is missing or materially erased")

    black = np.all(array == INK, axis=2)
    black_groups = _column_groups(black, minimum_pixels=12)
    if len(black_groups) != 2:
        raise ValueError("sample gates are missing or ambiguous")
    black_x = sorted((start + end) / 2.0 for start, end, _ in black_groups)
    section_width = black_x[1] - black_x[0]
    if not 40.0 <= section_width <= 56.0:
        raise ValueError("sample section width is malformed")
    reference = _route_word(blue, black_x[0], section_width)

    markers = []
    for name, rgb in COLORS.items():
        x, y = _marker(np.all(array == rgb, axis=2))
        markers.append((x, name))
    markers.sort()
    if any(markers[i + 1][0] - markers[i][0] < section_width + 5 for i in range(3)):
        raise ValueError("candidate gates are not distinctly ordered")

    scores = []
    for x, name in markers:
        word = _route_word(blue, x, section_width)
        scores.append((sum(a != b for a, b in zip(reference, word)), name))
    scores.sort()
    if scores[0][0] != 0 or scores[1][0] - scores[0][0] < 1:
        raise ValueError("no unique exact match with runner-up margin")
    return scores[0][1]
