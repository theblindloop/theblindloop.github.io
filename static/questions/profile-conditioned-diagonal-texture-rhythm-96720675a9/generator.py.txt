"""Renderer and analytic model for diagonal texture-energy rhythm."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


RENDERER_VERSION = "diagonal_texture_rhythm-0.1.0"
COLORS = ("red", "blue", "green")
PALETTE = {
    "red": (214, 62, 70),
    "blue": (52, 100, 208),
    "green": (44, 156, 96),
}
WIDTH = 720
HEIGHT = 440
PANEL_HALF = 78.0
DOT_RADIUS = 2.4
MARKS_PER_FIELD = 48
BIN_COUNT = 8
FOURIER_HARMONIC = 4
QUARANTINE_MARGIN = 0.16
SUPERSAMPLE = 3

_SLOT_CENTERS = ((140.0, 220.0), (360.0, 220.0), (580.0, 220.0))
_BACKGROUND = (248, 245, 239)
_PANEL_FILL = (253, 250, 245)
_PANEL_EDGE = (224, 216, 203)


def _band_count(band: int, phase: int) -> int:
    # A 5:1 high/low ratio makes the global rhythm stable with only 48 marks,
    # while every field still has exactly the same total ink.
    return 10 if (band + phase) % 2 == 0 else 2


def _band_interval(band: int) -> tuple[float, float]:
    low = -1.0 + 2.0 * band / BIN_COUNT
    high = -1.0 + 2.0 * (band + 1) / BIN_COUNT
    return low, high


def _candidate_norm_point(
    rng: random.Random, kind: str, band: int
) -> tuple[float, float]:
    """Draw a point from one of three equal-mass band arrangements."""

    low, high = _band_interval(band)
    # Avoid the exact band edges so rasterized dots do not create a boundary cue.
    band_value = rng.uniform(low + 0.025, high - 0.025)
    other = rng.uniform(-0.94, 0.94)
    if kind == "horizontal":
        return other, band_value
    if kind == "vertical":
        return band_value, other
    # Diagonal bands use u=(x+y)/2. The admissible v=(x-y)/2 interval keeps
    # both coordinates in the square [-1, 1].
    if kind != "diagonal":
        raise ValueError(f"unknown texture kind {kind!r}")
    u = band_value
    v_limit = max(0.04, 1.0 - abs(u))
    v = rng.uniform(-0.94 * v_limit, 0.94 * v_limit)
    return u + v, u - v


def _sample_cloud(
    rng: random.Random,
    kind: str,
    phase: int,
    sign: int,
    center: tuple[float, float],
) -> list[list[float]]:
    """Sample an equal-count cloud with a protected minimum dot separation."""

    points: list[tuple[float, float]] = []
    # The rasterized dot core remains disconnected at this spacing, while the
    # lower floor keeps narrow diagonal bands feasible for every stress seed.
    min_distance = 6.0 / PANEL_HALF
    for band in range(BIN_COUNT):
        for _ in range(_band_count(band, phase)):
            for _attempt in range(2500):
                x, y = _candidate_norm_point(rng, kind, band)
                if kind == "diagonal" and sign < 0:
                    # The same density rhythm may run along the other diagonal.
                    x, y = x, -y
                if all(math.hypot(x - px, y - py) >= min_distance for px, py in points):
                    points.append((x, y))
                    break
            else:
                raise RuntimeError("could not place separated texture marks")
    cx, cy = center
    return [[cx + x * PANEL_HALF, cy + y * PANEL_HALF] for x, y in points]


def sample_scene(seed: int) -> dict:
    """Sample one balanced three-field texture scene from an integer seed."""

    rng = random.Random(int(seed))
    target_color = rng.choice(COLORS)
    decoy_kinds = ["horizontal", "vertical"]
    rng.shuffle(decoy_kinds)
    kinds = {target_color: "diagonal"}
    for color, kind in zip((c for c in COLORS if c != target_color), decoy_kinds):
        kinds[color] = kind
    slots = list(_SLOT_CENTERS)
    rng.shuffle(slots)

    fields: list[dict] = []
    for color, slot in zip(COLORS, slots):
        center = (slot[0] + rng.uniform(-7.0, 7.0), slot[1] + rng.uniform(-6.0, 6.0))
        phase = rng.randrange(2)
        sign = rng.choice((-1, 1))
        kind = kinds[color]
        fields.append(
            {
                "color": color,
                "center": [round(center[0], 5), round(center[1], 5)],
                "kind": kind,
                "phase": phase,
                "sign": sign,
                "points": _sample_cloud(rng, kind, phase, sign, center),
            }
        )
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(_BACKGROUND),
        "dot_radius": DOT_RADIUS,
        "fields": fields,
    }


def render(scene: dict) -> Image.Image:
    """Render only visible panel geometry and colored microtexture."""

    width = int(scene["width"])
    height = int(scene["height"])
    scale = SUPERSAMPLE
    image = Image.new(
        "RGB",
        (width * scale, height * scale),
        tuple(int(value) for value in scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    panel_radius = 12.0 * scale
    for field in scene["fields"]:
        cx, cy = (float(value) for value in field["center"])
        box = (
            (cx - PANEL_HALF) * scale,
            (cy - PANEL_HALF) * scale,
            (cx + PANEL_HALF) * scale,
            (cy + PANEL_HALF) * scale,
        )
        draw.rounded_rectangle(box, radius=panel_radius, fill=_PANEL_FILL, outline=_PANEL_EDGE, width=2 * scale)
        ink = PALETTE[str(field["color"])]
        radius = float(scene["dot_radius"]) * scale
        for x, y in field["points"]:
            draw.ellipse(
                (float(x) * scale - radius, float(y) * scale - radius,
                 float(x) * scale + radius, float(y) * scale + radius),
                fill=ink,
            )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _field_texture_score(points: Iterable[Iterable[float]]) -> float:
    """Measure periodic density energy in the two diagonal projections."""

    raw = [(float(point[0]), float(point[1])) for point in points]
    if len(raw) < 12:
        return 0.0
    min_x = min(point[0] for point in raw)
    max_x = max(point[0] for point in raw)
    min_y = min(point[1] for point in raw)
    max_y = max(point[1] for point in raw)
    center_x = (min_x + max_x) / 2.0
    center_y = (min_y + max_y) / 2.0
    half_extent = max((max_x - min_x) / 2.0, (max_y - min_y) / 2.0, 1.0)
    normalized = [
        ((x - center_x) / half_extent, (y - center_y) / half_extent)
        for x, y in raw
    ]

    def harmonic(values: Iterable[float]) -> float:
        cos_sum = 0.0
        sin_sum = 0.0
        count = 0
        for value in values:
            t = min(1.0, max(0.0, (float(value) + 1.0) / 2.0))
            angle = 2.0 * math.pi * FOURIER_HARMONIC * t
            cos_sum += math.cos(angle)
            sin_sum += math.sin(angle)
            count += 1
        return math.hypot(cos_sum, sin_sum) / max(1, count)

    diagonal_sum = harmonic((x + y) / 2.0 for x, y in normalized)
    diagonal_difference = harmonic((x - y) / 2.0 for x, y in normalized)
    return max(diagonal_sum, diagonal_difference)


def _scores(scene: dict) -> list[tuple[float, str]]:
    return [
        (_field_texture_score(field["points"]), str(field["color"]))
        for field in scene["fields"]
    ]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return max(scores, key=lambda item: (item[0], -COLORS.index(item[1])))[1]


def margin(scene: dict) -> float:
    ordered = sorted((score for score, _color in _scores(scene)), reverse=True)
    if len(ordered) < 2:
        return 0.0
    return round(float(ordered[0] - ordered[1]), 8)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse disjoint field draw order, preserving pixels and the selected color."""

    twin = copy.deepcopy(scene)
    twin["fields"] = list(reversed(twin["fields"]))
    return [("reverse_disjoint_field_order", twin)]
