"""Independent PNG-only inverse for the texture-density question."""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_COLORS = ("red", "blue", "green")
_PALETTE = {
    "red": np.asarray((214, 62, 70), dtype=np.int16),
    "blue": np.asarray((52, 100, 208), dtype=np.int16),
    "green": np.asarray((44, 156, 96), dtype=np.int16),
}
_EXPECTED_MARKS = 48
_HARMONIC = 4
_MIN_DECISION_SCORE = 0.18
_MIN_TOP_GAP = 0.12


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Find solid-color dot components directly in the supplied raster."""

    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    rows, columns = np.nonzero(mask)
    for row, column in zip(rows.tolist(), columns.tolist()):
        if visited[row, column]:
            continue
        stack = [(row, column)]
        visited[row, column] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            current_row, current_column = stack.pop()
            pixels.append((current_row, current_column))
            for delta_row in (-1, 0, 1):
                for delta_column in (-1, 0, 1):
                    if not (delta_row or delta_column):
                        continue
                    next_row = current_row + delta_row
                    next_column = current_column + delta_column
                    if (
                        0 <= next_row < height
                        and 0 <= next_column < width
                        and mask[next_row, next_column]
                        and not visited[next_row, next_column]
                    ):
                        visited[next_row, next_column] = True
                        stack.append((next_row, next_column))
        if 10 <= len(pixels) <= 180:
            found.append(np.asarray(pixels, dtype=np.float64))
    return found


def _centers(rgb: np.ndarray, color: str) -> np.ndarray:
    target = _PALETTE[color]
    distance = np.max(np.abs(rgb.astype(np.int16) - target[None, None, :]), axis=2)
    components = _components(distance <= 28)
    centers = []
    for component in components:
        row, column = np.mean(component, axis=0)
        centers.append((column, row))
    return np.asarray(centers, dtype=np.float64)


def _harmonic(values: np.ndarray) -> float:
    if values.size == 0:
        return 0.0
    t = np.clip((values + 1.0) / 2.0, 0.0, 1.0)
    angle = 2.0 * math.pi * _HARMONIC * t
    return float(math.hypot(float(np.mean(np.cos(angle))), float(np.mean(np.sin(angle)))))


def _texture_score(points: np.ndarray) -> float:
    if points.shape != (_EXPECTED_MARKS, 2):
        return 0.0
    minimum = np.min(points, axis=0)
    maximum = np.max(points, axis=0)
    center = (minimum + maximum) / 2.0
    half_extent = max(float(np.max((maximum - minimum) / 2.0)), 1.0)
    normalized = (points - center[None, :]) / half_extent
    diagonal_sum = _harmonic((normalized[:, 0] + normalized[:, 1]) / 2.0)
    diagonal_difference = _harmonic((normalized[:, 0] - normalized[:, 1]) / 2.0)
    return max(diagonal_sum, diagonal_difference)


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected color using only RGB pixels."""

    if not isinstance(image, Image.Image):
        raise TypeError("decision_from_image expects a PIL image")
    rgb = np.asarray(image.convert("RGB"))
    scores = [(float(_texture_score(_centers(rgb, color))), color) for color in _COLORS]
    scores.sort(reverse=True)
    best_score, best_color = scores[0]
    runner_score = scores[1][0]
    if best_score < _MIN_DECISION_SCORE or best_score - runner_score < _MIN_TOP_GAP:
        return "ambiguous"
    return best_color
