"""Deterministic renderer and analytic arm for the clipped-corner world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
CUT_CORNERS = ("top-right", "top-left", "bottom-right", "bottom-left")
MIN_VISIBLE_CUT_PX = 20.0

_BODY_PALETTE = (
    (39, 105, 172),
    (28, 132, 148),
    (67, 96, 177),
    (35, 137, 108),
)
_BACKGROUNDS = (
    (245, 247, 249),
    (248, 244, 236),
    (238, 245, 242),
    (244, 241, 248),
)
_PAD_PALETTE = (
    (232, 177, 55),
    (221, 158, 43),
    (239, 191, 72),
    (210, 146, 35),
)


def sample_scene(seed: int) -> dict[str, Any]:
    """Return one deterministic member of a four-way matched group.

    A block of four consecutive seeds shares all style and geometry nuisances.
    Only the equal-area triangular cut moves between the four outer corners.
    """

    seed = int(seed)
    group_seed = seed // 4
    rng = random.Random((group_seed * 1_000_003) ^ 0x51C0A7D)
    return {
        "cut_corner": CUT_CORNERS[seed % 4],
        "center_x": round(rng.uniform(242.0, 270.0), 3),
        "center_y": round(rng.uniform(250.0, 266.0), 3),
        "card_width": round(rng.uniform(218.0, 252.0), 3),
        "card_height": round(rng.uniform(304.0, 340.0), 3),
        "cut_depth": round(rng.uniform(28.0, 54.0), 3),
        "body_rgb": list(rng.choice(_BODY_PALETTE)),
        "background_rgb": list(rng.choice(_BACKGROUNDS)),
        "pad_rgb": list(rng.choice(_PAD_PALETTE)),
        "pad_offset_x": round(rng.uniform(-10.0, 10.0), 3),
        "pad_offset_y": round(rng.uniform(-12.0, 12.0), 3),
        "pad_rounding": rng.choice((5, 7, 9)),
    }


def _bounds(scene: dict[str, Any]) -> tuple[float, float, float, float]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    width = float(scene["card_width"])
    height = float(scene["card_height"])
    return cx - width / 2, cy - height / 2, cx + width / 2, cy + height / 2


def _body_vertices(scene: dict[str, Any]) -> list[tuple[float, float]]:
    x0, y0, x1, y1 = _bounds(scene)
    depth = float(scene["cut_depth"])
    corner = str(scene["cut_corner"])
    if corner == "top-left":
        points = [(x0 + depth, y0), (x1, y0), (x1, y1), (x0, y1), (x0, y0 + depth)]
    elif corner == "top-right":
        points = [(x0, y0), (x1 - depth, y0), (x1, y0 + depth), (x1, y1), (x0, y1)]
    elif corner == "bottom-right":
        points = [(x0, y0), (x1, y0), (x1, y1 - depth), (x1 - depth, y1), (x0, y1)]
    elif corner == "bottom-left":
        points = [(x0, y0), (x1, y0), (x1, y1), (x0 + depth, y1), (x0, y1 - depth)]
    else:
        raise ValueError(f"invalid cut_corner {corner!r}")
    return [(x * SUPERSAMPLE, y * SUPERSAMPLE) for x, y in points]


def _contact_boxes(scene: dict[str, Any]) -> list[tuple[float, float, float, float]]:
    """Six centered contacts whose nuisance offsets do not reveal the label."""

    cx = float(scene["center_x"]) + float(scene["pad_offset_x"])
    cy = float(scene["center_y"]) + float(scene["pad_offset_y"])
    width = float(scene["card_width"])
    height = float(scene["card_height"])
    block_w = width * 0.56
    block_h = height * 0.50
    gap_x = width * 0.045
    gap_y = height * 0.032
    pad_w = (block_w - gap_x) / 2
    pad_h = (block_h - 2 * gap_y) / 3
    left = cx - block_w / 2
    top = cy - block_h / 2 + height * 0.035
    boxes = []
    for row in range(3):
        for col in range(2):
            x0 = left + col * (pad_w + gap_x)
            y0 = top + row * (pad_h + gap_y)
            boxes.append((x0, y0, x0 + pad_w, y0 + pad_h))
    return boxes


def render(scene: dict[str, Any]) -> Image.Image:
    """Render an upright card and its visible contact pattern to a final RGB PNG."""

    background = tuple(int(value) for value in scene["background_rgb"])
    body = tuple(int(value) for value in scene["body_rgb"])
    pad = tuple(int(value) for value in scene["pad_rgb"])
    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    vertices = _body_vertices(scene)
    draw.polygon(vertices, fill=body)

    rounding = int(scene["pad_rounding"]) * SUPERSAMPLE
    for index, box in enumerate(_contact_boxes(scene)):
        scaled = tuple(value * SUPERSAMPLE for value in box)
        draw.rounded_rectangle(scaled, radius=rounding, fill=pad)
        # A small inset line gives each contact a chip-like engraved detail. It
        # remains inside the already-foreground card and never changes its outer silhouette.
        if index in (1, 4):
            inset = 7 * SUPERSAMPLE
            x0, y0, x1, y1 = scaled
            draw.line(
                (x0 + inset, (y0 + y1) / 2, x1 - inset, (y0 + y1) / 2),
                fill=(112, 86, 43),
                width=2 * SUPERSAMPLE,
            )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    value = str(scene["cut_corner"])
    if value not in CUT_CORNERS:
        raise ValueError(f"invalid cut_corner {value!r}")
    return value


def margin(scene: dict[str, Any]) -> float:
    """Pixel headroom of the diagonal cut above the public visibility floor."""

    return float(scene["cut_depth"]) - MIN_VISIBLE_CUT_PX


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) <= 0.0:
        return True
    x0, y0, x1, y1 = _bounds(scene)
    return x0 < 12 or y0 < 12 or x1 > WIDTH - 12 or y1 > HEIGHT - 12


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No nontrivial pixel-identical latent aliases are claimed."""

    return []
