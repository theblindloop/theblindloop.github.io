"""Independent raster-only inverse for the SIM-card outer clipped corner.

The inverse estimates the uniform background from the image border, segments
all visible foreground without knowing renderer colors, and compares missing
foreground fractions in four normalized bounding-box corner patches.  It never
imports or receives renderer state, scene data, prompt code, or analytic gold.
"""

from __future__ import annotations

from dataclasses import asdict, dataclass

import numpy as np
from PIL import Image

_CORNERS = ("top-left", "top-right", "bottom-left", "bottom-right")
BACKGROUND_DISTANCE = 32.0
PATCH_FRACTION = 0.17
MIN_FOREGROUND_PIXELS = 35_000
MIN_PATCH_MISSING = 0.20
MIN_RUNNER_UP_MARGIN = 0.16


class PixelOracleError(ValueError):
    """Raised when the raster does not expose one unambiguous clipped corner."""


@dataclass(frozen=True)
class PixelMeasurement:
    decision: str
    missing_fractions: dict[str, float]
    runner_up_margin: float
    foreground_pixel_count: int
    foreground_fill_fraction: float
    body_bbox: tuple[int, int, int, int]

    def to_dict(self) -> dict:
        return asdict(self)


def _foreground_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.shape[0] < 64 or arr.shape[1] < 64:
        raise PixelOracleError("image is too small for normalized corner measurement")
    band = max(3, min(arr.shape[:2]) // 64)
    border = np.concatenate(
        [
            arr[:band, :, :].reshape(-1, 3),
            arr[-band:, :, :].reshape(-1, 3),
            arr[:, :band, :].reshape(-1, 3),
            arr[:, -band:, :].reshape(-1, 3),
        ],
        axis=0,
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(arr - background, axis=2)
    return distance >= BACKGROUND_DISTANCE


def _corner_patches(mask: np.ndarray, x0: int, y0: int, x1: int, y1: int) -> dict[str, np.ndarray]:
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    patch = max(12, int(round(min(width, height) * PATCH_FRACTION)))
    return {
        "top-left": mask[y0 : y0 + patch, x0 : x0 + patch],
        "top-right": mask[y0 : y0 + patch, x1 - patch + 1 : x1 + 1],
        "bottom-left": mask[y1 - patch + 1 : y1 + 1, x0 : x0 + patch],
        "bottom-right": mask[y1 - patch + 1 : y1 + 1, x1 - patch + 1 : x1 + 1],
    }


def measure_from_image(image: Image.Image) -> PixelMeasurement:
    """Return the clipped corner plus an explicit best-versus-runner-up gap."""

    mask = _foreground_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < MIN_FOREGROUND_PIXELS:
        raise PixelOracleError("too little foreground ink to expose a card body")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width < 170 or height < 240:
        raise PixelOracleError("foreground body is too small for corner recovery")
    if x0 <= 3 or y0 <= 3 or x1 >= mask.shape[1] - 4 or y1 >= mask.shape[0] - 4:
        raise PixelOracleError("foreground boundary touches the raster edge")
    aspect = width / height
    if not 0.55 <= aspect <= 0.90:
        raise PixelOracleError("foreground silhouette is not an upright card")
    fill_fraction = float(mask[y0 : y1 + 1, x0 : x1 + 1].mean())
    if not 0.90 <= fill_fraction <= 0.995:
        raise PixelOracleError(
            f"foreground silhouette has implausible fill fraction {fill_fraction:.4f}"
        )

    patches = _corner_patches(mask, x0, y0, x1, y1)
    missing = {name: 1.0 - float(patch.mean()) for name, patch in patches.items()}
    ranked = sorted(((value, name) for name, value in missing.items()), reverse=True)
    best_value, decision = ranked[0]
    second_value = ranked[1][0]
    runner_up_margin = best_value - second_value
    if decision not in _CORNERS:
        raise PixelOracleError("internal corner-label failure")
    if best_value < MIN_PATCH_MISSING or runner_up_margin < MIN_RUNNER_UP_MARGIN:
        raise PixelOracleError(
            "ambiguous clipped corner: "
            f"best_missing={best_value:.4f}, runner_up_margin={runner_up_margin:.4f}"
        )
    return PixelMeasurement(
        decision=decision,
        missing_fractions={key: round(value, 8) for key, value in sorted(missing.items())},
        runner_up_margin=round(runner_up_margin, 8),
        foreground_pixel_count=int(len(xs)),
        foreground_fill_fraction=round(fill_fraction, 8),
        body_bbox=(x0, y0, x1, y1),
    )


def decision_from_image(image: Image.Image) -> str:
    """Select the visibly clipped corner from final image pixels alone."""

    return measure_from_image(image).decision
