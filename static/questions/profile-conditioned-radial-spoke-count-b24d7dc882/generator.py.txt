"""Radial-spoke-count world: how many radial spokes does the dot burst trace?

A set of small dark dots is drawn on a white canvas. The dots are arranged on
straight radial spokes that all emanate from one shared center, so together
they trace a star-burst of spokes: dots along the same spoke lie very close in
direction while consecutive spokes are separated by a much larger empty angular
gap. The global-structure decision is how many such radial spokes the whole
constellation forms (3, 4, 5, or 6).

The renderer owns scene sampling + rasterization + the structural analytic gold
(which derives the spoke count from the latent dot coordinates). The independent
pixel-only inverse arm (``oracle.py``) reconstructs the dot constellation from the
final PNG alone and recomputes the same global spoke count from the recovered
dot centroids.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
CENTER_X = WIDTH // 2
CENTER_Y = HEIGHT // 2
DOT_R = 9
INK = (30, 30, 30)
BACKGROUND = (255, 255, 255)

RADIUS_MIN = 90.0
RADIUS_MAX = 210.0
ANGLE_JIT_DEG = 2.5
VALID_COUNTS = (3, 4, 5, 6)
CLUSTER_THETA_DEG = 25.0
QUARANTINE_MARGIN = 20.0


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    for attempt in range(200):
        candidate = _build(seed + attempt * 1000003)
        if _spoke_structure(candidate["dots"])[1] >= QUARANTINE_MARGIN:
            return candidate
    return _build(seed)


def _build(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    k = int(rng.integers(3, 7))  # 3..6 spokes
    dpr = int(rng.integers(2, 4))  # 2..3 dots per spoke
    rotation = float(rng.uniform(0.0, 2.0 * np.pi))
    base_angles = rotation + np.arange(k) * (2.0 * np.pi / k)

    dots = []
    for a in base_angles:
        radii = np.sort(rng.uniform(RADIUS_MIN, RADIUS_MAX, dpr))
        for r in radii:
            aj = a + float(rng.uniform(-1.0, 1.0)) * np.deg2rad(ANGLE_JIT_DEG)
            dots.append(
                [
                    int(round(CENTER_X + r * np.cos(aj))),
                    int(round(CENTER_Y + r * np.sin(aj))),
                ]
            )
    return {"dots": dots}


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    for (x, y) in scene["dots"]:
        d.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=INK)
    return img


def _dot_angles(dots: list) -> tuple[np.ndarray, tuple[float, float]]:
    """Return (circular angles of dots about the constellation centroid, centroid)."""
    pts = np.asarray(dots, dtype=float)
    center = pts.mean(axis=0)
    rel = pts - center
    angles = np.degrees(np.arctan2(rel[:, 1], rel[:, 0])) % 360.0
    return angles, (float(center[0]), float(center[1]))


def _spoke_structure(dots: list) -> tuple[int, float]:
    """Return (spoke count, min inter-spoke angular gap in degrees).

    Spokes are recovered by clustering dots that are contiguous in angle: a
    circular angular gap larger than ``CLUSTER_THETA_DEG`` separates two
    consecutive spokes, so the number of such large gaps equals the spoke count.
    The margin is the smallest inter-spoke angular gap.
    """
    angles, _ = _dot_angles(dots)
    angles = np.sort(angles)
    n = len(angles)
    if n == 0:
        return 0, 0.0
    diffs = np.roll(angles, -1) - angles
    diffs[-1] += 360.0
    big = diffs > CLUSTER_THETA_DEG
    count = int(big.sum())
    if count == 0:
        count = 1
    inter = diffs[big]
    margin = float(inter.min()) if len(inter) else 0.0
    return count, margin


def analytic_gold(scene: dict) -> str:
    """Structural gold: the number of radial spokes traced by the dots."""
    count, _ = _spoke_structure(scene["dots"])
    return str(count)


def margin(scene: dict) -> float:
    """Decision commitment: the smallest angular gap between two adjacent spokes."""
    _, gap = _spoke_structure(scene["dots"])
    return gap


def is_quarantined(scene: dict) -> bool:
    """Quarantine constellations whose spoke read is angularly ambiguous."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
