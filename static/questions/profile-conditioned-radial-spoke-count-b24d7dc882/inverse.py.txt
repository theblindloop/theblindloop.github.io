"""Independent pixel-only inverse arm for the radial-spoke-count world.

The oracle receives only the final rendered PNG and the public question
contract. It reconstructs the dark dot constellation from the raster alone,
takes every recovered dot centroid as an equal-mass mark, computes the global
angular arrangement around the constellation centroid, and counts how many
radial spokes the dots trace by separating the tiny within-spoke angular gaps
from the conspicuously large between-spoke gaps. It imports only numpy/PIL and
never receives the latent scene or analytic gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
INK_THRESHOLD = 128
CLUSTER_THETA_DEG = 25.0
VALID_COUNTS = (3, 4, 5, 6)


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB")).astype(int)
    return (arr[..., 0] < INK_THRESHOLD) & (
        (arr[..., 1] < INK_THRESHOLD) | (arr[..., 2] < INK_THRESHOLD) | (arr[..., 0] < INK_THRESHOLD)
    )


def _centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    """Recover each connected dark dot's centroid by BFS flood fill."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    centroids: list[tuple[float, float]] = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(y, x)])
                visited[y, x] = True
                sx = 0
                sy = 0
                count = 0
                while queue:
                    cy, cx = queue.popleft()
                    sx += cx
                    sy += cy
                    count += 1
                    for ny in (cy - 1, cy, cy + 1):
                        for nx in (cx - 1, cx, cx + 1):
                            if 0 <= ny < h and 0 <= nx < w:
                                if mask[ny, nx] and not visited[ny, nx]:
                                    visited[ny, nx] = True
                                    queue.append((ny, nx))
                if count > 0:
                    centroids.append((float(sx) / count, float(sy) / count))
    return centroids


def decision_from_image(image: Image.Image) -> str:
    """Image-only inverse: the number of radial spokes traced by the dots."""
    mask = _dark_mask(image)
    centroids = _centroids(mask)
    if len(centroids) < 2:
        return str(1)
    pts = np.asarray(centroids, dtype=float)
    center = pts.mean(axis=0)
    angles = np.degrees(np.arctan2(pts[:, 1] - center[1], pts[:, 0] - center[0])) % 360.0
    angles = np.sort(angles)
    diffs = np.roll(angles, -1) - angles
    diffs[-1] += 360.0
    count = int((diffs > CLUSTER_THETA_DEG).sum())
    if count == 0:
        count = 1
    if count in VALID_COUNTS:
        return str(count)
    return str(count)
