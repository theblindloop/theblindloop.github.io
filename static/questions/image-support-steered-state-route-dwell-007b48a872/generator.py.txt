"""Renderer and analytic arm for the connected state-dwell route world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "state-route-dwell-0.1.0"
CANVAS = (640, 480)
STATION_COUNT = 8
STATE_LABELS = ("A", "B", "C")
PALETTE = {
    "background": (249, 251, 253),
    "route": (36, 61, 84),
    "operation": (216, 100, 58),
    "state_a": (192, 70, 67),
    "state_b": (42, 148, 94),
    "state_c": (67, 106, 190),
    "neutral": (169, 177, 187),
    "text": (29, 40, 52),
}


def _colors(scene: dict[str, Any]) -> dict[str, tuple[int, int, int]]:
    return {
        name: tuple(int(channel) for channel in value)
        for name, value in scene["palette"].items()
    }


def _trace(initial_state: int, operations: list[int]) -> list[int]:
    state = int(initial_state) % 3
    trace: list[int] = []
    for operation in operations:
        state = (state + int(operation)) % 3
        trace.append(state)
    return trace


def _run_lengths(trace: list[int]) -> list[int]:
    lengths = [0, 0, 0]
    if not trace:
        return lengths
    previous = trace[0]
    current_length = 0
    for state in trace:
        if state == previous:
            current_length += 1
        else:
            lengths[previous] = max(lengths[previous], current_length)
            previous = state
            current_length = 1
    lengths[previous] = max(lengths[previous], current_length)
    return lengths


def _best_state(trace: list[int]) -> tuple[int, float]:
    lengths = _run_lengths(trace)
    order = sorted(range(3), key=lambda state: lengths[state], reverse=True)
    if len(order) < 2:
        raise ValueError("state trace is empty")
    return order[0], float(lengths[order[0]] - lengths[order[1]])


def _sample_points(rng: random.Random) -> tuple[list[list[int]], int]:
    """Return a central, monotone-x thin route with a randomized phase."""

    base_y = [250, 306, 190, 328, 205, 338, 224, 302, 244, 270]
    phase = rng.randrange(4)
    offsets = (-8, 0, 8, 4)
    xs = [86, 132, 186, 240, 294, 348, 402, 456, 510, 554]
    points: list[list[int]] = []
    for index, (x, y) in enumerate(zip(xs, base_y, strict=True)):
        jitter = 0 if index in (0, len(xs) - 1) else rng.randint(-3, 3)
        points.append([x + jitter, y + offsets[phase]])
    return points, phase


def _sample_distractors(rng: random.Random) -> list[dict[str, int]]:
    anchors = [(48, 68), (590, 70), (52, 416), (588, 414)]
    rng.shuffle(anchors)
    return [
        {
            "kind": index % 2,
            "x": x + rng.randint(-6, 6),
            "y": y + rng.randint(-6, 6),
            "size": rng.randint(8, 12),
        }
        for index, (x, y) in enumerate(anchors)
    ]


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced scenes with a unique longest post-station run."""

    rng = random.Random(int(seed))
    route_points, route_phase = _sample_points(rng)

    # Keep the operation sequence independent of the eventual state label.
    # A cyclic relabeling of the initial state shifts every derived state while
    # leaving all visible operation glyphs unchanged.
    for _ in range(200):
        operations = [rng.choice((-1, 0, 1)) for _ in range(STATION_COUNT)]
        base_initial = rng.randrange(3)
        base_trace = _trace(base_initial, operations)
        base_state, base_margin = _best_state(base_trace)
        if base_margin > 0:
            break
    else:
        raise RuntimeError("could not sample a unique longest run")

    desired_state = int(seed) % 3
    shift = (desired_state - base_state) % 3
    initial_state = (base_initial + shift) % 3
    return {
        "initial_state": initial_state,
        "operations": operations,
        "route_points": route_points,
        "route_phase": route_phase,
        "distractors": _sample_distractors(rng),
        "route_width": 6,
        "marker_size": 14,
        "palette": {name: list(value) for name, value in PALETTE.items()},
        "canvas": list(CANVAS),
    }


def _draw_operation(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    operation: int,
    radius: int,
    color: tuple[int, int, int],
) -> None:
    x, y = center
    if operation == 1:
        # A filled triangle pointing right means advance.
        draw.polygon([(x - radius, y - radius), (x - radius, y + radius), (x + radius, y)], fill=color)
    elif operation == -1:
        # A filled triangle pointing left means retreat.
        draw.polygon([(x + radius, y - radius), (x + radius, y + radius), (x - radius, y)], fill=color)
    else:
        # A hollow square means hold.
        draw.rectangle(
            (x - radius, y - radius, x + radius, y + radius),
            outline=color,
            width=4,
        )


def _draw_state_badge(
    draw: ImageDraw.ImageDraw,
    center: list[int],
    state: int,
    radius: int,
    colors: dict[str, tuple[int, int, int]],
    font: ImageFont.ImageFont,
) -> None:
    x, y = center
    state_color = colors["state_a" if state == 0 else "state_b" if state == 1 else "state_c"]
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=state_color, outline=colors["route"], width=3)
    label = STATE_LABELS[state]
    bbox = draw.textbbox((0, 0), label, font=font)
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((x - tw / 2, y - th / 2 - 1), label, fill=colors["background"], font=font)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the route, its visible operations, and a marked initial state."""

    width, height = (int(value) for value in scene["canvas"])
    colors = _colors(scene)
    image = Image.new("RGB", (width, height), colors["background"])
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    for distractor in scene["distractors"]:
        x, y, size = (int(distractor[key]) for key in ("x", "y", "size"))
        if int(distractor["kind"]) == 0:
            draw.ellipse((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)
        else:
            draw.rectangle((x - size, y - size, x + size, y + size), outline=colors["neutral"], width=3)

    points = [tuple(int(value) for value in point) for point in scene["route_points"]]
    draw.line(points, fill=colors["route"], width=int(scene["route_width"]), joint="curve")

    marker_radius = int(scene["marker_size"])
    for index, (point, operation) in enumerate(zip(points[1:-1], scene["operations"], strict=True), start=1):
        center = [int(point[0]), int(point[1])]
        _draw_operation(draw, center, int(operation), marker_radius, colors["operation"])
        label = str(index)
        bbox = draw.textbbox((0, 0), label, font=font)
        tw = bbox[2] - bbox[0]
        draw.text((center[0] - tw / 2, center[1] - marker_radius - 18), label, fill=colors["text"], font=font)

    _draw_state_badge(
        draw,
        [int(value) for value in points[0]],
        int(scene["initial_state"]),
        marker_radius + 3,
        colors,
        font,
    )
    end_x, end_y = points[-1]
    draw.ellipse((end_x - 9, end_y - 9, end_x + 9, end_y + 9), outline=colors["neutral"], width=3)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Compute only from the two visible label-bearing scene fields."""

    trace = _trace(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    state, _ = _best_state(trace)
    return STATE_LABELS[state]


def margin(scene: dict[str, Any]) -> float:
    trace = _trace(int(scene["initial_state"]), [int(value) for value in scene["operations"]])
    _, gap = _best_state(trace)
    return round(float(gap), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    if len(scene.get("operations", [])) != STATION_COUNT:
        return True
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    # Every answer-bearing field is explicitly visible.  No hidden
    # label-preserving pixel-identical transform is claimed.
    return []
