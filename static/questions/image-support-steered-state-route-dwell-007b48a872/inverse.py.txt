"""Independent pixel-only inverse for the connected state-dwell route."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_COLORS = {
    "background": (249, 251, 253),
    "route": (36, 61, 84),
    "operation": (216, 100, 58),
    "state_a": (192, 70, 67),
    "state_b": (42, 148, 94),
    "state_c": (67, 106, 190),
}


def _masks(image: Image.Image) -> dict[str, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    return {
        name: np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
        for name, color in _COLORS.items()
    }


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue: deque[tuple[int, int]] = deque([(y, x)])
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _centroid(component: np.ndarray) -> tuple[float, float]:
    y, x = component.mean(axis=0)
    return float(y), float(x)


def _bbox_stats(component: np.ndarray) -> tuple[float, float, float, float]:
    ys, xs = component[:, 0], component[:, 1]
    height = float(ys.max() - ys.min() + 1)
    width = float(xs.max() - xs.min() + 1)
    fill = len(component) / (width * height)
    return fill, width, height, float(xs.mean() - (xs.min() + xs.max()) / 2.0)


def _operation(component: np.ndarray) -> int:
    """Classify right triangle, left triangle, or outlined square."""

    fill, width, height, centroid_offset = _bbox_stats(component)
    if not (22.0 <= width <= 32.0 and 22.0 <= height <= 32.0):
        raise ValueError("station glyph has invalid size")
    if abs(centroid_offset) <= 1.5 and 0.30 < fill < 0.70:
        return 0
    if fill < 0.70 and centroid_offset < -1.5:
        return 1
    if fill < 0.70 and centroid_offset > 1.5:
        return -1
    raise ValueError("station glyph silhouette is ambiguous")


def _longest_state(trace: list[int]) -> str:
    lengths = [0, 0, 0]
    previous = trace[0]
    current = 0
    for state in trace:
        if state == previous:
            current += 1
        else:
            lengths[previous] = max(lengths[previous], current)
            previous = state
            current = 1
    lengths[previous] = max(lengths[previous], current)
    order = sorted(range(3), key=lambda state: lengths[state], reverse=True)
    if lengths[order[0]] <= lengths[order[1]]:
        raise ValueError("longest dwell run is tied")
    return ("A", "B", "C")[order[0]]


def decision_from_image(image: Image.Image) -> str:
    """Recover the start state and all eight ordered station operations from pixels."""

    masks = _masks(image)
    state_components: list[tuple[int, np.ndarray]] = []
    for state, name in enumerate(("state_a", "state_b", "state_c")):
        components = _components(masks[name], minimum=250)
        if len(components) > 1:
            raise ValueError("multiple components use one state color")
        if components:
            component = components[0]
            fill, width, height, _ = _bbox_stats(component)
            if not (28.0 <= width <= 42.0 and 28.0 <= height <= 42.0 and fill > 0.55):
                raise ValueError("start badge is ambiguous")
            state_components.append((state, component))
    if len(state_components) != 1:
        raise ValueError("expected exactly one colored start badge")
    initial_state = state_components[0][0]

    operation_components = _components(masks["operation"], minimum=60)
    if len(operation_components) != 8:
        raise ValueError("expected eight visible station glyphs")
    ordered: list[tuple[float, np.ndarray]] = []
    for component in operation_components:
        _, center_x = _centroid(component)
        ordered.append((center_x, component))
    ordered.sort(key=lambda item: item[0])
    if any(right[0] - left[0] < 30.0 for left, right in zip(ordered, ordered[1:])):
        raise ValueError("station order is unresolved")
    operations = [_operation(component) for _, component in ordered]

    route = masks["route"]
    traversable = route | masks["operation"] | masks["state_a"] | masks["state_b"] | masks["state_c"]
    route_components = _components(traversable, minimum=80)
    if len(route_components) != 1:
        raise ValueError("route is missing or disconnected")

    state = initial_state
    trace: list[int] = []
    for operation in operations:
        state = (state + operation) % 3
        trace.append(state)
    return _longest_state(trace)
