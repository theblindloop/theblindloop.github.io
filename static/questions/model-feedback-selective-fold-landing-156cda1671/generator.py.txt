"""Deterministic renderer for an ordered, side-selective folding program."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1200
HEIGHT = 800
GRID_N = 9
SOURCE_CELL = 58
SOURCE_ORIGIN = (55, 158)
OPTION_CELL = 24
OPTION_ORIGINS = ((680, 190), (948, 190), (680, 480), (948, 480))

WHITE = (255, 255, 255)
INK = (31, 41, 55)
GRID = (205, 214, 224)
PAPER = (249, 250, 252)
BLUE = (37, 99, 235)
AMBER = (245, 158, 11)
RED = (220, 38, 38)
VIOLET = (124, 58, 237)

FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_AXIS_K = {"horizontal": 4, "diag_down": 0, "vertical": 4, "diag_up": 8}
_ORIENTATIONS = ("horizontal", "diag_down", "vertical", "diag_up")


def _axis_value(point: tuple[int, int], fold: dict) -> int:
    x, y = point
    kind = str(fold["kind"])
    k = int(fold["k"])
    if kind == "vertical":
        return x - k
    if kind == "horizontal":
        return y - k
    if kind == "diag_down":
        return y - x - k
    if kind == "diag_up":
        return x + y - k
    raise ValueError(f"unknown axis kind {kind!r}")


def _reflect_point(point: tuple[int, int], fold: dict) -> tuple[int, int]:
    x, y = point
    kind = str(fold["kind"])
    k = int(fold["k"])
    if kind == "vertical":
        return 2 * k - x, y
    if kind == "horizontal":
        return x, 2 * k - y
    if kind == "diag_down":
        return y - k, x + k
    if kind == "diag_up":
        return k - y, k - x
    raise ValueError(f"unknown axis kind {kind!r}")


def _moves(point: tuple[int, int], fold: dict) -> bool:
    return _axis_value(point, fold) * int(fold["sign"]) > 0


def _fold_point(point: tuple[int, int], fold: dict) -> tuple[int, int]:
    return _reflect_point(point, fold) if _moves(point, fold) else point


def _apply_fold(points: Iterable[tuple[int, int]], fold: dict) -> set[tuple[int, int]]:
    return {_fold_point(point, fold) for point in points}


def _serial(points: Iterable[tuple[int, int]]) -> list[list[int]]:
    return [[x, y] for x, y in sorted(points, key=lambda point: (point[1], point[0]))]


def _inside(points: Iterable[tuple[int, int]]) -> bool:
    return all(0 <= x < GRID_N and 0 <= y < GRID_N for x, y in points)


def _final_set(scene: dict) -> set[tuple[int, int]]:
    points = {tuple(point) for point in scene["source"]}
    for fold in scene["folds"]:
        points = _apply_fold(points, fold)
    return points


def _option_score(scene: dict, option: Iterable[Iterable[int]]) -> int:
    return len(_final_set(scene) & {tuple(point) for point in option})


def _marker_for(
    rng: random.Random,
    fold: dict,
    other: dict,
    forbidden: set[tuple[int, int]],
) -> tuple[int, int] | None:
    choices = [
        (x, y)
        for y in range(GRID_N)
        for x in range(GRID_N)
        if _axis_value((x, y), fold) * int(fold["sign"]) >= 3
        and abs(_axis_value((x, y), other)) >= 1
        and (x, y) not in forbidden
    ]
    rng.shuffle(choices)
    return choices[0] if choices else None


def _sample_program(rng: random.Random, correct_index: int) -> dict:
    for _ in range(30000):
        first_orientation = rng.randrange(4)
        step = rng.choice((-1, 1))
        kinds = (
            _ORIENTATIONS[first_orientation],
            _ORIENTATIONS[(first_orientation + step) % 4],
        )
        folds = [
            {"kind": kind, "k": _AXIS_K[kind], "sign": rng.choice((-1, 1))}
            for kind in kinds
        ]

        pool = [
            (x, y)
            for y in range(GRID_N)
            for x in range(GRID_N)
            if _axis_value((x, y), folds[0]) != 0
            and _axis_value((x, y), folds[1]) != 0
        ]
        source_list = rng.sample(pool, 7)
        source = set(source_list)
        intermediate_by_point = [_fold_point(point, folds[0]) for point in source_list]
        intermediate = set(intermediate_by_point)
        if len(intermediate) != 7 or not _inside(intermediate):
            continue
        if any(_axis_value(point, folds[1]) == 0 for point in intermediate):
            continue
        first_moves = sum(_moves(point, folds[0]) for point in source_list)
        second_moves = sum(_moves(point, folds[1]) for point in intermediate_by_point)
        membership_changes = sum(
            _moves(before, folds[1]) != _moves(after, folds[1])
            for before, after in zip(source_list, intermediate_by_point)
        )
        if not (2 <= first_moves <= 5 and 2 <= second_moves <= 5 and membership_changes >= 2):
            continue

        final = _apply_fold(intermediate, folds[1])
        if len(final) != 7 or not _inside(final):
            continue

        stale_membership = {
            _reflect_point(after, folds[1]) if _moves(before, folds[1]) else after
            for before, after in zip(source_list, intermediate_by_point)
        }
        global_second = {_reflect_point(point, folds[1]) for point in intermediate}
        reverse_order = _apply_fold(_apply_fold(source, folds[1]), folds[0])
        wrong_side = _apply_fold(
            intermediate,
            {**folds[1], "sign": -int(folds[1]["sign"])},
        )
        first_only = intermediate
        global_both = {
            _reflect_point(_reflect_point(point, folds[0]), folds[1])
            for point in source
        }
        wrong_pool: list[set[tuple[int, int]]] = []
        for wrong in (
            stale_membership,
            global_second,
            reverse_order,
            wrong_side,
            first_only,
            global_both,
        ):
            if (
                len(wrong) == 7
                and _inside(wrong)
                and wrong != final
                and len(wrong & final) <= 5
                and wrong not in wrong_pool
            ):
                wrong_pool.append(wrong)
        if len(wrong_pool) < 3:
            continue
        rng.shuffle(wrong_pool)

        marker1 = _marker_for(rng, folds[0], folds[1], source)
        if marker1 is None:
            continue
        marker2 = _marker_for(rng, folds[1], folds[0], source | {marker1})
        if marker2 is None or max(abs(marker1[0] - marker2[0]), abs(marker1[1] - marker2[1])) < 2:
            continue
        folds[0]["marker"] = list(marker1)
        folds[1]["marker"] = list(marker2)

        options: list[set[tuple[int, int]]] = []
        wrong_iter = iter(wrong_pool[:3])
        for index in range(4):
            options.append(final if index == correct_index else next(wrong_iter))
        order = list(range(4))
        rng.shuffle(order)
        return {
            "source": _serial(source),
            "folds": folds,
            "options": [_serial(option) for option in options],
            "option_draw_order": order,
        }
    raise RuntimeError("could not sample a clear selective-fold program")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x51EC71F0)
    scene = _sample_program(rng, int(seed) % 4)
    analytic_gold(scene)
    return scene


def analytic_gold(scene: dict) -> str:
    # Unpack every label-bearing operand from the declared top-level fields.
    points = {tuple(point) for point in scene["source"]}
    for fold in scene["folds"]:
        points = _apply_fold(points, fold)
    scores = [len(points & {tuple(point) for point in option}) for option in scene["options"]]
    winners = [index for index, score in enumerate(scores) if score == len(points)]
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} exact options: {scores}")
    return f"option-{winners[0] + 1}"


def margin(scene: dict) -> float:
    scores = sorted((_option_score(scene, option) for option in scene["options"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["option_draw_order"] = list(reversed(scene["option_draw_order"]))
    return [("reverse_disjoint_option_painter_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_BOLD if bold else FONT_REGULAR, size)


def _center(origin: tuple[int, int], cell: int, point: tuple[float, float]) -> tuple[float, float]:
    x, y = point
    return origin[0] + (x + 0.5) * cell, origin[1] + (y + 0.5) * cell


def _axis_endpoints(fold: dict) -> tuple[tuple[float, float], tuple[float, float]]:
    low, high = -0.42, GRID_N - 0.58
    kind, k = str(fold["kind"]), float(fold["k"])
    if kind == "vertical":
        a, b = (k, low), (k, high)
    elif kind == "horizontal":
        a, b = (low, k), (high, k)
    elif kind == "diag_down":
        x0, x1 = max(low, low - k), min(high, high - k)
        a, b = (x0, x0 + k), (x1, x1 + k)
    else:
        x0, x1 = max(low, k - high), min(high, k - low)
        a, b = (x0, k - x0), (x1, k - x1)
    return _center(SOURCE_ORIGIN, SOURCE_CELL, a), _center(SOURCE_ORIGIN, SOURCE_CELL, b)


def _dashed_line(
    draw: ImageDraw.ImageDraw,
    start: tuple[float, float],
    end: tuple[float, float],
    color: tuple[int, int, int],
) -> None:
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    distance = 0.0
    while distance < length:
        stop = min(distance + 20.0, length)
        draw.line(
            (
                start[0] + ux * distance,
                start[1] + uy * distance,
                start[0] + ux * stop,
                start[1] + uy * stop,
            ),
            fill=color,
            width=8,
        )
        distance += 32.0


def _draw_grid(draw: ImageDraw.ImageDraw, origin: tuple[int, int], cell: int) -> None:
    size = GRID_N * cell
    draw.rectangle((origin[0], origin[1], origin[0] + size, origin[1] + size), fill=PAPER, outline=INK, width=3)
    for step in range(1, GRID_N):
        draw.line((origin[0] + step * cell, origin[1], origin[0] + step * cell, origin[1] + size), fill=GRID, width=1)
        draw.line((origin[0], origin[1] + step * cell, origin[0] + size, origin[1] + step * cell), fill=GRID, width=1)


def _draw_badge(draw: ImageDraw.ImageDraw, fold: dict, number: int, color: tuple[int, int, int]) -> None:
    cx, cy = _center(SOURCE_ORIGIN, SOURCE_CELL, tuple(fold["marker"]))
    draw.ellipse((cx - 23, cy - 23, cx + 23, cy + 23), fill=color, outline=WHITE, width=2)
    draw.text((cx, cy - 1), str(number), fill=WHITE, font=_font(27, True), anchor="mm")


def _draw_option(draw: ImageDraw.ImageDraw, option: list[list[int]], index: int) -> None:
    origin = OPTION_ORIGINS[index]
    draw.rounded_rectangle(
        (origin[0] - 13, origin[1] - 47, origin[0] + 54, origin[1] - 10),
        radius=15,
        fill=INK,
    )
    draw.text((origin[0] + 20, origin[1] - 29), str(index + 1), fill=WHITE, font=_font(21, True), anchor="mm")
    _draw_grid(draw, origin, OPTION_CELL)
    for point in option:
        cx, cy = _center(origin, OPTION_CELL, tuple(point))
        draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=AMBER)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    draw.text((34, 20), "FOLD MARKED SIDE 1, THEN 2", fill=INK, font=_font(31, True))
    draw.ellipse((704, 29, 728, 53), fill=BLUE)
    draw.text((740, 28), "start dots", fill=INK, font=_font(19))
    draw.ellipse((874, 29, 898, 53), fill=AMBER)
    draw.text((910, 28), "final options", fill=INK, font=_font(19))
    draw.text((55, 112), "START", fill=INK, font=_font(24, True))
    draw.text((680, 108), "WHICH FINAL SET?", fill=INK, font=_font(24, True))

    _draw_grid(draw, SOURCE_ORIGIN, SOURCE_CELL)
    for point in scene["source"]:
        cx, cy = _center(SOURCE_ORIGIN, SOURCE_CELL, tuple(point))
        draw.ellipse((cx - 17, cy - 17, cx + 17, cy + 17), fill=BLUE)

    colors = (RED, VIOLET)
    for index, fold in enumerate(scene["folds"]):
        start, end = _axis_endpoints(fold)
        _dashed_line(draw, start, end, colors[index])
    for index, fold in enumerate(scene["folds"]):
        _draw_badge(draw, fold, index + 1, colors[index])

    for index in scene.get("option_draw_order", list(range(4))):
        _draw_option(draw, scene["options"][int(index)], int(index))
    draw.text(
        (55, 704),
        "After fold 1, use the dots' new positions for fold 2.",
        fill=INK,
        font=_font(20, True),
    )
    return image
