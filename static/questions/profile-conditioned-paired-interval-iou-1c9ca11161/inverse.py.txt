"""Independent inverse arm using only final RGB pixels."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = "ABCD"


def _active_runs(values: np.ndarray) -> list[tuple[int, int]]:
    indices = np.flatnonzero(values)
    if len(indices) == 0:
        return []
    runs: list[tuple[int, int]] = []
    start = previous = int(indices[0])
    for value in indices[1:]:
        current = int(value)
        if current != previous + 1:
            runs.append((start, previous))
            start = current
        previous = current
    runs.append((start, previous))
    return runs


def _iou(first: tuple[int, int], second: tuple[int, int]) -> float:
    a, b = first
    c, d = second
    intersection = max(0, min(b, d) - max(a, c))
    union = (b - a) + (d - c) - intersection
    if union <= 0:
        raise ValueError("degenerate interval union")
    return intersection / union


def decision_from_image(image: Image.Image) -> str:
    """Segment saturated marks, recover eight bounds, and compare four IoUs."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = array.max(axis=2) - array.min(axis=2)
    mask = (spread > 70) & (array.max(axis=2) < 248)
    runs = _active_runs(mask.any(axis=1))
    if len(runs) != 8:
        raise ValueError(f"expected eight separated interval marks, found {len(runs)}")

    components = []
    for y0, y1 in runs:
        ys, xs = np.nonzero(mask[y0 : y1 + 1])
        if len(xs) < 80:
            raise ValueError("interval mark has too little colored evidence")
        components.append((y0, y1, int(xs.min()), int(xs.max())))

    scores = []
    for row in range(4):
        first, second = components[2 * row : 2 * row + 2]
        if second[0] - first[1] > 10 or max(first[1], second[1]) - min(first[0], second[0]) > 42:
            raise ValueError("colored marks cannot be paired into rows")
        scores.append(_iou((first[2], first[3]), (second[2], second[3])))
    order = sorted(range(4), key=lambda index: scores[index], reverse=True)
    if scores[order[0]] - scores[order[1]] < 0.004:
        raise ValueError("pixel evidence is tied at raster resolution")
    return LABELS[order[0]]
