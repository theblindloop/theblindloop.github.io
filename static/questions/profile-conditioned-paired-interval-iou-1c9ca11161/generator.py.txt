"""Deterministic scene sampler and raster renderer."""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 680, 420
PLOT_WIDTH = 400
LABELS = "ABCD"
FONT_REGULAR = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
FONT_BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

PALETTES = (
    {"blue": [36, 105, 196], "orange": [224, 111, 36]},
    {"blue": [30, 119, 180], "orange": [230, 126, 34]},
    {"blue": [48, 92, 184], "orange": [214, 96, 30]},
)

# The runner-up has more raw overlap, more total ink, and a smaller center gap
# than the winner. Thus those local proxies cannot solve the scene.
SAFE_TEMPLATES = (
    (
        ((10, 80), (30, 90)),
        ((0, 100), (22, 76)),
        ((8, 92), (38, 60)),
        ((5, 50), (44, 96)),
    ),
    (
        ((15, 82), (26, 94)),
        ((2, 98), (16, 77)),
        ((10, 88), (39, 61)),
        ((4, 47), (41, 93)),
    ),
    (
        ((12, 76), (25, 91)),
        ((0, 100), (20, 72)),
        ((7, 89), (34, 59)),
        ((9, 51), (46, 97)),
    ),
    (
        ((18, 84), (29, 95)),
        ((1, 99), (18, 79)),
        ((11, 91), (37, 62)),
        ((3, 46), (40, 94)),
    ),
)

# Deliberate visible near-boundary scenes for quarantine/gallery coverage.
NEAR_TEMPLATE = (
    ((10, 80), (28, 92)),
    ((0, 100), (18, 80)),
    ((8, 91), (35, 61)),
    ((6, 49), (43, 95)),
)


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    path = FONT_BOLD if bold else FONT_REGULAR
    if not Path(path).is_file():
        raise RuntimeError(f"required deterministic font is missing: {path}")
    return ImageFont.truetype(path, size=size)


def _iou(pair: list[list[int]] | tuple[tuple[int, int], tuple[int, int]]) -> float:
    (a, b), (c, d) = pair
    intersection = max(0, min(b, d) - max(a, c))
    union = (b - a) + (d - c) - intersection
    return intersection / union if union else 0.0


def _reflect(pair: tuple[tuple[int, int], tuple[int, int]]):
    return tuple((100 - b, 100 - a) for a, b in pair)


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; ``seed % 4`` pins the winning row."""
    rng = random.Random(int(seed))
    template = NEAR_TEMPLATE if seed % 13 == 0 else SAFE_TEMPLATES[seed % len(SAFE_TEMPLATES)]
    pairs = [tuple(interval for interval in pair) for pair in template]
    if rng.random() < 0.5:
        pairs = [_reflect(pair) for pair in pairs]
    rng.shuffle(pairs)
    winner = max(range(4), key=lambda index: _iou(pairs[index]))
    target = seed % 4
    pairs[winner], pairs[target] = pairs[target], pairs[winner]
    return {
        "pairs": [[list(interval) for interval in pair] for pair in pairs],
        "palette": copy.deepcopy(PALETTES[rng.randrange(len(PALETTES))]),
        "blue_on_top": [bool(rng.getrandbits(1)) for _ in range(4)],
        "plot_left": [146, 158, 170][rng.randrange(3)],
        "plot_top": [112, 116, 120][rng.randrange(3)],
        "row_gap": [66, 68, 70][rng.randrange(3)],
    }


def analytic_gold(scene: dict) -> str:
    scores = [_iou(pair) for pair in scene["pairs"]]
    return LABELS[max(range(4), key=lambda index: scores[index])]


def margin(scene: dict) -> float:
    scores = sorted((_iou(pair) for pair in scene["pairs"]), reverse=True)
    return 100.0 * (scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_interval(draw: ImageDraw.ImageDraw, bounds, y: int, color) -> None:
    x1, x2 = bounds
    draw.rectangle((x1, y - 3, x2, y + 3), fill=tuple(color))
    draw.rectangle((x1 - 2, y - 8, x1 + 2, y + 8), fill=tuple(color))
    draw.rectangle((x2 - 2, y - 8, x2 + 2, y + 8), fill=tuple(color))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (249, 249, 247))
    draw = ImageDraw.Draw(image)
    left = int(scene["plot_left"])
    top = int(scene["plot_top"])
    gap = int(scene["row_gap"])
    right = left + PLOT_WIDTH
    bottom = top + 3 * gap + 28

    draw.rounded_rectangle((left - 18, top - 32, right + 18, bottom), radius=9,
                           fill=(255, 255, 255), outline=(184, 188, 193), width=2)
    draw.text((34, 24), "PAIRED INTERVAL OVERLAP", font=_font(22, True), fill=(35, 38, 42))
    draw.text((35, 56), "compare overlap / union", font=_font(14), fill=(86, 91, 97))

    for tick in (0, 25, 50, 75, 100):
        x = left + tick * 4
        draw.line((x, top - 20, x, bottom - 10), fill=(226, 228, 230), width=1)
        label = str(tick)
        box = draw.textbbox((0, 0), label, font=_font(11))
        draw.text((x - (box[2] - box[0]) / 2, bottom - 4), label,
                  font=_font(11), fill=(92, 96, 101))

    palette = scene["palette"]
    for index, pair in enumerate(scene["pairs"]):
        y = top + index * gap
        draw.text((left - 58, y - 12), LABELS[index], font=_font(21, True), fill=(35, 38, 42))
        draw.line((left, y, right, y), fill=(205, 208, 211), width=1)
        blue_y, orange_y = (y - 10, y + 10) if scene["blue_on_top"][index] else (y + 10, y - 10)
        blue = [left + 4 * int(value) for value in pair[0]]
        orange = [left + 4 * int(value) for value in pair[1]]
        _draw_interval(draw, blue, blue_y, palette["blue"])
        _draw_interval(draw, orange, orange_y, palette["orange"])
    return image
