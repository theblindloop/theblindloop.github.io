"""Latent sampler, analytic arm, and deterministic Pillow renderer."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 430
TRACKS = tuple(range(6))
PALETTE = (
    (204, 55, 61),
    (43, 108, 176),
    (42, 150, 83),
    (232, 126, 38),
    (132, 84, 166),
    (25, 157, 174),
)


def _inversions(values: list[int] | tuple[int, ...]) -> int:
    return sum(values[i] > values[j] for i in range(6) for j in range(i + 1, 6))


_BY_INV: dict[int, list[tuple[int, ...]]] = {value: [] for value in range(16)}
for _perm in itertools.permutations(TRACKS):
    _BY_INV[_inversions(_perm)].append(_perm)


def _counts(left_orders: list[list[int]], right_orders: list[list[int]]) -> list[int]:
    counts: list[int] = []
    for left, right in zip(left_orders, right_orders):
        right_rank = {track: rank for rank, track in enumerate(right)}
        relative = [right_rank[track] for track in left]
        counts.append(_inversions(relative))
    return counts


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    winner_count = rng.choice((10, 11, 12, 13, 14))
    gap = 1 if int(seed) % 7 == 0 else rng.choice((2, 3, 4))
    runner = winner_count - gap
    lower_pool = [value for value in range(max(0, runner - 7), runner) if value != runner]
    rng.shuffle(lower_pool)
    other_counts = [runner, lower_pool[0], lower_pool[1]]
    rng.shuffle(other_counts)
    target_counts: list[int] = []
    other_iter = iter(other_counts)
    for panel in range(4):
        target_counts.append(winner_count if panel == winner else next(other_iter))

    left_orders: list[list[int]] = []
    right_orders: list[list[int]] = []
    for target in target_counts:
        left = list(TRACKS)
        rng.shuffle(left)
        relative = rng.choice(_BY_INV[target])
        right = [left[index] for index in relative]
        left_orders.append(left)
        right_orders.append(right)

    palette_assignment = list(TRACKS)
    rng.shuffle(palette_assignment)
    return {
        "left_orders": left_orders,
        "right_orders": right_orders,
        "palette_assignment": palette_assignment,
        "x_shift": rng.randint(-5, 5),
        "y_shift": rng.randint(-4, 4),
        "dot_radius": rng.choice((7, 8, 9)),
        "background_tone": rng.randint(0, 5),
    }


def analytic_gold(scene: dict) -> str:
    left_orders = scene["left_orders"]
    right_orders = scene["right_orders"]
    counts = _counts(left_orders, right_orders)
    best = max(counts)
    if counts.count(best) != 1:
        raise ValueError("scene has no unique inversion-count winner")
    return "ABCD"[counts.index(best)]


def margin(scene: dict) -> float:
    counts = sorted(_counts(scene["left_orders"], scene["right_orders"]), reverse=True)
    return float(counts[0] - counts[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["left_orders"] = [[(track + 1) % 6 for track in order] for order in scene["left_orders"]]
    twin["right_orders"] = [[(track + 1) % 6 for track in order] for order in scene["right_orders"]]
    assignment = [0] * 6
    for old_track, palette_index in enumerate(scene["palette_assignment"]):
        assignment[(old_track + 1) % 6] = palette_index
    twin["palette_assignment"] = assignment
    return [("cyclic_hidden_track_rename", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size=size)
    except OSError:
        return ImageFont.load_default()


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.ImageFont, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    draw.text((xy[0] - width // 2, xy[1]), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    left_orders = scene["left_orders"]
    right_orders = scene["right_orders"]
    palette_assignment = scene["palette_assignment"]
    x_shift = int(scene["x_shift"])
    y_shift = int(scene["y_shift"])
    dot_radius = int(scene["dot_radius"])
    background_tone = int(scene["background_tone"])

    background = (250 - background_tone, 249 - background_tone, 246 - background_tone)
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    title_font = _font(21, bold=True)
    label_font = _font(18, bold=True)
    small_font = _font(12, bold=True)
    ink = (41, 45, 52)
    guide = (216 - background_tone, 216 - background_tone, 212 - background_tone)
    border = (169, 174, 180)
    _centered(draw, (WIDTH // 2, 16), "PAIRWISE RANK REVERSALS", title_font, ink)
    _centered(draw, (WIDTH // 2, 46), "Match colors; compare top-to-bottom order", _font(13), (79, 84, 91))

    slot_top = 128 + y_shift
    slot_gap = 47
    for panel in range(4):
        panel_x = 18 + panel * 236 + x_shift
        left_x = panel_x + 61
        right_x = panel_x + 157
        draw.rounded_rectangle((panel_x, 79 + y_shift, panel_x + 214, 403 + y_shift), radius=8, outline=border, width=2)
        _centered(draw, (panel_x + 107, 86 + y_shift), "ABCD"[panel], label_font, ink)
        _centered(draw, (left_x, 109 + y_shift), "BEFORE", small_font, ink)
        _centered(draw, (right_x, 109 + y_shift), "AFTER", small_font, ink)
        for rank in range(6):
            y = slot_top + rank * slot_gap
            draw.line((panel_x + 25, y, panel_x + 189, y), fill=guide, width=1)
        for column_x, order in ((left_x, left_orders[panel]), (right_x, right_orders[panel])):
            for rank, track in enumerate(order):
                y = slot_top + rank * slot_gap
                color = PALETTE[palette_assignment[track]]
                r = dot_radius
                draw.ellipse((column_x - r, y - r, column_x + r, y + r), fill=color, outline=(255, 255, 255), width=1)
    return image
