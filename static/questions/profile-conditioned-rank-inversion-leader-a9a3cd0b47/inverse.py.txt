"""Independent PNG-only inverse arm for pairwise rank reversals.

The implementation discovers colored components and rank columns from the raster;
it has no scene input and no renderer/generator/prompt imports.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """The visible evidence is malformed, ambiguous, or tied."""


def _components(mask: np.ndarray, rgb: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[dict] = []
    for y, x in zip(*np.nonzero(mask)):
        y = int(y)
        x = int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if 90 <= len(points) <= 400:
            ys = np.fromiter((p[0] for p in points), dtype=float)
            xs = np.fromiter((p[1] for p in points), dtype=float)
            colors = np.array([rgb[p[0], p[1]] for p in points], dtype=float)
            found.append({"x": float(xs.mean()), "y": float(ys.mean()), "rgb": colors.mean(axis=0), "area": len(points)})
    return found


def _cluster_columns(marks: list[dict]) -> list[list[dict]]:
    ordered = sorted(marks, key=lambda mark: mark["x"])
    columns: list[list[dict]] = []
    for mark in ordered:
        if not columns or abs(mark["x"] - np.mean([item["x"] for item in columns[-1]])) > 24:
            columns.append([mark])
        else:
            columns[-1].append(mark)
    if len(columns) != 8 or any(len(column) != 6 for column in columns):
        raise OracleAbstention(f"expected eight six-dot columns, found {[len(c) for c in columns]}")
    gaps = [np.mean([m["x"] for m in columns[i + 1]]) - np.mean([m["x"] for m in columns[i]]) for i in range(7)]
    if any(gap < 35 for gap in gaps):
        raise OracleAbstention("rank columns are not visibly separated")
    return columns


def _panel_inversions(left: list[dict], right: list[dict]) -> int:
    left = sorted(left, key=lambda mark: mark["y"])
    right = sorted(right, key=lambda mark: mark["y"])
    if min(np.diff([mark["y"] for mark in left])) < 20 or min(np.diff([mark["y"] for mark in right])) < 20:
        raise OracleAbstention("rank slots touch or overlap")

    distances = np.array([[np.linalg.norm(a["rgb"] - b["rgb"]) for b in right] for a in left])
    mapping = distances.argmin(axis=1)
    if len(set(int(value) for value in mapping)) != 6:
        raise OracleAbstention("same-color pairing is not one-to-one")
    if any(distances[row, mapping[row]] > 18 for row in range(6)):
        raise OracleAbstention("a color lacks a reliable mate")
    ranks = [int(mapping[row]) for row in range(6)]
    return sum(ranks[i] > ranks[j] for i in range(6) for j in range(i + 1, 6))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = rgb.max(axis=2)
    low = rgb.min(axis=2)
    saturation = high - low
    mask = (saturation >= 55) & (high >= 90) & (low <= 190)
    marks = _components(mask, rgb)
    if len(marks) != 48:
        raise OracleAbstention(f"expected 48 colored dots, found {len(marks)}")
    columns = _cluster_columns(marks)
    counts = [_panel_inversions(columns[2 * panel], columns[2 * panel + 1]) for panel in range(4)]
    best = max(counts)
    if counts.count(best) != 1:
        raise OracleAbstention(f"largest visible inversion count is tied: {counts}")
    return "ABCD"[counts.index(best)]
