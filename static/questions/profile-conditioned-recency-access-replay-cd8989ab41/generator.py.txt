"""Deterministic sampler, replay logic, and Pillow renderer."""

from __future__ import annotations

from copy import deepcopy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1040, 640
LABELS = ("A", "B", "C", "D")
INK = "#17212B"
MUTED = "#64717D"
PALETTES = (
    {"rack": "#CFEAF7", "cards": "#FFE5A6", "accent": "#D64B4B"},
    {"rack": "#DEE8FA", "cards": "#FADC B2".replace(" ", ""), "accent": "#C53D55"},
)


def _replay_from(initial: list[str], accesses: list[str]) -> tuple[list[str], int]:
    order = list(initial)
    effective = 0
    if sorted(order) != list(LABELS):
        raise ValueError("initial_order must be a permutation of A-D")
    for label in accesses:
        if label not in LABELS:
            raise ValueError("unknown access label")
        index = order.index(label)
        if index < len(order) - 1:
            order[index], order[index + 1] = order[index + 1], order[index]
            effective += 1
    return order, effective


def _replay(scene: dict) -> tuple[list[str], int]:
    return _replay_from(list(scene["initial_order"]), list(scene["accesses"]))


def analytic_gold(scene: dict) -> str:
    return _replay(scene)[0][0]


def margin(scene: dict) -> float:
    _replay(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        order, _ = _replay(scene)
        return len(order) != 4
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def sample_scene(seed: int) -> dict:
    """Sample balanced scenes with equal access counts and useful counterfactuals."""

    rng = random.Random(int(seed) ^ 0x5A17C9)
    target = LABELS[int(seed) % len(LABELS)]
    for _ in range(6000):
        initial = list(LABELS)
        rng.shuffle(initial)
        accesses = [label for label in LABELS for _ in range(2)]
        rng.shuffle(accesses)
        final, effective = _replay_from(initial, accesses)
        if final[0] != target or not 4 <= effective <= 7:
            continue
        reversed_final, _ = _replay_from(initial, list(reversed(accesses)))
        swapped_initial = list(initial)
        swapped_initial[0], swapped_initial[1] = swapped_initial[1], swapped_initial[0]
        counterfactual_final, _ = _replay_from(swapped_initial, accesses)
        if reversed_final[0] == target or counterfactual_final[0] == target:
            continue
        return {
            "initial_order": initial,
            "accesses": accesses,
            "palette": rng.randrange(len(PALETTES)),
            "shift": [rng.randint(-10, 10), rng.randint(-5, 5)],
        }
    raise RuntimeError(f"could not sample a balanced scene for seed {seed}")


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(f"/usr/share/fonts/truetype/dejavu/{name}", size=size)


def _pip_centers(count: int, cx: int, cy: int, spread: int = 21) -> list[tuple[int, int]]:
    layouts = {
        1: [(0, 0)],
        2: [(-1, -1), (1, 1)],
        3: [(-1, -1), (0, 0), (1, 1)],
        4: [(-1, -1), (1, -1), (-1, 1), (1, 1)],
    }
    return [(cx + dx * spread, cy + dy * spread) for dx, dy in layouts[count]]


def _pipped_tile(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    count: int,
    fill: str,
    *,
    radius: int = 8,
) -> None:
    draw.rounded_rectangle(box, radius=14, fill=fill, outline=INK, width=4)
    cx = (box[0] + box[2]) // 2
    cy = (box[1] + box[3]) // 2
    spread = 20 if box[2] - box[0] >= 90 else 17
    for px, py in _pip_centers(count, cx, cy, spread):
        draw.ellipse((px - radius, py - radius, px + radius, py + radius), fill=INK)


def render(scene: dict) -> Image.Image:
    initial = list(scene["initial_order"])
    accesses = list(scene["accesses"])
    palette_index = int(scene["palette"])
    dx, dy = (int(value) for value in scene["shift"])
    if sorted(initial) != list(LABELS) or len(accesses) != 8:
        raise ValueError("invalid visible program")
    if not 0 <= palette_index < len(PALETTES):
        raise ValueError("invalid palette")
    palette = PALETTES[palette_index]

    image = Image.new("RGB", (WIDTH, HEIGHT), "#FFFFFF")
    draw = ImageDraw.Draw(image)

    def p(x: int, y: int) -> tuple[int, int]:
        return x + dx, y + dy

    title_font = _font(30, bold=True)
    section_font = _font(20, bold=True)
    label_font = _font(17, bold=True)
    small_font = _font(15)
    draw.text(p(520, 28), "ONE-STEP RECENCY PROMOTION", fill=INK, font=title_font, anchor="ma")
    draw.text(
        p(520, 72),
        "PIP KEY:  1 = A     2 = B     3 = C     4 = D",
        fill=MUTED,
        font=label_font,
        anchor="ma",
    )

    draw.text(p(520, 116), "STARTING RECENCY RACK", fill=INK, font=section_font, anchor="ma")
    draw.text(p(238, 151), "LEAST RECENT", fill=palette["accent"], font=label_font, anchor="ma")
    draw.text(p(802, 151), "MOST RECENT", fill=palette["accent"], font=label_font, anchor="ma")
    draw.line([p(340, 151), p(700, 151)], fill=palette["accent"], width=4)
    draw.polygon([p(700, 151), p(682, 142), p(682, 160)], fill=palette["accent"])

    tile_size = 104
    rack_left = 249
    rack_top = 174
    rack_gap = 42
    for index, label in enumerate(initial):
        left = rack_left + index * (tile_size + rack_gap)
        box = (*p(left, rack_top), *p(left + tile_size, rack_top + tile_size))
        _pipped_tile(draw, box, LABELS.index(label) + 1, palette["rack"], radius=8)
        draw.text(p(left + tile_size // 2, rack_top + 121), f"slot {index + 1}", fill=MUTED, font=small_font, anchor="ma")
        if index < 3:
            ax = left + tile_size + 21
            draw.text(p(ax, rack_top + tile_size // 2), "→", fill=MUTED, font=section_font, anchor="mm")

    draw.text(p(520, 337), "READ ACCESS CARDS LEFT TO RIGHT", fill=INK, font=section_font, anchor="ma")
    card_w, card_h, card_gap = 84, 108, 28
    card_left, card_top = 86, 378
    for index, label in enumerate(accesses):
        left = card_left + index * (card_w + card_gap)
        box = (*p(left, card_top), *p(left + card_w, card_top + card_h))
        _pipped_tile(draw, box, LABELS.index(label) + 1, palette["cards"], radius=7)
        draw.text(p(left + card_w // 2, card_top - 13), str(index + 1), fill=MUTED, font=small_font, anchor="ms")
        draw.text(p(left + card_w // 2, card_top + card_h + 17), "ACCESS", fill=INK, font=small_font, anchor="ma")

    draw.rounded_rectangle((*p(170, 550), *p(870, 608)), radius=12, fill="#F5F7F9", outline="#CDD5DC", width=2)
    draw.text(
        p(520, 579),
        "Each access swaps its tile ONE slot right; rightmost access = no change.",
        fill=INK,
        font=label_font,
        anchor="mm",
    )
    return image


def palette_equivalent(scene: dict) -> dict:
    twin = deepcopy(scene)
    twin["palette"] = (int(scene["palette"]) + 1) % len(PALETTES)
    return twin


def translated_equivalent(scene: dict, dx: int = 7, dy: int = -3) -> dict:
    twin = deepcopy(scene)
    twin["shift"] = [int(scene["shift"][0]) + dx, int(scene["shift"][1]) + dy]
    return twin
