"""Independent inverse arm: decode and replay only the final PNG pixels."""

from __future__ import annotations

from PIL import Image
import numpy as np

LABELS = ("A", "B", "C", "D")


class OracleAbstention(ValueError):
    """Raised when the raster does not contain one unambiguous visible program."""


def _runs(flags: np.ndarray, minimum: int) -> list[tuple[int, int]]:
    padded = np.pad(flags.astype(np.int8), (1, 1))
    edges = np.diff(padded)
    starts = np.flatnonzero(edges == 1)
    ends = np.flatnonzero(edges == -1) - 1
    return [
        (int(start), int(end))
        for start, end in zip(starts, ends)
        if end - start + 1 >= minimum
    ]


def _colored_rows(arr: np.ndarray) -> list[list[tuple[int, int, int, int]]]:
    red = arr[:, :, 0].astype(np.int16)
    green = arr[:, :, 1].astype(np.int16)
    blue = arr[:, :, 2].astype(np.int16)
    cool = (blue - red > 15) & (blue - green > 7) & (red > 180)
    warm = (red - blue > 50) & (green - blue > 28) & (blue > 130)
    mask = cool | warm
    bands = _runs(mask.any(axis=1), minimum=45)
    rows: list[list[tuple[int, int, int, int]]] = []
    for y0, y1 in bands:
        slab = mask[y0 : y1 + 1]
        boxes = []
        for x0, x1 in _runs(slab.any(axis=0), minimum=45):
            active = slab[:, x0 : x1 + 1]
            if int(active.sum()) < 2500:
                continue
            ys, xs = np.nonzero(active)
            boxes.append((x0 + int(xs.min()), y0 + int(ys.min()), x0 + int(xs.max()), y0 + int(ys.max())))
        if boxes:
            rows.append(sorted(boxes, key=lambda box: box[0]))
    return rows


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[int, int, int, int, int]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            stack = [(x, y)]
            seen[y, x] = True
            count = 0
            min_x = max_x = x
            min_y = max_y = y
            while stack:
                px, py = stack.pop()
                count += 1
                min_x, max_x = min(min_x, px), max(max_x, px)
                min_y, max_y = min(min_y, py), max(max_y, py)
                for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((nx, ny))
            found.append((min_x, min_y, max_x, max_y, count))
    return found


def _pip_label(arr: np.ndarray, box: tuple[int, int, int, int]) -> str:
    x0, y0, x1, y1 = box
    inset = 10
    crop = arr[y0 + inset : y1 - inset + 1, x0 + inset : x1 - inset + 1]
    if crop.shape[0] < 40 or crop.shape[1] < 40:
        raise OracleAbstention("colored tile is too small")
    dark = np.max(crop, axis=2) < 70
    pips = []
    for cx0, cy0, cx1, cy1, area in _components(dark):
        width = cx1 - cx0 + 1
        height = cy1 - cy0 + 1
        if 80 <= area <= 320 and 10 <= width <= 22 and 10 <= height <= 22:
            pips.append((cx0, cy0, cx1, cy1, area))
    if not 1 <= len(pips) <= 4:
        raise OracleAbstention(f"tile has {len(pips)} decodable pips")
    return LABELS[len(pips) - 1]


def _decode(image: Image.Image) -> tuple[list[str], list[str]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    rows = _colored_rows(arr)
    counts = [len(row) for row in rows]
    if counts != [4, 8]:
        raise OracleAbstention(f"expected visible rows [4, 8], found {counts}")
    initial = [_pip_label(arr, box) for box in rows[0]]
    accesses = [_pip_label(arr, box) for box in rows[1]]
    if sorted(initial) != list(LABELS):
        raise OracleAbstention("starting rack is not a permutation of A-D")
    if sorted(accesses) != sorted(label for label in LABELS for _ in range(2)):
        raise OracleAbstention("access program must show every identity exactly twice")
    return initial, accesses


def decision_from_image(image: Image.Image) -> str:
    initial, accesses = _decode(image)
    order = list(initial)
    for label in accesses:
        index = order.index(label)
        if index < len(order) - 1:
            order[index], order[index + 1] = order[index + 1], order[index]
    return order[0]
