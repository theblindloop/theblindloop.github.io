"""Independent pixel-only inverse for the shortest-route multiplicity world."""

from __future__ import annotations

import math
from collections import deque

import numpy as np

PALETTE = {
    "black": (32, 36, 42),
    "border": (170, 180, 192),
    "green": (46, 164, 79),
    "purple": (111, 82, 200),
    "red": (211, 63, 73),
    "blue": (41, 121, 214),
    "orange": (242, 142, 43),
}


def _near(array: np.ndarray, color: tuple[int, int, int], tolerance: float) -> np.ndarray:
    delta = array.astype(np.int32) - np.asarray(color, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= tolerance * tolerance


def _components(mask: np.ndarray, minimum: int) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    result = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        xs, ys = [], []
        while queue:
            cy, cx = queue.popleft()
            xs.append(cx)
            ys.append(cy)
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(xs) >= minimum:
            result.append({
                "size": len(xs),
                "center": (float(sum(xs)) / len(xs), float(sum(ys)) / len(ys)),
                "bbox": (min(xs), min(ys), max(xs), max(ys)),
            })
    return result


def _point_segment_distance(point: tuple[float, float], left: tuple[float, float], right: tuple[float, float]) -> tuple[float, float]:
    px, py = point
    ax, ay = left
    bx, by = right
    length2 = (bx - ax) ** 2 + (by - ay) ** 2
    if length2 == 0:
        return math.inf, 0.0
    t = ((px - ax) * (bx - ax) + (py - ay) * (by - ay)) / length2
    qx, qy = ax + t * (bx - ax), ay + t * (by - ay)
    return math.hypot(px - qx, py - qy), t


def _line_present(black: np.ndarray, left: tuple[float, float], right: tuple[float, float], others: list[tuple[float, float]]) -> bool:
    distance = math.dist(left, right)
    if distance < 35:
        return False
    for point in others:
        clearance, position = _point_segment_distance(point, left, right)
        if 0.12 < position < 0.88 and clearance < 16:
            return False
    inset = min(0.24, 16.0 / distance)
    sample_count = max(20, int(distance / 2.0))
    hits = []
    height, width = black.shape
    for index in range(sample_count):
        t = inset + (1.0 - 2.0 * inset) * index / max(1, sample_count - 1)
        x = int(round(left[0] + t * (right[0] - left[0])))
        y = int(round(left[1] + t * (right[1] - left[1])))
        x0, x1 = max(0, x - 3), min(width, x + 4)
        y0, y1 = max(0, y - 3), min(height, y + 4)
        hits.append(bool(black[y0:y1, x0:x1].any()))
    return sum(hits) / len(hits) >= 0.90


def _shortest_count(nodes: list[dict], black: np.ndarray) -> int:
    centers = [node["center"] for node in nodes]
    adjacency = {index: set() for index in range(len(nodes))}
    for left_index in range(len(nodes)):
        for right_index in range(left_index + 1, len(nodes)):
            others = [center for index, center in enumerate(centers) if index not in {left_index, right_index}]
            if _line_present(black, centers[left_index], centers[right_index], others):
                adjacency[left_index].add(right_index)
                adjacency[right_index].add(left_index)
    starts = [index for index, node in enumerate(nodes) if node["kind"] == "green"]
    terminals = [index for index, node in enumerate(nodes) if node["kind"] in {"red", "blue", "orange"}]
    if len(starts) != 1 or len(terminals) != 1:
        raise ValueError("a panel must contain one green start and one colored terminal")
    start, terminal = starts[0], terminals[0]
    distance = {index: 10**9 for index in adjacency}
    ways = {index: 0 for index in adjacency}
    distance[start] = 0
    ways[start] = 1
    queue = deque([start])
    while queue:
        node = queue.popleft()
        for neighbor in sorted(adjacency[node]):
            candidate = distance[node] + 1
            if candidate < distance[neighbor]:
                distance[neighbor] = candidate
                ways[neighbor] = ways[node]
                queue.append(neighbor)
            elif candidate == distance[neighbor]:
                ways[neighbor] += ways[node]
    if distance[terminal] == 10**9:
        raise ValueError("terminal is disconnected from start")
    if distance[terminal] != 4 or len(adjacency) != 7 or sum(map(len, adjacency.values())) // 2 != 8:
        raise ValueError("recovered graph violates the public matched-panel contract")
    return ways[terminal]


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected an RGB image")
    borders = _components(_near(array, PALETTE["border"], 28), minimum=300)
    borders = [component for component in borders if component["bbox"][2] - component["bbox"][0] > 150 and component["bbox"][3] - component["bbox"][1] > 300]
    if len(borders) != 3:
        raise ValueError(f"expected three visible panel boundaries, found {len(borders)}")
    detected = []
    for kind in ("green", "purple", "red", "blue", "orange"):
        components = _components(_near(array, PALETTE[kind], 48), minimum=120)
        for component in components:
            detected.append({"kind": kind, "center": component["center"], "size": component["size"]})
    expected = {"green": 3, "purple": 15, "red": 1, "blue": 1, "orange": 1}
    observed = {kind: sum(node["kind"] == kind for node in detected) for kind in expected}
    if observed != expected:
        raise ValueError(f"node segmentation mismatch: {observed}")
    black = _near(array, PALETTE["black"], 55)
    outcomes = []
    for border in sorted(borders, key=lambda item: item["center"][0]):
        left, top, right, bottom = border["bbox"]
        nodes = [node for node in detected if left < node["center"][0] < right and top < node["center"][1] < bottom]
        if len(nodes) != 7:
            raise ValueError(f"panel contains {len(nodes)} detected circles instead of seven")
        route_count = _shortest_count(nodes, black)
        terminal_color = next(node["kind"] for node in nodes if node["kind"] in {"red", "blue", "orange"})
        outcomes.append((route_count, terminal_color))
    if sorted(route_count for route_count, _ in outcomes) != [1, 2, 3]:
        raise ValueError(f"route-count evidence is ambiguous: {outcomes}")
    return next(color for route_count, color in outcomes if route_count == 2)
