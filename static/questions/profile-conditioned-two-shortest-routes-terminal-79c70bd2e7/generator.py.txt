"""Deterministic planar graph world and analytic shortest-route decision."""

from __future__ import annotations

import copy
import math
import random
from collections import deque

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 768, 512
BACKGROUND = "#FAFBFC"
EDGE = "#20242A"
BORDER = "#AAB4C0"
START = "#2EA44F"
JUNCTION = "#6F52C8"
TERMINALS = {"red": "#D33F49", "blue": "#2979D6", "orange": "#F28E2B"}
NODE_RADIUS = 11
EDGE_WIDTH = 6

NODE_ORDER = ("S", "A", "B", "C", "D", "E", "T")
BASE_POSITIONS = {
    "S": (0.0, 0.0),
    "A": (-1.0, 1.0),
    "B": (1.0, 1.0),
    "C": (-1.0, 2.0),
    "D": (1.0, 2.0),
    "E": (0.0, 3.0),
    "T": (0.0, 4.0),
}

# Each graph has seven nodes, eight edges, shortest distance four, and sorted
# degree multiset [1,2,2,2,3,3,3].  Their shortest-route counts are 1, 2, 3.
TEMPLATE_EDGES = {
    1: (
        ("S", "A"), ("S", "B"), ("A", "B"), ("A", "C"),
        ("B", "D"), ("C", "D"), ("C", "E"), ("E", "T"),
    ),
    2: (
        ("S", "A"), ("S", "B"), ("A", "C"), ("B", "D"),
        ("C", "D"), ("C", "E"), ("D", "E"), ("E", "T"),
    ),
    3: (
        ("S", "A"), ("S", "B"), ("A", "C"), ("A", "D"),
        ("B", "D"), ("C", "E"), ("D", "E"), ("E", "T"),
    ),
}


def _graph(panel: dict) -> tuple[dict[str, set[str]], str]:
    nodes = {item["id"]: item for item in panel["nodes"]}
    adjacency = {node_id: set() for node_id in nodes}
    for left, right in panel["edges"]:
        adjacency[left].add(right)
        adjacency[right].add(left)
    terminal = next(node_id for node_id, item in nodes.items() if item["kind"] == "terminal")
    return adjacency, terminal


def _shortest_stats(panel: dict) -> tuple[int, int]:
    adjacency, terminal = _graph(panel)
    distance = {node: 10**9 for node in adjacency}
    ways = {node: 0 for node in adjacency}
    distance["S"] = 0
    ways["S"] = 1
    queue = deque(["S"])
    while queue:
        node = queue.popleft()
        for neighbor in sorted(adjacency[node]):
            candidate = distance[node] + 1
            if candidate < distance[neighbor]:
                distance[neighbor] = candidate
                ways[neighbor] = ways[node]
                queue.append(neighbor)
            elif candidate == distance[neighbor]:
                ways[neighbor] += ways[node]
    return distance[terminal], ways[terminal]


def _terminal_color(panel: dict) -> str:
    return next(item["color"] for item in panel["nodes"] if item["kind"] == "terminal")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    colors = ("red", "blue", "orange")
    offset = int(seed) % len(colors)
    color_for_count = {
        2: colors[offset],
        1: colors[(offset + 1) % 3],
        3: colors[(offset + 2) % 3],
    }
    counts = [1, 2, 3]
    rng.shuffle(counts)
    centers = (128, 384, 640)
    panels = []
    for slot, route_count in enumerate(counts):
        center_x = centers[slot]
        reflect = -1 if rng.randrange(2) else 1
        x_scale = rng.randint(39, 45)
        y_step = rng.randint(78, 83)
        x_jitter = {name: rng.randint(-3, 3) for name in NODE_ORDER}
        y_jitter = {name: rng.randint(-3, 3) for name in NODE_ORDER}
        # Keep the endpoints centered so start/terminal alignment cannot encode count.
        x_jitter["S"] = x_jitter["T"] = 0
        nodes = []
        for node_id in NODE_ORDER:
            base_x, base_y = BASE_POSITIONS[node_id]
            x = center_x + reflect * base_x * x_scale + x_jitter[node_id]
            y = 82 + base_y * y_step + y_jitter[node_id]
            if node_id == "S":
                kind, color = "start", "green"
            elif node_id == "T":
                kind, color = "terminal", color_for_count[route_count]
            else:
                kind, color = "junction", "purple"
            nodes.append({"id": node_id, "x": int(round(x)), "y": int(round(y)), "kind": kind, "color": color})
        panels.append({
            "bbox": [center_x - 108, 34, center_x + 108, 478],
            "nodes": nodes,
            "edges": [list(edge) for edge in TEMPLATE_EDGES[route_count]],
        })
    return {"panels": panels}


def _rgb_for(node: dict) -> str:
    if node["kind"] == "start":
        return START
    if node["kind"] == "junction":
        return JUNCTION
    return TERMINALS[node["color"]]


def render(scene: dict) -> Image.Image:
    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for panel in scene["panels"]:
        left, top, right, bottom = panel["bbox"]
        draw.rounded_rectangle(
            (left * scale, top * scale, right * scale, bottom * scale),
            radius=12 * scale,
            outline=BORDER,
            width=2 * scale,
        )
        nodes = {item["id"]: item for item in panel["nodes"]}
        for left_id, right_id in panel["edges"]:
            left_node, right_node = nodes[left_id], nodes[right_id]
            draw.line(
                (left_node["x"] * scale, left_node["y"] * scale,
                 right_node["x"] * scale, right_node["y"] * scale),
                fill=EDGE,
                width=EDGE_WIDTH * scale,
            )
        for node in panel["nodes"]:
            x, y = node["x"] * scale, node["y"] * scale
            radius = NODE_RADIUS * scale
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=_rgb_for(node), outline="white", width=2 * scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    # Kept self-contained so the verifier's conservative rendered-field audit
    # can see that the sole top-level label input is ``panels``.
    outcomes = []
    for panel in scene["panels"]:
        nodes = {item["id"]: item for item in panel["nodes"]}
        adjacency = {node_id: set() for node_id in nodes}
        for left, right in panel["edges"]:
            adjacency[left].add(right)
            adjacency[right].add(left)
        terminal = next(node_id for node_id, item in nodes.items() if item["kind"] == "terminal")
        distance = {node_id: 10**9 for node_id in adjacency}
        ways = {node_id: 0 for node_id in adjacency}
        distance["S"] = 0
        ways["S"] = 1
        queue = deque(["S"])
        while queue:
            node_id = queue.popleft()
            for neighbor in sorted(adjacency[node_id]):
                candidate = distance[node_id] + 1
                if candidate < distance[neighbor]:
                    distance[neighbor] = candidate
                    ways[neighbor] = ways[node_id]
                    queue.append(neighbor)
                elif candidate == distance[neighbor]:
                    ways[neighbor] += ways[node_id]
        outcomes.append((ways[terminal], nodes[terminal]["color"]))
    winners = [color for route_count, color in outcomes if route_count == 2]
    if len(winners) != 1:
        raise ValueError("scene must contain exactly one graph with two shortest routes")
    return winners[0]


def margin(scene: dict) -> float:
    counts = [_shortest_stats(panel)[1] for panel in scene["panels"]]
    if counts.count(2) != 1:
        return 0.0
    return float(min(abs(value - 2) for value in counts if value != 2))


def _orientation(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float]) -> float:
    return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0])


def _strict_cross(a: tuple[float, float], b: tuple[float, float], c: tuple[float, float], d: tuple[float, float]) -> bool:
    return _orientation(a, b, c) * _orientation(a, b, d) < 0 and _orientation(c, d, a) * _orientation(c, d, b) < 0


def is_quarantined(scene: dict) -> bool:
    try:
        if sorted(_shortest_stats(panel)[1] for panel in scene["panels"]) != [1, 2, 3]:
            return True
        for panel in scene["panels"]:
            if len(panel["nodes"]) != 7 or len(panel["edges"]) != 8 or _shortest_stats(panel)[0] != 4:
                return True
            nodes = {item["id"]: (float(item["x"]), float(item["y"])) for item in panel["nodes"]}
            edges = [tuple(edge) for edge in panel["edges"]]
            for index, (a, b) in enumerate(edges):
                for c, d in edges[index + 1:]:
                    if len({a, b, c, d}) == 4 and _strict_cross(nodes[a], nodes[b], nodes[c], nodes[d]):
                        return True
                ax, ay = nodes[a]
                bx, by = nodes[b]
                length2 = (bx - ax) ** 2 + (by - ay) ** 2
                for node_id, (px, py) in nodes.items():
                    if node_id in {a, b}:
                        continue
                    t = max(0.0, min(1.0, ((px - ax) * (bx - ax) + (py - ay) * (by - ay)) / length2))
                    clearance = math.hypot(px - (ax + t * (bx - ax)), py - (ay + t * (by - ay)))
                    if clearance < 2.4 * NODE_RADIUS:
                        return True
    except (KeyError, StopIteration, TypeError, ValueError, ZeroDivisionError):
        return True
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["panels"].reverse()
    for panel in twin["panels"]:
        panel["edges"].reverse()
        panel["nodes"].reverse()
    return [("reverse_storage_order", twin)]
