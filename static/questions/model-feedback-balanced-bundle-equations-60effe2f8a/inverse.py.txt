"""Independent PNG-only inverse for balanced bundle equations."""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

PALETTE = (
    (201, 67, 64),
    (48, 111, 176),
    (43, 147, 103),
    (222, 153, 35),
)
INK = (35, 42, 54)
CARD_RECTS = (
    (30, 58, 585, 250),
    (615, 58, 1170, 250),
    (30, 274, 585, 466),
    (615, 274, 1170, 466),
)
TRAY_RECTS = (
    (20, 538, 240, 798),
    (260, 538, 480, 798),
    (500, 538, 720, 798),
    (740, 538, 960, 798),
    (980, 538, 1180, 798),
)


def _components(mask: np.ndarray) -> list[int]:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    sizes: list[int] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        size = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            size += 1
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        sizes.append(size)
    return sizes


def _read_bundle(arr: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, int, int, int] | None:
    x0, y0, x1, y1 = box
    crop = arr[y0:y1 + 1, x0:x1 + 1]
    counts: list[int] = []
    for color in PALETTE:
        mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
        sizes = _components(mask)
        if any(not 720 <= size <= 920 for size in sizes):
            return None
        counts.append(len(sizes))
    return tuple(counts)  # type: ignore[return-value]


def _det3(matrix: list[list[int]]) -> int:
    a, b, c = matrix
    return (
        a[0] * (b[1] * c[2] - b[2] * c[1])
        - a[1] * (b[0] * c[2] - b[2] * c[0])
        + a[2] * (b[0] * c[1] - b[1] * c[0])
    )


def _solve_weights(rows: list[list[int]]) -> tuple[int, int, int, int] | None:
    solutions: set[tuple[int, int, int, int]] = set()
    for chosen in itertools.combinations(rows, 3):
        vector: list[int] = []
        for column in range(4):
            minor = [[row[j] for j in range(4) if j != column] for row in chosen]
            vector.append(((-1) ** column) * _det3(minor))
        if not any(vector):
            continue
        if next(v for v in vector if v) < 0:
            vector = [-v for v in vector]
        divisor = math.gcd(*[abs(v) for v in vector])
        vector = [v // divisor for v in vector]
        if any(v <= 0 or v > 20 for v in vector):
            continue
        if all(sum(a * b for a, b in zip(row, vector)) == 0 for row in rows):
            solutions.add(tuple(vector))  # type: ignore[arg-type]
    return next(iter(solutions)) if len(solutions) == 1 else None


def _structure_ok(arr: np.ndarray) -> bool:
    ink = np.asarray(INK, dtype=np.uint8)
    points = []
    for x0, y0, x1, _y1 in CARD_RECTS:
        points.extend((((x0 + x1) // 2, y0), (x0, y0 + 96)))
    for x0, y0, x1, _y1 in TRAY_RECTS[1:]:
        points.append(((x0 + x1) // 2, y0))
    return all(np.array_equal(arr[y, x], ink) for x, y in points)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (820, 1200, 3) or not _structure_ok(arr):
        return "abstain"

    equation_rows: list[list[int]] = []
    expected_totals = ((5, 3), (2, 1), (2, 1), (2, 2))
    for rect, expected in zip(CARD_RECTS, expected_totals):
        x0, y0, x1, y1 = rect
        middle = (x0 + x1) // 2
        left = _read_bundle(arr, (x0 + 55, y0 + 55, middle - 50, y1 - 20))
        right = _read_bundle(arr, (middle + 50, y0 + 55, x1 - 55, y1 - 20))
        if left is None or right is None or (sum(left), sum(right)) != expected:
            return "abstain"
        equation_rows.append([a - b for a, b in zip(left, right)])
    weights = _solve_weights(equation_rows)
    if weights is None:
        return "abstain"

    bundles: list[tuple[int, int, int, int]] = []
    for rect in TRAY_RECTS:
        x0, y0, x1, y1 = rect
        counts = _read_bundle(arr, (x0 + 22, y0 + 88, x1 - 22, y1 - 25))
        if counts is None or sum(counts) != 4:
            return "abstain"
        bundles.append(counts)

    target_total = sum(a * b for a, b in zip(bundles[0], weights))
    matches = [index for index, counts in enumerate(bundles[1:], 1) if sum(a * b for a, b in zip(counts, weights)) == target_total]
    return f"tray-{matches[0]}" if len(matches) == 1 else "abstain"
