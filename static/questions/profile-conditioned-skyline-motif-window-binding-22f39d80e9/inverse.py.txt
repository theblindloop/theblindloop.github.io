"""Independent pixel-only decision arm for skyline_motif_window_binding.

This module receives a rasterized RGB image and nothing else.  It imports no
renderer, prompt, generator, verifier or analytic-gold code, and it never sees
latent scene data.  Everything below is measured from pixels:

1. locate the widest solid black rule -- the panel separator;
2. segment the gray bars above and below it into connected column runs and
   measure each bar's pixel height from its own panel baseline;
3. segment the colored flags and order them left to right;
4. slide the 3-bar template over the 9-bar track and keep the unique window
   whose three measured heights all agree exactly;
5. only then read the bound attribute: the flag color three bars to the right
   of the window start.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

# Public question contract: the four answerable flag colors.
_PALETTE: dict[str, tuple[int, int, int]] = {
    "crimson": (201, 32, 48),
    "amber": (235, 160, 20),
    "teal": (24, 150, 150),
    "violet": (139, 61, 191),
}

_GRAY = (95, 105, 120)
_TOL = 26
_READOUT_OFFSET = 3

AMBIGUOUS = "ambiguous"


def _mask_near(array: np.ndarray, color, tol: int = _TOL) -> np.ndarray:
    target = np.array(color, dtype=np.int16)
    return (np.abs(array.astype(np.int16) - target).max(axis=2) <= tol)


def _column_runs(mask: np.ndarray, min_width: int = 6) -> list[tuple[int, int]]:
    """Contiguous column intervals in which the mask has any pixel."""
    occupied = mask.any(axis=0)
    runs: list[tuple[int, int]] = []
    start = None
    for x, flag in enumerate(occupied):
        if flag and start is None:
            start = x
        elif not flag and start is not None:
            if x - start >= min_width:
                runs.append((start, x - 1))
            start = None
    if start is not None and len(occupied) - start >= min_width:
        runs.append((start, len(occupied) - 1))
    return runs


def _bar_heights(mask: np.ndarray) -> list[int]:
    """Pixel height of every gray bar in a panel, ordered left to right."""
    heights: list[int] = []
    for x0, x1 in _column_runs(mask):
        rows = np.nonzero(mask[:, x0 : x1 + 1].any(axis=1))[0]
        if rows.size == 0:
            continue
        heights.append(int(rows.max() - rows.min() + 1))
    return heights


def _flag_colors(array: np.ndarray) -> list[str]:
    """Flag color names, ordered left to right, from the colored squares."""
    combined = np.zeros(array.shape[:2], dtype=bool)
    per_color = {}
    for name, rgb in _PALETTE.items():
        mask = _mask_near(array, rgb)
        per_color[name] = mask
        combined |= mask
    names: list[str] = []
    for x0, x1 in _column_runs(combined, min_width=8):
        best_name, best_count = None, 0
        for name, mask in per_color.items():
            count = int(mask[:, x0 : x1 + 1].sum())
            if count > best_count:
                best_name, best_count = name, count
        if best_name is not None:
            names.append(best_name)
    return names


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    height, width = array.shape[:2]

    black = _mask_near(array, (0, 0, 0), tol=40)
    separator_y, separator_span = None, 0
    for y in range(height):
        xs = np.nonzero(black[y])[0]
        if xs.size == 0:
            continue
        span = int(xs.max() - xs.min() + 1)
        if span > separator_span:
            separator_y, separator_span = y, span
    if separator_y is None:
        return AMBIGUOUS

    gray = _mask_near(array, _GRAY)
    motif = _bar_heights(gray[:separator_y, :])
    track = _bar_heights(gray[separator_y + 1 :, :])
    flags = _flag_colors(array[separator_y + 1 :, :])

    if len(motif) < 2 or len(track) <= len(motif) or len(flags) != len(track):
        return AMBIGUOUS

    hits = [
        start
        for start in range(len(track) - len(motif) + 1)
        if all(track[start + k] == motif[k] for k in range(len(motif)))
    ]
    if len(hits) != 1:
        return AMBIGUOUS

    readout = hits[0] + _READOUT_OFFSET
    if not 0 <= readout < len(flags):
        return AMBIGUOUS
    return flags[readout]
