"""skyline_motif_window_binding -- template-search correspondence world.

An upper panel shows a short TEMPLATE of three gray bars.  A lower panel shows a
TRACK of nine gray bars drawn at the same width, pitch and vertical scale on a
common baseline, each carrying one colored flag directly above it.  Exactly one
run of three consecutive track bars reproduces the template heights bar for bar.
The decision is the flag color above the track bar immediately to the right of
that unique matching run.

The correspondence is established by matching a *sequence of measured local
parts*; only after the run is located does the bound attribute (the flag color)
become readable.  Neither color, position, size, count nor total ink identifies
the answer.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

# ---------------------------------------------------------------- public palette

# The four flag colors.  These names are part of the public question contract.
FLAG_COLORS: dict[str, tuple[int, int, int]] = {
    "crimson": (201, 32, 48),
    "amber": (235, 160, 20),
    "teal": (24, 150, 150),
    "violet": (139, 61, 191),
}
COLOR_ORDER: tuple[str, ...] = ("crimson", "amber", "teal", "violet")

BAR_GRAY = (95, 105, 120)
BACKGROUND = (255, 255, 255)
RULE_COLOR = (0, 0, 0)

# ---------------------------------------------------------------- geometry (px)

WIDTH = 560
HEIGHT = 360

BAR_W = 34
PITCH = 46

N_TRACK = 9
N_MOTIF = 3
READOUT_OFFSET = 3  # flag read above bar (window_start + READOUT_OFFSET)

MOTIF_BASELINE = 150
TRACK_BASELINE = 330
SEPARATOR_Y = 163

FLAG_TOP = 176
FLAG_SIZE = 22

TRACK_X0 = (WIDTH - (N_TRACK * PITCH - (PITCH - BAR_W))) // 2
MOTIF_X0 = (WIDTH - (N_MOTIF * PITCH - (PITCH - BAR_W))) // 2

# Height grid: multiples of 6 keep well-separated decoys, tight decoys are
# injected explicitly by the sampler.
HEIGHT_LEVELS = tuple(range(24, 109, 6))

QUARANTINE_MARGIN = 6.0


# ---------------------------------------------------------------- scene sampling


def _windows(track: list[int]) -> int:
    return len(track) - N_MOTIF + 1


def _window_gap(track: list[int], motif: list[int], start: int) -> int:
    """Chebyshev height discrepancy, in pixels, of one candidate window."""
    return max(abs(track[start + k] - motif[k]) for k in range(N_MOTIF))


def _decoy_margin(track: list[int], motif: list[int], match_index: int) -> int:
    gaps = [
        _window_gap(track, motif, j)
        for j in range(_windows(track))
        if j != match_index
    ]
    return min(gaps) if gaps else 10 ** 6


def _solve(track: list[int], motif: list[int]) -> int:
    """Index of the unique exactly-matching window (-1 when not unique)."""
    hits = [j for j in range(_windows(track)) if _window_gap(track, motif, j) == 0]
    return hits[0] if len(hits) == 1 else -1


def _sample_flags(rng: random.Random) -> list[str]:
    """Nine flags with color multiplicities {3,2,2,2}: no color is a singleton,
    so neither rarity nor abundance can identify the answer."""
    counts = [3, 2, 2, 2]
    colors = list(COLOR_ORDER)
    rng.shuffle(colors)
    pool: list[str] = []
    for name, count in zip(colors, counts):
        pool.extend([name] * count)
    rng.shuffle(pool)
    return pool


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    for _attempt in range(4000):
        track = [rng.choice(HEIGHT_LEVELS) for _ in range(N_TRACK)]
        match_index = rng.randrange(0, N_TRACK - READOUT_OFFSET)
        motif = track[match_index : match_index + N_MOTIF]

        if _solve(track, motif) != match_index:
            continue

        tag = "wide"
        # Roughly one scene in three gets an explicit near-miss decoy so that the
        # near-boundary and quarantine bands are actually populated.
        if rng.random() < 0.38:
            far = [
                j
                for j in range(_windows(track))
                if abs(j - match_index) >= N_MOTIF
            ]
            if far:
                j = rng.choice(far)
                delta = rng.choice([2, 3, 4, 5, 6, 8, 10])
                sign = rng.choice([-1, 1])
                odd = rng.randrange(N_MOTIF)
                trial = list(track)
                ok = True
                for k in range(N_MOTIF):
                    value = motif[k] + (sign * delta if k == odd else 0)
                    if not 18 <= value <= 114:
                        ok = False
                        break
                    trial[j + k] = value
                if ok and _solve(trial, motif) == match_index:
                    track = trial
                    tag = "tight"

        margin = _decoy_margin(track, motif, match_index)
        if margin < 2:  # never leave a pixel-indistinguishable duplicate run
            continue

        return {
            "seed": int(seed),
            "motif": [int(h) for h in motif],
            "track_heights": [int(h) for h in track],
            "flag_colors": _sample_flags(rng),
            "match_index": int(match_index),
            "sampler_tag": tag,
        }

    raise RuntimeError(f"could not sample a valid scene for seed {seed}")


# ---------------------------------------------------------------- rendering


def render(scene: dict) -> Image.Image:
    motif = [int(h) for h in scene["motif"]]
    track = [int(h) for h in scene["track_heights"]]
    flags = list(scene["flag_colors"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    # Panel rule: the template panel above, the track panel below.
    draw.rectangle([24, SEPARATOR_Y, WIDTH - 25, SEPARATOR_Y + 1], fill=RULE_COLOR)

    for index, height in enumerate(motif):
        x = MOTIF_X0 + index * PITCH
        draw.rectangle(
            [x, MOTIF_BASELINE - height, x + BAR_W - 1, MOTIF_BASELINE - 1],
            fill=BAR_GRAY,
        )
    draw.rectangle(
        [MOTIF_X0 - 8, MOTIF_BASELINE, MOTIF_X0 + N_MOTIF * PITCH - 4, MOTIF_BASELINE + 2],
        fill=RULE_COLOR,
    )

    for index, height in enumerate(track):
        x = TRACK_X0 + index * PITCH
        draw.rectangle(
            [x, TRACK_BASELINE - height, x + BAR_W - 1, TRACK_BASELINE - 1],
            fill=BAR_GRAY,
        )
        flag_x = x + (BAR_W - FLAG_SIZE) // 2
        draw.rectangle(
            [flag_x, FLAG_TOP, flag_x + FLAG_SIZE - 1, FLAG_TOP + FLAG_SIZE - 1],
            fill=FLAG_COLORS[flags[index]],
        )
    draw.rectangle(
        [TRACK_X0 - 8, TRACK_BASELINE, TRACK_X0 + N_TRACK * PITCH - 4, TRACK_BASELINE + 2],
        fill=RULE_COLOR,
    )
    return image


# ---------------------------------------------------------------- analytic gold


def analytic_gold(scene: dict) -> str:
    """Flag color above the bar immediately right of the unique matching run.

    Reads exactly three scene fields, all of which are drawn into the raster:
    ``motif``, ``track_heights`` and ``flag_colors``.  The stored
    ``match_index`` is never consulted -- the run is re-located by search.
    """
    motif = [int(h) for h in scene["motif"]]
    track = [int(h) for h in scene["track_heights"]]
    flags = list(scene["flag_colors"])

    hits = []
    for start in range(len(track) - len(motif) + 1):
        if all(track[start + k] == motif[k] for k in range(len(motif))):
            hits.append(start)
    if len(hits) != 1:
        raise ValueError("scene does not contain a unique matching run")
    return str(flags[hits[0] + READOUT_OFFSET])


def margin(scene: dict) -> float:
    """Runner-up decision margin: the Chebyshev height discrepancy, in pixels, of
    the best non-matching window.  Zero would mean two runs match equally well."""
    motif = [int(h) for h in scene["motif"]]
    track = [int(h) for h in scene["track_heights"]]
    return float(_decoy_margin(track, motif, _solve(track, motif)))


def is_quarantined(scene: dict) -> bool:
    """A runner-up within QUARANTINE_MARGIN pixels is not safely separable at
    finite raster resolution and is excluded from headline claims."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Render-invariant reparametrizations that must preserve the gold label.

    Both bookkeeping fields below are excluded from rendering *and* from the
    analytic gold; scrambling them is pixel-identical and label-preserving,
    which is the concrete evidence that the label is recomputed from the drawn
    structure rather than read off a stored answer key.
    """
    twins: list[tuple[str, dict]] = []

    scrambled = dict(scene)
    upper = len(scene["track_heights"]) - len(scene["motif"])
    scrambled["match_index"] = (int(scene["match_index"]) + 2) % (upper + 1)
    twins.append(("redundant_answer_key_scramble", scrambled))

    relabeled = dict(scene)
    relabeled["sampler_tag"] = "relabeled"
    twins.append(("sampler_tag_relabel", relabeled))

    return twins
