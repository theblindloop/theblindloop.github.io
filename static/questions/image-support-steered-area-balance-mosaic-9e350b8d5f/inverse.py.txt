"""Independent image-only inverse for the area-balance mosaic.

It segments the saturated landmass, verifies the public exact-area contract,
then compares the mask's horizontal centroid with the neutral board center.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

CELL_COUNT = 42
ALLOWED_TILE_SIZES = (46, 48, 50)


def _colored_mask(array: np.ndarray) -> np.ndarray:
    values = array.astype(np.int16)
    chroma = values.max(axis=2) - values.min(axis=2)
    return (chroma >= 55) & (values.min(axis=2) < 205)


def _board_center(array: np.ndarray) -> float:
    values = array.astype(np.int16)
    chroma = values.max(axis=2) - values.min(axis=2)
    gray = values.mean(axis=2)
    # Neutral board fill/border, excluding the near-white outer background.
    neutral_board = (chroma < 25) & (gray < 246)
    counts = neutral_board.sum(axis=0)
    xs = np.flatnonzero(counts >= 40)
    if len(xs) < 100:
        raise ValueError("board reference is missing or corrupted")
    return (float(xs.min()) + float(xs.max())) / 2.0


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (512, 512, 3):
        raise ValueError("unexpected canvas")
    mask = _colored_mask(array)
    area = int(mask.sum())
    expected = [CELL_COUNT * size * size for size in ALLOWED_TILE_SIZES]
    if min(abs(area - value) for value in expected) > 4:
        raise ValueError("the declared 42-tile landmass is missing or damaged")
    ys, xs = np.nonzero(mask)
    if len(xs) != area:
        raise ValueError("landmass segmentation failed")
    center = _board_center(array)
    centroid = float(xs.mean())
    if abs(centroid - center) < 0.20:
        raise ValueError("balance is raster-ambiguous")
    return "left" if centroid < center else "right"
