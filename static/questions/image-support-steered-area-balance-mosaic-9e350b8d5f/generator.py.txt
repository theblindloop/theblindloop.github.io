"""Deterministic latent sampler and renderer for the area-balance mosaic."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

BOARD_N = 8
CELL_COUNT = 42
CANVAS = 512
PALETTES = (
    ("#168C8C", "#FCFBF7", "#D9D8D2", "#242424"),
    ("#C85A28", "#FBFCF8", "#D7D9D3", "#202020"),
    ("#6D55B3", "#FCFAFD", "#DAD6DC", "#252225"),
    ("#2878B5", "#FBFCFE", "#D6DADF", "#22252A"),
)


def _connected(cells: set[tuple[int, int]]) -> bool:
    if not cells:
        return False
    seen = {next(iter(cells))}
    queue = deque(seen)
    while queue:
        x, y = queue.popleft()
        for point in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if point in cells and point not in seen:
                seen.add(point)
                queue.append(point)
    return len(seen) == len(cells)


def _balance(cells: set[tuple[int, int]]) -> float:
    return sum(x - 3.5 for x, _ in cells) / len(cells)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    wanted = "left" if int(seed) % 2 == 0 else "right"
    near = int(seed) % 11 == 0
    universe = [(x, y) for y in range(BOARD_N) for x in range(BOARD_N)]
    for _ in range(30000):
        cells = set(rng.sample(universe, CELL_COUNT))
        if not _connected(cells):
            continue
        # A broad landmass must span the board in both dimensions.
        if {x for x, _ in cells} != set(range(BOARD_N)):
            continue
        if {y for _, y in cells} != set(range(BOARD_N)):
            continue
        value = _balance(cells)
        if near:
            if not (0.0 < abs(value) <= 0.095):
                continue
        elif wanted == "left":
            if value > -0.26:
                continue
        elif value < 0.26:
            continue
        tile_size = rng.choice((46, 48, 50))
        board_size = BOARD_N * tile_size
        origin_x = (CANVAS - board_size) // 2 + rng.randint(-10, 10)
        origin_y = (CANVAS - board_size) // 2 + rng.randint(-10, 10)
        ink, background, grid_color, line_color = rng.choice(PALETTES)
        return {
            "cells": [[x, y] for x, y in sorted(cells, key=lambda p: (p[1], p[0]))],
            "tile_size": tile_size,
            "origin_x": origin_x,
            "origin_y": origin_y,
            "ink": ink,
            "background": background,
            "grid_color": grid_color,
            "line_color": line_color,
        }
    raise RuntimeError("could not sample a balanced connected landmass")


def render(scene: dict) -> Image.Image:
    size = int(scene["tile_size"])
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    board = BOARD_N * size
    image = Image.new("RGB", (CANVAS, CANVAS), scene["background"])
    draw = ImageDraw.Draw(image)
    # A visible neutral board fixes the reference frame without encoding the answer.
    draw.rectangle(
        (ox - 4, oy - 4, ox + board + 3, oy + board + 3),
        fill="#F0F0EC",
        outline=scene["grid_color"],
        width=3,
    )
    center = ox + board // 2
    draw.line((center, oy - 18, center, oy + board + 17), fill=scene["line_color"], width=3)
    draw.polygon(
        ((center - 7, oy - 14), (center + 7, oy - 14), (center, oy - 5)),
        fill=scene["line_color"],
    )
    draw.polygon(
        ((center - 7, oy + board + 13), (center + 7, oy + board + 13), (center, oy + board + 4)),
        fill=scene["line_color"],
    )
    # Tiles meet without gaps: the colored evidence is one connected area.
    for x, y in scene["cells"]:
        x0, y0 = ox + int(x) * size, oy + int(y) * size
        draw.rectangle((x0, y0, x0 + size - 1, y0 + size - 1), fill=scene["ink"])
    # Redraw only the external ticks; the line stays behind the landmass.
    draw.line((center, oy - 18, center, oy - 5), fill=scene["line_color"], width=3)
    draw.line((center, oy + board + 4, center, oy + board + 17), fill=scene["line_color"], width=3)
    return image


def analytic_gold(scene: dict) -> str:
    cells = {(int(x), int(y)) for x, y in scene["cells"]}
    return "left" if _balance(cells) < 0 else "right"


def margin(scene: dict) -> float:
    cells = {(int(x), int(y)) for x, y in scene["cells"]}
    return abs(_balance(cells))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.10


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
