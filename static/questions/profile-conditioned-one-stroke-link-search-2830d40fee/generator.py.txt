"""Deterministic scene sampler and renderer for One-Stroke Link Search.

Each panel is a connected nine-node graph on a 3x3 lattice with exactly nine
unit-length links.  The winner, and only the winner, has an Euler trail: a
connected graph has such a trail precisely when it has zero or two odd-degree
vertices.  Equal node/link counts and equal link lengths block ink and count
shortcuts.
"""

from __future__ import annotations

import copy
import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 700
SCALE = 2
BACKGROUND = (248, 247, 242)
INK = (39, 45, 55)
NODE = (24, 151, 164)
FRAME = (184, 190, 196)
ACCENT = (220, 125, 55)
LABELS = ("A", "B", "C", "D")


def _all_grid_edges() -> list[tuple[int, int]]:
    edges = []
    for row in range(3):
        for col in range(3):
            node = row * 3 + col
            if col < 2:
                edges.append((node, node + 1))
            if row < 2:
                edges.append((node, node + 3))
    return edges


GRID_EDGES = _all_grid_edges()


def _graph_stats(edges: list[list[int]] | tuple[tuple[int, int], ...]) -> tuple[bool, int]:
    neighbors = [set() for _ in range(9)]
    for raw_u, raw_v in edges:
        u, v = int(raw_u), int(raw_v)
        neighbors[u].add(v)
        neighbors[v].add(u)
    seen = {0}
    pending = [0]
    while pending:
        node = pending.pop()
        for other in neighbors[node]:
            if other not in seen:
                seen.add(other)
                pending.append(other)
    odd = sum(len(items) % 2 for items in neighbors)
    return len(seen) == 9, odd


def _pools() -> tuple[list[tuple[tuple[int, int], ...]], list[tuple[tuple[int, int], ...]]]:
    winners, distractors = [], []
    for edges in itertools.combinations(GRID_EDGES, 9):
        connected, odd = _graph_stats(edges)
        if not connected:
            continue
        (winners if odd == 2 else distractors).append(edges)
    return winners, distractors


WINNERS, DISTRACTORS = _pools()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = rng.choice(WINNERS)
    distractors = rng.sample(DISTRACTORS, 3)
    graphs = [winner, *distractors]
    panel_order = list(range(4))
    rng.shuffle(panel_order)
    return {
        "panel_edges": [
            [[int(u), int(v)] for u, v in graph]
            for graph in graphs
        ],
        "panel_order": panel_order,
        "style_variant": rng.randrange(1_000_000),
    }


def _display_graphs(scene: dict) -> list[list[list[int]]]:
    graphs = scene["panel_edges"]
    order = scene["panel_order"]
    return [graphs[int(index)] for index in order]


def _defect(edges: list[list[int]]) -> int:
    connected, odd = _graph_stats(edges)
    if not connected:
        return 10
    return min(abs(odd), abs(odd - 2))


def analytic_gold(scene: dict) -> str:
    defects = [_defect(edges) for edges in _display_graphs(scene)]
    winners = [index for index, defect in enumerate(defects) if defect == 0]
    if len(winners) != 1:
        raise ValueError("scene does not have a unique one-stroke network")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    defects = sorted(_defect(edges) for edges in _display_graphs(scene))
    return float(defects[1] - defects[0])


def is_quarantined(scene: dict) -> bool:
    defects = [_defect(edges) for edges in _display_graphs(scene)]
    return defects.count(0) != 1 or sorted(defects)[1] - sorted(defects)[0] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["style_variant"] = (int(scene.get("style_variant", 0)) + 1) % 1_000_000
    return [("unused_style_nonce", twin)]


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size * SCALE)
    except OSError:
        return ImageFont.load_default()


def _point(panel_x: int, panel_y: int, node: int) -> tuple[int, int]:
    row, col = divmod(node, 3)
    return (panel_x + 150 + col * 70, panel_y + 57 + row * 70)


def render(scene: dict) -> Image.Image:
    graphs = _display_graphs(scene)
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def box(coords, **kwargs):
        draw.rounded_rectangle(tuple(int(v * SCALE) for v in coords), radius=14 * SCALE, **kwargs)

    draw.text((40 * SCALE, 22 * SCALE), "ONE-STROKE LINK SEARCH", fill=INK,
              font=_font(24, True))
    draw.text((40 * SCALE, 60 * SCALE),
              "Use every dark link exactly once. Do not lift or retrace; start at any circle.",
              fill=INK, font=_font(17))
    panel_origins = ((40, 120), (520, 120), (40, 405), (520, 405))
    for index, ((px, py), edges) in enumerate(zip(panel_origins, graphs)):
        box((px, py, px + 440, py + 245), fill=(255, 255, 255),
            outline=FRAME, width=3 * SCALE)
        draw.ellipse(((px + 16) * SCALE, (py + 14) * SCALE,
                      (px + 54) * SCALE, (py + 52) * SCALE), fill=ACCENT)
        label = LABELS[index]
        bbox = draw.textbbox((0, 0), label, font=_font(21, True))
        tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
        draw.text(((px + 35) * SCALE - tw / 2, (py + 33) * SCALE - th / 2 - 2 * SCALE),
                  label, fill=(255, 255, 255), font=_font(21, True))
        points = [_point(px, py, node) for node in range(9)]
        for u, v in edges:
            a, b = points[int(u)], points[int(v)]
            draw.line((a[0] * SCALE, a[1] * SCALE, b[0] * SCALE, b[1] * SCALE),
                      fill=INK, width=9 * SCALE)
        for x, y in points:
            radius = 15
            draw.ellipse(((x - radius) * SCALE, (y - radius) * SCALE,
                          (x + radius) * SCALE, (y + radius) * SCALE),
                         fill=NODE, outline=(255, 255, 255), width=3 * SCALE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
