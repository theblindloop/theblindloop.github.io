"""Independent pixel-only inverse for One-Stroke Link Search.

The implementation receives a raster only.  It segments the teal node disks,
discovers the four panels from the largest gaps between node centroids, rebuilds
the twelve possible lattice adjacencies by measuring dark pixels at link
midpoints, and evaluates every recovered graph.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        pending = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while pending:
            cy, cx = pending.pop()
            xs.append(cx)
            ys.append(cy)
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    pending.append((ny, nx))
        if len(xs) >= 250:
            result.append((float(sum(xs)) / len(xs), float(sum(ys)) / len(ys), len(xs)))
    return result


def _largest_split(values: list[float]) -> float:
    ordered = sorted(values)
    gaps = [(ordered[i + 1] - ordered[i], i) for i in range(len(ordered) - 1)]
    _, index = max(gaps)
    return (ordered[index] + ordered[index + 1]) / 2.0


def _dark_fraction(array: np.ndarray, x: float, y: float, radius: int = 7) -> float:
    xi, yi = int(round(x)), int(round(y))
    crop = array[max(0, yi - radius):yi + radius + 1, max(0, xi - radius):xi + radius + 1]
    if crop.size == 0:
        return 0.0
    dark = (crop[:, :, 0] < 90) & (crop[:, :, 1] < 90) & (crop[:, :, 2] < 100)
    return float(dark.mean())


def _recover_edges(array: np.ndarray, nodes: list[tuple[float, float, int]]) -> list[tuple[int, int]] | None:
    if len(nodes) != 9:
        return None
    ordered_x = sorted(nodes, key=lambda item: item[0])
    columns = [ordered_x[i:i + 3] for i in (0, 3, 6)]
    x_centers = [sum(item[0] for item in group) / 3.0 for group in columns]
    ordered_y = sorted(nodes, key=lambda item: item[1])
    rows = [ordered_y[i:i + 3] for i in (0, 3, 6)]
    y_centers = [sum(item[1] for item in group) / 3.0 for group in rows]
    if min(x_centers[i + 1] - x_centers[i] for i in range(2)) < 35:
        return None
    if min(y_centers[i + 1] - y_centers[i] for i in range(2)) < 35:
        return None
    grid: dict[tuple[int, int], tuple[float, float]] = {}
    for x, y, _ in nodes:
        col = min(range(3), key=lambda i: abs(x - x_centers[i]))
        row = min(range(3), key=lambda i: abs(y - y_centers[i]))
        if (row, col) in grid:
            return None
        grid[(row, col)] = (x, y)
    if len(grid) != 9:
        return None
    edges = []
    for row in range(3):
        for col in range(3):
            u = row * 3 + col
            for dr, dc in ((0, 1), (1, 0)):
                other = (row + dr, col + dc)
                if other not in grid:
                    continue
                a, b = grid[(row, col)], grid[other]
                if _dark_fraction(array, (a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0) > 0.28:
                    edges.append((u, other[0] * 3 + other[1]))
    return edges if len(edges) == 9 else None


def _has_euler_trail(edges: list[tuple[int, int]]) -> bool:
    neighbors = [set() for _ in range(9)]
    for u, v in edges:
        neighbors[u].add(v)
        neighbors[v].add(u)
    seen = {0}
    pending = deque([0])
    while pending:
        node = pending.popleft()
        for other in neighbors[node]:
            if other not in seen:
                seen.add(other)
                pending.append(other)
    return len(seen) == 9 and sum(len(items) % 2 for items in neighbors) in (0, 2)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    teal = (array[:, :, 0] < 90) & (array[:, :, 1] > 105) & (array[:, :, 2] > 110)
    nodes = _components(teal)
    if len(nodes) != 36:
        return "ABSTAIN"
    x_split = _largest_split([item[0] for item in nodes])
    y_split = _largest_split([item[1] for item in nodes])
    panels = [[], [], [], []]
    for node in nodes:
        col = int(node[0] > x_split)
        row = int(node[1] > y_split)
        panels[row * 2 + col].append(node)
    graphs = [_recover_edges(array, panel) for panel in panels]
    if any(graph is None for graph in graphs):
        return "ABSTAIN"
    winners = [index for index, graph in enumerate(graphs) if _has_euler_trail(graph)]
    return LABELS[winners[0]] if len(winners) == 1 else "ABSTAIN"
