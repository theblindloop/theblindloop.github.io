"""Rectangle-area (width x height) argmax selection world.

Latent scene z carries a set of colored filled rectangles, each confined to its
own horizontal column on a shared light panel. Each rectangle has an
independently sampled width and height. The task is a genuine 2D compositional
measurement: find the rectangle whose AREA (width times height) is largest.

Because both extents vary independently across the three rectangles, neither
the tallest rectangle, nor the widest rectangle, nor any single extent, total
ink, or position discloses the winner. Only recovering BOTH the width and the
height of every rectangle and forming their product separates the colours. This
isolates whether the model can compose two independent axis extents into an
area quantity, distinct from measuring a single extent (mean_height_compare,
widest_gap), a width/height ratio (nearest_square_aspect), or a circular
sector's area by pixel count (pie_sector_fraction).

The independent pixel oracle recovers each rectangle's width and height from
its colour-segmented bounding box and forms the product, then selects the
colour with the largest product.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "rectangle_area_argmax-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
NCOLUMNS = 3
COLUMN_WIDTH = WIDTH // NCOLUMNS

EDGE_MARGIN = 8

# Extent sampling ranges (px).
WIDTH_RANGE = (40.0, 90.0)
HEIGHT_RANGE = (60.0, 175.0)

# Minimum top-two AREA gap (px^2) required for a decidable selection. The pixel
# oracle measures each bounding-box extent exactly (aliased solid fill), so its
# area product is accurate to a few hundred px^2; a gap of 2600 px^2 is far
# above that noise.
MIN_MARGIN = 2600.0
# Below this top-two area gap the scene is quarantined as a finite-resolution
# near-boundary case.
QUARANTINE_MARGIN = 3200.0


def _sample_rectangle(rng, color: str, width: float, height: float, center_x: float) -> dict:
    """Place one filled rectangle of the given width and height centred on center_x."""
    x0 = center_x - width / 2.0
    x1 = center_x + width / 2.0
    y0 = (HEIGHT - height) / 2.0 + rng.uniform(-40.0, 40.0)
    y1 = y0 + height
    if (
        EDGE_MARGIN <= x0 <= WIDTH - EDGE_MARGIN
        and EDGE_MARGIN <= x1 <= WIDTH - EDGE_MARGIN
        and EDGE_MARGIN <= y0 <= HEIGHT - EDGE_MARGIN
        and EDGE_MARGIN <= y1 <= HEIGHT - EDGE_MARGIN
    ):
        return {"color": color, "x0": x0, "y0": y0, "x1": x1, "y1": y1}
    return None


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed, counterbalanced largest-area bar."""
    rng = random.Random(seed * 1049 + 13)
    while True:
        widths = [rng.uniform(*WIDTH_RANGE) for _ in range(NCOLUMNS)]
        heights = [rng.uniform(*HEIGHT_RANGE) for _ in range(NCOLUMNS)]
        areas = [w * h for w, h in zip(widths, heights)]
        areas_sorted = sorted(areas, reverse=True)
        if areas_sorted[0] - areas_sorted[1] >= MIN_MARGIN:
            break
    colors = list(PALETTE)
    rng.shuffle(colors)
    rectangles: list[dict] = []
    while True:
        rectangles = []
        valid = True
        for idx, (color, width, height) in enumerate(zip(colors, widths, heights)):
            center_x = idx * COLUMN_WIDTH + COLUMN_WIDTH / 2.0
            rect = _sample_rectangle(rng, color, width, height, center_x)
            if rect is None:
                valid = False
                break
            rectangles.append(rect)
        if valid:
            break
    return {
        "rectangles": rectangles,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three filled rectangles, one per colour, in three columns."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for rect in scene["rectangles"]:
        draw.rectangle(
            [(rect["x0"], rect["y0"]), (rect["x1"], rect["y1"])],
            fill=COLOR_RGB[rect["color"]],
        )
    return img


def _area(x0: float, y0: float, x1: float, y1: float) -> float:
    return (x1 - x0) * (y1 - y0)


def _largest(rectangles: list[dict]) -> tuple[dict, float, float]:
    """Return (largest rectangle, its area, the top-two area gap)."""
    keyed = [
        (rect, _area(rect["x0"], rect["y0"], rect["x1"], rect["y1"]))
        for rect in rectangles
    ]
    keyed.sort(key=lambda item: item[1], reverse=True)
    top, second = keyed[0], keyed[1]
    return top[0], top[1], top[1] - second[1]


def analytic_gold(scene: dict) -> str:
    """The colour of the rectangle with the largest area (width times height)."""
    top, _area_v, _gap = _largest(scene["rectangles"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two AREA gap in px^2."""
    _top, _area_v, gap = _largest(scene["rectangles"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two area gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each rectangle's colour label and its four corners) is
    directly rendered as visible ink. Swapping two rectangles' colour labels
    changes which colour has the largest area, and any global reflection,
    rotation, or permutation turns the raster into a distinct image, so there is
    no hidden alias transformation that is both pixel-identical and
    gold-preserving.
    """
    return []
