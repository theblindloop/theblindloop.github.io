"""Independent pixel-only inverse arm for the rectangle-area world.

This module receives ONLY the rasterized PNG (plus the world's fixed public
palette) and re-measures which coloured rectangle has the largest area (width
times height). It shares no geometry with renderer.render() and never imports
the renderer, prompts, generator, verifier, or any analytical-gold code. It
cannot be fed latent scene data.

For each colour it labels its ink mask, takes its axis-aligned bounding box,
recovers the rectangle's width and height as the box extents, and forms the
area as their product (a composition of two independent extents). The answer is
the colour whose recovered area product is largest.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "rectangle_area_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

_MIN_PIXELS = 10


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colours + background)."""
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _area_from_mask(mask: np.ndarray) -> float:
    """Area = (bounding-box width) * (bounding-box height) of the ink mask."""
    ys, xs = np.nonzero(mask)
    if len(ys) < _MIN_PIXELS:
        return -1.0
    width = float(xs.max() - xs.min() + 1)
    height = float(ys.max() - ys.min() + 1)
    return width * height


def decision_from_image(image: Image.Image) -> str:
    """Return the colour of the largest-area rectangle from pixels."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_area = -1.0
    for idx, color in enumerate(PALETTE):
        area = _area_from_mask(labels == idx)
        if area > best_area:
            best_area = area
            best_color = color
    return best_color
