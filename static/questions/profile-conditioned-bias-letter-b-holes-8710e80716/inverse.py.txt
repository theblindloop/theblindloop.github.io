"""Independent image-only oracle based on complement topology."""

from __future__ import annotations

from collections import deque

from PIL import Image


def _components(mask: list[bytearray], wanted: int) -> list[tuple[int, bool, tuple[int, int, int, int]]]:
    height = len(mask)
    width = len(mask[0]) if height else 0
    seen = [bytearray(width) for _ in range(height)]
    found: list[tuple[int, bool, tuple[int, int, int, int]]] = []
    for y0 in range(height):
        for x0 in range(width):
            if seen[y0][x0] or mask[y0][x0] != wanted:
                continue
            queue = deque([(x0, y0)])
            seen[y0][x0] = 1
            area = 0
            border = False
            min_x = max_x = x0
            min_y = max_y = y0
            while queue:
                x, y = queue.popleft()
                area += 1
                border = border or x == 0 or y == 0 or x == width - 1 or y == height - 1
                min_x = min(min_x, x)
                max_x = max(max_x, x)
                min_y = min(min_y, y)
                max_y = max(max_y, y)
                for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                    if 0 <= nx < width and 0 <= ny < height and not seen[ny][nx] and mask[ny][nx] == wanted:
                        seen[ny][nx] = 1
                        queue.append((nx, ny))
            found.append((area, border, (min_x, min_y, max_x, max_y)))
    return found


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 64 or height < 64:
        return "abstain"
    pixels = rgb.load()
    blue_mask = [bytearray(width) for _ in range(height)]
    for y in range(height):
        row = blue_mask[y]
        for x in range(width):
            r, g, b = pixels[x, y]
            row[x] = int(b >= 120 and b - r >= 55 and b - g >= 35)

    blue_components = [item for item in _components(blue_mask, 1) if item[0] >= 800]
    if len(blue_components) != 1:
        return "abstain"
    blue_area, touches_border, bbox = blue_components[0]
    if touches_border or blue_area < 12000:
        return "abstain"
    min_x, min_y, max_x, max_y = bbox
    if max_x - min_x < 150 or max_y - min_y < 230:
        return "abstain"

    # In the complement, only components disconnected from the image border are holes.
    enclosed = []
    for area, border, component_bbox in _components(blue_mask, 0):
        if border or area < 90:
            continue
        x0, y0, x1, y1 = component_bbox
        if x0 <= min_x or x1 >= max_x or y0 <= min_y or y1 >= max_y:
            continue
        if x1 - x0 < 18 or y1 - y0 < 12:
            continue
        enclosed.append(area)
    if len(enclosed) not in (1, 2, 3):
        return "abstain"
    return str(len(enclosed))
