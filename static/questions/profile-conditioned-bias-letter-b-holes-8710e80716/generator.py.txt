"""Deterministic capital-B enclosure world.

The analytic arm reads only ``hole_count``.  Rendering is custom geometry rather
than a font: a stem plus overlapping lobes forms a recognizable B, after which
one to three white elliptical counters are cut from the blue silhouette.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 440
SCALE = 4
QUARANTINE_WALL_PX = 10.0
PALETTE = (
    (24, 86, 190),
    (32, 105, 214),
    (18, 74, 171),
    (38, 96, 184),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed) ^ 0xB17E)
    hole_count = int(seed) % 3 + 1
    width = rng.randint(224, 264)
    height = rng.randint(304, 344)
    x = rng.randint(72, CANVAS - width - 58)
    y = rng.randint(38, CANVAS - height - 38)
    wall_choices = (14, 19, 27, 38)
    wall_px = wall_choices[(int(seed) // 3) % len(wall_choices)]
    return {
        "hole_count": hole_count,
        "canvas_size": CANVAS,
        "x": x,
        "y": y,
        "width": width,
        "height": height,
        "blue_rgb": list(PALETTE[rng.randrange(len(PALETTE))]),
        "hole_total_height_ratio": round(rng.uniform(0.235, 0.275), 5),
        "wall_px": wall_px,
    }


def _scaled_box(values: tuple[float, float, float, float]) -> tuple[int, int, int, int]:
    return tuple(int(round(value * SCALE)) for value in values)  # type: ignore[return-value]


def render(scene: dict[str, Any]) -> Image.Image:
    size = int(scene["canvas_size"])
    x = float(scene["x"])
    y = float(scene["y"])
    width = float(scene["width"])
    height = float(scene["height"])
    blue = tuple(int(value) for value in scene["blue_rgb"])
    count = int(scene["hole_count"])
    total_ratio = float(scene["hole_total_height_ratio"])
    wall = float(scene["wall_px"])

    image = Image.new("RGB", (size * SCALE, size * SCALE), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    color = blue

    # A heavy stem and two substantially overlapping lobes keep the waist closed.
    draw.rounded_rectangle(
        _scaled_box((x, y, x + 0.34 * width, y + height)),
        radius=int(0.025 * height * SCALE),
        fill=color,
    )
    draw.ellipse(_scaled_box((x + 0.10 * width, y, x + width, y + 0.62 * height)), fill=color)
    draw.ellipse(
        _scaled_box((x + 0.10 * width, y + 0.38 * height, x + width, y + height)),
        fill=color,
    )
    # Close the lobe overlap explicitly so the middle counter has the declared
    # wall even in the thinnest sampled-clearance scenes.
    draw.rectangle(
        _scaled_box((x + 0.10 * width, y + 0.38 * height, x + width, y + 0.62 * height)),
        fill=color,
    )

    centers_by_count = {
        1: (0.50,),
        2: (0.29, 0.71),
        3: (0.19, 0.50, 0.81),
    }
    centers = centers_by_count[count]
    hole_height = total_ratio * height / count
    hole_left = x + 0.35 * width
    hole_right = x + width - wall
    for center_ratio in centers:
        cy = y + center_ratio * height
        draw.ellipse(
            _scaled_box((hole_left, cy - hole_height / 2, hole_right, cy + hole_height / 2)),
            fill=(250, 249, 246),
        )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict[str, Any]) -> float:
    return float(scene["wall_px"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) <= QUARANTINE_WALL_PX


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
