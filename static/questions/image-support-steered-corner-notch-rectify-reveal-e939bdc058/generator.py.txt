"""Corner-notch rectification reveal: four separated tiles, each turned upright.

A symbol lives on an 8x8 lattice. The lattice is cut into its four 4x4
quadrants, and every quadrant is rendered as its own small bordered tile at the
matching corner of a wide 2x2 arrangement, independently turned by a multiple of
90 degrees. Each tile's border is broken by a paper-coloured notch at the corner
that was the quadrant's top-left corner before that turn, so the declared
rectification (turn each tile until its notch is top-left, keep the tiles where
they are) is fully determined by visible fiducials. Nothing anywhere in the
picture draws the symbol: it exists only after all four separated tiles are
rectified and read together.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "corner_notch_rectify_reveal-renderer-0.1.0"

CANVAS = 640
CELLS = 4
GLYPH_CELLS = 8
CELL_PX = 8
FRAME_PX = 3
TILE_PX = CELLS * CELL_PX + 2 * FRAME_PX  # 38
NOTCH_PX = FRAME_PX + 8  # 11
NEAR_ORIGIN = 182
FAR_ORIGIN = 420
JITTER = 3

PAPER = (250, 250, 250)
BACKGROUND = (236, 236, 238)
TINTS = ((0, 0, 0), (0, 5, 11), (11, 5, 0), (4, 11, 0))

MIN_CONTRAST = 62.0
MAX_CONTRAST = 150.0
QUARANTINE_CLEARANCE = 34.0

GLYPHS = ("plus", "box", "hourglass", "diamond")
FOURFOLD_GLYPHS = frozenset({"plus", "box", "diamond"})

# Corner order used for the notch: index k means the quadrant was turned k
# quarter-turns counter-clockwise, which carries its top-left corner to this
# corner of the rendered tile.
CORNERS = ("top_left", "bottom_left", "bottom_right", "top_right")


def _template(cells: set[tuple[int, int]]) -> np.ndarray:
    grid = np.zeros((GLYPH_CELLS, GLYPH_CELLS), dtype=np.uint8)
    for row, col in cells:
        grid[row, col] = 1
    return grid


def _plus() -> set[tuple[int, int]]:
    cells = {(row, col) for row in (3, 4) for col in range(1, 7)}
    cells |= {(row, col) for col in (3, 4) for row in range(1, 7)}
    return cells


def _box() -> set[tuple[int, int]]:
    cells = {(1, col) for col in range(1, 7)} | {(6, col) for col in range(1, 7)}
    cells |= {(row, 1) for row in range(2, 6)} | {(row, 6) for row in range(2, 6)}
    return cells


def _hourglass() -> set[tuple[int, int]]:
    cells = {(1, col) for col in range(1, 7)} | {(6, col) for col in range(1, 7)}
    cells |= {(2, 2), (3, 3), (4, 4), (5, 5), (2, 5), (3, 4), (4, 3), (5, 2)}
    return cells


def _diamond() -> set[tuple[int, int]]:
    cells = {
        (0, 3), (0, 4), (1, 2), (1, 5), (2, 1), (2, 6), (3, 0), (3, 7),
        (4, 0), (4, 7), (5, 1), (5, 6), (6, 2), (6, 5), (7, 3), (7, 4),
    }
    cells |= {(3, 3), (3, 4), (4, 3), (4, 4)}
    return cells


TEMPLATES: dict[str, np.ndarray] = {
    "plus": _template(_plus()),
    "box": _template(_box()),
    "hourglass": _template(_hourglass()),
    "diamond": _template(_diamond()),
}

# Counterbalance guarantees: every glyph inks exactly 20 of the 64 lattice
# positions and exactly 5 of the 16 positions of every quadrant, so total ink,
# per-tile ink, tile order and tile size carry no information about the answer.
for _name, _grid in TEMPLATES.items():
    assert int(_grid.sum()) == 20, _name
    for _qr in (0, 1):
        for _qc in (0, 1):
            _quad = _grid[_qr * 4:_qr * 4 + 4, _qc * 4:_qc * 4 + 4]
            assert int(_quad.sum()) == 5, (_name, _qr, _qc)

MIN_GLYPH_SEPARATION = min(
    int(np.abs(TEMPLATES[a].astype(int) - TEMPLATES[b].astype(int)).sum())
    for i, a in enumerate(GLYPHS)
    for b in GLYPHS[i + 1:]
)
assert MIN_GLYPH_SEPARATION >= 12


def _luma(color: tuple[int, int, int]) -> float:
    return 0.299 * color[0] + 0.587 * color[1] + 0.114 * color[2]


def sample_scene(seed: int) -> dict:
    """Sample one scene: glyph, four tile turns, tile placement and tones."""

    # Two-word seeding decorrelates the first draws of consecutive scene seeds,
    # which keeps the symbol classes balanced across a contiguous dataset.
    rng = np.random.default_rng([int(seed), 0x9E37])
    glyph = GLYPHS[int(rng.integers(len(GLYPHS)))]
    turns = [int(rng.integers(4)) for _ in range(4)]
    if all(turn == 0 for turn in turns):
        turns[int(rng.integers(4))] = 1 + int(rng.integers(3))
    # One scene in five is drawn from the band that straddles the quarantine
    # clearance so that near-boundary and quarantined cases are always covered.
    if rng.random() < 0.2:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MIN_CONTRAST + 14.0), 3))
    else:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MAX_CONTRAST), 3))
    tint = TINTS[int(rng.integers(len(TINTS)))]
    jitter = [[int(rng.integers(-JITTER, JITTER + 1)) for _ in range(2)] for _ in range(4)]
    return {
        "canvas": CANVAS,
        "cell_px": CELL_PX,
        "frame_px": FRAME_PX,
        "notch_px": NOTCH_PX,
        "glyph": glyph,
        "turns": turns,
        "jitter": jitter,
        "contrast": contrast,
        "tint": list(tint),
    }


def _ink_color(scene: dict) -> tuple[int, int, int]:
    base = int(round(_luma(PAPER) - float(scene["contrast"])))
    tint = tuple(int(value) for value in scene["tint"])
    return (
        max(0, min(255, base + tint[0])),
        max(0, min(255, base + tint[1])),
        max(0, min(255, base + tint[2])),
    )


def _clearance(scene: dict) -> float:
    """Half the rendered paper-to-ink luma gap: the per-cell decision clearance."""

    return 0.5 * (_luma(PAPER) - _luma(_ink_color(scene)))


def _tile_origins(scene: dict) -> list[tuple[int, int]]:
    base = ((NEAR_ORIGIN, NEAR_ORIGIN), (NEAR_ORIGIN, FAR_ORIGIN),
            (FAR_ORIGIN, NEAR_ORIGIN), (FAR_ORIGIN, FAR_ORIGIN))
    return [
        (base[index][0] + scene["jitter"][index][0], base[index][1] + scene["jitter"][index][1])
        for index in range(4)
    ]


def _quadrants(glyph: str) -> list[np.ndarray]:
    grid = TEMPLATES[glyph]
    return [
        grid[0:4, 0:4].copy(), grid[0:4, 4:8].copy(),
        grid[4:8, 0:4].copy(), grid[4:8, 4:8].copy(),
    ]


def render(scene: dict) -> Image.Image:
    """Rasterise the scene with flat, hard-edged blocks and no antialiasing."""

    canvas = int(scene["canvas"])
    frame = int(scene["frame_px"])
    cell = int(scene["cell_px"])
    notch = int(scene["notch_px"])
    ink = np.array(_ink_color(scene), dtype=np.uint8)
    array = np.zeros((canvas, canvas, 3), dtype=np.uint8)
    array[:, :] = np.array(BACKGROUND, dtype=np.uint8)

    tile_px = CELLS * cell + 2 * frame
    for index, (top, left) in enumerate(_tile_origins(scene)):
        quad = _quadrants(scene["glyph"])[index]
        turned = np.rot90(quad, scene["turns"][index])
        array[top:top + tile_px, left:left + tile_px] = ink
        array[top + frame:top + tile_px - frame, left + frame:left + tile_px - frame] = np.array(
            PAPER, dtype=np.uint8
        )
        corner = CORNERS[scene["turns"][index] % 4]
        ny = top if corner.startswith("top") else top + tile_px - notch
        nx = left if corner.endswith("left") else left + tile_px - notch
        array[ny:ny + notch, nx:nx + notch] = np.array(PAPER, dtype=np.uint8)
        for row in range(CELLS):
            for col in range(CELLS):
                if not turned[row, col]:
                    continue
                y0 = top + frame + row * cell
                x0 = left + frame + col * cell
                array[y0:y0 + cell, x0:x0 + cell] = ink
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """The symbol carried jointly by the four rectified tiles."""

    return str(scene["glyph"])


def margin(scene: dict) -> float:
    """Per-cell luma clearance, the binding decision margin of the world."""

    return float(round(_clearance(scene), 6))


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose per-cell tone clearance is too small to read by eye."""

    return bool(_clearance(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered injectively."""

    return []
