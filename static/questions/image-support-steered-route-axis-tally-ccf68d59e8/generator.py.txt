"""Deterministic renderer for the route-axis tally world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 247, 242)
NAVY = (25, 45, 66)
CYAN = (20, 177, 199)
BLUE = (48, 101, 211)
ORANGE = (235, 132, 39)
PALE = (202, 207, 207)

# Same bounding box, endpoint displacement, total centerline length, segment
# count, and bend count.  Their answer differs only after lengths are grouped
# by axis and accumulated over the whole trace.
_H_BASE = [
    (0, 0), (240, 0), (240, 15), (120, 15), (120, 30),
    (180, 30), (180, 45), (210, 45), (210, 60), (240, 60),
]
_V_BASE = [
    (0, 0), (0, 60), (60, 60), (60, 0), (120, 0),
    (120, 60), (180, 60), (180, 0), (240, 0), (240, 60),
]


def _normalize(points: Iterable[tuple[int, int]]) -> list[list[int]]:
    values = list(points)
    min_x = min(x for x, _ in values)
    min_y = min(y for _, y in values)
    return [[x - min_x, y - min_y] for x, y in values]


def sample_scene(seed: int) -> dict:
    """Sample one balanced route plus answer-irrelevant pale dash geometry."""
    variant = int(seed) % 4
    rng = random.Random(int(seed) // 4)
    points = list(_H_BASE if variant < 2 else _V_BASE)
    if variant in (1, 3):
        points = [(y, -x) for x, y in points]
    points = _normalize(points)
    max_x = max(x for x, _ in points)
    max_y = max(y for _, y in points)
    if rng.random() < 0.5:
        points = [[max_x - x, y] for x, y in points]
    if rng.random() < 0.5:
        points = [[x, max_y - y] for x, y in points]
    if rng.random() < 0.5:
        points.reverse()

    box_w = max(x for x, _ in points) - min(x for x, _ in points)
    box_h = max(y for _, y in points) - min(y for _, y in points)
    left = (WIDTH - box_w) // 2 + rng.randint(-22, 22)
    top = (HEIGHT - box_h) // 2 + rng.randint(-22, 22)
    points = [[x + left, y + top] for x, y in points]

    # Dashes sit outside the route band and vary independently of the answer.
    dashes: list[list[list[int]]] = []
    for side in (-1, 1):
        for step in range(3):
            if box_w >= box_h:
                x = left + 25 + step * 78 + rng.randint(-5, 5)
                y = top + (side * 42 if side < 0 else box_h + 42)
                dashes.append([[x, y], [x + 25, y]])
            else:
                x = left + (side * 42 if side < 0 else box_w + 42)
                y = top + 25 + step * 78 + rng.randint(-5, 5)
                dashes.append([[x, y], [x, y + 25]])
    return {"points": points, "dashes": dashes}


def _axis_totals(scene: dict) -> tuple[float, float]:
    points = scene["points"]
    horizontal = 0.0
    vertical = 0.0
    for first, second in zip(points, points[1:]):
        dx = abs(float(second[0]) - float(first[0]))
        dy = abs(float(second[1]) - float(first[1]))
        if dx > 0 and dy > 0:
            raise ValueError("route contains a diagonal run")
        if dx == 0 and dy == 0:
            raise ValueError("route contains a zero-length run")
        horizontal += dx
        vertical += dy
    return horizontal, vertical


def analytic_gold(scene: dict) -> str:
    horizontal, vertical = _axis_totals(scene)
    if horizontal == vertical:
        return "tie"
    return "horizontal" if horizontal > vertical else "vertical"


def margin(scene: dict) -> float:
    horizontal, vertical = _axis_totals(scene)
    return abs(horizontal - vertical)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for first, second in scene["dashes"]:
        draw.line([tuple(first), tuple(second)], fill=PALE, width=3)
    points = [tuple(point) for point in scene["points"]]
    draw.line(points, fill=NAVY, width=13, joint="curve")
    draw.line(points, fill=CYAN, width=5, joint="curve")

    sx, sy = points[0]
    radius = 10
    draw.ellipse(
        (sx - radius, sy - radius, sx + radius, sy + radius),
        fill=BACKGROUND,
        outline=BLUE,
        width=5,
    )
    ex, ey = points[-1]
    half = 10
    draw.rectangle(
        (ex - half, ey - half, ex + half, ey + half),
        fill=BACKGROUND,
        outline=ORANGE,
        width=5,
    )
    return image
