"""Independent image-only inverse arm for route axis-length comparison."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_CYAN = np.array([20, 177, 199], dtype=np.int32)


def _route_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    distance = np.sqrt(np.sum((array - _CYAN) ** 2, axis=2))
    mask = distance < 36.0
    count = int(mask.sum())
    if not 900 <= count <= 4200:
        raise ValueError(f"cyan route ink is missing or malformed ({count} pixels)")
    return mask


def _component_size(mask: np.ndarray) -> int:
    locations = np.argwhere(mask)
    if not len(locations):
        raise ValueError("cyan route is absent")
    start = tuple(int(value) for value in locations[0])
    queue: deque[tuple[int, int]] = deque([start])
    seen = {start}
    height, width = mask.shape
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            point = (ny, nx)
            if (
                0 <= ny < height
                and 0 <= nx < width
                and mask[ny, nx]
                and point not in seen
            ):
                seen.add(point)
                queue.append(point)
    return len(seen)


def decision_from_image(image: Image.Image) -> str:
    """Segment and validate one trace, then compare its axis-conditioned ink."""
    mask = _route_mask(image)
    connected = _component_size(mask)
    if connected != int(mask.sum()):
        raise ValueError("cyan centerline is broken into multiple components")

    ys, xs = np.nonzero(mask)
    span_x = int(xs.max() - xs.min())
    span_y = int(ys.max() - ys.min())
    if max(span_x, span_y) < 180 or min(span_x, span_y) < 45:
        raise ValueError("cyan trace does not expose the required regional route")

    horizontal_pairs = int(np.sum(mask[:, :-1] & mask[:, 1:]))
    vertical_pairs = int(np.sum(mask[:-1, :] & mask[1:, :]))
    signed = horizontal_pairs - vertical_pairs
    if abs(signed) < 30:
        raise ValueError("axis-length evidence is too close to the raster boundary")
    return "horizontal" if signed > 0 else "vertical"
