"""Pixel-only oracle: receives the final PNG and nothing else.

It segments the junction discs by exact fill colour, recovers a link whenever a
gray bar fully covers the straight span between two disc centres with no third
disc lying on that span, and then strips dead-end junctions until only the
closed loop and its interior survive. The coloured junction that survives is
the decision.
"""

from __future__ import annotations

import math
from collections import deque

from PIL import Image

ORACLE_VERSION = "loop_member_node-oracle-0.1.0"
ABSTAIN = "unknown"

DISC_COLORS = {
    "grey": (60, 60, 60),
    "blue": (40, 90, 220),
    "orange": (235, 140, 20),
    "purple": (150, 60, 190),
}
BAR_COLOR = (120, 120, 120)
COLOR_TOL = 30
MIN_DISC_PIXELS = 200
MAX_LINK_SPAN = 148.0
END_SKIP = 20.0
INK_COVERAGE = 0.94
ON_SPAN_TOL = 17.0


def _classify(px, w, h):
    """One pass over the raster: a mask per disc colour plus all non-background ink."""
    masks = {label: set() for label in DISC_COLORS}
    ink = set()
    palette = list(DISC_COLORS.items()) + [("_bar", BAR_COLOR)]
    for y in range(h):
        for x in range(w):
            r, g, b = px[x, y]
            if r > 245 and g > 245 and b > 245:
                continue
            for label, (r0, g0, b0) in palette:
                if abs(r - r0) <= COLOR_TOL and abs(g - g0) <= COLOR_TOL and abs(b - b0) <= COLOR_TOL:
                    ink.add((x, y))
                    if label != "_bar":
                        masks[label].add((x, y))
                    break
    return masks, ink


def _components(mask):
    seen, out = set(), []
    for start in mask:
        if start in seen:
            continue
        seen.add(start)
        q, blob = deque([start]), []
        while q:
            x, y = q.popleft()
            blob.append((x, y))
            for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                p = (x + dx, y + dy)
                if p in mask and p not in seen:
                    seen.add(p)
                    q.append(p)
        out.append(blob)
    return out


def _centroid(blob):
    return (sum(p[0] for p in blob) / len(blob), sum(p[1] for p in blob) / len(blob))


def _point_segment_distance(p, a, b):
    vx, vy = b[0] - a[0], b[1] - a[1]
    L2 = vx * vx + vy * vy
    t = 0.0 if L2 == 0 else max(0.0, min(1.0, ((p[0] - a[0]) * vx + (p[1] - a[1]) * vy) / L2))
    return math.hypot(a[0] + t * vx - p[0], a[1] + t * vy - p[1])


def decision_from_image(image: Image.Image) -> str:
    img = image.convert("RGB")
    w, h = img.size
    px = img.load()

    masks, ink = _classify(px, w, h)

    nodes = []  # (x, y, label)
    for label in DISC_COLORS:
        for blob in _components(masks[label]):
            if len(blob) < MIN_DISC_PIXELS:
                continue
            cx, cy = _centroid(blob)
            nodes.append((cx, cy, label))
    if len(nodes) < 8:
        return ABSTAIN
    for colour in ("blue", "orange", "purple"):
        if sum(1 for n in nodes if n[2] == colour) != 1:
            return ABSTAIN

    if not ink:
        return ABSTAIN

    def inked(x, y):
        ix, iy = int(round(x)), int(round(y))
        for dx in (-2, -1, 0, 1, 2):
            for dy in (-2, -1, 0, 1, 2):
                if (ix + dx, iy + dy) in ink:
                    return True
        return False

    edges = []
    for i in range(len(nodes)):
        for j in range(i + 1, len(nodes)):
            ax, ay, _ = nodes[i]
            bx, by, _ = nodes[j]
            span = math.hypot(bx - ax, by - ay)
            if span > MAX_LINK_SPAN or span <= 2 * END_SKIP:
                continue
            blocked = False
            for k, n in enumerate(nodes):
                if k in (i, j):
                    continue
                if _point_segment_distance((n[0], n[1]), (ax, ay), (bx, by)) < ON_SPAN_TOL:
                    blocked = True
                    break
            if blocked:
                continue
            lo, hi = END_SKIP / span, 1.0 - END_SKIP / span
            samples = 48
            good = 0
            for s in range(samples):
                t = lo + (hi - lo) * s / (samples - 1)
                if inked(ax + (bx - ax) * t, ay + (by - ay) * t):
                    good += 1
            if good >= INK_COVERAGE * samples:
                edges.append((i, j))
    if not edges:
        return ABSTAIN

    adj = {i: set() for i in range(len(nodes))}
    for a, b in edges:
        adj[a].add(b)
        adj[b].add(a)
    alive = set(range(len(nodes)))
    changed = True
    while changed:
        changed = False
        for v in sorted(alive):
            if len(adj[v] & alive) <= 1:
                alive.discard(v)
                changed = True
    if not alive:
        return ABSTAIN

    survivors = sorted({nodes[i][2] for i in alive} & {"blue", "orange", "purple"})
    if len(survivors) != 1:
        return ABSTAIN
    return survivors[0]
