"""Renderer and analytic arm for a half-turn directed-subtraction mosaic."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw, ImageFont

GRID_SIZE = 11
LABELS = ("fork", "hook", "stair", "rake")
TEMPLATES = {
    "fork": {(0, 0), (0, 4), (1, 1), (1, 3), (2, 2), (3, 2), (4, 1), (4, 2), (4, 3)},
    "hook": {(0, 0), (1, 0), (2, 0), (3, 0), (4, 0), (4, 1), (4, 2), (4, 3), (4, 4)},
    "stair": {(0, 0), (1, 0), (1, 1), (2, 1), (2, 2), (3, 2), (3, 3), (4, 3), (4, 4)},
    "rake": {(0, 0), (0, 1), (0, 2), (0, 3), (0, 4), (1, 0), (1, 2), (1, 4), (2, 2)},
}
POSITIONS = ((0, 0), (0, 6), (6, 0), (6, 6))
DARK_COLORS = ("#1E2D4F", "#51295C", "#185342", "#623426")
LIGHT_COLORS = ("#FAFAF4", "#F4F7F8", "#FFF9F0", "#F7F3FA")
FRAME_COLOR = "#0796B5"
PIVOT_COLOR = "#E1A51A"
BACKGROUND = "#EEF1F3"


def _rotate(cell: tuple[int, int]) -> tuple[int, int]:
    return GRID_SIZE - 1 - cell[0], GRID_SIZE - 1 - cell[1]


def _directed_result(cells: list[list[int]] | list[tuple[int, int]]) -> set[tuple[int, int]]:
    active = {tuple(cell) for cell in cells}
    return {cell for cell in active if _rotate(cell) not in active}


def _scores(result: set[tuple[int, int]]) -> dict[str, int]:
    scores: dict[str, int] = {}
    for label, template in TEMPLATES.items():
        best = 10_000
        for row0 in range(GRID_SIZE - 4):
            for col0 in range(GRID_SIZE - 4):
                placed = {(row0 + row, col0 + col) for row, col in template}
                best = min(best, len(result.symmetric_difference(placed)))
        scores[label] = best
    return scores


def _classify_result(result: set[tuple[int, int]]) -> tuple[str, float, int]:
    scores = _scores(result)
    ranked = sorted((cost, label) for label, cost in scores.items())
    best_cost, label = ranked[0]
    decision_margin = float(ranked[1][0] - best_cost)
    return label, decision_margin, best_cost


def _make_result(
    label: str,
    position: tuple[int, int],
    difficulty: int,
    rng: random.Random,
) -> set[tuple[int, int]]:
    row0, col0 = position
    pristine = {(row0 + row, col0 + col) for row, col in TEMPLATES[label]}
    if difficulty == 0:
        return pristine

    # Replace one or two glyph cells while preserving its nine-cell ink count.
    # Among valid trials choose the smallest still-safe top-two margin, which
    # supplies deterministic near-boundary examples without ambiguous labels.
    best_result = pristine
    best_margin = 1_000.0
    universe = [
        (row, col)
        for row in range(GRID_SIZE)
        for col in range(GRID_SIZE)
        if (row, col) != (GRID_SIZE // 2, GRID_SIZE // 2)
    ]
    for _ in range(800):
        removed = set(rng.sample(sorted(pristine), difficulty))
        available = [
            cell
            for cell in universe
            if cell not in pristine
            and _rotate(cell) not in pristine
            and cell not in removed
        ]
        added = set(rng.sample(available, difficulty))
        trial = (pristine - removed) | added
        if any(_rotate(cell) in trial for cell in trial):
            continue
        winner, decision_margin, best_cost = _classify_result(trial)
        if winner == label and decision_margin >= 4.0 and best_cost <= 4:
            if decision_margin < best_margin:
                best_result, best_margin = trial, decision_margin
                if best_margin == 4.0:
                    break
    return best_result


def sample_scene(seed: int) -> dict:
    """Sample a balanced, fully rendered scene from one integer seed."""
    rng = random.Random((int(seed) ^ 0x5A17C9D3) & 0xFFFFFFFF)
    label = LABELS[int(seed) % len(LABELS)]
    group = int(seed) // len(LABELS)
    position = POSITIONS[group % len(POSITIONS)]
    difficulty = group % 3
    result = _make_result(label, position, difficulty, rng)

    forbidden = result | {_rotate(cell) for cell in result}
    center = (GRID_SIZE // 2, GRID_SIZE // 2)
    representatives = []
    for row in range(GRID_SIZE):
        for col in range(GRID_SIZE):
            cell = (row, col)
            mate = _rotate(cell)
            if cell == center or cell in forbidden or mate in forbidden or cell > mate:
                continue
            representatives.append(cell)
    noise_reps = rng.sample(representatives, 16)
    active = set(result)
    for cell in noise_reps:
        active.add(cell)
        active.add(_rotate(cell))
    assert len(active) == 41
    assert _directed_result([list(cell) for cell in active]) == result

    paint_order = [list(cell) for cell in active]
    rng.shuffle(paint_order)
    cell_size = (28, 30, 32)[group % 3]
    board_size = GRID_SIZE * cell_size
    board_x = (640 - board_size) // 2 + rng.randint(-16, 16)
    board_y = 154 + rng.randint(-12, 12)
    palette_index = (3 * group + group // 4) % len(DARK_COLORS)
    return {
        "width": 640,
        "height": 640,
        "board_x": board_x,
        "board_y": board_y,
        "cell_size": cell_size,
        "grid_size": GRID_SIZE,
        "paint_order": paint_order,
        "dark_color": DARK_COLORS[palette_index],
        "light_color": LIGHT_COLORS[(group + 1) % len(LIGHT_COLORS)],
        "frame_color": FRAME_COLOR,
        "pivot_color": PIVOT_COLOR,
    }


def render(scene: dict) -> Image.Image:
    """Render only public scene fields; no answer field exists in the scene."""
    width = int(scene["width"])
    height = int(scene["height"])
    board_x = int(scene["board_x"])
    board_y = int(scene["board_y"])
    cell_size = int(scene["cell_size"])
    grid_size = int(scene["grid_size"])
    dark_color = str(scene["dark_color"])
    light_color = str(scene["light_color"])
    frame_color = str(scene["frame_color"])
    pivot_color = str(scene["pivot_color"])
    paint_order = scene["paint_order"]

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    board_size = grid_size * cell_size

    # Compact visible transform legend above the actual mosaic.
    title = "ROTATE COPY 180 degrees   ORIGINAL minus ROTATED"
    title_box = draw.textbbox((0, 0), title, font=font)
    title_w = title_box[2] - title_box[0]
    draw.rounded_rectangle(
        [width // 2 - title_w // 2 - 14, 57, width // 2 + title_w // 2 + 14, 91],
        radius=8,
        fill="#FFFFFF",
        outline=frame_color,
        width=2,
    )
    draw.text((width // 2 - title_w // 2, 69), title, fill="#25323A", font=font)

    # A gold half-turn arrow is separate from the gold pivot and purely public.
    cx = board_x + board_size // 2
    draw.arc([cx - 36, 102, cx + 36, 142], 195, 525, fill=pivot_color, width=4)
    draw.polygon([(cx + 35, 125), (cx + 27, 116), (cx + 25, 130)], fill=pivot_color)

    draw.rectangle(
        [board_x, board_y, board_x + board_size - 1, board_y + board_size - 1],
        fill=light_color,
    )
    for row, col in paint_order:
        row, col = int(row), int(col)
        x0 = board_x + col * cell_size + 3
        y0 = board_y + row * cell_size + 3
        x1 = board_x + (col + 1) * cell_size - 4
        y1 = board_y + (row + 1) * cell_size - 4
        draw.rounded_rectangle([x0, y0, x1, y1], radius=3, fill=dark_color)

    grid_color = "#B8C0C5"
    for index in range(grid_size + 1):
        x = board_x + index * cell_size
        y = board_y + index * cell_size
        draw.line([x, board_y, x, board_y + board_size], fill=grid_color, width=1)
        draw.line([board_x, y, board_x + board_size, y], fill=grid_color, width=1)

    frame_pad = 8
    draw.rectangle(
        [
            board_x - frame_pad,
            board_y - frame_pad,
            board_x + board_size + frame_pad - 1,
            board_y + board_size + frame_pad - 1,
        ],
        outline=frame_color,
        width=5,
    )

    pivot_x = board_x + (grid_size * cell_size) // 2
    pivot_y = board_y + (grid_size * cell_size) // 2
    radius = max(6, cell_size // 5)
    draw.ellipse(
        [pivot_x - radius, pivot_y - radius, pivot_x + radius, pivot_y + radius],
        fill=pivot_color,
        outline="#8B5E00",
        width=2,
    )
    draw.ellipse([pivot_x - 2, pivot_y - 2, pivot_x + 2, pivot_y + 2], fill="#FFF7D1")

    footer = "KEEP: original DARK and rotated LIGHT     orientation stays fixed"
    footer_box = draw.textbbox((0, 0), footer, font=font)
    footer_w = footer_box[2] - footer_box[0]
    draw.text((width // 2 - footer_w // 2, board_y + board_size + 36), footer, fill="#3D4A52", font=font)
    return image


def analytic_gold(scene: dict) -> str:
    result = _directed_result(scene["paint_order"])
    label, _, best_cost = _classify_result(result)
    if best_cost > 4:
        raise ValueError("analytic glyph is outside the supported template tolerance")
    return label


def margin(scene: dict) -> float:
    result = _directed_result(scene["paint_order"])
    _, decision_margin, _ = _classify_result(result)
    return decision_margin


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["paint_order"] = list(reversed(twin["paint_order"]))
    return [("reverse_nonoverlapping_paint_order", twin)]
