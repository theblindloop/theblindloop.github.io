"""Independent image-only inverse for residual bundle orders."""
from __future__ import annotations
import numpy as np
from PIL import Image

TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91)); RUN_COLOR=(72,79,89)
RECIPE_X=(22,336,650); ORDER_X=(18,254,490,726); ROW_Y=(540,610,680)

def _components(mask):
    h,w=mask.shape; remaining=set(int(v) for v in np.flatnonzero(mask)); sizes=[]
    while remaining:
        start=remaining.pop(); stack=[start]; size=0
        while stack:
            v=stack.pop(); y,x=divmod(v,w); size+=1
            for n in (v-1,v+1,v-w,v+w):
                if n in remaining:
                    ny,nx=divmod(n,w)
                    if abs(nx-x)+abs(ny-y)==1: remaining.remove(n); stack.append(n)
        if size>=70: sizes.append(size)
    if any(not 120<=size<=190 for size in sizes): return None
    return len(sizes)
def _count_color(arr,box,color):
    x0,y0,x1,y1=box; crop=arr[y0:y1,x0:x1]
    delta=np.max(np.abs(crop.astype(np.int16)-np.asarray(color,dtype=np.int16)),axis=2)
    return _components(delta<=2)
def _counts(arr,box):
    values=tuple(_count_color(arr,box,c) for c in TOKEN_COLORS)
    return None if any(v is None for v in values) else values
def _multiple(residual,kit):
    if any(v<=0 for v in kit) or any(v<0 for v in residual): return False
    q=[]
    for have,need in zip(residual,kit):
        if have%need: return False
        q.append(have//need)
    return q[0]>=1 and len(set(q))==1

def decision_from_image(image:Image.Image)->str:
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(720,960,3): return "abstain"
    recipes=[]
    for x0 in RECIPE_X:
        costs=_counts(arr,(x0+20,78,x0+268,168))
        if costs is None or sum(v>0 for v in costs)!=2 or sum(costs)!=3: return "abstain"
        recipes.append(costs)
    stock=_counts(arr,(125,218,835,308)); kit=_counts(arr,(235,354,725,430))
    if stock is None or kit is None or any(not 1<=x<=3 for x in kit) or len(set(kit))<3: return "abstain"
    orders=[]
    for x0 in ORDER_X:
        plan=[]
        for cy in ROW_Y:
            value=_count_color(arr,(x0+50,cy-28,x0+206,cy+28),RUN_COLOR)
            if value is None or not 0<=value<=4: return "abstain"
            plan.append(value)
        orders.append(tuple(plan))
    winners=[]
    for i,plan in enumerate(orders):
        used=tuple(sum(plan[r]*recipes[r][c] for r in range(3)) for c in range(4))
        residual=tuple(stock[c]-used[c] for c in range(4))
        if _multiple(residual,kit): winners.append(i)
    return f"order-{chr(65+winners[0])}" if len(winners)==1 else "abstain"
