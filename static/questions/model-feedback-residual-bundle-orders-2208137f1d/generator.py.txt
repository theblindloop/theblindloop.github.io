"""Deterministic renderer for affine residual-to-kit proportionality."""
from __future__ import annotations
import itertools, random
from PIL import Image, ImageDraw, ImageFont

SIZE=(960,720); BACKGROUND=(246,244,238); PANEL=(255,254,250); INK=(38,42,48); MUTED=(214,211,203)
ACCENT=(178,76,54); KIT_ACCENT=(38,126,106)
TOKEN_COLORS=((24,137,181),(146,75,163),(228,130,30),(52,153,91)); RUN_COLOR=(72,79,89)
RECIPE_COLORS=((208,73,73),(29,151,151),(128,92,184)); DECISIONS=("order-A","order-B","order-C","order-D")
RECIPE_X=(22,336,650); ORDER_X=(18,254,490,726); ROW_Y=(540,610,680)
FONT_PATH="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _font(size): return ImageFont.truetype(FONT_PATH,size)
def _usage(plan,recipes): return tuple(sum(int(plan[r])*int(recipes[r][c]) for r in range(3)) for c in range(4))
def _residual(plan,stock,recipes):
    used=_usage(plan,recipes); return tuple(int(stock[c])-used[c] for c in range(4))
def _multiplier(residual,kit):
    if any(int(v)<=0 for v in kit) or any(int(v)<0 for v in residual): return None
    q=[]
    for have,need in zip(residual,kit):
        if int(have)%int(need): return None
        q.append(int(have)//int(need))
    return q[0] if q and q[0]>=1 and len(set(q))==1 else None

def sample_scene(seed:int)->dict:
    rng=random.Random(int(seed))
    for _ in range(1200):
        pairs=[(0,1),(1,2),(2,3)]; rng.shuffle(pairs); recipes=[]
        for a,b in pairs:
            costs=[0,0,0,0]
            costs[a],costs[b]=(1,2) if rng.randrange(2) else (2,1)
            recipes.append(tuple(costs))
        recipes=tuple(recipes); kit=tuple(rng.randint(1,3) for _ in range(4)); total=rng.randint(5,8)
        if len(set(kit))<3: continue
        possible=[p for p in itertools.product(range(5),repeat=3) if sum(p)==total]; rng.shuffle(possible)
        winning=possible[0]; copies=rng.randint(2,3); used=_usage(winning,recipes)
        stock=tuple(used[c]+copies*kit[c] for c in range(4))
        if max(stock)>15 or min(stock)<3 or sum(stock)>48: continue
        target=tuple(copies*v for v in kit); distractors=[]
        for plan in possible[1:]:
            residual=_residual(plan,stock,recipes)
            if min(residual)<1 or _multiplier(residual,kit) is not None: continue
            if sum(abs(a-b) for a,b in zip(residual,target))<2: continue
            distractors.append(plan)
        if len(distractors)<3: continue
        rng.shuffle(distractors); shown=[winning]+distractors[:3]; wi=int(seed)%4
        shown[0],shown[wi]=shown[wi],shown[0]
        scene={"stock_counts":list(stock),"recipes":[list(x) for x in recipes],"kit_counts":list(kit),
               "order_runs":[list(x) for x in shown],"layout_variant":rng.randrange(64)}
        if not is_quarantined(scene): return scene
    raise RuntimeError("unable to sample a valid residual-bundle scene")

def _validated(scene):
    stock=tuple(int(x) for x in scene["stock_counts"]); recipes=tuple(tuple(int(x) for x in row) for row in scene["recipes"])
    kit=tuple(int(x) for x in scene["kit_counts"]); orders=tuple(tuple(int(x) for x in p) for p in scene["order_runs"]); variant=int(scene["layout_variant"])
    if len(stock)!=4 or len(recipes)!=3 or len(kit)!=4 or len(orders)!=4: raise ValueError("wrong collection size")
    if any(len(row)!=4 or sum(v>0 for v in row)!=2 or sum(row)!=3 for row in recipes): raise ValueError("bad recipe")
    if any(not 0<=x<=15 for x in stock) or any(not 0<=x<=2 for row in recipes for x in row): raise ValueError("bad count")
    if any(not 1<=x<=3 for x in kit) or len(set(kit))<3: raise ValueError("bad kit")
    if any(len(p)!=3 or any(not 0<=x<=4 for x in p) for p in orders): raise ValueError("bad order")
    if not 0<=variant<64: raise ValueError("bad variant")
    return stock,recipes,kit,orders,variant

def analytic_gold(scene):
    try:
        stock,recipes,kit,orders,_=_validated(scene)
        winners=[i for i,p in enumerate(orders) if _multiplier(_residual(p,stock,recipes),kit) is not None]
        return DECISIONS[winners[0]] if len(winners)==1 else "abstain"
    except (KeyError,TypeError,ValueError,IndexError): return "abstain"
def margin(scene):
    stock,recipes,kit,orders,_=_validated(scene); winner=analytic_gold(scene)
    if winner=="abstain": return 0.0
    wi=DECISIONS.index(winner); target=_residual(orders[wi],stock,recipes)
    return float(min(sum(abs(a-b) for a,b in zip(_residual(p,stock,recipes),target)) for i,p in enumerate(orders) if i!=wi))
def is_quarantined(scene):
    try:
        stock,recipes,kit,orders,_=_validated(scene); residuals=[_residual(p,stock,recipes) for p in orders]
        return len(set(orders))!=4 or any(min(r)<1 for r in residuals) or len({sum(r) for r in residuals})!=1 or sum(_multiplier(r,kit) is not None for r in residuals)!=1 or margin(scene)<2
    except (KeyError,TypeError,ValueError,IndexError): return True
def latent_symmetries(scene): return []

def _disks(draw,counts,box,columns,reverse,colors=TOKEN_COLORS,radius=7):
    x0,y0,x1,y1=box; kinds=[k for k,n in enumerate(counts) for _ in range(int(n))]
    if reverse: kinds.reverse()
    gap=20; rows=max(1,(len(kinds)+columns-1)//columns); shown=min(columns,max(1,len(kinds)))
    sx=(x0+x1)//2-(shown-1)*gap//2; sy=(y0+y1)//2-(rows-1)*gap//2
    for i,k in enumerate(kinds):
        cx,cy=sx+(i%columns)*gap,sy+(i//columns)*gap
        draw.ellipse((cx-radius,cy-radius,cx+radius,cy+radius),fill=colors[k],outline=INK,width=1)
def _same_disks(draw,count,box,columns,color,reverse=False): _disks(draw,[count],box,columns,reverse,colors=(color,),radius=7)

def render(scene):
    stock,recipes,kit,orders,variant=_validated(scene); image=Image.new("RGB",SIZE,BACKGROUND); draw=ImageDraw.Draw(image)
    title,label,small=_font(23),_font(17),_font(14)
    draw.text((22,10),"RECIPES: TOKENS USED EACH RUN",font=title,fill=INK)
    for i,(costs,x0) in enumerate(zip(recipes,RECIPE_X)):
        draw.rounded_rectangle((x0,44,x0+288,174),radius=11,fill=PANEL,outline=RECIPE_COLORS[i],width=4)
        draw.text((x0+12,54),f"RECIPE {i+1}",font=small,fill=INK)
        _disks(draw,costs,(x0+24,82,x0+264,164),8,bool((variant>>i)&1))
    draw.text((22,190),"START STOCK",font=label,fill=INK)
    draw.rounded_rectangle((22,216,938,310),radius=11,fill=PANEL,outline=ACCENT,width=4)
    _disks(draw,stock,(130,222,830,304),14,bool(variant&8))
    draw.text((22,326),"ONE TARGET KIT",font=label,fill=KIT_ACCENT)
    draw.text((585,328),"PACK ALL LEFTOVERS • NO EXTRAS",font=small,fill=INK)
    draw.rounded_rectangle((22,352,938,432),radius=11,fill=PANEL,outline=KIT_ACCENT,width=4)
    _disks(draw,kit,(240,358,720,426),10,bool(variant&16))
    draw.text((22,446),"ORDERS: ONE MARK = ONE RECIPE RUN",font=label,fill=INK)
    for oi,(plan,x0) in enumerate(zip(orders,ORDER_X)):
        draw.rounded_rectangle((x0,474,x0+216,710),radius=11,fill=PANEL,outline=INK,width=3)
        draw.text((x0+66,484),f"order {chr(65+oi)}",font=small,fill=INK); draw.line((x0+10,516,x0+206,516),fill=MUTED,width=2)
        for ri,(runs,cy) in enumerate(zip(plan,ROW_Y)):
            draw.ellipse((x0+13,cy-14,x0+41,cy+14),fill=RECIPE_COLORS[ri],outline=INK,width=2)
            tw=draw.textbbox((0,0),str(ri+1),font=small)[2]; draw.text((x0+27-tw//2,cy-10),str(ri+1),font=small,fill=PANEL)
            _same_disks(draw,runs,(x0+54,cy-25,x0+202,cy+25),5,RUN_COLOR,bool((variant+oi+ri)&1))
            if ri<2: draw.line((x0+10,cy+34,x0+206,cy+34),fill=MUTED,width=1)
    return image
