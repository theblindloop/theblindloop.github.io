"""Renderer for the tally-sort reveal world.

A bordered seven-by-seven panel carries the seven rows of an abstract symbol in a
scrambled vertical order.  To the left of the panel, separated from it and from
each other by background, lies a gutter of seven free-floating tally bars, one
beside each panel row.  The declared operation reorders the panel's rows by the
length of the bar beside each row, shortest at the top, carrying each row's cells
with it; the reordered cells form the symbol.  The panel alone fixes the multiset
of rows but not their order, and the gutter alone fixes only a permutation, so
both disconnected components are decisive.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "tally_sort_reveal-renderer-0.1.0"

CANVAS = (1000, 700)
GRID = 7
CELL_PX = 30
BORDER_PX = 4
PAD_PX = 6
PANEL_PX = 2 * BORDER_PX + 2 * PAD_PX + GRID * CELL_PX  # 230
BAR_H = 10
GUTTER_GAP = 22
BASE_BAR = 40
QUARANTINE_CLEARANCE = 24.0

# Every symbol is a seven-by-seven bitmap with exactly 13 marked cells, so total
# ink, cell count and panel size are constant across answers.
SYMBOLS: dict[str, tuple[str, ...]] = {
    "cross": (
        "...#...",
        "...#...",
        "...#...",
        "#######",
        "...#...",
        "...#...",
        "...#...",
    ),
    "tee": (
        "#######",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
        "...#...",
    ),
    "zigzag": (
        ".#####.",
        "....#..",
        "...#...",
        "..#....",
        ".#####.",
        ".......",
        ".......",
    ),
    "corners": (
        "##...##",
        "#.....#",
        ".......",
        "...#...",
        ".......",
        "#.....#",
        "##...##",
    ),
    "stripes": (
        ".#####.",
        ".......",
        ".......",
        ".####..",
        ".......",
        ".......",
        ".####..",
    ),
    "columns": (
        ".......",
        ".#.#.#.",
        ".#.#.#.",
        ".#.#.#.",
        ".#.#.#.",
        "...#...",
        ".......",
    ),
}

SYMBOL_NAMES = tuple(sorted(SYMBOLS))
SINGLE_PIECE = ("cross", "tee", "zigzag")
INK_CELLS = 13


def _matrix(name: str) -> tuple[tuple[int, ...], ...]:
    return tuple(tuple(1 if ch == "#" else 0 for ch in row) for row in SYMBOLS[name])


def _hamming(a, b) -> int:
    return sum(1 for ra, rb in zip(a, b) for ca, cb in zip(ra, rb) if ca != cb)


def _runner_up_gap(name: str) -> int:
    mine = _matrix(name)
    return min(_hamming(mine, _matrix(other)) for other in SYMBOL_NAMES if other != name)


def _nearest_symbol_distance(rows) -> int:
    return min(_hamming(tuple(rows), _matrix(other)) for other in SYMBOL_NAMES)


def sample_scene(seed: int) -> dict:
    """Sample one scene deterministically from an integer seed."""

    rng = random.Random(seed * 7919 + 13)
    symbol = SYMBOL_NAMES[seed % len(SYMBOL_NAMES)]
    matrix = _matrix(symbol)

    # Row order: display row d shows glyph row row_order[d].  Reject orders that
    # leave the unsorted panel close to any symbol, so the answer is unreadable
    # without the gutter.
    for _ in range(4000):
        order = list(range(GRID))
        rng.shuffle(order)
        displayed = [matrix[g] for g in order]
        if sum(1 for d, g in enumerate(order) if d == g) > 2:
            continue
        if _nearest_symbol_distance(displayed) < 6:
            continue
        break
    else:  # pragma: no cover - the loop always succeeds for the sampled sizes
        raise RuntimeError("no admissible row order")

    gaps = [rng.randint(12, 22) for _ in range(GRID - 1)]
    sorted_lengths = [BASE_BAR]
    for gap in gaps:
        sorted_lengths.append(sorted_lengths[-1] + gap)
    bar_lengths = [sorted_lengths[order[d]] for d in range(GRID)]

    clearance = rng.randint(18, 62)
    ink_value = rng.randint(26, 58)
    pale_value = ink_value + 2 * clearance
    bg_value = min(252, max(pale_value + 45, rng.randint(232, 252)))
    tint = rng.choice([(0, 0, 0), (4, 0, -4), (-4, 2, 4), (0, -4, 4), (3, 3, -3)])

    panel_left = rng.randint(BASE_BAR + 6 * 22 + GUTTER_GAP + 30, CANVAS[0] - PANEL_PX - 30)
    panel_top = rng.randint(40, CANVAS[1] - PANEL_PX - 40)

    cluster = (
        panel_left - GUTTER_GAP - max(sorted_lengths) - 14,
        panel_top - 14,
        panel_left + PANEL_PX + 14,
        panel_top + PANEL_PX + 14,
    )
    distractors = []
    for _ in range(rng.randint(4, 7)):
        for _attempt in range(200):
            kind = rng.choice(["square", "rule", "diamond"])
            if kind == "square":
                w = h = rng.randint(8, 14)
            elif kind == "rule":
                w, h = 4, rng.randint(20, 40)
            else:
                w = h = rng.randint(9, 13)
            x = rng.randint(8, CANVAS[0] - w - 8)
            y = rng.randint(8, CANVAS[1] - h - 8)
            if x + w >= cluster[0] and x <= cluster[2] and y + h >= cluster[1] and y <= cluster[3]:
                continue
            distractors.append([kind, x, y, w, h])
            break

    return {
        "canvas": list(CANVAS),
        "cell_px": CELL_PX,
        "grid": GRID,
        "symbol": symbol,
        "row_order": order,
        "bar_lengths": bar_lengths,
        "bar_h": BAR_H,
        "gutter_gap": GUTTER_GAP,
        "panel_left": panel_left,
        "panel_top": panel_top,
        "ink_value": ink_value,
        "pale_value": pale_value,
        "background_value": bg_value,
        "tint": list(tint),
        "clearance": clearance,
        "distractors": distractors,
    }


def _tone(value: int, tint) -> tuple[int, int, int]:
    return tuple(max(0, min(255, value + t)) for t in tint)


def render(scene: dict) -> Image.Image:
    """Rasterise a scene with flat, hard-edged fills (no antialiasing)."""

    width, height = scene["canvas"]
    tint = tuple(scene["tint"])
    ink = _tone(scene["ink_value"], tint)
    pale = _tone(scene["pale_value"], tint)
    bg = _tone(scene["background_value"], tint)
    image = Image.new("RGB", (width, height), bg)
    draw = ImageDraw.Draw(image)

    left, top = scene["panel_left"], scene["panel_top"]
    right, bottom = left + PANEL_PX - 1, top + PANEL_PX - 1
    # Panel: an ink frame around a pale field; cells never touch the frame.
    draw.rectangle([left, top, right, bottom], fill=ink)
    draw.rectangle(
        [left + BORDER_PX, top + BORDER_PX, right - BORDER_PX, bottom - BORDER_PX], fill=pale
    )

    matrix = _matrix(scene["symbol"])
    grid_left = left + BORDER_PX + PAD_PX
    grid_top = top + BORDER_PX + PAD_PX
    cell = scene["cell_px"]
    for display_row, glyph_row in enumerate(scene["row_order"]):
        for col in range(scene["grid"]):
            if not matrix[glyph_row][col]:
                continue
            x0 = grid_left + col * cell
            y0 = grid_top + display_row * cell
            draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=ink)

    bar_h = scene["bar_h"]
    bar_right = left - scene["gutter_gap"]
    for display_row, length in enumerate(scene["bar_lengths"]):
        y_center = grid_top + display_row * cell + cell // 2
        y0 = y_center - bar_h // 2
        draw.rectangle([bar_right - length + 1, y0, bar_right, y0 + bar_h - 1], fill=ink)

    for kind, x, y, w, h in scene["distractors"]:
        if kind == "diamond":
            cx, cy = x + w // 2, y + h // 2
            draw.polygon([(cx, y), (x + w - 1, cy), (cx, y + h - 1), (x, cy)], fill=ink)
        else:
            draw.rectangle([x, y, x + w - 1, y + h - 1], fill=ink)
    return image


def analytic_gold(scene: dict) -> str:
    """The symbol revealed by the declared reordering."""

    return scene["symbol"]


def margin(scene: dict) -> float:
    """Composite runner-up margin in luma-equivalent units."""

    lengths = sorted(scene["bar_lengths"])
    length_gap = min(b - a for a, b in zip(lengths, lengths[1:]))
    return float(
        min(
            float(scene["clearance"]),
            2.0 * length_gap,
            4.0 * _runner_up_gap(scene["symbol"]),
        )
    )


def is_quarantined(scene: dict) -> bool:
    """Flag finite-resolution boundary cases."""

    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every sampled field is rendered."""

    return []
