"""Deterministic latent sampler and raster renderer for spiral-path reflow."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

N = 12
SYMBOLS = ("arrow", "cup", "bolt", "key")

_ART = {
    "arrow": (
        "............", ".....##.....", "....####....", "...######...",
        ".....##.....", ".....##.....", ".....##.....", ".....##.....",
        ".....##.....", ".....##.....", "............", "............",
    ),
    "cup": (
        "............", "............", "..#......#..", "..#......#..",
        "..#......#..", "..#......#..", "..#......#..", "..#......#..",
        "..########..", "....####....", "............", "............",
    ),
    "bolt": (
        "............", ".......###..", "......###...", ".....###....",
        "....###.....", "......###...", ".....###....", "....###.....",
        "...###......", "............", "............", "............",
    ),
    "key": (
        "............", "............", "..####......", "..#..#......",
        "..#..##.....", "..####......", ".....##.....", ".....###....",
        ".....####...", ".....##.....", "............", "............",
    ),
}


def _bits(name: str) -> list[int]:
    return [int(ch == "#") for row in _ART[name] for ch in row]


def _base_spiral() -> list[tuple[int, int]]:
    out: list[tuple[int, int]] = []
    top = left = 0
    bottom = right = N - 1
    while top <= bottom and left <= right:
        out.extend((top, c) for c in range(left, right + 1))
        top += 1
        out.extend((r, right) for r in range(top, bottom + 1))
        right -= 1
        if top <= bottom:
            out.extend((bottom, c) for c in range(right, left - 1, -1))
            bottom -= 1
        if left <= right:
            out.extend((r, left) for r in range(bottom, top - 1, -1))
            left += 1
    return out


def _rotate(rc: tuple[int, int], turns: int) -> tuple[int, int]:
    r, c = rc
    for _ in range(turns):
        r, c = c, N - 1 - r
    return r, c


def _path(corner: str) -> list[tuple[int, int]]:
    turns = {"tl": 0, "tr": 1, "br": 2, "bl": 3}[corner]
    return [_rotate(rc, turns) for rc in _base_spiral()]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17)
    cell = rng.choice((20, 21, 22, 23))
    span = N * cell
    ox = rng.randint(62, 420 - span - 40)
    oy = rng.randint(62, 420 - span - 40)
    darks = ((28, 35, 48), (48, 28, 42), (24, 48, 40), (42, 38, 24))
    lights = ((246, 246, 241), (241, 247, 248), (248, 243, 239), (244, 242, 249))
    frames = ((20, 169, 184), (12, 154, 174), (32, 180, 190))
    return {
        "symbol": SYMBOLS[int(seed) % len(SYMBOLS)],
        "start_corner": ("tl", "tr", "br", "bl")[rng.randrange(4)],
        "cell_size": cell,
        "origin_x": ox,
        "origin_y": oy,
        "dark_rgb": list(rng.choice(darks)),
        "light_rgb": list(rng.choice(lights)),
        "frame_rgb": list(rng.choice(frames)),
        "start_rgb": [210, 43, 151],
        "end_rgb": [238, 181, 24],
    }


def render(scene: dict) -> Image.Image:
    symbol = scene["symbol"]
    corner = scene["start_corner"]
    cell = int(scene["cell_size"])
    ox, oy = int(scene["origin_x"]), int(scene["origin_y"])
    dark, light = tuple(scene["dark_rgb"]), tuple(scene["light_rgb"])
    cyan = tuple(scene["frame_rgb"])
    magenta, yellow = tuple(scene["start_rgb"]), tuple(scene["end_rgb"])
    path = _path(corner)
    source = [[0] * N for _ in range(N)]
    for value, (r, c) in zip(_bits(symbol), path):
        source[r][c] = value

    image = Image.new("RGB", (420, 420), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    centers = [(ox + (c + 0.5) * cell, oy + (r + 0.5) * cell) for r, c in path]
    draw.line(centers, fill=cyan, width=max(3, cell // 5), joint="curve")
    draw.rectangle((ox, oy, ox + N * cell, oy + N * cell), outline=cyan, width=4)
    inset = max(4, cell // 5)
    for r in range(N):
        for c in range(N):
            x0, y0 = ox + c * cell, oy + r * cell
            fill = dark if source[r][c] else light
            draw.rounded_rectangle(
                (x0 + inset, y0 + inset, x0 + cell - inset, y0 + cell - inset),
                radius=2, fill=fill,
            )

    sr, sc = path[0]
    sx, sy = ox + (sc + 0.5) * cell, oy + (sr + 0.5) * cell
    d = max(10, cell // 2)
    if corner == "tl":
        tri = [(sx, oy - 5), (sx - d, oy - 5 - d), (sx + d, oy - 5 - d)]
    elif corner == "tr":
        tri = [(ox + N * cell + 5, sy), (ox + N * cell + 5 + d, sy - d), (ox + N * cell + 5 + d, sy + d)]
    elif corner == "br":
        tri = [(sx, oy + N * cell + 5), (sx - d, oy + N * cell + 5 + d), (sx + d, oy + N * cell + 5 + d)]
    else:
        tri = [(ox - 5, sy), (ox - 5 - d, sy - d), (ox - 5 - d, sy + d)]
    draw.polygon(tri, fill=magenta)

    er, ec = path[-1]
    ex, ey = ox + (ec + 0.5) * cell, oy + (er + 0.5) * cell
    rad = max(7, cell // 3)
    diamond = [(ex, ey - rad), (ex + rad, ey), (ex, ey + rad), (ex - rad, ey)]
    draw.line(diamond + [diamond[0]], fill=yellow, width=3, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    truth = _bits(scene["symbol"])
    return float(min(sum(a != b for a, b in zip(truth, _bits(other)))
                     for other in SYMBOLS if other != scene["symbol"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
