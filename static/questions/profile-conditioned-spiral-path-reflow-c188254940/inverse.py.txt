"""Independent image-only inverse for the spiral-path raster."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 12
NAMES = ("arrow", "cup", "bolt", "key")
_ART = {
    "arrow": ("............", ".....##.....", "....####....", "...######...", ".....##.....", ".....##.....", ".....##.....", ".....##.....", ".....##.....", ".....##.....", "............", "............"),
    "cup": ("............", "............", "..#......#..", "..#......#..", "..#......#..", "..#......#..", "..#......#..", "..#......#..", "..########..", "....####....", "............", "............"),
    "bolt": ("............", ".......###..", "......###...", ".....###....", "....###.....", "......###...", ".....###....", "....###.....", "...###......", "............", "............", "............"),
    "key": ("............", "............", "..####......", "..#..#......", "..#..##.....", "..####......", ".....##.....", ".....###....", ".....####...", ".....##.....", "............", "............"),
}


def _template(name: str) -> np.ndarray:
    return np.array([ch == "#" for row in _ART[name] for ch in row], dtype=np.uint8)


def _base_spiral() -> list[tuple[int, int]]:
    out = []
    top = left = 0
    bottom = right = N - 1
    while top <= bottom and left <= right:
        out += [(top, c) for c in range(left, right + 1)]
        top += 1
        out += [(r, right) for r in range(top, bottom + 1)]
        right -= 1
        if top <= bottom:
            out += [(bottom, c) for c in range(right, left - 1, -1)]
            bottom -= 1
        if left <= right:
            out += [(r, left) for r in range(bottom, top - 1, -1)]
            left += 1
    return out


def _rotate(rc: tuple[int, int], turns: int) -> tuple[int, int]:
    r, c = rc
    for _ in range(turns):
        r, c = c, N - 1 - r
    return r, c


def _measure(image: Image.Image) -> tuple[str, float]:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (arr[:, :, 0] < 100) & (arr[:, :, 1] > 120) & (arr[:, :, 2] > 130)
    yy, xx = np.where(cyan)
    if len(xx) < 300:
        return "abstain", 0.0
    ox, rx, oy, by = int(xx.min()), int(xx.max()), int(yy.min()), int(yy.max())
    span_x, span_y = rx - ox, by - oy
    if abs(span_x - span_y) > 5 or not 220 <= span_x <= 280:
        return "abstain", 0.0
    cell = (span_x + span_y) / (2 * N)

    magenta = (arr[:, :, 0] > 160) & (arr[:, :, 1] < 100) & (arr[:, :, 2] > 100)
    my, mx = np.where(magenta)
    yellow = (arr[:, :, 0] > 190) & (arr[:, :, 1] > 125) & (arr[:, :, 2] < 90)
    if len(mx) < 30 or int(yellow.sum()) < 20:
        return "abstain", 0.0
    mcx, mcy = float(mx.mean()), float(my.mean())
    corners = ((ox, oy), (rx, oy), (rx, by), (ox, by))
    turns = int(np.argmin([(mcx - x) ** 2 + (mcy - y) ** 2 for x, y in corners]))
    path = [_rotate(rc, turns) for rc in _base_spiral()]

    values = []
    radius = max(2, int(cell * 0.13))
    for r, c in path:
        cx = int(round(ox + (c + 0.5) * cell))
        cy = int(round(oy + (r + 0.5) * cell))
        patch = arr[cy - radius:cy + radius + 1, cx - radius:cx + radius + 1]
        if patch.size == 0:
            return "abstain", 0.0
        luminance = float(np.median(patch.mean(axis=2)))
        values.append(luminance < 135.0)
    observed = np.array(values, dtype=np.uint8)
    distances = sorted((int(np.count_nonzero(observed != _template(name))), name) for name in NAMES)
    gap = float(distances[1][0] - distances[0][0])
    if distances[0][0] > 5 or gap < 10:
        return "abstain", gap
    return distances[0][1], gap


def decision_from_image(image: Image.Image) -> str:
    return _measure(image)[0]


def decision_and_margin(image: Image.Image) -> tuple[str, float]:
    return _measure(image)
