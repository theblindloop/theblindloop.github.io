"""Deterministic renderer and analytic arm for Local Rail Crossing Count."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 800
BACKGROUND = (248, 249, 246)
NAVY = (24, 47, 72)
GOLD = (221, 166, 35)
BLUE = (38, 110, 210)
ORANGE = (232, 111, 32)
PALE = (207, 216, 219)

_LEVELS = {
    2: (18, 6, -6, -18, -6, -18, -6, 6, 18),
    4: (18, 6, -6, -18, -6, 6, -6, 6, 18),
}


def _rigid_point(x: int, y: int, quarter_turns: int, mirror: int, cx: int, cy: int) -> list[int]:
    y *= mirror
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return [cx + x, cy + y]


def sample_scene(seed: int) -> dict:
    """Sample balanced equal-ink paths with randomized rigid placement."""
    rng = random.Random(int(seed) ^ 0x51A7C0DE)
    crossing_count = 2 if int(seed) % 2 == 0 else 4
    quarter_turns = rng.randrange(4)
    mirror = rng.choice((-1, 1))
    centers = (150, 250, 350, 450, 550, 650)
    cx, cy = rng.choice(centers), rng.choice(centers)

    canonical = [[-60 + 15 * index, level] for index, level in enumerate(_LEVELS[crossing_count])]
    route_points = [_rigid_point(x, y, quarter_turns, mirror, cx, cy) for x, y in canonical]
    if rng.randrange(2):
        route_points.reverse()
    rail_points = [
        _rigid_point(-70, 0, quarter_turns, mirror, cx, cy),
        _rigid_point(70, 0, quarter_turns, mirror, cx, cy),
    ]

    decorations: list[list[int]] = []
    for _ in range(12):
        for _attempt in range(100):
            x = rng.randrange(45, WIDTH - 45)
            y = rng.randrange(45, HEIGHT - 45)
            if math.hypot(x - cx, y - cy) > 145:
                decorations.append([x, y, rng.randrange(2)])
                break

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "route_points": route_points,
        "rail_points": rail_points,
        "start_point": list(route_points[0]),
        "finish_point": list(route_points[-1]),
        "decorations": decorations,
    }


def _signed_distance(point: list[int], rail_a: list[int], rail_b: list[int]) -> float:
    dx = rail_b[0] - rail_a[0]
    dy = rail_b[1] - rail_a[1]
    return (dx * (point[1] - rail_a[1]) - dy * (point[0] - rail_a[0])) / math.hypot(dx, dy)


def _crossing_count(scene: dict) -> int:
    points = scene["route_points"]
    rail = scene["rail_points"]
    sides = [_signed_distance(point, rail[0], rail[1]) for point in points]
    if any(abs(value) < 1e-9 for value in sides):
        raise ValueError("a route vertex lies on the rail")
    return sum((a < 0) != (b < 0) for a, b in zip(sides, sides[1:]))


def analytic_gold(scene: dict) -> str:
    count = _crossing_count(scene)
    if count == 2:
        return "two"
    if count == 4:
        return "four"
    raise ValueError(f"unsupported crossing count: {count}")


def margin(scene: dict) -> float:
    points = scene["route_points"]
    rail = scene["rail_points"]
    vertex_clearance = min(abs(_signed_distance(point, rail[0], rail[1])) for point in points)
    crossing_positions = []
    for index, (left, right) in enumerate(zip(points, points[1:])):
        a = _signed_distance(left, rail[0], rail[1])
        b = _signed_distance(right, rail[0], rail[1])
        if (a < 0) != (b < 0):
            crossing_positions.append(index + abs(a) / (abs(a) + abs(b)))
    separation = min(
        (b - a) * 15.0 for a, b in zip(crossing_positions, crossing_positions[1:])
    ) if len(crossing_positions) > 1 else 999.0
    return float(min(vertex_clearance, separation / 2.0))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 5.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["route_points"] = list(reversed(twin["route_points"]))
    return [("reverse_undirected_vertex_storage", twin)]


def _disk(draw: ImageDraw.ImageDraw, point: list[int], radius: int, fill: tuple[int, int, int]) -> None:
    x, y = point
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)

    for x, y, vertical in scene["decorations"]:
        if vertical:
            draw.line((x, y - 8, x, y + 8), fill=PALE, width=3)
        else:
            draw.line((x - 8, y, x + 8, y), fill=PALE, width=3)

    rail = [tuple(point) for point in scene["rail_points"]]
    draw.line(rail, fill=GOLD, width=3)
    route = [tuple(point) for point in scene["route_points"]]
    draw.line(route, fill=NAVY, width=5, joint="curve")
    for point in scene["route_points"][1:-1]:
        _disk(draw, point, 2, NAVY)

    background = tuple(scene["background"])
    for point, color in ((scene["start_point"], BLUE), (scene["finish_point"], ORANGE)):
        _disk(draw, point, 6, color)
        _disk(draw, point, 3, background)
    return image
