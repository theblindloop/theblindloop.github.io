"""Deterministic scene sampler and renderer for the route-prefix task."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
ROUTE = [35, 55, 85]
BLUE = [25, 125, 230]
ORANGE = [245, 140, 25]
GREEN = [30, 175, 85]
RED = [220, 55, 60]
BACKGROUND = [250, 250, 247]


def _polyline_length(points: list[list[float]]) -> float:
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:]))


def _point_at(points: list[list[float]], distance: float) -> tuple[float, float]:
    remaining = distance
    for a, b in zip(points, points[1:]):
        length = math.hypot(b[0] - a[0], b[1] - a[1])
        if remaining <= length:
            t = remaining / length
            return a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])
        remaining -= length
    return tuple(points[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    x0 = 175 + rng.randint(-10, 10)
    x1 = 455 + rng.randint(-10, 10)
    x_finish = 350 + rng.randint(-8, 8)
    y0 = 135 + rng.randint(-8, 8)
    y_low = 330 + rng.randint(-8, 8)
    points = [
        [x0, y0], [x1, y0], [x1, y_low], [x_finish, y_low],
    ]
    # Reflecting/reversing counters endpoint position and traversal direction.
    if rng.randrange(2):
        points = [[WIDTH - x, y] for x, y in points]
    if rng.randrange(2):
        points = [[x, HEIGHT - y] for x, y in points]
    if rng.randrange(2):
        points.reverse()

    answer = 1 + (int(seed) % 3)
    tail = ["blue"] * (4 - answer) + ["orange"]
    rng.shuffle(tail)
    order = ["blue"] * answer + ["orange"] + tail
    length = _polyline_length(points)
    # Fixed broad spacing with seed-specific jitter; order, not total ink, carries gold.
    fractions = [0.105, 0.265, 0.425, 0.585, 0.745, 0.895]
    distances = [length * (f + rng.uniform(-0.012, 0.012)) for f in fractions]
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "path_points": points,
        "marker_order": order,
        "marker_distances": distances,
        "route_color": ROUTE,
        "background": BACKGROUND,
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    points = [[float(x), float(y)] for x, y in scene["path_points"]]
    order = list(scene["marker_order"])
    distances = [float(v) for v in scene["marker_distances"]]
    route_color = tuple(scene["route_color"])
    background = tuple(scene["background"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    scaled = [(round(x * scale), round(y * scale)) for x, y in points]
    draw.line(scaled, fill=route_color, width=8 * scale, joint="curve")

    length = _polyline_length(points)
    for color_name, distance in zip(order, distances):
        x, y = _point_at(points, distance)
        r = 11
        color = BLUE if color_name == "blue" else ORANGE
        box = [(x - r) * scale, (y - r) * scale, (x + r) * scale, (y + r) * scale]
        draw.ellipse(box, fill=tuple(color), outline=route_color, width=2 * scale)

    sx, sy = points[0]
    ex, ey = points[-1]
    r = 14
    draw.ellipse([(sx-r)*scale, (sy-r)*scale, (sx+r)*scale, (sy+r)*scale],
                 fill=tuple(GREEN), outline=route_color, width=2*scale)
    draw.rectangle([(ex-r)*scale, (ey-r)*scale, (ex+r)*scale, (ey+r)*scale],
                   fill=tuple(RED), outline=route_color, width=2*scale)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    count = 0
    for value in scene["marker_order"]:
        if value == "orange":
            return str(count)
        if value == "blue":
            count += 1
    raise ValueError("scene has no orange checkpoint")


def margin(scene: dict) -> float:
    distances = [float(v) for v in scene["marker_distances"]]
    order = list(scene["marker_order"])
    first_orange = order.index("orange")
    neighbors = []
    if first_orange:
        neighbors.append(distances[first_orange] - distances[first_orange - 1])
    if first_orange + 1 < len(distances):
        neighbors.append(distances[first_orange + 1] - distances[first_orange])
    return float(min(neighbors))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 34.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
