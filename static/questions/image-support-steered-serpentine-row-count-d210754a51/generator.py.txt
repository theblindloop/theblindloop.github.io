"""Serpentine route row-count world (profile: prior-conflict; QD support target:
regional | pathlike).

One single connected winding route (a serpentine polyline) is drawn within a
medium-sized central region of the canvas. The route meanders back and forth
across ``rows`` distinct horizontal rows, joined by short vertical connectors at
alternating ends, so it is exactly ONE connected thin component whose decisive
evidence is spread across a region and must be *traced* along its whole length.

The familiar canonical prior this image may contradict: a single drawn stroke is
one undifferentiated line and (for a reader) occupies one row / travels a single
path. In reality the one connected stroke sweeps through ``rows`` distinct
horizontal rows. The question asks to report the *visible* row-span of that one
line (2, 3, or 4 rows), which can disagree with the flat one-stroke / one-line
prior.

Decisive support is deliberately pathlike and regional: the independent pixel
oracle must recover the row-span by following the single connected route over a
medium region. Ablating any interior portion of the route disconnects the one
stroke and makes the oracle abstain - so many interior positions matter, and the
support is one connected thin component contained within a medium region (not
limited to endpoints, not a disconnected card set).

Latent state z = {rows, row_spacing, x_left, x_right, y_top, stroke_width,
route_color}. The analytic gold is ``rows`` (a scene parameter), and the margin
is the vertical clearance between adjacent route ink bands (row_spacing minus the
stroke width) - large margin => clearly separated rows, small margin => rows nearly
fuse and the count is pixel-ambiguous (quarantined near zero).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "serpentine_row_count-0.1.0"

CANVAS = (512, 512)
BACKGROUND = "white"
ROUTE_COLOR = (24, 84, 200)  # vivid "blue" route on white

ROWS_RANGE = (2, 4)          # inclusive: 2, 3, or 4 distinct rows
ROW_SPACING_RANGE = (36.0, 60.0)
STROKE_WIDTH = 13.0
XSPAN_RANGE = (300.0, 348.0)  # horizontal span of each row (kept regional)
PAD = 60.0                    # route box stays clear of all four edges

# A row-count is pixel-ambiguous when adjacent ink bands come too close: below
# this clearance the oracle cannot certify separation. Approached from above.
QUARANTINE_MARGIN = 12.0


def sample_scene(seed: int) -> dict:
    """Deterministic scene sampler. Returns a JSON-friendly latent dict."""
    rng = random.Random(seed)
    rows = rng.randint(*ROWS_RANGE)
    spacing = rng.uniform(*ROW_SPACING_RANGE)
    xspan = rng.uniform(*XSPAN_RANGE)
    x_left = rng.uniform(PAD, CANVAS[0] - PAD - xspan)
    x_right = x_left + xspan
    # Vertical extent of the serpentine (rows + connectors) kept within a medium
    # band near the canvas centre so the decisive support stays regional.
    ext = (rows - 1) * spacing + STROKE_WIDTH
    y_top = rng.uniform(max(PAD, CANVAS[1] / 2 - ext / 2), CANVAS[1] / 2 + ext / 2 - ext)
    return {
        "rows": int(rows),
        "row_spacing": round(spacing, 3),
        "x_left": round(x_left, 3),
        "x_right": round(x_right, 3),
        "y_top": round(y_top, 3),
        "stroke_width": STROKE_WIDTH,
        "route_color": "rgb(24,84,200)",
    }


def _route_points(scene: dict) -> list[tuple[float, float]]:
    """The single connected serpentine polyline (row strokes + vertical joins)."""
    rows = int(scene["rows"])
    spacing = float(scene["row_spacing"])
    x_left = float(scene["x_left"])
    x_right = float(scene["x_right"])
    y_top = float(scene["y_top"])
    pts: list[tuple[float, float]] = [(x_left, y_top), (x_right, y_top)]
    dir_right = True
    for i in range(1, rows):
        ny = y_top + i * spacing
        if dir_right:      # previous row ended at the right edge -> drop then go left
            pts.append((x_right, ny))
            pts.append((x_left, ny))
        else:              # previous row ended at the left edge -> drop then go right
            pts.append((x_left, ny))
            pts.append((x_right, ny))
        dir_right = not dir_right
    return pts


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(img)
    pts = _route_points(scene)
    draw.line(pts, fill=ROUTE_COLOR, width=int(scene["stroke_width"]), joint="curve")
    return img


def analytic_gold(scene: dict) -> str:
    """Closed-form gold: the number of distinct horizontal rows swept by the route."""
    return str(int(scene["rows"]))


def margin(scene: dict) -> float:
    """Vertical clearance between adjacent route ink bands (row_spacing - width)."""
    return float(scene["row_spacing"]) - float(scene["stroke_width"])


def is_quarantined(scene: dict) -> bool:
    """Quarantine when rows are out of range or adjacent ink bands nearly fuse."""
    rows = int(scene["rows"])
    if rows not in (2, 3, 4):
        return True
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    """not_applicable mode: no pixel-identical nontrivial transform is declared."""
    return []
