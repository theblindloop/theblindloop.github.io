"""Twin seal depth duel: two disconnected ring stacks inside one small inset.

Each stack draws two or three concentric black square rings around a small white
core.  A ring is either unbroken or carries one visible gap.  A gapped ring lets
the core reach the outside, so the core's enclosure depth is the number of
unbroken rings in its own stack.  The two stacks always differ by exactly one
unbroken ring and the decision names the stack with the deeper core.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (960, 620)
INK = (0, 0, 0)
BG = (255, 255, 255)
STROKE = 2

# Decoration inks are all far brighter than the ink threshold used by the pixel
# arm, so no decoration can ever act as a barrier.
DECOR_COLORS = [
    (216, 78, 78),
    (72, 108, 216),
    (232, 148, 44),
    (150, 96, 196),
    (60, 158, 108),
    (176, 176, 176),
]


def _rings(rng: random.Random) -> tuple[int, int, int]:
    corridor = rng.randint(5, 6)
    r1 = rng.randint(6, 7)
    return corridor, r1, corridor + STROKE


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)

    # Unbroken-ring counts differ by exactly one; both are at least one so both
    # stacks always show at least one enclosed core.
    deeper_left = (seed % 2 == 0) if rng.random() < 0.5 else rng.random() < 0.5
    low = rng.randint(1, 2)
    high = low + 1
    left_sealed, right_sealed = (high, low) if deeper_left else (low, high)

    stacks = []
    for sealed in (left_sealed, right_sealed):
        count = rng.randint(max(2, sealed), 3)
        corridor, r1, step = _rings(rng)
        radii = [r1 + k * step for k in range(count)]
        # Gaps never land on the innermost ring: its side is too short to host an
        # opening with clear stubs at either end, which is a raster-soundness
        # constraint only, since enclosure depth does not depend on which rings
        # are ajar.
        eligible = [k for k in range(count) if 2 * radii[k] >= 24]
        gapped = sorted(rng.sample(eligible, count - sealed))
        gaps = {}
        for idx in gapped:
            side = rng.choice(("top", "bottom", "left", "right"))
            span = 2 * radii[idx]
            width = min(rng.randint(5, 11), max(5, span - 14))
            offset = rng.randint(-(span // 2 - width // 2 - 5), span // 2 - width // 2 - 5)
            gaps[idx] = {"side": side, "width": width, "offset": offset}
        stacks.append(
            {
                "count": count,
                "sealed": sealed,
                "corridor": corridor,
                "radii": radii,
                "gaps": gaps,
            }
        )

    span_left = 2 * stacks[0]["radii"][-1] + STROKE
    span_right = 2 * stacks[1]["radii"][-1] + STROKE
    separation = rng.randint(235, 255)
    inset_w = span_left + separation + span_right
    inset_h = max(span_left, span_right)
    ox = rng.randint(40, CANVAS[0] - inset_w - 40)
    oy = rng.randint(40, CANVAS[1] - inset_h - 40)
    cy = oy + inset_h // 2
    stacks[0]["center"] = [ox + span_left // 2, cy]
    stacks[1]["center"] = [ox + span_left + separation + span_right // 2, cy]

    inset_box = [ox - 14, oy - 14, ox + inset_w + 14, oy + inset_h + 14]

    # Black distractor glyphs live inside the same inset but enclose nothing: an
    # always-open ring and one right-angled hook.
    black_decoys = []
    for _ in range(2):
        for _try in range(40):
            r = rng.randint(8, 13)
            bx = rng.randint(ox - 4, ox + inset_w + 4)
            by = rng.choice([oy - 34 - r, oy + inset_h + 34 + r])
            if not (34 <= bx <= CANVAS[0] - 34 and 34 <= by <= CANVAS[1] - 34):
                continue
            ok = True
            for stack in stacks:
                cxs, cys = stack["center"]
                reach = stack["radii"][-1] + r + 20
                if abs(bx - cxs) < reach and abs(by - cys) < reach:
                    ok = False
            for prev in black_decoys:
                if abs(bx - prev["center"][0]) < 2 * r + 26 and abs(by - prev["center"][1]) < 2 * r + 26:
                    ok = False
            if ok:
                kind = "open_ring" if not black_decoys else "hook"
                black_decoys.append({"kind": kind, "center": [bx, by], "radius": r})
                break
        if len(black_decoys) == 2:
            break

    decor = []
    for _ in range(rng.randint(9, 13)):
        for _try in range(60):
            kind = rng.choice(("ring", "dot", "bar"))
            w = rng.randint(10, 30)
            h = rng.randint(10, 30) if kind != "bar" else rng.randint(4, 8)
            x = rng.randint(16, CANVAS[0] - w - 16)
            y = rng.randint(16, CANVAS[1] - h - 16)
            if not (x + w + 8 < inset_box[0] or x - 8 > inset_box[2] or y + h + 8 < inset_box[1] or y - 8 > inset_box[3]):
                continue
            clear = True
            for glyph in black_decoys:
                gx, gy = glyph["center"]
                pad = glyph["radius"] + 12
                if not (x + w < gx - pad or x > gx + pad or y + h < gy - pad or y > gy + pad):
                    clear = False
            if not clear:
                continue
            decor.append(
                {
                    "kind": kind,
                    "box": [x, y, x + w, y + h],
                    "color": list(DECOR_COLORS[rng.randrange(len(DECOR_COLORS))]),
                }
            )
            break

    return {
        "canvas": list(CANVAS),
        "stroke_px": STROKE,
        "stacks": stacks,
        "black_decoys": black_decoys,
        "decor": decor,
    }


def _ring_bands(cx: int, cy: int, r: int, gap: dict | None) -> list[tuple[int, int, int, int]]:
    s = STROKE
    x0, y0, x1, y1 = cx - r, cy - r, cx + r, cy + r
    bands = {
        "top": (x0, y0, x1, y0 + s - 1),
        "bottom": (x0, y1 - s + 1, x1, y1),
        "left": (x0, y0, x0 + s - 1, y1),
        "right": (x1 - s + 1, y0, x1, y1),
    }
    out = []
    for side, box in bands.items():
        if gap is None or gap["side"] != side:
            out.append(box)
            continue
        half = gap["width"] // 2
        if side in ("top", "bottom"):
            mid = cx + gap["offset"]
            out.append((box[0], box[1], mid - half, box[3]))
            out.append((mid + half, box[1], box[2], box[3]))
        else:
            mid = cy + gap["offset"]
            out.append((box[0], box[1], box[2], mid - half))
            out.append((box[0], mid + half, box[2], box[3]))
    return [b for b in out if b[2] >= b[0] and b[3] >= b[1]]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), BG)
    draw = ImageDraw.Draw(img)

    for mark in scene["decor"]:
        color = tuple(mark["color"])
        box = mark["box"]
        if mark["kind"] == "ring":
            draw.rectangle(box, outline=color, width=3)
        elif mark["kind"] == "dot":
            draw.ellipse(box, fill=color)
        else:
            draw.rectangle(box, fill=color)

    for glyph in scene["black_decoys"]:
        cx, cy = glyph["center"]
        r = glyph["radius"]
        if glyph["kind"] == "open_ring":
            gap = {"side": "right", "width": 9, "offset": 0}
            for band in _ring_bands(cx, cy, r, gap):
                draw.rectangle(band, fill=INK)
        else:
            draw.rectangle((cx - r, cy - r, cx - r + STROKE - 1, cy + r), fill=INK)
            draw.rectangle((cx - r, cy + r - STROKE + 1, cx + r, cy + r), fill=INK)

    for stack in scene["stacks"]:
        cx, cy = stack["center"]
        for idx, r in enumerate(stack["radii"]):
            gap = stack["gaps"].get(idx) or stack["gaps"].get(str(idx))
            for band in _ring_bands(cx, cy, r, gap):
                draw.rectangle(band, fill=INK)
    return img


def analytic_gold(scene: dict) -> str:
    left, right = scene["stacks"]
    return "left" if left["sealed"] > right["sealed"] else "right"


def margin(scene: dict) -> float:
    """Weakest decisive clearance in pixels: a corridor width or a gap width."""
    worst = 99.0
    for stack in scene["stacks"]:
        worst = min(worst, float(stack["corridor"]))
        for gap in stack["gaps"].values():
            worst = min(worst, float(gap["width"]))
    return worst


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list:
    """No latent alias is declared: every gold input is drawn into the raster."""
    return []
