"""Pixel-only inverse arm: enclosure depth of each ring stack from the PNG.

Reads nothing but the rasterised image.  Dark ink is segmented, free space is
peeled outward-in so every white pixel gets an enclosure depth, enclosed pockets
are grouped into stacks by spatial clustering, and the deeper stack wins.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

INK_MAX = 90
DILATE = 5
MAX_DEPTH = 8


def _grow(cur: np.ndarray, mask: np.ndarray, diagonal: bool) -> np.ndarray:
    out = cur.copy()
    out[1:, :] |= cur[:-1, :]
    out[:-1, :] |= cur[1:, :]
    out[:, 1:] |= cur[:, :-1]
    out[:, :-1] |= cur[:, 1:]
    if diagonal:
        out[1:, 1:] |= cur[:-1, :-1]
        out[1:, :-1] |= cur[:-1, 1:]
        out[:-1, 1:] |= cur[1:, :-1]
        out[:-1, :-1] |= cur[1:, 1:]
    return out & mask


def _flood(mask: np.ndarray, seeds: np.ndarray, diagonal: bool = False) -> np.ndarray:
    cur = seeds & mask
    total = int(cur.sum())
    while True:
        cur = _grow(cur, mask, diagonal)
        grown = int(cur.sum())
        if grown == total:
            return cur
        total = grown


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    cur = mask.copy()
    ones = np.ones_like(mask)
    for _ in range(rounds):
        cur = _grow(cur, ones, True)
    return cur


def _components(mask: np.ndarray) -> list[np.ndarray]:
    remaining = mask.copy()
    comps = []
    while remaining.any():
        ys, xs = np.nonzero(remaining)
        seed = np.zeros_like(mask)
        seed[ys[0], xs[0]] = True
        comp = _flood(mask, seed, True)
        comps.append(comp)
        remaining &= ~comp
        if len(comps) > 24:
            break
    return comps


def _depth_map(ink: np.ndarray) -> np.ndarray:
    free = ~ink
    border = np.zeros_like(ink)
    border[0, :] = True
    border[-1, :] = True
    border[:, 0] = True
    border[:, -1] = True
    visited = _flood(free, border & free)
    depth = np.full(ink.shape, -1, dtype=np.int16)
    depth[visited] = 0
    crossed = np.zeros_like(ink)
    for level in range(1, MAX_DEPTH + 1):
        touching = _dilate(visited, 1) & ink & ~crossed
        if not touching.any():
            break
        layer = _flood(ink & ~crossed, touching, True)
        crossed |= layer
        seeds = _dilate(layer, 1) & free & ~visited
        if not seeds.any():
            break
        region = _flood(free, seeds)
        depth[region & ~visited] = level
        visited |= region
    return depth


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    ink = (arr[:, :, 0] < INK_MAX) & (arr[:, :, 1] < INK_MAX) & (arr[:, :, 2] < INK_MAX)
    if not ink.any():
        return "abstain"

    ys, xs = np.nonzero(ink)
    pad = 14
    y0 = max(int(ys.min()) - pad, 0)
    y1 = min(int(ys.max()) + pad + 1, ink.shape[0])
    x0 = max(int(xs.min()) - pad, 0)
    x1 = min(int(xs.max()) + pad + 1, ink.shape[1])
    crop = ink[y0:y1, x0:x1]

    depth = _depth_map(crop)
    pockets = depth >= 1
    if not pockets.any():
        return "abstain"

    clusters = _components(_dilate(pockets, DILATE))
    scored = []
    for cluster in clusters:
        inside = pockets & cluster
        if not inside.any():
            continue
        cys, cxs = np.nonzero(inside)
        scored.append((float(cxs.mean()), int(depth[inside].max())))
    if len(scored) != 2:
        return "abstain"

    scored.sort(key=lambda item: item[0])
    left_depth = scored[0][1]
    right_depth = scored[1][1]
    if left_depth == right_depth:
        return "abstain"
    return "left" if left_depth > right_depth else "right"
