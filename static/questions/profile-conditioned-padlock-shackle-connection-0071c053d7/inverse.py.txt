"""Independent raster-only endpoint-gap oracle for the padlock world."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _body_bounds(pixels: np.ndarray) -> tuple[int, int, int, int]:
    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    saturated = (high - low > 48) & (high > 105) & (low < 155)
    ys, xs = np.nonzero(saturated)
    if xs.size < 12000:
        raise ValueError("colored lock body evidence is absent")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _side_gap(dark: np.ndarray, x0: int, x1: int, body_top: int) -> float:
    ys, _ = np.nonzero(dark[:, x0:x1])
    if ys.size < 180:
        raise ValueError("shackle endpoint evidence is absent")
    return float(body_top - int(ys.max()))


def decision_from_image(image: Image.Image) -> str:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int16)
    left, top, right, bottom = _body_bounds(pixels)
    width = right - left + 1
    height = bottom - top + 1
    if width < 150 or height < 95:
        raise ValueError("lock body is too small for a reliable decision")

    high = pixels.max(axis=2)
    low = pixels.min(axis=2)
    dark = (high < 92) & (high - low < 28)
    center = (left + right) / 2.0
    left_gap = _side_gap(dark, int(left + 0.06 * width), int(center - 0.08 * width), top)
    right_gap = _side_gap(dark, int(center + 0.08 * width), int(right - 0.06 * width), top)

    threshold = 13.0
    guard = 4.0
    for gap in (left_gap, right_gap):
        if abs(gap - threshold) < guard:
            raise ValueError("endpoint gap is ambiguous at pixel resolution")
    separated = [name for name, gap in (("left", left_gap), ("right", right_gap)) if gap > threshold]
    if not separated:
        return "neither"
    if len(separated) == 1:
        seated_gap = right_gap if separated[0] == "left" else left_gap
        if seated_gap > 7.0:
            raise ValueError("no shackle endpoint is clearly seated")
        return separated[0]
    raise ValueError("both shackle endpoints appear separated")
