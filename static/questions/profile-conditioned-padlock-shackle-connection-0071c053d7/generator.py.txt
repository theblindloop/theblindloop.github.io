"""Matched padlock scenes with a rigid, constant-length shackle."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 384
SCALE = 4
SHACKLE = (42, 48, 57)
BACKGROUND = (239, 242, 245)
BODY_PALETTES = ((35, 121, 184), (27, 145, 128), (126, 86, 174))


def sample_scene(seed: int) -> dict:
    # The three consecutive labels share all nuisance variables exactly.
    # Only the rigid shackle connection state changes within a matched triplet.
    rng = random.Random((seed // 3) ^ 0x5A17C1E)
    body_width = rng.randrange(202, 221, 2)
    body_height = rng.randrange(124, 141, 2)
    return {
        "connection": ("left", "neither", "right")[seed % 3],
        "center_x": 192 + rng.randint(-11, 11),
        "body_top": 196 + rng.randint(-7, 7),
        "body_width": body_width,
        "body_height": body_height,
        "shackle_width": rng.randrange(120, 139, 2),
        "shackle_height": rng.randrange(102, 119, 2),
        "rotation_deg": rng.choice((23.0, 24.5, 26.0)),
        "body_color": list(BODY_PALETTES[rng.randrange(len(BODY_PALETTES))]),
        "screw_order": ["left", "right"] if rng.randrange(2) else ["right", "left"],
    }


def _rotate(point: tuple[float, float], pivot: tuple[float, float], degrees: float) -> tuple[float, float]:
    angle = math.radians(degrees)
    px, py = point
    ox, oy = pivot
    dx, dy = px - ox, py - oy
    return (
        ox + dx * math.cos(angle) - dy * math.sin(angle),
        oy + dx * math.sin(angle) + dy * math.cos(angle),
    )


def _shackle_points(scene: dict) -> list[tuple[float, float]]:
    cx = float(scene["center_x"])
    body_top = float(scene["body_top"])
    width = float(scene["shackle_width"])
    height = float(scene["shackle_height"])
    radius = width / 2.0
    left = (cx - radius, body_top + 16.0)
    right = (cx + radius, body_top + 16.0)
    arc_cy = body_top + 16.0 - height + radius
    points = [left, (left[0], arc_cy)]
    for index in range(1, 41):
        angle = math.pi + math.pi * index / 40.0
        points.append((cx + radius * math.cos(angle), arc_cy + radius * math.sin(angle)))
    points.append(right)

    connection = scene["connection"]
    if connection == "left":
        return [_rotate(point, right, float(scene["rotation_deg"])) for point in points]
    if connection == "right":
        return [_rotate(point, left, -float(scene["rotation_deg"])) for point in points]
    return points


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE * SCALE, SIZE * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)

    def scaled(point: tuple[float, float]) -> tuple[int, int]:
        return int(round(point[0] * SCALE)), int(round(point[1] * SCALE))

    shackle = [scaled(point) for point in _shackle_points(scene)]
    draw.line(shackle, fill=SHACKLE, width=18 * SCALE, joint="curve")

    cx = float(scene["center_x"])
    top = float(scene["body_top"])
    width = float(scene["body_width"])
    height = float(scene["body_height"])
    left = cx - width / 2.0
    right = cx + width / 2.0
    bottom = top + height
    body_color = tuple(scene["body_color"])
    body_box = tuple(value * SCALE for value in (left, top, right, bottom))
    draw.rounded_rectangle(body_box, radius=18 * SCALE, fill=body_color)

    # Nonoverlapping screw heads provide a render-invariant draw-order alias.
    screw_y = top + 25.0
    screw_x = {"left": left + 27.0, "right": right - 27.0}
    for name in scene["screw_order"]:
        x = screw_x[name]
        box = tuple(value * SCALE for value in (x - 6, screw_y - 6, x + 6, screw_y + 6))
        draw.ellipse(box, fill=(218, 224, 229))
        draw.line(
            ((x - 3) * SCALE, screw_y * SCALE, (x + 3) * SCALE, screw_y * SCALE),
            fill=(116, 126, 136), width=2 * SCALE,
        )

    # A neutral keyhole reinforces the familiar object without encoding the class.
    key_y = top + height * 0.53
    draw.ellipse(
        tuple(value * SCALE for value in (cx - 12, key_y - 12, cx + 12, key_y + 12)),
        fill=(235, 226, 186),
    )
    draw.polygon(
        [scaled((cx - 7, key_y + 8)), scaled((cx + 7, key_y + 8)), scaled((cx + 10, key_y + 33)), scaled((cx - 10, key_y + 33))],
        fill=(235, 226, 186),
    )
    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["connection"])


def margin(scene: dict) -> float:
    if scene["connection"] == "neither":
        return 28.0
    lift = float(scene["shackle_width"]) * math.sin(math.radians(float(scene["rotation_deg"])))
    return lift - 18.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["screw_order"] = list(reversed(scene["screw_order"]))
    return [("reverse_nonoverlapping_screw_order", twin)]
