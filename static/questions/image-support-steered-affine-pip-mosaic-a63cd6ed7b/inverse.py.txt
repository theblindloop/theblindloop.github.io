from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

ABSTAIN = "__abstain__"

def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points = []
        while queue:
            yy, xx = queue.popleft()
            points.append((yy, xx))
            for ny, nx in ((yy-1, xx), (yy+1, xx), (yy, xx-1), (yy, xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 55:
            a = np.asarray(points, dtype=float)
            out.append((len(points), float(a[:, 1].mean()), float(a[:, 0].mean())))
    return out

def _kind(pixels: np.ndarray) -> tuple[str, float]:
    r, g, b = pixels[..., 0], pixels[..., 1], pixels[..., 2]
    cyan = (b > 150) & (g > 125) & (r < 80)
    coral = (r > 180) & (g < 145) & (b < 145)
    gold = (r > 190) & (g > 140) & (g < 225) & (b < 120)
    fractions = {"add": float(cyan.mean()), "double": float(coral.mean()), "start": float(gold.mean())}
    kind = max(fractions, key=fractions.get)
    return kind, fractions[kind]

def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.ndim != 3 or arr.shape[0] < 300 or arr.shape[1] < 300:
        return ABSTAIN
    # The independently inferred compact board is every substantially non-paper pixel.
    nonpaper = np.any(arr < 225, axis=2)
    ys, xs = np.nonzero(nonpaper)
    if len(xs) < 35000:
        return ABSTAIN
    xmin, xmax, ymin, ymax = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = xmax - xmin + 1, ymax - ymin + 1
    if not (260 <= width <= 270 and 260 <= height <= 270):
        return ABSTAIN
    x0, y0 = xmin + 7, ymin + 7
    cell = (width - 14) / 3.0
    kinds = []
    start_crop = None
    for index in range(9):
        row, col = divmod(index, 3)
        xa = int(round(x0 + col * cell + 3))
        xb = int(round(x0 + (col + 1) * cell - 3)) + 1
        ya = int(round(y0 + row * cell + 3))
        yb = int(round(y0 + (row + 1) * cell - 3)) + 1
        crop = arr[ya:yb, xa:xb]
        if crop.shape[0] < 72 or crop.shape[1] < 72:
            return ABSTAIN
        kind, coverage = _kind(crop)
        # Coverage is an integrity check: a grid ablation or erased tile cannot be replayed.
        if coverage < 0.72:
            return ABSTAIN
        kinds.append(kind)
        if index == 0:
            start_crop = crop
    if kinds[0] != "start" or any(kind == "start" for kind in kinds[1:]):
        return ABSTAIN
    assert start_crop is not None
    dark = np.all(start_crop < 95, axis=2)
    comps = _components(dark)
    # Ignore border spill; ordinary pips/ring lie in the lower 70% of the start cell.
    comps = [(area, cx, cy) for area, cx, cy in comps if 18 < cx < 62 and 27 < cy < 72 and area < 450]
    if len(comps) == 1:
        area, cx, cy = comps[0]
        # The zero ring is centered; one pip is visibly upper-left.
        start = 0 if (34 <= cx <= 44 and 44 <= cy <= 54 and area > 180) else 1
    elif 2 <= len(comps) <= 4:
        start = len(comps)
    else:
        return ABSTAIN
    value = start
    for kind in kinds[1:]:
        value = (value + 1) % 5 if kind == "add" else (2 * value) % 5
    return str(value)
