from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 512
PALETTES = [
    {"add": (23, 176, 204), "double": (235, 105, 92), "start": (235, 190, 45)},
    {"add": (25, 165, 199), "double": (224, 91, 105), "start": (239, 185, 52)},
]

def _replay(start: int, ops: list[str]) -> int:
    value = start
    for op in ops:
        value = (value + 1) % 5 if op == "add" else (2 * value) % 5
    return value

def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start = rng.randrange(5)
    # Both operations always occur; parity of seed-independent shuffle prevents a count shortcut.
    ops = ["add"] * 4 + ["double"] * 4
    rng.shuffle(ops)
    return {
        "start": start,
        "ops": ops,
        "palette_variant": rng.randrange(2),
        "layout_variant": rng.randrange(3),
    }

def analytic_gold(scene: dict) -> str:
    return str(_replay(int(scene["start"]), list(scene["ops"])))

def margin(scene: dict) -> float:
    return 28.0

def is_quarantined(scene: dict) -> bool:
    return not (0 <= int(scene.get("start", -1)) <= 4 and len(scene.get("ops", [])) == 8 and set(scene["ops"]) <= {"add", "double"})

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    alt = dict(scene)
    alt["palette_variant"] = int(scene["palette_variant"]) + 2
    moved = dict(scene)
    moved["layout_variant"] = int(scene["layout_variant"]) + 3
    return [("palette_alias", alt), ("layout_alias", moved)]

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), (248, 247, 242))
    draw = ImageDraw.Draw(image)
    palette = PALETTES[int(scene["palette_variant"]) % 2]
    offsets = [(110, 110), (102, 114), (114, 102)]
    x0, y0 = offsets[int(scene["layout_variant"]) % 3]
    cell = 84
    gap = 3
    dark = (35, 43, 52)
    # A dark connected backing makes the nine cells one compact visual component.
    draw.rounded_rectangle((x0 - 7, y0 - 7, x0 + 3 * cell + 7, y0 + 3 * cell + 7), radius=16, fill=dark)
    tiles = ["start"] + list(scene["ops"])
    for index, kind in enumerate(tiles):
        row, col = divmod(index, 3)
        xa, ya = x0 + col * cell + gap, y0 + row * cell + gap
        xb, yb = x0 + (col + 1) * cell - gap, y0 + (row + 1) * cell - gap
        draw.rounded_rectangle((xa, ya, xb, yb), radius=11, fill=palette[kind])
        if kind == "start":
            draw.rectangle((xa + 8, ya + 8, xb - 8, ya + 23), fill=(255, 244, 194))
            # Zero is visibly an empty ring; 1..4 use separated dark pips.
            count = int(scene["start"])
            if count == 0:
                draw.ellipse((xa + 28, ya + 38, xa + 50, ya + 60), outline=dark, width=5)
            else:
                points = [(xa + 28, ya + 38), (xa + 50, ya + 38), (xa + 28, ya + 59), (xa + 50, ya + 59)]
                for px, py in points[:count]:
                    draw.ellipse((px - 6, py - 6, px + 6, py + 6), fill=dark)
        elif kind == "add":
            # Broad white plus sign, deliberately not text-dependent.
            cx, cy = (xa + xb) // 2, (ya + yb) // 2
            draw.rounded_rectangle((cx - 9, ya + 18, cx + 9, yb - 18), radius=5, fill=(250, 250, 247))
            draw.rounded_rectangle((xa + 18, cy - 9, xb - 18, cy + 9), radius=5, fill=(250, 250, 247))
        else:
            # Broad white double bar signifies x2; tile color remains the declared operation cue.
            draw.rounded_rectangle((xa + 21, ya + 22, xb - 21, ya + 34), radius=5, fill=(250, 250, 247))
            draw.rounded_rectangle((xa + 21, yb - 34, xb - 21, yb - 22), radius=5, fill=(250, 250, 247))
    return image
