"""Independent pixel-only inverse for the Pac-Man mouth direction.

The inverse sees only the final RGB raster.  It isolates the single saturated
body component, estimates four candidate circle centres from the intact
orthogonal extrema, and selects the candidate whose boundary has the strongest
continuous circular arc.  It does not import renderer, prompts, generator,
verifier, scene data, or analytic gold.
"""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "pacman_facing_direction-oracle-0.1.0"
DIRECTIONS = ("right", "down", "left", "up")


def _body_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = arr.max(axis=2) - arr.min(axis=2)
    # The body is the only large, strongly chromatic mark.  Neutral outlines,
    # eye, pupil, and pale background are excluded without reading their colors.
    return (spread >= 70) & (arr.max(axis=2) >= 100)


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y = int(y)
        x = int(x)
        if visited[y, x]:
            continue
        queue = deque([(y, x)])
        visited[y, x] = True
        current: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            current.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    queue.append((ny, nx))
        if len(current) > len(best):
            best = current
    if len(best) < 500:
        raise ValueError("no measurable colored character silhouette")
    component = np.zeros_like(mask, dtype=bool)
    ys, xs = zip(*best)
    component[np.asarray(ys), np.asarray(xs)] = True
    return component


def _boundary(component: np.ndarray) -> np.ndarray:
    interior = component.copy()
    for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
        shifted = np.zeros_like(component)
        ys = slice(max(dy, 0), component.shape[0] + min(dy, 0))
        xs = slice(max(dx, 0), component.shape[1] + min(dx, 0))
        ysrc = slice(max(-dy, 0), component.shape[0] + min(-dy, 0))
        xsrc = slice(max(-dx, 0), component.shape[1] + min(-dx, 0))
        shifted[ys, xs] = component[ysrc, xsrc]
        interior &= shifted
    points = np.argwhere(component & ~interior)
    if len(points) < 100:
        raise ValueError("colored silhouette has too few boundary pixels")
    # Pixel-centre coordinates in x,y order.
    return points[:, ::-1].astype(np.float64)


def _candidate_centres(xs: np.ndarray, ys: np.ndarray) -> dict[str, tuple[float, float, float]]:
    xmin, xmax = float(xs.min()), float(xs.max())
    ymin, ymax = float(ys.min()), float(ys.max())
    radius_x = (xmax - xmin) / 2.0
    radius_y = (ymax - ymin) / 2.0
    cx_mid = (xmin + xmax) / 2.0
    cy_mid = (ymin + ymax) / 2.0
    # A mouth removes one cardinal sector.  The perpendicular axis retains both
    # extrema, and the opposite cardinal extremum supplies the circle centre.
    return {
        "right": (xmin + radius_y, cy_mid, radius_y),
        "left": (xmax - radius_y, cy_mid, radius_y),
        "down": (cx_mid, ymin + radius_x, radius_x),
        "up": (cx_mid, ymax - radius_x, radius_x),
    }


def _fit_scores(component: np.ndarray, boundary: np.ndarray) -> dict[str, float]:
    ys, xs = np.nonzero(component)
    candidates = _candidate_centres(xs.astype(float), ys.astype(float))
    scores: dict[str, float] = {}
    for direction, (cx, cy, radius) in candidates.items():
        if radius < 20.0:
            scores[direction] = float("inf")
            continue
        distances = np.hypot(boundary[:, 0] - cx, boundary[:, 1] - cy)
        residual = np.abs(distances - radius)
        # The mouth's two radial edges are intentionally treated as outliers;
        # the 55th percentile retains the majority of the curved outer arc while
        # rejecting centres shifted toward the missing wedge.  This is a
        # geometry-only score and does not assume a particular fill color.
        scores[direction] = float(np.percentile(residual, 55.0))
    return scores


def _estimate(image: Image.Image) -> tuple[str, float]:
    component = _largest_component(_body_mask(image))
    boundary = _boundary(component)
    scores = _fit_scores(component, boundary)
    ordered = sorted(scores, key=lambda direction: (scores[direction], DIRECTIONS.index(direction)))
    winner, runner = ordered[0], ordered[1]
    # Lower residual is better, so the gap is runner - winner.
    return winner, round(float(scores[runner] - scores[winner]), 6)


def decision_from_image(image: Image.Image) -> str:
    """Return only the facing direction recovered from the PNG."""

    return _estimate(image)[0]


def decision_margin(image: Image.Image) -> float:
    """Return the winner-versus-runner-up geometric residual gap."""

    return _estimate(image)[1]
