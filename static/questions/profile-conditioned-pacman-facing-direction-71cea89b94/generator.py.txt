"""Deterministic raster renderer for the visible Pac-Man direction world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "pacman_facing_direction-renderer-0.1.0"
WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
DIRECTIONS = ("right", "down", "left", "up")
_ANGLE_DEG = {"right": 0.0, "down": 90.0, "left": 180.0, "up": 270.0}

# Every body colour is strongly separated from every pale background.  The
# oracle uses only this public visual fact, not these Python constants.
FILL_PALETTE = (
    (235, 193, 39),
    (224, 68, 55),
    (47, 128, 218),
    (49, 171, 103),
    (159, 79, 202),
    (26, 161, 171),
)
BACKGROUND_PALETTE = (
    (255, 255, 255),
    (248, 245, 237),
    (239, 247, 249),
    (247, 242, 250),
    (244, 248, 239),
)
OUTLINE_PALETTE = (
    (25, 30, 40),
    (35, 39, 48),
    (40, 32, 45),
)
EYE_WHITE = (255, 250, 235)
EYE_DARK = (25, 30, 40)


def _rgb(value: Any) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _box(cx: float, cy: float, radius: float, scale: int) -> tuple[float, ...]:
    return (
        (cx - radius) * scale,
        (cy - radius) * scale,
        (cx + radius) * scale,
        (cy + radius) * scale,
    )


def _point(cx: float, cy: float, radius: float, angle_deg: float, scale: int) -> tuple[int, int]:
    angle = math.radians(angle_deg)
    return (
        int(round((cx + radius * math.cos(angle)) * scale)),
        int(round((cy + radius * math.sin(angle)) * scale)),
    )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, JSON-friendly scene from one integer seed.

    The direction is on an independent four-cycle: the public generator's
    stride of 1009 visits each direction equally often.  Style and geometry are
    sampled from a separate RNG so direction does not choose the palette or
    pose.
    """

    seed = int(seed)
    direction = DIRECTIONS[seed % len(DIRECTIONS)]
    rng = random.Random(seed * 1_000_003 + 7919)
    near_boundary = seed % 17 == 0
    if near_boundary:
        radius = rng.uniform(54.0, 66.0)
        mouth_half_angle = rng.uniform(22.0, 25.0)
    else:
        radius = rng.uniform(86.0, 132.0)
        mouth_half_angle = rng.uniform(28.0, 34.0)

    clearance = radius + 34.0
    center = [
        round(rng.uniform(clearance, WIDTH - clearance), 4),
        round(rng.uniform(clearance, HEIGHT - clearance), 4),
    ]
    return {
        "direction": direction,
        "center": center,
        "radius": round(radius, 4),
        "mouth_half_angle_deg": round(mouth_half_angle, 4),
        "fill_color": list(rng.choice(FILL_PALETTE)),
        "background": list(rng.choice(BACKGROUND_PALETTE)),
        "outline_color": list(rng.choice(OUTLINE_PALETTE)),
        "outline_width": rng.choice((4, 5, 6)),
        "eye_scale": round(rng.uniform(0.075, 0.105), 5),
        "eye_offset": round(rng.uniform(0.18, 0.27), 5),
        "width": WIDTH,
        "height": HEIGHT,
        "antialias": True,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible marks; the mouth gap is cut out of the body disk."""

    width = int(scene["width"])
    height = int(scene["height"])
    supersample = SUPERSAMPLE if bool(scene["antialias"]) else 1
    image = Image.new("RGB", (width * supersample, height * supersample), _rgb(scene["background"]))
    draw = ImageDraw.Draw(image)
    cx, cy = (float(value) for value in scene["center"])
    radius = float(scene["radius"])
    half = float(scene["mouth_half_angle_deg"])
    angle = _ANGLE_DEG[str(scene["direction"])]
    fill = _rgb(scene["fill_color"])
    background = _rgb(scene["background"])
    outline = _rgb(scene["outline_color"])
    outline_width = max(1, int(scene["outline_width"])) * supersample
    body_box = _box(cx, cy, radius, supersample)

    # Draw the circular body and its neutral outline first.
    draw.ellipse(body_box, fill=fill, outline=outline, width=outline_width)

    # Remove one wedge.  Its straight radial boundaries are redrawn as a
    # neutral outline so the opening remains explicit at the raster boundary.
    draw.pieslice(
        body_box,
        start=angle - half,
        end=angle + half,
        fill=background,
    )
    for boundary_angle in (angle - half, angle + half):
        draw.line(
            [(int(round(cx * supersample)), int(round(cy * supersample))),
             _point(cx, cy, radius, boundary_angle, supersample)],
            fill=outline,
            width=outline_width,
        )

    # A small eye makes the glyph readable without adding a second saturated
    # component.  It rotates with the visible facing direction but is ignored
    # by the independent inverse, which uses only the colored silhouette.
    facing = math.radians(angle)
    ux, uy = math.cos(facing), math.sin(facing)
    px, py = -uy, ux
    eye_distance = float(scene["eye_offset"]) * radius
    eye_lateral = 0.22 * radius
    eye_cx = cx + ux * eye_distance + px * (-eye_lateral)
    eye_cy = cy + uy * eye_distance + py * (-eye_lateral)
    eye_radius = float(scene["eye_scale"]) * radius
    draw.ellipse(
        _box(eye_cx, eye_cy, eye_radius, supersample),
        fill=EYE_WHITE,
        outline=outline,
        width=max(1, 2 * supersample),
    )
    pupil_radius = eye_radius * 0.38
    pupil_cx = eye_cx + ux * eye_radius * 0.28
    pupil_cy = eye_cy + uy * eye_radius * 0.28
    draw.ellipse(
        _box(pupil_cx, pupil_cy, pupil_radius, supersample),
        fill=EYE_DARK,
    )

    if supersample > 1:
        image = image.resize((width, height), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the visible cardinal direction encoded by the mouth opening."""

    return str(scene["direction"])


def margin(scene: dict[str, Any]) -> float:
    """Radial mouth-edge clearance, a continuous proxy for visible evidence."""

    radius = float(scene["radius"])
    half = float(scene["mouth_half_angle_deg"])
    return round(min(radius * math.sin(math.radians(half)), radius - 24.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Flag the small/narrow subset whose visible mouth edge is near the floor."""

    return margin(scene) < 24.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared; every label field changes visible pixels."""

    return []
