"""Image-only inverse arm for Nesting Rank Reversal; imports no world state."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the raster no longer exposes the declared robust topology."""


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    remaining = {tuple(int(v) for v in point) for point in np.argwhere(mask)}
    result: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        component = [start]
        while stack:
            y, x = stack.pop()
            for neighbor in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                ny, nx = neighbor
                if 0 <= ny < height and 0 <= nx < width and neighbor in remaining:
                    remaining.remove(neighbor)
                    stack.append(neighbor)
                    component.append(neighbor)
        result.append(component)
    return result


def _centroid(component: list[tuple[int, int]]) -> tuple[float, float]:
    return (
        sum(point[1] for point in component) / len(component),
        sum(point[0] for point in component) / len(component),
    )


def _marker_components(mask: np.ndarray, name: str) -> list[tuple[float, float]]:
    components = _components(mask)
    if len(components) != 2:
        raise OracleAbstention(f"expected two {name} dots, found {len(components)}")
    centers: list[tuple[float, float]] = []
    for component in components:
        if not 80 <= len(component) <= 120:
            raise OracleAbstention(f"{name} marker has corrupted area {len(component)}")
        ys = [p[0] for p in component]
        xs = [p[1] for p in component]
        if max(max(xs) - min(xs), max(ys) - min(ys)) > 11:
            raise OracleAbstention(f"{name} marker is not compact")
        centers.append(_centroid(component))
    return centers


def _encloses(component: list[tuple[int, int]], point: tuple[float, float]) -> bool:
    ys = [p[0] for p in component]
    xs = [p[1] for p in component]
    y0, y1 = min(ys) - 1, max(ys) + 2
    x0, x1 = min(xs) - 1, max(xs) + 2
    if y0 < 0 or x0 < 0:
        raise OracleAbstention("navy loop touches image border")
    px, py = int(round(point[0])), int(round(point[1]))
    if not (x0 <= px < x1 and y0 <= py < y1):
        return False
    local = np.zeros((y1 - y0, x1 - x0), dtype=bool)
    for y, x in component:
        local[y - y0, x - x0] = True
    q: list[tuple[int, int]] = []
    seen = np.zeros_like(local)
    height, width = local.shape
    for x in range(width):
        if not local[0, x]:
            q.append((0, x)); seen[0, x] = True
        if not local[height - 1, x] and not seen[height - 1, x]:
            q.append((height - 1, x)); seen[height - 1, x] = True
    for y in range(height):
        if not local[y, 0] and not seen[y, 0]:
            q.append((y, 0)); seen[y, 0] = True
        if not local[y, width - 1] and not seen[y, width - 1]:
            q.append((y, width - 1)); seen[y, width - 1] = True
    for y, x in q:
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and not local[ny, nx] and not seen[ny, nx]:
                seen[ny, nx] = True
                q.append((ny, nx))
    ly, lx = py - y0, px - x0
    if local[ly, lx]:
        raise OracleAbstention("a colored marker touches navy ink")
    return not bool(seen[ly, lx])


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    dark = np.max(array, axis=2) < 155
    orange = (red >= 190) & (green >= 70) & (green <= 180) & (blue <= 105) & (red - green >= 45)
    purple = (red >= 90) & (red <= 190) & (green <= 125) & (blue >= 130) & (blue - green >= 25)
    orange_centers = _marker_components(orange, "orange")
    purple_centers = _marker_components(purple, "purple")
    rings = [component for component in _components(dark) if len(component) >= 300]
    if len(rings) != 8:
        raise OracleAbstention(f"expected eight closed navy loops, found {len(rings)}")
    for ring in rings:
        ys = [p[0] for p in ring]
        xs = [p[1] for p in ring]
        bw, bh = max(xs) - min(xs) + 1, max(ys) - min(ys) + 1
        if min(bw, bh) < 26 or max(bw, bh) > 86 or abs(bw - bh) > 2:
            raise OracleAbstention("a navy component is not a robust closed loop")
    markers = [("orange", center) for center in orange_centers] + [
        ("purple", center) for center in purple_centers
    ]
    depths = {
        (color, center): sum(_encloses(ring, center) for ring in rings)
        for color, center in markers
    }
    # Pair orange and purple markers by the minimum-distance perfect matching.
    matchings: list[tuple[float, tuple[int, int]]] = []
    for perm in itertools.permutations(range(2)):
        total = sum(
            ((orange_centers[i][0] - purple_centers[perm[i]][0]) ** 2
             + (orange_centers[i][1] - purple_centers[perm[i]][1]) ** 2) ** 0.5
            for i in range(2)
        )
        matchings.append((total, perm))
    matchings.sort(key=lambda item: item[0])
    if matchings[1][0] - matchings[0][0] < 100:
        raise OracleAbstention("emblem pairing is spatially ambiguous")
    signs: list[int] = []
    for i, j in enumerate(matchings[0][1]):
        orange_depth = depths[("orange", orange_centers[i])]
        purple_depth = depths[("purple", purple_centers[j])]
        if sorted((orange_depth, purple_depth)) != [1, 3]:
            raise OracleAbstention(
                f"unsupported or corrupted depth pair {orange_depth}, {purple_depth}"
            )
        signs.append(1 if orange_depth > purple_depth else -1)
    return "yes" if signs[0] != signs[1] else "no"
