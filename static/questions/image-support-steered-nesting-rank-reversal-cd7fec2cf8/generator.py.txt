"""Deterministic renderer and independent analytic arm for Nesting Rank Reversal."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
LOOP_SIZES = (84, 56, 28)
LOOP_WIDTH = 4
MARKER_RADIUS = 5
NEST_SEPARATION = 100
MIN_CLEARANCE = 3.0

INKS = ("#173f5f", "#263b66", "#194b58", "#333f63")
ORANGES = ("#ef7d24", "#e87925", "#f28a2e")
PURPLES = ("#8756b3", "#9952ae", "#7658b5")
PALES = ("#d8dee2", "#d3dce0", "#e1dadd")


def _rect_loop(draw: ImageDraw.ImageDraw, center: tuple[int, int], size: int, ink: str) -> None:
    cx, cy = center
    x0 = cx - size // 2
    y0 = cy - size // 2
    draw.rectangle(
        (x0, y0, x0 + size - 1, y0 + size - 1),
        outline=ink,
        width=LOOP_WIDTH,
    )


def _sample_decorations(
    rng: random.Random, marker_centers: list[tuple[int, int]]
) -> list[list[int]]:
    decorations: list[list[int]] = []
    attempts = 0
    while len(decorations) < 8 and attempts < 800:
        attempts += 1
        x = rng.randrange(24, CANVAS - 24)
        y = rng.randrange(24, CANVAS - 24)
        if min(math.dist((x, y), center) for center in marker_centers) < 78:
            continue
        if any(math.dist((x, y), (old_x, old_y)) < 25 for old_x, old_y, _ in decorations):
            continue
        decorations.append([x, y, rng.randrange(2)])
    if len(decorations) != 8:
        raise RuntimeError("could not place separated decorations")
    return decorations


def _emblem(
    rng: random.Random,
    center: tuple[int, int],
    split_axis: str,
) -> dict:
    cx, cy = center
    half = NEST_SEPARATION // 2
    if split_axis == "vertical":
        slots = [(cx, cy - half), (cx, cy + half)]
    elif split_axis == "horizontal":
        slots = [(cx - half, cy), (cx + half, cy)]
    else:
        raise ValueError(f"unknown split axis {split_axis!r}")
    if rng.randrange(2):
        slots.reverse()
    return {
        "markers": [
            {"color": "orange", "center": list(slots[0])},
            {"color": "purple", "center": list(slots[1])},
        ]
    }


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x6E57A91)
    # The four ordered sign pairs balance yes/no, color identity, and panel identity.
    sign_pairs = ((True, True), (True, False), (False, True), (False, False))
    left_or_top, right_or_bottom = sign_pairs[int(seed) % len(sign_pairs)]
    layout_axis = rng.choice(("horizontal", "vertical"))
    if layout_axis == "horizontal":
        cross = rng.choice((246, 256, 266))
        panel_centers = [(58, cross), (454, cross)]
        split_axis = "vertical"
    else:
        cross = rng.choice((246, 256, 266))
        panel_centers = [(cross, 58), (cross, 454)]
        split_axis = "horizontal"
    emblems = [
        _emblem(rng, panel_centers[0], split_axis),
        _emblem(rng, panel_centers[1], split_axis),
    ]
    marker_centers = [
        tuple(marker["center"])
        for emblem in emblems
        for marker in emblem["markers"]
    ]
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": "#ffffff",
        "ink": rng.choice(INKS),
        "orange": rng.choice(ORANGES),
        "purple": rng.choice(PURPLES),
        "decoration_color": rng.choice(PALES),
        "emblems": emblems,
        "depth_signs": [1 if left_or_top else -1, 1 if right_or_bottom else -1],
        "decorations": _sample_decorations(rng, marker_centers),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB", (int(scene["width"]), int(scene["height"])), str(scene["background"])
    )
    draw = ImageDraw.Draw(image)
    pale = str(scene["decoration_color"])
    for x, y, kind in scene["decorations"]:
        x, y, kind = int(x), int(y), int(kind)
        if kind == 0:
            draw.rectangle((x - 5, y - 2, x + 4, y + 1), fill=pale)
        else:
            draw.ellipse((x - 3, y - 3, x + 3, y + 3), outline=pale, width=2)
    ink = str(scene["ink"])
    colors = {"orange": str(scene["orange"]), "purple": str(scene["purple"])}
    depth_signs = [int(value) for value in scene["depth_signs"]]
    if len(depth_signs) != 2 or any(value not in {-1, 1} for value in depth_signs):
        raise ValueError("depth_signs must contain two values from {-1, 1}")
    for emblem, sign in zip(scene["emblems"], depth_signs, strict=True):
        for marker in emblem["markers"]:
            center = tuple(int(value) for value in marker["center"])
            color = str(marker["color"])
            if color not in colors:
                raise ValueError("each emblem needs one orange and one purple marker")
            depth = 3 if (color == "orange") == (sign > 0) else 1
            for size in LOOP_SIZES[:depth]:
                _rect_loop(draw, center, size, ink)
            cx, cy = center
            draw.ellipse(
                (cx - MARKER_RADIUS, cy - MARKER_RADIUS, cx + MARKER_RADIUS, cy + MARKER_RADIUS),
                fill=colors[color],
            )
    return image


def analytic_gold(scene: dict) -> str:
    signs = [int(value) for value in scene["depth_signs"]]
    if len(signs) != 2 or any(value not in {-1, 1} for value in signs):
        raise ValueError("exactly two nonzero depth signs are required")
    return "yes" if signs[0] != signs[1] else "no"


def margin(scene: dict) -> float:
    centers = [
        tuple(int(value) for value in marker["center"])
        for emblem in scene["emblems"]
        for marker in emblem["markers"]
    ]
    if len(centers) != 4:
        return 0.0
    half_outer = LOOP_SIZES[0] / 2.0
    border = min(
        min(cx - half_outer, int(scene["width"]) - cx - half_outer,
            cy - half_outer, int(scene["height"]) - cy - half_outer)
        for cx, cy in centers
    )
    pairwise = sorted(
        math.dist(centers[i], centers[j]) - LOOP_SIZES[0]
        for i in range(4) for j in range(i + 1, 4)
    )[0]
    loop_gap = min(
        (LOOP_SIZES[i] - LOOP_SIZES[i + 1]) / 2.0 - LOOP_WIDTH
        for i in range(len(LOOP_SIZES) - 1)
    )
    marker_gap = LOOP_SIZES[-1] / 2.0 - LOOP_WIDTH - MARKER_RADIUS
    return float(min(LOOP_WIDTH, loop_gap, marker_gap, pairwise, border))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < MIN_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["emblems"] = list(reversed(twin["emblems"]))
    twin["depth_signs"] = list(reversed(twin["depth_signs"]))
    return [("reverse_disjoint_emblem_draw_order", twin)]
