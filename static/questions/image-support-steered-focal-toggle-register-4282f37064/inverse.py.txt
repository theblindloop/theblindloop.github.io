"""Independent pixel-only inverse arm for focal_toggle_register.

Receives ONLY the final PNG. It finds the one connected coloured register disk
(blue = OFF, orange = ON) anywhere on the canvas and reads its state from its
colour. It then segments every red dot and counts only those red dots whose
centroid lies within the small on-region radius around the register centre --
these are the real on-cluster toggles. Red dots far away are decoys and are
ignored. The final state is the initial state flipped once per real toggle.

The oracle raises (refuses to decide) when there is not exactly one substantial
coloured register disk, or when any red dot lies in the ambiguous band between
the toggle region and the decoy clear distance. This makes the decisive pixel
support exactly the compact register cluster: destroying the cluster removes the
register or a toggle (changing or preventing the answer), while destroying empty
canvas or distant decoys changes nothing.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

MIN_REG = 400
MIN_RED = 20
REGION_R = 28.0
MIN_DECOY_CLEAR = 52.0
AMBIGUOUS_LOW = REGION_R + 4.0
AMBIGUOUS_HIGH = MIN_DECOY_CLEAR - 6.0


def _colors(arr):
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    red = (r > 150) & (g < 110) & (b < 110)
    blue = (r < 80) & (g < 160) & (b > 140)
    orange = (r > 170) & (g > 90) & (g < 210) & (b < 110)
    return red, blue, orange


def _components(mask):
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    comps = []
    ys_idx, xs_idx = np.nonzero(mask)
    for y, x in zip(ys_idx.tolist(), xs_idx.tolist()):
        if visited[y, x]:
            continue
        stack = [(y, x)]
        visited[y, x] = True
        ys = []
        xs = []
        while stack:
            cy, cx = stack.pop()
            ys.append(cy)
            xs.append(cx)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                    visited[ny, nx] = True
                    stack.append((ny, nx))
        comps.append((np.asarray(ys, dtype=np.int64), np.asarray(xs, dtype=np.int64)))
    return comps


def _centroid(comp):
    ys, xs = comp
    return float(np.mean(ys)), float(np.mean(xs))


def _register_info(arr):
    red, blue, orange = _colors(arr)
    register_mask = blue | orange
    comps = [c for c in _components(register_mask) if len(c[0]) >= MIN_REG]
    if len(comps) != 1:
        raise ValueError(f"expected exactly one register disk, found {len(comps)}")
    ys, xs = comps[0]
    n_blue = int((blue[ys, xs]).sum())
    n_orange = int((orange[ys, xs]).sum())
    if n_blue == n_orange:
        raise ValueError("register colour is ambiguous")
    start_state = 1 if n_orange > n_blue else 0
    return _centroid(comps[0]), start_state


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    red, _, _ = _colors(arr)

    (cy, cx), start_state = _register_info(arr)

    red_comps = [c for c in _components(red) if len(c[0]) >= MIN_RED]
    for ys, xs in red_comps:
        rcy, rcx = _centroid((ys, xs))
        dist = float(np.hypot(rcy - cy, rcx - cx))
        if AMBIGUOUS_LOW <= dist <= AMBIGUOUS_HIGH:
            raise ValueError(f"red dot lies in the ambiguous band at distance {dist:.1f}")
    toggles = sum(
        1
        for ys, xs in red_comps
        if float(np.hypot(*np.subtract(_centroid((ys, xs)), (cy, cx)))) < REGION_R
    )

    final = start_state ^ (toggles % 2)
    return "yes" if final else "no"
