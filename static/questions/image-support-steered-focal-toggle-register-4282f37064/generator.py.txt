"""Focal toggle-register world renderer.

A single compact "state register" -- one filled coloured disk -- sits at a
randomised position on an otherwise neutral canvas. The register disk declares
the initial state (blue = OFF, orange = ON). A small cluster of red toggle dots
is placed tightly around the disk, each resting on its circumference so that the
disk and its toggles form ONE connected, area-like, localised cue. Each red
toggle flips the tracked binary state; the answer is the FINAL state after
applying every toggle in the cluster.

Remote red decoy dots are scattered far across the rest of the canvas. They use
the identical red colour and identical size as the real toggles, so total red
colour, total red count, and total ink are all nondiagnostic. The decisive pixel
support is entirely the small connected register cluster; the surrounding canvas
is deliberately inert.

The analytic gold is start_state XOR (parity of the toggle cluster count).
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "focal_toggle_register-0.1.0"

CANVAS = 512
REGISTER_R = 18.0
TOGGLE_R = 5.0
BLUE = (0, 90, 200)
ORANGE = (240, 130, 30)
RED = (215, 45, 45)

REGION_R = 28.0          # oracle on-toggle radius around the register centre
MIN_DECOY_CLEAR = 52.0   # nearest decoy must sit well beyond the toggle region
EDGE_MARGIN = 88.0       # keep the whole cluster clear of the canvas edge
MIN_TOGGLE_GAP = 0.75    # min angular gap (rad) so adjacent toggles stay apart


def _start_color(state: int):
    return ORANGE if state else BLUE


def _angular_gap(a: float, b: float) -> float:
    d = abs(a - b) % (2.0 * math.pi)
    return min(d, 2.0 * math.pi - d)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start_state = rng.choice((0, 1))
    cx = float(rng.randint(int(EDGE_MARGIN), int(CANVAS - EDGE_MARGIN)))
    cy = float(rng.randint(int(EDGE_MARGIN), int(CANVAS - EDGE_MARGIN)))
    target_toggles = rng.randint(0, 4)
    angles = []
    tries = 0
    while len(angles) < target_toggles and tries < 300:
        tries += 1
        a = rng.uniform(0.0, 2.0 * math.pi)
        if all(_angular_gap(a, b) >= MIN_TOGGLE_GAP for b in angles):
            angles.append(a)
    angles.sort()
    toggle_points = [
        (
            round(cx + REGISTER_R * math.cos(a), 1),
            round(cy + REGISTER_R * math.sin(a), 1),
        )
        for a in angles
    ]

    decoy_points = []
    nd = rng.randint(1, 4)
    tries = 0
    while len(decoy_points) < nd and tries < 600:
        tries += 1
        tx = float(rng.randint(12, CANVAS - 12))
        ty = float(rng.randint(12, CANVAS - 12))
        if math.hypot(tx - cx, ty - cy) >= MIN_DECOY_CLEAR:
            decoy_points.append((round(tx, 1), round(ty, 1)))

    return {
        "start_state": start_state,
        "register_center": [cx, cy],
        "toggle_points": toggle_points,
        "decoy_points": decoy_points,
    }


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (CANVAS, CANVAS), "white")
    draw = ImageDraw.Draw(img)
    cx, cy = scene["register_center"]
    draw.ellipse(
        (cx - REGISTER_R, cy - REGISTER_R, cx + REGISTER_R, cy + REGISTER_R),
        fill=_start_color(scene["start_state"]),
    )
    for tx, ty in scene["toggle_points"] + scene["decoy_points"]:
        draw.ellipse(
            (tx - TOGGLE_R, ty - TOGGLE_R, tx + TOGGLE_R, ty + TOGGLE_R),
            fill=RED,
        )
    return img


def analytic_gold(scene: dict) -> str:
    n_toggles = len(scene["toggle_points"])
    final = int(scene["start_state"]) ^ (n_toggles % 2)
    return "yes" if final else "no"


def margin(scene: dict) -> float:
    """Nearest decoy distance beyond the on-toggle region radius.

    A large positive value means every red decoy sits safely beyond the radius at
    which the oracle would count it as an on-cluster toggle, so no decoy can be
    mistaken for a real toggle.
    """
    cx, cy = scene["register_center"]
    best = float("inf")
    for tx, ty in scene["decoy_points"]:
        best = min(best, math.hypot(tx - cx, ty - cy))
    return 60.0 if best == float("inf") else best - REGION_R


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict):
    # not_applicable: there is no pixel-identical latent transform. The register
    # colour and every toggle in the cluster are load-bearing and rendered; every
    # scene field that influences the gold is rendered, and no nontrivial
    # transform leaves the image unchanged while preserving the answer.
    return []
