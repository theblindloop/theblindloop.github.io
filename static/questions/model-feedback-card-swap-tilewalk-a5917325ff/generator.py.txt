"""Deterministic renderer and analytic arm for a card-swapping cross-coupled walk."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = (660, 660)
N, CELL, X0, Y0, STEPS = 5, 100, 80, 80, 8
DIRS = ((0, -1), (1, 0), (0, 1), (-1, 0))
TOKEN_COLORS = ((220, 45, 55), (25, 105, 220))
GOAL_COLORS = {"amber": (235, 150, 20), "cyan": (0, 165, 185), "violet": (125, 70, 195)}


def _inside(cell):
    return 0 <= cell[0] < N and 0 <= cell[1] < N


def _step(cell, direction):
    dx, dy = DIRS[direction]
    return cell[0] + dx, cell[1] + dy


def _trace(instructions, red_start, blue_start, rule="correct"):
    board = [[pair[:] for pair in row] for row in instructions]
    card_ids = [[(x, y) for x in range(N)] for y in range(N)]
    red, blue = tuple(red_start), tuple(blue_start)
    states, reds = set(), [red]
    moved_card_read = False
    for _ in range(STEPS):
        old_red, old_blue = red, blue
        if old_red == old_blue:
            raise ValueError("tokens share a tile")
        moved_card_read = moved_card_read or (
            card_ids[old_blue[1]][old_blue[0]] != old_blue
            or card_ids[old_red[1]][old_red[0]] != old_red
        )
        red_dir = board[old_blue[1]][old_blue[0]][0]
        blue_dir = board[old_red[1]][old_red[0]][1]
        red = _step(old_red, red_dir)
        blue = _step(old_blue, blue_dir)
        if not _inside(red) or not _inside(blue) or red == blue:
            raise ValueError("invalid token landing")
        if rule == "correct":
            a, b = old_red, old_blue
            board[a[1]][a[0]], board[b[1]][b[0]] = board[b[1]][b[0]], board[a[1]][a[0]]
            card_ids[a[1]][a[0]], card_ids[b[1]][b[0]] = card_ids[b[1]][b[0]], card_ids[a[1]][a[0]]
        elif rule == "landing":
            a, b = red, blue
            board[a[1]][a[0]], board[b[1]][b[0]] = board[b[1]][b[0]], board[a[1]][a[0]]
            card_ids[a[1]][a[0]], card_ids[b[1]][b[0]] = card_ids[b[1]][b[0]], card_ids[a[1]][a[0]]
        elif rule != "static":
            raise ValueError("unknown trace rule")
        signature = (red, blue, tuple(tuple(tuple(pair) for pair in row) for row in board))
        if signature in states:
            raise ValueError("complete state repeats")
        states.add(signature)
        reds.append(red)
    return red, blue, moved_card_read, reds


def _construct(rng):
    for _attempt in range(300000):
        instructions = [[[rng.randrange(4), rng.randrange(4)] for _ in range(N)] for _ in range(N)]
        red_start = (rng.randrange(1, N - 1), rng.randrange(1, N - 1))
        blue_start = (rng.randrange(1, N - 1), rng.randrange(1, N - 1))
        if red_start == blue_start:
            continue
        try:
            correct = _trace(instructions, red_start, blue_start, "correct")
            static = _trace(instructions, red_start, blue_start, "static")
            landing = _trace(instructions, red_start, blue_start, "landing")
        except ValueError:
            continue
        endpoints = [correct[0], static[0], landing[0]]
        if not correct[2] or len(set(endpoints)) != 3:
            continue
        if any(cell in correct[3][:-1] for cell in endpoints):
            continue
        return instructions, (red_start, blue_start), endpoints
    raise RuntimeError("could not construct card-swap walk")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5CA4D71E)
    instructions, (red_start, blue_start), endpoints = _construct(rng)
    answer = ("amber", "cyan", "violet")[int(seed) % 3]
    others = [name for name in GOAL_COLORS if name != answer]
    rng.shuffle(others)
    goals = {answer: list(endpoints[0]), others[0]: list(endpoints[1]), others[1]: list(endpoints[2])}
    return {"instructions": instructions, "red_start": list(red_start),
            "blue_start": list(blue_start), "goals": goals}


def _simulate(scene, rule="correct"):
    return _trace(scene["instructions"], scene["red_start"], scene["blue_start"], rule)


def analytic_gold(scene):
    endpoint = _simulate(scene)[0]
    matches = [name for name, cell in scene["goals"].items() if tuple(cell) == endpoint]
    if len(matches) != 1:
        raise ValueError("red endpoint does not have exactly one goal")
    return matches[0]


def margin(scene):
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene):
    try:
        ins = scene["instructions"]
        if len(ins) != N or any(len(row) != N for row in ins):
            return True
        if any(len(pair) != 2 or any(d not in range(4) for d in pair) for row in ins for pair in row):
            return True
        starts = tuple(scene["red_start"]), tuple(scene["blue_start"])
        if any(not _inside(start) for start in starts) or starts[0] == starts[1]:
            return True
        correct = _simulate(scene, "correct")
        static = _simulate(scene, "static")
        landing = _simulate(scene, "landing")
        endpoints = [correct[0], static[0], landing[0]]
        if not correct[2] or len(set(endpoints)) != 3:
            return True
        if any(cell in correct[3][:-1] for cell in endpoints):
            return True
        goal_cells = [tuple(cell) for cell in scene["goals"].values()]
        return set(scene["goals"]) != set(GOAL_COLORS) or set(goal_cells) != set(endpoints)
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene):
    return []


def _arrow(draw, center, direction, color):
    cx, cy = center
    if direction == 0:
        draw.line((cx, cy + 12, cx, cy - 5), fill=color, width=8)
        draw.polygon(((cx, cy - 17), (cx - 10, cy - 3), (cx + 10, cy - 3)), fill=color)
    elif direction == 1:
        draw.line((cx - 12, cy, cx + 5, cy), fill=color, width=8)
        draw.polygon(((cx + 17, cy), (cx + 3, cy - 10), (cx + 3, cy + 10)), fill=color)
    elif direction == 2:
        draw.line((cx, cy - 12, cx, cy + 5), fill=color, width=8)
        draw.polygon(((cx, cy + 17), (cx - 10, cy + 3), (cx + 10, cy + 3)), fill=color)
    else:
        draw.line((cx + 12, cy, cx - 5, cy), fill=color, width=8)
        draw.polygon(((cx - 17, cy), (cx - 3, cy - 10), (cx - 3, cy + 10)), fill=color)


def render(scene):
    image = Image.new("RGB", CANVAS, (248, 248, 246))
    draw = ImageDraw.Draw(image)
    for y in range(N):
        for x in range(N):
            fill = (255, 255, 255) if (x + y) % 2 == 0 else (243, 246, 248)
            draw.rectangle((X0 + x * CELL, Y0 + y * CELL,
            X0 + (x + 1) * CELL, Y0 + (y + 1) * CELL), fill=fill)
            draw.rounded_rectangle((X0 + x * CELL + 11, Y0 + y * CELL + 10,
                                    X0 + (x + 1) * CELL - 11, Y0 + (y + 1) * CELL - 10),
                                   radius=8, fill=(252, 252, 250), outline=(165, 172, 178), width=3)
    for name, cell in scene["goals"].items():
        x, y = cell
        draw.rectangle((X0 + x * CELL + 5, Y0 + y * CELL + 5,
                        X0 + (x + 1) * CELL - 5, Y0 + (y + 1) * CELL - 5),
                       outline=GOAL_COLORS[name], width=8)
    for i in range(N + 1):
        draw.line((X0 + i * CELL, Y0, X0 + i * CELL, Y0 + N * CELL), fill=(45, 50, 55), width=3)
        draw.line((X0, Y0 + i * CELL, X0 + N * CELL, Y0 + i * CELL), fill=(45, 50, 55), width=3)
    for y in range(N):
        for x in range(N):
            cx, cy = X0 + x * CELL + CELL // 2, Y0 + y * CELL + CELL // 2
            _arrow(draw, (cx, cy - 22), scene["instructions"][y][x][0], TOKEN_COLORS[0])
            _arrow(draw, (cx, cy + 22), scene["instructions"][y][x][1], TOKEN_COLORS[1])
    rx, ry = scene["red_start"]
    bx, by = scene["blue_start"]
    draw.ellipse((X0 + rx * CELL + 7, Y0 + ry * CELL + 7,
                  X0 + rx * CELL + 29, Y0 + ry * CELL + 29), fill=TOKEN_COLORS[0], outline="black", width=3)
    draw.ellipse((X0 + bx * CELL + 71, Y0 + by * CELL + 7,
                  X0 + bx * CELL + 93, Y0 + by * CELL + 29), fill=TOKEN_COLORS[1], outline="black", width=3)
    return image
