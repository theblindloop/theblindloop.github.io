"""Independent image-only reconstruction of the colored cut vertex."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        group: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            group.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def _centroid(group: list[tuple[int, int]]) -> tuple[float, float]:
    return (sum(x for _, x in group) / len(group), sum(y for y, _ in group) / len(group))


def _graph_components(graph: list[set[int]], removed: int | None = None) -> int:
    unseen = set(range(len(graph)))
    if removed is not None:
        unseen.discard(removed)
    count = 0
    while unseen:
        count += 1
        queue = deque([unseen.pop()])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor != removed and neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
    return count


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = (array[:, :, 0] > 160) & (array[:, :, 1] < 100) & (array[:, :, 2] < 110)
    blue = (array[:, :, 2] > 150) & (array[:, :, 0] < 100) & (array[:, :, 1] < 150)
    dark = np.max(array, axis=2) < 75
    gray = (
        (array[:, :, 0] >= 115) & (array[:, :, 0] <= 180)
        & (np.max(array, axis=2) - np.min(array, axis=2) <= 28)
    )

    red_groups = [g for g in _components(red) if 120 <= len(g) <= 260]
    blue_groups = [g for g in _components(blue) if 120 <= len(g) <= 260]
    dark_groups = [g for g in _components(dark) if 120 <= len(g) <= 260]
    if len(red_groups) != 1 or len(blue_groups) != 1 or len(dark_groups) != 9:
        raise ValueError("expected one red, one blue, and nine black dot components")

    points = [_centroid(red_groups[0]), _centroid(blue_groups[0])]
    points.extend(_centroid(group) for group in dark_groups)
    candidate_indices = {"red": 0, "blue": 1}

    bars = [g for g in _components(gray) if len(g) >= 300]
    if len(bars) != 1:
        raise ValueError(f"expected one gray scale bar, found {len(bars)}")
    threshold = max(x for _, x in bars[0]) - min(x for _, x in bars[0]) + 1

    graph = [set() for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if math.dist(points[i], points[j]) < threshold:
                graph[i].add(j)
                graph[j].add(i)
    base = _graph_components(graph)
    if base != 1 or len(graph[0]) != len(graph[1]):
        raise ValueError("pixel graph is disconnected or colored degrees differ")
    scores = {color: _graph_components(graph, index) - base for color, index in candidate_indices.items()}
    winners = [color for color, score in scores.items() if score > 0]
    if len(winners) != 1 or sorted(scores.values()) != [0, 1]:
        raise ValueError(f"expected one colored cut vertex, got {scores}")
    return winners[0]
