"""Independent fixed-layout pixel inverse for the turn-and-flip walk."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 900, 740
N, CELL = 5, 88
GRID_X, GRID_Y = 230, 154
BACKGROUND = (247, 246, 241)
GOLD = (242, 188, 54)
BLUE = (61, 130, 194)
RED = (207, 52, 62)
DIRS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _near(pixel: np.ndarray, color: tuple[int, int, int], tolerance: int = 5) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (HEIGHT, WIDTH, 3) or not _near(arr[725, 880], BACKGROUND):
        return "abstain"

    tiles: list[list[int]] = []
    start_hits: list[tuple[int, int]] = []
    for row in range(N):
        values = []
        for col in range(N):
            x0 = GRID_X + col * CELL
            y0 = GRID_Y + row * CELL
            pixel = arr[y0 + 12, x0 + 12]
            if _near(pixel, GOLD):
                values.append(0)
            elif _near(pixel, BLUE):
                values.append(1)
            else:
                return "abstain"
            patch = arr[y0 + 8:y0 + CELL - 8, x0 + 8:x0 + CELL - 8]
            red = np.all(np.abs(patch.astype(np.int16) - np.array(RED)) <= 5, axis=2)
            if int(red.sum()) > 100:
                start_hits.append((row, col))
        tiles.append(values)
    if len(start_hits) != 1:
        return "abstain"
    row, col = start_hits[0]
    cx = GRID_X + col * CELL + CELL // 2
    cy = GRID_Y + row * CELL + CELL // 2
    red_y, red_x = np.nonzero(np.all(np.abs(arr.astype(np.int16) - np.array(RED)) <= 5, axis=2))
    inside = (red_x >= GRID_X) & (red_x < GRID_X + N * CELL) & (red_y >= GRID_Y) & (red_y < GRID_Y + N * CELL)
    red_x = red_x[inside]
    red_y = red_y[inside]
    if len(red_x) < 100:
        return "abstain"
    extents = (cy - int(red_y.min()), int(red_x.max()) - cx, int(red_y.max()) - cy, cx - int(red_x.min()))
    best = max(extents)
    if extents.count(best) != 1 or best < 20:
        return "abstain"
    heading = extents.index(best)

    board = [line[:] for line in tiles]
    for _ in range(100):
        heading = (heading + (1 if board[row][col] == 0 else -1)) % 4
        board[row][col] ^= 1
        dr, dc = DIRS[heading]
        row += dr
        col += dc
        if row < 0:
            return "gate-N"
        if col >= N:
            return "gate-E"
        if row >= N:
            return "gate-S"
        if col < 0:
            return "gate-W"
    return "abstain"
