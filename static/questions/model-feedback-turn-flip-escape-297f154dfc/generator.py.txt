"""Sampler and renderer for a self-modifying turn-and-flip walk."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 740
N = 5
CELL = 88
GRID_X, GRID_Y = 230, 154
BACKGROUND = (247, 246, 241)
INK = (29, 37, 48)
GOLD = (242, 188, 54)
BLUE = (61, 130, 194)
RED = (207, 52, 62)
WHITE = (255, 255, 255)
DECISIONS = ("gate-N", "gate-E", "gate-S", "gate-W")
DIRS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = Path("/usr/share/fonts/truetype/dejavu") / name
    return ImageFont.truetype(str(path), size) if path.exists() else ImageFont.load_default()


def evolve(tiles: list[list[int]], start: list[int], direction: int, limit: int = 100) -> dict:
    board = [[int(value) for value in row] for row in tiles]
    if len(board) != N or any(len(row) != N for row in board):
        raise ValueError("five by five board required")
    if any(value not in (0, 1) for row in board for value in row):
        raise ValueError("tile states must be binary")
    row, col = map(int, start)
    heading = int(direction)
    if not (0 <= row < N and 0 <= col < N and 0 <= heading < 4):
        raise ValueError("invalid start")
    visits: dict[tuple[int, int], int] = {}
    trace: list[list[int]] = []
    for step in range(1, int(limit) + 1):
        visits[(row, col)] = visits.get((row, col), 0) + 1
        old = board[row][col]
        heading = (heading + (1 if old == 0 else -1)) % 4
        board[row][col] ^= 1
        trace.append([row, col, old, heading])
        dr, dc = DIRS[heading]
        row += dr
        col += dc
        if row < 0:
            gate = "gate-N"
        elif col >= N:
            gate = "gate-E"
        elif row >= N:
            gate = "gate-S"
        elif col < 0:
            gate = "gate-W"
        else:
            continue
        return {
            "decision": gate,
            "steps": step,
            "revisits": sum(count - 1 for count in visits.values()),
            "max_visits": max(visits.values()),
            "trace": trace,
        }
    return {"decision": "abstain", "steps": int(limit), "revisits": 0, "max_visits": 0, "trace": trace}


def _static_walk(tiles: list[list[int]], start: list[int], direction: int, limit: int = 100) -> str:
    row, col = map(int, start)
    heading = int(direction)
    for _ in range(limit):
        heading = (heading + (1 if int(tiles[row][col]) == 0 else -1)) % 4
        dr, dc = DIRS[heading]
        row += dr
        col += dc
        if row < 0:
            return "gate-N"
        if col >= N:
            return "gate-E"
        if row >= N:
            return "gate-S"
        if col < 0:
            return "gate-W"
    return "abstain"


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5455524E)
    target = DECISIONS[int(seed) % len(DECISIONS)]
    for _ in range(200000):
        tiles = [[rng.randrange(2) for _col in range(N)] for _row in range(N)]
        start = [rng.randrange(1, N - 1), rng.randrange(1, N - 1)]
        direction = rng.randrange(4)
        outcome = evolve(tiles, start, direction)
        if outcome["decision"] != target or not (12 <= outcome["steps"] <= 22):
            continue
        if outcome["revisits"] < 3 or outcome["max_visits"] < 2:
            continue
        if _static_walk(tiles, start, direction) == target:
            continue
        return {"tiles": tiles, "start": start, "direction": direction}
    raise RuntimeError("could not sample a balanced consequential revisit scene")


def _validate(scene: dict) -> tuple[list[list[int]], list[int], int]:
    tiles = [[int(value) for value in row] for row in scene["tiles"]]
    start = [int(value) for value in scene["start"]]
    direction = int(scene["direction"])
    evolve(tiles, start, direction)
    return tiles, start, direction


def analytic_gold(scene: dict) -> str:
    try:
        tiles, start, direction = _validate(scene)
        outcome = evolve(tiles, start, direction)
        return str(outcome["decision"])
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _arrow_points(cx: int, cy: int, direction: int, scale: int = 38) -> list[tuple[int, int]]:
    points = [(-0.45, -0.25), (0.05, -0.25), (0.05, -0.52), (0.78, 0), (0.05, 0.52), (0.05, 0.25), (-0.45, 0.25)]
    rotated = []
    for x, y in points:
        for _ in range((int(direction) + 3) % 4):
            x, y = -y, x
        rotated.append((round(cx + x * scale), round(cy + y * scale)))
    return rotated


def render(scene: dict) -> Image.Image:
    tiles, start, direction = _validate(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((WIDTH // 2, 24), "TURN · FLIP · MOVE", font=_font(29, True), fill=INK, anchor="ma")
    draw.text((WIDTH // 2, 66), "At each tile: turn by its color  →  flip the tile  →  move one cell", font=_font(19), fill=INK, anchor="ma")

    draw.rounded_rectangle((46, 145, 198, 286), radius=15, fill=WHITE, outline=INK, width=3)
    draw.text((122, 160), "TURN KEY", font=_font(18, True), fill=INK, anchor="ma")
    draw.rectangle((65, 193, 101, 229), fill=GOLD, outline=INK, width=2)
    draw.text((112, 211), "RIGHT", font=_font(17, True), fill=INK, anchor="lm")
    draw.rectangle((65, 241, 101, 277), fill=BLUE, outline=INK, width=2)
    draw.text((112, 259), "LEFT", font=_font(17, True), fill=INK, anchor="lm")

    draw.rounded_rectangle((700, 145, 854, 286), radius=15, fill=WHITE, outline=INK, width=3)
    draw.text((777, 160), "EACH VISIT", font=_font(18, True), fill=INK, anchor="ma")
    draw.text((777, 200), "1  TURN", font=_font(17, True), fill=INK, anchor="ma")
    draw.text((777, 233), "2  FLIP", font=_font(17, True), fill=INK, anchor="ma")
    draw.text((777, 266), "3  MOVE", font=_font(17, True), fill=INK, anchor="ma")

    for row in range(N):
        for col in range(N):
            x0 = GRID_X + col * CELL
            y0 = GRID_Y + row * CELL
            fill = GOLD if tiles[row][col] == 0 else BLUE
            draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), fill=fill, outline=INK, width=4)

    gate_specs = (
        ("N", (GRID_X + N * CELL // 2, GRID_Y - 39)),
        ("E", (GRID_X + N * CELL + 39, GRID_Y + N * CELL // 2)),
        ("S", (GRID_X + N * CELL // 2, GRID_Y + N * CELL + 39)),
        ("W", (GRID_X - 39, GRID_Y + N * CELL // 2)),
    )
    for label, (cx, cy) in gate_specs:
        draw.rounded_rectangle((cx - 28, cy - 25, cx + 28, cy + 25), radius=10, fill=WHITE, outline=INK, width=3)
        draw.text((cx, cy), label, font=_font(25, True), fill=INK, anchor="mm")

    sr, sc = start
    cx = GRID_X + sc * CELL + CELL // 2
    cy = GRID_Y + sr * CELL + CELL // 2
    arrow = _arrow_points(cx, cy, direction)
    draw.polygon(arrow, fill=RED)
    draw.line(arrow + [arrow[0]], fill=WHITE, width=4, joint="curve")
    draw.text((WIDTH // 2, 662), "Start at the red arrow. Stop at the first side crossed.", font=_font(20, True), fill=INK, anchor="ma")
    draw.text((WIDTH // 2, 700), "Which gate: N, E, S, or W?", font=_font(24, True), fill=RED, anchor="ma")
    return image
