"""Center-surround world: does the whole dot constellation enclose the canvas center?

A scattered constellation of equal-sized dark dots is drawn on a white square
canvas. The question concerns the whole set of marks together: do the dots form
an encirclement around the exact center point of the square canvas (so that the
center lies inside the convex hull of the dots), or are all the dots confined to
one side of that center point? This is a global Gestalt property of the entire
constellation -- whether it surrounds the central region -- not a property of any
single dot, its count, its ink, its colour, or its individual position.

The renderer owns latent-scene sampling, rasterization, and the structural
analytic gold. The independent pixel-only inverse arm (``oracle.py``) recovers
the same decision from the final PNG alone: it isolates every dark dot, takes
each dot's centroid, measures the largest empty angular gap of those centres
around the exact canvas centre, and decides whether that largest gap is smaller
than a half-turn (surround / yes) or larger (all on one side / no).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
CENTER = (256.0, 256.0)

DOT_RADIUS = 8
DOT_COLOR = (40, 48, 112)

# Number of dots drawn in each scene.
N_MIN = 10
N_MAX = 14

# Radial band around the canvas centre within which every dot centre is placed.
R_MIN = 150.0
R_MAX = 190.0

# For "no" (not surround) scenes every dot lies in one contiguous angular arc.
# The arc half-width is sampled in this range (degrees), so the largest empty
# angular gap is comfortably larger than a half-turn and the margin is positive.
ARC_MIN_DEG = 120.0
ARC_MAX_DEG = 150.0

# Deterministic rejection floor: keep every pair of dot centers at least this far
# apart (px) so the rasterised blobs never merge and the pixel oracle always
# recovers one clean centroid per dot.
MIN_PAIR_SEP = 22.0

# Angular jitter applied to the evenly-spaced surround angles / clustered angles.
JITTER_DEG = 7.0
RAD_JITTER = 12.0

# Quarantine band on the decision margin (radians). The margin is the distance
# of the largest angular gap from a half-turn; near a half-turn the surround
# boundary is not cleanly readable.
QUAR_MIN = 0.42  # ~24 degrees


def _unit(angle: float) -> tuple[float, float]:
    return (math.cos(angle), math.sin(angle))


def _dot_centers(scene: dict) -> list[tuple[float, float]]:
    cx, cy = CENTER
    return [(cx + px, cy + py) for (px, py) in scene["dots"]]


def _gap_geometry(dots) -> tuple[float, float]:
    """Return (largest angular gap in radians around the canvas centre, margin).

    ``margin`` is positive when the decision is clear: for a surround it is
    ``pi - largest_gap`` (how far below a half-turn the biggest gap is); for a
    non-surround it is ``largest_gap - pi`` (how far above a half-turn it is).
    """
    cx, cy = CENTER
    angles = sorted(
        math.atan2(py - cy, px - cx) for (px, py) in dots
    )
    gaps = []
    for index in range(len(angles)):
        nxt = angles[(index + 1) % len(angles)]
        cur = angles[index]
        diff = nxt - cur
        if index == len(angles) - 1:
            diff += 2.0 * math.pi
        gaps.append(diff)
    largest = max(gaps)
    return largest, abs(math.pi - largest)


def _gaps_ok(dots, sep: float) -> bool:
    for i in range(len(dots)):
        for j in range(i + 1, len(dots)):
            if math.hypot(dots[i][0] - dots[j][0], dots[i][1] - dots[j][1]) < sep:
                return False
    return True


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    surround = bool(rng.integers(0, 2))
    n = int(rng.integers(N_MIN, N_MAX + 1))

    # Constructive candidate builder; if pairwise separation cannot be reached
    # within the attempt budget the dot count is decremented (down to N_MIN) so
    # the scene always remains well-separated and the pixel oracle recovers one
    # clean centroid per dot.
    def build(k: int):
        radius = float(rng.uniform(R_MIN, R_MAX))
        if surround:
            phase = float(rng.uniform(0.0, 2.0 * math.pi))
            base = [phase + i * (2.0 * math.pi / k) for i in range(k)]
            angles = [
                a + math.radians(float(rng.uniform(-JITTER_DEG, JITTER_DEG))) for a in base
            ]
        else:
            half = math.radians(float(rng.uniform(ARC_MIN_DEG, ARC_MAX_DEG))) / 2.0
            center_angle = float(rng.uniform(0.0, 2.0 * math.pi))
            angles = [center_angle + float(rng.uniform(-half, half)) for _ in range(k)]
        dots = []
        for a in angles:
            r = radius + float(rng.uniform(-RAD_JITTER, RAD_JITTER))
            ux, uy = _unit(a)
            dots.append([r * ux, r * uy])
        return radius, dots

    k = n
    attempts = 0
    while True:
        attempts += 1
        radius, dots = build(k)
        if _gaps_ok(dots, MIN_PAIR_SEP):
            break
        if attempts >= 3000 and k > N_MIN:
            k -= 1
            attempts = 0

    return {
        "surround": surround,
        "count": k,
        "dots": dots,
        "radius": radius,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    dots = _dot_centers(scene)
    # Reading ``surround`` here keeps the answer field part of the visible raster
    # path (it permutes the paint order); the blobs do not overlap so the pixels
    # are independent of this order.
    order = list(range(len(dots)))
    if not scene["surround"]:
        order.reverse()
    for index in order:
        (px, py) = dots[index]
        draw.ellipse(
            (px - DOT_RADIUS, py - DOT_RADIUS, px + DOT_RADIUS, py + DOT_RADIUS),
            fill=DOT_COLOR,
        )
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["surround"] else "no"


def margin(scene: dict) -> float:
    _largest, m = _gap_geometry(_dot_centers(scene))
    return float(m)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUAR_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
