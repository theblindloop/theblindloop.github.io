"""Independent pixel-only inverse arm for the center-surround world.

Recovers whether the whole constellation of dark dots surrounds the exact center
of the square canvas, from the final PNG alone. It never sees the latent scene,
the analytic gold, or the renderer code.

Pipeline: threshold every non-white (dark) pixel on the white canvas, label the
resulting 8-connected blobs (one per dot) and take each blob's centroid, then
measure the largest empty angular gap of those dot centres around the exact
canvas center (both dimensions are known public constants, not scene data). The
answer is "yes" when that largest gap is smaller than a half-turn (the dots
encircle the centre) and "no" when it is larger (every dot lies in some open
half-plane through the centre, so the centre is not enclosed).
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
CX = WIDTH / 2.0
CY = HEIGHT / 2.0
PI = 3.141592653589793


def _dark_mask(arr: np.ndarray) -> np.ndarray:
    """A pixel is a dot pixel if it is not near-white (anti-aliased/white omitted)."""
    return (arr[..., 0] < 180) | (arr[..., 1] < 180) | (arr[..., 2] < 180)


def _blob_centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    ys, xs = np.nonzero(mask)
    pixels = set(zip(ys.tolist(), xs.tolist()))
    centers: list[tuple[float, float]] = []
    seen = set()
    for (y, x) in pixels:
        if (y, x) in seen:
            continue
        queue = deque([(y, x)])
        comp: list[tuple[int, int]] = []
        seen.add((y, x))
        while queue:
            cy, cx = queue.popleft()
            comp.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx2 = cy + dy, cx + dx
                    if (ny, nx2) in pixels and (ny, nx2) not in seen:
                        seen.add((ny, nx2))
                        queue.append((ny, nx2))
        sy = sum(c[0] for c in comp)
        sx = sum(c[1] for c in comp)
        if len(comp) >= 10:
            centers.append((sx / len(comp), sy / len(comp)))
    return centers


def _largest_gap(centers) -> float:
    if not centers:
        return float("inf")
    angles = sorted((np.arctan2(cy - CY, cx - CX) for (cx, cy) in centers))
    largest = 0.0
    for index in range(len(angles)):
        cur = angles[index]
        nxt = angles[(index + 1) % len(angles)]
        diff = nxt - cur
        if index == len(angles) - 1:
            diff += 2.0 * PI
        if diff > largest:
            largest = diff
    return largest


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    centers = _blob_centroids(_dark_mask(arr))
    if not centers:
        return "no"
    largest = _largest_gap(centers)
    return "yes" if largest < PI else "no"
