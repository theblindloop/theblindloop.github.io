"""Independent image-only inverse arm for the mirror-copy board."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1000, 650)
SIZE = 7
START_X, START_Y, START_CELL, START_GAP = 40, 93, 30, 3
CARD_XS, CARD_Y = (345, 555, 765), 84
OPTION_XS, OPTION_Y, OPTION_CELL, OPTION_GAP = (18, 220, 422, 624, 826), 452, 20, 2
INK = np.array((32, 54, 74), dtype=np.float32)
PAPER = np.array((247, 243, 232), dtype=np.float32)
BLUE = np.array((21, 101, 192), dtype=np.float32)


class OracleError(ValueError):
    pass


def _bit(array: np.ndarray, left: int, top: int, cell: int) -> int:
    radius = 7 if cell >= 30 else 4
    cx, cy = left + cell // 2, top + cell // 2
    patch = array[cy - radius : cy + radius + 1, cx - radius : cx + radius + 1].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ink_distance = float(np.linalg.norm(mean - INK))
    paper_distance = float(np.linalg.norm(mean - PAPER))
    if ink_distance < 6.0 and ink_distance + 30.0 < paper_distance:
        return 1
    if paper_distance < 6.0 and paper_distance + 30.0 < ink_distance:
        return 0
    raise OracleError("a cell has no clean dark-or-light interior")


def _grid(array: np.ndarray, x: int, y: int, cell: int, gap: int) -> np.ndarray:
    return np.array(
        [
            [_bit(array, x + col * (cell + gap), y + row * (cell + gap), cell) for col in range(SIZE)]
            for row in range(SIZE)
        ],
        dtype=np.uint8,
    )


def _axis_from_card(array: np.ndarray, x: int) -> str:
    gx, gy = x + 31, CARD_Y + 65
    crop = array[gy : gy + 113, gx : gx + 113].astype(np.float32)
    mask = np.linalg.norm(crop - BLUE, axis=2) < 10.0
    ys, xs = np.nonzero(mask)
    if len(xs) < 550 or np.ptp(xs) < 5 or np.ptp(ys) < 5:
        raise OracleError("a mirror card has no full blue axis")
    x0 = xs.astype(np.float64) - xs.mean()
    y0 = ys.astype(np.float64) - ys.mean()
    var_x, var_y = float(np.mean(x0 * x0)), float(np.mean(y0 * y0))
    covariance = float(np.mean(x0 * y0))
    if var_x < 0.05 * var_y:
        return "vertical"
    if var_y < 0.05 * var_x:
        return "horizontal"
    correlation = covariance / max((var_x * var_y) ** 0.5, 1e-9)
    if correlation > 0.93:
        return "falling"
    if correlation < -0.93:
        return "rising"
    raise OracleError("blue axis orientation is not one of the four supported mirrors")


def _validate_option_labels(array: np.ndarray) -> None:
    for x in OPTION_XS:
        cx, cy = x + 76, 416
        patch = array[cy - 20 : cy + 21, cx - 20 : cx + 21].astype(np.float32)
        ink_count = int((np.linalg.norm(patch - INK, axis=2) < 10.0).sum())
        white_count = int((patch.mean(axis=2) > 245).sum())
        if ink_count < 950 or white_count < 28:
            raise OracleError("an option label disk or glyph is missing")


def _mirror(bits: np.ndarray, axis: str) -> np.ndarray:
    if axis == "vertical":
        return np.fliplr(bits)
    if axis == "horizontal":
        return np.flipud(bits)
    if axis == "falling":
        return bits.T
    if axis == "rising":
        return np.flipud(np.fliplr(bits)).T
    raise OracleError("unknown recovered axis")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_option_labels(array)
    start = _grid(array, START_X, START_Y, START_CELL, START_GAP)
    if int(start.sum()) not in (2, 3):
        raise OracleError("START must contain two or three stamps")
    axes = [_axis_from_card(array, x) for x in CARD_XS]
    state = start.copy()
    for axis in axes:
        state = np.maximum(state, _mirror(state, axis))
    options = [_grid(array, x, OPTION_Y, OPTION_CELL, OPTION_GAP) for x in OPTION_XS]
    signatures = {option.tobytes() for option in options}
    if len(signatures) != 5:
        raise OracleError("option patterns are not distinct")
    matches = [index for index, option in enumerate(options) if np.array_equal(option, state)]
    if len(matches) != 1:
        raise OracleError("the recovered final set does not match exactly one option")
    return "ABCDE"[matches[0]]
