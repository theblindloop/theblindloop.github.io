"""Deterministic sampler, renderer, and analytic mirror-copy program."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 650
SIZE = 7
START_X, START_Y, START_CELL, START_GAP = 40, 93, 30, 3
CARD_XS = (345, 555, 765)
CARD_Y = 84
OPTION_XS = (18, 220, 422, 624, 826)
OPTION_Y, OPTION_CELL, OPTION_GAP = 452, 20, 2
BACKGROUND = "#ffffff"
INK = "#20364a"
PAPER = "#f7f3e8"
BORDER = "#6d7e8b"
PALE = "#dce4e8"
BLUE = "#1565c0"
AMBER = "#bd6b13"
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
AXES = ("vertical", "horizontal", "rising", "falling")


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _reflect(cells: Iterable[tuple[int, int]], axis: str) -> set[tuple[int, int]]:
    points = {(int(row), int(col)) for row, col in cells}
    if axis == "vertical":
        return {(row, SIZE - 1 - col) for row, col in points}
    if axis == "horizontal":
        return {(SIZE - 1 - row, col) for row, col in points}
    if axis == "falling":
        return {(col, row) for row, col in points}
    if axis == "rising":
        return {(SIZE - 1 - col, SIZE - 1 - row) for row, col in points}
    raise ValueError(f"unknown mirror axis {axis!r}")


def _evolve(cells: Iterable[tuple[int, int]], axes: Iterable[str]) -> set[tuple[int, int]]:
    state = {(int(row), int(col)) for row, col in cells}
    for axis in axes:
        state |= _reflect(state, axis)
    return state


def _matrix(cells: Iterable[tuple[int, int]]) -> list[list[int]]:
    points = set(cells)
    return [[int((row, col) in points) for col in range(SIZE)] for row in range(SIZE)]


def _cells(matrix: Iterable[Iterable[int]]) -> set[tuple[int, int]]:
    rows = [[int(value) for value in row] for row in matrix]
    if len(rows) != SIZE or any(len(row) != SIZE for row in rows):
        raise ValueError("every stamp grid must be 7 by 7")
    if any(value not in (0, 1) for row in rows for value in row):
        raise ValueError("stamp grids must be binary")
    return {(row, col) for row in range(SIZE) for col in range(SIZE) if rows[row][col]}


def _parse_scene(scene: dict) -> tuple[set[tuple[int, int]], list[str], list[list[list[int]]]]:
    if set(scene) != {"start_cells", "mirror_sequence", "options"}:
        raise ValueError("scene fields must be exactly start_cells, mirror_sequence, and options")
    raw_cells = scene["start_cells"]
    if not isinstance(raw_cells, list):
        raise ValueError("start_cells must be a list")
    start = {(int(pair[0]), int(pair[1])) for pair in raw_cells if isinstance(pair, list) and len(pair) == 2}
    if len(start) != len(raw_cells) or not start:
        raise ValueError("start cells must be unique coordinate pairs")
    if any(not (0 <= row < SIZE and 0 <= col < SIZE) for row, col in start):
        raise ValueError("start coordinate is outside the grid")
    axes = [str(axis) for axis in scene["mirror_sequence"]]
    if len(axes) != 3 or any(axis not in AXES for axis in axes):
        raise ValueError("exactly three known mirror axes are required")
    raw_options = scene["options"]
    if not isinstance(raw_options, list) or len(raw_options) != 5:
        raise ValueError("exactly five options are required")
    options = [_matrix(_cells(matrix)) for matrix in raw_options]
    signatures = {tuple(value for row in matrix for value in row) for matrix in options}
    if len(signatures) != 5:
        raise ValueError("options must be distinct")
    return start, axes, options


def _original_only(start: set[tuple[int, int]], axes: list[str]) -> set[tuple[int, int]]:
    result = set(start)
    for axis in axes:
        result |= _reflect(start, axis)
    return result


def _replacement(start: set[tuple[int, int]], axes: list[str]) -> set[tuple[int, int]]:
    state = set(start)
    for axis in axes:
        state = _reflect(state, axis)
    return state


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    universe = [(row, col) for row in range(SIZE) for col in range(SIZE)]
    for _ in range(1200):
        axes = list(AXES)
        rng.shuffle(axes)
        sequence = axes[:3]
        start = set(rng.sample(universe, rng.choice((2, 2, 3))))
        states = [set(start)]
        for axis in sequence:
            states.append(states[-1] | _reflect(states[-1], axis))
        target = states[-1]
        if not all(len(states[index + 1]) > len(states[index]) for index in range(3)):
            continue
        if not (8 <= len(target) <= 22):
            continue
        preferred = [
            states[-2],
            _replacement(start, sequence),
            _original_only(start, sequence),
            _evolve(start, reversed(sequence)),
        ]
        for index, axis in enumerate(sequence):
            if axis in ("rising", "falling"):
                mistaken = list(sequence)
                mistaken[index] = "falling" if axis == "rising" else "rising"
                preferred.append(_evolve(start, mistaken))
        distractors: list[set[tuple[int, int]]] = []
        for value in preferred:
            if value != target and value not in distractors:
                distractors.append(value)
        # Deterministic, whole-cell fallback mutations retain a visible margin.
        fallback = list(universe)
        rng.shuffle(fallback)
        for point in fallback:
            mutated = set(target)
            if point in mutated:
                mutated.remove(point)
            else:
                mutated.add(point)
            if mutated and mutated not in distractors and mutated != target:
                distractors.append(mutated)
            if len(distractors) >= 4:
                break
        if len(distractors) < 4:
            continue
        distractors = distractors[:4]
        rng.shuffle(distractors)
        options = [_matrix(value) for value in distractors]
        options.insert(int(seed) % 5, _matrix(target))
        return {
            "start_cells": [[row, col] for row, col in sorted(start)],
            "mirror_sequence": sequence,
            "options": options,
        }
    raise RuntimeError("could not sample a nondegenerate mirror-copy board")


def analytic_gold(scene: dict) -> str:
    start, axes, options = _parse_scene(scene)
    target = _matrix(_evolve(start, axes))
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("final mirror-copy set must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, axes, options = _parse_scene(scene)
    target = _matrix(_evolve(start, axes))
    distances = [
        sum(a != b for left, right in zip(target, option) for a, b in zip(left, right))
        for option in options
        if option != target
    ]
    return float(min(distances)) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, axes, options = _parse_scene(scene)
        target = _matrix(_evolve(start, axes))
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _parse_scene(scene)
    return []


def _draw_grid(
    draw: ImageDraw.ImageDraw,
    matrix: list[list[int]],
    x: int,
    y: int,
    cell: int,
    gap: int,
    *,
    accent: str = BORDER,
) -> None:
    for row in range(SIZE):
        for col in range(SIZE):
            left = x + col * (cell + gap)
            top = y + row * (cell + gap)
            draw.rectangle(
                (left, top, left + cell, top + cell),
                fill=INK if matrix[row][col] else PAPER,
                outline=accent,
                width=2,
            )


def _draw_card(draw: ImageDraw.ImageDraw, x: int, number: int, axis: str) -> None:
    draw.rounded_rectangle((x, CARD_Y, x + 175, CARD_Y + 238), radius=12, fill="#f8fafb", outline=PALE, width=3)
    draw.text((x + 87, CARD_Y + 27), f"MIRROR {number}", fill=BORDER, font=_font(18), anchor="mm")
    gx, gy, span = x + 31, CARD_Y + 65, 112
    for step in range(8):
        pos = step * 16
        draw.line((gx + pos, gy, gx + pos, gy + span), fill=PALE, width=1)
        draw.line((gx, gy + pos, gx + span, gy + pos), fill=PALE, width=1)
    draw.rectangle((gx, gy, gx + span, gy + span), outline=BORDER, width=2)
    if axis == "vertical":
        endpoints = (gx + 56, gy, gx + 56, gy + span)
    elif axis == "horizontal":
        endpoints = (gx, gy + 56, gx + span, gy + 56)
    elif axis == "falling":
        endpoints = (gx, gy, gx + span, gy + span)
    elif axis == "rising":
        endpoints = (gx, gy + span, gx + span, gy)
    else:
        raise ValueError(axis)
    draw.line(endpoints, fill=BLUE, width=8)
    draw.text((x + 87, CARD_Y + 207), "COPY + KEEP", fill=BLUE, font=_font(16), anchor="mm")


def render(scene: dict) -> Image.Image:
    start, axes, options = _parse_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((500, 26), "COPY + KEEP  •  LEFT TO RIGHT", fill=INK, font=_font(24), anchor="mm")
    draw.text((154, 68), "START", fill=AMBER, font=_font(23), anchor="mm")
    _draw_grid(draw, _matrix(start), START_X, START_Y, START_CELL, START_GAP, accent=AMBER)
    for number, (x, axis) in enumerate(zip(CARD_XS, axes), 1):
        _draw_card(draw, x, number, axis)
    draw.line((28, 352, 972, 352), fill=PALE, width=2)
    draw.text((500, 377), "FINAL PATTERN", fill=BORDER, font=_font(20), anchor="mm")
    for label, matrix, x in zip("ABCDE", options, OPTION_XS):
        centre = (x + 76, 416)
        draw.ellipse((centre[0] - 22, centre[1] - 22, centre[0] + 22, centre[1] + 22), fill=INK)
        draw.text(centre, label, fill="white", font=_font(25), anchor="mm")
        _draw_grid(draw, matrix, x, OPTION_Y, OPTION_CELL, OPTION_GAP)
    return image
