"""Independent pixel-only inverse arm for the figure-signature-match world.

Receives ONLY the final PNG. It never imports or reads the renderer, prompts,
generator, verifier, or any latent scene data.

The picture has a fixed visible four-panel layout: one reference figure near the
top centre and three candidate figures (left, centre, right) in a row below.
For each panel the oracle:

1. segments the dark ink from the white background;
2. counts the 4-connected black components (connected pieces) inside the panel;
3. floods the white background from the panel border and counts the white
   components it cannot reach (fully sealed white holes);

giving a (pieces, holes) signature per panel. It then returns the panel whose
candidate signature equals the reference signature.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "figure_signature_match-oracle-0.4.0"


def _black_mask(arr: np.ndarray) -> np.ndarray:
    r = arr[..., 0].astype(np.int16)
    g = arr[..., 1].astype(np.int16)
    b = arr[..., 2].astype(np.int16)
    return (r <= 140) & (g <= 140) & (b <= 140)


def _component_signature(panel_black: np.ndarray) -> tuple[int, int]:
    """(connected black pieces, enclosed white holes) within one panel crop."""
    h, w = panel_black.shape
    pieces = 0
    seen = np.zeros_like(panel_black, dtype=bool)
    for y0 in range(h):
        for x0 in range(w):
            if panel_black[y0, x0] and not seen[y0, x0]:
                pieces += 1
                stack = [(y0, x0)]
                seen[y0, x0] = True
                while stack:
                    y, x = stack.pop()
                    for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                        if 0 <= ny < h and 0 <= nx < w and panel_black[ny, nx] and not seen[ny, nx]:
                            seen[ny, nx] = True
                            stack.append((ny, nx))

    white = ~panel_black
    border_reached = np.zeros_like(panel_black, dtype=bool)
    stack = []
    for y in range(h):
        for x in range(w):
            if (y == 0 or y == h - 1 or x == 0 or x == w - 1) and white[y, x]:
                border_reached[y, x] = True
                stack.append((y, x))
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
            if 0 <= ny < h and 0 <= nx < w and white[ny, nx] and not border_reached[ny, nx]:
                border_reached[ny, nx] = True
                stack.append((ny, nx))

    holes = 0
    unseen_white = np.zeros_like(panel_black, dtype=bool)
    for y0 in range(h):
        for x0 in range(w):
            if white[y0, x0] and not border_reached[y0, x0] and not unseen_white[y0, x0]:
                holes += 1
                stack = [(y0, x0)]
                unseen_white[y0, x0] = True
                while stack:
                    y, x = stack.pop()
                    for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                        if (
                            0 <= ny < h
                            and 0 <= nx < w
                            and white[ny, nx]
                            and not border_reached[ny, nx]
                            and not unseen_white[ny, nx]
                        ):
                            unseen_white[ny, nx] = True
                            stack.append((ny, nx))
    return (pieces, holes)


def decision_from_image(image: Image.Image) -> int:
    """0/1/2 for which candidate figure matches the reference's signature."""
    arr = np.asarray(image.convert("RGB"))
    h, w = arr.shape[0], arr.shape[1]
    black = _black_mask(arr)

    def panel(x0, x1, y0, y1):
        return black[int(round(y0)):int(round(y1)), int(round(x0)):int(round(x1))]

    ref_box = panel(0.25 * w, 0.75 * w, 0.0, 0.5 * h)
    cand_boxes = [
        panel(0.0, w / 3.0, 0.5 * h, h),
        panel(w / 3.0, 2.0 * w / 3.0, 0.5 * h, h),
        panel(2.0 * w / 3.0, w, 0.5 * h, h),
    ]
    ref_sig = _component_signature(ref_box)
    for index, box in enumerate(cand_boxes):
        if _component_signature(box) == ref_sig:
            return index
    return -1
