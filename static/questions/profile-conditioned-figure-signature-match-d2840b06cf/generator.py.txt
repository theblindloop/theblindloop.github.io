"""Figure topological-signature match world renderer.

Four filled dark shapes ("figures") are drawn on a white background: one
reference figure near the top centre and three candidate figures in a row below
(left, centre, right). Each figure is a simple set of one or two filled closed
components -- a solid disk, or a ring (a solid disk with a white hole cut out).

The *topological signature* of a figure is the pair

    signature = (number of connected black pieces, number of sealed white holes)

A solid disk contributes (1, 0); a ring contributes (1, 1); two pieces add their
contribution, so a figure of two solids is (2, 0) and a solid plus a ring is
(2, 1). Exactly one of the three candidate figures has the SAME signature as the
reference figure; the decision is which candidate (left/centre/right) that is.

The scene is a plain dict with a fixed 4-panel layout. Every field either draws
ink or is read back by the analytic gold, so there is no hidden label state.

    z = {
        "width": 800, "height": 560, "background": "white",
        "ref":  [ {component}, ... ],          # components of the reference figure
        "cands": [ [ {component}, ... ], ... ], # three candidate figures
    }

Component dicts:
    {"kind": "solid", "cx": x, "cy": y, "r": R}            -> filled disk
    {"kind": "ring",  "cx": x, "cy": y, "r": R, "ri": Ri}  -> filled annulus
Coordinates cx, cy are absolute canvas coordinates for that component.
"""

from __future__ import annotations

from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "figure_signature_match-0.4.0"

WHITE = (255, 255, 255)
DARK = (45, 45, 45)

WIDTH_DEFAULT = 800
HEIGHT_DEFAULT = 560
QUARANTINE_MIN = 6.0

# Figure anchors: reference on top-centre, three candidates along the bottom.
REF_ANCHOR = (400.0, 130.0)
CAND_ANCHORS = [(133.0, 400.0), (400.0, 400.0), (667.0, 400.0)]

SOLID_R = 38.0
RING_R = 44.0
RING_RI = 26.0
PAIR_OFFSET = 52.5

# Signature pool: (pieces, holes) -> component template relative to an anchor.
_SIG_TEMPLATES: dict[tuple[int, int], list[dict]] = {
    (1, 0): [{"kind": "solid", "dx": 0.0}],
    (1, 1): [{"kind": "ring", "dx": 0.0}],
    (2, 0): [
        {"kind": "solid", "dx": -PAIR_OFFSET},
        {"kind": "solid", "dx": PAIR_OFFSET},
    ],
    (2, 1): [
        {"kind": "solid", "dx": -PAIR_OFFSET},
        {"kind": "ring", "dx": PAIR_OFFSET},
    ],
}
SIGNATURE_POOL = sorted(_SIG_TEMPLATES)


def _resolve(anchor: tuple[float, float], template: dict) -> dict:
    kind = template["kind"]
    ax, ay = anchor
    comp = {
        "kind": kind,
        "cx": round(ax + template["dx"], 1),
        "cy": round(ay + 0.0, 1),
        "r": SOLID_R if kind == "solid" else RING_R,
    }
    if kind == "ring":
        comp["ri"] = RING_RI
    return comp


def _figure_from_signature(sig: tuple[int, int], anchor: tuple[float, float]) -> list[dict]:
    return [_resolve(anchor, template) for template in _SIG_TEMPLATES[sig]]


def sample_scene(rng) -> dict:
    """Sample a balanced scene with exactly one matching candidate figure."""
    ref_sig = rng.choice(SIGNATURE_POOL)
    others = [s for s in SIGNATURE_POOL if s != ref_sig]
    rng.shuffle(others)
    cand_sigs = [ref_sig] + others[:2]
    rng.shuffle(cand_sigs)
    ref = _figure_from_signature(ref_sig, REF_ANCHOR)
    cands = [
        _figure_from_signature(cand_sigs[0], CAND_ANCHORS[0]),
        _figure_from_signature(cand_sigs[1], CAND_ANCHORS[1]),
        _figure_from_signature(cand_sigs[2], CAND_ANCHORS[2]),
    ]
    return {
        "width": WIDTH_DEFAULT,
        "height": HEIGHT_DEFAULT,
        "background": "white",
        "ref": ref,
        "cands": cands,
    }


def _draw_components(image: Image.Image, components: list[dict]) -> Image.Image:
    draw = ImageDraw.Draw(image)
    for comp in components:
        kind = comp["kind"]
        cx, cy = comp["cx"], comp["cy"]
        r = comp["r"]
        if kind == "solid":
            draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=DARK)
        else:
            ri = comp["ri"]
            draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=DARK)
            draw.ellipse([cx - ri, cy - ri, cx + ri, cy + ri], fill=WHITE)
    return image


def render(scene: dict, supersample: int = 3) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    if ss == 1:
        img = Image.new("RGB", (width, height), background)
    else:
        img = Image.new("RGB", (width * ss, height * ss), background)
        scale = ss
        ref = [
            dict(c, cx=c["cx"] * scale, cy=c["cy"] * scale, r=c["r"] * scale,
                 ri=c.get("ri", 0) * scale)
            for c in scene["ref"]
        ]
        cands = [
            [
                dict(c, cx=c["cx"] * scale, cy=c["cy"] * scale, r=c["r"] * scale,
                     ri=c.get("ri", 0) * scale)
                for c in figure
            ]
            for figure in scene["cands"]
        ]
        img = _draw_components(img, ref)
        for figure in cands:
            img = _draw_components(img, figure)
        img = img.resize((width, height), Image.LANCZOS)
        return img
    img = _draw_components(img, scene["ref"])
    for figure in scene["cands"]:
        img = _draw_components(img, figure)
    return img


def _signature(components: list[dict]) -> tuple[int, int]:
    pieces = len(components)
    holes = sum(1 for comp in components if comp["kind"] == "ring")
    return (pieces, holes)


def analytic_gold(scene: dict) -> int:
    """0/1/2 for which candidate figure shares the reference's signature."""
    ref_sig = _signature(scene["ref"])
    sigs = [_signature(figure) for figure in scene["cands"]]
    return sigs.index(ref_sig)


def _clearance(scene: dict) -> float:
    """Smallest robustness clearance across all ink, in canvas pixels."""
    all_components = list(scene["ref"])
    for figure in scene["cands"]:
        all_components.extend(figure)
    width = scene["width"]
    height = scene["height"]
    best = float("inf")
    for comp in all_components:
        r = comp["r"]
        cx, cy = comp["cx"], comp["cy"]
        best = min(best, cx - r, cy - r, width - (cx + r), height - (cy + r))
        if comp["kind"] == "ring":
            best = min(best, comp["ri"], (comp["r"] - comp["ri"]) / 2.0)
    for figure in (scene["ref"], *scene["cands"]):
        if len(figure) == 2:
            a, b = figure
            gap = abs(a["cx"] - b["cx"]) - a["r"] - b["r"]
            best = min(best, gap / 2.0)
    return best


def margin(scene: dict) -> float:
    """Robustness margin = smallest ink clearance from borders, holes, or pieces."""
    return float(_clearance(scene))


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose features are too thin or close to trust."""
    return _clearance(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render-invariant alias: every field draws ink or is read by gold.

    The colours are fixed (dark ink on white), the four panels are part of the
    visible layout, and the answer (which candidate matches the reference
    signature) is recovered purely from the rendered pixels. There is no
    reparametrization that keeps pixels identical while carrying a different
    label, so the world declares not_applicable for the latent-alias gate.
    """
    return iter(())
