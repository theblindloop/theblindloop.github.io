"""Renderer for intervention search over a causally pruned swap log."""
from __future__ import annotations
import math, random
from PIL import Image, ImageDraw, ImageFont

CANVAS=(1000,660); BOARD_W,BOARD_H=960,620
REGISTERS=("A","B","C","D"); SHAPES=("circle","triangle","star","diamond")
SHIELDS=("amber","cyan","violet")
PAIRS=tuple((a,b) for i,a in enumerate(REGISTERS) for b in REGISTERS[i+1:])
RGB={"white":(255,255,255),"board":(246,248,244),"card":(255,255,252),"ink":(37,43,49),
     "muted":(105,112,118),"border":(55,64,72),"cancel":(204,43,55),"target":(196,119,16),
     "amber":(232,145,23),"cyan":(13,151,184),"violet":(139,72,181),
     "A":(198,82,52),"B":(22,139,150),"C":(121,75,180),"D":(65,104,174),
     "circle":(42,124,202),"triangle":(38,157,91),"star":(230,158,26),"diamond":(205,69,119)}
FONT="/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"; BOLD="/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"

def _dependencies(pairs):
    last={}; deps=[]
    for i,pair in enumerate(pairs):
        deps.append({last[s] for s in pair if s in last})
        for s in pair: last[s]=i
    return deps

def _removed(pairs,cancel,shield):
    deps=_dependencies(pairs); removed={cancel}; depth={cancel:0}
    for i in range(cancel+1,len(pairs)):
        if i==shield: continue
        hit=deps[i]&removed
        if hit:
            removed.add(i); depth[i]=1+max(depth[p] for p in hit)
    return removed,depth

def _run(initial,pairs,skipped):
    state=list(initial)
    for i,pair in enumerate(pairs):
        if i in skipped: continue
        a,b=(REGISTERS.index(s) for s in pair); state[a],state[b]=state[b],state[a]
    return state

def _outcomes(initial,pairs,cancel,shields,target):
    ti=REGISTERS.index(target)
    return [_run(initial,pairs,_removed(pairs,cancel,s)[0])[ti] for s in shields]

def sample_scene(seed:int)->dict:
    rng=random.Random((int(seed)^0x5A13D7C9)&0xffffffffffffffff); correct=int(seed)%3
    for _ in range(240000):
        initial=list(SHAPES); rng.shuffle(initial)
        pairs=[list(rng.choice(PAIRS)) for _ in range(7)]; cancel=rng.randrange(1,3)
        baseline,depth=_removed(pairs,cancel,None); descendants=sorted(baseline-{cancel})
        if len(descendants)<3 or max(depth.values())<2: continue
        shields=sorted(rng.sample(descendants,3))
        sets=[_removed(pairs,cancel,s)[0] for s in shields]
        if len({tuple(sorted(x)) for x in sets})<3: continue
        targets=list(REGISTERS); rng.shuffle(targets)
        for target in targets:
            outcomes=_outcomes(initial,pairs,cancel,shields,target)
            if len(set(outcomes))!=3: continue
            goal=outcomes[correct]; ti=REGISTERS.index(target)
            if goal in {_run(initial,pairs,baseline)[ti],_run(initial,pairs,{cancel})[ti]}: continue
            return {"initial":initial,"pairs":pairs,"cancel_card":cancel,"shield_cards":shields,
                    "target_slot":target,"goal_shape":goal,"anchor_x":rng.randint(18,22),"anchor_y":rng.randint(18,22)}
    raise RuntimeError("could not sample a unique three-intervention rescue scene")

def _validate(scene):
    initial=[str(v) for v in scene["initial"]]; pairs=[[str(v) for v in p] for p in scene["pairs"]]
    cancel=int(scene["cancel_card"]); shields=[int(v) for v in scene["shield_cards"]]
    target=str(scene["target_slot"]); goal=str(scene["goal_shape"])
    if sorted(initial)!=sorted(SHAPES) or len(pairs)!=7: raise ValueError("invalid start or card count")
    if cancel not in range(1,3) or target not in REGISTERS or goal not in SHAPES: raise ValueError("invalid selector")
    if any(len(p)!=2 or tuple(p) not in PAIRS for p in pairs): raise ValueError("invalid pair")
    baseline,depth=_removed(pairs,cancel,None); descendants=baseline-{cancel}
    if len(shields)!=3 or shields!=sorted(set(shields)) or not set(shields)<=descendants: raise ValueError("invalid shields")
    if len(descendants)<3 or max(depth.values())<2: raise ValueError("shallow cascade")
    sets=[_removed(pairs,cancel,s)[0] for s in shields]
    if len({tuple(sorted(x)) for x in sets})!=3: raise ValueError("equal causal slices")
    outcomes=_outcomes(initial,pairs,cancel,shields,target)
    if len(set(outcomes))!=3 or outcomes.count(goal)!=1: raise ValueError("nonunique goal")
    ti=REGISTERS.index(target)
    if goal in {_run(initial,pairs,baseline)[ti],_run(initial,pairs,{cancel})[ti]}: raise ValueError("shortcut")

def analytic_gold(scene):
    _validate(scene)
    out=_outcomes(scene["initial"],scene["pairs"],int(scene["cancel_card"]),[int(v) for v in scene["shield_cards"]],str(scene["target_slot"]))
    return SHIELDS[out.index(str(scene["goal_shape"]))]

def margin(scene):
    try: return 1.0 if analytic_gold(scene) in SHIELDS else 0.0
    except (KeyError,TypeError,ValueError): return 0.0
def is_quarantined(scene): return margin(scene)<1.0
def latent_symmetries(scene): return []
def _font(n,bold=False): return ImageFont.truetype(BOLD if bold else FONT,n)
def _text(d,xy,s,n,fill=None,bold=False,anchor="mm"): d.text(xy,s,font=_font(n,bold),fill=fill or RGB["ink"],anchor=anchor)

def _shape(d,c,k,scale=20):
    x,y=c; color=RGB[k]
    if k=="circle": d.ellipse((x-scale,y-scale,x+scale,y+scale),fill=color,outline=RGB["ink"],width=2)
    elif k=="triangle":
        p=((x,y-scale-2),(x-scale-2,y+scale),(x+scale+2,y+scale)); d.polygon(p,fill=color); d.line(p+(p[0],),fill=RGB["ink"],width=2)
    elif k=="diamond":
        p=((x,y-scale-2),(x-scale-2,y),(x,y+scale+2),(x+scale+2,y)); d.polygon(p,fill=color); d.line(p+(p[0],),fill=RGB["ink"],width=2)
    elif k=="star":
        p=[]
        for i in range(10):
            r=scale+2 if i%2==0 else scale*.45; a=-math.pi/2+i*math.pi/5; p.append((x+r*math.cos(a),y+r*math.sin(a)))
        d.polygon(p,fill=color); d.line(p+[p[0]],fill=RGB["ink"],width=2)
    else: raise ValueError(k)

def _badge(d,c,label,large=False):
    x,y=c; w,h=(52,38) if large else (42,31)
    d.rounded_rectangle((x-w//2,y-h//2,x+w//2,y+h//2),radius=6,fill=RGB[label],outline=RGB["ink"],width=2)
    _text(d,c,label,20 if large else 17,fill=RGB["white"],bold=True)

def render(scene):
    _validate(scene); ax,ay=int(scene["anchor_x"]),int(scene["anchor_y"])
    im=Image.new("RGB",CANVAS,RGB["white"]); d=ImageDraw.Draw(im)
    d.rectangle((ax,ay,ax+BOARD_W-1,ay+BOARD_H-1),fill=RGB["board"],outline=RGB["border"],width=4)
    _text(d,(ax+480,ay+24),"CHOOSE ONE CAUSAL SHIELD",22,bold=True); _text(d,(ax+57,ay+58),"START",16,bold=True)
    for label,kind,x in zip(REGISTERS,scene["initial"],(145,285,425,565),strict=True):
        d.rounded_rectangle((ax+x-60,ay+43,ax+x+60,ay+130),radius=9,fill=RGB["card"],outline=RGB[label],width=4)
        _badge(d,(ax+x-36,ay+66),label); _shape(d,(ax+x+16,ay+98),str(kind),20)
    _text(d,(ax+682,ay+56),"TARGET",15,fill=RGB["target"],bold=True); _badge(d,(ax+739,ay+94),str(scene["target_slot"]),True)
    _text(d,(ax+827,ay+56),"GOAL",15,fill=RGB["target"],bold=True); _shape(d,(ax+890,ay+94),str(scene["goal_shape"]),23)
    d.rounded_rectangle((ax+34,ay+143,ax+926,ay+286),radius=10,fill=(255,252,238),outline=RGB["target"],width=3)
    _text(d,(ax+480,ay+166),"Dependencies use FULL row: latest earlier swap touching either box.",16,bold=True)
    _text(d,(ax+480,ay+197),"Try ONE shield at a time: shielded card STAYS.",17,fill=RGB["violet"],bold=True)
    _text(d,(ax+480,ay+228),"Cancel red X; repeatedly cancel each other dependent card.",16,fill=RGB["cancel"],bold=True)
    _text(d,(ax+480,ay+260),"Restart from START, do swaps left  →  which shield puts GOAL in TARGET?",16,bold=True)
    cancel=int(scene["cancel_card"]); shield_map={c:SHIELDS[i] for i,c in enumerate(scene["shield_cards"])}
    for i,pair in enumerate(scene["pairs"]):
        x,y=ax+19+i*133,ay+304; selected=i==cancel; outline=RGB["cancel"] if selected else RGB["border"]
        d.rounded_rectangle((x,y,x+112,y+274),radius=9,fill=RGB["card"],outline=outline,width=5 if selected else 3)
        d.ellipse((x+38,y+10,x+74,y+46),fill=RGB["cancel"] if selected else RGB["ink"]); _text(d,(x+56,y+28),str(i+1),17,fill=RGB["white"],bold=True)
        _badge(d,(x+56,y+88),str(pair[0]),True); _text(d,(x+56,y+136),"↕",27,bold=True); _badge(d,(x+56,y+184),str(pair[1]),True)
        if selected: _text(d,(x+56,y+240),"X",34,fill=RGB["cancel"],bold=True)
        elif i in shield_map:
            color=shield_map[i]; d.rounded_rectangle((x+13,y+223,x+99,y+260),radius=9,fill=RGB[color],outline=RGB["ink"],width=2)
            _text(d,(x+56,y+241),"SHIELD",12,fill=RGB["white"],bold=True)
        else: _text(d,(x+56,y+241),"SWAP",13,fill=RGB["muted"],bold=True)
    return im
