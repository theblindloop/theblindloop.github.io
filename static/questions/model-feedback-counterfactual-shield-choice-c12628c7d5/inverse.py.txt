"""Independent pixel-only inverse for causal shield intervention search."""
from __future__ import annotations
import numpy as np
from PIL import Image

CANVAS=(1000,660); BOARD_W,BOARD_H=960,620
BORDER=(55,64,72); CANCEL=(204,43,55); INK=(37,43,49)
REGISTERS=("A","B","C","D"); SHIELDS=("amber","cyan","violet")
BADGES={(198,82,52):"A",(22,139,150):"B",(121,75,180):"C",(65,104,174):"D"}
SHIELD_RGB={(232,145,23):"amber",(13,151,184):"cyan",(139,72,181):"violet"}
SHAPES_BY_RGB={(42,124,202):"circle",(38,157,91):"triangle",(230,158,26):"star",(205,69,119):"diamond"}
SHAPES=("circle","triangle","star","diamond")
PAIRS=tuple((a,b) for i,a in enumerate(REGISTERS) for b in REGISTERS[i+1:])

class OracleError(ValueError): pass

def _patch(a,x,y,r=2):
    p=a[y-r:y+r+1,x-r:x+r+1]; colors={tuple(int(v) for v in row) for row in p.reshape(-1,3)}
    if len(colors)!=1: raise OracleError("a required marker is damaged")
    return next(iter(colors))

def _shape(a,x,y):
    c=_patch(a,x,y)
    if c not in SHAPES_BY_RGB: raise OracleError("unknown shape")
    return SHAPES_BY_RGB[c]

def _badge(a,x,y):
    c=_patch(a,x-18,y)
    if c not in BADGES: raise OracleError("unknown badge")
    return BADGES[c]

def _dependencies(pairs):
    last={}; deps=[]
    for i,pair in enumerate(pairs):
        deps.append({last[s] for s in pair if s in last})
        for s in pair: last[s]=i
    return deps

def _removed(pairs,cancel,shield):
    deps=_dependencies(pairs); removed={cancel}; depth={cancel:0}
    for i in range(cancel+1,len(pairs)):
        if i==shield: continue
        hit=deps[i]&removed
        if hit:
            removed.add(i); depth[i]=1+max(depth[p] for p in hit)
    return removed,depth

def _run(initial,pairs,skipped):
    state=list(initial)
    for i,pair in enumerate(pairs):
        if i in skipped: continue
        x,y=(REGISTERS.index(s) for s in pair); state[x],state[y]=state[y],state[x]
    return state

def decision_from_image(image:Image.Image)->str:
    a=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if (a.shape[1],a.shape[0])!=CANVAS: raise OracleError("unexpected canvas")
    mask=np.all(a==np.asarray(BORDER,dtype=np.uint8),axis=2); pts=np.argwhere(mask)
    if not len(pts): raise OracleError("board absent")
    top,left=pts.min(axis=0); bottom,right=pts.max(axis=0); ax,ay=int(left),int(top)
    if int(right-left+1)!=BOARD_W or int(bottom-top+1)!=BOARD_H: raise OracleError("wrong board extent")
    if not (np.all(mask[ay,ax:ax+BOARD_W]) and np.all(mask[ay+BOARD_H-1,ax:ax+BOARD_W]) and np.all(mask[ay:ay+BOARD_H,ax]) and np.all(mask[ay:ay+BOARD_H,ax+BOARD_W-1])): raise OracleError("broken border")
    initial=[_shape(a,ax+x+16,ay+98) for x in (145,285,425,565)]
    if sorted(initial)!=sorted(SHAPES): raise OracleError("bad start")
    target=_badge(a,ax+739,ay+94); goal=_shape(a,ax+890,ay+94)
    pairs=[]; selectors=[]; shield_cards={}
    for i in range(7):
        x,y=ax+19+i*133,ay+304
        selectors.append(_patch(a,x+43,y+28)); pair=(_badge(a,x+56,y+88),_badge(a,x+56,y+184))
        if pair not in PAIRS: raise OracleError("bad pair")
        pairs.append(pair); c=_patch(a,x+22,y+241)
        if c in SHIELD_RGB: shield_cards[SHIELD_RGB[c]]=i
    selected=[i for i,c in enumerate(selectors) if c==CANCEL]
    if len(selected)!=1 or any(c not in {CANCEL,INK} for c in selectors): raise OracleError("bad cancellation")
    cancel=selected[0]
    if set(shield_cards)!=set(SHIELDS): raise OracleError("bad shields")
    shields=[shield_cards[s] for s in SHIELDS]
    if shields!=sorted(set(shields)): raise OracleError("shield order mismatch")
    baseline,depth=_removed(pairs,cancel,None); descendants=baseline-{cancel}
    if cancel not in range(1,3) or len(descendants)<3 or max(depth.values())<2 or not set(shields)<=descendants: raise OracleError("invalid cascade")
    sets=[_removed(pairs,cancel,s)[0] for s in shields]
    if len({tuple(sorted(x)) for x in sets})!=3: raise OracleError("equal slices")
    ti=REGISTERS.index(target); outcomes=[_run(initial,pairs,s)[ti] for s in sets]
    if len(set(outcomes))!=3 or outcomes.count(goal)!=1: raise OracleError("nonunique goal")
    if goal in {_run(initial,pairs,baseline)[ti],_run(initial,pairs,{cancel})[ti]}: raise OracleError("shortcut")
    return SHIELDS[outcomes.index(goal)]
