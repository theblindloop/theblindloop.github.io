"""Last-in-first-out (LIFO) color-stack top register (state tracking).

The image is a left-to-right program. The leftmost panel shows the INITIAL
STACK: a vertical column of colored discs sitting on a base, read bottom to
top. The top disc (highest present) is marked with an orange ring. To the
right sit ordered operator panels, read left to right. Each operator panel is
either a PUSH panel (a coloured disc shown with a matching upward arrow,
meaning "place this colour onto the top of the stack") or a POP panel (a dark
downward arrow over an empty slot, meaning "remove the current top disc").

The stack is last-in-first-out: a push adds its declared colour on top, and a
pop removes whichever colour is currently on top (revealing the colour that
was below it). The FINAL stack is never drawn anywhere; it must be recovered
by replaying the declared push/pop operators from the initial stack. The
answer is the colour of the final top disc, or "empty" if all discs were
popped.

The latent scene stores only ``initial_colors`` (the ordered list of disc
colours in the initial stack, bottom to top) and ``ops`` (the ordered list of
push/pop operators). The decision is the final top-colour.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 1000
HEIGHT = 320
CENTER_Y = 160

START_LEFT = 20
START_TOP = 40
START_W = 220
START_H = 250
START_RIGHT = START_LEFT + START_W

STACK_X = START_LEFT + 110
STACK_BASE = START_TOP + START_H - 35
SLOT_PITCH = 46
DISC_R = 18

OP_COUNT = 4
OP_W = 110
OP_H = 140
OP_TOP = CENTER_Y - OP_H // 2
OP_FIRST = START_RIGHT + 40
OP_PITCH = 140
PUSH_DISC_R = 18

BG = (255, 255, 255)
BORDER = (200, 200, 200)
INK = (0, 0, 0)
BASE_LINE = (150, 150, 150)
RING_MARK = (240, 150, 20)
DARK = (60, 60, 60)

RED = (200, 45, 45)
GREEN = (40, 170, 60)
BLUE = (40, 60, 200)

COLORS = {"red": RED, "green": GREEN, "blue": BLUE}
COLOR_NAMES = ["red", "green", "blue"]
MAX_INITIAL = 4


def _slot_y(k: int) -> int:
    return STACK_BASE - k * SLOT_PITCH


def _op_center(index: int) -> tuple[int, int]:
    return (OP_FIRST + index * OP_PITCH + OP_W // 2, OP_TOP + OP_H // 2)


def sample_scene(seed: int) -> dict:
    """Deterministic latent scene from an integer seed."""
    rng = random.Random(seed)
    height = rng.randint(1, MAX_INITIAL)
    initial_colors = [rng.choice(COLOR_NAMES) for _ in range(height)]
    ops = []
    for _ in range(OP_COUNT):
        if rng.random() < 0.5:
            ops.append(("push", rng.choice(COLOR_NAMES)))
        else:
            ops.append(("pop", None))
    return {"initial_colors": initial_colors, "ops": ops}


def _replay(initial_colors: list, ops: list) -> str:
    stack = list(initial_colors)
    for op, color in ops:
        if op == "push":
            stack.append(color)
        elif op == "pop" and stack:
            stack.pop()
    return stack[-1] if stack else "empty"


def _draw_arrow(draw: ImageDraw.ImageDraw, cx: int, cy: int, color, up: bool) -> None:
    if up:
        draw.polygon(
            [(cx, cy + 8), (cx - 20, cy + 40), (cx + 20, cy + 40)],
            fill=color,
        )
    else:
        draw.polygon(
            [(cx, cy + 42), (cx - 20, cy + 10), (cx + 20, cy + 10)],
            fill=color,
        )


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)

    draw.rectangle(
        [START_LEFT, START_TOP, START_RIGHT, START_TOP + START_H], outline=BORDER, width=3
    )
    draw.line(
        [START_LEFT + 40, STACK_BASE + 5, START_RIGHT - 40, STACK_BASE + 5],
        fill=BASE_LINE,
        width=3,
    )

    initial = scene["initial_colors"]
    for k, color in enumerate(initial):
        y = _slot_y(k)
        draw.ellipse(
            [STACK_X - DISC_R, y - DISC_R, STACK_X + DISC_R, y + DISC_R], fill=COLORS[color]
        )
    top_k = len(initial) - 1
    ty = _slot_y(top_k)
    draw.ellipse(
        [STACK_X - (DISC_R + 5), ty - (DISC_R + 5), STACK_X + (DISC_R + 5), ty + (DISC_R + 5)],
        outline=RING_MARK,
        width=4,
    )

    for i, (op, color) in enumerate(scene["ops"]):
        cx, cy = _op_center(i)
        draw.rectangle(
            [cx - OP_W // 2, OP_TOP, cx + OP_W // 2, OP_TOP + OP_H], outline=BORDER, width=3
        )
        if op == "push":
            draw.ellipse(
                [
                    cx - PUSH_DISC_R,
                    cy - 22 - PUSH_DISC_R,
                    cx + PUSH_DISC_R,
                    cy - 22 + PUSH_DISC_R,
                ],
                fill=COLORS[color],
            )
            _draw_arrow(draw, cx, cy + 4, COLORS[color], up=True)
        else:
            draw.rectangle([cx - 18, cy - 40, cx + 18, cy - 6], outline=DARK, width=3)
            _draw_arrow(draw, cx, cy + 4, DARK, up=False)

    return img


def analytic_gold(scene: dict) -> str:
    return _replay(scene["initial_colors"], scene["ops"])


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry.

    ``initial_colors`` fixes the drawn discs (bottom to top, with the top one
    ringed) and ``ops`` fixes the drawn push/pop operators in their fixed
    left-to-right order. Reordering either field changes the pixels, and any
    other field would be render-irrelevant to the decision.
    """
    return []
