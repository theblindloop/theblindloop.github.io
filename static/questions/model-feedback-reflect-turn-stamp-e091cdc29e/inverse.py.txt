"""Independent image-only inverse for reflect-then-turn matching.

Only the final RGB raster enters this module.  It rediscovers the source,
target, dashed axis, pivot, and turn sign from their exact visible colors, then
applies the two geometric operations to every source-mask pixel.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

WIDTH = 1120
HEIGHT = 760
PANEL_W = 530
PANEL_H = 350
MARGIN_X = 20
MARGIN_Y = 20
GAP_X = 20
GAP_Y = 20

SOURCE = (28, 65, 105)
TARGET_FILL = (255, 244, 216)
AXIS = (105, 78, 151)
TURN_TAIL = (137, 224, 204)
TURN_HEAD = (7, 92, 96)
PIVOT = (229, 167, 45)


def _origin(slot: int) -> tuple[int, int]:
    return (MARGIN_X + (slot % 2) * (PANEL_W + GAP_X), MARGIN_Y + (slot // 2) * (PANEL_H + GAP_Y))


def _exact(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(color, dtype=np.uint8), axis=2)


def _points(mask: np.ndarray) -> np.ndarray:
    y, x = np.nonzero(mask)
    if len(x) == 0:
        raise ValueError("empty raster component")
    return np.column_stack((x.astype(np.float64), y.astype(np.float64)))


def _center(mask: np.ndarray) -> np.ndarray:
    return _points(mask).mean(axis=0)


def _fit_axis(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    points = _points(mask)
    center = points.mean(axis=0)
    centered = points - center
    covariance = centered.T @ centered
    values, vectors = np.linalg.eigh(covariance)
    direction = vectors[:, int(np.argmax(values))]
    norm = np.linalg.norm(direction)
    return center, direction / (norm or 1.0)


def _transform_mask(
    source: np.ndarray,
    axis_center: np.ndarray,
    axis_direction: np.ndarray,
    pivot: np.ndarray,
    turn_sign: float,
) -> set[tuple[int, int]]:
    points = _points(source)
    direction = axis_direction / (np.linalg.norm(axis_direction) or 1.0)
    relative = points - axis_center
    reflected = 2.0 * (relative @ direction)[:, None] * direction[None, :] - relative + axis_center
    shifted = reflected - pivot
    if turn_sign > 0:
        rotated = np.column_stack((-shifted[:, 1], shifted[:, 0]))
    else:
        rotated = np.column_stack((shifted[:, 1], -shifted[:, 0]))
    result = rotated + pivot
    rounded = np.rint(result).astype(np.int32)
    height, width = source.shape
    inside = (
        (rounded[:, 0] >= 0)
        & (rounded[:, 0] < width)
        & (rounded[:, 1] >= 0)
        & (rounded[:, 1] < height)
    )
    return {(int(x), int(y)) for x, y in rounded[inside]}


def _overlap_score(predicted: set[tuple[int, int]], target: np.ndarray) -> float:
    actual = {(int(x), int(y)) for y, x in np.argwhere(target)}
    union = len(predicted | actual)
    if not union:
        return -1.0
    return len(predicted & actual) / union


def _panel_score(arr: np.ndarray, slot: int) -> float:
    x0, y0 = _origin(slot)
    crop = arr[y0 : y0 + PANEL_H + 1, x0 : x0 + PANEL_W + 1]
    source = _exact(crop, SOURCE)
    target = _exact(crop, TARGET_FILL)
    axis = _exact(crop, AXIS)
    tail = _exact(crop, TURN_TAIL)
    head = _exact(crop, TURN_HEAD)
    pivot_mask = _exact(crop, PIVOT)
    if (
        source.sum() < 900
        or target.sum() < 600
        or axis.sum() < 120
        or tail.sum() < 20
        or head.sum() < 20
        or pivot_mask.sum() < 100
    ):
        raise ValueError("missing panel component")
    axis_center, axis_direction = _fit_axis(axis)
    pivot = _center(pivot_mask)
    tail_center = _center(tail)
    head_center = _center(head)
    # The arrow always starts at the right side of the pivot.  Its head is
    # visibly above or below the pivot, which encodes the signed turn.
    if abs(head_center[1] - pivot[1]) < 8.0:
        raise ValueError("turn head has no readable sign")
    turn_sign = 1.0 if head_center[1] > pivot[1] else -1.0
    predicted = _transform_mask(source, axis_center, axis_direction, pivot, turn_sign)
    target_center = _center(target)
    predicted_array = np.asarray(list(predicted), dtype=np.float64)
    predicted_center = predicted_array.mean(axis=0)
    iou = _overlap_score(predicted, target)
    center_error = float(np.linalg.norm(predicted_center - target_center))
    return iou - 0.0012 * center_error


def decision_from_image(image: Image.Image) -> str:
    """Return the tile selected using only the supplied raster."""

    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError(f"unexpected image size {arr.shape[:2]}")
    scores = [_panel_score(arr, slot) for slot in range(4)]
    winner = max(range(4), key=lambda index: scores[index])
    return f"tile-{chr(65 + winner)}"
