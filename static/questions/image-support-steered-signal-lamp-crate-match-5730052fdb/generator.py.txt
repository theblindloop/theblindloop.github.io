"""Renderer for signal_lamp_crate_match.

Two visually separate components sit at opposite corners of the canvas:

* a grey *signal plate* - a framed grey rectangle carrying a vertical stack of
  ``K`` dark round lamps, where ``K`` is any number from 2 to 5;
* a *crate block* - four coloured crates (red, amber, green, blue) whose dark
  upright bar counts are a permutation of ``(2, 3, 4, 5)``.

Exactly one crate therefore holds as many bars as the plate holds lamps, and the
answer is that crate's colour. Neither component can answer alone: the plate
fixes the target count and the crate block fixes the colour that carries it.
Nothing is drawn between the two components, so the decisive evidence is two
disconnected blobs at opposite ends of the image.

The canonical prior - a traffic signal has three lamps - is recorded for analysis
and never consulted by ``analytic_gold``.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "signal_lamp_crate_match-renderer-0.1.0"

CANVAS = 600
SUPERSAMPLE = 2
BACKGROUND = (255, 255, 255)
FRAME = (96, 96, 96)  # achromatic, too bright to be "dark", too dark to be plate grey
DARK = (26, 26, 26)  # lamps and bars

CANONICAL_PRIOR = "3"  # a traffic signal has three lamps
QUARANTINE_MARGIN = 4.0

COLOUR_NAMES = ("red", "amber", "green", "blue")
BASE_COLOURS = {
    "red": (204, 38, 38),
    "amber": (226, 148, 22),
    "green": (44, 182, 92),
    "blue": (36, 84, 200),
}
BAR_COUNTS = (2, 3, 4, 5)

_SLOTS = {
    "tl": (0.24, 0.24),
    "tr": (0.76, 0.24),
    "bl": (0.24, 0.76),
    "br": (0.76, 0.76),
}
_OPPOSITE = {"tl": "br", "br": "tl", "tr": "bl", "bl": "tr"}
_CRATE_OFFSETS = ((-1, -1), (1, -1), (-1, 1), (1, 1))


def _r(value: float) -> float:
    return round(float(value), 4)


def sample_scene(seed: int) -> dict:
    """Sample one scene. All geometry is stored explicitly and rounded."""
    rng = random.Random(seed)

    plate_slot = rng.choice(sorted(_SLOTS))
    crate_slot = _OPPOSITE[plate_slot]
    pfx, pfy = _SLOTS[plate_slot]
    cfx, cfy = _SLOTS[crate_slot]
    pcx = pfx * CANVAS + rng.uniform(-8.0, 8.0)
    pcy = pfy * CANVAS + rng.uniform(-8.0, 8.0)
    ccx = cfx * CANVAS + rng.uniform(-8.0, 8.0)
    ccy = cfy * CANVAS + rng.uniform(-8.0, 8.0)

    # --- signal plate ------------------------------------------------------
    plate_w = rng.uniform(94.0, 110.0)
    plate_h = rng.uniform(220.0, 250.0)
    frame = 5.0
    plate_fill = rng.choice((172, 180, 188, 196))
    lamp_count = rng.choice(BAR_COUNTS)
    lamp_scale = rng.uniform(30.0, 36.0)
    lamp_r = lamp_scale / math.sqrt(lamp_count)  # total lamp ink is pi*scale**2

    inner = frame + 7.0
    usable = plate_h - 2.0 * inner
    lamp_gap = (usable - lamp_count * 2.0 * lamp_r) / (lamp_count + 1)
    top = pcy - plate_h / 2.0 + inner
    lamps = []
    for index in range(lamp_count):
        cy = top + lamp_gap * (index + 1) + lamp_r * (2 * index + 1)
        lamps.append([_r(pcx), _r(cy), _r(lamp_r)])

    plate_bottom = pcy + plate_h / 2.0
    pole_len = min(34.0, CANVAS - 8.0 - plate_bottom)
    if pole_len < 10.0:
        pole_len = 0.0

    # --- crate block -------------------------------------------------------
    crate_w = rng.uniform(88.0, 102.0)
    crate_h = rng.uniform(64.0, 76.0)
    crate_gap = rng.uniform(13.0, 20.0)
    pad_x = rng.uniform(9.0, 14.0)
    pad_y = rng.uniform(9.0, 13.0)
    usable_w = crate_w - 2.0 * pad_x

    roll = rng.random()
    ceiling = min(9.0, (usable_w - 30.0) / 6.0)
    if ceiling < 3.4:
        ceiling = 3.4
    if roll < 0.25:  # deliberately near-boundary / quarantined
        gap_unit = rng.uniform(3.2, min(4.0, ceiling))
    elif roll < 0.55:  # near-boundary but off-quarantine
        gap_unit = rng.uniform(4.0, min(5.6, max(4.2, ceiling)))
    else:
        gap_unit = rng.uniform(min(5.6, ceiling), ceiling)
    gap_unit = max(3.2, min(gap_unit, ceiling))
    bar_ink_width = usable_w - 6.0 * gap_unit  # equals 5 * (bar width at n = 5)

    counts = list(BAR_COUNTS)
    rng.shuffle(counts)
    names = list(COLOUR_NAMES)
    rng.shuffle(names)
    shades = {name: rng.uniform(0.86, 1.1) for name in COLOUR_NAMES}

    crates = []
    for slot, (ox, oy) in enumerate(_CRATE_OFFSETS):
        cx = ccx + ox * (crate_w + crate_gap) / 2.0
        cy = ccy + oy * (crate_h + crate_gap) / 2.0
        count = counts[slot]
        name = names[slot]
        base = BASE_COLOURS[name]
        factor = shades[name]
        rgb = [int(round(min(255.0, channel * factor))) for channel in base]
        bar_w = bar_ink_width / count
        bar_gap = (usable_w - bar_ink_width) / (count + 1)
        left = cx - crate_w / 2.0 + pad_x
        bar_top = cy - crate_h / 2.0 + pad_y
        bar_bottom = cy + crate_h / 2.0 - pad_y
        bars = []
        for index in range(count):
            x0 = left + bar_gap * (index + 1) + bar_w * index
            bars.append([_r(x0), _r(bar_top), _r(x0 + bar_w), _r(bar_bottom)])
        crates.append(
            {
                "slot": slot,
                "colour": name,
                "rgb": rgb,
                "center": [_r(cx), _r(cy)],
                "w": _r(crate_w),
                "h": _r(crate_h),
                "bars": bars,
                "bar_gap": _r(bar_gap),
            }
        )

    return {
        "width": CANVAS,
        "height": CANVAS,
        "plate_slot": plate_slot,
        "crate_slot": crate_slot,
        "plate": {
            "center": [_r(pcx), _r(pcy)],
            "w": _r(plate_w),
            "h": _r(plate_h),
            "frame": _r(frame),
            "fill": [plate_fill, plate_fill, plate_fill],
        },
        "pole_len": _r(pole_len),
        "lamps": lamps,
        "crates": crates,
        "crate_gap": _r(crate_gap),
        "pad_y": _r(pad_y),
        "gap_unit": _r(gap_unit),
    }


def _rect(draw: ImageDraw.ImageDraw, box, scale: int, **kwargs) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle([x0 * scale, y0 * scale, x1 * scale, y1 * scale], **kwargs)


def render(scene: dict) -> Image.Image:
    """Rasterise a scene. Order within the crate list is irrelevant."""
    scale = SUPERSAMPLE
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)

    plate = scene["plate"]
    pcx, pcy = plate["center"]
    pw, ph = float(plate["w"]), float(plate["h"])
    frame = float(plate["frame"])
    if float(scene["pole_len"]) > 0.0:
        _rect(
            draw,
            (pcx - 7.0, pcy + ph / 2.0, pcx + 7.0, pcy + ph / 2.0 + float(scene["pole_len"])),
            scale,
            fill=FRAME,
        )
    _rect(
        draw,
        (pcx - pw / 2.0, pcy - ph / 2.0, pcx + pw / 2.0, pcy + ph / 2.0),
        scale,
        fill=tuple(plate["fill"]),
        outline=FRAME,
        width=int(round(frame * scale)),
    )
    for lamp_x, lamp_y, lamp_r in scene["lamps"]:
        draw.ellipse(
            [
                (lamp_x - lamp_r) * scale,
                (lamp_y - lamp_r) * scale,
                (lamp_x + lamp_r) * scale,
                (lamp_y + lamp_r) * scale,
            ],
            fill=DARK,
        )

    for crate in scene["crates"]:
        cx, cy = crate["center"]
        cw, ch = float(crate["w"]), float(crate["h"])
        _rect(
            draw,
            (cx - cw / 2.0, cy - ch / 2.0, cx + cw / 2.0, cy + ch / 2.0),
            scale,
            fill=tuple(crate["rgb"]),
            outline=FRAME,
            width=int(round(3 * scale)),
        )
        for x0, y0, x1, y1 in crate["bars"]:
            _rect(draw, (x0, y0, x1, y1), scale, fill=DARK)

    return image.resize((width, height), Image.BOX)


def analytic_gold(scene: dict) -> str:
    """Colour of the crate whose bar count equals the lamp count."""
    target = len(scene["lamps"])
    hits = [crate["colour"] for crate in scene["crates"] if len(crate["bars"]) == target]
    if len(hits) != 1:
        raise ValueError("scene does not have exactly one matching crate")
    return hits[0]


def margin(scene: dict) -> float:
    """Pixel clearance to the nearest visible configuration change."""
    clearances = []
    for crate in scene["crates"]:
        clearances.append(float(crate["bar_gap"]))  # two bars merging
        bars = crate["bars"]
        clearances.append(bars[0][2] - bars[0][0])  # a bar shrinking away
        clearances.append(float(scene["pad_y"]))  # a bar reaching the crate rim
    clearances.append(float(scene["crate_gap"]))  # two crates touching
    lamps = scene["lamps"]
    plate = scene["plate"]
    inner_top = plate["center"][1] - plate["h"] / 2.0 + plate["frame"] + 7.0
    inner_bottom = plate["center"][1] + plate["h"] / 2.0 - plate["frame"] - 7.0
    clearances.append(lamps[0][1] - lamps[0][2] - inner_top)
    clearances.append(inner_bottom - (lamps[-1][1] + lamps[-1][2]))
    clearances.append(plate["w"] / 2.0 - plate["frame"] - 7.0 - lamps[0][2])
    for first, second in zip(lamps, lamps[1:]):
        clearances.append((second[1] - second[2]) - (first[1] + first[2]))
    return _r(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the disjoint drawing lists."""
    out: list[tuple[str, dict]] = []
    crates = list(scene["crates"])
    lamps = list(scene["lamps"])

    reversed_crates = dict(scene)
    reversed_crates["crates"] = list(reversed(crates))
    out.append(("crate_list_reversed", reversed_crates))

    rotated_crates = dict(scene)
    rotated_crates["crates"] = crates[1:] + crates[:1]
    out.append(("crate_list_rotated", rotated_crates))

    reversed_lamps = dict(scene)
    reversed_lamps["lamps"] = list(reversed(lamps))
    out.append(("lamp_list_reversed", reversed_lamps))
    return out
