"""Deterministic renderer and analytic arm for a tangential orientation field."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


WIDTH = 480
HEIGHT = 480
COLOR_LABELS = ("red", "blue", "green", "orange")
PALETTES = (
    {
        "red": "#c73e4c",
        "blue": "#287bb5",
        "green": "#2d9d69",
        "orange": "#e58b2e",
        "dark": "#263238",
        "background": "#f7f4ed",
    },
    {
        "red": "#d14b57",
        "blue": "#337ab7",
        "green": "#269667",
        "orange": "#ed9638",
        "dark": "#303039",
        "background": "#fbf7ef",
    },
    {
        "red": "#bf3548",
        "blue": "#246fa8",
        "green": "#379f70",
        "orange": "#dc8128",
        "dark": "#302d2a",
        "background": "#f4f2ea",
    },
)
QUARANTINE_MARGIN = 10.0


def _scores(scene: dict) -> dict[str, float]:
    """Mean absolute radial alignment; zero means perfectly tangential."""
    scores: dict[str, float] = {}
    for candidate in scene["candidates"]:
        cx = float(candidate["x"])
        cy = float(candidate["y"])
        errors = []
        for segment in scene["segments"]:
            x1, y1 = float(segment["x1"]), float(segment["y1"])
            x2, y2 = float(segment["x2"]), float(segment["y2"])
            mx, my = (x1 + x2) / 2.0, (y1 + y2) / 2.0
            dx, dy = x2 - x1, y2 - y1
            rx, ry = mx - cx, my - cy
            dn = math.hypot(dx, dy)
            rn = math.hypot(rx, ry)
            if dn <= 1e-9 or rn <= 1e-9:
                errors.append(1.0)
            else:
                errors.append(abs((dx * rx + dy * ry) / (dn * rn)))
        scores[str(candidate["color"])] = sum(errors) / len(errors)
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return min(scores, key=lambda label: (scores[label], label))


def margin(scene: dict) -> float:
    ranked = sorted(_scores(scene).values())
    return 100.0 * (ranked[1] - ranked[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _make_segments(
    rng: random.Random,
    center: tuple[float, float],
    candidates: list[dict],
) -> list[dict]:
    cx, cy = center
    segments: list[dict] = []
    # Positions come from a jittered canvas-wide field independent of the answer.
    # Thus the dash centroid, envelope, density, and count do not reveal the center.
    cells = [(48.0 + 77.0 * col, 48.0 + 77.0 * row) for row in range(6) for col in range(6)]
    rng.shuffle(cells)
    for cell_x, cell_y in cells:
        mx = cell_x + rng.uniform(-10.0, 10.0)
        my = cell_y + rng.uniform(-10.0, 10.0)
        if any(
            math.hypot(mx - float(item["x"]), my - float(item["y"])) < 31.0
            for item in candidates
        ):
            continue
        theta = math.atan2(my - cy, mx - cx)
        length = rng.uniform(17.0, 22.0)
        # A small label-independent orientation perturbation prevents a rigid template.
        angle = theta + math.pi / 2.0 + rng.gauss(0.0, math.radians(1.15))
        ux, uy = math.cos(angle), math.sin(angle)
        x1, y1 = mx - length * ux / 2.0, my - length * uy / 2.0
        x2, y2 = mx + length * ux / 2.0, my + length * uy / 2.0
        if min(x1, x2, y1, y2) < 12:
            continue
        if max(x1, x2) > WIDTH - 12 or max(y1, y2) > HEIGHT - 12:
            continue
        segments.append(
            {
                "x1": x1,
                "y1": y1,
                "x2": x2,
                "y2": y2,
                "width": rng.choice((5.0, 5.5, 6.0)),
            }
        )
        if len(segments) == 24:
            break
    if len(segments) != 24:
        raise RuntimeError("could not populate the orientation field")
    return segments


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Consecutive seeds cycle the answer color."""
    rng = random.Random(int(seed))
    answer_color = COLOR_LABELS[int(seed) % len(COLOR_LABELS)]
    palette = PALETTES[rng.randrange(len(PALETTES))]
    slots = [
        (150.0 + rng.uniform(-5, 5), 150.0 + rng.uniform(-5, 5)),
        (330.0 + rng.uniform(-5, 5), 150.0 + rng.uniform(-5, 5)),
        (330.0 + rng.uniform(-5, 5), 330.0 + rng.uniform(-5, 5)),
        (150.0 + rng.uniform(-5, 5), 330.0 + rng.uniform(-5, 5)),
    ]
    rng.shuffle(slots)
    candidates = [
        {"color": label, "rgb": palette[label], "x": slots[i][0], "y": slots[i][1]}
        for i, label in enumerate(COLOR_LABELS)
    ]
    answer = next(item for item in candidates if item["color"] == answer_color)

    for _ in range(100):
        segments = _make_segments(rng, (float(answer["x"]), float(answer["y"])), candidates)
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "background": palette["background"],
            "dark_color": palette["dark"],
            "ring_radius": 12.0,
            "segments": segments,
            "candidates": candidates,
        }
        if analytic_gold(scene) == answer_color and margin(scene) >= 16.0:
            return scene
    raise RuntimeError("could not sample a decisive orientation field")


def render(scene: dict) -> Image.Image:
    """Render only the declared scene. Endpoint order has no raster effect."""
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    for segment in scene["segments"]:
        x1, y1 = float(segment["x1"]), float(segment["y1"])
        x2, y2 = float(segment["x2"]), float(segment["y2"])
        # Canonicalize an undirected primitive so its endpoint alias is byte-exact.
        if (x2, y2) < (x1, y1):
            x1, y1, x2, y2 = x2, y2, x1, y1
        draw.line(
            (
                x1 * scale,
                y1 * scale,
                x2 * scale,
                y2 * scale,
            ),
            fill=scene["dark_color"],
            width=max(1, int(round(float(segment["width"]) * scale))),
        )
    radius = float(scene["ring_radius"]) * scale
    for candidate in scene["candidates"]:
        cx, cy = float(candidate["x"]) * scale, float(candidate["y"]) * scale
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            outline=candidate["rgb"],
            width=6 * scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for segment in twin["segments"]:
        segment["x1"], segment["x2"] = segment["x2"], segment["x1"]
        segment["y1"], segment["y2"] = segment["y2"], segment["y1"]
    return [("swap_all_undirected_dash_endpoints", twin)]
