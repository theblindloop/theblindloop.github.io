"""Pixel-only inverse arm for the tangential orientation-flow world.

The oracle segments the four colored rings and every disconnected dark dash
directly from RGB pixels. It fits each dash by PCA, then selects the ring whose
radial vectors are most nearly perpendicular to the fitted dash axes.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


LABEL_ANCHORS = {
    "red": (199.0, 62.0, 76.0),
    "blue": (40.0, 123.0, 181.0),
    "green": (45.0, 157.0, 105.0),
    "orange": (229.0, 139.0, 46.0),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected components as arrays of (x, y) pixel centers."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        pixels = []
        while stack:
            py, px = stack.pop()
            pixels.append((px + 0.5, py + 0.5))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = py + dy, px + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= 24:
            components.append(np.asarray(pixels, dtype=np.float64))
    return components


def _ring_centers(arr: np.ndarray) -> dict[str, tuple[float, float]]:
    centers: dict[str, tuple[float, float]] = {}
    work = arr.astype(np.float64)
    for label, rgb in LABEL_ANCHORS.items():
        dist = np.linalg.norm(work - np.asarray(rgb), axis=2)
        ys, xs = np.nonzero(dist < 62.0)
        if len(xs) < 90:
            raise ValueError(f"colored ring {label} is missing")
        centers[label] = (float(xs.mean() + 0.5), float(ys.mean() + 0.5))
    return centers


def _dash_axes(arr: np.ndarray) -> list[tuple[float, float, float, float]]:
    # All allowed charcoal variants have every channel below 90 in their solid cores.
    dark = arr.max(axis=2) < 105
    axes = []
    for points in _components(dark):
        if len(points) < 45:
            continue
        center = points.mean(axis=0)
        centered = points - center
        covariance = centered.T @ centered
        values, vectors = np.linalg.eigh(covariance)
        direction = vectors[:, int(np.argmax(values))]
        major = float(values.max())
        minor = float(values.min())
        if major < 4.0 * max(minor, 1e-9):
            continue
        axes.append((float(center[0]), float(center[1]), float(direction[0]), float(direction[1])))
    if not 20 <= len(axes) <= 28:
        raise ValueError(f"expected a field of separate dashes, found {len(axes)}")
    return axes


def _scores(
    centers: dict[str, tuple[float, float]],
    axes: list[tuple[float, float, float, float]],
) -> dict[str, float]:
    scores: dict[str, float] = {}
    for label, (cx, cy) in centers.items():
        errors = []
        for mx, my, ux, uy in axes:
            rx, ry = mx - cx, my - cy
            norm = math.hypot(rx, ry)
            if norm <= 1e-9:
                errors.append(1.0)
            else:
                errors.append(abs((ux * rx + uy * ry) / norm))
        scores[label] = sum(errors) / len(errors)
    return scores


def decision_from_image(image: Image.Image) -> str:
    """Return a color label using only pixels from ``image``."""
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    centers = _ring_centers(arr)
    axes = _dash_axes(arr)
    scores = _scores(centers, axes)
    ranked = sorted(scores, key=lambda label: (scores[label], label))
    if 100.0 * (scores[ranked[1]] - scores[ranked[0]]) < 6.0:
        raise ValueError("ambiguous orientation-flow center")
    return ranked[0]
