"""Independent pixel-only inverse arm for the circumcenter-marker-selection world.

Recovers which color marker sits on the circumcenter of the triangle formed by
the three corner dots from the final PNG alone. It never sees the latent scene,
analytic gold, or renderer code. It segments the three distinct-colored corner
dots, computes their triangle's circumcenter as the intersection of two
perpendicular bisectors, then selects the colored marker nearest that point.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

BACKGROUND = (255, 255, 255)
CORNER_COLORS = {
    "brown": (139, 69, 19),
    "darkcyan": (0, 139, 139),
    "gray": (88, 88, 96),
}
MARKER_COLORS = {
    "red": (230, 50, 50),
    "green": (50, 190, 90),
    "blue": (50, 90, 230),
    "purple": (170, 60, 200),
}
MARKER_NAMES = ["red", "green", "blue", "purple"]

_TH = 40


def _mask(arr, rgb):
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    return (
        (np.abs(r - rgb[0]) < _TH) & (np.abs(g - rgb[1]) < _TH) & (np.abs(b - rgb[2]) < _TH)
    )


def _center(mask: np.ndarray) -> tuple[float, float] | None:
    coords = np.argwhere(mask)
    if len(coords) < 20:
        return None
    return (float(coords[:, 1].mean()), float(coords[:, 0].mean()))


def _circumcenter(a, b, c) -> tuple[float, float]:
    ax, ay = a
    bx, by = b
    cx, cy = c
    d = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(d) < 1e-9:
        return ((ax + bx + cx) / 3.0, (ay + by + cy) / 3.0)
    ux = (
        (ax * ax + ay * ay) * (by - cy)
        + (bx * bx + by * by) * (cy - ay)
        + (cx * cx + cy * cy) * (ay - by)
    ) / d
    uy = (
        (ax * ax + ay * ay) * (cx - bx)
        + (bx * bx + by * by) * (ax - cx)
        + (cx * cx + cy * cy) * (bx - ax)
    ) / d
    return (float(ux), float(uy))


def decision_from_image(image: Image.Image) -> str:
    """The color name of the marker sitting on the triangle's circumcenter."""
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)

    corners = []
    for color in CORNER_COLORS:
        center = _center(_mask(arr, CORNER_COLORS[color]))
        if center is not None:
            corners.append(center)
    if len(corners) < 3:
        return MARKER_NAMES[0]

    ox, oy = _circumcenter(*corners[:3])

    best_color = None
    best_dist = math.inf
    for color in MARKER_NAMES:
        center = _center(_mask(arr, MARKER_COLORS[color]))
        if center is None:
            continue
        dist = math.hypot(center[0] - ox, center[1] - oy)
        if dist < best_dist:
            best_dist = dist
            best_color = color

    if best_color is None:
        return MARKER_NAMES[0]
    return best_color
