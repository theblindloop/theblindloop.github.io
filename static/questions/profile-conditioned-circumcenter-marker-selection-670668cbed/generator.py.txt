"""Circumcenter-marker-selection world.

Three corner dots, each in a distinct color (brown, dark-cyan, gray), mark the
vertices of a triangle on a white canvas. The single point equally distant from
all three corners -- the triangle's circumcenter -- is the shared center of the
one circle that passes through all three vertices. A set of colored marker dots
(red, green, blue, purple) is drawn; exactly one of them sits exactly on that
circumcenter. The decision is which color that marker is.

The global Gestalt structure is the circumcenter of the triangle formed by the
whole corner triplet (an equidistance / circle-through-three-points grouping).
The renderer owns latent-scene sampling, rasterization, and the structural
analytic gold. The independent pixel-only inverse arm (``oracle.py``) recovers
the same decision from the final PNG alone: it segments the three distinct
corner dots, computes their triangle's circumcenter as the intersection of two
perpendicular bisectors, and selects the colored marker nearest that point.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)

# Three distinct, visibly separate corner colors (one dot each, so every color
# mask maps to exactly one dot centre).
CORNER_COLORS = {
    "brown": (139, 69, 19),
    "darkcyan": (0, 139, 139),
    "gray": (88, 88, 96),
}
CORNER_NAMES = ["brown", "darkcyan", "gray"]
CORNER_RADIUS = 9

# Corner dots sit on a base circle centred at the canvas centre; mild angle
# jitter keeps the triangle acute so the circumcenter stays near the centre.
BASE_CENTER = (256.0, 256.0)
CORNER_RADIUS_MIN = 120.0
CORNER_RADIUS_MAX = 165.0
ANGLE_JITTER = 0.38  # rad

MARKER_RADIUS = 10
MARKER_COLORS = {
    "red": (230, 50, 50),
    "green": (50, 190, 90),
    "blue": (50, 90, 230),
    "purple": (170, 60, 200),
}
MARKER_NAMES = ["red", "green", "blue", "purple"]

# Distractor markers sit between these distances (px) from the circumcenter.
_DIST_MIN = 45.0
_DIST_MAX = 140.0

# Quarantine band on the decision margin (px from the recovered circumcenter to
# the nearest distractor marker; the true marker is right on the circumcenter).
QUAR_MIN = 60.0


def _unit(angle: float) -> tuple[float, float]:
    return (math.cos(angle), math.sin(angle))


def _circumcenter(a, b, c) -> tuple[float, float]:
    """Return the circumcenter of three non-collinear points."""
    ax, ay = a
    bx, by = b
    cx, cy = c
    d = 2.0 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
    if abs(d) < 1e-9:
        return ((ax + bx + cx) / 3.0, (ay + by + cy) / 3.0)
    ux = (
        (ax * ax + ay * ay) * (by - cy)
        + (bx * bx + by * by) * (cy - ay)
        + (cx * cx + cy * cy) * (ay - by)
    ) / d
    uy = (
        (ax * ax + ay * ay) * (cx - bx)
        + (bx * bx + by * by) * (ax - cx)
        + (cx * cx + cy * cy) * (bx - ax)
    ) / d
    return (float(ux), float(uy))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)

    r = float(rng.uniform(CORNER_RADIUS_MIN, CORNER_RADIUS_MAX))
    a0 = float(rng.uniform(0.0, 2.0 * math.pi))
    corner_angles = [
        float(a0 + 2.0 * math.pi * k / 3.0 + float(rng.uniform(-ANGLE_JITTER, ANGLE_JITTER)))
        for k in range(3)
    ]
    cx, cy = BASE_CENTER
    corners = []
    for angle in corner_angles:
        ux, uy = _unit(angle)
        corners.append([cx + r * ux, cy + r * uy])

    circumcenter = [float(v) for v in _circumcenter(*corners)]

    # Ordered corner colors (one per corner dot; purely geometric, never read by
    # the analytic gold).
    corner_colors = list(CORNER_NAMES)
    rng.shuffle(corner_colors)

    # Ordered marker colors; markers[0] is the true marker placed on the
    # circumcenter. This is the only field the analytic gold reads.
    markers = list(MARKER_NAMES)
    rng.shuffle(markers)

    # Three distractor markers: distances and directions away from circumcenter.
    dist_vals = [float(rng.uniform(_DIST_MIN, _DIST_MAX)) for _ in range(3)]
    dirs = [float(rng.uniform(0.0, 2.0 * math.pi)) for _ in range(3)]
    distractors = []
    for d, a in zip(dist_vals, dirs):
        dx, dy = _unit(a)
        distractors.append([circumcenter[0] + d * dx, circumcenter[1] + d * dy])

    return {
        "circumcenter": circumcenter,
        "corner_angles": corner_angles,
        "corner_radius": r,
        "corners": corners,
        "corner_colors": corner_colors,
        "markers": markers,
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic RGB PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    d = ImageDraw.Draw(img)
    ox, oy = scene["circumcenter"]

    # Three distinct-colored corner dots at the triangle vertices.
    for (cx, cy), color in zip(scene["corners"], scene["corner_colors"]):
        d.ellipse(
            [cx - CORNER_RADIUS, cy - CORNER_RADIUS, cx + CORNER_RADIUS, cy + CORNER_RADIUS],
            fill=CORNER_COLORS[color],
        )

    # Distractor markers (colors markers[1:]), then the true marker last on top
    # at the circumcenter so it always covers exactly that point.
    for (mx, my), color in zip(scene["distractors"], scene["markers"][1:]):
        d.ellipse(
            [mx - MARKER_RADIUS, my - MARKER_RADIUS, mx + MARKER_RADIUS, my + MARKER_RADIUS],
            fill=MARKER_COLORS[color],
        )
    true_color = scene["markers"][0]
    d.ellipse(
        [ox - MARKER_RADIUS, oy - MARKER_RADIUS, ox + MARKER_RADIUS, oy + MARKER_RADIUS],
        fill=MARKER_COLORS[true_color],
    )
    return img


def analytic_gold(scene: dict) -> str:
    """Structural gold: the color name of the marker sitting on the circumcenter."""
    return scene["markers"][0]


def margin(scene: dict) -> float:
    """Decision margin: distance from the circumcenter to the nearest distractor."""
    ox, oy = scene["circumcenter"]
    dists = [math.hypot(mx - ox, my - oy) for (mx, my) in scene["distractors"]]
    return float(min(dists))


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes whose nearest distractor is too close to the circumcenter."""
    return margin(scene) < QUAR_MIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
