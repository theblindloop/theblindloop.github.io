"""Deterministic renderer and analytic arm for relative uncertainty."""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

PALETTE = {
    "blue": (31, 119, 180),
    "orange": (230, 110, 20),
    "green": (38, 145, 80),
}


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(int(seed))
    names = ["blue", "orange", "green"]
    winner = names[int(seed) % 3]
    remaining = [name for name in names if name != winner]
    rng.shuffle(remaining)
    near = int(seed) % 11 == 0
    ratios = {
        winner: float(rng.uniform(0.74, 0.79) if near else rng.uniform(0.88, 1.08)),
        remaining[0]: float(rng.uniform(0.61, 0.64) if near else rng.uniform(0.51, 0.66)),
        remaining[1]: float(rng.uniform(0.28, 0.43)),
    }
    zero_x = int(rng.integers(72, 101))
    rows = [122, 220, 318]
    rng.shuffle(rows)
    colors = names.copy()
    rng.shuffle(colors)
    intervals: list[list[int]] = []
    centers: list[int] = []
    for name in colors:
        # Distance and ratio are independently varied, preventing raw width or
        # absolute position from being a reliable answer shortcut.
        distance = int(rng.integers(190, 391))
        center = zero_x + distance
        width = int(round(ratios[name] * distance))
        width += width % 2
        intervals.append([center - width // 2, center + width // 2])
        centers.append(center)
    return {
        "canvas_width": 760,
        "canvas_height": 400,
        "zero_x": zero_x,
        "rows": rows,
        "colors": colors,
        "intervals": intervals,
        "centers": centers,
    }


def _ratios(scene: dict) -> dict[str, float]:
    zero = float(scene["zero_x"])
    intervals = scene["intervals"]
    centers = scene["centers"]
    colors = scene["colors"]
    return {
        colors[i]: (float(intervals[i][1]) - float(intervals[i][0]))
        / abs(float(centers[i]) - zero)
        for i in range(3)
    }


def analytic_gold(scene: dict) -> str:
    values = _ratios(scene)
    return max(values, key=values.get)


def margin(scene: dict) -> float:
    values = sorted(_ratios(scene).values(), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.10


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    scale = 2
    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    zero_x = int(scene["zero_x"])
    rows = scene["rows"]
    colors = scene["colors"]
    intervals = scene["intervals"]
    centers = scene["centers"]
    image = Image.new("RGB", (width * scale, height * scale), (250, 251, 253))
    draw = ImageDraw.Draw(image)

    # A pale plotting field and subtle row guides provide a figure-like world.
    draw.rounded_rectangle(
        (36 * scale, 52 * scale, (width - 28) * scale, (height - 30) * scale),
        radius=12 * scale,
        fill=(255, 255, 255),
        outline=(222, 226, 232),
        width=2 * scale,
    )
    for y in rows:
        draw.line((48 * scale, y * scale, (width - 46) * scale, y * scale), fill=(237, 240, 244), width=2 * scale)

    # The shared, explicitly named zero reference is dashed and visually distinct.
    for y0 in range(70, height - 42, 18):
        draw.line(
            (zero_x * scale, y0 * scale, zero_x * scale, min(y0 + 10, height - 42) * scale),
            fill=(70, 74, 82),
            width=3 * scale,
        )
    font = ImageFont.load_default(size=18 * scale)
    draw.text((max(42, zero_x - 20) * scale, 24 * scale), "ZERO", fill=(70, 74, 82), font=font)

    for i, name in enumerate(colors):
        y = int(rows[i])
        left, right = (int(v) for v in intervals[i])
        center = int(centers[i])
        color = PALETTE[name]
        draw.line((left * scale, y * scale, right * scale, y * scale), fill=color, width=6 * scale)
        draw.line((left * scale, (y - 8) * scale, left * scale, (y + 8) * scale), fill=color, width=5 * scale)
        draw.line((right * scale, (y - 8) * scale, right * scale, (y + 8) * scale), fill=color, width=5 * scale)
        draw.ellipse(
            ((center - 12) * scale, (y - 12) * scale, (center + 12) * scale, (y + 12) * scale),
            fill=color,
            outline=(255, 255, 255),
            width=2 * scale,
        )

    return image.resize((width, height), Image.Resampling.LANCZOS)
