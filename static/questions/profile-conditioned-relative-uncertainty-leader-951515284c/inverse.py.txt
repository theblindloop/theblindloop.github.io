"""Independent RGB-only inverse arm for the relative-uncertainty figure."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "blue": (31, 119, 180),
    "orange": (230, 110, 20),
    "green": (38, 145, 80),
}


def _mask_near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float = 62.0) -> np.ndarray:
    delta = arr.astype(np.float32) - np.asarray(rgb, dtype=np.float32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _zero_column(arr: np.ndarray) -> float:
    h, _ = arr.shape[:2]
    neutral_dark = (
        (arr[:, :, 0] < 125)
        & (arr[:, :, 1] < 125)
        & (arr[:, :, 2] < 125)
        & (np.max(arr, axis=2) - np.min(arr, axis=2) < 24)
    )
    neutral_dark[:64, :] = False
    neutral_dark[max(0, h - 38) :, :] = False
    counts = neutral_dark.sum(axis=0)
    if int(counts.max()) < 45:
        raise ValueError("visible dashed zero line was not recovered")
    peak = np.flatnonzero(counts >= counts.max() * 0.82)
    return float(np.mean(peak))


def _measure_color(arr: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float]:
    mask = _mask_near(arr, rgb)
    ys, xs = np.nonzero(mask)
    if len(xs) < 260:
        raise ValueError("colored interval has insufficient visible pixels")
    left = float(np.percentile(xs, 0.15))
    right = float(np.percentile(xs, 99.85))
    span = right - left
    if span < 55:
        raise ValueError("colored interval is too short to measure")
    column_counts = mask.sum(axis=0)
    lo = max(0, int(left) + 13)
    hi = min(mask.shape[1], int(right) - 12)
    interior = column_counts[lo:hi]
    if interior.size == 0 or int(interior.max()) < 14:
        raise ValueError("circular center marker was not recovered")
    marker_columns = np.flatnonzero(interior >= max(13, interior.max() - 3)) + lo
    weights = column_counts[marker_columns].astype(float)
    center = float(np.average(marker_columns, weights=weights))
    return span, center


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 200 or arr.shape[1] < 400:
        raise ValueError("image is not a supported measurement figure")
    zero = _zero_column(arr)
    ratios: dict[str, float] = {}
    for name, rgb in PALETTE.items():
        span, center = _measure_color(arr, rgb)
        distance = abs(center - zero)
        if distance < 40:
            raise ValueError("center marker is unresolved from zero")
        ratios[name] = span / distance
    ordered = sorted(ratios, key=ratios.get, reverse=True)
    if ratios[ordered[0]] - ratios[ordered[1]] < 0.055:
        raise ValueError("pixel decision is ambiguous")
    return ordered[0]
