from __future__ import annotations

import random
from collections import Counter

from PIL import Image, ImageDraw


COLORS = {
    "orange": (238, 126, 34),
    "purple": (145, 72, 196),
    "blue": (42, 112, 210),
}
BG = (247, 245, 239)
DARK = (35, 38, 45)
GREEN = (35, 166, 94)
RED = (211, 55, 57)

TEMPLATES = [
    [(68, 78), (300, 78), (300, 182), (132, 182), (132, 322), (424, 322), (424, 218), (568, 218)],
    [(72, 338), (274, 338), (274, 236), (116, 236), (116, 92), (402, 92), (402, 264), (566, 264)],
    [(66, 116), (238, 116), (238, 294), (388, 294), (388, 156), (520, 156), (520, 336), (574, 336)],
    [(72, 286), (202, 286), (202, 106), (352, 106), (352, 246), (494, 246), (494, 92), (570, 92)],
]


def _second_positions(markers: list[dict]) -> dict[str, int]:
    seen: Counter[str] = Counter()
    result: dict[str, int] = {}
    for index, marker in enumerate(markers):
        color = marker["color"]
        seen[color] += 1
        if seen[color] == 2:
            result[color] = index
    return result


def _route_locations(points: list[list[int]], fractions: list[float]) -> list[tuple[float, float, float, float]]:
    lengths = []
    total = 0.0
    for a, b in zip(points, points[1:]):
        length = abs(b[0] - a[0]) + abs(b[1] - a[1])
        lengths.append(length)
        total += length
    output = []
    for fraction in fractions:
        target = fraction * total
        walked = 0.0
        for (x1, y1), (x2, y2), length in zip(points, points[1:], lengths):
            if walked + length >= target:
                t = (target - walked) / length
                x, y = x1 + t * (x2 - x1), y1 + t * (y2 - y1)
                dx, dy = x2 - x1, y2 - y1
                norm = max(abs(dx), abs(dy))
                output.append((x, y, -dy / norm, dx / norm))
                break
            walked += length
    return output


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = ("orange", "purple", "blue")[int(seed) % 3]
    sequence = None
    for _ in range(500):
        trial = ["orange", "orange", "purple", "purple", "blue", "blue"]
        rng.shuffle(trial)
        positions = _second_positions([{"color": value} for value in trial])
        ordered = sorted(positions, key=positions.get)
        if ordered[0] == target and positions[ordered[1]] - positions[ordered[0]] >= 1:
            sequence = trial
            break
    if sequence is None:
        raise RuntimeError("could not construct balanced sequence")

    points = [list(p) for p in TEMPLATES[rng.randrange(len(TEMPLATES))]]
    # Small whole-route translations vary absolute position without changing topology.
    tx, ty = rng.choice([-5, 0, 5]), rng.choice([-5, 0, 5])
    points = [[x + tx, y + ty] for x, y in points]
    fractions = [0.105, 0.255, 0.405, 0.565, 0.715, 0.865]
    locations = _route_locations(points, fractions)
    markers = []
    for index, (color, (x, y, nx, ny)) in enumerate(zip(sequence, locations)):
        side = -1 if index % 2 else 1
        markers.append({
            "x": round(x + side * 18 * nx, 2),
            "y": round(y + side * 18 * ny, 2),
            "attach_x": round(x, 2),
            "attach_y": round(y, 2),
            "color": color,
        })
    return {
        "points": points,
        "markers": markers,
        "start": points[0],
        "end": points[-1],
        "canvas": [640, 420],
    }


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (width, height), BG)
    draw = ImageDraw.Draw(image)
    points = [tuple(p) for p in scene["points"]]
    draw.line(points, fill=DARK, width=9, joint="curve")
    for x, y in points[1:-1]:
        draw.ellipse((x - 4, y - 4, x + 4, y + 4), fill=DARK)

    for marker in scene["markers"]:
        color = COLORS[marker["color"]]
        x, y = marker["x"], marker["y"]
        ax, ay = marker["attach_x"], marker["attach_y"]
        draw.line((ax, ay, x, y), fill=color, width=3)
        r = 8
        draw.polygon([(x, y-r), (x+r, y), (x, y+r), (x-r, y)], fill=color)

    sx, sy = scene["start"]
    ex, ey = scene["end"]
    draw.ellipse((sx-13, sy-13, sx+13, sy+13), outline=GREEN, width=6)
    draw.rectangle((ex-11, ey-11, ex+11, ey+11), fill=RED, outline=(255, 255, 255), width=2)
    # Preserve cable continuity through the endpoint markers while leaving most
    # of each marker visible.
    draw.line((points[0], points[1]), fill=DARK, width=5)
    return image


def analytic_gold(scene: dict) -> str:
    positions = _second_positions(scene["markers"])
    if len(positions) != 3:
        raise ValueError("every checkpoint color must occur twice")
    return min(positions, key=positions.get)


def margin(scene: dict) -> float:
    positions = sorted(_second_positions(scene["markers"]).values())
    return float(positions[1] - positions[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
