"""Deterministic renderer for colored chord-crossing correspondence."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

W, H = 900, 700
SCALE = 3
SEGMENT_COLORS = ["cyan", "magenta", "green", "orange"]
RGB = {
    "cyan": (0, 150, 200), "magenta": (196, 55, 150),
    "green": (30, 155, 85), "orange": (235, 125, 30),
    "red": (210, 48, 55), "blue": (45, 95, 205), "gold": (210, 160, 25),
}
PAIRING = [(0, 4), (1, 6), (2, 7), (3, 5)]


def _cross(a, b, c, d):
    def orient(p, q, r):
        return (q[0]-p[0])*(r[1]-p[1]) - (q[1]-p[1])*(r[0]-p[0])
    return orient(a,b,c)*orient(a,b,d) < 0 and orient(c,d,a)*orient(c,d,b) < 0


def _signature(order):
    pts = [(math.cos(2*math.pi*i/8), math.sin(2*math.pi*i/8)) for i in range(8)]
    by_color = {}
    for chord_idx, color in enumerate(order):
        a, b = PAIRING[chord_idx]
        by_color[color] = (pts[a], pts[b])
    out = []
    for i, c1 in enumerate(SEGMENT_COLORS):
        for c2 in SEGMENT_COLORS[i+1:]:
            if _cross(*by_color[c1], *by_color[c2]):
                out.append(f"{c1}-{c2}")
    return sorted(out)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    base = SEGMENT_COLORS[:]
    rng.shuffle(base)
    target_sig = _signature(base)
    perms = []
    allp = []
    import itertools
    for p in itertools.permutations(SEGMENT_COLORS):
        p = list(p)
        if _signature(p) != target_sig and _signature(p) not in [_signature(x) for x in allp]:
            allp.append(p)
    rng.shuffle(allp)
    perms = [base, allp[0], allp[1]]
    rng.shuffle(perms)
    borders = ["red", "blue", "gold"]
    rng.shuffle(borders)
    panels = [base] + perms
    transforms = []
    for _ in panels:
        transforms.append({"rotation": rng.uniform(-0.7, 0.7), "reflection": bool(rng.randrange(2)), "scale": rng.uniform(0.92, 1.04)})
    scene = {"seed": int(seed), "pairing": [list(x) for x in PAIRING], "panel_color_orders": panels,
             "panel_transforms": transforms, "border_colors": borders}
    return scene


def _panel_specs():
    return [(450, 190, 235, 155), (160, 505, 205, 145), (450, 505, 205, 145), (740, 505, 205, 145)]


def _points(cx, cy, radius, transform):
    sign = -1 if transform["reflection"] else 1
    rot = transform["rotation"]
    return [(cx + radius*transform["scale"]*math.cos(rot + sign*2*math.pi*i/8),
             cy + radius*transform["scale"]*math.sin(rot + sign*2*math.pi*i/8)) for i in range(8)]


def render(scene: dict) -> Image.Image:
    im = Image.new("RGB", (W*SCALE, H*SCALE), "white")
    d = ImageDraw.Draw(im)
    specs = _panel_specs()
    for idx, ((cx,cy,pw,ph), order, trans) in enumerate(zip(specs, scene["panel_color_orders"], scene["panel_transforms"])):
        box = [(cx-pw/2)*SCALE, (cy-ph/2)*SCALE, (cx+pw/2)*SCALE, (cy+ph/2)*SCALE]
        fill = (242,242,242)
        border = (35,35,35) if idx == 0 else RGB[scene["border_colors"][idx-1]]
        d.rounded_rectangle(box, radius=12*SCALE, fill=fill, outline=border, width=7*SCALE)
        pts = _points(cx, cy, 60 if idx == 0 else 54, trans)
        for a,b in PAIRING:
            d.line([tuple(v*SCALE for v in pts[a]), tuple(v*SCALE for v in pts[b])], fill=(75,75,75), width=5*SCALE)
        for chord_idx, color in enumerate(order):
            for endpoint in PAIRING[chord_idx]:
                x,y = pts[endpoint]
                r=9
                d.ellipse([(x-r)*SCALE,(y-r)*SCALE,(x+r)*SCALE,(y+r)*SCALE], fill=RGB[color], outline=(255,255,255), width=2*SCALE)
    d.text((388*SCALE, 18*SCALE), "REFERENCE", fill=(25,25,25), stroke_width=0)
    d.text((397*SCALE, 332*SCALE), "OPTIONS", fill=(25,25,25), stroke_width=0)
    return im.resize((W,H), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    ref = _signature(scene["panel_color_orders"][0])
    hits = [i for i,p in enumerate(scene["panel_color_orders"][1:]) if _signature(p) == ref]
    if len(hits) != 1:
        raise ValueError("non-unique match")
    return scene["border_colors"][hits[0]]


def margin(scene: dict) -> float:
    ref = set(_signature(scene["panel_color_orders"][0]))
    distances = sorted(len(ref.symmetric_difference(set(_signature(p)))) for p in scene["panel_color_orders"][1:])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 1.0


def latent_symmetries(scene: dict):
    other = __import__("copy").deepcopy(scene)
    other["panel_transforms"][0]["rotation"] += 2 * math.pi
    return [("reference_rotation_plus_full_turn", other)]
