"""Independent pixel-only inverse for colored chord crossings."""

from __future__ import annotations

import numpy as np

SEGMENT_RGB = {
    "cyan": (0, 150, 200), "magenta": (196, 55, 150),
    "green": (30, 155, 85), "orange": (235, 125, 30),
}
BORDER_RGB = {"red": (210, 48, 55), "blue": (45, 95, 205), "gold": (210, 160, 25)}


def _components(mask, minimum=20):
    h, w = mask.shape
    seen = np.zeros((h,w), dtype=bool)
    out=[]
    for y,x in zip(*np.nonzero(mask & ~seen)):
        if seen[y,x]:
            continue
        stack=[(int(y),int(x))]; seen[y,x]=True; pts=[]
        while stack:
            yy,xx=stack.pop(); pts.append((yy,xx))
            for ny,nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0<=ny<h and 0<=nx<w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx]=True; stack.append((ny,nx))
        if len(pts)>=minimum:
            a=np.asarray(pts)
            out.append((len(pts), float(a[:,1].mean()), float(a[:,0].mean()),
                        int(a[:,1].min()),int(a[:,1].max()),int(a[:,0].min()),int(a[:,0].max())))
    return out


def _cross(a,b,c,d):
    def o(p,q,r): return (q[0]-p[0])*(r[1]-p[1])-(q[1]-p[1])*(r[0]-p[0])
    return o(a,b,c)*o(a,b,d)<0 and o(c,d,a)*o(c,d,b)<0


def _panel_signature(arr, box):
    x0,x1,y0,y1=box
    crop=arr[y0:y1+1,x0:x1+1]
    endpoints={}
    for name,rgb in SEGMENT_RGB.items():
        dist=np.linalg.norm(crop.astype(float)-np.asarray(rgb),axis=2)
        comps=sorted(_components(dist<28, 55), reverse=True)[:2]
        if len(comps)!=2:
            raise ValueError(f"expected two {name} endpoints, got {len(comps)}")
        endpoints[name]=[(c[1]+x0,c[2]+y0) for c in comps]
    colors=list(SEGMENT_RGB)
    sig=[]
    for i,a in enumerate(colors):
        for b in colors[i+1:]:
            if _cross(*endpoints[a],*endpoints[b]): sig.append(f"{a}-{b}")
    return tuple(sorted(sig))


def decision_from_image(image):
    arr=np.asarray(image.convert("RGB"))
    # The filled panel cards are solid non-white components. This discovers
    # their raster bounds without importing or duplicating renderer coordinates.
    panel_ink=np.linalg.norm(arr.astype(float)-np.asarray((255,255,255)),axis=2)>8
    regions=sorted(_components(panel_ink, 6000), reverse=True)[:4]
    if len(regions)!=4: raise ValueError(f"expected four panels, got {len(regions)}")
    panels=[]
    for _,cx,cy,x0,x1,y0,y1 in regions:
        panels.append({"cx":cx,"cy":cy,"box":(max(0,x0-12),min(arr.shape[1]-1,x1+12),max(0,y0-12),min(arr.shape[0]-1,y1+12))})
    ref=min(panels,key=lambda p:p["cy"])
    opts=sorted([p for p in panels if p is not ref],key=lambda p:p["cx"])
    target=_panel_signature(arr,ref["box"])
    hits=[]
    for p in opts:
        if _panel_signature(arr,p["box"])==target: hits.append(p)
    if len(hits)!=1: raise ValueError(f"expected unique matching option, got {len(hits)}")
    p=hits[0]; x0,x1,y0,y1=p["box"]; crop=arr[y0:y1+1,x0:x1+1]
    scores={name:int((np.linalg.norm(crop.astype(float)-np.asarray(rgb),axis=2)<35).sum()) for name,rgb in BORDER_RGB.items()}
    ranked=sorted(scores,key=scores.get,reverse=True)
    if scores[ranked[0]]<250 or scores[ranked[0]]<2*scores[ranked[1]]:
        raise ValueError("border color is ambiguous")
    return ranked[0]
