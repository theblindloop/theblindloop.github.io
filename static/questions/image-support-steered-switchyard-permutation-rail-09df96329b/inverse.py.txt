"""Independent pixel-only inverse arm for the adjacent-swap switchyard."""

from __future__ import annotations

import numpy as np
from PIL import Image

TAG_RGB = {
    "gold": np.array((226, 174, 38), dtype=np.int16),
    "red": np.array((214, 60, 60), dtype=np.int16),
    "green": np.array((45, 157, 91), dtype=np.int16),
    "blue": np.array((53, 105, 210), dtype=np.int16),
}
TOKEN_RGB = np.array((112, 191, 201), dtype=np.int16)


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    results = []
    height, width = mask.shape
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        stack = [(int(sy), int(sx))]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        results.append(np.asarray(points, dtype=np.int32))
    return results


def _centroid_for(array: np.ndarray, rgb: np.ndarray, name: str) -> tuple[float, float]:
    delta = np.linalg.norm(array.astype(np.int16) - rgb, axis=2)
    comps = [c for c in _components(delta < 18.0) if len(c) >= 20]
    if len(comps) != 1:
        raise OracleAbstention(f"expected one intact {name} dock")
    y, x = comps[0].mean(axis=0)
    return float(x), float(y)


def _token_parts(array: np.ndarray) -> list[tuple[float, float, str]]:
    delta = np.linalg.norm(array.astype(np.int16) - TOKEN_RGB, axis=2)
    comps = [c for c in _components(delta < 18.0) if len(c) >= 35]
    if len(comps) != 12:
        raise OracleAbstention(f"expected twelve intact shape tokens, saw {len(comps)}")
    tokens = []
    for comp in comps:
        y0, x0 = comp.min(axis=0)
        y1, x1 = comp.max(axis=0)
        width, height, area = int(x1-x0+1), int(y1-y0+1), len(comp)
        density = area / float(width * height)
        if density > 0.92:
            shape = "square"
        elif density > 0.72:
            shape = "circle"
        else:
            shape = "triangle"
        y, x = comp.mean(axis=0)
        tokens.append((float(x), float(y), shape))
    return tokens


def _groups(values: np.ndarray) -> list[np.ndarray]:
    if not len(values):
        return []
    cuts = np.where(np.diff(values) > 1)[0] + 1
    return [v for v in np.split(values, cuts) if len(v)]


def _read_gates(dark: np.ndarray, gold: tuple[float, float], x_min: int, x_max: int) -> list[int]:
    bx, by = gold
    y = int(round(by))
    direction = 1 if bx < (x_min + x_max) / 2 else -1
    candidates = []
    for x in range(x_min + 88, x_max - 10):
        above = int(dark[max(0,y-12):max(0,y-3), x].sum())
        below = int(dark[min(dark.shape[0],y+4):min(dark.shape[0],y+13), x].sum())
        if max(above, below) >= 5:
            candidates.append(x)
    groups = _groups(np.asarray(candidates, dtype=np.int32))
    if len(groups) != 6:
        raise OracleAbstention(f"expected six intact swap gates, saw {len(groups)}")
    observed = []
    for group in groups:
        x = int(round(float(group.mean())))
        above = int(dark[max(0,y-12):max(0,y-3), group].sum())
        below = int(dark[min(dark.shape[0],y+4):min(dark.shape[0],y+13), group].sum())
        if above == below:
            raise OracleAbstention("swap gate side is ambiguous")
        observed.append((x, 0 if above > below else 1))
    observed.sort(key=lambda item: direction * (item[0] - bx))
    return [gate for _, gate in observed]


def _execute(gates: list[int]) -> list[str]:
    order = ["circle", "square", "triangle"]
    for gate in gates:
        order[gate], order[gate+1] = order[gate+1], order[gate]
    return order


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    tags = {name: _centroid_for(array, rgb, name) for name, rgb in TAG_RGB.items()}
    ys = sorted(round(point[1]) for point in tags.values())
    if len(ys) != 4 or any(b-a < 15 for a,b in zip(ys,ys[1:])):
        raise OracleAbstention("docks do not define four separated rail runs")
    dark = np.all(array < 100, axis=2)
    points = np.argwhere(dark)
    if len(points) < 1300:
        raise OracleAbstention("connected switchyard ink is missing")
    x_min, x_max = int(points[:,1].min()), int(points[:,1].max())
    if x_max - x_min < 235:
        raise OracleAbstention("switchyard rail is truncated")
    target = _execute(_read_gates(dark, tags["gold"], x_min, x_max))
    tokens = _token_parts(array)
    distances = {}
    for name in ("red", "green", "blue"):
        bx, by = tags[name]
        direction = 1 if bx < (x_min + x_max) / 2 else -1
        row_tokens = [(x, shape) for x,y,shape in tokens if abs(y-by) < 5]
        if len(row_tokens) != 3:
            raise OracleAbstention(f"{name} dock lacks three intact tokens")
        row_tokens.sort(key=lambda item: direction * (item[0] - bx))
        order = [shape for _,shape in row_tokens]
        distances[name] = sum(a != b for a,b in zip(target, order, strict=True))
    winners = [name for name,value in distances.items() if value == min(distances.values())]
    if len(winners) != 1 or distances[winners[0]] != 0 or sorted(distances.values())[1] < 2:
        raise OracleAbstention("pixels do not yield one separated exact dock match")
    return winners[0]
