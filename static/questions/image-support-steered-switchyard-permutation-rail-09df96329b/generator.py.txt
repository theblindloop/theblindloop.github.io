"""Latent sampler and raster renderer for the adjacent-swap switchyard."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

WIDTH = 768
HEIGHT = 768
RUN_LENGTH = 250
ROW_OFFSETS = (0, 24, 48, 72)
START_TOKEN_DISTANCES = (25, 48, 71)
GATE_DISTANCES = (105, 130, 155, 180, 205, 230)
CANDIDATE_TOKEN_DISTANCES = (65, 130, 195)
ANSWER_COLORS = ("red", "green", "blue")
SHAPES = ("circle", "square", "triangle")
RGB = {
    "red": (214, 60, 60),
    "green": (45, 157, 91),
    "blue": (53, 105, 210),
    "gold": (226, 174, 38),
}
TOKEN_FILL = (112, 191, 201)
PALETTES = (
    ((28, 34, 40), (246, 247, 242)),
    ((24, 47, 57), (248, 245, 239)),
    ((49, 35, 47), (244, 248, 246)),
)


def _execute(gates: list[int]) -> list[str]:
    order = list(SHAPES)
    for gate in gates:
        index = 0 if int(gate) == 0 else 1
        order[index], order[index + 1] = order[index + 1], order[index]
    return order


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C4D)
    gates = [rng.randrange(2) for _ in range(6)]
    target = _execute(gates)
    answer = ANSWER_COLORS[int(seed) % 3]
    dock_colors = list(ANSWER_COLORS)
    rng.shuffle(dock_colors)
    target_row = dock_colors.index(answer)
    alternatives = [list(p) for p in itertools.permutations(SHAPES) if list(p) != target]
    distractors = rng.sample(alternatives, 2)
    dock_orders = []
    distractor_index = 0
    for row in range(3):
        if row == target_row:
            dock_orders.append(list(target))
        else:
            dock_orders.append(distractors[distractor_index])
            distractor_index += 1
    x0 = rng.choice((54, 246, 438)) + rng.randint(-3, 3)
    y0 = rng.choice((72, 264, 456, 648)) + rng.randint(-2, 2)
    ink, background = rng.choice(PALETTES)
    return {
        "gates": gates,
        "dock_orders": dock_orders,
        "dock_colors": dock_colors,
        "x0": x0,
        "y0": y0,
        "ink_color": list(ink),
        "background": list(background),
    }


def _distances(scene: dict) -> list[int]:
    target = _execute([int(v) for v in scene["gates"]])
    return [sum(a != b for a, b in zip(target, order, strict=True)) for order in scene["dock_orders"]]


def analytic_gold(scene: dict) -> str:
    distances = _distances(scene)
    return str(scene["dock_colors"][min(range(3), key=distances.__getitem__)])


def margin(scene: dict) -> float:
    values = sorted(_distances(scene))
    return float(values[1] - values[0])


def is_quarantined(scene: dict) -> bool:
    values = _distances(scene)
    return values.count(0) != 1 or sorted(values)[1] < 2


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_token(draw: ImageDraw.ImageDraw, x: int, y: int, shape: str, ink: tuple[int, int, int]) -> None:
    if shape == "circle":
        draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=TOKEN_FILL, outline=ink, width=2)
    elif shape == "square":
        draw.rectangle((x - 8, y - 8, x + 8, y + 8), fill=TOKEN_FILL, outline=ink, width=2)
    elif shape == "triangle":
        draw.polygon(((x, y - 9), (x - 9, y + 8), (x + 9, y + 8)), fill=TOKEN_FILL, outline=ink, width=2)
    else:
        raise ValueError(shape)


def render(scene: dict) -> Image.Image:
    background = tuple(int(v) for v in scene["background"])
    ink = tuple(int(v) for v in scene["ink_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), background)
    draw = ImageDraw.Draw(image)
    x0 = int(scene["x0"])
    x1 = x0 + RUN_LENGTH
    ys = [int(scene["y0"]) + offset for offset in ROW_OFFSETS]

    for y in ys:
        draw.line((x0, y, x1, y), fill=ink, width=3)
    draw.line((x1, ys[0], x1, ys[1]), fill=ink, width=3)
    draw.line((x0, ys[1], x0, ys[2]), fill=ink, width=3)
    draw.line((x1, ys[2], x1, ys[3]), fill=ink, width=3)

    starts = [x0, x1, x0, x1]
    directions = [1, -1, 1, -1]
    for distance, gate in zip(GATE_DISTANCES, scene["gates"], strict=True):
        x = starts[0] + directions[0] * distance
        page_sign = -1 if int(gate) == 0 else 1
        draw.line((x, ys[0], x, ys[0] + page_sign * 10), fill=ink, width=3)

    for distance, shape in zip(START_TOKEN_DISTANCES, SHAPES, strict=True):
        _draw_token(draw, starts[0] + directions[0] * distance, ys[0], shape, ink)
    for row, order in enumerate(scene["dock_orders"], start=1):
        for distance, shape in zip(CANDIDATE_TOKEN_DISTANCES, order, strict=True):
            _draw_token(draw, starts[row] + directions[row] * distance, ys[row], shape, ink)

    for name, x, y in zip(("gold", *scene["dock_colors"]), starts, ys, strict=True):
        draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=RGB[name], outline=ink, width=1)
    return image
