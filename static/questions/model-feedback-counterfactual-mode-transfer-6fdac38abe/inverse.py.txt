"""Independent PNG-only inverse for the mode-transfer visual grammar."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 900
HEIGHT = 780
SOURCE_CENTERS = {
    "S": (80, 105), "T": (330, 105), "U": (570, 105), "V": (820, 105),
}
GATE_CENTERS = {
    "J": (180, 245), "K": (450, 245), "L": (720, 245),
    "M": (300, 405), "N": (600, 405), "P": (360, 565), "Q": (660, 565),
}
PARENTS = {
    "J": ("S", "T"), "K": ("T", "U"), "L": ("U", "V"),
    "M": ("J", "K"), "N": ("K", "L"),
    "P": ("M", "N"), "Q": ("N", "L"),
}
EDGES = tuple((parent, child) for child, parents in PARENTS.items() for parent in parents)
GATE_IDS = tuple(GATE_CENTERS)
TEST_NODES = ("J", "K", "L")

GATE_RGB = {"XOR": (36, 116, 216), "AND": (242, 169, 0), "OR": (135, 84, 200)}
NODE_RGB = {
    "J": (216, 74, 91), "K": (0, 141, 153), "L": (58, 157, 77),
    "M": (209, 63, 165), "N": (229, 104, 30), "P": (74, 77, 179),
    "Q": (138, 90, 43),
}
SOURCE_RGB = {
    "S": (216, 74, 91), "T": (0, 141, 153),
    "U": (58, 157, 77), "V": (135, 84, 200),
}
INK_RGB = (23, 33, 43)
WHITE_RGB = (255, 255, 255)


class OracleError(ValueError):
    pass


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _recover_bit(arr: np.ndarray, center: tuple[int, int], label: str) -> int:
    cx, cy = center
    patch = arr[cy - 13:cy + 14, cx - 13:cx + 14]
    dark = int(_near(patch, INK_RGB, 48.0).sum())
    light = int(_near(patch, WHITE_RGB, 24.0).sum())
    if dark >= 590 and light <= 60:
        return 1
    if light >= 590 and dark <= 60:
        return 0
    raise OracleError(f"{label} bit is missing or ambiguous")


def _recover_gate(arr: np.ndarray, center: tuple[int, int], label: str) -> str:
    cx, cy = center
    patch = arr[cy - 28:cy + 29, cx - 45:cx - 5]
    scores = {kind: int(_near(patch, rgb, 55.0).sum()) for kind, rgb in GATE_RGB.items()}
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] < 1150 or ranked[1][1] > 160:
        raise OracleError(f"gate {label} type is missing or ambiguous: {scores}")
    return ranked[0][0]


def _validate_source_rims(arr: np.ndarray) -> None:
    for name, (cx, cy) in SOURCE_CENTERS.items():
        patch = arr[cy - 29:cy + 30, cx - 29:cx + 30]
        score = int(_near(patch, SOURCE_RGB[name], 52.0).sum())
        if score < 420:
            raise OracleError(f"source identity rim {name} is missing")


def _validate_node_rims(arr: np.ndarray) -> None:
    for name, (cx, cy) in GATE_CENTERS.items():
        patch = arr[cy - 45:cy + 46, cx - 59:cx + 60]
        score = int(_near(patch, NODE_RGB[name], 52.0).sum())
        if score < 1350:
            raise OracleError(f"node identity rim {name} is missing")


def _validate_wires(arr: np.ndarray) -> None:
    ink = _near(arr, INK_RGB, 68.0)
    centers = {**SOURCE_CENTERS, **GATE_CENTERS}
    for parent, child in EDGES:
        x0, y0 = centers[parent]
        x1, y1 = centers[child]
        mx = int(round(x0 * 0.48 + x1 * 0.52))
        my = int(round(y0 * 0.48 + y1 * 0.52))
        if int(ink[my - 7:my + 8, mx - 7:mx + 8].sum()) < 28:
            raise OracleError(f"wire {parent}->{child} is missing")


def _recover_test(arr: np.ndarray) -> tuple[str, int]:
    patch = arr[707:746, 334:480]
    scores = {node: int(_near(patch, NODE_RGB[node], 52.0).sum()) for node in TEST_NODES}
    ranked = sorted(scores.items(), key=lambda item: item[1], reverse=True)
    if ranked[0][1] < 3600 or ranked[1][1] > 180:
        raise OracleError(f"TEST node binding is missing or ambiguous: {scores}")
    return ranked[0][0], _recover_bit(arr, (548, 726), "TEST")


def _gate(kind: str, left: int, right: int) -> int:
    if kind == "XOR":
        return int(left != right)
    if kind == "AND":
        return int(left == 1 and right == 1)
    if kind == "OR":
        return int(left == 1 or right == 1)
    raise OracleError(f"unknown gate type {kind!r}")


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    _validate_wires(arr)
    _validate_source_rims(arr)
    _validate_node_rims(arr)

    actual = {
        name: _recover_bit(arr, center, f"source {name}")
        for name, center in SOURCE_CENTERS.items()
    }
    kinds = {
        name: _recover_gate(arr, center, name)
        for name, center in GATE_CENTERS.items()
    }
    for name, (cx, cy) in GATE_CENTERS.items():
        actual[name] = _recover_bit(arr, (cx + 31, cy + 9), f"node {name}")

    if set(actual[name] for name in SOURCE_CENTERS) != {0, 1}:
        raise OracleError("ACTUAL sources are not visibly varied")
    if set(kinds.values()) != {"XOR", "AND", "OR"}:
        raise OracleError("ACTUAL circuit does not visibly use all gate types")

    modes: dict[str, int] = {}
    for node in GATE_IDS:
        left, right = PARENTS[node]
        modes[node] = actual[node] ^ _gate(kinds[node], actual[left], actual[right])
    if not 2 <= sum(modes.values()) <= 5:
        raise OracleError("ACTUAL trace does not encode a varied mode set")

    test_node, test_bit = _recover_test(arr)
    if test_bit != 1 - actual[test_node]:
        raise OracleError("TEST does not visibly oppose the selected node's ACTUAL bit")

    tested = {name: actual[name] for name in SOURCE_CENTERS}
    for node in GATE_IDS:
        left, right = PARENTS[node]
        value = _gate(kinds[node], tested[left], tested[right]) ^ modes[node]
        tested[node] = test_bit if node == test_node else value

    changed = sum(actual[node] != tested[node] for node in GATE_IDS)
    if changed < 3 or (actual["P"], actual["Q"]) == (tested["P"], tested["Q"]):
        raise OracleError("TEST intervention has no valid visible downstream effect")
    return f"{tested['P']}{tested['Q']}"
