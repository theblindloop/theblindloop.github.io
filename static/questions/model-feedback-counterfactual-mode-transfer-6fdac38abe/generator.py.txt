"""Deterministic visual world for abducting modes before intervention."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 900
HEIGHT = 780
SCALE = 3

SOURCE_CENTERS = {
    "S": (80, 105),
    "T": (330, 105),
    "U": (570, 105),
    "V": (820, 105),
}
GATE_CENTERS = {
    "J": (180, 245),
    "K": (450, 245),
    "L": (720, 245),
    "M": (300, 405),
    "N": (600, 405),
    "P": (360, 565),
    "Q": (660, 565),
}
PARENTS = {
    "J": ("S", "T"),
    "K": ("T", "U"),
    "L": ("U", "V"),
    "M": ("J", "K"),
    "N": ("K", "L"),
    "P": ("M", "N"),
    "Q": ("N", "L"),
}
EDGES = tuple((parent, child) for child, parents in PARENTS.items() for parent in parents)
GATE_IDS = tuple(GATE_CENTERS)
GATE_TYPES = ("XOR", "AND", "OR")
TEST_NODES = ("J", "K", "L")
OPTIONS = ("00", "01", "10", "11")

GATE_COLORS = {
    "XOR": "#2474D8",
    "AND": "#F2A900",
    "OR": "#8754C8",
}
NODE_COLORS = {
    "J": "#D84A5B",
    "K": "#008D99",
    "L": "#3A9D4D",
    "M": "#D13FA5",
    "N": "#E5681E",
    "P": "#4A4DB3",
    "Q": "#8A5A2B",
}
SOURCE_COLORS = {
    "S": "#D84A5B",
    "T": "#008D99",
    "U": "#3A9D4D",
    "V": "#8754C8",
}
INK = "#17212B"
GRAY = "#66727E"
RED = "#D5262F"
PALE = "#F4F6F8"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


def _gate(kind: str, left: int, right: int) -> int:
    if kind == "XOR":
        return int(left != right)
    if kind == "AND":
        return int(left == 1 and right == 1)
    if kind == "OR":
        return int(left == 1 or right == 1)
    raise ValueError(f"unknown gate type {kind!r}")


def _evaluate(
    scene: dict[str, Any], override: tuple[str, int] | None = None
) -> dict[str, int]:
    values = {name: int(bit) for name, bit in zip(SOURCE_CENTERS, scene["source_bits"])}
    kinds = {str(name): str(kind) for name, kind in scene["gate_types"].items()}
    modes = {str(name): int(bit) for name, bit in scene["mode_bits"].items()}
    forced_node = None if override is None else str(override[0])
    forced_bit = None if override is None else int(override[1])
    if forced_node is not None and (forced_node not in TEST_NODES or forced_bit not in (0, 1)):
        raise ValueError("invalid intervention")
    for node in GATE_IDS:
        left, right = PARENTS[node]
        value = _gate(kinds[node], values[left], values[right]) ^ modes[node]
        values[node] = forced_bit if node == forced_node else value
    return values


def _test_values(scene: dict[str, Any]) -> dict[str, int]:
    actual = _evaluate(scene)
    node = str(scene["test_node"])
    return _evaluate(scene, (node, 1 - actual[node]))


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    desired = OPTIONS[int(seed) % len(OPTIONS)]
    for _ in range(50000):
        source_bits = [rng.randrange(2) for _ in SOURCE_CENTERS]
        gate_types = {name: rng.choice(GATE_TYPES) for name in GATE_IDS}
        mode_bits = {name: rng.randrange(2) for name in GATE_IDS}
        test_node = rng.choice(TEST_NODES)
        scene: dict[str, Any] = {
            "source_bits": source_bits,
            "gate_types": gate_types,
            "mode_bits": mode_bits,
            "test_node": test_node,
        }
        if set(source_bits) != {0, 1} or set(gate_types.values()) != set(GATE_TYPES):
            continue
        inverted = sum(mode_bits.values())
        if not 2 <= inverted <= 5:
            continue
        actual = _evaluate(scene)
        tested = _test_values(scene)
        answer = f"{tested['P']}{tested['Q']}"
        changed = sum(actual[node] != tested[node] for node in GATE_IDS)
        if answer != desired or changed < 3:
            continue
        if (actual["P"], actual["Q"]) == (tested["P"], tested["Q"]):
            continue
        return scene
    raise RuntimeError("could not sample a balanced abduct-then-intervene scene")


def analytic_gold(scene: dict[str, Any]) -> str:
    if is_quarantined(scene):
        raise ValueError("scene does not expose a valid intervention")
    tested = _test_values(scene)
    return f"{tested['P']}{tested['Q']}"


def margin(scene: dict[str, Any]) -> float:
    return 14.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        sources = list(scene["source_bits"])
        kinds = dict(scene["gate_types"])
        modes = dict(scene["mode_bits"])
        node = str(scene["test_node"])
        if (
            len(sources) != 4
            or any(type(bit) is not int or bit not in (0, 1) for bit in sources)
            or set(sources) != {0, 1}
            or set(kinds) != set(GATE_IDS)
            or any(kind not in GATE_TYPES for kind in kinds.values())
            or set(kinds.values()) != set(GATE_TYPES)
            or set(modes) != set(GATE_IDS)
            or any(type(bit) is not int or bit not in (0, 1) for bit in modes.values())
            or not 2 <= sum(modes.values()) <= 5
            or node not in TEST_NODES
        ):
            return True
        actual = _evaluate(scene)
        tested = _test_values(scene)
        changed = sum(actual[name] != tested[name] for name in GATE_IDS)
        return changed < 3 or (actual["P"], actual["Q"]) == (tested["P"], tested["Q"])
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _center_text(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    text: str,
    font: ImageFont.FreeTypeFont,
    fill: str,
) -> None:
    cx, cy = center
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text(
        (int(round(cx * SCALE - width / 2)), int(round(cy * SCALE - height / 2 - SCALE))),
        text,
        fill=fill,
        font=font,
    )


def _arrow(
    draw: ImageDraw.ImageDraw,
    start: tuple[int, int],
    end: tuple[int, int],
    start_pad: float,
    end_pad: float,
) -> None:
    dx = float(end[0] - start[0])
    dy = float(end[1] - start[1])
    length = math.hypot(dx, dy)
    ux, uy = dx / length, dy / length
    p0 = (start[0] + ux * start_pad, start[1] + uy * start_pad)
    tip = (end[0] - ux * end_pad, end[1] - uy * end_pad)
    base = (tip[0] - ux * 12, tip[1] - uy * 12)
    normal = (-uy, ux)
    left = (base[0] + normal[0] * 7, base[1] + normal[1] * 7)
    right = (base[0] - normal[0] * 7, base[1] - normal[1] * 7)
    sp = lambda point: (int(round(point[0] * SCALE)), int(round(point[1] * SCALE)))
    draw.line((*sp(p0), *sp(base)), fill=INK, width=5 * SCALE)
    draw.polygon([sp(tip), sp(left), sp(right)], fill=INK)


def _draw_bit(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    bit: int,
    outline: str,
    half: int = 22,
) -> None:
    cx, cy = center
    sx = lambda value: int(round(value * SCALE))
    draw.rounded_rectangle(
        (sx(cx - half), sx(cy - half), sx(cx + half), sx(cy + half)),
        radius=sx(6),
        fill=INK if bit else "white",
        outline=outline,
        width=sx(5),
    )


def render(scene: dict[str, Any]) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    actual = _evaluate(scene)
    kinds = {str(name): str(kind) for name, kind in scene["gate_types"].items()}
    test_node = str(scene["test_node"])
    test_bit = 1 - actual[test_node]

    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(canvas)
    sx = lambda value: int(round(value * SCALE))

    draw.text((sx(15), sx(10)), "ACTUAL", fill=INK, font=_font(20))
    draw.text((sx(125), sx(14)), "every square is a readout", fill=GRAY, font=_font(12))
    for x, kind, meaning in ((430, "XOR", "different"), (585, "AND", "both"), (725, "OR", "either")):
        draw.rounded_rectangle(
            (sx(x), sx(10), sx(x + 54), sx(40)), radius=sx(5),
            fill=GATE_COLORS[kind], outline=INK, width=sx(2),
        )
        _center_text(draw, (x + 27, 25), kind, _font(10), INK if kind == "AND" else "white")
        draw.text((sx(x + 59), sx(16)), meaning, fill=GRAY, font=_font(9))

    centers = {**SOURCE_CENTERS, **GATE_CENTERS}
    for parent, child in EDGES:
        _arrow(
            draw,
            centers[parent],
            centers[child],
            29.0 if parent in SOURCE_CENTERS else 58.0,
            58.0,
        )

    for name, (cx, cy) in SOURCE_CENTERS.items():
        _draw_bit(draw, (cx, cy), actual[name], SOURCE_COLORS[name], half=24)
        _center_text(draw, (cx, cy - 42), name, _font(16), SOURCE_COLORS[name])

    for name, (cx, cy) in GATE_CENTERS.items():
        kind = kinds[name]
        node_color = NODE_COLORS[name]
        draw.rounded_rectangle(
            (sx(cx - 57), sx(cy - 43), sx(cx + 57), sx(cy + 43)),
            radius=sx(12), fill=PALE, outline=node_color, width=sx(6),
        )
        draw.rounded_rectangle(
            (sx(cx - 48), sx(cy - 31), sx(cx + 1), sx(cy + 31)),
            radius=sx(7), fill=GATE_COLORS[kind], outline=INK, width=sx(2),
        )
        _center_text(draw, (cx - 24, cy - 13), name, _font(17), INK if kind == "AND" else "white")
        _center_text(draw, (cx - 24, cy + 14), kind, _font(10), INK if kind == "AND" else "white")
        draw.text((sx(cx + 12), sx(cy - 34)), "BIT", fill=GRAY, font=_font(8))
        _draw_bit(draw, (cx + 31, cy + 9), actual[name], node_color, half=21)

    draw.rounded_rectangle(
        (sx(220), sx(650), sx(680), sx(765)), radius=sx(15),
        fill="#FFF7F7", outline=RED, width=sx(5),
    )
    draw.text((sx(242), sx(663)), "TEST", fill=RED, font=_font(18))
    draw.text((sx(330), sx(668)), "force this node; ignore incoming arrows", fill=GRAY, font=_font(11))
    draw.rounded_rectangle(
        (sx(327), sx(703), sx(487), sx(749)), radius=sx(9),
        fill=NODE_COLORS[test_node], outline=INK, width=sx(2),
    )
    _center_text(draw, (407, 726), f"FORCE {test_node} =", _font(16), "white")
    _draw_bit(draw, (548, 726), test_bit, NODE_COLORS[test_node], half=24)
    draw.text((sx(586), sx(716)), "same modes", fill=GRAY, font=_font(10))

    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
