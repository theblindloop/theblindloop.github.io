"""Independent image-only inverse arm for radial-order badge search."""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = np.asarray(
    [(22, 164, 184), (235, 158, 38), (126, 79, 181), (54, 151, 91)],
    dtype=np.int16,
)
EXPECTED = (0, 1, 2, 3)  # cyan, amber, violet, green from center outward


def _axis_clusters(projection: np.ndarray) -> list[tuple[int, int]]:
    """Find separated badge columns/rows, merging damage gaps inside a badge."""
    active = np.flatnonzero(projection)
    if not len(active):
        return []
    cuts = np.flatnonzero(np.diff(active) > 60)
    groups = np.split(active, cuts + 1)
    return [(int(group[0]), int(group[-1])) for group in groups if len(group) >= 8]


def _components(mask: np.ndarray) -> list[tuple[int, np.ndarray]]:
    x_groups = _axis_clusters(mask.any(axis=0))
    y_groups = _axis_clusters(mask.any(axis=1))
    output = []
    for row_index, (y0, y1) in enumerate(y_groups[:2]):
        for column_index, (x0, x1) in enumerate(x_groups[:2]):
            local_y, local_x = np.where(mask[y0 : y1 + 1, x0 : x1 + 1])
            if len(local_y) >= 900:
                points = np.column_stack((local_y + y0, local_x + x0)).astype(np.int32)
                output.append((2 * row_index + column_index, points))
    return output


def _badge_record(nearest: np.ndarray, close: np.ndarray, points: np.ndarray) -> tuple[float, float, int, float]:
    ys = points[:, 0]
    xs = points[:, 1]
    cy = float((ys.min() + ys.max()) / 2.0)
    cx = float((xs.min() + xs.max()) / 2.0)
    radius = max(1.0, min(float(xs.max() - xs.min()), float(ys.max() - ys.min())) / 2.0)
    pad = int(radius) + 3
    y0, y1 = max(0, int(cy) - pad), min(nearest.shape[0], int(cy) + pad + 1)
    x0, x1 = max(0, int(cx) - pad), min(nearest.shape[1], int(cx) + pad + 1)
    yy, xx = np.indices((y1 - y0, x1 - x0))
    rr = np.sqrt((xx + x0 - cx) ** 2 + (yy + y0 - cy) ** 2) / radius
    local_nearest = nearest[y0:y1, x0:x1]
    local_close = close[y0:y1, x0:x1]
    bands = ((0.00, 0.25), (0.28, 0.49), (0.52, 0.73), (0.76, 0.96))
    observed: list[int] = []
    confidences: list[float] = []
    for low, high in bands:
        region = (rr >= low) & (rr <= high) & local_close
        counts = np.bincount(local_nearest[region], minlength=4)
        if counts.sum() < 25:
            observed.append(-1)
            confidences.append(0.0)
        else:
            observed.append(int(np.argmax(counts)))
            confidences.append(float(counts.max() / counts.sum()))
    score = sum(value == expected for value, expected in zip(observed, EXPECTED, strict=True))
    confidence = min(confidences) if confidences else 0.0
    return cy, cx, score, confidence


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    distances = np.linalg.norm(array.astype(np.int16)[:, :, None, :] - PALETTE[None, None, :, :], axis=3)
    colored = np.min(distances, axis=2) < 18.0
    nearest = np.argmin(distances, axis=2)
    records = [(index, _badge_record(nearest, colored, points)) for index, points in _components(colored)]
    if len(records) < 3:
        raise ValueError("too few intact badge candidates")
    scores = [-1, -1, -1, -1]
    for index, record in records:
        scores[index] = record[2] if record[3] >= 0.72 else -1
    winners = [index for index, score in enumerate(scores) if score == 4]
    runner = sorted(scores, reverse=True)[1] if len(scores) >= 2 else 4
    if len(winners) != 1 or 4 - runner < 2:
        raise ValueError("no unique radial-order winner with runner-up margin")
    return ("A", "B", "C", "D")[winners[0]]
