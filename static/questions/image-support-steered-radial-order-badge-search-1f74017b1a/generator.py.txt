"""Deterministic renderer and analytic arm for radial-order badge search."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 246, 240)
INK = (34, 39, 46)
COLORS = {
    "cyan": (22, 164, 184),
    "amber": (235, 158, 38),
    "violet": (126, 79, 181),
    "green": (54, 151, 91),
}
TARGET = ["cyan", "amber", "violet", "green"]
DISTRACTORS = [
    ["cyan", "amber", "green", "violet"],
    ["cyan", "violet", "amber", "green"],
    ["green", "amber", "violet", "cyan"],
]
RADII_OUTWARD = [12, 23, 34, 45]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    base_centers = [(146, 137), (366, 137), (146, 357), (366, 357)]
    centers = [[x + rng.randint(-13, 13), y + rng.randint(-13, 13)] for x, y in base_centers]
    target_index = int(seed) % 4
    distractors = [list(value) for value in DISTRACTORS]
    rng.shuffle(distractors)
    orders: list[list[str]] = []
    cursor = 0
    for index in range(4):
        if index == target_index:
            orders.append(list(TARGET))
        else:
            orders.append(distractors[cursor])
            cursor += 1
    return {"centers": centers, "orders": orders, "labels": ["A", "B", "C", "D"]}


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    centers = scene["centers"]
    orders = scene["orders"]
    labels = scene["labels"]
    for center, order, label in zip(centers, orders, labels, strict=True):
        cx, cy = (int(center[0]), int(center[1]))
        # Draw outside inward so every declared color occupies one thick annulus/core.
        for radius, color_name in reversed(list(zip(RADII_OUTWARD, order, strict=True))):
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                fill=COLORS[color_name],
            )
        draw.ellipse((cx - 47, cy - 47, cx + 47, cy + 47), outline=INK, width=2)
        box = draw.textbbox((0, 0), str(label), font=font)
        tw = box[2] - box[0]
        draw.rounded_rectangle((cx - 15, cy + 57, cx + 15, cy + 79), radius=5, fill=INK)
        draw.text((cx - tw / 2, cy + 63), str(label), font=font, fill=(255, 255, 255))
    return image


def _scores(scene: dict[str, Any]) -> list[int]:
    return [sum(left == right for left, right in zip(order, TARGET, strict=True)) for order in scene["orders"]]


def analytic_gold(scene: dict[str, Any]) -> str:
    orders = scene["orders"]
    labels = scene["labels"]
    scores = [sum(left == right for left, right in zip(order, TARGET, strict=True)) for order in orders]
    best = max(scores)
    winners = [index for index, score in enumerate(scores) if score == best]
    if best != 4 or len(winners) != 1:
        raise ValueError("scene has no unique exact radial-order winner")
    return str(labels[winners[0]])


def margin(scene: dict[str, Any]) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict[str, Any]) -> bool:
    scores = _scores(scene)
    return scores.count(max(scores)) != 1 or max(scores) != 4 or margin(scene) < 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
