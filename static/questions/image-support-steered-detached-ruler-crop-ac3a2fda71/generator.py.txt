"""Deterministic latent sampler and renderer for Detached Ruler Crop."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 800
CELL = 32
N = 7
BOARD = CELL * N
SYMBOLS = ("plus", "X", "T", "L")
TEMPLATES = {
    "plus": (
        "00100", "00100", "11111", "00100", "00100",
    ),
    "X": (
        "10001", "01010", "00100", "01010", "10001",
    ),
    "T": (
        "11111", "00100", "00100", "00100", "00100",
    ),
    "L": (
        "10000", "10000", "10000", "10000", "11111",
    ),
}
FRAME = (68, 72, 78)
BACKGROUND = (249, 248, 244)
PALETTE = (
    (20, 126, 153),
    (211, 82, 67),
    (119, 76, 169),
    (37, 116, 194),
    (207, 133, 25),
    (35, 146, 91),
)


def _template_bits(name: str) -> list[list[int]]:
    return [[int(ch) for ch in row] for row in TEMPLATES[name]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    delete_rows = sorted(rng.sample(range(N), 2))
    delete_cols = sorted(rng.sample(range(N), 2))
    keep_rows = [i for i in range(N) if i not in delete_rows]
    keep_cols = [i for i in range(N) if i not in delete_cols]
    # Exactly twelve of the 24 deleted cells are dark. Since every template has
    # nine dark cells, total board ink is fixed at 21 across all answers.
    junk_positions = [
        [r, c]
        for r in range(N)
        for c in range(N)
        if r in delete_rows or c in delete_cols
    ]
    junk_cells = rng.sample(junk_positions, 12)
    palette = list(PALETTE)
    rng.shuffle(palette)
    return {
        "symbol": symbol,
        "delete_rows": delete_rows,
        "delete_cols": delete_cols,
        "junk_cells": junk_cells,
        "cluster_x": 300 + rng.randint(-22, 22),
        "cluster_y": 340 + rng.randint(-5, 5),
        "palette": [list(color) for color in palette[:3]],
    }


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    target = _template_bits(str(scene["symbol"]))
    distances = []
    for name in SYMBOLS:
        if name == scene["symbol"]:
            continue
        other = _template_bits(name)
        distances.append(sum(target[r][c] != other[r][c] for r in range(5) for c in range(5)))
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["junk_cells"] = list(reversed(twin["junk_cells"]))
    return [("reverse_set_serialization", twin)]


def _board_bits(scene: dict) -> list[list[int]]:
    bits = [[0 for _ in range(N)] for _ in range(N)]
    keep_rows = [i for i in range(N) if i not in scene["delete_rows"]]
    keep_cols = [i for i in range(N) if i not in scene["delete_cols"]]
    template = _template_bits(str(scene["symbol"]))
    for rr, r in enumerate(keep_rows):
        for cc, c in enumerate(keep_cols):
            bits[r][c] = template[rr][cc]
    for r, c in scene["junk_cells"]:
        bits[int(r)][int(c)] = 1
    return bits


def _draw_grid(draw: ImageDraw.ImageDraw, x0: int, y0: int, rows: int, cols: int) -> None:
    draw.rectangle((x0, y0, x0 + cols * CELL, y0 + rows * CELL), outline=FRAME, width=4)
    for index in range(1, cols):
        x = x0 + index * CELL
        draw.line((x, y0, x, y0 + rows * CELL), fill=FRAME, width=3)
    for index in range(1, rows):
        y = y0 + index * CELL
        draw.line((x0, y, x0 + cols * CELL, y), fill=FRAME, width=3)


def _fill_cell(draw: ImageDraw.ImageDraw, x0: int, y0: int, row: int, col: int, color: tuple[int, int, int]) -> None:
    pad = 7
    left = x0 + col * CELL + pad
    top = y0 + row * CELL + pad
    draw.rounded_rectangle((left, top, left + CELL - 2 * pad, top + CELL - 2 * pad), radius=3, fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    bx = int(scene["cluster_x"])
    by = int(scene["cluster_y"])
    # The horizontal ruler and board are separated by an inactive row even
    # under the public 8x8 probe's 25-pixel ablation halo.
    hx, hy = bx, by - 270
    vx, vy = bx - 58, by
    _draw_grid(draw, bx, by, N, N)
    _draw_grid(draw, hx, hy, 1, N)
    _draw_grid(draw, vx, vy, N, 1)
    board_color, col_color, row_color = (tuple(v) for v in scene["palette"])
    bits = _board_bits(scene)
    for r in range(N):
        for c in range(N):
            if bits[r][c]:
                _fill_cell(draw, bx, by, r, c, board_color)
    for c in scene["delete_cols"]:
        _fill_cell(draw, hx, hy, 0, int(c), col_color)
    for r in scene["delete_rows"]:
        _fill_cell(draw, vx, vy, int(r), 0, row_color)
    return image
