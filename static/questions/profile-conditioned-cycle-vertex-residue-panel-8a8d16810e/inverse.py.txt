"""Independent pixel-only inverse arm for cycle-vertex residue groups."""

from __future__ import annotations

import numpy as np

CYAN = np.array([38, 176, 190])
LINK = np.array([105, 112, 120])
LABELS = ("left", "center", "right")


def _components(mask: np.ndarray, minimum: int = 90) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            result.append((sum(xs) / len(xs), sum(ys) / len(ys), len(xs)))
    return result


def _has_link(rgb: np.ndarray, first: tuple[float, float], second: tuple[float, float]) -> bool:
    x1, y1 = first
    x2, y2 = second
    distance = float(np.hypot(x2 - x1, y2 - y1))
    # The two closest irregular-decagon slots are about 50 px apart; station
    # disks still leave a clear 20 px exposed band between their rims.
    if distance < 40:
        return False
    hits = 0
    total = 0
    start = 19.0 / distance
    for t in np.linspace(start, 1.0 - start, max(24, int(distance - 34))):
        x = int(round(x1 + t * (x2 - x1)))
        y = int(round(y1 + t * (y2 - y1)))
        patch = rgb[max(0, y - 2):y + 3, max(0, x - 2):x + 3]
        if patch.size:
            delta = np.max(np.abs(patch.astype(int) - LINK), axis=2)
            hits += int(np.any(delta <= 5))
            total += 1
    return total > 0 and hits / total > 0.79


def _graph_components(adjacency: list[set[int]]) -> list[set[int]]:
    unseen = set(range(len(adjacency)))
    result = []
    while unseen:
        start = unseen.pop()
        component = {start}
        stack = [start]
        while stack:
            vertex = stack.pop()
            for other in adjacency[vertex]:
                if other in unseen:
                    unseen.remove(other)
                    component.add(other)
                    stack.append(other)
        result.append(component)
    return result


def _cycle_vertices(adjacency: list[set[int]]) -> set[int]:
    alive = set(range(len(adjacency)))
    queue = [vertex for vertex in alive if len(adjacency[vertex]) < 2]
    while queue:
        vertex = queue.pop()
        if vertex not in alive:
            continue
        alive.remove(vertex)
        for other in adjacency[vertex]:
            if other in alive and len(adjacency[other] & alive) < 2:
                queue.append(other)
    return alive


def _residual_groups(adjacency: list[set[int]]) -> int:
    unseen = set(range(len(adjacency))) - _cycle_vertices(adjacency)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        stack = [start]
        while stack:
            vertex = stack.pop()
            for other in adjacency[vertex]:
                if other in unseen:
                    unseen.remove(other)
                    stack.append(other)
    return count


def decision_from_image(image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan_delta = np.max(np.abs(rgb.astype(int) - CYAN), axis=2)
    nodes = _components(cyan_delta <= 5)
    if len(nodes) != 30:
        raise ValueError("could not recover thirty cyan stations")
    groups = [[], [], []]
    panel_width = rgb.shape[1] / 3
    for x, y, _ in nodes:
        panel_index = min(2, max(0, int(x // panel_width)))
        groups[panel_index].append((x, y))
    if any(len(group) != 10 for group in groups):
        raise ValueError("could not separate ten stations in each panel")
    winners = []
    for panel_index, points in enumerate(groups):
        adjacency = [set() for _ in points]
        for a in range(10):
            for b in range(a + 1, 10):
                if _has_link(rgb, points[a], points[b]):
                    adjacency[a].add(b)
                    adjacency[b].add(a)
        edge_count = sum(map(len, adjacency)) // 2
        degrees = sorted(map(len, adjacency))
        if edge_count != 10 or degrees != [1, 1, 1, 1, 2, 2, 2, 3, 3, 4]:
            raise ValueError("ambiguous station-link graph")
        if len(_cycle_vertices(adjacency)) != 4:
            raise ValueError("pixel graph does not expose one four-station loop")
        if _residual_groups(adjacency) == 2:
            winners.append(LABELS[panel_index])
    if len(winners) != 1:
        raise ValueError("pixel graph has no unique two-residue-group panel")
    return winners[0]
