"""Renderer and analytic arm for cycle-vertex residue groups."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 420
BG = (250, 249, 245)
FRAME = (196, 199, 201)
OUTLINE = (31, 35, 39)
LINK = (105, 112, 120)
CYAN = (38, 176, 190)
LABELS = ("left", "center", "right")

# All templates are connected unicyclic graphs with ten vertices, ten edges,
# a four-station cycle, six off-cycle stations, and the same degree multiset
# (1,1,1,1,2,2,2,3,3,4). After deleting the cycle vertices, their six-vertex
# forest residues have respectively two, one, and three components.
WINNER = [
    [0, 1], [1, 2], [2, 3], [3, 0],
    [1, 9], [0, 5], [5, 4], [5, 6], [5, 8], [6, 7],
]
LOSER_ONE_GROUP = [
    [0, 1], [1, 2], [2, 3], [3, 0],
    [2, 8], [8, 6], [6, 4], [6, 7], [6, 9], [8, 5],
]
LOSER_THREE_GROUPS = [
    [0, 1], [1, 2], [2, 3], [3, 0],
    [2, 4], [4, 9], [2, 8], [4, 5], [9, 7], [0, 6],
]

# Pre-screened layout permutations avoid a non-edge chord being visually
# indistinguishable from a chain of collinear links. Multiple layouts prevent
# any answer class from being bound to one fixed geometry.
SAFE_ORDERS = {
    "winner": [
        [7, 9, 8, 2, 0, 6, 1, 3, 5, 4],
        [1, 5, 4, 3, 7, 9, 6, 0, 8, 2],
        [3, 1, 5, 4, 0, 9, 7, 2, 6, 8],
        [2, 5, 9, 4, 0, 6, 1, 3, 7, 8],
        [4, 3, 6, 2, 7, 5, 0, 1, 8, 9],
        [3, 5, 8, 2, 0, 4, 6, 1, 7, 9],
    ],
    "one": [
        [9, 2, 8, 0, 5, 6, 7, 1, 3, 4],
        [7, 5, 1, 0, 2, 4, 3, 9, 6, 8],
        [5, 7, 6, 0, 1, 3, 2, 9, 8, 4],
        [6, 3, 2, 4, 7, 9, 5, 8, 0, 1],
        [5, 4, 2, 0, 7, 9, 6, 1, 3, 8],
        [0, 6, 2, 1, 3, 5, 9, 8, 7, 4],
    ],
    "three": [
        [5, 4, 2, 9, 7, 3, 8, 1, 0, 6],
        [7, 1, 3, 6, 0, 5, 8, 2, 4, 9],
        [0, 6, 4, 7, 9, 3, 5, 2, 1, 8],
        [1, 0, 2, 5, 6, 8, 4, 7, 9, 3],
        [0, 7, 8, 5, 2, 4, 9, 3, 1, 6],
        [9, 0, 4, 2, 1, 7, 6, 5, 8, 3],
    ],
}

# An intentionally irregular decagon avoids collinear non-edge chords.
SLOTS = [
    [0, -126], [73, -104], [122, -43], [117, 39], [68, 107],
    [-8, 127], [-78, 101], [-124, 35], [-110, -52], [-49, -116],
]


def _adjacency(edges: list[list[int]]) -> list[set[int]]:
    adjacency = [set() for _ in range(10)]
    for a, b in edges:
        adjacency[a].add(b)
        adjacency[b].add(a)
    return adjacency


def _components(edges: list[list[int]]) -> list[set[int]]:
    adjacency = _adjacency(edges)
    unseen = set(range(10))
    result = []
    while unseen:
        start = unseen.pop()
        component = {start}
        stack = [start]
        while stack:
            vertex = stack.pop()
            for other in adjacency[vertex]:
                if other in unseen:
                    unseen.remove(other)
                    component.add(other)
                    stack.append(other)
        result.append(component)
    return result


def _cycle_vertices(edges: list[list[int]]) -> set[int]:
    adjacency = _adjacency(edges)
    alive = set(range(10))
    queue = [vertex for vertex in alive if len(adjacency[vertex]) < 2]
    while queue:
        vertex = queue.pop()
        if vertex not in alive:
            continue
        alive.remove(vertex)
        for other in adjacency[vertex]:
            if other in alive and len(adjacency[other] & alive) < 2:
                queue.append(other)
    return alive


def _residual_group_count(edges: list[list[int]]) -> int:
    adjacency = _adjacency(edges)
    unseen = set(range(10)) - _cycle_vertices(edges)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        stack = [start]
        while stack:
            vertex = stack.pop()
            for other in adjacency[vertex]:
                if other in unseen:
                    unseen.remove(other)
                    stack.append(other)
    return count


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    templates = [
        (WINNER, SAFE_ORDERS["winner"]),
        (LOSER_ONE_GROUP, SAFE_ORDERS["one"]),
        (LOSER_THREE_GROUPS, SAFE_ORDERS["three"]),
    ]
    rng.shuffle(templates)
    panels = []
    for template, safe_orders in templates:
        panels.append({
            "edges": [list(edge) for edge in template],
            "order": list(rng.choice(safe_orders)),
        })
    return {"panels": panels, "offset": [rng.randint(-4, 4), rng.randint(-6, 6)]}


def _positions(panel_index: int, panel: dict, offset: list[int]) -> list[list[int]]:
    cx = 150 + 300 * panel_index + offset[0]
    cy = 216 + offset[1]
    result = [[0, 0] for _ in range(10)]
    for vertex, slot_index in enumerate(panel["order"]):
        sx, sy = SLOTS[slot_index]
        result[vertex] = [cx + sx, cy + sy]
    return result


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for panel_index, panel in enumerate(scene["panels"]):
        x0 = 10 + 300 * panel_index + scene["offset"][0]
        draw.rounded_rectangle(
            (x0, 45 + scene["offset"][1], x0 + 280, 383 + scene["offset"][1]),
            radius=14, outline=FRAME, width=3,
        )
        positions = _positions(panel_index, panel, scene["offset"])
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=OUTLINE, width=12)
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=LINK, width=8)
        for x, y in positions:
            draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=OUTLINE)
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=CYAN)
    return image


def analytic_gold(scene: dict) -> str:
    counts = [_residual_group_count(panel["edges"]) for panel in scene["panels"]]
    winners = [label for label, count in zip(LABELS, counts) if count == 2]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one two-residue-group panel")
    return winners[0]


def margin(scene: dict) -> float:
    counts = [_residual_group_count(panel["edges"]) for panel in scene["panels"]]
    distances = sorted(abs(count - 2) for count in counts)
    return float(distances[1] - distances[0]) if len(distances) == 3 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if set(scene) != {"panels", "offset"} or len(scene["panels"]) != 3:
            return True
        if len(scene["offset"]) != 2 or any(abs(int(value)) > 8 for value in scene["offset"]):
            return True
        counts = []
        for panel in scene["panels"]:
            if set(panel) != {"edges", "order"} or sorted(panel["order"]) != list(range(10)):
                return True
            edges = panel["edges"]
            normalized = [tuple(sorted(edge)) for edge in edges]
            if len(edges) != 10 or len(set(normalized)) != 10:
                return True
            if any(len(edge) != 2 or edge[0] == edge[1] or min(edge) < 0 or max(edge) > 9 for edge in edges):
                return True
            if len(_components(edges)) != 1 or len(_cycle_vertices(edges)) != 4:
                return True
            degrees = sorted(len(neighbors) for neighbors in _adjacency(edges))
            if degrees != [1, 1, 1, 1, 2, 2, 2, 3, 3, 4]:
                return True
            counts.append(_residual_group_count(edges))
        return sorted(counts) != [1, 2, 3] or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "panels": [
            {"edges": [list(edge) for edge in reversed(panel["edges"])], "order": list(panel["order"])}
            for panel in scene["panels"]
        ],
        "offset": list(scene["offset"]),
    }
    return [("reverse_edge_serialization_order", twin)]
