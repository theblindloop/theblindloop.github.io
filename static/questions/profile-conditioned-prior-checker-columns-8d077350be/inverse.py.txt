"""Independent pixel-only inverse arm for the checkerboard-column world.

Receives ONLY the final PNG and returns the number of columns as a string. It
shares no scene constants with the renderer (only the generic fact that the
board uses two square colours distinct from the white background, which is part
of the visible image). It never imports renderer, prompts, generate, verify, or
analytic-gold code, and accepts no latent scene data.

Method: the checkerboard is the only non-white structure on the canvas. The
oracle thresholds the raster into three bands -- dark square, light square, and
white background -- and takes the bounding box of all non-white pixels as the
board. A checkerboard always alternates colour across columns in every row, so
an interior horizontal scanline that stays inside the board crosses exactly one
dark<->light transition between every adjacent pair of columns. Counting those
transitions and adding one yields the number of columns, robustly across the
board's offset, size and column count.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "prior_checker_columns-oracle-0.1.0"

_DARK_MAX = 120.0
_LIGHT_MIN = 120.0
_LIGHT_MAX = 245.0
_INSET = 3


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    gray = arr[..., 0]
    dark = gray < _DARK_MAX
    light = (gray >= _LIGHT_MIN) & (gray < _LIGHT_MAX)
    board = dark | light
    ys, xs = np.nonzero(board)
    if len(ys) == 0:
        return "0"
    x0 = int(xs.min()) + _INSET
    x1 = int(xs.max()) - _INSET
    y0 = int(ys.min()) + _INSET
    y1 = int(ys.max()) - _INSET
    if x1 <= x0 or y1 <= y0:
        return "1"
    y = (y0 + y1) // 2
    seg = np.where(dark[y], 1, np.where(light[y], 0, -1))[x0:x1]
    seg = seg[(seg == 0) | (seg == 1)]
    if seg.size == 0:
        return "1"
    transitions = int(np.sum(seg[:-1] != seg[1:]))
    return str(transitions + 1)


def measure(image: Image.Image) -> dict:
    """Diagnostic measurement (not used for the decision)."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    gray = arr[..., 0]
    dark = gray < _DARK_MAX
    light = (gray >= _LIGHT_MIN) & (gray < _LIGHT_MAX)
    board = dark | light
    ys, xs = np.nonzero(board)
    return {
        "dark_px": int(dark.sum()),
        "board_px": int(board.sum()),
        "board_bbox": (
            (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()))
            if len(xs)
            else None
        ),
        "size": image.size,
    }
