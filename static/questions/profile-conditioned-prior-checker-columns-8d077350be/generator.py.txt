"""Checkerboard-column count world under a contradictory canonical prior.

Profile: paper-semantic@0.3.0:prior_conflict. Report visible truth (the actual
number of columns) when a familiar canonical prior ("an ordinary chessboard is
8 squares by 8 squares, i.e. it has 8 columns") is contradicted by the actual
image, which in the counterfactual scenes shows a checkerboard with a visibly
different number of columns.

Latent z = {columns, cell, x0, y0, dark, light, width, height}.
``columns`` is the number of columns (and rows, the board is square) actually
drawn on the checkerboard. ``cell`` is the side length of one square in pixels,
``x0``/``y0`` the board's top-left offset, ``dark``/``light`` the two square
colours, and ``width``/``height`` the canvas size. The closed-form gold is

    decision = str(columns)

a scene parameter, not a pixel measurement.

The board is drawn as an n x n grid of alternating dark and light squares on a
plain white background, so the image unambiguously reads as a checkerboard.
There is no drawn seam between squares; the alternating dark/light cells define
the grid, so every horizontal scanline crossing a cell row alternates colour
once per column. The pixel-only inverse arm therefore counts columns by
segmenting the board's two square colours against the white background, finding
the board bounding box, and counting the dark<->light colour alternations along
an interior horizontal scanline: the number of alternations plus one equals the
number of columns.

The continuous verifiability margin is the cell side length in pixels: the
minimum feature size that makes each square visually resolvable. When the cells
are too small the columns crowd together and the count becomes pixel-ambiguous,
so such scenes are quarantined.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "prior_checker_columns-0.1.0"

CANVAS = (512, 512)
BG = (255, 255, 255)       # white paper background
DARK = (28, 28, 38)        # dark squares
LIGHT = (210, 210, 218)    # light squares (clearly distinct from white BG)

SUPPORTED_COLUMNS = (3, 4, 5, 6, 7, 8, 9)
MIN_CELL = 12              # below this, columns are too crowded to count reliably


def margin(scene: dict) -> float:
    """Cell side length (px): the minimum resolvable feature of the grid."""
    return float(scene["cell"])


def is_quarantined(scene: dict) -> bool:
    n = int(scene["columns"])
    cell = int(scene["cell"])
    if n not in SUPPORTED_COLUMNS:
        return True
    if cell < MIN_CELL:
        return True
    w, h = int(scene["width"]), int(scene["height"])
    if n * cell + 6 > min(w, h):
        return True
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        columns = rng.choice(SUPPORTED_COLUMNS)
        cell = rng.randint(10, 34)
        w, h = CANVAS
        board_px = columns * cell
        margin_x = w - board_px - 12
        margin_y = h - board_px - 12
        if margin_x < 1 or margin_y < 1:
            continue
        x0 = rng.randint(6, margin_x)
        y0 = rng.randint(6, margin_y)
        scene = {
            "columns": columns,
            "cell": cell,
            "x0": x0,
            "y0": y0,
            "dark": DARK,
            "light": LIGHT,
            "width": w,
            "height": h,
        }
        if not is_quarantined(scene):
            return scene


def render(scene: dict) -> Image.Image:
    w, h = int(scene["width"]), int(scene["height"])
    n = int(scene["columns"])
    cell = int(scene["cell"])
    x0, y0 = int(scene["x0"]), int(scene["y0"])
    dark = tuple(int(v) for v in scene["dark"])
    light = tuple(int(v) for v in scene["light"])

    img = Image.new("RGB", (w, h), BG)
    draw = ImageDraw.Draw(img)
    for r in range(n):
        for c in range(n):
            colour = dark if (r + c) % 2 == 0 else light
            px = x0 + c * cell
            py = y0 + r * cell
            draw.rectangle([px, py, px + cell - 1, py + cell - 1], fill=colour)
    return img


def analytic_gold(scene: dict) -> str:
    return str(scene["columns"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No pixel-identical latent alias exists: the scene maps injectively.

    The column count, cell size, board offset and both square colours are all
    directly visible in the raster, and changing any declared field changes the
    pixels. This is a `not_applicable` latent-alias mode, so no transforms are
    declared.
    """
    return []
