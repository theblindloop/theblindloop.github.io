"""Independent pixel-only inverse for the parity-weave patch.

This module intentionally duplicates the small raster specification instead of
importing the renderer.  It locates the largest foreground patch, audits every
interior cell, deinterleaves the two checkerboard parity streams, and classifies
the pairwise equality bitmap.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
PANEL_SIDE = 96
FRAME = 8
INNER_SIDE = 80
GRID_SIDE = 16
CELL = 5

PETAL = (
    "....####....####",
    "...######...####",
    "..########..####",
    ".##########.####",
    ".##########.####",
    "..########..####",
    "...######...####",
    "....####....####",
)
GATE = (
    "################",
    "####........####",
    "####........####",
    "################",
    "####........####",
    "####........####",
    "####........####",
    "################",
)
TEMPLATES = {
    "A": tuple(int(char == "#") for row in PETAL for char in row),
    "B": tuple(int(char == "#") for row in GATE for char in row),
}
EVEN_COORDINATES = tuple(
    (row, col)
    for row in range(GRID_SIDE)
    for col in range(GRID_SIDE)
    if (row + col) % 2 == 0
)
ODD_COORDINATES = tuple(
    (row, col)
    for row in range(GRID_SIDE)
    for col in range(GRID_SIDE)
    if (row + col) % 2 == 1
)


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.uint8)


def _foreground(array: np.ndarray, background: np.ndarray) -> np.ndarray:
    distance = np.linalg.norm(array.astype(np.float32) - background.astype(np.float32), axis=2)
    return distance > 16.0


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        component: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            component.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        if len(component) > len(best):
            best = component
    if len(best) < 7000:
        raise ValueError("no complete frame-sized patch found")
    output = np.zeros_like(mask, dtype=bool)
    points = np.asarray(best, dtype=np.int32)
    output[points[:, 0], points[:, 1]] = True
    return output


def _locate_patch(mask: np.ndarray) -> tuple[int, int, np.ndarray]:
    component = _largest_component(mask)
    ys, xs = np.nonzero(component)
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    if (x1 - x0, y1 - y0) != (PANEL_SIDE, PANEL_SIDE):
        raise ValueError("patch bounding box is not the declared square")
    patch = component[y0:y1, x0:x1]
    if float(patch.mean()) < 0.96:
        raise ValueError("patch foreground coverage is incomplete")
    return x0, y0, patch


def _cell_centers(array: np.ndarray, coverage: np.ndarray, x0: int, y0: int) -> np.ndarray:
    inner_mask = coverage[
        FRAME : FRAME + INNER_SIDE,
        FRAME : FRAME + INNER_SIDE,
    ]
    if inner_mask.shape != (INNER_SIDE, INNER_SIDE):
        raise ValueError("patch interior is clipped")
    centers: list[np.ndarray] = []
    inner = array[
        y0 + FRAME : y0 + FRAME + INNER_SIDE,
        x0 + FRAME : x0 + FRAME + INNER_SIDE,
    ]
    for row in range(GRID_SIDE):
        for col in range(GRID_SIDE):
            top, left = row * CELL, col * CELL
            block_mask = inner_mask[top : top + CELL, left : left + CELL]
            if float(block_mask.mean()) < 0.94:
                raise ValueError("an interior cell has incomplete coverage")
            block = inner[top : top + CELL, left : left + CELL]
            centers.append(np.median(block.reshape(-1, 3), axis=0))
    return np.asarray(centers, dtype=np.float32)


def _cell_labels(centers: np.ndarray) -> np.ndarray:
    first = centers[0]
    distances = np.linalg.norm(centers - first, axis=1)
    second = centers[int(np.argmax(distances))]
    if float(np.linalg.norm(first - second)) < 30.0:
        raise ValueError("the two interior cell tones are not separable")
    distances_a = np.linalg.norm(centers - first, axis=1)
    distances_b = np.linalg.norm(centers - second, axis=1)
    return (distances_b < distances_a).astype(np.int8)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using only the final rasterized image."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (HEIGHT, WIDTH):
        raise ValueError("unexpected raster dimensions")
    background = _background(array)
    mask = _foreground(array, background)
    x0, y0, patch = _locate_patch(mask)
    centers = _cell_centers(array, patch, x0, y0)
    labels = _cell_labels(centers)
    code: list[int] = []
    for even, odd in zip(EVEN_COORDINATES, ODD_COORDINATES, strict=True):
        even_index = even[0] * GRID_SIDE + even[1]
        odd_index = odd[0] * GRID_SIDE + odd[1]
        code.append(int(labels[even_index] != labels[odd_index]))
    distance_a = sum(value != expected for value, expected in zip(code, TEMPLATES["A"], strict=True))
    distance_b = sum(value != expected for value, expected in zip(code, TEMPLATES["B"], strict=True))
    if distance_a == distance_b:
        raise ValueError("parity bitmap is tied between the two templates")
    return "A" if distance_a < distance_b else "B"
