"""Deterministic sampler and renderer for reciprocal-time guarantees."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1200, 960
SIZE = (WIDTH, HEIGHT)
BACKGROUND = (247, 245, 238)
CARD = (255, 253, 248)
CLUE_CARD = (236, 246, 250)
INK = (35, 42, 54)
MUTED = (100, 108, 119)
ACCENT = (207, 91, 45)
COLORS = (
    (201, 67, 64),
    (48, 111, 176),
    (43, 147, 103),
    (222, 153, 35),
    (137, 88, 174),
    (27, 153, 164),
)
DECISIONS = ("route-1", "route-2", "route-3", "route-4")

# Every clue states sum(left speeds) > sum(right speeds).
CLUE_SPECS = (
    ((3, 5), (0, 1)),
    ((3, 4), (1, 2)),
    ((1, 3), (4, 5)),
    ((0, 4), (2, 5)),
    ((2, 3), (0, 1)),
    ((1, 4), (0, 5)),
    ((0, 3), (1, 2)),
)

# Each route contains two sequential one-unit jobs. Only (3, 4) has
# 1/speed[3] + 1/speed[4] < 3/4 throughout the feasible family. Every
# distractor is strictly below the deadline for some calibrations and above it
# for others; none ever lands exactly on the deadline.
BASE_ROUTE_SPECS = (
    (3, 4),
    (0, 3),
    (1, 3),
    (2, 3),
)

CLUE_RECTS = (
    (20, 124, 295, 244),
    (315, 124, 590, 244),
    (610, 124, 885, 244),
    (905, 124, 1180, 244),
    (167, 264, 442, 384),
    (462, 264, 737, 384),
    (757, 264, 1032, 384),
)
ROUTE_RECTS = (
    (30, 515, 585, 710),
    (615, 515, 1170, 710),
    (30, 735, 585, 930),
    (615, 735, 1170, 930),
)

_BOLD_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
FONT_SMALL = ImageFont.truetype(_BOLD_PATH, 16)
FONT_MEDIUM = ImageFont.truetype(_BOLD_PATH, 24)
FONT_LARGE = ImageFont.truetype(_BOLD_PATH, 36)


def _clue_difference(
    spec: tuple[tuple[int, ...], tuple[int, ...]], speeds: tuple[int, ...]
) -> int:
    left, right = spec
    return sum(speeds[index] for index in left) - sum(speeds[index] for index in right)


FEASIBLE_SPEEDS = tuple(
    speeds
    for speeds in itertools.permutations(range(1, 10), 6)
    if all(_clue_difference(spec, speeds) > 0 for spec in CLUE_SPECS)
)


def _deadline_gap(route: tuple[int, int], speeds: tuple[int, ...]) -> int:
    """Positive exactly when 1/a + 1/b < 3/4."""
    a, b = (speeds[index] for index in route)
    return 3 * a * b - 4 * (a + b)


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    role_colors = list(range(6))
    rng.shuffle(role_colors)
    distractors = list(BASE_ROUTE_SPECS[1:])
    rng.shuffle(distractors)
    answer_slot = seed % 4
    routes = distractors[:]
    routes.insert(answer_slot, BASE_ROUTE_SPECS[0])
    return {
        "role_colors": role_colors,
        "route_specs": [list(route) for route in routes],
    }


def _validated(scene: dict) -> tuple[list[int], tuple[tuple[int, int], ...]]:
    role_colors = [int(value) for value in scene["role_colors"]]
    raw_routes = scene["route_specs"]
    if sorted(role_colors) != list(range(6)):
        raise ValueError("role_colors must be a permutation of six colors")
    if not isinstance(raw_routes, list) or len(raw_routes) != 4:
        raise ValueError("four route specs are required")
    routes: list[tuple[int, int]] = []
    for raw in raw_routes:
        if not isinstance(raw, list) or len(raw) != 2:
            raise ValueError("each route needs two stages")
        route = tuple(int(value) for value in raw)
        if route != tuple(sorted(route)) or len(set(route)) != 2:
            raise ValueError("route roles must be a canonical distinct pair")
        if any(value not in range(6) for value in route):
            raise ValueError("invalid route role")
        routes.append(route)
    if sorted(routes) != sorted(BASE_ROUTE_SPECS):
        raise ValueError("route specs must be the fixed comparison set")
    return role_colors, tuple(routes)


def _guaranteed(route: tuple[int, int]) -> bool:
    return bool(FEASIBLE_SPEEDS) and all(
        _deadline_gap(route, speeds) > 0 for speeds in FEASIBLE_SPEEDS
    )


def analytic_gold(scene: dict) -> str:
    try:
        _role_colors, routes = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return "abstain"
    matches = [index for index, route in enumerate(routes) if _guaranteed(route)]
    return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"


def margin(scene: dict) -> float:
    try:
        _role_colors, routes = _validated(scene)
    except (KeyError, TypeError, ValueError):
        return 0.0
    guaranteed = [route for route in routes if _guaranteed(route)]
    distractors = [route for route in routes if not _guaranteed(route)]
    if len(guaranteed) != 1 or len(distractors) != 3:
        return 0.0
    clearances = [
        min(_deadline_gap(guaranteed[0], speeds) for speeds in FEASIBLE_SPEEDS)
    ]
    for route in distractors:
        gaps = [_deadline_gap(route, speeds) for speeds in FEASIBLE_SPEEDS]
        positive = [gap for gap in gaps if gap > 0]
        negative = [-gap for gap in gaps if gap < 0]
        if not positive or not negative or any(gap == 0 for gap in gaps):
            return 0.0
        clearances.extend((min(positive), min(negative)))
    return float(min(clearances))


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered_text(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    label: str,
    font: ImageFont.FreeTypeFont,
    fill: tuple[int, int, int],
) -> None:
    box = draw.textbbox((0, 0), label, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - box[1]),
        label,
        font=font,
        fill=fill,
    )


def _machine(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], color: tuple[int, int, int]
) -> None:
    x, y = center
    radius = 21
    draw.ellipse(
        (x - radius, y - radius, x + radius, y + radius),
        fill=color,
        outline=INK,
        width=3,
    )
    draw.ellipse((x - 5, y - 5, x + 5, y + 5), fill=(255, 255, 255), outline=INK, width=2)


def _draw_clue(
    draw: ImageDraw.ImageDraw,
    rect: tuple[int, int, int, int],
    spec: tuple[tuple[int, ...], tuple[int, ...]],
    role_colors: list[int],
    label: str,
) -> None:
    x0, y0, x1, y1 = rect
    draw.rounded_rectangle(rect, radius=15, fill=CLUE_CARD, outline=INK, width=4)
    draw.rounded_rectangle((x0 + 10, y0 + 9, x0 + 92, y0 + 39), radius=8, fill=INK)
    _centered_text(draw, (x0 + 51, y0 + 24), label, FONT_SMALL, (255, 255, 255))
    cx = (x0 + x1) // 2
    cy = y0 + 82
    offsets = (-78, -36, 36, 78)
    left, right = spec
    for role, offset in zip(left + right, offsets):
        _machine(draw, (cx + offset, cy), COLORS[role_colors[role]])
    _centered_text(draw, (cx, cy), ">", FONT_LARGE, INK)


def _draw_route(
    draw: ImageDraw.ImageDraw,
    rect: tuple[int, int, int, int],
    route: tuple[int, int],
    role_colors: list[int],
    label: str,
) -> None:
    x0, y0, x1, y1 = rect
    draw.rounded_rectangle(rect, radius=17, fill=CARD, outline=INK, width=4)
    draw.rounded_rectangle((x0 + 12, y0 + 11, x0 + 122, y0 + 45), radius=8, fill=INK)
    _centered_text(draw, (x0 + 67, y0 + 28), label, FONT_SMALL, (255, 255, 255))
    cx = (x0 + x1) // 2
    cy = y0 + 126
    left_x, right_x = cx - 120, cx + 120
    _centered_text(draw, (left_x, y0 + 72), "1-UNIT JOB", FONT_SMALL, MUTED)
    _centered_text(draw, (right_x, y0 + 72), "1-UNIT JOB", FONT_SMALL, MUTED)
    _machine(draw, (left_x, cy), COLORS[role_colors[route[0]]])
    _machine(draw, (right_x, cy), COLORS[role_colors[route[1]]])
    draw.line((cx - 70, cy, cx + 63, cy), fill=INK, width=6)
    draw.polygon(((cx + 75, cy), (cx + 57, cy - 12), (cx + 57, cy + 12)), fill=INK)
    _centered_text(draw, (cx, y0 + 174), "one after another", FONT_SMALL, MUTED)


def render(scene: dict) -> Image.Image:
    role_colors, routes = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered_text(draw, (WIDTH // 2, 26), "DIFFERENT WHOLE-NUMBER SPEEDS 1-9", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 62), "EVERY CLUE:  LEFT PAIR SPEED SUM  >  RIGHT PAIR SPEED SUM", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 98), "SEVEN CLUES", FONT_SMALL, MUTED)
    for index, (rect, spec) in enumerate(zip(CLUE_RECTS, CLUE_SPECS), 1):
        _draw_clue(draw, rect, spec, role_colors, f"CLUE {index}")
    _centered_text(draw, (WIDTH // 2, 417), "TWO 1-UNIT JOBS, ONE AFTER ANOTHER", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 453), "ROUTE TIME = 1 / SPEED A  +  1 / SPEED B", FONT_MEDIUM, INK)
    _centered_text(draw, (WIDTH // 2, 489), "WHICH ROUTE IS ALWAYS UNDER 3 / 4 TIME?", FONT_MEDIUM, ACCENT)
    for index, (rect, route) in enumerate(zip(ROUTE_RECTS, routes), 1):
        _draw_route(draw, rect, route, role_colors, f"ROUTE {index}")
    return image
