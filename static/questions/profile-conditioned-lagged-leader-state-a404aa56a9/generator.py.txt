"""PIL renderer and analytic answer for a cross-identity lag task.

The raster shows six ordered checkpoints.  A separate LEADER row supplies a
six-state code, while four labeled rows supply identity-preserving tracks.  The
answer is the unique label whose P2--P6 suffix equals the LEADER's P1--P5
prefix.  Palette, row order, panel tint, shape, and marker size are rendered
nuisances; the analytic answer reads only the two state-track fields.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 1440
HEIGHT = 900
PANEL_LEFTS = (20, 255, 490, 725, 960, 1195)
PANEL_CENTERS = (130, 365, 600, 835, 1070, 1305)
PANEL_WIDTH = 220
PANEL_TOP = 205
PANEL_BOTTOM = 810
LEADER_Y = 302
ROW_Y = (442, 535, 628, 721)
STATE_KEY_X = (930, 1030, 1130)
STATE_KEY_Y = 104
LABEL_KEY_X = (100, 220, 340, 460)
LABEL_KEY_Y = 858
LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")

# The renderer samples a permutation of each palette.  All colors are well
# separated so an inverse can recover the keyed categories from pixel interiors.
LABEL_COLORS = (
    (214, 70, 70),
    (48, 116, 216),
    (58, 166, 103),
    (153, 78, 190),
)
STATE_COLORS = (
    (241, 164, 43),
    (25, 166, 177),
    (220, 69, 145),
)
PANEL_TINTS = (
    (250, 247, 240),
    (244, 249, 250),
    (249, 245, 251),
    (246, 250, 244),
    (250, 246, 243),
    (245, 248, 252),
)
STATE_SHAPES = ("circle", "square", "diamond")


def _font(size: int = 19) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _rgb(value: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(channel) for channel in value)  # type: ignore[return-value]


def _text_center(
    draw: ImageDraw.ImageDraw,
    xy: tuple[int, int],
    text: str,
    fill: tuple[int, int, int],
    size: int = 18,
) -> None:
    font = _font(size)
    box = draw.textbbox((0, 0), text, font=font)
    draw.text(
        (xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2 - 2),
        text,
        font=font,
        fill=fill,
    )


def _draw_state(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    radius: int,
    shape: str,
    color: tuple[int, int, int],
) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    elif shape == "square":
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(3, radius // 5),
            fill=color,
        )
    elif shape == "diamond":
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
        )
    else:
        raise ValueError(f"unknown state shape {shape!r}")


def _scores(leader_states: list[int], tile_states: dict[str, list[int]]) -> list[int]:
    if len(leader_states) != 6 or any(value not in (0, 1, 2) for value in leader_states):
        raise ValueError("leader_states must contain six three-state values")
    reference = leader_states[:-1]
    scores: list[int] = []
    for label in LABELS:
        trajectory = tile_states[label]
        if len(trajectory) != 6 or any(value not in (0, 1, 2) for value in trajectory):
            raise ValueError("each tile trajectory must contain six three-state values")
        scores.append(sum(int(value == expected) for value, expected in zip(trajectory[1:], reference)))
    return scores


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene with a unique exact one-step lag match."""

    rng = random.Random(int(seed))
    leader_states = [rng.randrange(3) for _ in range(6)]
    answer_index = (int(seed) + int(seed) // 7) % 4
    reference = leader_states[:-1]
    near_boundary = int(seed) % 7 == 0

    trajectories: list[list[int] | None] = [None, None, None, None]
    trajectories[answer_index] = [rng.randrange(3), *reference]
    used_suffixes = {tuple(reference)}

    # A near-boundary distractor differs from the reference at one checkpoint;
    # ordinary scenes keep every distractor at least two mismatches away.
    for identity in range(4):
        if trajectories[identity] is not None:
            continue
        for _ in range(10000):
            suffix = [rng.randrange(3) for _ in range(5)]
            score = sum(a == b for a, b in zip(suffix, reference))
            if near_boundary and not any(
                trajectory is not None and sum(a == b for a, b in zip(trajectory[1:], reference)) == 4
                for trajectory in trajectories
            ):
                acceptable = score == 4
            else:
                acceptable = score <= 3
            candidate = tuple(suffix)
            if acceptable and candidate not in used_suffixes:
                trajectories[identity] = [rng.randrange(3), *suffix]
                used_suffixes.add(candidate)
                break
        if trajectories[identity] is None:
            raise RuntimeError("could not construct a unique lagged-track scene")

    label_palette = [list(color) for color in LABEL_COLORS]
    state_palette = [list(color) for color in STATE_COLORS]
    rng.shuffle(label_palette)
    rng.shuffle(state_palette)
    panel_tints = [list(PANEL_TINTS[index]) for index in rng.sample(range(6), 6)]
    row_orders = [rng.sample(range(4), 4) for _ in range(6)]
    state_shapes = list(STATE_SHAPES)
    rng.shuffle(state_shapes)

    return {
        "leader_states": leader_states,
        "tile_states": {label: list(trajectories[index] or []) for index, label in enumerate(LABELS)},
        "label_palette": label_palette,
        "state_palette": state_palette,
        "panel_tints": panel_tints,
        "row_orders": row_orders,
        "state_shapes": state_shapes,
        "marker_radius": rng.choice((19, 21, 23)),
        "tile_outline_width": rng.choice((2, 3)),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (247, 244, 237))
    draw = ImageDraw.Draw(image)
    dark = (35, 45, 58)
    muted = (101, 112, 124)
    label_palette = [_rgb(color) for color in scene["label_palette"]]
    state_palette = [_rgb(color) for color in scene["state_palette"]]
    panel_tints = [_rgb(color) for color in scene["panel_tints"]]
    row_orders = scene["row_orders"]
    state_shapes = scene["state_shapes"]
    radius = int(scene["marker_radius"])
    outline_width = int(scene["tile_outline_width"])

    draw.rounded_rectangle((18, 15, 1422, 65), radius=14, fill=(34, 50, 70))
    draw.text((42, 28), "CHECKPOINT FOLLOWER", font=_font(22), fill=(255, 255, 255))
    draw.text(
        (43, 76),
        "The LEADER changes state across six checkpoints. Find the tile that follows it one step later.",
        font=_font(16),
        fill=muted,
    )

    draw.rounded_rectangle((860, 78, 1204, 133), radius=12, fill=(255, 253, 248), outline=(207, 199, 186), width=2)
    draw.text((876, 91), "STATE KEY", font=_font(15), fill=dark)
    for state, x in enumerate(STATE_KEY_X):
        _draw_state(draw, (x, STATE_KEY_Y), 13, state_shapes[state], state_palette[state])
        draw.text((x + 18, 94), STATE_NAMES[state], font=_font(14), fill=dark)

    for panel, left in enumerate(PANEL_LEFTS):
        center_x = PANEL_CENTERS[panel]
        draw.rounded_rectangle(
            (left, PANEL_TOP, left + PANEL_WIDTH, PANEL_BOTTOM),
            radius=14,
            fill=panel_tints[panel],
            outline=(183, 191, 199),
            width=2,
        )
        draw.rectangle((left + 1, PANEL_TOP + 1, left + PANEL_WIDTH - 1, PANEL_TOP + 48), fill=(235, 239, 241))
        _text_center(draw, (center_x, PANEL_TOP + 25), f"P{panel + 1}", dark, size=18)
        if panel < len(PANEL_CENTERS) - 1:
            draw.line((left + PANEL_WIDTH + 5, PANEL_TOP + 25, PANEL_LEFTS[panel + 1] - 5, PANEL_TOP + 25), fill=(151, 161, 171), width=2)
            tip = PANEL_LEFTS[panel + 1] - 5
            draw.polygon(((tip - 8, PANEL_TOP + 20), (tip, PANEL_TOP + 25), (tip - 8, PANEL_TOP + 30)), fill=(151, 161, 171))

        draw.rounded_rectangle((left + 10, LEADER_Y - 35, left + PANEL_WIDTH - 10, LEADER_Y + 35), radius=9, fill=(255, 255, 255), outline=(104, 125, 137), width=outline_width)
        draw.text((left + 23, LEADER_Y - 11), "LEADER", font=_font(14), fill=dark)
        _draw_state(draw, (center_x + 55, LEADER_Y), radius, state_shapes[scene["leader_states"][panel]], state_palette[scene["leader_states"][panel]])

        for row, identity in enumerate(row_orders[panel]):
            y = ROW_Y[row]
            draw.rounded_rectangle((left + 10, y - 35, left + PANEL_WIDTH - 10, y + 35), radius=9, fill=(255, 255, 255), outline=(151, 160, 168), width=outline_width)
            draw.rounded_rectangle((left + 19, y - 23, left + 59, y + 23), radius=6, fill=label_palette[identity])
            _text_center(draw, (left + 39, y), LABELS[identity], (255, 255, 255), size=20)
            state = scene["tile_states"][LABELS[identity]][panel]
            _draw_state(draw, (left + 164, y), radius, state_shapes[state], state_palette[state])

    draw.rounded_rectangle((28, 832, 596, 884), radius=12, fill=(255, 253, 248), outline=(207, 199, 186), width=2)
    draw.text((44, 846), "IDENTITY KEY", font=_font(15), fill=dark)
    for identity, x in enumerate(LABEL_KEY_X):
        draw.rectangle((x - 18, LABEL_KEY_Y - 12, x + 18, LABEL_KEY_Y + 12), fill=label_palette[identity])
        _text_center(draw, (x, LABEL_KEY_Y + 28), LABELS[identity], dark, size=16)
    draw.text((650, 850), "Compare tile P2 with LEADER P1, tile P3 with LEADER P2, and so on.", font=_font(15), fill=muted)
    return image


def analytic_gold(scene: dict) -> str:
    """Compute the answer from the semantic state tracks, not from pixels."""

    scores = _scores(scene["leader_states"], scene["tile_states"])
    best = max(scores)
    if best != 5 or scores.count(best) != 1:
        raise ValueError(f"scene has no unique exact lag match: {scores}")
    return LABELS[scores.index(best)]


def margin(scene: dict) -> float:
    scores = _scores(scene["leader_states"], scene["tile_states"])
    ordered = sorted(scores, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No separate latent alias is claimed for this fully rendered encoding."""

    return []
