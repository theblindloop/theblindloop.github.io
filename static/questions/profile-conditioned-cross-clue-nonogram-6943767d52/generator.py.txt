"""Renderer and analytic arm for the cross-clue nonogram world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

GLYPHS = ("circle", "kite", "plus", "X")
PATTERNS = {
    "circle": (
        "..####..", ".##..##.", "##....##", "#......#",
        "#......#", "##....##", ".##..##.", "..####..",
    ),
    "kite": (
        "...##...", "..####..", ".######.", "##....##",
        ".##..##.", "..####..", "...##...", "...##...",
    ),
    "plus": (
        "...##...", "...##...", "...##...", "########",
        "########", "...##...", "...##...", "...##...",
    ),
    "X": (
        "##....##", ".##..##.", "..####..", "...##...",
        "...##...", "..####..", ".##..##.", "##....##",
    ),
}
PAPERS = ((250, 250, 247), (250, 247, 240), (242, 248, 250))
GRID_INK = (45, 48, 54)
CLUE_INK = (25, 105, 190)
MARK_INK = (238, 177, 30)


def _bits(name: str) -> list[list[int]]:
    return [[int(ch == "#") for ch in row] for row in PATTERNS[name]]


def _runs(values: list[int]) -> tuple[int, ...]:
    out: list[int] = []
    count = 0
    for value in values + [0]:
        if value:
            count += 1
        elif count:
            out.append(count)
            count = 0
    return tuple(out)


def _clues(name: str) -> tuple[list[tuple[int, ...]], list[tuple[int, ...]]]:
    grid = _bits(name)
    rows = [_runs(row) for row in grid]
    cols = [_runs([grid[y][x] for y in range(8)]) for x in range(8)]
    return rows, cols


def sample_scene(seed: int) -> dict:
    """Balanced deterministic sampling with answer-independent style factors."""
    rng = random.Random(seed)
    # The modulo supplies exact four-way balance for consecutive generator seeds.
    glyph = GLYPHS[seed % len(GLYPHS)]
    return {
        "glyph": glyph,
        "palette": (seed // 4) % len(PAPERS),
        "spacing_variant": rng.randrange(2),
    }


def render(scene: dict) -> Image.Image:
    glyph = scene["glyph"]
    paper_index = int(scene["palette"]) % len(PAPERS)
    # Both aliases intentionally quantize to the same visible integer gap.
    spacing = 8 if int(scene["spacing_variant"]) in (0, 1) else 8
    image = Image.new("RGB", (512, 512), PAPERS[paper_index])
    draw = ImageDraw.Draw(image)
    x0, y0, pitch = 190, 190, 30
    x1 = x0 + 8 * pitch
    y1 = y0 + 8 * pitch

    # Empty solve grid and a visible orientation fiducial at its upper-left.
    for k in range(9):
        x = x0 + k * pitch
        y = y0 + k * pitch
        draw.line((x, y0, x, y1), fill=GRID_INK, width=2)
        draw.line((x0, y, x1, y), fill=GRID_INK, width=2)
    draw.rectangle((x0 - 13, y0 - 13, x0 - 5, y0 - 5), fill=MARK_INK)

    row_clues, col_clues = _clues(glyph)
    unit, thick = 10, 8
    # Row sequences are displayed left-to-right and right-aligned to each row.
    for row, clues in enumerate(row_clues):
        total = sum(clues) * unit + (len(clues) - 1) * spacing
        x = x0 - 18 - total
        cy = y0 + row * pitch + pitch // 2
        for length in clues:
            width = length * unit
            draw.rectangle((x, cy - thick // 2, x + width - 1, cy + thick // 2 - 1), fill=CLUE_INK)
            x += width + spacing

    # Column sequences are displayed top-to-bottom and bottom-aligned above.
    for col, clues in enumerate(col_clues):
        total = sum(clues) * unit + (len(clues) - 1) * spacing
        y = y0 - 18 - total
        cx = x0 + col * pitch + pitch // 2
        for length in clues:
            height = length * unit
            draw.rectangle((cx - thick // 2, y, cx + thick // 2 - 1, y + height - 1), fill=CLUE_INK)
            y += height + spacing
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["glyph"])


def margin(scene: dict) -> float:
    target = _bits(scene["glyph"])
    gaps = []
    for other in GLYPHS:
        if other == scene["glyph"]:
            continue
        candidate = _bits(other)
        gaps.append(sum(a != b for ra, rb in zip(target, candidate) for a, b in zip(ra, rb)))
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["spacing_variant"] = 1 - int(scene["spacing_variant"])
    return [("quantized_padding_alias", twin)]
