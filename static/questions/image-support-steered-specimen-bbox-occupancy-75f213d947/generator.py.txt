"""Deterministic renderer for self-normalized specimen occupancy."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = 384
PALETTE = [
    [20, 150, 145],
    [31, 119, 180],
    [197, 88, 52],
    [112, 78, 170],
]

# Both masks are one 4-connected component and touch all four sides of their
# 7x7 grid. Rotations/reflections diversify contour orientation without changing
# the tight-box occupancy.
YES_MASK = [
    "0001000",
    "0011100",
    "0111110",
    "1111111",
    "0111110",
    "0011100",
    "0001100",
]
NO_MASK = [
    "0001000",
    "0011100",
    "0111110",
    "1111111",
    "0111110",
    "0010000",
    "0010000",
]


def _transform(rows: list[str], turns: int, mirror: bool) -> list[str]:
    grid = [list(row) for row in rows]
    if mirror:
        grid = [list(reversed(row)) for row in grid]
    for _ in range(turns % 4):
        grid = [list(row) for row in zip(*grid[::-1])]
    return ["".join(row) for row in grid]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    positive = (int(seed) & 1) == 0
    mask = _transform(YES_MASK if positive else NO_MASK, rng.randrange(4), bool(rng.randrange(2)))
    cell_size = rng.randrange(29, 33)
    span = 7 * cell_size
    # The specimen remains regional: 203..224 px on a 384 px canvas.
    origin_x = rng.randrange(58, CANVAS - span - 57)
    origin_y = rng.randrange(58, CANVAS - span - 57)
    return {
        "mask": mask,
        "origin_x": origin_x,
        "origin_y": origin_y,
        "cell_size": cell_size,
        "color": list(rng.choice(PALETTE)),
        "canvas_size": CANVAS,
        "alias_nonce": rng.randrange(1_000_000),
    }


def render(scene: dict) -> Image.Image:
    mask = scene["mask"]
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    cell = int(scene["cell_size"])
    color = tuple(int(value) for value in scene["color"])
    size = int(scene["canvas_size"])
    image = Image.new("RGB", (size, size), (248, 247, 243))
    draw = ImageDraw.Draw(image)
    for row_index, row in enumerate(mask):
        for column_index, value in enumerate(row):
            if value == "1":
                left = x0 + column_index * cell
                top = y0 + row_index * cell
                draw.rectangle((left, top, left + cell - 1, top + cell - 1), fill=color)
    return image


def _occupancy_numerator(scene: dict) -> int:
    mask = scene["mask"]
    filled = sum(row.count("1") for row in mask)
    width = max(index for row in mask for index, value in enumerate(row) if value == "1") - min(
        index for row in mask for index, value in enumerate(row) if value == "1"
    ) + 1
    occupied_rows = [index for index, row in enumerate(mask) if "1" in row]
    height = max(occupied_rows) - min(occupied_rows) + 1
    return 2 * filled - width * height


def analytic_gold(scene: dict) -> str:
    return "yes" if _occupancy_numerator(scene) > 0 else "no"


def margin(scene: dict) -> float:
    mask = scene["mask"]
    total = len(mask) * len(mask[0])
    return abs(_occupancy_numerator(scene)) / total


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.04


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["alias_nonce"] = (int(twin["alias_nonce"]) + 1) % 1_000_000
    return [("audit_nonce_increment", twin)]
