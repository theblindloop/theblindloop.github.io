"""Independent pixel-only inverse arm for tight-box occupancy."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _ink_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Every public palette color is saturated while the warm background is not.
    chroma = array.max(axis=2) - array.min(axis=2)
    distance_from_background = np.linalg.norm(array - np.array([248, 247, 243]), axis=2)
    return (chroma > 45) & (distance_from_background > 80)


def _component_count(mask: np.ndarray) -> int:
    seen = np.zeros(mask.shape, dtype=bool)
    count = 0
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        count += 1
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
    return count


def decision_from_image(image: Image.Image) -> str:
    mask = _ink_mask(image)
    area = int(mask.sum())
    if area < 2000:
        raise ValueError("no substantial colored specimen")
    if _component_count(mask) != 1:
        raise ValueError("specimen must be one connected component")
    ys, xs = np.nonzero(mask)
    box_area = (int(xs.max()) - int(xs.min()) + 1) * (int(ys.max()) - int(ys.min()) + 1)
    # The image supplies both terms: colored pixels and their tight box.
    return "yes" if 2 * area > box_area else "no"
