"""Deterministic world for searching detached arc pairs for cocircularity."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = [248, 247, 242]
COLORS = (
    ("red", [220, 48, 57]),
    ("blue", [37, 99, 196]),
    ("green", [32, 150, 90]),
    ("purple", [139, 70, 185]),
)

# Each location consists of two diagonal-neighbor 8x8-grid cell centers.  A
# short arc stays inside the exclusive core of its cell under the public
# overlapping ablation probe; the two decisive components remain detached.
PAIR_LOCATIONS = (
    ((96.0, 96.0), (160.0, 160.0)),
    ((352.0, 96.0), (416.0, 160.0)),
    ((96.0, 352.0), (160.0, 416.0)),
    ((352.0, 352.0), (416.0, 416.0)),
)

ARC_LENGTH = 28.0
STROKE_WIDTH = 5
QUARANTINE_MARGIN = 8.0


def _angle_of(dx: float, dy: float) -> float:
    return math.degrees(math.atan2(dy, dx)) % 360.0


def _arc_from_midpoint(
    midpoint: tuple[float, float], radius: float, radial_angle: float
) -> dict:
    theta = math.radians(radial_angle)
    cx = midpoint[0] - radius * math.cos(theta)
    cy = midpoint[1] - radius * math.sin(theta)
    return {
        "circle_center": [round(cx, 6), round(cy, 6)],
        "radius": round(radius, 6),
        "mid_angle": round(radial_angle % 360.0, 6),
        "span": round(math.degrees(ARC_LENGTH / radius), 6),
        "width": STROKE_WIDTH,
    }


def _make_pair(
    color_name: str,
    rgb: list[int],
    location: tuple[tuple[float, float], tuple[float, float]],
    kind: str,
    sign: int,
) -> dict:
    p1, p2 = location
    mx = (p1[0] + p2[0]) / 2.0
    my = (p1[1] + p2[1]) / 2.0
    base_radius = math.hypot(p1[0] - mx, p1[1] - my)
    a1 = _angle_of(p1[0] - mx, p1[1] - my)
    a2 = _angle_of(p2[0] - mx, p2[1] - my)

    first = _arc_from_midpoint(p1, base_radius, a1)
    if kind == "cocircular":
        second = _arc_from_midpoint(p2, base_radius, a2)
    elif kind == "direction_only":
        # Equal curvature but a different extrapolated center.
        second = _arc_from_midpoint(p2, base_radius, a2 + sign * 48.0)
    elif kind == "curvature_only":
        # Parallel local tangents, but unequal radius and center.
        second = _arc_from_midpoint(p2, 22.0, a2)
    elif kind == "double_mismatch":
        second = _arc_from_midpoint(p2, 30.0, a2 - sign * 36.0)
    else:
        raise ValueError(f"unknown pair kind {kind!r}")
    return {
        "color": color_name,
        "rgb": list(rgb),
        "arcs": [first, second],
    }


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; answer color is exactly ``seed mod 4``."""
    rng = random.Random(int(seed))
    target_index = int(seed) % len(COLORS)
    locations = list(PAIR_LOCATIONS)
    rng.shuffle(locations)
    distractor_kinds = ["direction_only", "curvature_only", "double_mismatch"]
    rng.shuffle(distractor_kinds)
    sign = -1 if rng.randrange(2) else 1

    pairs = []
    d_index = 0
    for index, (name, rgb) in enumerate(COLORS):
        if index == target_index:
            kind = "cocircular"
        else:
            kind = distractor_kinds[d_index]
            d_index += 1
        pairs.append(_make_pair(name, rgb, locations[index], kind, sign))
        sign *= -1
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(BACKGROUND),
        "pairs": pairs,
    }


def _draw_arc(draw: ImageDraw.ImageDraw, arc: dict, fill: tuple[int, int, int]) -> None:
    cx, cy = arc["circle_center"]
    radius = float(arc["radius"])
    start = float(arc["mid_angle"]) - float(arc["span"]) / 2.0
    end = float(arc["mid_angle"]) + float(arc["span"]) / 2.0
    box = [cx - radius, cy - radius, cx + radius, cy + radius]
    draw.arc(box, start=start, end=end, fill=fill, width=int(arc["width"]))


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (int(scene["width"]), int(scene["height"])),
        tuple(scene["background"]),
    )
    draw = ImageDraw.Draw(image)
    for pair in scene["pairs"]:
        color = tuple(pair["rgb"])
        for arc in pair["arcs"]:
            _draw_arc(draw, arc, color)
    return image


def analytic_gold(scene: dict) -> str:
    # Deliberately inline: the only top-level label input is the fully rendered
    # ``pairs`` field, which the latent rendered-field audit can observe.
    scored = []
    for pair in scene["pairs"]:
        a, b = pair["arcs"]
        center_gap = math.hypot(
            float(a["circle_center"][0]) - float(b["circle_center"][0]),
            float(a["circle_center"][1]) - float(b["circle_center"][1]),
        )
        radius_gap = abs(float(a["radius"]) - float(b["radius"]))
        scored.append((center_gap + 2.0 * radius_gap, str(pair["color"])))
    scored.sort()
    if len(scored) < 2 or scored[1][0] - scored[0][0] <= 0.0:
        raise ValueError("scene has no unique cocircular pair")
    return scored[0][1]


def _scores(scene: dict) -> list[tuple[float, str]]:
    values = []
    for pair in scene["pairs"]:
        first, second = pair["arcs"]
        c1 = first["circle_center"]
        c2 = second["circle_center"]
        center_gap = math.hypot(float(c1[0]) - float(c2[0]), float(c1[1]) - float(c2[1]))
        radius_gap = abs(float(first["radius"]) - float(second["radius"]))
        values.append((center_gap + 2.0 * radius_gap, str(pair["color"])))
    return sorted(values)


def margin(scene: dict) -> float:
    scores = _scores(scene)
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for pair in twin["pairs"]:
        pair["arcs"].reverse()
    return [("reverse_same_color_arc_draw_order", twin)]
