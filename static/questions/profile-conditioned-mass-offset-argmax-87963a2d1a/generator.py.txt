"""Mass-offset (lopsidedness) selection world: which color's solid is most off-center.

Latent scene z carries three L-shaped solid regions (one red, one blue, one green),
each a union of two axis-aligned rectangles forming an "L" that occupies its own
horizontal column. All three shapes share the SAME bounding-box footprint (the
same width and the same height), but the inner arm geometry of each L differs, so
its center of mass (area centroid) sits at a different horizontal/vertical offset
from the center of its own bounding box.

The task is a pure measurement: for every color, recover the distance between the
color's own center of mass and its own bounding-box center, then select the color
with the largest recovered offset (the most lopsided shape).

Because every shape shares one identical bounding box, the decision is NOT
recoverable from absolute size, footprint, position, or total ink: those are
either held constant across the three shapes or shuffled independently of the
offset. Only an internal 2D mass-eccentricity measurement separates the colors.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "mass_offset_argmax-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)

WIDTH = 560
HEIGHT = 300

# Each L sits in its own column with one SHARED bounding-box footprint (bbox_w x bbox_h).
BBOX_W = 150.0
BBOX_H = 200.0
COL_X0 = (30.0, 205.0, 380.0)
COL_Y0 = 45.0

# Per-shape inner arm geometry ranges.
A_MIN, A_MAX = 40.0, 110.0   # vertical column (left leg) width
C_MIN, C_MAX = 40.0, 160.0   # bottom arm height

# Top-two measured offset gap required for a decidable selection (pixels).
MIN_MARGIN_PX = 4.0
# Below this top-two offset gap the scene is quarantined as near-boundary yet
# still decidable (well above pixel-fit noise).
QUARANTINE_PX = 3.0


def _offset(a: float, b: float, c: float, h: float) -> float:
    """Distance between the L's area centroid and its bounding-box center.

    The L is the union of a full-height left column M_a = [0,a]x[0,h] and a
    bottom arm M_b = [0,b]x[0,c] (overlap [0,a]x[0,c]). Union area and first
    moments are summed with the overlap subtracted exactly once.
    """
    area = a * h + b * c - a * c
    cx = (a * h * (a / 2.0) + b * c * (b / 2.0) - a * c * (a / 2.0)) / area
    cy = (a * h * (h / 2.0) + b * c * (c / 2.0) - a * c * (c / 2.0)) / area
    return float(math.hypot(cx - b / 2.0, cy - h / 2.0))


def _shape_metrics(shape: dict) -> tuple[float, float, float]:
    """Return (offset, top-gap position holder, value) for an analytic shape."""
    offset = _offset(shape["a"], BBOX_W, shape["c"], BBOX_H)
    return offset, shape["color"], offset


def sample_scene(seed: int) -> dict:
    """Sample a deterministic scene with a guaranteed top-two offset gap.

    Returns a plain dict (JSON-serialisable). ``shapes`` is a list of
    {"color", "x0", "y0", "a", "c"} where x0/y0 is the lower-left of the L's
    bounding box and (a, c) are the inner arm geometry.
    """
    rng = random.Random(seed * 1009 + 7)
    while True:
        shapes = []
        cols = list(COL_X0)
        rng.shuffle(cols)
        colors = list(PALETTE)
        rng.shuffle(colors)
        offsets = []
        for color, x0 in zip(colors, cols):
            a = rng.uniform(A_MIN, A_MAX)
            c = rng.uniform(C_MIN, C_MAX)
            off = _offset(a, BBOX_W, c, BBOX_H)
            shapes.append(
                {
                    "color": color,
                    "x0": round(x0, 2),
                    "y0": COL_Y0,
                    "a": round(a, 2),
                    "c": round(c, 2),
                }
            )
            offsets.append(off)
        sorted_offsets = sorted(offsets, reverse=True)
        if len(sorted_offsets) >= 2 and sorted_offsets[0] - sorted_offsets[1] >= MIN_MARGIN_PX:
            break
    return {
        "shapes": shapes,
        "width": WIDTH,
        "height": HEIGHT,
        "background": "light",
    }


def render(scene: dict) -> Image.Image:
    """Render I = R(z): three L-shaped solid regions in their own columns."""
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for shape in scene["shapes"]:
        color = COLOR_RGB[shape["color"]]
        x0, y0 = shape["x0"], shape["y0"]
        a, c = shape["a"], shape["c"]
        # Left vertical leg (full height) and bottom horizontal arm (full width):
        # their union is the L.
        draw.rectangle([x0, y0, x0 + a, y0 + BBOX_H], fill=color)
        draw.rectangle([x0, y0, x0 + BBOX_W, y0 + c], fill=color)
    return img


def _most_lopsided(shapes: list[dict]) -> tuple[dict, float, float]:
    """Return (winning shape, its offset, top-two offset gap)."""
    keyed = sorted(shapes, key=lambda s: -_offset(s["a"], BBOX_W, s["c"], BBOX_H))
    top, second = keyed[0], keyed[1]
    top_off = _offset(top["a"], BBOX_W, top["c"], BBOX_H)
    second_off = _offset(second["a"], BBOX_W, second["c"], BBOX_H)
    return top, top_off, float(top_off - second_off)


def analytic_gold(scene: dict) -> str:
    """The color whose shape is most lopsided (largest own centroid offset)."""
    top, _off, _gap = _most_lopsided(scene["shapes"])
    return top["color"]


def margin(scene: dict) -> float:
    """Verifiability margin: the top-two centroid-offset gap in pixels."""
    _top, _off, gap = _most_lopsided(scene["shapes"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    """Near-boundary: the top-two offset gap is finite-resolution small."""
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list:
    """not_applicable: no nontrivial pixel-identical latent symmetry.

    Every latent field (each shape's color label and its (x0, y0, a, c) arm
    geometry) is directly rendered as visible filled ink. Changing a shape's arm
    geometry moves its centroid and thus changes which color is most lopsided,
    and the three shapes share one bounding box so the decision is an internal
    mass measurement, not a global transform. Any global flip/rotation turns the
    raster into a different image. So no hidden scene field can change the gold
    without changing the raster.
    """
    return []
