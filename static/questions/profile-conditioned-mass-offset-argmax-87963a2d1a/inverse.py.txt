"""Independent pixel-only inverse arm for the mass-offset world.

This module receives ONLY the rasterized PNG plus the public palette (the world's
fixed naming convention) and re-measures which color's L-shaped solid has the
largest distance between its own center of mass and its own bounding-box center.
It shares no geometry with renderer.render() and never imports the renderer,
prompts, generator, verifier, or any analytic-gold code.
"""

from __future__ import annotations

from typing import Any

import numpy as np
from PIL import Image

ORACLE_VERSION = "mass_offset_argmax-oracle-0.1.0"

PALETTE = ["red", "blue", "green"]
COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
    "green": (30, 150, 70),
}
BACKGROUND = (250, 250, 250)


def _nearest_color(arr: np.ndarray) -> np.ndarray:
    """Label every pixel to its nearest palette entry (segment colors + background).

    argmin leaves no unclassified antialias band: edge blends split at their
    ~50% point, dominated by the background, so the labeled solid masks stay
    clean regardless of any anti-aliasing the renderer introduced.
    """
    refs = np.stack(
        [np.array(COLOR_RGB[c], dtype=np.float32) for c in PALETTE]
        + [np.array(BACKGROUND, dtype=np.float32)],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _offset_from_mask(mask: np.ndarray) -> float:
    """Recover a color's own off-center offset: |centroid - bbox-center|.

    centroid is the pixel mean of the color's filled region (its area center of
    mass); the bounding-box center is the midpoint of the tight bounding box of
    that same filled region. The L's shape makes these two points differ; the
    returned scalar is their Euclidean separation, measured purely from pixels.
    """
    ys, xs = np.nonzero(mask)
    if len(ys) < 10:
        return -1.0
    cx = float(xs.mean())
    cy = float(ys.mean())
    bx = (float(xs.min()) + float(xs.max())) / 2.0
    by = (float(ys.min()) + float(ys.max())) / 2.0
    return float(np.hypot(cx - bx, cy - by))


def decision_from_image(image: Image.Image) -> str:
    """Return the color of the most lopsided (largest own centroid offset) shape."""
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _nearest_color(arr)
    best_color = PALETTE[0]
    best_offset = -1.0
    for idx, color in enumerate(PALETTE):
        mask = labels == idx
        offset = _offset_from_mask(mask)
        if offset > best_offset:
            best_offset = offset
            best_color = color
    return best_color
