"""Independent raster-only inverse for standardized_reference_offset.

This module intentionally has no access to latent data, renderer values, prompts,
generation code, filenames, or metadata. Geometry is recovered from exact RGB
regions in the supplied Pillow image.
"""

from __future__ import annotations

from PIL import Image
import numpy as np

REFERENCE_RGB = (92, 99, 108)
COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "red": (211, 55, 65),
    "blue": (45, 105, 205),
    "green": (35, 148, 92),
    "orange": (231, 132, 31),
}
MIN_TOP_GAP = 0.020


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2)


def _reference_x(array: np.ndarray) -> float:
    mask = _mask(array, REFERENCE_RGB)
    ys, xs = np.nonzero(mask)
    if xs.size < 40:
        raise ValueError("dashed reference is absent or too corrupted")
    counts = np.bincount(xs, minlength=array.shape[1])
    active = np.flatnonzero(counts >= 20)
    if active.size < 1 or active[-1] - active[0] > 4:
        raise ValueError("dashed reference does not form one vertical band")
    return float(active.mean())


def _row_measurement(array: np.ndarray, rgb: tuple[int, int, int]) -> tuple[float, float, float]:
    mask = _mask(array, rgb)
    ys, xs = np.nonzero(mask)
    if xs.size < 500:
        raise ValueError("a colored interval is absent or materially corrupted")
    left = int(xs.min())
    right = int(xs.max())
    length = right - left
    if length < 80:
        raise ValueError("colored interval is too short")

    column_counts = mask.sum(axis=0)
    inner = np.arange(left + 6, right - 5)
    marker_columns = inner[column_counts[inner] >= 20]
    if marker_columns.size < 3 or marker_columns[-1] - marker_columns[0] > 12:
        raise ValueError("diamond marker is absent, duplicated, or malformed")
    marker_x = float(marker_columns.mean())

    # Endcaps must remain visible; this also makes erasing either interval end fail.
    if column_counts[left : left + 3].max(initial=0) < 15:
        raise ValueError("left interval endcap is not visible")
    if column_counts[right - 2 : right + 1].max(initial=0) < 15:
        raise ValueError("right interval endcap is not visible")
    return marker_x, float(left), float(right)


def decision_from_image(image: Image.Image) -> str:
    """Recover all four ratios from pixels and return the unique winning color."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 100 or array.shape[1] < 100:
        raise ValueError("image is too small")
    reference_x = _reference_x(array)
    scores: list[tuple[float, str]] = []
    for name, rgb in COLOR_RGB.items():
        marker_x, left, right = _row_measurement(array, rgb)
        scores.append((abs(marker_x - reference_x) / (right - left), name))
    scores.sort(reverse=True)
    if scores[0][0] - scores[1][0] < MIN_TOP_GAP:
        raise ValueError("pixel decision is ambiguous at the top-two boundary")
    return scores[0][1]
