"""Latent sampler, Pillow renderer, and analytic arm for one visual world."""

from __future__ import annotations

import copy
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 480
REFERENCE_RGB = (92, 99, 108)
INK_RGB = (35, 40, 48)
GRID_RGB = (226, 229, 234)
BACKGROUND_RGB = (250, 250, 248)
COLOR_RGB: dict[str, tuple[int, int, int]] = {
    "red": (211, 55, 65),
    "blue": (45, 105, 205),
    "green": (35, 148, 92),
    "orange": (231, 132, 31),
}
COLORS = tuple(COLOR_RGB)
QUARANTINE_GAP_PP = 3.0


def _ranked_scores(scene: dict[str, Any]) -> list[tuple[float, str]]:
    reference_x = int(scene["reference_x"])
    rows = scene["rows"]
    return sorted(
        (
            (
                abs(int(row["marker_x"]) - reference_x)
                / (int(row["right"]) - int(row["left"])),
                str(row["color"]),
            )
            for row in rows
        ),
        reverse=True,
    )


def sample_scene(seed: int) -> dict:
    """Sample a balanced, fully visible scene from one integer seed."""

    rng = random.Random(int(seed))
    reference_x = 350 + rng.randint(-18, 18)
    target_color = COLORS[int(seed) % len(COLORS)]

    # Every sixth seed approaches (but stays outside) the declared tie quarantine.
    if int(seed) % 6 == 0:
        desired = [0.300, 0.245, 0.155, 0.085]
    elif int(seed) % 5 == 0:
        # Scheduled cue-conflict case: scale normalization reverses raw displacement.
        desired = [0.330, 0.250, 0.165, 0.090]
    else:
        desired = [rng.uniform(0.335, 0.365), rng.uniform(0.225, 0.250),
                   rng.uniform(0.140, 0.185), rng.uniform(0.070, 0.115)]
    other_scores = desired[1:]
    rng.shuffle(other_scores)
    score_for = {target_color: desired[0]}
    for color, value in zip((c for c in COLORS if c != target_color), other_scores):
        score_for[color] = value

    # Length rank and y position are each permuted independently of the answer color.
    lengths = [232, 258, 284, 310]
    if int(seed) % 5 == 0:
        second_color = max((c for c in COLORS if c != target_color), key=score_for.__getitem__)
        remaining_colors = [c for c in COLORS if c not in (target_color, second_color)]
        remaining_lengths = [258, 284]
        rng.shuffle(remaining_lengths)
        length_for = {target_color: 232, second_color: 310}
        length_for.update(zip(remaining_colors, remaining_lengths))
    else:
        rng.shuffle(lengths)
        length_for = dict(zip(COLORS, lengths))
    slots = [112, 202, 292, 382]
    rng.shuffle(slots)
    rows = []
    for index, color in enumerate(COLORS):
        length = length_for[color]
        # The reference sits well inside every interval, but not always at its midpoint.
        reference_fraction = rng.uniform(0.46, 0.54)
        left = reference_x - round(reference_fraction * length)
        right = left + length
        offset = round(score_for[color] * length)
        side = -1 if rng.randrange(2) == 0 else 1
        marker_x = reference_x + side * offset
        if not left + 16 <= marker_x <= right - 16:
            side *= -1
            marker_x = reference_x + side * offset
        rows.append(
            {
                "color": color,
                "y": slots[index],
                "left": left,
                "right": right,
                "marker_x": marker_x,
            }
        )

    scene = {
        "width": WIDTH,
        "height": HEIGHT,
        "reference_x": reference_x,
        "rows": rows,
    }
    if is_quarantined(scene):
        raise RuntimeError(f"sampler produced a quarantined scene for seed {seed}")
    return scene


def render(scene: dict) -> Image.Image:
    """Render only the visible fields of the latent scene to a deterministic RGB PNG."""

    width = int(scene["width"])
    height = int(scene["height"])
    reference_x = int(scene["reference_x"])
    rows = scene["rows"]
    image = Image.new("RGB", (width, height), BACKGROUND_RGB)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()

    draw.text((22, 17), "STANDARDIZED REFERENCE OFFSET", fill=INK_RGB, font=font)
    draw.text(
        (22, 35),
        "score = |diamond center - dashed reference| / full interval length",
        fill=INK_RGB,
        font=font,
    )
    draw.text((reference_x - 30, 58), "REFERENCE", fill=REFERENCE_RGB, font=font)

    for y in (112, 202, 292, 382):
        draw.line((82, y, width - 35, y), fill=GRID_RGB, width=1)
    for top in range(73, 430, 16):
        draw.rectangle((reference_x - 1, top, reference_x + 1, min(top + 8, 430)), fill=REFERENCE_RGB)

    for row in rows:
        color_name = str(row["color"])
        rgb = COLOR_RGB[color_name]
        y = int(row["y"])
        left = int(row["left"])
        right = int(row["right"])
        marker_x = int(row["marker_x"])
        draw.text((24, y - 6), color_name.upper(), fill=INK_RGB, font=font)
        draw.rectangle((left, y - 2, right, y + 2), fill=rgb)
        draw.rectangle((left, y - 8, left + 2, y + 8), fill=rgb)
        draw.rectangle((right - 2, y - 8, right, y + 8), fill=rgb)
        draw.polygon(
            (
                (marker_x, y - 13),
                (marker_x + 13, y),
                (marker_x, y + 13),
                (marker_x - 13, y),
            ),
            fill=rgb,
        )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the color with the largest normalized absolute offset."""

    return _ranked_scores(scene)[0][1]


def margin(scene: dict) -> float:
    """Top-two ratio gap in percentage points."""

    ranked = _ranked_scores(scene)
    return 100.0 * (ranked[0][0] - ranked[1][0])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) < QUARANTINE_GAP_PP:
        return True
    reference_x = int(scene["reference_x"])
    for row in scene["rows"]:
        left = int(row["left"])
        right = int(row["right"])
        marker_x = int(row["marker_x"])
        if right - left < 80 or not left + 15 <= marker_x <= right - 15:
            return True
        if abs(marker_x - reference_x) < 15:
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse disjoint latent draw records; pixels and color-bound gold are unchanged."""

    twin = copy.deepcopy(scene)
    twin["rows"] = list(reversed(twin["rows"]))
    return [("reverse_disjoint_row_records", twin)]
