"""Pixel-only decision arm. Receives a rendered image and nothing else.

It locates the two rails, the landmark ticks on each rail, the pointer, and the
coloured dots directly from pixels, rebuilds the piecewise-uniform transfer from
the tick columns, and names the colour of the dot nearest the transferred
position. No renderer, prompt, generator, verifier, or gold code is imported and
no latent scene is accepted.
"""

from __future__ import annotations

from PIL import Image

_REFERENCE_COLOURS = {
    "red": (214, 40, 40),
    "blue": (37, 84, 204),
    "green": (26, 140, 60),
    "orange": (240, 150, 20),
    "purple": (140, 60, 180),
    "brown": (128, 80, 40),
}
_COLOUR_TOLERANCE = 60
_DARK = 110
_GREY_SPREAD = 26


def _is_dark_neutral(pixel: tuple[int, int, int]) -> bool:
    r, g, b = pixel
    return max(r, g, b) < _DARK and max(r, g, b) - min(r, g, b) <= _GREY_SPREAD


def _runs(columns: list[int]) -> list[float]:
    groups: list[list[int]] = []
    for x in sorted(columns):
        if groups and x - groups[-1][-1] <= 2:
            groups[-1].append(x)
        else:
            groups.append([x])
    return [sum(group) / float(len(group)) for group in groups]


def _rail_rows(pixels, width: int, height: int) -> list[int]:
    counts = []
    for y in range(height):
        dark = sum(1 for x in range(0, width, 2) if _is_dark_neutral(pixels[x, y]))
        counts.append(dark)
    threshold = 0.6 * (width / 2)
    rows = [y for y, count in enumerate(counts) if count >= threshold]
    bands: list[list[int]] = []
    for y in rows:
        if bands and y - bands[-1][-1] <= 2:
            bands[-1].append(y)
        else:
            bands.append([y])
    return [int(round(sum(band) / float(len(band)))) for band in bands]


def _ticks(pixels, width: int, row: int) -> list[float]:
    columns = [x for x in range(width) if _is_dark_neutral(pixels[x, row])]
    return _runs(columns)


def _pointer_x(pixels, width: int, top_rail: int) -> float:
    total = 0.0
    count = 0
    for y in range(max(0, top_rail - 40), top_rail - 10):
        for x in range(width):
            if _is_dark_neutral(pixels[x, y]):
                total += x
                count += 1
    if not count:
        raise ValueError("pointer not found")
    return total / count


def _dots(pixels, width: int, height: int, bottom_rail: int) -> list[tuple[float, str]]:
    found: dict[str, list[int]] = {}
    for name, reference in _REFERENCE_COLOURS.items():
        columns = []
        for y in range(bottom_rail, min(height, bottom_rail + 60)):
            for x in range(width):
                r, g, b = pixels[x, y]
                if (
                    abs(r - reference[0]) <= _COLOUR_TOLERANCE
                    and abs(g - reference[1]) <= _COLOUR_TOLERANCE
                    and abs(b - reference[2]) <= _COLOUR_TOLERANCE
                ):
                    columns.append(x)
        if len(columns) >= 30:
            found[name] = columns
    return [(sum(cols) / float(len(cols)), name) for name, cols in found.items()]


def _transfer(top: list[float], bottom: list[float], x: float) -> float:
    for i in range(len(top) - 1):
        if top[i] <= x <= top[i + 1]:
            frac = (x - top[i]) / (top[i + 1] - top[i])
            return bottom[i] + frac * (bottom[i + 1] - bottom[i])
    return bottom[-1] if x > top[-1] else bottom[0]


def decision_from_image(image: Image.Image) -> str:
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()

    rails = _rail_rows(pixels, width, height)
    if len(rails) < 2:
        raise ValueError("could not locate two rails")
    top_rail, bottom_rail = rails[0], rails[-1]

    top_ticks = _ticks(pixels, width, top_rail - 7)
    bottom_ticks = _ticks(pixels, width, bottom_rail - 7)
    if len(top_ticks) != len(bottom_ticks) or len(top_ticks) < 2:
        raise ValueError("landmark ticks do not correspond")

    pointer = _pointer_x(pixels, width, top_rail)
    target = _transfer(top_ticks, bottom_ticks, pointer)

    dots = _dots(pixels, width, height, bottom_rail)
    if len(dots) < 2:
        raise ValueError("insufficient coloured dots")
    best = min(dots, key=lambda dot: abs(dot[0] - target))
    return best[1]
