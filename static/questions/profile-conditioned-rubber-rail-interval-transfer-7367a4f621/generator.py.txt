"""Rubber-rail interval transfer: sampler, renderer, analytic gold, margin.

Two horizontal rails carry the same five landmark ticks; grey tie-lines join
matching ticks. Between consecutive ticks the lower rail is a uniformly
stretched copy of the upper rail, and the stretch factor differs from interval
to interval, so a single global scale or offset cannot transfer a position.
A black pointer marks a spot on the upper rail; the answer is the colour of the
lower-rail dot standing at the transferred spot.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 760
HEIGHT = 380
X0 = 70
X1 = 690
Y_TOP = 110
Y_BOTTOM = 280

TICK_HALF = 10
DOT_RADIUS = 8
DOT_DY = 22
POINTER_APEX_DY = 14
POINTER_BASE_DY = 32
POINTER_HALF = 9

BACKGROUND = (255, 255, 255)
RAIL = (60, 60, 60)
TICK = (0, 0, 0)
TIE = (170, 170, 170)
POINTER = (0, 0, 0)

PALETTE = {
    "red": (214, 40, 40),
    "blue": (37, 84, 204),
    "green": (26, 140, 60),
    "orange": (240, 150, 20),
    "purple": (140, 60, 180),
    "brown": (128, 80, 40),
}
COLOR_NAMES = tuple(sorted(PALETTE))

QUARANTINE_MARGIN = 30.0
MIN_DOT_SEPARATION = 22


def _sample_anchors(rng: random.Random) -> list[int]:
    while True:
        interior = sorted(rng.randint(X0 + 70, X1 - 70) for _ in range(3))
        anchors = [X0] + interior + [X1]
        if all(anchors[i + 1] - anchors[i] >= 70 for i in range(4)):
            return anchors


def _transfer(top: list[int], bottom: list[int], x: float) -> float:
    for i in range(len(top) - 1):
        if top[i] <= x <= top[i + 1]:
            frac = (x - top[i]) / float(top[i + 1] - top[i])
            return bottom[i] + frac * (bottom[i + 1] - bottom[i])
    return float(bottom[-1] if x > top[-1] else bottom[0])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    while True:
        top = _sample_anchors(rng)
        bottom = _sample_anchors(rng)
        ratios = [
            (bottom[i + 1] - bottom[i]) / float(top[i + 1] - top[i]) for i in range(4)
        ]
        if max(ratios) < 1.5 or min(ratios) > 0.7:
            continue

        pointer = rng.randint(X0 + 20, X1 - 20)
        if min(abs(pointer - a) for a in top) < 18:
            continue
        target = _transfer(top, bottom, pointer)
        if not X0 + 20 <= target <= X1 - 20:
            continue
        aligned = rng.random() < 0.25
        displacement = abs(target - pointer)
        if aligned and displacement > 14:
            continue
        if not aligned and displacement < 55:
            continue

        target_rank = rng.randrange(4)
        tight = rng.random() < 0.25
        placed = _place_distractors(rng, target, tight)
        if placed is None:
            continue
        if sum(1 for x in placed if x < target) != target_rank:
            continue

        names = rng.sample(COLOR_NAMES, 4)
        answer_index = rng.randrange(4)
        answer_name = names[answer_index]
        others = [name for name in names if name != answer_name]
        dots = [{"x": int(round(target)), "color": answer_name}]
        for x, name in zip(placed, others):
            dots.append({"x": int(x), "color": name})
        dots.sort(key=lambda dot: dot["x"])
        rng.shuffle(dots)
        return {
            "top_anchors": top,
            "bottom_anchors": bottom,
            "pointer_x": pointer,
            "dots": dots,
        }


def _place_distractors(rng: random.Random, target: float, tight: bool) -> list[int] | None:
    close = rng.randint(20, 44) if tight else rng.randint(46, 130)
    for _ in range(80):
        side = rng.choice((-1, 1))
        first = int(round(target + side * close))
        if not X0 + 15 <= first <= X1 - 15:
            continue
        placed = [first]
        ok = True
        for _ in range(2):
            for _attempt in range(60):
                x = rng.randint(X0 + 15, X1 - 15)
                if abs(x - target) <= close:
                    continue
                if any(abs(x - other) < MIN_DOT_SEPARATION for other in placed):
                    continue
                placed.append(x)
                break
            else:
                ok = False
                break
        if ok and len(placed) == 3:
            return placed
    return None


def _target_position(scene: dict) -> float:
    return _transfer(scene["top_anchors"], scene["bottom_anchors"], scene["pointer_x"])


def analytic_gold(scene: dict) -> str:
    target = _target_position(scene)
    best = min(scene["dots"], key=lambda dot: abs(dot["x"] - target))
    return str(best["color"])


def margin(scene: dict) -> float:
    target = _target_position(scene)
    distances = sorted(abs(dot["x"] - target) for dot in scene["dots"])
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    dots = list(scene["dots"])
    rotated = dict(scene, dots=dots[1:] + dots[:1])
    reversed_dots = dict(scene, dots=list(reversed(dots)))
    return [("dot_list_rotation", rotated), ("dot_list_reversal", reversed_dots)]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    top = scene["top_anchors"]
    bottom = scene["bottom_anchors"]

    for a, b in zip(top, bottom):
        draw.line(
            [(a, Y_TOP + TICK_HALF), (b, Y_BOTTOM - TICK_HALF)], fill=TIE, width=2
        )

    for y in (Y_TOP, Y_BOTTOM):
        draw.rectangle([X0 - 6, y - 1, X1 + 6, y + 1], fill=RAIL)

    for a in top:
        draw.rectangle([a - 1, Y_TOP - TICK_HALF, a + 1, Y_TOP + TICK_HALF], fill=TICK)
    for b in bottom:
        draw.rectangle(
            [b - 1, Y_BOTTOM - TICK_HALF, b + 1, Y_BOTTOM + TICK_HALF], fill=TICK
        )

    px = scene["pointer_x"]
    draw.polygon(
        [
            (px, Y_TOP - POINTER_APEX_DY),
            (px - POINTER_HALF, Y_TOP - POINTER_BASE_DY),
            (px + POINTER_HALF, Y_TOP - POINTER_BASE_DY),
        ],
        fill=POINTER,
    )

    for dot in scene["dots"]:
        x = dot["x"]
        colour = PALETTE[dot["color"]]
        draw.rectangle([x - 1, Y_BOTTOM + 2, x + 1, Y_BOTTOM + DOT_DY], fill=RAIL)
        draw.ellipse(
            [
                x - DOT_RADIUS,
                Y_BOTTOM + DOT_DY - DOT_RADIUS,
                x + DOT_RADIUS,
                Y_BOTTOM + DOT_DY + DOT_RADIUS,
            ],
            fill=colour,
        )
    return image
