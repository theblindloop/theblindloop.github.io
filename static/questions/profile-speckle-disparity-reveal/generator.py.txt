"""Scene sampling, rendering and analytic gold for the speckle-disparity world.

Two speckle plates are printed side by side. The right plate repeats the left
plate's speckle cell by cell; in the cells of the answer symbol the copy is
taken from a strip that lies a printed distance further right (the row is
treated as a loop). Neither plate alone carries the symbol: every cell of both
plates is the same unbiased speckle, so the symbol exists only in the
correspondence between the two plates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 7
DOTS_PER_CELL = 7
DOT = 2
CELL = DOTS_PER_CELL * DOT
PANEL = GRID * CELL
BORDER = 3
CANVAS = (400, 200)
GAUGE_H = 8
GAUGE_W = 2
QUARANTINE_MARGIN = 3.0

SYMBOLS: tuple[str, ...] = ("crescent", "funnel", "pinwheel", "spiral", "staircase")

TEMPLATES: dict[str, tuple[str, ...]] = {
    "crescent": (
        "..###..",
        ".##....",
        ".#.....",
        ".#.....",
        ".#.....",
        ".##....",
        "..###..",
    ),
    "funnel": (
        "#######",
        ".#####.",
        "..###..",
        "...#...",
        "...#...",
        "...#...",
        "..###..",
    ),
    "pinwheel": (
        "##.....",
        "##.....",
        "##.####",
        "...#...",
        "####.##",
        ".....##",
        ".....##",
    ),
    "spiral": (
        ".#####.",
        ".#.....",
        ".#.###.",
        ".#.#.#.",
        ".#...#.",
        ".#####.",
        ".......",
    ),
    "staircase": (
        ".....##",
        ".....#.",
        "...###.",
        "...#...",
        ".###...",
        ".#.....",
        "##.....",
    ),
}

FRAME_COLORS: tuple[tuple[int, int, int], ...] = (
    (30, 90, 200),
    (180, 45, 45),
    (25, 130, 75),
    (150, 60, 175),
    (200, 120, 20),
)

SHIFT_DOTS: tuple[int, ...] = (2, 3, 4)


def _template_mask(name: str) -> np.ndarray:
    rows = TEMPLATES[name]
    return np.array([[1 if ch == "#" else 0 for ch in row] for row in rows], dtype=np.int8)


def _cells_to_mask(cells) -> np.ndarray:
    mask = np.zeros((GRID, GRID), dtype=np.int8)
    for row, col in cells:
        mask[int(row), int(col)] = 1
    return mask


def _rank(mask: np.ndarray) -> list[tuple[float, str]]:
    scored = [(float(np.sum(mask != _template_mask(name))), name) for name in SYMBOLS]
    scored.sort(key=lambda item: (item[0], item[1]))
    return scored


def sample_scene(seed: int) -> dict:
    rng = np.random.RandomState(seed % (2**32))
    symbol = SYMBOLS[int(rng.randint(len(SYMBOLS)))]
    mask = _template_mask(symbol).copy()
    for _ in range(int(rng.randint(0, 7))):
        row = int(rng.randint(GRID))
        col = int(rng.randint(GRID))
        mask[row, col] = 1 - mask[row, col]
    cells = [[int(r), int(c)] for r in range(GRID) for c in range(GRID) if mask[r, c]]

    gap = int(rng.randint(16, 41))
    block = PANEL + 2 * BORDER
    max_x = CANVAS[0] - (2 * block + gap) - 10
    origin_x = int(rng.randint(10, max(11, max_x + 1)))
    origin_y = int(rng.randint(40, CANVAS[1] - block - 9))
    gauge_y = int(rng.randint(12, 31))
    shift = int(SHIFT_DOTS[int(rng.randint(len(SHIFT_DOTS)))]) * DOT
    frame = FRAME_COLORS[int(rng.randint(len(FRAME_COLORS)))]
    tone = int(rng.randint(0, 4))
    return {
        "grid": GRID,
        "cell": CELL,
        "dot": DOT,
        "shift": shift,
        "slid_cells": cells,
        "texture_seed": int(rng.randint(1, 2**31 - 1)),
        "origin_x": origin_x,
        "origin_y": origin_y,
        "gap": gap,
        "gauge_y": gauge_y,
        "gauge_x": int(rng.randint(40, CANVAS[0] - 60)),
        "bg": [232 + tone * 4] * 3,
        "paper": [246 + tone] * 3,
        "ink": [30 + tone * 6] * 3,
        "frame": list(frame),
        "seed_tag": int(seed),
    }


def _textures(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    """Dot-level left and right speckle fields."""
    side = GRID * DOTS_PER_CELL
    rng = np.random.RandomState(int(scene["texture_seed"]) % (2**32))
    left = rng.randint(0, 2, size=(side, side)).astype(np.int8)
    shift_dots = int(scene["shift"]) // DOT
    rolled = np.roll(left, -shift_dots, axis=1)
    right = left.copy()
    for row, col in scene["slid_cells"]:
        r0 = int(row) * DOTS_PER_CELL
        c0 = int(col) * DOTS_PER_CELL
        right[r0 : r0 + DOTS_PER_CELL, c0 : c0 + DOTS_PER_CELL] = rolled[
            r0 : r0 + DOTS_PER_CELL, c0 : c0 + DOTS_PER_CELL
        ]
    return left, right


def _paint_plate(canvas: np.ndarray, dots: np.ndarray, x0: int, y0: int, scene: dict) -> None:
    frame = np.array(scene["frame"], dtype=np.uint8)
    canvas[y0 - BORDER : y0 + PANEL + BORDER, x0 - BORDER : x0 + PANEL + BORDER] = frame
    field = np.kron(dots, np.ones((DOT, DOT), dtype=np.int8))
    ink = np.array(scene["ink"], dtype=np.uint8)
    paper = np.array(scene["paper"], dtype=np.uint8)
    block = np.where(field[:, :, None] == 1, ink[None, None, :], paper[None, None, :])
    canvas[y0 : y0 + PANEL, x0 : x0 + PANEL] = block.astype(np.uint8)


def render(scene: dict) -> Image.Image:
    width, height = CANVAS
    canvas = np.zeros((height, width, 3), dtype=np.uint8)
    canvas[:, :] = np.array(scene["bg"], dtype=np.uint8)
    left, right = _textures(scene)
    x0 = int(scene["origin_x"]) + BORDER
    y0 = int(scene["origin_y"]) + BORDER
    _paint_plate(canvas, left, x0, y0, scene)
    x1 = x0 + PANEL + 2 * BORDER + int(scene["gap"])
    _paint_plate(canvas, right, x1, y0, scene)

    frame = np.array(scene["frame"], dtype=np.uint8)
    gx = int(scene["gauge_x"])
    gy = int(scene["gauge_y"])
    canvas[gy : gy + GAUGE_H, gx : gx + GAUGE_W] = frame
    x2 = gx + int(scene["shift"])
    canvas[gy : gy + GAUGE_H, x2 : x2 + GAUGE_W] = frame
    return Image.fromarray(canvas, mode="RGB")


def analytic_gold(scene: dict) -> str:
    return _rank(_cells_to_mask(scene["slid_cells"]))[0][1]


def margin(scene: dict) -> float:
    ranked = _rank(_cells_to_mask(scene["slid_cells"]))
    return float(ranked[1][0] - ranked[0][0])


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Scene rewrites that are pixel-identical and must preserve the gold."""
    out: list[tuple[str, dict]] = []
    reordered = dict(scene)
    reordered["slid_cells"] = list(reversed(scene["slid_cells"]))
    out.append(("slid_cell_order_reversed", reordered))
    if scene["slid_cells"]:
        duplicated = dict(scene)
        duplicated["slid_cells"] = list(scene["slid_cells"]) + [list(scene["slid_cells"][0])]
        out.append(("slid_cell_repeated", duplicated))
    retagged = dict(scene)
    retagged["seed_tag"] = int(scene["seed_tag"]) + 991
    out.append(("sampling_tag_shifted", retagged))
    return out
