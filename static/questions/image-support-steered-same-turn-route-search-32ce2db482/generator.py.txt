"""Deterministic latent sampler and renderer for same-turn route search."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = HEIGHT = 640
BACKGROUND = [248, 247, 243]
BLACK = (34, 38, 43)
CYAN = (0, 170, 190)
TERMINALS = {
    "red": (214, 58, 63),
    "gold": (222, 159, 24),
    "violet": (132, 82, 190),
    "green": (42, 151, 91),
}
LENGTHS = (200, 45, 170, 20, 140)
MIXED_WORDS = ("LLLR", "LLRL", "LLRR", "RRLL", "RRLR", "RRRL")


def _turn_score(points: list[list[float]]) -> float:
    vectors = [(points[i+1][0]-points[i][0], points[i+1][1]-points[i][1]) for i in range(5)]
    signs = []
    for first, second in zip(vectors, vectors[1:]):
        cross = first[0] * second[1] - first[1] * second[0]
        signs.append(1 if cross > 0 else -1)
    return float(max(signs.count(1), signs.count(-1)))


def _points(word: str, heading: int) -> list[list[float]]:
    angle = heading % 4
    x = y = 0.0
    out = [[x, y]]
    for index, length in enumerate(LENGTHS):
        radians = math.radians(angle * 90)
        x += length * math.cos(radians)
        y += length * math.sin(radians)
        out.append([round(x, 6), round(y, 6)])
        if index < len(word):
            angle = (angle + (1 if word[index] == "R" else -1)) % 4
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = rng.randrange(4)
    target_color = tuple(TERMINALS)[int(seed) % 4]
    remaining = [c for c in TERMINALS if c != target_color]
    rng.shuffle(remaining)
    slot_colors = remaining[:]
    slot_colors.insert(target_slot, target_color)
    distractors = list(MIXED_WORDS)
    rng.shuffle(distractors)
    winner = rng.choice(("LLLL", "RRRR"))
    offset = [rng.randint(-10, 10), rng.randint(90, 110)]
    routes = []
    all_points = []
    di = 0
    for slot in range(4):
        word = winner if slot == target_slot else distractors[di]
        di += slot != target_slot
        raw = _points(word, rng.choice((0, 2)))
        cx = 320 + offset[0]
        cy = offset[1] + slot * 110 + 55
        min_x, max_x = min(p[0] for p in raw), max(p[0] for p in raw)
        min_y, max_y = min(p[1] for p in raw), max(p[1] for p in raw)
        dx = cx - (min_x + max_x) / 2
        dy = cy - (min_y + max_y) / 2
        points = [[p[0] + dx, p[1] + dy] for p in raw]
        routes.append({"terminal": slot_colors[slot], "slot": slot})
        all_points.append(points)
    return {"routes": routes, "points": all_points, "offset": offset, "canvas": [WIDTH, HEIGHT], "background": list(BACKGROUND)}


def render(scene: dict) -> Image.Image:
    width, height = [int(v) for v in scene["canvas"]]
    image = Image.new("RGB", (width, height), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    _ = scene["offset"]
    for route, points in zip(scene["routes"], scene["points"]):
        pts = [(round(x), round(y)) for x, y in points]
        draw.line(pts, fill=BLACK, width=1)
        sx, sy = pts[0]
        ex, ey = pts[-1]
        radius = 6
        draw.ellipse((sx-radius, sy-radius, sx+radius, sy+radius), fill=CYAN, outline=(255, 255, 255), width=1)
        draw.ellipse((ex-radius, ey-radius, ex+radius, ey+radius), fill=TERMINALS[route["terminal"]], outline=(255, 255, 255), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    scored = sorted(((_turn_score(points), route["terminal"]) for route, points in zip(scene["routes"], scene["points"])), reverse=True)
    return scored[0][1]


def margin(scene: dict) -> float:
    scores = sorted((_turn_score(points) for points in scene["points"]), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = sorted((_turn_score(points) for points in scene["points"]), reverse=True)
    return scores[0] != 4 or scores[1] >= 4


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
