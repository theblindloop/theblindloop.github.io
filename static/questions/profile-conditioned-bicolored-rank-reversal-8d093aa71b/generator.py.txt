"""Cross-panel rank-reversal selection world renderer.

A two-panel plot. Both panels share one gray baseline and contain one red and
one blue vertical bar of equal width standing on it. By construction one color
-- the designated ``winner`` -- is the taller bar in the left panel and the
shorter bar in the right panel, so its rank flips across the two panels. The
analytic decision is that flipping color, recovered independently from the
raster by counting each color's fill pixels per panel.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "bicolored_rank_reversal-0.1.0"

WIDTH = 360
HEIGHT = 360
BASELINE = 290
BAR_WIDTH = 24
LEFT_X = (68, 108)
RIGHT_X = (248, 288)
PANEL_SPLIT = 180

COLOR_RGB = {
    "red": (200, 40, 40),
    "blue": (40, 80, 200),
}
GRAY = (150, 150, 150)
WHITE = (255, 255, 255)

QUARANTINE_MARGIN = 0.08

COLORS = ["red", "blue"]


def _height_left(scene: dict, color: str) -> int:
    return int(scene["bars"][color]["left"])


def _height_right(scene: dict, color: str) -> int:
    return int(scene["bars"][color]["right"])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 104729 + (seed % 1000))
    winner = COLORS[seed % 2]
    bars = {}
    for color in COLORS:
        if color == winner:
            left = rng.randint(50, 66)
            right = rng.randint(22, 44)
        else:
            left = rng.randint(22, 42)
            right = rng.randint(50, 66)
        bars[color] = {"left": left, "right": right}
    return {"bars": bars}


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)
    for panel, xs in (("left", LEFT_X), ("right", RIGHT_X)):
        for color, center in zip(COLORS, xs):
            height = _height_left(scene, color) if panel == "left" else _height_right(scene, color)
            x0 = int(center - BAR_WIDTH / 2)
            x1 = int(center + BAR_WIDTH / 2)
            top = BASELINE - int(height)
            draw.rectangle([x0, top, x1, BASELINE], fill=COLOR_RGB[color])
        draw.line(
            [(xs[0] - 60, BASELINE), (xs[1] + 60, BASELINE)], fill=GRAY, width=2
        )
    return image


def analytic_gold(scene: dict) -> str:
    return max(COLORS, key=lambda color: _height_left(scene, color) - _height_right(scene, color))


def margin(scene: dict) -> float:
    winner = analytic_gold(scene)
    loser = "blue" if winner == "red" else "red"
    left_gap = _height_left(scene, winner) - _height_left(scene, loser)
    right_gap = _height_right(scene, loser) - _height_right(scene, winner)
    return float(min(left_gap, right_gap) / 66.0)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list:
    return []
