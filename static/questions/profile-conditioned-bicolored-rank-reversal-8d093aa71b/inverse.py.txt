"""Independent scene-free pixel oracle for the bicolored_rank_reversal world.

Recovers the crossing-panel rank-reversal color from the raster alone. The
oracle imports no renderer, prompt, generator, verifier, or analytic-gold code
and never sees the scene. It labels every pixel by its nearest palette color,
counts each color's fill pixels left and right of the panel split, and returns
the color with the larger left-minus-right fill difference (the color that is
tallest in the left panel and shortest in the right panel).
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PALETTE = {
    "red": (200, 40, 40),
    "blue": (40, 80, 200),
    "white": (255, 255, 255),
}
COLORS = ["red", "blue"]
PANEL_SPLIT = 180


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    rgb_list = [np.array(value, dtype=np.float32) for value in PALETTE.values()]
    labels = np.argmin(
        np.stack([np.linalg.norm(arr - color, axis=-1) for color in rgb_list], axis=-1),
        axis=-1,
    )
    left = labels[:, :PANEL_SPLIT]
    right = labels[:, PANEL_SPLIT:]
    diffs = {}
    for index, color in enumerate(COLORS):
        left_count = int((left == index).sum())
        right_count = int((right == index).sum())
        diffs[color] = left_count - right_count
    winner = max(COLORS, key=lambda color: (diffs[color], -COLORS.index(color)))
    return winner
