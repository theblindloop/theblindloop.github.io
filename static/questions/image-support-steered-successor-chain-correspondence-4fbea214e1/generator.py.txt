"""Deterministic analytic arm and renderer for Shuffled-Link Chain."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 720
BACKGROUND = (250, 248, 243)
PLAQUE = (229, 224, 210)
PANEL = (248, 246, 238)
INK = (43, 48, 56)
CLUE = (105, 116, 128)
ARROW = (44, 116, 179)
TAG_RGB = {
    "amber": (230, 156, 31),
    "cyan": (25, 177, 194),
    "violet": (139, 83, 205),
}
TAG_NAMES = tuple(TAG_RGB)
SHAPES = ("circle", "square", "triangle")

CLUE_BOXES = ((55, 65, 260, 205), (700, 65, 905, 205))
CANDIDATE_BOXES = ((40, 500, 250, 645), (375, 500, 585, 645), (710, 500, 920, 645))
FRAME = 8
SHAPE_RADIUS = 19


def _shape(value: object) -> str:
    if value not in SHAPES:
        raise ValueError(f"invalid shape {value!r}")
    return str(value)


def _scene_parts(scene: dict) -> tuple[list[list[str]], list[list[str]], list[str]]:
    raw_clues = scene["clues"]
    raw_candidates = scene["candidates"]
    raw_tags = scene["tag_colors"]
    if not isinstance(raw_clues, list) or len(raw_clues) != 2:
        raise ValueError("clues must contain two directed pairs")
    clues = []
    for clue in raw_clues:
        if not isinstance(clue, list) or len(clue) != 2:
            raise ValueError("each clue must be a pair")
        clues.append([_shape(item) for item in clue])
    if not isinstance(raw_candidates, list) or len(raw_candidates) != 3:
        raise ValueError("candidates must contain three sequences")
    candidates = []
    for candidate in raw_candidates:
        if not isinstance(candidate, list) or len(candidate) != 3:
            raise ValueError("each candidate must contain three shapes")
        candidates.append([_shape(item) for item in candidate])
    if (
        not isinstance(raw_tags, list)
        or len(raw_tags) != 3
        or set(raw_tags) != set(TAG_NAMES)
    ):
        raise ValueError("tag_colors must be one permutation of the answer colors")
    return clues, candidates, [str(tag) for tag in raw_tags]


def _join(clues: list[list[str]]) -> list[str]:
    if len(clues) != 2 or any(len(pair) != 2 for pair in clues):
        raise ValueError("exactly two directed pairs are required")
    successors: dict[str, str] = {}
    indegree: dict[str, int] = {}
    nodes: set[str] = set()
    for left, right in clues:
        if left == right or left in successors:
            raise ValueError("clues do not define a simple successor relation")
        successors[left] = right
        nodes.update((left, right))
        indegree[right] = indegree.get(right, 0) + 1
        indegree.setdefault(left, indegree.get(left, 0))
    starts = [node for node in nodes if indegree.get(node, 0) == 0]
    if len(nodes) != 3 or len(starts) != 1:
        raise ValueError("clues do not define one three-shape chain")
    chain = [starts[0]]
    while chain[-1] in successors:
        chain.append(successors[chain[-1]])
        if len(chain) > 3:
            raise ValueError("successor cycle")
    if len(chain) != 3 or len(set(chain)) != 3:
        raise ValueError("incomplete successor chain")
    return chain


def sample_scene(seed: int) -> dict:
    value = int(seed)
    rng = random.Random(value ^ 0x6D7A31C5)
    chain = list(SHAPES)
    rng.shuffle(chain)
    clues = [[chain[0], chain[1]], [chain[1], chain[2]]]
    if rng.randrange(2):
        clues.reverse()

    rotations = [chain, chain[1:] + chain[:1], chain[2:] + chain[:2]]
    answer_index = value % 3
    correct_position = (value // 3) % 3
    distractors = rotations[1:]
    if (value // 9) % 2:
        distractors.reverse()
    candidates = list(distractors)
    candidates.insert(correct_position, rotations[0])

    answer_tag = TAG_NAMES[answer_index]
    other_tags = [tag for tag in TAG_NAMES if tag != answer_tag]
    if (value // 27) % 2:
        other_tags.reverse()
    tag_colors = [""] * 3
    tag_colors[correct_position] = answer_tag
    cursor = 0
    for position in range(3):
        if position != correct_position:
            tag_colors[position] = other_tags[cursor]
            cursor += 1
    scene = {"clues": clues, "candidates": candidates, "tag_colors": tag_colors}
    if is_quarantined(scene):
        raise RuntimeError("sampler produced a quarantined scene")
    return scene


def analytic_gold(scene: dict) -> str:
    clues, candidates, tags = _scene_parts(scene)
    chain = _join(clues)
    matches = [index for index, candidate in enumerate(candidates) if candidate == chain]
    if len(matches) != 1:
        raise ValueError("scene has no unique matching candidate")
    return tags[matches[0]]


def margin(scene: dict) -> float:
    clues, candidates, _ = _scene_parts(scene)
    links = {(left, right) for left, right in clues}
    scores = [sum((candidate[i], candidate[i + 1]) in links for i in range(2)) for candidate in candidates]
    scores.sort(reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        clues, candidates, tags = _scene_parts(scene)
        chain = _join(clues)
        if set(chain) != set(SHAPES) or set(tags) != set(TAG_NAMES):
            return True
        if len({tuple(candidate) for candidate in candidates}) != 3:
            return True
        if any(set(candidate) != set(SHAPES) for candidate in candidates):
            return True
        matches = sum(candidate == chain for candidate in candidates)
        links = {(left, right) for left, right in clues}
        distractor_scores = [
            sum((candidate[i], candidate[i + 1]) in links for i in range(2))
            for candidate in candidates
            if candidate != chain
        ]
        return matches != 1 or sorted(distractor_scores) != [1, 1] or margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _draw_shape(draw: ImageDraw.ImageDraw, center: tuple[int, int], name: str) -> None:
    cx, cy = center
    r = SHAPE_RADIUS
    if name == "circle":
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=INK)
    elif name == "square":
        draw.rectangle((cx - r, cy - r, cx + r, cy + r), fill=INK)
    elif name == "triangle":
        draw.polygon(((cx, cy - r), (cx - r, cy + r), (cx + r, cy + r)), fill=INK)
    else:
        raise ValueError(f"unknown shape {name!r}")


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=color)
    draw.rectangle((x0 + FRAME, y0 + FRAME, x1 - FRAME, y1 - FRAME), fill=PANEL)


def render(scene: dict) -> Image.Image:
    clues, candidates, tags = _scene_parts(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((12, 12, WIDTH - 13, HEIGHT - 13), fill=PLAQUE)

    for box, pair in zip(CLUE_BOXES, clues, strict=True):
        _panel(draw, box, CLUE)
        x0, y0, x1, y1 = box
        cy = (y0 + y1) // 2
        _draw_shape(draw, (x0 + 52, cy), pair[0])
        _draw_shape(draw, (x1 - 52, cy), pair[1])
        draw.line((x0 + 91, cy, x1 - 91, cy), fill=ARROW, width=7)
        draw.polygon(((x1 - 91, cy - 11), (x1 - 72, cy), (x1 - 91, cy + 11)), fill=ARROW)

    for box, candidate, tag in zip(CANDIDATE_BOXES, candidates, tags, strict=True):
        _panel(draw, box, TAG_RGB[tag])
        x0, y0, x1, y1 = box
        cy = (y0 + y1) // 2
        centers = (x0 + 48, (x0 + x1) // 2, x1 - 48)
        for cx, shape in zip(centers, candidate, strict=True):
            _draw_shape(draw, (cx, cy), shape)
    return image
