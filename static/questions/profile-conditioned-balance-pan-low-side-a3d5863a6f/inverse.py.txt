"""Pixel-only inverse arm for `balance_pan_low_side`.

Receives one rasterised PNG and nothing else: no latent scene, no renderer,
prompt, generator, verifier or gold code. It segments the pan ink (dark and
nearly colourless, so the light grey stand, beam, cords and page decoration and
the saturated coloured blocks are all excluded), keeps the two large connected
pan bodies, measures each body's centroid, and reports the pan whose centroid
sits nearer the bottom of the raster -- named `left` or `right` by the order of
the two centroids across the page.

It abstains whenever the raster does not show exactly two intact pan bars of
comparable size, or when the two centroids sit too close in height for the
question to have a readable answer.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "balance_pan_low_side-oracle-0.1.0"
ABSTAIN = "abstain"

DARK_MAX = 95  # every channel this dark or darker
CHROMA_MAX = 42  # and near-colourless: pan ink, not block ink
MIN_PAN_AREA = 420
MIN_PAN_WIDTH = 48.0
PAN_HEIGHT_RANGE = (5.0, 34.0)
AREA_RATIO_TOL = 0.45  # the two pans are drawn identically
MIN_X_SEPARATION = 120.0
READ_BAND_PX = 8.0  # measured height difference below which we abstain


def _components(mask: np.ndarray) -> list:
    """4-connected components as (area, sum_r, sum_c, r0, r1, c0, c1)."""
    h, w = mask.shape
    flat = mask.reshape(-1).copy()
    out = []
    for start in np.flatnonzero(flat):
        if not flat[start]:
            continue
        stack = [int(start)]
        flat[start] = False
        area = 0
        sum_r = 0
        sum_c = 0
        r0 = r1 = int(start) // w
        c0 = c1 = int(start) % w
        while stack:
            p = stack.pop()
            r, c = divmod(p, w)
            area += 1
            sum_r += r
            sum_c += c
            if r < r0:
                r0 = r
            if r > r1:
                r1 = r
            if c < c0:
                c0 = c
            if c > c1:
                c1 = c
            if c > 0 and flat[p - 1]:
                flat[p - 1] = False
                stack.append(p - 1)
            if c < w - 1 and flat[p + 1]:
                flat[p + 1] = False
                stack.append(p + 1)
            if r > 0 and flat[p - w]:
                flat[p - w] = False
                stack.append(p - w)
            if r < h - 1 and flat[p + w]:
                flat[p + w] = False
                stack.append(p + w)
        out.append((area, sum_r, sum_c, r0, r1, c0, c1))
    return out


def decision_from_image(image: "Image.Image") -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=-1) - arr.min(axis=-1)
    ink = (arr.max(axis=-1) <= DARK_MAX) & (chroma <= CHROMA_MAX)
    if int(ink.sum()) < 2 * MIN_PAN_AREA:
        return ABSTAIN

    bodies = [c for c in _components(ink) if c[0] >= MIN_PAN_AREA]
    if len(bodies) != 2:
        return ABSTAIN

    measured = []
    for area, sum_r, sum_c, r0, r1, c0, c1 in bodies:
        width = float(c1 - c0 + 1)
        height = float(r1 - r0 + 1)
        if width < MIN_PAN_WIDTH or not (PAN_HEIGHT_RANGE[0] <= height <= PAN_HEIGHT_RANGE[1]):
            return ABSTAIN  # not an intact pan bar
        measured.append((sum_c / area, sum_r / area, float(area)))

    a0, a1 = measured[0][2], measured[1][2]
    if abs(a0 - a1) > AREA_RATIO_TOL * max(a0, a1):
        return ABSTAIN  # the two bodies are not a matched pair of pans

    measured.sort(key=lambda t: t[0])
    if measured[1][0] - measured[0][0] < MIN_X_SEPARATION:
        return ABSTAIN

    left_y, right_y = measured[0][1], measured[1][1]
    if abs(right_y - left_y) < READ_BAND_PX:
        return ABSTAIN
    return "right" if right_y > left_y else "left"
