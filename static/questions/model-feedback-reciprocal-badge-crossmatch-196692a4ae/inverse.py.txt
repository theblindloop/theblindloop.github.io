"""Independent image-only inverse for Reciprocal Badge Crossmatch."""

from __future__ import annotations

import itertools
from PIL import Image, ImageDraw

CANVAS = (1000, 800)
INK = (35, 40, 48)
MUTED = (218, 223, 228)
SHAPE = (25, 126, 111)
BADGE = (61, 49, 94)
PANEL = (255, 255, 255)
COLORS = {"amber": (219, 139, 22), "cyan": (18, 153, 177), "violet": (133, 76, 181)}
SHAPES = ("circle", "square", "triangle", "diamond")
ROW_Y = (210, 325, 440, 555)
ANSWER_X = (175, 500, 825)


class OracleError(ValueError):
    pass


def _shape_template(name: str, scale: float) -> tuple[int, ...]:
    image = Image.new("RGB", (80, 80), PANEL)
    draw, cx, cy = ImageDraw.Draw(image), 40, 40
    if name == "circle":
        r=round(25*scale); draw.ellipse((cx-r, cy-r, cx+r, cy+r), fill=SHAPE)
    elif name == "square":
        r=round(23*scale); draw.rectangle((cx-r, cy-r, cx+r, cy+r), fill=SHAPE)
    elif name == "triangle":
        a,b=round(29*scale),round(23*scale); draw.polygon(((cx,cy-a),(cx-a,cy+b),(cx+a,cy+b)), fill=SHAPE)
    elif name == "diamond":
        r=round(30*scale); draw.polygon(((cx,cy-r),(cx-r,cy),(cx,cy+r),(cx+r,cy)), fill=SHAPE)
    return tuple(pixel == SHAPE for pixel in image.getdata())


def _read_shape(image: Image.Image, cx: int, cy: int, scale: float = 1.0) -> str:
    crop = image.crop((cx-40, cy-40, cx+40, cy+40))
    observed = tuple(pixel == SHAPE for pixel in crop.getdata())
    scores = {name: sum(a != b for a,b in zip(observed, _shape_template(name, scale))) for name in SHAPES}
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 900:
        raise OracleError("damaged or unknown key shape")
    return best


def _read_badge(image: Image.Image, cx: int, cy: int) -> tuple[int, ...]:
    px = image.load()
    if px[cx-60, cy] != INK or px[cx+60, cy] != INK:
        raise OracleError("badge border missing")
    bits = []
    for i in range(6):
        col, row = i % 3, i // 3
        x0, y0 = cx-43+col*31, cy-29+row*36
        samples = [px[x,y] for x in range(x0+4,x0+21,4) for y in range(y0+4,y0+21,4)]
        if all(pixel == BADGE for pixel in samples): bits.append(1)
        elif all(pixel == MUTED for pixel in samples): bits.append(0)
        else: raise OracleError("badge cell damaged")
    if sum(bits) != 3:
        raise OracleError("invalid badge")
    return tuple(bits)


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != CANVAS:
        raise OracleError("wrong canvas")
    px = image.load()
    for x,y in ((500,95),(90,350),(910,350)):
        if px[x,y] != INK: raise OracleError("table boundary missing")
    records = []
    for y in ROW_Y:
        if px[320,y] != INK: raise OracleError("row arrow missing")
        records.append((_read_shape(image,220,y), _read_badge(image,495,y), _read_badge(image,745,y)))
    if len({record[0] for record in records}) != 4:
        raise OracleError("duplicate key")
    pairs = []
    for first, second in itertools.combinations(records,2):
        if first[1] == second[2] and first[2] == second[1]:
            pairs.append(frozenset((first[0],second[0])))
    if len(pairs) != 1:
        raise OracleError("reciprocal pair is not unique")
    target = pairs[0]
    answers = []
    seen = set()
    for cx in ANSWER_X:
        border = px[cx,655]
        names = [name for name,value in COLORS.items() if border == value]
        if len(names) != 1: raise OracleError("answer border damaged")
        pair = frozenset((_read_shape(image,cx-52,724,0.82), _read_shape(image,cx+52,724,0.82)))
        if len(pair) != 2 or pair in seen: raise OracleError("malformed answer pair")
        seen.add(pair)
        if pair == target: answers.append(names[0])
    if len(answers) != 1:
        raise OracleError("answer is not unique")
    return answers[0]
