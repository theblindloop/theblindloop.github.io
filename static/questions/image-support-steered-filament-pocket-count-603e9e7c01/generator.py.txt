"""Deterministic renderer for compact filament-complement topology."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw


SCALE = 4
PALETTES = (
    ((248, 247, 242), (255, 253, 246), (25, 91, 156)),
    ((247, 246, 250), (254, 252, 255), (133, 45, 112)),
    ((247, 249, 246), (253, 255, 249), (24, 116, 92)),
    ((249, 247, 244), (255, 253, 248), (174, 70, 34)),
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    background, card, ink = PALETTES[rng.randrange(len(PALETTES))]
    width = 512
    height = 512
    center_x = rng.randint(112, width - 112)
    # Keep the thin trace inside one fine-grid row while randomizing which row.
    center_y = rng.choice([96, 160, 224, 288, 352, 416])
    ornaments = []
    while len(ornaments) < 7:
        ox, oy = rng.randint(24, width - 24), rng.randint(24, height - 24)
        if abs(ox - center_x) > 92 or abs(oy - center_y) > 66:
            ornaments.append([ox, oy, rng.randint(2, 4)])
    return {
        "width": width,
        "height": height,
        "center_x": center_x,
        "center_y": center_y,
        "pocket_count": seed % 3,
        "loop_rx": rng.randint(21, 25),
        "loop_ry": rng.randint(9, 11),
        "rotation": rng.uniform(-6.0, 6.0),
        "stroke": rng.choice([4, 5]),
        "background": list(background),
        "card": list(card),
        "ink": list(ink),
        "ornaments": ornaments,
        "wiggle": [rng.uniform(-2.5, 2.5) for _ in range(4)],
    }


def _transform(points: list[tuple[float, float]], cx: float, cy: float, angle: float) -> list[tuple[int, int]]:
    radians = math.radians(angle)
    cosine, sine = math.cos(radians), math.sin(radians)
    return [
        (
            round(SCALE * (cx + x * cosine - y * sine)),
            round(SCALE * (cy + x * sine + y * cosine)),
        )
        for x, y in points
    ]


def _ellipse(cx: float, cy: float, rx: float, ry: float, start: float, steps: int = 120) -> list[tuple[float, float]]:
    return [
        (
            cx + rx * math.cos(start + 2.0 * math.pi * index / steps),
            cy + ry * math.sin(start + 2.0 * math.pi * index / steps),
        )
        for index in range(steps + 1)
    ]


def _filament(count: int, rx: float, ry: float, wiggle: list[float]) -> list[tuple[float, float]]:
    if count == 0:
        # An x-monotone meander cannot bound a pocket, despite having comparable ink.
        points = []
        for index in range(121):
            t = index / 120.0
            x = -58.0 + 116.0 * t
            y = (ry * 0.72) * math.sin(3.0 * math.pi * t) + wiggle[index % 4] * math.sin(math.pi * t)
            points.append((x, y))
        return points
    if count == 1:
        # One continuous stroke: tail to a junction, once around a deformed loop.
        tail = [(-58.0 + (58.0 - rx) * index / 35.0, wiggle[0] * math.sin(math.pi * index / 35.0)) for index in range(36)]
        loop = _ellipse(0.0, 0.0, rx, ry, math.pi)
        return tail + loop[1:]
    # One continuous stroke visits two loops sharing a central junction.
    left = _ellipse(-rx, 0.0, rx, ry, 0.0)
    right = _ellipse(rx, 0.0, rx, ry, math.pi)
    return left + right[1:]


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    center_x = float(scene["center_x"])
    center_y = float(scene["center_y"])
    pocket_count = int(scene["pocket_count"])
    loop_rx = float(scene["loop_rx"])
    loop_ry = float(scene["loop_ry"])
    rotation = float(scene["rotation"])
    stroke = int(scene["stroke"])
    background = tuple(int(value) for value in scene["background"])
    card = tuple(int(value) for value in scene["card"])
    ink = tuple(int(value) for value in scene["ink"])
    ornaments = scene["ornaments"]
    wiggle = [float(value) for value in scene["wiggle"]]

    image = Image.new("RGB", (width * SCALE, height * SCALE), background)
    draw = ImageDraw.Draw(image)
    # The pale card gives a local inspection neighborhood but is ignored by the inverse arm.
    card_box = [
        round(SCALE * (center_x - 78)), round(SCALE * (center_y - 52)),
        round(SCALE * (center_x + 78)), round(SCALE * (center_y + 52)),
    ]
    draw.rounded_rectangle(card_box, radius=12 * SCALE, fill=card, outline=(224, 224, 220), width=SCALE)
    for ox, oy, radius in ornaments:
        r = int(radius) * SCALE
        gray = (224, 224, 222)
        draw.ellipse([int(ox) * SCALE - r, int(oy) * SCALE - r, int(ox) * SCALE + r, int(oy) * SCALE + r], fill=gray)
    points = _transform(_filament(pocket_count, loop_rx, loop_ry, wiggle), center_x, center_y, rotation)
    draw.line(points, fill=ink, width=stroke * SCALE, joint="curve")
    endpoint_radius = max(2, stroke // 2) * SCALE
    for px, py in (points[0], points[-1]):
        draw.ellipse([px - endpoint_radius, py - endpoint_radius, px + endpoint_radius, py + endpoint_radius], fill=ink)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return ("zero", "one", "two")[int(scene["pocket_count"])]


def margin(scene: dict) -> float:
    count = int(scene["pocket_count"])
    clearance = 2.0 * float(scene["loop_ry"]) - float(scene["stroke"])
    return clearance if count else float(scene["loop_ry"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
