"""Independent image-only inverse arm for the proximity graph."""

from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[list[tuple[int, int]]] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        group: list[tuple[int, int]] = []
        while queue:
            yy, xx = queue.popleft()
            group.append((yy, xx))
            for ny, nx in ((yy - 1, xx), (yy + 1, xx), (yy, xx - 1), (yy, xx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        groups.append(group)
    return groups


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = np.max(array, axis=2) < 70
    gray = (
        (array[:, :, 0] >= 120)
        & (array[:, :, 0] <= 180)
        & (np.max(array, axis=2) - np.min(array, axis=2) <= 28)
    )

    dot_groups = [group for group in _components(dark) if 120 <= len(group) <= 260]
    if len(dot_groups) != 12:
        raise ValueError(f"expected 12 black dot components, found {len(dot_groups)}")
    points = []
    for group in dot_groups:
        ys = [p[0] for p in group]
        xs = [p[1] for p in group]
        points.append((sum(xs) / len(xs), sum(ys) / len(ys)))

    gray_groups = [group for group in _components(gray) if len(group) >= 300]
    if len(gray_groups) != 1:
        raise ValueError(f"expected one gray scale bar, found {len(gray_groups)}")
    bar = gray_groups[0]
    threshold = max(x for _, x in bar) - min(x for _, x in bar) + 1

    graph = [set() for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if math.dist(points[i], points[j]) < threshold:
                graph[i].add(j)
                graph[j].add(i)
    if any(len(neighbors) != 2 for neighbors in graph):
        raise ValueError("pixel graph is not entirely degree two")

    unseen = set(range(len(points)))
    cycles = 0
    while unseen:
        cycles += 1
        queue = deque([unseen.pop()])
        while queue:
            for neighbor in graph[queue.popleft()]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
    if cycles == 1:
        return "one"
    if cycles == 2:
        return "two"
    raise ValueError(f"unsupported pixel cycle count {cycles}")
