"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw


def _point(cx: float, cy: float, radius: float, angle: float) -> list[float]:
    return [round(cx + radius * math.cos(angle), 4), round(cy + radius * math.sin(angle), 4)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    threshold = rng.randint(66, 70)
    side = rng.uniform(44.0, 49.0)
    rotation = rng.uniform(0.0, 2.0 * math.pi)
    cx = 256.0 + rng.uniform(-18.0, 18.0)
    cy = 218.0 + rng.uniform(-14.0, 14.0)
    target_cycles = 1 if int(seed) % 2 == 0 else 2

    if target_cycles == 1:
        radius = side / (2.0 * math.sin(math.pi / 12.0))
        points = [_point(cx, cy, radius, rotation + i * math.pi / 6.0) for i in range(12)]
    else:
        separation = rng.uniform(181.0, 192.0)
        axis = rng.uniform(0.0, 2.0 * math.pi)
        dx = 0.5 * separation * math.cos(axis)
        dy = 0.5 * separation * math.sin(axis)
        centers = ((cx - dx, cy - dy), (cx + dx, cy + dy))
        points = []
        for ring_index, (rx, ry) in enumerate(centers):
            phase = rotation + ring_index * rng.uniform(0.18, 0.48)
            points.extend(_point(rx, ry, side, phase + i * math.pi / 3.0) for i in range(6))

    rng.shuffle(points)
    bar_x = rng.randint(190, 255)
    bar_y = rng.randint(459, 471)
    return {
        "points": points,
        "dot_radius": 7,
        "threshold": threshold,
        "bar_box": [bar_x, bar_y, 9],
        "palette": {"background": [250, 250, 247], "dots": [23, 25, 29], "bar": [145, 153, 162]},
        "canvas": [512, 512],
    }


def _graph(scene: dict) -> list[set[int]]:
    points = scene["points"]
    threshold = float(scene["threshold"])
    graph = [set() for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if math.dist(points[i], points[j]) < threshold:
                graph[i].add(j)
                graph[j].add(i)
    return graph


def _component_count(graph: list[set[int]]) -> int:
    unseen = set(range(len(graph)))
    count = 0
    while unseen:
        count += 1
        queue = deque([unseen.pop()])
        while queue:
            for neighbor in graph[queue.popleft()]:
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    queue.append(neighbor)
    return count


def analytic_gold(scene: dict) -> str:
    # Direct graph derivation is intentional: no sampled class tag exists.
    points = scene["points"]
    threshold = scene["threshold"]
    graph = [set() for _ in points]
    for i in range(len(points)):
        for j in range(i + 1, len(points)):
            if math.dist(points[i], points[j]) < threshold:
                graph[i].add(j)
                graph[j].add(i)
    if any(len(neighbors) != 2 for neighbors in graph):
        raise ValueError("constellation is not a union of cycles")
    cycles = _component_count(graph)
    if cycles not in (1, 2):
        raise ValueError(f"unsupported cycle count {cycles}")
    return "one" if cycles == 1 else "two"


def margin(scene: dict) -> float:
    points = scene["points"]
    threshold = float(scene["threshold"])
    return min(
        abs(math.dist(points[i], points[j]) - threshold)
        for i in range(len(points))
        for j in range(i + 1, len(points))
    )


def is_quarantined(scene: dict) -> bool:
    try:
        graph = _graph(scene)
        cycles = _component_count(graph)
        return margin(scene) < 6.0 or any(len(n) != 2 for n in graph) or cycles not in (1, 2)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    canvas = scene["canvas"]
    palette = scene["palette"]
    image = Image.new("RGB", (int(canvas[0]), int(canvas[1])), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    radius = int(scene["dot_radius"])
    for x, y in scene["points"]:
        draw.ellipse(
            [round(x) - radius, round(y) - radius, round(x) + radius, round(y) + radius],
            fill=tuple(palette["dots"]),
        )
    bar_x, bar_y, bar_h = [int(v) for v in scene["bar_box"]]
    threshold = int(scene["threshold"])
    draw.rectangle(
        [bar_x, bar_y, bar_x + threshold - 1, bar_y + bar_h - 1],
        fill=tuple(palette["bar"]),
    )
    return image
