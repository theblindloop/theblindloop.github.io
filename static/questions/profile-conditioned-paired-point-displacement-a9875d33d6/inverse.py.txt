"""Independent inverse arm using only raster pixels."""

from __future__ import annotations

import math
import numpy as np
from PIL import Image

PALETTE = {
    "red": (216, 58, 50),
    "blue": (40, 120, 200),
    "green": (39, 150, 87),
    "orange": (228, 134, 36),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    parts = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = cy + dy, cx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(pixels) >= 120:
            parts.append(np.asarray(pixels, dtype=float))
    return parts


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int32)
    distances = {}
    for name, rgb in PALETTE.items():
        delta = arr - np.asarray(rgb, dtype=np.int32)
        mask = np.sqrt(np.sum(delta * delta, axis=2)) < 48
        parts = _components(mask)
        if len(parts) != 2:
            raise ValueError(f"expected two separated {name} measurements, found {len(parts)}")
        centers = [(float(part[:, 1].mean()), float(part[:, 0].mean())) for part in parts]
        distances[name] = math.hypot(centers[0][0] - centers[1][0], centers[0][1] - centers[1][1])
    ordered = sorted(distances, key=distances.get, reverse=True)
    if distances[ordered[0]] - distances[ordered[1]] < 4.0:
        raise ValueError("pixel distance comparison is ambiguous")
    return ordered[0]
