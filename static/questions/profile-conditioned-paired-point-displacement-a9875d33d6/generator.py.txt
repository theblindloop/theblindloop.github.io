"""Renderer for a paired-measurement displacement plot."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

PALETTE = {
    "red": "#d83a32",
    "blue": "#2878c8",
    "green": "#279657",
    "orange": "#e48624",
}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    names = list(PALETTE)
    winner = names[int(seed) % len(names)]
    remaining = [name for name in names if name != winner]
    rng.shuffle(remaining)

    if int(seed) % 11 == 0:
        lengths = {winner: rng.uniform(116.0, 120.0), remaining[0]: rng.uniform(108.5, 111.0),
                   remaining[1]: rng.uniform(67.0, 82.0), remaining[2]: rng.uniform(39.0, 55.0)}
    else:
        lengths = {winner: rng.uniform(128.0, 145.0), remaining[0]: rng.uniform(91.0, 106.0),
                   remaining[1]: rng.uniform(64.0, 82.0), remaining[2]: rng.uniform(38.0, 55.0)}

    colors = names[:]
    rng.shuffle(colors)
    slots = [(170, 125), (470, 125), (170, 325), (470, 325)]
    centers = []
    offsets = []
    for color, (sx, sy) in zip(colors, slots):
        angle = rng.uniform(0.12, math.pi - 0.12)
        if rng.random() < 0.5:
            angle = -angle
        length = lengths[color]
        dx = length * math.cos(angle)
        dy = length * math.sin(angle)
        centers.append([sx + rng.randint(-12, 12), sy + rng.randint(-8, 8)])
        offsets.append([dx, dy])
    return {"colors": colors, "centers": centers, "offsets": offsets, "width": 640, "height": 440}


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), "white")
    draw = ImageDraw.Draw(image)
    for x in range(70, width - 30, 50):
        draw.line((x * scale, 30 * scale, x * scale, (height - 35) * scale), fill="#edf0f2", width=scale)
    for y in range(35, height - 30, 50):
        draw.line((45 * scale, y * scale, (width - 30) * scale, y * scale), fill="#edf0f2", width=scale)
    draw.line((45 * scale, (height - 35) * scale, (width - 30) * scale, (height - 35) * scale), fill="#7d858b", width=2 * scale)
    draw.line((45 * scale, 30 * scale, 45 * scale, (height - 35) * scale), fill="#7d858b", width=2 * scale)

    radius = 12
    for color, center, offset in zip(scene["colors"], scene["centers"], scene["offsets"]):
        cx, cy = center
        dx, dy = offset
        endpoints = ((cx - dx / 2.0, cy - dy / 2.0), (cx + dx / 2.0, cy + dy / 2.0))
        for px, py in endpoints:
            draw.ellipse(((px - radius) * scale, (py - radius) * scale,
                          (px + radius) * scale, (py + radius) * scale), fill=PALETTE[color])
            draw.ellipse(((px - 4) * scale, (py - 4) * scale,
                          (px + 4) * scale, (py + 4) * scale), fill="white")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _lengths(scene: dict) -> list[float]:
    return [math.hypot(float(dx), float(dy)) for dx, dy in scene["offsets"]]


def analytic_gold(scene: dict) -> str:
    values = _lengths(scene)
    return scene["colors"][max(range(len(values)), key=values.__getitem__)]


def margin(scene: dict) -> float:
    values = sorted(_lengths(scene), reverse=True)
    return values[0] - values[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
