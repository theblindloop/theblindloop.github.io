"""Independent RGB-only inverse for reciprocal-route races."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

PALETTE = (
    (201, 67, 64),
    (48, 111, 176),
    (43, 147, 103),
    (222, 153, 35),
    (137, 88, 174),
    (27, 153, 164),
)
INK = (35, 42, 54)
CLUE_RECTS = (
    (20, 124, 295, 244),
    (315, 124, 590, 244),
    (610, 124, 885, 244),
    (905, 124, 1180, 244),
    (167, 264, 442, 384),
    (462, 264, 737, 384),
    (757, 264, 1032, 384),
)
RACE_RECTS = (
    (30, 515, 585, 710),
    (615, 515, 1170, 710),
    (30, 735, 585, 930),
    (615, 735, 1170, 930),
)
ASSIGNMENTS = np.asarray(list(itertools.permutations(range(1, 10), 6)), dtype=np.int16)


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    height, width = mask.shape
    remaining = set(int(value) for value in np.flatnonzero(mask))
    found: list[tuple[int, float, float]] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            xs.append(x)
            ys.append(y)
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        found.append((len(xs), sum(xs) / len(xs), sum(ys) / len(ys)))
    return found


def _colored_components(
    arr: np.ndarray, rect: tuple[int, int, int, int], expected_y: float
) -> list[tuple[int, float]] | None:
    x0, y0, x1, y1 = rect
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    found: list[tuple[int, float]] = []
    for color_index, color in enumerate(PALETTE):
        mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
        for size, x, y in _components(mask):
            if not 850 <= size <= 1300 or abs(y - expected_y) > 1.0:
                return None
            found.append((color_index, x))
    return found


def _read_clue(
    arr: np.ndarray, rect: tuple[int, int, int, int]
) -> np.ndarray | None:
    components = _colored_components(arr, rect, 82.0)
    if components is None or len(components) != 4:
        return None
    x0, _y0, x1, _y1 = rect
    center = (x1 - x0) / 2
    vector = np.zeros(6, dtype=np.int16)
    sides = {-1: 0, 1: 0}
    for color_index, x in components:
        if abs(x - center) < 25:
            return None
        side = -1 if x < center else 1
        vector[color_index] += -side
        sides[side] += 1
    if sides != {-1: 2, 1: 2} or np.count_nonzero(vector) != 4:
        return None
    return vector


def _read_race(
    arr: np.ndarray, rect: tuple[int, int, int, int]
) -> tuple[tuple[int, int], tuple[int, int]] | None:
    components = _colored_components(arr, rect, 126.0)
    if components is None or len(components) != 4:
        return None
    components.sort(key=lambda item: item[1])
    colors = tuple(item[0] for item in components)
    if len(set(colors)) != 4:
        return None
    return (colors[0:2], colors[2:4])


def _structure_ok(arr: np.ndarray) -> bool:
    if arr.shape != (960, 1200, 3):
        return False
    ink = np.asarray(INK, dtype=np.uint8)
    border_points = [
        (x0, (y0 + y1) // 2) for x0, y0, _x1, y1 in CLUE_RECTS + RACE_RECTS
    ]
    return all(np.array_equal(arr[y, x], ink) for x, y in border_points)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if not _structure_ok(arr):
        return "abstain"
    clue_rows: list[np.ndarray] = []
    for rect in CLUE_RECTS:
        vector = _read_clue(arr, rect)
        if vector is None:
            return "abstain"
        clue_rows.append(vector)
    clue_matrix = np.stack(clue_rows)
    feasible = ASSIGNMENTS[np.all(ASSIGNMENTS @ clue_matrix.T > 0, axis=1)]
    if not 2 <= len(feasible) <= 5000:
        return "abstain"
    matches: list[int] = []
    for index, rect in enumerate(RACE_RECTS, 1):
        race = _read_race(arr, rect)
        if race is None:
            return "abstain"
        left, right = race
        a = feasible[:, left[0]].astype(np.int32)
        b = feasible[:, left[1]].astype(np.int32)
        c = feasible[:, right[0]].astype(np.int32)
        d = feasible[:, right[1]].astype(np.int32)
        gaps = a * b * (c + d) - c * d * (a + b)
        if np.all(gaps > 0):
            matches.append(index)
        elif not (np.any(gaps > 0) and np.any(gaps < 0)) or np.any(gaps == 0):
            return "abstain"
    return f"race-{matches[0]}" if len(matches) == 1 else "abstain"
