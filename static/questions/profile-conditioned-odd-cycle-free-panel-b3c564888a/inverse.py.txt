"""Independent PNG-only inverse arm: recover three graphs and test bipartiteness."""

from __future__ import annotations

from collections import deque
from math import hypot

import numpy as np
from PIL import Image

JUNCTION = np.array((22, 27, 34), dtype=float)
LINK = np.array((93, 103, 118), dtype=float)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = [(int(y), int(x))]
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.pop()
            pixels.append((cy, cx))
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(pixels) >= 220:
            result.append(pixels)
    return result


def _line_ink_fraction(array: np.ndarray, first, second) -> float:
    ax, ay = first
    bx, by = second
    distance = hypot(bx - ax, by - ay)
    count = max(12, int(distance - 30))
    hits = 0
    for index in range(count):
        along = (15.0 + (distance - 30.0) * (index + 0.5) / count) / distance
        x = int(round(ax + along * (bx - ax)))
        y = int(round(ay + along * (by - ay)))
        patch = array[max(0, y - 1): y + 2, max(0, x - 1): x + 2]
        if patch.size and np.min(np.linalg.norm(patch - LINK, axis=2)) < 48.0:
            hits += 1
    return hits / count


def _recover_graph(array: np.ndarray, nodes) -> dict[int, set[int]]:
    graph = {index: set() for index in range(len(nodes))}
    for i, first in enumerate(nodes):
        for j in range(i + 1, len(nodes)):
            if _line_ink_fraction(array, first, nodes[j]) >= 0.86:
                graph[i].add(j)
                graph[j].add(i)
    degrees = sorted(len(values) for values in graph.values())
    if len(nodes) != 8 or sum(degrees) != 24 or degrees != [3] * 8:
        raise ValueError(f"panel graph failed cubic integrity check: degrees={degrees}")
    return graph


def _is_bipartite(graph: dict[int, set[int]]) -> bool:
    colors: dict[int, int] = {}
    for start in graph:
        if start in colors:
            continue
        colors[start] = 0
        queue = deque([start])
        while queue:
            node = queue.popleft()
            for nxt in graph[node]:
                if nxt not in colors:
                    colors[nxt] = 1 - colors[node]
                    queue.append(nxt)
                elif colors[nxt] == colors[node]:
                    return False
    return True


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=float)
    if array.shape[1] < 600 or array.shape[0] < 240:
        raise ValueError("raster is too small")
    junction_mask = np.linalg.norm(array - JUNCTION, axis=2) < 42.0
    blobs = _components(junction_mask)
    nodes = []
    for pixels in blobs:
        ys = np.array([pixel[0] for pixel in pixels], dtype=float)
        xs = np.array([pixel[1] for pixel in pixels], dtype=float)
        nodes.append((float(xs.mean()), float(ys.mean())))
    if len(nodes) != 24:
        raise ValueError(f"expected 24 round junctions, recovered {len(nodes)}")
    panel_nodes = [[], [], []]
    width = array.shape[1]
    for point in nodes:
        panel_nodes[min(2, int(3 * point[0] / width))].append(point)
    if [len(group) for group in panel_nodes] != [8, 8, 8]:
        raise ValueError("junctions do not divide into three eight-node panels")
    matches = []
    for index, group in enumerate(panel_nodes):
        graph = _recover_graph(array, group)
        if _is_bipartite(graph):
            matches.append(index)
    if len(matches) != 1:
        raise ValueError(f"expected one odd-cycle-free raster graph, found {matches}")
    return ("left", "center", "right")[matches[0]]
