"""Deterministic renderer and analytic arm for odd-cycle exclusion."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 360
BACKGROUND = (250, 250, 247)
LINK = (93, 103, 118)
JUNCTION = (22, 27, 34)
FRAME = (216, 220, 224)
PANEL_CENTERS = ((150, 180), (450, 180), (750, 180))
RADIUS = 108

_CYCLE = tuple((i, (i + 1) % 8) for i in range(8))
_BIPARTITE_MATCHINGS = (
    ((0, 3), (1, 6), (2, 5), (4, 7)),
    ((0, 5), (1, 4), (2, 7), (3, 6)),
)
# These have the same 8 nodes, 12 edges, cubic degree sequence, and chord ink
# within four percent of the bipartite constructions on a regular octagon.
_ODD_MATCHINGS = (
    ((0, 2), (1, 5), (3, 6), (4, 7)),
    ((0, 3), (1, 4), (2, 6), (5, 7)),
    ((0, 3), (1, 5), (2, 7), (4, 6)),
    ((0, 4), (1, 6), (2, 7), (3, 5)),
    ((0, 4), (1, 7), (2, 5), (3, 6)),
    ((0, 5), (1, 3), (2, 6), (4, 7)),
    ((0, 5), (1, 6), (2, 4), (3, 7)),
    ((0, 6), (1, 4), (2, 5), (3, 7)),
)


def _relabel(edges, shift: int, reflected: bool) -> list[list[int]]:
    def label(value: int) -> int:
        value = (-value) % 8 if reflected else value
        return (value + shift) % 8
    return [[label(a), label(b)] for a, b in edges]


def _nodes(center: tuple[int, int], angle_offset: int, jitter) -> list[list[int]]:
    # Integer unit directions make raster recovery stable while retaining an
    # octagonal embedding. angle_offset is in eighth turns.
    directions = ((0, -108), (76, -76), (108, 0), (76, 76),
                  (0, 108), (-76, 76), (-108, 0), (-76, -76))
    cx, cy = center
    result = []
    for i in range(8):
        dx, dy = directions[(i + angle_offset) % 8]
        jx, jy = jitter[i]
        result.append([cx + dx + jx, cy + dy + jy])
    return result


def _is_bipartite(edges) -> bool:
    graph = {i: set() for i in range(8)}
    for a, b in edges:
        graph[int(a)].add(int(b))
        graph[int(b)].add(int(a))
    colors: dict[int, int] = {}
    for start in graph:
        if start in colors:
            continue
        colors[start] = 0
        stack = [start]
        while stack:
            node = stack.pop()
            for nxt in graph[node]:
                if nxt not in colors:
                    colors[nxt] = 1 - colors[node]
                    stack.append(nxt)
                elif colors[nxt] == colors[node]:
                    return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_position = int(seed) % 3
    bip = rng.choice(_BIPARTITE_MATCHINGS)
    odd = rng.sample(_ODD_MATCHINGS, 2)
    matchings = list(odd)
    matchings.insert(answer_position, bip)
    panels = []
    for position, matching in enumerate(matchings):
        shift = rng.randrange(8)
        reflected = bool(rng.randrange(2))
        angle_offset = rng.randrange(8)
        jitter = [[rng.randint(-3, 3), rng.randint(-3, 3)] for _ in range(8)]
        edges = _relabel((*_CYCLE, *matching), shift, reflected)
        panels.append({
            "nodes": _nodes(PANEL_CENTERS[position], angle_offset, jitter),
            "edges": edges,
        })
    return {"panels": panels}


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for position, panel in enumerate(panels):
        cx, cy = PANEL_CENTERS[position]
        draw.rounded_rectangle((cx - 139, cy - 157, cx + 139, cy + 157),
                               radius=14, outline=FRAME, width=3)
        nodes = panel["nodes"]
        for a, b in panel["edges"]:
            draw.line([tuple(nodes[int(a)]), tuple(nodes[int(b)])], fill=LINK, width=9)
        for x, y in nodes:
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=JUNCTION)
    return image


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    matches = [index for index, panel in enumerate(panels) if _is_bipartite(panel["edges"])]
    if len(matches) != 1:
        raise ValueError(f"expected one odd-cycle-free panel, found {matches}")
    return ("left", "center", "right")[matches[0]]


def margin(scene: dict) -> float:
    panels = scene["panels"]
    if sum(_is_bipartite(panel["edges"]) for panel in panels) != 1:
        return 0.0
    # One edge insertion/deletion is the nearest discrete parity boundary.
    return 1.0


def is_quarantined(scene: dict) -> bool:
    panels = scene["panels"]
    return any(len(panel["nodes"]) != 8 or len(panel["edges"]) != 12 for panel in panels)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    for panel in twin["panels"]:
        panel["edges"] = list(reversed(panel["edges"]))
    return [("reverse_edge_storage_order", twin)]
