"""Scene-free pixel oracle for the barred-holder binding world.

Receives only the rendered PNG.  It segments the vivid flat pip fills from the
pale card and page greys, groups the pips into connected discs, splits them by
page side into two-pip badge rows and four-pip holder rows, reads each row's
colour set, binds the pure black tag to its badge row, and then reports the
one holder that the tagged badge fits yet no complete injective badge-to-holder
assignment ever hands it.

It imports no renderer, prompt, generator, verifier or gold code and never
sees latent scene data.
"""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

_SPREAD = 40      # a pip fill is far more saturated than any grey on the page
_SIDE_SPLIT = 550  # badges are drawn left of this column, holders right of it
_ROW_TOL = 34      # pip centres of one card share a row within this many pixels
_TAG_MAX_X = 130   # the black tag is drawn in the left page margin

_NO_ANSWER = "holder-0"


def _components(mask: np.ndarray) -> list[tuple[float, float, int, int]]:
    """Return (cy, cx, y, x) for each 4-connected blob of a boolean mask."""
    height, width = mask.shape
    parent: dict[int, int] = {}

    def find(a: int) -> int:
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a: int, b: int) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    runs_by_row: list[list[tuple[int, int, int]]] = []
    next_id = 0
    for y in range(height):
        row = mask[y]
        runs: list[tuple[int, int, int]] = []
        x = 0
        while x < width:
            if row[x]:
                start = x
                while x < width and row[x]:
                    x += 1
                parent[next_id] = next_id
                runs.append((start, x - 1, next_id))
                next_id += 1
            else:
                x += 1
        if y and runs and runs_by_row[-1]:
            for a0, a1, aid in runs:
                for b0, b1, bid in runs_by_row[-1]:
                    if a0 <= b1 and b0 <= a1:
                        union(aid, bid)
        runs_by_row.append(runs)

    acc: dict[int, list[float]] = {}
    for y, runs in enumerate(runs_by_row):
        for x0, x1, rid in runs:
            root = find(rid)
            slot = acc.setdefault(root, [0.0, 0.0, 0.0])
            count = x1 - x0 + 1
            slot[0] += count
            slot[1] += count * y
            slot[2] += (x0 + x1) * count / 2.0
    out = []
    for slot in acc.values():
        if slot[0] < 60:
            continue
        cy = slot[1] / slot[0]
        cx = slot[2] / slot[0]
        out.append((cy, cx, int(round(cy)), int(round(cx))))
    return out


def _rows(pips: list[tuple[float, float, tuple[int, int, int]]]) -> list[list]:
    groups: list[list] = []
    for pip in sorted(pips, key=lambda p: p[0]):
        if groups and abs(pip[0] - groups[-1][0][0]) <= _ROW_TOL:
            groups[-1].append(pip)
        else:
            groups.append([pip])
    return [sorted(g, key=lambda p: p[1]) for g in groups]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)
    spread = arr.max(axis=2) - arr.min(axis=2)
    vivid = spread > _SPREAD

    pips: list[tuple[float, float, tuple[int, int, int]]] = []
    for cy, cx, y, x in _components(vivid):
        pips.append((cy, cx, tuple(int(v) for v in arr[y, x])))
    if not pips:
        return _NO_ANSWER

    badge_rows = _rows([p for p in pips if p[1] < _SIDE_SPLIT])
    holder_rows = _rows([p for p in pips if p[1] >= _SIDE_SPLIT])
    if len(badge_rows) != 5 or len(holder_rows) != 5:
        return _NO_ANSWER
    if any(len(r) != 2 for r in badge_rows) or any(len(r) != 4 for r in holder_rows):
        return _NO_ANSWER

    badges = [frozenset(p[2] for p in row) for row in badge_rows]
    holders = [frozenset(p[2] for p in row) for row in holder_rows]
    if any(len(s) != 2 for s in badges) or any(len(s) != 4 for s in holders):
        return _NO_ANSWER

    dark = (arr.max(axis=2) < 40) & (
        np.arange(arr.shape[1])[None, :] < _TAG_MAX_X
    )
    tags = _components(dark)
    if len(tags) != 1:
        return _NO_ANSWER
    tag_y = tags[0][0]
    marked = min(
        range(5), key=lambda i: abs(badge_rows[i][0][0] - tag_y)
    )
    if abs(badge_rows[marked][0][0] - tag_y) > _ROW_TOL:
        return _NO_ANSWER

    compat = [
        sum(1 << h for h, group in enumerate(holders) if pair <= group)
        for pair in badges
    ]
    found = [
        perm
        for perm in itertools.permutations(range(5))
        if all(compat[b] >> perm[b] & 1 for b in range(5))
    ]
    if not found:
        return _NO_ANSWER
    reached = {perm[marked] for perm in found}
    barred = [
        h for h in range(5) if (compat[marked] >> h & 1) and h not in reached
    ]
    if len(barred) != 1:
        return _NO_ANSWER
    return f"holder-{barred[0] + 1}"
