"""Barred-holder binding world.

Five badges on the left each carry two coloured pips; five holders on the
right each carry four coloured pips.  A badge fits a holder when both of the
badge's colours appear on that holder.  A full arrangement puts every badge
into a holder it fits, one badge per holder.  Many full arrangements exist,
so no badge is forced by simple elimination.  The black-tagged badge fits at
least three holders, at least two of which really do receive it in some full
arrangement, and exactly one of which no full arrangement ever gives it.  The
question asks for that barred holder, so the label is a negative fact about
every arrangement rather than an entry of any one arrangement.
"""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

# ------------------------------------------------------------------ geometry

WIDTH = 1080
HEIGHT = 700

BG = (244, 242, 236)
CARD_FILL = (253, 253, 251)
CARD_EDGE = (118, 118, 112)
TAG = (0, 0, 0)

BADGE_X0, BADGE_X1 = 140, 500
BADGE_TOP, BADGE_PITCH, CARD_H = 74, 118, 84
BADGE_PIP_X = (280, 370)

HOLDER_X0, HOLDER_X1 = 592, 1040
HOLDER_TOP, HOLDER_PITCH = 46, 124

MAX_JITTER = 12
HOLDER_PIP_X = (660, 760, 860, 960)

PIP_D = 52
EDGE_W = 4
TAG_X0, TAG_SIDE = 62, 46

N = 5
COLOURS = 6

PALETTE = (
    (215, 35, 35),
    (40, 155, 55),
    (25, 55, 190),
    (240, 105, 170),
    (245, 215, 40),
    (40, 205, 210),
)

QUARANTINE_MARGIN = 24.0

PAIRS = [frozenset(p) for p in itertools.combinations(range(COLOURS), 2)]
QUADS = [frozenset(q) for q in itertools.combinations(range(COLOURS), 4)]

_NO_ANSWER = "holder-0"


# ------------------------------------------------------------------ mechanism

def _compat(badges: list[frozenset], holders: list[frozenset]) -> list[int]:
    return [
        sum(1 << h for h, group in enumerate(holders) if pair <= group)
        for pair in badges
    ]


def _perfect_matchings(compat: list[int]) -> list[tuple[int, ...]]:
    return [
        perm
        for perm in itertools.permutations(range(N))
        if all(compat[b] >> perm[b] & 1 for b in range(N))
    ]


def _sets(scene: dict) -> tuple[list[frozenset], list[frozenset]]:
    badges = [frozenset(p) for p in scene["badge_pips"]]
    holders = [frozenset(q) for q in scene["holder_pips"]]
    return badges, holders


def _barred(compat: list[int], marked: int) -> list[int]:
    """Holders the tagged badge fits that no full arrangement ever gives it."""
    fits = [h for h in range(N) if compat[marked] >> h & 1]
    reached = {perm[marked] for perm in _perfect_matchings(compat)}
    return [h for h in fits if h not in reached]


def _solve(scene: dict) -> int:
    badges, holders = _sets(scene)
    compat = _compat(badges, holders)
    barred = _barred(compat, int(scene["marked"]))
    if len(barred) != 1:
        raise ValueError("scene does not bar exactly one fitting holder")
    return barred[0]


def analytic_gold(scene: dict) -> str:
    return f"holder-{_solve(scene) + 1}"


# ------------------------------------------------------------------ sampling

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    for _ in range(400000):
        badges = rng.sample(PAIRS, N)
        holders = rng.sample(QUADS, N)
        compat = _compat(badges, holders)
        # no badge and no holder may be settled by simple elimination
        if any(bin(mask).count("1") < 2 for mask in compat):
            continue
        if any(
            sum(1 for b in range(N) if compat[b] >> h & 1) < 2 for h in range(N)
        ):
            continue
        matchings = _perfect_matchings(compat)
        if len(matchings) < 3:
            continue  # several full arrangements must exist
        order = list(range(N))
        rng.shuffle(order)
        for marked in order:
            fits = [h for h in range(N) if compat[marked] >> h & 1]
            if len(fits) < 3:
                continue  # the tag's own pips must leave three legal holders
            barred = _barred(compat, marked)
            if len(barred) != 1:
                continue
            answer = barred[0]
            if answer == marked:
                continue  # row position of the tag never names the answer
            degrees = {
                h: sum(1 for b in range(N) if compat[b] >> h & 1) for h in fits
            }
            values = list(degrees.values())
            low, high = min(values), max(values)
            if degrees[answer] == low and values.count(low) == 1:
                continue  # the least popular fitting holder is never the answer
            if degrees[answer] == high and values.count(high) == 1:
                continue  # nor the most popular one
            # at least three badges must float across the arrangements
            floating = sum(
                1
                for b in range(N)
                if len({perm[b] for perm in matchings}) >= 2
            )
            if floating < 3:
                continue
            palette = [list(c) for c in rng.sample(PALETTE, COLOURS)]
            badge_dy = [rng.randint(0, MAX_JITTER) for _ in range(N)]
            holder_dy = [rng.randint(0, MAX_JITTER) for _ in range(N)]
            badge_pips = []
            for pair in badges:
                slots = sorted(pair)
                rng.shuffle(slots)
                badge_pips.append(slots)
            holder_pips = []
            for group in holders:
                slots = sorted(group)
                rng.shuffle(slots)
                holder_pips.append(slots)
            paint_order = list(range(2 * N))
            rng.shuffle(paint_order)
            return {
                "badge_pips": badge_pips,
                "holder_pips": holder_pips,
                "palette": palette,
                "badge_dy": badge_dy,
                "holder_dy": holder_dy,
                "marked": int(marked),
                "paint_order": paint_order,
            }
    raise RuntimeError("sampler exhausted")


def margin(scene: dict) -> float:
    """Rows of clear background between the closest two cards of one column.

    Every input to the answer is a card's own pip set, so the visible boundary
    is whether neighbouring cards stay separate objects; the six pip colours
    are fixed and pairwise at least 150 RGB units apart.
    """
    gaps = []
    for index in range(N - 1):
        gaps.append(_badge_box(index + 1, scene)[1] - _badge_box(index, scene)[3] - 1)
        gaps.append(_holder_box(index + 1, scene)[1] - _holder_box(index, scene)[3] - 1)
    return float(min(gaps))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    order = list(scene["paint_order"])
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    flipped = dict(scene)
    flipped["paint_order"] = list(reversed(order))
    return [("paint_order_rotate", rotated), ("paint_order_reverse", flipped)]


# ------------------------------------------------------------------ renderer

def _badge_box(index: int, scene: dict) -> tuple[int, int, int, int]:
    top = BADGE_TOP + index * BADGE_PITCH + int(scene["badge_dy"][index])
    return BADGE_X0, top, BADGE_X1, top + CARD_H


def _holder_box(index: int, scene: dict) -> tuple[int, int, int, int]:
    top = HOLDER_TOP + index * HOLDER_PITCH + int(scene["holder_dy"][index])
    return HOLDER_X0, top, HOLDER_X1, top + CARD_H


def _pip(draw: ImageDraw.ImageDraw, cx: int, cy: int, colour: tuple[int, int, int]) -> None:
    half = PIP_D // 2
    draw.ellipse([cx - half, cy - half, cx + half - 1, cy + half - 1], fill=colour)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    palette = [tuple(int(v) for v in c) for c in scene["palette"]]

    for card in scene["paint_order"]:
        card = int(card)
        if card < N:
            x0, y0, x1, y1 = _badge_box(card, scene)
            draw.rectangle([x0, y0, x1, y1], fill=CARD_FILL, outline=CARD_EDGE, width=EDGE_W)
            cy = (y0 + y1) // 2
            for slot, colour_index in enumerate(scene["badge_pips"][card]):
                _pip(draw, BADGE_PIP_X[slot], cy, palette[int(colour_index)])
            if card == int(scene["marked"]):
                draw.rectangle(
                    [TAG_X0, cy - TAG_SIDE // 2, TAG_X0 + TAG_SIDE, cy + TAG_SIDE // 2],
                    fill=TAG,
                )
        else:
            holder = card - N
            x0, y0, x1, y1 = _holder_box(holder, scene)
            draw.rectangle([x0, y0, x1, y1], fill=CARD_FILL, outline=CARD_EDGE, width=EDGE_W)
            cy = (y0 + y1) // 2
            for slot, colour_index in enumerate(scene["holder_pips"][holder]):
                _pip(draw, HOLDER_PIP_X[slot], cy, palette[int(colour_index)])
    return image
