"""Deterministic renderer for depot-valued bilateral transfers."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw, ImageFont

SIZE = (960, 720)
BACKGROUND = (246, 244, 238)
PANEL = (255, 254, 250)
INK = (38, 42, 48)
MUTED = (214, 211, 203)
ACCENT = (181, 83, 55)
TOKEN_COLORS = ((24, 137, 181), (146, 75, 163), (228, 130, 30), (52, 153, 91))
PIP_COLOR = (238, 185, 45)
LR_COLOR = (22, 132, 143)
RL_COLOR = (128, 82, 168)
DECISIONS = ("plan-A", "plan-B", "plan-C", "plan-D")
CARD_X = (18, 254, 490, 726)
PLAN_X = CARD_X
START_X = (22, 495)
STOCK_Y = (250, 282, 314, 346)
TRANSFER_Y = (500, 585, 670)
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size)


def _final(scene_values, left_stock, right_stock, plan):
    left_values, right_values = scene_values
    left = list(left_stock)
    right = list(right_stock)
    for color, count, direction in plan:
        color, count, direction = int(color), int(count), int(direction)
        if direction == 1:
            left[color] -= count
            right[color] += count
        else:
            right[color] -= count
            left[color] += count
    left_total = sum(left[c] * left_values[c] for c in range(4))
    right_total = sum(right[c] * right_values[c] for c in range(4))
    return tuple(left), tuple(right), left_total, right_total


def _valid_plan(left_stock, right_stock, plan) -> bool:
    if len(plan) != 3 or len({int(move[0]) for move in plan}) != 3:
        return False
    for color, count, direction in plan:
        if not 0 <= int(color) < 4 or not 1 <= int(count) <= 3 or int(direction) not in (-1, 1):
            return False
        available = left_stock[int(color)] if int(direction) == 1 else right_stock[int(color)]
        if int(count) > available:
            return False
    return True


def _all_plans(left_stock, right_stock):
    plans = []
    for colors in itertools.combinations(range(4), 3):
        for counts in itertools.product(range(1, 4), repeat=3):
            for directions in itertools.product((-1, 1), repeat=3):
                plan = tuple((colors[i], counts[i], directions[i]) for i in range(3))
                if _valid_plan(left_stock, right_stock, plan):
                    plans.append(plan)
    return plans


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    for _ in range(800):
        left_values = list(range(1, 5))
        right_values = list(range(1, 5))
        rng.shuffle(left_values)
        rng.shuffle(right_values)
        left_stock = tuple(rng.randint(3, 7) for _ in range(4))
        right_stock = tuple(rng.randint(3, 7) for _ in range(4))
        scored = []
        for plan in _all_plans(left_stock, right_stock):
            _, _, left_total, right_total = _final(
                (left_values, right_values), left_stock, right_stock, plan
            )
            scored.append((plan, left_total - right_total))
        targets = [plan for plan, difference in scored if difference == 0]
        distractors = [(plan, difference) for plan, difference in scored if 2 <= abs(difference) <= 14]
        if not targets or len(distractors) < 3:
            continue
        rng.shuffle(targets)
        rng.shuffle(distractors)
        target = targets[0]
        selected = []
        seen_differences = set()
        for plan, difference in distractors:
            if difference in seen_differences:
                continue
            selected.append(plan)
            seen_differences.add(difference)
            if len(selected) == 3:
                break
        if len(selected) != 3:
            continue
        shown = [target, *selected]
        winner = int(seed) % 4
        shown[0], shown[winner] = shown[winner], shown[0]
        return {
            "left_values": left_values,
            "right_values": right_values,
            "left_stock": list(left_stock),
            "right_stock": list(right_stock),
            "plans": [[[int(v) for v in move] for move in plan] for plan in shown],
            "layout_variant": rng.randrange(256),
        }
    raise RuntimeError("unable to sample a valid transfer scene")


def _validated(scene: dict):
    left_values = tuple(int(x) for x in scene["left_values"])
    right_values = tuple(int(x) for x in scene["right_values"])
    left_stock = tuple(int(x) for x in scene["left_stock"])
    right_stock = tuple(int(x) for x in scene["right_stock"])
    plans = tuple(tuple(tuple(int(v) for v in move) for move in plan) for plan in scene["plans"])
    variant = int(scene["layout_variant"])
    if len(left_values) != 4 or len(right_values) != 4 or sorted(left_values) != [1, 2, 3, 4] or sorted(right_values) != [1, 2, 3, 4]:
        raise ValueError("bad value keys")
    if len(left_stock) != 4 or len(right_stock) != 4 or any(not 3 <= x <= 7 for x in (*left_stock, *right_stock)):
        raise ValueError("bad starting stock")
    if len(plans) != 4 or any(not _valid_plan(left_stock, right_stock, plan) for plan in plans):
        raise ValueError("bad transfer plan")
    if not 0 <= variant < 256:
        raise ValueError("bad layout variant")
    return left_values, right_values, left_stock, right_stock, plans, variant


def analytic_gold(scene: dict) -> str:
    try:
        left_values, right_values, left_stock, right_stock, plans, _ = _validated(scene)
        winners = []
        for index, plan in enumerate(plans):
            left, right, left_total, right_total = _final(
                (left_values, right_values), left_stock, right_stock, plan
            )
            if min(*left, *right) < 0:
                return "abstain"
            if left_total == right_total:
                winners.append(index)
        return DECISIONS[winners[0]] if len(winners) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    left_values, right_values, left_stock, right_stock, plans, _ = _validated(scene)
    differences = []
    for plan in plans:
        _, _, left_total, right_total = _final((left_values, right_values), left_stock, right_stock, plan)
        if left_total != right_total:
            differences.append(abs(left_total - right_total))
    return float(min(differences)) if differences else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        _, _, _, _, plans, _ = _validated(scene)
        return analytic_gold(scene) == "abstain" or margin(scene) < 2 or len(set(plans)) != 4
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _disks(draw, count: int, box, color, radius: int, reverse: bool = False) -> None:
    x0, y0, x1, y1 = box
    count = int(count)
    gap = radius * 2 + 5
    xs = [((x0 + x1) // 2 - (count - 1) * gap // 2 + i * gap) for i in range(count)]
    if reverse:
        xs.reverse()
    cy = (y0 + y1) // 2
    for cx in xs:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=1)


def _arrow(draw, x0: int, x1: int, cy: int, direction: int) -> None:
    color = LR_COLOR if direction == 1 else RL_COLOR
    if direction == 1:
        draw.line((x0, cy, x1 - 10, cy), fill=color, width=7)
        draw.polygon(((x1, cy), (x1 - 14, cy - 11), (x1 - 14, cy + 11)), fill=color)
    else:
        draw.line((x1, cy, x0 + 10, cy), fill=color, width=7)
        draw.polygon(((x0, cy), (x0 + 14, cy - 11), (x0 + 14, cy + 11)), fill=color)


def render(scene: dict) -> Image.Image:
    left_values, right_values, left_stock, right_stock, plans, variant = _validated(scene)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    title, label, small = _font(22), _font(17), _font(14)
    draw.text((20, 10), "CRATE VALUE IN EACH DEPOT", font=title, fill=INK)
    for color_index, x0 in enumerate(CARD_X):
        draw.rounded_rectangle((x0, 42, x0 + 216, 178), radius=10, fill=PANEL, outline=TOKEN_COLORS[color_index], width=4)
        draw.ellipse((x0 + 100, 51, x0 + 116, 67), fill=TOKEN_COLORS[color_index], outline=INK, width=1)
        draw.text((x0 + 18, 78), "LEFT", font=small, fill=INK)
        draw.text((x0 + 139, 78), "RIGHT", font=small, fill=INK)
        draw.line((x0 + 108, 76, x0 + 108, 168), fill=MUTED, width=2)
        _disks(draw, left_values[color_index], (x0 + 10, 112, x0 + 100, 164), PIP_COLOR, 6, bool((variant >> color_index) & 1))
        _disks(draw, right_values[color_index], (x0 + 118, 112, x0 + 208, 164), PIP_COLOR, 6, bool((variant >> (color_index + 4)) & 1))

    draw.text((20, 187), "START INVENTORIES", font=label, fill=INK)
    for side, x0, stock in (("LEFT", START_X[0], left_stock), ("RIGHT", START_X[1], right_stock)):
        draw.rounded_rectangle((x0, 216, x0 + 443, 365), radius=10, fill=PANEL, outline=ACCENT, width=4)
        draw.text((x0 + 12, 222), side, font=small, fill=INK)
        for color_index, (count, cy) in enumerate(zip(stock, STOCK_Y)):
            draw.ellipse((x0 + 22, cy - 9, x0 + 40, cy + 9), fill=TOKEN_COLORS[color_index], outline=INK, width=1)
            _disks(draw, count, (x0 + 62, cy - 12, x0 + 421, cy + 12), TOKEN_COLORS[color_index], 7, bool((variant + color_index + x0) & 1))

    draw.text((20, 374), "TRANSFER PLANS", font=label, fill=INK)
    for plan_index, (plan, x0) in enumerate(zip(plans, PLAN_X)):
        draw.rounded_rectangle((x0, 404, x0 + 216, 706), radius=10, fill=PANEL, outline=INK, width=3)
        draw.text((x0 + 72, 416), f"plan {chr(65 + plan_index)}", font=small, fill=INK)
        draw.line((x0 + 10, 452, x0 + 206, 452), fill=MUTED, width=2)
        for row, ((color_index, count, direction), cy) in enumerate(zip(plan, TRANSFER_Y)):
            draw.text((x0 + 10, cy - 10), "L", font=small, fill=INK)
            draw.text((x0 + 197, cy - 10), "R", font=small, fill=INK)
            _disks(draw, count, (x0 + 39, cy - 16, x0 + 116, cy + 16), TOKEN_COLORS[color_index], 8, bool((variant + plan_index + row) & 1))
            _arrow(draw, x0 + 126, x0 + 188, cy, direction)
            if row < 2:
                draw.line((x0 + 10, cy + 39, x0 + 206, cy + 39), fill=MUTED, width=1)
    return image
