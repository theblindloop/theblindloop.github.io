"""Independent image-only inverse for directional value transfer."""

from __future__ import annotations

import numpy as np
from PIL import Image

TOKEN_COLORS = ((24, 137, 181), (146, 75, 163), (228, 130, 30), (52, 153, 91))
PIP_COLOR = (238, 185, 45)
LR_COLOR = (22, 132, 143)
RL_COLOR = (128, 82, 168)
CARD_X = (18, 254, 490, 726)
PLAN_X = CARD_X
START_X = (22, 495)
STOCK_Y = (250, 282, 314, 346)
TRANSFER_Y = (500, 585, 670)


def _component_count(mask: np.ndarray, low: int = 55, high: int = 260) -> int | None:
    height, width = mask.shape
    remaining = set(int(v) for v in np.flatnonzero(mask))
    sizes = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        size = 0
        while stack:
            value = stack.pop()
            y, x = divmod(value, width)
            size += 1
            for neighbor in (value - 1, value + 1, value - width, value + width):
                if neighbor in remaining:
                    ny, nx = divmod(neighbor, width)
                    if abs(nx - x) + abs(ny - y) == 1:
                        remaining.remove(neighbor)
                        stack.append(neighbor)
        if size >= low:
            sizes.append(size)
    if any(size > high for size in sizes):
        return None
    return len(sizes)


def _mask(crop: np.ndarray, color) -> np.ndarray:
    delta = np.max(np.abs(crop.astype(np.int16) - np.asarray(color, dtype=np.int16)), axis=2)
    return delta <= 2


def _count_color(array: np.ndarray, box, color) -> int | None:
    x0, y0, x1, y1 = box
    return _component_count(_mask(array[y0:y1, x0:x1], color))


def _direction(array: np.ndarray, box) -> int | None:
    x0, y0, x1, y1 = box
    crop = array[y0:y1, x0:x1]
    lr = int(_mask(crop, LR_COLOR).sum())
    rl = int(_mask(crop, RL_COLOR).sum())
    if lr >= 180 and rl == 0:
        return 1
    if rl >= 180 and lr == 0:
        return -1
    return None


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (720, 960, 3):
        return "abstain"

    left_values = []
    right_values = []
    for x0 in CARD_X:
        left = _count_color(array, (x0 + 8, 108, x0 + 102, 168), PIP_COLOR)
        right = _count_color(array, (x0 + 116, 108, x0 + 210, 168), PIP_COLOR)
        if left is None or right is None or not 1 <= left <= 4 or not 1 <= right <= 4:
            return "abstain"
        left_values.append(left)
        right_values.append(right)

    stocks = []
    for x0 in START_X:
        counts = []
        for color, cy in zip(TOKEN_COLORS, STOCK_Y):
            count = _count_color(array, (x0 + 58, cy - 13, x0 + 425, cy + 13), color)
            if count is None or not 3 <= count <= 7:
                return "abstain"
            counts.append(count)
        stocks.append(counts)

    plans = []
    for x0 in PLAN_X:
        plan = []
        for cy in TRANSFER_Y:
            color_counts = [
                _count_color(array, (x0 + 36, cy - 18, x0 + 119, cy + 18), color)
                for color in TOKEN_COLORS
            ]
            if any(value is None for value in color_counts):
                return "abstain"
            used = [index for index, value in enumerate(color_counts) if value]
            direction = _direction(array, (x0 + 121, cy - 18, x0 + 193, cy + 18))
            if len(used) != 1 or not 1 <= color_counts[used[0]] <= 3 or direction is None:
                return "abstain"
            plan.append((used[0], color_counts[used[0]], direction))
        if len({move[0] for move in plan}) != 3:
            return "abstain"
        plans.append(plan)

    winners = []
    for index, plan in enumerate(plans):
        left = list(stocks[0])
        right = list(stocks[1])
        for color, count, direction in plan:
            if direction == 1:
                left[color] -= count
                right[color] += count
            else:
                right[color] -= count
                left[color] += count
        if min(*left, *right) < 0:
            return "abstain"
        left_total = sum(left[color] * left_values[color] for color in range(4))
        right_total = sum(right[color] * right_values[color] for color in range(4))
        if left_total == right_total:
            winners.append(index)
    return f"plan-{chr(65 + winners[0])}" if len(winners) == 1 else "abstain"
