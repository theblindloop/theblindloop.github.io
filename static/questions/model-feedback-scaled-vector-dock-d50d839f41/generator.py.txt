"""Code-native renderer for a two-step affine point transformation."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1240
HEIGHT = 820
BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
INK = (24, 32, 48)
MUTED = (84, 96, 115)
GRID = (194, 202, 214)
GOLD = (232, 171, 24)
CYAN = (10, 154, 171)
SCALE_BAND = (111, 76, 176)

BOARD_X = 82
BOARD_Y = 190
CELL = 58
BOARD_N = 9
BOARD_RIGHT = BOARD_X + BOARD_N * CELL
BOARD_BOTTOM = BOARD_Y + BOARD_N * CELL

TARGETS: dict[str, tuple[int, int]] = {
    "red": (0, 0),
    "blue": (8, 0),
    "green": (0, 8),
    "violet": (8, 8),
}
TARGET_COLORS: dict[str, tuple[int, int, int]] = {
    "red": (211, 61, 70),
    "blue": (42, 111, 205),
    "green": (31, 157, 100),
    "violet": (133, 75, 183),
}

# Each vector is shown by a saturated full-width card band and repeated by an
# arrow plus a signed cell-count label. Coordinates are (delta-column, delta-row).
SLIDE_VECTORS: dict[str, tuple[int, int]] = {
    "east": (1, 0),
    "west": (-1, 0),
    "south": (0, 1),
    "north": (0, -1),
    "southeast": (1, 1),
    "southwest": (-1, 1),
    "northeast": (1, -1),
    "northwest": (-1, -1),
}
SLIDE_BANDS: dict[str, tuple[int, int, int]] = {
    "east": (0, 132, 196),
    "west": (205, 78, 55),
    "south": (0, 151, 113),
    "north": (207, 132, 24),
    "southeast": (33, 105, 164),
    "southwest": (181, 70, 139),
    "northeast": (85, 118, 38),
    "northwest": (155, 83, 150),
}
SECTOR_NAMES = ("northwest", "northeast", "southwest", "southeast")


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _on_board(cell: tuple[int, int]) -> bool:
    return 0 <= cell[0] < BOARD_N and 0 <= cell[1] < BOARD_N


def _cell_center(cell: tuple[int, int]) -> tuple[int, int]:
    return BOARD_X + cell[0] * CELL + CELL // 2, BOARD_Y + cell[1] * CELL + CELL // 2


def _final_cell(start: tuple[int, int], hub: tuple[int, int], slide: str) -> tuple[int, int]:
    dx, dy = SLIDE_VECTORS[slide]
    return (hub[0] + 2 * (start[0] - hub[0]) + dx, hub[1] + 2 * (start[1] - hub[1]) + dy)


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, int], tuple[int, int], str]:
    try:
        start_raw = scene["start_cell"]
        hub_raw = scene["hub_cell"]
        slide = str(scene["slide_vector"])
    except (KeyError, TypeError) as exc:
        raise ValueError("scene is missing a transformation field") from exc
    if not isinstance(start_raw, (list, tuple)) or len(start_raw) != 2:
        raise ValueError("start_cell must contain two coordinates")
    if not isinstance(hub_raw, (list, tuple)) or len(hub_raw) != 2:
        raise ValueError("hub_cell must contain two coordinates")
    start = (int(start_raw[0]), int(start_raw[1]))
    hub = (int(hub_raw[0]), int(hub_raw[1]))
    if not _on_board(start) or not _on_board(hub):
        raise ValueError("start and hub must be on the 9 by 9 board")
    if start == hub:
        raise ValueError("start and hub must be distinct")
    if start in TARGETS.values() or hub in TARGETS.values():
        raise ValueError("start and hub cannot cover a docking cell")
    if slide not in SLIDE_VECTORS:
        raise ValueError("unknown slide vector")
    doubled = (hub[0] + 2 * (start[0] - hub[0]), hub[1] + 2 * (start[1] - hub[1]))
    final = _final_cell(start, hub, slide)
    if not _on_board(doubled):
        raise ValueError("doubled point must remain on the board")
    if doubled in TARGETS.values():
        raise ValueError("doubled point cannot cover a docking cell")
    if not _on_board(final) or final not in TARGETS.values():
        raise ValueError("final point must land exactly on a docking cell")
    return start, hub, slide


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, readable integer scene for one desired dock color."""
    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 431)
    wanted = tuple(TARGETS)[seed % len(TARGETS)]
    options: list[tuple[tuple[int, int], tuple[int, int], str]] = []
    for slide, (dx, dy) in SLIDE_VECTORS.items():
        target = TARGETS[wanted]
        doubled = (target[0] - dx, target[1] - dy)
        if not _on_board(doubled) or doubled in TARGETS.values():
            continue
        for hub_x in range(BOARD_N):
            for hub_y in range(BOARD_N):
                hub = (hub_x, hub_y)
                if hub in TARGETS.values():
                    continue
                # doubled = 2*start - hub, so the parity of doubled+hub selects start.
                total_x = doubled[0] + hub[0]
                total_y = doubled[1] + hub[1]
                if total_x % 2 or total_y % 2:
                    continue
                start = (total_x // 2, total_y // 2)
                if not _on_board(start) or start == hub or start in TARGETS.values():
                    continue
                if _on_board((hub[0] + 2 * (start[0] - hub[0]), hub[1] + 2 * (start[1] - hub[1]))):
                    options.append((start, hub, slide))
    if not options:
        raise RuntimeError(f"could not sample an on-board scene for seed {seed}")
    start, hub, slide = rng.choice(options)
    return {
        "start_cell": [start[0], start[1]],
        "hub_cell": [hub[0], hub[1]],
        "slide_vector": slide,
    }


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=(207, 214, 224), width=2)


def _draw_dock(draw: ImageDraw.ImageDraw, name: str, cell: tuple[int, int]) -> None:
    left = BOARD_X + cell[0] * CELL + 6
    top = BOARD_Y + cell[1] * CELL + 6
    right = BOARD_X + (cell[0] + 1) * CELL - 6
    bottom = BOARD_Y + (cell[1] + 1) * CELL - 6
    color = TARGET_COLORS[name]
    draw.rounded_rectangle((left, top, right, bottom), radius=9, fill=(255, 255, 255), outline=color, width=5)
    draw.ellipse((left + 13, top + 13, right - 13, bottom - 13), fill=color)
    draw.text(((left + right) // 2, (top + bottom) // 2), name[0].upper(), font=_font(12, True), fill=(255, 255, 255), anchor="mm")


def _draw_hub(draw: ImageDraw.ImageDraw, cell: tuple[int, int]) -> None:
    cx, cy = _cell_center(cell)
    draw.ellipse((cx - 19, cy - 19, cx + 19, cy + 19), fill=(255, 255, 255), outline=CYAN, width=5)
    draw.line((cx - 24, cy, cx + 24, cy), fill=CYAN, width=5)
    draw.line((cx, cy - 24, cx, cy + 24), fill=CYAN, width=5)
    draw.text((cx, cy - 1), "H", font=_font(12, True), fill=INK, anchor="mm")


def _draw_slide_arrow(draw: ImageDraw.ImageDraw, center: tuple[int, int], vector: tuple[int, int]) -> None:
    cx, cy = center
    dx, dy = vector
    scale = 54
    ex, ey = cx + dx * scale, cy + dy * scale
    draw.line((cx - dx * 12, cy - dy * 12, ex, ey), fill=INK, width=9)
    ux, uy = dx / (abs(dx) + abs(dy)), dy / (abs(dx) + abs(dy))
    px, py = -uy, ux
    tip = (ex, ey)
    base = (ex - ux * 28, ey - uy * 28)
    draw.polygon([tip, (base[0] + px * 17, base[1] + py * 17), (base[0] - px * 17, base[1] - py * 17)], fill=INK)
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    start, hub, slide = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 23), "SCALE · SLIDE · DOCK", font=_font(30, True), fill=INK)
    draw.text((40, 63), "Carry the gold point through both cards in order.", font=_font(17), fill=MUTED)

    _panel(draw, (30, 98, 665, 790))
    draw.text((55, 117), "9 BY 9 CELL MAP", font=_font(19, True), fill=INK)
    draw.text((55, 146), "H is the cyan hub; colored corners are docks.", font=_font(14), fill=MUTED)
    for index in range(BOARD_N + 1):
        x = BOARD_X + index * CELL
        y = BOARD_Y + index * CELL
        draw.line((x, BOARD_Y, x, BOARD_BOTTOM), fill=GRID, width=2)
        draw.line((BOARD_X, y, BOARD_RIGHT, y), fill=GRID, width=2)
    for index in range(BOARD_N):
        draw.text((BOARD_X + index * CELL + CELL // 2, BOARD_Y - 12), str(index + 1), font=_font(11, True), fill=MUTED, anchor="ms")
        draw.text((BOARD_X - 12, BOARD_Y + index * CELL + CELL // 2), str(index + 1), font=_font(11, True), fill=MUTED, anchor="rm")
    for name, cell in TARGETS.items():
        _draw_dock(draw, name, cell)
    _draw_hub(draw, hub)
    sx, sy = _cell_center(start)
    draw.ellipse((sx - 18, sy - 18, sx + 18, sy + 18), fill=GOLD, outline=INK, width=4)
    draw.ellipse((sx - 7, sy - 9, sx - 1, sy - 3), fill=(255, 239, 164))
    draw.text((55, 730), "START = gold point    HUB = cyan cross-pin", font=_font(14, True), fill=MUTED)

    _panel(draw, (700, 98, 1210, 350))
    draw.rectangle((727, 120, 1183, 158), fill=SCALE_BAND)
    draw.text((750, 139), "CARD 1  ·  DOUBLE FROM HUB", font=_font(17, True), fill=(255, 255, 255), anchor="lm")
    draw.text((748, 184), "Keep H fixed. Double the point's", font=_font(18, True), fill=INK)
    draw.text((748, 213), "displacement from H.", font=_font(18, True), fill=INK)
    # A neutral diagram explains the operation without adding a second encoded input.
    cx, cy = 835, 294
    draw.ellipse((cx - 9, cy - 9, cx + 9, cy + 9), fill=INK)
    draw.line((cx, cy, cx + 56, cy - 38), fill=INK, width=6)
    draw.ellipse((cx + 48, cy - 46, cx + 64, cy - 30), fill=INK)
    draw.line((cx, cy, cx + 112, cy - 76), fill=INK, width=6)
    draw.polygon([(cx + 128, cy - 87), (cx + 104, cy - 82), (cx + 119, cy - 64)], fill=INK)
    draw.text((1000, 284), "×2", font=_font(31, True), fill=INK, anchor="mm")
    draw.text((1000, 320), "from H", font=_font(14, True), fill=MUTED, anchor="mm")

    _panel(draw, (700, 378, 1210, 650))
    draw.rectangle((727, 400, 1183, 438), fill=SLIDE_BANDS[slide])
    draw.text((750, 419), "CARD 2  ·  SLIDE BY GRID-VECTOR", font=_font(17, True), fill=(255, 255, 255), anchor="lm")
    draw.text((748, 462), "Translate the doubled point by:", font=_font(18, True), fill=INK)
    dx, dy = SLIDE_VECTORS[slide]
    signed = f"dcol {dx:+d}, drow {dy:+d}  (cells)"
    draw.text((748, 493), signed, font=_font(20, True), fill=INK)
    _draw_slide_arrow(draw, (840, 592), (dx, dy))
    draw.text((1010, 572), slide.upper(), font=_font(24, True), fill=INK, anchor="mm")
    draw.text((1010, 610), "one cell each way shown", font=_font(14), fill=MUTED, anchor="mm")

    _panel(draw, (700, 680, 1210, 790), radius=14)
    draw.text((728, 708), "ORDER", font=_font(15, True), fill=MUTED, anchor="lm")
    draw.text((812, 708), "1  then  2", font=_font(21, True), fill=INK, anchor="lm")
    draw.text((728, 748), "Which colored dock is on the final cell?", font=_font(16, True), fill=INK)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    start, hub, slide = _validate_scene(scene)
    final = _final_cell(start, hub, slide)
    for name, cell in TARGETS.items():
        if cell == final:
            return name
    raise ValueError("final cell has no dock")


def margin(scene: dict[str, Any]) -> float:
    _validate_scene(scene)
    # The final answer is a whole lattice cell; every other dock is at least
    # eight Manhattan cells away. This is a stable discrete docking margin.
    return 8.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    _validate_scene(scene)
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
