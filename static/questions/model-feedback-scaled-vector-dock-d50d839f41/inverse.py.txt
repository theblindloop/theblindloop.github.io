"""Independent pixel-only inverse for the scale-and-slide world."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1240
HEIGHT = 820
BOARD_X = 82
BOARD_Y = 190
CELL = 58
BOARD_N = 9
GOLD = np.array((232, 171, 24), dtype=np.int16)
CYAN = np.array((10, 154, 171), dtype=np.int16)
SCALE_BAND = np.array((111, 76, 176), dtype=np.int16)

TARGETS = {
    "red": (0, 0),
    "blue": (8, 0),
    "green": (0, 8),
    "violet": (8, 8),
}
TARGET_COLORS = {
    "red": np.array((211, 61, 70), dtype=np.int16),
    "blue": np.array((42, 111, 205), dtype=np.int16),
    "green": np.array((31, 157, 100), dtype=np.int16),
    "violet": np.array((133, 75, 183), dtype=np.int16),
}
SLIDE_VECTORS = {
    "east": (1, 0),
    "west": (-1, 0),
    "south": (0, 1),
    "north": (0, -1),
    "southeast": (1, 1),
    "southwest": (-1, 1),
    "northeast": (1, -1),
    "northwest": (-1, -1),
}
SLIDE_BANDS = {
    "east": np.array((0, 132, 196), dtype=np.int16),
    "west": np.array((205, 78, 55), dtype=np.int16),
    "south": np.array((0, 151, 113), dtype=np.int16),
    "north": np.array((207, 132, 24), dtype=np.int16),
    "southeast": np.array((33, 105, 164), dtype=np.int16),
    "southwest": np.array((181, 70, 139), dtype=np.int16),
    "northeast": np.array((85, 118, 38), dtype=np.int16),
    "northwest": np.array((155, 83, 150), dtype=np.int16),
}


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray, minimum: int, label: str) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < minimum:
        raise ValueError(f"{label} is not recoverable")
    return float(xs.mean()), float(ys.mean())


def _cell_from_xy(x: float, y: float) -> tuple[int, int]:
    col = int(round((x - (BOARD_X + CELL / 2)) / CELL))
    row = int(round((y - (BOARD_Y + CELL / 2)) / CELL))
    if not (0 <= col < BOARD_N and 0 <= row < BOARD_N):
        raise ValueError(f"mark is outside the board: {(col, row)}")
    return col, row


def _recover_cell(arr: np.ndarray, color: np.ndarray, minimum: int, label: str) -> tuple[int, int]:
    x, y = _centroid(_near(arr, color, 10), minimum, label)
    return _cell_from_xy(x, y)


def _slide_value(arr: np.ndarray) -> str:
    crop = arr[400:439, 727:1184]
    counts = {name: int(_near(crop, color, 8).sum()) for name, color in SLIDE_BANDS.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    other = max(value for key, value in counts.items() if key != name)
    if count < 5000 or count <= other * 2:
        raise ValueError(f"slide card is unreadable: {counts}")
    return name


def _dock_at(arr: np.ndarray, cell: tuple[int, int]) -> str:
    if cell not in TARGETS.values():
        raise ValueError(f"computed cell is not a dock: {cell}")
    cx = BOARD_X + cell[0] * CELL + CELL // 2
    cy = BOARD_Y + cell[1] * CELL + CELL // 2
    crop = arr[cy - 20 : cy + 21, cx - 20 : cx + 21]
    counts = {name: int(_near(crop, color, 10).sum()) for name, color in TARGET_COLORS.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    if count < 150:
        raise ValueError(f"dock color is unreadable at {cell}: {counts}")
    if TARGETS[name] != cell:
        raise ValueError(f"dock layout is inconsistent at {cell}: {name}")
    return name


def decision_from_image(image: Image.Image) -> str:
    """Recover the dock color from final RGB pixels, without latent scene data."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected canvas size {rgb.size}")
    arr = np.asarray(rgb, dtype=np.int16)
    start = _recover_cell(arr, GOLD, 500, "gold start point")
    hub = _recover_cell(arr, CYAN, 500, "cyan hub")
    slide = _slide_value(arr)
    dx, dy = SLIDE_VECTORS[slide]
    doubled = (hub[0] + 2 * (start[0] - hub[0]), hub[1] + 2 * (start[1] - hub[1]))
    final = (doubled[0] + dx, doubled[1] + dy)
    return _dock_at(arr, final)
