"""Independent PNG-only inverse arm for the quarter-turn board world."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen), strict=True):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        min_x = max_x = x0
        min_y = max_y = y0
        area = 0
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((min_x, min_y, max_x, max_y, area))
    return output


def _rotate90(bits: tuple[int, ...]) -> tuple[int, ...]:
    return tuple(bits[(3 - x) * 4 + y] for y in range(4) for x in range(4))


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    dark = (rgb[:, :, 0] < 90) & (rgb[:, :, 1] < 90) & (rgb[:, :, 2] < 90)
    candidates = []
    for x0, y0, x1, y1, area in _components(dark):
        width, height = x1 - x0 + 1, y1 - y0 + 1
        if 130 <= width <= 170 and 130 <= height <= 170 and area > 2500:
            candidates.append((x0, y0, x1, y1))
    candidates.sort(key=lambda box: box[0])
    if len(candidates) != 4:
        raise ValueError(f"expected four visible grid components, found {len(candidates)}")

    blue = (rgb[:, :, 2] > 120) & (rgb[:, :, 1] > 70) & (rgb[:, :, 0] < 90)
    boards: list[tuple[int, ...]] = []
    for x0, y0, x1, y1 in candidates:
        span_x = x1 - x0
        span_y = y1 - y0
        bits = []
        for row in range(4):
            for col in range(4):
                xa = round(x0 + span_x * col / 4) + 5
                xb = round(x0 + span_x * (col + 1) / 4) - 4
                ya = round(y0 + span_y * row / 4) + 5
                yb = round(y0 + span_y * (row + 1) / 4) - 4
                region = blue[ya:yb, xa:xb]
                if region.size < 100:
                    raise ValueError("grid cell region is too small")
                bits.append(int(float(region.mean()) > 0.35))
        if sum(bits) != 8:
            raise ValueError(f"equal-ink contract violated: recovered {sum(bits)} filled cells")
        boards.append(tuple(bits))

    mismatches = [sum(a != b for a, b in zip(board, _rotate90(board), strict=True)) for board in boards]
    winners = [index for index, value in enumerate(mismatches) if value == 0]
    if len(winners) != 1:
        raise ValueError(f"expected unique quarter-turn board, found {winners}")
    if min(value for value in mismatches if value) < 2:
        raise ValueError("runner-up mismatch margin is too small")
    return _LABELS[winners[0]]
