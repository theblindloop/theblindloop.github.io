"""Deterministic renderer and analytic arm for quarter-turn board search."""

from __future__ import annotations

import itertools
import random
from functools import lru_cache

from PIL import Image, ImageDraw, ImageFont

LABELS = ("A", "B", "C", "D")
N = 4
FILLED = (31, 119, 180)
INK = (36, 43, 50)
PAPER = (248, 247, 243)


def _rotate90(bits: tuple[int, ...]) -> tuple[int, ...]:
    return tuple(bits[(N - 1 - x) * N + y] for y in range(N) for x in range(N))


def _rotate180(bits: tuple[int, ...]) -> tuple[int, ...]:
    return _rotate90(_rotate90(bits))


def _mirror_vertical(bits: tuple[int, ...]) -> tuple[int, ...]:
    return tuple(bits[y * N + (N - 1 - x)] for y in range(N) for x in range(N))


def _symmetry_flags(bits: tuple[int, ...]) -> tuple[bool, bool, bool]:
    q = _rotate90(bits) == bits
    h = _rotate180(bits) == bits
    reflections = []
    value = bits
    for _ in range(4):
        reflections.append(_mirror_vertical(value) == bits)
        value = _rotate90(value)
    return q, h, any(reflections)


@lru_cache(maxsize=1)
def _pattern_pools() -> dict[str, tuple[tuple[int, ...], ...]]:
    pools: dict[str, list[tuple[int, ...]]] = {
        "quarter": [],
        "half": [],
        "mirror": [],
        "asymmetric": [],
    }
    for occupied in itertools.combinations(range(16), 8):
        bitset = set(occupied)
        bits = tuple(int(i in bitset) for i in range(16))
        q, h, mirror = _symmetry_flags(bits)
        if q:
            pools["quarter"].append(bits)
        elif h:
            pools["half"].append(bits)
        elif mirror:
            pools["mirror"].append(bits)
        else:
            pools["asymmetric"].append(bits)
    return {name: tuple(values) for name, values in pools.items()}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    pools = _pattern_pools()
    target = rng.choice(pools["quarter"])
    distractors = [
        rng.choice(pools["half"]),
        rng.choice(pools["mirror"]),
        rng.choice(pools["asymmetric"]),
    ]
    rng.shuffle(distractors)
    target_position = int(seed) % 4
    panels: list[list[int]] = []
    iterator = iter(distractors)
    for index in range(4):
        panels.append(list(target if index == target_position else next(iterator)))
    return {"panels": panels}


def _panel_mismatch(panel: list[int] | tuple[int, ...]) -> int:
    bits = tuple(int(v) for v in panel)
    return sum(a != b for a, b in zip(bits, _rotate90(bits), strict=True))


def analytic_gold(scene: dict) -> str:
    scores = [_panel_mismatch(panel) for panel in scene["panels"]]
    winners = [index for index, value in enumerate(scores) if value == 0]
    if len(winners) != 1:
        raise ValueError(f"expected one quarter-turn invariant board, found {winners}")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_panel_mismatch(panel) for panel in scene["panels"])
    if len(scores) != 4 or scores[0] != 0:
        return 0.0
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    try:
        panels = scene["panels"]
        if len(panels) != 4:
            return True
        if any(len(panel) != 16 or sum(int(v) for v in panel) != 8 for panel in panels):
            return True
        values = [_panel_mismatch(panel) for panel in panels]
        return values.count(0) != 1 or margin(scene) < 2.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    if len(panels) != 4 or any(len(panel) != 16 for panel in panels):
        raise ValueError("render requires four 4-by-4 panels")
    image = Image.new("RGB", (900, 280), PAPER)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=22)
    board_size = 144
    cell = board_size // 4
    top = 74
    starts = (45, 255, 465, 675)
    for panel_index, (left, panel) in enumerate(zip(starts, panels, strict=True)):
        label = LABELS[panel_index]
        label_box = draw.textbbox((0, 0), label, font=font)
        label_width = label_box[2] - label_box[0]
        draw.text((left + (board_size - label_width) / 2, 30), label, fill=INK, font=font)
        for y in range(4):
            for x in range(4):
                if int(panel[y * 4 + x]):
                    x0 = left + x * cell + 4
                    y0 = top + y * cell + 4
                    draw.rectangle((x0, y0, x0 + cell - 8, y0 + cell - 8), fill=FILLED)
        for i in range(5):
            xx = left + i * cell
            yy = top + i * cell
            draw.line((xx, top, xx, top + board_size), fill=INK, width=3)
            draw.line((left, yy, left + board_size, yy), fill=INK, width=3)
    return image
