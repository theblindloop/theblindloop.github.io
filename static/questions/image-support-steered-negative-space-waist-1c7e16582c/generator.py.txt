"""Deterministic world for a two-bank negative-space width relation."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (247, 246, 241)
PALE_MARK = (229, 229, 224)
PALE_MARK_2 = (235, 233, 228)
PALETTES = (
    (25, 96, 112),
    (119, 53, 99),
    (181, 77, 48),
    (44, 79, 151),
    (42, 120, 77),
)
DECISION_THRESHOLD_PX = 6.0


def sample_scene(seed: int) -> dict:
    """Sample paired nuisance-identical scenes with opposite relation labels.

    Consecutive even/odd seeds share every nuisance field.  The top bank is
    therefore pixel-identical within a pair; the bottom bank changes only as
    needed to switch the relational answer.
    """

    rng = random.Random(int(seed) // 2)
    orientation = rng.randrange(2)
    # Grid-centered coordinates keep the compact motif translated while its
    # two decisive contours occupy separated 8x8 ablation cells.
    centers = (160, 224, 288, 352)
    center_x = rng.choice(centers)
    center_y = rng.choice(centers)
    label = "even" if int(seed) % 2 == 0 else "changing"
    distractors = []
    for _ in range(7):
        for _attempt in range(100):
            x = rng.randint(35, CANVAS - 36)
            y = rng.randint(35, CANVAS - 36)
            # Keep faint decoration away from the focal motif.
            if math.hypot(x - center_x, y - center_y) > 150:
                break
        distractors.append([x, y, rng.choice((3, 4, 5)), rng.randrange(2)])
    return {
        "label": label,
        "center_x": center_x,
        "center_y": center_y,
        "orientation": orientation,
        "half_length": rng.choice((48, 50, 52)),
        "base_gap": 128,
        "amplitude": rng.choice((9, 10, 11, 12)),
        "bank_thickness": rng.choice((21, 23, 25)),
        "palette_index": rng.randrange(len(PALETTES)),
        "phase": rng.choice((-1, 1)),
        "distractors": distractors,
    }


def _profile_signs(scene: dict) -> tuple[int, int]:
    top = int(scene["phase"])
    bottom = top if scene["label"] == "even" else -top
    return top, bottom


def _world_point(scene: dict, local_x: float, local_y: float) -> tuple[float, float]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    if int(scene["orientation"]) == 0:
        return cx + local_x, cy + local_y
    # Quarter-turn preserves the label and makes the corridor left/right.
    return cx - local_y, cy + local_x


def _curve_samples(half_length: float, count: int = 81) -> Iterable[tuple[float, float]]:
    for index in range(count):
        x = -half_length + 2.0 * half_length * index / (count - 1)
        u = x / half_length
        # Broad, unambiguous central displacement with visible shoulders.
        yield x, (1.0 - u * u) ** 2


def _polygon_points(scene: dict) -> tuple[list[tuple[float, float]], list[tuple[float, float]]]:
    half = float(scene["half_length"])
    gap = float(scene["base_gap"])
    amp = float(scene["amplitude"])
    thickness = float(scene["bank_thickness"])
    top_sign, bottom_sign = _profile_signs(scene)
    samples = list(_curve_samples(half))

    top_local = [(-half, -gap / 2.0 - thickness), (half, -gap / 2.0 - thickness)]
    top_local.extend(
        (x, -gap / 2.0 + top_sign * amp * q) for x, q in reversed(samples)
    )
    bottom_local = [(x, gap / 2.0 + bottom_sign * amp * q) for x, q in samples]
    bottom_local.extend([(half, gap / 2.0 + thickness), (-half, gap / 2.0 + thickness)])
    top = [_world_point(scene, x, y) for x, y in top_local]
    bottom = [_world_point(scene, x, y) for x, y in bottom_local]
    return top, bottom


def render(scene: dict) -> Image.Image:
    """Render at 4x resolution and downsample for stable antialiased contours."""

    scale = 4
    image = Image.new("RGB", (CANVAS * scale, CANVAS * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius, variant in scene["distractors"]:
        color = PALE_MARK if int(variant) == 0 else PALE_MARK_2
        box = tuple((value * scale) for value in (x - radius, y - radius, x + radius, y + radius))
        draw.ellipse(box, fill=color)
    ink = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    top, bottom = _polygon_points(scene)
    draw.polygon([(x * scale, y * scale) for x, y in top], fill=ink)
    draw.polygon([(x * scale, y * scale) for x, y in bottom], fill=ink)
    return image.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["label"])


def margin(scene: dict) -> float:
    """Runner-up margin in reconstructed-width pixels."""

    if scene["label"] == "even":
        return DECISION_THRESHOLD_PX
    # At |u|=.72 the shoulder profile has the value below; the two profiles
    # move oppositely in the changing class.
    shoulder_q = (1.0 - 0.72**2) ** 2
    return 2.0 * float(scene["amplitude"]) * (1.0 - shoulder_q) - DECISION_THRESHOLD_PX


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The canonical scene representation has no unused, reordered, or aliased
    # fields; quarter-turns are visible transforms rather than pixel aliases.
    return []
