"""Pixel-only inverse arm for the last-marker-on-route tracing world.

Reads only the final RGB PNG. It segments the slate-grey route stroke by
nearest-colour labelling, locates the orange start ring, seeds a geodesic
breadth-first distance field over the grey route centreline from that ring, and
for each coloured marker ring computes its centroid. The marker whose trajectory
is farthest along the route (largest geodesic distance from the ring) is the
answer colour. This requires tracing the stroke's ordered centreline from the
ring to the terminal: the answer is an on-path order fact, not a straight-line
distance, a bounding box, or an ink measure.

No renderer, prompt, generator, verifier or scene data is imported or read.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_PALETTE = {
    "white": (255, 255, 255),
    "slate": (112, 112, 118),
    "blue": (40, 90, 210),
    "green": (40, 170, 70),
    "red": (208, 60, 60),
    "orange": (245, 150, 20),
    "dark": (30, 30, 30),
}
_NAMES = ["white", "slate", "blue", "green", "red", "orange", "dark"]
_MARKER_NAMES = ["blue", "green", "red"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _largest_component_centroid(labels: np.ndarray, idx: int) -> tuple[float, float]:
    ys, xs = np.nonzero(labels == idx)
    if len(ys) == 0:
        raise ValueError(f"colour {_NAMES[idx]} not found")
    return (float(ys.mean()), float(xs.mean()))


def _bfs_dist(mask: np.ndarray, seed) -> np.ndarray:
    h, w = mask.shape
    dist = np.full((h, w), -1, dtype=np.int64)
    dq = deque()
    sy, sx = seed
    dist[sy, sx] = 0
    dq.append((sy, sx))
    while dq:
        y, x = dq.popleft()
        nd = dist[y, x] + 1
        for ny in range(y - 1, y + 2):
            for nx in range(x - 1, x + 2):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and dist[ny, nx] == -1:
                    dist[ny, nx] = nd
                    dq.append((ny, nx))
    return dist


def _route_dist_at_centroid(mask: np.ndarray, dist: np.ndarray, cy: float, cx: float) -> int:
    """Distance at the nearest mask pixel to a marker centroid.

    The centroid lies on the route figure, so its own BFS distance is used when
    that pixel is part of the mask; otherwise the closest mask pixel in a window
    is chosen.
    """
    h, w = mask.shape
    iy, ix = int(round(cy)), int(round(cx))
    if 0 <= iy < h and 0 <= ix < w and mask[iy, ix] and dist[iy, ix] != -1:
        return int(dist[iy, ix])
    best = -1
    best_d = 1 << 30
    radius = 20
    iy0 = max(0, iy - radius)
    iy1 = min(h, iy + radius + 1)
    ix0 = max(0, ix - radius)
    ix1 = min(w, ix + radius + 1)
    for y in range(iy0, iy1):
        for x in range(ix0, ix1):
            if mask[y, x] and dist[y, x] != -1:
                d = (y - cy) ** 2 + (x - cx) ** 2
                if d < best_d:
                    best_d = d
                    best = dist[y, x]
    if best < 0:
        raise ValueError("marker centroid has no nearby route ink")
    return best


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    white_idx = idx["white"]
    trace_mask = labels != white_idx
    if int(trace_mask.sum()) < 8:
        raise ValueError("figure not found")
    oy, ox = _largest_component_centroid(labels, idx["orange"])
    dist = _bfs_dist(trace_mask, (int(round(oy)), int(round(ox))))
    best_color = None
    best_val = -1
    for name in _MARKER_NAMES:
        cy, cx = _largest_component_centroid(labels, idx[name])
        v = _route_dist_at_centroid(trace_mask, dist, cy, cx)
        if v > best_val:
            best_val = v
            best_color = name
    if best_color is None:
        raise ValueError("no marker colours found")
    return best_color
