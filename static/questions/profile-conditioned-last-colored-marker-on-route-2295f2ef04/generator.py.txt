"""Last-marker-on-route tracing world: which colour marker is last on the path.

A single winding slate-grey route runs from an orange ring to a dark terminal
dot. Three coloured rings (blue, green, red) are centred on the route at three
well-separated points along it. Tracing the route from the ring toward the
terminal, the question asks which colour ring is encountered LAST (i.e. is
farthest along the route before the terminal). Recovering the answer requires
following the ordered centreline of the stroke and ordering the three markers by
their on-path (geodesic) position; it is a topological tracing fact that cannot
be recovered from straight-line distance, marker size, or ink area.
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "last_colored_marker_on_route-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_SLATE = (112, 112, 118)
COLOR_BLUE = (40, 90, 210)
COLOR_GREEN = (40, 170, 70)
COLOR_RED = (208, 60, 60)
COLOR_ORANGE = (245, 150, 20)
COLOR_DARK = (30, 30, 30)

STROKE = 16
PORT_R = 4
TERM_R = 6
MARKER_R = 7
X_LEFT = 70.0
X_RIGHT = 410.0
Y_TOP = 110.0
LANE_STEP = 34.0
FOLDS = 4
FRACTIONS = [0.30, 0.55, 0.78]

MARKER_COLORS = ["blue", "green", "red"]
FILL = {
    "blue": COLOR_BLUE,
    "green": COLOR_GREEN,
    "red": COLOR_RED,
}


def _serpentine(
    x0: float, x1: float, y0: float, n_bends: int, dy: float
) -> list[list[float]]:
    pts: list[list[float]] = [[x0, y0]]
    y = y0
    x = x0
    for i in range(n_bends):
        x = x1 if i % 2 == 0 else x0
        pts.append([x, y])
        y = y + dy
        pts.append([x, y])
    x = x1 if n_bends % 2 == 0 else x0
    pts.append([x, y])
    return pts


def _arc_lengths(poly: list) -> list[float]:
    out = [0.0]
    for a, b in zip(poly, poly[1:]):
        out.append(out[-1] + math.hypot(b[0] - a[0], b[1] - a[1]))
    return out


def _point_at_frac(poly: list, frac: float) -> list[float]:
    arcs = _arc_lengths(poly)
    total = arcs[-1]
    target = frac * total
    for i in range(1, len(arcs)):
        if arcs[i] >= target:
            seg = arcs[i] - arcs[i - 1]
            t = (target - arcs[i - 1]) / seg if seg > 0 else 0.0
            a = poly[i - 1]
            b = poly[i]
            return [a[0] + t * (b[0] - a[0]), a[1] + t * (b[1] - a[1])]
    return list(poly[-1])


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    poly = _serpentine(X_LEFT, X_RIGHT, Y_TOP, FOLDS, LANE_STEP)
    positions = [_point_at_frac(poly, f) for f in FRACTIONS]
    colors = list(MARKER_COLORS)
    rng.shuffle(colors)
    markers = [
        {"color": colors[i], "pos": positions[i]} for i in range(len(positions))
    ]
    route = {
        "color": "slate",
        "polyline": poly,
        "ring": list(poly[0]),
        "terminal": list(poly[-1]),
    }
    return {"route": route, "markers": markers}


def _draw_route(draw: ImageDraw.ImageDraw, poly, color, width):
    for a, b in zip(poly, poly[1:]):
        draw.line([tuple(a), tuple(b)], fill=color, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    route = scene["route"]
    poly = route["polyline"]
    _draw_route(draw, poly, COLOR_SLATE, STROKE)
    for marker in scene["markers"]:
        cx, cy = marker["pos"]
        r = MARKER_R
        draw.ellipse(
            [cx - r, cy - r, cx + r, cy + r],
            fill=FILL[marker["color"]],
        )
    sx, sy = route["ring"]
    draw.ellipse(
        [sx - PORT_R, sy - PORT_R, sx + PORT_R, sy + PORT_R],
        fill=COLOR_ORANGE,
    )
    tx, ty = route["terminal"]
    draw.ellipse(
        [tx - TERM_R, ty - TERM_R, tx + TERM_R, ty + TERM_R], fill=COLOR_DARK
    )
    return img


def _marker_arclen(scene: dict, idx: int) -> float:
    arcs = _arc_lengths(scene["route"]["polyline"])
    return FRACTIONS[idx] * arcs[-1]


def _answer_index(scene: dict) -> int:
    vals = [(_marker_arclen(scene, i), i) for i in range(len(scene["markers"]))]
    vals.sort(reverse=True)
    return vals[0][1]


def analytic_gold(scene: dict) -> str:
    return scene["markers"][_answer_index(scene)]["color"]


def margin(scene: dict) -> float:
    vals = sorted(_marker_arclen(scene, i) for i in range(len(scene["markers"])))
    return float(vals[-1] - vals[-2])


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < 40.0:
        return True
    route = scene["route"]
    poly = route["polyline"]
    for x, y in poly:
        if not (10 <= x <= w - 10 and 10 <= y <= h - 10):
            return True
    ring = route["ring"]
    terminal = route["terminal"]
    positions = [marker["pos"] for marker in scene["markers"]]
    for i in range(len(positions)):
        for j in range(i + 1, len(positions)):
            if math.dist(positions[i], positions[j]) < 30.0:
                return True
        if math.dist(positions[i], ring) < 24.0:
            return True
        if math.dist(positions[i], terminal) < 24.0:
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
