"""Renderer for the midturn-stance-recovery world.

A single rotation of the plane carries the green figure onto the orange figure.
Exactly one of four congruent blue figures is the halfway stance of that motion:
one and the same rotation carries green onto it and carries it onto orange.
The three decoys are the classic wrong mechanisms - halving the straight-line
displacement, halving the angle about the wrong centre, and swinging the wrong
way round the right centre - and every blue figure is drawn in the same
orientation, so orientation carries no information and only the recovered centre
of rotation discriminates.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 1600
HEIGHT = 1300

BACKGROUND = (247, 246, 241)
GREEN = (36, 150, 74)
ORANGE = (232, 124, 26)
BLUE = (66, 84, 196)

_RAW = (
    (112.0, 0.0),
    (16.0, 58.0),
    (-46.0, 74.0),
    (-84.0, 10.0),
    (-58.0, -62.0),
    (18.0, -52.0),
)
_SCALE = 0.52

CLEARANCE_FLOOR = 34.0
DX_FLOOR = 62.0
QUARANTINE_MARGIN = 40.0


def _poly_centroid(points) -> tuple[float, float]:
    twice_area = 0.0
    cx = 0.0
    cy = 0.0
    count = len(points)
    for index in range(count):
        x0, y0 = points[index]
        x1, y1 = points[(index + 1) % count]
        cross = x0 * y1 - x1 * y0
        twice_area += cross
        cx += (x0 + x1) * cross
        cy += (y0 + y1) * cross
    if abs(twice_area) < 1e-9:
        raise ValueError("degenerate polygon")
    return (cx / (3.0 * twice_area), cy / (3.0 * twice_area))


def _centred_shape() -> tuple[tuple[float, float], ...]:
    scaled = [(x * _SCALE, y * _SCALE) for x, y in _RAW]
    cx, cy = _poly_centroid(scaled)
    return tuple((x - cx, y - cy) for x, y in scaled)


LOCAL = _centred_shape()
RMAX = max(math.hypot(x, y) for x, y in LOCAL)


def _rotate(point, degrees: float) -> tuple[float, float]:
    radians = math.radians(degrees)
    cos = math.cos(radians)
    sin = math.sin(radians)
    return (point[0] * cos - point[1] * sin, point[0] * sin + point[1] * cos)


def _rotate_about(point, centre, degrees: float) -> tuple[float, float]:
    shifted = _rotate((point[0] - centre[0], point[1] - centre[1]), degrees)
    return (centre[0] + shifted[0], centre[1] + shifted[1])


def _pose_polygon(ref, rotation: float) -> list[list[int]]:
    points = []
    for local in LOCAL:
        dx, dy = _rotate(local, rotation)
        points.append([int(round(ref[0] + dx)), int(round(ref[1] + dy))])
    return points


def _far_angle(points, centre) -> float:
    best = None
    best_distance = -1.0
    for x, y in points:
        distance = math.hypot(x - centre[0], y - centre[1])
        if distance > best_distance:
            best_distance = distance
            best = (x, y)
    return math.degrees(math.atan2(best[1] - centre[1], best[0] - centre[0]))


def _wrap(degrees: float) -> float:
    return (degrees + 180.0) % 360.0 - 180.0


def _rotation_centre(source, image, degrees: float) -> tuple[float, float]:
    """The centre of the unique rotation by ``degrees`` taking source to image."""
    radians = math.radians(degrees)
    cos = math.cos(radians)
    sin = math.sin(radians)
    # (I - R) c = image - R source
    a = 1.0 - cos
    b = sin
    rx = image[0] - (source[0] * cos - source[1] * sin)
    ry = image[1] - (source[0] * sin + source[1] * cos)
    determinant = a * a + b * b
    if determinant < 1e-9:
        raise ValueError("rotation angle too close to zero")
    # inverse of [[a, b], [-b, a]]
    return ((a * rx - b * ry) / determinant, (b * rx + a * ry) / determinant)


def _alpha_band(rng: random.Random) -> float:
    if rng.random() < 0.5:
        magnitude = rng.uniform(62.0, 76.0)
    else:
        magnitude = rng.uniform(104.0, 142.0)
    return magnitude if rng.random() < 0.5 else -magnitude


def sample_scene(seed: int) -> dict:
    wanted = int(seed) % 4
    for attempt in range(30000):
        rng = random.Random((int(seed) * 7919 + attempt * 104729) & 0xFFFFFFFF)
        alpha = _alpha_band(rng)
        radius = rng.uniform(200.0, 380.0)
        phi = rng.uniform(0.0, 360.0)
        psi = rng.uniform(0.0, 360.0)
        centre = (0.0, 0.0)

        def on_circle(angle: float) -> tuple[float, float]:
            radians = math.radians(angle)
            return (
                centre[0] + radius * math.cos(radians),
                centre[1] + radius * math.sin(radians),
            )

        green_ref = on_circle(phi)
        orange_ref = on_circle(phi + 2.0 * alpha)
        truth_ref = on_circle(phi + alpha)
        other_root = (2.0 * centre[0] - truth_ref[0], 2.0 * centre[1] - truth_ref[1])
        mid = (
            0.5 * (green_ref[0] + orange_ref[0]),
            0.5 * (green_ref[1] + orange_ref[1]),
        )
        decoy_hinge = _rotate_about(green_ref, mid, alpha)
        decoy_chord = mid
        decoy_reverse = on_circle(phi - alpha)

        blue_refs = [truth_ref, decoy_hinge, decoy_chord, decoy_reverse]
        every = [green_ref, orange_ref] + blue_refs

        # Re-centre the whole rigid configuration inside the canvas; a rigid
        # translation changes no relation the answer depends on.
        limit_low = RMAX + 18.0
        span_x = max(p[0] for p in every) - min(p[0] for p in every)
        span_y = max(p[1] for p in every) - min(p[1] for p in every)
        if span_x > WIDTH - 2.0 * limit_low or span_y > HEIGHT - 2.0 * limit_low:
            continue
        shift = (
            limit_low + rng.uniform(0.0, WIDTH - 2.0 * limit_low - span_x)
            - min(p[0] for p in every),
            limit_low + rng.uniform(0.0, HEIGHT - 2.0 * limit_low - span_y)
            - min(p[1] for p in every),
        )
        green_ref = (green_ref[0] + shift[0], green_ref[1] + shift[1])
        orange_ref = (orange_ref[0] + shift[0], orange_ref[1] + shift[1])
        other_root = (other_root[0] + shift[0], other_root[1] + shift[1])
        blue_refs = [(p[0] + shift[0], p[1] + shift[1]) for p in blue_refs]
        every = [green_ref, orange_ref] + blue_refs

        clearance = min(
            math.hypot(every[i][0] - every[j][0], every[i][1] - every[j][1])
            for i in range(len(every))
            for j in range(i + 1, len(every))
        ) - 2.0 * RMAX
        if clearance < CLEARANCE_FLOOR:
            continue

        if any(
            math.hypot(other_root[0] - ref[0], other_root[1] - ref[1]) < 2.0 * RMAX + 34.0
            for ref in blue_refs
        ):
            continue

        min_dx = min(
            abs(blue_refs[i][0] - blue_refs[j][0])
            for i in range(4)
            for j in range(i + 1, 4)
        )
        if min_dx < DX_FLOOR:
            continue

        # Every decoy must fail the two-step story: the rotation by alpha that
        # carries green onto it and the one carrying it onto orange differ.
        ok = True
        for ref in blue_refs[1:]:
            first = _rotation_centre(green_ref, ref, alpha)
            second = _rotation_centre(ref, orange_ref, alpha)
            if math.hypot(first[0] - second[0], first[1] - second[1]) < 45.0:
                ok = False
                break
        if not ok:
            continue

        order = [0, 1, 2, 3]
        rng.shuffle(order)
        blues = [_pose_polygon(blue_refs[index], psi + alpha) for index in order]
        scene = {
            "canvas": [WIDTH, HEIGHT],
            "green": _pose_polygon(green_ref, psi),
            "orange": _pose_polygon(orange_ref, psi + 2.0 * alpha),
            "blues": blues,
        }
        try:
            decision = analytic_gold(scene)
        except ValueError:
            continue
        rank = int(decision.split("-")[1]) - 1
        truth_slot = order.index(0)
        expected = sorted(range(4), key=lambda k: _poly_centroid(blues[k])[0]).index(truth_slot)
        if rank != expected:
            continue
        if rank != wanted:
            continue
        if margin(scene) < min(CLEARANCE_FLOOR, DX_FLOOR) - 1e-9:
            continue
        return scene
    raise RuntimeError(f"failed to sample a scene for seed {seed}")


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for polygon in scene["blues"]:
        draw.polygon([tuple(point) for point in polygon], fill=BLUE)
    draw.polygon([tuple(point) for point in scene["green"]], fill=GREEN)
    draw.polygon([tuple(point) for point in scene["orange"]], fill=ORANGE)
    return image


def _scene_geometry(scene: dict):
    green = [(float(x), float(y)) for x, y in scene["green"]]
    orange = [(float(x), float(y)) for x, y in scene["orange"]]
    blues = [[(float(x), float(y)) for x, y in polygon] for polygon in scene["blues"]]
    green_centre = _poly_centroid(green)
    orange_centre = _poly_centroid(orange)
    blue_centres = [_poly_centroid(polygon) for polygon in blues]
    green_angle = _far_angle(green, green_centre)
    orange_angle = _far_angle(orange, orange_centre)
    blue_angle = _far_angle(blues[0], blue_centres[0])
    return green_centre, orange_centre, blue_centres, green_angle, orange_angle, blue_angle


def _solve_halfway(green_centre, orange_centre, green_angle, orange_angle, blue_angle):
    theta = _wrap(orange_angle - green_angle)
    if abs(theta) < 80.0:
        raise ValueError("demonstrated turn is too small to halve reliably")
    hinge = _rotation_centre(green_centre, orange_centre, theta)
    best = None
    for half in (theta / 2.0, theta / 2.0 + 180.0):
        error = abs(_wrap(green_angle + half - blue_angle))
        if best is None or error < best[0]:
            best = (error, half)
    if best[0] > 25.0:
        raise ValueError("no halved turn matches the drawn stance orientation")
    return hinge, best[1]


def analytic_gold(scene: dict) -> str:
    geometry = _scene_geometry(scene)
    green_centre, orange_centre, blue_centres, green_angle, orange_angle, blue_angle = geometry
    hinge, half = _solve_halfway(
        green_centre, orange_centre, green_angle, orange_angle, blue_angle
    )
    predicted = _rotate_about(green_centre, hinge, half)
    order = sorted(range(4), key=lambda k: blue_centres[k][0])
    distances = [
        math.hypot(blue_centres[k][0] - predicted[0], blue_centres[k][1] - predicted[1])
        for k in order
    ]
    best = min(range(4), key=lambda k: distances[k])
    return f"blue-{best + 1}"


def margin(scene: dict) -> float:
    green = [(float(x), float(y)) for x, y in scene["green"]]
    orange = [(float(x), float(y)) for x, y in scene["orange"]]
    blues = [[(float(x), float(y)) for x, y in polygon] for polygon in scene["blues"]]
    polygons = [green, orange] + blues
    centres = [_poly_centroid(polygon) for polygon in polygons]
    reach = max(
        math.hypot(x - centre[0], y - centre[1])
        for polygon, centre in zip(polygons, centres)
        for x, y in polygon
    )
    closest = min(
        math.hypot(centres[i][0] - centres[j][0], centres[i][1] - centres[j][1])
        for i in range(len(centres))
        for j in range(i + 1, len(centres))
    )
    blue_centres = centres[2:]
    min_dx = min(
        abs(blue_centres[i][0] - blue_centres[j][0])
        for i in range(4)
        for j in range(i + 1, 4)
    )
    return float(min(closest - 2.0 * reach, min_dx))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent rewrites that must preserve the gold answer."""
    transforms: list[tuple[str, dict]] = []

    reversed_blues = dict(scene)
    reversed_blues["blues"] = list(reversed(list(scene["blues"])))
    transforms.append(("reverse_blue_serialization", reversed_blues))

    cycled = dict(scene)
    cycled["green"] = list(scene["green"][2:]) + list(scene["green"][:2])
    cycled["orange"] = list(scene["orange"][3:]) + list(scene["orange"][:3])
    cycled["blues"] = [list(poly[1:]) + list(poly[:1]) for poly in scene["blues"]]
    transforms.append(("cycle_vertex_serialization", cycled))
    return transforms
