"""Deterministic renderer and analytic arm for disjoint ordinal-slot search."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (249, 248, 244)
FRAME = (205, 207, 203)
ROW_GUIDE = (232, 232, 226)
COLORS = {
    "red": (210, 55, 65),
    "blue": (48, 105, 202),
    "green": (35, 151, 91),
    "purple": (139, 72, 186),
}
COLOR_NAMES = tuple(COLORS)
BASE_POSITIONS = ((128, 128), (320, 128), (128, 320), (320, 320))
SLOT_COUNT = 7
OCCUPIED_PER_ROW = 3
DIAL_RADIUS = 12
DIAL_SHIFT = 32


def _make_pair(rng: random.Random, overlap: int) -> list[list[int]]:
    """Return two three-of-seven sets with the requested intersection size."""
    left = set(rng.sample(range(SLOT_COUNT), OCCUPIED_PER_ROW))
    shared = set(rng.sample(sorted(left), overlap))
    outside = [value for value in range(SLOT_COUNT) if value not in left]
    right = shared | set(rng.sample(outside, OCCUPIED_PER_ROW - overlap))
    rows = [sorted(left), sorted(right)]
    if rng.randrange(2):
        rows.reverse()
    return rows


def sample_scene(seed: int) -> dict:
    """Sample four equal-ink colored pairs with overlap counts 0, 1, 2, 3."""
    seed = int(seed)
    rng = random.Random(seed)
    target = COLOR_NAMES[seed % len(COLOR_NAMES)]
    nonzero = [1, 2, 3]
    rng.shuffle(nonzero)
    overlap_for: dict[str, int] = {target: 0}
    for color, overlap in zip((c for c in COLOR_NAMES if c != target), nonzero, strict=True):
        overlap_for[color] = overlap

    records = [
        {"color": color, "rows": _make_pair(rng, overlap_for[color])}
        for color in COLOR_NAMES
    ]
    shuffled_positions = list(BASE_POSITIONS)
    rng.shuffle(shuffled_positions)
    positions = {
        color: list(position)
        for color, position in zip(COLOR_NAMES, shuffled_positions, strict=True)
    }
    return {
        "slot_pairs": records,
        "positions": positions,
        "offset": list(rng.choice(((0, 0), (64, 0), (0, 64), (64, 64)))),
    }


def _intersection_counts(scene: dict) -> dict[str, int]:
    counts: dict[str, int] = {}
    for record in scene["slot_pairs"]:
        left, right = (set(row) for row in record["rows"])
        counts[record["color"]] = len(left & right)
    return counts


def analytic_gold(scene: dict) -> str:
    counts = _intersection_counts(scene)
    winners = [color for color, value in counts.items() if value == 0]
    if len(winners) != 1:
        raise ValueError(f"expected one empty intersection, found {len(winners)}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = sorted(_intersection_counts(scene).values())
    if len(counts) != 4 or counts[0] != 0:
        return 0.0
    return float(counts[1] - counts[0])


def is_quarantined(scene: dict) -> bool:
    records = scene.get("slot_pairs", [])
    if len(records) != 4:
        return True
    colors = [record.get("color") for record in records]
    if set(colors) != set(COLOR_NAMES) or len(colors) != len(set(colors)):
        return True
    for record in records:
        rows = record.get("rows", [])
        if len(rows) != 2:
            return True
        if any(len(row) != OCCUPIED_PER_ROW or len(set(row)) != OCCUPIED_PER_ROW for row in rows):
            return True
        if any(any(not 0 <= int(value) < SLOT_COUNT for value in row) for row in rows):
            return True
    values = sorted(_intersection_counts(scene).values())
    return values != [0, 1, 2, 3]


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Record order is a genuine draw-order alias because badges never overlap."""
    twin = {
        "slot_pairs": list(reversed(scene["slot_pairs"])),
        "positions": {key: list(value) for key, value in scene["positions"].items()},
        "offset": list(scene["offset"]),
    }
    return [("reverse_record_order", twin)]


def _draw_dial(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    occupied: set[int],
    color: tuple[int, int, int],
) -> None:
    cx, cy = center
    draw.ellipse(
        (cx - DIAL_RADIUS - 4, cy - DIAL_RADIUS - 4,
         cx + DIAL_RADIUS + 4, cy + DIAL_RADIUS + 4),
        outline=ROW_GUIDE,
        width=2,
    )
    for index in range(SLOT_COUNT):
        angle = -math.pi / 2.0 + index * (2.0 * math.pi / SLOT_COUNT)
        x = int(round(cx + DIAL_RADIUS * math.cos(angle)))
        y = int(round(cy + DIAL_RADIUS * math.sin(angle)))
        if index in occupied:
            draw.rectangle((x - 3, y - 3, x + 3, y + 3), fill=color)
        else:
            draw.rectangle((x - 1, y - 1, x + 1, y + 1), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    offset_x, offset_y = (int(value) for value in scene["offset"])
    for record in scene["slot_pairs"]:
        color_name = record["color"]
        rgb = COLORS[color_name]
        base_x, base_y = scene["positions"][color_name]
        cx, cy = int(base_x) + offset_x, int(base_y) + offset_y
        draw.rounded_rectangle(
            (cx - 58, cy - 58, cx + 58, cy + 58),
            radius=14,
            outline=FRAME,
            width=2,
        )
        rows = record["rows"]
        _draw_dial(draw, (cx - DIAL_SHIFT, cy - DIAL_SHIFT), set(rows[0]), rgb)
        _draw_dial(draw, (cx + DIAL_SHIFT, cy + DIAL_SHIFT), set(rows[1]), rgb)
    return image
