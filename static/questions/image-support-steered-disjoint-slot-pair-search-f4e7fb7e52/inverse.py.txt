"""Independent pixel-only inverse for disjoint ordinal-slot pair search.

The inverse receives only a final raster. It segments the four public palette
colors, reconstructs complete seven-position rows from connected components,
classifies large squares versus small dots by visible area, and searches the
decoded pairs for the unique empty occupied-set intersection.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((210, 55, 65), dtype=np.int16),
    "blue": np.array((48, 105, 202), dtype=np.int16),
    "green": np.array((35, 151, 91), dtype=np.int16),
    "purple": np.array((139, 72, 186), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 8:
            output.append(np.asarray(points, dtype=np.int32))
    return output


def _decode_complete_pair(components: list[np.ndarray]) -> tuple[set[int], set[int]] | None:
    """Decode one color, returning None for a damaged/ablated distractor."""
    if len(components) != 14:
        return None
    marks = []
    for points in components:
        ys, xs = points[:, 0], points[:, 1]
        area = int(len(points))
        width = int(xs.max() - xs.min() + 1)
        height = int(ys.max() - ys.min() + 1)
        if area == 49 and width == 7 and height == 7:
            occupied = True
        elif area == 9 and width == 3 and height == 3:
            occupied = False
        else:
            return None
        marks.append((float(xs.mean()), float(ys.mean()), occupied))

    marks.sort(key=lambda item: item[1])
    dials = [marks[:7], marks[7:]]
    if dials[1][0][1] - dials[0][-1][1] < 25.0:
        return None
    decoded: list[set[int]] = []
    for dial in dials:
        center_x = sum(item[0] for item in dial) / 7.0
        center_y = sum(item[1] for item in dial) / 7.0
        by_slot: dict[int, bool] = {}
        for x, y, occupied_value in dial:
            clockwise = math.atan2(x - center_x, center_y - y) % (2.0 * math.pi)
            slot = int(round(clockwise * 7.0 / (2.0 * math.pi))) % 7
            if slot in by_slot:
                return None
            by_slot[slot] = occupied_value
        if set(by_slot) != set(range(7)):
            return None
        occupied = {slot for slot, value in by_slot.items() if value}
        if len(occupied) != 3:
            return None
        decoded.append(occupied)
    return decoded[0], decoded[1]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8).astype(np.int16)
    decoded: dict[str, tuple[set[int], set[int]]] = {}
    for name, rgb in PALETTE.items():
        distance = np.linalg.norm(array - rgb, axis=2)
        pair = _decode_complete_pair(_components(distance < 12.0))
        if pair is not None:
            decoded[name] = pair
    if len(decoded) < 2:
        raise ValueError("fewer than two complete color candidates")
    scores = {name: len(left & right) for name, (left, right) in decoded.items()}
    winners = [name for name, score in scores.items() if score == 0]
    runners = [score for score in scores.values() if score > 0]
    if len(winners) != 1 or not runners or min(runners) < 1:
        raise ValueError("no unique zero-intersection winner with a runner-up margin")
    return winners[0]
