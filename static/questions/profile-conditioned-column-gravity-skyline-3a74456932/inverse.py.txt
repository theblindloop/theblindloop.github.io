"""Independent PNG-only inverse arm for the gravity-packing task."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 9

# Public answer shapes are independently encoded here. No renderer state,
# latent scene, filename, metadata, or analytic-gold implementation is read.
_PROFILES: dict[str, tuple[int, ...]] = {
    "rising stairs": (1, 2, 3, 4, 5, 6, 7, 8, 9),
    "falling stairs": (9, 8, 7, 6, 5, 4, 3, 2, 1),
    "peak": (1, 3, 5, 7, 9, 8, 6, 4, 2),
    "valley": (9, 7, 5, 3, 1, 2, 4, 6, 8),
}


def _frame_geometry(array: np.ndarray) -> tuple[int, int, int]:
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    cyan = (red < 70) & (green > 125) & (green < 195) & (blue > 145)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 500:
        raise ValueError("cyan frame missing")
    xmin, xmax = int(xs.min()), int(xs.max())
    ymin, ymax = int(ys.min()), int(ys.max())
    horizontal = cyan[(ymin + ymax) // 2, xmin : xmax + 1]
    vertical = cyan[ymin : ymax + 1, (xmin + xmax) // 2]
    left = int(np.argmax(~horizontal)) if np.any(~horizontal) else 0
    right = int(np.argmax(~horizontal[::-1])) if np.any(~horizontal) else 0
    top = int(np.argmax(~vertical)) if np.any(~vertical) else 0
    bottom = int(np.argmax(~vertical[::-1])) if np.any(~vertical) else 0
    if min(left, right, top, bottom) < 3:
        raise ValueError("cyan frame is incomplete")
    inner_w = xmax - xmin + 1 - left - right
    inner_h = ymax - ymin + 1 - top - bottom
    if inner_w != inner_h or inner_w % GRID_N:
        raise ValueError("cyan frame does not bound a square 9-by-9 field")
    cell = inner_w // GRID_N
    if not 36 <= cell <= 44:
        raise ValueError("implausible grid cell size")
    return xmin + left, ymin + top, cell


def _check_down_arrow(array: np.ndarray) -> None:
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    gold = (red > 180) & (green > 115) & (green < 200) & (blue < 90)
    ys, xs = np.nonzero(gold)
    if len(xs) < 200:
        raise ValueError("gold gravity arrow missing")
    width = int(xs.max() - xs.min() + 1)
    height = int(ys.max() - ys.min() + 1)
    if height < 2 * width:
        raise ValueError("gold gravity marker is not a vertical arrow")
    cut = int(ys.min() + 2 * height / 3)
    if int((gold & (np.indices(gold.shape)[0] >= cut)).sum()) <= int(
        (gold & (np.indices(gold.shape)[0] < int(ys.min() + height / 3))).sum()
    ):
        raise ValueError("gold arrow does not point downward")


def measure(image: Image.Image) -> dict:
    array = np.asarray(image.convert("RGB"), dtype=np.int32)
    x0, y0, cell = _frame_geometry(array)
    _check_down_arrow(array)
    luminance = (299 * array[..., 0] + 587 * array[..., 1] + 114 * array[..., 2]) // 1000
    occupied = np.zeros((GRID_N, GRID_N), dtype=bool)
    confidence = []
    for row in range(GRID_N):
        for col in range(GRID_N):
            patch = luminance[
                y0 + row * cell + 7 : y0 + (row + 1) * cell - 7,
                x0 + col * cell + 7 : x0 + (col + 1) * cell - 7,
            ]
            dark_fraction = float(np.mean(patch < 150))
            occupied[row, col] = dark_fraction > 0.5
            confidence.append(abs(dark_fraction - 0.5))
    if min(confidence) < 0.30:
        raise ValueError("tile occupancy is ambiguous")

    counts = [int(occupied[:, col].sum()) for col in range(GRID_N)]
    ranked = sorted(
        (sum(abs(a - b) for a, b in zip(counts, profile)), name)
        for name, profile in _PROFILES.items()
    )
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("skyline classification tie")
    gap_percent = 100.0 * (ranked[1][0] - ranked[0][0]) / (GRID_N * GRID_N)
    return {
        "decision": ranked[0][1],
        "template_margin": gap_percent,
        "column_counts": counts,
        "cell_px": cell,
        "occupied_tiles": int(occupied.sum()),
    }


def decision_from_image(image: Image.Image) -> str:
    return str(measure(image)["decision"])
