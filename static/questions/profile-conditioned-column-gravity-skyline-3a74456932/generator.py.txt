"""Deterministic renderer and analytic arm for a gravity-packing reveal."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (640, 560)
GRID_N = 9
FRAME = (24, 162, 185)
GOLD = (222, 158, 35)
BACKGROUND = (248, 247, 242)
CELL_BACKGROUND = (239, 242, 239)
GRID_LINE = (190, 202, 203)
DISTRACTOR = (173, 177, 174)
INK_PALETTES = (
    (25, 39, 66),
    (72, 34, 62),
    (23, 65, 53),
)

# Each class is a permutation of heights 1..9. Consequently every class has
# 45 tiles and the same multiset of column counts; answer identity is encoded
# only by the order in which those equal-histogram columns appear.
PROFILES: dict[str, tuple[int, ...]] = {
    "rising stairs": (1, 2, 3, 4, 5, 6, 7, 8, 9),
    "falling stairs": (9, 8, 7, 6, 5, 4, 3, 2, 1),
    "peak": (1, 3, 5, 7, 9, 8, 6, 4, 2),
    "valley": (9, 7, 5, 3, 1, 2, 4, 6, 8),
}
LABELS = tuple(PROFILES)


def _profile_distances(counts: list[int] | tuple[int, ...]) -> list[tuple[int, str]]:
    return sorted(
        (sum(abs(int(a) - int(b)) for a, b in zip(counts, profile)), name)
        for name, profile in PROFILES.items()
    )


def _classify_counts(counts: list[int] | tuple[int, ...]) -> tuple[str, float]:
    if len(counts) != GRID_N or any(not 0 <= int(value) <= GRID_N for value in counts):
        raise ValueError("column counts must contain nine values in [0, 9]")
    ranked = _profile_distances(counts)
    if ranked[0][0] == ranked[1][0]:
        raise ValueError("skyline template tie")
    gap_percent = 100.0 * (ranked[1][0] - ranked[0][0]) / (GRID_N * GRID_N)
    return ranked[0][1], gap_percent


def _counts_from_occupied(occupied: list[list[int]]) -> list[int]:
    if len(occupied) != GRID_N or any(len(row) != GRID_N for row in occupied):
        raise ValueError("occupied must be a 9-by-9 matrix")
    return [sum(int(bool(occupied[row][col])) for row in range(GRID_N)) for col in range(GRID_N)]


def _near_boundary_profile(label: str) -> list[int]:
    """Move paired tiles toward the closest rival while preserving total ink."""
    base = list(PROFILES[label])
    rivals = sorted(
        (sum(abs(a - b) for a, b in zip(base, profile)), name, profile)
        for name, profile in PROFILES.items()
        if name != label
    )
    rival = list(rivals[0][2])
    counts = base[:]
    best = counts[:]
    while True:
        donors = [i for i in range(GRID_N) if counts[i] > rival[i]]
        receivers = [i for i in range(GRID_N) if counts[i] < rival[i]]
        if not donors or not receivers:
            break
        trial = counts[:]
        trial[donors[-1]] -= 1
        trial[receivers[0]] += 1
        try:
            decision, gap = _classify_counts(trial)
        except ValueError:
            break
        if decision != label:
            break
        counts = trial
        best = counts[:]
        if gap < 12.0:
            return counts
    return best


def sample_scene(seed: int) -> dict:
    """Sample balanced quartets with answer-independent rendering nuisances."""
    label_index = int(seed) % len(LABELS)
    group = int(seed) // len(LABELS)
    label = LABELS[label_index]
    counts = _near_boundary_profile(label) if group % 6 == 0 else list(PROFILES[label])
    nuisance = random.Random(group ^ 0x6C39A7)
    scatter = random.Random((group << 9) ^ label_index ^ 0x31D4B5)
    occupied = [[0 for _ in range(GRID_N)] for _ in range(GRID_N)]
    for col, count in enumerate(counts):
        rows = list(range(GRID_N))
        scatter.shuffle(rows)
        for row in rows[:count]:
            occupied[row][col] = 1

    distractors = []
    for index in range(group % 4):
        distractors.append(
            [
                74 + index * 86 + nuisance.randrange(-8, 9),
                514 + nuisance.randrange(-7, 8),
                5 + nuisance.randrange(0, 4),
            ]
        )
    return {
        "occupied": occupied,
        "palette_index": group % len(INK_PALETTES),
        "origin_x": 66 + 4 * nuisance.randrange(0, 7),
        "origin_y": 54 + 4 * nuisance.randrange(0, 7),
        "cell_px": (38, 40, 42)[(group // 3) % 3],
        "distractors": distractors,
    }


def render(scene: dict) -> Image.Image:
    occupied = scene["occupied"]
    palette_index = int(scene["palette_index"])
    x0 = int(scene["origin_x"])
    y0 = int(scene["origin_y"])
    cell = int(scene["cell_px"])
    distractors = scene["distractors"]
    _counts_from_occupied(occupied)
    if palette_index not in range(len(INK_PALETTES)):
        raise ValueError("invalid palette_index")
    if cell not in (38, 40, 42):
        raise ValueError("invalid cell size")

    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    span = GRID_N * cell
    border = 5
    draw.rectangle(
        (x0 - border, y0 - border, x0 + span + border - 1, y0 + span + border - 1),
        fill=FRAME,
    )
    draw.rectangle((x0, y0, x0 + span - 1, y0 + span - 1), fill=CELL_BACKGROUND)
    ink = INK_PALETTES[palette_index]
    inset = 5
    for row in range(GRID_N):
        for col in range(GRID_N):
            if bool(occupied[row][col]):
                left = x0 + col * cell + inset
                top = y0 + row * cell + inset
                draw.rectangle(
                    (left, top, x0 + (col + 1) * cell - inset - 1,
                     y0 + (row + 1) * cell - inset - 1),
                    fill=ink,
                )
    for k in range(1, GRID_N):
        gx = x0 + k * cell
        gy = y0 + k * cell
        draw.line((gx, y0, gx, y0 + span - 1), fill=GRID_LINE, width=1)
        draw.line((x0, gy, x0 + span - 1, gy), fill=GRID_LINE, width=1)

    # A visible gold arrow fixes the declared direction of gravity.
    arrow_x = x0 + span + 58
    arrow_top = y0 + span // 2 - 56
    arrow_base = arrow_top + 78
    draw.line((arrow_x, arrow_top, arrow_x, arrow_base), fill=GOLD, width=7)
    draw.polygon(
        [(arrow_x - 17, arrow_base - 7), (arrow_x + 17, arrow_base - 7),
         (arrow_x, arrow_base + 20)],
        fill=GOLD,
    )
    for x, y, radius in distractors:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius),
                     outline=DISTRACTOR, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    return _classify_counts(_counts_from_occupied(scene["occupied"]))[0]


def margin(scene: dict) -> float:
    return _classify_counts(_counts_from_occupied(scene["occupied"]))[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
