"""Renderer and analytic arm for paired checkpoint-order agreement."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 384
BACKGROUND = [248, 247, 242]
NAVY = (31, 55, 82)
GREEN = (34, 160, 92)
RED = (218, 67, 64)
BLUE = (45, 119, 210)
ORANGE = (238, 145, 42)

# Self-avoiding folds. A square symmetry is applied separately to each route.
TEMPLATES = (
    ((0.08, 0.12), (0.77, 0.12), (0.77, 0.34), (0.24, 0.34),
     (0.24, 0.58), (0.84, 0.58), (0.84, 0.84), (0.10, 0.84)),
    ((0.12, 0.10), (0.12, 0.76), (0.36, 0.76), (0.36, 0.24),
     (0.61, 0.24), (0.61, 0.84), (0.86, 0.84), (0.86, 0.13)),
    ((0.09, 0.16), (0.72, 0.16), (0.72, 0.43), (0.19, 0.43),
     (0.19, 0.76), (0.48, 0.76), (0.48, 0.57), (0.87, 0.57), (0.87, 0.86)),
)


def _transform(points, rotation: int, mirror: bool):
    output = []
    for x, y in points:
        if mirror:
            x = 1.0 - x
        for _ in range(rotation % 4):
            x, y = 1.0 - y, x
        output.append((x, y))
    return output


def _place(points, x0: float, y0: float, size: float, jitter_x: float, jitter_y: float):
    return [[round(x0 + x * size + jitter_x, 3), round(y0 + y * size + jitter_y, 3)]
            for x, y in points]


def _point_at(points, fraction: float):
    lengths = [math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:])]
    target = fraction * sum(lengths)
    walked = 0.0
    for a, b, length in zip(points, points[1:], lengths):
        if target <= walked + length:
            ratio = (target - walked) / length
            return (a[0] + ratio * (b[0] - a[0]), a[1] + ratio * (b[1] - a[1]))
        walked += length
    return tuple(points[-1])


def _route_length(points):
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    same = int(seed) % 2 == 0
    first_blue_first = bool(rng.getrandbits(1))
    orders = (first_blue_first, first_blue_first if same else not first_blue_first)
    routes = []
    for index, x0 in enumerate((25.0, 345.0)):
        template = TEMPLATES[rng.randrange(len(TEMPLATES))]
        transformed = _transform(template, rng.randrange(4), bool(rng.getrandbits(1)))
        points = _place(
            transformed, x0, 42.0, 270.0,
            rng.uniform(-4.0, 4.0), rng.uniform(-4.0, 4.0),
        )
        start_at = rng.randrange(2)  # 0 = first point, 1 = last point
        low = rng.uniform(0.25, 0.34)
        high = rng.uniform(0.66, 0.75)
        blue_first_from_start = orders[index]
        # Fractions are measured from points[0]. Convert the desired directed order.
        blue_low = blue_first_from_start if start_at == 0 else not blue_first_from_start
        blue_t, orange_t = (low, high) if blue_low else (high, low)
        routes.append({
            "points": points,
            "start_at": start_at,
            "blue_t": round(blue_t, 6),
            "orange_t": round(orange_t, 6),
            "stroke": rng.choice([10, 11, 12]),
        })
    return {
        "routes": routes,
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND,
    }


def _directed_blue_first(route: dict) -> bool:
    blue_t = float(route["blue_t"])
    orange_t = float(route["orange_t"])
    return blue_t < orange_t if int(route["start_at"]) == 0 else blue_t > orange_t


def analytic_gold(scene: dict) -> str:
    routes = scene["routes"]
    first_blue = float(routes[0]["blue_t"])
    first_orange = float(routes[0]["orange_t"])
    first_order = first_blue < first_orange if int(routes[0]["start_at"]) == 0 else first_blue > first_orange
    second_blue = float(routes[1]["blue_t"])
    second_orange = float(routes[1]["orange_t"])
    second_order = second_blue < second_orange if int(routes[1]["start_at"]) == 0 else second_blue > second_orange
    return "same" if first_order == second_order else "reversed"


def margin(scene: dict) -> float:
    routes = scene["routes"]
    return round(min(abs(float(route["blue_t"]) - float(route["orange_t"])) * _route_length(route["points"])
                     for route in routes), 6)


def is_quarantined(scene: dict) -> bool:
    routes = scene["routes"]
    return margin(scene) < 120.0 or any(
        min(float(route["blue_t"]), float(route["orange_t"])) < 0.18
        or max(float(route["blue_t"]), float(route["orange_t"])) > 0.82
        for route in routes
    )


def _disk(draw, center, radius, fill):
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)


def _ring(draw, center, radius, color, width):
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=color, width=width)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    background = tuple(scene["background"])
    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=12 * scale)
    for route in scene["routes"]:
        points = [(round(x * scale), round(y * scale)) for x, y in route["points"]]
        stroke = int(route["stroke"]) * scale
        draw.line(points, fill=NAVY, width=stroke, joint="curve")
        # Rounded navy joins improve graph connectivity after downsampling.
        for point in points[1:-1]:
            _disk(draw, point, stroke / 2.0, NAVY)
        blue = tuple(v * scale for v in _point_at(route["points"], float(route["blue_t"])))
        orange = tuple(v * scale for v in _point_at(route["points"], float(route["orange_t"])))
        for center, color in ((blue, BLUE), (orange, ORANGE)):
            _ring(draw, center, 15 * scale, color, 6 * scale)
        first = tuple(v * scale for v in route["points"][0])
        last = tuple(v * scale for v in route["points"][-1])
        start, finish = (first, last) if int(route["start_at"]) == 0 else (last, first)
        _ring(draw, start, 13 * scale, GREEN, 6 * scale)
        _ring(draw, finish, 13 * scale, RED, 6 * scale)
        # Restore an uninterrupted centerline through every marker ring. The
        # inverse arm groups the two visible arcs of each ring by route island.
        draw.line(points, fill=NAVY, width=stroke, joint="curve")
        for point in points[1:-1]:
            _disk(draw, point, stroke / 2.0, NAVY)
        sx, sy = start
        fx, fy = finish
        draw.text((sx + 14 * scale, sy - 6 * scale), "START", fill=(42, 92, 65), font=font)
        draw.text((fx + 14 * scale, fy - 6 * scale), "FINISH", fill=(128, 55, 53), font=font)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
