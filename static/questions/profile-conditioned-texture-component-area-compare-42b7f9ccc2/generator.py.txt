"""Deterministic renderer for the slash-texture component-area world."""

from __future__ import annotations

import random
from collections import deque
from copy import deepcopy

from PIL import Image, ImageDraw

SIZE = 512
N = 12
OFFSET = 58
STEP = 36

# A connected, non-self-touching prefix path.  For every used prefix length,
# the first cell has exactly one occupied edge-neighbor and the same 3x3 patch.
PATH = [
    (2, 2), (2, 1), (2, 0), (1, 0), (0, 0),
    (0, 1), (0, 2), (0, 3), (0, 4), (1, 4),
    (2, 4), (3, 4), (4, 4), (4, 3), (4, 2),
]
ZONES = [(0, 0), (7, 7), (0, 7), (7, 0)]
RED_PALETTES = [(214, 59, 52), (194, 45, 92), (222, 89, 32)]
BLUE_PALETTES = [(38, 112, 201), (38, 139, 160), (83, 76, 180)]
INKS = [(22, 25, 29), (37, 39, 43), (49, 43, 38)]


def _variant(cell: tuple[int, int], code: int) -> tuple[int, int]:
    r, c = cell
    if code >= 4:
        c = 4 - c
    for _ in range(code % 4):
        r, c = c, 4 - r
    return r, c


def _global(cell: tuple[int, int], code: int) -> tuple[int, int]:
    r, c = cell
    if code >= 4:
        c = N - 1 - c
    for _ in range(code % 4):
        r, c = c, N - 1 - r
    return r, c


def _component_cells(zone: tuple[int, int], count: int, code: int) -> list[tuple[int, int]]:
    r0, c0 = zone
    return [(r0 + r, c0 + c) for r, c in (_variant(cell, code) for cell in PATH[:count])]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Sequential seeds alternate the answer color."""
    rng = random.Random(int(seed))
    large = 11 + ((int(seed) // 2) % 3)
    small = 20 - large
    codes = [rng.randrange(8) for _ in range(4)]
    parts = [
        _component_cells(ZONES[0], large, codes[0]),
        _component_cells(ZONES[1], small, codes[1]),
        _component_cells(ZONES[2], 10, codes[2]),
        _component_cells(ZONES[3], 10, codes[3]),
    ]
    red_large = int(seed) % 2 == 0
    red_cell = parts[0][0] if red_large else parts[1][0]
    blue_cell = parts[1][0] if red_large else parts[0][0]

    grid = [[0 for _ in range(N)] for _ in range(N)]
    for part in parts:
        for r, c in part:
            grid[r][c] = 1

    # Global dihedral variation destroys fixed absolute zone positions.
    global_code = rng.randrange(8)
    transformed = [[0 for _ in range(N)] for _ in range(N)]
    for r in range(N):
        for c in range(N):
            rr, cc = _global((r, c), global_code)
            transformed[rr][cc] = grid[r][c]
    red_cell = _global(red_cell, global_code)
    blue_cell = _global(blue_cell, global_code)

    cell_order = [[r, c] for r in range(N) for c in range(N)]
    rng.shuffle(cell_order)
    return {
        "grid": transformed,
        "red_cell": list(red_cell),
        "blue_cell": list(blue_cell),
        "cell_order": cell_order,
        "palette_index": rng.randrange(3),
        "ink_index": rng.randrange(3),
        "seed": int(seed),
    }


def _center(cell: list[int] | tuple[int, int]) -> tuple[int, int]:
    r, c = int(cell[0]), int(cell[1])
    return OFFSET + c * STEP, OFFSET + r * STEP


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    grid = scene["grid"]
    ink = INKS[int(scene.get("ink_index", 0)) % len(INKS)]
    order = scene.get("cell_order", [[r, c] for r in range(N) for c in range(N)])
    for r, c in order:
        x, y = _center((r, c))
        if int(grid[r][c]) == 1:  # up-right /
            endpoints = (x - 9, y + 9, x + 9, y - 9)
        else:  # down-right backslash
            endpoints = (x - 9, y - 9, x + 9, y + 9)
        draw.line(endpoints, fill=ink, width=4)

    palette = int(scene.get("palette_index", 0)) % len(RED_PALETTES)
    for cell, color in (
        (scene["red_cell"], RED_PALETTES[palette]),
        (scene["blue_cell"], BLUE_PALETTES[palette]),
    ):
        x, y = _center(cell)
        draw.ellipse((x - 16, y - 16, x + 16, y + 16), outline=color, width=4)
    return image


def _component_size(scene: dict, start: list[int]) -> int:
    grid = scene["grid"]
    origin = (int(start[0]), int(start[1]))
    if int(grid[origin[0]][origin[1]]) != 1:
        return 0
    seen = {origin}
    queue = deque([origin])
    while queue:
        r, c = queue.popleft()
        for rr, cc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= rr < N and 0 <= cc < N and int(grid[rr][cc]) == 1 and (rr, cc) not in seen:
                seen.add((rr, cc))
                queue.append((rr, cc))
    return len(seen)


def analytic_gold(scene: dict) -> str:
    red_size = _component_size(scene, scene["red_cell"])
    blue_size = _component_size(scene, scene["blue_cell"])
    if red_size == blue_size:
        return "abstain"
    return "red" if red_size > blue_size else "blue"


def margin(scene: dict) -> float:
    return float(abs(_component_size(scene, scene["red_cell"]) - _component_size(scene, scene["blue_cell"])))


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain" or margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["cell_order"] = list(reversed(scene["cell_order"]))
    return [("reverse_disjoint_dash_painter_order", twin)]
