"""Independent pixel-only inverse for the slash-texture lattice."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum: int = 1) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    groups: list[np.ndarray] = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = deque([(int(y), int(x))])
        pixels = []
        while queue:
            yy, xx = queue.popleft()
            pixels.append((yy, xx))
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = yy + dy, xx + dx
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            groups.append(np.asarray(pixels, dtype=float))
    return groups


def _cluster_axis(values: list[float]) -> list[float]:
    ordered = sorted(values)
    clusters: list[list[float]] = []
    for value in ordered:
        if not clusters or value - clusters[-1][-1] > 14.0:
            clusters.append([value])
        else:
            clusters[-1].append(value)
    return [float(sum(group) / len(group)) for group in clusters]


def _marker_center(array: np.ndarray, which: str) -> tuple[float, float]:
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    if which == "red":
        mask = (red > 145) & (red > blue + 45) & (red > green + 45)
    else:
        mask = (blue > 135) & (blue > red + 40) & (blue > green + 15)
    yy, xx = np.nonzero(mask)
    if len(xx) < 25:
        raise ValueError(f"missing {which} marker")
    return float(xx.mean()), float(yy.mean())


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Ink is neutral and dark; colored rings are excluded by the channel-spread test.
    dark = array.max(axis=2) < 90
    neutral = array.max(axis=2) - array.min(axis=2) < 25
    dash_groups = _components(dark & neutral, minimum=25)
    if len(dash_groups) != 144:
        raise ValueError(f"expected 144 visible dash components, found {len(dash_groups)}")

    centers = []
    orientations = []
    for pixels in dash_groups:
        ys, xs = pixels[:, 0], pixels[:, 1]
        x0, y0 = float(xs.mean()), float(ys.mean())
        centers.append((x0, y0))
        covariance = float(np.mean((xs - x0) * (ys - y0)))
        orientations.append(1 if covariance < 0.0 else 0)

    x_axis = _cluster_axis([point[0] for point in centers])
    y_axis = _cluster_axis([point[1] for point in centers])
    if len(x_axis) != 12 or len(y_axis) != 12:
        raise ValueError("could not reconstruct the 12 by 12 lattice")
    grid = np.zeros((12, 12), dtype=np.uint8)
    for (x, y), orientation in zip(centers, orientations):
        c = int(np.argmin(np.abs(np.asarray(x_axis) - x)))
        r = int(np.argmin(np.abs(np.asarray(y_axis) - y)))
        grid[r, c] = orientation

    def marker_cell(which: str) -> tuple[int, int]:
        x, y = _marker_center(array, which)
        return (
            int(np.argmin(np.abs(np.asarray(y_axis) - y))),
            int(np.argmin(np.abs(np.asarray(x_axis) - x))),
        )

    def component_size(start: tuple[int, int]) -> int:
        if grid[start] != 1:
            raise ValueError("marker does not identify an up-right dash")
        seen = {start}
        queue = deque([start])
        while queue:
            r, c = queue.popleft()
            for rr, cc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
                if 0 <= rr < 12 and 0 <= cc < 12 and grid[rr, cc] == 1 and (rr, cc) not in seen:
                    seen.add((rr, cc))
                    queue.append((rr, cc))
        return len(seen)

    red_size = component_size(marker_cell("red"))
    blue_size = component_size(marker_cell("blue"))
    if red_size == blue_size:
        return "abstain"
    return "red" if red_size > blue_size else "blue"
