"""Independent pixel-only inverse arm for the ring/tilt conjunction world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")
COLORS = np.array([
    (39, 105, 205), (231, 123, 35), (38, 151, 88), (139, 79, 188)
], dtype=np.float32)


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        q = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while q:
            yy, xx = q.pop()
            pts.append((yy, xx))
            for ny, nx in ((yy-1,xx),(yy+1,xx),(yy,xx-1),(yy,xx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True
                    q.append((ny,nx))
        if len(pts) >= minimum:
            ys = [p[0] for p in pts]
            xs = [p[1] for p in pts]
            out.append((min(xs), min(ys), max(xs)+1, max(ys)+1, len(pts)))
    return out


def _regions(arr: np.ndarray) -> tuple[tuple[int,int,int,int], list[tuple[int,int,int,int]]]:
    # Border gray is neutral with all channels close and far from the white field.
    spread = arr.max(axis=2) - arr.min(axis=2)
    mean = arr.mean(axis=2)
    mask = (spread < 16) & (mean > 150) & (mean < 225)
    comps = _components(mask, 120)
    boxes = [(x0,y0,x1,y1) for x0,y0,x1,y1,n in comps if (x1-x0) > 90 and (y1-y0) > 100]
    if len(boxes) != 5:
        raise ValueError(f"expected five visible panels, found {len(boxes)}")
    boxes.sort(key=lambda b: (b[1], b[0]))
    reference = boxes[0]
    candidates = sorted(boxes[1:], key=lambda b: b[0])
    if len(candidates) != 4:
        raise ValueError("candidate panel detection failed")
    return reference, candidates


def _attributes(arr: np.ndarray, box: tuple[int,int,int,int]) -> tuple[int, str, float]:
    x0,y0,x1,y1 = box
    crop = arr[y0:y1, x0:x1].astype(np.float32)
    # Select the ring palette using nearest-color support; text and bar are excluded.
    dist = ((crop[:,:,None,:] - COLORS[None,None,:,:]) ** 2).sum(axis=3)
    nearest = dist.argmin(axis=2)
    near_palette = dist.min(axis=2) < 2600
    support = np.array([((nearest == i) & near_palette).sum() for i in range(len(COLORS))])
    color_idx = int(support.argmax())
    color_mask = (nearest == color_idx) & near_palette
    ys, xs = np.nonzero(color_mask)
    if len(xs) < 500:
        raise ValueError("ring color support too small")
    cx, cy = float(xs.mean()), float(ys.mean())

    # Dark bar pixels are measured only near the colored ring centroid, excluding labels.
    yy, xx = np.indices(crop.shape[:2])
    dark = (crop.max(axis=2) < 95) & (((xx-cx)**2 + (yy-cy)**2) < 31**2)
    by, bx = np.nonzero(dark)
    if len(bx) < 120:
        raise ValueError("bar support too small")
    covariance = float(np.mean((bx-bx.mean()) * (by-by.mean())))
    tilt = "falling" if covariance > 0 else "rising"
    confidence = abs(covariance)
    if confidence < 40:
        raise ValueError("ambiguous bar tilt")
    return color_idx, tilt, confidence


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    reference_box, candidate_boxes = _regions(arr)
    ref_color, ref_tilt, _ = _attributes(arr, reference_box)
    scores = []
    for box in candidate_boxes:
        color, tilt, _ = _attributes(arr, box)
        scores.append(int(color == ref_color) + int(tilt == ref_tilt))
    best = max(scores)
    if best != 2 or scores.count(best) != 1:
        raise ValueError(f"no unique two-attribute match: {scores}")
    ordered = sorted(scores, reverse=True)
    if ordered[0] - ordered[1] < 1:
        raise ValueError("insufficient runner-up margin")
    return LABELS[scores.index(best)]
