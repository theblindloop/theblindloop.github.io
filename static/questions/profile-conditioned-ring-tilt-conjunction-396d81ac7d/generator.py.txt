"""Deterministic scene sampler, analytic arm, and Pillow renderer."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 430
PALETTE = ("blue", "orange", "green", "purple")
RGB = {
    "blue": (39, 105, 205),
    "orange": (231, 123, 35),
    "green": (38, 151, 88),
    "purple": (139, 79, 188),
}
LABELS = ("A", "B", "C", "D")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    reference_color, alternate_color = rng.sample(PALETTE, 2)
    reference_tilt = rng.choice(("rising", "falling"))
    alternate_tilt = "falling" if reference_tilt == "rising" else "rising"
    target = [reference_color, reference_tilt]
    distractors = [
        [reference_color, alternate_tilt],
        [alternate_color, reference_tilt],
        [alternate_color, alternate_tilt],
    ]
    rng.shuffle(distractors)
    target_slot = int(seed) % 4
    candidate_pairs = distractors[:]
    candidate_pairs.insert(target_slot, target)
    return {
        "reference_color": reference_color,
        "reference_tilt": reference_tilt,
        "candidate_pairs": candidate_pairs,
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _glyph(draw: ImageDraw.ImageDraw, center: tuple[int, int], color: str, tilt: str) -> None:
    cx, cy = center
    r = 34
    draw.ellipse((cx-r, cy-r, cx+r, cy+r), outline=RGB[color], width=12)
    dx, dy = 23, 23
    if tilt == "rising":
        ends = (cx-dx, cy+dy, cx+dx, cy-dy)
    else:
        ends = (cx-dx, cy-dy, cx+dx, cy+dy)
    draw.line(ends, fill=(35, 38, 44), width=9)
    for x, y in ((ends[0], ends[1]), (ends[2], ends[3])):
        draw.ellipse((x-4, y-4, x+4, y+4), fill=(35, 38, 44))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 249, 246))
    draw = ImageDraw.Draw(image)
    border = (186, 190, 197)
    ink = (35, 38, 44)
    title_font = _font(17, True)
    label_font = _font(24, True)

    draw.rounded_rectangle((250, 28, 390, 172), radius=12, outline=border, width=3, fill=(255,255,255))
    draw.text((320, 47), "REFERENCE", font=title_font, fill=ink, anchor="mm")
    _glyph(draw, (320, 118), scene["reference_color"], scene["reference_tilt"])

    centers = (92, 244, 396, 548)
    for label, cx, pair in zip(LABELS, centers, scene["candidate_pairs"], strict=True):
        draw.rounded_rectangle((cx-65, 215, cx+65, 397), radius=12, outline=border, width=3, fill=(255,255,255))
        draw.text((cx, 242), label, font=label_font, fill=ink, anchor="mm")
        _glyph(draw, (cx, 320), pair[0], pair[1])
    return image


def _scores(scene: dict) -> list[int]:
    return [
        int(pair[0] == scene["reference_color"]) + int(pair[1] == scene["reference_tilt"])
        for pair in scene["candidate_pairs"]
    ]


def analytic_gold(scene: dict) -> str:
    # Access exactly the declared label inputs; alternate_color is a rendered nuisance.
    _ = scene["reference_color"], scene["reference_tilt"], scene["candidate_pairs"]
    scores = _scores(scene)
    if scores.count(max(scores)) != 1:
        raise ValueError("non-unique winner")
    return LABELS[scores.index(max(scores))]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    scores = _scores(scene)
    return scores.count(2) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
