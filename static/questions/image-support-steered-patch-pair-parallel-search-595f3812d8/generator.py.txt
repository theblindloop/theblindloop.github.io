"""Renderer and analytic gold for the patch-pair parallel search world.

A 1440 by 1080 page is divided into four horizontal bands of equal height. Each
band carries one station. A station is two disconnected clusters of twelve
short straight strokes; the two clusters of a station share one ink colour and
sit two grid cells apart with wide empty background between them. Inside a
cluster every stroke has the same orientation, so the orientation is coded
redundantly twelve times. Exactly one station has both of its clusters at the
same orientation; the answer is that station's band.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "patch_pair_parallel_search-renderer-0.1.0"

WIDTH = 1440
HEIGHT = 1080
BANDS = 4
BAND_H = HEIGHT // BANDS          # 270
CELL_W = WIDTH // 8               # 180
CELL_H = HEIGHT // 8              # 135
PATCH_W = 90                      # one cluster's ink box
PATCH_H = 68
PATCH_DX = 2 * CELL_W             # partner cluster sits two cells to the right
GLYPH_LEN = 18
GLYPH_THICK = 6
COLS = 4
ROWS = 3

WORDS = ("first", "second", "third", "fourth")

PALETTES = (
    {
        "background": (247, 245, 238),
        "inks": [(28, 42, 84), (140, 32, 44), (26, 92, 68), (96, 52, 120)],
    },
    {
        "background": (238, 240, 244),
        "inks": [(38, 38, 38), (150, 66, 20), (22, 74, 118), (110, 24, 90)],
    },
    {
        "background": (250, 248, 240),
        "inks": [(64, 22, 100), (18, 82, 96), (128, 44, 32), (44, 60, 24)],
    },
)

# Left origins that keep both clusters inside their two ablation cells.
_X_SLOTS = ((0, 95), (170, 275), (350, 455), (530, 635), (710, 815), (890, 990))
_Y_SLOTS = ((20, 75), (127, 202))


def _axial_gap(first: int, second: int) -> int:
    """Undirected angular difference between two stroke orientations."""

    raw = abs(int(first) - int(second)) % 180
    return min(raw, 180 - raw)


def _glyph_mask(angle: int) -> Image.Image:
    """Binary mask of one stroke, cropped tightly to its own ink."""

    size = GLYPH_LEN + GLYPH_THICK + 8
    canvas = Image.new("L", (size, size), 0)
    draw = ImageDraw.Draw(canvas)
    radians = math.radians(float(angle))
    centre = size / 2.0
    dx = math.cos(radians) * GLYPH_LEN / 2.0
    dy = -math.sin(radians) * GLYPH_LEN / 2.0
    px = -math.sin(radians) * GLYPH_THICK / 2.0
    py = -math.cos(radians) * GLYPH_THICK / 2.0
    draw.polygon(
        [
            (centre + dx + px, centre + dy + py),
            (centre + dx - px, centre + dy - py),
            (centre - dx - px, centre - dy - py),
            (centre - dx + px, centre - dy + py),
        ],
        fill=255,
    )
    box = canvas.getbbox()
    return canvas.crop(box)


def _patch_boxes(x0: int, y0: int, angle: int) -> list[tuple[int, int, Image.Image]]:
    """Lattice placement whose outer ink box is exactly PATCH_W by PATCH_H."""

    mask = _glyph_mask(angle)
    gw, gh = mask.size
    step_x = (PATCH_W - gw) / (COLS - 1)
    step_y = (PATCH_H - gh) / (ROWS - 1)
    places = []
    for row in range(ROWS):
        for col in range(COLS):
            places.append((x0 + round(col * step_x), y0 + round(row * step_y), mask))
    return places


def sample_scene(seed: int) -> dict:
    """Sample one balanced scene."""

    rng = random.Random(seed)
    palette = rng.randrange(len(PALETTES))
    winner = seed % BANDS
    inks = list(range(4))
    rng.shuffle(inks)

    winner_angle = rng.randrange(4, 176)
    base_gap = rng.randrange(20, 56)
    losers = [band for band in range(BANDS) if band != winner]
    rng.shuffle(losers)
    gaps = {losers[0]: base_gap}
    for band in losers[1:]:
        gaps[band] = rng.randrange(base_gap, 89)

    # One losing cluster repeats the winning orientation in a different band,
    # so a solver that hunts for a repeated orientation anywhere on the page
    # cannot name the band without binding each pair inside its own station.
    echo_band = losers[0]

    stations = []
    for band in range(BANDS):
        slot = rng.randrange(len(_X_SLOTS))
        low, high = _X_SLOTS[slot]
        x0 = rng.randrange(low, high + 1)
        low_y, high_y = _Y_SLOTS[rng.randrange(len(_Y_SLOTS))]
        y0 = band * BAND_H + rng.randrange(low_y, high_y + 1)
        if band == winner:
            angle_a = winner_angle
            angle_b = winner_angle
        else:
            gap = gaps[band]
            angle_a = winner_angle if band == echo_band else rng.randrange(0, 180)
            angle_b = (angle_a + gap * rng.choice((1, -1))) % 180
        if rng.random() < 0.5:
            angle_a, angle_b = angle_b, angle_a
        stations.append(
            {
                "band": band,
                "ink": inks[band],
                "x0": x0,
                "y0": y0,
                "angle_a": int(angle_a),
                "angle_b": int(angle_b),
            }
        )
    rng.shuffle(stations)

    boxes = [
        (station["x0"] - 24, station["y0"] - 24,
         station["x0"] + PATCH_DX + PATCH_W + 24, station["y0"] + PATCH_H + 24)
        for station in stations
    ]
    specks = []
    attempts = 0
    while len(specks) < rng.randrange(5, 10) and attempts < 400:
        attempts += 1
        sx = rng.randrange(8, WIDTH - 12)
        sy = rng.randrange(8, HEIGHT - 12)
        if any(x0 <= sx <= x1 and y0 <= sy <= y1 for x0, y0, x1, y1 in boxes):
            continue
        if any(abs(sx - px) < 20 and abs(sy - py) < 20 for px, py, _ in specks):
            continue
        specks.append((sx, sy, rng.randrange(4)))

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "palette": palette,
        "stations": stations,
        "specks": specks,
        "glyph_length": GLYPH_LEN,
        "glyph_thickness": GLYPH_THICK,
    }


def render(scene: dict) -> Image.Image:
    """Rasterize a scene; the inverse arm sees only this PNG."""

    palette = PALETTES[int(scene["palette"]) % len(PALETTES)]
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), palette["background"])
    for station in scene["stations"]:
        colour = tuple(palette["inks"][int(station["ink"]) % 4])
        x0 = int(station["x0"])
        y0 = int(station["y0"])
        for offset, angle in ((0, station["angle_a"]), (PATCH_DX, station["angle_b"])):
            for gx, gy, mask in _patch_boxes(x0 + offset, y0, int(angle)):
                patch = Image.new("RGB", mask.size, colour)
                image.paste(patch, (gx, gy), mask)
    draw = ImageDraw.Draw(image)
    for sx, sy, ink in scene["specks"]:
        draw.rectangle(
            [int(sx), int(sy), int(sx) + 3, int(sy) + 3],
            fill=tuple(palette["inks"][int(ink) % 4]),
        )
    return image


def _ranked_gaps(scene: dict) -> list[tuple[int, int]]:
    """Every station's own two-cluster orientation gap, smallest first."""

    scored = [
        (_axial_gap(int(station["angle_a"]), int(station["angle_b"])), int(station["band"]))
        for station in scene["stations"]
    ]
    scored.sort()
    return scored


def analytic_gold(scene: dict) -> str:
    """Band of the only station whose two clusters share one orientation."""

    scored = _ranked_gaps(scene)
    return WORDS[scored[0][1]]


def margin(scene: dict) -> float:
    """Runner-up decision margin in degrees of stroke orientation."""

    scored = _ranked_gaps(scene)
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose runner-up station is close to parallel as well."""

    return margin(scene) < 26.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared render-invariant, label-preserving latent rewrites."""

    reversed_specks = dict(scene)
    reversed_specks["specks"] = list(reversed(list(scene["specks"])))
    rotated = dict(scene)
    stations = list(scene["stations"])
    rotated["stations"] = stations[1:] + stations[:1]
    swapped = dict(scene)
    swapped["stations"] = list(reversed(stations))
    return [
        ("reversed_specks", reversed_specks),
        ("rotated_stations", rotated),
        ("reversed_stations", swapped),
    ]
