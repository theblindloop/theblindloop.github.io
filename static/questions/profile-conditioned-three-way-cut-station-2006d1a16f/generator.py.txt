"""Deterministic renderer and analytic arm for the three-way station cut task."""

from __future__ import annotations

import math
import random
from collections import deque

from PIL import Image, ImageDraw

COLORS = {
    "orange": (235, 132, 28),
    "green": (44, 160, 90),
    "purple": (142, 80, 190),
}
NODE = (28, 30, 36)
EDGE = (126, 132, 142)
BACKGROUND = (248, 249, 246)

BASE_NODES = {
    "t": (320, 320),
    "a": (160, 150), "u1": (80, 90), "u2": (240, 90), "u3": (160, 230),
    "b": (160, 480), "b1": (80, 420), "b2": (240, 420), "b3": (160, 570),
    "r1": (460, 210), "r2": (560, 320), "r3": (460, 430),
}
BASE_EDGES = [
    ("a", "u1"), ("a", "u2"), ("a", "u3"),
    ("u1", "u2"), ("u2", "u3"), ("u3", "u1"), ("t", "u3"),
    ("b", "b1"), ("b", "b2"), ("b", "b3"), ("b1", "b2"), ("t", "b2"),
    ("r1", "r2"), ("r2", "r3"), ("r3", "r1"), ("t", "r1"),
]


def _transform(point: tuple[int, int], transform: int) -> tuple[int, int]:
    x, y = point
    if transform == 0:
        return x, y
    if transform == 1:
        return 640 - y, x
    if transform == 2:
        return 640 - x, 640 - y
    if transform == 3:
        return y, 640 - x
    if transform == 4:
        return 640 - x, y
    return x, 640 - y


def sample_scene(seed: int) -> dict:
    """Sample a balanced color binding plus a global rigid layout transform."""
    rng = random.Random(int(seed) ^ 0x3A71C9)
    transform = rng.randrange(6)
    nodes = {name: list(_transform(point, transform)) for name, point in BASE_NODES.items()}
    names = list(nodes)
    # A rigid translation is visible and keeps generous canvas clearance.
    dx = rng.choice([-8, -4, 0, 4, 8])
    dy = rng.choice([-8, -4, 0, 4, 8])
    for name in names:
        nodes[name][0] += dx
        nodes[name][1] += dy

    palette = ["orange", "green", "purple"]
    target_index = int(seed) % 3
    target_color = palette[target_index]
    remaining = [c for c in palette if c != target_color]
    if (int(seed) // 3) % 2:
        remaining.reverse()
    # t -> 3 pieces; b -> 2 pieces; a -> 1 piece.
    candidate_nodes = {target_color: "t", remaining[0]: "b", remaining[1]: "a"}
    return {
        "width": 640,
        "height": 640,
        "nodes": nodes,
        "edges": [list(edge) for edge in BASE_EDGES],
        "candidate_nodes": candidate_nodes,
    }


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    nodes = scene["nodes"]
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for left, right in edges:
        draw.line([tuple(nodes[left]), tuple(nodes[right])], fill=EDGE, width=11)
    node_colors = {node: COLORS[color] for color, node in candidates.items()}
    radius = 15
    for name, (x, y) in nodes.items():
        fill = node_colors.get(name, NODE)
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=fill)
    return image


def _component_count_after(edges: list[list[str]], removed: str) -> int:
    vertices = {value for edge in edges for value in edge} - {removed}
    adjacency = {vertex: set() for vertex in vertices}
    for left, right in edges:
        if removed not in (left, right):
            adjacency[left].add(right)
            adjacency[right].add(left)
    unseen = set(vertices)
    count = 0
    while unseen:
        count += 1
        start = unseen.pop()
        queue = deque([start])
        while queue:
            here = queue.popleft()
            for nxt in adjacency[here] & unseen:
                unseen.remove(nxt)
                queue.append(nxt)
    return count


def analytic_gold(scene: dict) -> str:
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    matches = [color for color, node in candidates.items() if _component_count_after(edges, node) == 3]
    if len(matches) != 1:
        raise ValueError(f"expected one three-component deletion, found {matches}")
    return matches[0]


def margin(scene: dict) -> float:
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    counts = [_component_count_after(edges, node) for node in candidates.values()]
    if 3 not in counts:
        return 0.0
    return float(min(abs(3 - value) for value in counts if value != 3))


def is_quarantined(scene: dict) -> bool:
    edges = scene["edges"]
    candidates = scene["candidate_nodes"]
    counts = sorted(_component_count_after(edges, node) for node in candidates.values())
    points = list(scene["nodes"].values())
    min_sep = min(math.dist(a, b) for i, a in enumerate(points) for b in points[i + 1 :])
    return counts != [1, 2, 3] or min_sep < 48


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
