"""Deterministic sampler, renderer, and analytic arm for difference mosaics."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS_WIDTH = 768
CANVAS_HEIGHT = 768
BOARD_SIZE = 300
GRID_SIDE = 5
CELL_COUNT = GRID_SIDE * GRID_SIDE

BACKGROUND = (255, 255, 255)
BOARD = (226, 213, 181)
OUTLINE = (72, 64, 55)
ACCENT = (112, 98, 77)

PALETTES = {
    "slate_sand": ((42, 57, 73), (244, 235, 207)),
    "plum_oat": ((76, 45, 88), (239, 229, 199)),
    "pine_clay": ((38, 77, 59), (244, 229, 211)),
}
TAG_RGB = {
    "amber": (244, 181, 35),
    "cyan": (30, 174, 190),
    "violet": (145, 84, 185),
}
TAG_NAMES = tuple(TAG_RGB)

SOURCE_ORIGINS = ((45, 42), (175, 42))
SOURCE_CELL = 16
CANDIDATE_ORIGINS = ((20, 158), (115, 158), (210, 158))
CANDIDATE_CELL = 14
TAG_TOP = 248


def _difference(source_a: list[int], source_b: list[int]) -> list[int]:
    return [int(left != right) for left, right in zip(source_a, source_b, strict=True)]


def _mutate_equal_ink(bits: list[int], rng: random.Random, pairs: int) -> list[int]:
    output = list(bits)
    dark = [index for index, value in enumerate(output) if value]
    light = [index for index, value in enumerate(output) if not value]
    for index in rng.sample(dark, pairs):
        output[index] = 0
    for index in rng.sample(light, pairs):
        output[index] = 1
    return output


def sample_scene(seed: int) -> dict:
    """Sample balanced equal-ink inputs and candidates from one integer seed."""

    seed = int(seed)
    rng = random.Random(seed)
    source_a = [1] * 12 + [0] * 13
    rng.shuffle(source_a)
    dark = [index for index, value in enumerate(source_a) if value]
    light = [index for index, value in enumerate(source_a) if not value]
    toggled = set(rng.sample(dark, 6) + rng.sample(light, 6))
    source_b = [1 - value if index in toggled else value for index, value in enumerate(source_a)]
    correct = _difference(source_a, source_b)
    near = _mutate_equal_ink(correct, rng, 1)
    far = _mutate_equal_ink(correct, rng, 3)

    correct_slot = (seed // 3) % 3
    correct_tag = TAG_NAMES[seed % 3]
    remaining_tags = [tag for tag in TAG_NAMES if tag != correct_tag]
    rng.shuffle(remaining_tags)
    candidate_tags = list(remaining_tags)
    candidate_tags.insert(correct_slot, correct_tag)
    distractors = [near, far]
    rng.shuffle(distractors)
    candidates = list(distractors)
    candidates.insert(correct_slot, correct)

    return {
        "canvas_width": CANVAS_WIDTH,
        "canvas_height": CANVAS_HEIGHT,
        "left": 234 + rng.randrange(-8, 9),
        "top": 234 + rng.randrange(-8, 9),
        "source_a": source_a,
        "source_b": source_b,
        "candidates": candidates,
        "candidate_tags": candidate_tags,
        "bit_palette": rng.choice(tuple(PALETTES)),
    }


def _draw_grid(
    draw: ImageDraw.ImageDraw,
    origin_x: int,
    origin_y: int,
    cell: int,
    bits: list[int],
    colors: tuple[tuple[int, int, int], tuple[int, int, int]],
    *,
    heavy: bool,
) -> None:
    dark, light = colors
    size = GRID_SIDE * cell
    draw.rectangle(
        (origin_x - 3, origin_y - 3, origin_x + size + 2, origin_y + size + 2),
        fill=OUTLINE,
    )
    for index, value in enumerate(bits):
        row, column = divmod(index, GRID_SIDE)
        x0 = origin_x + column * cell
        y0 = origin_y + row * cell
        inset = 2 if heavy else 1
        draw.rectangle(
            (x0 + inset, y0 + inset, x0 + cell - 1, y0 + cell - 1),
            fill=dark if value else light,
        )


def render(scene: dict) -> Image.Image:
    """Render one connected, medium-span square board on a white canvas."""

    width = int(scene["canvas_width"])
    height = int(scene["canvas_height"])
    left = int(scene["left"])
    top = int(scene["top"])
    source_a = [int(value) for value in scene["source_a"]]
    source_b = [int(value) for value in scene["source_b"]]
    candidates = [[int(value) for value in row] for row in scene["candidates"]]
    tags = [str(value) for value in scene["candidate_tags"]]
    palette_name = str(scene["bit_palette"])
    colors = PALETTES[palette_name]

    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((left, top, left + BOARD_SIZE - 1, top + BOARD_SIZE - 1), fill=BOARD)
    draw.rectangle((left, top, left + BOARD_SIZE - 1, top + BOARD_SIZE - 1), outline=OUTLINE, width=4)

    for origin, bits in zip(SOURCE_ORIGINS, (source_a, source_b), strict=True):
        _draw_grid(draw, left + origin[0], top + origin[1], SOURCE_CELL, bits, colors, heavy=True)

    # A purely visible composition emblem: two differently filled tiles converge
    # onto one dark result tile. The public prompt supplies the complete rule.
    dark, light = colors
    draw.rounded_rectangle((left + 137, top + 53, left + 162, top + 102), radius=7, fill=(201, 187, 153), outline=ACCENT, width=2)
    draw.rectangle((left + 141, top + 58, left + 149, top + 70), fill=dark)
    draw.rectangle((left + 151, top + 58, left + 159, top + 70), fill=light)
    draw.line((left + 145, top + 76, left + 155, top + 76), fill=OUTLINE, width=2)
    draw.line((left + 150, top + 72, left + 150, top + 87), fill=OUTLINE, width=2)
    draw.polygon(((left + 146, top + 84), (left + 154, top + 84), (left + 150, top + 90)), fill=OUTLINE)
    draw.rectangle((left + 141, top + 91, left + 159, top + 98), fill=dark)

    draw.line((left + 30, top + 143, left + 270, top + 143), fill=ACCENT, width=3)
    for (origin_x, origin_y), bits, tag in zip(CANDIDATE_ORIGINS, candidates, tags, strict=True):
        _draw_grid(draw, left + origin_x, top + origin_y, CANDIDATE_CELL, bits, colors, heavy=False)
        tag_left = left + origin_x + 10
        draw.rounded_rectangle(
            (tag_left, top + TAG_TOP, tag_left + 49, top + TAG_TOP + 27),
            radius=8,
            fill=TAG_RGB[tag],
            outline=OUTLINE,
            width=2,
        )
    return image


def _expected(scene: dict) -> list[int]:
    return _difference(
        [int(value) for value in scene["source_a"]],
        [int(value) for value in scene["source_b"]],
    )


def _candidate_distances(scene: dict) -> list[int]:
    expected = _expected(scene)
    candidates = [[int(value) for value in row] for row in scene["candidates"]]
    return [sum(left != right for left, right in zip(expected, row, strict=True)) for row in candidates]


def analytic_gold(scene: dict) -> str:
    distances = _candidate_distances(scene)
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one matching output mosaic")
    tags = [str(value) for value in scene["candidate_tags"]]
    return tags[matches[0]]


def margin(scene: dict) -> float:
    distances = sorted(_candidate_distances(scene))
    return float(distances[1] - distances[0]) if len(distances) >= 2 else 0.0


def is_quarantined(scene: dict) -> bool:
    distances = _candidate_distances(scene)
    return distances.count(0) != 1 or margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
