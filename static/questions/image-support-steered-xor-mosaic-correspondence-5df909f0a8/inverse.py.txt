"""Independent image-only inverse arm for the difference-mosaic board."""

from __future__ import annotations

import numpy as np
from PIL import Image

_WHITE = np.array((255, 255, 255), dtype=np.uint8)
_BOARD_SIZE = 300
_SIDE = 5
_SOURCE_LAYOUT = ((45 / 300, 42 / 300), (175 / 300, 42 / 300))
_SOURCE_CELL = 16 / 300
_CANDIDATE_LAYOUT = ((20 / 300, 158 / 300), (115 / 300, 158 / 300), (210 / 300, 158 / 300))
_CANDIDATE_CELL = 14 / 300
_TAG_Y = 261.5 / 300
_TAG_RGB = {
    (244, 181, 35): "amber",
    (30, 174, 190): "cyan",
    (145, 84, 185): "violet",
}


def _role(color: np.ndarray) -> int:
    luminance = float(np.mean(color.astype(np.float64)))
    if luminance < 120.0:
        return 1
    if luminance > 175.0 and not np.all(color == _WHITE):
        return 0
    raise ValueError("tile center is neither a dark nor a light role color")


def _read_grid(crop: np.ndarray, origin: tuple[float, float], cell: float) -> tuple[int, ...]:
    height, width = crop.shape[:2]
    bits = []
    for row in range(_SIDE):
        for column in range(_SIDE):
            x = min(width - 1, int((origin[0] + (column + 0.5) * cell) * width))
            y = min(height - 1, int((origin[1] + (row + 0.5) * cell) * height))
            bits.append(_role(crop[y, x]))
    return tuple(bits)


def _tag_at(crop: np.ndarray, origin_x: float) -> str:
    height, width = crop.shape[:2]
    x = min(width - 1, int((origin_x + 35 / 300) * width))
    y = min(height - 1, int(_TAG_Y * height))
    color = tuple(int(value) for value in crop[y, x])
    if color not in _TAG_RGB:
        raise ValueError("candidate answer tag has an unknown color")
    return _TAG_RGB[color]


def decision_from_image(image: Image.Image) -> str:
    """Recover the winning tag from raster evidence alone, or abstain."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    foreground = np.any(array != _WHITE, axis=2)
    ys, xs = np.nonzero(foreground)
    if not len(xs):
        raise ValueError("no mosaic board foreground")
    x0, x1 = int(xs.min()), int(xs.max()) + 1
    y0, y1 = int(ys.min()), int(ys.max()) + 1
    crop = array[y0:y1, x0:x1]
    if crop.shape[:2] != (_BOARD_SIZE, _BOARD_SIZE):
        raise ValueError("foreground is not the declared square board")
    if np.any(np.all(crop == _WHITE, axis=2)):
        raise ValueError("the connected board surface has been erased")

    source_a, source_b = (
        _read_grid(crop, origin, _SOURCE_CELL) for origin in _SOURCE_LAYOUT
    )
    candidates = [
        _read_grid(crop, origin, _CANDIDATE_CELL) for origin in _CANDIDATE_LAYOUT
    ]
    tags = [_tag_at(crop, origin[0]) for origin in _CANDIDATE_LAYOUT]
    if len(set(tags)) != 3:
        raise ValueError("answer tags are missing or repeated")
    if sum(source_a) != 12 or sum(source_b) != 12:
        raise ValueError("source masks do not have the declared equal ink")
    expected = tuple(int(left != right) for left, right in zip(source_a, source_b, strict=True))
    if sum(expected) != 12 or any(sum(candidate) != 12 for candidate in candidates):
        raise ValueError("candidate masks do not preserve the declared equal ink")
    distances = [
        sum(left != right for left, right in zip(expected, candidate, strict=True))
        for candidate in candidates
    ]
    matches = [index for index, distance in enumerate(distances) if distance == 0]
    if len(matches) != 1:
        raise ValueError("difference correspondence is not unique")
    if sorted(distances)[1] < 2:
        raise ValueError("runner-up lies inside the quarantine boundary")
    return tags[matches[0]]
