"""Independent image-only inverse for colored graph correspondence.

This module deliberately duplicates the public palette and panel geometry rather
than importing the renderer.  It receives only a final RGB PNG.  It segments
each palette color inside each fixed panel, obtains the component bounding box,
and infers a side-sharing relation from the centers: neighboring tile centers
are one tile-width apart on the same row or column; diagonal/corner contact is
not an edge.  It then compares unordered color-pair edge sets.
"""

from __future__ import annotations

from itertools import combinations

import numpy as np
from PIL import Image


_PANEL_BOXES = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}
_PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
    "teal": (0, 145, 155),
}
_CENTER_TOLERANCE = 5.0
_ADJACENT_DISTANCE = 53.0


def _component_centers(image: Image.Image, box: tuple[int, int, int, int]) -> dict[str, tuple[float, float]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    left, top, right, bottom = box
    crop = arr[top + 2 : bottom - 2, left + 2 : right - 2]
    centers: dict[str, tuple[float, float]] = {}
    for name, rgb in _PALETTE.items():
        mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
        ys, xs = np.where(mask)
        if len(xs) < 500:
            raise ValueError(f"missing or tiny {name} component")
        centers[name] = (float(xs.mean() + left + 2), float(ys.mean() + top + 2))
    return centers


def _edge_signature(nodes: dict[str, tuple[float, float]]) -> frozenset[tuple[str, str]]:
    edges: set[tuple[str, str]] = set()
    for left_name, right_name in combinations(sorted(nodes), 2):
        x1, y1 = nodes[left_name]
        x2, y2 = nodes[right_name]
        same_row = abs(y1 - y2) <= _CENTER_TOLERANCE and abs(x1 - x2) <= _ADJACENT_DISTANCE
        same_col = abs(x1 - x2) <= _CENTER_TOLERANCE and abs(y1 - y2) <= _ADJACENT_DISTANCE
        if same_row or same_col:
            edges.add((left_name, right_name))
    return frozenset(edges)


def decision_from_image(image: Image.Image) -> str:
    """Return A or B using pixels only; no latent scene or renderer access."""
    reference = _edge_signature(_component_centers(image, _PANEL_BOXES["reference"]))
    candidate_a = _edge_signature(_component_centers(image, _PANEL_BOXES["A"]))
    candidate_b = _edge_signature(_component_centers(image, _PANEL_BOXES["B"]))
    matches = (candidate_a == reference, candidate_b == reference)
    if sum(matches) != 1:
        raise ValueError("image has no unique candidate graph match")
    return "A" if matches[0] else "B"
