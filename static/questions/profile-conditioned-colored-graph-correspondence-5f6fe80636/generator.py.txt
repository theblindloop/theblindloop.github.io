"""Renderer and analytic world for colored-part graph correspondence.

The latent scene stores three visible collections of six uniquely colored
square parts.  Their local integer slots form an asymmetric tree.  A candidate
may move or rotate the tree, but the graph is defined by which colored squares
share a side.  One candidate is exact and the other is made non-exact by a
color swap, while both retain identical local statistics.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "colored_graph_correspondence-0.1.0"
CANVAS_SIZE = (1000, 560)
CELL_SIZE = 48
TILE_SIZE = 48

PANEL_BOXES = {
    "reference": (24, 72, 324, 528),
    "A": (355, 72, 645, 528),
    "B": (676, 72, 966, 528),
}

PALETTE = {
    "red": (220, 50, 47),
    "blue": (38, 91, 185),
    "green": (40, 151, 77),
    "gold": (235, 157, 24),
    "purple": (135, 70, 170),
    "teal": (0, 145, 155),
}
COLORS = tuple(PALETTE)

# The degree sequence is asymmetric, so no non-identity relabelling preserves
# all edges.  Side-sharing edges are the five unit horizontal/vertical pairs.
BASE_SLOTS = ((0, 0), (1, 0), (2, 0), (1, 1), (1, 2), (2, 2))
BASE_EDGE_COUNT = 5


def _rotate_slot(slot: tuple[int, int], turns: int) -> tuple[int, int]:
    x, y = slot
    for _ in range(turns % 4):
        x, y = 2 - y, x
    return x, y


def _parts_from_slots(slots: Iterable[tuple[int, int]], colors: Iterable[str]) -> list[dict]:
    return [
        {"x": int(x), "y": int(y), "color": str(color)}
        for (x, y), color in zip(slots, colors)
    ]


def _rotated_parts(parts: list[dict], turns: int) -> list[dict]:
    return [
        {
            "x": _rotate_slot((int(part["x"]), int(part["y"])), turns)[0],
            "y": _rotate_slot((int(part["x"]), int(part["y"])), turns)[1],
            "color": str(part["color"]),
        }
        for part in parts
    ]


def _swapped_colors(parts: list[dict], first: int, second: int) -> list[dict]:
    result = [dict(part) for part in parts]
    result[first]["color"], result[second]["color"] = (
        result[second]["color"],
        result[first]["color"],
    )
    return result


def _edge_signature(parts: list[dict]) -> frozenset[tuple[str, str]]:
    edges: set[tuple[str, str]] = set()
    for i, left in enumerate(parts):
        for right in parts[i + 1 :]:
            dx = abs(int(left["x"]) - int(right["x"]))
            dy = abs(int(left["y"]) - int(right["y"]))
            if (dx == 1 and dy == 0) or (dx == 0 and dy == 1):
                edges.add(tuple(sorted((str(left["color"]), str(right["color"])))))
    return frozenset(edges)


def _edge_overlap(reference: frozenset, candidate: frozenset) -> int:
    return len(reference & candidate)


def _well_formed_parts(parts: object) -> bool:
    if not isinstance(parts, list) or len(parts) != len(BASE_SLOTS):
        return False
    seen_slots: set[tuple[int, int]] = set()
    seen_colors: set[str] = set()
    for part in parts:
        if not isinstance(part, dict) or set(part) != {"x", "y", "color"}:
            return False
        try:
            slot = (int(part["x"]), int(part["y"]))
            color = str(part["color"])
        except (TypeError, ValueError):
            return False
        if not (0 <= slot[0] <= 2 and 0 <= slot[1] <= 2):
            return False
        if slot in seen_slots or color not in PALETTE or color in seen_colors:
            return False
        seen_slots.add(slot)
        seen_colors.add(color)
    return len(seen_slots) == len(BASE_SLOTS) and seen_colors == set(COLORS)


def sample_scene(seed: int) -> dict:
    """Sample one deterministic, balanced, fully visible scene."""
    rng = random.Random(int(seed))
    colors = list(COLORS)
    rng.shuffle(colors)
    reference = _parts_from_slots(BASE_SLOTS, colors)

    correct = _rotated_parts(reference, rng.randrange(4))
    wrong = _rotated_parts(reference, rng.randrange(4))
    for _ in range(20):
        first, second = rng.sample(range(len(wrong)), 2)
        trial = _swapped_colors(wrong, first, second)
        if _edge_signature(trial) != _edge_signature(reference):
            wrong = trial
            break
    else:
        raise RuntimeError("could not construct a non-matching graph")

    # Seed parity counterbalances answer identity while leaving no visible seed
    # cue: each panel still has exactly the same six colors and six tile areas.
    correct_is_a = int(seed) % 2 == 0
    candidate_a, candidate_b = (correct, wrong) if correct_is_a else (wrong, correct)

    def anchor(box: tuple[int, int, int, int]) -> list[int]:
        left, top, right, bottom = box
        return [
            left + 64 + rng.randrange(0, 43),
            top + 125 + rng.randrange(0, 43),
        ]

    return {
        "reference": reference,
        "candidate_a": candidate_a,
        "candidate_b": candidate_b,
        "anchors": {
            "reference": anchor(PANEL_BOXES["reference"]),
            "A": anchor(PANEL_BOXES["A"]),
            "B": anchor(PANEL_BOXES["B"]),
        },
    }


def _draw_parts(draw: ImageDraw.ImageDraw, parts: list[dict], anchor: list[int]) -> None:
    ax, ay = int(anchor[0]), int(anchor[1])
    outline = (30, 30, 38)
    for part in parts:
        x = ax + int(part["x"]) * CELL_SIZE
        y = ay + int(part["y"]) * CELL_SIZE
        color = PALETTE[str(part["color"])]
        draw.rectangle(
            (x, y, x + TILE_SIZE - 1, y + TILE_SIZE - 1),
            fill=color,
            outline=outline,
            width=1,
        )


def render(scene: dict) -> Image.Image:
    """Render only scene-visible fields; no answer-specific decoration is used."""
    image = Image.new("RGB", CANVAS_SIZE, (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    for name, box in PANEL_BOXES.items():
        left, top, right, bottom = box
        draw.rectangle(box, outline=(75, 78, 88), width=2)
        label = "REFERENCE" if name == "reference" else name
        draw.text((left + 11, top + 12), label, fill=(25, 28, 35), font=font)

    _draw_parts(draw, scene["reference"], scene["anchors"]["reference"])
    _draw_parts(draw, scene["candidate_a"], scene["anchors"]["A"])
    _draw_parts(draw, scene["candidate_b"], scene["anchors"]["B"])
    return image


def analytic_gold(scene: dict) -> str:
    """Select the unique candidate whose colored edge set equals the reference."""
    reference_edges = _edge_signature(scene["reference"])
    a_edges = _edge_signature(scene["candidate_a"])
    b_edges = _edge_signature(scene["candidate_b"])
    matches = (a_edges == reference_edges, b_edges == reference_edges)
    if sum(matches) != 1:
        raise ValueError("scene has no unique graph correspondence")
    return "A" if matches[0] else "B"


def margin(scene: dict) -> float:
    """Return the positive score gap between the matching candidate and runner-up."""
    reference_edges = _edge_signature(scene["reference"])
    a_score = _edge_overlap(reference_edges, _edge_signature(scene["candidate_a"]))
    b_score = _edge_overlap(reference_edges, _edge_signature(scene["candidate_b"]))
    return float(abs(a_score - b_score))


def is_quarantined(scene: dict) -> bool:
    """Reject malformed or tied graph comparisons; sampled scenes are well clear."""
    try:
        parts_ok = all(
            _well_formed_parts(scene[key])
            for key in ("reference", "candidate_a", "candidate_b")
        )
        if not parts_ok:
            return True
        reference_edges = _edge_signature(scene["reference"])
        a_edges = _edge_signature(scene["candidate_a"])
        b_edges = _edge_signature(scene["candidate_b"])
        return (a_edges == b_edges) or ((a_edges == reference_edges) == (b_edges == reference_edges))
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The visible fields are injective for the label; no latent aliases apply."""
    return []
