"""Matched renderer for a percent-like symbol with two dot-side relations."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


DARK_COLORS = ((38, 43, 52), (50, 49, 67), (38, 55, 52))
DOT_COLORS = ((31, 137, 171), (211, 83, 58), (126, 88, 177))
BACKGROUNDS = ((249, 248, 244), (245, 249, 251), (251, 246, 249))
MARK_COLORS = ((232, 176, 45), (65, 163, 110), (218, 115, 158))


def sample_scene(seed: int) -> dict:
    """Return deterministic even/odd control-counterfactual pairs.

    Seeds 2k and 2k+1 share every nuisance.  Only ``opposite_sides`` changes,
    so icon styling cannot predict whether the visible relation is canonical.
    """

    pair_seed = int(seed) // 2
    rng = random.Random(0xC0FFEE31 ^ (pair_seed * 1_000_003))
    return {
        "width": 420,
        "height": 360,
        "opposite_sides": int(seed) % 2 == 0,
        "same_side_sign": -1 if rng.randrange(2) == 0 else 1,
        "center_x": 210 + rng.randint(-10, 10),
        "center_y": 180 + rng.randint(-8, 8),
        "slash_angle_deg": rng.uniform(-53.0, -43.0),
        "slash_length": rng.randrange(246, 269, 2),
        "slash_width": rng.randrange(15, 20, 2),
        "dot_radius": rng.randrange(28, 35, 2),
        "perpendicular_offset": rng.randrange(78, 89, 2),
        "along_offset": rng.randrange(58, 69, 2),
        "palette_index": rng.randrange(len(DOT_COLORS)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "corner_pattern": rng.randrange(4),
        "dot_draw_order": [0, 1] if rng.randrange(2) == 0 else [1, 0],
    }


def _geometry(scene: dict) -> tuple[list[tuple[float, float]], tuple[float, float], tuple[float, float]]:
    angle = math.radians(float(scene["slash_angle_deg"]))
    direction = (math.cos(angle), math.sin(angle))
    normal = (-direction[1], direction[0])
    center = (float(scene["center_x"]), float(scene["center_y"]))
    along = float(scene["along_offset"])
    perpendicular = float(scene["perpendicular_offset"])
    if bool(scene["opposite_sides"]):
        signs = (1, -1)
    else:
        sign = int(scene["same_side_sign"])
        signs = (sign, sign)
    dots: list[tuple[float, float]] = []
    for distance, sign in zip((-along, along), signs):
        dots.append(
            (
                center[0] + distance * direction[0] + sign * perpendicular * normal[0],
                center[1] + distance * direction[1] + sign * perpendicular * normal[1],
            )
        )
    return dots, direction, normal


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * scale)) for value in box)


def render(scene: dict) -> Image.Image:
    """Render one balanced icon scene using only declared scene fields."""

    ss = 4
    width, height = int(scene["width"]), int(scene["height"])
    palette = int(scene["palette_index"])
    background = BACKGROUNDS[int(scene["background_index"])]
    dark = DARK_COLORS[palette]
    dot_color = DOT_COLORS[palette]
    mark_color = MARK_COLORS[palette]
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    cx, cy = float(scene["center_x"]), float(scene["center_y"])
    dots, direction, _ = _geometry(scene)
    half = float(scene["slash_length"]) / 2.0
    start = ((cx - half * direction[0]) * ss, (cy - half * direction[1]) * ss)
    end = ((cx + half * direction[0]) * ss, (cy + half * direction[1]) * ss)
    line_width = int(scene["slash_width"]) * ss
    draw.line((start, end), fill=dark, width=line_width)
    cap = line_width / 2.0
    for x, y in (start, end):
        draw.ellipse((x - cap, y - cap, x + cap, y + cap), fill=dark)

    radius = float(scene["dot_radius"])
    for index in scene["dot_draw_order"]:
        x, y = dots[int(index)]
        draw.ellipse(
            _scaled_box((x - radius, y - radius, x + radius, y + radius), ss),
            fill=dot_color,
        )

    # Small corner marks are paired nuisances and never use target colors.
    pattern = int(scene["corner_pattern"])
    corners = ((24, 24), (396, 24), (396, 336), (24, 336))
    for index, (x, y) in enumerate(corners):
        if (index + pattern) % 2 == 0:
            draw.polygon(
                ((x * ss, (y - 7) * ss), ((x + 7) * ss, (y + 7) * ss), ((x - 7) * ss, (y + 7) * ss)),
                fill=mark_color,
            )
        else:
            draw.rectangle(((x - 6) * ss, (y - 6) * ss, (x + 6) * ss, (y + 6) * ss), fill=mark_color)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if bool(scene["opposite_sides"]) else "no"


def margin(scene: dict) -> float:
    """Visible clearance from the nearer dot rim to the slash edge, in pixels."""

    return (
        float(scene["perpendicular_offset"])
        - float(scene["dot_radius"])
        - float(scene["slash_width"]) / 2.0
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0 or int(scene["dot_radius"]) < 24 or int(scene["slash_length"]) < 220


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["dot_draw_order"] = list(reversed(twin["dot_draw_order"]))
    return [("reverse_disjoint_dot_draw_order", twin)]
