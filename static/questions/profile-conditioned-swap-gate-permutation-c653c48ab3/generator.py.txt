"""Deterministic Pillow renderer for sequential visible swap composition."""

from __future__ import annotations

import itertools
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 720, 640, 2
COLORS = {
    "red": (218, 67, 67),
    "blue": (54, 112, 201),
    "gold": (230, 168, 35),
}
QUESTION = (
    "Start with the three colored beads shown above the tracks. Read the numbered "
    "gates from top to bottom. At each gate, the two tracks joined by the horizontal "
    "bar swap their beads; the third bead stays on its track. Which option shows the "
    "final left-to-right bead order? Answer A, B, or C."
)


def _font(size: int, bold: bool = False):
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    return ImageFont.truetype(name, size * SCALE)


def _apply_swaps(start_order: list[str], gates: list[int]) -> list[str]:
    order = list(start_order)
    for gate in gates:
        order[gate], order[gate + 1] = order[gate + 1], order[gate]
    return order


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start = list(COLORS)
    rng.shuffle(start)
    while True:
        gates = [rng.randrange(2) for _ in range(5)]
        if 0 in gates and 1 in gates and any(a != b for a, b in zip(gates, gates[1:])):
            break
    final = _apply_swaps(start, gates)
    permutations = [list(value) for value in itertools.permutations(COLORS)]
    distractors = [value for value in permutations if value != final]
    rng.shuffle(distractors)
    options = [final, distractors[0], distractors[1]]
    rng.shuffle(options)
    return {"start_order": start, "gates": gates, "options": options}


def analytic_gold(scene: dict) -> str:
    final = _apply_swaps(scene["start_order"], scene["gates"])
    matches = [index for index, option in enumerate(scene["options"]) if option == final]
    if len(matches) != 1:
        raise ValueError("scene does not have exactly one matching option")
    return "ABC"[matches[0]]


def margin(scene: dict) -> float:
    final = _apply_swaps(scene["start_order"], scene["gates"])
    distances = sorted(
        sum(left != right for left, right in zip(final, option))
        for option in scene["options"]
        if option != final
    )
    return float(distances[0]) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        valid = (
            sorted(scene["start_order"]) == sorted(COLORS)
            and len(scene["gates"]) == 5
            and all(gate in (0, 1) for gate in scene["gates"])
            and len(scene["options"]) == 3
            and len({tuple(option) for option in scene["options"]}) == 3
            and all(sorted(option) == sorted(COLORS) for option in scene["options"])
        )
        analytic_gold(scene)
        return not valid
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    start_order = scene["start_order"]
    gates = scene["gates"]
    options = scene["options"]
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), (248, 247, 243))
    draw = ImageDraw.Draw(image)

    def line(points, fill, width):
        draw.line([(x * SCALE, y * SCALE) for x, y in points], fill=fill, width=width * SCALE)

    def ellipse(box, fill, outline=None, width=1):
        draw.ellipse(tuple(value * SCALE for value in box), fill=fill, outline=outline, width=width * SCALE)

    def rect(box, fill=None, outline=None, width=1, radius=0):
        coords = tuple(value * SCALE for value in box)
        draw.rounded_rectangle(coords, radius=radius * SCALE, fill=fill, outline=outline, width=width * SCALE)

    def text(xy, value, font, fill, anchor=None):
        draw.text((xy[0] * SCALE, xy[1] * SCALE), value, font=font, fill=fill, anchor=anchor)

    dark = (43, 47, 53)
    gray = (114, 120, 126)
    text((360, 26), "SWAP THE BEADS", _font(22, True), dark, "mm")
    text((360, 54), "Follow gates 1–5 from top to bottom", _font(13), gray, "mm")

    xs = [190, 360, 530]
    top_y = 100
    track_bottom = 405
    for x in xs:
        line([(x, top_y + 16), (x, track_bottom)], (116, 122, 128), 5)
    for x, color in zip(xs, start_order):
        ellipse((x - 16, top_y - 16, x + 16, top_y + 16), COLORS[color], dark, 3)

    gate_ys = [155, 210, 265, 320, 375]
    for number, (gate, y) in enumerate(zip(gates, gate_ys), 1):
        left, right = xs[gate], xs[gate + 1]
        line([(left, y), (right, y)], (248, 247, 243), 15)
        line([(left, y), (right, y)], dark, 7)
        # Arrowheads make the horizontal bar visibly an exchange operation.
        line([(left + 14, y - 8), (left, y), (left + 14, y + 8)], dark, 5)
        line([(right - 14, y - 8), (right, y), (right - 14, y + 8)], dark, 5)
        text((135, y), str(number), _font(14, True), dark, "mm")

    text((360, 430), "FINAL ORDER", _font(14, True), dark, "mm")
    option_ys = [480, 535, 590]
    for label, option, y in zip("ABC", options, option_ys):
        rect((120, y - 23, 600, y + 23), fill=(255, 255, 255), outline=(186, 188, 187), width=2, radius=10)
        text((145, y), label, _font(16, True), dark, "mm")
        for x, color in zip(xs, option):
            ellipse((x - 13, y - 13, x + 13, y + 13), COLORS[color], dark, 2)

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
