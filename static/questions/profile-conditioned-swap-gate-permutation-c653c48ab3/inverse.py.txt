"""Independent raster-only inverse for the swap-gate world."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array((218, 67, 67)),
    "blue": np.array((54, 112, 201)),
    "gold": np.array((230, 168, 35)),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        sx = sy = count = 0
        while queue:
            cy, cx = queue.popleft()
            sx += cx
            sy += cy
            count += 1
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= 120:
            found.append((sx / count, sy / count, count))
    return found


def _colored_beads(array: np.ndarray) -> list[tuple[float, float, str]]:
    pixels = array.astype(np.int16)
    beads = []
    for name, color in PALETTE.items():
        distance = np.sqrt(np.sum((pixels - color) ** 2, axis=2))
        for x, y, _ in _components(distance < 45):
            beads.append((x, y, name))
    return beads


def _row_groups(beads: list[tuple[float, float, str]]) -> list[list[tuple[float, float, str]]]:
    rows = []
    for bead in sorted(beads, key=lambda item: item[1]):
        if not rows or bead[1] - np.mean([item[1] for item in rows[-1]]) > 30:
            rows.append([bead])
        else:
            rows[-1].append(bead)
    return [sorted(row, key=lambda item: item[0]) for row in rows]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    rows = _row_groups(_colored_beads(array))
    if len(rows) != 4 or any(len(row) != 3 for row in rows):
        raise ValueError("could not recover one start row and three option rows")
    start = [item[2] for item in rows[0]]
    options = [[item[2] for item in row] for row in rows[1:]]

    xs = [item[0] for item in rows[0]]
    top_bottom = max(item[1] for item in rows[0]) + 24
    option_top = min(item[1] for item in rows[1]) - 30
    dark = np.max(array, axis=2) < 105
    xlo, xhi = int(min(xs) - 4), int(max(xs) + 5)
    ylo, yhi = int(top_bottom), int(option_top)
    if not (0 <= xlo < xhi <= array.shape[1] and 0 <= ylo < yhi <= array.shape[0]):
        raise ValueError("invalid discovered track bounds")
    row_ink = dark[ylo:yhi, xlo:xhi].sum(axis=1)
    threshold = max(35, int((xhi - xlo) * 0.22))
    active = np.flatnonzero(row_ink > threshold)
    groups = []
    for value in active:
        if not groups or value > groups[-1][-1] + 1:
            groups.append([int(value)])
        else:
            groups[-1].append(int(value))
    if len(groups) != 5:
        raise ValueError("could not recover five swap gates")

    gates = []
    midpoints = [(xs[0] + xs[1]) / 2, (xs[1] + xs[2]) / 2]
    for group in groups:
        y = ylo + group[len(group) // 2]
        samples = []
        for midpoint in midpoints:
            cx = int(round(midpoint))
            samples.append(int(dark[max(0, y - 3):y + 4, cx - 4:cx + 5].sum()))
        if max(samples) < 20 or samples[0] == samples[1]:
            raise ValueError("ambiguous swap bar")
        gates.append(0 if samples[0] > samples[1] else 1)

    final = list(start)
    for gate in gates:
        final[gate], final[gate + 1] = final[gate + 1], final[gate]
    matches = [index for index, option in enumerate(options) if option == final]
    if len(matches) != 1:
        raise ValueError("computed order does not uniquely match an option")
    return "ABC"[matches[0]]
