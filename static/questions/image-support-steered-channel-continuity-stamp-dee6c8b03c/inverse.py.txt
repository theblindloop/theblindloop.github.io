"""Independent image-only inverse for the compact channel stamp."""

from __future__ import annotations

from collections import deque

import numpy as np

NAVY = np.asarray((27, 42, 58), dtype=np.int16)
CREAM = np.asarray((242, 226, 170), dtype=np.int16)
GREEN = np.asarray((35, 178, 92), dtype=np.int16)
ORANGE = np.asarray((238, 130, 36), dtype=np.int16)


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    # int32 avoids overflow when squared color differences approach 255**2.
    delta = array.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _centroid(mask: np.ndarray, name: str) -> tuple[float, float]:
    points = np.argwhere(mask)
    if not 80 <= len(points) <= 180:
        raise ValueError(f"missing or corrupted {name} endpoint")
    return float(points[:, 0].mean()), float(points[:, 1].mean())


def _components(mask: np.ndarray) -> tuple[np.ndarray, dict[int, int]]:
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    sizes: dict[int, int] = {}
    label = 0
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if labels[y0, x0]:
            continue
        label += 1
        queue = deque([(y0, x0)])
        labels[y0, x0] = label
        count = 0
        while queue:
            y, x = queue.popleft()
            count += 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = label
                    queue.append((ny, nx))
        sizes[label] = count
    return labels, sizes


def _nearby_labels(
    labels: np.ndarray, sizes: dict[int, int], center: tuple[float, float], radius: float = 13.0
) -> set[int]:
    cy, cx = center
    yy, xx = np.indices(labels.shape)
    near = (yy - cy) ** 2 + (xx - cx) ** 2 <= radius * radius
    return {
        int(value)
        for value in np.unique(labels[near])
        if int(value) > 0 and sizes.get(int(value), 0) >= 35
    }


def decision_from_image(image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3:
        raise ValueError("expected RGB raster")

    navy = _near(array, NAVY)
    navy_points = np.argwhere(navy)
    if len(navy_points) < 4300:
        raise ValueError("compact navy stamp is missing or materially erased")
    height = int(navy_points[:, 0].max() - navy_points[:, 0].min() + 1)
    width = int(navy_points[:, 1].max() - navy_points[:, 1].min() + 1)
    if not (78 <= height <= 86 and 78 <= width <= 86):
        raise ValueError("navy stamp boundary is corrupted")

    green_center = _centroid(_near(array, GREEN), "green")
    orange_center = _centroid(_near(array, ORANGE), "orange")
    cream = _near(array, CREAM)
    labels, sizes = _components(cream)
    green_labels = _nearby_labels(labels, sizes, green_center)
    orange_labels = _nearby_labels(labels, sizes, orange_center)
    if not green_labels or not orange_labels:
        raise ValueError("endpoint does not meet a recoverable cream channel")
    return "connected" if green_labels & orange_labels else "broken"
