"""Deterministic compact trace-continuity world."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

CANVAS = (512, 384)
BACKGROUND = (250, 250, 248)
NAVY = (27, 42, 58)
CREAM = (242, 226, 170)
GREEN = (35, 178, 92)
ORANGE = (238, 130, 36)
CYAN = (46, 181, 200)
MAGENTA = (190, 72, 158)
FLECK = (211, 215, 216)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    size = 84
    connected = (int(seed) % 2) == 0
    cx = rng.randint(92, CANVAS[0] - 92)
    cy = rng.randint(82, CANVAS[1] - 82)
    rotation = rng.choice((0, 90, 180, 270))
    trace_width = rng.choice((8, 9, 10))
    gap_size = rng.choice((12, 14, 16))
    path_points = [[14, 42], [28, 42], [28, 22], [56, 22], [56, 42], [70, 42]]
    flecks = []
    while len(flecks) < 8:
        x = rng.randint(22, CANVAS[0] - 23)
        y = rng.randint(22, CANVAS[1] - 23)
        if abs(x - cx) > 78 or abs(y - cy) > 78:
            flecks.append([x, y])
    return {
        "connected": connected,
        "cx": cx,
        "cy": cy,
        "rotation": rotation,
        "stamp_size": size,
        "trace_width": trace_width,
        "gap_size": gap_size,
        "path_points": path_points,
        "flecks": flecks,
    }


def _local_stamp(scene: dict) -> Image.Image:
    size = int(scene["stamp_size"])
    tile = Image.new("RGB", (size, size), NAVY)
    draw = ImageDraw.Draw(tile)
    width = int(scene["trace_width"])
    points = [tuple(int(v) for v in pair) for pair in scene["path_points"]]
    # Canonicalize an undirected centerline representation before rasterization.
    if points[0] > points[-1]:
        points.reverse()
    draw.line(points, fill=CREAM, width=width, joint="curve")

    # A second cream trace is a local, nondiagnostic tracing distractor.
    lower = [(16, 68), (30, 56), (54, 56), (68, 68)]
    draw.line(lower, fill=CREAM, width=width, joint="curve")

    if not bool(scene["connected"]):
        gap = int(scene["gap_size"])
        x = size // 2
        y = 22
        draw.rectangle(
            (x - gap // 2, y - width // 2 - 2, x + gap // 2, y + width // 2 + 2),
            fill=NAVY,
        )

    radius = 6
    endpoints = ((14, 42, GREEN), (70, 42, ORANGE), (16, 68, CYAN), (68, 68, MAGENTA))
    for x, y, color in endpoints:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    # A thin light rim makes the bounded local inset easy to locate visually.
    draw.rectangle((0, 0, size - 1, size - 1), outline=(92, 108, 120), width=2)
    return tile


def render(scene: dict) -> Image.Image:
    canvas = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    for x, y in scene["flecks"]:
        draw.ellipse((int(x) - 2, int(y) - 2, int(x) + 2, int(y) + 2), fill=FLECK)

    tile = _local_stamp(scene)
    rotation = int(scene["rotation"])
    if rotation:
        tile = tile.rotate(rotation, resample=Image.Resampling.NEAREST, expand=False)
    size = int(scene["stamp_size"])
    left = int(scene["cx"]) - size // 2
    top = int(scene["cy"]) - size // 2
    canvas.paste(tile, (left, top))
    return canvas


def analytic_gold(scene: dict) -> str:
    return "connected" if bool(scene["connected"]) else "broken"


def margin(scene: dict) -> float:
    if bool(scene["connected"]):
        return float(scene["trace_width"])
    return float(scene["gap_size"])


def is_quarantined(scene: dict) -> bool:
    half = int(scene["stamp_size"]) // 2
    boundary_clearance = min(
        int(scene["cx"]) - half,
        CANVAS[0] - (int(scene["cx"]) + half),
        int(scene["cy"]) - half,
        CANVAS[1] - (int(scene["cy"]) + half),
    )
    return (
        int(scene["trace_width"]) < 6
        or (not bool(scene["connected"]) and int(scene["gap_size"]) < 6)
        or boundary_clearance < 12
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["path_points"] = list(reversed(twin["path_points"]))
    return [("reverse_undirected_centerline", twin)]
