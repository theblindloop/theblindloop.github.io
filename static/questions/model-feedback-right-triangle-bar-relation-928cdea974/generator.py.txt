"""Renderer and analytic arm for a visible square-sum relation.

Each panel contains three exact-color horizontal bars.  The red and blue
lengths are the two operands on the left of the identity and the gold length
is the third operand.  All label-bearing fields are rendered directly.
"""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1400
HEIGHT = 760

BACKGROUND = (247, 249, 252)
PANEL = (255, 255, 255)
FRAME = (102, 116, 134)
INK = (34, 43, 56)
GUIDE = (218, 225, 233)
RED = (208, 60, 60)
BLUE = (49, 101, 194)
GOLD = (220, 162, 40)

PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
PANEL_BOXES = (
    (35, 75, 685, 345),
    (715, 75, 1365, 345),
    (35, 415, 685, 685),
    (715, 415, 1365, 685),
)
BAR_X_OFFSET = 150
BAR_Y_OFFSETS = (100, 160, 220)
BAR_HEIGHT = 28
BAR_CAPACITY = 450
MIN_BAR = 78
MAX_BAR = 420
TARGET_TOLERANCE = 0.000001
DECOY_CLEARANCE = 0.14
TARGET_HEADROOM = 0.000001

_BAR_NAMES = ("red", "blue", "gold")
_BAR_COLORS = {"red": RED, "blue": BLUE, "gold": GOLD}

# Primitive triples are intentionally not limited to the familiar 3:4:5
# family.  Integer scales make the target exact in the raster while leaving
# its apparent bar lengths unrelated to panel position or a fixed scale.
PYTHAGOREAN_TEMPLATES: tuple[tuple[int, int, int], ...] = (
    (20, 21, 29),
    (12, 35, 37),
    (9, 40, 41),
    (28, 45, 53),
    (33, 56, 65),
    (16, 63, 65),
    (48, 55, 73),
    (36, 77, 85),
)


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _residual(values: Iterable[int]) -> float:
    red, blue, gold = [int(value) for value in values]
    numerator = red * red + blue * blue - gold * gold
    denominator = max(red * red + blue * blue, gold * gold)
    if denominator <= 0:
        raise ValueError("bar lengths must be positive")
    return numerator / denominator


def _target_lengths(rng: random.Random) -> tuple[int, int, int]:
    for _ in range(5000):
        red_unit, blue_unit, gold_unit = rng.choice(PYTHAGOREAN_TEMPLATES)
        scales = [
            scale
            for scale in range(2, 16)
            if MIN_BAR <= red_unit * scale <= MAX_BAR
            and MIN_BAR <= blue_unit * scale <= MAX_BAR
            and MIN_BAR <= gold_unit * scale <= MAX_BAR
        ]
        if scales:
            scale = rng.choice(scales)
            return red_unit * scale, blue_unit * scale, gold_unit * scale
    raise RuntimeError("could not sample a readable Pythagorean triple")


def _decoy_lengths(rng: random.Random) -> tuple[int, int, int]:
    """Make a plausible but non-identity triple near the square-sum curve."""
    for _ in range(5000):
        red = rng.randint(86, 290)
        blue = rng.randint(86, 290)
        ideal = int(round(math.hypot(red, blue)))
        perturb = rng.randint(14, 34)
        gold = ideal + rng.choice((-perturb, perturb))
        if not (MIN_BAR <= gold <= MAX_BAR and gold > max(red, blue) + 18):
            continue
        residual = abs(_residual((red, blue, gold)))
        if residual >= DECOY_CLEARANCE:
            return red, blue, gold
    raise RuntimeError("could not sample a separated decoy triple")


def sample_scene(seed: int) -> dict:
    """Sample four panels with one exact square-sum match."""
    rng = random.Random(int(seed))
    target_index = rng.randrange(len(PANEL_LABELS))
    panels: list[dict[str, int]] = []
    for index in range(len(PANEL_LABELS)):
        values = _target_lengths(rng) if index == target_index else _decoy_lengths(rng)
        panels.append({"red": values[0], "blue": values[1], "gold": values[2]})
    scene = {"panels": panels}
    if is_quarantined(scene):
        raise AssertionError("sampler produced a quarantined square-sum scene")
    return scene


def _residuals(panels: Iterable[dict]) -> list[float]:
    return [_residual((panel["red"], panel["blue"], panel["gold"])) for panel in panels]


def _matches(panels: Iterable[dict]) -> list[int]:
    return [index for index, value in enumerate(_residuals(panels)) if abs(value) <= TARGET_TOLERANCE]


def render(scene: dict) -> Image.Image:
    """Render four framed panels with three common-baseline bar rows."""
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(27)
    legend_font = _font(18)
    panel_font = _font(28)
    row_font = _font(20)

    draw.text((40, 17), "Which panel obeys the square-sum rule?", fill=INK, font=title_font)
    draw.text(
        (40, 52),
        "Red and blue are the legs; gold is the hypotenuse.",
        fill=FRAME,
        font=legend_font,
    )

    for index, (box, panel) in enumerate(zip(PANEL_BOXES, scene["panels"])):
        x0, y0, x1, y1 = box
        draw.rounded_rectangle(box, radius=18, fill=PANEL, outline=FRAME, width=3)
        draw.text((x0 + 28, y0 + 20), PANEL_LABELS[index][-1], fill=INK, font=panel_font)
        bar_x = x0 + BAR_X_OFFSET
        for name, y_offset in zip(_BAR_NAMES, BAR_Y_OFFSETS):
            y = y0 + y_offset
            draw.line(
                (bar_x, y + BAR_HEIGHT + 12, bar_x + BAR_CAPACITY, y + BAR_HEIGHT + 12),
                fill=GUIDE,
                width=2,
            )
            draw.text((x0 + 52, y + 1), name[0].upper(), fill=INK, font=row_font)
            width = int(panel[name])
            draw.rectangle(
                (bar_x, y, bar_x + width - 1, y + BAR_HEIGHT - 1),
                fill=_BAR_COLORS[name],
            )
    return image


def analytic_gold(scene: dict) -> str:
    """Return the unique panel whose bar lengths satisfy Pythagoras."""
    matches = _matches(scene["panels"])
    if len(matches) != 1:
        raise ValueError(f"expected one square-sum match, found {matches}")
    return PANEL_LABELS[matches[0]]


def margin(scene: dict) -> float:
    """Return separation between the zero identity band and nearest decoy."""
    try:
        values = sorted(abs(value) for value in _residuals(scene["panels"]))
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return -1.0
    if len(values) != len(PANEL_LABELS):
        return -1.0
    return min(TARGET_TOLERANCE - values[0], values[1] - DECOY_CLEARANCE)


def is_quarantined(scene: dict) -> bool:
    """Exclude malformed, ambiguous, or finite-resolution-near-boundary scenes."""
    try:
        panels = scene["panels"]
        residuals = sorted(abs(value) for value in _residuals(panels))
        if len(panels) != len(PANEL_LABELS):
            return True
        for panel in panels:
            values = [int(panel[name]) for name in _BAR_NAMES]
            if any(value < MIN_BAR or value > MAX_BAR for value in values):
                return True
            if values[2] <= max(values[0], values[1]):
                return True
    except (KeyError, TypeError, ValueError, ZeroDivisionError):
        return True
    return not (
        residuals[0] <= TARGET_HEADROOM
        and residuals[1] >= DECOY_CLEARANCE
        and len(_matches(panels)) == 1
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """There are no hidden interchangeable fields in the bar encoding."""
    return []
