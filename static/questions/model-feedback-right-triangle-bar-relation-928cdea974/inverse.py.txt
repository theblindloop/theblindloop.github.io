"""Independent image-only inverse for the square-sum bar relation."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1400
HEIGHT = 760
PANEL_BOXES = (
    (35, 75, 685, 345),
    (715, 75, 1365, 345),
    (35, 415, 685, 685),
    (715, 415, 1365, 685),
)
PANEL_LABELS = ("panel-A", "panel-B", "panel-C", "panel-D")
RED = (208, 60, 60)
BLUE = (49, 101, 194)
GOLD = (220, 162, 40)
BAR_COLORS = (RED, BLUE, GOLD)
TARGET_TOLERANCE = 0.000001


def _bar_width(arr: np.ndarray, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> int:
    x0, y0, x1, y1 = box
    crop = arr[y0 : y1 + 1, x0 : x1 + 1]
    mask = np.all(crop == np.asarray(color, dtype=np.uint8), axis=2)
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        raise ValueError("bar color is missing or too small")
    if ys.max() - ys.min() < 20:
        raise ValueError("bar is not a readable horizontal rectangle")
    width = int(xs.max() - xs.min() + 1)
    if width < 70:
        raise ValueError("bar is too short")
    return width


def _residual(values: tuple[int, int, int]) -> float:
    red, blue, gold = values
    numerator = red * red + blue * blue - gold * gold
    denominator = max(red * red + blue * blue, gold * gold)
    return numerator / denominator


def decision_from_image(image: Image.Image) -> str:
    """Recover colored bar widths and select the unique square-sum panel."""
    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError("unexpected image size")
    arr = np.asarray(rgb, dtype=np.uint8)
    residuals = []
    for box in PANEL_BOXES:
        values = tuple(_bar_width(arr, box, color) for color in BAR_COLORS)
        residuals.append(_residual(values))
    matches = [index for index, value in enumerate(residuals) if abs(value) <= TARGET_TOLERANCE]
    if len(matches) != 1:
        raise ValueError(f"expected one square-sum match, found {matches}")
    return PANEL_LABELS[matches[0]]
