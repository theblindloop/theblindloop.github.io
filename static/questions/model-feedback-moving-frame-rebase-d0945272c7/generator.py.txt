"""Code-native renderer and analytic world for a moving coordinate frame."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1200
HEIGHT = 760
BACKGROUND = (246, 248, 251)
PANEL = (255, 255, 255)
INK = (24, 32, 48)
MUTED = (84, 96, 115)
GRID = (203, 211, 222)
AXIS = (103, 119, 143)
FRAME = (18, 157, 151)
GOLD = (232, 171, 24)

GRID_LEFT = 100
GRID_TOP = 155
CELL = 70
BOARD_CENTER = (GRID_LEFT + 3 * CELL + CELL // 2, GRID_TOP + 3 * CELL + CELL // 2)

SHIFT_VECTORS = {
    "east": (1, 0),
    "west": (-1, 0),
    "north": (0, -1),
    "south": (0, 1),
}
SHIFT_BANDS = {
    "east": (37, 112, 207),
    "west": (126, 72, 181),
    "north": (0, 148, 122),
    "south": (211, 87, 48),
}
TURN_BANDS = {
    "cw": (42, 137, 92),
    "ccw": (177, 77, 126),
}
QUADRANTS = ("northwest", "northeast", "southwest", "southeast")
POINT_CELLS = [
    (x, y)
    for x in range(-3, 4)
    for y in range(-3, 4)
    if x != 0 and y != 0
]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size=size)
    except OSError:
        return ImageFont.load_default()


def _validate_scene(scene: dict[str, Any]) -> tuple[tuple[int, int], str, str]:
    point = scene["point_cell"]
    if not isinstance(point, (list, tuple)) or len(point) != 2:
        raise ValueError("point_cell must contain two coordinates")
    px, py = int(point[0]), int(point[1])
    if (px, py) not in POINT_CELLS:
        raise ValueError("point_cell must be off both initial frame axes")
    shift = str(scene["frame_shift"])
    turn = str(scene["frame_turn"])
    if shift not in SHIFT_VECTORS:
        raise ValueError("frame_shift must be a compass direction")
    if turn not in TURN_BANDS:
        raise ValueError("frame_turn must be cw or ccw")
    return (px, py), shift, turn


def _relative_in_final_frame(
    point: tuple[int, int], shift: str, turn: str
) -> tuple[int, int]:
    """Return fixed-point coordinates in the frame after shift then turn."""

    ox, oy = SHIFT_VECTORS[shift]
    dx, dy = point[0] - ox, point[1] - oy
    # Image coordinates have +x right and +y down.  A clockwise turn of the
    # frame maps local (u,v) to world (-v,u), so the inverse is (dy,-dx).
    if turn == "cw":
        return (dy, -dx)
    # A counterclockwise frame maps local (u,v) to world (v,-u).
    return (-dy, dx)


def _quadrant(relative: tuple[int, int]) -> str:
    x, y = relative
    if x == 0 or y == 0:
        raise ValueError("point lies on a final frame axis")
    if x < 0 and y < 0:
        return "northwest"
    if x > 0 and y < 0:
        return "northeast"
    if x < 0 and y > 0:
        return "southwest"
    return "southeast"


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced point cells and two visible frame motions."""

    seed = int(seed)
    rng = random.Random(seed * 1_000_003 + 43)
    wanted = QUADRANTS[(seed * 2_654_435_761 + 101) % len(QUADRANTS)]
    options = []
    for point in POINT_CELLS:
        for shift in SHIFT_VECTORS:
            for turn in TURN_BANDS:
                relative = _relative_in_final_frame(point, shift, turn)
                if relative[0] != 0 and relative[1] != 0 and _quadrant(relative) == wanted:
                    options.append((point, shift, turn))
    if not options:
        raise ValueError("sampler found no off-axis scene for requested quadrant")
    point, shift, turn = rng.choice(options)
    return {
        "point_cell": [point[0], point[1]],
        "frame_shift": shift,
        "frame_turn": turn,
    }


def _panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], radius: int = 18) -> None:
    draw.rounded_rectangle(box, radius=radius, fill=PANEL, outline=(207, 214, 224), width=2)


def _arrow_head(
    draw: ImageDraw.ImageDraw,
    tip: tuple[int, int],
    direction: tuple[int, int],
    fill: tuple[int, int, int],
    size: int = 18,
) -> None:
    dx, dy = direction
    px, py = -dy, dx
    bx, by = tip[0] - dx * size, tip[1] - dy * size
    draw.polygon(
        [tip, (bx + px * size // 2, by + py * size // 2), (bx - px * size // 2, by - py * size // 2)],
        fill=fill,
    )


def _draw_shift_icon(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: str
) -> None:
    dx, dy = SHIFT_VECTORS[direction]
    cx, cy = center
    start = (cx - dx * 72, cy - dy * 72)
    end = (cx + dx * 72, cy + dy * 72)
    draw.line((*start, *end), fill=INK, width=11)
    _arrow_head(draw, end, (dx, dy), INK, size=27)
    draw.ellipse((cx - 11, cy - 11, cx + 11, cy + 11), fill=FRAME, outline=INK, width=3)


def _draw_turn_icon(draw: ImageDraw.ImageDraw, center: tuple[int, int], direction: str) -> None:
    cx, cy = center
    box = (cx - 67, cy - 67, cx + 67, cy + 67)
    draw.arc(box, start=30, end=325, fill=INK, width=10)
    if direction == "cw":
        draw.polygon([(cx + 53, cy - 64), (cx + 82, cy - 48), (cx + 53, cy - 31)], fill=INK)
    else:
        draw.polygon([(cx - 53, cy + 64), (cx - 82, cy + 48), (cx - 53, cy + 31)], fill=INK)
    draw.ellipse((cx - 10, cy - 10, cx + 10, cy + 10), fill=INK)


def _draw_frame(draw: ImageDraw.ImageDraw) -> None:
    ox, oy = BOARD_CENTER
    x_end = (ox + 92, oy)
    y_end = (ox, oy + 92)
    draw.line((ox, oy, *x_end), fill=FRAME, width=9)
    draw.line((ox, oy, *y_end), fill=FRAME, width=9)
    _arrow_head(draw, x_end, (1, 0), FRAME, size=22)
    _arrow_head(draw, y_end, (0, 1), FRAME, size=22)
    draw.polygon(
        [(ox, oy - 14), (ox + 14, oy), (ox, oy + 14), (ox - 14, oy)],
        fill=FRAME,
        outline=INK,
    )
    draw.text((ox + 101, oy - 7), "+X", font=_font(15, True), fill=FRAME, anchor="lm")
    draw.text((ox + 8, oy + 103), "+Y", font=_font(15, True), fill=FRAME, anchor="lm")


def _cell_center(cell: tuple[int, int]) -> tuple[int, int]:
    return (BOARD_CENTER[0] + cell[0] * CELL, BOARD_CENTER[1] + cell[1] * CELL)


def render(scene: dict[str, Any]) -> Image.Image:
    point, shift, turn = _validate_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.text((38, 24), "MOVING-FRAME REBASE", font=_font(30, True), fill=INK)
    draw.text(
        (40, 63),
        "Keep the gold point fixed; move the cyan frame in order.",
        font=_font(17),
        fill=MUTED,
    )

    _panel(draw, (35, 95, 700, 700))
    draw.text((58, 112), "7 BY 7 BOARD", font=_font(19, True), fill=INK)
    draw.ellipse((470, 112, 486, 128), fill=(242, 198, 82), outline=INK, width=1)
    draw.text((495, 108), "GOLD POINT · FIXED", font=_font(14, True), fill=MUTED)
    draw.line((470, 138, 502, 138), fill=FRAME, width=8)
    draw.text((512, 126), "CYAN FRAME START", font=_font(13, True), fill=MUTED)

    grid_right = GRID_LEFT + 7 * CELL
    grid_bottom = GRID_TOP + 7 * CELL
    for index in range(8):
        x = GRID_LEFT + index * CELL
        y = GRID_TOP + index * CELL
        draw.line((x, GRID_TOP, x, grid_bottom), fill=GRID, width=2)
        draw.line((GRID_LEFT, y, grid_right, y), fill=GRID, width=2)
    draw.line((BOARD_CENTER[0], GRID_TOP, BOARD_CENTER[0], grid_bottom), fill=AXIS, width=3)
    draw.line((GRID_LEFT, BOARD_CENTER[1], grid_right, BOARD_CENTER[1]), fill=AXIS, width=3)
    draw.text((GRID_LEFT + 14, GRID_TOP + 9), "NW", font=_font(17, True), fill=MUTED)
    draw.text((grid_right - 48, GRID_TOP + 9), "NE", font=_font(17, True), fill=MUTED)
    draw.text((GRID_LEFT + 14, grid_bottom - 37), "SW", font=_font(17, True), fill=MUTED)
    draw.text((grid_right - 48, grid_bottom - 37), "SE", font=_font(17, True), fill=MUTED)
    draw.text((BOARD_CENTER[0], GRID_TOP - 14), "N", font=_font(14, True), fill=AXIS, anchor="ma")
    draw.text((BOARD_CENTER[0], grid_bottom + 10), "S", font=_font(14, True), fill=AXIS, anchor="ma")
    draw.text((GRID_LEFT - 14, BOARD_CENTER[1]), "W", font=_font(14, True), fill=AXIS, anchor="mm")
    draw.text((grid_right + 14, BOARD_CENTER[1]), "E", font=_font(14, True), fill=AXIS, anchor="mm")

    _draw_frame(draw)
    px, py = _cell_center(point)
    draw.ellipse((px - 26, py - 26, px + 26, py + 26), fill=GOLD, outline=INK, width=4)
    draw.ellipse((px - 9, py - 13, px - 2, py - 6), fill=(255, 239, 164))

    _panel(draw, (735, 95, 1165, 320))
    draw.rectangle((760, 117, 1140, 152), fill=SHIFT_BANDS[shift])
    draw.text((780, 135), "MOTION 1  ·  SHIFT FRAME", font=_font(17, True), fill=(255, 255, 255), anchor="lm")
    _draw_shift_icon(draw, (846, 242), shift)
    draw.text((955, 181), "BOARD COMPASS", font=_font(15, True), fill=INK, anchor="lm")
    draw.text((955, 218), "ONE CELL", font=_font(21, True), fill=INK, anchor="lm")
    draw.text((955, 255), shift.upper(), font=_font(25, True), fill=MUTED, anchor="lm")
    draw.text((955, 288), "from the center", font=_font(14), fill=MUTED, anchor="lm")

    _panel(draw, (735, 340, 1165, 565))
    draw.rectangle((760, 362, 1140, 397), fill=TURN_BANDS[turn])
    draw.text((780, 380), "MOTION 2  ·  TURN FRAME", font=_font(17, True), fill=(255, 255, 255), anchor="lm")
    _draw_turn_icon(draw, (846, 493), turn)
    draw.text((955, 421), "ABOUT NEW ORIGIN", font=_font(15, True), fill=INK, anchor="lm")
    draw.text((955, 461), "QUARTER-TURN", font=_font(19, True), fill=INK, anchor="lm")
    draw.text((955, 505), "CLOCKWISE" if turn == "cw" else "COUNTERCLOCKWISE", font=_font(16, True), fill=MUTED, anchor="lm")
    draw.text((955, 544), "frame axes only", font=_font(14), fill=MUTED, anchor="lm")

    _panel(draw, (735, 590, 1165, 700), radius=14)
    draw.text((760, 620), "ORDER", font=_font(15, True), fill=MUTED, anchor="lm")
    draw.text((845, 620), "1  then  2", font=_font(21, True), fill=INK, anchor="lm")
    draw.text((1010, 620), "final frame quadrant", font=_font(15), fill=MUTED, anchor="lm")
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    point, shift, turn = _validate_scene(scene)
    return _quadrant(_relative_in_final_frame(point, shift, turn))


def margin(scene: dict[str, Any]) -> float:
    point, shift, turn = _validate_scene(scene)
    x, y = _relative_in_final_frame(point, shift, turn)
    return float(min(abs(x), abs(y)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    point, shift, turn = _validate_scene(scene)
    x, y = _relative_in_final_frame(point, shift, turn)
    return x == 0 or y == 0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    _validate_scene(scene)
    return []
