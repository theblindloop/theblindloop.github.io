"""Independent image-only inverse for the moving-frame rebase world."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH = 1200
HEIGHT = 760
GOLD = np.array((232, 171, 24), dtype=np.int16)
BOARD_CENTER = (345, 400)
CELL = 70
SHIFT_BANDS = {
    "east": np.array((37, 112, 207), dtype=np.int16),
    "west": np.array((126, 72, 181), dtype=np.int16),
    "north": np.array((0, 148, 122), dtype=np.int16),
    "south": np.array((211, 87, 48), dtype=np.int16),
}
TURN_BANDS = {
    "cw": np.array((42, 137, 92), dtype=np.int16),
    "ccw": np.array((177, 77, 126), dtype=np.int16),
}
SHIFT_VECTORS = {
    "east": (1, 0),
    "west": (-1, 0),
    "north": (0, -1),
    "south": (0, 1),
}


def _near(arr: np.ndarray, color: np.ndarray, tolerance: int) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 100:
        raise ValueError("gold point is not recoverable")
    return float(xs.mean()), float(ys.mean())


def _card_value(
    arr: np.ndarray,
    bounds: tuple[int, int, int, int],
    colors: dict[str, np.ndarray],
) -> str:
    x0, y0, x1, y1 = bounds
    crop = arr[y0:y1, x0:x1]
    counts = {name: int(_near(crop, color, 8).sum()) for name, color in colors.items()}
    name, count = max(counts.items(), key=lambda item: item[1])
    other = max(value for key, value in counts.items() if key != name)
    if count < 3000 or count <= other * 2:
        raise ValueError(f"operation card is unreadable: {counts}")
    return name


def _quadrant(relative: tuple[int, int]) -> str:
    x, y = relative
    if x == 0 or y == 0:
        raise ValueError("point lies on a final frame axis")
    if x < 0 and y < 0:
        return "northwest"
    if x > 0 and y < 0:
        return "northeast"
    if x < 0 and y > 0:
        return "southwest"
    return "southeast"


def decision_from_image(image: Image.Image) -> str:
    """Recover the final-frame quadrant from RGB pixels only."""

    rgb = image.convert("RGB")
    if rgb.size != (WIDTH, HEIGHT):
        raise ValueError(f"unexpected canvas size {rgb.size}")
    arr = np.asarray(rgb, dtype=np.int16)
    gx, gy = _centroid(_near(arr, GOLD, 12))
    point_x = int(round((gx - BOARD_CENTER[0]) / CELL))
    point_y = int(round((gy - BOARD_CENTER[1]) / CELL))
    if not (-3 <= point_x <= 3 and not point_x == 0):
        raise ValueError(f"gold point x coordinate is not recoverable: {point_x}")
    if not (-3 <= point_y <= 3 and not point_y == 0):
        raise ValueError(f"gold point y coordinate is not recoverable: {point_y}")
    shift = _card_value(arr, (760, 117, 1140, 153), SHIFT_BANDS)
    turn = _card_value(arr, (760, 362, 1140, 398), TURN_BANDS)
    ox, oy = SHIFT_VECTORS[shift]
    dx, dy = point_x - ox, point_y - oy
    if turn == "cw":
        relative = (dy, -dx)
    else:
        relative = (-dy, dx)
    return _quadrant(relative)
