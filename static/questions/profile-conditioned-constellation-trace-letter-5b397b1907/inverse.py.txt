"""Independent pixel-only inverse arm for the constellation-trace-letter world.

The oracle receives only the final rendered PNG and the public question
contract. It reconstructs the dark dot constellation from the raster alone,
takes every recovered dot centroid as an equal-mass mark, normalizes the marks
to a canonical box, and fits each of the four letter skeletons to recover which
capital letter (T, L, V, or X) the whole constellation traces. It imports only
numpy/PIL and never receives the latent scene or analytic gold.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 512
HEIGHT = 512
INK_THRESHOLD = 128

LETTERS = ("T", "L", "V", "X")

SEGMENTS = {
    "T": [((0.10, 0.18), (0.90, 0.18)), ((0.50, 0.18), (0.50, 0.88))],
    "L": [((0.12, 0.15), (0.12, 0.88)), ((0.12, 0.88), (0.88, 0.88))],
    "V": [((0.14, 0.18), (0.50, 0.88)), ((0.86, 0.18), (0.50, 0.88))],
    "X": [((0.10, 0.10), (0.90, 0.90)), ((0.90, 0.10), (0.10, 0.90))],
}

_SKEL_STEPS = 90


def _skeleton(letter: str) -> np.ndarray:
    points = []
    for (a, b) in SEGMENTS[letter]:
        for i in range(_SKEL_STEPS + 1):
            t = i / _SKEL_STEPS
            points.append((a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t))
    return np.asarray(points, dtype=float)


_SKEL = {letter: _skeleton(letter) for letter in LETTERS}


def _dark_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB")).astype(int)
    return arr[..., 0] < INK_THRESHOLD


def _centroids(mask: np.ndarray) -> list[tuple[float, float]]:
    """Recover each connected dark dot's centroid by BFS flood fill."""
    h, w = mask.shape
    visited = np.zeros((h, w), dtype=bool)
    centroids: list[tuple[float, float]] = []
    for y in range(h):
        for x in range(w):
            if mask[y, x] and not visited[y, x]:
                queue = deque([(y, x)])
                visited[y, x] = True
                sx = 0.0
                sy = 0.0
                count = 0
                while queue:
                    cy, cx = queue.popleft()
                    sx += cx
                    sy += cy
                    count += 1
                    for ny in (cy - 1, cy, cy + 1):
                        for nx in (cx - 1, cx, cx + 1):
                            if 0 <= ny < h and 0 <= nx < w:
                                if mask[ny, nx] and not visited[ny, nx]:
                                    visited[ny, nx] = True
                                    queue.append((ny, nx))
                if count > 0:
                    centroids.append((sx / count, sy / count))
    return centroids


def _shape_energy(pts: np.ndarray, letter: str) -> float:
    if pts.shape[0] == 0:
        return 1e9
    sk = _SKEL[letter]
    best = np.empty(pts.shape[0], dtype=float)
    for i, p in enumerate(pts):
        best[i] = float(np.min(np.sum((sk - p) ** 2, axis=1)) ** 0.5)
    return float(np.mean(best))


def _normalize(pts: np.ndarray) -> np.ndarray:
    x0, x1 = pts[:, 0].min(), pts[:, 0].max()
    y0, y1 = pts[:, 1].min(), pts[:, 1].max()
    w = x1 - x0
    h = y1 - y0
    if w < 1e-6 or h < 1e-6:
        return pts * 0.0
    return (pts - np.array([x0, y0])) / np.array([w, h])


def decision_from_image(image: Image.Image) -> str:
    """Image-only inverse: the capital letter traced by the dot constellation."""
    mask = _dark_mask(image)
    centroids = _centroids(mask)
    if len(centroids) == 0:
        return "L"
    pts = np.asarray(centroids, dtype=float)
    pts = _normalize(pts)
    return min(LETTERS, key=lambda letter: _shape_energy(pts, letter))
