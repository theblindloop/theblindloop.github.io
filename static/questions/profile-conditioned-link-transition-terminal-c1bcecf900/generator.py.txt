"""Deterministic renderer for link-color transition tracing."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 480
BACKGROUND = (250, 248, 243)
INK = (55, 58, 63)
GRAY = (150, 154, 158)
GREEN = (42, 166, 89)
LINK_COLORS = {"cyan": (25, 154, 191), "magenta": (201, 56, 142)}
TERMINAL_COLORS = {
    "orange": (235, 135, 38),
    "purple": (125, 78, 190),
    "blue": (48, 105, 207),
}


def _transition_count(values: list[str]) -> int:
    return sum(a != b for a, b in zip(values, values[1:]))


def _sequences_by_count() -> dict[int, list[list[str]]]:
    out = {1: [], 3: [], 5: []}
    for cyan_positions in itertools.combinations(range(6), 3):
        seq = ["magenta"] * 6
        for pos in cyan_positions:
            seq[pos] = "cyan"
        count = _transition_count(seq)
        if count in out:
            out[count].append(seq)
    return out


SEQUENCES = _sequences_by_count()


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_row = int(seed) % 3
    transition_layout = [1, 5]
    rng.shuffle(transition_layout)
    transition_layout.insert(target_row, 3)

    terminal_names = ["orange", "purple", "blue"]
    # Cycling before a seeded rotation guarantees all answer identities in any
    # run of several consecutive integer seeds while preserving randomness.
    shift = int(seed) % 3
    terminal_names = terminal_names[shift:] + terminal_names[:shift]
    if (int(seed) // 3) % 2:
        terminal_names[1], terminal_names[2] = terminal_names[2], terminal_names[1]

    routes = []
    base_ys = [105, 240, 375]
    for row, base_y in enumerate(base_ys):
        xs = [54, 142, 230, 318, 406, 494, 582]
        ys = [base_y + rng.randint(-15, 15) for _ in xs]
        points = [[x, y] for x, y in zip(xs, ys)]
        count = transition_layout[row]
        choices = SEQUENCES[count]
        colors = list(choices[rng.randrange(len(choices))])
        # Global palette exchange is a rendered nuisance, not a label cue.
        if rng.randrange(2):
            colors = ["magenta" if c == "cyan" else "cyan" for c in colors]
        routes.append({"points": points, "link_colors": colors})
    return {"routes": routes, "terminal_colors": terminal_names}


def render(scene: dict) -> Image.Image:
    routes = scene["routes"]
    terminal_colors = scene["terminal_colors"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for route in routes:
        points = [tuple(map(int, point)) for point in route["points"]]
        for index, color_name in enumerate(route["link_colors"]):
            draw.line([points[index], points[index + 1]], fill=LINK_COLORS[color_name], width=12)

    for row, route in enumerate(routes):
        points = [tuple(map(int, point)) for point in route["points"]]
        # Interior nodes make the sequence of links explicit and cover joins.
        for x, y in points[1:-1]:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=GRAY, outline=INK, width=2)
        sx, sy = points[0]
        draw.ellipse((sx - 15, sy - 15, sx + 15, sy + 15), fill=GREEN, outline=INK, width=3)
        tx, ty = points[-1]
        terminal_rgb = TERMINAL_COLORS[terminal_colors[row]]
        draw.rectangle((tx - 15, ty - 15, tx + 15, ty + 15), fill=terminal_rgb, outline=INK, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    routes = scene["routes"]
    terminal_colors = scene["terminal_colors"]
    matches = [
        index
        for index, route in enumerate(routes)
        if _transition_count(route["link_colors"]) == 3
    ]
    if len(matches) != 1:
        raise ValueError("scene does not have one three-transition route")
    return terminal_colors[matches[0]]


def margin(scene: dict) -> float:
    counts = [_transition_count(route["link_colors"]) for route in scene["routes"]]
    targets = [count for count in counts if count == 3]
    if len(targets) != 1:
        return 0.0
    return float(min(abs(count - 3) for count in counts if count != 3))


def is_quarantined(scene: dict) -> bool:
    routes = scene["routes"]
    if len(routes) != 3 or any(len(route.get("points", [])) != 7 for route in routes):
        return True
    if any(len(route.get("link_colors", [])) != 6 for route in routes):
        return True
    if any(sorted(route["link_colors"]) != ["cyan"] * 3 + ["magenta"] * 3 for route in routes):
        return True
    counts = [_transition_count(route["link_colors"]) for route in routes]
    if sorted(counts) != [1, 3, 5]:
        return True
    centers = [sum(point[1] for point in route["points"]) / 7 for route in routes]
    return min(b - a for a, b in zip(sorted(centers), sorted(centers)[1:])) < 90


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
