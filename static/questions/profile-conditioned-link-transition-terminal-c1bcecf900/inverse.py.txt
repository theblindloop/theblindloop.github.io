"""Independent image-only inverse for link-color transition tracing."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

GREEN = (42, 166, 89)
GRAY = (150, 154, 158)
CYAN = (25, 154, 191)
MAGENTA = (201, 56, 142)
TERMINALS = {
    "orange": (235, 135, 38),
    "purple": (125, 78, 190),
    "blue": (48, 105, 207),
}


def _components(mask: np.ndarray, minimum: int = 20) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= minimum:
            ys = np.fromiter((p[0] for p in pixels), dtype=float)
            xs = np.fromiter((p[1] for p in pixels), dtype=float)
            result.append((float(xs.mean()), float(ys.mean()), len(pixels)))
    return result


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=2)


def _cluster_rows(markers: list[tuple[float, float]]) -> list[list[tuple[float, float]]]:
    if len(markers) != 21:
        raise ValueError(f"expected 21 route markers, found {len(markers)}")
    ordered = sorted(markers, key=lambda p: p[1])
    rows = [ordered[index:index + 7] for index in (0, 7, 14)]
    for row in rows:
        row.sort(key=lambda p: p[0])
        if max(p[1] for p in row) - min(p[1] for p in row) > 45:
            raise ValueError("row clustering is ambiguous")
    return rows


def _link_color(arr: np.ndarray, start: tuple[float, float], end: tuple[float, float]) -> str:
    cyan = 0
    magenta = 0
    for fraction in np.linspace(0.22, 0.78, 25):
        x = int(round(start[0] + fraction * (end[0] - start[0])))
        y = int(round(start[1] + fraction * (end[1] - start[1])))
        patch = arr[max(0, y - 3):y + 4, max(0, x - 3):x + 4]
        cyan += int(np.all(patch == np.asarray(CYAN, dtype=np.uint8), axis=2).sum())
        magenta += int(np.all(patch == np.asarray(MAGENTA, dtype=np.uint8), axis=2).sum())
    if max(cyan, magenta) < 100 or abs(cyan - magenta) < 100:
        raise ValueError("link color is missing or ambiguous")
    return "cyan" if cyan > magenta else "magenta"


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    green_components = _components(_mask(arr, GREEN), minimum=100)
    gray_components = _components(_mask(arr, GRAY), minimum=80)
    terminal_positions: dict[str, tuple[float, float]] = {}
    for name, rgb in TERMINALS.items():
        pieces = _components(_mask(arr, rgb), minimum=100)
        if len(pieces) != 1:
            raise ValueError(f"expected one {name} terminal")
        terminal_positions[name] = (pieces[0][0], pieces[0][1])
    if len(green_components) != 3 or len(gray_components) != 15:
        raise ValueError("start or interior node markers are missing")
    green_positions = [(item[0], item[1]) for item in green_components]
    markers = green_positions + [(item[0], item[1]) for item in gray_components] + list(terminal_positions.values())
    rows = _cluster_rows(markers)

    found: list[str] = []
    for row in rows:
        sequence = [_link_color(arr, a, b) for a, b in zip(row, row[1:])]
        transitions = sum(a != b for a, b in zip(sequence, sequence[1:]))
        if transitions == 3:
            terminal = row[-1]
            name = min(terminal_positions, key=lambda key: (terminal_positions[key][0] - terminal[0]) ** 2 + (terminal_positions[key][1] - terminal[1]) ** 2)
            if (terminal_positions[name][0] - terminal[0]) ** 2 + (terminal_positions[name][1] - terminal[1]) ** 2 > 9:
                raise ValueError("target terminal binding is ambiguous")
            found.append(name)
    if len(found) != 1:
        raise ValueError(f"expected one three-transition route, found {len(found)}")
    return found[0]
