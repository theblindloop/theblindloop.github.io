"""Deterministic renderer for a one-sided fold transported across frames."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

CANVAS = (1100, 760)
N = 7
KEY_CELL, KEY_Y = 26, 62
KEY_X = (28, 240, 646, 858)
STAMP_X, STAMP_Y, STAMP_CELL = 459, 278, 26
CANDIDATE_X = (35, 459, 883)
CANDIDATE_Y, CANDIDATE_CELL = 530, 26
ANSWERS = ("amber", "cyan", "violet")
LANDMARKS = ("red", "blue", "green")
FRAMES = ("flip_lr", "flip_ud", "transpose", "anti")
ALL_FRAMES = ("identity", "rot90", "rot180", "rot270") + FRAMES
FOLDS = tuple((kind, index, side) for kind in ("h", "v")
              for index in range(1, N - 1) for side in (-1, 1))

RGB = {
    "background": (255, 255, 255), "paper": (247, 249, 251),
    "grid": (132, 141, 150), "ink": (29, 34, 40), "arrow": (71, 79, 91),
    "red": (214, 65, 70), "blue": (35, 113, 190), "green": (39, 151, 94),
    "gold": (238, 178, 24), "amber": (226, 151, 28),
    "cyan": (20, 157, 181), "violet": (137, 78, 191),
}


def _inside(point: tuple[int, int]) -> bool:
    return 0 <= point[0] < N and 0 <= point[1] < N


def _frame(point: tuple[int, int], name: str) -> tuple[int, int]:
    r, c = point
    if name == "identity": return r, c
    if name == "rot90": return c, N - 1 - r
    if name == "rot180": return N - 1 - r, N - 1 - c
    if name == "rot270": return N - 1 - c, r
    if name == "flip_lr": return r, N - 1 - c
    if name == "flip_ud": return N - 1 - r, c
    if name == "transpose": return c, r
    if name == "anti": return N - 1 - c, N - 1 - r
    raise ValueError(f"unknown frame {name}")


def _inverse_frame(name: str) -> str:
    return {"identity": "identity", "rot90": "rot270", "rot180": "rot180",
            "rot270": "rot90", "flip_lr": "flip_lr", "flip_ud": "flip_ud",
            "transpose": "transpose", "anti": "anti"}[name]


def _reflect(point: tuple[int, int], axis_kind: str, axis_index: int) -> tuple[int, int]:
    r, c = point
    if axis_kind == "h":
        return 2 * axis_index - r, c
    if axis_kind == "v":
        return r, 2 * axis_index - c
    raise ValueError(f"unknown axis {axis_kind}")


def _fold(point: tuple[int, int], axis_kind: str, axis_index: int,
          moving_side: int) -> tuple[int, int]:
    r, c = point
    coordinate = r if axis_kind == "h" else c
    return _reflect(point, axis_kind, axis_index) if moving_side * (coordinate - axis_index) > 0 else point


def _infer_fold(source: dict[str, list[int]], target: dict[str, list[int]]) -> tuple[str, int, int]:
    matches = [spec for spec in FOLDS if all(
        _fold(tuple(source[color]), *spec) == tuple(target[color]) for color in LANDMARKS)]
    if len(matches) != 1:
        raise ValueError("fold demonstration is ambiguous")
    return matches[0]


def _infer_frame(source: dict[str, list[int]], target: dict[str, list[int]]) -> str:
    matches = [name for name in ALL_FRAMES if all(
        _frame(tuple(source[color]), name) == tuple(target[color]) for color in LANDMARKS)]
    if len(matches) != 1:
        raise ValueError("frame demonstration is ambiguous")
    return matches[0]


def _map_set(points: set[tuple[int, int]], fn) -> set[tuple[int, int]]:
    return {fn(point) for point in points}


def _sets(scene: dict) -> tuple[set[tuple[int, int]], set[tuple[int, int]], set[tuple[int, int]]]:
    axis_kind, axis_index, moving_side = _infer_fold(scene["move_source"], scene["move_target"])
    frame = _infer_frame(scene["frame_source"], scene["frame_target"])
    inverse = _inverse_frame(frame)
    stamp = {tuple(point) for point in scene["stamp"]}

    correct = _map_set(stamp, lambda point: _frame(
        _fold(_frame(point, inverse), axis_kind, axis_index, moving_side), frame))
    whole_mirror = _map_set(stamp, lambda point: _frame(
        _reflect(_frame(point, inverse), axis_kind, axis_index), frame))
    opposite_fold = _map_set(stamp, lambda point: _frame(
        _fold(_frame(point, inverse), axis_kind, axis_index, -moving_side), frame))
    return correct, whole_mirror, opposite_fold


def _random_frame_key(rng: random.Random, name: str) -> tuple[dict[str, list[int]], dict[str, list[int]]]:
    while True:
        cells = rng.sample([(r, c) for r in range(N) for c in range(N)], 3)
        source = {color: list(cell) for color, cell in zip(LANDMARKS, cells)}
        target = {color: list(_frame(cell, name)) for color, cell in zip(LANDMARKS, cells)}
        try:
            if _infer_frame(source, target) == name:
                return source, target
        except ValueError:
            pass


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0xF01D51DE) & 0xFFFFFFFFFFFFFFFF)
    while True:
        axis_kind = rng.choice(("h", "v"))
        axis_index = rng.choice((1, 2, 4, 5))
        moving_side = rng.choice((-1, 1))
        frame = rng.choice(FRAMES)
        all_cells = [(r, c) for r in range(N) for c in range(N)]
        valid_band = [point for point in all_cells
                      if _inside(_reflect(point, axis_kind, axis_index))]
        moving = [point for point in valid_band if moving_side * ((point[0] if axis_kind == "h" else point[1]) - axis_index) > 0]
        stationary = [point for point in valid_band if moving_side * ((point[0] if axis_kind == "h" else point[1]) - axis_index) < 0]
        crease = [point for point in valid_band if (point[0] if axis_kind == "h" else point[1]) == axis_index]
        move_cells = rng.sample(moving, 2) + rng.sample(stationary, 1)
        rng.shuffle(move_cells)
        move_source = {color: list(cell) for color, cell in zip(LANDMARKS, move_cells)}
        move_target = {color: list(_fold(cell, axis_kind, axis_index, moving_side))
                       for color, cell in zip(LANDMARKS, move_cells)}
        if len({tuple(point) for point in move_target.values()}) != len(LANDMARKS):
            continue
        try:
            if _infer_fold(move_source, move_target) != (axis_kind, axis_index, moving_side):
                continue
        except ValueError:
            continue
        frame_source, frame_target = _random_frame_key(rng, frame)
        base_stamp = set(rng.sample(moving, 2) + rng.sample(stationary, 2) + rng.sample(crease, 1))
        stamp = {_frame(point, frame) for point in base_stamp}
        provisional = {
            "axis_kind": axis_kind, "axis_index": axis_index, "moving_side": moving_side,
            "move_source": move_source, "move_target": move_target,
            "frame_source": frame_source, "frame_target": frame_target,
            "stamp": sorted(map(list, stamp)),
        }
        correct, whole_mirror, opposite_fold = _sets(provisional)
        patterns = (correct, whole_mirror, opposite_fold)
        if len({frozenset(pattern) for pattern in patterns}) == 3 and all(len(pattern) == 5 for pattern in patterns):
            break

    target_slot = int(seed) % 3
    distractors = [whole_mirror, opposite_fold]
    rng.shuffle(distractors)
    arranged = list(distractors)
    arranged.insert(target_slot, correct)
    answer_color = ANSWERS[(int(seed) // 3) % 3]
    other_colors = [color for color in ANSWERS if color != answer_color]
    rng.shuffle(other_colors)
    tag_colors = [None, None, None]
    tag_colors[target_slot] = answer_color
    fill = iter(other_colors)
    for index in range(3):
        if tag_colors[index] is None:
            tag_colors[index] = next(fill)
    provisional["candidates"] = [sorted(map(list, pattern)) for pattern in arranged]
    provisional["tag_colors"] = [str(color) for color in tag_colors]
    return provisional


def _grid(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int,
          border: tuple[int, int, int], width: int = 4) -> None:
    size = N * cell
    draw.rectangle((x, y, x + size, y + size), fill=RGB["paper"], outline=border, width=width)
    for k in range(1, N):
        draw.line((x + k * cell, y, x + k * cell, y + size), fill=RGB["grid"], width=2)
        draw.line((x, y + k * cell, x + size, y + k * cell), fill=RGB["grid"], width=2)


def _dot(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, color: tuple[int, int, int]) -> None:
    cx, cy = x + cell // 2, y + cell // 2
    draw.ellipse((cx - 8, cy - 8, cx + 8, cy + 8), fill=color)


def _anchor(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    cx, cy = x + cell // 2, y + cell // 2
    draw.polygon(((cx, cy - 10), (cx + 10, cy), (cx, cy + 10), (cx - 10, cy)),
                 fill=RGB["gold"], outline=RGB["ink"])


def _cell_mark(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int) -> None:
    draw.rounded_rectangle((x + 7, y + 7, x + cell - 7, y + cell - 7), radius=3, fill=RGB["ink"])


def _arrow(draw: ImageDraw.ImageDraw, x1: int, x2: int, y: int) -> None:
    draw.line((x1, y, x2, y), fill=RGB["arrow"], width=6)
    draw.polygon(((x2, y), (x2 - 14, y - 10), (x2 - 14, y + 10)), fill=RGB["arrow"])


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, RGB["background"])
    draw = ImageDraw.Draw(image)
    move_source = scene["move_source"]
    move_target = scene["move_target"]
    frame_source = scene["frame_source"]
    frame_target = scene["frame_target"]
    draw.rounded_rectangle((103, 25, 165, 35), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((737, 20, 787, 29), radius=3, fill=RGB["ink"])
    draw.rounded_rectangle((758, 35, 808, 44), radius=3, fill=RGB["ink"])
    for x in KEY_X:
        _grid(draw, x, KEY_Y, KEY_CELL, RGB["ink"])
    for key_index, key in enumerate((move_source, move_target, frame_source, frame_target)):
        x = KEY_X[key_index]
        for color, (r, c) in key.items():
            _dot(draw, x + int(c) * KEY_CELL, KEY_Y + int(r) * KEY_CELL, KEY_CELL, RGB[color])
    _arrow(draw, KEY_X[0] + N * KEY_CELL + 12, KEY_X[1] - 12, KEY_Y + N * KEY_CELL // 2)
    _arrow(draw, KEY_X[2] + N * KEY_CELL + 12, KEY_X[3] - 12, KEY_Y + N * KEY_CELL // 2)

    _grid(draw, STAMP_X, STAMP_Y, STAMP_CELL, RGB["ink"])
    for r, c in scene["stamp"]:
        _cell_mark(draw, STAMP_X + int(c) * STAMP_CELL, STAMP_Y + int(r) * STAMP_CELL, STAMP_CELL)
    # Three distinct bars below the stamp reiterate FRAME^-1, FOLD, FRAME.
    draw.rounded_rectangle((395, 482, 433, 489), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((395, 494, 433, 501), radius=2, fill=RGB["ink"])
    _arrow(draw, 382, 337, 492)
    draw.rounded_rectangle((472, 488, 520, 496), radius=2, fill=RGB["ink"])
    _arrow(draw, 533, 578, 492)
    draw.rounded_rectangle((590, 482, 628, 489), radius=2, fill=RGB["ink"])
    draw.rounded_rectangle((590, 494, 628, 501), radius=2, fill=RGB["ink"])

    for slot, x in enumerate(CANDIDATE_X):
        color = str(scene["tag_colors"][slot])
        _grid(draw, x, CANDIDATE_Y, CANDIDATE_CELL, RGB[color], width=8)
        for r, c in scene["candidates"][slot]:
            _cell_mark(draw, x + int(c) * CANDIDATE_CELL, CANDIDATE_Y + int(r) * CANDIDATE_CELL, CANDIDATE_CELL)
    return image


def analytic_gold(scene: dict) -> str:
    target = _sets(scene)[0]
    matches = [index for index, pattern in enumerate(scene["candidates"])
               if {tuple(point) for point in pattern} == target]
    if len(matches) != 1:
        raise ValueError("no unique transported one-side fold result")
    return str(scene["tag_colors"][matches[0]])


def margin(scene: dict) -> float:
    target = _sets(scene)[0]
    distances = sorted(len(target.symmetric_difference({tuple(point) for point in pattern}))
                       for pattern in scene["candidates"])
    return float(distances[1]) if len(distances) == 3 and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        correct, whole_mirror, opposite_fold = _sets(scene)
        patterns = [{tuple(point) for point in pattern} for pattern in scene["candidates"]]
        return (len({frozenset(correct), frozenset(whole_mirror), frozenset(opposite_fold)}) != 3
                or any(len(pattern) != 5 for pattern in patterns)
                or any(not _inside(point) for pattern in patterns for point in pattern)
                or sum(pattern == correct for pattern in patterns) != 1
                or len(set(scene["tag_colors"])) != 3 or margin(scene) < 2)
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
