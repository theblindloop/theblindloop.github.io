"""Deterministic orthogonal-cord scene and analytic arm."""
from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw

SIZE = 640
BG = (248, 247, 242)
INK = (25, 28, 34)
CORD = (202, 205, 207)
COLORS = {
    "cyan": (0, 166, 205), "magenta": (218, 55, 157),
    "red": (218, 48, 56), "gold": (232, 163, 25),
    "violet": (123, 72, 194), "green": (38, 158, 82),
}
CANDIDATES = ("red", "gold", "violet", "green")


def _transform(point: tuple[float, float], code: int) -> tuple[float, float]:
    x, y = point
    mode = code % 8
    if mode >= 4:
        x = SIZE - x
        mode -= 4
    for _ in range(mode):
        x, y = SIZE - y, x
    return x, y


def _checkpoint(points: list[list[int]], segment: int, fraction: float) -> tuple[float, float]:
    a, b = points[segment], points[segment + 1]
    return (a[0] + fraction * (b[0] - a[0]), a[1] + fraction * (b[1] - a[1]))


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    # One elongated, non-self-intersecting cord with 13 segments and exactly
    # 12 visible corners. Small independent offsets vary its rhythm while the
    # decisive connected support continues to span nearly the full canvas.
    j = [rng.randint(-5, 5) for _ in range(6)]
    points = [
        [38, 272+j[0]], [86+j[0], 272+j[0]],
        [86+j[0], 342+j[1]], [170+j[1], 342+j[1]],
        [170+j[1], 286+j[2]], [255+j[2], 286+j[2]],
        [255+j[2], 365+j[3]], [342+j[3], 365+j[3]],
        [342+j[3], 258+j[4]], [430+j[4], 258+j[4]],
        [430+j[4], 332+j[5]], [516+j[5], 332+j[5]],
        [516+j[5], 278+j[0]], [602, 278+j[0]],
    ]
    low = rng.sample([1, 2, 3, 4], 2)
    high = rng.sample([8, 9, 10, 11], 1)
    segments = [6, *low, *high]
    rng.shuffle(segments)
    fractions = [rng.choice((0.34, 0.42, 0.58, 0.66)) for _ in segments]
    colors = list(CANDIDATES)
    rng.shuffle(colors)
    return {
        "points": points,
        "checkpoint_segments": segments,
        "checkpoint_fractions": fractions,
        "checkpoint_colors": colors,
        "transform": rng.randrange(8),
    }


def render(scene: dict) -> Image.Image:
    points = [[int(x), int(y)] for x, y in scene["points"]]
    transform = int(scene["transform"])
    route = [_transform((x, y), transform) for x, y in points]
    centers = [
        _transform(_checkpoint(points, int(seg), float(frac)), transform)
        for seg, frac in zip(scene["checkpoint_segments"], scene["checkpoint_fractions"], strict=True)
    ]
    image = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(image)

    def disk(center: tuple[float, float], radius: int, color: tuple[int, int, int]) -> None:
        x, y = center
        draw.ellipse((round(x-radius), round(y-radius), round(x+radius), round(y+radius)), fill=color)

    # The broad pale cord makes the visual route easy to see. Its exact black
    # one-pixel centerline is independently traceable and fixes every corner.
    draw.line(route, fill=CORD, width=7, joint="curve")
    disk(route[0], 17, COLORS["cyan"])
    disk(route[-1], 17, COLORS["magenta"])
    for center, name in zip(centers, scene["checkpoint_colors"], strict=True):
        disk(center, 15, COLORS[name])
    draw.line(route, fill=INK, width=1)
    return image


def _scores(scene: dict) -> list[int]:
    total = len(scene["points"]) - 2
    return [abs(2 * int(segment) - total) for segment in scene["checkpoint_segments"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    if len(scores) != 4 or scores.count(min(scores)) != 1:
        raise ValueError("ambiguous checkpoint scene")
    return scene["checkpoint_colors"][scores.index(min(scores))]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    points = scene.get("points", [])
    if len(points) != 14 or len(scene.get("checkpoint_colors", [])) != 4:
        return True
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["transform"] = int(twin["transform"]) + 8
    return [("periodic_dihedral_code", twin)]
