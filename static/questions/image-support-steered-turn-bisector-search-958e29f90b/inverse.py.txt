"""Pixel-only inverse arm: trace the exact centerline and count its corners."""
from __future__ import annotations

import numpy as np
from PIL import Image

INK = np.array((25, 28, 34), dtype=np.uint8)
COLORS = {
    "cyan": np.array((0, 166, 205), dtype=np.uint8),
    "magenta": np.array((218, 55, 157), dtype=np.uint8),
    "red": np.array((218, 48, 56), dtype=np.uint8),
    "gold": np.array((232, 163, 25), dtype=np.uint8),
    "violet": np.array((123, 72, 194), dtype=np.uint8),
    "green": np.array((38, 158, 82), dtype=np.uint8),
}
CANDIDATES = ("red", "gold", "violet", "green")


def _centroid(array: np.ndarray, color: np.ndarray, name: str) -> tuple[float, float]:
    mask = np.all(array == color, axis=2)
    pts = np.argwhere(mask)
    if len(pts) < 300:
        raise ValueError(f"missing {name} checkpoint")
    y, x = pts.mean(axis=0)
    return float(y), float(x)


def _trace(mask: np.ndarray, start_hint: tuple[float, float], end_hint: tuple[float, float]) -> list[tuple[int, int]]:
    pts = np.argwhere(mask)
    if len(pts) < 900:
        raise ValueError("cord centerline is missing")
    pixels = {tuple(map(int, p)) for p in pts}
    ends = []
    for y, x in pixels:
        degree = sum((y+dy, x+dx) in pixels for dy, dx in ((-1,0),(1,0),(0,-1),(0,1)))
        if degree == 1:
            ends.append((y, x))
        elif degree != 2:
            raise ValueError("cord is branched or corrupted")
    if len(ends) != 2:
        raise ValueError("cord is not a single simple trace")
    sy, sx = start_hint
    start = min(ends, key=lambda p: (p[0]-sy)**2 + (p[1]-sx)**2)
    ey, ex = end_hint
    finish = min(ends, key=lambda p: (p[0]-ey)**2 + (p[1]-ex)**2)
    if start == finish:
        raise ValueError("endpoint markers do not identify opposite ends")
    route = [start]
    previous = None
    current = start
    while current != finish:
        y, x = current
        nexts = [(y+dy, x+dx) for dy, dx in ((-1,0),(1,0),(0,-1),(0,1))
                 if (y+dy, x+dx) in pixels and (y+dy, x+dx) != previous]
        if len(nexts) != 1:
            raise ValueError("ambiguous cord traversal")
        previous, current = current, nexts[0]
        route.append(current)
        if len(route) > len(pixels):
            raise ValueError("cord traversal cycle")
    if len(route) != len(pixels):
        raise ValueError("unreached cord pixels")
    return route


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (640, 640, 3):
        raise ValueError("unexpected raster size")
    centers = {name: _centroid(array, color, name) for name, color in COLORS.items()}
    route = _trace(np.all(array == INK, axis=2), centers["cyan"], centers["magenta"])
    directions = [(b[0]-a[0], b[1]-a[1]) for a, b in zip(route, route[1:])]
    turns = [i + 1 for i in range(len(directions)-1) if directions[i] != directions[i+1]]
    if len(turns) < 4:
        raise ValueError("too few visible corners")
    coords = np.asarray(route, dtype=np.int32)
    scores: dict[str, int] = {}
    for name in CANDIDATES:
        y, x = centers[name]
        index = int(np.argmin((coords[:, 0]-y)**2 + (coords[:, 1]-x)**2))
        before = sum(turn < index for turn in turns)
        after = len(turns) - before
        scores[name] = abs(before - after)
    ranked = sorted(scores, key=lambda name: (scores[name], name))
    if scores[ranked[0]] != 0 or scores[ranked[1]] - scores[ranked[0]] < 2:
        raise ValueError("no unique equal-corner checkpoint")
    return ranked[0]
