"""Independent raster-only inverse for the calibration bracket.

This module intentionally knows nothing about the renderer's scene dictionary,
sampling process, prompt text, or analytic margin.  It identifies saturated
connected components in the PNG, recognizes the two tall tick glyphs and the
one square diamond glyph from their raster geometry, then applies the strict
two-sided interval test to their recovered centroids.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _background(array: np.ndarray) -> np.ndarray:
    height, width = array.shape[:2]
    strips = np.concatenate(
        (
            array[:18, :18].reshape(-1, 3),
            array[:18, width - 18 :].reshape(-1, 3),
            array[height - 18 :, :18].reshape(-1, 3),
            array[height - 18 :, width - 18 :].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(strips, axis=0).astype(np.float32)


def _saturated_mask(array: np.ndarray) -> np.ndarray:
    background = _background(array)
    spread = array.max(axis=2) - array.min(axis=2)
    distance = np.linalg.norm(array - background, axis=2)
    return (spread >= 50.0) & (distance >= 35.0)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected component pixel coordinates as y,x arrays."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue: deque[tuple[int, int]] = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dy or dx):
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) >= 30:
            found.append(np.asarray(points, dtype=np.float64))
    return found


def _bbox(component: np.ndarray) -> tuple[int, int, int, int]:
    ys = component[:, 0]
    xs = component[:, 1]
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _centroid_x(component: np.ndarray) -> float:
    return float(component[:, 1].mean())


def decision_from_image(image: Image.Image) -> str:
    """Recover yes/no from the final PNG alone."""

    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    mask = _saturated_mask(array)
    components = _components(mask)
    if len(components) != 3:
        raise ValueError(f"expected exactly three saturated glyphs, found {len(components)}")

    ticks: list[float] = []
    target: float | None = None
    for component in components:
        x0, y0, x1, y1 = _bbox(component)
        width = x1 - x0 + 1
        height = y1 - y0 + 1
        area = len(component)
        aspect = height / max(width, 1)
        if height >= 18 and aspect >= 1.65 and width <= 20:
            ticks.append(_centroid_x(component))
        elif abs(width - height) <= 4 and area >= 60:
            if target is not None:
                raise ValueError("multiple diamond-like components")
            target = _centroid_x(component)
        else:
            raise ValueError("saturated component has an unsupported glyph shape")

    if len(ticks) != 2 or target is None:
        raise ValueError("could not identify two endpoint ticks and one target diamond")
    left, right = sorted(ticks)
    if not left < right:
        raise ValueError("endpoint ticks are unresolved")
    return "yes" if left < target < right else "no"
