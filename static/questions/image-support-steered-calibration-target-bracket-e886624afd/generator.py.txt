"""Renderer and analytic arm for a compact calibration interval world.

The decisive evidence is deliberately only three disconnected saturated glyphs:
two identical vertical endpoint ticks and one diamond target.  Pale gray plot
context is present for a scientific-measurement appearance but is not used by
the pixel inverse.  The analytic arm reads only the three rendered center
coordinates that determine the interval decision.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
QUARANTINE_MARGIN = 10
MAX_CERTIFIED_MARGIN = 75

BACKGROUNDS = ("#f8f6f0", "#f6f8fb", "#fbf7f7", "#f4f8f3")
TICK_COLORS = ("#145a78", "#4c3978", "#265d48", "#7a4b20")
TARGET_COLORS = ("#c2493e", "#b05a18", "#a63170", "#2470a3")


def _draw_remote_context(draw: ImageDraw.ImageDraw, variant: int) -> None:
    """Add low-saturation measurement-chart texture that is nondiagnostic."""

    pale = "#d0d7dd"
    paler = "#dce1e5"
    for index in range(5):
        x = 28 + ((variant * 67 + index * 91) % 430)
        y = 42 + ((variant * 31 + index * 79) % 408)
        width = 26 + ((variant + index * 13) % 30)
        draw.line((x, y, x + width, y), fill=pale, width=2)
        draw.line((x, y - 7, x, y + 7), fill=paler, width=2)
        draw.rectangle((x + width - 4, y - 4, x + width + 4, y + 4), fill=paler)

    # A few faint calibration numerals represented as short typeset-like strokes.
    for index in range(4):
        x = 54 + ((variant * 43 + index * 117) % 390)
        y = 92 + ((variant * 59 + index * 71) % 340)
        draw.line((x, y, x + 5, y), fill=paler, width=2)
        draw.line((x + 5, y, x + 5, y + 9), fill=paler, width=2)
        draw.line((x + 5, y + 9, x + 11, y + 9), fill=paler, width=2)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, finite-resolution-safe calibration scene."""

    rng = random.Random(int(seed))
    # Odd scene seeds alternate the answer across the generator's 1009 stride,
    # while the absolute placement and the side of an outside target vary too.
    bracketed = int(seed) % 2 == 0
    near_boundary = int(seed) % 11 == 0
    left_column = rng.choice((1, 2, 3))
    tick_row = rng.choice((1, 2))
    if bracketed:
        # Independent ablation cells: left tick, right tick, and target form
        # the corners of a small triangle with one empty cell between them.
        left_tick_x = left_column * 64 + 32
        right_tick_x = (left_column + 2) * 64 + 32
        baseline_y = tick_row * 64 + 32
        right_tick_y = baseline_y
        target_y_offset = 128
    elif rng.randrange(2):
        # Outside-left layout: the endpoint rows are staggered so the target's
        # neighboring ablation cell remains separate from both endpoints.
        left_tick_x = left_column * 64 + 32
        right_tick_x = (left_column + 1) * 64 + 32
        baseline_y = tick_row * 64 + 32
        right_tick_y = baseline_y + 128
        target_y_offset = 64
    else:
        # Outside-right mirror of the staggered layout.
        left_tick_x = left_column * 64 + 32
        right_tick_x = (left_column + 1) * 64 + 32
        baseline_y = (tick_row + 2) * 64 + 32
        right_tick_y = baseline_y - 128
        target_y_offset = -64
    interval_width = right_tick_x - left_tick_x

    if near_boundary:
        # These are intentionally quarantined near-equality cases for gallery
        # coverage; the ordinary stress distribution remains well separated.
        target_x = left_tick_x + (6 if rng.randrange(2) else -6)
    elif bracketed:
        target_x = left_tick_x + rng.randint(62, interval_width - 62)
    elif right_tick_x > left_tick_x and right_tick_x - left_tick_x == 64 and right_tick_y > baseline_y:
        target_x = left_tick_x - rng.randint(22, 38)
    else:
        target_x = right_tick_x + rng.randint(22, 38)

    all_x = (left_tick_x, right_tick_x, target_x)
    all_y = (baseline_y, right_tick_y, baseline_y + target_y_offset)
    inset_x = min(all_x) - 56
    inset_y = min(all_y) - 46

    return {
        "left_tick_x": int(left_tick_x),
        "right_tick_x": int(right_tick_x),
        "target_x": int(target_x),
        "baseline_y": int(baseline_y),
        "right_tick_y": int(right_tick_y),
        "inset_x": int(inset_x),
        "inset_y": int(inset_y),
        "tick_width": int(rng.randint(8, 12)),
        "tick_height": int(rng.randint(23, 31)),
        "target_radius": int(rng.randint(9, 12)),
        "target_y_offset": int(target_y_offset),
        "tick_color": rng.choice(TICK_COLORS),
        "target_color": rng.choice(TARGET_COLORS),
        "background": rng.choice(BACKGROUNDS),
        "remote_variant": int(rng.randrange(12)),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render one PNG from the public scene dictionary."""

    image = Image.new("RGB", (WIDTH, HEIGHT), scene["background"])
    draw = ImageDraw.Draw(image)
    _draw_remote_context(draw, int(scene["remote_variant"]))

    inset_x = int(scene["inset_x"])
    inset_y = int(scene["inset_y"])
    all_x = (int(scene["left_tick_x"]), int(scene["right_tick_x"]), int(scene["target_x"]))
    all_y = (
        int(scene["baseline_y"]),
        int(scene["right_tick_y"]),
        int(scene["baseline_y"]) + int(scene["target_y_offset"]),
    )
    inset_w = max(all_x) - min(all_x) + 112
    inset_h = max(all_y) - min(all_y) + 92
    frame = "#b9c3ca"
    reference = "#d1d8dd"
    draw.rounded_rectangle(
        (inset_x, inset_y, inset_x + inset_w, inset_y + inset_h),
        radius=10,
        outline=frame,
        width=2,
    )
    # These pale reference tracks give the inset a measurement-chart grammar.
    # They are intentionally below the oracle's saturation mask and are
    # nondiagnostic; the two glyph rows remain visually separate.
    y_values = (
        int(scene["baseline_y"]),
        int(scene["right_tick_y"]),
        int(scene["baseline_y"]) + int(scene["target_y_offset"]),
    )
    for track_y in sorted(set(y_values)):
        draw.line((inset_x + 18, track_y, inset_x + inset_w - 18, track_y), fill=reference, width=2)
        for x in (inset_x + 28, inset_x + inset_w - 28):
            draw.line((x, track_y - 5, x, track_y + 5), fill=reference, width=2)

    tick_color = scene["tick_color"]
    tick_half = int(scene["tick_width"]) // 2
    half_height = int(scene["tick_height"]) // 2
    for x, tick_y in (
        (int(scene["left_tick_x"]), int(scene["baseline_y"])),
        (int(scene["right_tick_x"]), int(scene["right_tick_y"])),
    ):
        draw.rectangle(
            (x - tick_half, tick_y - half_height, x + tick_half, tick_y + half_height),
            fill=tick_color,
        )

    target_x = int(scene["target_x"])
    radius = int(scene["target_radius"])
    target_y = int(scene["baseline_y"]) + int(scene["target_y_offset"])
    draw.polygon(
        (
            (target_x, target_y - radius),
            (target_x + radius, target_y),
            (target_x, target_y + radius),
            (target_x - radius, target_y),
        ),
        fill=scene["target_color"],
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return yes exactly when the target center is strictly bracketed."""

    left = int(scene["left_tick_x"])
    right = int(scene["right_tick_x"])
    target = int(scene["target_x"])
    return "yes" if left < target < right else "no"


def margin(scene: dict[str, Any]) -> float:
    """Return signed nearest-boundary clearance in pixels."""

    left = int(scene["left_tick_x"])
    right = int(scene["right_tick_x"])
    target = int(scene["target_x"])
    left_clearance = target - left
    right_clearance = right - target
    if left_clearance > 0 and right_clearance > 0:
        return float(min(left_clearance, right_clearance))
    return float(left_clearance if left_clearance <= 0 else -right_clearance)


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Exclude near-equality and out-of-certification-band scenes."""

    value = abs(margin(scene))
    return value <= QUARANTINE_MARGIN or value > MAX_CERTIFIED_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases are declared for this injective scene parameterization."""

    return []
