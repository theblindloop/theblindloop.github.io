"""Riffle rank-binding correspondence world.

A KEY row shows N coloured tokens, each with a rank ruler beneath it: one
small filled cell per unit of that token's rank, so the ruler's filled-cell
count is the token's rank (1..N). The KEY records a colour-identity to-rank
binding. Two candidate rows A and B below show the same N colours without
rulers. The solver must bind each token to its KEY rank and then check, per
candidate row, whether the translated ranks reproduce the *riffled* order: rank
1, then the lowest HIGH-half rank, repeating so that LOW and HIGH halves are
interleaved with each half internally ascending. A candidate row is correct
exactly when its translated left-to-right ranks equal the fixed riffle target
[1, floor+1, 2, floor+2, ...], i.e. ``1,4,2,5,3,6`` for six tokens. This is a
correspondence-binding program: it binds each token identity to its keyed rank,
then carries that binding into each candidate row and evaluates a composed
parity-split (LOW/HIGH half) interleave predicate, rather than a plain
ascending-order check. The other candidate is a permutation whose translated
ranks are not riffled.

The renderer owns scene sampling, rasterization and the analytic gold. The
independent pixel-only inverse arm (``oracle.py``) recovers the same decision
from the final PNG alone: it detects each coloured token, reads the filled-cell
rank ruler under each KEY token to build the colour->rank map, then for each
candidate row translates its left-to-right colour sequence to ranks and checks
which row equals the riffled target.
"""

from __future__ import annotations

import numpy as np
from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 660
BACKGROUND = (255, 255, 255)
LABEL = (35, 35, 35)
PIP = (45, 45, 45)

N = 6
HALF = N // 2
COLOR_NAMES = ["red", "green", "blue", "orange", "purple", "brown"]
COLOR_RGB = {
    "red": (220, 40, 40),
    "green": (40, 170, 40),
    "blue": (40, 80, 220),
    "orange": (255, 150, 20),
    "purple": (150, 60, 190),
    "brown": (120, 80, 40),
}

TOKEN = 46
PITCH = 96
FIRST_CX = 150
KEY_TOP = 96
A_TOP = 408
B_TOP = 546
LABEL_Y = 52

CELL = 9
CELL_GAP = 2
RULER_GAP = 12
RULER_W = N * CELL + (N - 1) * CELL_GAP

RANK_MAX = N
QUAR_DECOY_MISPLACE = 2

_FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"

_CX = [FIRST_CX + i * PITCH for i in range(N)]


def riffle_target(n: int = N) -> list[int]:
    """Riffled rank order: interleave the LOW half (1..floor) ascending with
    the HIGH half (floor+1..n) ascending, starting with the LOW half."""
    half = n // 2
    low = list(range(1, half + 1))
    high = list(range(half + 1, n + 1))
    out = []
    for i in range(n):
        if i % 2 == 0:
            out.append(low[i // 2])
        else:
            out.append(high[i // 2])
    return out


TARGET = riffle_target()


def _perm(rng: np.random.Generator, n: int) -> list[int]:
    return list(rng.permutation(n))


def sample_scene(seed: int) -> dict:
    """Sample one latent scene deterministically from ``seed``."""
    rng = np.random.default_rng(seed)

    key_colors = [COLOR_NAMES[i] for i in _perm(rng, N)]
    key_ranks = [int(r) + 1 for r in _perm(rng, N)]
    color_to_rank = dict(zip(key_colors, key_ranks))

    rank_to_color = {rank: color for color, rank in color_to_rank.items()}
    riffled_seq = [rank_to_color[rank] for rank in TARGET]

    while True:
        decoy = [COLOR_NAMES[i] for i in _perm(rng, N)]
        misplaced = sum(color_to_rank[name] != TARGET[pos] for pos, name in enumerate(decoy))
        if misplaced >= QUAR_DECOY_MISPLACE:
            break

    correct_index = int(rng.integers(0, 2))
    if correct_index == 0:
        a_order = list(riffled_seq)
        b_order = decoy
    else:
        a_order = decoy
        b_order = list(riffled_seq)

    return {
        "key_colors": key_colors,
        "key_ranks": key_ranks,
        "a_order": a_order,
        "b_order": b_order,
        "correct_index": correct_index,
    }


def _draw_token(draw: ImageDraw.ImageDraw, cx: int, top: int, color: str) -> None:
    x0 = cx - TOKEN // 2
    draw.rectangle([x0, top, x0 + TOKEN, top + TOKEN], fill=COLOR_RGB[color], outline=(255, 255, 255), width=1)


def _draw_ruler(draw: ImageDraw.ImageDraw, cx: int, top: int, rank: int) -> None:
    x0 = cx - RULER_W // 2
    y0 = top
    for i in range(rank):
        x = x0 + i * (CELL + CELL_GAP)
        draw.rectangle([x, y0, x + CELL, y0 + CELL], fill=PIP)


def render(scene: dict) -> Image.Image:
    """Rasterize the latent scene (a deterministic PNG)."""
    img = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(img)
    try:
        font = ImageFont.truetype(_FONT_PATH, 26)
    except OSError:
        font = ImageFont.load_default()

    draw.text((24, LABEL_Y), "KEY", font=font, fill=LABEL)
    for i, color in enumerate(scene["key_colors"]):
        _draw_token(draw, _CX[i], KEY_TOP, color)
        _draw_ruler(draw, _CX[i], KEY_TOP + TOKEN + RULER_GAP, scene["key_ranks"][i])

    draw.text((24, A_TOP + 8), "A", font=font, fill=LABEL)
    for i, color in enumerate(scene["a_order"]):
        _draw_token(draw, _CX[i], A_TOP, color)

    draw.text((24, B_TOP + 8), "B", font=font, fill=LABEL)
    for i, color in enumerate(scene["b_order"]):
        _draw_token(draw, _CX[i], B_TOP, color)

    return img


def _color_to_rank(scene: dict) -> dict:
    return dict(zip(scene["key_colors"], scene["key_ranks"]))


def _is_riffled(seq: list, color_to_rank: dict) -> bool:
    ranks = [color_to_rank[name] for name in seq]
    return ranks == TARGET


def analytic_gold(scene: dict) -> str:
    """Structural gold: which candidate row is bound to the riffled rank order."""
    color_to_rank = _color_to_rank(scene)
    a_riff = _is_riffled(scene["a_order"], color_to_rank)
    b_riff = _is_riffled(scene["b_order"], color_to_rank)
    if a_riff:
        return "A"
    if b_riff:
        return "B"
    raise ValueError("neither candidate row is bound to the riffled rank order")


def margin(scene: dict) -> float:
    """Commitment magnitude: how disordered the decoy row is (misplaced ranks)."""
    color_to_rank = _color_to_rank(scene)
    a_riff = _is_riffled(scene["a_order"], color_to_rank)
    b_riff = _is_riffled(scene["b_order"], color_to_rank)
    decoy = scene["b_order"] if a_riff else scene["a_order"]
    misplaced = sum(color_to_rank[name] != TARGET[pos] for pos, name in enumerate(decoy))
    return float(misplaced)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where the decoy is barely different (ambiguous)."""
    return margin(scene) < QUAR_DECOY_MISPLACE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The scene is injectively rendered; no latent aliases are declared."""
    return []
