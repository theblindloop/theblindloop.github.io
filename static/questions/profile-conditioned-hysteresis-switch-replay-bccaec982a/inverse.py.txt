"""Independent pixel-only inverse for the hysteresis controller."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

BLUE = np.array((42, 119, 201), dtype=np.int16)
RED = np.array((211, 70, 62), dtype=np.int16)
GREEN = np.array((32, 150, 112), dtype=np.int16)


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray, minimum: int = 8) -> list[tuple[int, int, int, int, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))])
        seen[y, x] = True
        xs, ys = [], []
        while q:
            cy, cx = q.popleft()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        if len(xs) >= minimum:
            result.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    blue_components = _components(_mask(arr, BLUE), minimum=100)
    if len(blue_components) != 5:
        raise ValueError(f"expected five visible level bars, found {len(blue_components)}")
    blue_components.sort(key=lambda box: (box[0] + box[2]) / 2)

    green_components = _components(_mask(arr, GREEN), minimum=50)
    if len(green_components) != 1:
        raise ValueError("start lamp is missing or ambiguous")
    x0, y0, x1, y1, _ = green_components[0]
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    state = bool(_mask(arr, GREEN)[cy, cx])

    red_components = _components(_mask(arr, RED), minimum=20)
    if len(red_components) != 20:
        raise ValueError(f"expected twenty threshold ticks, found {len(red_components)}")

    for bx0, by0, bx1, by1, _ in blue_components:
        center_x = (bx0 + bx1) / 2
        nearby = []
        for rx0, ry0, rx1, ry1, _ in red_components:
            red_center_x = (rx0 + rx1) / 2
            if abs(red_center_x - center_x) < 60:
                nearby.append((ry0 + ry1) / 2)
        rows = []
        for value in sorted(nearby):
            if not rows or abs(value - rows[-1]) > 3:
                rows.append(value)
        if len(rows) != 2:
            raise ValueError("a gauge does not expose exactly two threshold rows")
        high_y, low_y = rows
        bar_top = by0
        clearance = min(abs(bar_top - high_y), abs(bar_top - low_y))
        if clearance <= 6:
            raise ValueError("bar top is too close to a threshold")
        if bar_top < high_y:
            state = True
        elif bar_top > low_y:
            state = False
        # Otherwise the hold band preserves state.
    return "ON" if state else "OFF"
