"""Deterministic renderer and analytic model for a visible hysteresis replay."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 960, 500
BG = (247, 249, 252)
INK = (31, 43, 58)
BLUE = (42, 119, 201)
RED = (211, 70, 62)
GREEN = (32, 150, 112)
CARD_A = (255, 255, 255)
CARD_B = (242, 247, 255)


def sample_scene(seed: int) -> dict:
    """Return balanced deterministic cases in three causal regimes."""
    rng = random.Random(int(seed) ^ 0x5A17C3)
    target_on = bool(int(seed) % 2)
    mode = (int(seed) // 2) % 3
    start_on = bool(rng.randrange(2))
    levels = ["MID"] * 5
    if mode == 0:  # pure hold: the start state is essential
        start_on = target_on
    elif mode == 1:  # equal HIGH/LOW ink; their temporal order decides
        a, b = sorted(rng.sample(range(4), 2))
        if target_on:
            levels[a], levels[b] = "LOW", "HIGH"
        else:
            levels[a], levels[b] = "HIGH", "LOW"
    else:  # one decisive crossing followed by one or more holds
        position = rng.randrange(4)
        levels[position] = "HIGH" if target_on else "LOW"
    return {
        "start_on": start_on,
        "levels": levels,
        "style_variant": (int(seed) // 6) % 2,
    }


def _font() -> ImageFont.ImageFont:
    return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str) -> None:
    box = draw.textbbox((0, 0), text, font=_font())
    w, h = box[2] - box[0], box[3] - box[1]
    draw.text((xy[0] - w // 2, xy[1] - h // 2), text, fill=INK, font=_font())


def render(scene: dict) -> Image.Image:
    start_on = bool(scene["start_on"])
    levels = list(scene["levels"])
    style_variant = int(scene["style_variant"])
    if len(levels) != 5 or any(value not in {"LOW", "MID", "HIGH"} for value in levels):
        raise ValueError("levels must contain five LOW/MID/HIGH values")

    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    accent = CARD_A if style_variant == 0 else CARD_B
    draw.rounded_rectangle((24, 22, 936, 86), radius=16, fill=accent, outline=INK, width=3)
    _center_text(draw, (480, 46), "HYSTERESIS CONTROLLER")
    _center_text(draw, (480, 68), "READ LEVEL CARDS LEFT TO RIGHT")

    # Start card: green geometry alone carries the starting state.
    draw.rounded_rectangle((28, 116, 174, 438), radius=18, fill=accent, outline=INK, width=3)
    _center_text(draw, (101, 145), "START LAMP")
    lamp_box = (64, 178, 138, 252)
    if start_on:
        draw.ellipse(lamp_box, fill=GREEN, outline=GREEN, width=7)
    else:
        draw.ellipse(lamp_box, fill=BG, outline=GREEN, width=7)
    _center_text(draw, (101, 284), "SOLID = ON")
    _center_text(draw, (101, 306), "HOLLOW = OFF")
    draw.line((54, 350, 148, 350), fill=INK, width=2)
    _center_text(draw, (101, 374), "INITIAL STATE")

    # Five cards. Blue components encode levels; paired red ticks expose both thresholds.
    card_xs = [194, 344, 494, 644, 794]
    top_shift = 0 if style_variant == 0 else 4
    for index, (card_x, level) in enumerate(zip(card_xs, levels), start=1):
        y0 = 116 + top_shift
        draw.rounded_rectangle((card_x, y0, card_x + 136, y0 + 322), radius=18,
                               fill=accent, outline=INK, width=3)
        _center_text(draw, (card_x + 68, y0 + 29), f"STEP {index}")
        gx0, gx1 = card_x + 51, card_x + 85
        gy0, gy1 = y0 + 62, y0 + 252
        draw.rectangle((gx0, gy0, gx1, gy1), outline=INK, width=3)
        top_offsets = {"HIGH": 20, "MID": 91, "LOW": 150}
        fill_top = gy0 + top_offsets[level]
        draw.rectangle((gx0 + 4, fill_top, gx1 - 4, gy1 - 4), fill=BLUE)
        high_y, low_y = gy0 + 55, gy0 + 128
        draw.line((gx0 - 17, high_y, gx0 - 3, high_y), fill=RED, width=5)
        draw.line((gx1 + 3, high_y, gx1 + 17, high_y), fill=RED, width=5)
        draw.line((gx0 - 17, low_y, gx0 - 3, low_y), fill=RED, width=5)
        draw.line((gx1 + 3, low_y, gx1 + 17, low_y), fill=RED, width=5)
        _center_text(draw, (card_x + 24, high_y), "H")
        _center_text(draw, (card_x + 24, low_y), "L")
        _center_text(draw, (card_x + 68, y0 + 278), "LEVEL")
    return image


def analytic_gold(scene: dict) -> str:
    state = bool(scene["start_on"])
    for level in scene["levels"]:
        if level == "HIGH":
            state = True
        elif level == "LOW":
            state = False
        elif level != "MID":
            raise ValueError(f"unknown level {level!r}")
    return "ON" if state else "OFF"


def margin(scene: dict) -> float:
    # Nearest designed bar-top-to-threshold separation: LOW 22, HIGH 35, MID 36 px.
    clearances = {"LOW": 22.0, "MID": 36.0, "HIGH": 35.0}
    return min(clearances[level] for level in scene["levels"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
