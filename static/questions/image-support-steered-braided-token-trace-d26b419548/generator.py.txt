"""Latent sampler, analytic arm, and rasterizer for Braided Token Trace."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (255, 255, 255)
DARK = (35, 43, 56)
CYAN = (18, 171, 193)
VIOLET = (133, 73, 184)
GOLD = (238, 174, 35)
PALE = (224, 230, 236)
LANES = ("top", "middle", "bottom")
OPERATIONS = ("swap_top_middle", "swap_middle_bottom")
TRACK_X0 = 28
TRACK_X1 = 484
LANE_OFFSET = 9


def _apply(lane: str, operation: str) -> str:
    if operation == "swap_top_middle":
        return {"top": "middle", "middle": "top", "bottom": "bottom"}[lane]
    if operation == "swap_middle_bottom":
        return {"top": "top", "middle": "bottom", "bottom": "middle"}[lane]
    raise ValueError(f"unknown operation {operation!r}")


def _start_for_target(target: str, operations: list[str]) -> str:
    for candidate in LANES:
        lane = candidate
        for operation in operations:
            lane = _apply(lane, operation)
        if lane == target:
            return candidate
    raise AssertionError("adjacent swaps must form a permutation")


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    while True:
        operations = [rng.choice(OPERATIONS) for _ in range(6)]
        if len(set(operations)) == 2 and any(
            operations[index] != operations[index + 1] for index in range(5)
        ):
            break
    target = LANES[int(seed) % 3]
    direction = 1 if rng.randrange(2) == 0 else -1
    station_xs = [82 + 68 * index + rng.randint(-3, 3) for index in range(6)]
    center_y = rng.choice([32 + 64 * index for index in range(8)])
    decorations: list[list[int]] = []
    for _ in range(10):
        for _attempt in range(80):
            x = rng.randint(18, WIDTH - 18)
            y = rng.randint(12, HEIGHT - 12)
            if abs(y - center_y) >= 34:
                decorations.append([x, y, rng.choice([2, 3])])
                break
    return {
        "center_y": center_y,
        "direction": direction,
        "start_lane": _start_for_target(target, operations),
        "operations": operations,
        "station_xs": station_xs,
        "decorations": decorations,
    }


def _lane_ys(center_y: int) -> tuple[int, int, int]:
    return center_y - LANE_OFFSET, center_y, center_y + LANE_OFFSET


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=PALE)

    ys = _lane_ys(int(scene["center_y"]))
    for y in ys:
        draw.line((TRACK_X0, y, TRACK_X1, y), fill=DARK, width=3)

    physical_xs = list(scene["station_xs"])
    if int(scene["direction"]) == -1:
        physical_xs.reverse()
    for operation, x in zip(scene["operations"], physical_xs, strict=True):
        if operation == "swap_top_middle":
            a, b = ys[0], ys[1]
            color = CYAN
        elif operation == "swap_middle_bottom":
            a, b = ys[1], ys[2]
            color = VIOLET
        else:
            raise ValueError(f"unknown operation {operation!r}")
        draw.line((x - 13, a, x + 13, b), fill=color, width=5)
        draw.line((x - 13, b, x + 13, a), fill=color, width=5)

    start_x = TRACK_X0 if int(scene["direction"]) == 1 else TRACK_X1
    start_y = ys[LANES.index(str(scene["start_lane"]))]
    draw.ellipse((start_x - 8, start_y - 8, start_x + 8, start_y + 8), fill=DARK)
    draw.ellipse((start_x - 5, start_y - 5, start_x + 5, start_y + 5), fill=GOLD)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    lane = str(scene["start_lane"])
    for operation in scene["operations"]:
        lane = _apply(lane, str(operation))
    return lane


def margin(scene: dict[str, Any]) -> float:
    xs = sorted(int(value) for value in scene["station_xs"])
    station_clearance = min(xs[index + 1] - xs[index] - 30 for index in range(5))
    edge_clearance = min(xs[0] - TRACK_X0 - 18, TRACK_X1 - xs[-1] - 18)
    return float(min(station_clearance, edge_clearance, LANE_OFFSET - 5))


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        operations = list(scene["operations"])
        xs = sorted(int(value) for value in scene["station_xs"])
        center_y = int(scene["center_y"])
        return bool(
            len(operations) != 6
            or len(xs) != 6
            or len(set(operations)) != 2
            or any(operation not in OPERATIONS for operation in operations)
            or str(scene["start_lane"]) not in LANES
            or int(scene["direction"]) not in (-1, 1)
            or center_y not in [32 + 64 * index for index in range(8)]
            or margin(scene) < 4.0
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
