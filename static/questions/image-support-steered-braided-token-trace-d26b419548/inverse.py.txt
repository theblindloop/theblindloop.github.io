"""Independent RGB-only inverse arm for Braided Token Trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_DARK = np.array((35, 43, 56), dtype=np.int16)
_CYAN = np.array((18, 171, 193), dtype=np.int16)
_VIOLET = np.array((133, 73, 184), dtype=np.int16)
_GOLD = np.array((238, 174, 35), dtype=np.int16)
_LANES = ("top", "middle", "bottom")


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 8) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _large(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    return [component for component in _components(mask) if len(component) >= minimum]


def _row_groups(row_counts: np.ndarray, threshold: int) -> list[tuple[int, int]]:
    rows = np.flatnonzero(row_counts >= threshold)
    if len(rows) == 0:
        return []
    groups: list[tuple[int, int]] = []
    start = prior = int(rows[0])
    for raw in rows[1:]:
        value = int(raw)
        if value != prior + 1:
            groups.append((start, prior))
            start = value
        prior = value
    groups.append((start, prior))
    return groups


def _apply(lane: str, operation: str) -> str:
    if operation == "swap_top_middle":
        return {"top": "middle", "middle": "top", "bottom": "bottom"}[lane]
    return {"top": "top", "middle": "bottom", "bottom": "middle"}[lane]


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape[:2] != (512, 512):
        raise ValueError("unexpected canvas size")

    dark = _mask(array, _DARK)
    cyan = _mask(array, _CYAN)
    violet = _mask(array, _VIOLET)
    gold = _mask(array, _GOLD)
    relevant = dark | cyan | violet | gold

    graph_components = _large(relevant, 250)
    if len(graph_components) != 1 or len(graph_components[0]) < 3500:
        raise ValueError("the connected braid is missing or broken")
    graph = graph_components[0]
    graph_x0 = int(graph[:, 1].min())
    graph_x1 = int(graph[:, 1].max())
    if graph_x0 > 28 or graph_x1 < 484 or graph_x1 - graph_x0 < 456:
        raise ValueError("braid does not span the canvas")

    groups = _row_groups(dark.sum(axis=1), 280)
    if len(groups) != 3 or any(end - start + 1 not in (3, 4) for start, end in groups):
        raise ValueError("three lane rails are not unambiguous")
    lane_ys = [0.5 * (start + end) for start, end in groups]
    if any(not 8.0 <= lane_ys[index + 1] - lane_ys[index] <= 10.0 for index in range(2)):
        raise ValueError("lane spacing is ambiguous")
    y0 = max(0, int(lane_ys[0]) - 11)
    y1 = min(512, int(lane_ys[-1]) + 12)
    occupied_columns = relevant[y0:y1, :].any(axis=0)
    missing = np.flatnonzero(~occupied_columns[28:485])
    if len(missing):
        raise ValueError("the ordered braid trace has a missing span")

    gold_components = _large(gold, 60)
    if len(gold_components) != 1:
        raise ValueError("starting ring is ambiguous")
    start_x = float(gold_components[0][:, 1].mean())
    start_y = float(gold_components[0][:, 0].mean())
    if min(abs(start_x - 28.0), abs(start_x - 484.0)) > 2.0:
        raise ValueError("starting ring is not attached to an endpoint")
    direction = 1 if start_x < 256.0 else -1
    lane_index = int(np.argmin([abs(start_y - value) for value in lane_ys]))
    if abs(start_y - lane_ys[lane_index]) > 2.0:
        raise ValueError("starting lane is ambiguous")
    lane = _LANES[lane_index]

    stations: list[tuple[float, str]] = []
    for name, mask in (("swap_top_middle", cyan), ("swap_middle_bottom", violet)):
        for component in _large(mask, 90):
            xs = component[:, 1]
            ys = component[:, 0]
            width = int(xs.max() - xs.min() + 1)
            height = int(ys.max() - ys.min() + 1)
            if not 27 <= width <= 31 or not 12 <= height <= 14:
                raise ValueError("crossover geometry is ambiguous")
            stations.append((float(xs.mean()), name))
    if len(stations) != 6 or {name for _, name in stations} != {
        "swap_top_middle", "swap_middle_bottom"
    }:
        raise ValueError("expected six crossovers of both types")
    stations.sort(key=lambda item: direction * item[0])
    ordered_xs = [direction * x for x, _ in stations]
    if any(ordered_xs[index + 1] - ordered_xs[index] < 50.0 for index in range(5)):
        raise ValueError("crossover order is crowded")

    for _, operation in stations:
        lane = _apply(lane, operation)
    return lane
