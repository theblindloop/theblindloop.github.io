"""Independent PNG-only inverse arm for the erode-then-expand reveal."""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 12


def _opening(cells: np.ndarray) -> np.ndarray:
    p = np.pad(cells.astype(np.uint8), 1)
    e = np.ones_like(cells, dtype=np.uint8)
    for dr in range(3):
        for dc in range(3):
            e &= p[dr:dr + N, dc:dc + N]
    p = np.pad(e, 1)
    d = np.zeros_like(cells, dtype=np.uint8)
    for dr in range(3):
        for dc in range(3):
            d |= p[dr:dr + N, dc:dc + N]
    return d


def _templates() -> dict[str, np.ndarray]:
    result = {}
    a = np.zeros((N, N), dtype=np.uint8); a[3:9,1:4]=1; a[3:9,8:11]=1; a[5:8,3:9]=1; result["barbell"]=_opening(a)
    a = np.zeros((N, N), dtype=np.uint8); a[2:10,2:5]=1; a[2:10,7:10]=1; a[2:5,4:8]=1; result["arch"]=_opening(a)
    a = np.zeros((N, N), dtype=np.uint8); a[2:5,2:10]=1; a[4:10,2:5]=1; a[7:10,4:10]=1; a[5:8,7:10]=1; result["spiral"]=_opening(a)
    a = np.zeros((N, N), dtype=np.uint8); a[3:10,1:4]=1; a[5:10,5:8]=1; a[3:10,8:11]=1; a[7:10,3:9]=1; result["crown"]=_opening(a)
    return result


TEMPLATES = _templates()


def _source_cells(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (arr[:, :, 0] < 40) & (arr[:, :, 1] > 145) & (arr[:, :, 1] < 205) & (arr[:, :, 2] > 165)
    ys, xs = np.nonzero(cyan)
    if len(xs) < 500:
        raise ValueError("cyan grid frame is missing")
    min_x, max_x, min_y, max_y = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = max_x - min_x + 1, max_y - min_y + 1
    if not (224 <= width <= 232 and 224 <= height <= 232):
        raise ValueError("cyan fiducial has invalid dimensions")
    x0, y0 = min_x + 6, min_y + 6
    cell_w, cell_h = (width - 12) / N, (height - 12) / N
    cells = np.zeros((N, N), dtype=np.uint8)
    for r in range(N):
        for c in range(N):
            cx = int(round(x0 + (c + 0.5) * cell_w))
            cy = int(round(y0 + (r + 0.5) * cell_h))
            patch = arr[cy - 3:cy + 4, cx - 3:cx + 4]
            cells[r, c] = int(float(patch.mean()) < 90.0)
    return cells


def decision_from_image(image: Image.Image) -> str:
    opened = _opening(_source_cells(image))
    scores = sorted((int(np.count_nonzero(opened ^ template)), name) for name, template in TEMPLATES.items())
    if scores[0][0] == scores[1][0] or scores[1][0] - scores[0][0] <= 4:
        raise ValueError("symbol classification is ambiguous")
    return scores[0][1]
