"""Scene sampler and renderer for the erode-then-expand reveal world."""

from __future__ import annotations

import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

N = 12
CELL = 18
GRID_PX = N * CELL
CYAN = (0, 174, 199)
MAGENTA = (220, 45, 135)
YELLOW = (244, 190, 35)


def _opening(cells: np.ndarray) -> np.ndarray:
    a = cells.astype(np.uint8)
    padded = np.pad(a, 1)
    eroded = np.ones_like(a)
    for dr in range(3):
        for dc in range(3):
            eroded &= padded[dr:dr + N, dc:dc + N]
    padded_e = np.pad(eroded, 1)
    expanded = np.zeros_like(a)
    for dr in range(3):
        for dc in range(3):
            expanded |= padded_e[dr:dr + N, dc:dc + N]
    return expanded


def _templates() -> dict[str, np.ndarray]:
    result: dict[str, np.ndarray] = {}
    a = np.zeros((N, N), dtype=np.uint8)
    a[3:9, 1:4] = 1; a[3:9, 8:11] = 1; a[5:8, 3:9] = 1
    result["barbell"] = _opening(a)
    a = np.zeros((N, N), dtype=np.uint8)
    a[2:10, 2:5] = 1; a[2:10, 7:10] = 1; a[2:5, 4:8] = 1
    result["arch"] = _opening(a)
    a = np.zeros((N, N), dtype=np.uint8)
    a[2:5, 2:10] = 1; a[4:10, 2:5] = 1; a[7:10, 4:10] = 1; a[5:8, 7:10] = 1
    result["spiral"] = _opening(a)
    a = np.zeros((N, N), dtype=np.uint8)
    a[3:10, 1:4] = 1; a[5:10, 5:8] = 1; a[3:10, 8:11] = 1; a[7:10, 3:9] = 1
    result["crown"] = _opening(a)
    return result


TEMPLATES = _templates()
DECISIONS = tuple(TEMPLATES)


def _scores(opened: np.ndarray) -> list[tuple[int, str]]:
    return sorted((int(np.count_nonzero(opened ^ template)), name) for name, template in TEMPLATES.items())


def _decoded(cells: list[list[int]]) -> tuple[str, float]:
    scores = _scores(_opening(np.asarray(cells, dtype=np.uint8)))
    return scores[0][1], float(scores[1][0] - scores[0][0])


def _noisy_source(target: np.ndarray, rng: random.Random) -> np.ndarray:
    """Add label-balanced dark interference that opening provably removes."""
    source = target.copy()
    candidates = [(r, c) for r in range(N) for c in range(N) if not source[r, c]]
    rng.shuffle(candidates)
    for r, c in candidates:
        if int(source.sum()) >= 88:
            break
        trial = source.copy()
        trial[r, c] = 1
        if np.array_equal(_opening(trial), target):
            source = trial
    if int(source.sum()) != 88 or not np.array_equal(_opening(source), target):
        raise RuntimeError("could not construct balanced removable interference")
    return source


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    label = DECISIONS[int(seed) % len(DECISIONS)]
    source = _noisy_source(TEMPLATES[label], rng)
    palettes = [
        ((231, 72, 82), (78, 110, 220)),
        ((126, 76, 186), (230, 126, 34)),
        ((54, 150, 95), (205, 75, 148)),
    ]
    colors = palettes[(int(seed) // 4) % len(palettes)]
    decorations = [
        {"x": 382, "y": 85, "color": list(colors[0]), "kind": "diamond"},
        {"x": 400, "y": 230, "color": list(colors[1]), "kind": "circle"},
    ]
    if rng.randrange(2):
        decorations.reverse()
    return {
        "cells": source.astype(int).tolist(),
        "grid_x": 72 + 12 * rng.randrange(4),
        "grid_y": 70 + 10 * rng.randrange(3),
        "decorations": decorations,
    }


def _draw_decorations(draw: ImageDraw.ImageDraw, decorations: list[dict]) -> None:
    for item in sorted(decorations, key=lambda value: (value["x"], value["y"], value["kind"])):
        x, y = int(item["x"]), int(item["y"])
        color = tuple(int(v) for v in item["color"])
        if item["kind"] == "diamond":
            draw.polygon([(x, y - 10), (x + 10, y), (x, y + 10), (x - 10, y)], fill=color)
        else:
            draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)


def render(scene: dict) -> Image.Image:
    cells = np.asarray(scene["cells"], dtype=np.uint8)
    x, y = int(scene["grid_x"]), int(scene["grid_y"])
    image = Image.new("RGB", (480, 360), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    _draw_decorations(draw, scene["decorations"])
    draw.rectangle((x - 6, y - 6, x + GRID_PX + 5, y + GRID_PX + 5), fill=CYAN)
    draw.rectangle((x, y, x + GRID_PX - 1, y + GRID_PX - 1), fill=(255, 255, 255))
    for r in range(N):
        for c in range(N):
            left, top = x + c * CELL, y + r * CELL
            if cells[r, c]:
                draw.rectangle((left + 2, top + 2, left + CELL - 3, top + CELL - 3), fill=(20, 22, 25))
    for k in range(N + 1):
        draw.line((x + k * CELL, y, x + k * CELL, y + GRID_PX), fill=(204, 207, 207), width=1)
        draw.line((x, y + k * CELL, x + GRID_PX, y + k * CELL), fill=(204, 207, 207), width=1)
    draw.rectangle((x - 3, y - 18, x + 13, y - 7), fill=MAGENTA)
    draw.rectangle((x + GRID_PX - 14, y - 18, x + GRID_PX + 2, y - 7), fill=YELLOW)
    return image


def analytic_gold(scene: dict) -> str:
    return _decoded(scene["cells"])[0]


def margin(scene: dict) -> float:
    return _decoded(scene["cells"])[1]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("decoration_serialization_order", twin)]
