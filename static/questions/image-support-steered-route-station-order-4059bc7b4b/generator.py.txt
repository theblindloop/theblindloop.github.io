"""Deterministic renderer for the connected-route station-order world."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


PROMPT_FAMILY = "pf1_route_shape_order"
SHAPE_NAMES = ("circle", "triangle", "square", "diamond")
_ROUTE_COLORS = (
    (36, 48, 64),
    (48, 49, 57),
    (35, 65, 56),
    (55, 43, 67),
)
_STATION_COLORS = (
    (223, 77, 61),
    (28, 157, 184),
    (223, 130, 35),
    (174, 71, 170),
    (41, 160, 94),
)
_BACKGROUNDS = (
    (248, 247, 242),
    (242, 248, 250),
    (249, 244, 247),
    (240, 247, 239),
)
_COUNTERFACTUAL_ORDERS = (
    (1, 0, 2, 3),
    (0, 2, 1, 3),
    (0, 1, 3, 2),
    (2, 1, 0, 3),
    (3, 0, 1, 2),
)


def _as_rgb(value: Iterable[int]) -> tuple[int, int, int]:
    values = tuple(int(item) for item in value)
    if len(values) != 3:
        raise ValueError("RGB value must have three channels")
    return values  # type: ignore[return-value]


def _path_points(scene: dict, count: int = 161) -> list[tuple[float, float]]:
    """Create a vertically advancing, horizontally backtracking open route."""

    width = float(scene["width"])
    height = float(scene["height"])
    sway = float(scene["x_sway"])
    phase = float(scene["y_phase"])
    amplitude = float(scene["y_amplitude"])
    baseline_x = 55.0
    horizontal_span = width - 120.0
    baseline_y = 60.0
    vertical_span = height - 120.0
    points = []
    for index in range(count):
        t = index / float(count - 1)
        x = baseline_x + horizontal_span * t + sway * math.sin(4.0 * math.pi * t)
        y = baseline_y + vertical_span * t + amplitude * math.sin(
            2.0 * math.pi * t + phase
        )
        points.append((x, y))
    return points


def _polyline_lengths(points: list[tuple[float, float]]) -> tuple[list[float], float]:
    cumulative = [0.0]
    for first, second in zip(points, points[1:]):
        cumulative.append(cumulative[-1] + math.dist(first, second))
    return cumulative, cumulative[-1]


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    cumulative, total = _polyline_lengths(points)
    target = max(0.0, min(1.0, float(fraction))) * total
    for index in range(1, len(cumulative)):
        if cumulative[index] >= target:
            before = cumulative[index - 1]
            span = cumulative[index] - before
            alpha = 0.0 if span <= 0.0 else (target - before) / span
            x0, y0 = points[index - 1]
            x1, y1 = points[index]
            return (x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0))
    return points[-1]


def _station_positions(scene: dict) -> list[tuple[float, float]]:
    points = _path_points(scene)
    return [_point_at_fraction(points, value) for value in scene["station_fractions"]]


def _draw_shape(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    radius: float,
    shape: int,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    cx, cy = center[0] * scale, center[1] * scale
    r = radius * scale
    if shape == 0:  # circle
        draw.ellipse((cx - r, cy - r, cx + r, cy + r), fill=fill)
    elif shape == 1:  # triangle, with its apex at the top
        draw.polygon(((cx, cy - r), (cx + r, cy + r), (cx - r, cy + r)), fill=fill)
    elif shape == 2:  # square
        draw.rectangle((cx - r, cy - r, cx + r, cy + r), fill=fill)
    elif shape == 3:  # diamond
        draw.polygon(((cx, cy - r), (cx + r, cy), (cx, cy + r), (cx - r, cy)), fill=fill)
    else:
        raise ValueError(f"unknown station shape {shape}")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    positive = int(seed) % 2 == 0
    if positive:
        order = [0, 1, 2, 3]
    else:
        order = list(rng.choice(_COUNTERFACTUAL_ORDERS))
    base_fractions = (0.18, 0.34, 0.60, 0.82)
    fractions = [
        round(value + rng.uniform(-0.012, 0.012), 6) for value in base_fractions
    ]
    radii = [rng.choice((16, 17, 18, 19, 20)) for _ in SHAPE_NAMES]
    return {
        "route_order": order,
        "station_fractions": fractions,
        "station_radii": radii,
        "route_width": rng.choice((7, 8, 9)),
        "route_color": list(rng.choice(_ROUTE_COLORS)),
        "station_color": list(rng.choice(_STATION_COLORS)),
        "background": list(rng.choice(_BACKGROUNDS)),
        "width": 800,
        "height": 500,
        "x_sway": rng.choice((68, 72, 76, 80)),
        "y_phase": round(rng.uniform(0.0, 1.2), 6),
        "y_amplitude": rng.choice((14, 16, 18, 20)),
        "antialias": rng.choice((3, 4)),
    }


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    scale = max(1, min(4, int(scene["antialias"])))
    background = _as_rgb(scene["background"])
    route_color = _as_rgb(scene["route_color"])
    station_color = _as_rgb(scene["station_color"])
    canvas = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(canvas)

    points = _path_points(scene)
    scaled_points = [(round(x * scale), round(y * scale)) for x, y in points]
    draw.line(
        scaled_points,
        fill=route_color,
        width=max(1, int(scene["route_width"]) * scale),
        joint="curve",
    )

    # The arrow is part of the same connected route and supplies a clear start.
    x0, y0 = points[0]
    x1, y1 = points[3]
    dx, dy = x1 - x0, y1 - y0
    length = math.hypot(dx, dy) or 1.0
    ux, uy = dx / length, dy / length
    nx, ny = -uy, ux
    tip = (x0 + ux * 22.0, y0 + uy * 22.0)
    base = (x0 - ux * 4.0, y0 - uy * 4.0)
    arrow = [
        (round(tip[0] * scale), round(tip[1] * scale)),
        (round((base[0] + nx * 11.0) * scale), round((base[1] + ny * 11.0) * scale)),
        (round((base[0] - nx * 11.0) * scale), round((base[1] - ny * 11.0) * scale)),
    ]
    draw.polygon(arrow, fill=route_color)

    for rank, shape in enumerate(scene["route_order"]):
        center = _point_at_fraction(points, scene["station_fractions"][rank])
        radius = float(scene["station_radii"][int(shape)])
        _draw_shape(draw, center, radius, int(shape), station_color, scale)

    if scale > 1:
        canvas = canvas.resize((width, height), Image.Resampling.LANCZOS)
    return canvas


def analytic_gold(scene: dict) -> str:
    order = tuple(int(value) for value in scene["route_order"])
    return "yes" if order == (0, 1, 2, 3) else "no"


def margin(scene: dict) -> float:
    points = _path_points(scene)
    cumulative, total = _polyline_lengths(points)
    fractions = [float(value) for value in scene["station_fractions"]]
    gaps = [max(0.0, (right - left) * total) for left, right in zip(fractions, fractions[1:])]
    min_gap = min(gaps) if gaps else 0.0
    max_radius = max(float(value) for value in scene["station_radii"])
    # This is a visible raster-clearance margin, not an answer-specific cue.
    value = min_gap - 2.0 * max_radius - float(scene["route_width"]) - 12.0
    return round(value, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 45.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # This world exposes every sampled field and declares no latent alias.
    return []
