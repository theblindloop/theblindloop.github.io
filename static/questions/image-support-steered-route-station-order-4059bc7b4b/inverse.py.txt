"""Independent pixel-only inverse for the route-station-order world.

This module intentionally contains no imports from the renderer, prompt, or
dataset code. It reconstructs a foreground graph, detects colorful station
symbols, classifies their silhouettes, and orders them by pixel geodesic from
the leftmost route start.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstain(ValueError):
    """The raster does not expose a reliable connected-route answer."""


def _background(array: np.ndarray) -> np.ndarray:
    border = np.concatenate(
        (
            array[:3].reshape(-1, 3),
            array[-3:].reshape(-1, 3),
            array[:, :3].reshape(-1, 3),
            array[:, -3:].reshape(-1, 3),
        ),
        axis=0,
    )
    return np.median(border, axis=0).astype(np.float32)


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        y0, x0 = int(y), int(x)
        if visited[y0, x0]:
            continue
        stack = [(y0, x0)]
        visited[y0, x0] = True
        pixels: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            pixels.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        stack.append((ny, nx))
        if len(pixels) >= minimum:
            components.append(np.asarray(pixels, dtype=np.int32))
    return components


def _distance_map(mask: np.ndarray, source: tuple[int, int]) -> np.ndarray:
    height, width = mask.shape
    distances = np.full((height, width), -1, dtype=np.int32)
    sy, sx = source
    if not (0 <= sy < height and 0 <= sx < width and mask[sy, sx]):
        raise OracleAbstain("route start is not foreground")
    distances[sy, sx] = 0
    queue: deque[tuple[int, int]] = deque([(sy, sx)])
    while queue:
        cy, cx = queue.popleft()
        next_distance = int(distances[cy, cx]) + 1
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if dy == 0 and dx == 0:
                    continue
                ny, nx = cy + dy, cx + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and distances[ny, nx] < 0
                ):
                    distances[ny, nx] = next_distance
                    queue.append((ny, nx))
    return distances


def _shape_from_component(component: np.ndarray) -> tuple[int, tuple[int, int], float]:
    ys = component[:, 0]
    xs = component[:, 1]
    min_y, max_y = int(ys.min()), int(ys.max())
    min_x, max_x = int(xs.min()), int(xs.max())
    height = max_y - min_y + 1
    width = max_x - min_x + 1
    occupancy = len(component) / float(height * width)
    center = ((min_y + max_y) // 2, (min_x + max_x) // 2)
    relative_y = (float(ys.mean()) - min_y) / float(max(1, height - 1))
    if occupancy >= 0.84:
        shape = 2  # square
    elif occupancy >= 0.66:
        shape = 0  # circle
    elif relative_y >= 0.57:
        shape = 1  # upward triangle
    else:
        shape = 3  # diamond
    return shape, center, occupancy


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or array.shape[2] != 3:
        raise OracleAbstain("image is not RGB")
    background = _background(array)
    foreground = np.linalg.norm(array - background[None, None, :], axis=2) > 18.0
    if int(foreground.sum()) < 300:
        raise OracleAbstain("too little visible route")

    ys, xs = np.nonzero(foreground)
    left = int(xs.min())
    candidates = np.argwhere(foreground[:, left])
    if not len(candidates):
        raise OracleAbstain("no left route endpoint")
    target_y = float(ys.mean())
    source_y = int(candidates[np.argmin(np.abs(candidates[:, 0] - target_y)), 0])
    distances = _distance_map(foreground, (source_y, left))

    channel_spread = array.max(axis=2) - array.min(axis=2)
    luminance = array.mean(axis=2)
    colorful = (channel_spread > 55.0) & (luminance > 35.0) & (luminance < 242.0)
    station_components = _components(colorful, minimum=80)
    if len(station_components) != 4:
        raise OracleAbstain(f"expected four station symbols, found {len(station_components)}")

    stations = []
    for component in station_components:
        shape, center, occupancy = _shape_from_component(component)
        cy, cx = center
        if distances[cy, cx] < 0:
            # A partially erased marker may have a center pixel outside the
            # remaining route. Use its nearest colored foreground pixel only
            # when that still lies on the connected route.
            reachable = distances[component[:, 0], component[:, 1]]
            reachable = reachable[reachable >= 0]
            if not len(reachable):
                raise OracleAbstain("station is disconnected from route start")
            path_distance = int(reachable.min())
        else:
            path_distance = int(distances[cy, cx])
        stations.append((path_distance, shape, occupancy))

    stations.sort(key=lambda item: item[0])
    order = [shape for _, shape, _ in stations]
    if len(set(order)) != 4:
        raise OracleAbstain("station silhouettes are ambiguous")
    if any(right - left < 30 for (left, _, _), (right, _, _) in zip(stations, stations[1:])):
        raise OracleAbstain("station path margin is too small")
    return "yes" if order == [0, 1, 2, 3] else "no"
