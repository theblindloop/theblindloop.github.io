"""Block-majority coarsening world.

A 24x24 lattice of small cells is printed inside a coloured frame, and heavier
coloured rules cut the lattice into an 8x8 arrangement of 3x3 blocks. Every
block holds either four or five dark cells out of nine, so no block is anywhere
near empty or full. The declared transform replaces each block by its own
majority vote (dark when at least five of its nine cells are dark), and the
resulting 8x8 pattern matches one of five named abstract symbols.

The fine scale is adversarial on purpose: the centre cell of every block is
dark exactly where a *different* symbol would be dark, so reading the lattice
without collapsing each block yields a decoy answer that is never the gold.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

BLOCKS = 8
BLOCK_SIDE = 3
CELLS = BLOCKS * BLOCK_SIDE
CENTER = 4
ON_COUNT = 5
OFF_COUNT = 4
SYMBOLS = ("plus", "chevron", "square", "ladder", "cross")
QUARANTINE_MARGIN = 2.0

_BACKGROUNDS = ([255, 255, 255], [250, 249, 245], [247, 250, 252])
_HAIRLINES = ([176, 176, 176], [182, 178, 172], [168, 172, 180])
_INKS = ([26, 26, 30], [42, 42, 42], [22, 28, 38], [48, 44, 40])
_RULES = ([202, 54, 42], [214, 122, 20], [172, 36, 126], [26, 128, 74], [38, 82, 204])
_QUANTA = (9, 10, 11)
_PAD = 22
_FRAME_W = 3
_RULE_W = 2


def _template(name: str) -> list[list[int]]:
    """Five 8x8 templates, each using exactly sixteen dark blocks."""

    grid = [[0] * BLOCKS for _ in range(BLOCKS)]
    for r in range(BLOCKS):
        for c in range(BLOCKS):
            if name == "plus":
                on = (r in (3, 4) and 2 <= c <= 5) or (c in (3, 4) and r in (1, 2, 5, 6))
            elif name == "chevron":
                stem = 5 - min(c, BLOCKS - 1 - c)
                on = r in (stem, stem - 1)
            elif name == "square":
                on = 2 <= r <= 6 and 2 <= c <= 6 and (r in (2, 6) or c in (2, 6))
            elif name == "ladder":
                on = r in (1, 3, 5, 7) and 2 <= c <= 5
            else:
                on = r == c or r + c == BLOCKS - 1
            grid[r][c] = 1 if on else 0
    return grid


TEMPLATES = {name: _template(name) for name in SYMBOLS}
INK_BLOCKS = {name: sum(map(sum, TEMPLATES[name])) for name in SYMBOLS}


def _distance(pattern: list[list[int]], template: list[list[int]]) -> int:
    return sum(
        1
        for r in range(BLOCKS)
        for c in range(BLOCKS)
        if pattern[r][c] != template[r][c]
    )


def _scored(pattern: list[list[int]]) -> list[tuple[int, int, str]]:
    """Template distances with a fixed public tie-break by declared symbol order."""

    return sorted(
        (_distance(pattern, TEMPLATES[name]), SYMBOLS.index(name), name)
        for name in SYMBOLS
    )


def _majority(scene: dict) -> list[list[int]]:
    """The 8x8 pattern the declared block-majority collapse reveals.

    Read only from the rendered field ``blocks``; no other scene key is used.
    """

    blocks = scene["blocks"]
    grid = [[0] * BLOCKS for _ in range(BLOCKS)]
    for r in range(BLOCKS):
        for c in range(BLOCKS):
            filled = len(set(int(i) for i in blocks[r * BLOCKS + c]))
            grid[r][c] = 1 if filled >= ON_COUNT else 0
    return grid


def _centers(scene: dict) -> list[list[int]]:
    """The decoy 8x8 pattern a reader gets from block centre cells alone."""

    blocks = scene["blocks"]
    return [
        [1 if CENTER in set(int(i) for i in blocks[r * BLOCKS + c]) else 0
         for c in range(BLOCKS)]
        for r in range(BLOCKS)
    ]


def _naive_center_symbol(scene: dict) -> str:
    return _scored(_centers(scene))[0][2]


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    for _ in range(800):
        symbol = SYMBOLS[rng.randrange(len(SYMBOLS))]
        decoy = SYMBOLS[rng.randrange(len(SYMBOLS))]
        if decoy == symbol:
            continue
        budget = rng.choice([0, 0, 1, 1, 2, 2, 3, 3, 4, 5, 6, 7, 8])
        coords = [[r, c] for r in range(BLOCKS) for c in range(BLOCKS)]
        if rng.random() < 0.5:
            rival = SYMBOLS[rng.randrange(len(SYMBOLS))]
            if rival == symbol:
                rival = SYMBOLS[(SYMBOLS.index(symbol) + 1) % len(SYMBOLS)]
            pool = [
                cell
                for cell in coords
                if TEMPLATES[symbol][cell[0]][cell[1]]
                != TEMPLATES[rival][cell[0]][cell[1]]
            ]
        else:
            pool = list(coords)
        rng.shuffle(pool)
        flips = sorted(pool[:budget])
        target = [row[:] for row in TEMPLATES[symbol]]
        for r, c in flips:
            target[r][c] = 1 - target[r][c]

        blocks: list[list[int]] = []
        for r in range(BLOCKS):
            for c in range(BLOCKS):
                want = ON_COUNT if target[r][c] else OFF_COUNT
                chosen = [CENTER] if TEMPLATES[decoy][r][c] else []
                others = [i for i in range(BLOCK_SIDE * BLOCK_SIDE) if i != CENTER]
                rng.shuffle(others)
                chosen.extend(others[: want - len(chosen)])
                blocks.append(sorted(chosen))

        scene = {
            "symbol": symbol,
            "decoy": decoy,
            "flips": flips,
            "blocks": blocks,
            "quantum": _QUANTA[rng.randrange(len(_QUANTA))],
            "jitter_x": rng.randint(-5, 5),
            "jitter_y": rng.randint(-5, 5),
            "bg": list(_BACKGROUNDS[rng.randrange(len(_BACKGROUNDS))]),
            "hairline": list(_HAIRLINES[rng.randrange(len(_HAIRLINES))]),
            "ink": list(_INKS[rng.randrange(len(_INKS))]),
            "rule": list(_RULES[rng.randrange(len(_RULES))]),
        }
        # The declared shortcut guard: the centre-cell reading must never be
        # the gold answer, so the transform is genuinely required.
        if _naive_center_symbol(scene) == analytic_gold(scene):
            continue
        return scene
    raise RuntimeError("scene sampling failed to satisfy the declared constraints")


def _canvas(scene: dict) -> tuple[int, int, int, int, int]:
    quantum = int(scene["quantum"])
    side = CELLS * quantum
    width = 2 * _PAD + side
    height = 2 * _PAD + side
    ox = _PAD + int(scene["jitter_x"])
    oy = _PAD + int(scene["jitter_y"])
    return width, height, ox, oy, quantum


def render(scene: dict) -> Image.Image:
    width, height, ox, oy, quantum = _canvas(scene)
    side = CELLS * quantum
    image = Image.new("RGB", (width, height), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    hair_rgb = tuple(scene["hairline"])
    ink_rgb = tuple(scene["ink"])
    rule_rgb = tuple(scene["rule"])

    for index in range(1, CELLS):
        offset = index * quantum
        draw.line([ox, oy + offset, ox + side - 1, oy + offset], fill=hair_rgb, width=1)
        draw.line([ox + offset, oy, ox + offset, oy + side - 1], fill=hair_rgb, width=1)

    blocks = scene["blocks"]
    for br in range(BLOCKS):
        for bc in range(BLOCKS):
            filled = set(int(i) for i in blocks[br * BLOCKS + bc])
            for index in filled:
                cr = br * BLOCK_SIDE + index // BLOCK_SIDE
                cc = bc * BLOCK_SIDE + index % BLOCK_SIDE
                x0 = ox + cc * quantum
                y0 = oy + cr * quantum
                draw.rectangle(
                    [x0 + 1, y0 + 1, x0 + quantum - 2, y0 + quantum - 2], fill=ink_rgb
                )

    for index in range(1, BLOCKS):
        offset = index * BLOCK_SIDE * quantum
        draw.line(
            [ox, oy + offset, ox + side - 1, oy + offset], fill=rule_rgb, width=_RULE_W
        )
        draw.line(
            [ox + offset, oy, ox + offset, oy + side - 1], fill=rule_rgb, width=_RULE_W
        )
    draw.rectangle(
        [ox - _FRAME_W, oy - _FRAME_W, ox + side + _FRAME_W - 1, oy + side + _FRAME_W - 1],
        outline=rule_rgb,
        width=_FRAME_W,
    )
    return image


def analytic_gold(scene: dict) -> str:
    return _scored(_majority(scene))[0][2]


def margin(scene: dict) -> float:
    ranking = _scored(_majority(scene))
    return float(ranking[1][0] - ranking[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings that must leave the analytic gold unchanged."""

    reordered = dict(scene)
    reordered["blocks"] = [list(reversed(list(cells))) for cells in scene["blocks"]]

    metadata = dict(scene)
    metadata["symbol"] = SYMBOLS[(SYMBOLS.index(scene["symbol"]) + 2) % len(SYMBOLS)]
    metadata["decoy"] = SYMBOLS[(SYMBOLS.index(scene["decoy"]) + 1) % len(SYMBOLS)]
    metadata["flips"] = list(reversed(list(scene["flips"])))
    return [("block_cell_listing_order", reordered), ("generative_metadata", metadata)]
