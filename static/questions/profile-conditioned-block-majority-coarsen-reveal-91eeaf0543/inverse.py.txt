"""Pixel-only inverse arm for the block-majority coarsening world.

This module receives a rendered image and nothing else. It locates the coloured
frame and block rules, recovers the 24x24 cell lattice, reads which small cells
are dark, executes the declared block-majority collapse on that raster
evidence, and names the closest 8x8 symbol template. It imports only PIL and
shares no code, constants file, or scene data with the renderer.
"""

from __future__ import annotations

from PIL import Image

ABSTAIN = "abstain"
_NAMES = ("plus", "chevron", "square", "ladder", "cross")
_SIDE = 8
_SUB = 3
_FINE = _SIDE * _SUB
_FRAME = 3
_MAJORITY = 5


def _templates() -> dict[str, list[list[int]]]:
    """Independently written copies of the five public 8x8 symbol templates."""

    out: dict[str, list[list[int]]] = {}
    for name in _NAMES:
        grid = [[0] * _SIDE for _ in range(_SIDE)]
        for row in range(_SIDE):
            for col in range(_SIDE):
                if name == "plus":
                    hit = (row in (3, 4) and 2 <= col <= 5) or (
                        col in (3, 4) and row in (1, 2, 5, 6)
                    )
                elif name == "chevron":
                    base = 5 - min(col, _SIDE - 1 - col)
                    hit = row == base or row == base - 1
                elif name == "square":
                    hit = (
                        2 <= row <= 6
                        and 2 <= col <= 6
                        and (row in (2, 6) or col in (2, 6))
                    )
                elif name == "ladder":
                    hit = row in (1, 3, 5, 7) and 2 <= col <= 5
                else:
                    hit = row == col or row + col == _SIDE - 1
                grid[row][col] = 1 if hit else 0
        out[name] = grid
    return out


def decision_from_image(image: Image.Image) -> str:
    return _decide(image)[0]


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    """Reported decision plus the runner-up gap in mismatching blocks."""

    return _decide(image)


def _decide(image: Image.Image) -> tuple[str, float]:
    rgb = image.convert("RGB")
    width, height = rgb.size
    if width < 120 or height < 120:
        return ABSTAIN, 0.0
    pixels = rgb.load()

    chroma_x: list[int] = []
    chroma_y: list[int] = []
    dark = [[False] * width for _ in range(height)]
    for y in range(height):
        dark_row = dark[y]
        for x in range(width):
            r, g, b = pixels[x, y]
            hi = max(r, g, b)
            lo = min(r, g, b)
            if hi - lo > 55:
                chroma_x.append(x)
                chroma_y.append(y)
            elif 0.299 * r + 0.587 * g + 0.114 * b < 110.0:
                dark_row[x] = True
    if len(chroma_x) < 200:
        return ABSTAIN, 0.0

    x0 = min(chroma_x) + _FRAME
    x1 = max(chroma_x) - _FRAME
    y0 = min(chroma_y) + _FRAME
    y1 = max(chroma_y) - _FRAME
    span_x = x1 - x0 + 1
    span_y = y1 - y0 + 1
    if span_x != span_y or span_x < 7 * _FINE or span_x % _FINE != 0:
        return ABSTAIN, 0.0
    quantum = span_x // _FINE
    reach = max(1, quantum // 4)

    fine = [[0] * _FINE for _ in range(_FINE)]
    for row in range(_FINE):
        cy = y0 + row * quantum + quantum // 2
        for col in range(_FINE):
            cx = x0 + col * quantum + quantum // 2
            hits = 0
            total = 0
            for yy in range(cy - reach, cy + reach + 1):
                if yy < 0 or yy >= height:
                    return ABSTAIN, 0.0
                dark_row = dark[yy]
                for xx in range(cx - reach, cx + reach + 1):
                    if xx < 0 or xx >= width:
                        return ABSTAIN, 0.0
                    total += 1
                    if dark_row[xx]:
                        hits += 1
            share = hits / total
            if 0.2 < share < 0.8:
                return ABSTAIN, 0.0
            fine[row][col] = 1 if share >= 0.8 else 0

    coarse = [[0] * _SIDE for _ in range(_SIDE)]
    for br in range(_SIDE):
        for bc in range(_SIDE):
            filled = sum(
                fine[br * _SUB + dr][bc * _SUB + dc]
                for dr in range(_SUB)
                for dc in range(_SUB)
            )
            if filled < 3 or filled > 6:
                return ABSTAIN, 0.0
            coarse[br][bc] = 1 if filled >= _MAJORITY else 0

    templates = _templates()
    scores = sorted(
        (
            sum(
                1
                for row in range(_SIDE)
                for col in range(_SIDE)
                if coarse[row][col] != templates[name][row][col]
            ),
            _NAMES.index(name),
            name,
        )
        for name in _NAMES
    )
    return scores[0][2], float(scores[1][0] - scores[0][0])
