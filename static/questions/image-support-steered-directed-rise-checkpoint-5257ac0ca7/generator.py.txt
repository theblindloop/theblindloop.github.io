"""Deterministic renderer and analytic arm for directed local route motion."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw

WIDTH = 800
HEIGHT = 600
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE_RGB = (27, 53, 76)
GREEN_RGB = (43, 174, 103)
RED_RGB = (218, 70, 66)
AMBER_RGB = (235, 151, 35)
VIOLET_RGB = (143, 84, 194)
QUARANTINE_MARGIN = 18.0


def _distance(first: Iterable[float], second: Iterable[float]) -> float:
    ax, ay = (float(value) for value in first)
    bx, by = (float(value) for value in second)
    return math.hypot(ax - bx, ay - by)


def _points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in scene["route_points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(cumulative[-1] + _distance(first, second))
    return cumulative, cumulative[-1]


def _point_at_fraction(
    points: list[tuple[float, float]], fraction: float
) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span <= 0.0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0)
    return points[-1]


def _segment_for_fraction(points: list[tuple[float, float]], fraction: float) -> int:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if target <= cumulative[index + 1] + 1e-9:
            return index
    return len(points) - 2


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a compact arch/valley route with one marker on each sloped leg."""

    rng = random.Random(int(seed))
    near_boundary = int(seed) % 6 == 0
    # The short route has a horizontal lead, an upward leg, a downward leg,
    # and a horizontal tail. A vertical reflection swaps which checkpoint is
    # upward without changing the route's local compactness.
    if near_boundary:
        top_y = 20.0
        left_y = 30.0
        right_y = 40.0
    else:
        top_y = 10.0
        left_y = 30.0
        right_y = 30.0
    base = [
        (0.0, left_y),
        (35.0, left_y),
        (70.0, top_y),
        (105.0, right_y),
        (160.0, right_y),
    ]
    mirror_x = bool(rng.randrange(2))
    mirror_y = bool(rng.randrange(2))
    max_x = max(point[0] for point in base)
    max_y = max(point[1] for point in base)
    transformed = []
    for x, y in base:
        if mirror_x:
            x = max_x - x
        if mirror_y:
            y = max_y - y
        transformed.append((x, y))

    # Place the trace across two neighboring 4x4 ablation columns and inside
    # one row. This randomizes absolute position while retaining a focal map.
    column = rng.randrange(7)
    tx = rng.uniform(column * 100.0 + 14.0, column * 100.0 + 20.0)
    row = rng.randrange(7)
    ty = rng.uniform(row * 75.0 + 24.0, row * 75.0 + 30.0)
    route_points = [[round(x + tx, 3), round(y + ty, 3)] for x, y in transformed]
    points = [(float(x), float(y)) for x, y in route_points]
    cumulative, total = _route_lengths(points)

    up_segment = 1 if not mirror_y else 2
    down_segment = 2 if not mirror_y else 1
    # Fractions are deliberately interior to the two sloped legs. The small
    # per-seed jitter prevents a fixed marker coordinate from being useful.
    up_t = rng.uniform(0.48, 0.52)
    down_t = rng.uniform(0.48, 0.52)
    up_fraction = (cumulative[up_segment] + up_t * (cumulative[up_segment + 1] - cumulative[up_segment])) / total
    down_fraction = (cumulative[down_segment] + down_t * (cumulative[down_segment + 1] - cumulative[down_segment])) / total
    up_label = "amber" if rng.randrange(2) == 0 else "violet"
    down_label = "violet" if up_label == "amber" else "amber"
    return {
        "route_points": route_points,
        "checkpoint_fractions": {
            up_label: round(up_fraction, 6),
            down_label: round(down_fraction, 6),
        },
        "route_width": round(rng.uniform(7.6, 9.2), 3),
        "marker_radius": round(rng.uniform(8.5, 9.5), 3),
        "orientation": int(mirror_x) + 2 * int(mirror_y),
        "translation": [round(tx, 3), round(ty, 3)],
        "scene_variant": "near_boundary" if near_boundary else "clear",
    }


def _scaled_points(points: list[tuple[float, float]]) -> list[tuple[int, int]]:
    return [(round(x * SCALE), round(y * SCALE)) for x, y in points]


def _draw_round_route(
    draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], width: int
) -> None:
    scaled = _scaled_points(points)
    radius = max(1, width // 2)
    draw.line(scaled, fill=ROUTE_RGB, width=width, joint="curve")
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ROUTE_RGB)


def _draw_port(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    r = round((radius + 1.5) * SCALE)
    draw.ellipse((x - r, y - r, x + r, y + r), fill=color)


def _draw_checkpoint(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    outer = round((radius + 3.0) * SCALE)
    width = max(2, round(4.0 * SCALE))
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), outline=color, width=width)


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize only visible route, ports, and checkpoint fields."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    route_width = max(1, round(float(scene["route_width"]) * SCALE))
    marker_radius = float(scene["marker_radius"])
    _draw_round_route(draw, points, route_width)
    colors = {"amber": AMBER_RGB, "violet": VIOLET_RGB}
    for label, fraction in scene["checkpoint_fractions"].items():
        _draw_checkpoint(draw, _point_at_fraction(points, float(fraction)), colors[str(label)], marker_radius)
    _draw_port(draw, points[0], GREEN_RGB, marker_radius)
    _draw_port(draw, points[-1], RED_RGB, marker_radius)
    # Restore a single connected dark centerline through every marker. The
    # colored checkpoint arcs remain exposed around the route.
    _draw_round_route(draw, points, route_width)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    points = _points(scene)
    labels = set(str(label) for label in scene["checkpoint_fractions"])
    if labels != {"amber", "violet"}:
        raise ValueError("scene needs amber and violet checkpoints")
    upward: list[str] = []
    for label, fraction in scene["checkpoint_fractions"].items():
        index = _segment_for_fraction(points, float(fraction))
        if points[index + 1][1] < points[index][1]:
            upward.append(str(label))
    if len(upward) != 1:
        raise ValueError("scene must contain exactly one upward checkpoint")
    return upward[0]


def margin(scene: dict[str, Any]) -> float:
    points = _points(scene)
    cumulative, total = _route_lengths(points)
    values: list[float] = []
    for fraction in scene["checkpoint_fractions"].values():
        index = _segment_for_fraction(points, float(fraction))
        segment_length = cumulative[index + 1] - cumulative[index]
        local = float(fraction) * total - cumulative[index]
        values.append(min(local, segment_length - local))
        values.append(abs(points[index + 1][1] - points[index][1]))
    return float(min(values))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # All label-bearing route geometry and checkpoint positions are exposed in
    # the final PNG; no hidden pixel-identical alias is claimed.
    return []
