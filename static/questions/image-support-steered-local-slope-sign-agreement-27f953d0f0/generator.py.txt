"""Deterministic scene sampler and raster renderer for local slope agreement."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH = 512
HEIGHT = 512
BACKGROUND = [255, 255, 255]
CYAN = [0, 151, 167]
ORANGE = [225, 104, 37]
INK = [41, 52, 66]
MUTED = [142, 153, 166]
PANEL = [247, 250, 252]
FRAME = [190, 202, 214]
GRID = 64
HALF_RUN = 20
LINE_WIDTH = 7
QUARANTINE_PX = 8.0


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _make_segment(cx: int, cy: int, rise_sign: int, magnitude: int) -> list[list[int]]:
    """Return endpoints whose right-minus-left page height has the requested sign.

    ``rise_sign == +1`` means the right endpoint is higher (smaller image y).
    """

    low_half = magnitude // 2
    high_half = magnitude - low_half
    left_y = cy + high_half if rise_sign > 0 else cy - high_half
    right_y = cy - low_half if rise_sign > 0 else cy + low_half
    return [[cx - HALF_RUN, left_y], [cx + HALF_RUN, right_y]]


def _outside_inset_distractors(
    rng: random.Random, inset_bbox: list[int], count: int = 10
) -> list[list[int]]:
    x0, y0, x1, y1 = inset_bbox
    marks: list[list[int]] = []
    attempts = 0
    while len(marks) < count and attempts < 500:
        attempts += 1
        x = rng.randrange(36, WIDTH - 36)
        y = rng.randrange(36, HEIGHT - 36)
        radius = rng.choice((3, 4, 5))
        if x0 - 30 <= x <= x1 + 30 and y0 - 30 <= y <= y1 + 30:
            continue
        if any((x - ox) ** 2 + (y - oy) ** 2 < 42**2 for ox, oy, _ in marks):
            continue
        marks.append([x, y, radius])
    return marks


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene deterministically from one integer seed."""

    rng = random.Random(int(seed))
    same = int(seed) % 2 == 0
    phase = (int(seed) // 2) % 2
    cyan_sign = 1 if phase == 0 else -1
    orange_sign = cyan_sign if same else -cyan_sign

    magnitudes = (5, 9, 14, 20, 28, 34)
    cyan_mag = magnitudes[(int(seed) // 4) % len(magnitudes)]
    orange_mag = magnitudes[(int(seed) // 7 + 2) % len(magnitudes)]

    left_col = rng.choice((2, 3))
    row = rng.choice((2, 3, 4, 5))
    jitter_x = rng.choice((-3, 0, 3))
    jitter_y = rng.choice((-3, 0, 3))
    cyan_center = [left_col * GRID + GRID // 2 + jitter_x, row * GRID + GRID // 2 + jitter_y]
    orange_center = [cyan_center[0] + 2 * GRID, cyan_center[1]]

    inset_bbox = [
        cyan_center[0] - 38,
        cyan_center[1] - 48,
        orange_center[0] + 38,
        orange_center[1] + 48,
    ]
    segments = {
        "cyan": _make_segment(*cyan_center, cyan_sign, cyan_mag),
        "orange": _make_segment(*orange_center, orange_sign, orange_mag),
    }
    palette = {
        "cyan": CYAN.copy(),
        "orange": ORANGE.copy(),
        "ink": INK.copy(),
        "muted": MUTED.copy(),
        "panel": PANEL.copy(),
        "frame": FRAME.copy(),
    }
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": BACKGROUND.copy(),
        "palette": palette,
        "inset_bbox": inset_bbox,
        "segments": segments,
        "distractors": _outside_inset_distractors(rng, inset_bbox),
    }


def _rgb(values: Iterable[int]) -> tuple[int, int, int]:
    return tuple(int(value) for value in values)


def _ordered(points: list[list[int]]) -> list[tuple[int, int]]:
    return sorted((tuple(map(int, point)) for point in points), key=lambda point: (point[0], point[1]))


def render(scene: dict) -> Image.Image:
    """Render only visible scene fields into a single RGB PNG-ready image."""

    width = int(scene["width"])
    height = int(scene["height"])
    background = _rgb(scene["background"])
    palette = scene["palette"]
    inset = [int(value) for value in scene["inset_bbox"]]
    segments = scene["segments"]
    distractors = scene["distractors"]

    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle(
        inset,
        radius=14,
        fill=_rgb(palette["panel"]),
        outline=_rgb(palette["frame"]),
        width=2,
    )
    mid_y = (inset[1] + inset[3]) // 2
    draw.line(
        [(inset[0] + 16, mid_y), (inset[2] - 16, mid_y)],
        fill=_rgb(palette["frame"]),
        width=1,
    )
    label_font = _font(13, bold=True)
    note_font = _font(12)
    draw.text(
        ((inset[0] + inset[2]) // 2, inset[1] + 8),
        "READ LEFT  →  RIGHT",
        fill=_rgb(palette["ink"]),
        anchor="ma",
        font=note_font,
    )

    for x, y, radius in distractors:
        x, y, radius = int(x), int(y), int(radius)
        draw.ellipse(
            [x - radius, y - radius, x + radius, y + radius],
            outline=_rgb(palette["muted"]),
            width=1,
        )

    for name in ("cyan", "orange"):
        points = _ordered(segments[name])
        color = _rgb(palette[name])
        draw.line(points, fill=color, width=LINE_WIDTH)
        for x, y in points:
            draw.ellipse([x - 4, y - 4, x + 4, y + 4], fill=color)
        center_x = round(sum(point[0] for point in points) / 2)
        draw.text(
            (center_x, inset[3] - 7),
            name.upper(),
            fill=_rgb(palette["ink"]),
            anchor="ms",
            font=label_font,
        )
    return image


def _vertical_change(points: list[list[int]]) -> float:
    left, right = _ordered(points)
    return float(left[1] - right[1])


def analytic_gold(scene: dict) -> str:
    segments = scene["segments"]
    cyan_change = _vertical_change(segments["cyan"])
    orange_change = _vertical_change(segments["orange"])
    if cyan_change == 0.0 or orange_change == 0.0:
        raise ValueError("horizontal segment has undefined binary trend")
    return "yes" if cyan_change * orange_change > 0.0 else "no"


def margin(scene: dict) -> float:
    segments = scene["segments"]
    return min(
        abs(_vertical_change(segments["cyan"])),
        abs(_vertical_change(segments["orange"])),
    )


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["segments"]["cyan"].reverse()
    twin["segments"]["orange"].reverse()
    return [("reverse_both_endpoint_lists", twin)]
