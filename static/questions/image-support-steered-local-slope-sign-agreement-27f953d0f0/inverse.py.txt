"""Independent PNG-only inverse arm for paired slope-sign agreement."""

from __future__ import annotations

import numpy as np
from PIL import Image


CYAN = np.array([0, 151, 167], dtype=np.int16)
ORANGE = np.array([225, 104, 37], dtype=np.int16)
MIN_PIXELS = 70
MIN_X_SPAN = 28
MIN_FITTED_CHANGE = 2.0


class OracleAbstention(ValueError):
    """The final raster does not expose both measurable colored segments."""


def _segment_change(array: np.ndarray, color: np.ndarray, name: str) -> float:
    delta = np.max(np.abs(array.astype(np.int16) - color[None, None, :]), axis=2)
    mask = delta <= 2
    ys, xs = np.nonzero(mask)
    if len(xs) < MIN_PIXELS:
        raise OracleAbstention(f"{name} segment has only {len(xs)} pixels")
    unique_x = np.unique(xs)
    if int(unique_x[-1] - unique_x[0]) < MIN_X_SPAN:
        raise OracleAbstention(f"{name} segment horizontal span is incomplete")
    column_y = np.array([ys[xs == x].mean() for x in unique_x], dtype=np.float64)
    design = np.column_stack([unique_x.astype(np.float64), np.ones(len(unique_x))])
    slope, _intercept = np.linalg.lstsq(design, column_y, rcond=None)[0]
    fitted_change = -float(slope) * float(unique_x[-1] - unique_x[0])
    if abs(fitted_change) < MIN_FITTED_CHANGE:
        raise OracleAbstention(f"{name} segment is too close to horizontal")
    return fitted_change


def decision_from_image(image: Image.Image) -> str:
    """Recover the binary decision from RGB pixels and no scene information."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    cyan_change = _segment_change(array, CYAN, "cyan")
    orange_change = _segment_change(array, ORANGE, "orange")
    return "yes" if cyan_change * orange_change > 0.0 else "no"
