"""Deterministic bubble-plot renderer and independent analytic arm."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 640
HEIGHT = 640
CELL = 80
BACKGROUND = (249, 248, 244)
INK = (54, 60, 66)
GRID = (207, 212, 211)
PALETTES = (
    (16, 132, 166),
    (196, 82, 48),
    (59, 139, 92),
    (126, 80, 166),
    (185, 116, 18),
)
PAPER_TONES = (
    (239, 242, 240),
    (241, 239, 235),
    (236, 240, 242),
)
PLACEMENTS = (
    ((2, 2), (4, 3)),
    ((5, 2), (3, 3)),
    ((2, 5), (4, 4)),
    ((5, 5), (3, 4)),
)
QUARANTINE_RADIUS_PX = 2
QUARANTINE_HEIGHT_PX = 12


def _font(size: int, *, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict[str, Any]:
    """Return pair-balanced scenes with no size-only or height-only label cue."""

    seed = int(seed)
    pair_seed = seed // 2
    rng = random.Random(0xBABB_1E20 + pair_seed * 1_000_003)
    (cell_ax, cell_ay), (cell_bx, cell_by) = PLACEMENTS[pair_seed % len(PLACEMENTS)]
    x_a, y_a = cell_ax * CELL + CELL // 2, cell_ay * CELL + CELL // 2
    x_b, y_b = cell_bx * CELL + CELL // 2, cell_by * CELL + CELL // 2

    small = 8 + rng.randrange(4)
    gap = 2 if pair_seed % 6 == 0 else 5 + pair_seed % 3
    large = small + gap
    if large > 18:
        small -= large - 18
        large = 18

    higher_is_a = y_a < y_b
    concordant = seed % 2 == 0
    larger_is_a = higher_is_a if concordant else not higher_is_a
    radius_a, radius_b = (large, small) if larger_is_a else (small, large)

    return {
        "x_a": x_a,
        "y_a": y_a,
        "radius_a": radius_a,
        "x_b": x_b,
        "y_b": y_b,
        "radius_b": radius_b,
        "bubble_rgb": list(PALETTES[pair_seed % len(PALETTES)]),
        "paper_rgb": list(PAPER_TONES[(pair_seed // 3) % len(PAPER_TONES)]),
    }


def _rank_values(scene: dict[str, Any]) -> tuple[int, int, int, int]:
    y_a = int(scene["y_a"])
    radius_a = int(scene["radius_a"])
    y_b = int(scene["y_b"])
    radius_b = int(scene["radius_b"])
    if radius_a == radius_b:
        raise ValueError("bubble areas must not tie")
    if y_a == y_b:
        raise ValueError("bubble centers must not share a height")
    return y_a, radius_a, y_b, radius_b


def analytic_gold(scene: dict[str, Any]) -> str:
    y_a, radius_a, y_b, radius_b = _rank_values(scene)
    larger_is_a = radius_a > radius_b
    higher_is_a = y_a < y_b
    return "yes" if larger_is_a == higher_is_a else "no"


def margin(scene: dict[str, Any]) -> float:
    """Minimum visible pixel distance to either rank-tie boundary."""

    y_a, radius_a, y_b, radius_b = _rank_values(scene)
    return float(min(abs(radius_a - radius_b), abs(y_a - y_b)))


def is_quarantined(scene: dict[str, Any]) -> bool:
    y_a, radius_a, y_b, radius_b = _rank_values(scene)
    return (
        abs(radius_a - radius_b) <= QUARANTINE_RADIUS_PX
        or abs(y_a - y_b) <= QUARANTINE_HEIGHT_PX
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    for suffix in ("x", "y", "radius"):
        key_a = f"{suffix}_a"
        key_b = f"{suffix}_b"
        twin[key_a], twin[key_b] = scene[key_b], scene[key_a]
    return [("swap_unrendered_bubble_identities", twin)]


def render(scene: dict[str, Any]) -> Image.Image:
    x_a = int(scene["x_a"])
    y_a = int(scene["y_a"])
    radius_a = int(scene["radius_a"])
    x_b = int(scene["x_b"])
    y_b = int(scene["y_b"])
    radius_b = int(scene["radius_b"])
    bubble = tuple(int(value) for value in scene["bubble_rgb"])
    paper = tuple(int(value) for value in scene["paper_rgb"])
    _rank_values(scene)

    if not (8 <= radius_a <= 18 and 8 <= radius_b <= 18):
        raise ValueError("bubble radii are outside the calibrated range")
    if min(x_a, x_b, y_a, y_b) < 80 or max(x_a, x_b, y_a, y_b) > 560:
        raise ValueError("bubble centers must stay inside the bounded plot field")
    if abs(x_a - x_b) < 120 or abs(y_a - y_b) < 40:
        raise ValueError("bubbles must be visibly separated in both measurements")
    if len(bubble) != 3 or len(paper) != 3:
        raise ValueError("RGB fields must contain three channels")

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    left = min(x_a, x_b) - 58
    right = max(x_a, x_b) + 58
    top = min(y_a, y_b) - 68
    bottom = max(y_a, y_b) + 68
    draw.rounded_rectangle((left, top, right, bottom), radius=14, fill=paper, outline=GRID, width=2)

    # Grayscale scientific context is deliberately nondiagnostic to the
    # saturated-color inverse arm.
    for fraction in (1, 2, 3):
        gx = left + fraction * (right - left) // 4
        gy = top + fraction * (bottom - top) // 4
        draw.line((gx, top + 35, gx, bottom - 20), fill=GRID, width=1)
        draw.line((left + 24, gy, right - 18, gy), fill=GRID, width=1)
    draw.text((left + 16, top + 10), "LOCAL BUBBLE RANK", font=_font(13, bold=True), fill=INK)
    draw.line((left + 17, bottom - 24, left + 17, top + 44), fill=INK, width=2)
    draw.polygon(
        ((left + 17, top + 37), (left + 12, top + 47), (left + 22, top + 47)),
        fill=INK,
    )
    draw.text((left + 27, top + 38), "higher", font=_font(11), fill=INK)

    for x, y, radius in ((x_a, y_a, radius_a), (x_b, y_b, radius_b)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=bubble)
    return image
