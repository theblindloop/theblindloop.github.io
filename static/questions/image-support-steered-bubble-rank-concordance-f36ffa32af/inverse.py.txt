"""Independent final-PNG inverse arm for local bubble rank concordance.

The inverse receives only raster pixels. It segments saturated ink, discovers
two filled circular components, measures each component's pixel area and
centroid, and compares area rank with vertical-center rank. It uses no scene,
renderer values, filenames, metadata, or analytic-gold implementation.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _saturated_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    high = array.max(axis=2)
    low = array.min(axis=2)
    return ((high - low) >= 58) & (high >= 100) & (low <= 190)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) < 40:
            raise ValueError("unexpected saturated speckle or damaged bubble fragment")
        found.append(np.asarray(points, dtype=np.int32))
    return found


def _measure_disk(component: np.ndarray) -> tuple[float, float, int]:
    ys = component[:, 0]
    xs = component[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if not (15 <= width <= 39 and 15 <= height <= 39 and abs(width - height) <= 1):
        raise ValueError("saturated component does not have a calibrated circular extent")
    area = int(len(component))
    radius = 0.25 * (width + height - 2)
    fill_ratio = area / (math.pi * radius * radius)
    if not 0.88 <= fill_ratio <= 1.20:
        raise ValueError("saturated component is not a filled disk")
    center_x = float(xs.mean())
    center_y = float(ys.mean())
    if abs(center_x - (x0 + x1) / 2.0) > 0.35 or abs(center_y - (y0 + y1) / 2.0) > 0.35:
        raise ValueError("saturated component is not centroid-symmetric")
    return center_x, center_y, area


def decision_from_image(image: Image.Image) -> str:
    components = _components(_saturated_mask(image))
    if len(components) != 2:
        raise ValueError("exactly two disconnected saturated bubbles are required")
    first, second = (_measure_disk(component) for component in components)
    if abs(first[1] - second[1]) < 8.0:
        raise ValueError("bubble height rank is unresolved")
    if abs(first[2] - second[2]) < 60:
        raise ValueError("bubble area rank is unresolved")
    larger = first if first[2] > second[2] else second
    smaller = second if larger is first else first
    return "yes" if larger[1] < smaller[1] else "no"
