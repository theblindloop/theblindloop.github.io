"""Deterministic raster world for an identity-bound zone-coverage race."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1280, 720
BACKGROUND = (245, 247, 251)
PAPER = (255, 255, 255)
CARD = (250, 251, 253)
INK = (28, 37, 51)
MUTED = (94, 107, 128)
BORDER = (207, 214, 225)
LABELS = tuple("ABCD")
ZONE_NAMES = ("BLUE", "AMBER", "GREEN")
ZONE_FILLS = ((222, 239, 252), (253, 241, 210), (222, 245, 232))
ZONE_INKS = ((38, 103, 158), (153, 103, 22), (35, 122, 77))

PALETTES = (
    ((220, 52, 77), (57, 92, 205), (137, 64, 190), (225, 112, 28)),
    ((198, 57, 128), (32, 137, 163), (112, 72, 196), (218, 91, 49)),
    ((192, 55, 48), (48, 103, 190), (146, 64, 166), (210, 124, 27)),
)

_PIP_OFFSETS = (
    ((0, 0),),
    ((-6, 0), (6, 0)),
    ((0, -6), (-6, 5), (6, 5)),
    ((-6, -6), (6, -6), (-6, 6), (6, 6)),
)


def _font(size: int, bold: bool = True) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _completion_panel(states: list[list[int]], identity: int) -> int | None:
    visited: set[int] = set()
    for panel, row in enumerate(states):
        visited.add(int(row[identity]))
        if len(visited) == 3:
            return panel
    return None


def _winner_and_gap(states: list[list[int]]) -> tuple[int, int, int]:
    horizon = len(states)
    completions = [
        (_completion_panel(states, identity) if _completion_panel(states, identity) is not None else horizon)
        for identity in range(4)
    ]
    earliest = min(completions)
    if earliest >= horizon:
        raise ValueError("no identity completes all three zones")
    winners = [identity for identity, panel in enumerate(completions) if panel == earliest]
    if len(winners) != 1:
        raise ValueError("earliest three-zone completion is not unique")
    runner_up = sorted(completions)[1]
    return winners[0], earliest, runner_up - earliest


def _path_for_completion(rng: random.Random, panel: int) -> list[int]:
    zones = [0, 1, 2]
    rng.shuffle(zones)
    a, b, c = zones
    if panel == 2:
        return [a, b, c, rng.randrange(3), rng.randrange(3)]
    if panel == 3:
        return [a, b, rng.choice((a, b)), c, rng.randrange(3)]
    if panel == 4:
        return [a, b, rng.choice((a, b)), rng.choice((a, b)), c]
    raise ValueError("completion panel must be 2, 3, or 4")


def _incomplete_path(rng: random.Random) -> list[int]:
    zones = [0, 1, 2]
    rng.shuffle(zones)
    a, b = zones[:2]
    path = [rng.choice((a, b)) for _ in range(5)]
    path[0], path[1] = a, b
    return path


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced five-panel race with a one-panel runner-up gap."""
    seed = int(seed)
    rng = random.Random(seed ^ 0x73C0A9)
    target = seed % 4
    target_completion = 2 + ((seed // 4) % 3)
    paths: list[list[int] | None] = [None] * 4
    paths[target] = _path_for_completion(rng, target_completion)

    distractors = [identity for identity in range(4) if identity != target]
    rng.shuffle(distractors)
    runner, other_a, other_b = distractors
    if target_completion < 4:
        paths[runner] = _path_for_completion(rng, target_completion + 1)
    else:
        paths[runner] = _incomplete_path(rng)
    paths[other_a] = _incomplete_path(rng)
    paths[other_b] = _incomplete_path(rng)

    states = [[int(paths[identity][panel]) for identity in range(4)] for panel in range(5)]  # type: ignore[index]
    winner, completion, gap = _winner_and_gap(states)
    if winner != target or completion != target_completion or gap != 1:
        raise AssertionError("sampler failed to construct the intended coverage race")

    target_color = (seed // 12) % 4
    color_order = [-1] * 4
    color_order[target] = target_color
    remaining_colors = [value for value in range(4) if value != target_color]
    rng.shuffle(remaining_colors)
    for identity, color in zip((i for i in range(4) if i != target), remaining_colors, strict=True):
        color_order[identity] = color

    zone_orders = []
    slot_flips = []
    for _ in range(5):
        order = [0, 1, 2]
        rng.shuffle(order)
        zone_orders.append(order)
        slot_flips.append(bool(rng.randrange(2)))

    return {
        "states": states,
        "zone_orders": zone_orders,
        "slot_flips": slot_flips,
        "color_order": color_order,
        "palette_variant": (seed // 48) % len(PALETTES),
        "layout_dx": rng.randint(-5, 5),
        "layout_dy": rng.randint(-4, 4),
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    states = [[int(value) for value in panel] for panel in scene["states"]]
    winner, _panel, _gap = _winner_and_gap(states)
    return LABELS[winner]


def margin(scene: dict[str, Any]) -> float:
    states = [[int(value) for value in panel] for panel in scene["states"]]
    _winner, _panel, gap = _winner_and_gap(states)
    return float(gap)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        return margin(scene) < 1.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _centered_text(draw: ImageDraw.ImageDraw, center: tuple[float, float], text: str, font: Any, fill: tuple[int, int, int]) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((center[0] - (box[2] - box[0]) / 2, center[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], identity: int, color: tuple[int, int, int]) -> None:
    cx, cy = center
    radius = 18
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color)
    for ox, oy in _PIP_OFFSETS[identity]:
        draw.ellipse((cx + ox - 2, cy + oy - 2, cx + ox + 2, cy + oy + 2), fill=INK)


def render(scene: dict[str, Any]) -> Image.Image:
    states = [[int(value) for value in panel] for panel in scene["states"]]
    zone_orders = [[int(value) for value in panel] for panel in scene["zone_orders"]]
    slot_flips = [bool(value) for value in scene["slot_flips"]]
    color_order = [int(value) for value in scene["color_order"]]
    palette_variant = int(scene["palette_variant"])
    dx, dy = int(scene["layout_dx"]), int(scene["layout_dy"])
    if len(states) != 5 or any(len(panel) != 4 or any(zone not in (0, 1, 2) for zone in panel) for panel in states):
        raise ValueError("states must be a five by four zone ledger")
    if len(zone_orders) != 5 or any(sorted(order) != [0, 1, 2] for order in zone_orders):
        raise ValueError("each panel needs all three zone rows")
    if len(slot_flips) != 5 or sorted(color_order) != list(range(4)):
        raise ValueError("invalid visible nuisance fields")
    palette = PALETTES[palette_variant]

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((17, 14, WIDTH - 17, HEIGHT - 14), radius=24, fill=PAPER, outline=(220, 226, 235), width=2)
    draw.text((42, 31), "FIRST COMPLETE THREE-ZONE TOUR", font=_font(24), fill=INK)
    draw.text((42, 67), "Read left to right. Track each pip identity's visited BLUE, AMBER, and GREEN bands.", font=_font(16, False), fill=MUTED)
    draw.text((42, 104), "PIP KEY", font=_font(14), fill=INK)
    draw.text((119, 104), "A = 1     B = 2     C = 3     D = 4", font=_font(15, False), fill=MUTED)
    draw.line((42, 142, WIDTH - 42, 142), fill=(221, 226, 235), width=2)

    panel_w, panel_h, gap = 224, 510, 24
    start_x = (WIDTH - (5 * panel_w + 4 * gap)) // 2 + dx
    top = 158 + dy
    zone_top, zone_h, zone_gap = top + 62, 126, 10
    for panel in range(5):
        left = start_x + panel * (panel_w + gap)
        right = left + panel_w
        draw.rounded_rectangle((left, top, right, top + panel_h), radius=15, fill=CARD, outline=BORDER, width=3)
        _centered_text(draw, ((left + right) / 2, top + 27), f"PANEL {panel + 1}", _font(16), INK)
        for row, zone in enumerate(zone_orders[panel]):
            y0 = zone_top + row * (zone_h + zone_gap)
            y1 = y0 + zone_h
            draw.rounded_rectangle((left + 12, y0, right - 12, y1), radius=10, fill=ZONE_FILLS[zone], outline=ZONE_INKS[zone], width=2)
            draw.text((left + 20, y0 + 9), ZONE_NAMES[zone], font=_font(10), fill=ZONE_INKS[zone])
            members = [identity for identity in range(4) if states[panel][identity] == zone]
            if slot_flips[panel]:
                members.reverse()
            count = len(members)
            if count:
                x_lo, x_hi = left + 64, right - 40
                xs = [(x_lo + x_hi) // 2] if count == 1 else [round(x_lo + i * (x_hi - x_lo) / (count - 1)) for i in range(count)]
                for x, identity in zip(xs, members, strict=True):
                    _draw_token(draw, (int(x), y0 + 75), identity, palette[color_order[identity]])
        if panel < 4:
            arrow_x, arrow_y = right + gap // 2, top + panel_h // 2
            draw.polygon(((arrow_x - 5, arrow_y - 8), (arrow_x + 8, arrow_y), (arrow_x - 5, arrow_y + 8)), fill=(137, 149, 168))
    return image
