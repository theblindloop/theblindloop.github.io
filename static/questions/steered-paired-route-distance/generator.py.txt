"""Deterministic renderer for two disconnected branched route networks."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

BACKGROUND = (247, 244, 235)
NAVY = (29, 52, 79)
GREEN = (28, 171, 95)
RED = (222, 66, 72)
MUTED = (104, 108, 109)
WIDTH = 9


def _network(rng: random.Random, row: str, offset: int) -> dict:
    x0 = rng.randint(170, 178)
    x3 = x0 + 170
    x1 = x0 + rng.randint(58, 70)
    x2 = x0 + rng.randint(110, 124)
    if row == "top":
        base = rng.randint(126, 136)
        sign = 1
    else:
        base = rng.randint(376, 386)
        sign = -1
    bend_y = base + sign * offset
    main = [[x0, base], [x1, base], [x1, bend_y], [x2, bend_y], [x2, base], [x3, base]]

    # Main length is 170 + 2*offset.  The three dead ends make total
    # centerline ink 360 in both networks, blocking a component-area shortcut.
    branch_total = 190 - 2 * offset
    a = branch_total // 3
    branch_lengths = [a, a, branch_total - 2 * a]
    rng.shuffle(branch_lengths)
    anchors = [x0 + 20, x0 + 40, x3 - 24]
    branches = [
        [[x, base], [x, base - sign * length]]
        for x, length in zip(anchors, branch_lengths)
    ]
    return {"row": row, "main": main, "branches": branches}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    short_offset = rng.randint(20, 27)
    long_offset = rng.randint(58, 66)
    top_is_long = int(seed) % 2 == 0
    offsets = [long_offset, short_offset] if top_is_long else [short_offset, long_offset]
    return {
        "canvas": [512, 512],
        "networks": [
            _network(rng, "top", offsets[0]),
            _network(rng, "bottom", offsets[1]),
        ],
    }


def _main_length(network: dict) -> float:
    points = network["main"]
    return float(sum(abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:])))


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for network in scene["networks"]:
        start = tuple(network["main"][0])
        finish = tuple(network["main"][-1])
        # Rings go down first so the navy centerline remains visibly continuous.
        for center, color in ((start, GREEN), (finish, RED)):
            x, y = center
            draw.ellipse((x - 13, y - 13, x + 13, y + 13), outline=color, width=6)
        draw.line([tuple(p) for p in network["main"]], fill=NAVY, width=WIDTH, joint="curve")
        for branch in network["branches"]:
            draw.line([tuple(p) for p in branch], fill=NAVY, width=WIDTH)
    # Labels state the answer vocabulary but are deliberately not used by the oracle.
    draw.text((43, 123), "TOP", fill=MUTED)
    draw.text((30, 373), "BOTTOM", fill=MUTED)
    return image


def analytic_gold(scene: dict) -> str:
    networks = scene["networks"]
    lengths = []
    for network in networks:
        points = network["main"]
        lengths.append(float(sum(abs(b[0] - a[0]) + abs(b[1] - a[1]) for a, b in zip(points, points[1:]))))
    return "top" if lengths[0] > lengths[1] else "bottom"


def margin(scene: dict) -> float:
    networks = scene["networks"]
    return abs(_main_length(networks[0]) - _main_length(networks[1]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
