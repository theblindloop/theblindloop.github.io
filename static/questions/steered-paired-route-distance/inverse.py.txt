"""Pixel-only graph-distance oracle; this module has no world-state imports."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

NAVY = np.array([29, 52, 79], dtype=np.int16)
GREEN = np.array([28, 171, 95], dtype=np.int16)
RED = np.array([222, 66, 72], dtype=np.int16)


def _mask(rgb: np.ndarray, target: np.ndarray, tolerance: int = 18) -> np.ndarray:
    return np.max(np.abs(rgb.astype(np.int16) - target), axis=2) <= tolerance


def _row_centers(mask: np.ndarray) -> list[tuple[float, float]]:
    centers: list[tuple[float, float]] = []
    split = mask.shape[0] // 2
    for lo, hi in ((0, split), (split, mask.shape[0])):
        y, x = np.nonzero(mask[lo:hi])
        if len(x) < 30:
            raise ValueError("endpoint ring is missing")
        centers.append((float(x.mean()), float((y + lo).mean())))
    return centers


def _graph_distance(mask: np.ndarray, start: tuple[float, float], finish: tuple[float, float]) -> int:
    height, width = mask.shape
    sx, sy = start
    fx, fy = finish
    yy, xx = np.ogrid[:height, :width]
    starts = mask & ((xx - sx) ** 2 + (yy - sy) ** 2 <= 36)
    goals = mask & ((xx - fx) ** 2 + (yy - fy) ** 2 <= 36)
    if not starts.any() or not goals.any():
        raise ValueError("endpoint ring does not touch a route")
    distance = np.full(mask.shape, -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for y, x in zip(*np.nonzero(starts)):
        distance[y, x] = 0
        queue.append((int(y), int(x)))
    while queue:
        y, x = queue.popleft()
        if goals[y, x]:
            return int(distance[y, x])
        nd = int(distance[y, x]) + 1
        for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
            ny, nx = y + dy, x + dx
            if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and distance[ny, nx] < 0:
                distance[ny, nx] = nd
                queue.append((ny, nx))
    raise ValueError("route is broken")


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"))
    navy = _mask(rgb, NAVY)
    greens = _row_centers(_mask(rgb, GREEN))
    reds = _row_centers(_mask(rgb, RED))
    distances = []
    for index in range(2):
        row_mask = navy.copy()
        split = rgb.shape[0] // 2
        if index == 0:
            row_mask[split:, :] = False
        else:
            row_mask[:split, :] = False
        distances.append(_graph_distance(row_mask, greens[index], reds[index]))
    if abs(distances[0] - distances[1]) < 30:
        raise ValueError("pixel distance comparison is ambiguous")
    return "top" if distances[0] > distances[1] else "bottom"
