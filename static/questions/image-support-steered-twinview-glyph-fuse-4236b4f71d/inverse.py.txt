"""Pixel-only inverse arm for the twin-view glyph fusion world.

Receives one rendered RGB image and nothing else. It segments the framed tiles
by their frame colour, recovers each tile's 5x5 cell states from the raster,
fuses the two shuttered slate-framed views into one partially known glyph, and
returns the frame colour of the unique complete tile consistent with the fused
evidence. Anything ambiguous, damaged or under-determined raises, which the
harness treats as abstention.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 5
STROKE = 6
CELL = 26
TILE_SIDE = GRID * CELL + 2 * STROKE

BOARD = (247, 243, 233)
CELL_BG = (252, 252, 250)
INK = (26, 28, 34)
SHUTTER = (135, 138, 146)
VIEW_FRAME = (58, 66, 84)

FRAME_COLOURS = {
    "red": (206, 42, 42),
    "orange": (232, 124, 28),
    "green": (46, 150, 62),
    "teal": (26, 158, 168),
    "blue": (48, 86, 208),
    "purple": (140, 58, 180),
}

N_CELLS = GRID * GRID
N_VIEWS = 2
N_TILES = 4
MIN_FRAME_PIXELS = 900


def _mask(array: np.ndarray, colour) -> np.ndarray:
    return np.all(array == np.asarray(colour, dtype=np.uint8), axis=-1)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int]]:
    """Bounding boxes of 4-connected true components, coarse but exact enough."""
    points = np.argwhere(mask)
    if points.size == 0:
        return []
    todo = {(int(y), int(x)) for y, x in points}
    boxes = []
    while todo:
        start = todo.pop()
        stack = [start]
        ys = [start[0]]
        xs = [start[1]]
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in todo:
                    todo.discard((ny, nx))
                    stack.append((ny, nx))
                    ys.append(ny)
                    xs.append(nx)
        boxes.append((min(ys), min(xs), max(ys), max(xs)))
    return boxes


def _read_tile(array: np.ndarray, box) -> list[int | None]:
    """Cell states of one framed tile: 1 filled, 0 empty, None shuttered."""
    top, left, bottom, right = box
    height = bottom - top + 1
    width = right - left + 1
    if abs(height - TILE_SIDE) > 2 or abs(width - TILE_SIDE) > 2:
        raise ValueError("framed tile has an unexpected size")
    cells: list[int | None] = []
    for index in range(N_CELLS):
        row, col = divmod(index, GRID)
        cy = top + STROKE + row * CELL + CELL // 2
        cx = left + STROKE + col * CELL + CELL // 2
        patch = array[cy - 2 : cy + 3, cx - 2 : cx + 3].reshape(-1, 3)
        first = patch[0]
        if not np.all(patch == first):
            raise ValueError("cell centre is not a uniform colour")
        value = tuple(int(v) for v in first)
        if value == INK:
            cells.append(1)
        elif value == CELL_BG:
            cells.append(0)
        elif value == SHUTTER:
            cells.append(None)
        else:
            raise ValueError(f"unreadable cell colour {value}")
    return cells


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)

    view_boxes = [
        box
        for box in _components(_mask(array, VIEW_FRAME))
        if (box[2] - box[0] + 1) * (box[3] - box[1] + 1) >= MIN_FRAME_PIXELS
    ]
    if len(view_boxes) != N_VIEWS:
        raise ValueError(f"expected {N_VIEWS} slate-framed views, found {len(view_boxes)}")

    fused: list[int | None] = [None] * N_CELLS
    for box in view_boxes:
        for index, value in enumerate(_read_tile(array, box)):
            if value is None:
                continue
            if fused[index] is not None and fused[index] != value:
                raise ValueError("the two views disagree on a revealed cell")
            fused[index] = value
    if all(value is None for value in fused):
        raise ValueError("the views reveal nothing")

    tiles: list[tuple[str, list[int | None]]] = []
    for name, colour in FRAME_COLOURS.items():
        boxes = [
            box
            for box in _components(_mask(array, colour))
            if (box[2] - box[0] + 1) * (box[3] - box[1] + 1) >= MIN_FRAME_PIXELS
        ]
        if not boxes:
            continue
        if len(boxes) != 1:
            raise ValueError(f"colour {name} frames more than one tile")
        tiles.append((name, _read_tile(array, boxes[0])))
    if len(tiles) != N_TILES:
        raise ValueError(f"expected {N_TILES} candidate tiles, found {len(tiles)}")
    if any(value is None for _, cells in tiles for value in cells):
        raise ValueError("a candidate tile is shuttered")

    survivors = [
        name
        for name, cells in tiles
        if all(f is None or cells[i] == f for i, f in enumerate(fused))
    ]
    if len(survivors) != 1:
        raise ValueError(f"the visible evidence leaves {len(survivors)} candidates")
    return survivors[0]
