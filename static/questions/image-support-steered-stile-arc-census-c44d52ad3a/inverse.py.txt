"""Pixel-only inverse arm for the stile-arc census.

The arm receives the final PNG and nothing else.  It segments the wire ink, the
two black stile rings and the orange beads, rebuilds the wire's connectivity
after the stiles have cut it, identifies the single arc that is joined to *both*
stiles, decides which arc each bead sits on, and reports how many beads live on
the arc between the stiles.  It abstains whenever the structure named by the
question (one wire cut by exactly two stiles into exactly three arcs, every bead
owned by exactly one arc) is not visible in the pixels.

Only numpy, PIL and the standard library are used.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_WIRE = (40, 70, 110)
_BEAD = (226, 118, 30)
_EDGE = (0, 0, 0)
_TOL = 40

_MIN_WIRE_AREA = 300
_MIN_STILE_AREA = 150
_MAX_STILE_AREA = 900
_MIN_BEAD_AREA = 16
_MAX_BEAD_AREA = 260
_TOUCH = 2

ABSTAIN = "abstain"
_MIN_COUNT = 1
_MAX_COUNT = 7


def _mask(rgb: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    diff = np.abs(rgb - np.asarray(color, dtype=np.int16))
    return diff.max(axis=2) <= _TOL


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    """Four-connected labelling of a boolean mask; 0 means background."""

    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    current = 0
    for y, x in zip(*np.nonzero(mask)):
        if labels[y, x]:
            continue
        current += 1
        labels[y, x] = current
        queue = deque([(int(y), int(x))])
        while queue:
            cy, cx = queue.popleft()
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not labels[ny, nx]:
                    labels[ny, nx] = current
                    queue.append((ny, nx))
    return labels, current


def _components(labels: np.ndarray, count: int, lo: int, hi: int | None) -> list[dict]:
    out = []
    for value in range(1, count + 1):
        ys, xs = np.nonzero(labels == value)
        area = int(ys.size)
        if area < lo or (hi is not None and area > hi):
            continue
        out.append(
            {
                "label": value,
                "area": area,
                "box": (int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())),
            }
        )
    return out


def _neighbour_labels(labels: np.ndarray, box: tuple[int, int, int, int], pad: int) -> set[int]:
    height, width = labels.shape
    x0 = max(0, box[0] - pad)
    y0 = max(0, box[1] - pad)
    x1 = min(width - 1, box[2] + pad)
    y1 = min(height - 1, box[3] + pad)
    window = labels[y0 : y1 + 1, x0 : x1 + 1]
    return {int(value) for value in np.unique(window) if value}


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)

    edge_labels, edge_count = _label(_mask(rgb, _EDGE))
    stiles = _components(edge_labels, edge_count, _MIN_STILE_AREA, _MAX_STILE_AREA)
    if len(stiles) != 2:
        return ABSTAIN
    for stile in stiles:
        box = stile["box"]
        if not (12 <= box[2] - box[0] <= 30 and 12 <= box[3] - box[1] <= 30):
            return ABSTAIN

    wire_labels, wire_count = _label(_mask(rgb, _WIRE))
    arcs = _components(wire_labels, wire_count, _MIN_WIRE_AREA, None)
    if len(arcs) != 3:
        return ABSTAIN
    allowed = {arc["label"] for arc in arcs}

    touching = [
        _neighbour_labels(wire_labels, stile["box"], _TOUCH) & allowed for stile in stiles
    ]
    shared = touching[0] & touching[1]
    if len(shared) != 1:
        return ABSTAIN
    middle = next(iter(shared))
    if len(touching[0]) != 2 or len(touching[1]) != 2:
        return ABSTAIN

    bead_labels, bead_count = _label(_mask(rgb, _BEAD))
    beads = _components(bead_labels, bead_count, _MIN_BEAD_AREA, _MAX_BEAD_AREA)
    if not 2 <= len(beads) <= 14:
        return ABSTAIN

    stile_boxes = [stile["box"] for stile in stiles]
    inside = 0
    for bead in beads:
        box = bead["box"]
        if box[2] - box[0] > 16 or box[3] - box[1] > 16:
            return ABSTAIN
        for stile_box in stile_boxes:
            if (
                box[0] - _TOUCH <= stile_box[2]
                and stile_box[0] <= box[2] + _TOUCH
                and box[1] - _TOUCH <= stile_box[3]
                and stile_box[1] <= box[3] + _TOUCH
            ):
                return ABSTAIN  # bead touches a stile: ownership is unreadable
        owners = _neighbour_labels(wire_labels, box, _TOUCH) & allowed
        if len(owners) != 1:
            return ABSTAIN  # bead straddles two arcs or floats free
        if next(iter(owners)) == middle:
            inside += 1

    if not _MIN_COUNT <= inside <= _MAX_COUNT:
        return ABSTAIN
    return str(inside)
