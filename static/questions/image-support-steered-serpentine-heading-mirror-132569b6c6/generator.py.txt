"""Deterministic latent sampler and raster renderer.

The answer is a state replay, not a statistic of the tile multiset: clockwise
turns and left-right mirrors do not commute.  The board is deliberately a
single dense connected visual structure so every distant portion contributes
operations to the same replay.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

CANVAS_SIZE = 640
HEADINGS = ("north", "east", "south", "west")
OPERATIONS = ("turn_cw", "mirror_lr")

BACKGROUND = (249, 250, 247)
INK = (31, 42, 58)
START_PALETTES = ((111, 78, 157), (126, 82, 166), (98, 75, 151))
TURN_PALETTES = ((235, 171, 48), (244, 181, 58), (224, 158, 42))
MIRROR_PALETTES = ((43, 168, 160), (38, 184, 173), (53, 153, 151))
ARROW = (249, 249, 242)


def _apply(heading: int, operation: str) -> int:
    if operation == "turn_cw":
        return (heading + 1) % 4
    if operation == "mirror_lr":
        return (-heading) % 4
    raise ValueError(f"unknown operation: {operation}")


def _replay(initial_heading: str, operations: Iterable[str]) -> str:
    value = HEADINGS.index(initial_heading)
    for operation in operations:
        value = _apply(value, operation)
    return HEADINGS[value]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    operations = [OPERATIONS[rng.randrange(2)] for _ in range(15)]

    # Make final classes exactly cycle-balanced over consecutive scene seeds,
    # while leaving operation type/count/ink independent of the target class.
    target = HEADINGS[int(seed) % 4]
    initial_heading = next(
        heading for heading in HEADINGS if _replay(heading, operations) == target
    )
    return {
        "initial_heading": initial_heading,
        "operations": operations,
        "palette_variant": rng.randrange(3),
        "board_offset_x": rng.choice((-4, 0, 4)),
        "board_offset_y": rng.choice((-4, 0, 4)),
        "cell_size": rng.choice((142, 143, 144)),
        "line_width": rng.choice((7, 8, 9)),
    }


def _snake_cells() -> list[tuple[int, int]]:
    cells: list[tuple[int, int]] = []
    for row in range(4):
        columns = range(4) if row % 2 == 0 else range(3, -1, -1)
        cells.extend((row, column) for column in columns)
    return cells


def _draw_start_arrow(
    draw: ImageDraw.ImageDraw, center: tuple[int, int], heading: str, scale: int
) -> None:
    cx, cy = center
    # Canonical north arrow: a broad triangular head and a narrow tail.
    points = [(-48, 4), (0, -52), (48, 4), (14, 4), (14, 43), (-14, 43), (-14, 4)]

    def rotate(point: tuple[int, int]) -> tuple[int, int]:
        x, y = point
        if heading == "north":
            rx, ry = x, y
        elif heading == "east":
            rx, ry = -y, x
        elif heading == "south":
            rx, ry = -x, -y
        elif heading == "west":
            rx, ry = y, -x
        else:
            raise ValueError(f"invalid initial heading: {heading}")
        factor = scale / 144.0
        return (round(cx + rx * factor), round(cy + ry * factor))

    draw.polygon([rotate(point) for point in points], fill=ARROW)


def _draw_turn_icon(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], width: int) -> None:
    left, top, right, bottom = box
    cx, cy = (left + right) // 2, (top + bottom) // 2
    radius = min(right - left, bottom - top) // 4
    arc_box = (cx - radius, cy - radius, cx + radius, cy + radius)
    draw.arc(arc_box, start=205, end=520, fill=INK, width=max(6, width))
    draw.polygon(
        [(cx + radius + 2, cy - 5), (cx + radius - 16, cy - 20), (cx + radius - 14, cy + 2)],
        fill=INK,
    )


def _draw_mirror_icon(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], width: int) -> None:
    left, top, right, bottom = box
    cx, cy = (left + right) // 2, (top + bottom) // 2
    span = min(right - left, bottom - top) // 3
    draw.line((cx, cy - span, cx, cy + span), fill=INK, width=max(6, width))
    draw.line((cx - span, cy, cx - 10, cy), fill=INK, width=max(5, width - 1))
    draw.line((cx + 10, cy, cx + span, cy), fill=INK, width=max(5, width - 1))
    draw.polygon([(cx - span - 2, cy), (cx - span + 15, cy - 12), (cx - span + 15, cy + 12)], fill=INK)
    draw.polygon([(cx + span + 2, cy), (cx + span - 15, cy - 12), (cx + span - 15, cy + 12)], fill=INK)


def render(scene: dict) -> Image.Image:
    initial_heading = scene["initial_heading"]
    operations = list(scene["operations"])
    palette_variant = int(scene["palette_variant"])
    offset_x = int(scene["board_offset_x"])
    offset_y = int(scene["board_offset_y"])
    cell = int(scene["cell_size"])
    line_width = int(scene["line_width"])

    if initial_heading not in HEADINGS or len(operations) != 15:
        raise ValueError("scene must contain one valid heading and fifteen operations")
    if any(operation not in OPERATIONS for operation in operations):
        raise ValueError("scene contains an invalid operation")
    if palette_variant not in range(3):
        raise ValueError("palette_variant must be in range(3)")

    board_size = 4 * cell
    left = (CANVAS_SIZE - board_size) // 2 + offset_x
    top = (CANVAS_SIZE - board_size) // 2 + offset_y
    right, bottom = left + board_size - 1, top + board_size - 1
    if left < 24 or top < 24 or right >= CANVAS_SIZE - 24 or bottom >= CANVAS_SIZE - 24:
        raise ValueError("board must retain its visible canvas margin")

    image = Image.new("RGB", (CANVAS_SIZE, CANVAS_SIZE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rectangle((left, top, right, bottom), fill=INK)

    snake = _snake_cells()
    op_index = 0
    for sequence_index, (row, column) in enumerate(snake):
        x0, y0 = left + column * cell, top + row * cell
        x1, y1 = x0 + cell - 1, y0 + cell - 1
        inner = (x0 + line_width, y0 + line_width, x1 - line_width, y1 - line_width)
        if sequence_index == 0:
            fill = START_PALETTES[palette_variant]
            draw.rectangle(inner, fill=fill)
            _draw_start_arrow(draw, ((x0 + x1) // 2, (y0 + y1) // 2), initial_heading, cell)
            continue

        operation = operations[op_index]
        op_index += 1
        fill = TURN_PALETTES[palette_variant] if operation == "turn_cw" else MIRROR_PALETTES[palette_variant]
        draw.rectangle(inner, fill=fill)
        icon_box = (
            inner[0] + cell // 6,
            inner[1] + cell // 6,
            inner[2] - cell // 6,
            inner[3] - cell // 6,
        )
        if operation == "turn_cw":
            _draw_turn_icon(draw, icon_box, line_width)
        else:
            _draw_mirror_icon(draw, icon_box, line_width)

    return image


def analytic_gold(scene: dict) -> str:
    return _replay(scene["initial_heading"], scene["operations"])


def margin(scene: dict) -> float:
    # One-sided visibility reserve after grid and icon occupancy.  A positive
    # value means the tile interiors retain a large pure-color region and the
    # START arrow retains a broad head beyond its narrow tail.
    return float(int(scene["cell_size"]) - 2 * int(scene["line_width"]) - 104)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 18.0 or len(scene["operations"]) != 15


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
