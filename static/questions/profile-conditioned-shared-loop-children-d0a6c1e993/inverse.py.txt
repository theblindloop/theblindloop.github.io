"""Pixel-only inverse: segment loops/markers and recover the enclosure tree."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "red": np.array([220, 45, 55]),
    "green": np.array([32, 158, 82]),
    "blue": np.array([38, 100, 220]),
}


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=np.uint8)
    output = []
    for sy, sx in zip(*np.nonzero(mask)):
        if seen[sy, sx]:
            continue
        queue = deque([(int(sy), int(sx))])
        seen[sy, sx] = 1
        xs, ys = [], []
        while queue:
            y, x = queue.popleft()
            xs.append(x)
            ys.append(y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = 1
                    queue.append((ny, nx))
        if len(xs) >= minimum:
            output.append((min(xs), min(ys), max(xs), max(ys), len(xs)))
    return output


def _inside(outer: tuple[int, int, int, int, int], inner: tuple[int, int, int, int, int]) -> bool:
    return outer[0] < inner[0] and outer[1] < inner[1] and outer[2] > inner[2] and outer[3] > inner[3]


def _contains(box: tuple[int, int, int, int, int], point: tuple[float, float]) -> bool:
    return box[0] < point[0] < box[2] and box[1] < point[1] < box[3]


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int32)
    black = np.max(rgb, axis=2) < 70
    loops = _components(black, minimum=180)
    if len(loops) != 15:
        raise ValueError(f"expected 15 closed black loop components, found {len(loops)}")

    child_counts = []
    for i, outer in enumerate(loops):
        contained = [j for j, inner in enumerate(loops) if i != j and _inside(outer, inner)]
        direct = [
            j for j in contained
            if not any(k != j and _inside(outer, loops[k]) and _inside(loops[k], loops[j])
                       for k in contained)
        ]
        child_counts.append(len(direct))

    counts = {}
    for color, target in PALETTE.items():
        distance = np.sqrt(np.sum((rgb - target) ** 2, axis=2))
        blobs = _components(distance < 45, minimum=45)
        if len(blobs) != 2:
            raise ValueError(f"expected two {color} dots, found {len(blobs)}")
        points = [((b[0] + b[2]) / 2.0, (b[1] + b[3]) / 2.0) for b in blobs]
        common = [i for i, loop in enumerate(loops) if _contains(loop, points[0]) and _contains(loop, points[1])]
        if not common:
            raise ValueError(f"{color} dots lack common loop")
        lca = min(common, key=lambda i: (loops[i][2] - loops[i][0]) * (loops[i][3] - loops[i][1]))
        counts[color] = child_counts[lca]

    winners = [color for color, count in counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"non-unique pixel decision: {counts}")
    return winners[0]
