"""Deterministic scenes of paired markers in a visible loop-containment tree."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

COLORS = {
    "red": (220, 45, 55),
    "green": (32, 158, 82),
    "blue": (38, 100, 220),
}


def _rect(x0: int, y0: int, x1: int, y1: int) -> list[int]:
    return [x0, y0, x1, y1]


def _structure(box: list[int], degree: int) -> tuple[list[list[int]], list[tuple[int, int]]]:
    """Return five loops and two marker centers; their LCA is the root."""
    x0, y0, x1, y1 = box
    loops = [box]
    if degree == 1:
        child = _rect(x0 + 24, y0 + 28, x1 - 24, y1 - 28)
        loops.append(child)
        # Three sibling grandchildren: total descendants stays four for every root.
        gy0, gy1 = y0 + 70, y1 - 38
        cuts = [(x0 + 38, x0 + 70), (x0 + 84, x0 + 116), (x0 + 130, x0 + 162)]
        loops.extend(_rect(a, gy0, b, gy1) for a, b in cuts)
        markers = [(x0 + 12, y0 + 90), (x0 + 46, y0 + 50)]
    elif degree == 2:
        left = _rect(x0 + 18, y0 + 28, x0 + 91, y1 - 28)
        right = _rect(x0 + 109, y0 + 28, x1 - 18, y1 - 28)
        loops.extend([left, right])
        loops.extend([
            _rect(x0 + 35, y0 + 65, x0 + 74, y1 - 65),
            _rect(x0 + 126, y0 + 65, x0 + 165, y1 - 65),
        ])
        markers = [(x0 + 54, y0 + 48), (x0 + 145, y0 + 48)]
    elif degree == 3:
        children = [
            _rect(x0 + 16, y0 + 32, x0 + 67, y1 - 32),
            _rect(x0 + 75, y0 + 32, x0 + 126, y1 - 32),
            _rect(x0 + 134, y0 + 32, x1 - 16, y1 - 32),
        ]
        loops.extend(children)
        loops.append(_rect(x0 + 146, y0 + 70, x1 - 28, y1 - 70))
        markers = [(x0 + 41, y0 + 54), (x0 + 100, y0 + 54)]
    else:
        raise ValueError("degree must be 1, 2, or 3")
    return loops, markers


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    degrees = [1, 2, 3]
    rng.shuffle(degrees)
    colors = ["red", "green", "blue"]
    rng.shuffle(colors)
    color_for_degree = dict(zip((1, 2, 3), colors))

    loops: list[list[int]] = []
    markers: list[dict] = []
    for slot, degree in enumerate(degrees):
        # Small whole-tree translations vary geometry while preserving generous gaps.
        dx = rng.randint(-3, 3)
        dy = rng.randint(-4, 4)
        x0 = 14 + slot * 235 + dx
        root = _rect(x0, 17 + dy, x0 + 200, 303 + dy)
        group_loops, centers = _structure(root, degree)
        loops.extend(group_loops)
        color = color_for_degree[degree]
        for cx, cy in centers:
            markers.append({"color": color, "center": [cx, cy], "radius": 8})

    return {"canvas": [720, 330], "loops": loops, "markers": markers}


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    image = Image.new("RGB", (int(width), int(height)), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    for bounds in scene["loops"]:
        draw.rectangle(tuple(bounds), outline=(18, 20, 24), width=5)
    for marker in scene["markers"]:
        cx, cy = marker["center"]
        radius = int(marker["radius"])
        fill = COLORS[marker["color"]]
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=fill,
                     outline=(255, 255, 255), width=2)
    return image


def _contains(bounds: list[int], point: list[int] | tuple[int, int]) -> bool:
    x0, y0, x1, y1 = bounds
    return x0 < point[0] < x1 and y0 < point[1] < y1


def _inside(outer: list[int], inner: list[int]) -> bool:
    return outer[0] < inner[0] and outer[1] < inner[1] and outer[2] > inner[2] and outer[3] > inner[3]


def _analysis(scene: dict) -> dict[str, int]:
    loops = scene["loops"]
    markers = scene["markers"]
    child_counts = []
    for i, outer in enumerate(loops):
        contained = [j for j, inner in enumerate(loops) if i != j and _inside(outer, inner)]
        direct = [
            j for j in contained
            if not any(k != j and _inside(outer, loops[k]) and _inside(loops[k], loops[j])
                       for k in contained)
        ]
        child_counts.append(len(direct))

    result: dict[str, int] = {}
    for color in COLORS:
        points = [m["center"] for m in markers if m["color"] == color]
        if len(points) != 2:
            raise ValueError(f"expected two {color} markers")
        common = [i for i, loop in enumerate(loops) if _contains(loop, points[0]) and _contains(loop, points[1])]
        if not common:
            raise ValueError("marker pair lacks a common loop")
        lca = min(common, key=lambda i: (loops[i][2] - loops[i][0]) * (loops[i][3] - loops[i][1]))
        result[color] = child_counts[lca]
    return result


def analytic_gold(scene: dict) -> str:
    counts = _analysis(scene)
    winners = [color for color, count in counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"non-unique target: {counts}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = _analysis(scene)
    distances = sorted(abs(count - 2) for count in counts.values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) <= 0.0
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["loops"] = list(reversed(twin["loops"]))
    return [("reverse_same_ink_loop_draw_order", twin)]
