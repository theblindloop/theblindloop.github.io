"""Pixel-only inverse arm: measure slate ink in units of its own cut window.

Receives the final PNG and nothing else.  It segments the slate ink, recovers
the one square window enclosed by that ink as the areal unit, and reports how
many whole units of ink there are.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "window_area_unit_tally-oracle-0.1.0"

_SLATE_REF = np.array([58, 66, 84], dtype=np.int16)
_SLATE_TOL = 30
ABSTAIN = "abstain"


def _slate_mask(arr: np.ndarray) -> np.ndarray:
    return (np.abs(arr - _SLATE_REF[None, None, :]) <= _SLATE_TOL).all(axis=2)


def _reachable_from_border(free: np.ndarray) -> np.ndarray:
    """Span flood fill of ``free`` seeded from every border pixel."""
    height, width = free.shape
    seen = np.zeros_like(free)
    stack: list[tuple[int, int]] = []
    for x in range(width):
        if free[0, x]:
            stack.append((0, x))
        if free[height - 1, x]:
            stack.append((height - 1, x))
    for y in range(height):
        if free[y, 0]:
            stack.append((y, 0))
        if free[y, width - 1]:
            stack.append((y, width - 1))
    while stack:
        y, x = stack.pop()
        row = free[y] & ~seen[y]
        if not row[x]:
            continue
        blocked = np.nonzero(~row)[0]
        left = blocked[blocked < x]
        right = blocked[blocked > x]
        x0 = int(left[-1]) + 1 if left.size else 0
        x1 = int(right[0]) - 1 if right.size else width - 1
        seen[y, x0 : x1 + 1] = True
        for ny in (y - 1, y + 1):
            if not 0 <= ny < height:
                continue
            nrow = free[ny, x0 : x1 + 1] & ~seen[ny, x0 : x1 + 1]
            idx = np.nonzero(nrow)[0]
            if idx.size == 0:
                continue
            breaks = np.nonzero(np.diff(idx) > 1)[0]
            starts = np.concatenate(([idx[0]], idx[breaks + 1]))
            for start in starts:
                stack.append((ny, x0 + int(start)))
    return seen


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    slate = _slate_mask(arr)
    ink = int(slate.sum())
    if ink < 4000:
        return ABSTAIN

    enclosed = ~slate & ~_reachable_from_border(~slate)
    if not enclosed.any():
        return ABSTAIN

    ys, xs = np.nonzero(enclosed)
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    unit = int(enclosed.sum())
    height = y1 - y0 + 1
    width = x1 - x0 + 1
    if unit != height * width:
        return ABSTAIN
    if abs(height - width) > 1:
        return ABSTAIN
    if not 20 <= width <= 120:
        return ABSTAIN
    if unit <= 0 or ink < unit:
        return ABSTAIN
    return str(int(ink // unit))
