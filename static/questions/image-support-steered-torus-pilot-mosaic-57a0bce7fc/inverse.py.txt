"""Independent pixel-only inverse for Torus Pilot Mosaic.

This module deliberately duplicates the public visual vocabulary. It imports no
renderer, scene, prompt, generator, verifier, or gold implementation.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

PAPER = np.array((248, 245, 236), dtype=np.int16)
INK = np.array((38, 54, 74), dtype=np.int16)
START = np.array((224, 218, 235), dtype=np.int16)
GOLD = np.array((245, 184, 46), dtype=np.int16)
TURN = (np.array((55, 187, 205), dtype=np.int16), np.array((35, 166, 187), dtype=np.int16))
FORWARD = (np.array((242, 107, 91), dtype=np.int16), np.array((224, 82, 72), dtype=np.int16))
HEADINGS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _near(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    return np.max(np.abs(array.astype(np.int16) - color), axis=2) <= tolerance


def _largest_bbox(mask: np.ndarray) -> tuple[int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if len(xs) < 1000:
        raise ValueError("mosaic perimeter is missing")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max())


def _classify_command(cell: np.ndarray) -> str:
    h, w = cell.shape[:2]
    inner = cell[max(5, h // 20) : h - max(5, h // 20), max(5, w // 20) : w - max(5, w // 20)]
    area = inner.shape[0] * inner.shape[1]
    turn_count = max(int(_near(inner, color).sum()) for color in TURN)
    forward_count = max(int(_near(inner, color).sum()) for color in FORWARD)
    dominant = max(turn_count, forward_count)
    # Full-area validation makes a missing command fragment explicit rather than guessed.
    if dominant < 0.74 * area:
        raise ValueError("command tile is incomplete or ambiguous")
    if min(turn_count, forward_count) > 0.08 * area:
        raise ValueError("command tile mixes operation colors")
    return "turn" if turn_count > forward_count else "forward"


def _start_pose(cell: np.ndarray) -> tuple[int, int, int]:
    h, w = cell.shape[:2]
    gold = _near(cell, GOLD, tolerance=15)
    ys, xs = np.nonzero(gold)
    if not 350 <= len(xs) <= 850:
        raise ValueError("gold start ring is missing or damaged")
    cx, cy = float(xs.mean()), float(ys.mean())

    white = _near(cell, np.array((255, 255, 255), dtype=np.int16), tolerance=5)
    wys, wxs = np.nonzero(white)
    if len(wxs) < 5000:
        raise ValueError("START board is missing or damaged")
    bx0, bx1 = float(np.percentile(wxs, 1)), float(np.percentile(wxs, 99))
    by0, by1 = float(np.percentile(wys, 1)), float(np.percentile(wys, 99))
    if min(bx1 - bx0, by1 - by0) < 85:
        raise ValueError("START board extent is ambiguous")
    col = int(np.clip(np.floor(3.0 * (cx - bx0) / (bx1 - bx0 + 1.0)), 0, 2))
    row = int(np.clip(np.floor(3.0 * (cy - by0) / (by1 - by0 + 1.0)), 0, 2))

    yy, xx = np.indices((h, w))
    local = _near(cell, INK, tolerance=15) & (((xx - cx) ** 2 + (yy - cy) ** 2) <= 12.5**2)
    ay, ax = np.nonzero(local)
    if len(ax) < 55:
        raise ValueError("start heading arrow is missing")
    vectors = np.stack((ay.astype(float) - cy, ax.astype(float) - cx), axis=1)
    scores = [float(np.max(vectors @ np.array(direction, dtype=float))) for direction in HEADINGS]
    ordering = np.argsort(scores)
    if scores[int(ordering[-1])] - scores[int(ordering[-2])] < 2.0:
        raise ValueError("start heading is ambiguous")
    heading = int(ordering[-1])
    return row, col, heading


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[2] != 3 or min(array.shape[:2]) < 300:
        raise ValueError("unsupported raster")
    ink = _near(array, INK, tolerance=15)
    x0, y0, x1, y1 = _largest_bbox(ink)
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (0.85 <= width / height <= 1.15 and width > 0.75 * array.shape[1]):
        raise ValueError("mosaic geometry is incomplete")

    x_edges = [round(x0 + k * (width - 1) / 4) for k in range(5)]
    y_edges = [round(y0 + k * (height - 1) / 4) for k in range(5)]
    cells = []
    for row_index in range(4):
        for col_index in range(4):
            xa, xb = x_edges[col_index], x_edges[col_index + 1] + 1
            ya, yb = y_edges[row_index], y_edges[row_index + 1] + 1
            cells.append(array[ya:yb, xa:xb])
    row, col, heading = _start_pose(cells[0])
    commands = [_classify_command(cell) for cell in cells[1:]]
    for command in commands:
        if command == "turn":
            heading = (heading + 1) % 4
        else:
            dr, dc = HEADINGS[heading]
            row = (row + dr) % 3
            col = (col + dc) % 3
    return "ABCDEFGHI"[row * 3 + col]
