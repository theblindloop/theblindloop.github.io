"""Deterministic latent sampler and renderer for Torus Pilot Mosaic."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

SIZE = 640
OUTER = 24
GRID = 4
CELL = 148
INK = (38, 54, 74)
PAPER = (248, 245, 236)
START_FILL = (224, 218, 235)
GOLD = (245, 184, 46)
WHITE = (255, 255, 255)
TURN_COLORS = ((55, 187, 205), (35, 166, 187))
FORWARD_COLORS = ((242, 107, 91), (224, 82, 72))
HEADINGS = ((-1, 0), (0, 1), (1, 0), (0, -1))


def _replay(scene: dict) -> tuple[int, int, int]:
    row = int(scene["start_row"])
    col = int(scene["start_col"])
    heading = int(scene["start_heading"])
    commands = list(scene["commands"])
    for command in commands:
        if command == "turn":
            heading = (heading + 1) % 4
        elif command == "forward":
            dr, dc = HEADINGS[heading]
            row = (row + dr) % 3
            col = (col + dc) % 3
        else:
            raise ValueError(f"unknown command {command!r}")
    return row, col, heading


def sample_scene(seed: int) -> dict:
    """Sample a balanced deterministic scene; consecutive seeds cycle answers."""
    target = int(seed) % 9
    rng = random.Random(int(seed) ^ 0x5A17D3)
    for _ in range(5000):
        start_row = rng.randrange(3)
        start_col = rng.randrange(3)
        start_heading = rng.randrange(4)
        commands = [rng.choice(("turn", "forward")) for _ in range(14)] + ["forward"]
        forward_count = commands.count("forward")
        if not 5 <= forward_count <= 10:
            continue
        scene = {
            "start_row": start_row,
            "start_col": start_col,
            "start_heading": start_heading,
            "commands": commands,
            "palette_variant": rng.randrange(2),
        }
        row, col, _ = _replay(scene)
        if row * 3 + col == target:
            return scene
    raise RuntimeError("could not sample requested balanced final cell")


def analytic_gold(scene: dict) -> str:
    row, col, _ = _replay(scene)
    return "ABCDEFGHI"[row * 3 + col]


def margin(scene: dict) -> float:
    # The narrowest designed gap is the arrow tip to a START-board grid line.
    _ = (scene["start_row"], scene["start_col"], scene["start_heading"], scene["commands"])
    return 5.0


def is_quarantined(scene: dict) -> bool:
    try:
        commands = list(scene["commands"])
        return not (
            0 <= int(scene["start_row"]) < 3
            and 0 <= int(scene["start_col"]) < 3
            and 0 <= int(scene["start_heading"]) < 4
            and len(commands) == 15
            and all(command in {"turn", "forward"} for command in commands)
        )
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    # Palette codes differing by two intentionally select the same visible palette.
    twin["palette_variant"] = int(scene["palette_variant"]) + 2
    return [("redundant_palette_code_plus_two", twin)]


def _arrow_polygon(cx: float, cy: float, heading: int, radius: float = 12.0):
    dr, dc = HEADINGS[heading]
    px, py = -dr, dc
    tip = (cx + dc * radius, cy + dr * radius)
    back_x = cx - dc * radius * 0.48
    back_y = cy - dr * radius * 0.48
    return [
        tip,
        (back_x + px * radius * 0.48, back_y + py * radius * 0.48),
        (back_x - px * radius * 0.48, back_y - py * radius * 0.48),
    ]


def _draw_start(scene: dict, draw: ImageDraw.ImageDraw, x0: int, y0: int) -> None:
    pad = 20
    bx0, by0 = x0 + pad, y0 + pad
    board = 108
    step = board // 3
    draw.rectangle((bx0, by0, bx0 + board, by0 + board), fill=WHITE, outline=INK, width=4)
    for k in (1, 2):
        draw.line((bx0 + k * step, by0, bx0 + k * step, by0 + board), fill=INK, width=3)
        draw.line((bx0, by0 + k * step, bx0 + board, by0 + k * step), fill=INK, width=3)
    font = ImageFont.load_default()
    for row in range(3):
        for col in range(3):
            draw.text((bx0 + col * step + 3, by0 + row * step + 2), "ABCDEFGHI"[row * 3 + col], fill=INK, font=font)
    cx = bx0 + int(scene["start_col"]) * step + step / 2
    cy = by0 + int(scene["start_row"]) * step + step / 2
    # A clean gold disk keeps the dark heading glyph separable from board lines.
    draw.ellipse((cx - 14, cy - 14, cx + 14, cy + 14), fill=GOLD)
    draw.polygon(_arrow_polygon(cx, cy, int(scene["start_heading"])), fill=INK)


def _draw_turn(draw: ImageDraw.ImageDraw, x0: int, y0: int) -> None:
    cx, cy = x0 + CELL / 2, y0 + CELL / 2
    box = (cx - 39, cy - 39, cx + 39, cy + 39)
    draw.arc(box, 205, 520, fill=WHITE, width=15)
    angle = math.radians(520)
    tx = cx + 39 * math.cos(angle)
    ty = cy + 39 * math.sin(angle)
    draw.polygon([(tx + 12, ty - 3), (tx - 7, ty - 14), (tx - 5, ty + 8)], fill=WHITE)


def _draw_forward(draw: ImageDraw.ImageDraw, x0: int, y0: int) -> None:
    cx, cy = x0 + CELL / 2, y0 + CELL / 2
    draw.polygon(
        [
            (cx, cy - 48),
            (cx + 43, cy),
            (cx + 17, cy),
            (cx + 17, cy + 46),
            (cx - 17, cy + 46),
            (cx - 17, cy),
            (cx - 43, cy),
        ],
        fill=WHITE,
    )


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), PAPER)
    draw = ImageDraw.Draw(image)
    palette_variant = int(scene["palette_variant"]) % 2
    turn_color = TURN_COLORS[palette_variant]
    forward_color = FORWARD_COLORS[palette_variant]
    commands = list(scene["commands"])
    if len(commands) != 15:
        raise ValueError("exactly fifteen commands are required")
    for index in range(16):
        row, col = divmod(index, 4)
        x0, y0 = OUTER + col * CELL, OUTER + row * CELL
        x1, y1 = x0 + CELL, y0 + CELL
        if index == 0:
            fill = START_FILL
        else:
            fill = turn_color if commands[index - 1] == "turn" else forward_color
        draw.rectangle((x0, y0, x1, y1), fill=fill, outline=INK, width=7)
        if index == 0:
            _draw_start(scene, draw, x0, y0)
        elif commands[index - 1] == "turn":
            _draw_turn(draw, x0, y0)
        else:
            _draw_forward(draw, x0, y0)
    # One heavy perimeter makes the command field a single bounded visual object.
    draw.rectangle((OUTER, OUTER, OUTER + 4 * CELL, OUTER + 4 * CELL), outline=INK, width=9)
    return image
