"""Independent image-only inverse arm for quadrant coverage.

This module intentionally imports no renderer, prompt, generator, verifier, or
gold code. It discovers the four colored crosses and sixteen colored dot
components directly from the final RGB raster, assigns dots to their nearest
cross, and evaluates all four quadrant-occupancy vectors.
"""

from __future__ import annotations

from collections import Counter, deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float]]:
    remaining = set(zip(*np.nonzero(mask), strict=True))
    found: list[tuple[int, float, float]] = []
    while remaining:
        start = remaining.pop()
        todo = deque([start])
        pixels = [start]
        while todo:
            y, x = todo.popleft()
            for p in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if p in remaining:
                    remaining.remove(p)
                    todo.append(p)
                    pixels.append(p)
        arr = np.asarray(pixels, dtype=np.float64)
        found.append((len(pixels), float(arr[:, 1].mean()), float(arr[:, 0].mean())))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    spread = arr.max(axis=2) - arr.min(axis=2)
    # Both allowed visual roles are saturated, while text, borders, paper, and
    # panels are neutral. Components are then classified by measured area.
    colorful = (spread > 42) & (arr.max(axis=2) < 245)
    comps = _components(colorful)
    crosses = [(a, x, y) for a, x, y in comps if 1700 <= a <= 3600]
    dots = [(a, x, y) for a, x, y in comps if 250 <= a <= 430]
    if len(crosses) != 4 or len(dots) != 16:
        raise ValueError(f"abstain: expected 4 crosses and 16 dots, found {len(crosses)} and {len(dots)}")
    # Recover reading order from the component centroids rather than fixed image
    # coordinates. The two rows are separated far more than any nuisance shift.
    crosses.sort(key=lambda c: c[2])
    top = sorted(crosses[:2], key=lambda c: c[1])
    bottom = sorted(crosses[2:], key=lambda c: c[1])
    ordered = top + bottom
    assigned: list[list[int]] = [[] for _ in range(4)]
    for _, x, y in dots:
        distances = [(x - cx) ** 2 + (y - cy) ** 2 for _, cx, cy in ordered]
        panel = int(np.argmin(distances))
        _, cx, cy = ordered[panel]
        if abs(x - cx) < 18 or abs(y - cy) < 18:
            raise ValueError("abstain: dot too close to a cross arm")
        q = (2 if y > cy else 0) + (1 if x > cx else 0)
        assigned[panel].append(q)
    if any(len(values) != 4 for values in assigned):
        raise ValueError("abstain: panel dot count is not four")
    scores = []
    for values in assigned:
        counts = Counter(values)
        scores.append(sum(counts.get(q, 0) == 1 for q in range(4)))
    best = max(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    runner = sorted(scores, reverse=True)[1]
    if best != 4 or runner > 2 or len(winners) != 1:
        raise ValueError(f"abstain: no unique complete coverage, scores={scores}")
    return "ABCD"[winners[0]]
