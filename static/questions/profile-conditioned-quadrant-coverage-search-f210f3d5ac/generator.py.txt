"""Deterministic latent sampler and raster renderer for quadrant coverage."""

from __future__ import annotations

import random
from collections import Counter

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 720, 660
BACKGROUND = (250, 249, 246)
PANEL_FILL = (255, 255, 255)
PANEL_BORDER = (174, 178, 180)
INK = (37, 41, 43)
PANEL_CENTERS = ((190, 270), (530, 270), (190, 520), (530, 520))
DOT_PALETTES = ((177, 62, 173), (220, 75, 78), (91, 76, 196), (202, 92, 38))
AXIS_PALETTES = ((25, 137, 145), (29, 128, 172), (28, 145, 102), (37, 121, 151))
LABELS = "ABCD"


def _font(size: int, bold: bool = False):
    name = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    target = seed % 4
    patterns: list[list[int]] = []
    # Each distractor has four dots but occupancy 2,1,1,0. The omitted and
    # repeated regions vary independently, preventing a fixed empty-region cue.
    for panel in range(4):
        if panel == target:
            pattern = [0, 1, 2, 3]
        else:
            missing = rng.randrange(4)
            repeated = (missing + rng.choice((1, 2, 3))) % 4
            pattern = [q for q in range(4) if q != missing] + [repeated]
        rng.shuffle(pattern)
        patterns.append(pattern)
    return {
        "patterns": patterns,
        "dot_color": list(rng.choice(DOT_PALETTES)),
        "axis_color": list(rng.choice(AXIS_PALETTES)),
        "layout_dx": rng.randint(-5, 5),
        "layout_dy": rng.randint(-4, 4),
    }


def _dot_centers(pattern: list[int], cx: int, cy: int) -> list[tuple[int, int]]:
    seen: Counter[int] = Counter()
    out = []
    for q in pattern:
        occurrence = seen[q]
        seen[q] += 1
        sx = -1 if q in (0, 2) else 1
        sy = -1 if q in (0, 1) else 1
        # A repeated dot moves farther toward the panel corner; both positions
        # remain clearly inside the same visible quadrant.
        xoff = 52 + 34 * occurrence
        yoff = 38 + 25 * occurrence
        out.append((cx + sx * xoff, cy + sy * yoff))
    return out


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font, body_font, label_font = _font(26, True), _font(16), _font(22, True)
    draw.text((32, 20), "FOUR-QUADRANT COVERAGE", fill=INK, font=title_font)
    draw.text((32, 60), "Find the panel with exactly one dot in each region made by its cross.", fill=INK, font=body_font)
    draw.text((32, 84), "Dots on a line would not count. Panels A-D are in reading order.", fill=INK, font=body_font)
    dx, dy = int(scene.get("layout_dx", 0)), int(scene.get("layout_dy", 0))
    axis_color = tuple(scene.get("axis_color", AXIS_PALETTES[0]))
    dot_color = tuple(scene.get("dot_color", DOT_PALETTES[0]))
    for i, ((base_x, base_y), pattern) in enumerate(zip(PANEL_CENTERS, scene["patterns"], strict=True)):
        cx, cy = base_x + dx, base_y + dy
        draw.rounded_rectangle((cx - 145, cy - 92, cx + 145, cy + 92), radius=8, fill=PANEL_FILL, outline=PANEL_BORDER, width=3)
        draw.line((cx - 125, cy, cx + 125, cy), fill=axis_color, width=7)
        draw.line((cx, cy - 73, cx, cy + 73), fill=axis_color, width=7)
        for px, py in _dot_centers(list(pattern), cx, cy):
            draw.ellipse((px - 10, py - 10, px + 10, py + 10), fill=dot_color)
        draw.rounded_rectangle((cx - 139, cy - 86, cx - 105, cy - 54), radius=5, fill=BACKGROUND)
        draw.text((cx - 132, cy - 84), LABELS[i], fill=INK, font=label_font)
    return image


def _scores(scene: dict) -> list[int]:
    scores = []
    for pattern in scene["patterns"]:
        counts = Counter(pattern)
        scores.append(sum(counts.get(q, 0) == 1 for q in range(4)))
    return scores


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    best = max(scores)
    winners = [i for i, score in enumerate(scores) if score == best]
    if best != 4 or len(winners) != 1:
        raise ValueError("scene does not have a unique complete-coverage panel")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    if len(scene.get("patterns", [])) != 4:
        return True
    return any(len(p) != 4 or any(q not in (0, 1, 2, 3) for q in p) for p in scene["patterns"])


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
