"""Ordered onset of a running level accumulator (state tracking).

The image shows four ordered panels, P1..P4, drawn left to right.  Each panel
shows a horizontal track with a marked half-way line, and a left-aligned LEVEL
BAR whose length is the running value of a tank/accumulator.  The tank starts
at a base level (given in the question) and, for each successive panel, one
declared signed step is applied, so panel Pk shows the value after steps 1..k.
The level bar can grow (step positive) or shrink (step negative) between panels.

The task: report, from the ordered run, the panel in which the running level
FIRST rises strictly above the half-way line -- P1, P2, P3, or P4 -- or
"never" if it never does.  This is a state-tracking decision: the onset panel
depends on the whole ordered accumulation, not on any single frame.

Latent scene: ``start`` is the unsigned base level in display units and
``steps`` is a list of N signed unit steps applied in order.  The running
levels are the clamped cumulative sums.  analytic_gold reads exactly these two
fields.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N = 4
HALF = 10
MAXU = 20
UNIT_PX = 9

WIDTH = 960
HEIGHT = 210
PANEL_W = 210
PANEL_H = 120
PANEL_TOP = 46
MARGIN_LEFT = 20
PANEL_GAP = 24

BG = (255, 255, 255)
BORDER = (50, 50, 50)
TRACK = (130, 130, 130)
BAR = (240, 140, 40)
TICK = (90, 90, 90)
LABEL = (40, 40, 40)


def _track_origin(panel_left: int) -> tuple[int, int]:
    return panel_left + 16, PANEL_TOP + 44


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    start = rng.randint(2, 8)
    steps = []
    for _ in range(N):
        step = 0
        while step == 0:
            step = rng.randint(-6, 6)
        steps.append(step)
    return {"start": start, "steps": steps}


def _clamp(value: int) -> int:
    return max(1, min(MAXU, value))


def _levels(scene: dict) -> list[int]:
    acc = scene["start"]
    levels = []
    for step in scene["steps"]:
        acc = _clamp(acc + step)
        levels.append(acc)
    return levels


def _draw_panel(draw: ImageDraw.ImageDraw, left: int, level: int, label: str) -> None:
    draw.rectangle(
        [left, PANEL_TOP, left + PANEL_W, PANEL_TOP + PANEL_H],
        outline=BORDER,
        width=3,
    )
    ox, oy = _track_origin(left)
    track_right = ox + MAXU * UNIT_PX
    draw.rectangle([ox, oy, track_right, oy + 40], outline=TRACK, width=3)

    bar_w = level * UNIT_PX
    draw.rectangle([ox, oy, ox + bar_w - 1, oy + 40 - 1], fill=BAR)

    tick_x = ox + HALF * UNIT_PX
    for dy in range(-10, 50, 4):
        if 0 <= dy <= 40:
            continue
        draw.line(
            [tick_x, oy + dy, tick_x, oy + dy + 1],
            fill=TICK,
            width=2,
        )
    draw.line([tick_x, oy + 10, tick_x, oy + 30], fill=TICK, width=2)

    draw.text((left + 8, PANEL_TOP + 6), label, fill=LABEL)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    levels = _levels(scene)
    for i, level in enumerate(levels):
        left = MARGIN_LEFT + i * (PANEL_W + PANEL_GAP)
        _draw_panel(draw, left, level, f"P{i + 1}")
    return img


def analytic_gold(scene: dict) -> str:
    levels = _levels(scene)
    for i, level in enumerate(levels):
        if level > HALF:
            return f"P{i + 1}"
    return "never"


def margin(scene: dict) -> float:
    levels = _levels(scene)
    best = min(abs(level - HALF) for level in levels)
    return float(best * UNIT_PX)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 15.0


def latent_symmetries(scene: dict) -> list:
    """No render-invariant reparametrization exists: the drawn level bars are
    the clamped cumulative sums of start plus the ordered steps, so any scalar
    change to start or any reordering/sign change of the steps shifts the bars
    and, if it moves the onset across the half-way line, the gold.  There is no
    cell or label whose permutation keeps both pixels and gold identical."""
    return []
