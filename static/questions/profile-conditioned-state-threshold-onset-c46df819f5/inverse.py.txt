"""Scene-free pixel inverse arm for the ordered level-accumulator onset world.

Receives only the final PNG.  From the raster it:
  1. finds the four panel frames by their dark border colour and orders them by
     x position (left to right = P1..P4);
  2. within each panel locates the orange level bar (its bounding box) and the
     marked half-way tick (the separate grey vertical indicator) by colour
     segmentation;
  3. decides whether that panel's bar rises strictly to the right of the tick
     (the running level strictly above the half-way line);
  4. reports the FIRST panel for which that holds, or "never".

Only the image and the public question rules are used; no scene, renderer,
prompt, generator, or gold code is accessed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BORDER = (50, 50, 50)
BAR = (240, 140, 40)
TICK = (90, 90, 90)


class UnreadableImageError(ValueError):
    pass


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    visited = np.zeros_like(mask, dtype=bool)
    ys, xs = np.nonzero(mask)
    comps = []
    for y, x in zip(ys.tolist(), xs.tolist()):
        if visited[y, x]:
            continue
        points = []
        queue = deque([(y, x)])
        visited[y, x] = True
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < mask.shape[0]
                        and 0 <= nx < mask.shape[1]
                        and mask[ny, nx]
                        and not visited[ny, nx]
                    ):
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        comps.append(points)
    return comps


def _bbox(points) -> tuple[int, int, int, int]:
    ys = [p[0] for p in points]
    xs = [p[1] for p in points]
    return min(xs), min(ys), max(xs), max(ys)


def _mask_equal(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    r, g, b = color
    return (arr[:, :, 0] == r) & (arr[:, :, 1] == g) & (arr[:, :, 2] == b)


def _panel_bar_and_tick(arr: np.ndarray, box: tuple[int, int, int, int]):
    x0, y0, x1, y1 = box
    x0 += 3
    y0 += 3
    x1 -= 3
    y1 -= 3
    slice_ = arr[y0:y1, x0:x1]

    bar_mask = _mask_equal(slice_, BAR)
    tick_mask = _mask_equal(slice_, TICK)

    bar = None
    if bar_mask.any():
        ys, xs = np.nonzero(bar_mask)
        bar = {
            "left": int(xs.min()) + x0,
            "right": int(xs.max()) + x0,
        }
    if not tick_mask.any():
        raise UnreadableImageError("panel missing the half-way tick")
    ys, xs = np.nonzero(tick_mask)
    tick_x = int(int(np.median(xs)) + x0)

    if bar is None:
        return {"crossed": False}
    return {"crossed": bar["right"] > tick_x}


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    boxes = []
    for comp in _components(_mask_equal(arr, BORDER)):
        if len(comp) < 50:
            continue
        boxes.append(_bbox(comp))
    boxes.sort(key=lambda box: box[0])
    if len(boxes) < 4:
        raise UnreadableImageError(f"found {len(boxes)} panels; expected 4")

    for index, box in enumerate(boxes):
        state = _panel_bar_and_tick(arr, box)
        if state["crossed"]:
            return f"P{index + 1}"
    return "never"
