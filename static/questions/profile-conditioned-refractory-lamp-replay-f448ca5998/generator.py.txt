"""Latent sampler, analytic replay, and Pillow renderer."""

from __future__ import annotations

import random
from copy import deepcopy
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 360
WHITE = (250, 250, 247)
DARK = (35, 42, 52)
GREEN = (34, 177, 104)
PURPLE = (145, 83, 205)
BLUE = (54, 126, 211)
ORANGE = (235, 139, 52)
PALE = (226, 230, 235)


def _replay_values(initial_on: bool, initial_cooldown: int, commands: list[str]) -> str:
    on = bool(initial_on)
    cooldown = int(initial_cooldown)
    for command in commands:
        if command == "WAIT":
            cooldown = max(0, cooldown - 1)
        elif command == "PRESS" and cooldown == 0:
            on = not on
            cooldown = 2
    return "ON" if on else "OFF"


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; seed parity fixes only the requested final class."""
    rng = random.Random(int(seed))
    desired = "ON" if int(seed) % 2 == 0 else "OFF"
    for _ in range(1000):
        initial_on = bool(rng.randrange(2))
        initial_cooldown = rng.randrange(3)
        commands = ["PRESS"] * 3 + ["WAIT"] * 4
        rng.shuffle(commands)
        if _replay_values(initial_on, initial_cooldown, commands) == desired:
            return {
                "initial_on": initial_on,
                "initial_cooldown": initial_cooldown,
                "commands": commands,
            }
    raise RuntimeError("could not sample requested balanced class")


def analytic_gold(scene: dict) -> str:
    return _replay_values(
        bool(scene["initial_on"]),
        int(scene["initial_cooldown"]),
        list(scene["commands"]),
    )


def margin(scene: dict) -> float:
    # Broad fixed gaps and saturated palette distances make every accepted scene
    # far from the pixel decoder's component thresholds.
    _ = (scene["initial_on"], scene["initial_cooldown"], scene["commands"])
    return 24.0


def is_quarantined(scene: dict) -> bool:
    return not (
        isinstance(scene["initial_on"], bool)
        and scene["initial_cooldown"] in (0, 1, 2)
        and len(scene["commands"]) == 7
        and all(item in ("WAIT", "PRESS") for item in scene["commands"])
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = deepcopy(scene)
    return []


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _center_text(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], text: str, size: int, fill=DARK) -> None:
    font = _font(size)
    bounds = draw.textbbox((0, 0), text, font=font)
    w, h = bounds[2] - bounds[0], bounds[3] - bounds[1]
    x = (box[0] + box[2] - w) // 2
    y = (box[1] + box[3] - h) // 2 - bounds[1]
    draw.text((x, y), text, font=font, fill=fill)


def render(scene: dict) -> Image.Image:
    initial_on = bool(scene["initial_on"])
    initial_cooldown = int(scene["initial_cooldown"])
    commands = list(scene["commands"])
    image = Image.new("RGB", (WIDTH, HEIGHT), WHITE)
    draw = ImageDraw.Draw(image)

    draw.rounded_rectangle((24, 35, 245, 322), radius=18, fill=(242, 244, 246), outline=DARK, width=3)
    _center_text(draw, (24, 45, 245, 82), "START", 22)
    draw.ellipse((72, 103, 142, 173), fill=GREEN if initial_on else WHITE, outline=DARK, width=5)
    _center_text(draw, (48, 175, 166, 210), "LAMP", 17)
    _center_text(draw, (35, 222, 112, 250), "COOL", 15)
    for index in range(2):
        left = 120 + index * 51
        fill = PURPLE if index < initial_cooldown else PALE
        draw.rounded_rectangle((left, 218, left + 42, 264), radius=6, fill=fill, outline=DARK, width=3)
    _center_text(draw, (38, 272, 225, 303), "WAIT drains 1", 14)

    draw.line((260, 178, 291, 178), fill=DARK, width=4)
    draw.polygon([(291, 178), (278, 170), (278, 186)], fill=DARK)
    for index, command in enumerate(commands):
        left = 300 + index * 98
        box = (left, 104, left + 82, 252)
        color = BLUE if command == "WAIT" else ORANGE
        draw.rounded_rectangle(box, radius=12, fill=color, outline=DARK, width=3)
        _center_text(draw, (left, 112, left + 82, 150), str(index + 1), 16, WHITE)
        if command == "WAIT":
            draw.line((left + 41, 166, left + 41, 211), fill=WHITE, width=8)
            draw.polygon([(left + 41, 226), (left + 25, 204), (left + 57, 204)], fill=WHITE)
            _center_text(draw, (left + 5, 225, left + 77, 247), "WAIT", 13, WHITE)
        else:
            draw.ellipse((left + 24, 164, left + 58, 198), fill=WHITE, outline=DARK, width=3)
            draw.line((left + 41, 198, left + 41, 221), fill=WHITE, width=6)
            _center_text(draw, (left + 4, 225, left + 78, 247), "PRESS", 11, WHITE)
    _center_text(draw, (285, 55, 990, 92), "READ LEFT TO RIGHT", 22)
    _center_text(draw, (280, 276, 990, 325), "PRESS works only at cooldown 0; then cooldown = 2", 18)
    return image
