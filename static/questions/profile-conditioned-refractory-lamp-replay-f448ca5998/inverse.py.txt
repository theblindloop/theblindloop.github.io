"""Independent inverse arm using only raster color components."""

from __future__ import annotations

import numpy as np
from PIL import Image

GREEN = np.array((34, 177, 104), dtype=np.int16)
PURPLE = np.array((145, 83, 205), dtype=np.int16)
BLUE = np.array((54, 126, 211), dtype=np.int16)
ORANGE = np.array((235, 139, 52), dtype=np.int16)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 12) -> np.ndarray:
    delta = np.abs(array.astype(np.int16) - color)
    return np.max(delta, axis=2) <= tolerance


def _column_components(mask: np.ndarray, minimum_pixels: int) -> list[tuple[float, int]]:
    counts = mask.sum(axis=0)
    active = counts > 3
    runs: list[tuple[float, int]] = []
    start = None
    for x, present in enumerate(active.tolist() + [False]):
        if present and start is None:
            start = x
        elif not present and start is not None:
            stop = x
            pixels = int(mask[:, start:stop].sum())
            if pixels >= minimum_pixels:
                runs.append(((start + stop - 1) / 2.0, pixels))
            start = None
    return runs


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 100 or array.shape[1] < 500:
        raise ValueError("image is too small to decode")

    green_pixels = int(_mask(array, GREEN).sum())
    if 100 < green_pixels < 6000:
        initial_on = True
    elif green_pixels <= 100:
        initial_on = False
    else:
        raise ValueError("ambiguous initial lamp component")

    purple = _column_components(_mask(array, PURPLE), 500)
    if len(purple) not in (0, 1, 2):
        raise ValueError("ambiguous cooldown cells")
    cooldown = len(purple)

    cards: list[tuple[float, str]] = []
    cards.extend((x, "WAIT") for x, _ in _column_components(_mask(array, BLUE), 3000))
    cards.extend((x, "PRESS") for x, _ in _column_components(_mask(array, ORANGE), 3000))
    cards.sort(key=lambda item: item[0])
    if len(cards) != 7 or any(cards[i + 1][0] - cards[i][0] < 40 for i in range(6)):
        raise ValueError("command cards are missing, merged, or not uniquely ordered")

    on = initial_on
    for _, command in cards:
        if command == "WAIT":
            cooldown = max(0, cooldown - 1)
        elif cooldown == 0:
            on = not on
            cooldown = 2
    return "ON" if on else "OFF"
