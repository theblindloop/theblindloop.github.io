"""Independent final-PNG inverse for local sampling-well density.

The inverse receives only raster pixels. It discovers role colors by channel
relationships, extracts connected components, validates two rectangular frame
outlines and every disk-like particle, measures each visible inner frame area,
counts particles inside it, and compares the resulting count-per-area rates.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


def _role_masks(image: Image.Image) -> tuple[np.ndarray, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red = array[:, :, 0]
    green = array[:, :, 1]
    blue = array[:, :, 2]
    frame = (green - red >= 55) & (blue - red >= 75) & (np.maximum(green, blue) >= 100)
    particle = (red - green >= 90) & (green - blue >= 20) & (red >= 180)
    if np.any(frame & particle):
        raise ValueError("frame and particle role masks overlap")
    return frame, particle


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    components: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        stack = [(y, x)]
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return components


def _measure_frame(component: np.ndarray) -> tuple[int, int, int, int, float, float]:
    ys = component[:, 0]
    xs = component[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    width = x1 - x0 + 1
    height = y1 - y0 + 1
    if width % 2 != 1 or height % 2 != 1 or not (25 <= width <= 37 and 25 <= height <= 37):
        raise ValueError("frame extent is outside the calibrated odd-size range")
    expected_area = width * height - (width - 4) * (height - 4)
    if len(component) != expected_area:
        raise ValueError("frame is not an intact two-pixel rectangular outline")
    center_x = float(xs.mean())
    center_y = float(ys.mean())
    if abs(center_x - (x0 + x1) / 2.0) > 1e-9 or abs(center_y - (y0 + y1) / 2.0) > 1e-9:
        raise ValueError("frame outline is not centroid-symmetric")
    return x0, y0, x1, y1, center_x, center_y


def _measure_particle(component: np.ndarray) -> tuple[float, float]:
    ys = component[:, 0]
    xs = component[:, 1]
    y0, y1 = int(ys.min()), int(ys.max())
    x0, x1 = int(xs.min()), int(xs.max())
    if x1 - x0 + 1 != 5 or y1 - y0 + 1 != 5 or not 19 <= len(component) <= 23:
        raise ValueError("particle is not a calibrated five-pixel disk")
    center_x = float(xs.mean())
    center_y = float(ys.mean())
    if abs(center_x - (x0 + x1) / 2.0) > 1e-9 or abs(center_y - (y0 + y1) / 2.0) > 1e-9:
        raise ValueError("particle is not centroid-symmetric")
    return center_x, center_y


def decision_from_image(image: Image.Image) -> str:
    frame_mask, particle_mask = _role_masks(image)
    frame_components = _components(frame_mask)
    if len(frame_components) != 2:
        raise ValueError("exactly two disconnected sampling-well frames are required")
    frames = sorted((_measure_frame(component) for component in frame_components), key=lambda f: f[4])
    if frames[1][4] - frames[0][4] < 80:
        raise ValueError("left and right well frames are not visibly separated")

    particle_components = _components(particle_mask)
    if not 4 <= len(particle_components) <= 12:
        raise ValueError("unexpected total number of particle components")
    counts = [0, 0]
    for component in particle_components:
        center_x, center_y = _measure_particle(component)
        owners = []
        for index, (x0, y0, x1, y1, _, _) in enumerate(frames):
            if x0 + 2 < center_x < x1 - 2 and y0 + 2 < center_y < y1 - 2:
                owners.append(index)
        if len(owners) != 1:
            raise ValueError("each particle must belong to exactly one visible frame interior")
        counts[owners[0]] += 1
    if min(counts) < 2 or max(counts) > 6:
        raise ValueError("per-well particle count is outside the calibrated range")

    areas = [(frame[2] - frame[0] - 3) * (frame[3] - frame[1] - 3) for frame in frames]
    if min(areas) <= 0:
        raise ValueError("visible frame interior has nonpositive area")
    left_density = counts[0] / areas[0]
    right_density = counts[1] / areas[1]
    if abs(left_density - right_density) * 1000.0 <= 0.25:
        raise ValueError("density ordering is at or inside the finite-resolution quarantine band")
    return "left" if left_density > right_density else "right"
