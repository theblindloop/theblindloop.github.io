from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _component_masks(mask: np.ndarray) -> list[np.ndarray]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result: list[np.ndarray] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        component = np.zeros_like(mask, dtype=bool)
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        while queue:
            y, x = queue.popleft()
            component[y, x] = True
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        result.append(component)
    return sorted(result, key=lambda m: int(m.sum()), reverse=True)


def _edge_neighbors(mask: np.ndarray) -> np.ndarray:
    expanded = np.zeros_like(mask)
    expanded[1:] |= mask[:-1]
    expanded[:-1] |= mask[1:]
    expanded[:, 1:] |= mask[:, :-1]
    expanded[:, :-1] |= mask[:, 1:]
    return expanded & ~mask


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    yellow = (arr[:, :, 0] > 190) & (arr[:, :, 1] > 120) & (arr[:, :, 2] < 110)
    black = (arr.max(axis=2) < 95) & (arr.mean(axis=2) < 72)
    yellow_components = _component_masks(yellow)
    black_components = _component_masks(black)
    if len(yellow_components) != 1:
        raise OracleAbstention("yellow hub is missing or fragmented")
    hub_area = int(yellow_components[0].sum())
    if not (1295 <= hub_area <= 1325 or 5230 <= hub_area <= 5265):
        raise OracleAbstention("yellow hub area is outside the supported range")
    if len(black_components) != 3:
        raise OracleAbstention("expected exactly three separate black islands")
    black_areas = sorted(int(component.sum()) for component in black_components)
    if not (all(545 <= area <= 575 for area in black_areas) or all(2220 <= area <= 2265 for area in black_areas)):
        raise OracleAbstention("black islands are damaged or outside the supported range")
    contact_band = _edge_neighbors(yellow_components[0])
    degree = sum(bool((contact_band & component).any()) for component in black_components)
    if degree not in (2, 3):
        raise OracleAbstention("hub has an unsupported contact degree")
    return "yes" if degree == 3 else "no"
