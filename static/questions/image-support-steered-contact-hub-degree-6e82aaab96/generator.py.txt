from __future__ import annotations

import random
from PIL import Image, ImageDraw

CANVAS = 384
BACKGROUND = (250, 249, 246)
ISLAND = (29, 35, 43)
HUB = (244, 181, 45)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    scale = rng.choice([1, 1, 1, 2])
    span = 88 * scale
    return {
        "contact_class": "three" if int(seed) % 2 == 0 else "two",
        "origin_x": rng.randint(22, CANVAS - span - 22),
        "origin_y": rng.randint(22, CANVAS - span - 22),
        "rotation": rng.choice([0, 90, 180, 270]),
        "scale": scale,
        "decoy_side": rng.choice(["left", "right"]),
        "canvas_width": CANVAS,
        "canvas_height": CANVAS,
    }


def _badge(scene: dict) -> Image.Image:
    s = int(scene["scale"])
    badge = Image.new("RGBA", (88 * s, 88 * s), (0, 0, 0, 0))
    draw = ImageDraw.Draw(badge)

    def rect(box: tuple[int, int, int, int], color: tuple[int, int, int]) -> None:
        x0, y0, x1, y1 = (v * s for v in box)
        draw.rectangle((x0, y0, x1 - 1, y1 - 1), fill=(*color, 255))

    # Three equal-area islands. Eight-pixel clearances keep them separate.
    rect((0, 30, 20, 58), ISLAND)
    rect((68, 30, 88, 58), ISLAND)
    rect((30, 0, 58, 20), ISLAND)

    # One connected, equal-area hub. Both classes touch the side islands.
    rect((28, 28, 60, 60), HUB)
    rect((20, 38, 28, 50), HUB)
    rect((60, 38, 68, 50), HUB)
    shift = -6 if scene["decoy_side"] == "left" else 6
    if scene["contact_class"] == "three":
        # This arm reaches the top island by a full 12-pixel-wide edge.
        rect((38 + shift, 20, 50 + shift, 28), HUB)
    elif scene["contact_class"] == "two":
        # Same yellow area, placed away from every island; the top gap is 8 px.
        rect((38 + shift, 60, 50 + shift, 68), HUB)
    else:
        raise ValueError("contact_class must be two or three")

    rotation = int(scene["rotation"])
    if rotation == 90:
        badge = badge.transpose(Image.Transpose.ROTATE_90)
    elif rotation == 180:
        badge = badge.transpose(Image.Transpose.ROTATE_180)
    elif rotation == 270:
        badge = badge.transpose(Image.Transpose.ROTATE_270)
    elif rotation != 0:
        raise ValueError("rotation must be a right angle")
    return badge


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["canvas_width"]), int(scene["canvas_height"])), BACKGROUND)
    badge = _badge(scene)
    image.paste(badge, (int(scene["origin_x"]), int(scene["origin_y"])), badge)
    return image


def analytic_gold(scene: dict) -> str:
    return "yes" if scene["contact_class"] == "three" else "no"


def margin(scene: dict) -> float:
    return float(8 * int(scene["scale"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
