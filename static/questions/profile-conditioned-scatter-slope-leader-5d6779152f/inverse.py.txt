"""Independent pixel-only inverse arm for the scatter-slope world."""

from __future__ import annotations

import numpy as np
from PIL import Image

_COLORS = {
    "blue": np.asarray((36, 111, 214), dtype=np.int16),
    "orange": np.asarray((232, 126, 38), dtype=np.int16),
}


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[tuple[float, float, int]] = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs: list[int] = []
        ys: list[int] = []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx)
            ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= 80:
            found.append((float(np.mean(xs)), float(np.mean(ys)), len(xs)))
    return found


def _pixel_slope(points: list[tuple[float, float, int]]) -> float:
    if len(points) != 7:
        raise ValueError(f"expected seven visible markers, found {len(points)}")
    xs = np.asarray([p[0] for p in points], dtype=float)
    # Raster y grows downward, so negate it to recover plot coordinates.
    ys = -np.asarray([p[1] for p in points], dtype=float)
    denominator = float(np.dot(xs - xs.mean(), xs - xs.mean()))
    if denominator <= 0:
        raise ValueError("degenerate horizontal marker geometry")
    return float(np.dot(xs - xs.mean(), ys - ys.mean()) / denominator)


def decision_from_image(image: Image.Image) -> str:
    """Return the color with the larger OLS slope using raster pixels alone."""
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    slopes: dict[str, float] = {}
    for name, rgb in _COLORS.items():
        distance = np.max(np.abs(array - rgb), axis=2)
        slopes[name] = _pixel_slope(_components(distance <= 8))
    return max(slopes, key=slopes.get)
