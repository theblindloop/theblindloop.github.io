"""Deterministic renderer and analytic arm for scatter-slope comparison."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 560, 520
LEFT, TOP, SIZE = 80, 54, 400
BLUE = (36, 111, 214)
ORANGE = (232, 126, 38)
BACKGROUND = (250, 250, 248)
QUARANTINE_GAP = 0.12


def _slope(points: list[list[float]]) -> float:
    values = np.asarray(points, dtype=float)
    x = values[:, 0]
    y = values[:, 1]
    return float(np.dot(x - x.mean(), y - y.mean()) / np.dot(x - x.mean(), x - x.mean()))


def _make_points(
    rng: random.Random, slope: float, intercept: float, horizontal_offset: float
) -> list[list[float]]:
    xs = np.linspace(0.16, 0.84, 7) + horizontal_offset
    xs += np.asarray([rng.uniform(-0.005, 0.005) for _ in xs])
    centered = xs - xs.mean()
    noise = np.asarray([rng.uniform(-0.035, 0.035) for _ in xs])
    # Remove the noise component parallel to x so the intended slope remains exact.
    noise -= centered * float(np.dot(centered, noise) / np.dot(centered, centered))
    noise -= noise.mean()
    ys = intercept + slope * centered + noise
    return [[round(float(x), 6), round(float(y), 6)] for x, y in zip(xs, ys)]


def sample_scene(seed: int) -> dict:
    """Sample balanced, separated slopes with seven points in each color."""
    rng = random.Random(int(seed))
    winner = "blue" if int(seed) % 2 == 0 else "orange"
    # Every eighth sample approaches the declared boundary without entering it.
    gap = rng.uniform(0.14, 0.19) if int(seed) % 8 == 0 else rng.uniform(0.25, 0.48)
    low = rng.uniform(0.24, 0.48)
    high = low + gap
    slopes = {winner: high, ("orange" if winner == "blue" else "blue"): low}
    for _ in range(100):
        series = {
            "blue": _make_points(rng, slopes["blue"], rng.uniform(0.47, 0.53), -0.025),
            "orange": _make_points(rng, slopes["orange"], rng.uniform(0.47, 0.53), 0.025),
        }
        pixels = []
        for color in ("blue", "orange"):
            pixels.extend((LEFT + p[0] * SIZE, TOP + (1.0 - p[1]) * SIZE) for p in series[color])
        if all((x1 - x2) ** 2 + (y1 - y2) ** 2 > 15**2 for i, (x1, y1) in enumerate(pixels) for x2, y2 in pixels[i + 1 :]):
            return {"series": series}
    raise RuntimeError("could not sample separated marker centers")


def render(scene: dict) -> Image.Image:
    """Render only fields that are available to the analytic decision."""
    series = scene["series"]
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    # A shared square coordinate frame makes pixel and data slopes comparable.
    for fraction in (0.25, 0.5, 0.75):
        x = round(LEFT + fraction * SIZE)
        y = round(TOP + fraction * SIZE)
        draw.line((x, TOP, x, TOP + SIZE), fill=(224, 226, 226), width=1)
        draw.line((LEFT, y, LEFT + SIZE, y), fill=(224, 226, 226), width=1)
    draw.rectangle((LEFT, TOP, LEFT + SIZE, TOP + SIZE), outline=(70, 74, 76), width=2)
    for color, rgb in (("blue", BLUE), ("orange", ORANGE)):
        for x_value, y_value in series[color]:
            x = round(LEFT + x_value * SIZE)
            y = round(TOP + (1.0 - y_value) * SIZE)
            radius = 7
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=rgb)
    return image


def analytic_gold(scene: dict) -> str:
    series = scene["series"]
    slopes = {color: _slope(series[color]) for color in ("blue", "orange")}
    return max(slopes, key=slopes.get)


def margin(scene: dict) -> float:
    series = scene["series"]
    values = sorted((_slope(series[color]) for color in ("blue", "orange")), reverse=True)
    return float(values[0] - values[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Point storage order is latent because the rendered series are unordered sets."""
    series = scene["series"]
    twin = {"series": {"blue": list(reversed(series["blue"])), "orange": list(reversed(series["orange"]))}}
    return [("reverse_point_storage_order", twin)]
