"""Renderer and analytic arm for bridge-pruned loop groups."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 900, 420
BG = (250, 249, 245)
FRAME = (196, 199, 201)
OUTLINE = (31, 35, 39)
LINK = (105, 112, 120)
CYAN = (38, 176, 190)
LABELS = ("left", "center", "right")

# All templates have ten vertices, twelve edges, two bridges, and degree
# multiset {2 x 6, 3 x 4}. WINNER has a rank-two theta core, a triangle core,
# and an intervening acyclic singleton. Each loser has three cycle cores.
WINNER = [
    [0, 2], [2, 1], [0, 3], [3, 1], [0, 4], [4, 5], [5, 1],
    [6, 7], [7, 8], [8, 6], [2, 9], [9, 6],
]
LOSER_END_SQUARE = [
    [0, 1], [1, 2], [2, 0], [3, 4], [4, 5], [5, 3],
    [6, 7], [7, 8], [8, 9], [9, 6], [2, 3], [5, 6],
]
LOSER_MIDDLE_SQUARE = [
    [0, 1], [1, 2], [2, 0], [3, 4], [4, 5], [5, 6], [6, 3],
    [7, 8], [8, 9], [9, 7], [2, 3], [6, 7],
]

# An intentionally irregular decagon avoids collinear non-edge chords.
SLOTS = [
    [0, -126], [73, -104], [122, -43], [117, 39], [68, 107],
    [-8, 127], [-78, 101], [-124, 35], [-110, -52], [-49, -116],
]


def _adjacency(edges: list[list[int]]) -> list[set[int]]:
    adjacency = [set() for _ in range(10)]
    for a, b in edges:
        adjacency[a].add(b)
        adjacency[b].add(a)
    return adjacency


def _components(edges: list[list[int]]) -> list[set[int]]:
    adjacency = _adjacency(edges)
    unseen = set(range(10))
    result = []
    while unseen:
        start = unseen.pop()
        component = {start}
        stack = [start]
        while stack:
            vertex = stack.pop()
            for other in adjacency[vertex]:
                if other in unseen:
                    unseen.remove(other)
                    component.add(other)
                    stack.append(other)
        result.append(component)
    return result


def _bridges(edges: list[list[int]]) -> list[list[int]]:
    base = len(_components(edges))
    result = []
    for index, edge in enumerate(edges):
        reduced = edges[:index] + edges[index + 1:]
        if len(_components(reduced)) > base:
            result.append(edge)
    return result


def _loop_group_count(edges: list[list[int]]) -> int:
    bridges = {tuple(sorted(edge)) for edge in _bridges(edges)}
    residual = [edge for edge in edges if tuple(sorted(edge)) not in bridges]
    components = _components(residual)
    count = 0
    for vertices in components:
        edge_count = sum(a in vertices and b in vertices for a, b in residual)
        if edge_count >= len(vertices):
            count += 1
    return count


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    templates = [WINNER, LOSER_END_SQUARE, LOSER_MIDDLE_SQUARE]
    rng.shuffle(templates)
    panels = []
    for template in templates:
        order = list(range(10))
        rng.shuffle(order)
        panels.append({"edges": [list(edge) for edge in template], "order": order})
    return {"panels": panels, "offset": [rng.randint(-4, 4), rng.randint(-6, 6)]}


def _positions(panel_index: int, panel: dict, offset: list[int]) -> list[list[int]]:
    cx = 150 + 300 * panel_index + offset[0]
    cy = 216 + offset[1]
    result = [[0, 0] for _ in range(10)]
    for vertex, slot_index in enumerate(panel["order"]):
        sx, sy = SLOTS[slot_index]
        result[vertex] = [cx + sx, cy + sy]
    return result


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    for panel_index, panel in enumerate(scene["panels"]):
        x0 = 10 + 300 * panel_index + scene["offset"][0]
        draw.rounded_rectangle(
            (x0, 45 + scene["offset"][1], x0 + 280, 383 + scene["offset"][1]),
            radius=14, outline=FRAME, width=3,
        )
        positions = _positions(panel_index, panel, scene["offset"])
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=OUTLINE, width=12)
        for a, b in panel["edges"]:
            draw.line([tuple(positions[a]), tuple(positions[b])], fill=LINK, width=8)
        for x, y in positions:
            draw.ellipse((x - 15, y - 15, x + 15, y + 15), fill=OUTLINE)
            draw.ellipse((x - 11, y - 11, x + 11, y + 11), fill=CYAN)
    return image


def analytic_gold(scene: dict) -> str:
    counts = [_loop_group_count(panel["edges"]) for panel in scene["panels"]]
    winners = [label for label, count in zip(LABELS, counts) if count == 2]
    if len(winners) != 1:
        raise ValueError("scene must have exactly one two-loop-group panel")
    return winners[0]


def margin(scene: dict) -> float:
    counts = [_loop_group_count(panel["edges"]) for panel in scene["panels"]]
    distances = sorted(abs(count - 2) for count in counts)
    return float(distances[1] - distances[0]) if len(distances) == 3 else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        if set(scene) != {"panels", "offset"} or len(scene["panels"]) != 3:
            return True
        if len(scene["offset"]) != 2 or any(abs(int(value)) > 8 for value in scene["offset"]):
            return True
        counts = []
        for panel in scene["panels"]:
            if set(panel) != {"edges", "order"} or sorted(panel["order"]) != list(range(10)):
                return True
            edges = panel["edges"]
            normalized = [tuple(sorted(edge)) for edge in edges]
            if len(edges) != 12 or len(set(normalized)) != 12:
                return True
            if any(len(edge) != 2 or edge[0] == edge[1] or min(edge) < 0 or max(edge) > 9 for edge in edges):
                return True
            if len(_components(edges)) != 1 or len(_bridges(edges)) != 2:
                return True
            degrees = sorted(len(neighbors) for neighbors in _adjacency(edges))
            if degrees != [2] * 6 + [3] * 4:
                return True
            counts.append(_loop_group_count(edges))
        return sorted(counts) != [2, 3, 3] or margin(scene) < 1.0
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = {
        "panels": [
            {"edges": [list(edge) for edge in reversed(panel["edges"])], "order": list(panel["order"])}
            for panel in scene["panels"]
        ],
        "offset": list(scene["offset"]),
    }
    return [("reverse_edge_serialization_order", twin)]
