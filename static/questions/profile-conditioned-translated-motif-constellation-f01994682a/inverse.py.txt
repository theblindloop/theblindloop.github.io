"""Independent raster-only inverse for the translated-motif world."""

from __future__ import annotations

import math

from PIL import Image


COLORS = ("red", "blue", "green")
RGB = {
    "red": (220, 55, 65),
    "blue": (48, 108, 220),
    "green": (42, 166, 98),
}
MIN_TRANSLATION = 45.0
MAX_TRANSLATION = 240.0


def _distance(left: tuple[float, float], right: tuple[float, float]) -> float:
    return math.hypot(left[0] - right[0], left[1] - right[1])


def _translation_score(points: list[tuple[float, float]]) -> float:
    hypotheses: list[tuple[float, float]] = []
    for source in points:
        for target in points:
            shift = (target[0] - source[0], target[1] - source[1])
            length = math.hypot(*shift)
            if MIN_TRANSLATION <= length <= MAX_TRANSLATION:
                hypotheses.append(shift)
    best = 999.0
    for shift in hypotheses:
        pairs = []
        for source_index, source in enumerate(points):
            moved = (source[0] + shift[0], source[1] + shift[1])
            for target_index, target in enumerate(points):
                pairs.append((_distance(moved, target), source_index, target_index))
        pairs.sort(key=lambda item: (item[0], item[1], item[2]))
        used_sources: set[int] = set()
        used_targets: set[int] = set()
        chosen: list[float] = []
        for residual, source_index, target_index in pairs:
            if source_index in used_sources or target_index in used_targets:
                continue
            used_sources.add(source_index)
            used_targets.add(target_index)
            chosen.append(residual)
            if len(chosen) == 4:
                break
        if len(chosen) == 4:
            best = min(best, sum(chosen) / 4.0)
    return best


def _centers_from_coordinates(
    coordinates: set[tuple[int, int]],
) -> list[tuple[float, float]]:
    centers: list[tuple[float, float]] = []
    while coordinates:
        start = coordinates.pop()
        stack = [start]
        count = 0
        sum_x = 0
        sum_y = 0
        while stack:
            x, y = stack.pop()
            count += 1
            sum_x += x
            sum_y += y
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in coordinates:
                    coordinates.remove(neighbor)
                    stack.append(neighbor)
        if count >= 80:
            centers.append((sum_x / count, sum_y / count))
    if len(centers) != 8:
        raise ValueError(f"expected eight components, found {len(centers)}")
    return centers


def _components(image: Image.Image) -> dict[str, list[tuple[float, float]]]:
    image = image.convert("RGB")
    width, height = image.size
    coordinates = {name: set() for name in COLORS}
    palette = {tuple(RGB[name]): name for name in COLORS}
    for index, pixel in enumerate(image.getdata()):
        name = palette.get(tuple(pixel))
        if name is not None:
            coordinates[name].add((index % width, index // width))
    return {
        name: _centers_from_coordinates(coordinates[name])
        for name in COLORS
    }


def decision_from_image(image: Image.Image) -> str:
    components = _components(image)
    scores = {}
    for color_name in COLORS:
        points = components[color_name]
        scores[color_name] = _translation_score(points)
    ordered = sorted((score, color) for color, score in scores.items())
    if ordered[0][0] >= 80.0 or ordered[1][0] - ordered[0][0] < 4.0:
        raise ValueError("translation-repeat evidence is ambiguous")
    return ordered[0][1]
