"""Deterministic renderer and analytic reference for a translated-motif world."""

from __future__ import annotations

import copy
import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


COLORS = ("red", "blue", "green")
RGB = {
    "red": (220, 55, 65),
    "blue": (48, 108, 220),
    "green": (42, 166, 98),
}
WIDTH = 960
HEIGHT = 560
LANE_X = (160, 480, 800)
LANE_Y = 280
MIN_TRANSLATION = 45.0
MAX_TRANSLATION = 240.0


def _distance(left: tuple[float, float], right: tuple[float, float]) -> float:
    return math.hypot(left[0] - right[0], left[1] - right[1])


def _normalise_points(points: Iterable[Iterable[float]]) -> list[list[float]]:
    values = [(float(point[0]), float(point[1])) for point in points]
    if not values:
        raise ValueError("a pattern needs points")
    min_x = min(point[0] for point in values)
    max_x = max(point[0] for point in values)
    min_y = min(point[1] for point in values)
    max_y = max(point[1] for point in values)
    center_x = (min_x + max_x) / 2.0
    center_y = (min_y + max_y) / 2.0
    width = max(max_x - min_x, 1.0)
    height = max(max_y - min_y, 1.0)
    return [
        [
            (point[0] - center_x) * (180.0 / width),
            (point[1] - center_y) * (130.0 / height),
        ]
        for point in values
    ]


def _translation_score(points: Iterable[Iterable[float]]) -> float:
    """Return the residual of the best four-pair translation hypothesis."""

    values = [(float(point[0]), float(point[1])) for point in points]
    if len(values) != 8:
        return 999.0
    hypotheses: list[tuple[float, float]] = []
    for source in values:
        for target in values:
            shift = (target[0] - source[0], target[1] - source[1])
            length = math.hypot(*shift)
            if MIN_TRANSLATION <= length <= MAX_TRANSLATION:
                hypotheses.append(shift)
    if not hypotheses:
        return 999.0
    best = 999.0
    for shift in hypotheses:
        pairs = []
        for source_index, source in enumerate(values):
            moved = (source[0] + shift[0], source[1] + shift[1])
            for target_index, target in enumerate(values):
                pairs.append((_distance(moved, target), source_index, target_index))
        pairs.sort(key=lambda item: (item[0], item[1], item[2]))
        used_sources: set[int] = set()
        used_targets: set[int] = set()
        chosen: list[float] = []
        for residual, source_index, target_index in pairs:
            if source_index in used_sources or target_index in used_targets:
                continue
            used_sources.add(source_index)
            used_targets.add(target_index)
            chosen.append(residual)
            if len(chosen) == 4:
                break
        if len(chosen) == 4:
            best = min(best, sum(chosen) / 4.0)
    return best


def _random_points(rng: random.Random, count: int = 8) -> list[list[float]]:
    points: list[list[float]] = []
    while len(points) < count:
        candidate = [rng.uniform(-90.0, 90.0), rng.uniform(-62.0, 62.0)]
        if all(_distance(tuple(candidate), tuple(point)) >= 26.0 for point in points):
            points.append(candidate)
    return _normalise_points(points)


def _target_motif(rng: random.Random) -> tuple[list[list[float]], list[float]]:
    motif: list[list[float]] = []
    while len(motif) < 4:
        candidate = [rng.uniform(-56.0, 16.0), rng.uniform(-48.0, 48.0)]
        if all(_distance(tuple(candidate), tuple(point)) >= 30.0 for point in motif):
            motif.append(candidate)
    translation = [rng.uniform(102.0, 126.0), rng.uniform(-18.0, 18.0)]
    return motif, translation


def _target_points(scene: dict) -> list[list[float]]:
    motif = scene["motif"]
    shift = scene["translation"]
    raw = [list(point) for point in motif]
    raw.extend([[point[0] + shift[0], point[1] + shift[1]] for point in motif])
    return _normalise_points(raw)


def _patterns(scene: dict) -> list[list[list[float]]]:
    return [_target_points(scene), scene["decoy_patterns"][0], scene["decoy_patterns"][1]]


def _scene_points(scene: dict) -> list[list[list[float]]]:
    target_lane = scene["lane_order"].index(scene["target_color"])
    patterns = _patterns(scene)
    selected: list[list[list[float]]] = []
    decoy_index = 1
    for lane in range(3):
        if lane == target_lane:
            selected.append(patterns[0])
        else:
            selected.append(patterns[decoy_index])
            decoy_index += 1
    return selected


def _minimum_separation(scene: dict) -> float:
    minimum = float("inf")
    for lane, points in enumerate(_scene_points(scene)):
        transformed = [
            _transformed(
                point,
                scene["lane_centers"][lane],
                float(scene["global_scale"]),
                float(scene["global_rotation"]),
            )
            for point in points
        ]
        for index, left in enumerate(transformed):
            for right in transformed[index + 1 :]:
                minimum = min(minimum, _distance(left, right))
    return minimum


def _transformed(point: Iterable[float], center: Iterable[float], scale: float, angle: float) -> tuple[float, float]:
    x, y = float(point[0]) * scale, float(point[1]) * scale
    cosine = math.cos(angle)
    sine = math.sin(angle)
    return (
        float(center[0]) + cosine * x - sine * y,
        float(center[1]) + sine * x + cosine * y,
    )


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_color = rng.choice(COLORS)
    lane_order = rng.sample(list(COLORS), k=3)
    motif, translation = _target_motif(rng)
    decoys: list[list[list[float]]] = []
    for _ in range(2):
        for _attempt in range(500):
            candidate = _random_points(rng)
            if _translation_score(candidate) >= 24.0:
                decoys.append(candidate)
                break
        else:
            raise RuntimeError("could not sample a non-repeating decoy")
    fields = [{"lane": lane} for lane in rng.sample([0, 1, 2], k=3)]
    scene = {
        "seed": int(seed),
        "target_color": target_color,
        "fields": fields,
        "lane_order": lane_order,
        "palette": {name: list(RGB[name]) for name in COLORS},
        "canvas_size": [WIDTH, HEIGHT],
        "global_rotation": rng.uniform(-math.pi, math.pi),
        "global_scale": rng.uniform(0.90, 1.10),
        "dot_radius": rng.randint(8, 10),
        "lane_centers": [[x, LANE_Y + rng.randint(-6, 6)] for x in LANE_X],
        "motif": motif,
        "translation": translation,
        "decoy_patterns": decoys,
    }
    if _minimum_separation(scene) < 2.0 * scene["dot_radius"] + 6.0 or margin(scene) < 18.0:
        return sample_scene(seed + 1_000_003)
    return scene


def render(scene: dict) -> Image.Image:
    width, height = (int(scene["canvas_size"][0]), int(scene["canvas_size"][1]))
    image = Image.new("RGB", (width, height), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    selected = _scene_points(scene)
    for lane, points in enumerate(selected):
        color_name = scene["lane_order"][lane]
        center = scene["lane_centers"][lane]
        radius = int(scene["dot_radius"])
        fill = tuple(int(value) for value in scene["palette"][color_name])
        for point in points:
            x, y = _transformed(
                point,
                center,
                float(scene["global_scale"]),
                float(scene["global_rotation"]),
            )
            box = [
                int(round(x - radius)),
                int(round(y - radius)),
                int(round(x + radius)),
                int(round(y + radius)),
            ]
            draw.ellipse(box, fill=fill)
    return image


def analytic_gold(scene: dict) -> str:
    return scene["target_color"]


def margin(scene: dict) -> float:
    scores = []
    for points in _scene_points(scene):
        transformed = [
            _transformed(
                point,
                (0.0, 0.0),
                float(scene["global_scale"]),
                float(scene["global_rotation"]),
            )
            for point in points
        ]
        scores.append(_translation_score(transformed))
    scores.sort()
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["fields"] = list(reversed(twin["fields"]))
    return [("reverse_field_record_order", twin)]
