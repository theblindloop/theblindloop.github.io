"""Independent image-only inverse for the rendered heatmap.

The implementation does not know scene coordinates or renderer palettes. It
finds solid-color components, recognizes the 4x4 cell grid by component shape,
decodes values from the nine visible key swatches, and reads row identity from
the fixed A-D badge colors that are simultaneously visible in the raster.
"""

from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image


LABEL_COLORS = {
    "A": (218, 55, 55),
    "B": (36, 150, 92),
    "C": (137, 79, 191),
    "D": (232, 142, 36),
}


@dataclass(frozen=True)
class Component:
    color: tuple[int, int, int]
    x0: int
    y0: int
    x1: int
    y1: int
    count: int

    @property
    def width(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def height(self) -> int:
        return self.y1 - self.y0 + 1

    @property
    def cx(self) -> float:
        return (self.x0 + self.x1) / 2.0

    @property
    def cy(self) -> float:
        return (self.y0 + self.y1) / 2.0


def _components(mask: np.ndarray, color: tuple[int, int, int]) -> list[Component]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[Component] = []
    for y_value, x_value in np.argwhere(mask):
        y = int(y_value)
        x = int(x_value)
        if seen[y, x]:
            continue
        stack = [(x, y)]
        seen[y, x] = True
        x0 = x1 = x
        y0 = y1 = y
        count = 0
        while stack:
            px, py = stack.pop()
            count += 1
            x0 = min(x0, px)
            x1 = max(x1, px)
            y0 = min(y0, py)
            y1 = max(y1, py)
            for nx, ny in ((px - 1, py), (px + 1, py), (px, py - 1), (px, py + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((nx, ny))
        output.append(Component(color, x0, y0, x1, y1, count))
    return output


def _solid_color_components(arr: np.ndarray) -> list[Component]:
    colors, counts = np.unique(arr.reshape(-1, 3), axis=0, return_counts=True)
    output: list[Component] = []
    for raw_color, count in zip(colors, counts):
        color = tuple(int(channel) for channel in raw_color)
        if int(count) < 250 or max(color) - min(color) < 7:
            continue
        mask = np.all(arr == raw_color, axis=2)
        output.extend(_components(mask, color))
    return output


def _measure(image: Image.Image) -> tuple[list[list[int]], list[str], list[int]]:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    components = _solid_color_components(arr)

    cells = [
        item
        for item in components
        if 68 <= item.width <= 76 and 68 <= item.height <= 76 and item.count >= 3500
    ]
    if len(cells) != 16:
        raise ValueError(f"expected 16 recoverable heatmap cells, found {len(cells)}")

    swatches = [
        item
        for item in components
        if 23 <= item.width <= 25 and 23 <= item.height <= 25 and item.count >= 520
    ]
    if len(swatches) != 9:
        raise ValueError(f"expected 9 recoverable value-key swatches, found {len(swatches)}")
    swatches.sort(key=lambda item: item.cx)
    value_for_color = {item.color: index for index, item in enumerate(swatches, start=1)}
    if len(value_for_color) != 9:
        raise ValueError("value key contains repeated colors")

    cells.sort(key=lambda item: (item.cy, item.cx))
    rows = [sorted(cells[index : index + 4], key=lambda item: item.cx) for index in range(0, 16, 4)]
    if any(max(cell.cy for cell in row) - min(cell.cy for cell in row) > 3 for row in rows):
        raise ValueError("heatmap cells do not form four horizontal rows")
    if any(len({round(cell.cx) for cell in row}) != 4 for row in rows):
        raise ValueError("heatmap columns are unresolved")

    values: list[list[int]] = []
    labels: list[str] = []
    grid_left = min(cell.x0 for cell in cells)
    for row in rows:
        try:
            values.append([value_for_color[cell.color] for cell in row])
        except KeyError as exc:
            raise ValueError("cell color is absent from the visible value key") from exc
        y0 = min(item.y0 for item in row)
        y1 = max(item.y1 for item in row)
        label_counts = {
            label: int(np.all(arr[y0 : y1 + 1, :grid_left] == np.array(color), axis=2).sum())
            for label, color in LABEL_COLORS.items()
        }
        label, count = max(label_counts.items(), key=lambda item: item[1])
        if count < 300:
            raise ValueError("row badge is missing or unreadable")
        labels.append(label)
    if len(set(labels)) != 4:
        raise ValueError("row badges are not a permutation of A-D")

    scores = [row[1] + row[2] - row[0] - row[3] for row in values]
    return values, labels, scores


def decision_from_image(image: Image.Image) -> str:
    """Return the winning visible row label using the PNG alone."""
    _values, labels, scores = _measure(image)
    ordered = sorted(scores, reverse=True)
    if len(ordered) < 2 or ordered[0] == ordered[1]:
        raise ValueError("top center-edge contrast is tied")
    return labels[max(range(4), key=scores.__getitem__)]


def diagnostics_from_image(image: Image.Image) -> dict:
    values, labels, scores = _measure(image)
    ordered = sorted(scores, reverse=True)
    return {
        "values": values,
        "row_labels": labels,
        "scores": scores,
        "decision": labels[max(range(4), key=scores.__getitem__)],
        "runner_up_margin": ordered[0] - ordered[1],
    }
