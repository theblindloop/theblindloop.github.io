"""Deterministic scene sampler and renderer for center-versus-edge heatmaps."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH = 720
HEIGHT = 540
CELL = 72
GAP = 10
GRID_X = 205
GRID_Y = 166
ROW_TOTAL = 22
LABELS = ("A", "B", "C", "D")

LABEL_COLORS = {
    "A": (218, 55, 55),
    "B": (36, 150, 92),
    "C": (137, 79, 191),
    "D": (232, 142, 36),
}

PALETTES = {
    "blue": [
        (222, 235, 247), (198, 219, 239), (172, 205, 232),
        (143, 188, 222), (112, 166, 210), (82, 145, 198),
        (55, 123, 181), (33, 101, 163), (18, 78, 140),
    ],
    "teal": [
        (222, 242, 238), (194, 229, 223), (165, 216, 207),
        (132, 198, 188), (99, 178, 168), (70, 157, 148),
        (45, 135, 128), (27, 111, 108), (15, 87, 88),
    ],
    "purple": [
        (237, 230, 244), (220, 205, 233), (201, 179, 220),
        (180, 151, 206), (158, 123, 191), (135, 96, 174),
        (112, 72, 154), (89, 52, 132), (68, 36, 108),
    ],
}


def _pair_with_sum(total: int, rng: random.Random) -> list[int]:
    choices = [(a, total - a) for a in range(1, 10) if 1 <= total - a <= 9]
    a, b = rng.choice(choices)
    pair = [a, b]
    rng.shuffle(pair)
    return pair


def _row_for_center_sum(center_sum: int, rng: random.Random) -> list[int]:
    center = _pair_with_sum(center_sum, rng)
    edge = _pair_with_sum(ROW_TOTAL - center_sum, rng)
    return [edge[0], center[0], center[1], edge[1]]


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene; consecutive seeds cycle the winning label."""
    rng = random.Random(int(seed))
    target_label = LABELS[int(seed) % len(LABELS)]
    row_labels = list(LABELS)
    rng.shuffle(row_labels)
    target_row = row_labels.index(target_label)

    near_boundary = int(seed) % 7 == 0
    center_sums = [13 if near_boundary else 15, 12, 10, 8]
    remaining = center_sums[1:]
    rng.shuffle(remaining)
    assigned = []
    for row in range(4):
        assigned.append(center_sums[0] if row == target_row else remaining.pop())
    values = [_row_for_center_sum(value, rng) for value in assigned]

    distractors = []
    for index in range(8):
        if index % 2 == 0:
            x = rng.randint(34, 104)
        else:
            x = rng.randint(616, 684)
        distractors.append([x, rng.randint(175, 492), rng.choice([3, 4, 5])])

    return {
        "values": values,
        "row_labels": row_labels,
        "palette_id": rng.choice(sorted(PALETTES)),
        "grid_dx": rng.choice([-12, -6, 0, 6, 12]),
        "grid_dy": rng.choice([-6, 0, 6]),
        "distractors": distractors,
    }


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    try:
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def render(scene: dict) -> Image.Image:
    values = scene["values"]
    row_labels = scene["row_labels"]
    palette = PALETTES[scene["palette_id"]]
    gx = GRID_X + int(scene["grid_dx"])
    gy = GRID_Y + int(scene["grid_dy"])
    distractors = scene["distractors"]

    image = Image.new("RGB", (WIDTH, HEIGHT), (248, 249, 251))
    draw = ImageDraw.Draw(image)
    dark = (37, 43, 52)
    muted = (91, 101, 113)

    draw.text((WIDTH // 2, 24), "CENTER vs EDGE", fill=dark, font=_font(25, True), anchor="ma")
    draw.text(
        (WIDTH // 2, 56),
        "score = (middle two) - (outer two)     |     every row total = 22",
        fill=muted,
        font=_font(15),
        anchor="ma",
    )

    legend_x = 177
    legend_y = 91
    draw.text((legend_x - 34, legend_y + 11), "value", fill=dark, font=_font(14, True), anchor="rm")
    for index, color in enumerate(palette, start=1):
        x = legend_x + (index - 1) * 39
        draw.rectangle((x, legend_y, x + 23, legend_y + 23), fill=color)
        draw.text((x + 12, legend_y + 31), str(index), fill=dark, font=_font(12, True), anchor="ma")

    for x, y, radius in distractors:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), outline=(177, 184, 193), width=2)

    value_font = _font(23, True)
    label_font = _font(18, True)
    for row, (label, row_values) in enumerate(zip(row_labels, values)):
        y = gy + row * (CELL + GAP)
        chip_x = gx - 59
        chip_y = y + CELL // 2
        chip_color = LABEL_COLORS[label]
        draw.ellipse((chip_x - 16, chip_y - 16, chip_x + 16, chip_y + 16), fill=chip_color)
        draw.text((chip_x, chip_y + 1), label, fill=(255, 255, 255), font=label_font, anchor="mm")
        for column, value in enumerate(row_values):
            x = gx + column * (CELL + GAP)
            color = palette[int(value) - 1]
            draw.rectangle((x, y, x + CELL - 1, y + CELL - 1), fill=color)
            ink = (250, 252, 254) if int(value) >= 6 else (28, 34, 42)
            draw.text(
                (x + CELL // 2, y + CELL // 2 + 1),
                str(value),
                fill=ink,
                font=value_font,
                anchor="mm",
            )

    draw.text(
        (WIDTH // 2, 516),
        "Rows are identified by the colored A-D badges at left.",
        fill=muted,
        font=_font(13),
        anchor="mm",
    )
    return image


def _scores(scene: dict) -> list[int]:
    return [int(row[1]) + int(row[2]) - int(row[0]) - int(row[3]) for row in scene["values"]]


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winner = max(range(len(scores)), key=scores.__getitem__)
    return str(scene["row_labels"][winner])


def margin(scene: dict) -> float:
    ordered = sorted(_scores(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Distractor draw order is latent but invisible because all use one style."""
    if len(scene["distractors"]) < 2:
        return []
    twin = copy.deepcopy(scene)
    twin["distractors"] = list(reversed(twin["distractors"]))
    return [("reverse_distractor_order", twin)]
