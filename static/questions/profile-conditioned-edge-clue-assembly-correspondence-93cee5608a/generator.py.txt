"""Deterministic scene sampler, analytic solver, and Pillow renderer."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 650
COLORS = ("red", "green", "blue", "orange")
RGB = {
    "red": (220, 55, 65),
    "green": (45, 150, 90),
    "blue": (45, 100, 210),
    "orange": (235, 135, 35),
}
NAVY = (22, 50, 82)
GRAY = (112, 118, 126)
MAGENTA = (202, 38, 156)
INK = (28, 32, 38)
PAPER = (255, 255, 255)
TILE_FILL = (247, 249, 251)

# Cell order is upper-left, upper-right, lower-left, lower-right. Edges are N,E,S,W.
_CLUE_SIDES = (0, 1, 3, 2)


def _checks(tiles: list[dict], clues: list[int], placement: tuple[int, ...]) -> list[bool]:
    a, b, c, d = (tiles[index]["edges"] for index in placement)
    return [
        a[1] == b[3],
        a[2] == c[0],
        b[2] == d[0],
        c[1] == d[3],
        a[0] == clues[0],
        b[1] == clues[1],
        c[3] == clues[2],
        d[2] == clues[3],
    ]


def _ranked(tiles: list[dict], clues: list[int]) -> list[tuple[int, tuple[int, ...]]]:
    ranked = []
    for placement in itertools.permutations(range(4)):
        ranked.append((sum(_checks(tiles, clues, placement)), placement))
    return sorted(ranked, key=lambda item: (-item[0], item[1]))


def _perfect(tiles: list[dict], clues: list[int]) -> list[tuple[int, ...]]:
    return [placement for score, placement in _ranked(tiles, clues) if score == 8]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    for _ in range(10000):
        edges = [[rng.randint(1, 4) for _side in range(4)] for _tile in range(4)]
        # Build a valid solution in canonical cell order.
        edges[1][3] = edges[0][1]
        edges[2][0] = edges[0][2]
        edges[3][0] = edges[1][2]
        edges[3][3] = edges[2][1]
        clues = [edges[0][0], edges[1][1], edges[2][3], edges[3][2]]
        bare = [{"tag": COLORS[i], "edges": edge} for i, edge in enumerate(edges)]
        # A shown outer clue must leave at least two locally plausible tiles.
        locally_ambiguous = all(
            sum(tile["edges"][side] == clues[cell] for tile in bare) >= 2
            for cell, side in enumerate(_CLUE_SIDES)
        )
        if locally_ambiguous and _perfect(bare, clues) == [(0, 1, 2, 3)]:
            break
    else:
        raise RuntimeError("could not sample a unique locally ambiguous assembly")

    # Consecutive seeds cycle every tag through every logical tile position.
    shift = int(seed) % 4
    tags = list(COLORS[shift:] + COLORS[:shift])
    if (int(seed) // 4) % 2:
        tags[1], tags[3] = tags[3], tags[1]
    tiles = [{"tag": tags[i], "edges": edges[i]} for i in range(4)]
    loose_order = list(range(4))
    rng.shuffle(loose_order)
    target_cell = (int(seed) // 2) % 4
    return {
        "tiles": tiles,
        "frame_clues": clues,
        "target_cell": target_cell,
        "loose_order": loose_order,
    }


def analytic_gold(scene: dict) -> str:
    tiles = scene["tiles"]
    clues = scene["frame_clues"]
    target = scene["target_cell"]
    solutions = _perfect(tiles, clues)
    if len(solutions) != 1:
        raise ValueError(f"expected one perfect assembly, found {len(solutions)}")
    return str(tiles[solutions[0][target]]["tag"])


def margin(scene: dict) -> float:
    ranked = _ranked(scene["tiles"], scene["frame_clues"])
    return float(ranked[0][0] - ranked[1][0])


def is_quarantined(scene: dict) -> bool:
    return len(_perfect(scene["tiles"], scene["frame_clues"])) != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    names = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf" if bold
        else "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    )
    for name in names:
        try:
            return ImageFont.truetype(name, size)
        except OSError:
            pass
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2, xy[1] - (box[3] - box[1]) / 2), text, font=font, fill=fill)


def _code_points(center: tuple[int, int], horizontal: bool, count: int) -> Iterable[tuple[int, int]]:
    spacing = 17
    start = -spacing * (count - 1) / 2
    for index in range(count):
        offset = int(round(start + spacing * index))
        yield (center[0] + offset, center[1]) if horizontal else (center[0], center[1] + offset)


def _draw_code(draw: ImageDraw.ImageDraw, center: tuple[int, int], horizontal: bool, count: int) -> None:
    for x, y in _code_points(center, horizontal, count):
        draw.rectangle((x - 4, y - 4, x + 4, y + 4), fill=NAVY)


def _draw_tile(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], tile: dict) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=TILE_FILL, outline=GRAY, width=5)
    cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
    edges = tile["edges"]
    _draw_code(draw, (cx, y0 + 15), True, edges[0])
    _draw_code(draw, (x1 - 15, cy), False, edges[1])
    _draw_code(draw, (cx, y1 - 15), True, edges[2])
    _draw_code(draw, (x0 + 15, cy), False, edges[3])
    draw.ellipse((cx - 20, cy - 20, cx + 20, cy + 20), fill=RGB[tile["tag"]], outline=PAPER, width=3)
    # Identical upright marker on every tile; orientation is intentionally fixed.
    draw.polygon([(cx, y0 + 31), (cx - 7, y0 + 42), (cx + 7, y0 + 42)], fill=INK)


def render(scene: dict) -> Image.Image:
    tiles = scene["tiles"]
    clues = scene["frame_clues"]
    target = scene["target_cell"]
    loose_order = scene["loose_order"]
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25, bold=True)
    body_font = _font(17)
    small_font = _font(15, bold=True)
    _centered_text(draw, (230, 55), "ASSEMBLY FRAME", title_font, INK)
    _centered_text(draw, (755, 55), "LOOSE TILES — KEEP UPRIGHT", title_font, INK)
    _centered_text(draw, (230, 92), "match all navy edge-square counts", body_font, GRAY)
    _centered_text(draw, (755, 92), "each color appears exactly once", body_font, GRAY)
    draw.line((500, 35, 500, 615), fill=(220, 224, 229), width=3)

    x0, y0, cell = 80, 190, 150
    x1, y1 = x0 + 2 * cell, y0 + 2 * cell
    draw.rectangle((x0, y0, x1, y1), fill=TILE_FILL, outline=GRAY, width=5)
    draw.line((x0 + cell, y0, x0 + cell, y1), fill=GRAY, width=5)
    draw.line((x0, y0 + cell, x1, y0 + cell), fill=GRAY, width=5)
    # One sparse outer clue per cell: A north, B east, C west, D south.
    _draw_code(draw, (x0 + cell // 2, y0 - 18), True, clues[0])
    _draw_code(draw, (x1 + 18, y0 + cell // 2), False, clues[1])
    _draw_code(draw, (x0 - 18, y0 + cell + cell // 2), False, clues[2])
    _draw_code(draw, (x0 + cell + cell // 2, y1 + 18), True, clues[3])
    tx = x0 + (target % 2) * cell + cell // 2
    ty = y0 + (target // 2) * cell + cell // 2
    draw.ellipse((tx - 27, ty - 27, tx + 27, ty + 27), outline=MAGENTA, width=8)
    _centered_text(draw, (tx, ty), "?", _font(25, bold=True), MAGENTA)

    boxes = [(555, 145, 675, 265), (785, 145, 905, 265), (555, 390, 675, 510), (785, 390, 905, 510)]
    for box, tile_index in zip(boxes, loose_order, strict=True):
        _draw_tile(draw, box, tiles[tile_index])
    _centered_text(draw, (730, 565), "NAVY COUNTS MATCH  •  DO NOT ROTATE", small_font, NAVY)
    return image
