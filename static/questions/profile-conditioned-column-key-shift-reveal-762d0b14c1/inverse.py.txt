"""Pixel-only inverse arm for the declared per-column wrap-shift reveal.

Receives only the rendered PNG. It finds the blue-framed panel, locates each
column's magenta key square to recover that column's vertical shift, reverses
each shift (unwrapping the columns), reassembles the upright triangle, and reads
whether it points up (A) or down (B) from its ink centroid within the panel.
No scene, latent, renderer, or gold data is used.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

N = 6
G = 300
COLW = G // N
MARKER_ROW = G - 14


def _mask(arr, color, tol):
    return np.abs(arr - np.array(color, dtype=np.float32)).sum(axis=2) < tol


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    arr = np.asarray(image, dtype=np.float32)

    # locate the panel by its blue frame, then take the interior below the border
    fm = _mask(arr, (0, 70, 140), 70)
    ys, xs = np.nonzero(fm)
    if len(ys) == 0:
        raise ValueError("no grid frame found in image")
    ox = int(xs.min()) + 3
    oy = int(ys.min()) + 3

    ink = _mask(arr, (0, 0, 0), 80)

    # recover each column's shift from its magenta key square and unwrap it
    out = np.zeros((G, G), dtype=bool)
    for c in range(N):
        cx0 = ox + c * COLW
        cx1 = cx0 + COLW
        col = ink[oy:oy + G, cx0:cx1]

        dm = _mask(arr[oy:oy + G, cx0:cx1, :], (230, 0, 220), 80)
        dys, _ = np.nonzero(dm)
        if len(dys) == 0:
            s = 0
        else:
            dot_row = dys.mean()
            s = (MARKER_ROW - dot_row) % G
        out[:, c * COLW:(c + 1) * COLW] = np.roll(col, s, axis=0)

    ys, xs = np.nonzero(out)
    if len(ys) == 0:
        raise ValueError("no ink found after unwrap")
    cy = ys.mean() / G
    # apex-down (B) centroid is in the upper half; apex-up (A) in the lower half
    return "B" if cy < 0.5 else "A"
