"""Declared per-column wrap-shift reassembly reveal world.

A single solid ``A`` (apex-up) or ``B`` (apex-down) triangle is drawn across a
square panel. The panel is then cut into N vertical columns and every column is
cyclically shifted vertically (wrapped around the panel top/bottom) by a random
whole-pixel amount. A small magenta key square is drawn inside each column at a
fixed reference row before shifting; after the shift it marks how far that
column moved, so reading the key square's row recovers the column's shift. The
declared transform is to reverse each column's shift (unwrap) and reassemble the
upright glyph; the answer is whether the reconstructed triangle points up (A) or
down (B).

The transform is a per-column cyclic vertical registration (a discrete 1D
unwrapping by keyed column shifts), distinct from the per-cell rotation of the
derotation world, the sinusoidal column bend of the rectification world, and the
rank-order panel collation worlds: no cell rotates about its center and no panel
changes grid position.
"""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

N = 6                    # number of vertical columns
G = 300                  # side of the square panel in pixels
COLW = G // N            # column width
X0 = 60
Y0 = 60
BORDER = 3               # frame outline width
MARKER_ROW = G - 14      # reference row (offset from panel top) of each key square
DOT = 12                 # key square side length
RENDERER_VERSION = "column_key_shift_reveal-0.1.0"

FRAME = "rgb(0,70,140)"
DOTCOL = "rgb(230,0,220)"
INK = "black"
BG = "white"


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    direction = rng.choice(["A", "B"])
    shifts = [rng.randrange(0, G) for _ in range(N)]
    return {
        "direction": direction,
        "cell": COLW,
        "n": N,
        "x0": X0,
        "y0": Y0,
        "shifts": shifts,
        "marker_row": MARKER_ROW,
        "seed": seed,
    }


def _tri_points(scene: dict) -> list[tuple[float, float]]:
    g = G
    x0, y0 = X0, Y0
    if scene["direction"] == "A":
        # apex at top center, base along the bottom edge
        return [(x0 + 0.5 * g, y0), (x0 + 0.05 * g, y0 + g), (x0 + 0.95 * g, y0 + g)]
    # apex at bottom center, base along the top edge
    return [(x0 + 0.5 * g, y0 + g), (x0 + 0.05 * g, y0), (x0 + 0.95 * g, y0)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (512, 512), BG)
    d = ImageDraw.Draw(img)
    d.polygon(_tri_points(scene), fill=INK)

    # key square at the reference row near the bottom of each column, before shift
    for c in range(N):
        cx = X0 + c * COLW
        d.rectangle(
            [cx + (COLW - DOT) // 2, Y0 + MARKER_ROW, cx + (COLW + DOT) // 2, Y0 + MARKER_ROW + DOT],
            fill=DOTCOL,
        )

    # cyclically shift each column vertically by a random amount (lossless)
    arr = np.asarray(img, dtype=np.uint8).copy()
    shifts = scene["shifts"]
    for c in range(N):
        cx0 = X0 + c * COLW
        cx1 = cx0 + COLW
        col = arr[Y0:Y0 + G, cx0:cx1, :]
        arr[Y0:Y0 + G, cx0:cx1, :] = np.roll(col, -shifts[c], axis=0)
    img = Image.fromarray(arr, mode="RGB")

    d = ImageDraw.Draw(img)
    d.rectangle(
        [X0 - BORDER, Y0 - BORDER, X0 + G + BORDER, Y0 + G + BORDER],
        outline=FRAME,
        width=BORDER,
    )
    return img


def analytic_gold(scene: dict) -> str:
    return scene["direction"]


def margin(scene: dict) -> float:
    # analytic verifiability margin: the triangle centroid sits G/6 pixels off
    # the panel centre in the vertical direction.
    return float(G) / 6.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list:
    # no pixel-identical label-preserving transform for not_applicable mode
    return []
