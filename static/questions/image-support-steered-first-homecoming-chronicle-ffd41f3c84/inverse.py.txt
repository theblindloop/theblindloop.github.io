"""Independent pixel-only inverse arm for First Homecoming Chronicle."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

_ORDER = np.array((236, 178, 52), dtype=np.uint8)
_COLORS = {
    "red": np.array((211, 64, 73), dtype=np.uint8),
    "green": np.array((42, 139, 91), dtype=np.uint8),
    "blue": np.array((47, 104, 190), dtype=np.uint8),
}


def _components(mask: np.ndarray, minimum: int = 1) -> list[tuple[int, int, int, int, int, float, float]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    result = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        xs, ys = [], []
        while stack:
            cy, cx = stack.pop()
            xs.append(cx); ys.append(cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(xs) >= minimum:
            result.append((min(xs), min(ys), max(xs), max(ys), len(xs), float(sum(xs))/len(xs), float(sum(ys))/len(ys)))
    return result


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    beads: dict[str, list[tuple[float, float]]] = {}
    all_centers = []
    for name, rgb in _COLORS.items():
        comps = _components(np.all(arr == rgb, axis=2), 180)
        if len(comps) != 4:
            raise ValueError(f"expected four visible {name} beads, found {len(comps)}")
        beads[name] = [(c[5], c[6]) for c in comps]
        all_centers.extend(beads[name])

    def largest_gap_split(values: list[float]) -> float:
        ordered_values = sorted(values)
        gaps = [(ordered_values[i + 1] - ordered_values[i], i) for i in range(len(ordered_values) - 1)]
        gap, index = max(gaps)
        if gap < 40:
            raise ValueError("panel groups are not visibly separated")
        return (ordered_values[index] + ordered_values[index + 1]) / 2.0

    split_x = largest_gap_split([p[0] for p in all_centers])
    split_y = largest_gap_split([p[1] for p in all_centers])
    grouped: dict[int, dict[str, tuple[float, float]]] = {i: {} for i in range(4)}
    for name, centers in beads.items():
        for x, y in centers:
            group = int(x > split_x) + 2 * int(y > split_y)
            if name in grouped[group]:
                raise ValueError("duplicate bead identity in one frame")
            grouped[group][name] = (x, y)
    if any(set(group) != set(_COLORS) for group in grouped.values()):
        raise ValueError("each frame must contain every bead identity")

    markers = _components(np.all(arr == _ORDER, axis=2), 25)
    if len(markers) != 10:
        raise ValueError("frame dots are missing or ambiguous")
    marker_counts = {i: 0 for i in range(4)}
    for marker in markers:
        marker_counts[int(marker[5] > split_x) + 2 * int(marker[6] > split_y)] += 1

    ordered: dict[int, list[int]] = {}
    names = ("red", "green", "blue")
    for group_index, group in grouped.items():
        marker_count = marker_counts[group_index]
        if marker_count not in (1, 2, 3, 4) or marker_count in ordered:
            raise ValueError("unary frame order is not unique")
        xs = [p[0] for p in group.values()]
        ys = [p[1] for p in group.values()]
        if max(xs) - min(xs) < 30 or max(ys) - min(ys) < 30:
            raise ValueError("frame does not expose both board rows and columns")
        cx_mid = (min(xs) + max(xs)) / 2.0
        cy_mid = (min(ys) + max(ys)) / 2.0
        positions = [int(group[name][0] > cx_mid) + 2 * int(group[name][1] > cy_mid) for name in names]
        if len(set(positions)) != 3:
            raise ValueError("two bead identities occupy one site")
        ordered[marker_count] = positions
    frames = [ordered[i] for i in range(1, 5)]
    home = frames[0]
    for step in range(1, 4):
        matches = [i for i in range(3) if frames[step][i] == home[i]]
        if matches:
            if len(matches) != 1:
                raise ValueError("ambiguous simultaneous homecoming")
            if step != 3:
                raise ValueError("unexpected early homecoming")
            return ("red", "green", "blue")[matches[0]]
    raise ValueError("no homecoming visible")
