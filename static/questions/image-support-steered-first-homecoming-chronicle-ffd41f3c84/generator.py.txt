"""Deterministic renderer and analytic arm for First Homecoming Chronicle."""

from __future__ import annotations

import itertools
import random
from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 560
BACKGROUND = (255, 255, 255)
PANEL = (238, 242, 247)
INK = (37, 47, 58)
ORDER = (236, 178, 52)
PALETTE = [(211, 64, 73), (42, 139, 91), (47, 104, 190)]
NAMES = ("red", "green", "blue")
PANEL_ORIGINS = ((145, 120), (385, 120), (145, 320), (385, 320))
PANEL_SIZE = 140
SITE_OFFSETS = ((40, 38), (100, 38), (40, 102), (100, 102))


def _valid_positions(home: list[int], avoid_home: bool, winner: int | None = None) -> list[tuple[int, int, int]]:
    out = []
    for positions in itertools.permutations(range(4), 3):
        matches = [i for i in range(3) if positions[i] == home[i]]
        if avoid_home and matches:
            continue
        if winner is not None and matches != [winner]:
            continue
        out.append(positions)
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    home = list(rng.sample(range(4), 3))
    away = _valid_positions(home, True)
    final = _valid_positions(home, False, winner)
    frame2 = list(rng.choice(away))
    frame3_options = [p for p in away if list(p) != frame2]
    frame3 = list(rng.choice(frame3_options))
    frame4 = list(rng.choice(final))
    order = list(range(4))
    rng.shuffle(order)
    palette = [list(rgb) for rgb in PALETTE]
    return {
        "frames": [home, frame2, frame3, frame4],
        "panel_order": order,
        "palette": palette,
        "width": WIDTH,
        "height": HEIGHT,
    }


def analytic_gold(scene: dict) -> str:
    frames = scene["frames"]
    home = frames[0]
    for step in range(1, len(frames)):
        matches = [i for i in range(3) if frames[step][i] == home[i]]
        if matches:
            if len(matches) != 1:
                raise ValueError("return event is not unique")
            return NAMES[matches[0]]
    raise ValueError("no bead returns home")


def margin(scene: dict) -> float:
    frames = scene["frames"]
    home = frames[0]
    counts = [sum(frames[t][i] == home[i] for t in range(1, 4)) for i in range(3)]
    ranked = sorted(counts, reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    frames = scene["frames"]
    if not isinstance(frames, list) or len(frames) != 4:
        return True
    if any(len(f) != 3 or len(set(f)) != 3 or any(p not in range(4) for p in f) for f in frames):
        return True
    home = frames[0]
    returns = [[i for i in range(3) if frames[t][i] == home[i]] for t in range(1, 4)]
    return bool(returns[0] or returns[1] or len(returns[2]) != 1)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    frames = scene["frames"]
    panel_order = scene["panel_order"]
    palette = [tuple(c) for c in scene["palette"]]
    image = Image.new("RGB", (int(scene.get("width", WIDTH)), int(scene.get("height", HEIGHT))), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot, frame_index in enumerate(panel_order):
        ox, oy = PANEL_ORIGINS[slot]
        draw.rounded_rectangle((ox, oy, ox + PANEL_SIZE, oy + PANEL_SIZE), radius=12, fill=PANEL, outline=INK, width=4)
        # Visible unary frame number: one through four separated gold center dots.
        for j in range(frame_index + 1):
            cx, cy = ox + 52 + 12 * j, oy + 70
            draw.ellipse((cx - 4, cy - 4, cx + 4, cy + 4), fill=ORDER, outline=INK, width=1)
        for sx, sy in SITE_OFFSETS:
            draw.ellipse((ox + sx - 18, oy + sy - 18, ox + sx + 18, oy + sy + 18), fill=(255, 255, 255), outline=(151, 162, 174), width=2)
        for bead, site in enumerate(frames[frame_index]):
            sx, sy = SITE_OFFSETS[site]
            draw.ellipse((ox + sx - 12, oy + sy - 12, ox + sx + 12, oy + sy + 12), fill=palette[bead], outline=INK, width=2)
    return image
