"""Spiral-chain world: a hub disc and a chain of equal dots winding outward.

Latent state z = canvas, hub centre and radius, dot radius, the chain's start
radius, radial step, start angle, signed angular step, and the dot list. The
handedness of the chain is the sign of the angular step read in image
coordinates (y down), i.e. positive turns are clockwise on screen.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "spiral_chain_handedness-0.1.0"

CLASS_NAMES = ("clockwise", "counter-clockwise")

WIDTH = 512
HEIGHT = 512
BG = (255, 255, 255)
INK = (20, 20, 20)

HUB_RADIUS = 16.0
DOT_RADIUS = 6.0

MIN_STEP_DEG = 24.0
MAX_STEP_DEG = 44.0
MIN_RADIAL_STEP = 22.0
MAX_RADIAL_STEP = 30.0
MAX_OUTER_RADIUS = 205.0
EDGE_PAD = 10.0

# Quarantine: below this clearance a finite-resolution reader can no longer tell
# the turning direction apart from a straight radial ray, or tell which of two
# dots is the outer one. The sampler floor is strictly larger.
QUARANTINE_CLEARANCE = 15.0
SAMPLING_FLOOR = 22.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(f"spiral_chain_handedness|{seed}")
    # Handedness is stratified on the seed so any run of consecutive scene seeds
    # is exactly class balanced; every other latent is sampled independently.
    sign = 1 if seed % 2 == 0 else -1
    for _ in range(10_000):
        count = rng.randint(6, 8)
        step_deg = rng.uniform(MIN_STEP_DEG, MAX_STEP_DEG)
        radial_step = rng.uniform(MIN_RADIAL_STEP, MAX_RADIAL_STEP)
        start_radius = rng.uniform(46.0, 60.0)
        start_deg = rng.uniform(0.0, 360.0)
        hub_x = 256.0 + rng.uniform(-26.0, 26.0)
        hub_y = 256.0 + rng.uniform(-26.0, 26.0)
        if start_radius + (count - 1) * radial_step > MAX_OUTER_RADIUS:
            continue
        dots = []
        for k in range(count):
            radius = start_radius + k * radial_step
            angle = math.radians(start_deg + sign * step_deg * k)
            dots.append(
                {
                    "x": hub_x + radius * math.cos(angle),
                    "y": hub_y + radius * math.sin(angle),
                }
            )
        if any(
            d["x"] < DOT_RADIUS + EDGE_PAD
            or d["x"] > WIDTH - DOT_RADIUS - EDGE_PAD
            or d["y"] < DOT_RADIUS + EDGE_PAD
            or d["y"] > HEIGHT - DOT_RADIUS - EDGE_PAD
            for d in dots
        ):
            continue
        scene = {
            "width": WIDTH,
            "height": HEIGHT,
            "hub": {"x": hub_x, "y": hub_y, "r": HUB_RADIUS},
            "dot_radius": DOT_RADIUS,
            "dots": dots,
        }
        if margin(scene) < SAMPLING_FLOOR:
            continue
        return scene
    raise RuntimeError(f"could not sample a spiral chain for seed {seed}")


def _ordered(scene: dict) -> list[dict]:
    """Dots ordered outward from the hub. The stored list order is never used."""
    hub = scene["hub"]
    return sorted(dots_of(scene), key=lambda d: math.hypot(d["x"] - hub["x"], d["y"] - hub["y"]))


def dots_of(scene: dict) -> list[dict]:
    return [dict(d) for d in scene["dots"]]


def _turns(scene: dict) -> list[float]:
    """Signed turn in degrees between consecutive dots read outward, wrapped."""
    hub = scene["hub"]
    angles = [
        math.degrees(math.atan2(d["y"] - hub["y"], d["x"] - hub["x"])) for d in _ordered(scene)
    ]
    turns = []
    for a, b in zip(angles, angles[1:]):
        turns.append((b - a + 180.0) % 360.0 - 180.0)
    return turns


def analytic_gold(scene: dict) -> str:
    total = sum(_turns(scene))
    return "clockwise" if total > 0 else "counter-clockwise"


def margin(scene: dict) -> float:
    """Smallest legibility clearance: the smallest turn magnitude in degrees and
    the smallest outward radial gap in pixels, whichever is smaller."""
    hub = scene["hub"]
    radii = [math.hypot(d["x"] - hub["x"], d["y"] - hub["y"]) for d in _ordered(scene)]
    gaps = [b - a for a, b in zip(radii, radii[1:])]
    turns = [abs(t) for t in _turns(scene)]
    return float(min(min(turns), min(gaps)))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical relabelings of the latent scene.

    The dots are painted identically and independently, so the stored dot list
    is an unordered set: reversing it or cyclically shifting it repaints the
    same raster. Both arms recover the chain order from the geometry (distance
    from the hub), never from the list order, so the gold is unchanged.
    """
    dots = dots_of(scene)
    reversed_scene = dict(scene)
    reversed_scene["dots"] = list(reversed(dots))
    shifted_scene = dict(scene)
    shifted_scene["dots"] = dots[3:] + dots[:3]
    return [("dot_list_reversal", reversed_scene), ("dot_list_cyclic_shift", shifted_scene)]


def render(scene: dict) -> Image.Image:
    ss = 4
    img = Image.new("RGB", (scene["width"] * ss, scene["height"] * ss), BG)
    draw = ImageDraw.Draw(img)
    hub = scene["hub"]
    draw.ellipse(
        [
            (hub["x"] - hub["r"]) * ss,
            (hub["y"] - hub["r"]) * ss,
            (hub["x"] + hub["r"]) * ss,
            (hub["y"] + hub["r"]) * ss,
        ],
        fill=INK,
    )
    rad = scene["dot_radius"]
    for dot in scene["dots"]:
        draw.ellipse(
            [
                (dot["x"] - rad) * ss,
                (dot["y"] - rad) * ss,
                (dot["x"] + rad) * ss,
                (dot["y"] + rad) * ss,
            ],
            fill=INK,
        )
    return img.resize((scene["width"], scene["height"]), Image.LANCZOS)
