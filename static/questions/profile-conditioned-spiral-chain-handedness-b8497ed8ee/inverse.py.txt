"""Image-only inverse arm for the spiral-chain world.

Receives the final PNG and the public question contract only. It reconstructs
the constellation structure from pixels: dark ink is segmented, connected
components are labelled, the single largest blob is the hub, the remaining
blobs are the chain dots, and the chain is ordered by distance from the hub.
The declared global evidence is the accumulated signed turn of that ordered
chain; only its sign is reported.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "spiral_chain_handedness-oracle-0.1.0"

INK_LUMA_MAX = 128.0
MIN_BLOB_PIXELS = 12


class OracleReadError(ValueError):
    """The picture does not show a readable hub plus chain."""


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Label 8-connected true regions; returns per-component (y, x) index arrays."""
    height, width = mask.shape
    labels = np.zeros((height, width), dtype=np.int32)
    blobs: list[list[tuple[int, int]]] = []
    for sy, sx in zip(*np.nonzero(mask)):
        if labels[sy, sx]:
            continue
        current = len(blobs) + 1
        labels[sy, sx] = current
        stack = [(int(sy), int(sx))]
        pixels: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            pixels.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not labels[ny, nx]:
                        labels[ny, nx] = current
                        stack.append((ny, nx))
        blobs.append(pixels)
    return [np.asarray(p, dtype=np.float64) for p in blobs if len(p) >= MIN_BLOB_PIXELS]


def _centroid(pixels: np.ndarray) -> tuple[float, float]:
    return float(pixels[:, 1].mean()) + 0.5, float(pixels[:, 0].mean()) + 0.5


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    luma = arr @ np.array([0.299, 0.587, 0.114], dtype=np.float32)
    blobs = _components(luma <= INK_LUMA_MAX)
    if len(blobs) < 4:
        raise OracleReadError(f"only {len(blobs)} ink blobs found")
    sizes = [len(b) for b in blobs]
    hub_index = int(np.argmax(sizes))
    rest = sorted(sizes)[:-1]
    if sizes[hub_index] <= 1.5 * max(rest):
        raise OracleReadError("no single blob is clearly the largest")
    hub_x, hub_y = _centroid(blobs[hub_index])

    dots = []
    for index, blob in enumerate(blobs):
        if index == hub_index:
            continue
        cx, cy = _centroid(blob)
        dots.append((math.hypot(cx - hub_x, cy - hub_y), cx, cy))
    dots.sort()
    angles = [math.degrees(math.atan2(cy - hub_y, cx - hub_x)) for _, cx, cy in dots]
    total = 0.0
    for a, b in zip(angles, angles[1:]):
        total += (b - a + 180.0) % 360.0 - 180.0
    if total == 0.0:
        raise OracleReadError("chain accumulates no net turn")
    return "clockwise" if total > 0 else "counter-clockwise"
