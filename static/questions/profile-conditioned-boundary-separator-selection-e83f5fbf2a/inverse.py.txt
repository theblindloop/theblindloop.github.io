"""Independent raster-only inverse for boundary-separator panels."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = (255, 255, 255)
COLORS = {
    "red": (220, 68, 72),
    "blue": (54, 107, 198),
    "green": (48, 161, 91),
}
PANEL_X = (20, 340, 660)
PANEL_Y = 48
PANEL_W = 280
PANEL_H = 290
INTERIOR_MARGIN = 12


def _white_components(mask: np.ndarray) -> int:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    sizes: list[int] = []
    for sy, sx in zip(*np.where(mask)):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue: deque[tuple[int, int]] = deque([(sy, sx)])
        seen[sy, sx] = True
        size = 0
        while queue:
            y, x = queue.popleft()
            size += 1
            for dy, dx in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                ny, nx = y + dy, x + dx
                if (
                    0 <= ny < height
                    and 0 <= nx < width
                    and mask[ny, nx]
                    and not seen[ny, nx]
                ):
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        sizes.append(size)
    # Thick Pillow line joins can leave isolated white speckles at a bend.
    # They are below the observable topology scale and must not become extra
    # regions; the deliberately sampled gaps are thousands of pixels.
    return sum(size >= 64 for size in sizes)


def _inner_crop(arr: np.ndarray, slot: int) -> np.ndarray:
    left = PANEL_X[slot]
    top = PANEL_Y
    x0 = left + INTERIOR_MARGIN
    x1 = left + PANEL_W - INTERIOR_MARGIN
    y0 = top + INTERIOR_MARGIN
    y1 = top + PANEL_H - INTERIOR_MARGIN
    return arr[y0:y1, x0:x1]


def decision_from_image(image: Image.Image) -> str:
    """Recover the wall color whose white complement has two components."""

    if image.size != (960, 380):
        raise ValueError(f"unexpected raster size {image.size}")
    pixels = np.asarray(image.convert("RGB"), dtype=np.uint8)
    matches: list[str] = []
    for slot in range(3):
        crop = _inner_crop(pixels, slot)
        white = np.all(crop == np.asarray(BACKGROUND, dtype=np.uint8), axis=2)
        components = _white_components(white)
        if components == 2:
            color_hits = []
            for name, rgb in COLORS.items():
                mask = np.all(crop == np.asarray(rgb, dtype=np.uint8), axis=2)
                if int(mask.sum()) >= 100:
                    color_hits.append(name)
            if len(color_hits) != 1:
                raise ValueError(f"separator panel has ambiguous color set {color_hits}")
            matches.append(color_hits[0])
        elif components != 1:
            raise ValueError(f"panel {slot} has ambiguous white-component count {components}")
    if len(matches) != 1:
        raise ValueError(f"expected one separator panel, recovered {matches}")
    return matches[0]
