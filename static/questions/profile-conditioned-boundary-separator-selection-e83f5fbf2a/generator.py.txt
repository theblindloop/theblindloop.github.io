"""Deterministic renderer and analytic complement-connectivity label."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "boundary_separator_selection-0.1.0"
WIDTH, HEIGHT = 960, 380
BACKGROUND = (255, 255, 255)
FRAME = (35, 42, 52)
COLORS = {
    "red": (220, 68, 72),
    "blue": (54, 107, 198),
    "green": (48, 161, 91),
}
PANEL_X = (20, 340, 660)
PANEL_Y = 48
PANEL_W = 280
PANEL_H = 290
FRAME_WIDTH = 8
INTERIOR_MARGIN = 12
WALL_WIDTH_MIN = 24
WALL_WIDTH_MAX = 30
MIN_SPLIT_GAP = 34
MIN_FLOAT_CLEARANCE = 30


def _panel_box(slot: int) -> tuple[int, int, int, int]:
    left = PANEL_X[slot]
    top = PANEL_Y
    return left, top, left + PANEL_W - 1, top + PANEL_H - 1


def _inner_box(slot: int) -> tuple[int, int, int, int]:
    left, top, right, bottom = _panel_box(slot)
    return (
        left + INTERIOR_MARGIN,
        top + INTERIOR_MARGIN,
        right - INTERIOR_MARGIN,
        bottom - INTERIOR_MARGIN,
    )


def _path_points(panel: dict[str, Any], slot: int, start: float, end: float) -> list[tuple[int, int]]:
    """Return a gently bent vertical or horizontal centerline subpath."""

    x0, y0, x1, y1 = _inner_box(slot)
    span = float((y1 - y0) if panel["orientation"] == "vertical" else (x1 - x0))
    bend = float(panel["bend"])
    phase = float(panel["phase"])
    points: list[tuple[int, int]] = []
    steps = max(8, int(abs(end - start) / 5.0))
    for index in range(steps + 1):
        distance = start + (end - start) * index / steps
        u = max(0.0, min(1.0, distance / span))
        offset = bend * math.sin(math.pi * u + phase)
        if panel["orientation"] == "vertical":
            x = (x0 + x1) / 2.0 + offset
            y = y0 + distance
        else:
            x = x0 + distance
            y = (y0 + y1) / 2.0 + offset
        points.append((int(round(x)), int(round(y))))
    return points


def _draw_wall(draw: ImageDraw.ImageDraw, panel: dict[str, Any], slot: int) -> None:
    x0, y0, x1, y1 = _inner_box(slot)
    span = float((y1 - y0) if panel["orientation"] == "vertical" else (x1 - x0))
    extension = 9.0
    mode = panel["mode"]
    fill = COLORS[panel["color"]]
    width = int(panel["wall_width"])
    if mode == "through":
        paths = [_path_points(panel, slot, -extension, span + extension)]
    elif mode == "floating":
        clearance = float(panel["clearance"])
        paths = [_path_points(panel, slot, clearance, span - clearance)]
    elif mode == "split":
        gap = float(panel["gap"])
        center = float(panel["gap_position"]) * span
        paths = [
            _path_points(panel, slot, -extension, center - gap / 2.0),
            _path_points(panel, slot, center + gap / 2.0, span + extension),
        ]
    else:
        raise ValueError(f"unknown wall mode {mode!r}")
    for path in paths:
        draw.line(path, fill=fill, width=width, joint="curve")


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one scene with one continuous separator and two non-separators."""

    rng = random.Random(int(seed))
    colors = ["red", "blue", "green"]
    rng.shuffle(colors)
    separator_slot = rng.randrange(3)
    panels: list[dict[str, Any]] = []
    for slot, color in enumerate(colors):
        if slot == separator_slot:
            mode = "through"
        else:
            mode = rng.choice(("floating", "split"))
        panels.append(
            {
                "color": color,
                "mode": mode,
                "orientation": rng.choice(("vertical", "horizontal")),
                "wall_width": rng.randint(WALL_WIDTH_MIN, WALL_WIDTH_MAX),
                "bend": round(rng.uniform(-20.0, 20.0), 3),
                "phase": round(rng.uniform(-0.18, 0.18), 4),
                "gap": rng.randint(36, 48) if mode == "split" else 0,
                "gap_position": round(rng.uniform(0.34, 0.66), 4) if mode == "split" else 0.5,
                "clearance": rng.randint(32, 44) if mode == "floating" else 0,
            }
        )
    return {"panels": panels}


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible scene fields into a flat RGB image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for slot, panel in enumerate(scene["panels"]):
        left, top, right, bottom = _panel_box(slot)
        _draw_wall(draw, panel, slot)
        draw.rectangle(
            (left, top, right, bottom),
            outline=FRAME,
            width=FRAME_WIDTH,
        )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the color of the unique visible boundary-to-boundary wall."""

    matches = [panel["color"] for panel in scene["panels"] if panel["mode"] == "through"]
    if len(matches) != 1:
        raise ValueError(f"expected one separator wall, found {matches}")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    """Return a one-sided raster safety margin for the topology boundary."""

    values: list[float] = []
    for panel in scene["panels"]:
        half_width = float(panel["wall_width"]) / 2.0
        if panel["mode"] == "through":
            values.append(half_width - 1.0)
        elif panel["mode"] == "split":
            values.append(float(panel["gap"]) / 2.0 - 2.0)
        elif panel["mode"] == "floating":
            values.append(float(panel["clearance"]) - 2.0)
        else:
            values.append(-1.0)
    return float(min(values))


def is_quarantined(scene: dict[str, Any]) -> bool:
    """Reject narrow walls, small gaps, and malformed topology labels."""

    if margin(scene) < 8.0:
        return True
    modes = [panel.get("mode") for panel in scene.get("panels", [])]
    if modes.count("through") != 1 or len(modes) != 3:
        return True
    for panel in scene["panels"]:
        if int(panel["wall_width"]) < WALL_WIDTH_MIN:
            return True
        if panel["mode"] == "split" and int(panel["gap"]) < MIN_SPLIT_GAP:
            return True
        if panel["mode"] == "floating" and int(panel["clearance"]) < MIN_FLOAT_CLEARANCE:
            return True
    return False


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent aliases: all panel fields affect visible pixels or label identity."""

    return []
