"""Loop-separation world renderer.

A single closed black elliptical loop sits on a white background. Two small
colored dots (one red, one blue) are drawn on the white surface. A closed loop
splits the plane into an inside (the enclosed white pocket it surrounds) and an
outside (the rest of the white background). The two dots are on OPPOSITE sides
of the loop exactly when one of them lies inside the loop and the other lies
outside it.

Decision : whether exactly one of the two colored dots lies inside the closed
black loop (i.e. the two dots are separated by the loop).

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "center": [cx, cy],
        "loop_rx": float,          # horizontal half-axis of the loop
        "loop_ry": float,          # vertical half-axis of the loop
        "wall": float,             # stroke thickness of the loop band
        "dots": [{"color": "red",  "x": .., "y": ..},
                 {"color": "blue", "x": .., "y": ..}],
        "width": 400,
        "height": 400,
        "background": "white",
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "loop_separation-0.4.0"

BLACK = (20, 20, 20)
RED = (230, 40, 40)
BLUE = (40, 70, 230)
WHITE = (255, 255, 255)

DOT_R = 7.0            # dot radius (pixels)
WALL_MIN = 14.0        # loop stroke thickness floor
QUARANTINE_MIN = 6.0   # below this dot clearance the scene is ambiguous


def _dot_inside(scene: dict, dot: dict) -> bool:
    """Whether a dot lies inside the enclosed pocket of the loop (inner ellipse)."""
    (cx, cy) = (float(scene["center"][0]), float(scene["center"][1]))
    in_rx = float(scene["loop_rx"]) - float(scene["wall"])
    in_ry = float(scene["loop_ry"]) - float(scene["wall"])
    dx = float(dot["x"]) - cx
    dy = float(dot["y"]) - cy
    s_in = math.sqrt((dx * dx) / (in_rx * in_rx) + (dy * dy) / (in_ry * in_ry))
    return s_in < 1.0


def _dot_by_color(scene: dict, color: str):
    for dot in scene["dots"]:
        if dot["color"] == color:
            return dot
    raise KeyError(f"dot color {color!r} not present")


def sample_scene(
    rng: random.Random,
    separated: bool | None = None,
    width: int = 400,
    height: int = 400,
) -> dict:
    """Sample a scene with two dots either separated or not separated by the loop."""
    wall = WALL_MIN + rng.uniform(0.0, 6.0)
    rx = rng.uniform(72.0, 118.0)
    ry = rng.uniform(64.0, 106.0)
    in_rx = rx - wall
    in_ry = ry - wall
    min_ax = min(in_rx, in_ry)

    clear = QUARANTINE_MIN + DOT_R + 2.0
    cx = rng.uniform(rx + clear, width - rx - clear)
    cy = rng.uniform(ry + clear, height - ry - clear)

    if separated is None:
        separated = rng.random() < 0.5
    inner = [rng.random() < 0.5, rng.random() < 0.5]
    if separated:
        inner = [True, False] if rng.random() < 0.5 else [False, True]
    else:
        both_in = rng.random() < 0.5
        inner = [both_in, both_in]
    colors = ["red", "blue"]

    dots: list[dict] = []
    for index in range(2):
        inside = inner[index]
        placed = False
        for _ in range(300):
            angle = rng.uniform(0.0, 2.0 * math.pi)
            if inside:
                max_s = max(0.25, 1.0 - clear / min_ax)
                s = rng.uniform(0.15, max_s)
                x = cx + s * in_rx * math.cos(angle)
                y = cy + s * in_ry * math.sin(angle)
            else:
                min_s = 1.0 + clear / min(rx, ry)
                max_s = 2.0
                s = rng.uniform(min_s, max_s)
                x = cx + s * rx * math.cos(angle)
                y = cy + s * ry * math.sin(angle)
            if clear < x < width - clear and clear < y < height - clear:
                # keep the two dots from overlapping each other
                ok = True
                for other in dots:
                    if math.hypot(x - other["x"], y - other["y"]) < 2.0 * DOT_R + 2.0:
                        ok = False
                        break
                if ok:
                    dots.append({"color": colors[index], "x": x, "y": y})
                    placed = True
                    break
        if not placed:
            dots.append({"color": colors[index], "x": cx, "y": cy + (DOT_R + 4.0) * (1 if index else -1)})
    return {
        "center": [cx, cy],
        "loop_rx": rx,
        "loop_ry": ry,
        "wall": wall,
        "dots": dots,
        "width": width,
        "height": height,
        "background": "white",
    }


def render(scene: dict, supersample: int = 4) -> Image.Image:
    """Render I = R(z). Deterministic pure function of the scene dict."""
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    (cx, cy) = (float(scene["center"][0]), float(scene["center"][1]))
    rx = float(scene["loop_rx"])
    ry = float(scene["loop_ry"])
    wall = float(scene["wall"])
    dots = scene["dots"]

    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    outer = [
        (cx - rx) * ss,
        (cy - ry) * ss,
        (cx + rx) * ss,
        (cy + ry) * ss,
    ]
    draw.ellipse(outer, fill=BLACK)
    inner = [
        (cx - (rx - wall)) * ss,
        (cy - (ry - wall)) * ss,
        (cx + (rx - wall)) * ss,
        (cy + (ry - wall)) * ss,
    ]
    draw.ellipse(inner, fill=background)
    for dot in dots:
        col = BLUE if dot["color"] == "blue" else RED
        box = [
            (dot["x"] - DOT_R) * ss,
            (dot["y"] - DOT_R) * ss,
            (dot["x"] + DOT_R) * ss,
            (dot["y"] + DOT_R) * ss,
        ]
        draw.ellipse(box, fill=col)
    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def analytic_gold(scene: dict) -> str:
    """Whether the red and blue dots lie on opposite sides of the loop."""
    inside_red = _dot_inside(scene, _dot_by_color(scene, "red"))
    inside_blue = _dot_inside(scene, _dot_by_color(scene, "blue"))
    return "yes" if inside_red != inside_blue else "no"


def margin(scene: dict) -> float:
    """Robustness margin: the closest a dot comes to the loop band.

    Continuous and deterministic; small values denote scenes whose separation
    answer is fragile at pixel scale (quarantined by ``is_quarantined``).
    """
    (cx, cy) = (float(scene["center"][0]), float(scene["center"][1]))
    rx = float(scene["loop_rx"])
    ry = float(scene["loop_ry"])
    wall = float(scene["wall"])
    in_rx = rx - wall
    in_ry = ry - wall
    best: float = float("inf")
    for dot in scene["dots"]:
        dx = float(dot["x"]) - cx
        dy = float(dot["y"]) - cy
        s_in = math.sqrt(dx * dx / (in_rx * in_rx) + dy * dy / (in_ry * in_ry))
        if s_in < 1.0:
            clearance = (1.0 - s_in) * min(in_rx, in_ry)
        else:
            s_out = math.sqrt(dx * dx / (rx * rx) + dy * dy / (ry * ry))
            clearance = (s_out - 1.0) * min(rx, ry)
        best = min(best, clearance - DOT_R)
    return float(best)


def is_quarantined(scene: dict) -> bool:
    """Quarantine scenes where a dot sits too close to the loop band."""
    return margin(scene) < QUARANTINE_MIN


def latent_symmetries(scene: dict) -> Iterator[dict]:
    """No nontrivial render symmetry: every field directly changes the picture.

    Swapping the red and blue dot colors is a label-transforming relabel, not a
    pixel-identical transform, and the separation answer is unchanged by it;
    declared not_applicable.
    """
    return iter(())
