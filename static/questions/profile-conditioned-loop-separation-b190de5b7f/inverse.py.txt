"""Independent pixel-only inverse arm for the loop-separation world.

receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking "white-like" pixels (the white background and the loop's inner pocket);
2. flood-filling the 4-connected white region reachable from the image border;
3. locating each colored dot by color and deciding whether it sits in white
   reachable from the border (outside the loop) or in white the border flood
   does NOT reach (inside the enclosed pocket of the loop);

A dot inside the enclosed pocket is surrounded by white sealed off by the loop
band; a dot outside the pocket sits in the border-connected white. The two dots
are separated (answer "yes") exactly when one is inside and the other outside.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "loop_separation-oracle-0.4.0"

_WHITE_THRESHOLD = 240
_COLOR_TOLERANCE = 90.0
_RED = np.array([230.0, 40.0, 40.0])
_BLUE = np.array([40.0, 70.0, 230.0])


def _is_white(arr: np.ndarray, threshold: int = _WHITE_THRESHOLD) -> np.ndarray:
    return arr.min(axis=-1) > threshold


def _border_component_mask(white: np.ndarray) -> np.ndarray:
    """Boolean mask of white pixels 4-connected to any border white pixel."""
    h, w = white.shape
    reached = np.zeros_like(white, dtype=bool)
    stack: list[tuple[int, int]] = []
    for y in (0, h - 1):
        for x in range(w):
            if white[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    for y in range(h):
        for x in (0, w - 1):
            if white[y, x] and not reached[y, x]:
                reached[y, x] = True
                stack.append((x, y))
    while stack:
        x, y = stack.pop()
        for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
            if 0 <= nx < w and 0 <= ny < h and white[ny, nx] and not reached[ny, nx]:
                reached[ny, nx] = True
                stack.append((nx, ny))
    return reached


def _color_mask(arr: np.ndarray, target: np.ndarray, tolerance: float) -> np.ndarray:
    diff = arr.astype(np.float32) - target
    return np.sqrt((diff * diff).sum(axis=-1)) < tolerance


def _dilate(mask: np.ndarray) -> np.ndarray:
    out = mask
    for dy in (-1, 0, 1):
        for dx in (-1, 0, 1):
            if dx == 0 and dy == 0:
                continue
            out = out | np.roll(mask, (dy, dx), axis=(0, 1))
    return out


def _dot_is_inside(arr: np.ndarray, white: np.ndarray, reach: np.ndarray, target: np.ndarray) -> bool:
    """A dot is inside the loop pocket iff none of its adjacent white is border-reachable."""
    mask = _color_mask(arr, target, _COLOR_TOLERANCE)
    if not mask.any():
        return False
    adjacent_white = _dilate(mask) & white
    if adjacent_white.any():
        if reach[adjacent_white].any():
            return False
        return True
    return False


def decision_from_image(image: Image.Image) -> str:
    """Whether the red and blue dots are separated by the loop, from pixels alone."""
    arr = np.asarray(image.convert("RGB")).astype(np.float32)
    white = _is_white(arr)
    reach = _border_component_mask(white)
    inside_red = _dot_is_inside(arr, white, reach, _RED)
    inside_blue = _dot_is_inside(arr, white, reach, _BLUE)
    return "yes" if inside_red != inside_blue else "no"
