"""Independent fixed-layout pixel inverse for conditioned probe/patch search."""

from __future__ import annotations
import numpy as np
from PIL import Image

BACKGROUND=(247,245,238); ONE=(30,55,78); ZERO=(229,232,226); XOR=(126,73,163); AND=(28,126,125)
NOW=(238,163,38); SAMPLE=(35,137,190); PLAIN=(238,239,235)
CHECK_COLORS=((42,139,95),(70,157,104),(32,121,91),(78,143,82))
PATCH_COLORS=((91,77,166),(126,78,157),(76,92,166),(112,63,137))
INPUT_X=(12,203,394,585); GATE_X=(8,264,520); GATE_Y=(166,302); OUTPUT_X=(5,198,391,584); OUTPUT_Y=466
DECISIONS=("pair-A","pair-B","pair-C","pair-D")
GATE_PARENTS=(("I1","I2"),("I2","I3"),("I3","I4"),("G1","G2"),("G2","G3"),("G1","G3"))
OUTPUT_PARENTS=(("G4","G5"),("G4","G6"),("G5","G6"),("G4","G3"))

def _near(pixel,color,tolerance=4):
    return bool(np.max(np.abs(pixel.astype(np.int16)-np.array(color,dtype=np.int16)))<=tolerance)

def _bit(arr,x,y):
    pixel=arr[y,x+10]
    if _near(pixel,ONE): return 1
    if _near(pixel,ZERO): return 0
    return None

def _apply(op,a,b): return int(bool(a)!=bool(b)) if op=="XOR" else int(bool(a) and bool(b))

def _evaluate(inputs,operations,overrides):
    values={f"I{i+1}":int(v) for i,v in enumerate(inputs)}; gates=[]; sockets=[]
    for i,(left,right) in enumerate(GATE_PARENTS):
        raw_a,raw_b=values[left],values[right]; sockets.extend((raw_a,raw_b))
        a=overrides.get(2*i,raw_a); b=overrides.get(2*i+1,raw_b)
        value=_apply(operations[i],a,b); gates.append(value); values[f"G{i+1}"]=value
    outputs=[_apply(operations[6+i],values[a],values[b]) for i,(a,b) in enumerate(OUTPUT_PARENTS)]
    return gates,outputs,sockets

def decision_from_image(image:Image.Image)->str:
    arr=np.asarray(image.convert("RGB"),dtype=np.uint8)
    if arr.shape!=(640,768,3) or not _near(arr[630,760],BACKGROUND): return "abstain"
    inputs=[_bit(arr,x+142,102) for x in INPUT_X]; forced=_bit(arr,721,27)
    if forced is None or any(v is None for v in inputs): return "abstain"
    operations=[]; now_hits=[]; sample_hits=[]; check_hits=[[] for _ in range(4)]; patch_hits=[[] for _ in range(4)]; shown_gates=[]
    for i in range(6):
        x,y=GATE_X[i%3],GATE_Y[i//3]; pixel=arr[y+56,x+72]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        value=_bit(arr,x+220,y+74)
        if value is None: return "abstain"
        shown_gates.append(value)
        for side in range(2):
            sx=x+(12 if side==0 else 140); mark=arr[y+55,sx+5]; slot=2*i+side
            if _near(mark,NOW): now_hits.append(slot)
            elif _near(mark,SAMPLE): sample_hits.append(slot)
            elif _near(mark,PLAIN): pass
            else:
                cm=[j for j,c in enumerate(CHECK_COLORS) if _near(mark,c)]
                pm=[j for j,c in enumerate(PATCH_COLORS) if _near(mark,c)]
                if len(cm)+len(pm)!=1: return "abstain"
                (check_hits if cm else patch_hits)[(cm or pm)[0]].append(slot)
    target=[]
    for x in OUTPUT_X:
        pixel=arr[OUTPUT_Y+57,x+51]
        if _near(pixel,XOR): operations.append("XOR")
        elif _near(pixel,AND): operations.append("AND")
        else: return "abstain"
        value=_bit(arr,x+89,OUTPUT_Y+118)
        if value is None: return "abstain"
        target.append(value)
    if len(now_hits)!=1 or len(sample_hits)!=1 or any(len(h)!=1 for h in check_hits+patch_hits): return "abstain"
    now,sample=now_hits[0],sample_hits[0]; checks=[h[0] for h in check_hits]; patches=[h[0] for h in patch_hits]
    if len({now,sample,*checks,*patches})!=10: return "abstain"
    current_gates,_current,_s=_evaluate(inputs,operations,{now:forced})
    if current_gates!=shown_gates: return "abstain"
    hits=[]
    for i,(check,patch) in enumerate(zip(checks,patches)):
        _ng,_no,natural_sockets=_evaluate(inputs,operations,{})
        flip=1-natural_sockets[check]
        _pg,_po,probe_sockets=_evaluate(inputs,operations,{check:flip})
        carried=probe_sockets[sample]
        _fg,future,_fs=_evaluate(inputs,operations,{patch:carried})
        if future==target: hits.append(i)
    return DECISIONS[hits[0]] if len(hits)==1 else "abstain"
