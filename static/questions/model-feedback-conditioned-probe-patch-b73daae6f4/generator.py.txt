"""Sampler and renderer for candidate-conditioned probe/patch interventions."""

from __future__ import annotations
import random
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

SIZE=(768,640); BACKGROUND=(247,245,238); INK=(35,42,52); CARD=(255,255,252); BORDER=(105,113,124)
ONE=(30,55,78); ZERO=(229,232,226); XOR=(126,73,163); AND=(28,126,125)
NOW=(238,163,38); SAMPLE=(35,137,190); PLAIN=(238,239,235)
CHECK_COLORS=((42,139,95),(70,157,104),(32,121,91),(78,143,82))
PATCH_COLORS=((91,77,166),(126,78,157),(76,92,166),(112,63,137))
INPUT_X=(12,203,394,585); GATE_X=(8,264,520); GATE_Y=(166,302); OUTPUT_X=(5,198,391,584); OUTPUT_Y=466
DECISIONS=("pair-A","pair-B","pair-C","pair-D")
GATE_PARENTS=(("I1","I2"),("I2","I3"),("I3","I4"),("G1","G2"),("G2","G3"),("G1","G3"))
OUTPUT_PARENTS=(("G4","G5"),("G4","G6"),("G5","G6"),("G4","G3"))

def _font(size:int):
    path=Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path),size=size) if path.exists() else ImageFont.load_default()

def _apply(op,a,b):
    if op=="XOR": return int(bool(a)!=bool(b))
    if op=="AND": return int(bool(a) and bool(b))
    raise ValueError(op)

def _evaluate_overrides(inputs,operations,overrides):
    if len(inputs)!=4 or len(operations)!=10: raise ValueError("wrong circuit size")
    values={f"I{i+1}":int(v) for i,v in enumerate(inputs)}; gates=[]; sockets=[]
    for i,(left,right) in enumerate(GATE_PARENTS):
        raw_a,raw_b=values[left],values[right]; sockets.extend((raw_a,raw_b))
        a=overrides.get(2*i,raw_a); b=overrides.get(2*i+1,raw_b)
        value=_apply(operations[i],a,b); gates.append(value); values[f"G{i+1}"]=value
    outputs=[_apply(operations[6+i],values[a],values[b]) for i,(a,b) in enumerate(OUTPUT_PARENTS)]
    return gates,outputs,sockets

def _pair_result(inputs,operations,sample,check,patch):
    _g,_o,natural_sockets=_evaluate_overrides(inputs,operations,{})
    flipped=1-natural_sockets[check]
    _pg,_po,probe_sockets=_evaluate_overrides(inputs,operations,{check:flipped})
    carried=probe_sockets[sample]
    _fg,future,_fs=_evaluate_overrides(inputs,operations,{patch:carried})
    return future,carried,flipped,natural_sockets[sample]

def sample_scene(seed:int)->dict:
    rng=random.Random(int(seed)^0x43484543); wanted=int(seed)%4
    for _ in range(20000):
        inputs=[rng.randrange(2) for _ in range(4)]; operations=[rng.choice(("XOR","AND")) for _ in range(10)]
        forced=rng.randrange(2); now=rng.randrange(6)
        current_gates,_current_outputs,_current_sockets=_evaluate_overrides(inputs,operations,{now:forced})
        sample=rng.choice([s for s in range(6,12) if s!=now]); pool=[s for s in range(12) if s not in (now,sample)]
        rng.shuffle(pool); checks=pool[:4]; patches=pool[4:8]
        results=[]; carried=[]; natural_sample=None
        for check,patch in zip(checks,patches):
            result,bit,_flip,nat=_pair_result(inputs,operations,sample,check,patch)
            results.append(result); carried.append(bit); natural_sample=nat
        target=results[wanted]
        if len(set(target))<2 or sum(result==target for result in results)!=1: continue
        # The winning probe must change SAMPLE; omitting it must not solve the scene.
        if carried[wanted]==natural_sample: continue
        no_probe=_evaluate_overrides(inputs,operations,{patches[wanted]:natural_sample})[1]
        _n,_o,natural_sockets=_evaluate_overrides(inputs,operations,{})
        flip=1-natural_sockets[checks[wanted]]
        accumulated=_evaluate_overrides(inputs,operations,{checks[wanted]:flip,patches[wanted]:carried[wanted]})[1]
        if no_probe==target or accumulated==target: continue
        return {"inputs":inputs,"operations":operations,"forced_value":forced,"now_slot":now,
                "sample_slot":sample,"check_slots":checks,"patch_slots":patches,
                "current_gates":current_gates,"target_outputs":target}
    raise RuntimeError("could not sample conditioned probe patch")

def _validate(scene):
    inputs=[int(v) for v in scene["inputs"]]; operations=[str(v) for v in scene["operations"]]
    forced=int(scene["forced_value"]); now=int(scene["now_slot"]); sample=int(scene["sample_slot"])
    checks=[int(v) for v in scene["check_slots"]]; patches=[int(v) for v in scene["patch_slots"]]
    gates=[int(v) for v in scene["current_gates"]]; outputs=[int(v) for v in scene["target_outputs"]]
    if len(inputs)!=4 or len(operations)!=10 or len(gates)!=6 or len(outputs)!=4: raise ValueError("wrong size")
    if any(v not in (0,1) for v in inputs+gates+outputs+[forced]): raise ValueError("nonbinary")
    if any(op not in ("XOR","AND") for op in operations): raise ValueError("bad operation")
    roles=[now,sample,*checks,*patches]
    if len(checks)!=4 or len(patches)!=4 or len(set(roles))!=10 or any(not 0<=s<12 for s in roles): raise ValueError("bad sockets")
    eg,_eo,_=_evaluate_overrides(inputs,operations,{now:forced})
    if gates!=eg: raise ValueError("inconsistent current")
    return inputs,operations,forced,now,sample,checks,patches,gates,outputs

def analytic_gold(scene):
    try:
        inputs,operations,_forced,_now,sample,checks,patches,_gates,current=_validate(scene)
        hits=[]
        for i,(check,patch) in enumerate(zip(checks,patches)):
            if _pair_result(inputs,operations,sample,check,patch)[0]==current: hits.append(i)
        return DECISIONS[hits[0]] if len(hits)==1 else "abstain"
    except (KeyError,TypeError,ValueError,IndexError): return "abstain"

def margin(scene): return 1.0 if analytic_gold(scene) in DECISIONS else 0.0
def is_quarantined(scene): return analytic_gold(scene)=="abstain"
def latent_symmetries(scene): return []

def _centered(draw,xy,text,font,fill):
    box=draw.textbbox((0,0),text,font=font); draw.text((xy[0]-(box[2]-box[0])/2,xy[1]-(box[3]-box[1])/2-box[1]),text,font=font,fill=fill)

def _lamp(draw,center,value,radius=20):
    x,y=center; draw.ellipse((x-radius,y-radius,x+radius,y+radius),fill=ONE if value else ZERO,outline=INK,width=3)
    _centered(draw,center,str(value),_font(max(16,radius+2)),(255,255,255) if value else INK)

def _operation(draw,box,op):
    draw.rounded_rectangle(box,radius=9,fill=XOR if op=="XOR" else AND)
    _centered(draw,((box[0]+box[2])//2,(box[1]+box[3])//2),op,_font(16),(255,255,255))

def render(scene):
    inputs,operations,forced,now,sample,checks,patches,gates,outputs=_validate(scene)
    image=Image.new("RGB",SIZE,BACKGROUND); draw=ImageDraw.Draw(image)
    _centered(draw,(300,22),"DISPLAYED NOW CIRCUIT",_font(21),INK)
    draw.text((600,7),"NOW BIT",font=_font(14),fill=INK); _lamp(draw,(721,27),forced,18)
    _centered(draw,(384,51),"amber NOW   blue SAMPLE   green CHECK / violet PATCH (A-D)",_font(14),INK)
    for i,(x,value) in enumerate(zip(INPUT_X,inputs)):
        draw.rounded_rectangle((x,66,x+171,137),radius=12,fill=CARD,outline=BORDER,width=3)
        draw.text((x+16,86),f"INPUT I{i+1}",font=_font(19),fill=INK); _lamp(draw,(x+142,102),value)
    for i,((left,right),value) in enumerate(zip(GATE_PARENTS,gates)):
        x,y=GATE_X[i%3],GATE_Y[i//3]; draw.rectangle((x,y,x+240,y+115),fill=CARD,outline=BORDER,width=3)
        draw.text((x+10,y+10),f"G{i+1}",font=_font(19),fill=INK)
        for side,label in enumerate((left,right)):
            slot=2*i+side; sx=x+(12 if side==0 else 140); role=""; color=PLAIN
            if slot==now: color=NOW; role="NOW"
            elif slot==sample: color=SAMPLE; role="SAMPLE"
            elif slot in checks:
                j=checks.index(slot); color=CHECK_COLORS[j]; role=f"CHECK {chr(65+j)}"
            elif slot in patches:
                j=patches.index(slot); color=PATCH_COLORS[j]; role=f"PATCH {chr(65+j)}"
            draw.rounded_rectangle((sx,y+50,sx+48,y+96),radius=7,fill=color,outline=INK,width=2)
            _centered(draw,(sx+24,y+73),label,_font(17),(255,255,255) if color not in (PLAIN,NOW) else INK)
            if role: _centered(draw,(sx+24,y+42),role,_font(9),color)
        _operation(draw,(x+69,y+53,x+133,y+93),operations[i]); draw.text((x+191,y+66),"=",font=_font(18),fill=INK); _lamp(draw,(x+220,y+74),value,18)
    draw.line((18,438,750,438),fill=BORDER,width=3); _centered(draw,(384,450),"TARGET OUTPUTS",_font(17),INK)
    for i,((left,right),value) in enumerate(zip(OUTPUT_PARENTS,outputs)):
        x,y=OUTPUT_X[i],OUTPUT_Y; draw.rounded_rectangle((x,y,x+179,y+143),radius=12,fill=CARD,outline=BORDER,width=3)
        draw.text((x+12,y+12),f"TARGET {chr(65+i)}",font=_font(18),fill=INK); draw.text((x+10,y+69),left,font=_font(15),fill=INK)
        _operation(draw,(x+48,y+54,x+113,y+94),operations[6+i]); draw.text((x+121,y+69),right,font=_font(15),fill=INK); _lamp(draw,(x+89,y+118),value,18)
    return image
