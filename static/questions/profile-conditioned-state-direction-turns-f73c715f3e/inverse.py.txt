"""Independent pixel-only inverse for state-direction turns.

The inverse reads only RGB samples from the public raster: the two visible keys
map colors to identities and categorical states, and one sample per tile then
reconstructs the six ordered identity trajectories.  It independently counts
consecutive opposite nonzero state differences.
"""

from __future__ import annotations

from PIL import Image


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
CANVAS = (1700, 1060)
PANEL_LEFTS = (30, 590, 1150, 1150, 590, 30)
PANEL_TOPS = (180, 180, 180, 630, 630, 630)
SLOT_OFFSETS = ((18, 74), (18, 140), (18, 206), (18, 272))
TILE_H = 56
IDENTITY_STRIP_SAMPLE_X = 8
STATE_MARKER_OFFSET_X = 370
STATE_MARKER_OFFSET_Y = TILE_H // 2
IDENTITY_KEY_X = 30
IDENTITY_KEY_Y = 110
IDENTITY_KEY_STEP = 104
STATE_KEY_X = 520
STATE_KEY_Y = 110
STATE_KEY_STEP = 180


def _pixel(frame: Image.Image, xy: tuple[int, int]) -> tuple[int, int, int]:
    value = frame.getpixel(xy)
    if not isinstance(value, tuple) or len(value) != 3:
        raise ValueError("raster is not RGB")
    return tuple(int(channel) for channel in value)


def _read_keys(frame: Image.Image) -> tuple[dict[tuple[int, int, int], int], dict[tuple[int, int, int], int]]:
    identities: dict[tuple[int, int, int], int] = {}
    for label in range(len(LABELS)):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        color = _pixel(frame, (left + IDENTITY_STRIP_SAMPLE_X, IDENTITY_KEY_Y + 16))
        if color not in IDENTITY_PALETTE or color in identities:
            raise ValueError("identity key is unreadable")
        identities[color] = label

    states: dict[tuple[int, int, int], int] = {}
    for state in range(len(STATE_NAMES)):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        color = _pixel(frame, (cx, STATE_KEY_Y + 16))
        if color not in STATE_PALETTE or color in states:
            raise ValueError("state key is unreadable")
        states[color] = state
    return identities, states


def _snapshot(
    frame: Image.Image,
    identities: dict[tuple[int, int, int], int],
    states: dict[tuple[int, int, int], int],
    panel: int,
) -> list[int]:
    values: list[int | None] = [None] * len(LABELS)
    seen: set[int] = set()
    left, top = PANEL_LEFTS[panel], PANEL_TOPS[panel]
    for dx, dy in SLOT_OFFSETS:
        identity_color = _pixel(
            frame,
            (left + dx + IDENTITY_STRIP_SAMPLE_X, top + dy + TILE_H // 2),
        )
        if identity_color not in identities:
            raise ValueError("tile identity evidence is missing")
        label = identities[identity_color]
        if label in seen:
            raise ValueError("panel repeats an identity")
        seen.add(label)
        state_color = _pixel(
            frame,
            (left + dx + STATE_MARKER_OFFSET_X, top + dy + STATE_MARKER_OFFSET_Y),
        )
        if state_color not in states:
            raise ValueError("tile state evidence is missing")
        values[label] = states[state_color]
    if seen != set(range(len(LABELS))) or any(value is None for value in values):
        raise ValueError("panel does not expose all four identities")
    return [int(value) for value in values]


def _direction(before: int, after: int) -> int:
    if after > before:
        return 1
    if after < before:
        return -1
    return 0


def _turns(path: list[int]) -> int:
    directions = [_direction(before, after) for before, after in zip(path, path[1:])]
    return sum(
        1
        for before, after in zip(directions, directions[1:])
        if before != 0 and after != 0 and before == -after
    )


def decision_from_image(image: Image.Image) -> str:
    """Decode the six panels and return the unique identity with most turns."""
    frame = image.convert("RGB")
    if frame.size != CANVAS:
        raise ValueError(f"unexpected raster size {frame.size}")
    identities, states = _read_keys(frame)
    snapshots = [_snapshot(frame, identities, states, panel) for panel in range(6)]
    paths = [[snapshots[panel][label] for panel in range(6)] for label in range(len(LABELS))]
    counts = [_turns(path) for path in paths]
    ranking = sorted(range(len(LABELS)), key=lambda label: (-counts[label], label))
    if counts[ranking[0]] == counts[ranking[1]]:
        raise ValueError("pixel evidence has no unique reversal winner")
    return LABELS[ranking[0]]
