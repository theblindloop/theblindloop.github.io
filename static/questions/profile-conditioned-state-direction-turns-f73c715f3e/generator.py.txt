"""Renderer and analytic side for the state-direction-reversal world.

The image is a serpentine filmstrip of six ordered snapshots.  Each snapshot
contains the same four identity-coded tiles, but row positions are shuffled.
The answer is the identity whose ordered LOW/MID/HIGH trajectory has the most
immediate rise-to-fall or fall-to-rise turns.  A stay between two changes does
not form a turn.
"""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont


LABELS = ("A", "B", "C", "D")
STATE_NAMES = ("LOW", "MID", "HIGH")
CANVAS = (1700, 1060)
BACKGROUND = (245, 248, 251)
INK = (35, 44, 58)
MUTED = (91, 104, 119)
GRID = (187, 197, 208)
CARD = (255, 255, 255)
IDENTITY_PALETTE = (
    (207, 73, 88),
    (38, 126, 180),
    (111, 82, 169),
    (43, 148, 105),
)
STATE_PALETTE = (
    (226, 124, 54),
    (40, 145, 185),
    (157, 89, 163),
)
PANEL_TINTS = (
    (255, 255, 255),
    (247, 251, 254),
    (253, 249, 244),
    (249, 253, 247),
)

# The six panels form a visible left-to-right serpentine path:
# P1 -> P2 -> P3, then down, P4 -> P5 -> P6.
PANEL_LEFTS = (30, 590, 1150, 1150, 590, 30)
PANEL_TOPS = (180, 180, 180, 630, 630, 630)
PANEL_W = 520
PANEL_H = 370
SLOT_OFFSETS = ((18, 74), (18, 140), (18, 206), (18, 272))
TILE_W = 484
TILE_H = 56
IDENTITY_STRIP_W = 42
STATE_MARKER_OFFSET_X = 370
STATE_MARKER_OFFSET_Y = TILE_H // 2

IDENTITY_KEY_X = 30
IDENTITY_KEY_Y = 110
IDENTITY_KEY_STEP = 104
STATE_KEY_X = 520
STATE_KEY_Y = 110
STATE_KEY_STEP = 180
KEY_RADIUS = 16

_FONT_PATHS = (
    "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
    "/usr/share/fonts/dejavu/DejaVuSans.ttf",
)


def _font(size: int):
    for candidate in _FONT_PATHS:
        if Path(candidate).is_file():
            return ImageFont.truetype(candidate, size=size)
    return ImageFont.load_default()


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[float, float], text: str, font, fill):
    box = draw.textbbox((0, 0), text, font=font)
    width = box[2] - box[0]
    height = box[3] - box[1]
    draw.text((xy[0] - width / 2, xy[1] - height / 2 - box[1]), text, font=font, fill=fill)


def _identity_color(scene: dict, label: int) -> tuple[int, int, int]:
    return IDENTITY_PALETTE[int(scene["label_palette_order"][int(label)])]


def _state_color(scene: dict, state: int) -> tuple[int, int, int]:
    return STATE_PALETTE[int(scene["state_palette_order"][int(state)])]


def _draw_state_marker(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    state: int,
    color: tuple[int, int, int],
    radius: int,
) -> None:
    cx, cy = center
    radius = max(4, int(radius))
    if int(state) == 0:
        draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline=INK, width=2)
    elif int(state) == 1:
        draw.rounded_rectangle(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            radius=max(2, radius // 4),
            fill=color,
            outline=INK,
            width=2,
        )
    else:
        draw.polygon(
            ((cx, cy - radius), (cx + radius, cy), (cx, cy + radius), (cx - radius, cy)),
            fill=color,
            outline=INK,
        )


def _draw_identity_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((IDENTITY_KEY_X, IDENTITY_KEY_Y - 32), "IDENTITY KEY", font=small_font, fill=MUTED)
    for label, name in enumerate(LABELS):
        left = IDENTITY_KEY_X + label * IDENTITY_KEY_STEP
        draw.rectangle(
            (left, IDENTITY_KEY_Y, left + 42, IDENTITY_KEY_Y + 32),
            fill=_identity_color(scene, label),
            outline=INK,
            width=1,
        )
        _centered_text(draw, (left + 66, IDENTITY_KEY_Y + 16), name, label_font, INK)


def _draw_state_key(draw: ImageDraw.ImageDraw, scene: dict, label_font, small_font) -> None:
    draw.text((STATE_KEY_X, STATE_KEY_Y - 32), "STATE KEY  (ordered LOW < MID < HIGH)", font=small_font, fill=MUTED)
    for state, name in enumerate(STATE_NAMES):
        cx = STATE_KEY_X + state * STATE_KEY_STEP + 18
        cy = STATE_KEY_Y + 16
        _draw_state_marker(draw, (cx, cy), state, _state_color(scene, state), KEY_RADIUS)
        _centered_text(draw, (cx + 72, cy), name, label_font, INK)


def _draw_tile(
    draw: ImageDraw.ImageDraw,
    scene: dict,
    left: int,
    top: int,
    label: int,
    state: int,
    label_font,
) -> None:
    draw.rounded_rectangle(
        (left, top, left + TILE_W, top + TILE_H),
        radius=10,
        fill=CARD,
        outline=GRID,
        width=2,
    )
    identity_color = _identity_color(scene, label)
    draw.rounded_rectangle(
        (left, top, left + IDENTITY_STRIP_W, top + TILE_H),
        radius=8,
        fill=identity_color,
    )
    draw.rectangle((left + IDENTITY_STRIP_W - 8, top, left + IDENTITY_STRIP_W, top + TILE_H), fill=identity_color)
    _centered_text(
        draw,
        (left + IDENTITY_STRIP_W / 2, top + TILE_H / 2),
        LABELS[label],
        label_font,
        (255, 255, 255),
    )
    _draw_state_marker(
        draw,
        (left + STATE_MARKER_OFFSET_X, top + STATE_MARKER_OFFSET_Y),
        state,
        _state_color(scene, state),
        int(scene["state_marker_radius"]),
    )
    _centered_text(draw, (left + 158, top + TILE_H / 2), "state", label_font, MUTED)


def _arrow(draw: ImageDraw.ImageDraw, start: tuple[int, int], end: tuple[int, int]) -> None:
    draw.line((*start, *end), fill=MUTED, width=4)
    dx, dy = end[0] - start[0], end[1] - start[1]
    length = max(1.0, (dx * dx + dy * dy) ** 0.5)
    ux, uy = dx / length, dy / length
    px, py = -uy, ux
    bx, by = end[0] - ux * 16, end[1] - uy * 16
    draw.polygon(
        ((end[0], end[1]), (bx + px * 8, by + py * 8), (bx - px * 8, by - py * 8)),
        fill=MUTED,
    )


def _draw_order_arrows(draw: ImageDraw.ImageDraw) -> None:
    centers = [
        (PANEL_LEFTS[index] + PANEL_W // 2, PANEL_TOPS[index] + PANEL_H // 2)
        for index in range(6)
    ]
    # Explicit gaps between the panel cards.
    _arrow(draw, (centers[0][0] + PANEL_W // 2 + 8, centers[0][1]), (centers[1][0] - PANEL_W // 2 - 8, centers[1][1]))
    _arrow(draw, (centers[1][0] + PANEL_W // 2 + 8, centers[1][1]), (centers[2][0] - PANEL_W // 2 - 8, centers[2][1]))
    _arrow(draw, (centers[2][0], centers[2][1] + PANEL_H // 2 + 8), (centers[3][0], centers[3][1] - PANEL_H // 2 - 8))
    _arrow(draw, (centers[3][0] - PANEL_W // 2 - 8, centers[3][1]), (centers[4][0] + PANEL_W // 2 + 8, centers[4][1]))
    _arrow(draw, (centers[4][0] - PANEL_W // 2 - 8, centers[4][1]), (centers[5][0] + PANEL_W // 2 + 8, centers[5][1]))


def _direction(before: int, after: int) -> int:
    if int(after) > int(before):
        return 1
    if int(after) < int(before):
        return -1
    return 0


def _reversal_count(path: list[int]) -> int:
    directions = [_direction(before, after) for before, after in zip(path, path[1:])]
    return sum(
        1
        for before, after in zip(directions, directions[1:])
        if before != 0 and after != 0 and before == -after
    )


def _reversal_counts(paths: list[list[int]]) -> list[int]:
    return [_reversal_count(path) for path in paths]


def _winner_and_gap(panel_states: list[list[int]]) -> tuple[int, int]:
    if len(panel_states) != 6 or any(len(panel) != len(LABELS) for panel in panel_states):
        raise ValueError("six ordered panels with four identity states are required")
    paths = [[int(panel_states[panel][label]) for panel in range(6)] for label in range(len(LABELS))]
    counts = _reversal_counts(paths)
    ranking = sorted(range(len(LABELS)), key=lambda label: (-counts[label], label))
    return ranking[0], counts[ranking[0]] - counts[ranking[1]]


def _random_path(rng: random.Random) -> list[int]:
    return [rng.randrange(len(STATE_NAMES)) for _ in range(6)]


def _make_trajectories(target: int, rng: random.Random) -> list[list[int]]:
    """Sample paths until the requested label has a unique reversal winner."""
    best: tuple[int, int, list[list[int]]] | None = None
    for _ in range(60000):
        paths = [_random_path(rng) for _ in LABELS]
        panels = [[paths[label][panel] for label in range(len(LABELS))] for panel in range(6)]
        counts = _reversal_counts(paths)
        ranking = sorted(range(len(LABELS)), key=lambda label: (-counts[label], label))
        gap = counts[ranking[0]] - counts[ranking[1]]
        score = (int(ranking[0] == target), gap, counts[target])
        if best is None or score > (best[0], best[1], _reversal_count(best[2][target])):
            best = (score[0], score[1], paths)
        if ranking[0] == target and gap >= 1:
            return paths
    if best is None:
        raise RuntimeError("could not sample direction-reversal trajectories")
    return best[2]


def sample_scene(seed: int) -> dict:
    """Sample a balanced trajectory winner and independent visible nuisances."""
    logic_rng = random.Random(int(seed) * 1000003 + 17)
    style_rng = random.Random(int(seed) * 1000033 + 31)
    target = logic_rng.randrange(len(LABELS))
    paths = _make_trajectories(target, logic_rng)
    panel_states = [
        [int(paths[label][panel]) for label in range(len(LABELS))]
        for panel in range(6)
    ]

    label_palette_order = list(range(len(LABELS)))
    state_palette_order = list(range(len(STATE_NAMES)))
    style_rng.shuffle(label_palette_order)
    style_rng.shuffle(state_palette_order)
    slot_orders: list[list[int]] = []
    for _ in range(6):
        order = list(range(len(LABELS)))
        style_rng.shuffle(order)
        slot_orders.append(order)
    panel_tints = [style_rng.randrange(len(PANEL_TINTS)) for _ in range(6)]
    return {
        "canvas": list(CANVAS),
        "panel_states": panel_states,
        "label_palette_order": label_palette_order,
        "state_palette_order": state_palette_order,
        "slot_orders": slot_orders,
        "panel_tints": panel_tints,
        "state_marker_radius": 15 + style_rng.randrange(5),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", tuple(scene["canvas"]), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(30)
    section_font = _font(22)
    label_font = _font(19)
    small_font = _font(15)

    draw.text((30, 20), "STATE DIRECTION TURNS", font=title_font, fill=INK)
    draw.text(
        (30, 57),
        "Follow each keyed identity through the filmstrip; a turn is rise↔fall on consecutive changes.",
        font=small_font,
        fill=MUTED,
    )
    _draw_identity_key(draw, scene, label_font, small_font)
    _draw_state_key(draw, scene, label_font, small_font)
    draw.text((1255, 75), "ORDER", font=small_font, fill=MUTED)
    draw.text((1255, 99), "P1  →  P2  →  P3", font=section_font, fill=INK)
    draw.text((1255, 130), "↓  P4  ←  P5  ←  P6", font=section_font, fill=INK)

    for panel in range(6):
        left, top = PANEL_LEFTS[panel], PANEL_TOPS[panel]
        draw.rounded_rectangle(
            (left, top, left + PANEL_W, top + PANEL_H),
            radius=16,
            fill=PANEL_TINTS[int(scene["panel_tints"][panel])],
            outline=INK,
            width=2,
        )
        _centered_text(draw, (left + PANEL_W / 2, top + 28), f"P{panel + 1}", section_font, INK)
        draw.line((left + 22, top + 52, left + PANEL_W - 22, top + 52), fill=GRID, width=1)
        for slot, (dx, dy) in enumerate(SLOT_OFFSETS):
            label = int(scene["slot_orders"][panel][slot])
            state = int(scene["panel_states"][panel][label])
            _draw_tile(draw, scene, left + dx, top + dy, label, state, label_font)
    _draw_order_arrows(draw)
    return image


def analytic_gold(scene: dict) -> str:
    winner, gap = _winner_and_gap(scene["panel_states"])
    if gap <= 0:
        raise ValueError("direction-reversal counts have no unique winner")
    return LABELS[winner]


def margin(scene: dict) -> float:
    _winner, gap = _winner_and_gap(scene["panel_states"])
    return float(gap)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No render-invariant answer-bearing alias is declared."""
    return []
