"""Pixel-only inverse arm for the fork-branch marker-count world.

Reads only the final RGB PNG. It segments the rendered palette (blue network,
orange ring, black waypoints, red/green terminal dots, white background). It
separates the two blue arms at the fork's horizontal level (the upper arm and
the lower arm) and, for each arm, counts the black waypoint dots that lie on it
(connected components of black pixels whose centroid falls in the arm's band).
It returns the colour of the terminal dot that lies on the arm with the greater
black-waypoint count.

Imports only numpy/PIL; it never reads renderer, prompt, generator, verifier or
gold code and it never receives latent scene data.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_PALETTE = {
    "net": (35, 95, 210),
    "ring": (250, 150, 40),
    "red": (225, 40, 40),
    "green": (35, 165, 75),
    "wp": (30, 30, 30),
    "white": (255, 255, 255),
}
_NAMES = ["net", "ring", "red", "green", "wp", "white"]


def _label_pixels(arr: np.ndarray) -> np.ndarray:
    palette = np.array([*_PALETTE.values()], dtype=np.float32)
    dists = np.stack([np.linalg.norm(arr - c, axis=-1) for c in palette], axis=-1)
    return dists.argmin(axis=-1)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return the (ys, xs) index arrays of each 8-connected component of mask."""
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return []
    labels = np.zeros_like(mask, dtype=np.int32)
    seen = np.zeros_like(mask, dtype=bool)
    comps: list[list[tuple[int, int]]] = []
    for (sy, sx) in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        cells: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            cells.append((cy, cx))
            y0, y1 = max(0, cy - 1), min(mask.shape[0] - 1, cy + 1)
            x0, x1 = max(0, cx - 1), min(mask.shape[1] - 1, cx + 1)
            for ny in range(y0, y1 + 1):
                for nx in range(x0, x1 + 1):
                    if mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        comps.append(cells)
    return [np.array(c, dtype=np.float64) for c in comps]


def _terminal_dot_band(arr_mask: np.ndarray, idx: int) -> str | None:
    """Return 'a' (upper) or 'b' (lower) if a terminal dot of the given colour
    is found, else None."""
    ys, xs = np.nonzero(arr_mask == idx)
    if len(ys) == 0:
        return None
    cy = float(np.mean(ys))
    return "a" if cy < 240 else "b"


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label_pixels(arr)
    idx = {name: i for i, name in enumerate(_NAMES)}
    wp_mask = labels == idx["wp"]
    if not wp_mask.any():
        raise ValueError("no black waypoints found")
    comps = _components(wp_mask)
    counts = {"a": 0, "b": 0}
    for comp in comps:
        cy = float(np.mean(comp[:, 0]))
        if cy < 240:
            counts["a"] += 1
        else:
            counts["b"] += 1
    band_a = _terminal_dot_band(labels, idx["red"])
    band_b = _terminal_dot_band(labels, idx["green"])
    if band_a is None or band_b is None:
        raise ValueError("could not locate terminal dots")
    m = "a" if counts["a"] > counts["b"] else "b"
    if band_a == m:
        return "red"
    if band_b == m:
        return "green"
    raise ValueError("ambiguous terminal band assignment")
