"""Fork-branch marker-count world: which coloured terminal is on the MORE-marked branch.

A single connected blue network is drawn. It begins at an orange ring on the
left, runs along a shared trunk to a junction (fork) in the middle, and then
splits into two mutually exclusive branches. Each branch ends in one coloured
terminal dot (red or green). In addition, N_branch black waypoint dots are drawn
ON each branch (pick a distinct horizontal lane of that branch's serpentine).
The question asks which coloured terminal, red or green, is on the branch that
carries MORE black waypoints along it.

Because both branches share the same trunk up to the junction, and the black
waypoints are drawn exclusively on the post-fork arms, the decisive fact is a
per-branch count of black waypoints recovered by tracing the shared trunk to the
fork and then separating the two arms, not by any property of the fortuitous
colour labels (which are assigned at random to the two arms).
"""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "fork_branch_marker_count-0.1.0"

CANVAS = (480, 480)
BG = (255, 255, 255)
COLOR_NET = (35, 95, 210)
COLOR_RING = (250, 150, 40)
COLOR_RED = (225, 40, 40)
COLOR_GREEN = (35, 165, 75)
COLOR_WP = (30, 30, 30)

STROKE = 18
TERM_R = 6
WP_R = 5
PAD = 40
MIN_MARGIN = 2.0
BRANCH_CLEAR = 30.0

TERMINAL_COLORS = ["red", "green"]


def _pts(points: Any) -> list[tuple[float, float]]:
    return [(float(p[0]), float(p[1])) for p in points]


def _path_length(points: list[tuple[float, float]]) -> float:
    return sum(math.dist(a, b) for a, b in zip(points, points[1:]))


def _seg_dist(p, a, b):
    ax, ay = a
    bx, by = b
    px, py = p
    dx, dy = bx - ax, by - ay
    if dx == 0 and dy == 0:
        return math.dist(p, a)
    t = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
    t = max(0.0, min(1.0, t))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _dist_to_polyline(p, pts):
    best = float("inf")
    for a, b in zip(pts, pts[1:]):
        best = min(best, _seg_dist(p, a, b))
    return best


def _branch_upper(rng: random.Random, fx: float) -> list:
    w, h = CANVAS
    x_lo = fx + 8.0
    x_hi = float(w - PAD - 10)
    n_lanes = rng.randint(3, 5)
    y_top = float(PAD + 18)
    y_bot = float(218.0)
    step = (y_bot - y_top) / (n_lanes - 1) if n_lanes > 1 else 0.0
    pts = [(float(fx), 240.0)]
    for lane in range(n_lanes):
        y = y_bot - lane * step
        if lane % 2 == 0:
            pts.append((x_lo, y))
            pts.append((x_hi, y))
        else:
            pts.append((x_hi, y))
            pts.append((x_lo, y))
    return pts


def _branch_lower(rng: random.Random, fx: float) -> list:
    w, h = CANVAS
    x_lo = fx + 8.0
    x_hi = float(w - PAD - 10)
    n_lanes = rng.randint(3, 5)
    y_top = float(262.0)
    y_bot = float(h - PAD - 18)
    step = (y_bot - y_top) / (n_lanes - 1) if n_lanes > 1 else 0.0
    pts = [(float(fx), 240.0)]
    for lane in range(n_lanes):
        y = y_top + lane * step
        if lane % 2 == 0:
            pts.append((x_lo, y))
            pts.append((x_hi, y))
        else:
            pts.append((x_hi, y))
            pts.append((x_lo, y))
    return pts


def _lane_midpoints(pts: list) -> list:
    """Return (x, y) midpoints of the horizontal (even-index) lane segments."""
    mids = []
    for a, b in zip(pts, pts[1:]):
        if abs(a[1] - b[1]) < 1e-6 and (b[0] - a[0]) > 20:
            mx = (a[0] + b[0]) / 2.0
            mids.append((mx, float(a[1])))
    return mids


def _place_waypoints(rng: random.Random, n: int, midpoints: list) -> list:
    """Pick n well-separated waypoint positions from the branch's lane midpoints."""
    if n > len(midpoints):
        return None
    chosen = []
    used = set()
    for _ in range(n):
        picks = [i for i in range(len(midpoints)) if i not in used]
        i = picks[int(rng.uniform(0, len(picks) - 1))] if picks else None
        if i is None:
            return None
        used.add(i)
        chosen.append(midpoints[i])
    return chosen


def _sample_scene(rng: random.Random) -> dict:
    w, h = CANVAS
    for _ in range(80000):
        fx = rng.uniform(150.0, 220.0)
        rx = rng.uniform(55.0, 100.0)
        ring = [float(rx), 240.0]
        fork = [float(fx), 240.0]
        trunk = [ring, fork]
        try:
            up = _branch_upper(rng, fx)
            lo = _branch_lower(rng, fx)
        except Exception:
            continue
        mids_a = _lane_midpoints(up)
        mids_b = _lane_midpoints(lo)
        if len(mids_a) < 2 or len(mids_b) < 2:
            continue
        max_wp = min(len(mids_a), len(mids_b))
        na = rng.randint(1, max_wp)
        nb = rng.randint(1, max_wp)
        if abs(na - nb) < MIN_MARGIN:
            continue
        wa = _place_waypoints(rng, na, mids_a)
        wb = _place_waypoints(rng, nb, mids_b)
        if wa is None or wb is None:
            continue
        more_branch = "a" if na > nb else "b"
        colors = list(TERMINAL_COLORS)
        rng.shuffle(colors)
        color_a, color_b = colors[0], colors[1]
        terminals = [
            {"color": color_a, "pos": [float(up[-1][0]), float(up[-1][1])], "branch": "a"},
            {"color": color_b, "pos": [float(lo[-1][0]), float(lo[-1][1])], "branch": "b"},
        ]
        waypoints = [{"branch": "a", "pos": [float(x), float(y)]} for x, y in wa] + [
            {"branch": "b", "pos": [float(x), float(y)]} for x, y in wb
        ]
        answer = color_a if more_branch == "a" else color_b

        # validity / position checks
        ok = True
        if not (20 <= up[-1][0] <= w - 20 and 20 <= up[-1][1] <= h - 20):
            ok = False
        if not (20 <= lo[-1][0] <= w - 20 and 20 <= lo[-1][1] <= h - 20):
            ok = False
        for wp in waypoints:
            if not (20 <= wp["pos"][0] <= w - 20 and 20 <= wp["pos"][1] <= h - 20):
                ok = False
                break
            pos = (wp["pos"][0], wp["pos"][1])
            other = lo if wp["branch"] == "a" else up
            if min(
                _dist_to_polyline(pos, _pts(other)),
                _dist_to_polyline(pos, _pts(trunk)),
                math.dist(pos, fork),
            ) < BRANCH_CLEAR:
                ok = False
                break
        if not ok:
            continue
        pa = (terminals[0]["pos"][0], terminals[0]["pos"][1])
        pb = (terminals[1]["pos"][0], terminals[1]["pos"][1])
        if min(_dist_to_polyline(pa, _pts(lo)), math.dist(pa, fork)) < BRANCH_CLEAR:
            continue
        if min(_dist_to_polyline(pb, _pts(up)), math.dist(pb, fork)) < BRANCH_CLEAR:
            continue
        for i in range(len(waypoints)):
            for j in range(i + 1, len(waypoints)):
                if math.dist(waypoints[i]["pos"], waypoints[j]["pos"]) < 2 * (WP_R + 3):
                    ok = False
        if not ok:
            continue
        network = [
            {"seg": "trunk", "pts": trunk},
            {"seg": "branch_a", "pts": up},
            {"seg": "branch_b", "pts": lo},
        ]
        scene = {
            "network": network,
            "ring": ring,
            "fork": fork,
            "terminals": terminals,
            "waypoints": waypoints,
            "answer": answer,
        }
        return scene
    raise RuntimeError("could not satisfy fork-branch marker constraints")


def sample_scene(seed: int) -> dict:
    return _sample_scene(random.Random(seed))


def _draw_polyline(draw: ImageDraw.ImageDraw, pts, width):
    for a, b in zip(pts, pts[1:]):
        draw.line([a, b], fill=COLOR_NET, width=width, joint="curve")


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", CANVAS, BG)
    draw = ImageDraw.Draw(img)
    for route in scene["network"]:
        _draw_polyline(draw, _pts(route["pts"]), STROKE)
    term_rgb = {"red": COLOR_RED, "green": COLOR_GREEN}
    rx, ry = float(scene["ring"][0]), float(scene["ring"][1])
    rr = 9
    draw.ellipse([rx - rr, ry - rr, rx + rr, ry + rr], fill=COLOR_RING)
    for wp in scene["waypoints"]:
        wx, wy = float(wp["pos"][0]), float(wp["pos"][1])
        draw.ellipse([wx - WP_R, wy - WP_R, wx + WP_R, wy + WP_R], fill=COLOR_WP)
    for term in scene["terminals"]:
        tx, ty = float(term["pos"][0]), float(term["pos"][1])
        draw.ellipse([tx - TERM_R, ty - TERM_R, tx + TERM_R, ty + TERM_R],
                     fill=term_rgb[term["color"]])
    return img


def _branch_counts(scene: dict) -> tuple[int, int]:
    ca = sum(1 for w in scene["waypoints"] if w["branch"] == "a")
    cb = sum(1 for w in scene["waypoints"] if w["branch"] == "b")
    return ca, cb


def analytic_gold(scene: dict) -> str:
    """Colour of the terminal on the branch with MORE black waypoints."""
    ca, cb = _branch_counts(scene)
    more = "a" if ca > cb else "b"
    for t in scene["terminals"]:
        if t["branch"] == more:
            return t["color"]
    raise RuntimeError("more-marked branch has no terminal")


def margin(scene: dict) -> float:
    ca, cb = _branch_counts(scene)
    return float(abs(ca - cb))


def is_quarantined(scene: dict) -> bool:
    w, h = CANVAS
    if margin(scene) < MIN_MARGIN:
        return True
    branches = {r["seg"]: r for r in scene["network"] if r["seg"] != "trunk"}
    up = _pts(branches["branch_a"]["pts"])
    lo = _pts(branches["branch_b"]["pts"])
    trunk = _pts(scene["network"][0]["pts"])
    fork = (float(scene["fork"][0]), float(scene["fork"][1]))
    for term in scene["terminals"]:
        pos = (float(term["pos"][0]), float(term["pos"][1]))
        if not (20 <= pos[0] <= w - 20 and 20 <= pos[1] <= h - 20):
            return True
        other = up if term["branch"] == "b" else lo
        if min(_dist_to_polyline(pos, other), _dist_to_polyline(pos, trunk),
               math.dist(pos, fork)) < BRANCH_CLEAR:
            return True
    for wp in scene["waypoints"]:
        pos = (float(wp["pos"][0]), float(wp["pos"][1]))
        if not (20 <= pos[0] <= w - 20 and 20 <= pos[1] <= h - 20):
            return True
        other = up if wp["branch"] == "b" else lo
        if min(_dist_to_polyline(pos, other), _dist_to_polyline(pos, trunk),
               math.dist(pos, fork)) < BRANCH_CLEAR:
            return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
