"""Renderer, analytic gold and margin for the corridor-reach reveal world.

A 13 x 33 lattice prints a closed corridor loop around three sealed rooms. Each
room holds one symbol drawn as a void of light cells inside dark fill and opens
onto the bottom corridor through a single door cell. The corridor is broken by
dark plugs, so only one room can actually be walked to from the marked start
cell. The declared transform is that walk; the reached void names the symbol.
"""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

ROWS = 13
COLS = 33
CHAMBER = 9
CHAMBER_COLS = (2, 12, 22)
CHAMBER_ROW = 2
DOOR_ROW = 11
DOOR_LOCAL_COL = 4

SYMBOL_NAMES: tuple[str, ...] = ("comb", "cross", "cup", "ell", "tee")

QUARANTINE_MARGIN = 4.0
MIN_EVIDENCE_CELLS = 6


def _template(name: str) -> frozenset:
    cells: set = set()
    if name == "cross":
        cells |= {(r, 4) for r in range(9)}
        cells |= {(4, c) for c in range(9)}
    elif name == "tee":
        cells |= {(0, c) for c in range(9)}
        cells |= {(r, 4) for r in range(9)}
    elif name == "ell":
        cells |= {(r, 1) for r in range(9)}
        cells |= {(8, c) for c in range(1, 9)}
    elif name == "cup":
        cells |= {(r, 1) for r in range(1, 9)}
        cells |= {(r, 7) for r in range(1, 9)}
        cells |= {(8, c) for c in range(1, 8)}
    elif name == "comb":
        cells |= {(8, c) for c in range(9)}
        for c in (0, 2, 4, 6):
            cells |= {(r, c) for r in range(2, 9)}
    else:
        raise ValueError(f"unknown symbol {name!r}")
    return frozenset(cells)


TEMPLATES: dict = {name: _template(name) for name in SYMBOL_NAMES}


def _loop_cycle() -> list:
    cycle = [(0, c) for c in range(COLS)]
    cycle += [(r, COLS - 1) for r in range(1, ROWS)]
    cycle += [(ROWS - 1, c) for c in range(COLS - 2, -1, -1)]
    cycle += [(r, 0) for r in range(ROWS - 2, 0, -1)]
    return cycle


LOOP_CYCLE = _loop_cycle()
LOOP_SET = frozenset(LOOP_CYCLE)


def _door_col(chamber: int) -> int:
    return CHAMBER_COLS[chamber] + DOOR_LOCAL_COL


def _arc(cycle_len: int, start: int, stop: int) -> list:
    """Indices strictly between two cut points, walking forward from start."""
    out = []
    index = (start + 1) % cycle_len
    while index != stop:
        out.append(index)
        index = (index + 1) % cycle_len
    return out


def _component(cells: set, start: tuple) -> set:
    if start not in cells:
        return set()
    seen = {start}
    stack = [start]
    while stack:
        r, c = stack.pop()
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            cell = (r + dr, c + dc)
            if cell in cells and cell not in seen:
                seen.add(cell)
                stack.append(cell)
    return seen


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    names = rng.sample(list(SYMBOL_NAMES), 3)
    target = rng.randrange(3)

    damage_choices = (0, 0, 1, 1, 2, 2, 3, 4, 5, 6)
    shapes = []
    for name in names:
        template = TEMPLATES[name]
        removable = sorted(template - {(8, DOOR_LOCAL_COL)})
        cells = set(template)
        for _ in range(40):
            drop = min(rng.choice(damage_choices), len(removable))
            trial = set(template) - set(rng.sample(removable, drop))
            # The door-connected part is the only evidence the walk can expose,
            # so damage that would starve it below a readable size is redeemed.
            if len(_component(trial, (8, DOOR_LOCAL_COL))) >= 7:
                cells = trial
                break
        shapes.append(sorted(cells))

    cycle_len = len(LOOP_CYCLE)
    index_of = {cell: i for i, cell in enumerate(LOOP_CYCLE)}
    target_foot = index_of[(ROWS - 1, _door_col(target))]

    scene = None
    for _ in range(400):
        cut_a, cut_b = rng.sample(range(cycle_len), 2)
        near = _arc(cycle_len, cut_a, cut_b)
        far = _arc(cycle_len, cut_b, cut_a)
        if target_foot in far:
            near, far = far, near
        if target_foot not in near or len(near) < 6 or len(far) < 4:
            continue
        plugs = {LOOP_CYCLE[cut_a], LOOP_CYCLE[cut_b]}
        extra = rng.randint(0, 2)
        for index in rng.sample(far, min(extra, len(far))):
            plugs.add(LOOP_CYCLE[index])
        seed_index = rng.choice([i for i in near if LOOP_CYCLE[i] not in plugs])
        seed_cell = LOOP_CYCLE[seed_index]

        doors = []
        for chamber in range(3):
            foot = index_of[(ROWS - 1, _door_col(chamber))]
            if chamber == target:
                doors.append(True)
            elif foot in near:
                doors.append(False)
            else:
                doors.append(rng.random() < 0.75)

        candidate = {
            "rows": ROWS,
            "cols": COLS,
            "cell": 14,
            "pad": 18,
            "chamber_cols": list(CHAMBER_COLS),
            "shapes": [[list(c) for c in shape] for shape in shapes],
            "doors": doors,
            "plugs": sorted([list(p) for p in plugs]),
            "seed_cell": list(seed_cell),
            "seed_tag": rng.randrange(10_000),
            "palette": {
                "bg": [246, 246, 242],
                "ink": [35, 35, 40],
                "light": [255, 255, 255],
                "grid": [203, 203, 209],
                "fiducial": [0, 90, 220],
                "room": [0, 150, 60],
                "start": [220, 30, 30],
            },
        }
        reached = _reached_by_chamber(candidate)
        live = [k for k, cells in enumerate(reached) if cells]
        if live == [target] and len(reached[target]) >= 3:
            scene = candidate
            break
    if scene is None:  # pragma: no cover - the constructive sampler always succeeds
        raise RuntimeError(f"scene sampling failed for seed {seed}")
    return scene


def _light_cells(scene: dict) -> set:
    plugs = {tuple(p) for p in scene["plugs"]}
    light = {cell for cell in LOOP_SET if cell not in plugs}
    for chamber, shape in enumerate(scene["shapes"]):
        base_col = scene["chamber_cols"][chamber]
        for r, c in shape:
            light.add((CHAMBER_ROW + r, base_col + c))
        if scene["doors"][chamber]:
            light.add((DOOR_ROW, base_col + DOOR_LOCAL_COL))
    return light


def _reached_by_chamber(scene: dict) -> list:
    light = _light_cells(scene)
    start = tuple(scene["seed_cell"])
    seen = {start}
    stack = [start]
    while stack:
        r, c = stack.pop()
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            cell = (r + dr, c + dc)
            if cell in light and cell not in seen:
                seen.add(cell)
                stack.append(cell)
    out = []
    for chamber in range(3):
        base_col = scene["chamber_cols"][chamber]
        local = {
            (r - CHAMBER_ROW, c - base_col)
            for (r, c) in seen
            if CHAMBER_ROW <= r < CHAMBER_ROW + CHAMBER
            and base_col <= c < base_col + CHAMBER
        }
        out.append(local)
    return out


def _scored(scene: dict) -> list:
    reached = _reached_by_chamber(scene)
    best = max(range(3), key=lambda k: (len(reached[k]), -k))
    evidence = reached[best]
    scores = sorted(
        (len(evidence ^ TEMPLATES[name]), name) for name in SYMBOL_NAMES
    )
    return [scores, len(evidence)]


def analytic_gold(scene: dict) -> str:
    scores, _ = _scored(scene)
    return scores[0][1]


def margin(scene: dict) -> float:
    scores, _ = _scored(scene)
    return float(scores[1][0] - scores[0][0])


def is_quarantined(scene: dict) -> bool:
    scores, evidence = _scored(scene)
    return bool(
        float(scores[1][0] - scores[0][0]) < QUARANTINE_MARGIN
        or evidence < MIN_EVIDENCE_CELLS
    )


def naive_read_symbol(scene: dict) -> str:
    """Diagnostic: naming the biggest printed void ignores the declared walk."""
    reached = [
        {
            (r, c)
            for (r, c) in ({tuple(x) for x in shape})
        }
        for shape in scene["shapes"]
    ]
    best = max(range(3), key=lambda k: (len(reached[k]), -k))
    scores = sorted((len(reached[best] ^ TEMPLATES[n]), n) for n in SYMBOL_NAMES)
    return scores[0][1]


def latent_symmetries(scene: dict) -> list:
    reversed_plugs = dict(scene)
    reversed_plugs["plugs"] = list(reversed(scene["plugs"]))

    duplicated = dict(scene)
    duplicated["shapes"] = [list(shape) for shape in scene["shapes"]]
    if duplicated["shapes"][0]:
        duplicated["shapes"][0] = duplicated["shapes"][0] + [
            list(duplicated["shapes"][0][0])
        ]

    retagged = dict(scene)
    retagged["seed_tag"] = (scene["seed_tag"] + 4321) % 10_000

    return [
        ("plug_order_reversed", reversed_plugs),
        ("shape_cell_duplicated", duplicated),
        ("sampling_tag_shifted", retagged),
    ]


def render(scene: dict) -> Image.Image:
    cell = int(scene["cell"])
    pad = int(scene["pad"])
    palette = scene["palette"]
    width = COLS * cell + 2 * pad
    height = ROWS * cell + 2 * pad
    image = Image.new("RGB", (width, height), tuple(palette["bg"]))
    draw = ImageDraw.Draw(image)

    light = _light_cells(scene)
    for r in range(ROWS):
        for c in range(COLS):
            x0 = pad + c * cell
            y0 = pad + r * cell
            fill = palette["light"] if (r, c) in light else palette["ink"]
            draw.rectangle([x0, y0, x0 + cell - 1, y0 + cell - 1], fill=tuple(fill))

    grid = tuple(palette["grid"])
    for c in range(COLS + 1):
        x = pad + c * cell
        draw.line([(x, pad), (x, pad + ROWS * cell)], fill=grid, width=1)
    for r in range(ROWS + 1):
        y = pad + r * cell
        draw.line([(pad, y), (pad + COLS * cell, y)], fill=grid, width=1)

    room = tuple(palette["room"])
    for base_col in scene["chamber_cols"]:
        x0 = pad + base_col * cell
        y0 = pad + CHAMBER_ROW * cell
        draw.rectangle(
            [x0 - 1, y0 - 1, x0 + CHAMBER * cell, y0 + CHAMBER * cell],
            outline=room,
            width=2,
        )

    sr, sc = scene["seed_cell"]
    cx = pad + sc * cell + cell // 2
    cy = pad + sr * cell + cell // 2
    draw.ellipse([cx - 4, cy - 4, cx + 4, cy + 4], fill=tuple(palette["start"]))

    fid = tuple(palette["fiducial"])
    for x in (pad, pad + COLS * cell):
        for y in (pad, pad + ROWS * cell):
            draw.rectangle([x - 4, y - 4, x + 4, y + 4], fill=fid)
    return image
