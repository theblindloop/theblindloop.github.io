"""Pixel-only inverse arm for the corridor-reach reveal world.

Receives the final PNG and the public question contract only. It locates the
four blue corner fiducials, rebuilds the declared 13 x 33 cell lattice, reads
each cell as light or dark, finds the red start dot, walks the light cells, and
names the symbol formed by the cells it reaches inside a room.
"""

from __future__ import annotations

from PIL import Image

ROWS = 13
COLS = 33
CHAMBER = 9
CHAMBER_ROW = 2
CHAMBER_COLS = (2, 12, 22)
SYMBOL_NAMES = ("comb", "cross", "cup", "ell", "tee")


def _template(name: str) -> frozenset:
    cells: set = set()
    if name == "cross":
        cells |= {(r, 4) for r in range(9)}
        cells |= {(4, c) for c in range(9)}
    elif name == "tee":
        cells |= {(0, c) for c in range(9)}
        cells |= {(r, 4) for r in range(9)}
    elif name == "ell":
        cells |= {(r, 1) for r in range(9)}
        cells |= {(8, c) for c in range(1, 9)}
    elif name == "cup":
        cells |= {(r, 1) for r in range(1, 9)}
        cells |= {(r, 7) for r in range(1, 9)}
        cells |= {(8, c) for c in range(1, 8)}
    elif name == "comb":
        cells |= {(8, c) for c in range(9)}
        for c in (0, 2, 4, 6):
            cells |= {(r, c) for r in range(2, 9)}
    return frozenset(cells)


_TEMPLATES = {name: _template(name) for name in SYMBOL_NAMES}


def _lattice_box(pixels, width: int, height: int):
    xs: list = []
    ys: list = []
    for y in range(height):
        for x in range(width):
            r, g, b = pixels[x, y]
            if b > 150 and r < 110 and g < 150 and b - r > 70:
                xs.append(x)
                ys.append(y)
    if len(xs) < 4:
        raise ValueError("no corner fiducials found")
    mid_x = (min(xs) + max(xs)) / 2.0
    mid_y = (min(ys) + max(ys)) / 2.0
    # Each fiducial square is centred on a lattice corner: cluster and average.
    groups: dict = {}
    for x, y in zip(xs, ys):
        groups.setdefault((x > mid_x, y > mid_y), []).append((x, y))
    if len(groups) != 4:
        raise ValueError("expected four corner fiducials")
    centres = {
        key: (sum(p[0] for p in pts) / len(pts), sum(p[1] for p in pts) / len(pts))
        for key, pts in groups.items()
    }
    left = (centres[(False, False)][0] + centres[(False, True)][0]) / 2.0
    right = (centres[(True, False)][0] + centres[(True, True)][0]) / 2.0
    top = (centres[(False, False)][1] + centres[(True, False)][1]) / 2.0
    bottom = (centres[(False, True)][1] + centres[(True, True)][1]) / 2.0
    return left, top, right, bottom


def _read_cells(image: Image.Image):
    rgb = image.convert("RGB")
    pixels = rgb.load()
    width, height = rgb.size
    x0, y0, x1, y1 = _lattice_box(pixels, width, height)
    cw = (x1 - x0) / COLS
    ch = (y1 - y0) / ROWS

    light: set = set()
    reds: dict = {}
    for r in range(ROWS):
        for c in range(COLS):
            cx = x0 + (c + 0.5) * cw
            cy = y0 + (r + 0.5) * ch
            total = 0.0
            count = 0
            red = 0
            for dy in (-2, 0, 2):
                for dx in (-2, 0, 2):
                    px = int(round(cx + dx))
                    py = int(round(cy + dy))
                    if not (0 <= px < width and 0 <= py < height):
                        continue
                    pr, pg, pb = pixels[px, py]
                    total += 0.299 * pr + 0.587 * pg + 0.114 * pb
                    count += 1
                    if pr > 150 and pg < 110 and pb < 110:
                        red += 1
            if count == 0:
                continue
            if red >= 3:
                reds[(r, c)] = red
                light.add((r, c))
            elif total / count > 140.0:
                light.add((r, c))
    if not reds:
        raise ValueError("no start dot found")
    start = max(reds, key=lambda cell: (reds[cell], -cell[0], -cell[1]))
    return light, start


def _scores(image: Image.Image):
    light, start = _read_cells(image)
    seen = {start}
    stack = [start]
    while stack:
        r, c = stack.pop()
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            cell = (r + dr, c + dc)
            if cell in light and cell not in seen:
                seen.add(cell)
                stack.append(cell)
    rooms = []
    for base_col in CHAMBER_COLS:
        rooms.append(
            {
                (r - CHAMBER_ROW, c - base_col)
                for (r, c) in seen
                if CHAMBER_ROW <= r < CHAMBER_ROW + CHAMBER
                and base_col <= c < base_col + CHAMBER
            }
        )
    best = max(range(3), key=lambda k: (len(rooms[k]), -k))
    evidence = rooms[best]
    return sorted((len(evidence ^ _TEMPLATES[n]), n) for n in SYMBOL_NAMES), len(evidence)


def decision_from_image(image: Image.Image) -> str:
    scores, _ = _scores(image)
    return scores[0][1]


def margin_from_image(image: Image.Image) -> float:
    scores, _ = _scores(image)
    return float(scores[1][0] - scores[0][0])
