"""Independent RGB-only inverse for the notched-tile correspondence board.

The inverse receives only the final raster.  It finds the expected nine local
teal components from public panel geometry, rejects incomplete components, and
infers each notch side from four boundary-strip occupancy measurements before
performing the half-turn code comparison.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

TEAL = np.asarray((43, 125, 143), dtype=np.float32)
CARD = np.asarray((231, 235, 230), dtype=np.float32)
PANEL_CENTERS = {"R": (160, 220), "A": (160, 380), "B": (600, 300)}
SLOTS = ((-45, -30), (0, 30), (45, -30))
TILE_SIZE = 40
OPPOSITE = {"N": "S", "E": "W", "S": "N", "W": "E"}
_DIRECTION_ORDER = ("N", "E", "S", "W")


class OracleFitError(ValueError):
    """The raster does not preserve all nine local marks clearly enough."""


def _teal_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.linalg.norm(array - TEAL, axis=2)
    return distance < 72.0


def _tile_direction(mask: np.ndarray, cx: int, cy: int) -> str:
    half_window = 34
    left = max(0, cx - half_window)
    right = min(mask.shape[1], cx + half_window + 1)
    top = max(0, cy - half_window)
    bottom = min(mask.shape[0], cy + half_window + 1)
    crop = mask[top:bottom, left:right]
    local_y, local_x = cy - top, cx - left
    if not (0 <= local_y < crop.shape[0] and 0 <= local_x < crop.shape[1]) or not crop[local_y, local_x]:
        raise OracleFitError("tile center is missing")

    # The broad crop can contain a neighboring tile.  Keep only the connected
    # component reached from this tile's public center, rather than letting a
    # neighboring component contaminate the four side measurements.
    component = np.zeros_like(crop, dtype=bool)
    stack = [(local_y, local_x)]
    component[local_y, local_x] = True
    while stack:
        point_y, point_x = stack.pop()
        for next_y, next_x in (
            (point_y - 1, point_x),
            (point_y + 1, point_x),
            (point_y, point_x - 1),
            (point_y, point_x + 1),
        ):
            if (
                0 <= next_y < crop.shape[0]
                and 0 <= next_x < crop.shape[1]
                and crop[next_y, next_x]
                and not component[next_y, next_x]
            ):
                component[next_y, next_x] = True
                stack.append((next_y, next_x))

    points = np.argwhere(component)
    if len(points) < 1100:
        raise OracleFitError("a tile has too little visible ink")
    y0, x0 = points.min(axis=0)
    y1, x1 = points.max(axis=0)
    if x1 - x0 + 1 < 36 or y1 - y0 + 1 < 36:
        raise OracleFitError("a tile bounding box is incomplete")
    tile = component[y0 : y1 + 1, x0 : x1 + 1]
    strip = 8
    scores = {
        "N": float(tile[:strip, :].mean()),
        "E": float(tile[:, -strip:].mean()),
        "S": float(tile[-strip:, :].mean()),
        "W": float(tile[:, :strip].mean()),
    }
    ordered = sorted(scores.values())
    if ordered[1] - ordered[0] < 0.12:
        raise OracleFitError(f"notch side is ambiguous: {scores}")
    return min(scores, key=scores.get)


def _panel_code(mask: np.ndarray, name: str) -> tuple[str, ...]:
    cx, cy = PANEL_CENTERS[name]
    observed = []
    for dx, dy in SLOTS:
        observed.append(_tile_direction(mask, cx + dx, cy + dy))
    # Sorting by horizontal center is explicit, and the fixed SLOTS are already
    # in that order.  The y offsets are deliberately irrelevant to the code.
    return tuple(observed)


def _mismatch(reference: tuple[str, ...], candidate: tuple[str, ...]) -> int:
    transformed = tuple(OPPOSITE[value] for value in reversed(candidate))
    return sum(a != b for a, b in zip(reference, transformed, strict=True))


def decision_from_image(image: Image.Image) -> str:
    """Return ``A`` or ``B`` using only the supplied PNG."""

    if image.size != (960, 560):
        raise OracleFitError(f"unexpected image size {image.size}")
    mask = _teal_mask(image)
    codes = {name: _panel_code(mask, name) for name in ("R", "A", "B")}
    error_a = _mismatch(codes["R"], codes["A"])
    error_b = _mismatch(codes["R"], codes["B"])
    if error_a == error_b:
        raise OracleFitError("candidate code comparison is tied")
    return "A" if error_a < error_b else "B"
