"""Renderer and analytic world for half-turned notch-triplet correspondence.

Each panel contains three equal filled tiles.  A rectangular bite from one
edge is the tile's local opening-side attribute.  Candidate records are
addressed by their visible slot, so their latent list order is a tested alias.
"""

from __future__ import annotations

import copy
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

RENDERER_VERSION = "notch-triplet-correspondence-0.1.0"
WIDTH, HEIGHT = 960, 560
PAPER = (248, 246, 240)
INK = (35, 54, 75)
CARD = (231, 235, 230)
TEAL = (43, 125, 143)

PANEL_CENTERS = {"R": (160, 220), "A": (160, 380), "B": (600, 300)}
SLOTS = ((-45, -30), (0, 30), (45, -30))
TILE_SIZE = 40
NOTCH_WIDTH = 14
NOTCH_DEPTH = 14
DIRECTIONS = ("N", "E", "S", "W")
OPPOSITE = {"N": "S", "E": "W", "S": "N", "W": "E"}
QUARANTINE_MARGIN = 2.5


def _font(size: int):
    for name in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/truetype/liberation2/LiberationSans-Regular.ttf",
    ):
        if Path(name).is_file():
            return ImageFont.truetype(name, size=size)
    return ImageFont.load_default()


def _slot_records(scene: dict) -> dict[int, dict]:
    return {int(record["slot"]): record for record in scene["candidate_panels"]}


def _transformed_code(signature: list[str] | tuple[str, ...]) -> tuple[str, ...]:
    return tuple(OPPOSITE[value] for value in reversed(signature))


def _mismatch(reference: tuple[str, ...], candidate: list[str] | tuple[str, ...]) -> int:
    observed = _transformed_code(candidate)
    return sum(left != right for left, right in zip(reference, observed, strict=True))


def sample_scene(seed: int) -> dict:
    """Sample balanced codes with equal candidate component geometry."""

    rng = random.Random(int(seed))
    reference = rng.sample(list(DIRECTIONS), 3)
    matching_raw = [OPPOSITE[value] for value in reversed(reference)]
    if int(seed) % 7 == 0:
        # A finite-resolution near boundary: a transposition leaves one of the
        # three transformed positions correct and two incorrect.
        wrong_raw = list(matching_raw)
        wrong_raw[0], wrong_raw[1] = wrong_raw[1], wrong_raw[0]
    else:
        wrong_raw = matching_raw[1:] + matching_raw[:1]

    match_slot = 1 if int(seed) % 2 else 2
    records = [
        {
            "slot": 1,
            "signature": list(matching_raw if match_slot == 1 else wrong_raw),
        },
        {
            "slot": 2,
            "signature": list(matching_raw if match_slot == 2 else wrong_raw),
        },
    ]
    rng.shuffle(records)
    return {
        "reference_signature": list(reference),
        "candidate_panels": records,
    }


def analytic_gold(scene: dict) -> str:
    """Select the slot whose transformed three-side code has fewer mismatches."""

    reference = tuple(scene["reference_signature"])
    panels = _slot_records(scene)
    error_a = _mismatch(reference, panels[1]["signature"])
    error_b = _mismatch(reference, panels[2]["signature"])
    return "A" if error_a < error_b else "B"


def margin(scene: dict) -> float:
    """Return the runner-up mismatch-count gap."""

    reference = tuple(scene["reference_signature"])
    panels = _slot_records(scene)
    error_a = _mismatch(reference, panels[1]["signature"])
    error_b = _mismatch(reference, panels[2]["signature"])
    return float(abs(error_a - error_b))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def _scaled(value: float, scale: int) -> int:
    return int(round(value * scale))


def _draw_text(draw: ImageDraw.ImageDraw, xy, text: str, font, fill, scale: int) -> None:
    draw.text((_scaled(xy[0], scale), _scaled(xy[1], scale)), text, font=font, fill=fill)


def _draw_panel(draw: ImageDraw.ImageDraw, name: str, label: str, scale: int) -> None:
    cx, cy = PANEL_CENTERS[name]
    draw.rounded_rectangle(
        (_scaled(cx - 96, scale), _scaled(cy - 76, scale), _scaled(cx + 96, scale), _scaled(cy + 76, scale)),
        radius=_scaled(14, scale),
        fill=CARD,
        outline=INK,
        width=_scaled(4, scale),
    )
    font = _font(15 * scale)
    bbox = draw.textbbox((0, 0), label, font=font)
    label_width = bbox[2] - bbox[0]
    _draw_text(draw, (cx - label_width / (2 * scale), cy - 66), label, font, INK, scale)


def _draw_tile(draw: ImageDraw.ImageDraw, cx: float, cy: float, direction: str, scale: int) -> None:
    half = TILE_SIZE / 2.0
    left, top, right, bottom = cx - half, cy - half, cx + half, cy + half
    draw.rectangle(
        (_scaled(left, scale), _scaled(top, scale), _scaled(right, scale), _scaled(bottom, scale)),
        fill=TEAL,
    )
    notch_half = NOTCH_WIDTH / 2.0
    if direction == "N":
        notch = (cx - notch_half, top, cx + notch_half, top + NOTCH_DEPTH)
    elif direction == "S":
        notch = (cx - notch_half, bottom - NOTCH_DEPTH, cx + notch_half, bottom)
    elif direction == "W":
        notch = (left, cy - notch_half, left + NOTCH_DEPTH, cy + notch_half)
    elif direction == "E":
        notch = (right - NOTCH_DEPTH, cy - notch_half, right, cy + notch_half)
    else:
        raise ValueError(f"unknown opening side {direction!r}")
    draw.rectangle(tuple(_scaled(value, scale) for value in notch), fill=CARD)


def render(scene: dict) -> Image.Image:
    """Render one deterministic antialiased RGB board."""

    scale = 3
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), PAPER)
    draw = ImageDraw.Draw(image)
    title_font = _font(25 * scale)
    small_font = _font(15 * scale)
    panel_font = _font(14 * scale)

    _draw_text(draw, (42, 26), "NOTCH-SIDE CORRESPONDENCE", title_font, INK, scale)
    _draw_text(
        draw,
        (42, 64),
        "Rotate A and B by 180 degrees; compare the three opening sides from left to right.",
        small_font,
        INK,
        scale,
    )
    draw.rounded_rectangle(
        (_scaled(42, scale), _scaled(118, scale), _scaled(770, scale), _scaled(494, scale)),
        radius=_scaled(20, scale),
        fill=INK,
    )
    draw.rounded_rectangle(
        (_scaled(50, scale), _scaled(126, scale), _scaled(762, scale), _scaled(486, scale)),
        radius=_scaled(16, scale),
        fill=PAPER,
    )

    _draw_panel(draw, "R", "R  REFERENCE", scale)
    _draw_panel(draw, "A", "A  CANDIDATE", scale)
    _draw_panel(draw, "B", "B  CANDIDATE", scale)

    records = _slot_records(scene)
    reference = scene["reference_signature"]
    for name, signature in (
        ("R", reference),
        ("A", records[1]["signature"]),
        ("B", records[2]["signature"]),
    ):
        cx, cy = PANEL_CENTERS[name]
        for (dx, dy), direction in zip(SLOTS, signature, strict=True):
            _draw_tile(draw, cx + dx, cy + dy, direction, scale)

    # A small legend declares the local attribute without providing an answer
    # cue: every direction has the same tile area and the same teal color.
    _draw_text(draw, (800, 186), "Each tile", panel_font, INK, scale)
    _draw_text(draw, (800, 207), "has one", panel_font, INK, scale)
    _draw_text(draw, (800, 228), "edge opening.", panel_font, INK, scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Exercise slot-addressed candidate-record order as a pixel alias."""

    twin = copy.deepcopy(scene)
    twin["candidate_panels"] = list(reversed(twin["candidate_panels"]))
    return [("candidate_record_order", twin)]
