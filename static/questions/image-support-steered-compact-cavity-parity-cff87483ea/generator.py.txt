"""Compact sealed dark blob matter: decide whether the enclosed-cavity count is
odd (one white hole) or even (two white holes).

Latent scene z describes one THICK, connected, area-like dark blob that is always
fully sealed (a closed boundary), located in a small focal window whose absolute
position is randomized across scenes. The blob encloses either one interior
white cavity (hole) -> decision "odd", or two interior white cavities -> decision
"even". Everything remote from the blob is blank and nondiagnostic.

The decision is recovered purely from the raster by flood-filling the exterior
white and counting the enclosed white regions of the single connected dark blob.
"""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = (512, 512)
W, H = CANVAS

# A small, thick, sealed blob so the whole decisive cue stays within a compact
# local span (focal, one connected area-like component).
BLOB_R_MIN, BLOB_R_MAX = 38.0, 46.0
HOLE_R = 11.0
HOLE_OFFSET = 16.0
EDGE_CLEAR = 26.0  # canvas margin so the blob plus clearances always fit
MIN_HOLE_BOUNDARY_GAP = 10.0
MIN_HOLE_TO_HOLE_GAP = 6.0


def sample_scene(seed: int) -> dict:
    """Deterministically sample one scene dict from a seed (no external state)."""
    rng = random.Random(seed)
    radius = rng.uniform(BLOB_R_MIN, BLOB_R_MAX)
    lo = EDGE_CLEAR + radius
    hi_w = W - EDGE_CLEAR - radius
    hi_h = H - EDGE_CLEAR - radius
    cx = rng.uniform(lo, hi_w)
    cy = rng.uniform(lo, hi_h)
    center = (cx, cy)
    n_holes = 1 if rng.random() < 0.5 else 2
    return {
        "center": center,
        "radius": radius,
        "hole_r": HOLE_R,
        "hole_offset": HOLE_OFFSET,
        "n_holes": n_holes,
    }


def render(scene: dict) -> Image.Image:
    """Render the raster I = R(z). Deterministic, no noise, no text."""
    img = Image.new("RGB", CANVAS, "white")
    d = ImageDraw.Draw(img)
    cx, cy = scene["center"]
    radius = scene["radius"]
    hole_r = scene["hole_r"]
    # Drawn geometry is fully determined by the sampled fields; the gold field
    # n_holes is used here to decide how many interior white cavities to punch.
    # Punching fewer/more cavities changes the visible enclosed-cavity count, so
    # the answer-relevant field is directly painted into the pixels.
    bbox = [cx - radius, cy - radius, cx + radius, cy + radius]
    d.ellipse(bbox, fill="black")
    if scene["n_holes"] == 1:
        d.ellipse([cx - hole_r, cy - hole_r, cx + hole_r, cy + hole_r], fill="white")
    else:
        off = scene["hole_offset"]
        d.ellipse([cx - off - hole_r, cy - hole_r, cx - off + hole_r, cy + hole_r], fill="white")
        d.ellipse([cx + off - hole_r, cy - hole_r, cx + off + hole_r, cy + hole_r], fill="white")
    return img


def analytic_gold(scene: dict) -> str:
    return "odd" if scene["n_holes"] == 1 else "even"


def margin(scene: dict) -> float:
    """Robustness clearance in pixels: the smallest gap protecting the parity
    decision from a small raster perturbation."""
    radius = scene["radius"]
    hole_r = scene["hole_r"]
    if scene["n_holes"] == 1:
        boundary_gap = radius - hole_r
        return float(boundary_gap)
    off = scene["hole_offset"]
    boundary_gap = radius - (off + hole_r)
    hole_gap = 2.0 * off - 2.0 * hole_r
    return float(min(boundary_gap, hole_gap))


def is_quarantined(scene: dict) -> bool:
    """Exclude geometries too close to a parity-changing boundary event."""
    if margin(scene) < MIN_HOLE_BOUNDARY_GAP:
        return True
    if scene["n_holes"] == 2 and (2.0 * scene["hole_offset"] - 2.0 * scene["hole_r"]) < MIN_HOLE_TO_HOLE_GAP:
        return True
    if scene["radius"] < BLOB_R_MIN - 1e-9:
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    """Declared empty: the raster fully determines the gold (not_applicable)."""
    return []
