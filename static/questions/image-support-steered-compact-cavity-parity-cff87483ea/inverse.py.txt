"""Independent pixel-only inverse arm for compact_cavity_parity.

Recovers the odd/even cavity-parity decision from the raster alone: segment the
dark ink, require exactly one connected dark region (the sealed blob), flood-fill
the exterior white from the border, and count the enclosed white regions (holes).
Exactly one enclosed white region -> "odd"; exactly two -> "even"; any other
configuration (no blob, several dark components, or a count not in {1,2}) is
ambiguous and the arm abstains by raising.

This module imports neither the renderer nor any scene data; it reads only pixels.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _mask_dark(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    return (arr[:, :, 0] < 128) & (arr[:, :, 1] < 128) & (arr[:, :, 2] < 128)


def _components(size: tuple[int, int], mask_flat: np.ndarray) -> tuple[int, int]:
    """Return (4-connected component count, labels) over True pixels."""
    h, w = size
    labels = np.zeros(h * w, dtype=np.int32)
    count = 0
    seen = mask_flat.copy()
    for start_idx in np.nonzero(mask_flat)[0]:
        if not seen[start_idx]:
            continue
        count += 1
        seen[start_idx] = False
        labels[start_idx] = count
        frontier = deque([start_idx])
        while frontier:
            idx = frontier.popleft()
            r, c = divmod(idx, w)
            for ni in (
                idx - w if r > 0 else -1,
                idx + w if r < h - 1 else -1,
                idx - 1 if c > 0 else -1,
                idx + 1 if c < w - 1 else -1,
            ):
                if ni >= 0 and seen[ni]:
                    seen[ni] = False
                    labels[ni] = count
                    frontier.append(ni)
    return count, labels


def _flood(size: tuple[int, int], starts, allow, visited) -> int:
    h, w = size
    total = 0
    frontier = deque()
    for r, c in starts:
        idx = r * w + c
        if allow[idx] and not visited[idx]:
            visited[idx] = True
            total += 1
            frontier.append(idx)
    while frontier:
        idx = frontier.popleft()
        r, c = divmod(idx, w)
        for ni in (
            idx - w if r > 0 else -1,
            idx + w if r < h - 1 else -1,
            idx - 1 if c > 0 else -1,
            idx + 1 if c < w - 1 else -1,
        ):
            if ni >= 0 and allow[ni] and not visited[ni]:
                visited[ni] = True
                total += 1
                frontier.append(ni)
    return total


def decision_from_image(image: Image.Image) -> str:
    """Return 'odd'/'even', or abstain (raise) when the image is ambiguous."""
    dark = _mask_dark(image)
    h, w = dark.shape
    dark_flat = dark.reshape(-1)

    n_components, labels = _components((h, w), dark_flat)
    if n_components != 1:
        raise ValueError(f"expected exactly one dark blob, found {n_components}")

    white = ~dark
    white_flat = white.reshape(-1)
    visited = np.zeros(h * w, dtype=bool)
    border_starts = []
    for c in range(w):
        if white[0, c]:
            border_starts.append((0, c))
        if white[h - 1, c]:
            border_starts.append((h - 1, c))
    for r in range(1, h - 1):
        if white[r, 0]:
            border_starts.append((r, 0))
        if white[r, w - 1]:
            border_starts.append((r, w - 1))
    _flood((h, w), border_starts, white_flat, visited)
    interior = white_flat & ~visited
    # Of the interior white pixels, count their 4-connected components (holes).
    interior_components = 0
    unseen = interior.copy()
    for start_idx in np.nonzero(interior)[0]:
        if not unseen[start_idx]:
            continue
        interior_components += 1
        unseen[start_idx] = False
        frontier = deque([start_idx])
        while frontier:
            idx = frontier.popleft()
            r, c = divmod(idx, w)
            for ni in (
                idx - w if r > 0 else -1,
                idx + w if r < h - 1 else -1,
                idx - 1 if c > 0 else -1,
                idx + 1 if c < w - 1 else -1,
            ):
                if ni >= 0 and unseen[ni]:
                    unseen[ni] = False
                    frontier.append(ni)

    if interior_components == 1:
        return "odd"
    if interior_components == 2:
        return "even"
    raise ValueError(f"expected 1 or 2 enclosed cavities, found {interior_components}")
