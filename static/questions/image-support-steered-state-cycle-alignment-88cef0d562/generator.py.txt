"""Renderer and analytic arm for paired cyclic state-history alignment."""

from __future__ import annotations

import itertools
import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


WIDTH = 640
HEIGHT = 640
BACKGROUND = (250, 251, 247)
RING_COLOR = (214, 70, 66)

PALETTES = (
    {
        "ink": (34, 43, 53),
        "panel": (35, 91, 108),
        "grid": (132, 155, 165),
        "panel_fill": (239, 246, 246),
        "faint": (206, 218, 219),
    },
    {
        "ink": (54, 38, 62),
        "panel": (113, 63, 105),
        "grid": (164, 139, 161),
        "panel_fill": (247, 241, 247),
        "faint": (225, 212, 224),
    },
    {
        "ink": (45, 57, 38),
        "panel": (77, 107, 64),
        "grid": (147, 164, 137),
        "panel_fill": (243, 248, 239),
        "faint": (216, 226, 211),
    },
)

LEFT_PANEL = (112, 226, 240, 414)
RIGHT_PANEL = (400, 226, 528, 414)
LEFT_START_X = 130
RIGHT_START_X = 418
STATE_STEP = 18
STATE_Y = 337


def _font(size: int) -> ImageFont.ImageFont:
    try:
        return ImageFont.truetype("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _translate_box(box: tuple[int, int, int, int], dx: int, dy: int) -> tuple[int, int, int, int]:
    return (box[0] + dx, box[1] + dy, box[2] + dx, box[3] + dy)


def _draw_history(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    title: str,
    states: list[int],
    palette: dict[str, tuple[int, int, int]],
    dx: int,
    dy: int,
    stroke: int,
) -> None:
    x0, y0, x1, y1 = _translate_box(box, dx, dy)
    draw.rounded_rectangle(
        (x0, y0, x1, y1),
        radius=12,
        fill=palette["panel_fill"],
        outline=palette["panel"],
        width=max(2, stroke),
    )
    font_title = _font(16)
    font_small = _font(13)
    draw.text((x0 + 15, y0 + 14), title, fill=palette["panel"], font=font_title)
    draw.text((x0 + 15, y0 + 43), "marked start", fill=palette["grid"], font=font_small)

    start_x = LEFT_START_X if title.endswith("A") else RIGHT_START_X
    centers = [(start_x + STATE_STEP * index + dx, STATE_Y + dy) for index in range(6)]
    draw.line(
        (centers[0][0], centers[0][1], centers[-1][0], centers[-1][1]),
        fill=palette["grid"],
        width=2,
    )
    # A small terminal arrow makes the left-to-right order explicit without
    # carrying any answer-bearing state.
    ax, ay = centers[-1]
    draw.polygon(
        [(ax + 8, ay), (ax + 1, ay - 5), (ax + 1, ay + 5)],
        fill=palette["grid"],
    )
    for index, state in enumerate(states):
        cx, cy = centers[index]
        radius = 8
        if int(state):
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                fill=palette["ink"],
            )
        else:
            draw.ellipse(
                (cx - radius, cy - radius, cx + radius, cy + radius),
                fill=palette["panel_fill"],
                outline=palette["ink"],
                width=max(2, stroke),
            )
        draw.text((cx - 4, y0 + 72), str(index + 1), fill=palette["panel"], font=font_small)
    # Both cards visibly identify their first checkpoint, allowing the inverse
    # to recover the shared translation without reading text.
    cx, cy = centers[0]
    draw.ellipse(
        (cx - 11, cy - 11, cx + 11, cy + 11),
        outline=RING_COLOR,
        width=max(2, stroke),
    )


def _draw_distractors(
    draw: ImageDraw.ImageDraw,
    variant: int,
    palette: dict[str, tuple[int, int, int]],
) -> None:
    positions = (
        ((48, 92), (592, 92)),
        ((52, 558), (588, 558)),
        ((76, 140), (564, 140)),
    )
    pair = positions[int(variant) % len(positions)]
    for index, (cx, cy) in enumerate(pair):
        radius = 5 + (int(variant) + index) % 3
        draw.ellipse(
            (cx - radius, cy - radius, cx + radius, cy + radius),
            fill=palette["faint"],
        )


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced same-count histories with deterministic nuisance choices."""

    rng = random.Random(int(seed))
    patterns = [
        tuple(1 if index in on_positions else 0 for index in range(6))
        for on_positions in itertools.combinations(range(6), 3)
    ]
    history_a = list(rng.choice(patterns))
    cyclic = history_a[1:] + history_a[:1]
    wants_yes = (int(seed) % 2) == 0
    if wants_yes:
        history_b = list(cyclic)
    else:
        alternatives = [pattern for pattern in patterns if list(pattern) != cyclic]
        history_b = list(rng.choice(alternatives))
    boundary = int(seed) % 13 == 0
    return {
        "history_a": [int(value) for value in history_a],
        "history_b": [int(value) for value in history_b],
        "translation": [rng.randint(-10, 10), rng.randint(-10, 10)],
        "palette_id": int(rng.randrange(len(PALETTES))),
        "stroke": 1.8 if boundary else float(rng.choice((3.0, 3.5, 4.0))),
        "distractor_variant": int(rng.randrange(3)),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    """Render the two detached histories and answer-independent distractors."""

    palette = PALETTES[int(scene["palette_id"]) % len(PALETTES)]
    dx, dy = (int(scene["translation"][0]), int(scene["translation"][1]))
    stroke = max(1, int(round(float(scene["stroke"]))))
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    _draw_distractors(draw, int(scene["distractor_variant"]), palette)
    _draw_history(
        draw,
        LEFT_PANEL,
        "HISTORY A",
        list(scene["history_a"]),
        palette,
        dx,
        dy,
        stroke,
    )
    _draw_history(
        draw,
        RIGHT_PANEL,
        "HISTORY B",
        list(scene["history_b"]),
        palette,
        dx,
        dy,
        stroke,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the cyclic alignment label from the two visible state histories."""

    history_a = tuple(int(value) for value in scene["history_a"])
    history_b = tuple(int(value) for value in scene["history_b"])
    expected = history_a[1:] + history_a[:1]
    return "YES" if history_b == expected else "NO"


def margin(scene: dict[str, Any]) -> float:
    """Return the one-sided raster clarity margin."""

    return round(min(float(scene["stroke"]) - 1.0, 12.0), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return float(scene["stroke"]) <= 2.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # The candidate declares no latent aliases.  All label-bearing fields are
    # explicit and no pixel-identical transform is claimed.
    return []
