"""Independent pixel-only inverse for paired cyclic state histories."""

from __future__ import annotations

from math import hypot

import numpy as np
from PIL import Image


LEFT_PANEL = (112, 226, 240, 414)
RIGHT_PANEL = (400, 226, 528, 414)
LEFT_START_X = 130
RIGHT_START_X = 418
STATE_STEP = 18
STATE_Y = 337


def _luma(pixel: np.ndarray) -> float:
    return float(0.2126 * pixel[0] + 0.7152 * pixel[1] + 0.0722 * pixel[2])


def _red_mask(array: np.ndarray) -> np.ndarray:
    red = array[:, :, 0].astype(np.int16)
    green = array[:, :, 1].astype(np.int16)
    blue = array[:, :, 2].astype(np.int16)
    return (red > 145) & (red > green + 45) & (red > blue + 40)


def _ring_centers(array: np.ndarray) -> tuple[tuple[float, float], tuple[float, float]]:
    mask = _red_mask(array)
    ys, xs = np.nonzero(mask)
    if len(xs) < 40:
        raise ValueError("both red start rings must be visible")
    left = xs < (LEFT_START_X + RIGHT_START_X) / 2
    right = ~left
    if int(left.sum()) < 18 or int(right.sum()) < 18:
        raise ValueError("one history start ring is absent")
    return (
        (float(xs[left].mean()), float(ys[left].mean())),
        (float(xs[right].mean()), float(ys[right].mean())),
    )


def _dark(array: np.ndarray, x: int, y: int, threshold: float = 205.0) -> bool:
    height, width = array.shape[:2]
    if x < 0 or y < 0 or x >= width or y >= height:
        return False
    return _luma(array[y, x]) < threshold


def _panel_present(array: np.ndarray, box: tuple[int, int, int, int]) -> None:
    x0, y0, x1, y1 = box
    samples: list[bool] = []
    for x in range(x0 + 18, x1 - 17, 10):
        samples.append(_dark(array, x, y0 + 1))
        samples.append(_dark(array, x, y1 - 1))
    for y in range(y0 + 18, y1 - 17, 10):
        samples.append(_dark(array, x0 + 1, y))
        samples.append(_dark(array, x1 - 1, y))
    if sum(samples) < 18:
        raise ValueError("history panel frame is absent or damaged")


def _state_at(array: np.ndarray, cx: int, cy: int) -> int:
    inside = []
    perimeter = []
    for y in range(cy - 11, cy + 12):
        for x in range(cx - 11, cx + 12):
            radius = hypot(x - cx, y - cy)
            if radius <= 6.5:
                inside.append(_dark(array, x, y, 190.0))
            elif 7.0 <= radius <= 10.5:
                perimeter.append(_dark(array, x, y, 190.0))
    inside_dark = sum(inside)
    perimeter_dark = sum(perimeter)
    if inside_dark >= 100 and perimeter_dark >= 8:
        return 1
    if inside_dark <= 75 and perimeter_dark >= 12:
        return 0
    raise ValueError("state glyph is missing or ambiguous")


def _history(array: np.ndarray, start_x: int, dx: int, dy: int) -> list[int]:
    return [_state_at(array, start_x + STATE_STEP * index + dx, STATE_Y + dy) for index in range(6)]


def _shift_box(box: tuple[int, int, int, int], dx: int, dy: int) -> tuple[int, int, int, int]:
    return (box[0] + dx, box[1] + dy, box[2] + dx, box[3] + dy)


def decision_from_image(image: Image.Image) -> str:
    """Recover YES/NO using only the final RGB raster."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    (left_ring, right_ring) = _ring_centers(array)
    dx = int(round(left_ring[0] - LEFT_START_X))
    dy = int(round(left_ring[1] - STATE_Y))
    if abs(dx) > 14 or abs(dy) > 14:
        raise ValueError("translation is outside the declared layout")
    if abs(right_ring[0] - (RIGHT_START_X + dx)) > 5 or abs(right_ring[1] - (STATE_Y + dy)) > 5:
        raise ValueError("history starts are not aligned")
    _panel_present(array, _shift_box(LEFT_PANEL, dx, dy))
    _panel_present(array, _shift_box(RIGHT_PANEL, dx, dy))
    history_a = _history(array, LEFT_START_X, dx, dy)
    history_b = _history(array, RIGHT_START_X, dx, dy)
    expected = history_a[1:] + history_a[:1]
    return "YES" if history_b == expected else "NO"
