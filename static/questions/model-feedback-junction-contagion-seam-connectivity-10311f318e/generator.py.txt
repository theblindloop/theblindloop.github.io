"""Deterministic renderer for junction-contagion connectivity through seams."""

from __future__ import annotations

from collections import deque
import copy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1020, 760
BACKGROUND = (250, 248, 244)
INK = (42, 47, 55)
JOINT = (218, 230, 241)
PANEL = (164, 176, 189)
LABEL = (67, 76, 88)
SAME_FILL = (207, 232, 219)
FLIP_FILL = (242, 215, 221)
SOCKET = (185, 201, 218)
SPARK = (211, 55, 58)
COLORS = {
    "amber": (232, 156, 37),
    "cyan": (31, 163, 190),
    "violet": (132, 79, 188),
}
COLOR_NAMES = tuple(COLORS)
PATCH_BASES = (25, 365, 705)
GRID_X = (18, 130, 242)
GRID_Y = (150, 250, 350, 450, 550, 650)
NROWS = len(GRID_Y)


def _node(patch: int, row: int, col: int) -> str:
    return f"P{patch}r{row}c{col}"


def _decode(node_id: str) -> tuple[int, int, int]:
    return int(node_id[1]), int(node_id[3]), int(node_id[5])


def _edge(a: str, b: str) -> list[str]:
    return sorted((a, b))


def _all_nodes() -> list[str]:
    return [_node(p, r, c) for p in range(3) for r in range(NROWS) for c in range(3)]


def _base_edges() -> tuple[tuple[tuple[str, str], ...], ...]:
    patches: list[list[tuple[str, str]]] = [[], [], []]
    for even in (0, 2, 4):
        odd = even + 1
        patches[0] += [(_node(0, even, 1), _node(0, even, 2)),
                       (_node(0, even, 1), _node(0, odd, 1)),
                       (_node(0, odd, 1), _node(0, odd, 2))]
        patches[2] += [(_node(2, even, 0), _node(2, even, 1)),
                       (_node(2, odd, 0), _node(2, odd, 1)),
                       (_node(2, even, 1), _node(2, odd, 1))]
    for row in range(NROWS):
        patches[1] += [(_node(1, row, 0), _node(1, row, 1)),
                       (_node(1, row, 1), _node(1, row, 2))]
    patches[1] += [(_node(1, row, 1), _node(1, row + 1, 1)) for row in (0, 1, 2, 4)]
    return tuple(tuple(edges) for edges in patches)


BASE_PATCH_EDGES = _base_edges()
BASE_PAIRS = (
    (_node(0, 4, 1), _node(2, 4, 1)),  # survives contagion through rows 4 and 5
    (_node(0, 0, 1), _node(2, 0, 1)),  # loses both routes in infected rows 0 and 1
    (_node(0, 2, 1), _node(2, 2, 1)),  # loses both routes in infected rows 2 and 3
)
BASE_SPARK = _node(1, 0, 1)


def _graph(scene: dict) -> dict[str, list[str]]:
    adjacency = {node: [] for node in _all_nodes()}
    for patch in scene["patches"]:
        for a, b in patch["edges"]:
            adjacency[a].append(b)
            adjacency[b].append(a)
    for seam in scene["seams"]:
        left, mode = int(seam["left"]), str(seam["mode"])
        for row in range(NROWS):
            other = row if mode == "same" else NROWS - 1 - row
            a, b = _node(left, row, 2), _node(left + 1, other, 0)
            adjacency[a].append(b)
            adjacency[b].append(a)
    return adjacency


def _is_socket(node_id: str) -> bool:
    patch, _, col = _decode(node_id)
    return (patch < 2 and col == 2) or (patch > 0 and col == 0)


def _removable(scene: dict) -> set[str]:
    terminals = {node for pair in scene["terminals"].values() for node in pair}
    return {node for node in _all_nodes() if node not in terminals and not _is_socket(node)}


def _infected(scene: dict) -> set[str]:
    adjacency, removable = _graph(scene), _removable(scene)
    spark = str(scene["spark"])
    if spark not in removable:
        return set()
    reached, queue = {spark}, deque([spark])
    while queue:
        here = queue.popleft()
        for nxt in adjacency[here]:
            if nxt in removable and nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    return reached


def _connected(scene: dict, pair: list[str] | tuple[str, str], removed: set[str] | None = None) -> bool:
    adjacency = _graph(scene)
    gone = set() if removed is None else set(removed)
    start, goal = pair
    if start in gone or goal in gone:
        return False
    reached, queue = {start}, deque([start])
    while queue:
        here = queue.popleft()
        for nxt in adjacency[here]:
            if nxt not in gone and nxt not in reached:
                reached.add(nxt)
                queue.append(nxt)
    return goal in reached


def _single_junction_resilient(scene: dict, pair: list[str] | tuple[str, str]) -> bool:
    return _connected(scene, pair) and all(_connected(scene, pair, {node}) for node in _removable(scene))


def _answer(scene: dict) -> str:
    gone = _infected(scene)
    answers = [color for color, pair in scene["terminals"].items() if _connected(scene, pair, gone)]
    if len(answers) != 1:
        raise ValueError(f"expected one post-contagion pair, got {answers}")
    return answers[0]


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    flips = tuple(bool((seed // 6) >> patch & 1) for patch in range(3))

    def transform(node_id: str) -> str:
        patch, row, col = _decode(node_id)
        return _node(patch, NROWS - 1 - row if flips[patch] else row, col)

    patches = []
    for patch, base_edges in enumerate(BASE_PATCH_EDGES):
        edges = sorted(_edge(transform(a), transform(b)) for a, b in base_edges)
        patches.append({"name": chr(65 + patch), "edges": edges})
    seams = []
    for left, base_mode in enumerate(("same", "same")):
        parity = (base_mode == "flip") ^ flips[left] ^ flips[left + 1]
        seams.append({"left": left, "right": left + 1, "mode": "flip" if parity else "same"})
    pairs = [sorted(transform(node) for node in pair) for pair in BASE_PAIRS]
    answer = COLOR_NAMES[seed % 3]
    wrong = [color for color in COLOR_NAMES if color != answer]
    if (seed // 3) % 2:
        wrong.reverse()
    scene = {
        "patches": patches,
        "seams": seams,
        "terminals": {answer: pairs[0], wrong[0]: pairs[1], wrong[1]: pairs[2]},
        "spark": transform(BASE_SPARK),
    }
    if is_quarantined(scene):
        raise RuntimeError(f"constructed invalid scene for seed {seed}")
    return scene


def analytic_gold(scene: dict) -> str:
    return _answer(scene)


def margin(scene: dict) -> float:
    _answer(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        if len(scene["patches"]) != 3 or len(scene["seams"]) != 2:
            return True
        if {p["name"] for p in scene["patches"]} != {"A", "B", "C"}:
            return True
        nodes = set(_all_nodes())
        edges = []
        for patch_index, patch in enumerate(sorted(scene["patches"], key=lambda p: p["name"])):
            for edge in patch["edges"]:
                if len(edge) != 2 or not set(edge) <= nodes:
                    return True
                p1, r1, c1 = _decode(edge[0]); p2, r2, c2 = _decode(edge[1])
                if p1 != patch_index or p2 != patch_index or abs(r1-r2) + abs(c1-c2) != 1:
                    return True
                edges.append(tuple(sorted(edge)))
        if len(edges) != len(set(edges)) or len(edges) != 34:
            return True
        seams = sorted(scene["seams"], key=lambda s: int(s["left"]))
        if [(s["left"], s["right"]) for s in seams] != [(0, 1), (1, 2)]:
            return True
        if any(s["mode"] not in {"same", "flip"} for s in seams):
            return True
        if set(scene["terminals"]) != set(COLOR_NAMES):
            return True
        marked = [node for pair in scene["terminals"].values() for node in pair]
        if len(marked) != 6 or len(set(marked)) != 6 or not set(marked) <= nodes:
            return True
        if scene["spark"] not in _removable(scene):
            return True
        infected = _infected(scene)
        if len(infected) != 4:
            return True
        if not all(_single_junction_resilient(scene, pair) for pair in scene["terminals"].values()):
            return True
        answer = _answer(scene)
        if any(_connected(scene, pair, infected) != (color == answer)
               for color, pair in scene["terminals"].items()):
            return True
        # The survivor must still depend on pale junctions, not on a trivial safe-only route.
        if _connected(scene, scene["terminals"][answer], _removable(scene)):
            return True
        return False
    except Exception:
        return True


def _font(size: int) -> ImageFont.ImageFont:
    for path in ("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
                 "/usr/share/fonts/truetype/liberation2/LiberationSans-Bold.ttf"):
        try:
            return ImageFont.truetype(path, size=size)
        except OSError:
            pass
    return ImageFont.load_default()


def _xy(node_id: str) -> tuple[int, int]:
    patch, row, col = _decode(node_id)
    return PATCH_BASES[patch] + GRID_X[col], GRID_Y[row]


def render(scene: dict) -> Image.Image:
    if is_quarantined(scene):
        raise ValueError("cannot render quarantined scene")
    scale = 2
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    patches = {str(patch["name"]): patch for patch in scene["patches"]}
    for patch_index, base in enumerate(PATCH_BASES):
        draw.rounded_rectangle((base*scale, 95*scale, (base+260)*scale, 705*scale),
                               radius=15*scale, outline=PANEL, width=3*scale)
        draw.text(((base+130)*scale, 62*scale), f"PATCH {chr(65+patch_index)}",
                  fill=LABEL, anchor="mm", font=_font(22*scale))
        for a, b in sorted(_edge(*edge) for edge in patches[chr(65+patch_index)]["edges"]):
            draw.line([tuple(v*scale for v in _xy(a)), tuple(v*scale for v in _xy(b))],
                      fill=INK, width=7*scale)
    terminal_of = {node: color for color, pair in scene["terminals"].items() for node in pair}
    for node_id in sorted(_all_nodes()):
        x, y = _xy(node_id)
        patch, _, col = _decode(node_id)
        radius = 18 if node_id in terminal_of else 11
        fill = COLORS[terminal_of[node_id]] if node_id in terminal_of else (
            SOCKET if _is_socket(node_id) else JOINT)
        if fill == SOCKET:
            draw.rounded_rectangle(((x-radius)*scale, (y-radius)*scale,
                                    (x+radius)*scale, (y+radius)*scale),
                                   radius=4*scale, fill=fill, outline=INK, width=3*scale)
        else:
            draw.ellipse(((x-radius)*scale, (y-radius)*scale,
                          (x+radius)*scale, (y+radius)*scale),
                         fill=fill, outline=INK, width=3*scale)
    sx, sy = _xy(str(scene["spark"]))
    draw.ellipse(((sx-21)*scale, (sy-21)*scale, (sx+21)*scale, (sy+21)*scale),
                 outline=SPARK, width=6*scale)
    for seam in sorted(scene["seams"], key=lambda s: int(s["left"])):
        left, mode = int(seam["left"]), str(seam["mode"])
        cx = 325 + 340 * left
        fill = SAME_FILL if mode == "same" else FLIP_FILL
        draw.rounded_rectangle(((cx-27)*scale, 102*scale, (cx+27)*scale, 698*scale),
                               radius=12*scale, fill=fill, outline=PANEL, width=2*scale)
        draw.rounded_rectangle(((cx-43)*scale, 74*scale, (cx+43)*scale, 124*scale),
                               radius=9*scale, fill=fill, outline=INK, width=2*scale)
        draw.text((cx*scale, 92*scale), mode.upper(), fill=INK, anchor="mm", font=_font(16*scale))
        draw.text((cx*scale, 113*scale), "↑ ↑" if mode == "same" else "↑ ↓",
                  fill=INK, anchor="mm", font=_font(15*scale))
        for row in range(NROWS):
            other = row if mode == "same" else NROWS - 1 - row
            y1, y2 = GRID_Y[row], GRID_Y[other]
            draw.line(((cx-18)*scale, y1*scale, (cx+18)*scale, y2*scale),
                      fill=LABEL, width=3*scale)
            draw.ellipse(((cx-22)*scale, (y1-4)*scale, (cx-14)*scale, (y1+4)*scale), fill=INK)
            draw.ellipse(((cx+14)*scale, (y2-4)*scale, (cx+22)*scale, (y2+4)*scale), fill=INK)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["patches"] = list(reversed(twin["patches"]))
    for patch in twin["patches"]:
        patch["edges"] = list(reversed(patch["edges"]))
    twin["seams"] = list(reversed(twin["seams"]))
    twin["terminals"] = {k: list(reversed(v)) for k, v in reversed(list(twin["terminals"].items()))}
    return [("reverse_record_orders", twin)]
