"""Renderer and analytic gold for the sluice-pour replay world.

One small framed sluice block sits somewhere on a plain canvas.  The block holds
two upright tanks of six cells each, six cells of liquid shared between them,
and three pour instructions stacked underneath.  Each instruction is a thick
arrow (the tank the liquid is poured into) and one to three pips (how many cells
of liquid are asked for).  A pour is saturating: it moves what the arrow asks
for unless the source tank holds less, in which case it moves only what is
there.  The decision is the pair of tank levels after the three pours.

Everything decisive lives inside the one small block; every grey mark elsewhere
is inert decoration.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "sluice_pour_replay-renderer-0.1.0"

CANVAS = 640
CAPACITY = 6
TOTAL = 6
STEPS = 3

BG = (250, 250, 250)
PANEL = (255, 255, 255)
INK = (35, 35, 40)
PIP = (25, 25, 30)
LIQUID = (0, 128, 140)
ARROW = (230, 120, 20)
RULE = (205, 205, 210)
DECOR = (168, 168, 172)

FRAME_PX = 3
PAD_PX = 6
BORDER_PX = 2
TANK_GAP = 14
TANK_ROW_GAP = 8
PIP_LEAD = 10
LEFT = "left"
RIGHT = "right"


# --------------------------------------------------------------------------- #
# replay
# --------------------------------------------------------------------------- #
def _replay(left: int, ops: list) -> tuple[int, int]:
    """Apply saturating pours to (left, right) and return the final levels."""
    if not 0 <= left <= CAPACITY:
        raise ValueError(f"illegal start level {left}")
    right = TOTAL - left
    for op in ops:
        direction, amount = op[0], int(op[1])
        if direction not in (LEFT, RIGHT) or not 1 <= amount <= 3:
            raise ValueError(f"illegal instruction {op!r}")
        if direction == RIGHT:  # arrow points at the right tank: left pours out
            moved = min(amount, left, CAPACITY - right)
            left -= moved
            right += moved
        else:
            moved = min(amount, right, CAPACITY - left)
            right -= moved
            left += moved
    return left, right


def _clamped_steps(left: int, ops: list) -> int:
    """How many pours moved strictly less liquid than the pips asked for."""
    right = TOTAL - left
    clamped = 0
    for direction, amount in ops:
        if direction == RIGHT:
            moved = min(amount, left, CAPACITY - right)
            left -= moved
            right += moved
        else:
            moved = min(amount, right, CAPACITY - left)
            right -= moved
            left += moved
        if moved < amount:
            clamped += 1
    return clamped


def _naive_left(left: int, ops: list) -> int:
    """The level a reader gets by adding the pips and ignoring saturation."""
    for direction, amount in ops:
        left += -amount if direction == RIGHT else amount
    return left


# --------------------------------------------------------------------------- #
# geometry
# --------------------------------------------------------------------------- #
def _metrics(scene: dict) -> dict:
    cell = scene["cell_px"]
    tank_w = scene["tank_w_px"]
    tri_h = scene["tri_h_px"]
    tank_h = CAPACITY * cell + 2 * BORDER_PX
    row_h = tri_h + 4
    content_w = 2 * tank_w + TANK_GAP
    content_h = tank_h + TANK_ROW_GAP + STEPS * row_h
    inset = FRAME_PX + PAD_PX
    return {
        "cell": cell,
        "tank_w": tank_w,
        "tank_h": tank_h,
        "row_h": row_h,
        "content_w": content_w,
        "content_h": content_h,
        "block_w": content_w + 2 * inset,
        "block_h": content_h + 2 * inset,
        "content_x": scene["origin_x"] + inset,
        "content_y": scene["origin_y"] + inset,
    }


def block_extent(scene: dict) -> tuple[int, int, int, int]:
    """Outer pixel box of the decisive block, as (x0, y0, x1, y1)."""
    m = _metrics(scene)
    return (
        scene["origin_x"],
        scene["origin_y"],
        scene["origin_x"] + m["block_w"],
        scene["origin_y"] + m["block_h"],
    )


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) * 7919 + 13)
    cell = rng.choice([7, 8])
    tank_w = rng.choice([22, 24, 26])
    tri_w = rng.choice([18, 20])
    tri_h = rng.choice([10, 11])
    pip_side = rng.choice([5, 6])
    tight = (int(seed) % 13) == 0
    pip_gap = 2 if tight else rng.choice([3, 4])

    target = int(seed) % (CAPACITY + 1)
    start = 0
    ops: list = []
    for _ in range(6000):
        start = rng.randint(0, CAPACITY)
        ops = [(rng.choice([LEFT, RIGHT]), rng.randint(1, 3)) for _ in range(STEPS)]
        final_left, _ = _replay(start, ops)
        if final_left != target:
            continue
        if final_left == start:
            continue  # the start reading must never be the answer
        if _clamped_steps(start, ops) == 0:
            continue  # saturation must matter somewhere in the replay
        if _naive_left(start, ops) == final_left:
            continue  # adding the pips blindly must not land on the answer
        fuller = LEFT if final_left * 2 > TOTAL else (RIGHT if final_left * 2 < TOTAL else "equal")
        # counterbalance the last arrow against the side that ends up fuller
        balanced = fuller != "equal" and 0 < target < TOTAL
        if balanced and (ops[-1][0] == fuller) != (int(seed) % 3 == 0):
            continue
        break
    else:  # pragma: no cover - the rejection loop always succeeds in practice
        raise RuntimeError(f"could not sample a scene for seed {seed}")

    scene = {
        "width": CANVAS,
        "height": CANVAS,
        "cell_px": cell,
        "tank_w_px": tank_w,
        "tri_w_px": tri_w,
        "tri_h_px": tri_h,
        "pip_side_px": pip_side,
        "pip_gap_px": pip_gap,
        "origin_x": 0,
        "origin_y": 0,
        "start_left": start,
        "ops": [[direction, amount] for direction, amount in ops],
        "decor": [],
    }
    m = _metrics(scene)
    scene["origin_x"] = rng.randint(26, CANVAS - 26 - m["block_w"])
    scene["origin_y"] = rng.randint(26, CANVAS - 26 - m["block_h"])
    scene["decor"] = _sample_decor(rng, scene)
    return scene


def _sample_decor(rng: random.Random, scene: dict) -> list:
    x0, y0, x1, y1 = block_extent(scene)
    keep_out = (x0 - 16, y0 - 16, x1 + 16, y1 + 16)
    decor: list = []
    attempts = 0
    while len(decor) < rng.randint(6, 10) and attempts < 400:
        attempts += 1
        kind = rng.choice(["disc", "box", "bar"])
        w = rng.randint(14, 40)
        h = rng.randint(14, 40) if kind != "bar" else rng.randint(3, 5)
        px = rng.randint(8, CANVAS - 8 - w)
        py = rng.randint(8, CANVAS - 8 - h)
        if px < keep_out[2] and px + w > keep_out[0] and py < keep_out[3] and py + h > keep_out[1]:
            continue
        decor.append([kind, px, py, w, h])
    return decor


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (scene["width"], scene["height"]), BG)
    draw = ImageDraw.Draw(image)

    for kind, px, py, w, h in scene["decor"]:
        box = (px, py, px + w, py + h)
        if kind == "disc":
            draw.ellipse(box, outline=DECOR, width=3)
        elif kind == "box":
            draw.rectangle(box, outline=DECOR, width=3)
        else:
            draw.rectangle(box, fill=DECOR)

    m = _metrics(scene)
    x0, y0, x1, y1 = block_extent(scene)
    draw.rectangle((x0, y0, x1, y1), fill=PANEL, outline=INK, width=FRAME_PX)

    cell = m["cell"]
    tank_w = m["tank_w"]
    tank_h = m["tank_h"]
    levels = (scene["start_left"], TOTAL - scene["start_left"])
    for index, level in enumerate(levels):
        tx = m["content_x"] + index * (tank_w + TANK_GAP)
        ty = m["content_y"]
        draw.rectangle((tx, ty, tx + tank_w, ty + tank_h), outline=INK, width=BORDER_PX)
        ix0 = tx + BORDER_PX
        iy0 = ty + BORDER_PX
        ix1 = tx + tank_w - BORDER_PX
        iy1 = ty + tank_h - BORDER_PX
        for step in range(1, CAPACITY):
            ry = iy1 - step * cell
            draw.line((ix0, ry, ix1, ry), fill=RULE, width=1)
        if level > 0:
            draw.rectangle((ix0, iy1 - level * cell, ix1, iy1), fill=LIQUID)

    tri_w = scene["tri_w_px"]
    tri_h = scene["tri_h_px"]
    pip_side = scene["pip_side_px"]
    pip_gap = scene["pip_gap_px"]
    for index, (direction, amount) in enumerate(scene["ops"]):
        ry = m["content_y"] + tank_h + TANK_ROW_GAP + index * m["row_h"]
        ax = m["content_x"] + 2
        if direction == RIGHT:
            points = [(ax, ry), (ax, ry + tri_h), (ax + tri_w, ry + tri_h // 2)]
        else:
            points = [(ax + tri_w, ry), (ax + tri_w, ry + tri_h), (ax, ry + tri_h // 2)]
        draw.polygon(points, fill=ARROW)
        px = ax + tri_w + PIP_LEAD
        py = ry + (tri_h - pip_side) // 2
        for pip in range(int(amount)):
            left = px + pip * (pip_side + pip_gap)
            draw.rectangle((left, py, left + pip_side - 1, py + pip_side - 1), fill=PIP)
    return image


# --------------------------------------------------------------------------- #
# gold, margin, quarantine, symmetries
# --------------------------------------------------------------------------- #
def analytic_gold(scene: dict) -> str:
    left, right = _replay(int(scene["start_left"]), [tuple(op) for op in scene["ops"]])
    return f"{left}-{right}"


def margin(scene: dict) -> float:
    """Thinnest decisive pixel feature, in pixels."""
    features = [
        scene["cell_px"] / 2.0,            # slack of a fill edge from a cell line
        float(scene["pip_side_px"]),       # pip body
        float(scene["pip_gap_px"]),        # separation between adjacent pips
        scene["tri_w_px"] / 6.0,           # arrow centroid offset from its own centre
        float(BORDER_PX + 1),              # tank wall
    ]
    return float(min(features))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list:
    """No latent aliases are declared: every sampled field is drawn."""
    return []
