"""Independent pixel-only inverse arm for convexity_search.

Receives only the final RGB image. It recovers the dark filled region in each
of the four quadrants, measures how much its convex-hull area exceeds its own
filled area (a concavity proxy), and selects the quadrant whose shape is the
most convex (minimum hull-to-area excess). Imports nothing from the renderer,
prompts, generator, or verifier.
"""

from __future__ import annotations

import numpy as np


def _poly_area(pts: list) -> float:
    n = len(pts)
    area = 0.0
    for i in range(n):
        x0, y0 = pts[i]
        x1, y1 = pts[(i + 1) % n]
        area += x0 * y1 - x1 * y0
    return abs(area) / 2.0


def _hull_area(coords: np.ndarray) -> float:
    if len(coords) < 3:
        area = len(coords)
        return float(area)
    pts = sorted({(int(x), int(y)) for x, y in coords})
    pts = [(float(x), float(y)) for x, y in pts]

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    return _poly_area(hull)


def _concavity(ys, xs) -> float:
    if xs.size < 3:
        return 0.0
    coords = np.column_stack([xs.astype(float), ys.astype(float)])
    area = float(xs.size)
    hull_area = _hull_area(coords)
    if hull_area <= 0:
        return 0.0
    return max(0.0, hull_area / area - 1.0)


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < 150

    ys, xs = np.nonzero(dark)
    if xs.size == 0:
        return "TL"

    x_split = w * 0.5
    y_split = h * 0.5

    best_sector = "TL"
    best_score = float("inf")
    for sector in ("TL", "TR", "BL", "BR"):
        if sector in ("TL", "BL"):
            left = xs < x_split
        else:
            left = xs >= x_split
        if sector in ("TL", "TR"):
            band = ys < y_split
        else:
            band = ys >= y_split
        sel = band & left
        if sel.sum() < 3:
            score = 0.0
        else:
            score = _concavity(ys[sel], xs[sel])
        if score < best_score:
            best_score = score
            best_sector = sector

    return best_sector
