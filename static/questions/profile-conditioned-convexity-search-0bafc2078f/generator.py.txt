"""Convexity visual-search world.

Four filled dark polygons sit one per quadrant of a 2x2 grid. Exactly one
polygon is convex (all interior angles < 180 degrees, no inward notch); the
other three are concave (each has exactly one clearly visible inward notch).
The decision is the quadrant containing the only convex polygon. All rendering
is deterministic and every decision-relevant corner is visible in the raster.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image, ImageDraw

SIZE = 480
BG = (255, 255, 255)
POLY = (20, 20, 20)
DIV = (205, 205, 205)

SECTORS = ("TL", "TR", "BL", "BR")
SECTOR_CENTERS = {
    "TL": (SIZE // 4, SIZE // 4),
    "TR": (3 * SIZE // 4, SIZE // 4),
    "BL": (SIZE // 4, 3 * SIZE // 4),
    "BR": (3 * SIZE // 4, 3 * SIZE // 4),
}

RADIUS_MIN = 50.0
RADIUS_MAX = 74.0
JITTER_LOW = 0.98
JITTER_HIGH = 1.03
DEPTH_FRAC_MIN = 0.52
DEPTH_FRAC_MAX = 0.68
SHOULDER1 = 0.06
SHOULDER2 = 0.94
QUARANTINE_THRESH = 0.03


def _signed_area(verts: list[list[float]]) -> float:
    n = len(verts)
    area = 0.0
    for i in range(n):
        x0, y0 = verts[i]
        x1, y1 = verts[(i + 1) % n]
        area += x0 * y1 - x1 * y0
    return area / 2.0


def _reflex_count(verts: list[list[float]]) -> int:
    n = len(verts)
    area = _signed_area(verts)
    orient = 1.0 if area >= 0 else -1.0
    count = 0
    for i in range(n):
        p0 = verts[i]
        p1 = verts[(i + 1) % n]
        p2 = verts[(i + 2) % n]
        a = (p1[0] - p0[0], p1[1] - p0[1])
        b = (p2[0] - p1[0], p2[1] - p1[1])
        cross = a[0] * b[1] - a[1] * b[0]
        if (cross * orient) < 0:
            count += 1
    return count


def _poly_area(verts: list[list[float]]) -> float:
    return abs(_signed_area(verts))


def _convex_hull_area(verts: list[list[float]]) -> float:
    pts = sorted({(round(x, 6), round(y, 6)) for x, y in verts})
    if len(pts) <= 1:
        return 0.0

    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    lower = []
    for p in pts:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], p) <= 0:
            lower.pop()
        lower.append(p)
    upper = []
    for p in reversed(pts):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], p) <= 0:
            upper.pop()
        upper.append(p)
    hull = lower[:-1] + upper[:-1]
    return _poly_area(hull)


def _concavity(verts: list[list[float]]) -> float:
    area = _poly_area(verts)
    hull_area = _convex_hull_area(verts)
    if hull_area <= 0:
        return 0.0
    return max(0.0, (hull_area - area) / hull_area)


def _convex_outer(rng, cx: float, cy: float) -> list[list[float]]:
    radius = float(rng.uniform(RADIUS_MIN, RADIUS_MAX))
    m = int(rng.integers(5, 7))
    rot = float(rng.uniform(0.0, 2.0 * math.pi))
    verts = []
    for i in range(m):
        angle = rot + 2.0 * math.pi * i / m
        r = radius * float(rng.uniform(JITTER_LOW, JITTER_HIGH))
        verts.append([cx + r * math.cos(angle), cy + r * math.sin(angle)])
    return verts


def _make_concave(rng, cx: float, cy: float, outer: list[list[float]]) -> list[list[float]]:
    m = len(outer)
    for _ in range(400):
        radius = float(rng.uniform(RADIUS_MIN, RADIUS_MAX))
        rot = float(rng.uniform(0.0, 2.0 * math.pi))
        e = int(rng.integers(0, m))
        depth = float(rng.uniform(DEPTH_FRAC_MIN, DEPTH_FRAC_MAX)) * radius

        base = []
        for i in range(m):
            angle = rot + 2.0 * math.pi * i / m
            r = radius * float(rng.uniform(JITTER_LOW, JITTER_HIGH))
            base.append([cx + r * math.cos(angle), cy + r * math.sin(angle)])

        a = base[e]
        b = base[(e + 1) % m]
        s1 = [a[0] + SHOULDER1 * (b[0] - a[0]), a[1] + SHOULDER1 * (b[1] - a[1])]
        s2 = [a[0] + SHOULDER2 * (b[0] - a[0]), a[1] + SHOULDER2 * (b[1] - a[1])]
        mid = [(a[0] + b[0]) / 2.0, (a[1] + b[1]) / 2.0]
        inx = cx - mid[0]
        iny = cy - mid[1]
        norm = math.hypot(inx, iny) or 1.0
        tip = [mid[0] + inx / norm * depth, mid[1] + iny / norm * depth]

        verts = base[: e + 1] + [s1, tip, s2] + base[e + 1 :]
        if _reflex_count(verts) == 1 and _concavity(verts) >= QUARANTINE_THRESH:
            return verts
    return outer  # pragma: no cover - extremely unlikely


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]
    polygons: dict[str, dict] = {}
    for sector in SECTORS:
        cx, cy = SECTOR_CENTERS[sector]
        if sector == winner_sector:
            outer = _convex_outer(rng, float(cx), float(cy))
            polygons[sector] = {"vertices": [[float(x), float(y)] for x, y in outer]}
        else:
            outer = _convex_outer(rng, float(cx), float(cy))
            verts = _make_concave(rng, float(cx), float(cy), outer)
            polygons[sector] = {
                "vertices": [[float(x), float(y)] for x, y in verts]
            }
    return {"polygons": polygons}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (SIZE, SIZE), BG)
    draw = ImageDraw.Draw(img)

    draw.line([SIZE // 2, 0, SIZE // 2, SIZE], fill=DIV, width=2)
    draw.line([0, SIZE // 2, SIZE, SIZE // 2], fill=DIV, width=2)

    for sector in SECTORS:
        draw.polygon([(x, y) for x, y in scene["polygons"][sector]["vertices"]], fill=POLY)

    return img


def _decision(scene: dict) -> tuple[str, float]:
    concavities = {}
    for sector in SECTORS:
        verts = scene["polygons"][sector]["vertices"]
        if _reflex_count(verts) == 0:
            concavities[sector] = 0.0
        else:
            concavities[sector] = _concavity(verts)
    winner = min(concavities, key=lambda s: (concavities[s], s))
    runner_up = min(
        (s for s in SECTORS if s != winner),
        key=lambda s: (concavities[s], s),
    )
    margin = concavities[runner_up] - concavities[winner]
    return winner, float(max(0.0, margin))


def analytic_gold(scene: dict) -> str:
    winner, _ = _decision(scene)
    return winner


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
