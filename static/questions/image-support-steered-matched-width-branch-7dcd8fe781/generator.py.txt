"""Deterministic renderer and analytic arm for Width-Matched Fork Trace."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS = 512
BACKGROUND = (248, 247, 242)
INSET_FILL = (238, 241, 243)
INSET_OUTLINE = (184, 191, 198)
NAVY = (24, 45, 69)
BLUE = (31, 119, 207)
PURPLE = (139, 68, 181)
RED = (215, 55, 58)
GOLD = (225, 164, 31)


def sample_scene(seed: int) -> dict:
    """Sample one balanced compact fork from an integer seed."""
    rng = random.Random(int(seed))
    grid_centers = (96.0, 160.0, 224.0, 288.0, 352.0, 416.0)
    matching_terminal = "red" if int(seed) % 2 == 0 else "gold"
    matching_side = "upper" if (int(seed) // 2) % 2 == 0 else "lower"
    trunk_width = float(rng.choice((10.0, 11.0, 12.0)))
    delta = 5.5
    other_width = trunk_width + delta if (int(seed) // 4) % 2 == 0 else trunk_width - delta
    return {
        "center_x": rng.choice(grid_centers) + rng.choice((-3.0, 0.0, 3.0)),
        "center_y": rng.choice(grid_centers) + rng.choice((-3.0, 0.0, 3.0)),
        "rotation": float(((int(seed) // 8) % 4) * 90 + rng.uniform(-6.0, 6.0)),
        "matching_terminal": matching_terminal,
        "matching_side": matching_side,
        "trunk_width": trunk_width,
        "other_width": float(other_width),
        "scale": float(rng.choice((0.94, 1.0, 1.06))),
    }


def _rotate(point: tuple[float, float], angle_degrees: float) -> tuple[float, float]:
    angle = math.radians(angle_degrees)
    cosine, sine = math.cos(angle), math.sin(angle)
    return point[0] * cosine - point[1] * sine, point[0] * sine + point[1] * cosine


def _geometry(scene: dict) -> dict[str, tuple[float, float]]:
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    scale = float(scene["scale"])
    angle = float(scene["rotation"])
    local = {
        "blue": (-29.0, 0.0),
        "junction": (-4.0, 0.0),
        "upper": (25.0, -22.0),
        "lower": (25.0, 22.0),
    }
    output: dict[str, tuple[float, float]] = {}
    for name, (x, y) in local.items():
        rx, ry = _rotate((x * scale, y * scale), angle)
        output[name] = (cx + rx, cy + ry)
    return output


def _scaled(point: tuple[float, float], factor: int) -> tuple[int, int]:
    return round(point[0] * factor), round(point[1] * factor)


def render(scene: dict) -> Image.Image:
    """Render the visible fork. All label inputs are expressed in final pixels."""
    factor = 4
    size = CANVAS * factor
    canvas = Image.new("RGB", (size, size), BACKGROUND)
    draw = ImageDraw.Draw(canvas)
    points = _geometry(scene)
    scale = float(scene["scale"])
    center = (float(scene["center_x"]), float(scene["center_y"]))
    inset_radius = 43.0 * scale
    draw.ellipse(
        [
            round((center[0] - inset_radius) * factor),
            round((center[1] - inset_radius) * factor),
            round((center[0] + inset_radius) * factor),
            round((center[1] + inset_radius) * factor),
        ],
        fill=INSET_FILL,
        outline=INSET_OUTLINE,
        width=1 * factor,
    )

    matching_side = str(scene["matching_side"])
    matching_terminal = str(scene["matching_terminal"])
    other_terminal = "gold" if matching_terminal == "red" else "red"
    terminal_by_side = {
        matching_side: matching_terminal,
        "lower" if matching_side == "upper" else "upper": other_terminal,
    }
    width_by_side = {
        matching_side: float(scene["trunk_width"]) * scale,
        "lower" if matching_side == "upper" else "upper": float(scene["other_width"]) * scale,
    }

    # The three corridors form one marker-mediated graph. Draw wide first so the
    # shared junction does not create a width-order artifact.
    for side in sorted(("upper", "lower"), key=lambda item: -width_by_side[item]):
        draw.line(
            [_scaled(points["junction"], factor), _scaled(points[side], factor)],
            fill=NAVY,
            width=max(2, round(width_by_side[side] * factor)),
        )
    draw.line(
        [_scaled(points["blue"], factor), _scaled(points["junction"], factor)],
        fill=NAVY,
        width=max(2, round(float(scene["trunk_width"]) * scale * factor)),
    )

    marker_radius = 6.4 * scale
    marker_colors = {"blue": BLUE, "junction": PURPLE, "red": RED, "gold": GOLD}
    endpoint_for_terminal = {
        terminal_by_side["upper"]: points["upper"],
        terminal_by_side["lower"]: points["lower"],
    }
    marker_points = {
        "blue": points["blue"],
        "junction": points["junction"],
        "red": endpoint_for_terminal["red"],
        "gold": endpoint_for_terminal["gold"],
    }
    for name in ("blue", "junction", "red", "gold"):
        x, y = marker_points[name]
        draw.ellipse(
            [
                round((x - marker_radius) * factor),
                round((y - marker_radius) * factor),
                round((x + marker_radius) * factor),
                round((y + marker_radius) * factor),
            ],
            fill=marker_colors[name],
            outline=BACKGROUND,
            width=1 * factor,
        )

    return canvas.resize((CANVAS, CANVAS), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["matching_terminal"])


def margin(scene: dict) -> float:
    return abs(float(scene["other_width"]) - float(scene["trunk_width"])) * float(scene["scale"])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
