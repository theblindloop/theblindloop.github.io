"""Deterministic Pillow renderer and latent analytic arm."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
WHITE = (255, 255, 255)
NAVY = (31, 49, 67)
CYAN = (41, 190, 195)
CORAL = (239, 105, 93)
PALE = (230, 235, 236)
GOLD = (241, 181, 50)

STEP = 16
GROUP_OFFSET = 128
TOKEN_RADIUS = 7


def _rotate(values: list[str], delta: int) -> list[str]:
    """Move each source item by delta clockwise corners."""
    out = [""] * 4
    for source, value in enumerate(values):
        out[(source + delta) % 4] = value
    return out


def _expected(scene: dict[str, Any]) -> tuple[list[str], list[str]]:
    return _rotate(scene["source_shapes"], 1), _rotate(scene["source_colors"], -1)


def sample_scene(seed: int) -> dict[str, Any]:
    """Create matched adjacent-seed pairs with identical nuisance variables."""
    pair_seed = int(seed) // 2
    rng = random.Random(0xC0A7 + pair_seed * 10007)

    source_shapes = ["circle", "circle", "square", "square"]
    source_colors = ["cyan", "cyan", "coral", "coral"]
    rng.shuffle(source_shapes)
    rng.shuffle(source_colors)
    # Whole-cell placement randomizes absolute position while keeping the two
    # decisive groups on separated ablation columns in one focal neighborhood.
    # Centers sit inside the 32-pixel exclusive cores of the overlapping
    # 8x8 ablation windows. Each whole group therefore occupies one grid cell,
    # while the 128-pixel inter-group gap leaves one inactive cell between them.
    origin = [88 + 64 * rng.randrange(4), 88 + 64 * rng.randrange(6)]
    target_shapes = _rotate(source_shapes, 1)
    target_colors = _rotate(source_colors, -1)

    if int(seed) % 2:
        # Alternate the corrupted attribute. A swap preserves shape/color counts,
        # total token count, layout, and palette while breaking two bindings.
        values = target_shapes if pair_seed % 2 == 0 else target_colors
        pairs = [(a, b) for a in range(4) for b in range(a + 1, 4) if values[a] != values[b]]
        a, b = rng.choice(pairs)
        values[a], values[b] = values[b], values[a]

    return {
        "source_shapes": source_shapes,
        "source_colors": source_colors,
        "target_shapes": target_shapes,
        "target_colors": target_colors,
        "origin": origin,
        "phase_turns": 0,
    }


def _mismatch_count(scene: dict[str, Any]) -> int:
    expected_shapes, expected_colors = _expected(scene)
    return sum(
        (scene["target_shapes"][i], scene["target_colors"][i])
        != (expected_shapes[i], expected_colors[i])
        for i in range(4)
    )


def analytic_gold(scene: dict[str, Any]) -> str:
    # Direct scene accesses intentionally match latent_alias.label_inputs.
    expected_shapes = _rotate(scene["source_shapes"], 1)
    expected_colors = _rotate(scene["source_colors"], -1)
    target_shapes = scene["target_shapes"]
    target_colors = scene["target_colors"]
    return "yes" if all(
        (target_shapes[i], target_colors[i]) == (expected_shapes[i], expected_colors[i])
        for i in range(4)
    ) else "no"


def margin(scene: dict[str, Any]) -> float:
    mismatches = _mismatch_count(scene)
    return 1.0 if mismatches == 0 else float(mismatches)


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        shapes = scene["source_shapes"]
        colors = scene["source_colors"]
        target_shapes = scene["target_shapes"]
        target_colors = scene["target_colors"]
        ox, oy = map(int, scene["origin"])
        phase = int(scene["phase_turns"])
        fields_ok = all(len(v) == 4 for v in (shapes, colors, target_shapes, target_colors))
        source_ok = sorted(shapes) == ["circle", "circle", "square", "square"] and sorted(colors) == ["coral", "coral", "cyan", "cyan"]
        marginals_ok = sorted(target_shapes) == sorted(shapes) and sorted(target_colors) == sorted(colors)
        placement_ok = 32 <= ox <= CANVAS - GROUP_OFFSET - STEP - 32 and 32 <= oy <= CANVAS - STEP - 32
        mismatch_ok = _mismatch_count(scene) in (0, 2)
        return not (fields_ok and source_ok and marginals_ok and placement_ok and mismatch_ok and phase % 4 == 0)
    except (KeyError, TypeError, ValueError):
        return True


def _center(ox: int, oy: int, group: int, corner: int) -> tuple[int, int]:
    positions = ((0, 0), (STEP, 0), (STEP, STEP), (0, STEP))
    dx, dy = positions[corner]
    return ox + group * GROUP_OFFSET + dx, oy + dy


def _draw_token(draw: ImageDraw.ImageDraw, center: tuple[int, int], shape: str, color: str) -> None:
    cx, cy = center
    fill = CYAN if color == "cyan" else CORAL
    box = (cx - TOKEN_RADIUS, cy - TOKEN_RADIUS, cx + TOKEN_RADIUS, cy + TOKEN_RADIUS)
    if shape == "circle":
        draw.ellipse(box, fill=fill, outline=NAVY, width=3)
    elif shape == "square":
        draw.rectangle(box, fill=fill, outline=NAVY, width=3)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), WHITE)
    draw = ImageDraw.Draw(image)
    ox, oy = map(int, scene["origin"])
    # A periodic representational phase is declared for the pixel-identical
    # latent-alias test. Submitted scenes use phase 0; adding four is identical.
    phase = int(scene["phase_turns"]) % 4
    if phase:
        raise ValueError("only full-cycle phase aliases are renderable")

    # Nondiagnostic context remains pale and does not enter the inverse arm.
    mid_x = ox + GROUP_OFFSET // 2 + STEP // 2
    mid_y = oy + STEP // 2
    draw.line((mid_x - 8, mid_y, mid_x + 7, mid_y), fill=PALE, width=3)
    draw.polygon(((mid_x + 7, mid_y), (mid_x + 2, mid_y - 4), (mid_x + 2, mid_y + 4)), fill=PALE)
    for group, (shapes, colors) in enumerate(((scene["source_shapes"], scene["source_colors"]), (scene["target_shapes"], scene["target_colors"]))):
        for corner in range(4):
            _draw_token(draw, _center(ox, oy, group, corner), shapes[corner], colors[corner])
    # Tiny gold anchors make the direction of panel correspondence visible but
    # are deliberately ignored by the oracle and cannot determine the label.
    draw.ellipse((ox - 18, oy + 7, ox - 12, oy + 13), fill=GOLD)
    draw.ellipse((ox + GROUP_OFFSET + STEP + 12, oy + 7, ox + GROUP_OFFSET + STEP + 18, oy + 13), fill=GOLD)
    return image


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = dict(scene)
    twin["phase_turns"] = int(scene["phase_turns"]) + 4
    return [("add_full_cycle_to_periodic_phase", twin)]
