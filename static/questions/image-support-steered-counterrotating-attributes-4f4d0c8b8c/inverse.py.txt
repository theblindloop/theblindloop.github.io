"""Independent inverse arm that receives only the final raster image.

It segments the eight token components, classifies each component's geometry
and interior color, infers the two nearby 2x2 groups from component centroids,
and applies the two opposite corner transports. It imports no world module and
contains no absolute scene coordinates.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_NAVY = np.array((31, 49, 67), dtype=np.int16)
_CYAN = np.array((41, 190, 195), dtype=np.int16)
_CORAL = np.array((239, 105, 93), dtype=np.int16)


def _near(arr: np.ndarray, color: np.ndarray, radius: float = 20.0) -> np.ndarray:
    delta = arr.astype(np.int32) - color.astype(np.int32)
    return np.sum(delta * delta, axis=2) <= radius * radius


def _components(mask: np.ndarray) -> list[np.ndarray]:
    seen = np.zeros(mask.shape, dtype=bool)
    found: list[np.ndarray] = []
    height, width = mask.shape
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= 100:
            found.append(np.asarray(points, dtype=np.int32))
    return found


def _classify(arr: np.ndarray, points: np.ndarray) -> dict[str, object]:
    ys, xs = points[:, 0], points[:, 1]
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (14 <= width <= 16 and 14 <= height <= 16 and 150 <= len(points) <= 240):
        raise ValueError("token component geometry is malformed")
    shape = "square" if len(points) >= 205 else "circle"
    patch = arr[y0:y1 + 1, x0:x1 + 1]
    cyan_count = int(_near(patch, _CYAN, 18.0).sum())
    coral_count = int(_near(patch, _CORAL, 18.0).sum())
    if max(cyan_count, coral_count) < 45 or min(cyan_count, coral_count) > 5:
        raise ValueError("token interior color is missing or mixed")
    color = "cyan" if cyan_count > coral_count else "coral"
    return {"cx": float(xs.mean()), "cy": float(ys.mean()), "shape": shape, "color": color}


def _decode_group(columns: list[list[dict[str, object]]]) -> tuple[list[str], list[str]]:
    if len(columns) != 2 or any(len(column) != 2 for column in columns):
        raise ValueError("group does not contain a 2x2 token layout")
    left = sorted(columns[0], key=lambda item: float(item["cy"]))
    right = sorted(columns[1], key=lambda item: float(item["cy"]))
    if any(abs(float(column[1]["cy"]) - float(column[0]["cy"])) < 12 for column in (left, right)):
        raise ValueError("token rows are not separated")
    ordered = [left[0], right[0], right[1], left[1]]
    return [str(item["shape"]) for item in ordered], [str(item["color"]) for item in ordered]


def _move(values: list[str], delta: int) -> list[str]:
    out = [""] * 4
    for source, value in enumerate(values):
        out[(source + delta) % 4] = value
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    token_mask = _near(arr, _NAVY) | _near(arr, _CYAN) | _near(arr, _CORAL)
    components = _components(token_mask)
    if len(components) != 8:
        raise ValueError(f"expected eight disconnected tokens, found {len(components)}")
    tokens = sorted((_classify(arr, component) for component in components), key=lambda item: float(item["cx"]))

    # Four x-columns, each containing top and bottom. The largest center gap is
    # the panel boundary; no absolute coordinate or renderer state is used.
    columns = [tokens[index:index + 2] for index in range(0, 8, 2)]
    if any(abs(float(column[0]["cx"]) - float(column[1]["cx"])) > 1.5 for column in columns):
        raise ValueError("tokens do not form four vertical columns")
    centers = [sum(float(item["cx"]) for item in column) / 2.0 for column in columns]
    gaps = [centers[i + 1] - centers[i] for i in range(3)]
    if not (gaps[1] > gaps[0] + 5 and gaps[1] > gaps[2] + 5):
        raise ValueError("two panel groups are not uniquely separable")

    source_shapes, source_colors = _decode_group(columns[:2])
    target_shapes, target_colors = _decode_group(columns[2:])
    expected_shapes = _move(source_shapes, 1)
    expected_colors = _move(source_colors, -1)
    exact = all(
        (target_shapes[i], target_colors[i]) == (expected_shapes[i], expected_colors[i])
        for i in range(4)
    )
    return "yes" if exact else "no"
