from __future__ import annotations

import random
from PIL import Image, ImageDraw

PALETTE = {"blue": (30, 105, 210), "orange": (232, 119, 34), "green": (38, 153, 83)}
MAGENTA = (190, 45, 145)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    colors = ["blue", "orange", "green"]
    # The role rotation balances the answer independently of fixed palette identity.
    winner = seed % 3
    order = [winner, (winner + 1) % 3, (winner + 2) % 3]
    patterns = [
        [0.22, -0.20, 0.18, -0.18],
        [-0.13, 0.13, -0.12, 0.12],
        [0.05, -0.05, 0.04, -0.04],
    ]
    series = {}
    for role, color_index in enumerate(order):
        # A small answer-independent perturbation prevents rote coordinate lookup.
        series[colors[color_index]] = [round(v + rng.uniform(-0.006, 0.006), 4) for v in patterns[role]]
    x_values = [round(x + rng.uniform(-0.008, 0.008), 4) for x in (0.18, 0.39, 0.61, 0.82)]
    dx = rng.randint(-14, 14)
    dy = rng.randint(-10, 10)
    return {
        "x_values": x_values,
        "series": series,
        "colors": colors,
        "plot_box": [90 + dx, 55 + dy, 550 + dx, 415 + dy],
        "radius": rng.choice([7, 8]),
    }


def _residual_maxima(scene: dict) -> dict[str, float]:
    xs = scene["x_values"]
    series = scene["series"]
    return {color: max(abs(y - x) for x, y in zip(xs, ys)) for color, ys in series.items()}


def analytic_gold(scene: dict) -> str:
    # Explicit accesses are audited: every label input also affects render().
    _ = scene["colors"]
    scores = _residual_maxima(scene)
    return max(scene["colors"], key=lambda color: scores[color])


def margin(scene: dict) -> float:
    scores = sorted(_residual_maxima(scene).values(), reverse=True)
    height = scene["plot_box"][3] - scene["plot_box"][1]
    return (scores[0] - scores[1]) * height


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    xs = scene["x_values"]
    series = scene["series"]
    colors = scene["colors"]
    x0, y0, x1, y1 = scene["plot_box"]
    radius = scene["radius"]
    image = Image.new("RGB", (640, 480), (250, 250, 247))
    draw = ImageDraw.Draw(image)
    draw.rectangle((x0, y0, x1, y1), fill=(255, 255, 255), outline=(55, 59, 65), width=2)
    # Common ticks communicate equal linear scales without introducing text parsing.
    for frac in (0.2, 0.4, 0.6, 0.8):
        px = x0 + frac * (x1 - x0)
        py = y1 - frac * (y1 - y0)
        draw.line((px, y1, px, y1 + 6), fill=(55, 59, 65), width=2)
        draw.line((x0 - 6, py, x0, py), fill=(55, 59, 65), width=2)
    # Visible ideal y=x boundary.
    steps = 24
    for i in range(0, steps, 2):
        a, b = i / steps, (i + 1) / steps
        draw.line((x0 + a * (x1-x0), y1 - a * (y1-y0),
                   x0 + b * (x1-x0), y1 - b * (y1-y0)), fill=MAGENTA, width=3)
    for color in colors:
        rgb = PALETTE[color]
        for xv, residual in zip(xs, series[color]):
            yv = xv + residual
            cx = x0 + xv * (x1 - x0)
            cy = y1 - yv * (y1 - y0)
            draw.ellipse((cx-radius, cy-radius, cx+radius, cy+radius), fill=rgb, outline=(255,255,255), width=2)
    return image
