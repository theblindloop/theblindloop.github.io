from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

PALETTE = {"blue": (30, 105, 210), "orange": (232, 119, 34), "green": (38, 153, 83)}
MAGENTA = (190, 45, 145)


def _mask(arr: np.ndarray, rgb: tuple[int, int, int], tolerance: float = 45.0) -> np.ndarray:
    delta = arr.astype(float) - np.array(rgb, dtype=float)
    return np.sum(delta * delta, axis=2) <= tolerance * tolerance


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for yy, xx in np.argwhere(mask):
        if seen[yy, xx]:
            continue
        q = deque([(int(yy), int(xx))]); seen[yy, xx] = True; pts = []
        while q:
            y, x = q.popleft(); pts.append((x, y))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny,nx] and not seen[ny,nx]:
                    seen[ny,nx] = True; q.append((ny,nx))
        if len(pts) >= 20:
            found.append((sum(p[0] for p in pts)/len(pts), sum(p[1] for p in pts)/len(pts), len(pts)))
    return found


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    line_y, line_x = np.nonzero(_mask(arr, MAGENTA, 35.0))
    if len(line_x) < 100:
        raise ValueError("ideal diagonal is missing")
    slope, intercept = np.polyfit(line_x.astype(float), line_y.astype(float), 1)
    maxima = {}
    for color, rgb in PALETTE.items():
        comps = _components(_mask(arr, rgb))
        if len(comps) != 4:
            raise ValueError(f"expected four {color} checkpoints, found {len(comps)}")
        maxima[color] = max(abs(cy - (slope * cx + intercept)) for cx, cy, _ in comps)
    ranked = sorted(maxima, key=maxima.get, reverse=True)
    if maxima[ranked[0]] - maxima[ranked[1]] < 5.0:
        raise ValueError("ambiguous top residual")
    return ranked[0]
