"""Deterministic sampler and raster renderer for a USB-terminal prior conflict."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw


_PERMUTATIONS = (
    ("circle", "square", "triangle"),
    ("circle", "triangle", "square"),
    ("square", "circle", "triangle"),
    ("square", "triangle", "circle"),
    ("triangle", "circle", "square"),
    ("triangle", "square", "circle"),
)
_BACKGROUNDS = ("#f7f8f3", "#f4f7f8", "#faf6f2", "#f6f4f9")
_TRUNKS = ("#25323c", "#263836", "#343044", "#273746")
_MARKERS = ("#168f88", "#2873b5", "#9855a6", "#c26432")
_DECORATION_COLORS = ("#d89527", "#5c8f3d", "#b54c68", "#5277a8")


def sample_scene(seed: int) -> dict:
    """Return a balanced scene; each six consecutive seeds cover all assignments."""
    value = int(seed)
    rng = random.Random(value ^ 0x5A17C3)
    assignment = _PERMUTATIONS[value % len(_PERMUTATIONS)]
    terminals = [
        {"slot": slot, "shape": shape}
        for slot, shape in zip(("left", "right", "top"), assignment)
    ]
    rng.shuffle(terminals)
    palette = (value // len(_PERMUTATIONS)) % len(_BACKGROUNDS)
    center_x = rng.randint(246, 266)
    hub_y = rng.randint(276, 294)
    top_y = hub_y - rng.randint(150, 166)
    side_y = hub_y - rng.randint(48, 64)
    reach = rng.randint(116, 134)
    decorations = [
        {
            "x": rng.randint(40, 62),
            "y": rng.randint(38, 58),
            "radius": rng.choice((7, 8, 9)),
            "color": _DECORATION_COLORS[(palette + 1) % len(_DECORATION_COLORS)],
        },
        {
            "x": rng.randint(450, 472),
            "y": rng.randint(448, 470),
            "radius": rng.choice((7, 8, 9)),
            "color": _DECORATION_COLORS[(palette + 2) % len(_DECORATION_COLORS)],
        },
    ]
    if rng.randrange(2):
        decorations.reverse()
    return {
        "width": 512,
        "height": 512,
        "center_x": center_x,
        "hub_y": hub_y,
        "top_y": top_y,
        "side_y": side_y,
        "reach": reach,
        "bottom_y": hub_y + rng.randint(112, 130),
        "stroke_width": rng.choice((16, 18, 20)),
        "marker_scale": rng.choice((0.94, 0.98, 1.02, 1.06)),
        "background_color": _BACKGROUNDS[palette],
        "trunk_color": _TRUNKS[(palette + value // 24) % len(_TRUNKS)],
        "marker_color": _MARKERS[(palette + value // 12) % len(_MARKERS)],
        "decorations": decorations,
        "terminals": terminals,
    }


def _terminal_map(scene: dict) -> dict[str, str]:
    return {str(item["slot"]): str(item["shape"]) for item in scene["terminals"]}


def _slot_center(scene: dict, slot: str) -> tuple[float, float]:
    center_x = float(scene["center_x"])
    if slot == "left":
        return center_x - float(scene["reach"]), float(scene["side_y"])
    if slot == "right":
        return center_x + float(scene["reach"]), float(scene["side_y"])
    if slot == "top":
        return center_x, float(scene["top_y"])
    raise ValueError(f"unknown terminal slot {slot!r}")


def _triangle_points(
    center: tuple[float, float], radius: float, slot: str
) -> list[tuple[float, float]]:
    heading = {"top": -90.0, "left": 180.0, "right": 0.0}[slot]
    return [
        (
            center[0] + radius * math.cos(math.radians(heading + 120.0 * index)),
            center[1] + radius * math.sin(math.radians(heading + 120.0 * index)),
        )
        for index in range(3)
    ]


def _draw_shape(
    draw: ImageDraw.ImageDraw,
    shape: str,
    slot: str,
    center: tuple[float, float],
    size: float,
    scale: int,
    color: str,
) -> None:
    cx, cy = center
    if shape == "circle":
        draw.ellipse(
            tuple(int(round(value * scale)) for value in (cx - size, cy - size, cx + size, cy + size)),
            fill=color,
        )
    elif shape == "square":
        half = size * 0.90
        draw.rectangle(
            tuple(int(round(value * scale)) for value in (cx - half, cy - half, cx + half, cy + half)),
            fill=color,
        )
    elif shape == "triangle":
        draw.polygon(
            [(int(round(x * scale)), int(round(y * scale))) for x, y in _triangle_points(center, size * 1.16, slot)],
            fill=color,
        )
    else:
        raise ValueError(f"unknown terminal shape {shape!r}")


def render(scene: dict) -> Image.Image:
    """Draw one antialiased USB-like symbol; every latent field affects pixels."""
    supersample = 4
    width = int(scene["width"])
    height = int(scene["height"])
    center_x = float(scene["center_x"])
    hub_y = float(scene["hub_y"])
    top_y = float(scene["top_y"])
    side_y = float(scene["side_y"])
    reach = float(scene["reach"])
    bottom_y = float(scene["bottom_y"])
    stroke_width = int(scene["stroke_width"])
    marker_scale = float(scene["marker_scale"])
    background_color = str(scene["background_color"])
    trunk_color = str(scene["trunk_color"])
    marker_color = str(scene["marker_color"])
    decorations = scene["decorations"]
    terminal_map = _terminal_map(scene)

    image = Image.new(
        "RGB", (width * supersample, height * supersample), background_color
    )
    draw = ImageDraw.Draw(image)

    for decoration in decorations:
        x = int(decoration["x"]) * supersample
        y = int(decoration["y"]) * supersample
        radius = int(decoration["radius"]) * supersample
        draw.polygon(
            ((x, y - radius), (x + radius, y), (x, y + radius), (x - radius, y)),
            fill=str(decoration["color"]),
        )

    line_width = stroke_width * supersample

    def point(x: float, y: float) -> tuple[int, int]:
        return int(round(x * supersample)), int(round(y * supersample))

    left_center = _slot_center(scene, "left")
    right_center = _slot_center(scene, "right")
    top_center = _slot_center(scene, "top")
    branch_y = hub_y - 24.0
    draw.line(
        (point(center_x, bottom_y), point(center_x, top_y)),
        fill=trunk_color,
        width=line_width,
    )
    draw.line(
        (
            point(center_x, branch_y),
            point(center_x - reach * 0.46, branch_y),
            point(left_center[0], side_y),
        ),
        fill=trunk_color,
        width=line_width,
        joint="curve",
    )
    draw.line(
        (
            point(center_x, branch_y),
            point(center_x + reach * 0.46, branch_y),
            point(right_center[0], side_y),
        ),
        fill=trunk_color,
        width=line_width,
        joint="curve",
    )
    hub_radius = stroke_width * 0.72
    draw.ellipse(
        tuple(
            int(round(value * supersample))
            for value in (
                center_x - hub_radius,
                branch_y - hub_radius,
                center_x + hub_radius,
                branch_y + hub_radius,
            )
        ),
        fill=trunk_color,
    )
    base_radius = stroke_width * 0.64
    draw.ellipse(
        tuple(
            int(round(value * supersample))
            for value in (
                center_x - base_radius,
                bottom_y - base_radius,
                center_x + base_radius,
                bottom_y + base_radius,
            )
        ),
        fill=trunk_color,
    )

    outer_size = 33.0 * marker_scale
    inner_size = outer_size - 6.0
    for slot in ("left", "right", "top"):
        shape = terminal_map[slot]
        center = {"left": left_center, "right": right_center, "top": top_center}[slot]
        _draw_shape(draw, shape, slot, center, outer_size, supersample, trunk_color)
        _draw_shape(draw, shape, slot, center, inner_size, supersample, marker_color)

    return image.resize((width, height), Image.Resampling.LANCZOS).convert("RGB")


def analytic_gold(scene: dict) -> str:
    """Return the geometric shape assigned to the left branch terminal."""
    return _terminal_map(scene)["left"]


def margin(scene: dict) -> float:
    """Return the horizontal centroid gap from the leftmost terminal to its runner-up."""
    centers = sorted(_slot_center(scene, slot)[0] for slot in ("left", "right", "top"))
    return float(centers[1] - centers[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 90.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the disjoint corner-decoration storage order without changing pixels."""
    twin = deepcopy(scene)
    twin["decorations"] = list(reversed(twin["decorations"]))
    return [("reverse_disjoint_decoration_order", twin)]
