"""Independent pixel-only inverse for USB-like endpoint-shape binding.

The inverse sees only a raster. It isolates saturated endpoint interiors, rejects
the much smaller corner diamonds by component scale, classifies circle/square/
triangle from component occupancy, then reports the class of the leftmost large
endpoint. It imports no renderer or latent-world implementation.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    """Raised when the required raster evidence is absent or ambiguous."""


def _components(mask: np.ndarray) -> list[dict]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found: list[dict] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        ys = np.fromiter((point[0] for point in points), dtype=np.float64)
        xs = np.fromiter((point[1] for point in points), dtype=np.float64)
        box_width = int(xs.max() - xs.min() + 1)
        box_height = int(ys.max() - ys.min() + 1)
        found.append(
            {
                "area": len(points),
                "cx": float(xs.mean()),
                "cy": float(ys.mean()),
                "width": box_width,
                "height": box_height,
                "fill_ratio": float(len(points) / (box_width * box_height)),
            }
        )
    return found


def _shape_for(component: dict) -> str:
    ratio = float(component["fill_ratio"])
    if ratio < 0.64:
        return "triangle"
    if ratio > 0.90:
        return "square"
    if 0.69 <= ratio <= 0.86:
        return "circle"
    raise OracleAbstention(f"endpoint fill ratio {ratio:.3f} is ambiguous")


def _measure(image: Image.Image) -> dict:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    if rgb.ndim != 3 or rgb.shape[2] != 3 or min(rgb.shape[:2]) < 128:
        raise OracleAbstention("unsupported or too-small raster")

    channel_max = rgb.max(axis=2)
    channel_min = rgb.min(axis=2)
    chroma = channel_max - channel_min
    colored = (chroma >= 40) & (channel_max >= 88)
    large = [
        component
        for component in _components(colored)
        if component["area"] >= 650
        and 34 <= component["width"] <= 90
        and 34 <= component["height"] <= 90
    ]
    if len(large) != 3:
        raise OracleAbstention(f"expected three colored endpoints, found {len(large)}")

    for component in large:
        component["shape"] = _shape_for(component)
    inventory = {str(component["shape"]) for component in large}
    if inventory != {"circle", "square", "triangle"}:
        raise OracleAbstention(f"endpoint shape inventory is malformed: {sorted(inventory)}")

    ranked = sorted(large, key=lambda component: float(component["cx"]))
    measured_margin = float(ranked[1]["cx"] - ranked[0]["cx"])
    if measured_margin < 70.0:
        raise OracleAbstention("leftmost endpoint has an ambiguous horizontal runner-up gap")
    return {
        "decision": str(ranked[0]["shape"]),
        "margin": measured_margin,
        "components": large,
    }


def decision_from_image(image: Image.Image) -> str:
    """Return circle, square, or triangle from raster evidence alone."""
    return str(_measure(image)["decision"])
