"""Independent pixel-only inverse for paired tail-lamp color equality."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_MATCH_THRESHOLD = 55.0
_MIN_CHROMA = 62
_MIN_AREA = 360
_MAX_AREA = 760


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected component coordinate arrays for a sparse mask."""

    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for raw_y, raw_x in np.argwhere(mask):
        y, x = int(raw_y), int(raw_x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dy == 0 and dx == 0:
                        continue
                    ny, nx = cy + dy, cx + dx
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        queue.append((ny, nx))
        output.append(np.asarray(points, dtype=np.int32))
    return output


def _lamp_components(array: np.ndarray) -> list[tuple[np.ndarray, np.ndarray]]:
    high = array.max(axis=2).astype(np.int16)
    low = array.min(axis=2).astype(np.int16)
    chroma = high - low
    mask = (chroma >= _MIN_CHROMA) & (high >= 120)
    lamps = []
    for points in _components(mask):
        area = len(points)
        if area < 24:
            continue
        if not _MIN_AREA <= area <= _MAX_AREA:
            raise OracleAbstention(f"chromatic component has unsupported area {area}")
        ys, xs = points[:, 0], points[:, 1]
        height = int(ys.max() - ys.min() + 1)
        width = int(xs.max() - xs.min() + 1)
        aspect = width / height
        fill = area / float(width * height)
        if not (23 <= width <= 31 and 23 <= height <= 31):
            raise OracleAbstention("chromatic component has unsupported diameter")
        if not 0.88 <= aspect <= 1.14 or not 0.68 <= fill <= 0.86:
            raise OracleAbstention("chromatic component is not a solid circular lamp")
        colors = array[ys, xs].astype(np.float64)
        # Medians over the high-chroma component recover the solid interior and
        # are insensitive to the thin antialiased rim.
        lamps.append((points, np.median(colors, axis=0)))
    if len(lamps) != 2:
        raise OracleAbstention(f"expected two circular chromatic lamps, found {len(lamps)}")
    return lamps


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[0] < 256 or array.shape[1] < 256:
        raise OracleAbstention("raster is too small")
    lamps = _lamp_components(array)
    centers = []
    for points, _ in lamps:
        centers.append((float(points[:, 1].mean()), float(points[:, 0].mean())))
    centers.sort()
    horizontal_gap = centers[1][0] - centers[0][0]
    if not 105.0 <= horizontal_gap <= 150.0 or abs(centers[1][1] - centers[0][1]) > 3.0:
        raise OracleAbstention("the two lamps do not form a plausible horizontal rear pair")
    colors = [item[1] for item in lamps]
    distance = float(np.linalg.norm(colors[0] - colors[1]))
    if abs(distance - _MATCH_THRESHOLD) <= 20.0:
        raise OracleAbstention("lamp colors are too close to the match boundary")
    return "yes" if distance < _MATCH_THRESHOLD else "no"
