"""Deterministic renderer for the paired tail-lamp color world."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

CANVAS_SIZE = 512
CELL_SIZE = 64
SUPERSAMPLE = 4
MATCH_THRESHOLD = 55.0

LAMP_PALETTE = (
    (220, 48, 62),
    (43, 101, 222),
    (35, 166, 91),
    (236, 158, 27),
)
BACKGROUND_TONES = ((247, 244, 237), (239, 244, 246), (246, 241, 246))
BODY_TONES = ((50, 59, 66), (65, 68, 74), (45, 62, 68))


def sample_scene(seed: int) -> dict:
    """Return matched adjacent-seed control/counterfactual pairs.

    Seeds 2k and 2k+1 share position, body, decoration, and their left lamp.
    The even control repeats that color on the right; the odd counterfactual
    substitutes a distant palette entry. Position varies only among fine-grid
    cell centers so the two decisive support regions stay separated.
    """

    seed = int(seed)
    pair = seed // 2
    rng = random.Random(pair ^ 0x71A11A9)
    left_code = pair % len(LAMP_PALETTE)
    matching = seed % 2 == 0
    offset = 1 + (pair // len(LAMP_PALETTE)) % (len(LAMP_PALETTE) - 1)
    right_code = left_code if matching else (left_code + offset) % len(LAMP_PALETTE)
    if not matching and ((pair // 12) % 2):
        left_code, right_code = right_code, left_code
    return {
        "background_tone": pair % len(BACKGROUND_TONES),
        "body_tone": (pair // 3) % len(BODY_TONES),
        "canvas_size": CANVAS_SIZE,
        "decor_variant": rng.randrange(8),
        "lamp_left_code": left_code,
        "lamp_right_code": right_code,
        "left_cell": rng.randrange(1, 5),
        "row_cell": rng.randrange(1, 7),
    }


def _scaled_box(box: tuple[float, float, float, float], scale: int) -> tuple[int, int, int, int]:
    return tuple(round(value * scale) for value in box)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    background = BACKGROUND_TONES[int(scene["background_tone"]) % len(BACKGROUND_TONES)]
    body = BODY_TONES[int(scene["body_tone"]) % len(BODY_TONES)]
    decor = int(scene["decor_variant"])
    left_cell = int(scene["left_cell"])
    row_cell = int(scene["row_cell"])
    left_code = int(scene["lamp_left_code"])
    right_code = int(scene["lamp_right_code"])

    ss = SUPERSAMPLE
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)

    # Remote, achromatic road marks vary independently and are nondiagnostic.
    road = tuple(max(0, channel - 18) for channel in background)
    for index in range(3):
        x = 38 + ((decor * 41 + index * 173) % 430)
        y = 34 if (decor + index) % 2 == 0 else size - 36
        draw.rounded_rectangle(
            _scaled_box((x - 22, y - 3, x + 22, y + 3), ss),
            radius=3 * ss,
            fill=road,
        )

    lamp_y = row_cell * CELL_SIZE + CELL_SIZE // 2
    lamp_left_x = left_cell * CELL_SIZE + CELL_SIZE // 2
    lamp_right_x = (left_cell + 2) * CELL_SIZE + CELL_SIZE // 2
    center_x = (lamp_left_x + lamp_right_x) / 2.0

    outline = tuple(max(0, channel - 22) for channel in body)
    glass = tuple(min(255, channel + 88) for channel in body)
    trim = tuple(min(255, channel + 35) for channel in body)

    # A compact rear-car silhouette. Every non-lamp color has low chroma.
    draw.rounded_rectangle(
        _scaled_box((center_x - 90, lamp_y - 40, center_x + 90, lamp_y + 39), ss),
        radius=18 * ss,
        fill=body,
        outline=outline,
        width=3 * ss,
    )
    draw.polygon(
        [
            (round((center_x - 58) * ss), round((lamp_y - 39) * ss)),
            (round((center_x - 43) * ss), round((lamp_y - 63) * ss)),
            (round((center_x + 43) * ss), round((lamp_y - 63) * ss)),
            (round((center_x + 58) * ss), round((lamp_y - 39) * ss)),
        ],
        fill=body,
    )
    draw.rounded_rectangle(
        _scaled_box((center_x - 38, lamp_y - 57, center_x + 38, lamp_y - 42), ss),
        radius=4 * ss,
        fill=glass,
    )
    for wheel_x in (center_x - 60, center_x + 60):
        draw.ellipse(
            _scaled_box((wheel_x - 15, lamp_y + 25, wheel_x + 15, lamp_y + 50), ss),
            fill=outline,
        )
    draw.rounded_rectangle(
        _scaled_box((center_x - 25, lamp_y + 13, center_x + 25, lamp_y + 29), ss),
        radius=3 * ss,
        fill=trim,
    )
    draw.line(
        [(round((center_x - 70) * ss), round((lamp_y + 34) * ss)),
         (round((center_x + 70) * ss), round((lamp_y + 34) * ss))],
        fill=trim,
        width=3 * ss,
    )

    radius = 13
    for x, color_code in ((lamp_left_x, left_code), (lamp_right_x, right_code)):
        draw.ellipse(
            _scaled_box((x - radius - 3, lamp_y - radius - 3, x + radius + 3, lamp_y + radius + 3), ss),
            fill=outline,
        )
        draw.ellipse(
            _scaled_box((x - radius, lamp_y - radius, x + radius, lamp_y + radius), ss),
            fill=LAMP_PALETTE[color_code % len(LAMP_PALETTE)],
        )

    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    left = int(scene["lamp_left_code"]) % len(LAMP_PALETTE)
    right = int(scene["lamp_right_code"]) % len(LAMP_PALETTE)
    return "yes" if left == right else "no"


def margin(scene: dict) -> float:
    left = LAMP_PALETTE[int(scene["lamp_left_code"]) % len(LAMP_PALETTE)]
    right = LAMP_PALETTE[int(scene["lamp_right_code"]) % len(LAMP_PALETTE)]
    distance = math.sqrt(sum((float(a) - float(b)) ** 2 for a, b in zip(left, right)))
    return abs(distance - MATCH_THRESHOLD)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["lamp_left_code"] = int(twin["lamp_left_code"]) + len(LAMP_PALETTE)
    twin["lamp_right_code"] = int(twin["lamp_right_code"]) + len(LAMP_PALETTE)
    return [("one_full_palette_cycle", twin)]
