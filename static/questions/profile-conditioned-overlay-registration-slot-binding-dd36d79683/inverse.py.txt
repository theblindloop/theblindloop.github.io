"""Pixel-only inverse arm for the overlay-registration world.

This module receives a single ``PIL.Image`` and nothing else.  It re-derives the
whole decision from ink alone:

1. segment the two solid black registration crosses, split them into the upper
   (board) cross and the lower (sheet) cross by the vertical gap between the two
   panels, and take their centroids;
2. segment the light-gray shaded square and take its centroid;
3. segment each colored dot and take its centroid;
4. translate every dot by ``board_cross - sheet_cross`` and report the color of
   the translated dot closest to the shaded centroid.

It shares no geometry code with the renderer: it never reconstructs squares,
frames or scene fields, and it abstains whenever the ink it needs is missing or
structurally implausible (a corrupted or blanked raster).
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "overlay_registration_slot_binding-oracle-0.1.0"

ABSTAIN = "abstain"

_INK_TOLERANCE = 26
_MIN_CROSS_PIXELS = 60
_MIN_SHADE_PIXELS = 400
_MIN_DOT_PIXELS = 90

_PROBE_COLORS = {
    "red": (214, 36, 36),
    "blue": (36, 74, 214),
    "green": (22, 140, 64),
    "orange": (240, 150, 24),
}
_BLACK = (0, 0, 0)
_SHADE = (206, 206, 206)


def _mask(array: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    target = np.array(rgb, dtype=np.int16)
    delta = np.abs(array.astype(np.int16) - target).max(axis=2)
    return delta <= _INK_TOLERANCE


def _centroid(mask: np.ndarray) -> tuple[float, float]:
    ys, xs = np.nonzero(mask)
    return float(xs.mean()), float(ys.mean())


def _split_rows(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray] | None:
    """Cut a mask into an upper and a lower group at its widest empty band."""
    rows = np.nonzero(mask.any(axis=1))[0]
    if rows.size < 2:
        return None
    gaps = np.diff(rows)
    index = int(np.argmax(gaps))
    if gaps[index] < 20:
        return None
    boundary = rows[index]
    upper = mask.copy()
    upper[boundary + 1 :, :] = False
    lower = mask.copy()
    lower[: boundary + 1, :] = False
    return upper, lower


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"))
    if array.ndim != 3 or array.shape[0] < 32 or array.shape[1] < 32:
        return ABSTAIN

    black = _mask(array, _BLACK)
    if int(black.sum()) < 2 * _MIN_CROSS_PIXELS:
        return ABSTAIN
    split = _split_rows(black)
    if split is None:
        return ABSTAIN
    upper, lower = split
    if int(upper.sum()) < _MIN_CROSS_PIXELS or int(lower.sum()) < _MIN_CROSS_PIXELS:
        return ABSTAIN
    board_fid = _centroid(upper)
    sheet_fid = _centroid(lower)

    shade = _mask(array, _SHADE)
    if int(shade.sum()) < _MIN_SHADE_PIXELS:
        return ABSTAIN
    target = _centroid(shade)

    tx = board_fid[0] - sheet_fid[0]
    ty = board_fid[1] - sheet_fid[1]

    ranked = []
    for name, rgb in _PROBE_COLORS.items():
        blob = _mask(array, rgb)
        if int(blob.sum()) < _MIN_DOT_PIXELS:
            return ABSTAIN
        cx, cy = _centroid(blob)
        distance = math.hypot(cx + tx - target[0], cy + ty - target[1])
        ranked.append((distance, name))
    ranked.sort()
    return ranked[0][1]
