"""Independent PNG-only inverse arm for rotated-glyph-match.

Recovers the magenta-framed reference glyph and each tagged tile glyph directly
from the raster, rotates the reference by a quarter-turn clockwise, and returns
the colour tag of the single tile whose 5x5 grid equals that rotated reference.
Uses no renderer geometry, scene parameters, prompt, or gold code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 5
N_TILES = 4
BORDER_W = 6

TAG_RGB = {
    "crimson": (200, 28, 40),
    "teal": (0, 150, 150),
    "amber": (235, 165, 25),
    "violet": (135, 60, 185),
}


def _rotcw(arr: np.ndarray) -> np.ndarray:
    return np.rot90(arr, k=-1)


def _components(mask: np.ndarray, minsize: int) -> list[tuple[int, int, int, int]]:
    pixels = set(zip(*np.nonzero(mask)))
    boxes: list[tuple[int, int, int, int]] = []
    while pixels:
        start = pixels.pop()
        stack = [start]
        ys, xs = [start[0]], [start[1]]
        size = 1
        while stack:
            y, x = stack.pop()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if (ny, nx) in pixels:
                    pixels.remove((ny, nx))
                    size += 1
                    ys.append(ny)
                    xs.append(nx)
                    stack.append((ny, nx))
        if size >= minsize:
            boxes.append((min(ys), min(xs), max(ys), max(xs)))
    return boxes


def _classify_tag(crop: np.ndarray) -> str:
    mean = crop.reshape(-1, 3).mean(axis=0)
    best = None
    best_dist = None
    for name, rgb in TAG_RGB.items():
        dist = float(np.sum((mean - np.asarray(rgb)) ** 2))
        if best is None or dist < best_dist:
            best, best_dist = name, dist
    return best


def _read_grid(arr: np.ndarray, bbox: tuple[int, int, int, int]) -> list[list[int]]:
    y0, x0, y1, x1 = bbox
    inner_w = x1 - x0 - 2 * BORDER_W
    inner_h = y1 - y0 - 2 * BORDER_W
    cell = min(inner_w, inner_h) // GRID
    grid = []
    for r in range(GRID):
        row = []
        for k in range(GRID):
            cx = x0 + BORDER_W + (k + 0.5) * cell
            cy = y0 + BORDER_W + (r + 0.5) * cell
            window = max(3, cell // 6)
            xa, xb = int(cx) - window, int(cx) + window + 1
            ya, yb = int(cy) - window, int(cy) + window + 1
            block = arr[ya:yb, xa:xb, :]
            dark = float(block.mean()) < 150.0
            row.append(1 if dark else 0)
        grid.append(row)
    return grid


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)

    magenta = (
        (arr[:, :, 0] > 150)
        & (arr[:, :, 2] > 150)
        & (arr[:, :, 1] < 130)
    )
    ref_boxes = _components(magenta, 80)
    if len(ref_boxes) != 1:
        raise ValueError(f"expected one reference frame, found {len(ref_boxes)}")
    ref_box = ref_boxes[0]
    reference = np.asarray(_read_grid(arr, ref_box), dtype=np.int8)
    target = _rotcw(reference)

    gray = (
        (abs(arr[:, :, 0] - 110) < 45)
        & (abs(arr[:, :, 1] - 110) < 45)
        & (abs(arr[:, :, 2] - 110) < 45)
    )
    tile_boxes = [b for b in _components(gray, 40) if (b[2] - b[0]) >= 40]
    if len(tile_boxes) != N_TILES:
        raise ValueError(f"expected {N_TILES} tile frames, found {len(tile_boxes)}")
    tile_boxes.sort(key=lambda b: (b[1], b[0]))

    for box in tile_boxes:
        y0, x0, y1, x1 = box
        grid = np.asarray(_read_grid(arr, box), dtype=np.int8)
        if np.array_equal(grid, target):
            tag_cx = int((x0 + x1) // 2)
            tag_cy = int(y0 - 3 * 12)
            crop = arr[tag_cy - 8 : tag_cy + 9, tag_cx - 8 : tag_cx + 9, :]
            return _classify_tag(crop)
    raise ValueError("no tile matched the quarter-turn reference")
