"""Deterministic renderer and analytic arm for rotated-glyph-match.

One 5x5 asymmetric reference glyph is drawn upright inside a magenta-bordered
frame. Four smaller frames below each hold another rendering of the SAME 5x5
glyph, but rotated by 0, 90, 180 or 270 degrees clockwise (each frame gets one
of the four rotations in a random order), and each frame is tagged with one of
four saturated colours above it.

The counterfactual is a geometric transformation: a quarter-turn clockwise
rotation of the reference glyph.  The question asks which tagged tile equals the
reference after exactly that rotation.  Because all four tiles are rotations of
the same underlying glyph (including one tile holding an exact, rotation-zero
copy of the reference), the answer can only be recovered by actually rotating
the reference in the specified direction and matching orientation, never by
outline or identity matching.
"""

from __future__ import annotations

import random

import numpy as np

from PIL import Image, ImageDraw

CANVAS = 720
GRID = 5
N_TILES = 4

BG = (250, 250, 250)
INK = (42, 42, 42)
CELL_LIGHT = (238, 238, 238)
WHITE_LINE = (255, 255, 255)

REF_COLOR = (200, 20, 200)
TILE_COLOR = (110, 110, 110)
BORDER_W = 6

REF_ORIGIN = (230, 40)
REF_SIZE = 282

TILE_X = [24, 204, 384, 564]
TILE_Y = 430
TILE_SIZE = 152

TAG_NAMES = ["crimson", "teal", "amber", "violet"]
TAG_RGB = {
    "crimson": (200, 28, 40),
    "teal": (0, 150, 150),
    "amber": (235, 165, 25),
    "violet": (135, 60, 185),
}
TAG_HALF = 12


def rotcw_web(grid: list) -> list:
    """Rotate a GRID x GRID 0/1 grid by a quarter turn clockwise."""

    arr = np.asarray(grid, dtype=np.int8).reshape(GRID, GRID)
    return np.rot90(arr, k=-1).tolist()


def _rotk(grid: list, k: int) -> list:
    result = [list(row) for row in grid]
    k = int(k) % 4
    for _ in range(k):
        result = rotcw_web(result)
    return result


def _connected(grid: list) -> bool:
    arr = np.asarray(grid, dtype=np.int8)
    filled = list(zip(*np.nonzero(arr)))
    if not filled:
        return False
    pixels = set(filled)
    start = pixels.pop()
    stack = [start]
    while stack:
        y, x = stack.pop()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if (ny, nx) in pixels:
                pixels.remove((ny, nx))
                stack.append((ny, nx))
    return not pixels


def _valid_glyph(grid: list) -> bool:
    if len(grid) != GRID or any(len(row) != GRID for row in grid):
        return False
    flat = [int(v) for row in grid for v in row]
    if any(v not in (0, 1) for v in flat):
        return False
    if not (7 <= sum(flat) <= 15):
        return False
    if not _connected(grid):
        return False
    arr = np.asarray(grid, dtype=np.int8)
    for k in (1, 2, 3):
        if np.array_equal(np.rot90(arr, k=k), arr):
            return False
    return True


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    while True:
        cells = [[rng.randint(0, 1) for _ in range(GRID)] for _ in range(GRID)]
        if _valid_glyph(cells):
            break
    rotations = [0, 1, 2, 3]
    rng.shuffle(rotations)
    return {
        "canvas_size": CANVAS,
        "glyph": cells,
        "rotations": rotations,
        "grid": GRID,
        "tag_names": list(TAG_NAMES),
    }


def _draw_glyph(draw, origin, cell, grid) -> None:
    x0, y0 = origin
    for r in range(GRID):
        for k in range(GRID):
            left = x0 + BORDER_W + k * cell
            top = y0 + BORDER_W + r * cell
            fill = INK if int(grid[r][k]) == 1 else CELL_LIGHT
            draw.rectangle(
                (left + 1, top + 1, left + cell - 1, top + cell - 1),
                fill=fill,
                outline=WHITE_LINE,
            )


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    glyph = [[int(v) for v in row] for row in scene["glyph"]]
    rotations = [int(v) for v in scene["rotations"]]
    image = Image.new("RGB", (size, size), BG)
    draw = ImageDraw.Draw(image)

    ref_cell = (REF_SIZE - 2 * BORDER_W) // GRID
    draw.rectangle(
        (REF_ORIGIN[0], REF_ORIGIN[1], REF_ORIGIN[0] + REF_SIZE, REF_ORIGIN[1] + REF_SIZE),
        fill=None,
        outline=REF_COLOR,
        width=BORDER_W,
    )
    _draw_glyph(draw, REF_ORIGIN, ref_cell, glyph)

    tile_cell = (TILE_SIZE - 2 * BORDER_W) // GRID
    for slot in range(N_TILES):
        x0 = TILE_X[slot]
        y0 = TILE_Y
        draw.rectangle(
            (x0, y0, x0 + TILE_SIZE, y0 + TILE_SIZE),
            fill=None,
            outline=TILE_COLOR,
            width=BORDER_W,
        )
        grid_rot = _rotk(glyph, rotations[slot])
        _draw_glyph(draw, (x0, y0), tile_cell, grid_rot)
        name = scene["tag_names"][slot]
        cx = x0 + TILE_SIZE // 2
        tag_cy = TILE_Y - 3 * TAG_HALF
        draw.rectangle(
            (cx - TAG_HALF, tag_cy - TAG_HALF, cx + TAG_HALF, tag_cy + TAG_HALF),
            fill=TAG_RGB[name],
        )
    return image


def _match_slot(scene: dict) -> int:
    rotations = [int(v) for v in scene["rotations"]]
    return rotations.index(1)


def analytic_gold(scene: dict) -> str:
    glyph = [[int(v) for v in row] for row in scene["glyph"]]
    slot = _match_slot(scene)
    return scene["tag_names"][slot]


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    glyph = scene["glyph"]
    rotations = [int(v) for v in scene["rotations"]]
    if len(rotations) != N_TILES or sorted(rotations) != [0, 1, 2, 3]:
        return True
    if not _valid_glyph(glyph):
        return True
    return False


def latent_symmetries(scene: dict) -> list:
    return []
