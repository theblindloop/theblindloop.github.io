"""Scene sampler, renderer, analytic gold, margin and latent symmetries for slide_blank_replay.

A 3x3 board of square cells is drawn once. Eight cells hold coloured tiles, one
cell is a light empty gap, and one cell is marked with a red ring. To the right,
an ordered row of cream cards is drawn; each card carries a solid black triangle
pointing up, down, left or right. A card's arrow says which way the empty gap
moves: the gap exchanges places with the neighbouring cell in that direction, so
the tile in that neighbouring cell slides into the gap. The board after the
cards have been applied is never drawn, so the answer is a consequence of
replaying the drawn program, not a measurement of any drawn configuration.

Tiles and cards are stored as unordered bags. A tile's drawn position comes from
its own row and column, a card's drawn position from its own ``slot``, and the
marked cell is recovered from the drawn ring geometry, so gold never depends on
Python list order; ``latent_symmetries`` exercises exactly that.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "slide_blank_replay-renderer-0.1.0"

WHITE = (255, 255, 255)
GRID_EDGE = (90, 90, 90)
EMPTY_FILL = (236, 236, 236)
RED = (214, 38, 38)
CARD_BG = (252, 246, 228)
CARD_EDGE = (176, 168, 140)
ARROW_BLACK = (15, 15, 15)
ORDER_GREY = (120, 120, 120)

PALETTE: dict[str, tuple[int, int, int]] = {
    "blue": (36, 72, 204),
    "brown": (140, 84, 40),
    "green": (26, 150, 72),
    "orange": (240, 140, 20),
    "pink": (240, 130, 180),
    "purple": (128, 48, 178),
    "teal": (24, 168, 168),
    "yellow": (238, 214, 44),
}
COLOR_NAMES = tuple(sorted(PALETTE))
EMPTY_NAME = "empty"

N = 3
CELL = 88
GAP = 10
PITCH = CELL + GAP  # 98
ORIGIN = (44, 66)

RING_R = 32
RING_W = 5

N_MOVES = 6
CARD = 74
CGAP = 16
CARD_PITCH = CARD + CGAP  # 90
CARD_ORIGIN = (392, 140)
TRI_H = 54
TRI_B = 48

CANVAS = (960, 420)
QUARANTINE_MARGIN = 6.0

DIRS = ("up", "down", "left", "right")
_STEP = {"up": (-1, 0), "down": (1, 0), "left": (0, -1), "right": (0, 1)}
_OPPOSITE = {"up": "down", "down": "up", "left": "right", "right": "left"}

# clearance of the drawn arrow-direction reading: the centroid of an isoceles
# triangle sits TRI_H/6 px off its bounding-box centre along the axis it points
# on, and exactly 0 px off it on the perpendicular axis.
ARROW_MARGIN = TRI_H / 6.0


# --------------------------------------------------------------------------- geometry


def _cell_center(r: int, c: int) -> tuple[float, float]:
    return (
        ORIGIN[0] + c * PITCH + CELL / 2.0,
        ORIGIN[1] + r * PITCH + CELL / 2.0,
    )


def _ring_center(scene: dict) -> tuple[float, float]:
    ring = scene["ring"]
    cx, cy = _cell_center(int(ring["r"]), int(ring["c"]))
    return cx + float(ring["dx"]), cy + float(ring["dy"])


def _ring_cell(scene: dict) -> tuple[int, int]:
    """The marked cell is the drawn cell whose centre is nearest the ring centre."""
    rx, ry = _ring_center(scene)
    best = None
    for r in range(N):
        for c in range(N):
            cx, cy = _cell_center(r, c)
            d = math.hypot(rx - cx, ry - cy)
            if best is None or d < best[0]:
                best = (d, r, c)
    return best[1], best[2]


def _board(scene: dict) -> dict[tuple[int, int], str]:
    """The drawn board: cell -> colour name, with the empty cell absent."""
    board: dict[tuple[int, int], str] = {}
    for tile in scene["tiles"]:
        board[(int(tile["r"]), int(tile["c"]))] = str(tile["color"])
    return board


def _blank_start(scene: dict) -> tuple[int, int]:
    board = _board(scene)
    missing = [(r, c) for r in range(N) for c in range(N) if (r, c) not in board]
    if len(missing) != 1:
        raise ValueError("a scene must draw exactly one empty cell")
    return missing[0]


def _moves_in_drawn_order(scene: dict) -> list[dict]:
    return sorted(scene["moves"], key=lambda m: int(m["slot"]))


def _initial_content(scene: dict, r: int, c: int) -> str:
    return _board(scene).get((r, c), EMPTY_NAME)


def _replay(scene: dict) -> tuple[dict[tuple[int, int], str], tuple[int, int]]:
    """Slide the gap through the drawn cards in drawn order."""
    board = _board(scene)
    blank = _blank_start(scene)
    for move in _moves_in_drawn_order(scene):
        dr, dc = _STEP[str(move["dir"])]
        target = (blank[0] + dr, blank[1] + dc)
        if not (0 <= target[0] < N and 0 <= target[1] < N):
            raise ValueError("scene draws an illegal move; sampling must not emit one")
        board[blank] = board.pop(target)
        blank = target
    return board, blank


def _cell_index(r: int, c: int) -> int:
    """Cells are numbered 1..9 in reading order, exactly as the question states."""
    return N * r + c + 1


# --------------------------------------------------------------------------- gold


def analytic_gold(scene: dict) -> str:
    """``"<content of the ringed cell>|<number of the cell holding the gap>"``."""
    mr, mc = _ring_cell(scene)
    board, blank = _replay(scene)
    content = board.get((mr, mc), EMPTY_NAME)
    return f"{content}|{_cell_index(*blank)}"


def margin(scene: dict) -> float:
    """Smallest drawn clearance any reading the answer depends on has to cross."""
    rx, ry = _ring_center(scene)
    dists = sorted(
        math.hypot(rx - _cell_center(r, c)[0], ry - _cell_center(r, c)[1])
        for r in range(N)
        for c in range(N)
    )
    assign = (dists[1] - dists[0]) / 2.0

    mr, mc = _ring_cell(scene)
    cx, cy = _cell_center(mr, mc)
    off = math.hypot(rx - cx, ry - cy)
    stroke = abs(off - RING_R) - RING_W / 2.0

    return round(min(assign, stroke, ARROW_MARGIN), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Re-orderings of the unordered bags: pixel-identical and answer preserving."""
    out: list[tuple[str, dict]] = []
    tiles = list(scene["tiles"])
    moves = list(scene["moves"])
    out.append(("tiles_bag_reversed", dict(scene, tiles=list(reversed(tiles)))))
    out.append(("tiles_bag_rotated", dict(scene, tiles=tiles[3:] + tiles[:3])))
    out.append(("moves_bag_reversed", dict(scene, moves=list(reversed(moves)))))
    out.append(("moves_bag_rotated", dict(scene, moves=moves[2:] + moves[:2])))
    return out


# --------------------------------------------------------------------------- sampling


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    tight = rng.random() < 0.18

    cells = [(r, c) for r in range(N) for c in range(N)]
    blank0 = cells[rng.randrange(len(cells))]
    names = list(COLOR_NAMES)
    rng.shuffle(names)
    tiles = [
        {"r": r, "c": c, "color": name}
        for (r, c), name in zip([x for x in cells if x != blank0], names)
    ]

    # a legal random walk of the gap; every drawn card is a legal move
    moves: list[dict] = []
    blank = blank0
    visited = {blank0}
    prev: str | None = None
    for slot in range(N_MOVES):
        opts = [
            d
            for d in DIRS
            if 0 <= blank[0] + _STEP[d][0] < N and 0 <= blank[1] + _STEP[d][1] < N
        ]
        if prev is not None and len(opts) > 1:
            forward = [d for d in opts if d != _OPPOSITE[prev]]
            if forward:
                opts = forward
        d = opts[rng.randrange(len(opts))]
        moves.append({"slot": slot, "dir": d})
        blank = (blank[0] + _STEP[d][0], blank[1] + _STEP[d][1])
        visited.add(blank)
        prev = d

    # The ringed cell's content changes during the replay exactly when the gap
    # visits it, so the drawn starting content of that cell is held to roughly
    # chance level as a predictor of the answer.
    touched = rng.random() < 0.85
    pool = [x for x in cells if (x in visited) is touched] or sorted(visited)
    rr, rc = pool[rng.randrange(len(pool))]

    if tight:
        legal = [d for d in DIRS if 0 <= rr + _STEP[d][0] < N and 0 <= rc + _STEP[d][1] < N]
        d = legal[rng.randrange(len(legal))]
        t = rng.uniform(43.0, 47.0)
        dx = _STEP[d][1] * t
        dy = _STEP[d][0] * t
    else:
        ang = rng.uniform(0.0, 2.0 * math.pi)
        rad = rng.uniform(0.0, 20.0)
        dx = rad * math.cos(ang)
        dy = rad * math.sin(ang)

    rng.shuffle(tiles)
    rng.shuffle(moves)
    return {
        "canvas": list(CANVAS),
        "origin": list(ORIGIN),
        "cards_origin": list(CARD_ORIGIN),
        "n_moves": N_MOVES,
        "tiles": tiles,
        "ring": {"r": rr, "c": rc, "dx": round(dx, 4), "dy": round(dy, 4)},
        "moves": moves,
    }


# --------------------------------------------------------------------------- rendering


def _triangle(ccx: float, ccy: float, direction: str) -> list[tuple[float, float]]:
    h2 = TRI_H / 2.0
    b2 = TRI_B / 2.0
    if direction == "up":
        return [(ccx, ccy - h2), (ccx - b2, ccy + h2), (ccx + b2, ccy + h2)]
    if direction == "down":
        return [(ccx, ccy + h2), (ccx - b2, ccy - h2), (ccx + b2, ccy - h2)]
    if direction == "left":
        return [(ccx - h2, ccy), (ccx + h2, ccy - b2), (ccx + h2, ccy + b2)]
    return [(ccx + h2, ccy), (ccx - h2, ccy - b2), (ccx - h2, ccy + b2)]


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(CANVAS), WHITE)
    d = ImageDraw.Draw(img)

    board = _board(scene)
    for r in range(N):
        for c in range(N):
            x0 = ORIGIN[0] + c * PITCH
            y0 = ORIGIN[1] + r * PITCH
            name = board.get((r, c))
            fill = PALETTE[name] if name is not None else EMPTY_FILL
            d.rectangle([x0, y0, x0 + CELL, y0 + CELL], fill=fill, outline=GRID_EDGE, width=3)

    rx, ry = _ring_center(scene)
    d.ellipse([rx - RING_R, ry - RING_R, rx + RING_R, ry + RING_R], outline=RED, width=RING_W)

    for move in _moves_in_drawn_order(scene):
        slot = int(move["slot"])
        x0 = CARD_ORIGIN[0] + slot * CARD_PITCH
        y0 = CARD_ORIGIN[1]
        d.rectangle([x0, y0, x0 + CARD, y0 + CARD], fill=CARD_BG, outline=CARD_EDGE, width=2)
        d.polygon(_triangle(x0 + CARD / 2.0, y0 + CARD / 2.0, str(move["dir"])), fill=ARROW_BLACK)

    ax0 = CARD_ORIGIN[0]
    ax1 = CARD_ORIGIN[0] + (N_MOVES - 1) * CARD_PITCH + CARD
    ay = CARD_ORIGIN[1] + CARD + 34
    d.line([ax0, ay, ax1 - 16, ay], fill=ORDER_GREY, width=4)
    d.polygon([(ax1, ay), (ax1 - 18, ay - 10), (ax1 - 18, ay + 10)], fill=ORDER_GREY)
    return img
