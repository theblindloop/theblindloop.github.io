from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

EXPECTED_TILES = 112

def _components(cells: set[tuple[int, int]]) -> list[set[tuple[int, int]]]:
    unseen = set(cells)
    out = []
    while unseen:
        start = min(unseen)
        part = {start}
        unseen.remove(start)
        q = deque([start])
        while q:
            x, y = q.popleft()
            for p in ((x-1,y),(x+1,y),(x,y-1),(x,y+1)):
                if p in unseen:
                    unseen.remove(p); part.add(p); q.append(p)
        out.append(part)
    return out

def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    colors, inverse, counts = np.unique(arr.reshape(-1, 3), axis=0, return_inverse=True, return_counts=True)
    if len(colors) != 3:
        raise ValueError("expected exactly background, land, and marker colors")
    order = np.argsort(counts)
    marker_index, land_index, background_index = map(int, order)
    if counts[background_index] <= counts[land_index] or counts[land_index] <= 20 * counts[marker_index]:
        raise ValueError("color areas do not match one marker in a broad landmass")
    labels = inverse.reshape(arr.shape[:2])
    marker_mask = labels == marker_index
    land_mask = labels == land_index
    ys, xs = np.nonzero(marker_mask)
    if not len(xs):
        raise ValueError("marker absent")
    x0, x1, y0, y1 = int(xs.min()), int(xs.max()), int(ys.min()), int(ys.max())
    side = x1 - x0 + 1
    if side < 20 or side > 40 or y1 - y0 + 1 != side or int(marker_mask.sum()) != side * side:
        raise ValueError("marker is not one intact square tile")
    foreground = marker_mask | land_mask
    cx, cy = x0 + side // 2, y0 + side // 2
    cells: set[tuple[int, int]] = set()
    for gy in range(-20, 21):
        py = cy + gy * side
        if not (0 <= py < arr.shape[0]):
            continue
        for gx in range(-20, 21):
            px = cx + gx * side
            if 0 <= px < arr.shape[1] and foreground[py, px]:
                cells.add((gx, gy))
    if len(cells) != EXPECTED_TILES or (0, 0) not in cells:
        raise ValueError(f"expected {EXPECTED_TILES} complete tiles")
    reconstructed = np.zeros_like(foreground)
    for gx, gy in cells:
        tx0 = x0 + gx * side
        ty0 = y0 + gy * side
        if tx0 < 0 or ty0 < 0 or tx0 + side > arr.shape[1] or ty0 + side > arr.shape[0]:
            raise ValueError("tile exceeds canvas")
        reconstructed[ty0:ty0+side, tx0:tx0+side] = True
    if not np.array_equal(reconstructed, foreground):
        raise ValueError("landmass is not an intact equal-tile raster")
    before = _components(cells)
    after = _components(cells - {(0, 0)})
    if len(before) != 1 or len(after) not in (1, 2) or min(map(len, after)) < 20:
        raise ValueError("landmass violates the declared connectivity boundary")
    return "stay" if len(after) == 1 else "split"
