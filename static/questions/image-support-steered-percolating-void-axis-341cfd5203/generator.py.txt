from __future__ import annotations

import random
from collections import deque
from PIL import Image, ImageDraw

SIZE = 512
GRID = 13
TOTAL = 112
PALETTES = [
    ("#2867b2", "#f2b43c", "#f7f4ec"),
    ("#237a70", "#f0ad36", "#f6f1e8"),
    ("#6046a8", "#efb637", "#f8f3e9"),
]

def _components(cells: set[tuple[int, int]]) -> list[set[tuple[int, int]]]:
    unseen = set(cells)
    out = []
    while unseen:
        start = min(unseen)
        part = {start}
        unseen.remove(start)
        q = deque([start])
        while q:
            x, y = q.popleft()
            for p in ((x-1,y),(x+1,y),(x,y-1),(x,y+1)):
                if p in unseen:
                    unseen.remove(p); part.add(p); q.append(p)
        out.append(part)
    return out

def _transform(p: tuple[int, int], turn: int, mirror: bool) -> tuple[int, int]:
    x, y = p
    if mirror:
        x = GRID - 1 - x
    for _ in range(turn):
        x, y = GRID - 1 - y, x
    return x, y

def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    should_stay = (int(seed) % 2 == 0)
    cells = {(x, y) for x in range(1, 6) for y in range(1, 12)}
    cells |= {(x, y) for x in range(7, 12) for y in range(1, 12)}
    marker = (6, 6)
    cells.add(marker)
    # Equalize count, span, and bounding box while changing only a distant bypass.
    cells.add((3, 0))
    cells.add((6, 1) if should_stay else (4, 0))
    cells.remove((3, 9))
    turn = rng.randrange(4)
    mirror = bool(rng.randrange(2))
    cells = {_transform(p, turn, mirror) for p in cells}
    marker = _transform(marker, turn, mirror)
    ink, marker_ink, background = PALETTES[rng.randrange(len(PALETTES))]
    cell_size = rng.choice((28, 29, 30, 31))
    footprint = GRID * cell_size
    jitter = max(0, (SIZE - footprint) // 2 - 8)
    origin = [
        (SIZE - footprint) // 2 + rng.randint(-jitter, jitter),
        (SIZE - footprint) // 2 + rng.randint(-jitter, jitter),
    ]
    scene = {
        "occupied": [[x, y] for x, y in sorted(cells)],
        "marker": [marker[0], marker[1]],
        "ink": ink,
        "marker_ink": marker_ink,
        "background": background,
        "origin": origin,
        "cell_size": cell_size,
    }
    assert len(cells) == TOTAL and len(_components(cells)) == 1
    return scene

def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (SIZE, SIZE), scene["background"])
    draw = ImageDraw.Draw(image)
    ox, oy = scene["origin"]
    side = int(scene["cell_size"])
    marker = tuple(scene["marker"])
    for raw in scene["occupied"]:
        x, y = raw
        color = scene["marker_ink"] if (x, y) == marker else scene["ink"]
        draw.rectangle((ox+x*side, oy+y*side, ox+(x+1)*side-1, oy+(y+1)*side-1), fill=color)
    return image

def analytic_gold(scene: dict) -> str:
    cells = {tuple(p) for p in scene["occupied"]}
    cells.discard(tuple(scene["marker"]))
    return "stay" if len(_components(cells)) == 1 else "split"

def margin(scene: dict) -> float:
    cells = {tuple(p) for p in scene["occupied"]}
    cells.discard(tuple(scene["marker"]))
    parts = sorted((len(p) for p in _components(cells)), reverse=True)
    return float(parts[-1] if len(parts) > 1 else 1.0)

def is_quarantined(scene: dict) -> bool:
    cells = {tuple(p) for p in scene["occupied"]}
    marker = tuple(scene["marker"])
    before = _components(cells)
    after = _components(cells - {marker})
    return not (len(cells) == TOTAL and marker in cells and len(before) == 1 and len(after) in (1, 2) and min(map(len, after)) >= 20)

def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["occupied"] = list(reversed(scene["occupied"]))
    return [("reverse_cell_serialization", twin)]
