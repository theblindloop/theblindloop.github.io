"""Independent PNG-only inverse arm for counterfactual unmasking scenes."""

from __future__ import annotations

import numpy as np
from PIL import Image

WIDTH, HEIGHT = 960, 760
INK, OFF, RED = (np.array(c, dtype=np.float64) for c in
                 ((35, 40, 47), (255, 255, 255), (205, 54, 54)))
GATE_COLORS = {
    "and": np.array((48, 112, 190), dtype=np.float64),
    "or": np.array((224, 148, 39), dtype=np.float64),
    "xor": np.array((132, 76, 183), dtype=np.float64),
}
SWITCH_COLORS = tuple(np.array(color, dtype=np.float64) for color in
                      ((0, 143, 163), (43, 139, 72), (205, 91, 27), (179, 54, 132)))
ROW_Y, TOP_X0, CELL_STEP = (134, 229, 324, 419, 514), 84, 60


class OracleAbstention(ValueError):
    pass


def _patch(arr: np.ndarray, x: int, y: int, radius: int = 4) -> np.ndarray:
    pixels = arr[y - radius:y + radius + 1, x - radius:x + radius + 1]
    return np.median(pixels.reshape(-1, 3), axis=0)


def _binary(arr: np.ndarray, x: int, y: int) -> int:
    rgb = _patch(arr, x, y)
    d_on, d_off = float(np.linalg.norm(rgb - INK)), float(np.linalg.norm(rgb - OFF))
    if min(d_on, d_off) > 35 or abs(d_on - d_off) < 60:
        raise OracleAbstention(f"unclear binary cell at {(x, y)}: {rgb.tolist()}")
    return int(d_on < d_off)


def _gate(arr: np.ndarray, y: int) -> str:
    rgb = _patch(arr, 662, y, 5)
    distances = {kind: float(np.linalg.norm(rgb - color)) for kind, color in GATE_COLORS.items()}
    kind = min(distances, key=distances.get)  # type: ignore[arg-type]
    if distances[kind] > 45:
        raise OracleAbstention(f"unclear row rule at y={y}: {rgb.tolist()}")
    return kind


def _annulus(arr: np.ndarray, x: int, y: int) -> np.ndarray:
    crop = arr[y - 31:y + 32, x - 31:x + 32].astype(np.float64)
    yy, xx = np.ogrid[-31:32, -31:32]
    mask = ((xx * xx + yy * yy) >= 21 * 21) & ((xx * xx + yy * yy) <= 30 * 30)
    return crop[mask]


def _color_count(pixels: np.ndarray, color: np.ndarray, tolerance: float = 58.0) -> int:
    return int(np.count_nonzero(np.linalg.norm(pixels - color, axis=1) < tolerance))


def _has_source_ring(arr: np.ndarray, x: int, y: int) -> bool:
    return _color_count(_annulus(arr, x, y), RED) >= 100


def _switch_color_at(arr: np.ndarray, x: int, y: int) -> int | None:
    pixels = _annulus(arr, x, y)
    counts = [_color_count(pixels, color) for color in SWITCH_COLORS]
    best, ordered = max(range(4), key=counts.__getitem__), sorted(counts, reverse=True)
    if counts[best] < 100:
        return None
    if ordered[0] - ordered[1] < 50:
        raise OracleAbstention(f"ambiguous switch ring at {(x, y)}: {counts}")
    return best


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise OracleAbstention("unknown gate")


def _reduce(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[i], row[i + 1]) for i in range(len(row) - 1)]


def _run(top: list[int], gates: list[str], switch: tuple[int, int] | None = None,
         forced_value: int | None = None) -> list[list[int]]:
    rows = [list(top)]
    for row_index, kind in enumerate(gates, 1):
        row = _reduce(rows[-1], kind)
        if switch is not None and row_index == switch[0]:
            if forced_value not in (0, 1):
                raise OracleAbstention("missing forced value")
            row[switch[1]] = int(forced_value)
        rows.append(row)
    return rows


def decision_from_image(image: Image.Image) -> str:
    raster = image.convert("RGB")
    if raster.size != (WIDTH, HEIGHT):
        raise OracleAbstention(f"unexpected canvas {raster.size}")
    arr = np.asarray(raster)
    visible_rows: list[list[int]] = []
    source_marks: list[bool] = []
    switches: list[tuple[int, int] | None] = [None] * 4
    for row_index, y in enumerate(ROW_Y):
        x0 = TOP_X0 + row_index * (CELL_STEP // 2)
        row = []
        for cell_index in range(9 - row_index):
            x = x0 + cell_index * CELL_STEP
            row.append(_binary(arr, x, y))
            if row_index == 0:
                source_marks.append(_has_source_ring(arr, x, y))
            elif row_index <= 3:
                index = _switch_color_at(arr, x, y)
                if index is not None:
                    if switches[index] is not None:
                        raise OracleAbstention(f"duplicate switch-{index + 1} marker")
                    switches[index] = (row_index, cell_index)
        visible_rows.append(row)
    if sum(source_marks) != 1:
        raise OracleAbstention(f"found {sum(source_marks)} source rings")
    marked = source_marks.index(True)
    if not 1 <= marked <= 7 or any(item is None for item in switches):
        raise OracleAbstention(f"invalid source or missing switches: {switches}")
    gates = [_gate(arr, ROW_Y[i]) for i in range(1, 5)]
    factual_rows = _run(visible_rows[0], gates)
    if factual_rows != visible_rows:
        raise OracleAbstention("visible factual lattice contradicts its row rules")
    source_top = list(visible_rows[0])
    source_top[marked] ^= 1
    source_rows = _run(source_top, gates)
    if source_rows[-1] != factual_rows[-1]:
        raise OracleAbstention("the unconditioned source flip is not masked")
    outcomes = []
    for item in switches:
        assert item is not None
        row, cell = item
        if source_rows[row][cell] != factual_rows[row][cell]:
            raise OracleAbstention("a switch is not locally stable across the two trials")
        forced = 1 - factual_rows[row][cell]
        left = _run(visible_rows[0], gates, item, forced)[-1]
        right = _run(source_top, gates, item, forced)[-1]
        if left == factual_rows[-1] or right == factual_rows[-1]:
            raise OracleAbstention("a switch does not visibly alter both trial bottoms")
        outcomes.append((left, right))
    matches = [i for i, (left, right) in enumerate(outcomes) if left != right]
    if len(matches) != 1:
        raise OracleAbstention(f"expected one unmasking switch, recovered {matches}")
    if sum(a != b for a, b in zip(*outcomes[matches[0]], strict=True)) < 2:
        raise OracleAbstention("unmasked effect lacks the required two-cell gap")
    return f"switch-{matches[0] + 1}"
