"""Deterministic renderer for paired-world counterfactual unmasking."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 960, 760, 2
BACKGROUND, PANEL_FILL = (246, 245, 240), (255, 255, 255)
INK, OFF, RED, WIRE = (35, 40, 47), (255, 255, 255), (205, 54, 54), (158, 164, 172)
GATE_COLORS = {"and": (48, 112, 190), "or": (224, 148, 39), "xor": (132, 76, 183)}
GATE_LABELS = {"and": "ALL", "or": "ANY", "xor": "ONE"}
SWITCH_COLORS = ((0, 143, 163), (43, 139, 72), (205, 91, 27), (179, 54, 132))
ROW_Y = (134, 229, 324, 419, 514)
TOP_X0, CELL_STEP, CELL_RADIUS = 84, 60, 16
GATE_X0, GATE_X1 = 650, 758
SWITCH_BOXES = ((38, 626, 238, 732), (266, 626, 466, 732),
                (494, 626, 694, 732), (722, 626, 922, 732))


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise ValueError(f"unknown gate {kind!r}")


def _next_row(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[i], row[i + 1]) for i in range(len(row) - 1)]


def _run(top: list[int], gates: list[str], switch: tuple[int, int] | None = None,
         forced_value: int | None = None) -> list[list[int]]:
    rows = [list(top)]
    for row_index, kind in enumerate(gates, 1):
        row = _next_row(rows[-1], kind)
        if switch is not None and row_index == switch[0]:
            if forced_value not in (0, 1):
                raise ValueError("a switch needs a binary forced value")
            row[switch[1]] = int(forced_value)
        rows.append(row)
    return rows


def _intervened_top(top: list[int], marked: int) -> list[int]:
    changed = list(top)
    changed[marked] ^= 1
    return changed


def _switch_outcomes(top: list[int], gates: list[str], marked: int,
                     switch: tuple[int, int]) -> tuple[list[int], list[int]]:
    factual_rows = _run(top, gates)
    forced = 1 - factual_rows[switch[0]][switch[1]]
    baseline = _run(top, gates, switch, forced)[-1]
    source_flipped = _run(_intervened_top(top, marked), gates, switch, forced)[-1]
    return baseline, source_flipped


def _candidate_sets(top: list[int], gates: list[str], marked: int) -> tuple[list[tuple], list[tuple]]:
    factual_rows = _run(top, gates)
    source_rows = _run(_intervened_top(top, marked), gates)
    unmaskers, masked = [], []
    for row_index in range(1, 4):
        for cell_index in range(9 - row_index):
            # Every switch changes the same locally stable cell in both trials.
            if source_rows[row_index][cell_index] != factual_rows[row_index][cell_index]:
                continue
            coordinate = (row_index, cell_index)
            left, right = _switch_outcomes(top, gates, marked, coordinate)
            # Every forced switch visibly affects the bottom in both trials.
            if left == factual_rows[-1] or right == factual_rows[-1]:
                continue
            distance = sum(a != b for a, b in zip(left, right, strict=True))
            item = (coordinate, tuple(left), tuple(right), distance)
            (unmaskers if distance else masked).append(item)
    return unmaskers, masked


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    for _ in range(50000):
        top = [rng.randrange(2) for _ in range(9)]
        gates = [rng.choice(("and", "or", "xor")) for _ in range(4)]
        if len(set(gates)) < 3:
            continue
        marked = rng.randrange(1, 8)
        factual_rows = _run(top, gates)
        source_rows = _run(_intervened_top(top, marked), gates)
        internal_changes = sum(
            a != b
            for row_a, row_b in zip(factual_rows[1:-1], source_rows[1:-1], strict=True)
            for a, b in zip(row_a, row_b, strict=True)
        )
        if source_rows[-1] != factual_rows[-1] or internal_changes < 2:
            continue
        unmaskers, masked = _candidate_sets(top, gates, marked)
        strong = [item for item in unmaskers if item[3] >= 2]
        rng.shuffle(strong)
        rng.shuffle(masked)
        selected = None
        for unmasker in strong:
            for wrongs in itertools.combinations(masked, 3):
                if {unmasker[0][0], *(item[0][0] for item in wrongs)} != {1, 2, 3}:
                    continue
                if len({item[1] for item in wrongs}) < 2:
                    continue
                selected = (unmasker, wrongs)
                break
            if selected is not None:
                break
        if selected is None:
            continue
        unmasker, wrongs = selected
        wrong_list = list(wrongs)
        rng.shuffle(wrong_list)
        switches, cursor = [], 0
        for slot in range(4):
            coordinate = unmasker[0] if slot == target_slot else wrong_list[cursor][0]
            if slot != target_slot:
                cursor += 1
            switches.append([coordinate[0], coordinate[1]])
        return {"top": top, "gates": gates, "marked": marked, "switches": switches,
                "switch_draw_order": [0, 1, 2, 3]}
    raise RuntimeError("could not sample a separated counterfactual unmasking scene")


def _all_switch_outcomes(scene: dict) -> list[tuple[list[int], list[int]]]:
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    marked = int(scene["marked"])
    return [_switch_outcomes(top, gates, marked, (int(item[0]), int(item[1])))
            for item in scene["switches"]]


def analytic_gold(scene: dict) -> str:
    matches = [i for i, (baseline, source) in enumerate(_all_switch_outcomes(scene))
               if baseline != source]
    if len(matches) != 1:
        raise ValueError(f"expected one unmasking switch, got {matches}")
    return f"switch-{matches[0] + 1}"


def margin(scene: dict) -> float:
    correct = int(analytic_gold(scene).split("-")[1]) - 1
    baseline, source = _all_switch_outcomes(scene)[correct]
    return float(sum(a != b for a, b in zip(baseline, source, strict=True)))


def is_quarantined(scene: dict) -> bool:
    try:
        top = [int(value) for value in scene["top"]]
        gates = [str(value) for value in scene["gates"]]
        switches, marked = scene["switches"], int(scene["marked"])
        if len(top) != 9 or any(value not in (0, 1) for value in top):
            return True
        if len(gates) != 4 or any(value not in GATE_COLORS for value in gates):
            return True
        if len(set(gates)) < 3 or not 1 <= marked <= 7:
            return True
        if len(switches) != 4 or any(len(item) != 2 for item in switches):
            return True
        coordinates = [(int(item[0]), int(item[1])) for item in switches]
        if len(set(coordinates)) != 4 or {row for row, _ in coordinates} != {1, 2, 3}:
            return True
        if any(not (1 <= row <= 3 and 0 <= cell < 9 - row) for row, cell in coordinates):
            return True
        if sorted(scene["switch_draw_order"]) != [0, 1, 2, 3]:
            return True
        factual_rows = _run(top, gates)
        source_rows = _run(_intervened_top(top, marked), gates)
        if source_rows[-1] != factual_rows[-1]:
            return True
        internal_changes = sum(
            a != b
            for row_a, row_b in zip(factual_rows[1:-1], source_rows[1:-1], strict=True)
            for a, b in zip(row_a, row_b, strict=True)
        )
        if internal_changes < 2:
            return True
        if any(source_rows[row][cell] != factual_rows[row][cell] for row, cell in coordinates):
            return True
        outcomes = _all_switch_outcomes(scene)
        if any(left == factual_rows[-1] or right == factual_rows[-1] for left, right in outcomes):
            return True
        matches = [i for i, pair in enumerate(outcomes) if pair[0] != pair[1]]
        if len(matches) != 1 or margin(scene) < 2.0:
            return True
        wrong = [tuple(left) for i, (left, _right) in enumerate(outcomes) if i != matches[0]]
        return len(set(wrong)) < 2
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["switch_draw_order"] = list(reversed(twin["switch_draw_order"]))
    return [("reverse_disjoint_switch_legend_draw_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _pt(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _row_centers(row_index: int) -> list[tuple[int, int]]:
    x0 = TOP_X0 + row_index * (CELL_STEP // 2)
    return [(x0 + i * CELL_STEP, ROW_Y[row_index]) for i in range(9 - row_index)]


def _draw_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int) -> None:
    x, y = center
    draw.ellipse(_box((x - CELL_RADIUS, y - CELL_RADIUS, x + CELL_RADIUS, y + CELL_RADIUS)),
                 fill=INK if value else OFF, outline=INK, width=3 * SCALE)


def _draw_switch_marker(draw: ImageDraw.ImageDraw, center: tuple[int, int], index: int) -> None:
    x, y = center
    color = SWITCH_COLORS[index]
    draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=color, width=6 * SCALE)
    draw.rounded_rectangle(_box((x - 13, y + 27, x + 13, y + 50)), radius=6 * SCALE,
                           fill=color, outline=INK, width=2 * SCALE)
    font, label = _font(15, True), str(index + 1)
    bbox = draw.textbbox((0, 0), label, font=font)
    draw.text((x * SCALE - (bbox[2] - bbox[0]) // 2,
               (y + 37) * SCALE - (bbox[3] - bbox[1]) // 2 - SCALE),
              label, fill=(255, 255, 255), font=font)


def _draw_switch_box(draw: ImageDraw.ImageDraw, slot: int) -> None:
    x0, y0, x1, y1 = SWITCH_BOXES[slot]
    color = SWITCH_COLORS[slot]
    draw.rounded_rectangle(_box((x0, y0, x1, y1)), radius=13 * SCALE,
                           fill=PANEL_FILL, outline=color, width=5 * SCALE)
    for text, y, font, fill in ((f"SWITCH {slot + 1}", y0 + 24, _font(20, True), INK),
                                ("FORCE OPPOSITE", y0 + 63, _font(12, True), color)):
        bbox = draw.textbbox((0, 0), text, font=font)
        draw.text(((x0 + x1) * SCALE // 2 - (bbox[2] - bbox[0]) // 2, y * SCALE),
                  text, fill=fill, font=font)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text(_pt((28, 16)), "COMPARE TWO FRESH TRIALS PER SWITCH", fill=INK, font=_font(25, True))
    draw.text(_pt((30, 53)),
              "A: shown TOP   •   B: flip red TOP   •   force the same opposite value",
              fill=INK, font=_font(17))
    draw.text(_pt((794, 22)), "dark = ON", fill=INK, font=_font(16, True))
    top, gates = [int(v) for v in scene["top"]], [str(v) for v in scene["gates"]]
    rows, centers = _run(top, gates), [_row_centers(i) for i in range(5)]
    for row_index in range(1, 5):
        for i, child in enumerate(centers[row_index]):
            for parent in (centers[row_index - 1][i], centers[row_index - 1][i + 1]):
                draw.line([_pt(parent), _pt(child)], fill=WIRE, width=3 * SCALE)
    draw.text(_pt((17, ROW_Y[0] - 10)), "TOP", fill=INK, font=_font(14, True))
    for row_index, row in enumerate(rows):
        for i, value in enumerate(row):
            center = centers[row_index][i]
            if row_index == 0 and i == int(scene["marked"]):
                x, y = center
                draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=RED, width=6 * SCALE)
            _draw_cell(draw, center, int(value))
        if row_index:
            kind, y = gates[row_index - 1], ROW_Y[row_index]
            draw.rounded_rectangle(_box((GATE_X0, y - 23, GATE_X1, y + 23)), radius=9 * SCALE,
                                   fill=GATE_COLORS[kind], outline=INK, width=2 * SCALE)
            label, font = GATE_LABELS[kind], _font(17, True)
            bbox = draw.textbbox((0, 0), label, font=font)
            draw.text((704 * SCALE - (bbox[2] - bbox[0]) // 2,
                       y * SCALE - (bbox[3] - bbox[1]) // 2 - SCALE),
                      label, fill=(255, 255, 255), font=font)
            row_label = "FACTUAL BOTTOM" if row_index == 4 else f"ROW {row_index}"
            draw.text(_pt((776, y - 10)), row_label, fill=INK, font=_font(14, True))
    for index, item in enumerate(scene["switches"]):
        row, cell = int(item[0]), int(item[1])
        _draw_switch_marker(draw, centers[row][cell], index)
    draw.text(_pt((28, 583)), "WHICH SWITCH MAKES A AND B END DIFFERENT?", fill=INK,
              font=_font(20, True))
    for slot in scene["switch_draw_order"]:
        _draw_switch_box(draw, int(slot))
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
