"""Image-only inverse arm for the cyclic-flow world.

This module intentionally knows only the public raster design: a red center,
three light shafts, and three darker triangular heads. It imports no world or
gold implementation and accepts no latent scene data.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[tuple[int, float, float, int, int]]:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    results: list[tuple[int, float, float, int, int]] = []
    for y0, x0 in np.argwhere(mask):
        y0 = int(y0)
        x0 = int(x0)
        if visited[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        visited[y0, x0] = True
        count = 0
        sum_x = 0.0
        sum_y = 0.0
        min_x = max_x = x0
        min_y = max_y = y0
        while queue:
            y, x = queue.popleft()
            count += 1
            sum_x += x
            sum_y += y
            min_x = min(min_x, x)
            max_x = max(max_x, x)
            min_y = min(min_y, y)
            max_y = max(max_y, y)
            for ny in range(max(0, y - 1), min(height, y + 2)):
                for nx in range(max(0, x - 1), min(width, x + 2)):
                    if mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        results.append((count, sum_x / count, sum_y / count, max_x - min_x + 1, max_y - min_y + 1))
    return results


def _distance_sq(left: tuple[float, float], right: tuple[float, float]) -> float:
    return (left[0] - right[0]) ** 2 + (left[1] - right[1]) ** 2


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.float64)
    red = rgb[:, :, 0]
    green = rgb[:, :, 1]
    blue = rgb[:, :, 2]
    maximum = rgb.max(axis=2)
    minimum = rgb.min(axis=2)
    chroma = maximum - minimum
    luminance = 0.2126 * red + 0.7152 * green + 0.0722 * blue

    center_mask = (red > 165) & (red > 1.55 * green) & (red > 1.35 * blue)
    height, width = center_mask.shape
    center_parts = [
        part
        for part in _components(center_mask)
        if 150 <= part[0] <= 600
        and 0.25 * width < part[1] < 0.75 * width
        and 0.25 * height < part[2] < 0.75 * height
    ]
    if len(center_parts) != 1:
        raise ValueError("abstain: the red center is not uniquely recoverable")
    center = (center_parts[0][1], center_parts[0][2])

    head_mask = (chroma > 32) & (luminance < 125)
    head_parts = [
        part
        for part in _components(head_mask)
        if 900 <= part[0] <= 2600 and max(part[3], part[4]) <= 95
    ]
    shaft_mask = (chroma > 30) & (luminance >= 125) & (luminance < 210)
    shaft_parts = [
        part
        for part in _components(shaft_mask)
        if 850 <= part[0] <= 3000 and max(part[3], part[4]) <= 115
    ]
    if len(head_parts) != 3 or len(shaft_parts) != 3:
        raise ValueError("abstain: expected exactly three arrowheads and three shafts")

    heads = [(part[1], part[2]) for part in head_parts]
    shafts = [(part[1], part[2]) for part in shaft_parts]
    available = set(range(3))
    bindings: list[tuple[tuple[float, float], tuple[float, float]]] = []
    for shaft in shafts:
        head_index = min(available, key=lambda index: _distance_sq(shaft, heads[index]))
        if _distance_sq(shaft, heads[head_index]) > 85.0**2:
            raise ValueError("abstain: an arrowhead is too far from its shaft")
        available.remove(head_index)
        bindings.append((shaft, heads[head_index]))
    if available:
        raise ValueError("abstain: arrow binding is not one-to-one")

    signs: list[int] = []
    for shaft, head in bindings:
        radial_x = shaft[0] - center[0]
        radial_y = shaft[1] - center[1]
        heading_x = head[0] - shaft[0]
        heading_y = head[1] - shaft[1]
        denominator = (radial_x**2 + radial_y**2) ** 0.5 * (heading_x**2 + heading_y**2) ** 0.5
        if denominator <= 0:
            raise ValueError("abstain: degenerate arrow geometry")
        normalized_cross = (radial_x * heading_y - radial_y * heading_x) / denominator
        if abs(normalized_cross) < 0.70:
            raise ValueError("abstain: an arrow is not visibly tangent")
        signs.append(1 if normalized_cross > 0 else -1)
    return "yes" if len(set(signs)) == 1 else "no"
