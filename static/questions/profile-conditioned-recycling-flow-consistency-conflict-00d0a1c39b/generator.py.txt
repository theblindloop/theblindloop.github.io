"""Deterministic renderer for a three-arrow cyclic-flow consistency world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 246, 239)
GUIDE = (215, 216, 209)
CENTER_FILL = (218, 55, 62)
CENTER_OUTLINE = (116, 31, 43)
PALETTES = (
    ((95, 190, 125), (20, 110, 65)),
    ((92, 177, 222), (24, 88, 150)),
    ((188, 135, 215), (100, 54, 143)),
)
CORNER_COLORS = ((225, 166, 45), (76, 112, 126), (207, 112, 76), (114, 126, 82))


def sample_scene(seed: int) -> dict:
    """Return one balanced scene; 2k/2k+1 are matched yes/no variants."""

    value = int(seed)
    pair_index = value // 2
    rng = random.Random(0x5EEDC0DE + pair_index * 7919)
    base_direction = 1 if rng.randrange(2) == 0 else -1
    directions = [base_direction, base_direction, base_direction]
    if value % 2:
        directions[pair_index % 3] *= -1
    corner_pattern = list(range(4))
    rng.shuffle(corner_pattern)
    return {
        "directions": directions,
        "rotation_deg": rng.randrange(0, 360),
        "center_x": 256 + rng.randrange(-18, 19),
        "center_y": 256 + rng.randrange(-18, 19),
        "scale_pct": rng.randrange(92, 109),
        "palette_index": pair_index % len(PALETTES),
        "skew_degrees": [rng.randrange(-6, 7) for _ in range(3)],
        "corner_pattern": corner_pattern,
    }


def _point(cx: float, cy: float, angle_deg: float, x: float, y: float) -> tuple[int, int]:
    angle = math.radians(angle_deg % 360.0)
    cosine = math.cos(angle)
    sine = math.sin(angle)
    return (
        int(round(cx + x * cosine - y * sine)),
        int(round(cy + x * sine + y * cosine)),
    )


def _arrow_polygons(
    x: float,
    y: float,
    direction_angle: float,
    scale: float,
) -> tuple[list[tuple[int, int]], list[tuple[int, int]]]:
    shaft = [
        _point(x, y, direction_angle, -58 * scale, -12 * scale),
        _point(x, y, direction_angle, 20 * scale, -12 * scale),
        _point(x, y, direction_angle, 20 * scale, 12 * scale),
        _point(x, y, direction_angle, -58 * scale, 12 * scale),
    ]
    head = [
        _point(x, y, direction_angle, 10 * scale, -28 * scale),
        _point(x, y, direction_angle, 68 * scale, 0),
        _point(x, y, direction_angle, 10 * scale, 28 * scale),
    ]
    return shaft, head


def _draw_corner_mark(draw: ImageDraw.ImageDraw, x: int, y: int, kind: int) -> None:
    color = CORNER_COLORS[kind % len(CORNER_COLORS)]
    if kind % 2 == 0:
        draw.polygon([(x, y - 10), (x + 10, y), (x, y + 10), (x - 10, y)], fill=color)
    else:
        draw.ellipse((x - 9, y - 9, x + 9, y + 9), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    directions = [int(value) for value in scene["directions"]]
    rotation = float(scene["rotation_deg"]) % 360.0
    center_x = float(scene["center_x"])
    center_y = float(scene["center_y"])
    scale = float(scene["scale_pct"]) / 100.0
    palette = PALETTES[int(scene["palette_index"]) % len(PALETTES)]
    skews = [float(value) for value in scene["skew_degrees"]]
    pattern = [int(value) for value in scene["corner_pattern"]]

    guide_radius = int(round(128 * scale))
    draw.ellipse(
        (
            int(center_x - guide_radius),
            int(center_y - guide_radius),
            int(center_x + guide_radius),
            int(center_y + guide_radius),
        ),
        outline=GUIDE,
        width=4,
    )

    radial_distance = 126 * scale
    for index, direction in enumerate(directions):
        radial_angle = rotation + 120.0 * index
        radial_radians = math.radians(radial_angle % 360.0)
        arrow_x = center_x + radial_distance * math.cos(radial_radians)
        arrow_y = center_y + radial_distance * math.sin(radial_radians)
        direction_angle = radial_angle + direction * 90.0 + skews[index]
        shaft, head = _arrow_polygons(arrow_x, arrow_y, direction_angle, scale)
        draw.polygon(shaft, fill=palette[0])
        draw.polygon(head, fill=palette[1])

    radius = 11
    draw.ellipse(
        (center_x - radius, center_y - radius, center_x + radius, center_y + radius),
        fill=CENTER_FILL,
        outline=CENTER_OUTLINE,
        width=3,
    )
    locations = ((35, 35), (477, 35), (35, 477), (477, 477))
    for location, kind in zip(locations, pattern, strict=True):
        _draw_corner_mark(draw, location[0], location[1], kind)
    return image


def analytic_gold(scene: dict) -> str:
    directions = [int(value) for value in scene["directions"]]
    return "yes" if len(directions) == 3 and len(set(directions)) == 1 else "no"


def margin(scene: dict) -> float:
    directions = [int(value) for value in scene["directions"]]
    skews = [float(value) for value in scene["skew_degrees"]]
    if len(directions) != 3 or len(skews) != 3:
        return 0.0
    clearances = [abs(math.sin(math.radians(sign * 90.0 + skew))) for sign, skew in zip(directions, skews, strict=True)]
    return round(100.0 * min(clearances), 6)


def is_quarantined(scene: dict) -> bool:
    directions = [int(value) for value in scene["directions"]]
    return len(directions) != 3 or any(value not in (-1, 1) for value in directions) or margin(scene) < 70.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["rotation_deg"] = float(twin["rotation_deg"]) + 360.0
    return [("rotation_plus_full_turn", twin)]
