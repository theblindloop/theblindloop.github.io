"""Pixel-only inverse arm for the proportion-tick-rail world.

The oracle receives a rasterized image and nothing else.  It thresholds ink
against the page, splits the image into horizontal ink bands, identifies the
one full-width band as the dividing rule, and reads every remaining band as
either a long thin rail or a small marker.  For the reference pair above the
rule and for each candidate pair below it, the marker's centre column is
expressed as a fraction of that rail's own drawn length, measured from that
rail's own left end.  The reported rail is the one whose fraction is closest to
the reference fraction.  The oracle abstains whenever the band structure, the
pairing, or a unique clear winner is not recovered.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

N_RAILS = 5
INK_MAX = 190
RULE_WIDTH_FRACTION = 0.90
MIN_RAIL_WIDTH = 150
MAX_MARKER_WIDTH = 34
MAX_BAND_HEIGHT = 26
ORACLE_MARGIN = 3.0


def _runs(flags: np.ndarray) -> list:
    """Contiguous index ranges where the boolean profile is true."""
    spans = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            spans.append((start, index - 1))
            start = None
    if start is not None:
        spans.append((start, len(flags) - 1))
    return spans


def _bands(ink: np.ndarray) -> list:
    """Horizontal ink bands with their column extents."""
    bands = []
    for top, bottom in _runs(ink.any(axis=1)):
        columns = np.flatnonzero(ink[top : bottom + 1, :].any(axis=0))
        if columns.size == 0:
            return None
        bands.append(
            {
                "top": top,
                "bottom": bottom,
                "height": bottom - top + 1,
                "left": int(columns[0]),
                "right": int(columns[-1]),
                "width": int(columns[-1] - columns[0] + 1),
            }
        )
    return bands


def _pair_fraction(marker: dict, rail: dict) -> float:
    """Marker centre as a fraction of the rail's own length, or None."""
    if marker["width"] > MAX_MARKER_WIDTH or marker["height"] > MAX_BAND_HEIGHT:
        return None
    if rail["width"] < MIN_RAIL_WIDTH or rail["height"] > MAX_BAND_HEIGHT:
        return None
    centre = (marker["left"] + marker["right"]) / 2.0
    if not rail["left"] <= centre <= rail["right"]:
        return None
    return (centre - rail["left"]) / (rail["width"] - 1)


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB")).astype(np.int32)
    ink = rgb.mean(axis=2) < INK_MAX

    bands = _bands(ink)
    if not bands:
        return AMBIGUOUS

    limit = RULE_WIDTH_FRACTION * ink.shape[1]
    rules = [index for index, band in enumerate(bands) if band["width"] >= limit]
    if len(rules) != 1:
        return AMBIGUOUS
    divider = rules[0]

    above = bands[:divider]
    below = bands[divider + 1 :]
    if len(above) != 2 or len(below) != 2 * N_RAILS:
        return AMBIGUOUS

    reference = _pair_fraction(above[0], above[1])
    if reference is None:
        return AMBIGUOUS

    deltas = []
    for index in range(N_RAILS):
        fraction = _pair_fraction(below[2 * index], below[2 * index + 1])
        if fraction is None:
            return AMBIGUOUS
        deltas.append(abs(fraction - reference))

    order = sorted(range(N_RAILS), key=lambda index: deltas[index])
    if 100.0 * (deltas[order[1]] - deltas[order[0]]) < ORACLE_MARGIN:
        return AMBIGUOUS
    return f"rail-{order[0] + 1}"
