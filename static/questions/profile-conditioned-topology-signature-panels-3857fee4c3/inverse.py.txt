"""Independent pixel-only inverse arm.

This module deliberately contains its own segmentation and connectivity code and
does not import any world-state or gold-producing module.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _components(mask: np.ndarray, minimum_area: int) -> list[tuple[int, int, int, int, int]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    for y0, x0 in zip(*np.nonzero(mask & ~seen)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(y0), int(x0))])
        seen[y0, x0] = True
        area = 0
        min_x = max_x = int(x0)
        min_y = max_y = int(y0)
        while queue:
            y, x = queue.popleft()
            area += 1
            min_x, max_x = min(min_x, x), max(max_x, x)
            min_y, max_y = min(min_y, y), max(max_y, y)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if area >= minimum_area:
            output.append((min_x, min_y, max_x, max_y, area))
    return output


def _find_frames(rgb: np.ndarray) -> list[tuple[int, int, int, int]]:
    red = rgb[:, :, 0].astype(int)
    green = rgb[:, :, 1].astype(int)
    blue = rgb[:, :, 2].astype(int)
    blue_frame = (blue > 145) & (green > 70) & (green < 160) & (red < 80)
    gray_frame = (red > 105) & (red < 190) & (green > 125) & (green < 205) & (blue > 130) & (blue < 215)
    found = _components(blue_frame | gray_frame, minimum_area=450)
    boxes = [(a, b, c, d) for a, b, c, d, _ in found if c - a > 170 and d - b > 210]
    boxes.sort(key=lambda box: box[0])
    if len(boxes) != 4:
        raise ValueError(f"could not recover four visible frames: {boxes}")
    return boxes


def _signature(rgb: np.ndarray, box: tuple[int, int, int, int]) -> tuple[int, int]:
    left, top, right, bottom = box
    crop = rgb[top + 9 : bottom - 8, left + 9 : right - 8]
    black = np.max(crop, axis=2) < 90
    foreground = _components(black, minimum_area=90)
    # Holes are connected components of non-black pixels that do not touch the
    # crop boundary. Antialiased edge pixels are background for this topological
    # computation; the thick black core remains closed.
    background = ~black
    regions = _components(background, minimum_area=70)
    height, width = background.shape
    holes = sum(
        1
        for x0, y0, x1, y1, _ in regions
        if x0 > 0 and y0 > 0 and x1 < width - 1 and y1 < height - 1
    )
    return len(foreground), holes


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    boxes = _find_frames(rgb)
    signatures = [_signature(rgb, box) for box in boxes]
    target = signatures[0]
    matches = [index for index, value in enumerate(signatures[1:]) if value == target]
    if len(matches) != 1:
        raise ValueError(f"ambiguous pixel signatures {signatures}")
    return ("first", "second", "third")[matches[0]]
