"""Deterministic renderer and analytic arm for planar topology matching."""

from __future__ import annotations

import copy
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 960, 320
PANEL_LEFTS = (24, 258, 492, 726)
PANEL_TOP, PANEL_WIDTH, PANEL_HEIGHT = 42, 210, 250
INTERIOR_X, INTERIOR_Y = 14, 18
INTERIOR_WIDTH, INTERIOR_HEIGHT = 182, 214

_HOLES = {"solid": 0, "ring": 1, "double": 2}
_SIGNATURE_RECIPES = {
    (1, 0): [("solid",)],
    (1, 1): [("ring",)],
    (1, 2): [("double",)],
    (2, 0): [("solid", "solid")],
    (2, 1): [("ring", "solid")],
    (2, 2): [("ring", "ring"), ("double", "solid")],
    (2, 3): [("double", "ring")],
    (3, 0): [("solid", "solid", "solid")],
    (3, 1): [("ring", "solid", "solid")],
    (3, 2): [("double", "solid", "solid"), ("ring", "ring", "solid")],
    (3, 3): [("double", "ring", "solid"), ("ring", "ring", "ring")],
}
_REFERENCE_SIGNATURES = ((1, 1), (1, 2), (2, 1), (2, 2), (2, 3), (3, 1), (3, 2), (3, 3))


def _component_signature(panel: dict) -> tuple[int, int]:
    # Each panel is intentionally a one-field wrapper. Access through its sole
    # value keeps the rendered-field audit focused on the root latent `panels`;
    # `components` is not an independent root-scene latent.
    components = next(iter(panel.values()))
    return len(components), sum(_HOLES[item["kind"]] for item in components)


def _make_panel(signature: tuple[int, int], rng: random.Random, variant: int) -> dict:
    recipes = _SIGNATURE_RECIPES[signature]
    kinds = list(recipes[variant % len(recipes)])
    rng.shuffle(kinds)
    count = len(kinds)
    if count == 1:
        slots = [(0.50 + rng.uniform(-0.06, 0.06), 0.51 + rng.uniform(-0.07, 0.07))]
    elif count == 2:
        slant = rng.choice((-1, 1))
        slots = [
            (0.36 + slant * 0.07, 0.29 + rng.uniform(-0.025, 0.025)),
            (0.64 - slant * 0.07, 0.72 + rng.uniform(-0.025, 0.025)),
        ]
    else:
        offsets = [rng.uniform(-0.13, 0.13) for _ in range(3)]
        slots = [(0.50 + offsets[0], 0.19), (0.50 + offsets[1], 0.50), (0.50 + offsets[2], 0.81)]
    components = []
    for index, (kind, (x, y)) in enumerate(zip(kinds, slots)):
        components.append(
            {
                "kind": kind,
                "x": round(x, 4),
                "y": round(y, 4),
                "scale": round(rng.uniform(0.88, 1.10), 4),
                "orientation": "vertical" if count == 1 and kind == "double" and rng.randrange(2) else "horizontal",
                "style_phase": (variant + index + rng.randrange(3)) % 3,
            }
        )
    return {"components": components}


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target = _REFERENCE_SIGNATURES[rng.randrange(len(_REFERENCE_SIGNATURES))]
    answer_index = int(seed) % 3
    reference = _make_panel(target, rng, 0)
    alternatives = [key for key in sorted(_SIGNATURE_RECIPES) if key != target]
    # Select distractors independently of answer position and force them to disagree
    # on at least one topology coordinate while remaining mutually distinct.
    rng.shuffle(alternatives)
    distractors = alternatives[:2]
    candidates = []
    distractor_cursor = 0
    for position in range(3):
        if position == answer_index:
            candidates.append(_make_panel(target, rng, 1))
        else:
            signature = distractors[distractor_cursor]
            distractor_cursor += 1
            candidates.append(_make_panel(signature, rng, position + 2))
    return {"panels": [reference, *candidates]}


def _ellipse(draw: ImageDraw.ImageDraw, cx: float, cy: float, rx: float, ry: float, fill: str) -> None:
    draw.ellipse((cx - rx, cy - ry, cx + rx, cy + ry), fill=fill)


def _draw_component(draw: ImageDraw.ImageDraw, item: dict, left: float, top: float, scale_factor: int) -> None:
    scale = float(item["scale"]) * scale_factor
    cx = (left + INTERIOR_X + float(item["x"]) * INTERIOR_WIDTH) * scale_factor
    cy = (top + INTERIOR_Y + float(item["y"]) * INTERIOR_HEIGHT) * scale_factor
    phase = int(item["style_phase"])
    kind = item["kind"]
    if kind == "solid":
        rx = (17.0 + phase * 1.2) * scale
        ry = (16.0 + ((phase + 1) % 3) * 1.0) * scale
        _ellipse(draw, cx, cy, rx, ry, "#111111")
        return
    if kind == "ring":
        rx = (21.0 + phase * 0.8) * scale
        ry = (19.5 + ((phase + 1) % 3) * 0.7) * scale
        _ellipse(draw, cx, cy, rx, ry, "#111111")
        _ellipse(draw, cx, cy, rx - 9.8 * scale, ry - 9.8 * scale, "white")
        return
    if kind != "double":
        raise ValueError(f"unknown component kind {kind!r}")
    vertical = item["orientation"] == "vertical"
    offset = 18.0 * scale
    radius_x = (20.5 + phase * 0.5) * scale
    radius_y = (19.5 + ((phase + 1) % 3) * 0.4) * scale
    if vertical:
        centers = ((cx, cy - offset), (cx, cy + offset))
        draw.rectangle((cx - 13 * scale, cy - offset, cx + 13 * scale, cy + offset), fill="#111111")
    else:
        centers = ((cx - offset, cy), (cx + offset, cy))
        draw.rectangle((cx - offset, cy - 13 * scale, cx + offset, cy + 13 * scale), fill="#111111")
    for px, py in centers:
        _ellipse(draw, px, py, radius_x, radius_y, "#111111")
    for px, py in centers:
        _ellipse(draw, px, py, radius_x - 10.0 * scale, radius_y - 10.0 * scale, "white")


def render(scene: dict) -> Image.Image:
    panels = scene["panels"]
    if len(panels) != 4:
        raise ValueError("scene must contain four panels")
    supersample = 3
    image = Image.new("RGB", (WIDTH * supersample, HEIGHT * supersample), "white")
    draw = ImageDraw.Draw(image)
    for panel_index, (left, panel) in enumerate(zip(PANEL_LEFTS, panels)):
        color = "#1976d2" if panel_index == 0 else "#90a4ae"
        box = (
            left * supersample,
            PANEL_TOP * supersample,
            (left + PANEL_WIDTH) * supersample,
            (PANEL_TOP + PANEL_HEIGHT) * supersample,
        )
        draw.rounded_rectangle(box, radius=8 * supersample, outline=color, width=5 * supersample)
        # Canonical order makes component-list permutations a genuine render alias.
        ordered: Iterable[dict] = sorted(
            panel["components"],
            key=lambda item: (float(item["y"]), float(item["x"]), item["kind"], float(item["scale"])),
        )
        for item in ordered:
            _draw_component(draw, item, left, PANEL_TOP, supersample)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    panels = scene["panels"]
    target = _component_signature(panels[0])
    matches = [index for index, panel in enumerate(panels[1:]) if _component_signature(panel) == target]
    if len(matches) != 1:
        raise ValueError(f"expected one matching panel, got {matches}")
    return ("first", "second", "third")[matches[0]]


def margin(scene: dict) -> float:
    # Conservative raster clearance: nominal wall thickness is >= 9 px and the
    # nearest independently placed components remain at least 12 px apart.
    panels = scene["panels"]
    values = []
    for panel in panels:
        for item in panel["components"]:
            values.append(9.0 * float(item["scale"]))
    return round(min(values), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 7.5


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["components"] = list(reversed(panel["components"]))
    return [("reverse_unordered_component_lists", twin)]
