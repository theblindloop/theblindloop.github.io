"""Deterministic RGB renderer for the visible flag-band count world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 560
HEIGHT = 420
SUPERSAMPLE = 4
OUTLINE_WIDTH = 5.0

# Every fill colour is strongly chromatic and well separated from the neutral
# scene furniture.  The sampler permutes this palette independently of count.
BAND_PALETTE = (
    (214, 52, 62),
    (40, 104, 203),
    (40, 160, 100),
    (224, 168, 38),
    (133, 78, 190),
    (24, 154, 166),
    (224, 101, 38),
    (194, 52, 139),
)
BACKGROUND_PALETTE = (
    (248, 249, 246),
    (250, 247, 240),
    (244, 248, 251),
    (248, 245, 250),
)
POLE_PALETTE = (
    (64, 72, 82),
    (77, 82, 89),
    (83, 75, 70),
    (59, 83, 84),
)
LABEL_COLOR = (92, 98, 103)


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one scene; styling is independent from the answer field."""

    answer_rng = random.Random(int(seed) ^ 0x5EED_431)
    style_rng = random.Random(int(seed) ^ 0xA17E_9B3)
    stripe_count = answer_rng.choice((2, 3, 4, 5, 6))
    near_boundary = style_rng.random() < 0.16

    flag_width = style_rng.uniform(268.0, 342.0)
    flag_height = style_rng.uniform(182.0, 230.0)
    if near_boundary:
        # Keep a few valid but deliberately thin examples in the dataset.  The
        # independent oracle still measures them, while the manifest marks them.
        flag_height = style_rng.uniform(118.0, 151.0)

    min_x = 154.0
    max_x = WIDTH - flag_width - 25.0
    min_y = 58.0
    max_y = HEIGHT - flag_height - 54.0
    flag_x = style_rng.uniform(min_x, max_x)
    flag_y = style_rng.uniform(min_y, max_y)
    pole_width = style_rng.uniform(14.0, 22.0)
    finial_radius = style_rng.uniform(9.0, 13.0)

    stripe_colors = list(style_rng.sample(BAND_PALETTE, stripe_count))
    return {
        "stripe_count": int(stripe_count),
        "flag_x": round(flag_x, 3),
        "flag_y": round(flag_y, 3),
        "flag_width": round(flag_width, 3),
        "flag_height": round(flag_height, 3),
        "stripe_colors": [list(color) for color in stripe_colors],
        "background": list(style_rng.choice(BACKGROUND_PALETTE)),
        "pole_color": list(style_rng.choice(POLE_PALETTE)),
        "outline_color": [35, 43, 51],
        "pole_width": round(pole_width, 3),
        "finial_radius": round(finial_radius, 3),
        "near_boundary": bool(near_boundary),
        "label_text": "FLAG",
    }


def _font(size: int):
    try:
        return ImageFont.truetype("DejaVuSans-Bold.ttf", size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, float, float, float], scale: int) -> tuple[float, ...]:
    return tuple(float(value) * scale for value in values)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only visible geometry; no metadata is embedded in the PNG."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    fx = float(scene["flag_x"])
    fy = float(scene["flag_y"])
    fw = float(scene["flag_width"])
    fh = float(scene["flag_height"])
    pole_color = tuple(scene["pole_color"])
    outline = tuple(scene["outline_color"])
    pole_width = float(scene["pole_width"])
    finial = float(scene["finial_radius"])

    pole_x = fx - pole_width * 0.65
    pole_top = fy - finial * 2.4
    pole_bottom = fy + fh + 48.0
    draw.rectangle(_box((pole_x - pole_width / 2, pole_top, pole_x + pole_width / 2, pole_bottom), scale), fill=pole_color)
    draw.ellipse(
        _box((pole_x - finial, pole_top - finial * 1.3, pole_x + finial, pole_top + finial * 0.7), scale),
        fill=pole_color,
    )
    draw.ellipse(
        _box((pole_x - finial * 0.48, pole_top - finial * 0.65, pole_x + finial * 0.48, pole_top + finial * 0.31), scale),
        fill=(170, 177, 180),
    )

    # Draw the dark outline first, then equal-height colored interior bands.
    draw.rectangle(_box((fx, fy, fx + fw, fy + fh), scale), fill=outline)
    inner = OUTLINE_WIDTH
    ix0, iy0 = fx + inner, fy + inner
    ix1, iy1 = fx + fw - inner, fy + fh - inner
    band_height = (iy1 - iy0) / float(scene["stripe_count"])
    for index, color in enumerate(scene["stripe_colors"]):
        y0 = iy0 + index * band_height
        y1 = iy0 + (index + 1) * band_height
        draw.rectangle(_box((ix0, y0, ix1, y1), scale), fill=tuple(color))

    # A small neutral caption gives the familiar object a stable context but is
    # deliberately outside the colored interior and irrelevant to the oracle.
    label_y = fy + fh + 18.0
    draw.text(
        (fx + fw / 2.0 * scale, label_y * scale),
        str(scene["label_text"]),
        anchor="ma",
        fill=LABEL_COLOR,
        font=_font(14 * scale),
    )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return str(scene["stripe_count"])


def margin(scene: dict[str, Any]) -> float:
    inner_height = float(scene["flag_height"]) - 2.0 * OUTLINE_WIDTH
    inner_width = float(scene["flag_width"]) - 2.0 * OUTLINE_WIDTH
    return float(min(inner_height / float(scene["stripe_count"]), inner_width))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 25.0 or float(scene["flag_width"]) - 2.0 * OUTLINE_WIDTH < 120.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No latent transform is asserted pixel-identical for this renderer."""

    return []
