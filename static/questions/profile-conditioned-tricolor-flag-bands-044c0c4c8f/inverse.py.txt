"""Independent image-only inverse for counting colored flag bands."""

from __future__ import annotations

import numpy as np


def _largest_row_block(mask: np.ndarray) -> tuple[int, int]:
    """Find the longest run of rows containing a wide chromatic region."""

    height, width = mask.shape
    row_counts = mask.sum(axis=1)
    eligible = row_counts >= max(80, int(width * 0.18))
    blocks: list[tuple[int, int]] = []
    start = None
    for index, value in enumerate(eligible.tolist() + [False]):
        if value and start is None:
            start = index
        elif not value and start is not None:
            blocks.append((start, index))
            start = None
    if not blocks:
        raise ValueError("no wide colored flag region found")
    # A supersampled boundary between two saturated fills can become one
    # neutral antialias row.  Join only tiny gaps; this cannot join the flag to
    # the neutral pole or caption, which are separated by many rows.
    merged: list[list[int]] = []
    for start, stop in blocks:
        if merged and start - merged[-1][1] <= 3:
            merged[-1][1] = stop
        else:
            merged.append([start, stop])
    return max(
        ((int(start), int(stop)) for start, stop in merged),
        key=lambda block: (block[1] - block[0], int(row_counts[block[0] : block[1]].max())),
    )


def _transition_count(colors: np.ndarray) -> int:
    """Count robust color changes between piecewise-constant band runs."""

    if len(colors) < 24:
        raise ValueError("colored scanline is too short")
    # A short median filter suppresses antialiased seam pixels without knowing
    # the renderer's palette or any latent coordinate.
    padded = np.pad(colors.astype(np.float64), ((2, 2), (0, 0)), mode="edge")
    smooth = np.stack([np.median(padded[index : index + 5], axis=0) for index in range(len(colors))])
    scores: list[float] = []
    radius = 5
    for index in range(radius, len(smooth) - radius):
        left = smooth[index - radius : index].mean(axis=0)
        right = smooth[index : index + radius].mean(axis=0)
        scores.append(float(np.linalg.norm(left - right)))
    candidate_indices = [index + radius for index, score in enumerate(scores) if score >= 32.0]
    groups: list[list[int]] = []
    for index in candidate_indices:
        if not groups or index - groups[-1][-1] > 6:
            groups.append([index])
        else:
            groups[-1].append(index)
    if not groups:
        raise ValueError("no visible band boundaries found")
    return len(groups) + 1


def decision_from_image(image) -> str:
    """Recover the count from the raster alone, or raise on an ambiguous image."""

    array = np.asarray(image.convert("RGB"), dtype=np.float64)
    chroma = array.max(axis=2) - array.min(axis=2)
    bright = array.max(axis=2)
    # Pole, outline, caption, and background are neutral; only the flag fills
    # are wide and chromatic under this public visual contract.
    mask = (chroma >= 48.0) & (bright >= 100.0)
    y0, y1 = _largest_row_block(mask)
    interior = mask[y0:y1]
    column_strength = interior.sum(axis=0)
    stable_columns = np.flatnonzero(column_strength >= max(12, int((y1 - y0) * 0.72)))
    if len(stable_columns) < 20:
        raise ValueError("flag interior has no stable scanline")
    x = int(np.median(stable_columns))
    valid_rows = np.flatnonzero(mask[y0:y1, x])
    if len(valid_rows) < 24:
        raise ValueError("central flag scanline is ambiguous")
    start = int(valid_rows.min())
    stop = int(valid_rows.max()) + 1
    # Trim only the outer antialiasing fringe, retaining all band interiors.
    start += 2
    stop -= 2
    count = _transition_count(array[y0 + start : y0 + stop, x])
    if count not in (2, 3, 4, 5, 6):
        raise ValueError(f"ambiguous band count {count}")
    return str(count)
