"""Independent pixel-only decision arm.

Receives a rasterised PNG and nothing else: no latent scene, no renderer
constants, no analytic gold. It segments dark ink, isolates the three plates by
maximal component bounding boxes, and counts, inside each plate, the wall pieces
that are detached from that plate's outer wall. A detached wall piece is a
bounded complementary component of the corridor space, so the number of such
pieces is the corridor cycle rank. The answer is the plate with none.
"""

from __future__ import annotations

from collections import deque

from PIL import Image

DARK_THRESHOLD = 110
MIN_PLATE_SIDE = 40


def _dark_mask(image: Image.Image) -> tuple[list[list[bool]], int, int]:
    grey = image.convert("L")
    width, height = grey.size
    data = list(grey.getdata())
    mask = [
        [data[y * width + x] < DARK_THRESHOLD for x in range(width)]
        for y in range(height)
    ]
    return mask, width, height


def _components(mask, width, height, region=None):
    if region is None:
        x_lo, y_lo, x_hi, y_hi = 0, 0, width - 1, height - 1
    else:
        x_lo, y_lo, x_hi, y_hi = region
    seen = [[False] * width for _ in range(height)]
    out = []
    for y in range(y_lo, y_hi + 1):
        for x in range(x_lo, x_hi + 1):
            if not mask[y][x] or seen[y][x]:
                continue
            queue = deque([(x, y)])
            seen[y][x] = True
            pixels = []
            while queue:
                cx, cy = queue.popleft()
                pixels.append((cx, cy))
                for dx in (-1, 0, 1):
                    for dy in (-1, 0, 1):
                        nx, ny = cx + dx, cy + dy
                        if x_lo <= nx <= x_hi and y_lo <= ny <= y_hi:
                            if mask[ny][nx] and not seen[ny][nx]:
                                seen[ny][nx] = True
                                queue.append((nx, ny))
            xs = [p[0] for p in pixels]
            ys = [p[1] for p in pixels]
            out.append(
                {
                    "size": len(pixels),
                    "box": (min(xs), min(ys), max(xs), max(ys)),
                }
            )
    return out


def _contains(outer, inner) -> bool:
    return (
        outer[0] <= inner[0]
        and outer[1] <= inner[1]
        and outer[2] >= inner[2]
        and outer[3] >= inner[3]
        and outer != inner
    )


def decision_from_image(image: Image.Image) -> str:
    """Return 'left', 'center', 'right', or 'abstain'."""
    mask, width, height = _dark_mask(image)
    comps = _components(mask, width, height)
    if not comps:
        return "abstain"

    plates = []
    for comp in comps:
        box = comp["box"]
        side_x = box[2] - box[0] + 1
        side_y = box[3] - box[1] + 1
        if side_x < MIN_PLATE_SIDE or side_y < MIN_PLATE_SIDE:
            continue
        if any(_contains(other["box"], box) for other in comps if other is not comp):
            continue
        plates.append(comp)

    if len(plates) != 3:
        return "abstain"
    plates.sort(key=lambda comp: comp["box"][0])

    boxes = [plate["box"] for plate in plates]
    for left, right in zip(boxes, boxes[1:]):
        if right[0] <= left[2]:
            return "abstain"

    loop_counts = []
    for box in boxes:
        inner = _components(mask, width, height, region=box)
        detached = 0
        for comp in inner:
            cbox = comp["box"]
            touches_frame = (
                cbox[0] == box[0]
                or cbox[1] == box[1]
                or cbox[2] == box[2]
                or cbox[3] == box[3]
            )
            if not touches_frame:
                detached += 1
        loop_counts.append(detached)

    zeros = [index for index, count in enumerate(loop_counts) if count == 0]
    if len(zeros) != 1:
        return "abstain"
    return ("left", "center", "right")[zeros[0]]
