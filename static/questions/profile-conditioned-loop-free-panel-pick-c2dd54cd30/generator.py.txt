"""Deterministic scene sampling and rendering for the loop-free panel world.

Three walled panels are drawn side by side. Every panel is a rectangular plate
whose interior is cut into corridors by black walls. In exactly one panel the
corridor space is loop free (its cycle rank is zero, so no closed round trip
exists); the other two panels carry one to three independent circuits. Black
ink is equalised exactly across the three panels with blind wall stubs, so ink,
wall length and corridor area cannot be used as shortcuts.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 5
CELL = 18
WALL = 4
PANEL = WALL + GRID * (CELL + WALL)
PAD = 18
GAP = 28
CANVAS_W = PAD * 2 + 3 * PANEL + 2 * GAP
CANVAS_H = PAD * 2 + PANEL

INK = (17, 17, 17)
PAPER = (255, 255, 255)

STUB_MIN = 5
STUB_MAX = 13
MIN_CLEARANCE = 4
QUARANTINE_MARGIN = 5.0
TIGHT_PROBABILITY = 0.18

POSITIONS = ("left", "center", "right")
SIDES = ("N", "E", "S", "W")
STUB_BUDGETS = (18, 27, 36)


def _neighbours(idx: int) -> dict[str, int]:
    row, col = divmod(idx, GRID)
    out: dict[str, int] = {}
    if row > 0:
        out["N"] = (row - 1) * GRID + col
    if col < GRID - 1:
        out["E"] = row * GRID + col + 1
    if row < GRID - 1:
        out["S"] = (row + 1) * GRID + col
    if col > 0:
        out["W"] = row * GRID + col - 1
    return out


def _all_edges() -> list[tuple[int, int]]:
    edges = set()
    for cell in range(GRID * GRID):
        for other in _neighbours(cell).values():
            edges.add((min(cell, other), max(cell, other)))
    return sorted(edges)


def _spanning_tree(rng: random.Random) -> list[tuple[int, int]]:
    start = rng.randrange(GRID * GRID)
    seen = {start}
    stack = [start]
    edges: list[tuple[int, int]] = []
    while stack:
        cur = stack[-1]
        options = [v for v in _neighbours(cur).values() if v not in seen]
        if not options:
            stack.pop()
            continue
        nxt = options[rng.randrange(len(options))]
        edges.append((min(cur, nxt), max(cur, nxt)))
        seen.add(nxt)
        stack.append(nxt)
    return edges


def cycle_rank(passages) -> int:
    """Number of independent circuits of the corridor graph (E - V + C)."""
    parent = list(range(GRID * GRID))

    def find(node: int) -> int:
        while parent[node] != node:
            parent[node] = parent[parent[node]]
            node = parent[node]
        return node

    extra = 0
    for pair in passages:
        left, right = int(pair[0]), int(pair[1])
        root_l, root_r = find(left), find(right)
        if root_l == root_r:
            extra += 1
        else:
            parent[root_l] = root_r
    return extra


def _split_budget(total: int, rng: random.Random, max_len: int = STUB_MAX) -> list[int]:
    count = max(1, -(-total // max_len))
    while count > 1 and count * STUB_MIN > total:
        count -= 1
    parts = [STUB_MIN] * count
    remainder = total - STUB_MIN * count
    guard = 0
    while remainder > 0 and guard < 5000:
        guard += 1
        room = [i for i, value in enumerate(parts) if value < max_len]
        if not room:
            parts.append(STUB_MIN)
            remainder -= STUB_MIN
            continue
        index = room[rng.randrange(len(room))]
        add = rng.randint(1, min(max_len - parts[index], remainder))
        parts[index] += add
        remainder -= add
    return sorted(parts)


def _sample_panel(rng: random.Random, loops: int, stub_total: int, floor: int) -> dict | None:
    tree = set(_spanning_tree(rng))
    edges = set(tree)
    if loops:
        pool = [edge for edge in _all_edges() if edge not in tree]
        rng.shuffle(pool)
        if len(pool) < loops:
            return None
        edges.update(pool[:loops])
    passages = sorted(edges)
    open_set = set(passages)

    closed: dict[int, list[str]] = {}
    for cell in range(GRID * GRID):
        sides = []
        for side, other in _neighbours(cell).items():
            if (min(cell, other), max(cell, other)) not in open_set:
                sides.append(side)
        if sides:
            closed[cell] = sorted(sides)

    lengths = _split_budget(stub_total, rng, CELL - floor)
    cells = sorted(closed)
    rng.shuffle(cells)
    if len(cells) < len(lengths):
        return None

    stubs = []
    for length, cell in zip(lengths, cells):
        sides = closed[cell]
        side = sides[rng.randrange(len(sides))]
        offset = rng.randint(floor, CELL - WALL - floor)
        stubs.append([cell, side, int(length), int(offset)])
    stubs.sort()
    return {
        "passages": [[int(u), int(v)] for u, v in passages],
        "stubs": stubs,
    }


def sample_scene(seed: int) -> dict:
    """Sample one deterministic latent scene for ``seed``."""
    for attempt in range(24):
        rng = random.Random(f"loop_free_panel_pick|{int(seed)}|{attempt}")
        answer_index = rng.randrange(3)
        budget = STUB_BUDGETS[rng.randrange(len(STUB_BUDGETS))]
        floor = rng.choice([5, 5, 6, 7])
        panels = []
        ok = True
        for index in range(3):
            loops = 0 if index == answer_index else rng.randint(1, 3)
            panel = _sample_panel(rng, loops, 18 * loops + budget, floor)
            if panel is None:
                ok = False
                break
            panels.append(panel)
        if not ok:
            continue
        scene = {
            "seed": int(seed),
            "canvas": [CANVAS_W, CANVAS_H],
            "grid": GRID,
            "cell": CELL,
            "wall": WALL,
            "stub_budget": budget,
            "clearance_floor": floor,
            "panels": panels,
        }
        if rng.random() < TIGHT_PROBABILITY:
            panel = panels[rng.randrange(3)]
            stub = panel["stubs"][rng.randrange(len(panel["stubs"]))]
            stub[3] = MIN_CLEARANCE if rng.random() < 0.5 else CELL - WALL - MIN_CLEARANCE
            panel["stubs"].sort()
        ranks = [cycle_rank(panel["passages"]) for panel in scene["panels"]]
        if sum(1 for rank in ranks if rank == 0) != 1:
            continue
        return scene
    raise RuntimeError(f"could not sample a scene for seed {seed}")


def analytic_gold(scene: dict) -> str:
    """Panel whose corridor space carries no closed circuit."""
    ranks = [cycle_rank(panel["passages"]) for panel in scene["panels"]]
    zeros = [index for index, rank in enumerate(ranks) if rank == 0]
    if len(zeros) != 1:
        return "abstain"
    return POSITIONS[zeros[0]]


def margin(scene: dict) -> float:
    """Smallest white clearance whose closure could move a circuit count."""
    best = float(CELL)
    for panel in scene["panels"]:
        for _cell, _side, length, offset in panel["stubs"]:
            best = min(best, float(offset))
            best = min(best, float(CELL - WALL - offset))
            best = min(best, float(CELL - length))
    return round(best, 3)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical latent relabellings that must preserve the gold answer."""
    import copy

    reversed_passages = copy.deepcopy(scene)
    for panel in reversed_passages["panels"]:
        panel["passages"] = [list(reversed(edge)) for edge in reversed(panel["passages"])]

    rotated_stubs = copy.deepcopy(scene)
    for panel in rotated_stubs["panels"]:
        stubs = panel["stubs"]
        if stubs:
            panel["stubs"] = stubs[1:] + stubs[:1]

    return [
        ("passage_record_order_reversed", reversed_passages),
        ("stub_record_order_rotated", rotated_stubs),
    ]


def panel_origin(index: int) -> tuple[int, int]:
    return PAD + index * (PANEL + GAP), PAD


def render(scene: dict) -> Image.Image:
    """Rasterise a latent scene; drawing is order invariant by construction."""
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), PAPER)
    draw = ImageDraw.Draw(image)
    for index, panel in enumerate(scene["panels"]):
        ox, oy = panel_origin(index)
        draw.rectangle([ox, oy, ox + PANEL - 1, oy + PANEL - 1], fill=INK)
        for cell in range(GRID * GRID):
            row, col = divmod(cell, GRID)
            x0 = ox + WALL + col * (CELL + WALL)
            y0 = oy + WALL + row * (CELL + WALL)
            draw.rectangle([x0, y0, x0 + CELL - 1, y0 + CELL - 1], fill=PAPER)
        for edge in sorted(tuple(sorted(int(v) for v in pair)) for pair in panel["passages"]):
            low, high = edge
            row, col = divmod(low, GRID)
            x0 = ox + WALL + col * (CELL + WALL)
            y0 = oy + WALL + row * (CELL + WALL)
            if high - low == 1:
                draw.rectangle([x0 + CELL, y0, x0 + CELL + WALL - 1, y0 + CELL - 1], fill=PAPER)
            else:
                draw.rectangle([x0, y0 + CELL, x0 + CELL - 1, y0 + CELL + WALL - 1], fill=PAPER)
        for cell, side, length, offset in sorted(tuple(stub) for stub in panel["stubs"]):
            row, col = divmod(int(cell), GRID)
            x0 = ox + WALL + col * (CELL + WALL)
            y0 = oy + WALL + row * (CELL + WALL)
            if side == "N":
                box = [x0 + offset, y0, x0 + offset + WALL - 1, y0 + length - 1]
            elif side == "S":
                box = [x0 + offset, y0 + CELL - length, x0 + offset + WALL - 1, y0 + CELL - 1]
            elif side == "W":
                box = [x0, y0 + offset, x0 + length - 1, y0 + offset + WALL - 1]
            else:
                box = [x0 + CELL - length, y0 + offset, x0 + CELL - 1, y0 + offset + WALL - 1]
            draw.rectangle(box, fill=INK)
    return image


def panel_ink(scene: dict) -> list[int]:
    """Black pixel count of each panel; equal by construction (counterbalance)."""
    image = render(scene)
    pixels = image.load()
    counts = []
    for index in range(3):
        ox, oy = panel_origin(index)
        total = 0
        for y in range(oy, oy + PANEL):
            for x in range(ox, ox + PANEL):
                if pixels[x, y] == INK:
                    total += 1
        counts.append(total)
    return counts
