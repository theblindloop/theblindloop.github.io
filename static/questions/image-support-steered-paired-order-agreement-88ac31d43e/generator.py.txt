"""Deterministic renderer for paired checkpoint-order tracing."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

BG = (249, 248, 243)
FRAME = (218, 216, 207)
DECOR = (229, 226, 216)
PALE = (194, 198, 196)
QUARANTINE_MARGIN = 60.0

# All variants remain plainly navy/blue/orange/red/green to a human while
# preventing one exact RGB tuple from acting as an answer-correlated cue.
PALETTES = (
    {"navy": (29, 44, 61), "blue": (31, 118, 188), "orange": (231, 128, 43), "red": (207, 61, 64), "green": (43, 151, 96)},
    {"navy": (37, 50, 69), "blue": (37, 109, 179), "orange": (224, 119, 48), "red": (193, 54, 60), "green": (48, 139, 91)},
    {"navy": (24, 39, 56), "blue": (25, 127, 198), "orange": (236, 139, 48), "red": (216, 70, 68), "green": (35, 159, 101)},
)


def _make_route(rng: random.Random, center_y: int) -> list[list[int]]:
    """Create an ordered, non-self-intersecting route from blue to orange."""

    reflection = -1 if rng.randrange(2) else 1
    vertical = -1 if rng.randrange(2) else 1
    a = rng.choice((13, 16, 18))
    b = rng.choice((10, 13, 16))
    c = rng.choice((7, 10, 12))
    xs = (-92, -66, -40, -14, 14, 40, 66, 92)
    ys = (0, 0, vertical * a, vertical * a, -vertical * b, -vertical * b, vertical * c, 0)
    center_x = 320 + rng.randrange(-10, 11)
    center_y = center_y + rng.randrange(-6, 7)
    return [[center_x + reflection * x, center_y + y] for x, y in zip(xs, ys, strict=True)]


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed)
    # Consecutive seeds cycle through both answer classes exactly: same, same,
    # opposite, opposite.  The first route's order changes independently.
    first = "red_green" if ((seed // 2) % 2 == 0) else "green_red"
    same = seed % 4 in (0, 1)
    second = first if same else ("green_red" if first == "red_green" else "red_green")
    routes = [_make_route(rng, 140), _make_route(rng, 500)]

    decorations: list[list[int]] = []
    while len(decorations) < 14:
        point = [rng.randrange(45, 596), rng.randrange(45, 596)]
        # Keep pale dots away from either decisive island.
        if all(math.dist(point, vertex) > 42 for route in routes for vertex in route):
            decorations.append(point)
    return {
        "canvas_size": 640,
        "route_vertices": routes,
        "checkpoint_orders": [first, second],
        "palette_variant": rng.randrange(len(PALETTES)),
        "decorations": decorations,
    }


def _scaled(point: list[int] | tuple[int, int], scale: int) -> tuple[int, int]:
    return int(point[0]) * scale, int(point[1]) * scale


def _ring(draw: ImageDraw.ImageDraw, center: tuple[int, int], radius: int, color: tuple[int, int, int], width: int, scale: int) -> None:
    cx, cy = center[0] * scale, center[1] * scale
    r = radius * scale
    draw.ellipse([cx - r, cy - r, cx + r, cy + r], outline=color, width=width * scale)


def _draw_route(draw: ImageDraw.ImageDraw, vertices: list[list[int]], order: str, palette: dict[str, tuple[int, int, int]], route_index: int, scale: int) -> None:
    points = [_scaled(point, scale) for point in vertices]
    navy = palette["navy"]
    draw.line(points, fill=navy, width=7 * scale, joint="curve")

    # Short dark stems make the station on the centerline explicit while
    # leaving the main route unbroken for tracing.  The two islands point their
    # markers inward, keeping all decisive evidence away from canvas edges.
    side = 1 if route_index == 0 else -1
    station_points = [vertices[2], vertices[5]]
    marker_centers: list[tuple[int, int]] = []
    for x, y in station_points:
        marker = (int(x), int(y) + side * 20)
        marker_centers.append(marker)
        draw.line([_scaled((x, y), scale), _scaled(marker, scale)], fill=navy, width=3 * scale)

    _ring(draw, (int(vertices[0][0]), int(vertices[0][1])), 11, palette["blue"], 4, scale)
    ex, ey = _scaled(vertices[-1], scale)
    er = 8 * scale
    draw.ellipse([ex - er, ey - er, ex + er, ey + er], fill=palette["orange"])

    colors = ("red", "green") if order == "red_green" else ("green", "red")
    for marker, color_name in zip(marker_centers, colors, strict=True):
        _ring(draw, marker, 8, palette[color_name], 4, scale)


def render(scene: dict) -> Image.Image:
    size = int(scene["canvas_size"])
    routes = scene["route_vertices"]
    orders = scene["checkpoint_orders"]
    palette = PALETTES[int(scene["palette_variant"])]
    decorations = scene["decorations"]
    scale = 4
    image = Image.new("RGB", (size * scale, size * scale), BG)
    draw = ImageDraw.Draw(image)

    # A pale dashed inset groups the visual world but never enters the oracle.
    x0, y0, x1, y1 = 188, 88, 452, 552
    dash = 12
    for x in range(x0, x1, 2 * dash):
        draw.line([_scaled((x, y0), scale), _scaled((min(x + dash, x1), y0), scale)], fill=FRAME, width=2 * scale)
        draw.line([_scaled((x, y1), scale), _scaled((min(x + dash, x1), y1), scale)], fill=FRAME, width=2 * scale)
    for y in range(y0, y1, 2 * dash):
        draw.line([_scaled((x0, y), scale), _scaled((x0, min(y + dash, y1)), scale)], fill=FRAME, width=2 * scale)
        draw.line([_scaled((x1, y), scale), _scaled((x1, min(y + dash, y1)), scale)], fill=FRAME, width=2 * scale)

    for x, y in decorations:
        cx, cy = x * scale, y * scale
        r = 2 * scale
        draw.ellipse([cx - r, cy - r, cx + r, cy + r], fill=DECOR)
    # Pale bars are nondiagnostic clutter and never touch a route.
    for x, y in ((112, 310), (510, 302), (131, 343), (492, 337)):
        draw.line([_scaled((x - 12, y), scale), _scaled((x + 12, y), scale)], fill=PALE, width=3 * scale)

    for index in range(2):
        _draw_route(draw, routes[index], orders[index], palette, index, scale)
    return image.resize((size, size), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    orders = scene["checkpoint_orders"]
    return "same_order" if orders[0] == orders[1] else "opposite_order"


def _polyline_length(points: list[list[int]]) -> float:
    return sum(math.dist(left, right) for left, right in zip(points, points[1:]))


def margin(scene: dict) -> float:
    routes = scene["route_vertices"]
    return float(min(_polyline_length(route[2:6]) for route in routes))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
