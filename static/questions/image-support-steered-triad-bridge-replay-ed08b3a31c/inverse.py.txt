"""Independent image-only inverse arm for the bridge replay raster.

The oracle uses only exact rendered colors, connected components, relative
component size, line aspect ratio, and command vertical order. It imports no
world-side latent or gold implementation.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

VIVID = ((214, 53, 70), (47, 103, 211), (232, 151, 35))
PALE = ((246, 190, 198), (181, 205, 247), (250, 220, 173))


def _components(mask: np.ndarray) -> list[tuple[int, int, int, float, float]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    output: list[tuple[int, int, int, float, float]] = []
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        queue = deque([(sy, sx)])
        seen[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        ys = [point[0] for point in points]
        xs = [point[1] for point in points]
        output.append(
            (
                len(points),
                max(xs) - min(xs) + 1,
                max(ys) - min(ys) + 1,
                float(sum(xs)) / len(xs),
                float(sum(ys)) / len(ys),
            )
        )
    return output


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    initial_by_palette: dict[int, int] = {}
    large_geometry: dict[int, tuple[int, int]] = {}
    commands: list[tuple[float, int, int]] = []

    for palette in range(3):
        large_seen = 0
        for state, rgb in ((1, VIVID[palette]), (0, PALE[palette])):
            mask = np.all(array == np.asarray(rgb, dtype=np.uint8), axis=2)
            for area, box_w, box_h, center_x, center_y in _components(mask):
                if area >= 85:
                    large_seen += 1
                    initial_by_palette[palette] = state
                    large_geometry[palette] = (box_w, box_h)
                elif 20 <= area <= 80 and box_w <= 10 and box_h <= 10:
                    commands.append((center_y, palette, state))
        if large_seen != 1:
            raise ValueError("each bridge color must have exactly one intact large component")

    if len(commands) != 4:
        raise ValueError("four intact command marks are required")
    commands.sort(key=lambda item: item[0])
    if any(commands[index + 1][0] - commands[index][0] < 5 for index in range(3)):
        raise ValueError("command order is ambiguous")

    direct = [
        palette
        for palette, (box_w, box_h) in large_geometry.items()
        if box_w >= 3 * box_h
    ]
    if len(direct) != 1:
        raise ValueError("the direct bottom bridge cannot be recovered")
    direct_palette = direct[0]
    sloped_palettes = [palette for palette in range(3) if palette != direct_palette]

    states = dict(initial_by_palette)
    for _, palette, state in commands:
        states[palette] = state
    connected = bool(
        states[direct_palette]
        or (states[sloped_palettes[0]] and states[sloped_palettes[1]])
    )
    return "yes" if connected else "no"
