"""Deterministic renderer for the compact notched-tile measurement world."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 512
PALETTES = (
    ((247, 246, 240), (13, 148, 160)),
    ((250, 248, 242), (18, 142, 155)),
    ((245, 247, 243), (10, 151, 145)),
)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    background, tile_color = PALETTES[rng.randrange(len(PALETTES))]
    # Seed parity deliberately counterbalances the answer in every adjacent pair.
    decision_yes = int(seed) % 2 == 0
    stem_width = rng.randint(28, 43)
    gap = rng.randint(10, 20)
    depth = stem_width + gap if decision_yes else stem_width - gap
    # Keep the smaller span comfortably visible and all dimensions local.
    if depth < 16:
        depth = 16
        stem_width = depth + gap
    width = stem_width + depth
    height = rng.randint(76, 104)
    top_arm = rng.randint(16, 25)
    bottom_arm = rng.randint(16, 25)
    notch_height = height - top_arm - bottom_arm
    left = rng.randint(38, CANVAS - width - 38)
    top = rng.randint(38, CANVAS - height - 38)
    return {
        "canvas": [CANVAS, CANVAS],
        "background": list(background),
        "tile_color": list(tile_color),
        "left": left,
        "top": top,
        "width": width,
        "height": height,
        "stem_width": stem_width,
        "notch_top": top + top_arm,
        "notch_height": notch_height,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    canvas = tuple(scene["canvas"])
    background = tuple(scene["background"])
    tile_color = tuple(scene["tile_color"])
    left = int(scene["left"])
    top = int(scene["top"])
    width = int(scene["width"])
    height = int(scene["height"])
    stem_width = int(scene["stem_width"])
    notch_top = int(scene["notch_top"])
    notch_height = int(scene["notch_height"])

    image = Image.new("RGB", canvas, background)
    draw = ImageDraw.Draw(image)
    # Half-open geometry expressed through Pillow's inclusive rectangle endpoints.
    draw.rectangle((left, top, left + width - 1, top + height - 1), fill=tile_color)
    draw.rectangle(
        (left + stem_width, notch_top, left + width - 1, notch_top + notch_height - 1),
        fill=background,
    )
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if int(scene["width"]) - int(scene["stem_width"]) > int(scene["stem_width"]) else "no"


def margin(scene: dict[str, Any]) -> float:
    return float(abs((int(scene["width"]) - int(scene["stem_width"])) - int(scene["stem_width"])))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
