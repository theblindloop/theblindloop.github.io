"""Independent pixel-only inverse for the compact notched tile.

The inverse segments saturated teal ink, validates a single left-aligned concave
component through its scanline profile, and compares the two recovered widths.
It has no access to latent scene values or renderer code.
"""

from __future__ import annotations

import numpy as np
from PIL import Image


class OracleAbstention(ValueError):
    pass


def _tile_mask(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3), array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)),
        axis=0,
    )
    background = np.median(border, axis=0)
    delta = np.linalg.norm(array - background, axis=2)
    red, green, blue = array[..., 0], array[..., 1], array[..., 2]
    return (delta > 55.0) & (green > red + 55) & (blue > red + 45)


def _measure(image: Image.Image) -> tuple[int, int]:
    mask = _tile_mask(image)
    ys, xs = np.nonzero(mask)
    if len(xs) < 1800:
        raise OracleAbstention("missing or too-small tile")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    width, height = x1 - x0 + 1, y1 - y0 + 1
    if not (44 <= width <= 130 and 58 <= height <= 120):
        raise OracleAbstention("tile extent is outside the public design range")

    sub = mask[y0 : y1 + 1, x0 : x1 + 1]
    counts: list[int] = []
    for row in sub:
        columns = np.flatnonzero(row)
        if len(columns) == 0 or int(columns[0]) != 0:
            raise OracleAbstention("scanline is empty or no longer shares the left stem edge")
        if len(columns) != int(columns[-1]) + 1:
            raise OracleAbstention("scanline contains a hole")
        counts.append(len(columns))

    full_width = max(counts)
    stem_width = min(counts)
    if full_width != width or not (14 <= stem_width < full_width - 8):
        raise OracleAbstention("notch widths are not recoverable")
    # The ideal tile has exactly one internal run of narrow rows and full rows elsewhere.
    if any(value not in {full_width, stem_width} for value in counts):
        raise OracleAbstention("scanline profile has more than two widths")
    narrow = [index for index, value in enumerate(counts) if value == stem_width]
    if not narrow or narrow != list(range(narrow[0], narrow[-1] + 1)):
        raise OracleAbstention("bite is not one rectangular run")
    if narrow[0] < 12 or height - narrow[-1] - 1 < 12 or len(narrow) < 20:
        raise OracleAbstention("bite or arms are too thin")
    return full_width, stem_width


def decision_from_image(image: Image.Image) -> str:
    full_width, stem_width = _measure(image)
    depth = full_width - stem_width
    if abs(depth - stem_width) < 7:
        raise OracleAbstention("comparison is too near the raster boundary")
    return "yes" if depth > stem_width else "no"
