from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 420, 360
START = (42, 180)
LINE_COLOR = (31, 38, 49)
BACKGROUND = (248, 247, 243)
PALETTE = {
    "green": (26, 166, 91),
    "red": (219, 61, 63),
    "blue": (48, 105, 215),
    "orange": (236, 145, 35),
}
LANE_Y = (70, 180, 290)
PERMS = tuple(itertools.permutations((0, 1, 2)))


def _lane_path(lane: int, tier: int) -> list[list[int]]:
    y = LANE_Y[lane]
    points: list[tuple[int, int]] = [START, (90, 180), (90, y)]
    if lane == 1:
        # This compulsory excursion equalizes the three lanes' base lengths.
        points += [(120, y), (120, 125), (160, 125), (160, y)]
        blocks = ((205, 235, 30), (270, 300, -30))
    else:
        outward = -30 if lane == 0 else 30
        blocks = ((155, 190, outward), (245, 280, outward))
    for a, b, dy in blocks[:tier]:
        points += [(a, y), (a, y + dy), (b, y + dy), (b, y)]
    points.append((377, y))
    # Remove consecutive duplicates while keeping a JSON-friendly representation.
    compact: list[list[int]] = []
    for point in points:
        item = [int(point[0]), int(point[1])]
        if not compact or compact[-1] != item:
            compact.append(item)
    return compact


def sample_scene(seed: int) -> dict:
    # Consecutive stress seeds traverse every route permutation; color binding
    # changes on a separate six-seed cadence to avoid a fixed color/length cue.
    tier_by_lane = PERMS[seed % len(PERMS)]
    color_order = PERMS[(seed // len(PERMS)) % len(PERMS)]
    colors = ("red", "blue", "orange")
    return {
        "branch_paths": [_lane_path(lane, tier_by_lane[lane]) for lane in range(3)],
        "terminal_colors": [colors[index] for index in color_order],
        "seed": int(seed),
    }


def _path_length(path: list[list[int]]) -> float:
    return sum(
        math.hypot(b[0] - a[0], b[1] - a[1])
        for a, b in zip(path, path[1:])
    )


def _lengths(scene: dict) -> list[float]:
    return [_path_length(path) for path in scene["branch_paths"]]


def analytic_gold(scene: dict) -> str:
    lengths = _lengths(scene)
    winner = max(range(3), key=lengths.__getitem__)
    return scene["terminal_colors"][winner]


def margin(scene: dict) -> float:
    ordered = sorted(_lengths(scene), reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 35.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["branch_paths"] = [list(reversed(path)) for path in scene["branch_paths"]]
    return [("reverse_polyline_storage", twin)]


def render(scene: dict) -> Image.Image:
    scale = 4
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for path in scene["branch_paths"]:
        # Canonicalize storage direction so the declared reversal alias is
        # raster-identical even at antialiased line caps.
        ordered = path if tuple(path[0]) <= tuple(path[-1]) else list(reversed(path))
        points = [(x * scale, y * scale) for x, y in ordered]
        draw.line(points, fill=LINE_COLOR, width=8 * scale, joint="curve")

    # Markers are painted last, visibly terminating the underlying routes.
    sx, sy = START
    r = 13
    draw.ellipse(
        ((sx - r) * scale, (sy - r) * scale, (sx + r) * scale, (sy + r) * scale),
        fill=PALETTE["green"],
        outline=(255, 255, 255),
        width=2 * scale,
    )
    for lane, color in enumerate(scene["terminal_colors"]):
        x, y = 377, LANE_Y[lane]
        draw.ellipse(
            ((x - r) * scale, (y - r) * scale, (x + r) * scale, (y + r) * scale),
            fill=PALETTE[color],
            outline=(255, 255, 255),
            width=2 * scale,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
