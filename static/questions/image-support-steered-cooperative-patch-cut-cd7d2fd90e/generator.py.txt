"""Deterministic sampler and renderer for cooperative patch cuts."""

from __future__ import annotations

import copy
import random
from typing import Any

import numpy as np
from PIL import Image

WIDTH = 360
HEIGHT = 300
CELL = 28
X0 = 12
Y0 = 38
TEAL = [23, 126, 137]
CYAN = [30, 190, 210]
ORANGE = [235, 139, 48]
BACKGROUND = [248, 248, 244]
EXPECTED_FOREGROUND_PIXELS = 82 * CELL * CELL


def _rect(x0: int, y0: int, x1: int, y1: int) -> list[int]:
    return [X0 + x0 * CELL, Y0 + y0 * CELL, X0 + x1 * CELL, Y0 + y1 * CELL]


def _flip_rect(rect: list[int], horizontal: bool, vertical: bool) -> list[int]:
    x0, y0, x1, y1 = rect
    if horizontal:
        x0, x1 = WIDTH - x1, WIDTH - x0
    if vertical:
        y0, y1 = HEIGHT - y1, HEIGHT - y0
    return [x0, y0, x1, y1]


def _mask(scene: dict[str, Any], key: str) -> np.ndarray:
    mask = np.zeros((int(scene["height"]), int(scene["width"])), dtype=bool)
    rectangles = scene[key] if key == "land_rectangles" else [scene[key]]
    for x0, y0, x1, y1 in rectangles:
        mask[int(y0):int(y1), int(x0):int(x1)] = True
    return mask


def _component_areas(mask: np.ndarray) -> list[int]:
    """Four-connected areas via unions of overlapping horizontal runs."""
    parents: list[int] = []
    run_areas: list[int] = []
    previous: list[tuple[int, int, int]] = []

    def root(label: int) -> int:
        while parents[label] != label:
            parents[label] = parents[parents[label]]
            label = parents[label]
        return label

    def union(left: int, right: int) -> None:
        left, right = root(left), root(right)
        if left != right:
            parents[right] = left

    for row in mask:
        padded = np.pad(row, (1, 1), constant_values=False)
        edges = np.flatnonzero(padded[1:] != padded[:-1])
        current: list[tuple[int, int, int]] = []
        for start, end in zip(edges[::2], edges[1::2], strict=True):
            label = len(parents)
            parents.append(label)
            run_areas.append(int(end - start))
            for old_start, old_end, old_label in previous:
                if old_start < end and start < old_end:
                    union(label, old_label)
            current.append((int(start), int(end), label))
        previous = current
    totals: dict[int, int] = {}
    for label, area in enumerate(run_areas):
        representative = root(label)
        totals[representative] = totals.get(representative, 0) + area
    return sorted(totals.values(), reverse=True)


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    positive = int(seed) % 2 == 0
    top_row = rng.choice((1, 2))
    bottom_row = rng.choice((5, 6))

    if positive:
        # Two 5x8 slabs with opposite corner notches (78 cells) and two
        # two-cell bridges (4 cells). Each colored cell cuts one bridge.
        land = [
            _rect(1, 0, 5, 1), _rect(0, 1, 5, 8),
            _rect(7, 0, 12, 7), _rect(7, 7, 11, 8),
            _rect(5, top_row, 7, top_row + 1),
            _rect(5, bottom_row, 7, bottom_row + 1),
        ]
        patch_a_col = 5 if (int(seed) // 2) % 2 == 0 else 6
        patch_b_col = 6 if patch_a_col == 5 else 5
        patch_a = _rect(patch_a_col, top_row, patch_a_col + 1, top_row + 1)
        patch_b = _rect(patch_b_col, bottom_row, patch_b_col + 1, bottom_row + 1)
    else:
        # Three 3/4/3-column slabs (80 cells), joined in series by exactly
        # two one-cell colored bridges. Either singleton deletion disconnects.
        land = [
            _rect(0, 0, 3, 8), _rect(4, 0, 8, 8), _rect(9, 0, 12, 8),
            _rect(3, top_row, 4, top_row + 1),
            _rect(8, bottom_row, 9, bottom_row + 1),
        ]
        patch_a = _rect(3, top_row, 4, top_row + 1)
        patch_b = _rect(8, bottom_row, 9, bottom_row + 1)

    if rng.random() < 0.5:
        patch_a, patch_b = patch_b, patch_a
    horizontal = bool((int(seed) // 4) % 2)
    vertical = bool((int(seed) // 8) % 2)
    land = [_flip_rect(rect, horizontal, vertical) for rect in land]
    patch_a = _flip_rect(patch_a, horizontal, vertical)
    patch_b = _flip_rect(patch_b, horizontal, vertical)
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "land_rectangles": land,
        "cyan_patch": patch_a,
        "orange_patch": patch_b,
        "teal": list(TEAL),
        "cyan": list(CYAN),
        "orange": list(ORANGE),
        "background": list(BACKGROUND),
    }


def render(scene: dict[str, Any]) -> Image.Image:
    land = _mask(scene, "land_rectangles")
    cyan = _mask(scene, "cyan_patch")
    orange = _mask(scene, "orange_patch")
    if np.any(cyan & ~land) or np.any(orange & ~land) or np.any(cyan & orange):
        raise ValueError("colored patches must be disjoint subsets of land")
    array = np.empty((int(scene["height"]), int(scene["width"]), 3), dtype=np.uint8)
    array[:] = np.asarray(scene["background"], dtype=np.uint8)
    array[land] = np.asarray(scene["teal"], dtype=np.uint8)
    array[cyan] = np.asarray(scene["cyan"], dtype=np.uint8)
    array[orange] = np.asarray(scene["orange"], dtype=np.uint8)
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict[str, Any]) -> str:
    land = _mask(scene, "land_rectangles")
    cyan = _mask(scene, "cyan_patch")
    orange = _mask(scene, "orange_patch")
    cyan_only = len(_component_areas(land & ~cyan))
    orange_only = len(_component_areas(land & ~orange))
    both = len(_component_areas(land & ~cyan & ~orange))
    return "yes" if cyan_only == 1 and orange_only == 1 and both > 1 else "no"


def margin(scene: dict[str, Any]) -> float:
    land = _mask(scene, "land_rectangles")
    cyan = _mask(scene, "cyan_patch")
    orange = _mask(scene, "orange_patch")
    areas = _component_areas(land & ~cyan & ~orange)
    smallest_piece_width = min(areas) ** 0.5 if areas else 0.0
    return float(min(CELL, smallest_piece_width))


def is_quarantined(scene: dict[str, Any]) -> bool:
    land = _mask(scene, "land_rectangles")
    cyan = _mask(scene, "cyan_patch")
    orange = _mask(scene, "orange_patch")
    return (
        int(land.sum()) != EXPECTED_FOREGROUND_PIXELS
        or int(cyan.sum()) != CELL * CELL
        or int(orange.sum()) != CELL * CELL
        or len(_component_areas(land)) != 1
        or margin(scene) < 18.0
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    twin = copy.deepcopy(scene)
    twin["land_rectangles"] = list(reversed(twin["land_rectangles"]))
    return [("reverse_land_rectangle_draw_order", twin)]
