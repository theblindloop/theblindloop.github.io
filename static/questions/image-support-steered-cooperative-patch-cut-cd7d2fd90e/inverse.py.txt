"""Independent image-only oracle for the cooperative cut predicate."""

from __future__ import annotations

import numpy as np
from PIL import Image

_TEAL = np.array([23, 126, 137], dtype=np.int16)
_CYAN = np.array([30, 190, 210], dtype=np.int16)
_ORANGE = np.array([235, 139, 48], dtype=np.int16)
_COLOR_TOLERANCE = 38.0
_EXPECTED_FOREGROUND = 82 * 28 * 28
_EXPECTED_PATCH = 28 * 28


def _near(array: np.ndarray, color: np.ndarray) -> np.ndarray:
    return np.linalg.norm(array.astype(np.int16) - color, axis=2) <= _COLOR_TOLERANCE


def _component_areas(mask: np.ndarray) -> list[int]:
    """Four-connected areas recovered from image-row runs."""
    parents: list[int] = []
    run_areas: list[int] = []
    previous: list[tuple[int, int, int]] = []

    def root(label: int) -> int:
        while parents[label] != label:
            parents[label] = parents[parents[label]]
            label = parents[label]
        return label

    def union(left: int, right: int) -> None:
        left, right = root(left), root(right)
        if left != right:
            parents[right] = left

    for row in mask:
        padded = np.pad(row, (1, 1), constant_values=False)
        edges = np.flatnonzero(padded[1:] != padded[:-1])
        current: list[tuple[int, int, int]] = []
        for start, end in zip(edges[::2], edges[1::2], strict=True):
            label = len(parents)
            parents.append(label)
            run_areas.append(int(end - start))
            for old_start, old_end, old_label in previous:
                if old_start < end and start < old_end:
                    union(label, old_label)
            current.append((int(start), int(end), label))
        previous = current
    totals: dict[int, int] = {}
    for label, area in enumerate(run_areas):
        representative = root(label)
        totals[representative] = totals.get(representative, 0) + area
    return sorted(totals.values(), reverse=True)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.shape != (300, 360, 3):
        raise ValueError("unexpected canvas dimensions")
    teal = _near(array, _TEAL)
    cyan = _near(array, _CYAN)
    orange = _near(array, _ORANGE)
    foreground = teal | cyan | orange
    if int(foreground.sum()) != _EXPECTED_FOREGROUND:
        raise ValueError("landmass has missing or extraneous area")
    if int(cyan.sum()) != _EXPECTED_PATCH or int(orange.sum()) != _EXPECTED_PATCH:
        raise ValueError("each colored patch must be one full visible cell")
    if np.any(cyan & orange):
        raise ValueError("colored patch masks overlap")
    if len(_component_areas(foreground)) != 1:
        raise ValueError("initial landmass is not connected")
    after_cyan = len(_component_areas(foreground & ~cyan))
    after_orange = len(_component_areas(foreground & ~orange))
    after_both = len(_component_areas(foreground & ~cyan & ~orange))
    return "yes" if after_cyan == 1 and after_orange == 1 and after_both > 1 else "no"
