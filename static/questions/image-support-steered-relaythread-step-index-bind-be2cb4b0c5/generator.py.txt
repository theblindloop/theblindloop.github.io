"""Relay-thread step-index world.

One short thin slate thread lies inside a small local patch of a cream board.
Small coloured marks and exactly one black GATE mark sit on the thread.  Walking
the thread from its capped end, the number of coloured marks met before the gate
is a step count k; the answer is the colour of the k-th coloured mark met after
the gate.  The count harvested on one stretch of the thread indexes a position on
the other stretch, so the answer exists only as a correspondence between the two
sides of one connected trace and can only be resolved by following its interior.
"""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "relaythread_step_index_bind-0.1.0"

CANVAS_W = 320
CANVAS_H = 288
BACKGROUND = (245, 240, 228)
THREAD_INK = (52, 62, 78)
GATE_INK = (12, 12, 12)

PITCH = 14
GRID_W = 8
GRID_H = 6
PATH_EDGES = 15
STROKE_HALF = 1          # 3-pixel thread
CAP_HALF = 4             # 9-pixel square cap
DECOR = 5                # decorative square side
CLEARANCE = 8            # decoration clearance from the thread patch

COLORS = {
    "red": (196, 46, 44),
    "orange": (226, 132, 32),
    "yellow": (222, 198, 46),
    "green": (46, 148, 74),
    "blue": (48, 96, 200),
    "purple": (136, 62, 176),
}
COLOR_NAMES = tuple(COLORS)
WARM = frozenset({"red", "orange", "yellow"})

QUARANTINE_MARGIN = 4.0
RUN_CLEARANCE = float(PITCH - (2 * STROKE_HALF + 1))   # 11.0 px between runs


# --------------------------------------------------------------------------- #
# lattice path: self-avoiding and self-non-touching, so runs stay 11px apart
# --------------------------------------------------------------------------- #
def _neighbours(v: tuple[int, int]) -> list[tuple[int, int]]:
    x, y = v
    out = []
    for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
        nx, ny = x + dx, y + dy
        if 0 <= nx < GRID_W and 0 <= ny < GRID_H:
            out.append((nx, ny))
    return out


def _sample_path(rng: random.Random) -> list[tuple[int, int]] | None:
    """Randomised backtracking search for a PATH_EDGES-edge non-touching path."""
    target = PATH_EDGES + 1
    start = (rng.randrange(GRID_W), rng.randrange(GRID_H))
    path = [start]
    visited = {start}
    budget = [40000]

    def ok(nxt: tuple[int, int], prev: tuple[int, int]) -> bool:
        if nxt in visited:
            return False
        for other in path:
            if other is prev or other == prev:
                continue
            if abs(other[0] - nxt[0]) + abs(other[1] - nxt[1]) <= 1:
                return False
        return True

    def dfs() -> bool:
        if len(path) == target:
            return True
        budget[0] -= 1
        if budget[0] <= 0:
            return False
        prev = path[-1]
        options = _neighbours(prev)
        rng.shuffle(options)
        for nxt in options:
            if not ok(nxt, prev):
                continue
            path.append(nxt)
            visited.add(nxt)
            if dfs():
                return True
            path.pop()
            visited.discard(nxt)
        return False

    return list(path) if dfs() else None


def _pixel(scene: dict, v: Iterable[int]) -> tuple[int, int]:
    gx, gy = v
    return (scene["ox"] + gx * PITCH, scene["oy"] + gy * PITCH)


def _edge_midpoint(scene: dict, edge: int) -> tuple[int, int]:
    a = _pixel(scene, scene["path"][edge])
    b = _pixel(scene, scene["path"][edge + 1])
    return ((a[0] + b[0]) // 2, (a[1] + b[1]) // 2)


# --------------------------------------------------------------------------- #
# reading orders
# --------------------------------------------------------------------------- #
def _marks_in_path_order(scene: dict) -> list[dict]:
    return [it for it in sorted(scene["items"], key=lambda it: it["edge"]) if it["kind"] == "mark"]


def _gate_edge(scene: dict) -> int:
    return next(it["edge"] for it in scene["items"] if it["kind"] == "gate")


def _answer_for_order(scene: dict, ordered: list[dict]) -> str | None:
    """Apply the world rule to an arbitrary reading order of the items."""
    gate_slot = next((i for i, it in enumerate(ordered) if it["kind"] == "gate"), None)
    if gate_slot is None:
        return None
    before = [it for it in ordered[:gate_slot] if it["kind"] == "mark"]
    after = [it for it in ordered[gate_slot + 1:] if it["kind"] == "mark"]
    k = len(before)
    if k < 1 or len(after) < k:
        return None
    return after[k - 1]["color"]


def _path_order(scene: dict) -> list[dict]:
    return sorted(scene["items"], key=lambda it: it["edge"])


def _coordinate_order(scene: dict) -> list[dict]:
    return sorted(scene["items"], key=lambda it: (_edge_midpoint(scene, it["edge"])[1],
                                                  _edge_midpoint(scene, it["edge"])[0]))


def _column_order(scene: dict) -> list[dict]:
    return sorted(scene["items"], key=lambda it: (_edge_midpoint(scene, it["edge"])[0],
                                                  _edge_midpoint(scene, it["edge"])[1]))


def _radial_order(scene: dict) -> list[dict]:
    cap = _pixel(scene, scene["path"][0])

    def key(it: dict) -> float:
        mx, my = _edge_midpoint(scene, it["edge"])
        return (mx - cap[0]) ** 2 + (my - cap[1]) ** 2

    return sorted(scene["items"], key=key)


# --------------------------------------------------------------------------- #
# sampling
# --------------------------------------------------------------------------- #
def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed * 7919 + 13)
    hard = (seed % 4) == 3
    answer_target = COLOR_NAMES[seed % len(COLOR_NAMES)]

    for _ in range(4000):
        path = _sample_path(rng)
        if path is None:
            continue
        mark_size = 11 if hard else rng.choice((7, 7, 9))
        gate_step = 1 if hard else rng.choice((1, 2))
        k = rng.choice((2, 3))
        n_after = k + rng.choice((1, 2))
        n_items = k + 1 + n_after
        # index gaps: 2 between marks, gate_step on both sides of the gate
        gaps = []
        for slot in range(n_items - 1):
            gaps.append(gate_step if slot in (k - 1, k) else 2)
        span = sum(gaps)
        if span > PATH_EDGES - 2:
            continue
        # the first mark never sits on edge 0: it must not paint over the cap
        start = rng.randrange(1, PATH_EDGES - span)
        edges = [start]
        for gap in gaps:
            edges.append(edges[-1] + gap)

        colors = [rng.choice(COLOR_NAMES) for _ in range(k + n_after)]
        colors[2 * k - 1] = answer_target
        counts = {name: colors.count(name) for name in COLOR_NAMES}
        if counts[answer_target] < 2:
            continue
        if max(counts[n] for n in COLOR_NAMES if n != answer_target) < counts[answer_target]:
            continue        # the answer colour is never the strict plurality
        if answer_target in (colors[0], colors[k - 1], colors[k], colors[-1]):
            continue        # first, last-before-gate, first-after-gate, last are distractors

        items = []
        ci = 0
        for slot, edge in enumerate(edges):
            if slot == k:
                items.append({"edge": edge, "kind": "gate", "color": None, "size": mark_size})
            else:
                items.append({"edge": edge, "kind": "mark", "color": colors[ci], "size": mark_size})
                ci += 1

        ox = rng.randrange(12, CANVAS_W - (GRID_W - 1) * PITCH - 12)
        oy = rng.randrange(12, CANVAS_H - (GRID_H - 1) * PITCH - 12)
        scene = {"path": [list(v) for v in path], "items": items, "ox": ox, "oy": oy,
                 "mark_size": mark_size, "gate_step": gate_step, "step_count": k,
                 "decor": [], "cap_end": 0}

        answer = _answer_for_order(scene, _path_order(scene))
        if answer != answer_target:
            continue
        # cheap coordinate reading orders must not reproduce the answer
        if any(_answer_for_order(scene, order) == answer
               for order in (_coordinate_order(scene), _column_order(scene), _radial_order(scene))):
            continue

        scene["decor"] = _sample_decor(rng, scene)
        return scene
    raise RuntimeError(f"scene sampling failed for seed {seed}")


def _patch_rect(scene: dict) -> tuple[int, int, int, int]:
    xs = [_pixel(scene, v)[0] for v in scene["path"]]
    ys = [_pixel(scene, v)[1] for v in scene["path"]]
    pad = CAP_HALF + scene["mark_size"] // 2 + CLEARANCE
    return (min(xs) - pad, min(ys) - pad, max(xs) + pad, max(ys) + pad)


def _sample_decor(rng: random.Random, scene: dict) -> list[dict]:
    x0, y0, x1, y1 = _patch_rect(scene)
    inks = list(COLOR_NAMES) + ["gate", "thread"]
    decor: list[dict] = []
    placed: list[tuple[int, int]] = []
    attempts = 0
    want = rng.randrange(16, 25)
    while len(decor) < want and attempts < 4000:
        attempts += 1
        cx = rng.randrange(DECOR, CANVAS_W - DECOR)
        cy = rng.randrange(DECOR, CANVAS_H - DECOR)
        if x0 - DECOR <= cx <= x1 + DECOR and y0 - DECOR <= cy <= y1 + DECOR:
            continue
        if any(abs(cx - px) < 3 * DECOR and abs(cy - py) < 3 * DECOR for px, py in placed):
            continue
        ink = inks[len(decor) % len(inks)] if len(decor) < len(inks) else rng.choice(inks)
        decor.append({"x": cx, "y": cy, "ink": ink})
        placed.append((cx, cy))
    return decor


# --------------------------------------------------------------------------- #
# rendering
# --------------------------------------------------------------------------- #
def _ink(name: str) -> tuple[int, int, int]:
    if name == "gate":
        return GATE_INK
    if name == "thread":
        return THREAD_INK
    return COLORS[name]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (CANVAS_W, CANVAS_H), BACKGROUND)
    draw = ImageDraw.Draw(image)

    for spot in scene["decor"]:
        half = DECOR // 2
        draw.rectangle([spot["x"] - half, spot["y"] - half, spot["x"] + half, spot["y"] + half],
                       fill=_ink(spot["ink"]))

    path = scene["path"]
    for i in range(len(path) - 1):
        ax, ay = _pixel(scene, path[i])
        bx, by = _pixel(scene, path[i + 1])
        draw.rectangle([min(ax, bx) - STROKE_HALF, min(ay, by) - STROKE_HALF,
                        max(ax, bx) + STROKE_HALF, max(ay, by) + STROKE_HALF], fill=THREAD_INK)

    cx, cy = _pixel(scene, path[0])
    draw.rectangle([cx - CAP_HALF, cy - CAP_HALF, cx + CAP_HALF, cy + CAP_HALF], fill=THREAD_INK)

    for item in sorted(scene["items"], key=lambda it: it["edge"]):
        mx, my = _edge_midpoint(scene, item["edge"])
        half = item["size"] // 2
        ink = GATE_INK if item["kind"] == "gate" else COLORS[item["color"]]
        draw.rectangle([mx - half, my - half, mx + half, my + half], fill=ink)
    return image


# --------------------------------------------------------------------------- #
# gold, margin, quarantine, aliases
# --------------------------------------------------------------------------- #
def analytic_gold(scene: dict) -> str:
    answer = _answer_for_order(scene, _path_order(scene))
    if answer is None:
        raise ValueError("scene does not determine an answer")
    return answer


def margin(scene: dict) -> float:
    """Smallest visible clearance in pixels on the decisive thread structure.

    The cream gap between the black gate and its neighbouring coloured mark is
    the decision boundary: when it closes, which side of the gate a mark lies on
    is no longer readable and the step count k changes.  By construction that
    gate gap is the minimum, except that no two thread runs ever come closer
    than RUN_CLEARANCE.
    """
    ordered = _path_order(scene)
    gaps = [(ordered[i + 1]["edge"] - ordered[i]["edge"]) * PITCH - scene["mark_size"]
            for i in range(len(ordered) - 1)]
    return float(min(gaps + [RUN_CLEARANCE]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical re-indexings of the stored item list.

    The raster and the answer depend on the set of items and on the edge each
    one sits on, never on the slot each item occupies in the stored list, so
    rotating or reversing that list is a genuine latent alias.
    """
    items = scene["items"]
    out: list[tuple[str, dict]] = []
    if len(items) > 1:
        rotated = dict(scene)
        rotated["items"] = items[1:] + items[:1]
        out.append(("item_list_rotation", rotated))
        reversed_scene = dict(scene)
        reversed_scene["items"] = list(reversed(items))
        out.append(("item_list_reversal", reversed_scene))
        shuffled = dict(scene)
        shuffled["decor"] = list(reversed(scene["decor"]))
        out.append(("decor_list_reversal", shuffled))
    return out
