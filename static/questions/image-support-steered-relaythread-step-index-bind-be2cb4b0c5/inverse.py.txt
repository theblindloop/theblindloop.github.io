"""Pixel-only inverse arm for the relay-thread step-index world.

Input: the final PNG only.  The public question contract states the palette
names, that the thread is one thin slate trace with one thickened capped end,
that the single black mark on the thread is the gate, and that marks not touching
the thread are decoration.  Those contract facts are hardcoded here; no renderer
code, latent scene, coordinate table, filename or metadata is read.

Procedure: segment inks, find the thickened cap, rebuild the drawn trace as the
connected component of thread ink plus the marks that touch it, walk that
component with a geodesic breadth-first search from the cap, order the marks by
geodesic distance, count the marks before the gate and report the colour of the
mark that many steps after it.  Anything unreadable raises.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

THREAD_INK = (52, 62, 78)
GATE_INK = (12, 12, 12)
PALETTE = {
    "red": (196, 46, 44),
    "orange": (226, 132, 32),
    "yellow": (222, 198, 46),
    "green": (46, 148, 74),
    "blue": (48, 96, 200),
    "purple": (136, 62, 176),
}
CAP_WINDOW = 7          # the cap is the only place a full 7x7 slate window fits
MIN_MARK_AREA = 40      # decorative squares on the bare board are far smaller


class Abstain(ValueError):
    """Raised when the picture does not determine an answer."""


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return np.all(arr == np.asarray(rgb, dtype=np.uint8), axis=-1)


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    seen = np.zeros(mask.shape, dtype=bool)
    out: list[list[tuple[int, int]]] = []
    ys, xs = np.nonzero(mask)
    for sy, sx in zip(ys.tolist(), xs.tolist()):
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        blob: list[tuple[int, int]] = []
        while stack:
            y, x = stack.pop()
            blob.append((y, x))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < mask.shape[0] and 0 <= nx < mask.shape[1] \
                        and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        out.append(blob)
    return out


def _touches(blob: list[tuple[int, int]], target: np.ndarray) -> bool:
    h, w = target.shape
    for y, x in blob:
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < h and 0 <= nx < w and target[ny, nx]:
                return True
    return False


def _cap_pixel(thread: np.ndarray) -> tuple[int, int]:
    pad = CAP_WINDOW
    integral = np.cumsum(np.cumsum(thread.astype(np.int32), axis=0), axis=1)
    integral = np.pad(integral, ((1, 0), (1, 0)))
    h, w = thread.shape
    box = (integral[pad:, pad:] - integral[:h - pad + 1, pad:]
           - integral[pad:, :w - pad + 1] + integral[:h - pad + 1, :w - pad + 1])
    ys, xs = np.nonzero(box == pad * pad)
    if ys.size == 0:
        raise Abstain("no thickened thread end (cap) found")
    if ys.max() - ys.min() > 6 or xs.max() - xs.min() > 6:
        raise Abstain("more than one thickened thread end")
    cy = int(round(ys.mean())) + pad // 2
    cx = int(round(xs.mean())) + pad // 2
    if not thread[cy, cx]:
        raise Abstain("cap centre is not thread ink")
    return cy, cx


def _geodesic(mask: np.ndarray, start: tuple[int, int]) -> np.ndarray:
    dist = np.full(mask.shape, -1, dtype=np.int32)
    dist[start] = 0
    frontier = [start]
    h, w = mask.shape
    while frontier:
        nxt = []
        for y, x in frontier:
            d = dist[y, x] + 1
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and dist[ny, nx] < 0:
                    dist[ny, nx] = d
                    nxt.append((ny, nx))
        frontier = nxt
    return dist


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    thread = _mask(arr, THREAD_INK)
    if not thread.any():
        raise Abstain("no thread ink in the picture")

    cap = _cap_pixel(thread)
    # the marks paint over the stretches of thread they straddle, so the raw
    # thread ink is in pieces; the marks are what bridges them back together
    trace = thread.copy()

    marks: list[dict] = []
    for name, rgb in list(PALETTE.items()) + [("gate", GATE_INK)]:
        for blob in _components(_mask(arr, rgb)):
            if len(blob) < MIN_MARK_AREA or not _touches(blob, thread):
                continue        # decoration off the thread, or too small to be a mark
            marks.append({"name": name, "blob": blob})
            for y, x in blob:
                trace[y, x] = True

    full = _components(trace)
    reach = next((blob for blob in full if cap in blob), None)
    if reach is None:
        raise Abstain("cap is not on the reconstructed trace")
    walkable = np.zeros(trace.shape, dtype=bool)
    for y, x in reach:
        walkable[y, x] = True

    dist = _geodesic(walkable, cap)
    ordered = []
    for mark in marks:
        values = [dist[y, x] for y, x in mark["blob"] if dist[y, x] >= 0]
        if not values:
            raise Abstain("a mark on the thread cannot be reached from the capped end")
        ordered.append((int(min(values)), mark["name"]))
    ordered.sort()

    gates = [i for i, (_, name) in enumerate(ordered) if name == "gate"]
    if len(gates) != 1:
        raise Abstain(f"expected exactly one gate on the thread, found {len(gates)}")
    slot = gates[0]
    before = [name for _, name in ordered[:slot]]
    after = [name for _, name in ordered[slot + 1:]]
    if any(name == "gate" for name in before + after):
        raise Abstain("more than one gate on the thread")
    steps = len(before)
    if steps < 1:
        raise Abstain("no coloured mark before the gate")
    if len(after) < steps:
        raise Abstain("not enough coloured marks after the gate")
    return after[steps - 1]
