"""Independent PNG-only inverse arm for the chromatic mosaic."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID = 7
CELL = 28
BORDER = 8
SIDE = 212
FRAME_RGB = np.array([92, 50, 132], dtype=np.int16)
NAMES = ("X", "plus", "T", "L")
PATTERNS = {
    "X": ("1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"),
    "plus": ("0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"),
    "T": ("1111111", "0001000", "0001000", "0001000", "0001000", "0001000", "0001000"),
    "L": ("1000000", "1000000", "1000000", "1000000", "1000000", "1000000", "1111111"),
}
CYCLIC_RANKS = {(0, 1, 2), (2, 0, 1), (1, 2, 0)}


def _frame_box(array: np.ndarray) -> tuple[int, int] | None:
    delta = array.astype(np.int16) - FRAME_RGB
    mask = np.max(np.abs(delta), axis=2) <= 8
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        return None
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (SIDE - 2 <= x1 - x0 + 1 <= SIDE + 2 and SIDE - 2 <= y1 - y0 + 1 <= SIDE + 2):
        return None
    return x0, y0


def _template(name: str) -> np.ndarray:
    return np.array([[int(value) for value in row] for row in PATTERNS[name]], dtype=np.uint8)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    located = _frame_box(array)
    if located is None:
        return "__abstain__"
    x0, y0 = located
    displayed = np.zeros((GRID, GRID, 3), dtype=np.float64)
    for y in range(GRID):
        for x in range(GRID):
            left = x0 + BORDER + x * CELL
            top = y0 + BORDER + y * CELL
            patch = array[top + 6:top + CELL - 6, left + 6:left + CELL - 6]
            if patch.size == 0:
                return "__abstain__"
            displayed[y, x] = np.median(patch.reshape(-1, 3), axis=0)

    # The declared toroidal registrations: red right, green down, blue fixed.
    registered = np.empty_like(displayed)
    registered[:, :, 0] = np.roll(displayed[:, :, 0], shift=1, axis=1)
    registered[:, :, 1] = np.roll(displayed[:, :, 1], shift=1, axis=0)
    registered[:, :, 2] = displayed[:, :, 2]
    output = np.zeros((GRID, GRID), dtype=np.uint8)
    for y in range(GRID):
        for x in range(GRID):
            values = registered[y, x]
            if min(abs(float(values[a]) - float(values[b])) for a in range(3) for b in range(a)) < 28:
                return "__abstain__"
            ordering = np.argsort(values)
            ranks = [0, 0, 0]
            for rank, channel in enumerate(ordering.tolist()):
                ranks[channel] = rank
            output[y, x] = int(tuple(ranks) in CYCLIC_RANKS)

    scored = sorted((int(np.sum(output == _template(name))), name) for name in NAMES)
    if scored[-1][0] - scored[-2][0] < 4:
        return "__abstain__"
    return scored[-1][1]
