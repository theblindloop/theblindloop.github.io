"""Deterministic renderer and analytic arm for chromatic rank registration."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw

CANVAS = 448
GRID = 7
CELL = 28
BORDER = 8
SIDE = GRID * CELL + 2 * BORDER
VIOLET = (92, 50, 132)
BACKGROUND = (247, 245, 239)
LEVELS = (54, 137, 220)
SYMBOLS = ("X", "plus", "T", "L")

GLYPHS = {
    "X": (
        "1000001", "0100010", "0010100", "0001000", "0010100", "0100010", "1000001"
    ),
    "plus": (
        "0001000", "0001000", "0001000", "1111111", "0001000", "0001000", "0001000"
    ),
    "T": (
        "1111111", "0001000", "0001000", "0001000", "0001000", "0001000", "0001000"
    ),
    "L": (
        "1000000", "1000000", "1000000", "1000000", "1000000", "1000000", "1111111"
    ),
}

CYCLIC = ((0, 1, 2), (2, 0, 1), (1, 2, 0))
ANTICYCLIC = ((0, 2, 1), (1, 0, 2), (2, 1, 0))
DECOR_COLORS = ((230, 172, 74), (77, 158, 154), (208, 113, 121), (128, 154, 203))


def _target(symbol: str) -> list[list[int]]:
    return [[int(value) for value in row] for row in GLYPHS[symbol]]


def _derived_rgb(permutations: list[list[list[int]]], y: int, x: int) -> tuple[int, int, int]:
    """Compose displayed channels whose declared shifts recover permutations[y][x]."""
    red_rank = permutations[y][(x + 1) % GRID][0]
    green_rank = permutations[(y + 1) % GRID][x][1]
    blue_rank = permutations[y][x][2]
    return (LEVELS[red_rank], LEVELS[green_rank], LEVELS[blue_rank])


def _decoded_without_registration(permutations: list[list[list[int]]]) -> list[list[int]]:
    decoded = []
    for y in range(GRID):
        row = []
        for x in range(GRID):
            rgb = _derived_rgb(permutations, y, x)
            order = tuple(sorted(range(3), key=lambda index: rgb[index]))
            # Convert ordering of channel indices into their rank tuple.
            ranks = [0, 0, 0]
            for rank, channel in enumerate(order):
                ranks[channel] = rank
            row.append(int(tuple(ranks) in CYCLIC))
        decoded.append(row)
    return decoded


def _score(mask: list[list[int]], symbol: str) -> int:
    target = _target(symbol)
    return sum(mask[y][x] == target[y][x] for y in range(GRID) for x in range(GRID))


def _registered_mask(permutations: list[list[list[int]]]) -> list[list[int]]:
    return [[int(tuple(permutations[y][x]) in CYCLIC) for x in range(GRID)] for y in range(GRID)]


def _classify(mask: list[list[int]]) -> tuple[str, float]:
    scores = sorted((_score(mask, name), name) for name in SYMBOLS)
    return scores[-1][1], float(scores[-1][0] - scores[-2][0])


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random((int(seed) + 11) * 1000003)
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    target = _target(symbol)

    # Rejection prevents the already-composited display from accidentally naming the answer.
    for _ in range(500):
        permutations = []
        for y in range(GRID):
            row = []
            for x in range(GRID):
                family = CYCLIC if target[y][x] else ANTICYCLIC
                row.append(list(family[rng.randrange(3)]))
            permutations.append(row)
        raw = _decoded_without_registration(permutations)
        raw_scores = sorted((_score(raw, name), name) for name in SYMBOLS)
        if raw_scores[-1][1] != symbol or raw_scores[-1][0] - raw_scores[-2][0] <= 1:
            break
    else:
        raise RuntimeError("could not suppress unregistered shortcut")

    badge_x = 106 + rng.randrange(-12, 13)
    badge_y = 112 + rng.randrange(-12, 13)
    decorations = []
    while len(decorations) < 14:
        x = rng.randrange(24, CANVAS - 24)
        y = rng.randrange(24, CANVAS - 24)
        radius = rng.randrange(3, 7)
        if badge_x - 22 <= x <= badge_x + SIDE + 22 and badge_y - 22 <= y <= badge_y + SIDE + 22:
            continue
        decorations.append([x, y, radius, rng.randrange(len(DECOR_COLORS))])
    return {
        "seed": int(seed),
        "permutations": permutations,
        "badge_x": badge_x,
        "badge_y": badge_y,
        "decorations": decorations,
    }


def render(scene: dict[str, Any]) -> Image.Image:
    image = Image.new("RGB", (CANVAS, CANVAS), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius, color_index in scene["decorations"]:
        color = DECOR_COLORS[int(color_index)]
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)

    x0 = int(scene["badge_x"])
    y0 = int(scene["badge_y"])
    draw.rectangle((x0, y0, x0 + SIDE - 1, y0 + SIDE - 1), fill=VIOLET)
    permutations = scene["permutations"]
    for y in range(GRID):
        for x in range(GRID):
            color = _derived_rgb(permutations, y, x)
            left = x0 + BORDER + x * CELL
            top = y0 + BORDER + y * CELL
            draw.rectangle((left, top, left + CELL - 1, top + CELL - 1), fill=color)

    # Four visible registration fiducials are cut into the frame corners.
    pip = (238, 196, 82)
    for px, py in ((x0 + 2, y0 + 2), (x0 + SIDE - 6, y0 + 2),
                   (x0 + 2, y0 + SIDE - 6), (x0 + SIDE - 6, y0 + SIDE - 6)):
        draw.rectangle((px, py, px + 3, py + 3), fill=pip)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    decision, _ = _classify(_registered_mask(scene["permutations"]))
    return decision


def margin(scene: dict[str, Any]) -> float:
    _, gap = _classify(_registered_mask(scene["permutations"]))
    return gap


def is_quarantined(scene: dict[str, Any]) -> bool:
    if margin(scene) < 4.0:
        return True
    try:
        return any(sorted(cell) != [0, 1, 2] for row in scene["permutations"] for cell in row)
    except (KeyError, TypeError):
        return True


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []
