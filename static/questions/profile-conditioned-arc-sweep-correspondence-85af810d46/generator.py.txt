"""Arc-sweep gauge correspondence renderer (paper-semantic family).

A cross-panel angular-sweep correspondence task. A neutral gray "compass" gauge
is shown at the top (the reference): a dark pivot hub at the object's center
with a thick gray annular arc that sweeps out a specific central angle
(arc length) taken from a finite canonical set. Below a divider sit three
colored gauges A, B, C (red / green / blue), each a pivot hub with a thick
colored annular arc of a distinct sweep. Exactly one candidate has the same arc
sweep as the reference. The question asks for that candidate's color.

The pixel-only inverse arm must, for every panel, recover the gauge's pivot hub
and the angular extent of its colored arc around that hub, establish the
cross-panel correspondence by arc-sweep equality, and only then read out the
bound color of the single matched gauge.

Latent state z (every field that influences the gold is rendered -- nothing
hidden):

    ref_sweep  : central angle (degrees) swept by the gray reference arc, drawn
                 from the canonical set CANONICAL_SWEEPS.
    cand_sweeps: length-3 list of candidate arc sweeps; exactly one equals
                 ``ref_sweep``, the other two are distinct and different.
    cand_colors: the fixed 3-color palette bound one-to-one to the candidates.
                 Each candidate gauge's arc is drawn in its bound color.
    arc_start  : a per-object random start angle (degrees) for every arc so that
                 the absolute orientation of the sweep is counterbalanced and
                 only its angular extent (not its direction) is meaningful.
    width, height, divider_y, ref_center, cand_centers, hub_r, arc_r,
    arc_width : fixed geometry.

The analytic gold is the color of the candidate whose arc sweep equals the
reference's. Because the reference sweep, each candidate sweep, and the bound
colors are all rendered verbatim, the gold is recoverable from the pixels alone.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "arc_sweep_correspondence-0.1.0"

BACKGROUND = "white"
DIVIDER = (70, 70, 70)
GRAY = (150, 150, 150)
HUB = (30, 30, 30)

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
CAND_COLOR_ORDER = ["red", "green", "blue"]

# Canonical angular sweeps (degrees) an arc may span. Adjacent values are
# separated by 45 degrees, far above the finite-resolution oracle tolerance.
CANONICAL_SWEEPS = [60, 105, 150, 195, 240]

WIDTH = 900
HEIGHT = 760
DIVIDER_Y = 195
DIVIDER_THICK = 3
HUB_R = 14          # radius of the dark pivot hub at each object's center
ARC_R = 84          # midline radius of the annular arc stroke
ARC_WIDTH = 16      # stroke thickness; arc band spans [ARC_R-8, ARC_R+8]
REF_CENTER = (450, 100)
CAND_CENTERS = [(450, 330), (450, 470), (450, 610)]

_QUARANTINE_FLOOR = 30.0  # degrees


def _arc_box(center) -> tuple[float, float, float, float]:
    cx, cy = center
    return (cx - ARC_R, cy - ARC_R, cx + ARC_R, cy + ARC_R)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic latent scene (dict)."""
    rng = random.Random(seed)
    ref_sweep = rng.choice(CANONICAL_SWEEPS)
    pool = [s for s in CANONICAL_SWEEPS if s != ref_sweep]
    rng.shuffle(pool)
    distractors = pool[:2]
    cand_sweeps = [ref_sweep] + distractors
    rng.shuffle(cand_sweeps)

    starts = [rng.uniform(0.0, 55.0) for _ in range(4)]

    return {
        "ref_sweep": ref_sweep,
        "cand_sweeps": cand_sweeps,
        "cand_colors": list(PALETTE.keys()),
        "arc_start": starts,
        "ref_center": list(REF_CENTER),
        "cand_centers": [[c[0], c[1]] for c in CAND_CENTERS],
        "hub_r": HUB_R,
        "arc_r": ARC_R,
        "arc_width": ARC_WIDTH,
        "divider_y": DIVIDER_Y,
        "width": WIDTH,
        "height": HEIGHT,
    }


def _draw_gauge(draw, center, sweep: float, start: float, arc_color, hub_color) -> None:
    cx, cy = center
    xy = _arc_box(center)
    draw.arc(xy, start=start, end=start + sweep, fill=arc_color, width=ARC_WIDTH)
    draw.ellipse([cx - HUB_R, cy - HUB_R, cx + HUB_R, cy + HUB_R], fill=hub_color)


def render(scene: dict) -> Image.Image:
    """Render I = R(z) deterministically at pixel resolution."""
    w = int(scene["width"])
    h = int(scene["height"])
    img = Image.new("RGB", (w, h), BACKGROUND)
    draw = ImageDraw.Draw(img)
    _draw_gauge(
        draw, tuple(scene["ref_center"]), scene["ref_sweep"], scene["arc_start"][0], GRAY, HUB
    )
    for i, center in enumerate(scene["cand_centers"]):
        color = scene["cand_colors"][i]
        _draw_gauge(
            draw, tuple(center), scene["cand_sweeps"][i], scene["arc_start"][1 + i], PALETTE[color], HUB
        )
    draw.rectangle([0, scene["divider_y"], w, scene["divider_y"] + DIVIDER_THICK], fill=DIVIDER)
    return img


def analytic_gold(scene: dict) -> str:
    """Decision: color (red/green/blue) of the gauge with the reference sweep."""
    idx = scene["cand_sweeps"].index(scene["ref_sweep"])
    return scene["cand_colors"][idx]


def margin(scene: dict) -> float:
    """Runner-up margin: minimum absolute sweep difference (degrees) between the
    reference gauge and any distractor candidate."""
    ref = scene["ref_sweep"]
    diffs = [abs(ref - s) for s in scene["cand_sweeps"] if s != ref]
    return float(min(diffs)) if diffs else 0.0


def is_quarantined(scene: dict) -> bool:
    """Quarantine only an ambiguous scene where a distractor comes within
    _QUARANTINE_FLOOR degrees of the reference sweep (never produced because
    canonical sweeps are separated by 45 degrees)."""
    return margin(scene) < _QUARANTINE_FLOOR


def latent_symmetries(scene: dict) -> list:
    """not_applicable: every label-influencing field (reference sweep, each
    candidate sweep, and the bound colors) is rendered verbatim, so the raster
    uniquely determines the gold; no pixel-identical transform exists."""
    return []
