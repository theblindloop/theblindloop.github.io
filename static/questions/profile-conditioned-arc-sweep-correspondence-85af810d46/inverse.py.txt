"""Independent pixel-only inverse arm for the arc-sweep gauge correspondence.

Receives ONLY the final rasterized PNG and the public question contract (this
file's own visual spec). It never imports the renderer, prompts, generator,
verifier, or analytic-gold code, and never reads scene data.

Public visual spec:

  * the top panel holds one neutral gray "compass" gauge (the reference): a
    dark pivot hub at its center with a thick gray annular arc stroke that
    sweeps out a central angle from the canonical set {60,105,150,195,240};
  * a horizontal dark divider separates it from a column of three colored
    gauges, one red, one green, one blue, each a dark pivot hub with a thick
    colored annular arc of a distinct sweep;
  * the task: decide which gauge color has exactly the same arc sweep as the
    gray reference.

The oracle, from pixels alone: finds the divider, locates the four dark pivot
hubs (one reference above the divider, three candidates below), and for each
hub measures the angular extent of its surrounding arc by scanning a circle of
the arc's midline radius and measuring the covered fraction of angles, binned
to the nearest canonical sweep. It returns the color of the candidate gauge
whose binned sweep equals the reference's.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

ORACLE_VERSION = "arc_sweep_correspondence-oracle-0.1.0"

PALETTE: dict[str, tuple[int, int, int]] = {
    "red": (235, 40, 40),
    "green": (0, 170, 70),
    "blue": (40, 60, 235),
}
GRAY = (150, 150, 150)
CANONICAL_SWEEPS = [60, 105, 150, 195, 240]
ARC_R = 84.0
_SAMPLE_RADII = (78.0, 84.0, 90.0)
_N_DARK = 90.0
_COLOR_DIST = 70.0
_SAMPLES = 720


def _classify(arr: np.ndarray, color: tuple[int, int, int]) -> np.ndarray:
    c = np.asarray(color, dtype=np.float32)
    d = np.linalg.norm(arr - c[None, None, :], axis=-1)
    return d < _COLOR_DIST


def _label_components(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    labels = np.zeros((h, w), dtype=np.int32)
    comps: dict[int, list] = {}
    cur = 0
    for i in range(h):
        for j in range(w):
            if not mask[i, j] or labels[i, j]:
                continue
            cur += 1
            labels[i, j] = cur
            stack = [(i, j)]
            rows = []
            cols = []
            while stack:
                r, c = stack.pop()
                rows.append(r)
                cols.append(c)
                for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nr, nc = r + dr, c + dc
                    if 0 <= nr < h and 0 <= nc < w and mask[nr, nc] and labels[nr, nc] == 0:
                        labels[nr, nc] = cur
                        stack.append((nr, nc))
            comps[cur] = {"rows": rows, "cols": cols}
    hubs = []
    for comp in comps.values():
        rows = comp["rows"]
        cols = comp["cols"]
        rmin, rmax = min(rows), max(rows)
        cmin, cmax = min(cols), max(cols)
        area = len(rows)
        if 80 < area < 6000 and (rmax - rmin) < 60 and (cmax - cmin) < 60:
            hubs.append((float(np.mean(rows)), float(np.mean(cols))))
    return hubs


def _covered_any(mask: np.ndarray, center, angle_rad: float) -> bool:
    """True if any radial sample inside the arc band at ``angle_rad`` is colored."""
    cx, cy = center
    h, w = mask.shape
    for rr in _SAMPLE_RADII:
        r = int(round(cy + rr * math.sin(angle_rad)))
        c = int(round(cx + rr * math.cos(angle_rad)))
        if 0 <= r < h and 0 <= c < w and mask[r, c]:
            return True
    return False


def _measure_sweep(arr: np.ndarray, center, color_mask: np.ndarray) -> float:
    """Angular extent (degrees) of the arc around ``center``, binned to the
    nearest canonical sweep."""
    count = 0
    for k in range(_SAMPLES):
        a = 2.0 * math.pi * k / _SAMPLES
        if _covered_any(color_mask, center, a):
            count += 1
    sweep = 360.0 * count / _SAMPLES
    return min(CANONICAL_SWEEPS, key=lambda value: abs(value - sweep))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.float32)

    dark = arr.max(axis=2) < _N_DARK
    hubs = _label_components(dark)

    line_counts = dark.sum(axis=1)
    grid = line_counts > 0.45 * dark.shape[1]
    rows = np.where(grid)[0]
    if rows.size == 0:
        raise ValueError("no divider line found")
    divider_mid = float(rows.mean())

    ref_hub = None
    cand_hubs = []
    for r, c in hubs:
        if r < divider_mid:
            ref_hub = (c, r)
        else:
            cand_hubs.append((c, r))
    if ref_hub is None or len(cand_hubs) != 3:
        raise ValueError(f"expected 1 reference and 3 candidate hubs, got {hubs}")

    gray_mask = _classify(arr, GRAY)
    ref_sweep = _measure_sweep(arr, ref_hub, gray_mask)

    color_masks = {name: _classify(arr, rgb) for name, rgb in PALETTE.items()}

    for hub in cand_hubs:
        cx, cy = hub
        # determine this gauge's color by which palette color best fills its arc
        counts = {}
        for name, mask in color_masks.items():
            total = 0
            for k in range(_SAMPLES):
                a = 2.0 * math.pi * k / _SAMPLES
                if _covered_any(mask, hub, a):
                    total += 1
            counts[name] = total
        color = max(counts, key=counts.get)
        if counts[color] == 0:
            raise ValueError("candidate gauge has no detectable arc color")
        sweep = _measure_sweep(arr, hub, color_masks[color])
        if sweep == ref_sweep:
            return color
    raise ValueError(f"no candidate matched reference sweep {ref_sweep}")


if __name__ == "__main__":
    import sys
    img = Image.open(sys.argv[1]).convert("RGB")
    print(decision_from_image(img))
