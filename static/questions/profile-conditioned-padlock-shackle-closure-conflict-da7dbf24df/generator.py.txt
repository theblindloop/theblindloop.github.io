"""Deterministic matched renderer for visible padlock-shackle closure.

Consecutive seeds are a control/counterfactual pair.  They share canvas,
palette, body geometry, shackle stroke and shackle path length.  The even member
has both ends seated in the body; in the odd member one fixed-length shackle leg
is swung clear of the body.  Gold is derived only from the rendered gap size.
"""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


BODY_COLORS = (
    (42, 111, 177),
    (24, 137, 118),
    (177, 92, 43),
    (116, 78, 164),
)
SHACKLE_COLORS = (
    (28, 34, 43),
    (48, 38, 55),
    (27, 52, 50),
    (58, 45, 31),
)
BACKGROUNDS = (
    (250, 249, 246),
    (246, 250, 252),
    (252, 247, 250),
)
RIVET_COLOR = (236, 221, 164)


def sample_scene(seed: int) -> dict:
    """Return one deterministic scene; even/odd neighbors are style matched."""

    pair_seed = int(seed) // 2
    rng = random.Random(0x10C4A11 ^ (pair_seed * 1_000_003))
    width, height = 384, 320
    body_width = rng.randrange(184, 201, 4)
    body_height = rng.randrange(116, 133, 4)
    body_left = rng.randint(92, 108)
    body_top = rng.randint(158, 169)
    shackle_outer_width = rng.randrange(108, 125, 4)
    shackle_top = rng.randint(35, 45)
    contact_depth = rng.randrange(20, 27, 2)
    paired_open_gap = rng.randrange(18, 25, 2)
    gap_size = 0 if int(seed) % 2 == 0 else paired_open_gap
    rivet_order = [0, 1]
    if rng.randrange(2):
        rivet_order.reverse()
    return {
        "width": width,
        "height": height,
        "body_left": body_left,
        "body_top": body_top,
        "body_width": body_width,
        "body_height": body_height,
        "body_radius": rng.randrange(18, 25, 2),
        "shackle_outer_width": shackle_outer_width,
        "shackle_top": shackle_top,
        "shackle_stroke": rng.randrange(14, 19, 2),
        "contact_depth": contact_depth,
        "gap_side": rng.randrange(2),
        "gap_size": gap_size,
        "body_color_index": rng.randrange(len(BODY_COLORS)),
        "shackle_color_index": rng.randrange(len(SHACKLE_COLORS)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "rivet_draw_order": rivet_order,
    }


def _geometry(scene: dict) -> dict:
    """Derive the two equal-length legs and the fixed upper semicircle."""

    body_left = float(scene["body_left"])
    body_top = float(scene["body_top"])
    body_width = float(scene["body_width"])
    center_x = body_left + body_width / 2.0
    radius = float(scene["shackle_outer_width"]) / 2.0
    top = float(scene["shackle_top"])
    side_y = top + radius
    contact_y = body_top + float(scene["contact_depth"])
    leg_length = contact_y - side_y
    if leg_length <= 55.0:
        raise ValueError("shackle leg is too short")

    ends = [(center_x - radius, contact_y), (center_x + radius, contact_y)]
    if float(scene["gap_size"]) > 0.0:
        side = int(scene["gap_side"])
        end_y = body_top - float(scene["gap_size"])
        dy = end_y - side_y
        if not 0.0 < dy < leg_length:
            raise ValueError("invalid open-shackle geometry")
        outward = math.sqrt(max(0.0, leg_length * leg_length - dy * dy))
        end_x = center_x + (-1.0 if side == 0 else 1.0) * (radius + outward)
        ends[side] = (end_x, end_y)
    return {
        "center_x": center_x,
        "radius": radius,
        "top": top,
        "side_y": side_y,
        "pivots": [(center_x - radius, side_y), (center_x + radius, side_y)],
        "ends": ends,
    }


def _scale_box(box: tuple[float, float, float, float], ss: int) -> tuple[int, int, int, int]:
    return tuple(int(round(value * ss)) for value in box)


def render(scene: dict) -> Image.Image:
    """Render a high-contrast padlock diagram as a final antialiased RGB PNG."""

    ss = 4
    width, height = int(scene["width"]), int(scene["height"])
    background = BACKGROUNDS[int(scene["background_index"])]
    body_color = BODY_COLORS[int(scene["body_color_index"])]
    shackle_color = SHACKLE_COLORS[int(scene["shackle_color_index"])]
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    left = float(scene["body_left"])
    top = float(scene["body_top"])
    right = left + float(scene["body_width"])
    bottom = top + float(scene["body_height"])
    draw.rounded_rectangle(
        _scale_box((left, top, right, bottom), ss),
        radius=int(scene["body_radius"]) * ss,
        fill=body_color,
    )

    geometry = _geometry(scene)
    radius = geometry["radius"]
    center_x = geometry["center_x"]
    arc_box = (
        center_x - radius,
        geometry["top"],
        center_x + radius,
        geometry["top"] + 2.0 * radius,
    )
    stroke = int(scene["shackle_stroke"]) * ss
    draw.arc(_scale_box(arc_box, ss), start=180, end=360, fill=shackle_color, width=stroke)
    for pivot, end in zip(geometry["pivots"], geometry["ends"]):
        draw.line(
            [(int(round(pivot[0] * ss)), int(round(pivot[1] * ss))),
             (int(round(end[0] * ss)), int(round(end[1] * ss)))],
            fill=shackle_color,
            width=stroke,
        )
        # A round joint makes the arc/leg articulation visibly continuous even
        # when the open leg leaves its pivot at a steep angle.
        joint_radius = float(scene["shackle_stroke"]) / 2.0
        draw.ellipse(
            _scale_box(
                (
                    pivot[0] - joint_radius,
                    pivot[1] - joint_radius,
                    pivot[0] + joint_radius,
                    pivot[1] + joint_radius,
                ),
                ss,
            ),
            fill=shackle_color,
        )

    # Two disjoint decorative rivets provide a genuine list-order latent alias.
    rivet_y = bottom - 22.0
    rivet_xs = (left + 25.0, right - 25.0)
    for index in scene["rivet_draw_order"]:
        x = rivet_xs[int(index)]
        draw.ellipse(_scale_box((x - 5, rivet_y - 5, x + 5, rivet_y + 5), ss), fill=RIVET_COLOR)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Answer whether the rasterized shackle has two seated ends."""

    return "yes" if float(scene["gap_size"]) == 0.0 else "no"


def margin(scene: dict) -> float:
    """One-sided distance from the nearest contact/gap ambiguity boundary."""

    if float(scene["gap_size"]) == 0.0:
        return float(scene["contact_depth"])
    return min(float(scene["contact_depth"]), float(scene["gap_size"]))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 12.0 or int(scene["shackle_stroke"]) < 10


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the drawing order of disjoint same-colored rivets."""

    twin = copy.deepcopy(scene)
    twin["rivet_draw_order"] = list(reversed(twin["rivet_draw_order"]))
    return [("reverse_disjoint_rivet_draw_order", twin)]
