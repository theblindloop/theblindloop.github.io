"""Deterministic renderer for a multi-panel trace-relay board."""

from __future__ import annotations

import itertools
import random
from typing import Iterable

from PIL import Image, ImageDraw

BACKGROUND = (248, 249, 246)
INK = (25, 29, 35)
PORT_GRAY = (173, 180, 188)
GREEN = (31, 160, 88)
TERMINAL_RGB = {
    "red": (220, 64, 64),
    "blue": (55, 110, 220),
    "orange": (238, 142, 38),
}
TERMINAL_NAMES = tuple(TERMINAL_RGB)
PERMUTATIONS = tuple(itertools.permutations(range(3)))


def _trace_end(start_row: int, panel_permutations: Iterable[Iterable[int]]) -> int:
    row = int(start_row)
    for mapping in panel_permutations:
        row = int(tuple(mapping)[row])
    return row


def sample_scene(seed: int) -> dict:
    """Sample a balanced scene. Consecutive seeds cycle answer identity."""
    rng = random.Random(int(seed))
    start_row = rng.randrange(3)
    panels = [list(rng.choice(PERMUTATIONS)) for _ in range(4)]
    if sum(tuple(panel) != (0, 1, 2) for panel in panels) < 3:
        panels[0] = list(rng.choice(PERMUTATIONS[1:]))
        panels[1] = [1, 0, 2]

    end_row = _trace_end(start_row, panels)
    answer_color = TERMINAL_NAMES[int(seed) % len(TERMINAL_NAMES)]
    remaining = [name for name in TERMINAL_NAMES if name != answer_color]
    rng.shuffle(remaining)
    terminal_colors: list[str | None] = [None, None, None]
    terminal_colors[end_row] = answer_color
    for row, color in zip((r for r in range(3) if r != end_row), remaining):
        terminal_colors[row] = color

    return {
        "start_row": start_row,
        "panel_permutations": panels,
        "terminal_colors": terminal_colors,
        "panel_count": 4,
        "width": 700 + 20 * rng.randrange(3),
        "height": 400 + 20 * rng.randrange(3),
    }


def _layout(width: int, height: int, panel_count: int) -> tuple[list[float], list[float]]:
    columns = [70.0 + i * (width - 140.0) / panel_count for i in range(panel_count + 1)]
    rows = [100.0, height / 2.0, height - 100.0]
    return columns, rows


def render(scene: dict) -> Image.Image:
    """Render the board. X intersections are intentionally ordinary crossings."""
    width = int(scene["width"])
    height = int(scene["height"])
    panel_count = int(scene["panel_count"])
    start_row = int(scene["start_row"])
    panel_permutations = [tuple(int(v) for v in p) for p in scene["panel_permutations"]]
    terminal_colors = [str(v) for v in scene["terminal_colors"]]
    if panel_count != len(panel_permutations) or panel_count < 2:
        raise ValueError("panel_count must equal the number of routing permutations")
    if sorted(terminal_colors) != sorted(TERMINAL_NAMES):
        raise ValueError("terminal_colors must be a permutation of the public palette")

    scale = 3
    image = Image.new("RGB", (width * scale, height * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    columns, rows = _layout(width, height, panel_count)
    sx = lambda value: int(round(value * scale))

    # Pale panel cards make stage boundaries visible without encoding a mapping.
    for stage in range(panel_count):
        x0 = columns[stage] + 8
        x1 = columns[stage + 1] - 8
        fill = (244, 246, 242) if stage % 2 == 0 else (239, 243, 246)
        draw.rounded_rectangle(
            (sx(x0), sx(64), sx(x1), sx(height - 64)),
            radius=sx(12),
            fill=fill,
            outline=(218, 222, 218),
            width=scale,
        )

    # All three traces are rendered with identical ink and thickness.
    for stage, mapping in enumerate(panel_permutations):
        if sorted(mapping) != [0, 1, 2]:
            raise ValueError("each panel mapping must be a permutation of three rows")
        for source_row, destination_row in enumerate(mapping):
            draw.line(
                (
                    sx(columns[stage]),
                    sx(rows[source_row]),
                    sx(columns[stage + 1]),
                    sx(rows[destination_row]),
                ),
                fill=INK,
                width=6 * scale,
            )

    radius = 12.0
    for column_index, x in enumerate(columns):
        for row_index, y in enumerate(rows):
            if column_index == 0 and row_index == start_row:
                fill = GREEN
            elif column_index == panel_count:
                fill = TERMINAL_RGB[terminal_colors[row_index]]
            else:
                fill = PORT_GRAY
            box = (sx(x - radius), sx(y - radius), sx(x + radius), sx(y + radius))
            draw.ellipse(box, fill=fill, outline=INK, width=2 * scale)

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    row = int(scene["start_row"])
    for mapping in scene["panel_permutations"]:
        row = int(mapping[row])
    return str(scene["terminal_colors"][row])


def margin(scene: dict) -> float:
    """Visible clearance scale in pixels for the discrete trace assignment."""
    row_gap = (int(scene["height"]) - 200.0) / 2.0
    panel_width = (int(scene["width"]) - 140.0) / int(scene["panel_count"])
    return round(min(row_gap, panel_width) / 3.0, 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 30.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
