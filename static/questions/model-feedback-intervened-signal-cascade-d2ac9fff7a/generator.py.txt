"""Deterministic renderer for a surgically intervened Boolean cascade."""

from __future__ import annotations

import random

from PIL import Image, ImageDraw, ImageFont

CANVAS = (720, 460)
BOARD_W, BOARD_H = 640, 400
OPS = ("AND", "OR", "XOR")
ANSWERS = ("amber", "cyan", "violet")

RGB = {
    "white": (255, 255, 255),
    "board": (246, 248, 250),
    "border": (52, 61, 70),
    "ink": (31, 38, 45),
    "muted": (104, 115, 125),
    "and": (205, 74, 72),
    "or": (54, 150, 105),
    "xor": (55, 112, 190),
    "force": (194, 52, 125),
    "amber": (230, 155, 32),
    "cyan": (18, 157, 181),
    "violet": (137, 78, 191),
}

FONT = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
BOLD = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"


def _op(kind: str, left: int, right: int) -> int:
    if kind == "AND":
        return int(bool(left) and bool(right))
    if kind == "OR":
        return int(bool(left) or bool(right))
    if kind == "XOR":
        return int(bool(left) != bool(right))
    raise ValueError(f"unknown gate {kind!r}")


def _states(scene: dict) -> tuple[dict[str, int], dict[str, int]]:
    inputs = [int(value) for value in scene["inputs"]]
    gates = [str(value) for value in scene["gates"]]
    intervention = str(scene["intervention_node"])
    if len(inputs) != 3 or any(value not in {0, 1} for value in inputs):
        raise ValueError("inputs must be three bits")
    if len(gates) != 6 or any(value not in OPS for value in gates):
        raise ValueError("six valid gates are required")
    if intervention not in {"D", "E"}:
        raise ValueError("intervention must target D or E")

    a, b, c = inputs
    d = _op(gates[0], a, b)
    e = _op(gates[1], b, c)
    f = _op(gates[2], d, e)
    factual = {
        "A": a,
        "B": b,
        "C": c,
        "D": d,
        "E": e,
        "F": f,
        "L0": _op(gates[3], d, f),
        "L1": _op(gates[4], f, c),
        "L2": _op(gates[5], e, f),
    }

    d_cf, e_cf = d, e
    if intervention == "D":
        d_cf = 1 - d
    else:
        e_cf = 1 - e
    f_cf = _op(gates[2], d_cf, e_cf)
    counterfactual = {
        "D": d_cf,
        "E": e_cf,
        "F": f_cf,
        "L0": _op(gates[3], d_cf, f_cf),
        "L1": _op(gates[4], f_cf, c),
        "L2": _op(gates[5], e_cf, f_cf),
    }
    return factual, counterfactual


def _changed_slots(scene: dict) -> list[int]:
    factual, counterfactual = _states(scene)
    return [
        index
        for index, key in enumerate(("L0", "L1", "L2"))
        if factual[key] != counterfactual[key]
    ]


def sample_scene(seed: int) -> dict:
    """Sample a balanced valid intervention scene from an integer seed."""

    rng = random.Random((int(seed) ^ 0xC0A57A1) & 0xFFFFFFFFFFFFFFFF)
    wanted_color = ANSWERS[int(seed) % len(ANSWERS)]
    for _ in range(20000):
        inputs = [rng.randrange(2) for _ in range(3)]
        gates = [OPS[rng.randrange(len(OPS))] for _ in range(6)]
        intervention = ("D", "E")[rng.randrange(2)]
        if len(set(gates)) != 3:
            continue
        probe = {
            "inputs": inputs,
            "gates": gates,
            "intervention_node": intervention,
            "lamp_colors": list(ANSWERS),
            "anchor_x": 0,
            "anchor_y": 0,
        }
        factual, counterfactual = _states(probe)
        changed = _changed_slots(probe)
        if len(changed) != 1 or factual["F"] == counterfactual["F"]:
            continue
        changed_slot = changed[0]
        other_colors = [color for color in ANSWERS if color != wanted_color]
        rng.shuffle(other_colors)
        lamp_colors: list[str | None] = [None, None, None]
        lamp_colors[changed_slot] = wanted_color
        fill = iter(other_colors)
        for index in range(3):
            if lamp_colors[index] is None:
                lamp_colors[index] = next(fill)
        return {
            "inputs": inputs,
            "gates": gates,
            "intervention_node": intervention,
            "lamp_colors": [str(value) for value in lamp_colors],
            "anchor_x": rng.randint(28, 52),
            "anchor_y": rng.randint(24, 36),
        }
    raise RuntimeError("could not sample a uniquely changing cascade")


def analytic_gold(scene: dict) -> str:
    """Return the color bound to the uniquely changed terminal."""

    changed = _changed_slots(scene)
    colors = [str(value) for value in scene["lamp_colors"]]
    if len(colors) != 3 or sorted(colors) != sorted(ANSWERS):
        raise ValueError("lamp colors must be a permutation of the answer set")
    if len(changed) != 1:
        raise ValueError("scene does not have one uniquely changed lamp")
    return colors[changed[0]]


def margin(scene: dict) -> float:
    """One discrete changed-lamp unit separates a valid scene from its boundary."""

    factual, counterfactual = _states(scene)
    changes = sum(factual[key] != counterfactual[key] for key in ("L0", "L1", "L2"))
    return 1.0 if changes == 1 and factual["F"] != counterfactual["F"] else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0 or analytic_gold(scene) not in ANSWERS
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """All parameters are explicitly rendered; there are no intended aliases."""

    return []


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(BOLD if bold else FONT, size)


def _text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], value: str, size: int, *,
          fill: tuple[int, int, int] | None = None, bold: bool = False) -> None:
    draw.text(xy, value, font=_font(size, bold), fill=fill or RGB["ink"], anchor="mm")


def _status(draw: ImageDraw.ImageDraw, x: int, y: int, bit: int) -> None:
    draw.ellipse((x - 12, y - 12, x + 12, y + 12), outline=RGB["ink"], width=3,
                 fill=RGB["board"])
    draw.rectangle((x - 4, y - 4, x + 4, y + 4),
                   fill=RGB["ink"] if bit else RGB["board"])


def _gate(draw: ImageDraw.ImageDraw, x: int, y: int, kind: str) -> None:
    color = RGB[kind.lower()]
    draw.rounded_rectangle((x, y - 15, x + 76, y + 15), radius=6, fill=color)
    _text(draw, (x + 38, y), kind, 15, fill=RGB["white"], bold=True)


def _rule(draw: ImageDraw.ImageDraw, y: int, out: str, left: str, kind: str,
          right: str, bit: int, ax: int) -> None:
    _text(draw, (ax + 192, y), out, 20, bold=True)
    _text(draw, (ax + 224, y), "=", 18, fill=RGB["muted"])
    _text(draw, (ax + 267, y), left, 18, bold=True)
    _gate(draw, ax + 295, y, kind)
    _text(draw, (ax + 397, y), right, 18, bold=True)
    _text(draw, (ax + 421, y), "->", 17, fill=RGB["muted"])
    _status(draw, ax + 455, y, bit)


def render(scene: dict) -> Image.Image:
    """Render all factual evidence and the visible intervention marker."""

    inputs = [int(value) for value in scene["inputs"]]
    gates = [str(value) for value in scene["gates"]]
    intervention = str(scene["intervention_node"])
    lamp_colors = [str(value) for value in scene["lamp_colors"]]
    ax = int(scene["anchor_x"])
    ay = int(scene["anchor_y"])
    factual, _ = _states(scene)

    image = Image.new("RGB", CANVAS, RGB["white"])
    draw = ImageDraw.Draw(image)
    draw.rectangle((ax, ay, ax + BOARD_W - 1, ay + BOARD_H - 1),
                   fill=RGB["board"], outline=RGB["border"], width=4)
    _text(draw, (ax + 320, ay + 24), "FACTUAL SIGNAL BOARD", 19, bold=True)
    _text(draw, (ax + 110, ay + 52), "filled = 1    hollow = 0", 14,
          fill=RGB["muted"])
    _text(draw, (ax + 438, ay + 52), "AND: both   OR: at least one   XOR: exactly one",
          13, fill=RGB["muted"])

    _text(draw, (ax + 93, ay + 78), "SOURCES", 15, bold=True)
    for label, bit, yoff in zip(("A", "B", "C"), inputs, (105, 165, 225), strict=True):
        _text(draw, (ax + 55, ay + yoff), label, 20, bold=True)
        _status(draw, ax + 95, ay + yoff, bit)

    _text(draw, (ax + 330, ay + 78), "CAUSES -> CURRENT VALUE", 14, bold=True)
    internal_rows = (105, 165, 225)
    _rule(draw, ay + internal_rows[0], "D", "A", gates[0], "B", factual["D"], ax)
    _rule(draw, ay + internal_rows[1], "E", "B", gates[1], "C", factual["E"], ax)
    _rule(draw, ay + internal_rows[2], "F", "D", gates[2], "E", factual["F"], ax)

    target_row = internal_rows[0] if intervention == "D" else internal_rows[1]
    draw.rounded_rectangle((ax + 500, ay + target_row - 16, ax + 618, ay + target_row + 16),
                           radius=6, fill=RGB["force"])
    _text(draw, (ax + 559, ay + target_row), "WRENCH: FLIP", 13,
          fill=RGB["white"], bold=True)

    _text(draw, (ax + 350, ay + 260), "TERMINAL LAMPS (current)", 14, bold=True)
    lamp_specs = (
        ("L0", "D", gates[3], "F", 280),
        ("L1", "F", gates[4], "C", 330),
        ("L2", "E", gates[5], "F", 375),
    )
    for index, (key, left, kind, right, yoff) in enumerate(lamp_specs):
        y = ay + yoff
        _rule(draw, y, "", left, kind, right, factual[key], ax)
        color = lamp_colors[index]
        draw.rectangle((ax + 545, y - 12, ax + 575, y + 12), fill=RGB[color])
        draw.rectangle((ax + 552, y - 5, ax + 568, y + 5),
                       fill=RGB["ink"] if factual[key] else RGB["board"])
        _text(draw, (ax + 603, y), color, 14, bold=True)
    return image
