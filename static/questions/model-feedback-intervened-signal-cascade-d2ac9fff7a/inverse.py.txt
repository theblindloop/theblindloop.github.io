"""Independent pixel-only inverse for the intervened signal board."""

from __future__ import annotations

import numpy as np
from PIL import Image

CANVAS = (720, 460)
BOARD_W, BOARD_H = 640, 400
BOARD = (246, 248, 250)
BORDER = (52, 61, 70)
INK = (31, 38, 45)
FORCE = (194, 52, 125)
GATE_RGB = {
    (205, 74, 72): "AND",
    (54, 150, 105): "OR",
    (55, 112, 190): "XOR",
}
LAMP_RGB = {
    (230, 155, 32): "amber",
    (18, 157, 181): "cyan",
    (137, 78, 191): "violet",
}


class OracleError(ValueError):
    """Raised when the raster is not one intact, decidable signal board."""


def _op(kind: str, left: int, right: int) -> int:
    if kind == "AND":
        return left & right
    if kind == "OR":
        return left | right
    if kind == "XOR":
        return left ^ right
    raise OracleError("unknown visible gate")


def _pixel(array: np.ndarray, x: int, y: int) -> tuple[int, int, int]:
    return tuple(int(value) for value in array[y, x])


def _uniform_patch(array: np.ndarray, x: int, y: int) -> tuple[int, int, int]:
    patch = array[y - 3:y + 4, x - 3:x + 4]
    colors = {tuple(int(value) for value in row) for row in patch.reshape(-1, 3)}
    if len(colors) != 1:
        raise OracleError("a signal center is damaged or ambiguous")
    return next(iter(colors))


def _bit(array: np.ndarray, x: int, y: int) -> int:
    color = _uniform_patch(array, x, y)
    if color == INK:
        return 1
    if color == BOARD:
        return 0
    raise OracleError("a signal center has an invalid color")


def decision_from_image(image: Image.Image) -> str:
    """Decode the visible factual program, perform the intervention, and select a lamp."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if (array.shape[1], array.shape[0]) != CANVAS:
        raise OracleError("unexpected canvas size")
    border_mask = np.all(array == np.asarray(BORDER, dtype=np.uint8), axis=2)
    points = np.argwhere(border_mask)
    if not len(points):
        raise OracleError("signal board is absent")
    top, left = points.min(axis=0)
    bottom, right = points.max(axis=0)
    if int(right - left + 1) != BOARD_W or int(bottom - top + 1) != BOARD_H:
        raise OracleError("signal board has the wrong extent")
    ax, ay = int(left), int(top)
    if not (
        np.all(border_mask[ay, ax:ax + BOARD_W])
        and np.all(border_mask[ay + BOARD_H - 1, ax:ax + BOARD_W])
        and np.all(border_mask[ay:ay + BOARD_H, ax])
        and np.all(border_mask[ay:ay + BOARD_H, ax + BOARD_W - 1])
    ):
        raise OracleError("signal board border is broken")

    inputs = [_bit(array, ax + 95, ay + y) for y in (105, 165, 225)]
    rows = (105, 165, 225, 280, 330, 375)
    gates = []
    for y in rows:
        color = _pixel(array, ax + 300, ay + y - 11)
        if color not in GATE_RGB:
            raise OracleError("gate badge is missing or damaged")
        gates.append(GATE_RGB[color])

    force_hits = []
    for name, y in (("D", 105), ("E", 165)):
        color = _pixel(array, ax + 505, ay + y - 11)
        if color == FORCE:
            force_hits.append(name)
        elif color != BOARD:
            raise OracleError("intervention marker is damaged")
    if len(force_hits) != 1:
        raise OracleError("exactly one internal node must be wrench-marked")

    a, b, c = inputs
    d = _op(gates[0], a, b)
    e = _op(gates[1], b, c)
    f = _op(gates[2], d, e)
    outputs = [
        _op(gates[3], d, f),
        _op(gates[4], f, c),
        _op(gates[5], e, f),
    ]
    shown_internal = [_bit(array, ax + 455, ay + y) for y in (105, 165, 225)]
    if shown_internal != [d, e, f]:
        raise OracleError("displayed internal state disagrees with visible causes")
    shown_outputs = [_bit(array, ax + 455, ay + y) for y in (280, 330, 375)]
    if shown_outputs != outputs:
        raise OracleError("displayed terminal state disagrees with visible causes")

    lamp_colors = []
    for y in (280, 330, 375):
        color = _pixel(array, ax + 547, ay + y - 10)
        if color not in LAMP_RGB:
            raise OracleError("lamp binding is missing or damaged")
        lamp_colors.append(LAMP_RGB[color])
    if sorted(lamp_colors) != sorted(LAMP_RGB.values()):
        raise OracleError("lamp colors must be unique")

    d_cf, e_cf = d, e
    if force_hits[0] == "D":
        d_cf = 1 - d
    else:
        e_cf = 1 - e
    f_cf = _op(gates[2], d_cf, e_cf)
    counterfactual = [
        _op(gates[3], d_cf, f_cf),
        _op(gates[4], f_cf, c),
        _op(gates[5], e_cf, f_cf),
    ]
    changed = [index for index in range(3) if outputs[index] != counterfactual[index]]
    if len(changed) != 1 or f == f_cf:
        raise OracleError("visible intervention does not uniquely change one lamp through F")
    return lamp_colors[changed[0]]
