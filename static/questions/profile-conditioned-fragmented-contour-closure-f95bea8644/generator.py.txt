"""Renderer and analytic model for fragmented contour completion."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw


RENDERER_VERSION = "fragmented_contour_closure-0.1.0"
CANVAS = (720, 300)
SUPERSAMPLE = 3
BACKGROUND = (248, 248, 244)
PANEL_FILL = (252, 252, 249)
PANEL_EDGE = (211, 214, 209)
COLORS = ("red", "blue", "green")
PALETTE = {
    "red": (211, 62, 65),
    "blue": (52, 99, 192),
    "green": (42, 153, 92),
}
SLOT_X = (18.0, 250.0, 482.0)
SLOT_Y = 30.0
PANEL_SIZE = (220.0, 240.0)
_TAU = 2.0 * math.pi


def _normalize(values: list[float], total: float) -> list[float]:
    scale = total / sum(values)
    return [value * scale for value in values]


def _gap_pattern(
    rng: random.Random, total_gap: float, role: str, near: bool
) -> list[float]:
    base = total_gap / 10.0
    if near:
        if role == "closed":
            values = [base + rng.uniform(-0.015, 0.015) for _ in range(10)]
        elif role == "open":
            values = [base + 0.18] + [base + rng.uniform(-0.012, 0.012) for _ in range(9)]
        else:
            values = [base + 0.16, base + 0.13] + [base + rng.uniform(-0.012, 0.012) for _ in range(8)]
    else:
        if role == "closed":
            values = [base + rng.uniform(-0.012, 0.012) for _ in range(10)]
        elif role == "open":
            values = [base + 0.70] + [base + rng.uniform(-0.015, 0.015) for _ in range(9)]
        else:
            values = [base + 0.48, base + 0.34] + [base + rng.uniform(-0.015, 0.015) for _ in range(8)]
    return _normalize(values, total_gap)


def _make_fragments(
    rng: random.Random,
    total_span: float,
    phase: float,
    role: str,
    near: bool,
) -> list[dict[str, float]]:
    total_gap = _TAU - total_span
    gaps = _gap_pattern(rng, total_gap, role, near)
    # Equal angular stroke spans make the total visible ink independent of
    # contour class.  Only the global distribution of the intervening gaps
    # carries the answer.
    span = total_span / 10.0
    cursor = phase
    fragments: list[dict[str, float]] = []
    for gap in gaps:
        fragments.append({"start": round(cursor, 8), "span": round(span, 8)})
        cursor += span + gap
    return fragments


def sample_scene(seed: int) -> dict:
    """Sample three color-counterbalanced fragmented contour fields."""

    rng = random.Random((seed + 9176) * 10007)
    answer_index = seed % 3
    color_order = list(COLORS)
    rng.shuffle(color_order)
    slots = [0, 1, 2]
    rng.shuffle(slots)
    near = seed % 7 == 0
    total_span = round(rng.uniform(3.62, 4.05), 6)
    stroke_width = round(rng.uniform(4.5, 6.5), 3)
    panels: list[dict] = []
    fragment_sets: list[list[dict[str, float]]] = []
    for index, color in enumerate(color_order):
        role = "closed" if index == answer_index else ("open" if index % 2 else "double_gap")
        slot = slots[index]
        cx = SLOT_X[slot] + 110.0 + rng.uniform(-5.0, 5.0)
        cy = SLOT_Y + 120.0 + rng.uniform(-5.0, 5.0)
        panel_rng = random.Random(rng.randrange(1 << 30))
        panel = {
            "slot": slot,
            "color": color,
            "cx": round(cx, 3),
            "cy": round(cy, 3),
            "rx": round(rng.uniform(72.0, 82.0), 3),
            "ry": round(rng.uniform(58.0, 72.0), 3),
            "rotation": round(rng.uniform(-0.28, 0.28), 6),
            "stroke_width": stroke_width,
        }
        fragment_sets.append(
            _make_fragments(
                panel_rng,
                total_span,
                rng.uniform(0.0, _TAU),
                role,
                near,
            )
        )
        panels.append(panel)
    return {"panels": panels, "fragments": fragment_sets}


def _point(panel: dict, angle: float) -> tuple[float, float]:
    ca = math.cos(float(panel["rotation"]))
    sa = math.sin(float(panel["rotation"]))
    ux = float(panel["rx"]) * math.cos(angle)
    uy = float(panel["ry"]) * math.sin(angle)
    return (
        float(panel["cx"]) + ca * ux - sa * uy,
        float(panel["cy"]) + sa * ux + ca * uy,
    )


def _fragment_points(panel: dict, fragment: dict) -> list[tuple[float, float]]:
    start = float(fragment["start"])
    span = float(fragment["span"])
    # Short polylines visibly retain curvature while keeping every fragment
    # disconnected from its neighbors at the final raster resolution.
    count = max(6, int(round(span * 18.0)))
    return [_point(panel, start + span * i / count) for i in range(count + 1)]


def render(scene: dict) -> Image.Image:
    image = Image.new(
        "RGB",
        (CANVAS[0] * SUPERSAMPLE, CANVAS[1] * SUPERSAMPLE),
        BACKGROUND,
    )
    draw = ImageDraw.Draw(image)
    for slot in range(3):
        x0 = int(round(SLOT_X[slot] * SUPERSAMPLE))
        y0 = int(round(SLOT_Y * SUPERSAMPLE))
        x1 = int(round((SLOT_X[slot] + PANEL_SIZE[0]) * SUPERSAMPLE))
        y1 = int(round((SLOT_Y + PANEL_SIZE[1]) * SUPERSAMPLE))
        draw.rounded_rectangle((x0, y0, x1, y1), radius=10 * SUPERSAMPLE, fill=PANEL_FILL, outline=PANEL_EDGE, width=2 * SUPERSAMPLE)
    for panel in scene["panels"]:
        ink = PALETTE[str(panel["color"])]
        width = max(1, int(round(float(panel["stroke_width"]) * SUPERSAMPLE)))
        fragments = scene["fragments"][scene["panels"].index(panel)]
        # Fragment order is deliberately irrelevant: all strokes in a panel
        # have one opaque color, so reversing the list is pixel-identical.
        for fragment in fragments:
            points = [
                (int(round(x * SUPERSAMPLE)), int(round(y * SUPERSAMPLE)))
                for x, y in _fragment_points(panel, fragment)
            ]
            draw.line(points, fill=ink, width=width, joint="curve")
    return image.resize(CANVAS, Image.Resampling.LANCZOS)


def _fragment_mid_angles(fragments: list[dict]) -> list[float]:
    # The analytic field is represented by the fragment parameterization, not
    # by a hidden categorical tag.
    return [
        (float(fragment["start"]) + 0.5 * float(fragment["span"])) % _TAU
        for fragment in fragments
    ]


def _closure_score(fragments: list[dict]) -> float:
    angles = sorted(_fragment_mid_angles(fragments))
    gaps = [
        (angles[(i + 1) % len(angles)] - angles[i]) % _TAU
        for i in range(len(angles))
    ]
    # A complete contour has no single dominant opening.  The score is a
    # smooth, one-sided global closure measure with a visible top-two margin.
    return 1.0 - max(gaps) / _TAU


def analytic_gold(scene: dict) -> str:
    panels = list(scene["panels"])
    scores = [_closure_score(fragments) for fragments in scene["fragments"]]
    best = max(range(len(panels)), key=lambda index: (scores[index], str(panels[index]["color"])))
    return str(panels[best]["color"])


def margin(scene: dict) -> float:
    scores = sorted((_closure_score(fragments) for fragments in scene["fragments"]), reverse=True)
    return round(scores[0] - scores[1], 9)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.035


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["fragments"] = [list(reversed(fragments)) for fragments in twin["fragments"]]
    return [("permute_same_colour_fragments", twin)]
