"""Deterministic Pillow world for composing two visible socket matchings."""

from __future__ import annotations

import random
from functools import lru_cache
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 820
HEIGHT = 560
SCALE = 3
LETTERS = tuple("ABCDEFGH")
CARD_LEFT = (25, 425)
SOCKET_X = (78, 478)
SOCKET_Y = (116, 166, 216, 266, 316, 366, 416, 466)
CORD_COLORS = ("#247BA0", "#F28E2B")
LABEL_COLORS = {
    "A": "#C9485B",
    "B": "#087E8B",
    "C": "#4C956C",
    "D": "#7B2CBF",
    "E": "#D97706",
    "F": "#8C564B",
    "G": "#C23B8F",
    "H": "#52606D",
}
INK = "#17212B"
MUTED = "#667085"


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", size * SCALE
    )


@lru_cache(maxsize=1)
def _all_matchings() -> tuple[tuple[tuple[str, str], ...], ...]:
    def build(items: tuple[str, ...]) -> list[tuple[tuple[str, str], ...]]:
        if not items:
            return [tuple()]
        first = items[0]
        results = []
        for index in range(1, len(items)):
            second = items[index]
            rest = items[1:index] + items[index + 1 :]
            pair = tuple(sorted((first, second)))
            for tail in build(rest):
                results.append(tuple(sorted((pair, *tail))))
        return results

    return tuple(build(LETTERS))


@lru_cache(maxsize=1)
def _noncrossing_position_matchings() -> tuple[tuple[tuple[int, int], ...], ...]:
    def build(items: tuple[int, ...]) -> list[tuple[tuple[int, int], ...]]:
        if not items:
            return [tuple()]
        first = items[0]
        results = []
        # Pairing positions have an even number of points inside the arc.
        for offset in range(1, len(items), 2):
            second = items[offset]
            for inside in build(items[1:offset]):
                for outside in build(items[offset + 1 :]):
                    results.append(tuple(sorted(((first, second), *inside, *outside))))
        return results

    return tuple(build(tuple(range(8))))


def _component_count(
    left_pairs: list[list[str]] | tuple[tuple[str, str], ...],
    right_pairs: list[list[str]] | tuple[tuple[str, str], ...],
) -> int:
    parent = {letter: letter for letter in LETTERS}

    def find(value: str) -> str:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    def union(a: str, b: str) -> None:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    for pair in (*left_pairs, *right_pairs):
        union(str(pair[0]), str(pair[1]))
    return len({find(letter) for letter in LETTERS})


def _display_order(pairs: tuple[tuple[str, str], ...], rng: random.Random) -> list[str]:
    pattern = list(rng.choice(_noncrossing_position_matchings()))
    assigned = list(pairs)
    rng.shuffle(assigned)
    order: list[str | None] = [None] * 8
    for (p0, p1), pair in zip(pattern, assigned):
        a, b = pair
        if rng.randrange(2):
            a, b = b, a
        order[p0], order[p1] = a, b
    return [str(value) for value in order]


def sample_scene(seed: int) -> dict[str, Any]:
    rng = random.Random(int(seed))
    target = 1 + (int(seed) % 4)
    left = rng.choice(_all_matchings())
    compatible = [right for right in _all_matchings() if _component_count(left, right) == target]
    right = rng.choice(compatible)
    return {
        "left_order": _display_order(left, rng),
        "right_order": _display_order(right, rng),
        "left_pairs": [list(pair) for pair in left],
        "right_pairs": [list(pair) for pair in right],
    }


def analytic_gold(scene: dict[str, Any]) -> str:
    left_pairs = scene["left_pairs"]
    right_pairs = scene["right_pairs"]
    return str(_component_count(left_pairs, right_pairs))


def margin(scene: dict[str, Any]) -> float:
    # Conservative minimum colored-core and label-patch clearance in pixels.
    return 14.0


def _canonical_pairs(value: object) -> tuple[tuple[str, str], ...] | None:
    if not isinstance(value, list) or len(value) != 4:
        return None
    try:
        pairs = tuple(tuple(str(x) for x in pair) for pair in value)
    except TypeError:
        return None
    if any(len(pair) != 2 or pair[0] >= pair[1] for pair in pairs):
        return None
    if pairs != tuple(sorted(pairs)):
        return None
    if sorted(letter for pair in pairs for letter in pair) != list(LETTERS):
        return None
    return pairs


def _noncrossing(order: list[str], pairs: tuple[tuple[str, str], ...]) -> bool:
    position = {letter: index for index, letter in enumerate(order)}
    intervals = [tuple(sorted((position[a], position[b]))) for a, b in pairs]
    for index, (a, b) in enumerate(intervals):
        for c, d in intervals[index + 1 :]:
            if a < c < b < d or c < a < d < b:
                return False
    return True


def is_quarantined(scene: dict[str, Any]) -> bool:
    try:
        left_order = [str(value) for value in scene["left_order"]]
        right_order = [str(value) for value in scene["right_order"]]
        left_pairs = _canonical_pairs(scene["left_pairs"])
        right_pairs = _canonical_pairs(scene["right_pairs"])
    except (KeyError, TypeError):
        return True
    return not (
        sorted(left_order) == list(LETTERS)
        and sorted(right_order) == list(LETTERS)
        and left_pairs is not None
        and right_pairs is not None
        and _noncrossing(left_order, left_pairs)
        and _noncrossing(right_order, right_pairs)
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict]]:
    return []


def _curve_points(x: int, y0: int, y1: int) -> list[tuple[int, int]]:
    span = abs(y1 - y0) // 50
    outer_x = x + 68 + 18 * span
    points = []
    for step in range(81):
        t = step / 80.0
        u = 1.0 - t
        px = u**3 * x + 3 * u * u * t * outer_x + 3 * u * t * t * outer_x + t**3 * x
        py = u**3 * y0 + 3 * u * u * t * y0 + 3 * u * t * t * y1 + t**3 * y1
        points.append((int(round(px * SCALE)), int(round(py * SCALE))))
    return points


def _centered_text(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str, font: ImageFont.FreeTypeFont, **kwargs: Any) -> None:
    x, y = xy
    box = draw.textbbox((0, 0), text, font=font, stroke_width=int(kwargs.get("stroke_width", 0)))
    width, height = box[2] - box[0], box[3] - box[1]
    draw.text((x * SCALE - width / 2, y * SCALE - height / 2 - 2 * SCALE), text, font=font, **kwargs)


def render(scene: dict[str, Any]) -> Image.Image:
    left_order = [str(value) for value in scene["left_order"]]
    right_order = [str(value) for value in scene["right_order"]]
    left_pairs = [tuple(str(x) for x in pair) for pair in scene["left_pairs"]]
    right_pairs = [tuple(str(x) for x in pair) for pair in scene["right_pairs"]]

    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "#F1F5F9")
    draw = ImageDraw.Draw(canvas)
    draw.text((24 * SCALE, 17 * SCALE), "MATCH SAME LETTERS  •  COUNT CLOSED LOOPS", fill=INK, font=_font(22))
    draw.text((24 * SCALE, 48 * SCALE), "Each socket has one cord on each card.", fill=MUTED, font=_font(13))

    orders = (left_order, right_order)
    pair_sets = (left_pairs, right_pairs)
    for card_index, (card_x, socket_x, color, order, pairs) in enumerate(
        zip(CARD_LEFT, SOCKET_X, CORD_COLORS, orders, pair_sets), start=1
    ):
        draw.rounded_rectangle(
            (card_x * SCALE, 76 * SCALE, (card_x + 370) * SCALE, 516 * SCALE),
            radius=16 * SCALE,
            fill="white",
            outline="#94A3B8",
            width=2 * SCALE,
        )
        draw.text(((card_x + 18) * SCALE, 86 * SCALE), f"CARD {card_index}", fill=INK, font=_font(15))
        positions = {letter: SOCKET_Y[index] for index, letter in enumerate(order)}
        for a, b in pairs:
            y0, y1 = sorted((positions[a], positions[b]))
            points = _curve_points(socket_x + 17, y0, y1)
            draw.line(points, fill="#FFFFFF", width=13 * SCALE, joint="curve")
            draw.line(points, fill=color, width=8 * SCALE, joint="curve")

        for y, letter in zip(SOCKET_Y, order):
            radius = 21
            draw.ellipse(
                ((socket_x - radius) * SCALE, (y - radius) * SCALE,
                 (socket_x + radius) * SCALE, (y + radius) * SCALE),
                fill=LABEL_COLORS[letter],
                outline=INK,
                width=3 * SCALE,
            )
            _centered_text(
                draw,
                (socket_x, y),
                letter,
                _font(24),
                fill="white",
                stroke_width=1 * SCALE,
                stroke_fill=INK,
            )

    draw.text((244 * SCALE, 530 * SCALE), "snap A↔A, B↔B, …, H↔H", fill=MUTED, font=_font(14))
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
