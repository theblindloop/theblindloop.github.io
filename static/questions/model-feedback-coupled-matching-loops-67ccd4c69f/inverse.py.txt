"""Independent image-only inverse for the two-card matching grammar."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

WIDTH = 820
HEIGHT = 560
LETTERS = tuple("ABCDEFGH")
SOCKET_X = (78, 478)
SOCKET_Y = (116, 166, 216, 266, 316, 366, 416, 466)
CORD_RGB = ((36, 123, 160), (242, 142, 43))
LABEL_RGB = {
    "A": (201, 72, 91),
    "B": (8, 126, 139),
    "C": (76, 149, 108),
    "D": (123, 44, 191),
    "E": (217, 119, 6),
    "F": (140, 86, 75),
    "G": (194, 59, 143),
    "H": (82, 96, 109),
}


class OracleError(ValueError):
    pass


def _near(arr: np.ndarray, rgb: tuple[int, int, int], radius: float) -> np.ndarray:
    delta = arr.astype(np.int32) - np.asarray(rgb, dtype=np.int32)
    return np.sum(delta * delta, axis=2, dtype=np.int32) <= radius * radius


def _recover_order(arr: np.ndarray, socket_x: int) -> list[str]:
    order = []
    for socket_y in SOCKET_Y:
        patch = arr[socket_y - 17 : socket_y + 18, socket_x - 17 : socket_x + 18]
        scores = {letter: int(_near(patch, rgb, 48.0).sum()) for letter, rgb in LABEL_RGB.items()}
        ranked = sorted(scores, key=scores.get, reverse=True)
        if scores[ranked[0]] < 330 or scores[ranked[0]] < 4 * max(1, scores[ranked[1]]):
            raise OracleError(f"socket identity is not unique at y={socket_y}: {scores}")
        order.append(ranked[0])
    if sorted(order) != list(LETTERS):
        raise OracleError(f"card does not contain each identity exactly once: {order}")
    return order


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    remaining = {int(y) * width + int(x) for y, x in np.argwhere(mask)}
    components: list[list[tuple[int, int]]] = []
    while remaining:
        start = remaining.pop()
        queue = deque([start])
        component = []
        while queue:
            code = queue.popleft()
            y, x = divmod(code, width)
            component.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width:
                        neighbor = ny * width + nx
                        if neighbor in remaining:
                            remaining.remove(neighbor)
                            queue.append(neighbor)
        if len(component) >= 220:
            components.append(component)
    return components


def _recover_pairs(arr: np.ndarray, card_index: int, order: list[str]) -> list[tuple[str, str]]:
    socket_x = SOCKET_X[card_index]
    cord = _near(arr, CORD_RGB[card_index], 52.0)
    roi = np.zeros_like(cord)
    roi[96:492, socket_x + 21 : socket_x + 285] = True
    components = _components(cord & roi)
    if len(components) != 4:
        raise OracleError(f"card {card_index + 1} has {len(components)} recoverable cords")
    pairs = []
    for component in components:
        endpoints = []
        for slot, socket_y in enumerate(SOCKET_Y):
            count = sum(
                1
                for y, x in component
                if socket_x + 21 <= x <= socket_x + 43 and abs(y - socket_y) <= 7
            )
            if count >= 18:
                endpoints.append(slot)
        if len(endpoints) != 2:
            raise OracleError(f"cord has endpoint slots {endpoints}, expected exactly two")
        pairs.append(tuple(sorted((order[endpoints[0]], order[endpoints[1]]))))
    pairs.sort()
    if sorted(letter for pair in pairs for letter in pair) != list(LETTERS):
        raise OracleError("recovered cords do not use every socket exactly once")
    return pairs


def _component_count(left_pairs: list[tuple[str, str]], right_pairs: list[tuple[str, str]]) -> int:
    parent = {letter: letter for letter in LETTERS}

    def find(value: str) -> str:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    for a, b in (*left_pairs, *right_pairs):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra
    return len({find(letter) for letter in LETTERS})


def decision_from_image(image: Image.Image) -> str:
    if image.size != (WIDTH, HEIGHT):
        raise OracleError(f"expected {(WIDTH, HEIGHT)} image, got {image.size}")
    arr = np.asarray(image.convert("RGB"))
    left_order = _recover_order(arr, SOCKET_X[0])
    right_order = _recover_order(arr, SOCKET_X[1])
    left_pairs = _recover_pairs(arr, 0, left_order)
    right_pairs = _recover_pairs(arr, 1, right_order)
    answer = str(_component_count(left_pairs, right_pairs))
    if answer not in {"1", "2", "3", "4"}:
        raise OracleError(f"invalid component count {answer}")
    return answer
