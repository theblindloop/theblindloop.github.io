"""Image-only oracle for route_required_node.

Receives a rendered PNG and nothing else.  It segments the drawn ink, rebuilds
the junction network as a graph (stroke components are the edges, disks are the
vertices), walks the unique route from the green terminal to the red terminal,
and reports the colour of the marked junction sitting on that route.  It
abstains whenever the reconstruction is not unambiguous.

This module imports only numpy and PIL.  It never sees a latent scene and never
imports renderer, prompts, generate or verify code.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "route_required_node-oracle-0.1.0"
ABSTAIN = "unknown"

_STROKE = (150, 150, 150)
_PLAIN = (70, 70, 70)
_NODE_COLORS = {
    "green": (0, 150, 60),
    "red": (210, 35, 35),
    "blue": (30, 70, 210),
    "orange": (240, 140, 0),
    "purple": (140, 45, 175),
}
_MARKS = ("blue", "orange", "purple")

_MIN_STROKE_PIXELS = 60
_MIN_DISK_PIXELS = 200
_DILATE = 2


def _mask(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    return (
        (arr[:, :, 0] == rgb[0]) & (arr[:, :, 1] == rgb[1]) & (arr[:, :, 2] == rgb[2])
    )


def _components(mask: np.ndarray, min_pixels: int) -> list[np.ndarray]:
    """Four-connected components of a boolean mask, as boolean masks."""
    h, w = mask.shape
    seen = np.zeros((h, w), dtype=bool)
    out: list[np.ndarray] = []
    ys, xs = np.nonzero(mask)
    for y0, x0 in zip(ys.tolist(), xs.tolist()):
        if seen[y0, x0]:
            continue
        comp = np.zeros((h, w), dtype=bool)
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        comp[y0, x0] = True
        while queue:
            y, x = queue.popleft()
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    comp[ny, nx] = True
                    queue.append((ny, nx))
        if int(comp.sum()) >= min_pixels:
            out.append(comp)
    return out


def _dilate(mask: np.ndarray, rounds: int) -> np.ndarray:
    out = mask
    for _ in range(rounds):
        grown = out.copy()
        grown[1:, :] |= out[:-1, :]
        grown[:-1, :] |= out[1:, :]
        grown[:, 1:] |= out[:, :-1]
        grown[:, :-1] |= out[:, 1:]
        out = grown
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    if arr.ndim != 3 or arr.shape[0] < 32 or arr.shape[1] < 32:
        return ABSTAIN

    # --- junctions -------------------------------------------------------
    nodes: list[tuple[str, np.ndarray]] = []
    for name, rgb in _NODE_COLORS.items():
        comps = _components(_mask(arr, rgb), _MIN_DISK_PIXELS)
        if len(comps) != 1:
            return ABSTAIN  # every special junction must appear exactly once
        nodes.append((name, comps[0]))
    for comp in _components(_mask(arr, _PLAIN), _MIN_DISK_PIXELS):
        nodes.append(("plain", comp))
    if len(nodes) < 5:
        return ABSTAIN

    # --- strokes ---------------------------------------------------------
    strokes = _components(_mask(arr, _STROKE), _MIN_STROKE_PIXELS)
    if not strokes:
        return ABSTAIN

    grown_nodes = [_dilate(m, _DILATE) for _, m in nodes]
    adjacency: dict[int, set[int]] = {i: set() for i in range(len(nodes))}
    for stroke in strokes:
        touching = [i for i, gm in enumerate(grown_nodes) if bool((gm & stroke).any())]
        if len(touching) != 2:
            return ABSTAIN  # a stroke must join exactly two junctions
        a, b = touching
        adjacency[a].add(b)
        adjacency[b].add(a)

    if len(strokes) != len(nodes) - 1:
        return ABSTAIN  # the picture must show one tree, so routes stay unique
    labels = [name for name, _ in nodes]
    src = labels.index("green")
    dst = labels.index("red")

    prev = {src: -1}
    queue = deque([src])
    while queue:
        cur = queue.popleft()
        if cur == dst:
            break
        for nxt in sorted(adjacency[cur]):
            if nxt not in prev:
                prev[nxt] = cur
                queue.append(nxt)
    if dst not in prev:
        return ABSTAIN

    route = [dst]
    while route[-1] != src:
        route.append(prev[route[-1]])

    on_route = [labels[i] for i in route if labels[i] in _MARKS]
    if len(on_route) != 1:
        return ABSTAIN
    return on_route[0]
