"""Renderer and analytic replay for two independent phase registers."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont


RENDERER_VERSION = "dual-register-concordance-0.1.0"
WIDTH = 768
HEIGHT = 512
SUPERSAMPLE = 4

PANEL_WIDTH = 260
PANEL_HEIGHT = 424
PANEL_X_CHOICES = (24, 30, 36)
PANEL_Y_CHOICES = (38, 44, 50)

BACKGROUND = (255, 255, 255)
PANEL_FILL = (237, 242, 246)
PANEL_EDGE = (47, 66, 78)
CARD_FILL = (250, 250, 248)
CARD_EDGE = (166, 178, 184)
DIAL_FILL = (253, 253, 251)
DIAL_EDGE = (112, 130, 140)
INK = (54, 76, 89)
TEXT = (43, 61, 72)
ACCENT_PALETTE = (
    (220, 78, 89),
    (48, 123, 201),
    (55, 161, 111),
    (216, 150, 42),
)
OPS = ("A", "R", "S")
PHASE_LABELS = ("1", "2", "3")
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    try:
        return ImageFont.truetype(FONT_PATH, size)
    except OSError:
        return ImageFont.load_default()


def _box(values: tuple[float, ...], scale: int) -> list[int]:
    return [int(round(value * scale)) for value in values]


def _center_text(
    draw: ImageDraw.ImageDraw,
    text: str,
    center_x: float,
    top_y: float,
    font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    fill: tuple[int, int, int],
    scale: int,
) -> None:
    bounds = draw.textbbox((0, 0), text, font=font)
    width = bounds[2] - bounds[0]
    draw.text(
        (int(round(center_x * scale - width / 2)), int(round(top_y * scale))),
        text,
        fill=fill,
        font=font,
    )


def _valid_register(initial: Any, operations: Any) -> bool:
    return (
        isinstance(initial, int)
        and not isinstance(initial, bool)
        and initial in (0, 1, 2)
        and isinstance(operations, list)
        and len(operations) == 5
        and all(isinstance(item, str) and item in OPS for item in operations)
    )


def _replay(initial: int, operations: list[str]) -> int:
    """Apply the three visible permutations to a phase index."""

    phase = int(initial)
    for operation in operations:
        if operation == "A":
            phase = (phase + 1) % 3
        elif operation == "R":
            phase = (phase - 1) % 3
        elif operation == "S":
            if phase == 1:
                phase = 2
            elif phase == 2:
                phase = 1
    return phase


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample balanced, unambiguous paired register replays."""

    rng = random.Random(int(seed))
    left_ops = ["A", "R", "S", "A", "R"]
    right_ops = ["A", "R", "S", "A", "R"]
    rng.shuffle(left_ops)
    rng.shuffle(right_ops)
    left_initial = rng.randrange(3)
    left_final = _replay(left_initial, left_ops)

    same = rng.randrange(2) == 0
    matching_right = [
        phase for phase in range(3) if _replay(phase, right_ops) == left_final
    ]
    if same:
        right_initial = matching_right[0]
    else:
        right_initial = next(phase for phase in range(3) if phase not in matching_right)

    accent = [list(color) for color in ACCENT_PALETTE]
    rng.shuffle(accent)
    left_x = rng.choice(PANEL_X_CHOICES)
    top_y = rng.choice(PANEL_Y_CHOICES)
    return {
        "left_initial": left_initial,
        "left_ops": left_ops,
        "right_initial": right_initial,
        "right_ops": right_ops,
        "origin": [left_x, top_y],
        "palette": [accent[0], accent[1]],
    }


def _draw_operation(
    draw: ImageDraw.ImageDraw,
    center_x: float,
    center_y: float,
    operation: str,
    scale: int,
) -> None:
    width = 5 * scale
    ink = INK
    cx, cy = center_x * scale, center_y * scale
    if operation == "A":
        draw.line(
            [(int(round(cx - 20 * scale)), int(round(cy))),
             (int(round(cx + 5 * scale)), int(round(cy)))],
            fill=ink,
            width=width,
        )
        draw.polygon(
            [
                (int(round(cx + 18 * scale)), int(round(cy))),
                (int(round(cx + 3 * scale)), int(round(cy - 11 * scale))),
                (int(round(cx + 3 * scale)), int(round(cy + 11 * scale))),
            ],
            fill=ink,
        )
    elif operation == "R":
        draw.line(
            [(int(round(cx - 5 * scale)), int(round(cy))),
             (int(round(cx + 20 * scale)), int(round(cy)))],
            fill=ink,
            width=width,
        )
        draw.polygon(
            [
                (int(round(cx - 18 * scale)), int(round(cy))),
                (int(round(cx - 3 * scale)), int(round(cy - 11 * scale))),
                (int(round(cx - 3 * scale)), int(round(cy + 11 * scale))),
            ],
            fill=ink,
        )
    else:
        draw.polygon(
            [
                (int(round(cx)), int(round(cy - 13 * scale))),
                (int(round(cx + 13 * scale)), int(round(cy))),
                (int(round(cx)), int(round(cy + 13 * scale))),
                (int(round(cx - 13 * scale)), int(round(cy))),
            ],
            fill=ink,
        )


def _draw_register(
    draw: ImageDraw.ImageDraw,
    x: float,
    y: float,
    title: str,
    initial: int,
    operations: list[str],
    accent: tuple[int, int, int],
    scale: int,
    title_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    body_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
    phase_font: ImageFont.FreeTypeFont | ImageFont.ImageFont,
) -> None:
    draw.rectangle(
        _box((x, y, x + PANEL_WIDTH, y + PANEL_HEIGHT), scale),
        fill=PANEL_FILL,
        outline=PANEL_EDGE,
        width=3 * scale,
    )
    _center_text(draw, title, x + PANEL_WIDTH / 2, y + 14, title_font, TEXT, scale)
    _center_text(draw, "START PHASE", x + PANEL_WIDTH / 2, y + 45, body_font, TEXT, scale)

    phase_y = y + 91
    phase_xs = (x + 58, x + 130, x + 202)
    for index, phase_x in enumerate(phase_xs):
        draw.ellipse(
            _box((phase_x - 22, phase_y - 22, phase_x + 22, phase_y + 22), scale),
            fill=DIAL_FILL,
            outline=DIAL_EDGE,
            width=2 * scale,
        )
        _center_text(draw, PHASE_LABELS[index], phase_x, phase_y - 13, phase_font, TEXT, scale)
    selected_x = phase_xs[initial]
    draw.ellipse(
        _box((selected_x - 30, phase_y - 30, selected_x + 30, phase_y + 30), scale),
        outline=accent,
        width=5 * scale,
    )

    _center_text(draw, "OPERATIONS", x + PANEL_WIDTH / 2, y + 130, body_font, TEXT, scale)
    for index, operation in enumerate(operations):
        top = y + 151 + index * 49
        draw.rectangle(
            _box((x + 18, top, x + PANEL_WIDTH - 18, top + 38), scale),
            fill=CARD_FILL,
            outline=CARD_EDGE,
            width=1 * scale,
        )
        _center_text(draw, str(index + 1), x + 42, top + 8, body_font, TEXT, scale)
        _draw_operation(draw, x + 183, top + 19, operation, scale)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render two disconnected registers with no answer-specific decoration."""

    left_initial = scene["left_initial"]
    left_ops = scene["left_ops"]
    right_initial = scene["right_initial"]
    right_ops = scene["right_ops"]
    origin = scene["origin"]
    palette = scene["palette"]
    left_x, top_y = (float(value) for value in origin)
    right_x = WIDTH - PANEL_WIDTH - left_x
    accents = [tuple(int(value) for value in color) for color in palette]

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    title_font = _font(13 * scale)
    body_font = _font(9 * scale)
    phase_font = _font(13 * scale)
    _draw_register(
        draw, left_x, top_y, "LEFT REGISTER", left_initial, left_ops,
        accents[0], scale, title_font, body_font, phase_font,
    )
    _draw_register(
        draw, right_x, top_y, "RIGHT REGISTER", right_initial, right_ops,
        accents[1], scale, title_font, body_font, phase_font,
    )
    if scale > 1:
        image = image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    """Replay both label-bearing registers and compare their terminal phases."""

    left_initial = scene.get("left_initial")
    left_ops = scene.get("left_ops")
    right_initial = scene.get("right_initial")
    right_ops = scene.get("right_ops")
    if not _valid_register(left_initial, left_ops) or not _valid_register(right_initial, right_ops):
        raise ValueError("scene does not contain two valid five-step phase registers")
    left_final = _replay(left_initial, left_ops)
    right_final = _replay(right_initial, right_ops)
    return "yes" if left_final == right_final else "no"


def margin(scene: dict[str, Any]) -> float:
    """Unit margin for a discrete equality boundary; invalid scenes are quarantined."""

    return 0.0 if is_quarantined(scene) else 1.0


def is_quarantined(scene: dict[str, Any]) -> bool:
    if not isinstance(scene, dict):
        return True
    if not _valid_register(scene.get("left_initial"), scene.get("left_ops")):
        return True
    if not _valid_register(scene.get("right_initial"), scene.get("right_ops")):
        return True
    origin = scene.get("origin")
    palette = scene.get("palette")
    return not (
        isinstance(origin, list)
        and len(origin) == 2
        and all(isinstance(value, int) for value in origin)
        and isinstance(palette, list)
        and len(palette) == 2
        and all(
            isinstance(color, list)
            and len(color) == 3
            and all(isinstance(value, int) and 0 <= value <= 255 for value in color)
            for color in palette
        )
    )


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """This explicitly labeled register encoding claims no latent aliases."""

    return []
