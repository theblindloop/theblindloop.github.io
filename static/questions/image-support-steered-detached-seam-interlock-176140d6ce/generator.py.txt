from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 384
BANDS = 8
QUARANTINE_MARGIN = 0.20
_OFFSETS = (-21, -15, -9, -3, 3, 9, 15, 21)
_PALETTES = (
    ((40, 122, 164), (25, 66, 86)),
    ((205, 91, 88), (104, 42, 48)),
    ((68, 145, 112), (30, 75, 58)),
    ((137, 102, 176), (67, 45, 92)),
)


def _profile_pair(seed: int, rng: random.Random) -> tuple[list[int], list[int]]:
    left = list(_OFFSETS)
    rng.shuffle(left)
    # Avoid a nearly monotone edge: the intended evidence is the complete irregular
    # contour, not one gross diagonal lean.
    while abs(sum((i - 3.5) * value for i, value in enumerate(left))) > 250:
        rng.shuffle(left)
    right = left.copy() if seed % 2 == 0 else [-value for value in left]
    return left, right


def sample_scene(seed: int) -> dict:
    rng = random.Random((int(seed) ^ 0x5EAD1A7) & 0xFFFFFFFFFFFFFFFF)
    profile_left, profile_right = _profile_pair(int(seed), rng)
    background = (249, 247, 241)
    ink, outline = _PALETTES[(int(seed) // 2) % len(_PALETTES)]
    top = rng.randint(48, 68)
    band_height = rng.randint(31, 34)
    left_outer = rng.randint(22, 32)
    body_width = rng.randint(110, 117)
    left_face = left_outer + body_width
    right_outer = WIDTH - left_outer
    right_face = right_outer - body_width
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": list(background),
        "ink": list(ink),
        "outline": list(outline),
        "top": top,
        "band_height": band_height,
        "left_outer": left_outer,
        "left_face": left_face,
        "right_face": right_face,
        "right_outer": right_outer,
        "profile_left": profile_left,
        "profile_right": profile_right,
    }


def _face_path(face: int, top: int, band_height: int, profile: Iterable[int]) -> list[tuple[int, int]]:
    values = [int(value) for value in profile]
    points = [(face + values[0], top)]
    for index, value in enumerate(values):
        y_next = top + (index + 1) * band_height
        points.append((face + value, y_next))
        if index + 1 < len(values):
            points.append((face + values[index + 1], y_next))
    return points


def _piece_polygon(scene: dict, side: str) -> list[tuple[int, int]]:
    top = int(scene["top"])
    band_height = int(scene["band_height"])
    bottom = top + BANDS * band_height
    if side == "left":
        outer = int(scene["left_outer"])
        path = _face_path(int(scene["left_face"]), top, band_height, scene["profile_left"])
        return [(outer, top), *path, (outer, bottom)]
    outer = int(scene["right_outer"])
    path = _face_path(int(scene["right_face"]), top, band_height, scene["profile_right"])
    return [*path, (outer, bottom), (outer, top)]


def render(scene: dict) -> Image.Image:
    scale = 4
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    ink = tuple(int(value) for value in scene["ink"])
    outline = tuple(int(value) for value in scene["outline"])
    image = Image.new("RGB", (width * scale, height * scale), background)
    draw = ImageDraw.Draw(image)
    for side in ("left", "right"):
        points = [(x * scale, y * scale) for x, y in _piece_polygon(scene, side)]
        draw.polygon(points, fill=ink)
        draw.line([*points, points[0]], fill=outline, width=3 * scale, joint="curve")
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _residuals(scene: dict) -> tuple[float, float]:
    left = [float(value) for value in scene["profile_left"]]
    right = [float(value) for value in scene["profile_right"]]
    left_mean = sum(left) / len(left)
    right_mean = sum(right) / len(right)
    a = [value - left_mean for value in left]
    b = [value - right_mean for value in right]
    same = sum((x - y) ** 2 for x, y in zip(a, b)) / len(a)
    opposite = sum((x + y) ** 2 for x, y in zip(a, b)) / len(a)
    return same, opposite


def analytic_gold(scene: dict) -> str:
    same, opposite = _residuals(scene)
    return "fit" if same < opposite else "clash"


def margin(scene: dict) -> float:
    same, opposite = _residuals(scene)
    return abs(opposite - same) / max(1.0, opposite + same)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
