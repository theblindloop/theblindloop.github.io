from __future__ import annotations

import math

import numpy as np
from PIL import Image


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    border = np.concatenate(
        (array[:3].reshape(-1, 3), array[-3:].reshape(-1, 3),
         array[:, :3].reshape(-1, 3), array[:, -3:].reshape(-1, 3)), axis=0
    )
    background = np.median(border, axis=0)
    distance = np.linalg.norm(array.astype(np.float32) - background, axis=2)
    return distance > 54.0


def _two_components(mask: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    height, width = mask.shape
    active_columns = np.flatnonzero(mask.sum(axis=0) >= 4)
    if len(active_columns) < width // 5:
        raise ValueError("too little foreground to reconstruct two pieces")
    gaps = np.diff(active_columns)
    if not len(gaps):
        raise ValueError("only one foreground span")
    gap_index = int(np.argmax(gaps))
    if int(gaps[gap_index]) < width * 0.20:
        raise ValueError("the two detached components are not visibly separated")
    split = (int(active_columns[gap_index]) + int(active_columns[gap_index + 1])) // 2
    left = mask.copy()
    left[:, split + 1:] = False
    right = mask.copy()
    right[:, :split + 1] = False
    if left.sum() < 6000 or right.sum() < 6000:
        raise ValueError("a reconstructed piece is incomplete")
    return left, right


def _bounds(mask: np.ndarray) -> tuple[int, int, int, int, int]:
    ys, xs = np.nonzero(mask)
    if not len(xs):
        raise ValueError("empty component")
    return int(xs.min()), int(ys.min()), int(xs.max()), int(ys.max()), int(len(xs))


def _validate_pair(left: np.ndarray, right: np.ndarray) -> tuple[tuple[int, int, int, int, int], tuple[int, int, int, int, int]]:
    lb = _bounds(left)
    rb = _bounds(right)
    image_height, image_width = left.shape
    lw, lh = lb[2] - lb[0] + 1, lb[3] - lb[1] + 1
    rw, rh = rb[2] - rb[0] + 1, rb[3] - rb[1] + 1
    if min(lh, rh) < image_height * 0.58 or min(lw, rw) < image_width * 0.19:
        raise ValueError("piece extent is too small")
    if abs(lh - rh) > 4 or abs(lw - rw) > 5:
        raise ValueError("paired piece extents disagree")
    if abs(lb[1] - rb[1]) > 3 or abs(lb[3] - rb[3]) > 3:
        raise ValueError("facing edges are not vertically registered")
    area_ratio = min(lb[4], rb[4]) / max(lb[4], rb[4])
    if area_ratio < 0.955:
        raise ValueError("one detached component has been materially erased")
    return lb, rb


def _facing_profiles(left: np.ndarray, right: np.ndarray, lb: tuple[int, int, int, int, int], rb: tuple[int, int, int, int, int]) -> tuple[np.ndarray, np.ndarray]:
    y0 = max(lb[1], rb[1]) + 4
    y1 = min(lb[3], rb[3]) - 4
    left_edge = []
    right_edge = []
    for y in range(y0, y1 + 1):
        left_x = np.flatnonzero(left[y])
        right_x = np.flatnonzero(right[y])
        if not len(left_x) or not len(right_x):
            raise ValueError("a facing contour has a missing scanline")
        left_edge.append(float(left_x.max()))
        right_edge.append(float(right_x.min()))
    a = np.asarray(left_edge, dtype=np.float64)
    b = np.asarray(right_edge, dtype=np.float64)
    # Remove only each component's horizontal translation. The remaining vectors
    # are the full raster-recovered facing contour profiles.
    a -= np.mean(a)
    b -= np.mean(b)
    if np.std(a) < 5.0 or np.std(b) < 5.0:
        raise ValueError("jagged contour energy is insufficient")
    return a, b


def decision_from_image(image: Image.Image) -> str:
    mask = _foreground(image)
    left, right = _two_components(mask)
    lb, rb = _validate_pair(left, right)
    a, b = _facing_profiles(left, right, lb, rb)
    same_residual = float(np.mean((a - b) ** 2))
    opposite_residual = float(np.mean((a + b) ** 2))
    total = same_residual + opposite_residual
    if not math.isfinite(total) or total <= 1.0:
        raise ValueError("contour relation is unresolved")
    normalized_gap = abs(opposite_residual - same_residual) / total
    if normalized_gap <= 0.20:
        raise ValueError("contour relation lies in the quarantine band")
    return "fit" if same_residual < opposite_residual else "clash"
