"""Unique-pair intersection search world.

The image is split into four quadrants (top-left, top-right, bottom-left,
bottom-right) by two light-gray divider lines. Each quadrant holds one dark
filled circle and one dark bar (a thick line segment). In exactly one quadrant
the circle and the bar overlap each other (their filled regions intersect, so
they merge into a single connected dark blob); in the other three quadrants the
circle sits clearly away from the bar with a visible gap between the two
shapes. The decision is the quadrant containing the only overlapping pair. No
reference stimulus is shown; the answer is fixed purely by internal spatial
intersection within each quadrant. All rendering is deterministic and every
relevant relation is visible in the raster.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

SIZE = 480
QUADRANT = SIZE // 2
BG = (255, 255, 255)
SEG = (24, 24, 24)
GRAY = (198, 198, 198)

SECTORS = ("TL", "TR", "BL", "BR")
# Quadrant centers slightly inset so pairs stay clear of the divider lines.
SECTOR_CENTERS = {
    "TL": (120, 120),
    "TR": (360, 120),
    "BL": (120, 360),
    "BR": (360, 360),
}

BAR_THICK = 14
BAR_LEN_RANGE = (62, 76)
RADIUS_RANGE = (13, 18)
# Bar angles (undirected, degrees from horizontal) used purely for variety.
BAR_ANGLES_DEG = (0, 30, 60, 90, 120, 150)

# A pair is "overlapping" when the disk center-to-segment distance is below
# R + T/2. We require a clear positive overlap in the winner and a clear
# separation gap in every distractor so the pixel oracle and analytic gold
# always agree.
OVERLAP_MIN = 10
CLEAR_MIN = 28
OVERLAP_MAX = 20
CLEAR_MAX = 44
QUARANTINE_THRESH = 9.0


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    winner_sector = SECTORS[int(rng.integers(0, len(SECTORS)))]

    bars = []
    circles = []
    for sector in SECTORS:
        qcx, qcy = SECTOR_CENTERS[sector]
        length = int(rng.integers(*BAR_LEN_RANGE))
        radius = int(rng.integers(*RADIUS_RANGE))
        angle_deg = int(BAR_ANGLES_DEG[int(rng.integers(0, len(BAR_ANGLES_DEG)))])
        side = 1 if int(rng.integers(0, 2)) == 0 else -1

        # Small deterministic jitter around the quadrant center so positions are
        # not locked to a fixed lattice while the whole pair stays inside its
        # quadrant box (each box is QUADRANT x QUADRANT centered on the sector
        # center, with the paired objects extending at most ~90 px from it).
        cx, cy = qcx, qcy
        max_jitter = 18
        jx = int(rng.integers(-max_jitter, max_jitter + 1))
        jy = int(rng.integers(-max_jitter, max_jitter + 1))
        cx = qcx + jx
        cy = qcy + jy

        theta = np.deg2rad(angle_deg)
        ux, uy = np.cos(theta), np.sin(theta)
        # Perpendicular unit vector (the circle is offset across the bar).
        nx, ny = -uy, ux
        thick = BAR_THICK / 2.0
        hit = (radius + thick)  # center-to-axis distance that touches

        if sector == winner_sector:
            overlap = float(rng.uniform(OVERLAP_MIN, OVERLAP_MAX))
            dist = hit - overlap
        else:
            clearance = float(rng.uniform(CLEAR_MIN, CLEAR_MAX))
            dist = hit + clearance

        c_cx = int(round(cx + side * dist * nx))
        c_cy = int(round(cy + side * dist * ny))

        bars.append(
            {
                "sector": sector,
                "cx": int(cx),
                "cy": int(cy),
                "length": int(length),
                "thickness": int(BAR_THICK),
                "angle_deg": angle_deg,
            }
        )
        circles.append(
            {
                "sector": sector,
                "cx": int(c_cx),
                "cy": int(c_cy),
                "radius": int(radius),
            }
        )

    return {"bars": bars, "circles": circles}


def _stamp_bar(arr: np.ndarray, bar: dict) -> None:
    theta = np.deg2rad(bar["angle_deg"])
    ys, xs = np.ogrid[:SIZE, :SIZE]
    dx = xs - bar["cx"]
    dy = ys - bar["cy"]
    u = dx * np.cos(-theta) - dy * np.sin(-theta)
    v = dx * np.sin(-theta) + dy * np.cos(-theta)
    half = bar["length"] / 2.0
    thick = bar["thickness"] / 2.0
    mask = (np.abs(u) <= half) & (np.abs(v) <= thick)
    arr[mask] = SEG


def _stamp_circle(arr: np.ndarray, circle: dict) -> None:
    ys, xs = np.ogrid[:SIZE, :SIZE]
    dx = xs - circle["cx"]
    dy = ys - circle["cy"]
    mask = (dx * dx + dy * dy) <= (circle["radius"] * circle["radius"])
    arr[mask] = SEG


def render(scene: dict) -> Image.Image:
    arr = np.empty((SIZE, SIZE, 3), dtype=np.uint8)
    arr[:] = BG
    arr[:, QUADRANT - 1 : QUADRANT + 2] = GRAY
    arr[QUADRANT - 1 : QUADRANT + 2, :] = GRAY
    for bar in scene["bars"]:
        _stamp_bar(arr, bar)
    for circle in scene["circles"]:
        _stamp_circle(arr, circle)
    return Image.fromarray(arr, "RGB")


def _pair_state(sector: str, bars: list, circles: dict[str, dict]) -> tuple[float, bool]:
    bar = next(b for b in bars if b["sector"] == sector)
    circle = circles[sector]
    theta = np.deg2rad(bar["angle_deg"])
    ux, uy = np.cos(theta), np.sin(theta)
    bx, by = bar["cx"], bar["cy"]
    c_cx, c_cy = circle["cx"], circle["cy"]
    # Distance from the circle center to the bar's central axis line.
    vec_x, vec_y = c_cx - bx, c_cy - by
    dist = abs(vec_x * (-uy) + vec_y * ux)
    hit = circle["radius"] + bar["thickness"] / 2.0
    gap = dist - hit
    return gap, gap < 0  # negative gap means overlapping


def _decision(scene: dict) -> tuple[str, float]:
    bars = scene["bars"]
    circles = {c["sector"]: c for c in scene["circles"]}
    gaps = {}
    for sector in SECTORS:
        gap, overlapping = _pair_state(sector, bars, circles)
        gaps[sector] = gap
    winners = [s for s in SECTORS if gaps[s] < 0]
    winner = winners[0] if winners else SECTORS[0]
    margin = min(abs(g) for g in gaps.values())
    return winner, float(margin)


def analytic_gold(scene: dict) -> str:
    sector, _ = _decision(scene)
    return sector


def margin(scene: dict) -> float:
    _, m = _decision(scene)
    return m


def is_quarantined(scene: dict) -> bool:
    _, m = _decision(scene)
    return bool(m < QUARANTINE_THRESH)


def latent_symmetries(scene: dict) -> list:
    return []
