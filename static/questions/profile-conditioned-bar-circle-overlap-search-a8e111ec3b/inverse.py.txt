"""Independent pixel-only inverse arm for bar_circle_overlap_search.

Receives only a PIL image. It isolates the four quadrants, extracts the dark
objects in each, runs connected-component labeling on the dark pixels, and
selects as the winner the single quadrant whose dark objects form exactly ONE
connected component (the circle merged into the overlapping bar). The three
distractor quadrants each show the circle and bar as two separate components.

Imports nothing from the renderer, prompts, generator, or verifier.
"""

from __future__ import annotations

from collections import deque

import numpy as np

DARK_SUM = 150
QUADRANT = 240


def _components_in_quadrant(dark: np.ndarray, x0: int, x1: int, y0: int, y1: int) -> int:
    """Count 4-connected dark components inside the [x0,x1)x[y0,y1) sub-box."""
    sub = dark[y0:y1, x0:x1]
    ys, xs = np.nonzero(sub)
    if xs.size == 0:
        return 0
    h, w = sub.shape
    coords = set(zip(xs.tolist(), ys.tolist()))
    seen = set()
    count = 0
    for start in coords:
        if start in seen:
            continue
        count += 1
        queue = deque([start])
        seen.add(start)
        while queue:
            x, y = queue.popleft()
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < w and 0 <= ny < h and (nx, ny) in coords and (nx, ny) not in seen:
                    seen.add((nx, ny))
                    queue.append((nx, ny))
    return count


def decision_from_image(image) -> str:
    w, h = image.size
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    dark = arr.sum(axis=2) < DARK_SUM

    q = QUADRANT
    counts = {
        "TL": _components_in_quadrant(dark, 0, q, 0, q),
        "TR": _components_in_quadrant(dark, q, w, 0, q),
        "BL": _components_in_quadrant(dark, 0, q, q, h),
        "BR": _components_in_quadrant(dark, q, w, q, h),
    }
    one = [sector for sector, count in counts.items() if count == 1]
    if len(one) == 1:
        return one[0]
    # Fallback: the fewest-component quadrant is the overlapping pair.
    if counts:
        return min(counts, key=lambda k: counts[k])
    return "TL"
