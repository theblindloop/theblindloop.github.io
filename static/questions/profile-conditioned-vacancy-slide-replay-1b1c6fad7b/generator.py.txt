"""Deterministic latent sampler and Pillow renderer."""

from __future__ import annotations

import random
from typing import Any

from PIL import Image, ImageDraw, ImageFont

WIDTH = 1500
HEIGHT = 620
CELL = 126
BOARD_W = CELL * 3
BOARD_H = CELL * 2

PALETTES = (
    {
        "background": (250, 248, 242),
        "ink": (32, 43, 57),
        "cell": (224, 235, 241),
        "tile": (246, 194, 68),
        "card": (96, 184, 205),
        "accent": (211, 52, 65),
        "empty": (207, 216, 220),
    },
    {
        "background": (247, 250, 246),
        "ink": (38, 43, 46),
        "cell": (232, 225, 244),
        "tile": (111, 194, 134),
        "card": (238, 155, 72),
        "accent": (202, 45, 78),
        "empty": (211, 216, 210),
    },
    {
        "background": (249, 247, 252),
        "ink": (35, 38, 54),
        "cell": (224, 232, 250),
        "tile": (225, 142, 193),
        "card": (100, 188, 158),
        "accent": (213, 57, 48),
        "empty": (211, 213, 224),
    },
)

_PIP_OFFSETS = {
    1: ((0, 0),),
    2: ((-1, -1), (1, 1)),
    3: ((-1, -1), (0, 0), (1, 1)),
    4: ((-1, -1), (1, -1), (-1, 1), (1, 1)),
    5: ((-1, -1), (1, -1), (0, 0), (-1, 1), (1, 1)),
}


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
    path = "/usr/share/fonts/truetype/dejavu/" + name
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _adjacent(left: int, right: int) -> bool:
    lr, lc = divmod(left, 3)
    rr, rc = divmod(right, 3)
    return abs(lr - rr) + abs(lc - rc) == 1


def _replay(board: list[int], moves: list[int]) -> tuple[list[int], int, int, set[int]]:
    state = list(board)
    legal = 0
    illegal = 0
    blank_path = {state.index(0)}
    for tile in moves:
        vacancy = state.index(0)
        tile_pos = state.index(tile)
        if _adjacent(vacancy, tile_pos):
            state[vacancy], state[tile_pos] = state[tile_pos], state[vacancy]
            legal += 1
            blank_path.add(tile_pos)
        else:
            illegal += 1
    return state, legal, illegal, blank_path


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced scene; final answer is deterministically seed mod six."""

    rng = random.Random(int(seed))
    desired = int(seed) % 6
    for _ in range(20000):
        board = list(range(6))
        rng.shuffle(board)
        target = rng.randint(1, 5)
        moves = [rng.randint(1, 5) for _ in range(8)]
        final_board, legal, illegal, blank_path = _replay(board, moves)
        start_position = board.index(target)
        final_position = final_board.index(target)
        if (
            final_position == desired
            and legal >= 3
            and illegal >= 2
            and start_position != final_position
            and len(blank_path) >= 3
            and len(set(moves)) >= 3
        ):
            return {
                "board": board,
                "target": target,
                "moves": moves,
                "palette_id": rng.randrange(len(PALETTES)),
                "dx": rng.randint(-8, 8),
                "dy": rng.randint(-6, 6),
                "layout": rng.randrange(2),
                "scale_id": rng.randrange(2),
            }
    raise RuntimeError(f"could not sample balanced scene for seed {seed}")


def analytic_gold(scene: dict[str, Any]) -> str:
    board = list(scene["board"])
    target = int(scene["target"])
    moves = list(scene["moves"])
    final_board, _, _, _ = _replay(board, moves)
    return "ABCDEF"[final_board.index(target)]


def margin(scene: dict[str, Any]) -> float:
    """Pixel runner-up margin for component/slot assignment."""

    return 38.0 - 4.0 * int(scene["scale_id"])


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 20.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []


def _draw_pips(
    draw: ImageDraw.ImageDraw,
    center: tuple[int, int],
    count: int,
    ink: tuple[int, int, int],
    radius: int,
    spread: int,
) -> None:
    cx, cy = center
    for ox, oy in _PIP_OFFSETS[count]:
        x = cx + ox * spread
        y = cy + oy * spread
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ink)


def render(scene: dict[str, Any]) -> Image.Image:
    board = list(scene["board"])
    target = int(scene["target"])
    moves = list(scene["moves"])
    palette_id = int(scene["palette_id"])
    dx = int(scene["dx"])
    dy = int(scene["dy"])
    layout = int(scene["layout"])
    scale_id = int(scene["scale_id"])

    colors = PALETTES[palette_id]
    image = Image.new("RGB", (WIDTH, HEIGHT), colors["background"])
    draw = ImageDraw.Draw(image)
    title_font = _font(31, bold=True)
    label_font = _font(25, bold=True)
    small_font = _font(19, bold=True)
    draw.text((48 + dx, 34 + dy), "SLIDE A NAMED TILE ONLY WHEN IT SHARES A SIDE WITH EMPTY", fill=colors["ink"], font=title_font)

    board_x = (70 if layout == 0 else 1060) + dx
    board_y = 188 + dy
    cards_x = (520 if layout == 0 else 56) + dx
    cards_y = 232 + dy

    draw.rounded_rectangle(
        (board_x - 22, board_y - 55, board_x + BOARD_W + 22, board_y + BOARD_H + 22),
        radius=18,
        fill=colors["cell"],
        outline=colors["ink"],
        width=4,
    )
    draw.text((board_x, board_y - 44), "START BOARD", fill=colors["ink"], font=small_font)
    tile_size = 76 + 8 * scale_id
    pip_radius = 7 + scale_id
    pip_spread = 18 + 2 * scale_id
    for index, value in enumerate(board):
        row, col = divmod(index, 3)
        x0 = board_x + col * CELL
        y0 = board_y + row * CELL
        draw.rectangle((x0, y0, x0 + CELL, y0 + CELL), outline=colors["ink"], width=4)
        draw.text((x0 + 8, y0 + 5), "ABCDEF"[index], fill=colors["ink"], font=small_font)
        cx = x0 + CELL // 2
        cy = y0 + CELL // 2 + 7
        if value == 0:
            empty_box = (cx - 36, cy - 27, cx + 36, cy + 27)
            draw.rounded_rectangle(empty_box, radius=8, fill=colors["empty"], outline=colors["ink"], width=3)
            draw.text((cx - 31, cy - 11), "EMPTY", fill=colors["ink"], font=_font(15, bold=True))
            continue
        half = tile_size // 2
        box = (cx - half, cy - half, cx + half, cy + half)
        draw.rounded_rectangle(box, radius=12, fill=colors["tile"], outline=colors["ink"], width=4)
        _draw_pips(draw, (cx, cy), value, colors["ink"], pip_radius, pip_spread)
        if value == target:
            pad = 8
            draw.rounded_rectangle(
                (box[0] - pad, box[1] - pad, box[2] + pad, box[3] + pad),
                radius=17,
                outline=colors["accent"],
                width=7,
            )

    draw.text((cards_x, cards_y - 74), "ATTEMPT CARDS  1 → 8", fill=colors["ink"], font=small_font)
    card_w = 92
    card_h = 154
    card_gap = 14
    for index, value in enumerate(moves):
        x0 = cards_x + index * (card_w + card_gap)
        y0 = cards_y
        draw.rounded_rectangle(
            (x0, y0, x0 + card_w, y0 + card_h),
            radius=13,
            fill=colors["card"],
            outline=colors["ink"],
            width=4,
        )
        draw.text((x0 + 35, y0 + 9), str(index + 1), fill=colors["ink"], font=_font(17, bold=True))
        _draw_pips(draw, (x0 + card_w // 2, y0 + 91), value, colors["ink"], 7, 18)
        if index < len(moves) - 1:
            ax = x0 + card_w + 2
            ay = y0 + card_h // 2
            draw.line((ax, ay, ax + card_gap - 4, ay), fill=colors["ink"], width=3)
            draw.polygon(((ax + card_gap - 4, ay), (ax + card_gap - 10, ay - 5), (ax + card_gap - 10, ay + 5)), fill=colors["ink"])

    legend_y = 535 + dy
    draw.rectangle((50 + dx, legend_y, 82 + dx, legend_y + 18), fill=colors["accent"])
    draw.text((94 + dx, legend_y - 5), "red outline = tracked tile", fill=colors["ink"], font=_font(18))
    draw.text((540 + dx, legend_y - 5), "diagonal contact does not count", fill=colors["ink"], font=_font(18))
    return image
