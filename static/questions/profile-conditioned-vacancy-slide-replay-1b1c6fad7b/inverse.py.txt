"""Independent pixel-only inverse for vacancy-slide scenes.

This module deliberately duplicates no renderer imports or latent access. It
recovers semantic components from solid-color raster regions and relative
geometry, then replays the visible cards.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_TILE_COLORS = {
    (246, 194, 68),
    (111, 194, 134),
    (225, 142, 193),
}
_CARD_COLORS = {
    (96, 184, 205),
    (238, 155, 72),
    (100, 188, 158),
}
_ACCENT_COLORS = {
    (211, 52, 65),
    (202, 45, 78),
    (213, 57, 48),
}
_INK_COLORS = {
    (32, 43, 57),
    (38, 43, 46),
    (35, 38, 54),
}


def _mask_for(array: np.ndarray, colors: set[tuple[int, int, int]]) -> np.ndarray:
    mask = np.zeros(array.shape[:2], dtype=bool)
    for color in colors:
        mask |= np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    return mask


def _components(mask: np.ndarray, minimum: int = 1) -> list[dict[str, float | int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[dict[str, float | int]] = []
    for y, x in zip(*np.nonzero(mask & ~seen)):
        if seen[y, x]:
            continue
        queue = deque([(int(y), int(x))])
        seen[y, x] = True
        count = 0
        sum_x = 0
        sum_y = 0
        min_x = max_x = int(x)
        min_y = max_y = int(y)
        while queue:
            cy, cx = queue.popleft()
            count += 1
            sum_x += cx
            sum_y += cy
            min_x = min(min_x, cx)
            max_x = max(max_x, cx)
            min_y = min(min_y, cy)
            max_y = max(max_y, cy)
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if count >= minimum:
            output.append(
                {
                    "area": count,
                    "cx": sum_x / count,
                    "cy": sum_y / count,
                    "x0": min_x,
                    "x1": max_x,
                    "y0": min_y,
                    "y1": max_y,
                }
            )
    return output


def _contained_pip_count(
    ink_components: list[dict[str, float | int]],
    region: dict[str, float | int],
    lower_y_fraction: float = 0.0,
) -> int:
    x0 = float(region["x0"])
    x1 = float(region["x1"])
    y0 = float(region["y0"])
    y1 = float(region["y1"])
    cutoff = y0 + lower_y_fraction * (y1 - y0)
    matches = [
        item
        for item in ink_components
        if 90 <= int(item["area"]) <= 280
        and x0 + 3 < float(item["cx"]) < x1 - 3
        and cutoff < float(item["cy"]) < y1 - 3
    ]
    return len(matches)


def _cluster_axis(values: list[float], expected: int) -> list[float]:
    ordered = sorted(values)
    if len(ordered) < expected:
        raise ValueError("too few visible objects for slot clustering")
    if expected == 2:
        gaps = [(ordered[i + 1] - ordered[i], i) for i in range(len(ordered) - 1)]
        _, split = max(gaps)
        groups = (ordered[: split + 1], ordered[split + 1 :])
        if not all(groups):
            raise ValueError("ambiguous row clustering")
        return [sum(group) / len(group) for group in groups]
    if expected == 3:
        unique = []
        for value in ordered:
            if not unique or value - unique[-1][-1] > 45:
                unique.append([value])
            else:
                unique[-1].append(value)
        if len(unique) != 3:
            raise ValueError("ambiguous column clustering")
        return [sum(group) / len(group) for group in unique]
    raise ValueError("unsupported cluster count")


def _nearest(value: float, centers: list[float]) -> int:
    distances = sorted((abs(value - center), index) for index, center in enumerate(centers))
    if len(distances) > 1 and distances[1][0] - distances[0][0] < 12:
        raise ValueError("ambiguous slot assignment")
    return distances[0][1]


def _adjacent(left: int, right: int) -> bool:
    lr, lc = divmod(left, 3)
    rr, rc = divmod(right, 3)
    return abs(lr - rr) + abs(lc - rc) == 1


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    tile_components = [
        item
        for item in _components(_mask_for(array, _TILE_COLORS), minimum=1000)
        if 55 <= int(item["x1"]) - int(item["x0"]) + 1 <= 90
        and 55 <= int(item["y1"]) - int(item["y0"]) + 1 <= 90
    ]
    card_components = [
        item
        for item in _components(_mask_for(array, _CARD_COLORS), minimum=2000)
        if 70 <= int(item["x1"]) - int(item["x0"]) + 1 <= 100
        and 120 <= int(item["y1"]) - int(item["y0"]) + 1 <= 160
    ]
    accent_components = _components(_mask_for(array, _ACCENT_COLORS), minimum=80)
    ink_components = _components(_mask_for(array, _INK_COLORS), minimum=40)

    if len(tile_components) != 5:
        raise ValueError(f"expected five tile components, found {len(tile_components)}")
    if len(card_components) != 8:
        raise ValueError(f"expected eight ordered cards, found {len(card_components)}")

    columns = _cluster_axis([float(item["cx"]) for item in tile_components], 3)
    rows = _cluster_axis([float(item["cy"]) for item in tile_components], 2)
    board = [0] * 6
    identities: dict[int, int] = {}
    for tile in tile_components:
        identity = _contained_pip_count(ink_components, tile)
        if identity not in range(1, 6) or identity in identities:
            raise ValueError("tile pip identities are missing or duplicated")
        slot = 3 * _nearest(float(tile["cy"]), rows) + _nearest(float(tile["cx"]), columns)
        if board[slot] != 0:
            raise ValueError("two tiles map to one board cell")
        board[slot] = identity
        identities[identity] = slot
    if sorted(identities) != [1, 2, 3, 4, 5] or board.count(0) != 1:
        raise ValueError("board reconstruction is not a five-tile permutation")

    ring_candidates = [
        accent
        for accent in accent_components
        if 70 <= int(accent["x1"]) - int(accent["x0"]) + 1 <= 115
        and 70 <= int(accent["y1"]) - int(accent["y0"]) + 1 <= 115
    ]
    if len(ring_candidates) != 1:
        raise ValueError("tracked-tile outline is missing or ambiguous")
    ring = ring_candidates[0]
    near = sorted(
        (
            (float(tile["cx"]) - float(ring["cx"])) ** 2
            + (float(tile["cy"]) - float(ring["cy"])) ** 2,
            identity,
        )
        for identity, slot in identities.items()
        for tile in tile_components
        if 3 * _nearest(float(tile["cy"]), rows) + _nearest(float(tile["cx"]), columns) == slot
    )
    if not near or near[0][0] > 100:
        raise ValueError("tracked outline does not surround a tile")
    target = near[0][1]

    moves = []
    for card in sorted(card_components, key=lambda item: float(item["cx"])):
        identity = _contained_pip_count(ink_components, card, lower_y_fraction=0.35)
        if identity not in range(1, 6):
            raise ValueError("card pip identity is unreadable")
        moves.append(identity)

    for tile in moves:
        vacancy = board.index(0)
        position = board.index(tile)
        if _adjacent(vacancy, position):
            board[vacancy], board[position] = board[position], board[vacancy]
    return "ABCDEF"[board.index(target)]
