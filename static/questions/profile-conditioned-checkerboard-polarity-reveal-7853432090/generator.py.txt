"""Renderer for the checkerboard-polarity reveal question world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw


SYMBOLS = ("arch", "fork", "knot", "stair")
CANVAS = (320, 300)
BASE_LEVEL = 128
MARKER_PALETTE = (
    (220, 45, 65),
    (35, 150, 75),
    (45, 105, 215),
    (190, 70, 175),
)


def _segments(symbol: str) -> tuple[tuple[tuple[float, float], tuple[float, float]], ...]:
    """Normalized centerlines for equal-stroke abstract glyphs."""
    if symbol == "arch":
        points = ((0.25, 0.72), (0.28, 0.56), (0.37, 0.43), (0.50, 0.38),
                  (0.63, 0.43), (0.72, 0.56), (0.75, 0.72))
    elif symbol == "fork":
        return (
            ((0.50, 0.74), (0.50, 0.46)),
            ((0.50, 0.47), (0.30, 0.28)),
            ((0.50, 0.47), (0.70, 0.28)),
        )
    elif symbol == "knot":
        return (
            ((0.33, 0.33), (0.67, 0.67)),
            ((0.67, 0.33), (0.33, 0.67)),
        )
    elif symbol == "stair":
        points = ((0.27, 0.72), (0.27, 0.58), (0.43, 0.58), (0.43, 0.43),
                  (0.59, 0.43), (0.59, 0.28), (0.75, 0.28))
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return tuple(zip(points[:-1], points[1:]))


def _symbol_mask(symbol: str, size: int) -> np.ndarray:
    yy, xx = np.indices((size, size), dtype=np.float32)
    nx = xx / float(size - 1)
    ny = yy / float(size - 1)
    best = np.full((size, size), np.inf, dtype=np.float32)
    for (x1, y1), (x2, y2) in _segments(symbol):
        vx, vy = x2 - x1, y2 - y1
        denom = vx * vx + vy * vy
        t = np.clip(((nx - x1) * vx + (ny - y1) * vy) / denom, 0.0, 1.0)
        dx = nx - (x1 + t * vx)
        dy = ny - (y1 + t * vy)
        best = np.minimum(best, dx * dx + dy * dy)
    return (best <= 0.026**2).astype(np.float32)


def sample_scene(seed: int) -> dict:
    """Sample a balanced, nuisance-randomized scene."""
    rng = random.Random(int(seed))
    size = rng.choice((144, 152, 160, 168))
    tile_x = rng.randint(38, CANVAS[0] - size - 38)
    tile_y = rng.randint(34, CANVAS[1] - size - 34)
    return {
        "symbol": SYMBOLS[(int(seed) * 7 + 3) % len(SYMBOLS)],
        "tile_x": tile_x,
        "tile_y": tile_y,
        "tile_size": size,
        "marker_color": list(rng.choice(MARKER_PALETTE)),
        "marker_size": rng.choice((9, 11, 13)),
        "checker_phase": rng.randrange(2),
        "carrier_amplitude": rng.choice((70, 76, 82)),
        "signal_strength": rng.choice((24, 32, 40, 48)),
        "background": rng.choice((238, 242, 246, 250)),
    }


def render(scene: dict) -> Image.Image:
    width, height = CANVAS
    background = int(scene["background"])
    canvas = np.full((height, width, 3), background, dtype=np.uint8)
    size = int(scene["tile_size"])
    yy, xx = np.indices((size, size), dtype=np.int16)
    phase = int(scene["checker_phase"]) % 2
    parity = np.where((xx + yy + phase) % 2 == 0, 1.0, -1.0)
    mask = _symbol_mask(str(scene["symbol"]), size)
    tile = BASE_LEVEL + parity * (
        float(scene["carrier_amplitude"]) + float(scene["signal_strength"]) * mask
    )
    tile = np.clip(np.rint(tile), 0, 255).astype(np.uint8)
    tx, ty = int(scene["tile_x"]), int(scene["tile_y"])
    canvas[ty : ty + size, tx : tx + size] = tile[:, :, None]

    output = Image.fromarray(canvas, mode="RGB")
    draw = ImageDraw.Draw(output)
    half = int(scene["marker_size"]) // 2
    color = tuple(int(value) for value in scene["marker_color"])
    centers = (
        (tx - 10, ty - 10),
        (tx + size + 9, ty - 10),
        (tx - 10, ty + size + 9),
        (tx + size + 9, ty + size + 9),
    )
    for cx, cy in centers:
        draw.rectangle((cx - half, cy - half, cx + half, cy + half), fill=color)
    return output


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    """One-sided transform margin above the quarantined signal floor."""
    return float(scene["signal_strength"]) - 28.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= 0.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["checker_phase"] = int(scene["checker_phase"]) + 2
    return [("checker_phase_period", twin)]
