"""Annulus-unwrap world: a ring of polar cells rectifies into a 5x7 symbol.

The scene is a single annulus divided into 5 concentric bands and 7 angular
sectors.  A coloured tab printed outside the ring marks the sector that becomes
the leftmost column; the innermost band becomes the top row.  Reading the ring
straight off the raster is not possible: neighbouring cells of one lattice row
are bent around the circle and the innermost band is compressed, so the symbol
only exists after the declared polar-to-rectangular rectification.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image

RENDERER_VERSION = "annulus_unwrap_reveal-renderer-0.1.0"

BANDS = 5
SECTORS = 7
TAU = 2.0 * math.pi
DELTA = TAU / SECTORS

# Five named 5x7 templates, each with exactly 15 marked cells so that the total
# printed ink cannot identify the answer.
TEMPLATES: dict[str, tuple[str, ...]] = {
    "cross": (
        "..###..",
        "...#...",
        "#######",
        "...#...",
        "..###..",
    ),
    "diamond": (
        "...#...",
        "..###..",
        "#######",
        "..###..",
        "...#...",
    ),
    "flag": (
        "#######",
        "#.##...",
        "#.##...",
        "#......",
        "#......",
    ),
    "goalpost": (
        "#######",
        "#.....#",
        "#.....#",
        "#.....#",
        "#.....#",
    ),
    "zigzag": (
        "#####..",
        "....#..",
        "..###..",
        "..#....",
        "..#####",
    ),
}

SYMBOL_NAMES: tuple[str, ...] = tuple(sorted(TEMPLATES))

# Label-preserving palettes.  In every palette the marked cells are the darkest
# ring colour, which is exactly what the question states.
PALETTES: tuple[dict[str, tuple[int, int, int]], ...] = (
    {
        "bg": (247, 246, 242),
        "plate": (226, 228, 232),
        "ink": (34, 38, 46),
        "hub": (214, 118, 44),
        "rim": (58, 106, 168),
        "mark": (176, 52, 122),
    },
    {
        "bg": (242, 244, 247),
        "plate": (214, 220, 214),
        "ink": (28, 44, 34),
        "hub": (196, 74, 62),
        "rim": (94, 78, 158),
        "mark": (26, 132, 128),
    },
    {
        "bg": (250, 248, 238),
        "plate": (232, 224, 208),
        "ink": (46, 34, 30),
        "hub": (68, 132, 74),
        "rim": (188, 96, 32),
        "mark": (60, 82, 176),
    },
)


def _bits_from_template(name: str) -> list[int]:
    rows = TEMPLATES[name]
    return [1 if rows[r][c] == "#" else 0 for r in range(BANDS) for c in range(SECTORS)]


def _bits_from_cells(cells) -> list[int]:
    bits = [0] * (BANDS * SECTORS)
    for pair in cells:
        band, sector = int(pair[0]), int(pair[1])
        bits[band * SECTORS + sector] = 1
    return bits


def _hamming(a: list[int], b: list[int]) -> int:
    return sum(1 for x, y in zip(a, b) if x != y)


def _ranked(bits: list[int]) -> list[tuple[int, str]]:
    scored = [(_hamming(bits, _bits_from_template(name)), name) for name in SYMBOL_NAMES]
    scored.sort()
    return scored


def sample_scene(seed: int) -> dict:
    rng = np.random.default_rng(seed)
    name = SYMBOL_NAMES[int(rng.integers(len(SYMBOL_NAMES)))]
    bits = _bits_from_template(name)
    n_flips = int(rng.integers(0, 4))
    if n_flips:
        idx = rng.choice(BANDS * SECTORS, size=n_flips, replace=False)
        for i in idx:
            bits[int(i)] ^= 1
    cells = [
        [i // SECTORS, i % SECTORS] for i in range(BANDS * SECTORS) if bits[i]
    ]
    palette = PALETTES[int(rng.integers(len(PALETTES)))]
    scene = {
        "size": 620,
        "cx": 310 + int(rng.integers(-12, 13)),
        "cy": 310 + int(rng.integers(-12, 13)),
        "r_in": int(rng.integers(46, 79)),
        "band_w": int(rng.integers(14, 35)),
        "start_deg": float(rng.integers(0, 360)),
        "cells": cells,
        "bg": list(palette["bg"]),
        "plate": list(palette["plate"]),
        "ink": list(palette["ink"]),
        "hub": list(palette["hub"]),
        "rim": list(palette["rim"]),
        "mark": list(palette["mark"]),
    }
    return scene


def _lum(color) -> float:
    r, g, b = (float(v) for v in color)
    return 0.299 * r + 0.587 * g + 0.114 * b


def render(scene: dict) -> Image.Image:
    size = int(scene["size"])
    cx, cy = float(scene["cx"]), float(scene["cy"])
    r_in = float(scene["r_in"])
    w = float(scene["band_w"])
    r_out = r_in + BANDS * w
    start = math.radians(float(scene["start_deg"]) % 360.0)

    ys, xs = np.mgrid[0:size, 0:size]
    dx = xs - cx
    dy = ys - cy
    r = np.hypot(dx, dy)
    # Screen angle measured from the 3-o'clock direction, increasing clockwise
    # because the raster y axis points down.
    phi = np.mod(np.arctan2(dy, dx), TAU)
    rel = np.mod(phi - start, TAU)

    img = np.empty((size, size, 3), dtype=np.uint8)
    img[:, :] = np.array(scene["bg"], dtype=np.uint8)

    plate_mask = (r >= r_in) & (r < r_out)
    img[plate_mask] = np.array(scene["plate"], dtype=np.uint8)

    band = np.clip(((r - r_in) / w).astype(np.int64), 0, BANDS - 1)
    sector = np.clip((rel / DELTA).astype(np.int64), 0, SECTORS - 1)
    ink = np.array(scene["ink"], dtype=np.uint8)
    for pair in scene["cells"]:
        b, s = int(pair[0]), int(pair[1])
        img[plate_mask & (band == b) & (sector == s)] = ink

    # Hairline separators, in the background colour, on the interior band and
    # sector boundaries only.  The outer and inner ring edges stay clean.
    bgc = np.array(scene["bg"], dtype=np.uint8)
    radial_off = np.mod(r - r_in, w)
    inner_edge = plate_mask & (r >= r_in + w - 0.5) & (radial_off < 1.5)
    img[inner_edge] = bgc
    ang_off = np.mod(rel, DELTA)
    arc = np.minimum(ang_off, DELTA - ang_off) * r
    img[plate_mask & (arc < 1.2)] = bgc

    # Fiducials: hub disk (centre), rim ring (outer radius), tab (start sector).
    img[r < r_in - 6.0] = np.array(scene["hub"], dtype=np.uint8)
    img[(r >= r_out) & (r < r_out + 6.0)] = np.array(scene["rim"], dtype=np.uint8)
    tab = (r >= r_out + 13.0) & (r < r_out + 30.0) & (rel < DELTA)
    img[tab] = np.array(scene["mark"], dtype=np.uint8)
    return Image.fromarray(img, mode="RGB")


def analytic_gold(scene: dict) -> str:
    bits = _bits_from_cells(scene["cells"])
    return _ranked(bits)[0][1]


def margin(scene: dict) -> float:
    bits = _bits_from_cells(scene["cells"])
    ranked = _ranked(bits)
    gap = float(ranked[1][0] - ranked[0][0])
    geo = min(float(scene["band_w"]), float(scene["r_in"]) * DELTA)
    contrast = abs(_lum(scene["ink"]) - _lum(scene["plate"]))
    return round(min(2.0 * gap, 0.5 * (geo - 12.0), contrast / 4.0), 6)


def is_quarantined(scene: dict) -> bool:
    return float(margin(scene)) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared pixel-identical, gold-preserving latent aliases."""
    reordered = dict(scene)
    reordered["cells"] = [list(pair) for pair in reversed(scene["cells"])]
    duplicated = dict(scene)
    first = list(scene["cells"][0])
    duplicated["cells"] = [list(pair) for pair in scene["cells"]] + [first, first]
    turned = dict(scene)
    turned["start_deg"] = float(scene["start_deg"]) + 360.0
    return [
        ("cell_draw_order", reordered),
        ("duplicate_cell_entries", duplicated),
        ("start_angle_full_turn", turned),
    ]


def _naive_read_distance(scene: dict) -> int:
    """Cells mis-read when the tab is ignored and sectors are read from east.

    Diagnostic only: it reports how much evidence the declared rectification
    actually contributes on this scene.
    """
    start = float(scene["start_deg"]) % 360.0
    shift = int(round(start / (360.0 / SECTORS))) % SECTORS
    bits = _bits_from_cells(scene["cells"])
    naive = [0] * (BANDS * SECTORS)
    for b in range(BANDS):
        for s in range(SECTORS):
            naive[b * SECTORS + s] = bits[b * SECTORS + (s - shift) % SECTORS]
    return _hamming(bits, naive)
