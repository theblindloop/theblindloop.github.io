from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

IDENTITIES = {
    "cyan": (20, 158, 190),
    "orange": (232, 126, 34),
    "purple": (135, 76, 176),
    "green": (36, 156, 92),
}
BORDERS = {
    "red": (211, 55, 55),
    "blue": (52, 102, 196),
    "gold": (205, 151, 25),
}
BLACK = (35, 39, 45)


def _mask(arr: np.ndarray, rgb, tolerance=18) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - np.array(rgb, dtype=np.int16)), axis=2) <= tolerance


def _components(mask: np.ndarray):
    seen = np.zeros(mask.shape, dtype=bool)
    h, w = mask.shape
    out = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = [(int(y), int(x))]
        seen[y, x] = True
        pts = []
        while q:
            cy, cx = q.pop()
            pts.append((cy, cx))
            for ny, nx in ((cy-1,cx),(cy+1,cx),(cy,cx-1),(cy,cx+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    q.append((ny, nx))
        ys = [p[0] for p in pts]
        xs = [p[1] for p in pts]
        out.append((len(pts), min(xs), min(ys), max(xs), max(ys)))
    return out


def _find_cards(arr: np.ndarray):
    options = []
    for name, rgb in BORDERS.items():
        comps = [c for c in _components(_mask(arr, rgb)) if c[3]-c[1] > 180 and c[4]-c[2] > 160]
        if len(comps) != 1:
            raise ValueError(f"cannot discover {name} option border")
        _, x0, y0, x1, y1 = comps[0]
        options.append((x0, y0, x1, y1, name))
    options.sort(key=lambda b: b[0])
    black = [c for c in _components(_mask(arr, BLACK)) if c[3]-c[1] > 400 and c[4]-c[2] > 170]
    if len(black) != 1:
        raise ValueError("cannot discover reference border")
    _, x0, y0, x1, y1 = black[0]
    return (x0, y0, x1, y1), options


def _word(arr: np.ndarray, box) -> tuple[str, ...]:
    x0, y0, x1, y1 = box[:4]
    crop = arr[y0+25:y1-5, x0+5:x1-4]
    endpoints = []
    for name, rgb in IDENTITIES.items():
        ys, xs = np.where(_mask(crop, rgb))
        if len(xs) < 80:
            raise ValueError(f"missing {name} interval")
        endpoints.append((float(xs.min()), f"{name}:S"))
        endpoints.append((float(xs.max()), f"{name}:E"))
    endpoints.sort()
    if min(b[0] - a[0] for a, b in zip(endpoints, endpoints[1:])) < 7:
        raise ValueError("ambiguous endpoint order")
    return tuple(token for _, token in endpoints)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"))
    reference_box, options = _find_cards(arr)
    reference = _word(arr, reference_box)
    words = [_word(arr, box) for box in options]
    hits = [i for i, word in enumerate(words) if word == reference]
    if len(hits) != 1:
        raise ValueError("pixel evidence has no unique endpoint-order match")
    return options[hits[0]][4]
