from __future__ import annotations

import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 650
IDENTITIES = ("cyan", "orange", "purple", "green")
RGB = {
    "cyan": (20, 158, 190),
    "orange": (232, 126, 34),
    "purple": (135, 76, 176),
    "green": (36, 156, 92),
    "red": (211, 55, 55),
    "blue": (52, 102, 196),
    "gold": (205, 151, 25),
    "black": (35, 39, 45),
}


def _valid_word(rng: random.Random) -> list[str]:
    # Uniform random interleaving of four two-event chains (start precedes end).
    remaining = {c: 0 for c in IDENTITIES}
    word: list[str] = []
    while len(word) < 8:
        available = []
        for color in IDENTITIES:
            if color not in remaining:
                continue
            if remaining[color] == 0:
                available.append((color, "S"))
            elif remaining[color] == 1:
                available.append((color, "E"))
        color, role = rng.choice(available)
        word.append(f"{color}:{role}")
        remaining[color] += 1
        if remaining[color] == 2:
            del remaining[color]
    return word


def _matches(a: list[str], b: list[str]) -> int:
    return sum(x == y for x, y in zip(a, b))


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    reference = _valid_word(rng)
    distractors: list[list[str]] = []
    for _ in range(2000):
        w = _valid_word(rng)
        if w != reference and w not in distractors and _matches(w, reference) <= 6:
            distractors.append(w)
            if len(distractors) == 2:
                break
    if len(distractors) != 2:
        raise RuntimeError("could not make unique interval distractors")

    gold_slot = seed % 3
    options = distractors[:]
    options.insert(gold_slot, list(reference))
    borders = ["red", "blue", "gold"]
    rng.shuffle(borders)

    layouts = []
    row_orders = []
    for _ in range(4):
        gaps = [rng.randint(85, 115) for _ in range(7)]
        layouts.append({"gaps": gaps, "jitter": rng.randint(-5, 5)})
        rows = list(IDENTITIES)
        rng.shuffle(rows)
        row_orders.append(rows)
    return {
        "reference_word": reference,
        "option_words": options,
        "layouts": layouts,
        "row_orders": row_orders,
        "border_colors": borders,
    }


def _positions(word: list[str], left: float, right: float, layout: dict) -> dict[str, float]:
    gaps = layout["gaps"]
    cumulative = [0]
    for gap in gaps:
        cumulative.append(cumulative[-1] + gap)
    usable_left = left + 25 + layout["jitter"]
    usable_right = right - 25 + layout["jitter"]
    scale = (usable_right - usable_left) / cumulative[-1]
    return {token: usable_left + cumulative[i] * scale for i, token in enumerate(word)}


def _draw_card(draw: ImageDraw.ImageDraw, box, word, rows, layout, border, label):
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=13, fill=(250, 250, 247), outline=RGB[border], width=5)
    try:
        font = ImageFont.truetype("DejaVuSans-Bold.ttf", 18)
    except OSError:
        font = ImageFont.load_default()
    draw.text(((x0 + x1) / 2, y0 + 16), label, anchor="mm", fill=RGB["black"], font=font)
    pos = _positions(word, x0 + 8, x1 - 8, layout)
    row_y = {color: y0 + 48 + i * ((y1 - y0 - 66) / 3) for i, color in enumerate(rows)}
    for color in IDENTITIES:
        xa, xb = pos[f"{color}:S"], pos[f"{color}:E"]
        y = row_y[color]
        draw.line((xa, y, xb, y), fill=RGB[color], width=10)
        for x in (xa, xb):
            draw.ellipse((x - 8, y - 8, x + 8, y + 8), fill=RGB[color], outline=RGB["black"], width=2)


def render(scene: dict) -> Image.Image:
    reference = scene["reference_word"]
    options = scene["option_words"]
    layouts = scene["layouts"]
    rows = scene["row_orders"]
    borders = scene["border_colors"]
    image = Image.new("RGB", (WIDTH, HEIGHT), (242, 240, 234))
    draw = ImageDraw.Draw(image)
    _draw_card(draw, (145, 42, 755, 276), reference, rows[0], layouts[0], "black", "REFERENCE")
    boxes = [(35, 360, 285, 596), (325, 360, 575, 596), (615, 360, 865, 596)]
    for i, box in enumerate(boxes):
        _draw_card(draw, box, options[i], rows[i + 1], layouts[i + 1], borders[i], chr(65 + i))
    return image


def analytic_gold(scene: dict) -> str:
    reference = scene["reference_word"]
    options = scene["option_words"]
    borders = scene["border_colors"]
    matches = [i for i, word in enumerate(options) if word == reference]
    if len(matches) != 1:
        raise ValueError("scene does not contain exactly one match")
    return borders[matches[0]]


def margin(scene: dict) -> float:
    reference = scene["reference_word"]
    options = scene["option_words"]
    scores = sorted((_matches(reference, word) for word in options), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    if margin(scene) <= 0:
        return True
    boxes = [(153, 747), (43, 277), (333, 567), (623, 857)]
    words = [scene["reference_word"], *scene["option_words"]]
    for word, layout, (left, right) in zip(words, scene["layouts"], boxes):
        xs = sorted(_positions(word, left, right, layout).values())
        if min(b - a for a, b in zip(xs, xs[1:])) < 12:
            return True
    return False
