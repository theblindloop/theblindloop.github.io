"""Renderer and analytic arm for the box-blurred halftone panel world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


WIDTH = 512
HEIGHT = 512
MICRO_GRID = 24
BLOCK = 3
FRAME_WIDTH = 8

TEMPLATES: dict[str, tuple[str, ...]] = {
    "ORBIT": (
        "01111110",
        "11000011",
        "10000001",
        "10000001",
        "10000001",
        "10000001",
        "11000011",
        "01111110",
    ),
    "PRISM": (
        "00011000",
        "00111100",
        "01100110",
        "11000011",
        "11000011",
        "01100110",
        "00111100",
        "00011000",
    ),
}


def _hamming(left: tuple[str, ...], right: tuple[str, ...]) -> int:
    return sum(
        a != b
        for row_a, row_b in zip(left, right)
        for a, b in zip(row_a, row_b)
    )


TEMPLATE_DISTANCE = _hamming(TEMPLATES["ORBIT"], TEMPLATES["PRISM"])


PALETTES = (
    {
        "background": (248, 246, 240),
        "panel": (218, 226, 216),
        "dark": (53, 67, 79),
        "light": (203, 213, 203),
        "frame": (31, 61, 76),
    },
    {
        "background": (243, 248, 245),
        "panel": (224, 216, 210),
        "dark": (72, 52, 58),
        "light": (211, 201, 196),
        "frame": (81, 47, 58),
    },
    {
        "background": (247, 243, 249),
        "panel": (215, 219, 232),
        "dark": (49, 54, 86),
        "light": (202, 207, 222),
        "frame": (49, 52, 91),
    },
    {
        "background": (242, 246, 250),
        "panel": (226, 221, 204),
        "dark": (74, 65, 42),
        "light": (213, 208, 193),
        "frame": (71, 63, 39),
    },
)


def _panel_spec(index: int) -> tuple[int, int]:
    # The three sizes all leave a 24-pixel frame-to-raster margin and make the
    # decisive area medium-scale on the 512-pixel canvas.
    microcell = (7, 8, 9)[index % 3]
    return microcell, MICRO_GRID * microcell + 48


def _carrier_cells(symbol: str, carrier_seed: int) -> list[list[int]]:
    """Create a random-looking 24-by-24 carrier with a one-cell count signal.

    Each 3-by-3 block contains four dark microcells for a zero and five for a
    one. The two templates have equal ink, so total dark-cell count is not a
    label cue. The arrangement inside every block is independently shuffled.
    """

    rng = random.Random(int(carrier_seed))
    template = TEMPLATES[symbol]
    cells = [[0 for _ in range(MICRO_GRID)] for _ in range(MICRO_GRID)]
    for block_y in range(8):
        for block_x in range(8):
            wanted = 5 if template[block_y][block_x] == "1" else 4
            positions = list(range(9))
            rng.shuffle(positions)
            for position in positions[:wanted]:
                local_y, local_x = divmod(position, 3)
                cells[block_y * 3 + local_y][block_x * 3 + local_x] = 1
    return cells


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, deterministic scene from one integer seed."""

    seed_value = int(seed)
    layout_rng = random.Random(seed_value ^ 0x3D2A7B)
    nuisance_rng = random.Random(seed_value ^ 0x71C4E9)
    palette = PALETTES[nuisance_rng.randrange(len(PALETTES))]
    microcell, panel_size = _panel_spec(nuisance_rng.randrange(3))
    panel_x = layout_rng.randint(70, WIDTH - panel_size - 70)
    panel_y = layout_rng.randint(70, HEIGHT - panel_size - 70)
    symbol = "ORBIT" if seed_value % 2 == 0 else "PRISM"

    remote_marks: list[dict[str, Any]] = []
    for index in range(7):
        for _ in range(100):
            x = nuisance_rng.uniform(24.0, WIDTH - 24.0)
            y = nuisance_rng.uniform(24.0, HEIGHT - 24.0)
            if not (
                panel_x - 34.0 <= x <= panel_x + panel_size + 34.0
                and panel_y - 34.0 <= y <= panel_y + panel_size + 34.0
            ):
                break
        if index % 2 == 0:
            remote_marks.append(
                {
                    "kind": "dot",
                    "x": round(x, 3),
                    "y": round(y, 3),
                    "radius": round(2.5 + 0.7 * (index % 3), 3),
                }
            )
        else:
            remote_marks.append(
                {
                    "kind": "line",
                    "x": round(x, 3),
                    "y": round(y, 3),
                    "dx": round(nuisance_rng.uniform(-22.0, 22.0), 3),
                    "dy": round(nuisance_rng.uniform(-22.0, 22.0), 3),
                }
            )

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "panel_x": panel_x,
        "panel_y": panel_y,
        "microcell": microcell,
        "panel_margin": 24,
        "symbol": symbol,
        "panel_fill": list(palette["panel"]),
        "tone_dark": list(palette["dark"]),
        "tone_light": list(palette["light"]),
        "frame_color": list(palette["frame"]),
        "background": list(palette["background"]),
        "carrier_seed": nuisance_rng.randrange(1, 2**31 - 1),
        "remote_marks": remote_marks,
    }


def _draw_remote_marks(draw: ImageDraw.ImageDraw, scene: dict[str, Any]) -> None:
    remote = (157, 164, 171)
    for mark in scene["remote_marks"]:
        if mark["kind"] == "dot":
            x = float(mark["x"])
            y = float(mark["y"])
            radius = float(mark["radius"])
            draw.ellipse(
                (x - radius, y - radius, x + radius, y + radius),
                fill=remote,
            )
        else:
            x = float(mark["x"])
            y = float(mark["y"])
            draw.line(
                (x, y, x + float(mark["dx"]), y + float(mark["dy"])),
                fill=remote,
                width=3,
            )


def _draw_panel(draw: ImageDraw.ImageDraw, scene: dict[str, Any], symbol: str) -> None:
    x0 = int(scene["panel_x"])
    y0 = int(scene["panel_y"])
    microcell = int(scene["microcell"])
    panel_size = MICRO_GRID * microcell + 2 * int(scene["panel_margin"])
    x1 = x0 + panel_size - 1
    y1 = y0 + panel_size - 1
    panel_fill = tuple(int(value) for value in scene["panel_fill"])
    frame_color = tuple(int(value) for value in scene["frame_color"])
    tone_dark = tuple(int(value) for value in scene["tone_dark"])
    tone_light = tuple(int(value) for value in scene["tone_light"])

    draw.rectangle((x0, y0, x1, y1), fill=panel_fill)
    draw.rectangle(
        (x0, y0, x1, y1),
        outline=frame_color,
        width=FRAME_WIDTH,
    )

    cells = _carrier_cells(symbol, int(scene["carrier_seed"]))
    inner_x = x0 + int(scene["panel_margin"])
    inner_y = y0 + int(scene["panel_margin"])
    for row in range(MICRO_GRID):
        for col in range(MICRO_GRID):
            left = inner_x + col * microcell
            top = inner_y + row * microcell
            tone = tone_dark if cells[row][col] else tone_light
            draw.rectangle(
                (left, top, left + microcell - 1, top + microcell - 1),
                fill=tone,
            )


def render(scene: dict[str, Any]) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = tuple(int(value) for value in scene["background"])
    image = Image.new("RGB", (width, height), background)
    draw = ImageDraw.Draw(image)
    _draw_remote_marks(draw, scene)
    symbol = str(scene["symbol"])
    _draw_panel(draw, scene, symbol)
    return image


def analytic_gold(scene: dict[str, Any]) -> str:
    symbol = str(scene["symbol"])
    if symbol not in TEMPLATES:
        raise ValueError(symbol)
    return symbol


def margin(scene: dict[str, Any]) -> float:
    # The renderer writes the exact selected template, so the visible score
    # gap is the fixed Hamming distance between the two displayed templates.
    analytic_gold(scene)
    return float(TEMPLATE_DISTANCE)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 8.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    return []
