"""Pixel-only inverse arm for the median orientation of a five-leg chain."""

from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image


def _centerline_mask(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = rgb[:, :, 0], rgb[:, :, 1], rgb[:, :, 2]
    # Dark blue centerline palettes satisfy this; pale cyan halo and neutral
    # ticks do not. No renderer constants or scene values enter this inverse.
    return (blue > red + 55) & (blue > green + 28) & (blue < 170) & (red < 70)


def _component(mask: np.ndarray) -> set[tuple[int, int]]:
    height, width = mask.shape
    pixels = {(int(x), int(y)) for y, x in np.argwhere(mask)}
    if not pixels:
        raise ValueError("measurement centerline is absent")
    start = next(iter(pixels))
    seen = {start}
    queue = deque([start])
    while queue:
        x, y = queue.popleft()
        for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if neighbor in pixels and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    if seen != pixels:
        raise ValueError("centerline is split or contains extra blue components")
    if not (110 <= len(seen) <= 280):
        raise ValueError("centerline has implausible total length")
    return seen


def _recover_legs(pixels: set[tuple[int, int]]) -> list[tuple[str, int]]:
    def neighbors(point: tuple[int, int]) -> list[tuple[int, int]]:
        x, y = point
        return [q for q in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)) if q in pixels]

    endpoints = [point for point in pixels if len(neighbors(point)) == 1]
    if len(endpoints) != 2 or any(len(neighbors(point)) not in (1, 2) for point in pixels):
        raise ValueError("centerline is not one simple unbranched path")
    ordered = [endpoints[0]]
    previous = None
    current = endpoints[0]
    while current != endpoints[1]:
        options = [point for point in neighbors(current) if point != previous]
        if len(options) != 1:
            raise ValueError("centerline traversal is ambiguous")
        previous, current = current, options[0]
        ordered.append(current)
        if len(ordered) > len(pixels):
            raise ValueError("centerline contains a cycle")
    if len(ordered) != len(pixels):
        raise ValueError("centerline traversal missed pixels")

    legs: list[tuple[str, int]] = []
    last_direction = None
    run = 0
    for (x0, y0), (x1, y1) in zip(ordered, ordered[1:]):
        direction = "horizontal" if y0 == y1 else "vertical"
        if last_direction is None or direction == last_direction:
            run += 1
        else:
            legs.append((last_direction, run))
            run = 1
        last_direction = direction
    if last_direction is not None:
        legs.append((last_direction, run))
    if len(legs) != 5:
        raise ValueError("expected exactly five visible straight legs")
    lengths = [length for _, length in legs]
    if min(lengths) < 15 or len(set(lengths)) != 5:
        raise ValueError("leg lengths are missing, tied, or implausibly short")
    return legs


def decision_from_image(image: Image.Image) -> str:
    legs = _recover_legs(_component(_centerline_mask(image)))
    ranked = sorted(legs, key=lambda item: item[1])
    median_orientation, median_length = ranked[2]
    cross_gap = min(abs(length - median_length) for orientation, length in ranked if orientation != median_orientation)
    if cross_gap < 6:
        raise ValueError("pixel measurement is too close to the rank boundary")
    return median_orientation
