"""Deterministic renderer for a compact five-leg orthogonal measurement chain."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw

CANVAS = 768
PALETTES = [
    ((19, 58, 112), (150, 216, 228), (250, 248, 242)),
    ((22, 64, 128), (154, 222, 215), (248, 250, 247)),
    ((31, 68, 122), (166, 211, 230), (251, 249, 245)),
]


def _legs(points: list[list[int]]) -> list[tuple[str, int]]:
    legs = []
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        dx, dy = x1 - x0, y1 - y0
        if (dx == 0) == (dy == 0):
            raise ValueError("every leg must be nonzero and orthogonal")
        legs.append(("horizontal" if dy == 0 else "vertical", abs(dx) + abs(dy)))
    if len(legs) != 5:
        raise ValueError("the chain must have exactly five legs")
    return legs


def _rotate(points: list[list[int]]) -> list[list[int]]:
    """Rotate 90 degrees in local coordinates; translation is added afterward."""
    return [[-y, x] for x, y in points]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    desired = "horizontal" if int(seed) % 2 == 0 else "vertical"
    rotate = bool(rng.randrange(2))
    base_desired = desired if not rotate else ("vertical" if desired == "horizontal" else "horizontal")

    lengths = [16, 23, 30, 37, 44]
    # Base orientations alternate H,V,H,V,H. Put the median value 30 in a
    # randomly chosen slot of the required orientation, then permute nuisances.
    base_orientations = ["horizontal", "vertical", "horizontal", "vertical", "horizontal"]
    target_slots = [i for i, orientation in enumerate(base_orientations) if orientation == base_desired]
    target_slot = rng.choice(target_slots)
    remaining = [16, 23, 37, 44]
    rng.shuffle(remaining)
    assigned = []
    for index in range(5):
        assigned.append(30 if index == target_slot else remaining.pop())

    # A folded orthogonal chain remains elongated along its three horizontal
    # legs. Randomized rotation and length permutation prevent global extent in
    # either axis from identifying the median leg's orientation.
    points = [[0, 0]]
    vertical_sign = rng.choice([-1, 1])
    for index, length in enumerate(assigned):
        x, y = points[-1]
        if index % 2 == 0:
            points.append([x + length, y])
        else:
            points.append([x, y + vertical_sign * length])
            vertical_sign *= -1
    if rotate:
        points = _rotate(points)

    min_x = min(p[0] for p in points)
    max_x = max(p[0] for p in points)
    min_y = min(p[1] for p in points)
    max_y = max(p[1] for p in points)
    pad = 15
    x_span, y_span = max_x - min_x, max_y - min_y
    # Keep the short axis inside the center half of a randomly selected 96px
    # measurement band. This preserves translation variation while ensuring the
    # decisive ablation footprint is a thin connected strip.
    band_center = rng.randint(1, 6) * 96 + 48
    if x_span >= y_span:
        tx = rng.randint(96 - min_x + pad, CANVAS - 96 - max_x - pad)
        ty = band_center - round((min_y + max_y) / 2)
    else:
        tx = band_center - round((min_x + max_x) / 2)
        ty = rng.randint(96 - min_y + pad, CANVAS - 96 - max_y - pad)
    points = [[x + tx, y + ty] for x, y in points]

    chain_color, halo_color, background = PALETTES[rng.randrange(len(PALETTES))]
    distractors = []
    # Pale neutral calibration ticks are remote and nondiagnostic. Their order
    # is immaterial because they never overlap.
    chain_box = (min(p[0] for p in points) - 30, min(p[1] for p in points) - 30,
                 max(p[0] for p in points) + 30, max(p[1] for p in points) + 30)
    for band_x in (58, CANVAS - 62):
        for offset in range(3):
            y = 150 + offset * 210 + rng.randint(-24, 24)
            if not (chain_box[0] <= band_x <= chain_box[2] and chain_box[1] <= y <= chain_box[3]):
                distractors.append([band_x, y, rng.choice([10, 14, 18])])
    return {
        "width": CANVAS,
        "height": CANVAS,
        "background": list(background),
        "chain_color": list(chain_color),
        "halo_color": list(halo_color),
        "stroke": rng.choice([7, 9]),
        "points": points,
        "distractors": distractors,
    }


def _draw_axis_chain(draw: ImageDraw.ImageDraw, points: list[list[int]], color: tuple[int, int, int], width: int) -> None:
    radius = width // 2
    # The union of axis-aligned rectangles and joint squares is traversal-order
    # invariant, which makes waypoint reversal a real pixel-identical alias.
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        draw.rectangle((min(x0, x1) - radius, min(y0, y1) - radius,
                        max(x0, x1) + radius, max(y0, y1) + radius), fill=color)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    for x, y, length in scene["distractors"]:
        neutral = (210, 211, 207)
        draw.line((x - length, y, x + length, y), fill=neutral, width=2)
    points = [[int(x), int(y)] for x, y in scene["points"]]
    _draw_axis_chain(draw, points, tuple(scene["halo_color"]), int(scene["stroke"]))
    # A one-pixel centerline is itself the connected measurement path. It makes
    # turn centers and leg lengths exactly visible without latent coordinates.
    for (x0, y0), (x1, y1) in zip(points, points[1:]):
        draw.line((x0, y0, x1, y1), fill=tuple(scene["chain_color"]), width=1)
    return image


def analytic_gold(scene: dict) -> str:
    ranked = sorted(_legs(scene["points"]), key=lambda item: item[1])
    return ranked[2][0]


def margin(scene: dict) -> float:
    ranked = sorted(_legs(scene["points"]), key=lambda item: item[1])
    median_orientation, median_length = ranked[2]
    cross_gaps = [abs(length - median_length) for orientation, length in ranked if orientation != median_orientation]
    return float(min(cross_gaps))


def is_quarantined(scene: dict) -> bool:
    lengths = [length for _, length in _legs(scene["points"])]
    return len(set(lengths)) != 5 or margin(scene) < 6.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    reversed_scene = copy.deepcopy(scene)
    reversed_scene["points"] = list(reversed(reversed_scene["points"]))
    return [("reverse_chain_traversal", reversed_scene)]
