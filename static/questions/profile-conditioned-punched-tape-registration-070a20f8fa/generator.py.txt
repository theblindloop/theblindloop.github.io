"""Deterministic punched-tape registration world."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 900, 540
REFERENCE_LEN, TARGET_LEN, PUNCH_ROWS = 5, 12, 4
CELL_W, CELL_H = 58, 38
BACKGROUND = (248, 249, 252)
INK = (28, 32, 40)
MUTED = (103, 110, 123)
TAPE = (244, 235, 201)
MAGENTA = (207, 34, 132)
TAG_RGB = {
    "red": (218, 57, 58),
    "green": (43, 151, 82),
    "blue": (48, 101, 211),
}


def _font(size: int, bold: bool = False) -> ImageFont.ImageFont:
    path = (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
        if bold
        else "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"
    )
    try:
        return ImageFont.truetype(path, size)
    except OSError:
        return ImageFont.load_default()


def _hamming(reference: Iterable[int], window: Iterable[int]) -> int:
    return sum((int(a) ^ int(b)).bit_count() for a, b in zip(reference, window))


def _alignment_costs(scene: dict) -> list[int]:
    reference = scene["reference_codes"]
    target = scene["target_codes"]
    return [
        _hamming(reference, target[start : start + len(reference)])
        for start in range(len(target) - len(reference) + 1)
    ]


def _validate_scene(scene: dict) -> None:
    reference = scene["reference_codes"]
    target = scene["target_codes"]
    offset = scene["offset"]
    marker = scene["marker_index"]
    tags = scene["tag_colors"]
    if len(reference) != REFERENCE_LEN or len(target) != TARGET_LEN:
        raise ValueError("unexpected tape length")
    if len(tags) != TARGET_LEN or any(c not in TAG_RGB for c in tags):
        raise ValueError("invalid tag colors")
    if not 0 <= offset <= TARGET_LEN - REFERENCE_LEN:
        raise ValueError("invalid alignment offset")
    if not 0 <= marker < REFERENCE_LEN:
        raise ValueError("invalid marker index")
    if list(target[offset : offset + REFERENCE_LEN]) != list(reference):
        raise ValueError("declared offset does not contain reference")
    if any(not isinstance(code, int) or not 0 <= code < 2**PUNCH_ROWS for code in reference + target):
        raise ValueError("punch code outside four-bit range")


def _make_candidate(rng: random.Random, seed: int) -> dict:
    # A minority of scenes deliberately have a one-cell runner-up gap so the
    # submitted evidence contains declared near-boundary/quarantined examples.
    requested_gap = 1 if seed % 11 == 0 else (2 if seed % 5 == 0 else None)
    reference = [rng.randrange(16) for _ in range(REFERENCE_LEN)]
    target = [rng.randrange(16) for _ in range(TARGET_LEN)]
    if requested_gap is None:
        offset = rng.randrange(TARGET_LEN - REFERENCE_LEN + 1)
        target[offset : offset + REFERENCE_LEN] = reference
    else:
        offset = 0 if rng.randrange(2) == 0 else TARGET_LEN - REFERENCE_LEN
        alternative = TARGET_LEN - REFERENCE_LEN if offset == 0 else 0
        target[offset : offset + REFERENCE_LEN] = reference
        near = list(reference)
        order = list(range(REFERENCE_LEN))
        rng.shuffle(order)
        for bit_index in range(requested_gap):
            column = order[bit_index]
            near[column] ^= 1 << rng.randrange(PUNCH_ROWS)
        target[alternative : alternative + REFERENCE_LEN] = near

    costs = [
        _hamming(reference, target[start : start + REFERENCE_LEN])
        for start in range(TARGET_LEN - REFERENCE_LEN + 1)
    ]
    ranked = sorted(costs)
    if costs[offset] != 0 or ranked.count(0) != 1:
        raise ValueError("non-unique exact registration")
    if requested_gap is None and ranked[1] < 4:
        raise ValueError("ordinary runner-up too close")
    if requested_gap is not None and ranked[1] != requested_gap:
        raise ValueError("near-boundary construction was masked by another alignment")

    marker_index = rng.randrange(REFERENCE_LEN)
    desired = ("red", "green", "blue")[seed % 3]
    tag_colors = ["red"] * 4 + ["green"] * 4 + ["blue"] * 4
    rng.shuffle(tag_colors)
    answer_index = offset + marker_index
    donor = next(i for i, color in enumerate(tag_colors) if color == desired)
    tag_colors[answer_index], tag_colors[donor] = tag_colors[donor], tag_colors[answer_index]
    return {
        "reference_codes": reference,
        "target_codes": target,
        "offset": offset,
        "marker_index": marker_index,
        "tag_colors": tag_colors,
    }


def sample_scene(seed: int) -> dict:
    """Sample a balanced, uniquely registrable scene from one integer seed."""
    rng = random.Random(int(seed))
    for _ in range(20_000):
        try:
            return _make_candidate(rng, int(seed))
        except ValueError:
            continue
    raise RuntimeError("could not sample a well-separated registration")


def analytic_gold(scene: dict) -> str:
    """Transport the marked column through the declared, visibly checked match."""
    _validate_scene(scene)
    reference = scene["reference_codes"]
    target = scene["target_codes"]
    offset = scene["offset"]
    marker = scene["marker_index"]
    tags = scene["tag_colors"]
    if list(target[offset : offset + len(reference)]) != list(reference):
        raise ValueError("reference does not occur at declared offset")
    return str(tags[offset + marker])


def margin(scene: dict) -> float:
    """Runner-up minus best alignment cost, measured in punch-cell mismatches."""
    costs = sorted(_alignment_costs(scene))
    return float(costs[1] - costs[0]) if len(costs) > 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    """One punch-cell separates the best and runner-up: explicit quarantine."""
    return margin(scene) < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """The declared scene is injective; there are no nuisance-only fields."""
    return []


def _draw_tape(draw: ImageDraw.ImageDraw, codes: list[int], x0: int, y0: int) -> None:
    width = len(codes) * CELL_W
    height = PUNCH_ROWS * CELL_H
    draw.rectangle((x0, y0, x0 + width, y0 + height), fill=TAPE, outline=INK, width=4)
    for column in range(1, len(codes)):
        x = x0 + column * CELL_W
        draw.line((x, y0, x, y0 + height), fill=INK, width=3)
    for row in range(1, PUNCH_ROWS):
        y = y0 + row * CELL_H
        draw.line((x0, y, x0 + width, y), fill=INK, width=3)
    for column, code in enumerate(codes):
        for row in range(PUNCH_ROWS):
            if code & (1 << row):
                cx = x0 + column * CELL_W + CELL_W // 2
                cy = y0 + row * CELL_H + CELL_H // 2
                draw.ellipse((cx - 10, cy - 10, cx + 10, cy + 10), fill=INK)


def render(scene: dict) -> Image.Image:
    """Render the complete question world as a single RGB raster."""
    _validate_scene(scene)
    reference = list(scene["reference_codes"])
    target = list(scene["target_codes"])
    marker = int(scene["marker_index"])
    # These reads are intentional: all label inputs are also visible inputs.
    _ = int(scene["offset"])
    tags = list(scene["tag_colors"])

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    ref_x, ref_y = (WIDTH - REFERENCE_LEN * CELL_W) // 2, 69
    target_x, target_y = (WIDTH - TARGET_LEN * CELL_W) // 2, 300

    draw.text((WIDTH // 2, 19), "REFERENCE TAPE", fill=INK, anchor="ma", font=_font(22, True))
    draw.text(
        (WIDTH // 2, 255),
        "TARGET TAPE — FIND THE SAME FIVE-COLUMN RUN",
        fill=INK,
        anchor="ma",
        font=_font(20, True),
    )
    _draw_tape(draw, reference, ref_x, ref_y)
    _draw_tape(draw, target, target_x, target_y)

    left = ref_x + marker * CELL_W
    right = left + CELL_W
    draw.rectangle((left + 2, ref_y + 2, right - 2, ref_y + PUNCH_ROWS * CELL_H - 2), outline=MAGENTA, width=5)
    cx = (left + right) // 2
    draw.polygon(((cx, ref_y - 5), (cx - 12, ref_y - 22), (cx + 12, ref_y - 22)), fill=MAGENTA)

    tape_bottom = target_y + PUNCH_ROWS * CELL_H
    for column, color in enumerate(tags):
        cx = target_x + column * CELL_W + CELL_W // 2
        cy = tape_bottom + 40
        draw.line((cx, tape_bottom + 2, cx, cy - 17), fill=MUTED, width=3)
        draw.ellipse((cx - 16, cy - 16, cx + 16, cy + 16), fill=TAG_RGB[color], outline=INK, width=2)

    draw.text(
        (WIDTH // 2, 512),
        "Match punch positions left-to-right; do not reverse the sequence.",
        fill=MUTED,
        anchor="mm",
        font=_font(17),
    )
    return image
