"""Renderer, sampler and analytic gold for macro_expansion_walk.

A track of seven coloured cells carries a marker that starts on the ringed cell.
A single programme row of badge circles is executed left to right; each badge
does not encode a move by itself, it names the legend row drawn with the same
badge colour, and that row's dark triangles are the moves. The same legend row
may be referenced several times, so the answer requires dereferencing a drawn
definition and replaying it, not reading a drawn move sequence.

Everything the answer depends on is drawn: the ring, the badge colours, the
triangle directions, the programme order and the cell colours.
"""

from __future__ import annotations

import random
from collections import Counter

from PIL import Image, ImageDraw

RENDERER_VERSION = "macro_expansion_walk-renderer-0.1.0"

W, H = 920, 560
BG = (250, 247, 240)
ARROW_RGB = (55, 60, 70)
RING_RGB = (20, 20, 20)

# Named track palette. Only these eight names can ever be an answer.
PALETTE: tuple[tuple[str, tuple[int, int, int]], ...] = (
    ("red", (214, 45, 40)),
    ("orange", (235, 130, 25)),
    ("yellow", (240, 205, 40)),
    ("green", (40, 150, 70)),
    ("blue", (40, 95, 200)),
    ("purple", (130, 60, 175)),
    ("brown", (140, 85, 45)),
    ("teal", (20, 150, 150)),
)
COLOR_NAMES: tuple[str, ...] = tuple(name for name, _ in PALETTE)

# Badge colours are disjoint from the track palette and are never named.
BADGE_RGB: tuple[tuple[int, int, int], ...] = (
    (200, 30, 140),
    (0, 160, 205),
    (125, 120, 35),
)

N_CELLS = 7
CELL = 66
RADIUS = 22
ARROW_H = 22  # half height


def _dist(a, b) -> float:
    return sum((int(x) - int(y)) ** 2 for x, y in zip(a, b)) ** 0.5


def name_of_rgb(rgb) -> str:
    """Nearest named palette colour. Analytic gold reads only drawn colour."""
    return min(PALETTE, key=lambda p: _dist(p[1], rgb))[0]


def _name_margin(rgb) -> float:
    """How far the drawn colour is from being renamed, in palette units."""
    ds = sorted(_dist(c, rgb) for _, c in PALETTE)
    return (ds[1] - ds[0]) / 10.0


# --------------------------------------------------------------------------
# sampling


def _walk(start: int, cards: dict[int, list[int]], program: list[int]):
    """Return (positions visited, final) or None when the marker leaves."""
    pos = start
    seen = [pos]
    for badge in program:
        for step in cards[badge]:
            pos += step
            if pos < 0 or pos >= N_CELLS:
                return None
            seen.append(pos)
    return seen, pos


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    target_color = seed % len(PALETTE)
    want_right = (((seed // len(PALETTE)) + seed) % 2) == 0
    tight = (seed % 5) == 4

    for _ in range(4000):
        start = rng.randrange(N_CELLS)
        cards = {b: [rng.choice((1, -1)) for _ in range(rng.randint(1, 3))] for b in range(3)}
        n_prog = rng.choice((3, 4))
        program = [rng.randrange(3) for _ in range(n_prog)]

        used = Counter(program)
        if len(used) < 2 or max(used.values()) < 2:
            continue  # a legend row must be reused: indirection has to bite
        if not any(len(cards[b]) >= 2 for b in used):
            continue
        walked = _walk(start, cards, program)
        if walked is None:
            continue
        _, final = walked
        if final == start:
            continue
        if (final > start) != want_right:
            continue

        # shortcut blocks: cheap readings must not land on the answer cell
        net = {b: sum(v) for b, v in cards.items()}
        naive_sign = start + sum((1 if net[b] > 0 else -1) if net[b] else 0 for b in program)
        naive_each_card_once = start + sum(net.values())
        naive_all_right = start + sum(len(cards[b]) for b in program)
        naive_first_arrow = start + sum(cards[b][0] for b in program)
        if final in (naive_sign, naive_each_card_once, naive_all_right, naive_first_arrow):
            continue

        # colours: the answer colour is fixed by the seed, so the answer name is
        # counterbalanced and independent of the landing position.
        others = [i for i in range(len(PALETTE)) if i != target_color]
        rng.shuffle(others)
        chosen = others[: N_CELLS - 1]
        idxs = [target_color] + chosen
        positions = [final] + [p for p in range(N_CELLS) if p != final]
        cells = []
        for pos, ci in zip(positions, idxs):
            cells.append({"pos": pos, "rgb": list(PALETTE[ci][1])})
        if tight:
            # explicit near-boundary mode: the landing cell is drawn 48% of the
            # way towards a rival palette colour, so its name is still well
            # defined but one more step would rename it.
            rival = PALETTE[chosen[0]][1]
            base = PALETTE[target_color][1]
            blend = [int(round(b + 0.48 * (r - b))) for b, r in zip(base, rival)]
            for cell in cells:
                if cell["pos"] == final:
                    cell["rgb"] = blend

        scene = {
            "seed": int(seed),
            "start": start,
            "cells": cells,
            "cards": [{"row": b, "badge": b, "arrows": cards[b]} for b in range(3)],
            "program": [{"pos": i, "badge": b} for i, b in enumerate(program)],
            "arrow_w": rng.choice((36, 40, 44)),
            "gap": rng.choice((21, 22, 24)),
            "row_jitter": [rng.randint(-3, 3) for _ in range(5)],
            "tight": tight,
        }
        # rotate the card bag so drawn row order is not list order
        k = rng.randrange(3)
        scene["cards"] = scene["cards"][k:] + scene["cards"][:k]
        return scene
    raise RuntimeError(f"sampling failed for seed {seed}")


# --------------------------------------------------------------------------
# gold and margin


def _cards_by_badge(scene: dict) -> dict[int, list[int]]:
    return {int(c["badge"]): [int(a) for a in c["arrows"]] for c in scene["cards"]}


def _cells_by_pos(scene: dict) -> list[list[int]]:
    ordered = sorted(scene["cells"], key=lambda c: int(c["pos"]))
    return [list(c["rgb"]) for c in ordered]


def _final_index(scene: dict) -> int:
    program = [int(p["badge"]) for p in sorted(scene["program"], key=lambda p: int(p["pos"]))]
    walked = _walk(int(scene["start"]), _cards_by_badge(scene), program)
    if walked is None:
        raise ValueError("scene walks off the drawn track")
    return walked[1]


def analytic_gold(scene: dict) -> str:
    final = _final_index(scene)
    rgbs = _cells_by_pos(scene)
    direction = "right" if final > int(scene["start"]) else "left"
    return f"{name_of_rgb(rgbs[final])}|{direction}"


def margin(scene: dict) -> float:
    rgbs = _cells_by_pos(scene)
    m_name = min(_name_margin(rgb) for rgb in rgbs)
    m_arrow = int(scene["arrow_w"]) / 5.0
    m_gap = int(scene["gap"]) / 3.0
    return round(min(m_name, m_arrow, m_gap), 4)


QUARANTINE_THRESHOLD = 6.0


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_THRESHOLD


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Card and programme bags are unordered: every drawn place is carried by a
    row/pos field inside the item itself, so re-ordering either list must be
    pixel-identical and answer-preserving."""
    out = []
    a = dict(scene)
    a["cards"] = list(reversed(scene["cards"]))
    out.append(("cards_bag_reversed", a))
    b = dict(scene)
    b["program"] = list(reversed(scene["program"]))
    out.append(("program_bag_reversed", b))
    c = dict(scene)
    c["cards"] = scene["cards"][1:] + scene["cards"][:1]
    c["program"] = scene["program"][2:] + scene["program"][:2]
    out.append(("both_bags_rotated", c))
    return out


# --------------------------------------------------------------------------
# rendering


def _triangle(draw, x0: int, yc: int, w: int, direction: int) -> None:
    if direction > 0:
        pts = [(x0, yc - ARROW_H), (x0, yc + ARROW_H), (x0 + w, yc)]
    else:
        pts = [(x0 + w, yc - ARROW_H), (x0 + w, yc + ARROW_H), (x0, yc)]
    draw.polygon(pts, fill=ARROW_RGB)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (W, H), BG)
    d = ImageDraw.Draw(img)
    jit = [int(v) for v in scene["row_jitter"]]
    arrow_w = int(scene["arrow_w"])

    # legend rows, drawn at the row index each card carries
    for card in sorted(scene["cards"], key=lambda c: int(c["row"])):
        r = int(card["row"])
        yc = 92 + r * 72 + jit[r]
        badge = BADGE_RGB[int(card["badge"])]
        d.ellipse([150 - RADIUS, yc - RADIUS, 150 + RADIUS, yc + RADIUS], fill=badge)
        for j, step in enumerate(card["arrows"]):
            x0 = 216 + j * (arrow_w + 22)
            _triangle(d, x0, yc, arrow_w, int(step))

    # programme row: badges only, no triangles
    prog = sorted(scene["program"], key=lambda p: int(p["pos"]))
    yc = 336 + jit[3]
    span = (len(prog) - 1) * 76
    x0 = 460 - span // 2
    for i, entry in enumerate(prog):
        cx = x0 + i * 76
        d.ellipse(
            [cx - RADIUS, yc - RADIUS, cx + RADIUS, yc + RADIUS],
            fill=BADGE_RGB[int(entry["badge"])],
        )

    # track
    gap = int(scene["gap"])
    total = N_CELLS * CELL + (N_CELLS - 1) * gap
    tx0 = (W - total) // 2
    ty = 428 + jit[4]
    for cell in sorted(scene["cells"], key=lambda c: int(c["pos"])):
        p = int(cell["pos"])
        left = tx0 + p * (CELL + gap)
        d.rectangle([left, ty, left + CELL - 1, ty + CELL - 1], fill=tuple(cell["rgb"]))
    sp = int(scene["start"])
    left = tx0 + sp * (CELL + gap)
    cx, cy = left + CELL // 2, ty + CELL // 2
    d.ellipse([cx - 24, cy - 24, cx + 24, cy + 24], outline=RING_RGB, width=5)
    return img
