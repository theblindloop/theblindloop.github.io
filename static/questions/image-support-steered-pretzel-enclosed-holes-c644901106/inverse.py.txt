"""Independent pixel-only inverse arm for visible pretzel holes.

This module deliberately duplicates only the public visual notion of brown ink.
It receives one raster and has no access to sampling state, source geometry, or
analytic labels.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

MIN_FOREGROUND_PIXELS = 4500
MIN_HOLE_AREA = 500
MIN_HOLE_EXTENT = 16
MIN_MAIN_COMPONENT_FRACTION = 0.985


def _brown_mask(image: Image.Image) -> np.ndarray:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    r, g, b = arr[..., 0], arr[..., 1], arr[..., 2]
    # Broad hue rule covers every declared baked-brown shade and its dark outline,
    # while excluding white background and pale salt grains.
    return (r > 60) & (r < 235) & (g > 20) & (g < 170) & (b < 115) & (r - g > 28) & (g - b > 7)


def _components(mask: np.ndarray) -> list[tuple[int, int, int, int, int]]:
    """Return (area, min_y, max_y, min_x, max_x) for 4-connected components."""
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[tuple[int, int, int, int, int]] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        seen[y0, x0] = True
        queue = deque([(y0, x0)])
        area = 0
        min_y = max_y = y0
        min_x = max_x = x0
        while queue:
            y, x = queue.popleft()
            area += 1
            min_y, max_y = min(min_y, y), max(max_y, y)
            min_x, max_x = min(min_x, x), max(max_x, x)
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append((area, min_y, max_y, min_x, max_x))
    return output


def decision_from_image(image: Image.Image) -> str:
    """Count large non-brown components enclosed by the connected brown rope."""
    foreground = _brown_mask(image)
    total = int(foreground.sum())
    if total < MIN_FOREGROUND_PIXELS:
        raise ValueError("brown rope is absent or materially erased")

    ys, xs = np.nonzero(foreground)
    if len(ys) == 0:
        raise ValueError("no foreground")
    y0, y1 = max(0, int(ys.min()) - 2), min(foreground.shape[0], int(ys.max()) + 3)
    x0, x1 = max(0, int(xs.min()) - 2), min(foreground.shape[1], int(xs.max()) + 3)
    crop = foreground[y0:y1, x0:x1]

    ink_components = _components(crop)
    largest = max((item[0] for item in ink_components), default=0)
    if largest / max(total, 1) < MIN_MAIN_COMPONENT_FRACTION:
        raise ValueError("brown rope is no longer one connected component")

    free_components = _components(~crop)
    height, width = crop.shape
    holes = []
    for area, min_y, max_y, min_x, max_x in free_components:
        touches_border = min_y == 0 or min_x == 0 or max_y == height - 1 or max_x == width - 1
        extent_y = max_y - min_y + 1
        extent_x = max_x - min_x + 1
        if not touches_border and area >= MIN_HOLE_AREA and min(extent_x, extent_y) >= MIN_HOLE_EXTENT:
            holes.append((area, extent_y, extent_x))
    count = len(holes)
    if count not in (2, 3, 4):
        raise ValueError(f"visible enclosed-region count {count} is outside the public answer set")
    return str(count)
