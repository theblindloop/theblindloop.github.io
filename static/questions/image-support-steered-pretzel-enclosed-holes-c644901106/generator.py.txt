"""Deterministic renderer and analytic arm for the pretzel-hole world."""

from __future__ import annotations

import itertools
import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SUPERSAMPLE = 4
RING_RADIUS = 42.0
RING_CENTERS_X = (136.0, 216.0, 296.0, 376.0)
RING_CENTER_Y = 236.0
BASELINE_Y = 288.0
BASELINE_X = (88.0, 424.0)
OUTLINE_EXTRA = 8
QUARANTINE_CLEARANCE = 6.0

FILL_COLORS = (
    (184, 103, 42),
    (169, 87, 34),
    (197, 119, 50),
    (177, 95, 38),
)
OUTLINE_COLORS = (
    (92, 45, 22),
    (83, 38, 18),
    (101, 50, 24),
    (88, 41, 20),
)


def sample_scene(seed: int) -> dict:
    """Sample answer-balanced scenes while randomizing every cheap style cue."""
    rng = random.Random(int(seed))
    return {
        "hole_count": 2 + (int(seed) % 3),
        "pattern_index": rng.randrange(0, 100_000),
        "offset_x": rng.randint(-12, 12),
        "offset_y": rng.randint(-12, 12),
        "rope_width": rng.choice((18, 19, 20, 21, 22)),
        "shade_index": rng.randrange(len(FILL_COLORS)),
        "salt_seed": rng.randrange(0, 1_000_000),
        "gap_degrees": rng.choice((54, 62, 70)),
    }


def _closed_slots(scene: dict) -> tuple[int, ...]:
    count = int(scene["hole_count"])
    patterns = tuple(itertools.combinations(range(4), count))
    return patterns[int(scene["pattern_index"]) % len(patterns)]


def _scaled_points(points: list[tuple[float, float]], ss: int) -> list[tuple[int, int]]:
    return [(round(x * ss), round(y * ss)) for x, y in points]


def _ring_points(
    cx: float, cy: float, *, closed: bool, gap_degrees: float
) -> list[tuple[float, float]]:
    if closed:
        angles = [(-90.0 + 360.0 * i / 180.0) for i in range(181)]
    else:
        start = -90.0 + gap_degrees / 2.0
        sweep = 360.0 - gap_degrees
        angles = [(start + sweep * i / 170.0) for i in range(171)]
    return [
        (
            cx + RING_RADIUS * math.cos(math.radians(angle)),
            cy + RING_RADIUS * math.sin(math.radians(angle)),
        )
        for angle in angles
    ]


def _paths(scene: dict) -> list[list[tuple[float, float]]]:
    ox = float(scene["offset_x"])
    oy = float(scene["offset_y"])
    gap = float(scene["gap_degrees"])
    closed = set(_closed_slots(scene))
    paths: list[list[tuple[float, float]]] = [
        [(BASELINE_X[0] + ox, BASELINE_Y + oy), (BASELINE_X[1] + ox, BASELINE_Y + oy)]
    ]
    horn_length = RING_RADIUS * math.radians(gap) / 2.0
    for slot, base_x in enumerate(RING_CENTERS_X):
        cx, cy = base_x + ox, RING_CENTER_Y + oy
        ring = _ring_points(cx, cy, closed=slot in closed, gap_degrees=gap)
        paths.append(ring)
        if slot not in closed:
            # Replace the missing top arc with two nonjoining outward horns.  This
            # nearly equalizes total ink without manufacturing another pocket.
            left, right = ring[-1], ring[0]
            paths.append([left, (left[0] - horn_length * 0.72, left[1] - horn_length * 0.70)])
            paths.append([right, (right[0] + horn_length * 0.72, right[1] - horn_length * 0.70)])
    return paths


def render(scene: dict) -> Image.Image:
    """Render only visible geometry; analytic labels are never embedded."""
    # Explicit reads keep the public rendered-field audit honest.
    width = int(scene["rope_width"])
    shade = int(scene["shade_index"]) % len(FILL_COLORS)
    salt_seed = int(scene["salt_seed"])
    _ = int(scene["hole_count"]), int(scene["pattern_index"]), float(scene["gap_degrees"])
    _ = int(scene["offset_x"]), int(scene["offset_y"])

    ss = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    paths = _paths(scene)
    for path in paths:
        draw.line(
            _scaled_points(path, ss),
            fill=OUTLINE_COLORS[shade],
            width=(width + OUTLINE_EXTRA) * ss,
            joint="curve",
        )
    for path in paths:
        draw.line(
            _scaled_points(path, ss),
            fill=FILL_COLORS[shade],
            width=width * ss,
            joint="curve",
        )

    # Small salt grains are answer-independent texture.  They are far below the
    # oracle's minimum enclosed-region area and never span the rope width.
    rng = random.Random(salt_seed)
    ox, oy = int(scene["offset_x"]), int(scene["offset_y"])
    for _index in range(14):
        x = rng.randint(102 + ox, 410 + ox)
        y = BASELINE_Y + oy + rng.randint(-5, 5)
        radius = rng.choice((2, 2, 3)) * ss
        draw.ellipse(
            [x * ss - radius, y * ss - radius, x * ss + radius, y * ss + radius],
            fill=(244, 218, 151),
        )

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["hole_count"]))


def margin(scene: dict) -> float:
    """Smallest pixel clearance to a topology-changing visual ambiguity."""
    width = float(scene["rope_width"])
    outer_width = width + OUTLINE_EXTRA
    open_gap = 2.0 * RING_RADIUS * math.sin(math.radians(float(scene["gap_degrees"])) / 2.0)
    open_gap -= outer_width
    pocket_diameter = 2.0 * (RING_RADIUS - outer_width / 2.0)
    connector_overlap = outer_width - abs(BASELINE_Y - (RING_CENTER_Y + RING_RADIUS))
    if int(scene["hole_count"]) == 4:
        open_gap = pocket_diameter
    return round(min(open_gap, pocket_diameter, connector_overlap), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_CLEARANCE


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    patterns = tuple(itertools.combinations(range(4), int(scene["hole_count"])))
    twin = dict(scene)
    twin["pattern_index"] = int(scene["pattern_index"]) + len(patterns)
    return [("cyclic_closed_slot_index", twin)]
