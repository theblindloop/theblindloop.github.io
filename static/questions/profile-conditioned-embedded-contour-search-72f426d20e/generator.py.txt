"""Deterministic latent sampler and raster renderer for exact-contour search."""

from __future__ import annotations

import copy
import itertools
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 600
BACKGROUND = (249, 248, 244)
NAVY = (23, 74, 116)
ORANGE = (224, 110, 35)
CLUTTER = (194, 199, 202)
PANEL = (119, 128, 134)
TEXT = (43, 48, 52)
OUTER_RADIUS = 47
CARDINAL_RADIUS = 52
INNER_RADII = (34, 38, 41)
PANEL_CENTERS = ((200, 292), (560, 292), (200, 492), (560, 492))
LABELS = ("A", "B", "C", "D")
PATTERNS = tuple(tuple(1 if i in pair else 0 for i in range(4)) for pair in itertools.combinations(range(4), 2))


def _font(size: int, bold: bool = False):
    try:
        name = "DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf"
        return ImageFont.truetype(name, size)
    except OSError:
        return ImageFont.load_default()


def _contour_points(code, inner_radius: int, center: tuple[int, int]):
    """Eight clockwise vertices: four fixed cardinals and four coded diagonals."""
    cx, cy = center
    points = []
    diagonal_index = 0
    for step in range(8):
        angle = -math.pi / 2 + step * math.pi / 4
        if step % 2 == 0:
            radius = CARDINAL_RADIUS
        else:
            radius = OUTER_RADIUS if int(code[diagonal_index]) else inner_radius
            diagonal_index += 1
        points.append((round(cx + radius * math.cos(angle)), round(cy + radius * math.sin(angle))))
    return points


def _make_clutter(rng: random.Random):
    """One scene-level stroke template translated identically into all panels."""
    vectors = ((38, 0), (0, 38), (28, 28), (28, -28), (44, 0), (0, 44), (32, 32))
    anchors = ((-112, -55), (98, -52), (-108, 48), (96, 50), (-88, 0), (84, 2), (0, 66))
    order = list(range(len(vectors)))
    rng.shuffle(order)
    template = []
    for slot, vector_index in enumerate(order):
        ax, ay = anchors[slot]
        dx, dy = vectors[vector_index]
        # A small shared jitter changes scenes but is repeated in every panel.
        ax += rng.randint(-5, 5)
        ay += rng.randint(-4, 4)
        template.append((ax - dx // 2, ay - dy // 2, ax + dx // 2, ay + dy // 2))
    panels = []
    for cx, cy in PANEL_CENTERS:
        panels.append([[cx + x1, cy + y1, cx + x2, cy + y2] for x1, y1, x2, y2 in template])
    return panels


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5EEDC0DE)
    reference = list(PATTERNS[rng.randrange(len(PATTERNS))])
    distractors = [list(code) for code in PATTERNS if list(code) != reference]
    rng.shuffle(distractors)
    target_index = int(seed) % 4
    candidates = [None, None, None, None]
    candidates[target_index] = reference[:]
    for index, code in zip((i for i in range(4) if i != target_index), distractors[:3], strict=True):
        candidates[index] = code
    # Four consecutive seeds expose all answer positions at the same difficulty.
    inner_radius = INNER_RADII[(int(seed) // 4) % len(INNER_RADII)]
    offsets = [[rng.randint(-6, 6), rng.randint(-5, 5)] for _ in range(4)]
    return {
        "reference_code": reference,
        "candidate_codes": candidates,
        "inner_radius": inner_radius,
        "contour_offsets": offsets,
        "clutter_segments": _make_clutter(rng),
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    label_font = _font(22, bold=True)
    caption_font = _font(16, bold=True)

    draw.rounded_rectangle((300, 14, 460, 180), radius=12, outline=PANEL, width=2, fill=(255, 254, 251))
    draw.text((327, 24), "REFERENCE", fill=TEXT, font=caption_font)
    ref_points = _contour_points(scene["reference_code"], int(scene["inner_radius"]), (380, 108))
    draw.line(ref_points + [ref_points[0]], fill=ORANGE, width=7, joint="curve")

    boxes = ((35, 200, 365, 382), (395, 200, 725, 382), (35, 400, 365, 582), (395, 400, 725, 582))
    for index, (box, center) in enumerate(zip(boxes, PANEL_CENTERS, strict=True)):
        draw.rounded_rectangle(box, radius=10, outline=PANEL, width=2, fill=(255, 254, 251))
        draw.text((box[0] + 12, box[1] + 9), LABELS[index], fill=TEXT, font=label_font)
        for x1, y1, x2, y2 in scene["clutter_segments"][index]:
            draw.line((x1, y1, x2, y2), fill=CLUTTER, width=4)
        ox, oy = scene["contour_offsets"][index]
        shifted = (center[0] + int(ox), center[1] + int(oy))
        points = _contour_points(scene["candidate_codes"][index], int(scene["inner_radius"]), shifted)
        draw.line(points + [points[0]], fill=NAVY, width=7, joint="curve")
    return image


def analytic_gold(scene: dict) -> str:
    reference = tuple(int(value) for value in scene["reference_code"])
    matches = [index for index, code in enumerate(scene["candidate_codes"]) if tuple(int(value) for value in code) == reference]
    if len(matches) != 1:
        raise ValueError(f"expected one contour match, found {len(matches)}")
    return LABELS[matches[0]]


def margin(scene: dict) -> float:
    reference = tuple(int(value) for value in scene["reference_code"])
    candidates = [tuple(int(value) for value in code) for code in scene["candidate_codes"]]
    distances = [sum(left != right for left, right in zip(reference, code, strict=True)) for code in candidates if code != reference]
    if not distances:
        return 0.0
    radial_gap = OUTER_RADIUS - int(scene["inner_radius"])
    return float(min(distances) * radial_gap)


def is_quarantined(scene: dict) -> bool:
    try:
        analytic_gold(scene)
    except ValueError:
        return True
    return margin(scene) <= 8.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["clutter_segments"] = [list(reversed(panel)) for panel in twin["clutter_segments"]]
    return [("reverse_same_color_clutter_draw_order", twin)]
