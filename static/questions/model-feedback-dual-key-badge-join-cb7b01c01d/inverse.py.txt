"""Independent image-only inverse for Dual-Key Badge Join."""

from __future__ import annotations

from PIL import Image, ImageDraw

CANVAS = (1000, 800)
INK = (35, 40, 48)
SHAPE = (25, 126, 111)
STRIPE = (51, 91, 170)
BADGE = (58, 48, 92)
MUTED = (222, 226, 229)
COLORS = {"amber": (219, 139, 22), "cyan": (18, 153, 177), "violet": (133, 76, 181)}
ROW_Y = (320, 400, 480, 560)


class OracleError(ValueError):
    pass


def _mask_crop(image: Image.Image, box: tuple[int, int, int, int], color: tuple[int, int, int]) -> tuple[int, ...]:
    crop = image.crop(box)
    return tuple(1 if p == color else 0 for p in crop.getdata())


def _shape_template(name: str) -> tuple[int, ...]:
    im = Image.new("RGB", (70, 70), "white")
    d, cx, cy = ImageDraw.Draw(im), 35, 35
    if name == "circle": d.ellipse((10, 10, 60, 60), fill=SHAPE)
    elif name == "square": d.rectangle((12, 12, 58, 58), fill=SHAPE)
    elif name == "triangle": d.polygon(((35, 6), (6, 58), (64, 58)), fill=SHAPE)
    elif name == "diamond": d.polygon(((35, 5), (5, 35), (35, 65), (65, 35)), fill=SHAPE)
    return tuple(1 if p == SHAPE else 0 for p in im.getdata())


def _read_shape(image: Image.Image, cx: int, cy: int) -> str:
    observed = _mask_crop(image, (cx - 35, cy - 35, cx + 35, cy + 35), SHAPE)
    scores = {name: sum(a != b for a, b in zip(observed, _shape_template(name))) for name in ("circle", "square", "triangle", "diamond")}
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 1500:
        raise OracleError("damaged or unknown shape")
    return best


def _stripe_template(name: str) -> tuple[int, ...]:
    im = Image.new("RGB", (76, 76), "white")
    d, cx, cy, w = ImageDraw.Draw(im), 38, 38, 7
    if name == "horizontal":
        for dy in (-18, 0, 18): d.line((8, cy + dy, 68, cy + dy), fill=STRIPE, width=w)
    elif name == "vertical":
        for dx in (-18, 0, 18): d.line((cx + dx, 8, cx + dx, 68), fill=STRIPE, width=w)
    elif name == "rising":
        for off in (-16, 0, 16): d.line((10, cy + 22 + off, 66, cy - 22 + off), fill=STRIPE, width=w)
    elif name == "falling":
        for off in (-16, 0, 16): d.line((10, cy - 22 + off, 66, cy + 22 + off), fill=STRIPE, width=w)
    return tuple(1 if p == STRIPE else 0 for p in im.getdata())


def _read_stripe(image: Image.Image, cx: int, cy: int) -> str:
    observed = _mask_crop(image, (cx - 38, cy - 38, cx + 38, cy + 38), STRIPE)
    scores = {name: sum(a != b for a, b in zip(observed, _stripe_template(name))) for name in ("horizontal", "vertical", "rising", "falling")}
    best = min(scores, key=scores.get)
    if scores[best] != 0 or sum(observed) < 700:
        raise OracleError("damaged or unknown stripe")
    return best


def _read_badge(image: Image.Image, cx: int, cy: int) -> tuple[int, ...]:
    px = image.load()
    if px[cx - 48, cy] != INK or px[cx + 48, cy] != INK:
        raise OracleError("badge border missing")
    bits = []
    for i in range(6):
        col, row = i % 3, i // 3
        x0, y0 = cx - 34 + col * 24, cy - 23 + row * 28
        samples = [px[x, y] for x in range(x0 + 3, x0 + 16, 4) for y in range(y0 + 3, y0 + 16, 4)]
        if all(p == BADGE for p in samples): bits.append(1)
        elif all(p == MUTED for p in samples): bits.append(0)
        else: raise OracleError("badge cell damaged")
    if sum(bits) != 3:
        raise OracleError("invalid badge")
    return tuple(bits)


def decision_from_image(image: Image.Image) -> str:
    image = image.convert("RGB")
    if image.size != CANVAS:
        raise OracleError("wrong canvas")
    px = image.load()
    for point in ((500, 48), (45, 400), (480, 400), (520, 400), (955, 400)):
        if px[point[0], point[1]] != INK:
            raise OracleError("panel boundary missing")
    start_shape = _read_shape(image, 405, 151)
    start_stripe = _read_stripe(image, 595, 151)
    shape_map, stripe_map = {}, {}
    for y in ROW_Y:
        if px[225, y] != INK or px[700, y] != INK:
            raise OracleError("key arrow missing")
        shape_map[_read_shape(image, 135, y)] = _read_badge(image, 360, y)
        stripe_map[_read_stripe(image, 610, y)] = _read_badge(image, 835, y)
    if len(shape_map) != 4 or len(stripe_map) != 4:
        raise OracleError("duplicate key")
    target = {shape_map[start_shape], stripe_map[start_stripe]}
    matches = []
    for cx in (175, 500, 825):
        border = px[cx, 649]
        names = [name for name, rgb in COLORS.items() if border == rgb]
        if len(names) != 1:
            raise OracleError("answer border damaged")
        pair = {_read_badge(image, cx - 58, 713), _read_badge(image, cx + 58, 713)}
        if pair == target:
            matches.append(names[0])
    if len(matches) != 1:
        raise OracleError("non-unique joined pair")
    return matches[0]
