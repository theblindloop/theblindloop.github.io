"""Deterministic renderer and analytic arm for paired midpoint alignment."""

from __future__ import annotations

import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SCALE = 4
BACKGROUND = (255, 255, 255)
CORAL = (224, 78, 70)
BLUE = (48, 112, 184)
GOLD = (226, 168, 38)
FRAME = (232, 235, 239)
QUARANTINE_MARGIN = 8.0


def sample_scene(seed: int) -> dict:
    """Sample a balanced, translated scene with a stable raster margin."""
    rng = random.Random(int(seed))
    left_x = rng.choice((96, 160, 224, 288))
    right_x = left_x + 128
    center_y = rng.choice((160, 224, 288, 352))
    reference_y = center_y + rng.choice((-3, 0, 3))

    # Two independent bits counterbalance winner and horizontal color order.
    winner = "coral" if int(seed) % 2 == 0 else "blue"
    coral_left = (int(seed) // 2) % 2 == 0
    coral_x, blue_x = (left_x, right_x) if coral_left else (right_x, left_x)

    near_error = rng.choice((0, 1, 2, 3))
    decision_margin = rng.choice((8, 10, 12, 15))
    far_error = near_error + decision_margin
    near_sign = rng.choice((-1, 1))
    far_sign = rng.choice((-1, 1))
    coral_mid = reference_y + (near_sign * near_error if winner == "coral" else far_sign * far_error)
    blue_mid = reference_y + (near_sign * near_error if winner == "blue" else far_sign * far_error)
    coral_half = rng.choice((12, 14, 16))
    blue_half = rng.choice((12, 14, 16))

    return {
        "width": WIDTH,
        "height": HEIGHT,
        "coral_x": coral_x,
        "blue_x": blue_x,
        "coral_ys": [coral_mid - coral_half, coral_mid + coral_half],
        "blue_ys": [blue_mid - blue_half, blue_mid + blue_half],
        # Ten pixels keeps the gold disk visibly disconnected from a same-column
        # endpoint while placing both inside one fine-grid ablation support cell.
        "reference_x": left_x + 10,
        "reference_y": reference_y,
        "radius": 5,
    }


def _dot(draw: ImageDraw.ImageDraw, x: int, y: int, radius: int, fill: tuple[int, int, int]) -> None:
    box = ((x - radius) * SCALE, (y - radius) * SCALE, (x + radius) * SCALE, (y + radius) * SCALE)
    draw.ellipse(box, fill=fill)


def render(scene: dict) -> Image.Image:
    """Render only observable scene fields; no answer-specific styling is used."""
    width = int(scene["width"])
    height = int(scene["height"])
    coral_x = int(scene["coral_x"])
    blue_x = int(scene["blue_x"])
    coral_ys = [int(value) for value in scene["coral_ys"]]
    blue_ys = [int(value) for value in scene["blue_ys"]]
    reference_x = int(scene["reference_x"])
    reference_y = int(scene["reference_y"])
    radius = int(scene["radius"])

    image = Image.new("RGB", (width * SCALE, height * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    xs = coral_x, blue_x, reference_x
    ys = coral_ys + blue_ys + [reference_y]
    pad = 17
    draw.rounded_rectangle(
        ((min(xs) - pad) * SCALE, (min(ys) - pad) * SCALE,
         (max(xs) + pad) * SCALE, (max(ys) + pad) * SCALE),
        radius=10 * SCALE,
        outline=FRAME,
        width=2 * SCALE,
    )
    for y in coral_ys:
        _dot(draw, coral_x, y, radius, CORAL)
    for y in blue_ys:
        _dot(draw, blue_x, y, radius, BLUE)
    _dot(draw, reference_x, reference_y, radius, GOLD)
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _errors(scene: dict) -> tuple[float, float]:
    coral_ys = scene["coral_ys"]
    blue_ys = scene["blue_ys"]
    reference_y = scene["reference_y"]
    coral_error = abs((float(coral_ys[0]) + float(coral_ys[1])) / 2.0 - float(reference_y))
    blue_error = abs((float(blue_ys[0]) + float(blue_ys[1])) / 2.0 - float(reference_y))
    return coral_error, blue_error


def analytic_gold(scene: dict) -> str:
    coral_ys = scene["coral_ys"]
    blue_ys = scene["blue_ys"]
    reference_y = scene["reference_y"]
    coral_error = abs((float(coral_ys[0]) + float(coral_ys[1])) / 2.0 - float(reference_y))
    blue_error = abs((float(blue_ys[0]) + float(blue_ys[1])) / 2.0 - float(reference_y))
    if coral_error == blue_error:
        raise ValueError("tied pair-midpoint errors have no decision")
    return "coral" if coral_error < blue_error else "blue"


def margin(scene: dict) -> float:
    coral_error, blue_error = _errors(scene)
    return abs(coral_error - blue_error)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # No pixel-identical latent aliases are claimed under protocol v0.4.
    _ = deepcopy(scene)
    return []
