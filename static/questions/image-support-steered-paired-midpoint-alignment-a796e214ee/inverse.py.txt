"""Independent pixel-only inverse arm for paired midpoint alignment."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

_COLORS = {
    "coral": np.array((224, 78, 70), dtype=np.int16),
    "blue": np.array((48, 112, 184), dtype=np.int16),
    "gold": np.array((226, 168, 38), dtype=np.int16),
}
_MAX_COLOR_DISTANCE = 72.0
_MIN_COMPONENT_AREA = 35
_MAX_COMPONENT_AREA = 150


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[np.ndarray] = []
    for y0, x0 in np.argwhere(mask):
        y0, x0 = int(y0), int(x0)
        if seen[y0, x0]:
            continue
        queue = deque([(y0, x0)])
        seen[y0, x0] = True
        points = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for yy, xx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= yy < height and 0 <= xx < width and mask[yy, xx] and not seen[yy, xx]:
                    seen[yy, xx] = True
                    queue.append((yy, xx))
        if _MIN_COMPONENT_AREA <= len(points) <= _MAX_COMPONENT_AREA:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centers(array: np.ndarray, color: np.ndarray, expected: int) -> list[tuple[float, float]]:
    delta = array.astype(np.int16) - color[None, None, :]
    mask = np.sqrt(np.sum(delta.astype(np.float32) ** 2, axis=2)) < _MAX_COLOR_DISTANCE
    components = _components(mask)
    if len(components) != expected:
        raise ValueError(f"expected {expected} disconnected markers, found {len(components)}")
    centers = [(float(points[:, 1].mean()), float(points[:, 0].mean())) for points in components]
    return sorted(centers, key=lambda point: point[1])


def decision_from_image(image: Image.Image) -> str:
    """Recover marker components and repeat the declared midpoint comparison."""
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    coral = _centers(array, _COLORS["coral"], 2)
    blue = _centers(array, _COLORS["blue"], 2)
    gold = _centers(array, _COLORS["gold"], 1)
    coral_mid = (coral[0][1] + coral[1][1]) / 2.0
    blue_mid = (blue[0][1] + blue[1][1]) / 2.0
    reference_y = gold[0][1]
    coral_error = abs(coral_mid - reference_y)
    blue_error = abs(blue_mid - reference_y)
    if abs(coral_error - blue_error) < 3.0:
        raise ValueError("pixel decision is too close to the finite-resolution boundary")
    return "coral" if coral_error < blue_error else "blue"
