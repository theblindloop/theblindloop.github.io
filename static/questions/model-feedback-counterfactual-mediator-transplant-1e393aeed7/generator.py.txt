"""Deterministic renderer for nested counterfactual mediator transplants."""

from __future__ import annotations

import itertools
import random
from collections import Counter
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT, SCALE = 960, 780, 2
BACKGROUND, PANEL_FILL = (246, 245, 240), (255, 255, 255)
INK, OFF, RED, WIRE = (35, 40, 47), (255, 255, 255), (205, 54, 54), (158, 164, 172)
GATE_COLORS = {"and": (48, 112, 190), "or": (224, 148, 39), "xor": (132, 76, 183)}
GATE_LABELS = {"and": "ALL", "or": "ANY", "xor": "ONE"}
TOKEN_COLORS = (
    (0, 150, 180), (0, 140, 70), (225, 100, 0), (190, 45, 135),
    (45, 75, 180), (130, 80, 25), (210, 180, 20), (105, 55, 150),
)
TOKEN_LABELS = tuple("ABCDEFGH")
ROW_Y = (130, 225, 320, 415, 510)
TOP_X0, CELL_STEP, CELL_RADIUS = 84, 60, 16
GATE_X0, GATE_X1 = 650, 758
CARD_BOXES = ((35, 625, 245, 755), (260, 625, 470, 755),
              (485, 625, 695, 755), (710, 625, 920, 755))


def _apply(kind: str, left: int, right: int) -> int:
    if kind == "and":
        return int(bool(left) and bool(right))
    if kind == "or":
        return int(bool(left) or bool(right))
    if kind == "xor":
        return int(bool(left) != bool(right))
    raise ValueError(f"unknown gate {kind!r}")


def _next_row(row: list[int], kind: str) -> list[int]:
    return [_apply(kind, row[i], row[i + 1]) for i in range(len(row) - 1)]


def _run(top: list[int], gates: list[str]) -> list[list[int]]:
    rows = [list(top)]
    for kind in gates:
        rows.append(_next_row(rows[-1], kind))
    return rows


def _intervened_top(top: list[int], marked: int) -> list[int]:
    changed = list(top)
    changed[marked] ^= 1
    return changed


def _transplant_run(top: list[int], gates: list[str], treated_rows: list[list[int]],
                    bundle: tuple[tuple[int, int], tuple[int, int]]) -> list[list[int]]:
    """Run factual TOP while clamping named mediators to treated-world values."""
    rows = [list(top)]
    selected = set(bundle)
    for row_index, kind in enumerate(gates, 1):
        row = _next_row(rows[-1], kind)
        for selected_row, selected_cell in selected:
            if selected_row == row_index:
                row[selected_cell] = int(treated_rows[row_index][selected_cell])
        rows.append(row)
    return rows


def _bundle_coordinates(scene: dict, bundle_index: int) -> tuple[tuple[int, int], tuple[int, int]]:
    mediators = [tuple(int(value) for value in item) for item in scene["mediators"]]
    pair = [int(value) for value in scene["bundles"][bundle_index]]
    return mediators[pair[0]], mediators[pair[1]]


def _bundle_outcomes(scene: dict) -> list[list[int]]:
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    marked = int(scene["marked"])
    treated = _run(_intervened_top(top, marked), gates)
    return [_transplant_run(top, gates, treated, _bundle_coordinates(scene, index))[-1]
            for index in range(4)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    target_slot = int(seed) % 4
    for _ in range(60000):
        top = [rng.randrange(2) for _ in range(9)]
        gates = [rng.choice(("and", "or", "xor")) for _ in range(4)]
        if len(set(gates)) < 3:
            continue
        marked = rng.randrange(1, 8)
        factual = _run(top, gates)
        treated = _run(_intervened_top(top, marked), gates)
        if sum(a != b for a, b in zip(factual[-1], treated[-1], strict=True)) < 2:
            continue
        changed_row_2 = [(2, cell) for cell in range(7)
                         if factual[2][cell] != treated[2][cell]]
        changed_row_3 = [(3, cell) for cell in range(6)
                         if factual[3][cell] != treated[3][cell]]
        possibilities = []
        for first in changed_row_2:
            for second in changed_row_3:
                bundle = (first, second)
                outcome = _transplant_run(top, gates, treated, bundle)[-1]
                from_target = sum(a != b for a, b in zip(outcome, treated[-1], strict=True))
                from_factual = sum(a != b for a, b in zip(outcome, factual[-1], strict=True))
                possibilities.append((bundle, tuple(outcome), from_target, from_factual))
        correct = [item for item in possibilities if item[2] == 0]
        wrong = [item for item in possibilities if item[2] >= 2 and item[3] >= 2]
        rng.shuffle(correct)
        rng.shuffle(wrong)
        selection = None
        for survivor in correct:
            for distractors in itertools.combinations(wrong, 3):
                chosen = (survivor, *distractors)
                if len({item[1] for item in distractors}) != 3:
                    continue
                row_2_cells = {item[0][0] for item in chosen}
                row_3_cells = {item[0][1] for item in chosen}
                union = {coordinate for item in chosen for coordinate in item[0]}
                frequencies = Counter(coordinate for item in chosen for coordinate in item[0])
                if len(row_2_cells) < 3 or len(row_3_cells) < 3 or len(union) < 6:
                    continue
                if max(frequencies.values()) > 2 or len(union) > len(TOKEN_COLORS):
                    continue
                selection = chosen
                break
            if selection is not None:
                break
        if selection is None:
            continue
        survivor, *distractors = selection
        rng.shuffle(distractors)
        ordered = []
        cursor = 0
        for slot in range(4):
            item = survivor if slot == target_slot else distractors[cursor]
            cursor += int(slot != target_slot)
            ordered.append(item)
        mediator_coordinates = sorted({coordinate for item in ordered for coordinate in item[0]})
        coordinate_to_id = {coordinate: index for index, coordinate in enumerate(mediator_coordinates)}
        bundles = [[coordinate_to_id[item[0][0]], coordinate_to_id[item[0][1]]]
                   for item in ordered]
        return {
            "top": top,
            "gates": gates,
            "marked": marked,
            "mediators": [list(item) for item in mediator_coordinates],
            "bundles": bundles,
            "card_draw_order": [0, 1, 2, 3],
        }
    raise RuntimeError("could not sample a separated mediator-transplant scene")


def analytic_gold(scene: dict) -> str:
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    treated_bottom = _run(_intervened_top(top, int(scene["marked"])), gates)[-1]
    matches = [index for index, outcome in enumerate(_bundle_outcomes(scene))
               if outcome == treated_bottom]
    if len(matches) != 1:
        raise ValueError(f"expected one sufficient transplant card, got {matches}")
    return f"card-{matches[0] + 1}"


def margin(scene: dict) -> float:
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    target = _run(_intervened_top(top, int(scene["marked"])), gates)[-1]
    correct = int(analytic_gold(scene).split("-")[1]) - 1
    distances = [sum(a != b for a, b in zip(outcome, target, strict=True))
                 for index, outcome in enumerate(_bundle_outcomes(scene)) if index != correct]
    return float(min(distances))


def is_quarantined(scene: dict) -> bool:
    try:
        top = [int(value) for value in scene["top"]]
        gates = [str(value) for value in scene["gates"]]
        marked = int(scene["marked"])
        mediators = [tuple(int(value) for value in item) for item in scene["mediators"]]
        bundles = [[int(value) for value in item] for item in scene["bundles"]]
        if len(top) != 9 or any(value not in (0, 1) for value in top):
            return True
        if len(gates) != 4 or any(kind not in GATE_COLORS for kind in gates):
            return True
        if len(set(gates)) < 3 or not 1 <= marked <= 7:
            return True
        if not 6 <= len(mediators) <= len(TOKEN_COLORS) or len(set(mediators)) != len(mediators):
            return True
        if any(row not in (2, 3) or not 0 <= cell < 9 - row for row, cell in mediators):
            return True
        if len(bundles) != 4 or any(len(item) != 2 or item[0] == item[1] for item in bundles):
            return True
        if any(not 0 <= token < len(mediators) for item in bundles for token in item):
            return True
        if len({tuple(item) for item in bundles}) != 4:
            return True
        if sorted(scene["card_draw_order"]) != [0, 1, 2, 3]:
            return True
        coordinates = [_bundle_coordinates(scene, index) for index in range(4)]
        if any({first[0], second[0]} != {2, 3} for first, second in coordinates):
            return True
        if len({first for first, _ in coordinates}) < 3 or len({second for _, second in coordinates}) < 3:
            return True
        if max(Counter(point for pair in coordinates for point in pair).values()) > 2:
            return True
        factual = _run(top, gates)
        treated = _run(_intervened_top(top, marked), gates)
        if sum(a != b for a, b in zip(factual[-1], treated[-1], strict=True)) < 2:
            return True
        if any(factual[row][cell] == treated[row][cell] for row, cell in mediators):
            return True
        outcomes = _bundle_outcomes(scene)
        matches = [index for index, outcome in enumerate(outcomes) if outcome == treated[-1]]
        if len(matches) != 1:
            return True
        wrong = [outcome for index, outcome in enumerate(outcomes) if index != matches[0]]
        if len({tuple(outcome) for outcome in wrong}) != 3:
            return True
        if any(sum(a != b for a, b in zip(outcome, treated[-1], strict=True)) < 2
               for outcome in wrong):
            return True
        return margin(scene) < 2.0
    except (KeyError, TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["card_draw_order"] = list(reversed(twin["card_draw_order"]))
    return [("reverse_disjoint_card_draw_order", twin)]


def _font(size: int, bold: bool = False) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype("DejaVuSans-Bold.ttf" if bold else "DejaVuSans.ttf", size * SCALE)


def _pt(point: tuple[int, int]) -> tuple[int, int]:
    return point[0] * SCALE, point[1] * SCALE


def _box(box: tuple[int, int, int, int]) -> tuple[int, int, int, int]:
    return tuple(value * SCALE for value in box)  # type: ignore[return-value]


def _row_centers(row_index: int) -> list[tuple[int, int]]:
    x0 = TOP_X0 + row_index * (CELL_STEP // 2)
    return [(x0 + index * CELL_STEP, ROW_Y[row_index]) for index in range(9 - row_index)]


def _draw_cell(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int) -> None:
    x, y = center
    draw.ellipse(_box((x - CELL_RADIUS, y - CELL_RADIUS, x + CELL_RADIUS, y + CELL_RADIUS)),
                 fill=INK if value else OFF, outline=INK, width=3 * SCALE)


def _draw_token_marker(draw: ImageDraw.ImageDraw, center: tuple[int, int], token: int) -> None:
    x, y = center
    color = TOKEN_COLORS[token]
    draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=color, width=6 * SCALE)
    draw.rounded_rectangle(_box((x - 13, y + 27, x + 13, y + 50)), radius=6 * SCALE,
                           fill=color, outline=INK, width=2 * SCALE)
    label, font = TOKEN_LABELS[token], _font(15, True)
    bbox = draw.textbbox((0, 0), label, font=font)
    draw.text((x * SCALE - (bbox[2] - bbox[0]) // 2,
               (y + 37) * SCALE - (bbox[3] - bbox[1]) // 2 - SCALE),
              label, fill=(255, 255, 255), font=font)


def _draw_card(draw: ImageDraw.ImageDraw, slot: int, bundle: list[int]) -> None:
    x0, y0, x1, y1 = CARD_BOXES[slot]
    draw.rounded_rectangle(_box((x0, y0, x1, y1)), radius=13 * SCALE,
                           fill=PANEL_FILL, outline=INK, width=3 * SCALE)
    heading, font = f"CARD {slot + 1}", _font(19, True)
    bbox = draw.textbbox((0, 0), heading, font=font)
    draw.text(((x0 + x1) * SCALE // 2 - (bbox[2] - bbox[0]) // 2, (y0 + 14) * SCALE),
              heading, fill=INK, font=font)
    for x, token in zip((x0 + 65, x0 + 145), bundle, strict=True):
        color = TOKEN_COLORS[int(token)]
        draw.ellipse(_box((x - 22, y0 + 66, x + 22, y0 + 110)), fill=color, outline=INK,
                     width=2 * SCALE)
        label, token_font = TOKEN_LABELS[int(token)], _font(20, True)
        bbox = draw.textbbox((0, 0), label, font=token_font)
        draw.text((x * SCALE - (bbox[2] - bbox[0]) // 2,
                   (y0 + 88) * SCALE - (bbox[3] - bbox[1]) // 2 - SCALE),
                  label, fill=(255, 255, 255), font=token_font)
    draw.text(_pt((x0 + 101, y0 + 78)), "+", fill=INK, font=_font(18, True))


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text(_pt((28, 14)), "TRANSPLANT VALUES BETWEEN TWO WORLDS", fill=INK, font=_font(24, True))
    draw.text(_pt((30, 50)), "1 flip red + record letters   2 restore TOP   3 transplant one card",
              fill=INK, font=_font(16))
    draw.text(_pt((794, 20)), "dark = ON", fill=INK, font=_font(16, True))
    top = [int(value) for value in scene["top"]]
    gates = [str(value) for value in scene["gates"]]
    rows = _run(top, gates)
    centers = [_row_centers(index) for index in range(5)]
    for row_index in range(1, 5):
        for index, child in enumerate(centers[row_index]):
            for parent in (centers[row_index - 1][index], centers[row_index - 1][index + 1]):
                draw.line([_pt(parent), _pt(child)], fill=WIRE, width=3 * SCALE)
    draw.text(_pt((17, ROW_Y[0] - 10)), "TOP", fill=INK, font=_font(14, True))
    mediator_lookup = {tuple(int(value) for value in coordinate): token
                       for token, coordinate in enumerate(scene["mediators"])}
    for row_index, row in enumerate(rows):
        for cell_index, value in enumerate(row):
            center = centers[row_index][cell_index]
            if row_index == 0 and cell_index == int(scene["marked"]):
                x, y = center
                draw.ellipse(_box((x - 27, y - 27, x + 27, y + 27)), outline=RED, width=6 * SCALE)
            token = mediator_lookup.get((row_index, cell_index))
            if token is not None:
                _draw_token_marker(draw, center, token)
            _draw_cell(draw, center, int(value))
        if row_index:
            kind, y = gates[row_index - 1], ROW_Y[row_index]
            draw.rounded_rectangle(_box((GATE_X0, y - 23, GATE_X1, y + 23)), radius=9 * SCALE,
                                   fill=GATE_COLORS[kind], outline=INK, width=2 * SCALE)
            label, font = GATE_LABELS[kind], _font(17, True)
            bbox = draw.textbbox((0, 0), label, font=font)
            draw.text((704 * SCALE - (bbox[2] - bbox[0]) // 2,
                       y * SCALE - (bbox[3] - bbox[1]) // 2 - SCALE),
                      label, fill=(255, 255, 255), font=font)
            row_label = "FACTUAL BOTTOM" if row_index == 4 else f"ROW {row_index}"
            draw.text(_pt((776, y - 10)), row_label, fill=INK, font=_font(14, True))
    draw.text(_pt((28, 577)), "WHICH TRANSPLANT RECREATES THE RED-FLIPPED BOTTOM?", fill=INK,
              font=_font(19, True))
    for slot in scene["card_draw_order"]:
        _draw_card(draw, int(slot), [int(value) for value in scene["bundles"][int(slot)]])
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)
