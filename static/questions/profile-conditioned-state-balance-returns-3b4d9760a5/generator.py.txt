"""Balance return-to-origin counter across ordered operator panels.

A running balance register starts at 0 (marked by a labelled start panel). Six
operator panels run left to right; each shows a single step: a solid green
up-triangle is +1 and a solid red down-triangle is -1. Replaying the six steps
in order from the marked start of 0 yields the running balance after each step.
A step is a RETURN to the origin when, after applying it, the running balance is
exactly 0 again. The decision is the number of such returns across the six steps
(excluding the start itself).

This is a whole-sequence temporal aggregate: it depends on the running balance
state carried step to step and on where it touches zero, never on any single
frame or on the raw counts of up/down arrows alone (e.g. three ups and three
downs can produce anywhere from zero to three returns depending on their order).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

NPANEL = 6
STEP = 1

BG = (255, 255, 255)
UP = (34, 139, 34)
DOWN = (220, 20, 60)
TEXT = (60, 60, 60)
LABEL = (240, 240, 240)
BORDER = (140, 140, 140)

WIDTH = 760
HEIGHT = 160
START_W = 130
START_H = 80
PANEL_W = 80
PANEL_H = 80
GAP = 18
HEADER_H = 40


def _x_start() -> int:
    return 16


def _panel_x(k: int) -> int:
    return _x_start() + START_W + 30 + k * (PANEL_W + GAP)


def _returns(ops: tuple) -> int:
    balance = 0
    count = 0
    for op in ops:
        balance += STEP if op == 1 else -STEP
        if balance == 0:
            count += 1
    return count


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 104729 + 17)
    ops = tuple(rng.choice((1, -1)) for _ in range(NPANEL))
    return {"ops": ops}


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(img)
    ops = scene["ops"]

    sx = _x_start()
    draw.rectangle([sx, HEADER_H, sx + START_W, HEADER_H + START_H], fill=LABEL)
    draw.rectangle([sx, HEADER_H, sx + START_W, HEADER_H + START_H], outline=BORDER, width=2)
    draw.text((sx + 8, HEADER_H + 8), "start =", fill=TEXT)
    draw.text((sx + 8, HEADER_H + 34), "0", fill=TEXT)

    for k, op in enumerate(ops):
        x0 = _panel_x(k)
        y0 = HEADER_H
        color = UP if op == 1 else DOWN
        cx = x0 + PANEL_W // 2
        cy_top = y0 + 20
        cy_bottom = y0 + PANEL_H - 20
        if op == 1:
            pts = [(cx, cy_top), (x0 + 8, cy_bottom), (x0 + PANEL_W - 8, cy_bottom)]
        else:
            pts = [(cx, cy_bottom), (x0 + 8, cy_top), (x0 + PANEL_W - 8, cy_top)]
        draw.polygon(pts, fill=color)
        label = "+1" if op == 1 else "-1"
        draw.text((cx - 8, y0 + PANEL_H + 2), label, fill=TEXT)

    draw.text((sx, 6), "Order of steps (left to right), starting balance 0", fill=TEXT)
    return img


def analytic_gold(scene: dict) -> int:
    return _returns(scene["ops"])


def margin(scene: dict) -> float:
    return 1.0


def is_quarantined(scene: dict) -> bool:
    return False


def latent_symmetries(scene: dict) -> list:
    """No nontrivial render-invariant latent symmetry exists.

    ``ops`` fixes the exact ordered sequence of drawn step triangles in their own
    panels, and that order is the decision's temporal axis: swapping any two steps
    or flipping any sign changes the raster and, generically, the number of
    returns to the origin. Because identity is positional (which panel owns which
    step) and every step is directly visible as a sign, no reparametrization
    yields pixel-identical renders with a different gold.
    """
    return []
