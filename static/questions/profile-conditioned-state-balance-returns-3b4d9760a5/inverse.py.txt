"""Independent pixel-only inverse arm for the balance-return counter world.

Receives only the final PNG. From the raster alone it:
  1. labels pixels against the fixed palette (white background, solid green
     up-triangle, solid red down-triangle);
  2. for each of the six ordered operator panels (fixed left-to-right geometry)
     determines whether it holds a green up-triangle (+1) or a red down-triangle
     (-1) by which solid colour is present;
  3. replays the six steps from the marked starting balance 0 through a running
     balance register and counts how many steps return the balance exactly to 0.

No latent scene, renderer, prompt, or gold code is accessed.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

BG = (255, 255, 255)
UP = (34, 139, 34)
DOWN = (220, 20, 60)

PALETTE = [BG, UP, DOWN]
_BG, _UP, _DOWN = range(3)

NPANEL = 6
START_W = 130
PANEL_W = 80
PANEL_H = 80
GAP = 18
HEADER_H = 40
STEP = 1

PANEL_AREA_THRESHOLD = 200


class UnreadableImageError(ValueError):
    """The raster does not expose the expected ordered-operator structure."""


def _label(arr: np.ndarray) -> np.ndarray:
    dists = np.stack(
        [np.linalg.norm(arr - np.array(c, dtype=np.float32), axis=-1) for c in PALETTE],
        axis=-1,
    )
    return dists.argmin(axis=-1)


def _panel_x(k: int) -> int:
    return 16 + START_W + 30 + k * (PANEL_W + GAP)


def decision_from_image(image: Image.Image) -> int:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _label(arr)
    up_mask = labels == _UP
    down_mask = labels == _DOWN
    if not up_mask.any() and not down_mask.any():
        raise UnreadableImageError("no step triangle ink found in the image")

    ops = []
    for k in range(NPANEL):
        x0 = _panel_x(k)
        x1 = x0 + PANEL_W
        y0 = HEADER_H
        y1 = HEADER_H + PANEL_H
        up_count = int(up_mask[y0:y1, x0:x1].sum())
        down_count = int(down_mask[y0:y1, x0:x1].sum())
        if up_count >= PANEL_AREA_THRESHOLD and down_count < PANEL_AREA_THRESHOLD:
            ops.append(1)
        elif down_count >= PANEL_AREA_THRESHOLD and up_count < PANEL_AREA_THRESHOLD:
            ops.append(-1)
        else:
            raise UnreadableImageError(
                f"panel {k + 1} has ambiguous step ink (up={up_count}, down={down_count})"
            )

    balance = 0
    returns = 0
    for op in ops:
        balance += STEP if op == 1 else -STEP
        if balance == 0:
            returns += 1
    return returns
