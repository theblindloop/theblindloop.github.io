"""Scene sampler, raster renderer, analytic gold and latent aliases.

World: five framed panels stand side by side.  Every panel carries the same
five pillar heights and the same five pillar shades; only the left-to-right
arrangement differs between panels.  Exactly one panel presents its pillars in
strictly ascending height order.  The decision is that panel's position.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

N_PANELS = 5
N_PILLARS = 5

CANVAS_MARGIN = 22
TOP_MARGIN = 20
BOTTOM_MARGIN = 32
PANEL_W = 178
PANEL_H = 216
PANEL_GAP = 18
FRAME_PX = 3

PILLAR_W = 22
PILLAR_PITCH = 32
BASE_INSET = 18
OUTLINE_PX = 2

MIN_HEIGHT = 30
MAX_MIN_HEIGHT = 44
MIN_GAP = 17
MAX_GAP = 31

QUARANTINE_PX = 12.0
AMBIGUOUS = "ambiguous"

SHADES = (96, 124, 152, 180, 208)

PALETTES = (
    {"bg": (247, 246, 242), "frame": (78, 84, 96), "outline": (34, 38, 46)},
    {"bg": (243, 247, 247), "frame": (66, 92, 92), "outline": (28, 44, 46)},
    {"bg": (248, 245, 247), "frame": (92, 78, 96), "outline": (40, 32, 44)},
)

CANVAS_W = 2 * CANVAS_MARGIN + N_PANELS * PANEL_W + (N_PANELS - 1) * PANEL_GAP
CANVAS_H = TOP_MARGIN + PANEL_H + BOTTOM_MARGIN


# --------------------------------------------------------------------------
# geometry helpers shared by the renderer only
# --------------------------------------------------------------------------


def _panel_origin(index: int) -> tuple:
    return CANVAS_MARGIN + index * (PANEL_W + PANEL_GAP), TOP_MARGIN


def _pillar_x(panel_x: int, slot: int) -> int:
    span = N_PILLARS * PILLAR_W + (N_PILLARS - 1) * (PILLAR_PITCH - PILLAR_W)
    left = panel_x + (PANEL_W - span) // 2
    return left + slot * PILLAR_PITCH


def _ordered(panel: list) -> list:
    """Pillars of one panel, read by slot.  The stored order is irrelevant."""
    return sorted(panel, key=lambda pillar: pillar["slot"])


def _heights(panel: list) -> list:
    return [pillar["height"] for pillar in _ordered(panel)]


def _shades(panel: list) -> list:
    return [pillar["shade"] for pillar in _ordered(panel)]


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------


def _is_monotone(values: list) -> bool:
    ascending = all(b > a for a, b in zip(values, values[1:]))
    descending = all(b < a for a, b in zip(values, values[1:]))
    return ascending or descending


def _height_pool(rng: random.Random) -> list:
    pool = [rng.randint(MIN_HEIGHT, MAX_MIN_HEIGHT)]
    for _ in range(N_PILLARS - 1):
        pool.append(pool[-1] + rng.randint(MIN_GAP, MAX_GAP))
    return pool


def _distractor_perms(rng: random.Random) -> list:
    """Four distinct non-ascending rank orders with balanced end positions."""
    identity = tuple(range(N_PILLARS))
    for _ in range(400):
        kinds = ["adjacent", "adjacent", "shuffled", "shuffled"]
        rng.shuffle(kinds)
        perms = []
        for kind in kinds:
            for _ in range(60):
                if kind == "adjacent":
                    cut = rng.randrange(N_PILLARS - 1)
                    perm = list(identity)
                    perm[cut], perm[cut + 1] = perm[cut + 1], perm[cut]
                    perm = tuple(perm)
                else:
                    perm = tuple(rng.sample(range(N_PILLARS), N_PILLARS))
                if perm != identity and perm not in perms:
                    perms.append(perm)
                    break
        if len(perms) != 4:
            continue
        tall_right = sum(1 for perm in perms if perm[-1] == N_PILLARS - 1)
        short_left = sum(1 for perm in perms if perm[0] == 0)
        if tall_right >= 2 and short_left >= 2:
            return perms
    raise RuntimeError("could not sample balanced distractor arrangements")


def _shade_orders(rng: random.Random, graded: list) -> list:
    orders = []
    for index in range(N_PANELS):
        if index in graded:
            order = list(SHADES) if rng.random() < 0.5 else list(reversed(SHADES))
        else:
            for _ in range(80):
                order = rng.sample(SHADES, N_PILLARS)
                if not _is_monotone(order):
                    break
        orders.append(list(order))
    return orders


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    palette = rng.randrange(len(PALETTES))
    pool = _height_pool(rng)
    winner = rng.randrange(N_PANELS)
    distractors = _distractor_perms(rng)

    perms = []
    cursor = 0
    for index in range(N_PANELS):
        if index == winner:
            perms.append(tuple(range(N_PILLARS)))
        else:
            perms.append(distractors[cursor])
            cursor += 1

    heights = [[pool[rank] for rank in perm] for perm in perms]
    graded = sorted(rng.sample(range(N_PANELS), 2))

    orders = _shade_orders(rng, graded)
    for _ in range(200):
        weighted = [
            sum(h * (255 - s) for h, s in zip(heights[i], orders[i]))
            for i in range(N_PANELS)
        ]
        rank = sorted(range(N_PANELS), key=lambda i: weighted[i]).index(winner)
        if 1 <= rank <= N_PANELS - 2:
            break
        orders = _shade_orders(rng, graded)

    panels = []
    for index in range(N_PANELS):
        pillars = [
            {"slot": slot, "height": heights[index][slot], "shade": orders[index][slot]}
            for slot in range(N_PILLARS)
        ]
        rng.shuffle(pillars)
        panels.append(pillars)

    return {
        "width": CANVAS_W,
        "height": CANVAS_H,
        "palette": palette,
        "panels": panels,
    }


# --------------------------------------------------------------------------
# rendering
# --------------------------------------------------------------------------


def render(scene: dict) -> Image.Image:
    palette = PALETTES[scene["palette"] % len(PALETTES)]
    image = Image.new("RGB", (scene["width"], scene["height"]), palette["bg"])
    draw = ImageDraw.Draw(image)
    for index, panel in enumerate(scene["panels"]):
        panel_x, panel_y = _panel_origin(index)
        draw.rectangle(
            [panel_x, panel_y, panel_x + PANEL_W - 1, panel_y + PANEL_H - 1],
            outline=palette["frame"],
            width=FRAME_PX,
        )
        base_y = panel_y + PANEL_H - BASE_INSET
        for pillar in _ordered(panel):
            left = _pillar_x(panel_x, pillar["slot"])
            top = base_y - pillar["height"] + 1
            shade = int(pillar["shade"])
            draw.rectangle(
                [left, top, left + PILLAR_W - 1, base_y],
                fill=(shade, shade, shade),
                outline=palette["outline"],
                width=OUTLINE_PX,
            )
    return image


# --------------------------------------------------------------------------
# analytic arm
# --------------------------------------------------------------------------


def _ascending_panels(scene: dict) -> list:
    found = []
    for index, panel in enumerate(scene["panels"]):
        heights = _heights(panel)
        if all(b > a for a, b in zip(heights, heights[1:])):
            found.append(index)
    return found


def analytic_gold(scene: dict) -> str:
    found = _ascending_panels(scene)
    if len(found) != 1:
        return AMBIGUOUS
    return f"panel-{found[0] + 1}"


def margin(scene: dict) -> float:
    found = _ascending_panels(scene)
    if len(found) != 1:
        return 0.0
    winner = found[0]
    heights = _heights(scene["panels"][winner])
    rise = min(b - a for a, b in zip(heights, heights[1:]))
    drops = []
    for index, panel in enumerate(scene["panels"]):
        if index == winner:
            continue
        other = _heights(panel)
        drops.append(max(a - b for a, b in zip(other, other[1:])))
    return float(min([rise] + drops))


def is_quarantined(scene: dict) -> bool:
    if analytic_gold(scene) == AMBIGUOUS:
        return True
    return margin(scene) < QUARANTINE_PX


# --------------------------------------------------------------------------
# declared latent aliases
# --------------------------------------------------------------------------


def latent_symmetries(scene: dict) -> list:
    """Transforms that leave the raster byte-identical and the gold unchanged."""
    period = dict(scene)
    period["palette"] = scene["palette"] + len(PALETTES)
    period["panels"] = [
        [dict(pillar) for pillar in panel] for panel in scene["panels"]
    ]

    reversed_order = dict(scene)
    reversed_order["panels"] = [
        [dict(pillar) for pillar in reversed(panel)] for panel in scene["panels"]
    ]

    rotated = dict(scene)
    rotated["panels"] = [
        [dict(pillar) for pillar in panel[2:] + panel[:2]]
        for panel in scene["panels"]
    ]

    return [
        ("palette_period", period),
        ("pillar_list_reversed", reversed_order),
        ("pillar_list_rotated", rotated),
    ]
