"""Pixel-only inverse arm for the rank-ramp panel world.

The oracle receives a rasterized image and nothing else.  It segments the row
of framed panels from the ink columns, recovers each panel's frame rectangle
from the long horizontal and vertical ink runs, measures the pixel height of
every pillar standing on the panel's common baseline, and reports the single
panel whose pillar heights increase strictly from left to right.  It abstains
whenever the layout, the pillar count, the shared baseline, or the winner is
not recovered unambiguously.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

AMBIGUOUS = "ambiguous"

N_PANELS = 5
N_PILLARS = 5
INK_MAX = 228
FRAME_RUN_FRACTION = 0.7
FRAME_INSET = 6
MIN_PANEL_W = 60
MIN_PANEL_H = 60
MIN_PILLAR_W = 10
BASE_TOLERANCE = 3
STEP_TOLERANCE = 6.0


def _runs(flags: np.ndarray) -> list:
    """Contiguous index ranges where the boolean profile is true."""
    spans = []
    start = None
    for index, value in enumerate(flags):
        if value and start is None:
            start = index
        elif not value and start is not None:
            spans.append((start, index - 1))
            start = None
    if start is not None:
        spans.append((start, len(flags) - 1))
    return spans


def _panel_heights(ink: np.ndarray) -> list:
    """Pillar heights inside one panel, left to right, or None."""
    rows, cols = ink.shape
    if rows < MIN_PANEL_H or cols < MIN_PANEL_W:
        return None

    row_runs = ink.sum(axis=1) >= FRAME_RUN_FRACTION * cols
    if not row_runs.any():
        return None
    row_index = np.flatnonzero(row_runs)
    top, bottom = int(row_index[0]), int(row_index[-1])
    if bottom - top < MIN_PANEL_H:
        return None

    band = ink[top : bottom + 1, :]
    col_runs = band.sum(axis=0) >= FRAME_RUN_FRACTION * band.shape[0]
    if not col_runs.any():
        return None
    col_index = np.flatnonzero(col_runs)
    left, right = int(col_index[0]), int(col_index[-1])
    if right - left < MIN_PANEL_W:
        return None

    interior = ink[
        top + FRAME_INSET : bottom - FRAME_INSET + 1,
        left + FRAME_INSET : right - FRAME_INSET + 1,
    ]
    if interior.size == 0:
        return None

    spans = [
        span
        for span in _runs(interior.any(axis=0))
        if span[1] - span[0] + 1 >= MIN_PILLAR_W
    ]
    if len(spans) != N_PILLARS:
        return None

    heights = []
    bases = []
    for start, end in spans:
        block = interior[:, start : end + 1]
        rows_with_ink = np.flatnonzero(block.any(axis=1))
        if rows_with_ink.size == 0:
            return None
        heights.append(int(rows_with_ink[-1] - rows_with_ink[0] + 1))
        bases.append(int(rows_with_ink[-1]))
    if max(bases) - min(bases) > BASE_TOLERANCE:
        return None
    return heights


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB")).astype(np.int32)
    ink = rgb.mean(axis=2) < INK_MAX

    columns = [
        span
        for span in _runs(ink.any(axis=0))
        if span[1] - span[0] + 1 >= MIN_PANEL_W
    ]
    if len(columns) != N_PANELS:
        return AMBIGUOUS

    ascending = []
    for panel_index, (start, end) in enumerate(columns):
        heights = _panel_heights(ink[:, start : end + 1])
        if heights is None:
            return AMBIGUOUS
        steps = [b - a for a, b in zip(heights, heights[1:])]
        if any(abs(step) < STEP_TOLERANCE for step in steps):
            return AMBIGUOUS
        if all(step > 0 for step in steps):
            ascending.append(panel_index)

    if len(ascending) != 1:
        return AMBIGUOUS
    return f"panel-{ascending[0] + 1}"
