"""Latent sampler and renderer for a polygon implied by fragmented sides."""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 512
SIDE_OPTIONS = (3, 4, 5, 6)
TOTAL_DASHES = 60
SUPERSAMPLE = 4


def sample_scene(seed: int) -> dict:
    """Sample a balanced, deterministic scene.

    Side class is cycled by seed. All potential cheap cues are independently
    sampled: every class has 60 equal-length dashes, the same radius range,
    the same pose range, and the same palette range.
    """

    seed = int(seed)
    rng = random.Random(seed ^ 0x51DECAFE)
    n_sides = SIDE_OPTIONS[seed % len(SIDE_OPTIONS)]
    dark = rng.choice(((30, 39, 48), (42, 35, 48), (31, 45, 38), (38, 38, 38)))
    paper = rng.choice(((248, 246, 239), (244, 247, 249), (249, 244, 241), (246, 246, 242)))
    return {
        "n_sides": n_sides,
        "center": [round(256.0 + rng.uniform(-22.0, 22.0), 4), round(256.0 + rng.uniform(-22.0, 22.0), 4)],
        "radius": round(rng.uniform(143.0, 164.0), 4),
        "rotation": round(rng.uniform(0.0, 2.0 * math.pi), 7),
        "dash_length": round(rng.uniform(8.4, 9.2), 4),
        "stroke_width": 3,
        "dash_rgb": list(dark),
        "background_rgb": list(paper),
    }


def _vertices(scene: dict) -> list[tuple[float, float]]:
    n = int(scene["n_sides"])
    cx, cy = (float(v) for v in scene["center"])
    radius = float(scene["radius"])
    rotation = float(scene["rotation"])
    return [
        (
            cx + radius * math.cos(rotation + 2.0 * math.pi * j / n),
            cy + radius * math.sin(rotation + 2.0 * math.pi * j / n),
        )
        for j in range(n)
    ]


def render(scene: dict) -> Image.Image:
    """Render exactly 60 mutually disconnected dashes on polygon supports."""

    n = int(scene["n_sides"])
    center = scene["center"]  # Explicitly rendered and retained for field audit.
    radius = float(scene["radius"])
    rotation = float(scene["rotation"])
    dash_length = float(scene["dash_length"])
    stroke_width = int(scene["stroke_width"])
    dash_rgb = tuple(int(v) for v in scene["dash_rgb"])
    background_rgb = tuple(int(v) for v in scene["background_rgb"])
    # These no-op validations make malformed latent fields fail rather than hide.
    if len(center) != 2 or radius <= 0 or not math.isfinite(rotation):
        raise ValueError("invalid polygon pose")
    if n not in SIDE_OPTIONS or TOTAL_DASHES % n:
        raise ValueError("unsupported side count")

    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), background_rgb)
    draw = ImageDraw.Draw(image)
    vertices = _vertices(scene)
    per_side = TOTAL_DASHES // n
    for side in range(n):
        x0, y0 = vertices[side]
        x1, y1 = vertices[(side + 1) % n]
        dx, dy = x1 - x0, y1 - y0
        side_length = math.hypot(dx, dy)
        ux, uy = dx / side_length, dy / side_length
        half = dash_length / 2.0
        for index in range(per_side):
            # One full pitch of vertex clearance prevents adjacent-side dashes
            # from touching at sharp corners after rasterization.
            t = (index + 1.0) / (per_side + 1.0)
            mx, my = x0 + t * dx, y0 + t * dy
            points = (
                ((mx - half * ux) * SUPERSAMPLE, (my - half * uy) * SUPERSAMPLE),
                ((mx + half * ux) * SUPERSAMPLE, (my + half * uy) * SUPERSAMPLE),
            )
            draw.line(points, fill=dash_rgb, width=stroke_width * SUPERSAMPLE)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(int(scene["n_sides"]))


def margin(scene: dict) -> float:
    """Ideal runner-up gap in recovered support count (one whole support)."""

    n = int(scene["n_sides"])
    alternatives = [abs(n - other) for other in SIDE_OPTIONS if other != n]
    return float(min(alternatives))


def is_quarantined(scene: dict) -> bool:
    """Quarantine only if raster clearance could merge neighboring dashes."""

    n = int(scene["n_sides"])
    side_length = 2.0 * float(scene["radius"]) * math.sin(math.pi / n)
    pitch = side_length / (TOTAL_DASHES // n + 1)
    clearance = pitch - float(scene["dash_length"])
    return clearance < 2.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # The parameterization is render-injective over declared fields.
    return []
