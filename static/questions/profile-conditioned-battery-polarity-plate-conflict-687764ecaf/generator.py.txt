"""Deterministic latent sampler and raster renderer for a battery-symbol prior conflict.

The two members of each matched pair share all geometry and styling.  The even
seed places ``+`` above the longer plate (the conventional control); the odd
seed swaps the two glyphs (the counterfactual).  Gold depends only on the two
visible alignment fields, never on a hidden control flag.
"""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw


PLATE_COLORS = (
    (28, 35, 52),
    (49, 36, 62),
    (21, 55, 50),
    (56, 43, 27),
)
SIGN_COLORS = (
    (196, 45, 52),
    (33, 104, 190),
    (135, 62, 170),
    (208, 112, 25),
)
BACKGROUNDS = (
    (250, 249, 246),
    (246, 250, 252),
    (251, 247, 252),
)
WIRE_COLOR = (170, 176, 184)


def sample_scene(seed: int) -> dict:
    """Sample one scene; consecutive even/odd seeds form an exact style match."""

    pair_seed = int(seed) // 2
    rng = random.Random(0xB477E9 ^ (pair_seed * 1_000_003))
    width, height = 384, 320
    center_x = rng.randint(180, 204)
    center_y = rng.randint(188, 202)
    plate_gap = rng.randrange(76, 105, 4)
    long_length = rng.randrange(136, 165, 4)
    short_length = rng.randrange(72, 101, 4)
    plate_width = rng.choice((12, 14, 16))
    long_slot = rng.randrange(2)
    # Even = conventional long/plus; odd = matched polarity swap.
    plus_slot = long_slot if int(seed) % 2 == 0 else 1 - long_slot
    sign_size = rng.choice((28, 30, 32, 34))
    sign_y = rng.randint(53, 66)
    order = [0, 1]
    if rng.randrange(2):
        order.reverse()
    return {
        "width": width,
        "height": height,
        "center_x": center_x,
        "center_y": center_y,
        "plate_gap": plate_gap,
        "long_length": long_length,
        "short_length": short_length,
        "plate_width": plate_width,
        "long_slot": long_slot,
        "plus_slot": plus_slot,
        "sign_y": sign_y,
        "sign_size": sign_size,
        "wire_y": center_y,
        "plate_color_index": rng.randrange(len(PLATE_COLORS)),
        "sign_color_index": rng.randrange(len(SIGN_COLORS)),
        "background_index": rng.randrange(len(BACKGROUNDS)),
        "plate_draw_order": order,
    }


def _slot_x(scene: dict, slot: int) -> float:
    return float(scene["center_x"]) + (-0.5 if slot == 0 else 0.5) * float(
        scene["plate_gap"]
    )


def _plate_length(scene: dict, slot: int) -> float:
    return float(scene["long_length"] if slot == int(scene["long_slot"]) else scene["short_length"])


def _rect(draw: ImageDraw.ImageDraw, box: tuple[float, float, float, float], fill, ss: int) -> None:
    draw.rectangle(tuple(int(round(value * ss)) for value in box), fill=fill)


def render(scene: dict) -> Image.Image:
    """Render a clean cell diagram as a final RGB raster."""

    ss = 4
    width = int(scene["width"])
    height = int(scene["height"])
    background = BACKGROUNDS[int(scene["background_index"])]
    plate_color = PLATE_COLORS[int(scene["plate_color_index"])]
    sign_color = SIGN_COLORS[int(scene["sign_color_index"])]
    image = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(image)

    x0 = _slot_x(scene, 0)
    x1 = _slot_x(scene, 1)
    wire_y = float(scene["wire_y"])
    wire_width = 5
    # Pale terminal wires make the object read as a circuit cell, but the
    # question explicitly excludes them and the inverse arm segments them out.
    draw.line(
        [(34 * ss, int(round(wire_y * ss))), (int(round(x0 * ss)), int(round(wire_y * ss)))],
        fill=WIRE_COLOR,
        width=wire_width * ss,
    )
    draw.line(
        [(int(round(x1 * ss)), int(round(wire_y * ss))), ((width - 34) * ss, int(round(wire_y * ss)))],
        fill=WIRE_COLOR,
        width=wire_width * ss,
    )

    plate_width = float(scene["plate_width"])
    center_y = float(scene["center_y"])
    for slot in scene["plate_draw_order"]:
        x = _slot_x(scene, int(slot))
        length = _plate_length(scene, int(slot))
        _rect(
            draw,
            (
                x - plate_width / 2,
                center_y - length / 2,
                x + plate_width / 2,
                center_y + length / 2,
            ),
            plate_color,
            ss,
        )

    sign_y = float(scene["sign_y"])
    sign_size = float(scene["sign_size"])
    thickness = max(6.0, round(sign_size / 5.0))
    plus_slot = int(scene["plus_slot"])
    for slot in (0, 1):
        x = _slot_x(scene, slot)
        _rect(
            draw,
            (
                x - sign_size / 2,
                sign_y - thickness / 2,
                x + sign_size / 2,
                sign_y + thickness / 2,
            ),
            sign_color,
            ss,
        )
        if slot == plus_slot:
            _rect(
                draw,
                (
                    x - thickness / 2,
                    sign_y - sign_size / 2,
                    x + thickness / 2,
                    sign_y + sign_size / 2,
                ),
                sign_color,
                ss,
            )

    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    """Return the glyph visibly aligned with the longer plate."""

    return "plus" if int(scene["plus_slot"]) == int(scene["long_slot"]) else "minus"


def margin(scene: dict) -> float:
    """Runner-up minus winner horizontal distance to the long-plate center."""

    long_x = _slot_x(scene, int(scene["long_slot"]))
    plus_x = _slot_x(scene, int(scene["plus_slot"]))
    minus_x = _slot_x(scene, 1 - int(scene["plus_slot"]))
    distances = sorted((abs(plus_x - long_x), abs(minus_x - long_x)))
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    length_gap = float(scene["long_length"]) - float(scene["short_length"])
    return length_gap < 24.0 or margin(scene) < 24.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Reverse the draw order of disjoint plates; pixels and gold are unchanged."""

    twin = copy.deepcopy(scene)
    twin["plate_draw_order"] = list(reversed(twin["plate_draw_order"]))
    return [("reverse_nonoverlapping_plate_draw_order", twin)]
