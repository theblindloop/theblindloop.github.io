"""Independent fixed-layout PNG inverse for path-specific splicing."""

from __future__ import annotations

import numpy as np
from PIL import Image

BACKGROUND = (247, 245, 238)
ONE = (30, 55, 78)
ZERO = (229, 232, 226)
XOR = (126, 73, 163)
AND = (28, 126, 125)
MED_X = (12, 249, 486, 723)
D_X = (80, 560)
Y_X = (15, 335, 655)
DECISIONS = ("path-A", "path-B", "path-C", "path-D")


def _near(pixel, color, tolerance: int = 4) -> bool:
    return bool(np.max(np.abs(pixel.astype(np.int16) - np.array(color, dtype=np.int16))) <= tolerance)


def _bit(arr, center: tuple[int, int], offset: int) -> int | None:
    pixel = arr[center[1], center[0] + offset]
    if _near(pixel, ONE):
        return 1
    if _near(pixel, ZERO):
        return 0
    return None


def _op(arr, box_left: int, box_top: int) -> str | None:
    pixel = arr[box_top + 6, box_left + 6]
    if _near(pixel, XOR):
        return "XOR"
    if _near(pixel, AND):
        return "AND"
    return None


def _apply(op: str, a: int, b: int) -> int:
    return int(bool(a) != bool(b)) if op == "XOR" else int(bool(a) and bool(b))


def _outputs(mediators: list[int], downstream_ops: list[str]) -> list[int]:
    d1 = _apply(downstream_ops[0], mediators[0], mediators[1])
    d2 = _apply(downstream_ops[1], mediators[2], mediators[3])
    return [
        _apply(downstream_ops[2], d1, d2),
        _apply(downstream_ops[3], d1, mediators[2]),
        _apply(downstream_ops[4], d2, mediators[1]),
    ]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if arr.shape != (720, 960, 3) or not _near(arr[715, 955], BACKGROUND):
        return "abstain"
    source = _bit(arr, (205, 72), 16)
    test_source = _bit(arr, (430, 72), 16)
    if source is None or test_source != 1 - source:
        return "abstain"
    contexts: list[int] = []
    mediator_ops: list[str] = []
    shown_actual: list[int] = []
    for x in MED_X:
        context = _bit(arr, (x + 190, 196), 14)
        op = _op(arr, x + 55, 176)
        actual = _bit(arr, (x + 112, 277), 15)
        if context is None or op is None or actual is None:
            return "abstain"
        contexts.append(context)
        mediator_ops.append(op)
        shown_actual.append(actual)
    expected_actual = [_apply(op, source, context) for op, context in zip(mediator_ops, contexts)]
    if shown_actual != expected_actual:
        return "abstain"
    downstream_ops = [_op(arr, x + 126, 393) for x in D_X]
    downstream_ops.extend(_op(arr, x + 110, 536) for x in Y_X)
    if any(op is None for op in downstream_ops):
        return "abstain"
    target = [_bit(arr, (x + 145, 660), 17) for x in Y_X]
    if any(value is None for value in target):
        return "abstain"
    test = [_apply(op, test_source, context) for op, context in zip(mediator_ops, contexts)]
    matches = []
    for index in range(4):
        hybrid = shown_actual.copy()
        hybrid[index] = test[index]
        if _outputs(hybrid, downstream_ops) == target:
            matches.append(index)
    return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
