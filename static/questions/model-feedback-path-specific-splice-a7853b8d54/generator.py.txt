"""Sampler and renderer for nested path-specific counterfactual splicing."""

from __future__ import annotations

import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

SIZE = (960, 720)
BACKGROUND = (247, 245, 238)
INK = (35, 42, 52)
CARD = (255, 255, 252)
BORDER = (105, 113, 124)
ONE = (30, 55, 78)
ZERO = (229, 232, 226)
XOR = (126, 73, 163)
AND = (28, 126, 125)
ACTUAL = (35, 137, 190)
TEST = (238, 163, 38)
PATH_COLORS = ((176, 62, 78), (53, 132, 90), (94, 76, 171), (183, 102, 35))
MED_X = (12, 249, 486, 723)
D_X = (80, 560)
Y_X = (15, 335, 655)
DECISIONS = ("path-A", "path-B", "path-C", "path-D")


def _font(size: int) -> ImageFont.ImageFont:
    path = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")
    return ImageFont.truetype(str(path), size=size) if path.exists() else ImageFont.load_default()


def _apply(op: str, a: int, b: int) -> int:
    if op == "XOR":
        return int(bool(a) != bool(b))
    if op == "AND":
        return int(bool(a) and bool(b))
    raise ValueError(op)


def _evaluate(source: int, contexts: list[int], mediator_ops: list[str],
              downstream_ops: list[str], mediator_values: list[int] | None = None) -> tuple[list[int], list[int]]:
    mediators = ([_apply(op, source, context) for op, context in zip(mediator_ops, contexts)]
                 if mediator_values is None else [int(v) for v in mediator_values])
    d1 = _apply(downstream_ops[0], mediators[0], mediators[1])
    d2 = _apply(downstream_ops[1], mediators[2], mediators[3])
    outputs = [
        _apply(downstream_ops[2], d1, d2),
        _apply(downstream_ops[3], d1, mediators[2]),
        _apply(downstream_ops[4], d2, mediators[1]),
    ]
    return mediators, outputs


def _candidate_outputs(source: int, contexts: list[int], mediator_ops: list[str],
                       downstream_ops: list[str]) -> tuple[list[int], list[int], list[list[int]]]:
    actual, _actual_outputs = _evaluate(source, contexts, mediator_ops, downstream_ops)
    test, _test_outputs = _evaluate(1 - source, contexts, mediator_ops, downstream_ops)
    worlds: list[list[int]] = []
    for index in range(4):
        hybrid = actual.copy()
        hybrid[index] = test[index]
        worlds.append(_evaluate(source, contexts, mediator_ops, downstream_ops, hybrid)[1])
    return actual, test, worlds


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x53504C49)
    wanted = int(seed) % 4
    for _ in range(300000):
        source = rng.randrange(2)
        contexts = [rng.randrange(2) for _ in range(4)]
        mediator_ops = [rng.choice(("XOR", "AND")) for _ in range(4)]
        downstream_ops = [rng.choice(("XOR", "AND")) for _ in range(5)]
        actual, test, worlds = _candidate_outputs(source, contexts, mediator_ops, downstream_ops)
        target = worlds[wanted]
        actual_outputs = _evaluate(source, contexts, mediator_ops, downstream_ops, actual)[1]
        test_outputs = _evaluate(1 - source, contexts, mediator_ops, downstream_ops, test)[1]
        if sum(a != b for a, b in zip(actual, test)) < 3:
            continue
        if target in (actual_outputs, test_outputs) or sum(target) not in (1, 2):
            continue
        if [i for i, out in enumerate(worlds) if out == target] != [wanted]:
            continue
        reverse_worlds = []
        for index in range(4):
            reverse = test.copy()
            reverse[index] = actual[index]
            reverse_worlds.append(_evaluate(source, contexts, mediator_ops, downstream_ops, reverse)[1])
        reverse_hits = [i for i, out in enumerate(reverse_worlds) if out == target]
        if len(reverse_hits) != 1 or reverse_hits == [wanted]:
            continue
        return {
            "source_bit": source,
            "contexts": contexts,
            "mediator_ops": mediator_ops,
            "downstream_ops": downstream_ops,
            "target_outputs": target,
        }
    raise RuntimeError("could not sample balanced path-specific splice")


def _validate(scene: dict) -> tuple[int, list[int], list[str], list[str], list[int]]:
    source = int(scene["source_bit"])
    contexts = [int(v) for v in scene["contexts"]]
    mediator_ops = [str(v) for v in scene["mediator_ops"]]
    downstream_ops = [str(v) for v in scene["downstream_ops"]]
    target = [int(v) for v in scene["target_outputs"]]
    if source not in (0, 1) or len(contexts) != 4 or any(v not in (0, 1) for v in contexts):
        raise ValueError("invalid source or context bits")
    if len(mediator_ops) != 4 or len(downstream_ops) != 5:
        raise ValueError("wrong operation count")
    if any(op not in ("XOR", "AND") for op in mediator_ops + downstream_ops):
        raise ValueError("invalid operation")
    if len(target) != 3 or any(v not in (0, 1) for v in target):
        raise ValueError("invalid target")
    return source, contexts, mediator_ops, downstream_ops, target


def analytic_gold(scene: dict) -> str:
    try:
        source, contexts, mediator_ops, downstream_ops, target = _validate(scene)
        _actual, _test, worlds = _candidate_outputs(source, contexts, mediator_ops, downstream_ops)
        matches = [i for i, outputs in enumerate(worlds) if outputs == target]
        return DECISIONS[matches[0]] if len(matches) == 1 else "abstain"
    except (KeyError, TypeError, ValueError, IndexError):
        return "abstain"


def margin(scene: dict) -> float:
    return 1.0 if analytic_gold(scene) in DECISIONS else 0.0


def is_quarantined(scene: dict) -> bool:
    return analytic_gold(scene) == "abstain"


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _centered(draw: ImageDraw.ImageDraw, xy: tuple[int, int], text: str,
              font: ImageFont.ImageFont, fill) -> None:
    box = draw.textbbox((0, 0), text, font=font)
    draw.text((xy[0] - (box[2] - box[0]) / 2,
               xy[1] - (box[3] - box[1]) / 2 - box[1]), text, font=font, fill=fill)


def _lamp(draw: ImageDraw.ImageDraw, center: tuple[int, int], value: int, radius: int = 21) -> None:
    x, y = center
    draw.ellipse((x - radius, y - radius, x + radius, y + radius),
                 fill=ONE if value else ZERO, outline=INK, width=3)
    _centered(draw, center, str(value), _font(19), (255, 255, 255) if value else INK)


def _operation(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], op: str) -> None:
    draw.rounded_rectangle(box, radius=8, fill=XOR if op == "XOR" else AND)
    _centered(draw, ((box[0] + box[2]) // 2, (box[1] + box[3]) // 2),
              op, _font(16), (255, 255, 255))


def render(scene: dict) -> Image.Image:
    source, contexts, mediator_ops, downstream_ops, target = _validate(scene)
    actual, _actual_outputs = _evaluate(source, contexts, mediator_ops, downstream_ops)
    image = Image.new("RGB", SIZE, BACKGROUND)
    draw = ImageDraw.Draw(image)
    _centered(draw, (480, 24), "PATH-SPECIFIC TEST", _font(24), INK)
    draw.text((74, 52), "ACTUAL X", font=_font(17), fill=ACTUAL)
    _lamp(draw, (205, 72), source)
    draw.text((286, 52), "TEST flips X", font=_font(17), fill=TEST)
    _lamp(draw, (430, 72), 1 - source)
    draw.rounded_rectangle((510, 45, 944, 100), radius=10, fill=CARD, outline=BORDER, width=2)
    _centered(draw, (727, 63), "Try one PATH at a time", _font(17), INK)
    _centered(draw, (727, 84), "chosen M uses TEST; other M stay ACTUAL", _font(15), INK)
    for i, x in enumerate(MED_X):
        draw.rounded_rectangle((x, 120, x + 225, 310), radius=12, fill=CARD, outline=BORDER, width=3)
        draw.rounded_rectangle((x + 10, 130, x + 78, 164), radius=8, fill=PATH_COLORS[i])
        _centered(draw, (x + 44, 147), f"PATH {chr(65 + i)}", _font(14), (255, 255, 255))
        draw.text((x + 91, 134), f"MEDIATOR M{i + 1}", font=_font(15), fill=INK)
        draw.text((x + 20, 184), "X", font=_font(22), fill=INK)
        _operation(draw, (x + 55, 176, x + 121, 216), mediator_ops[i])
        draw.text((x + 135, 184), f"C{i + 1}", font=_font(20), fill=INK)
        _lamp(draw, (x + 190, 196), contexts[i], 19)
        _centered(draw, (x + 112, 240), "ACTUAL result", _font(14), ACTUAL)
        _lamp(draw, (x + 112, 277), actual[i], 20)
    _centered(draw, (480, 335), "DOWNSTREAM (recompute after each one-path splice)", _font(18), INK)
    d_specs = (("D1", "M1", "M2"), ("D2", "M3", "M4"))
    for i, (name, left, right) in enumerate(d_specs):
        x = D_X[i]
        draw.rounded_rectangle((x, 352, x + 320, 465), radius=12, fill=CARD, outline=BORDER, width=3)
        draw.text((x + 16, 366), name, font=_font(20), fill=INK)
        draw.text((x + 68, 405), left, font=_font(20), fill=INK)
        _operation(draw, (x + 126, 393, x + 194, 433), downstream_ops[i])
        draw.text((x + 220, 405), right, font=_font(20), fill=INK)
    y_specs = (("Y1", "D1", "D2"), ("Y2", "D1", "M3"), ("Y3", "D2", "M2"))
    for i, (name, left, right) in enumerate(y_specs):
        x = Y_X[i]
        draw.rounded_rectangle((x, 482, x + 290, 700), radius=12, fill=CARD, outline=BORDER, width=3)
        _centered(draw, (x + 145, 506), name, _font(20), INK)
        draw.text((x + 42, 548), left, font=_font(20), fill=INK)
        _operation(draw, (x + 110, 536, x + 180, 576), downstream_ops[2 + i])
        draw.text((x + 207, 548), right, font=_font(20), fill=INK)
        _centered(draw, (x + 145, 610), "TARGET", _font(16), TEST)
        _lamp(draw, (x + 145, 660), target[i], 23)
    return image
