"""Corner-mark integral world.

One 11x11 ruled panel carries a handful of dots. The dots are not the symbol:
they are the two-dimensional parity derivative of a hidden cell pattern, taken
with respect to the panel corner that the border visibly thickens. Accumulating
the dots back -- a cell is inked when the block running from the marked corner
to that cell holds an odd number of dots -- restores the hidden pattern, which
is matched against a closed set of five named symbols.

Perturbing cell flips are chosen so that every scene shows exactly the same
number of dots whatever the answer, so dot count and total ink carry no class
information, and so that a declared share of scenes sits near the decision
boundary.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

GRID = 11
NCELL = GRID * GRID

SYMBOLS: tuple[str, ...] = ("arch", "bars", "ring", "tee", "zed")

SIZES: tuple[int, ...] = (8, 9, 10, 11)
THICKNESSES: tuple[int, ...] = (2, 3)
CORNERS: tuple[str, ...] = ("tl", "tr", "bl", "br")

CELL = 30
BORDER = 5
PAD = 24
DOT_INSET = 6

TARGET_DOTS = 8
QUARANTINE_MARGIN = 2.0
CLEAR_MARGIN = 4.0
BOUNDARY_PERIOD = 6
SEARCH_TRIES = 4000

BACKGROUNDS = [(247, 244, 237), (239, 243, 248), (244, 239, 245), (237, 246, 240)]
FRAMES = [(36, 42, 56), (58, 38, 32), (28, 54, 48), (50, 34, 62)]
RULES = [(206, 203, 196), (200, 206, 213), (210, 202, 211), (199, 210, 203)]
INKS = [(22, 60, 126), (130, 32, 42), (24, 90, 58), (92, 46, 118), (38, 38, 42)]


# --------------------------------------------------------------------------
# the five named symbols, as filled cell sets inside an r0/c0/h/w box
# --------------------------------------------------------------------------
def symbol_cells(name: str, r0: int, c0: int, h: int, w: int, t: int) -> set[tuple[int, int]]:
    cells: set[tuple[int, int]] = set()

    def block(ra: int, rb: int, ca: int, cb: int) -> None:
        for r in range(ra, rb):
            for c in range(ca, cb):
                cells.add((r, c))

    if name == "ring":
        block(r0, r0 + h, c0, c0 + w)
        for r in range(r0 + t, r0 + h - t):
            for c in range(c0 + t, c0 + w - t):
                cells.discard((r, c))
    elif name == "tee":
        block(r0, r0 + t, c0, c0 + w)
        cs = c0 + (w - t) // 2
        block(r0, r0 + h, cs, cs + t)
    elif name == "arch":
        block(r0, r0 + h, c0, c0 + t)
        block(r0, r0 + h, c0 + w - t, c0 + w)
        block(r0 + h - t, r0 + h, c0, c0 + w)
    elif name == "bars":
        block(r0, r0 + t, c0, c0 + w)
        block(r0 + h - t, r0 + h, c0, c0 + w)
    elif name == "zed":
        h1 = h // 2
        block(r0, r0 + h1, c0, c0 + w - t)
        block(r0 + h1, r0 + h, c0 + t, c0 + w)
    else:
        raise ValueError(f"unknown symbol {name!r}")
    return cells


def _mask(cells: set[tuple[int, int]]) -> int:
    bits = 0
    for r, c in cells:
        bits |= 1 << (r * GRID + c)
    return bits


def placements(name: str) -> list[int]:
    """Every admissible drawing of one named symbol on the lattice."""
    out: list[int] = []
    for h in SIZES:
        for w in SIZES:
            for t in THICKNESSES:
                for r0 in range(GRID - h + 1):
                    for c0 in range(GRID - w + 1):
                        out.append(_mask(symbol_cells(name, r0, c0, h, w, t)))
    return out


_FAMILIES: dict[str, list[int]] = {name: placements(name) for name in SYMBOLS}


# --------------------------------------------------------------------------
# the declared transform and its inverse
# --------------------------------------------------------------------------
def _to_grid(bits: int, corner: str) -> list[list[int]]:
    grid = [[(bits >> (r * GRID + c)) & 1 for c in range(GRID)] for r in range(GRID)]
    if corner in ("bl", "br"):
        grid = grid[::-1]
    if corner in ("tr", "br"):
        grid = [row[::-1] for row in grid]
    return grid


def _from_grid(grid: list[list[int]], corner: str) -> int:
    if corner in ("tr", "br"):
        grid = [row[::-1] for row in grid]
    if corner in ("bl", "br"):
        grid = grid[::-1]
    bits = 0
    for r in range(GRID):
        for c in range(GRID):
            if grid[r][c]:
                bits |= 1 << (r * GRID + c)
    return bits


def parity_derivative(pattern: int, corner: str) -> int:
    """Dots whose corner-anchored running parity reproduces ``pattern``."""
    src = _to_grid(pattern, corner)
    out = [[0] * GRID for _ in range(GRID)]
    for r in range(GRID):
        for c in range(GRID):
            value = src[r][c]
            if r:
                value ^= src[r - 1][c]
            if c:
                value ^= src[r][c - 1]
            if r and c:
                value ^= src[r - 1][c - 1]
            out[r][c] = value
    return _from_grid(out, corner)


def running_parity(dots: int, corner: str) -> int:
    """The declared transform: odd dot count in the block from the marked corner."""
    src = _to_grid(dots, corner)
    out = [[0] * GRID for _ in range(GRID)]
    for r in range(GRID):
        for c in range(GRID):
            value = src[r][c]
            if r:
                value ^= out[r - 1][c]
            if c:
                value ^= out[r][c - 1]
            if r and c:
                value ^= out[r - 1][c - 1]
            out[r][c] = value
    return _from_grid(out, corner)


# --------------------------------------------------------------------------
# scene sampling and analytic gold
# --------------------------------------------------------------------------
def _apply(base: int, flips: list[int]) -> int:
    bits = base
    for index in flips:
        bits ^= 1 << int(index)
    return bits


def _scored_mask(pattern: int) -> list[tuple[float, str]]:
    scored = [
        (float(min(bin(pattern ^ drawing).count("1") for drawing in _FAMILIES[name])), name)
        for name in SYMBOLS
    ]
    scored.sort(key=lambda item: (item[0], SYMBOLS.index(item[1])))
    return scored


def _choose_flips(rng: random.Random, base: int, corner: str, near_boundary: bool) -> list[int]:
    """Flips that hold the drawn dot count fixed and land in the declared band.

    The dot count is pinned to ``TARGET_DOTS`` for every scene and every answer,
    so the printed evidence cannot be classified by counting ink.
    """
    fallback: list[int] | None = None
    tightest: tuple[float, list[int]] | None = None
    for attempt in range(SEARCH_TRIES):
        size = rng.choice([1, 2, 2, 3, 4, 5]) if attempt else 0
        flips = sorted(rng.sample(range(NCELL), size))
        pattern = _apply(base, flips)
        dots = parity_derivative(pattern, corner)
        if bin(dots).count("1") != TARGET_DOTS:
            continue
        scored = _scored_mask(pattern)
        gap = scored[1][0] - scored[0][0]
        if fallback is None:
            fallback = flips
        if near_boundary:
            if tightest is None or gap < tightest[0]:
                tightest = (gap, flips)
            if 0.0 < gap <= QUARANTINE_MARGIN:
                return flips
        elif gap >= CLEAR_MARGIN:
            return flips
    if near_boundary and tightest is not None:
        return tightest[1]
    return fallback if fallback is not None else []


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    h = rng.choice(SIZES)
    w = rng.choice(SIZES)
    t = rng.choice(THICKNESSES)
    r0 = rng.randrange(GRID - h + 1)
    c0 = rng.randrange(GRID - w + 1)
    corner = CORNERS[rng.randrange(len(CORNERS))]
    base = _mask(symbol_cells(symbol, r0, c0, h, w, t))
    flips = _choose_flips(rng, base, corner, near_boundary=(seed % BOUNDARY_PERIOD == 5))
    return {
        "symbol": symbol,
        "r0": r0,
        "c0": c0,
        "h": h,
        "w": w,
        "t": t,
        "flips": flips,
        "corner": corner,
        "bg": list(BACKGROUNDS[rng.randrange(len(BACKGROUNDS))]),
        "frame": list(FRAMES[rng.randrange(len(FRAMES))]),
        "rule": list(RULES[rng.randrange(len(RULES))]),
        "ink": list(INKS[rng.randrange(len(INKS))]),
        "cell": CELL,
        "border": BORDER,
        "pad": PAD,
    }


def revealed_pattern(scene: dict) -> int:
    """The cell pattern the declared running-parity transform restores."""
    base = _mask(
        symbol_cells(
            scene["symbol"], int(scene["r0"]), int(scene["c0"]),
            int(scene["h"]), int(scene["w"]), int(scene["t"]),
        )
    )
    return _apply(base, list(scene["flips"]))


def dot_pattern(scene: dict) -> int:
    return parity_derivative(revealed_pattern(scene), scene["corner"])


def analytic_gold(scene: dict) -> str:
    return _scored_mask(revealed_pattern(scene))[0][1]


def margin(scene: dict) -> float:
    scored = _scored_mask(revealed_pattern(scene))
    return float(scored[1][0] - scored[0][0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_MARGIN


def dot_count(scene: dict) -> int:
    return bin(dot_pattern(scene)).count("1")


def naive_read_distance(scene: dict) -> int:
    """Cell mismatch between the drawn dots and the nearest symbol drawing.

    A small value would mean the printed evidence already looks like the answer
    without performing the declared accumulation.
    """
    return int(min(score for score, _ in _scored_mask(dot_pattern(scene))))


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Declared pixel-identical, gold-preserving latent aliases."""
    reordered = dict(scene)
    reordered["flips"] = list(reversed(scene["flips"]))
    cancelled = dict(scene)
    cancelled["flips"] = list(scene["flips"]) + [5, 5]
    return [
        ("flips_reordered", reordered),
        ("flip_pair_cancellation", cancelled),
    ]


# --------------------------------------------------------------------------
# raster
# --------------------------------------------------------------------------
def render(scene: dict) -> Image.Image:
    cell = int(scene["cell"])
    border = int(scene["border"])
    pad = int(scene["pad"])
    panel = GRID * cell + 2 * border
    side = 2 * pad + panel
    image = Image.new("RGB", (side, side), tuple(scene["bg"]))
    draw = ImageDraw.Draw(image)
    frame = tuple(scene["frame"])
    x0 = y0 = pad
    x1 = y1 = pad + panel - 1
    inner = x0 + border
    for index in range(1, GRID):
        offset = inner + index * cell
        draw.line([(offset, inner), (offset, inner + GRID * cell - 1)], fill=tuple(scene["rule"]))
        draw.line([(inner, offset), (inner + GRID * cell - 1, offset)], fill=tuple(scene["rule"]))
    draw.rectangle([x0, y0, x1, y1], outline=frame, width=border)
    mark = 3 * border
    corner = scene["corner"]
    mx = x0 if corner in ("tl", "bl") else x1 - mark + 1
    my = y0 if corner in ("tl", "tr") else y1 - mark + 1
    draw.rectangle([mx, my, mx + mark - 1, my + mark - 1], fill=frame)
    dots = dot_pattern(scene)
    ink = tuple(scene["ink"])
    for r in range(GRID):
        for c in range(GRID):
            if not (dots >> (r * GRID + c)) & 1:
                continue
            cx = inner + c * cell
            cy = inner + r * cell
            draw.ellipse(
                [cx + DOT_INSET, cy + DOT_INSET, cx + cell - 1 - DOT_INSET, cy + cell - 1 - DOT_INSET],
                fill=ink,
            )
    return image
