"""Pixel-only inverse arm for the corner-mark integral world.

Written against the public question contract alone: an 11 by 11 ruled panel
inside a border, one border corner thickened, some cells carrying a dot, and
the declared rule that a cell of the revealed pattern is inked when the block
running from the thickened corner cell to that cell holds an odd number of
dots. The revealed pattern is then matched against the five named symbols.

This module receives a PIL image and nothing else. It imports no renderer,
prompt, generator, verifier or gold code, and it builds its own symbol
templates from the wording of the question.
"""

from __future__ import annotations

from collections import Counter

from PIL import Image

SIDE = 11
NCELL = SIDE * SIDE
NAMES: tuple[str, ...] = ("arch", "bars", "ring", "tee", "zed")
BOX_SIDES: tuple[int, ...] = (8, 9, 10, 11)
STROKES: tuple[int, ...] = (2, 3)
ABSTAIN = "abstain"

_INK_DISTANCE = 120.0
_CORNER_RATIO = 1.25


# ---------------------------------------------------------------------------
# symbol templates, transcribed from the question's five names
# ---------------------------------------------------------------------------
def _template(name: str, top: int, left: int, height: int, width: int, stroke: int) -> int:
    """One drawing of a named symbol, as a bitmask over the 11x11 lattice."""
    bits = 0
    for row in range(SIDE):
        for col in range(SIDE):
            dr = row - top
            dc = col - left
            if not (0 <= dr < height and 0 <= dc < width):
                continue
            near_top = dr < stroke
            near_bottom = dr >= height - stroke
            near_left = dc < stroke
            near_right = dc >= width - stroke
            if name == "ring":
                on = near_top or near_bottom or near_left or near_right
            elif name == "bars":
                on = near_top or near_bottom
            elif name == "arch":
                on = near_left or near_right or near_bottom
            elif name == "tee":
                stem = (width - stroke) // 2
                on = near_top or (stem <= dc < stem + stroke)
            elif name == "zed":
                half = height // 2
                on = (dr < half and dc < width - stroke) or (dr >= half and dc >= stroke)
            else:
                raise ValueError(name)
            if on:
                bits |= 1 << (row * SIDE + col)
    return bits


def _families() -> dict[str, list[int]]:
    table: dict[str, list[int]] = {}
    for name in NAMES:
        drawings: list[int] = []
        for height in BOX_SIDES:
            for width in BOX_SIDES:
                for stroke in STROKES:
                    for top in range(SIDE - height + 1):
                        for left in range(SIDE - width + 1):
                            drawings.append(_template(name, top, left, height, width, stroke))
        table[name] = drawings
    return table


_TEMPLATES = _families()


# ---------------------------------------------------------------------------
# raster reading
# ---------------------------------------------------------------------------
def _distance(a: tuple[int, int, int], b: tuple[int, int, int]) -> float:
    return ((a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2 + (a[2] - b[2]) ** 2) ** 0.5


def _read_panel(image: Image.Image):
    """Locate the border, its thickness, and the thickened corner."""
    rgb = image.convert("RGB")
    width, height = rgb.size
    pixels = rgb.load()
    background = Counter(rgb.getdata()).most_common(1)[0][0]

    xs: list[int] = []
    ys: list[int] = []
    for y in range(height):
        for x in range(width):
            if _distance(pixels[x, y], background) > 30:
                xs.append(x)
                ys.append(y)
    if not xs:
        return None
    x0, x1, y0, y1 = min(xs), max(xs), min(ys), max(ys)
    if x1 - x0 < 40 or y1 - y0 < 40:
        return None

    mid = (y0 + y1) // 2
    border_colour = pixels[x0, mid]
    thickness = 0
    while thickness < 40 and _distance(pixels[x0 + thickness, mid], border_colour) <= 40:
        thickness += 1
    if thickness < 2:
        return None

    span_x = (x1 - thickness) - (x0 + thickness) + 1
    span_y = (y1 - thickness) - (y0 + thickness) + 1
    if span_x < SIDE or span_y < SIDE:
        return None

    window = 3 * thickness
    scores: dict[str, int] = {}
    for tag, (cx, cy) in {
        "tl": (x0, y0),
        "tr": (x1 - window + 1, y0),
        "bl": (x0, y1 - window + 1),
        "br": (x1 - window + 1, y1 - window + 1),
    }.items():
        count = 0
        for y in range(cy, cy + window):
            for x in range(cx, cx + window):
                if 0 <= x < width and 0 <= y < height:
                    if _distance(pixels[x, y], border_colour) <= 40:
                        count += 1
        scores[tag] = count
    ranked = sorted(scores.items(), key=lambda item: (-item[1], item[0]))
    if ranked[1][1] == 0 or ranked[0][1] < _CORNER_RATIO * ranked[1][1]:
        return None
    corner = ranked[0][0]

    dots = 0
    for row in range(SIDE):
        for col in range(SIDE):
            ax = x0 + thickness + (col * span_x) // SIDE
            bx = x0 + thickness + ((col + 1) * span_x) // SIDE
            ay = y0 + thickness + (row * span_y) // SIDE
            by = y0 + thickness + ((row + 1) * span_y) // SIDE
            lo_x = ax + (bx - ax) * 3 // 10
            hi_x = bx - (bx - ax) * 3 // 10
            lo_y = ay + (by - ay) * 3 // 10
            hi_y = by - (by - ay) * 3 // 10
            hit = total = 0
            for y in range(lo_y, hi_y):
                for x in range(lo_x, hi_x):
                    total += 1
                    if _distance(pixels[x, y], background) > _INK_DISTANCE:
                        hit += 1
            if total and hit * 2 > total:
                dots |= 1 << (row * SIDE + col)
    return dots, corner


# ---------------------------------------------------------------------------
# declared transform and decision
# ---------------------------------------------------------------------------
def _cells(bits: int) -> list[list[int]]:
    return [[(bits >> (r * SIDE + c)) & 1 for c in range(SIDE)] for r in range(SIDE)]


def _pack(cells: list[list[int]]) -> int:
    bits = 0
    for r in range(SIDE):
        for c in range(SIDE):
            if cells[r][c]:
                bits |= 1 << (r * SIDE + c)
    return bits


def accumulate(dots: int, corner: str) -> int:
    """Ink a cell when the block from the marked corner holds an odd dot count."""
    grid = _cells(dots)
    flip_rows = corner in ("bl", "br")
    flip_cols = corner in ("tr", "br")
    if flip_rows:
        grid = grid[::-1]
    if flip_cols:
        grid = [row[::-1] for row in grid]
    out = [[0] * SIDE for _ in range(SIDE)]
    for r in range(SIDE):
        for c in range(SIDE):
            value = grid[r][c]
            if r:
                value ^= out[r - 1][c]
            if c:
                value ^= out[r][c - 1]
            if r and c:
                value ^= out[r - 1][c - 1]
            out[r][c] = value
    if flip_cols:
        out = [row[::-1] for row in out]
    if flip_rows:
        out = out[::-1]
    return _pack(out)


def _rank(pattern: int) -> list[tuple[int, str]]:
    scored = [
        (min(bin(pattern ^ drawing).count("1") for drawing in _TEMPLATES[name]), name)
        for name in NAMES
    ]
    scored.sort(key=lambda item: (item[0], NAMES.index(item[1])))
    return scored


def decide_with_margin(image: Image.Image) -> tuple[str, float]:
    read = _read_panel(image)
    if read is None:
        return ABSTAIN, 0.0
    dots, corner = read
    scored = _rank(accumulate(dots, corner))
    return scored[0][1], float(scored[1][0] - scored[0][0])


def decision_from_image(image: Image.Image) -> str:
    return decide_with_margin(image)[0]
