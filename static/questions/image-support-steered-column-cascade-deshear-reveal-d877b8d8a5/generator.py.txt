"""Column-cascade de-shear reveal: one compact panel, one staircase column shear.

A symbol lives on a 7x7 lattice. The renderer does not draw that lattice. It
draws the lattice after a staircase shear: the contents of lattice column j are
slid downward, cyclically, by j cells, where j counts from the panel's left
border. Reading the panel as drawn therefore shows a smeared cascade that
matches no declared symbol; the reader must execute the declared inverse
operation - slide column j back upward by j cells, wrapping - before any symbol
exists.

All decisive evidence lives inside one small bordered panel: a single connected,
solid, area-like region whose interior carries every cell of the sheared
lattice. The panel is placed uniformly at random on a much larger canvas, and
the small chaff marks scattered outside it are drawn independently of the symbol
and carry no evidence at all.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "column_cascade_deshear_reveal-renderer-0.1.0"

CANVAS = 560
CELLS = 7
FRAME_PX = 4
CELL_CHOICES = (18, 20, 22)
EDGE_MARGIN = 22
CHAFF_CLEARANCE = 26
CHAFF_MIN_PX = 7
CHAFF_MAX_PX = 15
CHAFF_GAP = 14

PAPER = (250, 250, 250)
BACKGROUND = (236, 236, 238)
TINTS = ((0, 0, 0), (0, 6, 12), (12, 6, 0), (5, 12, 0))

MIN_CONTRAST = 62.0
MAX_CONTRAST = 152.0
QUARANTINE_CLEARANCE = 34.0

GLYPHS = ("bars", "caret", "vee", "zigzag")
EDGE_BAR_GLYPHS = frozenset({"caret", "vee"})


def _template(cells: set[tuple[int, int]]) -> np.ndarray:
    grid = np.zeros((CELLS, CELLS), dtype=np.uint8)
    for row, col in cells:
        grid[row, col] = 1
    return grid


def _bars() -> set[tuple[int, int]]:
    return {(1, c) for c in range(CELLS)} | {(5, c) for c in range(CELLS)}


def _caret() -> set[tuple[int, int]]:
    return {(abs(c - 3), c) for c in range(CELLS)} | {(6, c) for c in range(CELLS)}


def _vee() -> set[tuple[int, int]]:
    return {(6 - abs(c - 3), c) for c in range(CELLS)} | {(0, c) for c in range(CELLS)}


def _zigzag() -> set[tuple[int, int]]:
    return {(3, c) for c in range(CELLS)} | {
        (1 if c % 2 == 0 else 5, c) for c in range(CELLS)
    }


TEMPLATES: dict[str, np.ndarray] = {
    "bars": _template(_bars()),
    "caret": _template(_caret()),
    "vee": _template(_vee()),
    "zigzag": _template(_zigzag()),
}

# Counterbalance guarantee. Every declared symbol inks exactly 14 of the 49
# lattice positions and exactly two of the seven cells of every single column.
# A per-column cyclic slide permutes cells inside their own column, so it
# preserves every column count: the rendered panel therefore shows exactly two
# dark cells in each of its seven columns, and 14 in total, whichever symbol is
# present. Total ink, the per-column ink profile - the one statistic that the
# declared operation leaves invariant - panel size, panel position and colour
# are all uninformative about the answer.
for _name, _grid in TEMPLATES.items():
    assert int(_grid.sum()) == 14, _name
    assert _grid.sum(axis=0).tolist() == [2] * CELLS, _name

MIN_GLYPH_SEPARATION = min(
    int(np.abs(TEMPLATES[a].astype(int) - TEMPLATES[b].astype(int)).sum())
    for i, a in enumerate(GLYPHS)
    for b in GLYPHS[i + 1:]
)
assert MIN_GLYPH_SEPARATION >= 12


def shear(grid: np.ndarray) -> np.ndarray:
    """Slide lattice column j downward, cyclically, by j cells."""

    out = np.zeros_like(grid)
    for col in range(CELLS):
        out[:, col] = np.roll(grid[:, col], col)
    return out


def unshear(grid: np.ndarray) -> np.ndarray:
    """The declared inverse: slide column j back upward by j cells."""

    out = np.zeros_like(grid)
    for col in range(CELLS):
        out[:, col] = np.roll(grid[:, col], -col)
    return out


# The sheared panel must not itself read as any declared symbol, otherwise the
# answer would be recoverable without executing the declared operation.
for _name, _grid in TEMPLATES.items():
    _drawn = shear(_grid)
    for _other, _cand in TEMPLATES.items():
        assert int(np.abs(_drawn.astype(int) - _cand.astype(int)).sum()) >= 8, (_name, _other)


def _luma(color: tuple[int, int, int]) -> float:
    return 0.299 * color[0] + 0.587 * color[1] + 0.114 * color[2]


def sample_scene(seed: int) -> dict:
    """Sample the symbol, the panel geometry and placement, tones, and chaff."""

    # Two-word seeding decorrelates the first draws of consecutive scene seeds,
    # which keeps the symbol classes balanced across a contiguous dataset.
    rng = np.random.default_rng([int(seed), 0x5C4D])
    glyph = GLYPHS[int(rng.integers(len(GLYPHS)))]
    cell_px = int(CELL_CHOICES[int(rng.integers(len(CELL_CHOICES)))])
    panel_px = CELLS * cell_px + 2 * FRAME_PX
    span = CANVAS - panel_px - 2 * EDGE_MARGIN
    top = int(EDGE_MARGIN + rng.integers(span + 1))
    left = int(EDGE_MARGIN + rng.integers(span + 1))
    # One scene in five is drawn from the band straddling the quarantine
    # clearance so that near-boundary and quarantined cases are always covered.
    if rng.random() < 0.2:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MIN_CONTRAST + 14.0), 3))
    else:
        contrast = float(np.round(rng.uniform(MIN_CONTRAST, MAX_CONTRAST), 3))
    tint = TINTS[int(rng.integers(len(TINTS)))]

    chaff: list[list[int]] = []
    wanted = 5 + int(rng.integers(7))
    for _ in range(400):
        if len(chaff) >= wanted:
            break
        size = int(CHAFF_MIN_PX + rng.integers(CHAFF_MAX_PX - CHAFF_MIN_PX + 1))
        cy = int(rng.integers(6, CANVAS - size - 6))
        cx = int(rng.integers(6, CANVAS - size - 6))
        if (
            cy + size + CHAFF_CLEARANCE > top
            and cy < top + panel_px + CHAFF_CLEARANCE
            and cx + size + CHAFF_CLEARANCE > left
            and cx < left + panel_px + CHAFF_CLEARANCE
        ):
            continue
        clash = False
        for oy, ox, osize in chaff:
            if (
                cy + size + CHAFF_GAP > oy
                and cy < oy + osize + CHAFF_GAP
                and cx + size + CHAFF_GAP > ox
                and cx < ox + osize + CHAFF_GAP
            ):
                clash = True
                break
        if clash:
            continue
        chaff.append([cy, cx, size])
    return {
        "canvas": CANVAS,
        "cells": CELLS,
        "cell_px": cell_px,
        "frame_px": FRAME_PX,
        "glyph": glyph,
        "top": top,
        "left": left,
        "contrast": contrast,
        "tint": list(tint),
        "chaff": chaff,
    }


def _ink_color(scene: dict) -> tuple[int, int, int]:
    base = int(round(_luma(PAPER) - float(scene["contrast"])))
    tint = tuple(int(value) for value in scene["tint"])
    return (
        max(0, min(255, base + tint[0])),
        max(0, min(255, base + tint[1])),
        max(0, min(255, base + tint[2])),
    )


def _clearance(scene: dict) -> float:
    """Half the rendered paper-to-ink luma gap: the per-cell decision clearance."""

    return 0.5 * (_luma(PAPER) - _luma(_ink_color(scene)))


def render(scene: dict) -> Image.Image:
    """Rasterise the scene with flat, hard-edged blocks and no antialiasing."""

    canvas = int(scene["canvas"])
    cell = int(scene["cell_px"])
    frame = int(scene["frame_px"])
    top = int(scene["top"])
    left = int(scene["left"])
    ink = np.array(_ink_color(scene), dtype=np.uint8)
    paper = np.array(PAPER, dtype=np.uint8)
    array = np.zeros((canvas, canvas, 3), dtype=np.uint8)
    array[:, :] = np.array(BACKGROUND, dtype=np.uint8)

    for cy, cx, size in scene["chaff"]:
        array[cy:cy + size, cx:cx + size] = ink

    panel_px = CELLS * cell + 2 * frame
    array[top:top + panel_px, left:left + panel_px] = ink
    array[top + frame:top + panel_px - frame, left + frame:left + panel_px - frame] = paper
    drawn = shear(TEMPLATES[scene["glyph"]])
    for row in range(CELLS):
        for col in range(CELLS):
            if not drawn[row, col]:
                continue
            y0 = top + frame + row * cell
            x0 = left + frame + col * cell
            array[y0:y0 + cell, x0:x0 + cell] = ink
    return Image.fromarray(array, mode="RGB")


def analytic_gold(scene: dict) -> str:
    """The symbol carried by the panel once the column cascade is undone."""

    return str(scene["glyph"])


def margin(scene: dict) -> float:
    """Per-cell luma clearance, the binding decision margin of the world."""

    return float(round(_clearance(scene), 6))


def is_quarantined(scene: dict) -> bool:
    """Flag scenes whose per-cell tone clearance is too small to read by eye."""

    return bool(_clearance(scene) < QUARANTINE_CLEARANCE)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No latent aliases exist: every sampled field is rendered injectively."""

    return []
