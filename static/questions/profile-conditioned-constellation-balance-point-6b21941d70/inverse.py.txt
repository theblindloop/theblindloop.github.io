"""Independent pixel-only inverse arm for constellation_balance_point."""

from __future__ import annotations

from collections import deque
import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (210, 52, 62),
    "blue": (42, 103, 184),
    "green": (36, 145, 91),
    "orange": (226, 126, 33),
}
DARK_RGB = np.array((28, 34, 43), dtype=np.float32)


def _components(mask: np.ndarray) -> list[np.ndarray]:
    work = mask.copy()
    height, width = work.shape
    output: list[np.ndarray] = []
    for sy, sx in np.argwhere(work):
        if not work[sy, sx]:
            continue
        work[sy, sx] = False
        queue = deque([(int(sy), int(sx))])
        pixels: list[tuple[float, float]] = []
        while queue:
            y, x = queue.popleft()
            pixels.append((x + 0.5, y + 0.5))
            for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
                if 0 <= ny < height and 0 <= nx < width and work[ny, nx]:
                    work[ny, nx] = False
                    queue.append((ny, nx))
        output.append(np.asarray(pixels, dtype=np.float64))
    return output


def _colored_ring_center(arr: np.ndarray, rgb: tuple[int, int, int]) -> np.ndarray:
    target = np.asarray(rgb, dtype=np.float32)
    mask = np.linalg.norm(arr - target, axis=2) < 28.0
    yy, xx = np.nonzero(mask)
    if not 130 <= len(xx) <= 430:
        raise ValueError("colored candidate ring is missing or malformed")
    return np.array([xx.mean() + 0.5, yy.mean() + 0.5], dtype=np.float64)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    if arr.ndim != 3 or arr.shape[0] < 64 or arr.shape[1] < 64:
        raise ValueError("image is too small")

    dark_mask = np.linalg.norm(arr - DARK_RGB, axis=2) < 48.0
    components = [component for component in _components(dark_mask) if 100 <= len(component) <= 260]
    if len(components) != 28:
        raise ValueError(f"expected 28 dark dots, recovered {len(components)}")
    points = np.asarray([component.mean(axis=0) for component in components], dtype=np.float64)
    balance = points.mean(axis=0)

    scores: list[tuple[float, str]] = []
    for name, rgb in PALETTE.items():
        center = _colored_ring_center(arr, rgb)
        scores.append((float(np.linalg.norm(center - balance)), name))
    scores.sort()
    if not math.isfinite(scores[0][0]) or scores[1][0] - scores[0][0] < 8.0:
        raise ValueError("pixels do not reveal a unique balance-point ring")
    return scores[0][1]
