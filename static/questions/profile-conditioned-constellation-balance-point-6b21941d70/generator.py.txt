"""Deterministic scene sampler and Pillow renderer for constellation_balance_point."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

COLORS = (
    ("red", (210, 52, 62)),
    ("blue", (42, 103, 184)),
    ("green", (36, 145, 91)),
    ("orange", (226, 126, 33)),
)
BACKGROUND = (248, 247, 243)
DARK = (28, 34, 43)
DOT_COUNT = 28


def _mean_point(points: list[list[float]]) -> tuple[float, float]:
    if not points:
        raise ValueError("point set is empty")
    count = float(len(points))
    return sum(p[0] for p in points) / count, sum(p[1] for p in points) / count


def _candidate_scores(scene: dict) -> list[tuple[float, str]]:
    mx, my = _mean_point(scene["points"])
    return [
        (math.hypot(float(item["center"][0]) - mx, float(item["center"][1]) - my), item["color"])
        for item in scene["candidates"]
    ]


def _candidate_layout(rng: random.Random, seed: int) -> tuple[list[list[float]], int]:
    """Return four exchangeable rectangle corners and the counterbalanced winning slot."""
    # Keep every exchangeable slot far enough from the canvas edge that a
    # centered cloud has ample construction room for every integer seed.
    base_x = rng.uniform(244.0, 268.0)
    base_y = rng.uniform(244.0, 268.0)
    half_x = rng.uniform(32.0, 42.0)
    half_y = rng.uniform(32.0, 42.0)
    angle = rng.uniform(0.0, 2.0 * math.pi)
    ca, sa = math.cos(angle), math.sin(angle)
    local = [(-half_x, -half_y), (half_x, -half_y), (half_x, half_y), (-half_x, half_y)]
    slots = [
        [round(base_x + ca * x - sa * y, 4), round(base_y + sa * x + ca * y, 4)]
        for x, y in local
    ]
    winner_slot = (int(seed) // 4) % 4
    if int(seed) % 7 == 0:
        # A reduced-margin case: move one adjacent slot toward the winning slot.
        runner_slot = (winner_slot + 1) % 4
        wx, wy = slots[winner_slot]
        rx, ry = slots[runner_slot]
        length = math.hypot(rx - wx, ry - wy)
        target_gap = 32.0
        slots[runner_slot] = [
            round(wx + (rx - wx) * target_gap / length, 4),
            round(wy + (ry - wy) * target_gap / length, 4),
        ]
    return slots, winner_slot


def _make_points(
    rng: random.Random, target: list[float], candidate_centers: list[list[float]]
) -> list[list[float]]:
    """Sample a separated, asymmetric cloud whose coordinate mean is target."""
    for _ in range(1400):
        offsets: list[tuple[float, float]] = []
        for _point_index in range(DOT_COUNT):
            for _placement_try in range(500):
                angle = rng.uniform(0.0, 2.0 * math.pi)
                radius = math.sqrt(rng.uniform(70.0**2, 148.0**2))
                stretch = rng.uniform(0.78, 1.18)
                ox = radius * math.cos(angle) * stretch
                oy = radius * math.sin(angle) / stretch
                if all(math.hypot(ox - qx, oy - qy) >= 20.0 for qx, qy in offsets):
                    offsets.append((ox, oy))
                    break
            else:
                break
        if len(offsets) != DOT_COUNT:
            continue
        mean_x = sum(x for x, _ in offsets) / DOT_COUNT
        mean_y = sum(y for _, y in offsets) / DOT_COUNT
        points = [
            [round(target[0] + x - mean_x, 4), round(target[1] + y - mean_y, 4)]
            for x, y in offsets
        ]
        if any(not (28.0 <= x <= 484.0 and 28.0 <= y <= 484.0) for x, y in points):
            continue
        if any(
            math.hypot(points[i][0] - points[j][0], points[i][1] - points[j][1]) < 19.0
            for i in range(DOT_COUNT)
            for j in range(i)
        ):
            continue
        if any(
            math.hypot(x - center[0], y - center[1]) < 27.0
            for x, y in points
            for center in candidate_centers
        ):
            continue
        # Rounding can move the mean by only a tiny fraction of a pixel.
        mx, my = _mean_point(points)
        if math.hypot(mx - target[0], my - target[1]) > 0.001:
            continue
        return points
    raise RuntimeError("could not sample a separated balance-point constellation")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    slots, winner_slot = _candidate_layout(rng, int(seed))
    winner_color_index = int(seed) % len(COLORS)
    target = slots[winner_slot]
    points = _make_points(rng, target, slots)

    remaining_slots = [index for index in range(4) if index != winner_slot]
    rng.shuffle(remaining_slots)
    slot_for_color: dict[int, int] = {winner_color_index: winner_slot}
    for color_index, slot_index in zip(
        [index for index in range(4) if index != winner_color_index], remaining_slots
    ):
        slot_for_color[color_index] = slot_index
    candidates = [
        {"color": name, "rgb": list(rgb), "center": slots[slot_for_color[index]]}
        for index, (name, rgb) in enumerate(COLORS)
    ]
    return {
        "width": 512,
        "height": 512,
        "background": list(BACKGROUND),
        "dark_color": list(DARK),
        "point_radius": 7,
        "ring_radius": 13,
        "ring_width": 4,
        "points": points,
        "candidates": candidates,
    }


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), tuple(scene["background"]))
    draw = ImageDraw.Draw(image)
    radius = float(scene["point_radius"])
    for x, y in scene["points"]:
        draw.ellipse(
            ((x - radius) * scale, (y - radius) * scale, (x + radius) * scale, (y + radius) * scale),
            fill=tuple(scene["dark_color"]),
        )
    ring_radius = float(scene["ring_radius"])
    ring_width = int(scene["ring_width"])
    for item in scene["candidates"]:
        x, y = item["center"]
        draw.ellipse(
            (
                (x - ring_radius) * scale,
                (y - ring_radius) * scale,
                (x + ring_radius) * scale,
                (y + ring_radius) * scale,
            ),
            fill=tuple(scene["background"]),
            outline=tuple(item["rgb"]),
            width=ring_width * scale,
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    scores = sorted(_candidate_scores(scene))
    if len(scores) != 4 or scores[1][0] - scores[0][0] <= 0.0:
        raise ValueError("scene has no unique closest candidate to the balance point")
    return scores[0][1]


def margin(scene: dict) -> float:
    scores = sorted(score for score, _ in _candidate_scores(scene))
    return float(scores[1] - scores[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 40.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["points"] = list(reversed(twin["points"]))
    return [("reverse_dark_dot_draw_order", twin)]
