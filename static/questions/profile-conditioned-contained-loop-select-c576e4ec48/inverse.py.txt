"""Independent pixel-only inverse arm for the contained_loop_select world.

Receives ONLY the final PNG. It never imports or reads the renderer, the
analytic scene, prompts, or any renderer state. It recovers the decision by:

1. marking the red and blue ring pixels and, for each colored ring component,
   taking its centroid (the ring's center) and the min/max pixel distance from
   that centroid (the ring's inner and outer radius);
2. deciding containment between the two colored rings: one closed loop is fully
   inside the other's interior exactly when the center-to-center distance plus
   the inner ring's outer radius is strictly less than the outer ring's inner
   radius (a real white gap remains between the contained ring and the
   container ring's inner edge);
3. returning the color of the ring that is the contained (inner) loop.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "contained_loop_select-oracle-0.4.0"

_RING_RGB = {
    "red": (255, 0, 0),
    "blue": (0, 0, 255),
}
_CONTAIN_MARGIN = 4.0
_MIN_COUNT = 30


def _component_features(rgb: np.ndarray, r: int, g: int, b: int):
    mask = (rgb[..., 0] == r) & (rgb[..., 1] == g) & (rgb[..., 2] == b)
    ys, xs = np.nonzero(mask)
    if len(xs) < _MIN_COUNT:
        return None
    cx = float(xs.mean())
    cy = float(ys.mean())
    d = np.sqrt((xs.astype(np.float64) - cx) ** 2 + (ys.astype(np.float64) - cy) ** 2)
    return {"center": (cx, cy), "inner": float(d.min()), "outer": float(d.max())}


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB")).astype(np.int16)

    features = {}
    for name, (r, g, b) in _RING_RGB.items():
        feats = _component_features(arr, r, g, b)
        if feats is None:
            return "red"
        features[name] = feats

    (rx, ry) = features["red"]["center"]
    (bx, by) = features["blue"]["center"]
    center_dist = float(np.hypot(rx - bx, ry - by))

    red_inside_blue = (
        center_dist + features["red"]["outer"] < features["blue"]["inner"] - _CONTAIN_MARGIN
    )
    blue_inside_red = (
        center_dist + features["blue"]["outer"] < features["red"]["inner"] - _CONTAIN_MARGIN
    )

    if red_inside_blue and not blue_inside_red:
        return "red"
    if blue_inside_red and not red_inside_blue:
        return "blue"
    return "red"
