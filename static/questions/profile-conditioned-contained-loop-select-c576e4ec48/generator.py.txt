"""Loop-containment (closed-curve nesting) world renderer.

Two colored closed ring loops, one red and one blue, are drawn on a white
background. Exactly one of the two ring loops lies entirely inside the interior
(hole) of the other ring loop -- that is, one closed loop is fully contained by
the other's closed boundary, with a visible white gap between the contained
loop's outer edge and the container loop's inner edge. The other ring is the
outer container and is not inside the first.

Decision: which colored ring loop (red or blue) is the fully contained (inner)
loop -- the one that lies inside the other ring's interior.

The scene is a plain dict so the verifier's AST reachability audit is free of
side effects.

    z = {
        "width": 400,
        "height": 400,
        "background": "white",
        "stroke": 10.0,
        "outer_color": "blue",          # color of the containing (outer) ring
        "outer_center": [cx, cy],
        "outer_radius": R_o,
        "inner_radius": R_i,
        "inner_center": [cx2, cy2],     # center of the contained (inner) ring
        "contained_color": "red",       # color of the contained (inner) ring
    }
"""

from __future__ import annotations

import math
import random
from collections.abc import Iterator

from PIL import Image, ImageDraw

RENDERER_VERSION = "contained_loop_select-0.4.0"

RING_COLORS = {"red": (255, 0, 0), "blue": (0, 0, 255)}
WHITE = (255, 255, 255)

WIDTH = 400
HEIGHT = 400
STROKE = 10.0
QUARANTINE_MIN = 6.0


def _ring_center(rng: random.Random) -> tuple[float, float]:
    return (rng.uniform(185.0, 215.0), rng.uniform(185.0, 215.0))


def sample_scene(rng: random.Random, width: int = WIDTH, height: int = HEIGHT) -> dict:
    for _attempt in range(400):
        outer_radius = rng.uniform(105.0, 130.0)
        inner_radius = rng.uniform(28.0, 45.0)
        half = STROKE / 2.0
        outer_cx, outer_cy = _ring_center(rng)

        contained_color = rng.choice(["red", "blue"])
        outer_color = "blue" if contained_color == "red" else "red"

        max_offset = outer_radius - half - (inner_radius + half) - QUARANTINE_MIN
        if max_offset <= 2.0:
            continue
        off = rng.uniform(0.0, max_offset)
        ang = rng.uniform(0.0, 2.0 * math.pi)
        inner_cx = outer_cx + off * math.cos(ang)
        inner_cy = outer_cy + off * math.sin(ang)

        # keep both rings (and their band) strictly inside the canvas with a margin
        pad = half + QUARANTINE_MIN
        if (
            outer_cx - outer_radius - pad < 4
            or outer_cx + outer_radius + pad > width - 4
            or outer_cy - outer_radius - pad < 4
            or outer_cy + outer_radius + pad > height - 4
            or inner_cx - inner_radius - pad < 4
            or inner_cx + inner_radius + pad > width - 4
            or inner_cy - inner_radius - pad < 4
            or inner_cy + inner_radius + pad > height - 4
        ):
            continue

        scene = {
            "width": width,
            "height": height,
            "background": "white",
            "stroke": STROKE,
            "outer_color": outer_color,
            "outer_center": [outer_cx, outer_cy],
            "outer_radius": outer_radius,
            "inner_radius": inner_radius,
            "inner_center": [inner_cx, inner_cy],
            "contained_color": contained_color,
        }
        if not is_quarantined(scene):
            return scene
    return _fallback_scene()


def _fallback_scene() -> dict:
    return {
        "width": WIDTH,
        "height": HEIGHT,
        "background": "white",
        "stroke": STROKE,
        "outer_color": "blue",
        "outer_center": [200.0, 200.0],
        "outer_radius": 120.0,
        "inner_radius": 34.0,
        "inner_center": [200.0, 200.0],
        "contained_color": "red",
    }


def _draw_ring(
    draw: ImageDraw.ImageDraw,
    color: str,
    cx: float,
    cy: float,
    radius: float,
    stroke: int,
) -> None:
    step = 0.02
    theta = 0.0
    prev = (cx + radius * math.cos(theta), cy - radius * math.sin(theta))
    while theta < 2.0 * math.pi:
        theta = min(theta + step, 2.0 * math.pi)
        cur = (cx + radius * math.cos(theta), cy - radius * math.sin(theta))
        draw.line([prev, cur], fill=RING_COLORS[color], width=stroke)
        prev = cur


def render(scene: dict, supersample: int = 1) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    background = scene["background"]
    ss = supersample
    img = Image.new("RGB", (width * ss, height * ss), background)
    draw = ImageDraw.Draw(img)
    stroke = max(1, int(round(float(scene["stroke"]) * ss)))

    ocx, ocy = (float(v) * ss for v in scene["outer_center"])
    orad = float(scene["outer_radius"]) * ss
    _draw_ring(draw, scene["outer_color"], ocx, ocy, orad, stroke)

    ccx, ccy = (float(v) * ss for v in scene["inner_center"])
    crad = float(scene["inner_radius"]) * ss
    _draw_ring(draw, scene["contained_color"], ccx, ccy, crad, stroke)

    if ss > 1:
        img = img.resize((width, height), Image.LANCZOS)
    return img


def margin(scene: dict) -> float:
    """Robustness margin: the narrowest visible gap that fixes containment.

    The decision is stable as long as (a) the contained ring's outer edge stays
    clearly clear of the container ring's inner edge (the white gap), and (b)
    every ring band keeps a clear margin to the canvas edge. The margin is the
    minimum of those clearances in pixels (minus a small lead-in) and is
    reported as a non-negative number.
    """
    half = float(scene["stroke"]) / 2.0
    ocx, ocy = (float(v) for v in scene["outer_center"])
    ccx, ccy = (float(v) for v in scene["inner_center"])
    orad = float(scene["outer_radius"])
    crad = float(scene["inner_radius"])

    gap = (orad - half) - (math.hypot(ocx - ccx, ocy - ccy) + (crad + half))
    edge = half + 4.0
    mx = min(
        gap - 4.0,
        ocx - orad - edge,
        (float(scene["width"]) - (ocx + orad) - edge),
        ocy - orad - edge,
        (float(scene["height"]) - (ocy + orad) - edge),
    )
    return max(0.0, float(mx))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN


def analytic_gold(scene: dict) -> str:
    """Return the color of the ring that is fully contained (the inner loop)."""
    return str(scene["contained_color"])


def latent_symmetries(scene: dict) -> Iterator[dict]:
    return iter(())
