"""Deterministic Pillow renderer for a three-arrow recycling-cycle symbol."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

SIZE = 512
AA = 4
HEAD_RGB = (238, 116, 36)
SHAFT_PALETTE = [(24, 132, 92), (42, 105, 176), (113, 76, 164), (16, 145, 156)]


def sample_scene(seed: int) -> dict:
    """Pair adjacent seeds: identical nuisances, opposite circulation labels."""
    rng = random.Random(int(seed) // 2)
    direction = "clockwise" if int(seed) % 2 == 0 else "counterclockwise"
    distractors = []
    for corner_x, corner_y in ((48, 48), (464, 48), (48, 464), (464, 464)):
        distractors.append([
            corner_x + rng.randint(-12, 12),
            corner_y + rng.randint(-12, 12),
            rng.randint(4, 7),
        ])
    return {
        "direction": direction,
        "center_x": rng.randint(246, 266),
        "center_y": rng.randint(246, 266),
        "radius": rng.randint(122, 145),
        "rotation_deg": round(rng.uniform(0.0, 120.0), 6),
        "head_skew_deg": round(rng.uniform(-48.0, 48.0), 6),
        "shaft_color": list(rng.choice(SHAFT_PALETTE)),
        "distractors": distractors,
    }


def _point(cx: float, cy: float, radius: float, theta: float) -> tuple[float, float]:
    return cx + radius * math.cos(theta), cy + radius * math.sin(theta)


def render(scene: dict) -> Image.Image:
    # Keep every declared latent field visibly load-bearing and auditable.
    direction = scene["direction"]
    cx = float(scene["center_x"])
    cy = float(scene["center_y"])
    radius = float(scene["radius"])
    rotation = math.radians(float(scene["rotation_deg"]))
    skew = math.radians(float(scene["head_skew_deg"]))
    shaft_color = tuple(int(v) for v in scene["shaft_color"])
    distractors = scene["distractors"]

    sign = 1.0 if direction == "clockwise" else -1.0
    image = Image.new("RGB", (SIZE * AA, SIZE * AA), (250, 249, 245))
    draw = ImageDraw.Draw(image)

    # Neutral registration marks are sampled and permuted independently of label.
    for x, y, r in distractors:
        draw.ellipse(
            ((x - r) * AA, (y - r) * AA, (x + r) * AA, (y + r) * AA),
            fill=(205, 207, 204),
        )

    # A visible white center makes the circulation reference explicit.
    cr = radius * 0.45
    draw.ellipse(
        ((cx - cr) * AA, (cy - cr) * AA, (cx + cr) * AA, (cy + cr) * AA),
        fill=(255, 255, 255),
        outline=(176, 180, 176),
        width=2 * AA,
    )

    for k in range(3):
        theta_head = rotation + k * 2.0 * math.pi / 3.0
        # The shaft approaches the arrowhead in the declared circulation direction.
        angles = [theta_head - sign * math.radians(v) for v in range(76, 13, -3)]
        shaft = [_point(cx, cy, radius, angle) for angle in angles]
        draw.line(
            [(round(x * AA), round(y * AA)) for x, y in shaft],
            fill=shaft_color,
            width=18 * AA,
            joint="curve",
        )

        hx, hy = _point(cx, cy, radius, theta_head)
        radial = (math.cos(theta_head), math.sin(theta_head))
        tangent = (-math.sin(theta_head) * sign, math.cos(theta_head) * sign)
        # Symmetric radial skew varies difficulty without changing the signed tangent.
        forward = (
            tangent[0] * math.cos(skew) + radial[0] * math.sin(skew),
            tangent[1] * math.cos(skew) + radial[1] * math.sin(skew),
        )
        normal = (-forward[1], forward[0])
        tip = (hx + 25 * forward[0], hy + 25 * forward[1])
        base = (hx - 12.5 * forward[0], hy - 12.5 * forward[1])
        tri = [
            tip,
            (base[0] + 15 * normal[0], base[1] + 15 * normal[1]),
            (base[0] - 15 * normal[0], base[1] - 15 * normal[1]),
        ]
        draw.polygon([(round(x * AA), round(y * AA)) for x, y in tri], fill=HEAD_RGB)

    return image.resize((SIZE, SIZE), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return str(scene["direction"])


def margin(scene: dict) -> float:
    """Normalized distance from radial ambiguity: |tangential projection|."""
    return round(abs(math.cos(math.radians(float(scene["head_skew_deg"])))), 12)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 0.55


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
