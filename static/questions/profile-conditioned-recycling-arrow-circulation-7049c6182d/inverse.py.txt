"""Independent pixel-only inverse for global arrow circulation."""

from __future__ import annotations

import numpy as np
from PIL import Image


def _components(mask: np.ndarray) -> list[np.ndarray]:
    """Return 8-connected orange-pixel coordinate arrays without external CV code."""
    remaining = {tuple(v) for v in np.argwhere(mask).tolist()}
    groups: list[np.ndarray] = []
    while remaining:
        start = remaining.pop()
        stack = [start]
        pixels = [start]
        while stack:
            y, x = stack.pop()
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if dx == 0 and dy == 0:
                        continue
                    q = (y + dy, x + dx)
                    if q in remaining:
                        remaining.remove(q)
                        stack.append(q)
                        pixels.append(q)
        if len(pixels) >= 150:
            groups.append(np.asarray(pixels, dtype=np.float64))
    return groups


def _tip_direction(component_yx: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    xy = component_yx[:, ::-1]
    center = xy.mean(axis=0)
    centered = xy - center
    covariance = centered.T @ centered
    values, vectors = np.linalg.eigh(covariance)
    axis = vectors[:, int(np.argmax(values))]
    projections = centered @ axis
    # The arrow tip extends twice as far from the triangle centroid as its base.
    if abs(float(projections.min())) > abs(float(projections.max())):
        axis = -axis
    return center, axis


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    # Public semantic cue: orange arrowheads. Broad thresholds tolerate AA rims.
    mask = (arr[:, :, 0] > 190) & (arr[:, :, 1] > 65) & (arr[:, :, 1] < 175) & (arr[:, :, 2] < 100)
    components = _components(mask)
    if len(components) != 3:
        raise ValueError(f"expected three visible orange arrowheads, found {len(components)}")
    measurements = [_tip_direction(comp) for comp in components]
    symbol_center = np.mean([center for center, _ in measurements], axis=0)
    votes = []
    for center, forward in measurements:
        radial = center - symbol_center
        denom = float(np.linalg.norm(radial) * np.linalg.norm(forward))
        if denom <= 0:
            raise ValueError("degenerate arrow geometry")
        votes.append(float((radial[0] * forward[1] - radial[1] * forward[0]) / denom))
    score = float(np.median(votes))
    if abs(score) < 0.50 or any(np.sign(v) != np.sign(score) for v in votes):
        raise ValueError(f"ambiguous circulation score {score:.3f}, votes={votes}")
    # Image y increases downward, so a positive cross product is visually clockwise.
    return "clockwise" if score > 0 else "counterclockwise"
