"""Scene sampling, rendering and analytic gold for pipe_bottleneck_tank.

A rectilinear pipe network is drawn on a lattice. Every pipe is drawn with one
of three clearly different visible thicknesses. A green pump square and three
coloured tank squares sit on lattice junctions. A route's capacity is the
thickness of its thinnest pipe; the decision is which tank has the highest
capacity route from the pump (a widest-path / bottleneck quantity, not a
length, a count of turns or an ordinal along one stroke).
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "pipe_bottleneck_tank-renderer-0.1.0"

CANVAS = (560, 440)
COLS, ROWS = 5, 4
STEP = 100
ORIGIN = (80, 70)
WIDTHS = (8, 14, 20)
MARKER = 36
BACKGROUND = (255, 255, 255)
PIPE_RGB = (80, 80, 80)
PUMP_RGB = (0, 150, 60)
TANK_PALETTE = {
    "red": (215, 40, 40),
    "blue": (40, 70, 210),
    "orange": (240, 150, 20),
}
TANK_COLORS = ("red", "blue", "orange")
QUARANTINE_PX = 6.0


# --------------------------------------------------------------------------
# lattice helpers


def _node_xy(node: list[int] | tuple[int, int]) -> tuple[int, int]:
    c, r = node
    return ORIGIN[0] + STEP * int(c), ORIGIN[1] + STEP * int(r)


def _all_edges() -> list[tuple[tuple[int, int], tuple[int, int]]]:
    edges = []
    for c in range(COLS):
        for r in range(ROWS):
            if c + 1 < COLS:
                edges.append(((c, r), (c + 1, r)))
            if r + 1 < ROWS:
                edges.append(((c, r), (c, r + 1)))
    return edges


def _spanning_plus(rng: random.Random) -> list[tuple[tuple[int, int], tuple[int, int]]]:
    """Random spanning tree over all lattice nodes plus a few extra edges."""
    pool = _all_edges()
    rng.shuffle(pool)
    parent: dict[tuple[int, int], tuple[int, int]] = {}

    def find(x):
        while parent.setdefault(x, x) != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    chosen, rest = [], []
    for a, b in pool:
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[ra] = rb
            chosen.append((a, b))
        else:
            rest.append((a, b))
    chosen.extend(rest[: rng.randint(4, 7)])
    return chosen


def _bottlenecks(
    edges: list, widths: list, source: tuple[int, int]
) -> dict[tuple[int, int], float]:
    """Widest-path capacity: maximise the minimum edge width along a route."""
    adj: dict[tuple[int, int], list] = {}
    for (a, b), w in zip(edges, widths):
        a, b = tuple(a), tuple(b)
        adj.setdefault(a, []).append((b, float(w)))
        adj.setdefault(b, []).append((a, float(w)))
    best = {tuple(source): float("inf")}
    todo = {tuple(source)}
    while todo:
        node = max(todo, key=lambda n: best[n])
        todo.discard(node)
        for nxt, w in adj.get(node, []):
            cap = min(best[node], w)
            if cap > best.get(nxt, -1.0):
                best[nxt] = cap
                todo.add(nxt)
    return best


def _line_coverage_ok(edges: list) -> bool:
    """Every lattice line carries at least one edge along it, so the pixel arm
    can recover the full lattice from stroke profiles alone."""
    cols = {a[0] for a, b in edges if a[0] == b[0]}
    rows = {a[1] for a, b in edges if a[1] == b[1]}
    return len(cols) == COLS and len(rows) == ROWS


# --------------------------------------------------------------------------
# sampling


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    for _ in range(4000):
        edges = _spanning_plus(rng)
        if not _line_coverage_ok(edges):
            continue
        widths = [rng.choice(WIDTHS) for _ in edges]
        nodes = [(c, r) for c in range(COLS) for r in range(ROWS)]
        picks = rng.sample(nodes, 4)
        pump, tanks = picks[0], picks[1:]
        if any(
            abs(p[0] - q[0]) + abs(p[1] - q[1]) < 2
            for i, p in enumerate(picks)
            for q in picks[i + 1 :]
        ):
            continue
        caps = _bottlenecks(edges, widths, pump)
        vals = [caps.get(t, 0.0) for t in tanks]
        if min(vals) <= 0.0:
            continue
        order = sorted(range(3), key=lambda i: -vals[i])
        if vals[order[0]] - vals[order[1]] < QUARANTINE_PX:
            continue
        # counterbalance: the winning tank node is forced to carry the colour
        # chosen by the seed, so answer identity is independent of geometry.
        target = TANK_COLORS[seed % 3]
        others = [c for c in TANK_COLORS if c != target]
        rng.shuffle(others)
        names = [None, None, None]
        names[order[0]] = target
        names[order[1]] = others[0]
        names[order[2]] = others[1]
        return {
            "canvas": list(CANVAS),
            "grid": [COLS, ROWS],
            "step": STEP,
            "origin": list(ORIGIN),
            "marker": MARKER,
            "edges": [[list(a), list(b)] for a, b in edges],
            "widths": [int(w) for w in widths],
            "pump": list(pump),
            "tanks": [
                {"color": names[i], "node": list(tanks[i])} for i in range(3)
            ],
            "palette": {k: list(v) for k, v in TANK_PALETTE.items()},
            "pump_rgb": list(PUMP_RGB),
            "pipe_rgb": list(PIPE_RGB),
            "background_rgb": list(BACKGROUND),
        }
    raise RuntimeError(f"no admissible scene for seed {seed}")


# --------------------------------------------------------------------------
# rendering


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", tuple(scene["canvas"]), tuple(scene["background_rgb"]))
    d = ImageDraw.Draw(img)
    pipe = tuple(scene["pipe_rgb"])
    incident: dict[tuple[int, int], int] = {}
    for (a, b), w in zip(scene["edges"], scene["widths"]):
        ax, ay = _node_xy(a)
        bx, by = _node_xy(b)
        h = w // 2
        if ay == by:
            d.rectangle([min(ax, bx), ay - h, max(ax, bx), ay + h - 1], fill=pipe)
        else:
            d.rectangle([ax - h, min(ay, by), ax + h - 1, max(ay, by)], fill=pipe)
        for n in (tuple(a), tuple(b)):
            incident[n] = max(incident.get(n, 0), w)
    for node, w in incident.items():  # clean square joint at each junction
        x, y = _node_xy(node)
        h = w // 2
        d.rectangle([x - h, y - h, x + h - 1, y + h - 1], fill=pipe)
    m = scene["marker"] // 2
    px, py = _node_xy(scene["pump"])
    d.rectangle([px - m, py - m, px + m - 1, py + m - 1], fill=tuple(scene["pump_rgb"]))
    for tank in scene["tanks"]:
        tx, ty = _node_xy(tank["node"])
        d.rectangle(
            [tx - m, ty - m, tx + m - 1, ty + m - 1],
            fill=tuple(scene["palette"][tank["color"]]),
        )
    return img


# --------------------------------------------------------------------------
# analytic gold


def _caps(scene: dict) -> list[tuple[str, float]]:
    caps = _bottlenecks(scene["edges"], scene["widths"], tuple(scene["pump"]))
    return [
        (t["color"], caps.get(tuple(t["node"]), 0.0)) for t in scene["tanks"]
    ]


def analytic_gold(scene: dict) -> str:
    ranked = sorted(_caps(scene), key=lambda kv: (-kv[1], kv[0]))
    return ranked[0][0]


def margin(scene: dict) -> float:
    ranked = sorted((v for _, v in _caps(scene)), reverse=True)
    return float(ranked[0] - ranked[1])


def is_quarantined(scene: dict) -> bool:
    vals = [v for _, v in _caps(scene)]
    return min(vals) <= 0.0 or margin(scene) < QUARANTINE_PX


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Label-preserving latent relabellings that leave the raster identical."""
    import copy

    flipped = copy.deepcopy(scene)
    flipped["edges"] = [[b, a] for a, b in scene["edges"]]

    reordered = copy.deepcopy(scene)
    pairs = list(zip(scene["edges"], scene["widths"]))[::-1]
    reordered["edges"] = [e for e, _ in pairs]
    reordered["widths"] = [w for _, w in pairs]

    relisted = copy.deepcopy(scene)
    relisted["tanks"] = list(scene["tanks"])[::-1]

    return [
        ("reverse_edge_endpoint_order", flipped),
        ("reverse_edge_record_order", reordered),
        ("reverse_tank_record_order", relisted),
    ]
