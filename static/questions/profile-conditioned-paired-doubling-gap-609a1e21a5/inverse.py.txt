"""Independent image-only inverse for the paired-doubling plot."""

from __future__ import annotations

import numpy as np
from PIL import Image

LABELS = ("A", "B", "C", "D")


def _components(mask: np.ndarray, minimum: int = 35) -> list[tuple[float, float, int]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    found = []
    for y, x in zip(*np.nonzero(mask)):
        if seen[y, x]:
            continue
        stack = [(int(y), int(x))]
        seen[y, x] = True
        points = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= minimum:
            ys = np.fromiter((point[0] for point in points), dtype=float)
            xs = np.fromiter((point[1] for point in points), dtype=float)
            found.append((float(xs.mean()), float(ys.mean()), len(points)))
    return found


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    red, green, blue = array[:, :, 0], array[:, :, 1], array[:, :, 2]
    blue_mask = (blue > 125) & (blue > red * 1.18) & (blue > green * 1.05) & (red < 100)
    orange_mask = (red > 180) & (green > 65) & (green < 175) & (blue < 90) & (red > green * 1.35)
    magenta_mask = (red > 140) & (blue > 100) & (green < 105) & (red > green * 1.5)

    blue_marks = [item for item in _components(blue_mask) if item[1] > 110]
    orange_marks = [item for item in _components(orange_mask) if item[1] > 110]
    magenta = _components(magenta_mask, minimum=150)
    if len(blue_marks) != 4 or len(orange_marks) != 4 or not magenta:
        raise ValueError("required ruler evidence is missing or ambiguous")
    blue_marks.sort(key=lambda item: item[1])
    orange_marks.sort(key=lambda item: item[1])
    if any(abs(blue_marks[i][1] - orange_marks[i][1]) > 8 for i in range(4)):
        raise ValueError("paired markers cannot be bound by row")
    origin = max(magenta, key=lambda item: item[2])[0]
    residuals = [abs((orange_marks[i][0] - origin) - 2.0 * (blue_marks[i][0] - origin)) for i in range(4)]
    order = sorted(range(4), key=lambda index: residuals[index])
    if residuals[order[1]] - residuals[order[0]] < 5.0:
        raise ValueError("pixel decision is too close to resolve")
    return LABELS[order[0]]
