"""Deterministic paired-doubling measurement world."""

from __future__ import annotations

import copy
import random
from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 640, 480
SCALE = 45.0
ROW_Y = (145, 225, 305, 385)
LABELS = ("A", "B", "C", "D")
PALETTES = (
    {"blue": (31, 119, 180), "orange": (238, 119, 35)},
    {"blue": (42, 105, 190), "orange": (225, 132, 28)},
    {"blue": (25, 132, 168), "orange": (244, 104, 30)},
)


def _residuals(scene: dict) -> list[float]:
    return [abs(float(row["later"]) - 2.0 * float(row["first"])) for row in scene["rows"]]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, well-separated four-row scene."""
    rng = random.Random(int(seed))
    winner = int(seed) % 4
    near = (int(seed) // 4) % 6 == 0
    rows = []
    for index in range(4):
        first = rng.uniform(1.25, 3.65)
        if index == winner:
            magnitude = rng.uniform(0.07, 0.15)
        elif near and index == (winner + 1) % 4:
            magnitude = rng.uniform(0.38, 0.48)
        else:
            magnitude = rng.uniform(0.75, 1.65)
        sign = -1.0 if rng.random() < 0.5 else 1.0
        later = 2.0 * first + sign * magnitude
        if later < 0.55 or later > 9.45 or abs(later - first) < 0.62:
            later = 2.0 * first - sign * magnitude
        rows.append({"first": round(first, 5), "later": round(later, 5)})
    return {
        "rows": rows,
        "palette": int(seed) % len(PALETTES),
        "x_shift": rng.randint(-12, 12),
    }


def analytic_gold(scene: dict) -> str:
    residuals = _residuals(scene)
    return LABELS[min(range(4), key=lambda index: residuals[index])]


def margin(scene: dict) -> float:
    ordered = sorted(_residuals(scene))
    return (ordered[1] - ordered[0]) * SCALE


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 9.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def render(scene: dict) -> Image.Image:
    ss = 3
    image = Image.new("RGB", (WIDTH * ss, HEIGHT * ss), (250, 250, 248))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=18 * ss)
    small = ImageFont.load_default(size=14 * ss)
    palette = PALETTES[int(scene.get("palette", 0)) % len(PALETTES)]
    origin = 105 + int(scene.get("x_shift", 0))
    top, bottom = 112, 416

    draw.text((WIDTH * ss / 2, 31 * ss), "Paired measurements on a common linear ruler", fill=(28, 28, 28), anchor="mm", font=font)
    draw.ellipse((70 * ss, 67 * ss, 84 * ss, 81 * ss), fill=palette["blue"])
    draw.text((91 * ss, 74 * ss), "first", fill=(35, 35, 35), anchor="lm", font=small)
    diamond = [(176 * ss, 66 * ss), (184 * ss, 74 * ss), (176 * ss, 82 * ss), (168 * ss, 74 * ss)]
    draw.polygon(diamond, fill=palette["orange"])
    draw.text((193 * ss, 74 * ss), "later", fill=(35, 35, 35), anchor="lm", font=small)

    for tick in range(11):
        x = (origin + tick * SCALE) * ss
        color = (219, 219, 215) if tick else (205, 205, 201)
        draw.line((x, top * ss, x, bottom * ss), fill=color, width=1 * ss)
        draw.text((x, 431 * ss), str(tick), fill=(70, 70, 70), anchor="mt", font=small)
    draw.line((origin * ss, (top - 5) * ss, origin * ss, (bottom + 4) * ss), fill=(196, 45, 154), width=4 * ss)
    draw.text((origin * ss, 99 * ss), "zero", fill=(160, 25, 124), anchor="ms", font=small)

    for index, row in enumerate(scene["rows"]):
        y = ROW_Y[index]
        x_first = origin + float(row["first"]) * SCALE
        x_later = origin + float(row["later"]) * SCALE
        draw.text((66 * ss, y * ss), LABELS[index], fill=(22, 22, 22), anchor="mm", font=font)
        draw.line((origin * ss, y * ss, (origin + 10 * SCALE) * ss, y * ss), fill=(185, 185, 181), width=2 * ss)
        draw.line((x_first * ss, y * ss, x_later * ss, y * ss), fill=(120, 120, 116), width=3 * ss)
        radius = 9
        draw.ellipse(((x_first - radius) * ss, (y - radius) * ss, (x_first + radius) * ss, (y + radius) * ss), fill=palette["blue"], outline=(255, 255, 255), width=2 * ss)
        radius = 10
        draw.polygon([
            (x_later * ss, (y - radius) * ss),
            ((x_later + radius) * ss, y * ss),
            (x_later * ss, (y + radius) * ss),
            ((x_later - radius) * ss, y * ss),
        ], fill=palette["orange"], outline=(255, 255, 255))

    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def transformed(scene: dict, *, palette: int | None = None, x_shift: int | None = None, row_order: list[int] | None = None) -> dict:
    """Public metamorphic helper used only by tests, never by the oracle."""
    twin = copy.deepcopy(scene)
    if palette is not None:
        twin["palette"] = int(palette)
    if x_shift is not None:
        twin["x_shift"] = int(x_shift)
    if row_order is not None:
        twin["rows"] = [twin["rows"][index] for index in row_order]
    return twin
