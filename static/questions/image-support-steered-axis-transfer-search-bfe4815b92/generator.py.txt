"""Deterministic compact cross-axis sequence-search world."""

from __future__ import annotations

import random
from PIL import Image, ImageDraw

SIZE = 1200
ANSWERS = ("upper-left", "upper-right", "lower-left", "lower-right")
BASE_COLORS = ((220, 55, 82), (20, 151, 163), (238, 166, 35))
INK = (43, 51, 59)
PANEL = (236, 239, 241)
WHITE = (255, 255, 255)


def _scores(scene: dict) -> list[int]:
    reference = list(scene["reference"])
    return [sum(int(pair[i] == reference[i]) for i in range(2)) for pair in scene["candidates"]]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    a, b = rng.sample(range(3), 2)
    c = next(value for value in range(3) if value not in (a, b))
    target = int(seed) % 4
    distractors = [[a, c], [c, b], [b, a]]
    rng.shuffle(distractors)
    candidates = []
    for index in range(4):
        candidates.append([a, b] if index == target else distractors.pop())
    palette_order = list(range(3))
    rng.shuffle(palette_order)
    # Centers lie in the exclusive cores of 8x8 overlapping-ablation cells.
    # Reference and candidates occupy cells two columns apart with a blank cell between.
    ref_cell = 1 + rng.randrange(4)
    row_cell = 1 + rng.randrange(6)
    origin = [ref_cell * 150 + 75, row_cell * 150 + 75]
    return {
        "origin": origin,
        "palette_order": palette_order,
        "reference": [a, b],
        "candidates": candidates,
    }


def render(scene: dict) -> Image.Image:
    ox, oy = map(int, scene["origin"])
    palette = [BASE_COLORS[int(i)] for i in scene["palette_order"]]
    image = Image.new("RGB", (SIZE, SIZE), WHITE)
    draw = ImageDraw.Draw(image)
    # Pale grouping cards and arrow are explanatory but nondiagnostic.
    draw.rounded_rectangle((ox - 30, oy - 38, ox + 30, oy + 38), radius=10, fill=PANEL, outline=INK, width=3)
    cx = ox + 300
    draw.rounded_rectangle((cx - 40, oy - 40, cx + 40, oy + 40), radius=10, fill=PANEL, outline=INK, width=3)
    draw.line((cx, oy - 36, cx, oy + 36), fill=(180, 185, 190), width=2)
    draw.line((cx - 36, oy, cx + 36, oy), fill=(180, 185, 190), width=2)
    draw.line((ox + 48, oy, cx - 55, oy), fill=(145, 151, 156), width=4)
    draw.polygon(((cx - 55, oy), (cx - 70, oy - 9), (cx - 70, oy + 9)), fill=(145, 151, 156))

    radius = 8
    for y, value in ((oy - 16, scene["reference"][0]), (oy + 16, scene["reference"][1])):
        color = palette[int(value)]
        draw.ellipse((ox-radius, y-radius, ox+radius, y+radius), fill=color, outline=INK, width=2)

    centers = (
        ((cx - 27, oy - 22), (cx - 14, oy - 22)),
        ((cx + 14, oy - 22), (cx + 27, oy - 22)),
        ((cx - 27, oy + 22), (cx - 14, oy + 22)),
        ((cx + 14, oy + 22), (cx + 27, oy + 22)),
    )
    for pair, points in zip(scene["candidates"], centers, strict=True):
        for value, (x, y) in zip(pair, points, strict=True):
            color = palette[int(value)]
            draw.ellipse((x-radius, y-radius, x+radius, y+radius), fill=color, outline=INK, width=2)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    winners = [i for i, score in enumerate(scores) if score == 2]
    if len(winners) != 1:
        raise ValueError("scene lacks a unique ordered match")
    return ANSWERS[winners[0]]


def margin(scene: dict) -> float:
    scores = sorted(_scores(scene), reverse=True)
    return float(scores[0] - scores[1])


def is_quarantined(scene: dict) -> bool:
    try:
        scores = sorted(_scores(scene), reverse=True)
        return not (len(scene["candidates"]) == 4 and scores[0] == 2 and scores[1] <= 1)
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
