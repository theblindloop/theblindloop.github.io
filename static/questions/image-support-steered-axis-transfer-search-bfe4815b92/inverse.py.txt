"""Independent pixel-only inverse using color components and relative geometry."""

from __future__ import annotations

import numpy as np
from PIL import Image

ANSWERS = ("upper-left", "upper-right", "lower-left", "lower-right")
COLORS = np.asarray(((220, 55, 82), (20, 151, 163), (238, 166, 35)), dtype=np.float32)


def _components(mask: np.ndarray) -> list[tuple[float, float, int]]:
    seen = np.zeros(mask.shape, dtype=bool)
    output = []
    h, w = mask.shape
    for sy, sx in np.argwhere(mask):
        sy, sx = int(sy), int(sx)
        if seen[sy, sx]:
            continue
        stack = [(sy, sx)]
        seen[sy, sx] = True
        points = []
        while stack:
            y, x = stack.pop()
            points.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    stack.append((ny, nx))
        if len(points) >= 80:
            arr = np.asarray(points, dtype=np.float64)
            output.append((float(arr[:, 1].mean()), float(arr[:, 0].mean()), len(points)))
    return output


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    distances = np.linalg.norm(arr[:, :, None, :] - COLORS[None, None, :, :], axis=3)
    nearest = distances.argmin(axis=2)
    minimum = distances.min(axis=2)
    marks: list[tuple[float, float, int, int]] = []
    for color in range(3):
        for x, y, area in _components((nearest == color) & (minimum < 55.0)):
            if not 120 <= area <= 260:
                raise ValueError("colored mark has invalid area")
            marks.append((x, y, color, area))
    if len(marks) != 10:
        raise ValueError("expected exactly ten colored dots")

    marks.sort(key=lambda item: item[0])
    reference = marks[:2]
    rest = marks[2:]
    if max(p[0] for p in reference) - min(p[0] for p in reference) > 3:
        raise ValueError("reference is not a vertical pair")
    if min(p[0] for p in rest) - max(p[0] for p in reference) < 180:
        raise ValueError("reference and candidate groups are not separated")
    reference.sort(key=lambda item: item[1])
    ref_colors = [reference[0][2], reference[1][2]]
    if ref_colors[0] == ref_colors[1]:
        raise ValueError("reference colors must differ")

    rest.sort(key=lambda item: item[1])
    upper = sorted(rest[:4], key=lambda item: item[0])
    lower = sorted(rest[4:], key=lambda item: item[0])
    if abs(np.mean([p[1] for p in upper]) - np.mean([p[1] for p in lower])) < 25:
        raise ValueError("candidate rows are not separated")
    pairs = (upper[:2], upper[2:], lower[:2], lower[2:])
    scores = []
    for pair in pairs:
        if len(pair) != 2 or abs(pair[0][1] - pair[1][1]) > 3:
            raise ValueError("candidate is not a horizontal pair")
        scores.append(sum(int(pair[i][2] == ref_colors[i]) for i in range(2)))
    winners = [i for i, score in enumerate(scores) if score == 2]
    ordered = sorted(scores, reverse=True)
    if len(winners) != 1 or ordered[0] - ordered[1] < 1:
        raise ValueError("no unique ordered sequence match")
    return ANSWERS[winners[0]]
