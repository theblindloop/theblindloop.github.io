"""Deterministic analytic scene sampler and independent visual renderer."""

from __future__ import annotations

import itertools
import random

from PIL import Image, ImageDraw

CANONICAL = ("S", "S", "S", "L", "L", "L", "S", "S", "S")
PALETTE = ("#146bce", "#0b8f87", "#8b4bc2", "#d45b28")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C0DE)
    is_control = int(seed) % 2 == 0
    if is_control:
        pattern = CANONICAL
    else:
        long_positions = [c for c in itertools.combinations(range(9), 3) if c != (3, 4, 5)]
        chosen = long_positions[rng.randrange(len(long_positions))]
        pattern = tuple("L" if i in chosen else "S" for i in range(9))
    return {
        "pattern": list(pattern),
        "trace_color": PALETTE[rng.randrange(len(PALETTE))],
        "background": "#fbfaf6",
        "width": 1024,
        "height": 384,
        "left": rng.randint(46, 54),
        "baseline_y": rng.randint(270, 278),
        "pulse_top_y": rng.randint(198, 206),
        "short_width": rng.randint(32, 34),
        "long_width": rng.randint(74, 76),
        "gap": rng.randint(44, 45),
        "line_width": rng.choice((9, 10, 11)),
    }


def _path(scene: dict, scale: int = 1) -> list[tuple[int, int]]:
    x = int(scene["left"]) * scale
    baseline = int(scene["baseline_y"]) * scale
    top = int(scene["pulse_top_y"]) * scale
    gap = int(scene["gap"]) * scale
    short_width = int(scene["short_width"]) * scale
    long_width = int(scene["long_width"]) * scale
    points = [(x, baseline)]
    for symbol in scene["pattern"]:
        x += gap
        points.append((x, baseline))
        points.append((x, top))
        x += long_width if symbol == "L" else short_width
        points.append((x, top))
        points.append((x, baseline))
    x += gap
    points.append((x, baseline))
    return points


def render(scene: dict) -> Image.Image:
    scale = 3
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width * scale, height * scale), scene["background"])
    draw = ImageDraw.Draw(image)
    # Faint instrument rails are visible context but deliberately achromatic.
    rail = "#ddd9cf"
    draw.rounded_rectangle(
        (20 * scale, 24 * scale, (width - 20) * scale, (height - 24) * scale),
        radius=18 * scale,
        outline=rail,
        width=2 * scale,
    )
    baseline = int(scene["baseline_y"]) * scale
    draw.line(
        [(34 * scale, baseline), ((width - 34) * scale, baseline)],
        fill="#e7e3da",
        width=2 * scale,
    )
    points = _path(scene, scale)
    if points[-1][0] >= (width - 30) * scale:
        raise ValueError("trace does not fit canvas")
    draw.line(
        points,
        fill=scene["trace_color"],
        width=int(scene["line_width"]) * scale,
        joint="curve",
    )
    # Endpoint caps clarify reading direction while remaining part of the same trace.
    radius = int(scene["line_width"]) * scale // 2
    for x, y in (points[0], points[-1]):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=scene["trace_color"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    return "yes" if tuple(scene["pattern"]) == CANONICAL else "no"


def margin(scene: dict) -> float:
    threshold = 64.0
    widths = [
        (scene["long_width"] if symbol == "L" else scene["short_width"]) + scene["line_width"]
        for symbol in scene["pattern"]
    ]
    return float(min(abs(float(value) - threshold) for value in widths))


def is_quarantined(scene: dict) -> bool:
    if len(scene["pattern"]) != 9 or set(scene["pattern"]) - {"S", "L"}:
        return True
    if scene["pattern"].count("S") != 6 or scene["pattern"].count("L") != 3:
        return True
    if margin(scene) < 8.0:
        return True
    try:
        points = _path(scene)
    except Exception:
        return True
    return points[-1][0] >= int(scene["width"]) - 30 or int(scene["pulse_top_y"]) < 35


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = dict(scene)
    twin["pattern"] = list(scene["pattern"])
    twin["trace_color"] = str(scene["trace_color"]).upper()
    return [("hex_case_alias", twin)]
