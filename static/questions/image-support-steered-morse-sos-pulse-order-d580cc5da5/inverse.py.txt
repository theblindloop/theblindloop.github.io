"""Image-only inverse arm for the connected pulse-width trace."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANONICAL = ("S", "S", "S", "L", "L", "L", "S", "S", "S")


def _components(mask: np.ndarray) -> list[int]:
    visited = np.zeros(mask.shape, dtype=bool)
    sizes: list[int] = []
    height, width = mask.shape
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if visited[y, x]:
            continue
        queue = deque([(y, x)])
        visited[y, x] = True
        size = 0
        while queue:
            cy, cx = queue.popleft()
            size += 1
            for ny in range(max(0, cy - 1), min(height, cy + 2)):
                for nx in range(max(0, cx - 1), min(width, cx + 2)):
                    if mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        sizes.append(size)
    return sorted(sizes, reverse=True)


def _runs(values: np.ndarray) -> list[tuple[int, int]]:
    padded = np.pad(values.astype(np.int8), (1, 1))
    changes = np.diff(padded)
    starts = np.flatnonzero(changes == 1)
    ends = np.flatnonzero(changes == -1)
    return [(int(a), int(b)) for a, b in zip(starts, ends, strict=True)]


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    # The trace is the only saturated component; rails and background are achromatic.
    mask = (chroma >= 72) & (arr.min(axis=2) <= 205)
    if int(mask.sum()) < 5000:
        raise ValueError("telegraph trace missing or severely erased")
    significant = [size for size in _components(mask) if size >= 18]
    if len(significant) != 1:
        raise ValueError("trace is not one connected visible component")
    ys, xs = np.nonzero(mask)
    xmin, xmax = int(xs.min()), int(xs.max())
    if xmax - xmin < 840:
        raise ValueError("trace does not span the display")
    centers = np.full(xmax - xmin + 1, np.nan, dtype=np.float64)
    for offset, x in enumerate(range(xmin, xmax + 1)):
        column_y = np.flatnonzero(mask[:, x])
        if len(column_y):
            centers[offset] = float(np.median(column_y))
    if np.isnan(centers).any():
        raise ValueError("trace has a horizontal discontinuity")
    baseline = float(np.percentile(centers, 88))
    raised = centers < baseline - 58.0
    runs = [(a, b) for a, b in _runs(raised) if b - a >= 12]
    if len(runs) != 9:
        raise ValueError(f"expected nine raised pulses, found {len(runs)}")
    widths = [b - a for a, b in runs]
    if any(abs(width - 44) < 12 for width in widths):
        raise ValueError("ambiguous short/long pulse width")
    if any(width < 20 or width > 100 for width in widths):
        raise ValueError("pulse width outside the declared visual regime")
    pattern = tuple("S" if width < 44 else "L" for width in widths)
    if pattern.count("S") != 6 or pattern.count("L") != 3:
        raise ValueError("balanced pulse inventory is not visibly recoverable")
    return "yes" if pattern == CANONICAL else "no"
