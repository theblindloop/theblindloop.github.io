"""Deterministic paired switchboards for global correspondence binding."""

from __future__ import annotations

import itertools
import random
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1280, 720
N = 6
DECISIONS = tuple("ABCDEF")
INK = (35, 39, 48)
PAPER = (246, 247, 244)
GRID = (112, 120, 130)
RED = (218, 52, 62)
BLUE = (35, 105, 210)
START = (200, 38, 145)
WHITE = (255, 255, 255)
CELL = 64
GRID_Y = 205
LEFT_X = 120
RIGHT_X = 755
FONT_PATH = Path("/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf")

# Two colored permutations with a trivial shared centralizer. Every node has
# one incoming and outgoing link of each color, so degree cannot identify it.
_BASE_RED = (1, 0, 3, 4, 5, 2)
_BASE_BLUE = (2, 3, 0, 5, 1, 4)


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(str(FONT_PATH), size=size)


def _is_permutation(values: list[int] | tuple[int, ...]) -> bool:
    return len(values) == N and sorted(values) == list(range(N))


def _bijections(scene: dict) -> list[tuple[int, ...]]:
    lr, lb = scene["left_red"], scene["left_blue"]
    rr, rb = scene["right_red"], scene["right_blue"]
    out = []
    for q in itertools.permutations(range(N)):
        if all(q[lr[i]] == rr[q[i]] and q[lb[i]] == rb[q[i]] for i in range(N)):
            out.append(q)
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    left_order = list(range(N))
    rng.shuffle(left_order)
    left_pos = {node: pos for pos, node in enumerate(left_order)}
    left_red = [left_pos[_BASE_RED[node]] for node in left_order]
    left_blue = [left_pos[_BASE_BLUE[node]] for node in left_order]

    start_index = rng.randrange(N)
    target = int(seed) % N
    q: list[int | None] = [None] * N
    q[start_index] = target
    remaining = [j for j in range(N) if j != target]
    rng.shuffle(remaining)
    for i, value in zip((i for i in range(N) if i != start_index), remaining):
        q[i] = value
    mapping = [int(x) for x in q]

    right_red = [0] * N
    right_blue = [0] * N
    for i in range(N):
        right_red[mapping[i]] = mapping[left_red[i]]
        right_blue[mapping[i]] = mapping[left_blue[i]]
    scene = {
        "left_red": left_red,
        "left_blue": left_blue,
        "right_red": right_red,
        "right_blue": right_blue,
        "start_index": start_index,
    }
    if is_quarantined(scene):
        raise RuntimeError("constructed scene failed switchboard invariants")
    return scene


def analytic_gold(scene: dict) -> str:
    solutions = _bijections(scene)
    if len(solutions) != 1:
        raise ValueError("scene does not have a unique colored-link renaming")
    target = solutions[0][int(scene["start_index"])]
    return DECISIONS[target]


def margin(scene: dict) -> float:
    return 1.0 if not is_quarantined(scene) else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        keys = ("left_red", "left_blue", "right_red", "right_blue")
        if any(not _is_permutation(scene[key]) for key in keys):
            return True
        if int(scene["start_index"]) not in range(N):
            return True
        if any(scene["left_red"][i] == scene["left_blue"][i] for i in range(N)):
            return True
        if any(scene["right_red"][i] == scene["right_blue"][i] for i in range(N)):
            return True
        return len(_bijections(scene)) != 1
    except Exception:
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []


def _node_label(draw: ImageDraw.ImageDraw, cx: int, cy: int, text: str, start: bool = False) -> None:
    fill = START if start else WHITE
    outline = START if start else INK
    draw.ellipse((cx - 22, cy - 22, cx + 22, cy + 22), fill=fill, outline=outline, width=4)
    bbox = draw.textbbox((0, 0), text, font=_font(24))
    tw, th = bbox[2] - bbox[0], bbox[3] - bbox[1]
    draw.text((cx - tw / 2, cy - th / 2 - 2), text, font=_font(24), fill=WHITE if start else INK)


def _draw_board(
    draw: ImageDraw.ImageDraw,
    x0: int,
    red: list[int],
    blue: list[int],
    labels: tuple[str, ...],
    caption: str,
    start_index: int | None,
) -> None:
    draw.text((x0 + 105, 92), caption, font=_font(29), fill=INK)
    for col in range(N):
        _node_label(draw, x0 + col * CELL + CELL // 2, GRID_Y - 37, labels[col], col == start_index)
    for row in range(N):
        _node_label(draw, x0 - 38, GRID_Y + row * CELL + CELL // 2, labels[row], row == start_index)
    for k in range(N + 1):
        draw.line((x0, GRID_Y + k * CELL, x0 + N * CELL, GRID_Y + k * CELL), fill=GRID, width=2)
        draw.line((x0 + k * CELL, GRID_Y, x0 + k * CELL, GRID_Y + N * CELL), fill=GRID, width=2)
    for row in range(N):
        cy = GRID_Y + row * CELL + CELL // 2
        rcx = x0 + red[row] * CELL + CELL // 2
        bcx = x0 + blue[row] * CELL + CELL // 2
        draw.rectangle((rcx - 15, cy - 15, rcx + 15, cy + 15), fill=RED, outline=INK, width=2)
        draw.ellipse((bcx - 16, cy - 16, bcx + 16, cy + 16), fill=BLUE, outline=INK, width=2)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), PAPER)
    draw = ImageDraw.Draw(image)
    draw.text((365, 22), "SAME SWITCHES, NEW NAMES", font=_font(36), fill=INK)
    _draw_board(
        draw, LEFT_X, scene["left_red"], scene["left_blue"],
        tuple(str(i + 1) for i in range(N)), "ORIGINAL", int(scene["start_index"]),
    )
    _draw_board(
        draw, RIGHT_X, scene["right_red"], scene["right_blue"],
        DECISIONS, "RENAMED", None,
    )
    draw.ellipse((118, 632, 152, 666), fill=START, outline=START, width=2)
    draw.text((163, 632), "START switch", font=_font(22), fill=INK)
    draw.rectangle((394, 634, 424, 664), fill=RED, outline=INK, width=2)
    draw.text((434, 632), "red outgoing link", font=_font(22), fill=INK)
    draw.ellipse((711, 633, 743, 665), fill=BLUE, outline=INK, width=2)
    draw.text((753, 632), "blue outgoing link", font=_font(22), fill=INK)
    return image
