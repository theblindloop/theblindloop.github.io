"""Independent image-only inverse for Renamed Switchboards."""

from __future__ import annotations

import itertools

import numpy as np
from PIL import Image

N = 6
DECISIONS = tuple("ABCDEF")
RED = np.array((218, 52, 62), dtype=np.int16)
BLUE = np.array((35, 105, 210), dtype=np.int16)
START = np.array((200, 38, 145), dtype=np.int16)
CELL = 64
GRID_Y = 205
LEFT_X = 120
RIGHT_X = 755


def _mask(arr: np.ndarray, color: np.ndarray, tolerance: int = 10) -> np.ndarray:
    return np.max(np.abs(arr.astype(np.int16) - color), axis=2) <= tolerance


def _read_board(arr: np.ndarray, x0: int) -> tuple[list[int], list[int]] | None:
    red, blue = [], []
    for row in range(N):
        red_hits, blue_hits = [], []
        for col in range(N):
            y0 = GRID_Y + row * CELL + 10
            x = x0 + col * CELL + 10
            crop = arr[y0:y0 + CELL - 20, x:x + CELL - 20]
            if int(_mask(crop, RED).sum()) > 300:
                red_hits.append(col)
            if int(_mask(crop, BLUE).sum()) > 300:
                blue_hits.append(col)
        if len(red_hits) != 1 or len(blue_hits) != 1 or red_hits[0] == blue_hits[0]:
            return None
        red.append(red_hits[0])
        blue.append(blue_hits[0])
    if sorted(red) != list(range(N)) or sorted(blue) != list(range(N)):
        return None
    return red, blue


def decision_from_image(image: Image.Image) -> str:
    try:
        if image.size != (1280, 720):
            return "abstain"
        arr = np.asarray(image.convert("RGB"))
        left = _read_board(arr, LEFT_X)
        right = _read_board(arr, RIGHT_X)
        if left is None or right is None:
            return "abstain"
        starts = []
        for row in range(N):
            cy = GRID_Y + row * CELL + CELL // 2
            crop = arr[cy - 24:cy + 25, LEFT_X - 64:LEFT_X - 12]
            starts.append(int(_mask(crop, START).sum()) > 800)
        if sum(starts) != 1:
            return "abstain"
        lr, lb = left
        rr, rb = right
        solutions = [
            q for q in itertools.permutations(range(N))
            if all(q[lr[i]] == rr[q[i]] and q[lb[i]] == rb[q[i]] for i in range(N))
        ]
        if len(solutions) != 1:
            return "abstain"
        return DECISIONS[solutions[0][starts.index(1)]]
    except Exception:
        return "abstain"
