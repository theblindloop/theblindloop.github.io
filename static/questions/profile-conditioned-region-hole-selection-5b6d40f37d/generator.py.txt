"""Deterministic renderer and analytic label for the hole-selection world.

The latent scene stores only public, rendered geometry: three disjoint colored
regions, each with a polygon, an open cut channel, and zero, one, or two closed
circular cavities.  The answer is the color whose rendered region has one
enclosed background component.
"""

from __future__ import annotations

import copy
import math
import random
from typing import Any

from PIL import Image, ImageDraw

RENDERER_VERSION = "region_hole_selection-0.1.0"
WIDTH, HEIGHT = 720, 380
BACKGROUND = (247, 247, 243)
COLORS = {
    "red": (210, 63, 70),
    "green": (53, 157, 96),
    "blue": (67, 104, 198),
}
SLOTS = ((125.0, 190.0), (360.0, 190.0), (595.0, 190.0))
SUPERSAMPLE = 3


def _outer_polygon(
    cx: float, cy: float, rx: float, ry: float, phase: float, waviness: float
) -> list[list[float]]:
    points: list[list[float]] = []
    for index in range(20):
        angle = 2.0 * math.pi * index / 20.0
        radial = 1.0 + waviness * (
            0.62 * math.sin(3.0 * angle + phase)
            + 0.38 * math.cos(5.0 * angle - 0.7 * phase)
        )
        points.append(
            [
                cx + rx * radial * math.cos(angle),
                cy + ry * radial * math.sin(angle),
            ]
        )
    return points


def _hole(cx: float, cy: float, x: float, y: float, radius: float) -> dict[str, float]:
    return {"x": float(cx + x), "y": float(cy + y), "r": float(radius)}


def _make_region(
    rng: random.Random,
    color: str,
    slot: tuple[float, float],
    hole_count: int,
) -> dict[str, Any]:
    cx = slot[0] + rng.uniform(-8.0, 8.0)
    cy = slot[1] + rng.uniform(-12.0, 12.0)
    rx = rng.uniform(72.0, 86.0)
    ry = rng.uniform(65.0, 78.0)
    phase = rng.uniform(-math.pi, math.pi)
    waviness = rng.uniform(0.035, 0.085)
    outer = _outer_polygon(cx, cy, rx, ry, phase, waviness)

    holes: list[dict[str, float]] = []
    if hole_count == 1:
        holes.append(
            _hole(
                cx,
                cy,
                rng.uniform(-8.0, -1.0),
                rng.uniform(-8.0, 8.0),
                rng.uniform(15.0, 19.0),
            )
        )
    elif hole_count == 2:
        holes.append(
            _hole(
                cx,
                cy,
                rng.uniform(-25.0, -18.0),
                rng.uniform(-23.0, -16.0),
                rng.uniform(13.0, 17.0),
            )
        )
        holes.append(
            _hole(
                cx,
                cy,
                rng.uniform(-11.0, -4.0),
                rng.uniform(16.0, 23.0),
                rng.uniform(13.0, 17.0),
            )
        )

    # This cut always reaches the exterior. It is intentionally near the right
    # edge, away from the closed cavities, and gives every region a similar
    # white opening that must not be counted as a hole.
    notch_angle = rng.uniform(-0.20, 0.20)
    open_start = rng.uniform(0.38, 0.46) * min(rx, ry)
    open_end = max(rx, ry) + rng.uniform(24.0, 34.0)
    notch_width = rng.uniform(11.0, 15.0)
    clearance_values = []
    for cavity in holes:
        nx = (cavity["x"] - cx) / rx
        ny = (cavity["y"] - cy) / ry
        radial = math.sqrt(nx * nx + ny * ny)
        clearance_values.append((1.0 - radial) * min(rx, ry) - cavity["r"])
    clearance = min(clearance_values, default=32.0)

    return {
        "color": color,
        "outer": outer,
        "holes": holes,
        "notch": {
            "angle": float(notch_angle),
            "start": float(open_start),
            "end": float(open_end),
            "width": float(notch_width),
        },
        "clearance": float(clearance),
    }


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample one balanced scene from a single integer seed."""

    rng = random.Random(int(seed))
    colors = list(COLORS)
    slots = list(SLOTS)
    counts = [0, 1, 2]
    rng.shuffle(colors)
    rng.shuffle(slots)
    rng.shuffle(counts)
    regions = [
        _make_region(rng, colors[index], slots[index], counts[index])
        for index in range(3)
    ]
    return {"regions": regions}


def _scaled_point(point: list[float] | tuple[float, float], scale: int) -> tuple[int, int]:
    return (int(round(point[0] * scale)), int(round(point[1] * scale)))


def _draw_region(draw: ImageDraw.ImageDraw, region: dict[str, Any], scale: int) -> None:
    fill = COLORS[region["color"]]
    outer = [_scaled_point(point, scale) for point in region["outer"]]
    draw.polygon(outer, fill=fill)

    for cavity in region["holes"]:
        x, y, radius = cavity["x"], cavity["y"], cavity["r"]
        box = (
            int(round((x - radius) * scale)),
            int(round((y - radius) * scale)),
            int(round((x + radius) * scale)),
            int(round((y + radius) * scale)),
        )
        draw.ellipse(box, fill=BACKGROUND)

    notch = region["notch"]
    cx = sum(point[0] for point in region["outer"]) / len(region["outer"])
    cy = sum(point[1] for point in region["outer"]) / len(region["outer"])
    direction = (math.cos(notch["angle"]), math.sin(notch["angle"]))
    start = (
        cx + direction[0] * notch["start"],
        cy + direction[1] * notch["start"],
    )
    end = (
        cx + direction[0] * notch["end"],
        cy + direction[1] * notch["end"],
    )
    width = max(1, int(round(notch["width"] * scale)))
    draw.line([_scaled_point(start, scale), _scaled_point(end, scale)], fill=BACKGROUND, width=width)
    cap = width // 2
    sx, sy = _scaled_point(start, scale)
    draw.ellipse((sx - cap, sy - cap, sx + cap, sy + cap), fill=BACKGROUND)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render only the public scene fields into a deterministic RGB PNG image."""

    scale = SUPERSAMPLE
    image = Image.new("RGB", (WIDTH * scale, HEIGHT * scale), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for region in scene["regions"]:
        _draw_region(draw, region, scale)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    """Return the color whose rendered region has exactly one closed cavity."""

    matches = [region["color"] for region in scene["regions"] if len(region["holes"]) == 1]
    if len(matches) != 1:
        raise ValueError(f"scene has {len(matches)} one-hole regions")
    return matches[0]


def margin(scene: dict[str, Any]) -> float:
    """Return the smallest approximate clearance of a closed cavity."""

    return float(min(float(region["clearance"]) for region in scene["regions"]))


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < 6.0


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """Exercise a render-invariant permutation of disjoint regions and cavities."""

    twin = copy.deepcopy(scene)
    twin["regions"] = list(reversed(twin["regions"]))
    for region in twin["regions"]:
        region["holes"] = list(reversed(region["holes"]))
    return [("reverse_disjoint_region_order", twin)]
