"""Independent image-only inverse for the region-hole selection world.

This module intentionally receives a final RGB image and nothing from the
renderer. It labels pixels by the public color palette, finds each largest
colored connected component, flood-fills its exterior background, and counts
background components that remain enclosed.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

BACKGROUND = (247, 247, 243)
PALETTE = {
    "red": (210, 63, 70),
    "green": (53, 157, 96),
    "blue": (67, 104, 198),
}


def _nearest_masks(image: Image.Image) -> dict[str, np.ndarray]:
    pixels = np.asarray(image.convert("RGB"), dtype=np.int32)
    labels = list(PALETTE) + ["background"]
    colors = np.asarray([PALETTE[name] for name in PALETTE] + [BACKGROUND], dtype=np.int32)
    distances = ((pixels[:, :, None, :] - colors[None, None, :, :]) ** 2).sum(axis=3)
    nearest = np.argmin(distances, axis=2)
    return {name: nearest == index for index, name in enumerate(labels[:-1])}


def _largest_component(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    best: list[tuple[int, int]] = []
    for sy, sx in zip(*np.where(mask & ~visited)):
        if visited[sy, sx]:
            continue
        queue: deque[tuple[int, int]] = deque([(int(sy), int(sx))])
        visited[sy, sx] = True
        points: list[tuple[int, int]] = []
        while queue:
            y, x = queue.popleft()
            points.append((y, x))
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        if len(points) > len(best):
            best = points
    component = np.zeros_like(mask, dtype=bool)
    if best:
        ys, xs = zip(*best)
        component[np.asarray(ys), np.asarray(xs)] = True
    return component


def _background_hole_count(component: np.ndarray) -> int:
    ys, xs = np.where(component)
    if len(xs) == 0:
        raise ValueError("missing colored component")
    pad = 2
    y0, y1 = max(0, int(ys.min()) - pad), min(component.shape[0], int(ys.max()) + pad + 1)
    x0, x1 = max(0, int(xs.min()) - pad), min(component.shape[1], int(xs.max()) + pad + 1)
    local_colored = component[y0:y1, x0:x1]
    background = ~local_colored
    outside = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()
    h, w = background.shape
    for x in range(w):
        for y in (0, h - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(h):
        for x in (0, w - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for dy in (-1, 0, 1):
            for dx in (-1, 0, 1):
                if not (dx or dy):
                    continue
                ny, nx = y + dy, x + dx
                if 0 <= ny < h and 0 <= nx < w and background[ny, nx] and not outside[ny, nx]:
                    outside[ny, nx] = True
                    queue.append((ny, nx))

    enclosed = background & ~outside
    visited = np.zeros_like(enclosed, dtype=bool)
    holes = 0
    for sy, sx in zip(*np.where(enclosed)):
        if visited[sy, sx]:
            continue
        component_size = 0
        queue = deque([(int(sy), int(sx))])
        visited[sy, sx] = True
        while queue:
            y, x = queue.popleft()
            component_size += 1
            for dy in (-1, 0, 1):
                for dx in (-1, 0, 1):
                    if not (dx or dy):
                        continue
                    ny, nx = y + dy, x + dx
                    if 0 <= ny < h and 0 <= nx < w and enclosed[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((ny, nx))
        # Antialiased polygon edges can leave isolated background-colored
        # pixels inside the nearest-color mask. Real cavities have hundreds
        # of pixels at the radii used by the renderer; tiny specks are not
        # topological evidence.
        if component_size >= 120:
            holes += 1
    return holes


def decision_from_image(image: Image.Image) -> str:
    """Recover the one-hole color from the raster alone."""

    masks = _nearest_masks(image)
    counts = {
        name: _background_hole_count(_largest_component(mask))
        for name, mask in masks.items()
    }
    matches = [name for name, count in counts.items() if count == 1]
    if len(matches) != 1:
        raise ValueError(f"expected one one-hole region, measured {counts}")
    return matches[0]
