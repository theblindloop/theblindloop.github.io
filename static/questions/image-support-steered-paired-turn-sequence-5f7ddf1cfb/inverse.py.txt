"""Independent pixel-only inverse for two corner-marked routes."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = (23, 50, 77)
GREEN = (42, 157, 91)
RED = (214, 73, 65)
YELLOW = (242, 193, 78)


class OracleAbstention(ValueError):
    pass


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output: list[list[tuple[int, int]]] = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        seen[y, x] = True
        queue = [(y, x)]
        component: list[tuple[int, int]] = []
        while queue:
            cy, cx = queue.pop()
            component.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        output.append(component)
    return output


def _centers(array: np.ndarray, color: tuple[int, int, int], count: int) -> list[tuple[int, int]]:
    mask = np.all(array == np.asarray(color, dtype=np.uint8), axis=2)
    components = [item for item in _components(mask) if len(item) >= 30]
    if len(components) != count:
        raise OracleAbstention(f"expected {count} {color} markers, found {len(components)}")
    centers = []
    for component in components:
        ys = [p[0] for p in component]
        xs = [p[1] for p in component]
        centers.append((round(sum(xs) / len(xs)), round(sum(ys) / len(ys))))
    return centers


def _connector(array: np.ndarray, a: tuple[int, int], b: tuple[int, int]) -> bool:
    ax, ay = a
    bx, by = b
    if ax == bx:
        lo, hi = sorted((ay, by))
        values = array[lo + 9 : hi - 8, ax]
    elif ay == by:
        lo, hi = sorted((ax, bx))
        values = array[ay, lo + 9 : hi - 8]
    else:
        return False
    if len(values) < 12:
        return False
    navy = np.all(values == np.asarray(NAVY, dtype=np.uint8), axis=1)
    # Exact core color is deliberately strict: a materially erased connector
    # causes abstention rather than a guessed answer.
    return bool(navy.mean() >= 0.94)


def _route_turns(
    array: np.ndarray,
    start: tuple[int, int],
    finish: tuple[int, int],
    corners: list[tuple[int, int]],
) -> str:
    nodes = [start, *corners, finish]
    graph = {index: [] for index in range(len(nodes))}
    for i, a in enumerate(nodes):
        for j in range(i + 1, len(nodes)):
            if _connector(array, a, nodes[j]):
                graph[i].append(j)
                graph[j].append(i)
    if len(graph[0]) != 1 or len(graph[len(nodes) - 1]) != 1:
        raise OracleAbstention("route endpoint degree is not one")
    if any(len(graph[index]) != 2 for index in range(1, len(nodes) - 1)):
        raise OracleAbstention("corner graph is not a simple chain")
    order = [0]
    previous = -1
    current = 0
    while current != len(nodes) - 1:
        options = [item for item in graph[current] if item != previous]
        if len(options) != 1 or len(order) > len(nodes):
            raise OracleAbstention("route traversal is ambiguous")
        previous, current = current, options[0]
        if current in order:
            raise OracleAbstention("route graph contains a cycle")
        order.append(current)
    if len(order) != len(nodes):
        raise OracleAbstention("finish reached without all corners")
    points = [nodes[index] for index in order]
    turns: list[str] = []
    for a, b, c in zip(points, points[1:], points[2:]):
        ux, uy = b[0] - a[0], b[1] - a[1]
        vx, vy = c[0] - b[0], c[1] - b[1]
        cross = ux * vy - uy * vx
        if cross == 0:
            raise OracleAbstention("marked corner does not visibly turn")
        turns.append("R" if cross > 0 else "L")
    return "".join(turns)


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or array.shape[1] < 400:
        raise OracleAbstention("unexpected raster")
    green = sorted(_centers(array, GREEN, 2))
    red = sorted(_centers(array, RED, 2))
    yellow = _centers(array, YELLOW, 8)
    midpoint = array.shape[1] // 2
    left_corners = [point for point in yellow if point[0] < midpoint]
    right_corners = [point for point in yellow if point[0] >= midpoint]
    if len(left_corners) != 4 or len(right_corners) != 4:
        raise OracleAbstention("each route must contain four corners")
    left = _route_turns(array, green[0], red[0], left_corners)
    right = _route_turns(array, green[1], red[1], right_corners)
    return "same" if left == right else "different"
