"""Deterministic scene model and renderer for paired turn-sequence comparison."""

from __future__ import annotations

import math
import random
from typing import Iterable

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 640, 360
BACKGROUND = (247, 243, 234)
NAVY = (23, 50, 77)
GREEN = (42, 157, 91)
RED = (214, 73, 65)
YELLOW = (242, 193, 78)
BORDER = (217, 210, 195)
STROKE = 9
MARKER_RADIUS = 8

# Every sequence has four turns, two left and two right. Thus neither turn count
# nor per-route left/right totals predicts the class.
TURN_SEQUENCES = ("LLRR", "LRLR", "LRRL", "RLLR", "RLRL", "RRLL")


def _turns(points: list[list[int]]) -> str:
    symbols: list[str] = []
    for a, b, c in zip(points, points[1:], points[2:]):
        ux, uy = b[0] - a[0], b[1] - a[1]
        vx, vy = c[0] - b[0], c[1] - b[1]
        cross = ux * vy - uy * vx
        if cross == 0:
            raise ValueError("route contains a non-turning corner")
        # Image y grows downward. The naming is consistently inverted from the
        # Cartesian cross sign in both the analytic and independent pixel arms.
        symbols.append("R" if cross > 0 else "L")
    return "".join(symbols)


def _segments(points: list[list[int]]) -> list[tuple[tuple[int, int], tuple[int, int]]]:
    return [((a[0], a[1]), (b[0], b[1])) for a, b in zip(points, points[1:])]


def _point_segment_distance(p: tuple[int, int], seg: tuple[tuple[int, int], tuple[int, int]]) -> float:
    (x, y), ((x1, y1), (x2, y2)) = p, seg
    dx, dy = x2 - x1, y2 - y1
    if dx == dy == 0:
        return math.hypot(x - x1, y - y1)
    t = max(0.0, min(1.0, ((x - x1) * dx + (y - y1) * dy) / (dx * dx + dy * dy)))
    return math.hypot(x - (x1 + t * dx), y - (y1 + t * dy))


def _valid_path(points: list[list[int]], bounds: tuple[int, int, int, int]) -> bool:
    x0, y0, x1, y1 = bounds
    if not all(x0 <= x <= x1 and y0 <= y <= y1 for x, y in points):
        return False
    if len({(x, y) for x, y in points}) != len(points):
        return False
    segs = _segments(points)
    if any(math.dist(a, b) < 34 for a, b in segs):
        return False
    # Non-neighboring strokes stay well separated, excluding ambiguous crossings
    # and near-touches at raster resolution.
    for i, seg_a in enumerate(segs):
        for j, seg_b in enumerate(segs):
            if j <= i + 1:
                continue
            if min(
                _point_segment_distance(seg_a[0], seg_b),
                _point_segment_distance(seg_a[1], seg_b),
                _point_segment_distance(seg_b[0], seg_a),
                _point_segment_distance(seg_b[1], seg_a),
            ) < 24:
                return False
    return True


def _make_path(rng: random.Random, sequence: str, bounds: tuple[int, int, int, int]) -> list[list[int]]:
    directions = ((1, 0), (0, 1), (-1, 0), (0, -1))
    x0, y0, x1, y1 = bounds
    for _ in range(10000):
        heading = rng.randrange(4)
        x = rng.randrange(x0, x1 + 1)
        y = rng.randrange(y0, y1 + 1)
        points = [[x, y]]
        for segment_index in range(5):
            length = rng.randrange(38, 59)
            dx, dy = directions[heading]
            x += dx * length
            y += dy * length
            points.append([x, y])
            if segment_index < 4:
                heading = (heading + (1 if sequence[segment_index] == "R" else -1)) % 4
        if _valid_path(points, bounds):
            return points
    raise RuntimeError("could not place a valid route")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    left_sequence = TURN_SEQUENCES[rng.randrange(len(TURN_SEQUENCES))]
    same = int(seed) % 2 == 0
    if same:
        right_sequence = left_sequence
    else:
        alternatives = [item for item in TURN_SEQUENCES if item != left_sequence]
        right_sequence = alternatives[rng.randrange(len(alternatives))]
    left_points = _make_path(rng, left_sequence, (38, 46, 286, 318))
    right_points = _make_path(rng, right_sequence, (354, 46, 602, 318))
    return {
        "left_points": left_points,
        "right_points": right_points,
        "width": WIDTH,
        "height": HEIGHT,
        "stroke": STROKE,
        "marker_radius": MARKER_RADIUS,
    }


def _draw_route(draw: ImageDraw.ImageDraw, points: list[list[int]], stroke: int, radius: int) -> None:
    xy = [(int(x), int(y)) for x, y in points]
    draw.line(xy, fill=NAVY, width=stroke, joint="curve")
    # Corner squares make the bend event explicitly visible and remove a hidden
    # curvature/angle threshold from the question.
    for x, y in xy[1:-1]:
        draw.rectangle((x - radius, y - radius, x + radius, y + radius), fill=YELLOW)
    for (x, y), color in ((xy[0], GREEN), (xy[-1], RED)):
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=color)


def render(scene: dict) -> Image.Image:
    width, height = int(scene["width"]), int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((18, 22, 304, 338), radius=14, outline=BORDER, width=2)
    draw.rounded_rectangle((336, 22, 622, 338), radius=14, outline=BORDER, width=2)
    _draw_route(draw, scene["left_points"], int(scene["stroke"]), int(scene["marker_radius"]))
    _draw_route(draw, scene["right_points"], int(scene["stroke"]), int(scene["marker_radius"]))
    return image


def analytic_gold(scene: dict) -> str:
    left = _turns(scene["left_points"])
    right = _turns(scene["right_points"])
    return "same" if left == right else "different"


def margin(scene: dict) -> float:
    points = scene["left_points"] + scene["right_points"]
    radius = float(scene["marker_radius"])
    clear_runs = [math.dist(a, b) - 2.0 * radius for a, b in zip(points, points[1:])]
    # Pixel clearance of the shortest visible connector is the runner-up margin:
    # corruption must consume this run before the ordered marker graph can change.
    return float(min(clear_runs))


def is_quarantined(scene: dict) -> bool:
    left = scene["left_points"]
    right = scene["right_points"]
    if not _valid_path(left, (28, 36, 296, 328)):
        return True
    if not _valid_path(right, (344, 36, 612, 328)):
        return True
    return margin(scene) < 16.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
