"""Deterministic latent sampler, analytic arm, and raster renderer."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw

_BASE_NODES = [
    (256.0, 256.0),  # shared articulation
    (176.0, 176.0),
    (96.0, 256.0),
    (176.0, 336.0),
    (336.0, 176.0),
    (416.0, 256.0),
    (336.0, 336.0),
]
_EDGES = [(0, 1), (1, 2), (2, 3), (3, 0), (0, 4), (4, 5), (5, 6), (6, 0)]
_BASE_COLORS = ["#376eb8", "#d17a2d", "#287f9a"]


def _transform_nodes(rng: random.Random) -> list[list[float]]:
    turns = rng.randrange(4)
    mirror = rng.choice((False, True))
    scale = rng.uniform(0.90, 1.06)
    dx = rng.uniform(-10.0, 10.0)
    dy = rng.uniform(-10.0, 10.0)
    result: list[list[float]] = []
    for x, y in _BASE_NODES:
        x -= 256.0
        y -= 256.0
        if mirror:
            x = -x
        for _ in range(turns):
            x, y = -y, x
        result.append([round(256.0 + dx + scale * x, 3), round(256.0 + dy + scale * y, 3)])
    return result


def sample_scene(seed: int) -> dict:
    # Adjacent seeds are a counterbalanced pair: identical geometry and ink,
    # with equal-distance degree-two marker pairs selecting opposite answers.
    rng = random.Random(seed // 2)
    nodes = _transform_nodes(rng)
    selector = rng.randrange(2)
    if seed % 2 == 0:
        marked = (1, 3) if selector == 0 else (4, 6)  # same lobe
    else:
        marked = (1, 4) if selector == 0 else (3, 6)  # different lobes
    if rng.choice((False, True)):
        marked = marked[::-1]
    return {
        "nodes": nodes,
        "edges": [list(edge) for edge in _EDGES],
        "green": marked[0],
        "purple": marked[1],
        "node_radius": rng.choice((17, 18, 19)),
        "stroke_width": rng.choice((9, 10, 12, 13)),
        "base_color": rng.choice(_BASE_COLORS),
        "background": "#faf8f2",
        "edge_color": "#252a31",
        "canvas_size": 512,
    }


def _biconnected_components(node_count: int, edges: list[list[int]]) -> list[set[int]]:
    adjacency = [[] for _ in range(node_count)]
    for a, b in edges:
        adjacency[a].append(b)
        adjacency[b].append(a)
    disc = [-1] * node_count
    low = [0] * node_count
    parent = [-1] * node_count
    stack: list[tuple[int, int]] = []
    components: list[set[int]] = []
    clock = 0

    def visit(u: int) -> None:
        nonlocal clock
        disc[u] = low[u] = clock
        clock += 1
        for v in adjacency[u]:
            if disc[v] == -1:
                parent[v] = u
                stack.append((u, v))
                visit(v)
                low[u] = min(low[u], low[v])
                if low[v] >= disc[u]:
                    vertices: set[int] = set()
                    while stack:
                        edge = stack.pop()
                        vertices.update(edge)
                        if edge == (u, v):
                            break
                    components.append(vertices)
            elif v != parent[u] and disc[v] < disc[u]:
                low[u] = min(low[u], disc[v])
                stack.append((u, v))

    for start in range(node_count):
        if disc[start] == -1:
            visit(start)
    return components


def _common_loop(scene: dict) -> bool:
    edges = scene["edges"]
    green = scene["green"]
    purple = scene["purple"]
    node_count = 1 + max(max(edge) for edge in edges)
    return any(
        len(component) >= 3 and green in component and purple in component
        for component in _biconnected_components(node_count, edges)
    )


def analytic_gold(scene: dict) -> str:
    return "yes" if _common_loop(scene) else "no"


def _point_segment_distance(point: list[float], a: list[float], b: list[float]) -> float:
    px, py = point
    ax, ay = a
    bx, by = b
    vx, vy = bx - ax, by - ay
    denom = vx * vx + vy * vy
    t = 0.0 if denom == 0 else max(0.0, min(1.0, ((px - ax) * vx + (py - ay) * vy) / denom))
    return math.hypot(px - (ax + t * vx), py - (ay + t * vy))


def margin(scene: dict) -> float:
    nodes = scene["nodes"]
    radius = float(scene["node_radius"])
    stroke = float(scene["stroke_width"])
    edge_set = {tuple(sorted(edge)) for edge in scene["edges"]}
    node_gap = min(
        math.dist(nodes[i], nodes[j]) - 2.0 * radius
        for i in range(len(nodes))
        for j in range(i + 1, len(nodes))
    )
    nonincident = []
    for i, point in enumerate(nodes):
        for a, b in edge_set:
            if i not in (a, b):
                nonincident.append(_point_segment_distance(point, nodes[a], nodes[b]) - radius - stroke / 2)
    return round(min(stroke / 2.0, node_gap / 4.0, min(nonincident) / 3.0), 6)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < 4.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    # Relabel the three noncentral vertices of the two lobes. Edge-list order
    # and each stroke's geometric endpoint order stay byte-for-byte equivalent;
    # only the internal indices change.
    new_order = [0, 4, 5, 6, 1, 2, 3]
    inverse = {old: new for new, old in enumerate(new_order)}
    twin = copy.deepcopy(scene)
    twin["nodes"] = [copy.deepcopy(scene["nodes"][old]) for old in new_order]
    twin["edges"] = [[inverse[a], inverse[b]] for a, b in scene["edges"]]
    twin["green"] = inverse[scene["green"]]
    twin["purple"] = inverse[scene["purple"]]
    return [("permute_internal_vertex_indices", twin)]


def render(scene: dict) -> Image.Image:
    nodes = scene["nodes"]
    edges = scene["edges"]
    green = scene["green"]
    purple = scene["purple"]
    radius = scene["node_radius"]
    stroke = scene["stroke_width"]
    base = scene["base_color"]
    background = scene["background"]
    edge_color = scene["edge_color"]
    size = scene["canvas_size"]
    ss = 3
    image = Image.new("RGB", (size * ss, size * ss), background)
    draw = ImageDraw.Draw(image)
    scaled = [(round(x * ss), round(y * ss)) for x, y in nodes]
    for a, b in edges:
        draw.line([scaled[a], scaled[b]], fill=edge_color, width=stroke * ss)
    colors = [base] * len(nodes)
    colors[green] = "#249a61"
    colors[purple] = "#9950c7"
    rr = radius * ss
    outline = max(2, 2 * ss)
    for (x, y), color in zip(scaled, colors):
        draw.ellipse((x - rr, y - rr, x + rr, y + rr), fill=color, outline="#17202a", width=outline)
    return image.resize((size, size), Image.Resampling.LANCZOS)
