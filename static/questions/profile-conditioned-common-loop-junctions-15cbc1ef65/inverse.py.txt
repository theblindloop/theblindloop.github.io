"""Independent image-only inverse arm for the common-loop decision."""

from __future__ import annotations

import hashlib
import math
from collections import deque

import numpy as np
from PIL import Image

_GREEN = np.array([36.0, 154.0, 97.0])
_PURPLE = np.array([153.0, 80.0, 199.0])
_CACHE: dict[str, str] = {}


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    groups: list[np.ndarray] = []
    for yy, xx in np.argwhere(mask):
        y, x = int(yy), int(xx)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        pixels = []
        while queue:
            cy, cx = queue.popleft()
            pixels.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(pixels) >= 250:
            groups.append(np.asarray(pixels, dtype=np.int32))
    return groups


def _biconnected(node_count: int, edges: list[tuple[int, int]]) -> list[set[int]]:
    adjacency = [[] for _ in range(node_count)]
    for a, b in edges:
        adjacency[a].append(b)
        adjacency[b].append(a)
    disc = [-1] * node_count
    low = [0] * node_count
    parent = [-1] * node_count
    stack: list[tuple[int, int]] = []
    blocks: list[set[int]] = []
    clock = 0

    def visit(u: int) -> None:
        nonlocal clock
        disc[u] = low[u] = clock
        clock += 1
        for v in adjacency[u]:
            if disc[v] < 0:
                parent[v] = u
                stack.append((u, v))
                visit(v)
                low[u] = min(low[u], low[v])
                if low[v] >= disc[u]:
                    block: set[int] = set()
                    while stack:
                        edge = stack.pop()
                        block.update(edge)
                        if edge == (u, v):
                            break
                    blocks.append(block)
            elif v != parent[u] and disc[v] < disc[u]:
                low[u] = min(low[u], disc[v])
                stack.append((u, v))

    for start in range(node_count):
        if disc[start] < 0:
            visit(start)
    return blocks


def _stroke_present(dark: np.ndarray, a: tuple[float, float], b: tuple[float, float]) -> bool:
    hits = 0
    samples = 29
    for t in np.linspace(0.22, 0.78, samples):
        x = int(round(a[0] + t * (b[0] - a[0])))
        y = int(round(a[1] + t * (b[1] - a[1])))
        patch = dark[max(0, y - 3) : y + 4, max(0, x - 3) : x + 4]
        hits += bool(patch.size and float(patch.mean()) > 0.18)
    return hits >= 26


def _measure(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    chroma = arr.max(axis=2) - arr.min(axis=2)
    colored = (chroma > 38.0) & (arr.max(axis=2) > 90.0)
    groups = _components(colored)
    if len(groups) != 7:
        raise ValueError(f"expected seven visible junctions, recovered {len(groups)}")
    centers: list[tuple[float, float]] = []
    colors: list[np.ndarray] = []
    for pixels in groups:
        centers.append((float(pixels[:, 1].mean()), float(pixels[:, 0].mean())))
        colors.append(arr[pixels[:, 0], pixels[:, 1]].mean(axis=0))
    green = int(np.argmin([np.linalg.norm(color - _GREEN) for color in colors]))
    purple = int(np.argmin([np.linalg.norm(color - _PURPLE) for color in colors]))
    if green == purple:
        raise ValueError("marked junction colors are not independently visible")
    if np.linalg.norm(colors[green] - _GREEN) > 75 or np.linalg.norm(colors[purple] - _PURPLE) > 75:
        raise ValueError("green or purple marker is missing")

    distances = np.full((len(centers), len(centers)), np.inf)
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            distances[i, j] = distances[j, i] = math.dist(centers[i], centers[j])
    unit = float(np.median(distances.min(axis=1)))
    if not 70.0 <= unit <= 145.0:
        raise ValueError(f"junction spacing is out of range: {unit:.2f}")
    dark = (arr.max(axis=2) < 105.0) & (arr.mean(axis=2) < 80.0)
    edges: list[tuple[int, int]] = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            if 0.72 * unit <= distances[i, j] <= 1.25 * unit and _stroke_present(
                dark, centers[i], centers[j]
            ):
                edges.append((i, j))
    if len(edges) != 8:
        raise ValueError(f"expected eight visible links, recovered {len(edges)}")
    answer = any(
        len(block) >= 3 and green in block and purple in block
        for block in _biconnected(len(centers), edges)
    )
    return "yes" if answer else "no"


def decision_from_image(image: Image.Image) -> str:
    pixels = image.convert("RGB").tobytes()
    key = hashlib.sha256(pixels).hexdigest()
    if key not in _CACHE:
        if len(_CACHE) > 256:
            _CACHE.clear()
        _CACHE[key] = _measure(image)
    return _CACHE[key]
