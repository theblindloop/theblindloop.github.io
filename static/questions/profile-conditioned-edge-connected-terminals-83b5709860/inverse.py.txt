"""Independent final-PNG inverse arm for terminal reachability."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

LABELS = "ABCD"


def _components(mask: np.ndarray, minimum: int) -> list[tuple[int, int, int, int, np.ndarray]]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    parts = []
    for y0, x0 in zip(*np.nonzero(mask)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(x0), int(y0))])
        seen[y0, x0] = True
        pixels = []
        while queue:
            x, y = queue.popleft()
            pixels.append((x, y))
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((nx, ny))
        if len(pixels) < minimum:
            continue
        xs = [p[0] for p in pixels]
        ys = [p[1] for p in pixels]
        left, top, right, bottom = min(xs), min(ys), max(xs), max(ys)
        crop = np.zeros((bottom - top + 1, right - left + 1), dtype=bool)
        for x, y in pixels:
            crop[y - top, x - left] = True
        parts.append((left, top, right, bottom, crop))
    return parts


def _one_component_joins(passable: np.ndarray, first: np.ndarray, second: np.ndarray) -> bool:
    seen = np.zeros_like(passable, dtype=bool)
    height, width = passable.shape
    for y0, x0 in zip(*np.nonzero(passable & first)):
        if seen[y0, x0]:
            continue
        queue = deque([(int(x0), int(y0))])
        seen[y0, x0] = True
        touches_second = False
        count = 0
        while queue:
            x, y = queue.popleft()
            count += 1
            touches_second = touches_second or bool(second[y, x])
            for nx, ny in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if 0 <= nx < width and 0 <= ny < height and passable[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((nx, ny))
        if count >= 150 and touches_second:
            return True
    return False


def decision_from_image(image: Image.Image) -> str:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    frame_mask = (rgb[:, :, 0] >= 35) & (rgb[:, :, 0] <= 75) & (rgb[:, :, 1] >= 105) & (rgb[:, :, 1] <= 145) & (rgb[:, :, 2] >= 135) & (rgb[:, :, 2] <= 175)
    frames = [part for part in _components(frame_mask, minimum=1000) if part[2] - part[0] > 300 and part[3] - part[1] > 240]
    if len(frames) != 4:
        raise ValueError(f"expected four visible panel frames, found {len(frames)}")
    frames.sort(key=lambda part: ((part[1] + part[3]) / 2, (part[0] + part[2]) / 2))

    dark = np.max(rgb, axis=2) < 90
    magenta = (rgb[:, :, 0] > 170) & (rgb[:, :, 1] < 90) & (rgb[:, :, 2] > 100) & (rgb[:, :, 2] < 175)
    amber = (rgb[:, :, 0] > 210) & (rgb[:, :, 1] > 125) & (rgb[:, :, 1] < 200) & (rgb[:, :, 2] < 80)
    winners = []
    for index, frame in enumerate(frames):
        left, top, right, bottom = frame[:4]
        inset = np.zeros_like(dark)
        inset[top + 4:bottom - 3, left + 4:right - 3] = True
        first, second = magenta & inset, amber & inset
        if int(first.sum()) < 80 or int(second.sum()) < 80:
            raise ValueError("each panel must expose both colored terminals")
        passable = (dark | magenta | amber) & inset
        if _one_component_joins(passable, first, second):
            winners.append(index)
    if len(winners) != 1:
        raise ValueError("image does not contain one unique connected terminal pair")
    return LABELS[winners[0]]
