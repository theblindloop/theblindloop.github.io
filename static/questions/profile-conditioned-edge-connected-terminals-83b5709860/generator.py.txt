"""Deterministic renderer for visual search over terminal connectivity."""

from __future__ import annotations

from collections import deque
import copy
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 760, 700
LABELS = "ABCD"
GRID = 11
TRACK_INKS = ((24, 28, 34), (48, 27, 54), (24, 54, 48), (59, 43, 24))
FRAME = (55, 126, 156)
BACKGROUND = (248, 247, 243)
MAGENTA = (205, 55, 135)
AMBER = (240, 165, 35)


def _path(rng: random.Random) -> list[tuple[int, int]]:
    """An x-monotone simple path with exactly six vertical steps."""
    while True:
        change_cols = set(rng.sample(range(10), 6))
        ys = [5]
        for x in range(10):
            y = ys[-1]
            if x in change_cols:
                choices = [step for step in (-1, 1) if 4 <= y + step <= 6]
                y += rng.choice(choices)
            ys.append(y)
        if sum(abs(ys[i + 1] - ys[i]) for i in range(10)) != 6:
            continue
        cells = [(0, ys[0])]
        for x in range(10):
            current = cells[-1][1]
            next_y = ys[x + 1]
            if next_y != current:
                cells.append((x, next_y))
            cells.append((x + 1, next_y))
        if len(cells) == 17 and len(set(cells)) == 17:
            return cells


def _target_decoys(rng: random.Random) -> set[tuple[int, int]]:
    top_x = rng.randint(1, 7)
    bottom_x = rng.randint(1, 7)
    top = {(top_x, 0), (top_x + 1, 0), (top_x + 1, 1)}
    bottom = {(bottom_x, 9), (bottom_x + 1, 9), (bottom_x + 2, 9), (bottom_x + 2, 10)}
    return top | bottom


def _runner_decoy(rng: random.Random) -> set[tuple[int, int]]:
    x = rng.randint(1, 6)
    shape = {(x, 0), (x + 1, 0), (x + 2, 0), (x, 1), (x + 1, 1), (x + 2, 1), (x + 1, 2), (x + 2, 2)}
    if rng.randrange(2):
        return {(px, 10 - py) for px, py in shape}
    return shape


def _reachable(cells: set[tuple[int, int]], start: tuple[int, int], goal: tuple[int, int]) -> bool:
    seen = {start}
    queue = deque([start])
    while queue:
        x, y = queue.popleft()
        if (x, y) == goal:
            return True
        for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
            if neighbor in cells and neighbor not in seen:
                seen.add(neighbor)
                queue.append(neighbor)
    return False


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    answer_index = int(seed) % 4
    panels = []
    for index in range(4):
        path = _path(rng)
        magenta, amber = path[0], path[-1]
        if index == answer_index:
            cells = set(path) | _target_decoys(rng)
        else:
            articulation_indices = [
                position for position in range(2, len(path) - 2)
                if not _reachable(set(path) - {path[position]}, path[0], path[-1])
            ]
            break_index = rng.choice(articulation_indices)
            cells = (set(path) - {path[break_index]}) | _runner_decoy(rng)
        if len(cells) != 24:
            raise RuntimeError("tile-count invariant failed")
        panels.append({
            "cells": [[x, y] for x, y in sorted(cells)],
            "magenta": [magenta[0], magenta[1]],
            "amber": [amber[0], amber[1]],
        })
    return {
        "panels": panels,
        "palette": {
            "track": list(TRACK_INKS[rng.randrange(len(TRACK_INKS))]),
            "frame": list(FRAME),
            "background": list(BACKGROUND),
            "magenta": list(MAGENTA),
            "amber": list(AMBER),
        },
        "cell_size": 18,
        "layout_dx": rng.randint(-7, 7),
        "layout_dy": rng.randint(-5, 5),
    }


def render(scene: dict) -> Image.Image:
    palette = scene["palette"]
    cell = int(scene["cell_size"])
    dx, dy = int(scene["layout_dx"]), int(scene["layout_dy"])
    image = Image.new("RGB", (WIDTH, HEIGHT), tuple(palette["background"]))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default(size=20)
    boxes = [
        (25 + dx, 45 + dy, 365 + dx, 325 + dy),
        (395 + dx, 45 + dy, 735 + dx, 325 + dy),
        (25 + dx, 375 + dy, 365 + dx, 655 + dy),
        (395 + dx, 375 + dy, 735 + dx, 655 + dy),
    ]
    for index, (box, panel) in enumerate(zip(boxes, scene["panels"], strict=True)):
        frame = tuple(palette["frame"])
        draw.rectangle(box, outline=frame, width=3)
        draw.text((box[0] + 12, box[1] + 9), LABELS[index], fill=frame, font=font)
        left = box[0] + (box[2] - box[0] - GRID * cell) // 2
        top = box[1] + 52
        track = tuple(palette["track"])
        for x, y in panel["cells"]:
            x0, y0 = left + int(x) * cell, top + int(y) * cell
            draw.rectangle((x0, y0, x0 + cell - 1, y0 + cell - 1), fill=track)
        radius = 6
        for key in ("magenta", "amber"):
            x, y = panel[key]
            cx, cy = left + int(x) * cell + cell // 2, top + int(y) * cell + cell // 2
            fill = tuple(palette[key])
            draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=fill)
    return image


def _components(cells: set[tuple[int, int]]) -> list[set[tuple[int, int]]]:
    unseen = set(cells)
    parts = []
    while unseen:
        start = min(unseen)
        unseen.remove(start)
        part = {start}
        queue = deque([start])
        while queue:
            x, y = queue.popleft()
            for neighbor in ((x - 1, y), (x + 1, y), (x, y - 1), (x, y + 1)):
                if neighbor in unseen:
                    unseen.remove(neighbor)
                    part.add(neighbor)
                    queue.append(neighbor)
        parts.append(part)
    return parts


def _panel_connected(_audit_parent: object, panel: dict) -> bool:
    cells = {tuple(cell) for cell in panel["cells"]}
    terminals = {tuple(panel["magenta"]), tuple(panel["amber"])}
    return any(terminals <= part for part in _components(cells))


def _missing_tile_gap(_audit_parent: object, panel: dict) -> int:
    cells = {tuple(cell) for cell in panel["cells"]}
    parts = _components(cells)
    a, b = tuple(panel["magenta"]), tuple(panel["amber"])
    left = next(part for part in parts if a in part)
    right = next(part for part in parts if b in part)
    if left is right:
        return 0
    return min(abs(x1 - x2) + abs(y1 - y2) - 1 for x1, y1 in left for x2, y2 in right)


def analytic_gold(scene: dict) -> str:
    winners = [index for index, panel in enumerate(scene["panels"]) if _panel_connected(scene, panel)]
    if len(winners) != 1:
        raise ValueError("scene must contain exactly one connected terminal pair")
    return LABELS[winners[0]]


def margin(scene: dict) -> float:
    winner = LABELS.index(analytic_gold(scene))
    return float(min(_missing_tile_gap(scene, panel) for index, panel in enumerate(scene["panels"]) if index != winner))


def is_quarantined(scene: dict) -> bool:
    try:
        return margin(scene) < 1.0
    except (ValueError, TypeError, StopIteration):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    for panel in twin["panels"]:
        panel["cells"] = list(reversed(panel["cells"]))
    return [("reverse_occupied_cell_serialization", twin)]
