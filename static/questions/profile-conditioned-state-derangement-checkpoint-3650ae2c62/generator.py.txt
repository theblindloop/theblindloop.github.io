"""Deterministic raster world for a whole-configuration state derangement.

The latent scene is deliberately small: four immutable identity glyphs carry one
of three keyed states in six ordered panels.  Rows are shuffled independently so
the answer cannot be read from a row position.  A target panel preserves the
P1 state-count vector while changing the state of every identity.
"""

from __future__ import annotations

import random
from collections import Counter
from pathlib import Path
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont


WIDTH, HEIGHT = 1880, 900
PANEL_W, PANEL_H = 285, 680
PANEL_GAP, LEFT = 18, 30
PANEL_TOP = 185
ROW_TOP, ROW_STEP, CARD_H = 280, 126, 94
CARD_X, CARD_W = 18, 249
ICON_CX, ICON_HALF = 50, 20
SWATCH_X, SWATCH_W = 155, 60

BACKGROUND = (247, 248, 250)
INK = (35, 43, 55)
MUTED = (92, 104, 119)
CARD = (255, 255, 255)
BASE_STATE_COLORS = ((224, 86, 91), (68, 140, 224), (73, 177, 119))
PANEL_TINTS = (
    (235, 242, 250),
    (244, 239, 249),
    (238, 247, 241),
    (249, 243, 232),
    (237, 246, 247),
    (246, 238, 240),
)


def _font(size: int) -> ImageFont.FreeTypeFont | ImageFont.ImageFont:
    """Use a fixed installed font, with Pillow's deterministic fallback."""

    for path in (
        "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf",
        "/usr/share/fonts/dejavu/DejaVuSans.ttf",
    ):
        if Path(path).is_file():
            return ImageFont.truetype(path, size)
    return ImageFont.load_default()


FONT_TITLE = _font(27)
FONT_LABEL = _font(21)
FONT_SMALL = _font(17)
FONT_PANEL = _font(25)


def _state_counts(values: Iterable[int]) -> Counter[int]:
    return Counter(int(value) for value in values)


def _is_derangement(initial: list[int], later: list[int]) -> bool:
    return (
        len(initial) == len(later) == 4
        and all(int(a) != int(b) for a, b in zip(initial, later))
        and _state_counts(initial) == _state_counts(later)
    )


def _find_derangement(initial: list[int], rng: random.Random) -> list[int] | None:
    for _ in range(800):
        candidate = list(initial)
        rng.shuffle(candidate)
        if _is_derangement(initial, candidate):
            return candidate
    return None


def sample_scene(seed: int) -> dict:
    """Sample one balanced six-panel scene from an integer seed."""

    rng = random.Random(int(seed))
    for _ in range(300):
        initial = [rng.randrange(3) for _ in range(4)]
        target = _find_derangement(initial, rng)
        if target is None:
            continue
        target_panel = rng.randrange(1, 6)
        panels: list[list[int]] = []
        for panel_index in range(6):
            if panel_index == 0:
                panels.append(list(initial))
            elif panel_index == target_panel:
                panels.append(list(target))
            else:
                panels.append([rng.randrange(3) for _ in range(4)])
        if sum(_is_derangement(initial, panel) for panel in panels[1:]) != 1:
            continue

        row_orders: list[list[int]] = []
        for _panel in range(6):
            order = list(range(4))
            rng.shuffle(order)
            row_orders.append(order)

        palette = [list(color) for color in BASE_STATE_COLORS]
        rng.shuffle(palette)
        tints = [list(PANEL_TINTS[(rng.randrange(len(PANEL_TINTS)))]) for _ in range(6)]
        return {
            "panel_states": panels,
            "row_orders": row_orders,
            "state_palette": palette,
            "panel_tints": tints,
        }
    raise RuntimeError("could not construct a unique derangement scene")


def _derangement_panels(panel_states: list[list[int]]) -> list[int]:
    initial = [int(value) for value in panel_states[0]]
    return [
        index
        for index, later in enumerate(panel_states[1:], start=1)
        if _is_derangement(initial, [int(value) for value in later])
    ]


def analytic_gold(scene: dict) -> str:
    """Return the first satisfying panel, or an explicit quarantine token."""

    hits = _derangement_panels(scene["panel_states"])
    return f"P{hits[0] + 1}" if len(hits) == 1 else "QUARANTINED"


def margin(scene: dict) -> float:
    """One-sided uniqueness margin: one for a unique hit, zero otherwise."""

    hits = _derangement_panels(scene["panel_states"])
    return 1.0 if len(hits) == 1 else 0.0


def is_quarantined(scene: dict) -> bool:
    return len(_derangement_panels(scene["panel_states"])) != 1


def _panel_x(panel_index: int) -> int:
    return LEFT + panel_index * (PANEL_W + PANEL_GAP)


def _draw_shape(draw: ImageDraw.ImageDraw, cx: int, cy: int, identity: int, size: int = ICON_HALF) -> None:
    if identity == 0:  # circle
        draw.ellipse((cx - size, cy - size, cx + size, cy + size), fill=INK)
    elif identity == 1:  # triangle
        draw.polygon(((cx, cy - size), (cx - size, cy + size), (cx + size, cy + size)), fill=INK)
    elif identity == 2:  # square
        draw.rectangle((cx - size, cy - size, cx + size, cy + size), fill=INK)
    elif identity == 3:  # diamond
        draw.polygon(((cx, cy - size), (cx + size, cy), (cx, cy + size), (cx - size, cy)), fill=INK)
    else:
        raise ValueError(f"unknown identity {identity}")


def _draw_state_key(draw: ImageDraw.ImageDraw, palette: list[list[int]]) -> None:
    draw.text((42, 25), "STATE KEY", font=FONT_TITLE, fill=INK)
    names = ("LOW", "MID", "HIGH")
    for state, name in enumerate(names):
        x = 48 + state * 128
        draw.rounded_rectangle((x, 73, x + 34, 107), radius=7, fill=tuple(palette[state]), outline=INK, width=2)
        draw.text((x + 47, 78), name, font=FONT_LABEL, fill=INK)


def _draw_identity_key(draw: ImageDraw.ImageDraw) -> None:
    draw.text((500, 25), "ID KEY", font=FONT_TITLE, fill=INK)
    names = ("A", "B", "C", "D")
    for identity, name in enumerate(names):
        x = 605 + identity * 105
        _draw_shape(draw, x, 91, identity, size=15)
        draw.text((x + 27, 78), name, font=FONT_LABEL, fill=INK)
    draw.text((1045, 78), "circle=A  triangle=B  square=C  diamond=D", font=FONT_SMALL, fill=MUTED)


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    panel_index: int,
    states: list[int],
    row_order: list[int],
    palette: list[list[int]],
    tint: list[int],
) -> None:
    px = _panel_x(panel_index)
    draw.rounded_rectangle(
        (px, PANEL_TOP, px + PANEL_W, PANEL_TOP + PANEL_H),
        radius=18,
        fill=tuple(tint),
        outline=(178, 187, 198),
        width=2,
    )
    draw.text((px + 18, 202), f"P{panel_index + 1}", font=FONT_PANEL, fill=INK)
    if panel_index == 0:
        draw.text((px + 78, 207), "initial", font=FONT_SMALL, fill=MUTED)
    else:
        draw.text((px + 78, 207), "checkpoint", font=FONT_SMALL, fill=MUTED)
    draw.line((px + 18, 250, px + PANEL_W - 18, 250), fill=(196, 204, 214), width=2)
    for row, identity in enumerate(row_order):
        y = ROW_TOP + row * ROW_STEP
        x = px + CARD_X
        draw.rounded_rectangle((x, y, x + CARD_W, y + CARD_H), radius=14, fill=CARD, outline=(203, 211, 220), width=2)
        _draw_shape(draw, px + ICON_CX, y + CARD_H // 2, identity)
        draw.text((px + 77, y + 30), f"{('A', 'B', 'C', 'D')[identity]}", font=FONT_LABEL, fill=INK)
        state = int(states[identity])
        sx = px + SWATCH_X
        draw.rounded_rectangle((sx, y + 22, sx + SWATCH_W, y + 72), radius=9, fill=tuple(palette[state]), outline=INK, width=2)
        draw.text((sx + 76, y + 31), "state", font=FONT_SMALL, fill=MUTED)


def render(scene: dict) -> Image.Image:
    """Render only the public visual fields of a scene to an RGB PNG image."""

    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((42, 135), "Compare each later configuration with P1: every identity must change, but state totals must stay fixed.", font=FONT_SMALL, fill=MUTED)
    _draw_state_key(draw, scene["state_palette"])
    _draw_identity_key(draw)
    draw.text((1535, 32), "P1  →  P6", font=FONT_LABEL, fill=INK)
    for panel_index in range(6):
        _draw_panel(
            draw,
            panel_index,
            scene["panel_states"][panel_index],
            scene["row_orders"][panel_index],
            scene["state_palette"],
            scene["panel_tints"][panel_index],
        )
    return image


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """No aliases are claimed: all answer-bearing scene fields are visible."""

    return []
