"""Independent pixel-only inverse for the derangement checkpoint world.

This module intentionally duplicates the public layout constants and uses only
the raster.  It does not import or inspect renderer state, latent scenes, or
analytic gold.  Identity is recovered from four different dark glyph shapes;
state is recovered by sampling each colored swatch and the visible state key.
"""

from __future__ import annotations

import math

import numpy as np
from PIL import Image


_WIDTH, _HEIGHT = 1880, 900
_PANEL_W, _PANEL_GAP, _LEFT = 285, 18, 30
_PANEL_TOP = 185
_ROW_TOP, _ROW_STEP, _CARD_H = 280, 126, 94
_ICON_CX, _ICON_HALF = 50, 20
_SWATCH_X, _SWATCH_W = 155, 60
_KEY_X, _KEY_Y, _KEY_STEP = 48, 73, 128
_IDENTITIES = ("A", "B", "C", "D")


def _panel_x(index: int) -> int:
    return _LEFT + index * (_PANEL_W + _PANEL_GAP)


def _sample_key_colors(arr: np.ndarray) -> list[np.ndarray]:
    colors = []
    for state in range(3):
        x = _KEY_X + state * _KEY_STEP + 17
        colors.append(arr[_KEY_Y + 17, x].astype(float))
    return colors


def _classify_identity(arr: np.ndarray, panel_x: int, row_y: int) -> int:
    x0 = panel_x + _ICON_CX - _ICON_HALF - 7
    x1 = panel_x + _ICON_CX + _ICON_HALF + 8
    y0 = row_y + _CARD_H // 2 - _ICON_HALF - 7
    y1 = row_y + _CARD_H // 2 + _ICON_HALF + 8
    crop = arr[y0:y1, x0:x1]
    mask = crop.max(axis=2) < 115
    ys, xs = np.nonzero(mask)
    if len(xs) < 80:
        raise ValueError("identity glyph is missing or too faint")
    bw = int(xs.max() - xs.min() + 1)
    bh = int(ys.max() - ys.min() + 1)
    box_area = max(1, bw * bh)
    fill_ratio = len(xs) / box_area
    centroid_y = float(ys.mean() - ys.min()) / max(1, bh - 1)
    if fill_ratio > 0.88:
        return 2  # square
    if fill_ratio > 0.67:
        return 0  # circle
    if centroid_y > 0.57:
        return 1  # triangle
    return 3  # diamond


def _nearest_state(pixel: np.ndarray, key_colors: list[np.ndarray]) -> int:
    distances = [float(np.linalg.norm(pixel.astype(float) - color)) for color in key_colors]
    state = int(np.argmin(distances))
    if distances[state] > 35.0:
        raise ValueError("state swatch does not match the visible state key")
    return state


def _decode_panel(arr: np.ndarray, panel_x: int, key_colors: list[np.ndarray]) -> list[int]:
    states = [-1, -1, -1, -1]
    for row in range(4):
        y = _ROW_TOP + row * _ROW_STEP
        identity = _classify_identity(arr, panel_x, y)
        sx = panel_x + _SWATCH_X + _SWATCH_W // 2
        sy = y + _CARD_H // 2
        state = _nearest_state(arr[sy, sx], key_colors)
        if states[identity] != -1:
            raise ValueError("identity glyph appears more than once in a panel")
        states[identity] = state
    if any(value < 0 for value in states):
        raise ValueError("panel does not contain all four identities")
    return states


def _is_derangement(initial: list[int], later: list[int]) -> bool:
    if any(a == b for a, b in zip(initial, later)):
        return False
    return all(initial.count(state) == later.count(state) for state in range(3))


def decision_from_image(image: Image.Image) -> str:
    """Recover the first derangement panel from the PNG alone."""

    rgb = image.convert("RGB")
    if rgb.size != (_WIDTH, _HEIGHT):
        raise ValueError(f"unexpected raster size {rgb.size}")
    arr = np.asarray(rgb)
    key_colors = _sample_key_colors(arr)
    panels = [_decode_panel(arr, _panel_x(index), key_colors) for index in range(6)]
    hits = [index for index in range(1, 6) if _is_derangement(panels[0], panels[index])]
    if len(hits) != 1:
        raise ValueError("image has no unique derangement checkpoint")
    return f"P{hits[0] + 1}"
