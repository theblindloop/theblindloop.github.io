"""Deterministic landmark-inferred rigid-transformation relay."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1000, 650
SIZE = 5
START_X, START_Y, START_CELL, START_GAP = 28, 137, 30, 3
CARD_XS, CARD_Y = (210, 474, 738), 66
CARD_BEFORE_DX, CARD_AFTER_DX = 12, 132
CARD_GRID_Y, CARD_CELL, CARD_GAP = 145, 20, 3
OPTION_XS = (24, 220, 416, 612, 808)
OPTION_Y, OPTION_CELL, OPTION_GAP = 468, 25, 3
BACKGROUND = "#ffffff"
INK = "#20364a"
PAPER = "#f7f3e8"
BORDER = "#6d7e8b"
PALE = "#dce4e8"
RED = "#d32f2f"
TEAL = "#00897b"
AMBER = "#d98212"
VIOLET = "#7b1fa2"
LANDMARKS = ("amber", "teal", "violet")
LANDMARK_COLORS = {"amber": AMBER, "teal": TEAL, "violet": VIOLET}
FONT_PATH = "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf"
TRANSFORMS = (
    "r90", "r180", "r270", "mirror_v", "mirror_h", "mirror_diag", "mirror_anti"
)
ROTATIONS = {"r90", "r180", "r270"}
REFLECTIONS = set(TRANSFORMS) - ROTATIONS


def _font(size: int) -> ImageFont.FreeTypeFont:
    return ImageFont.truetype(FONT_PATH, size=size)


def _transform(point: tuple[int, int], name: str) -> tuple[int, int]:
    row, col = point
    last = SIZE - 1
    if name == "r90":
        return col, last - row
    if name == "r180":
        return last - row, last - col
    if name == "r270":
        return last - col, row
    if name == "mirror_v":
        return row, last - col
    if name == "mirror_h":
        return last - row, col
    if name == "mirror_diag":
        return col, row
    if name == "mirror_anti":
        return last - col, last - row
    raise ValueError(f"unknown transform {name!r}")


def _point(raw: object) -> tuple[int, int]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a cell must be a two-integer list")
    row, col = int(raw[0]), int(raw[1])
    if [row, col] != raw or not (0 <= row < SIZE and 0 <= col < SIZE):
        raise ValueError("cell coordinate is invalid")
    return row, col


def _token(raw: object) -> tuple[tuple[int, int], tuple[int, int]]:
    if not isinstance(raw, list) or len(raw) != 2:
        raise ValueError("a token needs navy tail then red head")
    tail, head = _point(raw[0]), _point(raw[1])
    if abs(tail[0] - head[0]) + abs(tail[1] - head[1]) != 1:
        raise ValueError("token dots must occupy adjacent cells")
    return tail, head


def _landmark_map(raw: object) -> dict[str, tuple[int, int]]:
    if not isinstance(raw, dict) or set(raw) != set(LANDMARKS):
        raise ValueError("each peg grid needs amber, teal, and violet")
    result = {name: _point(raw[name]) for name in LANDMARKS}
    if len(set(result.values())) != len(LANDMARKS):
        raise ValueError("colored pegs must occupy different cells")
    return result


def _infer_transform(card: list) -> str:
    if not isinstance(card, list) or len(card) != 2:
        raise ValueError("a card must contain ordered before and after peg grids")
    before, after = _landmark_map(card[0]), _landmark_map(card[1])
    matches = [
        name for name in TRANSFORMS
        if all(_transform(before[color], name) == after[color] for color in LANDMARKS)
    ]
    if len(matches) != 1:
        raise ValueError("colored correspondences must specify one nonidentity rigid move")
    return matches[0]


def _parse_scene(scene: dict) -> tuple[
    tuple[tuple[int, int], tuple[int, int]], list[dict], list[tuple[tuple[int, int], tuple[int, int]]]
]:
    if set(scene) != {"start_token", "transform_cards", "options"}:
        raise ValueError("scene fields must be exactly start_token, transform_cards, and options")
    start = _token(scene["start_token"])
    cards = scene["transform_cards"]
    if not isinstance(cards, list) or len(cards) != 3:
        raise ValueError("exactly three transformation cards are required")
    for card in cards:
        _infer_transform(card)
    raw_options = scene["options"]
    if not isinstance(raw_options, list) or len(raw_options) != 5:
        raise ValueError("exactly five options are required")
    options = [_token(raw) for raw in raw_options]
    if len(set(options)) != 5:
        raise ValueError("option tokens must be distinct")
    return start, cards, options


def _move_token(
    token: tuple[tuple[int, int], tuple[int, int]], name: str
) -> tuple[tuple[int, int], tuple[int, int]]:
    return _transform(token[0], name), _transform(token[1], name)


def _evolve(
    token: tuple[tuple[int, int], tuple[int, int]], names: Iterable[str]
) -> tuple[tuple[int, int], tuple[int, int]]:
    state = token
    for name in names:
        state = _move_token(state, name)
    return state


def _as_json_token(token: tuple[tuple[int, int], tuple[int, int]]) -> list[list[int]]:
    return [[token[0][0], token[0][1]], [token[1][0], token[1][1]]]


def _random_landmarks(rng: random.Random, name: str) -> dict:
    cells = [(row, col) for row in range(SIZE) for col in range(SIZE)]
    for _ in range(100):
        chosen = rng.sample(cells, len(LANDMARKS))
        before = {color: chosen[index] for index, color in enumerate(LANDMARKS)}
        after = {color: _transform(before[color], name) for color in LANDMARKS}
        card = [
            {color: list(before[color]) for color in LANDMARKS},
            {color: list(after[color]) for color in LANDMARKS},
        ]
        try:
            if _infer_transform(card) == name:
                return card
        except ValueError:
            pass
    raise RuntimeError("could not sample transform-identifying landmark pegs")


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    oriented_tokens = []
    for row in range(SIZE):
        for col in range(SIZE):
            for dr, dc in ((-1, 0), (0, 1), (1, 0), (0, -1)):
                head = row + dr, col + dc
                if 0 <= head[0] < SIZE and 0 <= head[1] < SIZE:
                    oriented_tokens.append(((row, col), head))
    for _ in range(1200):
        names = [rng.choice(tuple(ROTATIONS)), rng.choice(tuple(REFLECTIONS)), rng.choice(TRANSFORMS)]
        rng.shuffle(names)
        start = rng.choice(oriented_tokens)
        states = [start]
        for name in names:
            states.append(_move_token(states[-1], name))
        if len(set(states)) != 4 or states[-1] == start:
            continue
        target = states[-1]
        preferred = [
            _evolve(start, reversed(names)),
            states[-2],
            _evolve((start[1], start[0]), names),
            _evolve(start, (names[0], names[2])),
            _move_token(start, names[-1]),
            _evolve(start, names[1:]),
        ]
        distractors: list[tuple[tuple[int, int], tuple[int, int]]] = []
        for token in preferred:
            if token != target and token not in distractors:
                distractors.append(token)
        fallback = list(oriented_tokens)
        rng.shuffle(fallback)
        for token in fallback:
            if token != target and token not in distractors:
                distractors.append(token)
            if len(distractors) >= 4:
                break
        if len(distractors) < 4:
            continue
        rng.shuffle(distractors)
        options = [_as_json_token(token) for token in distractors[:4]]
        options.insert(int(seed) % 5, _as_json_token(target))
        scene = {
            "start_token": _as_json_token(start),
            "transform_cards": [_random_landmarks(rng, name) for name in names],
            "options": options,
        }
        _parse_scene(scene)
        return scene
    raise RuntimeError("could not sample a nondegenerate landmark relay")


def analytic_gold(scene: dict) -> str:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, (_infer_transform(card) for card in cards))
    matches = [index for index, option in enumerate(options) if option == target]
    if len(matches) != 1:
        raise ValueError("relayed token must match exactly one option")
    return "ABCDE"[matches[0]]


def margin(scene: dict) -> float:
    start, cards, options = _parse_scene(scene)
    target = _evolve(start, (_infer_transform(card) for card in cards))
    distances = [
        float(sum(a != b for a, b in zip(target, option)))
        for option in options if option != target
    ]
    return min(distances) if distances else 0.0


def is_quarantined(scene: dict) -> bool:
    try:
        start, cards, options = _parse_scene(scene)
        target = _evolve(start, (_infer_transform(card) for card in cards))
        return sum(option == target for option in options) != 1 or margin(scene) < 1.0
    except (TypeError, ValueError, IndexError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _parse_scene(scene)
    return []


def _grid_frame(draw: ImageDraw.ImageDraw, x: int, y: int, cell: int, gap: int) -> None:
    for row in range(SIZE):
        for col in range(SIZE):
            left, top = x + col * (cell + gap), y + row * (cell + gap)
            draw.rectangle((left, top, left + cell, top + cell), fill=PAPER, outline=BORDER, width=2)


def _marker(draw: ImageDraw.ImageDraw, point: tuple[int, int], color: str, x: int, y: int, cell: int, gap: int) -> None:
    row, col = point
    cx = x + col * (cell + gap) + cell // 2
    cy = y + row * (cell + gap) + cell // 2
    radius = max(6, cell // 3)
    draw.ellipse((cx - radius, cy - radius, cx + radius, cy + radius), fill=color, outline="white", width=2)


def _draw_token(draw: ImageDraw.ImageDraw, token, x: int, y: int, cell: int, gap: int) -> None:
    _grid_frame(draw, x, y, cell, gap)
    _marker(draw, token[0], INK, x, y, cell, gap)
    _marker(draw, token[1], RED, x, y, cell, gap)


def _draw_landmarks(draw: ImageDraw.ImageDraw, values: dict, x: int, y: int) -> None:
    _grid_frame(draw, x, y, CARD_CELL, CARD_GAP)
    for name in LANDMARKS:
        _marker(draw, values[name], LANDMARK_COLORS[name], x, y, CARD_CELL, CARD_GAP)


def render(scene: dict) -> Image.Image:
    start, cards, options = _parse_scene(scene)
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)
    draw.text((500, 25), "INFER EACH MOVE  •  APPLY 1 → 2 → 3", fill=INK, font=_font(24), anchor="mm")
    draw.rounded_rectangle((14, CARD_Y, 198, 359), radius=12, fill="#f8fafb", outline=PALE, width=3)
    draw.text((106, 94), "START TOKEN", fill=INK, font=_font(20), anchor="mm")
    draw.text((106, 119), "NAVY TAIL  •  RED HEAD", fill=BORDER, font=_font(12), anchor="mm")
    _draw_token(draw, start, START_X, START_Y, START_CELL, START_GAP)
    for number, (x, card) in enumerate(zip(CARD_XS, cards), 1):
        draw.rounded_rectangle((x, CARD_Y, x + 250, 359), radius=12, fill="#f8fafb", outline=PALE, width=3)
        draw.text((x + 125, 89), f"MOVE {number}", fill=INK, font=_font(19), anchor="mm")
        draw.text((x + 66, 120), "BEFORE", fill=BORDER, font=_font(13), anchor="mm")
        draw.text((x + 186, 120), "AFTER", fill=BORDER, font=_font(13), anchor="mm")
        draw.text((x + 125, 201), "→", fill=INK, font=_font(23), anchor="mm")
        _draw_landmarks(draw, card[0], x + CARD_BEFORE_DX, CARD_GRID_Y)
        _draw_landmarks(draw, card[1], x + CARD_AFTER_DX, CARD_GRID_Y)
        draw.text((x + 125, 333), "COLORS KEEP IDENTITY", fill=BORDER, font=_font(11), anchor="mm")
    draw.line((24, 383, 976, 383), fill=PALE, width=2)
    draw.text((500, 407), "FINAL TOKEN", fill=BORDER, font=_font(19), anchor="mm")
    for label, token, x in zip("ABCDE", options, OPTION_XS):
        cx, cy = x + 68, 438
        draw.ellipse((cx - 21, cy - 21, cx + 21, cy + 21), fill=INK)
        draw.text((cx, cy), label, fill="white", font=_font(23), anchor="mm")
        _draw_token(draw, token, x, OPTION_Y, OPTION_CELL, OPTION_GAP)
    return image
