"""Independent PNG-only inverse for the landmark transformation relay."""

from __future__ import annotations

import numpy as np
from PIL import Image

EXPECTED_SIZE = (1000, 650)
SIZE = 5
START_X, START_Y, START_CELL, START_GAP = 28, 137, 30, 3
CARD_XS = (210, 474, 738)
CARD_BEFORE_DX, CARD_AFTER_DX = 12, 132
CARD_GRID_Y, CARD_CELL, CARD_GAP = 145, 20, 3
OPTION_XS = (24, 220, 416, 612, 808)
OPTION_Y, OPTION_CELL, OPTION_GAP = 468, 25, 3
PALETTE = {
    "tail": np.array((32, 54, 74), dtype=np.float32),
    "head": np.array((211, 47, 47), dtype=np.float32),
    "amber": np.array((217, 130, 18), dtype=np.float32),
    "teal": np.array((0, 137, 123), dtype=np.float32),
    "violet": np.array((123, 31, 162), dtype=np.float32),
}
LANDMARKS = ("amber", "teal", "violet")
TRANSFORMS = ("r90", "r180", "r270", "mirror_v", "mirror_h", "mirror_diag", "mirror_anti")


class OracleError(ValueError):
    pass


def _map(point: tuple[int, int], name: str) -> tuple[int, int]:
    row, col = point
    edge = 4
    rules = {
        "r90": (col, edge - row),
        "r180": (edge - row, edge - col),
        "r270": (edge - col, row),
        "mirror_v": (row, edge - col),
        "mirror_h": (edge - row, col),
        "mirror_diag": (col, row),
        "mirror_anti": (edge - col, edge - row),
    }
    return rules[name]


def _classify_center(array: np.ndarray, cx: int, cy: int, allowed: tuple[str, ...]) -> str | None:
    # The five-pixel core stays inside even the smallest broad marker and avoids
    # its deliberately redundant white outline.
    patch = array[cy - 2:cy + 3, cx - 2:cx + 3].astype(np.float32)
    mean = patch.mean(axis=(0, 1))
    ranked = sorted((float(np.linalg.norm(mean - PALETTE[name])), name) for name in allowed)
    if ranked[0][0] < 12.0 and (len(ranked) == 1 or ranked[1][0] - ranked[0][0] > 25.0):
        return ranked[0][1]
    if float(mean.mean()) > 190.0:
        return None
    raise OracleError("a grid cell center is neither clean paper nor an allowed marker")


def _scan(array: np.ndarray, x: int, y: int, cell: int, gap: int, allowed: tuple[str, ...]) -> dict[str, tuple[int, int]]:
    found: dict[str, tuple[int, int]] = {}
    for row in range(SIZE):
        for col in range(SIZE):
            cx = x + col * (cell + gap) + cell // 2
            cy = y + row * (cell + gap) + cell // 2
            name = _classify_center(array, cx, cy, allowed)
            if name is not None:
                if name in found:
                    raise OracleError(f"duplicate {name} marker")
                found[name] = (row, col)
    if set(found) != set(allowed):
        raise OracleError("a marker grid is incomplete")
    return found


def _token(array: np.ndarray, x: int, y: int, cell: int, gap: int):
    found = _scan(array, x, y, cell, gap, ("tail", "head"))
    tail, head = found["tail"], found["head"]
    if abs(tail[0] - head[0]) + abs(tail[1] - head[1]) != 1:
        raise OracleError("the two token dots are not adjacent")
    return tail, head


def _infer(before: dict, after: dict) -> str:
    matches = [
        name for name in TRANSFORMS
        if all(_map(before[color], name) == after[color] for color in LANDMARKS)
    ]
    if len(matches) != 1:
        raise OracleError("peg correspondences do not determine one move")
    return matches[0]


def _validate_labels(array: np.ndarray) -> None:
    ink = PALETTE["tail"]
    for x in OPTION_XS:
        cx, cy = x + 68, 438
        patch = array[cy - 19:cy + 20, cx - 19:cx + 20].astype(np.float32)
        if int((np.linalg.norm(patch - ink, axis=2) < 12.0).sum()) < 880:
            raise OracleError("an option label disk is missing")


def decision_from_image(image: Image.Image) -> str:
    if image.size != EXPECTED_SIZE:
        raise OracleError(f"unexpected raster size {image.size}")
    array = np.asarray(image.convert("RGB"))
    _validate_labels(array)
    state = _token(array, START_X, START_Y, START_CELL, START_GAP)
    moves = []
    for x in CARD_XS:
        before = _scan(array, x + CARD_BEFORE_DX, CARD_GRID_Y, CARD_CELL, CARD_GAP, LANDMARKS)
        after = _scan(array, x + CARD_AFTER_DX, CARD_GRID_Y, CARD_CELL, CARD_GAP, LANDMARKS)
        moves.append(_infer(before, after))
    for name in moves:
        state = _map(state[0], name), _map(state[1], name)
    options = [_token(array, x, OPTION_Y, OPTION_CELL, OPTION_GAP) for x in OPTION_XS]
    if len(set(options)) != 5:
        raise OracleError("final options are not distinct")
    matches = [index for index, option in enumerate(options) if option == state]
    if len(matches) != 1:
        raise OracleError("relayed token does not match exactly one option")
    return "ABCDE"[matches[0]]
