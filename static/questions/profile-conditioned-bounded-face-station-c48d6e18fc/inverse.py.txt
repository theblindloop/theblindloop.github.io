"""Scene-free inverse arm using only pixels from the final PNG."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = {
    "orange": np.array((230, 112, 42), dtype=np.int16),
    "green": np.array((32, 151, 86), dtype=np.int16),
    "purple": np.array((132, 82, 190), dtype=np.int16),
}


def _components(white: np.ndarray) -> tuple[np.ndarray, set[int]]:
    height, width = white.shape
    labels = np.zeros((height, width), dtype=np.int32)
    next_label = 0
    for y in range(height):
        for x in range(width):
            if not white[y, x] or labels[y, x]:
                continue
            next_label += 1
            labels[y, x] = next_label
            queue = deque([(x, y)])
            while queue:
                px, py = queue.popleft()
                if px and white[py, px - 1] and not labels[py, px - 1]:
                    labels[py, px - 1] = next_label
                    queue.append((px - 1, py))
                if px + 1 < width and white[py, px + 1] and not labels[py, px + 1]:
                    labels[py, px + 1] = next_label
                    queue.append((px + 1, py))
                if py and white[py - 1, px] and not labels[py - 1, px]:
                    labels[py - 1, px] = next_label
                    queue.append((px, py - 1))
                if py + 1 < height and white[py + 1, px] and not labels[py + 1, px]:
                    labels[py + 1, px] = next_label
                    queue.append((px, py + 1))
    border = set(labels[0]) | set(labels[-1]) | set(labels[:, 0]) | set(labels[:, -1])
    border.discard(0)
    return labels, border


def _marker_mask(array: np.ndarray, target: np.ndarray) -> np.ndarray:
    # int32 avoids overflow when squaring channel differences near 255.
    delta = array.astype(np.int32) - target.astype(np.int32)
    return np.sum(delta * delta, axis=2) < 38 * 38


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if array.ndim != 3 or min(array.shape[:2]) < 100:
        raise ValueError("unsupported raster")
    white = np.all(array >= 248, axis=2)
    labels, border_labels = _components(white)
    yy, xx = np.indices(white.shape)
    incidence: dict[str, int] = {}
    for name, rgb in PALETTE.items():
        mask = _marker_mask(array, rgb)
        ys, xs = np.nonzero(mask)
        if len(xs) < 500:
            raise ValueError(f"missing {name} station")
        cx, cy = float(xs.mean()), float(ys.mean())
        radius = float(np.sqrt(len(xs) / np.pi))
        distance2 = (xx - cx) ** 2 + (yy - cy) ** 2
        annulus = white & (distance2 >= (radius - 1.0) ** 2) & (distance2 <= (radius + 8.0) ** 2)
        adjacent = {int(value) for value in np.unique(labels[annulus]) if int(value) > 0}
        incidence[name] = len(adjacent - border_labels)
    winners = [name for name, count in incidence.items() if count == 3]
    if len(winners) != 1:
        raise ValueError(f"ambiguous face incidence {incidence}")
    return winners[0]
