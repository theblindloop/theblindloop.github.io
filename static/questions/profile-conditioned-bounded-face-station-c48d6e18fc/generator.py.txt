"""Deterministic planar-wall renderer and combinatorial analytic arm."""

from __future__ import annotations

import copy
import random

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 620, 500
WALL_WIDTH = 14
MARKER_RADIUS = 18
SCALE = 3
COLORS = {
    "orange": (230, 112, 42),
    "green": (32, 151, 86),
    "purple": (132, 82, 190),
}


def _transform_point(point: tuple[int, int], transform: int, dx: int, dy: int) -> tuple[int, int]:
    x, y = point
    x -= WIDTH // 2
    y -= HEIGHT // 2
    if transform >= 4:
        x = -x
    for _ in range(transform % 4):
        x, y = -y, x
    return (x + WIDTH // 2 + dx, y + HEIGHT // 2 + dy)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed)
    xs = (145, 255, 365, 475)
    ys = (85, 195, 305, 415)
    missing = {
        tuple(sorted(((xs[1], ys[0]), (xs[1], ys[1])))),
        tuple(sorted(((xs[1], ys[1]), (xs[1], ys[2])))),
    }
    base_walls: list[tuple[tuple[int, int], tuple[int, int]]] = []
    for y in ys:
        for i in range(3):
            edge = tuple(sorted(((xs[i], y), (xs[i + 1], y))))
            if edge not in missing:
                base_walls.append(edge)
    for x in xs:
        for j in range(3):
            edge = tuple(sorted(((x, ys[j]), (x, ys[j + 1]))))
            if edge not in missing:
                base_walls.append(edge)

    transform = (seed // 3) % 8
    dx = ((seed * 17) % 17) - 8
    dy = ((seed * 29) % 15) - 7
    walls = []
    for a, b in base_walls:
        ta = _transform_point(a, transform, dx, dy)
        tb = _transform_point(b, transform, dx, dy)
        walls.append([list(ta), list(tb)])
    rng.shuffle(walls)

    # These base vertices have incidence counts 2, 4, and 3 respectively.
    base_markers = [(xs[1], ys[1]), (xs[2], ys[1]), (xs[1], ys[2])]
    shift = seed % 3
    names = list(COLORS)
    names = names[shift:] + names[:shift]
    if (seed // 3) % 2:
        names[0], names[1] = names[1], names[0]
    markers = [
        {
            "color": color,
            "center": list(_transform_point(point, transform, dx, dy)),
            "radius": MARKER_RADIUS,
        }
        for point, color in zip(base_markers, names)
    ]
    rng.shuffle(markers)
    return {"canvas": [WIDTH, HEIGHT], "walls": walls, "markers": markers}


def _edge_key(a, b):
    return tuple(sorted((tuple(a), tuple(b))))


def _incidence_counts(scene: dict) -> dict[str, int]:
    width, height = scene["canvas"]
    if width < 100 or height < 100:
        raise ValueError("canvas too small")
    walls = scene["walls"]
    markers = scene["markers"]
    xs = sorted({int(p[0]) for edge in walls for p in edge})
    ys = sorted({int(p[1]) for edge in walls for p in edge})
    if len(xs) != 4 or len(ys) != 4:
        raise ValueError("expected a 4 by 4 wall lattice")
    wall_set = {_edge_key(edge[0], edge[1]) for edge in walls}
    ncols, nrows = len(xs) - 1, len(ys) - 1
    parent = list(range(ncols * nrows))

    def find(value: int) -> int:
        while parent[value] != value:
            parent[value] = parent[parent[value]]
            value = parent[value]
        return value

    def union(left: int, right: int) -> None:
        left, right = find(left), find(right)
        if left != right:
            parent[right] = left

    for row in range(nrows):
        for boundary in range(1, ncols):
            edge = _edge_key((xs[boundary], ys[row]), (xs[boundary], ys[row + 1]))
            if edge not in wall_set:
                union(row * ncols + boundary - 1, row * ncols + boundary)
    for boundary in range(1, nrows):
        for col in range(ncols):
            edge = _edge_key((xs[col], ys[boundary]), (xs[col + 1], ys[boundary]))
            if edge not in wall_set:
                union((boundary - 1) * ncols + col, boundary * ncols + col)

    result: dict[str, int] = {}
    for marker in markers:
        cx, cy = marker["center"]
        ix, iy = xs.index(cx), ys.index(cy)
        if not (0 < ix < len(xs) - 1 and 0 < iy < len(ys) - 1):
            raise ValueError("marker must be at an interior lattice vertex")
        cells = (
            (iy - 1) * ncols + ix - 1,
            (iy - 1) * ncols + ix,
            iy * ncols + ix - 1,
            iy * ncols + ix,
        )
        result[marker["color"]] = len({find(cell) for cell in cells})
    return result


def render(scene: dict) -> Image.Image:
    width, height = scene["canvas"]
    walls = scene["walls"]
    markers = scene["markers"]
    image = Image.new("RGB", (width * SCALE, height * SCALE), "white")
    draw = ImageDraw.Draw(image)
    for a, b in walls:
        draw.line(
            (a[0] * SCALE, a[1] * SCALE, b[0] * SCALE, b[1] * SCALE),
            fill=(18, 23, 28),
            width=WALL_WIDTH * SCALE,
        )
    for marker in markers:
        cx, cy = marker["center"]
        radius = marker["radius"]
        draw.ellipse(
            (
                (cx - radius) * SCALE,
                (cy - radius) * SCALE,
                (cx + radius) * SCALE,
                (cy + radius) * SCALE,
            ),
            fill=COLORS[marker["color"]],
        )
    return image.resize((width, height), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    counts = _incidence_counts(scene)
    winners = [color for color, count in counts.items() if count == 3]
    if len(winners) != 1:
        raise ValueError(f"non-unique analytic decision: {counts}")
    return winners[0]


def margin(scene: dict) -> float:
    counts = _incidence_counts(scene)
    decision = analytic_gold(scene)
    return float(min(abs(count - 3) for color, count in counts.items() if color != decision))


def is_quarantined(scene: dict) -> bool:
    try:
        counts = _incidence_counts(scene)
        radii = [int(marker["radius"]) for marker in scene["markers"]]
        return sorted(counts.values()) != [2, 3, 4] or min(radii) < 16
    except (KeyError, TypeError, ValueError):
        return True


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["walls"] = list(reversed(twin["walls"]))
    return [("reverse_unordered_wall_serialization", twin)]
