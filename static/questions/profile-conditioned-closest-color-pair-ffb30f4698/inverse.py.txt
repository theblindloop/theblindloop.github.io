"""Independent pixel-only inverse arm for the closest-pair measurement world.

Receives ONLY the rasterized PNG. It segments red and blue disks, recovers each
dot centre as a connected-component centroid, forms the unique within-colour
pair, measures its centre distance, and decides which colour's pair is closer
together. Imports only numpy and PIL; shares no geometry, coordinates, or code
with the renderer and never reads latent scene data or analytic gold.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

ORACLE_VERSION = "closest_color_pair-oracle-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)

DOT_PIXEL_MIN = 16


def _labels(arr: np.ndarray) -> np.ndarray:
    refs = np.stack(
        [
            np.array(COLOR_RGB["red"], dtype=np.float32),
            np.array(COLOR_RGB["blue"], dtype=np.float32),
            np.array(BACKGROUND, dtype=np.float32),
        ],
        axis=0,
    )
    diffs = arr[:, :, np.newaxis, :].astype(np.float32) - refs[np.newaxis, np.newaxis, :, :]
    return np.linalg.norm(diffs, axis=-1).argmin(axis=-1)


def _centroids(mask: np.ndarray):
    ys, xs = np.nonzero(mask)
    if len(ys) == 0:
        return []
    points = list(set(zip(xs.tolist(), ys.tolist())))
    seen = set()
    centroids = []
    for start in points:
        if start in seen:
            continue
        comp = [start]
        seen.add(start)
        idx = 0
        sx = sy = 0
        while idx < len(comp):
            x, y = comp[idx]
            idx += 1
            sx += x
            sy += y
            for dx, dy in ((-1, 0), (1, 0), (0, -1), (0, 1)):
                nb = (x + dx, y + dy)
                if nb in seen or nb not in points:
                    continue
                seen.add(nb)
                comp.append(nb)
        if len(comp) >= DOT_PIXEL_MIN:
            centroids.append((sx / len(comp), sy / len(comp)))
    return centroids


def _pair_distance(centroids):
    if len(centroids) < 2:
        return None
    (x1, y1), (x2, y2) = centroids[:2]
    return float(np.hypot(x2 - x1, y2 - y1))


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.float32)
    labels = _labels(arr)
    red = _centroids(labels == 0)
    blue = _centroids(labels == 1)
    d_red = _pair_distance(red)
    d_blue = _pair_distance(blue)
    if d_red is None or d_blue is None:
        return "blue"
    return "red" if d_red < d_blue else "blue"
