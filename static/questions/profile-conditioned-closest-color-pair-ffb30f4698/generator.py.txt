"""Intra-color closest-pair measurement world.

A light panel carries exactly two red dots and exactly two blue dots. The
decision is a recovered measurement: recover each dot's position from its
colored disk, form the unique pair per color, compute that pair's centre-to-
centre distance, and decide which color's two dots are CLOSER together.

The analysed quantity is an intra-class distance extremum recovered from the
raster: neither the total colored ink, the dot count (both colours have exactly
two dots), the positions alone, nor a color-area proxy determines which colour's
pair is tighter. Ties between the two pair distances are resolved by a clear
margin and any near-tie scene is quarantined. The oracle and analytic gold share
no geometry and never read latent scene data.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

RENDERER_VERSION = "closest_color_pair-0.1.0"

COLOR_RGB = {
    "red": (205, 40, 40),
    "blue": (30, 70, 205),
}
BACKGROUND = (250, 250, 250)

WIDTH = 480
HEIGHT = 340
DOT_R = 8
RED_COUNT = 2
BLUE_COUNT = 2
EDGE = 60
MIN_PAIR = 44          # min distance between a colour's two dots
MAX_PAIR = 220         # max distance between a colour's two dots
MIN_INTRA_CLR = 22     # min distance between any red and any blue dot
QUARANTINE = 6.0
MIN_MARGIN = 9.0


def _pair_distance(dots):
    (x1, y1), (x2, y2) = dots
    return math.hypot(x2 - x1, y2 - y1)


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 977 + 17)
    while True:
        def pair():
            x1 = rng.uniform(EDGE, WIDTH - EDGE)
            y1 = rng.uniform(EDGE, HEIGHT - EDGE)
            while True:
                ang = rng.uniform(0.0, 2.0 * math.pi)
                dist = rng.uniform(MIN_PAIR, MAX_PAIR)
                x2 = x1 + dist * math.cos(ang)
                y2 = y1 + dist * math.sin(ang)
                if EDGE <= x2 <= WIDTH - EDGE and EDGE <= y2 <= HEIGHT - EDGE:
                    return [
                        [round(x1, 3), round(y1, 3)],
                        [round(x2, 3), round(y2, 3)],
                    ]

        red = pair()
        blue = pair()
        all_dots = red + blue
        ok = True
        for i in range(len(all_dots)):
            for j in range(i + 1, len(all_dots)):
                if (i < 2) == (j < 2):
                    continue
                if _pair_distance_frame(all_dots[i], all_dots[j]) < MIN_INTRA_CLR:
                    ok = False
        if not ok:
            continue
        d_red = _pair_distance(red)
        d_blue = _pair_distance(blue)
        if abs(d_red - d_blue) < MIN_MARGIN:
            continue
        scene = {
            "red_dots": red,
            "blue_dots": blue,
            "width": WIDTH,
            "height": HEIGHT,
            "background": "light",
        }
        return scene


def _pair_distance_frame(a, b):
    return math.hypot(b[0] - a[0], b[1] - a[1])


def _distances(scene: dict):
    d_red = _pair_distance(scene["red_dots"])
    d_blue = _pair_distance(scene["blue_dots"])
    return d_red, d_blue


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (scene["width"], scene["height"]), BACKGROUND)
    draw = ImageDraw.Draw(img)
    for x, y in scene["red_dots"]:
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=COLOR_RGB["red"])
    for x, y in scene["blue_dots"]:
        draw.ellipse([x - DOT_R, y - DOT_R, x + DOT_R, y + DOT_R], fill=COLOR_RGB["blue"])
    return img


def analytic_gold(scene: dict) -> str:
    d_red, d_blue = _distances(scene)
    return "red" if d_red < d_blue else "blue"


def margin(scene: dict) -> float:
    d_red, d_blue = _distances(scene)
    return float(abs(d_red - d_blue))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE


def latent_symmetries(scene: dict) -> list:
    return []
