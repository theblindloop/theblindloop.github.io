"""Independent image-only inverse for the paired route arrival relation."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

NAVY = np.array((28, 58, 88), dtype=np.int32)
GREEN = np.array((20, 158, 117), dtype=np.int32)
ORANGE = np.array((230, 126, 34), dtype=np.int32)


def _mask(array: np.ndarray, color: np.ndarray, tolerance: int = 18) -> np.ndarray:
    delta = array.astype(np.int32) - color
    return np.sqrt(np.sum(delta * delta, axis=2)) <= tolerance


def _components(mask: np.ndarray, minimum: int) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    output = []
    for y, x in np.argwhere(mask):
        y, x = int(y), int(x)
        if seen[y, x]:
            continue
        queue = deque([(y, x)])
        seen[y, x] = True
        points = []
        while queue:
            cy, cx = queue.popleft()
            points.append((cy, cx))
            for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True
                    queue.append((ny, nx))
        if len(points) >= minimum:
            output.append(np.asarray(points, dtype=np.float64))
    return output


def _centroid(component: np.ndarray) -> np.ndarray:
    # Return x,y for geometric work.
    return component[:, ::-1].mean(axis=0)


def _nearest_distance(points_yx: np.ndarray, center_xy: np.ndarray) -> float:
    points_xy = points_yx[:, ::-1]
    return float(np.hypot(points_xy[:, 0] - center_xy[0], points_xy[:, 1] - center_xy[1]).min())


def _finish_axis(route_yx: np.ndarray, finish_xy: np.ndarray) -> str:
    points = route_yx[:, ::-1]
    radii = np.hypot(points[:, 0] - finish_xy[0], points[:, 1] - finish_xy[1])
    local = points[(radii >= 5.0) & (radii <= 12.0)]
    if len(local) < 10:
        raise ValueError("finish-adjacent route stub is missing")
    centered = local - local.mean(axis=0)
    covariance = centered.T @ centered
    values, vectors = np.linalg.eigh(covariance)
    if values[-1] < 1.5 * max(values[0], 1.0):
        raise ValueError("arrival direction is not stably axis-like")
    direction = vectors[:, -1]
    dominance = abs(abs(direction[0]) - abs(direction[1]))
    if dominance < 0.30:
        raise ValueError("arrival axis is ambiguous")
    return "horizontal" if abs(direction[0]) > abs(direction[1]) else "vertical"


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    routes = _components(_mask(array, NAVY), minimum=28)
    starts = _components(_mask(array, GREEN), minimum=35)
    finishes = _components(_mask(array, ORANGE), minimum=35)
    if len(routes) != 2 or len(starts) != 2 or len(finishes) != 2:
        raise ValueError("expected exactly two disconnected routes, starts, and finishes")

    start_centers = [_centroid(component) for component in starts]
    finish_centers = [_centroid(component) for component in finishes]
    remaining = set(range(2))
    axes = []
    # Reconstruct route ownership from marker adjacency. Each route must bind
    # exactly one start and one finish before its ordered arrival tangent is fit.
    for finish in finish_centers:
        candidate = min(remaining, key=lambda idx: _nearest_distance(routes[idx], finish))
        if _nearest_distance(routes[candidate], finish) > 7.5:
            raise ValueError("finish is detached from its route")
        start_distances = [_nearest_distance(routes[candidate], start) for start in start_centers]
        if min(start_distances) > 7.5:
            raise ValueError("route has no attached start")
        axes.append(_finish_axis(routes[candidate], finish))
        remaining.remove(candidate)
    return "parallel" if axes[0] == axes[1] else "perpendicular"
