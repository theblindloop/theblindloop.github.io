"""Deterministic renderer and analytic arm for paired arrival-axis relations."""

from __future__ import annotations

import math
import random
from PIL import Image, ImageDraw

BACKGROUND = (250, 248, 242)
NAVY = (28, 58, 88)
GREEN = (20, 158, 117)
ORANGE = (230, 126, 34)
GRAY = (210, 207, 198)


def _rotate(point: tuple[int, int], quarter_turns: int) -> tuple[int, int]:
    x, y = point
    for _ in range(quarter_turns % 4):
        x, y = -y, x
    return x, y


def _route(center: tuple[int, int], quarter_turns: int) -> list[list[int]]:
    # The base trace ends eastward. Its earlier bends prevent treating it as a
    # free-standing orientation bar.
    base = [(-9, -9), (-9, -3), (-3, -3), (-3, 4), (10, 4)]
    cx, cy = center
    return [[cx + x, cy + y] for x, y in (_rotate(p, quarter_turns) for p in base)]


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    # The complete miniature routes occupy two nonadjacent cells of the public
    # 8x8 ablation lattice. The pair moves among rows/columns and can be laid
    # out horizontally or vertically, while one blank cell separates them.
    if rng.random() < 0.5:
        row = rng.choice([2, 3, 4, 5])
        col = rng.choice([2, 3])
        centers = [(64 * col + 32, 64 * row + 32), (64 * (col + 2) + 32, 64 * row + 32)]
    else:
        col = rng.choice([2, 3, 4, 5])
        row = rng.choice([2, 3])
        centers = [(64 * col + 32, 64 * row + 32), (64 * col + 32, 64 * (row + 2) + 32)]
    if rng.random() < 0.5:
        centers.reverse()

    first_rotation = rng.randrange(4)
    # Seed parity guarantees balance in every consecutive generated batch.
    parallel = bool(int(seed) % 2)
    if parallel:
        second_rotation = (first_rotation + rng.choice([0, 2])) % 4
    else:
        second_rotation = (first_rotation + rng.choice([1, 3])) % 4
    routes = [_route(centers[0], first_rotation), _route(centers[1], second_rotation)]

    decorations = []
    # Pale, remote, answer-independent marks prevent absolute inset location
    # or total foreground extent from being useful.
    for box in ((38, 48, 112, 118), (400, 394, 474, 464)):
        x0, y0, x1, y1 = box
        for _ in range(3):
            decorations.append(
                [rng.randint(x0, x1), rng.randint(y0, y1), rng.choice([3, 4])]
            )
    return {
        "width": width,
        "height": height,
        "routes": routes,
        "decorations": decorations,
    }


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (int(scene["width"]), int(scene["height"])), BACKGROUND)
    draw = ImageDraw.Draw(image)
    for x, y, radius in scene["decorations"]:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=GRAY)
    for points in scene["routes"]:
        xy = [tuple(point) for point in points]
        draw.line(xy, fill=NAVY, width=4, joint="curve")
        sx, sy = xy[0]
        fx, fy = xy[-1]
        draw.ellipse((sx - 4, sy - 4, sx + 4, sy + 4), fill=GREEN)
        draw.ellipse((fx - 4, fy - 4, fx + 4, fy + 4), fill=ORANGE)
    return image


def _arrival_axis(points: list[list[int]]) -> str:
    x0, y0 = points[-2]
    x1, y1 = points[-1]
    return "horizontal" if abs(x1 - x0) > abs(y1 - y0) else "vertical"


def analytic_gold(scene: dict) -> str:
    routes = scene["routes"]
    return "parallel" if _arrival_axis(routes[0]) == _arrival_axis(routes[1]) else "perpendicular"


def margin(scene: dict) -> float:
    routes = scene["routes"]
    lengths = []
    clearances = []
    for points in routes:
        x0, y0 = points[-2]
        x1, y1 = points[-1]
        lengths.append(math.hypot(x1 - x0, y1 - y0) - 8.0)
    for p in routes[0]:
        for q in routes[1]:
            clearances.append(math.hypot(p[0] - q[0], p[1] - q[1]) - 14.0)
    return float(min(lengths + clearances))


def is_quarantined(scene: dict) -> bool:
    routes = scene["routes"]
    if len(routes) != 2 or any(len(points) != 5 for points in routes):
        return True
    for points in routes:
        x0, y0 = points[-2]
        x1, y1 = points[-1]
        if not ((x0 == x1) ^ (y0 == y1)) or math.hypot(x1 - x0, y1 - y0) < 12:
            return True
    return margin(scene) < 3.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
