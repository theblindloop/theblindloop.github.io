"""Independent PNG-only inverse for paired enclosure topology."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


ORACLE_VERSION = "paired_void_agreement-oracle-0.1.0"
MIN_COMPONENT_AREA = 2200
MIN_WIDTH = 70
MIN_HEIGHT = 100
MAX_DIMENSION_DELTA = 10
MAX_AREA_RATIO = 1.30
OPEN_MAX_HOLE_PIXELS = 450
CLOSED_MIN_HOLE_PIXELS = 1500


class OracleAbstention(ValueError):
    """The raster does not expose two intact, comparable seals."""


def _foreground(image: Image.Image) -> np.ndarray:
    array = np.asarray(image.convert("RGB"), dtype=np.int16)
    # The public renderer uses a white background and dark colored ink.  The
    # threshold admits antialiased contour pixels without admitting the white
    # interior or exterior.
    return np.max(np.abs(array - 255), axis=2) > 30


def _components(mask: np.ndarray) -> list[np.ndarray]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[np.ndarray] = []
    for y, x in np.argwhere(mask):
        iy, ix = int(y), int(x)
        if seen[iy, ix]:
            continue
        stack = [(iy, ix)]
        seen[iy, ix] = True
        points: list[tuple[int, int]] = []
        while stack:
            cy, cx = stack.pop()
            points.append((cy, cx))
            for ny in range(cy - 1, cy + 2):
                for nx in range(cx - 1, cx + 2):
                    if (
                        0 <= ny < height
                        and 0 <= nx < width
                        and mask[ny, nx]
                        and not seen[ny, nx]
                    ):
                        seen[ny, nx] = True
                        stack.append((ny, nx))
        components.append(np.asarray(points, dtype=np.int32))
    return sorted(components, key=len, reverse=True)


def _bbox(component: np.ndarray) -> tuple[int, int, int, int]:
    y0 = int(component[:, 0].min())
    y1 = int(component[:, 0].max())
    x0 = int(component[:, 1].min())
    x1 = int(component[:, 1].max())
    return x0, y0, x1, y1


def _enclosed_background_pixels(mask: np.ndarray, box: tuple[int, int, int, int]) -> int:
    """Count background pixels in a component crop not connected to its edge."""

    x0, y0, x1, y1 = box
    pad = 3
    height, width = mask.shape
    xa, ya = max(0, x0 - pad), max(0, y0 - pad)
    xb, yb = min(width - 1, x1 + pad), min(height - 1, y1 + pad)
    foreground = mask[ya : yb + 1, xa : xb + 1]
    background = ~foreground
    crop_h, crop_w = background.shape
    outside = np.zeros_like(background, dtype=bool)
    queue: deque[tuple[int, int]] = deque()

    for x in range(crop_w):
        for y in (0, crop_h - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    for y in range(crop_h):
        for x in (0, crop_w - 1):
            if background[y, x] and not outside[y, x]:
                outside[y, x] = True
                queue.append((y, x))
    while queue:
        y, x = queue.popleft()
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < crop_h and 0 <= nx < crop_w:
                if background[ny, nx] and not outside[ny, nx]:
                    outside[ny, nx] = True
                    queue.append((ny, nx))
    return int(np.count_nonzero(background & ~outside))


def _seal_status(mask: np.ndarray, component: np.ndarray) -> bool:
    x0, y0, x1, y1 = _bbox(component)
    if x1 - x0 + 1 < MIN_WIDTH or y1 - y0 + 1 < MIN_HEIGHT:
        raise OracleAbstention("seal component is too small")
    hole_pixels = _enclosed_background_pixels(mask, (x0, y0, x1, y1))
    if hole_pixels <= OPEN_MAX_HOLE_PIXELS:
        return False
    if hole_pixels >= CLOSED_MIN_HOLE_PIXELS:
        return True
    raise OracleAbstention("enclosure status is too close to the raster boundary")


def decision_from_image(image: Image.Image) -> str:
    """Return yes/no from the final pixels alone, or abstain on damaged evidence."""

    mask = _foreground(image)
    components = [item for item in _components(mask) if len(item) >= MIN_COMPONENT_AREA]
    if len(components) != 2:
        raise OracleAbstention("expected exactly two substantial seal components")
    components = sorted(components, key=lambda item: float(item[:, 1].mean()))
    boxes = [_bbox(component) for component in components]
    widths = [box[2] - box[0] + 1 for box in boxes]
    heights = [box[3] - box[1] + 1 for box in boxes]
    areas = [len(component) for component in components]
    if min(widths) < MIN_WIDTH or min(heights) < MIN_HEIGHT:
        raise OracleAbstention("seal dimensions are too small")
    if abs(widths[0] - widths[1]) > MAX_DIMENSION_DELTA:
        raise OracleAbstention("seal widths are not paired")
    if abs(heights[0] - heights[1]) > MAX_DIMENSION_DELTA:
        raise OracleAbstention("seal heights are not paired")
    if max(areas) / float(min(areas)) > MAX_AREA_RATIO:
        raise OracleAbstention("seal foreground areas are not paired")

    left_closed = _seal_status(mask, components[0])
    right_closed = _seal_status(mask, components[1])
    return "yes" if left_closed == right_closed else "no"
