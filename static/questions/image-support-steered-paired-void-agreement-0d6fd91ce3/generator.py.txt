"""Renderer for a paired, disconnected enclosure-topology question world."""

from __future__ import annotations

import math
import random
from typing import Any

from PIL import Image, ImageDraw


RENDERER_VERSION = "paired_void_agreement-0.1.0"
CANVAS = (512, 512)
BACKGROUND = (255, 255, 255)
PALETTE = (
    (31, 105, 113),
    (160, 70, 74),
    (86, 76, 143),
    (52, 117, 77),
    (149, 104, 38),
)
SUPERSAMPLE = 4
HOLE_THRESHOLD = 1200.0
QUARANTINE_MARGIN = 500.0


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced pair of large, separated annular seals.

    The four seed residues deliberately cover both agreeing states and both
    left/right disagreement orders.  Geometry and color are sampled separately
    from that class pattern.
    """

    rng = random.Random(int(seed))
    pattern = int(seed) % 4
    if pattern == 0:
        closed_left, closed_right = True, True
    elif pattern == 1:
        closed_left, closed_right = False, False
    elif pattern == 2:
        closed_left, closed_right = True, False
    else:
        closed_left, closed_right = False, True

    outer_rx = round(rng.uniform(42.0, 45.0), 3)
    outer_ry = round(rng.uniform(61.0, 66.0), 3)
    stroke = round(rng.uniform(12.0, 15.0), 3)
    return {
        "width": CANVAS[0],
        "height": CANVAS[1],
        # These narrow placement bands keep the two decisive islands in one
        # regional support band while leaving a full ablation cell between them.
        "cx_left": round(rng.uniform(126.0, 130.0), 3),
        "cx_right": round(rng.uniform(329.0, 333.0), 3),
        "cy": round(rng.uniform(255.0, 259.0), 3),
        "outer_rx": outer_rx,
        "outer_ry": outer_ry,
        "stroke": stroke,
        "closed_left": closed_left,
        "closed_right": closed_right,
        "slit_angle_left": round(rng.uniform(-math.pi, math.pi), 5),
        "slit_angle_right": round(rng.uniform(-math.pi, math.pi), 5),
        "slit_half_angle": round(rng.uniform(0.17, 0.21), 5),
        "color": list(rng.choice(PALETTE)),
    }


def _point(
    cx: float,
    cy: float,
    rx: float,
    ry: float,
    radius: float,
    angle: float,
    scale: int,
) -> tuple[int, int]:
    return (
        int(round((cx + radius * rx * math.cos(angle)) * scale)),
        int(round((cy + radius * ry * math.sin(angle)) * scale)),
    )


def _draw_seal(
    draw: ImageDraw.ImageDraw,
    *,
    cx: float,
    cy: float,
    outer_rx: float,
    outer_ry: float,
    inner_rx: float,
    inner_ry: float,
    closed: bool,
    slit_angle: float,
    slit_half_angle: float,
    color: tuple[int, int, int],
    scale: int,
) -> None:
    outer_box = (
        int(round((cx - outer_rx) * scale)),
        int(round((cy - outer_ry) * scale)),
        int(round((cx + outer_rx) * scale)),
        int(round((cy + outer_ry) * scale)),
    )
    inner_box = (
        int(round((cx - inner_rx) * scale)),
        int(round((cy - inner_ry) * scale)),
        int(round((cx + inner_rx) * scale)),
        int(round((cy + inner_ry) * scale)),
    )
    draw.ellipse(outer_box, fill=color)
    draw.ellipse(inner_box, fill=BACKGROUND)
    if not closed:
        # A background wedge reaches from the inner chamber past the outer
        # boundary.  The extra radial length makes the opening survive the
        # final downsample and guarantees exterior connectivity.
        a0 = slit_angle - slit_half_angle
        a1 = slit_angle + slit_half_angle
        wedge = [
            _point(cx, cy, outer_rx, outer_ry, 1.18, a0, scale),
            _point(cx, cy, outer_rx, outer_ry, 1.18, a1, scale),
            _point(cx, cy, inner_rx, inner_ry, 0.82, a1, scale),
            _point(cx, cy, inner_rx, inner_ry, 0.82, a0, scale),
        ]
        draw.polygon(wedge, fill=BACKGROUND)


def render(scene: dict[str, Any]) -> Image.Image:
    """Render two disconnected colored seals on a white RGB canvas."""

    scale = SUPERSAMPLE
    image = Image.new(
        "RGB", (int(scene["width"]) * scale, int(scene["height"]) * scale), BACKGROUND
    )
    draw = ImageDraw.Draw(image)
    common = {
        "cy": float(scene["cy"]),
        "outer_rx": float(scene["outer_rx"]),
        "outer_ry": float(scene["outer_ry"]),
        "inner_rx": float(scene["outer_rx"]) - float(scene["stroke"]),
        "inner_ry": float(scene["outer_ry"]) - float(scene["stroke"]),
        "slit_half_angle": float(scene["slit_half_angle"]),
        "color": tuple(int(value) for value in scene["color"]),
        "scale": scale,
    }
    _draw_seal(
        draw,
        cx=float(scene["cx_left"]),
        closed=bool(scene["closed_left"]),
        slit_angle=float(scene["slit_angle_left"]),
        **common,
    )
    _draw_seal(
        draw,
        cx=float(scene["cx_right"]),
        closed=bool(scene["closed_right"]),
        slit_angle=float(scene["slit_angle_right"]),
        **common,
    )
    return image.resize((int(scene["width"]), int(scene["height"])), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    return "yes" if bool(scene["closed_left"]) == bool(scene["closed_right"]) else "no"


def margin(scene: dict[str, Any]) -> float:
    """Continuous distance of each rendered void from the pixel status split."""

    inner_rx = float(scene["outer_rx"]) - float(scene["stroke"])
    inner_ry = float(scene["outer_ry"]) - float(scene["stroke"])
    hole_capacity = math.pi * inner_rx * inner_ry
    closed_margin = abs(hole_capacity - HOLE_THRESHOLD)
    open_margin = HOLE_THRESHOLD
    return round(float(min(closed_margin, open_margin)), 6)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    """No distinct pixel-identical latent alias exists in this parametrization."""

    return []
