"""Pixel-only inverse arm for socket-fit admission binding.

Receives a rendered PNG and nothing else: no scene, no renderer, no gold.
"""

from __future__ import annotations

from itertools import permutations

import numpy as np

HUES = {
    "blue": (0, 90, 200),
    "green": (0, 150, 60),
    "orange": (240, 140, 0),
    "purple": (140, 60, 190),
    "red": (220, 30, 40),
}
GRAY = (150, 150, 150)
COLS = 5


def _mask(arr, rgb):
    return np.all(arr == np.array(rgb, dtype=np.uint8), axis=2)


def _components(mask):
    """Row-run union-find labelling; returns list of bounding boxes."""
    h, w = mask.shape
    parent = []

    def find(a):
        while parent[a] != a:
            parent[a] = parent[parent[a]]
            a = parent[a]
        return a

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[max(ra, rb)] = min(ra, rb)

    prev_runs = []
    runs_all = []
    for y in range(h):
        row = mask[y]
        if not row.any():
            prev_runs = []
            continue
        idx = np.flatnonzero(np.diff(np.concatenate(([0], row.view(np.int8), [0]))))
        cur = []
        for a, b in zip(idx[0::2], idx[1::2]):
            lbl = len(parent)
            parent.append(lbl)
            cur.append((a, b, lbl))
            runs_all.append((y, a, b, lbl))
        for a, b, lbl in cur:
            for pa, pb, plbl in prev_runs:
                if pa < b and a < pb:
                    union(lbl, plbl)
        prev_runs = cur

    boxes = {}
    for y, a, b, lbl in runs_all:
        r = find(lbl)
        if r not in boxes:
            boxes[r] = [y, y, a, b - 1, 0]
        bx = boxes[r]
        bx[0] = min(bx[0], y)
        bx[1] = max(bx[1], y)
        bx[2] = min(bx[2], a)
        bx[3] = max(bx[3], b - 1)
        bx[4] += b - a
    return list(boxes.values())


def decision_from_image(image) -> str:
    arr = np.array(image.convert("RGB"))
    black = np.all(arr < 40, axis=2)

    # --- sockets -----------------------------------------------------------
    sockets = {}
    for name, rgb in HUES.items():
        m = _mask(arr, rgb)
        if not m.any():
            continue
        ys, xs = np.nonzero(m)
        y0, y1, x0, x1 = int(ys.min()), int(ys.max()), int(xs.min()), int(xs.max())
        cx = (x0 + x1) // 2
        col = m[y0 : y1 + 1, cx]
        f = 0
        while f < col.size and col[f]:
            f += 1
        ix0, iy0, ix1, iy1 = x0 + f, y0 + f, x1 - f, y1 - f
        s = (ix1 - ix0 + 1) / float(COLS)
        depths = []
        for j in range(COLS):
            xc = int(round(ix0 + (j + 0.5) * s))
            colb = black[iy0 : iy1 + 1, xc]
            k = 0
            while k < colb.size and not colb[k]:
                k += 1
            depths.append(int(round(k / s)))
        sockets[name] = tuple(depths)
    if len(sockets) != 5:
        return ""

    # --- pegs --------------------------------------------------------------
    gray = _mask(arr, GRAY)
    if not gray.any():
        return ""
    rule_bottom = int(np.nonzero(gray)[0].max())
    below = black.copy()
    below[: rule_bottom + 1, :] = False

    comps = _components(below)
    if len(comps) < 6:
        return ""
    ring = None
    for a in comps:
        for b in comps:
            if a is b:
                continue
            if a[0] < b[0] and a[1] > b[1] and a[2] < b[2] and a[3] > b[3]:
                ring = a
                break
        if ring is not None:
            break
    if ring is None:
        return ""
    pegs_boxes = [c for c in comps if c is not ring]
    if len(pegs_boxes) != 5:
        return ""
    pegs_boxes.sort(key=lambda c: c[2])

    pegs = []
    ringed = None
    for i, (py0, py1, px0, px1, _n) in enumerate(pegs_boxes):
        s = (px1 - px0 + 1) / float(COLS)
        teeth = []
        for j in range(COLS):
            xc = int(round(px0 + (j + 0.5) * s))
            colb = black[py0 : py1 + 1, xc]
            nz = np.nonzero(colb)[0]
            if nz.size == 0:
                return ""
            hgt = int(nz.max()) + 1
            teeth.append(int(round(hgt / s)) - 1)
        pegs.append(tuple(teeth))
        if ring[2] < px0 and ring[3] > px1 and ring[0] < py0 and ring[1] > py1:
            ringed = i
    if ringed is None:
        return ""

    names = sorted(sockets)
    depth_list = [sockets[n] for n in names]

    found = []
    for perm in permutations(range(5)):
        ok = True
        for i in range(5):
            d = depth_list[perm[i]]
            if any(p > q for p, q in zip(pegs[i], d)):
                ok = False
                break
        if ok:
            found.append(perm)
            if len(found) > 1:
                return ""
    if len(found) != 1:
        return ""
    return names[found[0][ringed]]
