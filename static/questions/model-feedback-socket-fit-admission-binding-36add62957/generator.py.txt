"""Socket-fit admission binding.

Five colored sockets are black blocks with a white opening cut down from the
top; the opening has a per-column depth.  Five black pegs are combs: a top bar
with five teeth of per-column length.  A peg fits a socket exactly when every
tooth is no longer than the opening in that column, so the peg can be pushed
straight down without any tooth overlapping black.

The admissibility relation is drawn nowhere.  It has to be constructed from all
twenty-five peg/socket pairs by comparing silhouettes column by column, and the
answer is then the unique system of distinct representatives over that derived
graph, read off at the ringed peg.
"""

from __future__ import annotations

import random
from itertools import permutations

from PIL import Image, ImageDraw

CANVAS_W = 1600
CANVAS_H = 660

WHITE = (255, 255, 255)
BLACK = (0, 0, 0)
GRAY = (150, 150, 150)

PALETTE = {
    "blue": (0, 90, 200),
    "green": (0, 150, 60),
    "orange": (240, 140, 0),
    "purple": (140, 60, 190),
    "red": (220, 30, 40),
}
COLOR_NAMES = tuple(sorted(PALETTE))

N = 5
COLS = 5
MAX_DEPTH = 4
MAX_TOOTH = 3

RULE_THICKNESS = 14
QUARANTINE_MARGIN = 11.0


# --------------------------------------------------------------------------
# combinatorics
# --------------------------------------------------------------------------

def _fits(peg, depth):
    return all(p <= d for p, d in zip(peg, depth))


def _edges(pegs, depths):
    return [[_fits(pegs[i], depths[j]) for j in range(N)] for i in range(N)]


def _perfect_matchings(edges):
    out = []
    for perm in permutations(range(N)):
        if all(edges[i][perm[i]] for i in range(N)):
            out.append(perm)
            if len(out) > 1:
                return out
    return out


def _greedy_tightest(pegs, depths):
    """Left-to-right greedy: every peg grabs the roomiest-fitting free socket.

    A locally plausible reading that ignores what other pegs are forced to take.
    """
    taken = set()
    assign = {}
    for i in range(N):
        opts = [j for j in range(N) if j not in taken and _fits(pegs[i], depths[j])]
        if not opts:
            return None
        j = max(opts, key=lambda j: (sum(depths[j]), j))
        assign[i] = j
        taken.add(j)
    return assign


def _area(profile):
    return sum(profile)


# --------------------------------------------------------------------------
# sampling
# --------------------------------------------------------------------------

def _sample_profiles(rng):
    depths = []
    while len(depths) < N:
        d = tuple(rng.randint(1, MAX_DEPTH) for _ in range(COLS))
        if min(d) > 2:
            continue
        if d in depths:
            continue
        depths.append(d)
    pegs = []
    while len(pegs) < N:
        p = tuple(rng.randint(0, MAX_TOOTH) for _ in range(COLS))
        if max(p) < 2:
            continue
        if p in pegs:
            continue
        pegs.append(p)
    return pegs, depths


def _structure(rng):
    """Sample pegs, sockets and the ringed peg with a unique valid assignment."""
    for _ in range(4000):
        pegs, depths = _sample_profiles(rng)
        edges = _edges(pegs, depths)
        matchings = _perfect_matchings(edges)
        if len(matchings) != 1:
            continue
        perm = matchings[0]
        degs = [sum(edges[i]) for i in range(N)]
        if sum(1 for d in degs if d >= 2) < 3:
            continue
        cands = [i for i in range(N) if degs[i] >= 3]
        if not cands:
            continue
        query = cands[rng.randrange(len(cands))]
        gold = perm[query]
        opts = [j for j in range(N) if edges[query][j]]
        # the tightest fitting socket is a decoy
        if gold == min(opts, key=lambda j: (_area(depths[j]), j)):
            continue
        # the roomiest fitting socket is a decoy
        if gold == max(opts, key=lambda j: (_area(depths[j]), j)):
            continue
        # the most similar drawn profile is a decoy
        near = min(
            opts,
            key=lambda j: (sum(abs(a - b) for a, b in zip(depths[j], pegs[query])), j),
        )
        if gold == near:
            continue
        # a socket at least as roomy as the peg that nevertheless does not fit
        if not any(
            _area(depths[j]) >= _area(pegs[query]) + 3 and not edges[query][j]
            for j in range(N)
        ):
            continue
        greedy = _greedy_tightest(pegs, depths)
        if greedy is not None and greedy.get(query) == gold:
            continue
        return pegs, depths, query, perm
    raise RuntimeError("structure sampling failed")


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    pegs, depths, query, perm = _structure(rng)
    colors = list(COLOR_NAMES)
    rng.shuffle(colors)
    cell = rng.choice([30, 33, 33, 36, 36, 39])
    scene = {
        "seed": int(seed),
        "pegs": [list(p) for p in pegs],
        "depths": [list(d) for d in depths],
        "query": int(query),
        "socket_colors": colors,
        "cell": int(cell),
        "frame": int(rng.randint(12, 16)),
        "ring_stroke": int(rng.randint(12, 14)),
        "ring_gap": int(rng.randint(16, 30)),
        "socket_gap": int(rng.randint(55, 75)),
        "peg_gap": int(rng.randint(110, 130)),
        "socket_dy": int(rng.randint(0, 10)),
        "socket_dx": [int(rng.randint(-6, 6)) for _ in range(N)],
        "peg_dx": [int(rng.randint(-5, 5)) for _ in range(N)],
    }
    return scene


# --------------------------------------------------------------------------
# layout
# --------------------------------------------------------------------------

def _layout(scene):
    s = scene["cell"]
    f = scene["frame"]
    inner_w = COLS * s
    inner_h = MAX_DEPTH * s
    outer_w = inner_w + 2 * f
    outer_h = inner_h + 2 * f
    gap = scene["socket_gap"]
    total = N * outer_w + (N - 1) * gap
    x0 = (CANVAS_W - total) // 2
    y0 = 50 + scene["socket_dy"]
    sockets = []
    for j in range(N):
        sx = x0 + j * (outer_w + gap) + scene["socket_dx"][j]
        sockets.append((sx, y0, outer_w, outer_h))

    rule_y = y0 + outer_h + 60
    peg_top = rule_y + RULE_THICKNESS + 80

    pgap = scene["peg_gap"]
    pw = COLS * s
    ptotal = N * pw + (N - 1) * pgap
    px0 = (CANVAS_W - ptotal) // 2
    pegs = []
    for i in range(N):
        px = px0 + i * (pw + pgap) + scene["peg_dx"][i]
        pegs.append((px, peg_top, pw))
    return {
        "cell": s,
        "frame": f,
        "inner_w": inner_w,
        "inner_h": inner_h,
        "sockets": sockets,
        "rule_y": rule_y,
        "peg_top": peg_top,
        "peg_w": pw,
        "pegs": pegs,
    }


def render(scene: dict) -> Image.Image:
    lay = _layout(scene)
    s = lay["cell"]
    f = lay["frame"]
    img = Image.new("RGB", (CANVAS_W, CANVAS_H), WHITE)
    dr = ImageDraw.Draw(img)

    for j, (sx, sy, ow, oh) in enumerate(lay["sockets"]):
        hue = PALETTE[scene["socket_colors"][j]]
        dr.rectangle([sx, sy, sx + ow - 1, sy + oh - 1], fill=hue)
        ix, iy = sx + f, sy + f
        dr.rectangle([ix, iy, ix + lay["inner_w"] - 1, iy + lay["inner_h"] - 1], fill=WHITE)
        depth = scene["depths"][j]
        for c in range(COLS):
            filled = MAX_DEPTH - depth[c]
            if filled <= 0:
                continue
            cx = ix + c * s
            cy = iy + depth[c] * s
            dr.rectangle([cx, cy, cx + s - 1, iy + lay["inner_h"] - 1], fill=BLACK)

    dr.rectangle(
        [80, lay["rule_y"], CANVAS_W - 81, lay["rule_y"] + RULE_THICKNESS - 1],
        fill=GRAY,
    )

    for i, (px, py, pw) in enumerate(lay["pegs"]):
        dr.rectangle([px, py, px + pw - 1, py + s - 1], fill=BLACK)
        teeth = scene["pegs"][i]
        for c in range(COLS):
            if teeth[c] <= 0:
                continue
            cx = px + c * s
            dr.rectangle([cx, py + s, cx + s - 1, py + s + teeth[c] * s - 1], fill=BLACK)

    q = scene["query"]
    px, py, pw = lay["pegs"][q]
    ph = s + max(scene["pegs"][q]) * s
    g = scene["ring_gap"]
    w = scene["ring_stroke"]
    dr.rectangle(
        [px - g - w, py - g - w, px + pw - 1 + g + w, py + ph - 1 + g + w],
        outline=BLACK,
        width=w,
    )
    return img


def solve(scene: dict):
    """The unique perfect matching, recomputed from the drawn profiles."""
    pegs = [tuple(p) for p in scene["pegs"]]
    depths = [tuple(d) for d in scene["depths"]]
    found = _perfect_matchings(_edges(pegs, depths))
    if len(found) != 1:
        raise ValueError("scene does not carry a unique assignment")
    return found[0]


def analytic_gold(scene: dict) -> str:
    return scene["socket_colors"][solve(scene)[scene["query"]]]


def margin(scene: dict) -> float:
    lay = _layout(scene)
    s = lay["cell"]
    f = lay["frame"]
    g = scene["ring_gap"]
    w = scene["ring_stroke"]
    q = scene["query"]

    terms = [
        float(f),                        # socket frame stroke
        float(w),                        # ring stroke
        float(g),                        # ring-to-peg clear gap
        float(RULE_THICKNESS),           # separator rule
        s / 3.0,                         # decisive depth-quantisation scale
    ]

    # socket-to-socket clearance
    socks = lay["sockets"]
    for a, b in zip(socks, socks[1:]):
        terms.append(float(b[0] - (a[0] + a[2])))
    terms.append(float(socks[0][0]))
    terms.append(float(CANVAS_W - (socks[-1][0] + socks[-1][2])))
    terms.append(float(socks[0][1]))

    # ring extent versus neighbours, rule and canvas
    px, py, pw = lay["pegs"][q]
    ph = s + max(scene["pegs"][q]) * s
    rx0, ry0 = px - g - w, py - g - w
    rx1, ry1 = px + pw - 1 + g + w, py + ph - 1 + g + w
    terms.append(float(ry0 - (lay["rule_y"] + RULE_THICKNESS)))
    terms.append(float(CANVAS_H - 1 - ry1))
    terms.append(float(rx0))
    terms.append(float(CANVAS_W - 1 - rx1))
    for i, (ox, oy, ow) in enumerate(lay["pegs"]):
        if i == q:
            continue
        if ox > px:
            terms.append(float(ox - rx1))
        else:
            terms.append(float(rx0 - (ox + ow)))
    # peg-to-peg clearance
    pegs = lay["pegs"]
    for a, b in zip(pegs, pegs[1:]):
        terms.append(float(b[0] - (a[0] + a[2])))
    # peg bottoms clear of the canvas
    for i, (ox, oy, ow) in enumerate(pegs):
        terms.append(float(CANVAS_H - 1 - (oy + s + max(scene["pegs"][i]) * s)))
    return float(min(terms))


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict):
    return []
