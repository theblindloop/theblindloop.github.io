"""Image-only inverse arm for gap_bridge_flag.

Receives one rendered PNG and nothing else: no scene, no renderer, no gold. It
segments the painted squares by colour, labels them as separate blobs, measures
the clear pixel gap between every pair of blobs, links pairs whose gap is
narrower than the typical gray stone, and reports the colour of the single flag
that is *not* reachable from the green pad.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

ORACLE_VERSION = "gap_bridge_flag-oracle-0.1.0"

_REFERENCES = {
    "stone": (60, 60, 68),
    "pad": (30, 120, 60),
    "orange": (230, 125, 30),
    "purple": (130, 55, 165),
    "blue": (35, 95, 205),
}
_FLAG_NAMES = ("orange", "purple", "blue")
_ABSTAIN = "abstain"


def _ink_mask(array: np.ndarray) -> np.ndarray:
    """Anything clearly darker or more saturated than the plain background."""

    return array.astype(np.int16).min(axis=2) < 210


def _label(mask: np.ndarray) -> tuple[np.ndarray, int]:
    labels = np.zeros(mask.shape, dtype=np.int32)
    count = 0
    height, width = mask.shape
    for start_y, start_x in zip(*np.nonzero(mask)):
        if labels[start_y, start_x]:
            continue
        count += 1
        labels[start_y, start_x] = count
        queue = deque([(int(start_y), int(start_x))])
        while queue:
            y, x = queue.popleft()
            for dy, dx in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                ny, nx = y + dy, x + dx
                if 0 <= ny < height and 0 <= nx < width:
                    if mask[ny, nx] and not labels[ny, nx]:
                        labels[ny, nx] = count
                        queue.append((ny, nx))
    return labels, count


def _classify(mean_rgb: np.ndarray) -> str:
    best = None
    for name, reference in _REFERENCES.items():
        distance = float(np.linalg.norm(mean_rgb - np.array(reference, dtype=float)))
        if best is None or distance < best[0]:
            best = (distance, name)
    return best[1]


def _blob_gap(a: dict, b: dict) -> float:
    dx = max(0, b["x0"] - a["x1"] - 1, a["x0"] - b["x1"] - 1)
    dy = max(0, b["y0"] - a["y1"] - 1, a["y0"] - b["y1"] - 1)
    return float(np.hypot(float(dx), float(dy)))


def _blobs(image: Image.Image) -> list[dict]:
    array = np.asarray(image.convert("RGB"))
    labels, count = _label(_ink_mask(array))
    blobs: list[dict] = []
    for index in range(1, count + 1):
        ys, xs = np.nonzero(labels == index)
        if ys.size < 24:  # stray speck, not a painted square
            continue
        mean_rgb = array[ys, xs].astype(float).mean(axis=0)
        blobs.append(
            {
                "kind": _classify(mean_rgb),
                "x0": int(xs.min()),
                "x1": int(xs.max()),
                "y0": int(ys.min()),
                "y1": int(ys.max()),
                "side": float(max(xs.max() - xs.min() + 1, ys.max() - ys.min() + 1)),
            }
        )
    return blobs


def decision_from_image(image: Image.Image) -> str:
    """Colour of the flag cut off from the green pad, or ``abstain``."""

    blobs = _blobs(image)
    pads = [index for index, blob in enumerate(blobs) if blob["kind"] == "pad"]
    flags = [index for index, blob in enumerate(blobs) if blob["kind"] in _FLAG_NAMES]
    stones = [blob["side"] for blob in blobs if blob["kind"] == "stone"]
    present = sorted({blobs[index]["kind"] for index in flags})
    if len(pads) != 1 or len(present) != 3 or not stones:
        return _ABSTAIN

    threshold = float(np.median(np.asarray(stones)))
    seen = {pads[0]}
    queue = deque([pads[0]])
    while queue:
        current = queue.popleft()
        for index, blob in enumerate(blobs):
            if index in seen:
                continue
            if _blob_gap(blobs[current], blob) < threshold:
                seen.add(index)
                queue.append(index)

    reached = {blobs[index]["kind"] for index in seen if index in flags}
    cut = sorted(set(present) - reached)
    if len(cut) != 1:
        return _ABSTAIN
    return cut[0]
