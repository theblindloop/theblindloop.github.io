"""Independent pixel-only inverse arm for the ledge-shade world.

Reads the final PNG alone: no latent scene, no renderer import, no gold.

Procedure: segment the four flat inks by exact colour, group the ledge ink into
horizontal bands top to bottom, measure each band's reach in unit cells from
the common left edge of the ledges, note which bands carry a black tag in the
left margin, then recompute the lighting of the five worlds in which one
tagged ledge is absent and report the tag that frees the most shadowed ledges.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

_WALL = (40, 42, 48)
_SHADE = (120, 126, 140)
_LIT = (250, 205, 45)
_TAG = (0, 0, 0)
_CELL = 56


def _mask(arr: np.ndarray, colour) -> np.ndarray:
    return np.all(arr == np.array(colour, dtype=arr.dtype), axis=-1)


def _bands(rows: np.ndarray) -> list[tuple[int, int]]:
    out = []
    start = None
    for y, on in enumerate(rows):
        if on and start is None:
            start = y
        elif not on and start is not None:
            out.append((start, y - 1))
            start = None
    if start is not None:
        out.append((start, len(rows) - 1))
    return out


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    ledge = _mask(arr, _SHADE) | _mask(arr, _LIT)
    wall = _mask(arr, _WALL)
    tag = _mask(arr, _TAG)

    if not ledge.any():
        return "mark-1"
    xs = np.where(ledge.any(axis=0))[0]
    left = int(xs.min())
    wall_xs = np.where(wall.any(axis=0))[0]
    wall_left = int(wall_xs.min()) if wall_xs.size else left

    reaches: list[int] = []
    tagged: list[bool] = []
    for y0, y1 in _bands(ledge.any(axis=1)):
        strip = ledge[y0 : y1 + 1]
        cols = np.where(strip.any(axis=0))[0]
        reach_px = int(cols.max()) + 1 - left
        reaches.append(int(round(reach_px / _CELL)))
        margin = tag[y0 : y1 + 1, :wall_left]
        tagged.append(bool(margin.any()))

    marks = [i for i, t in enumerate(tagged) if t]

    def prefix_max(upto: int, skip: int = -1) -> int:
        best = 0
        for k in range(upto):
            if k != skip and reaches[k] > best:
                best = reaches[k]
        return best

    scores = []
    for m in marks:
        freed = 0
        for j in range(m + 1, len(reaches)):
            if reaches[j] > prefix_max(j):
                continue
            if reaches[j] > prefix_max(j, skip=m):
                freed += 1
        scores.append(freed)

    if not scores:
        return "mark-1"
    return f"mark-{scores.index(max(scores)) + 1}"
