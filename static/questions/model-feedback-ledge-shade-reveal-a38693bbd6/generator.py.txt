"""Ledge-shade world: which tagged ledge, if taken away, frees the most shade?

Eleven ledges stick out to the right from a dark wall, each by a whole number
of unit cells, and light falls straight down the page. The lit part of a ledge
is the part that no ledge above it covers, painted yellow; a ledge that no
light reaches at all is drawn wholly grey. The factual lighting is therefore
fully rendered.

Five ledges carry a black tag. The question asks which single tagged ledge,
taken away, would put light on the greatest number of ledges that are wholly
shadowed as drawn. The answer names an intervention rather than the value of
one: five hypothetical worlds must each be re-lit, and a removal only frees
shade if the removed ledge is itself lit and if the ledges below it form a
rising run of reaches, so no tagged ledge can be scored from its own drawing.

The renderer owns scene sampling, rasterization and the structural analytic
gold. The independent pixel-only inverse arm (``oracle.py``) recovers the same
decision from the final PNG alone.
"""

from __future__ import annotations

import random

from PIL import Image, ImageDraw

WIDTH = 812
HEIGHT = 660
BACKGROUND = (255, 255, 255)

N_ROWS = 11
N_MARKS = 5
CELL = 56
MAX_REACH = 12

TAG_X0 = 16
TAG_X1 = 56
WALL_X0 = 68
WALL_X1 = 112
BAR_X0 = WALL_X1

ROW_Y0 = 22
ROW_PITCH = 56
BAR_H = 38

WALL_COLOR = (40, 42, 48)
SHADE_COLOR = (120, 126, 140)
LIT_COLOR = (250, 205, 45)
TAG_COLOR = (0, 0, 0)

QUARANTINE_MIN_MARGIN = 2.0 * CELL


def _row_top(row: int) -> int:
    return ROW_Y0 + row * ROW_PITCH


def _prefix_max(reaches, upto: int, skip: int = -1) -> int:
    """Largest reach strictly above row ``upto``, ignoring row ``skip``."""
    best = 0
    for k in range(upto):
        if k == skip:
            continue
        if reaches[k] > best:
            best = reaches[k]
    return best


def lit_cells(reaches) -> list[int]:
    """Number of lit unit cells at the tip of every ledge, as drawn."""
    out = []
    for i, r in enumerate(reaches):
        cover = _prefix_max(reaches, i)
        out.append(max(0, r - cover))
    return out


def freed_count(reaches, removed: int) -> int:
    """How many wholly shadowed ledges catch light once ``removed`` is gone."""
    freed = 0
    for j in range(removed + 1, len(reaches)):
        if reaches[j] > _prefix_max(reaches, j):
            continue  # already lit as drawn
        if reaches[j] > _prefix_max(reaches, j, skip=removed):
            freed += 1
    return freed


def _scores(scene: dict) -> list[int]:
    reaches, marks = scene["reaches"], scene["marks"]
    return [freed_count(reaches, m) for m in marks]


def _sorted_scores(scene: dict) -> list[int]:
    return sorted(_scores(scene), reverse=True)


def _valid(reaches, marks) -> bool:
    if len(set(marks)) != N_MARKS or sorted(marks) != list(marks):
        return False
    if len(set(reaches)) != len(reaches):
        return False
    scores = [freed_count(reaches, m) for m in marks]
    order = sorted(scores, reverse=True)
    best, second = order[0], order[1]
    if best < 2 or best == second:
        return False
    winner = marks[scores.index(best)]
    # A tagged ledge that is itself wholly shadowed frees nothing at all.
    if sum(1 for s in scores if s == 0) < 2:
        return False
    # The longest tagged ledge is never the answer.
    if max(range(N_MARKS), key=lambda k: reaches[marks[k]]) == scores.index(best):
        return False
    # Nor is the tagged ledge carrying the longest lit run.
    lit = lit_cells(reaches)
    if max(range(N_MARKS), key=lambda k: lit[marks[k]]) == scores.index(best):
        return False
    return True


def sample_scene(seed: int) -> dict:
    """Sample one scene: ledge reaches, the five tagged rows, and paint order.

    Reaches are built first and every row is scored by how much shade its own
    removal would free; the five tags are then chosen so that the winner is
    unique, at least two tags are wholly shadowed and so free nothing at all,
    and both the longest tag and the tag with the longest lit run are wrong.
    """
    rng = random.Random(seed * 6700417 + 11)
    # The winner's ordinal position among the five tags is fixed before the
    # scene is built, so no tag position is more often the answer.
    target_rank = rng.choice([2, 3, 4])
    for _ in range(4000):
        built = _build_reaches(rng)
        if built is None:
            continue
        reaches, winner = built
        scores = [freed_count(reaches, i) for i in range(N_ROWS)]
        if scores[winner] < 2 or max(scores) != scores[winner]:
            continue
        pool = [i for i in range(N_ROWS) if i != winner and scores[i] < scores[winner]]
        above = [i for i in pool if i < winner]
        below = [i for i in pool if i > winner]
        longer = [i for i in below if reaches[i] > reaches[winner]]
        if not longer:
            continue
        rank = target_rank
        if len(above) < rank - 1 or len(below) < N_MARKS - rank:
            continue
        for _ in range(40):
            picked = set(rng.sample(above, rank - 1)) | {rng.choice(longer)}
            rest = [i for i in below if i not in picked]
            need = N_MARKS - 1 - len(picked)
            if len(rest) < need:
                break
            picked |= set(rng.sample(rest, need))
            marks = sorted(picked | {winner})
            if marks.index(winner) + 1 != rank:
                continue
            if not _valid(reaches, marks):
                continue
            paint_order = list(range(N_ROWS))
            rng.shuffle(paint_order)
            return {
                "n_rows": N_ROWS,
                "reaches": reaches,
                "marks": marks,
                "paint_order": paint_order,
            }
    raise RuntimeError(f"no valid scene for seed {seed}")


def _build_reaches(rng: random.Random):
    """Build a reach profile with one row whose removal frees a rising run.

    Rows above the winner rise gently to a plateau ``m``; the winner jumps
    clear of that plateau; ``k`` rows below it hold values inside the jump, in
    increasing order, so all of them are shadowed as drawn and all of them are
    freed together when the winner goes; fillers stay under the plateau and a
    tail of longer, already-lit ledges closes the picture.
    """
    k = rng.choice([2, 2, 3, 3, 4])
    tail_n = rng.choice([1, 2])
    lo_m, hi_m = max(2, N_ROWS - 1 - k - tail_n), MAX_REACH - tail_n - k - 1
    if lo_m > hi_m:
        return None
    m = rng.randint(lo_m, hi_m)
    n_above = rng.randint(1, min(m, 4))
    fillers_n = N_ROWS - 1 - n_above - k - tail_n
    if fillers_n < 0 or fillers_n > m - n_above:
        return None
    lo_r, hi_r = m + k + 1, MAX_REACH - tail_n
    if lo_r > hi_r:
        return None
    reach_w = rng.randint(lo_r, hi_r)

    above = sorted(rng.sample(range(1, m), n_above - 1) + [m])
    filler_pool = [v for v in range(1, m) if v not in above]
    fillers = rng.sample(filler_pool, fillers_n)
    freed = sorted(rng.sample(range(m + 1, reach_w), k))
    tail = rng.sample(range(reach_w + 1, MAX_REACH + 1), tail_n)
    rng.shuffle(tail)

    middle = [None] * (k + fillers_n)
    slots = sorted(rng.sample(range(len(middle)), k))
    for value, slot in zip(freed, slots):
        middle[slot] = value
    spare = iter(fillers)
    middle = [v if v is not None else next(spare) for v in middle]

    reaches = above + [reach_w] + middle + tail
    if len(reaches) != N_ROWS or len(set(reaches)) != N_ROWS:
        return None
    return reaches, n_above


def render(scene: dict) -> Image.Image:
    reaches = list(scene["reaches"])
    marks = set(scene["marks"])
    image = Image.new("RGB", (WIDTH, HEIGHT), BACKGROUND)
    draw = ImageDraw.Draw(image)

    draw.rectangle([WALL_X0, 0, WALL_X1 - 1, HEIGHT - 1], fill=WALL_COLOR)

    for row in scene["paint_order"]:
        top = _row_top(row)
        bottom = top + BAR_H - 1
        reach = reaches[row]
        cover = min(_prefix_max(reaches, row), reach)
        x_end = BAR_X0 + reach * CELL - 1
        x_cover = BAR_X0 + cover * CELL
        if cover > 0:
            draw.rectangle([BAR_X0, top, x_cover - 1, bottom], fill=SHADE_COLOR)
        if reach > cover:
            draw.rectangle([x_cover, top, x_end, bottom], fill=LIT_COLOR)
        if row in marks:
            draw.rectangle([TAG_X0, top, TAG_X1 - 1, bottom], fill=TAG_COLOR)
    return image


def analytic_gold(scene: dict) -> str:
    scores = _scores(scene)
    return f"mark-{scores.index(max(scores)) + 1}"


def margin(scene: dict) -> float:
    order = _sorted_scores(scene)
    return float((order[0] - order[1]) * CELL)


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MIN_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Paint order is a sampled field that no pixel can express."""
    order = list(scene["paint_order"])
    twins = []
    rotated = dict(scene)
    rotated["paint_order"] = order[1:] + order[:1]
    twins.append(("paint_order_rotated", rotated))
    reversed_twin = dict(scene)
    reversed_twin["paint_order"] = list(reversed(order))
    twins.append(("paint_order_reversed", reversed_twin))
    return twins
