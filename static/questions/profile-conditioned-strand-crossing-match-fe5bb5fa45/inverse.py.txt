"""Pixel-only inverse arm; it has no access to renderer state or latent scenes."""

from __future__ import annotations

from PIL import Image

PALETTE = (
    (13, 132, 137),
    (230, 105, 28),
    (126, 65, 158),
)
LABELS = ("A", "B", "C", "D")


def _color_masks(image: Image.Image) -> list[list[tuple[int, int]]]:
    rgb = image.convert("RGB")
    pixels = rgb.load()
    masks: list[list[tuple[int, int]]] = [[] for _ in PALETTE]
    for y in range(rgb.height):
        for x in range(rgb.width):
            value = pixels[x, y]
            for index, target in enumerate(PALETTE):
                if sum((value[channel] - target[channel]) ** 2 for channel in range(3)) <= 32 ** 2:
                    masks[index].append((x, y))
                    break
    if any(len(mask) < 100 for mask in masks):
        raise ValueError("colored strand evidence is missing")
    return masks


def _bands(values: set[int]) -> list[tuple[int, int]]:
    if not values:
        return []
    ordered = sorted(values)
    runs: list[list[int]] = [[ordered[0]]]
    for value in ordered[1:]:
        if value > runs[-1][-1] + 1:
            runs.append([value])
        else:
            runs[-1].append(value)
    return [(run[0], run[-1]) for run in runs]


def _relation_for_region(
    masks: list[list[tuple[int, int]]],
    region: tuple[int, int, int, int],
) -> frozenset[tuple[int, int]]:
    x0, y0, x1, y1 = region
    per_color = [
        [(x, y) for x, y in mask if x0 <= x <= x1 and y0 <= y <= y1]
        for mask in masks
    ]
    if any(len(points) < 40 for points in per_color):
        raise ValueError("a panel is missing one colored strand")
    colored_x = [x for points in per_color for x, _ in points]
    left_edge, right_edge = min(colored_x), max(colored_x)
    span = right_edge - left_edge
    if span < 80:
        raise ValueError("panel strand span is too small")
    left_cut = left_edge + 0.13 * span
    right_cut = right_edge - 0.13 * span
    left_y: list[float] = []
    right_y: list[float] = []
    for points in per_color:
        ys_left = sorted(y for x, y in points if x <= left_cut)
        ys_right = sorted(y for x, y in points if x >= right_cut)
        if len(ys_left) < 8 or len(ys_right) < 8:
            raise ValueError("strand endpoints cannot be recovered")
        left_y.append(float(ys_left[len(ys_left) // 2]))
        right_y.append(float(ys_right[len(ys_right) // 2]))
    if min(abs(a - b) for i, a in enumerate(left_y) for b in left_y[i + 1 :]) < 18:
        raise ValueError("left endpoint order is ambiguous")
    if min(abs(a - b) for i, a in enumerate(right_y) for b in right_y[i + 1 :]) < 18:
        raise ValueError("right endpoint order is ambiguous")
    return frozenset(
        (a, b)
        for a in range(3)
        for b in range(a + 1, 3)
        if (left_y[a] - left_y[b]) * (right_y[a] - right_y[b]) < 0
    )


def decision_from_image(image: Image.Image) -> str:
    masks = _color_masks(image)
    all_points = [point for mask in masks for point in mask]
    y_runs = _bands({y for _, y in all_points})
    # Merge any nearby projection runs created by shallow strands.
    merged_y: list[list[int]] = []
    for lo, hi in y_runs:
        if merged_y and lo - merged_y[-1][1] < 100:
            merged_y[-1][1] = hi
        else:
            merged_y.append([lo, hi])
    if len(merged_y) != 2:
        raise ValueError("expected one reference row and one candidate row")
    top, bottom = merged_y
    reference_region = (0, top[0] - 2, image.width - 1, top[1] + 2)
    bottom_points = [(x, y) for x, y in all_points if bottom[0] <= y <= bottom[1]]
    x_runs = _bands({x for x, _ in bottom_points})
    merged_x: list[list[int]] = []
    for lo, hi in x_runs:
        if merged_x and lo - merged_x[-1][1] < 35:
            merged_x[-1][1] = hi
        else:
            merged_x.append([lo, hi])
    if len(merged_x) != 4:
        raise ValueError("expected four candidate panels")
    target = _relation_for_region(masks, reference_region)
    relations = [
        _relation_for_region(masks, (lo - 2, bottom[0] - 2, hi + 2, bottom[1] + 2))
        for lo, hi in merged_x
    ]
    matches = [index for index, relation in enumerate(relations) if relation == target]
    if len(matches) != 1:
        raise ValueError("pixel evidence does not identify a unique panel")
    return LABELS[matches[0]]
