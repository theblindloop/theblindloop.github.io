"""Deterministic renderer for reference-guided strand crossing search."""

from __future__ import annotations

import itertools
import random
from copy import deepcopy

from PIL import Image, ImageDraw, ImageFont

COLORS = (
    (13, 132, 137),   # teal
    (230, 105, 28),   # orange
    (126, 65, 158),   # purple
)
INK = (31, 38, 48)
FRAME = (150, 158, 168)
PANEL_LABELS = ("A", "B", "C", "D")
QUESTION = (
    "Which labeled panel has exactly the same pairs of colored strands "
    "crossing as the REFERENCE? A-D run left to right."
)


def _relation(pair: list[list[int]]) -> tuple[tuple[int, int], ...]:
    left, right = pair[0], pair[1]
    lp = {color: index for index, color in enumerate(left)}
    rp = {color: index for index, color in enumerate(right)}
    return tuple(
        (a, b)
        for a in range(3)
        for b in range(a + 1, 3)
        if (lp[a] - lp[b]) * (rp[a] - rp[b]) < 0
    )


def _pairs_for_relation(relation: frozenset[tuple[int, int]]) -> list[list[list[int]]]:
    choices: list[list[list[int]]] = []
    for left in itertools.permutations(range(3)):
        for right in itertools.permutations(range(3)):
            pair = [list(left), list(right)]
            if frozenset(_relation(pair)) == relation:
                choices.append(pair)
    return choices


ALL_RELATIONS = tuple(
    frozenset(edges)
    for size in range(4)
    for edges in itertools.combinations(((0, 1), (0, 2), (1, 2)), size)
)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    medium = [relation for relation in ALL_RELATIONS if len(relation) in (1, 2)]
    target_relation = rng.choice(medium)
    reference_order = deepcopy(rng.choice(_pairs_for_relation(target_relation)))

    same_count = [
        relation
        for relation in medium
        if relation != target_relation and len(relation) == len(target_relation)
    ]
    distractor_relations = [rng.choice(same_count)]
    remaining = [
        relation
        for relation in ALL_RELATIONS
        if relation != target_relation and relation not in distractor_relations
    ]
    distractor_relations.extend(rng.sample(remaining, 2))
    relations = [target_relation, *distractor_relations]
    candidate_orders = [deepcopy(rng.choice(_pairs_for_relation(rel))) for rel in relations]
    panel_order = list(range(4))
    rng.shuffle(panel_order)
    return {
        "reference_order": reference_order,
        "candidate_orders": candidate_orders,
        "panel_order": panel_order,
        "style_variant": rng.randrange(2),
    }


def _draw_panel(
    draw: ImageDraw.ImageDraw,
    box: tuple[int, int, int, int],
    pair: list[list[int]],
) -> None:
    x0, y0, x1, y1 = box
    draw.rounded_rectangle(box, radius=9, fill=(249, 250, 252), outline=FRAME, width=2)
    lx, rx = x0 + 28, x1 - 28
    levels = [y0 + 34, (y0 + y1) // 2, y1 - 34]
    left_pos = {color: index for index, color in enumerate(pair[0])}
    right_pos = {color: index for index, color in enumerate(pair[1])}
    # Equal widths and equal endpoint sizes keep total colored ink essentially fixed.
    for color in range(3):
        start = (lx, levels[left_pos[color]])
        end = (rx, levels[right_pos[color]])
        draw.line((start, end), fill=(255, 255, 255), width=15)
        draw.line((start, end), fill=COLORS[color], width=9)
    radius = 9
    for color in range(3):
        for x, y in (
            (lx, levels[left_pos[color]]),
            (rx, levels[right_pos[color]]),
        ):
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=COLORS[color])


def render(scene: dict) -> Image.Image:
    reference_order = scene["reference_order"]
    candidate_orders = scene["candidate_orders"]
    panel_order = scene["panel_order"]
    _ = scene["style_variant"]  # explicit render-invariant serialization alias
    image = Image.new("RGB", (1000, 640), (255, 255, 255))
    draw = ImageDraw.Draw(image)
    font = ImageFont.load_default()
    draw.text((30, 18), QUESTION, fill=INK, font=font)
    draw.text((463, 61), "REFERENCE", fill=INK, font=font)
    _draw_panel(draw, (350, 82, 650, 282), reference_order)
    boxes = (
        (25, 358, 245, 566),
        (268, 358, 488, 566),
        (511, 358, 731, 566),
        (754, 358, 974, 566),
    )
    for position, box in enumerate(boxes):
        _draw_panel(draw, box, candidate_orders[panel_order[position]])
        draw.text(((box[0] + box[2]) // 2 - 3, 584), PANEL_LABELS[position], fill=INK, font=font)
    return image


def analytic_gold(scene: dict) -> str:
    reference_order = scene["reference_order"]
    candidate_orders = scene["candidate_orders"]
    panel_order = scene["panel_order"]
    target = _relation(reference_order)
    matches = [
        position
        for position, source_index in enumerate(panel_order)
        if _relation(candidate_orders[source_index]) == target
    ]
    if len(matches) != 1:
        raise ValueError("scene does not have a unique matching panel")
    return PANEL_LABELS[matches[0]]


def margin(scene: dict) -> float:
    reference_order = scene["reference_order"]
    candidate_orders = scene["candidate_orders"]
    panel_order = scene["panel_order"]
    target = set(_relation(reference_order))
    distances = sorted(
        len(target.symmetric_difference(set(_relation(candidate_orders[index]))))
        for index in panel_order
    )
    return float(distances[1]) if len(distances) >= 2 and distances[0] == 0 else 0.0


def is_quarantined(scene: dict) -> bool:
    reference_order = scene["reference_order"]
    candidate_orders = scene["candidate_orders"]
    panel_order = scene["panel_order"]
    target = _relation(reference_order)
    matches = sum(_relation(candidate_orders[index]) == target for index in panel_order)
    return matches != 1 or margin(scene) < 1.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["style_variant"] = 1 - int(twin["style_variant"])
    return [("serialization_style_alias", twin)]
