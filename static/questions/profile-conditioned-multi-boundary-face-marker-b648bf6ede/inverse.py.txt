"""Independent pixel-only inverse for the two-boundary face question."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


def _label_components(mask: np.ndarray) -> np.ndarray:
    height, width = mask.shape
    labels = np.full((height, width), -1, dtype=np.int32)
    component = 0
    for row in range(height):
        for column in range(width):
            if not bool(mask[row, column]) or labels[row, column] >= 0:
                continue
            labels[row, column] = component
            queue = deque([(row, column)])
            while queue:
                y, x = queue.popleft()
                if y and mask[y - 1, x] and labels[y - 1, x] < 0:
                    labels[y - 1, x] = component
                    queue.append((y - 1, x))
                if y + 1 < height and mask[y + 1, x] and labels[y + 1, x] < 0:
                    labels[y + 1, x] = component
                    queue.append((y + 1, x))
                if x and mask[y, x - 1] and labels[y, x - 1] < 0:
                    labels[y, x - 1] = component
                    queue.append((y, x - 1))
                if x + 1 < width and mask[y, x + 1] and labels[y, x + 1] < 0:
                    labels[y, x + 1] = component
                    queue.append((y, x + 1))
            component += 1
    return labels


def _marker_center(mask: np.ndarray, label: str) -> tuple[int, int]:
    points = np.argwhere(mask)
    if len(points) < 40:
        raise ValueError(f"{label} marker is not recoverable from pixels")
    y, x = points.mean(axis=0)
    iy, ix = int(round(float(y))), int(round(float(x)))
    if iy < 0 or ix < 0 or iy >= mask.shape[0] or ix >= mask.shape[1] or not mask[iy, ix]:
        distances = (points[:, 0] - y) ** 2 + (points[:, 1] - x) ** 2
        iy, ix = points[int(np.argmin(distances))]
    return int(iy), int(ix)


def _incident_wall_counts(wall: np.ndarray, marker_centres: dict[str, tuple[int, int]]) -> dict[str, int]:
    wall_labels = _label_components(wall)
    face_labels = _label_components(~wall)
    incident: dict[int, set[int]] = {}
    height, width = wall.shape
    for row in range(height):
        for column in range(width):
            if not wall[row, column]:
                continue
            wall_component = int(wall_labels[row, column])
            neighbours = []
            if row:
                neighbours.append((row - 1, column))
            if row + 1 < height:
                neighbours.append((row + 1, column))
            if column:
                neighbours.append((row, column - 1))
            if column + 1 < width:
                neighbours.append((row, column + 1))
            for neighbour_y, neighbour_x in neighbours:
                face = int(face_labels[neighbour_y, neighbour_x])
                if face >= 0:
                    incident.setdefault(face, set()).add(wall_component)

    result = {}
    for label, centre in marker_centres.items():
        face = int(face_labels[centre[0], centre[1]])
        if face < 0:
            raise ValueError(f"{label} marker lies on a dark contour")
        result[label] = len(incident.get(face, set()))
    return result


def decision_from_image(image: Image.Image) -> str:
    """Recover the selected color from only the final RGB raster."""

    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    channels = array.astype(np.int16)
    red, green, blue = channels[..., 0], channels[..., 1], channels[..., 2]

    red_marker = (red > 145) & (red > green + 52) & (red > blue + 38)
    green_marker = (green > 105) & (green > red + 24) & (green > blue + 14)
    blue_marker = (blue > 120) & (blue > red + 38) & (blue > green + 22)

    wall_max = channels.max(axis=2)
    wall_min = channels.min(axis=2)
    wall = (wall_max < 135) & ((wall_max - wall_min) < 42)
    centres = {
        "red": _marker_center(red_marker, "red"),
        "green": _marker_center(green_marker, "green"),
        "blue": _marker_center(blue_marker, "blue"),
    }
    counts = _incident_wall_counts(wall, centres)
    winners = [label for label, count in counts.items() if count == 2]
    if len(winners) != 1:
        raise ValueError(f"pixel face-boundary counts are ambiguous: {counts}")
    return winners[0]
