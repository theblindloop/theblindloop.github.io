"""Renderer and analytic arm for the two-boundary face marker question."""

from __future__ import annotations

import copy
import math
import random
from typing import Sequence

from PIL import Image, ImageDraw

WIDTH = 512
HEIGHT = 384
SUPERSAMPLE = 3
BACKGROUND = (249, 247, 242)
WALL_CHOICES = ((31, 39, 48), (39, 45, 54), (47, 48, 56), (27, 42, 48))
MARKER_BASE = {
    "red": (220, 63, 76),
    "green": (48, 153, 101),
    "blue": (54, 105, 214),
}


def _rounded_rect(cx: float, cy: float, width: float, height: float, radius: float) -> list[list[float]]:
    """Return a gently rounded, closed polygon in clockwise order."""

    radius = min(float(radius), float(width) / 2.0 - 2.0, float(height) / 2.0 - 2.0)
    result: list[list[float]] = []
    corners = (
        (cx - width / 2.0 + radius, cy - height / 2.0 + radius, 180.0, 270.0),
        (cx + width / 2.0 - radius, cy - height / 2.0 + radius, 270.0, 360.0),
        (cx + width / 2.0 - radius, cy + height / 2.0 - radius, 0.0, 90.0),
        (cx - width / 2.0 + radius, cy + height / 2.0 - radius, 90.0, 180.0),
    )
    for corner_x, corner_y, start, finish in corners:
        for step in range(5):
            angle = math.radians(start + (finish - start) * step / 4.0)
            result.append([round(corner_x + radius * math.cos(angle), 2), round(corner_y + radius * math.sin(angle), 2)])
    return result


def _layout(rng: random.Random) -> tuple[list[tuple[float, float, float, float]], list[tuple[float, float]]]:
    """Choose two separated loop boxes and several safe outside positions."""

    layout_id = rng.randrange(4)
    if layout_id == 0:
        boxes = [(140.0, 190.0, 154.0, 186.0), (372.0, 190.0, 154.0, 186.0)]
        outside = [(256.0, 48.0), (256.0, 336.0), (256.0, 190.0)]
    elif layout_id == 1:
        boxes = [(256.0, 105.0, 248.0, 108.0), (256.0, 279.0, 248.0, 108.0)]
        outside = [(62.0, 190.0), (450.0, 190.0), (256.0, 190.0)]
    elif layout_id == 2:
        boxes = [(137.0, 112.0, 156.0, 132.0), (375.0, 272.0, 156.0, 132.0)]
        outside = [(256.0, 190.0), (256.0, 42.0), (256.0, 342.0)]
    else:
        boxes = [(145.0, 110.0, 174.0, 132.0), (370.0, 274.0, 174.0, 132.0)]
        outside = [(270.0, 190.0), (480.0, 190.0), (40.0, 190.0)]

    adjusted = []
    for cx, cy, width, height in boxes:
        scale = rng.uniform(0.94, 1.045)
        adjusted.append(
            (
                cx + rng.uniform(-6.0, 6.0),
                cy + rng.uniform(-6.0, 6.0),
                width * scale,
                height * rng.uniform(0.95, 1.04),
            )
        )
    shifted_outside = [
        (x + rng.uniform(-7.0, 7.0), y + rng.uniform(-6.0, 6.0))
        for x, y in outside
    ]
    return adjusted, shifted_outside


def _inside_point(box: tuple[float, float, float, float], rng: random.Random) -> list[float]:
    cx, cy, width, height = box
    x = cx + rng.uniform(-0.22 * width, 0.22 * width)
    y = cy + rng.uniform(-0.22 * height, 0.22 * height)
    return [round(x, 2), round(y, 2)]


def _segment_distance(point: Sequence[float], first: Sequence[float], second: Sequence[float]) -> float:
    px, py = float(point[0]), float(point[1])
    ax, ay = float(first[0]), float(first[1])
    bx, by = float(second[0]), float(second[1])
    dx, dy = bx - ax, by - ay
    denominator = dx * dx + dy * dy
    if denominator <= 1e-12:
        return math.hypot(px - ax, py - ay)
    t = max(0.0, min(1.0, ((px - ax) * dx + (py - ay) * dy) / denominator))
    return math.hypot(px - (ax + t * dx), py - (ay + t * dy))


def _point_in_polygon(point: Sequence[float], polygon: Sequence[Sequence[float]]) -> bool:
    x, y = float(point[0]), float(point[1])
    inside = False
    previous = polygon[-1]
    for current in polygon:
        x0, y0 = float(previous[0]), float(previous[1])
        x1, y1 = float(current[0]), float(current[1])
        if (y0 > y) != (y1 > y):
            crossing_x = (x1 - x0) * (y - y0) / (y1 - y0) + x0
            if x < crossing_x:
                inside = not inside
        previous = current
    return inside


def _boundary_count(point: Sequence[float], contours: Sequence[Sequence[Sequence[float]]]) -> int:
    """Count contour boundaries incident to a face in the disjoint-loop scene."""

    enclosed = sum(_point_in_polygon(point, contour) for contour in contours)
    return len(contours) - enclosed


def _scene_slack(scene: dict) -> float:
    points = [marker["xy"] for marker in scene["markers"]]
    contours = scene["contours"]
    distances = []
    for point in points:
        distances.append(float(point[0]))
        distances.append(float(point[1]))
        distances.append(WIDTH - float(point[0]))
        distances.append(HEIGHT - float(point[1]))
        for contour in contours:
            for index, vertex in enumerate(contour):
                distances.append(_segment_distance(point, vertex, contour[(index + 1) % len(contour)]))
    return min(distances) - float(scene["marker_radius"]) - float(scene["wall_width"]) / 2.0


def _boxes_separated(contours: Sequence[Sequence[Sequence[float]]]) -> bool:
    boxes = []
    for contour in contours:
        xs = [float(point[0]) for point in contour]
        ys = [float(point[1]) for point in contour]
        boxes.append((min(xs), min(ys), max(xs), max(ys)))
    first, second = boxes
    horizontal_gap = max(first[0], second[0]) - min(first[2], second[2])
    vertical_gap = max(first[1], second[1]) - min(first[3], second[3])
    return horizontal_gap > 12.0 or vertical_gap > 12.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    boxes, outside_choices = _layout(rng)
    contours = []
    for box in boxes:
        cx, cy, width, height = box
        contours.append(_rounded_rect(cx, cy, width, height, min(width, height) * 0.16))

    outside = list(rng.choice(outside_choices))
    inside_first = _inside_point(boxes[0], rng)
    inside_second = _inside_point(boxes[1], rng)
    role_points = [outside, inside_first, inside_second]
    # Cycle the selected color across consecutive generated scenes while
    # shuffling the two nonselected labels.  This gives exact three-way
    # counterbalance for the public 24-scene batches without making geometry
    # depend on the color beyond this label assignment.
    color_labels = ["red", "green", "blue"]
    target_label = color_labels[int(seed) % len(color_labels)]
    remaining_labels = [label for label in color_labels if label != target_label]
    rng.shuffle(remaining_labels)
    labels = [target_label, remaining_labels[0], remaining_labels[1]]
    markers = [
        {"label": labels[index], "xy": [round(float(point[0]), 2), round(float(point[1]), 2)]}
        for index, point in enumerate(role_points)
    ]

    palette = {}
    for label, base in MARKER_BASE.items():
        palette[label] = [
            max(0, min(255, int(channel + rng.randint(-5, 5))))
            for channel in base
        ]
    return {
        "contours": contours,
        "markers": markers,
        "palette": palette,
        "wall_rgb": list(rng.choice(WALL_CHOICES)),
        "wall_width": rng.choice([9, 10, 11, 12]),
        "marker_radius": rng.choice([10, 11, 12, 13]),
        "background_rgb": list(BACKGROUND),
    }


def _scaled_points(contour: Sequence[Sequence[float]]) -> list[tuple[int, int]]:
    return [(round(float(point[0]) * SUPERSAMPLE), round(float(point[1]) * SUPERSAMPLE)) for point in contour]


def render(scene: dict) -> Image.Image:
    background = tuple(int(value) for value in scene["background_rgb"])
    image = Image.new("RGB", (WIDTH * SUPERSAMPLE, HEIGHT * SUPERSAMPLE), background)
    draw = ImageDraw.Draw(image)
    wall = tuple(int(value) for value in scene["wall_rgb"])
    width = max(1, int(round(float(scene["wall_width"]) * SUPERSAMPLE)))
    for contour in scene["contours"]:
        points = _scaled_points(contour)
        draw.line(points + [points[0]], fill=wall, width=width, joint="curve")

    palette = scene["palette"]
    radius = float(scene["marker_radius"]) * SUPERSAMPLE
    for marker in scene["markers"]:
        cx = float(marker["xy"][0]) * SUPERSAMPLE
        cy = float(marker["xy"][1]) * SUPERSAMPLE
        color = tuple(int(value) for value in palette[marker["label"]])
        draw.ellipse(
            (round(cx - radius), round(cy - radius), round(cx + radius), round(cy + radius)),
            fill=color,
        )
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    counts = [(_boundary_count(marker["xy"], scene["contours"]), marker["label"]) for marker in scene["markers"]]
    winners = [label for count, label in counts if count == 2]
    if len(winners) != 1:
        raise ValueError(f"scene has {len(winners)} two-boundary faces")
    return winners[0]


def margin(scene: dict) -> float:
    """Return the one-sided raster clearance, capped by the one-count runner-up gap."""

    return round(min(1.0, max(0.0, _scene_slack(scene) / 14.0)), 3)


def is_quarantined(scene: dict) -> bool:
    counts = [_boundary_count(marker["xy"], scene["contours"]) for marker in scene["markers"]]
    return (
        len(scene["contours"]) != 2
        or not _boxes_separated(scene["contours"])
        or sorted(counts).count(2) != 1
        or min(counts) < 1
        or _scene_slack(scene) < 10.0
    )


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["contours"] = list(reversed(twin["contours"]))
    return [("reverse_disjoint_contour_order", twin)]
