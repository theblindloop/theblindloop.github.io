"""Deterministic renderer and analytic arm for one compact crest trace."""

from __future__ import annotations

import math
import random
from typing import Any, Iterable

from PIL import Image, ImageDraw

RENDERER_VERSION = "crest_route_checkpoint-0.1.0"
WIDTH = 512
HEIGHT = 384
SCALE = 4
BACKGROUND = (255, 255, 255)
ROUTE = (31, 68, 103)
GREEN = (39, 177, 105)
RED = (220, 67, 63)
AMBER = (241, 157, 35)
VIOLET = (167, 78, 190)
BLACK = (24, 29, 38)
QUARANTINE_GAP = 18.0

_T_VALUES = (0.0, 0.12, 0.25, 0.38, 0.50, 0.62, 0.75, 0.88, 1.0)
_SHAPE = (0.0, 0.34, 0.68, 0.91, 1.0, 0.91, 0.68, 0.34, 0.0)


def _distance(a: Iterable[float], b: Iterable[float]) -> float:
    ax, ay = (float(value) for value in a)
    bx, by = (float(value) for value in b)
    return math.hypot(ax - bx, ay - by)


def _points(scene: dict[str, Any]) -> list[tuple[float, float]]:
    return [(float(point[0]), float(point[1])) for point in scene["route_points"]]


def _route_lengths(points: Iterable[tuple[float, float]]) -> tuple[list[float], float]:
    values = list(points)
    cumulative = [0.0]
    for first, second in zip(values, values[1:]):
        cumulative.append(cumulative[-1] + _distance(first, second))
    return cumulative, cumulative[-1]


def _point_at_fraction(points: list[tuple[float, float]], fraction: float) -> tuple[float, float]:
    cumulative, total = _route_lengths(points)
    target = float(fraction) * total
    for index in range(len(points) - 1):
        if cumulative[index + 1] >= target:
            span = cumulative[index + 1] - cumulative[index]
            alpha = 0.0 if span <= 0.0 else (target - cumulative[index]) / span
            x0, y0 = points[index]
            x1, y1 = points[index + 1]
            return x0 + alpha * (x1 - x0), y0 + alpha * (y1 - y0)
    return points[-1]


def _crest_index(points: list[tuple[float, float]]) -> int:
    interior = range(1, len(points) - 1)
    return min(interior, key=lambda index: (points[index][1], abs(index - (len(points) - 1) / 2.0)))


def sample_scene(seed: int) -> dict[str, Any]:
    """Sample a balanced, compact, visibly cresting trace."""

    rng = random.Random(int(seed))
    x0 = rng.uniform(178.0, 182.0)
    width = rng.uniform(118.0, 124.0)
    # Keep the whole route inside the same 48-pixel-high row of the 8-by-8
    # descriptor grid, while retaining a visibly separated interior crest.
    baseline = rng.uniform(180.0, 186.0)
    depth = rng.uniform(28.0, 34.0)
    tilt = rng.uniform(-5.5, 5.5)
    route_width = rng.uniform(6.2, 7.5)
    marker_radius = rng.uniform(8.0, 10.0)

    points: list[list[float]] = []
    for index, (t, shape) in enumerate(zip(_T_VALUES, _SHAPE, strict=True)):
        # The apex has no jitter and is separated from its neighbors by at
        # least four pixels, making the topmost event observable after rasterization.
        taper = 0.0 if index in (0, 4, 8) else math.sin(math.pi * t)
        jitter = rng.uniform(-0.9, 0.9) * taper
        x = x0 + width * t
        y = baseline + tilt * (t - 0.5) - depth * shape + jitter
        points.append([round(x, 4), round(y, 4)])

    if rng.randrange(2):
        points.reverse()
    route_points = [(float(point[0]), float(point[1])) for point in points]
    cumulative, total = _route_lengths(route_points)
    apex_fraction = cumulative[_crest_index(route_points)] / total

    near_label = "amber" if rng.randrange(2) == 0 else "violet"
    far_label = "violet" if near_label == "amber" else "amber"
    if rng.randrange(2) == 0:
        near_fraction = apex_fraction - rng.uniform(0.085, 0.12)
        far_fraction = apex_fraction + rng.uniform(0.28, 0.34)
    else:
        near_fraction = apex_fraction + rng.uniform(0.085, 0.12)
        far_fraction = apex_fraction - rng.uniform(0.28, 0.34)
    near_fraction = max(0.16, min(0.84, near_fraction))
    far_fraction = max(0.16, min(0.84, far_fraction))
    checkpoint_fractions = {
        near_label: round(float(near_fraction), 5),
        far_label: round(float(far_fraction), 5),
    }
    return {
        "route_points": [[round(x, 4), round(y, 4)] for x, y in route_points],
        "checkpoint_fractions": checkpoint_fractions,
        "route_width": round(route_width, 3),
        "marker_radius": round(marker_radius, 3),
    }


def _marker_color(label: str) -> tuple[int, int, int]:
    if label == "amber":
        return AMBER
    if label == "violet":
        return VIOLET
    raise ValueError(f"unknown checkpoint label {label!r}")


def _draw_route(draw: ImageDraw.ImageDraw, points: list[tuple[float, float]], width: int) -> None:
    scaled = [(round(x * SCALE), round(y * SCALE)) for x, y in points]
    radius = max(1, width // 2)
    draw.line(scaled, fill=ROUTE, width=width, joint="curve")
    for x, y in scaled:
        draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=ROUTE)


def _draw_port(draw: ImageDraw.ImageDraw, center: tuple[float, float], color: tuple[int, int, int], radius: float) -> None:
    x, y = (round(value * SCALE) for value in center)
    inner = round(radius * SCALE)
    outer = inner + 3 * SCALE
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=BLACK)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=color)


def _draw_checkpoint(
    draw: ImageDraw.ImageDraw,
    center: tuple[float, float],
    color: tuple[int, int, int],
    radius: float,
) -> None:
    x, y = (round(value * SCALE) for value in center)
    inner = round(radius * SCALE)
    outer = inner + 3 * SCALE
    draw.ellipse((x - outer, y - outer, x + outer, y + outer), fill=BLACK)
    draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=color)


def render(scene: dict[str, Any]) -> Image.Image:
    """Rasterize the visible trace without hidden payloads."""

    image = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), BACKGROUND)
    draw = ImageDraw.Draw(image)
    points = _points(scene)
    route_width = max(1, round(float(scene["route_width"]) * SCALE))
    marker_radius = float(scene["marker_radius"])
    _draw_route(draw, points, route_width)

    for label in ("amber", "violet"):
        center = _point_at_fraction(points, float(scene["checkpoint_fractions"][label]))
        _draw_checkpoint(draw, center, _marker_color(label), marker_radius)

    _draw_port(draw, points[0], GREEN, marker_radius)
    _draw_port(draw, points[-1], RED, marker_radius)

    # Repaint the route centerline through the checkpoint disks so the dark
    # trace remains one connected visible component and the marker positions
    # stay visibly attached to it.
    _draw_route(draw, points, route_width)
    return image.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict[str, Any]) -> str:
    points = _points(scene)
    cumulative, total = _route_lengths(points)
    crest_distance = cumulative[_crest_index(points)]
    amber = float(scene["checkpoint_fractions"]["amber"]) * total
    violet = float(scene["checkpoint_fractions"]["violet"]) * total
    amber_gap = abs(amber - crest_distance)
    violet_gap = abs(violet - crest_distance)
    if abs(amber_gap - violet_gap) < 1e-9:
        raise ValueError("crest checkpoint tie is not a valid scene")
    return "amber" if amber_gap <= violet_gap else "violet"


def margin(scene: dict[str, Any]) -> float:
    points = _points(scene)
    cumulative, total = _route_lengths(points)
    crest_distance = cumulative[_crest_index(points)]
    amber_gap = abs(float(scene["checkpoint_fractions"]["amber"]) * total - crest_distance)
    violet_gap = abs(float(scene["checkpoint_fractions"]["violet"]) * total - crest_distance)
    return abs(amber_gap - violet_gap)


def is_quarantined(scene: dict[str, Any]) -> bool:
    return margin(scene) < QUARANTINE_GAP


def latent_symmetries(scene: dict[str, Any]) -> list[tuple[str, dict[str, Any]]]:
    # Every label-bearing field is visible; this injective scene has no
    # untested render-invariant alias to exercise.
    return []
