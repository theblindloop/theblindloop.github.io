"""Independent image-only inverse arm.

It recovers the grid from the framed raster, segments the seven exact public
palette colors, validates connected districts, constructs side adjacency, and
tests the recovered graph for bipartiteness. It has no access to latent scenes.
"""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

PALETTE = (
    (30, 132, 150), (239, 151, 46), (222, 72, 93), (92, 105, 190),
    (74, 170, 102), (180, 91, 176), (235, 196, 55),
)
BACKGROUND = (250, 249, 245)
FRAME = (25, 29, 33)
SIZE = 512
GRID = 14
CELL = 32
INNER = 32
OUTER = 24


def _connected(cells: np.ndarray, label: int) -> bool:
    points = {tuple(p) for p in np.argwhere(cells == label)}
    if not points:
        return False
    seen = {next(iter(points))}
    queue = deque(seen)
    while queue:
        r, c = queue.popleft()
        for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nxt = (r + dr, c + dc)
            if nxt in points and nxt not in seen:
                seen.add(nxt)
                queue.append(nxt)
    return seen == points


def _recover(image: Image.Image) -> np.ndarray:
    rgb = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if rgb.shape != (SIZE, SIZE, 3):
        raise ValueError("unexpected image dimensions")
    bg = np.array(BACKGROUND, dtype=np.uint8)
    frame = np.array(FRAME, dtype=np.uint8)
    if not (np.all(rgb[:OUTER] == bg) and np.all(rgb[-OUTER:] == bg)
            and np.all(rgb[:, :OUTER] == bg) and np.all(rgb[:, -OUTER:] == bg)):
        raise ValueError("outer background is damaged")
    if not (np.all(rgb[OUTER:INNER, OUTER:SIZE-OUTER] == frame)
            and np.all(rgb[SIZE-INNER:SIZE-OUTER, OUTER:SIZE-OUTER] == frame)
            and np.all(rgb[OUTER:SIZE-OUTER, OUTER:INNER] == frame)
            and np.all(rgb[OUTER:SIZE-OUTER, SIZE-INNER:SIZE-OUTER] == frame)):
        raise ValueError("map frame is damaged")
    cells = np.empty((GRID, GRID), dtype=np.int8)
    palette = {color: index for index, color in enumerate(PALETTE)}
    for row in range(GRID):
        for col in range(GRID):
            y0, x0 = INNER + row * CELL, INNER + col * CELL
            block = rgb[y0:y0 + CELL, x0:x0 + CELL]
            color = tuple(int(v) for v in block[CELL // 2, CELL // 2])
            if color not in palette or not np.all(block == np.array(color, dtype=np.uint8)):
                raise ValueError("district cell is unknown or nonuniform")
            cells[row, col] = palette[color]
    if set(int(v) for v in np.unique(cells)) != set(range(7)):
        raise ValueError("expected exactly seven palette districts")
    if any(not _connected(cells, label) for label in range(7)):
        raise ValueError("a same-color district is disconnected")
    return cells


def decision_from_image(image: Image.Image) -> str:
    cells = _recover(image)
    graph = {label: set() for label in range(7)}
    for row in range(GRID):
        for col in range(GRID):
            here = int(cells[row, col])
            for dr, dc in ((1, 0), (0, 1)):
                rr, cc = row + dr, col + dc
                if rr < GRID and cc < GRID:
                    other = int(cells[rr, cc])
                    if other != here:
                        graph[here].add(other)
                        graph[other].add(here)
    colors: dict[int, int] = {}
    for start in graph:
        if start in colors:
            continue
        colors[start] = 0
        queue = deque([start])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in colors:
                    colors[neighbor] = 1 - colors[node]
                    queue.append(neighbor)
                elif colors[neighbor] == colors[node]:
                    return "yes"
    return "no"
