"""Analytic sampler and renderer for compact colored district maps."""

from __future__ import annotations

import random
from collections import deque

from PIL import Image, ImageDraw

SIZE = 512
GRID = 14
CELL = 32
INNER = 32
OUTER = 24
PALETTE = (
    (30, 132, 150),   # teal
    (239, 151, 46),   # orange
    (222, 72, 93),    # coral
    (92, 105, 190),   # indigo
    (74, 170, 102),   # green
    (180, 91, 176),   # violet
    (235, 196, 55),   # yellow
)


def _base_cells(has_odd_cycle: bool) -> list[list[int]]:
    if not has_odd_cycle:
        return [[col // 2 for col in range(GRID)] for _ in range(GRID)]
    cuts = (0, 2, 4, 6, 8, 11, 14)
    cells = []
    for row in range(GRID):
        band = next(i for i in range(6) if cuts[i] <= row < cuts[i + 1])
        cells.append([0 if col < 2 else band + 1 for col in range(GRID)])
    return cells


def _transform(cells: list[list[int]], code: int) -> list[list[int]]:
    out = [row[:] for row in cells]
    if code & 1:
        out = [list(row) for row in zip(*out[::-1])]
    if code & 2:
        out = [row[::-1] for row in out]
    if code & 4:
        out = out[::-1]
    return out


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    cells = _transform(_base_cells(bool(seed & 1)), rng.randrange(8))
    permutation = list(range(7))
    rng.shuffle(permutation)
    cells = [[permutation[value] for value in row] for row in cells]
    return {"cells": cells}


def render(scene: dict) -> Image.Image:
    cells = scene["cells"]
    if len(cells) != GRID or any(len(row) != GRID for row in cells):
        raise ValueError("cells must be a 14 by 14 array")
    image = Image.new("RGB", (SIZE, SIZE), (250, 249, 245))
    draw = ImageDraw.Draw(image)
    draw.rectangle((OUTER, OUTER, SIZE - OUTER - 1, SIZE - OUTER - 1), fill=(25, 29, 33))
    for row in range(GRID):
        for col in range(GRID):
            value = int(cells[row][col])
            if not 0 <= value < len(PALETTE):
                raise ValueError("invalid district palette index")
            x0 = INNER + col * CELL
            y0 = INNER + row * CELL
            draw.rectangle((x0, y0, x0 + CELL - 1, y0 + CELL - 1), fill=PALETTE[value])
    return image


def _graph(scene: dict) -> dict[int, set[int]]:
    cells = scene["cells"]
    labels = {int(value) for row in cells for value in row}
    graph = {value: set() for value in labels}
    for row in range(GRID):
        for col in range(GRID):
            here = int(cells[row][col])
            for dr, dc in ((1, 0), (0, 1)):
                rr, cc = row + dr, col + dc
                if rr < GRID and cc < GRID:
                    other = int(cells[rr][cc])
                    if other != here:
                        graph[here].add(other)
                        graph[other].add(here)
    return graph


def _is_bipartite(graph: dict[int, set[int]]) -> bool:
    colors: dict[int, int] = {}
    for start in graph:
        if start in colors:
            continue
        colors[start] = 0
        queue = deque([start])
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if neighbor not in colors:
                    colors[neighbor] = 1 - colors[node]
                    queue.append(neighbor)
                elif colors[neighbor] == colors[node]:
                    return False
    return True


def analytic_gold(scene: dict) -> str:
    return "no" if _is_bipartite(_graph(scene)) else "yes"


def margin(scene: dict) -> float:
    # A graph edge is supported by at least one full 32-pixel cell side.
    _ = scene["cells"]
    return float(CELL)


def is_quarantined(scene: dict) -> bool:
    cells = scene["cells"]
    if len(cells) != GRID or any(len(row) != GRID for row in cells):
        return True
    labels = {int(value) for row in cells for value in row}
    if labels != set(range(7)):
        return True
    for label in labels:
        positions = {(r, c) for r in range(GRID) for c in range(GRID) if int(cells[r][c]) == label}
        seen = {next(iter(positions))}
        queue = deque(seen)
        while queue:
            r, c = queue.popleft()
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nxt = (r + dr, c + dc)
                if nxt in positions and nxt not in seen:
                    seen.add(nxt)
                    queue.append(nxt)
        if seen != positions:
            return True
    return False


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    _ = scene["cells"]
    return []
