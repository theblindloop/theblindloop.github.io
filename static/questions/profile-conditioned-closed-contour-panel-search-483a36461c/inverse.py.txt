"""Independent image-only inverse for open-versus-closed contours."""

from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image

CANVAS_SIZE = (1040, 760)
LABELS = ("A", "B", "C", "D")
PANEL_BOXES = (
    (24, 24, 504, 354),
    (536, 24, 1016, 354),
    (24, 406, 504, 736),
    (536, 406, 1016, 736),
)


class OracleAmbiguous(ValueError):
    """Raised when the PNG does not expose one closure winner."""


def _candidate_mask(array: np.ndarray, box: tuple[int, int, int, int]) -> np.ndarray:
    x0, y0, x1, y1 = box
    # The crop excludes the label and border but includes all translated masks.
    crop = array[y0 + 62 : y1 - 42, x0 + 70 : x1 - 40]
    dark = np.max(crop, axis=2) < 100
    ys, xs = np.nonzero(dark)
    if len(ys) == 0:
        raise OracleAmbiguous("panel contains no dark contour")
    return dark[int(ys.min()) : int(ys.max()) + 1, int(xs.min()) : int(xs.max()) + 1]


def _component_count(mask: np.ndarray) -> int:
    height, width = mask.shape
    visited = np.zeros_like(mask, dtype=bool)
    count = 0
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or visited[y, x]:
                continue
            count += 1
            visited[y, x] = True
            queue: deque[tuple[int, int]] = deque([(x, y)])
            while queue:
                cx, cy = queue.popleft()
                for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                    nx, ny = cx + dx, cy + dy
                    if 0 <= nx < width and 0 <= ny < height and mask[ny, nx] and not visited[ny, nx]:
                        visited[ny, nx] = True
                        queue.append((nx, ny))
    return count


def _closure_score(mask: np.ndarray) -> int:
    if _component_count(mask) != 1:
        raise OracleAmbiguous("candidate contour is missing or fragmented")
    height, width = mask.shape
    if min(height, width) < 70:
        raise OracleAmbiguous("candidate contour is too small")
    band = 12
    coverage = (
        float(np.any(mask[:band, :], axis=0).mean()),
        float(np.any(mask[:, -band:], axis=1).mean()),
        float(np.any(mask[-band:, :], axis=0).mean()),
        float(np.any(mask[:, :band], axis=1).mean()),
    )
    return int(min(coverage) >= 0.90)


def decision_from_image(image: Image.Image) -> str:
    """Enumerate the four panels from final pixels and select the closed loop."""

    if image.size != CANVAS_SIZE:
        raise OracleAmbiguous(f"expected {CANVAS_SIZE}, received {image.size}")
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    scores = [_closure_score(_candidate_mask(array, box)) for box in PANEL_BOXES]
    if scores.count(1) != 1:
        raise OracleAmbiguous(f"closure scores are ambiguous: {scores}")
    return LABELS[scores.index(1)]
