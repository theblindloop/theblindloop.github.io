"""Independent pixel-only inverse for the mirror-fold intersection world."""

from __future__ import annotations

import numpy as np
from PIL import Image

GRID_N = 15
QUARANTINE_GAP = 4.0


def _templates() -> dict[str, np.ndarray]:
    result: dict[str, np.ndarray] = {}
    for name in ("circle", "diamond", "plus", "X"):
        mask = np.zeros((15, 15), dtype=bool)
        for row in range(15):
            for col in range(15):
                x, y = col - 7, row - 7
                radius = (x * x + y * y) ** 0.5
                if name == "circle":
                    keep = 4.0 <= radius <= 6.0 or (row in (7, 8) and col in (7, 8))
                elif name == "diamond":
                    keep = 5 <= abs(x) + abs(y) <= 7
                elif name == "plus":
                    keep = ((abs(x) <= 1 and abs(y) <= 6) or (abs(y) <= 1 and abs(x) <= 6) or (row, col) in ((5, 5), (5, 9), (9, 9)))
                else:
                    keep = ((abs(abs(x) - abs(y)) <= 1 and abs(x) <= 6 and abs(y) <= 6) or (row, col) in ((7, 5), (5, 7), (7, 9)))
                mask[row, col] = keep
        result[name] = mask
    return result


def _groups(values: np.ndarray) -> list[list[int]]:
    groups: list[list[int]] = []
    for value in map(int, values):
        if not groups or value > groups[-1][-1] + 1:
            groups.append([value])
        else:
            groups[-1].append(value)
    return groups


def _recover(image: Image.Image) -> tuple[np.ndarray, float]:
    rgb = np.asarray(image.convert("RGB"), dtype=np.int16)
    cyan = (rgb[:, :, 0] < 25) & (rgb[:, :, 1] > 135) & (rgb[:, :, 2] > 150)
    column_counts = cyan.sum(axis=0)
    row_counts = cyan.sum(axis=1)
    x_groups = _groups(np.flatnonzero(column_counts > 150))
    y_groups = _groups(np.flatnonzero(row_counts > 300))
    if len(x_groups) != 4 or len(y_groups) != 2:
        raise ValueError("two cyan grid frames cannot be located")
    edges_x = [float(np.mean(group)) for group in x_groups]
    edges_y = [float(np.mean(group)) for group in y_groups]
    outer_a = edges_x[1] - edges_x[0]
    outer_b = edges_x[3] - edges_x[2]
    outer_h = edges_y[1] - edges_y[0]
    if not (243 <= outer_a <= 245 and 243 <= outer_b <= 245 and 243 <= outer_h <= 245):
        raise ValueError("cyan frame geometry is ambiguous")
    panel_a, panel_b, height = outer_a - 4.0, outer_b - 4.0, outer_h - 4.0
    top = edges_y[0] + 2.0
    left_a = edges_x[0] + 2.0
    left_b = edges_x[2] + 2.0
    cell_a = panel_a / GRID_N
    cell_b = panel_b / GRID_N
    cell_y = height / GRID_N
    layers: list[np.ndarray] = []
    for panel_left, cell_x in ((left_a, cell_a), (left_b, cell_b)):
        mask = np.zeros((GRID_N, GRID_N), dtype=bool)
        for row in range(GRID_N):
            for col in range(GRID_N):
                x = int(round(panel_left + (col + 0.5) * cell_x))
                y = int(round(top + (row + 0.5) * cell_y))
                patch = rgb[y - 3 : y + 4, x - 3 : x + 4]
                lum = 0.299 * patch[:, :, 0] + 0.587 * patch[:, :, 1] + 0.114 * patch[:, :, 2]
                mask[row, col] = float(np.median(lum)) < 135.0
        layers.append(mask)
    folded = layers[0] & layers[1][:, ::-1]
    distances = {name: int(np.count_nonzero(folded != ref)) for name, ref in _templates().items()}
    ranked = sorted(distances.items(), key=lambda item: (item[1], item[0]))
    gap = float(ranked[1][1] - ranked[0][1])
    if gap <= QUARANTINE_GAP:
        raise ValueError(f"ambiguous folded intersection: runner-up gap {gap:g}")
    return folded, gap


def decision_with_margin(image: Image.Image) -> tuple[str, float]:
    folded, gap = _recover(image)
    distances = {name: int(np.count_nonzero(folded != ref)) for name, ref in _templates().items()}
    winner = min(distances, key=lambda name: (distances[name], name))
    return winner, gap


def decision_from_image(image: Image.Image) -> str:
    return decision_with_margin(image)[0]
