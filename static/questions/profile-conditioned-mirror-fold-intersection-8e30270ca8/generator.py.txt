"""Renderer and analytic arm for the mirror-fold intersection world."""

from __future__ import annotations

import random

import numpy as np
from PIL import Image, ImageDraw

CANVAS = (640, 360)
GRID_N = 15
CELL = 16
PANEL = GRID_N * CELL
GAP = 18
CYAN = (0, 169, 188)
MAGENTA = (218, 28, 145)
GRID = (211, 216, 221)
BACKGROUND = (250, 249, 246)
INKS = ((25, 28, 34), (28, 47, 82), (65, 37, 77), (31, 70, 62))
NOISE_PER_LAYER = 62
QUARANTINE_GAP = 4.0
SYMBOLS = ("circle", "diamond", "plus", "X")


def _templates() -> dict[str, np.ndarray]:
    yy, xx = np.mgrid[-7:8, -7:8]
    circle = np.abs(np.sqrt(xx * xx + yy * yy) - 5.0) <= 1.0
    circle[7:9, 7:9] = True
    diamond = np.abs(np.abs(xx) + np.abs(yy) - 6) <= 1
    plus = ((np.abs(xx) <= 1) & (np.abs(yy) <= 6)) | ((np.abs(yy) <= 1) & (np.abs(xx) <= 6))
    plus[5, 5] = plus[5, 9] = plus[9, 9] = True
    x_mask = (np.abs(np.abs(xx) - np.abs(yy)) <= 1) & (np.abs(xx) <= 6) & (np.abs(yy) <= 6)
    x_mask[7, 5] = x_mask[5, 7] = x_mask[7, 9] = True
    result = {"circle": circle, "diamond": diamond, "plus": plus, "X": x_mask}
    assert {int(mask.sum()) for mask in result.values()} == {72}
    return result


def _target(scene: dict) -> np.ndarray:
    mask = _templates()[str(scene["symbol"])].copy()
    for row, col in scene["flip_off"]:
        mask[int(row), int(col)] = False
    for row, col in scene["flip_on"]:
        mask[int(row), int(col)] = True
    return mask


def _layers(scene: dict) -> tuple[np.ndarray, np.ndarray]:
    target = _target(scene)
    left = target.copy()
    folded_right = target.copy()
    for row, col in scene["left_noise"]:
        left[int(row), int(col)] = True
    for row, col in scene["right_noise"]:
        folded_right[int(row), int(col)] = True
    if bool(scene["swap_panels"]):
        left, folded_right = folded_right, left
    visible_right = folded_right[:, ::-1]
    return left, visible_right


def _distances(mask: np.ndarray) -> dict[str, int]:
    return {name: int(np.count_nonzero(mask != template)) for name, template in _templates().items()}


def sample_scene(seed: int) -> dict:
    seed = int(seed)
    rng = random.Random(seed ^ 0x4D17F01D)
    symbol = SYMBOLS[seed % len(SYMBOLS)]
    flip_count = (seed // 4) % 4
    base = _templates()[symbol]
    on = [tuple(map(int, p)) for p in np.argwhere(base)]
    off = [tuple(map(int, p)) for p in np.argwhere(~base)]
    rng.shuffle(on)
    rng.shuffle(off)
    flip_off = on[:flip_count]
    flip_on = off[:flip_count]
    target = base.copy()
    for row, col in flip_off:
        target[row, col] = False
    for row, col in flip_on:
        target[row, col] = True
    free = [tuple(map(int, p)) for p in np.argwhere(~target)]
    rng.shuffle(free)
    scene = {
        "symbol": symbol,
        "offset_x": 62 + rng.randrange(-8, 9),
        "offset_y": 63 + rng.randrange(-8, 9),
        "ink_index": (seed // 4) % len(INKS),
        "swap_panels": bool((seed // 2) % 2),
        "left_noise": [list(p) for p in free[:NOISE_PER_LAYER]],
        "right_noise": [list(p) for p in free[NOISE_PER_LAYER : 2 * NOISE_PER_LAYER]],
        "flip_off": [list(p) for p in flip_off],
        "flip_on": [list(p) for p in flip_on],
    }
    assert analytic_gold(scene) == symbol
    assert margin(scene) > QUARANTINE_GAP
    return scene


def _draw_panel(draw: ImageDraw.ImageDraw, mask: np.ndarray, left: int, top: int, ink: tuple[int, int, int]) -> None:
    for row in range(GRID_N):
        for col in range(GRID_N):
            if bool(mask[row, col]):
                x0 = left + col * CELL + 2
                y0 = top + row * CELL + 2
                draw.rectangle((x0, y0, x0 + CELL - 4, y0 + CELL - 4), fill=ink)
    for i in range(GRID_N + 1):
        x = left + i * CELL
        y = top + i * CELL
        draw.line((x, top, x, top + PANEL), fill=GRID, width=1)
        draw.line((left, y, left + PANEL, y), fill=GRID, width=1)
    draw.rectangle((left - 3, top - 3, left + PANEL + 3, top + PANEL + 3), outline=CYAN, width=3)


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", CANVAS, BACKGROUND)
    draw = ImageDraw.Draw(image)
    left = int(scene["offset_x"])
    top = int(scene["offset_y"])
    right = left + PANEL + GAP
    ink = INKS[int(scene["ink_index"])]
    left_mask, right_mask = _layers(scene)
    _draw_panel(draw, left_mask, left, top, ink)
    _draw_panel(draw, right_mask, right, top, ink)
    hinge_x = left + PANEL + GAP // 2
    for y in range(top - 10, top + PANEL + 11, 12):
        draw.line((hinge_x, y, hinge_x, min(y + 6, top + PANEL + 10)), fill=MAGENTA, width=3)
    draw.polygon(((right + PANEL - 8, top - 21), (right + PANEL - 8, top - 9), (right + PANEL - 23, top - 15)), fill=MAGENTA)
    draw.line((left + PANEL + 22, top - 15, right + PANEL - 11, top - 15), fill=MAGENTA, width=3)
    return image


def analytic_gold(scene: dict) -> str:
    return str(scene["symbol"])


def margin(scene: dict) -> float:
    distances = sorted(_distances(_target(scene)).values())
    return float(distances[1] - distances[0])


def is_quarantined(scene: dict) -> bool:
    return margin(scene) <= QUARANTINE_GAP


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
