"""Renderer and analytic arm for the circular-gauge alignment world."""

from __future__ import annotations

import math
import random
from copy import deepcopy

from PIL import Image, ImageDraw

WIDTH, HEIGHT = 720, 300
SCALE = 4
RADIUS = 72
TRACK_WIDTH = 17
COLORS = ("blue", "orange", "green")
RGB = {
    "blue": (31, 119, 180),
    "orange": (255, 127, 14),
    "green": (44, 160, 44),
}
MAGENTA = (202, 35, 150)
TRACK = (221, 225, 230)
INK = (30, 34, 40)


def _angular_gap(a: float, b: float) -> float:
    delta = abs(float(a) - float(b)) % 360.0
    return min(delta, 360.0 - delta)


def _gaps(fill_angles: list[float], target_angles: list[float]) -> list[float]:
    return [_angular_gap(a, b) for a, b in zip(fill_angles, target_angles)]


def sample_scene(seed: int) -> dict:
    """Sample a balanced, comfortably separated three-gauge scene."""
    rng = random.Random(int(seed))
    winner = int(seed) % 3
    other = [i for i in range(3) if i != winner]
    rng.shuffle(other)
    desired_gaps = [0.0, 0.0, 0.0]
    desired_gaps[winner] = rng.uniform(7.0, 14.0)
    desired_gaps[other[0]] = rng.uniform(40.0, 62.0)
    desired_gaps[other[1]] = rng.uniform(82.0, 118.0)

    fill_angles: list[float] = []
    target_angles: list[float] = []
    for gap in desired_gaps:
        # Keep both endpoints away from the 12 o'clock seam. This makes the
        # visible smaller-angle rule stable under rasterization.
        for _ in range(100):
            fill = rng.uniform(62.0, 298.0)
            sign = rng.choice((-1.0, 1.0))
            target = (fill + sign * gap) % 360.0
            if 24.0 <= target <= 336.0:
                break
        else:  # pragma: no cover - defensive only
            raise RuntimeError("could not sample a seam-safe target")
        fill_angles.append(round(fill, 4))
        target_angles.append(round(target, 4))

    slots = [140, 360, 580]
    rng.shuffle(slots)
    centers = [
        [slots[i] + rng.randint(-6, 6), 151 + rng.randint(-7, 7)]
        for i in range(3)
    ]
    return {
        "fill_angles": fill_angles,
        "target_angles": target_angles,
        "centers": centers,
        "serialization_variant": rng.randrange(2),
    }


def _point(cx: int, cy: int, radius: float, clockwise_degrees: float) -> tuple[float, float]:
    radians = math.radians(clockwise_degrees - 90.0)
    return cx + radius * math.cos(radians), cy + radius * math.sin(radians)


def render(scene: dict) -> Image.Image:
    canvas = Image.new("RGB", (WIDTH * SCALE, HEIGHT * SCALE), "white")
    draw = ImageDraw.Draw(canvas)
    centers = scene["centers"]
    fills = scene["fill_angles"]
    targets = scene["target_angles"]
    for index, name in enumerate(COLORS):
        cx, cy = (int(centers[index][0]), int(centers[index][1]))
        box = tuple(
            int(v * SCALE)
            for v in (cx - RADIUS, cy - RADIUS, cx + RADIUS, cy + RADIUS)
        )
        draw.ellipse(box, outline=TRACK, width=TRACK_WIDTH * SCALE)
        draw.arc(
            box,
            start=-90.0,
            end=-90.0 + float(fills[index]),
            fill=RGB[name],
            width=TRACK_WIDTH * SCALE,
        )
        # A visible center pin makes the polar center recoverable without any
        # renderer coordinates.
        pin = 7 * SCALE
        draw.ellipse(
            (cx * SCALE - pin, cy * SCALE - pin, cx * SCALE + pin, cy * SCALE + pin),
            fill=INK,
        )
        # Target ticks sit just outside the track so that near-alignment does
        # not erase the colored endpoint evidence.
        inner = _point(cx, cy, RADIUS + 6, float(targets[index]))
        outer = _point(cx, cy, RADIUS + 25, float(targets[index]))
        draw.line(
            tuple(int(v * SCALE) for point in (inner, outer) for v in point),
            fill=MAGENTA,
            width=7 * SCALE,
        )
    return canvas.resize((WIDTH, HEIGHT), Image.Resampling.LANCZOS)


def analytic_gold(scene: dict) -> str:
    gaps = _gaps(scene["fill_angles"], scene["target_angles"])
    return COLORS[min(range(3), key=gaps.__getitem__)]


def margin(scene: dict) -> float:
    ordered = sorted(_gaps(scene["fill_angles"], scene["target_angles"]))
    return (ordered[1] - ordered[0]) * math.pi * RADIUS / 180.0


def is_quarantined(scene: dict) -> bool:
    seam_clearance = min(
        min(float(angle) % 360.0, 360.0 - (float(angle) % 360.0))
        for angle in scene["target_angles"]
    )
    return margin(scene) < 18.0 or seam_clearance < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = deepcopy(scene)
    twin["serialization_variant"] = 1 - int(scene["serialization_variant"])
    return [("serialization_variant_flip", twin)]
