"""Renderer and analytic arm for a distributed principal-axis task."""
from __future__ import annotations

import math
import random
from copy import deepcopy

import numpy as np
from PIL import Image, ImageDraw

PALETTE = {"orange": (244, 126, 24), "blue": (38, 104, 218)}


def _axis_angle(points: list[list[float]]) -> tuple[float, float]:
    a = np.asarray(points, dtype=float)
    centered = a - a.mean(axis=0)
    cov = centered.T @ centered / len(a)
    vals, vecs = np.linalg.eigh(cov)
    v = vecs[:, int(np.argmax(vals))]
    return math.atan2(float(v[1]), float(v[0])), float(vals.max() / max(vals.min(), 1e-9))


def _angle_error(a: float, b: float) -> float:
    d = abs((a - b) % math.pi)
    return min(d, math.pi - d)


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    width = height = 512
    cx, cy = rng.uniform(238, 274), rng.uniform(238, 274)
    theta = rng.uniform(0, math.pi)
    ct, st = math.cos(theta), math.sin(theta)
    points: list[list[float]] = []
    # A jittered anisotropic field: no individual row or pair determines its axis.
    for _ in range(24):
        for _attempt in range(500):
            u = rng.uniform(-158, 158)
            v = rng.uniform(-42, 42)
            x = cx + u * ct - v * st
            y = cy + u * st + v * ct
            if not (38 <= x <= 474 and 38 <= y <= 474):
                continue
            if all((x-px)**2 + (y-py)**2 >= 18**2 for px, py in points):
                points.append([round(x, 3), round(y, 3)])
                break
        else:
            raise RuntimeError("could not sample separated dots")

    principal, _ = _axis_angle(points)
    near = (10.5 if seed % 7 == 0 else rng.uniform(2.5, 7.0)) * math.pi / 180
    far = (23.5 if seed % 7 == 0 else rng.uniform(29.0, 42.0)) * math.pi / 180
    signs = [rng.choice((-1, 1)), rng.choice((-1, 1))]
    correct_slot = int(seed) & 1
    errors = [far, far]
    errors[correct_slot] = near
    angles = [float((principal + signs[i] * errors[i]) % math.pi) for i in range(2)]
    colors = ["orange", "blue"]
    return {
        "width": width,
        "height": height,
        "points": points,
        "axis_angles": angles,
        "axis_colors": colors,
        "dot_radius": 6,
        "line_width": 4,
        "dash_length": 13,
        "gap_length": 10,
    }


def _draw_dashed(draw: ImageDraw.ImageDraw, center: tuple[float, float], angle: float,
                 color: tuple[int, int, int], width: int, dash: int, gap: int) -> None:
    ux, uy = math.cos(angle), math.sin(angle)
    half = 280
    step = dash + gap
    for start in range(-half, half, step):
        end = min(start + dash, half)
        p0 = (center[0] + start*ux, center[1] + start*uy)
        p1 = (center[0] + end*ux, center[1] + end*uy)
        draw.line([p0, p1], fill=color, width=width)


def render(scene: dict) -> Image.Image:
    img = Image.new("RGB", (int(scene["width"]), int(scene["height"])), "white")
    draw = ImageDraw.Draw(img)
    pts = np.asarray(scene["points"], dtype=float)
    center = (float(pts[:, 0].mean()), float(pts[:, 1].mean()))
    for angle, color in zip(scene["axis_angles"], scene["axis_colors"]):
        _draw_dashed(draw, center, float(angle), PALETTE[color], int(scene["line_width"]),
                     int(scene["dash_length"]), int(scene["gap_length"]))
    r = int(scene["dot_radius"])
    for x, y in scene["points"]:
        draw.ellipse([x-r, y-r, x+r, y+r], fill=(18, 20, 24))
    return img


def _decision_metrics(scene: dict) -> tuple[list[float], float]:
    principal, ratio = _axis_angle(scene["points"])
    errors = [_angle_error(principal, float(a)) for a in scene["axis_angles"]]
    return errors, ratio


def analytic_gold(scene: dict) -> str:
    errors, _ = _decision_metrics(scene)
    return str(scene["axis_colors"][int(np.argmin(errors))])


def margin(scene: dict) -> float:
    errors, _ = _decision_metrics(scene)
    e = sorted(errors)
    return float(math.degrees(e[1] - e[0]))


def is_quarantined(scene: dict) -> bool:
    errors, ratio = _decision_metrics(scene)
    return ratio < 2.2 or math.degrees(abs(errors[0] - errors[1])) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
