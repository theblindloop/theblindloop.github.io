"""Independent pixel-only inverse arm; this module receives only a PNG."""
from __future__ import annotations

import math
from collections import deque

import numpy as np
from PIL import Image

COLORS = {"orange": np.array([244, 126, 24]), "blue": np.array([38, 104, 218])}


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for y, x in np.argwhere(mask):
        if seen[y, x]:
            continue
        q = deque([(int(y), int(x))]); seen[y, x] = True; comp = []
        while q:
            yy, xx = q.popleft(); comp.append((yy, xx))
            for dy, dx in ((1,0),(-1,0),(0,1),(0,-1)):
                ny, nx = yy+dy, xx+dx
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; q.append((ny, nx))
        out.append(comp)
    return out


def _orientation(xy: np.ndarray) -> float:
    centered = xy - xy.mean(axis=0)
    cov = centered.T @ centered / len(xy)
    vals, vecs = np.linalg.eigh(cov)
    if vals.max() < 4 or vals.max() / max(vals.min(), 1e-9) < 1.2:
        raise ValueError("orientation is not recoverable")
    v = vecs[:, int(np.argmax(vals))]
    return math.atan2(float(v[1]), float(v[0]))


def _error(a: float, b: float) -> float:
    d = abs((a-b) % math.pi)
    return min(d, math.pi-d)


def decision_from_image(image: Image.Image) -> str:
    arr = np.asarray(image.convert("RGB"), dtype=np.int16)
    black = (arr[:,:,0] < 60) & (arr[:,:,1] < 60) & (arr[:,:,2] < 60)
    centers = []
    for comp in _components(black):
        if len(comp) >= 50:
            p = np.asarray(comp, dtype=float)
            centers.append([float(p[:,1].mean()), float(p[:,0].mean())])
    if len(centers) < 8:
        raise ValueError("too few dot components")
    principal = _orientation(np.asarray(centers))
    errors = {}
    for name, rgb in COLORS.items():
        dist = np.max(np.abs(arr - rgb), axis=2)
        yy, xx = np.nonzero(dist <= 2)
        if len(xx) < 80:
            raise ValueError(f"missing {name} axis")
        errors[name] = _error(principal, _orientation(np.column_stack([xx, yy])))
    ordered = sorted(errors, key=errors.get)
    if math.degrees(errors[ordered[1]] - errors[ordered[0]]) < 7:
        raise ValueError("ambiguous pixel margin")
    return ordered[0]
