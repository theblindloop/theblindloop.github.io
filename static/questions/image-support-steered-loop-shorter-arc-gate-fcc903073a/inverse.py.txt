from __future__ import annotations

from collections import deque

import numpy as np
from PIL import Image


_PALETTES = (
    {"background": "#fbfaf5", "track": "#173754", "start": "#18a45f", "finish": "#d83c4e", "cyan": "#16b7c9", "orange": "#ed8628"},
    {"background": "#f8fbf7", "track": "#253650", "start": "#259c58", "finish": "#d94855", "cyan": "#24adbf", "orange": "#e9792d"},
    {"background": "#fff9f1", "track": "#123b5b", "start": "#10a06a", "finish": "#d53f50", "cyan": "#0eb4c7", "orange": "#ef9022"},
)


def _rgb(value: str) -> tuple[int, int, int]:
    value = value.lstrip("#")
    return tuple(int(value[index:index + 2], 16) for index in (0, 2, 4))


def _masks(image: Image.Image) -> dict[str, np.ndarray]:
    array = np.asarray(image.convert("RGB"), dtype=np.float32)
    if array.ndim != 3 or array.shape[2] != 3 or array.shape[:2] != (512, 512):
        raise ValueError("unsupported raster")
    corners = np.stack((array[0, 0], array[0, -1], array[-1, 0], array[-1, -1]))
    corner = np.median(corners, axis=0)
    palette_index = int(np.argmin([
        float(((corner - np.asarray(_rgb(palette["background"]), dtype=np.float32)) ** 2).sum())
        for palette in _PALETTES
    ]))
    palette = _PALETTES[palette_index]
    result = {}
    for name in ("background", "track", "start", "finish", "cyan", "orange"):
        color = np.asarray(_rgb(palette[name]), dtype=np.float32)
        result[name] = ((array - color) ** 2).sum(axis=2) <= 38.0 ** 2
    return result


def _adjacent(mask: np.ndarray, domain: np.ndarray) -> np.ndarray:
    result = np.zeros_like(mask)
    result[1:] |= mask[:-1]
    result[:-1] |= mask[1:]
    result[:, 1:] |= mask[:, :-1]
    result[:, :-1] |= mask[:, 1:]
    return result & domain


def _distances(domain: np.ndarray, seeds: np.ndarray) -> np.ndarray:
    if int(seeds.sum()) < 8:
        raise ValueError("marker does not visibly touch the track")
    distances = np.full(domain.shape, -1, dtype=np.int32)
    queue: deque[tuple[int, int]] = deque()
    for y, x in np.argwhere(seeds):
        yi, xi = int(y), int(x)
        distances[yi, xi] = 0
        queue.append((yi, xi))
    height, width = domain.shape
    while queue:
        y, x = queue.popleft()
        next_distance = int(distances[y, x]) + 1
        for ny, nx in ((y - 1, x), (y + 1, x), (y, x - 1), (y, x + 1)):
            if 0 <= ny < height and 0 <= nx < width and domain[ny, nx] and distances[ny, nx] < 0:
                distances[ny, nx] = next_distance
                queue.append((ny, nx))
    return distances


def _distance_to(distance: np.ndarray, target: np.ndarray, domain: np.ndarray) -> float:
    boundary = _adjacent(target, domain)
    values = distance[boundary]
    values = values[values >= 0]
    if len(values) < 8:
        raise ValueError("track route to marker is broken")
    return float(values.min())


def decision_from_image(image: Image.Image) -> str:
    masks = _masks(image)
    track_area = int(masks["track"].sum())
    # Clean randomized renders occupy 37,945--38,087 solid track pixels over the
    # calibration sweep.  The tight floor makes loss of any grid-scale piece of
    # this single regional body explicit instead of silently tracing a remnant.
    if not 37800 <= track_area <= 38300:
        raise ValueError(f"closed track area is invalid: {track_area}")
    for name in ("start", "finish", "cyan", "orange"):
        area = int(masks[name].sum())
        if not 360 <= area <= 720:
            raise ValueError(f"{name} marker area is invalid: {area}")

    start_seeds = _adjacent(masks["start"], masks["track"])
    start_distance = _distances(masks["track"], start_seeds)
    if int((start_distance >= 0).sum()) < track_area - 40:
        raise ValueError("track contains a detached or erased arc")
    finish_distance = _distances(masks["track"], _adjacent(masks["finish"], masks["track"]))
    scores = {}
    for gate in ("cyan", "orange"):
        scores[gate] = _distance_to(start_distance, masks[gate], masks["track"]) + _distance_to(finish_distance, masks[gate], masks["track"])
    gap = abs(scores["cyan"] - scores["orange"])
    if gap < 44.0:
        raise ValueError(f"ambiguous pixel geodesic gap: {gap:.1f}")
    return min(scores, key=scores.get)
