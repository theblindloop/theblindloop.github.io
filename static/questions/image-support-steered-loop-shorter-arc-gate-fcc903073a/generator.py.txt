from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw


PALETTES = (
    {"background": "#fbfaf5", "track": "#173754", "start": "#18a45f", "finish": "#d83c4e", "cyan": "#16b7c9", "orange": "#ed8628"},
    {"background": "#f8fbf7", "track": "#253650", "start": "#259c58", "finish": "#d94855", "cyan": "#24adbf", "orange": "#e9792d"},
    {"background": "#fff9f1", "track": "#123b5b", "start": "#10a06a", "finish": "#d53f50", "cyan": "#0eb4c7", "orange": "#ef9022"},
)

QUARANTINE_MARGIN = 52.0


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    top_is_long = bool(rng.getrandbits(1))
    cyan_should_win = int(seed) % 2 == 0
    cyan_arc = ("bottom" if top_is_long else "top") if cyan_should_win else ("top" if top_is_long else "bottom")
    return {
        "width": 512,
        "height": 512,
        "center": [rng.randint(248, 264), rng.randint(248, 264)],
        "rotation_deg": rng.randrange(4) * 90 + rng.uniform(-7.0, 7.0),
        "half_span": 124,
        "short_amplitude": 95,
        "long_amplitude": 150,
        "top_is_long": top_is_long,
        "cyan_arc": cyan_arc,
        "belt_width": 56,
        "marker_radius": 15,
        "palette_variant": rng.randrange(len(PALETTES)),
    }


def _raw_arcs(scene: dict) -> dict[str, list[tuple[float, float]]]:
    span = float(scene["half_span"])
    short = float(scene["short_amplitude"])
    long = float(scene["long_amplitude"])
    top_amp = long if bool(scene["top_is_long"]) else short
    bottom_amp = short if bool(scene["top_is_long"]) else long
    return {
        "top": [(-span, 0.0), (-0.58 * span, -0.78 * top_amp), (0.0, -top_amp), (0.58 * span, -0.78 * top_amp), (span, 0.0)],
        "bottom": [(-span, 0.0), (-0.58 * span, 0.78 * bottom_amp), (0.0, bottom_amp), (0.58 * span, 0.78 * bottom_amp), (span, 0.0)],
    }


def _transform(scene: dict, point: tuple[float, float]) -> tuple[float, float]:
    theta = math.radians(float(scene["rotation_deg"]))
    c, s = math.cos(theta), math.sin(theta)
    x, y = point
    return (float(scene["center"][0]) + c * x - s * y, float(scene["center"][1]) + s * x + c * y)


def _arcs(scene: dict) -> dict[str, list[tuple[float, float]]]:
    return {name: [_transform(scene, point) for point in points] for name, points in _raw_arcs(scene).items()}


def _length(points: list[tuple[float, float]]) -> float:
    return sum(math.hypot(x1 - x0, y1 - y0) for (x0, y0), (x1, y1) in zip(points, points[1:]))


def _point_at_half(points: list[tuple[float, float]]) -> tuple[float, float]:
    target = _length(points) / 2.0
    walked = 0.0
    for p0, p1 in zip(points, points[1:]):
        segment = math.hypot(p1[0] - p0[0], p1[1] - p0[1])
        if walked + segment >= target:
            ratio = (target - walked) / segment
            return (p0[0] + ratio * (p1[0] - p0[0]), p0[1] + ratio * (p1[1] - p0[1]))
        walked += segment
    return points[-1]


def render(scene: dict) -> Image.Image:
    scale = 4
    width, height = int(scene["width"]), int(scene["height"])
    palette = PALETTES[int(scene["palette_variant"])]
    image = Image.new("RGB", (width * scale, height * scale), palette["background"])
    draw = ImageDraw.Draw(image)
    arcs = _arcs(scene)
    belt_width = int(scene["belt_width"]) * scale
    for points in arcs.values():
        scaled = [(round(x * scale), round(y * scale)) for x, y in points]
        draw.line(scaled, fill=palette["track"], width=belt_width, joint="curve")
        radius = belt_width // 2
        for x, y in scaled:
            draw.ellipse((x - radius, y - radius, x + radius, y + radius), fill=palette["track"])

    cyan_arc = str(scene["cyan_arc"])
    gate_points = {cyan_arc: "cyan", "bottom" if cyan_arc == "top" else "top": "orange"}
    markers = {
        "start": arcs["top"][0],
        "finish": arcs["top"][-1],
        gate_points["top"]: _point_at_half(arcs["top"]),
        gate_points["bottom"]: _point_at_half(arcs["bottom"]),
    }
    marker_radius = int(scene["marker_radius"]) * scale
    for name, point in markers.items():
        x, y = round(point[0] * scale), round(point[1] * scale)
        draw.ellipse((x - marker_radius, y - marker_radius, x + marker_radius, y + marker_radius), fill=palette[name])
        inner = max(4 * scale, marker_radius // 3)
        draw.ellipse((x - inner, y - inner, x + inner, y + inner), fill=palette["background"])
    return image.resize((width, height), Image.Resampling.LANCZOS)


def _arc_lengths(scene: dict) -> dict[str, float]:
    return {name: _length(points) for name, points in _raw_arcs(scene).items()}


def analytic_gold(scene: dict) -> str:
    lengths = _arc_lengths(scene)
    short_arc = min(lengths, key=lengths.get)
    return "cyan" if str(scene["cyan_arc"]) == short_arc else "orange"


def margin(scene: dict) -> float:
    values = sorted(_arc_lengths(scene).values())
    return values[1] - values[0]


def is_quarantined(scene: dict) -> bool:
    return margin(scene) < QUARANTINE_MARGIN


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
