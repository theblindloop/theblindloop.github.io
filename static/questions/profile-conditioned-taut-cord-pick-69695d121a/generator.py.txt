"""Taut-cord world: three pinned cords and black pegs on white paper.

A cord can be pulled completely straight exactly when no peg lies inside the
area shut in between the cord as drawn and the straight line joining that same
cord's own two pinned ends.
"""

from __future__ import annotations

import math
import random

from PIL import Image, ImageDraw

CANVAS_W = 480
CANVAS_H = 540
SUPERSAMPLE = 3
STROKE_W = 8
PEG_R = 9
BAND_CENTERS = (90, 270, 450)
COLORS = ("red", "green", "blue")
RGB = {
    "red": (208, 32, 44),
    "green": (18, 138, 62),
    "blue": (36, 74, 206),
}
PEG_RGB = (16, 16, 16)
PAPER = (255, 255, 255)
QUARANTINE_MARGIN = 8.0
SAMPLE_FLOOR = 12.0


# ---------------------------------------------------------------- geometry ---
def _chord_point(cord, t):
    ax, ay = cord["a"]
    bx, by = cord["b"]
    return (ax + (bx - ax) * t, ay + (by - ay) * t)


def _normal(cord):
    ax, ay = cord["a"]
    bx, by = cord["b"]
    dx, dy = bx - ax, by - ay
    ln = math.hypot(dx, dy)
    return (-dy / ln, dx / ln)


def _cord_points(cord):
    nx, ny = _normal(cord)
    amp = cord["amp"]
    mode = cord["mode"]
    sign = cord["sign"]
    pts = []
    steps = 24
    for i in range(steps + 1):
        t = i / steps
        cx, cy = _chord_point(cord, t)
        off = sign * amp * math.sin(mode * math.pi * t)
        pts.append((cx + nx * off, cy + ny * off))
    return pts


def _lobe_sign(cord, t):
    s = math.sin(cord["mode"] * math.pi * t)
    return cord["sign"] * (1.0 if s >= 0 else -1.0)


def _seg_dist(p, a, b):
    px, py = p
    ax, ay = a
    bx, by = b
    dx, dy = bx - ax, by - ay
    den = dx * dx + dy * dy
    if den <= 0:
        return math.hypot(px - ax, py - ay)
    u = ((px - ax) * dx + (py - ay) * dy) / den
    u = max(0.0, min(1.0, u))
    return math.hypot(px - (ax + u * dx), py - (ay + u * dy))


def _closed_polygon(points):
    """Cord polyline closed by the straight line between its own two ends."""
    return [tuple(p) for p in points]


def _dist_to_closed_curve(p, points):
    poly = _closed_polygon(points)
    best = 1e9
    for i in range(len(poly)):
        best = min(best, _seg_dist(p, poly[i], poly[(i + 1) % len(poly)]))
    return best


def _point_in_polygon(p, poly):
    px, py = p
    inside = False
    n = len(poly)
    for i in range(n):
        x1, y1 = poly[i]
        x2, y2 = poly[(i + 1) % n]
        if (y1 > py) != (y2 > py):
            xin = x1 + (py - y1) * (x2 - x1) / (y2 - y1)
            if px < xin:
                inside = not inside
    return inside


def _traps(points, peg):
    return _point_in_polygon(peg, _closed_polygon(points))


# ---------------------------------------------------------------- sampling ---
def _build_cord(rng, color, yc, forced_free):
    return {
        "color": color,
        "a": [round(rng.uniform(46.0, 74.0), 3), round(yc + rng.uniform(-13.0, 13.0), 3)],
        "b": [round(rng.uniform(406.0, 434.0), 3), round(yc + rng.uniform(-13.0, 13.0), 3)],
        "amp": round(rng.uniform(40.0, 58.0), 3),
        "mode": rng.choice([1, 2]),
        "sign": rng.choice([-1, 1]),
        "free": bool(forced_free),
    }


def _band_pegs(rng, cord):
    """One peg outside the shut-in area; a blocked cord also shuts one in."""
    nx, ny = _normal(cord)
    pegs = []
    t_out = rng.choice([0.22, 0.3, 0.7, 0.78])
    d_out = rng.uniform(27.0, 44.0)
    s_out = -_lobe_sign(cord, t_out)
    cx, cy = _chord_point(cord, t_out)
    pegs.append([round(cx + nx * s_out * d_out, 3), round(cy + ny * s_out * d_out, 3)])

    if cord["free"]:
        t_b = rng.choice([0.24, 0.34, 0.66, 0.76])
        d_b = rng.uniform(27.0, 44.0)
        s_b = -_lobe_sign(cord, t_b)
        cx, cy = _chord_point(cord, t_b)
        pegs.append([round(cx + nx * s_b * d_b, 3), round(cy + ny * s_b * d_b, 3)])
    else:
        t_in = (rng.uniform(0.36, 0.62) if cord["mode"] == 1 else rng.choice([0.25, 0.75]))
        t_in += rng.uniform(-0.03, 0.03)
        depth = abs(cord["amp"] * math.sin(cord["mode"] * math.pi * t_in))
        u = rng.uniform(0.4, 0.62)
        s_in = _lobe_sign(cord, t_in)
        cx, cy = _chord_point(cord, t_in)
        pegs.append([round(cx + nx * s_in * depth * u, 3), round(cy + ny * s_in * depth * u, 3)])
    return pegs


def sample_scene(seed: int) -> dict:
    rng = random.Random(seed * 7919 + 13)
    free_color = COLORS[seed % 3]
    for _ in range(600):
        order = list(COLORS)
        rng.shuffle(order)
        cords = [
            _build_cord(rng, color, yc, color == free_color)
            for color, yc in zip(order, BAND_CENTERS)
        ]
        pegs = []
        for cord in cords:
            pegs.extend(_band_pegs(rng, cord))
        drawn = [
            [cord["color"], [[round(x, 3), round(y, 3)] for x, y in _cord_points(cord)]]
            for cord in cords
        ]
        scene = {
            "seed": int(seed),
            "canvas": [CANVAS_W, CANVAS_H],
            "supersample": SUPERSAMPLE,
            "stroke_w": STROKE_W,
            "peg_r": PEG_R,
            "cords": drawn,
            "pegs": pegs,
        }
        if _valid(scene, free_color):
            return scene
    raise RuntimeError(f"no valid scene for seed {seed}")


def _valid(scene, free_colour) -> bool:
    pegs = scene["pegs"]
    for p in pegs:
        if not (26.0 <= p[0] <= CANVAS_W - 26.0 and 26.0 <= p[1] <= CANVAS_H - 26.0):
            return False
    for i in range(len(pegs)):
        for j in range(i + 1, len(pegs)):
            if math.hypot(pegs[i][0] - pegs[j][0], pegs[i][1] - pegs[j][1]) < 4 * PEG_R + 14:
                return False
    for cord in scene["cords"]:
        for p in cord[1]:
            if not (16.0 <= p[0] <= CANVAS_W - 16.0 and 16.0 <= p[1] <= CANVAS_H - 16.0):
                return False
    free = [c[0] for c in scene["cords"] if _trapped_count(scene, c[1]) == 0]
    if len(free) != 1 or free[0] != free_colour:
        return False
    return margin(scene) >= SAMPLE_FLOOR


def _trapped_count(scene, points) -> int:
    return sum(1 for peg in scene["pegs"] if _traps(points, tuple(peg)))


# ------------------------------------------------------------------- gold ----
def analytic_gold(scene: dict) -> str:
    free = [c[0] for c in scene["cords"] if _trapped_count(scene, c[1]) == 0]
    if len(free) != 1:
        return "abstain"
    return free[0]


def margin(scene: dict) -> float:
    best = 1e9
    for cord in scene["cords"]:
        for peg in scene["pegs"]:
            best = min(best, _dist_to_closed_curve(tuple(peg), cord[1]) - scene["peg_r"])
    return round(float(best), 3)


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < QUARANTINE_MARGIN or analytic_gold(scene) == "abstain")


def latent_symmetries(scene: dict) -> list:
    """Pixel-identical latent relabellings that must preserve the answer."""
    out = []

    rot = dict(scene)
    rot["pegs"] = scene["pegs"][1:] + scene["pegs"][:1]
    out.append(("peg_list_rotation", rot))

    rev = dict(scene)
    rev["pegs"] = list(reversed(scene["pegs"]))
    out.append(("peg_list_reversal", rev))

    order = dict(scene)
    order["cords"] = scene["cords"][1:] + scene["cords"][:1]
    out.append(("cord_draw_order", order))
    return out


# ----------------------------------------------------------------- render ----
def render(scene: dict) -> Image.Image:
    s = scene["supersample"]
    w, h = scene["canvas"]
    img = Image.new("RGB", (w * s, h * s), PAPER)
    d = ImageDraw.Draw(img)
    for peg in scene["pegs"]:
        x, y = peg[0] * s, peg[1] * s
        r = scene["peg_r"] * s
        d.ellipse([x - r, y - r, x + r, y + r], fill=PEG_RGB)
    for cord in sorted(scene["cords"], key=lambda c: c[0]):
        pts = [(p[0] * s, p[1] * s) for p in cord[1]]
        d.line(pts, fill=RGB[cord[0]], width=scene["stroke_w"] * s, joint="curve")
        r = scene["stroke_w"] * s / 2.0
        for p in (pts[0], pts[-1]):
            d.ellipse([p[0] - r, p[1] - r, p[0] + r, p[1] + r], fill=RGB[cord[0]])
    return img.resize((w, h), Image.BOX)
