"""Deterministic visual world for Checkerboard Carrier Weave."""
from __future__ import annotations

import math
import random
from copy import deepcopy
from PIL import Image, ImageDraw

WIDTH = HEIGHT = 800
SYMBOLS = ("X", "plus", "T", "L")
PALETTES = (
    ((105, 70, 165), (232, 164, 42), (31, 42, 67), (238, 235, 222)),
    ((31, 116, 145), (218, 112, 55), (55, 34, 69), (234, 239, 226)),
    ((142, 61, 106), (210, 151, 30), (28, 61, 59), (239, 232, 218)),
    ((55, 122, 98), (202, 126, 48), (61, 43, 27), (231, 236, 226)),
)
PANEL_BOXES = {"left": (150, 350, 249, 449), "right": (550, 350, 649, 449)}
GRID_OFFSET = 8
STEP = 12
CELL = 10


def glyph(symbol: str) -> list[list[int]]:
    g = [[0] * 7 for _ in range(7)]
    if symbol == "X":
        for i in range(7):
            g[i][i] = g[i][6 - i] = 1
    elif symbol == "plus":
        for i in range(7):
            g[3][i] = g[i][3] = 1
    elif symbol == "T":
        for i in range(7):
            g[0][i] = 1
            g[i][3] = 1
    elif symbol == "L":
        for i in range(7):
            g[i][0] = 1
            g[6][i] = 1
    else:
        raise ValueError(f"unknown symbol {symbol!r}")
    return g


def _noise_grid(rng: random.Random, active_even: bool, target: list[list[int]]) -> list[list[int]]:
    grid = [[0] * 7 for _ in range(7)]
    inactive = []
    active_ink = 0
    for r in range(7):
        for c in range(7):
            active = ((r + c) % 2 == 0) == active_even
            if active:
                grid[r][c] = target[r][c]
                active_ink += target[r][c]
            else:
                inactive.append((r, c))
    # Every source panel has exactly 15 dark cells, eliminating total-ink cues.
    needed = 15 - active_ink
    if not 0 <= needed <= len(inactive):
        raise AssertionError("template cannot be ink-balanced")
    for r, c in rng.sample(inactive, needed):
        grid[r][c] = 1
    return grid


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed) ^ 0x5A17C3)
    symbol = SYMBOLS[int(seed) % len(SYMBOLS)]
    target = glyph(symbol)
    one = _noise_grid(rng, True, target)
    two = _noise_grid(rng, False, target)
    one_side = "left" if (int(seed) // 4) % 2 == 0 else "right"
    frame, dot, dark, light = PALETTES[(int(seed) // 8) % len(PALETTES)]
    return {
        "symbol": symbol,
        "one_dot_grid": one,
        "two_dot_grid": two,
        "one_dot_side": one_side,
        "frame_color": list(frame),
        "dot_color": list(dot),
        "dark_color": list(dark),
        "light_color": list(light),
    }


def _validate_grid(value: object) -> list[list[int]]:
    if not isinstance(value, list) or len(value) != 7:
        raise ValueError("grid must have seven rows")
    rows = []
    for row in value:
        if not isinstance(row, list) or len(row) != 7 or any(v not in (0, 1) for v in row):
            raise ValueError("grid rows must be seven binary values")
        rows.append([int(v) for v in row])
    return rows


def _draw_panel(draw: ImageDraw.ImageDraw, box: tuple[int, int, int, int], grid: list[list[int]],
                dots: int, frame: tuple[int, int, int], dot: tuple[int, int, int],
                dark: tuple[int, int, int], light: tuple[int, int, int]) -> None:
    x0, y0, x1, y1 = box
    draw.rectangle(box, fill=frame)
    draw.rectangle((x0 + 4, y0 + 4, x1 - 4, y1 - 4), fill=(250, 248, 242))
    gx, gy = x0 + GRID_OFFSET, y0 + GRID_OFFSET
    for r in range(7):
        for c in range(7):
            px, py = gx + c * STEP, gy + r * STEP
            draw.rectangle((px, py, px + CELL - 1, py + CELL - 1),
                           fill=dark if grid[r][c] else light)
    cx = (x0 + x1) // 2
    badge_y = y0 - 22
    offsets = (0,) if dots == 1 else (-8, 8)
    for dx in offsets:
        draw.ellipse((cx + dx - 4, badge_y - 4, cx + dx + 4, badge_y + 4), fill=dot)


def render(scene: dict) -> Image.Image:
    # Explicit accesses are part of the rendered-field audit.
    _ = scene["symbol"]
    one = _validate_grid(scene["one_dot_grid"])
    two = _validate_grid(scene["two_dot_grid"])
    side = scene["one_dot_side"]
    if side not in PANEL_BOXES:
        raise ValueError("invalid one_dot_side")
    frame = tuple(scene["frame_color"])
    dot = tuple(scene["dot_color"])
    dark = tuple(scene["dark_color"])
    light = tuple(scene["light_color"])
    image = Image.new("RGB", (WIDTH, HEIGHT), (250, 248, 242))
    draw = ImageDraw.Draw(image)
    other = "right" if side == "left" else "left"
    _draw_panel(draw, PANEL_BOXES[side], one, 1, frame, dot, dark, light)
    _draw_panel(draw, PANEL_BOXES[other], two, 2, frame, dot, dark, light)
    return image


def _woven(scene: dict) -> list[list[int]]:
    one = _validate_grid(scene["one_dot_grid"])
    two = _validate_grid(scene["two_dot_grid"])
    return [[one[r][c] if (r + c) % 2 == 0 else two[r][c] for c in range(7)] for r in range(7)]


def analytic_gold(scene: dict) -> str:
    symbol = scene["symbol"]
    if _woven(scene) != glyph(symbol):
        raise ValueError("source panels do not weave to declared symbol")
    return symbol


def margin(scene: dict) -> float:
    woven = _woven(scene)
    scores = sorted(sum(a != b for row_a, row_b in zip(woven, glyph(s)) for a, b in zip(row_a, row_b))
                    for s in SYMBOLS)
    return float(scores[1] - scores[0])


def _color_distance(a: tuple[int, ...], b: tuple[int, ...]) -> float:
    return math.sqrt(sum((int(x) - int(y)) ** 2 for x, y in zip(a, b)))


def is_quarantined(scene: dict) -> bool:
    colors = [tuple(scene[k]) for k in ("frame_color", "dot_color", "dark_color", "light_color")]
    colors.append((250, 248, 242))
    return min(_color_distance(colors[i], colors[j]) for i in range(len(colors)) for j in range(i + 1, len(colors))) < 10.0


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
