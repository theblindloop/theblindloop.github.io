"""Independent final-PNG inverse for Checkerboard Carrier Weave.

This module intentionally shares no imports or constants with the renderer.
It locates visible colored frame components and role dots, reads cell centers,
executes the declared checkerboard selection, and reports the best glyph only
when the runner-up Hamming margin is large.
"""
from __future__ import annotations

from collections import deque
import numpy as np
from PIL import Image

FRAME_COLORS = {(105, 70, 165), (31, 116, 145), (142, 61, 106), (55, 122, 98)}
DOT_COLORS = {(232, 164, 42), (218, 112, 55), (210, 151, 30), (202, 126, 48)}
DARK_COLORS = {(31, 42, 67), (55, 34, 69), (28, 61, 59), (61, 43, 27)}
LIGHT_COLORS = {(238, 235, 222), (234, 239, 226), (239, 232, 218), (231, 236, 226)}


def _components(mask: np.ndarray) -> list[dict]:
    h, w = mask.shape
    seen = np.zeros_like(mask, dtype=bool)
    out = []
    for sy, sx in np.argwhere(mask):
        if seen[sy, sx]:
            continue
        q = deque([(int(sy), int(sx))]); seen[sy, sx] = True
        pts = []
        while q:
            y, x = q.popleft(); pts.append((y, x))
            for ny, nx in ((y-1,x),(y+1,x),(y,x-1),(y,x+1)):
                if 0 <= ny < h and 0 <= nx < w and mask[ny, nx] and not seen[ny, nx]:
                    seen[ny, nx] = True; q.append((ny, nx))
        ys = [p[0] for p in pts]; xs = [p[1] for p in pts]
        out.append({"area": len(pts), "box": (min(xs), min(ys), max(xs), max(ys)),
                    "cx": sum(xs)/len(xs), "cy": sum(ys)/len(ys)})
    return out


def _exact_mask(a: np.ndarray, colors: set[tuple[int, int, int]]) -> np.ndarray:
    return np.any(np.all(a[:, :, None, :] == np.array(sorted(colors), dtype=np.uint8)[None, None, :, :], axis=3), axis=2)


def _template(name: str) -> list[list[int]]:
    g = [[0] * 7 for _ in range(7)]
    if name == "X":
        for i in range(7): g[i][i] = g[i][6-i] = 1
    elif name == "plus":
        for i in range(7): g[3][i] = g[i][3] = 1
    elif name == "T":
        for i in range(7): g[0][i] = 1; g[i][3] = 1
    elif name == "L":
        for i in range(7): g[i][0] = 1; g[6][i] = 1
    return g


def _read_grid(a: np.ndarray, box: tuple[int, int, int, int]) -> list[list[int]]:
    x0, y0, x1, y1 = box
    if not (95 <= x1-x0+1 <= 105 and 95 <= y1-y0+1 <= 105):
        raise ValueError("invalid panel dimensions")
    grid = []
    for r in range(7):
        row = []
        for c in range(7):
            rgb = tuple(int(v) for v in a[y0 + 12 + r*12, x0 + 12 + c*12])
            if rgb in DARK_COLORS: row.append(1)
            elif rgb in LIGHT_COLORS: row.append(0)
            else: raise ValueError("unreadable cell sample")
        grid.append(row)
    return grid


def decision_from_image(image: Image.Image) -> str:
    a = np.asarray(image.convert("RGB"), dtype=np.uint8)
    if a.shape != (800, 800, 3):
        raise ValueError("unexpected canvas")
    frames = [c for c in _components(_exact_mask(a, FRAME_COLORS)) if c["area"] >= 1000]
    if len(frames) != 2:
        raise ValueError("expected two intact panel frames")
    frames.sort(key=lambda c: c["cx"])
    dots = [c for c in _components(_exact_mask(a, DOT_COLORS)) if 40 <= c["area"] <= 100]
    roles = {}
    for panel in frames:
        x0, y0, x1, _ = panel["box"]
        local = [d for d in dots if x0 <= d["cx"] <= x1 and y0-35 <= d["cy"] <= y0-10]
        if len(local) not in (1, 2) or len(local) in roles:
            raise ValueError("ambiguous role badge")
        roles[len(local)] = _read_grid(a, panel["box"])
    if set(roles) != {1, 2}:
        raise ValueError("missing role badge")
    woven = [[roles[1][r][c] if (r+c)%2 == 0 else roles[2][r][c] for c in range(7)] for r in range(7)]
    scored = []
    for name in ("X", "plus", "T", "L"):
        t = _template(name)
        scored.append((sum(woven[r][c] != t[r][c] for r in range(7) for c in range(7)), name))
    scored.sort()
    if scored[0][0] != 0 or scored[1][0] - scored[0][0] < 8:
        raise ValueError("ambiguous woven glyph")
    return scored[0][1]
