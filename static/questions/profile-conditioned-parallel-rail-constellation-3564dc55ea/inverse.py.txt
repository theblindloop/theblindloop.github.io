"""Independent pixel-only inverse for the parallel-rail constellation.

This module deliberately contains its own palette segmentation, connected
component extraction, centroid measurement, and line-fit search. It receives a
PNG only; no renderer, prompt, latent scene, or analytic gold is imported.
"""

from __future__ import annotations

import itertools
import math

import numpy as np
from PIL import Image

PALETTE = {
    "red": (214.0, 64.0, 72.0),
    "blue": (55.0, 101.0, 220.0),
    "green": (48.0, 166.0, 103.0),
}


def _components(mask: np.ndarray) -> list[list[tuple[int, int]]]:
    height, width = mask.shape
    seen = np.zeros(mask.shape, dtype=bool)
    components: list[list[tuple[int, int]]] = []
    for y in range(height):
        for x in range(width):
            if not mask[y, x] or seen[y, x]:
                continue
            stack = [(y, x)]
            seen[y, x] = True
            pixels: list[tuple[int, int]] = []
            while stack:
                cy, cx = stack.pop()
                pixels.append((cy, cx))
                for ny, nx in ((cy - 1, cx), (cy + 1, cx), (cy, cx - 1), (cy, cx + 1)):
                    if 0 <= ny < height and 0 <= nx < width and mask[ny, nx] and not seen[ny, nx]:
                        seen[ny, nx] = True
                        stack.append((ny, nx))
            if len(pixels) >= 12:
                components.append(pixels)
    return components


def _centers(image: Image.Image, color: tuple[float, float, float]) -> list[tuple[float, float]]:
    data = np.asarray(image.convert("RGB"), dtype=np.float32)
    distance = np.sqrt(np.sum((data - np.asarray(color, dtype=np.float32)) ** 2, axis=2))
    mask = distance < 115.0
    centers = []
    for component in _components(mask):
        ys = np.asarray([pixel[0] for pixel in component], dtype=np.float64)
        xs = np.asarray([pixel[1] for pixel in component], dtype=np.float64)
        centers.append((float(xs.mean()), float(ys.mean())))
    if len(centers) != 8:
        raise ValueError(f"expected eight separated components, found {len(centers)}")
    return centers


def _stats(points: list[tuple[float, float]]) -> tuple[float, float, float, float, float]:
    cx = sum(p[0] for p in points) / len(points)
    cy = sum(p[1] for p in points) / len(points)
    xx = sum((p[0] - cx) ** 2 for p in points)
    yy = sum((p[1] - cy) ** 2 for p in points)
    xy = sum((p[0] - cx) * (p[1] - cy) for p in points)
    heading = 0.5 * math.atan2(2.0 * xy, xx - yy)
    along = (math.cos(heading), math.sin(heading))
    normal = (-along[1], along[0])
    error = sum(abs((p[0] - cx) * normal[0] + (p[1] - cy) * normal[1]) for p in points) / len(points)
    projections = sorted((p[0] - cx) * along[0] + (p[1] - cy) * along[1] for p in points)
    gaps = [projections[i + 1] - projections[i] for i in range(3)]
    average = sum(gaps) / 3.0
    variation = math.sqrt(sum((gap - average) ** 2 for gap in gaps) / 3.0) / max(average, 1e-6)
    return cx, cy, heading, error, variation


def _score(points: list[tuple[float, float]]) -> float:
    best = 0.0
    for first in itertools.combinations(range(8), 4):
        if 0 not in first:
            continue
        chosen = set(first)
        left = [points[index] for index in first]
        right = [points[index] for index in range(8) if index not in chosen]
        lx, ly, lt, le, lc = _stats(left)
        rx, ry, rt, re, rc = _stats(right)
        cosine = abs(math.cos(lt) * math.cos(rt) + math.sin(lt) * math.sin(rt))
        angle = math.degrees(math.acos(max(-1.0, min(1.0, cosine))))
        separation = abs((rx - lx) * (-math.sin(lt)) + (ry - ly) * math.cos(lt))
        error = max(le, re)
        variation = max(lc, rc)
        value = math.exp(-error / 5.5 - angle / 14.0 - variation / 0.40) * min(1.0, separation / 28.0)
        best = max(best, value)
    return best


def decision_from_image(image: Image.Image) -> str:
    """Classify the highest-scoring complete field using pixels alone."""

    scores = [(name, _score(_centers(image, color))) for name, color in PALETTE.items()]
    scores.sort(key=lambda item: (-item[1], item[0]))
    return scores[0][0]
