"""Renderer for the halftone-density reveal world.

A 5x5 panel of speckled cells hides an abstract symbol in the *packing
density* of tiny grey blocks: every cell is filled with the same two grey
tones, so the symbol only exists at coarse scale, after the declared blur.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

RENDERER_VERSION = "halftone_density_reveal-renderer-0.1.0"

GRID = 5
ON_CELLS_PER_SYMBOL = 9

# Each template marks exactly nine cells, so total ink is constant across
# symbols and cannot be used as a shortcut cue.
_TEMPLATE_ROWS: dict[str, tuple[str, ...]] = {
    "plus": ("..X..", "..X..", "XXXXX", "..X..", "..X.."),
    "ex": ("X...X", ".X.X.", "..X..", ".X.X.", "X...X"),
    "diamond": ("..X..", ".X.X.", "X.X.X", ".X.X.", "..X.."),
    "ell": ("X....", "X....", "X....", "X....", "XXXXX"),
    "flag": ("XXX..", "XXX..", "X....", "X....", "X...."),
    "zee": ("XXXXX", "...X.", "..X..", ".X...", "X...."),
}

SYMBOL_NAMES: tuple[str, ...] = tuple(sorted(_TEMPLATE_ROWS))


def _template(name: str) -> frozenset[tuple[int, int]]:
    rows = _TEMPLATE_ROWS[name]
    return frozenset(
        (r, c)
        for r, row in enumerate(rows)
        for c, ch in enumerate(row)
        if ch == "X"
    )


TEMPLATES: dict[str, frozenset[tuple[int, int]]] = {
    name: _template(name) for name in SYMBOL_NAMES
}

for _name, _cells in TEMPLATES.items():
    if len(_cells) != ON_CELLS_PER_SYMBOL:
        raise AssertionError(f"template {_name} does not carry nine cells")

SIZE = 560
BLOCK = 4
FRAME_W = 7


# --------------------------------------------------------------------------
# scene sampling
# --------------------------------------------------------------------------
def sample_scene(seed: int) -> dict:
    rng = np.random.RandomState(int(seed) % (2 ** 31))
    symbol = SYMBOL_NAMES[int(rng.randint(len(SYMBOL_NAMES)))]
    swaps = int(rng.randint(0, 7))
    cells = set(TEMPLATES[symbol])
    # Perturb by cell swaps that preserve the nine-cell ink budget, and reject
    # exact ties: a pattern equidistant from two symbols has no single nearest
    # match, so it is never printed.
    for _ in range(16):
        trial = set(TEMPLATES[symbol])
        for _ in range(swaps):
            on_list = sorted(trial)
            off_list = [
                (r, c)
                for r in range(GRID)
                for c in range(GRID)
                if (r, c) not in trial
            ]
            trial.discard(on_list[int(rng.randint(len(on_list)))])
            trial.add(off_list[int(rng.randint(len(off_list)))])
        ordered = sorted(len(trial ^ TEMPLATES[n]) for n in SYMBOL_NAMES)
        if ordered[1] > ordered[0]:
            cells = trial
            break
    on_cells = [[int(r), int(c)] for r, c in sorted(cells)]

    gold_now = _nearest(frozenset(cells))
    others = [n for n in SYMBOL_NAMES if n != gold_now]
    decoy = others[int(rng.randint(len(others)))]
    decoy_cells = [[int(r), int(c)] for r, c in sorted(TEMPLATES[decoy])]

    cell = int([80, 88][int(rng.randint(2))])
    panel = cell * GRID
    slack = SIZE - panel - 2 * (FRAME_W + 6)
    ox = int(FRAME_W + 6 + rng.randint(0, max(1, slack)))
    oy = int(FRAME_W + 6 + rng.randint(0, max(1, slack)))

    contrast = float(round(0.07 + 0.25 * rng.rand(), 4))
    ink = int(rng.randint(30, 56))
    plate = int(rng.randint(198, 226))

    return {
        "on_cells": on_cells,
        "decoy_cells": decoy_cells,
        "cell": cell,
        "block": BLOCK,
        "ox": ox,
        "oy": oy,
        "size": SIZE,
        "contrast": contrast,
        "ink": ink,
        "plate": plate,
        "bg": [int(rng.randint(238, 250))] * 3,
        "frame": [
            int(rng.randint(0, 40)),
            int(rng.randint(40, 92)),
            int(rng.randint(150, 212)),
        ],
        "mark": [
            int(rng.randint(214, 246)),
            int(rng.randint(108, 152)),
            int(rng.randint(0, 42)),
        ],
        "dither_seed": int(rng.randint(1, 2 ** 30)),
    }


# --------------------------------------------------------------------------
# rasterisation
# --------------------------------------------------------------------------
def _on_set(scene: dict) -> frozenset[tuple[int, int]]:
    return frozenset((int(r), int(c)) for r, c in scene["on_cells"])


def _dark_fraction(scene: dict, on: bool) -> float:
    contrast = float(scene["contrast"])
    return 0.5 + (contrast / 2.0 if on else -contrast / 2.0)


def render(scene: dict) -> Image.Image:
    size = int(scene["size"])
    cell = int(scene["cell"])
    block = int(scene["block"])
    ox, oy = int(scene["ox"]), int(scene["oy"])
    panel = cell * GRID
    ink = int(scene["ink"])
    plate = int(scene["plate"])

    canvas = np.zeros((size, size, 3), dtype=np.uint8)
    canvas[:, :] = np.array(scene["bg"], dtype=np.uint8)

    # solid frame ring drawn just outside the panel
    fw = FRAME_W
    canvas[oy - fw : oy + panel + fw, ox - fw : ox + panel + fw] = np.array(
        scene["frame"], dtype=np.uint8
    )

    on_cells = _on_set(scene)
    per_side = cell // block
    n_blocks = per_side * per_side
    base_seed = int(scene["dither_seed"]) % (2 ** 30)

    for r in range(GRID):
        for c in range(GRID):
            frac = _dark_fraction(scene, (r, c) in on_cells)
            n_dark = int(round(frac * n_blocks))
            rng = np.random.RandomState((base_seed * 31 + r * GRID + c) % (2 ** 31))
            order = rng.permutation(n_blocks)
            flat = np.full(n_blocks, plate, dtype=np.uint8)
            flat[order[:n_dark]] = ink
            tile = np.repeat(
                np.repeat(flat.reshape(per_side, per_side), block, axis=0),
                block,
                axis=1,
            )
            y0 = oy + r * cell
            x0 = ox + c * cell
            patch = canvas[y0 : y0 + cell, x0 : x0 + cell]
            patch[:, :] = np.full((cell, cell, 3), plate, dtype=np.uint8)
            patch[: tile.shape[0], : tile.shape[1]] = tile[:, :, None]

    # decorative orange squares: a *different* symbol, declared irrelevant
    mark = np.array(scene["mark"], dtype=np.uint8)
    side = max(6, cell // 3)
    pad = (cell - side) // 2
    for r, c in scene["decoy_cells"]:
        y0 = oy + int(r) * cell + pad
        x0 = ox + int(c) * cell + pad
        canvas[y0 : y0 + side, x0 : x0 + side] = mark

    return Image.fromarray(canvas, mode="RGB")


# --------------------------------------------------------------------------
# analytic gold
# --------------------------------------------------------------------------
def _nearest(cells: frozenset) -> str:
    best_name, best = None, None
    for name in SYMBOL_NAMES:
        dist = len(cells ^ TEMPLATES[name])
        if best is None or dist < best:
            best_name, best = name, dist
    return str(best_name)


def _distances(scene: dict) -> list[tuple[str, int]]:
    cells = _on_set(scene)
    return [(name, len(cells ^ TEMPLATES[name])) for name in SYMBOL_NAMES]


def analytic_gold(scene: dict) -> str:
    return _nearest(_on_set(scene))


def cell_gap(scene: dict) -> float:
    """Grey-level separation between a dense cell and a sparse cell."""
    return float(
        (int(scene["plate"]) - int(scene["ink"])) * float(scene["contrast"])
    )


def margin(scene: dict) -> float:
    ordered = sorted(d for _, d in _distances(scene))
    return float(round(cell_gap(scene) * (ordered[1] - ordered[0]), 6))


def is_quarantined(scene: dict) -> bool:
    return bool(margin(scene) < 60.0)


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    """Pixel-identical rewrites of the latent scene."""
    twins: list[tuple[str, dict]] = []

    reversed_cells = dict(scene)
    reversed_cells["on_cells"] = list(reversed(scene["on_cells"]))
    twins.append(("reversed_on_cell_order", reversed_cells))

    duplicated = dict(scene)
    duplicated["on_cells"] = list(scene["on_cells"]) + [list(scene["on_cells"][0])]
    twins.append(("duplicated_on_cell", duplicated))

    shifted = dict(scene)
    shifted["dither_seed"] = int(scene["dither_seed"]) + 2 ** 30
    twins.append(("dither_seed_modulus", shifted))

    return twins


def naive_decoy_symbol(scene: dict) -> str:
    """Diagnostic: the symbol printed by the orange decoration."""
    cells = frozenset((int(r), int(c)) for r, c in scene["decoy_cells"])
    for name in SYMBOL_NAMES:
        if TEMPLATES[name] == cells:
            return name
    return "unknown"
