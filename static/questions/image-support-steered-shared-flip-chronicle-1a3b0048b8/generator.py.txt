"""Deterministic renderer and analytic arm for Shared-flip chronicle."""

from __future__ import annotations

import random
from typing import Iterable

from PIL import Image, ImageDraw, ImageFont

WIDTH, HEIGHT = 1024, 768
CARD = 156
BG = (250, 248, 242)
CARD_COLOR = (232, 224, 248)
INK = (45, 43, 61)
EMPTY = (252, 248, 237)
FILLED = (35, 166, 170)
ARROW = (116, 91, 175)
GOLD = (224, 167, 47)
POSITIONS = ("NW", "NE", "SW", "SE")
FRAME_ORIGINS = ((16, 16), (94, 16), (55, 94))


def _toggle(bits: list[int], indices: Iterable[int]) -> list[int]:
    output = list(bits)
    for index in indices:
        output[index] = 1 - output[index]
    return output


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    shared = int(seed) % 4
    others = [value for value in range(4) if value != shared]
    rng.shuffle(others)
    flip_a = sorted((shared, others[0]))
    flip_b = sorted((shared, others[1]))
    start_bits = [rng.randrange(2) for _ in range(4)]
    # Translation varies independently of the answer and keeps the full card in view.
    card_x = rng.randint(90, WIDTH - CARD - 90)
    card_y = rng.randint(75, HEIGHT - CARD - 75)
    return {
        "start_bits": start_bits,
        "flip_a": flip_a,
        "flip_b": flip_b,
        "card_x": card_x,
        "card_y": card_y,
    }


def _states(scene: dict) -> list[list[int]]:
    first = list(scene["start_bits"])
    second = _toggle(first, scene["flip_a"])
    third = _toggle(second, scene["flip_b"])
    return [first, second, third]


def render(scene: dict) -> Image.Image:
    image = Image.new("RGB", (WIDTH, HEIGHT), BG)
    draw = ImageDraw.Draw(image)
    x, y = int(scene["card_x"]), int(scene["card_y"])
    draw.rounded_rectangle((x, y, x + CARD - 1, y + CARD - 1), radius=18, fill=CARD_COLOR)

    # The arrows make the chronology explicit while remaining inside the compact card.
    draw.line((x + 64, y + 38, x + 91, y + 38), fill=ARROW, width=7)
    draw.polygon(((x + 91, y + 38), (x + 80, y + 30), (x + 80, y + 46)), fill=ARROW)
    draw.line((x + 117, y + 66, x + 91, y + 92), fill=ARROW, width=7)
    draw.polygon(((x + 91, y + 92), (x + 94, y + 78), (x + 105, y + 89)), fill=ARROW)

    font = ImageFont.load_default()
    for number, ((ox, oy), bits) in enumerate(zip(FRAME_ORIGINS, _states(scene)), start=1):
        left, top = x + ox, y + oy
        draw.rounded_rectangle((left, top, left + 45, top + 45), radius=5, fill=EMPTY, outline=INK, width=3)
        for index, bit in enumerate(bits):
            row, col = divmod(index, 2)
            cx = left + 11 + 19 * col
            cy = top + 11 + 19 * row
            draw.rectangle((cx - 7, cy - 7, cx + 7, cy + 7), fill=FILLED if bit else EMPTY, outline=INK, width=2)
        # Small numbered tabs reinforce panel order for human solvers.
        draw.ellipse((left - 5, top - 5, left + 9, top + 9), fill=GOLD, outline=INK, width=1)
        draw.text((left, top - 4), str(number), fill=INK, font=font, anchor="mm")
    return image


def analytic_gold(scene: dict) -> str:
    common = set(scene["flip_a"]) & set(scene["flip_b"])
    if len(common) != 1:
        raise ValueError("flip sets must have exactly one shared position")
    return POSITIONS[next(iter(common))]


def margin(scene: dict) -> float:
    counts = [int(i in scene["flip_a"]) + int(i in scene["flip_b"]) for i in range(4)]
    ordered = sorted(counts, reverse=True)
    return float(ordered[0] - ordered[1])


def is_quarantined(scene: dict) -> bool:
    a, b = set(scene["flip_a"]), set(scene["flip_b"])
    return len(a) != 2 or len(b) != 2 or len(a & b) != 1


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    return []
