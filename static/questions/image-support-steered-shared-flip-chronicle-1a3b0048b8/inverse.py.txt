"""Independent pixel-only inverse arm for Shared-flip chronicle.

This module deliberately contains no renderer, prompt, generator, verifier, scene,
or gold imports. It localizes the pastel card from the raster, samples the four
cell interiors at relative locations in each frame, reconstructs consecutive XOR
events, and returns their unique spatial intersection.
"""

from __future__ import annotations

import numpy as np
from PIL import Image

POSITIONS = ("NW", "NE", "SW", "SE")
FRAME_CENTERS = (
    ((27, 27), (46, 27), (27, 46), (46, 46)),
    ((105, 27), (124, 27), (105, 46), (124, 46)),
    ((66, 105), (85, 105), (66, 124), (85, 124)),
)


def _card_bounds(array: np.ndarray) -> tuple[int, int, int, int]:
    r = array[:, :, 0].astype(int)
    g = array[:, :, 1].astype(int)
    b = array[:, :, 2].astype(int)
    mask = (r > 190) & (g > 190) & (b > r + 7) & (b > g + 5)
    ys, xs = np.nonzero(mask)
    if len(xs) < 500:
        raise ValueError("compact chronicle card is missing")
    x0, x1 = int(xs.min()), int(xs.max())
    y0, y1 = int(ys.min()), int(ys.max())
    if not (145 <= x1 - x0 + 1 <= 165 and 145 <= y1 - y0 + 1 <= 165):
        raise ValueError("chronicle card geometry is incomplete")
    return x0, y0, x1, y1


def _filled(array: np.ndarray, x: int, y: int) -> int:
    patch = array[max(0, y - 4): y + 5, max(0, x - 4): x + 5].reshape(-1, 3)
    if len(patch) < 50:
        raise ValueError("cell sample is clipped")
    rgb = np.median(patch, axis=0)
    # Filled teal has strong green/blue chroma; hollow cells are warm and bright.
    teal_score = float(rgb[1] + rgb[2] - 2.0 * rgb[0])
    if teal_score > 75 and rgb[1] < 215:
        return 1
    if float(rgb.mean()) > 220 and abs(float(rgb[0]) - float(rgb[1])) < 25:
        return 0
    raise ValueError("cell state is visually ambiguous")


def decision_from_image(image: Image.Image) -> str:
    array = np.asarray(image.convert("RGB"), dtype=np.uint8)
    x0, y0, x1, y1 = _card_bounds(array)
    scale_x = (x1 - x0) / 155.0
    scale_y = (y1 - y0) / 155.0
    states: list[list[int]] = []
    for centers in FRAME_CENTERS:
        state = []
        for local_x, local_y in centers:
            x = int(round(x0 + local_x * scale_x))
            y = int(round(y0 + local_y * scale_y))
            state.append(_filled(array, x, y))
        states.append(state)
    first = {i for i in range(4) if states[0][i] != states[1][i]}
    second = {i for i in range(4) if states[1][i] != states[2][i]}
    shared = first & second
    if len(first) != 2 or len(second) != 2 or len(shared) != 1:
        raise ValueError("visible transitions do not define one shared two-step flip")
    return POSITIONS[next(iter(shared))]
