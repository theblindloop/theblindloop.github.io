"""Deterministic latent sampler and raster renderer for Switchback Relay."""

from __future__ import annotations

import copy
import math
import random

from PIL import Image, ImageDraw, ImageFont

WIDTH = 512
HEIGHT = 512
BACKGROUND = (248, 247, 242)
ROUTE = (35, 71, 105)
INK = (24, 25, 28)
RED = (202, 52, 55)
MUTED = (104, 107, 110)
TAG_RGB = {
    "amber": (232, 164, 38),
    "cyan": (31, 174, 190),
    "violet": (139, 88, 181),
}
SHAPES = ("circle", "square", "triangle")
TAG_NAMES = ("amber", "cyan", "violet")


def _apply_gates(scene: dict) -> list[str]:
    slots = list(scene["start_order"])
    for code in scene["gates"]:
        if int(code) == 1:
            slots[0], slots[1] = slots[1], slots[0]
        elif int(code) == 2:
            slots[1], slots[2] = slots[2], slots[1]
        else:
            raise ValueError("gate code must be 1 or 2")
    return slots


def sample_scene(seed: int) -> dict:
    rng = random.Random(int(seed))
    start = list(SHAPES)
    rng.shuffle(start)
    gates = [1 + rng.randrange(2) for _ in range(5)]
    if len(set(gates)) == 1:
        gates[2] = 3 - gates[2]

    probe = {"start_order": start, "gates": gates}
    slots = _apply_gates(probe)
    query_shape = "circle"
    query_slot = slots.index(query_shape)
    desired = TAG_NAMES[int(seed) % len(TAG_NAMES)]
    remainder = [name for name in TAG_NAMES if name != desired]
    rng.shuffle(remainder)
    finish = [None, None, None]
    finish[query_slot] = desired
    other_slots = [index for index in range(3) if index != query_slot]
    for index, name in zip(other_slots, remainder, strict=True):
        finish[index] = name

    bases = [(98, 96), (154, 146), (214, 202), (274, 258), (334, 314), (394, 370), (430, 416)]
    route_points = []
    for index, (x, y) in enumerate(bases):
        if index in (0, len(bases) - 1):
            route_points.append([x, y])
        else:
            route_points.append([x + rng.randint(-5, 5), y + rng.randint(-5, 5)])
    return {
        "seed": int(seed),
        "width": WIDTH,
        "height": HEIGHT,
        "start_order": start,
        "gates": gates,
        "finish_colors": finish,
        "query_shape": query_shape,
        "route_points": route_points,
        "gate_phase": float(rng.choice((0, 45, 90, 135, 180, 225, 270, 315))),
    }


def _draw_shape(draw: ImageDraw.ImageDraw, shape: str, center: tuple[int, int], fill, size: int = 20) -> None:
    cx, cy = center
    half = size // 2
    box = (cx - half, cy - half, cx + half, cy + half)
    if shape == "circle":
        draw.ellipse(box, fill=fill)
    elif shape == "square":
        draw.rectangle(box, fill=fill)
    elif shape == "triangle":
        draw.polygon([(cx, cy - half - 1), (cx - half - 1, cy + half), (cx + half + 1, cy + half)], fill=fill)
    else:
        raise ValueError(f"unknown shape {shape!r}")


def render(scene: dict) -> Image.Image:
    width = int(scene["width"])
    height = int(scene["height"])
    image = Image.new("RGB", (width, height), BACKGROUND)
    draw = ImageDraw.Draw(image)
    start_order = list(scene["start_order"])
    gates = list(scene["gates"])
    finish_colors = list(scene["finish_colors"])
    query_shape = str(scene["query_shape"])
    points = [tuple(map(int, point)) for point in scene["route_points"]]
    phase = math.radians(float(scene["gate_phase"]))

    start_ys = (50, 96, 142)
    finish_ys = (370, 416, 462)
    start_hub = points[0]
    finish_hub = points[-1]
    for y in start_ys:
        draw.line([(61, y), (78, y), start_hub], fill=ROUTE, width=5, joint="curve")
    draw.line(points, fill=ROUTE, width=5, joint="curve")
    for y in finish_ys:
        draw.line([finish_hub, (452, y), (468, y)], fill=ROUTE, width=5, joint="curve")

    for shape, y in zip(start_order, start_ys, strict=True):
        _draw_shape(draw, shape, (48, y), INK)

    # The query glyph is visibly repeated in a light legend, while the public
    # question also names it. It is deliberately not part of the route-color mask.
    draw.ellipse((16, 170, 80, 214), outline=MUTED, width=2)
    _draw_shape(draw, query_shape, (34, 192), MUTED, size=13)
    draw.text((45, 185), "TARGET", fill=MUTED, font=ImageFont.load_default())

    for code, (x, y) in zip(gates, points[1:-1], strict=True):
        cx, cy = x + 15, y - 15
        if int(code) == 1:
            centers = [(cx, cy)]
        else:
            dx = int(round(6 * math.cos(phase)))
            dy = int(round(6 * math.sin(phase)))
            if dx == 0 and dy == 0:
                dx = 6
            centers = [(cx - dx, cy - dy), (cx + dx, cy + dy)]
        for bx, by in centers:
            draw.ellipse((bx - 4, by - 4, bx + 4, by + 4), fill=RED)

    for name, y in zip(finish_colors, finish_ys, strict=True):
        draw.rectangle((473, y - 11, 495, y + 11), fill=TAG_RGB[name])

    draw.text((18, 14), "START", fill=MUTED, font=ImageFont.load_default())
    draw.text((454, 488), "FINISH", fill=MUTED, font=ImageFont.load_default())
    return image


def analytic_gold(scene: dict) -> str:
    slots = _apply_gates(scene)
    query_slot = slots.index(scene["query_shape"])
    return str(scene["finish_colors"][query_slot])


def margin(scene: dict) -> float:
    # Exact one-hot selection: the computed slot scores 1 and both alternatives 0.
    _ = analytic_gold(scene)
    return 1.0


def is_quarantined(scene: dict) -> bool:
    try:
        valid = (
            sorted(scene["start_order"]) == sorted(SHAPES)
            and len(scene["gates"]) == 5
            and all(int(code) in (1, 2) for code in scene["gates"])
            and sorted(scene["finish_colors"]) == sorted(TAG_NAMES)
            and scene["query_shape"] in SHAPES
            and len(scene["route_points"]) == 7
        )
    except (KeyError, TypeError, ValueError):
        return True
    return not valid


def latent_symmetries(scene: dict) -> list[tuple[str, dict]]:
    twin = copy.deepcopy(scene)
    twin["gate_phase"] = float(twin["gate_phase"]) + 360.0
    return [("bead_phase_plus_full_turn", twin)]
